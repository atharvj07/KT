#include <iostream>
#include <vector>

using namespace std;

char opr(int n) {
    return n ? '+' : '-';
}

int main() {
    string e;
    int p, m;
    cin >> e >> p >> m;
    vector<int> minus;
    vector<char> oprs;
    while(e.find('?') != -1) {
        int cur = e.find('?');
        if(e[cur - 1] == '(')
            minus.push_back(cur);
        else
            oprs.push_back(opr(m > 0 && (int)oprs.size() < p + m - minus.size() - 1));
        e[cur] = oprs.back();
        if(minus.size() && cur == minus.back()) {
            minus.pop_back();
            e[e.find(')')] = oprs.back();
            oprs.pop_back();
        }
    }
    int ans = 0;
    for(int i = 0; i < e.size(); i++) {
        if(e[i] >= '0' && e[i] <= '9')
            ans += e[i] - '0';
    }
    cout << ans << endl;
    return 0;
}
