#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x;
    cin >> n;
    vector<int> a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    for (int i = 1; i <= n; i++) {
        x = rand() % (n - i + 1) + 1;
        b[i] = a[x], a[x] = a[n - i + 1];
        if (x == n - i + 1) x--;
        a[n - i + 1] = a[x];
    }
    for (int i = 2; i <= n; i++) {
        if (b[i] == b[i - 1]) {
            for (int j = i; j + 1 <= n; j++) b[j] = b[j + 1], b[j + 1] = b[j];
            break;
        }
    }
    if (is_sorted(b.begin() + 1, b.end())) {
        for (int i = n - 1; i >= 1; i--) {
            if (b[i] == b[i + 1]) {
                for (int j = i; j <= n - 1; j++) b[j] = b[j + 1], b[j + 1] = b[j];
                break;
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << b[i] << ' ';
    return 0;
}
