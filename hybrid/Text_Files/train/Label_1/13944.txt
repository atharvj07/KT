#include<bits/stdc++.h>
using namespace std;
const int maxn = 105;
int n, m, k, pos[maxn], a[maxn][maxn], ans;
int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < k; i++) {
        cin >> pos[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            ans += pos[a[i][j] - 1];
            for (int l = a[i][j] - 1; l > 0; l--) pos[l] = pos[l - 1];
            pos[0] = a[i][j];
        }
    }
    cout << ans << '\n';
    return 0;
}
