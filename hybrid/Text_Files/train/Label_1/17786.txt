#include <iostream>
#include <algorithm>
using namespace std;
int t, n, m;
string s, p, ss;
bool solve() {
  ss = p;
  sort(s.begin(), s.end());
  sort(ss.begin(), ss.end());
  ss.erase(unique(ss.begin(), ss.end()), ss.end());
  for (int i = 0; i < n; i++)
    if (s[i] != ss[i]) return 0;
  return 1;
}
int main() {
  cin >> t;
  while (t--) {
    cin >> p >> s;
    n = p.length();
    m = s.length();
    if (n > m || !solve())
      cout << "NO\n";
    else
      cout << "YES\n";
  }
  return 0;
}
