import java.util.Scanner;
import java.util.HashMap;
import java.util.Arrays;
import java.util.List;

public class Main {

    static int[] getCounts(int n, int k, int[][] coordinates) {
        int[] counts = new int[n + 1];
        HashMap<Integer, List<Integer>> map = new HashMap<>();
        for (int[] coord : coordinates) {
            map.computeIfAbsent(coord[0], key -> new java.util.ArrayList<>()).add(coord[1]);
        }
        for (int x1 = -109; x1 <= 109 - k; x1++) {
            for (int y1 = -109; y1 <= 109 - k; y1++) {
                int count = 0;
                for (int dx = 0; dx < k; dx++) {
                    for (int dy = 0; dy < k; dy++) {
                        if (map.containsKey(x1 + dx) && map.get(x1 + dx).contains(y1 + dy)) {
                            count++;
                        }
                    }
                }
                counts[count]++;
            }
        }
        return counts;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] coordinates = new int[n][2];
        for (int i = 0; i < n; i++) {
            coordinates[i][0] = sc.nextInt();
            coordinates[i][1] = sc.nextInt();
        }
        int[] counts = getCounts(n, k, coordinates);
        for (int i = 1; i <= n; i++) {
            System.out.print(counts[i] + " ");
        }
        System.out.println();
    }
}
