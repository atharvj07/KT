#include <iostream>
#include <cstring>
#include <vector>

using namespace std;

const int maxn = 1e6 + 10;
bool visited[30][30][2];
vector<string> paths;
int vis[maxn];
int dx[] = {0,0,1,-1};
int dy[] = {1,-1,0,0};

int dfs(int t, int x, int y, char c, char opp) {
    if (t > paths[x].length()) return 0;
    if (x == paths[x].length() && opp == paths[y][t - 1]) return 1;
    int ax = x, ay = y;
    if (visited[x][y][c == opp]) return 0;
    if (c != opp) visited[x][y][c == opp] = 1;
    int ok = 0;
    if (c == 'N') {
        y -= 1;
        if (t <= paths[x].length() && paths[x][t - 1] == 'N') ay = y + 1;
        if (y >= 0 && paths[x][t - 1] != 'S' && paths[y][t - 1] != 'N') ok |= dfs(t + 1, x, y, 'N', opp);
    }
    if (c == 'E') {
        x += 1;
        if (t <= paths[y].length() && paths[y][t - 1] == 'E') ax = x - 1;
        if (x < paths[y].length() && paths[x][t - 1] != 'W' && paths[y][t - 1] != 'E') ok |= dfs(t + 1, x, y, 'E', opp);
    }
    if (c == 'S') {
        y += 1;
        if (t <= paths[x].length() && paths[x][t - 1] == 'S') ay = y - 1;
        if (y < paths[x].length() && paths[x][t - 1] != 'N' && paths[y][t - 1] != 'S') ok |= dfs(t + 1, x, y, 'S', opp);
    }
    if (c == 'W') {
        x -= 1;
        if (t <= paths[y].length() && paths[y][t - 1] == 'W') ax = x + 1;
        if (x >= 0 && paths[x][t - 1] != 'E' && paths[y][t - 1] != 'W') ok |= dfs(t + 1, x, y, 'W', opp);
    }
    if (c == opp) {
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx < 0 || ny < 0 || nx >= paths[x].length() || ny >= paths[y].length()) continue;
            if (paths[x][t - 1] == 'N' && dy[i] != -1) continue;
            if (paths[x][t - 1] == 'S' && dy[i] != 1) continue;
            if (paths[x][t - 1] == 'E' && dx[i] != 1) continue;
            if (paths[x][t - 1] == 'W' && dx[i] != -1) continue;
            if (paths[y][t - 1] == 'N' && dy[i] != 1) continue;
            if (paths[y][t - 1] == 'S' && dy[i] != -1) continue;
            if (paths[y][t - 1] == 'E' && dx[i] != -1) continue;
            if (paths[y][t - 1] == 'W' && dx[i] != 1) continue;
            ok |= dfs(t + 1, nx, ny, paths[x][t - 1], paths[y][t - 1]);
        }
    }
    visited[ax][ay][opp == c] = 0;
    return ok;
}

int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s;
    paths.push_back(s);
    cin >> t;
    paths.push_back(t);
    int ans = dfs(1, 0, 0, s[0], t[0]);
    if (ans) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
