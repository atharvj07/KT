#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int n;
vector<int> p;
vector<bool> changed;

int main() {
    cin >> n;
    p.resize(n);
    changed.resize(n);
    for (int& x : p) cin >> x;
    int ans = 0;
    for (int i = 0; i < n - 1; i++) {
        if (p[i] > p[i + 1]) {
            ans++;
            if (p[i] > 0) {
                p[i] *= -1;
                changed[i] = true;
            } else {
                p[i + 1] *= -1;
                changed[i + 1] = true;
            }
        }
    }
    for (int i = 1; i + 1 < n; i++) {
        if (changed[i - 1] && changed[i] && p[i] < 0) {
            ans--;
            p[i] *= -1;
        }
    }
    cout << ans << '\n';
}
