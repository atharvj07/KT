#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  if (is_sorted(s.begin(), s.end())) {
    cout << "YES\n";
    return cout << string(n, '0') << "\n", 0;
  }
  vector<int> cnt(26);
  for (char c : s) ++cnt[c - 'a'];
  for (int i = 1; i < 26; ++i)
    if (cnt[i] && cnt[i - 1]) { cout << "NO\n"; return 0; }
  cout << "YES\n";
  for (char c : s) cout << (c == 'a' ? '0' : '1');
  cout << "\n";
}
