import java.util.*;

public class Main {
    static int n, m;
    static char[][] table;
    static int[] pre, suf;
    static int ans = -1;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        table = new char[n][m];
        pre = new int[m];
        suf = new int[m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                table[i][j] = s.charAt(j);
                if (table[i][j] == '1') {
                    pre[j]++;
                }
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < m; j++) {
                if (table[i][j] == '1') {
                    suf[j]++;
                }
            }
        }
        for (int j = 0; j < m; j++) {
            ans = Math.max(ans, pre[j] + suf[j]);
        }
        System.out.println(ans == n ? -1 : n - ans);
    }
}
