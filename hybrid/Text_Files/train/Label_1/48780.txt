import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] ab = new long[n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2; j++) {
                ab[i][j] = sc.nextInt();
            }
        }
        sc.close();
        long ans = get(ab);
        System.out.println(ans);
    }
    static long get(long[][] ab) {
        int n = ab.length;
        long[] g = new long[n];
        for (int i = 0; i < n; i++) {
            g[i] = gcd(ab[i][0], ab[i][1]);
        }
        long gcd = g[0];
        for (int i = 0; i < n; i++) {
            gcd = gcd(gcd, g[i]);
        }
        if (gcd > 1) return gcd;
        for (int i = 2; i <= 1000000; i++) {
            boolean ok = true;
            for (int j = 0; j < n; j++) {
                if (ab[j][0] % i == 0 || ab[j][1] % i == 0) {
                    continue;
                } else {
                    ok = false;
                    break;
                }
            }
            if (ok) return i;
        }
        return -1;
    }
    static long gcd(long a, long b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
