#include<bits/stdc++.h>
using namespace std;
const int MAXN = 1e6 + 5;
int n, a[MAXN], cnt[MAXN], prime[MAXN], mu[MAXN], mx;
vector<pair<int, int>> v;
bool vis[MAXN];
int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> a[i];
    mu[1] = 1;
    for(int i = 2; i <= MAXN; i++) {
        if(!vis[i]) {
            prime[++cnt[0]] = i;
            mu[i] = -1;
        }
        for(int j = 1; j <= cnt[0] && prime[j] * i <= MAXN; j++) {
            vis[i * prime[j]] = 1;
            if(i % prime[j] == 0) {
                mu[i * prime[j]] = 0;
                break;
            }
            mu[i * prime[j]] = -mu[i];
        }
    }
    for(int i = 1; i <= n; i++) {
        int x = a[i], ans = 0;
        for(int j = 1; j <= cnt[0] && prime[j] * prime[j] <= x; j++) {
            if(x % prime[j] == 0) {
                while(x % prime[j] == 0) x /= prime[j];
                ans++;
            }
        }
        if(x != 1) ans++;
        mx = max(mx, ans);
        if(ans == mx) v.emplace_back(i, i);
    }
    int ans = 0;
    for(int i = 1; i <= n; i++) {
        int x = a[i], sum = 0;
        for(int j = 1; j <= cnt[0] && prime[j] * prime[j] <= x; j++) {
            if(x % prime[j] == 0) {
                while(x % prime[j] == 0) x /= prime[j], sum++;
            }
        }
        if(x != 1) sum++;
        if(sum == mx) {
            int l = i, r = i;
            while(l >= 1 && r <= n) {
                int x = a[l], y = a[r], ll = 0, rr = 0;
                for(int j = 1; j <= cnt[0] && prime[j] * prime[j] <= x; j++) {
                    if(x % prime[j] == 0) {
                        while(x % prime[j] == 0) x /= prime[j], ll++;
                    }
                }
                if(x != 1) ll++;
                for(int j = 1; j <= cnt[0] && prime[j] * prime[j] <= y; j++) {
                    if(y % prime[j] == 0) {
                        while(y % prime[j] == 0) y /= prime[j], rr++;
                    }
                }
                if(y != 1) rr++;
                if(ll == mx && rr == mx) {
                    ans++, r++;
                    if(r - l > v.back().second - v.back().first) {
                        v.clear();
                        v.emplace_back(l, r - 1);
                    }
                }
                else if(ll == mx && rr != mx) r++;
                else if(ll != mx && rr == mx) l--;
                else break;
            }
        }
    }
    cout << ans << ' ' << v[0].second - v[0].first << '\n';
    for(auto x: v) cout << x.first << ' ';
}
