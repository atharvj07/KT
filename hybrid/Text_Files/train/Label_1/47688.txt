import java.io.*;
import java.util.*;

class Main {
    static final int maxn = 1000;
    
    static int[][] a = new int[maxn][maxn];
    static int f[] = new int[maxn];
    static int n, m, k;
    static int ansx, ansy;
    static int max;

    public static void main(String[] args) throws IOException {
        input();
        solve();
        output();
    }

    static void input() throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] s = in.readLine().split(" ");
        n = Integer.parseInt(s[0]);
        m = Integer.parseInt(s[1]);
        k = Integer.parseInt(s[2]);
        for (int i = 0; i < n; i++) {
            s = in.readLine().split(" ");
            for (int j = 0; j < m; j++)
                a[i][j] = Integer.parseInt(s[j]);
        }
        in.close();
    }

    static void solve() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++)
                f[(i + j) % (n + m - 2 * k + 1)] = Math.max(f[(i + j) % (n + m - 2 * k + 1)], a[i][j]);
        }

        max = 0;
        for (int i = 0; i < n + m - 2 * k + 1; i++) {
            if (f[i] > max) {
                max = f[i];
                ansx = i % (n + m - 2 * k + 1);
                ansy = (i + 1) % (n + m - 2 * k + 1);
            }
        }
    }

    static void output() {
        System.out.println((ansx + 1) + " " + (ansy + 1));
    }
}
