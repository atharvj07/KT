n, m, p = map(int, input().split())
s = input()
for i in input():
    if i=='L': p=max(p-1,1)
    elif i=='R': p=min(p+1,n)
    else:
        q, o = p, 0
        while s[p-1]==s[p] and p>0: p, o = p-1, o+1
        while s[p]==s[p+1] and p<n-1: p, o = p+1, o+1
        s = s[:p-o]+s[p+1:]
        p = max(p-o,1)
print(s)
