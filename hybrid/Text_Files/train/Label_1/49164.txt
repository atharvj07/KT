import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextLong();
    long p = sc.nextLong();
    long ans = -1;
    if (n % 2 == 0 && n - p >= 0) {
      long cnt = 2;
      n -= p;
      while (n > 0) {
        cnt++;
        n /= 2;
      }
      ans = cnt;
    }
    System.out.println(ans);
  }
}
