n, k = map(int, input().split())
fighters = [list(map(int, input().split())) for _ in range(n)]

def min_effort(n, k, fighters):
    fighters.sort(reverse=True)
    efforts = [e for _, e in fighters]
    points = [p for p, _ in fighters]

    # Calculate minimum efforts for all possible victories
    min_efforts = [float('inf')] * (n + 1)
    min_efforts[0] = 0

    for i in range(n):
        for j in range(i, -1, -1):
            min_efforts[j+1] = min(min_efforts[j+1], min_efforts[j] + efforts[i])

    # Find the rank Manao can achieve with minimum effort
    for wins in range(n, -1, -1):
        manao_points = wins
        rank = 1
        for i in range(n):
            if points[i] > manao_points or (points[i] == manao_points and i < wins):
                rank += 1
        if rank <= k:
            return min_efforts[wins]

    return -1

print(min_effort(n, k, fighters))
