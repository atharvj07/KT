import sys

n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = [list(map(int, input().split())) for _ in range(m)]

a.sort()
b.sort(key=lambda x: x[1])

t = i = 0
for l, r in b:
    while t < l and i < n:
        t += a[i]
        i += 1
    if i == n:
        break
    t = max(t, l)
    t += a[i]
    i += 1

print(-1 if i < n else t)
