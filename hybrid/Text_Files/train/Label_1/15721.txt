#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<vector<int>> m(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) { cin >> m[i][j]; }
    }
    int ans = 0; vector<int> res;
    for (int car = 0; car < n; car++) {
        bool ok = true;
        for (int i = 0; i < n; i++) {
            if (i == car) continue;
            if (m[car][i] == 1 || m[i][car] == 2) { ok = false; }
        }
        if (ok) { ans++; res.push_back(car + 1); }
    }
    cout << ans << endl;
    for (int i = 0; i < ans; i++) cout << res[i] << ' ';
    cout << endl;
}
