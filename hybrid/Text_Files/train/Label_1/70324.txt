import sys
readline = sys.stdin.readline

t = int(readline())
for _ in range(t):
    n, x = map(int, readline().split())
    a = list(map(int, readline().split()))
    b = sorted(a)
    f = 1
    for i in range(n):
        if a[i] != b[i]:
            if abs(a.index(b[i]) - i) < x:
                f = 0
    if f: print("YES")
    else: print("NO")
