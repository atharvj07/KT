#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a;
  cin >> n >> a;
  vector<int> x(n);
  for (int &xi : x) cin >> xi;
  sort(x.begin(), x.end());
  int ans = 0;
  if (a < x[0])
    ans += x[0] - a;
  else if (a > x.back())
    ans += a - x.back();
  int r = 0;
  for (int i = 0; i < n - 1;) {
    while (i < n - 1 && x[i] <= a) ++i;
    if (i == n - 1) break;
    int j = i + 1;
    while (j < n && x[j] >= a) ++j;
    if (j == n) break;
    while (i < n - 1 && x[i] > a) {
      ans += x[i + 1] - x[i];
      ++i;
    }
    while (j < n - 1 && x[j] < a) {
      ++j;
      ++r;
    }
  }
  if (r > 0) ans += min(r - 1, n - 1 - r) * x.back();
  cout << ans << endl;
  return 0;
}
