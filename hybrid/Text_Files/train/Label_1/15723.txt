#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 1e9;
    int l = -1;
    int r = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] <= k) {
            if (i < m) l = i;
            if (i > m) r = i;
        }
    }
    if (l == -1) l = r;
    if (r == -1) r = l;
    ans = min(ans, abs(l - m) * 10);
    ans = min(ans, abs(r - m) * 10);
    cout << ans << endl;
}
