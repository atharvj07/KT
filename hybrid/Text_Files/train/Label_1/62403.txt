n, q = map(int, input().split())
s = input()
counts = sum(s[i:i+3] == 'abc' for i in range(n-2))

for _ in range(q):
    pos, c = input().split()
    pos = int(pos) - 1
    
    for i in range(max(0, pos-2), min(n-2, pos)+1):
        if s[i:i+3] == 'abc':
            counts -= 1
            
    s = s[:pos] + c + s[pos+1:]
    
    for i in range(max(0, pos-2), min(n-2, pos)+1):
        if s[i:i+3] == 'abc':
            counts += 1
    
    print(counts)
