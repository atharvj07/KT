import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, s, k = list(map(int, input().split()))
    a = set(list(map(int, input().split())))
    ans = n
    for i in range(s, n+1):
        if i not in a:
            ans = min(ans, i-s)
    for i in range(s, 0, -1):
        if i not in a:
            ans = min(ans, s-i)
    print(ans)
