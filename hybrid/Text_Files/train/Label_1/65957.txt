def min_operations_to_sort(t, test_cases):
    results = []
    for n, a in test_cases:
        b = sorted((v, i) for i, v in enumerate(a))
        dp = [0] * (n + 1)
        for v, i in b:
            dp[i + 1] = dp[i] + 1
        l = max(dp)
        results.append(n - l)
    return results

t = int(input())
test_cases = []
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    test_cases.append((n, a))

results = min_operations_to_sort(t, test_cases)
for result in results:
    print(result)
