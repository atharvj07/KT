import sys

s, t = map(list, sys.stdin.read().split())
ans = []
for letter in s:
    if letter not in t:
        print(-1)
        exit()
for i in range(len(s)):
    if s[i] != t[i]:
        j = 0
        while i+j < len(s) and s[i+j] != t[i]:
            j += 1
        if i+j == len(s):
            print(-1)
            exit()
        ans.append((s[i], t[i]))
        # swap
        s[i], t[i+j] = s[i+j], t[i]
print(len(ans))
for a in ans:
    print(*a)
