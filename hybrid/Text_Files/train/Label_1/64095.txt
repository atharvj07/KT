import sys
MAXN = int(1e9)
INF = 2*MAXN
N, M = map(int, input().split())
C = int(input())
hotels = [list(map(int, input().split())) for i in range(C)]
H = int(input())
restaurants = [list(map(int, input().split())) for i in range(H)]
ans = INF
for rest in restaurants:
    (x, y) = rest
    t = 0
    for hotel in hotels:
        (a, b) = hotel
        t = max(t, abs(x-a)+abs(y-b))
    if t < ans:
        ans = t
        res = rest
print(ans)
print(restaurants.index(res)+1)
