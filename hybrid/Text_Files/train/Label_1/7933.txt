#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) { cin >> x; }
  vector<int> ans(n - 1);
  for (int k = 1; k < n; k++) {
    int cnt = 0;
    for (int i = 1; i + k <= n; i++) {
      if (a[i] < a[(i + k) / k]) { cnt++; }
    }
    ans[k - 1] = cnt;
  }
  for (int x : ans) { cout << x << " "; }
  cout << endl;
  return 0;
}
