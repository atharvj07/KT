import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;

public class Main {
    private static int[] zFunction(char[] s, int n) {
        int[] z = new int[n];
        for (int i = 1, l = 0, r = 0; i < n; i++) {
            if (i <= r) {
                z[i] = Math.min(r - i + 1, z[i - l]);
            }
            while (i + z[i] < n && s[z[i]] == s[i + z[i]]) {
                z[i]++;
            }
            if (i + z[i] - 1 > r) {
                l = i;
                r = i + z[i] - 1;
            }
        }
        return z;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String s = reader.readLine();
        char[] t = s.toCharArray();
        int n = s.length();
        int[] z = zFunction(t, n);
        int q = Integer.parseInt(reader.readLine());
        HashMap<String, Integer> map = new HashMap<>();
        while (q-- > 0) {
            String[] parts = reader.readLine().split(" ");
            int type = Integer.parseInt(parts[0]);
            if (type == 1) {
                int index = Integer.parseInt(parts[1]) - 1;
                t[index] = parts[2].charAt(0);
                z = zFunction(t, n);
            } else if (type == 2) {
                int l = Integer.parseInt(parts[1]) - 1;
                int r = Integer.parseInt(parts[2]) - 1;
                String y = parts[3];
                int yLen = y.length();
                if (!map.containsKey(y)) {
                    map.put(y, Arrays.stream(zFunction(y.toCharArray(), yLen)).max().getAsInt());
                }
                int count = 0;
                for (int i = l; i + yLen <= r; i++) {
                    if (z[i] >= yLen && t[i + yLen - 1] == y.charAt(yLen - 1)) {
                        count++;
                    }
                }
                System.out.println(count);
            }
        }
    }
}
