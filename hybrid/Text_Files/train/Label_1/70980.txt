import sys

n, a, b = map(int, input().split())
s = list(map(int, input().split()))

if s[0] != s[-1]:
    print(-1)
    sys.exit()

L = [0] * n
R = [0] * n

if s[0] == 2:
    L[0] = min(a, b)
else:
    L[0] = 0

for i in range(1, n):
    if s[i] == 2:
        if s[i - 1] == 0:
            L[i] = L[i - 1] + a
        elif s[i - 1] == 1:
            L[i] = L[i - 1] + b
    elif s[i] == s[i - 1]:
        L[i] = L[i - 1]

if s[-1] == 2:
    R[-1] = min(a, b)
else:
    R[-1] = 0

for i in range(n - 2, -1, -1):
    if s[i] == 2:
        if s[i + 1] == 0:
            R[i] = R[i + 1] + a
        elif s[i + 1] == 1:
            R[i] = R[i + 1] + b
    elif s[i] == s[i + 1]:
        R[i] = R[i + 1]

ans = []

for i in range(n):
    if s[i] == 2:
        ans.append(min(L[i] + R[i], a + b))

print(sum(ans))
