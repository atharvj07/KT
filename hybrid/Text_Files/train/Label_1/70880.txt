import sys

n, q = [int(i) for i in input().split()]
data = [0]*n
unread = 0

for _ in range(q):
    t, *args = [int(i) for i in input().split()]
    if t == 1:
        data[args[0]-1] += 1
        unread += 1
    elif t == 2:
        unread -= data[args[0]-1]
        data[args[0]-1] = 0
    else:
        unread = 0
        for i in range(args[0]):
            unread += data[i]
    sys.stdout.write(str(unread)+'\n')
