import java.util.*;
import java.io.*;

public class Main {
    static int n, x;
    static int[] a;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            n = in.nextInt();
            x = in.nextInt();
            a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int ans = 0;
            boolean infected = false;
            for (int i = 0; i < n; i++) {
                if (a[i] == x) {
                    infected = true;
                }
            }
            if (!infected) {
                ans++;
                for (int i = 0; i < n; i++) {
                    a[i] = x;
                }
            }
            System.out.println(ans);
        }
    }
}
