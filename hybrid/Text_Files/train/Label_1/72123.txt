import sys
input = sys.stdin.readline

n, x1, y1, x2, y2 = map(int, input().split())
flowers = []
for _ in range(n):
    flowers.append(tuple(map(int, input().split())))

max_x = max(map(lambda p: abs(p[0] - x1), flowers)) + 1
max_y = max(map(lambda p: abs(p[1] - y1), flowers)) + 1
min_x = max(map(lambda p: abs(p[0] - x2), flowers)) + 1
min_y = max(map(lambda p: abs(p[1] - y2), flowers)) + 1
print(min(max_x ** 2 + max_y ** 2, min_x ** 2 + min_y ** 2))
