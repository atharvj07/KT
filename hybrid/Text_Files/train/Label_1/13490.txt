#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int MAXN = 1e5 + 5;

int n, arr[MAXN], ans[MAXN];
ll dp[MAXN][3];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr), cout.tie(nullptr);
  cin >> n;
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    cin >> arr[i];
    cnt += arr[i];
  }
  if (cnt == n) {
    cout << "0\n";
    return 0;
  }
  dp[n][0] = INT_MAX;
  for (int i = n - 1; i >= 1; i--) {
    dp[i][0] = 0;
    if (arr[i] == 2) dp[i][0] = 1;
    dp[i][1] = dp[i + 1][0];
    dp[i][2] = dp[i + 1][1];
    if (arr[i] == 1 || arr[i] == 2) dp[i][1]++, ans[++ans[0]] = i;
    if (arr[i] == 1 || arr[i] == 2 || arr[i] == 3) dp[i][2]++, ans[++ans[0]] = i;
  }
  cout << ans[0] << '\n';
  for (int i = 1; i <= ans[0]; i++) cout << ans[i] << ' ';
  cout << '\n';
  return 0;
}
