#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
    if (sum > s) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
