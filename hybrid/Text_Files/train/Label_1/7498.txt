#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int l, r;
    cin >> l >> r;
    if (l == r)
      cout << l << " " << l + 1 << "\n";
    else if (l != 1)
      cout << l << " " << l * 2 << "\n";
    else
      cout << l + 1 << " " << l + 3 << "\n";
  }
  return 0;
}
