from collections import defaultdict, deque
def dfs(u, p):
    vis[u] = True
    for v in adj[u]:
        if not vis[v]:
            dfs(v, u)
            sum_nodes[u] += sum_nodes[v]
            sum_ones[u] += sum_ones[v]
    sum_ones[u] += ones[u]
    sum_nodes[u] += nodes[u]

def bfs(u):
    q = deque([u])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if not vis[v]:
                if u == par[v]:
                    sum_ones[v] = sum_ones[u] + nodes[v] - sum_ones[v]
                else:
                    sum_ones[v] = sum_ones[p] - sum_ones[v] + sum_ones[u]
                sum_nodes[v] = sum_nodes[u] + nodes[v]
                par[v] = u
                q.append(v)

n = int(input())
adj = defaultdict(list)
for i in range(n - 1):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)
vis = [False] * (n + 1)
sum_ones = [0] * (n + 1)
sum_nodes = [0] * (n + 1)
nodes = [0] * (n + 1)
ones = [0] * (n + 1)
par = [0] * (n + 1)
for i in range(1, n + 1):
    a, b, c = map(int, input().split())
    ones[i] = b
    nodes[i] = a
    if b != c:
        vis[i] = True
for i in range(1, n + 1):
    if not vis[i]:
        dfs(i, i)
        bfs(i)
ans = 0
for i in range(1, n + 1):
    ans += (sum_nodes[i] - sum_ones[i]) * nodes[i]
if ans == -1:
    print(-1)
else:
    print(ans)
