#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<pair<int, int>> landings(n);
  for (int i = 0; i < n; i++) { cin >> landings[i].first >> landings[i].second; }
  sort(landings.begin(), landings.end());
  for (int i = 0; i < n; i++) {
    int h0 = landings[i].first;
    int m0 = landings[i].second;
    if (i > 0 && h0 - landings[i - 1].first > 2 * s) {
      cout << landings[i - 1].first + s << " " << landings[i - 1].second << endl;
      return 0;
    }
    if (i == 0 || h0 - landings[i - 1].first > s) {
      cout << h0 - s << " " << m0 << endl;
      return 0;
    }
  }
  cout << landings.back().first + s << " " << landings.back().second << endl;
  return 0;
}
