#include <iostream>
using namespace std;
int main() {
    int h, w;
    cin >> h >> w;
    string s[h];
    for (int i = 0; i < h; i++)
        cin >> s[i];
    bool ok = 1;
    int x, y;
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            if (s[i][j] == '*') {
                if (!ok) {
                    ok = 0;
                    break;
                }
                x = i, y = j;
                ok = 1;
            }
        }
    }
    for (int j = y - 1; j >= 0; j--) { if (s[x][j] != '*') ok = 0; }
    for (int j = y + 1; j < w; j++) { if (s[x][j] != '*') ok = 0; }
    for (int i = x - 1; i >= 0; i--) { if (s[i][y] != '*') ok = 0; }
    for (int i = x + 1; i < h; i++) { if (s[i][y] != '*') ok = 0; }
    cout << (ok ? "YES" : "NO") << endl;
}
