import java.util.Scanner;

public class CaracalAndMonster {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long H = scanner.nextLong();

        long attacks = 0;
        while (H > 0) {
            attacks++;
            H = (long) Math.floor(H / 2);
        }

        System.out.println(attacks);
    }
}
