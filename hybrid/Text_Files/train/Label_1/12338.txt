#include <bits/stdc++.h>

using namespace std;
using ll = long long;
const ll mod = 1e9 + 7;

int main() {
    ll n, q;
    cin >> n >> q;
    while (q--) {
        ll x, y;
        cin >> x >> y;
        (x += y, x += n - 1) %= n;
        cout << (x & 1 ? n * n - x / 2 : x / 2 + 1) << '\n';
    }
    return 0;
}
