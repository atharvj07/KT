import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
d, ans = {}, ''
for i in range(n):
    for j in range(i+1, n):
        if a[i]+a[j] not in d:
            d[a[i]+a[j]] = [[i+1, j+1]]
        else:
            d[a[i]+a[j]].append([i+1, j+1])
for i in d.values():
    if len(i)>1:
        for j in i:
            for k in i:
                if j!=k and j[0]!=k[0] and j[0]!=k[1] and j[1]!=k[0] and j[1]!=k[1]:
                    ans = 'YES\n'+' '.join(map(str, j+k))
                    print(ans);exit(0)
print('NO')
