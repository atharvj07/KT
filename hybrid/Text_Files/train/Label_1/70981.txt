n,a,b=map(int,input().split())
x=[int(x) for x in input().split()]
if x[0]!=x[-1]:
    print(-1)
else:
    c=0
    for i in range(1,n//2):
        if x[i]!=x[n-i-1] and not (x[i]==2 and x[n-i-1]==2):
            print(-1)
            exit()
        if x[i]==2 or x[n-i-1]==2:
            c+=min(a,b)
    print(c)
