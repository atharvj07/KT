#include <iostream>
#include <vector>
#include <algorithm>
#include <string>

using namespace std;

string s;

int main() {
    cin >> s;
    vector<int> f(26, 0);
    for (char a : s) f[a - 'a']++;
    int ans = 0;
    for (int i = 0; i < f['n' - 'a']; i++) {
        int now = 1;
        for (int j = i + 1; j < i + 7; j++) {
            now &= f[j - 'a'] > 0;
        }
        ans += now;
        if (!now) break;
    }
    cout << ans << endl;
    return 0;
}
