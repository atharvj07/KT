import math
for _ in range(int(input())):
    n = int(input())
    a, b, tm = [0] * n, [0] * n, [0] * n
    for i in range(n):
        a[i], b[i] = map(int, input().split())
    tm = list(map(int, input().split()))
    cur = 0
    for i in range(n):
        x = math.ceil((b[i] - a[i]) / 2)
        if cur + x <= b[i]:
            cur = b[i]
        else:
            cur += x
        cur += tm[i]
    print(cur)
