from collections import deque

def find_min_height(n, a):
    depth = 0
    queue = deque([1])
    for i in range(1, n):
        while a[i] > queue[-1]:
            queue.pop()
            if not queue:
                break
        queue.append(a[i])
        depth = max(depth, len(queue))
    return depth

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(find_min_height(n, a))
