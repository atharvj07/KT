#include <cstdio>

const int MAXN = 1e5 + 10;
const int MOD = 1e9 + 7;

int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  static long long a[MAXN], pref[MAXN], suf[MAXN], inv[MAXN];
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
  }
  long long ans = 0;
  pref[0] = suf[n + 1] = 1;
  inv[1] = 1;
  for (int i = 2; i <= n; i++) { inv[i] = (MOD - (long long)(MOD / i) * inv[MOD % i]) % MOD; }
  for (int i = 1; i <= n; i++) { pref[i] = (pref[i - 1] * a[i]) % MOD; }
  for (int i = n; i >= 1; i--) { suf[i] = (suf[i + 1] * a[i]) % MOD; }
  for (int x = 0; x <= n; x++) {
    for (int y = 0; x + y + 1 <= n; y++) {
      long long p = (x ? pref[x - 1] : 1) * (y ? suf[y + 1] : 1) % MOD;
      ans += (p * inv[k]) % MOD;
    }
  }
  printf("%lld\n", ans % MOD);
  return 0;
}
