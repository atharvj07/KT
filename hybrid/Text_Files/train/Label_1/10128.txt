#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  sort(a.begin(), a.end());
  sort(b.begin(), b.end(), greater<int>());
  int sa = 0, sb = 0;
  for (int i = 0; i < min(n, (int)(a.size())); i++) { sa += a[i]; }
  for (int i = 0; i < min(n, (int)(b.size())); i++) { sb += b[i]; }
  cout << sa - sb << endl;
  return 0;
}
