import java.util.ArrayDeque;
import java.util.Queue;
import java.util.Scanner;

public class Jaber {

    static int n, m, k;
    static int[][] grid;
    static boolean[][] visited;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        k = scanner.nextInt();

        grid = new int[n][m];
        visited = new boolean[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt() - 1;
            }
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;
            System.out.println(bfs(r1, c1, r2, c2));
        }
    }

    static int bfs(int r1, int c1, int r2, int c2) {
        Queue<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{r1, c1, 0});
        visited[r1][c1] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int row = current[0];
            int col = current[1];
            int time = current[2];

            if (row == r2 && col == c2) {
                return time;
            }

            // Check adjacent cells
            for (int i = -1; i <= 1; i++) {
                for (int j = -1; j <= 1; j++) {
                    int newRow = row + i;
                    int newCol = col + j;
                    if (newRow >= 0 && newRow < n && newCol >= 0 && newCol < m && !visited[newRow][newCol] && (i != 0 || j != 0 || grid[row][col] == grid[newRow][newCol])) {
                        queue.offer(new int[]{newRow, newCol, time + 1});
                        visited[newRow][newCol] = true;
                    }
                }
            }
        }

        return -1;
    }
}
