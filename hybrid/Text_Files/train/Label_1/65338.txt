from sys import stdin

n = int(stdin.readline())
l = list(map(int, stdin.readline().split()))
for x, y in [map(int, stdin.readline().split()) for i in range(int(stdin.readline()))]:
    for i in range(y - 1, len(l)):
        l[i] += l[i - 1]
    l[y - 1] = -1
for x in l: print(x + 1 if x >= 0 else 0)
