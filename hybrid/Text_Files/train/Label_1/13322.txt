#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ll n, f;
    cin >> n >> f;
    vector<pair<ll, ll>> v(n);
    for (ll i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end(), [](const pair<ll, ll> &a, const pair<ll, ll> &b) {
        return a.second < b.second;
    });
    ll ans = 0;
    for (ll i = 0; i < min(n, f); i++)
        ans += min(v[i].first * 2, v[i].second);
    for (ll i = f; i < n; i++)
        ans += min(v[i].first, v[i].second);
    cout << ans << endl;
    return 0;
}
