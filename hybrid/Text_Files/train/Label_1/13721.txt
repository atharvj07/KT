#include<bits/stdc++.h>
#define fast ios::sync_with_stdio(false);cin.tie(0);cout.tie(0)
using namespace std;

typedef long long ll;
const int maxn=1e3+10;
const int mod=1e9+7;

int n,m,v[maxn][maxn];
char grid[maxn][maxn];

int dx[]={0,0,1,-1};
int dy[]={1,-1,0,0};

int get(int x, int y){
    int ans=0, val;
    queue<pair<int,int>> q;
    q.push({x,y});
    v[x][y]=1;
    while(!q.empty()){
        x=q.front().first;
        y=q.front().second;
        q.pop();
        ans++;
        for(int i=0; i<4; i++){
            int nx=x+dx[i];
            int ny=y+dy[i];
            if(nx<0 || nx>=n || ny<0 || ny>=m || v[nx][ny] || grid[nx][ny]=='*') continue;
            v[nx][ny]=1;
            q.push({nx,ny});
        }
    }
    return ans%10;
}

int main(){
    fast;
    cin>>n>>m;
    for(int i=0; i<n; i++){
        for(int j=0; j<m; j++){
            cin>>grid[i][j];
        }
    }
    for(int i=0; i<n; i++){
        for(int j=0; j<m; j++){
            if(grid[i][j]=='.') cout<<'.';
            else cout<<get(i,j);
        }
    }
}
