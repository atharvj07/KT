#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 9;
int n, col[N], par[N], sz[N], mark[N];
vector<int> adj[N];
void dfs1(int u, int p)
{
    par[u] = p;
    sz[u] = 1;
    for(auto v : adj[u])
    {
        if(v == p)    continue;
        dfs1(v, u);
        sz[u] += sz[v];
    }
}
void dfs2(int u, int p, int color)
{
    col[u] = color;
    mark[color]++;
    for(auto v : adj[u])
    {
        if(v == p || mark[color] > 1)  continue;
        dfs2(v, u, color);
    }
    for(auto v : adj[u])
    {
        if(v == p)  continue;
        dfs2(v, u, color + 1);
    }
}
int main()
{
    cin >> n;
    for(int i = 2; i <= n; i++)
    {
        int p;  cin >> p;
        adj[i].push_back(p);
        adj[p].push_back(i);
    }
    dfs1(1, 1);
    dfs2(1, 1, 1);
    vector<int> ans(n + 1);
    for(int i = 1; i <= n; i++)
    {
        if(mark[i] > 0)    ans[mark[i]]++;
    }
    for(int i = 1; i <= n; i++)
    {
        cout << ans[i] << ' ';
    }
    return 0;
}
