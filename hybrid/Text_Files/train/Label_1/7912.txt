#include<bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
#define deb(x) cout<<#x<<':'<<x<<endl

int bs(int n){
    return ((n-2)*(n+1)*(n+1))/8;
}
void solve(){
    int n; cin>>n;
    int mx=0, sum=0;
    for(int i=0,x;i<n;i++){
        cin>>x;
        sum+=(x-i-1);
        mx=max(mx,x);
    }
    int ans=sum-bs(mx);
    cout<<ans<<endl;
}
int32_t main(){
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t; cin>>t;
    while(t--) solve();
}
