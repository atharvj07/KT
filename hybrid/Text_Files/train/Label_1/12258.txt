#include <iostream>
#include <string>
#include <bitset>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    bitset<26> b;
    for (int i = 0; i < k; i++) {
        char c;
        cin >> c;
        b[c - 'a'] = 1;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        bool ok = true;
        for (int j = i; j < n; j++) {
            if (!b[s[j] - 'a']) {
                ok = false;
                break;
            }
            ans += ok;
        }
    }
    cout << ans << endl;
    return 0;
}
