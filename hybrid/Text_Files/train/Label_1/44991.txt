import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int h = scanner.nextInt();
        int[] height = new int[n];
        for (int i = 0; i < n; i++) {
            height[i] = scanner.nextInt();
        }
        int width = 0;
        for (int i = 0; i < n; i++) {
            if (height[i] > h) {
                width += 2;
            } else {
                width += 1;
            }
        }
        System.out.println(width);
    }
}
