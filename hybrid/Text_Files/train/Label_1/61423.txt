t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    b.sort()
    total = sum(b[:-1])
    # Check if removing the last element results in a valid array
    if total - b[-2] == b[-2]:
        print(*b[:-2])
    else:
        total -= b[-1]
        if total - b[-1] in b[:-1]:
            b.remove(total - b[-1])
            print(*b[:-1])
        else:
            print(-1)
