import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long p = scanner.nextLong();
        long ans = 0;
        if (p < 0) {
            ans++;
            n += p;
        }
        while (n > 0) {
            if (n % 2 == 0) {
                n /= 2;
            } else {
                n += p;
                ans++;
            }
            if (n < 0) {
                ans = -1;
                break;
            }
        }
        System.out.print(ans);
    }
}
