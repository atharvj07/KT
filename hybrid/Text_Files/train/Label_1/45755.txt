import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        if (k % 2 == 1) {
            System.out.println("-1");
            return;
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < k; j++) {
                if ((i + j) % 2 == 0) {
                    System.out.print("w");
                } else {
                    System.out.print("b");
                }
            }
            System.out.println();
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < k; j++) {
                if ((i + j) % 2 == 0) {
                    System.out.print("b");
                } else {
                    System.out.print("w");
                }
            }
            System.out.println();
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < k; j++) {
                if ((i + j) % 2 == 0) {
                    System.out.print("w");
                } else {
                    System.out.print("b");
                }
            }
            System.out.println();
        }
    }
}
