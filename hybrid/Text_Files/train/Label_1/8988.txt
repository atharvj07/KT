#include <iostream>
#include <vector>
#include <set>
#include <stack>

int main() {
    int n;
    std::cin >> n;

    std::vector<int> hands(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> hands[i];
    }

    std::set<int> available;
    for (int i = 0; i < n; ++i) {
        available.insert(i+1);
    }

    std::stack<int> team;
    std::vector<int> order;

    for (int i = 0; i < n; ++i) {
        if (team.empty()) {
            available.erase(hands[i]+1);
            order.push_back(hands[i]+1);
            team.push(hands[i]+1);
        } else {
            if (team.size() == 2 && hands[i] == 0) {
                available.erase(hands[i]+1);
                team.push(hands[i]+1);
                order.push_back(hands[i]+1);
            } else if (team.size() == 2) {
                int top = team.top();
                team.pop();
                int next = team.top();
                team.pop();
                if (hands[i] == next) {
                    available.erase(hands[i]+1);
                    team.push(hands[i]+1);
                    order.push_back(hands[i]+1);
                    team.push(next);
                    team.push(top);
                } else if (hands[i] == top) {
                    available.erase(top);
                    order.push_back(top);
                    team.push(next);
                    team.push(top);
                } else {
                    std::cout << "Impossible" << std::endl;
                    return 0;
                }
            } else if (team.size() == 1) {
                if (hands[i] == 0) {
                    team.push(hands[i]+1);
                    order.push_back(hands[i]+1);
                } else if (hands[i] == 1) {
                    int top = team.top();
                    team.pop();
                    available.erase(top);
                    order.push_back(top);
                    team.push(hands[i]+1);
                } else {
                    std::cout << "Impossible" << std::endl;
                    return 0;
                }
            }
        }

        while (!team.empty() && available.find(team.top()) == available.end()) {
            int top = team.top();
            team.pop();
            order.push_back(top);
        }
    }

    if (team.empty() && available.empty()) {
        std::cout << "Possible" << std::endl;
        for (auto& i : order) {
            std::cout << i << " ";
        }
        std::cout << std::endl;
    } else {
        std::cout << "Impossible" << std::endl;
    }

    return 0;
}
