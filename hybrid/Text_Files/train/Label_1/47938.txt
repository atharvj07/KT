import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        Arrays.fill(dp, n);
        dp[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (h[i] > h[i + 1]) {
                dp[i] = dp[i + 1];
            } else {
                int min = Integer.MAX_VALUE;
                for (int j = i + 1; j < n; j++) {
                    if (h[j] > h[i]) {
                        min = Math.min(min, dp[j]);
                    }
                }
                if (min < Integer.MAX_VALUE) {
                    dp[i] = min + 1;
                }
            }
        }
        System.out.println(dp[0]);
    }
}
