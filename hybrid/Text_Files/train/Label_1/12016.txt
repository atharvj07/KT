#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int64_t> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  int l = 0, r = 0, ans = 1;
  int64_t lst = -1e18;
  while (r < n) {
    while (r < n && lst < v[r]) { lst = v[r]; r++; }
    while (r < n && lst >= v[r]) { r++; }
    ans = max(ans, r - l);
    lst = v[l];
    l++;
  }
  cout << ans << endl;
}
