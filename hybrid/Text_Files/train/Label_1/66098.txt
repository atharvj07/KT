import sys

v = int(sys.stdin.readline())
a = [int(x) for x in sys.stdin.readline().split()]
paint = v
num = []
i = 9
while paint >= a[0] and i >= 1:
    if paint >= a[i]:
        num.append(str(i))
        paint -= a[i]
    else:
        i -= 1
if len(num) == 0:
    print(-1)
else:
    print(''.join(num))
