#include <iostream>
bool check(int x, int n){
    int sum=x;
    while(x>0){
        sum+=x%10;
        x/=10;
    }
    if(sum==n) return true;
    else return false;
}
int main() {
    int n;
    std::cin>>n;
    int cnt=0;
    for(int i=0;i*10<=n;++i){
        if(check(i,n)) ++cnt;
    }
    std::cout<<cnt<<'\n';
    for(int i=0;i*10<=n;++i){
        if(check(i,n)) std::cout<<i<<'\n';
    }
}
