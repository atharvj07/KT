import sys
input = sys.stdin.readline

n, m = map(int, input().split())
s = input()
open = 0
mod = 1000000007
dp = [[0] * 2 for _ in range(m + 1)]
dp[m][0] = dp[m][1] = 1
for i in range(m - 1, -1, -1):
    if s[i] == '(':  open += 1
    for opened in range(open, -1, -1):
        dp[i][0] = dp[i + 1][1]
        if opened > 0:  dp[i][1] = (dp[i + 1][0] + dp[i + 1][1]) % mod
print(dp[0][0])
