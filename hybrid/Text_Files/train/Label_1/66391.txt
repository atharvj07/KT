def find_bob_first_move(n, k, a, m, shots):
    # Minimum space needed to place k ships with a gap of at least 1 between them
    min_space_needed = k * (a + 1) - 1
    if min_space_needed > n:  # If the minimum space needed is more than the board size, Alice definitely cheated
        return 1

    # Create an array to track hits
    hit_tracker = [0] * (n + 2)
    for shot in shots:
        hit_tracker[shot] = 1

    # Calculate prefix sums to quickly find the number of hits in any segment
    for i in range(1, n + 2):
        hit_tracker[i] += hit_tracker[i - 1]

    def can_place_ships(x):
        # Function to check if it's possible to place all ships given the current state of hits
        remaining_ships = k
        i = 1
        while i <= n - a + 1:
            # If the segment from i to i+a-1 is clear (no hits), place a ship
            if hit_tracker[i + a - 1] - hit_tracker[i - 1] == 0:
                remaining_ships -= 1
                i += a  # Move to the next possible start after placing a ship
            else:
                i += 1
            if remaining_ships == 0:
                return True
        return False

    # Binary search through Bob's moves to find the first move after which Alice must have cheated
    left, right = 1, m + 1
    while left < right:
        mid = (left + right) // 2
        if can_place_ships(mid):
            left = mid + 1
        else:
            right = mid
    return left if left <= m else -1

# Input reading
n, k, a = map(int, input().split())
m = int(input())
shots = list(map(int, input().split()))

# Function call and print the result
print(find_bob_first_move(n, k, a, m, shots))
