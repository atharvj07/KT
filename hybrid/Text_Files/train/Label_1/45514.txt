import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int cases = 0;
        while (input.hasNext()) {
            int an = input.nextInt(), bn = input.nextInt(), r = input.nextInt();
            if (an == 0 && bn == 0 && r == 0) break;
            int[][] acm = new int[an][2], icpc = new int[bn][2];
            for (int i = 0; i < an; i++) {
                acm[i][0] = input.nextInt();
                acm[i][1] = input.nextInt();
            }
            for (int i = 0; i < bn; i++) {
                icpc[i][0] = input.nextInt();
                icpc[i][1] = input.nextInt();
            }
            int ans = 0;
            for (int[] p : acm) {
                int cnt = 0;
                for (int[] b : icpc) {
                    if (Math.sqrt(Math.pow(p[0] - b[0], 2) + Math.pow(p[1] - b[1], 2)) <= 2 * r) cnt++;
                }
                ans += cnt;
            }
            System.out.println(ans);
            cases++;
        }
    }
}
