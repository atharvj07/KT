#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define INF 1000000000

int n,a[300005],b[300005];
bool vis[300005];
pair<int,ll> dp[300005];
vector<int> ans;

pair<int,ll> solve(ll x){
    if(x==0) return {0,0};
    if(vis[x]) return dp[x];
    vis[x]=true;
    pair<int,ll> anss={INF,INF};
    for(ll i=1;i<=a[x];i++){
        auto p=solve(x-i);
        if(p.first!=INF){
            anss=min(anss,{p.first+1,p.second+x-i});
        }
    }
    dp[x]=anss;
    return anss;
}

int main(){
    cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++) cin>>b[i];
    auto p=solve(n);
    if(p.first==INF) cout<<-1<<endl;
    else{
        cout<<p.first<<endl;
        ll x=n;
        while(x){
            for(ll i=1;i<=a[x];i++){
                auto p=solve(x-i);
                if(p.first+1==p.second+x-i){
                    ans.push_back(x);
                    x-=i;
                    break;
                }
            }
        }
        reverse(ans.begin(),ans.end());
        for(auto it:ans) cout<<it<<" ";
        cout<<endl;
    }
}
