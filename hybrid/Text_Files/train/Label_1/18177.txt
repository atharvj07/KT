#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> intervals(n);
    for (auto &interval : intervals) cin >> interval.first >> interval.second;
    sort(intervals.begin(), intervals.end());
    int ans = 0;
    int last = 0;
    for (auto interval : intervals) {
        if (interval.first > last) ans += interval.first - last;
        last = max(last, interval.second);
    }
    ans += 100 - last;
    cout << ans << endl;
    return 0;
}
