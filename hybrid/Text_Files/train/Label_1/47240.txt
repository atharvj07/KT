import java.util.Scanner;

public class RegularBracketSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        while (t-- > 0) {
            String s = scanner.nextLine();
            int maxMoves = countMaxMoves(s);
            System.out.println(maxMoves);
        }
    }

    private static int countMaxMoves(String s) {
        int n = s.length();
        char[] chars = s.toCharArray();

        int openCount = 0;
        int maxMoves = 0;

        for (char c : chars) {
            if (c == '(' || c == '[') {
                openCount++;
            } else if (c == ')' || c == ']') {
                openCount--;
                if (openCount < 0) {
                    openCount = 0;
                } else {
                    maxMoves++;
                }
            }
        }

        return maxMoves;
    }
}
