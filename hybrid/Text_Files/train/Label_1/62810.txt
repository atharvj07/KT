n, m = map(int, input().split())
image = [input() for _ in range(n)]

def min_toggle_to_compressible(n, m, image):
    max_k = min(n, m)
    min_toggle = float('inf')
    
    for k in range(2, max_k + 1):
        if n % k == 0 and m % k == 0:
            continue
        toggle_count = 0
        for row in range(0, n, k):
            for col in range(0, m, k):
                ones = 0
                zeros = 0
                for i in range(k):
                    if row + i < n:
                        for j in range(k):
                            if col + j < m:
                                if image[row + i][col + j] == '1':
                                    ones += 1
                                else:
                                    zeros += 1
                            else:
                                zeros += 1
                    else:
                        zeros += k
                toggle_count += min(ones, zeros)
        min_toggle = min(min_toggle, toggle_count)
    
    return min_toggle

print(min_toggle_to_compressible(n, m, image))
