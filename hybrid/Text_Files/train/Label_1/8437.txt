#include <bits/stdc++.h>
using namespace std;
int main() {
    int w, l, a[100000];
    cin >> w >> l;
    for (int i = 0; i < w - 1; i++) {
        cin >> a[i];
        if (!a[i]) {
            a[i] = -1;
        }
    }
    int ans = 0, p = -1;
    for (int i = 0; i < w - 1; i++) {
        if (a[i] == 1) {
            a[i] = a[p] + 1;
            ans += a[p] + 1;
            p = i;
        } else if (a[i] > 0) {
            a[i] = p < i - l ? -1 : a[p] + 1;
            if (a[i] != -1) {
                ans += a[i];
            }
            p = i;
        }
    }
    cout << ans << endl;
}
