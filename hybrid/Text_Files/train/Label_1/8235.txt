#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        string s;
        cin >> n >> s;
        vector<string> ans;
        long long cur = 0, lst = 0;
        for (int i = 0; i < n; i++) {
            cur = cur * 10 + s[i] - '0';
            if (cur > lst || i == 0) {
                ans.push_back(s.substr(i, 1));
                lst = cur;
            } else {
                ans.back() += s[i];
            }
        }
        if (ans.size() < 2) {
            cout << "NO" << endl;
        } else {
            cout << "YES" << endl;
            cout << ans.size() << endl;
            for (auto x : ans) { cout << x << ' '; }
            cout << endl;
        }
    }
}
