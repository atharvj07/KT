import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] a = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int sum = 0;
                for (int k = j; k < n; k++) {
                    sum += a[i][k];
                    max = Math.max(max, sum);
                }
            }
        }
        for (int j = 0; j < n; j++) {
            for (int i = 0; i < n; i++) {
                int sum = 0;
                for (int k = i; k < n; k++) {
                    sum += a[k][j];
                    max = Math.max(max, sum);
                }
            }
        }
        System.out.println(max);
    }
}
