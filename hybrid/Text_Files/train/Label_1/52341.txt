import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] b = new int[2 * n];
        for (int i = 0; i < n; i++) {
            b[i] = 1;
            b[2 * n - 1 - i] = 2;
        }
        System.out.println(n * (n - 1) + 1);
        for (int i = 0; i < 2 * n; i++) {
            System.out.print(b[i] + " ");
        }
        System.out.println();
    }
}
