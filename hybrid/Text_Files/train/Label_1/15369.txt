#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int n, k, a[MAXN];
bool fl;
int main() {
  int t;
  cin >> t;
  while (t--) {
    fl = 1;
    cin >> n >> k;
    int sum = 0;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      sum += a[i];
      if (a[i] > k) { fl = 0; }
    }
    sort(a + 1, a + 1 + n);
    if (a[n] > k) fl = 0;
    if (sum % n > k) fl = 0;
    cout << (fl ? "yes" : "no") << endl;
  }
  return 0;
}
