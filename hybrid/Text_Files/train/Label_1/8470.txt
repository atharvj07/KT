#include <iostream>
#include <string>
using namespace std;

int main() {
  string s, a, b;
  getline(cin, s);
  int l = s.length(), i = 0;
  while (i < l) {
    if (s[i] == ',' || s[i] == ';') {
      if (a.size()) a += ',';
      if (b.size()) b += ',';
      i++;
      continue;
    }
    bool flag = false;
    string num;
    while (i < l && s[i] != ',' && s[i] != ';') {
      if (s[i] == '0' && num.empty()) {
        if (flag) break;
      } else if ('0' <= s[i] && s[i] <= '9') {
        flag = true;
        num += s[i];
      } else {
        break;
      }
      i++;
    }
    if (num.size())
      a.size() ? a += ',' + num : a += num;
    else
      b.size() ? b += ',' + s.substr(i - num.size(), num.size()) : b += s.substr(i - num.size(), num.size());
  }
  cout << (a.size() ? '"' + a + '"' : "-") << endl;
  cout << (b.size() ? '"' + b + '"' : "-") << endl;
}
