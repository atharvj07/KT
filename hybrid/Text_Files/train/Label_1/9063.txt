#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
vector<int> is_prime(MAXN, true);
vector<int> ans(MAXN);
int main() {
  int n;
  cin >> n;
  if (n == 3) {
    cout << "2 1";
  } else {
    is_prime[0] = is_prime[1] = false;
    for (int i = 2; i <= n; i++) {
      if (is_prime[i]) {
        for (int j = i * i; j <= n; j += i) {
          is_prime[j] = false;
        }
        for (int j = 2; i * j <= n; j++) {
          ans[i * j] = i;
        }
      }
    }
    for (int i = 2; i <= n; i++) {
      if (!ans[i]) {
        int cnt = 0;
        for (int j = 1; j <= n; j++) {
          if (i % j == 0 && is_prime[j]) {
            cnt++;
          }
          if (cnt > 1) break;
        }
        for (int j = 2; j * j <= i; j++) {
          if (i % j == 0) {
            ans[i] = ans[i / j];
            break;
          }
        }
        if (!ans[i]) ans[i] = i;
      }
    }
    for (int i = 2; i <= n; i++) {
      if (ans[i] != 0) cout << ans[i] << ' ';
    }
  }
  cout << '\n';
  return 0;
}
