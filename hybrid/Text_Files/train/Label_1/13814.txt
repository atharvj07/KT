#include<iostream>
#include<vector>
#include<string>
#include<sstream>
using namespace std;
int main(){
    string str;
    cin >> str;
    int n = str.size();
    for (int i = 1; i < n - 1; ++i){
        string a = str.substr(0, i);
        stringstream ss1(a);
        int aa; ss1 >> aa;
        for (int j = i + 1; j < n; ++j){
            string b = str.substr(i, j - i);
            stringstream ss2(b);
            int bb; ss2 >> bb;
            string c = str.substr(j);
            stringstream ss3(c);
            int cc; ss3 >> cc;
            if (aa + bb == cc){
                cout << a << '+' << b << '=' << c << '\n';
                return 0;
            }
        }
    }
}
