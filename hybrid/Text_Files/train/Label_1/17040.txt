#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    string s;
    cin >> s;
    vector<int> pre(26, -1);
    for (char c : s) {
        int i = c - 'a';
        if (pre[i] != -1) {
            cout << "NO" << endl;
            return 0;
        }
        pre[i] = 1;
    }
    cout << "YES" << endl;
}
