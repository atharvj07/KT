#include <bits/stdc++.h>

using namespace std;

int x;
const int N = 31;
int ans[N];

int main() {
  cin >> x;
  for (int i = 0; i < N; i++) {
    if ((x ^ (1 << i)) < x) {
      ans[i] = i;
      x ^= 1 << i;
    }
  }
  if (x != 0) {
    cout << -1 << '\n';
    return 0;
  }
  int cnt = 0;
  for (int i = 0; i < N; i++) {
    if (ans[i]) cnt++;
  }
  cout << cnt << '\n';
  for (int i = 0; i < N; i++) {
    if (ans[i]) cout << ans[i] << ' ';
  }
  return 0;
}
