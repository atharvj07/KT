#include <iostream>
#include <cstdio>

int main() {
  long long a[100005];
  int n, x, y;
  std::cin >> n >> x >> y;
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    if (a[i] % (x + y) <= x) {
      std::cout << "Vanya\n";
    } else if (a[i] % (x + y) == x + y) {
      std::cout << "Both\n";
    } else {
      std::cout << "Vova\n";
    }
  }
  return 0;
}
