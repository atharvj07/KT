import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0 || i == n - 1) {
                b[i] = a[i];
            } else {
                int[] c = {a[i - 1], a[i], a[i + 1]};
                java.util.Arrays.sort(c);
                b[i] = c[1];
            }
        }
        boolean stable = true;
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                stable = false;
                break;
            }
        }
        if (stable) {
            System.out.println("0");
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        } else {
            int cnt = 1;
            while (!stable) {
                a = b;
                b = new int[n];
                for (int i = 0; i < n; i++) {
                    if (i == 0 || i == n - 1) {
                        b[i] = a[i];
                    } else {
                        int[] c = {a[i - 1], a[i], a[i + 1]};
                        java.util.Arrays.sort(c);
                        b[i] = c[1];
                    }
                }
                stable = true;
                for (int i = 0; i < n; i++) {
                    if (a[i] != b[i]) {
                        stable = false;
                        break;
                    }
                }
                cnt++;
            }
            System.out.println(cnt);
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        }
    }
}
