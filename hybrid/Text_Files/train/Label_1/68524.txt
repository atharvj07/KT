a=input()
b=[0 for i in range(11)]
for i in range(len(a)):
    if a[i]=="1":b[i]+=1
c=int(input())
d=0
for i in range(10,-1,-1):
    if d<=c-i and b[i]>=c-d-i:
        for j in range(c-d-i):
            print(i,end=' ')
            d+=i
            b[i]-=1
        if d!=c:print(10);break
    elif d==c:
        break
    if d==c:
        print("YES")
        break
    elif d==c-1:
        for i in range(1,11):
            if b[i]>=1:
                print(i);b[i]-=1;d+=1
                break
    else:print("NO");break
