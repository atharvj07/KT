#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &ai : a) cin >> ai;
  int m;
  cin >> m;
  vector<int> q(m);
  for (int &qi : q) cin >> qi;

  for (int i = 0, j = 0; i < m; i++) {
    while (q[i] > a[j]) {
      q[i] -= a[j];
      j++;
    }
    cout << j + 1 << endl;
  }

  return 0;
}
