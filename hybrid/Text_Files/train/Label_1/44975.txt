import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int x = sc.nextInt();
            int a = x / 2;
            int b = x - a;
            System.out.println(a + " " + b);
        }
    }
}
