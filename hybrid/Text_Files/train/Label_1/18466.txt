#include <bits/stdc++.h>
using namespace std;
#define ll long long int
const ll mxN = 2e6 + 5;
ll n, m, ans, u, v, ara[mxN];
pair<ll, ll> cp[100005];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> m;
  for (ll i = 0; i < m; i++) {
    cin >> cp[i].first >> cp[i].second;
  }
  sort(cp, cp + m);
  for (ll i = 0; i < m; i++) {
    if (!ara[cp[i].first] && !ara[cp[i].first + 1]) {
      ara[cp[i].first] = ara[cp[i].first + 1] = 1;
      ans += cp[i].second;
    } else if (!ara[cp[i].first]) {
      ara[cp[i].first] = 1;
      ans += cp[i].second;
    } else if (!ara[cp[i].first + 1]) {
      ara[cp[i].first + 1] = 1;
      ans += cp[i].second;
    }
  }
  cout << ans << "\n";
}
