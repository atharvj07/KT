#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long sum = 0;
    int d = 0;
    for (int i = 0; i < n; ++i) {
        sum += a[i];
        if (sum > k) {
            break;
        }
        ++d;
    }
    cout << d << endl;
    return 0;
}
