#include <bits/stdc++.h>
using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        long long l, r, d;
        cin >> l >> r >> d;
        bool flag = false;
        for (long long i = l; i <= r; i += d) {
            if (i != l and i != r) {
                cout << i << endl;
                flag = true;
                break;
            }
        }
        if (!flag) {
            if ((l + d > r))
                cout << l + d << endl;
            else
                cout << r + d << endl;
        }
    }
    return 0;
}
