import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] ans = new int[n];
        int cnt = 1;
        int last = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] - last >= d) {
                ans[i] = cnt;
                last = a[i];
            } else {
                ans[i] = cnt;
            }

            if (last == m) {
                cnt++;
                last = 0;
            }
        }

        System.out.println(cnt);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
