import sys

input = sys.stdin.readline

n,k = map(int,input().split())
a = [[] for _ in range(n+1)]
for _ in range(n-1):
    x,y = map(int,input().split())
    a[x].append(y)
    a[y].append(x)

d = {}
nxt = [0]*(n+1)
col = [-1]*(n+1)
def dfs(now,p):
    ans = 0
    for to in a[now]:
        if to == p:
            continue
        if col[to] != -1:
            ans += col[to] == col[now]
            continue
        if not nxt[to]:
            nxt[to] = len(d)+1
            d[nxt[to]] = 1
        col[to] = col[now]
        ans +=dfs(to,now)
    return ans

ans = []
for i in range(1,n+1):
    if col[i] != -1:
        continue
    nxt[i] = len(d)+1
    d[nxt[i]] = 1
    col[i] = len(d)
    ans.append(dfs(i,i))

ans.sort(reverse=True)
print(len(ans))
for i in a:
    if len(i) < 3:
        continue
    i.sort(key=lambda x: nxt[x])
    tmp = []
    for j in i:
        if nxt[j] in d:
            tmp.append(nxt[j])
            del d[nxt[j]]
    tmp.sort()
    for j in tmp:
        print(j,end=" ")
    if len(tmp) < len(i):
        for j in range(1,len(d)+1):
            if j not in tmp:
                print(j,end=" ")
                break
        del d[j]
    print()
