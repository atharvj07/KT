import java.util.ArrayDeque;
import java.util.LinkedList;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int n = sc.nextInt();
        ArrayDeque<Integer> stack = new ArrayDeque<>();
        LinkedList<Integer> queue = new LinkedList<>();
        LinkedList<Integer> deck = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            if (x == 0) {
                int k = 0;
                while (!stack.isEmpty() && k < 3) {
                    System.out.print("popStack ");
                    stack.pop();
                    k++;
                }
                while (!queue.isEmpty() && k < 3) {
                    System.out.print("popQueue ");
                    queue.removeFirst();
                    k++;
                }
                while (!deck.isEmpty() && k < 3) {
                    System.out.print("popFront ");
                    deck.removeFirst();
                    k++;
                }
                System.out.println(k);
            } else {
                if (!stack.isEmpty() || i > 0) System.out.println("pushStack");
                stack.push(x);
            }
        }
    }
}
