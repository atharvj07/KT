#include <iostream>
using namespace std;

int main() {
    long long w, m, k;
    cin >> w >> m >> k;
    long long ans = 0;
    while (w >= 0) {
        long long tmp = m;
        int cnt = 0;
        while (tmp > 0) {
            tmp /= 10;
            cnt++;
        }
        if ((w - cnt * k) >= 0) {
            ans++;
            w -= cnt * k;
            m++;
        } else
            break;
    }
    cout << ans << endl;
    return 0;
}
