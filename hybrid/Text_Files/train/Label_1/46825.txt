import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] s = sc.next().toCharArray();
        int ans = n;
        char c = s[0];
        for (int i = 1; i < n; i++) {
            if (c != s[i]) {
                ans = ans - 2;
                c = s[i];
            }
        }
        System.out.println(Math.max(0, ans));
    }
}
