from collections import Counter
for _ in range(int(input())):
    s = input()
    m = int(input())
    b = list(map(int, input().split()))
    cnt = Counter(s)
    t = []
    for i in range(m):
        for ch in cnt:
            if b[i] == 0:
                break
            cnt[ch] -= 1
            t.append(ch)
            b[i] -= i * cnt[ch] + (m - i - 1) * cnt[ch]
    if any(b):
        print(-1)
    else:
        print(''.join(t))
