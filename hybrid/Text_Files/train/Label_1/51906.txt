import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        BigInteger[][] a = new BigInteger[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                a[i][j] = in.nextBigInteger();
            }
        }
        BigInteger ans = BigInteger.ZERO;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    BigInteger x1 = a[j][1].multiply(a[k][2]).subtract(a[j][2].multiply(a[k][1]));
                    BigInteger y1 = a[j][2].multiply(a[k][0]).subtract(a[j][0].multiply(a[k][2]));
                    BigInteger x2 = a[i][1].multiply(a[k][2]).subtract(a[i][2].multiply(a[k][1]));
                    BigInteger y2 = a[i][2].multiply(a[k][0]).subtract(a[i][0].multiply(a[k][2]));
                    BigInteger x3 = a[i][1].multiply(a[j][2]).subtract(a[i][2].multiply(a[j][1]));
                    BigInteger y3 = a[i][2].multiply(a[j][0]).subtract(a[i][0].multiply(a[j][2]));
                    BigInteger det = x1.multiply(y2).subtract(x2.multiply(y1)).add(x3.multiply(y1).subtract(x1.multiply(y3))).add(x2.multiply(y3).subtract(x3.multiply(y2)));
                    if (det.equals(BigInteger.ZERO)) ans = ans.add(BigInteger.ONE);
                }
            }
        }
        System.out.println(ans);
    }
}
