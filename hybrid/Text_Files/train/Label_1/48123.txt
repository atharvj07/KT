import java.util.Arrays;
import java.util.Scanner;

public class RomaGrind {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        int[] weaponAttack = new int[n];
        int[] weaponCost = new int[n];
        for (int i = 0; i < n; i++) {
            weaponAttack[i] = scanner.nextInt();
            weaponCost[i] = scanner.nextInt();
        }

        int[] armorDefense = new int[m];
        int[] armorCost = new int[m];
        for (int i = 0; i < m; i++) {
            armorDefense[i] = scanner.nextInt();
            armorCost[i] = scanner.nextInt();
        }

        Monster[] monsters = new Monster[p];
        for (int i = 0; i < p; i++) {
            monsters[i] = new Monster(scanner.nextInt(), scanner.nextInt(), scanner.nextInt());
        }

        Arrays.sort(monsters);

        int maxProfit = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int profit = 0;
                for (Monster monster : monsters) {
                    if (weaponAttack[i] > monster.defense && armorDefense[j] > monster.attack) {
                        profit += monster.coins;
                    }
                }
                profit -= weaponCost[i] + armorCost[j];
                maxProfit = Math.max(maxProfit, profit);
            }
        }

        System.out.println(maxProfit);
    }

    static class Monster implements Comparable<Monster> {
        int defense;
        int attack;
        int coins;

        Monster(int defense, int attack, int coins) {
            this.defense = defense;
            this.attack = attack;
            this.coins = coins;
        }

        @Override
        public int compareTo(Monster other) {
            return Integer.compare(other.coins, coins);
        }
    }
}
