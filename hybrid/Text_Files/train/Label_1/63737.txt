import sys

n, q = map(int, input().split())
a = list(map(int, input().split()))

count = 0
for i in range(n - 1):
    if a[i] != a[i + 1]:
        count += 1

print(count)

for i in range(q):
    p, x = map(int, input().split())
    p -= 1
    if a[p] == x:
        continue

    if p > 0 and a[p - 1] == a[p]:
        count -= 1
    if p < n - 1 and a[p + 1] == a[p]:
        count -= 1

    if x == a[p - 1] or x == a[p + 1]:
        count += 1

    a[p] = x
    print(count)
