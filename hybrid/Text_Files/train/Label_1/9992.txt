#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
using namespace std;
ifstream fi("input.txt");
typedef long long ll;
const ll inf = LLONG_MAX, mod = 1e9 + 7;
ll n, k, ans;
vector<ll> adj[1000001], child[1000001];
bool visited[1000001];
unordered_map<ll, bool> fin_ans;
unordered_map<ll, ll> inp;
void dfs(ll node, ll prev) {
  visited[node] = 1;
  child[node].pb(prev);
  for (ll x : adj[node]) {
    if (!visited[x]) { dfs(x, node); }
  }
}
string solve(ll node) {
  string ans = "0";
  for (ll x : adj[node]) {
    if (x == child[node][0]) { ans = solve(x); } else { ans += solve(x); }
  }
  if (inp.find(node) != inp.end()) {
    if (inp[node] == 1) {
      if (ans.length() == 1 && ans == "0") { ans = "10"; } else {
        if (ans[ans.length() - 1] == '0') { ans = ans.substr(0, ans.length() - 1) + "1"; }
      }
    }
  } else {
    if (ans.length() % 2 == 1) {
      swap(ans[0], ans[ans.length() - 1]);
    }
    for (ll i = 1; i < ans.length() - 1; i++) {
      if (ans[i] == '0' && ans[i + 1] == '1') {
        swap(ans[i], ans[i + 1]);
        i++;
      }
    }
    if (ans.length() % 2 == 1) {
      swap(ans[0], ans[ans.length() - 1]);
    }
  }
  return ans;
}
int main() {
  cin >> n;
  for (ll i = 1; i <= n; i++) {
    string s;
    cin >> s;
    if (s == "IN") {
      ll num;
      cin >> num;
      inp[i] = num;
    } else if (s == "NOT") {
      ll x;
      cin >> x;
      adj[i].pb(x);
    } else {
      ll x, y;
      cin >> x >> y;
      adj[i].pb(x);
      adj[i].pb(y);
    }
  }
  for (ll i = 1; i <= n; i++) {
    if (!visited[i]) { dfs(i, -1); }
  }
  for (ll i = 1; i <= n; i++) {
    if (inp.find(i) != inp.end()) {
      ll temp = inp[i];
      inp[i] = 1 - temp;
      string ans1 = solve(1);
      inp[i] = temp;
      fin_ans[i] = 1 - (ans1[0] - '0');
    }
  }
  for (ll i = 1; i <= n; i++) {
    if (inp.find(i) != inp.end()) { cout << fin_ans[i]; }
  }
}
