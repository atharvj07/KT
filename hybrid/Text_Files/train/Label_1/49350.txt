import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[4], b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        sc.close();
        int d1 = Math.max(a[0], a[1]), a1 = Math.max(b[0], b[1]), d2 = Math.max(a[2], a[3]), a2 = Math.max(b[2], b[3]);
        if (d1 > a2 && a1 > d2) System.out.println("Team 1");
        else if (d2 > a1 && a2 > d1) System.out.println("Team 2");
        else System.out.println("Draw");
    }
}
