#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int k;
    cin >> k;
    vector<int> ranks(k);
    for (int i = 0; i < k; i++) {
        cin >> ranks[i];
    }
    sort(ranks.begin(), ranks.end());
    int ans = 0;
    for (int i = 0; i < k; i++) {
        if (ranks[i] > 25 - i) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
