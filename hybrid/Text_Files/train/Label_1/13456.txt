#include<bits/stdc++.h>
using namespace std;
int n, a[55][55], cx, cy, c;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            cin >> a[i][j];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) {
            c = 0;
            for (int x = i; x <= n; x++)
                if (a[x][j] > 0)
                    c++;
                else
                    break;
            if (c) {
                cx = i - 1;
                break;
            }
        }
    for (int j = 1; j <= n; j++)
        for (int i = 1; i <= n; i++) {
            c = 0;
            for (int x = i; x <= n; x++)
                if (a[j][x] > 0)
                    c++;
                else
                    break;
            if (c) {
                cy = j - 1;
                break;
            }
        }
    //
    for (int i = cx + 1; i <= n; i++)
        for (int j = cy + 1; j <= n; j++)
            if (a[i][j] != 2) {
                puts("No");
                return 0;
            }
    puts("Yes");
}
