import java.util.*;

public class Main {
    static int[] p, d;
    static boolean[] v;
    static ArrayList<Integer>[] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int b = sc.nextInt();
        p = new int[m];
        d = new int[n + 1];
        v = new boolean[n + 1];
        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) p[i] = sc.nextInt();
        for (int i = 1; i < n; i++) {
            int a = sc.nextInt();
            int c = sc.nextInt();
            g[a].add(c);
            g[c].add(a);
        }
        v[0] = true;
        bfs(1, 0);
        int r = 0;
        for (int i : p) {
            if (d[i] == b) r++;
        }
        if (r == 0) {
            System.out.println(0);
        } else {
            System.out.println(r);
        }
    }

    static void bfs(int n, int c) {
        LinkedList<Integer> q = new LinkedList<>();
        q.add(n);
        while (!q.isEmpty()) {
            int a = q.poll();
            v[a] = true;
            for (int s : g[a]) {
                if (!v[s]) {
                    q.add(s);
                    d[s] = c + 1;
                }
            }
        }
    }
}
