#include <bits/stdc++.h>

using namespace std;
int amin[1005], bmin[1005];

int main() {
  int a, b, c1, c2, c3, c4;
  cin >> c1 >> c2 >> c3 >> c4;
  cin >> a >> b;
  for (int i = 1; i <= a; i++) {
    cin >> amin[i];
  }
  for (int i = 1; i <= b; i++) {
    cin >> bmin[i];
  }
  int ans = 1e9;
  for (int i = 1; i <= a; i++) {
    ans = min(ans, c1 * amin[i]);
  }
  for (int i = 1; i <= b; i++) {
    ans = min(ans, c1 * bmin[i]);
  }
  ans = min(ans, c2);
  ans = min(ans, c3);
  ans = min(ans, c4);
  cout << ans << endl;
}
