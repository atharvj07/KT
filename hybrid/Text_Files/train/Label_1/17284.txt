#include <bits/stdc++.h>
using namespace std;
string s,t,p;
int main(){
    cin>>s>>t;
    int c[2]={};
    for(int i=0;i<s.size();i++){
        c[s[i]-'0']++;
        c[t[i]-'0']--;
    }
    if(c[0]||c[1]){cout<<"impossible\n";return 0;}
    for(int i=0;i<s.size();i++) p+='(01)[s[i]==t[i]]';
    cout<<p<<endl;
}
