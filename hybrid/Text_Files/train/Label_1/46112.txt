import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            board[i] = scanner.nextLine().toCharArray();
        }
        boolean[][] visited = new boolean[n][n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !visited[i][j]) {
                    if (dfs(board, visited, i, j)) {
                        count++;
                    }
                }
            }
        }
        System.out.println(count == n * n / 5 ? "YES" : "NO");
    }

    private static boolean dfs(char[][] board, boolean[][] visited, int x, int y) {
        if (x < 0 || x >= board.length || y < 0 || y >= board[0].length || board[x][y] == '#' || visited[x][y]) {
            return false;
        }
        visited[x][y] = true;
        boolean result = dfs(board, visited, x - 1, y - 1) || dfs(board, visited, x - 1, y) || dfs(board, visited, x - 1, y + 1) || dfs(board, visited, x, y - 1) || dfs(board, visited, x, y + 1);
        if (result) {
            visited[x][y] = false;
        }
        return result;
    }
}
