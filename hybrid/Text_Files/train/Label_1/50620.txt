import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double A = sc.nextDouble();
        double B = sc.nextDouble();
        int H = sc.nextInt();
        int M = sc.nextInt();
        double hourAngle = H * 30 + M * 0.5;
        double minuteAngle = M * 6;
        double angleDiff = Math.abs(hourAngle - minuteAngle);
        if (angleDiff > 180) {
            angleDiff = 360 - angleDiff;
        }
        double distance = Math.sqrt(A * A + B * B - 2 * A * B * Math.cos(angleDiff * Math.PI / 180));
        System.out.println(String.format("%.12f", distance));
    }
}
