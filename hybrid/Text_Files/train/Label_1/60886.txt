import math

def solve():
    t = int(input())
    for _ in range(t):
        n, u, v = map(int, input().split())
        a = list(map(int, input().split()))
        row = [0] * (1000001)
        for i in a:
            row[i] = 1
        ans = 0
        need = 1
        for j in range(1, 1000001):
            if row[j] == 1:
                if need == 1:
                    ans += u
                    need = 0
                else:
                    ans += v
            else:
                need = 1
        print(ans)

solve()
