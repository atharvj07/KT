#include <iostream>
#include <vector>
#include <cstring>
using namespace std;
const int N = 3e5 + 5;
vector<int> g[N];
int bal[N], vis[N], lst[N], np;
void dfs(int u) {
  vis[u] = 1;
  int cur = bal[u];
  for (auto v : g[u]) {
    if (!vis[v]) dfs(v);
    if (bal[v] + cur == -1) np++;
    cur += bal[v];
  }
  lst[cur]++;
}
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    string s;
    cin >> s;
    for (char c : s) {
      if (c == '(')
        bal[i]++;
      else
        bal[i]--;
    }
    if (bal[i] != 0) continue;
    g[i].push_back(++np);
    g[np].push_back(i);
  }
  for (int i = 1; i <= np; i++)
    if (!vis[i]) dfs(i);
  long long ans = 0;
  for (int i = 1; i <= np; i++) ans += 1ll * lst[i] * (lst[i] + lst[i] - 1) / 2;
  cout << ans << endl;
  return 0;
}
