import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++) a[i] = sc.nextInt();
        int q = sc.nextInt();
        int[] b = new int[q];
        int[] c = new int[q];
        for(int i=0; i<q; i++) {
            b[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        long[] s = new long[q];
        for(int i=0; i<n; i++) {
            s[0] += a[i];
            for(int j=1; j<q; j++){
                if(a[i] == b[j]) a[i] = c[j];
                s[j] += a[i];
            }
        }
        for(int i=0; i<q; i++) System.out.println(s[i]);
    }
}
