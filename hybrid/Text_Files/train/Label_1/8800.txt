#include<bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 5;
int ans[maxn], n, k;
string s;

int main() {
    cin >> n >> k >> s;
    for (int i = 0; i < k; ++i) {
        int cnt_b = 0, cnt_w = 0;
        for (int j = 1; j <= n; ++j) {
            if (s[j] == 'B') ++cnt_b;
            else ++cnt_w;
            if (cnt_b > cnt_w) ans[j] = 1;
            else ans[j] = 0;
        }
        for (int j = 1; j <= n; ++j) s[j] = (ans[j] ? 'B' : 'W');
    }
    for (int j = 1; j <= n; ++j) cout << s[j];
    cout << endl;
    return 0;
}
