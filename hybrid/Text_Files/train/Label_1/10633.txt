#include <bits/stdc++.h>
#define ll long long
#define gcd __gcd
using namespace std;

ll binpow(ll b, ll e){
    if (e == 0)
        return 1LL;
    if (e == 1)
        return b;
    ll res = binpow(b, e/2);
    res = res * res;
    if (e%2 == 1)
        res = res * b;
    return res;
}
int main(){
    ll n, f2;
    cin >> n;
    vector<ll> a(n), p(n);
    bool all2 = true;
    for(ll i = 0; i < n; i++){
        cin >> a[i];
        p[i] = a[i];
        if (a[i] < 2)
            all2 = false;
    }
    if (all2){
        cout << a[0] << ' ';
        for(ll i = 1; i < n; i++)
            cout << p[i] << ' ';
        cout << '\n';
        return 0;
    }
    f2 = 2;
    for(ll i = 0; i < n; i++){
        if (a[i] == 1){
            p[i] = f2;
            f2 += 2;
        }
        else if (a[i] % 2 == 0){
            p[i] = a[i]+1;
        }
    }
    for(ll i = 0; i < n; i++){
        if (p[i]%2 == 0)
            p[i]++;
        if (i > 0 && p[i-1] == p[i]){
            bool found = false;
            for(ll j = 3; j * j <= p[i]; j += 2){
                if (p[i]%j == 0){
                    p[i] += 2;
                    found = true;
                    break;
                }
            }
            if (!found)
                p[i] += 2;
        }
    }
    for(ll i = 0; i < n; i++)
        cout << p[i] << ' ';
    cout << '\n';
    return 0;
}
