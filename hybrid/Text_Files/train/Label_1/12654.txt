#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, a, b;
        cin >> n >> m >> a >> b;
        if (a * n != b * m)
            cout << "NO" << endl;
        else {
            cout << "YES" << endl;
            vector<vector<int>> v(n, vector<int>(m, 0));
            int ra = 0, r = 0, ca = 0, c = 0;
            while (ra < a && r != n) {
                int f = 0;
                while (ca < b && c != m) {
                    if (v[r][c] == 0) {
                        v[r][c] = 1;
                        ra++;
                        ca++;
                        f = 1;
                        break;
                    }
                    c++;
                }
                if (f == 0)
                    r++;
                else
                    ca = 0;
            }
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) { cout << v[i][j]; }
                cout << endl;
            }
        }
    }
    return 0;
}
