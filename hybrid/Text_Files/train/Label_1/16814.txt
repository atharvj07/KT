#include <iostream>
#include <vector>
#include <queue>
using namespace std;
#define MAX 1000
#define N_DIR 4
#define BLANK '.'
#define OBSTACLE '*'
#define START 'X'
#define EMPTY -1
typedef pair<int, int> pii;

char maze[MAX][MAX];
int dist[MAX][MAX], vis[MAX][MAX][N_DIR*2];
deque<pair<pii, char>> q;

char comp(int x) {
    if (x % N_DIR == 0) return 'U';
    if (x % N_DIR == 1) return 'R';
    if (x % N_DIR == 2) return 'D';
    return 'L';
}

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> maze[i][j];
            for (int d = 0; d < N_DIR; d++) {
                dist[i][j] = EMPTY;
                vis[i][j][d * 2] = vis[i][j][d * 2 + 1] = EMPTY;
            }
        }
    }

    int x, y;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (maze[i][j] == START) {
                x = i, y = j;
                break;
            }
        }
    }

    vis[x][y][0 * 2] = 0;
    q.push_back(make_pair(make_pair(x, y), 0 * 2));
    while (!q.empty()) {
        int x = q.front().first.first, y = q.front().first.second, d = q.front().second;
        q.pop_front();

        for (int i = 0; i < N_DIR; i++) {
            int nx = x + "2101"[i] - '1', ny = y + "1012"[i] - '1', nd = d + (i - (d % N_DIR) + N_DIR) % N_DIR;
            if (nx < 0 || nx >= n || ny < 0 || ny >= m || maze[nx][ny] == OBSTACLE) continue;
            if (dist[nx][ny] == EMPTY || dist[nx][ny] > dist[x][y] + 1) {
                dist[nx][ny] = dist[x][y] + 1;
                if (vis[nx][ny][nd * 2] == EMPTY) {
                    vis[nx][ny][nd * 2] = vis[x][y][d * 2] + 1;
                    q.push_front(make_pair(make_pair(nx, ny), nd * 2));
                }
                if (vis[nx][ny][nd * 2 + 1] == EMPTY) {
                    vis[nx][ny][nd * 2 + 1] = vis[x][y][d * 2] + 1;
                    q.push_back(make_pair(make_pair(nx, ny), nd * 2 + 1));
                }
            }
        }
    }

    string ans = "";
    int nx = x, ny = y, nd = 0;
    if (dist[x][y] == EMPTY) {
        cout << "IMPOSSIBLE\n";
        return 0;
    }
    for (int i = 0; i < k; i++) {
        for (int j = 0; j < N_DIR; j++) {
            int dx = nx + "2101"[j] - '1', dy = ny + "1012"[j] - '1', dj = nd + (j - (nd % N_DIR) + N_DIR) % N_DIR;
            if (vis[dx][dy][dj * 2] == vis[nx][ny][nd * 2] - 1) {
                ans += comp(nd);
                nx = dx, ny = dy, nd = dj;
                break;
            }
            else if (vis[dx][dy][dj * 2 + 1] == vis[nx][ny][nd * 2] - 1) {
                ans += comp(nd);
                nx = dx, ny = dy, nd = dj;
                break;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
