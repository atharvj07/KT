n = int(input())
a, b = [], []
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
graph = {i: [] for i in range(1, 1001)}
for i in range(n):
    graph[b[i]].append(a[i])
vis = [False] * 1001
cnt = 0
for i in range(1, 1001):
    if not vis[i]:
        stack = [i]
        vis[i] = True
        can_open = False
        while stack:
            curr = stack.pop()
            can_open |= curr == i
            for j in graph[curr]:
                if not vis[j]:
                    vis[j] = True
                    stack.append(j)
        cnt += not can_open
print(cnt)
