#include <bits/stdc++.h>
using namespace std;

int main() {
  int k, n;
  cin >> k >> n;
  int a[k], b[n];
  for (int i = 0; i < k; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  map<int, bool> has;
  for (int i = 0; i < n; i++) has[b[i]] = true;
  int cnt = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (cnt == 0) {
      if (has[b[i] + a[0]]) {
        cnt++;
        ans += b[i];
      }
    } else {
      if (has[ans + a[cnt]]) {
        cnt++;
        ans += a[cnt];
      }
    }
  }
  if (ans == 0)
    cout << 0 << endl;
  else if (cnt == k)
    cout << 1 << endl;
  else
    cout << -1 << endl;
  return 0;
}
