#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll INF = 1e18;
int main() {
    ll n;
    cin >> n;
    vector<ll> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    vector<ll> dp(n, INF);
    dp[n - 1] = 0;
    multiset<ll> st = {h[n - 1]};
    for (ll i = n - 2; i >= 0; i--) {
        auto it = st.lower_bound(h[i]);
        ll j = -1;
        if (it != st.end()) j = *it;
        if (j != -1) {
            dp[i] = min(dp[i], dp[j]);
        } else {
            it = st.begin();
            dp[i] = min(dp[i], dp[*it] + 1);
        }
        st.insert(h[i]);
    }
    cout << dp[0] << "\n";
    return 0;
}
