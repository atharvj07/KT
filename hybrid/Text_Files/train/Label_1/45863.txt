import java.util.Scanner;

public class StrongPasswordGenerator {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String[] dangerousPasswords = new String[n];
        for (int i = 0; i < n; i++) {
            dangerousPasswords[i] = scanner.nextLine();
        }

        Trie trie = new Trie();
        for (String dangerousPassword : dangerousPasswords) {
            trie.insert(dangerousPassword);
        }

        String password = "";
        for (char c = 'a'; c <= 'z'; c++) {
            if (!trie.hasPrefix(password + c)) {
                password += c;
                break;
            }
        }

        System.out.println(password);
    }

    static class TrieNode {
        boolean isEnd;
        TrieNode[] children;

        TrieNode() {
            isEnd = false;
            children = new TrieNode[26];
        }
    }

    static class Trie {
        TrieNode root;

        Trie() {
            root = new TrieNode();
        }

        void insert(String word) {
            TrieNode node = root;
            for (char c : word.toCharArray()) {
                int index = c - 'a';
                if (node.children[index] == null) {
                    node.children[index] = new TrieNode();
                }
                node = node.children[index];
            }
            node.isEnd = true;
        }

        boolean hasPrefix(String prefix) {
            TrieNode node = root;
            for (char c : prefix.toCharArray()) {
                int index = c - 'a';
                if (node.children[index] == null) {
                    return false;
                }
                node = node.children[index];
            }
            return true;
        }
    }
}
