def find_company_name(s, t):
    s = sorted(s)
    t = sorted(t, reverse=True)
    
    n = len(s)
    company_name = [''] * n
    
    oleg_turn = True
    left, right = 0, n - 1
    
    for i in range(n):
        if oleg_turn:
            if s[left] < t[left]:
                company_name[i] = s.pop(left)
            else:
                company_name[right] = s.pop()
                right -= 1
        else:
            if t[left] > s[left]:
                company_name[i] = t.pop(left)
            else:
                company_name[right] = t.pop()
                right -= 1
        oleg_turn = not oleg_turn
        
    return ''.join(company_name)

s = input()
t = input()

print(find_company_name(s, t))
