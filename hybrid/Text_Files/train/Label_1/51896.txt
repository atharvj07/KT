import java.util.*;

public class PowerOfTwoPairs {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] cnt = new int[30];
        for (int x : a) {
            int i = 0;
            while (x > 0) {
                if ((x & 1) == 1) {
                    cnt[i]++;
                }
                x >>= 1;
                i++;
            }
        }

        int ans = 0;
        for (int i = 0; i < 30; i++) {
            if (cnt[i] % 2 == 0) {
                ans += cnt[i] / 2;
            } else {
                ans += cnt[i] / 2;
            }
        }

        System.out.println(ans);
    }
}
