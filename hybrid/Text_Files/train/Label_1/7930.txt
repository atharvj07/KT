#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int MAXN = 3e5 + 1;
int a[MAXN], par[MAXN], vis[MAXN];

int find(int x) {
    if (x == par[x]) return x;
    return par[x] = find(par[x]);
}

int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i], par[i] = i;
    for (int i = 1; i < n; i++) {
        int u, v; cin >> u >> v;
        int pu = find(u), pv = find(v);
        par[pu] = pv;
    }

    priority_queue<int> pq;
    for (int i = 1; i <= n; i++) {
        int pi = find(i);
        pq.push(a[i]);
        vis[pi] = 1;
    }

    int ans = 0;
    while (!pq.empty()) {
        int cur = pq.top(); pq.pop();
        if (cur > ans) ans = cur;
        for (int i = 1; i <= n; i++) {
            int pi = find(i);
            if (pi != find(cur) && vis[pi] && !a[i]) {
                pq.push(a[i] = cur + 1);
                vis[pi] = 0;
            }
        }
    }

    cout << ans << endl;
    return 0;
}
