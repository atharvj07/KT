#include <bits/stdc++.h>
using namespace std;
const int mxN = 3e5, lg = 18;
int n, sol[mxN], a[mxN], par[mxN][lg], mx[lg][mxN];
vector<int> ans;
int get_mx(int l, int r) {
    int log = 31 - __builtin_clz(r - l + 1);
    return max(mx[log][l], mx[log][r - (1 << log) + 1]);
}
int get_par(int l, int r) {
    int log = 31 - __builtin_clz(r - l + 1);
    return max(par[log][l], par[log][r - (1 << log) + 1]);
}
void upd(int l, int r) {
    int l1 = l, r1 = r, x = get_mx(l, r);
    while (l <= r) {
        if (a[l] % x == 0) { l1 = l; }
        if (a[r] % x == 0) { r1 = r; }
        l++;
        r--;
    }
    if (l1 <= r1) {
        ans.push_back(l1);
        sol[l1] = r1 - l1 + 1;
    }
}
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        mx[0][i] = a[i];
        par[0][i] = -1;
        if (i > 0) {
            par[0][i] = par[0][i - 1];
            if (a[i] > a[par[0][i]]) { par[0][i] = i; }
        }
    }
    for (int j = 1; (1 << j) <= n; j++) {
        for (int i = 0; i + (1 << j) - 1 < n; i++) {
            mx[j][i] = mx[j - 1][i];
            par[j][i] = par[j - 1][i];
            if (mx[j - 1][i + (1 << (j - 1))] > mx[j][i]) {
                mx[j][i] = mx[j - 1][i + (1 << (j - 1))];
                par[j][i] = par[j - 1][i + (1 << (j - 1))];
            }
        }
    }
    int mxv = 0;
    for (int l = 0; l < n; l++) {
        int r = l;
        while (r < n && a[r] % a[l] == 0) { r++; }
        if (r - l > mxv) {
            mxv = r - l;
            ans.clear();
            upd(l, r - 1);
        } else if (r - l == mxv) {
            upd(l, r - 1);
        }
        l = r;
    }
    cout << ans.size() << " " << mxv << "\n";
    for (auto x : ans) { cout << x << " "; }
    cout << "\n";
    return 0;
}
