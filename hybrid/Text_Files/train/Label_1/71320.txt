import sys

input = sys.stdin.readline

n, k = map(int, input().split())

seginf = []
for i in range(n):
    l, r = map(int, input().split())
    seginf.append((l, r, i + 1))

seginf.sort()

covcnt = [0] * (2 * 10 ** 5 + 1)

res = []
currmx = 0
for l, r, i in seginf:
    for j in range(l, r + 1):
        covcnt[j] += 1
    if covcnt[l] <= k:
        currmx = max(currmx, covcnt[l])
    else:
        res.append(i)
        for j in range(l, r + 1):
            covcnt[j] -= 1

print(len(res))
print(*res)
