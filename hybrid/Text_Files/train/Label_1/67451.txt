def solve(s, b):
    n = len(b)
    t = [''] * n  # Placeholder for final string
    s_count = sorted([(ch, s.count(ch)) for ch in set(s)], key=lambda x: -ord(x[0]))
    while n > 0:
        indices = [i for i in range(len(b)) if b[i] == 0 and t[i] == '']
        for ch, count in s_count:
            if count >= len(indices):
                for i in indices:
                    t[i] = ch
                    for j in range(len(b)):
                        b[j] -= abs(i - j)
                s_count = [(ch, count-len(indices))] + s_count
                s_count.remove((ch, count))
                break
        n -= len(indices)
    return ''.join(t)

q = int(input().strip())
for _ in range(q):
    s = input().strip()
    m = int(input().strip())
    b = list(map(int, input().strip().split()))
    print(solve(s, b))
