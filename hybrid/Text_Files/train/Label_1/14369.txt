#include <iostream>
#include <climits>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, k;
        std::cin >> n >> k;
        int ans = n;
        while (true) {
            int max = INT32_MIN, min = INT32_MAX, cnt = 0;
            int tmp = ans;
            while (tmp) {
                int d = tmp % 10;
                tmp /= 10;
                max = std::max(max, d);
                min = std::min(min, d);
                if (cnt < d) cnt = d;
            }
            if (cnt <= k && max - min <= k) break;
            ++ans;
        }
        std::cout << ans << std::endl;
    }
    return 0;
}
