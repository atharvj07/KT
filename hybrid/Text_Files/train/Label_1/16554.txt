#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = n;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < n && a[j] <= a[i] + k) j++;
    ans = min(ans, n - j + i);
  }
  cout << ans << endl;
  return 0;
}
