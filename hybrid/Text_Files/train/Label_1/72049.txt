import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = [int(x) for x in input().split()]
t = 0
while len(a) != 1:
    b = []
    for i in range(len(a) - 1):
        b.append(a[i] | a[i + 1])
    if t == 1:
        a = [int(x) for x in b]
    else:
        a = [int(x) for x in b]
    t ^= 1
for i in range(m):
    p, b = map(int, input().split())
    a[p - 1] = b
    t = 0
    while len(a) != 1:
        b = []
        for i in range(len(a) - 1):
            b.append(a[i] | a[i + 1])
        if t == 1:
            a = [int(x) for x in b]
        else:
            a = [int(x) for x in b]
        t ^= 1
    print(a[0])
