#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> ans;
  int m = 1;
  while (m * m < n) {
    m++;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i + j >= n - 1) {
        ans.push_back({i + 1, j + 1});
      }
    }
  }
  cout << m << endl;
  for (auto &p : ans) {
    cout << p.first << " " << p.second << endl;
  }
}
