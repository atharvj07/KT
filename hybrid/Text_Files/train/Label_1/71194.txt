r, c = map(int, input().split())
cake = [input() for _ in range(r)]

rows_with_strawberry = set()
cols_with_strawberry = set()

for i in range(r):
    for j in range(c):
        if cake[i][j] == 'S':
            rows_with_strawberry.add(i)
            cols_with_strawberry.add(j)

safe_rows = r - len(rows_with_strawberry)
safe_cols = c - len(cols_with_strawberry)

max_eaten = safe_rows * c + safe_cols * (r - safe_rows)
print(max_eaten)
