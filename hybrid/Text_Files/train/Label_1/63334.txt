import sys

t = int(sys.stdin.readline())

for _ in range(t):
    n, k = map(int, sys.stdin.readline().split())

    a = list(range(1, k+1)) + list(range(k-1, 0, -1))
    p = [a[i-1] for i in a]

    if k != 1:
        m = max(p)
        for i in range(k):
            if p[i] != m:
                p[i], p[k-1] = p[k-1], p[i]
                break

    print(*p)
