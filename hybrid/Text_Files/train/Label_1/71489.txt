import sys

input = sys.stdin.readline

test_cases = int(input())

for _ in range(test_cases):
    n, s = map(int, input().split())
    arr = []

    for _ in range(n):
        l, r = map(int, input().split())
        arr.append((l, r))

    arr.sort(key=lambda x: x[0])

    prefix = [0] * n

    for i in range(n):
        if i == 0:
            prefix[i] = arr[i][0]
        else:
            prefix[i] = arr[i][0] + prefix[i - 1]

    ans = 0

    for i in range(n):
        low, high = arr[i][0], arr[i][1]

        l = i
        r = n - 1

        idx = -1

        while l <= r:
            mid = (l + r) // 2

            curr = 0

            if i == 0:
                curr = prefix[mid]
            else:
                curr = prefix[mid] - prefix[i - 1]

            if curr <= s:
                idx = mid
                l = mid + 1
            else:
                r = mid - 1

        if idx == -1:
            ans = max(ans, low)
        else:
            salary = (s - prefix[idx] + arr[i][0]) // (mid - i + 1)
            ans = max(ans, salary)

    print(ans)
