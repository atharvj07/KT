import java.util.*;
import java.io.*;
import java.math.*;

class Prob1 {

  static int[] solve(String s, int l, int r) {
    int ans[] = new int[10];
    int n = r - l + 1;
    int dp = 1;
    int cp = l - 1;
    for (int i = 0; i < n; i++) {
      cp += dp;
      if (cp < 0 || cp >= r)
        break;
      int val = s.charAt(cp) - '0';
      if (val >= 0 && val <= 9) {
        ans[val]++;
        if (val == 0) {
          s = s.substring(0, cp) + s.substring(cp + 1);
          n--;
          i--;
          cp--;
        } else {
          s = s.substring(0, cp) + (val - 1) + s.substring(cp + 1);
        }
      } else {
        dp = dp == 1 ? -1 : 1;
        if (s.charAt(cp) == '>') {
          cp++;
          if (cp < r && s.charAt(cp) == '>') {
            s = s.substring(0, cp - 1) + s.substring(cp);
            n--;
            i--;
          }
        }
      }
    }
    return ans;
  }

  public static void main(String[] args) {
    try {
      BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

      String tokens[] = br.readLine().split("\\s+");
      int n = Integer.parseInt(tokens[0]);
      int q = Integer.parseInt(tokens[1]);
      String s = br.readLine();

      for (int i = 0; i < q; i++) {
        tokens = br.readLine().split("\\s+");
        int l = Integer.parseInt(tokens[0]);
        int r = Integer.parseInt(tokens[1]);
        int[] ans = solve(s.substring(l - 1, r), l - 1, r - 1);
        for (int j = 0; j < ans.length; j++) {
          System.out.print(ans[j] + " ");
        }
        System.out.println();
      }

      br.close();
    } catch (Exception e) {
      e.printStackTrace();
    }
  }
}
