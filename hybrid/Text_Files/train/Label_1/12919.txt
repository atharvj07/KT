#include <iostream>
#include <vector>
using namespace std;
int main() {
    int s,x1,x2,t1,t2,p,d;
    cin>>s>>x1>>x2>>t1>>t2>>p>>d;
    int ans=1e9;
    ans=min(ans,abs(x2-x1)*t2+t1);  // never get the tram
    if(x1<x2){
        if(d==1){
            ans=min(ans,(s-p)*t1+abs(x2-s)*t2); // get on at the tram's position and get off at x2
            ans=min(ans,abs(x2-x1)*t2+t1+abs(s-x2)*t1); // get on at x1 and get off at the tram's position
        }
        else{
            ans=min(ans,p*t1+abs(x2-0)*t2); // get on at the tram's position and get off at x2
            ans=min(ans,abs(x2-x1)*t2+t1+abs(0-x2)*t1); // get on at x1 and get off at the tram's position
        }
    }
    else{
        if(d==1){
            ans=min(ans,abs(x1-p)*t1+abs(s-x1)*t2); // get on at the tram's position and get off at x1
            ans=min(ans,abs(x1-x2)*t2+t1+abs(s-x1)*t1); // get on at x2 and get off at the tram's position
        }
        else{
            ans=min(ans,abs(x1-0)*t2+p*t1); // get on at the tram's position and get off at x1
            ans=min(ans,abs(x1-x2)*t2+t1+abs(0-x1)*t1); // get on at x2 and get off at the tram's position
        }
    }
    cout<<ans<<endl;
    return 0;
}
