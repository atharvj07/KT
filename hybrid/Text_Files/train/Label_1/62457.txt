import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    x = list(map(int, input().split()))
    s, ans = 0, []
    for i in range(n):
        s += a[i]
        for j in range(m):
            if s >= x[j]:
                ans.append(i + 1)
                break
        if s >= x[j]:
            break
    print(*ans + [-1] * (m - len(ans)))
