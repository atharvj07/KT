import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int maxb = sc.nextInt();
        int t = sc.nextInt();
        int[] a = new int[n * t];
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                int b = sc.nextInt();
                for (int l = 0; l < t; l++) {
                    a[i * n * t + j * t + l] = b % maxb;
                    b = b / maxb;
                }
            }
        }
        int[] dp = new int[k];
        for (int i = 0; i < k; i++) {
            dp[i] = 1;
        }
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                boolean ok = true;
                for (int l = 0; l < n * t; l++) {
                    if (a[i * n * t + l] > a[j * n * t + l]) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    dp[j] = Math.max(dp[j], dp[i] + 1);
                }
            }
        }
        for (int i = 0; i < k; i++) {
            System.out.println(dp[i]);
        }
    }
}
