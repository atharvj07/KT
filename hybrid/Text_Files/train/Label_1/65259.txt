for x in[int(input())]:
    ans=[0 for i in range(21)]
    i=1
    while i<=40 and x>0:
        if x%2==0:
            ans[i]+=1
            x-=1
        else:
            now=0
            for j in range(29,-1,-1):
                if x>>j&1:
                    now=j
                    break
            x-=1<<now
            ans[i]+=now
        i+=2
    print(i//2-1)
    for j in range(1,i//2):
        if ans[j]:
            print(ans[j],end=" ")
