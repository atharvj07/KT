#include <bits/stdc++.h>
#define int ll

using namespace std;

//==============================================================================================================================

typedef long long ll;
typedef unsigned long long ull;
typedef long double ld;
#define PB emplace_back
#define EB emplace
#define PF emplace_front
#define INF 1e18
#define nl "\n"
const int MOD = 998244353;
const ld pi = acos(-1);

//==============================================================================================================================

vector<int> prefix_sum(100000);
void func(string s, vector<ll> a) {
	s += "0";
	a.PB(0);
	int n = s.size();
	prefix_sum = vector<int>(n, 0);
	for (int i = 1; i < n; i++) prefix_sum[i] = prefix_sum[i - 1] + a[i - 1];
	vector<int> dp(n, MOD);
	dp[n - 1] = 0;
	for (int i = n - 2; i >= 0; i--) {
		if (s[i] == 'a' || s[i] == 'r' || s[i] == 'd') dp[i] = min(dp[i + 1], prefix_sum[i]);
		else if (s[i - 1] == 'h' && s[i] == 'h') dp[i] = min(dp[i + 1], dp[i + 2] + a[i]);
		else dp[i] = min(dp[i + 1], prefix_sum[i]);
	}
	cout << dp[0] << nl;
}
//==============================================================================================================================
int32_t main() {
	ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
	int n, x;
	cin >> n;
	string s;
	cin >> s;
	vector<ll> a(n);
	for (int i = 0; i < n; i++) { cin >> x; a[i] = x; }
	func(s, a);
	return 0;
}
