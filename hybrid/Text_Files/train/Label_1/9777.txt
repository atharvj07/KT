#include <algorithm>
#include <iostream>
#include <vector>
#include <set>
using namespace std;

constexpr long long inf = 1e18;
constexpr int N = 201;
long long a[N], dp[N][N][N];
int main() {
  int n, k, x;
  cin >> n >> k >> x;
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  for (int j = 1; j <= x; ++j) {
    for (int seg_len = 1; seg_len + j - 1 <= n; ++seg_len) {
      for (int l = 1; l + seg_len - 1 <= n; ++l) {
        int r = l + seg_len - 1;
        dp[l][r][j] = max(dp[l + 1][r][j], dp[l][r - 1][j]);
        if (r - l + 1 >= k && dp[l][r][j] == -inf) dp[l][r][j] = 0;
        dp[l][r][j] = max(dp[l][r][j], dp[l][r][j - 1] + a[r]);
      }
    }
  }
  cout << (dp[1][n][x] < 0 ? -1 : dp[1][n][x]) << endl;
}
