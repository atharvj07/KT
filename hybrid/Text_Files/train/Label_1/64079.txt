import sys
sys.setrecursionlimit(500000)

n = int(input())
a = list(map(int, input().split()))
b = [0]*n
ans = []
for i in range(n):
    b[a[i]-1] = i
ma = -1
idx = -1
for i in range(1, n):
    if ma<b[i]:
        ma = b[i]
        idx = i
for i in range(1, n):
    if b[i]<b[i-1]:
        ans.append([b[i]+1, b[i-1]+1])
        tmp = b[i]
        b[i] = b[i-1]
        b[i-1] = tmp
        if i<=idx:
            idx+=1
if idx==0:
    if ans:
        print(len(ans))
        for el in ans:
            print(*el)
    else:
        print(0)
else:
    for i in range(n):
        if i==idx:
            continue
        ans.append([idx+1, i+1])
        tmp = b[idx]
        b[idx] = b[i]
        b[i] = tmp
    print(len(ans))
    for el in ans:
        print(*el)
