#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  rep(i, n) { cin >> a[i]; }
  int sum = 0;
  vector<int> b;
  for (int i = 0; i < n; i++) {
    if (sum == 0) b.push_back(i);
    sum += a[i];
    if (sum == 0) b.push_back(i + 1);
  }
  if (sum != 0)
    cout << -1 << endl;
  else {
    cout << b.size() / 2 << endl;
    rep(i, b.size() / 2) { cout << b[2 * i + 1] - b[2 * i] << " "; }
    cout << endl;
  }
}
