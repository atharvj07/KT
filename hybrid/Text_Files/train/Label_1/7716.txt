#include <bits/stdc++.h>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<int64_t> a(n), minDist(m);
    std::unordered_map<int64_t, std::vector<int64_t>> idx;
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
        idx[a[i]].push_back(i);
    }

    for (int i = 0; i < m; i++) {
        int l, r;
        std::cin >> l >> r;
        l--, r--;
        minDist[i] = LLONG_MAX;

        for (auto &v : idx) {
            std::vector<int64_t> &vec = v.second;
            int indL = std::lower_bound(vec.begin(), vec.end(), l) - vec.begin();
            int indR = std::upper_bound(vec.begin(), vec.end(), r) - vec.begin() - 1;

            if (indL <= indR) minDist[i] = std::min(minDist[i], vec[indR] - vec[indL]);
        }
        if (minDist[i] == LLONG_MAX) minDist[i] = -1;
    }
    for (int i = 0; i < m; i++) std::cout << minDist[i] << '\n';
}
