import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Edge {
        int from;
        int to;
        int weight;

        public Edge(int from, int to, int weight) {
            this.from = from;
            this.to = to;
            this.weight = weight;
        }
    }

    static int n, m;
    static List<Edge> edges;
    static int[] dist;
    static int[][] dp;
    static int INF = Integer.MAX_VALUE / 2;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int c = sc.nextInt();
            edges.add(new Edge(a, b, c));
        }
        dist = new int[n];
        dp = new int[n][1 << n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < (1 << n); j++) {
                dp[i][j] = INF;
            }
        }
        dijkstra(0);
        int ans = 0;
        for (Edge edge : edges) {
            if (dist[edge.from] + edge.weight + dist[edge.to] == dist[n - 1]) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    static void dijkstra(int s) {
        dp[s][1 << s] = 0;
        for (int mask = 1 << s; mask < (1 << n); mask++) {
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) == 0) {
                    continue;
                }
                for (Edge edge : edges) {
                    int nextMask = mask | (1 << edge.to);
                    dp[edge.to][nextMask] = Math.min(dp[edge.to][nextMask], dp[i][mask] + edge.weight);
                }
            }
        }
        for (int i = 0; i < n; i++) {
            dist[i] = dp[i][(1 << n) - 1];
        }
    }
}
