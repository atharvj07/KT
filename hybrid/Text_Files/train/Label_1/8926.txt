#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n, a[maxn], dp[maxn][2];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        unordered_map<int, int> mp;
        for (int i = 1; i <= n; ++i) {
            cin >> a[i];
            mp[a[i]] = 1;
            dp[i][0] = dp[i - 1][0], dp[i][1] = dp[i - 1][1];
            if (i == 1) { dp[i][0] = 1, dp[i][1] = 1; continue; }
            for (auto &j : mp) {
                if (i == 1) { dp[i][0] = 1, dp[i][1] = 1; break; }
                if (a[i] == j.first) {
                    if (a[i - 1] % a[i] == 0 or a[i] % a[i - 1] == 0) break;
                }
                if (a[i] % j.first == 0 or j.first % a[i] == 0) { dp[i][1] = 0; break; }
            }
            dp[i][0] += dp[i][1];
        }
        cout << dp[n][0] - 1 << endl;
        mp.clear();
    }
    return 0;
}
