import sys
input = sys.stdin.readline
def solve():
    n=input();
    if(n[len(n)-1]=="1" or n[len(n)-1]=="3" or n[len(n)-1]=="5" or n[len(n)-1]=="7" or n[len(n)-1]=="9"):
        return -1;
    elif(n[len(n)-2]=="1" or n[len(n)-2]=="3" or n[len(n)-2]=="5" or n[len(n)-2]=="7" or n[len(n)-2]=="9"):
        return 1;
    else:
        return 0;
for i in range(int(input())):
    print(solve());
