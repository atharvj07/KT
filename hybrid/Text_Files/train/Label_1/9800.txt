#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 1, inf = 1e9;
int n, a[maxn], ans;
char c[maxn];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> c[i];
    }
    for (int i = 1; i < n; i++) {
        if (c[i] == c[i + 1]) {
            ans += a[i + 1] - a[i];
            i++;
        }
    }
    cout << ans << endl;
    return 0;
}
