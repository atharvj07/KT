#include <cstdio>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
	int t;
	scanf("%d", &t);
	while(t--) {
		int n, k;
		scanf("%d %d", &n, &k);
		vector<int> a(n);
		for (int i = 0; i < n; i++) scanf("%d", &a[i]);
		int ans = 0;
		for (int i = 0; i < n; i++) {
			ans += (k - a[i] % k) % k;
		}
		printf("%d\n", ans);
	}
	return 0;
}
