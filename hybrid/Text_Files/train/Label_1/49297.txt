import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int candles = input.nextInt();
        int reuse = input.nextInt();
        int hours = candles;
        while (candles >= reuse) {
            hours += candles / reuse;
            candles = candles % reuse + candles / reuse;
        }
        System.out.println(hours);
    }
}
