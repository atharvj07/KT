#include <iostream>
#include <string>
#include <map>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  map<char, char> mp;
  for (int i = 0; i < m; i++) {
    char a, b;
    cin >> a >> b;
    mp[a] = b;
    mp[b] = a;
  }
  for (int i = 0; i < n; i++) {
    if (mp.count(s[i])) s[i] = mp[s[i]];
  }
  cout << s << endl;
  return 0;
}
