from collections import deque
def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        if n == 1:
            print(1)
            continue
        per = []
        q = deque([i for i in range(1, n+1)])
        for i in range(n):
            if len(q) == 1:
                per.append(q.popleft())
            elif i&1:
                per.append(q.pop())
            else:
                per.append(q.popleft())
        print(*per)
solve()
