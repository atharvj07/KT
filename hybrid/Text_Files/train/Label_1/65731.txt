t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    limits = []
    for _ in range(m):
        x, y, s = map(int, input().split())
        limits.append(s)
    limits.sort()
    lower = limits[n-2] if n-2 < len(limits) else limits[-1]
    upper = limits[n-1] if n-1 < len(limits) else limits[-1]
    if k < lower:
        print(lower - k)
    elif k > upper:
        print(k - upper)
    else:
        print(0)
