import sys

input = sys.stdin.readline
n, m, p = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Calculate the product of the two polynomials using fast Fourier transform
c = [0] * (n + m - 1)
for i in range(n):
    for j in range(m):
        c[i + j] += a[i] * b[j]

# Find the first coefficient of the product that is not divisible by p
t = -1
for i in range(n + m - 1):
    if c[i] % p != 0:
        t = i
        break

print(t)
