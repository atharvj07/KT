import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    c = [0] * 3
    for x in a:
        c[x%3] += 1
    ans = 0
    for i in range(3):
        if c[i] < c[(i+1)%3]:
            ans += c[(i+1)%3] - c[i]
            c[i] = c[(i+1)%3]
    print(ans)
