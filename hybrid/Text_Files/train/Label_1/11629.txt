#include <bits/stdc++.h>
using namespace std;
#define nl "\n"
const int M = 1e5 + 5, MOD = 1e9 + 7;
typedef long long ll;

int main() {
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;
    vector<ll> pref(n + 1);
    for (int i = 0; i < n; i++)
        pref[i + 1] = pref[i] + s[i] - 'a' + 1;
    while (q--) {
        int l, r;
        cin >> l >> r;
        cout << pref[r] - pref[l - 1] << nl;
    }
    return 0;
}
