#include <iostream>
#include <vector>
#include <queue>
using namespace std;
int n, m;
vector<vector<int>> g;
vector<int> color;
bool dfs(int v, int c, vector<bool>& used) {
	used[v] = true;
	color[v] = c;
	for (const auto& u : g[v])
		if (color[u] == c || (!used[u] && !dfs(u, c, used)))
			return false;
	return true;
}
int main() {
	cin >> n >> m;
	g.resize(n + 1);
	color.resize(n + 1);
	for (int i = 0; i < m; i++) {
		int a, b; cin >> a >> b;
		g[a].push_back(b);
	}
	vector<bool> used(n + 1);
	int ans = 0;
	queue<int> q;
	for (int i = 1; i <= n; i++)
		if (!used[i])
			q.push(i);
	while (!q.empty()) {
		ans++;
		vector<bool> u(n + 1);
		while (!q.empty()) {
			int cur = q.front();
			q.pop();
			if (!dfs(cur, ans, u)) {
				ans++;
				u.assign(n + 1, false);
				dfs(cur, ans, u);
			}
		}
	}
	cout << ans << endl;
	for (int i = 0; i < m; i++)
		cout << color[i + 1] << ' ';
	cout << endl;
}
