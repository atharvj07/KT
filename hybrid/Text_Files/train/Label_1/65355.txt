q = (1 + 5 ** 0.5) / 2
a = input()
b = input()
p = len(a) - len(b)
if p > 0:
    b = '0' * p + b
elif p < 0:
    a = '0' * -p + a
sa = 0
sb = 0
for ai, bi in zip(a, b):
    sa += int(ai) * q ** (len(a) - sa - 1)
    sb += int(bi) * q ** (len(b) - sb - 1)
if sa > sb:
    print('>')
elif sa < sb:
    print('<')
else:
    print('=')
