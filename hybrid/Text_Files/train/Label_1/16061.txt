#include<bits/stdc++.h>
using namespace std;
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector <int> v(n);
    int sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        sum += v[i];
    }
    sort(v.begin(), v.end());
    int ans = sum - v[(n / k) - 1];
    cout << ans << endl;
    return 0;
}
