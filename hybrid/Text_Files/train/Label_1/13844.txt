#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  int qa = 0, qaq = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'Q') qa++;
    else if (s[i] == 'A') qaq += qa;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'Q') ans += qaq;
    else if (s[i] == 'A') qaq--;
  }
  cout << ans << endl;
  return 0;
}
