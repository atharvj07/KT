#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    long long x, y;
    cin >> x >> y;
    if (x * y > 0) {
        cout << 0 << ' ' << 2 * abs(y) << ' ' << 2 * abs(x) << ' ' << 0 << endl;
    } else {
        cout << -2 * abs(x) << ' ' << 0 << ' ' << 0 << ' ' << 2 * abs(y) << endl;
    }
    return 0;
}
