import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int m = input.nextInt();
    int[] p = new int[n];
    for (int i = 0; i < n; i++) {
      p[i] = input.nextInt();
    }
    int[] c = new int[n];
    for (int i = 0; i < n; i++) {
      c[i] = input.nextInt();
    }
    int d = input.nextInt();
    int[] k = new int[d];
    for (int i = 0; i < d; i++) {
      k[i] = input.nextInt();
    }
    int[] removed = new int[n + 1];
    int[] mex = new int[d];
    int[] clubSize = new int[m + 1];
    for (int i = 0; i < d; i++) {
      clubSize[c[k[i] - 1]]--;
      removed[k[i]] = 1;
    }
    for (int i = 0; i < n; i++) {
      if (removed[i] == 0) {
        clubSize[c[i]]++;
      }
    }
    for (int i = 0; i < d; i++) {
      Set<Integer> set = new HashSet<>();
      for (int j = 1; j <= m; j++) {
        if (clubSize[j] > 0) {
          set.add(p[k[i] - 1]);
        }
      }
      int ans = 0;
      while (set.contains(ans)) {
        ans++;
      }
      mex[i] = ans;
    }
    for (int i = 0; i < d; i++) {
      System.out.println(mex[i]);
    }
  }
}
