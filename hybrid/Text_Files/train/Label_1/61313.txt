n, k = map(int, input().split())
c = list(map(int, input().split()))
a = [0] * n
b = [0] * n
for j in range(n):
    cur = k
    for i in range(n * k):
        if c[i] == j + 1:
            if cur > 0:
                if a[j] == 0:
                    a[j] = i + 1
                b[j] = i + 1
                cur -= 1
            else:
                b[j] = i
for i in range(n):
    print(a[i], b[i])
