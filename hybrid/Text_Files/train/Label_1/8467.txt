#include <iostream>
#include <string>
using namespace std;
int main() {
  string s, a, b;
  getline(cin, s);
  int n = s.size();
  for (int i = 0, j; i < n; i = j + 1) {
    for (j = i + 1; j < n; j++)
      if (s[j] == ',' || s[j] == ';') break;
    bool ok = 1;
    for (int k = i; k < j; k++)
      if (!isdigit(s[k])) {
        ok = 0;
        break;
      }
    if (!ok || (i > 0 && s[i - 1] == '0' && j - i > 1))
      b += s.substr(i, j - i), b += ',';
    else
      a += s.substr(i, j - i), a += ',';
  }
  if (a.size()) a.pop_back();
  if (b.size()) b.pop_back();
  cout << (a.size() ? '"' + a + '"' : "-") << endl;
  cout << (b.size() ? '"' + b + '"' : "-") << endl;
  return 0;
}
