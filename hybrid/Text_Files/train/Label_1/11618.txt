#include<bits/stdc++.h>
using namespace std;
string s;
int n;
int main(){
    int t; cin>>t;
    while(t--){
        cin>>s;
        n=s.size();
        for(int i=1;i<n;i++){
            if(s[i]==s[i-1]){
                if(s[i]!='?'){
                    cout<<-1<<endl;
                    goto out;
                }
            }
        }
        for(int i=0;i<n;i++){
            if(s[i]=='?'){
                if(i==0) s[i]=s[i+1]=='b'?'c':'b';
                else if(i==n-1) s[i]=s[i-1]=='b'?'c':'b';
                else{
                    if(s[i-1]==s[i+1]) s[i]=s[i-1]=='b'?'c':'b';
                    else s[i]=s[i-1];
                }
            }
        }
        cout<<s<<endl;
        out:;
    }
}
