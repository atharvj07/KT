from sys import stdin as input

n, m = map(int, input.readline().strip().split())
grid = []
for _ in range(n): grid.append(input.readline().strip())
ops = []
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.': continue
        ok = True
        for x in range(-1, 2):
            for y in range(-1, 2):
                if i+x >= n or j+y >= m or i+x < 0 or j+y < 0: continue
                if x == y == 0: ok = True
                else: ok = ok and grid[i+x][j+y] == '*'
        if ok: ops.append((i+1, j+1, 1))
        else: ops.append((i+1, j+1, 2))
        grid[i][j] = '.'
        for x in range(-1, 2):
            for y in range(-1, 2):
                if i+x >= n or j+y >= m or i+x < 0 or j+y < 0: continue
                if x == 0 and y == 0: continue
                grid[i+x][j+y] = '.'
print(len(ops))
print(*ops, sep='\n')
