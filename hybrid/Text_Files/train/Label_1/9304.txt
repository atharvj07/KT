#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2005;

int n, k;
int a[MAXN];
vector<array<int, 2>> b;
int f[MAXN][MAXN];

int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    b.push_back({a[i], i});
  }
  sort(b.rbegin(), b.rend());
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 0; j <= k; j++) {
      f[i][j] = f[i + 1][j];
      if (j > 0) f[i][j] = max(f[i][j], f[i + 1][j - 1] + b[i][0]);
    }
  }
  vector<int> ans;
  int i = 0, j = k;
  while (j > 0) {
    if (f[i][j] == f[i + 1][j]) {
      i++;
      continue;
    }
    ans.push_back(b[i][1] - i + 1);
    j--;
    i = b[i][1] + 1;
  }
  cout << f[0][k] << endl;
  for (int x : ans) cout << x << " ";
  cout << endl;
  return 0;
}
