import sys

mod = 1000000007

def fastpow(x, p):
    if p == 0:
        return 1
    if p == 1:
        return x

    val = fastpow(x, int(p / 2)) % mod
    val = (val * val) % mod
    if p % 2 != 0:
        return (val * x) % mod
    return val % mod

for line in sys.stdin:
    t, k = map(int, line.strip().split())
    for _ in range(t):
        a, b = map(int, input().strip().split())
        print((fastpow(2, b - a + 1) - 1 + mod) % mod)
