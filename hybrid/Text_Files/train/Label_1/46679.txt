import java.util.Scanner;

public class AlexeyAndSegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int[][] segments = new int[n][2];
        for (int i = 0; i < n; i++) {
            segments[i][0] = scanner.nextInt();
            segments[i][1] = scanner.nextInt();
        }

        int left = 0, right = 100, totalLength = 0;
        for (int i = 1; i < n; i++) {
            left = Math.max(left, segments[i][1]);
            right = Math.min(right, segments[i][0]);
        }

        totalLength = Math.max(0, right - left);
        System.out.println(totalLength);
    }
}
