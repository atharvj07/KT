n, p, k = map(int, input().split())
if p - k <= 1:
    start = 1
else:
    start = p - k
if p + k >= n:
    end = n
else:
    end = p + k
if start > 1:
    print("<<", end=" ")
for i in range(start, p):
    print(i, end=" ")
print("({})".format(p), end=" ")
for i in range(p + 1, end + 1):
    print(i, end=" ")
if end < n:
    print(">>", end=" ")
