#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<int,int> pi;

int a[100005];

int gcd(int a,int b){
    if(b==0) return a;
    return gcd(b,a%b);
}

int main() {
    int n;
    cin >> n;
    for(int i=0; i<n; i++) cin >> a[i];
    sort(a,a+n);
    int gc=a[0];
    for(int i=0; i<n; i++){
        for(int j=i+1; j<n; j++){
            gc=gcd(gc,a[i]*a[j]/__gcd(a[i],a[j]));
            if(gc==1) break;
        }
        if(gc==1) break;
    }
    cout << gc << endl;
    return 0;
}
