#include <iostream>
#include <string>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::string s;
  std::cin >> s;
  int cnt0 = 0, cnt1 = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') cnt0++;
    else cnt1++;
  }
  if (k == 1) {
    if (cnt0 > cnt1) std::cout << "tokitsukaze" << std::endl;
    else if (cnt0 < cnt1) std::cout << "quailty" << std::endl;
    else std::cout << "once again" << std::endl;
  } else if (cnt0 % k == 0 || cnt1 % k == 0) std::cout << "tokitsukaze" << std::endl;
  else if (cnt0 % k == 1 && cnt1 % k == 1) std::cout << "once again" << std::endl;
  else std::cout << "quailty" << std::endl;
  return 0;
}
