#include <iostream>
#include <cstdio>
#include <algorithm>

using namespace std;

int main() {
    long long n, m, k, l;
    cin >> n >> m >> k >> l;
    if (l > (m - 1) * (n - k)) {
        cout << "-1" << endl;
        return 0;
    }
    cout << (n - k + l - 1) / (m - 1) << endl;
    return 0;
}
