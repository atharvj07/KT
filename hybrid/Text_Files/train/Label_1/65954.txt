import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n = int(input())
    s = input().strip()
    a, b = ['']*n, ['']*n
    L = R = 0
    for i in range(n):
        if s[i]=='1':
            a[i], b[i] = '(', ')'
        else:
            if L:
                L -= 1
                a[i], b[i] = '(', ')'
            elif R:
                R -= 1
                a[i], b[i] = ')', '('
            else:
                a[i], b[i] = '(', ')'
                L += 1
    if L==0 and R==0:
        print("YES")
        print(''.join(a))
        print(''.join(b))
    else:
        print("NO")
