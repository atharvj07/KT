#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string n;
  cin >> n;
  int len = n.length();
  vector<int> digits(len);
  for (int i = 0; i < len; i++) {
    digits[i] = n[i] - '0';
  }
  bool found = false;
  for (int i = 0; i < len - 1; i++) {
    for (int j = i + 1; j < len; j++) {
      if (digits[i] % 2 && digits[j] % 2) {
        swap(digits[i], digits[j]);
        found = true;
        break;
      }
    }
    if (found) break;
  }
  if (!found) {
    cout << -1 << endl;
    return 0;
  }
  sort(digits.begin(), digits.end(), greater<int>());
  for (int i = 0; i < len; i++) {
    cout << digits[i];
  }
  cout << endl;
  return 0;
}
