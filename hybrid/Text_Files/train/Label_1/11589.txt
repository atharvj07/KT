#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, a, b, k;
    std::cin >> n >> a >> b >> k;
    std::string s;
    std::cin >> s;

    std::vector<bool> shot(n, false);
    for (int i = 0; i < n; ++i) {
        if (s[i] == '1') shot[i] = true;
    }

    std::vector<int> ans;
    int cnt = 0;
    for (int i = 0; i < n; ++i) {
        if (!shot[i]) {
            if (cnt == 0) {
                ans.push_back(i + 1);
            }
            cnt++;
            if (cnt == a) {
                break;
            }
        } else {
            cnt = 0;
        }
    }

    int sz = ans.size();
    std::cout << sz << std::endl;
    for (int i = 0; i < sz; ++i) {
        std::cout << ans[i] << " ";
    }
    std::cout << std::endl;

    return 0;
}
