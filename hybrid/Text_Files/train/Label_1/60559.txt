import sys
input = sys.stdin.readline

n = int(input())
rings = []
for _ in range(n):
    a, b, h = map(int, input().split())
    rings.append((a, b, h))

rings.sort(key=lambda x: x[1])
dp = [0] * n
dp[0] = rings[0][2]
for i in range(1, n):
    for j in range(i):
        if rings[i][1] > rings[j][0] and dp[i] < dp[j] + rings[i][2]:
            dp[i] = dp[j] + rings[i][2]
print(max(dp))
