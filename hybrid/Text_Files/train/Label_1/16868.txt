#include<bits/stdc++.h>
using namespace std;
int main()
{
long long int n,i,pos,neg,ans;
cin>>n;
long long int a[n];
pos=0;
neg=0;
for(i=0;i<n;++i)
{
cin>>a[i];
}
for(i=1;i<n;++i)
{
if(a[i]==a[i-1])
continue;
else if(a[i]>a[i-1])
{
pos+=a[i]-a[i-1];
}
else
{
neg+=a[i-1]-a[i];
}
if(pos>neg)
{
cout<<"NO"<<endl;
return 0;
}
}
cout<<"YES"<<endl;
return 0;
}
