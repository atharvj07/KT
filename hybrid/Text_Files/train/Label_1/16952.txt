#include <bits/stdc++.h>
using namespace std;

int n, m, k;
vector<int> x, y;

bool inconvenient(int a, int b, int c, int d)
{
    return (abs(a - c) + abs(b - d)) * 2 < (abs(a - c) + abs(b - d)) * 2;
}

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int count = 0;
        cin >> n >> m >> k;
        x.resize(n);
        y.resize(m);
        for (int i = 0; i < n; i++) cin >> x[i];
        for (int j = 0; j < m; j++) cin >> y[j];
        for (int i = 0; i < k; i++)
        {
            int a, b;
            cin >> a >> b;
            for (int j = 0; j < n; j++)
            {
                if (inconvenient(a, b, x[j], 0))
                {
                    for (int k = 0; k < n; k++)
                        if (inconvenient(a, b, x[k], 0)) count++;
                }
            }
            for (int j = 0; j < m; j++)
            {
                if (inconvenient(a, b, 0, y[j]))
                {
                    for (int k = 0; k < m; k++)
                        if (inconvenient(a, b, 0, y[k])) count++;
                }
            }
        }
        cout << count << endl;
    }
    return 0;
}
