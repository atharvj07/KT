#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), c(n);
  for (int i = 0; i < n; i++)
    cin >> a[i];
  for (int i = 0; i < n; i++)
    cin >> b[i];
  for (int i = 0; i < n; i++) {
    c[i] = (a[i] + b[i]) % n;
  }
  sort(b.begin(), b.end());
  sort(c.begin(), c.end());
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (c[i] == (a[j] + b[i]) % n) {
        b[i] = b[j];
        break;
      }
    }
  }
  for (auto x : b) cout << x << ' ';
  cout << endl;
}
