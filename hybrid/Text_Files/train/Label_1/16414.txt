#include <bits/stdc++.h>

using namespace std;

#define modulo 998244353

typedef long long ll;

ll binpow(ll a, ll b) {
    if (b == 0)
        return 1;
    if (b % 2 == 0) {
        ll ret = binpow(a, b / 2) % modulo;
        return (ret * ret) % modulo;
    }
    return (a * binpow(a, b - 1)) % modulo;
}

int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    vector<int> b(m);
    for (int i = 0; i < m; i++)
        cin >> b[i];
    vector<int> v(n, 1);
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < b[i]; j++)
            v[j] = 0, v[n - j - 1] = 0;
    }
    ll ans = 1;
    for (auto x : v)
        ans = (ans * x) % modulo;
    ans = binpow(2, ans);
    cout << ((ans % modulo) + modulo) % modulo << endl;
    return 0;
}
