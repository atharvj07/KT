#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> edges;
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        edges.push_back({a, b});
    }
    sort(edges.begin(), edges.end());
    vector<int> par(n + 1, -1);
    for (int i = 0; i < n - 1; i++) {
        int a = edges[i].first, b = edges[i].second;
        if (par[a] == -1) par[a] = b;
        else if (par[b] == -1) par[b] = a;
        else {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) {
        cout << i << " " << par[i] << endl;
    }
    return 0;
}
