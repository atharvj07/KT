#include<bits/stdc++.h>
using namespace std;
int main() {
    int t; cin >> t;
    string s; cin >> s;
    unordered_map<int, int> m;
    for(char ch: s) {
        m[ch - '0']++;
    }
    int ans = 0;
    ans += min(m[2], m[5]);
    m[2] -= ans; m[5] -= ans;
    ans += min(m[6], m[9]);
    m[6] -= ans; m[9] -= ans;
    ans += min(m[1], m[4]);
    m[1] -= ans; m[4] -= ans;
    ans += min(m[0], m[8]);
    m[0] -= ans; m[8] -= ans;
    ans += m[3]; ans += m[7];
    cout << ans << endl;
}
