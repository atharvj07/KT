#include <bits/stdc++.h>
#define ll long long
#define int ll
using namespace std;
vector<pair<int, int>> a;
int32_t main() {
    int t = 8;
    while (t--) {
        int x, y;
        cin >> x >> y;
        a.push_back({x, y});
    }
    sort(a.begin(), a.end());
    bool ok = 1;
    vector<int> x, y;
    for (int i = 0; i < 8; ++i) {
        if (i % 2 == 0) x.push_back(a[i].first);
        else y.push_back(a[i].second);
    }
    if (x[1] != x[2] && y[1] != y[2]) ok = 0;
    if (ok) cout << "respectable\n";
    else cout << "ugly\n";
    return 0;
}
