import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
  static class Edge {
    int to;
    int weight;

    public Edge(int to, int weight) {
      this.to = to;
      this.weight = weight;
    }
  }

  static int n, k;
  static List<List<Edge>> graph;
  static boolean[] visited;
  static Map<Integer, Boolean> inPath;
  static int minCost = Integer.MAX_VALUE;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] line1 = br.readLine().split(" ");
    n = Integer.parseInt(line1[0]);
    k = Integer.parseInt(line1[1]);
    line1 = br.readLine().split(" ");
    int m = Integer.parseInt(line1[0]);
    graph = new ArrayList<>();
    for (int i = 0; i <= n; i++) {
      graph.add(new ArrayList<>());
    }

    for (int i = 0; i < m; i++) {
      line1 = br.readLine().split(" ");
      int u = Integer.parseInt(line1[0]);
      int v = Integer.parseInt(line1[1]);
      int w = Integer.parseInt(line1[2]);
      graph.get(u).add(new Edge(v, w));
      graph.get(v).add(new Edge(u, w));
    }

    inPath = new HashMap<>();
    visited = new boolean[n + 1];
    dfs(1, 1, 0);
    if (minCost == Integer.MAX_VALUE) {
      System.out.println(-1);
    } else {
      System.out.println(minCost);
    }
  }

  static void dfs(int curr, int cnt, int cost) {
    if (cnt == k) {
      minCost = Math.min(minCost, cost);
      return;
    }
    for (Edge next : graph.get(curr)) {
      if (!visited[next.to] && !inPath.containsKey(next.to)) {
        inPath.put(curr, true);
        visited[next.to] = true;
        dfs(next.to, cnt + 1, cost + next.weight);
        inPath.remove(curr);
        visited[next.to] = false;
      }
    }
  }
}
