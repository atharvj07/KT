import java.util.*;

public class Main {
	static int n, k;
	static int[] p;
	static int[] q;
	static List<Integer>[] g;
	static int[][] dp;
	static boolean[] vis;

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		n = sc.nextInt();
		k = sc.nextInt();
		g = new ArrayList[n];
		for (int i = 0; i < n; i++) {
			g[i] = new ArrayList<>();
		}
		for (int i = 0; i < n - 1; i++) {
			int a = sc.nextInt() - 1;
			int b = sc.nextInt() - 1;
			g[a].add(b);
			g[b].add(a);
		}
		p = new int[n];
		q = new int[n];
		Arrays.fill(p, -1);
		dp = new int[n][k + 1];
		for (int i = 0; i < n; i++) {
			Arrays.fill(dp[i], -1);
		}
		vis = new boolean[n];
		int ans = dfs(0, 0);
		System.out.println(ans);
	}

	static int dfs(int u, int d) {
		if (vis[u]) {
			return dp[u][d];
		}
		vis[u] = true;
		dp[u][d] = 0;
		for (int v : g[u]) {
			if (v == p[u]) {
				continue;
			}
			p[v] = u;
			q[v] = q[u] + 1;
			int sub = 2 * q[u] - q[v];
			if (sub > k) {
				dp[u][d] += dfs(v, d + 1);
			} else {
				dp[u][d] += dfs(v, d);
			}
		}
		if (d == 0) {
			int ans = n - 1;
			for (int v : g[u]) {
				if (v == p[u]) {
					continue;
				}
				int sub = 2 * q[u] - q[v];
				if (sub <= k) {
					ans -= dfs(v, 0);
				}
			}
			dp[u][d] = Math.min(dp[u][d], ans);
		}
		return dp[u][d];
	}
}
