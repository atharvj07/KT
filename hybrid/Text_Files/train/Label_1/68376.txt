from collections import defaultdict, deque

adj = defaultdict(list)
n = int(input())
for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

q = deque([1])
par = [-1 for i in range(n + 1)]
dep = [0 for i in range(n + 1)]
while q:
    cur = q.popleft()
    for u in adj[cur]:
        if u == par[cur]:
            continue
        q.append(u)
        par[u] = cur
        dep[u] = dep[cur] + 1

ans = 0
rem = n
lst = []

def dfs(node):
    global ans, rem, lst
    if len(adj[node]) == 1 and par[node] != -1:
        ans += dep[node]
        lst.append([node, par[node], node if dep[node] % 2 == 1 else par[node]])
        rem -= 1
        adj[par[node]].remove(node)
        return

    for u in adj[node]:
        if u == par[node]:
            continue
        dfs(u)

dfs(1)

print(ans)

for x in lst:
    print(*x)
