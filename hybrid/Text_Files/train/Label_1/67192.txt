n = int(input())
v = list(map(int, input().split()))
m = int(input())
a, p, x = [], [], []
for _ in range(m):
    t, *query = map(int, input().split())
    a.append(t)
    p.append(query[0])
    x.append(query[1] if t == 1 else 0)
w = [0] * n
for t, i, j in zip(a, p, x):
    if t == 1:
        w[i - 1] += j
        if w[i - 1] > v[i - 1]:
            w[i - 1] = v[i - 1]
            w[i] += (j - (v[i - 1] - w[i - 1]))
    else:
        print(w[i - 1])
