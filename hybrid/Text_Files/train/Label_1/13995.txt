#include <bits/stdc++.h>
using namespace std;

const int maxn = 1000;
int a[maxn][maxn];
int n, m;

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      char c;
      cin >> c;
      if (c == '.' || c == '#' || (c >= '1' && c <= '3')) a[i][j] = c - '0';
      else
        a[i][j] = -1;
    }
  }
  bool ok = false;
  for (int c = 1; c <= 3 && !ok; ++c) {
    queue<pair<int, int>> q;
    for (int i = 0; i < n && !ok; ++i) {
      for (int j = 0; j < m && !ok; ++j) {
        if (a[i][j] == c) {
          ok = true;
          q.push({i, j});
          a[i][j] = 0;
        }
      }
    }
    while (!q.empty()) {
      pair<int, int> v = q.front();
      q.pop();
      for (int i = -1; i <= 1; i += 2) {
        if (v.second + i >= 0 && v.second + i < m && a[v.first][v.second + i] != 0 &&
            a[v.first][v.second + i] != c && a[v.first][v.second + i] != -1) {
          a[v.first][v.second + i] = 0;
          q.push({v.first, v.second + i});
        }
      }
      for (int i = -1; i <= 1; i += 2) {
        if (v.first + i >= 0 && v.first + i < n && a[v.first + i][v.second] != 0 &&
            a[v.first + i][v.second] != c && a[v.first + i][v.second] != -1) {
          a[v.first + i][v.second] = 0;
          q.push({v.first + i, v.second});
        }
      }
    }
  }
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (a[i][j] == '.') ++cnt;
    }
  }
  if (ok)
    cout << cnt << endl;
  else
    cout << -1 << endl;
  return 0;
}
