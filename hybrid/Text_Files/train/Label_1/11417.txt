#include <iostream>
#include <map>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  map<int, int> freq;
  int l = 1, r = 0, max_len = 0, start = 0;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    freq[a]++;
    r++;
    while (freq.size() > k) {
      freq[a]--;
      if (freq[a] == 0) freq.erase(a);
      l++;
    }
    if (r - l + 1 > max_len) {
      max_len = r - l + 1;
      start = l;
    }
  }
  cout << start << " " << start + max_len - 1 << endl;
}
