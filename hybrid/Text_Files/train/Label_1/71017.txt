mod = int(1e9)+7
for _ in range(int(input())):
    n,m,k = map(int,input().split())
    a = list(map(int,input().split()))
    b = [0]*n
    for i in a: b[i-1]+=1
    for i in range(1,n): b[i]+=b[i-1]

    ans = 0
    for i in range(n):
        if not b[i]: continue
        l = max(0,i-k)
        r = i
        ans += (b[r] - b[l] + b[i]) * b[i] % mod
    print(ans % mod)
