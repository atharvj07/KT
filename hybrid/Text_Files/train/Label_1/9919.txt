#include <bits/stdc++.h>

using namespace std;

int g[400005];
vector<int> vec[400005];

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        for (int i = 0; i <= n; i++) vec[i].clear(), g[i] = 0;
        for (int i = 0; i < m; i++) {
            int u, v;
            cin >> u >> v;
            vec[u].push_back(v);
        }
        queue<int> q;
        for (int i = 1; i <= n; i++) {
            if (vec[i].empty()) {
                q.push(i);
                g[i] = 1;
            }
        }
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (auto &v : vec[u]) {
                g[v]--;
                if (g[v] == 0) {
                    q.push(v);
                    g[v] = 1;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (g[i] == 1) cout << 1;
            else if (g[i] > 1) cout << 2;
            else if (g[i] == 0) cout << 0;
            else cout << -1;
        }
        cout << "\n";
    }
}
