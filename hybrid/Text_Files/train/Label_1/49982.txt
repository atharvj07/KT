import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong();
        long x = input.nextLong();
        long y = input.nextLong();
        long t = n / 2 * y + (n % 2) * x;
        System.out.println(t);
    }
}
