#include <iostream>
#include <cmath>

const long long MOD = 1e9 + 7;

int main() {
    long long a, b;
    std::cin >> a >> b;
    if (a == 1)
        std::cout << 0 << std::endl;
    else {
        long long ans = (a * 2 - 1) * b % MOD;
        ans = (ans * (ans + 1) / 2) % MOD;
        ans = (ans + MOD - a) % MOD;
        std::cout << ans << std::endl;
    }
    return 0;
}
