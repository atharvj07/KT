from sys import stdin
input = stdin.readline

t = int(input())
for i in range(t):
    n, m, k = map(int, input().split())
    a = [list(input()) for i in range(n)]
    f = [[False] * m for i in range(n)]
    for j in range(n):
        for l in range(m):
            if a[j][l] == '*':
                f[j][l] = True
                for p in range(1, k):
                    if j - p < 0 or l - p < 0: break
                    if a[j - p][l - p] == '.':
                        f[j][l] = False
                        break
            else:
                for p in range(1, k):
                    if j + p >= n or l + p >= m: break
                    if a[j + p][l + p] == '*':
                        f[j][l] = False
                        break
    for j in range(n):
        for l in range(m):
            if a[j][l] != '.' and not f[j][l]:
                print('NO')
                break
        else: continue
        break
    else: print('YES')
