import java.util.Scanner;

public class DotGame {

    static class Dot {
        int x, y;

        public Dot(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();
        int n = scanner.nextInt();
        int d = scanner.nextInt();
        Dot dot = new Dot(x, y);

        for (int i = 0; i < n; i++) {
            int xi = scanner.nextInt();
            int yi = scanner.nextInt();
            if (AntonWins(dot, xi, yi, d)) {
                System.out.println("Anton");
                return;
            }

            dot = reflect(dot);
            if (DashaWins(dot, d)) {
                System.out.println("Dasha");
                return;
            }

            dot = reflect(dot);
        }

        // If no one can win, Anton wins by default
        System.out.println("Anton");
    }

    static boolean AntonWins(Dot dot, int xi, int yi, int d) {
        int newX = dot.x + xi;
        int newY = dot.y + yi;
        return Math.sqrt(newX * newX + newY * newY) > d;
    }

    static boolean DashaWins(Dot dot, int d) {
        return Math.sqrt(dot.x * dot.x + dot.y * dot.y) > d;
    }

    static Dot reflect(Dot dot) {
        return new Dot(dot.y, dot.x);
    }
}
