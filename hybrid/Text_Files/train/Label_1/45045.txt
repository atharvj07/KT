import java.util.Scanner;

public class SecondPriceAuction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] bids = new int[n];
        for (int i = 0; i < n; i++) {
            bids[i] = scanner.nextInt();
        }

        int winnerIndex = 0;
        int winningBid = Integer.MIN_VALUE;
        for (int i = 1; i < n; i++) {
            if (bids[i] > bids[winnerIndex]) {
                winnerIndex = i;
                winningBid = bids[i - 1];
            }
        }

        System.out.println((winnerIndex + 1) + " " + winningBid);
    }
}
