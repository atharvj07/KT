#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    if (n > k) {
      cout << n - k << endl;
    } else if (k % 2 != n % 2) {
      cout << 1 << endl;
    } else {
      cout << (k - n) / 2 << endl;
    }
  }
  return 0;
}
