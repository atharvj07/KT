import sys

N, p = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(N)]

arr.sort(key=lambda x: x[1] / x[0])

ans = 0
cur = 0
for a, b in arr:
    t = (b - cur + a - 1) // a
    ans += t
    cur += t * a - b
    if cur < 0:
        print(-1)
        sys.exit()
print(ans + cur / p)
