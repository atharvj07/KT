#include <bits/stdc++.h>
using namespace std;
int p[100005];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, ans = 0;
    cin >> n;
    for (int i = 1, k; i <= n; i++) {
      cin >> k;
      int mx = 0;
      for (int j = 1, a; j <= k; j++) {
        cin >> a;
        mx = max(mx, a);
      }
      ans = max(ans, mx);
    }
    cout << ++ans << endl;
  }
  return 0;
}
