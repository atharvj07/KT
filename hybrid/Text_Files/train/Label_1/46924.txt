import java.util.Scanner;

public class Main {
    static void catBSoft(int n, int k) {
        k--;
        if (k < n) {
            System.out.println(k + 1);
        } else {
            int A = k / n;
            int B = k % n;
            int p = B - A;
            if (p <= 0) {
                p += n;
            }
            System.out.println((p + 1));
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- != 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            catBSoft(n, k);
        }
    }
}
