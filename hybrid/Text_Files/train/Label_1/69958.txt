import sys

n = int(input())
l = list(map(int, input().split()))
r = list(map(int, input().split()))
a = [0] * n

for i in range(1, n):
    a[i] = max(a[i-1] - r[i-1], 1)
for i in range(n-2,-1,-1):
    a[i] = max(a[i+1] - l[i+1], a[i])

print("YES" if all(l[i] <= r[i] for i in range(n)) and all(1 <= ai <= n for ai in a) else "NO")
if "YES" in sys.stdout.getvalue():
    print(*a)
