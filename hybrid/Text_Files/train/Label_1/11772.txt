#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
struct point {int x, y;};
int n, sx, sy, sz, ex, ey, ez;
char G[75][75];
vector<pair<int, int> > V;
int main() {
    cin >> n >> sx >> sy >> sz;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) cin >> G[i][j];
    }
    ex = 1;
    ey = 2;
    ez = 3;
    int ans = 0;
    while (sx ^ ex || sy ^ ey || sz ^ ez) {
        int nx = -1, ny = -1, nz = -1;
        if (G[sx][ex] == G[sy][ey] && sy != ex) {
            ans++;
            nx = ex, ny = sy;
        }
        else if (G[sy][ez] == G[sx][ex] && sx != ez) {
            ans++;
            nx = ez, ny = sx;
        }
        else if (G[sz][ey] == G[sx][ex] && sz != ey) {
            ans++;
            nx = ey, ny = sz;
        }
        else if (G[sx][ez] == G[sy][ey] && sy != ez) {
            ans++;
            nx = ez, ny = sy;
        }
        else if (G[sy][ez] == G[sz][ey] && sz != ey) {
            ans++;
            nx = ey, ny = sz;
        }
        else if (G[sx][ez] == G[sz][ey] && sx != ez) {
            ans++;
            nx = ez, ny = sx;
        }
        if (nx == -1) {cout << -1 << endl; return 0;}
        V.push_back(make_pair(sx, ny));
        sx = nx, sy = ny, sz = nz;
    }
    cout << ans << endl;
    for (int i = 0, size = V.size(); i < size; i++)
        cout << V[i].first << " " << V[i].second << endl;
    return 0;
}
