#include<bits/stdc++.h>
using namespace std;
vector<int>g[200001];
bitset<200001>vis;
vector<pair<int,int>>ed;
void dfs(int x,int p){
    vis[x]=1;
    for(auto y:g[x]){
        if(y!=p){
            if(vis[y]==0){
                ed.push_back({x,y});
                dfs(y,x);
            }
        }
    }
}
int main(){
    int n,m,D;
    cin>>n>>m>>D;
    if(D==n-1){
        cout<<"NO\n";
        return 0;
    }
    cout<<"YES\n";
    for(int i=1;i<=m;i++){
        int x,y;
        cin>>x>>y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    dfs(1,-1);
    if(ed.size()!=n-1){
        cout<<"NO\n";
        return 0;
    }
    for(auto p:ed){
        cout<<p.first<<" "<<p.second<<endl;
    }
    return 0;
}
