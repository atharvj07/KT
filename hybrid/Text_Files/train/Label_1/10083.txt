#include <iostream>
#include <algorithm>
#include <vector>
#include <queue>

using namespace std;

const int N = 2e3 + 5;

int n, m;
char g[N][N];

int cnt[N][N];
int ans = 2e9;

int main() {
	cin >> n >> m;
	for (int i = 1; i <= n; ++i) {
		for (int j = 1; j <= m; ++j) {
			cin >> g[i][j];
			g[i][j] = (g[i][j] == '0' ? 0 : 1);
		}
	}

	for (int k = 1; k <= n + m; ++k) {
		int cur = 0;
		for (int i = 1; i <= n; ++i) {
			if (i % k != 1) continue;
			for (int j = 1; j <= m; ++j) {
				if (j % k != 1) continue;
				cnt[i][j] = !cnt[i][j];
				cur += !cnt[i][j];
			}
		}

		for (int i = 1; i <= n; ++i) {
			for (int j = 1; j <= m; ++j) {
				if (i % k != 1 && j % k != 1) {
					cnt[i][j] = cnt[i - i % k + 1][j - j % k + 1];
				}
				cur += !cnt[i][j];
			}
		}

		for (int i = 1; i <= n; ++i) {
			for (int j = 1; j <= m; ++j) {
				int x = (i - 1) / k + 1;
				int y = (j - 1) / k + 1;
				cur -= !cnt[i][j];
				cnt[i][j] ^= cnt[x][y];
				cur += !cnt[i][j];
			}
		}

		ans = min(ans, cur);
	}

	cout << ans << endl;
}
