n, m = map(int, input().split())
sx, sy, fx, fy = map(int, input().split())
loc = [(x, y) for x, y in (map(int, input().split()) for _ in range(m))]
mn = 1e9
for x, y in loc:
    if x == sx or y == sy:
        mn = min(mn, abs(fx - sx) + abs(fy - y))
    if y == fx or y == fy:
        mn = min(mn, abs(fx - x) + abs(fy - sy))
if mn == 1e9:
    mn = abs(fx - sx) + abs(fy - sy)
print(mn)
