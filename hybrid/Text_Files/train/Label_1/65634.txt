def bs(ar, l, r, x):
    while l < r:
        m = (l + r) // 2
        if ar[m] >= x:
            r = m
        else:
            l = m + 1
    return l

a = list(map(int, input().split())) + [0]
mod = 1000000007
ans = 1
k = 0

for i in range(len(a) - 1):
    op = a[i] & 1
    p = a[i] >> 1
    if not op:
        pos = bs(a, i + 1, len(a), p) - i - 1
        ans = (ans * (2 ** pos)) % mod
    else:
        pos = bs(a, 0, i, p)
        ans = (ans * max(1, i - pos)) % mod
    k += not op

print(k == ans and ans or 0)
