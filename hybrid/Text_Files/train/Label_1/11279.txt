#include <bits/stdc++.h>
#define int long long
using namespace std;
struct st{
    int a,b;
};
vector<vector<int>> adj;
vector<int> indeg;
vector<st> edg;
int n,m;

int32_t main(){
    cin>>n>>m;
    adj.resize(n+1);
    indeg.assign(n+1,0);
    for(int i=0;i<m;i++){
        int u,v;
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        edg.push_back({u,v});
        indeg[u]++, indeg[v]++;
    }
    priority_queue<int> q;
    for(int i=1;i<=n;i++){
        if(indeg[i]==1)
            q.push(i);
    }
    int tail_len=0;
    vector<int> tail;
    while(!q.empty()){
        int u=q.top();
        q.pop();
        tail.push_back(u);
        tail_len++;
        for(int v:adj[u]){
            indeg[v]--;
            if(indeg[v]==1){
                q.push(v);
            }
        }
    }
    int ans=0;
    for(auto x:edg){
        int u=x.a,v=x.b;
        if((indeg[u]!=1 or indeg[v]!=2) and (indeg[v]!=1 or indeg[u]!=2)){
            ans++;
        }
    }
    cout<<tail_len*ans<<"\n";
}
