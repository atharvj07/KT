from collections import defaultdict

n, m = map(int, input().split())
edges = defaultdict(list)
for _ in range(n - 1):
    u, v, w = map(int, input().split())
    edges[u].append((v, w))
    edges[v].append((u, w))

queries = list(map(int, input().split()))

# Perform DFS to compute the maximum edge weight on all paths from a given root
max_edge_weights = [-1] * (n + 1)

def dfs(u, p):
    for v, w in edges[u]:
        if v == p:
            continue
        max_edge_weights[v] = max(max_edge_weights[u], w)
        dfs(v, u)

dfs(1, 1)

# Compute the answer for each query
for q in queries:
    ans = 0
    for i in range(1, n + 1):
        if max_edge_weights[i] <= q:
            ans += i - 1
    print(ans)
