k1,k2,k3=map(int,input().split())
a1=[int(x) for x in input().split()]
a2=[int(x) for x in input().split()]
a3=[int(x) for x in input().split()]
x=sorted(a1+a2+a3)
p1=0
p2=0
ans=0
for i in x:
    if p1<k1 and i==a1[p1]:
        p1+=1
    elif p2<k2 and i==a2[p2]:
        p2+=1
    else:
        ans+=1
print(ans)
