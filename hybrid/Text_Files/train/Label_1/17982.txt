#include <bits/stdc++.h>

using namespace std;

int main()
{
    long long n;
    cin >> n;
    map<long long, long long> m;
    for (long long i = 0; i < n; i++)
    {
        long long x;
        cin >> x;
        if (m.count(x) > 0)
            m[x]++;
        else
            m[x] = 1;
    }
    set<long long> ans;
    for (auto i : m)
    {
        long long x = 1;
        while (1)
        {
            if (i.first / x <= 0)
                break;
            ans.insert(x);
            x *= 2;
        }
    }
    if (ans.empty())
        cout << -1 << endl;
    else
        for (auto x : ans)
            cout << x << " ";
    cout << endl;
    return 0;
}
