#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    vector<int> cnt(n + 1, 0);
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
      ++cnt[a[i]];
    }
    long long ans = 0;
    for (int i = 1; i <= n; ++i) { ans += (long long)cnt[i] * (cnt[i] - 1) / 2; }
    cout << ans << endl;
  }
  return 0;
}
