#include <vector>
#include <iostream>
using namespace std;
int main() {
    int n; cin >> n; vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int cnt = 0, last = 4;
    for (int i = 0; i < n; i++) {
        if (a[i] >= 2 && last % 2 == a[i] % 2) cnt++;
        last = a[i];
    }
    cout << cnt << endl;
}
