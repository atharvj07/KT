for _ in range(int(input())):
    s, t = input(), input()
    n = len(s)
    m = len(t)
    if n + m - 1 == 2 * n:
        print("YES")
        continue
    i, j, ok = 0, 0, True
    while i < n and j < m:
        if s[i] == t[j]:
            i += 1
            j += 1
        elif s[i] != t[j - 1]:
            ok = False
            break
        i += 1
    if ok and i == n and j == m:
        print("YES")
    else:
        print("NO")
