#include <iostream>
#include <map>

int main() {
    int n, m, ans = 0;
    std::string s, h;
    std::map<char, int> mp;
    std::cin >> n >> s >> m;
    while (m--) {
        std::cin >> h;
        for (int i = 0; i < n; i++) {
            if (s[i] != '*') mp[h[i]]++;
        }
    }
    for (char c = 'a'; c <= 'z'; c++) {
        if (mp[c] == 1) ans++;
    }
    std::cout << ans << std::endl;
    return 0;
}
