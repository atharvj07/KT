def string_product(s, t):
    result = t
    for char in s:
        result += char + t
    return result

def beauty_of_string(s):
    max_beauty = 1
    current_beauty = 1
    for i in range(1, len(s)):
        if s[i] == s[i-1]:
            current_beauty += 1
            max_beauty = max(max_beauty, current_beauty)
        else:
            current_beauty = 1
    return max_beauty

n = int(input())
strings = [input() for _ in range(n)]

result_string = strings[0]
for i in range(1, n):
    result_string = string_product(result_string, strings[i])

print(beauty_of_string(result_string))
