#include<iostream>
#include<vector>
#include<queue>
using namespace std;
struct edge {
    int to,w;
    edge(int _to,int _w):to(_to),w(_w) {}
};
int n,m,s,t;
vector<int> dis,vis;
vector<vector<edge>> g;
bool spfa(int md) {
    queue<int> q;
    q.push(s);
    vis[s]=1;
    while(!q.empty()) {
        int u=q.front();q.pop();
        if(u==t) return true;
        vis[u]=0;
        for(edge &e:g[u]) {
            if(e.w>md) continue;
            if(dis[e.to]>dis[u]+e.w) {
                dis[e.to]=dis[u]+e.w;
                if(!vis[e.to]) {
                    vis[e.to]=1;
                    q.push(e.to);
                }
            }
        }
    }
    return false;
}
int main() {
    cin>>n>>m>>s>>t;
    g.resize(n+1);
    vis.resize(n+1);
    dis.resize(n+1,1e9);
    dis[s]=0;
    while(m--) {
        int x,y,w;
        cin>>x>>y>>w;
        g[x].push_back(edge(y,w));
    }
    if(spfa(1e9))
        cout<<"YES\n";
    else {
        int l=0,r=(int)1e9,ans=-1;
        while(l<=r) {
            int mid=(l+r)>>1;
            if(spfa(mid)) {
                ans=mid;
                r=mid-1;
            }
            else l=mid+1;
        }
        if(ans==-1) cout<<"NO\n";
        else {
            cout<<"CAN ";
            spfa(ans);
            cout<<dis[t]-ans<<'\n';
        }
    }
    return 0;
}
