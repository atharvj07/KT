import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    g = [[] for i in range(n+1)]
    for i in range(n-1):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)

    vis = [False] * (n+1)
    XOR = -1

    def dfs(u, p):
        nonlocal XOR
        if vis[u]: return
        vis[u] = True

        if XOR == -1: XOR = a[u-1]
        else: XOR ^= a[u-1]

        for v in g[u]:
            if v == p: continue
            dfs(v, u)

    dfs(1, 1)

    for i in range(1, n+1):
        if not vis[i]:
            XOR = -1
            dfs(i, i)
        if XOR != 0:
            print("NO")
            break
    else:
        print("YES")
