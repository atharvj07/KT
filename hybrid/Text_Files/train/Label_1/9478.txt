#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int even = 0, odd = 0;
  for (int i = 0; i < s.size(); i++) {
    int j = i;
    while (j < s.size() && s[j] == s[i]) j++;
    int len = j - i;
    if (len % 2 == 0) even++;
    else odd++;
    i = j - 1;
  }
  cout << even << " " << odd << endl;
  return 0;
}
