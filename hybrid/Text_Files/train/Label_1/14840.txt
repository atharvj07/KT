#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n;
    cin >> n;
    if (n == 3)
      cout << 2 << "\n"
           << 3 << " " << 2 << "\n"
           << 3 << " " << 2 << "\n";
    else {
      cout << n + 5 << "\n";
      for (ll i = 1; i <= n; i++) cout << n << " " << (i % n == 0 ? n : i % n) << "\n";
    }
  }
}
