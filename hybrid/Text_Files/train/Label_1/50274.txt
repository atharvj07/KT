import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] h = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextLong();
        }
        long[][] g = new long[n][];
        for (int i = 0; i < n; i++) {
            g[i] = new long[n];
            Arrays.fill(g[i], Long.MAX_VALUE);
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            g[u][v] = 0;
        }
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    g[i][j] = Math.min(g[i][j], g[i][k] + g[k][j]);
                }
            }
        }
        boolean canWin = false;
        long[] h2 = new long[n];
        for (int i = 0; i < n; i++) {
            boolean ok = true;
            for (int j = 0; j < n; j++) {
                if (g[i][j] == Long.MAX_VALUE) {
                    ok = false;
                } else if (h[j] != 0) {
                    h2[i] = Math.min(h2[i], h[j] - g[i][j]);
                }
            }
            if (ok && h2[i] >= 0) {
                canWin = true;
                break;
            }
        }
        if (canWin) {
            System.out.println("WIN");
            for (long x : h2) {
                System.out.print(x + " ");
            }
        } else {
            System.out.println("LOSE");
        }
    }
}
