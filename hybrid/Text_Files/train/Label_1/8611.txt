#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n, a, r, m, sum = 0, min_h = 1e9;
  cin >> n >> a >> r >> m;
  vector<long long> h(n);
  for (int i = 0; i < n; i++) {
    cin >> h[i];
    sum += h[i];
    min_h = min(min_h, h[i]);
  }
  sort(h.begin(), h.end());
  long long ans = 1e9;
  for (long long i = min_h; i <= h[n / 2]; i++) {
    long long op1 = 0, op2 = 0;
    for (long long j = 0; j < n; j++) {
      if (h[j] > i) {
        op1 += (h[j] - i) * r;
        op2 += (h[j] - i) * min(m, a + r);
      }
      if (h[j] < i) {
        op1 += (i - h[j]) * a;
        op2 += (i - h[j]) * min(m, a + r);
      }
    }
    ans = min({ans, op1, op2});
  }
  ans += (sum - (n * min_h));
  cout << ans << "\n";
}
