import java.io.*;

public class Main {

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		String[] input = br.readLine().split(" ");
		int n = Integer.parseInt(input[0]);
		int r = Integer.parseInt(input[1]);
		Duck[] ducks = new Duck[n];
		for (int i = 0; i < n; i++) {
			input = br.readLine().split(" ");
			ducks[i] = new Duck(Integer.parseInt(input[0]), Integer.parseInt(input[1]));
		}

		int maxDucks = 0;
		int lastShotTime = -1;
		for (int i = 0; i < n; i++) {
			if (lastShotTime + r <= ducks[i].tail) {
				maxDucks++;
				lastShotTime = ducks[i].tail;
			}
		}
		System.out.println(maxDucks);
	}

	private static class Duck {
		int head;
		int tail;

		public Duck(int head, int tail) {
			this.head = head;
			this.tail = tail;
		}
	}
}
