import sys
from collections import defaultdict, deque

input = sys.stdin.readline

def bfs(adj, in_degree, n):
    q = deque([1])
    cnt = defaultdict(int)
    dp = [0] * (n + 1)
    dp[1] = 1

    while q:
        u = q.popleft()
        for v in adj[u]:
            in_degree[v] -= 1
            cnt[v] += 1
            if in_degree[v] == 0:
                q.append(v)
                dp[v] = dp[u] + (cnt[v] > 1)

    return dp

for _ in range(int(input())):
    input()
    n, m = map(int, input().split())
    adj = defaultdict(list)
    in_degree = defaultdict(int)

    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        in_degree[v] += 1

    res = bfs(adj, in_degree, n)
    print(*res[1:])
