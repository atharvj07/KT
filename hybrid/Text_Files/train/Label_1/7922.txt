#include<bits/stdc++.h>
using namespace std;
const int mxn = 1e3+10;
bool poss[mxn][mxn];
int main(){
    int t; cin >> t;
    while(t--){
        int n; cin >> n;
        memset(poss, false, sizeof(poss));
        for(int i = 0; i < n; i++){
            int a,b,c,d,e; cin >> a >> b >> c >> d >> e;
            poss[i][0] = a;
            poss[i][1] = b;
            poss[i][2] = c;
            poss[i][3] = d;
            poss[i][4] = e;
        }
        bool flag = false;
        for(int i = 0; i < n; i++){
            for(int j = i+1; j < n; j++){
                int cnt = 0;
                bool f = false;
                for(int k = 0; k < 5; k++){
                    cnt+= (poss[i][k] && poss[j][k]);
                    f |= (poss[i][k] && poss[j][k]);
                }
                if(cnt == 1 && f){
                    flag = true;
                    break;
                }
            }
            if(flag) break;
        }
        cout << (flag ? "YES\n" : "NO\n");
    }
}
