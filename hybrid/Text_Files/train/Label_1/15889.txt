#include <iostream>

int main(int, char**) {
  int n, aa, bb;
  std::cin >> n >> aa >> bb;
  int aa2 = aa * 2;
  int tt;
  int denied = 0;
  while (n--) {
    std::cin >> tt;
    if (tt == 1) {
      if (aa > 0) {
        aa--;
      } else if (bb > 0) {
        bb--;
      } else if (aa2 > 0) {
        aa2--;
      } else {
        denied++;
      }
    } else {
      if (bb > 0) {
        bb--;
      } else {
        denied += 2;
      }
    }
  }
  std::cout << denied << std::endl;
  return 0;
}
