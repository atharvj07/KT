#include <vector>
#include <iostream>
using namespace std;

int main() {
    int n, d;
    cin >> n >> d;
    vector<vector<int>> graph(n + 1);
    vector<int> deg(n + 1);
    for (int i = 0; i < d; i++) {
        int u, v;
        cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
        deg[u]++;
        deg[v]++;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        ans = max(ans, deg[i]);
    }
    cout << ans << endl;
}
