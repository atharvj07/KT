#include <bits/stdc++.h>

#define int long long
#define mod 1000000007

using namespace std;

int power(int a, int b) {
  int ans = 1;
  while (b > 0) {
    if (b % 2 == 1) { ans = (ans * a) % mod; }
    a = (a * a) % mod;
    b /= 2;
  }
  return ans;
}

int32_t main() {
  int n, l, r;
  cin >> n >> l >> r;
  int sum = (l + r) * n;
  if (sum % 3 != 0) { cout << 0 << '\n'; } else {
    int x = (sum / 3 - l);
    int y = (r - sum / 3);
    if (x < 0 || y < 0) { cout << 0 << '\n'; } else {
      cout << (power(2LL, y) % mod * power(3LL, x) % mod) % mod << '\n';
    }
  }
}
