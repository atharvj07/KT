import java.util.Scanner;

public class CandyShop {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        long[] allowances = new long[n];
        for (int i = 0; i < n; i++) {
            allowances[i] = scanner.nextLong();
        }

        long maxCandies = 0;
        for (int i = 1; i <= m; i++) {
            long totalCost = i * (long) i;
            int kidsWithEnoughAllowance = 0;
            for (long allowance : allowances) {
                if (allowance >= totalCost) {
                    kidsWithEnoughAllowance++;
                }
            }
            maxCandies = Math.max(maxCandies, kidsWithEnoughAllowance * (long) i);
        }

        System.out.println(maxCandies);
    }
}
