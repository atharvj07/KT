import sys

n = int(input())
p = list(map(int, input().split()))
s = list(map(int, input().split()))
ans = []

for c in s:
    for i in range(n):
        if p[i] == c:
            if i != n - 1:
                ans.append((i + 1, n))
                p[i], p[n - 1] = p[n - 1], p[i]
            n -= 1
            break

print(len(ans))
print(len(ans))
for a in ans:
    print(*a)
