#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define rep(i, n) for(ll i = 0; i < n; i++)
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    ll t, n, temp;
    cin >> t;
    while (t--) {
        cin >> n;
        unordered_map<ll, ll> mp;
        rep(i, n) { cin >> temp; mp[temp]++; }
        ll ans = -1;
        for (auto it = mp.begin(); it != mp.end(); it++) {
            if (it->second == 1) { ans = it->first; break; }
        }
        cout << ans << "\n";
    }
}
