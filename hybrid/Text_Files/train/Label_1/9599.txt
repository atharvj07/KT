#include <bits/stdc++.h>
using namespace std;
const int maxn=55;
int cnt[maxn][26],c[maxn];
bool palin[maxn];
int main() {
    int q; cin>>q; while(q--) {
        int n; cin>>n;
        memset(cnt,0,sizeof(cnt)); memset(palin,0,sizeof(palin)); memset(c,0,sizeof(c));
        for(int i=1; i<=n; i++) {
            string s; cin>>s;
            for(int j=0; j<s.size(); j++) cnt[i][s[j]-'a']++;
            int odd=0; for(int k=0; k<26; k++) if(cnt[i][k]%2) odd++;
            palin[i]=odd<=1;
        }
        for(int i=1; i<=n; i++) for(int j=0; j<26; j++) c[j]+=cnt[i][j]%2;
        int ans=0; for(int j=0; j<26; j++) if(c[j]%2) ans++;
        if(ans%2) ans++;
        if(ans==0) ans=1;
        cout<<n-ans/2<<endl;
    }
    return 0;
}
