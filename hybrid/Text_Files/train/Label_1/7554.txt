#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<char> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    int ans = 0;
    int l = -1, r = -1;
    for (int i = 0; i < n; i++) {
        if (arr[i] == 'G') {
            if (l == -1) {
                l = i;
            }
            r = i;
        }
    }
    if (l != -1) {
        ans = max(ans, r - l + 1);
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] == 'S' && arr[j] == 'G') {
                int cur = 0;
                int ll = i, rr = j;
                if (r < ll) {
                    cur += rr - ll + 1;
                } else if (ll < l) {
                    cur += r - rr + 1;
                } else if (l < ll && rr < r) {
                    cur += r - ll + ll - l + 1;
                }
                ans = max(ans, cur);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
