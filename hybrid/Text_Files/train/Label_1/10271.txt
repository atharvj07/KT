#include <bits/stdc++.h>
using namespace std;

int n, cnt;
char grid[105][105];
int dx[] = {-1, -1, 0, 1, 1};
int dy[] = {-1, 0, 1, 1, 0};

bool valid(int x, int y) {
    if (x >= 0 && x < n && y >= 0 && y < n && grid[x][y] == '#') return true;
    return false;
}

void dfs(int x, int y) {
    if (cnt >= 5) return;
    grid[x][y] = '.';
    cnt++;
    for (int i = 0; i < 5; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        if (valid(nx, ny)) dfs(nx, ny);
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) { cin >> grid[i][j]; }
    }
    int ans = 1;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == '#') {
                cnt = 0;
                dfs(i, j);
                if (cnt != 5) ans = 0;
            }
        }
    }
    cout << (ans ? "YES" : "NO") << endl;
}
