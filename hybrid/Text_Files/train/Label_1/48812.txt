import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] target = {"the", "this", "that"};
        while (sc.hasNextLine()) {
            String c = sc.nextLine().strip();
            outer:
            for (int i = 0; i < 26; i++) {
                for (String t : target) {
                    String d = sa(c, i);
                    if (d.contains(t)) {
                        System.out.println(d);
                        break outer;
                    }
                }
            }
        }
    }

    static String sa(String s, int n) {
        char[] a = new char[s.length()];
        for (int i = 0; i < s.length(); i++) {
            char b = s.charAt(i);
            if (b >= 'a' && b <= 'z') {
                int j = b - 'a' + n;
                j %= 26;
                if (j < 0) j += 26;
                a[i] = (char) (j + 'a');
            } else {
                a[i] = b;
            }
        }
        return new String(a);
    }
}
