#include <iostream>
#include <unordered_set>
#include <string>

int main() {
  int n; std::cin >> n;
  std::unordered_set<std::string> set;
  for (int i = 0; i < n; ++i) {
    std::string s; std::cin >> s;
    if (!set.insert(s).second)
      std::cout << "YES" << std::endl;
    else
      std::cout << "NO" << std::endl;
  }
}
