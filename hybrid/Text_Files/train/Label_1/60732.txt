def fastpow(x: int, p: int) -> int:
	if p == 0:
		return 1
	if p == 1:
		return x
	val = fastpow(x, int(p / 2)) % 998244353
	val = (val * val) % 998244353
	if p % 2 != 0:
		val = (val * x) % 998244353
	return val % 998244353


n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
freq = {}
for ai in a:
	if ai not in freq:
		freq[ai] = 0
	freq[ai] += 1
cnt = 0
last = -1
for bi in b:
	if bi in freq:
		if freq[bi] > 0:
			freq[bi] -= 1
			if last != -1:
				cnt = (cnt * fastpow(2, freq[last])) % 998244353
			last = bi
		else:
			cnt = 0
			break
	else:
		cnt = 0
		break
if last != -1:
	cnt = (cnt * fastpow(2, freq[last])) % 998244353
print(cnt)
