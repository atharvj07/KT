import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = sc.nextInt();
    }
    int[] pref = new int[n];
    pref[0] = arr[0];
    for (int i = 1; i < n; i++) {
      pref[i] = pref[i - 1] * arr[i];
    }
    int[] suf = new int[n];
    suf[n - 1] = arr[n - 1];
    for (int i = n - 2; i >= 0; i--) {
      suf[i] = suf[i + 1] * arr[i];
    }
    int[] ans = new int[n];
    for (int i = 0; i < n; i++) {
      if (pref[i] == suf[i]) {
        ans[i] = 1;
      }
    }
    for (int i = 1; i < n; i++) {
      for (int j = i - 1; j >= 0; j--) {
        if (pref[j] * suf[i] == pref[i - 1] * pref[i - 1]) {
          ans[i] = ans[i - 1] + 1;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      System.out.print(ans[i] + " ");
    }
    System.out.println();
  }
}
