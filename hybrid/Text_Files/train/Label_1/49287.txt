import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int caseNum = 1;
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) break;
            int result = 0;
            int[] players = new int[n];
            while (true) {
                int cnt = 0;
                for (int i = 0; i < n; i++) {
                    if (players[i] == m) cnt++;
                }
                if (cnt > 0) break;
                result++;
                for (int i = 0; i < n; i++) {
                    for (int j = i + 1; j < n; j++) {
                        if (players[i] == players[j]) {
                            if (Math.random() < 0.5) players[i]++;
                            else players[j]++;
                        }
                    }
                }
            }
            System.out.println("Case " + caseNum + ": " + result);
            caseNum++;
        }
    }
}
