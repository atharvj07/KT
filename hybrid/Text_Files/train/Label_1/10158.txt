#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int n, m, t, h, M, s;
    cin >> n >> m >> t;
    vector<int> need(t * 60, 0);
    int id = 0;
    for (int i = 0; i < n; i++) {
        cin >> h >> M >> s;
        id++;
        for (int j = max(0, s - t); j < 60; j++) need[h * 60 + j]++;
        for (int j = 0; j < M; j++) {
            h++;
            if (h == 24) h = 0;
            for (int j = 0; j < 60; j++) need[h * 60 + j]++;
        }
    }
    int now = 0, R = 0;
    for (int i = 0; i < need.size(); i++) now = max(now - 1, need[i]), R = max(R, now);
    if (R > m)
        cout << "No solution";
    else {
        cout << R << endl;
        id = 0;
        for (int i = 0; i < n; i++) {
            cin >> h >> M >> s;
            now = 0;
            for (int j = max(0, s - t); j < 60; j++) now = max(now, need[h * 60 + j]);
            if (now >= m) {
                id++;
                for (int j = 0; j < M; j++) {
                    h++;
                    if (h == 24) h = 0;
                    for (int j = 0; j < 60; j++) need[h * 60 + j]++;
                }
            }
            cout << id << endl;
        }
    }
}
