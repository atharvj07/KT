#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;

  vector<long long> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }

  vector<vector<long long>> dp(n + 1, vector<long long>(k + 1));

  dp[0][0] = 0;
  for (int i = 1; i <= n; ++i) {
    for (int j = 0; j <= k; ++j) {
      dp[i][j] = dp[i - 1][j];
      if (j > 0) { dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] & a[i - 1]); }
    }
  }

  cout << dp[n][k] << endl;

  return 0;
}
