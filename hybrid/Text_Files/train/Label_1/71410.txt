import sys

n = int(input())
s = input()

if s[0] == ')' or s[-1] == '(':
    print(-1)
    sys.exit()

open = 0
need = 0
cost = 0
for c in s:
    if c == '(':
        open += 1
    else:
        if open > 0:
            open -= 1
        else:
            need += 1
            cost += 1
if open > 0:
    cost += open - 1
print(cost * need)
