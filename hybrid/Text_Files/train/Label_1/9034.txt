#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;
ll fastpow(ll x, ll y) {
  ll ret = 1;
  while (y) {
    if (y & 1) ret = (ret * x) % MOD;
    y >>= 1;
    x = (x * x) % MOD;
  }
  return ret;
}
int main() {
  ll n, m;
  cin >> n >> m;
  cout << ((m - n + 1) * fastpow(2, (n - 2)) % MOD) % MOD << '\n';
}
