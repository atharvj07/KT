import java.util.Scanner;
import java.util.HashMap;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String s = sc.nextLine();
        int n = sc.nextInt();
        HashMap<String, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++)
            map.put(sc.next(), 1);

        int maxLen = 0, start = 0;

        for (int i = 0, j = 0, k = 0; i < s.length(); i++) {
            while (k < s.length() && !map.containsKey(s.substring(k, k + 1)))
                k++;
            if (k == s.length())
                break;
            while (j < k) {
                if (map.containsKey(s.substring(j, j + 1))) {
                    if (i - j > maxLen) {
                        maxLen = i - j;
                        start = j;
                    }
                    j++;
                } else {
                    j = k;
                    break;
                }
            }
            k++;
        }

        System.out.println(maxLen + " " + start);
    }
}
