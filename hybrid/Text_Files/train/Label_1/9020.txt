#include <iostream>

long long solve(long long k, long long n, long long a, long long b) {
    if (a - b > k || (n - 1) * (a - b) > k)
        return -1;
    return (k - (n - 1) * (a - b)) / a;
}

int main() {
    long long q;
    std::cin >> q;
    while (q--) {
        long long k, n, a, b;
        std::cin >> k >> n >> a >> b;
        std::cout << solve(k, n, a, b) << "\n";
    }
    return 0;
}
