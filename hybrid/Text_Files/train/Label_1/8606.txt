#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n; vector<int>y(n);
    for (auto&i:y) cin>>i;
    sort(y.begin(), y.end());
    for (int i = 0; i < n; i++) {
        int x= y[i];
        while (x>1) {
            if ((x&1)==0) x>>=1;
            else if ((x-1)&1) x= (x-1)>>1;
            else x=(x-2)>>1;
        }
        y[i]=x;
    }
    for (auto&i:y) cout<<i<<' '; cout<<endl;
    return 0;
}
