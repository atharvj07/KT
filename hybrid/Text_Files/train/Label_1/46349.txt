import java.util.Scanner;

public class RyoukosMemoryNote {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        long[] prefixSum = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            prefixSum[i] = prefixSum[i - 1] + i;
        }

        long minPagesTurned = Long.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            long leftSum = 0;
            for (int j = 0; j < m; j++) {
                if (a[j] < i) {
                    leftSum += a[j];
                } else {
                    break;
                }
            }
            long rightSum = 0;
            for (int j = m - 1; j >= 0; j--) {
                if (a[j] > i) {
                    rightSum += n - a[j] + 1;
                } else {
                    break;
                }
            }
            minPagesTurned = Math.min(minPagesTurned, leftSum + prefixSum[n] - prefixSum[i - 1] + rightSum);
        }

        System.out.println(minPagesTurned);
    }
}
