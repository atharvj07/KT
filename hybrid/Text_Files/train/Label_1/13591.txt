#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        bool flag = true;
        map<ll, ll> freq;
        for (int i = 0; i < n; i++) {
            ll x;
            cin >> x;
            freq[x]++;
            if (freq[x] > 2) {
                flag = false;
            }
        }
        int sum = 0;
        for (auto it : freq) {
            sum += it.second;
        }
        if (sum % 2 == 1) {
            flag = false;
        }
        if (flag) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
