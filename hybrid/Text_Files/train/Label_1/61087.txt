t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    
    possible_divisions = []
    count_left = [0] * (n + 1)
    count_right = [0] * (n + 1)
    max_left = 0
    max_right = 0
    
    # Count frequencies on the right (full array initially)
    for val in a:
        count_right[val] += 1
        max_right = max(max_right, val)
    
    valid = True
    for i in range(n - 1):
        # Move one element from right to left
        val = a[i]
        count_left[val] += 1
        count_right[val] -= 1
        max_left = max(max_left, val)
        if count_left[val] > 1:  # If any number appears more than once on the left, it's invalid
            valid = False
            break
        if max_left == i + 1:  # Check if left part can form a permutation
            right_part_is_valid = True
            for j in range(1, n - i):
                if count_right[j] != 1:
                    right_part_is_valid = False
                    break
            if right_part_is_valid:
                possible_divisions.append((i + 1, n - i - 1))
    
    print(len(possible_divisions))
    for division in possible_divisions:
        print(division[0], division[1])
