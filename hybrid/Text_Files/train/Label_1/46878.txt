import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        String S = sc.next();
        if (S.length() == A + B + 1 && S.charAt(A) == '-' && S.substring(0, A).matches("[0-9]+") && S.substring(A + 1, A + B + 1).matches("[0-9]+")) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
