def swap(n, i, j):
        n = list(str(n))
        n[i - 1], n[j - 1] = n[j - 1], n[i - 1]
        return int(''.join(n))

n = int(input())
for i in range(1, len(str(n))):
        for j in range(i + 1, len(str(n)) + 1):
                if (n % 2 == 1 and swap(n, i, j) % 2 == 0):
                        if (swap(n, i, j) > n):
                                n = swap(n, i, j)
if (n % 2 == 1):
        print(-1)
elif (n % 2 == 0):
        print(n)
