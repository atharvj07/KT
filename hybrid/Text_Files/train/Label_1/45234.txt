import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        StringBuilder sb = new StringBuilder(sc.next());
        while (k-- > 0) {
            int i = sb.indexOf("47");
            if (i != -1) {
                sb.setCharAt(i, '4');
                sb.setCharAt(i + 1, '4');
            } else {
                i = sb.indexOf("74");
                if (i != -1) {
                    sb.setCharAt(i, '7');
                    sb.setCharAt(i + 1, '7');
                }
            }
        }
        System.out.print(sb);
    }
}
