import sys
input=lambda: sys.stdin.readline().strip()
for _ in range(int(input())):
    x, y = map(int, input().split())
    c = list(map(int, input().split()))
    if x == 0 and y == 0:
        print(0)
        continue
    if abs(x) == abs(y):
        if x < 0:
            x = -x
            y = -y
        print(c[(x+y)//2 - 1])
        continue
    if x == -y:
        if x > 0:
            x = -x
            y = -y
        print(c[6 - (x+y)//2 - 1])
        continue
    print(-1)
