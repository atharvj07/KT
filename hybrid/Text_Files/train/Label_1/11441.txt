#include <bits/stdc++.h>
#define endl '\n'

using namespace std;

vector<int> g[200005];
int vis[200005], deg[200005], ans;

void dfs(int u, int p = 1){
    vis[u] = true;
    deg[u] = deg[p] + 1;
    for(auto v : g[u]){
        if(!vis[v]){
            dfs(v, u);
            if(deg[v] > deg[u]) ans++;
        }
    }
}

int main(){
    int n, m; cin >> n >> m;
    for(int i = 1; i <= m; i++){
        int u, v; cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1);
    cout << ans << endl;
    return 0;
}
