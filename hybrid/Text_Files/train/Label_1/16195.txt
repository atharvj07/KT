#include <bits/stdc++.h>
using namespace std;
#define int long long
const int MOD = 1e9 + 7;
int32_t main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(n);
  for (auto &i : a) cin >> i;
  for (auto &i : b) cin >> i;
  unordered_map<int, int> mp;
  for (int i = 0; i < n; i++) {
    int key = (b[i] - a[i] + m) % m;
    if (mp.find(key) != mp.end()) {
      mp[key]++;
    } else
      mp[key] = 0;
  }
  int ans = MOD;
  for (auto it : mp) {
    if (it.second == n - 1) { ans = (it.first) % m; }
  }
  cout << ans << "\n";
}
