#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e5 + 5;

int n, k, a[maxn], b[maxn], cnt[maxn], sum[maxn], ans;

int main() {
    cin >> n >> k;
    for (int i = 1; i <= k; ++i) {
        cin >> a[i] >> b[i];
        ++cnt[a[i]], ++cnt[b[i]];
    }
    for (int i = 1; i <= n; ++i) {
        sum[i] = sum[i - 1] + cnt[i];
    }
    for (int i = 1; i <= k; ++i) {
        if (sum[a[i]] == 1) ++ans;
        if (sum[b[i]] == 1) ++ans;
    }
    cout << ans << endl;
}
