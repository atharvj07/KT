import java.util.*;
import java.math.*;

public class Main {
    static Pair[] dp;
    static int P;
    static int N;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        int M = sc.nextInt();
        int[] A = new int[N+1];
        int maxVal = 0;
        for(int i=1; i<=N; i++){
            A[i] = sc.nextInt();
            maxVal = Math.max(A[i], maxVal);
        }
        int MAX = 0;
        while(maxVal > 0){
            maxVal >>= 1;
            MAX++;
        }
        P = MAX;
        dp = new Pair[N+1];
        for(int i=0; i<=N; i++) dp[i] = new Pair(0,0);

        for(int i=1; i<=N; i++){
            dp[i] = dp[i-1].clone();


            for(int j=P-1; j>=0; j--){
                int now = dp[i].v;
                now = Math.max(now, dp[i-1].u + (A[i]+(1<<j)-1)/ (1<<j));
                dp[i].v = Math.max(dp[i].v, now);
                if(j == 0){
                    dp[i].u = dp[i].v;
                }else{
                    dp[i].u = Math.max(dp[i].u, dp[i].v);
                }
            }

        }
        System.out.println(dp[N].v);
    }
}

class Pair{
    int u,v;
    public Pair(int u, int v){
        this.u = u;
        this.v = v;
    }
    public Pair clone(){
        return new Pair(this.u, this.v);
    }
}
