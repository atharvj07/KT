import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        boolean[] reachable = new boolean[n];
        reachable[0] = true;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (reachable[j] && d[j] >= i - j) {
                    reachable[i] = true;
                    break;
                }
            }
        }
        System.out.println(reachable[n - 1] ? "yes" : "no");
    }
}
