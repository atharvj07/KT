import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            double sq = Math.sqrt(arr[i]);
            if (sq - (int) sq != 0) {
                max = Math.max(max, arr[i]);
            }
        }
        System.out.println(max);
    }
}
