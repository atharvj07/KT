#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
typedef vector<int> vi;
typedef vector<pi> vpi;

void print_result(int n, vi a, vi b) {
  cout << n << '\n';
  for (int i = 0; i < n; i++) {
    cout << a[i] << " " << b[i] << '\n';
  }
}

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  vi a, b;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      a.push_back(i + 1);
      b.push_back(i + 1);
    }
  }
  if (a.size() % 2) {
    cout << -1 << '\n';
    return 0;
  }
  print_result(a.size() / 2, a, b);
  return 0;
}
