#include <iostream>
#include <string>
using namespace std;
 
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.length();
    int cntA = 0, cntB = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == 'a') cntA++;
      if (s[i] == 'b') cntB++;
    }
    int ans = abs(cntA - cntB);
    for (int i = 0; i < n; i++) {
      if (cntA > cntB) {
        if (s[i] == 'a') {
          s[i] = 'b';
          ans--;
        }
      } else if (cntA < cntB) {
        if (s[i] == 'b') {
          s[i] = 'a';
          ans--;
        }
      }
      if (ans == 0) break;
    }
    cout << s << '\n';
  }
  return 0;
}
