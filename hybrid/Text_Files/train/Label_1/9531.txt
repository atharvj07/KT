#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<vector<int>> a(n, vector<int>(m));
  for (auto &row : a) {
    for (auto &x : row) {
      cin >> x;
    }
  }

  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (!a[i][j]) {
        if (i == 0 || a[i - 1][j]) ans++;
        if (i == n - 1 || a[i + 1][j]) ans++;
        if (j == 0 || a[i][j - 1]) ans++;
        if (j == m - 1 || a[i][j + 1]) ans++;
      }
    }
  }

  cout << ans << endl;

  return 0;
}
