import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            if (n == 1 || (n % 3 == 0 && n % 2 == 0))
                System.out.println(1);
            else if (n % 6 == 0)
                System.out.println(2);
            else
                System.out.println(-1);
        }
        sc.close();
    }
}
