import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Trees {
    static final int MODULO = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int k = Integer.parseInt(line[1]);

        long[][] dp = new long[n + 1][k + 1];

        // Base cases
        dp[1][0] = 1;
        dp[1][1] = 0;

        // Iterate over the number of vertices
        for (int i = 2; i <= n; i++) {
            // Iterate over the size of the maximum matching
            for (int j = 0; j <= k; j++) {
                // Calculate the number of trees for the current vertex
                dp[i][j] = (dp[i - 1][j] * (i - 2)) % MODULO;

                // If the size of the maximum matching is greater than 0,
                // add the number of trees with a matching edge from the previous vertex
                if (j > 0) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - 1] * (i - 1)) % MODULO;
                }
            }
        }

        System.out.println(dp[n][k]);
    }
}
