#include <bits/stdc++.h>
using namespace std;

const int N = 1e2 + 5;
int n, m;
int G[N][N];
int visR[N], visC[N];
vector<string> ans;
void addRow(int x) { visR[x] = 1; ans.push_back("row " + to_string(x)); }
void addCol(int y) { visC[y] = 1; ans.push_back("col " + to_string(y)); }

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> G[i][j];
        }
    }
    bool ok = true;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            bool found = false;
            if (i > 1) {
                if (G[i][j] < G[i - 1][j]) found = true;
            }
            if (j > 1) {
                if (G[i][j] < G[i][j - 1]) found = true;
            }
            if (!found) { ok = false; break; }
        }
        if (!ok) break;
    }
    if (!ok) {
        cout << -1 << endl;
        return 0;
    }
    int k = 0;
    for (int i = 1; i <= n; i++) {
        int cur = G[i][1];
        for (int j = 1; j <= m; j++) {
            if (G[i][j] != cur) {
                if (visC[j]) {
                    ok = false;
                    break;
                } else
                    addCol(j);
                k++;
            }
            cur = G[i][j];
        }
        if (!ok) break;
    }
    for (int j = 1; j <= m; j++) {
        int cur = G[1][j];
        for (int i = 1; i <= n; i++) {
            if (G[i][j] != cur) {
                if (visR[i]) {
                    ok = false;
                    break;
                } else
                    addRow(i);
                k++;
            }
            cur = G[i][j];
        }
        if (!ok) break;
    }
    cout << k << endl;
    for (string x : ans) cout << x << '\n';
    return 0;
}
