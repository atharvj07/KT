#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; i++) { cin >> p[i]; }
    vector<int> ans(n + 1);
    for (int i = 1; i <= n; i++) {
      if (!ans[i]) {
        int j = i;
        int cnt = 0;
        while (!ans[j]) {
          ans[j] = ++cnt;
          j = p[j];
        }
      }
    }
    for (int i = 1; i <= n; i++) { cout << ans[i] << ' '; }
    cout << '\n';
  }
  return 0;
}
