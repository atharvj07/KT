def min_draws_needed(tiles):
    # Count occurrences of each tile
    counts = {}
    for tile in tiles:
        if tile in counts:
            counts[tile] += 1
        else:
            counts[tile] = 1

    # Check for koutsu
    for count in counts.values():
        if count >= 3:  # If any tile occurs thrice, it's a koutsu, no draw needed
            return 0

    # Check for shuntsu or potential shuntsu
    suits = ['m', 'p', 's']
    for suit in suits:
        nums = [int(tile[0]) for tile in tiles if tile.endswith(suit)]
        nums.sort()

        # Check for existing shuntsu
        if len(nums) == 3 and nums[2] - nums[0] == 2:  # Consecutive numbers form a shuntsu
            return 0

        # Check for potential shuntsu with one draw
        if len(nums) == 3 and (nums[2] - nums[1] <= 2 or nums[1] - nums[0] <= 2):
            return 1

    # Check for potential koutsu or shuntsu with one or two draws
    if len(set(tiles)) < 3:  # If any duplicate, one draw might create koutsu
        return 1
    else:  # Otherwise, two draws are needed for either a koutsu or shuntsu
        return 2

tiles = input().split()
print(min_draws_needed(tiles))
