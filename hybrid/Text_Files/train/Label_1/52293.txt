import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Set<Integer> x = new HashSet<>();
        int p = in.nextInt();
        while (p-- > 0) {
            x.add(in.nextInt());
        }
        int q = in.nextInt();
        while (q-- > 0) {
            x.add(in.nextInt());
        }
        in.close();
        System.out.println(x.size() == n ? "I become the guy." : "Oh, my keyboard!");
    }
}
