n, r = map(int, input().split())
a = list(map(int, input().split()))
ans = 0
i = 0
while i < n:
    if a[i] == 1:
        ans += 1
        i += r
    else:
        j = i + 1
        skip = True
        while j < n and j < i + r:
            if a[j] == 1:
                skip = False
                break
            j += 1
        if skip:
            return -1
        i = j
print(ans)
