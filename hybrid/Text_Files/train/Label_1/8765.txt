#include <bits/stdc++.h>
#define mod 1000000007
using namespace std;
int n,k,bad,dp[1001][1001];
void solve(int pos,int ke){
    if(dp[pos][ke]!=-1)return;
    if(pos == n){
        if(ke == bad)dp[pos][ke]=1;
        else dp[pos][ke]=0;
        return;
    }
    dp[pos][ke]=0;
    if(ke < bad){
        solve(pos+1,ke+1);
        dp[pos][ke]=(dp[pos][ke]+dp[pos+1][ke+1]%mod)%mod;
    }
    solve(pos+1,ke);
    dp[pos][ke]=(dp[pos][ke]+dp[pos+1][ke]%mod)%mod;
}
int main(){
    cin>>n>>k;
    bad=n-k-1;
    if(bad<0)bad=0;
    memset(dp,-1,sizeof(dp));
    solve(1,0);
    cout<<dp[1][0]%mod<<endl;
}
