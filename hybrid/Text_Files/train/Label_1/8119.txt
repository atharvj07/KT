#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        for (int i = 0; i < n; i++) {
            if (s[i] == '?') {
                if (i > 0 && s[i - 1] == s[i + 1]) {
                    if (s[i - 1] == 'B') {
                        s[i] = 'R';
                    } else {
                        s[i] = 'B';
                    }
                } else {
                    s[i] = s[i - 1] == 'B' ? 'R' : 'B';
                }
            }
        }
        cout << s << '\n';
    }
    return 0;
}
