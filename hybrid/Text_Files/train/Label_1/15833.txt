#include <iostream>

using namespace std;

int main() {
  string time;
  int a;
  cin >> time >> a;
  int h = stoi(time.substr(0, 2));
  int m = stoi(time.substr(3, 2));

  m += a;
  
  h += m / 60;
  m %= 60;

  h %= 24;
  
  cout << (h < 10 ? "0" : "") << h << ":" << (m < 10 ? "0" : "") << m << endl;

  return 0;
}
