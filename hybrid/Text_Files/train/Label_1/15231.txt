#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int n;
vector<pair<int, int>> adj[101];
int vis[101];
int ans;

void dfs(int u, int p, int w) {
  vis[u] = true;
  for (auto& edge : adj[u]) {
    int v = edge.first;
    int wt = edge.second;
    if (v == p) {
      continue;
    }
    if (!vis[v]) {
      dfs(v, u, wt);
      ans = max(ans, w + wt);
    }
  }
}

int main() {
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    adj[u].push_back({v, w});
    adj[v].push_back({u, w});
  }
  dfs(0, -1, 0);
  cout << ans << endl;
  return 0;
}
