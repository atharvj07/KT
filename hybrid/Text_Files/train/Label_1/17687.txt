#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll dp[1000005][4]; // dp[i][0] = last -1, dp[i][1] = last 0, dp[i][2] = last 1, dp[i][3] = last 2
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 4; j++) dp[i][j] = 1e18;
  }
  dp[0][1] = 0;
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < 4; j++) {
      if (i == 1) dp[i][j] = abs(a[i - 1] - a[i]);
      dp[i][j] = min(dp[i][j], dp[i - 1][0] + abs(a[i - 1] - 1 - a[i]));
      dp[i][j] = min(dp[i][j], dp[i - 1][1] + abs(a[i - 1] - a[i]));
      dp[i][j] = min(dp[i][j], dp[i - 1][2] + abs(a[i - 1] + 1 - a[i]));
      dp[i][j] = min(dp[i][j], dp[i - 1][3] + abs(a[i - 1] + 2 - a[i]));
    }
    if (a[i] > 2) dp[i][3] = min(dp[i][3], dp[i - 1][2] + a[i] - 2);
  }
  ll ans = 1e18;
  for (int i = 0; i < 4; i++) ans = min(ans, dp[n - 1][i]);
  cout << ans << endl;
  return 0;
}
