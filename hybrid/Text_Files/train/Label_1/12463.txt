#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> x(n), y(n);
    for (int i = 0; i < n; i++) cin >> x[i];
    for (int i = 0; i < n; i++) cin >> y[i];
    int cntx = 0, cnty = 0;
    for (int i = 0; i < n; i++) {
        cntx += x[i];
        cnty += y[i];
        if (cntx > k || cnty > k) {
            cout << "NO" << endl;
            return 0;
        }
        if (i < n - 1 && x[i + 1] > k - cntx) {
            cout << "NO" << endl;
            return 0;
        }
        if (i < n - 1 && y[i + 1] > k - cnty) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    return 0;
}
