from collections import deque
import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    points = [list(map(int, input().split())) for _ in range(n)]
    w = list(map(int, input().split()))

    graph = [[] for _ in range(100001)]
    for p in points:
        graph[p[0] + p[1]].append(p)

    vis = [False for _ in range(100001)]
    q = deque([0])
    ans = []

    while q:
        cur = q.popleft()

        if vis[cur] == True:
            continue

        vis[cur] = True

        for p in graph[cur]:
            if p[0] + p[1] > cur:
                q.append(p[0] + p[1])

            if p[1] - p[0] == w[ans.count(p)]:
                ans.append(p)

    if len(ans) == n:
        print("YES")
        for p in ans:
            print(p[0], p[1])
    else:
        print("NO")

solve()
