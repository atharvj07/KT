#include<iostream>
#define LL long long
using namespace std;
int main(){
    int a,b,h,w,n;
    cin>>a>>b>>h>>w>>n;
    LL t1=h,t2=w,x;
    int ans=-1;
    for (int i =0; i<n; i++){
        cin>>x;
        if (x>=a && h<a && w<b){
            t1*=x;
            if (t1>=a && t2>=b) {ans=i+1; break;}
        }
        if (x>=b && h<b && w<a){
            t2*=x;
            if (t1>=b && t2>=a) {ans=i+1; break;}
        }
    }
    if (h>=a && w>=b) ans=0;
    cout<<ans<<endl;
    return 0;
}
