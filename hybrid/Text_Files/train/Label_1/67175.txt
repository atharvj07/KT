import sys
n, m = map(int, input().split())

a = list(map(int, input().split()))
a.sort()

need = n // 2 if n % 4 == 0 else n // 2 - 1

ev = 0
for i in range(n - 1, -1, -1):
  if a[i] % 2 == 0:
    ev += 1

res = []
ex = []

if need <= ev:
  for i in range(n):
    if a[i] % 2 == 0 and need > 0:
      res += [m]
      ex += [a[i]]
      need -= 1
    else:
      res += [a[i]]
elif m >= n - need:
  for i in range(n):
    if a[i] % 2 == 0:
      res += [a[i]]
    else:
      res += [m]
      ex += [a[i]]
      m -= 1
else:
  res = [-1]

print(len(ex))
print(*res)
