#include <bits/stdc++.h>

using namespace std;

vector<int> a, perm;

int main() {
  int n, q, m;
  cin >> n >> q >> m;
  a.resize(n + 1);
  perm.resize(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    perm[i] = i;
  }
  while (q--) {
    int t, l, r;
    cin >> t >> l >> r;
    if (t == 1) {
      for (int i = 0; i < r - l + 1; ++i) {
        swap(perm[l + i], perm[r - i]);
      }
    } else {
      reverse(perm.begin() + l, perm.begin() + r + 1);
    }
  }
  for (int i = 0; i < m; ++i) {
    int b;
    cin >> b;
    cout << a[perm[b]] << ' ';
  }
  cout << '\n';
}
