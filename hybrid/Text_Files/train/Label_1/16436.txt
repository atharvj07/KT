#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    long long n, ans = 0, cnt = 0, mn = LLONG_MAX;
    cin >> n;
    for (long long i = 0; i < n; i++) {
        long long tmp;
        cin >> tmp;
        ans += tmp / 2;
        tmp -= tmp / 2;
        if (tmp == 1) cnt += 1;
        mn = min(mn, tmp);
    }
    cout << ans + min(cnt, mn) << endl;
}
