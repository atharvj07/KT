#include<bits/stdc++.h>
using namespace std;
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> cnt(n + 1);
    int pos = 0;
    for (int i = 0; i < n; i++) {
      cnt[a[i]]++;
      if (a[i] == i + 1) pos++;
    }
    if (pos >= k) {
      cout << 0 << endl;
      continue;
    }
    int ans = INT_MAX;
    int cur = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == i + 1) {
        continue;
      }
      if (cnt[i + 1] > 0) {
        ans = min(ans, cur + 1);
      } else {
        cur++;
      }
      cnt[a[i]]--;
      pos--;
      if (pos >= k) {
        ans = min(ans, cur);
      }
    }
    if (ans == INT_MAX)
      ans = -1;
    cout << ans << endl;
  }
  return 0;
}
