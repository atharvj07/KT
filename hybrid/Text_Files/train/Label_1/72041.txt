for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    for i in range(n):
        for j in range(i+1, n):
            for k in range(j+1, n):
                if a[i] + a[j] <= a[k]:
                    print(i+1, j+1, k+1)
                    break
            else:
                continue
            break
    else:
        print(-1)
