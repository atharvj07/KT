import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] board = new int[14];

        // Read the initial state of the board
        for (int i = 0; i < 14; i++) {
            board[i] = scanner.nextInt();
        }

        // Find the hole with the maximum number of stones
        int maxStones = 0;
        int maxHole = 0;
        for (int i = 0; i < 14; i++) {
            if (board[i] > maxStones && board[i] % 2 == 1) {
                maxStones = board[i];
                maxHole = i;
            }
        }

        // Redistribute the stones and collect the even numbers
        int score = 0;
        board[maxHole] = 0;
        int currentHole = maxHole;
        for (int i = 0; i < maxStones; i++) {
            currentHole = (currentHole + 1) % 14;
            board[currentHole]++;
            if (board[currentHole] % 2 == 0) {
                score += board[currentHole];
                board[currentHole] = 0;
            }
        }

        // Print the maximum score
        System.out.println(score);
    }
}
