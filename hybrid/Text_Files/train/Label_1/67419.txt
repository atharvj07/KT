n, m, k, q = map(int, input().split())
a = [[-1 for j in range(m + 1)] for i in range(n + 1)]
ans = 10**9
for i in range(q):
    x, y, t = map(int, input().split())
    a[x][y] = t
    for ii in range(x - k + 1, x + 1):
        for jj in range(y - k + 1, y + 1):
            if ii > 0 and jj > 0 and a[ii][jj] == -1:
                break
            if a[ii][jj] != -1:
                ans = min(ans, a[ii][jj])
if ans < 10**9:
    print(ans)
else:
    print(-1)
