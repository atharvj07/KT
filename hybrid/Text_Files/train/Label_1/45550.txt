import java.util.*;
import java.math.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] arr = new int[m + 1];
        for (int i = 0; i < n; i++) {
            arr[in.nextInt()]++;
        }
        int ans = 0;
        for (int i = 1; i <= m; i++) {
            ans += arr[i] / 3;
            arr[i] %= 3;
        }
        int[] cnt = new int[3];
        for (int i = 1; i <= m; i++) {
            cnt[arr[i]]++;
        }
        ans += Math.min(cnt[1], cnt[2]);
        System.out.println(ans);
    }
}
