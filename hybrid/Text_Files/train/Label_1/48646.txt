import java.util.Scanner;

public class WormEvolution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] lengths = new int[n];

        for (int i = 0; i < n; i++) {
            lengths[i] = scanner.nextInt();
        }

        // Sort the lengths in descending order
        sort(lengths);

        // Check if there are three lengths that satisfy the condition
        for (int i = 0; i < n - 2; i++) {
            int j = i + 1;
            int k = n - 1;

            while (j < k) {
                int sum = lengths[j] + lengths[k];

                if (sum == lengths[i]) {
                    System.out.println(i + 1 + " " + (j + 1) + " " + (k + 1));
                    return;
                } else if (sum < lengths[i]) {
                    j++;
                } else {
                    k--;
                }
            }
        }

        // If no solution is found, output -1
        System.out.println(-1);
    }

    // Sort the array in descending order
    private static void sort(int[] arr) {
        for (int i = 0; i < arr.length - 1; i++) {
            for (int j = i + 1; j < arr.length; j++) {
                if (arr[i] < arr[j]) {
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = temp;
                }
            }
        }
    }
}
