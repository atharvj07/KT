def count_inconvenient_pairs(n, m, k, vertical_streets, horizontal_streets, people):
    vertical_streets.sort()
    horizontal_streets.sort()
    
    # Create a dictionary to hold vertical and horizontal lines with their people
    vertical_lines = {}
    horizontal_lines = {}
    
    for x, y in people:
        if x in vertical_lines:
            vertical_lines[x].append(y)
        else:
            vertical_lines[x] = [y]
            
        if y in horizontal_lines:
            horizontal_lines[y].append(x)
        else:
            horizontal_lines[y] = [x]
    
    def count_pairs(lines):
        count = 0
        for line in lines.values():
            line.sort()
            for i in range(len(line)):
                for j in range(i + 1, len(line)):
                    if abs(line[i] - line[j]) < j - i:
                        count += 1
        return count
    
    return count_pairs(vertical_lines) + count_pairs(horizontal_lines)

t = int(input().strip())
for _ in range(t):
    n, m, k = map(int, input().strip().split())
    vertical_streets = list(map(int, input().strip().split()))
    horizontal_streets = list(map(int, input().strip().split()))
    people = [tuple(map(int, input().strip().split())) for _ in range(k)]
    
    print(count_inconvenient_pairs(n, m, k, vertical_streets, horizontal_streets, people))
