#include <cstdio>
#include <vector>

int main()
{
	int n, q;
	scanf("%d %d", &n, &q);
	std::vector<int> lastRead(n);
	int unread = 0;
	while (q--)
	{
		int type, x;
		scanf("%d", &type);
		if (type <= 2)
			scanf("%d", &x), --x;
		else
		{
			int t;
			scanf("%d", &t);
		}
		switch (type)
		{
		case 1:
			++unread;
			lastRead[x] = q;
			break;
		case 2:
			unread -= q - lastRead[x];
			break;
		case 3:
			unread = 0;
			break;
		}
		printf("%d\n", unread);
	}
	return 0;
}
