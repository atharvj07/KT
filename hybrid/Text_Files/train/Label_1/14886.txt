#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<int> height(n + 1);
    int points = 0;
    for (int i = 0; i < m; i++) {
        int c;
        std::cin >> c;
        height[c]++;
        if (height[c] == 1) {
            points++;
            for (int j = 1; j <= n; j++) {
                height[j]--;
            }
        }
    }
    std::cout << points << std::endl;
    return 0;
}
