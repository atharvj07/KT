#include <iostream>
#include <deque>

using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    deque<int> dq;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        dq.push_back(x);
    }
    while (q--) {
        int m;
        cin >> m;
        for (int i = 0; i < m % (n - 1); i++) {
            int a = dq.front();
            dq.pop_front();
            int b = dq.front();
            dq.pop_front();
            if (a > b) {
                dq.push_front(b);
                dq.push_back(a);
            } else {
                dq.push_front(a);
                dq.push_back(b);
            }
        }
        int a = dq.front();
        dq.pop_front();
        int b = dq.front();
        dq.pop_front();
        cout << a << " " << b << endl;
    }
    return 0;
}
