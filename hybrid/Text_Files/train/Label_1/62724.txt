n=int(input());a=[*map(int,input().split())];up=down=[-1]*n;up[0]=0
a.sort()
for i in range(1,n):
	if a[i]!=a[i-1]:
		up[i]=up[i-1]+1
	else:
		up[i]=up[i-1]
for i in range(n-2,-1,-1):
	if a[i]!=a[i+1]:
		down[i]=down[i+1]+1
	else:
		down[i]=down[i+1]
ans=max(up[i]+down[i+1] if i+1<n else up[i] for i in range(n))
print(ans+1)
