#include<iostream>
using namespace std;
int main()
{
    long long a,b;
    cin>>a>>b;
    int o=0;
    while(a<b)
    {
        o++;
        if(b%10==1){
            b/=10;
        }
        else{
            if(b%2==0)
            {
                b/=2;
            }
            else
            {
                cout<<"NO";
                return 0;
            }
        }
    }
    if(a!=b)
    {
        cout<<"NO";
        return 0;
    }
    cout<<"YES"<<endl;
    cout<<o<<endl;
    while(a!=b)
    {
        if(b%10==1)
        {
            cout<<b*10+1<<" ";
            b/=10;
        }
        else
        {
            cout<<b*2<<" ";
            b/=2;
        }
    }
    cout<<b;
    return 0;
}
