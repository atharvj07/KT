from collections import Counter
tiles = input().split()
cnt = Counter(tiles)
ans = 0
if cnt[tiles[0]] == 3:
    ans = 0
elif cnt[tiles[0]] == 2:
    del cnt[tiles[0]]
    tiles = list(cnt.keys())
    if tiles[0][1] == tiles[1][1]:
        ans = 1
    else:
        ans = 2
else:
    tiles = list(cnt.keys())
    if len(tiles) == 1:
        ans = 2
    elif tiles[0][1] == tiles[1][1]:
        ans = 1
    else:
        ans = 2
print(ans)
