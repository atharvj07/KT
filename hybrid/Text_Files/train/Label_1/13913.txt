#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    while (n > 1 && *min_element(a.begin(), a.end() - 1) < *(a.end() - 1)) {
        int i = 0;
        while (a[i] <= a[i + 1] && i + 1 < n) ++i;
        long long sum = accumulate(a.begin(), a.begin() + i + 1, 0LL);
        a.resize(i + 1);
        a[i] = sum / (i + 1);
        n = i + 1;
    }
    for (int i = 0; i < a.size(); i++) { cout << a[i] << '\n'; }
}
