#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    pair<double, double> p[4];
    for (int i = 0; i < 4; i++) cin >> p[i].first >> p[i].second;
    pair<double, double> q[4];
    for (int i = 0; i < 4; i++) cin >> q[i].first >> q[i].second;
    sort(p, p + 4);
    sort(q, q + 4);
    if (max(p[0].first, q[0].first) < min(p[2].first, q[2].first) &&
        max(p[0].second, q[0].second) < min(p[2].second, q[2].second))
        cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
