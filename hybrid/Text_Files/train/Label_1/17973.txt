#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = (int)1e5 + 5;
int par[N], sz[N];
void init(int n) {
    for (int i = 0; i < n; i++) par[i] = i, sz[i] = 1;
}
int find(int u) {
    if (u == par[u]) return u;
    return par[u] = find(par[u]);
}
void merge(int a, int b) {
    a = find(a), b = find(b);
    if (a != b) {
        if (sz[a] > sz[b]) swap(a, b);
        par[a] = b, sz[b] += sz[a];
    }
}
bool same(int a, int b) {
    return find(a) == find(b);
}
int main() {
    ll n, m;
    cin >> n >> m;
    init(n);
    map<pair<ll, ll>, bool> mp;
    while (m--) {
        ll u, v;
        cin >> u >> v;
        if (u > v) swap(u, v);
        mp[{u, v}] = true;
    }
    ll ans = 1;
    for (ll i = 1; i <= n; i++) {
        for (ll j = i + 1; j <= n; j++) {
            if (mp[{i, j}] == false && !same(i, j)) {
                merge(i, j);
                ans *= 2;
            }
        }
    }
    cout << ans % 1000000007 << '\n';
}
