#include <iostream>

using namespace std;

int main() {
  int y;
  cin >> y;
  for (++y; y < 100000; ++y) {
    if ((y % 4 == 0 && y % 100 != 0) || y % 400 == 0) {
      if ((y + 1) % 4 == 0 && (y + 1) % 100 != 0 || (y + 1) % 400 == 0) {
        cout << y + 1 << endl;
        return 0;
      }
    } else {
      if ((y + 1) % 4 == 0 && (y + 1) % 100 != 0 || (y + 1) % 400 == 0) {
        cout << y + 2 << endl;
        return 0;
      }
    }
  }
  return 0;
}
