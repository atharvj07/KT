#include<bits/stdc++.h>
using namespace std;
#define sqm(x) ((x)*(x))
#define F first
#define S second
#define MAX_N 200010

vector<bool> visited(MAX_N,false);
vector<int> adj[MAX_N];

void solve()
{
    int n,k;
    cin>>n>>k;
    vector<int> pos(n+1,-1);
    for(int i=0,a;i<k;i++)
    {
        cin>>a;
        pos[a]=i;
    }
    for(int i=1;i<n;i++)
    {
        int a,b;
        cin>>a>>b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    int cur=1,dep=0;
    while(!visited[cur])
    {
        visited[cur]=true;
        if(pos[cur]!=-1 && pos[cur]!=dep)
        {
            cout<<"NO"<<endl;
            return;
        }
        dep++;
        for(auto i:adj[cur])
        {
            if(!visited[i])
            {
                cur=i;break;
            }
        }
    }
    cout<<"YES"<<endl;
}

int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        for(int i=1;i<=200009;i++)
        {
            visited[i]=false;
            adj[i].clear();
        }
        solve();
    }
}
