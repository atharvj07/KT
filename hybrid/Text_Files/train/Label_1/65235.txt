import sys

def solve():
    n, m = map(int, input().split())
    cmp = [[] for i in range(n+1)]
    for i in range(m):
        a, b = map(int, input().split())
        cmp[a].append(b)
        cmp[b].append(a)
    vis = [False for i in range(n+1)]
    cycle = []
    def dfs(cur, prv):
        if vis[cur]:
            return True
        vis[cur] = True
        cycle.append(cur)
        for nxt in cmp[cur]:
            if nxt != prv and dfs(nxt, cur):
                return True
        cycle.pop()
        return False
    ans1 = []
    ans2 = []
    for i in range(1, n+1):
        if vis[i]:
            continue
        if dfs(i, i):
            print("YES")
            for j in cycle:
                if j not in ans1:
                    ans1.append(j)
                ans2.append(j)
            print(*ans1)
            print(*ans2)
            return
    print("NO")

solve()
