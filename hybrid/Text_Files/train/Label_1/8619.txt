#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = n;
    for (int i = 1; i < n; i++) {
        if (s.substr(0, i) == s.substr(i, i)) {
            ans = n / i + (n % i != 0);
        }
    }
    cout << ans << endl;
    return 0;
}
