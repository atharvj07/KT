import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner scan = new java.util.Scanner(System.in);
        int n = scan.nextInt();
        List<String> list = Arrays.stream(String.valueOf(n).split("")).sorted().collect(Collectors.toList());
        String res = "";
        for (String s : list)
            res += s;
        System.out.println(res.equals(String.valueOf(scan.nextInt())) ? "OK" : "WRONG_ANSWER");
    }
}
