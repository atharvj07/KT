import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        boolean[][] obstacle = new boolean[n][m];
        while (k-- > 0) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            obstacle[x][y] = true;
        }
        int[][] dp = new int[n][m];
        for (int i = 1; i < n; i++)
            for (int j = 1; j < m; j++) {
                if (obstacle[i][j]) {
                    dp[i][j] = -1;
                    continue;
                }
                int min = Integer.MAX_VALUE;
                if (dp[i - 1][j] != -1) min = Math.min(min, dp[i - 1][j] + 1);
                if (dp[i][j - 1] != -1) min = Math.min(min, dp[i][j - 1] + 1);
                dp[i][j] = (min == Integer.MAX_VALUE) ? -1 : min;
            }
        System.out.println(dp[n - 1][m - 1]);
    }
}
