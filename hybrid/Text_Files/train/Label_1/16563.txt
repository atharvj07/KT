#include<bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
int main() {
  ll n, x; cin >> n >> x; ll a[n], sum = 0, mx = 0, ans = LLONG_MAX;
  for(ll i = 0; i < n; i++) { cin >> a[i]; sum += (a[i] <= x); mx = max(mx, a[i]); }
  for(ll i = 0; i <= mx; i++) { ll cnt = sum; for(ll j = 0; j < n; j++) if(a[j] < i) cnt--;
    if(i - cnt == x) ans = min(ans, i - sum + cnt); }
  cout << ans << endl;
  return 0;
}
