import java.util.*;
import java.io.*;

public class Main {
    static class Edge {
        int to, ch;
        Edge next;
        public Edge(int to, int ch, Edge next) {
            this.to = to;
            this.ch = ch;
            this.next = next;
        }
    }

    static final int MAXN = 500005;
    static int n;
    static int[] a = new int[MAXN];
    static Edge[] head = new Edge[MAXN];
    static void addEdge(int x, int y, int ch) {
        head[x] = new Edge(y, ch, head[x]);
    }
    
    static int[] len = new int[MAXN];
    static int[] maxLen = new int[MAXN];
    static void dfs1(int x, int fa) {
        len[x] = maxLen[x] = 0;
        for (Edge e=head[x]; e!=null; e=e.next) {
            if (e.to == fa)   continue;
            dfs1(e.to, x);
            if (e.ch == a[x]) {
                len[x] = Math.max(len[x], len[e.to] + 1);
                maxLen[x] = Math.max(maxLen[x], maxLen[e.to] + 2);
            }
        }
    }

    static void dfs2(int x, int fa) {
        for (Edge e=head[x]; e!=null; e=e.next) {
            if (e.to == fa)   continue;
            if (maxLen[x] > maxLen[e.to]) {
                len[e.to] = len[x] + 1;
            }
            else if (e.ch == a[x] && len[x] > 0) {
                len[e.to] = len[x] + 1;
            }
            else {
                len[e.to] = 1;
            }
            maxLen[e.to] = Math.max(maxLen[e.to], maxLen[x]);
            if (len[x] > 1 && e.ch == a[x]) {
                maxLen[e.to] = Math.max(maxLen[e.to], len[e.to]);
            }
            dfs2(e.to, x);
        }
    }

    public static void main(String args[]) {
        Scanner cin = new Scanner(System.in);
        n = cin.nextInt();
        for (int i=1; i<n; i++) {
            int fa = cin.nextInt() + 1;
            char ch = cin.next().charAt(0);
            a[i+1] = ch - 'a' + 1;
            addEdge(i+1, fa, a[i+1]);
            addEdge(fa, i+1, a[i+1]);
        }
        dfs1(1, 1);
        dfs2(1, 1);
        for (int i=1; i<=n; i++) {
            System.out.print(maxLen[i] + " ");
        }
    }
}
