#include<bits/stdc++.h>
using namespace std;
int main() {
    long long n, l, r, mini, maxi, cur;
    cin >> n >> l >> r;
    if (n == l) {
        mini = n;
        maxi = n + (n - 1) * n / 2;
    } else {
        mini = 1 + n + (n - 1) * n / 2;
        maxi = n + (n - 1) * n / 2;
        cur = 3;
        while (cur <= r && n > 1) {
            maxi += n - 1;
            cur++;
            n /= 2;
        }
        maxi += cur + (n - 1);
    }
    cout << mini << ' ' << maxi << endl;
}
