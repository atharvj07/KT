#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl "\n"
int main()
{
  int n;
  cin>>n;
  ll sum=0;
  vector<ll> v(n);
  for(int i=0; i<n; i++){
    cin>>v[i];
    sum+=v[i];
  }
  if(sum%2==0){
    cout<<-1<<endl;
    return 0;
  }
  for(int i=0; i<n; i++){
    for(int j=i+1; j<n; j++){
      if(v[i]+v[j]==sum-v[i]-v[j]){
        cout<<2<<endl;
        cout<<i+1<<" "<<j+1<<endl;
        return 0;
      }
    }
  }
  cout<<-1<<endl;
  return 0;
}
