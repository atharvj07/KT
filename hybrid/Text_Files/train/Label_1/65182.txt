def find_minimum_swaps(p, s):
    n = len(p)
    pos = {i: p.index(i) for i in range(1, n + 1)}
    ans = 0
    res = []
    for i in range(n - 1, -1, -1):
        if s[i] == p[i]: continue
        j = pos[s[i]]
        ans += abs(i - j)
        res.append((i + 1, j + 1))
        p[i], p[j] = p[j], p[i]
        pos[p[i]] = i
        pos[p[j]] = j
    return ans, res

n = int(input())
p = [int(x) for x in input().split()]
s = [int(x) for x in input().split()]
ans, res = find_minimum_swaps(p, s)
print(ans)
print(len(res))
for r in res:
    print(*r)
