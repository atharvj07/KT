from heapq import heappush, heappop
k1, k2, k3 = map(int, input().split())
a1 = sorted(map(int, input().split()))
a2 = sorted(map(int, input().split()))
a3 = sorted(map(int, input().split()))
ans = 0
h = []
for x in a1: heappush(h, x)
for x in a3: heappush(h, x)
while k2:
    x = heappop(h)
    k2 -= 1
    if x in a1:
        if k2 == 0: break
        x = heappop(h)
    if x in a3: ans += 1
    a1.append(x)
    heappush(h, x)
print(ans)
