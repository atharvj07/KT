#include <bits/stdc++.h>

char a[6][9];

int main() {
  int x, y;
  for (int i = 0; i < 6; ++i) {
    std::cin >> a[i];
    for (int j = 0; j < 8; ++j) {
      if (a[i][j] == '.' && i > x) { x = i; y = j; }
    }
  }
  a[x][y] = 'P';
  for (int i = 0; i < 6; ++i) std::cout << a[i] << '\n';
  return 0;
}
