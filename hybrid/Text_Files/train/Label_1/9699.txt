#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> b(n + 1);
    for (int i = 1; i <= n; ++i) cin >> b[i];
    vector<int> dp(n + 1);
    dp[1] = b[1];
    for (int i = 2; i <= n; ++i) {
        dp[i] = b[i];
        for (int j = 1; j < i; ++j) {
            if (b[j] < b[i] && i - j == b[i] - b[j]) {
                dp[i] = max(dp[i], dp[j] + b[i]);
            }
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; ++i) ans = max(ans, dp[i]);
    cout << ans << endl;
    return 0;
}
