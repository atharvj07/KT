#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

int main() {
    int n, m; cin >> n >> m;
    if (n * m & 1) return puts("NO"), 0;
    puts("YES");
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            a[i][j] = (i + j) % 2 + (i * m + j + 1) / 2;
        }
    }
    for (auto &row : a) {
        for (auto &x : row) {
            cout << x << ' ';
        }
        cout << endl;
    }
}
