#include <iostream>
int main() {
  int n;
  std::string s;
  std::cin >> n >> s;
  int ac = 0, dc = 0;
  for (auto &c : s) {
    if (c == 'A')
      ac++;
    else
      dc++;
  }
  if (ac > dc)
    std::cout << "Anton";
  else if (dc > ac)
    std::cout << "Danik";
  else
    std::cout << "Friendship";
}
