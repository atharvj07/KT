#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> points;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        points.push_back({x, y});
    }
    sort(points.begin(), points.end());
    int min_x = points[0].first;
    int max_x = points[0].first;
    int min_y = points[0].second;
    int max_y = points[0].second;
    for (int i = 1; i < n; i++) {
        min_x = min(min_x, points[i].first);
        max_x = max(max_x, points[i].first);
        min_y = min(min_y, points[i].second);
        max_y = max(max_y, points[i].second);
    }
    cout << (max_x - min_x) * (max_y - min_y) << endl;
    return 0;
}
