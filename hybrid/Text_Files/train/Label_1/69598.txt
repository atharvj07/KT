n = int(input())
bombs = [tuple(map(int, input().split())) for _ in range(n)]

moves = []

x, y = 0, 0
for bx, by in bombs:
    if bx > x:
        moves.extend(['1 {} R'.format(bx - x), '1 {} U'.format(by - y), '2', '1 {} L'.format(bx - x), '1 {} D'.format(by - y), '3'])
        x, y = bx, by
    else:
        moves.extend(['1 {} L'.format(x - bx), '1 {} U'.format(by - y), '2', '1 {} R'.format(x - bx), '1 {} D'.format(by - y), '3'])
        x, y = bx, by

print(len(moves))
for move in moves:
    print(move)
