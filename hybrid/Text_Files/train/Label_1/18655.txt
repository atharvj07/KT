#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.length();
  if (n % 2)
    cout << -1 << endl;
  else {
    int balance = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == 'U')
        balance++;
      else if (s[i] == 'D')
        balance--;
    }
    if (balance != 0)
      cout << -1 << endl;
    else {
      int changes = 0;
      for (int i = 0; i < n; i++) {
        if (s[i] != 'U' && s[i] != 'D') {
          s[i] = 'U';
          changes++;
        }
      }
      for (int i = n - 1, j = 1; i >= 0; i -= 2, j += 2) {
        if (s[i] != s[j]) {
          s[i] = s[j];
          changes++;
        }
      }
      cout << changes << endl;
    }
  }
  return 0;
}
