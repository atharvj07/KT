import java.util.Arrays;

class Main {

    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (m > 0 && a[i] > 1) {
                m--;
                ans += a[i] / 2;
            } else {
                ans += a[i];
            }
        }
        System.out.println(ans);
    }
}
