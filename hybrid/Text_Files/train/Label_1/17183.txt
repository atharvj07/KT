#include <iostream>
#include <string>
#include <algorithm>
#include <vector>

std::vector<int> solve(std::string& s, int k) {
    int n = s.length();
    std::vector<int> ans(n);
    std::vector<int> now;
    for (int i = 0; i < n; i++) {
        while (!now.empty() && now.back() > s[i] && k > 0) {
            k--;
            now.pop_back();
        }
        if (s[i] == '0') continue;
        now.push_back(i);
    }
    while (!now.empty()) {
        ans[now.back()] = 1;
        now.pop_back();
    }
    return ans;
}

int main() {
    int q; std::cin >> q;
    while (q--) {
        int n, k; std::cin >> n >> k;
        std::string s; std::cin >> s;
        auto ans = solve(s, k);
        for (int i = 0; i < n; i++) {
            std::cout << ans[i];
        }
        std::cout << std::endl;
    }
    return 0;
}
