for i in range(int(input())):
    x,y,z=map(int,input().split())
    if x<y:x,y=y,x
    if y<z:y,z=z,y
    if x==y==z:print("YES\n"+str(x)+' '+str(y)+' '+str(z))
    elif x==y and x!=z:print("YES\n"+str(x)+' '+str(x)+' '+str(z))
    elif x!=y and y==z:print("YES\n"+str(x)+' '+str(y)+' '+str(y))
    elif x==z and y!=x:print("YES\n"+str(x)+' '+str(z)+' '+str(y))
    else:print("NO")
