import sys
input = sys.stdin.readline
n = int(input())
adj = [[] for _ in range(n)]
for _ in range(n - 1):
    a, b, c = map(int, input().split())
    adj[a].append((b, c))
    adj[b].append((a, c))
vis = [False] * n
costs = [0] * n
que = [(0, 0)]
while que:
    cur, c = que.pop()
    if vis[cur]: continue
    vis[cur] = True
    costs[cur] = c
    for nxt, cost in adj[cur]:
        if not vis[nxt]: que.append((nxt, c + cost))
ans = 0
for c in costs: ans = max(ans, c)
print(ans)
