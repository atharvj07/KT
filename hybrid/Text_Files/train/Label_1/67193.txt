import sys

n, sx, sy = map(int, sys.stdin.readline().split())
locations = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]

count = 0
x_min, x_max = sx, sx
y_min, y_max = sy, sy

for x, y in locations:
    if x != sx or y != sy:
        if x == sx:
            y_min = min(y_min, y)
            y_max = max(y_max, y)
        elif y == sy:
            x_min = min(x_min, x)
            x_max = max(x_max, x)
        else:
            count += 1
            x_min = min(x_min, x)
            x_max = max(x_max, x)
            y_min = min(y_min, y)
            y_max = max(y_max, y)

x_opt = (x_min + x_max) // 2
y_opt = (y_min + y_max) // 2

print(count + 1 if count > 0 else 0)
print(x_opt, y_opt)
