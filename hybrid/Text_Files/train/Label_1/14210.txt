#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e5;
int freq[mxN + 5], sum[mxN + 5];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        freq[x]++;
        sum[x] += y;
    }
    int ans = 0;
    for (int i = 1; i <= m; i++) {
        if (freq[i]) {
            ans += max(sum[i] / freq[i] * freq[i], 0);
        }
    }
    cout << ans;
}
