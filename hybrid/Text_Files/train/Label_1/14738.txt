#include <iostream>
#include <map>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  map<ll, ll> mp;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    mp[a[i]]++;
  }
  if (mp.size() >= 2) {
    ll res = 0, ans = -1;
    for (map<ll, ll>::iterator it = mp.begin(); it != mp.end(); it++) {
      if (res == it->second) {
        ans = it->first;
        break;
      }
      if (ans == -1) ans = it->first;
      res = it->second;
    }
    cout << "1\n" << ans << ' ' << res << '\n';
    return 0;
  }
  if (mp.size() == 1) {
    cout << "1\n" << mp.begin()->second << ' ' << mp.begin()->second << '\n';
    return 0;
  }
  cout << "0\n";
  return 0;
}
