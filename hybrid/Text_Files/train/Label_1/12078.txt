#include <bits/stdc++.h>
using namespace std;
int main() {
    int a[6], b[100000], n;
    for (int i = 0; i < 6; i++) { cin >> a[i]; }
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    sort(b, b + n);
    sort(a, a + 6);
    int ans = b[n - 1] - b[0];
    int now = b[0] - a[0];
    for (int i = 0; i < 6; i++) {
        if (now < 0) now += 1e9;
        now = min(now, b[n - 1] - a[i]);
        ans = min(ans, now + b[n - 1] - a[i]);
        ans = min(ans, b[0] - a[i]);
    }
    cout << ans << endl;
}
