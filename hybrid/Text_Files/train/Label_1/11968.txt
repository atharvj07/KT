#include <bits/stdc++.h>
using namespace std;
const int maxn = 105;
int dp[maxn][maxn], n, m, k;
string s1, s2, v;
int main() {
    cin >> s1 >> s2 >> v;
    n = (int)s1.size();
    m = (int)s2.size();
    k = (int)v.size();
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (s1[i - 1] == s2[j - 1]) {
                int f = 0;
                for (int l = 0; v[l] && l < k; l++) {
                    if (s1[i - l - 1] != v[l]) {
                        f = 1;
                        break;
                    }
                }
                if (f == 0)
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                else
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            } else {
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    if (dp[n][m]) {
        string ans;
        int i = n, j = m;
        while (i > 0 && j > 0) {
            int f = 0;
            for (int l = 0; v[l] && l < k; l++) {
                if (s1[i - l - 1] != v[l]) {
                    f = 1;
                    break;
                }
            }
            if (f == 0 && s1[i - 1] == s2[j - 1]) {
                ans += s1[i - 1];
                i--, j--;
            } else if (dp[i - 1][j] > dp[i][j - 1])
                i--;
            else
                j--;
        }
        reverse(ans.begin(), ans.end());
        cout << ans << endl;
    } else
        cout << 0 << endl;
    return 0;
}
