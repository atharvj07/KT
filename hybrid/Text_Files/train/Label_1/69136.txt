import sys
input = sys.stdin.readline

n,k = map(int,input().split())
a = [list(map(int,input().split())) for _ in range(k)]
b = [[] for _ in range(n+1)]
for x,y in a:
    b[x].append(y)
    b[y].append(x)

f = [0]*(n+1)
g = []

def dfs(x):
    for y in b[x]:
        if f[y]==0:
            f[y]=1
            dfs(y)
            g.append(y)

dfs(1)
print(len(g)-len(set(g)))
