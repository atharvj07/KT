#include <bits/stdc++.h>

using namespace std;

const int Mod = 998244353;
const int N = 2e5 + 9;

int32_t main() {
  ios_base::sync_with_stdio(false);
  int n, k, h[N];
  cin >> n >> k;
  vector<int> b(n);
  for (int i = 0; i < n; ++i) {
    cin >> h[i];
    --h[i];
  }
  long long pw[N];
  pw[0] = 1;
  for (int i = 1; i < N; ++i) pw[i] = pw[i - 1] * 2 % Mod;
  vector<vector<long long>> dp(2, vector<long long>(n, -1));
  long long ans = 0;
  dp[0][n - 1] = 0;
  dp[1][n - 1] = pw[n - 1];
  for (int i = n - 2; i >= 0; --i) {
    for (int j = 0; j < 2; ++j) {
      int nj = (h[i] + j + 1) % k;
      if (nj == h[(i + 1) % n])
        dp[j][i] = (dp[j][(i + 1) % n] + pw[i]) % Mod;
      if (nj != h[(i + 1) % n])
        dp[j][i] = dp[0][(i + 1) % n];
      if (j == 0)
        dp[1][i] = (dp[1][(i + 1) % n] * 2) % Mod;
    }
    ans += dp[1][i];
    ans %= Mod;
  }
  cout << ans << '\n';
  return 0;
}
