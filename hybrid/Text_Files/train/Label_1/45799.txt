import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        int cost = 0;
        for (int i = 0; i < n - 1; i++) {
            int min = i;
            for (int j = i + 1; j < n; j++) {
                if (w[j] < w[min]) {
                    min = j;
                }
            }
            if (min != i) {
                cost += w[i] + w[min];
                int tmp = w[i];
                w[i] = w[min];
                w[min] = tmp;
            }
        }
        System.out.println(cost);
    }
}
