t, x = map(int, input().split())
a = list(map(int, input().split()))
c = [0]*t
for i in a:
    c[i-1]+=1
mn = 0; ans = 0
for i in range(t):
    if c[i]>0:
        mn+=1
    if mn>x:
        ans+=c[i]*(mn-x-1)
    if mn==x:
        ans+=c[i]*(c[i]+1)//2 - (c[i]%2==0)
        c[i]-=c[i]%2
        mn-=c[i]%2
print(ans)
