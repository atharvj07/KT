#include<bits/stdc++.h>

using namespace std;

const int MaxN = 2e5+10;

typedef long long ll;

ll tank, d, n, m;
ll gas[MaxN], pr[MaxN];

ll solve(){
    ll ans=0, last =-1;
    for(int i=0; i<m; i++){
        if(last!=-1){
            ans+= (gas[i] - last) * pr[last];
        }
        last=gas[i];
    }
    if(last==-1 || gas[last] < d){
        return -1;
    }
    ans+= (d - last) * pr[last];

    return ans;
}

int main(){
    cin>>d>>n>>m;
    for(int i=0; i<m; i++){
        cin>>gas[i]>>pr[i];
    }
    sort(gas,gas+m);
    sort(pr,pr+m);

    cout<<solve()<<endl;

    return 0;
}
