#include <bits/stdc++.h>
using namespace std;
#define ll long long

set<ll> p;
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, x;
    cin >> n >> x;
    p.clear();
    ll cnt = 0;
    for (ll i = 0; i < n; i++) {
      ll s;
      cin >> s;
      if (s == x) cnt++;
      p.insert(s);
    }
    if (p.size() == 1)
      cout << 0 << endl;
    else if (p.size() == 2 && cnt > 0)
      cout << 1 << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
