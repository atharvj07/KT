#include <iostream>
#include <vector>

using namespace std;

int main() {
  int k;
  cin >> k;
  int n = 1 << k;
  vector<vector<int>> ans(n, vector<int>(n));
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if ((i >> j) & 1) {
        ans[i][j] = -1;
      } else {
        ans[i][j] = 1;
      }
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      cout << ((ans[i][j] == 1) ? '+' : '*');
    }
    cout << endl;
  }
  return 0;
}
