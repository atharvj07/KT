#include <iostream>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  char grid[n][m];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> grid[i][j]; }
  }

  int ans = 0;
  for (int i = 1; i < n - 1; i++) {
    for (int j = 1; j < m - 1; j++) {
      if (grid[i][j] == '.' &&
          (grid[i - 1][j] == '.' || grid[i - 1][j + 1] == '.' ||
           grid[i][j - 1] == '.' || grid[i][j + 1] == '.') &&
          (grid[i + 1][j] == '.' || grid[i + 1][j + 1] == '.' ||
           grid[i][j - 1] == '.' || grid[i][j + 1] == '.')) {
        ans++;
      }
    }
  }

  cout << ans << endl;

  return 0;
}
