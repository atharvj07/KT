import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[][] a = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    a[i][j] = sc.nextInt();
                }
            }
            if (checkWin(a, n, m)) {
                System.out.println("Ashish");
            } else {
                System.out.println("Vivek");
            }
        }
    }

    public static boolean checkWin(int[][] a, int n, int m) {
        int available = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 0) {
                    available++;
                }
            }
        }
        if (available == 0) {
            return false;
        }
        return available % 2 == 1;
    }
}
