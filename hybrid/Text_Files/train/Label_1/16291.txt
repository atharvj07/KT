#include <iostream>

using namespace std;

bool solve(long long n, long long x, long long y) {
  if (x == 1 && y == 1) return false;
  if (n - x == n - y) return (x + y) & 1;
  return (n - y) >= (n - x);
}

int main() {
  long long n, x, y;
  cin >> n >> x >> y;
  cout << (solve(n, x, y) ? "White" : "Black") << endl;
  return 0;
}
