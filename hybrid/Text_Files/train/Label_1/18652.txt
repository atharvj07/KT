#include<bits/stdc++.h>
using namespace std;
int main(){
	string s;
	cin>>s;
	int n=s.size();
	int x=0,y=0;
	int ans=0;
	for(int i=0;i<n;i++){
		if(s[i]=='L')	x-=1;
		else if(s[i]=='R')	x+=1;
		else if(s[i]=='U')	y+=1;
		else if(s[i]=='D')	y-=1;
		else ans+=1;
	}
	int a=abs(x),b=abs(y);
	if(a==b)	ans+=a;
	else ans=-1;
	cout<<ans<<endl;
	return 0;
}
