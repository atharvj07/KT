import java.util.*;

class Main {
    private static int h1, a1, c1, h2, a2;
    private static int ptr = 0;

    private static String[] ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        h1 = sc.nextInt();
        a1 = sc.nextInt();
        c1 = sc.nextInt();
        h2 = sc.nextInt();
        a2 = sc.nextInt();

        if (a1 >= a2) {
            System.out.println(1);
            System.out.println("STRIKE");
            return;
        }

        int x = (h2 + a2 - 1) / a1;
        x++; // healing potion before the 1st attack
        ans = new String[x];

        solve(h1, h2, 0);
        System.out.println(x);
        for (String s: ans) {
            System.out.println(s);
        }
    }

    private static void solve(int cv, int cr, int cnt) {
        ans[ptr++] = "STRIKE";
        cr -= a1;
        cv += c1 - a2;
        if (cr <= 0) {
            return;
        }
        solve(cv, cr, ++cnt);
    }
}
