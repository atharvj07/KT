import java.util.Scanner;

public class CompetitionPrize {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int codingContestPlace = scanner.nextInt();
        int robotManeuverPlace = scanner.nextInt();

        int codingContestPrize = 0;
        int robotManeuverPrize = 0;

        if (codingContestPlace <= 3) {
            switch (codingContestPlace) {
                case 1:
                    codingContestPrize = 300000;
                    break;
                case 2:
                    codingContestPrize = 200000;
                    break;
                case 3:
                    codingContestPrize = 100000;
                    break;
            }
        }

        if (robotManeuverPlace <= 3) {
            switch (robotManeuverPlace) {
                case 1:
                    robotManeuverPrize = 300000;
                    break;
                case 2:
                    robotManeuverPrize = 200000;
                    break;
                case 3:
                    robotManeuverPrize = 100000;
                    break;
            }
        }

        int totalPrize = codingContestPrize + robotManeuverPrize;

        if (codingContestPlace == 1 && robotManeuverPlace == 1) {
            totalPrize += 400000;
        }

        System.out.println(totalPrize);
    }
}
