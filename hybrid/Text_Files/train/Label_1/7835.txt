#include <cstdio>
#include <vector>
#include <queue>
using namespace std;

const int MAX = 500010;
const int MAX_DAG = 50010;
vector<int> g[MAX], dag[MAX_DAG], topo;
int dis[MAX], in_degree[MAX];
queue<int> q;

int find_spanning_tree(int n) {
    int cnt = 0;
    for (int i = 1; i < n; i++) {
        int u;
        while (q.size()) {
            u = q.front();
            q.pop();
            for (vector<int>::iterator it = dag[u].begin(); it != dag[u].end(); it++) {
                int v = *it;
                in_degree[v]--;
                if (in_degree[v] == 0) {
                    q.push(v);
                }
            }
        }
        topo.push_back(u);
        cnt++;
    }
    return cnt;
}

int main() {
    int n, m;
    scanf("%d%d", &n, &m);
    for (int i = 0; i < m; i++) {
        int a, b;
        scanf("%d%d", &a, &b);
        g[a].push_back(b);
        g[b].push_back(a);
    }
    q.push(1);
    dis[1] = 1;
    while (q.size()) {
        int u = q.front();
        q.pop();
        for (vector<int>::iterator it = g[u].begin(); it != g[u].end(); it++) {
            int v = *it;
            if (dis[v] != 0) {
                if (dis[u] > dis[v]) {
                    dag[dis[u]].push_back(dis[v]);
                    in_degree[dis[v]]++;
                }
                continue;
            }
            dis[v] = dis[u] + 1;
            q.push(v);
        }
    }
    printf("%d\n", find_spanning_tree(n));
    for (int i = 0; i < topo.size(); i++) {
        printf("%d ", topo[i]);
    }
    puts("");
    return 0;
}
