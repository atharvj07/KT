import sys

input = sys.stdin.readline
n, t = map(int, input().split())
a = list(map(int, input().split()))
temp = list(map(int, input().split()))
aSum = sum(a)
tempSum = sum(a[i]*temp[i] for i in range(n))
if tempSum*aSum == 0 or tempSum*aSum % (aSum * t) != 0:
    print(0)
else:
    ans = tempSum / (aSum * t)
    print(ans)
