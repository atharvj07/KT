import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = in.nextLong();
        }
        long gcd = 0;
        for (int i = 0; i < n; i++) {
            gcd = gcd(gcd, Math.abs(a[i] - b[i]));
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == b[i]) {
                ans++;
            } else if ((a[i] - b[i]) % gcd == 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
