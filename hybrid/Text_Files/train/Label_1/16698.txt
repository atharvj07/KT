#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9;
const int MAXN = 1e5;
const int mod = 1e9 + 7;
#define int long long

int n, k;
pair<int, int> seg[MAXN];
int dp[MAXN][21][2];
int calc(int id, int rem, bool is_pre) {
    if (id == n) {
        if (k == 1 || rem == 0) return 0;
        return INF;
    }
    int &ret = dp[id][rem][is_pre];
    if (ret != -1) return ret;
    ret = INF;
    ret = min(ret, calc(id + 1, rem, is_pre));
    if (!is_pre) {
        // extend left
        ret = min(ret, calc(id + 1, (rem + (seg[id].first % k)) % k, true));
        // extend right
        ret = min(ret, calc(id + 1, (rem + (seg[id].second % k)) % k, true));
    }
    return ret;
}
int best_val(int l, int r, int rem) {
    if (l > r) return INF;
    int ans = 0;
    for (int i = l; i <= r; i++) {
        if (seg[i].first % k <= rem) ans++;
    }
    return ans;
}
int32_t main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> seg[i].first >> seg[i].second;
    memset(dp, -1, sizeof(dp));
    int ans = calc(0, 0, 0);
    if (ans == INF)
        cout << -1 << endl;
    else
        cout << ans << endl;
    return 0;
}
