def find_time(n, m, sensors):
    def gcd(a, b):
        while b:
            a, b = b, a % b
        return a

    def lcm(a, b):
        return a * b // gcd(a, b)

    def solve(x, y, n, m):
        # The ray will repeat its pattern every LCM(n, m) time units
        period = lcm(n, m)
        # Calculate how many times the sensor position is "mirrored" in the repeated pattern
        x_periods, y_periods = (x // n, y // m) if x % n != 0 or y % m != 0 else ((x - 1) // n, (y - 1) // m)
        # Check if the sensor is in the path of the laser based on its "mirrored" position
        if (x_periods % 2 == y_periods % 2):
            # Find the actual positions within one period
            x_position = x % (2 * n) if x_periods % 2 == 0 else 2 * n - x % (2 * n)
            y_position = y % (2 * m) if y_periods % 2 == 0 else 2 * m - y % (2 * m)
            # Check if both positions match, meaning the laser passes through the sensor
            if x_position == x % (2 * n) and y_position == y % (2 * m):
                return min(x_position, y_position)
        return -1

    results = []
    for sensor in sensors:
        x, y = sensor
        time = solve(x, y, n, m)
        results.append(time)

    return results

n, m, k = map(int, input().split())
sensors = [list(map(int, input().split())) for _ in range(k)]
results = find_time(n, m, sensors)
for result in results:
    print(result)
