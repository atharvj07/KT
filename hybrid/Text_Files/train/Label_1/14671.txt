#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 5;
int n, a[maxn], b[maxn];
long long tot, ans;
priority_queue<long long> pq;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
  }
  for (int i = 1; i <= n; i++) {
    if (b[i] == 'R' || b[i] == 'G') {
      pq.push(a[i]);
      tot += a[i];
    }
  }
  while (pq.size() > 1) {
    long long x = pq.top();
    pq.pop();
    long long y = pq.top();
    pq.pop();
    ans += x + y;
    pq.push(x + y);
  }
  for (int i = 1; i <= n; i++) {
    if (b[i] == 'G' || b[i] == 'B') {
      pq.push(a[i]);
      tot += a[i];
    }
  }
  while (pq.size() > 1) {
    long long x = pq.top();
    pq.pop();
    long long y = pq.top();
    pq.pop();
    ans += x + y;
    pq.push(x + y);
  }
  cout << ans << endl;
  return 0;
}
