import sys
import math
n, m = map(int, input().split())
sx, sy, fx, fy = map(int, input().split())
L = []
for _ in range(m):
    L.append(list(map(int, input().split())))
ans = math.inf
for p in L:
    if p[0] == sx or p[1] == sy:
        ans = min(ans, abs(fx - p[0]) + abs(fy - p[1]))
    elif p[0] == fx or p[1] == fy:
        ans = min(ans, abs(fx - sx) + abs(fy - sy) + abs(fx - p[0]) + abs(fy - p[1]))
print(ans)
