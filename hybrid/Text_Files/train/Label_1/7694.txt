#include<bits/stdc++.h>
using namespace std;
const double eps=1e-9;

int n,m;
double arr[100005];

int main() {
    cin>>n>>m;
    for(int i=1; i<=m; i++) {
        int x,d;
        cin>>x>>d;
        double inc=x-d;
        double dec=x+d;
        for(int j=1; j<=n; j++) {
            if(abs(j-(n+1)/2)>abs(j-(n+1)/2-d)) arr[j]+=dec;
            else arr[j]+=inc;
        }
    }
    double ans=0;
    for(int i=1; i<=n; i++) ans+=arr[i];
    ans/=n;
    cout<<fixed<<setprecision(9)<<ans<<endl;
    return 0;
}
