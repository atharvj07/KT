#include <bits/stdc++.h>

using namespace std;

int main() {
  string t;
  cin >> t;
  string s(t);
  reverse(t.begin(), t.end());
  for (int i = 1; i < t.size(); i++) {
    if (t[i] == t[i - 1]) {
      s.insert(i, string(1, t[i]));
      t.erase(i, 1);
    }
  }
  cout << s << endl;
  return 0;
}
