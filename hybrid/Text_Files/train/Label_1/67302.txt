import sys
n = int(input())
a, b, c = [0]*n, [0]*n, [0]*n
for i in range(n):
    a[i] = b[i] = i
for i in range(n):
    c[i] = (2*i + 1)%n
if n%2==1: print(*a,'\n',*b,'\n',*c); sys.exit(0)
for i in range(1,(n+1)//2):
    a[(2*i-2)%n] = (2*i-1)%n
    b[(2*i-1)%n] = (2*i-2)%n
print(*a,'\n',*b,'\n',*c)
