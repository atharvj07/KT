import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextLong();
        }
        sc.close();

        long l = 1, r = 1000000000000000L;
        while (l < r) {
            long mid = (l + r) / 2;
            boolean isValid = true;
            long cur = mid;
            for (long ai : a) {
                if (cur % ai != 0) {
                    isValid = false;
                    break;
                }
                cur /= ai;
            }
            if (isValid && cur == 2) {
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        if (l == 1000000000000000L || l % 2 != 0) {
            System.out.println(-1);
        } else {
            System.out.println(l - 1 + " " + l);
        }
    }
}
