#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll INF = 1e18;
const int mod = 1e9 + 7;
const double EPS = 1e-9;

int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (ll i = 0; i < n; ++i) cin >> a[i];
    sort(a.begin(), a.end());
    ll ans = 0;
    for (ll i = 0; i < n; ++i) {
        ll j = upper_bound(a.begin(), a.end(), a[i] + k) - a.begin() - 1;
        ans += j - i;
    }
    cout << ans << endl;
    return 0;
}
