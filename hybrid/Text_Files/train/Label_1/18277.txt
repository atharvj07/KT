#include <iostream>
#include <vector>

template <typename T>
struct BIT{
    int N;
    std::vector<T> BITree;

    BIT(int N) : N(N), BITree(N+1, 0) {}

    void update(int idx, int val){
        while(idx <= N){
            BITree[idx] += val;
            idx += idx & -idx;
        }
    }

    T query(int idx){
        T sum = 0;
        while(idx > 0){
            sum += BITree[idx];
            idx -= idx & -idx;
        }
        return sum;
    }
};

int main() {
    int n, k, q;
    std::cin >> n >> k >> q;

    BIT<int> bit(200000);

    for(int i = 0; i < n; i++){
        int l, r;
        std::cin >> l >> r;
        bit.update(l, 1);
        bit.update(r+1, -1);
    }

    for(int i = 1; i <= 200000; i++){
        bit.update(i, bit.query(i-1));
    }

    for(int i = 0; i < q; i++){
        int a, b;
        std::cin >> a >> b;
        std::cout << bit.query(b) - bit.query(a-1) << '\n';
    }

    return 0;
}
