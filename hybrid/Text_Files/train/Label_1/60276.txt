def min_operations(n, a, b):
    # Compute the positions (0-indexed) of each number in arrays a and b
    pos_a = [0] * (2 * n + 1)
    pos_b = [0] * (2 * n + 1)
    for i in range(n):
        pos_a[a[i]] = i
        pos_b[b[i]] = i
    
    # Compute the minimum position in b for each possible first element of a
    min_pos_b = [float('inf')] * (2 * n + 1)
    for i in range(2 * n, 0, -2):
        min_pos_b[i] = min(min_pos_b[i], pos_b[i])
        if i > 2:
            min_pos_b[i-2] = min_pos_b[i]
    
    # Compute the minimum number of operations needed
    ans = float('inf')
    for i in range(1, 2 * n + 1, 2):
        ans = min(ans, pos_a[i] + min_pos_b[i + 1])
    
    return ans

# Reading number of test cases
t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))
    print(min_operations(n, a, b))
