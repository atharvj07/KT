import java.util.*;
class Main {
    public static void main(String[] argv) {
        Scanner in = new Scanner(System.in);
        int maxx = in.nextInt();
        int maxy = in.nextInt();
        int w = in.nextInt();
        int[] luck = new int[maxy+1];
        for (int i = 1; i <= maxy; ++i)
            if (i < 10) luck[i] = 0;
            else if (i < 100) luck[i] = 1;
            else luck[i] = 2;
        int x = maxx / 2 + (maxx % 2 == 1 ? 1 : 0);
        if (w <= maxx && w <= maxy && luck[x] == luck[w]) {
            System.out.println(x + " " + w);
            return;
        }
        int y = maxy;
        if (w <= x * y) {
            System.out.println(x + " " + w / x);
            return;
        }
        w -= x * maxy;
        if (w <= y) {
            System.out.println(x + 1 + " " + w);
            return;
        }
        System.out.println(-1);
    }
}
