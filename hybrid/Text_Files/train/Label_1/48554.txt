import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        List<Integer> nums = new ArrayList<>();
        while (sc.hasNextInt()) {
            nums.add(sc.nextInt());
        }
        Collections.sort(nums);
        if (nums.size() < 24) {
            System.out.println("no");
            return;
        }
        int[] counts = new int[12];
        for (int i = 0; i < 12; i++) {
            counts[nums.get(i) - 1]++;
        }
        boolean[] used = new boolean[12];
        for (int i = 0; i < 12; i++) {
            for (int j = 0; j < used.length; j++) {
                if (!used[j] && counts[i] >= 4 && nums.get(j) == nums.get(i)) {
                    for (int k = 0; k < used.length; k++) {
                        if (!used[k] && nums.get(k) == nums.get(i)) {
                            if (counts[nums.get(j) - 1] >= 8) {
                                System.out.println("yes");
                                return;
                            }
                            used[k] = true;
                        }
                    }
                    used[j] = true;
                } else if (!used[j] && nums.get(j) != nums.get(i) && counts[nums.get(i)] >= 2 && counts[nums.get(j)] >= 2) {
                    for (int k = 0; k < used.length; k++) {
                        if (!used[k] && nums.get(k) == nums.get(i)) {
                            used[k] = true;
                        } else if (!used[k] && nums.get(k) == nums.get(j)) {
                            used[k] = true;
                        }
                    }
                }
            }
        }
        System.out.println("no");
    }
}
