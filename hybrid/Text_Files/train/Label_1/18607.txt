#include <iostream>
using namespace std;
int main() {
  int a, b, l, r;
  cin >> a >> b >> l >> r;
  l += b - 1; // make l refer to last letter added by Mr. B
  int n = (r - b) / a;
  int d = (r - b) % a;
  if (n % 2 == 1)
    d = a - d;
  int dist = d + n % 2;
  if (l > n * a)
    cout << 1;
  else if (l <= dist)
    cout << l + 2;
  else
    cout << dist + 2;
}
