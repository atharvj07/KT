from collections import deque

def bfs(graph, start, n):
    distances = [-1] * (n + 1)
    queue = deque([start])
    distances[start] = 0
    while queue:
        v = queue.popleft()
        for w in graph[v]:
            if distances[w] == -1:
                distances[w] = distances[v] + 1
                queue.append(w)
    return distances

n, m, k = map(int, input().split())
special = list(map(int, input().split()))
graph = [[] for _ in range(n + 1)]
for _ in range(m):
    x, y = map(int, input().split())
    graph[x].append(y)
    graph[y].append(x)

dist1 = bfs(graph, 1, n)
distn = bfs(graph, n, n)

special.sort(key=lambda x: dist1[x] - distn[x])

max_dist = -1
max_diff = -1
for i in range(k-1):
    max_diff = max(max_diff, dist1[special[i]] + 1 + distn[special[i+1]])
max_dist = max(max_dist, max_diff, dist1[n])

print(max_dist)
