import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

class Main {
    static class Vertex {
        int x;
        int t;
        ArrayList<Integer> neighbours;
        int parent;
        int l;

        public Vertex(int x, int t) {
            this.x = x;
            this.t = t;
            neighbours = new ArrayList<>();
            parent = -1;
            l = 0;
        }
    }

    static long maxCookies;
    static Vertex[] tree;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long T = sc.nextInt();

        tree = new Vertex[n + 1];
        for (int i = 1; i <= n; i++) {
            tree[i] = new Vertex(sc.nextInt(), sc.nextInt());
        }

        for (int i = 2; i <= n; i++) {
            Vertex vertex = tree[i];
            int parent = sc.nextInt();
            int l = sc.nextInt();
            vertex.parent = parent;
            vertex.l = l;
            tree[parent].neighbours.add(i);
        }

        dfs(1, 0, 0, T);
        System.out.println(maxCookies);
    }

    static void dfs(int curr, int time, int cookies, long T) {
        maxCookies = Math.max(maxCookies, cookies);

        for (int u : tree[curr].neighbours) {
            int edgeTime = tree[curr].l;
            int descendTime = tree[u].t * tree[u].x;
            int ascendTime = tree[u].l;

            long remTime = T - time - descendTime - ascendTime - edgeTime;
            if (remTime >= 0) {
                dfs(u, time + edgeTime, cookies + tree[u].x, T);
            }
        }
    }
}
