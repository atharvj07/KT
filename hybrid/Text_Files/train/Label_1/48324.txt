import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] strength = new int[n][k];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                strength[i][j] = sc.nextInt();
            }
            Arrays.sort(strength[i]);
        }

        for (int[] arr : strength) {
            int count = 0;
            for (int i = 0; i < k; i++) {
                if (arr[i] == arr[k - 1]) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
