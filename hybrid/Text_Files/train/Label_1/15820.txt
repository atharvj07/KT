#include <bits/stdc++.h>
using namespace std;
vector<int> G[1005];
bool vis[1005];
bool flag;
int a, b, c, d;
vector<int> ans1, ans2;
void dfs(int u, int v1, int v2) {
  vis[u] = true;
  if (u == b)
    ans1.push_back(u);
  else if (u == d)
    ans2.push_back(u);
  if (ans1.size() == d - a + 1 && ans2.size() == d - c + 1)
    return;
  for (auto i : G[u]) {
    if (!vis[i]) {
      if (v1 == u)
        dfs(i, u, v2);
      else if (v2 == u)
        dfs(i, v1, u);
      else
        dfs(i, v1, v2);
      if (flag)
        return;
    }
  }
}
int main() {
  int n, k;
  cin >> n >> k;
  cin >> a >> b >> c >> d;
  if (k > 2 * n - 2) {
    cout << "-1" << endl;
    return 0;
  }
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    G[u].push_back(v);
    G[v].push_back(u);
  }
  dfs(a, 0, 0);
  if (!flag) {
    cout << "-1" << endl;
    return 0;
  }
  reverse(ans1.begin(), ans1.end());
  for (auto i : ans1)
    cout << i << ' ';
  cout << endl;
  reverse(ans2.begin(), ans2.end());
  for (auto i : ans2)
    cout << i << ' ';
  cout << endl;
  return 0;
}
