#include <bits/stdc++.h>
using namespace std;
const int N = 200005;
int fre[N];
char s[N], ans[N];
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    scanf("%s", s);
    for (int i = 'a'; i <= 'z'; i++) fre[i] = 0;
    int len = strlen(s);
    for (int i = 0; i < len; i++) fre[s[i]]++;
    int l = 0, r = 0;
    for (int i = 'z'; i >= 'a'; i--) {
      while (fre[i] > 0) {
        while (l < r && ans[r - 1] >= i) r--;
        ans[r++] = i, fre[i]--;
      }
    }
    ans[r] = '\0';
    printf("%s\n", ans);
  }
}
