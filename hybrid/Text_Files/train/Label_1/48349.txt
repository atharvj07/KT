import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int C = sc.nextInt();
        int[][] wrongness = new int[C + 1][C + 1];
        for (int i = 1; i <= C; i++) {
            for (int j = 1; j <= C; j++) {
                wrongness[i][j] = sc.nextInt();
            }
        }
        int[][] grid = new int[N + 1][N + 1];
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
        long ans = 0;
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                int x = (i + j) % 3;
                int ord = -1;
                if (x == 0) {
                    ord = 1;
                } else if (x == 1) {
                    ord = 2;
                } else {
                    ord = 3;
                }
                int minColor = grid[i][j];
                for (int c = 1; c <= C; c++) {
                    if (c == grid[i][j]) continue;
                    if (x == 0) {
                        if (minColor > c) minColor = c;
                    } else if (x == 1) {
                        if (c % 3 == ord && minColor > c) minColor = c;
                    } else if (x == 2) {
                        if (c / 3 == ord && minColor > c) minColor = c;
                    }
                }
                if (minColor != grid[i][j]) ans += wrongness[grid[i][j]][minColor];
            }
        }
        System.out.println(ans);
    }
}
