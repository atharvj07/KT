import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Set;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        ArrayList<ArrayList<Edge>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            String c = line[2];
            adj.get(x).add(new Edge(y, c));
            adj.get(y).add(new Edge(x, c));
        }
        if (!check(adj, n, m)) {
            System.out.println("-1");
        } else {
            Set<Integer> cleared = new HashSet<>();
            Queue<Integer> queue = new LinkedList<>();
            queue.add(1);
            while (!queue.isEmpty()) {
                int u = queue.poll();
                for (Edge v : adj.get(u)) {
                    if (!cleared.contains(v.id) && v.person.equals("S")) {
                        cleared.add(v.id);
                        queue.add(v.to);
                    }
                }
            }
            System.out.println(cleared.size());
            for (int id : cleared) {
                System.out.print(id + " ");
            }
            System.out.println();
        }
    }

    public static boolean check(ArrayList<ArrayList<Edge>> adj, int n, int m) {
        int odd = 0;
        for (int i = 1; i <= n; i++) {
            if (adj.get(i).size() % 2 == 1) {
                odd++;
            }
        }
        return odd == 2 && m % 2 == 0;
    }

    public static class Edge {
        int to;
        String person;
        int id;
        static int count = 1;

        public Edge(int to, String person) {
            this.to = to;
            this.person = person;
            this.id = count++;
        }
    }
}
