#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5;
int n, m;
long long v[MAXN], t[MAXN];
vector<pair<int, int>> adj[MAXN];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int tt;
  cin >> tt;
  while (tt--) {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
      cin >> v[i];
      adj[i].clear();
    }
    for (int i = 0; i < n; i++) cin >> t[i];
    for (int i = 0; i < m; i++) {
      int a, b;
      cin >> a >> b;
      --a, --b;
      adj[a].emplace_back(b, i);
      adj[b].emplace_back(a, i);
    }
    long long sum = 0;
    for (int i = 0; i < n; i++) sum += t[i] - v[i];
    if (sum % n != 0) {
      cout << "NO\n";
      continue;
    }
    long long avg = sum / n;
    bool ok = true;
    for (int i = 0; i < n; i++) {
      ok &= t[i] - v[i] == avg;
    }
    if (!ok) {
      cout << "NO\n";
      continue;
    }
    cout << "YES\n";
    for (int i = 0; i < m; i++) {
      if ((v[adj[i].front().first] - avg) % 2 ==
          (v[adj[i].front().second] - avg) % 2)
        continue;
      cout << i + 1 << '\n';
    }
  }
}
