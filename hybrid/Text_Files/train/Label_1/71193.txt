import sys

r, c = [int(x) for x in sys.stdin.readline().split()]
grid = [list(sys.stdin.readline().strip()) for _ in range(r)]
max_cells = 0

for i in range(r):
    if 'S' not in grid[i]:
        max_cells += c
for j in range(c):
    col = [grid[i][j] for i in range(r)]
    if 'S' not in col:
        max_cells += r

print(max_cells)
