import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), r = sc.nextInt();
        int dp[] = new int[n];
        int[][] a = new int[n][2];
        for (int i = 0; i < n; i++) {
            a[i][0] = sc.nextInt();
            a[i][1] = sc.nextInt();
        }
        Arrays.sort(a, (x, y) -> x[1] - y[1]);
        for (int i = 0; i < n; i++) {
            int idx = Arrays.binarySearch(a, new int[]{0, a[i][1] - r}, (x, y) -> x[1] - y[1]);
            if (idx < 0)
                idx = ~idx;
            if (idx > 0)
                dp[i] = dp[idx - 1] + 1;
        }
        System.out.println(dp[n - 1]);
    }
}
