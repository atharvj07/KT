import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ArrayList<Long> a = new ArrayList<>();
        ArrayList<Long> b = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            a.add(sc.nextLong());
        }
        for (int i = 0; i < n; i++) {
            b.add(sc.nextLong());
        }

        boolean[][] can = new boolean[n][n];
        int[][] op = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (i == j) {
                    can[i][j] = true;
                    op[i][j] = 0;
                } else if (j - i == 1) {
                    can[i][j] = (a.get(i) + a.get(j) == b.get(i) && a.get(j) + a.get(i) == b.get(j));
                    op[i][j] = can[i][j] ? 1 : -1;
                } else {
                    can[i][j] = false;
                    op[i][j] = -1;
                    for (int k = i + 1; k < j; k++) {
                        if (can[i][k] && can[k + 1][j] && (a.get(k) + b.get(k) == b.get(i) && a.get(k) + b.get(k) == b.get(j))) {
                            can[i][j] = true;
                            op[i][j] = op[i][k] + op[k + 1][j] + 1;
                            break;
                        }
                    }
                }
            }
        }

        int ans = op[0][n - 1];
        if (ans == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
