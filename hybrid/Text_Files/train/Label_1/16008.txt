#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  int odd = 0;
  for (int i = 0; i < n; i++) {
    odd += a[i] % 2;
  }

  if (odd == 0)
    cout << "Second";
  else
    cout << "First";

  return 0;
}
