import sys

for t in range(int(input())):
    n, m, x, y = map(int, input().split())
    a = [list(input()) for i in range(n)]
    ans = 0
    for i in range(n):
        can = 1
        for j in range(m):
            if a[i][j] == '*':
                can = 0
            if a[i][j] == '.' and can:
                ans += x
            if a[i][j] == '.' and not can:
                ans += y
                can = 1
    print(ans)
