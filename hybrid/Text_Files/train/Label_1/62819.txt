from collections import defaultdict
from heapq import *

n, m = map(int, input().split())
v = list(map(int, input().split()))
edges = [[] for _ in range(n)]
for _ in range(m):
    x, y = map(int, input().split())
    x, y = x-1, y-1
    edges[x].append(y)
    edges[y].append(x)

PQ = [(0, 0)]
dist = defaultdict(lambda: float('inf'))
dist[0] = 0

while PQ:
    cost, node = heappop(PQ)
    if node == n-1:
        print(cost)
        break
    for u in edges[node]:
        if cost + v[u] < dist[u]:
            dist[u] = cost + v[u]
            heappush(PQ, (cost + v[u], u))
