#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 1e5 + 5;
ll bottle[maxn];
int n, l;
int main() {
    cin >> n >> l;
    for (int i = 1; i <= n; i++) cin >> bottle[i];
    ll mini = 1e9;
    for (int i = n; i >= 1; i--) {
        ll val = l / (1LL << (i - 1));
        mini = min(mini, val * bottle[i] + l % (1LL << (i - 1)));
    }
    cout << mini << endl;
    return 0;
}
