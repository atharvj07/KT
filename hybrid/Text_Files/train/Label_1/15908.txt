#include<bits/stdc++.h>
using namespace std;
long long a[100001];
int main()
{
    long long ans,sum=0,i,j,n,l=0,r=0;
    cin>>n;
    for(i=0;i<n;i++)
        cin>>a[i];
    for(i=1;i<n;i++)
    {
        while(j<n and a[j]-a[j-1]==a[i]-a[i-1])
            j++;
        if((j-i)>(r-l))
        {
            r=j;
            l=i;
        }
        i=j;
    }
    cout<<r-l+1<<endl;
    return 0;
}
