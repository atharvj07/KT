import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int[] freq = new int[n + 1];
        for (int i = 0; i < n; i++) {
            freq[arr[i]]++;
        }

        int maxFreq = 0;
        for (int i = 1; i <= n; i++) {
            maxFreq = Math.max(maxFreq, freq[i]);
        }

        if (maxFreq <= 1) {
            System.out.println(0);
            return;
        }

        int l = 0, r = 0;
        int ans = 0;
        int[] cnt = new int[n + 1];
        while (r < n) {
            cnt[arr[r]]++;
            while (l <= r && cnt[arr[r]] > maxFreq) {
                cnt[arr[l]]--;
                l++;
            }
            if (cnt[arr[r]] == maxFreq && cnt[arr[r]] > 1) {
                ans = Math.max(ans, r - l + 1);
            }
            r++;
        }

        System.out.println(ans);
    }
}
