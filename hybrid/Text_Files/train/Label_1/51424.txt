import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            long n = input.nextLong(), k = input.nextLong();
            k = (k - 1) * n + 1;
            System.out.println(k);
        }
        input.close();
    }
}
