n = int(input())
friend_counts = list(map(int, input().split()))

total_possible_friends = sum(friend_counts)
possible_bob_friends = []

# Check if total possible friends is odd or even
if total_possible_friends % 2 == 0:
    for bob_friends in range(n + 1):
        # Calculate the total number of handshakes excluding Bob
        total_handshakes_excl_bob = total_possible_friends // 2
        # Calculate Bob's potential handshakes
        bob_handshakes = bob_friends * (n - bob_friends) + bob_friends * (bob_friends - 1) // 2
        # Check if the total handshakes match
        if total_handshakes_excl_bob == bob_handshakes:
            possible_bob_friends.append(bob_friends)
else:
    possible_bob_friends.append(-1)

if possible_bob_friends:
    print(" ".join(map(str, possible_bob_friends)))
else:
    print("-1")
