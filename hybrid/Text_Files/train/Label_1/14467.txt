#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  map<int, int> exist;
  vector<int> ans, m(n + 1);
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    if (!exist[x]) exist[x] = i;
    m[x]++;
  }
  for (int i = 1; i <= n; i++) {
    if (!m[i]) ans.push_back(exist[i] + 1);
  }
  for (int i = 1; i <= n; i++) {
    if (m[i] > 1) {
      int tmp = ans.back();
      ans.pop_back();
      ans.push_back(exist[i] + 1);
      exist[i] = exist[tmp];
      exist[tmp] = -1;
    }
  }
  for (const auto &i : ans) cout << i << ' ';
  return 0;
}
