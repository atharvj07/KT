n,a,b=map(int,input().split())
s=[list(map(int,input().split())) for i in range(n)]
for i in range(n):s+=[[s[i][1],s[i][0]]]
s.sort(key=lambda p:p[0]*p[1],reverse=True)
ans=0
for x1,y1 in s:
    for x2,y2 in s[s.index([x1,y1])+1:]:
        if x1<=a and x2<=a:y=min(b,y1+y2)
        elif x1<=b and x2<=b:y=min(a,y1+y2)
        elif y1<=a and y2<=a:y=min(b,x1+x2)
        elif y1<=b and y2<=b:y=min(a,x1+x2)
        else:continue
        ans=max(ans,x1*y1+x2*y2)
print(ans)
