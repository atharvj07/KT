import sys
input=sys.stdin.readline

n,m=map(int,input().split())
score=[list(map(int,input().split())) for i in range(n)]
hate=[[0]*n for i in range(n)]
for i in range(m):
    a,b=map(int,input().split())
    hate[a-1][b-1]=1
    hate[b-1][a-1]=1
ans=[0]*n
for i in range(n):
    for j in range(n):
        if i==j or hate[i][j]:continue
        ans[i]+=min(score[i][0]+score[j][1],score[j][0]+score[i][1])
print(*ans)
