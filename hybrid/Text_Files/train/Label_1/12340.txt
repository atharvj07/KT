#include <bits/stdc++.h>
using namespace std;
#define fst first
#define snd second

int main() {
    int n, q, x, y;
    cin >> n >> q;
    for (int i = 0; i < q; i++) {
        cin >> x >> y;
        if ((x + y) % 2 == 0)
            cout << (x + y) * (x + y) / 4 - x + n;
        else
            cout << (x + y) * (x + y) / 4 + y + n;
        cout << endl;
    }
    return 0;
}
