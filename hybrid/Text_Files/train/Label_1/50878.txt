import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), p = sc.nextInt() ;
        int[] t = new int[n+1];
        for (int i = 1; i <= n; i++)
            t[i] = sc.nextInt();
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        int[] ans = new int[n+1];
        for (int i = 1, tm = 0; i <= n; i++) {
            tm = Math.max(tm, t[i]);
            while (!pq.isEmpty() && pq.peek() <= tm)
                pq.poll();
            pq.add(tm + p);
            ans[i] = tm;
            tm++;
        }
        for (int i = 1; i <= n; i++)
            System.out.println(ans[i]);
    }
}
