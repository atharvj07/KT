#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 1e5 + 5;
int n, m, k;
vector<int> b;

int main() {
    cin >> n >> m >> k;
    b.resize(n);
    for (int &x : b) { cin >> x; }
    if (n > k) {
        ll ans = 0;
        for (int i = 1; i <= n - k; i++) { ans += b[i + k - 1] - b[i]; }
        cout << ans << endl;
    } else {
        int cur_r = 0;
        ll ans = 0;
        for (int i = 1; i <= k; i++) {
            int r = lower_bound(b.begin(), b.end(), m + 1) - b.begin();
            if (r <= cur_r) {
                ans = -1;
                break;
            }
            ans += b[r] - b[cur_r + 1];
            cur_r = r;
        }
        cout << ans << endl;
    }
    return 0;
}
