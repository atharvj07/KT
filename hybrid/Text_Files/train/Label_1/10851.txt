#include <iostream>
#include <limits.h>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> h(n);
  for (int i = 0; i < n; i++) cin >> h[i];
  long long sum = 0;
  for (int i = 0; i < k; i++) sum += h[i];
  long long min_sum = sum;
  int min_idx = 0;
  for (int i = k; i < n; i++) {
    sum += h[i] - h[i - k];
    if (sum < min_sum) {
      min_sum = sum;
      min_idx = i - k + 1;
    }
  }
  cout << min_idx + 1 << endl;
  return 0;
}
