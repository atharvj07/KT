#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    string s, t;
    cin >> s >> t;
    if (s == t)
      cout << "YES\n";
    else {
      int sum1 = 0, sum2 = 0;
      for (int i = 0; s[i]; i++) {
        sum1 += s[i] - 'a';
        sum2 += t[i] - 'a';
      }
      if (sum1 == sum2)
        cout << "YES\n";
      else
        cout << "NO\n";
    }
  }
}
