#include <iostream>
#include <string>

using namespace std;

int diff(string s, int k) {
  int ans = 0, sum = 0;
  for (int i = 0; i < (int)s.size(); i++) {
    sum += s[i] - '0';
    ans += s[i] != '0' && sum < k;
  }
  return ans;
}

int main() {
  int k;
  string s;
  cin >> k >> s;
  cout << diff(s, k) << endl;
  return 0;
}
