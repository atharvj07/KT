#include <iostream>
#include <algorithm>
using namespace std;
struct doc{
    int s, d;
};
bool cmp(doc a, doc b){
    return a.s < b.s;
}
int main(){
    int n;
    cin >> n;
    doc v[n];
    for (int i = 0; i < n; ++i) cin >> v[i].s >> v[i].d;
    sort(v, v + n, cmp);
    int day = v[0].s;
    for (int i = 0; i < n; ++i){
        if ((day - v[i].s) % v[i].d == 0) continue;
        day = v[i].s + v[i].d - (day - v[i].s) % v[i].d;
    }
    cout << day << endl;
    return 0;
}
