n,a,b=map(int,input().split())
l=[list(map(int,input().split())) for x in range(n)]
l.sort(key=lambda x:x[0],reverse=1)
for x in l:
    if a and x[0]<x[1]:
        x[0],x[1]=x[1],x[0]
        a-=1
for x in l:
    if b and x[0]>x[1]:
        x[0],x[1]=x[1],x[0]
        b-=1
print(sum(x[1]for x in l))
