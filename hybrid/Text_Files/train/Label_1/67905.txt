n, p = map(int, input().split())
devices = [list(map(int, input().split())) for _ in range(n)]

total_power_needed = sum(a for a, _ in devices)
if total_power_needed <= p:
    print(-1)
else:
    left, right = 0, 1e18
    while right - left > 1e-7:
        mid = (left + right) / 2
        total_extra_power_needed = sum(max(0, a * mid - b) for a, b in devices)
        if total_extra_power_needed > p * mid:
            right = mid
        else:
            left = mid
    print(f'{left:.10f}')
