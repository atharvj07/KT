import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (a[j] < x) {
                    int diff = x - a[j];
                    if (diff % 2 == 0) {
                        if (cnt % 2 == 0) {
                            cnt += diff / 2;
                        } else {
                            cnt += diff / 2 + 1;
                        }
                    } else {
                        if (cnt % 2 == 0) {
                            cnt += diff / 2 + 1;
                        } else {
                            cnt += diff / 2;
                        }
                    }
                    a[j] = x;
                }
            }
            System.out.println(cnt);
        }
    }
}
