#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>
using namespace std;

int n, a[20000];

bool check(int x, int y) {
    int cnt = 0;
    for (int i = x; i <= y; i++)
        if (a[i] > 1)
            cnt++;
    return cnt <= 1;
}

bool ok() {
    vector<pair<int, pair<int, int>>> b;
    for (int i = 1; i <= n * n; i++)
        b.push_back({a[i], {0, i}});
    sort(b.begin(), b.end());
    int l = 1, r = n * n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (l <= r && b[l].second.first == i && b[l].second.second == j)
                l++;
            else if (l <= r && b[r].second.first == i && b[r].second.second == j)
                r--;
            else
                return false;
        }
    }
    return true;
}

void solve(int x) {
    if (x == n * n + 1) {
        if (ok()) {
            cout << "YES\n";
            vector<pair<int, pair<int, int>>> b;
            for (int i = 1; i <= n * n; i++)
                b.push_back({a[i], {0, i}});
            sort(b.begin(), b.end());
            int l = 1, r = n * n;
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (l <= r && b[l].second.first == i && b[l].second.second == j) {
                        cout << b[l].first << " ";
                        l++;
                    } else if (l <= r && b[r].second.first == i && b[r].second.second == j) {
                        cout << b[r].first << " ";
                        r--;
                    } else
                        return;
                }
                cout << "\n";
            }
            exit(0);
        }
        return;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (check(x, i * n + j)) {
                a[i * n + j]++;
                solve(x + 1);
                a[i * n + j]--;
            }
        }
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n * n; i++)
        cin >> a[i];
    solve(1);
    cout << "NO\n";
    return 0;
}
