import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x2 = scanner.nextInt();
        if (x2 % 2 == 0) {
            System.out.println(x2 / 2);
        } else if (x2 % 3 == 0) {
            System.out.println(x2 / 3);
        } else {
            System.out.println(x2 - 1);
        }
    }
}
