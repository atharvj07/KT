#include <iostream>
#include <array>
#include <vector>
#include <queue>

using namespace std;

const int MAXN = 510;
const int MAXM = 510;

int main() {
  int n, m;
  cin >> n >> m;
  bool ok = true;
  int ans = 0;
  array<array<int, MAXM>, MAXN> a;
  vector<vector<int>> p(n, vector<int>(m, -1));
  vector<int> d(n + m);
  vector<bool> inq(n + m, false);
  queue<int> q;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
      if (a[i][j] == 0) {
        ok = ok && (i > 0 && j > 0 && i < n - 1 && j < m - 1);
        q.push(n + j);
        q.push(i);
        d[n + j] = 0;
        d[i] = 0;
        inq[n + j] = inq[i] = true;
      }
      if (i > 0) p[i][j] = n + j;
      if (j > 0) p[i][j] = i;
    }
  }
  while (!q.empty()) {
    int k = q.front();
    q.pop();
    inq[k] = false;
    if (k < n) {
      for (int j = 0; j < m; j++) {
        if (a[k][j] != 0 && p[k][j] != -1 &&
            d[k] + 1 > d[p[k][j]]) {
          d[p[k][j]] = d[k] + 1;
          if (!inq[p[k][j]]) {
            q.push(p[k][j]);
            inq[p[k][j]] = true;
          }
        }
      }
    } else {
      for (int i = 0; i < n; i++) {
        if (a[i][k - n] != 0 && p[i][k - n] != -1 &&
            d[k] + 1 > d[p[i][k - n]]) {
          d[p[i][k - n]] = d[k] + 1;
          if (!inq[p[i][k - n]]) {
            q.push(p[i][k - n]);
            inq[p[i][k - n]] = true;
          }
        }
      }
    }
  }
  if (!ok) {
    cout << "-1" << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] == 0) {
        ans += (d[i] + d[n + j] - d[p[i][j]]) / 2 + 1;
      } else
        ans += a[i][j];
    }
  }
  cout << ans << endl;
  return 0;
}
