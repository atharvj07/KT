import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] cnt = new int[m];
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
                cnt[j] += a[i][j];
            }
        }
        int ans = -1;
        for (int j = 0; j < m; j++) {
            if (cnt[j] == n) {
                ans = 0;
            } else if (cnt[j] == 0) {
                ans = -1;
            } else {
                int can = n - cnt[j];
                if (ans == -1) {
                    ans = can;
                } else {
                    ans = Math.min(ans, can);
                }
            }
        }
        System.out.println(ans);
    }
}
