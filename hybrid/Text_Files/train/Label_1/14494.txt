#include<bits/stdc++.h>
using namespace std;
int main(){
	int t; cin >> t;
	while(t--){
		int n; cin >> n;
		bool pal = false;
		map<int,int> mp;
		for(int i = 0; i < n; i++){
			int x; cin >> x;
			mp[x]++;
		}
		for(auto x : mp){
			int cnt = x.second;
			pal |= cnt >= 3 || (cnt >= 2 && mp.find(x.first) != mp.end());
		}
		cout << (pal ? "YES" : "NO") << endl;
	}
}
