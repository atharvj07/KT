import sys
n, m = map(int, input().split())
g = [[] for i in range(n)]
for _ in range(m):
    a, b = map(int, input().split())
    a, b = a-1, b-1
    g[a].append(b)
    g[b].append(a)
oddeg = [0 for i in range(n)]
for i in range(n):
    for j in g[i]:
        oddeg[i] += (j < i)
oddcy = 0
for i in range(n):
    oddcy += (oddeg[i] % 2)
if oddcy <= 2:
    t, w = 0, 1
elif oddcy == n:
    t, w = (n-1)//2, 1
else:
    t = n-oddcy+1
    w = 1
    for i in range(n):
        if oddeg[i]%2:
            for j in g[i]:
                if not oddeg[j]%2:
                    w *= 2
                    break
print(t, w)
