import java.io.*;
import java.util.*;

class Main {
    public static class Dict {
        private final NavigableMap<String, List<Integer>> dict = new TreeMap<>();

        public void put(String key, int val) {
            List<Integer> list = dict.get(key);
            if (list == null) {
                list = new ArrayList<>();
                dict.put(key, list);
            }
            list.add(val);
        }

        public List<Integer> get(String key) {
            return dict.get(key);
        }

        public void remove(String key) {
            dict.remove(key);
        }

        public NavigableMap<String, List<Integer>> range(String from, String to) {
            return dict.subMap(from, true, to, true);
        }

        @Override
        public String toString() {
            return String.valueOf(dict);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        Dict dict = new Dict();
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            switch (type) {
                case 0:
                    dict.put(sc.next(), sc.nextInt());
                    break;
                case 1:
                    List<Integer> vals = dict.get(sc.next());
                    if (vals != null) {
                        for (Integer val : vals) {
                            sb.append(val).append('\n');
                        }
                    }
                    break;
                case 2:
                    dict.remove(sc.next());
                    break;
                case 3:
                    String from = sc.next();
                    String to = sc.next();
                    NavigableMap<String, List<Integer>> range = dict.range(from, to);
                    for (Map.Entry<String, List<Integer>> entry : range.entrySet()) {
                        for (Integer val : entry.getValue()) {
                            sb.append(entry.getKey()).append(' ').append(val).append('\n');
                        }
                    }
                    break;
            }
        }
        System.out.println(sb);
    }
}
