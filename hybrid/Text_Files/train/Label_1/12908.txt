#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll maxn = 1e5 + 5;
ll n, m, u[maxn], deg[maxn];
vector<ll>adj[maxn];
map<pair<ll, ll>, bool>mp;
void dfs(ll x, ll p) {
    u[x] = 1;
    for (auto y : adj[x]) {
        if (y == p) continue;
        if (u[y] == 0) {
            dfs(y, x);
        }
        else if (u[y] == 1) {
            cout << 1 << " " << 2 << endl;
            exit(0);
        }
    }
    u[x] = 2;
}
int main() {
    cin >> n >> m;
    for (ll i = 0; i < m; i++) {
        ll x, y; cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
        mp[{x, y}] = 1;
        mp[{y, x}] = 1;
        deg[x]++;
        deg[y]++;
    }
    ll ans1 = n - 1, ans2 = 1;
    for (ll i = 1; i <= n; i++) {
        if (deg[i] % 2) {
            ans1 = max(ans1, (deg[i] + 1) / 2);
            ans2 *= deg[i] / 2;
        }
    }
    for (ll i = 1; i <= n; i++) {
        if (u[i] == 0) {
            dfs(i, -1);
        }
    }
    cout << ans1 << " " << ans2 << endl;
}
