import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    x = list(map(int, input().split()))
    c = [0] * m
    s = 0
    for i in range(n):
        s += a[i]
        for j in range(m):
            if s >= x[j]:
                c[j] = i + 1
                s -= x[j]
    for v in c:
        if v: print(v)
        else: print(-1)
