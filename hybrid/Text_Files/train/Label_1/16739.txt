#include <bits/stdc++.h>
using namespace std;
const int N = 1005;
int dp[N][N], a[5][N];
int32_t main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= n; j++) { cin >> a[i][j]; }
    }
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= n; j++) {
            for (int z = 1; z <= k; z++) {
                if (z == i) continue;
                if (a[i][j] != a[z][j]) {
                    dp[i][j] = max(dp[i][j], dp[z][j - 1] + 1);
                }
            }
        }
    }
    int ans = 0;
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= n; j++) { ans = max(ans, dp[i][j]); }
    }
    cout << ans << endl;
}
