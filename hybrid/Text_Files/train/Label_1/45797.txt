import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = scanner.nextInt();
        }
        scanner.close();

        int[] sorted = w.clone();
        Arrays.sort(sorted);

        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (w[i] != sorted[i]) {
                for (int j = i + 1; j < n; j++) {
                    if (w[j] == sorted[i]) {
                        ans += (long) (w[i] + w[j]);
                        w[i] = w[j];
                        w[j] = sorted[i];
                        break;
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
