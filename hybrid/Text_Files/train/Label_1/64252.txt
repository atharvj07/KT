from math import gcd

def solve(r, b, k):
    if r > b:
        r, b = b, r
    g = gcd(r, b)
    if (b - g + r - 1) // r < k:
        return "OBEY"
    else:
        return "REBEL"

T = int(input().strip())
for _ in range(T):
    r, b, k = map(int, input().strip().split())
    print(solve(r, b, k))
