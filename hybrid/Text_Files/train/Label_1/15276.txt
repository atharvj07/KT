#include <iostream>
int main() {
    long long n, k, l, r, a, ans(0), now(0);
    std::cin >> n >> k;
    while (n--) {
        std::cin >> l >> r >> a;
        if (l >= now) { now = l; } else { ans += (now - l + k - 1) / k; }
        now += a;
    }
    if (now > ans * k) ++ans;
    std::cout << ans * k << "\n";
    return 0;
}
