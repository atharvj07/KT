#include <iostream>

const int MOD = 1e9 + 7;

using namespace std;

template <typename T> inline bool chmin(T &x, const T &y) {
  if (x > y) {
    x = y;
    return true;
  }
  return false;
}

long long power(long long x, long long y, long long mod) {
  if (y == 0) {
    return 1 % mod;
  }
  if (y % 2 == 1) {
    return (x * power((x * x) % mod, y / 2, mod)) % mod;
  }
  return power((x * x) % mod, y / 2, mod) % mod;
}

int main() {
  long long n, x, a, sum_a;
  cin >> n >> x;
  sum_a = 0;
  for (int i = 0; i < n; i++) {
    cin >> a;
    sum_a += a;
  }
  cout << power(x, sum_a, MOD) << endl;
}
