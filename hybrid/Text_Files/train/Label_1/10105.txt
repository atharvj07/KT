#include <iostream>
#include <vector>
#include <map>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> f(n + 1), b(m);
  for (int i = 1; i <= n; i++) { cin >> f[i]; }
  for (int i = 0; i < m; i++) { cin >> b[i]; }
  map<int, int> cnt;
  for (int i = 1; i <= n; i++) { cnt[f[i]]++; }
  for (int i = 0; i < m; i++) {
    cnt[b[i]]--;
    if (cnt[b[i]] < 0) {
      cout << "Impossible";
      return 0;
    }
  }
  for (map<int, int>::iterator it = cnt.begin(); it != cnt.end(); it++) {
    if (it->second != 0) {
      cout << "Ambiguity";
      return 0;
    }
  }
  cout << "Possible\n";
  for (int i = 0; i < m; i++) {
    for (int j = 1; j <= n; j++) {
      if (f[j] == b[i]) {
        cout << j << " ";
        break;
      }
    }
  }
  return 0;
}
