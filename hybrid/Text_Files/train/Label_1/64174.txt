n, m, d = map(int, input().split())
c = list(map(int, input().split()))
a = [0] * n
if sum(c) <= d or d == 1000: print('YES'); print(*a); exit()
for k in range(1, m + 1):
    mn = 10000
    for i in range(n - c[k - 1]):
        if sum(a[i:i + c[k - 1]]) == 0: mn = i; break
    if mn == 10000: print('NO'); exit()
    for i in range(mn, mn + c[k - 1]): a[i] = k
print('YES'); print(*a)
