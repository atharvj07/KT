import java.util.Scanner;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] victor = new int[2];
        int[] peter = new int[2];
        int[] wall = new int[4];
        int[] mirror = new int[4];
        for (int i = 0; i < 2; i++) {
            victor[i] = in.nextInt();
            peter[i] = in.nextInt();
        }
        for (int i = 0; i < 4; i++) {
            wall[i] = in.nextInt();
            mirror[i] = in.nextInt();
        }
        List<Segment> obstacles = new ArrayList<>();
        obstacles.add(new Segment(new Point(wall[0], wall[1]), new Point(wall[2], wall[3])));
        obstacles.add(new Segment(new Point(mirror[0], mirror[1]), new Point(mirror[2], mirror[3])));
        if (checkVisibility(obstacles, victor, peter)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean checkVisibility(List<Segment> obstacles, int[] victor, int[] peter) {
        return new Line(new Point(victor[0], victor[1]), new Point(peter[0], peter[1])).checkVisibility(obstacles);
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Line {
        Point p1;
        Point p2;

        public Line(Point p1, Point p2) {
            this.p1 = p1;
            this.p2 = p2;
        }

        public boolean checkVisibility(List<Segment> obstacles) {
            for (Segment obstacle : obstacles) {
                if (obstacle.crosses(this)) {
                    return false;
                }
                if (obstacle.reflects(this)) {
                    return true;
                }
            }
            return true;
        }

        private boolean crosses(Segment segment) {
            return (ccw(p1, p2, segment.p1) * ccw(p1, p2, segment.p2) <= 0) && (ccw(segment.p1, segment.p2, p1) * ccw(segment.p1, segment.p2, p2) <= 0);
        }

        private boolean reflects(Line line) {
            return ccw(p1, p2, line.p1) * ccw(p1, p2, line.p2) >= 0 && ccw(line.p1, line.p2, p1) * ccw(line.p1, line.p2, p2) >= 0;
        }

        private int ccw(Point p1, Point p2, Point p3) {
            return (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
        }
    }

    private static class Segment extends Line {
        public Segment(Point p1, Point p2) {
            super(p1, p2);
        }

        public boolean crosses(Line line) {
            return line.crosses(this);
        }

        public boolean reflects(Line line) {
            return line.reflects(this);
        }
    }
}
