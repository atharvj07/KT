import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        double[] xyz = new double[12];
        double[] vals = new double[3];
        for (int i = 0; i < 3; i++) {
            vals[i] = scanner.nextDouble();
        }
        for (int i = 0; i < 12; i++) {
            xyz[i] = Math.pow(vals[(i / 4) % 3], Math.pow(vals[(i / 2) % 3], vals[i % 3]));
        }
        double max = Double.MIN_VALUE;
        int index = -1;
        for (int i = 0; i < 12; i++) {
            if (xyz[i] > max) {
                max = xyz[i];
                index = i;
            }
        }
        switch (index) {
            case 0:
                System.out.println("x^y^z");
                break;
            case 1:
                System.out.println("xzy");
                break;
            case 2:
                System.out.println("(x^y)^z");
                break;
            case 3:
                System.out.println("(xz)^y");
                break;
            case 4:
                System.out.println("yxz");
                break;
            case 5:
                System.out.println("yzx");
                break;
            case 6:
                System.out.println("(y^x)^z");
                break;
            case 7:
                System.out.println("(y^z)^x");
                break;
            case 8:
                System.out.println("zxy");
                break;
            case 9:
                System.out.println("zyx");
                break;
            case 10:
                System.out.println("(z^x)^y");
                break;
            case 11:
                System.out.println("(z^y)^x");
                break;
        }
    }
}
