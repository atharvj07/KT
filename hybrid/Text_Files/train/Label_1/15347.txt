#include <bits/stdc++.h>
using namespace std;
int a[100];
int main() {
    int n, j = 0, k = 0;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] < 0) j++;
        else if (a[i] > 0) k++;
    }
    cout << j << " ";
    for (int i = 0; i < n; i++)
        if (a[i] < 0) cout << a[i] << " ";
    cout << endl;
    cout << k << " ";
    for (int i = 0; i < n; i++)
        if (a[i] > 0) cout << a[i] << " ";
    cout << endl;
    cout << n - j - k << " ";
    for (int i = 0; i < n; i++)
        if (a[i] == 0) cout << a[i] << " ";
    cout << endl;
    return 0;
}
