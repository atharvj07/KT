import java.util.*;
import java.io.*;

class Main {
    static int h1, a1, c1, h2, a2;
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        h1 = Integer.parseInt(st.nextToken());
        a1 = Integer.parseInt(st.nextToken());
        c1 = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(in.readLine());
        h2 = Integer.parseInt(st.nextToken());
        a2 = Integer.parseInt(st.nextToken());
        int strike = (h2 + a2 - 1) / a1;
        int heal = h1 + c1 - 1 != 0 ? (h2 + a2 - 1) / (h1 + c1 - 1) : 0;
        System.out.println(strike + heal);
        for (int i = 0; i < strike; i++) System.out.println("STRIKE");
        for (int i = 0; i < heal; i++) System.out.println("HEAL");
    }
}
