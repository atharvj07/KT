#include <iostream>
#include <vector>
#include <tuple>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int m, k;
    cin >> m >> k;
    vector<int> a(k);
    for (int i = 0; i < k; i++) cin >> a[i];
    vector<tuple<int, int>> obs;
    for (int i = 0; i < m - 1; i++) {
      int x, y;
      cin >> x >> y;
      obs.emplace_back(x, y);
    }
    vector<char> ans(k, 'N');
    for (auto &[d, l] : obs) {
      if (!l) continue;
      a[d - 1]--;
      if (a[d - 1] < 0) ans[d - 1] = 'Y';
    }
    for (char c : ans) cout << c;
    cout << '\n';
  }
  return 0;
}
