#include <iostream>

using namespace std;

int main() {
	long long a, b, c;
	cin >> a >> b >> c;
	long long t = max(a / c, b / c);
	long long s = c;
	if (a / c != b / c) {
		if (a / c < b / c) {
			if (a % c == 0) {
				t = a / c;
				s = b / c - a / c;
			} else {
				t = a / c + 1;
				if (c * t <= a + b) {
					s = b / c - t;
				}
			}
		} else {
			if (b % c == 0) {
				t = b / c;
				s = a / c - b / c;
			} else {
				t = b / c + 1;
				if (c * t <= a + b) {
					s = a / c - t;
				}
			}
		}
	}
	cout << t << ' ' << s << '\n';
	return 0;
}
