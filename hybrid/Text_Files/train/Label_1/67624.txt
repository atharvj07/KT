from math import factorial
from itertools import permutations

def ord_permutation(p):
    n = len(p)
    order = 0
    fact = factorial(n)
    used = [False] * n

    for i in range(n):
        fact //= n - i
        count = 0
        for j in range(p[i]):
            if not used[j]:
                count += 1
        order += count * fact
        used[p[i]] = True

    return order

def perm(n, k):
    seq = list(range(n))
    fact = factorial(n)
    k %= fact  # Ensure k is in a valid range
    result = []
    for i in reversed(range(n)):
        fact //= n - i
        index = k // fact
        k %= fact
        result.append(seq.pop(index))

    return result

n = int(input())
p = list(map(int, input().split()))
q = list(map(int, input().split()))

p_order = ord_permutation(p)
q_order = ord_permutation(q)

result = perm(n, p_order + q_order)
print(' '.join(map(str, result)))
