import java.util.Arrays;
import java.util.Scanner;
class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        long[] c = new long[k];
        for (int i = 0; i < k; i++) {
            c[i] = scanner.nextInt();
        }
        Arrays.sort(c);
        long[] f = new long[k];
        for (int i = 0; i < k; i++) {
            if (i == 0) {
                f[i] = 1;
            } else {
                for (int j = i; j >= 0; j--) {
                    if (c[j] == c[i]) {
                        f[i] += f[j];
                        f[i] %= MOD;
                    }
                }
            }
        }
        long ans = 0;
        for (int i = 0; i < k; i++) {
            ans += f[i];
            ans %= MOD;
        }
        System.out.println(ans);
    }
}
