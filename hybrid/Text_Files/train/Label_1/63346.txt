from collections import Counter

n, m = map(int, input().split())
f = list(map(int, input().split()))
b = list(map(int, input().split()))

fc = Counter(f)
bc = Counter(b)
if fc != bc:
    print("Impossible")
else:
    a = [-1] * m
    for i in range(m):
        for j in range(n):
            if f[j] == b[i] and a[i] == -1:
                a[i] = j + 1
                break
    if -1 in a:
        print("Ambiguity")
    else:
        print("Possible")
        print(' '.join(map(str, a)))
