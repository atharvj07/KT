n, k = map(int, input().split())
a = list(map(int, input().split()))
ans = ['0'] * n
a_r = [0] * (n + 2 * k)

for i in range(n):
    a_r[i + k] = a[i]

for i in range(n // (2 * k + 1)):
    m = a_r[(2 * k + 1) * i + k:].index(max(a_r[(2 * k + 1) * i + k:]))
    for j in range(m - k, m + k + 1):
        ans[a[j] - 1] = '1'
        a_r[j + k] = 0

print(''.join(ans))
