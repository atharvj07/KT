import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] players = new int[4][2];
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 2; j++) {
                players[i][j] = sc.nextInt();
            }
        }
        int[] team1 = {0, 1};
        int[] team2 = {2, 3};
        if (players[team1[0]][0] > players[team2[1]][1] && players[team1[1]][1] > players[team2[0]][0]) {
            System.out.println("Team 1");
        } else if (players[team2[0]][0] > players[team1[1]][1] && players[team2[1]][1] > players[team1[0]][0]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
