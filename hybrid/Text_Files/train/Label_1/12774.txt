#include <bits/stdc++.h>
using namespace std;

int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        if (n % 3) {
            cout << -1 << '\n';
        } else {
            int a = n / 3;
            int b = 0;
            int c = 0;
            while (a >= 0) {
                b = (n - a * 3) / 5;
                c = (n - a * 3 - b * 5) / 7;
                if (a * 3 + b * 5 + c * 7 == n) {
                    cout << a << ' ' << b << ' ' << c << '\n';
                    break;
                }
                a--;
            }
            if (a < 0) cout << -1 << '\n';
        }
    }
}
