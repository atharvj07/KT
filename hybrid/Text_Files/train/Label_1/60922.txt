import sys

n, k = map(int, input().split())
data = []
for _ in range(n):
    a, b = map(int, input().split())
    data.append((a, b))
data.sort()
ans = []
mx = 0
for i in range(n):
    l, r = data[i]
    cnt = 1
    for j in range(i + 1, n):
        nl, nr = data[j]
        if nr >= l and nl <= r:
            cnt += 1
    if cnt > mx:
        mx = cnt
        ans = [i + 1]
    elif cnt == mx:
        ans.append(i + 1)

print(mx)
print(*ans[:k])
