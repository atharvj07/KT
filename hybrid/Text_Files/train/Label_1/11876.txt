#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    int ans = 0, mxdif = 0;
    for (int i = 0, x, y; i < n; i++) {
        cin >> x >> y;
        if (abs(x - y) > mxdif) {
            mxdif = abs(x - y);
            ans = i + 1;
        }
    }
    cout << ans << endl;
    return 0;
}
