import java.util.*;

class Main {
    static int N;
    static long[] X;
    static long[] W;
    static List<Integer>[] G;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        X = new long[N + 1];
        W = new long[N + 1];
        G = new List[N + 1];
        for (int i = 0; i <= N; i++) {
            G[i] = new ArrayList<>();
        }
        for (int i = 2; i <= N; i++) {
            int p = sc.nextInt();
            G[p].add(i);
        }
        for (int i = 1; i <= N; i++) {
            X[i] = sc.nextInt();
        }
        dfs(1, 1);
        for (int i = 1; i <= N; i++) {
            if (X[i] != W[i]) {
                System.out.println("IMPOSSIBLE");
                return;
            }
        }
        System.out.println("POSSIBLE");
    }

    static long dfs(int v, int p) {
        long total = 0;
        for (int next : G[v]) {
            total += dfs(next, v);
        }
        W[v] = total + 1;
        return total + 1;
    }
}
