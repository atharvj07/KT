#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int MOD = 998244353;
const int MAXN = 2e5 + 5;
vector<int> adj[MAXN];
ll cost[MAXN], a[MAXN], b[MAXN], c[MAXN];
ll dp[MAXN][2][2];

ll dfs(int u, int p, int t) {
  if (dp[u][b[u]][t] != -1) return dp[u][b[u]][t];
  ll &res = dp[u][b[u]][t];
  res = LLONG_MAX;
  if (t == 1 && b[u] == c[u]) res = 0;
  for (int v : adj[u]) {
    if (v == p) continue;
    res = min(res, dfs(v, u, 0) + cost[u]);
    res = min(res, dfs(v, u, 1) + cost[u] * a[u]);
  }
  return res;
}

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i] >> c[i];
  }
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  memset(dp, -1, sizeof dp);
  ll ans = dfs(1, 1, 0);
  if (ans == LLONG_MAX) ans = -1;
  cout << ans << endl;
  return 0;
}
