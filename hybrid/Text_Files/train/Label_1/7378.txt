#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 100005;
#define SZ(x) (int)(x).size()

ll n, l, r, ql, qr;
ll w[N], dp[2][N];
int main(){
    cin >> n >> l >> r >> ql >> qr;
    for (ll i = 1; i <= n; i++) cin >> w[i];
    if (l > r){
        swap(l, r);
        swap(ql, qr);
    }
    memset(dp, 0x3f, sizeof dp);
    dp[0][0] = dp[1][0] = 0;
    for (ll i = 1; i <= n; i++){
        dp[0][i] = min(dp[0][i - 1] + w[i] * l + ql, dp[1][i - 1] + w[i] * r + qr);
        dp[1][i] = min(dp[1][i - 1] + w[i] * r + qr, dp[0][i - 1] + w[i] * l + ql);
    }
    cout << dp[0][n] << endl;
    return 0;
}
