#include <iostream>
#include <string>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int sh, sm, th, tm;
  sh = (s[0]-'0')*10 + (s[1]-'0');
  sm = (s[3]-'0')*10 + (s[4]-'0');
  th = (t[0]-'0')*10 + (t[1]-'0');
  tm = (t[3]-'0')*10 + (t[4]-'0');
  if (sm < tm) {
    sh--;
    sm += 60;
  }
  sh -= th;
  sm -= tm;
  if (sh < 0) sh += 24;
  if (sm < 0) sm += 60;
  cout.width(2);
  cout.fill('0');
  cout << sh << ':';
  cout.width(2);
  cout.fill('0');
  cout << sm << endl;
  return 0;
}
