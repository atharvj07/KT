#include <vector>
#include <iostream>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }

  std::sort(a.begin(), a.end());
  int ans = 0;
  bool bought = false;
  for (int i = 0; i < n; ++i) {
    if (a[i] > 0) {
      if (bought && a[i] <= a[i - 1]) {
        break;
      }
      ans += a[i];
      bought = true;
    }
  }

  std::cout << ans << std::endl;

  return 0;
}
