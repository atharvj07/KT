import java.util.*;
import java.io.*;

public class Main {
    static int n, k;
    static int[] cards;
    static boolean[] used;
    static Set<Long> ans;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        while (true) {
            n = Integer.parseInt(br.readLine());
            if (n == 0) break;
            k = Integer.parseInt(br.readLine());
            cards = new int[n + 1];
            used = new boolean[n + 1];
            for (int i = 1; i <= n; i++) cards[i] = Integer.parseInt(br.readLine());
            ans = new HashSet<>();
            dfs(0, 0L);
            System.out.println(ans.size());
        }
    }

    public static void dfs(int depth, long cur) {
        if (depth == k) {
            ans.add(cur);
            return;
        }
        for (int i = 1; i <= n; i++) {
            if (used[i]) continue;
            used[i] = true;
            dfs(depth + 1, cur * 100 + cards[i]);
            used[i] = false;
        }
    }
}
