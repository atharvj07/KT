import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long b = sc.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        Arrays.sort(a);
        int ind = 0;
        while (k > 0) {
            if (a[ind] <= b) {
                ind++;
                b -= a[ind - 1];
            } else break;
            k--;
        }
        System.out.println(ind + 1);
    }
}
