import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        boolean ok = true;
        while (ok) {
            ok = false;
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 1) {
                    ans = 0;
                    break;
                }
                a[i] /= 2;
                ok = true;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
