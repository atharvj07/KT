n,k1,k2=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
a.sort();b.sort()
ans=0
for i in range(min(k1,n-1)):
    if a[i]+1<b[i]:
        ans+=b[i]-(a[i]+1)
        a[i]+=1
        k1-=1
    elif a[i]>b[i]+1:
        ans+=a[i]-(b[i]+1)
        a[i]-=1
        k1-=1
    else: break
for i in range(n-1,max(0,n-1-k2),-1):
    if b[i]-1>a[i]:
        ans+=a[i]-(b[i]-1)
        b[i]-=1
        k2-=1
    elif b[i]<a[i]-1:
        ans+=b[i]-(a[i]-1)
        b[i]+=1
        k2-=1
    else: break
print(ans)
