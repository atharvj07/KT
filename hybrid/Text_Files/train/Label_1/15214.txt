#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> grid(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> grid[i][j];
    }
  }
  int ans = 0;
  bool found = false;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == 'D') {
        int cnt = 1;
        int x = i, y = j;
        while (grid[x][y] != 'D' || x != i || y != j) {
          if (x + 1 < n && grid[x + 1][y] == 'I') {
            x++;
            cnt++;
          } else if (y + 1 < m && grid[x][y + 1] == 'M') {
            y++;
            cnt++;
          } else if (x - 1 >= 0 && grid[x - 1][y] == 'A') {
            x--;
            cnt++;
          } else if (y - 1 >= 0 && grid[x][y - 1] == 'D') {
            y--;
            cnt++;
          } else {
            break;
          }
        }
        if (grid[x][y] == 'D' && x == i && y == j) {
          found = true;
          ans = max(ans, cnt / 4);
        }
      }
    }
  }
  if (!found) {
    cout << "Poor Dima!\n";
  } else if (ans == 0) {
    cout << "Poor Inna!\n";
  } else {
    cout << ans << "\n";
  }
  return 0;
}
