def bfs(graph, node, n):
    visited = [False] * (n + 1)
    queue = []
    visited[node] = True
    queue.append(node)
    depth = [0] * (n + 1)
    
    while queue:
        s = queue.pop(0)
        
        for i in graph[s]:
            if not visited[i]:
                depth[i] = depth[s] + 1
                queue.append(i)
                visited[i] = True
    return depth

n = int(input())
edges = []
graph = [[] for _ in range(n + 1)]

for _ in range(n - 1):
    u, v = map(int, input().split())
    edges.append((u, v))
    graph[u].append(v)
    graph[v].append(u)

depth = bfs(graph, 1, n)
edge_labels = []

for u, v in edges:
    if depth[u] > depth[v]:
        edge_labels.append((depth[u], 0))
    else:
        edge_labels.append((depth[v], 0))

edge_labels.sort(reverse=True)

for i in range(len(edge_labels)):
    edge_labels[i] = (edge_labels[i][0], i)

edge_labels.sort()

for label in edge_labels:
    print(label[1])
