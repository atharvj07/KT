import sys
input = sys.stdin.readline
mod = 1000000007

def gcd(a, b):
    if a == 0:
        return b
    return gcd(b % a, a)

n = int(input())
a = list(map(int, input().split()))
a.sort()
ans = 0
for i in range(n):
    for j in range(i + 1, n):
        g = gcd(a[i], a[j])
        ans = (ans + (j - i + 1) * g) % mod
print(ans)
