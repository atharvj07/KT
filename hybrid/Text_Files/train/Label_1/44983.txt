import java.util.*;
import java.io.*;

class Main {
    static int n;
    static int[] a;
    static int[] b;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String[] ss = in.nextLine().split(" ");
        a = new int[ss[0].length()];
        Arrays.fill(a, -1);
        for (int i = 0; i < ss[0].length(); i++) {
            a[i] = ss[0].charAt(i) - '0';
        }
        ss = in.nextLine().split(" ");
        b = new int[ss[0].length()];
        Arrays.fill(b, -1);
        for (int i = 0; i < ss[0].length(); i++) {
            b[i] = ss[0].charAt(i) - '0';
        }
        n = a.length;
        int[][] dp = new int[n][n];
        dp[n - 1][n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            dp[i][n - 1] = 1;
            if (b[n - 1] == 1) {
                dp[i][n - 1] = dp[i + 1][n - 1];
            }
        }
        for (int j = n - 2; j >= 0; j--) {
            dp[n - 1][j] = 1;
            if (a[n - 1] == 1) {
                dp[n - 1][j] = dp[n - 1][j + 1];
            }
        }
        for (int i = n - 2; i >= 0; i--) {
            for (int j = n - 2; j >= 0; j--) {
                if (a[i] == 0 && b[j] == 0) {
                    dp[i][j] = 0;
                } else {
                    dp[i][j] = Math.min(dp[i + 1][j], dp[i][j + 1]) + 1;
                }
            }
        }
        int a_ptr = 0;
        int b_ptr = 0;
        StringBuilder sb = new StringBuilder();
        while (a_ptr < n && b_ptr < n) {
            if (a[a_ptr] == 1 && b[b_ptr] == 1) {
                a_ptr++;
                b_ptr++;
                continue;
            }
            if (dp[a_ptr][b_ptr] == dp[a_ptr][b_ptr + 1]) {
                sb.append('0');
                b_ptr++;
            } else {
                sb.append('0');
                a_ptr++;
            }
        }
        while (a_ptr < n) {
            sb.append('0');
            a_ptr++;
        }
        while (b_ptr < n) {
            sb.append('0');
            b_ptr++;
        }
        System.out.println(sb.toString());
    }
}
