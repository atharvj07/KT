import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n * 2];
        for (int i = 0; i < a.length; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int minX = a[0], maxX = a[n * 2 - 1];
        int minY = a[1], maxY = a[n * 2 - 2];
        int area = (maxX - minX) * (maxY - minY);
        System.out.println(area);
    }
}
