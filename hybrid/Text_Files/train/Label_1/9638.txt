#include <iostream>
using namespace std;
typedef long long ll;
int main() {
    ll n, m, a, b;
    cin >> n >> m >> a >> b;
    if (n % m == 0) {
        cout << 0 << endl;
        return 0;
    }
    ll x = n / m + 1;
    ll extra = x * m - n;
    ll pay = min(extra * a, n * b);
    cout << pay << endl;
    return 0;
}
