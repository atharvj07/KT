#include <iostream>
#include <string>
using namespace std;

int main() {
	int n;
	cin >> n;
	for (int i = 0; i < n; i++) {
		string s;
		cin >> s;
		int lala = s.find("lala.");
		int miao = s.find("miao");
		if (lala != string::npos && lala + 5 == s.length()) {
			cout << "Freda's" << endl;
		} else if (miao == 0) {
			cout << "Rainbow's" << endl;
		} else {
			cout << "OMG>.< I don't know!" << endl;
		}
	}
	return 0;
}
