#include <iostream>
#include <algorithm>
using namespace std;
const int mod = 1e9 + 7;
int main() {
    int n; cin >> n;
    string s, t; cin >> s >> t;
    int eq = 0;
    for (int i = 0; i < n; ++i) eq += s[i] == t[i];
    if (eq == n) {
        cout << 0 << endl;
    } else if (eq == n - 1) {
        cout << 1 << endl;
    } else {
        cout << 2 << endl;
    }
    return 0;
}
