import sys
n = int(sys.stdin.readline())
s = sys.stdin.readline().strip()
t = sys.stdin.readline().strip()
if s in ['ab', 'ba'] and t in ['ab', 'ba']:
    print('NO')
else:
    print('YES')
    if s in ['ab', 'ba']:
        if t == 'ac':
            print('c'*n+'b'*n+'a'*n)
        if t == 'bc':
            print('b'*n+'c'*n+'a'*n)
    else:
        if t in ['ab', 'ba']:
            print('a'*n+'b'*n+'c'*n)
        else:
            print('a'*n+'c'*n+'b'*n)
