def compress(s):
    t = []
    for c in s:
        if c in "[]":
            t.append(c)
            if len(t) >= 4 and t[-4:] == "[]:":
                t.pop();t.pop();t.pop()
    if t[-1] == '[' and len(t) > 2 and t[-3:] == "[]:":
        t.pop();t.pop()
    return -1 if len(t)%2 else len(t)//2

s = input()
print(compress(s))
