#include<cstdio>
#include<vector>
using namespace std;
vector< pair<int,int> >z[1001];
int main(){
  int n,k,a,b,lcs=0,mx=0,ans=0;
  scanf("%d%d",&n,&k);
  for(int i=1;i<=k;++i){
    for(int j=1;j<=n;++j){
      scanf("%d",&a);
      z[a].push_back(pair<int,int>(i,j));
    }
  }
  for(int i=1;i<=n;++i){
    mx=0;
    for(int j=1;j<=n;++j){
      lcs=0;
      for(int x=0;x<z[i].size();++x){
        for(int y=0;y<z[j].size();++y){
          if(z[i][x].first==z[j][y].first){
            if(z[i][x].second<z[j][y].second)
              ++lcs;
            else
              break;
          }
        }
      }
      if(lcs>mx)
        mx=lcs;
    }
    if(mx>ans)
      ans=mx;
  }
  printf("%d\n",ans);
  return 0;
}
