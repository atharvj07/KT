import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int I = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int l = Integer.MAX_VALUE;
        int r = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            l = Math.min(l, a[i]);
            r = Math.max(r, a[i]);
        }
        long size = (long) (Math.ceil(Math.log(r - l + 1) / Math.log(2)) * n);
        if (size > I * 8) {
            int z = 0;
            while (true) {
                int nl = Integer.MAX_VALUE;
                int nr = Integer.MIN_VALUE;
                for (int i = 0; i < n; i++) {
                    if (a[i] > z) {
                        nl = Math.min(nl, a[i]);
                    }
                    if (a[i] < z) {
                        nr = Math.max(nr, a[i]);
                    }
                }
                if (nl == Integer.MAX_VALUE) {
                    break;
                }
                size = (long) (Math.ceil(Math.log(nr - nl + 1) / Math.log(2)) * n);
                if (size > I * 8) {
                    z++;
                } else {
                    break;
                }
            }
            System.out.println(z);
        } else {
            System.out.println(0);
        }
    }
}
