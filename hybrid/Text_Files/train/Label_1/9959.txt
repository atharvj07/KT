#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> is_left(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> is_left[i];
  }
  for (int i = 1; i <= n / 2; i++) {
    int left = 0, right = 0;
    for (int j = 1; j <= n; j++) {
      if (is_left[j] == 'L' && !left) {
        left = j;
      } else if (is_left[j] == 'R' && !right) {
        right = j;
      }
    }
    cout << left << " " << right << endl;
    is_left[left] = is_left[right] = 0;
  }
  return 0;
}
