import java.util.Scanner;

public class Main {
    static int n, m;
    static int[][] a;
    static int[][] dp;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        a = new int[n][m];
        dp = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = in.nextInt();
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            dp[i][0] = a[i][0];
            ans = Math.max(ans, dp[i][0]);
        }
        for (int j = 1; j < m; j++) {
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    dp[i][j] = a[i][j];
                    for (int k = i - 1; k >= 0; k--) {
                        if (dp[k][j - 1] > dp[i][j]) {
                            dp[i][j] = dp[k][j - 1] + a[i][j];
                        }
                    }
                } else {
                    dp[i][j] = a[i][j];
                    for (int k = i + 1; k < n; k++) {
                        if (dp[k][j - 1] > dp[i][j]) {
                            dp[i][j] = dp[k][j - 1] + a[i][j];
                        }
                    }
                }
                ans = Math.max(ans, dp[i][j]);
            }
        }
        System.out.println(ans);
    }
}
