from collections import defaultdict

def dfs(v, p, g, dp, a, k):
    children = []
    for u in g[v]:
        if u == p: continue
        dfs(u, v, g, dp, a, k)
        children.append(u)
    
    dp[v][0] = sum(dp[u][k] for u in children)
    dp[v][1] = a[v-1] + sum(dp[u][max(0, k-1)] for u in children)
    for i in range(2, k+2):
        for u in children:
            dp[v][i] = max(dp[v][i], dp[v][i-1], dp[u][i-2] + a[v-1] + sum(dp[uc][max(0, k-1)] for uc in children if uc != u))

n, k = map(int, input().split())
a = list(map(int, input().split()))
edges = [list(map(int, input().split())) for _ in range(n-1)]

g = defaultdict(list)
for u, v in edges:
    g[u].append(v)
    g[v].append(u)

dp = [[0 for _ in range(k+2)] for _ in range(n+1)]
dfs(1, -1, g, dp, a, k)
print(max(dp[1]))
