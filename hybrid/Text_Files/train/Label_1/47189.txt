import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Node {
        int id;
        List<Node> adj = new ArrayList<>();

        Node(int id) {
            this.id = id;
        }
    }

    static int n, m, k;
    static List<Node> graph = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        for (int i = 0; i < n; i++) {
            graph.add(new Node(i + 1));
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            graph.get(u).adj.add(graph.get(v));
            graph.get(v).adj.add(graph.get(u));
        }
        Node start = null;
        for (Node node : graph) {
            if (node.adj.size() >= k) {
                start = node;
                break;
            }
        }
        List<Node> path = new ArrayList<>();
        path.add(start);
        dfs(start, start, path);
    }

    static void dfs(Node u, Node start, List<Node> path) {
        if (path.size() > k) {
            System.out.println(path.size());
            for (Node node : path) {
                System.out.print(node.id + " ");
            }
            System.out.println();
            System.exit(0);
        }
        for (Node v : u.adj) {
            if (!path.contains(v)) {
                path.add(v);
                dfs(v, start, path);
                path.remove(v);
            }
        }
    }
}
