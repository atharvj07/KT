import sys
n, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
mx = max(a)
mn = 1
j = 1
even = 0
odd = 0
ans = []
for val in a:
    if val % 2 == 1:
        odd += 1
    else:
        even += 1
if abs(even - odd) <= 2:
    for val in a:
        if odd == even and (mn <= mx):
            ans.append(mn)
            mn += 1
        elif odd < even and (mx >= j):
            ans.append(mx)
            mx -= 1
            odd += 1
        elif even < odd and (mn <= j):
            ans.append(mn)
            mn += 1
            even += 1
        elif odd == even and val % 2 == 0:
            ans.append(val)
        else:
            ans.append(val)
    print(0)
    print(*ans)
else:
    print(-1)
