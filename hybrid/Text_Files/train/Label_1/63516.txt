from math import gcd

def can_move(n, m, q, queries):
    lcm = n * m // gcd(n, m)
    for i in range(q):
        s_x, s_y, e_x, e_y = queries[i]
        if s_x == 2:
            s_y *= n // gcd(n, m)
        else:
            s_y *= m // gcd(n, m)
        if e_x == 2:
            e_y *= n // gcd(n, m)
        else:
            e_y *= m // gcd(n, m)
        
        if (s_y - 1) // (lcm // m) == (e_y - 1) // (lcm // m) or (s_y - 1) // (lcm // n) == (e_y - 1) // (lcm // n):
            print("YES")
        else:
            print("NO")

n, m, q = map(int, input().split())
queries = [list(map(int, input().split())) for _ in range(q)]
can_move(n, m, q, queries)
