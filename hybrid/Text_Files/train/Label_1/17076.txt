#include <bits/stdc++.h>
using namespace std;
const int mod = 1000000007;
int fpow(int a, int p) {
    if (p == 0) return 1;
    if (p == 1) return a;
    int t = fpow(a, p / 2);
    t = (1LL * t * t) % mod;
    if (p & 1) t = (1LL * t * a) % mod;
    return t;
}
int32_t main() {
    int n;
    cin >> n;
    int ans = 0, inv2 = fpow(2, mod - 2);
    for (int i = n - 1; i >= 0; --i) {
        for (int j = i - 2; j >= 0; --j) {
            int x = (1LL * i * (i + 1)) % mod;
            x = (1LL * x * inv2) % mod;
            x = (1LL * x * j) % mod;
            ans = (ans + x) % mod;
        }
    }
    cout << ans << endl;
    return 0;
}
