#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef pair<int,int> pi;

int n, q, a[1505], ans[200005];
vector<pi> vec;

int get_max_len(int color){
	int cnt = 0, mx = 0;
	for (int i = 0; i < n; i++){
		if (a[i] == color) cnt++;
		else{
			mx = max(mx,cnt);
			cnt = 0;
		}
	}
	mx = max(mx,cnt);
	return mx;
}

int main() {
	cin >> n;
	string s; cin >> s;
	for (int i = 0; i < n; i++) a[i] = s[i]-'a';
	cin >> q;
	for (int i = 0; i < q; i++){
		int num; char c;
		cin >> num >> c;
		vec.push_back({num,c-'a'});
	}
	for (int i = 0; i < q; i++){
		int num = vec[i].first, color = vec[i].second;
		for (int j = 0; j < n; j++){
			if (a[j] == color && num){
				a[j] = -1;
				num--;
			}
		}
		for (int j = 0; j < n; j++)
			if (a[j] != -1) a[j] = get_max_len(a[j]);
		ans[i] = *max_element(a,a+n);
	}
	for (int i = 0; i < q; i++) cout << ans[i] << endl;
}
