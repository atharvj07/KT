q = int(input())
for _ in range(q):
    n, k = map(int, input().split())
    s = input()
    patterns = ["RGB", "GBR", "BRG"]
    min_changes = float('inf')
    for pattern in patterns:
        changes = [0] * (n + 1)
        for i in range(1, n + 1):
            changes[i] = changes[i - 1] + (s[i - 1] != pattern[(i - 1) % 3])
        for i in range(k, n + 1):
            min_changes = min(min_changes, changes[i] - changes[i - k])
    print(min_changes)
