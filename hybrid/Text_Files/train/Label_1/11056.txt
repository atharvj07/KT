#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    string s;
    int k;
    cin >> s >> k;
    k %= 26;
    vector<bool> was(26);
    for (int i = 0; i < s.size(); i++) {
        int shift = s[i] - 'a';
        int newPos = (shift + k) % 26;
        if (s[i] >= 'a')
            s[i] = 'a' + newPos;
        else
            s[i] = 'A' + newPos;
    }
    cout << s << endl;
    return 0;
}
