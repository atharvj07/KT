#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define ld long double
#define endl '\n'
#define io                        \
    ios_base::sync_with_stdio(0); \
    cin.tie(0);                  \
    cout.tie(0)

const int MOD = 1e9 + 7;
const int MXN = 2e5 + 5;

int main() {
    io;

    int n;
    cin >> n;

    vector<ll> b(n);
    for (int i = 0; i < n; i++) cin >> b[i];

    sort(b.begin(), b.end());
    int flip = 1;
    ll x = b[0];
    for (int i = 1; i < n; i++) {
        if (b[i] == x) {
            flip = 0;
            continue;
        }
        x ^= b[i];
    }
    if (flip == 0) {
        cout << "No" << endl;
        return 0;
    }
    cout << "Yes" << endl;
    for (auto &x : b) cout << x << " ";
    cout << endl;

    return 0;
}
