import sys
input = sys.stdin.readline

n, h = list(map(int, input().split()))
a = [int(x) for x in input().split()]
dp = [[0 for i in range(h)] for j in range(n)]
dp[0][a[0]] = 1
for i in range(1, n):
    for j in range(h):
        if a[i] != j:
            dp[i][j] = dp[i - 1][j]
        else:
            dp[i][j] = (dp[i - 1][j] + dp[i - 1][j - 1]) % 1000000007
            
print(dp[n - 1][h - 1])
