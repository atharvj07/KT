#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;
const int L = 260;

int n, q;
string u;
int a[3][N], b[3][N], id[N], now;
char w[3][L];
bool f[3];

void build(int x) {
  int l = N;
  for(int i = id[x]; i <= n; i++) {
    f[x] |= u[i] != u[id[x] - 1];
    w[x][now] = u[i];
    b[x][now++] = i;
    if(a[x][i] != a[x][i - 1]) {
      l = min(l, i);
    }
  }
  id[x + 1] = l;
}

void del(int x) {
  now--;
  f[x] |= u[b[x][now]] != u[b[x][now] - 1];
}

void add(int x, char c) {
  u += c;
  ++n;
  f[x] |= u[n] != u[n - 1];
  w[x][now++] = c;
  b[x][now++] = n;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> q;
  cin >> u;
  for(int i = 0; i < 3; i++) {
    for(int j = n; j >= 0; j--) {
      a[i][j] = (j > 0 ? a[i][j - 1] : 0) + (j < n && u[j] != u[j - 1]);
    }
    now = 0;
    id[0] = 0;
    build(i);
  }
  while(q--) {
    int x;
    cin >> x;
    if(--x < 0) {
      del(0), del(1), del(2);
    } else {
      char c;
      cin >> c;
      add(x, c);
      build(x);
    }
    cout << (f[0] && f[1] && f[2] ? "YES" : "NO") << '\n';
  }
}
