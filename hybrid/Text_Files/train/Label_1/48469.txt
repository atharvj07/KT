import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            long k = sc.nextLong();
            if (a == b) {
                System.out.println("YES");
            } else if (k == 0) {
                System.out.println("NO");
            } else if (a > b) {
                if ((a - b) % k == 0 && (a - b) / k <= a && (a - b) / k <= b) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            } else {
                if ((b - a) % k == 0 && (b - a) / k <= a && (b - a) / k <= b) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            }
        }
    }
}
