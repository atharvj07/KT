import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        ArrayList<Integer> a = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            a.add(input.nextInt());
        }
        Collections.sort(a);
        int d = a.get(n / 2);
        for (int i = 0; i < n; i++) {
            if (a.get(i) > d + 1) {
                System.out.println(-1);
                return;
            }
            if (a.get(i) < d) {
                d = a.get(i);
            }
        }
        System.out.println(d);
    }
}
