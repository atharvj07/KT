import sys
input = sys.stdin.readline

n = int(input())
ps = [tuple(map(int, input().split())) for _ in range(n)]
l = [[0] * n for _ in range(n)]
for i in range(n):
    for j in range(i):
        x1, y1 = ps[i]
        x2, y2 = ps[j]
        for k in range(j):
            x3, y3 = ps[k]
            if (x1 - x2) * (y3 - y2) == (x3 - x2) * (y1 - y2):
                l[i][j] += 1
                l[j][i] += 1
ans = 0
for a in l:
    for b in a:
        ans += b
print(ans // 2)
