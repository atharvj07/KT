q = int(input())
for _ in range(q):
    n = int(input())
    top_row = input()
    bottom_row = input()
    y = 1
    for i in range(n):
        if y == 1:
            if top_row[i] in "3456":
                y = 2
            elif top_row[i] in "12":
                continue
            else:
                break
        else:
            if bottom_row[i] in "3456":
                y = 1
            elif bottom_row[i] in "12":
                continue
            else:
                break
    if y == 2 and bottom_row[-1] in "3456":
        print("YES")
    else:
        print("NO")
