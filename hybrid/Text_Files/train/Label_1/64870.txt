import sys

class Node:
    def __init__(self, cost, init, goal):
        self.cost = cost
        self.init = init
        self.goal = goal
        self.visited = False
        self.adj = []

INF = sys.maxsize

def minCost(node, par):
    node.visited = True
    if node.init == node.goal:
        cost = 0
    else:
        cost = node.cost
    
    for u in node.adj:
        if u != par:
            cost += minCost(u, node)
    
    return cost

def dfs(node, par, cost):
    node.visited = True
    if node.init != node.goal:
        cost -= node.cost
        for u in node.adj:
            if u != par:
                dfs(u, node, cost + node.cost)

n = int(input())

nodes = []

for i in range(n):
    a, b, c = map(int, input().split())
    nodes.append(Node(a, b, c))

for i in range(n - 1):
    u, v = map(int, input().split())
    nodes[u - 1].adj.append(nodes[v - 1])
    nodes[v - 1].adj.append(nodes[u - 1])

cost = minCost(nodes[0], -1)

if cost == INF:
    print(-1)
else:
    dfs(nodes[0], -1, cost)
    print(cost)
