MOD = 998244353

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, m, A, b):
    distinct_strings = power(A, n, MOD)
    for i in range(m):
        k = b[i]
        fixed_positions = 2 * k
        free_positions = n - fixed_positions
        distinct_strings -= power(A, free_positions, MOD)
        distinct_strings %= MOD
    return distinct_strings

n, m, A = map(int, input().split())
b = list(map(int, input().split()))
print(solve(n, m, A, b))
