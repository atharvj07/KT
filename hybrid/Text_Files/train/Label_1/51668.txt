import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }
        int[] color = new int[n+1];
        int colorCnt = 1;
        for (int i = 0; i < m; i++) {
            if (color[a[i]] == 0 && color[b[i]] == 0) {
                color[a[i]] = colorCnt;
                color[b[i]] = colorCnt;
                colorCnt++;
            } else if (color[a[i]] == 0) {
                color[a[i]] = color[b[i]];
            } else if (color[b[i]] == 0) {
                color[b[i]] = color[a[i]];
            } else if (color[a[i]] != color[b[i]]) {
                int tmp = color[b[i]];
                for (int j = 1; j <= n; j++) {
                    if (color[j] == tmp) {
                        color[j] = color[a[i]];
                    }
                }
            }
            long ans = 0;
            for (int j = 1; j <= n; j++) {
                for (int k = j + 1; k <= n; k++) {
                    if (color[j] != color[k]) {
                        ans++;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
