#include<bits/stdc++.h>
using namespace std;
const int mod=998244353;
int fexp(int a,int n){int ret=1;while(n){if(n&1)ret=1ll*ret*a%mod;a=1ll*a*a%mod;n>>=1;}return ret;}
int32_t main()
{
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	int t;cin>>t;
	while(t--)
	{
	    const int MX=2e5+5;
	    int n,ans=0;cin>>n;
        vector<long long>v(MX);
        
	    for(int i=0;i<n;i++)
	    {
	        int x;cin>>x;
	        v[x]++;
	    }
	    sort(v.rbegin(),v.rend());
	    int sum=0,op=0;
	    for(int x:v)
	    {
	        if(sum==0&&x==0)break;
	        sum+=x;
	        if(sum>=n)op++;
	        else sum=0;
	    }
	    cout<<(fexp(2,op)-1+mod)%mod<<'\n';
	}
}
