import java.util.*;
public class Main {
    static int[] d, par, col;
    static boolean[] vis;
    static List<Integer>[] g;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        g = new List[n + 1];
        for (int i = 0; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        d = new int[n + 1];
        par = new int[n + 1];
        col = new int[n + 1];
        vis = new boolean[n + 1];
        if (!dfs(1, 1, 1)) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 2; i <= n; i++) {
            int p = par[i];
            System.out.print((col[p] ^ col[i]) ^ 0);
        }
        System.out.println();
    }
    static boolean dfs(int u, int p, int c) {
        vis[u] = true;
        col[u] = c;
        for (int v : g[u]) {
            if (!vis[v]) {
                par[v] = u;
                if (!dfs(v, u, c ^ 1)) return false;
            } else if (v != p && col[v] == col[u]) return false;
        }
        return true;
    }
}
