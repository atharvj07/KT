#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5;
int a[MAXN + 10], u[MAXN + 10];

int main() {
  int n, x;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> x;
    if (!u[x]) u[x] = 1;
  }
  int ans = -1;
  for (int i = 1; i <= n; ++i) {
    if (u[i]) ++u[i];
    if (u[i] == 1) {
      ans = i;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
