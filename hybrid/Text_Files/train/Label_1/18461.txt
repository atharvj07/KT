#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> v(k);
    for (int i = 0; i < k; i++) { cin >> v[i]; }
    int ans = n;
    for (int i = 0; i < k; i++) {
      int j = i + 1, cur = v[i];
      while (j < k && v[j] <= cur + j - i) j++;
      j--;
      ans = min(ans, max(max(v[i] - i, i + 1), max(v[j] - j, j + 1)));
      i = j;
    }
    cout << ans << '\n';
  }
  return 0;
}
