#include <iostream>

using namespace std;

const int big = 1e6, mod = 1e9 + 7;

int main() {
  long long n, ans = 1;
  while (cin >> n) {
    if (n <= 1) ans = 1;
    if (n <= 2) ans = 2;
    if (n <= 3) ans = 6;
    if (n >= 4) ans = (n + 1) * (n + 1);
    cout << ans % mod << endl;
  }
  return 0;
}
