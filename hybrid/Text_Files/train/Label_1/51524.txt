import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            char[] s = sc.next().toCharArray();
            int[] ans = new int[n];
            for (int i = 0; i < n; i++) {
                int[] vis = new int[n];
                Arrays.fill(vis, -1);
                int cur = i;
                while (vis[cur] != i) {
                    vis[cur] = i;
                    if (s[cur] == '>' || s[(cur - 1 + n) % n] == '-') {
                        cur = (cur + 1) % n;
                    } else {
                        cur = (cur - 1 + n) % n;
                    }
                }
                ans[cur]++;
            }
            int sum = 0;
            for (int i : ans) {
                if (i > 0) {
                    sum++;
                }
            }
            System.out.println(sum);
        }
    }
}
