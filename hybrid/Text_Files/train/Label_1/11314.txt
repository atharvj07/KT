#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int main(){
    int n; cin >> n;
    vector<string> codes(n);

    for (int i = 0; i < n; i++){
        cin >> codes[i];
    }

    int k = 0;

    while(1){
        for (int i = 0; i < n; i++){
            for (int j = 0; j < n; j++){
                if (i == j) continue;

                int diff = 0;
                for (int l = 0; l < 6; l++){
                    if (codes[i][l] != codes[j][l]) diff++;
                }
                if (diff <= k) goto outer;
            }
        }

        cout << k << endl;
        return 0;

        outer:;
        k++;
    }

    return 0;
}
