import sys
n,m=map(int, input().split())
l=list(map(int, input().split()))
vis=[True]*n
avail=[]
for i in range(n):
    if vis[i]:
        mn=i
        for j in range(i+1, n):
            if j+l[j]-1>i+l[i]-1:break
            if not vis[j]:break
            if j+l[j]-1<i+l[i]-1:mn=j
        for j in range(i,mn+l[mn]):
            vis[j]=False
        avail.append([i,mn,l[mn]])
#print(avail)
if len(avail)==m:
    print(*[i[0] for i in avail])
else:print(-1)
