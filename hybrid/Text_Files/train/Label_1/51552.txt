import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        if (n == 1) {
            System.out.println(-1);
            return;
        }
        for (int i = 0; i < 1000000; i++) {
            if (i % 2 == 0 && i % 3 == 0 && i % 5 == 0 && i % 7 == 0) {
                String s = String.valueOf(i);
                if (s.length() == n) {
                    System.out.println(i);
                    return;
                }
            }
        }
        System.out.println(-1);
    }
}
