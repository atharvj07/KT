#include <iostream>
#include <vector>
#include <string>
using namespace std;
pair<char, char> st[100][100];
int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;
  for (int l = 0; l < q; l++) {
    int x, y;
    cin >> x >> y;
    --x;
    --y;
    vector<char> left(10, 0), right(10, 0);
    for (int i = x; i <= y; i++) {
      if (s[i] >= '0' && s[i] <= '9') {
        if (st[x][i].first == s[i]) {
          if (st[x][i].second == 'r')
            right[s[i] - '0']++;
          if (st[x][i].second == 'l')
            left[s[i] - '0']++;
        } else {
          st[x][i].first = s[i];
          st[x][i].second = 'r';
          right[s[i] - '0']++;
        }
      }
      if (s[i] == '>') {
        for (int j = x; j < i; j++) {
          if (st[x][j].first == '<') {
            st[x][j].first = '>';
            st[x][j].second = 'r';
            right[s[j] - '0']++;
          }
        }
        st[x][i].first = '>';
        st[x][i].second = 'r';
      }
      if (s[i] == '<') {
        for (int j = i + 1; j <= y; j++) {
          if (st[i + 1][j].first == '>') {
            st[i + 1][j].first = '<';
            st[i + 1][j].second = 'l';
            left[s[j] - '0']++;
          }
        }
        st[i + 1][i].first = '<';
        st[i + 1][i].second = 'l';
      }
    }
    for (int i = 9; i >= 0; i--) {
      cout << left[i] + right[i] << " ";
    }
    cout << '\n';
  }
}
