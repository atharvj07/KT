#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n; cin >> n;
    long long a[n + 5], ans1 = -1, ans2 = -1;
    for (long long i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (long long i = 1; i <= n; i++) {
        for (long long j = i + 1; j <= n; j++) {
            if (i == 1) {
                ans1 = i, ans2 = j;
            }
            if ((a[i] < a[i - 1] && a[j] > a[j + 1]) || (a[i] > a[i - 1] && a[j] < a[j + 1])) {
                ans1 = i, ans2 = j;
            }
        }
    }
    cout << ans1 << ' ' << ans2 << endl;
    return 0;
}
