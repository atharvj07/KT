t=int(input())
for i in range(t):
    x=int(input())
    if x<=9: ans=x
    elif x==10: ans=-1
    else:
        di=[0 for _ in range(12)]
        di[1]=0
        di[2]=1
        rem=x-1
        ans=''
        for j in range(10,0,-1):
            if rem>=j:
                rem-=j
                ans+=str(j)
                di[j]=1
        if rem>0: ans=-1
        else:
            ans1=''
            for j in range(10,0,-1):
                if di[j]==0: ans1+=str(j)
            ans+=ans1
    print(ans)
