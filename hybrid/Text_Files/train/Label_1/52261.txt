import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        long n = Long.parseLong(in.readLine());
        int ans = 0;
        while (n % 2 == 0) {
            n /= 2;
            ans++;
        }
        while (n % 3 == 0) {
            n /= 3;
            ans++;
        }
        if (n != 1) ans = -1;
        System.out.println(n + " " + ans);
    }
}
