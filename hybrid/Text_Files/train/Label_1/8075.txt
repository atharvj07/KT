#include <bits/stdc++.h>
using namespace std;
struct in {int v,p;} a[1005];
int n,ans,cnt=3,ai[3];
char s[1005];
int main(){
    cin >> n;
    for(int i=1;i<=n;i++){
        cin >> a[i].p >> s;
        a[i].v=0;
        for(int j=0;s[j];j++){
            if(s[j]=='A') a[i].v|=4;
            if(s[j]=='B') a[i].v|=2;
            if(s[j]=='C') a[i].v|=1;
        }
    }
    sort(a+1,a+n+1,[&](in x,in y){return x.p<y.p;});
    for(int i=1;i<=n;i++){
        ans+=a[i].p;
        for(int j=0;j<3;j++){
            if(a[i].v&(1<<j)) ai[j]++;
            if(ai[j]>=1) cnt--;
            if(cnt==0) break;
        }
        if(cnt==0) break;
    }
    if(cnt>0) cout << -1;
    else cout << ans;
    return 0;
}
