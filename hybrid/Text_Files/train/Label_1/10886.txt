#include <bits/stdc++.h>
using namespace std;
long long n, s, a[100005], b[100005], dp[100005];
int main() {
  cin >> n >> s;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    b[i] = a[i];
  }
  sort(b + 1, b + n + 1);
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1] + b[i] + i * 1ll * i;
  }
  long long ans = 0, sum = 0;
  for (int i = 1; i <= n; i++) {
    if (sum + a[i] + i * 1ll * ans <= s) {
      ans++;
      sum += a[i] + i * 1ll * ans;
    } else
      break;
  }
  cout << ans << " " << dp[ans] << endl;
  return 0;
}
