import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            char[][] grid = new char[n][n];
            boolean[][] visited = new boolean[n][n];
            for (int j = 0; j < n; j++) {
                String line = scanner.next();
                for (int k = 0; k < n; k++) {
                    grid[j][k] = line.charAt(k);
                    visited[j][k] = false;
                }
            }
            int c = 0;
            boolean found = false;
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (!found && grid[j][k] == '0') {
                        if (dfs(grid, visited, j, k, n)) {
                            c++;
                            System.out.println(c);
                            System.out.println((j + 1) + " " + (k + 1));
                            found = true;
                            break;
                        }
                    } else if (!found && grid[j][k] == '1') {
                        if (dfs(grid, visited, j, k, n)) {
                            c++;
                            System.out.println(c);
                            System.out.println((j + 1) + " " + (k + 1));
                            found = true;
                            break;
                        }
                    }
                }
            }
            if (!found) {
                c = 2;
                System.out.println(c);
                for (int j = 1; j < n - 1; j++) {
                    for (int k = 1; k < n - 1; k++) {
                        if (grid[j][k] == '0') {
                            System.out.println((j + 1) + " " + (k + 1));
                            break;
                        }
                    }
                }
                for (int j = n - 2; j >= 1; j--) {
                    for (int k = n - 2; k >= 1; k--) {
                        if (grid[j][k] == '1') {
                            System.out.println((j + 1) + " " + (k + 1));
                            break;
                        }
                    }
                }
            }
        }
    }

    public static boolean dfs(char[][] grid, boolean[][] visited, int x, int y, int n) {
        if (x < 0 || x >= n || y < 0 || y >= n) {
            return false;
        }
        if (visited[x][y]) {
            return false;
        }
        if (grid[x][y] == 'F') {
            return true;
        }
        visited[x][y] = true;
        if (grid[x][y] == '0') {
            return dfs(grid, visited, x + 1, y, n) || dfs(grid, visited, x - 1, y, n) || dfs(grid, visited, x, y + 1, n) || dfs(grid, visited, x, y - 1, n);
        } else {
            return dfs(grid, visited, x + 1, y, n) || dfs(grid, visited, x - 1, y, n) || dfs(grid, visited, x, y + 1, n) || dfs(grid, visited, x, y - 1, n);
        }
    }
}
