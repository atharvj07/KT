#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, b, d, a, cnt = 0, cur = 0;
  cin >> n >> b >> d;
  for (int i = 0; i < n; i++) {
    cin >> a;
    if (a > b) continue;
    cur += a;
    if (cur > d) {
      cnt++;
      cur = a;
    }
  }
  cout << cnt << "\n";
  return 0;
}
