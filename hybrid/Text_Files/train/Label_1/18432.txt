#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int MAXN = 2e5 + 5;

int n, m, ta, tb, k;
ll a[MAXN], b[MAXN];

ll bs(ll li, ll ri) {
    if(li + 1 == ri) return ri;
    ll mi = (li + ri) / 2;
    deque<ll> dq;
    ll mx = -1e18;
    for(int i = 1; i <= m; i++) {
        while(!dq.empty() && dq.front() < b[i] - tb) dq.pop_front();
        if(!dq.empty()) mx = max(mx, dq.front() + ta);
        if(dq.size() <= k) dq.push_back(b[i]);
    }
    if(mx < li) return li;
    if(mx >= ri) return ri;
    if(mx >= mi) return bs(mi, ri);
    return bs(li, mi);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> ta >> tb >> k;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= m; i++) cin >> b[i];
    cout << bs(a[n] + ta, b[m] + 2 * tb) << "\n";
}
