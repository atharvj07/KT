import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class RainbowCycles {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int n = scanner.nextInt();
        int[] a = new int[m];
        int[] b = new int[n];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        List<List<Integer>> sets = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int s = scanner.nextInt();
            List<Integer> set = new ArrayList<>();
            for (int j = 0; j < s; j++) {
                set.add(scanner.nextInt());
            }
            sets.add(set);
        }

        long[][] dp = new long[1 << m][n + 1];
        for (int i = 0; i < (1 << m); i++) {
            Arrays.fill(dp[i], Long.MAX_VALUE / 2);
        }
        dp[0][0] = 0;

        for (int mask = 0; mask < (1 << m); mask++) {
            for (int last = 0; last <= n; last++) {
                if (dp[mask][last] == Long.MAX_VALUE / 2) {
                    continue;
                }
                for (int i = 0; i < m; i++) {
                    if ((mask & (1 << i)) == 0) {
                        for (int j = 0; j < n; j++) {
                            if (sets.get(i).contains(j + 1)) {
                                dp[mask | (1 << i)][j + 1] = Math.min(dp[mask | (1 << i)][j + 1], dp[mask][last] + a[i] + b[j]);
                            }
                        }
                    }
                }
                if (last > 0) {
                    for (int i = 0; i < m; i++) {
                        if ((mask & (1 << i)) != 0) {
                            dp[mask][last] = Math.min(dp[mask][last], dp[mask ^ (1 << i)][last - 1]);
                        }
                    }
                }
            }
        }

        long answer = Long.MAX_VALUE / 2;
        for (int i = 0; i <= n; i++) {
            answer = Math.min(answer, dp[(1 << m) - 1][i]);
        }

        System.out.println(answer);
    }
}
