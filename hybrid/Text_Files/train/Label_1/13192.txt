typedef long long ll;
typedef pair<ll,ll> pll;
inline int read(){
	int x=0;char c=getchar();bool f=1;while(!isdigit(c)){if(c=='-'){f=0;}c=getchar();}while(isdigit(c)){x=(x<<3)+(x<<1)+c-'0';c=getchar();}return f?x:-x;
}
#define max(x,y) ((x)>(y)?(x):(y))
#define min(x,y) ((x)<(y)?(x):(y))
ll gcd(ll a,ll b){return b?gcd(b,a%b):a;}
const double eps=1e-9;
const int N=100000+5;
int n;
pair<long long,long long>p[N];
int main()
{
	n=read();
	long long tmpx=-2e9,tmpy=-2e9;
	for(int i=1;i<=n;i++)
	{
		p[i].fi=read();p[i].se=read();
		tmpx=max(tmpx,p[i].fi);tmpy=max(tmpy,p[i].se);
	}
	long long ansx,ansy;
	if(tmpx+tmpy<=0){puts("-1");return 0;}
	ansx=(tmpx+tmpy)/2;ansy=(tmpy-tmpx)/2;
	double ans=sqrt(tmpx*tmpx+tmpy*tmpy)*0.5;
	for(int i=1;i<=n;i++)
	{
		double tmp=sqrt((p[i].fi-ansx)*(p[i].fi-ansx)+(p[i].se-ansy)*(p[i].se-ansy));
		if(fabs(tmp-ans)>eps)
		{
			puts("-1");return 0;
		}
	}
	printf("%.9lf\n",ans);
	return 0;
}
