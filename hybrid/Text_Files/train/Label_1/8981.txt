#include <bits/stdc++.h>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<int> a(n); for (auto& x : a) cin >> x;

    int c[3] = {0, 0, 0};
    for (auto x : a) c[x % 3]++;

    int ans = 0;
    int need = n / 3;
    for (int i = 0; i < 3; i++) {
      if (c[i] < need) ans += need - c[i];
    }

    cout << ans << '\n';
  }
}
