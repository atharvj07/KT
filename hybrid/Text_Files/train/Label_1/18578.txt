#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        string s;
        cin >> s;
        while (m--) {
            string t = s;
            for (int i = 1; i < n - 1; i++) {
                if (s[i] == '0' && s[i - 1] == s[i + 1]) t[i] = '1';
            }
            s = t;
        }
        cout << s << endl;
    }
    return 0;
}
