import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.next();
    String t = in.next();
    int n = s.length();
    int m = t.length();
    int[] pfx = new int[26];
    for (int i = 0; i < m; i++) {
      pfx[t.charAt(i) - 'a']++;
    }
    int ans = 0;
    for (char ch0 = 'a'; ch0 <= 'z'; ch0++) {
      int c0 = 0;
      for (int i = 0; i < n; i++) {
        if (s.charAt(i) == '0') {
          c0++;
        } else {
          c0--;
        }
        if (c0 == 0 && pfx[ch0 - 'a'] > 0) {
          int c1 = 0;
          for (int j = i + 1; j < n; j++) {
            if (s.charAt(j) == '1') {
              c1++;
            } else {
              c1--;
            }
            if (c1 == 0 && pfx[t.charAt(j) - 'a'] > 0) {
              ans++;
            }
          }
        }
      }
    }
    System.out.println(ans);
  }
}
