import java.util.*;

public class Main {
    static int[][] mat;
    static int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        mat = new int[n][n];
        for(int i = 0; i < n; i++) {
            String s = sc.next();
            for(int j = 0; j < n; j++) {
                mat[i][j] = s.charAt(j) - '0';
            }
        }

        System.out.println(solve());
    }

    static int solve() {
        int ans = 0;
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                if(mat[i][j] == 1) {
                    ans++;
                    toggle(i, j);
                }
            }
        }
        return ans;
    }

    static void toggle(int x, int y) {
        for(int i = 0; i < n; i++) {
            mat[i][y] = 1 - mat[i][y];
        }
        for(int j = 0; j < n; j++) {
            mat[x][j] = 1 - mat[x][j];
        }
    }
}
