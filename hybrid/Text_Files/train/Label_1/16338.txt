#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, s, t;
  cin >> n >> s >> t;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  if (s == t) {
    cout << 0 << endl;
    return 0;
  }
  int cur = s;
  int ans = 0;
  set<int> se;
  se.insert(s);
  while (true) {
    int nxt = p[cur - 1];
    if (nxt == t) {
      cout << ans << endl;
      return 0;
    }
    if(se.find(nxt)!=se.end()) {
      cout<<-1<<endl;
      return 0;
    }
    se.insert(nxt);
    ans++;
    cur = nxt;
  }
  return 0;
}
