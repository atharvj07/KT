import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int cnt = 0;
            boolean onGround = true;
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                if (s.equals("lu") || s.equals("ru")) {
                    if (onGround) {
                        onGround = false;
                        cnt++;
                    }
                } else if (s.equals("ld") || s.equals("rd")) {
                    if (!onGround) {
                        onGround = true;
                        cnt++;
                    }
                }
            }
            System.out.println(cnt);
        }
    }
}
