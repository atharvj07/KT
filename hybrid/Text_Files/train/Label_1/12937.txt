#include <bits/stdc++.h>
using namespace std;

const int N = 100005;

int n,k,m;
int a[N],b[N],cnt[N];
multiset<int>s;
int main(){
	cin>>n>>k>>m;
	for(int i=1;i<=n;i++){
		cin>>a[i];
		s.insert(a[i]);
	}
	for(multiset<int>::iterator it =s.begin();it !=s.end();it++){
		b[++cnt[0]]=*it;
		cnt[0]++;
	}
	if(k>cnt[0]){
		cout<<"No"<<endl;
		return 0;
	}
	sort(b+1,b+cnt[0]+1);
	cout<<"Yes"<<endl;
	for(int i=1;i<=k;i++){
		cout<<b[i]<<" ";
	}
	cout<<endl;
	return 0;
}
