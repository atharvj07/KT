import java.util.Scanner;

public class Main {
    static int[][] getScore(int[][] a, int[][] b, int[][] c) {
        int h = c.length;
        int w = c[0].length;
        int h1 = a.length;
        int w1 = a[0].length;
        int max = 0;
        int x = 0, y = 0;
        boolean found = false;
        for (int i = 0; i <= h1 - h; i++) {
            for (int j = 0; j <= w1 - w; j++) {
                int score = 0;
                boolean same = true;
                for (int k = 0; k < h; k++) {
                    for (int l = 0; l < w; l++) {
                        if (b[i + k][j + l] != c[k][l]) {
                            same = false;
                            break;
                        }
                        if (same)
                            score += a[i + k][j + l];
                    }
                }
                if (same && score > max) {
                    max = score;
                    x = i;
                    y = j;
                    found = true;
                }
            }
        }
        if (found)
            return new int[][]{{max, x, y}};
        else
            return new int[][]{{-1}};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt();
        int w1 = sc.nextInt();
        int[][] a = new int[h1][w1];
        for (int i = 0; i < h1; i++) {
            for (int j = 0; j < w1; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int[][] b = new int[h1][w1];
        for (int i = 0; i < h1; i++) {
            for (int j = 0; j < w1; j++) {
                b[i][j] = sc.nextInt();
            }
        }
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[][] c = new int[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                c[i][j] = sc.nextInt();
            }
        }
        int[][] result = getScore(a, b, c);
        if (result[0][0] == -1)
            System.out.println("NA");
        else
            System.out.println(result[0][0]);
    }
}
