#include <bits/stdc++.h>
using namespace std;

typedef long long int ll;

int gcd(ll a, ll b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

ll solve(ll n) {
    while (true) {
        ll sum = 0, num = n;
        while (num > 0) {
            sum += num % 10;
            num /= 10;
        }
        if (gcd(n, sum) > 1) return n;
        n++;
    }
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        cout << solve(n) << '\n';
    }
    return 0;
}
