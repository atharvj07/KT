from math import ceil
for _ in range(int(input())):
  n = int(input())
  a, b, c = map(int, input().split())
  s = input()
  x = ceil(n / 2)
  if a > 0 and (s.count('S') == 0 or s.count('R') < a):
    print('NO')
  elif b > 0 and (s.count('R') == 0 or s.count('P') < b):
    print('NO')
  elif c > 0 and (s.count('P') == 0 or s.count('S') < c):
    print('NO')
  else:
    print('YES')
    t = ''
    for i in range(n):
      if a > 0 and s[i] == 'S':
        t += 'R'
        a -= 1
      elif b > 0 and s[i] == 'R':
        t += 'P'
        b -= 1
      elif c > 0 and s[i] == 'P':
        t += 'S'
        c -= 1
      else:
        t += s[i]
    print(t)
