#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    sort(a.begin(), a.end());
    long long ans = 0;
    int cnt = 1;
    int i = 0;
    while (i < n) {
        int j = i;
        while (j < n && a[j] == a[i]) j++;
        ans += 1LL * cnt * cnt;
        cnt *= 2;
        i = j;
    }
    cout << ans << endl;
}
