#include<bits/stdc++.h>
using namespace std;
int n,m,a[51][51],b[51][51];
bool ok;
void dfs(int x,int y){
    if(x<=0||x>n||y<=0||y>m){
        return ;
    }
    if(b[x][y]==1){
        ok=false;
        return ;
    }
    if(b[x][y]==-1){
        return ;
    }
    b[x][y]=1;
    dfs(x-1,y);
    dfs(x+1,y);
    dfs(x,y-1);
    dfs(x,y+1);
}
int main(){
    int t;
    cin>>t;
    while(t--){
        ok=true;
        cin>>n>>m;
        for(int i=1;i<=n;i++){
            for(int j=1;j<=m;j++){
                char c;
                cin>>c;
                if(c=='R'){
                    a[i][j]=1;
                    b[i][j]=-1;
                }else if(c=='W'){
                    a[i][j]=2;
                    b[i][j]=-1;
                }else{
                    a[i][j]=0;
                    b[i][j]=0;
                }
            }
        }
        for(int i=1;i<=n;i++){
            for(int j=1;j<=m;j++){
                if(a[i][j]==0){
                    dfs(i,j);
                }
            }
        }
        if(!ok){
            cout<<"NO\n";
        }else{
            cout<<"YES\n";
            for(int i=1;i<=n;i++){
                for(int j=1;j<=m;j++){
                    if(b[i][j]==1){
                        cout<<"R";
                    }else if(b[i][j]==2){
                        cout<<"W";
                    }else{
                        cout<<".";
                    }
                }
                cout<<endl;
            }
        }
    }
    return 0;
}
