#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<int> cnt(3, 0);
  for (char c : s) {
    cnt[c - '0']++;
  }

  int need = (n + 2) / 3 - cnt[0];
  for (int i = 0; need > 0 && i < n; i++) {
    if (s[i] == '0' && cnt[1] < cnt[2]) {
      s[i] = '1';
      cnt[1]++;
      cnt[0]--;
      need--;
    } else if (s[i] == '0' && cnt[1] >= cnt[2]) {
      s[i] = '2';
      cnt[2]++;
      cnt[0]--;
      need--;
    }
  }

  for (int i = 0; need > 0 && i < n; i++) {
    if (s[i] == '2' && cnt[1] > cnt[0]) {
      s[i] = '1';
      cnt[1]++;
      cnt[2]--;
      need--;
    } else if (s[i] == '2' && cnt[1] <= cnt[0]) {
      s[i] = '0';
      cnt[0]++;
      cnt[2]--;
      need--;
    }
  }

  for (int i = 0; need > 0 && i < n; i++) {
    if (s[i] == '1' && cnt[2] > cnt[0]) {
      s[i] = '0';
      cnt[0]++;
      cnt[1]--;
      need--;
    } else if (s[i] == '1' && cnt[2] <= cnt[0]) {
      s[i] = '2';
      cnt[2]++;
      cnt[1]--;
      need--;
    }
  }

  cout << s << endl;

  return 0;
}
