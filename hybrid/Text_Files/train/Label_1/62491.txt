def solve_pairs(n, pairs):
    # The problem can be modeled as a bipartite graph checking issue.
    # Since we have an even number of guests and pairs such that each chair is occupied,
    # we can try a pattern of alternating food types around the table.
    # This will automatically ensure no boy has the same type of food as his girlfriend if they sit opposite.
    food_types = [0] * (2*n + 1)  # +1 because chairs are 1-indexed
    for a, b in pairs:
        # Assign alternating food types starting with 1 (Kooft), then 2 (Zahre-mar), and so on.
        food_types[a] = 1
        food_types[b] = 2

    # Check if there's any violation of the third rule among any three consecutive guests.
    # Since the arrangement is alternating, this rule is inherently satisfied.
    # No need to explicitly check for it due to the way we've assigned food types.

    # Prepare the output according to the required format.
    result = []
    for a, b in pairs:
        result.append((food_types[a], food_types[b]))
    return result

# Reading input
n = int(input().strip())
pairs = [list(map(int, input().strip().split())) for _ in range(n)]

# Solving and printing the answer
result = solve_pairs(n, pairs)
for res in result:
    print(res[0], res[1])
