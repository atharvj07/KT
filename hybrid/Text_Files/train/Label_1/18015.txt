#include<bits/stdc++.h>
using namespace std;
#define int long long
int a[1005],l[1005],r[1005];
int dp[1005][2];
int n,m;
int inf=1e18;
int32_t main(){
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    cin>>m;
    for(int i=0;i<m;i++){
        cin>>l[i]>>r[i];
    }
    for(int i=0;i<=n;i++){
        dp[i][0]=dp[i][1]=inf;
    }
    dp[0][0]=0;
    for(int i=0;i<m;i++){
        for(int j=0;j<=n;j++){
            dp[j][1]=min({dp[j][1],dp[j][0],dp[j+1][0]});
            if(j+1<=n)dp[j][1]=min({dp[j][1],dp[j+1][1]});
            if(l[i]<=dp[j][0] && dp[j][0]<r[i])dp[j][0]=dp[j][1];
            if(l[i]<=dp[j][1] && dp[j][1]<r[i])dp[j][1]=dp[j][1];
            if(j+1<=n && l[i]<=dp[j+1][0] && dp[j+1][0]<r[i])dp[j+1][0]=dp[j+1][1];
        }
    }
    int ans=inf;
    for(int i=0;i<=n;i++){
        ans=min({ans,dp[i][0],dp[i][1]});
    }
    for(int i=0;i<n;i++){
        ans+=a[i];
    }
    if(ans>=inf)cout<<-1<<endl;
    else cout<<ans<<endl;
}
