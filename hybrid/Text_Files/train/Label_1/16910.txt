#include <algorithm>
#include <cstdio>
#include <utility>
using namespace std;

const int MAXN = 110;
const int MAXM = 1e6 + 2;
const int INF = 2e9;

int n, u, v, a[MAXN], dp[MAXN][MAXM];
pair<int, int> ans[MAXN][MAXM];

int main() {
  int t;
  scanf("%d", &t);
  for (int cas = 1; cas <= t; ++cas) {
    scanf("%d %d %d", &n, &u, &v);
    for (int i = 1; i <= n; ++i) { scanf("%d", &a[i]); }

    for (int i = n; i >= 1; --i) {
      for (int j = 1; j < MAXM; ++j) {
        dp[i][j] = INF;
        int &res = dp[i][j];
        int &ansx = ans[i][j].first;
        int &ansy = ans[i][j].second;
        // move toward bottom
        if (i + 1 <= n && dp[i + 1][j] != INF) {
          if (dp[i + 1][j] < res) {
            res = dp[i + 1][j];
          }
        }
        // move toward top
        if (i - 1 >= 1 && a[i - 1] != j && dp[i - 1][j] != INF) {
          if (dp[i - 1][j] + u < res) {
            res = dp[i - 1][j] + u;
            ansx = i - 1;
            ansy = j;
          }
        }
        // move toward right
        if (j + 1 < MAXM && a[i] != j + 1 && dp[i][j + 1] != INF) {
          if (dp[i][j + 1] + v < res) {
            res = dp[i][j + 1] + v;
            ansx = i;
            ansy = j + 1;
          }
        }
        // move toward left
        if (j - 1 >= 1 && dp[i][j - 1] != INF) {
          if (dp[i][j - 1] + u < res) {
            res = dp[i][j - 1] + u;
            ansx = i;
            ansy = j - 1;
          }
        }
      }
    }
    if (dp[1][0] == INF)
      dp[1][0] = -1;
    printf("%d\n", dp[1][0]);
  }
  return 0;
}
