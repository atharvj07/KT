#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<double> a(2 * n);
    for (double &x : a) cin >> x;
    sort(a.begin(), a.end());
    double ans = 0;
    for (int i = 0; i < n; i++) ans += floor(a[i]) + ceil(a[2 * n - 1 - i]) - a[i] - a[2 * n - 1 - i];
    cout << fixed << setprecision(3) << abs(ans) << endl;
}
