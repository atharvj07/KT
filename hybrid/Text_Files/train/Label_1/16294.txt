#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, T;
  cin >> n >> T;
  vector<pair<int, int>> taps(n);
  for (int i = 0; i < n; i++) {
    int a, t;
    cin >> a >> t;
    taps[i] = {a, t};
  }
  sort(taps.begin(), taps.end(), [&](pair<int, int> &p1, pair<int, int> &p2) { return (1.0 * p1.second) / p1.first > (1.0 * p2.second) / p2.first; });
  double sum_a = 0, sum_ta = 0;
  for (int i = 0; i < n; i++) {
    double a = taps[i].first, t = taps[i].second;
    sum_a += a;
    sum_ta += a * t;
  }
  double x = (sum_ta - sum_a * T) / (double)(T * sum_a - sum_ta);
  double ans = 0;
  for (int i = 0; i < n; i++) {
    double a = taps[i].first, t = taps[i].second;
    ans += min(a, x * a / sum_a);
  }
  cout.precision(15);
  cout << ans << endl;
  return 0;
}
