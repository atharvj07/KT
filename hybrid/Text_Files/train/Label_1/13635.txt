#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<long long> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  std::sort(a.begin(), a.end());
  long long sum = 0;
  for (int i = 0; i < n; i++) { sum += a[i]; }
  int k = 0;
  while (k < n - 1 && a[k] == a[k + 1]) { k++; }
  std::cout << sum - k * a[k] << ' ' << k << '\n';
  for (int i = 0; i < k; i++) { std::cout << k + 1 + i << ' '; }
  return 0;
}
