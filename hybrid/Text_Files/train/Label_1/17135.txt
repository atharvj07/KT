#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
bool f(vector<int>& a){
    sort(a.begin(),a.end());
    int n=a.size();
    for(int i=0;i<n-2;i++){
        if((long long)a[i]+(long long)a[i+1]>(long long)a[i+2])return 1;
    }
    return 0;
}
int main(){
    int n;cin>>n;vector<int>a(n);
    for(int&x:a)cin>>x;
    cout<<(f(a)?"YES":"NO")<<endl;
}
