#include <bits/stdc++.h>
using namespace std;

const int N = 3005;

int n, m;
int a[N], b[N];
vector<pair<int, int>> v;

int main() {
    cin >> n >> m;
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        if (a[i] == 1) sum += b[i];
        else v.emplace_back(b[i], a[i]);
    }
    sort(v.begin(), v.end());
    int sz = v.size();
    for (int i = 0; i < sz && sum <= (n + 1) / 2; i++) {
        sum += v[i].first;
        a[v[i].second] = 1;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) ans += a[i] != 1 ? b[i] : 0;
    cout << ans << endl;
    return 0;
}
