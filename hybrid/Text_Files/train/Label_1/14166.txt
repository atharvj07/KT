#include<iostream>
#include<string>
#include<sstream>
#include<vector>
using namespace std;

int main() {
    string expr;
    getline(cin, expr);

    int ans = 0; bool negative = false;
    for (char c: expr) {
        if (c == '+') negative = false;
        else if (c == '-') negative = true;
        else {
            int val = c - '0';
            ans += negative ? -val : val;
        }
    }
    cout << ans;
    return 0;
}
