#include <iostream>
#include <cmath>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long n, m, k;
    cin >> n >> m >> k;
    if (abs(n - m) % 2 != k % 2 || k < abs(n - m))
      cout << -1 << endl;
    else
      cout << min(abs(n - m), k) / 2 << endl;
  }
  return 0;
}
