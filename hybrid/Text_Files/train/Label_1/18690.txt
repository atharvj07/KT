#include <bits/stdc++.h>
const int mod = 1e9 + 7;
int32_t main() {
  std::string s, t;
  getline(std::cin, s);
  getline(std::cin, t);
  int n = s.size(), m = t.size();
  std::vector<int> dp(n + 1, 0), lst(n + 2, -1), prv(n + 2, -1);
  for (int i = 0; i < n; i++) {
    if (s[i] == t[0]) lst[i + 1] = i;
    prv[i + 1] = lst[i + 1];
    lst[i + 1] = lst[i];
  }
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    dp[i] = (2 * dp[i - 1]) % mod;
    if (lst[i] != -1) dp[i] = (dp[i] - dp[lst[i]] + mod) % mod;
  }
  int ans = 0;
  for (int i = m - 1; i <= n; i++) {
    if (lst[i] == -1) continue;
    ans = (ans + dp[lst[i]]) % mod;
  }
  std::cout << ans << std::endl;
}
