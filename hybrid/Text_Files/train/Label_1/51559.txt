import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    static int n, d, k;
    static ArrayList<Integer>[] adj;
    static int[] dist;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]); // number of nodes
        d = Integer.parseInt(line[1]); // maximum diameter of the tree
        k = Integer.parseInt(line[2]); // maximum degree of each vertex

        if (n <= 2 || d == 0 || d % 2 != 0 || d > 2 * (n - 1) || k == 0) {
            System.out.println("NO");
            return;
        }

        // create adjacency list
        adj = new ArrayList[n + 1];
        for (int i = 0; i < n + 1; i++) {
            adj[i] = new ArrayList<>();
        }

        // create the tree
        createTree();

        // calculate distances from the root
        dist = new int[n + 1];
        dist = bfs();

        // check if the diameter of the tree is d
        int maxDist = Arrays.stream(dist).max().getAsInt();
        if (maxDist != d) {
            System.out.println("NO");
            return;
        }

        // check if the degree of each vertex is at most k
        int maxDegree = Arrays.stream(adj).mapToInt(ArrayList::size).max().getAsInt();
        if (maxDegree > k) {
            System.out.println("NO");
            return;
        }

        // print "YES" and the tree
        System.out.println("YES");
        for (int i = 2; i <= n; i++) {
            System.out.println(i + " " + adj[i].get(0));
        }
    }

    static void createTree() {
        for (int i = 1; i < n; i++) {
            adj[i].add(i + 1);
            adj[i + 1].add(i);
        }

        Queue<Integer> q = new LinkedList<>();
        q.add(1);

        while (q.size() > 0) {
            int u = q.poll();

            // if the current node is a leaf node, add a new node to its adjacency list
            if (adj[u].size() == 1) {
                int newNode = n + 1;
                adj[u].add(newNode);
                adj[newNode].add(u);
                q.add(newNode);
            }
        }
    }

    static int[] bfs() {
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        dist[1] = 0;

        while (q.size() > 0) {
            int u = q.poll();

            for (int v : adj[u]) {
                if (dist[v] == 0) {
                    dist[v] = dist[u] + 1;
                    q.add(v);
                }
            }
        }

        return dist;
    }
}
