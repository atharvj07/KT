#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)
#define all(v) (v).begin(), (v).end()
typedef long long ll;

int main() {
  int d;
  cin >> d;
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> v[d];
  rep(i, d) {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    if (y1 == y2) {
      if (x1 > x2) swap(x1, x2);
      for (int j = x1; j <= x2; ++j) {
        v[i].emplace_back(j, y1);
      }
    } else if (x1 == x2) {
      if (y1 > y2) swap(y1, y2);
      for (int j = y1; j <= y2; ++j) {
        v[i].emplace_back(x1, j);
      }
    }
  }
  int l, r, t, b;
  cin >> l >> r >> t >> b;
  rep(i, d) {
    vector<pair<int, int>> hs;
    for (auto&& [x, y] : v[i]) hs.emplace_back(x, y);
    sort(all(hs));
    int ls = 0, rs = 0, ts = 0, bs = 0;
    rep(j, hs.size()) {
      int x = hs[j].first, y = hs[j].second;
      if (hs[j] == hs[max(j - 1, 0)]) continue;
      if (y > hs[j - 1].second) ++rs;
      if (y < hs[j - 1].second) ++ls;
      if (x < hs[j - 1].first) ++ts;
      if (x > hs[j - 1].first) ++bs;
    }
    if (ls == l && rs == r && ts == t && bs == b) {
      cout << i + 1 << endl;
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
