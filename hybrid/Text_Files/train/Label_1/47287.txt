import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 1; i < n; i++) {
            p[i] = sc.nextInt() - 1;
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }

        boolean[] visited = new boolean[n];
        int[] color = new int[n];
        int[] weight = new int[n];
        if (dfs(0, -1, p, x, visited, color, weight)) {
            System.out.println("POSSIBLE");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }

    static boolean dfs(int v, int p, int[] parent, int[] x, boolean[] visited, int[] color, int[] weight) {
        visited[v] = true;
        int sum = 0;
        for (int child : getChildren(v, parent)) {
            if (!visited[child] && !dfs(child, v, parent, x, visited, color, weight)) {
                return false;
            }
            sum += weight[child];
        }
        if (sum != x[v]) {
            return false;
        }
        if (color[v] == 0) {
            color[v] = 1;
            weight[v] = x[v];
        } else {
            weight[v] = x[v] - sum;
            if (weight[v] < 0) {
                return false;
            }
        }
        return true;
    }

    static List<Integer> getChildren(int v, int[] parent) {
        List<Integer> children = new ArrayList<>();
        for (int i = 0; i < parent.length; i++) {
            if (parent[i] == v) {
                children.add(i);
            }
        }
        return children;
    }
}
