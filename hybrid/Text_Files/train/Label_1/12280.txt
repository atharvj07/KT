#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll gcd(ll a, ll b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}
int main() {
    int t; cin >> t;
    while(t--) {
        ll a, m; cin >> a >> m;
        ll g = gcd(a, m);
        ll ans = (m + g - 1) / g;
        cout << ans - 1 << endl;
    }
}
