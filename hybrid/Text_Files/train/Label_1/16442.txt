#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    int a[n]; for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a+n);

    int ans = 1;
    int res = a[0];
    for (int i = 1; i < n; i++) {
        if (a[i] - res <= 1) {
            ans++;
            res = a[i];
        }
    }

    cout << ans << endl;
    for (int i = 0; i < ans; i++) {
        cout << res << " ";
        res++;
    }
    cout << endl;
}
