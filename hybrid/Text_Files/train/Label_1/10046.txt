#include<bits/stdc++.h>
#define all(x) x.begin(), x.end()
using namespace std;

typedef long long int ll;
typedef pair<int, int> ii;

const int maxp = 2000000;

ll a[maxp], q, dp[maxp][2];
ll n, m, mx;

ll func(ll p, bool hit) {
    if(p == 0) return 0LL;

    ll &ret = dp[p][hit]; if(ret != -1) return ret;
    ret = 0LL;

    ll ch = min(q, 1LL<<(p + p));
    for(ll i = 1; i <= ch; i++) {
        mx = *max_element(a + i, a + i + ch);
        ret = max(ret, mx + func(p - 1, hit | 1));
    }

    return ret;
}

int main() {
    memset(dp, -1, sizeof dp); scanf("%lld", &q);
    for(ll i = 1; i <= q; i++) scanf("%lld", &a[i]);
    sort(a + 1, a + 1 + q, greater<ll>());

    n = log2(q) / 2; m = q / (1LL << (n + n));
    printf("%lld\n", func(n, 0));

    return 0;
}
