import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.List;

public class Main {

  static char[][] keyboard;
  static int n, m, x;
  static List<Character> letters = Arrays.asList('a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p',
      'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z');

  public static void main(String[] args) throws IOException {
    BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
    String[] line = input.readLine().split(" ");
    n = Integer.parseInt(line[0]);
    m = Integer.parseInt(line[1]);
    x = Integer.parseInt(line[2]);
    keyboard = new char[n][m];
    for (int i = 0; i < n; i++) {
      keyboard[i] = input.readLine().toCharArray();
    }
    String text = input.readLine();
    int result = solve(text);
    System.out.println(result);
  }

  private static int solve(String text) {
    int count = 0;
    int currentX = 0;
    int currentY = 0;
    for (char c : text.toCharArray()) {
      if (Character.isUpperCase(c)) {
        c = Character.toLowerCase(c);
        count++;
      }
      int[] nextPos = find(c);
      if (nextPos == null) {
        return -1;
      }
      int distance = Math.abs(currentX - nextPos[0]) + Math.abs(currentY - nextPos[1]);
      if (distance > x) {
        count++;
        currentX = nextPos[0];
        currentY = nextPos[1];
      }
      currentX = nextPos[0];
      currentY = nextPos[1];
    }
    return count;
  }

  private static int[] find(char c) {
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (keyboard[i][j] == 'S' && (j == 0 || keyboard[i][j - 1] != 'S')) {
          return new int[]{i, j};
        }
        if (keyboard[i][j] == c) {
          return new int[]{i, j};
        }
      }
    }
    return null;
  }
}
