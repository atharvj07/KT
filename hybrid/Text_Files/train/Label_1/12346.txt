#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;
    cin >> n;

    string love = "I hate";
    string hate = " that I love";

    for (int i = 0; i < n; i++)
        if (i % 2 == 0)
            love += hate;
        else
            hate += hate;

    cout << love << " it" << endl;

    return 0;
}
