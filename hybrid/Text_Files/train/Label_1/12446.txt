#include <bits/stdc++.h>
#define int long long
using namespace std;
const int inf = 1e18;
const int N = 1e5 + 5;
int n, m, a[N], c[N];
vector<pair<int, int>> order[N];
set<pair<int, int>> food; // price, amount

int32_t main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        for (int j = 1; j <= a[i]; j++)
            food.insert({c[i], 1});
    }
    for (int i = 1; i <= n; i++)
        cin >> c[i];
    while (m--)
    {
        int t, d;
        cin >> t >> d;
        order[t].push_back({d, 0});
    }
    int ans = 0;
    for (int i = 1; i <= n; i++)
    {
        for (auto &p : order[i])
        {
            if (!food.size() || a[i] == 0)
                break;
            p.second = min(p.first, a[i]);
            a[i] -= p.second;
            if (p.second)
                food.erase({c[i], 1});
        }
        for (auto &p : order[i])
        {
            if (p.second == 0)
                continue;
            if (p.second <= food.begin()->second)
            {
                food.erase(food.begin());
                ans += c[i] * p.second;
                continue;
            }
            food.erase({c[i], 1});
            ans += c[i];
            p.second--;
        }
        while (food.size() && a[i])
        {
            a[i]--;
            food.erase({c[i], 1});
            food.insert({inf, 1});
        }
    }
    cout << ans << '\n';
}
