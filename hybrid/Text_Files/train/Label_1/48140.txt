import java.io.*;
import java.util.*;

class Main {
    static char[][] map;
    static int n, m, k;
    static int[] dx = {0, 0, -1, 1};
    static int[] dy = {-1, 1, 0, 0};

    static boolean dfs(int x, int y, int[][] vis) {
        if (x < 0 || x >= n || y < 0 || y >= m) return false;
        if (map[x][y] == '#') return false;
        if (vis[x][y] == 2) return false;
        if (vis[x][y] == 1) return true;
        vis[x][y] = 1;
        boolean ok = true;
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            if (!dfs(nx, ny, vis)) ok = false;
        }
        if (!ok) vis[x][y] = 2;
        return ok;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        k = Integer.parseInt(line[2]);
        map = new char[n][m];
        for (int i = 0; i < n; i++) {
            map[i] = br.readLine().toCharArray();
        }
        int[][] vis = new int[n][m];
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == '.') {
                    if (!dfs(i, j, vis)) ok = false;
                    for (int x = 0; x < n; x++) {
                        for (int y = 0; y < m; y++) {
                            if (vis[x][y] == 1) {
                                k--;
                                vis[x][y] = 0;
                                map[x][y] = 'X';
                            }
                            if (k == 0) break;
                        }
                        if (k == 0) break;
                    }
                }
            }
            if (!ok) break;
        }
        for (char[] line1 : map) {
            System.out.println(line1);
        }
    }
}
