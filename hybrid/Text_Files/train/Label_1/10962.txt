#include<bits/stdc++.h>
using namespace std;
int32_t main(){
  long long n, l, c, ans = 10000000000, sum = 0;
  cin >> n;
  vector<long long> len, cost;
  for(long long i = 0; i < n; i++){
    cin >> l;
    len.push_back(l);
  }
  for(long long i = 0; i < n; i++){
    cin >> c;
    cost.push_back(c);
  }
  sort(len.begin(), len.end());
  sort(cost.begin(), cost.end());
  
  for(long long i = 0; i < n; i++){
    sum += cost[i];
    if(sum < len[i]) break;
    ans = min(ans, sum);
  }
  if(ans == 10000000000) cout << -1 << endl;
  else cout << ans << endl;
  return 0;
}
