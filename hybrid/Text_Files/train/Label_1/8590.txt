#include <algorithm>
#include <vector>
#include <map>
#include <iostream>

int main() {
  int n;
  std::cin >> n;
  std::map<std::pair<int, int>, bool> edges;
  for (int i = 0; i < n * 2; ++i) {
    int a, b;
    std::cin >> a >> b;
    edges[std::make_pair(a, b)] = true;
  }
  std::vector<int> perm(n);
  for (int i = 0; i < n; ++i) perm[i] = i + 1;
  do {
    bool valid = true;
    for (int i = 0; i < n; ++i) {
      if (!edges[std::make_pair(perm[i], perm[(i + 1) % n])] &&
          !edges[std::make_pair(perm[i], perm[(i - 1 + n) % n])]) {
        valid = false;
        break;
      }
    }
    if (valid) {
      for (int i = 0; i < n; ++i) {
        std::cout << perm[i] << ' ';
      }
      std::cout << '\n';
      return 0;
    }
  } while (std::next_permutation(perm.begin(), perm.end()));
  std::cout << "-1\n";
  return 0;
}
