x, y, n = map(int, input().split())
ans = [0, 1]
ans_num = float('inf')
for i in range(1, n + 1):
    num = x // y * i
    if abs(x / y - num / i) < ans_num:
        ans_num = abs(x / y - num / i)
        ans = [num, i]
    elif abs(x / y - num / i) == ans_num:
        if i < ans[1]:
            ans = [num, i]

print('/'.join(map(str, ans)))
