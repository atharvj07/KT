#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
int main() {
    int n, p;
    cin >> n >> p;
    string s;
    cin >> s;
    bool f[n + 1][2];
    memset(f, 0, sizeof(f));
    bool first = true;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j <= 1; j++) {
            int lo = max(0, s[i] - 'a' - j);
            int hi = p - 1;
            if (first && lo == 0) lo++;
            first = false;
            for (int k = lo; k <= hi; k++) {
                f[i][j] |= f[i + 1][j ^ (s[i] - 'a' != k)];
            }
        }
    }
    if (!f[0][0]) {
        cout << "NO" << endl;
        return 0;
    }
    string ans = "";
    int i = 0, j = 0;
    for (; i < n; i++) {
        int lo = max(0, s[i] - 'a' - j);
        int hi = p - 1;
        if (!f[i + 1][j ^ (s[i] - 'a' != hi)]) {
            ans += (char)(hi + 'a');
            j ^= 1;
        } else lo++;
        if (lo > hi) {
            cout << "NO" << endl;
            return 0;
        }
        ans += (char)(lo + 'a');
    }
    cout << ans << endl;
    return 0;
}
