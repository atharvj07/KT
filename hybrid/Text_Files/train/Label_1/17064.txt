#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  int a = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      a++;
    } else {
      a--;
    }
    if (a < 0) {
      a = 1;
      ans += i + 1;
    }
  }
  if (a == 0) {
    cout << ans << '\n';
  } else {
    cout << -1 << '\n';
  }
}
