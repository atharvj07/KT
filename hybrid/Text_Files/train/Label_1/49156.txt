import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long p = scanner.nextLong();
        if (p == 0) {
            long ans = 0;
            while (n > 0) {
                if ((n & 1) == 0) {
                    ans++;
                }
                n >>= 1;
            }
            System.out.println(ans);
        } else {
            long ans = 0;
            while (n > 0) {
                if (p > 0) {
                    while (n >= 2 * p) {
                        n -= 2 * p;
                        ans++;
                    }
                    if (n >= p) {
                        n -= p;
                        ans++;
                    }
                } else {
                    if (n >= 1 - p) {
                        n -= 1 - p;
                        ans++;
                    }
                }
                if (n > 0) {
                    ans++;
                    break;
                }
            }
            if (n == 0) {
                System.out.println(ans);
            } else {
                System.out.println(-1);
            }
        }
    }
}
