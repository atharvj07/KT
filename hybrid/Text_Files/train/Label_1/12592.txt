#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int max_n = 2e5 + 1;
int n, yay, whoops;
vector<int> cnt(max_n);
string s, t;

int main() {
    cin >> s >> t;
    n = s.size();
    for (int j = 0; j < n; ++j)
        cnt[t[j]]++;
    for (int j = 0; j < n; ++j) {
        char c = islower(s[j]) ? tolower(t[j]) : toupper(t[j]);
        if (cnt[c] > 0) {
            ++cnt[c], --cnt[t[j]];
            yay++, j--;
        }
        else
            whoops++;
    }
    cout << yay << ' ' << whoops << '\n';
}
