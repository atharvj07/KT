import sys
input = sys.stdin.readline

for i in range(int(input())):
    n,k = map(int,input().split())
    arr = ['a' for i in range(n)]
    arr[n-2:n] = 'b','b'
    k-=1
    for i in range(n-3,-1,-1):
        if k == 0: break
        if arr[i] == 'a' and arr[i+1] == 'b':
            arr[i],arr[i+1] = arr[i+1],arr[i]
            k-=1
    print(''.join(arr))
