#include<bits/stdc++.h>
using namespace std;
int n,x[100010],y[100010];
bool ok(int m){
    for(int i=2;i<=n;i+=2){
        if(x[i-1]+x[i]==m)continue;
        return 0;
    }
    return 1;
}
int main(){
    cin>>n;
    for(int i=2;i<=n;i+=2)cin>>x[i];
    int l=0,r=1e7,ans;
    while(l<=r){
        int m=(l+r)>>1;
        if(ok(m))r=m-1,ans=m;
        else l=m+1;
    }
    if(ok(ans)){
        cout<<"Yes\n";
        for(int i=2;i<=n;i+=2){
            if(i==2)cout<<ans-x[2]<<' ';
            else cout<<y[i-2]<<' ';
        }
        cout<<endl;
    }
    else cout<<"No\n";
}
