#include<iostream>
using namespace std;
const int INF = 1e9;
const int MAXN = 2e5 + 5;
const int MOD = 1e9 + 7;
int n, m, a[MAXN], b[MAXN], c[MAXN];

int sum(int l, int r) {
    return a[r] - (l == 0 ? 0 : a[l - 1]);
}

int main() {
    cin >> n >> m;
    for(int i = 0; i < n; i++) {
        char tmp;
        cin >> tmp;
        a[i] = tmp - '0';
    }
    for(int i = 0; i < m; i++) {
        char tmp;
        cin >> tmp;
        b[i] = tmp - '0';
    }
    for(int i = 0; i < m - n + 1; i++) {
        c[i] = sum(i, i + n - 1) != b[i];
        if(i > 0) c[i] += c[i - 1];
    }
    cout << c[m - n] << endl;
}
