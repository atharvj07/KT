import sys
input = sys.stdin.readline
n=int(input())
adj=[[] for i in range(n+1)]
for i in range(n-1):
    u,v=map(int,input().split())
    adj[u].append(v)
    adj[v].append(u)
x=[0 for i in range(n+1)]
y=[0 for i in range(n+1)]
vis=[False for i in range(n+1)]
def dfs(v,p):
    vis[v]=True
    for u in adj[v]:
        if vis[u]==False:
            if x[v]==x[p]:
                x[u]=x[v]+1
            else:
                y[u]=y[v]+1
            dfs(u,v)
dfs(1,1)
if x[n]==x[1] and y[n]==y[1]:
    print('NO')
else:
    print('YES')
    for i in range(1,n+1):
        print(x[i],y[i])
