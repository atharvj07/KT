#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int d, n, m;
    cin >> d >> n >> m;
    vector<vector<pair<int, int>>> sofas(d);
    for (int i = 0; i < d; i++) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        sofas[i].push_back({x1, y1});
        sofas[i].push_back({x2, y2});
    }
    int l, r, t, b;
    cin >> l >> r >> t >> b;
    int ans = -1;
    for (int i = 0; i < d; i++) {
        bool flag = true;
        int cntl = 0,cntr = 0,cntt = 0,cntb = 0;
        sort(sofas[i].begin(), sofas[i].end());
        for (int j = 0; j < d; j++) {
            if (i == j) continue;
            if (sofas[i][0].first > sofas[j][0].first && sofas[i][0].first < sofas[j][1].first) cntl++;
            if (sofas[i][0].first < sofas[j][0].first && sofas[i][1].first > sofas[j][0].first) cntr++;
            if (sofas[i][0].second < sofas[j][0].second && sofas[i][1].second > sofas[j][0].second) cntt++;
            if (sofas[i][0].second > sofas[j][0].second && sofas[i][1].second < sofas[j][1].second) cntb++;
        }
        if (cntl == l && cntr == r && cntt == t && cntb == b) {
            ans = i + 1;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
