import sys

input = sys.stdin.readline

n, x = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

depth = [0] * (n + 1)

def dfs(v, p, d):
    depth[v] = d
    for u in g[v]:
        if u == p:
            continue
        dfs(u, v, d + 1)

dfs(1, 1, 0)

y = -1
mydepth = -1
for i in range(1, n + 1):
    if i == x:
        continue
    if depth[i] > mydepth:
        mydepth = depth[i]
        y = i

print(mydepth * 2)
