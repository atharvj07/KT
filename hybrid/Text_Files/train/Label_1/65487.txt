import sys
import heapq as hq
input = sys.stdin.readline

n,k = map(int,input().split())
a = sorted([(x[0]-x[1],x[1]) for x in [list(map(int,input().split())) for _ in range(n)]])
s,mn = 0,0
for x,y in a:
    if s+x < mn:
        s += x
        mn += 1
    if mn >= k-1:
        break
print(-1 if s != 0 else s)
