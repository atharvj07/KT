#include <bits/stdc++.h>

using namespace std;
vector<int> ans;

void solve(int n, int k) {
    int cur = n;
    for (int i = 0; i < k; ++i) {
        ans.push_back(cur);
        cur--;
    }
    ans.push_back(n - k);
    while (ans.size() != n) {
        ans.push_back(n - k - ans.size());
    }
}

int main() {
    int n, k;
    cin >> n >> k;
    solve(n, k);
    for (int x : ans) cout << x << " ";
    cout << "\n";
}
