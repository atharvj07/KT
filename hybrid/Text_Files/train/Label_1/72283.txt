a,b=map(int,input().split())
l=[]
while a!=b:
    if b%2==0 and b/2>=a:
        b=int(b/2)
        l.append(b)
    elif b*10+1<=1e9 and b*10+1>=a:
        b=b*10+1
        l.append(b)
    else:
        l=[]
        print('NO')
        break
if l!=[]:
    print('YES')
    print(len(l))
    print(*l)
