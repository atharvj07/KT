from collections import defaultdict

def is_k_multihedgehog(n, k, edges):
    if n == 1:
        return k == 0
    degree = defaultdict(int)
    for u, v in edges:
        degree[u] += 1
        degree[v] += 1

    for _ in range(k):
        leaf_parents = defaultdict(list)
        for v, d in degree.items():
            if d == 1:
                for u, w in edges:
                    if u == v:
                        leaf_parents[w].append(v)
                        break
                    elif w == v:
                        leaf_parents[u].append(v)
                        break
        if not leaf_parents or any(len(leafs) < 3 for leafs in leaf_parents.values()):
            return False
        for parent, leafs in leaf_parents.items():
            degree[parent] -= len(leafs)
            if degree[parent] == 0:
                del degree[parent]
            for leaf in leafs:
                del degree[leaf]
                edges = [(u, w) for u, w in edges if u != leaf or w != leaf]
        if not degree:
            return False
    return True if all(d == 1 or d >= 3 for d in degree.values()) else False

n, k = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(n - 1)]

print("Yes" if is_k_multihedgehog(n, k, edges) else "No")
