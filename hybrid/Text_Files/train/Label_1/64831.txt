n = int(input())
a = list(map(int, input().split()))

MOD = 998244353
length = len(str(a[0]))
total = 0

for i in range(length):
    digit_sum = sum([(x // (10 ** i)) % 10 for x in a]) % MOD
    total = (total + (digit_sum * n * pow(10, 2*i, MOD))) % MOD
    total = (total + (digit_sum * n * pow(10, 2*i + 1, MOD))) % MOD

print(total)
