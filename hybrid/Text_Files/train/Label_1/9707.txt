#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    cin>>s;
    int lf=s.length(),k=s.length()+1;
    for(int i=0; i<lf; i++)
    {
        int j=i+1;
        while(j<lf && s[j]==s[i])
        {
            j++;
        }
        if(j-i+1<=k)
        {
            k=j-i+1;
        }
        i=j;
    }
    cout<<k<<endl;
    return 0;
}
