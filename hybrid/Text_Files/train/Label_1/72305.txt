n, m = map(int, input().split())
a = [list(input()) for i in range(n)]
ans = float('inf')
for j in range(m):
    b = []
    for i in range(n):
        b.append(a[i][j])
    c = sorted(b)
    f = False
    for i in range(n):
        if b[i] > c[i]:
            f = True
    if not f:
        ans = 0
    for i in range(n):
        if b[i] != c[i]:
            break
    if not f:
        ans = min(ans, j)
print(ans)
