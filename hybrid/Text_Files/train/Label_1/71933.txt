n, m = map(int, input().split())
a = [list(input().split()) for i in range(n)]
b = [sorted([i for i in a if i[1] == str(j)], key=lambda x: (-int(x[2]), x[0])) for j in range(1, m + 1)]
for i in range(m):
    if len(b[i]) < 2 or int(b[i][0][2]) == int(b[i][1][2]):
        print('?')
    else:
        print(b[i][0][0], b[i][1][0])
