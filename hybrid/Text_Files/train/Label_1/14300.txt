#include <iostream>
#include <vector>

using namespace std;

// Return the ordinal number of the given permutation
// (the smallest permutation has ordinal number 0)
long long ord(vector<int> &p) {
    int n = p.size();
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        // Count the number of elements in suffix that are smaller than p[i]
        int cnt = 0;
        for (int j = i + 1; j < n; j++) {
            if (p[j] < p[i]) cnt++;
        }
        // Update the ordinal number of p
        ans += cnt * (i == 0 ? 1 : i * (i - 1) / 2);
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    vector<int> p(n), q(n);
    for (int &x : p) cin >> x;
    for (int &x : q) cin >> x;
    // Find the sum of the two permutations
    vector<int> sum(n);
    long long o = ord(p) + ord(q);
    for (int i = 0; i < n; i++) {
        // Find the i-th element in the sum permutation
        for (int j = 0; j < n; j++) {
            if (o >= (i == 0 ? 1 : i * (i - 1) / 2)) {
                o -= (i == 0 ? 1 : i * (i - 1) / 2);
            } else {
                sum[i] = j;
                break;
            }
        }
    }
    for (int x : sum) cout << x << " ";
    cout << endl;
    return 0;
}
