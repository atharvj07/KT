#include <bits/stdc++.h>
using namespace std;
#define int long long

int power(int x, int y) {
  if (y == 0) return 1;
  if (y == 1) return x;
  int a = power(x, y / 2);
  a = (a * a) % 1000000007;
  if (y % 2 == 1) a = (a * x) % 1000000007;
  return a;
}

int32_t main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int cnt0 = count(a.begin(), a.end(), 0);
  int cnt1 = count(a.begin(), a.end(), 1);
  if (cnt0 > cnt1) {
    cout << 0 << endl;
    return 0;
  }
  int tot = n * (n - 1) / 2;
  int num = cnt0 * (cnt0 - 1) / 2 + cnt1 * (cnt1 - 1) / 2;
  int den = cnt0 * cnt1;
  int ans = (tot - num) * power(den, 1000000005) % 1000000007;
  cout << ans << endl;
  return 0;
}
