#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int ts, tf, t, n;
  cin >> ts >> tf >> t >> n;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) cin >> arr[i];
  int ans = 0, cur = 0;
  for (int i = n - 1; i >= 0; i--) {
    int x = arr[i] - cur;
    int need = ((x + t - 1) / t) * t;
    if (ts <= cur + need && cur + need <= tf - t) {
      ans = cur;
      break;
    }
    cur += max(need, t);
  }
  return cout << ans, 0;
}
