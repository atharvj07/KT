def dfs(curr_node, min_ans, max_ans, h, arr):
    if curr_node > (1 << h) - 1:
        return
    if min_ans <= curr_node <= max_ans:
        arr[0] = curr_node
        dfs(curr_node * 2, min_ans, (max_ans + 1) // 2 - 1, h, arr)
        dfs(curr_node * 2 + 1, (min_ans + 1) // 2, max_ans, h, arr)


h, q = map(int, input().split())
game_ans = []
exit_node = [0]
for i in range(q):
    node_num, min_ans, max_ans, ans = map(int, input().split())
    if i == 0:
        if ans == 1:
            dfs(1, 0, (1 << h) - 1, h, exit_node)
        if ans == 0:
            print("Game cheated!")
            exit(0)
    else:
        ans = 1 - ans
        if ans == 1:
            dfs(1, min_ans, max_ans, h, exit_node)
        if ans == 0:
            if min_ans <= exit_node[0] <= max_ans:
                print("Game cheated!")
                exit(0)
