n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

b.sort(reverse=True)
j = 0
for i in range(n):
    if a[i] == 0:
        a[i] = b[j]
        j += 1

increasing = True
for i in range(1, n):
    if a[i] <= a[i-1]:
        increasing = False
        break

if increasing:
    print("No")
else:
    print("Yes")
