#include <vector>
#include <iostream>
#include <algorithm>

int main() {
    int n, v;
    std::cin >> n >> v;
    std::vector<std::vector<int>> items(n);
    for (int i = 0; i < n; i++) {
        int k;
        std::cin >> k;
        items[i].resize(k);
        for (int j = 0; j < k; j++) {
            std::cin >> items[i][j];
        }
        std::sort(items[i].rbegin(), items[i].rend());
    }
    std::vector<int> sellers;
    for (int i = 0; i < n; i++) {
        if (!items[i].empty() && items[i][0] <= v) {
            sellers.push_back(i + 1);
        }
    }
    std::cout << sellers.size() << '\n';
    for (int seller : sellers) {
        std::cout << seller << ' ';
    }
    return 0;
}
