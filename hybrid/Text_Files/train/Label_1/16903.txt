#include <iostream>

using namespace std;

int main() {
    int c, v0, v1, a, l;
    cin >> c >> v0 >> v1 >> a >> l;
    int days = 0, pages = 0;
    while (pages < c) {
        if (days > 0) {
            pages -= l;
            if (pages < 0) pages = 0;
        }
        days++;
        int v = min(v0 + a * (days - 1), v1);
        pages += v;
    }
    cout << days << endl;
    return 0;
}
