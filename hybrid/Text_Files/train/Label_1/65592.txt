n = int(input().strip())
schedule = list(map(int, input().strip().split()))

# Extend the schedule to two days to consider the circular nature of the schedule
extended_schedule = schedule + schedule

max_rest = current_rest = 0

for hour in extended_schedule:
    if hour == 1:
        current_rest += 1
        max_rest = max(max_rest, current_rest)
    else:
        current_rest = 0

print(max_rest)
