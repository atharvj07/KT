import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt(), k = s.nextInt();
        s.nextLine();
        int[] t = Arrays.stream(s.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray(),
                c = Arrays.stream(s.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = dp.length - 1; j >= 0; j--) {
                if (dp[j] >= 0 || j == 0) {
                    if (Math.abs(((double) (dp[j] + t[i])) / (j + 1) - k) < 1e-6) {
                        dp[j + 1] = Math.max(dp[j] + t[i], dp[j + 1]);
                    }
                }
            }
        }
        Arrays.sort(dp);
        System.out.println(dp[dp.length - 1] == -1 ? -1 : dp[dp.length - 1]);
    }
}
