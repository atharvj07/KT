import java.util.*;

public class Main {
    static long[] a;
    static HashSet<Long> set;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            a = new long[n];
            for (int i = 0; i < n; i++)
                a[i] = sc.nextLong();
            set = new HashSet<>();
            for (int i = 0; i < n; i++)
                set.add(a[i]);
            boolean ok = false;
            for (int i = 0; i < n - 2; i++) {
                for (int j = i + 1; j < n - 1; j++) {
                    if (set.contains(a[i] + a[j] - 1) || set.contains(a[i] + a[j] + 1)) {
                        System.out.println(i + 1 + " " + (j + 1) + " " + (n));
                        ok = true;
                        break;
                    }
                }
            }
            if (!ok)
                System.out.println(-1);
        }
    }
}
