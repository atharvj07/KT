import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int y = in.nextInt(), k = in.nextInt(), n = in.nextInt();
        List<Integer> ans = new ArrayList<>();
        int t = n - y;
        if (t % k != 0) {
            System.out.println(-1);
        } else {
            for (int i = 1; i <= t / k; i++) {
                if (y < i * k) {
                    ans.add(i * k - y);
                }
            }
            if (ans.isEmpty()) {
                System.out.println(-1);
            } else {
                for (int x : ans) {
                    System.out.print(x + " ");
                }
                System.out.println();
            }
        }
    }
}
