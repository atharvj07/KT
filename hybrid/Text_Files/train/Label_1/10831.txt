#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  vector<int> dp(n + 1, 0);
  int ans = 0, zero_count = 0;
  for (int i = 0; i < n; i++) {
    if (a[i]) {
      dp[i + 1] = dp[i] + 1;
      ans = max(ans, dp[i + 1]);
    } else {
      zero_count++;
      if (zero_count <= k) { ans = max(ans, dp[i] + 1); }
      dp[i + 1] = 0;
    }
  }

  int j = n;
  while (j > 0 && zero_count > 0) {
    if (a[j - 1] == 0 && zero_count > 0) {
      a[j - 1] = 1;
      zero_count--;
    }
    j--;
  }

  cout << ans << endl;
  for (int i = 0; i < n; i++) { cout << a[i] << " "; }
  cout << endl;
  return 0;
}
