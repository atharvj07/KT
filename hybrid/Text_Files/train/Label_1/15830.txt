#include <iostream>

using namespace std;

int main() {
  int hour, minute, add, ans_hour, ans_minute;
  cin >> hour >> minute >> add;
  ans_minute = minute + add;
  ans_hour = hour + ans_minute / 60;
  ans_minute %= 60;
  ans_hour %= 24;
  cout.fill('0');
  cout.width(2);
  cout << ans_hour << ':';
  cout.width(2);
  cout << ans_minute << endl;
}
