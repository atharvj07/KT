#include<bits/stdc++.h>
using namespace std;
#define ll long long
void solve(){
ll n,p;cin>>n>>p;
vector<ll> a(n);for(ll &x:a) cin>>x;
sort(a.begin(),a.end()); 
vector<ll> ans;
for(ll x=1;x<=n;x++){
ll sum=0;
for(ll i=0;i<n;i++){
if(a[i]<=x) sum++;
}
if(sum%p) ans.push_back(x);
}
cout<<ans.size()<<endl;
for(ll x:ans) cout<<x<<" ";
}
int main()
{solve();}
