import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        sc.close();
        if ((a + b + c) % 2 == 0 && (a + b >= c) && (b + c >= a) && (c + a >= b))
            System.out.println("Yes");
        else
            System.out.println("No");
    }
}
