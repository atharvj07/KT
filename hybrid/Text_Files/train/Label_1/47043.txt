import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    int n = in.nextInt();
    int m = in.nextInt();

    int[] rowNums = new int[n];
    int[][] grid = new int[n][m];

    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        grid[i][j] = in.nextInt();
        rowNums[i] += grid[i][j];
      }
    }

    Arrays.sort(rowNums);

    int maxCoins = 0;
    boolean isComb = true;

    for (int i = 0; i < n; i++) {
      if (i % 2 == 0) {
        maxCoins += rowNums[n - i - 1];
      } else {
        isComb &= rowNums[n - i - 1] < rowNums[n - i];
      }
    }

    if (isComb) {
      System.out.println(maxCoins);
    } else {
      System.out.println("-1");
    }
  }
}
