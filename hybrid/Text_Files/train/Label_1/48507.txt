import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        boolean isBerlanese = true;
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == 'a' || c == 'o' || c == 'u' || c == 'e' || c == 'i' || c == 'n') {
                continue;
            } else {
                if (i + 1 < s.length() && (s.charAt(i + 1) == 'a' || s.charAt(i + 1) == 'o' || s.charAt(i + 1) == 'u' || s.charAt(i + 1) == 'e' || s.charAt(i + 1) == 'i')) {
                    continue;
                } else {
                    isBerlanese = false;
                    break;
                }
            }
        }
        System.out.println(isBerlanese ? "YES" : "NO");
    }
}
