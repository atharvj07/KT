import java.util.*;

class Main {
    static int[] getSolution(int k, int[] arr) {
        Arrays.sort(arr);
        List<Integer> sol = new ArrayList<>();

        int i = 0;
        while (i < k && arr[i] == 0) i++;
        if (i < k) {
            sol.add(arr[i++]);
        }

        while (i < k) {
            int n1 = arr[i++];
            while (i < k && arr[i] <= n1) i++;
            if (i < k) {
                sol.add(arr[i++]);
            }
        }

        Collections.sort(sol);

        int[] ans = new int[sol.size()];
        for (i = 0; i < sol.size(); i++) {
            ans[i] = sol.get(i);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        int[] arr = new int[k];

        for (int i = 0; i < k; i++) {
            arr[i] = input.nextInt();
        }

        int[] ans = getSolution(k, arr);
        System.out.println(ans.length);
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
