n, k = map(int, input().split())
if n % 2 == 0 and k > n * n // 2 - 1:
    print(-1)
elif n % 2 == 1 and k > n * n // 2:
    print(-1)
else:
    for i in range(n):
        for j in range(n):
            if (n % 2 == 1 and i == j and k > 0) or (n % 2 == 0 and (i + j == n - 1 and k > 0) or (i == j and k > 1)):
                print(1, end=' ')
                k -= 1
            else:
                print(0, end=' ')
        print()
