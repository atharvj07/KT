#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b, n;
    cin >> a >> b >> n;
    if (b == 1)
      cout << "NO" << endl;
    else {
      int sum = 0;
      for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        sum += x;
      }
      if (sum <= b)
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
    }
  }
  return 0;
}
