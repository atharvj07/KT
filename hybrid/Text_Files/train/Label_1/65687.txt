def min_mana(n, m, x, k, y, a, b):
    b_idx = 0
    mana = 0
    prev = -1
    for i in range(n):
        if b_idx < m and a[i] == b[b_idx]:
            mana += min_mana_sub(a, prev, i, x, k, y)
            prev = i
            b_idx += 1
    if b_idx != m:
        return -1
    mana += min_mana_sub(a, prev, n, x, k, y)
    return mana

def min_mana_sub(a, start, end, x, k, y):
    max_power = max(a[start+1:end], default=-1)
    len_sub = end - start - 1
    if len_sub < k and max_power > max(a[start], a[end-1], default=-1):
        return -1
    if y * len_sub < x or len_sub < k:
        return y * len_sub
    return min(len_sub // k * x + (len_sub % k) * y, (len_sub - k) * y + x)

n, m = map(int, input().split())
x, k, y = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

print(min_mana(n, m, x, k, y, a, b))
