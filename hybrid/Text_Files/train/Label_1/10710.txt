#define fr(i,n) for(int i=0;i<n;++i)

const double pi = acos(-1);
void solve(){
    int n,x,f[100005]={0};
    cin >> n;
    fr(i,n){
        cin >> x;
        f[x]++;
    }
    set<int>vis;
    set<int>:: iterator it;
    fr(i,100005){
        if(f[i]==0){
            vis.insert(i);
        }
    }
    fr(i,100005){
        if(f[i]>1){
            if(vis.find(f[i])==vis.end()){
                cout << -1 << endl;
                return;
            }
            else vis.erase(f[i]);
        }
    }
    int sz = vis.size();
    cout << sz << endl;
    it = vis.begin();
    fr(i,n){
        if(f[i]==0)cout << *it << " ";
        else cout << f[i] << " ";
    }
    cout << endl;
    fr(i,sz){
        it=vis.begin();
        cout << *it << " ";
        vis.erase(it);
    }
    cout << endl;
}
int main(){
    ios_base :: sync_with_stdio(false);
    cin.tie(NULL);
    solve();
    return 0;
}
