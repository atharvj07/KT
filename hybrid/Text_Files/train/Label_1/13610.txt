#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, v;
    std::cin >> n >> v;
    std::vector<std::pair<int, std::vector<int>>> s;
    for (int i = 0; i < n; i++) {
        int k;
        std::cin >> k;
        std::vector<int> t;
        for (int j = 0; j < k; j++) {
            int x;
            std::cin >> x;
            t.push_back(x);
        }
        s.push_back({k, t});
    }
    std::sort(s.begin(), s.end(), [](const std::pair<int, std::vector<int>>& a, const std::pair<int, std::vector<int>>& b) { return a.second[0] < b.second[0]; });
    int p = 0;
    std::vector<int> ans;
    for (int i = 0; i < n; i++) {
        if (v < s[i].second[0]) break;
        p++;
        ans.push_back(i + 1);
        v -= s[i].second[0];
    }
    std::cout << p << "\n";
    for (int i = 0; i < p; i++) {
        std::cout << ans[i] << " ";
    }
    return 0;
}
