#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int y;
  std::cin >> y;
  while (++y) {
    std::vector<int> digits;
    int tmp = y;
    while (tmp > 0) {
      digits.push_back(tmp % 10);
      tmp /= 10;
    }
    std::sort(digits.begin(), digits.end());
    for (int i = 1; i < digits.size(); i++) {
      if (digits[i] == digits[i - 1]) {
        goto ctn;
      }
    }
    std::cout << y << std::endl;
    return 0;
  ctn:;
  }
}
