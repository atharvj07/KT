#include <iostream>

int main() {
  int x1, y1, x2, y2, x, y;
  std::cin >> x1 >> y1 >> x2 >> y2;
  std::cin >> x >> y;

  int dx = x2 - x1, dy = y2 - y1;
  if ((dx + x) % x != 0 || (dy + y) % y != 0) {
    std::cout << "NO" << std::endl;
  } else {
    std::cout << "YES" << std::endl;
  }
  return 0;
}
