#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  int pearls = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'o') pearls++;
  }
  if (pearls == 1) {
    cout << "YES";
    return 0;
  }
  if ((n - pearls) % (pearls - 1) == 0) {
    cout << "YES";
  } else {
    cout << "NO";
  }
  return 0;
}
