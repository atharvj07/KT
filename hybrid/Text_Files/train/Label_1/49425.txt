import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0 && s.charAt(i) != s.charAt(i - 1)) {
                ans++;
            }
            if (i + 1 < n && s.charAt(i) != s.charAt(i + 1)) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
