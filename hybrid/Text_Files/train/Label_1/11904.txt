#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)
using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    vector<vector<int>> id(51);
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        id[a].push_back(i);
    }
    for (int i = 0; i < q; i++) {
        int t;
        cin >> t;
        cout << id[t].back() + 1 << endl;
        id[t].pop_back();
        id[t].insert(id[t].begin(), id[t].back());
        id[t].pop_back();
    }
}
