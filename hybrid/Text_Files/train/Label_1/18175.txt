#include <bits/stdc++.h>

using namespace std;

using ll = long long;

ll gcd(ll a, ll b) {
    if (a == 0) {
        return b;
    }
    return gcd(b % a, a);
}

int main() {
    ll n, m;
    cin >> n >> m;

    if (n < 2 || n * (n - 1) / 2 < m) {
        cout << "Impossible\n";
        return 0;
    }

    cout << "Possible\n";

    ll cur = 2;
    for (ll i = 1; i < n; i++) {
        if (m == 0) {
            break;
        }
        if (gcd(i, cur) == 1) {
            cout << i << " " << cur << "\n";
            m--;
        }
        cur++;
    }

    cur = n - 1;
    for (ll i = 1; i < n; i++) {
        if (m == 0) {
            break;
        }
        if (i == cur) {
            cur--;
            continue;
        }
        if (gcd(i, cur) == 1) {
            cout << i << " " << cur << "\n";
            m--;
        }
    }

    return 0;
}
