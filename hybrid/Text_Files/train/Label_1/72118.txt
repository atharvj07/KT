l = int(input())
x = 0
for _ in range(l):
    s = input()
    if s[0] == 'a':
        x += 1
    else:
        end = s.find('d')
        start = 3
        while start != end:
            x *= int(s[start:start+2])
            start += 3
        while end+3 < len(s):
            x += int(s[end+3:end+7])*int(s[start:start+2])
            start += 3
            end += 7
if x > 2**32-1:
    print('OVERFLOW!!!')
else:
    print(x%(2**32))
