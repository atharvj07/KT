#include <iostream>
#include <vector>
#include <map>
#include <unordered_map>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<string, pair<int, int>>> students(n);
  for (int i = 0; i < n; ++i) {
    string name;
    int region, score;
    cin >> name >> region >> score;
    students[i] = {name, {region, score}};
  }
  sort(students.begin(), students.end(), [](const auto& a, const auto& b) {
    return a.second.second > b.second.second;
  });
  unordered_map<int, vector<string>> teams;
  for (const auto& [name, regionScore] : students) {
    auto& team = teams[regionScore.first];
    if (team.size() < 2) {
      team.push_back(name);
    }
  }
  for (const auto& [region, team] : teams) {
    if (team.size() < 2) {
      cout << "?\n";
    } else {
      cout << team[0] << " " << team[1] << "\n";
    }
  }
  return 0;
}
