import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        HashMap<String, Integer> map = new HashMap<>();
        int max = 0;
        String result = "";
        for (int i = 0; i <= n - 2; i++) {
            String sub = s.substring(i, i + 2);
            map.put(sub, map.getOrDefault(sub, 0) + 1);
            if (map.get(sub) > max) {
                result = sub;
                max = map.get(sub);
            }
        }
        System.out.println(result);
    }
}
