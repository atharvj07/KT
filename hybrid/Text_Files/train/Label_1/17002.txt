#include<iostream>
#include<vector>
using namespace std;
const int maxn=20010;
int dp[maxn][2][2]={0};
vector<int> num;
vector<int> op;
int main()
{
    string s;
    while( cin>>s )
    {
        int p,m;
        cin>>p>>m;
        num.clear();
        op.clear();
        int tt=0;
        for(int i=0;i<s.size();i++)
        {
            if(s[i]>='0' && s[i]<='9')
            {
                tt=tt*10+s[i]-'0';
            }
            else if(s[i]=='?')
            {
                op.push_back(0);
            }
            else
            {
                num.push_back(tt);
                tt=0;
            }
        }
        num.push_back(tt);
        int n=num.size();
        int m1=op.size();
        dp[0][1][0]=dp[0][0][1]=num[0];
        for(int i=1;i<n;i++)
        {
            for(int j=0;j<=m1;j++)
            {
                for(int k=0;k<=1;k++)
                {
                    if(j)
                    {
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j-1][k]+num[i]);
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j-1][k]-num[i]);
                    }
                    if(k==0)
                    {
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j][1]+num[i]);
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j][1]-num[i]);
                    }
                    if(k==1)
                    {
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j][0]+num[i]);
                        dp[i][j][k]=max(dp[i][j][k],dp[i-1][j][0]-num[i]);
                    }
                }
            }
        }
        cout<<dp[n-1][m1][1]<<endl;
    }
    return 0;
}
