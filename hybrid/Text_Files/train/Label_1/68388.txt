import sys

input = sys.stdin.readline

s = input()
vs = {'I', 'E', 'A', 'O', 'U', 'Y'}
sp = 0
for i in range(len(s)):
    for j in range(i + 1, len(s) + 1):
        sp += int(s[i:j].count('I') + s[i:j].count('E') + s[i:j].count('A') + s[i:j].count('O') + s[i:j].count('U') + s[i:j].count('Y')) / (j - i)
print("{0:.6f}".format(sp))
