import sys

for _ in range(int(input())):
    n, k = map(int, input().split())
    if k > (n - 1) // 2:
        print(-1)
    else:
        a = [i + 1 for i in range(n)]
        i = 1
        while k:
            if i + 1 < n:
                a[i], a[i + 1] = a[i + 1], a[i]
                k -= 1
            if i - 1 >= 0:
                a[i], a[i - 1] = a[i - 1], a[i]
                k -= 1
            i += 2
        print(*a)
