#include <bits/stdc++.h>
using namespace std;
#define MAXN 101
int n, m, k;
vector<int> c;
vector<vector<int>> p;
long long dp[MAXN][MAXN][MAXN][2];
long long calc(int pos, int cnt, int prv, bool ok) {
  if (cnt == k)
    return 0;
  long long &ret = dp[pos][cnt][prv][ok];
  if (ret != -1)
    return ret;
  ret = 1e18;
  if (pos == n) {
    if (cnt == k)
      ret = ok ? 0 : 1e18;
    return ret;
  }
  if (c[pos] != 0) {
    if (prv != c[pos])
      ret = calc(pos + 1, cnt + (ok == 0), c[pos], 0);
    else
      ret = calc(pos + 1, cnt, c[pos], ok);
  } else {
    for (int i = 1; i <= m; i++) {
      if (prv != i)
        ret = min(ret, p[pos][i - 1] + calc(pos + 1, cnt + (ok == 0), i, 0));
      else
        ret = min(ret, p[pos][i - 1] + calc(pos + 1, cnt, i, ok));
    }
  }
  return ret;
}
int main() {
  cin >> n >> m >> k;
  c.resize(n);
  p.resize(n);
  for (int &x : c)
    cin >> x;
  for (int i = 0; i < n; i++) {
    p[i].resize(m);
    for (int &y : p[i])
      cin >> y;
  }
  memset(dp, -1, sizeof(dp));
  long long ans = calc(0, 0, 0, 1);
  cout << (ans == 1e18 ? -1 : ans);
  return 0;
}
