n,p,s = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
c = list(zip(a,b))
c.sort(key = lambda x:x[0]+x[1],reverse = True)
ans = 0
ta=[]
tb=[]
for i in range(p):
    ans+=c[i][0]
    ta.append(i+1)
for i in range(s):
    ans+=c[i+p][1]
    tb.append(i+p+1)
print(ans)
print(*ta)
print(*tb)
