#include <bits/stdc++.h>

using namespace std;

const int inf = 1e9;

int main() {
  int n;
  cin >> n;
  vector<int> a(n / 2);
  for (int i = 0; i < n / 2; i++) { cin >> a[i]; }
  bool ok = true;
  int y = a[0];
  for (int i = 0; i < n / 2 - 1; i++) {
    int x = sqrt(y + a[i + 1]);
    if (x * x != y + a[i + 1]) {
      ok = false;
      break;
    }
    y = x;
  }
  if (!ok) {
    cout << "No" << endl;
    return 0;
  }
  cout << "Yes" << endl;
  for (int i = n / 2 - 1; i >= 0; i--) { cout << y << ' '; }
  for (int x : a) { cout << x << ' '; }
  cout << endl;
  return 0;
}
