import sys

n, U = map(int, input().split())
E = list(map(int, input().split()))

if E[2] - E[0] > U:
    print(-1)
    sys.exit()

max_eff = (E[2] - E[1]) / (E[2] - E[0])
for i in range(1, n - 2):
    if E[i + 2] - E[i] > U:
        break
    max_eff = max(max_eff, (E[i + 2] - E[i + 1]) / (E[i + 2] - E[i]))

print(round(max_eff, 9))
