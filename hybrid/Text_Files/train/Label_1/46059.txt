import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int n = in.nextInt();
        String strN = String.valueOf(n);
        int sum = 0;
        for (char c : strN.toCharArray()) {
            sum += c - '0';
        }
        System.out.println(sum < k ? strN.length() : 0);
    }
}
