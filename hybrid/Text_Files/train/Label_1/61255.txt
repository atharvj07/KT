import itertools

n = int(input())
q = [list(map(int, input().split())) for i in range(n - 2)]

# sort triples by first and second number
q.sort(key=lambda x: (x[0], x[1]))

# dfs to find permutation
vis = [False] * n
p = []
for x, y, z in q:
    if not vis[x - 1]:
        vis[x - 1] = True
        vis[y - 1] = True
        vis[z - 1] = True
        p.extend([x, y, z])
for i in range(n):
    if not vis[i]:
        p.append(i + 1)
print(*p)
