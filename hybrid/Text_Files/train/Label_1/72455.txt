n = int(input())
a = list(map(int, input().split()))

prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sum[i] = prefix_sum[i - 1] + a[i - 1]

def g(i, j):
    return abs(prefix_sum[j] - prefix_sum[i])

min_value = float('inf')
for i in range(1, n + 1):
    for j in range(1, n + 1):
        if i != j:
            min_value = min(min_value, (i - j)**2 + g(min(i, j), max(i, j))**2)

print(min_value)
