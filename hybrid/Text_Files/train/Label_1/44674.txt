import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int sum = 0;
        for (int x : a) {
            sum += x;
        }
        if (sum % 2 == 0) {
            System.out.println("Second");
        } else {
            System.out.println("First");
        }
    }
}
