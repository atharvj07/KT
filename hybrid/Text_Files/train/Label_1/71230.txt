import sys

s = input()
t = input()

if s == t:
    print(0)
else:
    n = len(s)
    m = len(t)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        for j in range(m):
            if s[i] == t[j]:
                dp[i][j] = dp[i + 1][j + 1] + 1
            dp[i][j] = max(dp[i][j], dp[i + 1][j], dp[i][j + 1])
    if dp[0][0] != m:
        print(-1)
    else:
        i, j, ans = 0, 0, []
        while i < n and j < m:
            if s[i] == t[j]:
                ans.append((i + 1, j + 1))
                i, j = i + 1, j + 1
            elif dp[i + 1][j] > dp[i][j + 1]:
                i += 1
            else:
                j += 1
        print(len(ans))
        for x, y in reversed(ans):
            print(x, y)
