n, m = map(int, input().split())
a = list(map(int, input().split()))
g = [[] for i in range(n)]
for i in range(m):
    x, y, w = map(int, input().split())
    x -= 1
    y -= 1
    g[x].append((y, w))
    g[y].append((x, w))
mx = a[0]
for i in range(n):
    if a[i] > 0:
        mx = max(mx, a[i])
mn = 1e9
for i in range(n):
    if a[i] < 0:
        mn = min(mn, a[i])
if mn * mx > 0:
    print(0)
else:
    print(mx - mn)
