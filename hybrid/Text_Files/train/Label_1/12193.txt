#include <bits/stdc++.h>
using namespace std;
long long n, k, a[1000000], d[1000000], e[1000000];
long long dp[1000000][3], ans = LLONG_MIN;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> d[i]; }
  for (int i = n; i >= 1; i--) { cin >> e[i]; }
  for (int i = 1; i <= n; i++) {
    dp[i][0] = max(dp[i - 1][0], dp[i - 1][2]);
    dp[i][1] = max(dp[i - 1][1], dp[i - 1][2] + a[i] - d[i]);
    dp[i][2] = max(dp[i - 1][0], dp[i - 1][1] + (k ? a[e[i]] - d[e[i]]
                                                       : LLONG_MIN));
    if (!k) dp[i][2] = LLONG_MIN;
  }
  for (int i = 1; i <= n; i++) { ans = max(ans, dp[i][1]); }
  cout << ans << endl;
  return 0;
}
