#include <algorithm>
#include <iostream>
#include <vector>

struct Ghost {
    long long x, v_x, v_y;
    bool operator<(const Ghost& o) const { return x < o.x; }
};

int main() {
    int n, a, b;
    std::cin >> n >> a >> b;
    std::vector<Ghost> ghosts(n);
    for (auto& g : ghosts) { std::cin >> g.x >> g.v_x >> g.v_y; }
    std::sort(ghosts.begin(), ghosts.end());
    long long gx = 0;
    for (int l = 0, r = 0; r < n;) {
        long long l_x = ghosts[l].x;
        long long r_x = ghosts[r].x;
        long long intersects_at =
            (r_x * a - l_x * a + b - b) / (double)(a * (ghosts[r].v_x - ghosts[l].v_x));
        while (r + 1 < n && intersects_at >= ghosts[r + 1].x) r++;
        gx += (r - l) * (r - l + 1) / 2;
        l = r + 1;
        r++;
    }
    std::cout << gx << '\n';
    return 0;
}
