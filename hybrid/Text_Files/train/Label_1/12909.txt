#include <bits/stdc++.h>
#define rep(i,lo,hi) for(int i=(lo);i<=(hi);++i)
#define rev(i,hi,lo) for(int i=(hi);i>=(lo);--i)
using namespace std;
const int N = 100005;
vector<int> G[N];
int n, m, q[N], qh, qt, ans;
bool vis[N];
int main() {
	ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
	cin >> n >> m;
	while (m--) {
		int a, b; cin >> a >> b;
		G[a].push_back(b);
		G[b].push_back(a);
	}
	rep(i,1,n) if (!vis[i]) {
		qh = qt = 0;
		q[qt++] = i;
		vis[i] = 1;
		int l = 1;
		while (qh != qt) {
			int k = q[qh++];
			for (int j : G[k]) if (!vis[j]) {
				vis[j] = 1;
				q[qt++] = j;
				l += 2;
			}
		}
		if (l % 2 && l > 3) --l;
		l = max(0, l - 2);
		ans += l / 2;
	}
	cout << ans << ' ' << 1LL * ans * (ans + 1) / 2 << '\n';
	return 0;
}
