#include<bits/stdc++.h>
using namespace std;
int n,p[505],G[505][505],w[505][505],vis[505][505];
void dfs(int x,int y,int id){
    if(x<1||x>n||y<1||y>n||G[x][y]!=id||vis[x][y]) return;
    vis[x][y]=1;
    dfs(x-1,y-1,id);
    dfs(x-1,y,id);
    dfs(x-1,y+1,id);
    dfs(x,y-1,id);
    dfs(x,y+1,id);
    dfs(x+1,y-1,id);
    dfs(x+1,y,id);
    dfs(x+1,y+1,id);
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>p[i];
        G[i][i]=p[i];
        w[i][i]=1;
    }
    int tot=0,cnt=0;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j){
            if(!vis[i][j]){
                dfs(i,j,++cnt);
            }
            if(G[i][j]) tot+=w[i][j];
        }
    }
    if(tot!=n){
        cout<<-1;
        return 0;
    }
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j){
            if(G[i][j]) cout<<w[i][j]<<" ";
        }
        cout<<"\n";
    }
}
