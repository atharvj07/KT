from collections import defaultdict

n, q = map(int, input().split())
a = list(map(int, input().split()))
g = defaultdict(list)
for i in range(1, n):
    if a[i - 1] & a[i]:
        g[i - 1].append(i)

for _ in range(q):
    x, y = map(int, input().split())
    if (not g[x - 1]) or (x not in g) or (y not in g) or (y not in g[x - 1]):
        print('Fou')
    else:
        print('Shi')
