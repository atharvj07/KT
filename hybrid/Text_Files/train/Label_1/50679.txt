import java.util.Scanner;

public class AdjacentNumbersMatrix {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();

            if (n % 2 == 0) {
                System.out.println(-1);
            } else {
                int[][] matrix = new int[n][n];
                int num = 1;

                // Fill the first row
                for (int j = 0; j < n; j++) {
                    matrix[0][j] = num++;
                }

                // Fill the remaining rows in a spiral manner
                int row = 1, col = n - 1;
                while (row < n && col >= 0) {
                    if (row % 2 == 1) {
                        // Move up and fill the column
                        for (int i = row; i >= 0; i--) {
                            matrix[i][col] = num++;
                        }
                        col--;
                    } else {
                        // Move down and fill the column
                        for (int i = row; i < n; i++) {
                            matrix[i][col] = num++;
                        }
                        col++;
                    }
                    row++;
                }

                // Print the matrix
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) {
                        System.out.print(matrix[i][j] + " ");
                    }
                    System.out.println();
                }
            }
        }
    }
}
