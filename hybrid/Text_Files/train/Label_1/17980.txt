#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,a,b;
	cin >> n;
	map<int,int> mp;
	for(int i = 0; i < n; i++){
		cin >> a >> b;
		mp[b] = a;
	}
	int mx = mp.begin()->first, mn = mp.begin()->first;
	for(auto i : mp){
		if(i.first < mx) mx = i.first;
		if(i.first > mn) mn = i.first;
	}
	cout << (mp[mx] < mp[mn] ? "Happy Alex" : "Poor Alex") << endl;
}
