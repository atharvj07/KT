from sys import stdin

for _ in range(int(stdin.readline())):
    n = int(stdin.readline())
    if n % 2 and n != 1:
        print(-1)
    elif n % 2 == 0:
        a = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(i % 2, n, 2):
                a[i][j] = j * n + i + 1
        for i in range(n):
            for j in range(i % 2 + 1, n, 2):
                a[i][j] = (j + 1) * n - i
        for l in a:
            print(*l)
    else:
        print(1)
