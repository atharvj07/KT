import java.util.*;
import java.io.*;

public class Valerian {
    static class Edge {
        int to, weight;

        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static int n;
    static ArrayList<Edge>[] graph;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int w = Integer.parseInt(st.nextToken());
            graph[u].add(new Edge(v, w));
            graph[v].add(new Edge(u, w));
        }

        int minDist = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            minDist = Math.min(minDist, dfs(i, -1, 0));
        }
        System.out.println(minDist);
    }

    static int dfs(int node, int parent, int dist) {
        int minSubtreeDist = Integer.MAX_VALUE;
        for (Edge edge : graph[node]) {
            if (edge.to != parent) {
                minSubtreeDist = Math.min(minSubtreeDist, dfs(edge.to, node, dist + edge.weight));
            }
        }
        return minSubtreeDist == Integer.MAX_VALUE ? dist : minSubtreeDist + dist;
    }
}
