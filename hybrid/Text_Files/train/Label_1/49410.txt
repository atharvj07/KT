import java.util.*;
public class Main {
    static int[] arr;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            if (n == 0 && k == 0) break;
            arr = new int[n];
            for (int i = 0; i < n; i++)
                arr[i] = sc.nextInt();
            int score = findMaxScore(n, k);
            System.out.println(score == -1 ? "NO GAME" : score);
        }
    }
    static int findMaxScore(int n, int k) {
        int ans = -1;
        for (int i = 0; i + k - 1 < n; i++) {
            int prod = 1;
            for (int j = i; j < i + k; j++)
                prod *= arr[j];
            ans = Math.max(ans, prod);
        }
        return ans;
    }
}
