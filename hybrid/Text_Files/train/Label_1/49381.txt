import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int min = 1, max = n;
            for (int j = 0; j < n - 1; j++) {
                System.out.println("? " + min + " " + max);
                System.out.flush();
                char c = sc.next().charAt(0);
                if (c == '=') break;
                else if (c == '<') min++;
                else max--;
            }
            System.out.println("! " + min + " " + max);
            System.out.flush();
        }
    }
}
