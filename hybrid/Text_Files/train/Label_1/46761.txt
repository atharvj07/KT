import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();

        TreeMap<Integer, Integer> multiset = new TreeMap<>();
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            multiset.put(a, multiset.getOrDefault(a, 0) + 1);
        }

        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            if (k > 0) {
                multiset.put(k, multiset.getOrDefault(k, 0) + 1);
            } else {
                k = -k;
                int count = 0;
                for (Map.Entry<Integer, Integer> entry : multiset.entrySet()) {
                    count += entry.getValue();
                    if (count >= k) {
                        if (entry.getValue() > 1) {
                            multiset.put(entry.getKey(), entry.getValue() - 1);
                        } else {
                            multiset.remove(entry.getKey());
                        }
                        break;
                    }
                }
            }
        }

        if (multiset.isEmpty()) {
            System.out.println(0);
        } else {
            System.out.println(multiset.firstKey());
        }
    }
}
