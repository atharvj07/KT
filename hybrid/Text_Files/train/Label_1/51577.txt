import java.util.*;
public class Main {
    public static void main(String[] args) {
        var sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] lis = new long[n], ris = new long[n];
        long[][] abcs = new long[m][3];
        long ans = 0, a = 0, b = 0;
        for (int i = 0; i < n; i++) {
            lis[i] = sc.nextInt();
            ris[i] = sc.nextInt();
        }
        for (long[] abc : abcs) {
            abc[0] = sc.nextInt();
            abc[1] = sc.nextInt();
            abc[2] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            for (long[] abc : abcs) {
                long l = Math.max(lis[i], abc[0]), r = Math.min(ris[i], abc[1]);
                if (r < l) continue;
                long t = (r - l) * abc[2];
                if (t > ans) ans = t;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                long l = Math.max(lis[i], abcs[j][0]), r = Math.min(ris[i], abcs[j][1]);
                if (r < l) continue;
                long t = (r - l) * abcs[j][2];
                if (t == ans) a = i;
                if (t == ans) b = j;
            }
        }
        if (ans != 0) System.out.println(ans + "\n" + (a + 1) + " " + (b + 1));
        else System.out.println(ans);
    }
}
