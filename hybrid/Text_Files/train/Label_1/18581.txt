#include<bits/stdc++.h>
using namespace std;
int n,x[100010],y[100010];
bool cmp1(int a,int b){
    return x[a]<x[b];
}
bool cmp2(int a,int b){
    return y[a]<y[b];
}
int main(){
    cin>>n;
    for(int i=1;i<=n;++i) cin>>x[i]>>y[i];
    int z[100010];
    for(int i=1;i<=n;++i) cin>>z[i];
    sort(z+1,z+1+n,cmp1);
    sort(z+1,z+1+n,cmp2);
    for(int i=1;i<=n;++i) if(z[i]!=i){
        cout<<"NO"<<endl;
        return 0;
    }
    cout<<"YES"<<endl;
    return 0;
}
