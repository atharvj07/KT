#include <bits/stdc++.h>
using namespace std;
const int N = 105;
int n, m, a[N][3];
vector<int> ans;
int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= m; i++) cin >> a[i][0] >> a[i][1] >> a[i][2];
    for (int t = 0; t <= n + m; t++) {
        int cnt = 0;
        vector<int> sol;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (a[j][0] != a[j][1] && a[j][0] <= i && i <= a[j][1]) {
                    if (i == a[j][0] || i == a[j][1])
                        sol.push_back(j + N);
                    else
                        sol.push_back(j);
                    cnt++;
                    break;
                }
                if (a[j][0] == a[j][1] && a[j][0] == i) {
                    sol.push_back(j + N);
                    cnt++;
                    break;
                }
            }
            if (cnt == 0) sol.push_back(0);
        }
        if (cnt == m) {
            for (auto i : sol) ans.push_back(i);
            cout << ans.size() << "\n";
            for (auto i : ans) {
                if (i >= N)
                    cout << i - N << " ";
                else
                    cout << i << " ";
            }
            return 0;
        }
    }
    cout << -1 << "\n";
}
