#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
    ll n, b, k, x;
    cin >> n >> b >> k >> x;
    vector<ll> a(n);
    unordered_map<ll, ll> cnt;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    vector<ll> dp(x, 0);
    dp[0] = 1;
    for (auto &p : cnt) {
        vector<ll> dp_new(x, 0);
        for (ll i = 0; i < x; i++) {
            dp_new[(i + p.second % x) % x] = (dp_new[(i + p.second % x) % x] + dp[i]) % mod;
        }
        dp = dp_new;
    }
    cout << dp[k] << "\n";
    return 0;
}
