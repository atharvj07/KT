import sys
input = sys.stdin.readline

k = int(input())
A = []
sum_a = [0]
for _ in range(k):
    n = int(input())
    tmp = [0]
    tmp.extend([int(x) for x in input().split()])
    sum_a.append(sum(tmp))
    A.append(tmp)
sum_a = sum_a[1:]

ans = "NO"
for i in range(k):
    for j in range(k):
        if i == j:
            continue
        for x in range(1, 1 + A[i][0]):
            for y in range(1, 1 + A[j][0]):
                if sum_a[i] - A[i][x] == sum_a[j] - A[j][y]:
                    ans = "YES"
                    print(ans)
                    print(i+1, x)
                    print(j+1, y)
                    exit(0)
print(ans)
