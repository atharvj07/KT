#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    map<char, int> m;
    for (char c : s) { m[c]++; }
    int mx = 0;
    for (auto p : m) { mx = max(mx, p.second); }
    if (mx <= n / 2) { cout << "Yes"; } else { cout << "No"; }
    return 0;
}
