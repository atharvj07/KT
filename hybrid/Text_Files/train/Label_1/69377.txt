n = int(input().strip())
balances = list(map(int, input().strip().split()))
q = int(input().strip())

last_payout = -1
payout_updates = [0] * n

for _ in range(q):
    event = list(map(int, input().strip().split()))
    if event[0] == 1:
        p, x = event[1]-1, event[2]
        balances[p] = x
        payout_updates[p] = last_payout
    else:
        x = event[1]
        last_payout = x

for i in range(n):
    if payout_updates[i] != last_payout and balances[i] < last_payout:
        balances[i] = last_payout

print(' '.join(map(str, balances)))
