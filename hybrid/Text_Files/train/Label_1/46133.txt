import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[][] s = new int[m][2];
        for (int i = 0; i < m; i++) {
            s[i][0] = sc.nextInt() - 1;
            s[i][1] = sc.nextInt() - 1;
        }
        for (int i = 0; i < m; i++) {
            int x = s[i][0], y = s[i][1];
            a[x] -= y;
            if (x > 0) {
                a[x - 1] += y / 2;
            }
            if (x < n - 1) {
                a[x + 1] += y / 2 + y % 2;
            }
        }
        for (int x : a) {
            System.out.println(x);
        }
    }
}
