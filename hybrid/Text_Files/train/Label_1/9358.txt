#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  if (n * m % 2 == 1) {
    cout << "NO";
  } else {
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        a[i][j] = i * m + j + 1;
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j + 1 < m; j += 2) {
        swap(a[i][j], a[i][j + 1]);
      }
    }
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cout << a[i][j] << ' ';
      }
      cout << endl;
    }
  }
}
