#include <bits/stdc++.h>
typedef long long ll;

const ll mxN = 5e5 + 5;
ll n, k, d;
ll a[mxN];

int main() {
    std::cin >> n >> k >> d;
    for (ll i = 0; i < n; i++) std::cin >> a[i];
    std::sort(a, a + n);
    for (ll i = 0, j = 0; i < n; i = j) {
        while (j < n && a[j] - a[i] <= d) j++;
        if (j - i < k) { std::cout << "NO" << std::endl; return 0; }
    }
    std::cout << "YES" << std::endl;
    return 0;
}
