#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    int p[n];
    for (int i = 0; i < n; i++) p[i] = i;
    for (int i = 0; i < m - 1; i++) {
        int j = n - i - 2;
        if (p[j] < p[j + 1]) swap(p[j], p[j + 1]);
    }
    for (int i = 0; i < n; i++) cout << p[i] + 1 << ' ';
    cout << '\n';
    return 0;
}
