from collections import deque

n, m = map(int, input().split())
grid = [['0'] * (m+2) for _ in range(n+2)]

dx = [0, 0, 1, -1, 2, -2]
dy = [1, -1, 0, 0, 2, -2]

def bfs(x: int, y: int) -> bool:
    q = deque([(x, y, (0,0))])
    
    while q:
        x, y, prev = q.popleft()
        
        if x == n and y == m:
            return True
        
        for i in range(6):
            nx, ny = x + dx[i], y + dy[i]
            if grid[nx][ny] == '0' and prev != (dx[i], dy[i]):
                grid[nx][ny] = grid[x][y] + 1
                q.append((nx, ny, (dx[i], dy[i])))
    
    return False

if bfs(1, 1):
    for i in range(1, n+1):
        for j in range(1, m+1):
            print(i, j)
else:
    print(-1)
