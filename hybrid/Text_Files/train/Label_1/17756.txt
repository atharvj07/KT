#include <vector>
#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    v[i] = {l, r};
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (v[i].first != v[i].second) ans++;
  }
  cout << ans << endl;
  return 0;
}
