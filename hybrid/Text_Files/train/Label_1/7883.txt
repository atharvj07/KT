#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, M;
  cin >> n >> M;
  vector<int> t(n);
  for (int i = 0; i < n; i++) cin >> t[i];
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    int sum = 0;
    for (int j = i; j < n; j++) {
      sum += t[j];
      if (sum > M) { ans[i] = j - i; break; }
    }
  }
  for (int i = 0; i < n; i++) cout << ans[i] << ' ';
  cout << endl;
  return 0;
}
