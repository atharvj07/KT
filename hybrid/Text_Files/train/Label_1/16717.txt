const ll mod = 998244353;

ll binpow(ll x, ll p) {
    if (p == 0) return 0;
    if (p == 1) return x;
    ll z = binpow(x, p / 2);
    z = (z * z) % mod;
    if (p & 1) return (z * x) % mod;
    else return z;
}
ll modinv(ll x) {
    return binpow(x, mod - 2);
}
ll fact[2000001], ifact[2000001];
ll nCr(ll n, ll r) {
    if (r > n || r < 0) return 0;
    ll aa = fact[n];
    ll bb = ifact[n - r];
    ll cc = ifact[r];
    return (aa * bb) % mod * cc % mod;
}
int main() {
    int n;
    cin >> n;
    fact[0] = 1;
    for (int i = 1; i <= 2000000; i++) fact[i] = fact[i - 1] * i % mod;
    ifact[0] = 1;
    for (int i = 1; i <= 2000000; i++) ifact[i] = ifact[i - 1] * modinv(i) % mod;
    cout << nCr(n * n, n) * n % mod << '\n';
    return 0;
}
