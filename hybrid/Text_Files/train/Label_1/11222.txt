#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mxN = 1e6 + 5;
int32_t main() {
    int n;
    cin >> n;
    vector<int> v(n);
    vector<bool> isPrime(1e6 + 5, 1);
    for (int i = 0; i < n; i++) cin >> v[i];
    isPrime[0] = isPrime[1] = 0;
    for (int p = 2; p * p <= 1e6 + 5; p++) {
        if (isPrime[p]) {
            for (int i = p; i * p <= 1e6 + 5; i++) {
                isPrime[i * p] = 0;
            }
        }
    }
    vector<vector<int>> adj(1e6 + 5);
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (isPrime[v[i] + v[j]]) {
                adj[v[i]].push_back(v[j]);
                adj[v[j]].push_back(v[i]);
            }
        }
    }
    vector<int> vis(1e6 + 5), color(1e6 + 5), sz(1e6 + 5), res;
    int mx = 0;
    function<void(int, int)> dfs = [&](int node, int clr) {
        vis[node] = 1;
        color[node] = clr;
        sz[clr]++;
        for (int nbr : adj[node]) {
            if (vis[nbr] && color[nbr] == clr)
                continue;
            if (!vis[nbr]) {
                dfs(nbr, clr ^ 1);
            }
        }
    };
    for (int i = 0; i < n; i++) {
        if (!vis[v[i]]) {
            dfs(v[i], 0);
            mx = max(mx, sz[0] + sz[1]);
        }
    }
    cout << mx << "\n";
    set<int> ans;
    for (int i = 0; i < n; i++) {
        if (sz[color[v[i]]] + sz[color[v[i]] ^ 1] == mx) {
            ans.insert(v[i]);
        }
    }
    for (int x : ans) cout << x << " ";
    cout << "\n";
}
