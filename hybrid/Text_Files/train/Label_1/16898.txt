#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < n; i++)
const int MAX_N = 1005;
int n;
int pos[MAX_N][2];
int main() {
    cin >> n;
    int m = 1;
    while(m * m < n) m++;
    if(m * m == n) {
        cout << m << "\n";
        int cnt = 1;
        for(int i = 1; i <= m; i += 2) {
            for(int j = 1; j <= m; j += 2) {
                pos[cnt][0] = i;
                pos[cnt][1] = j;
                cnt++;
            }
        }
        for(int i = 1; i <= m; i += 2) {
            for(int j = 2; j <= m; j += 2) {
                pos[cnt][0] = i;
                pos[cnt][1] = j;
                cnt++;
            }
        }
    }
    else {
        m++;
        cout << m << "\n";
        int cnt = 1;
        for(int i = 1; i <= m; i += 2) {
            for(int j = 1; j <= m; j += 2) {
                pos[cnt][0] = i;
                pos[cnt][1] = j;
                cnt++;
            }
        }
        for(int i = 2; i <= m; i += 2) {
            for(int j = 2; j <= m; j += 2) {
                pos[cnt][0] = i;
                pos[cnt][1] = j;
                cnt++;
            }
        }
    }
    for(int i = 1; i <= n; i++) {
        cout << pos[i][0] << " " << pos[i][1] << "\n";
    }
    return 0;
}
