import sys
input = sys.stdin.readline
n, m, a = map(int, input().split())
boys = [int(x) for x in input().split()]
bikes = [int(x) for x in input().split()]
boys.sort(reverse=True)
bikes.sort()
r = 0
s = 0
for bike in bikes:
    if a >= bike:
        if r < n:
            a -= bike
            s += max(boys[r] - bike, 0)
            r += 1
    else:
        break
print(r, s)
