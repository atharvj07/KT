#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
    int t;
    cin >> t;

    for (int i = 0; i < t; ++i) {
        int n;
        cin >> n;

        map<string, int> users;
        map<string, string> messages;
        vector<string> users_list;
        for (int j = 0; j < n; ++j) {
            string user;
            cin >> user;
            users[user] = 0;
            users_list.push_back(user);
        }

        int m;
        cin >> m;
        for (int j = 0; j < m; ++j) {
            string user, message;
            cin >> user >> message;
            if (user[0] == '?') {
                messages[user] = message;
            } else {
                users[user]++;
                for (int k = 0; k < n; ++k) {
                    if (users[users_list[k]] == 0 && message.find(users_list[k]) == string::npos) {
                        messages[user] = message;
                        users[user]--;
                        users[users_list[k]]++;
                        break;
                    }
                }
            }
        }

        bool impossible = false;
        for (const auto& [user, message] : messages) {
            if (user[0] == '?' || users[user] == 0) {
                impossible = true;
                break;
            }
            cout << user << ":" << message << endl;
        }

        if (impossible) {
            cout << "Impossible" << endl;
        }
    }

    return 0;
}
