#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, tmp;
    cin >> n;
    vector<int> v(n);
    for (auto &i : v) {
        cin >> i;
    }
    sort(v.begin(), v.end());
    vector<int> ans(n / 2 + 1);
    for (int i = 0, j = 0; i < v.size(); i++) {
        while (j < i && v[i] <= v[j]) { j++; }
        ans[i - j] += (i - j) * v[i];
        for (int k = j; k < i; k++) {
            ans[i - k] += v[k];
        }
    }
    for (auto &i : ans) {
        cout << i << ' ';
    }
    return 0;
}
