import sys

sys.setrecursionlimit(2000)


def find_beauty(a, n, m):
    if n == 0:
        return a[0][0]
    max_num = 0
    for i in range(m):
        for j in range(m):
            max_num = max(max_num, a[i][j])
    ans = 0
    ans += max_num
    a[int(max_num / m)][max_num % m] = 0
    for i in range(m):
        for j in range(m):
            if a[i][j] != 0:
                b = [[0] * (m // 2) for _ in range(m // 2)]
                k = 0
                for p in range(i, i + m // 2):
                    l = 0
                    for q in range(j, j + m // 2):
                        b[k][l] = a[p][q]
                        l += 1
                    k += 1
                ans += find_beauty(b, n - 1, m // 2)
    return ans


n = int(input())
a = []
for i in range(n ** 2):
    a.append([int(x) for x in input().split()])
print(find_beauty(a, n, n))
