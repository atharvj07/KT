#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=20;
ll n,m,k;
map<ll,ll>freq;
int main(){
    cin>>n>>m>>k;
    for(ll i=0;i<m+1;++i){
        ll x;
        cin>>x;
        freq[x]++;
    }
    ll sm=0;
    for(auto it:freq){
        ll x=it.first;
        ll lz=0;
        for(ll i=0;i<n;++i){
            if((x&(1ll<<i)))
                lz++;
        }
        if(lz<=k)
            sm+=it.second-1;
    }
    cout<<sm<<endl;
}
