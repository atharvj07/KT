#include <iostream>
#include <stack>
#include <vector>
#include <algorithm>
#define endl "\n"
using namespace std;
int b[200005], a[200005],k,n,c=0;
stack<int> s;
bool valid=1;
int main() {
    cin>>n>>k;
    for(int i=0; i<k; i++){
        cin>>a[i];
        b[i]=a[i];
    }
    for(int i=k;i<n;i++){
        if(c<k){
            s.push(a[c]);
            c++;
        }
        int t=s.top();
        while(!s.empty()&&t>b[i]&&c<n){
            s.pop();
            b[c]=t;
            c++;
            if(!s.empty()) t=s.top();
        }
        if(t>b[i]){valid=0; break;}
        b[i]=a[c];
        c++;
    }
    while(!s.empty()){
        b[c]=s.top();
        c++;
        s.pop();
    }
    if(valid){
        for(int i=0;i<n;i++){
            cout<<b[i]<<" ";
        }
    }else cout<<-1<<endl;
}
