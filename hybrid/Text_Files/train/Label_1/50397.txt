import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static long mod = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        long[] dp = new long[n + 1];
        dfs(1, 0, adj, dp, k);
        System.out.println(dp[1]);
    }

    static void dfs(int u, int p, List<List<Integer>> adj, long[] dp, int k) {
        dp[u] = 1;
        for (int v : adj.get(u)) {
            if (v == p) continue;
            dfs(v, u, adj, dp, k);
            if (k == 0) dp[u] = (dp[u] * (dp[v] + 1)) % mod;
            else dp[u] = (dp[u] * (dp[v] + dp[v] * (k - 1))) % mod;
        }
    }
}
