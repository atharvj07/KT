import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int k = sc.nextInt();
    int[][] arr = new int[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        arr[i][j] = sc.nextInt();
      }
    }
    int max = 0;
    int mx = 0;
    int my = 0;
    for (int i = k - 1; i < n - k + 1; i++) {
      for (int j = k - 1; j < m - k + 1; j++) {
        int sum = 0;
        for (int x = -k + 1; x <= k - 1; x++) {
          for (int y = -k + 1; y <= k - 1; y++) {
            sum = Math.max(sum, arr[i + x][j + y]);
          }
        }
        if (sum > max) {
          max = sum;
          mx = i;
          my = j;
        }
      }
    }
    System.out.println((mx + 1) + " " + (my + 1));
  }
}
