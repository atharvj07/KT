import sys

n, m, k = list(map(int, input().split()))
a = list()
for _ in range(n):
    a.append(list(map(int, input().split())))

b = list()
for i in range(m):
    b.append(sum([x[i] for x in a]))

ans = [0 for i in range(m)]
for i in range(len(b)):
    ans[i] = min(k, b[i] // 2)
    k -= ans[i]
    b[i] -= ans[i] * 2

sort = sorted([(x, i) for i, x in enumerate(b)], reverse=True)
for x, i in sort:
    if k < x:
        break
    ans[i] += min(x, k)
    k -= min(x, k)

print(*ans)
