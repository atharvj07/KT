#include <iostream>
#include <cstring>
using namespace std;
int main() {
  int n; cin >> n;
  char s[200005];
  s[0] = s[1] = 'a';
  for (int i = 2; i < n; i++) {
    if (s[i - 1] == s[i - 2])
      s[i] = (s[i - 1] == 'a' ? 'b' : 'c');
    else
      s[i] = s[i - 1];
  }
  cout << s << "\n";
  return 0;
}
