#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        string s;
        cin>>s;
        bool f=0;
        for(int i=0; i<s.size(); i++)
        {
            if(i==(s.size()-1) && s[i]=='0')
            {
                f=0;
                break;
            }
            if(s[i]=='0' || (s[i]-'0')>=(s.size()-i))
            {
                f=0;
                break;
            }
            else if((s[i]-'0')<(s.size()-i))
            {
                f=1;
            }
        }
        if(f==0)
        {
            for(int i=s.size()-1; i>=0; i--)
            {
                if(s[i]!='9')
                {
                    s[i]=s[i]+1;
                    for(int j=i+1; j<s.size(); j++)
                    {
                        s[j]='0';
                    }
                    break;
                }
            }
        }
        cout<<s<<endl;
    }
}
