#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int f = 0;
    for (int i = 1; i < n; i++) {
      if (s[i] == s[i - 1] && s[i] != '?') {
        f = 1;
        break;
      }
    }
    if (f) {
      cout << -1 << endl;
      continue;
    }
    for (int i = 0; i < n; i++) {
      if (s[i] == '?') {
        if (i == 0) {
          if (n == 1)
            s[i] = 'a';
          else if (s[i + 1] != 'a')
            s[i] = 'a';
          else
            s[i] = 'b';
        } else if (i == n - 1) {
          if (s[i - 1] != 'c')
            s[i] = 'c';
          else
            s[i] = 'b';
        } else {
          if (s[i - 1] != s[i + 1] && s[i - 1] != '?' && s[i + 1] != '?')
            s[i] = s[i - 1];
          else if (s[i - 1] == s[i + 1]) {
            if (s[i - 1] == 'a')
              s[i] = 'b';
            else if (s[i - 1] == 'b')
              s[i] = 'c';
            else
              s[i] = 'a';
          } else if (s[i - 1] == '?') {
            if (s[i + 1] != 'a')
              s[i] = 'a';
            else
              s[i] = 'b';
          } else {
            if (s[i - 1] != 'c')
              s[i] = 'c';
            else
              s[i] = 'b';
          }
        }
      }
    }
    cout << s << endl;
  }
  return 0;
}
