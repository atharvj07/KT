import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextInt(), t = scanner.nextInt();
        String s = scanner.next();
        ArrayList<Integer> h = new ArrayList<>(), ss = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'H') h.add(i);
            else if (s.charAt(i) == 'S') ss.add(i);
        }
        if (h.size() * 2 - 1 > t) {
            System.out.println(-1);
            return ;
        }
        long cur = 0;
        for (int i = 0; i < ss.size(); i++) {
            cur += Math.abs(ss.get(i) - h.get(0));
        }
        t -= cur;
        if (t < 0) {
            System.out.println(-1);
            return ;
        }
        System.out.println(Math.max(0, (h.size() - ss.size()) * 2 - t));
    }
}
