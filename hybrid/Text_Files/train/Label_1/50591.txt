import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), b = in.nextInt();
        int ans = 0, cur = 0, x = a, y = b;
        StringBuilder sb = new StringBuilder();
        while (a > 0 || b > 0) {
            if (cur == 1) {
                if (a > 0) {
                    sb.append('o');
                    ans += x * x;
                    a--;
                    x++;
                } else if (b > 0) {
                    sb.append('x');
                    ans -= y * y;
                    b--;
                    y++;
                }
            } else {
                if (b > 0) {
                    sb.append('x');
                    ans -= y * y;
                    b--;
                    y++;
                } else if (a > 0) {
                    sb.append('o');
                    ans += x * x;
                    a--;
                    x++;
                }
            }
            cur = (cur + 1) % 2;
        }
        System.out.println(ans);
        System.out.println(sb);
    }
}
