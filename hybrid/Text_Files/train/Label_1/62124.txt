def count_classy(L, R):
    # Function to count the number of digits in a number
    def count_digits(n):
        return sum(1 for digit in str(n) if digit != '0')

    # Function to generate all classy numbers within a range
    def generate_classy(max_len):
        classy_nums = []
        # Generate numbers with up to 3 non-zero digits
        for length in range(1, max_len + 1):
            for i in range(1, 10):
                generate_classy_rec(str(i), length, 1, classy_nums)
        return sorted(classy_nums)
    
    # Recursive function to generate classy numbers of a certain length
    def generate_classy_rec(num_str, max_len, non_zero_count, classy_nums):
        if len(num_str) == max_len:
            classy_nums.append(int(num_str))
        else:
            # Add a zero and continue
            generate_classy_rec(num_str + '0', max_len, non_zero_count, classy_nums)
            # Only add a non-zero digit if we have less than 3 non-zero digits
            if non_zero_count < 3:
                for i in range(1, 10):
                    generate_classy_rec(num_str + str(i), max_len, non_zero_count + 1, classy_nums)
    
    # Find the upper limit for classy number generation based on R
    max_len = len(str(R))
    classy_numbers = generate_classy(max_len)
    
    # Count the classy numbers within the range [L, R]
    count = 0
    for num in classy_numbers:
        if L <= num <= R:
            count += 1
    return count

# Read the number of test cases
T = int(input().strip())

# Iterate over each test case and solve
for _ in range(T):
    L, R = map(int, input().strip().split())
    print(count_classy(L, R))
