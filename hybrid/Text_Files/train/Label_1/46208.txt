import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        for (int i = 0; i < t; i++) {
            int n = input.nextInt();
            int m = input.nextInt();
            int[][] a = new int[n][m];
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    a[j][k] = input.nextInt();
                }
            }
            boolean canWin = dfs(a, 0, 0, 1);
            System.out.println(canWin ? "Ashish" : "Vivek");
        }
    }

    private static boolean dfs(int[][] a, int x, int y, int turn) {
        if (x < 0 || x >= a.length || y < 0 || y >= a[0].length || a[x][y] != 0) {
            return false;
        }
        a[x][y] = turn;
        boolean canWin = false;
        if (turn == 1) {
            canWin = dfs(a, x, y + 1, turn) || dfs(a, x, y - 1, turn) || dfs(a, x + 1, y, turn) || dfs(a, x - 1, y, turn);
        } else {
            canWin = !dfs(a, x, y + 1, turn) && !dfs(a, x, y - 1, turn) && !dfs(a, x + 1, y, turn) && !dfs(a, x - 1, y, turn);
        }
        a[x][y] = 0;
        return canWin;
    }
}
