#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ll n, m, sx, sy, fx, fy;
    cin >> n >> m >> sx >> sy >> fx >> fy;
    vector<pair<ll, ll>> loc(m);
    for (auto &p : loc) cin >> p.first >> p.second;

    ll ans = -1;
    for (auto &l : loc) {
        if (l.first == sx || l.second == sy) ans = 1;
        if ((abs(l.first - fx) + abs(l.second - fy)) % 2 == 0) ans = max(ans, (abs(l.first - fx) + abs(l.second - fy)) / 2);
    }

    if (sx == fx && sy == fy) ans = 0;
    if ((abs(sx - fx) + abs(sy - fy)) % 2 == 1) ans = max(ans, (abs(sx - fx) + abs(sy - fy)) / 2 + 1);
    if (ans != -1) ans += 1;

    cout << ans << endl;
}
