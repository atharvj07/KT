n, m, k = map(int, input().split())
a = []
b = [int(i) for i in input().split()]
for i in range(m):
    a.append(int(input()))
    b[m] |= a[-1]
ans = 0
for i in a:
    x = b[m] ^ i
    c = 0
    for j in range(n):
        if ((x & (1 << j)) > 0):
            c += 1
    if c <= k:
        ans += 1
print(ans)
