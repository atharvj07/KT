#include <iostream>
using namespace std;
int main(){
    long long n,a,b,c;
    cin>>n;
    a=1;
    if(n==3){
        a=1;
        b=1;
        c=1;
    }
    else if(n%3==0){
        a=n/3;
        b=n/3;
        c=n/3;
    }
    else if(n%3==1){
        a=1;
        b=n/3;
        c=(n/3)+1;
    }
    else{
        a=n/3;
        b=(n/3)+1;
        c=n/3;
    }
    cout<<a<<' '<<b<<' '<<c<<endl;
}
