#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> b(1 << 20);
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      b[a[j] |= a[i]]++;
    }
  }
  int cnt = 0;
  for (int i = 0; i < (1 << 20); i++) {
    if (b[i]) cnt++;
  }
  cout << cnt << endl;
  return 0;
}
