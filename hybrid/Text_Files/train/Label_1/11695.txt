#include <cstdio>
 
const int MAXN = 100000 + 5;
 
int n, k, a[MAXN], p[MAXN], ans;
 
int findp(int x) { return x == p[x] ? x : p[x] = findp(p[x]); }
 
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 1; i <= n; i++) p[i] = i;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    int fx = findp(i), fy = findp(a[i]);
    if (fx != fy) ans++, p[fx] = fy;
  }
  k = k * ans;
  printf("%d\n", (k / 2) * (k / 2 + 1) + (k % 2) * ((k + 1) / 2) * ((k + 1) / 2));
  return 0;
}
