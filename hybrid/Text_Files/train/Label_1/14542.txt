#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> v(n);
  for (char& c : v) cin >> c;
  unordered_map<int, int> m;
  m[0] = 1;
  int x = 0, y = 0, cnt = 0;
  for (char c : v) {
    switch (c) {
      case 'U': y--; break;
      case 'D': y++; break;
      case 'L': x--; break;
      case 'R': x++; break;
    }
    cnt += m[x - y];
    m[x - y]++;
  }
  cout << cnt << endl;
  return 0;
}
