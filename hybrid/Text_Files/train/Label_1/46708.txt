import java.util.*;

public class Main {
    static long[] solve() {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        long[] res = new long[t];
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            long ans = 0;
            int l = 0, r = n - 1;
            while (l < r) {
                if (a[l] + a[r] == x) {
                    ans++;
                    l++;
                    r--;
                } else if (a[l] + a[r] < x) {
                    l++;
                } else {
                    r--;
                }
            }
            res[i] = ans;
        }
        return res;
    }

    public static void main(String[] args) {
        long[] result = solve();
        for (long r : result) {
            System.out.println(r);
        }
    }
}
