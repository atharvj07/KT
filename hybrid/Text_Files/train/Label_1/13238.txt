#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main()
{
    int t;
    cin >> t;
    while (t--) {
        ll x;
        cin >> x;
        if ((x - 1) % 2 == 0 && x % 2 == 0)
            cout << x << " " << 2 << endl;
        else if (x % 3 == 0)
            cout << x << " " << 3 << endl;
        else if (x % 2 == 0)
            cout << x << " " << 2 << endl;
        else
            cout << -1 << endl;
    }
    return 0;
}
