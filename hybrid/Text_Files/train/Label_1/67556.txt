import sys
input = sys.stdin.readline

n = int(input())
x = [0] * n
y = [0] * n
for i in range(n):
    xi, yi, xj, yj = map(int, input().split())
    x[i] = [xi, xj]
    y[i] = [yi, yj]
x.sort()
y.sort()
mx = -1
my = -1
cnt = 0
for i in range(n - 1):
    if x[i][1] == x[i + 1][0]:
        if y[i][1] == y[i + 1][0]:
            cnt += 1
            mx = x[i][0]
            my = y[i][0]
print(mx, my)
