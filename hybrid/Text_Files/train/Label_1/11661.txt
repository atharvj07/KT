#include <bits/stdc++.h>
using namespace std;
int mod = 1e9 + 7;
int main() {
  long long a, b, k, t;
  cin >> a >> b >> k >> t;
  long long ans = 1, two = 1;
  while (t--) two = (two << 1) % mod;
  for (int i = 1; i <= k; i++) {
    if (a + i < b - i) continue;
    ans = (ans * two) % mod;
  }
  cout << ans << endl;
  return 0;
}
