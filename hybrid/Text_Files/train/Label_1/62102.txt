# Read input
s = input().strip()
m = int(input())
queries = [tuple(map(int, input().split())) for _ in range(m)]

# Precompute the answers for query efficiency
precomputed = [0] * len(s)
for i in range(1, len(s)):
    precomputed[i] = precomputed[i-1] + (s[i] == s[i-1])

# Answer the queries
for l, r in queries:
    print(precomputed[r-1] - precomputed[l-1])
