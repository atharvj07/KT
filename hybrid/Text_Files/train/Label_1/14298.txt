#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> p(n), q(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> q[i];
    }
    vector<int> perm(1, 0);
    for (int i = 1; i < n; i++) {
        int number = i;
        for (int j = 0; j < i; j++) {
            if (p[j] == number) {
                number = q[j];
                break;
            }
        }
        perm.push_back(number);
    }
    for (int x : perm) {
        cout << x << ' ';
    }
    cout << endl;
    return 0;
}
