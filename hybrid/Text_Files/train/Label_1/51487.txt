import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static int MAX_N = 40001;
    static int[] a = new int[MAX_N], d = new int[MAX_N], s = new int[MAX_N], h = new int[MAX_N];
    static long[][] dp = new long[MAX_N][2];

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        for (int i = 1; i <= n; i++) {
            String[] l = in.readLine().split(" ");
            h[i] = Integer.parseInt(l[0]);
            a[i] = Integer.parseInt(l[1]);
            d[i] = Integer.parseInt(l[2]);
            s[i] = Integer.parseInt(l[3]);
        }
        Arrays.sort(s, 1, n + 1);
        boolean flag = true;
        dp[n + 1][0] = 1;
        dp[n + 1][1] = 0;
        for (int i = n; i >= 1; i--) {
            for (int j = 0; j <= 1; j++) {
                if (dp[i + 1][j] == 1) {
                    dp[i][j] = 0;
                    continue;
                }
                long t = (s[i] < s[n] || s[i] == s[n] && j == 0) ? dp[i + 1][1] : dp[i + 1][0];
                long now = a[n] - d[i];
                if (now > h[i]) {
                    dp[i][j] = 1;
                } else if (now > 0) {
                    t += now;
                    dp[i][j] = (t < h[i]) ? t : 1;
                } else {
                    dp[i][j] = t;
                }
            }
        }
        if (dp[1][1] > 0) {
            System.out.println(dp[1][1]);
        } else {
            System.out.println(-1);
        }
    }
}
