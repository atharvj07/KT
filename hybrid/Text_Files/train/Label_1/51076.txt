import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] d = new int[n + 1];
        Arrays.fill(d, -1);
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int l = sc.nextInt();
            if (d[a] == -1 || d[a] > d[b] + l) d[a] = d[b] + l;
            if (d[b] == -1 || d[b] > d[a] + l) d[b] = d[a] + l;
        }
        int[] s = new int[k];
        for (int i = 0; i < k; i++) {
            s[i] = sc.nextInt();
        }
        int ans = -1;
        for (int v : s) {
            if (ans == -1 || ans > d[v]) ans = d[v];
        }
        System.out.println(ans);
    }
}
