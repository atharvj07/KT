#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) cin >> arr[i];
    int ans = 0;
    int prev = arr[0];
    for (int i = 0; i < n * t; i++) {
        int cur = arr[i % n];
        if (i > 0 && cur < prev) cur = prev;
        prev = cur;
        ans++;
    }
    cout << ans << endl;
    return 0;
}
