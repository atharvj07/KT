import sys

input = sys.stdin.readline

n, m = map(int, input().split())
g = [[] for i in range(n + 1)]
for i in range(m):
    u, v = map(int, input().split())
    g[u].append(v)

vis = [0] * (n + 1)
col = [0] * m


def dfs(x, c):
    vis[x] = 1
    for y in g[x]:
        if vis[y] == 0:
            if dfs(y, c + 1):
                return True
            col[g[x].index(y)] = c
        elif vis[y] == 1 and col[g[x].index(y)] == c:
            return True
    vis[x] = 2
    return False


mn = n + 1
for i in range(1, n + 1):
    if vis[i] == 0:
        if dfs(i, 1):
            mn = 2
        else:
            mn = 1

print(mn)
for i in col:
    print(i, end=' ')
