#include <bits/stdc++.h>

using namespace std;

int n, m, a[501][501], b[501], c[501];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> a[i][j];
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int mx = 0;
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == 2) {
                b[i] = c[j] = 1;
                break;
            }
            mx = max(mx, a[i][j] % 2);
        }
        ans += mx;
    }
    for (int i = 1; i <= n; i++) {
        if (b[i]) continue;
        int mx = 0;
        for (int j = 1; j <= m; j++) {
            mx = max(mx, a[i][j]);
        }
        if (mx % 2 == 0) ++mx;
        ans += mx - a[i][1];
    }
    for (int i = 1; i <= m; i++) {
        if (c[i]) continue;
        int mx = 0;
        for (int j = 1; j <= n; j++) {
            mx = max(mx, a[j][i]);
        }
        if (mx % 2 == 0) ++mx;
        ans += mx - a[1][i];
    }
    cout << ans << '\n';
    return 0;
}
