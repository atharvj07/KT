#include <iostream>
using namespace std;
const int N = 1e5 + 5;
int32_t a[N], cnt[(1 << 15) + 5];
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n; ++i) {
    int cur = 0;
    for (int j = 0; j <= 14; ++j) {
      cur |= ((a[i] >> j) & 1) << j;
      cnt[cur] ^= (1 << i);
      cur ^= ((1 << j) & (1 << k));
      cnt[cur] ^= (1 << i);
    }
  }
  int ans = 0;
  for (int i = 0; i < (1 << 15); ++i) {
    ans += (__builtin_popcount(cnt[i]) * (__builtin_popcount(cnt[i]) - 1)) / 2;
  }
  cout << ans << "\n";
  return 0;
}
