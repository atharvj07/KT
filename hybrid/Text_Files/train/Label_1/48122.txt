import java.util.Scanner;

public class LexicographicallyMaximumSubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();

        int n = s.length();
        int[] dp = new int[n];
        dp[n - 1] = 1;

        for (int i = n - 2; i >= 0; i--) {
            int max = 0;
            for (int j = i + 1; j < n; j++) {
                if (s.charAt(j) >= s.charAt(i)) {
                    max = Math.max(max, dp[j]);
                }
            }
            dp[i] = max + 1;
        }

        int maxIndex = 0;
        for (int i = 0; i < n; i++) {
            if (dp[i] > dp[maxIndex]) {
                maxIndex = i;
            }
        }

        StringBuilder result = new StringBuilder();
        while (maxIndex >= 0) {
            result.append(s.charAt(maxIndex));
            int prevMaxIndex = maxIndex;
            maxIndex = -1;
            for (int i = prevMaxIndex - 1; i >= 0; i--) {
                if (s.charAt(i) <= s.charAt(prevMaxIndex) && dp[i] == dp[prevMaxIndex] - 1) {
                    maxIndex = i;
                    break;
                }
            }
        }

        System.out.println(result.reverse().toString());
    }
}
