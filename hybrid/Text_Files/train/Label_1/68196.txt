n = int(input())
measurements = list(map(int, input().split()))
measurements.sort()

# Initialize result
result = n

# Use two pointers approach
for i in range(n):
    for j in range(i, n):
        if measurements[j] > 2 * measurements[i]:
            break
        result = min(result, i + (n - 1 - j))

print(result)
