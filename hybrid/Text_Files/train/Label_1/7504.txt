#include <algorithm>
#include <vector>
#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    vector<pair<int, int> > ans;
    int bal = 0;
    for (int i = 0; i < n; ++i) {
        bal += (s[i] != t[i]);
    }
    if (bal % 2 != 0) {
        cout << "-1" << endl;
        return 0;
    }
    bal /= 2;
    for (int i = 0; i < n; ++i) {
        if (s[i] != t[i]) {
            if (bal > 0) {
                ans.push_back(make_pair(i, i));
                bal -= 1;
                s[i] = t[i];
            } else {
                for (int j = i + 1; j < n; ++j) {
                    if (s[i] == s[j] && t[i] == t[j]) {
                        ans.push_back(make_pair(j, i));
                        ans.push_back(make_pair(i, j));
                        swap(s[i], s[j]);
                        swap(t[i], t[j]);
                        break;
                    }
                }
            }
        }
    }
    cout << ans.size() << endl;
    for (size_t i = 0; i < ans.size(); ++i) {
        cout << ans[i].first + 1 << ' ' << ans[i].second + 1 << endl;
    }
    return 0;
}
