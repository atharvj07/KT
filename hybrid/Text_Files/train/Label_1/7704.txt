#include <cstdio>
#include <iostream>
#include <set>
#include <list>
#include <map>
#include <algorithm>
#include <stdlib.h>

using namespace std;

int main() {
    string a, b;

    cin >> a >> b;

    if (a == b) {
        cout << "=";
    } else {
        if (a.length() < b.length()) {
            cout << "<";
        } else if (a.length() > b.length()) {
            cout << ">";
        } else {
            for (int i = 0; i < a.length(); i++) {
                if (a[i] < b[i]) {
                    cout << "<";
                    return 0;
                } else if (a[i] > b[i]) {
                    cout << ">";
                    return 0;
                }
            }
        }
    }

    return 0;
}
