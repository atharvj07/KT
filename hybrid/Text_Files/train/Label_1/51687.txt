import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int n = sc.nextInt();
      if (n == 0) break;
      String[] words = new String[n];
      for (int i = 0; i < n; i++) words[i] = sc.next();
      for (int i = 0; i < n; i++) {
        boolean ok = true;
        int l = 0;
        int j = i;
        for (int k = 0; k < 5; k++) {
          int len = (k == 0 || k == 2) ? 5 : 7;
          for (; j < n && l < len; j++) l += words[j].length();
          if (l != len) {
            ok = false;
            break;
          }
          l = 0;
        }
        if (ok) {
          System.out.println(i + 1);
          break;
        }
      }
    }
  }
}
