#include <iostream>
#include <vector>
#include <queue>
#include <cstring>
using namespace std;

const int maxn = 205;
const int inf = 1e9;
int dx[] = {1, -1, 0, 0};
int dy[] = {0, 0, 1, -1};
char dir[] = {'D', 'U', 'R', 'L'};

struct node {
    int x;
    int y;
    int dir;
    int dist;
};

int n, m, k;
int sx[maxn], sy[maxn], fx[maxn], fy[maxn];
int vis[maxn][maxn];
int dist[maxn][maxn][4];

bool valid(int x, int y) {
    return x >= 1 && x <= n && y >= 1 && y <= m;
}

int bfs() {
    memset(vis, 0, sizeof(vis));
    memset(dist, inf, sizeof(dist));
    queue<node> q;
    for (int i = 0; i <= k; i++) {
        q.push({sx[i], sy[i], 0, 0});
        vis[sx[i]][sy[i]] |= 1 << i;
        dist[sx[i]][sy[i]][0] = 0;
    }
    while (!q.empty()) {
        node cur = q.front();
        q.pop();
        for (int i = 0; i < 4; i++) {
            int nx = cur.x + dx[i];
            int ny = cur.y + dy[i];
            if (!valid(nx, ny)) continue;
            int d = (cur.dir + i) % 4;
            if (dist[nx][ny][d] > dist[cur.x][cur.y][cur.dir] + 1) {
                dist[nx][ny][d] = dist[cur.x][cur.y][cur.dir] + 1;
                q.push({nx, ny, d, dist[nx][ny][d]});
                vis[nx][ny] |= vis[cur.x][cur.y];
            }
        }
    }
    int ans = inf;
    for (int i = 0; i <= k; i++) {
        for (int j = 0; j < 4; j++) {
            if (vis[fx[i]][fy[i]] & (1 << i) && dist[fx[i]][fy[i]][j] != inf) {
                ans = min(ans, dist[fx[i]][fy[i]][j]);
            }
        }
    }
    if (ans == inf) return -1;
    else return ans;
}

void print_path(int x, int y, int d, int ans) {
    if (ans == 0) return;
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        if (!valid(nx, ny)) continue;
        int d2 = (d + i) % 4;
        if (dist[nx][ny][d2] + 1 == dist[x][y][d]) {
            cout << dir[i];
            print_path(nx, ny, d2, ans - 1);
            break;
        }
    }
}

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i <= k; i++) {
        cin >> sx[i] >> sy[i];
    }
    for (int i = 0; i <= k; i++) {
        cin >> fx[i] >> fy[i];
    }
    int ans = bfs();
    if (ans == -1) cout << -1;
    else {
        cout << ans << endl;
        print_path(sx[0], sy[0], 0, ans);
        cout << endl;
    }
    return 0;
}
