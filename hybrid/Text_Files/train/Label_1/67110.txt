def calculate_swaps(n, permutation):
    def merge_sort(arr, temp, left, right):
        inv_count = 0
        if left < right:
            mid = (left + right) // 2
            inv_count += merge_sort(arr, temp, left, mid)
            inv_count += merge_sort(arr, temp, mid + 1, right)
            inv_count += merge(arr, temp, left, mid, right)
        return inv_count

    def merge(arr, temp, left, mid, right):
        i = left
        j = mid + 1
        k = left
        inv_count = 0
        while i <= mid and j <= right:
            if arr[i] <= arr[j]:
                temp[k] = arr[i]
                k += 1
                i += 1
            else:
                temp[k] = arr[j]
                k += 1
                j += 1
                inv_count += (mid - i + 1)
        while i <= mid:
            temp[k] = arr[i]
            k += 1
            i += 1
        while j <= right:
            temp[k] = arr[j]
            k += 1
            j += 1
        for loop_var in range(left, right + 1):
            arr[loop_var] = temp[loop_var]
        return inv_count

    total_inversions = merge_sort(permutation, [0]*n, 0, n-1)
    answer = 0
    for i in range(n):
        for j in range(i + 1, n):
            if permutation[i] > permutation[j]:
                current_inversions = total_inversions - (permutation[i] - permutation[j] - 1)
                if current_inversions == total_inversions - 1:
                    answer += 1
    print(total_inversions, answer)

n = int(input().strip())
permutation = list(map(int, input().strip().split()))
calculate_swaps(n, permutation)
