from collections import defaultdict

def dfs(node, depth):
    if len(children[node]) == 0:
        depths.append(depth)
        return
    for child in children[node]:
        dfs(child, depth + 1)

n = int(input())
children = defaultdict(list)
for _ in range(n - 1):
    u, v = map(int, input().split())
    children[u].append(v)

depths = []
dfs(1, 0)

max_depth = max(depths)
min_moves = max_moves = 0

for depth in depths:
    if depth % 2 == max_depth % 2:
        min_moves += 1

max_moves = 2 ** (max_depth // 2) if max_depth % 2 == 0 else 2 ** ((max_depth + 1) // 2)

print(max_moves, min_moves)
