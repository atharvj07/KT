#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, t = 0, k = 0;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    for (int i = 0; i < n; ++i) { cin >> b[i]; }
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    for (int i = 0; i < n; ++i) {
        if (k == 0) { ++k; }
        if (b[k - 1] < a[i]) { ++k; }
        t += min(b[k - 1], a[i]) - a[i];
    }
    cout << k << ' ' << t << '\n';
    return 0;
}
