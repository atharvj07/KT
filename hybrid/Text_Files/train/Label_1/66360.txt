import sys
from collections import defaultdict
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
se = defaultdict(list)
for i in range(m):
    l, r = map(int, input().split())
    for j in range(l, r + 1):
        se[j].append(i)

mn = min(a)
mx = max(a)
ans = mx - mn
cnt = 0
res = []
for i in range(1, n + 1):
    if len(se[i]) == 0:
        continue
    for j in se[i]:
        if a[i] == mx:
            a[i] -= 1
            ans += 1
            cnt += 1
            res.append(j + 1)
            break
        elif a[i] == mn:
            a[i] += 1
            ans += 1
            cnt += 1
            res.append(j + 1)
            break
print(ans, cnt)
print(*res)
