for _ in range(int(input())):
    n = int(input())
    s = input()
    i_t, i_r, i_y, i_g, i_u = (s.rfind(chr(ord('t') + i)) for i in range(5))
    while i_r > i_t > i_y > i_g > i_u:
        s = s[:i_t] + s[i_r + 1: i_y] + s[i_t + 1: i_r] + s[i_y + 1: i_g] + s[i_g + 1: i_u] + s[i_u + 1:]
        i_t, i_r, i_y, i_g, i_u = (s.rfind(chr(ord('t') + i)) for i in range(5))
    print(s)
