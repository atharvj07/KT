#include <iostream>
using namespace std;

long long w[3];
int main() {
  for (int i = 0; i < 3; ++i) { cin >> w[i]; }
  long long a = w[0], b = w[1], c = w[2];
  long long s = a / c + b / c;
  long long ans = 1e18;
  for (long long j = 0; j <= min(a, b); ++j) {
    long long na = a + j, nb = b - j;
    long long ns = na / c + nb / c;
    long long na_need = c * ns - na;
    long long nb_need = c * ns - nb;
    if (ns > s || (ns == s && na_need < ans)) {
      ans = na_need;
      s = ns;
    }
  }
  cout << s << " " << ans << endl;
  return 0;
}
