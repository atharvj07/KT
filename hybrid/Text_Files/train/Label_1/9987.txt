#include <iostream>
#include <cmath>
using namespace std;
int main() {
  long long y1, y2, yw, xb, yb, r;
  double ansx;
  cin >> y1 >> y2 >> yw >> xb >> yb >> r;
  if (y2 - y1 < 2 * r) {
    cout << "-1" << endl;
    return 0;
  }
  if (yb + r >= yw) {
    cout << "-1" << endl;
    return 0;
  }
  if (y2 - y1 > 2 * r) {
    ansx = xb * 1.0 * (yw - yb - r) / (yw - yb);
    double ansy = yb + (yw - yb - r) * yb / (yw - yb);
    if (ansx > 0.0 && yw - ansy > 0.0 && ansy >= 0.0 && ansy <= yw) {
      cout << ansx << endl;
      return 0;
    }
  }
  ansx = xb * 1.0 * (yw - yb - r - 2 * y1) / (yw - yb - 2 * r);
  double ansy = yb + (yw - yb - r - 2 * y1) * yb / (yw - yb - 2 * r);
  if (ansx > 0.0 && ansy >= 0.0 && ansy <= yw) {
    cout << ansx << endl;
    return 0;
  }
  cout << "-1" << endl;
  return 0;
}
