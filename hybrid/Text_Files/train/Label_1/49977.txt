import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long x = sc.nextLong();
        long y = sc.nextLong();
        long t = Math.min(n * x, (y >> 1) + (x + y) * n);
        System.out.println(t);
    }
}
