#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>
#include <algorithm>

using namespace std;

struct HashFunction {
    size_t operator()(const string& str) const {
        size_t hash = 0;
        for (char c : str) {
            hash = 31 * hash + c - 'a' + 1;
        }
        return hash;
    }
};

struct EqualFunction {
    bool operator()(const string& lhs, const string& rhs) const {
        return lhs == rhs;
    }
};

int main() {
    int n;
    cin >> n;
    vector<string> strs(n);
    unordered_map<string, vector<int>, HashFunction, EqualFunction> str_map;
    for (int i = 0; i < n; i++) {
        cin >> strs[i];
        str_map[strs[i]].push_back(i);
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        int l = strs[i].size();
        for (int j = 0; j <= l; j++) {
            string prefix = strs[i].substr(0, j);
            string suffix = strs[i].substr(j);
            reverse(suffix.begin(), suffix.end());
            if (str_map.count(prefix) && str_map.count(suffix) && str_map[prefix].size() > 0 && str_map[suffix].size() > 0) {
                if (prefix == suffix) {
                    cnt += str_map[prefix].size() - 1;
                } else {
                    cnt += str_map[prefix].size() * str_map[suffix].size();
                }
            }
        }
    }
    cout << cnt << endl;
    return 0;
}
