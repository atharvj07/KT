def can_assign(R, B, rebels, bases):
    if R != B:
        return "No"
    rebels.sort(key=lambda x: (x[0], x[1]))
    bases.sort(key=lambda x: (x[0], x[1]))
    for i in range(1, R):
        for j in range(i):
            x1, y1 = rebels[i]
            x2, y2 = rebels[j]
            x3, y3 = bases[i]
            x4, y4 = bases[j]
            if (x2 - x1) * (y4 - y3) - (y2 - y1) * (x4 - x3) == 0:
                continue
            t1 = ((x3 - x1) * (y4 - y3) - (y3 - y1) * (x4 - x3)) / ((x2 - x1) * (y4 - y3) - (y2 - y1) * (x4 - x3))
            t2 = ((x3 - x1) * (y2 - y1) - (y3 - y1) * (x2 - x1)) / ((x4 - x3) * (y2 - y1) - (y4 - y3) * (x2 - x1))
            if 0 < t1 < 1 and 0 < t2 < 1:
                return "No"
    return "Yes"

R, B = map(int, input().split())
rebels = [list(map(int, input().split())) for _ in range(R)]
bases = [list(map(int, input().split())) for _ in range(B)]
print(can_assign(R, B, rebels, bases))
