#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  vector<vector<int>> dp(2, vector<int>(k + 1, INT_MAX / 2));
  dp[0][0] = 0;
  for (int j = 1; j <= k; j++) {
    dp[0][j] = a[n - j];
    for (int l = n - j - 1; l >= 0; l--) {
      if (j % 2 == 0) {
        dp[j % 2][j] = min(dp[j % 2][j], max(a[l], dp[(j - 1) % 2][j - 1]));
      } else {
        dp[j % 2][j] = min(dp[j % 2][j], max(a[l], dp[(j - 1) % 2][j]));
      }
    }
    dp[(j + 1) % 2].assign(k + 1, INT_MAX / 2);
  }
  cout << dp[k % 2][k] << endl;
  return 0;
}
