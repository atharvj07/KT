#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
/*
 * @param {string} s
 * @return {ll}
 */
ll solve(string s) {
    ll n = s.length(), ans = 0;
    vector<vector<ll>> pal(n, vector<ll>(n, 0));
    for (ll i = 0; i < n; i++) pal[i][i] = 1;
    for (ll len = 2; len <= n; len++) {
        for (ll i = 0; i + len - 1 < n; i++) {
            ll j = i + len - 1;
            if (len == 2) pal[i][j] = (s[i] == s[j]);
            else pal[i][j] = (s[i] == s[j] && pal[i + 1][j - 1]);
        }
    }
    for (ll l = 0; l < n; l++) {
        for (ll r = l; r < n; r++) {
            if(pal[l][r]) ans++;
        }
    }
    return ans;
}
int main() {
    ll n;
    cin >> n;
    string s;
    cin >> s;
    cout << solve(s) << '\n';
    return 0;
}
