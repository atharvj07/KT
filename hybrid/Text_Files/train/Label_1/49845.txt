import java.util.Scanner;

public class DimaAndStaircase {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] stairs = new int[n];
        for (int i = 0; i < n; i++) {
            stairs[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        for (int i = 0; i < m; i++) {
            int width = scanner.nextInt();
            int height = scanner.nextInt();

            int bottomHeight = findBottomHeight(stairs, width, height);
            System.out.println(bottomHeight);
        }
    }

    private static int findBottomHeight(int[] stairs, int width, int height) {
        int bottomHeight = stairs[0];
        for (int i = 1; i < width; i++) {
            bottomHeight = Math.max(bottomHeight, stairs[i]);
        }
        return bottomHeight + height;
    }
}
