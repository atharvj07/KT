import java.util.Scanner;

public class Main {

    private static long query(int[] a, int[] b, int k, int l1, int r1, int l2, int r2) {
        int n = a.length;
        int[] fa = new int[n];
        int[] fb = new int[n];
        for (int i = 0; i < n; i++) {
            fa[i] = (a[i] + k - 1) % n + 1;
            fb[i] = (b[i] + k - 1) % n + 1;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (l1 <= fa[i] && fa[i] <= r1 && l2 <= fb[i] && fb[i] <= r2) {
                ans++;
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        int m = in.nextInt();
        for (int i = 0; i < m; i++) {
            int l1 = in.nextInt();
            int r1 = in.nextInt();
            int l2 = in.nextInt();
            int r2 = in.nextInt();
            System.out.println(query(a, b, i, l1, r1, l2, r2));
        }
        in.close();
    }
}
