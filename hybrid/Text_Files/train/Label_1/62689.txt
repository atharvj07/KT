from collections import defaultdict

MOD = int(1e9) + 7

n = int(input())
a = list(map(int, input().split()))

freq = defaultdict(int)
for x in a:
    freq[x] += 1

coprime_subsequences = 1
for _, v in freq.items():
    coprime_subsequences = (coprime_subsequences * ((1 << v) - 1)) % MOD

print(coprime_subsequences)
