import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int D = sc.nextInt();
        int[] d = new int[N];
        for (int i = 0; i < N; i++) {
            d[i] = sc.nextInt();
        }
        int Q = sc.nextInt();
        int[] q = new int[Q];
        for (int i = 0; i < Q; i++) {
            q[i] = sc.nextInt();
        }
        for (int i = 0; i < Q; i++) {
            int index = q[i] - 1;
            int sum = D;
            for (int j = 0; j < N; j++) {
                if (j == index) continue;
                sum -= d[j];
            }
            if (sum <= 0) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
