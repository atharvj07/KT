#include <cstdio>
#include <algorithm>
#pragma GCC optimize("Ofast")
using namespace std;

const int MAXN = 5005;
int n, m, a[MAXN], id[MAXN], p[MAXN], pre[MAXN], id2[MAXN], ans;
double ip[MAXN];

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) {
        scanf("%d%lf", p + i, ip + i);
        id[i] = id2[p[i]] = i;
    }
    sort(id2 + 1, id2 + m + 1);
    for (int i = 1; i <= n; i++) {
        if (id[i] > pre[p[i]]) {
            pre[p[i]] = id[i];
            if (p[i] != p[i - 1]) ans++;
        }
    }
    printf("%d\n", ans);
    return 0;
}
