#include <bits/stdc++.h>
using namespace std;

int main() {
    int t, a, b, c, m; cin >> t;
    while (t--) {
        cin >> a >> b >> c >> m;
        if ((a + b + c) / 2 < m) cout << "NO" << endl;
        else if ((a + b + c) % 2 == 1 && m % 2 == 1) cout << "NO" << endl;
        else if ((a + b + c) % 2 == 0 && m % 2 == 0) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
