from collections import defaultdict

n = int(input())
a = [int(input()) for i in range(n)]
a.sort()
dp = defaultdict(int)
dp[a[n - 1]] = 1
for i in range(n - 2, -1, -1):
    last = 0
    for j in range(i + 1, n):
        if a[j] >= 2 * a[i] and dp[a[j]] > last:
            last = dp[a[j]]
    dp[a[i]] = last + 1
print(n - max(dp.values()))
