#include<bits/stdc++.h>
using namespace std;
int n,k,a[400001],b[400001],c[400001],x,y,z,ans;
int main(){
  cin>>n>>k;
  for(int i=1;i<=n;i++){
    cin>>a[i];
    c[a[i]]=i;
  }
  for(int i=1;i<=k;i++){
    cin>>b[i];
    if(!c[b[i]])c[b[i]]=n+1;
  }
  x=c[b[1]];
  y=a[1];
  for(int i=2;i<=k;i++){
    z=c[b[i]];
    if(z<x)x=z;
    if(y<z)y=z;
  }
  ans=y-x+1;
  if(ans>n)ans=n;
  cout<<ans<<endl;
}
