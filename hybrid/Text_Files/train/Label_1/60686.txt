
n, k, q = map(int, input().split())
l = [0] * 200001
r = [0] * 200001
for i in range(n):
  x, y = map(int, input().split())
  l[x] += 1
  r[y] -= 1
for i in range(1, 200001):
  l[i] += l[i - 1]
  r[i] += r[i - 1]
for i in range(q):
  a, b = map(int, input().split())
  print(max(0, l[b] - r[a - 1] - ((l[b] > 0) and (r[a - 1] == 0))))
