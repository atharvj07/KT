#include <bits/stdc++.h>

using namespace std;

int n, ans;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      for (int k = j; k <= n; k++) {
        if ((i ^ j ^ k) == 0) {
          if (((i - j) * (i - j) + (j - k) * (j - k) - (i - k) * (i - k)) > 0)
            ans++;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
