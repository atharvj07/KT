import java.util.Scanner;

class Respect {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        int[] army = new int[0];

        for (int i = 0; i < q; i++) {
            int event = scanner.nextInt();
            int p = scanner.nextInt();

            if (event == 1) {
                army = addWarrior(army, p);
            } else if (event == 2) {
                army = removeWarrior(army, p);
            } else if (event == 3) {
                int l = scanner.nextInt();
                int count = countRespectingWarriors(army, p, l);
                System.out.println(count);
            }
        }
    }

    static int[] addWarrior(int[] army, int p) {
        int[] newArmy = new int[army.length + 1];
        System.arraycopy(army, 0, newArmy, 0, army.length);
        newArmy[newArmy.length - 1] = p;
        return newArmy;
    }

    static int[] removeWarrior(int[] army, int p) {
        int[] newArmy = new int[army.length - 1];
        int index = -1;
        for (int i = 0; i < army.length; i++) {
            if (army[i] == p) {
                index = i;
                break;
            }
        }
        if (index != -1) {
            System.arraycopy(army, 0, newArmy, 0, index);
            System.arraycopy(army, index + 1, newArmy, index, army.length - index - 1);
        }
        return newArmy;
    }

    static int countRespectingWarriors(int[] army, int p, int l) {
        int count = 0;
        for (int warrior : army) {
            if ((warrior ^ p) < l) {
                count++;
            }
        }
        return count;
    }
}
