t = int(input())
for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    sum, prod = 0, 1
    for x in a:
        sum += x
        prod *= x
    if (sum == 0 or prod == 0) and (len(set(a)) > 1):
        print(1)
    elif (sum != 0 and prod != 0) or (sum == 0 and len(set(a)) == 1) or (prod == 0 and len(set(a)) == 1):
        print(0)
    else:
        print(2)
