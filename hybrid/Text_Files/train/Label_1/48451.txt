import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	/**
	 * @param n upper bracket layer, you can say the power here
	 * @param f is the list of your favorite team
	 * @param x we will use to calculate 2^x
	 * @param fav which is the number of team which is your favorite
	 * @return return maximum matches of your favorite team
	 */
	static int solve(int n, int[] f, int x, int fav) {
		if (fav == 0 || n <= 0)
			return 0;

		x++;

		for ( int i = 0 ; i < fav ; i++ )
			if ( ((f[i] - 1) >> x) % 2 == 0 )
				fav--;
			else
				break;

		if (fav > 0)
			return 1 + solve(n - 1, f, x, fav);
		else
			return 0;
	}
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner input = new Scanner(System.in);
		int n = input.nextInt();
		int fav = input.nextInt();
		int[] f = new int[fav];
		for ( int i = 0 ; i < fav ; i++)
			f[i] = input.nextInt();

		Arrays.sort(f);
		int ans = solve(n, f, 0, fav);
		System.out.println(n * (ans + 1));
	}
}
