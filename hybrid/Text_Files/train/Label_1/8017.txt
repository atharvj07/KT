#include<bits/stdc++.h>
using namespace std;
int main()
{
 int n,a,b,x[105],y[105];
 cin>>n>>a>>b;
 for(int i=0;i<n;i++) cin>>x[i]>>y[i];
 sort(x,x+n);sort(y,y+n);
 int ans=min(a,b)*2;

 for(int i=0;i<n;i++)
 {
  if(x[i]<=a&&y[i]<=b) ans=max(ans,(a/x[i])*(b/y[i])*x[i]*y[i]);
  swap(x[i],y[i]);
  if(x[i]<=a&&y[i]<=b) ans=max(ans,(a/x[i])*(b/y[i])*x[i]*y[i]);
 }

 cout<<ans<<endl;
}
