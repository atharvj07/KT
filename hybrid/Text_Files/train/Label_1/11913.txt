#include<bits/stdc++.h>
using namespace std;
struct data
{
    int a,b,c;
    char x;
}x[100001];
int c[100001],a[100001][4],f[100001][4][4],k[100001],n,m,ans[100001];
bool cmp(const data &x1,const data &x2)
{
    return x1.b<x2.b;
}
int main()
{
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        string s;
        cin>>s;
        k[i]=s.size();
        for(int j=0;j<k[i];j++)
        {
            if(s[j]=='a'||s[j]=='e'||s[j]=='i'||s[j]=='o'||s[j]=='u')
            {
                x[i].a++;
                x[i].x=s[j];
                if(j==k[i]-1)
                {
                    x[i].c=j;
                }
            }
        }
        x[i].b=x[i].a;
    }
    sort(x+1,x+1+n,cmp);
    for(int i=1;i<=n;i++)
    {
        for(int j=0;j<=3;j++)
        {
            a[i][j]=a[i-1][j];
            if(j==x[i].a)
            {
                a[i][j]++;
            }
        }
    }
    for(int i=1;i<=n;i++)
    {
        for(int j=0;j<=3;j++)
        {
            for(int l=0;l<=3;l++)
            {
                f[i][j][l]=f[i-1][j][l];
                if(j==x[i].a&&l==x[i].a&&x[i].x==x[i-1].x)
                {
                    f[i][j][l]+=a[i-1][j-1];
                }
            }
        }
    }
    for(int i=n;i>=1;i--)
    {
        for(int j=3;j>=0;j--)
        {
            for(int l=3;l>=0;l--)
            {
                if(f[i][j][l]==f[i-1][j][l])
                {
                    continue;
                }
                c[i]=1;
                ans[++m]=i;
                int last=l;
                while(l!=j)
                {
                    while(x[i].a!=l||x[i].x!=x[ans[m]].x)
                    {
                        i--;
                    }
                    ans[++m]=i;
                    l--;
                    last=l;
                }
                while(x[ans[m]].c!=x[ans[m]+1].c)
                {
                    m--;
                }
                while(k[ans[m]]==k[ans[m]+1])
                {
                    m--;
                }
                m--;
                if(x[ans[m]].a!=last)
                {
                    while(x[ans[m]].a!=x[ans[m]+1].a||x[ans[m]].x!=x[ans[m]+1].x)
                    {
                        m--;
                    }
                }
                i=ans[m];
                break;
            }
        }
    }
    cout<<m<<endl;
    for(int i=1;i<=m;i++)
    {
        for(int j=i;j<=m;j++)
        {
            if(ans[j]<ans[i])
            {
                swap(ans[j],ans[i]);
            }
        }
    }
    for(int i=1;i<=m;i++)
    {
        for(int j=ans[i];j<=ans[i+1];j++)
        {
            cout<<x[j].b<<' ';
        }
        cout<<endl;
        for(int j=ans[i];j<=ans[i+1];j++)
        {
            cout<<k[j]-x[j].c-1<<' ';
        }
        cout<<endl;
        i++;
    }
}
