#include <bits/stdc++.h>
using namespace std;
void solve() {
    int n, m, a, b;
    cin >> n >> m >> a >> b;
    if (n * a != m * b) { cout << "NO" << endl; return; }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) cout << ((i % b == 1 and j % a == 1) ? 1 : 0);
        cout << '\n';
    }
}
int main() {
    int t;
    cin >> t;
    while (t--) solve();
}
