#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    vector<int> ans(n);
    for (int k = 0; k < n; k++) {
        int p = k & 1;
        if (k % 2 == n % 2) reverse(a.begin(), a.end());
        ans[k] = a[p];
        for (int i = k + 1; i < n; i++) {
            if (i % 2 == k % 2) a[i] = min(a[i], a[i - 1]);
            else a[i] = max(a[i], a[i - 1]);
        }
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << endl;
}
