import collections

n, m = map(int, input().split())
g = collections.defaultdict(set)
for _ in range(m):
    a, b = map(int, input().split())
    g[a].add(b)
    g[b].add(a)
ans = 0
for x in g:
    for y in g[x]:
        if x < y:
            for z in g[y]:
                if z != x and z in g[x]:
                    ans += 1
print(ans)
