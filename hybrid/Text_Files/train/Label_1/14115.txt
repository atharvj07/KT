#include <iostream>
#include <vector>
using namespace std;

int main() {
  int tc;
  cin >> tc;
  while (tc--) {
    int n, x, y;
    cin >> n >> x >> y;
    if (n == 2) {
      cout << x << " " << y << endl;
      continue;
    }
    vector<int> arr(n);
    int sum = (n - 2) * (y - x);
    arr[0] = x;
    for (int i = 1; i < n - 1; i++) {
      arr[i] = y - (sum / (n - 1 - i));
      sum -= arr[i] - arr[i - 1];
    }
    arr[n - 1] = y;
    for (int i = 0; i < n; i++) cout << arr[i] << " ";
    cout << endl;
  }
}
