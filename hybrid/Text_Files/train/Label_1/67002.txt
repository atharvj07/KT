t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    
    def min_moves(n, k, a):
        correct_positions = [0] * (n + 1)
        for i in range(n):
            if a[i] == i + 1:
                correct_positions[i + 1] = correct_positions[i] + 1
            else:
                correct_positions[i + 1] = correct_positions[i]
        
        if correct_positions[n] >= k:
            return 0
        
        max_additional_correct = 0
        for i in range(n):
            for j in range(i, n):
                original_correct = correct_positions[i]
                additional_correct = sum(1 for x in range(i+1, j+2) if a[x-1] == x)
                total_correct = original_correct + additional_correct + (correct_positions[n] - correct_positions[j+1])
                max_additional_correct = max(max_additional_correct, total_correct)
                if max_additional_correct >= k:
                    return i + (n - j - 1)
        
        return -1
    
    print(min_moves(n, k, a))
