import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int t = in.nextInt();
    while (t-- > 0) {
      int n = in.nextInt(), m = in.nextInt(), k = in.nextInt();
      System.out.println((n + m - 2 == k) ? "YES" : "NO");
    }
  }
}
