n1, y1 = map(int, input().split())
a = list(map(int, input().split()))
n2, y2 = map(int, input().split())
b = list(map(int, input().split()))
a.sort(); b.sort()
ans = -1
if(y2 - y1 > 0):
    for i in range(0, n1):
        for j in range(0, n2):
            A = min(a[i], b[j])
            B = max(a[i], b[j])
            C = round((a[i] * y2 - b[j] * y1) / (y2 - y1))
            if (min(C, A) <= B and B <= max(C, A)):
                ans = max(ans, j + i + 1)
print(ans)
