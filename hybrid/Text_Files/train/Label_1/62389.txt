n,q=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
st=[]
pos=[0 for i in range(51)]
for i in range(n-1,-1,-1):
    if(pos[a[i]]==0):st.append(i)
    pos[a[i]]=i
for i in b:
    if(len(st)==0):print(-1)
    else:
        ans=0
        while(st and a[st[-1]]!=i):st.pop()
        if(len(st)==0):print(-1)
        else:
            ans=st[-1]
            st.pop()
            for j in range(51):
                if(pos[j]==ans):pos[j]=n
            pos[i]=ans
            print(ans+1)
