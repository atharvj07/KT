import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] p = new int[n];
            for (int j = 0; j < n; j++) {
                p[j] = sc.nextInt();
            }
            boolean cw = true, ccw = true;
            for (int j = 0; j < n; j++) {
                if (p[j] != (j + 1) % n + 1) {
                    cw = false;
                }
                if (p[j] != (j - 1 + n) % n + 1) {
                    ccw = false;
                }
            }
            System.out.println(cw || ccw ? "YES" : "NO");
        }
    }
}
