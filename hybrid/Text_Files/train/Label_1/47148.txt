import java.util.Scanner;

public class CupcakeTastiness {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            long[] a = new long[n];
            long sum = 0;

            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
                sum += a[i];
            }

            boolean isHappy = true;
            for (int i = 0; i < n; i++) {
                long adelTastiness = sum - a[i];
                if (adelTastiness >= sum) {
                    isHappy = false;
                    break;
                }
            }

            System.out.println(isHappy ? "YES" : "NO");
        }
    }
}
