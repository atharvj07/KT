#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    bool ok = true;
    for (int l = 0, r = n - 1; l < r; l++, r--) {
      if (a[l] != a[r]) ok = false;
    }
    cout << (ok ? 0 : 1) << '\n';
  }
  return 0;
}
