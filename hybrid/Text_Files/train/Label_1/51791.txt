import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args)
  {
    Scanner scanner = new Scanner(System.in);

    String s = scanner.nextLine();
    int n = s.length();
    int[] bal = new int[n + 1];

    for (int i = 0; i < n; ++i) {
      int c = s.charAt(i);
      bal[i + 1] = bal[i] + ((c == '(') ? 1 : -1);
    }

    int bestStart = -1;
    int bestCnt = n + 1;
    for (int i = 0; i <= n; ++i) {
      if (bal[i] == 0 && bal[i] + bal[n] < bestCnt) {
        bestStart = i;
        bestCnt = bal[i] + bal[n];
      }
    }

    int curCnt = 0;
    String ans = "";
    for (int i = bestStart; i < n; ++i) {
      int c = s.charAt(i);
      if ((c == ')') && (curCnt == 0)) {
        continue;
      }
      ans += s.charAt(i);
      curCnt += ((c == '(') ? 1 : -1);
    }

    for (int i = 0; i < bestStart; ++i) {
      int c = s.charAt(i);
      if ((c == ')') && (curCnt == 0)) {
        continue;
      }
      ans += s.charAt(i);
      curCnt += ((c == '(') ? 1 : -1);
    }

    System.out.println(ans);
  }
}
