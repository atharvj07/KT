#include <bits/stdc++.h>
using namespace std;
#define MOD 1000000007

long long FastPow(long long base, long long pow) {
    if (!pow) return 1;
    if (pow == 1) return base;
    long long p = FastPow(base, pow >> 1);
    p = (p * p) % MOD;
    if (pow & 1) p = (p * base) % MOD;
    return p;
}

int main() {
    long long k;
    cin >> k;
    vector<long long> v(k);
    for (int i = 0; i < k; i++) {
        cin >> v[i];
    }
    long long ans_up = 1, ans_down = 2;
    for (int i = 0; i < k; i++) {
        ans_up = (ans_up * (v[i] - 1)) % MOD;
        ans_down = (ans_down * v[i]) % MOD;
    }
    long long p = __gcd(ans_up, ans_down);
    ans_up /= p;
    ans_down /= p;
    cout << ans_up % MOD << "/" << ans_down % MOD << endl;
}
