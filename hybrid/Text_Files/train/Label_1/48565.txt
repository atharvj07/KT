import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            long x = (long) Math.sqrt(a[i]);
            if (x * x == a[i]) {
                ans -= (a[i] % 2);
            } else {
                ans++;
            }
        }
        ans /= 2;
        System.out.println(ans);
    }
}
