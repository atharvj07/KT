def count_valid_pairs(l, r):
    # The condition a + b = a XOR b holds true only if there are no common set bits
    # Therefore, we only need to count the pairs where this condition holds
    answer = 0
    for i in range(32):
        bit = 1 << i
        if (l & bit) == (r & bit) == 0:
            # If the current bit is not set in both l and r, all pairs in this range are valid
            range_size = min(bit, r + 1) - l
            answer += range_size * (range_size - 1)
        elif (l & bit) and (r & bit):
            # If the current bit is set in both l and r, adjust l and r to exclude this bit
            l -= bit
            r -= bit
        elif (l & bit) == 0 and (r & bit):
            # If the current bit is set in r but not in l, split the range
            mid = bit - 1
            if l <= mid:
                range_size = mid + 1 - l
                answer += range_size * (range_size - 1)
            if r >= bit:
                range_size = r - bit + 1
                answer += range_size * (range_size - 1)
            break  # No need to check higher bits
    return answer

t = int(input())
for _ in range(t):
    l, r = map(int, input().split())
    print(count_valid_pairs(l, r))
