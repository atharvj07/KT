#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  while (n--) {
    string s;
    getline(cin, s);
    if (s.back() == 'a' && s.substr(s.size() - 5) == "lala.")
      cout << "Freda's" << endl;
    else if (s[0] == 'm' && s.substr(0, 5) == "miao.")
      cout << "Rainbow's" << endl;
    else
      cout << "OMG>.< I don't know!" << endl;
  }
  return 0;
}
