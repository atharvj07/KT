#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, cnt = 0;
    cin >> n;
    string s[n];
    for (int i = 0; i < n; i++)
        cin >> s[i];
    for (int i = 0; i < 7; i++) {
        int sum = 0;
        for (int j = 0; j < n; j++)
            sum += s[j][i] - '0';
        cnt = max(cnt, sum);
    }
    cout << cnt << '\n';
    return 0;
}
