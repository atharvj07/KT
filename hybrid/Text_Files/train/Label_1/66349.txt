import sys
input = sys.stdin.readline

n = int(input())
salaries = []
for _ in range(n):
    _, *s = map(int, input().split())
    salaries.append(s)

salaries.sort(key=lambda x: -x[0])
ans = 0
mn = max(salaries[0])
for i in range(1, n):
    mx = max(salaries[i])
    mn = min(mn, mx)
    ans += mx - mn
print(ans)
