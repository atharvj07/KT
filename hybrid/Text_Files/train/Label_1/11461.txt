#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  vector<int> dp(n, 1);
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < i; j++) {
      if (a[i] > a[j]) {
        dp[i] = max(dp[i], dp[j] + 1);
      }
    }
  }

  int mx = *max_element(dp.begin(), dp.end());
  vector<int> b(n, 0);
  for (int i = n - 1; i >= 0; i--) {
    if (dp[i] == mx) {
      b[i] = 3;
      mx--;
    }
  }

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < i; j++) {
      if (a[i] > a[j] && dp[i] == dp[j] + 1) {
        b[i] = max(b[i], b[j] + 1);
      }
    }
  }

  for (int i = 0; i < n; i++) {
    if (b[i] == 0) {
      b[i] = 1;
    }
  }

  for (int i = 0; i < n; i++) {
    cout << b[i];
  }

  return 0;
}
