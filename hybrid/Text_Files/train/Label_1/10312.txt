#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for (int i = a; i < b; i++)

void solve(int n, int k, vector<int> &p) {
  int sm = p[0], ans = 0;
  rep(i, 1, n) {
    if (p[i] * 100 > (sm + p[i]) * k) {
      ans += p[i] * 100 - (sm + p[i]) * k;
      sm += (p[i] * 100 - (sm + p[i]) * k) / 100;
    }
    sm += p[i];
  }
  cout << ans << endl;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> p(n);
    rep(i, 0, n) cin >> p[i];
    solve(n, k, p);
  }
}
