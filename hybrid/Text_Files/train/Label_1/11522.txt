#include<bits/stdc++.h>

using namespace std;
typedef long long LL;
const int maxn = 4e5 + 5;
const LL INF = 1e18;

int main() {
    int t;
    cin >> t;
    while (t--) {
        LL n;
        cin >> n;
        vector<LL> a(n);
        LL ans = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            ans += a[i];
        }
        sort(a.rbegin(), a.rend());
        for (int i = 1; i < n; i++) {
            if (a[i] > a[0]) {
                ans += a[i] * i;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
