#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

int p[200005], b[200005], a[200005], inv[200005], e[200005];
vector<int> v;

int main() {
  int n, m, q;
  cin >> n >> m >> q;
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
    inv[p[i]] = i;
  }
  for (int i = 1; i <= m; ++i) {
    cin >> a[i];
    b[a[i]]++;
  }
  for (int i = 1; i <= n; ++i) if (!b[i]) v.push_back(i);
  for (int i = 1; i <= m; ++i) {
    e[i] = e[i - 1] + b[a[i]];
  }
  while (q--) {
    int l, r;
    cin >> l >> r;
    int t = r - l + 1 - e[r] + e[l - 1];
    cout << ((t < n) ? '0' : '1');
  }
  return 0;
}
