#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    int w = 0, h = 0;
    for (int i = 0; i < n; i++) {
        w += max(v[i].first, v[i].second);
        h = max(h, max(v[i].first, v[i].second));
    }
    cout << w * h << endl;
    return 0;
}
