#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const double eps = 1e-10;
int main() {
    int n;
    cin >> n;
    ll sum1 = 0, sum2 = 0;
    for (int i = 0; i < 2 * n; i++) {
        double x;
        cin >> x;
        if (x - floor(x) < eps) { sum1 += (int)floor(x); sum2 += (int)floor(x); }
        else if (ceil(x) - x < eps) { sum1 += (int)ceil(x); sum2 += (int)ceil(x); }
        else { sum1 += (int)floor(x); sum2 += (int)ceil(x); }
    }
    printf("%.3lf\n", abs((double)(sum1 - sum2) / 2));
}
