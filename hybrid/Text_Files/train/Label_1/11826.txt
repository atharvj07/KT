#include <bits/stdc++.h>

using namespace std;
#define int long long
#define vi vector<int>

#define all(a) a.begin(), a.end()
const int inf = 1e18;
const int mod = 45678;

int dp[(1 << 13)][13][13];
vi adj[13];
int32_t main() {
  int n, m;
  cin >> n >> m;
  int a, b;
  for (int i = 0; i < m; i++) {
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  int s1, t1, l1, s2, t2, l2;
  cin >> s1 >> t1 >> l1 >> s2 >> t2 >> l2;
  for (int i = 0; i < (1 << 13); i++) {
    for (int j = 0; j < 13; j++) {
      for (int k = 0; k < 13; k++) { dp[i][j][k] = inf; }
    }
  }
  dp[(1 << s1) | (1 << t1)][s1][t1] = 0;
  queue<tuple<int, int, int, int>> q;
  q.push({(1 << s1) | (1 << t1), s1, t1, 0});
  while (!q.empty()) {
    int mask, x, y, dis;
    tie(mask, x, y, dis) = q.front();
    q.pop();
    if (dis > l1) continue;
    for (auto i : adj[x]) {
      if (!(mask & (1 << i))) {
        dp[(mask | (1 << i))][i][y] = min(dp[(mask | (1 << i))][i][y],
                                          dis + 1);
        dp[(mask | (1 << i))][y][i] = min(dp[(mask | (1 << i))][y][i],
                                          dis + 1);
        q.push({(mask | (1 << i)), i, y, dis + 1});
      }
    }
  }
  int ans = inf;
  for (int i = 0; i < (1 << 13); i++) {
    for (int j = 0; j < 13; j++) {
      for (int k = 0; k < 13; k++) {
        if (dp[i][j][k] != inf && dp[i][j][k] <= l1) {
          for (auto x : adj[j]) {
            if (x != k && (i & (1 << x))) ans = min(ans, dp[i][j][k]);
          }
        }
      }
    }
  }
  for (int i = 0; i < (1 << 13); i++) {
    for (int j = 0; j < 13; j++) {
      for (int k = 0; k < 13; k++) { dp[i][j][k] = inf; }
    }
  }
  dp[(1 << s2) | (1 << t2)][s2][t2] = 0;
  q = queue<tuple<int, int, int, int>>();
  q.push({(1 << s2) | (1 << t2), s2, t2, 0});
  while (!q.empty()) {
    int mask, x, y, dis;
    tie(mask, x, y, dis) = q.front();
    q.pop();
    if (dis > l2) continue;
    for (auto i : adj[x]) {
      if (!(mask & (1 << i))) {
        dp[(mask | (1 << i))][i][y] = min(dp[(mask | (1 << i))][i][y],
                                          dis + 1);
        dp[(mask | (1 << i))][y][i] = min(dp[(mask | (1 << i))][y][i],
                                          dis + 1);
        q.push({(mask | (1 << i)), i, y, dis + 1});
      }
    }
  }
  ans = min(ans, inf);
  for (int i = 0; i < (1 << 13); i++) {
    for (int j = 0; j < 13; j++) {
      for (int k = 0; k < 13; k++) {
        if (dp[i][j][k] != inf && dp[i][j][k] <= l2) {
          for (auto x : adj[j]) {
            if (x != k && (i & (1 << x))) ans = min(ans, dp[i][j][k]);
          }
        }
      }
    }
  }
  if (ans == inf) {
    cout << -1 << endl;
    return 0;
  }
  cout << ans << endl;
}
