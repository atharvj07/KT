import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        long k = input.nextLong();
        long chunks = (n+1)/2;
        if (k <= chunks * m) {
            long ans = 1 + (k-1)/m;
            long col = k - (k-1)/m * m;
            long row = (col+1)/2;
            if (row%2 == 1) ans = n-ans+1;
            System.out.println(ans*col);
        } else {
            k -= chunks*m;
            long ans = chunks + 1 + (k-1)/m;
            long col = k - (k-1)/m * m;
            long row = (col+1)/2 + (n%2==1? (n+1)/2 : n/2);
            if (row%2 == 1) ans = n-ans+1;
            System.out.println(ans*col);
        }
    }
}
