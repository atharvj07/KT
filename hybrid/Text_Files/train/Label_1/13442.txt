#include<bits/stdc++.h>
using namespace std;
int main() {
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a == b && b == c) {
        long long x = a - d / 3;
        cout << x << " " << x << " " << x << endl;
    }
    else if (a == b) {
        cout << a << " " << (d - 2 * a) << " " << c << endl;
    }
    else if (a == c) {
        cout << a << " " << (d - 2 * a) << " " << b << endl;
    }
    else if (b == c) {
        cout << b << " " << (d - 2 * b) << " " << a << endl;
    }
    else {
        cout << a << " " << b << " " << (d - a - b) << endl;
    }
    return 0;
}
