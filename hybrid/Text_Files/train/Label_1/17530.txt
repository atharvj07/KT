#include <bits/stdc++.h>
using namespace std;
const int inf = (1<<30);
struct seg{ int l, r; };
struct um{ int p, x; };
bool cmpl(const seg& a, const seg& b){ return a.l < b.l; }
bool cmpu(const um& a, const um& b){ return a.x < b.x; }
int main()
{
    int a, n, m, l, r, p, x, f[2005], ans = inf;
    vector<seg> rain;
    vector<um> umberllas;
    cin >> a >> n >> m;
    for(int i = 0; i < n; i++)
    {
        cin >> l >> r;
        rain.push_back({l, r});
    }
    for(int i = 0; i < m; i++)
    {
        cin >> x >> p;
        umberllas.push_back({p, x});
    }
    for(int i = 0; i <= a; i++) f[i] = inf;
    sort(rain.begin(), rain.end(), cmpl);
    sort(umberllas.begin(), umberllas.end(), cmpu);
    f[0] = 0;
    for(int i = 0, j = 0; i <= a; i++)
    {
        if(j < m && i == umberllas[j].x)
        {
            f[i] = min(f[i], f[max(i - 1, 0)] + umberllas[j].p);
            j++;
        }
        for(int k = 0; k < n; k++)
        {
            if(rain[k].l <= i && rain[k].r > i)
            {
                f[i] = inf;
                break;
            }
        }
        if(i < a) f[i+1] = min(f[i+1], f[i]);
    }
    if(f[a] == inf) cout << -1 << endl;
    else cout << f[a] << endl;
    return 0;
}
