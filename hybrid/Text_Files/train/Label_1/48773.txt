import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> itemCost = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String itemName = sc.next();
            int cost = sc.nextInt();
            itemCost.put(itemName, cost);
        }
        int m = sc.nextInt();
        Map<String, Set<String>> itemPairs = new HashMap<>();
        for (int i = 0; i < m; i++) {
            String item1 = sc.next();
            String item2 = sc.next();
            if (!itemPairs.containsKey(item1)) {
                itemPairs.put(item1, new HashSet<>());
            }
            itemPairs.get(item1).add(item2);
            if (!itemPairs.containsKey(item2)) {
                itemPairs.put(item2, new HashSet<>());
            }
            itemPairs.get(item2).add(item1);
        }
        int minCost = Integer.MAX_VALUE;
        for (Map.Entry<String, Integer> entry : itemCost.entrySet()) {
            minCost = Math.min(minCost, entry.getValue());
        }
        Set<String> visited = new HashSet<>();
        Queue<String> queue = new LinkedList<>();
        for (Map.Entry<String, Integer> entry : itemCost.entrySet()) {
            if (visited.contains(entry.getKey())) {
                continue;
            }
            queue.add(entry.getKey());
            int cost = 0;
            while (!queue.isEmpty()) {
                String item = queue.poll();
                visited.add(item);
                cost += itemCost.get(item);
                if (itemPairs.containsKey(item)) {
                    for (String nextItem : itemPairs.get(item)) {
                        if (!visited.contains(nextItem)) {
                            queue.add(nextItem);
                        }
                    }
                }
            }
            minCost = Math.min(minCost, cost);
        }
        System.out.println(minCost);
    }
}
