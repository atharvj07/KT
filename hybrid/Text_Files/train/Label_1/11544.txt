#include <bits/stdc++.h>
using namespace std;
const int mxN = 201;
int n, k;
int a[mxN], dp[mxN][mxN][11];
int solve(int l, int r, int swp) {
    if (dp[l][r][swp] != -1) return dp[l][r][swp];
    if (l > r) return 0;
    int &ret = dp[l][r][swp];
    ret = a[l] + a[r];
    if (l != r) ret = max(ret, -a[l] + a[r] + solve(l + 1, r, swp));
    if (l != r) ret = max(ret, a[l] - a[r] + solve(l, r - 1, swp));
    for (int x = l + 2; x <= r - 1; x++) {
        if (swp) {
            ret = max(ret, a[l] + a[x] + solve(l + 1, x - 1, swp - 1) +
                              solve(x + 1, r, swp - 1));
        }
    }
    ret = max(ret, solve(l + 1, r, swp) + a[l]);
    ret = max(ret, solve(l, r - 1, swp) + a[r]);
    return ret;
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    memset(dp, -1, sizeof dp);
    cout << solve(1, n, k) << '\n';
    return 0;
}
