#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int arr[n];
        for(int i=0;i<n;i++) cin>>arr[i];
        bool check=1;
        for(int i=0;i<n;i++){
            if(arr[i]==1){
                if(i==n-1){
                    cout<<"NO\n";
                    check=0;
                    break;
                }
                if(arr[i+1]==0&&arr[i+2]==0){
                    cout<<"NO\n";
                    check=0;
                    break;
                }
            }
        }
        if(check){
            cout<<"YES\n";
            if(arr[n-1]==1){
                int x=-1;
                for(int i=n-1;i>=0;i--){
                    if(arr[i]==0){
                        x=i;
                        break;
                    }
                }
                cout<<1<<endl;
                cout<<x+1<<endl;
            }
            else cout<<0<<endl;
        }
    }
}
