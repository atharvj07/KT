n, q = map(int, input().split())
painters = [tuple(map(int, input().split())) for _ in range(q)]

# Initialize a list to keep track of how many painters cover each section
cover_count = [0] * (n + 1)

# Fill cover_count based on painters' coverage
for l, r in painters:
    for i in range(l, r + 1):
        cover_count[i] += 1

# Initialize the result and calculate the total sections painted by all painters
result = 0
total_painted = sum(1 for x in cover_count if x > 0)

# Iterate over all pairs of painters to simulate not hiring them
for i in range(q):
    for j in range(i + 1, q):
        skipped_coverage = set(range(painters[i][0], painters[i][1] + 1)) | set(range(painters[j][0], painters[j][1] + 1))
        skipped_painted = sum(1 for x in skipped_coverage if cover_count[x] == 1)
        # Update the result with the maximum sections painted
        result = max(result, total_painted - skipped_painted)

print(result)
