#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n + 1, vector<int>(n + 1));
  for (int i = 0; i < m; ++i) {
    int a, b, c;
    cin >> a >> b >> c;
    adj[a][b] = c;
    adj[b][a] = c;
  }
  int q;
  cin >> q;
  while (q--) {
    int u, v;
    cin >> u >> v;
    int cnt = 0;
    for (int i = 1; i <= n; ++i) {
      if (adj[u][i] && adj[i][v]) ++cnt;
    }
    cout << cnt << '\n';
  }
  return 0;
}
