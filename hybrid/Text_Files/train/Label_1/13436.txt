#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
ll F[300010];
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	ll n;
	cin >> n;
	ll arr[n + 5];
	for (int i = 0;i < n;i++) { cin >> arr[i];F[i] = 0; }
	sort(arr, arr + n);
	for (int i = 0;i < n;i++) {
		if (i != 0) {
			F[i] = F[i - 1];
			F[i] += (arr[i] - arr[i - 1]);
			F[i] %= mod;
		}
		if (i != n - 1) {
			F[i] = F[i - 1];
			F[i] += (arr[i + 1] - arr[i]);
			F[i] %= mod;
		}
	}
	ll ans = 0;
	for (int i = 0;i < n;i++) {
		ans += F[i];
		ans %= mod;
	}
	cout << ans << endl;
	return 0;
}
