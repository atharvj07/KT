#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<long long int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    double sum = 0;
    for (int i = 0; i < n; i++) {
      sum += (a[i] * (n - i)) / (double)(n * (i + 1));
    }
    cout << (sum == m ? "YES" : "NO") << endl;
  }
  return 0;
}
