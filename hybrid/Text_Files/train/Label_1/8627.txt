#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
#include <bits/stdc++.h>

using namespace std;
using namespace __gnu_pbds;
template <typename T>
using ordered_set = tree<T, null_type, less<T>, rb_tree_tag, tree_order_statistics_node_update>;

#define int long long
const int mod = 998244353;

int power(int a, int b) {
    int ans = 1;
    a %= mod;
    while (b) {
        if (b & 1) ans = ans * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return ans;
}

const int MAXN = 1e3 + 10;
vector<int> g[MAXN];

int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n);
    map<int, int> last;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (last[a[i]]) g[last[a[i]]].push_back(i);
        last[a[i]] = i;
    }
    vector<int> dp(n), pref(n);
    dp[n - 1] = 1;
    for (int i = n - 2; i >= 0; i--) {
        dp[i] = dp[i + 1];
        for (auto j : g[i]) dp[i] = (dp[i] + dp[j + 1]) % mod;
    }
    for (int i = n - 2; i >= 0; i--) {
        pref[i] = (pref[i + 1] + dp[i]) % mod;
    }
    cout << pref[0] << '\n';
    return 0;
}
