#include <bits/stdc++.h>
#define ll long long
#define fi first
#define se second
#define pb push_back
using namespace std;
const ll mod = 1e9 + 7;
const int mxN = 1e5+3;
struct pair_hash {
    template <class T1, class T2>
    std::size_t operator() (const std::pair<T1, T2> &pair) const {
        return std::hash<T1>()(pair.first) ^ std::hash<T2>()(pair.second);
    }
};
unordered_map<pair<string, int>, int, pair_hash> dp;
bool palindrome(string s) {
    int i = 0;
    int j = s.size() - 1;
    while(i < j) {
        if(s[i] != s[j]) return false;
        i++;
        j--;
    }
    return true;
}
bool check(string s, string t) {
    int dp[s.size() + 1][t.size() + 1];
    int n = s.size(), m = t.size();
    for(int i = 0; i <= n; i++) dp[i][m] = 1;
    for(int i = n - 1; i >= 0; i--) {
        for(int j = m - 1; j >= 0; j--) {
            if(s[i] == t[j]) {
                dp[i][j] = dp[i+1][j+1] + dp[i][j+1];
            } else {
                dp[i][j] = dp[i][j+1];
            }
        }
    }
    return dp[0][0] > 0;
}
int main() {
    ll n; cin >> n;
    vector<string> a(n);
    for(string& s: a) cin >> s;
    dp.clear();
    ll ans = 0;
    for(int i = 0; i < n; i++) {
        string w = a[i];
        dp[{w, i}] = 1;
        reverse(w.begin(), w.end());
        for(int j = 0; j+1 <= w.size(); j++) {
            string x = w.substr(0, j);
            string y = w.substr(j);
            ll left, right;
            if(check(x, w)) left = dp[{x, i}];
            else left = 0;
            if(check(y, w)) right = dp[{y, i}];
            else right = 0;
            ans += left * right;
        }
        for(int j = 0; j < n; j++) {
            if(i == j) continue;
            string x = a[j];
            ll left, right;
            if(check(w, x)) left = dp[{x, j}];
            else left = 0;
            if(check(x, w)) right = dp[{w, i}];
            else right = 0;
            ans += left * right;
        }
    }
    cout << ans << "\n";
}
