import java.util.*;
import java.io.*;
public class Main {
    static class Node {
        ArrayList<Node> children;
        Node parent;
        boolean vis;
        boolean leaf;

        public Node() {
            children = new ArrayList<>();
            vis = false;
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] arr = br.readLine().split(" ");
        int[] par = new int[n + 1];
        for (int i = 2; i <= n; i++) {
            par[i] = Integer.parseInt(arr[i - 2]);
        }
        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node();
        }
        for (int i = 2; i <= n; i++) {
            nodes[par[i]].children.add(nodes[i]);
            nodes[i].parent = nodes[par[i]];
        }
        Stack<Node> st = new Stack<>();
        st.push(nodes[1]);
        while (!st.empty()) {
            Node top = st.pop();
            if (top.children.isEmpty()) {
                top.leaf = true;
            }
            for (Node child : top.children) {
                if (!child.vis) {
                    child.vis = true;
                    st.push(child);
                }
            }
        }
        int[] dp = new int[n + 1];
        int ans = 0;
        for (int i = n; i >= 1; i--) {
            int cur = 0;
            int leafCnt = 0;
            for (Node child : nodes[i].children) {
                if (child.leaf) {
                    leafCnt++;
                } else {
                    cur = Math.max(cur, dp[child.parent.children.indexOf(child) + 1]);
                }
            }
            if (leafCnt > 0) {
                ans++;
            }
            if (ans > leafCnt) {
                cur = Math.max(cur, ans);
            }
            dp[i] = cur + 1;
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(dp[i] + " ");
        }
    }
}
