def solve():
    for _ in range(int(input())):
        n = int(input())
        w = list(map(int, input().split()))
        deg = [0] * n
        for _ in range(n - 1):
            u, v = map(int, input().split())
            deg[u - 1] += 1
            deg[v - 1] += 1
        extra = []
        for i in range(n):
            extra.extend([w[i]] * (deg[i] - 1))
        extra.sort(reverse=True)
        ans = [sum(w)]
        s = ans[0]
        for i in range(n - 2):
            s += extra[i]
            ans.append(s)
        print(' '.join(map(str, ans)))

solve()
