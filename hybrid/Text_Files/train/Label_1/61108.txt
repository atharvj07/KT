MOD = 1000000007

def power(a:int, b:int) -> int:
    ret = 1
    while b > 0:
        if b % 2 == 1:
            ret = (ret * a) % MOD
        a = (a * a) % MOD
        b //= 2
    return ret

def solve(k:int) -> int:
    if k == 1:
        return 6
    return ((power(6, k - 1) * 5) % MOD)

k = int(input())
print(solve(k))
