import sys
n=int(input())
p=[0]+list(map(int,input().split()))
l=[0]*n
r=[0]*n
for i in range(1,n):
    if p[i]<p[i-1]:
        l[i]=i-1
    else:
        l[i]=l[i-1]
for i in range(n-2,-1,-1):
    if p[i]>p[i+1]:
        r[i]=i+1
    else:
        r[i]=r[i+1]
ans=0
for i in range(1,n):
    if l[i]!=0 and r[i]!=0:
        ans+=1
print(ans)
