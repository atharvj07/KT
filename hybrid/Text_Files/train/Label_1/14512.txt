#include<iostream>
using namespace std;

int main() {
    int n, x[2010], y[2010], z[4020], ans;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> x[i];
        z[x[i]] = 1;
    }
    for (int i = 1; i <= n; i++) {
        cin >> y[i];
        z[y[i]] = 1;
    }
    for (int i = 1; i <= n; i++)
        if (z[x[i]] * z[y[i]] == 0) ans++;
    if (ans % 2 == 0) cout << "Karen\n";
    else cout << "Koyomi\n";
    return 0;
}
