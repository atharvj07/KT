#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[10];
  cin >> n;
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  sort(a, a + n);
  int sum = 0;
  for (int i = 0; i < n; ++i) { sum += a[i]; }
  for (int i = 0; i < n; ++i) {
    int sum1 = a[i], sum2 = sum - a[i];
    if (sum1 != sum2 && sum1 > 0 && sum2 > 0) {
      cout << 1 << ' ' << i + 1 << endl;
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
