#include <bitset>
#include <iostream>

int main() {
  int n;
  std::cin >> n;
  unsigned int mask = 0;
  for (int i = 0, x; i < 2; ++i) {
    int c;
    std::cin >> c;
    while (c--) {
      std::cin >> x;
      mask |= 1 << (x - 1);
    }
  }
  if (mask == (1 << n) - 1)
    std::cout << "I become the guy." << std::endl;
  else
    std::cout << "Oh, my keyboard!" << std::endl;
  return 0;
}
