n, m, d = map(int, input().split())
c = list(map(int, input().split()))

# Calculate total free spaces
total_free_spaces = n - sum(c)

# Initialize the output array with zeros
a = [0] * n

# Check if it's possible to reach the end
if m * d - 1 < total_free_spaces:
    print("NO")
else:
    print("YES")
    pos = 0
    for i in range(m):
        # Calculate maximum shift for current platform
        shift = min(d - 1, total_free_spaces)
        pos += shift
        # Place the platform
        for j in range(c[i]):
            a[pos + j] = i + 1
        pos += c[i]
        # Subtract the used free spaces
        total_free_spaces -= shift

    print(*a)
