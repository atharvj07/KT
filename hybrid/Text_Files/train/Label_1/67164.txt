def solve(n, pairs):
    count = [0] * (n + 1)
    for a, b in pairs:
        count[a] += 1
        count[b] += 1
    one_count = sum(c == 1 for c in count)
    if one_count != 2:
        return ["NO"]
    edges = []
    edges_dict = {}
    for a, b in pairs:
        if a not in edges_dict:
            edges_dict[a] = []
        if b not in edges_dict:
            edges_dict[b] = []
        edges_dict[a].append(b)
        edges_dict[b].append(a)
    for i in range(1, n + 1):
        if len(edges_dict[i]) == 1:
            start = i
            break
    visited = set()
    while True:
        visited.add(start)
        next_node = None
        for node in edges_dict[start]:
            if node not in visited:
                next_node = node
                edges.append((start, node))
                break
        if next_node is None:
            break
        start = next_node
    if len(edges) != n - 1:
        return ["NO"]
    return ["YES"] + [f"{x[0]} {x[1]}" for x in edges]

n = int(input().strip())
pairs = [list(map(int, input().strip().split())) for _ in range(n - 1)]

result = solve(n, pairs)
for line in result:
    print(line)
