import sys

t = int(input())
for _ in range(t):
    _ = int(input())
    a = list(map(int, input().split()))
    b, c, ans = [], [], 0
    for v in a:
        if c and c[-1] > v:
            ans += 1
        elif b and b[0] > v:
            c.append(b.pop(0))
            ans += 1
        c.append(v)
        b.extend(c)
        c = []
    print(ans)
