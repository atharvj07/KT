int solve(vector<int> a, int n) {
	int ans = -1;
	int g = 0;
	for(int i = 0; i < n; i++) {
		g = __gcd(a[i], g);
	}
	if(g > 1) return 0;
	for(int i = 0; i < n; i++) {
		if(a[i]) return 1;
	}
	return -1;
}

	int n;
	cin >> n;
	vector<int> a(n);
	for(int &x : a) cin >> x;
	cout << solve(a, n)  << endl;
	return 0;
}
