import sys

input = sys.stdin.readline

d = int(input())
n, m = map(int, input().split())
sofas = [tuple(map(int, input().split())) for _ in range(d)]
l, r, t, b = map(int, input().split())

for i in range(d):
    x1, y1, x2, y2 = sofas[i]
    if (x1 == x2 and y1 < y2 and r == i or
        x1 < x2 and y1 == y2 and b == i or
        x1 == x2 and y1 > y2 and l == i or
        x1 > x2 and y1 == y2 and t == i):
        print(i + 1)
        exit()

print(-1)
