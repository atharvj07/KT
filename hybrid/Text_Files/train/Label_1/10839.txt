#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    vector<int> freq(26, 0);
    for (char c : s) freq[c - 'a']++;
    int k = 0;
    for (int f : freq) k += f / 2;
    cout << k << endl;
  }
  return 0;
}
