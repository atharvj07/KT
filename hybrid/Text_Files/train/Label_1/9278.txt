#include <bits/stdc++.h>

using namespace std;

void initial(vector<vector<int> > &a, const vector<int> &radius) {
  for (int i = 0; i < radius.size(); ++i) {
    a[i].push_back(radius[i]);
  }
}

bool is_possible(vector<vector<int> > &a) {
  int n = a.size();
  stack<int> ans;
  for (int i = 0; i < n; ++i) {
    while (a[i].size()) {
      if (ans.empty())
        ans.push(a[i].back());
      else if (a[i].back() > ans.top()) {
        ans.push(a[i].back());
        a[i].pop_back();
      } else {
        break;
      }
    }
  }
  return ans.size() == n;
}

int main() {
  int n;
  cin >> n;
  vector<int> radius(n);
  for (int i = 0; i < n; ++i) {
    cin >> radius[i];
  }
  vector<vector<int> > a(n);
  initial(a, radius);
  if (is_possible(a))
    cout << "YES\n";
  else
    cout << "NO\n";
  return 0;
}
