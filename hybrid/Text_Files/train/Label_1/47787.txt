import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] arr = sc.next().toCharArray();
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 'X') count++;
        }
        int need = n / 2;
        int ans = Math.abs(need - count);
        System.out.println(ans);
        if (count >= need) {
            for (int i = 0; i < n; i++) {
                if (arr[i] == 'X' && count > need) {
                    arr[i] = 'x';
                    count--;
                }
            }
        } else {
            for (int i = 0; i < n; i++) {
                if (arr[i] == 'x' && count < need) {
                    arr[i] = 'X';
                    count++;
                }
            }
        }
        System.out.println(String.valueOf(arr));
    }
}
