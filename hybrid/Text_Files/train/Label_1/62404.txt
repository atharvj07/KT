n, q = map(int, input().split())
s = list(input())
e, l = 0, n
last = [0] * 3
for i in range(q):
    pos, ch = input().split()
    pos = int(pos) - 1
    if s[pos] != ch:
        s[pos] = ch
        l = min(l, pos)
        e = max(e, pos)
    ans = 0
    for ch in [0, 1, 2]:
        if e - last[ch] >= 2: ans += 1
        last[ch] = max(last[ch], pos)
    print(ans)
