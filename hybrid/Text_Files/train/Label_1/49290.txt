import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        boolean good = false;
        for (int i = 0; i < n - 4; i++) {
            if (s.charAt(i) == '*' && s.charAt(i + 1) == '*' && s.charAt(i + 2) == '*' && s.charAt(i + 3) == '*' && s.charAt(i + 4) == '*') {
                good = true;
                break;
            }
        }
        System.out.println(good ? "yes" : "no");
    }
}
