n,m,k=map(int,input().split())
c=list(map(int,input().split()))
p=[]
for i in range(n):
    p.append(list(map(int,input().split())))
inf=1e9
dp=[[[-1]*k for j in range(m+1)] for i in range(n+1)]
def go(x,y,g):
    if(x==n):
        if(g==k):return 0
        else:return inf
    if(dp[x][y][g]!=-1):return dp[x][y][g]
    dp[x][y][g]=inf
    if(c[x]!=0 and c[x]!=y):dp[x][y][g]=go(x+1,c[x],g)
    else:
        for j in range(1,m+1):
            if(j!=y):
                dp[x][y][g]=min(dp[x][y][g],go(x+1,j,g+1)+p[x][j-1])
    return dp[x][y][g]
if(go(0,0,0)==inf):print(-1)
else:print(go(0,0,0))
