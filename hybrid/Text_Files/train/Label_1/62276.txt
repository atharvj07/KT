from heapq import *

n, m, k = map(int, input().split())
g = [[] for _ in range(n+1)]
for _ in range(m):
    a, b, c = map(int, input().split())
    g[a].append((b,c))
    g[b].append((a,c))

d = [[float('inf') for _ in range(n+1)] for _ in range(n+1)]
vis = [False for _ in range(n+1)]

hq = []
d[1][1] = 0
heappush(hq, (0, 1))
ans = -1

while hq:
    dd, i = heappop(hq)
    if len(hq)+1 == k:
        ans = dd
        break
    
    if vis[i]: continue
    vis[i] = True
    for j, w in g[i]:
        if d[i][j] > dd + w:
            d[i][j] = d[j][i] = dd + w
            heappush(hq, (dd + w, j))

print(ans)
