n = int(input())
board = []
for i in range(n):
    board.append(list(input()))
ans = 'YES'
for i in range(n):
    for j in range(n):
        cnt = 0
        if i > 0 and board[i - 1][j] == 'o':
            cnt += 1
        if i < n - 1 and board[i + 1][j] == 'o':
            cnt += 1
        if j > 0 and board[i][j - 1] == 'o':
            cnt += 1
        if j < n - 1 and board[i][j + 1] == 'o':
            cnt += 1
        if cnt % 2 == 1:
            ans = 'NO'
            break
    if ans == 'NO':
        break
print(ans)
