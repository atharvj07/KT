#include <bits/stdc++.h>
using namespace std;

int main() {
  int s, x1, x2, t1, t2, p, d;
  cin >> s >> x1 >> x2 >> t1 >> t2 >> p >> d;
  int l = abs(x1 - p), l1 = s - p, l2 = x2 - p;
  if (d == 1) {
    l = s - l;
    l1 = p;
    l2 = s - x2;
  }
  long long ans = l / t1;
  ans += l % t1 != 0;
  if (l2 <= l1) {
    ans += l2 / t2;
    l1 -= l2;
  } else {
    ans += l1 / t2;
    l2 -= l1;
  }
  ans += l2 / t1;
  ans += l2 % t1 != 0;
  cout << ans;
}
