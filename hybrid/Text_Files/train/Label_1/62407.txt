n = int(input())
days = list(map(int, input().split()))

rest_days = 0
prev_activity = 0  # 1 for contest, 2 for sport

for day in days:
    if day == 0:
        rest_days += 1
        prev_activity = 0
    elif day == 3:
        if prev_activity == 1:
            prev_activity = 2
        elif prev_activity == 2:
            prev_activity = 1
        else:
            prev_activity = 3
    else:
        if day == prev_activity:
            rest_days += 1
            prev_activity = 0
        else:
            prev_activity = day

print(rest_days)
