#include <iostream>
#include <cstdio>
#include <string>
#include <cstring>
#include <queue>
#include <vector>
using namespace std;
#define read(x) scanf("%d", &x)
#define readc(x) scanf("%c", &x)
#define rep(i, a, b) for(int i = a; i <= b; i++)

const int maxn = 1000;
int n, m, k; 
char maze[maxn][maxn + 1];
bool vis[maxn][maxn][4];
int stx, sty;
int dir[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
string path;
bool flag = false;

void DFS(int x, int y, int step, int d) {
    if (step == k) {
        if (x == stx && y == sty) {
            path += '0';
            flag = true;
            return;
        } else return;
    }
    int nowx, nowy;
    rep(i, 0, 3) {
        if (!vis[x][y][i] && maze[x + dir[i][0]][y + dir[i][1]] != '*') {
            nowx = x + dir[i][0];
            nowy = y + dir[i][1];
            vis[nowx][nowy][(i + 2) % 4] = true;
            path += "DLUR"[i];
            DFS(nowx, nowy, step + 1, i);
            if (flag) return;
            vis[nowx][nowy][(i + 2) % 4] = false;
            path.pop_back();
        }
    }
}

int main() {
    read(n); read(m); read(k);
    rep(i, 1, n) {
        scanf("%s", maze[i] + 1);
        rep(j, 1, m) {
            if (maze[i][j] == 'X') {
                stx = i;
                sty = j;
            }
        }
    }
    memset(vis, false, sizeof(vis));
    DFS(stx, sty, 0, 0);
    if (flag) {
        rep(i, 0, k - 1) {
            if (path[i] == '0') cout << path[i - 1];
            else cout << path[i];
        }
    } else puts("IMPOSSIBLE");
    return 0;
}
