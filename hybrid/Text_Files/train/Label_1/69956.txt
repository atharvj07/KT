n = int(input())
l, r = map(list, (map(int, input().split()), map(int, input().split())))
chk = [0] * n
f = 0
for i in range(n):
    if l[i] + 1 + r[i] > n or (chk[l[i]] == 1 and r[i] != 0) or (chk[l[i] + 1 + r[i]] == 1 and r[i] != n - l[i] - 1):
        f = 1
        break
    chk[l[i] + 1 + r[i]] = 1
if f:
    print("NO")
else:
    print("YES")
    for i in range(1, n + 1):
        print(chk[i], end=' ')
