import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k1 = in.nextInt(), k2 = in.nextInt(), k3 = in.nextInt();
        int n = k1 + k2 + k3;
        int[] a1 = new int[k1], a2 = new int[k2], a3 = new int[k3];
        for (int i = 0; i < k1; i++)
            a1[i] = in.nextInt();
        for (int i = 0; i < k2; i++)
            a2[i] = in.nextInt();
        for (int i = 0; i < k3; i++)
            a3[i] = in.nextInt();
        int[] b = new int[n];
        for (int i = 0; i < k1; i++)
            b[a1[i] - 1] = 1;
        for (int i = 0; i < k2; i++)
            b[a2[i] - 1] = 2;
        for (int i = 0; i < k3; i++)
            b[a3[i] - 1] = 3;
        int ans = 0;
        for (int i = 0, j = 0, k = n - 1; i < n; i++) {
            if (b[i] == 1)
                j++;
            else if (b[i] == 3)
                k--;
            else if (b[i] == 2 && j != k1 && k != n - k3)
                ans++;
        }
        System.out.println(ans);
    }
}
