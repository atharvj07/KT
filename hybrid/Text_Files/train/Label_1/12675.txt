#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  sort(v.begin(), v.end());
  int mx = 0;
  for (int i = 0; i < n; i++) {
    if (mx >= m) break;
    mx = max(mx, min(m, v[i].second));
    if (v[i].first <= mx) mx = max(mx, v[i].second);
  }
  cout << (mx >= m ? "YES" : "NO");
  return 0;
}
