import heapq
inp = [int(x) for x in input().split()]
n, k = inp[0], inp[1]

videos = sorted([list(map(int, input().split())) for i in range(n)])

servers = [0] * k

for vidTime, vidLen in videos:
    srvTime = heapq.heappop(servers)
    waitTime = max(srvTime, vidTime)
    heapq.heappush(servers, waitTime + vidLen)
    print(waitTime + vidLen)
