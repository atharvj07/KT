#include<bits/stdc++.h>
using namespace std;

int main() {
    int t, x;
    cin >> t;
    while (t--) {
        cin >> x;
        if (x < 10) {
            cout << x << endl;
            continue;
        }
        int a[10] = {0};
        a[0] = 9;
        x -= 9;
        while (x) {
            a[x % 10]++;
            x /= 10;
            if (a[x % 10] > 0) {
                cout << -1 << endl;
                continue;
            }
        }
        for (int i = 1; i < 10; i++) {
            for (int j = 0; j < a[i]; j++)
                cout << i;
        }
        cout << endl;
    }
}
