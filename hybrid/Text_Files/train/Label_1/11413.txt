#include <cstdio>
#include <algorithm>

constexpr int MAXN = 5e5;

int n, k;
int a[MAXN + 1];
int cnt[1000001];
int ans = 0, curl = 0, curr = 0;

int main() {
  scanf("%d%d", &n, &k);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  int l = 1;
  int r = 1;
  cnt[a[1]] = 1;
  curr = 1;
  while (r <= n) {
    if (cnt[a[r]] < k) {
      ans = std::max(ans, r - l + 1);
      cnt[a[r]]++;
      if (curr < k) curr++;
      r++;
    } else {
      l++;
      cnt[a[l - 1]]--;
      if (cnt[a[l - 1]] == 0) curr--;
      while (curr < k) {
        if (cnt[a[r]] == 1) curr++;
        cnt[a[r]]++;
        r++;
      }
    }
  }
  printf("%d %d", l, l + ans - 1);
  return 0;
}
