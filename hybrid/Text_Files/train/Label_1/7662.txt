#include <iostream>
#include <vector>
using namespace std;

const int MAX_DIG = 18;
int a[MAX_DIG], b[MAX_DIG], s[MAX_DIG];

void solve() {
  int n_a, n_b, n_s;
  cin >> n_a >> n_s;
  fill(s, s + MAX_DIG, 0);
  fill(a, a + MAX_DIG, 0);
  fill(b, b + MAX_DIG, 0);
  int d = MAX_DIG;
  bool ok = false;
  while (n_a > 0 || n_s > 0) {
    a[--d] = n_a % 10;
    s[--d] = n_s % 10;
    n_a /= 10;
    n_s /= 10;
  }
  n_b = d;
  for (int i = d; i < MAX_DIG; ++i) {
    b[i] = s[i] + 10 - a[i];
    s[i + 1] += b[i] / 10;
    b[i] %= 10;
    if (b[i] > 0) ok = true;
  }
  if (not ok || b[n_b] > 0) {
    cout << -1 << endl;
    return;
  }
  for (int i = n_b; i < MAX_DIG; ++i) {
    if (b[i] > 0) --n_b;
    else break;
  }
  for (int i = n_b; i < MAX_DIG; ++i) cout << b[i];
  cout << endl;
}

int main() {
  int t;
  cin >> t;
  while (t--) solve();
}
