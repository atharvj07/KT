import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    private static int N, K;
    private static char[][] layout;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        while (true) {
            String[] line = in.readLine().split(" ");
            N = Integer.parseInt(line[0]);
            K = Integer.parseInt(line[1]);
            if (N == 0 && K == 0) break;

            layout = new char[N][N];

            if (K >= Math.pow(2, N * N)) {
                System.out.println("No");
            } else {
                int count = 0;
                Queue<int[]> queue = new LinkedList<>();
                for (int i = 0; i < N; i++) {
                    for (int j = 0; j < N; j++) {
                        layout[i][j] = '.';
                        queue.add(new int[]{i, j});
                        while (!queue.isEmpty()) {
                            count++;
                            int[] curr = queue.poll();
                            if (count > K) {
                                layout[i][j] = 'E';
                                break;
                            }
                            checkRoom(curr[0], curr[1], queue);
                        }
                        if (count > K) break;
                    }
                    if (count > K) break;
                }

                if (count <= K) {
                    for (char[] row : layout) {
                        System.out.println(Arrays.toString(row).replace("[", "").replace("]", "").replace(",", ""));
                    }
                } else {
                    System.out.println("No");
                }
            }
            System.out.println();
        }
    }

    private static void checkRoom(int i, int j, Queue<int[]> queue) {
        if (i > 0 && layout[i - 1][j] == '.') {
            layout[i - 1][j] = layout[i][j];
            queue.add(new int[]{i - 1, j});
        }
        if (i < N - 1 && layout[i + 1][j] == '.') {
            layout[i + 1][j] = layout[i][j];
            queue.add(new int[]{i + 1, j});
        }
        if (j > 0 && layout[i][j - 1] == '.') {
            layout[i][j - 1] = layout[i][j];
            queue.add(new int[]{i, j - 1});
        }
        if (j < N - 1 && layout[i][j + 1] == '.') {
            layout[i][j + 1] = layout[i][j];
            queue.add(new int[]{i, j + 1});
        }
    }
}
