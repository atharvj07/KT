import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int maxb = sc.nextInt();
        long t = sc.nextInt();
        int[] b = new int[k * n];
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                b[i * n + j] = sc.nextInt();
            }
        }
        long[] dp = new long[k * n];
        int[] ans = new int[k];
        for (int i = 0; i < k * n; i++) {
            dp[i] = 1;
            for (int j = 0; j < i; j++) {
                if (b[i] % maxb > b[j] % maxb && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                }
            }
            if (dp[i] > ans[i / n]) {
                ans[i / n] = (int) dp[i];
            }
        }
        for (int i : ans) {
            System.out.println(i);
        }
    }
}
