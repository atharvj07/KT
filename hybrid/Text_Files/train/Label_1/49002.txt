import java.util.*;
import java.io.*;
class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.nextLine();
    boolean f = true;
    for (char c : s.toCharArray()) {
      if (c != s.charAt(0)) {
        f = false;
      }
    }
    if (f) System.out.println("Yes");
    else System.out.println("No");
  }
}
