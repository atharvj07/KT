import java.util.*;

class Main {

    static final int INF = (int) 1e9;
    static final int MOD = (int) 1e9 + 7;

    void solve() {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] s = sc.next().toCharArray();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt() - 1;
            b[i] = sc.nextInt() - 1;
        }
        int[][] g = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(g[i], INF);
        }
        for (int i = 0; i < n - 1; i++) {
            g[a[i]][b[i]] = g[b[i]][a[i]] = 1;
        }
        floydWarshall(g, n);
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (g[i][j] < INF) {
                    c[i] += g[i][j] % 2;
                }
            }
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '1') {
                if (c[i] % 2 == 1) {
                    System.out.println(-1);
                    return;
                } else {
                    sum += c[i] / 2;
                }
            }
        }
        System.out.println(sum);
    }

    void floydWarshall(int[][] g, int n) {
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (g[i][k] + g[k][j] < g[i][j]) {
                        g[i][j] = g[i][k] + g[k][j];
                    }
                }
            }
        }
    }

    public static void main(String[] args) {
        new Main().solve();
    }
}
