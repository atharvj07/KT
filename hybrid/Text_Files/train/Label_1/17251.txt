#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
vector<pair<ll, ll>> v, ans;
int main() {
  int n, m;
  cin >> n >> m;
  v.resize(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first; }
  sort(v.rbegin(), v.rend());
  for (int i = 0, cur = 0; i < n, cur + v[i].first <= m; i++) {
    ans.push_back(v[i]);
    cur += v[i].first;
  }
  cout << ans.size() << endl;
  for (auto x : ans) cout << x.first << " ";
  return 0;
}
