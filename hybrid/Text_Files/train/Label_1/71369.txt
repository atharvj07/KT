def solve():
    mod = 998244353
    n, k = [int(i) for i in input().split()]
    intervals = [(int(i) - 1, int(j) - 1) for j, i in zip(*[iter(input().split()) for _ in range(n)])]
    dp = [[0] * (k + 1) for _ in range(n + 1)]
    dp[0][0] = 1
    for i in range(n):
        l, r = intervals[i]
        for j in range(k, 0, -1):
            dp[i + 1][j] = (dp[i + 1][j] + dp[i][j - 1]) % mod
        for j in range(l, r + 1):
            dp[i + 1][j] = (dp[i + 1][j] + dp[i][j]) % mod
    return dp[n][k]

print(solve())
