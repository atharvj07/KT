#include<bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int n; cin >> n;
    int a[n + 1]; for (int i = 0; i < n; i++) cin >> a[i];
    bool flag = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] % 2 == 1) {
            cout << a[i] << endl;
            return 0;
        }
    }
    int even = 0, odd = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] % 2 == 1) odd += a[i];
        else even += a[i];
    }
    if (odd != 0) cout << odd << endl;
    else cout << even << endl;
    return 0;
}
