import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long l1 = in.nextLong();
        long r1 = in.nextLong();
        long l2 = in.nextLong();
        long r2 = in.nextLong();
        long n = (30L + 1) * 36 / 2;
        long len1 = Math.min(r1 - l1 + 1, n - l1 + 1);
        long len2 = Math.min(r2 - l2 + 1, n - l2 + 1);
        if (l1 > n || l2 > n || (l2 <= l1 && l1 + len1 - 1 < l2) || (l1 <= l2 && l2 + len2 - 1 < l1)) {
            System.out.println(0);
            return;
        }
        long ans = 0;
        for (long i = 0; i < len1; i++) {
            for (long j = 0; j < len2; j++) {
                if (i + l1 - 1 < l2 || j + l2 - 1 < l1 || (i + l1 - 1 < l2 && j + l2 - 1 < l1)) {
                    if (j + l2 - 1 < l1) {
                        ans = Math.max(ans, i);
                    }
                } else {
                    long lcp = Math.min(l1 + i - 1, l2 + j - 1);
                    for (long k = 0; k < lcp; k++) {
                        if (i + k < len1 && j + k < len2 && ((l1 + i + k - 1) / 36 == (l2 + j + k - 1) / 36 && (l1 + i + k - 1) % 36 == (l2 + j + k - 1) % 36)) {
                            ans = Math.max(ans, i + k + 1);
                        } else {
                            break;
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
