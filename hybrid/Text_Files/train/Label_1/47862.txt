import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static final int MOD = 998244353;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        Arrays.sort(a);

        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (long) (a[i] + a[2 * n - 1 - i]) * (a[i] + a[2 * n - 1 - i]) % MOD;
            ans %= MOD;
        }

        System.out.println(ans);
    }
}
