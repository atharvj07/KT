#include<bits/stdc++.h>
using namespace std;
#define ll long long

int main(){
    int n; cin>>n;
    vector<ll>a(n),b;
    ll sum=0;
    for(auto &x:a) cin>>x,sum+=x;
    if(sum%n){
        cout<<-1;
        return 0;
    }
    ll target=sum/n;
    for(auto x:a){
        if(x!=target){
            b.push_back(x);
        }
    }
    b.push_back(target);
    for(auto x:b) cout<<x<<' ';
}
