#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, cnt = 0;
    cin >> n;
    vector<int> a(n), b(n), c(1001);
    for (int i = 0; i < n; i++) {
        cin >> a[i] >> b[i];
        c[a[i]] = 1;
    }
    for (int i = 0; i < n; i++) {
        if (!c[b[i]]) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
