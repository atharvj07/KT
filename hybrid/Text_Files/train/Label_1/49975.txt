import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long x = in.nextLong();
        long y = in.nextLong();
        if (x >= y) System.out.println(n * x);
        else System.out.println((n + 1) / 2 * y + (n - 1) / 2 * x);
    }
}
