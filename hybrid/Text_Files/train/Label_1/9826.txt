#include <cstdio>
#include <iostream>
#include <stack>
#include <vector>
using namespace std;

const int MAXN = 1e5 + 5;
typedef long long ll;

int n, q;
ll width;
int query, x, y;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n >> q;
    width = n;
    stack<ll> W;
    W.push(width);
    while (q--) {
        cin >> query;
        if (query == 1) {
            cin >> x;
            W.push(x);
            width = x;
        } else if (query == 2) {
            cin >> x >> y;
            ll ans = 0;
            while (!W.empty() && W.top() != width) {
                ans += W.top();
                W.pop();
            }
            W.pop();  // remove current width
            while (!W.empty()) {
                W.pop();
            }
            ans += y - x;
            W.push(ans);
            width = ans;
            cout << ans << endl;
        }
    }
    return 0;
}
