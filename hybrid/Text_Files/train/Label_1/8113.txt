#include <vector>
#include <iostream>
#include <math.h>
#define ll long long
using namespace std;
vector<ll> pow10;
ll mod = 1e9 + 7;
ll fastPow(ll a, ll b) {
  ll ret = 1;
  while (b > 0) {
    if (b & 1) ret = (ret * a) % mod;
    a = (a * a) % mod;
    b >>= 1;
  }
  return ret;
}
int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(n / k), b(n / k);
  for (ll i = 0; i < n / k; i++) cin >> a[i];
  for (ll i = 0; i < n / k; i++) cin >> b[i];
  pow10.push_back(1);
  for (ll i = 0; i < k; i++) pow10.push_back(pow10[i] * 10 % mod);
  ll ans = 0;
  for (ll i = 0; i < (1 << k); i++) {
    vector<bool> ap(10), bp(10);
    ll num = 0;
    bool fail = false;
    for (ll j = 0; j < k; j++) {
      ll c = (i >> j) & 1;
      if (c) {
        if (bp[b[j / k]]) {
          fail = true;
          break;
        }
        bp[b[j / k]] = true;
      } else {
        if (ap[j / k]) {
          fail = true;
          break;
        }
        num = (num + c * pow10[j]) % a[j / k];
        ap[j / k] = true;
      }
    }
    if (fail) continue;
    if (num == 0) ans++;
  }
  cout << ans << endl;
  return 0;
}
