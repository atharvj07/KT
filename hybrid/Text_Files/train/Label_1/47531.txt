import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long r = 0, b = 0;
            String rs = sc.next();
            String bs = sc.next();
            for (int i = 0; i < n; i++) {
                r = r * 10 + (rs.charAt(i) - '0');
                b = b * 10 + (bs.charAt(i) - '0');
            }
            if (r > b) {
                System.out.println("RED");
            } else if (r < b) {
                System.out.println("BLUE");
            } else {
                System.out.println("EQUAL");
            }
        }
    }
}
