#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

double dist(pair<int, int> p1, pair<int, int> p2) {
    return sqrt(pow(p1.first - p2.first, 2) + pow(p1.second - p2.second, 2));
}

int main() {
    int x, y;
    cin >> x >> y;
    int n;
    cin >> n;
    pair<int, int> start = {x, y};
    double ans = 1e9;
    while (n--) {
        int a, b, v;
        cin >> a >> b >> v;
        ans = min(ans, dist(start, {a, b}) / v);
    }
    cout << ans << endl;
}
