from datetime import datetime, timedelta

def min_jury_size(n, olympiads):
    calendar = {}
    for olympiad in olympiads:
        m, d, p, t = olympiad
        olympiad_date = datetime(2013, m, d)
        start_date = olympiad_date - timedelta(days=t)
        for day in range(t):
            current_day = start_date + timedelta(days=day)
            if current_day not in calendar:
                calendar[current_day] = p
            else:
                calendar[current_day] += p
    return max(calendar.values()) if calendar else 0

n = int(input())
olympiads = [list(map(int, input().split())) for _ in range(n)]
print(min_jury_size(n, olympiads))
