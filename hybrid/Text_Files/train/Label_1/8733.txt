#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 26;
int n, c[maxn];
ll sum[1 << maxn];
bool flag;
void dfs(int dep, int mask, ll now) {
  sum[mask] = now;
  if (dep == n - 1) {
    if (now == n) flag = true;
    return;
  }
  for (int i = 1; i <= n; i++) {
    if (!(mask & (1 << i)) && now + c[i] <= n && sum[mask | (1 << i)] >= now + c[i]) {
      dfs(dep + 1, mask | (1 << i), now + c[i]);
      if (flag) return;
    }
  }
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> c[i];
  for (int i = 1; i < (1 << n); i++) sum[i] = n + 1;
  ll now = 0, mask = 0;
  dfs(0, mask, now);
  if (flag)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
