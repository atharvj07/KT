import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int[] b = new int[n * m + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                b[a[i][j]] = i * m + j;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n * m; i++) {
            if (b[i] / m == b[i - 1] / m) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
