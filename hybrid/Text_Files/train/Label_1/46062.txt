import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long k = in.nextLong();
        long n = in.nextLong();
        int cnt = 0;
        while (n > 0) {
            long d = n % 10;
            n /= 10;
            if (d < k) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
