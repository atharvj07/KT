import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // number of frogs
        int m = scanner.nextInt(); // number of mosquitoes
        int[] frogsX = new int[n]; // x-coordinates of frogs
        int[] frogsT = new int[n]; // initial tongue lengths of frogs
        for (int i = 0; i < n; i++) {
            frogsX[i] = scanner.nextInt();
            frogsT[i] = scanner.nextInt();
        }
        int[][] mosquitoes = new int[m][2]; // x-coordinates and sizes of mosquitoes
        int[] mosquitoesEaten = new int[n]; // number of mosquitoes eaten by each frog
        for (int i = 0; i < m; i++) {
            mosquitoes[i][0] = scanner.nextInt();
            mosquitoes[i][1] = scanner.nextInt();
        }

        Arrays.sort(frogsX);
        Arrays.sort(mosquitoes, (a, b) -> (a[0] - b[0]));
        int lastMosquito = 0;
        for (int i = 0; i < frogsX.length; i++) {
            int currentFrogX = frogsX[i];
            int currentFrogT = frogsT[i];
            int mosquitoesEatenByCurrentFrog = 0;
            for (int j = lastMosquito; j < mosquitoes.length; j++) {
                int currentMosquitoX = mosquitoes[j][0];
                int currentMosquitoSize = mosquitoes[j][1];
                if (currentFrogX <= currentMosquitoX && currentFrogT >= currentMosquitoX - currentFrogX) {
                    currentFrogT += currentMosquitoSize;
                    mosquitoesEatenByCurrentFrog++;
                    if (currentMosquitoSize >= currentFrogX - currentFrogT) {
                        break;
                    }
                    lastMosquito = j + 1;
                }
            }
            mosquitoesEaten[i] = mosquitoesEatenByCurrentFrog;
        }

        for (int i = 0; i < frogsX.length; i++) {
            System.out.println(mosquitoesEaten[i] + " " + frogsT[i]);
        }
    }
}
