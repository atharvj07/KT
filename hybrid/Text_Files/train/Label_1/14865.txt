#include<bits/stdc++.h>
using namespace std;
const int MN = 200005;
int A[MN], B[MN], preA[MN], preB[MN], sufA[MN], sufB[MN];
int32_t main(){
    int N; cin >> N;
    for(int i = 1; i <= N; i++) cin >> A[i];
    for(int i = 1; i <= N; i++) cin >> B[i];
    stack<int> st;
    for(int i = N; i >= 1; i--){
        while(!st.empty() && B[st.top()] > B[i]) st.pop();
        if(st.empty()) sufB[i] = N+1;
        else sufB[i] = st.top();
        st.push(i);
    }
    st = stack<int>();
    for(int i = N; i >= 1; i--){
        while(!st.empty() && A[st.top()] < A[i]) st.pop();
        if(st.empty()) sufA[i] = N+1;
        else sufA[i] = st.top();
        st.push(i);
    }
    while(!st.empty()) st.pop();
    for(int i = 1; i <= N; i++){
        while(!st.empty() && B[st.top()] < B[i]) st.pop();
        if(st.empty()) preB[i] = 0;
        else preB[i] = st.top();
        st.push(i);
    }
    st = stack<int>();
    for(int i = 1; i <= N; i++){
        while(!st.empty() && A[st.top()] > A[i]) st.pop();
        if(st.empty()) preA[i] = 0;
        else preA[i] = st.top();
        st.push(i);
    }
    int ans = 0;
    for(int i = 1; i <= N; i++){
        int l = preA[i], r = sufA[i];
        if(l < i && i < r && B[i] >= A[l] && B[i] <= A[r]) ans++;
        l = preB[i], r = sufB[i];
        if(l < i && i < r && A[i] <= B[l] && A[i] >= B[r]) ans++;
    }
    cout << ans << '\n';
}
