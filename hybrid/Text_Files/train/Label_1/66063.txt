a,b,c=map(int,input().split())
#bin search
low=1;high=10**9
while low<=high:
    mid=(low+high)//2
    if mid+b*sum([int(j) for j in list(str(mid))])**a==c:
        low=mid+1
    else:
        high=mid-1
print(low-1)
for i in range(1,low):
    if i+b*sum([int(j) for j in list(str(i))])**a==c:
        print(i)
