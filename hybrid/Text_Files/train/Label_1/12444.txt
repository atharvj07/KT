#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  int n, m;
  cin >> n >> m;
  vector<ll> a(n), c(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  for (int i = 0; i < n; i++) { b[i] = a[i]; }
  while (m--) {
    int t, d;
    cin >> t >> d;
    t--;
    ll ans = 0;
    while (d--) {
      if (b[t]) {
        ans += c[t];
        b[t]--;
      } else {
        int mn = 1e9, mni;
        for (int i = 0; i < n; i++) {
          if (b[i] && c[i] < mn) {
            mn = c[i];
            mni = i;
          }
        }
        if (mn != 1e9) {
          ans += mn;
          b[mni]--;
        } else {
          ans = -1;
          break;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
