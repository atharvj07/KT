#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
int n,m,k,ans=0,vis[105][105],par[1000005],a[105][105],rt,cnt,chk[1000005],siz,cc=0;
vector<int>v,mp[1000005];
char ch[105][105];
int dx[]={0,0,1,-1};
int dy[]={1,-1,0,0};
int findpar(int x)
{
    if(par[x]==x)return x;
    return par[x]=findpar(par[x]);
}
void makepar(int x,int y)
{
    int u=x,v=y;
    x=findpar(x);
    y=findpar(y);
    if(x!=y)
    {
        par[y]=x;
        v=x;
    }
    mp[u].push_back(v);
    mp[v].push_back(u);
    return;
}
void dfs(int x,int y)
{
    cc++;
    vis[x][y]=1;
    for(int j=0; j<4; j++)
    {
        int xx=x+dx[j];
        int yy=y+dy[j];
        if(xx>=1 && xx<=n && yy>=1 && yy<=m)
        {
            int aa=a[x][y];
            int bb=a[xx][yy];
            if(vis[xx][yy]==0 && aa==bb)
            {
                makepar(rt,cc);
                dfs(xx,yy);
            }
        }
    }
    return;
}
void clr()
{
    for(int i=0; i<=cc; i++)
    {
        vis[i/m+1][i%m+1]=0;
        par[i]=i;
        mp[i].clear();
    }
}
int32_t main()
{
    cin>>n>>m>>k;
    for(int i=1; i<=n; i++)
    {
        for(int j=1; j<=m; j++)
        {
            cin>>a[i][j];
            ch[i][j]='.';
        }
    }
    for(int i=1; i<=n; i++)
    {
        for(int j=1; j<=m; j++)
        {
            if(vis[i][j]==0)
            {
                rt=cc+1;
                dfs(i,j);
                v.push_back(rt);
            }
        }
    }
    for(int i=0; i<v.size(); i++)
    {
        int sz=mp[v[i]].size();
        if(sz>siz)
        {
            siz=sz;
            cnt=v[i];
        }
    }
    for(int i=0; i<v.size(); i++)
    {
        if(mp[cnt].size()==1)continue;
        int sz=mp[v[i]].size();
        if(cnt!=v[i] && chk[v[i]]==0)
        {
            chk[v[i]]=1;
            ans=ans+sz;
        }
    }
    int q=chk[cnt];
    if(mp[cnt].size()>1)q++;
    if(ans+q<=k)cout<<ans+q<<endl;
    else cout<<-1<<endl;
    return 0;
}
