#include <iostream>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        bool found = false;
        for (int i = 1; i < n - 10; i++) {
            if (s[i - 1] == '8' && s[i] == '0' && s[i + 1] == '0') {
                found = true;
            }
        }
        if (found) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
