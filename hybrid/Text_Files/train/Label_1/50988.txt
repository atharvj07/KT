import java.util.*;

class Main {
  static boolean isRight(Point a, Point b, Point c) {
    double[] x = { a.x, b.x, c.x };
    double[] y = { a.y, b.y, c.y };
    double[] len = { dist(a, b), dist(b, c), dist(c, a) };
    Arrays.sort(len);
    return len[2] * len[2] == len[0] * len[0] + len[1] * len[1];
  }

  static double dist(Point a, Point b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return Math.sqrt(dx * dx + dy * dy);
  }

  static boolean isAlmostRight(Point a, Point b, Point c) {
    Point[] points = { a, b, c };
    for (int i = 0; i < 3; i++) {
      for (int dx = -1; dx <= 1; dx++) {
        for (int dy = -1; dy <= 1; dy++) {
          if (dx == 0 && dy == 0) continue;
          Point p = new Point(points[i].x + dx, points[i].y + dy);
          if (isRight(p, points[(i + 1) % 3], points[(i + 2) % 3])) return true;
        }
      }
    }
    return false;
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    Point a = new Point(scanner.nextInt(), scanner.nextInt());
    Point b = new Point(scanner.nextInt(), scanner.nextInt());
    Point c = new Point(scanner.nextInt(), scanner.nextInt());
    scanner.close();

    if (isRight(a, b, c)) System.out.println("RIGHT");
    else if (isAlmostRight(a, b, c)) System.out.println("ALMOST");
    else System.out.println("NEITHER");
  }

  static class Point {
    int x, y;

    public Point(int x, int y) {
      this.x = x;
      this.y = y;
    }
  }
}
