#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<int> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  sort(v.begin(), v.end());
  int sum = 0, ans = -1;
  for (int i = 0; i < n; i++) {
    if (sum + v[i] <= s) {
      sum += v[i];
      ans = v[i];
    } else {
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
