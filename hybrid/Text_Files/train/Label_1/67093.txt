def rearrange_schedule(s, t):
    ones = s.count('1')
    zeros = s.count('0')
    ones_t = t.count('1')
    zeros_t = t.count('0')
    prefix = min(ones // ones_t, zeros // zeros_t)
    ones -= ones_t * prefix
    zeros -= zeros_t * prefix
    result = t * prefix
    if zeros_t > 0 and prefix > 0:
        extra_zeros = min(zeros, zeros_t - 1)
        result += '0' * extra_zeros
        zeros -= extra_zeros
    if ones_t > 0 and prefix > 0:
        extra_ones = min(ones, ones_t - 1)
        result += '1' * extra_ones
        ones -= extra_ones
    result += '0' * zeros + '1' * ones
    return result

s = input().strip()
t = input().strip()
print(rearrange_schedule(s, t))
