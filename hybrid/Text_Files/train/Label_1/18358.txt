#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 2e5 + 5;
int n, m, a[N];
pair<int, int> ans[N];

bool cmp(int x, int y) { return a[x] < a[y]; }

void solve() {
  cin >> n >> m;
  if (n % 2 == 1) {
    cout << -1 << '\n';
    return;
  }
  vector<int> odds, evens;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    if (a[i] % 2)
      odds.push_back(i);
    else
      evens.push_back(i);
  }

  int need = (n - m) / 2, p = 0;
  sort(odds.begin(), odds.end(), cmp);
  sort(evens.begin(), evens.end(), cmp);

  for (int i = 0; i < evens.size() && p < need; i++) {
    if (a[evens[i]] > m) break;
    ans[evens[i]] = {odds[p], m + 1 - a[evens[i]]};
    p++;
  }

  for (int i = 0; i < odds.size() && p < need; i++) {
    if (a[odds[i]] > m) break;
    ans[odds[i]] = {evens[p], m + 1 - a[odds[i]]};
    p++;
  }

  if (p < need) {
    cout << -1 << '\n';
    return;
  }

  cout << need << '\n';
  for (int i = 1; i <= n; i++) cout << (ans[i].first ? ans[i].second : a[i]) << ' ';
}

signed main() { solve(); }
