from datetime import datetime, timedelta

def find_moment(n, m, logs):
    log_times = []
    for log in logs:
        dt_str = log[:19]  # Extract datetime string
        dt = datetime.strptime(dt_str, "%Y-%m-%d %H:%M:%S")
        log_times.append(dt)
    for i in range(len(log_times)):
        current_time = log_times[i]
        past_time = current_time - timedelta(seconds=n)
        count = sum(1 for dt in log_times if past_time < dt <= current_time)
        if count >= m:
            return current_time.strftime("%Y-%m-%d %H:%M:%S")
    return -1

n, m = map(int, input().split())
logs = []
while True:
    try:
        log = input()
    except EOFError:
        break
    logs.append(log)

print(find_moment(n, m, logs))
