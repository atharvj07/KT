import java.util.Scanner;

public class VasyaTheSleuth {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String question = scanner.nextLine().trim();

        char lastLetter = question.charAt(question.length() - 2); // Ignore the question mark

        String vowels = "AEIOUYaeiouy";
        String consonants = "BCDFGHJKLMNPQRSTVWXZbcdfghjklmnpqrstvwxz";

        if (vowels.indexOf(lastLetter) != -1) {
            System.out.println("YES");
        } else if (consonants.indexOf(lastLetter) != -1) {
            System.out.println("NO");
        } else {
            System.out.println("Invalid question format.");
        }
    }
}
