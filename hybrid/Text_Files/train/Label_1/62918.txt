def solve_problems(n, k):
    min_problems = k * (k + 1) // 2
    if n < min_problems:
        return "NO"
    
    # Base problems per day
    problems_per_day = [i for i in range(1, k + 1)]
    additional_problems = n - min_problems
    
    # Distribute additional problems
    for i in range(k - 1, -1, -1):
        additional = min(additional_problems, problems_per_day[i])
        problems_per_day[i] += additional
        additional_problems -= additional
        if additional_problems == 0:
            break
    
    if additional_problems > 0:
        return "NO"
    else:
        return "YES\n" + " ".join(map(str, problems_per_day))

n, k = map(int, input().split())
print(solve_problems(n, k))
