import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        Arrays.sort(a);
        int ans = 1;
        long pre = a[0];
        long pow = 1;
        for (int i = 1; i < n; i++) {
            if (a[i] == pre) {
                continue;
            }
            if (a[i] - pre == pow) {
                ans++;
                pow *= 2;
            } else {
                pow = 1;
            }
            pre = a[i];
        }
        System.out.println(ans);
        System.out.println(a[0]);
        for (int i = 1; i < n; i++) {
            if (a[i] == pre || a[i] - pre != pow) {
                continue;
            }
            System.out.print(" ");
            System.out.print(a[i]);
            ans++;
            pow *= 2;
            pre = a[i];
        }
        System.out.println();
    }
}
