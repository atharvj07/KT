#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;

    int left = 0, right = 0;
    int ans = n + 1;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;

        if (x == 0) left++;
        else right++;

        if (left > 0 && right > 0) {
            ans = i + 1;
            break;
        }
    }

    cout << ans << endl;

    return 0;
}
