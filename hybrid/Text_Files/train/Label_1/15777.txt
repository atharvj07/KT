#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n;) {
    char c = s[i];
    int t = 1;
    for (int j = i + 1; j < n; j++) {
      if (s[j] == c)
        t++;
      else
        break;
    }
    cout << c;
    i += t;
  }
  return 0;
}
