#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> b(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> b[i][j];
      }
      sort(b[i].begin(), b[i].end());
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cout << b[i][j] << ' ';
      }
      cout << '\n';
    }
  }
  return 0;
}
