#include<bits/stdc++.h>
using namespace std;
#define ll long long
void solve()
{
	ll n,m,k,x,y,z;
	cin>>n>>m>>k;
	vector<vector<ll>>v(n+1,vector<ll>(m+1)),ans(n+1,vector<ll>(m+1));
	while(k--)
	{
		cin>>x>>y>>z;
		if(x==1)
		{
			for(ll j=1;j<=m;j++)
			{
				ans[y][j]=z;
			}
		}
		else
		{
			for(ll i=1;i<=n;i++)
			{
				ans[i][y]=z;
			}
		}
	}
	for(ll i=1;i<=n;i++)
	{
		for(ll j=1;j<=m;j++)
		{
			cout<<ans[i][j]<<" ";
		}cout<<endl;
	}
}
int main()
{
	solve();
	return 0;
}
