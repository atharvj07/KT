import java.util.Scanner;

public class ShopSellOut {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int f = scanner.nextInt();
        long[] products = new long[n];
        long[] clients = new long[n];
        for (int i = 0; i < n; i++) {
            products[i] = scanner.nextLong();
            clients[i] = scanner.nextLong();
        }

        long maxProducts = 0;
        for (int mask = 0; mask < (1 << n); mask++) {
            if (Integer.bitCount(mask) != f) {
                continue;
            }
            long currentProducts = 0;
            for (int i = 0; i < n; i++) {
                if (((mask >> i) & 1) == 1) {
                    currentProducts += products[i] * 2;
                } else {
                    currentProducts += products[i];
                }
                currentProducts -= Math.min(currentProducts, clients[i]);
            }
            maxProducts = Math.max(maxProducts, currentProducts);
        }

        System.out.println(maxProducts);
    }
}
