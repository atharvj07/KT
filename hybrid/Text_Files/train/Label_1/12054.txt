#include <iostream>
const int mod = 1e9 + 7;
using namespace std;
long long poww(long long a, long long b) {
    long long ans = 1;
    while (b) {
        if (b & 1) ans = (ans * a) % mod;
        a = (a * a) % mod;
        b >>= 1;
    }
    return ans % mod;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        cout << poww(2ll, n * 1ll * k - n) << endl;
    }
}
