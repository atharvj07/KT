import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s1 = scanner.nextLine();
        String s2 = scanner.nextLine();
        System.out.println(s1.equals(s2) ? "YES" : isEquivalent(s1, s2) ? "YES" : "NO");
    }

    private static boolean isEquivalent(String s1, String s2) {
        if (s1.length() % 2 == 1) {
            return false;
        }
        int mid = s1.length() / 2;
        String s11 = s1.substring(0, mid);
        String s12 = s1.substring(mid);
        String s21 = s2.substring(0, mid);
        String s22 = s2.substring(mid);
        return (isEquivalent(s11, s21) && isEquivalent(s12, s22)) || (isEquivalent(s11, s22) && isEquivalent(s12, s21));
    }
}
