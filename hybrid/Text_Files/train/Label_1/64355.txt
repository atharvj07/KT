n, m = map(int, input().split())
notes = [list(map(int, input().split())) for _ in range(m)]

max_height = 0
# Check the height change between the first day and the first note, and the last note and the last day
max_height = max(max_height, notes[0][1] + notes[0][0] - 1)
max_height = max(max_height, notes[-1][1] + n - notes[-1][0])

# Check the height change between each consecutive notes
for i in range(1, m):
    day_diff = notes[i][0] - notes[i-1][0]
    height_diff = abs(notes[i][1] - notes[i-1][1])

    if height_diff > day_diff:
        print("IMPOSSIBLE")
        break
    else:
        max_possible_height = (day_diff - height_diff) // 2 + max(notes[i][1], notes[i-1][1])
        max_height = max(max_height, max_possible_height)
else:
    print(max_height)
