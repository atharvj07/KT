import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int x = Integer.parseInt(br.readLine());
        int a = 0;
        char b = 'D';
        if (x % 4 == 3) {
            a = 1;
            b = 'B';
        } else if (x % 4 == 2) {
            a = 2;
            b = 'C';
        } else if (x % 4 == 1) {
            b = 'A';
        } else {
            a = 1;
            b = 'B';
        }
        System.out.println(a + " " + b);
    }
}
