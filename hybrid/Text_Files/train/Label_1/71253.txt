MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    print(power(2, k-1, MOD) * power(2, n-1, MOD) % MOD)
