#include <deque>
#include <vector>
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        vector<vector<int>> dp(n, vector<int>(2, 1e9));
        dp[0][0] = dp[0][1] = 0;
        for (int i = 1; i < n; i++) {
            dp[i][0] = min(dp[i][0], dp[i - 1][0] + (a[i] > a[i - 1]));
            dp[i][0] = min(dp[i][0], dp[i - 1][1] + (a[i] < a[i - 1]));
            dp[i][1] = min(dp[i][1], dp[i - 1][1] + (a[i] > a[i - 1]));
            dp[i][1] = min(dp[i][1], dp[i - 1][0] + (a[i] < a[i - 1]));
        }
        cout << min(dp[n - 1][0], dp[n - 1][1]) << "\n";
    }
}
