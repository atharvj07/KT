import sys

n, m = list(map(int, sys.stdin.readline().split()))
grid = []
for i in range(n):
    grid.append([0] + list(map(int, sys.stdin.readline().strip())) + [0])

ans = 0
for i in range(n - 1, -1, -1):
    for j in range(1, m + 1):
        if grid[i][j] == 0:
            continue
        ans += min(grid[i + 1][j], grid[i + 1][j + 1])
        grid[i + 1][j] = min(grid[i + 1][j], grid[i + 1][j + 1])
        grid[i][j] = 1
print(ans)
