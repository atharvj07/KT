def solve_sequence_problem(n, a):
    l, r = 0, n - 1
    last, ans, s = 0, 0, []
    while l <= r:
        choices = []
        if a[l] > last:
            choices.append((a[l], 'L'))
        if a[r] > last:
            choices.append((a[r], 'R'))
        if not choices:
            break
        next_val, direction = min(choices)
        last = next_val
        ans += 1
        s.append(direction)
        if direction == 'L':
            l += 1
        else:
            r -= 1
    print(ans)
    print(''.join(s))

n = int(input().strip())
a = list(map(int, input().strip().split()))
solve_sequence_problem(n, a)
