from sys import stdin
input = stdin.readline

def merge(a, b):
    if not a: return b
    if not b: return a
    if a[0] < b[0]: return [a[0]] + merge(a[1:], b)
    return [b[0]] + merge(a, b[1:])

def is_mergeable(p):
    n = len(p) // 2
    a = []
    b = []
    for x in p:
        if x <= n: a.append(x)
        else: b.append(x)
    if len(a) == n and len(b) == n and sorted(a + b) == p: return "YES"
    return "NO"

for _ in range(int(input())):
    n = int(input())
    p = [int(x) for x in input().split()]
    print(is_mergeable(p))
