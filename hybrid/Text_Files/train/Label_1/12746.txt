#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);

	int t;
	cin >> t;

	while (t--) {
		int n, k;
		cin >> n >> k;

		vector<int> a(n);
		for (int i = 0; i < n; i++) {
			cin >> a[i];
		}

		int ans = 0, l = -1;
		for (int i = 0; i <= n - k; i++) {
			int cnt = 0;
			for (int j = i + 1; j < i + k - 1; j++) {
				if (a[j] > a[j - 1] && a[j] > a[j + 1]) {
					cnt++;
				}
			}

			if (cnt > ans) {
				ans = cnt;
				l = i + 1;
			} else if (cnt == ans) {
				l = min(l, i + 1);
			}
		}

		cout << ans + 1 << ' ' << l << '\n';
	}

	return 0;
}
