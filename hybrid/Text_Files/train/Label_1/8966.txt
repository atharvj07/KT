#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mxN=1e5+5;
ll a[mxN];

int main() {
  ll n, d; cin>>n>>d;
  for(int i=0; i<n; ++i)
    cin>>a[i];
  ll need=0, ans=0;
  for(int i=0; i<n; ++i) {
    if(a[i]==0) {
      if(need<=0) ++ans;
    } else {
      need+=a[i];
      if(need>d) return cout<<"-1", 0;
    }
  }
  cout<<ans<<"\n";
  return 0;
}
