#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;
  vector<int> cnt(n + 1);
  for (int i = 0; i < n; i++) cnt[i + 1] = cnt[i] + (s[i] == 'a');
  for (int i = 0; i < q; i++) {
    int l, r;
    cin >> l >> r;
    cout << cnt[r] - cnt[l - 1] + r - l + 1 << '\n';
  }
}
