import sys
sys.setrecursionlimit(100000)
n = int(input())
l, s = list(map(int, input().split())), list(input())
t = [[1e18] * 3 for _ in range(n)]
def dfs(pos, last, sta):
  if pos == n:
    return 0
  if t[pos][last] != 1e18:
    return t[pos][last]
  ret = 1e18
  if s[pos] != 'L':
    ret = dfs(pos + 1, 1, sta + (1 if s[pos] == 'G' else 0)) + ((0 if last == 1 else 5) if s[pos] == 'G' else (0 if last == 2 else 3)) * l[pos]
  if s[pos] != 'W':
    ret = min(ret, dfs(pos + 1, 2, sta + (1 if s[pos] == 'G' else 0)) + ((0 if last == 2 else 5) if s[pos] == 'G' else (0 if last == 1 else 3)) * l[pos])
  if sta > 0 or last == 3:
    ret = min(ret, dfs(pos + 1, 3, sta - 1 if s[pos] != 'G' else sta) + l[pos])
  t[pos][last] = ret
  return ret
print(dfs(0, 0, 0))
