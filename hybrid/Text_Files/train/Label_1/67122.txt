for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    a = [0] * n
    vis = [0] * n
    for i in range(n):
        if vis[i] == 0:
            j = i
            cycle = 0
            while vis[j] == 0:
                vis[j] = 1
                j = p[j] - 1
                cycle += 1
            for j in range(i, i + cycle):
                a[j % n] = cycle
    print(*a)
