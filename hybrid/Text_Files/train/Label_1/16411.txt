#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=1e5+5;
int n;
string s[N];
int32_t main() {
    cin>>n;
    unordered_map<string,bool>mp;
    for(int i=0; i<n; ++i){
        cin>>s[i];
        sort(s[i].begin(),s[i].end());
        mp[s[i]]=true;
    }
    cout<<mp.size()<<endl;
    return 0;
}
