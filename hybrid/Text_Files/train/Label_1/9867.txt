#include <bits/stdc++.h>
using namespace std;

const int mx=140582;
long long a[mx];
int n;
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    if(a[0]==1){
        cout<<"YES"<<endl;
        for(int i=1;i<=n;i++){
            cout<<(a[(i-1)%n]+1)<<" ";
        }
        cout<<endl;
        return 0;
    }
    cout<<"NO"<<endl;
}
