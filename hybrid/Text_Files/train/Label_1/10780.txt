#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        set<pair<int, int>> s;
        while (n--) {
            int a, b;
            cin >> a >> b;
            s.insert({a, -1});
            s.insert({b, 1});
        }
        int x = 1e9, y = 0;
        for (auto &i : s) {
            int a = i.first;
            int b = i.second;
            if (b == -1) x = min(x, a);
            if (b == 1) y = max(y, a);
        }
        cout << y - x << "\n";
    }
}
