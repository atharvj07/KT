#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (auto &x : a) cin >> x;
  int lo = *max_element(a.begin(), a.end());
  int hi = accumulate(a.begin(), a.end(), m);
  cout << lo << ' ' << hi << '\n';
}
int main() {
  solve();
  return 0;
}
