import java.util.*;

class Main {
    static boolean[] used;
    static char[] stages;
    static int[] weight;
    static int n, k;
    static int ans = Integer.MAX_VALUE, total = 0;

    static void dfs(int pos) {
        if (pos == k) {
            if (ans > total) ans = total;
            return;
        }
        for (int i = 0; i < n; i++) {
            if (!used[i] && (pos == 0 || stages[i] >= stages[i - 1] + 2)) {
                used[i] = true;
                total += weight[i];
                dfs(pos + 1);
                total -= weight[i];
                used[i] = false;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        used = new boolean[n];
        stages = sc.next().toCharArray();
        weight = new int[n];
        for (int i = 0; i < n; i++) {
            weight[i] = stages[i] - 'a' + 1;
        }
        dfs(0);
        System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
    }
}
