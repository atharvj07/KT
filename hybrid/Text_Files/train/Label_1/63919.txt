import sys

in_lines = sys.stdin.readlines()
n, *edges = [int(x) for e in in_lines[:-1] for x in e.split()]
seq = [int(x) for x in in_lines[-1].split()]

e = [[] for _ in range(n)]
for v1, v2 in zip(*[iter(edges)] * 2):
    e[v1 - 1].append(v2 - 1)
    e[v2 - 1].append(v1 - 1)

if seq[0] != 1:
    print('No')
    sys.exit()

u = [0] * n
u[0] = 1
q = [0]
i = 1
while q:
    v = q.pop(0)
    if v != seq[i - 1]:
        print('No')
        sys.exit()
    i += 1
    for w in e[v]:
        if not u[w]:
            q.append(w)
            u[w] = 1

print('Yes')
