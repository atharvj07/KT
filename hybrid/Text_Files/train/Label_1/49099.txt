import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int ans = 0;
        int last = arr[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (arr[i] > last) {
                ans++;
            } else {
                last = arr[i];
            }
        }
        if (ans > 1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
