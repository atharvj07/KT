#include <iostream>
#include <string>
#include <vector>
#include <map>

using namespace std;

int main() {
  string s;
  getline(cin, s);
  string t;
  getline(cin, t);
  int k;
  cin >> k;
  map<string, int> mp;
  int ans = 0;
  int bad = 0;
  string sub = "";
  for (int i = 0; i < s.length(); i++) {
    if (t[s[i] - 'a'] == '0') bad++;
    sub += s[i];
    while (bad > k) {
      if (t[sub[0] - 'a'] == '0') bad--;
      sub.erase(sub.begin());
    }
    ans += sub.length();
    mp[sub]++;
  }
  cout << ans - mp.size() << endl;
  return 0;
}
