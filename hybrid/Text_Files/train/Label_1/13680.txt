#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxN = 2e5 + 5;
ll ans, arr[maxN], sub[maxN];
vector<int> adj[maxN];
void dfs(int node, int p) {
    sub[node] = arr[node];
    for (auto u : adj[node]) {
        if (u == p)
            continue;
        dfs(u, node);
        sub[node] += sub[u];
    }
}
void update(int node, int p, ll val, int add_rem) {
    if (add_rem == 1) {
        arr[node] += val;
        sub[node] += val;
        sub[p] -= val;
    } else {
        arr[node] -= val;
        sub[node] -= val;
        sub[p] += val;
    }
    for (auto u : adj[node]) {
        if (u == p)
            continue;
        update(u, node, val, add_rem);
    }
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 1; i < n; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    for (int i = 1; i <= n; i++) { cin >> arr[i]; }
    dfs(1, 0);
    for (int i = 1; i <= n; i++) {
        if (sub[i] < 0)
            ans += -sub[i];
        else if (sub[i] > 0) {
            ll x = sub[i] / n;
            update(1, i, x, 1);
            ans += x;
        }
    }
    dfs(1, 0);
    for (int i = 1; i <= n; i++) {
        if (arr[i] > 0)
            ans += arr[i];
        else if (arr[i] < 0) {
            ll x = -arr[i] / n;
            update(1, i, x, 0);
            ans += x;
        }
    }
    cout << ans << "\n";
    return 0;
}
