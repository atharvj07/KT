#include <bits/stdc++.h>
using namespace std;
#define N 110
char g[N][N],vis[N];
int n,m,a,b,ans,dis[N],head,tail;
queue <int> q;
int main()
{
    cin>>n>>m;
    for (int i=1;i<=m;i++)
    {
        cin>>a>>b;
        g[a][b]=g[b][a]=1;
    }
    q.push(1);vis[1]=1;dis[1]=1;
    while (!q.empty())
    {
        head=q.front();q.pop();
        if (head==n) ans=dis[head];
        for (int i=1;i<=n;i++)
            if (g[head][i]&&!vis[i])
            {
                q.push(i);
                vis[i]=1;
                dis[i]=dis[head]+1;
            }
    }
    if (ans&1) cout<<"Alice";
    else cout<<"Bob";
    return 0;
}
