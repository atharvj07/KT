#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, sx, sy;
  cin >> n >> sx >> sy;
  pair<int, int> ans = {0, 0};
  int maxcnt = 0;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    int cnt = 0;
    if (sx < x) cnt += 1;
    if (sx > x) cnt += 1;
    if (sy < y) cnt += 1;
    if (sy > y) cnt += 1;
    if (cnt > maxcnt) {
      maxcnt = cnt;
      ans = {x, y};
    }
  }
  cout << maxcnt << '\n' << ans.first << ' ' << ans.second << '\n';
  return 0;
}
