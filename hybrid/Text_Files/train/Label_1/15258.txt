#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  ll n, p, c;
  cin >> n;
  vector<ll> b(n - 1), c_(n - 1);
  for (int i = 0; i < n - 1; i++) { cin >> b[i]; }
  for (int i = 0; i < n - 1; i++) { cin >> c_[i]; }
  deque<ll> a;
  a.push_front(c_[0]);
  a.push_back(b[0]);
  for (int i = 1; i < n - 1; i++) {
    p = min(b[i], c_[i]);
    c = max(b[i], c_[i]);
    if (a.back() == p)
      a.push_back(c);
    else
      a.push_front(c);
  }
  for (ll x : a) { cout << x << " "; }
  return 0;
}
