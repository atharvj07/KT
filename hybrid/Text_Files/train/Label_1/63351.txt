import sys

def dfs(u, p, graph, out):
    out[u] = 1
    for v in graph[u]:
        if v != p:
            dfs(v, u, graph, out)
            out[u] += out[v]

for q in range(int(input())):
    n = int(input())
    graph = {i: [] for i in range(1, n+1)}
    for _ in range(n-1):
        u, v = map(int, input().split())
        graph[u].append(v)
        graph[v].append(u)
    out = [0 for _ in range(n+1)]
    dfs(1, 1, graph, out)
    print(max(out))
