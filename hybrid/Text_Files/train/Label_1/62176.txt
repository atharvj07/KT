import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = [0] * n
    for i in range(n):
        b[a[i] - 1] = i
    k = [0] * n
    for i in range(n):
        if i == 0 or (b[i] - b[i - 1] + n) % n != 1:
            k[i] = 1
    print(sum(k), *[i for i, v in enumerate(k) if v])
