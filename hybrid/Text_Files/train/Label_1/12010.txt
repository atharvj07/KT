#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int MAX_N = 1e5 + 10;
const int MAX_M = 1e5 + 10;

int n, m, d[MAX_N], a[MAX_M];
vector<int> g[MAX_M];
int dist[MAX_M];

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) {
    cin >> d[i];
    if (d[i]) g[d[i]].push_back(i);
  }
  for (int i = 1; i <= m; ++i) cin >> a[i];

  queue<int> q;
  for (int i = 1; i <= m; ++i) {
    if (!a[i]) {
      q.push(i);
    }
    dist[i] = n + 1;
  }

  int ans = 0;
  while (!q.empty()) {
    int v = q.front();
    q.pop();
    ans = max(ans, dist[v]);
    for (int u : g[v]) {
      dist[d[u]] = min(dist[d[u]], dist[v] + a[v]);
      --a[d[u]];
      if (!a[d[u]]) {
        q.push(d[u]);
      }
    }
  }

  int cnt = 0;
  for (int i = 1; i <= m; ++i) {
    cnt += (a[i] == 0);
  }
  cout << (cnt == m ? ans : -1) << endl;
  return 0;
}
