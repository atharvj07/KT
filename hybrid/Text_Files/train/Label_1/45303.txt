import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);

    final String trump = scanner.next();
    final String card1 = scanner.next();
    final String card2 = scanner.next();

    if (card1.charAt(1) == card2.charAt(1) && card1.charAt(0) > card2.charAt(0)) {
      System.out.println("YES");
    } else if (trump.equals(Character.toString(card1.charAt(1))) || (card1.charAt(1) == trump.charAt(0))) {
      System.out.println("YES");
    } else {
      System.out.println("NO");
    }
  }
}
