#include <iostream>
#include <set>
using namespace std;

const int MX = 1e3 + 5;
string w[MX];
int sz, ans;
set<string> st;

int main() {
    cin >> sz;
    for (int i = 1; i <= sz; i++) {
        cin >> w[i];
        string root;
        for (char j = 'a'; j <= 'z'; j++) {
            for (char k : w[i]) {
                if (k == j) root += j;
            }
        }
        if (!st.count(root)) ans++, st.insert(root);
    }
    cout << ans << endl;
}
