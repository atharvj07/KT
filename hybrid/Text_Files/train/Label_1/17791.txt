#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int r, g, b, w;
    cin >> r >> g >> b >> w;
    if (r > w || g > w || b > w) {
      cout << "No" << endl;
    } else {
      int mn = min({r, g, b});
      r -= mn;
      g -= mn;
      b -= mn;
      int mx = max({r, g, b});
      if (mx > w - mn) {
        cout << "No" << endl;
      } else {
        cout << "Yes" << endl;
      }
    }
  }
  return 0;
}
