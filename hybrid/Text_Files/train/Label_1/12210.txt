#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,i,j,mn=1e9,cnt=0,f=0;
    string s,t;
    cin>>n>>s>>m>>t;
    for(i=0;i<=n-m+1;i++){
        cnt=0;
        f=0;
        for(j=0;j<m;j++){
            if(t[j]==s[i+j]){
                cnt++;
            }
            if(s[i+j]=='?'){
                cnt++;
            }
        }
        if(cnt==m){
            f=1;
        }
        if(f){
            mn=0;
        }
        else{
            mn=min(mn,m-cnt);
        }
    }
    cout<<mn<<endl;
    return 0;
}
