#include <bits/stdc++.h>
using namespace std;
string s,t; int n;
vector<int> v;
int f(string p){
    int ans=0,x=0;
    for(int i=0;i<n;i++){
        x+=p[i]=='0';
        if(!i||p[i]!=p[i-1]) ans=max(ans,x);
    }
    return ans;
}
int main(){
    cin>>s;
    n=s.length();
    int w=n;
    while(w){
        if(s[w-1]=='0'){
            t+='0';
            v.push_back(w-1);
        }
        w/=2;
    }
    for(int i=n-1;i>=0;i--){
        if(!binary_search(v.begin(),v.end(),i)){
            t+='1';
        }
    }
    int x=f(s);
    if(f(t)!=x) return !printf("-1");
    cout<<t<<endl;
}
