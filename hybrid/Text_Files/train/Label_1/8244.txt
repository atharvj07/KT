#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    bool rated = false, unrated = false;
    for (int i = 0; i < n; i++) {
        if (a[i].first != a[i].second)
            rated = true;
        if (i > 0 && a[i - 1].second < a[i].second)
            unrated = true;
    }
    if (rated && unrated)
        cout << "maybe" << endl;
    else if (rated)
        cout << "rated" << endl;
    else
        cout << "unrated" << endl;
    return 0;
}
