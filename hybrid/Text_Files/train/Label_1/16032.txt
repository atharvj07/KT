#include <cstdio>
#include <iostream>
#include <map>
#include <string>
using namespace std;

int freqS[1001][26];
int freqT[1001][26];
int dp[1001][1001][11];

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  string s, t;
  cin >> s >> t;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 26; j++) freqS[i + 1][j] = freqS[i][j];
    freqS[i + 1][s[i] - 'a']++;
  }
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < 26; j++) freqT[i + 1][j] = freqT[i][j];
    freqT[i + 1][t[i] - 'a']++;
  }
  for (int i = 0; i <= n; i++)
    for (int j = 0; j <= m; j++)
      for (int l = 0; l <= k; l++) dp[i][j][l] = -1;
  dp[0][0][0] = 0;
  int ans = 0;
  for (int i = 0; i <= n; i++)
    for (int j = 0; j <= m; j++)
      for (int l = 0; l <= k; l++) {
        if (dp[i][j][l] == -1) continue;
        dp[i][j][l + 1] = dp[i][j][l];
        if (i == n || j == m) continue;
        if (s[i] != t[j]) continue;
        bool ok = true;
        for (int x = 0; x < 26; x++) {
          if (x & 1) continue;
          ok &= freqS[i + 1][x] <= freqT[j + 1][x];
        }
        for (int x = 0; x < 26; x++) {
          if (!(x & 1)) continue;
          ok &= freqS[i + 1][x] <= freqT[j + 1][x];
        }
        if (ok) {
          dp[i + 1][j + 1][l + 1] = max(dp[i + 1][j + 1][l + 1], dp[i][j][l] + 1);
          if (dp[i + 1][j + 1][l + 1] > ans) ans = dp[i + 1][j + 1][l + 1];
        }
      }
  cout << ans << endl;
  return 0;
}
