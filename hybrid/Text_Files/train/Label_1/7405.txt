#include <bits/stdc++.h>
using ll = long long;
using namespace std;
const int N = 2e5 + 5;
int n, p, k;
ll a[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> p >> k;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    sort(a + 1, a + n + 1);
    int ans = 0;
    for (int i = n; i >= 1; i--) {
      if (p >= a[i]) {
        ans += 1;
        p -= a[i];
        k -= 1;
      }
      if (k <= 0) { break; }
    }
    cout << ans << endl;
  }
  return 0;
}
