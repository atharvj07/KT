#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1e5 + 5;
const ll inf = 1e18;
ll d[N], a[N], e[N];
ll dp[N][21];
int main() {
    ll n, k;
    cin >> n >> k;
    for (ll i = 1; i <= n; ++i) cin >> a[i];
    for (ll i = 1; i <= n; ++i) cin >> d[i];
    for (ll i = 1; i < n; ++i) {
        cin >> e[i];
        --e[i];
    }
    for (ll j = 0; j <= k; ++j) {
        for (ll i = 1; i <= n; ++i) dp[i][j] = -inf;
    }
    dp[1][0] = 0;
    for (ll i = 1; i <= n; ++i) {
        for (ll j = 0; j <= k; ++j) {
            ll nj = j;
            if (e[i] > 0 && j) nj = j - 1;
            dp[i + 1][j] = max(dp[i + 1][j], dp[i][j]);
            // no exchange bond
            dp[i + 1][nj] = max(dp[i + 1][nj], dp[i][j] + a[i]);
            // exchange bond
            dp[e[i] + 1][j] = max(dp[e[i] + 1][j], dp[i][j] + a[i] - d[e[i]]);
        }
    }
    cout << *max_element(dp[n] + 1, dp[n] + k + 2) << endl;
    return 0;
}
