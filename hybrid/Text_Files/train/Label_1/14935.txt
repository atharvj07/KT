#include <bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt[26] = {0};
  for (int i = 0; i < n; i++) cnt[s[i] - 'a']++;
  int mx = *max_element(cnt, cnt + 26);
  cout << (mx > (n + 1) / 2 ? -1 : n - mx) << endl;
  return 0;
}
