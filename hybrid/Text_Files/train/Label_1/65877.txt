n, s, t = map(int, input().split())
p = list(map(int, input().split()))

if s == t:
    print(0)
else:
    visited = [False] * n
    current = s
    count = 0
    while not visited[current - 1]:
        visited[current - 1] = True
        current = p[current - 1]
        count += 1
        if current == t:
            print(count)
            break
    else:
        print(-1)
