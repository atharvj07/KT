#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  int ans = 0;
  for (int i = 0; i < n;) {
    int j = i;
    while (j < n && (a[j] == 0 || a[j] == (1 << k) - 1)) { j++; }
    if (j == n) { break; }
    int xor_sum = 0;
    while (j < n && (xor_sum ^ a[j]) != 0) {
      xor_sum ^= a[j];
      j++;
    }
    ans += j - i;
    i = j;
  }

  cout << ans << endl;
  return 0;
}
