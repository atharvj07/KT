#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  sort(a.begin(), a.end(), greater<int>());
  int sum = 0;
  for (int x : a) sum += x;
  if (sum & 1 || a[0] * 2 <= sum - a[0]) {
    cout << 0 << endl;
    return 0;
  }
  int sum2 = a[0];
  vector<int> ans;
  ans.push_back(1);
  for (int i = 1; i < n; i++) {
    sum2 += a[i];
    if (sum2 * 2 > sum) {
      ans.push_back(i + 1);
    }
  }
  cout << ans.size() << endl;
  for (int x : ans) { cout << x << ' '; }
  return 0;
}
