import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) break;
            int y = scanner.nextInt();
            double max = 0;
            int bank = 0;
            for (int i = 1; i <= n; i++) {
                int bi = scanner.nextInt();
                double ri = scanner.nextDouble();
                int ti = scanner.nextInt();
                double interest = ti == 1 ? ri * y : Math.pow(1 + ri / 100, y) - 1;
                double principalAndInterest = 1 + interest;
                if (principalAndInterest > max) {
                    max = principalAndInterest;
                    bank = bi;
                }
            }
            System.out.println(bank);
        }
        scanner.close();
    }
}
