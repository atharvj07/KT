n, m = map(int, input().split())
A = [[int(i) for i in input().split()] for j in range(n)]
B = [[int(i) for i in input().split()] for j in range(n)]
flag = 0
for i in range(n - 1):
    for j in range(m - 1):
        if (A[i][j] == A[i + 1][j] == A[i][j + 1] == A[i + 1][j + 1]) and (A[i][j] != B[i][j]):
            flag = 1
            break
    if flag:
        break
if flag:
    print("No")
else:
    print("Yes")
