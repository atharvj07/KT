import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        boolean[] sieve = new boolean[m + 1];
        for (int i = 2; i <= m; i++) {
            sieve[i] = true;
        }
        for (int i = 2; i * i <= m; i++) {
            if (sieve[i]) {
                for (int j = i * i; j <= m; j += i) {
                    sieve[j] = false;
                }
            }
        }
        System.out.println(sieve[m] && sieve[n + 1] ? "YES" : "NO");
    }
}
