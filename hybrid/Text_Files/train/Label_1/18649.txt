#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    if (n % 2 == 0) {
        cout << n << endl;
        for (int i = 1; i <= n; i += 2) {
            cout << i << ' ' << i + 1 << ' ';
        }
        cout << endl;
    } else {
        if (n == 3) {
            cout << 2 << endl;
            cout << 1 << ' ' << 3 << endl;
        } else {
            cout << n << endl;
            for (int i = 1; i <= n; i += 2) {
                if (i == n) {
                    cout << n << ' ';
                } else {
                    cout << i << ' ' << i + 2 << ' ';
                }
            }
            cout << endl;
        }
    }
    return 0;
}
