#include <vector>
#include <iostream>
using namespace std;

int main() {
    long long n;
    cin >> n;
    string ans;
    cin >> ans;
    vector<long long> dp(n + 2);
    dp[n] = 1;
    for (long long i = n - 1; i >= 0; --i) {
        long long cur = ans[i] - '0';
        dp[i] = dp[i + 1];
        if ((cur % 2 == 0) or (cur == 5 and ans[i + 1] <= '6')) {
            dp[i] += dp[i + 2];
        }
        dp[i] %= 1000000007;
    }
    cout << dp[0];
    return 0;
}
