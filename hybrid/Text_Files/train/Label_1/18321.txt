#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    while (cin >> n) {
        vector<string> state(n);
        for (int i = 0; i < n; ++i) cin >> state[i];
        int mx = -1;
        for (int j = 0; j < n; ++j) {
            int cnt = 0;
            for (int i = 0; i < n; ++i) {
                cnt += state[i][j] == '0';
            }
            mx = max(mx, min(cnt, n - cnt));
        }
        cout << mx * 2 << '\n';
    }
    return 0;
}
