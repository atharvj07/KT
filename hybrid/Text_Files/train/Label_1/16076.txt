#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long LL;
int main() {
  LL n, l, a, t, ll;
  cin >> n >> l >> a;
  vector<pair<LL, LL>> customer;
  while (n--) {
    cin >> t >> ll;
    customer.push_back(make_pair(t, ll));
  }
  LL last = 0, ans = 0;
  for (auto v : customer) {
    if (v.first - last >= a) {
      last = v.first + v.second;
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
