from collections import deque

s = input()
n = len(s)
bal = [0] * n
for i in range(n):
    if s[i] == '(':
        bal[i] = 1
    else:
        bal[i] = -1
bal2 = deque()
for i in range(n):
    bal2.append(bal[i])
    if not bal2 or bal2[0] == 1:
        bal2.appendleft(1)
    else:
        bal2.appendleft(0)
for i in range(n - 1, -1, -1):
    bal2[i] = max(bal2[i], bal2[i + 1])
m = int(input())
for i in range(m):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    print(bal2[l] if l == r else bal2[l] + bal2[r + 1])
