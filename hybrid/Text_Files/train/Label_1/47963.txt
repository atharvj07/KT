import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = bf.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int m = Integer.parseInt(line1[1]);

        Map<Integer, List<Integer>> adjList = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            adjList.put(i, new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            String[] line2 = bf.readLine().split(" ");
            int u = Integer.parseInt(line2[0]);
            int v = Integer.parseInt(line2[1]);
            adjList.get(u).add(v);
            adjList.get(v).add(u);
        }

        boolean[] visited = new boolean[n + 1];
        int[] cycleEdges = new int[m];
        int[] cycleStart = new int[m];
        int[] cycleEnd = new int[m];
        int cycleCount = 0;

        for (int i = 1; i <= n; i++) {
            if (visited[i]) {
                continue;
            }

            List<Integer> path = new ArrayList<>();
            int start = i;
            while (true) {
                visited[start] = true;
                path.add(start);

                boolean foundCycle = false;
                for (int neighbor : adjList.get(start)) {
                    if (!visited[neighbor]) {
                        start = neighbor;
                        foundCycle = true;
                        break;
                    } else if (path.contains(neighbor)) {
                        int cycleLength = path.size() - path.indexOf(neighbor);
                        if (cycleLength >= 3) {
                            for (int j = path.indexOf(neighbor); j < path.indexOf(neighbor) + cycleLength; j++) {
                                cycleEdges[j - path.indexOf(neighbor)] = cycleCount;
                                cycleStart[cycleCount] = path.get(j);
                                cycleEnd[cycleCount] = path.get((j + 1) % cycleLength);
                            }
                            cycleCount++;
                        }
                        foundCycle = true;
                        break;
                    }
                }

                if (!foundCycle) {
                    break;
                }
            }
        }

        int answer = 0;
        List<Integer> answerEdges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            if (cycleEdges[i] == 0) {
                answer++;
                answerEdges.add(i + 1);
            }
        }

        System.out.println(answer);
        Collections.sort(answerEdges);
        for (int edge : answerEdges) {
            System.out.print(edge + " ");
        }
    }
}
