#include <iostream>
#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, a, b, q;
  cin >> n >> k >> a >> b >> q;
  vector<int> orders(n + 1);
  while (q--) {
    int t, x;
    cin >> t >> x;
    orders[x] += t == 1 ? x : -x;
  }
  for (int i = 1; i <= n; i++) orders[i] += orders[i - 1];
  vector<int> cumsum(n - k + 1);
  cumsum[k] += orders[k];
  for (int i = k + 1; i <= n; i++) {
    cumsum[i] = max(cumsum[i - 1], orders[i]) + orders[i];
  }
  for (int i = 1; i <= n; i++) {
    cout << (i >= k ? cumsum[i - k] : 0) << endl;
  }
  return 0;
}
