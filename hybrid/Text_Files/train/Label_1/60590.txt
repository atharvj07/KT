import sys

def solve(s, t):
    if len(s) > len(t):
        return "NO"
    i = 0
    j = 0
    while j < len(t):
        if i < len(s) and s[i] == t[j]:
            i += 1
            j += 1
        else:
            if i == 0 or s[i-1] != t[j]:
                return "NO"
            j += 1
    if i == len(s):
        return "YES"
    else:
        return "NO"

n = int(input())
for i in range(n):
    s = input()
    t = input()
    print(solve(s, t))
