#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int l, r;
    cin >> l >> r;
    int ans = 0;
    for (int x = l; x <= r; x++) {
      int cnt = 0;
      for (int j = x; j > 0; j /= 10) cnt += j % 10 != 0;
      ans += cnt <= 3;
    }
    cout << ans << endl;
  }
  return 0;
}
