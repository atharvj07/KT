import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        int ans = 0;
        while (true) {
            boolean flag = false;
            for (int i = 0; i < n; i++) {
                if (h[i] == 0) {
                    continue;
                }
                if (i == 0 || i == n - 1 || h[i] <= h[i - 1] || h[i] <= h[i + 1]) {
                    h[i]--;
                    flag = true;
                }
            }
            if (!flag) {
                break;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
