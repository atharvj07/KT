x, k = map(int, input().split())
div2 = []
for _ in range(k):
    lst = list(map(int, input().split()))
    if lst[0] == 1:
        l, r = lst[1], lst[2]
        div2.extend([l, r])
    else:
        div2.append(lst[1])
div2.sort()
min_cnt = div2[0] - 1
max_cnt = x - max(div2) - 1
print(min_cnt, max_cnt)
