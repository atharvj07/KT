#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, M;
    cin >> n >> M;
    vector<int> t(n);
    for (int &i : t) cin >> i;
    vector<int> ans(n);
    for (int i = 0, s = 0; i < n; i++) {
        ans[i] = max(0, (s + t[i] - M + 1) / t[i]);
        s += t[i];
    }
    for (int i : ans) cout << i << ' ';
    cout << '\n';
}
