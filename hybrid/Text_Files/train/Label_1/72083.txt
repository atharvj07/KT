from datetime import datetime, timedelta

n = int(input())
times = [datetime.strptime(input(), '%H:%M') for _ in range(n)]
times.sort()
last_time = times[0]
answer = timedelta(hours=0, minutes=0)
for time in times[1:]:
    diff = time - last_time
    if diff > answer:
        answer = diff
    last_time = time
start = last_time + answer
print(start.strftime('%H:%M'))
