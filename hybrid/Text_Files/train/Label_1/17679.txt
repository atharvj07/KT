#include <bits/stdc++.h>
using namespace std;
int main() {
    int h, w;
    cin >> h >> w;
    vector<string> s(h);
    for (int i = 0; i < h; i++) { cin >> s[i]; }
    if (h == 1 && w == 1) {
        if (s[0][0] == '*') {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
        return 0;
    }
    int x = -1, y = -1;
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            if (s[i][j] == '*') {
                if (x == -1) {
                    x = i;
                    y = j;
                } else {
                    cout << "NO" << endl;
                    return 0;
                }
            }
        }
    }
    if (x == -1) {
        cout << "YES" << endl;
        return 0;
    }
    for (int i = x - 1; i >= 0; i--) { if (s[i][y] != '*') { cout << "NO" << endl; return 0; } }
    for (int i = x + 1; i < h; i++) { if (s[i][y] != '*') { cout << "NO" << endl; return 0; } }
    for (int i = y - 1; i >= 0; i--) { if (s[x][i] != '*') { cout << "NO" << endl; return 0; } }
    for (int i = y + 1; i < w; i++) { if (s[x][i] != '*') { cout << "NO" << endl; return 0; } }
    cout << "YES" << endl;
    return 0;
}
