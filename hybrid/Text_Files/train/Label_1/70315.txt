from collections import defaultdict

n, m = map(int, input().split())
edges = defaultdict(list)
for _ in range(m):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)

vis = [False] * (n + 1)
par = [0] * (n + 1)

def dfs(node, p):
    vis[node] = True
    par[node] = p
    mx = 0
    for u in edges[node]:
        if not vis[u]:
            mx = max(mx, dfs(u, node) + 1)
    return mx

dia = 0
node1, node2 = 0, 0
for i in range(1, n + 1):
    if not vis[i]:
        x = dfs(i, 0)
        if x > dia:
            node1, node2 = i, par[i]
            dia = x

vis = [False] * (n + 1)
dfs(node1, 0)
dia = 0
for i in range(1, n + 1):
    if not vis[i]:
        x = dfs(i, 0)
        if x > dia:
            dia = x

print(dia)
for i in range(2, n + 1):
    if par[i] != 0:
        print(par[i], i)
