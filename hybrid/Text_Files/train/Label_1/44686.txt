import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    static List<int[]> edgeList;
    static List<List<int[]>> adjacencyList;
    static int[] visited;
    static int[] parent;
    static int N;
    static int Q;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        edgeList = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            edgeList.add(new int[]{sc.nextInt(), sc.nextInt()});
        }
        adjacencyList = new ArrayList<>();
        for (int i = 0; i <= N; i++) {
            adjacencyList.add(new ArrayList<>());
        }
        for (int[] edge : edgeList) {
            adjacencyList.get(edge[0]).add(edge);
            adjacencyList.get(edge[1]).add(edge);
        }
        Q = sc.nextInt();
        for (int i = 0; i < Q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            visited = new int[N + 1];
            parent = new int[N + 1];
            Arrays.fill(parent, -1);
            int res = dfs(a, b);
            System.out.println(res);
        }
    }

    public static int dfs(int a, int b) {
        visited[a] = 1;
        for (int[] edge : adjacencyList.get(a)) {
            int next = edge[0] == a ? edge[1] : edge[0];
            if (next == b) {
                return 1;
            }
            if (visited[next] == 0) {
                parent[next] = a;
                int res = dfs(next, b);
                if (res > 0) {
                    return res + 1;
                }
            }
        }
        return -1;
    }
}
