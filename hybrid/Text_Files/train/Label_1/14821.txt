#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    int a[n+1],c[1000001];
    for (int i = 0; i < n; ++i) cin >> a[i];
    int cnt = 0, ansl = 0, ansr = 0;
    for (int i = 0, j = 0; i < n; ++i) {
        c[a[i]]++;
        if (c[a[i]] > cnt) {
            ansl = i;
        }
        cnt = max(cnt, c[a[i]]);
        while (c[a[i]] == cnt) {
            ansr = i;
            c[a[j]]--;
            j++;
        }
    }
    cout << ansl+1 << " " << ansr+1 << endl;
}
