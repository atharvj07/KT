import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] arr = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    arr[j][k] = scanner.nextInt();
                }
            }
            boolean flag = true;
            for (int j = 0; j < 5; j++) {
                if (arr[j][0] != arr[j][4] || arr[j][0] + arr[j][4] != 7) {
                    flag = false;
                }
            }
            for (int k = 0; k < 5; k++) {
                if (arr[0][k] != arr[4][k] || arr[0][k] + arr[4][k] != 7) {
                    flag = false;
                }
            }
            if (arr[0][0] != arr[4][4] || arr[0][0] + arr[4][4] != 7) {
                flag = false;
            }
            if (arr[0][4] != arr[4][0] || arr[0][4] + arr[4][0] != 7) {
                flag = false;
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (arr[j][k] == 0) {
                        continue;
                    }
                    for (int l = j; l < 5; l++) {
                        for (int m = k; m < 5; m++) {
                            if (arr[l][m] == arr[j][k]) {
                                flag = false;
                            }
                        }
                    }
                }
            }
            System.out.println(flag);
        }
    }
}
