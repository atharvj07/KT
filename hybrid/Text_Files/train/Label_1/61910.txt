import queue
n, m = map(int, input().split())
costs = [int(i) for i in input().split()]
graph = [[]] * n
vis = [False] * n
dist = [1e6] * n
for _ in range(m):
    u, v = map(lambda x: int(x) - 1, input().split())
    graph[u].append(v)
    graph[v].append(u)
q = queue.Queue()
q.put((0, 0))
while not q.empty():
    cost, node = q.get()
    if vis[node]:
        continue
    vis[node] = True
    dist[node] = cost
    for i in graph[node]:
        q.put((cost + costs[i], i))
print(sum(dist))
