import array

n = int(input())
s = input()
x = array.array('i', map(int, input().split()))
if s.count('L') == s.count('R'):
    if s[0] == 'L':
        t = -min(x)
        for i in range(n - 1):
            if (x[i + 1] - x[i]) % 2 == 0:
                t = min(t, (x[i + 1] - x[i]) // 2)
    else:
        t = x[0]
        for i in range(n - 1):
            if (x[i + 1] - x[i]) % 2 == 0:
                t = min(t, (x[i + 1] - x[i]) // 2)
else:
    t = -1
print(t)
