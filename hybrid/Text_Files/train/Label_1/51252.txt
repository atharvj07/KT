import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        BigInteger res = BigInteger.ONE;
        for (long i = 0; i < m; i++) res = res.multiply(BigInteger.valueOf(n - i));
        for (long i = 1; i <= m; i++) res = res.divide(BigInteger.valueOf(i));
        System.out.println(res.mod(BigInteger.valueOf(1000000007)));
    }
}
