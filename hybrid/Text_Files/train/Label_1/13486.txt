#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
long long fastpow(long long a, int b) {
    if (b == 0) return 1;
    if (b == 1) return a;
    long long val = fastpow(a, b / 2) % mod;
    val = (val * val) % mod;
    if (b % 2 != 0) val = (val * a) % mod;
    return val % mod;
}
int main() {
    int n, m;
    cin >> n >> m;
    long long ans = fastpow(2, n) % mod;
    ans -= 1;
    ans = (ans * fastpow(m, mod - 2)) % mod;
    cout << ans << "\n";
}
