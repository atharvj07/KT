import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        int[] sum = new int[100010];
        Arrays.fill(sum, 1);
        for (int i = 100009; i >= 2; i--) {
            for (int j = i; j <= 100009; j += i) {
                sum[j] += i;
            }
        }
        for (int i = 0; i < t; i++) {
            long n = scanner.nextLong();
            int temp = (int) (n % 100009);
            int temp1 = sum[temp];
            while (gcd(n, temp1) == 1) {
                temp++;
                if (temp == 100009) {
                    temp = 0;
                    temp1--;
                }
                n += 1;
            }
            System.out.println(n);
        }
        scanner.close();
    }

    static int gcd(long a, long b) {
        if (a == 0) {
            return (int) b;
        }
        return gcd(b % a, a);
    }
}
