#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, x;
        cin >> n >> m >> x;
        vector<int> h(n);
        for (auto &i : h) cin >> i;
        sort(h.rbegin(), h.rend());
        vector<int> res(n);
        bool ok = true;
        for (int i = 0; i < n; i++) {
            if (h[i] > x)
                ok = false;
            res[i] = (i % m) + 1;
        }
        if (!ok)
            cout << "NO" << endl;
        else {
            cout << "YES" << endl;
            for (auto &i : res) cout << i << " ";
            cout << endl;
        }
    }
}
