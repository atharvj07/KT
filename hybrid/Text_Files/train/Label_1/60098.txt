def is_valid(grid, n, m):
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                continue
            cnt = 0
            if i > 0 and grid[i-1][j] == grid[i][j]:
                cnt += 1
            if i < n-1 and grid[i+1][j] == grid[i][j]:
                cnt += 1
            if j > 0 and grid[i][j-1] == grid[i][j]:
                cnt += 1
            if j < m-1 and grid[i][j+1] == grid[i][j]:
                cnt += 1
            if cnt > 0:
                return False
    return True

def solve(grid, n, m):
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                grid[i][j] = 'R'
                if is_valid(grid, n, m):
                    return True
                grid[i][j] = 'W'
                if is_valid(grid, n, m):
                    return True
                grid[i][j] = '.'
    return False

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    grid = [list(input()) for _ in range(n)]
    if solve(grid, n, m):
        print("YES")
        for row in grid:
            print("".join(row))
    else:
        print("NO")
