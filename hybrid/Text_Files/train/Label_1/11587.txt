#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    ll n, a, b, k;
    cin >> n >> a >> b >> k;
    string s;
    cin >> s;
    vector<ll> v;
    for (int i = 0; i < n; i++) {
        if (s[i] == '0') { v.push_back(i + 1); }
    }
    if (v.size() <= 2 * (a - 1) + 2) {
        cout << v.size() << '\n';
        for (auto x : v) { cout << x << ' '; }
        cout << '\n';
        return 0;
    }
    for (int i = 0; i < v.size() && k > 0; i++) {
        if (v[i + 1] - v[i] > b) {
            v.erase(v.begin() + i + 1);
            i--;
            k--;
        }
    }
    cout << v.size() << '\n';
    for (auto x : v) { cout << x << ' '; }
    cout << '\n';
    return 0;
}
