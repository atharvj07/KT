import java.util.Scanner;

public class CupcakeShop {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine();

        while (t-- > 0) {
            int n = scanner.nextInt();
            scanner.nextLine();

            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            boolean isYasserHappy = isYasserHappy(a);
            System.out.println(isYasserHappy ? "YES" : "NO");
        }
    }

    private static boolean isYasserHappy(int[] a) {
        int sum = 0;
        for (int x : a) {
            sum += x;
        }

        int maxSegmentSum = 0;
        int currentSegmentSum = 0;
        for (int x : a) {
            currentSegmentSum = Math.max(x, currentSegmentSum + x);
            maxSegmentSum = Math.max(maxSegmentSum, currentSegmentSum);
        }

        return sum > maxSegmentSum;
    }
}
