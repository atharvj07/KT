p,q,l,r=map(int,input().split())
a=[0]*p
b=[0]*p
c=[0]*q
d=[0]*q
cnt=0
for i in range(p):
    a[i],b[i]=map(int,input().split())
for i in range(q):
    c[i],d[i]=map(int,input().split())
for i in range(l,min(r,a[0])+1):
    for j in c:
        if j<=i<=j+d[c.index(j)]:
            cnt+=1
for i in range(max(l,(min(r,a[0])-p+1)),r+1):
    for j in range(p):
        if (a[j]+i-min(r,(min(r,a[0])-p+1)))<=i<=a[j]+i-min(r,(min(r,a[0])-p+1))+b[j]-a[j]:
            cnt+=1
print(cnt)
