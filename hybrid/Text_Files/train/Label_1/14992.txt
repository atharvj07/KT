#include <iostream>
#include <string>

using namespace std;

int main() {
    string s1, s2;
    cin >> s1 >> s2;
    int n = s1.size();
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s1[i] == '0' && s2[i] == '0') {
            cnt++;
        }
    }
    cout << cnt << endl;
    return 0;
}
