import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();
    int[] jars = new int[n];
    for (int i = 0; i < n; i++) {
      jars[i] = input.nextInt();
    }
    Arrays.sort(jars);
    int piglet = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (jars[i] >= k) {
        piglet += jars[i] - k;
        jars[i] = k;
      } else {
        piglet += jars[i];
        jars[i] = 0;
      }
    }
    System.out.println(piglet);
  }
}
