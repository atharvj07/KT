#include <bits/stdc++.h>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> h(n);
    for (int &x : h) cin >> x;
    cout << (h[0] == h[n - 1] ? "YES\n" : "NO\n");
  }
  return 0;
}
