import java.util.*;
class Main {
    static double[] a;
    static int n;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new double[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        double l = -1e8, r = 1e8;
        for (int i = 0; i < 100; i++) {
            double m = (l + r) / 2;
            int p = 0;
            for (int j = 0; j < n; j++) if (a[j] / m > 0) p++;
            if (p >= (n + 1) / 2) r = m;
            else l = m;
        }
        if (r == 1e8) System.out.println(0);
        else System.out.println((int)Math.round(r));
    }
}
