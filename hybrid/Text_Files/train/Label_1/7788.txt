#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;
typedef long long ll;
int main() {
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int q; cin >> q; while (q--) {
        string s, t, p; cin >> s >> t >> p;
        vector<int> f(26), g(26);
        for (auto c : t) ++g[c - 'a'];
        for (auto c : s) ++f[c - 'a'];
        for (auto c : p) {
            if (f[c - 'a'] >= g[c - 'a']) {
                f[c - 'a']--;
                ++g[c - 'a'];
            } else break;
        }
        bool ok = true;
        for (int i = 0; i < 26; ++i) if (g[i] > f[i]) ok = false;
        (ok) ? cout << "YES\n" : cout << "NO\n";
    }
    return 0;
}
