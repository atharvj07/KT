#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define MOD 1000000007
int main() {
    int n;
    cin >> n;
    vector<int> adj[n + 1];
    for (int i = 1; i <= n; i++) {
        int u;
        cin >> u;
        adj[i].push_back(u);
        adj[u].push_back(i);
    }
    ll ans = (1LL << n) - 2;
    for (int i = 1; i <= n; i++) {
        int cnt = 0;
        for (int u : adj[i]) {
            if (u < i) cnt++;
        }
        if (cnt > 1) ans -= (1LL << (n - cnt)) - 1;
    }
    cout << ans % MOD << endl;
    return 0;
}
