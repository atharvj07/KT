import sys

def fastpow(x, p, MOD):
    if p == 0:
        return 1
    if p == 1:
        return x
    val = fastpow(x, int(p / 2), MOD)
    val = (val * val) % MOD
    if p % 2 != 0:
        val = (val * x) % MOD
    return val % MOD

m = int(sys.stdin.readline())
p = list(map(int, sys.stdin.readline().split()))
MOD = int(1e9) + 7
res = 1
inc = 0
p_unique = set()
for i in range(m):
    if p[i] not in p_unique:
        inc += 1
        p_unique.add(p[i])
    res = (res * fastpow(p[i], inc, MOD)) % MOD
print(res)
