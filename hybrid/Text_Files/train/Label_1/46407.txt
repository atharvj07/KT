import java.util.*;

class Main {
    static int N, M;
    static int[] p;
    static ArrayList<ArrayList<Edge>> g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        p = new int[N + 1];
        g = new ArrayList<>();
        for (int i = 0; i <= N; i++) {
            g.add(new ArrayList<>());
        }
        int s = sc.nextInt();
        int t = sc.nextInt();
        for (int i = 1; i <= N; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < M; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            g.get(u).add(new Edge(v, w));
            g.get(v).add(new Edge(u, w));
        }
        int[] sDist = dijkstra(s);
        int[] tDist = dijkstra(t);
        int sScore = 0;
        int tScore = 0;
        boolean sZero = false;
        boolean tZero = false;
        for (int i = 1; i <= N; i++) {
            if (sDist[i] == Integer.MAX_VALUE) {
                if (p[i] > 0) {
                    sZero = true;
                }
            } else if (sDist[i] <= 1000000000) {
                sScore += p[i];
            }
            if (tDist[i] == Integer.MAX_VALUE) {
                if (p[i] > 0) {
                    tZero = true;
                }
            } else if (tDist[i] <= 1000000000) {
                tScore += p[i];
            }
        }
        if (sScore > tScore) {
            System.out.println("Break a heart");
        } else if (tScore > sScore) {
            System.out.println("Cry");
        } else if (sZero && tZero) {
            System.out.println("Flowers");
        } else {
            System.out.println("Flowers");
        }
    }

    static int[] dijkstra(int start) {
        int[] dist = new int[N + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[start] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(start, 0));
        while (!pq.isEmpty()) {
            Edge e = pq.poll();
            if (e.dist > dist[e.v]) {
                continue;
            }
            for (Edge next : g.get(e.v)) {
                int newDist = e.dist + next.w;
                if (newDist < dist[next.v]) {
                    dist[next.v] = newDist;
                    pq.add(new Edge(next.v, newDist));
                }
            }
        }
        return dist;
    }

    static class Edge implements Comparable<Edge> {
        int v;
        int w;
        int dist;

        public Edge(int v, int w) {
            this.v = v;
            this.w = w;
        }

        @Override
        public int compareTo(Edge other) {
            return this.dist - other.dist;
        }
    }
}
