#include <vector>
#include <iostream>
using namespace std;

int main() {
  int t, n, m, x, y;
  cin >> t;
  while (t--) {
    cin >> n >> m >> x >> y;
    vector<string> a(n);
    for (auto &i : a) { cin >> i; }
    long long res = 0;
    for (int i = 0; i < n; i++) {
      bool has_black = false;
      int cnt = 0;
      for (int j = 0; j < m; j++) {
        if (a[i][j] == '*') has_black = true;
        else if (has_black) {
          res += (cnt % 2) * x + (cnt / 2) * y;
          cnt = 0;
          has_black = false;
        } else {
          cnt++;
        }
      }
      if (cnt) res += (cnt % 2) * x + (cnt / 2) * y;
    }
    cout << res << endl;
  }
  return 0;
}
