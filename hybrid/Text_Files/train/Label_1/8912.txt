#include <bits/stdc++.h>

using namespace std;

using ll = long long;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    ll n;
    cin >> n;
    vector<pair<ll, ll>> seg(n);
    for (auto &[l, r] : seg) { cin >> l >> r; }
    sort(seg.begin(), seg.end());
    ll id = -1;
    for (ll i = 1; i < n; i++)
    {
        if (seg[i].first <= seg[id].first && seg[i].second >= seg[id].second)
        {
            cout << i + 1 << " " << id + 1 << '\n';
            return 0;
        }
        if (seg[i].first > seg[id].first) { id = i; }
    }
    cout << "-1 -1\n";
}
