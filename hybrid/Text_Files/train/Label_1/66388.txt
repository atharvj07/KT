import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    l = list(map(int, input().split()))
    odd = [i for i in range(n) if l[i]%2]
    even = [i for i in range(n) if not l[i]%2]
    if (len(odd)+1)//2 <= len(even) or (len(even)+1)//2 <= len(odd): print(0)
    else: print(-1)
