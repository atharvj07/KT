#include <bits/stdc++.h>

using namespace std;

const int MAX_SIZE = 1e5+5;
int a[MAX_SIZE];

int main() {
    int n; cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    
    int left = -1, right = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] > 1) left = i;
        if (a[i] < n - 1) right = i;
    }
    if (left == -1 || right == -1) cout << 0 << endl;
    else cout << min(left, n - right - 1) << endl;
    return 0;
}
