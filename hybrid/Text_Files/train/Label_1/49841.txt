import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] cards = new int[N];
        for (int i = 0; i < N; i++) {
            cards[i] = sc.nextInt();
        }

        Map<Integer, Integer> cardCounts = new HashMap<>();
        for (int card : cards) {
            cardCounts.put(card, cardCounts.getOrDefault(card, 0) + 1);
        }

        int maxPairs = 0;
        for (int card : cardCounts.keySet()) {
            int count = cardCounts.get(card);
            maxPairs += Math.min(count, cardCounts.getOrDefault(card - 1, 0)) + Math.min(count, cardCounts.getOrDefault(card + 1, 0));
        }

        System.out.println(maxPairs);
    }
}
