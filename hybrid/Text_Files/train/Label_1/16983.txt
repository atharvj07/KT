#include <iostream>
#include <string>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  int flag = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' ||
        s[i] == 'u' || s[i] == 'y') {
      if (flag == 1) {
        s.erase(i, 1);
        n--;
        i--;
      } else {
        flag = 1;
      }
    } else {
      flag = 0;
    }
  }
  cout << s << endl;
}
