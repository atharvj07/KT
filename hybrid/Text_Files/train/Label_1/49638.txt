import java.util.Scanner;
import java.util.TreeMap;

public class TelevisionContest {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int teams = scanner.nextInt();
        int records = scanner.nextInt();
        int contestLength = scanner.nextInt();

        scanner.nextLine(); // Consume the newline character

        // Create a TreeMap to store the team ID and the total time they were on TV
        TreeMap<Integer, Integer> teamToTime = new TreeMap<>();

        // Process each record in the log
        for (int i = 0; i < records; i++) {
            int teamId = scanner.nextInt();
            int time = scanner.nextInt();
            int scoreChange = scanner.nextInt();

            // Update the team's total time on TV
            teamToTime.put(teamId, teamToTime.getOrDefault(teamId, 0) + time);
        }

        // Find the team with the maximum time on TV
        int maxTime = 0;
        int teamId = 0;
        for (int key : teamToTime.keySet()) {
            if (teamToTime.get(key) > maxTime) {
                maxTime = teamToTime.get(key);
                teamId = key;
            }
        }

        // Print the team ID of the team that was on TV for the longest time
        System.out.println(teamId);
    }
}
