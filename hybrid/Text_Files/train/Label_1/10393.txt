#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)

int main() {
  int n;
  cin >> n;
  vector<int> ans(n * 2);
  rep(i, n) {
    ans[i * 2] = ans[i * 2 + 1] = i + 1;
  }
  if (n % 2 == 0) {
    rep(i, n / 2 - 1) {
      swap(ans[i * 2 + 1], ans[i * 4 + 2]);
    }
    for (int i = n / 2 - 1, j = n - 1; i >= 0; i--, j--) {
      swap(ans[i * 2 + 1], ans[j * 2 + 1]);
    }
  } else {
    for (int i = n / 2 - 1, j = n - 1; i >= 0; i--, j--) {
      swap(ans[i * 2 + 1], ans[j * 2 + 1]);
    }
  }
  for (int &x : ans) cout << x << ' ';
  cout << endl;
  return 0;
}
