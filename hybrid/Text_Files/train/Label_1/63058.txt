from collections import defaultdict

def calculate_min_sum(n, numbers):
    weight = defaultdict(int)
    non_zero_chars = set()

    for number in numbers:
        length = len(number)
        non_zero_chars.add(number[0])
        for i, char in enumerate(number):
            weight[char] += 10**(length-i-1)

    # Sort characters by their weights
    sorted_chars = sorted(weight, key=lambda x: -weight[x])

    # Assign the smallest non-zero digit to the character with the highest weight not being the first character
    assign = {}
    digit = 1
    for char in sorted_chars:
        if char in non_zero_chars and digit == 1:
            assign[char] = digit
            digit += 1
            break

    # Assign remaining digits
    for char in sorted_chars:
        if char not in assign:
            assign[char] = digit
            digit += 1

    # Calculate sum of restored numbers
    total_sum = 0
    for number in numbers:
        restored_number = 0
        for char in number:
            restored_number = 10 * restored_number + assign[char]
        total_sum += restored_number

    return total_sum

# Reading input
n = int(input())
numbers = [input().strip() for _ in range(n)]

# Calculate and print minimum possible sum
print(calculate_min_sum(n, numbers))
