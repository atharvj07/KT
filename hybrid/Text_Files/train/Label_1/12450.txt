#include<bits/stdc++.h>
using namespace std;
long long arr[200005],ans[200005],L;
int main(){
  long long i,n;
  cin>>n;
  for(i=1;i<=n;i++) cin>>arr[i];
  long long cnt=1;
  ans[cnt]=arr[1];
  for(i=2;i<=n;i++){
    if(arr[i]>ans[cnt]){
      cnt++;
      ans[cnt]=arr[i];
    }
  }
  cout<<cnt<<endl;
  stack <char> s;
  for(i=n;i>0;i--){
    if(arr[i]==ans[cnt]){
      s.push('R');
      cnt--;
    }
    else{
      s.push('L');
    }
  }
  while(!s.empty()){
    cout<<s.top();
    s.pop();
  }
  cout<<endl;
  return 0;
}
