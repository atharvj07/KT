def check(arr, n, m):
    can_move = False
    for i in range(n):
        for j in range(m):
            if arr[i][j] == 0 and (not any(arr[i][k] for k in range(m)) or not any(arr[k][j] for k in range(n))):
                can_move = True

    return can_move


for _ in range(int(input())):
    n, m = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(n)]
    if check(arr, n, m):
        print("Ashish")
    else:
        print("Vivek")
