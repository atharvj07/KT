#include <vector>
#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> h(n), ans(n);

    for (int& x : h)
        cin >> x;

    ans[n-1] = h[n-1];

    for (int i=n-2; i>=0; i--)
        ans[i] = h[i] + ans[i+1];

    for (int x : ans)
        cout << x << " ";

    return 0;
}
