t = int(input().strip())
for _ in range(t):
    A, B, n = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))

    monsters = sorted(zip(a, b), key=lambda x: x[0])
    can_kill_all = True

    for attack, health in monsters:
        fights_needed = (health + A - 1) // A
        B -= fights_needed * attack
        if B - attack <= 0 and (health - ((fights_needed - 1) * A)) > 0:
            can_kill_all = False
            break
        
    if can_kill_all or B > 0:
        print("YES")
    else:
        print("NO")
