#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    int a[n];
    for (int& e : a) cin >> e;
    int ans = 0;
    for (int i = 1; i < n - 1; i++) {
        ans += ((a[i] > a[i - 1] && a[i] > a[i + 1]) || (a[i] < a[i - 1] && a[i] < a[i + 1]));
    }
    cout << ans << endl;
}
