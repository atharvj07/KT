import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            int n = in.nextInt(), r = in.nextInt();
            int[] x = new int[n];
            for (int j = 0; j < n; j++) {
                x[j] = in.nextInt();
            }
            Arrays.sort(x);
            int ans = 0;
            int cur = 1 << 30;
            for (int j = n - 1; j >= 0; j--) {
                if (x[j] <= cur) {
                    ans++;
                    cur = Math.max(0, cur - r);
                } else {
                    cur = x[j] + r;
                }
            }
            System.out.println(ans);
        }
    }
}
