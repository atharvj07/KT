#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2001;
const int MAXM = 2001;
int a[MAXN][MAXM];
int main() {
  int n, m, k, s;
  cin >> n >> m >> k >> s;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  vector<int> seq(s + 1);
  for (int i = 1; i <= s; i++) { cin >> seq[i]; }
  int max_diff = 0;
  int last_x = 1, last_y = 1;
  for (int i = 1; i <= s; i++) {
    for (int x = 1; x <= n; x++) {
      for (int y = 1; y <= m; y++) {
        if (a[x][y] == seq[i]) {
          max_diff = max(max_diff, abs(x - last_x) + abs(y - last_y));
          last_x = x;
          last_y = y;
        }
      }
    }
  }
  cout << max_diff << endl;
  return 0;
}
