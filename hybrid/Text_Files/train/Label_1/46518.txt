import java.util.*;
public class Main {
    static int n, m;
    static int[] p;
    static int[][] g;
    static List<Integer> res;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        p = new int[n + 1];
        g = new int[n + 1][n + 1];
        res = new ArrayList<Integer>();
        for (int i = 0; i < n + 1; i++) {
            p[i] = i;
        }
        int t = 0;
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            String c = sc.next();
            if (c.charAt(0) == 'S') {
                g[a][b] = 1;
                g[b][a] = 1;
            } else {
                p[find(a)] = find(b);
                t++;
            }
        }
        if (t != m / 2) {
            System.out.println("-1");
            return;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                if (i != j && find(i) != find(j) && g[i][j] == 1) {
                    res.add(i);
                    res.add(j);
                    union(i, j);
                }
            }
        }
        System.out.println(res.size() / 2);
        for (Integer x : res) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
    static int find(int x) {
        if (x == p[x])
            return x;
        return p[x] = find(p[x]);
    }
    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            p[y] = x;
        }
    }
}
