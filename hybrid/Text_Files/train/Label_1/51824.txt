import java.util.Scanner;

public class VasyaAndHello {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();

        // Create a Boolean array to track which characters in "hello" have been found
        boolean[] found = new boolean[5];

        // Iterate over the characters in the input string
        for (char c : s.toCharArray()) {
            // If the character is 'h', set the first element of 'found' to true
            if (c == 'h') {
                found[0] = true;
            } else if (found[0] && c == 'e') { // If 'h' has been found and the character is 'e', set the second element of 'found' to true
                found[1] = true;
            } else if (found[1] && c == 'l') { // If 'h' and 'e' have been found and the character is 'l', set the third element of 'found' to true
                found[2] = true;
            } else if (found[2] && c == 'l') { // If 'h', 'e', and one 'l' have been found and the character is 'l', set the fourth element of 'found' to true
                found[3] = true;
            } else if (found[3] && c == 'o') { // If 'h', 'e', and two 'l's have been found and the character is 'o', set the fifth element of 'found' to true
                found[4] = true;
            }
        }

        // Check if all elements of 'found' are true, indicating all characters in "hello" have been found
        if (found[0] && found[1] && found[2] && found[3] && found[4]) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
