#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,i,j,l,r;
    char a[200001];
    cin>>n>>m;
    cin>>a;
    for(i=1;i<=m;++i){
        cin>>l>>r;
        for(j=l;j<=r-1;++j)if(a[j]==a[j+1])break;
        if(j==r)cout<<"0\n";
        else if(l==r)cout<<"1\n";
        else cout<<"2\n";
    }
}
