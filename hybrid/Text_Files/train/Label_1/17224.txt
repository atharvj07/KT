#include <iostream>
#include <stdlib.h>
using namespace std;

typedef struct {double x, y;} point;

inline int abs(int a) { return a < 0 ? -a : a; }

// Calculate the determinant |A| = |[a b][c d]| = a*d - b*c
inline double det(const point &a, const point &b) { return a.x*b.y - a.y*b.x; }

// Check if 2 squares intersect
bool intersect(const point *sq1, const point *sq2)
{
    double a, b, c;
    point o1, o2;
    int i, j;

    // Calculate orientation of each square
    for(i = 0; i < 4; ++i) {
        a = det(sq1[(i+2)%4], sq1[(i+1)%4]);
        b = det(sq2[(i+2)%4], sq2[(i+1)%4]);

        // If opposite sides of the squares are parallel, then they don't intersect
        if(abs(a) < 1e-10 && abs(b) < 1e-10)
            return false;

        // Check orientation of the squares
        if(a < 0) c = 1; else c = -1;
        if(b < 0) o1.x = o1.y = 1; else o1.x = o1.y = -1;

        // Check if the orientations are the same
        if(c != o1.x || c != o1.y)
            return false;
    }

    // Calculate the orientation of the diagonals of each square
    a = det(sq1[2], sq1[0]) * det(sq1[3], sq1[1]);
    b = det(sq2[2], sq2[0]) * det(sq2[3], sq2[1]);

    // Check if the orientations are the same
    if(a < 0) c = 1; else c = -1;
    if(b < 0) o2.x = o2.y = 1; else o2.x = o2.y = -1;

    // If the orientations are different, then the squares don't intersect
    if(c != o2.x || c != o2.y)
        return false;

    // Check if the squares are co-linear
    for(i = 0; i < 4; ++i) {
        a = det(sq1[(i+2)%4], sq1[(i+1)%4]);
        b = det(sq2[(i+2)%4], sq2[(i+1)%4]);

        // If opposite sides of the squares are parallel, then they don't intersect
        if(abs(a) < 1e-10 && abs(b) < 1e-10)
            return false;

        // Check orientation of the squares
        if(a < 0) c = 1; else c = -1;
        if(b < 0) o1.x = o1.y = 1; else o1.x = o1.y = -1;

        // Check if the orientations are the same
        if(c != o1.x || c != o1.y)
            return false;
    }

    // If none of the above conditions are met, then the squares intersect
    return true;
}

int main()
{
    int i;
    point sq1[4], sq2[4];
    bool res;

    for(i = 0; i < 4; ++i)
        cin >> sq1[i].x >> sq1[i].y;

    for(i = 0; i < 4; ++i)
        cin >> sq2[i].x >> sq2[i].y;

    res = intersect(sq1, sq2);

    if(res)
        cout << "YES\n";
    else
        cout << "NO\n";

    return 0;
}
