#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;
ll mod = 1e9 + 7;

vector<int> primes;
vector<int> p;
vector<int> occur;

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    primes.push_back(x);
  }
  for (int i = 2; i <= 200000; i++) {
    if (p.size() == n) break;
    for (int j = 0; j < n; j++) {
      if (i % primes[j] == 0) {
        p.push_back(primes[j]);
        occur.push_back(0);
        break;
      }
    }
  }
  sort(p.begin(), p.end());
  sort(occur.begin(), occur.end());
  ll ans = 1;
  for (int i = 0; i < p.size(); i++) {
    occur[i]++;
    ll mul = 1;
    for (int j = 0; j < occur[i]; j++) {
      mul *= p[i];
      mul %= mod;
    }
    ans *= mul;
    ans %= mod;
  }
  cout << ans << "\n";
}
