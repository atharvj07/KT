from bisect import bisect_left, bisect_right

n, q, c = map(int, input().split())
stars = []
for _ in range(n):
    x, y, s = map(int, input().split())
    stars.append((x, y, s))
stars.sort()

for _ in range(q):
    t, x1, y1, x2, y2 = map(int, input().split())
    l = bisect_left(stars, (x1, y1))
    r = bisect_right(stars, (x2, y2))
    ans = sum(s + min(t, c - s) for x, y, s in stars[l:r])
    print(ans)
