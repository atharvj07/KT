#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    ll n,x; cin>>n>>x;
    vector<ll> freq(x+1);
    for (ll i=0, tmp; i<n; i++) {
        cin>>tmp; freq[tmp]++;
    }
    for (ll i=1; i<=x; i++) freq[i]+=freq[i-1];
    ll ans=0;
    for (ll i=1; i<=x; i++) {
        for (ll j=i; j<=x; j++) {
            if (freq[j]-freq[i-1]==j-i+1) ans++;
        }
    }
    cout<<ans<<'\n';
    return 0;
}
