from collections import defaultdict, Counter

n, m = map(int, input().split())
a = []
for i in range(n):
    a.append(Counter(list(map(int, input().split()))))

b = defaultdict(int)
for i in range(1, n*m+1):
    b[i] = 0

for i in a:
    for j in i:
        b[j] += i[j]

ans = 0
for i in b:
    if b[i] == 0:
        ans += 1
    elif b[i] < 0:
        ans += abs(b[i]) - 1
    elif b[i] > 1:
        ans += b[i] - 1

print(ans)
