def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n, m = map(int, input().split())
segments = [tuple(sorted(map(int, input().split()))) for _ in range(m)]

rotation_possibilities = set()

for a, b in segments:
    rotation_possibilities.add((b - a) % n)
    rotation_possibilities.add((n - (b - a)) % n)

result = "No"
for k in rotation_possibilities:
    if all(((a + k) % n, (b + k) % n) in segments or ((b + k) % n, (a + k) % n) in segments for a, b in segments):
        result = "Yes"
        break

if result == "Yes":
    gcd_result = gcd(n, k)
    if gcd_result > 1:
        for i in range(1, gcd_result):
            if n % gcd_result == 0 and all(((a + (k * i) % n) % n, (b + (k * i) % n) % n) in segments or ((b + (k * i) % n) % n, (a + (k * i) % n) % n) in segments for a, b in segments):
                print("Yes")
                break
        else:
            print("No")
    else:
        print("Yes")
else:
    print("No")
