#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,m; cin>>n>>m;
  int u,v;
  vector<vector<int>>adj(n+1);
  for(int i=0; i<m; i++){
    cin>>u>>v;
    adj[u].push_back(v);
  }
  vector<int>vis(n+1,0),stk(n+1,0);
  function<bool(int)> dfs=[&](int u){
    vis[u]=1; stk[u]=1;
    for(auto v:adj[u]){
      if(!vis[v]){
        if(dfs(v)) return true;
      }else if(stk[v]) return true;
    }
    stk[u]=0;
    return false;
  };
  for(int i=1; i<=n; i++)
    if(!vis[i]&&dfs(i)){
      cout<<"YES"; return 0;}
  cout<<"NO";
  return 0;
}
