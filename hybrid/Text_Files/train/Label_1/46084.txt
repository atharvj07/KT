import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        for (int i = 0; i < m; i++) {
            int type = sc.nextInt();

            switch (type) {
                case 1:
                    int index = sc.nextInt() - 1;
                    arr[index] = sc.nextInt();
                    break;

                case 2:
                    long increment = sc.nextInt();
                    for (int j = 0; j < n; j++) {
                        arr[j] += increment;
                    }
                    break;

                case 3:
                    index = sc.nextInt() - 1;
                    System.out.println(arr[index]);
                    break;
            }
        }
    }
}
