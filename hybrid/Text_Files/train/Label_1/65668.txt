import sys
input = sys.stdin.readline

def dfs(v, clr):
    vis[v] = clr
    for u in adj[v]:
        if vis[u] == -1:
            if not dfs(u, clr ^ 1):
                return False
        elif clr == vis[u]:
            return False
    return True

n, m = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
vis = [-1] * (n+1)
ans = [[], []]
for i in range(1, n+1):
    if vis[i] == -1:
        if not dfs(i, 0):
            print(-1)
            sys.exit()
        else:
            ans[0].append(i)
            for j in range(1, n+1):
                if vis[j] == 0:
                    ans[1].append(1)
                else:
                    ans[1].append(2)
k = len(set(ans[1]))
print(k)
print(*ans[1])
