#include <algorithm>
#include <iostream>
#include <utility>
#include <vector>

int main() {
  int t, n, l, r;
  std::cin >> t;
  while (t--) {
    std::cin >> n >> l >> r;
    std::vector<int> ans;
    for (int i = (n - 1); i >= 1; i--)
      for (int j = 1; j <= n; j++)
        if (ans.size() < r) {
          ans.push_back(j);
          ans.push_back(i);
        }
    for (int i = l; i <= r; i++)
      std::cout << ans[i - 1] << " ";
    std::cout << "\n";
  }
  return 0;
}
