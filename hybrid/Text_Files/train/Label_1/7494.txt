#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b, c;
        cin >> a >> b >> c;
        long long x = (c * 2 - a - b) % 2;
        if (x < 0)
            x += 2;
        x = b + x;
        if (a > b && (x < a || x > b))
            x += 2;
        if (x == a || x == b || x > c)
            cout << -1 << endl;
        else
            cout << x << endl;
    }
    return 0;
}
