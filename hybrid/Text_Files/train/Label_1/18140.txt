#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
#define int long long
#define all(v) (v).begin(), (v).end()
void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    set<int> b;
    int ans = 0;
    int l = 1e9;
    for (int i = 0; i < k; ++i) {
        int x;
        cin >> x;
        b.insert(x - 1);
    }
    for (int i = n - 1; i >= 0; --i) {
        if (b.find(i) != b.end()) {
            continue;
        }
        if (a[i] < l)
            ans++;
        l = a[i];
    }
    if (ans > n){
     cout<<-1<<endl;
     return;
    }
    cout << ans << endl;
}
int32_t main() { solve(); }
