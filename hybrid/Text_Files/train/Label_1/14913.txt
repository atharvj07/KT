#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m, k;
    scanf(" %d %d %d", &n, &m, &k);
    vector<int> b(n);
    for (int i = 0; i < n; ++i) scanf(" %d", &b[i]);
    if (k == n) {
        long long ans = 0;
        for (int x : b) ans += x;
        printf("%lld\n", ans);
        return 0;
    }
    vector<int> dif(n);
    for (int i = 1; i < n; ++i) dif[i] = b[i] - b[i - 1] - 1;
    long long ans = b[n - 1];
    sort(dif.begin(), dif.end());
    for (int i = 0; i < n - k; ++i) ans += dif[i];
    printf("%lld\n", ans);
}
