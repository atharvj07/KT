#include <algorithm>
#include <iostream>
#include <string>

int main() {
  std::string t, p;
  std::cin >> t >> p;
  int n = t.size(), m = p.size();
  int ans = 0;
  while (ans < n && t[ans] == p[ans]) ans++;
  std::string s = t;
  std::reverse(t.begin(), t.end());
  int anst = 0;
  while (anst < n && t[anst] == p[anst]) anst++;
  ans = std::max(ans, anst);
  for (int i = 0; i < n; i++) {
    char c = s[i];
    t[i] = c;
    int anst = 0;
    while (anst < n && t[anst] == p[anst]) anst++;
    ans = std::max(ans, anst);
  }
  std::cout << ans << std::endl;
}
