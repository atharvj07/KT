#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n];
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    int cnt=0;
    for(int i=0;i<n;i++)
    {
        if(a[i])
        {
            cnt++;
        }
    }
    if(cnt==0)
    {
        cout<<0<<endl;
    }
    if(cnt==1)
    {
        cout<<1<<endl;
    }
    if(cnt>1)
    {
        cout<<cnt-1<<endl;
    }
}
