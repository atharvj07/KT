import sys
input = sys.stdin.readline

n = int(input())
m = list(map(int, input().split()))
# Make a list of all possible tower heights for each plot
# For each plot, the max height is the min of its limit and the height of the previous plot
h = [min(m[i], h[i - 1] if i > 0 else 0) for i in range(n)]
# Make a list of all possible tower heights for each plot
# For each plot, the min height is the max of its limit and the height of the next plot
h = [max(m[i], h[i + 1] if i < n - 1 else 0) for i in range(n - 1, -1, -1)]
# Print the list of all possible tower heights for each plot
print(*h)
