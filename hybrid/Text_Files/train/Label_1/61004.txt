n = int(input().strip())
for _ in range(n):
    s = input().strip()
    s_sorted = sorted(s)
    is_diverse = True
    for i in range(1, len(s_sorted)):
        if ord(s_sorted[i]) - ord(s_sorted[i-1]) != 1:
            is_diverse = False
            break
    if is_diverse:
        print("Yes")
    else:
        print("No")
