import sys
input = sys.stdin.readline
mod = 10 ** 9 + 7
n = int(input())
a = list(map(int, input().split()))
dp = {x: [0 for j in range(10)] for x in a}
for i in sorted(list(set(a))):
    dp[i][1] = 1
for j in range(2, 10):
    for i in sorted(list(set(a))):
        for k in sorted(list(set(a))):
            if i % j == 0 and k % j == 0 and dp[k][j - 1] > 0:
                dp[i][j] = (dp[i][j] + dp[k][j - 1]) % mod
res = 0
for i in dp:
    for j in range(1, 10):
        res = (res + dp[i][j]) % mod
print(res)
