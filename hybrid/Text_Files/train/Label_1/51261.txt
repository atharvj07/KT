import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = scanner.nextInt();
            }
        }
        int[][] b = new int[n][m];
        ArrayList<String> ans = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m - 1; j++) {
                if (a[i][j] == 1 && a[i][j + 1] == 1 && a[i + 1][j] == 1 && a[i + 1][j + 1] == 1) {
                    continue;
                }
                if (a[i][j] == 0 && a[i][j + 1] == 0 && a[i + 1][j] == 0 && a[i + 1][j + 1] == 0) {
                    continue;
                }
                if (a[i][j] == 0 && a[i][j + 1] == 1 && a[i + 1][j] == 1 && a[i + 1][j + 1] == 0) {
                    b[i][j + 1] = 1;
                    b[i + 1][j + 1] = 1;
                    ans.add((i + 1) + " " + (j + 2));
                } else if (a[i][j] == 1 && a[i][j + 1] == 0 && a[i + 1][j] == 1 && a[i + 1][j + 1] == 0) {
                    b[i][j] = 1;
                    b[i + 1][j] = 1;
                    ans.add((i + 1) + " " + (j + 1));
                } else if (a[i][j] == 1 && a[i][j + 1] == 1 && a[i + 1][j] == 0 && a[i + 1][j + 1] == 0) {
                    b[i + 1][j] = 1;
                    b[i + 1][j + 1] = 1;
                    ans.add((i + 2) + " " + (j + 1));
                } else if (a[i][j] == 0 && a[i][j + 1] == 0 && a[i + 1][j] == 1 && a[i + 1][j + 1] == 1) {
                    b[i][j] = 1;
                    b[i][j + 1] = 1;
                    ans.add((i + 1) + " " + (j + 1));
                } else {
                    System.out.println(-1);
                    return;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] != b[i][j]) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        System.out.println(ans.size());
        for (String s : ans) {
            System.out.println(s);
        }
    }
}
