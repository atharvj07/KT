n, a = map(int, input().split())
cities = list(map(int, input().split()))

caught = cities[a-1]  # Catch the criminal in Limak's city if present.
for d in range(1, n):  # Check each distance.
    left = a - 1 - d
    right = a - 1 + d
    if left >= 0 and right < n:  # Both cities exist.
        caught += (cities[left] and cities[right])
    elif left < 0 and right < n:  # Only right city exists.
        caught += cities[right]
    elif right >= n and left >= 0:  # Only left city exists.
        caught += cities[left]

print(caught)
