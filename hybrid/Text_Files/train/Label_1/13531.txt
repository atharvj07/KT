#include <iostream>
using namespace std;
typedef long long ll;
int main() {
    ll a, b, x, y;
    cin >> a >> b >> x >> y;
    ll l = b / y + 1;
    ll r = a / x + 1;
    ll ans = (b / l) * (a / r);
    l = (b - 1) / y + 1;
    r = (a - 1) / x + 1;
    ans -= (b - 1) / l * (a - 1) / r;
    cout << ans << "\n";
}
