n, k = list(map(int, input().split()))
s = input()
suff_cost = [0] * n
cost = 0
cnt = 0

for i in range(n - 1, - 1, - 1):
    if s[i] != s[i - 1]:
        cost += i
        cnt += 1
    suff_cost[i] = (cost, cnt)

ans = -1
for i in range(n):
    if cnt + i >= k:
        x = suff_cost[i][0] + (cnt + i - k) * n
        if ans == -1 or ans > x:
            ans = x

print(ans)
