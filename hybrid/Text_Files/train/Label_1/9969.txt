#include <algorithm>
#include <iostream>
using namespace std;

int main() {
  string arr[8];
  for (int i = 0; i < 8; i++) { cin >> arr[i]; }

  for (int j = 0; j < 8; j++) {
    rotate(arr[j].begin(), arr[j].begin() + 1, arr[j].end());
    bool flag = true;
    for (int i = 1; i < 8; i++) {
      if (arr[i][0] == arr[i - 1][0]) {
        flag = false;
        break;
      }
    }
    if (flag) {
      cout << "YES" << endl;
      return 0;
    }
    rotate(arr[j].begin(), arr[j].begin() + 1, arr[j].end());
  }
  cout << "NO" << endl;
  return 0;
}
