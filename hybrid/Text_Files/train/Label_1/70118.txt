n, k = map(int, input().split())
s = input()
t = []
cnt = 0
ans = 0
for x in range(n):
    if s[x] == 'a':
        cnt += 1
    else:
        ans = max(ans, cnt)
        cnt = 0
ans = max(ans, cnt)
if ans == n:
    print(n)
else:
    aa = (ans + 1) // 2
    bb = ans - aa
    if aa + bb <= k:
        ans += min(aa, k - aa - bb)
    print(ans)
