#include <bits/stdc++.h>
#include <vector>
using namespace std;

int main() {
  int T;
  cin >> T;
  while (T--) {
    string s;
    cin >> s;
    int n = s.length();
    vector<vector<int>> dp(n + 2, vector<int>(n + 2, false));
    dp[n][n] = true;
    for (int i = n - 1; i >= 0; i--) {
      for (int j = n - 1; j >= 0; j--) {
        if (s[i] == s[j] && dp[i + 1][j + 1]) dp[i][j] = true;
        if (s[i] != s[j] && dp[i + 1][j]) dp[i][j] = true;
      }
    }
    string ans = s;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        if (dp[i][j] && lexicographical_compare(ans.begin(), ans.end(),
                                                  s.begin() + i, s.begin() + j))
          ans = s.substr(i, j - i + 1);
      }
    }
    string t = ans;
    reverse(t.begin(), t.end());
    while (t.length() < s.length()) t += '0';
    if (t == ans) cout << s << endl;
    else {
      int m = s.length() - ans.length();
      int k = m / 2;
      for (int i = 0; i < k; i++) {
        ans[i] = s[i] == '0' ? '1' : '0';
        ans[s.length() - 1 - i] = s[s.length() - 1 - i] == '0' ? '1' : '0';
      }
      cout << ans << endl;
    }
  }
}
