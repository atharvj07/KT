#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int n, a[51];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  for (int i = 0; i < n - 2; i++) {
    if (a[i + 1] == a[i + 2]) cout << "NO", exit(0);
    if (a[i + 2] - a[i] > 2) cout << "NO", exit(0);
  }
  cout << "YES";
  return 0;
}
