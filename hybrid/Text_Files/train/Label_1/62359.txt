def rook(x0, y0, x):
    global ans
    if abs(x0 - x) == abs(y0 - y):
        ans = "YES"

def bishop(x0, y0, x):
    global ans
    if abs(x0 - x) == abs(y0 - y):
        ans = "YES"

def queen(x0, y0, x):
    global ans
    ans = "YES"


n = int(input())
x0, y0 = map(int, input().split())
for i in range(n):
    c, x, y = input().split()
    x, y = int(x), int(y)
    ans = "NO"
    rook(x0, y0, x)
    bishop(x0, y0, x)
    queen(x0, y0, x)
    if ans == "YES":
        break
print(ans)
