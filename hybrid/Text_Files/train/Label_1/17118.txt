#include<bits/stdc++.h>
using namespace std;
const int N=5e5+10;
int n,m,p,vis[N];
char s[N];
int main(){
    cin>>n>>m>>p,cin>>s;
    for(int i=1;i<=m;i++){
        char ch;cin>>ch;
        if(ch=='L')p--;
        else if(ch=='R')p++;
        else{
            int tmp=p-1;
            vis[tmp]=1;
            while(s[tmp]!=s[p] && tmp>=1)tmp--;
            vis[p]=1;
            p=tmp+1;
        }
    }
    for(int i=1;i<=n;i++)if(!vis[i])cout<<s[i];
    cout<<endl;
}
