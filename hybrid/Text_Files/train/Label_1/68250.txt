n = int(input())
a = list(map(int, input().split()))
# calculate prefix xor array
b = [0] * n
b[0] = a[0]
for i in range(1, n):
    b[i] = b[i - 1] ^ a[i]
# find maximum xor using prefix xor array
mx = 0
for i in range(n):
    for j in range(i, n):
        mx = max(mx, b[j] ^ (b[i - 1] if i > 0 else 0))
print(mx)
