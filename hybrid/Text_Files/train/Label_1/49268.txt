import java.util.*;
import java.math.*;

class Main {
    static final int mod = 1000000007;

    static BigInteger pow(BigInteger a, long b, int p) {
        BigInteger ans = BigInteger.ONE;
        while (b > 0) {
            if ((b & 1) != 0) ans = ans.multiply(a).mod(BigInteger.valueOf(p));
            a = a.multiply(a).mod(BigInteger.valueOf(p));
            b >>= 1;
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        BigInteger ans = BigInteger.valueOf(1);
        for (int v : p) {
            ans = ans.multiply(BigInteger.valueOf(v)).mod(BigInteger.valueOf(mod));
        }
        System.out.println(ans);
    }
}
