import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int sLen = sc.nextInt();
        int[] s = new int[sLen];
        for (int i = 0; i < sLen; i++) {
            s[i] = sc.nextInt();
        }
        int qLen = sc.nextInt();
        int[] q = new int[qLen];
        for (int i = 0; i < qLen; i++) {
            q[i] = sc.nextInt();
        }

        int cLen = 0;
        for (int qItem : q) {
            int l = 0, r = sLen - 1;
            while (l <= r) {
                int m = (l + r) >> 1;
                if (s[m] == qItem) {
                    cLen++;
                    break;
                } else if (s[m] > qItem) {
                    r = m - 1;
                } else {
                    l = m + 1;
                }
            }
        }

        System.out.println(cLen);
    }
}
