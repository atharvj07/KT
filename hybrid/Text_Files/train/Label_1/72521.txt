n, k = map(int, input().split())
p = [int(x) for x in input().split()]
q = [int(x) for x in input().split()]
t = [0] * (n + 1)
for i in range(n):
    t[p[i]] = q[i]
if len(set(t)) < k:
    print("NO")
else:
    print("YES")
    s = ''
    for i in range(1, n + 1):
        if len(set(s)) == k:
            s += 'z'
            continue
        ch = chr(ord('a') + len(set(s)))
        while ch in s:
            ch = chr(ord(ch) + 1)
        s += ch
    print(s)
