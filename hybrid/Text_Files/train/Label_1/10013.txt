#include <bits/stdc++.h>
using namespace std;
#define int long long 
main(){
	int n, sumX = 0, x;
	cin >> n;
	vector<int> v(n);
	for(int i = 0; i < n; i++){
		cin >> x;
		v[i] = x;
		sumX += x;
	}
	sort(v.begin(), v.end());
	double l = v[(n - 1) / 2 - 1], r = v[n / 2];
	if((sumX - l * n) < (r * n - sumX)){
		cout << l;
	}
	else{
	   cout << r;
	}
	return 0;
}
