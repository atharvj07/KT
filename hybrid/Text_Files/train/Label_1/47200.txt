import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        short n = sc.nextShort();
        int m = sc.nextInt();
        short k = sc.nextShort();
        short s = sc.nextShort();
        long p = sc.nextInt();
        long q = sc.nextInt();
        boolean[] zombies = new boolean[n + 1];
        for (short i = 0; i < k; i++) zombies[sc.nextInt()] = true;
        int[] edges = new int[m * 2];
        for (int i = 0; i < m; i++) {
            short a = sc.nextShort(), b = sc.nextShort();
            edges[i * 2] = a; edges[i * 2 + 1] = b;
        }
        int[] dist = new int[n + 1];
        Arrays.fill(dist, -1);
        boolean[] queue = new boolean[n + 1];
        dist[1] = 0;
        queue[1] = true;
        for (int i = 0, j; i < n; i++) if (queue[i]) {
            for (j = 0; j < m * 2; j++)
                if (edges[j] == i) queue[edges[j ^ 1]] |= dist[edges[j ^ 1]] >= dist[i] - s;
            j = i * 2;
            if (dist[edges[j]] == dist[i] + 1) queue[edges[j]] |= queue[edges[j ^ 1]];
            if (dist[edges[j ^ 1]] == dist[i] + 1) queue[edges[j ^ 1]] |= queue[edges[j]];
            dist[edges[j]] = dist[edges[j ^ 1]] = dist[i] + 1;
        }
        System.out.println(queue[n] ? dist[n] * p : dist[n] * q);
    }
}
