n = int(input())
p = list(map(int, input().split()))

record_breaks = [False] * n
max_so_far = 0

for i in range(n):
    if p[i] > max_so_far:
        max_so_far = p[i]
        record_breaks[i] = True

to_remove = [0] * (n + 1)
for i in range(1, n):
    if record_breaks[i]:
        to_remove[p[i - 1]] += 1

to_remove[p[n - 1]] += 1
print(to_remove.index(max(to_remove)))
