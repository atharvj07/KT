n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
print(['maybe', 'rated', 'unrated'][
    all(ai == bi for ai, bi in a) and sorted(ai for ai, _ in a) == sorted(bi for _, bi in a) or
    any(ai != bi for ai, bi in a) or
    any(ai > bi for ai, bi in a) and any(bi > ai for ai, bi in a)
])
