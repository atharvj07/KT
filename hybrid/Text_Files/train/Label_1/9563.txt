#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  if (n == 1) {
    cout << a[0] << endl;
  } else if (n == 2) {
    cout << max(0LL, a[0] + a[1]) << endl;
  } else {
    long long sum = 0, sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; i++) { sum += a[i]; }
    sum /= 2;
    for (int i = 0; i < n; i++) {
      sum1 += a[i];
      if (sum1 == sum) {
        sum2 = sum - sum1;
        break;
      }
    }
    if (sum1 == sum && sum1 == sum2) {
      cout << sum1 << endl;
    } else {
      cout << 0 << endl;
    }
  }
  return 0;
}
