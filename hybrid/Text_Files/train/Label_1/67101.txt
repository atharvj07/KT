import sys

def main():
    t = int(sys.stdin.readline())
    for _ in range(t):
        s = sys.stdin.readline().strip()
        if len(s) % 2 == 0:
            print(''.join(sorted(s)))
        else:
            if s[0:len(s) // 2] != s[len(s) // 2 + 1:len(s)]:
                print(''.join(sorted(s)))
            else:
                print(-1)

main()
