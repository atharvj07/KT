#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> t(n);
  for (int i = 0; i < n; ++i) {
    cin >> t[i];
  }

  int d = t[1] - t[0];
  bool is_ap = true;
  for (int i = 2; i < n; ++i) {
    if (t[i] - t[i - 1] != d) {
      is_ap = false;
      break;
    }
  }

  if (is_ap) {
    cout << t[n - 1] + d << "\n";
  } else {
    cout << t[n - 1] << "\n";
  }
}
