#include<vector>
#include<iostream>
using namespace std;
const int maxn = 3e5 + 123;
typedef long long ll;
vector<pair<ll, ll>> v[maxn];
int vis[maxn];
ll dfs(int p, ll sum) {
  vis[p] = 1;
  ll ans = sum;
  for (auto x : v[p]) {
    if (!vis[x.first]) {
      ans = min(ans, dfs(x.first, sum + x.second) + 1);
    }
  }
  return ans;
}
int main() {
  ll n, m, a, b;
  ios::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i <= n; i++) {
      v[i].clear();
      vis[i] = 0;
    }
    for (int i = 1; i <= n; i++) {
      cin >> a >> b;
      v[i].push_back({(i + 1) % (n + 1), b});
    }
    cout << dfs(1, 0) << endl;
  }
}
