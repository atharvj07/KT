import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == -1)
                break;
            boolean[] display = new boolean[7];
            for (int i = 0; i < n; i++) {
                int num = sc.nextInt();
                if (num == 0) {
                    display[0] = true;
                    display[1] = true;
                    display[2] = true;
                    display[4] = true;
                    display[5] = true;
                    display[6] = true;
                } else if (num == 1) {
                    display[1] = true;
                    display[2] = true;
                } else if (num == 2) {
                    display[0] = true;
                    display[1] = true;
                    display[3] = true;
                    display[4] = true;
                    display[6] = true;
                } else if (num == 3) {
                    display[0] = true;
                    display[1] = true;
                    display[2] = true;
                    display[3] = true;
                    display[6] = true;
                } else if (num == 4) {
                    display[2] = true;
                    display[3] = true;
                    display[4] = true;
                    display[5] = true;
                } else if (num == 5) {
                    display[0] = true;
                    display[2] = true;
                    display[3] = true;
                    display[5] = true;
                    display[6] = true;
                } else if (num == 6) {
                    display[0] = true;
                    display[2] = true;
                    display[3] = true;
                    display[4] = true;
                    display[5] = true;
                    display[6] = true;
                } else if (num == 7) {
                    display[0] = true;
                    display[1] = true;
                    display[2] = true;
                } else if (num == 8) {
                    display[0] = true;
                    display[1] = true;
                    display[2] = true;
                    display[3] = true;
                    display[4] = true;
                    display[5] = true;
                    display[6] = true;
                } else if (num == 9) {
                    display[0] = true;
                    display[1] = true;
                    display[2] = true;
                    display[3] = true;
                    display[5] = true;
                    display[6] = true;
                }
            }
            for (int i = 6; i >= 0; i--) {
                System.out.print(display[i] ? '1' : '0');
            }
            System.out.println();
        }
    }
}
