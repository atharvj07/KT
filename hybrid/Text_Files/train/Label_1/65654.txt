n, x = map(int, input().split())
a = list(map(int, input().split()))
ans = 0
pref = [0]
for i in a: pref.append(i)
for i in range(1, n+1): pref[i] += pref[i-1]
for i in range(n):
    l, r = i, i
    while l >= 0 and r < n and a[l] * x >= 0 and a[r] * x >= 0:
        l -= 1
        r += 1
    ans = max(ans, pref[r] - pref[l])
print(ans)
