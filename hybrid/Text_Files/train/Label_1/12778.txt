#include <iostream>
#include <algorithm>
using namespace std;
#define rep(i,n) for(int i=0;i<(n);i++)

const int N = 105;
double f[N][N][N];
int n, k, p[N];

double dp(int l, int r, int inv) {
  if (l > r) return 0;
  double& res = f[l][r][inv];
  if (res != -1.0) return res;
  res = (inv + double(r - l + 1) * (r - l)) / (r - l + 2);
  for (int i = l; i <= r; i++) {
    res += dp(l, i - 1, inv + i - l) * dp(i + 1, r, inv + r - i) / (r - l + 1);
  }
  return res;
}

int main() {
  cin >> n >> k;
  rep(i,n) {
    int x; cin >> x;
    x--;
    p[x] = i;
  }
  for (int _loop = 0; _loop <= k; _loop++)
    for (int l = 0; l < n; l++)
      for (int r = 0; r < n; r++)
        f[l][r][_loop] = -1.0;
  double ans = dp(0, n - 1, 0);
  printf("%.10f\n", ans);
  return 0;
}
