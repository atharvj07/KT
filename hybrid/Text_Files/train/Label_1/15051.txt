#include <iostream>
#include <set>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  set<int> s;
  for (int i = 0; i < k; i++) {
    int x;
    cin >> x;
    s.insert(a[x - 1]);
  }
  if (s.size() == k)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
