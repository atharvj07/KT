#include <bits/stdc++.h>
using namespace std;
int main() {
  int t = 0;
  cin >> t;
  while (t--) {
    int n, c0, c1, h;
    cin >> n >> c0 >> c1 >> h;
    string s;
    cin >> s;
    int ans = 0;
    for (auto x : s) {
      if (x == '1')
        ans += min(c1, c0 + h);
      else
        ans += min(c0, c1 + h);
    }
    cout << ans << endl;
  }
}
