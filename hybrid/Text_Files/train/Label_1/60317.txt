def can_transform(q, queries):
    results = []
    for i in range(q):
        n = queries[i][0]
        s = queries[i][1]
        t = queries[i][2]
        
        # Check for character frequency
        if sorted(s) != sorted(t):
            results.append("NO")
            continue
        
        # Check for at least one character appearing more than once
        if len(set(s)) < n:
            results.append("YES")
            continue
        
        # Check for parity of inversions
        inv_s = sum(1 for i in range(n) for j in range(i + 1, n) if s[i] > s[j])
        inv_t = sum(1 for i in range(n) for j in range(i + 1, n) if t[i] > t[j])
        
        if inv_s % 2 == inv_t % 2:
            results.append("YES")
        else:
            results.append("NO")
    
    return results

# Reading from standard input
q = int(input().strip())
queries = []
for _ in range(q):
    n = int(input().strip())
    s = input().strip()
    t = input().strip()
    queries.append((n, s, t))

# Processing queries
results = can_transform(q, queries)

# Writing to standard output
for result in results:
    print(result)
