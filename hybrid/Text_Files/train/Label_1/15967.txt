#include <bits/stdc++.h>
using namespace std;

const int N = 205;

struct edge {
    int u, v, w;
};

vector<int> adj[N];
vector<edge> edges;
int n, k;
int a[N];
int dp[N][N][2]; // dp[v][msk][taken] = max weight if we are at v, mask shows deleted, taken = 0/1

int dfs(int v, int prv, int msk, bool taken) {
    int &ret = dp[v][msk][taken];
    if (~ret)
        return ret;
    ret = 0;
    if (!taken)
        ret = a[v];

    for (int u : adj[v]) {
        if (u == prv)
            continue;
        if ((msk & (1 << u)) == 0) {
            ret = max(ret, dfs(u, v, msk, taken));
            ret = max(ret, dfs(u, v, msk | (1 << u), true));
        }
    }
    return ret;
}

int main() {
    memset(dp, -1, sizeof dp);
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        edges.push_back({u, v, 1});
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int ans = dfs(1, 1, 0, false);

    cout << ans << '\n';

    return 0;
}
