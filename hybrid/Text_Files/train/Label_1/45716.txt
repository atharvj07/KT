import java.util.Scanner;

public class Main {
    static int n, m;
    static int[] a, b;
    static int[][] c;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        a = new int[n];
        b = new int[m];
        c = new int[n][m];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }
        if (!check()) {
            System.out.println("NO");
            return;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                c[i][j] = a[i] ^ b[j];
            }
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(c[i][j] + " ");
            }
            System.out.println();
        }
    }

    static boolean check() {
        int x = 0;
        for (int i = 0; i < n; i++) {
            x ^= a[i];
        }
        for (int i = 0; i < m; i++) {
            x ^= b[i];
        }
        return x == 0;
    }
}
