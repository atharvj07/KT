#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int ans = 0;
    for (int i = 0, j = 0; i < n; i++) {
        while (j < n && b[j] < a[i]) j++;
        if (j < n) {
            b[j] -= a[i];
            k -= a[i];
            ans++;
            j++;
        } else {
            if (k >= a[i]) {
                k -= a[i];
                ans++;
            } else
                break;
        }
    }
    cout << ans << endl;
    return 0;
}
