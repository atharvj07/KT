#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int nr, ng, nb;
    cin >> nr >> ng >> nb;
    vector<int> r(nr), g(ng), b(nb);
    for (int i = 0; i < nr; i++) cin >> r[i];
    for (int i = 0; i < ng; i++) cin >> g[i];
    for (int i = 0; i < nb; i++) cin >> b[i];
    sort(r.begin(), r.end());
    sort(g.begin(), g.end());
    sort(b.begin(), b.end());
    int ans = (r[nr / 2] - g[ng / 2]) * (r[nr / 2] - g[ng / 2]);
    ans += (g[ng / 2] - b[nb / 2]) * (g[ng / 2] - b[nb / 2]);
    ans += (b[nb / 2] - r[nr / 2]) * (b[nb / 2] - r[nr / 2]);
    cout << ans << endl;
  }
  return 0;
}
