import sys

n, m = map(int, sys.stdin.readline().split())
last = [0] * (n + 1)
for _ in range(m):
    cmd, id = sys.stdin.readline().split()
    id = int(id)
    if cmd[0] == '-':
        last[id] = -1
    else:
        last[id] = 1
L = filter(lambda x: x[1] == 1, enumerate(last))
k = sum(map(lambda x: x[1], L))
print(k)
if k > 0:
    print(' '.join(map(lambda x: str(x[0]), L)))
