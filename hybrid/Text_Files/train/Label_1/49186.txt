import java.util.Scanner;

public class CountryPathCounter {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] interLayerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            interLayerCosts[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        int[][] dp = new int[l + 1][m + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= l; i++) {
            for (int j = 0; j <= m; j++) {
                for (int k = 0; k < n; k++) {
                    int cost = (entryCosts[k] + (i - 1) * interLayerCosts[k] + exitCosts[k]) % m;
                    if (j - cost >= 0) {
                        dp[i][j] = (dp[i][j] + dp[i - 1][j - cost]) % MOD;
                    }
                }
            }
        }

        System.out.println(dp[l][m]);
    }
}
