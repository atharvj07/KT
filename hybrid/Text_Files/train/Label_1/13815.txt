#include <iostream>
#include <string>

int main() {
  std::string s;
  std::cin >> s;

  int a = 0, b = 0, c = 0;
  for (int i = 0; i < s.length(); i++) {
    int d = s[i] - '0';
    if (a + b + c == 0) {
      a = d;
    } else if (a + b <= d) {
      b = a;
      a = d;
    } else if (a <= d) {
      b = d;
    } else {
      c = d;
    }
  }
  std::cout << a << '+' << b << '=' << c << std::endl;

  return 0;
}
