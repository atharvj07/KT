n, L = map(int, input().split())
kefa = list(map(int, input().split()))
sasha = list(map(int, input().split()))

# Compute the differences between adjacent barriers for both Kefa and Sasha
kefa_diffs = [(kefa[i+1] - kefa[i]) % L for i in range(n-1)] + [(L - kefa[-1] + kefa[0]) % L]
sasha_diffs = [(sasha[i+1] - sasha[i]) % L for i in range(n-1)] + [(L - sasha[-1] + sasha[0]) % L]

# Check if one sequence can be rotated to match the other
for i in range(n):
    if kefa_diffs == sasha_diffs[i:] + sasha_diffs[:i]:
        print("YES")
        break
else:
    print("NO")
