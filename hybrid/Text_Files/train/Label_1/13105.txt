#include<bits/stdc++.h>
using namespace std;

int main()
{
    int tt ;
    cin>>tt;
    while(tt--)
    {
        int n,m;
        cin>>n>>m;
        long long sum=0;
        vector<long long>v(n);
        for(long long &x:v)
        {
            cin>>x;
            sum+=x;
        }
        long long dif=(m-1)*n-sum;
        cout<<(dif>=0 && dif<=m?dif:m)<<endl;
    }
    return 0;
}
