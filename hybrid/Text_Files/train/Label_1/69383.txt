from collections import defaultdict, deque

n,m = map(int,input().split())
roads={}
for i in range(m):
	u,v,c = map(int,input().split())
	roads[(u,v)]=c
	roads[(v,u)] = c
graph = defaultdict(list)
for node in range(1,n+1):
	graph[node] = [node]
for a,b in roads:
	graph[a]+=[b]

par = [0]*(n+1)
ans = []
def dfs(node,p):
	for u in graph[node]:
		if u==p: continue
		if par[u]:
			cycle = [u]
			v = node
			while v!=u: cycle+=[v] ; v=par[v]
			ans.extend(cycle)
			return
		par[u]=node
		dfs(u,node)
		
dfs(1,1)
print(len(ans),len(set(ans)))
print(*set(ans))
