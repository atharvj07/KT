#include <iostream>
using namespace std;

int main() {
    long long q;
    cin >> q;
    while (q--) {
        long long n, m, k;
        cin >> n >> m >> k;
        if ((n + m) % 2 != k % 2 || abs(n - m) > k) {
            cout << -1 << endl;
        } else {
            long long diag = 0;
            while (n > 0 && m > 0 && k >= 2) {
                diag++;
                n--;
                m--;
                k -= 2;
            }
            cout << diag << endl;
        }
    }
    return 0;
}
