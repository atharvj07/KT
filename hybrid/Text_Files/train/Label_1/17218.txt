#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    map<string, int> m;
    vector<string> v;
    for (int i = 0; i < n; i++) {
      string s;
      cin >> s;
      v.push_back(s);
      m[s]++;
    }
    int k = 0;
    for (int i = 0; i < n; i++) {
      if (m[v[i]] == 1)
        continue;
      k += m[v[i]] - 1;
      int x = 1;
      while (m[v[i] + to_string(x)]) x++;
      v[i] += to_string(x);
      m[v[i]] = 1;
    }
    cout << k << endl;
    for (auto x : v) cout << x << endl;
  }
}
