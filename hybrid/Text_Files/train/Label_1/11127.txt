#include <iostream>

using namespace std;

int main() {
    int m;
    cin >> m;
    int h1, a1;
    cin >> h1 >> a1;
    int x1, y1;
    cin >> x1 >> y1;
    int h2, a2;
    cin >> h2 >> a2;
    int x2, y2;
    cin >> x2 >> y2;

    long long nh1 = h1, nh2 = h2;
    int ans = -1;
    for (int i = 0; i < m; i++) {
        if (nh1 == a1 && nh2 == a2) {
            ans = i;
            break;
        }
        nh1 = (nh1 * x1 + y1) % m;
        nh2 = (nh2 * x2 + y2) % m;
    }
    cout << ans << endl;
    return 0;
}
