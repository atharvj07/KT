#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=998244353;
ll f[200000],pow2[200000];
int main(){
    ll n,k,i;
    cin>>n>>k;
    pow2[0]=1;
    for(i=1;i<=n;i++) pow2[i]=(2*pow2[i-1])%mod;
    f[0]=1;f[1]=k;
    for(i=2;i<=n;i++) f[i]=((f[i-1]*(k-i+1))%mod+f[i-2])%mod;
    cout<<f[n]<<endl;
}
