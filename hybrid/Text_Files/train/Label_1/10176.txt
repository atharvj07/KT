#include<bits/stdc++.h>
using namespace std;
string s,ans;
int main(){
  cin>>s;
  set<char> dig;
  for(char ch:s) if(ch-'0'%2) dig.insert(ch);
  int n=s.length();
  if(dig.size()>=2){
    for(int i=n-1;~i;i--){
      if(s[i]-'0'%2){
        for(int j=i-1;~j;j--){
          if(s[j]-'0'%2){
            if(s[i]>s[j]) swap(s[i],s[j]),ans=s;
          }
          else{
            swap(s[i],s[j]),ans=s;
            break;
          }
        }
        break;
      }
    }
  }
  if(ans.empty()) cout<<"-1";
  else while(ans[0]=='0'&&ans.length()>1) ans.erase(0,1);
  cout<<ans;
}
