import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int a = sc.nextInt();
    int b = sc.nextInt();
    if (Math.abs(a - b) > 1 || k == 1 && a == b) System.out.println("NO");
    else {
      char[] s = new char[n];
      for (int i = 0; i < Math.min(k, a); i++) s[i] = 'G';
      for (int i = Math.min(k, a); i < a; i++) s[n - 1 - i] = 'G';
      for (int i = 0; i < Math.min(k, b); i++) s[k + i] = 'B';
      for (int i = Math.min(k, b); i < b; i++) s[n - 1 - k - i] = 'B';
      System.out.println(s);
    }
  }
}
