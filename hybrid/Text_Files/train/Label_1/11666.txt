#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0;;) {
    int mid = n / 2;
    if (n % 2 == 0) cout << s[mid - 1];
    cout << s[mid];
    s.erase(mid, 1);
    if (n % 2 == 0) s.erase(mid - 1, 1);
    n -= 2;
    if (n <= 0) break;
    i++;
  }
  return 0;
}
