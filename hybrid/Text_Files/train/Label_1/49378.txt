import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int minIdx = 1;
            int maxIdx = 1;
            for (int i = 2; i <= n; i++) {
                System.out.println("? " + minIdx + " " + i);
                System.out.flush();
                String response = scanner.next();
                if (response.equals("<")) {
                    maxIdx = minIdx;
                    minIdx = i;
                } else if (response.equals("=")) {
                    minIdx = i;
                }
            }
            for (int i = 1; i <= n; i++) {
                if (i == minIdx) {
                    continue;
                }
                System.out.println("? " + i + " " + maxIdx);
                System.out.flush();
                String response = scanner.next();
                if (response.equals(">")) {
                    maxIdx = i;
                }
            }
            System.out.println("! " + minIdx + " " + maxIdx);
            System.out.flush();
        }
    }
}
