#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e3 + 5;
ll dp[N][N];
ll suf[N][N];
ll a[N];
int main() {
  ll n, k;
  cin >> n >> k;
  for (ll i = 1; i <= n; i++) cin >> a[i];
  for (ll i = n; i >= 1; i--) {
    for (ll j = 1; j <= k; j++) {
      dp[i][j] = LLONG_MIN;
      suf[i][j] = a[i];
      if (i != n) suf[i][j] = max(suf[i][j], suf[i + 1][j]);
    }
  }
  dp[n][1] = a[n];
  for (ll i = n; i >= 1; i--) {
    for (ll j = 2; j <= k; j++) {
      dp[i][j] = max(dp[i][j], dp[i + 1][j - 1] + suf[i][j - 1]);
      dp[i][j] = max(dp[i][j], dp[i + 1][j]);
    }
  }
  cout << dp[1][k] << endl;
  ll j = k;
  vector<ll> v;
  for (ll i = 1;; i++) {
    if (dp[i][j] == dp[i + 1][j]) continue;
    ll cnt = j - 1;
    while (cnt > 0 && dp[i + 1][cnt] == dp[i][j]) cnt--;
    v.push_back(j - cnt);
    j = cnt;
    if (j == 0) break;
  }
  reverse(v.begin(), v.end());
  for (auto x : v) cout << x << " ";
  cout << endl;
  return 0;
}
