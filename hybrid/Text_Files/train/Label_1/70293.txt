def find_dice_probabilities(n, max_probs, min_probs):
    a_probs = [0] * n
    b_probs = [0] * n
    
    a_probs[0] = min_probs[0]
    b_probs[0] = min_probs[0]
    
    for i in range(1, n):
        total_prob = (max_probs[i] - max_probs[i - 1]) * (i + 1)
        min_prob = min_probs[i]
        prev_min_prob = min_probs[i - 1]
        a_probs[i] = (total_prob - prev_min_prob) / (1 - prev_min_prob/min_prob)
        b_probs[i] = min_prob / a_probs[i]
        a_probs[i] += a_probs[i - 1]
        b_probs[i] += b_probs[i - 1]
        
    for i in range(n):
        a_probs[i] -= a_probs[i - 1] if i > 0 else 0
        b_probs[i] -= b_probs[i - 1] if i > 0 else 0
    
    return a_probs, b_probs

n = int(input().strip())
max_probs = list(map(float, input().strip().split()))
min_probs = list(map(float, input().strip().split()))

a_probs, b_probs = find_dice_probabilities(n, max_probs, min_probs)

print(" ".join(map(lambda x: f"{x:.8f}", a_probs)))
print(" ".join(map(lambda x: f"{x:.8f}", b_probs)))
