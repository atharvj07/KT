#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 1;
int n,f[maxn],vis[maxn],ans[maxn],sz;
vector<int> vec[maxn];
int main() {
    cin >> n;
    for(int i = 1;i <= n;i++) {
        cin >> f[i];
        if(!f[i]) vec[0].push_back(i);
        else vis[f[i]]++;
    }
    for(int i = 1;i <= n;i++) {
        if(vis[i] != 1) {
            vec[0].push_back(i);
        }
    }
    for(int i = 1;i <= n;i++) {
        if(!f[i]) {
            for(int j = 1;j <= n;j++) {
                if(!vis[j] && i != j) {
                    f[i] = j;
                    vis[j]++;
                    break;
                }
            }
        }
    }
    for(int i = 1;i <= n;i++) {
        if(!f[i]) {
            f[i] = vec[0][sz];
            vis[f[i]]++;
            sz++;
        }
        ans[i] = f[i];
    }
    for(int i = 1;i <= n;i++) cout << ans[i] << ' ';
    cout << endl;
    return 0;
}
