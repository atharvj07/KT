n = int(input())
M = [input() for i in range(n)]
ismatchX = M[0][0] == M[n-1][n-1] and M[0][n-1] == M[n-1][0]
ismatchO = True
for i in range(n):
    if any(M[i][i] != M[j][j] for j in range(n)):
        ismatchO = False
for i in range(n):
    if any(M[i][n - 1 - i] != M[j][n - 1 - j] for j in range(n)):
        ismatchO = False
if ismatchO: print("YES")
elif ismatchX: print("YES")
else: print("NO")
