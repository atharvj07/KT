t=int(input())
for i in range(t):
    n=int(input())
    l=[[0 for i in range(n+1)] for j in range(n+1)]
    for x in range(n):
        a,b=map(int,input().split())
        l[a][b]+=1
    ans=0
    for a in range(1,n+1):
        for b in range(1,n+1):
            if l[a][b]==0:continue
            for c in range(1,n+1):
                if l[c][b]==0 or a==c:continue
                ans+=l[a][b]*l[c][b]*max(l[a][c],l[b][c])
                ans+=l[a][b]*l[c][b]*l[a][c]*l[b][c]
    print(ans)
