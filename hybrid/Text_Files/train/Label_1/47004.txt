import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int na = sc.nextInt(), nb = sc.nextInt(), w = sc.nextInt(), ans = 0;
        ArrayList<int[]> sa = new ArrayList<>(), sb = new ArrayList<>();
        for(int i = 0; i < na; i++) sa.add(new int[]{sc.nextInt(), sc.nextInt()});
        for(int i = 0; i < nb; i++) sb.add(new int[]{sc.nextInt(), sc.nextInt()});
        Collections.sort(sa, (a, b) -> a[1] - b[1]);
        Collections.sort(sb, (a, b) -> a[1] - b[1]);
        int[] ama = new int[na], amb = new int[na], bma = new int[nb], bmb = new int[nb];
        for(int i = 0; i < na; i++) {
            ama[i] = sa.get(i)[0];
            amb[i] = ama[i] + w;
        }
        for(int i = 0; i < nb; i++) {
            bma[i] = sb.get(i)[0];
            bmb[i] = bma[i] + w;
        }
        int[] coma = new int[na], comb = new int[nb];
        for(int i = 0; i < na; i++) {
            coma[i] = sa.get(i)[1];
        }
        for(int i = 0; i < nb; i++) {
            comb[i] = sb.get(i)[1];
        }
        for(int i = 0; i < na; i++) {
            for(int j = 0; j < nb; j++) {
                if(ama[i] <= w && bma[j] <= w) ans = Math.max(ans, Math.max(Math.abs(coma[i] - comb[j]), ama[i] - bma[j]));
                if(ama[i] <= w && bmb[j] <= w) ans = Math.max(ans, Math.max(Math.abs(coma[i] - comb[j]), ama[i] - bmb[j]));
                if(amb[i] <= w && bma[j] <= w) ans = Math.max(ans, Math.max(Math.abs(coma[i] - comb[j]), amb[i] - bma[j]));
                if(amb[i] <= w && bmb[j] <= w) ans = Math.max(ans, Math.max(Math.abs(coma[i] - comb[j]), amb[i] - bmb[j]));
            }
        }
        System.out.println(ans);
    }
}
