import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] cards = new int[n];
        for (int i = 0; i < n; i++) {
            cards[i] = input.nextInt();
        }
        int sereja = 0, dima = 0;
        boolean serejaTurn = true;
        while (n > 0) {
            if (serejaTurn) {
                if (cards[0] > cards[n - 1]) {
                    sereja += cards[0];
                    n--;
                } else {
                    sereja += cards[n - 1];
                    n--;
                }
            } else {
                if (cards[0] > cards[n - 1]) {
                    dima += cards[0];
                    n--;
                } else {
                    dima += cards[n - 1];
                    n--;
                }
            }
            serejaTurn = !serejaTurn;
        }
        System.out.println(sereja + " " + dima);
    }
}
