import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int h = sc.nextInt();
            if (n == 0 && h == 0) break;
            int[][][] cube = new int[n][n][n];
            for (int i = 0; i < h; i++) {
                String c = sc.next();
                int a = sc.nextInt() - 1;
                int b = sc.nextInt() - 1;
                if (c.equals("xy")) {
                    for (int j = 0; j < n; j++) {
                        cube[a][b][j] = 1;
                    }
                } else if (c.equals("xz")) {
                    for (int j = 0; j < n; j++) {
                        cube[a][j][b] = 1;
                    }
                } else if (c.equals("yz")) {
                    for (int j = 0; j < n; j++) {
                        cube[j][a][b] = 1;
                    }
                }
            }
            int ans = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    for (int k = 0; k < n; k++) {
                        if (cube[i][j][k] == 0) ans++;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
