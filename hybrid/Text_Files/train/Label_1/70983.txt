n,a,b=map(int,input().split())
l=[int(i) for i in input().split()]
if l[0]!=l[-1]:
    print(-1)
    exit()
ans=0
for i in range(1,n//2+1):
    if l[i]!=l[n-1-i]:
        if l[i]==2:
            if l[n-1-i]==0:
                ans+=a
            else:
                ans+=b
        elif l[n-1-i]==2:
            if l[i]==0:
                ans+=b
            else:
                ans+=a
        else:
            print(-1)
            exit()

print(ans)
