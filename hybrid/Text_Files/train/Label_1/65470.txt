n, m = map(int, input().split())
a = []
b = [False for i in range(m+1)]
for i in range(n):
    _, *x = map(int, input().split())
    a.append(set(x))
vis = [False for i in range(n+1)]
for i in range(n):
    if not vis[i]:
        vis[i] = True
        q = [a[i]]
        while q:
            cur = q.pop(0)
            for j in cur:
                if not b[j]:
                    for k in range(n):
                        if j in a[k] and not vis[k]:
                            vis[k] = True
                            q.append(a[k])
                    b[j] = True
if all(b):
    print('YES')
else:
    print('NO')
