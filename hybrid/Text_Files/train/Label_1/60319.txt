import sys
lines = int(input())
put = list(map(int, input().split()))
for line in sys.stdin:
    cnt,lst,s = 0,[],''
    for j in line.split():
        suc=False
        for c in j:
            if c in 'aeiouy':
                if suc==False:
                    suc=True
                    lst.append(s)
                    s=''
            s+=c
        if suc==True:
            lst.append(s)
            s=''
    lst.append(s)
    if len(lst)-1==put[cnt]:
        cnt+=1
    else:
        break
if cnt==lines:
    print('YES')
else:
    print('NO')
