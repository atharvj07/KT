for _ in range(int(input())):
    n = int(input())
    a = input()
    b = input()
    if a.count('1') != b.count('1'):
        print(-1)
    else:
        ans = 0
        for i in range(n):
            if a[i] != b[i]:
                ans += 1
                for j in range(n):
                    if i != j:
                        a = a[:j] + str(1 - int(a[j])) + a[j + 1:]
        print(ans)
