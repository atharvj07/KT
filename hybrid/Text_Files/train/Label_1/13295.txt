#include <bits/stdc++.h>
using namespace std;
#define ll long long

const int N = 2e5 + 5;
int32_t main() {
    cin.tie(nullptr)->sync_with_stdio(false);
    int n;
    cin >> n;
    vector<ll> p(n);
    for (auto& it : p) { cin >> it; }
    vector<int> a(n), b(n);
    for (auto& it : a) { cin >> it; }
    for (auto& it : b) { cin >> it; }
    sort(begin(p), end(p));
    int m;
    cin >> m;
    vector<int> c(m);
    for (auto& it : c) { cin >> it; }
    auto get = [&](ll v) -> pair<bool, int> {
        auto it = lower_bound(begin(p), end(p), v);
        if (it == end(p) || *it != v) {
            return {false, -1};
        }
        int idx = it - begin(p);
        return {true, a[idx] == c[idx] || b[idx] == c[idx]};
    };
    for (int v : c) {
        if (get(v).second == true) {
            cout << v << ' ';
        } else {
            auto r = get(v + 1);
            if (r.first)
                cout << r.second << ' ';
            else
                cout << -1 << ' ';
        }
    }
    cout << '\n';
    return 0;
}
