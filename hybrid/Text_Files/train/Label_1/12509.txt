#include <iostream>

using namespace std;

int main() {
    long long k;
    cin >> k;
    // 0 to 9 has one digit for each number
    if (k <= 9) {
        cout << k << endl;
        return 0;
    }
    // 10 to 99 has two digits for each number
    else if (k <= 99) {
        cout << (k + 1) / 10 << endl;
        return 0;
    }
    // 100 to 999 has three digits for each number
    else if (k <= 999) {
        cout << (k + 2) / 10 % 10 << endl;
        return 0;
    }
    // 1000 to 9999 has four digits for each number
    else if (k <= 9999) {
        cout << (k + 3) / 100 % 10 << endl;
        return 0;
    }
    // 10000 to 99999 has five digits for each number
    else if (k <= 99999) {
        cout << (k + 4) / 1000 % 10 << endl;
        return 0;
    }
    // 100000 to 999999 has six digits for each number
    else if (k <= 999999) {
        cout << (k + 5) / 10000 % 10 << endl;
        return 0;
    }
    // 1000000 to 9999999 has seven digits for each number
    else if (k <= 9999999) {
        cout << (k + 6) / 100000 % 10 << endl;
        return 0;
    }
    // 10000000 to 99999999 has eight digits for each number
    else if (k <= 99999999) {
        cout << (k + 7) / 1000000 % 10 << endl;
        return 0;
    }
    // 100000000 to 999999999 has nine digits for each number
    else if (k <= 999999999) {
        cout << (k + 8) / 10000000 % 10 << endl;
        return 0;
    }
    // 1000000000 to 9999999999 has ten digits for each number
    else if (k <= 9999999999) {
        cout << (k + 9) / 100000000 % 10 << endl;
        return 0;
    }
    // 10000000000 to 99999999999 has eleven digits for each number
    else if (k <= 99999999999) {
        cout << (k + 10) / 1000000000 % 10 << endl;
        return 0;
    }
    // 100000000000 to 999999999999 has twelve digits for each number
    else if (k <= 999999999999) {
        cout << (k + 11) / 10000000000 % 10 << endl;
        return 0;
    }
}
