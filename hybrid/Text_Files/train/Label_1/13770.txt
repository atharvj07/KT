#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  string s;
  int q;
  cin >> s >> q;
  vector<int> cnt(26, 0);
  for (auto c : s) ++cnt[c - 'a'];
  int ans = 0;
  for (int i = 0; i < 26; ++i) if (cnt[i]) ++ans;

  while (q--) {
    int t;
    cin >> t;
    if (t == 1) {
      int p;
      char c;
      cin >> p >> c;
      --cnt[s[p - 1] - 'a'];
      ++cnt[c - 'a'];
      s[p - 1] = c;
      if (cnt[s[p - 1] - 'a'] == 1)
        ++ans;
      else if (cnt[s[p - 1] - 'a'] == 0)
        --ans;
    } else {
      int l, r;
      cin >> l >> r;
      int cnt2 = 0;
      for (int i = 0; i < 26; ++i)
        if (cnt[i] && l <= r) ++cnt2;
      cout << cnt2 << endl;
    }
  }
  return 0;
}
