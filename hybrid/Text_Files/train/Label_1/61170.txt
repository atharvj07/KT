import sys

n = int(input())
a = [int(x) for x in input().split()]

a.sort()
b = a[:n]
c = a[n:]
b.reverse()

sa = sb = ca = cb = 0
for i in range(n):
    if sa <= sb:
        sa += c[i]
    else:
        sb += b[i]
    if ca <= cb:
        ca += a[2 * n - 1 - i]
    else:
        cb += a[2 * i]

print(sa + sb + ca + cb)
