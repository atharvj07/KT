for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    a.sort()
    k = 0
    for i in range(n):
        s = 0
        num = 0
        for j in range(i,n):
            s+=a[j]
            num+=1
            if s > num*a[i]:
                break
        k+=num - 1
    print(k)
