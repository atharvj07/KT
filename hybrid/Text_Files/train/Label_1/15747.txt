#include <bits/stdc++.h>

using namespace std;
const int mod = 2;

int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    int n;
    cin >> n;
    bool ok = true;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    for (int i = 0; i < n; ++i) { ok &= (a[i] % mod == i % mod); }
    if (ok) {
      cout << 0 << endl;
    } else {
      bool hasEven = false, hasOdd = false;
      for (int i = 0; i < n; ++i) {
        if (a[i] % mod == 0) hasEven = true;
        if (a[i] % mod) hasOdd = true;
      }
      if (hasEven && hasOdd) {
        cout << 1 << endl;
      } else {
        cout << -1 << endl;
      }
    }
  }
}
