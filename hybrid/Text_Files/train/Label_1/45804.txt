import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    int[] b = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
      b[i] = in.nextInt();
    }
    Arrays.sort(a);
    Arrays.sort(b);
    int i = 0, j = n - 1;
    int ans = 0;
    while (i < j) {
      if (a[i] <= b[j]) {
        ans++;
        i++;
        j--;
      } else {
        j--;
      }
    }
    System.out.println(ans);
  }
}
