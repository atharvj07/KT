import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) break;
            int[] a = new int[n];
            int x = 0;
            for (int i = 0; i < n; i++) {
                String s = scanner.next();
                if (s.equals("x")) {
                    a[i] = x;
                } else {
                    a[i] = Integer.parseInt(s);
                    x = a[i];
                }
            }
            if (isRecoverable(a)) {
                System.out.println(x);
            } else if (isAmbiguous(a)) {
                System.out.println("ambiguous");
            } else {
                System.out.println("none");
            }
        }
    }

    private static boolean isRecoverable(int[] a) {
        int n = a.length;
        for (int i = 1; i < n - 1; i++) {
            if ((a[i] >= a[i - 1] && a[i] >= a[i + 1]) || (a[i] <= a[i - 1] && a[i] <= a[i + 1]))
                return false;
        }
        return true;
    }

    private static boolean isAmbiguous(int[] a) {
        int n = a.length;
        for (int i = 1; i < n - 1; i++) {
            if (a[i] == a[i - 1] || a[i] == a[i + 1])
                return true;
        }
        return false;
    }
}
