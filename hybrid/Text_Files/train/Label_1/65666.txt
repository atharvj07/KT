import heapq

n1, n2, n3 = map(int, input().split())
a1 = [-x for x in map(int, input().split())]
a2 = [-x for x in map(int, input().split())]
a3 = [-x for x in map(int, input().split())]
heapq.heapify(a1)
heapq.heapify(a2)
heapq.heapify(a3)
ans = 0
while len(a1) > 0 and len(a2) > 0:
    x = heapq.heappop(a1)
    y = heapq.heappop(a2)
    ans -= x - y
    if x - y > 0:
        heapq.heappush(a1, x - y)
while len(a2) > 0 and len(a3) > 0:
    x = heapq.heappop(a2)
    y = heapq.heappop(a3)
    ans -= x - y
    if x - y > 0:
        heapq.heappush(a2, x - y)
while len(a1) > 0 and len(a3) > 0:
    x = heapq.heappop(a1)
    y = heapq.heappop(a3)
    ans -= x - y
    if x - y > 0:
        heapq.heappush(a1, x - y)
print(-ans)
