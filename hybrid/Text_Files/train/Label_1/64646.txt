a=int(input())
s=input()
t=len(s)
pref=[[0,0,0,0]for i in range(t+1)]
pref[0][0]=1
for i in range(t):
    for j in range(4):
        pref[i+1][j]=pref[i][j]
    l=int(s[i])
    pref[i+1][l%4]+=1
ans=0
for i in range(t):
    for j in range(i,t):
        k=s[i]+s[j]
        k=int(k)%4
        ans+=pref[i][k]*pref[j+1][k]
print(ans)
