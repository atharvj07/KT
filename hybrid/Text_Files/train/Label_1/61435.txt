n, a, cf, cm, m = [int(x) for x in input().split()]
sk = [int(x) for x in input().split()]
sk.sort()
msk = 0
mnv = 10**9
for x in sk:
  if x == a: msk += cf
  mnv = min(mnv, x)
res = msk + cm * mnv
fin = []
b = mnv + m // n
for x in sk:
  fin.append(min(x, b))
  m -= min(x, b) - x
  if m < 0: break
fin[-1] += m
print(res)
print(*fin)
