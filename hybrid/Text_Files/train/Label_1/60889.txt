def f(x):
    if x == 0:
        return 0
    return f(x//2) + x % 2

n = int(input())
a = list(map(int, input().split()))

occurrences = {}

for number in a:
    fx = f(number)
    if fx in occurrences:
        occurrences[fx] += 1
    else:
        occurrences[fx] = 1

result = 0
for key in occurrences:
    result += (occurrences[key] * (occurrences[key] - 1)) // 2

print(result)
