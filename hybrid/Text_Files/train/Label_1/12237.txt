#include <iostream>
#include <string>

const std::string ones[] = {"","one","two","three","four","five","six","seven","eight","nine"};
const std::string tens[] = {"","ten","twenty","thirty","forty","fifty","sixty","seventy","eighty","ninety"};
const std::string teens[] = {"ten","eleven","twelve","thirteen","fourteen","fifteen","sixteen","seventeen","eighteen","nineteen"};

std::string number(int n) {
  if (n == 0) { return "zero"; }

  int q = n / 10;
  int r = n % 10;

  std::string res;
  if (q == 0) {
    res = ones[r];
  } else if (q == 1) {
    res = teens[r];
  } else {
    res = tens[q]; if (r > 0) { res += "-" + ones[r]; }
  }

  return res;
}

int main() {
  int n;
  std::cin >> n;
  std::cout << number(n) << std::endl;
  return 0;
}
