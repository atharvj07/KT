#include <cstdio>

int main() {
    long long l, r, x, y, k;
    int p;
    scanf("%lld %lld %lld %lld %lld", &l, &r, &x, &y, &k);
    for (int i = x; i <= y; i++) {
        if (l <= k && k <= r) {
            p = 1;
            break;
        }
        k = k * 10 / i;
    }
    puts(p ? "YES" : "NO");
}
