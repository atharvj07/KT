#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> xs, ys;
  unordered_map<int, int> mx, my;
  for (int i = 0; i < 4 * n + 1; i++) {
    int x, y; cin >> x >> y;
    xs.push_back(x);
    ys.push_back(y);
    mx[x]++;
    my[y]++;
  }
  int ox = -1, oy = -1;
  for (int x : xs) {
    if (mx[x] == 1) {
      ox = x;
      break;
    }
  }
  for (int y : ys) {
    if (my[y] == 1) {
      oy = y;
      break;
    }
  }
  cout << ox << " " << oy << endl;
  return 0;
}
