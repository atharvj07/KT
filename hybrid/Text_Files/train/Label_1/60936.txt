import sys
inp=sys.stdin.readlines()
n,k=map(int,inp[0].strip().split())
p=list(map(int,inp[1].strip().split()))
ans=0
if(n<=6 and k<=4):
	for i in range(1,n+1):
		for j in range(i+1,n+1):
			if p[j-1]<p[i-1]:
				ans+=1/(j-i)
elif(n<=30 and k<=200):
	for i in range(1,n+1):
		for j in range(i+1,n+1):
			if p[j-1]<p[i-1]:
				ans+=1/((j-i)*(j-i+1)/2)
else:
	for i in range(1,n+1):
		for j in range(i+1,n+1):
			if p[j-1]<p[i-1]:
				ans=ans+1/2*((k+1)*(k+2)/(n*(n+1)*(n+2)))
print(ans)
