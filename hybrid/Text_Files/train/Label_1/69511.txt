def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def find_sequence(n):
    if n == 4 or is_prime(n):
        return ["YES"] + [str(i) for i in range(1, n + 1)]
    if n < 4:
        return ["NO"]
    for i in range(2, n):
        if n % i == 0:
            return ["NO"]
    return ["YES"] + [str(i) for i in range(1, n + 1)]

n = int(input().strip())
result = find_sequence(n)
print("\n".join(result))
