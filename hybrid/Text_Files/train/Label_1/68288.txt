from collections import defaultdict

def dfs(node, parent, graph):
    global ans
    size = 1
    for child in graph[node]:
        if child != parent:
            child_size = dfs(child, node, graph)
            if child_size % 2 == 0:
                ans += 1
            else:
                size += child_size
    return size

n = int(input().strip())
graph = defaultdict(list)

for _ in range(n - 1):
    u, v = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)

ans = 0
if dfs(1, -1, graph) % 2 == 0:
    print(ans)
else:
    print(-1)
