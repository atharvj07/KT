n,k=map(int,input().split())
l=list(map(int,input().split()))
for i in range(k):
    mn=10**9
    f=0
    for x in l:
        if x>0:
            mn=min(mn,x)
            f=1
    if f==0:
        print(0)
    else:
        print(mn)
        for i in range(n):
            if l[i]>0:
                l[i]-=mn
