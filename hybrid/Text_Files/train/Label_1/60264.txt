n = int(input())
before = list(map(int, input().split()))
after = list(map(int, input().split()))
z = [i for i in range(n) if before[i] != after[i]]
if not z:
    print('unrated')
    exit()
z.sort(key=lambda i: (before[i], after[i]))
for i in range(1, len(z)):
    if before[z[i]] >= before[z[i - 1]] and after[z[i]] >= after[z[i - 1]]: continue
    print('rated')
    exit()
print('maybe')
