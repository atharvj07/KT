n, k = map(int, input().split())
s = input()
ans = "YES"
for i in range(1, n - 1):
    if s[i] == '#' and s[i - 1] == '#' and s[i + 1] == '#':
        ans = "NO"
        break
    if s[i] == '#' and s[i + 1] == '.' and s[i - 1] != '.' and i + 1 + k >= n:
        ans = "NO"
        break
    if s[i] == '#' and s[i - 1] == '.' and s[i + 1] != '.' and i + 1 + k >= n:
        ans = "NO"
        break
print(ans)
