import java.util.*;
import java.math.BigInteger;
 
class Main {
  static long mod = 1_000_000_007L;
  static int[] dist;
  static ArrayList<Integer>[] adj;
  static long[] val;
  static long[] sum;
 
  static void dfs(int v, int d) {
    dist[v] = d;
    sum[v] = val[v];
    for (int u : adj[v]) {
      if (dist[u] != -1) {
        continue;
      }
      dfs(u, d + 1);
      sum[v] += sum[u];
    }
  }
 
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] p = new int[n];
    adj = new ArrayList[n];
    for (int i = 0; i < n; i++) {
      adj[i] = new ArrayList<>();
    }
 
    for (int i = 1; i < n; i++) {
      p[i] = sc.nextInt();
      --p[i];
      adj[i].add(p[i]);
      adj[p[i]].add(i);
    }
 
    int q = sc.nextInt();
    dist = new int[n];
    val = new long[n];
    sum = new long[n];
    for (int i = 0; i < n; i++) {
      dist[i] = -1;
    }
 
    dfs(0, 0);
 
    for (int i = 0; i < q; i++) {
      int t = sc.nextInt();
      if (t == 1) {
        int v = sc.nextInt();
        --v;
        long x = sc.nextLong();
        long k = sc.nextLong();
 
        val[v] = (val[v] + x) % mod;
        for (int u : adj[v]) {
          val[u] = (val[u] + x - k) % mod;
          k = (k + mod) % mod;
        }
      } else {
        int v = sc.nextInt();
        --v;
        System.out.println(sum[v] % mod);
      }
    }
  }
}
