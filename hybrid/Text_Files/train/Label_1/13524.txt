#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, ans = 0, t;
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < k; j++) {
            cin >> t;
            ans += t;
        }
    }
    if (ans <= n * k / 2)
        cout << "YES";
    else
        cout << "NO";
    return 0;
}
