n = int(input())
a = list(map(int, input().split()))

# Find the minimum element and its index
min_val = min(a)
min_index = a.index(min_val)

# If there are negative numbers, find the maximum among them
neg_max_val = max(filter(lambda x: x < 0, a), default=None)
if neg_max_val is not None:
    neg_max_index = a.index(neg_max_val)

# Operations
operations = []

# Case 1: If all are non-negative or all are negative, remove the minimum
if all(x >= 0 for x in a) or all(x < 0 for x in a):
    operations.append(f"2 {min_index + 1}")
    for i in range(n - 1):
        if i != min_index:
            operations.append(f"1 {min_index + 1} {i + 1 if i < min_index else i + 2}")
# Case 2: Mix of positive and negative, remove the maximum negative
else:
    operations.append(f"2 {neg_max_index + 1}")
    for i in range(n - 1):
        if i != neg_max_index:
            operations.append(f"1 {neg_max_index + 1} {i + 1 if i < neg_max_index else i + 2}")

print("\n".join(operations))
