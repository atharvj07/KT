#include <cmath>
#include <cstdio>
#include <iostream>
using namespace std;

void checkMax(double &x, double v) { if (x < v) x = v; }
void checkMin(double &x, double v) { if (x > v) x = v; }

int main() {
    long long R, x1, y1, x2, y2;
    cin >> R >> x1 >> y1 >> x2 >> y2;
    long long distx = abs(x1 - x2), disty = abs(y1 - y2);
    long long dist = sqrt(1ll * distx * distx + 1ll * disty * disty);
    if (dist > 2 * R) { cout << x2 << ' ' << y2 << ' ' << R << endl; return 0; }
    double x = (x1 + x2) / 2.0, y = (y1 + y2) / 2.0;
    double maxr = R;
    if (distx > disty) {
        double cand = (sqrt((1ll * dist * dist - 1ll * disty * disty)) - 1ll * abs(x1 - x)) / 2.0;
        checkMin(maxr, cand);
        cand = (sqrt((1ll * dist * dist - 1ll * disty * disty)) - 1ll * abs(x1 - x2)) / 2.0;
        checkMin(maxr, cand);
    } else if (disty > distx) {
        double cand = (sqrt((1ll * dist * dist - 1ll * distx * distx)) - 1ll * abs(y1 - y)) / 2.0;
        checkMin(maxr, cand);
        cand = (sqrt((1ll * dist * dist - 1ll * distx * distx)) - 1ll * abs(y1 - y2)) / 2.0;
        checkMin(maxr, cand);
    }
    if (maxr < 1e-6) { cout << x << ' ' << y << ' ' << maxr << endl; return 0; }
    double r = maxr;
    double left = -maxr, right = maxr;
    while (abs(right - left) > 1e-9) {
        double mid = (left + right) / 2.0;
        if (mid + r >= dist) right = mid;
        else left = mid;
    }
    r -= left;
    cout << x << ' ' << y << ' ' << r << endl;
    return 0;
}
