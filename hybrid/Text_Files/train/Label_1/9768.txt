#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      m[x]++;
    }
    if (m.size() % 2 == 0)
      cout << "Alice" << endl;
    else
      cout << "Bob" << endl;
  }
  return 0;
}
