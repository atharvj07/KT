#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    string s, t;
    cin >> s >> t;
    int si = s.size(), ti = t.size();
    int i = si - 1, j = ti - 1;
    while (i >= 0 && j >= 0) {
      if (s[i] == t[j]) {
        --i;
        --j;
      } else {
        ++i;
      }
    }
    cout << (j == -1 ? "YES\n" : "NO\n");
  }
}
