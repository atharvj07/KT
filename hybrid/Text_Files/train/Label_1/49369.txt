import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[][] map = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = in.next();
            for (int j = 0; j < m; j++) {
                char c = line.charAt(j);
                if (c == '#') {
                    map[i][j] = -1;
                } else if (c == '.') {
                    map[i][j] = 0;
                } else {
                    map[i][j] = c - '0';
                }
            }
        }
        boolean[][] visited = new boolean[n][m];
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] != 0 && !visited[i][j]) {
                    bfs(map, visited, dist, i, j);
                }
            }
        }
        int min = Integer.MAX_VALUE;
        for (int[] row : dist) {
            for (int cell : row) {
                if (cell > 0) {
                    min = Math.min(min, cell);
                }
            }
        }
        System.out.println(min == Integer.MAX_VALUE ? -1 : min - 1);
    }

    private static void bfs(int[][] map, boolean[][] visited, int[][] dist, int x, int y) {
        Queue<int[]> queue = new LinkedList<>();
        queue.add(new int[]{x, y});
        visited[x][y] = true;
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int curX = cur[0];
            int curY = cur[1];
            int curDist = dist[curX][curY];
            if (map[curX][curY] != 0 && map[curX][curY] != map[x][y]) {
                dist[curX][curY] = Math.min(dist[curX][curY], curDist);
            } else {
                dist[curX][curY] = curDist;
            }
            if (curX > 0 && !visited[curX - 1][curY] && map[curX - 1][curY] != -1) {
                queue.add(new int[]{curX - 1, curY});
                visited[curX - 1][curY] = true;
            }
            if (curY > 0 && !visited[curX][curY - 1] && map[curX][curY - 1] != -1) {
                queue.add(new int[]{curX, curY - 1});
                visited[curX][curY - 1] = true;
            }
            if (curX < map.length - 1 && !visited[curX + 1][curY] && map[curX + 1][curY] != -1) {
                queue.add(new int[]{curX + 1, curY});
                visited[curX + 1][curY] = true;
            }
            if (curY < map[0].length - 1 && !visited[curX][curY + 1] && map[curX][curY + 1] != -1) {
                queue.add(new int[]{curX, curY + 1});
                visited[curX][curY + 1] = true;
            }
        }
    }
}
