#include <bits/stdc++.h>
using namespace std;
int n, k, a[505], f[505][505];
set<int> s;
void solve() {
  for (int i = 0; i <= n; i++) for (int j = 0; j <= k; j++) f[i][j] = 0;
  f[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      f[i][j] = f[i - 1][j];
      if (j >= a[i]) f[i][j] |= f[i - 1][j - a[i]];
    }
  }
  for (int i = 0; i <= k; i++) {
    if (f[n][i]) s.insert(i);
  }
  cout << s.size() << endl;
  for (auto x : s) cout << x << " ";
}
int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  solve();
}
