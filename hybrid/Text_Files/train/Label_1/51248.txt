import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int k = sc.nextInt();
        int n = s.length();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int j = i;
            while (j < n && s.charAt(j) == s.charAt(j % n)) {
                j++;
            }
            if (j % n <= k) {
                ans = Math.max(ans, j - i + 1);
            }
        }
        System.out.println(ans);
    }
}
