import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args)
    {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] xs = new int[n];
        int[] vs = new int[n];
        for(int i = 0; i < n; i++)
        {
            xs[i] = in.nextInt();
            vs[i] = in.nextInt();
        }
        long sum = 0;
        for(int i = 0; i < n; i++)
        {
            for(int j = i + 1; j < n; j++)
            {
                long dx = xs[i] - xs[j];
                long dv = vs[j] - vs[i];
                if(dv == 0)
                {
                    sum += Math.abs(dx);
                }
                else
                {
                    if(dx * dv < 0)
                    {
                        sum += Math.abs(dx);
                    }
                    else
                    {
                        sum += Math.abs(dx) % Math.abs(dv);
                    }
                }
            }
        }
        System.out.println(sum);
    }
}
