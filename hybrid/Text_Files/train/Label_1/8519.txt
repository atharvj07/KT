#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
int main() {
    ll a1, b1, a2, b2, l, r; cin >> a1 >> b1 >> a2 >> b2 >> l >> r;
    if (a1 < a2) {
        swap(a1, a2);
        swap(b1, b2);
    }
    l =  max(l, (b2 - b1 + a1 - 1) / a1);
    r = min(r, (b2 - b1 + a2 - 1) / a2);
    cout << (r < l ? 0 : 1 + (r - l) / (a2 / __gcd(a2, a1))) << endl;
    return 0;
}
