import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        if (n == 3) {
            System.out.println("Yes");
        } else {
            boolean ok = true;
            for (int i = 1; i < n - 1; i++) {
                if ((a[i - 1] ^ a[i] ^ a[i + 1]) != a[i]) {
                    ok = false;
                }
            }
            if ((a[n - 2] ^ a[n - 1] ^ a[0]) != a[n - 1]) {
                ok = false;
            }
            System.out.println(ok ? "Yes" : "No");
        }
    }
}
