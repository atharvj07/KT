import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int n;
    static List<Integer>[] adj;
    static int[] parent;
    static int[] childCount;
    static boolean[] vis;
    static int[] a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        parent = new int[n + 1];
        Arrays.fill(parent, -1);
        childCount = new int[n + 1];
        vis = new boolean[n + 1];
        a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            int p = sc.nextInt();
            int c = sc.nextInt();
            parent[i] = p;
            childCount[i] = c;
            if (p != 0) {
                adj[p].add(i);
            }
        }
        if (!dfs(1)) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 1; i <= n; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }

    static boolean dfs(int u) {
        vis[u] = true;
        Queue<Integer> q = new ArrayDeque<>();
        q.add(u);
        while (!q.isEmpty()) {
            int cur = q.poll();
            int max = 0;
            for (int v : adj[cur]) {
                if (!vis[v]) {
                    max = Math.max(max, childCount[v]);
                    q.add(v);
                }
            }
            a[cur] = max + 1;
            if (childCount[cur] != max) {
                return false;
            }
        }
        return true;
    }
}
