n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Count frequency of elements in both arrays
freq_a = {i:0 for i in range(1, n+1)}
freq_b = {i:0 for i in range(1, n+1)}
for i in a:
    freq_a[i] += 1
for i in b:
    freq_b[i] += 1

# Calculate maximum possible matching pairs
ans = 0
for i in range(1, n+1):
    ans += min(freq_a[i], freq_b[i])

print(ans)
