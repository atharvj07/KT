n,l,r,ql,qr=[int(x) for x in input().split()]
a=[int(x) for x in input().split()]
b=[0]*n
b[0]=a[0]*l
for i in range(1,n):
    b[i]=min(b[i-1]+ql+a[i]*l,a[i]*l)
c=[0]*n
c[n-1]=a[n-1]*r
for i in range(n-2,-1,-1):
    c[i]=min(c[i+1]+qr+a[i]*r,a[i]*r)
ans=1e9
for i in range(n):
    ans=min(ans,b[i]+c[i])
print(ans)
