#include <bits/stdc++.h>
using namespace std;
constexpr int mod = 1e9 + 7;
constexpr int ulm = 1e9 + 7;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n), b;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
    }
    if (sum % 2)
      b.push_back(sum / 2 + 1), b.push_back(sum / 2);
    else
      b.push_back(sum / 2), b.push_back(sum / 2);
    for (auto e : b) cout << e << ' ';
    cout << '\n';
  }
  return 0;
}
