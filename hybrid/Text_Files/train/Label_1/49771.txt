import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long l = scanner.nextLong();
        long r = scanner.nextLong();
        long max = 0;
        long result = 0;
        for (long i = 2; i * i <= r; i++) {
            long cnt = 0;
            for (long j = l; j <= r; j++) {
                if (j % i == 0) {
                    cnt++;
                }
            }
            if (cnt > max) {
                max = cnt;
                result = i;
            }
        }
        System.out.println(result);
    }
}
