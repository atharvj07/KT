n=int(input())
l1=[]
for i in range(n):
    a,b=map(int,input().split())
    l1.append([a,b])
l1.sort()
ans=0
for i in range(n):
    x=l1[i][0]
    y=l1[i][1]
    ans=max(ans,x,y)
for i in range(n):
    x=l1[i][0]
    y=l1[i][1]
    flag=0
    for j in range(n):
        if j==i:
            continue
        a=l1[j][0]
        b=l1[j][1]
        if a<=x and b>=y:
            flag=1
            break
    if flag==0:
        ans=max(ans,y-x+1)
print(ans)
