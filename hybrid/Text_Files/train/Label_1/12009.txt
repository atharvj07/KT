#include<cstdio>
#include<vector>
#include<queue>

using namespace std;

const int Maxn = 100005;
int n, m, a[Maxn];
bool vis[Maxn];
vector<int> e[Maxn];
queue<int> q;

int main() {
    scanf("%d %d", &n, &m);
    for (int i = 1;i <= n;i++) {
        int x;
        scanf("%d", &x);
        if (x) e[i].push_back(x);
    }
    for (int i = 1;i <= m;i++) scanf("%d", &a[i]);
    for (int i = 1;i <= m;i++) if (!vis[i]) {
        q.push(i);
        while (!q.empty()) {
            int p = q.front(); q.pop();
            if (vis[p]) continue;
            vis[p] = true;
            for (vector<int>::iterator it = e[p].begin();it != e[p].end();it++) q.push(*it);
        }
    }
    int ans = 0;
    for (int i = 1;i <= m;i++) if (vis[i]) ans += a[i];
    if (ans <= n) printf("%d\n", ans); else printf("-1\n");
}
