#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[2][2],_;
void matmul(ll x[2][2],ll y[2][2])
{
    ll z[2][2]={{0,0},{0,0}};
    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++)
            for(int k=0;k<2;k++)
                z[i][j]+=x[i][k]*y[k][j];
    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++)
            x[i][j]=z[i][j];
}
char ans[1000010];
int main()
{
    cin>>a[0][0]>>a[0][1]>>a[1][0]>>a[1][1];
    if(a[0][0]+a[0][1]!=a[1][0]+a[1][1])
    {
        cout<<"Impossible\n";
        return 0;
    }
    _=(a[1][0]-a[0][0])/2;
    if(_<0)
    {
        cout<<"Impossible\n";
        return 0;
    }
    a[0][0]-=_;
    a[1][0]-=_;
    for(int i=0;i<_;i++)
        ans[i]='0';
    for(int i=0;i<a[0][1];i++)
        ans[_+i]='1' + '0';
    for(int i=0;i<a[1][1];i++)
        ans[_+a[0][1]+i]='1' + '1';
    for(int i=0;i<a[1][0];i++)
        ans[_+a[0][1]+a[1][1]+i]='0' + '0';
    cout<<ans<<'\n';
    return 0;
}
