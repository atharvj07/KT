#include <bits/stdc++.h>
using namespace std;
int a[2010];
int main () {
    int n;
    cin >> n;
    int sm = 0;
    for (int i = 0; i < 2 * n; i += 1) {
        cin >> a[i];
        sm += a[i];
    }
    if (sm % 2 == 0) {
        for (int i = 0; i < 2 * n; i += 1) cout << a[i] << " ";
        cout << endl;
        return 0;
    }
    int x = 0, y = 0;
    for (int i = 0; i < 2 * n; i += 1) {
        if (a[i] % 2 == 0) x = a[i];
        if (a[i] % 2 == 1) y = a[i];
    }
    if (x != 0) {
        a[2 * n - 1] = x;
        a[0] = y;
        for (int i = 0; i < 2 * n; i += 1) cout << a[i] << " ";
        cout << endl;
        return 0;
    }
    cout << "-1" << endl;
    return 0;
}
