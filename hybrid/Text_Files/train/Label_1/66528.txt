import sys
from collections import Counter

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    p = [int(x) for x in input().split()]

    s = []
    for i in range(len(p)):
        s.append(p[i] + p[(i + 1) % len(p)])

    s.sort()

    for i in range(len(p)):
        for j in range(i + 1, len(p)):
            if s[i] == s[j] and Counter(p[i:j + 1]) != Counter(p[i + 1:j + 2]):  # found
                p[i], p[j] = p[j], p[i]
                break

    print(*p)
