def shift_char(c):
    return chr((ord(c) - 97 - 1) % 26 + 97)

def min_lexico_shift(s):
    n = len(s)
    for i in range(n):
        if s[i] != 'a':
            j = i
            while j < n and s[j] != 'a':
                j += 1
            return s[:i] + ''.join(shift_char(c) for c in s[i:j]) + s[j:]
    return ''.join(shift_char(c) for c in s)

s = input().strip()
print(min_lexico_shift(s))
