def can_divide_array(n, arr):
    total_sum = sum(arr)
    if total_sum % 2 != 0:
        return "NO"
    half_sum = total_sum // 2

    prefix_sum = 0
    prefix_sums = set()
    for i in range(n):
        prefix_sum += arr[i]
        prefix_sums.add(prefix_sum)

    suffix_sum = 0
    for i in range(n - 1, -1, -1):
        suffix_sum += arr[i]
        if suffix_sum > half_sum:
            break
        if suffix_sum in prefix_sums or (suffix_sum + arr[i]) in prefix_sums:
            return "YES"

    return "NO"

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(can_divide_array(n, arr))
