import java.util.*;

class Main {
    static List<int[]> senators = new ArrayList<>();
    static int k, A;

    static double calc() {
        Collections.sort(senators, (a, b) -> (b[0] - b[1] - a[0] + a[1]));
        double p = 0;
        for (int[] senator : senators) {
            int l = senator[0], x = senator[1];
            if (k == 0) break;
            if (x == 100) continue;
            double p1 = (l * (100 - x)) / 1000.0;
            if (p1 > 1) p1 = 1;
            if (p + p1 > 1) return 1;
            A += l;
            p += p1;
            k--;
        }
        return p;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        k = sc.nextInt();
        A = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int l = sc.nextInt(), x = sc.nextInt();
            senators.add(new int[]{ l, x });
        }
        System.out.printf("%.10f\n", calc());
    }
}
