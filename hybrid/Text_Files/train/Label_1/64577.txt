import sys
MOD = 2
xs = [0] * 3000

for line in sys.stdin.readlines()[1:]:
    x = int(line)
    cnt = 0
    tmp = []
    for i in range(29, -1, -1):
        if (x >> i) & 1:
            if xs[i]:
                cnt += 1
                tmp.append(xs[i])
            else:
                cnt = 0
                tmp.clear()
                break
        xs[i] = x
    print((cnt, *tmp) if cnt else 0)
