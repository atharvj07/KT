#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> nums(n);
  for (int i = 0; i < n; ++i) {
    cin >> nums[i];
  }

  int zero = -1;
  for (int i = 0; i < n; ++i) {
    if (nums[i] == 0) {
      zero = i;
      break;
    }
  }

  if (zero == -1) {
    cout << -1 << endl;
    return 0;
  }

  int m = -1;
  for (int i = 0; i < n; ++i) {
    if (nums[i] > nums[zero]) {
      m = i;
      break;
    }
  }

  if (m == -1) {
    for (int i = n - 1; i >= 0; --i) {
      if (nums[i] > nums[zero]) {
        m = i;
        break;
      }
    }
  }

  if (m == -1) {
    cout << -1 << endl;
    return 0;
  }

  int max_dist = 0;
  int l = zero;
  int r = m;
  while (l >= 0 && r < n) {
    if (nums[l] == nums[r]) {
      max_dist = max(max_dist, nums[l]);
      l--;
      r++;
    } else {
      break;
    }
  }

  if (m - zero == max_dist) {
    cout << max_dist + 1 << " " << max_dist + 1 << endl;
    cout << zero / max_dist + 1 << " " << zero % max_dist + 1 << endl;
  } else {
    cout << -1 << endl;
  }

  return 0;
}
