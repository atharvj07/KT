#include<bits/stdc++.h>
using namespace std;

const int maxN = 110;
int n, m, k;
int a[maxN][maxN];

int dx[] = {0, 1, 0, -1};
int dy[] = {1, 0, -1, 0};

void dfs(int x, int y, int v) {
    if(x < 1 || x > n || y < 1 || y > m || a[x][y] != v)
        return;

    a[x][y] = 2;

    for(int i = 0; i < 4; i++) {
        dfs(x + dx[i], y + dy[i], v);
    }
}

int main() {
    cin >> n >> m >> k;

    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            cin >> a[i][j];
        }
    }

    int components = 0;

    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(a[i][j] == 1) {
                dfs(i, j, 1);
                components++;
            }
        }
    }

    int cells = 0;

    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(a[i][j] != 2) {
                cells++;
            }
        }
    }

    if(cells > k) {
        cout << -1 << endl;
        return 0;
    }

    if(components == 1) {
        cout << 0 << endl;
        return 0;
    }

    int ans = cells;

    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(a[i][j] == 1) {
                int cnt = 0;

                for(int p = 0; p < 4; p++) {
                    int nx = i + dx[p], ny = j + dy[p];

                    if(nx >= 1 && nx <= n && ny >= 1 && ny <= m && a[nx][ny] == 1) {
                        cnt++;
                    }
                }

                if(cnt == 1) {
                    ans = min(ans, cells - 1);
                }
                else if(cnt == 2) {
                    ans = min(ans, cells - 2);
                }
            }
        }
    }

    cout << ans << endl;

    return 0;
}
