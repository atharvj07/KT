#include <bits/stdc++.h>

using namespace std;

signed main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  string res(n, 'B');
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (a[j] % a[i] == 0) {
        res[i] = 'A';
        break;
      }
    }
  }
  cout << res << endl;
  return 0;
}
