import bisect
n, m, k, q = map(int, input().split())
a = [[-1] * (m + 1) for _ in range(n + 1)]
ans = -1
for _ in range(q):
    x, y, t = map(int, input().split())
    if a[x][y] == -1:
        a[x][y] = t
        for j in range(x, x + k):
            for i in range(y, y + k):
                if a[j][i] < t:
                    ans = t
                    break
            if ans != -1:
                break
if ans == -1:
    ans = sorted(a[x][y] for x in range(1, n + 1) for y in range(1, m + 1))[-k ** 2]
print(ans)
