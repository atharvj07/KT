#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    // input
    int n, k;
    cin >> n >> k;
    vector<int> types(n);
    for (int i = 0; i < n; i++) cin >> types[i];

    // solve
    int e = 0, s = 0;
    for (int i = 0; i < n; i++) {
        if ((i + 1) % k == 0) continue;
        if (types[i] == 1) e++;
        else s++;
    }

    // output
    cout << abs(e - s) << endl;
}
