#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> ans;
    for (int i = 0; i < n - 1; i++) {
      ans.push_back(9);
    }
    if (n % 2 == 1) {
      ans.push_back(8);
    } else {
      ans.push_back(6);
    }
    int p = 2;
    while (ans.size() < n) {
      ans.push_back(p);
      p = (p + 1) % 10;
    }
    for (int i = n - 1; i >= 0; i--) {
      cout << ans[i];
    }
    cout << endl;
  }
  return 0;
}
