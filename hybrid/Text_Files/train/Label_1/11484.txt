#include<vector>
#include<iostream>
using namespace std;
long long c[150100];
int main(){
    int n;
    cin>>n;
    vector<int> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
        c[a[i]]++;
    }
    long long ans=c[a[0]]*(c[a[0]]-1)/2,ans1,ans2;
    for(int i=1;i<n;i++){
        ans+=(c[a[i]]*(c[a[i]]-1))/2;
        ans+=c[a[i]]*c[a[i]-1];
        ans+=c[a[i]]*c[a[i]+1];
    }
    for(int i=0;i<n;i++){
        ans1=0,ans2=0;
        if(i>0){
            ans1+=c[a[i]]*(c[a[i]]-1)/2;
            ans1+=c[a[i]]*c[a[i]-1];
            ans1+=c[a[i]-1]*(c[a[i]-1]-1)/2;
            if(i-2>=0){
                ans1+=c[a[i]]*c[a[i]-2];
            }
        }
        if(i+1<n){
            ans2+=c[a[i]]*(c[a[i]]-1)/2;
            ans2+=c[a[i]]*c[a[i]+1];
            ans2+=c[a[i]+1]*(c[a[i]+1]-1)/2;
            if(i+2<n){
                ans2+=c[a[i]]*c[a[i]+2];
            }
        }
        ans-=max(ans1,ans2);
    }
    cout<<ans/2<<endl;
}
