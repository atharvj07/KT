n, m, k = map(int, input().split())
s, t = input(), input()
mn = min(n, m)
a = [[-1]*mn for _ in range(mn)]
for i in range(mn):
    for j in range(mn):
        if s[i] == t[j]:
            if i == 0 or j == 0:
                a[i][j] = 1
            else:
                a[i][j] = a[i - 1][j - 1] + 1
pos = []
for i in range(mn):
    for j in range(mn):
        if a[i][j] == k:
            pos.append((i, j))
ans = 0
for l, r in pos:
    ans += s[l - a[l][r] + 1: l + 1] == t[r - a[l][r] + 1: r + 1]
print(ans)
