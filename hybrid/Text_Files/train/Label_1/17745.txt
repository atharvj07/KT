#include <cstdio>
#include <vector>
using namespace std;
const int MAXN = 1e3;
unsigned int a[MAXN], b[MAXN];
decltype(a) pre[MAXN];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%u", a + i);
    scanf("%u", b + i);
  }
  for (int i = 1; i < n; i++) {
    a[i] |= a[i - 1];
    b[i] |= b[i - 1];
  }
  unsigned int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) { ans = max(ans, a[j] | b[j]); }
  }
  printf("%u\n", ans);
}
