n = int(input())
segments = [list(map(int, input().split())) for _ in range(n)]

max_left = max(segments, key=lambda x: x[0])
min_right = min(segments, key=lambda x: x[1])

cover_all = True
for segment in segments:
    if max_left[0] > segment[0] or min_right[1] < segment[1]:
        cover_all = False
        break

if cover_all and max_left[0] == min_right[0] and max_left[1] == min_right[1]:
    print(segments.index(max_left) + 1)
else:
    print(-1)
