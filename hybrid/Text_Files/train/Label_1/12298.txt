#include <iostream>
#include <vector>
#include <algorithm>
#include <bitset>

using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 1e5 + 5;
const int MOD = 1e9 + 7;

struct Pizza {
  int price;
  int num_ingredients;
  vector<int> ingredients;
};

int n, m;
vector<int> friends[MAXN];
vector<Pizza> pizzas[MAXM];
int dp[MAXN][1 << 9];

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    int num_ingredients;
    cin >> num_ingredients;
    for (int j = 0; j < num_ingredients; j++) {
      int ingredient;
      cin >> ingredient;
      friends[i].push_back(ingredient);
    }
  }
  for (int i = 0; i < m; i++) {
    Pizza pizza;
    cin >> pizza.price >> pizza.num_ingredients;
    for (int j = 0; j < pizza.num_ingredients; j++) {
      int ingredient;
      cin >> ingredient;
      pizza.ingredients.push_back(ingredient);
    }
    pizzas[i].push_back(pizza);
  }
  int best_score = 0;
  int best_pizza1 = -1;
  int best_pizza2 = -1;
  for (int i = 0; i < m; i++) {
    for (int j = i + 1; j < m; j++) {
      int score = 0;
      bitset<10> ingredients;
      for (int k = 0; k < pizzas[i].size(); k++) {
        for (int l = 0; l < pizzas[i][k].ingredients.size(); l++) {
          ingredients[pizzas[i][k].ingredients[l]] = true;
        }
      }
      for (int k = 0; k < pizzas[j].size(); k++) {
        for (int l = 0; l < pizzas[j][k].ingredients.size(); l++) {
          ingredients[pizzas[j][k].ingredients[l]] = true;
        }
      }
      for (int k = 0; k < n; k++) {
        bool satisfied = true;
        for (int l = 0; l < friends[k].size(); l++) {
          if (!ingredients[friends[k][l]]) {
            satisfied = false;
            break;
          }
        }
        if (satisfied) {
          score++;
        }
      }
      if (score > best_score) {
        best_score = score;
        best_pizza1 = i;
        best_pizza2 = j;
      }
    }
  }
  cout << best_pizza1 + 1 << " " << best_pizza2 + 1 << endl;
  return 0;
}
