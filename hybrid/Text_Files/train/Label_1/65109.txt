import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, sys.stdin.readline().split()))
    stack = []
    inversions = 0
    for i in range(n):
        if not stack or stack[-1] >= a[i]:
            stack.append(a[i])
        else:
            while stack and stack[-1] < a[i]:
                inversions += len(stack)
                stack.pop()
            stack.append(a[i])
    print(inversions)
