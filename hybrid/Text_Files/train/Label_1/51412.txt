import java.util.*;
import static java.lang.Math.ceil;

class Main {
    static List<String> minMoves = new ArrayList<String>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int min = minRides(n, k, p);
        System.out.println(min == -1 ? -1 : min);
        System.out.println(min == -1 ? 0 : minMoves.size() % 1000000007);
        sc.close();
    }

    static int minRides(int n, int k, int[] p) {
        List<Integer> a = new ArrayList<Integer>();
        for (int x : p) {
            if (x == 50) {
                a.add(x);
            }
        }
        int m = a.size();
        if (k < 50 || (k == 50 && m % 2 != 0)) {
            return -1;
        }
        int[] dp = new int[m + 1];
        for (int i = 0; i <= m; i++) {
            dp[i] = i % 2 == 0 ? i / 2 : (i + 1) / 2;
        }
        List<Integer> b = new ArrayList<Integer>();
        for (int x : p) {
            if (x == 100) {
                b.add(x);
            }
        }
        int l = b.size();
        if (m == 0 && (l * 100) > k) {
            return -1;
        }
        int r = (int) ceil((double) l * 100 / (double) k);
        for (int s = m + 1; s <= m + r; s++) {
            for (int j = m; j >= 0; j--) {
                dp[s] = Math.min(dp[s], dp[j] + dp[s - j]);
            }
        }
        int res = dp[m + r];
        if (res == -1) {
            return -1;
        }
        rec(k, l, p, res, res - 1, "", 0, 0);
        return res;
    }

    static void rec(int k, int l, int[] p, int res, int r, String resMoves, int x, int mx) {
        if (x == p.length) {
            if (mx == l) {
                minMoves.add(resMoves);
            }
            return;
        }
        if (p[x] == 50) {
            if (r >= 0) {
                rec(k, l, p, res, r - 1, resMoves + (r == 0 ? "" : " ") + "1", x + 1, mx);
            }
        } else {
            if (mx < res - 1) {
                rec(k, l, p, res, r, resMoves + (r == 0 ? "" : " ") + "2", x + 1, mx + 1);
            }
            if (r >= 1) {
                rec(k, l, p, res, r - 2, resMoves + (r == 1 ? "" : " ") + "3", x + 1, mx);
            }
        }
    }
}
