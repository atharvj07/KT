from collections import defaultdict
def main():
    n, m, nchar = map(int, input().split())
    b = list(map(int, input().split()))
    if nchar == 1:
        print(1)
        return
    k = pow(2, nchar - 1, 998244353)
    ans = pow(k, m, 998244353)
    sb = defaultdict(int)
    for x in b:
        sb[x] += 1
    for x in sb:
        if sb[x] > 1:
            t = pow(k, sb[x] - 1, 998244353)
            ans = (ans * t) % 998244353
    print(ans)
main()
