import java.util.*;
import java.io.*;

public class Main {
    final private static Long MOD = (long)1e9;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Long a = scanner.nextLong(), b = scanner.nextLong(), n = scanner.nextLong();
        System.out.println((Combination(a + b, b) < n ? "Masha" : "Stas".equalsIgnoreCase(n % 2 == 0 ? "Missing" : "Stas")));
    }

    private static Long Combination(long x, long y) {
        return (fact(x) * power(fact(x - y) * fact(y), MOD - 2)) % MOD;
    }

    private static Long fact(long x) {
        Long res = 1L;
        for (int i = 2; i <= x; ++i) res = (res * i) % MOD;
        return res;
    }

    private static Long power(Long x, Long y) {
        if (y == 0) return 1L;
        Long res = power(x, y / 2);
        res = (res * res) % MOD;
        return y % 2 != 0 ? (res * x) % MOD : res;
    }
}
