import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        long[] dp = new long[n];
        for (int i = 0; i < n; i++) {
            dp[i] = 1;
        }
        for (int i = 1; i < n; i++) {
            if (h[i] <= h[i - 1]) {
                dp[i] = dp[i - 1] + 1;
            }
        }
        long sum = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (h[i] >= h[i + 1]) {
                dp[i] = dp[i + 1] + 1;
            }
            sum += dp[i];
        }
        sum %= 1000000007;
        System.out.println(sum);
    }
}
