n, m = map( int, input().split() )
F = [ input() for _ in range(n) ]
pref = [[ 0 for j in range( m+1 ) ] for i in range( n+1 ) ]
for i in range( 1, n+1 ):
    for j in range( 1, m+1 ):
        pref[i][j] = pref[i-1][j] + pref[i][j-1] - pref[i-1][j-1] + (1 if F[i-1][j-1] == 'W' else -1)
def getr( x1, y1, x2, y2 ):
    return pref[x2][y2] - pref[x1-1][y2] - pref[x2][y1-1] + pref[x1-1][y1-1]
ans = 999999999
for x2 in range( 1, n+1 ):
    for y2 in range( 1, m+1 ):
        for x1 in range( x2 ):
            for y1 in range( y2 ):
                if getr( x1, y1, x2, y2 ) == 0:
                    ans = min( ans, 2 )
                else:
                    ans = min( ans, 3 )
print( ans )
