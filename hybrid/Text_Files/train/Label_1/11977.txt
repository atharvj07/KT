#include<bits/stdc++.h>
using namespace std ;
#define ll long long
string s ;
ll n , c0 , c1 ;
priority_queue <ll , vector<ll> , greater<ll> > pq ;
int main()
{
    cin >> n >> c0 >> c1 ;
    for(ll i = 2 , j = 1 ; i <= n ; i*=2)
    {
        pq.push(i*c0 + j*c1) ;
        j++ ;
    }
    ll ans = 0 ;
    while(pq.size() > 1)
    {
        ll a = pq.top() ;
        pq.pop() ;
        ll b = pq.top() ;
        pq.pop() ;
        ans += a + b ;
        pq.push(a + b) ;
    }
    cout << ans << endl ;
    return 0 ;
}
