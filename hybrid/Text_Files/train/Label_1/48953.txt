import java.util.*;

public class Main {
    static class Pair implements Comparable<Pair> {
        int id;
        long val;

        Pair(int id, long val) {
            this.id = id;
            this.val = val;
        }

        public int compareTo(Pair o) {
            if (this.val == o.val) {
                return this.id - o.id;
            }

            return o.val - this.val > 0 ? 1 : -1;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int q = sc.nextInt();
        PriorityQueue<Pair> pq = new PriorityQueue<>();
        HashMap<Integer, Pair> hm = new HashMap<>();
        long[] arr = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            arr[i] = sc.nextLong();
        }

        for (int i = 1; i <= n; i++) {
            pq.add(new Pair(i, 0));
            hm.put(i, new Pair(i, arr[i]));
        }

        for (int i = 0; i < q; i++) {
            int t = sc.nextInt();
            int id = sc.nextInt();
            if (t == 1) {
                pq.remove(hm.get(id));
                hm.get(id).val = arr[id];
                pq.add(hm.get(id));
            } else {
                boolean ok = false;
                for (int j = 0; j < k; j++) {
                    Pair p = pq.peek();
                    if (p.id == id) {
                        ok = true;
                        break;
                    }
                    pq.remove();
                    pq.add(p);
                }

                System.out.println(ok ? "YES" : "NO");
            }
        }
    }
}
