import java.util.Scanner;

public class AdjacentNumbersMatrix {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();

            if (n % 2 == 0) {
                System.out.println(-1);
            } else {
                int[][] matrix = new int[n][n];

                int count = 1;
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) {
                        if ((i + j) % 2 == 0) {
                            matrix[i][j] = count;
                            count += 2;
                        }
                    }
                }

                count = 2;
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) {
                        if ((i + j) % 2 != 0) {
                            matrix[i][j] = count;
                            count += 2;
                        }
                    }
                }

                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) {
                        System.out.print(matrix[i][j] + " ");
                    }
                    System.out.println();
                }
            }
        }
    }
}
