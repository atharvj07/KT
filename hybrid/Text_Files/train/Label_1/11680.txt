#include<bits/stdc++.h>
using namespace std;
vector<int>a;
const int N = 1e5+1;
int need[N], hav[2];
int main()
{
	int x, y, z, w, n;
	cin >> x >> y >> z >> w >> n;
	for (int i = 0; i < n; i++)
	{
		int t;
		cin >> t;
		a.push_back(t);
	}
	if (x > z)
	{
		swap(z, x);
		swap(w, y);
	}
	need[x] = (y + x - 1) / x;
	need[z] = (w + z - 1) / z;

	for (int i = 0; i < n; i++)
	{
		int t = a[i];
		if (t <= 1000)
		{
			hav[t]++;
			continue;
		}
		if (t <= x && t <= z)
		{
			int tt = (x + z) / t;
			hav[0] += tt / 2;
			hav[1] += tt - tt / 2;
			continue;
		}
		if (t <= x)
		{
			hav[0] += need[x] / (t / x);
			continue;
		}
		if (t <= z)
		{
			hav[1] += need[z] / (t / z);
		}
	}
	int ans = 1e9;
	if (hav[0] >= need[x])
		ans = min(ans, need[z]);
	if (hav[1] >= need[z])
		ans = min(ans, need[x]);
	int t = min(x, z);
	for (int i = 2; i <= 1000; i++)
	{
		if (x % i == 0 && z % i == 0)
			t = min(t, i);
	}
	int ggg = (x + z) / t;
	if (hav[0] >= ggg / 2 && hav[1] >= ggg / 2)
		ans = min(ans, ggg);
	if (ans == 1e9)
		ans = -1;
	cout << ans << endl;
	
}
