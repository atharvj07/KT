import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    long n = input.nextLong(), k = input.nextLong();
    int digits = (Long.toString(n).length() - 1), ans = 0;
    long factor = 1;
    for (int i = 0; i < digits; i++) {
      factor *= 10;
      long rem = n % factor;
      if (rem % (10 * k) == 0) ans = i;
    }
    System.out.println(digits - ans);
  }
}
