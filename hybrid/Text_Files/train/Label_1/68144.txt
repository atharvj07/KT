from sys import stdin
rd = stdin.readline

n,q = map(int,rd().split())
arr = list(map(int,rd().split()))
ans = [0]*q
cur = [0]*n

for i in range(1,n):
    diff = arr[i]-arr[i-1]
    cur[i] = max(cur[i-1],diff)
for i in range(q):
    l,r = map(int,rd().split())
    ans[i] = cur[r-1]-cur[l-2]
for i in ans:    print(i)
