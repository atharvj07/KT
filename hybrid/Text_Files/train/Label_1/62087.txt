import sys
input = sys.stdin.readline

n,m = map(int,input().split())
a = list(map(int,input().split()))
for i in range(m):
    l,r,x = map(int,input().split())
    l-=1
    r-=1
    found = False
    for j in range(l,r+1):
        if a[j]!=x:
            print(j+1)
            found = True
            break
    if not found:
        print(-1)
