n = int(input())
p = list(map(int, input().split()))

def deviation(shift_p):
    return sum(abs(shift_p[i] - (i + 1)) for i in range(n))

min_deviation = float('inf')
best_shift = 0

for k in range(n):
    current_deviation = deviation(p)
    if current_deviation < min_deviation:
        min_deviation = current_deviation
        best_shift = k
    p = [p[-1]] + p[:-1]

print(min_deviation, best_shift)
