import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            long n = in.nextLong();
            long k = in.nextLong();
            if (k % n <= (n + 1) / 2) {
                System.out.println((k % n) + ((n + 1) / 2));
            } else {
                System.out.println(k % n);
            }
        }
    }
}
