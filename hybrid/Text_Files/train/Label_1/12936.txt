#include <algorithm>
#include <iostream>
#include <map>
#include <vector>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k, m;
  cin >> n >> k >> m;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  if ((k > a[n - 1] - a[0]) || ((k - 1) * m > a[n - 1] - a[0])) {
    cout << "No" << endl;
    return 0;
  }
  cout << "Yes" << endl;
  int ans[k];
  map<int, int> mp;
  for (int i = 0; i < n; i++) mp[a[i]]++;
  int cur = a[0] + m * k;
  for (int i = 1; i < n; i++) {
    int cnt = 0;
    while (cur >= a[i] && cnt < mp[a[i]]) {
      ans[cnt] = a[i];
      cnt++;
      cur -= m;
    }
    mp[a[i]] -= cnt;
    if (i + 1 == n && cnt < k) {
      for (int j = i; j >= 0; j--) {
        if (mp[a[j]]) {
          ans[cnt] = a[j];
          mp[a[j]]--;
          cnt++;
        }
        if (cnt == k) break;
      }
    }
    if (cnt == k) break;
    cur = a[i] + m * ((k - cnt - 1) / 2);
  }
  for (int i = 0; i < k; i++) cout << ans[i] << " ";
  cout << endl;
}
