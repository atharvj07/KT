import sys

n, x = map(int, input().split())

if n < 3 or n > 5 or (n == 3 and x % 3 != 0) or (n == 4 and x % 4 != 0) or (n == 5 and x % 5 != 0):
    print("NO")
else:
    print("YES")
    if n == 3:
        print(1, x - 1, x)
    elif n == 4:
        print(1, x - 1, x // 2, x // 2 + x % 2)
    elif n == 5:
        print(1, x - 1, (x - 1) // 2, (x - 1) // 2 + 1, x)
