#include<bits/stdc++.h>
using namespace std;
int main()
{
  int n; cin>>n;
  vector<pair<int,int>>v(n);
  for(int i=0;i<n;i++)
  {
    int a,b; cin>>a>>b;
    v[i]={min(a,b),max(a,b)};
  }
  vector<int>h;
  for(pair<int,int> x: v)
  {
    if(h.empty()) h.push_back(x.second);
    else if(h.back()>=x.second) h.push_back(x.second);
    else if(h.back()>x.first) return puts("NO"),0;
    else h.back()=x.first;
  }
  puts("YES");
  return 0;
}
