#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<array<int, 5>> a(n);
  for (auto &i : a) {
    for (auto &j : i) { cin >> j; }
  }
  vector<int> ans;
  for (int i = 0; i < n; ++i) {
    bool ok = 0;
    for (int j = 0; j < n; ++j) {
      for (int k = 0; k < n; ++k) {
        if (i == j || i == k || j == k) continue;
        array<int, 5> x = {a[i][0] - a[j][0], a[i][1] - a[j][1],
                           a[i][2] - a[j][2], a[i][3] - a[j][3],
                           a[i][4] - a[j][4]};
        array<int, 5> y = {a[i][0] - a[k][0], a[i][1] - a[k][1],
                           a[i][2] - a[k][2], a[i][3] - a[k][3],
                           a[i][4] - a[k][4]};
        double dot = 0;
        for (int l = 0; l < 5; ++l) { dot += x[l] * 1LL * y[l]; }
        double norm_x = 0;
        for (int l = 0; l < 5; ++l) { norm_x += x[l] * 1LL * x[l]; }
        double norm_y = 0;
        for (int l = 0; l < 5; ++l) { norm_y += y[l] * 1LL * y[l]; }
        double cos = dot / (sqrt(norm_x) * sqrt(norm_y));
        if (cos < 0) { ok = 1; }
      }
    }
    if (!ok) ans.push_back(i + 1);
  }
  cout << ans.size() << endl;
  for (auto i : ans) { cout << i << endl; }
  return 0;
}
