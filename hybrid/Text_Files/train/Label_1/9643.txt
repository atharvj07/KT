#include<bits/stdc++.h>
using namespace std;

#define ll long long
#define MOD 1000000007

ll pw(ll x, ll y) {
    ll ret = 1;
    while (y) {
        if (y % 2 == 1) {
            ret *= x;
            ret %= MOD;
        }
        x *= x;
        x %= MOD;
        y /= 2;
    }
    return ret;
}

ll fact(ll x) {
    if (x == 0) return 1;
    ll ret = 1;
    for (ll i = 1; i <= x; i++) {
        ret *= i;
        ret %= MOD;
    }
    return ret;
}

int main() {
    ll n, m;
    cin >> n >> m;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    ll num = 0;
    for (ll i = 0; i < n - 1; i++) {
        num += a[i + 1] - a[i];
        num %= MOD;
    }
    ll den = fact(n - 1);
    den %= MOD;
    ll ans = (num * pw(den, MOD - 2)) % MOD;
    cout << ans << endl;
    return 0;
}
