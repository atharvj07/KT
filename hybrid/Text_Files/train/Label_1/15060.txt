#include<bits/stdc++.h>
using namespace std;
int n,m,vis[55][55],sx,sy;
char grid[55][55];
bool check(int x,int y) {
	if(x<0 || y<0 || x>=n || y>=m || grid[x][y]=='W' || vis[x][y]==1) return 0;
	return 1;
}
bool dfs(int x,int y,int prv) {
	if(x<0 || y<0 || x>=n || y>=m || grid[x][y]=='W') return 0;
	if(vis[x][y]==1 && prv!=5) return 0;
	if(vis[x][y]==2 && prv!=6) return 0;
	if(vis[x][y]==3 && prv==7) return 0;
	if(vis[x][y]==4 && prv==8) return 0;
	vis[x][y]=prv;
	if(prv==1) {
		if(!dfs(x+1,y,2)) return 0;
		if(!dfs(x-1,y,3)) return 0;
		if(!dfs(x,y+1,4)) return 0;
	}
	else if(prv==2) {
		if(!dfs(x+1,y,1)) return 0;
		if(!dfs(x,y-1,8)) return 0;
	}
	else if(prv==3) {
		if(!dfs(x-1,y,1)) return 0;
		if(!dfs(x,y+1,6)) return 0;
	}
	else if(prv==4) {
		if(!dfs(x,y-1,7)) return 0;
		if(!dfs(x+1,y,1)) return 0;
	}
	else if(prv==6) {
		if(!dfs(x-1,y,1)) return 0;
		if(!dfs(x,y-1,5)) return 0;
	}
	else if(prv==7) {
		if(!dfs(x,y-1,5)) return 0;
		if(!dfs(x+1,y,1)) return 0;
	}
	else if(prv==8) {
		if(!dfs(x+1,y,1)) return 0;
		if(!dfs(x,y+1,4)) return 0;
	}
	else if(prv==5) {
		if(!dfs(x,y+1,6)) return 0;
		if(!dfs(x-1,y,1)) return 0;
	}
	return 1;
}
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	cin>>n>>m;
	for(int i=0; i<n; i++) {
		for(int j=0; j<m; j++) {
			cin>>grid[i][j];
			if(grid[i][j]=='B') {
				sx=i,sy=j;
			}
		}
	}
	if(dfs(sx,sy,1)) cout<<"YES";
	else cout<<"NO";
}
