import sys

n,m=map(int,input().split())
a=[]
for i in range(m):
	x,y,z,w=map(int,input().split())
	a.extend([(x-1,y-1,z,w),(x-1,w,z,n),(z,y-1,n,w),(z,w,n,n)])
a.sort()
ans=0
j,x=-1,0
for i in range(len(a)):
	if i and a[i][0]==a[i-1][0]:continue
	y,z,w,t=a[i]
	x=min(w,t)
	ans+=min((z-y+1)*(x-j),x-j)
	j=x
print(ans)
