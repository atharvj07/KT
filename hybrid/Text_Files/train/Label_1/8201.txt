#include <cstdio>

const int MAXN = 1e5 + 5;

int n, m, k;
int p[MAXN], ans;
bool vis[MAXN];

int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i <= m; i++)
    scanf("%d", &p[i]);
  for (int i = 1; i <= m; i++)
    for (int j = p[i]; j <= n; j += k)
      vis[j] = 1;
  for (int i = 1; i <= n; i += k) {
    int ok = 0;
    for (int j = i; j < i + k; j++) {
      if (!vis[j]) {
        for (int u = j - 1; u >= i && u % k > ok; u--)
          vis[u] = 0;
      }
      ok = j % k + 1;
    }
    ans += vis[i];
  }
  printf("%d\n", ans);
  return 0;
}
