n,m=map(int,input().split())
t=[]
for i in range(n):
	a,b=map(int,input().split())
	t.append((a,b))
visit=[False for i in range(m+1)]
pos=[0]
while len(pos)>0:
	cur=pos.pop()
	if cur==m:
		print("YES")
		exit()
	visit[cur]=True
	for a,b in t:
		if a<=cur<=b and not visit[b]:
			pos.append(b)
print("NO")
