import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        Map<String, Integer> price = new HashMap<>();
        for (int i = 0; i < N; i++) {
            String name = sc.next();
            int x = sc.nextInt();
            price.put(name, x);
        }
        int M = sc.nextInt();
        Map<String, String> mat = new HashMap<>();
        for (int i = 0; i < M; i++) {
            String s = sc.next();
            String t = sc.next();
            mat.put(s, t);
            mat.put(t, s);
        }
        int min = Integer.MAX_VALUE;
        for (String start : price.keySet()) {
            Set<String> used = new HashSet<>();
            min = Math.min(min, dfs(start, mat, price, used));
        }
        System.out.println(min);
    }

    private static int dfs(String start, Map<String, String> mat, Map<String, Integer> price, Set<String> used) {
        used.add(start);
        int min = price.get(start);
        for (String next : mat.keySet()) {
            if (used.contains(next)) continue;
            min = Math.min(min, dfs(next, mat, price, used));
        }
        used.remove(start);
        return min;
    }
}
