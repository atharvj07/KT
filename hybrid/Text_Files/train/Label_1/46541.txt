import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int[] reversed = new int[s.length()];
        int[] lastReversed = new int[s.length()];
        int[] countA = new int[s.length()];
        int[] countAllA = new int[s.length()];
        for (int i = 0; i < s.length(); i++) {
            countA[i] = s.charAt(i) == 'a' ? 1 : 0;
            countAllA[i] = i > 0 ? countAllA[i - 1] + countA[i] : countA[i];
        }
        for (int i = s.length() - 1; i >= 0; i--) {
            if (s.charAt(i) == 'a') {
                if (i < s.length() - 1 && s.charAt(i + 1) == 'b' && (i < s.length() - 2 && s.charAt(i + 2) == 'a' || i == s.length() - 2)) {
                    reversed[i] = 1;
                    lastReversed[i] = i;
                } else if (i == s.length() - 1 || i > 0 && s.charAt(i - 1) == 'b') {
                    lastReversed[i] = i;
                }
            } else {
                lastReversed[i] = i;
            }
            if (i > 0) {
                lastReversed[i - 1] = Math.max(lastReversed[i - 1], lastReversed[i]);
            }
        }
        int aBefore = 0;
        for (int i = 0; i < s.length(); i++) {
            aBefore = i > 0 ? countAllA[i - 1] : 0;
            if (i > 0 && lastReversed[i - 1] == i && (i == s.length() - 1 || s.charAt(i + 1) == 'a')) {
                reversed[i] = 1;
            } else if (i < s.length() - 1 && countA[i] + aBefore >= countAllA[s.length() - 1] - countAllA[i]) {
                reversed[i] = 1;
            }
            System.out.print(reversed[i]);
        }
    }
}
