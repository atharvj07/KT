#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int cnt = 0;
    for (int i = 0; i < n - 1; i++) {
        if (a[i + 1] - a[i] > d) {
            cnt++;
            i++;
        }
    }
    cout << cnt;
    return 0;
}
