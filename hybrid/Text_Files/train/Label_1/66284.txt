def find_longest_k_good_segment(n, k, arr):
    from collections import defaultdict

    # Helper function to add an element to the current window
    def add_element(num):
        if counts[num] == 0:
            nonlocal unique
            unique += 1
        counts[num] += 1

    # Helper function to remove an element from the current window
    def remove_element(num):
        counts[num] -= 1
        if counts[num] == 0:
            nonlocal unique
            unique -= 1

    left, right = 0, 0
    max_len = 0
    counts = defaultdict(int)
    unique = 0
    res_l, res_r = 0, 0

    while right < n:
        add_element(arr[right])
        while unique > k:
            remove_element(arr[left])
            left += 1
        if right - left + 1 > max_len:
            max_len = right - left + 1
            res_l, res_r = left, right
        right += 1

    return res_l + 1, res_r + 1

# Read input
n, k = map(int, input().split())
arr = list(map(int, input().split()))

# Find and print the indices of the longest k-good segment
l, r = find_longest_k_good_segment(n, k, arr)
print(l, r)
