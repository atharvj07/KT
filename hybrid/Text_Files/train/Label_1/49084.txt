import java.util.*;

public class Main {
    static final int MOD = 998244353;
    int n, m;
    long[] dp, pw;
    List<Integer>[] g;
    boolean[] vis;
    int[] l, r;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        new Main().solve(sc);
    }

    void dfs(int u, int p) {
        vis[u] = true;
        for (int v: g[u]) {
            if (v == p) continue;
            dfs(v, u);
            dp[u] = (dp[u] + (pw[l[v]] - pw[l[v] - 1] + pw[r[v]] - pw[r[v] - 1]) % MOD * dp[v] % MOD) % MOD;
        }
    }

    void solve(Scanner sc) {
        n = sc.nextInt();
        m = sc.nextInt();
        dp = new long[n + 1];
        Arrays.fill(dp, 1);
        pw = new long[n + 1];
        g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) g[i] = new ArrayList<>();
        vis = new boolean[n + 1];
        l = new int[n + 1];
        r = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            g[u].add(v);
            g[v].add(u);
        }
        pw[0] = 1;
        for (int i = 1; i <= n; i++) pw[i] = (pw[i - 1] * 2) % MOD;
        dfs(1, 1);
        System.out.println(dp[1]);
    }
}
