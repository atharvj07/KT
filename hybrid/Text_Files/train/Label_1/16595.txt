#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;
#define vi(x) vector<x>
#define pb push_back
const ll mod = 1e9 + 7;
const char nl = '\n';
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    ll n, m;
    cin >> n >> m;
    vi(ll) a(n);
    for (ll i = 0; i < n; ++i) { cin >> a[i]; }
    ll ans = 2 * m - 1;
    for (ll i = 0; i < n; ++i) {
        for (ll x = 1; x < m; ++x) {
            if (a[i] % x == 0) { ans -= (a[i] / x >= m - x); }
        }
    }
    cout << ans << nl;
    return 0;
}
