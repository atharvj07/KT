#include <iostream>
using namespace std;

const int N = 2e5 + 7;
int n, m, d;
int a[N];
int days[N];

int main() {
  cin >> n >> m >> d;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  int day = 0;
  for (int i = 1; i <= n; i++) {
    if (day * d + 1 < a[i]) day++;
    days[i] = day;
  }
  cout << day << '\n';
  for (int i = 1; i <= n; i++) {
    cout << days[i] << ' ';
  }
  return 0;
}
