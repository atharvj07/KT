import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        String s = sc.next();
        if (s.length() != a + b + 1 || s.charAt(a) != '-') {
            System.out.println("No");
        } else {
            System.out.println("Yes");
        }
    }
}
