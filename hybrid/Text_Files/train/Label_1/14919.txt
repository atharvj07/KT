#include <iostream>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  if (s.size() != t.size())
    cout << "need tree";
  else if (s == t)
    cout << "automaton";
  else if (s[0] != t[0] || s[s.size() - 1] != t[t.size() - 1])
    cout << "need tree";
  else
    cout << "array";
}
