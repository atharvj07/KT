#include <bits/stdc++.h>
#define nl "\n"
using namespace std;
typedef long long ll;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int t, n, W;
    ll w;
    cin >> t;
    while (t--) {
        cin >> n >> W;
        vector<pair<ll, int>> weights(n);
        for (int i = 0; i < n; i++) {
            cin >> w;
            weights[i] = {w, i + 1};
        }
        sort(weights.begin(), weights.end());
        bool flag = true;
        ll sum = 0;
        vector<int> ans;
        for (int i = n - 1; i >= 0; i--) {
            if (sum + weights[i].first > W) break;
            sum += weights[i].first;
            ans.push_back(weights[i].second);
            if (sum >= (W + 1) / 2) { flag = false; break; }
        }
        if (!flag) {
            cout << ans.size() << nl;
            for (auto x : ans) cout << x << " ";
            cout << nl;
        } else
            cout << -1 << nl;
    }
}
