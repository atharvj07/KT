n = int(input())
x1, y1, x2, y2 = [], [], [], []

for _ in range(n):
    x1i, y1i, x2i, y2i = map(int, input().split())
    x1.append(x1i)
    y1.append(y1i)
    x2.append(x2i)
    y2.append(y2i)

x1.sort()
y1.sort()
x2.sort()
y2.sort()

if n % 2 == 0:
    print(x1[n//2 - 1], y1[n//2 - 1])
else:
    print(x1[n//2], y1[n//2])
