t = int(input())
for _ in range(t):
    x1, p1 = map(int, input().split())
    x2, p2 = map(int, input().split())

    # Normalize the exponents by reducing the larger one
    min_p = min(p1, p2)
    p1 -= min_p
    p2 -= min_p

    # Compare after normalization
    if p1 > 0:  # If p1 is still positive, it means x1 is definitely larger
        print(">")
    elif p2 > 0:  # If p2 is still positive, it means x2 is definitely larger
        print("<")
    else:  # If both p1 and p2 are zero, compare x1 and x2 directly
        if x1 > x2:
            print(">")
        elif x1 < x2:
            print("<")
        else:
            print("=")
