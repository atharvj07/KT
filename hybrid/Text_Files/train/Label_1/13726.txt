#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    string best = s;
    for (int k = 1; k <= n / 2; ++k) {
      string t = s;
      for (int i = 0; i <= n - k; ++i)
        reverse(t.begin() + i, t.begin() + i + k);
      if (t < best)
        best = t;
    }
    cout << best << endl;
    int k = best.find(s) + 1;
    cout << k << endl;
  }
  return 0;
}
