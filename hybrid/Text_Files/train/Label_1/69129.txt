from sys import stdin

n, m = map(int, stdin.readline().split())
u, d, l, r = [], [], [], []
for _ in range(m):
    x1, y1, x2, y2 = map(int, stdin.readline().split())
    u.append(n - x1)
    d.append(n - x2)
    l.append(y1 - 1)
    r.append(y2)
u.sort()
d.sort()
l.sort()
r.sort()
ans, i, j, k, p = 0, 0, 0, 0, 0
while i < m or j < m or k < m or p < m:
    v, w = 1 << 30, 1 << 30
    if i < m:
        v = n - u[i]
    if j < m:
        w = d[j] - 1
    if k < m:
        v = min(v, r[k] - 1)
    if p < m:
        w = min(w, l[p])
    if v < w:
        ans += l[p] * v
        p += 1
    else:
        ans += r[k] * w
        k += 1
    if p == k and p < m:
        k, p = p, k
    if i == j and i < m:
        i, j = j, i
print(ans)
