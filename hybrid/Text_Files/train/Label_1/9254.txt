#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int maxn = 2e6;

ll f[maxn];
ll ans, ans1;

ll find(ll x)
{
    if (f[x] == x)
        return x;
    return f[x] = find(f[x]);
}

const ll mod = 61678331;

int main()
{
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++)
        f[i] = i, ans1 = (ans1 * i) % mod;

    for (ll i = 1; i <= n; i++)
    {
        ll x;
        cin >> x;
        f[find(i)] = find(x);
    }

    for (ll i = 1; i <= n; i++)
    {
        if (f[i] == i)
            ans = (ans * f[i]) % mod;
    }

    if (ans == ans1)
        cout << "Petr" << endl;
    else
        cout << "Um_nik" << endl;

    return 0;
}
