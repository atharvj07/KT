def count_pairs(n, x, k, a):
    a.sort()
    answer = 0
    for i in range(n):
        left = (a[i] - 1) // x * x + k * x
        right = (a[i] // x) * x + k * x
        if k == 0:
            if a[i] % x == 0:
                left += x
            right = a[i] - 1
        left_index = lower_bound(a, left, i, n)
        right_index = upper_bound(a, right, i, n)
        answer += right_index - left_index
    return answer

def lower_bound(a, value, start, end):
    while start < end:
        mid = (start + end) // 2
        if a[mid] < value:
            start = mid + 1
        else:
            end = mid
    return start

def upper_bound(a, value, start, end):
    while start < end:
        mid = (start + end) // 2
        if a[mid] <= value:
            start = mid + 1
        else:
            end = mid
    return start

n, x, k = map(int, input().split())
a = list(map(int, input().split()))
print(count_pairs(n, x, k, a))
