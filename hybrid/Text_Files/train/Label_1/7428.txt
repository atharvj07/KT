#include<bits/stdc++.h>
using namespace std;
int n,a[100005], cnt = 0;
void get_prime()
{
    a[0] = a[1] = 1;
    for(int i = 2; i <= 100000; i++)
    {
        if(a[i] == 0)
        {
            for(int j = i+i; j <= 100000; j += i) a[j] = 1;
        }
    }
}

int main()
{
    get_prime();
    cin >> n;
    int sq = sqrt(n);
    if(n == 4) { cout << 2 << endl << 2 << " " << 2 << endl; return 0; }
    for(int i = 3; i < sq; i += 2)
    {
        if(a[i] == 0 && a[n-i] == 0)
        {
            cout << 2 << endl << i << " " << n-i << endl;
            return 0;
        }
    }
    if(n%2 == 1) {cout << 1 << endl << n << endl; return 0;}
    else
    {
        cout << 2 << endl << n/2 << " " << n/2 << endl;
        return 0;
    }
}
