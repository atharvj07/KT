import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] p1 = new int[n + 1], p2 = new int[n + 1], b1 = new int[n + 1], b2 = new int[n + 1];
        for (int i = 0; i < m; i++) {
            String action = sc.next();
            int team = sc.nextInt();
            if (action.charAt(0) == 'p') {
                if (team == 1) {
                    p1[i] = s[i];
                    b2[i] = -s[i];
                } else {
                    p2[i] = s[i];
                    b1[i] = -s[i];
                }
            } else {
                if (team == 1) {
                    b1[i] = -s[i];
                } else {
                    b2[i] = -s[i];
                }
            }
        }
        int[] dp1 = new int[m + 1], dp2 = new int[m + 1];
        for (int i = 0; i < m; i++) {
            dp1[i + 1] = Math.max(dp1[i] + p1[i], dp1[i] + b1[i]);
            dp2[i + 1] = Math.max(dp2[i] + p2[i], dp2[i] + b2[i]);
        }
        System.out.println(dp1[m] - dp2[m]);
    }
}
