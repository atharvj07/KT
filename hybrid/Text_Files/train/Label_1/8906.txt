#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    cin >> n;
    string s;
    cin >> s;

    vector<int> gaps;
    for (int j = 0; j < n; j++) {
      if (s[j] == '.') gaps.push_back(j);
    }

    int ans = 0;
    for (int j = 1; j < gaps.size(); j++) {
      int gap = gaps[j] - gaps[j - 1] - 1;
      ans += gap / 2;
    }

    cout << ans << endl;
  }
  return 0;
}
