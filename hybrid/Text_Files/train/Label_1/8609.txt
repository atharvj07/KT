#include<bits/stdc++.h>
using namespace std;

#define ll long long

ll a,r,m,h[100001];

int main(){
    ll n;
    cin>>n>>a>>r>>m;
    ll sum=0;
    for(ll i=0;i<n;i++){
        cin>>h[i];
        sum+=h[i];
    }
    sort(h,h+n);
    ll mi=h[0];
    ll mx=h[n-1];
    ll ans;
    if(sum%n==0){
        ll avg=sum/n;
        if(avg>=mi){
            ans=(avg-mi)*a;
        }else{
            ll diff=mi-avg;
            ans=(diff)*r;
            for(ll i=0;i<n;i++){
                if(h[i]>avg){
                    ans+=(h[i]-avg)*m;
                }
            }
        }
    }else{
        ll left=sum%n;
        ll avg=sum/n;
        if(avg>=mi){
            ans=(avg-mi)*a;
            ans+=((n-left)*m);
        }else{
            ll diff=mi-avg;
            ans=(diff)*r;
            for(ll i=0;i<n;i++){
                if(h[i]>avg){
                    ans+=(h[i]-avg)*m;
                }
            }
            ans+=((left)*a);
        }
    }
    cout<<ans<<endl;
    return 0;
}
