from collections import defaultdict

s = input()

# Hash table for storing suffixes
suff_hash = defaultdict(int)
for i in range(len(s) - 1, -1, -1):
    suff_hash[s[i:]] += 1

# Hash table for storing prefixes
pref_hash = defaultdict(int)
count = 0
for i in range(len(s)):
    pref_hash[s[:i + 1]] += 1
    if s[:i + 1] in suff_hash:
        count += 1
        print(i + 1, suff_hash[s[:i + 1]])
    suff_hash[s[i:]] -= 1

print(count)
