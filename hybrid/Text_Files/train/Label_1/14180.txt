#include <iostream>
#include <vector>
using ll = long long;
const int mod = 1e9 + 7;
ll qpow(ll a, ll b) {
  ll ans = 1;
  while (b) {
    if (b % 2 == 1) ans = ans * a % mod;
    a = a * a % mod;
    b /= 2;
  }
  return ans;
}
int main() {
  int n, x, pos;
  std::cin >> n >> x >> pos;
  ll ans = 0;
  if (x == 1)
    ans = qpow(n - 1, n - 1 - pos);
  else {
    for (int i = pos; i < n - 1; ++i) {
      ans = (ans + qpow(n - 1, n - 1 - pos) *
                   (i - pos + 1) % mod * qpow(n - 1, i) % mod) %
            mod;
    }
  }
  std::cout << ans << '\n';
  return 0;
}
