#include <bits/stdc++.h>

using namespace std;

using ll = long long;

const int MAXN = 1005;
int n;
ll x[MAXN], y[MAXN];

struct pt {
    ll x, y;
};
struct vec {
    ll x, y;
    vec(ll _x, ll _y) : x(_x), y(_y) {}
};

ll cross(vec a, vec b) { return a.x * b.y - a.y * b.x; }
ll dot(vec a, vec b) { return a.x * b.x + a.y * b.y; }

int sameLine(pt a, pt b, pt c) {
    return cross({b.x - a.x, b.y - a.y}, {c.x - a.x, c.y - a.y}) == 0;
}

bool segxseg(pt a, pt b, pt c, pt d) {
    int d1, d2, d3, d4;
    d1 = cross({c.x - a.x, c.y - a.y}, {d.x - c.x, d.y - c.y});
    d2 = cross({c.x - a.x, c.y - a.y}, {b.x - a.x, b.y - a.y});
    d3 = cross({a.x - d.x, a.y - d.y}, {b.x - d.x, b.y - d.y});
    d4 = cross({a.x - d.x, a.y - d.y}, {c.x - d.x, c.y - d.y});
    if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0)))
        return true;
    if (d1 == 0 && sameLine(a, c, d)) return true;
    if (d2 == 0 && sameLine(a, b, c)) return true;
    if (d3 == 0 && sameLine(c, a, b)) return true;
    if (d4 == 0 && sameLine(c, d, b)) return true;
    return false;
}

int segseg(pt a, pt b, pt c, pt d) {
    if (min(a.x, b.x) > max(c.x, d.x) || min(c.x, d.x) > max(a.x, b.x) || min(a.y, b.y) > max(c.y, d.y) || min(c.y, d.y) > max(a.y, b.y))
        return 0;
    return segxseg(a, b, c, d);
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> x[i] >> y[i];
    int ans = 0;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++)
            for (int k = j + 1; k < n; k++)
                for (int l = k + 1; l < n; l++) {
                    ans += segseg({x[i], y[i]}, {x[j], y[j]}, {x[k], y[k]}, {x[l], y[l]});
                }
    cout << ans << '\n';
    return 0;
}
