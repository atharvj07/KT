def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return a * b // gcd(a, b)

def solve(n):
    a = 1
    b = n - 1
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            if lcm(i, n-i) < lcm(a, b):
                a, b = i, n-i
            if lcm(n//i, n-n//i) < lcm(a, b):
                a, b = n//i, n-n//i
    return a, b

t = int(input())
for _ in range(t):
    n = int(input())
    a, b = solve(n)
    print(a, b)
