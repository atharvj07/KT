import java.util.*;

class Main {
    static int n, k;
    static char[][] grid;

    static boolean valid(int x, int y) {
        return x >= 0 && y >= 0 && x < 3 && y < n && grid[x][y] != 's';
    }

    static boolean check(int x, int y) {
        if (!valid(x, y)) {
            return false;
        }

        for (int i = 0; i < 26; i++) {
            int cnt = 0;
            for (int j = y; j < n; j++) {
                if (grid[x][j] == 'a' + i) {
                    cnt++;
                } else {
                    break;
                }
            }

            if (cnt >= 2) {
                return false;
            }
        }

        return true;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int t = in.nextInt();

        while (t-- > 0) {
            n = in.nextInt();
            k = in.nextInt();
            grid = new char[3][n];

            for (int i = 0; i < 3; i++) {
                String s = in.next();
                for (int j = 0; j < n; j++) {
                    grid[i][j] = s.charAt(j);
                }
            }

            int x = 0, y = 0;
            boolean can = true;

            while (y < n) {
                if (check(x, y)) {
                    x++;
                    if (x == 3) {
                        x--;
                        y++;
                    }
                } else {
                    can = false;
                    break;
                }
            }

            System.out.println(can ? "YES" : "NO");
        }
    }
}
