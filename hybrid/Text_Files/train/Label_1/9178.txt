#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        stack<pair<int, int>> st;
        for (int i = 0; i < s.size(); i++) {
            int cnt0 = 0, cnt1 = 0;
            while (!st.empty() && st.top().first == i && st.top().second) {
                if (s[i] == '0')
                    cnt0++;
                else
                    cnt1++;
                i++;
            }
            if (cnt0 == cnt1)
                st.push({i, 0});
            else if (cnt0 > cnt1)
                st.push({i, 1});
        }
        string ans;
        while (!st.empty()) {
            auto [l, f] = st.top();
            st.pop();
            if (f == 0)
                ans += s.substr(l, s.size() - l);
            else {
                string s1 = s.substr(l, s.size() - l);
                reverse(s1.begin(), s1.end());
                for (int i = 0; i < s1.size(); i++)
                    s1[i] = s1[i] == '1' ? '0' : '1';
                ans += s1;
            }
        }
        cout << ans << '\n';
    }
}
