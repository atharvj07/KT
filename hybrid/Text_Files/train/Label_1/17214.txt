#include<bits/stdc++.h>
using namespace std;
#define MOD 1000000007
long long power(long long x, long long y) {
  if (y == 0) return 1;
  long long t = power(x, y / 2);
  t = (t * t) % MOD;
  if (y % 2 == 0) return t;
  return (x * t) % MOD;
}
int main() {
  int n, m;
  cin >> n >> m;
  cout << (power(m + 1, n) * 2 - 2 + MOD) % MOD << endl;
  return 0;
}
