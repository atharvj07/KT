import sys, heapq
input = sys.stdin.readline
n, m, k = map(int, input().split())
b = list(map(int, input().split()))
c = [m]
for i in b:
    c.append(i)
    c.append(i+1)
c.append(0)
c = sorted(list(set(c)))
ans = 0
for i in range(1, len(c)):
    if c[i] - c[i-1] > 1:
        ans += c[i] - c[i-1] - 1
        if k == 0: break
        else: k -= 1
print(ans)
