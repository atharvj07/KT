#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define int ll
const int maxn = 2001, mod = 1e9+7;
int  n, p, f[maxn], inv[maxn], g[maxn], ans;
bool good[maxn];
int mypow(int x, int y) {
    int res = 1;
    while (y) {
        if (y&1) res = res * x % mod;
        y >>= 1;
        x = x * x % mod;
    }
    return res;
} 
void pre() {
    inv[1] = 1;
    for (int i = 2; i < maxn; i ++) inv[i] = (-(mod/i) * inv[mod%i]%mod + mod)%mod;
}
bool isprime(int x) {
    if (x==1) return 0;
    for (int i = 2; i <= x/i; i ++) if (x%i==0) return 0;
    return 1;
}
bool check(int x) {
    return inv[x%p]>0;
}
signed main() {
    cin >> n >> p;
    pre();
    vector<pair<int, int>> a(n);
    for (int i = 0; i < n; i ++) cin >> a[i].first, a[i].second = i+1;
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i ++) g[a[i].second] = a[i].first;
    for (int i = 1; i <= n; i ++) {
        for (int j = 1; j <= i; j ++) f[i] = (f[i] + f[i-j]*g[j])%mod;
        if (check(f[i])) {
            ans ++;
            good[i] = 1;
        }
    }
    cout << ans << endl;
    for (int i = 1; i <= n; i ++) if (good[i]) cout << i << ' ';
    return 0;
}
