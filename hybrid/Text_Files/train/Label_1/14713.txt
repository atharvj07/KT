#include <bits/stdc++.h>
using namespace std;

int main() {
  int x;
  cin >> x;
  const int MAX_N = 30;
  vector<int> ops;
  int xxor = x;
  bool turn = true;
  while (xxor != 0 && ops.size() <= 40) {
    if (turn) {
      for (int i = MAX_N; i >= 0; i--) {
        if ((xxor >> i) & 1) {
          ops.push_back(i);
          xxor ^= 1 << i;
          break;
        }
      }
    } else {
      xxor++;
    }
    turn = !turn;
  }
  cout << ops.size() << endl;
  for (const int &op : ops) cout << op << " ";
  cout << endl;
  return 0;
}
