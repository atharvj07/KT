import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] oldPath = new int[m + 1];
        for (int i = 0; i < m + 1; i++) {
            oldPath[i] = sc.nextInt();
        }
        int[] newPath = new int[m + 1];
        newPath[0] = newPath[m] = oldPath[0];
        Set<Integer> visited = new HashSet<>();
        visited.add(oldPath[0]);
        int idx = 1;
        for (int i = 1; i < m + 1; i++) {
            if (visited.contains(oldPath[i])) {
                continue;
            }
            visited.add(oldPath[i]);
            newPath[idx++] = oldPath[i];
        }
        if (idx != m + 1) {
            System.out.println("No solution");
            return;
        }
        for (int i = 0; i < m + 1; i++) {
            System.out.print(newPath[i] + " ");
        }
        System.out.println();
    }
}
