#include <bits/stdc++.h>
using namespace std;
#define fi first
#define se second
#define ll long long
const int mod = 1e9 + 7;
typedef pair<int, int> pi;
typedef pair<ll, ll> pl;
#define vi(x) vector<x>
#define pb push_back
const ll INF = 1e18 + 1;

int main(){
	int n, a, b;
	cin >> n >> a >> b;
	int arr[a][b];
	if(n%2){
		cout << -1 << '\n';
		return 0;
	}
	n /= 2;
	if(n > a*b){
		cout << -1 << '\n';
		return 0;
	}

	int cnt = 1;
	for(int i = 0; i < a; i+=2){
		for(int j = 0; j < b; j+=2){
			arr[i][j] = cnt++;
			arr[i+1][j] = cnt++;
		}
	}

	for(int i = 0; i < a; i++){
		for(int j = 0; j < b; j++){
			cout << arr[i][j] << ' ';
		}
		cout << '\n';
	}
}
