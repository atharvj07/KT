n = int(input().strip())
a = list(map(int, input().strip().split()))

def median_smooth(a):
    b = a[:]
    for i in range(1, len(a) - 1):
        b[i] = sorted([a[i-1], a[i], a[i+1]])[1]
    return b

def is_stable(a, b):
    return a == b

changes = 0
while True:
    b = median_smooth(a)
    if is_stable(a, b):
        break
    a = b
    changes += 1

print(changes)
print(*a)
