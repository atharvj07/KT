#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (auto &l : a) cin >> l;
  sort(begin(a), end(a));
  int rest = k;
  int res = 0;
  for (const auto &l : a) {
    if (rest - l >= 0) {
      res++;
      rest -= l;
    } else {
      res += rest / l + (rest % l != 0);
      rest = 0;
    }
  }
  cout << res << '\n';
  return 0;
}
