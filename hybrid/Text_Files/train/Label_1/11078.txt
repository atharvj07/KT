#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;

int main() {
  int h, w;
  cin >> h >> w;
  vector<string> grid(h);
  for (auto &row : grid) cin >> row;
  int q;
  cin >> q;
  vector<int> ans(q);
  for (int i = 0; i < q; i++) {
    int r1, c1, r2, c2;
    cin >> r1 >> c1 >> r2 >> c2;
    r1--, c1--, r2--, c2--;
    int R = r2 - r1 + 1, C = c2 - c1 + 1;
    int rowCount[500][500]{}, colCount[500][500]{};
    for (int r = r1; r <= r2; r++) {
      for (int c = c1; c <= c2; c++) {
        rowCount[r][c] = rowCount[r][c - 1] + rowCount[r - 1][c] - rowCount[r - 1][c - 1];
        colCount[r][c] = colCount[r - 1][c] + colCount[r][c - 1] - colCount[r - 1][c - 1];
        if (grid[r][c] == '.') { rowCount[r][c]++, colCount[r][c]++; }
      }
    }
    ans[i] = 0;
    for (int r = r1; r <= r2; r++) {
      for (int c = c1; c <= c2; c++) {
        if (grid[r][c] == '.') {
          int emptyRowCount = rowCount[r][c2] - rowCount[r][c - 1];
          ans[i] = (ans[i] + 1ll * emptyRowCount * (rowCount[r2][c] - rowCount[r2][c - 1])) % MOD;
          int emptyColCount = colCount[r2][c] - colCount[r - 1][c];
          ans[i] = (ans[i] + 1ll * emptyColCount * (colCount[r][c2] - colCount[r][c - 1])) % MOD;
        }
      }
    }
  }
  for (auto &x : ans) cout << x << endl;
}
