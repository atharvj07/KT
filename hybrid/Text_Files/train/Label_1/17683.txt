#include<bits/stdc++.h>

using namespace std;

const int N = 501;

char c[N][N];
int n, m;

bool valid(int x, int y){
    return (x >= 1 && x <= n && y >= 1 && y <= m);
}

bool check(){
    int sx, sy;
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            if(c[i][j] == '*'){
                sx = i, sy = j;
                break;
            }
        }
    }

    int cnt = 0;
    for(int i = 1; i <= m; ++i){
        if(c[sx][i] != '*') break;
        ++cnt;
    }

    if(cnt & 1) return false;

    cnt = 0;
    for(int i = sx; i >= 1; --i){
        if(c[i][sy] != '*') break;
        ++cnt;
    }

    if(cnt & 1) return false;

    cnt = 0;
    for(int i = 1; i <= n; ++i){
        if(c[i][sy] != '*') break;
        ++cnt;
    }

    if(cnt & 1) return false;

    cnt = 0;
    for(int i = sy; i >= 1; --i){
        if(c[sx][i] != '*') break;
        ++cnt;
    }

    if(cnt & 1) return false;

    for(int i = sx + 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            if(c[i][j] != '.') return false;
        }
    }

    for(int i = 1; i < sx; ++i){
        for(int j = 1; j <= m; ++j){
            if(c[i][j] != '.') return false;
        }
    }

    for(int i = sy + 1; i <= m; ++i){
        for(int j = 1; j <= n; ++j){
            if(c[j][i] != '.') return false;
        }
    }

    for(int i = 1; i < sy; ++i){
        for(int j = 1; j <= n; ++j){
            if(c[j][i] != '.') return false;
        }
    }

    return true;
}

int main(){
    cin >> n >> m;
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            cin >> c[i][j];
        }
    }

    if(check()) cout << "YES" << endl;
    else cout << "NO" << endl;

    return 0;
}
