#include <iostream>
#include <set>

int main() {
  int n, m, k;
  std::cin >> n >> m >> k;
  std::set<int> alice;
  for (int i = 0; i < n; i++) {
    int t;
    std::cin >> t;
    alice.insert(t);
  }
  std::set<int> bob;
  for (int i = 0; i < m; i++) {
    int t;
    std::cin >> t;
    bob.insert(t);
  }
  std::cout << (alice.size() == bob.size() ? "NO" : "YES") << std::endl;
  return 0;
}
