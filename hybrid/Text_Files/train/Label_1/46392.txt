import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++)
                a[j] = scanner.nextInt();
            int[] b = new int[m];
            for (int j = 0; j < m; j++)
                b[j] = scanner.nextInt();
            int count = 0;
            for (int j = 0, k = 0; j < n && k < m; j++) {
                if (a[j] == b[k]) {
                    k++;
                    count++;
                }
            }
            System.out.println(count > 0 ? "YES\n" + count + "\n" + a[count - 1] : "NO");
        }
    }
}
