import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] dices = new int[n][6];
        for (int i = 0; i < n; i++) for (int j = 0; j < 6; j++) dices[i][j] = sc.nextInt();
        boolean[] flag = new boolean[1000];
        for (int[] dice : dices) {
            int a = dice[0] * 10 + dice[1];
            int b = dice[2] * 10 + dice[3];
            int c = dice[4] * 10 + dice[5];
            if (flag[a] || flag[b] || flag[c]) {
                System.out.println("No");
                return;
            }
            flag[a] = flag[b] = flag[c] = true;
        }
        System.out.println("Yes");
    }
}
