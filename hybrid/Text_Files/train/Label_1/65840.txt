def is_leap_year(year):
    return year % 400 == 0 or (year % 4 == 0 and year % 100 != 0)

def find_next_same_calendar_year(y):
    initial_is_leap = is_leap_year(y)
    day_offset = 0
    while True:
        y += 1
        if is_leap_year(y):
            day_offset += 2
        else:
            day_offset += 1
        if day_offset % 7 == 0 and is_leap_year(y) == initial_is_leap:
            break
    return y

y = int(input())
print(find_next_same_calendar_year(y))
