import sys

n, m = map(int, input().split())
dislike = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, input().split())
    dislike[a].append(b)
    dislike[b].append(a)

q = int(input())
salary = [i for i in range(1, n+1)]
res = []
for i in range(q+1):
    ans = 0
    for j in range(n):
        for k in dislike[salary[j]]:
            if k > salary[j] and k in dislike[salary[k]]:
                ans += 1
    res.append(ans)
    if i < q:
        x = int(input())
        for j in range(n):
            if salary[j] == x:
                salary[j] = n + i + 1
                for k in dislike[x]:
                    if salary[j] > salary[k]:
                        ans -= 1
                break
    print(ans)
