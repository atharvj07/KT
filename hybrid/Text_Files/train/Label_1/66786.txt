n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
for p in a:
    for i in range(0, m - 1):
        for j in range(i + 1, m):
            if p[i] > p[j]:
                print('NO')
                exit()
for j in range(0, m - 1):
    for i in range(0, n):
        for l in range(i + 1, n):
            if a[i][j] > a[l][j]:
                print('NO')
                exit()
print('YES')
