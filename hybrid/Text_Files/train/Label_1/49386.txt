import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int even = 0, odd = 0;
        for (int i = 0; i < n; i++) {
            int j = i;
            while (j + 1 < n && s.charAt(j) == s.charAt(j + 1)) j++;
            int len = j - i + 1;
            if (len % 2 == 0) even++;
            else odd++;
            i = j + 1;
        }
        System.out.println(even + " " + odd);
    }
}
