#include <bits/stdc++.h>
using namespace std;

#define endl '\n'

typedef long long ll;

const ll mod = 1e9 + 7;

ll modpow(ll base, ll exp) {
  ll ans = 1;
  while (exp > 0) {
    if (exp % 2 == 1) ans = ans * base % mod;
    base = base * base % mod;
    exp /= 2;
  }
  return ans;
}

int main() {
  cin.tie(0);
  ios::sync_with_stdio(0);

  int t;
  cin >> t;
  while (t--) {
    ll x;
    string s;
    cin >> x >> s;
    int n = s.length();
    ll ans = (n + (n - 1) * modpow(3, x - 1) % mod) % mod;
    cout << ans << endl;
  }
  return 0;
}
