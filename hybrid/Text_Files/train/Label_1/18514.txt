#include<bits/stdc++.h>
using namespace std;
const int maxn=500+5;
int n,a[maxn],ans,sum;
vector<int> v;
int main(){
    cin>>n;
    for(int i=1;i<=n;++i) cin>>a[i],sum+=a[i];
    sort(a+1,a+n+1);
    ans=a[n];
    int j=n-1;
    for(int i=1;i<n;++i){
        v.push_back(a[i]);
        while(j&&v.back()>a[j]) --j,v.push_back(a[j]);
        ans+=max(1,(int)v.size());
        v.clear();
    }
    cout<<ans<<endl;
    return 0;
}
