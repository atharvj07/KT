import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            long n = sc.nextLong();
            long k = sc.nextLong();
            
            long catA = (k - 1) % n + 1;
            long catB = 1;

            if (catA == catB) catB = catB + 2;
            if (catB > n) catB = catB - n;

            System.out.println(catB);
        }
    }
}
