#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
const ll mod = 1e9 + 7;
#define endl "\n"

void solve() {
  ll n, k;
  cin >> n >> k;
  vector<pair<ll, ll>> v(n);
  for (ll i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  priority_queue<ll, vector<ll>, greater<ll>> pq;
  for (ll i = 0; i < k; i++) pq.push(0);
  ll ans[n];
  for (ll i = 0, j = 0; i < n; i++) {
    ll x = v[i].first, y = v[i].second;
    while (!pq.empty() && pq.top() <= x) pq.pop();
    ll cur = pq.empty() ? x : pq.top();
    pq.push(cur + y);
    ans[i] = cur + y;
  }
  for (auto x : ans) cout << x << endl;
}
int main() {
  solve();
  return 0;
}
