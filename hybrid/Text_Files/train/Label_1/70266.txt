import sys,math

def lcm(a,b):
    return a*b//math.gcd(a,b)

for _ in range(int(input())):
    n,k=map(int,input().split())
    a=[]
    for i in range(k-2):
        a.append(1)
    n-=k-2
    assert n%2==0, "Impossible"
    n//=2
    a.append(n-(n//(k-1)))
    for i in range(k-1):
        a[i]+=n//(k-1)
    print(*a)
