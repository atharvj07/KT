def rearrange_students(n, m):
    if n == 1 and m == 1:
        return ["YES", "1"]
    if n == 1 and m < 3 or m == 1 and n < 3:
        return ["NO"]
    if n == 2 and m == 2:
        return ["NO"]

    matrix = [[0 for _ in range(m)] for _ in range(n)]
    numbers = list(range(1, n*m + 1))
    if n <= m:
        for i in range(n):
            for j in range(m):
                matrix[i][j] = numbers[(i+j*n) % (n*m)]
    else:
        for i in range(n):
            for j in range(m):
                matrix[i][j] = numbers[(i*m+j) % (n*m)]

    return ["YES"] + [" ".join(map(str, row)) for row in matrix]

n, m = map(int, input().split())
result = rearrange_students(n, m)
for line in result:
    print(line)
