from collections import deque
n=int(input())
p=[int(x) for x in input().split()]
a=[list(input()) for x in range(n)]
vis=[0]*n
q=deque([1])
ans=[]
while q:
    i=q.popleft()
    ans.append(i)
    vis[i-1]=1
    for j in range(n):
        if a[i-1][j]=='1' and vis[j]==0:
            q.append(j+1)
print(*ans)
