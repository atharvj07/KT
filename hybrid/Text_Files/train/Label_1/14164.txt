#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int a[n];
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int ans = 0;
        int s1 = 0, s2 = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1)
                ans++;
            else if (a[i] == 2)
                ans--;
            else {
                if (s1 >= s2) {
                    s1 += 1;
                    ans++;
                } else {
                    s2 += 1;
                    ans--;
                }
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
