#include <iostream>
#include <set>

using namespace std;

int main() {
    int n;
    cin >> n;
    set<int> a, b, p;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a.insert(x);
    }
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        b.insert(x);
    }
    for (int i = 1; i <= n; i++) {
        if (a.find(i) == a.end()) {
            p.insert(i);
        }
    }
    for (int i = 1; i <= n; i++) {
        if (b.find(i) == b.end()) {
            p.insert(i);
        }
    }
    for (int x : p) {
        cout << x << " ";
    }
    cout << endl;
    return 0;
}
