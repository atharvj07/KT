#include <iostream>
#include <vector>

using namespace std;
vector<string> a;
int main() {
  int n, p;
  cin >> n >> p;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    a.push_back(s);
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i][0] == 'h')
      ans += p / 2;
    else
      ans += p;
  }
  cout << ans << endl;
  return 0;
}
