#include <algorithm>
#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    map<int, vector<pair<int, int>>> l, r;
    while (n--) {
      int a, b;
      cin >> a >> b;
      l[a].push_back(make_pair(0, b));
      r[b].push_back(make_pair(a, 0));
    }
    vector<int> ll, rr;
    for (auto x : l) {
      ll.push_back(x.first);
    }
    for (auto x : r) {
      rr.push_back(x.first);
    }
    sort(ll.begin(), ll.end());
    sort(rr.begin(), rr.end());
    int ans = 1e9;
    for (int i = 0; i < ll.size(); i++) {
      auto it = upper_bound(rr.begin(), rr.end(), ll[i]);
      if (it != rr.end()) { ans = min(ans, *it - ll[i]); }
    }
    cout << ans << endl;
  }
  return 0;
}
