#include <iostream>
#include <vector>
using namespace std;
int main() {
    int potatoes, maxSize, stepsPerPotato, totalSteps = 0;
    cin >> potatoes >> maxSize >> stepsPerPotato;
    vector<int> heights(potatoes);
    for (int i = 0; i < potatoes; i++) {
        cin >> heights[i];
    }
    for (int height : heights) {
        totalSteps += (height + stepsPerPotato - 1) / stepsPerPotato;
    }
    cout << totalSteps << endl;
}
