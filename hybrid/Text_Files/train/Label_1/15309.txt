#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
vector<pair<int, int>> g[maxn];
ll dis[maxn];
int p[maxn], sz[maxn];

int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
void join(int x, int y) {
    x = find(x), y = find(y);
    if (x != y) {
        if (sz[x] < sz[y]) swap(x, y);
        p[y] = x;
        sz[x] += sz[y];
    }
}
bool same(int a, int b) { return find(a) == find(b); }

int main() {
    int n, m, x, y, w, ans = 0;
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> x >> y >> w;
        g[x].push_back({y, w});
        g[y].push_back({x, w});
    }
    priority_queue<pair<ll, int>> pq;
    for (int i = 1; i <= n; i++) {
        dis[i] = 1e16;
        p[i] = i, sz[i] = 1;
    }
    pq.push({0, 1});
    dis[1] = 0;
    while (!pq.empty()) {
        int u = pq.top().second;
        pq.pop();
        for (pair<int, int> v : g[u]) {
            if (dis[u] + v.second < dis[v.first]) {
                join(u, v.first);
                dis[v.first] = dis[u] + v.second;
                pq.push({-dis[v.first], v.first});
            }
        }
    }
    for (int i = 1; i <= m; i++) {
        cin >> x >> y >> w;
        if (same(x, y)) ans++;
    }
    cout << ans << endl;
    return 0;
}
