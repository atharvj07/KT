import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 6; j++) {
        a[i] |= 1 << scanner.nextInt();
      }
    }
    int ans = -1;
    int[] cnt = new int[10];
    for (int i = 0; i < n; i++) {
      cnt[0] = 1;
      for (int j = 1; j < 10; j++) {
        cnt[j] = 10 * cnt[j % 10] + cnt[(j + 1) % 10];
      }
      for (int j = 1; j < 10; j++) {
        if ((a[i] >> j & 1) == 0) {
          if (cnt[j] > ans || (cnt[j] == ans && j > ans % 10)) {
            ans = cnt[j];
          }
        }
      }
    }
    System.out.println(ans);
  }
}
