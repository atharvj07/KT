#include <iostream>
#include <vector>
using namespace std;

int main() {
  int x;
  cin >> x;
  vector<int> v;
  for (int i = 30; i >= 0; i--) {
    if ((x ^ (1 << i)) < x) {
      x ^= 1 << i;
      v.push_back(i);
    }
  }
  x++;
  int cnt = 0;
  for (int i = v.size() - 1; i >= 0 && cnt < 40; i--) {
    x ^= 1 << v[i];
    v.pop_back();
    cnt++;
  }
  while (cnt < 40) {
    x++;
    cnt++;
  }
  int sz = v.size();
  cout << sz << endl;
  for (int i = 0; i < sz; i++) { cout << v[i] << " "; }
  cout << endl;
}
