import sys

input = sys.stdin.readlines()
n, m = map(int, input[0].split())
colors = list(map(int, input[1].split()))
edges = [map(int, line.split()) for line in input[2:]]

adj = [[] for _ in range(n + 1)]
for a, b in edges:
    adj[a].append(b)
    adj[b].append(a)

color_counts = [0] * (max(colors) + 1)
for color in colors:
    color_counts[color] += 1

neighbor_counts = [0] * (max(colors) + 1)

for i in range(1, n + 1):
    color = colors[i - 1]
    for neighbor in adj[i]:
        if colors[neighbor - 1] != color:
            neighbor_counts[color] += 1

max_count = -1
max_color = -1
for color in range(1, max(colors) + 1):
    if color_counts[color] > 0 and neighbor_counts[color] > max_count:
        max_count = neighbor_counts[color]
        max_color = color

print(max_color)
