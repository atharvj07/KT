for _ in range(int(input())):
  n = int(input())
  s = input()
  if s.count('a') == s.count('b'):
    print(1, n)
  else:
    ans = 0
    diff = 0
    l = 0
    for r in range(n):
      if s[r] == 'a':
        diff += 1
      else:
        diff -= 1
      if diff == 0:
        l = r + 1
        if ans < r - l + 1:
          ans = r - l + 1
      elif -diff == s[l:].count('a'):
        ans = max(ans, r - l + 1)
        l += 1
    if ans: print(l, l + ans - 1)
    else: print(-1, -1)
