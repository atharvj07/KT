import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        sc.close();

        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }

        if (sum % 2 == 0) {
            System.out.println("Second");
        } else {
            System.out.println("First");
        }
    }
}
