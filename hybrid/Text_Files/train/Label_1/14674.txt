#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 3e5 + 5;
int par[N];
ll sz[N];

ll find(int a){
    return (a == par[a]) ? a : par[a] = find(par[a]);
}
ll merge(int a, int b){
    a = find(a), b = find(b);
    if(a != b) par[b] = a, sz[a]+=sz[b];
    return sz[a];
}
ll GCD(ll a, ll b){
    if(b == 0) return a;
    return GCD(b, a%b);
}
int main(){
    int n; cin >> n;
    for(int i = 1; i <= n; ++i) par[i] = i, sz[i] = 1;
    vector<pair<ll, pair<int, char>>> p(n + 1);
    for(int i = 1; i <= n; ++i) cin >> p[i].first >> p[i].second.first >> p[i].second.second;
    p[0].first = 0; p[0].second.first = 1; p[0].second.second = 'G';
    sort(p.begin(), p.end());
    ll ans = 0, Red = 0, Blue = 0, Green = 0;
    for(int i = 1; i <= n; ++i){
        if(p[i].second.second == 'R') Red = p[i].first;
        else if(p[i].second.second == 'B') Blue = p[i].first;
        else{
            ll _Red = find(Red), _Blue = find(Blue);
            if(_Red != 1 && _Blue != 1){
                ans += p[i].first - (p[_Red].first + p[_Blue].first) / 2;
                merge(_Red, _Blue);
            }
            else if(_Red == 1){
                Green = p[i].first;
                ans += p[i].first - p[1].first;
            }
            else{
                Red = p[i].first;
                ans += p[i].first - p[2].first;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
