import java.util.Scanner;

public class SmallestIntegerWithNDivisors {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int smallestInteger = 1;
        int divisors = 1;

        while (divisors < n) {
            smallestInteger++;
            divisors = countDivisors(smallestInteger);
        }

        System.out.println(smallestInteger);
    }

    private static int countDivisors(int number) {
        int divisors = 0;
        for (int i = 1; i <= Math.sqrt(number); i++) {
            if (number % i == 0) {
                divisors += 2; // Count both i and number/i as divisors
            }
        }
        return divisors;
    }
}
