#include <iostream>
#include <vector>
using namespace std;

int main() {
  int a, b, k;
  cin >> a >> b >> k;
  vector<int> isPrime(b+1, 1);
  for (int i = 2; i * i <= b; i++) {
    if (isPrime[i]) {
      for (int j = i * i; j <= b; j += i) isPrime[j] = 0;
    }
  }
  int l = -1;
  for (int i = a; i <= b - k + 1; i++) {
    int cnt = 0;
    for (int j = i; j < i + k; j++) {
      cnt += isPrime[j];
    }
    if (cnt >= k) {
      l = i + k - 1 - a;
      break;
    }
  }
  cout << l << '\n';
  return 0;
}
