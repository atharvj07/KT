#include <bits/stdc++.h>
#define ll long long
#define MOD 1000000007
#define MAX 1000000
using namespace std;

int prime[MAX+5];
vector<int>primes;

void Sieve() {
    for (int i = 2; i <= MAX; i++) {
        if (!prime[i]) {
            primes.push_back(i);
            for (int j = i + i; j <= MAX; j += i)
                prime[j] = 1;
        }
    }
}

int g(ll x, ll p) {
    int ans = 0;
    while (x % p == 0) {
        x /= p;
        ans++;
    }
    return ans;
}

int f(ll x, ll y) {
    int ans = 1;
    for (ll p : primes) {
        if (p * p > y) break;
        if (x % p == 0) ans *= (int)pow(p, g(y, p));
    }
    if (x > 1) ans *= (int)pow(x, g(y, x));
    return ans;
}

int main() {
    Sieve();
    ll x, n;
    cin >> x >> n;
    ll ans = 1;
    for (ll i = 1; i <= n; i++) ans = (ans * f(x, i)) % MOD;
    cout << ans << endl;
    return 0;
}
