def assign_coordinates(n, edges):
    # Converting edge list to adjacency list
    graph = {i: [] for i in range(1, n+1)}
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    coordinates = {}

    def dfs(node, parent, direction, step=1):
        if node in coordinates:
            return

        if parent is None:
            coordinates[node] = (0, 0)
        else:
            x, y = coordinates[parent]
            if direction == 'horizontal':
                coordinates[node] = (x + step, y)
            else:  # direction == 'vertical'
                coordinates[node] = (x, y + step)

        if len(graph[node]) == 1 and parent is not None:
            return

        next_direction = 'vertical' if direction == 'horizontal' else 'horizontal'
        child_step = 1
        for child in graph[node]:
            if child != parent:
                dfs(child, node, next_direction, child_step)
                child_step += 1

    # Start DFS from the first node with horizontal direction
    dfs(1, None, 'horizontal')

    if len(coordinates) == n:
        return ["YES"] + [" ".join(map(str, coordinates[i])) for i in range(1, n+1)]
    else:
        return ["NO"]

# Input processing
n = int(input().strip())
edges = [list(map(int, input().strip().split())) for _ in range(n-1)]

# Solve and output
result = assign_coordinates(n, edges)
for line in result:
    print(line)
