#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int INF = 1e9;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<vector<vector<int>>> dp(n + 1, vector<vector<int>>(k + 1, vector<int>(k + 1, INF)));
    dp[0][0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            for (int l = 0; l <= k; l++) {
                dp[i][j][l] = dp[i - 1][j][l];
                if (j < k) {
                    dp[i][j + 1][l] = min(dp[i][j + 1][l], dp[i - 1][j][l] + n - i + l);
                }
                if (l < k && s[i - 1] == s[l]) {
                    dp[i][j][l + 1] = min(dp[i][j][l + 1], dp[i - 1][j][l]);
                }
            }
        }
    }
    if (dp[n][k][k] == INF) {
        cout << -1 << endl;
    } else {
        cout << dp[n][k][k] << endl;
    }
    return 0;
}
