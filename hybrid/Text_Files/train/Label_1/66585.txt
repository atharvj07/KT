from collections import deque

n, m, k = map(int, input().split())
grid = [list(input()) for _ in range(n)]
s = sum(c == '.' for row in grid for c in row)
if k == s - 1:
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                grid[i][j] = 'X'
                break
else:
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                grid[i][j] = 'X'
                q = deque([(i, j)])
                seen = {(i, j)}
                while q:
                    x, y = q.popleft()
                    for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < n and 0 <= ny < m and (nx, ny) not in seen:
                            if grid[nx][ny] != '#':
                                seen.add((nx, ny))
                                q.append((nx, ny))
                if len(seen) == s - k:
                    break
print('\n'.join(''.join(row) for row in grid))
