#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll p[100005], d[100005];
int main() {
  ll n, k;
  cin >> n >> k;
  ll a[n + 1];
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  p[n - 1] = a[n - 1];
  d[0] = a[0];
  for (ll i = n - 2; i >= 0; i--) { p[i] = p[i + 1] * a[i]; }
  for (ll i = 1; i < n; i++) { d[i] = d[i - 1] * a[i]; }
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    for (ll j = i; j < n; j++) {
      ll mul = 1;
      if (i > 0) mul *= d[i - 1];
      if (j + 1 < n) mul *= p[j + 1];
      if (mul % k == 0) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
