import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[][] q = new int[m][2];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 2; j++) {
                q[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            dp[1][i] = a[i];
        }
        for (int i = 2; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                dp[i][j] = Math.min(dp[i - 1][j], dp[i - 1][j - 1]) + a[j];
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.println(dp[q[i][0]][q[i][1]]);
        }
    }
}
