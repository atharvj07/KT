#include <bits/stdc++.h>

using namespace std;

#define MAX_N 500000
#define MAX_K 500000

int main() {
  long long m, k, n, s;
  cin >> m >> k >> n >> s;
  unordered_map<long long, long long> freq;
  vector<long long> a(m), b(s), pos;
  for (long long i = 0; i < m; i++) {
    cin >> a[i];
    freq[a[i]]++;
  }
  for (long long i = 0; i < s; i++) {
    cin >> b[i];
  }
  sort(a.begin(), a.end());
  for (long long i = 0; i < s; i++) {
    freq[b[i]]--;
    if (freq[b[i]] < 0) {
      cout << -1 << endl;
      return 0;
    }
  }
  long long d = 0;
  for (long long i = 0; i < m && d < n * k - s; i++) {
    if (freq[a[i]] == 0) {
      d++;
      pos.push_back(i + 1);
    }
  }
  cout << d << endl;
  for (auto p : pos) {
    cout << p << " ";
  }
  cout << endl;
  return 0;
}
