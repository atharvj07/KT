import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while(t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for(int i = 0; i < n; i++) a[i] = in.nextInt();
            boolean flag = true;
            for(int i = 0; i < n && flag; i++) {
                int newRoom = (i + a[i % n] + n) % n;
                if(newRoom != i) flag = false;
            }
            System.out.println(flag ? "YES" : "NO");
        }
        in.close();
    }
}
