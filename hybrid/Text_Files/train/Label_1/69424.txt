for _ in range(int(input())):
    n,m=map(int,input().split())
    a=[[int(x) for x in input().split()]for i in range(n)]
    flag=False
    for i in range(n):
        for j in range(m):
            if a[i][j]!=0:
                if i==0 or j==0 or i==n-1 or j==m-1:
                    flag=True
    if flag:
        print('Ashish')
    else:
        print('Jeel')
