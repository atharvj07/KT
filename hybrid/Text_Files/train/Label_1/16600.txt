#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
int main() {
  long long int n, a;
  cin >> n >> a;
  long long int arr[n], b[n];
  for (int i = 0; i < n; i++) { cin >> arr[i]; }
  for (int i = 0; i < n; i++) {
    b[i] = max(0LL, a - n + i + 1 - arr[i]);
  }
  for (int i = 0; i < n; i++) { cout << b[i] << " "; }
  cout << endl;
  return 0;
}
