import queue

n, m = map(int, input().split())
graph = [set() for _ in range(n + 1)]

for _ in range(m):
    u, v = map(int, input().split())
    graph[u].add(v)
    graph[v].add(u)

vis = [0] * (n + 1)

steps = 0
q = queue.LifoQueue()
q.put(1)
vis[1] = 1

while not q.empty():
    u = q.get()
    steps += 1
    for v in graph[u]:
        if vis[v] == 0:
            q.put(v)
            vis[v] = 1

print(steps - 1)
for u in range(1, n + 1):
    if vis[u] == 0:
        print(u, end=' ')
