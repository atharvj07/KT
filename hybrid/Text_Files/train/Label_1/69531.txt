n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))

max_score_diff = 2 * n - 2 * m
max_a_score = 2 * n

for d in set(a + b):
    a_score = 3 * len([x for x in a if x > d]) + 2 * len([x for x in a if x <= d])
    b_score = 3 * len([x for x in b if x > d]) + 2 * len([x for x in b if x <= d])
    score_diff = a_score - b_score
    if score_diff > max_score_diff or (score_diff == max_score_diff and a_score > max_a_score):
        max_score_diff = score_diff
        max_a_score = a_score

print(f"{max_a_score}:{max_a_score - max_score_diff}")
