import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        int[] a = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        SegmentTree stree = new SegmentTree(n);
        while (q-- > 0) {
            st = new StringTokenizer(br.readLine());
            int op = Integer.parseInt(st.nextToken());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            if (op == 1) {
                int d = Integer.parseInt(st.nextToken());
                stree.updateRange(l, r, d);
            } else {
                if (stree.query(l, r, m) > 0) {
                    bw.write("1\n");
                } else {
                    bw.write("2\n");
                }
            }
        }
        bw.flush();
        bw.close();
    }

    public static class SegmentTree {
        int n;
        long[] arr;
        long[] tree;

        public SegmentTree(int n) {
            this.n = n;
            arr = new long[n + 1];
            tree = new long[4 * n];
        }

        public void buildTree(int node, int start, int end) {
            if (start == end) {
                tree[node] = arr[start];
            } else {
                int mid = (start + end) / 2;
                buildTree(2 * node, start, mid);
                buildTree(2 * node + 1, mid + 1, end);
                tree[node] = tree[2 * node] + tree[2 * node + 1];
            }
        }

        public void update(int node, int start, int end, int idx, long val) {
            if (start == end) {
                arr[idx] = val;
                tree[node] = val;
            } else {
                int mid = (start + end) / 2;
                if (idx <= mid) {
                    update(2 * node, start, mid, idx, val);
                } else {
                    update(2 * node + 1, mid + 1, end, idx, val);
                }
                tree[node] = tree[2 * node] + tree[2 * node + 1];
            }
        }

        public void updateRange(int start, int end, long val) {
            updateRange(1, 1, n, start, end, val);
        }

        public void updateRange(int node, int start, int end, int l, int r, long val) {
            if (start > end || l > end || r < start) {
                return;
            }
            if (l <= start && end <= r) {
                tree[node] += (end - start + 1) * val;
                return;
            }
            int mid = (start + end) / 2;
            updateRange(2 * node, start, mid, l, r, val);
            updateRange(2 * node + 1, mid + 1, end, l, r, val);
            tree[node] = tree[2 * node] + tree[2 * node + 1];
        }

        public long query(int left, int right, int m) {
            return query(1, 1, n, left, right, m);
        }

        public long query(int node, int start, int end, int left, int right, int m) {
            if (start > end || left > end || right < start) {
                return 0;
            }
            if (left <= start && end <= right) {
                return tree[node] - Math.min(tree[node], m);
            }
            int mid = (start + end) / 2;
            long leftSum = query(2 * node, start, mid, left, right, m);
            long rightSum = query(2 * node + 1, mid + 1, end, left, right, m);
            return leftSum + rightSum;
        }
    }
}
