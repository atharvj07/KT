#include <bits/stdc++.h>
#define ll long long

using namespace std;

ll n, x[300005], y[300005];
struct point {
    ll x, y;
    point(ll x, ll y) : x(x), y(y){};
};
ll cross(const point &a, const point &b, const point &c) {
    return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
}
int main() {
    cin >> n;
    vector<point> poly;
    for (int i = 0; i < n; i++) {
        cin >> x[i] >> y[i];
        poly.push_back({x[i], y[i]});
    }
    vector<ll> ans(n + 3);
    for (int k = 3; k <= n; k++) {
        ll ans1 = 0;
        for (int i = 0; i < k; i++) {
            ans1 += abs(poly[i].x - poly[(i + 1) % k].x) +
                    abs(poly[i].y - poly[(i + 1) % k].y);
        }
        ll ans2 = 0;
        for (int i = 0; i < k; i++) {
            ans2 += abs(poly[i].x - poly[(i - 1 + k) % k].x) +
                    abs(poly[i].y - poly[(i - 1 + k) % k].y);
        }
        ans[k] = max(ans1, ans2);
    }
    for (int i = 3; i <= n; i++) cout << ans[i] << ' ';
    return 0;
}
