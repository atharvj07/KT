#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> v(n + 1, 0);
  for (int i = 0; i < m; i++) {
    char c;
    int x;
    std::cin >> c >> x;
    if (c == '+') {
      v[x] = 1;
    } else {
      v[x] = -1;
    }
    bool flag = false;
    for (int j = 1; j <= n; j++) {
      if (v[j] == 1) {
        flag = true;
      }
    }
    if (flag) {
      std::cout << "1\n" << x << "\n";
      return 0;
    }
  }
  int cnt = 0;
  std::vector<int> ans;
  for (int i = 1; i <= n; i++) {
    if (v[i] == 0) {
      cnt++;
      ans.push_back(i);
    }
  }
  std::cout << cnt << "\n";
  for (int i = 0; i < cnt; i++) {
    std::cout << ans[i] << " ";
  }
  std::cout << "\n";
  return 0;
}
