n = int(input().strip())
lengths = list(map(int, input().strip().split()))
terrains = input().strip()

time = 0
stamina = 0

for i in range(n):
    if terrains[i] == 'G':
        time += lengths[i] * 5
        stamina += lengths[i]
    elif terrains[i] == 'W':
        time += lengths[i] * 3
        stamina += lengths[i]
    else:  # Lava
        if stamina >= lengths[i]:
            time += lengths[i]
            stamina -= lengths[i]
        else:
            needed = lengths[i] - stamina
            time += needed + stamina * 2
            stamina = 0  # All stamina is used to cross the lava

print(time)
