import java.util.Scanner;

public class CoinChange {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }

        long totalSum = 0;
        for (long price : a) {
            totalSum += price;
        }

        int minCoins = 0;
        long currentSum = 0;
        for (int i = 0; i < n; i++) {
            if (currentSum + a[i] <= totalSum) {
                currentSum += a[i];
            } else {
                minCoins++;
                currentSum = a[i];
            }
        }

        System.out.println(minCoins + 1); // Add one for the last coin
    }
}
