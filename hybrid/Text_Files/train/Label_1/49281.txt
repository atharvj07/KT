import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }
            int[] freq = new int[101];
            for (int i = 0; i < n; i++) {
                freq[a[i]]++;
            }
            int mex1 = 0, mex2 = 0;
            for (int i = 0; i < 101; i++) {
                if (freq[i] == 0) {
                    mex1 = i;
                    break;
                }
            }
            for (int i = mex1 + 1; i < 101; i++) {
                if (freq[i] == 0) {
                    mex2 = i;
                    break;
                }
            }
            int ans = mex1 + mex2;
            System.out.println(ans);
        }
        scanner.close();
    }
}
