#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> deg(n + 1, 0);
    for (int i = 0; i < m; ++i) {
        int a, b;
        cin >> a >> b;
        ++deg[a]; ++deg[b];
    }
    bool ring = true, star = true, bus = true;
    for (int i = 1; i <= n; ++i) {
        if (deg[i] > 2) star = false;
        if (deg[i] != 2) ring = false;
        if (deg[i] != 2 && deg[i] != n - 1) bus = false;
    }
    if (bus) cout << "bus topology\n";
    else if (ring) cout << "ring topology\n";
    else if (star) cout << "star topology\n";
    else cout << "unknown topology\n";
    return 0;
}
