import java.io.IOException;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int v = scanner.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }

        int[] dp = new int[n];
        for (int i = 0; i < n; i++) {
            dp[i] = -1;
        }
        dp[0] = v;

        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (dp[j] >= x[i] - x[j]) {
                    dp[i] = Math.max(dp[i], dp[j] - (x[i] - x[j]));
                    if (dp[i] > 0) {
                        break;
                    }
                }
            }
            if (dp[i] >= 0) {
                dp[i] /= 2;
                dp[i] += v;
            }
        }

        for (int i = 0; i < n; i++) {
            if (dp[i] >= 0) {
                System.out.println("Possible");
            } else {
                System.out.println("Impossible");
            }
        }
    }
}
