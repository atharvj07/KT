t = int(input())

for _ in range(t):
    s = input()
    freq = {}
    for char in s:
        if char in freq:
            freq[char] += 1
        else:
            freq[char] = 1
    
    red = 0
    more_than_one = 0
    for key in freq:
        if freq[key] >= 2:
            more_than_one += 1
        else:
            red += 1
    
    red += more_than_one // 2
    print(red)
