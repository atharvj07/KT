#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int last = -1e9, cnt = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '1') {
        cnt = 0;
        last = i;
      } else {
        cnt++;
        if (i - last > k && i - last < 2 * k) {
          cnt--;
        }
      }
    }
    cout << cnt << endl;
  }
  return 0;
}
