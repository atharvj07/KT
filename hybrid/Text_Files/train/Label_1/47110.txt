import java.util.Arrays;

public class Main {
  static final int MOD = 1000_000_007;
  int h, w, k;

  public Main(int h, int w, int k) {
    this.h = h;
    this.w = w;
    this.k = k;
  }

  private long count(boolean[] opened, int y) {
    if (y == h) return 1;
    long res = 0;
    for (int i = 0; i < w - 1; i++) {
      if (!opened[i] && !opened[i + 1]) {
        opened[i] = opened[i + 1] = true;
        res += count(opened, y + 1);
        opened[i] = opened[i + 1] = false;
      }
    }
    return res;
  }

  public long solve() {
    boolean[] opened = new boolean[w];
    Arrays.fill(opened, false);
    opened[k - 1] = true;
    long res = count(opened, 0);
    if (h == 1) res++;
    return res % MOD;
  }

  public static void main(String[] args) {
    int h, w, k;
    java.util.Scanner stdin = new java.util.Scanner(System.in);
    h = stdin.nextInt();
    w = stdin.nextInt();
    k = stdin.nextInt();
    Main m = new Main(h, w, k);
    System.out.println(m.solve());
  }
}
