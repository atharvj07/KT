#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> m(n);
  for (int i = 0; i < n; ++i) { cin >> m[i]; }
  int ans = 0;
  for (int i = 1; i < n - 1; ++i) {
    for (int j = 1; j < n - 1; ++j) {
      if (m[i][j] == 'X' && m[i - 1][j - 1] == 'X' &&
          m[i - 1][j + 1] == 'X' && m[i + 1][j - 1] == 'X' &&
          m[i + 1][j + 1] == 'X') {
        ++ans;
      }
    }
  }
  cout << ans << "\n";
}
