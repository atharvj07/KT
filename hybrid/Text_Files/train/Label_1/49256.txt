import java.util.Scanner;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int d = sc.nextInt();
        int[] k = new int[d];
        for (int i = 0; i < d; i++) {
            k[i] = sc.nextInt();
        }
        int[] ans = new int[d];
        int[] last = new int[m + 1];
        Arrays.fill(last, -1);
        for (int i = d - 1; i >= 0; i--) {
            int del = k[i];
            last[c[del - 1]] = i;
            int mex = 0;
            for (int j = 1; j <= m; j++) {
                if (last[j] > i) {
                    mex = Math.max(mex, p[last[j] - 1]);
                }
            }
            ans[i] = mex;
        }
        for (int i = 0; i < d; i++) {
            System.out.println(ans[i]);
        }
    }
}
