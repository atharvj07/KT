n = int(input())
a = list(map(int, input().split()))

def min_operations(a):
    operations = []
    for value in a:
        op = 0
        while value % 2 == 0:
            value //= 2
            op += 1
        operations.append(op)
    min_op = min(operations)
    max_op = max(operations)
    
    total_operations = 0
    for op in operations:
        total_operations += abs(op - min_op)
        
    return total_operations + (max_op - min_op)

print(min_operations(a))
