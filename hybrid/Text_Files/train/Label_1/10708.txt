#include <cstdio>
#include <vector>
#include <unordered_map>
typedef long long ll;
using namespace std;

int main() {
    ll n, a;
    unordered_map<ll, ll> um;
    vector<ll> g, h;
    scanf("%lld", &n);
    for (ll i = 0; i < n; ++i) {
        scanf("%lld", &a);
        if (um.find(a) == um.end()) { um[a] = 1; } else { um[a]++; }
    }
    ll m = 0;
    for (auto p : um) {
        if (p.second > 1) { m++; }
    }
    if (m == 0) { printf("-1"); return 0; }
    printf("%lld\n", m);
    for (auto p : um) {
        if (p.second > 1) { g.push_back(p.first); }
    }
    for (auto p : um) {
        if (um.find(p.first) == um.end()) { h.push_back(p.first); } else if (um[p.first] > 1) { h.push_back(p.first); }
    }
    for (ll i = 0; i < n; ++i) { printf("%lld ", g[i]); } printf("\n");
    for (ll i = 0; i < m; ++i) { printf("%lld ", h[i]); }
    return 0;
}
