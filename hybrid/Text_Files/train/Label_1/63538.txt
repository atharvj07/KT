import sys

def check(px, py, s):
    x = y = 0
    for c in s:
        if c == 'U':
            y += 1
        elif c == 'D':
            y -= 1
        elif c == 'R':
            x += 1
        elif c == 'L':
            x -= 1
    return x == px and y == py

for _ in range(int(input())):
    px, py = map(int, input().split())
    s = input()
    print(['NO', 'YES'][(check(px, py, s) or check(px, py, s.replace('U', '').replace('D', '')))])
