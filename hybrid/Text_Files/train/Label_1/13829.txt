#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9+7;
vector<vector<ll>> adj;
vector<ll> sub;
vector<ll> apples;
vector<bool> vis;
void dfs(ll node, ll p) {
    vis[node] = true;
    sub[node] = apples[node];
    for(auto x: adj[node]) {
        if(p == x) continue;
        dfs(x, node);
        sub[node] += sub[x];
    }
}
int main() {
    ll n;
    cin >> n;
    adj.resize(n+1), sub.resize(n+1), apples.resize(n+1), vis.resize(n+1);
    for(ll i = 1; i <= n; i++) cin >> apples[i];
    for(ll i = 2; i <= n; i++) {
        ll x;
        cin >> x;
        adj[i].push_back(x);
        adj[x].push_back(i);
    }
    dfs(1, 1);
    ll ans = 0;
    for(ll i = 1; i <= n; i++) {
        for(auto x: adj[i]) {
            if(i == 1) continue;
            if(sub[i]-sub[x] == sub[x]) ans++;
        }
    }
    cout << ans << endl;
}
