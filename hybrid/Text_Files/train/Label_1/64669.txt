n=int(input())
a=list(map(int,input().split()))
lis=[1]*n
lds=[1]*n
for i in range(1,n):
    for j in range(i):
        if a[i]>a[j] and lis[i]<lis[j]+1:
            lis[i]=lis[j]+1
for i in range(n-2,-1,-1):
    for j in range(n-1,i,-1):
        if a[i]>a[j] and lds[i]<lds[j]+1:
            lds[i]=lds[j]+1
mx=0
for i in range(n):
    mx=max(mx,lis[i]+lds[i]-1)
print('3'*mx)
