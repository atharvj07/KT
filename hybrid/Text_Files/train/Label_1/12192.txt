#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 1e5 + 10;
int n, k, a[N], d[N], e[N];

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> d[i];
  for (int i = 1; i < n; i++) cin >> e[i];
  vector<int> v;
  for (int i = 1; i < n; i++) {
    int dif = abs(e[i] - (i + 1));
    if (dif) v.push_back(dif);
  }
  sort(v.begin(), v.end(), greater<int>());
  if (v.size() > k) {
    ll ans = 0;
    for (int i = 0; i < k; i++) ans += v[i];
    cout << ans << endl;
    return 0;
  }
  ll ans = 0;
  for (int i = 0; i < (int)v.size(); i++) {
    if (i < k) ans -= v[i];
    ans += a[e[i + 1]];
    ans -= d[i + 1];
  }
  cout << ans << endl;
  return 0;
}
