#include <bits/stdc++.h>

using namespace std;

void solve(long long a) {
    int res = 1;
    for (long long i = 2; i * i <= a; ++i) {
        if (a % i == 0) {
            res = i;
            while (a % i == 0) a /= i;
        }
    }
    if (a > 1) res = a;
    cout << res << '\n';
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        long long a;
        cin >> a;
        solve(a);
    }
    return 0;
}
