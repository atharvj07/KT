import java.util.*;
import java.lang.*;

class Main{
    static int mod = 1000000007;
    static long[][] fac;
    static ArrayList<Integer>[] adj;
    static int[] siz;
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long ans = 1;
        ArrayList<Long> x = new ArrayList<Long>();
        ArrayList<Long> y = new ArrayList<Long>();
        Point[] p = new Point[n + 1];
        for(int i = 0; i < n; ++i){
            x.add(in.nextLong());
            y.add(in.nextLong());
        }
        x.add(x.get(0));
        y.add(y.get(0));
        for(int i = 1; i <= n; ++i){
            p[i] = new Point(x.get(i - 1), y.get(i - 1));
        }
        for(int i = 1; i <= n; ++i){
            for(int j = i + 1; j <= n; ++j){
                for(int k = j + 1; k <= n; ++k){
                    if(isTriangle(p[i], p[j], p[k])){
                        ans *= 3;
                        ans %= mod;
                    }
                }
            }
        }
        System.out.println(ans);
    }

    static boolean isTriangle(Point p1, Point p2, Point p3){
        if(convex(p1, p2, p3) == -1){
            return false;
        }
        if(onSegment(p1, p2, p3) == true){
            return false;
        }
        if(onSegment(p2, p3, p1) == true){
            return false;
        }
        if(onSegment(p3, p1, p2) == true){
            return false;
        }
        return true;
    }

    static int convex(Point p1, Point p2, Point p3){
        long val = (p2.y - p1.y) * (p3.x - p2.x) - (p2.x - p1.x) * (p3.y - p2.y);
        if(val > 0){
            return 1;
        }else if(val < 0){
            return -1;
        }else{
            return 0;
        }
    }

    static boolean onSegment(Point p1, Point p2, Point p3){
        if(p2.x <= Math.max(p1.x, p3.x) && p2.x >= Math.min(p1.x, p3.x) && p2.y <= Math.max(p1.y, p3.y) && p2.y >= Math.min(p1.y, p3.y)){
            return true;
        }
        return false;
    }

    static class Point{
        long x;
        long y;
        public Point(long _x, long _y){
            x = _x;
            y = _y;
        }
    }
}
