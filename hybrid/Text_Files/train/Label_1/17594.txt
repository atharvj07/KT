#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_map<int, int> cnt;
    for (int i = 0, x; i < n; i++) {
      cin >> x;
      cnt[x]++;
    }
    int ans = 0;
    for (auto p : cnt) ans += p.second % 2;
    cout << (ans <= 1 ? 0 : ans - 1) << endl;
  }
  return 0;
}
