#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  array<ll, 6> g[6];
  for (ll i = 1; i <= 5; i++) {
    for (ll j = 1; j <= 5; j++) { cin >> g[i][j]; }
  }
  vector<ll> p = {0, 2, 3, 1, 5, 4};
  ll ans = 0;
  for (ll i = 1; i <= 5; i++) { ans += g[p[i]][p[i % 5 + 1]] + g[p[i % 5 + 1]][p[i]]; }
  for (ll i = 1; i <= 5; i++) {
    ll x = p[i], y = p[i % 5 + 1], t = p[i % 5 + 2];
    ans = max(ans, g[x][y] + g[y][x] + g[x][t] + g[t][x] - g[y][t] - g[t][y]);
  }
  cout << ans << endl;
  return 0;
}
