#include <iostream>
#include <vector>

int longestCommonSubsequence(const std::string& s, const std::string& t) {
  std::vector<std::vector<int>> dp(s.length() + 1, std::vector<int>(t.length() + 1, 0));
  for (int i = 1; i <= s.length(); i++) {
    for (int j = 1; j <= t.length(); j++) {
      dp[i][j] = std::max(dp[i][j - 1], dp[i - 1][j]);
      if (s[i - 1] == t[j - 1]) {
        dp[i][j] = std::max(dp[i][j], dp[i - 1][j - 1] + 1);
      }
    }
  }
  return s.length() - dp[s.length()][t.length()];
}

int main() {
  std::string s, t;
  std::cin >> s >> t;
  std::cout << longestCommonSubsequence(s, t) << std::endl;
  return 0;
}
