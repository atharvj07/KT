import sys

input = sys.stdin.readline
n = int(input())
data = []
for _ in range(n):
    data.append(list(map(int, input().split())))

x = [a[0] for a in data]
y = [a[1] for a in data]
xmin = min(x)
xmax = max(x)
ymin = min(y)
ymax = max(y)

x2 = xmin + xmax + 1
y2 = ymin + ymax + 1

print(min(x2, y2))
