#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define vvi vector<vi>

int main() {
  int n, m, h, u;
  cin >> n >> m >> h;
  vi v(n);
  for (int &x : v) cin >> x;
  vector<pair<int, int>> ans;
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    if (v[a] == v[b]) ans.push_back({a, b});
  }
  int k = ans.size();
  cout << k << endl;
  for (auto &[a, b] : ans) cout << a + 1 << " ";
}
