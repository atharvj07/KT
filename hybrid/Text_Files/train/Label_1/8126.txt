#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> y(n);
    for (auto &i: y) cin >> i;
    sort(y.begin(), y.end());
    int ans = false;
    for (int i = 1; i + 1 < y.size(); i++) {
        if (y[i] == y[i-1] && y[i] == y[i+1]) ans = true;
    }
    cout << (ans ? "Yes" : "No") << endl;
}
