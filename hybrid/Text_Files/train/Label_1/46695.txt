import java.util.Scanner;

public class VasyaAndThePainting {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] paper = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                paper[i][j] = line.charAt(j);
            }
        }

        Frame firstFrame = findFrame(paper);
        if (firstFrame == null) {
            System.out.println("NO");
            return;
        }

        Frame secondFrame = findFrame(removeFrame(paper, firstFrame));
        if (secondFrame == null) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        System.out.println(firstFrame.x1 + " " + firstFrame.y1 + " " + firstFrame.x2 + " " + firstFrame.y2);
        System.out.println(secondFrame.x1 + " " + secondFrame.y1 + " " + secondFrame.x2 + " " + secondFrame.y2);
    }

    private static Frame findFrame(char[][] paper) {
        int n = paper.length;
        int m = paper[0].length;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (paper[i][j] == '#') {
                    Frame frame = findFrame(paper, i, j);
                    if (frame != null) {
                        return frame;
                    }
                }
            }
        }
        return null;
    }

    private static Frame findFrame(char[][] paper, int x, int y) {
        int n = paper.length;
        int m = paper[0].length;
        int x1 = x;
        int y1 = y;
        int x2 = x;
        int y2 = y;
        while (x1 > 0 && paper[x1 - 1][y] == '#') {
            x1--;
        }
        while (y1 > 0 && paper[x][y1 - 1] == '#') {
            y1--;
        }
        while (x2 < n - 1 && paper[x2 + 1][y] == '#') {
            x2++;
        }
        while (y2 < m - 1 && paper[x][y2 + 1] == '#') {
            y2++;
        }
        if (x2 - x1 + 1 < 3 || y2 - y1 + 1 < 3) {
            return null;
        }
        return new Frame(x1, y1, x2, y2);
    }

    private static char[][] removeFrame(char[][] paper, Frame frame) {
        int n = paper.length;
        int m = paper[0].length;
        char[][] newPaper = new char[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                newPaper[i][j] = paper[i][j];
            }
        }
        for (int i = frame.x1; i <= frame.x2; i++) {
            for (int j = frame.y1; j <= frame.y2; j++) {
                if (i == frame.x1 || i == frame.x2 || j == frame.y1 || j == frame.y2) {
                    newPaper[i][j] = '.';
                }
            }
        }
        return newPaper;
    }

    private static class Frame {
        int x1;
        int y1;
        int x2;
        int y2;

        public Frame(int x1, int y1, int x2, int y2) {
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
        }
    }
}
