import sys
n = int(sys.stdin.readline())
lst = list(map(int, sys.stdin.readline().split()))
lst2 = []
for i in range(n):
    if lst[i] == 0:
        lst2.append(i)
for i in lst2:
    lst.remove(i)
lst2.sort()
ans = 0
for i in range(1, len(lst)):
    if lst[i] == 0:
        continue
    ans += 1
    l = lst2[0]
    r = lst2[-1]
    while lst2 and lst[l] == 0:
        l = lst2.pop(0)
    while lst2 and lst[r] == 0:
        r = lst2.pop()
    if l >= lst[i] and r <= lst[i]:
        lst2.remove(l)
        lst2.remove(r)
    elif l <= lst[i]:
        for j in range(l, i + 1):
            lst[j] -= 1
    else:
        for j in range(i, r + 1):
            lst[j] -= 1
print(ans)
