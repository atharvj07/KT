#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    std::ios::sync_with_stdio(false);
    long long n, m;
    std::cin >> n >> m;
    std::vector<long long> x1(m), y1(m), x2(m), y2(m);
    for (long long i = 0; i < m; ++i)
        std::cin >> x1[i] >> y1[i] >> x2[i] >> y2[i];
    std::vector<long long> X, Y;
    for (long long i = 0; i < m; ++i) {
        X.push_back(x1[i]);
        X.push_back(x2[i]);
        Y.push_back(y1[i]);
        Y.push_back(y2[i]);
    }
    std::sort(X.begin(), X.end());
    std::sort(Y.begin(), Y.end());
    X.resize(std::unique(X.begin(), X.end()) - X.begin());
    Y.resize(std::unique(Y.begin(), Y.end()) - Y.begin());
    long long ans = 0;
    for (long long i = 0; i < X.size() - 1; ++i)
        for (long long j = 0; j < Y.size() - 1; ++j) {
            long long cnt = 0;
            for (long long k = 0; k < m; ++k)
                if (x1[k] <= X[i] && x2[k] >= X[i + 1] && y1[k] <= Y[j] && y2[k] >= Y[j + 1])
                    ++cnt;
            if (cnt % 2)
                ans += (X[i + 1] - X[i]) * (Y[j + 1] - Y[j]);
        }
    std::cout << ans << '\n';
}
