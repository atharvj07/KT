import java.util.Scanner;

public class TreesWithMaximumMatching {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        // dp[i][j] is the number of trees with i vertices and a maximum matching of size j.
        long[][] dp = new long[n + 1][k + 1];
        dp[1][0] = 1; // Base case: a single vertex tree with no matching.

        for (int i = 2; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                // Case 1: Add a new vertex connected to the root.
                dp[i][j] = (dp[i][j] + dp[i - 1][j] * (i - 1)) % MOD;

                // Case 2: Add a new vertex connected to a non-root vertex with degree 1.
                if (j > 0) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - 1] * (i - 2)) % MOD;
                }

                // Case 3: Add a new vertex connected to a non-root vertex with degree 2.
                if (j > 1) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - 2] * (i - 3)) % MOD;
                }
            }
        }

        System.out.println(dp[n][k]);
    }
}
