#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '0' && t[i] == '0') ans += 2;
      if (s[i] == '1' && t[i] == '0') ans += 1;
      if (s[i] == '0' && t[i] == '1') ans += 1;
    }
    cout << ans << endl;
  }
  return 0;
}
