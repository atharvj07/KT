#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n, p;
  cin >> n >> p;
  string s;
  cin >> s;
  int i = n - 1;
  while (i >= 0 && s[i] == ('a' + (i % p))) i--;
  if (i < 0) {
    cout << "NO" << endl;
  } else {
    s[i]++;
    for (int j = i + 1; j < n; j++) {
      s[j] = 'a';
    }
    cout << s << endl;
  }
  return 0;
}
