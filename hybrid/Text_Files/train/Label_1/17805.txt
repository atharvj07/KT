#include <iostream>
#include <vector>

int main() {
    int n, m, k; std::cin >> n >> m >> k;
    std::vector<int> a(n), b(m);
    for (int& x : a) std::cin >> x;
    for (int& x : b) std::cin >> x;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (a[i] == b[j]) {
                int ni = i, nj = j;
                while (ni < n && nj < m && a[ni] == b[nj]) {
                    ++ni, ++nj;
                }
                --ni, --nj;
                ans += (ni - i + 1) * (nj - j + 1) * (a[i] == 1);
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
