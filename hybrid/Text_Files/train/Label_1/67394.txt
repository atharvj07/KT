import sys
read = sys.stdin.readline

s = read().strip()
n = len(s)
MOD = int(1e9) + 7

# dp[i][bombs_left][cur_num] = number of ways to fill the remaining cells with bombs and numbers from the current cell i
dp = [[[0 for _ in range(3)] for _ in range(n+1)] for _ in range(n+1)]
dp[n][0][0] = 1 # base case: all cells filled, no bombs left, current number is 0

for i in range(n-1,-1,-1):
    for bombs_left in range(n+1):
        for cur_num in range(3):
            # place a bomb
            if s[i] == '?':
                dp[i][bombs_left][cur_num] = (dp[i][bombs_left][cur_num] + dp[i+1][bombs_left-1][0]) % MOD
            # place a number
            if s[i] == '?' or s[i] == str(cur_num):
                dp[i][bombs_left][cur_num] = (dp[i][bombs_left][cur_num] + dp[i+1][bombs_left][cur_num+1]) % MOD
                # if current number is 0, can also place 1 or 2
                if cur_num == 0:
                    dp[i][bombs_left][cur_num] = (dp[i][bombs_left][cur_num] + dp[i+1][bombs_left][1] + dp[i+1][bombs_left][2]) % MOD
print(dp[0][n][0])
