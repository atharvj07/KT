def count_inversions(arr):
    inv_count = 0
    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):
            if arr[i] > arr[j]:
                inv_count += 1
    return inv_count

n = int(input().strip())
a = list(map(int, input().strip().split()))
m = int(input().strip())

for _ in range(m):
    l, r = map(int, input().strip().split())
    a[l-1:r] = reversed(a[l-1:r])
    if count_inversions(a) % 2 == 0:
        print("even")
    else:
        print("odd")
