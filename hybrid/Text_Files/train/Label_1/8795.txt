#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    long long n, x, sum = 0, ans = 0; cin >> n >> x;
    vector<long long> ar(n);
    for (long long &i : ar) { cin >> i; sum += i; }
    if (sum >= x * n) ans = n;
    else ans = sum / x;
    cout << ans << endl;
  }
  return 0;
}
