import sys
input=sys.stdin.readline

n=int(input())
s=input()
t=input()
ans=[]
for i in range(n-1):
	if s[i]!=t[i]:
		for j in range(i+1,n):
			if s[j]==t[i]:
				ans.append(j)
				ans.append(i)
				s=s[:i]+s[j]+s[i+1:j]+s[i]+s[j+1:]
				break
if s==t:
	print(len(ans))
	for i in ans:
		print(i,end=' ')
else:
	print(-1)
