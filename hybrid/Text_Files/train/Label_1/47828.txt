import java.util.*;

public class Main {
    static class Pair {
        int x, y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static int h, w;
    static int[][] a;
    static Pair[] dxdy = {new Pair(-1, 0), new Pair(1, 0), new Pair(0, 1), new Pair(0, -1)};
    static int ans = 0;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        h = sc.nextInt();
        w = sc.nextInt();
        a = new int[h + 1][w + 1];
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        solve(1, 1, 0, new int[h + 1][w + 1]);
        System.out.println(ans);
    }

    static void solve(int x, int y, int sum, int[][] vis) {
        if (x == h && y == w) {
            ans = Math.max(ans, sum);
            return;
        }
        vis[x][y] = 1;
        for (Pair dir : dxdy) {
            int nx = x + dir.x;
            int ny = y + dir.y;
            if (nx >= 1 && nx <= h && ny >= 1 && ny <= w && vis[nx][ny] == 0) {
                solve(nx, ny, sum + a[nx][ny], vis);
            }
        }
        vis[x][y] = 0;
    }
}
