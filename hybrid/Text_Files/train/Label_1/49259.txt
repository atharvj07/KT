import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int d = sc.nextInt();
        int[] k = new int[d];
        for (int i = 0; i < d; i++) {
            k[i] = sc.nextInt();
        }
        int[] club = new int[m + 1];
        for (int i = 0; i < n; i++) {
            club[c[i]]++;
        }
        List<Integer>[] clubList = new List[m + 1];
        for (int i = 1; i <= m; i++) {
            clubList[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            clubList[c[i]].add(p[i]);
        }
        int[] ans = new int[d];
        for (int i = 0; i < d; i++) {
            club[c[k[i] - 1]]--;
            clubList[c[k[i] - 1]].remove(Integer.valueOf(p[k[i] - 1]));
            int max = 0;
            for (int j = 1; j <= m; j++) {
                if (club[j] > 0) {
                    clubList[j].sort(Collections.reverseOrder());
                    max = Math.max(max, clubList[j].get(club[j] - 1));
                }
            }
            ans[i] = max;
        }
        for (int i = 0; i < d; i++) {
            System.out.println(ans[i]);
        }
    }
}
