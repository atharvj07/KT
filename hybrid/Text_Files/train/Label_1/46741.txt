import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int count = 0;
        int last = 0;
        while (n > 0) {
            int current = Math.min(n, last + 1);
            n -= current;
            last = current;
            count++;
        }
        System.out.println(count);
    }
}
