import sys

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [list() for _ in range(n + 1)]
for _ in range(m):
    x, y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)

q = int(input())
vis = [False for _ in range(n + 1)]
K = [0 for _ in range(n + 1)]
for i in range(1, n + 1):
    K[i] = i

def dfs1(x, p):
    vis[x] = True
    for u in adj[x]:
        if not vis[u]:
            dfs1(u, x)
            if K[u] < K[x]:
                K[x] = K[u]

def dfs2(x, p):
    vis[x] = True
    for u in adj[x]:
        if not vis[u]:
            dfs2(u, x)
            if K[x] == K[u] and x < u:
                K[u] = K[x] + 1

ans = [n for _ in range(q)]
for query in range(q):
    cmd = input().strip()
    if cmd[0] == '1':
        x, y = map(int, cmd.split()[1:])
        adj[x].append(y)
        adj[y].append(x)
    elif cmd[0] == '2':
        x, y = map(int, cmd.split()[1:])
        adj[x].remove(y)
        adj[y].remove(x)
    else:
        vis = [False for _ in range(n + 1)]
        for i in range(1, n + 1):
            if not vis[i]:
                dfs1(i, 0)
        vis = [False for _ in range(n + 1)]
        for i in range(1, n + 1):
            if not vis[i]:
                dfs2(i, 0)
        cnt = 0
        for i in range(1, n + 1):
            if vis[i]:
                cnt += 1
        ans[query] = cnt

for i in ans:
    print(i)
