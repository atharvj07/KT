#include <iostream>
#include <string>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::string s, t;
  std::cin >> s >> t;
  int i = 0, j = 0;
  char wildcard = '*';
  while (i < n && j < m) {
    if (t[j] == wildcard || s[i] == t[j]) {
      i++;
      j++;
    } else if (s[i] == wildcard) {
      i++;
    } else {
      std::cout << "NO" << std::endl;
      return 0;
    }
  }
  while (i < n) {
    if (s[i] != wildcard) {
      std::cout << "NO" << std::endl;
      return 0;
    }
    i++;
  }
  std::cout << "YES" << std::endl;
  return 0;
}
