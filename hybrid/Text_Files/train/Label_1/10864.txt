#include <bits/stdc++.h>
using namespace std;
const int M=20;
typedef long long ll;
vector<ll> p[M],ls;
int main(){
    ll q;
    cin>>q;
    while(q--){
        string s,t;
        cin>>s>>t;
        ll a=(1ll<<s.size())-1,b=(1ll<<t.size())-1;
        for(ll i=1;i*i<=a;i++){
            if(a%i==0){
                p[s.size()].push_back(i);
                if(i*i!=a)p[s.size()].push_back(a/i);
            }
        }
        for(ll i=1;i*i<=b;i++){
            if(b%i==0){
                p[t.size()].push_back(i);
                if(i*i!=b)p[t.size()].push_back(b/i);
            }
        }
        ll gcd=__gcd(a,b);
        for(ll e:p[s.size()]){
            for(ll f:p[t.size()]){
                if(gcd%(__gcd(e,f))==0)ls.push_back(__gcd(e,f));
            }
        }
        sort(ls.begin(),ls.end());
        ls.resize(unique(ls.begin(),ls.end())-ls.begin());
        bool flg=false;
        for(ll x:ls){
            if(a%x==0&&b%x==0){
                string ans="";
                for(ll i=0;i<x;i++)ans+=s;
                cout<<ans<<endl;
                flg=true;
                break;
            }
        }
        if(!flg)cout<<-1<<endl;
        ls.clear();
        p[s.size()].clear();
        p[t.size()].clear();
    }
}
