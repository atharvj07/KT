from collections import defaultdict,deque
n,k=map(int,input().split())
l=list(map(int,input().split()))
d=defaultdict(lambda:deque())
for i in map(int,input().split()):
    d[i]=deque([0]*(n+1))
ans=0
for i in range(1,n+1):
    for j in d:
        l[i-1]*=l[j-1]
        ans+=l[i-1]*d[j].pop()
        d[j].appendleft(l[i-1])
    l[i-1]=0
print(ans)
