import sys

n = int(input())

board = []
for _ in range(n):
    row = input()
    board.append(row)

cnt = 0
for i in range(n):
    for j in range(n):
        if board[i][j] == '#':
            cnt += 1

if cnt % 5 != 0:
    print("NO")
    sys.exit()

cover = [[False] * n for _ in range(n)]
def dfs(x, y, cnt):
    if not (0 <= x < n and 0 <= y < n):
        return
    if cover[x][y] or board[x][y] == '.':
        return
    cover[x][y] = True
    cnt += 1
    dfs(x + 1, y, cnt)
    dfs(x - 1, y, cnt)
    dfs(x, y + 1, cnt)
    dfs(x, y - 1, cnt)
    if cnt == 5:
        return True

flag = False
for i in range(n):
    for j in range(n):
        if board[i][j] == '#' and not cover[i][j]:
            if dfs(i, j, 1):
                flag = True
                break
    if flag:
        break

if flag:
    print("YES")
else:
    print("NO")
