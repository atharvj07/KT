#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7;
const int mxN = 1e5 + 5;
int dp[mxN][20][2];
int n, k;
int ans;
int go(int idx, int or_val, bool is_lead) {
  if (idx == n) {
    if (or_val == 0) return 1;
    return 0;
  }
  int &ret = dp[idx][or_val][is_lead];
  if (~ret) return ret;
  int to = is_lead ? 9 : (1 << k) - 1;
  ret = 0;
  for (int i = 0; i <= to; i++) {
    if ((or_val | i) != or_val) continue;
    ret = (ret + go(idx + 1, or_val | i, i > 0) % mod) % mod;
  }
  return ret;
}
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    memset(dp, -1, sizeof dp);
    ans = go(0, 0, 1);
    cout << ans << endl;
  }
}
