import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
    Set<Integer> special = new HashSet<>();
    for (int i = 0; i < k; i++)
      special.add(sc.nextInt());
    int[] dist = new int[n + 1];

    ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
    for (int i = 0; i <= n; i++)
      adj.add(new ArrayList<>());
    for (int i = 0; i < m; i++) {
      int a = sc.nextInt(), b = sc.nextInt();
      adj.get(a).add(b);
      adj.get(b).add(a);
    }

    Queue<Integer> q = new LinkedList<>();
    q.offer(1);
    Arrays.fill(dist, -1);
    dist[1] = 0;
    while (!q.isEmpty()) {
      int current = q.poll();
      for (Integer next : adj.get(current)) {
        if (dist[next] == -1) {
          dist[next] = dist[current] + 1;
          q.offer(next);
        }
      }
    }

    int maxDist = 0;
    for (Integer field : special)
      maxDist = Math.max(maxDist, dist[field]);
    int farthest = -1;
    for (Integer field : special)
      if (dist[field] == maxDist)
        farthest = field;

    maxDist = 0;
    q.offer(farthest);
    Arrays.fill(dist, -1);
    dist[farthest] = 0;
    while (!q.isEmpty()) {
      int current = q.poll();
      for (Integer next : adj.get(current)) {
        if (dist[next] == -1) {
          dist[next] = dist[current] + 1;
          q.offer(next);
        }
      }
    }

    for (Integer field : special)
      maxDist = Math.max(maxDist, dist[field]);

    System.out.println(maxDist);
    sc.close();
  }
}
