#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
 int t;
 cin>>t;
 while(t--){
  ll n,s,ans=0,sum=0;
  cin>>n>>s;
  vector<ll>a(n);
  for(int i=0;i<n;i++){
   cin>>a[i];
   sum+=a[i];
  }
  if(sum<=s){
   cout<<0<<"\n";
   continue;
  }
  for(int i=0;i<n;i++){
   if(sum-a[i]<=s){
    ans=i+1;
   }
  }
  cout<<ans<<"\n";
 }
 return 0;
}
