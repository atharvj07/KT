n, k = map(int, input().split())
if k > n * n / 2:
    print(-1)
else:
    a = [[0 for i in range(n)] for j in range(n)]
    r = k // n
    c = k % n
    for i in range(0, r):
        a[i][i] = 1
        a[n-i-1][i] = 1
    for i in range(n-1-r, n):
        a[i][i+1] = 1
        a[n-i-1][i+1] = 1
    if c > 0:
        a[r][r] = 1
        c -= 1
    for i in range(r+1, n-1-r):
        if c == 0:
            break
        if a[i][i] == 0:
            a[i][i] = 1
            c -= 1
        if c == 0:
            break
        if a[n-i-1][i] == 0:
            a[n-i-1][i] = 1
            c -= 1
    for i in range(0, n):
        for j in range(0, n):
            print(a[i][j], end=' ')
        print()
