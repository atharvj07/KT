import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.PriorityQueue;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int d = Integer.parseInt(input[1]);
        Pair[] employees = new Pair[n];
        for (int i = 0; i < n; i++) {
            input = br.readLine().split(" ");
            employees[i] = new Pair(Integer.parseInt(input[0]), Integer.parseInt(input[1]));
        }
        Arrays.sort(employees, Comparator.comparingInt(p -> p.floor));
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        int totalTime = 0;
        int cannotTake = 0;
        for (Pair employee : employees) {
            while (!pq.isEmpty() && pq.peek() <= employee.time) {
                pq.poll();
            }
            if (pq.size() < d) {
                pq.add(employee.time + employee.floor - 1);
                totalTime += employee.floor;
            } else {
                cannotTake++;
            }
        }
        System.out.println(cannotTake > 0 ? -1 : totalTime);
    }

    static class Pair {
        int floor;
        int time;

        public Pair(int floor, int time) {
            this.floor = floor;
            this.time = time;
        }
    }
}
