#include <iostream>
#include <cstring>
using namespace std;
int main() {
  int n;
  cin >> n;
  while (n--) {
    string s;
    cin >> s;
    int len = s.size();
    bool diverse = true;
    for (int i = 0; i < len - 1; i++) {
      if (s[i] + 1 != s[i + 1]) {
        diverse = false;
      }
    }
    if (diverse) {
      cout << "Yes" << endl;
    } else {
      cout << "No" << endl;
    }
  }
}
