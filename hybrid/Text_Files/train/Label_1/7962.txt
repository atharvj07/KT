#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ll r;
    cin >> r;
    for (ll x = 1; x * x + 2 * x + 1 <= r; ++x) {
        ll y = (r - x * x - x - 1) / (2 * x);
        if (y >= 1) {
            cout << x << ' ' << y << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
}
