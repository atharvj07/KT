import heapq
import sys

n = int(input())
a = []
heapq.heapify(a)
ans = []

for i in range(n):
    x = input().split()
    if x[0] == 'insert':
        heapq.heappush(a,int(x[1]))
        ans.append('insert '+x[1])
    elif x[0] == 'getMin':
        while a and a[0]>int(x[1]):
            heapq.heappop(a)
            ans.append('removeMin')
        if a and a[0]==int(x[1]):
            ans.append('getMin '+x[1])
        else:
            ans.append('insert '+x[1])
            heapq.heappush(a,int(x[1]))
            ans.append('getMin '+x[1])
    else:
        while a and a[0]<int(x[1]):
            heapq.heappop(a)
            ans.append('removeMin')
        if a and a[0]==int(x[1]):
            ans.append('removeMin')
        else:
            ans.append('insert '+x[1])
            heapq.heappush(a,int(x[1]))
            ans.append('removeMin')

print(len(ans))
for x in ans:
    print(x)
