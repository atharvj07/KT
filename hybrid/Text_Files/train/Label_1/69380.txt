import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edges = [[] for i in range(n+1)]
for i in range(m):
    a, b, w = map(int, input().split())
    edges[a].append((b, w))
    edges[b].append((a, w))
vis = [False] * (n+1)
ans = []
mins = 0
def dfs(x, p, w):
    global mins
    vis[x] = True
    for y, z in edges[x]:
        if y==p: continue
        if vis[y]:
            if z%2==w%2: return False
            else: ans.append((x, y)); mins+=z
            return True
        if dfs(y, x, (w+z)%2): return True
    return False
for i in range(1, n+1):
    if not vis[i]:
        if dfs(i, -1, 0): break
print(mins, len(ans))
print(*ans)
