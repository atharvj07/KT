#include <iostream>
#include <string>
using namespace std;

int main() {
  int L;
  string a;
  cin >> L >> a;

  while (a.size() < L) a += '0';
  a[L-1]++;
  for (int i = L - 1; i > 0; i--) {
    if (a[i] == '9' + 1) {
      a[i] = '0';
      a[i-1]++;
    }
  }

  if (a[0] == '9' + 1) { a.insert(0, "1"); }
  cout << a << endl;

  return 0;
}
