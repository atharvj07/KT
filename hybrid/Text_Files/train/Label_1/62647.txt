import sys

def possible(a, k):
    n = len(a)
    a.sort()
    if a[n // 2 - 1] > k:
        return 'no'
    if a[n // 2] < k:
        return 'yes'
    if n % 2 == 0:
        if a[n // 2 - 1] == k and a[n // 2] == k:
            return 'yes'
    return 'no'

for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    print(possible(a, k))
