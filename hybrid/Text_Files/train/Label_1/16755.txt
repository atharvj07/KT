#include <iostream>
#include <string>

int main()
{
	std::string name;
	std::cin >> name;
	int cnt = 0;
	for (int i = 0; i < name.length(); i++) {
		bool new_char = true;
		for (int j = 0; j < i; j++) {
			if (name[i] == name[j]) {
				new_char = false;
			}
		}
		if (new_char)
			cnt++;
	}
	if (cnt & 1)
		std::cout << "IGNORE HIM!" << std::endl;
	else
		std::cout << "CHAT WITH HER!" << std::endl;
	return 0;
}
