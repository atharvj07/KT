import java.util.*;
import java.io.*;

class Main {
    static boolean[][] cur, wish, bar, vis;
    static int n, ans;
    static boolean done = false;
    static Deque<String> ops = new ArrayDeque<>();

    static boolean same(int r1, int c1, int r2, int c2) {
        return cur[r1][c1] == cur[r2][c2] && wish[r1][c1] == wish[r2][c2];
    }

    static void applyRow(int r) {
        for (int c = 0; c < n; c++) {
            if (bar[r][c] != cur[r][c]) cur[r][c] = !cur[r][c];
        }
        ops.add("row " + r);
        ans++;
    }

    static void applyCol(int c) {
        for (int r = 0; r < n; r++) {
            if (bar[r][c] != cur[r][c]) cur[r][c] = !cur[r][c];
        }
        ops.add("col " + c);
        ans++;
    }

    static void flip(int r, int c) {
        cur[r][c] = !cur[r][c];
        ans++;
        ops.add("flip " + r + " " + c);
    }

    static void dfs(int r, int c) {
        if (r == n || c == n || vis[r][c]) return;
        if (cur[r][c] != wish[r][c]) {
            vis[r][c] = true;
            if (bar[r][c] == cur[r][c]) flip(r, c);
            else {
                if (r < n - 1 && bar[r + 1][c] == cur[r + 1][c] && !same(r + 1, c, r, c)) {
                    dfs(r + 1, c);
                    if (done) return;
                    applyRow(r);
                    dfs(r + 1, c);
                }
                if (c < n - 1 && bar[r][c + 1] == cur[r][c + 1] && !same(r, c + 1, r, c)) {
                    dfs(r, c + 1);
                    if (done) return;
                    applyCol(c);
                    dfs(r, c + 1);
                }
            }
        } else {
            if (r < n - 1 && same(r + 1, c, r, c) && !vis[r + 1][c]) dfs(r + 1, c);
            if (c < n - 1 && same(r, c + 1, r, c) && !vis[r][c + 1]) dfs(r, c + 1);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        int[] dr = {-1, 1, 0, 0};
        int[] dc = {0, 0, -1, 1};
        cur = new boolean[n][n];
        wish = new boolean[n][n];
        bar = new boolean[n][n];
        vis = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            String s = br.readLine();
            for (int j = 0; j < n; j++) {
                cur[i][j] = (s.charAt(j) == '1');
            }
        }
        for (int i = 0; i < n; i++) {
            String s = br.readLine();
            for (int j = 0; j < n; j++) {
                wish[i][j] = (s.charAt(j) == '1');
            }
        }
        String s = br.readLine();
        for (int i = 0; i < n; i++) bar[0][i] = (s.charAt(i) == '1');
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (cur[i][j] != wish[i][j]) vis[i][j] = true;
            }
        }
        dfs(0, 0);
        if (!done) {
            System.out.println(-1);
            return;
        }
        System.out.println(ans);
        for (String op : ops) {
            System.out.println(op);
        }
    }
}
