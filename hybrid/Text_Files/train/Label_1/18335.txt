#include <bits/stdc++.h>
using namespace std;

bool dfs(vector<vector<int>>& v, int idx, vector<int>& ans, int& sum, int& p) {
    if (idx == v.size()) {
        bool ok = true;
        for (int i = 1; i < v.size(); i++) {
            if (ans[i] != ans[i - 1]) {
                ok = false;
                break;
            }
        }
        if (ok) {
            p = ans[0];
            return true;
        }
        return false;
    }
    for (int i = 0; i < v[idx].size(); i++) {
        sum += v[idx][i];
        ans.push_back(sum);
        if (dfs(v, idx + 1, ans, sum, p)) {
            return true;
        }
        sum -= v[idx][i];
        ans.pop_back();
    }
    return false;
}

int main() {
    int k;
    cin >> k;
    vector<vector<int>> v(k + 1);
    for (int i = 1; i <= k; i++) {
        int n;
        cin >> n;
        for (int j = 1; j <= n; j++) {
            int x;
            cin >> x;
            v[i].push_back(x);
        }
    }
    vector<int> ans;
    int sum = 0, p = 0;
    if (dfs(v, 1, ans, sum, p)) {
        cout << "Yes\n";
        for (int i = 1; i <= k; i++) {
            cout << p << " " << i << "\n";
        }
    } else {
        cout << "No\n";
    }
    return 0;
}
