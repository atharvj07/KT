#include <bits/stdc++.h>

using namespace std;

const int INF = 1e9;

struct Flight {
    int id, cost, time;

    bool operator<(const Flight& other) const {
        return cost > other.cost;
    }
};

int main() {
    int n, k;
    cin >> n >> k;

    vector<Flight> flights(n);
    for (int i = 0; i < n; i++) {
        cin >> flights[i].cost;
        flights[i].time = i + 1;
        flights[i].id = i + 1;
    }

    sort(flights.begin(), flights.end());

    int ans = 0;
    vector<int> order(n);
    for (int i = 0; i < n; i++) {
        order[i] = flights[i].id;
        ans += flights[i].cost * (k + i + 1 - flights[i].time);
    }

    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        cout << k + i + 1 << " ";
    }
    cout << endl;

    return 0;
}
