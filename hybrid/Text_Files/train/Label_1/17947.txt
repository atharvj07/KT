#include <bits/stdc++.h>

using namespace std;

typedef long long int lli;

int main() {
    lli n, s;
    cin >> n >> s;
    int cnt = 0;
    for (lli i = 1; i <= n; i++) {
        lli sum = 0, num = i;
        while (num) {
            sum += num % 10;
            num /= 10;
        }
        if (i - sum >= s) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
