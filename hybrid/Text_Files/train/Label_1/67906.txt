import bisect
n,p=map(int,input().split())
a=[];b=[]
for i in range(n):
	c,d=map(int,input().split())
	a.append(c)
	b.append(d)
a.sort()
b.sort()
inf=1e200
start=0;end=inf
while(end-start>1e-9):
	mid=(start+end)/2
	t=0
	for x in b[:bisect.bisect(a,mid*p)]:
		if(x>0):t+=(mid*p-x)/a[bisect.bisect(a,mid*p)-1]
	if(t<=mid):end=mid
	else:start=mid
print(-1 if start==inf else start)
