import java.util.Scanner;
import java.util.Stack;

public class RBSMoves {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int testCases = scanner.nextInt();
        scanner.nextLine(); // Consume newline after the number of test cases

        for (int i = 0; i < testCases; i++) {
            String s = scanner.nextLine();
            int maxMoves = findMaxMoves(s);
            System.out.println(maxMoves);
        }
    }

    static int findMaxMoves(String s) {
        Stack<Character> stack = new Stack<>();
        int moves = 0;

        for (char c : s.toCharArray()) {
            if (c == '(' || c == '[') {
                stack.push(c);
            } else if (c == ')' || c == ']') {
                if (!stack.isEmpty() && stack.peek() == getMatchingBracket(c)) {
                    stack.pop();
                    moves++; // Found a valid RBS
                } else {
                    // Invalid RBS, reset stack and moves
                    stack.clear();
                    moves = 0;
                }
            }
        }

        return moves;
    }

    static char getMatchingBracket(char c) {
        switch (c) {
            case ')': return '(';
            case ']': return '[';
            default: return ' '; // Should not happen
        }
    }
}
