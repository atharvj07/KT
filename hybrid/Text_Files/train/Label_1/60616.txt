def solve_beauty_contest(n, k, beauties):
    beauties.sort()
    results = []

    # First, use individual soldiers to form detachments until we run out of k or soldiers.
    for i in range(n):
        if k > 0:
            results.append(f"1 {beauties[i]}")
            k -= 1
        else:
            break

    # Now, use combinations of soldiers to form unique detachments
    if k > 0:
        for length in range(2, n + 1):
            for i in range(n - length + 1):
                if k > 0:
                    detachment = " ".join(map(str, beauties[i:i + length]))
                    results.append(f"{length} {detachment}")
                    k -= 1
                else:
                    break
            if k <= 0:
                break

    return "\n".join(results)

# Input reading
n, k = map(int, input().split())
beauties = list(map(int, input().split()))

# Solve and output
print(solve_beauty_contest(n, k, beauties))
