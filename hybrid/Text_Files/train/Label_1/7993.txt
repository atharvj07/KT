#include <iostream>
#include <vector>

using namespace std;

typedef unsigned long long ull;
typedef long long ll;

int get_base(const string& number) {
    int max_base = 0;
    for (char c : number) {
        max_base = max(int(c - '0'), max_base);
    }

    return max_base + 1;
}

ull convert_to_decimal(const string& number, ull base) {
    ull decimal = 0;
    int power = 0;
    for (int i = number.length() - 1; i >= 0; --i) {
        decimal += (number[i] - '0') * (ull(base) << power);
        ++power;
    }

    return decimal;
}

int main() {
    ull base;
    cin >> base;

    string number;
    cin >> number;

    if (base == 10) {
        cout << number << endl;
        return 0;
    }

    ull decimal = convert_to_decimal(number, base);

    ull answer = 0;
    while (answer * base <= decimal) {
        ++answer;
    }

    cout << answer << endl;
    return 0;
}
