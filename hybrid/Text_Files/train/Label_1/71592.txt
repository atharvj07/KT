n, m = map(int, input().split())
colors = [0] * n

for _ in range(m):
    dancers = list(map(int, input().split()))
    for dancer in dancers:
        if colors[dancer - 1] == 0:
            if 1 not in [colors[d - 1] for d in dancers]:
                colors[dancer - 1] = 1
            elif 2 not in [colors[d - 1] for d in dancers]:
                colors[dancer - 1] = 2
            else:
                colors[dancer - 1] = 3

print(' '.join(map(str, colors)))
