#include <iostream>
#include <string>

using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int j = 0, ans = 0;
  for (int i = 0; i < a.size(); i++) {
    if (j < b.size() && a[i] == b[j]) {
      j++;
      continue;
    }
    ans++;
  }
  if (ans == a.size()) cout << "-";
  else cout << b.substr(ans);
  return 0;
}
