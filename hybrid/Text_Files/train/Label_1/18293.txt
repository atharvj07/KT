#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, z;
    cin >> n >> k >> z;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<vector<int>> dp(k + 1, vector<int>(z + 1, -1e9));
    dp[0][0] = a[0];

    for (int i = 1; i <= k; i++) {
      for (int j = 0; j <= z; j++) {
        if (j > 0) dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + a[i]);
        dp[i][j] = max(dp[i][j], dp[i - 1][j] + a[i]);
        if (i - 1 > 0) dp[i][j] = max(dp[i][j], dp[i - 1][j] + a[i - 1]);
      }
    }
    cout << dp[k][z] << endl;
  }
  return 0;
}
