import java.util.Scanner;
import java.util.Set;
import java.util.TreeSet;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        int p = scanner.nextInt();
        Set<Integer> levelsX = new TreeSet<>();
        for (int i = 0; i < p; i++) {
            levelsX.add(scanner.nextInt());
        }
        scanner.nextLine();
        int q = scanner.nextInt();
        Set<Integer> levelsY = new TreeSet<>();
        for (int i = 0; i < q; i++) {
            levelsY.add(scanner.nextInt());
        }
        Set<Integer> allLevels = new TreeSet<>();
        allLevels.addAll(levelsX);
        allLevels.addAll(levelsY);
        if (allLevels.size() == n) {
            System.out.println("I become the guy.");
        } else {
            System.out.println("Oh, my keyboard!");
        }
    }
}
