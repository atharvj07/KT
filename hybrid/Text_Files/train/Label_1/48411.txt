import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long ans = 0;
        for (int i = 1; i * i <= n; i++) {
            if (n % i == 0) {
                ans += i - 1 + (n / i - 1);
            }
        }
        System.out.println(ans);
    }
}
