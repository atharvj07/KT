#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const int mod = 1000000009;

int main() {
    int n;
    cin >> n;
    if (n == 1) {
        cout << 1 << endl;
        cout << "0 1" << endl;
        cout << 0 << endl;
        cout << "1" << endl;
        return 0;
    }
    if (n == 2) {
        cout << 2 << endl;
        cout << "-1 0 1" << endl;
        cout << 1 << endl;
        cout << "0 1" << endl;
        return 0;
    }
    vector<ll> p(n + 1), q(n);
    p[n] = 1;
    q[n - 1] = 1;
    for (int i = 0; i < n - 1; i++) p[i] = rand() % 3 - 1;
    for (int i = 0; i < n - 1; i++) q[i] = rand() % 3 - 1;
    while (n--) {
        ll t = rand() % 3 - 1;
        p[n] = t * q[n] + p[n + 1];
    }
    for (int i = n + 1; i >= 0; i--) cout << p[i] << " ";
    cout << endl;
    for (int i = n - 1; i >= 0; i--) cout << q[i] << " ";
    cout << endl;
}
