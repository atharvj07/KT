import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.PriorityQueue;

class Main {
    private static class Edge {
        int to;
        int w;

        public Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    private static class State implements Comparable<State> {
        int v;
        int d;

        public State(int v, int d) {
            this.v = v;
            this.d = d;
        }

        @Override
        public int compareTo(State other) {
            return Integer.compare(d, other.d);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int q = Integer.parseInt(parts[2]);
        List<List<Edge>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            parts = reader.readLine().split(" ");
            int a = Integer.parseInt(parts[0]);
            int b = Integer.parseInt(parts[1]);
            int w = Integer.parseInt(parts[2]);
            graph.get(a).add(new Edge(b, w));
        }
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[1] = 0;
        PriorityQueue<State> queue = new PriorityQueue<>();
        queue.add(new State(1, 0));
        while (!queue.isEmpty()) {
            State state = queue.poll();
            int v = state.v;
            int d = state.d;
            if (d > dist[v]) {
                continue;
            }
            for (Edge edge : graph.get(v)) {
                int to = edge.to;
                int w = edge.w;
                if (dist[v] + w < dist[to]) {
                    dist[to] = dist[v] + w;
                    queue.add(new State(to, dist[to]));
                }
            }
        }
        for (int i = 0; i < q; i++) {
            parts = reader.readLine().split(" ");
            int type = Integer.parseInt(parts[0]);
            int v = Integer.parseInt(parts[1]);
            if (type == 1) {
                if (dist[v] < Integer.MAX_VALUE) {
                    System.out.println(dist[v]);
                } else {
                    System.out.println(-1);
                }
            } else {
                for (int j = 2; j < parts.length; j++) {
                    int e = Integer.parseInt(parts[j]);
                    for (Edge edge : graph.get(e)) {
                        edge.w++;
                    }
                }
                Arrays.fill(dist, Integer.MAX_VALUE);
                dist[1] = 0;
                queue = new PriorityQueue<>();
                queue.add(new State(1, 0));
                while (!queue.isEmpty()) {
                    State state = queue.poll();
                    int v1 = state.v;
                    int d1 = state.d;
                    if (d1 > dist[v1]) {
                        continue;
                    }
                    for (Edge edge : graph.get(v1)) {
                        int to = edge.to;
                        int w = edge.w;
                        if (dist[v1] + w < dist[to]) {
                            dist[to] = dist[v1] + w;
                            queue.add(new State(to, dist[to]));
                        }
                    }
                }
            }
        }
    }
}
