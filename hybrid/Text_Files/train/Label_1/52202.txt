import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.StringTokenizer;
import java.util.stream.Collectors;
import java.util.stream.LongStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        List<long[]> islands = new ArrayList<>();
        long[][] width = new long[n][2];
        Set<Long> bridges = new HashSet<>();
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            width[i][0] = Long.parseLong(st.nextToken());
            width[i][1] = Long.parseLong(st.nextToken());
            islands.add(new long[]{width[i][0], width[i][1]});
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < m; i++) {
            bridges.add(Long.parseLong(st.nextToken()));
        }
        List<Long> ans = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            long curr = islands.get(i)[1] - islands.get(i)[0];
            long next = islands.get(i + 1)[0] - islands.get(i)[1];
            if (curr > 0 && next > 0) {
                if (bridges.stream().filter(s -> s > 0 && s <= curr + next).count() == 0) {
                    System.out.println("No");
                    return;
                }
                List<Long> validBridges = bridges.stream().filter(s -> s > 0 && s <= curr + next).sorted().collect(Collectors.toList());
                ans.add(validBridges.get(0));
                bridges.remove(validBridges.get(0));
                bridges.add(-validBridges.get(0));
            } else if (curr > 0) {
                if (bridges.stream().filter(s -> s > 0 && s <= curr).count() == 0) {
                    System.out.println("No");
                    return;
                }
                List<Long> validBridges = bridges.stream().filter(s -> s > 0 && s <= curr).sorted().collect(Collectors.toList());
                ans.add(validBridges.get(0));
                bridges.remove(validBridges.get(0));
            } else if (next > 0) {
                if (bridges.stream().filter(s -> s > 0 && s <= next).count() == 0) {
                    System.out.println("No");
                    return;
                }
                List<Long> validBridges = bridges.stream().filter(s -> s > 0 && s <= next).sorted().collect(Collectors.toList());
                ans.add(validBridges.get(0));
                bridges.remove(validBridges.get(0));
            }
        }
        System.out.println("Yes");
        System.out.println(ans.stream().map(Object::toString).collect(Collectors.joining(" ")));
    }
}
