#include<bits/stdc++.h>
#include<cassert>
#define pb push_back
#define sz size()
using namespace std;
int main(){
  int t; cin >> t;
  while(t--){
    string x, y; cin >> x >> y; int n = x.sz, m = y.sz;
    int ans = 0;
    reverse(x.begin(), x.end());
    reverse(y.begin(), y.end());
    int mx = n - 1, my = m - 1;
    while(mx >= 0 && my >= 0 && x[mx] == '1' && y[my] == '0'){
      --mx, --my;
      ++ans;
    }
    cout << ans << '\n';
  }
  return 0;
}
