n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

total_soda = sum(a)
b_sorted = sorted(b, reverse=True)

accumulated_volume = 0
k = 0
while accumulated_volume < total_soda:
    accumulated_volume += b_sorted[k]
    k += 1

t = 0
if k > 1:
    for i in range(k - 1):
        t += b_sorted[i]
    t += max(0, total_soda - sum(b_sorted[:k - 1]))

print(k, t)
