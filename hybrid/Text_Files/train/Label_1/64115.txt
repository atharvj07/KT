n, s = map(int, input().split())
v = list(map(int, input().split()))
v.sort()
if s > sum(v):
    print(-1)
else:
    ans = s
    for i in range(n):
        if s >= v[i]:
            s -= v[i]
            ans = min(ans, v[i])
        else:
            ans = min(ans, s)
            break
    print(ans)
