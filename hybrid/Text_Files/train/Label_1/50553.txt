import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int[] f = new int[26];
        for (char c : s.toCharArray()) {
            f[c - 'a']++;
        }
        int ans = 0;
        for (int i = 0; i < 26; i++) {
            if (f[i] > 0) {
                int x = f[i];
                while (x > 0) {
                    ans += x % 10 + 1;
                    x /= 10;
                }
            }
        }
        System.out.println(ans);
    }
}
