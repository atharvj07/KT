import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length(), ans = 0, cnt[] = new int[n + 1], st[] = new int[n + 2], en[] = new int[n + 2];
        for (int i = 0; i < n; i++) {
            cnt[i + 1] = cnt[i] + (s.charAt(i) - '0');
        }
        for (int i = 1; i <= n; i++) {
            if (cnt[i] % 3 == 0) {
                if (st[cnt[i]] == 0) {
                    st[cnt[i]] = en[cnt[i]] = i;
                } else {
                    en[cnt[i]] = i;
                }
            }
        }
        for (int i = 0; i <= cnt[n]; i++) {
            if (st[i] != 0) {
                ans += (en[i] - st[i]) / 3 * (en[i] - st[i]) / 3 * 3;
            }
        }
        System.out.println(ans);
    }
}
