n, m, k = map(int, input().split())
s = input()
t = input()

dp = [[[-1] * (k + 1) for _ in range(m + 1)] for _ in range(n + 1)]

def f(i, j, rem):
    if i == n or j == m:
        return 0 if rem == 0 else -1e9
    if dp[i][j][rem] != -1:
        return dp[i][j][rem]
    dp[i][j][rem] = f(i + 1, j, rem)
    if s[i] == t[j]:
        dp[i][j][rem] = max(dp[i][j][rem], f(i + 1, j + 1, rem - 1) + 1)
    return dp[i][j][rem]

print(f(0, 0, k))
