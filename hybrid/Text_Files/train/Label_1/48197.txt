import java.util.Scanner;

public class Main {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] intervals = new int[n][2];
        for (int i = 0; i < n; i++) {
            intervals[i][0] = sc.nextInt();
            intervals[i][1] = sc.nextInt();
        }
        int[] maxSouvenirs = new int[m];
        for (int d = 1; d <= m; d++) {
            int max = 0;
            for (int i = 0; i < n; i++) {
                if (intervals[i][0] <= d && d <= intervals[i][1]) {
                    max++;
                }
            }
            maxSouvenirs[d - 1] = max;
        }
        for (int max : maxSouvenirs) {
            System.out.println(max);
        }
    }
}
