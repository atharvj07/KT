#include <bits/stdc++.h>

using namespace std;

struct Item {
    int first;
    int second;
    int third;
};

int n;
vector<int> a;
vector<Item> dp;

void solve() {
    cin >> n;
    a.resize(n);
    dp.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    dp[0].first = 1;
    dp[0].second = 0;
    dp[0].third = 0;

    int max_len = 1;
    for (int i = 1; i < n; i++) {
        dp[i].first = 1;
        dp[i].second = 0;
        dp[i].third = 0;

        if (a[i] == a[i - 1]) {
            dp[i].first = max(dp[i].first, dp[i - 1].first + 1);
            dp[i].second = max(dp[i].second, dp[i - 1].second + 1);
        } else {
            if (dp[i - 1].second > 0) {
                dp[i].second = max(dp[i].second, dp[i - 1].second + 1);
                dp[i].third = dp[i - 1].first + 1;
                max_len = max(max_len, dp[i].third);
            }
        }
    }

    max_len = max(max_len, dp[n - 1].first);
    max_len = max(max_len, dp[n - 1].second);

    cout << max_len << endl;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
