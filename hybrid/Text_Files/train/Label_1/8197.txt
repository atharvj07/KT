#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> g(n);
  for (auto &i : g)cin >> i;
  int w[n + 4][m + 4];
  bool f = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (g[i][j] == '*') {
        memset(w, 0, sizeof w);
        w[i + 2][j + 2] = 1;
        for (int di = -2; di <= 2; di++)
          for (int dj = -2; dj <= 2; dj++)
            if (abs(di) + abs(dj) == 1) {
              int nx = i + di;
              int ny = j + dj;
              if (nx < 0 || ny < 0 || nx >= n || ny >= m) {
                continue;
              }
              if (g[nx][ny] == '*' && w[nx + 2][ny + 2] == 0) {
                f = 0;
                break;
              }
              if (g[nx][ny] == '.') {
                w[nx + 2][ny + 2] = -1;
              }
            }
      }
    }
  }
  if (!f) {
    return puts("-1"), 0;
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (w[i + 2][j + 2] == 1) {
        cout << 3 << endl;
        cout << i + 1 << " " << j + 1 << " " << 1 << endl;
        cout << i + 1 << " " << j + 1 << " " << 2 << endl;
        cout << i + 1 << " " << j + 1 << " " << 2 << endl;
        cnt += 3;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (g[i][j] == '*' && w[i + 2][j + 2] != 1) {
        cout << 1 << endl;
        cout << i + 1 << " " << j + 1 << " " << 1 << endl;
        cnt++;
      }
    }
  }
}
