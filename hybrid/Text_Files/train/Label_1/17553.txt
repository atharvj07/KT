#include <bits/stdc++.h>
typedef unsigned long long ll;
using namespace std;

ll pw(ll b, ll e) {
  if (e == 0) return (ll)1;
  ll ans = pw(b, e >> 1);
  ans = (ans * ans) % 1000000007;
  if (e % 2) ans = (ans * b) % 1000000007;
  return ans;
}

int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m;
  cin >> n >> m;
  ll ans;
  if (n % 2 == 0 && m % 2 == 0)
    ans = pw(4, (n / 2) * (m / 2)) % 1000000007;
  else if (n % 2 == 0)
    ans = pw(4, (n / 2) * (m / 2)) * 2 % 1000000007;
  else if (m % 2 == 0)
    ans = pw(4, (n / 2) * (m / 2)) * 2 % 1000000007;
  else
    ans = pw(4, (n / 2) * (m / 2)) * 4 % 1000000007;
  cout << ans << "\n";
  return 0;
}
