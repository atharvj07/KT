n, q, m = map(int, input().split())
a = list(map(int, input().split()))
c = [0] * (n + 6)
for _ in range(q):
    t, l, r = map(int, input().split())
    if t == 1:
        c[(l - 1) % n] += 1
        c[r % n] -= 1
    else:
        c[l - 1] += 1
        c[(r - 1) % n] -= 1
        c[(r) % n] += 1
ans = [a[0]]
for i in range(1, n):
    c[i] += c[(i - 1)]
    ans.append(a[(i - c[i] + n) % n])
for i in map(int, input().split()):
    print(ans[i - 1])
