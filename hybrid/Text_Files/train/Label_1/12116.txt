#include<bits/stdc++.h>
using namespace std;
int main() {
    int f, i, t, a, count = 0;
    cin >> f >> i >> t;
    while (f--) {
        a = 0;
        for (int j = 0; j < i; j++) {
            char x;
            cin >> x;
            a += (x == 'Y');
        }
        count += a >= t;
    }
    cout << count << endl;
    return 0;
}
