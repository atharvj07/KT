x = input().strip()

# Remove leading zeros and trailing zeros if it's a decimal
if '.' in x:
    x = x.rstrip('0')
x = x.lstrip('0')

# Handle case for pure 0 or after stripping it becomes empty which means it was 0
if x == '' or x == '.':
    print('0')
    exit()

# Extract pre-decimal and post-decimal parts
if '.' in x:
    pre_decimal, post_decimal = x.split('.')
else:
    pre_decimal, post_decimal = x, ''

# Calculate exponent
if pre_decimal:  # number is greater than or equal to 1
    exponent = len(pre_decimal) - 1
else:  # number is less than 1
    exponent = -len(post_decimal.lstrip('0')) - 1
    pre_decimal = post_decimal.lstrip('0')[0]
    post_decimal = post_decimal.lstrip('0')[1:]

# Construct the number 'a'
if post_decimal:
    a = pre_decimal + '.' + post_decimal
else:
    a = pre_decimal

# Remove trailing .0
if a.endswith('.0'):
    a = a[:-2]

# Construct final representation
if exponent == 0:
    print(a)
else:
    print(f"{a}E{exponent}")
