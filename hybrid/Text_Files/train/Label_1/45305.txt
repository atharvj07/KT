import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String trump = scanner.nextLine();
        char trumpChar = trump.charAt(0);
        String[] cards = scanner.nextLine().split(" ");
        char card1Rank = cards[0].charAt(0);
        char card1Suit = cards[0].charAt(1);
        char card2Rank = cards[1].charAt(0);
        char card2Suit = cards[1].charAt(1);
        if (card1Suit == card2Suit) {
            if (card1Rank > card2Rank) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        } else {
            if (card1Suit == trumpChar) {
                System.out.println("YES");
            } else if (card2Suit == trumpChar) {
                System.out.println("NO");
            } else {
                System.out.println("NO");
            }
        }
    }
}
