import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while(t-- > 0) {
            int n = in.nextInt();
            char[] str = in.next().toCharArray();
            int tcnt = 0,
                mcnt = 0;
            for(int i = 0; i < n; ++i) {
                if(str[i] == 'T') {
                    if(mcnt != 0) {
                        tcnt = -1;
                        break;
                    }
                    ++tcnt;
                }
                else if(str[i] == 'M') {
                    if(tcnt <= 0 || tcnt != mcnt) {
                        tcnt = -1;
                        break;
                    }
                    ++mcnt;
                }
            }
            if(tcnt > 0 && tcnt == mcnt && tcnt * 3 == n) {
                System.out.println("YES");
            }
            else {
                System.out.println("NO");
            }
        }
    }
}
