import java.util.Arrays;
import java.util.Queue;
import java.util.Scanner;
import java.util.LinkedList;
import java.util.HashMap;

class Main {
    static int n, m, count = 0, ans = 0;
    static boolean[] in = new boolean[100000];
    static HashMap<Integer, Integer> hm = new HashMap<>();
    static Queue<Integer> q = new LinkedList<>();
    static Box[] boxes;

    static class Box {
        int size;
        int[] presents;

        public Box(int[] presents) {
            this.presents = presents;
            size = presents.length;
        }

        public boolean add(int kind) {
            for (int i = 0; i < size; i++)
                if (presents[i] == kind)
                    return false;
            int[] presents1 = Arrays.copyOf(presents, size + 1);
            presents1[size++] = kind;
            presents = presents1;
            return true;
        }

        public int remove(int kind) {
            for (int i = 0; i < size; i++)
                if (presents[i] == kind)  {
                    presents[i] = presents[--size];
                    return i;
                }
            return -1;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        int[][] presents = new int[n][];
        boxes = new Box[n];
        for (int i = 0; i < n; i++) {
            int size = scanner.nextInt();
            presents[i] = new int[size];
            for (int j = 0; j < size; j++) {
                int kind = scanner.nextInt();
                presents[i][j] = kind;
                if (!in[kind])  {
                    in[kind] = true;
                    q.add(kind);
                }
                hm.put(kind, hm.getOrDefault(kind, 0) + 1);
            }
            boxes[i] = new Box(presents[i]);
        }
        int bal = 0, sum = 0;
        for (Integer kind : hm.keySet())
            if (hm.get(kind) >= n)
                bal++;
            else
                sum += hm.get(kind);
        if (n - bal - sum > 0) {
            System.out.print("NO");
            return;
        }
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = boxes[i].size;
        Arrays.sort(a);
        int min = a[0], max = a[n - 1];
        if (max - min > 1) {
            System.out.print("NO");
            return;
        }
        while (!q.isEmpty()) {
            int kind = q.poll();
            if (hm.get(kind) == n)
                continue;
            int need = n - hm.get(kind);
            a = new int[n];
            for (int i = 0; i < n; i++)
                a[i] = boxes[i].size;
            Arrays.sort(a);
            if (min + need > max) {
                System.out.print("NO");
                return;
            }
            for (int i = 0; i < n; i++) {
                if (!boxes[i].add(kind))
                    continue;
                count++;
                ans++;
                System.out.println(i + " " + n + " " + kind);
                hm.put(kind, hm.get(kind) + 1);
                a = new int[n];
                for (int j = 0; j < n; j++)
                    a[j] = boxes[j].size;
                Arrays.sort(a);
                min = a[0];
                max = a[n - 1];
                if (max - min <= 1)
                    break;
                else
                    q.add(kind);
            }
        }
        System.out.println(ans);
    }
}
