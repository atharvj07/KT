import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] p = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] q = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int m = Integer.parseInt(br.readLine());

        int[] ip = new int[n + 1];
        int[] iq = new int[n + 1];
        for (int i = 0; i < n; ++i) {
            ip[p[i]] = i + 1;
            iq[q[i]] = i + 1;
        }

        int x = 0;
        for (int i = 0; i < m; ++i) {
            int[] arr = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int a = arr[0], b = arr[1], c = arr[2], d = arr[3];
            x = ((x + f(a, b, n, x)) % n + n) % n;
            int l1 = Math.min(f(a, b, n, x), f(c, d, n, x));
            int r1 = Math.max(f(a, b, n, x), f(c, d, n, x));
            int l2 = Math.min(ip[c], ip[d]);
            int r2 = Math.max(ip[c], ip[d]);
            int ans = (r1 - l1 + 1) * (r2 - l2 + 1);
            System.out.println(ans);
        }
    }

    private static int f(int x, int y, int n, int z) {
        return ((x - 1 + z) % n + n) % n + 1;
    }
}
