import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long gcd = 0;
        boolean setwise = false;
        for (int i = 0; i < n; i++) {
            long a = sc.nextInt();
            gcd = gcd(gcd, a);
            if (i > 0 && gcd == 1) {
                setwise = true;
            }
        }
        if (gcd == 1) {
            System.out.println(setwise ? "setwise coprime" : "pairwise coprime");
        } else {
            System.out.println("not coprime");
        }
    }

    private static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
