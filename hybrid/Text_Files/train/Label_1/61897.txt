import sys
input = sys.stdin.readline
n, m, ta, tb, k = map(int, input().split())
a = sorted([int(i) for i in input().split()])
b = sorted([int(i) for i in input().split()])
ok = 1
ans = 1e9
for i in range(n):
    l = i+1
    r = m
    while l <= r:
        mid = (l+r)//2
        if b[mid] >= a[i]+ta:
            ans = min(ans, b[mid]+tb)
            l = mid+1
        else:
            r = mid-1
    if i==0:
        ok &= (ans-ta-tb <= k)
if not ok:
    print(-1)
else:
    print(ans)
