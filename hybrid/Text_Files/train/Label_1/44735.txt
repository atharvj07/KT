import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;

public class Main {
    private static long[] parent;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] nk = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int n = nk[0], q = nk[1];
        long[][] edges = new long[q][3];
        for (int i = 0; i < q; i++) {
            edges[i] = Arrays.stream(br.readLine().split(" ")).mapToLong(Long::parseLong).toArray();
        }
        Arrays.sort(edges, Comparator.comparingLong(a -> a[0]));

        long ans = 0;
        parent = new long[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }
        for (long[] edge : edges) {
            ans += edge[2];
            union(edge[0], edge[1]);
        }
        System.out.println(ans);
    }

    private static void union(long a, long b) {
        long pA = find(a);
        long pB = find(b);
        if (pA != pB) {
            parent[(int) pA] = pB;
        }
    }

    private static long find(long a) {
        if (parent[(int) a] == a) {
            return a;
        }
        return find(parent[(int) a]);
    }
}
