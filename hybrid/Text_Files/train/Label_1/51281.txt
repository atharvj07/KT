import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        int[] cntA = new int[m];
        int[] cntB = new int[m];
        for (int i = 0; i < n; i++) {
            cntA[a[i]]++;
            cntB[b[i]]++;
        }
        int ans = -1;
        for (int i = 0; i < m; i++) {
            if (cntA[i] != cntB[i]) {
                ans = -1;
                break;
            }
            if (ans == -1) {
                ans = i;
            } else {
                ans = Math.min(ans, i);
            }
        }
        System.out.println(ans);
    }
}
