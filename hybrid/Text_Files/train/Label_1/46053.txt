import java.util.*;

class Main {
    static int[] p;
    static int[] cnt;
    static int n;

    static int find(int x) {
        if (p[x] != x) p[x] = find(p[x]);
        return p[x];
    }

    static void unite(int x, int y) {
        p[find(y)] = find(x);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        String s = sc.next();
        cnt = new int[n];
        p = new int[n];
        for (int i = 0; i < n; i++) p[i] = i;
        for (int i = 0; i < n; i++) {
            cnt[i] = s.charAt(i) - '0';
            if (cnt[i] == 1) {
                unite(i, 0);
            }
        }
        int v = 0;
        for (int i = 1; i < n; i++) {
            if (cnt[i] == 1) {
                v = find(i);
                break;
            }
        }
        for (int i = 0; i < n; i++) {
            if (find(i) != v) {
                System.out.println(-1);
                return;
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (cnt[i] > 0) {
                ans += cnt[i] - 1;
            }
        }
        System.out.println(ans);
    }
}
