def count_good_substrings(s, good_chars, k):
    good = [1 if good_chars[ord(c)-ord('a')] == '1' else 0 for c in s]
    substrings = set()
    for start in range(len(s)):
        bad_count = 0
        for end in range(start, len(s)):
            if not good[end]:
                bad_count += 1
            if bad_count > k:
                break
            substrings.add(s[start:end+1])
    return len(substrings)

s = input().strip()
good_chars = input().strip()
k = int(input().strip())

print(count_good_substrings(s, good_chars, k))
