#include <cstdio>
#include <algorithm>

using namespace std;

typedef long long i64;
const int kMaxN = 1e5 + 10;

int main() {
  int N, D, M;
  scanf("%d%d%d", &N, &D, &M);
  i64 A[kMaxN], sm[kMaxN];
  for (int i = 1; i <= N; ++i) { scanf("%lld", A + i); }
  sort(A + 1, A + N + 1);
  for (int i = 1; i <= N; ++i) { sm[i] = sm[i - 1] + A[i]; }
  i64 Ans = 0;
  for (int i = D + 1; i <= N; ++i) {
    i64 R = min((i64)(i + D), (i64)N);
    int K = upper_bound(A + 1, A + N + 1, M) - (A + 1);
    if (K > R - i + 1) { Ans = sm[N]; break; }
    Ans = max(Ans, sm[R] - sm[i - 1]);
  }
  printf("%lld\n", Ans);
  return 0;
}
