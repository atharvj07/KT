import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++)
                a[i] = sc.nextLong();
            long min = 0, max = 0;
            for (long v : a) {
                min += (v + x - 1) / x;
                max += (v + x - 1) / x;
            }
            int i = 0;
            while (i < n - 1) {
                long v1 = a[i], v2 = a[i + 1];
                long sum = (v1 + v2 + x - 1) / x;
                if (sum < v1 + v2) {
                    min += sum - v1 - v2;
                    max += sum - v1 - v2;
                    i += 2;
                } else
                    i++;
            }
            System.out.println(min + " " + max);
        }
    }
}
