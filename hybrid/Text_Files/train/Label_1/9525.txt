#include <bits/stdc++.h>
using namespace std;
bool lucky(string s, int l) {
  vector<int> sum(l);
  sum[0] = s[0] - '0';
  for (int i = 1; i < l; i++) sum[i] = sum[i - 1] + s[i] - '0';
  for (int i = 0; i < l - 1; i++) {
    if (sum[i] == sum[l - 1] - sum[i]) return true;
    if (i > 0 && sum[i - 1] == sum[l - 1] - sum[i]) return true;
  }
  return false;
}
int main() {
  int n;
  string s;
  cin >> n >> s;
  cout << (lucky(s, n) ? "YES" : "NO");
}
