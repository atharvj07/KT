import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] x = new int[n];
        String[] line = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            x[i] = Integer.parseInt(line[i]);
        }
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int diff = Math.abs(x[i] - x[j]);
                int pow = 0;
                while (diff % 2 == 0) {
                    diff /= 2;
                    pow++;
                }
                if (diff == 1) {
                    map.computeIfAbsent(pow, k -> new ArrayList<>()).add(i);
                    map.computeIfAbsent(pow, k -> new ArrayList<>()).add(j);
                }
            }
        }
        int maxPow = 0;
        List<Integer> maxList = new ArrayList<>();
        for (int pow : map.keySet()) {
            if (map.get(pow).size() > maxList.size()) {
                maxPow = pow;
                maxList = map.get(pow);
            }
        }
        System.out.println(maxList.size());
        for (int idx : maxList) {
            System.out.print(x[idx] + " ");
        }
        System.out.println();
    }
}
