#include <bits/stdc++.h>
#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;

    vector<int> cnta(26, 0), cntb(26, 0);
    for (int i = 0; i < n; i++) cnta[a[i] - 'a']++, cntb[b[i] - 'a']++;

    int ans = 0;
    for (int i = 0; i < 26; i++) ans += abs(cnta[i] - cntb[i]);

    cout << ans / 2 << endl;

    return 0;
}
