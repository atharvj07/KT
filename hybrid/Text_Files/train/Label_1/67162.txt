import sys

def find(p, x):
    if p[x] == x:
        return x
    return find(p, p[x])

def union(p, a, b):
    pa = find(p, a)
    pb = find(p, b)
    p[pa] = pb

n = int(input())
edges = [list(map(int, input().split())) for _ in range(n - 1)]
edges.sort(key=lambda x: x[1] - x[0], reverse=True)

p = list(range(n + 1))
for a, b in edges:
    if find(p, a) != find(p, b):
        union(p, a, b)
        print(a, b)
    else:
        print("NO")
        sys.exit()
