#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<pair<int, int>> coupons(n);
    for (int i = 0; i < n; i++) {
        int l, r;
        cin >> l >> r;
        coupons[i] = {l, r};
    }
    sort(coupons.begin(), coupons.end(), [](pair<int, int>& a, pair<int, int>& b) {
        return a.second < b.second;
    });
    int max_products = 0, max_products_l;
    vector<pair<int, int>> ans_coupons;
    for (int i = 0; i < k; i++) {
        if (i == 0) {
            max_products = coupons[0].second - coupons[0].first + 1;
            max_products_l = coupons[0].first;
            ans_coupons.push_back(coupons[0]);
        } else {
            if (coupons[i].first > max_products_l) {
                max_products = coupons[i].second - coupons[i].first + 1;
                max_products_l = coupons[i].first;
                ans_coupons.clear();
                ans_coupons.push_back(coupons[i]);
            } else {
                int curr_products = max_products;
                for (int j = 0; j < i; j++) {
                    if (coupons[i].first <= ans_coupons[j].second) {
                        curr_products -= ans_coupons[j].second - coupons[i].first + 1;
                    }
                }
                if (curr_products > max_products) {
                    max_products = curr_products;
                    max_products_l = coupons[i].first;
                    ans_coupons.clear();
                    ans_coupons.push_back(coupons[i]);
                }
            }
        }
    }
    cout << max_products << endl;
    for (int i = 0; i < k; i++) {
        cout << ans_coupons[i].first << ' ';
    }
    cout << endl;
    return 0;
}
