#include <cstdio>
#include <vector>
const int mod=998244353;
char a[300005];
int b[300005];
int f[300005];
int fb[300005];
int fc[300005];
std::vector<std::pair<int,int>> d;
int main(){
	int n,m;
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;i++){
		scanf("%s",a+1);
		int lst=0;
		for(int j=1;j<=m;j++)if(a[j]=='o')lst=j;
		b[i]=lst;
		if(a[m]=='o')continue;
		for(int j=m;j>=1;j--)if(a[j]=='o'){
			d.push_back(std::make_pair(i,j));
			break;
		}
	}
	f[0]=1;
	for(int i=1;i<=n;i++){
		f[i]=f[i-1];
		if(b[i]==m)f[i]=(f[i]+f[i-1])%mod;
	}
	fb[n+1]=1;
	for(int i=n;i>=1;i--){
		fb[i]=fb[i+1];
		if(b[i])fb[i]=(fb[i]+fb[i+1])%mod;
	}
	fc[d.size()]=1;
	for(int i=d.size()-1;i>=1;i--){
		fc[i]=fc[i+1];
		if(d[i].second!=m)fc[i]=(fc[i]+fc[i+1])%mod;
	}
	long long ans=0;
	for(int i=0;i<d.size();i++){
		ans=(ans+(long long)f[d[i].first-1]*fb[d[i].first+1]*fc[i+1])%mod;
	}
	printf("%lld",ans);
	return 0;
}
