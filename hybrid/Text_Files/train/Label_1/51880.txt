import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        input.nextLine();
        Map<String, String> variableMap = new HashMap<>();
        Set<String> usedVariables = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String originalVariable = input.nextLine();
            String newVariable = input.nextLine();
            if (variableMap.containsKey(originalVariable) && !variableMap.get(originalVariable).equals(newVariable)) {
                System.out.println("NO");
                return;
            }
            if (usedVariables.contains(newVariable)) {
                System.out.println("NO");
                return;
            }
            variableMap.put(originalVariable, newVariable);
            usedVariables.add(newVariable);
        }
        System.out.println("YES");
        String s = "";
        String t = "";
        for (Map.Entry<String, String> entry: variableMap.entrySet()) {
            if (!entry.getKey().equals(entry.getValue())) {
                s = entry.getKey();
                t = entry.getValue();
                break;
            }
        }
        System.out.println(s);
        System.out.println(t);
    }
}
