import java.math.BigInteger;
import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] graph = new long[n][n];
        // Initialize graph with all distances as infinite
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) graph[i][j] = 0;
                else graph[i][j] = 999999999;
            }
        }
        for (int i = 0; i < m; i++) {
            int l = sc.nextInt() - 1, r = sc.nextInt() - 1, c = sc.nextInt();
            for (int x = l; x < r; x++) {
                for (int y = x + 1; y < r; y++) {
                    graph[x][y] = Math.min(graph[x][y], c);
                    graph[y][x] = graph[x][y];
                }
            }
        }
        floyd(graph, n);
        if (graph[0][n - 1] != 999999999) {
            System.out.println(graph[0][n - 1]);
        } else {
            System.out.println(-1);
        }
    }

    private static void floyd(long[][] graph, int n) {
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    graph[i][j] = Math.min(graph[i][j], graph[i][k] + graph[k][j]);
                }
            }
        }
    }
}
