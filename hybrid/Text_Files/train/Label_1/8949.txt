#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  bool alex, bob, carl;
  alex = bob = carl = true;
  while (n--) {
    int a;
    cin >> a;
    if (a == 1) {
      if (bob) bob = false;
      else if (carl) carl = false;
    } else if (a == 2) {
      if (alex) alex = false;
      else if (carl) carl = false;
    } else {
      if (alex) alex = false;
      else if (bob) bob = false;
    }
  }
  if (alex || bob || carl)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
