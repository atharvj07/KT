#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k, ans1 = 0, ans2 = 0, cnt = 0;
    cin >> n >> k;
    vector<string> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    string s;
    cin >> s;
    sort(v.begin(), v.end());
    for (int i = 0; i < n; i++) {
        ans1++;
        if (v[i] == s) break;
    }
    for (int i = 0; i < n; i++) {
        ans2++;
        if (v[i] == s) break;
        cnt++;
        if (cnt == k) {
            ans2 += 5;
            cnt = 0;
        }
    }
    cout << ans1 << ' ' << ans2 << endl;
}
