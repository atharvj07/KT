n = int(input())
points = []
for _ in range(n):
    x, w = map(int, input().split())
    points.append((x, w))

points.sort(key=lambda x: x[0] + x[1])

max_clique_size = 0
last_endpoint = -float('inf')
for point in points:
    if point[0] - point[1] >= last_endpoint:
        max_clique_size += 1
        last_endpoint = point[0] + point[1]

print(max_clique_size)
