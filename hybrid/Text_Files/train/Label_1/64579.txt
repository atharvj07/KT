import sys
prefix_xor = []
for line in sys.stdin.readlines():
    x = int(line)
    h = -1
    for i in range(len(prefix_xor)):
        if x == prefix_xor[-1] ^ prefix_xor[i]:
            h = i
            break
    if h == -1:
        prefix_xor.append(x)
        print(0)
    else:
        print(2, h + 1, len(prefix_xor))
        prefix_xor.append(x)
