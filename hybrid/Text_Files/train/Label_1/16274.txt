#include <iostream>
#include <string>
using namespace std;
int main() {
  string s;
  cin >> s;
  char c = s[0];
  int r = s[1] - '0';
  int cnt = 0;
  if (r > 1) cnt++;
  if (r < 8) cnt++;
  if (c > 'a') cnt++;
  if (c < 'h') cnt++;
  if (c > 'a' && r < 8) cnt++;
  if (c < 'h' && r > 1) cnt++;
  if (c > 'a' && r > 1) cnt++;
  if (c < 'h' && r < 8) cnt++;
  cout << cnt << endl;
  return 0;
}
