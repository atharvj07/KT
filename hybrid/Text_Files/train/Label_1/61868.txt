n, m = map(int, input().split())
a = list(map(int, input().split()))
b = [0] * (m)
for x in a:
    b[x % m] = 1
if 0 in b:
    print("NO")
elif b[m // 2] and m % 2 == 0:
    print("YES")
else:
    for i in range(1, (m + 1) // 2):
        if b[i] and b[m - i]:
            print("YES")
            break
    else:
        print("NO")
