import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[][] bc = new int[q][2];
        for (int i = 0; i < q; i++) {
            bc[i][0] = sc.nextInt();
            bc[i][1] = sc.nextInt();
        }
        long[] s = new long[q];
        for (int i = 0; i < q; i++) {
            long sum = 0;
            for (int j = 0; j < n; j++) {
                if (a[j] == bc[i][0]) {
                    sum += bc[i][1];
                } else {
                    sum += a[j];
                }
            }
            s[i] = sum;
        }
        for (long i : s) {
            System.out.println(i);
        }
    }
}
