import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] c = sc.next().toCharArray();
        for (int i = 0; i < c.length; i++) {
            System.out.print((char) ('A' + (c[i] + n - 'A') % 26));
        }
    }
}
