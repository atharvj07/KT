#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
#define p 1000000007

ll power(ll x,ll y){
    x %= p;
    ll ret = 1;
    while(y > 0){
        if(y & 1) ret = (ret * x) % p;
        x = (x * x) % p;
        y >>= 1;
    }
    return ret;
}

int main()
{
    ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    string s; cin >> s;
    vector<set<int>> v(26);
    int n; cin >> n;
    while(n--){
        char op; int x; cin >> op >> x, --x;
        if(op == '+'){
            char ch; cin >> ch;
            v[ch - 'a'].insert(x);
        }else v[s[x] - 'a'].erase(x);
        int ans = 0;
        for(char ch = 'a'; ch <= 'z'; ++ch){
            int l = 0, r = 0, cnt = 0;
            for(int pos : v[ch - 'a']){
                if(pos < l) ++cnt;
                else if(pos <= r) ++cnt, l = r + 1;
                else l = r = pos, ++cnt, ++r;
            }
            ans += (cnt == 1);
        }
        cout << (ans == 3 ? "YES" : "NO") << endl;
    }
    return 0;
}
