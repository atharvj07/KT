import sys

n = int(input())
a = list(map(int, input().split()))

inc = [False] * n
dec = [False] * n

if a[0] >= a[1]:
    inc[0] = True
else:
    dec[0] = True

for i in range(1, n):
    if a[i] < a[i - 1]:
        if dec[i - 1]:
            sys.stdout.write("NO")
            sys.exit()
        dec[i] = True
    if a[i] > a[i - 1]:
        if inc[i - 1]:
            sys.stdout.write("NO")
            sys.exit()
        inc[i] = True

sys.stdout.write("YES\n")
for i in range(n):
    if inc[i]:
        sys.stdout.write("0 ")
    else:
        sys.stdout.write("1 ")
