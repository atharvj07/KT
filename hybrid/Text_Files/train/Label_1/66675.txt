for _ in range(int(input())):
    n = int(input())
    p = [input() for _ in range(n)]
    d = [{} for _ in range(n)]
    for i in range(n):
        for j in range(4):
            for k in range(10):
                if k != int(p[i][j]):
                    p2 = p[i]
                    p2 = p2[:j] + str(k) + p2[j + 1:]
                    if p2 not in d[i]:
                        d[i][p2] = True
    k = 0
    while n > 1:
        for i in range(n):
            if len(d[i]) == 0:
                continue
            k += 1
            p[i] = d[i].popitem()[0]
            for j in range(n):
                if i == j:
                    continue
                while p[i] in d[j]:
                    d[j].pop(p[i])
            n -= 1
            break
    print(k)
    for s in p:
        print(s)
