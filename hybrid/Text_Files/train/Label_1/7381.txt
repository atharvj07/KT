#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, l, r, Ql, Qr;
  cin >> n >> l >> r >> Ql >> Qr;
  vector<int> w(n);
  for (int i = 0; i < n; i++) cin >> w[i];
  sort(w.begin(), w.end());
  long long ans = 0;
  int i = 0, j = n - 1, lh = l, rh = r;
  while (i <= j) {
    if (lh * w[i] <= rh * w[j]) {
      ans += lh * w[i];
      if (i == j) i++, j--;
      else i++;
      lh += Ql;
    } else {
      ans += rh * w[j];
      j--;
      rh += Qr;
    }
  }
  cout << ans << endl;
}
