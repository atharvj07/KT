#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N = 150005;
ll t[N];
ll onl[N];
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  int n, k, q;
  cin >> n >> k >> q;
  for (int i = 1; i <= n; i++) { cin >> t[i]; }
  while (q--) {
    int ty, x;
    cin >> ty >> x;
    if (ty == 1) {
      if (onl[x] == 0) {
        onl[x] = 1;
        priority_queue<int> pq;
        for (int i = 1; i <= n; i++) {
          if (onl[i]) pq.push(t[i]);
        }
        for (int i = 0; i < k && !pq.empty(); i++) {
          int mx = pq.top();
          pq.pop();
          if (t[x] == mx) onl[x] = 2;
        }
      }
    } else {
      if (onl[x] == 1 || onl[x] == 2) {
        cout << "YES" << endl;
      } else {
        cout << "NO" << endl;
      }
    }
  }
}
