#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<long long> x(n + m), t(n + m);
  for (int i = 0; i < n + m; i++) cin >> x[i];
  for (int i = 0; i < n + m; i++) cin >> t[i];

  vector<vector<int>> id(m);
  for (int i = 0; i < n + m; i++)
    if (t[i]) id[t[i] - 1].push_back(i);

  vector<int> ans(m, 0);
  vector<int> l(n + m, 0), r(n + m, 0);
  for (int i = 0; i < m; i++) {
    for (auto j : id[i]) {
      l[j] = i ? r[id[i - 1].back()] : -1;
      r[j] = id[i].back();
    }
  }

  for (int i = 0; i < n + m; i++) {
    if (t[i] == 0) {
      int a = l[i], b = r[i];
      ans[lower_bound(x.begin() + a + 1, x.begin() + b + 1, x[i]) - x.begin() - 1]++;
    }
  }

  for (int i = 0; i < m; i++) cout << ans[i] << " ";
  cout << endl;

  return 0;
}
