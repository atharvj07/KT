#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    bool odd = false;
    unordered_map<char, int> freq;
    for (int i = 0; i < n; i++) {
        freq[s[i]]++;
        if (freq[s[i]] % 2) odd = !odd;
    }
    if (odd) {
        char odd_char;
        for (auto &p : freq) {
            if (p.second % 2) {
                odd_char = p.first;
                break;
            }
        }
        int idx = -1;
        for (int i = 0; i < n; i++) {
            if (s[i] == odd_char) {
                idx = i;
                break;
            }
        }
        if (idx == -1) {
            cout << "NA" << endl;
            return 0;
        }
        for (int i = 0; i <= idx; i++) {
            cout << s[i];
        }
        for (int i = idx - 1; i >= 0; i--) {
            cout << s[i];
        }
        cout << endl;
    } else {
        for (int i = 0; i < n; i++) {
            cout << s[i];
        }
        for (int i = n - 1; i >= 0; i--) {
            cout << s[i];
        }
        cout << endl;
    }
    return 0;
}
