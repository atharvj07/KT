def beautiful_string(s):
    st = list(s)
    for i in range(1, len(st)):
        if st[i - 1] == st[i] or (st[i - 1] == '?' and st[i] != '?'):
            st[i - 1] = 'a' if st[i] in ['b', 'c'] else 'b'
    if st[0] == st[1]:
        st[0] = 'a' if st[1] in ['b', 'c'] else 'b'
    for i in range(1, len(st)):
        if st[i - 1] == st[i]:
            return -1
    return ''.join(st).replace('?', 'c')
for _ in range(int(input())):
    print(beautiful_string(input()))
