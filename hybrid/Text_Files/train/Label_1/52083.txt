import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] v = new int[5], p = new int[5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                int t = in.nextInt();
                if (i == 0) v[j] = t;
                else if (i == 1) p[j] = t;
            }
        }
        int ans = -1;
        for (int i = 0; i < 5; i++) {
            if (v[i] == -1) continue;
            int max = 1000 + (1000 * (n - 1 - (p[i] == -1 ? 0 : 1))) / n;
            int now = max - (max * v[i] / 250);
            if (p[i] == -1 || now > max - (max * p[i] / 250)) {
                ans = 0;
                break;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
