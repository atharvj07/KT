#include <bits/stdc++.h>
using namespace std;
const int maxn = 5005;

int n, a[maxn], vis[maxn], ans;
vector<int> g[maxn];

void dfs(int u, int c, int res) {
  if (a[u] != c) res = a[u];
  ans = max(ans, res);
  for (int v : g[u]) {
    if (!vis[v]) { vis[v] = 1; dfs(v, c, res ^ a[u]); }
  }
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    g[a[i]].push_back(i);
  }
  for (int i = 1; i <= n; ++i) {
    if (!vis[i]) { vis[i] = 1; dfs(i, i, 0); }
  }
  cout << ans << endl;
}
