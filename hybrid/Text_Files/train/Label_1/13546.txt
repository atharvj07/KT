#include <bits/stdc++.h>
#define int long long
using namespace std;
int a[200007];
map<int, int> freq;
int n, m;
int32_t main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    freq[a[i]]++;
  }
  for (int i = 0; i < m; i++) {
    int x, l, r;
    cin >> l >> r >> x;
    if (freq[x] <= 0 || freq[x] > (r - l + 1)) {
      cout << -1 << endl;
      continue;
    }
    for (int j = l; j <= r; j++) {
      if (a[j] != x) {
        cout << j << endl;
        break;
      }
    }
  }
}
