import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int m = 0;
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (a[i] == 1) {
                b[m++] = i;
            }
        }
        if (m % 2 == 1) {
            System.out.println("-1");
        } else {
            System.out.println(m);
            for (int i = 0; i < m; i++) {
                System.out.print(b[i] + " ");
            }
            System.out.println();
        }
    }
}
