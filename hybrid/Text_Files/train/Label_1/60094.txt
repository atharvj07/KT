t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    grid = [list(input().strip()) for _ in range(n)]

    def can_color(start_color):
        temp_grid = [row[:] for row in grid]
        for i in range(n):
            for j in range(m):
                if temp_grid[i][j] == '.':
                    temp_grid[i][j] = 'R' if (i + j + start_color) % 2 == 0 else 'W'
                elif (i + j + start_color) % 2 == 0 and temp_grid[i][j] == 'W':
                    return False
                elif (i + j + start_color) % 2 == 1 and temp_grid[i][j] == 'R':
                    return False
        return temp_grid

    result = can_color(0) or can_color(1)
    if result:
        print("YES")
        for row in result:
            print(''.join(row))
    else:
        print("NO")
