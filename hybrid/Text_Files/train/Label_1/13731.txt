#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int n, m;
    cin >> n >> m;
    
    vector<vector<char>> ans(n, vector<char>(m));
    vector<int> w(m);
    
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> ans[i][j];
            
    for (int i = 0; i < m; i++)
        cin >> w[i];
        
    int res = 0;
    
    for (int j = 0; j < m; j++)
    {
        vector<int> cnt(5);
        for (int i = 0; i < n; i++)
            cnt[ans[i][j] - 'A']++;
        
        res += w[j] * *max_element(cnt.begin(), cnt.end());
    }
    
    cout << res << '\n';
    return 0;
}
