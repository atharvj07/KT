#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    if (k == 1) {
        cout << 0 << '\n';
        return 0;
    }
    if (k == n) {
        cout << 0 << '\n';
        return 0;
    }
    int suffmin[n], suffmax[n];

    suffmax[n - 1] = arr[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        suffmax[i] = max(suffmax[i + 1], arr[i]);
    }
    suffmin[n - 1] = arr[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        suffmin[i] = min(suffmin[i + 1], arr[i]);
    }

    int ans = 1e9;
    for (int i = 0; i + k <= n; i++) {
        int maxi = suffmax[i];
        int mini = suffmin[i + k - 1];
        if (ans > maxi - mini) { ans = maxi - mini; }
    }
    cout << ans << '\n';

    return 0;
}
