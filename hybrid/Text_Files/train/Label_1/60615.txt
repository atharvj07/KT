from itertools import combinations

n,k=map(int,input().split())
a=list(map(int,input().split()))
assert len(a)==n
a.sort()
ans=[]
for _ in range(k):
    cur=[a.pop()]
    for p in range(1, min(len(a),n)):
        for c in combinations(a,p):
            if sum(c)!=sum(cur):
                cur=list(c)
                break
    ans.append((len(cur), *cur))
for k in ans:
    print(*k)
