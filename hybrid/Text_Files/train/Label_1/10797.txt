#include <bits/stdc++.h>
using namespace std;
const int maxn = 105;
int dp[31][maxn*maxn];
int main() {
    int n, m;
    cin >> n >> m;
    int a[maxn], b[maxn];
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= m; i++) cin >> b[i];
    int ans = 0;
    for(int i = 1; i <= n; i++) dp[0][i] = 0;
    for(int i = 1; i <= m; i++) {
        for(int j = 1; j <= n*maxn; j++) {
            dp[i][j] = dp[i-1][j];
            int id = b[i];
            if(j - a[id] >= 0) dp[i][j] = max(dp[i][j], dp[i-1][j-a[id]] + a[id]);
        }
    }
    for(int i = 1; i <= n*maxn; i++) {
        ans = max(ans, dp[m][i]);
    }
    cout << ans << endl;
    return 0;
}
