#include <bits/stdc++.h>
using namespace std;

#define pb push_back
const int N = 1e5 + 10, M = 1e5 + 10;
vector<int> adj[N], radj[N], topo;
int vis[N], d[N], r[N], g[N];
void dfs1(int u) {
  vis[u] = -1;
  for (int v : adj[u]) {
    if (!vis[v]) dfs1(v);
  }
  topo.pb(u);
}
void dfs2(int u) {
  vis[u] = 1;
  for (int v : radj[u]) {
    if (!vis[v]) dfs2(v);
  }
  r[u] = topo.size() - 1;
}
int32_t main() {
  int n;
  cin >> n;
  vector<int> a(n + 1);
  map<int, int> mp;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    mp[a[i]] = 0;
  }
  int L = 0;
  for (auto &x : mp) {
    x.second = ++L;
  }
  for (int i = 1; i <= n; ++i) {
    a[i] = mp[a[i]];
    d[a[i]]++;
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= L; ++j) {
      if (i == j) continue;
      adj[j].pb(i);
      radj[i].pb(j);
    }
  }
  for (int i = 1; i <= L; ++i) {
    if (!vis[i]) dfs1(i);
  }
  memset(vis, 0, sizeof vis);
  reverse(topo.begin(), topo.end());
  for (int u : topo) {
    if (!vis[u]) dfs2(u);
  }
  g[r[1]] = 3;
  for (int i = 2; i <= L; ++i) {
    g[r[i]] = 3;
    for (int j : adj[i]) {
      if (r[i] < r[j]) {
        if (d[i] <= d[j]) g[r[i]] = 1;
      }
    }
  }
  for (int i = 1; i <= n; ++i) {
    cout << g[r[a[i]]];
  }
  cout << '\n';
  return 0;
}
