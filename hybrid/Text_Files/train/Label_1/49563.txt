import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        // Read n*(n-1)/2-1 games
        boolean[][] played = new boolean[n + 1][n + 1];
        for (int i = 0; i < n * (n - 1) / 2 - 1; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            played[x][y] = played[y][x] = true;
        }

        // Find the missing game
        int x = -1, y = -1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (!played[i][j]) {
                    x = i;
                    y = j;
                    break;
                }
            }
        }

        System.out.println(x + " " + y);
    }
}
