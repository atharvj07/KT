#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[1000006];
int main() {
  ll n;
  cin >> n;
  ll mi = 1e18;
  ll ansx, ansy;
  for (ll i = 1; i <= n; i++) {
    cin >> a[i];
    for (ll j = i + 1; j <= n; j++) {
      ll g = __gcd(a[i], a[j]);
      if (g < mi) {
        mi = g;
        ansx = i;
        ansy = j;
      }
    }
  }
  cout << ansx << ' ' << ansy << endl;
  return 0;
}
