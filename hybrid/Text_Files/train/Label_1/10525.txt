#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long x, y, p, q;
        cin >> x >> y >> p >> q;
        if (x % y * q != p * y) {
            cout << -1 << endl;
        } else {
            cout << (p * y - x * q) / y << endl;
        }
    }
    return 0;
}
