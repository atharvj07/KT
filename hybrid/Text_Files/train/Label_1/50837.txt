import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        if (k < n) { System.out.println(-1); return; }
        List<Integer> p = new ArrayList<>(), q = new ArrayList<>();
        for (int i = n; i >= 1; i--) {
            if (k - i >= i) {
                p.add(i);
                q.add(k - i);
                k -= i;
            } else {
                p.add(k);
                q.add(1);
                break;
            }
        }
        System.out.println(p.stream().mapToInt(i -> i).sum());
        for (int x: p) System.out.print(x + " ");
        System.out.println();
        for (int x: q) System.out.print(x + " ");
        System.out.println();
    }
}
