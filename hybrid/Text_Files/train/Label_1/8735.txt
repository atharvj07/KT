#include <cstdio>

int par[25];
int c[25];

int f(int x) { return x == par[x] ? x : par[x] = f(par[x]); }

int main() {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) par[i] = i, scanf("%d", &c[i]);
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (c[i] > 1) {
      cnt++;
      if (c[i] == n) par[f(i)] = f(n + 1);
    }
    if (cnt >= 2) {
      puts("YES");
      return 0;
    }
  }
  puts("NO");
}
