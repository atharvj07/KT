def find_kth_permutation(n, k):
    # Finding the maximum length of such a sequence
    max_length = 1
    while (1 << max_length) - 1 < k:
        max_length += 1

    if max_length > n:
        return -1

    # Preparing the result array
    result = list(range(1, n - max_length + 2))
    base = n - max_length + 1

    # Constructing the permutation
    for i in range(max_length - 1, -1, -1):
        if k & (1 << i):
            result.append(base + i + 1)
        else:
            result.insert(-1, base + i + 1)

    return result

# Reading the number of test cases
t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    result = find_kth_permutation(n, k)
    if result == -1:
        print(-1)
    else:
        print(*result)
