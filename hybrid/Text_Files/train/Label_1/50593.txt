import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), k = sc.nextInt();
            String s = sc.next();
            int[] freq = new int[26];
            int minFreq = n;
            for (int i = 0; i < n; i++) {
                freq[s.charAt(i) - 'a']++;
                minFreq = Math.min(minFreq, freq[s.charAt(i) - 'a']);
            }
            if (minFreq % k != 0) {
                System.out.println(-1);
            } else {
                int[] count = new int[26];
                StringBuilder ans = new StringBuilder();
                for (int i = 0; i < n; i++) {
                    char c = s.charAt(i);
                    int index = c - 'a';
                    count[index]++;
                    if (count[index] <= minFreq) {
                        ans.append(c);
                    } else {
                        int nextIndex = -1;
                        for (int j = index + 1; j < 26; j++) {
                            if (count[j] < minFreq) {
                                nextIndex = j;
                                break;
                            }
                        }
                        if (nextIndex == -1) {
                            ans.append(c);
                        } else {
                            for (int j = i; j < n; j++) {
                                if (s.charAt(j) == c) {
                                    ans.append((char) ('a' + nextIndex));
                                    count[nextIndex]++;
                                } else {
                                    ans.append(s.charAt(j));
                                }
                            }
                            break;
                        }
                    }
                }
                if (ans.length() == n) {
                    System.out.println(ans);
                } else {
                    System.out.println(-1);
                }
            }
        }
    }
}
