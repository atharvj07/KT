#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, k, q, a[5005];
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i];
  cin >> q;
  for (int i = 0; i < q; i++) {
    int x;
    cin >> x;
    auto p = upper_bound(a, a + n, x / k);
    p--;
    if (p >= a) {
      int cost = (x + *p - 1) / *p;
      if (cost <= k)
        cout << cost << endl;
      else
        cout << -1 << endl;
    } else
      cout << -1 << endl;
  }
  return 0;
}
