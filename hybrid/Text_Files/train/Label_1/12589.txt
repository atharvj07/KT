#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    string s, t;
    cin >> s >> t;
    int n = s.length();
    vector<int> cnt(256);
    for (int i = 0; i < n; i++) cnt[s[i]]++;
    int ans1 = 0, ans2 = 0;
    for (int i = 0; i < n; i++) {
        if (t[i] == s[i]) ans1++;
        else if (cnt[t[i]] > 0) {
            ans2++;
            cnt[t[i]]--;
        }
    }
    cout << ans1 << ' ' << ans2 << '\n';
    return 0;
}
