#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n, q;
        cin >> n >> q;
        vector<ll> a(n), s(q);
        for (auto &x : a) { cin >> x; }
        for (auto &x : s) { cin >> x; }
        for (ll i = 0; i < q; i++) {
            ll sum = s[i];
            sort(a.begin(), a.end());
            while (sum > 0) {
                ll M = 0, m = 1e9;
                for (int j = 0; j < n; j++) {
                    M = max(M, a[j]);
                    m = min(m, a[j]);
                }
                ll mid = (M + m) / 2;
                vector<ll> L, R;
                for (int j = 0; j < n; j++) {
                    if (a[j] <= mid)
                        L.push_back(a[j]);
                    else
                        R.push_back(a[j]);
                }
                if (L.empty() || R.empty()) {
                    cout << "No" << endl;
                    break;
                }
                ll l = L.size(), r = R.size();
                ll lsum = 0, rsum = 0;
                for (auto &x : L) { lsum += x; }
                for (auto &x : R) { rsum += x; }
                if (lsum > sum || rsum > sum) {
                    cout << "No" << endl;
                    break;
                }
                if (lsum == sum) {
                    a = L;
                    break;
                } else if (rsum == sum) {
                    a = R;
                    break;
                } else if (lsum < sum) {
                    sum -= lsum;
                    a = R;
                } else if (rsum < sum) {
                    sum -= rsum;
                    a = L;
                }
            }
            if (sum <= 0)
                cout << "Yes" << endl;
        }
    }
}
