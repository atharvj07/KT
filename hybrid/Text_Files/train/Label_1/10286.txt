#include <iostream>
#include <vector>
#include <algorithm>
#include <limits.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  long long ans = LLONG_MAX;
  for (int i = 2; i <= 3; i++) {
    long long cur = 0;
    vector<long long> b = a;
    sort(b.begin(), b.end());
    for (int j = 0; j < n / i; j++)
      for (int k = 0; k < i; k++) cur += b[j * i + k];
    b.resize(n - n % i);
    for (int j = 0; j < n % i; j++) { cur += b[j]; }
    ans = min(ans, cur);
  }
  cout << ans << endl;
  return 0;
}
