#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;

ll n, k;
ll c[1001], p[1001], r[1001];
ll dp[1001][1001];
vector<pi> ans;

int main() {
  cin >> n;
  for (ll i = 1; i <= n; i++) cin >> c[i] >> p[i];
  cin >> k;
  for (ll i = 1; i <= k; i++) cin >> r[i];
  for (ll i = 1; i <= n; i++) {
    for (ll j = 1; j <= k; j++) {
      dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
      if (c[i] <= r[j]) dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + p[i]);
    }
  }
  ll i = n, j = k;
  while (i > 0 && j > 0) {
    if (dp[i][j] != dp[i - 1][j]) {
      ans.emplace_back(i, j);
      j -= 1;
    } else
      i -= 1;
  }
  reverse(ans.begin(), ans.end());
  cout << ans.size() << " " << dp[n][k] << "\n";
  for (auto &u : ans) cout << u.first << " " << u.second << "\n";
  return 0;
}
