import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            int q = in.nextInt();
            String s = in.next();
            for (int j = 0; j < q; j++) {
                int l = in.nextInt() - 1;
                int r = in.nextInt() - 1;
                boolean found = false;
                for (int k = l + 1; k < r; k++) {
                    if (s.charAt(k) != s.charAt(l) && s.charAt(k) != s.charAt(r)) {
                        found = true;
                        break;
                    }
                }
                System.out.println(found ? "YES" : "NO");
            }
        }
        in.close();
    }
}
