#include <iostream>
#include <vector>
using namespace std;

const int MAX_SIZE = 502;

int n, m;
char arr[MAX_SIZE][MAX_SIZE];

int sum;

void dfs(int r, int c, int h) {
    bool flag = true;
    for (int i = 0; i < h; i++) {
        for (int j = c - i; j <= c + i; j++) {
            if (arr[r + i][j] != '*') {
                flag = false;
                break;
            }
        }
        if (!flag) {
            break;
        }
    }
    if (flag) {
        sum++;
    }
    return;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> arr[i][j];
            }
        }
        sum = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (arr[i][j] == '*') {
                    for (int h = 1; h <= min(n - i + 1, m - j + 1); h++) {
                        dfs(i, j, h);
                    }
                }
            }
        }
        cout << sum << endl;
    }
    return 0;
}
