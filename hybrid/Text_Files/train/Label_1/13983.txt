#include <bits/stdc++.h>

using namespace std;

const int maxn = 305;

int n;
string a[maxn], b[maxn], s;
vector<string> v;

int main() {
	cin >> n;
	for (int i = 1; i <= n; i++) {
		cin >> a[i];
	}
	int ans = 0;
	for (int i = 1; i <= n; i++) {
		int len = a[i].size();
		ans += len;
		for (int j = i + 1; j <= n; j++) {
			b[j - i + 1] = a[j];
		}
		v.push_back(a[i]);
		for (int j = i + 1; j <= n; j++) {
			a[j - i + 1] = b[j - i + 1];
		}
	}
	if (v.size() == 1) {
		cout << ans << '\n';
		return 0;
	}
	s = v[0];
	for (int i = 1; i < v.size(); i++) {
		if (s == v[i]) {
			ans -= s.size();
			ans += s.size() - 1;
		}
	}
	cout << ans << '\n';
}
