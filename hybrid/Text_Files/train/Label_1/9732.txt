#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
int main(){
    int n,m; cin>>n>>m;
    vector<vector<int>>row(n),col(m);
    for(int i=0;i<n;++i){
        string s; cin>>s;
        for(int j=0;j<m;++j){
            if(s[j]=='#'){
                row[i].push_back(j);
                col[j].push_back(i);
            }
        }
    }
    for(int i=0;i<n;++i){
        for(auto j:row[i]){
            if(col[j].size()>1){
                cout<<"No"<<endl;
                return 0;
            }
        }
    }
    for(int i=0;i<m;++i){
        for(auto j:col[i]){
            if(row[j].size()>1){
                cout<<"No"<<endl;
                return 0;
            }
        }
    }
    cout<<"Yes"<<endl;
    return 0;
}
