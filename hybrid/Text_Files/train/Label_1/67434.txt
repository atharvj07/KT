import sys

def solve(keyboard, s):
    time = 0
    pos = ord(keyboard[0]) - ord('a')
    for c in s:
        next_pos = ord(c) - ord('a')
        time += abs(next_pos - pos)
        pos = next_pos
    return time

num_tests = int(sys.stdin.readline())
for i in range(num_tests):
    keyboard = sys.stdin.readline().strip()
    s = sys.stdin.readline().strip()
    result = solve(keyboard, s)
    print(result)
