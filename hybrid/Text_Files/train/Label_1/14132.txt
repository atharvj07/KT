#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct edge {
    int to, weight;
};

const int INF = 1e9;

int main() {
    int n, m, T;
    cin >> n >> m >> T;
    vector<vector<edge>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b, w;
        cin >> a >> b >> w;
        adj[a].push_back({b, w});
    }
    vector<int> dist(n + 1, INF);
    dist[1] = 0;
    priority_queue<pair<int, int>> pq;
    pq.push({0, 1});
    vector<int> path;
    while (!pq.empty()) {
        int v = pq.top().second;
        int d = -pq.top().first;
        pq.pop();
        if (d > dist[v]) continue;
        path.push_back(v);
        for (auto e : adj[v]) {
            if (d + e.weight <= T) {
                if (d + e.weight < dist[e.to]) {
                    dist[e.to] = d + e.weight;
                    pq.push({-(d + e.weight), e.to});
                }
            }
        }
    }
    cout << path.size() << endl;
    for (auto v : path) cout << v << " ";
    cout << endl;
    return 0;
}
