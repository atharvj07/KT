#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5;
const int INF = 0x3f3f3f3f;
int x[N], y[N], z[N], idx[N];
int32_t main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> x[i] >> y[i] >> z[i];
    }
    vector<pair<int, int>> ans;
    auto get = [&](int i1, int i2) -> bool {
        for (int i = 1; i <= n; i++)
            if (i != i1 && i != i2) {
                if (x[i] >= min(x[i1], x[i2]) && x[i] <= max(x[i1], x[i2]) &&
                    y[i] >= min(y[i1], y[i2]) && y[i] <= max(y[i1], y[i2]) &&
                    z[i] >= min(z[i1], z[i2]) && z[i] <= max(z[i1], z[i2]))
                    return 0;
            }
        return 1;
    };
    while (ans.size() != n / 2) {
        for (int i = 1; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (idx[i] || idx[j]) continue;
                if (get(i, j)) {
                    ans.push_back(make_pair(i, j));
                    idx[i] = idx[j] = 1;
                }
            }
        }
    }
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i].first << ' ' << ans[i].second << '\n';
    }
}
