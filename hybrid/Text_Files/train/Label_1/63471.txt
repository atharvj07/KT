n = int(input())
a = list(map(int, input().split()))

# Calculate the frequency of each statement
statement_freq = {}
for ai in a:
    if ai in statement_freq:
        statement_freq[ai] += 1
    else:
        statement_freq[ai] = 1

# Possible hat configuration
hat_config = [0] * n

# Sort statements by frequency (descending)
sorted_statements = sorted(statement_freq.items(), key=lambda x: -x[1])

# Assign hat types based on the sorted order
current_hat = 1
for statement, freq in sorted_statements:
    expected_diff = n - freq
    if expected_diff != statement:
        print("Impossible")
        break
    for i in range(n):
        if hat_config[i] == 0 and a[i] == statement:
            hat_config[i] = current_hat
    current_hat += 1
else:
    print("Possible")
    print(" ".join(map(str, hat_config)))
