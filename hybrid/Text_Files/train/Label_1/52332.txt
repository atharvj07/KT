import java.util.Scanner;

public class Main {
    public static final int MAX_N = 1000;
    public static final int MAX_K = 99;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();

        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
        }

        for (int i = 0; i < n; i++) {
            int mi = m[i];

            // Check if it is possible to find mi successive seats in one row
            if (mi > k) {
                System.out.println(-1);
                continue;
            }

            // Find the row with the minimum total seats remoteness from the center
            // and the segment [yl, yr] of the seats numbers in this row
            int x = (k + 1) / 2;
            int yl = (k + 1 - mi) / 2 + 1;
            int yr = yl + mi - 1;

            // Output the result
            System.out.println(x + " " + yl + " " + yr);
        }

        in.close();
    }
}
