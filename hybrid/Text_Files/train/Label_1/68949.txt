def min_operations_to_rearrange(n, colors):
    positions = [[] for _ in range(21)]
    for i, color in enumerate(colors):
        positions[color].append(i)
    
    answer = float('inf')
    
    for i in range(1, 21):
        if not positions[i]:
            continue
        dp = [[0] * len(positions[i]) for _ in range(len(positions[i]))]
        for length in range(1, len(positions[i])):
            for start in range(len(positions[i]) - length):
                end = start + length
                dp[start][end] = min(dp[start + 1][end], dp[start][end - 1]) + positions[i][end] - positions[i][start] - length
        
        current_answer = 0
        for j in range(1, 21):
            if j == i or not positions[j]:
                continue
            cnt = 0
            for pos in positions[j]:
                cnt += bisect.bisect_left(positions[i], pos)
            current_answer += min(cnt, len(positions[i]) * len(positions[j]) - cnt)
        
        answer = min(answer, current_answer + dp[0][len(positions[i]) - 1])
    
    return answer

import bisect
import sys

n = int(input().strip())
colors = list(map(int, input().strip().split()))

print(min_operations_to_rearrange(n, colors))
