#include<bits/stdc++.h>
using namespace std;
int main()
{
    int a[1005],n,i,j,ans=0,x=0;
    cin>>n;
    for(i=1;i<=n;i++)
    {
        cin>>a[i];
    }
    sort(a+1,a+n+1);
    for(i=n;i>=2;i--)
    {
        x+=a[i]*x+1;
    }
    ans=x+a[1];
    cout<<ans<<endl;
    for(i=n;i>=1;i--)
    {
        cout<<i<<' ';
    }
}
