import java.util.*;

/**
 * Finds the number of sequences of indices such that the density of the array is equal to the given value.
 */
public class Main {
    static long mod = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[n + 1];
        for (int i = 0; i < n; i++) {
            dp[i + 1] = dp[i];
            for (int j = i + 1; j < n; j++) {
                if (a[i] == a[j]) {
                    dp[j + 1] = dp[j];
                } else {
                    dp[j + 1] = Math.max(dp[j + 1], dp[i + 1] + 1);
                }
            }
        }
        long[] pw = new long[n + 1];
        pw[0] = 1;
        for (int i = 1; i < n + 1; i++) {
            pw[i] = pw[i - 1] * 2 % mod;
        }
        long[] ans = new long[n + 1];
        for (int i = 0; i < n + 1; i++) {
            ans[i] = pw[n - dp[i]] - 1;
            if (ans[i] < 0) ans[i] += mod;
            ans[i] %= mod;
        }
        ans[0] = 0;
        for (long x : ans) {
            System.out.println(x);
        }
    }
}
