#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    long long a, b;
    cin >> a >> b;
    if (a == b) {
        cout << 0 << endl;
        return 0;
    }
    int ans = 0;
    while (a != b) {
        if (a > b) {
            if (a % 5 == 0) {
                a = a / 5 * 4;
                ans++;
            } else if (a % 3 == 0) {
                a = a / 3 * 2;
                ans++;
            } else if (a % 2 == 0) {
                a = a / 2;
                ans++;
            } else {
                cout << -1 << endl;
                return 0;
            }
        } else {
            if (b % 5 == 0) {
                b = b / 5 * 4;
                ans++;
            } else if (b % 3 == 0) {
                b = b / 3 * 2;
                ans++;
            } else if (b % 2 == 0) {
                b = b / 2;
                ans++;
            } else {
                cout << -1 << endl;
                return 0;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
