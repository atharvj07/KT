#include <bits/stdc++.h>
using namespace std;

int main() {
  int t, n, k;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    vector<int> ans;
    int x = n / k + 1;
    int rem = n % k;
    while (rem) {
      ans.push_back(x);
      n -= x;
      k--;
      rem = n % k;
    }
    while (k--) { ans.push_back(x - 1); }
    for (int a : ans) { cout << a << ' '; }
    cout << '\n';
  }
}
