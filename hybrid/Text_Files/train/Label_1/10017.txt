#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long
using namespace std;
const int N = 5e5 + 5;

int mx_x = 0, mx_y = 0;
int n;

void add(ll x, ll y) {
  if (x > mx_x) mx_x = x;
  if (y > mx_y) mx_y = y;
}
bool que(ll x, ll y) {
  if (mx_x <= x && mx_y <= y) return true;
  if (mx_x <= y && mx_y <= x) return true;
  return false;
}

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    char c;
    cin >> c;
    if (c == '+') {
      ll x, y;
      cin >> x >> y;
      add(x, y);
    } else {
      ll x, y;
      cin >> x >> y;
      cout << (que(x, y) ? "YES" : "NO") << endl;
    }
  }
  return 0;
}
