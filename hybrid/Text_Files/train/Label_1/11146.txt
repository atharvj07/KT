#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, d;
    cin >> n >> d;
    string s;
    cin >> s;
    vector<int> dp(n + 1, -1);
    dp[n] = 0;
    for (int i = n - 1; i > 0; --i) {
        if (s[i] == '0') continue;
        for (int j = 1; j <= d; ++j) {
            if (i + j <= n && dp[i + j] != -1) {
                if (dp[i] == -1 || dp[i] > dp[i + j] + 1) {
                    dp[i] = dp[i + j] + 1;
                }
            }
        }
    }
    cout << dp[1] << endl;
    return 0;
}
