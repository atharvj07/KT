import sys

def input():
    return sys.stdin.readline()

n = int(input())
a = list(map(int, input().split()))
b = sorted(a)
ans = []
for i in range(n - 1):
    if a[i] > a[i + 1]:
        ans.append((2, i + 1, i + 2))
        a[i + 1] += a[i] - a[i + 1]
    else:
        ans.append((1, i + 1, i + 2))
        a[i] += a[i + 1] - a[i]
    if a[i] != b[i]:
        print(-1)
        sys.exit()

print(len(ans))
for i in ans:
    print(*i)
