import java.util.*;

public class Railroad {

    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static class Line {
        Point start, end;
        int owner, type;

        Line(Point start, Point end, int owner, int type) {
            this.start = start;
            this.end = end;
            this.owner = owner;
            this.type = type;
        }
    }

    static boolean isOnLine(Point p, Line line) {
        return (p.x - line.start.x) * (line.end.y - line.start.y) == (p.y - line.start.y) * (line.end.x - line.start.x);
    }

    static boolean isBetween(Point p, Line line) {
        return (p.x >= Math.min(line.start.x, line.end.x) && p.x <= Math.max(line.start.x, line.end.x)) &&
               (p.y >= Math.min(line.start.y, line.end.y) && p.y <= Math.max(line.start.y, line.end.y));
    }

    static int solve(Point a, Point b, List<Line> lines) {
        int count = 0;
        for (Line line : lines) {
            if (isOnLine(a, line) && isOnLine(b, line)) {
                // Both endpoints are on the same line
                continue;
            } else if (isOnLine(a, line) || isOnLine(b, line)) {
                // One endpoint is on the line
                count++;
            } else if (isBetween(a, line) && isBetween(b, line)) {
                // Both endpoints are between the line's endpoints
                count += 2;
            } else if (isBetween(a, line) || isBetween(b, line)) {
                // One endpoint is between the line's endpoints
                count++;
            }
        }
        return count;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int datasets = scanner.nextInt();
        for (int i = 0; i < datasets; i++) {
            Point a = new Point(scanner.nextInt(), scanner.nextInt());
            Point b = new Point(scanner.nextInt(), scanner.nextInt());
            int n = scanner.nextInt();
            List<Line> lines = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                lines.add(new Line(new Point(scanner.nextInt(), scanner.nextInt()),
                                   new Point(scanner.nextInt(), scanner.nextInt()),
                                   scanner.nextInt(), scanner.nextInt()));
            }
            int count = solve(a, b, lines);
            System.out.println(count);
        }
    }
}
