#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, T;
    cin >> n >> T;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<pair<long long, long long>> p(n);
    for (int i = 0; i < n; i++) p[i] = {a[i], i};
    sort(p.begin(), p.end());
    vector<int> ans(n);
    long long sum = 0;
    for (int i = 0; i < n; i++)
      if (sum + p[i].first <= T / 2) {
        sum += p[i].first;
        ans[p[i].second] = 0;
      } else
        ans[p[i].second] = 1;
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << "\n";
  }
}
