import sys

n, m = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))

for i in range(m):
    l, r = map(int, sys.stdin.readline().split())
    l -= 1
    r -= 1

    if a[l] > a[r]:
        print("No")
    elif a[l] == a[r]:
        print("Yes")
    else:
        inc = True
        for j in range(l + 1, r + 1):
            if a[j] < a[j - 1]:
                inc = False
                break
        dec = True
        for j in range(l + 1, r + 1):
            if a[j] > a[j - 1]:
                dec = False
                break
        if inc or dec:
            print("Yes")
        else:
            print("No")
