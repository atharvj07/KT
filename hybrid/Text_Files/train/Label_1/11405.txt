#include<bits/stdc++.h>
#define faster ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
using namespace std;

int n, k, p, a[1005], b[1005], ans;
set<int> pos;

int main() {
    faster
    freopen("input.txt", "r", stdin);
    cin >> n >> k >> p;
    for(int i=1; i<=n; i++) { cin >> a[i]; pos.insert(a[i]); }
    for(int i=1; i<=k; i++) { cin >> b[i]; pos.insert(b[i]); }

    for(int i=1; i<=n; i++) {
        auto it = pos.upper_bound(a[i]);
        if(it == pos.begin()) ans += abs(a[i] - p);
        else {
            it--;
            if(*it == a[i]) it++;
            int pos_key = *it;
            ans += abs(pos_key - a[i]) + abs(pos_key - p);
            pos.erase(it);
        }
    }

    cout << ans << endl;
}
