import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        int[] t = new int[n];
        for (int i = 0; i < n; i++) s[i] = sc.nextInt();
        for (int i = 0; i < n; i++) t[i] = sc.nextInt();
        System.out.println((valid(s, t))?"YES":"NO");
        sc.close();
    }
    static boolean valid(int[] s, int[] t) {
        Arrays.sort(s);
        Arrays.sort(t);
        for (int i = 0; i < s.length; i++) if (s[i] != t[i]) return false;
        return true;
    }
}
