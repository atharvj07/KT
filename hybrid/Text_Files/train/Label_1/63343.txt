n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = {}
for i in range(n):
    if a[i] in c: c[a[i]].append(b[i])
    else: c[a[i]] = [b[i]]
for i in range(1, n + 1):
    if i not in c: c[i] = []
if len({len(c[x]) for x in c}) > 1: print("Impossible")
elif len({len(c[x]) for x in c}) == 0: print("Ambiguity")
else:
    print("Possible")
    for i in range(1, n + 1):
        print(c[i][0], end=" ")
