import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int testCases = input.nextInt();

        while (testCases-- > 0) {
            int piranhaCount = input.nextInt();
            int[] piranhaSizes = new int[piranhaCount];
            for (int i = 0; i < piranhaCount; i++) {
                piranhaSizes[i] = input.nextInt();
            }

            int dominantPiranha = -1;
            for (int i = 0; i < piranhaCount; i++) {
                if (canEatAll(piranhaSizes, i)) {
                    dominantPiranha = i + 1;
                    break;
                }
            }

            System.out.println(dominantPiranha);
        }
    }

    public static boolean canEatAll(int[] piranhaSizes, int startIndex) {
        int currentSize = piranhaSizes[startIndex];
        for (int i = startIndex + 1; i < piranhaSizes.length; i++) {
            if (piranhaSizes[i] > currentSize) {
                return false;
            } else {
                currentSize++;
            }
        }
        return true;
    }
}
