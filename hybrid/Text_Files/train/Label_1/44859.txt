import java.util.*;
import java.io.*;

class Main {
    static final int MAX_N = 201, MAX_M = 10001;

    static boolean isPrime(int n) {
        if (n < 2) return false;
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) return false;
        }
        return true;
    }

    static int[] prime, pCnt;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++) a[i] = Integer.parseInt(s[i]);
        Arrays.sort(a);

        int tot = 0;
        for (int i : a) tot += i;
        if (tot % 2 == 1 || tot == 2) {
            System.out.println("Impossible");
            return;
        }
        tot >>= 1;

        prime = new int[MAX_M];
        pCnt = new int[MAX_M];
        for (int i = 4; i < MAX_M; i++) {
            if (!isPrime(i)) continue;
            prime[++prime[0]] = i;
        }

        int[][] dp = new int[n + 1][tot + 1];
        for (int i = 0; i <= n; i++) for (int j = 0; j <= tot; j++) dp[i][j] = -1;
        dp[0][0] = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= tot; j++) {
                if (dp[i][j] == -1) continue;
                dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j]);
                if (j + a[i] <= tot) {
                    int cnt = 0;
                    if (j + a[i] != 0)
                        cnt = pCnt[j + a[i]];
                    cnt++;
                    dp[i + 1][j + a[i]] = Math.max(dp[i + 1][j + a[i]], dp[i][j] + cnt);
                    pCnt[j + a[i]] = cnt;
                }
            }
        }

        if (dp[n][tot] == -1) {
            System.out.println("Impossible");
            return;
        }

        System.out.println(dp[n][tot]);
        int ind = n;
        int cur = tot;
        List<Integer> ans = new ArrayList<>();
        while (ind > 0) {
            if (dp[ind - 1][cur] == dp[ind][cur]) ind--;
            else {
                ans.add(ind);
                cur -= a[ind - 1];
                ind--;
            }
        }

        System.out.println(ans.size());
        for (int x : ans) System.out.print(x + " ");
    }
}
