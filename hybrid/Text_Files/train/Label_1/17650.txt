#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), pos(n + 1);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    pos[a[i]] = i;
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  int ans = 0, j = 0;
  for (int i = 1; i <= n; i++) {
    if (a[j] == i) {
      j++;
    } else {
      int p = pos[i];
      ans += p - j + 1;
      j = p + 1;
    }
  }
  cout << ans << endl;
  return 0;
}
