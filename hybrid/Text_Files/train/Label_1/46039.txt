import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), b = in.nextInt(), c = in.nextInt();
        int s = (a + b + c) / 2;
        System.out.println((int) Math.sqrt(s * (s - a) * (s - b) * (s - c)));
    }
}
