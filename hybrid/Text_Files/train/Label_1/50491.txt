import java.util.*;

public class Main {
    static char[][] keyboard;
    static int n, m, x;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        x = in.nextInt();
        keyboard = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = in.next();
            for (int j = 0; j < m; j++) {
                keyboard[i][j] = line.charAt(j);
            }
        }
        int q = in.nextInt();
        String text = in.next();
        int ans = 0;
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = -1;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (keyboard[i][j] == 's' || Character.isLetter(keyboard[i][j])) {
                    dist[i][j] = 0;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (dist[i][j] == -1) {
                    continue;
                }
                for (int k = 0; k < 4; k++) {
                    int nx = i + dx[k], ny = j + dy[k];
                    if (nx >= 0 && nx < n && ny >= 0 && ny < m) {
                        if (Character.isLetter(keyboard[i][j]) && Character.isLetter(keyboard[nx][ny])) {
                            if (Math.abs(keyboard[i][j] - keyboard[nx][ny]) >= 'a' - 'Z') {
                                dist[nx][ny] = Math.min(dist[nx][ny], dist[i][j] + 1);
                            }
                        } else if (keyboard[i][j] == 's' && (Character.isLetter(keyboard[nx][ny]) || keyboard[nx][ny] == 's')) {
                            dist[nx][ny] = Math.min(dist[nx][ny], dist[i][j] + 1);
                        }
                    }
                }
            }
        }
        for (int i = 0; i < q; i++) {
            char c = text.charAt(i);
            if (Character.isLetter(c)) {
                if (c >= 'a' && c <= 'z') {
                    int minDist = Integer.MAX_VALUE;
                    for (int j = 0; j < n; j++) {
                        for (int k = 0; k < m; k++) {
                            if (keyboard[j][k] == c) {
                                minDist = Math.min(minDist, dist[j][k]);
                            }
                        }
                    }
                    if (minDist == Integer.MAX_VALUE) {
                        ans = -1;
                        break;
                    }
                    ans = Math.max(ans, minDist);
                } else {
                    int minDist = Integer.MAX_VALUE;
                    for (int j = 0; j < n; j++) {
                        for (int k = 0; k < m; k++) {
                            if (keyboard[j][k] == c || (keyboard[j][k] >= 'a' && keyboard[j][k] <= 'z' && Character.toUpperCase(keyboard[j][k]) == c)) {
                                minDist = Math.min(minDist, dist[j][k]);
                            }
                        }
                    }
                    if (minDist == Integer.MAX_VALUE) {
                        ans = -1;
                        break;
                    }
                    ans = Math.max(ans, minDist);
                }
            }
        }
        System.out.println(ans);
    }
}
