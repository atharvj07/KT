#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long a, b, c;
    cin >> a >> b >> c;
    long long ans = abs(a - b) + abs(b - c) + abs(a - c);
    long long mid = (a + b + c) / 3;
    ans = min(ans, abs(a - mid) + abs(b - mid) + abs(c - mid));
    cout << ans << endl;
  }
  return 0;
}
