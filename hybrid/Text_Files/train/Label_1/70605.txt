import sys
input = sys.stdin.readline

n, x = map(int, input().split())
d = list(map(int, input().split()))
cumsum = [0]
for i in d:
    cumsum.append(cumsum[-1] + i)
mn = cumsum[-1]
ans = 0
for i in range(1, n + 1):
    for j in range(i, n + 1):
        sum = cumsum[j] - cumsum[i - 1]
        if sum > x:
            break
        if cumsum[j] + min(x - sum, cumsum[-1] - cumsum[j]) > ans:
            ans = cumsum[j] + min(x - sum, cumsum[-1] - cumsum[j])
print(ans)
