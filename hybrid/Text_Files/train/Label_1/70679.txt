for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    b = list(map(int,input().split()))
    x=0
    for j in range(n-1):
        for i in range(n-1):
            if i==j:
                continue
            if a[i]>a[i+1] and b[i]!=b[i+1]:
                a[i],a[i+1]=a[i+1],a[i]
                b[i],b[i+1]=b[i+1],b[i]
                x=1
    if x==1:
        print("YES")
    else:
        print("NO")
