#include <vector>
#include <algorithm>
#include <iostream>
#include <cmath>

int main() {
  long long n;
  std::cin >> n;
  std::vector<long long> v = {3, 4}, tmp;
  for (long long i = 6, j = 5; i <= n; i += 5, j++) {
    tmp.clear();
    for (const auto &x : v) {
      tmp.push_back(x);
      tmp.push_back(x + j);
    }
    std::swap(tmp, v);
  }
  if (n > 4 && v.back() != n) {
    v.push_back(n);
  }
  if (v.empty() || *v.begin() > n) {
    std::cout << -1 << std::endl;
  } else {
    for (const auto &x : v) {
      if (x >= 3) {
        std::cout << x << std::endl;
      }
    }
  }
  return 0;
}
