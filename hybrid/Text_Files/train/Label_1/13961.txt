#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll minAB(ll a, ll b, ll x, ll y, ll n) {
    if (b < y) { swap(a, b); swap(x, y); }
    while (a > x && n--) a--, b++;
    return a * b;
}
int main() {
    ll t, a, b, x, y, n;
    cin >> t;
    while (t--) cin >> a >> b >> x >> y >> n, cout << minAB(a, b, x, y, n) << '\n';
    return 0;
}
