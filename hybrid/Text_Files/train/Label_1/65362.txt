import sys
input = sys.stdin.readline

k, n = map(int, input().split())
s = list(input() for _ in range(k))

cnt = {}
for i in range(n):
    t = ''
    for j in range(k):
        t += s[j][i]
    cnt[t] = cnt.get(t, 0) + 1

res = []
for c in cnt:
    if cnt[c] % 2 != 0:
        if len(res) == 0:
            res.append(c)
        else:
            res = []
            break

if len(res) == 0:
    print(-1)
else:
    print(res[0])
