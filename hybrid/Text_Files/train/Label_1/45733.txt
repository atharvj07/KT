import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt();
    Map<Integer, List<Participant>> map = new HashMap<>();
    for (int i = 0; i < n; i++) {
      Participant p = new Participant(sc.next(), sc.nextInt(), sc.nextInt());
      map.computeIfAbsent(p.region, k -> new ArrayList<>()).add(p);
    }
    for (int i = 1; i <= m; i++) {
      List<Participant> list = map.get(i);
      if (list == null || list.size() < 2) {
        System.out.println("?");
        continue;
      }
      list.sort(Comparator.comparingInt(Participant::getScore).reversed());
      System.out.println(list.get(0).name + " " + list.get(1).name);
    }
  }

  static class Participant {
    String name;
    int region, score;

    Participant(String name, int region, int score) {
      this.name = name;
      this.region = region;
      this.score = score;
    }

    int getScore() {
      return score;
    }
  }
}
