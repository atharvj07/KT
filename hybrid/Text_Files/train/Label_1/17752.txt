#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;
const char nl = '\n';

int main() {
  int n, m;
  cin >> n >> m;
  ll x, k, y;
  cin >> x >> k >> y;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
  }
  vector<int> pos(n);
  for (int i = 0; i < n; i++) {
    pos[a[i]] = i;
  }
  ll ans = 0;
  int j = 0;
  for (int i = 0; i < m; i++) {
    while (j < n && a[j] != b[i]) {
      j++;
    }
    if (j == n) {
      return cout << -1 << nl, 0;
    }
    if (j + 1 < n && a[j + 1] == b[i + 1]) {
      ans += y;
      j++;
      i++;
    } else if (j > 0 && a[j - 1] == b[i - 1]) {
      ans += y;
      j--;
      i--;
    } else {
      ans += x - y;
    }
  }
  cout << ans << nl;
}
