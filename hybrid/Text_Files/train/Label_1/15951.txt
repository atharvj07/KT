#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  if (n == 2) {
    swap(a[0], a[1]);
    b = a;
  } else if (n == 3) {
    sort(a.begin(), a.end());
    b = {a[1], a[2], a[0]};
  } else {
    cout << -1 << endl;
    return 0;
  }

  for (int x : b) cout << x << ' ';
  cout << endl;

  return 0;
}
