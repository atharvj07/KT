import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        long sum = 0;
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
        }
        Arrays.sort(x);
        Arrays.sort(v);
        for (int i = 0; i < n - 1; i++) {
            int y = x[i + 1];
            int yv = v[i + 1];
            int xDiff = y - x[i];
            int vDiff = yv - v[i];
            if (vDiff >= 0) {
                sum += xDiff;
            }
        }
        System.out.println(sum);
    }
}
