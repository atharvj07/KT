#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;cin>>n;
    vector<int>good;
    vector<vector<int>>v(n,vector<int>(n));
    for(int i=0;i<n;i++)for(int j=0;j<n;j++)cin>>v[i][j];
    for(int i=0;i<n;i++){
        int cnt=0;
        for(int j=0;j<n;j++){
            if(i==j)continue;
            if(v[i][j]==0||v[i][j]==3)cnt++;
        }
        if(cnt==n-1)good.push_back(i+1);
    }
    cout<<good.size()<<endl;
    for(int x:good)cout<<x<<" ";
}
