#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, c;
    cin >> n >> c;
    vector<int> p(n), t(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    int limakscore = 0, radewooshscore = 0;
    for (int i = 0; i < n; i++) {
        limakscore += max(0, p[i] - c * (i + t[i]));
        radewooshscore += max(0, p[n - 1 - i] - c * (i + t[n - 1 - i]));
    }
    if (limakscore > radewooshscore) {
        cout << "Limak" << endl;
    } else if (limakscore < radewooshscore) {
        cout << "Radewoosh" << endl;
    } else {
        cout << "Tie" << endl;
    }
    return 0;
}
