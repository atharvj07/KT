def is_possible_layout(s):
    # Initialize the adjacency list.
    adj = {ch: set() for ch in s}
    
    # Fill the adjacency list based on the password.
    for i in range(len(s) - 1):
        adj[s[i]].add(s[i + 1])
        adj[s[i + 1]].add(s[i])
    
    # Check for the impossibility condition: any character having more than 2 neighbors.
    for ch in adj:
        if len(adj[ch]) > 2:
            return "NO", ""
    
    # Find the start character (with only 1 neighbor) for the layout.
    start = None
    for ch in adj:
        if len(adj[ch]) == 1:
            start = ch
            break
    
    # If there's no starting point but the password is not empty, it's a loop or isolated characters.
    if start is None and s:
        return "NO", ""
    
    # Generate the layout from the starting character.
    layout = [start]
    visited = {start}
    while True:
        next_chars = [ch for ch in adj[layout[-1]] if ch not in visited]
        if not next_chars:
            break
        layout.append(next_chars[0])
        visited.add(next_chars[0])
    
    # Fill the rest of the layout with the remaining alphabet characters.
    for ch in "abcdefghijklmnopqrstuvwxyz":
        if ch not in visited:
            layout.append(ch)
    
    return "YES", "".join(layout)

# Read the number of test cases.
T = int(input().strip())
for _ in range(T):
    s = input().strip()
    result, layout = is_possible_layout(s)
    print(result)
    if result == "YES":
        print(layout)
