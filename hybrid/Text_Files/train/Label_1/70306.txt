n, m = [int(i) for i in input().split()]
grid = [input() for _ in range(n)]
ans = 1000000000000000000
for white in grid:
    if white.count('W') > 0:
        ans = min(ans, white.count('B'))
for i in range(0, n):
    if grid[i].count('B') > 0:
        ans = min(ans, sum([j.count('B') for j in grid[i + 1:]]))
for i in range(0, m):
    cnt = 0
    for j in range(0, n):
        if grid[j][i] == 'B':
            cnt += 1
        else:
            ans = min(ans, cnt)
if ans == 1000000000000000000:
    print(-1)
else:
    print(ans)
