n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for i in range(0, m):
    if b[i] == 0:
        continue
    l = a.count(i+1)
    if l < b[i]:
        print('NO')
        exit(0)
    l -= b[i]
    for j in range(0, n):
        if l <= 0:
            break
        if a[j] == i+1:
            l -= 1
            a[j] = -1
print('YES')
