import sys
from collections import defaultdict,Counter,deque
input = sys.stdin.readline

N = int(input())

authors = []
for _ in range(N):
    authors.append(input().rstrip())

for a in authors:
    c = Counter(a)
    c['0'] = 0
    c['1'] = 0
    c['2'] = 0
    c['3'] = 0
    c['4'] = 0
    c['5'] = 0
    c['6'] = 0
    c['7'] = 0
    c['8'] = 0
    c['9'] = 0
    if c.most_common(1)[0][1] > 1:
        print("Impossible")
        exit(0)

for i in range(N):
    for j in range(i+1,N):
        x = list(authors[i])
        y = list(authors[j])
        for k in range(min(len(x),len(y))):
            if x[k] > y[k]:
                print("Impossible")
                exit(0)
            elif x[k] < y[k]:
                break

ans = list('abcdefghijklmnopqrstuvwxyz')
for i in authors:
    set_i = sorted(set(i))
    for j in range(len(set_i)):
        for k in range(len(ans)):
            if ans[k]==set_i[j]:
                ans[k],ans[j] = ans[j],ans[k]
                break

print(''.join(ans))
