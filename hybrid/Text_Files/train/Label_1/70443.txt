n, m = map(int, input().split())
a = list(map(int, input().split()))
s = [0]
for x in a:
    s.append(s[-1] + x)
b = list(map(int, input().split()))
for x in b:
    l, r = 0, n
    while l < r:
        mid = (l + r) // 2
        if x <= s[mid]:
            r = mid
        else:
            l = mid + 1
    print(l, x - s[l - 1])
