from sys import stdin

n = int(stdin.readline())

if n % 2 == 1:
    print(n)
    for i in range(1, n + 1):
        print(i, n + 1 - i)
else:
    print(n // 2 + 1)
    for i in range(1, n // 2 + 1):
        print(i, n // 2 + 1 - i)
    for i in range(1, n // 2 + 1):
        print(i, n // 2 + 1 + i)
