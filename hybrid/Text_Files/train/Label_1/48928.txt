import java.util.Scanner;

public class IvanAndLessons {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine();

        int[][] lessons = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m; j++) {
                lessons[i][j] = line.charAt(j) - '0';
            }
        }

        int[][] dp = new int[n][k + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }

        for (int j = 0; j <= k; j++) {
            for (int i = 0; i < n; i++) {
                if (j == 0) {
                    dp[i][j] = countHours(lessons, i);
                } else {
                    for (int l = 0; l <= i; l++) {
                        dp[i][j] = Math.min(dp[i][j], dp[l][j - 1] + countHours(lessons, i) - countHours(lessons, l));
                    }
                }
            }
        }

        System.out.println(dp[n - 1][k]);
    }

    private static int countHours(int[][] lessons, int day) {
        int hours = 0;
        for (int i = 0; i < lessons[day].length; i++) {
            if (lessons[day][i] == 1) {
                hours++;
            }
        }
        return hours;
    }
}
