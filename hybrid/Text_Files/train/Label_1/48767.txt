import java.util.Scanner;

public class ChampionshipWinners {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }

            int count = 0;
            int[] winners = new int[n];
            for (int j = 0; j < n; j++) {
                if (canWin(a, j)) {
                    winners[count++] = j + 1;
                }
            }

            System.out.println(count);
            for (int j = 0; j < count; j++) {
                System.out.print(winners[j] + " ");
            }
            System.out.println();
        }
    }

    private static boolean canWin(int[] a, int playerIndex) {
        int remainingTokens = a[playerIndex];
        int playersLeft = a.length;

        while (playersLeft > 1) {
            int opponentIndex = getRandomOpponent(a, playerIndex);
            int opponentTokens = a[opponentIndex];

            if (remainingTokens > opponentTokens) {
                remainingTokens += opponentTokens;
            } else {
                return false;
            }

            a[opponentIndex] = 0;
            playersLeft--;
        }

        return true;
    }

    private static int getRandomOpponent(int[] a, int playerIndex) {
        while (true) {
            int opponentIndex = (int) (Math.random() * a.length);
            if (opponentIndex != playerIndex && a[opponentIndex] > 0) {
                return opponentIndex;
            }
        }
    }
}
