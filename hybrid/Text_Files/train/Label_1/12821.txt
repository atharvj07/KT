#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; ++i) { cin >> p[i]; }
  vector<int> ans(n + 1, 0);
  for (int i = 1; i <= n; ++i) {
    int j = i;
    while (!ans[j]) {
      ans[j] = i;
      j = p[j];
    }
  }
  for (int i = 1; i <= n; ++i) { cout << ans[i] << " "; }
  return 0;
}
