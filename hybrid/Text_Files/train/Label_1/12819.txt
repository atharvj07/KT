#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  vector<int> res(n + 1);
  for (int i = 1; i <= n; i++) {
    int j = i;
    while (res[j] == 0) {
      res[j] = i;
      j = p[j];
    }
    res[j] = i;
  }
  for (int i = 1; i <= n; i++) {
    cout << res[i] << " ";
  }
  return 0;
}
