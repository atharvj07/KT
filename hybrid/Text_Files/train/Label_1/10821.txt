#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  int n;
  cin >> n;
  while (n--) {
    string a;
    cin >> a;
    int flag = 1;
    for (int i = 0; i < a.size() - 1; i++) {
      if (a[i] + 1 == a[i + 1]) {
        flag = 0;
      }
    }
    if (flag)
      cout << a << '\n';
    else
      cout << "No answer" << '\n';
  }
  return 0;
}
