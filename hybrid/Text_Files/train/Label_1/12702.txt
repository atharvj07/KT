#include<bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int rootlen = 5;
  for (int i = 4; i < s.size(); ++i)
    if (s[i] != s[i % rootlen]) {
      rootlen = i;
      break;
    }
  set<string> suf;
  for (int i = rootlen; i < s.size(); ++i)
    for (int j = 2; j <= 3; ++j)
      if (i - j >= rootlen)
        suf.insert(s.substr(i - j, j));
  cout << suf.size() << "\n";
  for (const string &i : suf) cout << i << "\n";
  return 0;
}
