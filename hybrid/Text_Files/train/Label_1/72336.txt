s = input().strip()
n = len(s)

# Check if it's possible to have a nice word
for i in range(n - 25):
    missing = set("ABCDEFGHIJKLMNOPQRSTUVWXYZ")
    question_marks = 0
    valid = True
    for j in range(26):
        if s[i + j] == '?':
            question_marks += 1
        elif s[i + j] in missing:
            missing.remove(s[i + j])
        else:
            valid = False
            break
    if valid and len(missing) <= question_marks:
        # Construct the nice word
        nice_word = list(s)
        for j in range(26):
            if nice_word[i + j] == '?':
                if missing:
                    nice_word[i + j] = missing.pop()
                else:
                    nice_word[i + j] = 'A'  # Use 'A' or any letter to fill extra '?'
        for k in range(n):
            if nice_word[k] == '?':
                nice_word[k] = 'A'  # Replace remaining '?' with 'A' or any letter
        print("".join(nice_word))
        break
else:
    print("-1")
