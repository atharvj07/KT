#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> dp(n + 1, 0);
    for (int i = 0; i < n; i++) {
      dp[i + 1] = dp[i];
      if (i + 2 < n) {
        if (s[i] == s[i + 1] && s[i + 1] == s[i + 2]) {
          dp[i + 1]++;
        }
      }
      if (i + 3 < n) {
        if (s[i] != s[i + 1] && s[i + 1] == s[i + 2] && s[i + 2] != s[i + 3]) {
          dp[i + 1] += dp[i];
        }
      }
    }
    cout << dp[n] << endl;
  }
  return 0;
}
