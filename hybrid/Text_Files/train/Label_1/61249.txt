def min_cost_shovels():
    n, m, k = map(int, input().split())
    shovel_costs = list(map(int, input().split()))
    offers = [tuple(map(int, input().split())) for _ in range(m)]

    shovel_costs.sort()

    dp = [0] + [float('inf')] * k

    for i in range(1, k + 1):
        dp[i] = dp[i - 1] + shovel_costs[i - 1]

        for x, y in offers:
            if x <= i:
                dp[i] = min(dp[i], dp[i - x] + sum(shovel_costs[i - x:i - y]))

    return dp[k]

print(min_cost_shovels())
