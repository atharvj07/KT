#include <iostream>
#include <cstring>
#define MAXLEN 500
using namespace std;
char a[MAXLEN][MAXLEN];
bool vis[MAXLEN][MAXLEN];
int n, m, dx[4]={0,0,1,-1}, dy[4]={1,-1,0,0};
bool check(int x1, int y1, int x2, int y2){
    memset(vis, false, sizeof(vis));
    vis[x1][y1] = true;
    for(int i=x1; i<=x2; i++){
        for(int j=y1; j<=y2; j++){
            if(a[i][j]=='X' && !vis[i][j]){
                bool flag=false;
                for(int k=0; k<4; k++){
                    int nx=i+dx[k], ny=j+dy[k];
                    if(nx>=x1 && nx<=x2 && ny>=y1 && ny<=y2){
                        if(a[nx][ny]=='X'){
                            vis[nx][ny]=true;
                            flag=true;
                            break;
                        }
                    }
                }
                if(!flag)
                    return false;
            }
        }
    }
    return true;
}
int main(){
    int x1=-1, y1=-1, x2=-1, y2=-1;
    cin>>n>>m;
    for(int i=1; i<=n; i++){
        for(int j=1; j<=m; j++){
            cin>>a[i][j];
            if(a[i][j]=='X'){
                if(x1==-1){
                    x1=i;
                    y1=j;
                }
                x2=i;
                y2=j;
            }
        }
    }
    if(check(x1, y1, x2, y2)){
        cout<<"YES"<<endl;
        return 0;
    }
    if(check(x1, n-y2+1, x2, m-y1+1)){
        cout<<"YES"<<endl;
        return 0;
    }
    cout<<"NO"<<endl;
    return 0;
}
