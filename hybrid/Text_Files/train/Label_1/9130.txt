# include <iostream>
# include <vector>
# include <algorithm>
using namespace std;
const int maxn = 1500 + 5;
int n, s, m, k, a[maxn];
vector<pair<int, int>> edges[maxn];
vector<int> tmp;
int main() {
	cin >> n >> s >> m >> k;
	for (int i = 1; i <= n; ++i)
		cin >> a[i];
	for (int i = 1; i <= s; ++i) {
		int l, r;
		cin >> l >> r;
		edges[l].push_back(pair<int, int>(r, i));
	}
	for (int i = 1; i <= n; ++i)
		for (pair<int, int> edge : edges[i]) {
			tmp.push_back(a[i]);
			if (edge.first == n)
				continue;
			edges[edge.first + 1].push_back(pair<int, int>(edge.first, edge.second));
		}
	sort(tmp.begin(), tmp.end());
	if (tmp.size() < k)
		cout << -1 << endl;
	else
		cout << tmp[k - 1] << endl;
	return 0;
}
