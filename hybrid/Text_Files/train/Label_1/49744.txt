import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] frogs = new long[n][2];
        for (int i = 0; i < n; i++) {
            frogs[i][0] = sc.nextInt();
            frogs[i][1] = sc.nextInt();
        }
        long[][] mosquitoes = new long[m][2];
        for (int i = 0; i < m; i++) {
            mosquitoes[i][0] = sc.nextInt();
            mosquitoes[i][1] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            long numEaten = 0, tongueLength = frogs[i][1];
            for (int j = 0; j < m; j++) {
                if (frogs[i][0] <= mosquitoes[j][0] && mosquitoes[j][0] <= frogs[i][0] + tongueLength) {
                    numEaten++;
                    tongueLength += mosquitoes[j][1];
                }
            }
            System.out.println(numEaten + " " + tongueLength);
        }
    }
}
