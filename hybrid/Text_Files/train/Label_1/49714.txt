import java.util.*;

public class Main {
    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long x = in.nextLong(), y = in.nextLong();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextLong();
        }
        if (n == 1) {
            System.out.println(0);
            return;
        }
        long g = 0;
        for (int i = 0; i < n; i++) {
            g = gcd(g, arr[i]);
        }
        if (g == 1) {
            System.out.println(0);
            return;
        }
        long ans = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            int curX = 0;
            int curY = 0;
            long curG = 0;
            while (g != 1) {
                if (arr[i] % g == 0) {
                    arr[i] /= g;
                    curY++;
                    g = 0;
                    for (int j = 0; j < n; j++) {
                        g = gcd(g, arr[j]);
                    }
                    if (g == 1) {
                        break;
                    }
                } else {
                    arr[i]++;
                    curX++;
                    curG = 0;
                    for (int j = 0; j < n; j++) {
                        curG = gcd(curG, arr[j]);
                    }
                    g = curG;
                }
                ans = Math.min(ans, curX * x + curY * y);
            }
        }
        System.out.println(ans);
    }
}
