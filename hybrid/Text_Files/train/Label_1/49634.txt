import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        List<Integer> order = new ArrayList<>();
        int cur = 1;
        for (int i = n - 1; i >= 0; i--) {
            ans += cur * a[i];
            order.add(n - i);
            cur++;
        }
        System.out.println(ans);
        for (int i = order.size() - 1; i >= 0; i--) {
            System.out.print(order.get(i) + " ");
        }
    }
}
