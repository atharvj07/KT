t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    res = 0
    for i in range(n):
        for k in range(i+2, n):
            count = 0
            for j in range(i+1, k):
                if a[j] == a[k]:
                    count += 1
            for l in range(k+1, n):
                if a[i] == a[l]:
                    res += count
    print(res)
