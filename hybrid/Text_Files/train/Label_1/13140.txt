#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;

int n;
long long a[N], c[N];
long long mx;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        mx = max(a[i], mx);
        if (a[i] >= mx / 2) c[i] = -1;
        else {
            int j = i;
            while (a[j] >= a[i] / 2) j++;
            while (a[j] < a[i] / 2 && a[j + 1] < a[i] / 2) {
                j += 2;
                if (j > n) j -= n;
            }
            c[i] = j - i + 1;
        }
    }
    for (int i = 1; i <= n; i++) cout << c[i] << ' ';
    return 0;
}
