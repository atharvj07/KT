import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        long[] c = new long[k];
        for (int i = 0; i < k; i++) {
            c[i] = input.nextInt();
        }
        long ans = 1;
        for (int i = 0; i < k - 1; i++) {
            ans = ans * c[i] % 1000000007;
        }
        System.out.println(ans);
    }
}
