import java.util.Arrays;
import java.util.LinkedList;
import java.util.PriorityQueue;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static class Friend implements Comparable<Friend> {
        int id, relation;

        public Friend(int id, int relation) {
            this.id = id;
            this.relation = relation;
        }

        @Override
        public int compareTo(Friend o) {
            return o.relation - this.relation;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), q = in.nextInt();
        int[] relations = new int[n];
        PriorityQueue<Friend> pq = new PriorityQueue<>();
        Queue<Friend> online = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            relations[i] = in.nextInt();
        }
        for (int i = 0; i < q; i++) {
            int type = in.nextInt(), id = in.nextInt();
            if (type == 1) {
                online.add(new Friend(id - 1, relations[id - 1]));
                online.stream().limit(k).forEach(pq::add);
                online.removeIf(f -> pq.size() < k || pq.peek().relation == f.relation);
            } else {
                System.out.println(online.stream().anyMatch(friend -> friend.id == id - 1) ? "YES" : "NO");
            }
        }
    }
}
