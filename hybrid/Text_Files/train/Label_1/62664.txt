mod = 998244353
n = int(input())
seq = list(map(int, input().split()))
dp = [0] * n
dp[n - 1] = 1 if seq[n - 1] == n - 1 else 0
for i in range(n - 2, -1, -1):
    if seq[i] > 0:
        dp[i] = (dp[i + 1] * 2) % mod
        if seq[i] == i:
            dp[i] = (dp[i] + 1) % mod
    else:
        dp[i] = dp[i + 1]
print(dp[0])
