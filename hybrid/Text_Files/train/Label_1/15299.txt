#include <bits/stdc++.h>

using namespace std;

#define ll long long

int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.rbegin(), a.rend());
  ll sum = 0, ans = 0;
  for (ll i = 0; i < n; i++) {
    if (sum + a[i] < m) {
      sum += a[i];
      ans++;
    } else {
      break;
    }
  }
  if (sum < m) {
    cout << -1 << endl;
  } else {
    cout << ans << endl;
  }
  return 0;
}
