#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<array<int, 2>> a(n);
    for (int i = 0; i < n; i++) cin >> a[i][0] >> a[i][1];
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 1; i < n; i++) {
      if (a[i][0] == a[i - 1][0]) {
        ans += min(a[i][1], a[i - 1][1]);
      }
    }
    cout << ans << endl;
  }
  return 0;
}
