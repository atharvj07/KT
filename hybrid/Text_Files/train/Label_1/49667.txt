import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] t = br.readLine().split(" ");
        int n = Integer.parseInt(t[0]), m = Integer.parseInt(t[1]);
        long[] x = new long[n];
        String[] tx = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            x[i] = Long.parseLong(tx[i]);
        }
        String[] tp = br.readLine().split(" ");
        long[] p = new long[m];
        for (int i = 0; i < m; i++) {
            p[i] = Long.parseLong(tp[i]);
        }
        for (int i = 1; i < n; i++) {
            if (x[i] - x[i - 1] > p[m - 1] || x[i] - x[0] > p[m - 1]) {
                System.out.println("NO");
                return;
            }
        }
        long y = Long.MAX_VALUE;
        int ind = -1;
        for (long j : x) {
            if (j % p[0] < y) {
                y = j % p[0];
                ind = 1;
            }
        }
        if (ind == -1) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        System.out.println(y + " " + ind);
    }
}
