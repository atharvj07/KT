import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] t = new int[n];
        int cur = 0, cnt = 0;
        for (int i = 0; i < n; i++) {
            if (cur + a[i] <= m) {
                cur += a[i];
            } else {
                t[i] = (cur + a[i]) / m - 1;
                cur = (cur + a[i]) % m;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(t[i] + " ");
        }
    }
}
