import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String singular = in.next();
        String plural = singular;
        if (!singular.endsWith("s")) {
            plural += "s";
        } else {
            plural += "es";
        }
        System.out.println(plural);
    }
}
