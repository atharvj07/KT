#include <bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
const int N = 2e5 + 10;
const ll oo = 1e18;
int n, k;
ll x[N], y[N];
vector<ll> adj[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
      cin >> x[i] >> y[i];
    }
    ll ans = 0;
    // compress the x coordinates
    vector<ll> X;
    for (int i = 1; i <= n; ++i) {
      X.pb(x[i]);
    }
    sort(X.begin(), X.end());
    X.erase(unique(X.begin(), X.end()), X.end());
    for (int i = 1; i <= n; ++i) {
      x[i] = lower_bound(X.begin(), X.end(), x[i]) - X.begin() + 1;
    }
    // build the adjacency list
    for (int i = 1; i <= n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (y[i] == y[j]) {
          if (abs(x[i] - x[j]) <= k) {
            adj[i].pb(j);
            adj[j].pb(i);
          }
        }
      }
    }
    vector<bool> vis(n + 1, false);
    // run dfs
    function<void(int)> dfs = [&dfs, &vis, &ans](int u) {
      if (vis[u]) return;
      vis[u] = true;
      ans++;
      for (int v : adj[u]) {
        dfs(v);
      }
    };
    for (int i = 1; i <= n; ++i) {
      if (!vis[i]) {
        dfs(i);
      }
    }
    cout << ans << '\n';
    // cleanup
    for (int i = 1; i <= n; ++i) {
      adj[i].clear();
      vis[i] = false;
    }
  }
  return 0;
}
