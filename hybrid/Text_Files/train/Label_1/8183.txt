#include <bits/stdc++.h>
using namespace std;
const int N = 2e5+5;
vector<int> g[N];
int subtree[N], ans[N];
int dfs(int u, int p) {
    subtree[u] = 1;
    for (auto v : g[u]) {
        if (v == p) continue;
        subtree[u] += dfs(v, u);
    }
    return subtree[u];
}
int main() {
    int n, k; cin >> n >> k;
    for (int i = 0; i < n-1; i++) {
        int u, v; cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 1);
    int need = k+1;
    vector<pair<int,int>> taken;
    function<void(int,int,int)> dfs2 = [&](int u, int p, int c) {
        if (subtree[u] >= need) {
            taken.push_back({u, c});
            return;
        }
        for (auto v : g[u]) {
            if (v == p) continue;
            dfs2(v, u, c+1);
            if ((int)taken.size() >= k+1) break;
        }
    };
    dfs2(1, 1, 0);
    cout << taken.size() << "\n";
    for (auto &t : taken) ans[t.first] = t.second;
    for (int i = 0; i < n-1; i++) {
        if (!ans[i+1]) cout << 1 << " ";
        else cout << ans[i+1] << " ";
    }
    cout << "\n";
    return 0;
}
