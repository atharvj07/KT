import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();
        int V = sc.nextInt();
        int[] x = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = sc.nextInt();
        }

        boolean[] possible = new boolean[N];

        for (int i = 0; i < N; i++) {
            if (x[i] <= V && V - x[i] >= x[0] - x[i]) {
                possible[i] = true;
                continue;
            }

            int sum = 0;
            int j = i + 1;
            while (j < N) {
                if (sum + x[j] - x[i] <= V) {
                    sum += x[j] - x[i];
                    possible[i] = true;
                    j++;
                } else {
                    break;
                }
            }
        }

        for (int i = 0; i < N; i++) {
            if (possible[i]) {
                System.out.println("Possible");
            } else {
                System.out.println("Impossible");
            }
        }
    }
}
