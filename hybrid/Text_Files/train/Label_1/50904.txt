import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        int g = 0;
        for (int i = 0; i < n; i++) {
            g = gcd(g, a[i]);
        }

        if (g > 1) {
            System.out.println(-1);
            return;
        }

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) continue;
            if (i == 0) {
                ans += a[i] - 1;
                a[i + 1] += a[i];
                a[i] = 0;
            } else if (i == n - 1) {
                ans += a[i] - 1;
                a[i - 1] += a[i];
                a[i] = 0;
            } else {
                int l = a[i - 1];
                int r = a[i + 1];
                if (l > r) {
                    ans += a[i] - r;
                    a[i] -= a[i] - r;
                    a[i - 1] += a[i] - r;
                } else if (l < r) {
                    ans += a[i] - l;
                    a[i] -= a[i] - l;
                    a[i + 1] += a[i] - l;
                } else {
                    ans += a[i] - 1;
                    a[i] = 0;
                    a[i - 1] += 1;
                    a[i + 1] += 1;
                }
            }
        }

        System.out.println(ans);
    }

    static int gcd(int a, int b) {
        if (a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }
}
