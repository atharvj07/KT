from collections import defaultdict
import sys
n, s = map(int, input().split())
d, p, q = [], [], []
for i in range(n):
    t,x,y=input().split()
    d.append(t);p.append(int(x));q.append(int(y))
t=defaultdict(list)
for i in range(n):
    t[d[i]].append((p[i],q[i]))
for key in t:
    t[key].sort(reverse=key=='S')
b = []
for i in t['B']:
    if len(b)==s:break
    b.append(i)
s = []
for i in t['S']:
    if len(s)==s:break
    s.append(i)
for i in s:print('S',i[0],i[1])
for i in b:print('B',i[0],i[1])
