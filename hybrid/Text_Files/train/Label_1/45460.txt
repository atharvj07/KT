import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int len = s.length();
        char[] stack = new char[len];
        int sp = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < len; i++) {
            char c = s.charAt(i);
            switch (c) {
                case '(':
                case '_':
                    stack[sp++] = c;
                    break;
                case ')':
                case '>':
                    if (sp > 0) {
                        sb.append(stack[sp - 1] == '(' ? 0 : 1);
                        sp--;
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                case '[':
                case '+':
                case '*':
                case '-':
                    stack[sp++] = c;
                    break;
                case '<':
                    if (sp > 1) {
                        sb.append(stack[sp - 1] == '[' ? 0 : 1);
                        sp--;
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                case '.':
                    if (sp > 0) {
                        stack[sp - 1] = (char) ((stack[sp - 1] - '0' + 1) % 10 + '0');
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                case ',':
                    if (sp > 0) {
                        sp--;
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                case ']':
                case '}':
                    System.out.println("false");
                    return;
            }
        }
        if (sp == 0) {
            System.out.println(sb);
        } else {
            System.out.println("false");
        }
    }
}
