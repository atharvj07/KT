import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        if (n % (long)Math.pow(10, k) == 0)
            System.out.println(0);
        else
            System.out.println(String.valueOf(n).length() - String.valueOf((n/(long)Math.pow(10, k))*Math.pow(10, k)).length());
    }
}
