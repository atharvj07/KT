#include <bits/stdc++.h>
using namespace std;

#define MAXN 2005

typedef long long ll;
typedef pair<int, int> pi;

const ll MOD = 998244353;

int n, m;
char grid[MAXN][MAXN];
int dx[] = {-1, 0, 1, 0};
int dy[] = {0, 1, 0, -1};

ll dp[MAXN][MAXN][2][2]; // dp[i][j][d][f] = number of ways to end at (i, j) with direction d and facing f
bool vis[MAXN][MAXN][2][2];

// Check if (i, j) is a valid cell
bool valid(int i, int j) {
    return 1 <= i && i <= n && 1 <= j && j <= m && grid[i][j] == '.';
}

// Calculate the number of ways to end at (i, j) with direction d and facing f
ll solve(int i, int j, int d, int f) {
    if (vis[i][j][d][f]) return dp[i][j][d][f];
    vis[i][j][d][f] = true;

    // Base case: (i, j) is not a valid cell
    if (!valid(i, j)) return 0;

    // If (i, j) is the start of a pipe
    if (grid[i][j] == '*') {
        dp[i][j][d][f] = 1;
        return 1;
    }

    // If (i, j) is not the start of a pipe
    dp[i][j][d][f] = 0;

    // Try all possible directions and facing
    for (int k = 0; k < 4; k++) {
        int ni = i + dx[k], nj = j + dy[k];

        // If the next cell is valid and not facing the current direction
        if (valid(ni, nj) && (k + 2) % 4 != d) {
            dp[i][j][d][f] = (dp[i][j][d][f] + solve(ni, nj, k, (k + 1) % 2)) % MOD;
        }

        // If the next cell is valid and facing the current direction
        if (valid(ni, nj) && (k + 2) % 4 == d) {
            dp[i][j][d][f] = (dp[i][j][d][f] + solve(ni, nj, k, f)) % MOD;
        }
    }

    return dp[i][j][d][f];
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> grid[i][j];
        }
    }

    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            for (int d = 0; d < 4; d++) {
                for (int f = 0; f < 2; f++) {
                    ans = (ans + solve(i, j, d, f)) % MOD;
                }
            }
        }
    }

    cout << ans << endl;

    return 0;
}
