#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k1, k2, a, b;
    cin >> n >> k1 >> k2;
    vector<int> vA, vB;
    while (n--) cin >> a >> b, vA.push_back(a - b);
    sort(vA.begin(), vA.end());
    long long ans = 0;
    for (auto &x : vA) ans += 1ll * x * x;
    int i = 0, j = vA.size() - 1;
    while (k1-- && k2--) {
        if (abs(vA[i]) < abs(vA[j])) {
            ans += -2ll * vA[i];
            i++;
        } else {
            ans += -2ll * vA[j];
            j--;
        }
    }
    while (k1--) {
        if (vA[i] > 0) {
            ans += -2ll * vA[i];
            i++;
        } else
            break;
    }
    while (k2--) {
        if (vA[j] < 0) {
            ans += -2ll * vA[j];
            j--;
        } else
            break;
    }
    cout << ans << endl;
    return 0;
}
