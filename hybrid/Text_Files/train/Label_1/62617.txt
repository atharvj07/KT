from math import gcd

def possible_split(n, arr):
    # Check if the overall GCD is 1
    overall_gcd = arr[0]
    for num in arr[1:]:
        overall_gcd = gcd(overall_gcd, num)
    if overall_gcd == 1:
        return "YES", [1] * n

    # Try to find a split
    for i in range(n):
        temp_gcd = arr[0] if i != 0 else arr[1]
        for j in range(1, n):
            if i == j:
                continue
            temp_gcd = gcd(temp_gcd, arr[j])
        if temp_gcd == 1:
            result = [2 if j == i else 1 for j in range(n)]
            return "YES", result

    return "NO", []

n = int(input().strip())
arr = list(map(int, input().strip().split()))

ans, split = possible_split(n, arr)
print(ans)
if ans == "YES":
    print(' '.join(map(str, split)))
