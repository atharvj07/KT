#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 3e5+5;
int n, k;
ll x[maxn], y[maxn];
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; ++i) cin >> x[i];
  for (int i = 0; i < n; ++i) cin >> y[i];
  ll tt = 0;
  for (int i = 0; i < n; ++i) {
    tt = tt + x[i] - k;
    if (tt < 0) {
      cout << "NO" << endl;
      return 0;
    }
    tt += y[i];
  }
  tt = 0;
  for (int i = 0; i < n; ++i) {
    tt = tt + y[i] - k;
    if (tt < 0) {
      cout << "NO" << endl;
      return 0;
    }
    tt += x[i];
  }
  cout << "YES" << endl;
  return 0;
}
