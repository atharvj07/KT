from collections import deque

n,m=map(int,input().split())
c=list(map(int,input().split()))
adj=[[] for i in range(n)]
for i in range(m):
    u,v=map(int,input().split())
    adj[u-1].append(v-1)
    adj[v-1].append(u-1)
vis=[0]*n
fin=[0]*n
mn=1e19
for i in range(n):
    if vis[i]==0:
        q=deque()
        ans=0
        vis[i]=1
        q.append(i)
        while q:
            p=q.popleft()
            if fin[p]==0:
                ans+=c[p]
                fin[p]=1
            for j in adj[p]:
                if vis[j]==0:
                    vis[j]=1
                    q.append(j)
        mn=min(mn,ans)
print(mn)
