#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define M 1000000009
ll phi[100005];
ll powMod(ll a, ll b){
    a %= M, b %= (M - 1);
    ll ret = 1;
    while(b > 0){
        if(b & 1) ret = ret * a % M;
        a = a * a % M;
        b >>= 1;
    }
    return ret;
}
ll nPhi(int n){
    if(n < 2) return n;
    if(phi[n] != -1) return phi[n];
    ll ans = n * (n - 1);
    for(int i = 2; i * i <= n; i++){
        if(n % i == 0){
            while(n % i == 0) n /= i;
            ans /= i;
            ans *= (i - 1);
        }
    }
    if(n > 1) ans /= n, ans *= (n - 1);
    return phi[n] = ans % M;
}
int main(){
    memset(phi,-1,sizeof phi);
    ll n;
    cin >> n;
    int cnt = 0;
    for(int i = 1; i * i <= n; i++){
        if(n % i == 0) {
            cnt += (n / i != i);
            cnt += (n / i % M * nPhi(i)) % M;
            cnt %= M;
        }
    }
    cout << cnt << endl;
    for(int i = 1; i <= n; i++){
        if(n % i == 0 && (n / i != i)){
            cout << i << " " << n / i << endl;
        }
    }
    for(int i = 1; i <= n; i++){
        if(n % i == 0){
            cout << i << " " << i << endl;
        }
    }
    return 0;
}
