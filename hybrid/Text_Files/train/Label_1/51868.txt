import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            long n = in.nextLong(), x = in.nextLong(), m = in.nextLong();
            long ans = 0;
            if (m == 0) {
                ans = 1;
            } else if (m == 1) {
                if (in.nextInt() == in.nextInt()) {
                    ans = 2;
                }
            } else if (m == 2) {
                long l1 = in.nextInt(), r1 = in.nextInt();
                long l2 = in.nextInt(), r2 = in.nextInt();
                if (l1 == r1 && l2 == r2) {
                    ans = 2;
                } else if (l1 == r1) {
                    if (l2 <= x && x <= r2) {
                        ans = 3;
                    } else {
                        ans = 1;
                    }
                } else if (l2 == r2) {
                    if (l1 <= x && x <= r1) {
                        ans = 3;
                    } else {
                        ans = 1;
                    }
                } else {
                    ans = 1;
                }
            } else {
                ans = 1;
            }
            System.out.println(ans);
        }
    }
}
