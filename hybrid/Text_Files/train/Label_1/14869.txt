#include<bits/stdc++.h>
#define LL long long
using namespace std;
LL n,m,a[1003],ans,sum,vis[1003];
struct edge{
    LL to,val,cost;
}e[2003];
vector<edge>g[1003];
void dfs(LL now){
    vis[now]=1;
    LL sz=g[now].size();
    for(LL i=0;i<sz;i++){
        LL to=g[now][i].to,val=g[now][i].val,cost=g[now][i].cost;
        if(!vis[to]){
            dfs(to);
            ans+=cost;
            sum+=val;
        }
    }
}
int main(){
    cin>>n>>m;
    for(LL i=1;i<=n;i++){
        cin>>a[i];
    }
    for(LL i=1;i<=m;i++){
        LL x,y;
        cin>>x>>y;
        e[i].to=y;
        e[i].val=a[y];
        e[i].cost=a[x];
        g[x].push_back(e[i]);
        e[i].to=x;
        e[i].val=a[x];
        e[i].cost=a[y];
        g[y].push_back(e[i]);
    }
    dfs(1);
    cout<<ans<<endl;
    return 0;
}
