#include <bits/stdc++.h>
using namespace std;
bool ok(vector<vector<char>> &v, int k) {
  for (int i = 0; i < 4; i++) {
    vector<int> t;
    for (int j = 0; j < 4; j++) {
      if (v[i][j] != '.') t.push_back(v[i][j] - '0');
    }
    if (t.size() > k * 2) return false;
  }
  for (int j = 0; j < 4; j++) {
    vector<int> t;
    for (int i = 0; i < 4; i++) {
      if (v[i][j] != '.') t.push_back(v[i][j] - '0');
    }
    if (t.size() > k * 2) return false;
  }
  return true;
}
int main() {
  int k;
  cin >> k;
  vector<vector<char>> v(4, vector<char>(4));
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) { cin >> v[i][j]; }
  }
  if (ok(v, k))
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
