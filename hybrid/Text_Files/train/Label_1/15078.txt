#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int power(int a, int b){
	int ret = 1;
	while(b){
		if(b & 1){
			ret *= a;
			ret %= mod;
		}
		a *= a;
		a %= mod;
		b >>= 1;
	}
	return ret;
}
int main(){
	int n;
	cin >> n;
	int a1 = 1, a2 = 1;
	while(n > 1){
		a1 *= n;
		a1 %= mod;
		a2 *= power(2, n - 1);
		a2 %= mod;
		a1 -= a2;
		a1 %= mod;
		n -= 2;
	}
	a1 = (a1 + mod) % mod;
	cout << a1 << endl;
	return 0;
}
