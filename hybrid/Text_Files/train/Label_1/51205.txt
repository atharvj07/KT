import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int m = in.nextInt();
        int y = in.nextInt();
        System.out.println(Math.max(n - Math.abs(x - y), 0) * Math.max(m - Math.abs(x - y), 0) + 4);
    }
}
