# include <bits/stdc++.h>
using namespace std;

const int mod = 998244353;
int pow2[200005];

int main() {
    int n; cin >> n;
    vector<long long> a(n), b(n), sum(n), pref(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    for (int i = 0; i < n; ++i) sum[i] = 1LL * a[i] * b[i];
    for (int i = 0; i < n; ++i) {
        if (i > 0) pref[i] = pref[i - 1];
        pref[i] += sum[i];
    }
    sort(b.begin(), b.end());
    sort(a.rbegin(), a.rend());
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        pow2[i] = 1LL * (i % 2 == 0 ? 1 : -1) * (1LL << i);
        ans += pow2[i] * 1LL * a[i] * b[i];
        if (i > 0) ans += pow2[i] * pref[i - 1];
    }
    cout << (ans % mod + mod) % mod << endl;
    return 0;
}
