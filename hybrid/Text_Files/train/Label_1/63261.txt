def max_happiness(n, m, d, fireworks):
    prev_time = 0
    dp = [0] * n
    for a, b, t in fireworks:
        a -= 1  # Convert to 0-based indexing
        max_move = d * (t - prev_time)
        new_dp = [0] * n
        maxl = [0] * n
        maxr = [0] * n
        # Calculate prefix max
        maxl[0] = dp[0] - abs(0 - a)
        for i in range(1, n):
            maxl[i] = max(maxl[i-1], dp[i] - abs(i - a))
        # Calculate suffix max
        maxr[n-1] = dp[n-1] - abs(n-1 - a)
        for i in range(n-2, -1, -1):
            maxr[i] = max(maxr[i+1], dp[i] - abs(i - a))
        for i in range(n):
            left = max(0, i - max_move)
            right = min(n-1, i + max_move)
            new_dp[i] = max(maxl[right], maxr[left]) + b
        dp = new_dp
        prev_time = t
    return max(dp)

n, m, d = map(int, input().split())
fireworks = [list(map(int, input().split())) for _ in range(m)]
print(max_happiness(n, m, d, fireworks))
