#include <iostream>
#include <utility>
int main() {
  int n, m, q;
  std::cin >> n >> m >> q;
  while (q--) {
    int sx, sy, ex, ey;
    std::cin >> sx >> sy >> ex >> ey;
    if (sx == ex && sy == ey ||
        (sx == 1 && ex == 2 && sy == ey) ||
        (sx == 2 && ex == 1 && (sy == ey || (sy % n == ey % m))))
      std::cout << "YES\n";
    else
      std::cout << "NO\n";
  }
  return 0;
}
