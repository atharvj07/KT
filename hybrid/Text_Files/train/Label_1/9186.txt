#include <iostream>

long long dp[100001];

int main() {
  int t;
  std::cin >> t;
  dp[2] = 1;
  dp[3] = 7;
  dp[4] = 11;
  dp[5] = 13;
  dp[6] = 14;
  dp[7] = 15;
  for (int n = 8; n <= 100000; ++n) {
    dp[n] = dp[n - 2] + dp[n - 4] + dp[n - 6] + dp[n - 7];
  }
  while (t--) {
    int n;
    std::cin >> n;
    std::cout << dp[n] << '\n';
  }
  return 0;
}
