#include <cstdio>
#include <cstring>
#include <algorithm>

using namespace std;

#define MAX_N 1000

int n;
int deg[MAX_N + 5], vis[MAX_N + 5];
int main() {
    scanf("%d", &n);
    for (int i = 1; i < n; i++) {
        int u, v;
        scanf("%d%d", &u, &v);
        deg[u]++;
        deg[v]++;
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (deg[i] == 1) { cnt++; vis[i] = true; }
    }
    for (int i = 1; i <= n; i++) {
        if (vis[i]) {
            deg[i] = 0;
            for (int j = 1; j <= n; j++) {
                if (vis[j] && i != j) {
                    deg[j]--;
                    if (deg[j] == 1) { cnt++; vis[j] = true; }
                }
            }
        }
    }
    printf("%d\n", cnt);
    return 0;
}
