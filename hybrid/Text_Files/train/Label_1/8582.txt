#include<iostream>
#include<vector>
using namespace std;
int main()
{
    int n,last=-1e9,k=0;
    cin>>n;
    vector<int>v(n);
    for(int i=0;i<n;i++)
        cin>>v[i];
 
    for(int i=0;i<n;i++)
    {
        if(v[i]>last)
        {
            last=v[i];
            if(k!=0)  cout<<endl;
            cout<<v[i]<<" ";
            k++;
        }
    }
    cout<<endl;
    return 0;
}
