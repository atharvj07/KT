#include <iostream>
#include <string>
#include <vector>
using namespace std;
long long n, m, ans, f[100005][26]; string s;
int main(){
    cin >> n >> m >> s;
    f[n][s[n-1]-'a'] = 1;
    for (long long i=n-1;i>=1;i--){
        for (long long j=0;j<26;j++){
            for (long long k=0;k<26;k++)
                f[i][j] += f[i+1][k];
            f[i][j] -= f[i+1][j];
            if (j==s[i-1]-'a') f[i][j]++;
        }
    }
    for (long long i=0;i<26;i++) ans+=f[1][i];
    cout << ans << endl;
    return 0;
}
