n = int(input())
sequence = input()

def min_time_to_correct(n, sequence):
    if n % 2 != 0:
        return -1
    balance = 0
    max_unbalance = 0
    for char in sequence:
        if char == '(':
            balance += 1
        else:
            balance -= 1
        max_unbalance = min(max_unbalance, balance)
    if balance != 0:
        return -1
    return abs(2 * max_unbalance)

print(min_time_to_correct(n, sequence))
