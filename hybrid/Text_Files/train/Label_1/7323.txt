#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9+7;
const int inf = 1e9;
#define all(v) (v).begin(),(v).end()
ll powmod (ll a, ll p) {
  ll ans = 1;
  while(p) {
    if(p&1) ans = ans * a % mod;
    a = a * a % mod;
    p >>= 1;
  }
  return ans;
}

bool valid(string s) {
  int bal = 0;
  for(char c: s) {
    bal += c == '(' ? 1: -1;
    if(bal < 0) return 0;
  }
  return bal == 0;
}

void solve() {
  string s;
  cin >> s;
  ll n = s.size();
  string a = "(", b = "(";
  for(ll i = 1; i < n; i++) {
    if(s[i] == s[i-1]) {
      if(s[i] == '0') {
        if(i&1) a += ")", b += "(";
        else a += "(", b += ")";
      } else {
        a += "(", b += "(";
      }
    } else {
      if(i&1) a += ")", b += "(";
      else a += "(", b += ")";
    }
  }
  if(a[n-1] == ')') {
    a[n-1] = '(';
    b[n-1] = ')';
  }
  if(valid(a) and valid(b)) {
    cout << "YES\n" << a << '\n' << b << '\n';
  } else {
    cout << "NO\n";
  }
}
int main()
{
  ll t;
  cin >> t;
  while(t--) {
    solve();
  }
  return 0;
}
