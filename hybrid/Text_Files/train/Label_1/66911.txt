import sys
def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
        ok = True
        for i in range(n):
            if a[i] > b[i]:
                ok = False
        if not ok: print('NO')
        else: print('YES')
solve()
