#include <bits/stdc++.h>
using namespace std;
int main() {
    cin.tie(NULL), ios_base::sync_with_stdio(false);
    int n, k;
    cin >> n >> k;
    vector<int> c(n), id(k);
    for (int i = 0; i < n; i++) cin >> c[i];
    for (int i = 0; i < k; i++) { cin >> id[i]; id[i]--; }
    sort(id.begin(), id.end());
    sort(c.rbegin(), c.rend());
    long long sum = 0;
    for (int i = 0, j = 0; i < n; i++) {
        if (j < k && i == id[j])
            j++, sum += 1ll * c[i] * c[0];
        else
            sum += 1ll * c[i] * c[i + 1];
    }
    cout << sum;
}
