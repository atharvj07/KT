import queue

n, T = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(n)]
arr.sort(key=lambda x: -x[1]/x[0])
q = queue.PriorityQueue()
sm = 0
ans = 0
for task in arr:
    dur, val = task
    if sm+dur <= T:
        sm += dur
        q.put((-val, dur))
        ans += val
    else:
        while(sm > T and q.empty() == False):
            sm-=q.get()[1]
        if sm+dur <= T:
            sm += dur
            q.put((-val, dur))
            ans += val
print(ans)
