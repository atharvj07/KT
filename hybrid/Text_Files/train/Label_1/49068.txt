import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int n, h;
            while ((n = sc.nextInt()) != 0 && (h = sc.nextInt()) != 0) {
                boolean[][][] visited = new boolean[n + 1][n + 1][n + 1];
                for (int i = 0; i < h; i++) {
                    String dir = sc.next();
                    int d = sc.nextInt(), s = sc.nextInt();
                    if (dir.equals("xy")) {
                        for (int k = 1; k <= n; k++) {
                            visited[d][s][k] = true;
                        }
                    } else if (dir.equals("xz")) {
                        for (int j = 1; j <= n; j++) {
                            visited[d][j][s] = true;
                        }
                    } else if (dir.equals("yz")) {
                        for (int k = 1; k <= n; k++) {
                            visited[s][d][k] = true;
                        }
                    }
                }
                int cnt = 0;
                for (int i = 1; i <= n; i++) {
                    for (int j = 1; j <= n; j++) {
                        for (int k = 1; k <= n; k++) {
                            if (!visited[i][j][k]) {
                                cnt++;
                            }
                        }
                    }
                }
                System.out.println(cnt);
            }
        }
    }
}
