#include<vector>
#include<iostream>
#include<functional>
using namespace std;
int main(){
    int n; cin >> n; vector<long long int> a(n);
    for(int i=0;i<n;++i) cin >> a[i];
    long long int s = 0; for(int i=0;i<n;++i) s += a[i];
    vector<int> res;
    for(int i=0;i<n;++i) if(s - a[i] == a[i]) res.push_back(i+1);
    cout << res.size() << '\n';
    for(int i: res) cout << i << ' ';
    cout << '\n';
}
