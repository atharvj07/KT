#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    while (n--) {
        long long p, q, b;
        cin >> p >> q >> b;
        if (q % b != 0) {
            cout << "Finite\n";
            continue;
        }
        while (p < q && q % b == 0) {
            p *= 10;
            q *= b;
        }
        if (p % q == 0) {
            cout << "Finite\n";
        } else {
            cout << "Infinite\n";
        }
    }
}
