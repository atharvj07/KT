#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 100005;
typedef long long ll;

int n, p;
ll a[MAXN];
ll f[MAXN]; // f[i] = the number of permutations for i candies
vector<ll> ans;

int main() {
    cin >> n >> p;
    for (int i = 1; i <= n; i++) cin >> a[i];

    // Calculate f[i] for all i
    f[0] = 1;
    for (int i = 1; i <= n; i++) {
        f[i] = 0;
        for (int j = 0; j < i; j++) {
            if ((a[i] >= a[j]) && (f[j] % p == 0)) {
                f[i] += f[j];
                f[i] %= p;
            }
        }
    }

    // Find all good integers x
    for (int i = 1; i <= n; i++) {
        if (f[i] % p != 0) ans.push_back(i);
    }

    // Print the number of good integers x
    cout << ans.size() << endl;

    // Print all good integers x in the ascending order
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;

    return 0;
}
