#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, K;
    cin >> n >> K;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = n;
    for (int i = 0; i < n; i++) {
        if (a[i] + K < a[i + 1]) {
            ans--;
        }
    }
    cout << ans << '\n';
    return 0;
}
