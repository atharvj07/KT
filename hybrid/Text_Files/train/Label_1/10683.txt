#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int g = a[0];
    for (int i = 1; i < n; i++) g = gcd(g, a[i]);
    for (int i = 0; i < n; i++) a[i] = (a[i] + k - g) % k + g;
    int ans = a[0];
    for (int i = 1; i < n; i++) ans = gcd(ans, a[i]);
    cout << ans << endl;
    return 0;
}
