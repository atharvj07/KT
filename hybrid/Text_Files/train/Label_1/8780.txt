#include <algorithm>
#include <iostream>
#include <set>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    set<int> s(begin(a), end(a));
    while (k--) {
      int mex = *s.begin();
      int max = *s.rbegin();
      int add = (mex + max + 1) / 2;
      if (!s.count(add)) {
        s.insert(add);
      }
    }
    cout << s.size() << '\n';
  }
  return 0;
}
