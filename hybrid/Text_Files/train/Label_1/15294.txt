#include <iostream>
#include <string>
using namespace std;

int main() {
    int n, a, b;
    string s;
    cin >> n >> a >> b;
    cin >> s;
    int ans = 0;
    bool a_prev = false, b_prev = false;
    for (int i = 0; i < n; i++) {
        if (s[i] == '.') {
            if (a_prev && b_prev) continue;
            if (a && !a_prev) {
                ans++;
                a--;
                a_prev = true;
                b_prev = false;
            } else if (b && !b_prev) {
                ans++;
                b--;
                b_prev = true;
                a_prev = false;
            }
        } else {
            a_prev = false;
            b_prev = false;
        }
    }
    cout << ans << endl;
    return 0;
}
