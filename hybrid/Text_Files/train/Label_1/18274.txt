#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
const int inf = 0x3f3f3f3f;
int n, k, q;
int pre[maxn << 1];
int main() {
  cin >> n >> k >> q;
  memset(pre, 0, sizeof(pre));
  for (int i = 1; i <= n; ++i) {
    int l, r;
    cin >> l >> r;
    pre[l]++;
    pre[r + 1]--;
  }
  for (int i = 1; i <= 200000; ++i) { pre[i] += pre[i - 1]; }
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << (pre[r] - ((l > k) ? pre[l - 1] : 0)) << endl;
  }
  return 0;
}
