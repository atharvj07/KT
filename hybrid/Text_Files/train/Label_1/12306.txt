#include <bits/stdc++.h>
using namespace std;

int n, ans;
struct node{
    int x, a;
    bool operator<(const node &other) const{
        return x < other.x;
    }
};
vector<node> v;

int main(){
    cin >> n;
    v.resize(n);
    for(int i = 0; i < n; ++i){
        cin >> v[i].x >> v[i].a;
    }
    sort(v.begin(), v.end());
    int sum = 0;
    for(int i = 0; i < n; ){
        int j = i;
        while(j < n && v[j].x == v[i].x){
            sum += v[j].a;
            ++j;
        }
        ans = max(ans, sum);
        sum = 0;
        i = j;
    }
    cout << ans << endl;
}
