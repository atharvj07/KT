import math

def solve(n, miners, mines):
    miners.sort()
    mines.sort()
    energy = 0
    for i in range(n):
        energy += math.sqrt(miners[i]**2 + mines[i]**2)
    return energy

t = int(input())
for _ in range(t):
    n = int(input())
    miners = []
    mines = []
    for _ in range(2*n):
        x, y = map(int, input().split())
        if x == 0:
            miners.append(abs(y))
        else:
            mines.append(abs(x))
    print(solve(n, miners, mines))
