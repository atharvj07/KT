import java.util.Scanner;

public class VovaAndNumber {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int k = 0;
        for (int x = 1; x <= n; x++) {
            int sumOfDigits = 0;
            int temp = x;
            while (temp > 0) {
                sumOfDigits += temp % 10;
                temp /= 10;
            }
            if (x + sumOfDigits == n) {
                k++;
                System.out.println(x);
            }
        }

        System.out.println(k);
    }
}
