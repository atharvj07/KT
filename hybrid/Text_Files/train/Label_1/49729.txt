import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        s = s.replaceAll(" ", "");

        boolean[] used = new boolean[26];
        for (int i = 0; i < s.length(); ++i) {
            if (Character.isLetter(s.charAt(i)) && !used[s.charAt(i) - 'a']) {
                used[s.charAt(i) - 'a'] = true;
                char c = 'a';
                while (s.indexOf(c) != -1) {
                    s = s.replaceAll(String.valueOf(c), String.valueOf((char) (i + 'a')));
                    c++;
                }
            }
        }
        boolean valid = true;
        for (int i = 0; i < s.length(); ++i) {
            if (Character.isLetter(s.charAt(i)) && s.charAt(i) != (char) (i + 'a')) {
                valid = false;
            }
        }

        System.out.println(valid ? "YES" : "NO");
    }
}
