import java.util.HashSet;
import java.util.Scanner;

public class Shockers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        HashSet<Character> possibleLetters = new HashSet<>();
        for (int i = 0; i < 26; i++) {
            possibleLetters.add((char) ('a' + i));
        }

        int excessiveShocks = 0;

        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(0);

            if (firstChar == '.') {
                String word = action.substring(2);
                for (char c : word.toCharArray()) {
                    possibleLetters.remove(c);
                }
            } else if (firstChar == '!') {
                String word = action.substring(2);
                if (word.contains(String.valueOf(possibleLetters.iterator().next()))) {
                    excessiveShocks++;
                }
            } else if (firstChar == '?') {
                char guess = action.charAt(2);
                if (possibleLetters.size() == 1 && possibleLetters.contains(guess)) {
                    break;
                } else {
                    possibleLetters.remove(guess);
                }
            }
        }

        System.out.println(excessiveShocks);
    }
}
