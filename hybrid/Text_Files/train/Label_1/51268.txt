        int N = sc.nextInt();
        long D = sc.nextLong();

        for (int i = 1; i <= N; i++) {
            d[i] = sc.nextLong();
            sum += d[i];
        }
        int Q = sc.nextInt();
        for (int i = 1; i <= Q; i++) {
            long q = sc.nextLong();
            if (sum - d[i] + D >= 0) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
