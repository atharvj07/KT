#include <algorithm>
#include <deque>
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    deque<int> s;
    s.push_back(1);
    while (s.size() != n) {
        int b = s.back();
        s.pop_back();
        int f = s.back();
        if (f == b)
            s.back()++, s.push_back(1);
        else
            s.push_back(1), s.push_back(b);
    }
    while (!s.empty()) {
        cout << s.front() << ' ';
        s.pop_front();
    }
    return 0;
}
