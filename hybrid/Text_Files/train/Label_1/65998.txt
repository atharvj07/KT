from itertools import combinations

n, k, p = map(int, input().split())
a = list(map(int, input().split()))
sump, sumk = 0, 0
for x in a:
    sump += x % 2
    sumk += 1 - x % 2

if sump < p or sumk < k - p:
    print('NO')
else:
    print('YES')
    odd = sorted([(x, i + 1) for i, x in enumerate(a) if x % 2])
    even = sorted([(x, i + 1) for i, x in enumerate(a) if not x % 2])
    for i in range(p):
        print(1, odd[i][1])
    for i in range(k - p - 1, -1, -1):
        print(1, even[i][1])
    print(len(odd) - p, *[x[1] for x in odd[p:]])
