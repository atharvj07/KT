import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong();
        long k = input.nextLong();
        long a = input.nextLong();
        long b = input.nextLong();
        long res = 0;
        while (n > 1) {
            if (n % k == 0 && a > b) {
                n /= k;
                res += b;
            } else {
                n -= 1;
                res += a;
            }
        }
        System.out.println(res);
    }
}
