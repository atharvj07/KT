#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> t;
    for (int i = 0; i < t; i++) {
        int x, y;
        cin >> x >> y;
        int MIN = 1e9, MAX = -1e9;
        for (int j = 0; j < x; j++) {
            int a;
            cin >> a;
            MIN = min(a, MIN);
            MAX = max(a, MAX);
        }
        if (MAX - MIN <= 1) { cout << "Yes\n"; }
        else { cout << "No\n"; }
    }
    return 0;
}
