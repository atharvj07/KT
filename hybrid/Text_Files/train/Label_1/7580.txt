#include <bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  string s; cin >> s;
  vector<int> v(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    b[i] = v[i];
  }
  for (int i = 0; i < n / 2; i++) {
    int j = n - 1 - i;
    if (s[i] != s[j]) {
      v[i] ^= v[j];
      v[j] ^= v[i];
      v[i] ^= v[j];
    }
  }
  sort(v.begin(), v.end());
  sort(b.begin(), b.end());
  int ans = 0;
  for (int i = 0; i < n; i++) ans += v[i] * b[i];
  cout << ans << endl;
  return 0;
}
