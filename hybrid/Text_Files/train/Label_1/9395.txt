#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int a[N];

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  long long ans = 0;
  for (int i = 1, r = -1; i <= n; i++) {
    if (a[i] == 1) r = i;
    if (r != -1 && a[i] == 0) ans += r - i, r = i;
  }
  cout << ans << '\n';
}
