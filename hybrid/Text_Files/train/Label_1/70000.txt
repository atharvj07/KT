import sys
input = sys.stdin.readline

from collections import Counter

n, m = map(int, input().split())

l = []
for i in range(n):
    l.append(Counter(map(int, input().split()[1:])))

c = [0]*m

for d in l:
    for i in d:
        c[i - 1] += d[i]

ans = 1
for i in c:
    ans = (ans * i) % (10**9 + 7)
print(ans)
