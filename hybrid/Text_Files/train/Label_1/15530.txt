#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll bit[500005], v[500005];
ll query(ll x) {
    ll sum = 0;
    for (; x; x -= x & -x) sum += bit[x];
    return sum;
}
void update(ll x, ll val) {
    for (; x <= 500000; x += x & -x) bit[x] += val;
}
int main() {
    ll q;
    cin >> q;
    while (q--) {
        ll t, x, y;
        cin >> t >> x >> y;
        if (t == 1) {
            v[x] += y;
            update(x, y);
        } else if (t == 2) {
            cout << query(500000) - query(x - 1) << '\n';
        }
    }
    return 0;
}
