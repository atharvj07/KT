import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int a = sc.nextInt() - 1;
    int b = sc.nextInt();
    int[] t = new int[n];
    int ans = 0;
    for (int i = 0; i < n; i++) {
      t[i] = sc.nextInt() - 1;
      if (!(t[i] >= a && t[i] < b)) {
        ans++;
      }
    }
    System.out.println(ans);
  }
}
