#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size(), ans = n;
        vector<int> dp(n + 1);
        dp[n] = 0;
        for (int i = n - 1; i >= 0; --i) {
            dp[i] = dp[i + 1] + 1;
            if (s[i] == s[i + 1]) { dp[i] = min(dp[i], dp[i + 2]); }
        }
        cout << dp[0] << endl;
    }
    return 0;
}
