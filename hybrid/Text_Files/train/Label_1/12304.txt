#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  bool a[n + 5] = {};
  int p[n + 5] = {};
  int st = 0;
  string s;
  for (int i = 0; i < n * 2; i++) {
    cin >> s;
    if (s == "+") st++;
    else {
      int x;
      cin >> x;
      a[x] = 1;
      while (st > 0 && a[p[st]]) { cout << p[st--] << " "; }
    }
    p[++st] = st;
  }
  if (st == 0) cout << "YES"; else cout << "NO";
}
