import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[6];
        for (int i = 0; i < 6; i++) {
            a[i] = sc.nextInt();
        }
        int n = sc.nextInt();
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }

        int max = b[0], min = b[0];
        for (int i = 1; i < n; i++) {
            max = Math.max(max, b[i]);
            min = Math.min(min, b[i]);
        }
        int minDiff = Integer.MAX_VALUE;
        for (int i = 0; i < 6; i++) {
            int mx = Integer.MIN_VALUE, mi = Integer.MAX_VALUE;
            for (int j = 0; j < n; j++) {
                int fret = b[j] - a[i];
                mx = Math.max(mx, fret);
                mi = Math.min(mi, fret);
            }
            minDiff = Math.min(minDiff, mx - mi);
        }

        System.out.println(minDiff);
    }
}
