from string import ascii_uppercase
from sys import stdin
s = stdin.readline().rstrip()
if len(s) < 26 or s.count('?') > 26 - len(set(''.join(filter(lambda i: i != '?', s)))):
    print(-1)
else:
    t = sorted(''.join(filter(lambda i: i != '?', s)))
    for i in range(len(s)):
        if s[i] == '?':
            s = s[:i] + t.pop() + s[i + 1:]
    print(s)
