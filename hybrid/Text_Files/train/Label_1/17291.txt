#include <bits/stdc++.h>
#define pdd pair<double,double>
#define eps 1e-10
using namespace std;
int sgn(double x){
    return (x<0?-1:(x>0));
}
int n,r[5],x[5],y[5];
const double pi=acos(-1);
int main(){
    while(cin>>n){
        for(int i=0; i<n; i++){
            cin>>x[i]>>y[i]>>r[i];
        }
        set<pdd>st;
        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                if(j==i)continue;
                double dis=sqrt((x[i]-x[j])*(x[i]-x[j])+(y[i]-y[j])*(y[i]-y[j]));
                if(dis>r[i]+r[j])continue;
                double l1,l2;
                if(sgn(x[i]-x[j])==0){
                    l1=0,l2=acos(0);
                }
                else{
                    l1=atan((y[i]-y[j])/(x[i]-x[j]));
                    l2=pi+atan((y[i]-y[j])/(x[i]-x[j]));
                }
                if(l1<l2){
                    swap(l1,l2);
                }
                double alpha=acos((dis*dis+r[i]*r[i]-r[j]*r[j])/(2.0*dis*r[i]));
                l1=l1-alpha;
                l2=l2+alpha;
                if(l1>l2){
                    swap(l1,l2);
                }
                l1=max(l1,0.0);
                l2=min(l2,2.0*pi);
                if(l1<l2){
                    st.insert({l1,l2});
                }
            }
        }
        int cnt=0;
        for(set<pdd>::iterator it=st.begin(); it!=st.end(); it++){
            cnt+=it->second-it->first>eps;
        }
        cout<<cnt+1<<endl;
    }
    return 0;
}
