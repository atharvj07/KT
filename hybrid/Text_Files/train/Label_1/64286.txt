from sys import stdin
input = stdin.readline
n, m = map(int, input().split())
a = [int(_) for _ in input().split()]
need = [0]*(m+1)
day, done = 0, 0
for i in range(m, -1, -1):
    if need[i]: done += 1
    if done >= 1: need[i-1] += 1
    x = min(need[i], len(a))
    while x:
        done += max(0, a.pop() - day)
        day += 1
        x -= 1
    if done >= 1: need[i-1] += 1
if done < 1: print(-1)
else: print(day)
