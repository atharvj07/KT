from collections import defaultdict
from functools import reduce
from operator import mul

n, m = map(int, input().split())
c = list(map(int, input().split()))
a = [0]*m
b = [0]*m
for i in range(m):
	a[c[i]-1] = i
	b[c[i]-1] = i
for i in range(1, m):
	if a[i] < a[i-1]:
		print(0)
		exit()
	if b[i] < b[i-1]:
		print(0)
		exit()
for l in range(m):
	for r in range(l, m):
		if a[r] < a[l]:
			print(0)
			exit()
		if b[r] < b[l]:
			print(0)
			exit()
f = defaultdict(int)
for x in c:
	f[x] += 1
p = 1
for x in f:
	p *= x**f[x]
q = 1
for x in f:
	q *= f[x]
print((m-n+1)**q * p // reduce(mul, [i+1 for i in range(m-n+1)]))
