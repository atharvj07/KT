n, m, k = map(int, input().split())
layers = []
for _ in range(n):
    layer = [input() for _ in range(m)]
    layers.append(layer)
    if _ < n - 1:
        input()  # to consume the empty line

critical_cpus = 0

for x in range(n):
    for y in range(m):
        for z in range(k):
            if layers[x][y][z] == '1':
                for dx, dy, dz in [(1, 0, 0), (0, 1, 0), (0, 0, 1)]:
                    nx, ny, nz = x + dx, y + dy, z + dz
                    if 0 <= nx < n and 0 <= ny < m and 0 <= nz < k:
                        if layers[nx][ny][nz] == '1':
                            px, py, pz = x - dx, y - dy, z - dz
                            if 0 <= px < n and 0 <= py < m and 0 <= pz < k:
                                if layers[px][py][pz] == '0':
                                    critical_cpus += 1
                                    break
                            else:
                                critical_cpus += 1
                                break

print(critical_cpus)
