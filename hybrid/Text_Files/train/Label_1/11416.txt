#include <cstdio>
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int maxN = 5e5 + 5;
int a[maxN], f[maxN], l[maxN], r[maxN], c[maxN];
pair<int, int> ans = {1, 1};
int n, k, lst, cnt;

int main() {
    scanf("%d%d", &n, &k);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &a[i]);
    }
    for (int i = 1; i <= n; i++) {
        if (i > 1 && a[i] == a[i - 1]) cnt++;
        else cnt = 1;
        c[i] = cnt;
        l[i] = max(l[i], i - cnt + 1);
        if (c[i - cnt] > 1) l[i] = max(l[i], l[i - cnt]);
        r[i] = i;
    }
    for (int i = n; i >= 1; i--) {
        f[i] = f[i + 1];
        if (c[i] > 1) f[i] = max(f[i], r[i]);
        else f[i] = max(f[i], r[i - 1]);
    }
    for (int i = 1; i <= n; i++) {
        if (r[i] - l[i] + 1 <= k) {
            int j = f[i + 1];
            if (j > ans.second - ans.first + 1) {
                ans = {l[i], r[i]};
            }
        }
    }
    printf("%d %d\n", ans.first, ans.second);
    return 0;
}
