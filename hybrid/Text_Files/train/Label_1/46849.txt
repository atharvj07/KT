import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int s = in.nextInt();
        int b = in.nextInt();
        int[] a = new int[s];
        for (int i = 0; i < s; i++) {
            a[i] = in.nextInt();
        }
        int[][] d = new int[b][2];
        for (int i = 0; i < b; i++) {
            d[i][0] = in.nextInt();
            d[i][1] = in.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(d, (x, y) -> Integer.compare(x[0], y[0]));
        long[] res = new long[s];
        for (int i = 0; i < s; i++) {
            for (int j = 0; j < b; j++) {
                if (a[i] >= d[j][0]) {
                    res[i] += d[j][1];
                } else {
                    break;
                }
            }
        }
        for (int i = 0; i < s; i++) {
            System.out.println(res[i]);
        }
    }
}
