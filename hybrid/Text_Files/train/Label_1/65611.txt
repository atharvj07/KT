n = int(input())
s = [input() for _ in range(n)]
l = []
for i in s:
    a = 0
    for x in i:
        if x == "(":
            a += 1
        else:
            a -= 1
        if a < 0:
            break
    else:
        if a == 0:
            l.append(i)
ans = 0
for i in l:
    for j in l:
        if i != j:
            ans += 1
        else:
            ans += 2
print(ans)
