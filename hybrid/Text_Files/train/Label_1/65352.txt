grid = [[1, 1, 1], [1, 1, 1], [1, 1, 1]]

for i in range(3):
    row = list(map(int, input().split()))
    for j in range(3):
        press_count = row[j]
        if press_count % 2 == 1:
            grid[i][j] ^= 1
            if i > 0:
                grid[i-1][j] ^= 1
            if i < 2:
                grid[i+1][j] ^= 1
            if j > 0:
                grid[i][j-1] ^= 1
            if j < 2:
                grid[i][j+1] ^= 1

for row in grid:
    print(''.join(map(str, row)))
