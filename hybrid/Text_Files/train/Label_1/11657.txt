#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> s(n);
    for (int i = 0; i < n; i++) cin >> s[i];
    int ans = 0;
    for (int i = 0; i < n;) {
      while (i < n && s[i] != 1) {
        s[i] = max(s[i] - 1, 1);
        i = i + s[i];
      }
      i++;
      ans++;
    }
    cout << ans << '\n';
  }
  return 0;
}
