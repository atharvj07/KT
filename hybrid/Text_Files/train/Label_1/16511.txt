#include <iostream>
#include <algorithm>
#define ll long long
using namespace std;

const int maxn = 5e5 + 5, maxm = 55;

struct rec {
    int x1, x2, y1, y2;
} a[maxm];

int main() {
    int n, m, ans = 0;
    cin >> n >> m;
    for (int i = 0; i < m; i++) cin >> a[i].x1 >> a[i].y1 >> a[i].x2 >> a[i].y2;
    int ypos = 1, cnt = 1;
    sort(a, a + m, [](const rec &a, const rec &b) {
        return a.y1 < b.y1;
    });
    for (int i = 0; i < m; i++) {
        cnt = max(cnt, a[i].y1 - ypos);
        ans += ((a[i].x2 - a[i].x1 + 1) * min(a[i].y2 - ypos, cnt));
        ypos = a[i].y2 + 1;
    }
    cout << ans << "\n";
    return 0;
}
