import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            long sum = 0;
            for (int j = 0; j < n; j++) {
                sum += scanner.nextInt();
            }

            System.out.println((sum + n - 1) / n);
        }
    }
}
