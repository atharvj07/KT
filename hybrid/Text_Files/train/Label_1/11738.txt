#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < n; i++)
#define all(x) x.begin(), x.end()
typedef long long int ll;

int main(){
    int n, k;
    cin >> n >> k;
    vector<ll> p(n), e(n);
    rep(i, n) cin >> p[i] >> e[i];
    vector<pair<ll, ll>>vp;
    rep(i, n) vp.push_back({p[i], e[i]});
    sort(all(vp));
    reverse(all(vp));
    ll ans = 0, cnt = 0;
    int flag = 0;
    for(auto x: vp){
        ll point = x.first, effort = x.second;
        if(cnt + max(point, 1ll) < k){
            ans += effort;
            cnt += max(point, 1ll);
        }else{
            if(cnt + 1 == k){
                if(point == 0) ans += effort;
                flag = 1;
                break;
            }else{
                flag = -1;
                break;
            }
        }
    }
    if(cnt < k) flag = -1;
    cout << (flag == 1 ? ans : -1) << endl;
}
