#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<long long> b(n);
  for (long long i = 0; i < n; i++) { cin >> b[i]; }

  vector<long long> a(n, 0);
  a[0] = 1;

  for (long long i = 1; i < n; i++) {
    a[i] = b[i - 1] + 1;
  }

  if (a[n - 1] % a[0] != b[n - 1]) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    for (long long i = 0; i < n; i++) { cout << a[i] << " "; }
    cout << endl;
  }

  return 0;
}
