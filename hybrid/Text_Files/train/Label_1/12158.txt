#include <iostream>
#include <algorithm>

int main() {
    long long a, h, w;
    std::cin >> a >> h >> w;
    long long x = std::min(h / a, w / a);
    if (!x) std::cout << "-1\n";
    else {
        long double eps = 1e-6;
        long double ans = (a - 1.0L * eps) / x;
        std::cout << ans << "\n";
    }
    return 0;
}
