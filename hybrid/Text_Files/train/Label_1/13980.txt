#include <iostream>
#include <vector>

using namespace std;

typedef unsigned long long ll;

int main() {
  int n;
  cin >> n;
  vector<string> v(n);
  for (auto &s : v) cin >> s;
  ll dp[n][n + 1];
  for (int i = n - 1; i >= 0; i--) {
    dp[i][n] = ll(v[i].size());
    for (int j = n - 1; j >= 0; j--) {
      dp[i][j] = dp[i][j + 1];
      int len = min(v[i].size(), v[j].size());
      for (int k = 1; k <= len; k++) {
        if (v[i].substr(0, k) != v[j].substr(0, k)) break;
        dp[i][j] = min(dp[i][j], dp[i + k][j + k] + v[i].size() - k + 1);
      }
    }
  }
  cout << dp[0][0] << endl;
  return 0;
}
