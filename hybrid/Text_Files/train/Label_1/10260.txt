#include <bits/stdc++.h>
using ll = long long;
const ll mod = 1e9 + 7;
ll pow(ll x, ll y) {
    ll ret = 1;
    while (y) {
        if (y & 1) ret = ret * x % mod;
        x = x * x % mod;
        y >>= 1;
    }
    return ret % mod;
}
ll inv(ll x) { return pow(x, mod - 2); }
ll gcd(ll x, ll y) { return y == 0 ? x : gcd(y, x % y); }
ll f(ll x, ll y) {
    if (x > y) std::swap(x, y);
    return x == 0 ? y : f(y % x, x);
}
ll dfs1(const std::vector<std::vector<int>>& adj, std::vector<ll>& g,
        const std::vector<ll>& x, int u, int p) {
    ll ret = 0;
    for (int v : adj[u]) {
        if (v == p) continue;
        ret += dfs1(adj, g, x, v, u);
        g[u] = f(g[u], g[v]);
        g[u] %= mod;
    }
    g[u] = (g[u] * x[u - 1]) % mod;
    return g[u];
}
ll dfs2(const std::vector<std::vector<int>>& adj, std::vector<ll>& g,
        const std::vector<ll>& x, int u, int p) {
    ll ret = 0;
    for (int v : adj[u]) {
        if (v == p) continue;
        ret += dfs2(adj, g, x, v, u);
        ret += (g[u] * inv(g[v])) % mod;
        ret %= mod;
    }
    return ret;
}
int main() {
    int n;
    std::cin >> n;
    std::vector<ll> x(n);
    for (int i = 0; i < n; ++i) std::cin >> x[i];
    std::vector<std::vector<int>> adj(n);
    for (int i = 0; i < n - 1; ++i) {
        int a, b;
        std::cin >> a >> b;
        adj[a - 1].push_back(b - 1);
        adj[b - 1].push_back(a - 1);
    }
    std::vector<ll> g(n, 0);
    dfs1(adj, g, x, 0, 0);
    ll ans = dfs2(adj, g, x, 0, 0);
    std::cout << ans << '\n';
    return 0;
}
