import java.math.BigDecimal;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static class Senator implements Comparable<Senator> {
        int level;
        int loyalty;

        public Senator(int level, int loyalty) {
            this.level = level;
            this.loyalty = loyalty;
        }

        @Override
        public int compareTo(Senator o) {
            return Integer.compare(this.loyalty, o.loyalty);
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int A = input.nextInt();

        Senator[] senators = new Senator[n];
        for (int i = 0; i < n; i++) {
            senators[i] = new Senator(input.nextInt(), input.nextInt());
        }

        Arrays.sort(senators);

        int totalLevel = 0;
        for (int i = 0; i < n; i++) {
            totalLevel += senators[i].level;
        }

        BigDecimal probability = BigDecimal.ZERO;
        for (int i = 0; i < n; i++) {
            int candies = Math.min(k, 10 - senators[i].loyalty / 10);
            k -= candies;
            senators[i].loyalty += candies * 10;
            BigDecimal p = BigDecimal.ONE;
            int totalLevelAgainst = 0;
            for (int j = i + 1; j < n; j++) {
                if (senators[j].loyalty < 50) {
                    totalLevelAgainst += senators[j].level;
                }
            }
            if (totalLevelAgainst > 0) {
                p = p.multiply(BigDecimal.valueOf(A).divide(BigDecimal.valueOf(A + totalLevelAgainst), 10, BigDecimal.ROUND_DOWN));
            }
            probability = probability.add(p);
        }

        System.out.printf("%.10f", probability);
    }
}
