import sys
from collections import defaultdict
for t in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    a = [*map(int,sys.stdin.readline().split())]
    f = defaultdict(int)
    cur,L,ans = 0,n+1,n+1
    for i,v in enumerate(a):
        f[v]+=1
        if f[v]==1:
            cur+=1
        while cur>1:
            f[a[i-L+1]]-=1
            if f[a[i-L+1]]==0:
                cur-=1
            L-=1
        if cur==1:
            ans=min(ans,L)
    print(ans if ans<n+1 else -1)
