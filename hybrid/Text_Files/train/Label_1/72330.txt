N, D = map(int, input().split())
S = list(map(int, input().split()))
P = list(map(int, input().split()))
sumS = [0]
sumP = [0]

for i in S:
    sumS.append(sumS[-1] + i)

for i in P:
    sumP.append(sumP[-1] + i)

now = sumS[D]
ans = N
for i in range(N):
    if now + sumP[i + 1] > sumS[i]:
        ans = i + 1

print(ans)
