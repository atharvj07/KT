N, K = map(int, input().split())
ip = [list(map(int, input().split('.'))) for _ in range(N)]
nm = [0, 0, 0, 0]
for i in range(0, 4):
    for j in range(0, 8):
        b = 1 << (7-j)
        pref = sum(1 for x in ip if x[i] & b)
        rng = pref * 2 + (K > 1 and i < 3)
        if rng <= N:
            nm[i] |= b
        K -= pref >= N - rng
print((".".join(str(x) for x in nm)) if K == 0 else -1)
