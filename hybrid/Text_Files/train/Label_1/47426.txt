import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int k = in.nextInt();
        int ans = (b / k) - ((a - 1) / k);
        System.out.println(ans);
    }
}
