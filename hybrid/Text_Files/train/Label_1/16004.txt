#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  int ans = 1;
  for (int i = 0; i < n; ++i) {
    if (i + 1 < a[i]) {
      ans = -1;
      break;
    }
    for (int j = 0; j < i; ++j) {
      if (a[i] % a[j] == 0 && i + 1 < a[i] / a[j]) {
        ans = -1;
        break;
      }
    }
  }
  cout << ans << '\n';
}
