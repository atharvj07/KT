#include <algorithm>
#include <iostream>
#include <queue>
#include <vector>

using namespace std;

const int MAXN = 2e5 + 10;
vector<int> adj[MAXN];
int val[MAXN];
long long dp[MAXN][2];
bool vis[MAXN];
long long ans;

void dfs(int u, int p) {
  vis[u] = true;
  dp[u][0] = dp[u][1] = val[u];
  for (int v : adj[u]) {
    if (!vis[v]) {
      dfs(v, u);
      dp[u][0] += max(dp[v][0], dp[v][1]);
      dp[u][1] += dp[v][0];
    }
  }
  if (p != -1) {
    ans = max(ans, dp[u][0] + dp[p][1]);
  }
}

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> val[i];
  }
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1, -1);
  if (ans <= 0) {
    cout << "Impossible" << endl;
  } else {
    cout << ans << endl;
  }
}
