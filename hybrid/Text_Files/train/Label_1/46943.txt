import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static int n;
    static int[][] bit;
    static int[][] a;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        String[] parts = line.split(" ");
        n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        bit = new int[n + 1][n + 1];
        a = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            line = br.readLine();
            parts = line.split(" ");
            int type = Integer.parseInt(parts[0]);
            int x0 = Integer.parseInt(parts[1]);
            int y0 = Integer.parseInt(parts[2]);
            int x1 = Integer.parseInt(parts[3]);
            int y1 = Integer.parseInt(parts[4]);
            if (type == 1) {
                System.out.println(query(x0, y0, x1, y1));
            } else {
                long v = Long.parseLong(parts[5]);
                update(x0, y0, x1, y1, v);
            }
        }
    }

    static void update(int x0, int y0, int x1, int y1, long v) {
        update(x0, y0, v);
        update(x0, y1 + 1, -v);
        update(x1 + 1, y0, -v);
        update(x1 + 1, y1 + 1, v);
    }

    static void update(int x, int y, long v) {
        for (; x <= n; x += (x & (-x))) {
            for (int j = y; j <= n; j += (j & (-j))) {
                bit[x][j] ^= v;
            }
        }
    }

    static long query(int x0, int y0, int x1, int y1) {
        return query(x1, y1) ^ query(x0 - 1, y1) ^ query(x1, y0 - 1) ^ query(x0 - 1, y0 - 1);
    }

    static long query(int x, int y) {
        long sum = 0;
        for (; x > 0; x -= (x & (-x))) {
            for (int j = y; j > 0; j -= (j & (-j))) {
                sum ^= bit[x][j];
            }
        }
        return sum;
    }
}
