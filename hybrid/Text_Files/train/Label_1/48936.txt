import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        Map<Integer, Integer> x = new HashMap<>();
        Map<Integer, Integer> y = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int xi = input.nextInt();
            int yi = input.nextInt();
            if (!x.containsKey(xi)) {
                x.put(xi, 0);
            }
            if (!y.containsKey(yi)) {
                y.put(yi, 0);
            }
            x.put(xi, x.get(xi) + 1);
            y.put(yi, y.get(yi) + 1);
        }
        long total = 0;
        for (Integer xi : x.keySet()) {
            total += (long) x.get(xi) * (x.get(xi) - 1) / 2;
        }
        for (Integer yi : y.keySet()) {
            total += (long) y.get(yi) * (y.get(yi) - 1) / 2;
        }
        System.out.println(total);
    }
}
