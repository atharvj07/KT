#include<bits/stdc++.h>
using namespace std;
int main(){
    int t,n,l,r,s;
    cin>>t;
    while(t--){
        cin>>n>>l>>r>>s;
        int sum=(r-l+1)*(l+r)/2;
        if(sum!=s){
            cout<<-1<<endl;
            continue;
        }
        vector<int>v;
        for(int i=1;i<=n;i++){
            if(l<=i && i<=r){
                v.push_back(0);
            }
            else{
                v.push_back(i);
            }
        }
        int cnt=l;
        int su=s;
        for(int i=l-1;i>=1;i--){
            if(su>v[i]){
                swap(v[i],v[cnt]);
                su-=v[cnt];
                cnt++;
            }
        }
        if(su){
            cout<<-1<<endl;
            continue;
        }
        for(auto x:v){
            cout<<x<<" ";
        }
        cout<<endl;
    }
    return 0;
}
