import java.util.HashMap;
import java.util.Scanner;

public class OgudarOlokWarehouse {

    private static final int MAX_SHELVES = 30;
    private static final int MAX_SECTIONS = 30;

    private static int[][] warehouse;
    private static HashMap<String, int[]> boxLocations;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        warehouse = new int[n][m];
        boxLocations = new HashMap<>();

        for (int i = 0; i < k; i++) {
            String operation = scanner.next();
            if (operation.equals("+1")) {
                int x = scanner.nextInt() - 1;
                int y = scanner.nextInt() - 1;
                String id = scanner.next();
                addBox(x, y, id);
            } else if (operation.equals("-1")) {
                String id = scanner.next();
                removeBox(id);
            }
        }

        scanner.close();
    }

    private static void addBox(int x, int y, String id) {
        if (warehouse[x][y] == 0) {
            warehouse[x][y] = 1;
            boxLocations.put(id, new int[]{x, y});
        } else {
            // Find the next empty section to the right
            for (int i = y + 1; i < warehouse[x].length; i++) {
                if (warehouse[x][i] == 0) {
                    warehouse[x][i] = 1;
                    boxLocations.put(id, new int[]{x, i});
                    return;
                }
            }

            // Find the first empty section on the next shelf
            for (int i = x + 1; i < warehouse.length; i++) {
                for (int j = 0; j < warehouse[i].length; j++) {
                    if (warehouse[i][j] == 0) {
                        warehouse[i][j] = 1;
                        boxLocations.put(id, new int[]{i, j});
                        return;
                    }
                }
            }

            // No empty section found, drink the box
            System.out.println("Drink " + id);
        }
    }

    private static void removeBox(String id) {
        int[] location = boxLocations.get(id);
        if (location != null) {
            warehouse[location[0]][location[1]] = 0;
            boxLocations.remove(id);
            System.out.println(location[0] + 1 + " " + (location[1] + 1));
        } else {
            System.out.println("-1 -1");
        }
    }
}
