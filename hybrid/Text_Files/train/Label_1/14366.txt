#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    bool found = false;
    for (long long x = n;; x++) {
      int cnt = 0;
      for (long long y = x;; y /= 10) {
        if (y == 0) break;
        cnt += (y % 10 != 0);
        if (cnt > k) break;
      }
      if (cnt <= k) {
        cout << x << endl;
        found = true;
        break;
      }
    }
  }
  return 0;
}
