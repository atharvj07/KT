import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] strengths = new int[n];
        for (int i = 0; i < n; i++) {
            strengths[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        int[] team1 = new int[n];
        int[] team2 = new int[n];
        int[] banned = new int[n];
        int picks = m / 2;
        int bans = m / 2;
        for (int i = 0; i < m; i++) {
            String action = scanner.next();
            int team = scanner.nextInt();
            if (action.equals("p")) {
                if (team == 1) {
                    for (int j = 0; j < n; j++) {
                        if (banned[j] == 0) {
                            team1[j]++;
                            if (team1[j] == picks) {
                                banned[j] = 1;
                                break;
                            }
                        }
                    }
                } else {
                    for (int j = 0; j < n; j++) {
                        if (banned[j] == 0) {
                            team2[j]++;
                            if (team2[j] == picks) {
                                banned[j] = 1;
                                break;
                            }
                        }
                    }
                }
            } else {
                for (int j = 0; j < n; j++) {
                    if (banned[j] == 0) {
                        banned[j] = 1;
                        break;
                    }
                }
            }
        }
        int team1Strength = 0;
        int team2Strength = 0;
        for (int i = 0; i < n; i++) {
            if (team1[i] > 0) {
                team1Strength += strengths[i];
            }
            if (team2[i] > 0) {
                team2Strength += strengths[i];
            }
        }
        System.out.println(team1Strength - team2Strength);
    }
}
