#include<bits/stdc++.h>
#define ld long double
using namespace std;

int main(){
    int t; cin >> t;
    for(int z = 0; z < t; z++){
        ld w, h, x1, y1, x2, y2, a, b;
        cin >> w >> h >> x1 >> y1 >> x2 >> y2 >> a >> b;
        if((w - (x2 - x1) < a) || (h - (y2 - y1) < b)) cout << -1 << endl;
        else {
            ld dx = min(max(0.0L, (w - a - (x2 - x1)) / 2), x1);
            ld dy = min(max(0.0L, (h - b - (y2 - y1)) / 2), y1);
            cout << fixed << setprecision(10);
            cout << sqrtl(dx * dx + dy * dy) << endl;
        }
    }
}
