import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = Long.parseLong(in.nextLine());
        int x = in.nextInt();
        if (n % 2 == 0) {
            x--;
        }
        System.out.println(x % 3);
    }
}
