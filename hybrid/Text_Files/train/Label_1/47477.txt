import java.util.*;

class Main {

    static List<Integer>[] graph;
    static boolean[] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        graph = new ArrayList[n];
        visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int s = sc.nextInt();
            int t = sc.nextInt();
            graph[s].add(t);
            graph[t].add(s);
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int s = sc.nextInt();
            int t = sc.nextInt();
            System.out.println(dfs(s, t) ? "yes" : "no");
        }
    }

    static boolean dfs(int s, int t) {
        if (s == t) {
            return true;
        }
        if (visited[s]) {
            return false;
        }
        visited[s] = true;
        for (int next : graph[s]) {
            if (dfs(next, t)) {
                return true;
            }
        }
        return false;
    }
}
