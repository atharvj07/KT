#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin>>n;
    string s; cin>>s;
    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if((abs(s[i]-s[j])==2&&s[i]%2!=s[j]%2)||(abs(s[i]-s[j])==1&&s[i]%2==s[j]%2)){
                cout<<"NO"<<endl;
                return 0;
            }
        }
    }
    cout<<"YES"<<endl;
}
