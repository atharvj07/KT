#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int v;
  cin >> v;
  vector<int> a(10);
  for (int i = 0; i < 10; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 9; i >= 1; i--) {
    if (v >= a[i]) {
      v -= a[i];
      ans *= 10;
      ans += i;
    }
  }
  if (v > 0) {
    cout << -1 << endl;
  } else {
    cout << ans << endl;
  }
  return 0;
}
