#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1000000007;
typedef long long ll;
#define pb push_back
#define endl '\n'
signed main() {
    ll n, c0, c1; cin >> n >> c0 >> c1;
    ll dp[n + 1][2];
    dp[0][0] = 0, dp[0][1] = 1;
    for (ll i = 1; i < n; i++) {
        dp[i][0] = dp[i - 1][0] + c0;
        dp[i][1] = min(dp[i - 1][0], dp[i - 1][1]) + c1;
    }
    cout << dp[n - 1][0] << endl;
}
