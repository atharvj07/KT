import java.util.Scanner;

public class Main {
    static final int MOD = 1000000007;
    static int[][][] dp = new int[51][51][2501];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int[] x = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextInt();
        }
        System.out.println(solve(n, a, x));
    }

    static int solve(int n, int a, int[] x) {
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= a; j++) {
                for (int k = 0; k < 2501; k++) {
                    dp[i][j][k] = -1;
                }
            }
        }
        return dfs(n, a, x, 0, 0, 0);
    }

    static int dfs(int n, int a, int[] x, int i, int sum, int cnt) {
        if (i == n + 1) {
            if (cnt == 0) {
                return 0;
            }
            if (sum * cnt == a * n) {
                return 1;
            }
            return 0;
        }
        if (dp[i][cnt][sum] != -1) {
            return dp[i][cnt][sum];
        }
        int ans = 0;
        ans = (ans + dfs(n, a, x, i + 1, sum + x[i], cnt + 1)) % MOD;
        ans = (ans + dfs(n, a, x, i + 1, sum, cnt)) % MOD;
        return dp[i][cnt][sum] = ans;
    }
}
