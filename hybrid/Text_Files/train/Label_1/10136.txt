#include <iostream>
using namespace std;

int a[101],cnt[101];

int main(){
    int n,k,ans=0;
    cin>>n>>k;
    for(int i=0;i<n;i++){
        cin>>a[i];
        while(a[i]){
            if(a[i]%10==4||a[i]%10==7) cnt[i]++;
            a[i] /= 10;
        }
        if(cnt[i]<=k) ans++;
    }
    cout<<ans<<endl;
    return 0;
}
