from sys import stdin

k = int(stdin.readline())
b = []
mn = 0
mx = 0
for i in range(k):
	n = []
	x = stdin.readline().split()
	n.append(int(x[0]))
	for j in range(1,int(x[0])+1):
		n.append(int(x[j]))
	mn += n[1]
	mx += n[-1]
	b.append(n)

if mn != mx:
	print("No")
else:
	print("Yes")
	for i in b:
		print(i[1],i[-1])
