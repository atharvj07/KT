import sys

input = sys.stdin.readline

n, x, y = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(n - 1)]
visited = [False] * (n + 1)
dist = [[0] * (n + 1) for _ in range(n + 1)]
for u, v in edges:
    dist[u][v] = x
    dist[v][u] = x
for i in range(1, n + 1):
    for j in range(1, n + 1):
        if i == j:
            continue
        if dist[i][j] == 0:
            dist[i][j] = y

ans = 1e18
for i in range(1, n + 1):
    if visited[i]:
        continue
    visited[i] = True
    q = [(0, i)]
    cur = 0
    while q:
        d, u = q.pop(0)
        cur = max(cur, d)
        for v in range(1, n + 1):
            if not visited[v] and dist[u][v] != 0:
                q.append((d + dist[u][v], v))
                visited[v] = True
    ans = min(ans, cur)

print(ans)
