#include <iostream>

int main() {
    long long a, b, f, k;
    std::cin >> a >> b >> f >> k;
    long long ans = (a + b - f - 1) / (b - f);
    long long jour_k = k - (a - f) / b;
    if (jour_k > ans && (a - f) % b != 0) ans++;
    if (jour_k > ans || b * ans < a) std::cout << -1;
    else std::cout << ans;
}
