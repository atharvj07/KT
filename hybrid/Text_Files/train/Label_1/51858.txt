import java.util.Scanner;

public class Main {

	  static long area(long x1, long y1, long x2, long y2, long x3, long y3) {
	    return (x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2));
	  }

	  public static void main(String[] args) {
	    Scanner sc = new Scanner(System.in);
	    long x0 = sc.nextInt();
	    long y0 = sc.nextInt();
	    long x1 = sc.nextInt();
	    long y1 = sc.nextInt();
	    int q = sc.nextInt();
	    sc.nextLine();

	    for (int i = 0; i < q; i++) {
	      long x2 = sc.nextInt();
	      long y2 = sc.nextInt();

	      long area = area(x0, y0, x1, y1, x2, y2);

	      if (area == 0) {
	        if (Math.min(x0, x1) <= x2 && x2 <= Math.max(x0, x1) &&
	          Math.min(y0, y1) <= y2 && y2 <= Math.max(y0, y1)) {
	          System.out.println("ON_SEGMENT");
	        } else {
	          System.out.println("ONLINE_BACK");
	        }
	      } else if (area > 0) {
	        System.out.println("COUNTER_CLOCKWISE");
	      } else {
	        System.out.println("CLOCKWISE");
	      }
	    }

	    sc.close();
	  }
	}
