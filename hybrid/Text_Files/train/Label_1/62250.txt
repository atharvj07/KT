n,k=map(int,input().split())
a=list(map(int,input().split()))
z=0
for i in range(n):
    if(i!=0):
        z=max(z,a[i-1])
    if(a[i]==1):
        z+=1
if(k==0):
    print(z)
    print(*a)
elif(k!=0):
    l=[]
    for i in range(n):
        if(a[i]!=1):
            if(k!=0):
                k-=1
                a[i]=1
                l.append(i+1)
            if(k==0):
                break
    z+=k
    print(z)
    print(*a)
