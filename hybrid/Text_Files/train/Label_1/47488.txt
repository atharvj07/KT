import java.util.Arrays;
import java.util.Scanner;

class Main {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] input1 = Arrays.stream(input.nextLine().split(" "))
                .mapToInt(Integer::parseInt).toArray();
        int[] x1 = new int[input1[0]];
        int[] x2 = new int[input1[1]];
        int[] y1 = new int[input1[0]];
        int[] y2 = new int[input1[1]];
        for (int i = 0; i < x1.length; i++) {
            int[] temp = Arrays.stream(input.nextLine().split(" "))
                    .mapToInt(Integer::parseInt).toArray();
            x1[i] = temp[0];
            y1[i] = temp[1];
        }
        for (int i = 0; i < x2.length; i++) {
            int[] temp = Arrays.stream(input.nextLine().split(" "))
                    .mapToInt(Integer::parseInt).toArray();
            x2[i] = temp[0];
            y2[i] = temp[1];
        }
        int result = 0;
        for (int a = 0; a < x2.length; a++) {
            boolean flag = false;
            for (int i = 0; i < x1.length; i++) {
                if (x2[a] <= Math.max(x1[i], x2[a])
                        && x2[a] >= Math.min(x1[i], x2[a])
                        && y2[a] <= Math.max(y1[i], y2[a])
                        && y2[a] >= Math.min(y1[i], y2[a])) {
                    if ((x1[i] == x2[a] && y2[a] > y1[i])
                            || (y1[i] == y2[a] && x2[a] > x1[i])
                            || (x1[i] == x2[a] && y2[a] < y1[i])
                            || (y1[i] == y2[a] && x2[a] < x1[i])) {
                        flag = true;
                        break;
                    }
                }
            }
            if (flag) {
                result++;
            }
        }
        System.out.print(result);
    }
}
