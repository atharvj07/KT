from collections import defaultdict

n, m = map(int, input().split())
c = list(map(int, input().split()))
g = defaultdict(lambda: set())
for _ in range(m):
    a, b = map(int, input().split())
    g[a-1].add(b-1)
    g[b-1].add(a-1)
deg = [0]*n
for x in g:
    for y in g[x]:
        if c[x] != c[y]:
            deg[c[x]-1] += 1

print(deg.index(max(deg))+1)
