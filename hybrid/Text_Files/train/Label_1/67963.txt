from functools import lru_cache

n = int(input())
a = input().split()
a = [int(x) for x in a]

@lru_cache(None)
def dp(l, r):
    if l == r:
        return 1 if a[l] == 1 else 0
    if r - l == 1:
        return 1 if a[l] != a[r] else 0
    ans = dp(l+1, r)
    if a[l] != a[l+1]:
        ans += dp(l+2, r)
    return ans

print(dp(0, n-1))
