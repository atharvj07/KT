import re

vowels = set('eyuioa')

n = int(input())
p = list(map(int, input().split()))
text = [input() for _ in range(n)]

for i in range(n):
    cur = 0
    for x in re.split(r'\s+', text[i]):
        if any(y in x for y in vowels):
            for s in re.split(r'[bcdfghjklmnpqrstvwxyz]', x):
                if s:
                    cur += 1
    if cur != p[i]:
        print('NO')
        exit(0)

print('YES')
