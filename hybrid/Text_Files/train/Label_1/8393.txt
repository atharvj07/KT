#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll n, A, B, ans, arr[100009][3];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin>>n;
    for(ll i=1;i<=n;i++) {
        cin>>arr[i][0]>>arr[i][1]>>arr[i][2];
    }
    for(ll P=1;P*P<=A;P++) {
        if(A%P==0) {
            B = A/P;
            for(ll i=1;i<=n;i++) {
                if(arr[i][0]==P && arr[i][1]==B) {
                    ans+=arr[i][2];
                }
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
