import sys

for line in sys.stdin.readlines()[1:]:
    a, b = map(int, line.split())
    if b % a:
        print(-1)
        continue
    ans = 0
    while a < b:
        if b % 2 and b % 4 and b % 8:
            print(-1)
            break
        if b % 8 == 0:
            b //= 8
            ans += 3
        elif b % 4 == 0:
            b //= 4
            ans += 2
        else:
            b //= 2
            ans += 1
    print(ans)
