#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    while (a[i] % 10 != 0 && k > 0) {
      a[i]++;
      k--;
    }
  }
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i] / 10;
  }
  cout << sum << endl;
  return 0;
}
