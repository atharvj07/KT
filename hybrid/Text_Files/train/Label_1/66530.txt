from itertools import permutations
from collections import Counter

for t in range(int(input())):
    n = int(input())
    p = [int(x) for x in input().split()]
    F = sorted([p[(i+1)%n] + p[i] for i in range(n)])
    for a in list(permutations(p)):
        if a != p and sorted([a[(i+1)%n] + a[i] for i in range(n)]) == F:
            print(*a)
            break
