#include <bits/stdc++.h>
using namespace std;

const int maxn = 1003, maxm = 1003;
int dp[maxn][maxm][2];
int a[maxn][maxm];

int main() {
    int n, m; cin >> n >> m;
    for (int i = 1; i <= n; i++) for (int j = 1; j <= m; j++) cin >> a[i][j];
    for (int j = m; j >= 0; j--) {
        for (int i = 1; i <= n; i++) {
            if (j == m) {
                dp[i][j][0] = a[i][j];
                dp[i][j][1] = a[i][j];
                continue;
            }
            dp[i][j][0] = max(dp[i][j + 1][0], dp[i][j + 1][1]) + a[i][j];
            dp[i][j][1] = max(dp[i + 1][j][0], dp[i + 1][j][1]) + a[i][j];
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            ans = max(ans, dp[i][j][0]);
            ans = max(ans, dp[i][j][1]);
        }
    }
    cout << ans << endl;
    return 0;
}
