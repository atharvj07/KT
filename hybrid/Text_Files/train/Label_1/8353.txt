#include<bits/stdc++.h>
const int mod = 1e9 + 7;
using namespace std;
int main() {
	int n, x, ans = 0;
	cin >> n >> x;
	vector<int>d(n);
	for (int i = 0; i < n; i++) {
		cin >> d[i];
		x -= d[i];
	}
	x += d[0]; ans++;
	if (x < 0) {}
	else {
		int m = x / d[0]; m++;
		ans += (1ll * n * m) % mod;
		x -= m * d[0];
		if (x > 0) ans += n;
		ans %= mod;
	}
	cout << ans << endl;
}
