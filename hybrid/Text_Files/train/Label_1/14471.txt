#include <iostream>
#include <vector>
#include <map>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  std::map<int, int> m;
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
    ++m[a[i]];
  }
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    while (m[cnt + 1]) ++cnt;
    if (a[i] == cnt + 1) continue;
    std::cout << cnt + 1 << ' ';
    --m[a[i]];
    ++m[cnt + 1];
  }
  std::cout << '\n';
  return 0;
}
