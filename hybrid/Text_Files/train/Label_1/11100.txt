#include <bits/stdc++.h>
using namespace std;
#define mp make_pair
typedef long long ll; typedef pair<int, int> pii; typedef vector<int> vi;
#define all(x) (x).begin(), (x).end()
const int MAXN = 5000, MAXM = 1000;
string a[MAXN];
int num[MAXN], c[MAXN], d[MAXN];
int ansmin[MAXM], ansmax[MAXM];
unordered_map<string, int> id;
unordered_map<int, string> rid;
int n, m, q, p, tot;
bool vis[MAXN];
inline int r(int x) { return x ? x : num[p]; }
inline int g(int x) { return id.find(a[x]) == id.end() ? num[x] : id[a[x]]; }
inline void upd(int x, int y) {
  while (x < m) {
    d[x] += y;
    x += x & -x;
  }
}
inline int get(int x) {
  int sum = 0;
  while (x > 0) {
    sum += d[x];
    x -= x & -x;
  }
  return sum % 2;
}
inline void dfs(int x) {
  vis[x] = true;
  if (x > 0 && c[x] != -1) {
    ansmax[tot] |= num[x] >> p & 1;
    ansmin[tot] |= num[x] >> p & 1;
  }
  if (c[x] == -2 || c[x] == 0) {
    ansmax[tot] |= get(g(x));
    if (c[x] == -2) ansmin[tot] |= get(g(x));
  } else if (c[x] == 1) {
    ansmax[tot] |= get(g(x)) | get(g(x + 1));
    ansmin[tot] |= get(g(x)) & get(g(x + 1));
  } else if (c[x] == 2) {
    ansmax[tot] |= get(g(x)) ^ get(g(x + 1));
    ansmin[tot] |= get(g(x)) ^ get(g(x + 1));
  }
  if (x + x < m) upd(x + x, num[x]);
  if (x > 0) upd(x, -num[x]);
}
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  cin >> n >> m;
  memset(c, -1, sizeof(c));
  for (int i = 0; i < n; i++) {
    string u, v;
    char op;
    cin >> u >> op >> v;
    if (id.find(u) == id.end()) id[u] = tot++, rid[tot - 1] = u;
    if (op == '=') {
      num[id[u]] = 0;
      for (int j = 0; j < m; j++) {
        char t;
        cin >> t;
        num[id[u]] <<= 1;
        num[id[u]] |= (t - '0');
      }
    } else {
      int x = id[u], y = id[v];
      cin >> op;
      if (op == '&')
        c[y] = 1;
      else if (op == '|')
        c[y] = 2;
      else
        c[y] = 0;
      c[x] = -2;
    }
  }
  for (int i = 0; i < n; i++) {
    p = i;
    if (!vis[i] && c[i] != -1) dfs(i);
  }
  for (int i = m - 1; i >= 0; i--) {
    if (ansmin[i] & 1)
      cout << '1';
    else
      cout << '0';
  }
  cout << "\n";
  for (int i = m - 1; i >= 0; i--) {
    if (ansmax[i] & 1)
      cout << '1';
    else
      cout << '0';
  }
  cout << "\n";
  return 0;
}
