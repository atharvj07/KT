import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        int x = sc.nextInt();
        int[] ans = new int[q + 1];
        int mex = 0;
        for (int i = 1; i <= q; i++) {
            int y = sc.nextInt();
            int can = (y >= mex && (y - mex) % x == 0) ? mex + x : mex;
            ans[i] = Math.max(ans[i - 1], can);
            mex = ans[i];
        }
        for (int i = 1; i <= q; i++) {
            System.out.println(ans[i]);
        }
    }
}
