#include<bits/stdc++.h>
using namespace std;
struct node
{
	int to, nxt;
}e[8000001];
int head[400001], dfn[400001], low[400001], Time = 0, bel[400001], cnt = 0, ans[400001], in[400001], out[400001];
void add(int x, int y)
{
	e[++cnt].to = y;
	e[cnt].nxt = head[x];
	head[x] = cnt;
}
bool dfs(int x, int fa)
{
	dfn[x] = low[x] = ++Time;
	in[x] = out[x] = 1;
	out[fa] = 0;
	bool flag = 0;
	for (int i = head[x]; i; i = e[i].nxt)
	{
		int j = e[i].to;
		if (!dfn[j])
		{
			flag |= dfs(j, x);
			low[x] = min(low[x], low[j]);
			if (low[j] > dfn[x])
				ans[j] = 1;
			if (in[j] - out[j] + 1 > 1)
			{
				flag = 1;
				ans[x] = 0;
			}
			in[x] += in[j];
			out[x] += out[j];
		}
		else
		{
			if (j == fa)
				continue;
			low[x] = min(low[x], dfn[j]);
		}
	}
	return flag;
}
int main()
{
	int T;
	scanf("%d", &T);
	while(T--)
	{
		memset(head, 0, sizeof(head));
		memset(dfn, 0, sizeof(dfn));
		memset(low, 0, sizeof(low));
		memset(ans, 0, sizeof(ans));
		memset(in, 0, sizeof(in));
		memset(out, 0, sizeof(out));
		cnt = Time = 0;
		int n, m;
		scanf("%d%d", &n, &m);
		for (; m; m--)
		{
			int x, y;
			scanf("%d%d", &x, &y);
			add(x, y);
		}
		for (int i = 1; i <= n; i++)
			if (!dfn[i])
			{
				ans[i] = dfs(i, i);
				if (ans[i] == 0)
					ans[i] = -1;
			}
		for (int i = 1; i <= n; i++)
		{
			if (ans[i] == -1)
				printf("-1 ");
			else
				printf("%d ", ans[i]);
		}
		printf("\n");
	}
}
