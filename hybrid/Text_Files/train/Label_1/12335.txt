#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  vector<pair<int, int>> segments(m);
  for (int i = 0; i < m; i++) {
    cin >> segments[i].first >> segments[i].second;
  }
  sort(segments.begin(), segments.end());
  vector<bool> used(m);
  int ans = 0, cnt = 0;
  vector<int> ans_segments;
  for (int i = 1; i <= n; i++) {
    int l = i, r = i;
    for (int j = 0; j < m; j++) {
      if (used[j]) continue;
      if (segments[j].first <= l && r <= segments[j].second) {
        l = segments[j].first;
        r = segments[j].second;
        used[j] = true;
        cnt++;
        ans_segments.push_back(j + 1);
      }
    }
    ans = max(ans, a[r] - a[l] + 1);
  }
  cout << ans << endl;
  cout << cnt << endl;
  for (int i = 0; i < cnt; i++) {
    cout << ans_segments[i] << " ";
  }
  cout << endl;
  return 0;
}
