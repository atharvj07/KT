import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {

    static class Edge {
        int to, inv;

        public Edge(int to, int inv) {
            this.to = to;
            this.inv = inv;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int[] ntk = Arrays.stream(br.readLine().split(" "))
                .mapToInt(Integer::parseInt)
                .toArray();

        List<List<Edge>> graph = new ArrayList<>();
        List<List<Edge>> graphInv = new ArrayList<>();

        for (int i = 0; i <= ntk[0]; i++) {
            graph.add(new ArrayList<>());
            graphInv.add(new ArrayList<>());
        }

        for (int i = 0; i < ntk[1]; i++) {
            int[] ab = Arrays.stream(br.readLine().split(" "))
                    .mapToInt(Integer::parseInt)
                    .toArray();
            graph.get(ab[0]).add(new Edge(ab[1], graph.get(ab[1]).size()));
            graphInv.get(ab[1]).add(new Edge(ab[0], graphInv.get(ab[0]).size()));
        }

        for (int i = 0; i < ntk[2]; i++) {
            int[] cd = Arrays.stream(br.readLine().split(" "))
                    .mapToInt(Integer::parseInt)
                    .toArray();
            graph.get(cd[0]).add(new Edge(cd[1], -1));
            graphInv.get(cd[1]).add(new Edge(cd[0], -1));
        }

        boolean[] visited = new boolean[ntk[0] + 1];
        int[] dist = new int[ntk[0] + 1];
        Arrays.fill(dist, -1);

        for (int i = 1; i <= ntk[0]; i++) {
            if (visited[i]) {
                continue;
            }
            List<Integer> q = new ArrayList<>();
            q.add(i);
            visited[i] = true;
            dist[i] = 0;

            while (!q.isEmpty()) {
                int now = q.remove(0);
                for (Edge edge : graph.get(now)) {
                    if (edge.to == -1) {
                        continue;
                    }
                    if (!visited[edge.to]) {
                        q.add(edge.to);
                        visited[edge.to] = true;
                        dist[edge.to] = dist[now] + 1;
                    }
                }
            }
        }

        Arrays.fill(visited, false);
        int[] ans = new int[ntk[0] + 1];

        for (int i = 1; i <= ntk[0]; i++) {
            if (visited[i]) {
                continue;
            }
            List<Integer> q = new ArrayList<>();
            q.add(i);
            visited[i] = true;
            ans[i] = 0;

            while (!q.isEmpty()) {
                int now = q.remove(0);
                for (Edge edge : graphInv.get(now)) {
                    if (dist[edge.to] == -1) {
                        ans[edge.to]++;
                    }
                    if (!visited[edge.to]) {
                        q.add(edge.to);
                        visited[edge.to] = true;
                    }
                }
            }
        }

        for (int i = 1; i <= ntk[0]; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
