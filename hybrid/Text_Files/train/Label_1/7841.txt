#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[2*n];
        int f[2] = {0};
        for(int i=0;i<2*n;i++){
            cin>>a[i];
            f[a[i]-1]++;
        }
        cout<<abs(f[0]-f[1])<<endl;
    }
}
