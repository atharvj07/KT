#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  long long n, t;
  cin >> n >> t;
  vector<long long> a(n);
  for (long long i = 0; i < n; i++) { cin >> a[i]; }
  long long ans = 0;
  while (t >= *min_element(a.begin(), a.end()) && n) {
    for (long long i = 0; i < n; i++) {
      if (t >= a[i]) {
        ans++;
        t -= a[i];
      } else {
        a.erase(a.begin() + i);
        n--;
        i--;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
