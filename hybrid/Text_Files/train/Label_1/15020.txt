#include<bits/stdc++.h>
using namespace std;
int main(){
  int n, a[105]={0}, b[105]={0}, c[6]={0};
  cin>>n;
  for(int i=1;i<=n;i++){
    cin>>a[i];
    c[a[i]]++;
  }
  for(int i=1;i<=n;i++){
    cin>>b[i];
    c[b[i]]--;
  }
  int ans=0;
  for(int i=1;i<=5;i++){
    if(c[i]>0)
      ans+=c[i]/2;
  }
  if(ans==0)
    cout<<-1<<endl;
  else
    cout<<ans<<endl;
}
