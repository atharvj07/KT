#include <bits/stdc++.h>
using namespace std;
#define fastio()                        \
    ios_base::sync_with_stdio(false); \
    cin.tie(nullptr);                  \
    cout.tie(nullptr)
int main() {
    fastio();
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1), t(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) { cin >> t[i]; }
    vector<int> dp(n + 1, 0);
    for (int i = 1; i < k; i++) { dp[i] = a[i] * t[i]; }
    for (int i = k; i <= n; i++) { dp[i] = max(dp[i - 1] + a[i] * t[i], a[i] * t[i]); }
    int ans = 0;
    for (int i = k; i <= n; i++) { ans = max(ans, dp[i]); }
    cout << ans << endl;
    return 0;
}
