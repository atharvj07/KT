#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100 + 5;
void solve() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> g(n, vector<int>(m));
  vector<vector<int>> vis(n, vector<int>(m, 0));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> g[i][j];
      vis[i][j] = 0;
    }
  }
  vector<tuple<int, int, int>> moves;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (!vis[i][j]) {
        int sum = g[i][j];
        queue<tuple<int, int, int>> q;
        q.push({i, j, g[i][j]});
        while (!q.empty()) {
          auto [x, y, s] = q.front();
          q.pop();
          if (vis[x][y]) continue;
          vis[x][y] = 1;
          sum += g[x][y];
          if (s > 1) {
            moves.push_back({x + 1, 0, -1});
            moves.push_back({x + 1, 0, s - 1});
          }
          if (x + 1 < n) q.push({x + 1, y, s});
          if (y + 1 < m) q.push({x, y + 1, s});
        }
        if (sum == 0) {
          cout << "-1" << endl;
          return;
        } else if (sum > 1) {
          moves.push_back({i + 1, 1, sum - 1});
        }
      }
    }
  }
  cout << moves.size() << endl;
  for (auto [x, y, s] : moves) {
    if (y == 0)
      cout << "row " << x << endl;
    else
      cout << "col " << x << endl;
  }
}
int main() { solve(); }
