import java.math.BigInteger;
import java.util.Scanner;

class Main {
  static long solve(String x) {
    BigInteger left = BigInteger.valueOf(0);
    BigInteger right = BigInteger.valueOf(0);
    // x & 1 == 1 -> x[i] == '1'
    // x & 2 == 2 -> x[i+1] == '1'
    for (int i = 0; i < x.length(); i++) {
      if ((x.charAt(i) - '0') % 2 == 1) {
        left = left.add(BigInteger.valueOf(2).pow(i));
        right = right.add(BigInteger.valueOf(2).pow(x.length() - i - 1));
      }
    }
    // 11
    // 1 1
    return left.multiply(right).mod(BigInteger.valueOf(1000000007)).longValue();
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String x = in.nextLine();
    System.out.println(solve(x));
  }
}
