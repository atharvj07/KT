n, k = map(int, input().split())
a = list(map(int, input().split()))

changes_needed = 0
for i in range(k):
    count_1 = 0
    count_2 = 0
    for j in range(i, n, k):
        if a[j] == 1:
            count_1 += 1
        else:
            count_2 += 1
    changes_needed += min(count_1, count_2)

print(changes_needed)
