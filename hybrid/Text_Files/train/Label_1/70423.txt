import sys

s = sys.stdin.readline().strip()
stack = []
ans = []
for i in range(len(s)):
    if s[i] == '(': stack.append(i)
    else:
        if stack:
            ans.append(i - stack.pop())
print(*ans)
for _ in range(int(sys.stdin.readline())):
    l, r = map(int, sys.stdin.readline().split())
    r = min(r, len(ans))
    for i in range(len(ans)):
        if ans[i] + l > r:
            print(r-l)
            break
    else:
        print(r-l+1)
