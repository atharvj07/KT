#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0, odd = 0;
  for (int x : a) {
    bool is_sq = false;
    for (long long j = 1; j * j <= x; j++) {
      if (j * j == x) {
        is_sq = true;
        break;
      }
    }
    ans += is_sq != (odd & 1);
    odd ^= is_sq;
  }
  cout << ans << endl;
  return 0;
}
