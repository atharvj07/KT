#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, a[200005], pos[1000005];
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i], pos[a[i]] = i;
  int it = 0;
  for (int i = 1; i <= n; i++) {
    if (pos[i] > it) {
      it = pos[i];
      cout << i << ' ';
    }
  }
  return 0;
}
