import java.util.*;

public class Main {
    static final long p = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), a[] = new int[10];
        for (int i = 0; i < 10; i++) a[i] = in.nextInt();
        long[][] dp = new long[n + 1][1 << 10];
        dp[0][0] = 1;
        for (int i = 0; i < n; i++)
            for (int mask = 0; mask < (1 << 10); mask++)
                for (int d = 0; d < 10; d++)
                    if ((mask & (1 << d)) == 0 && a[d] > 0)
                        dp[i + 1][mask | (1 << d)] = (dp[i + 1][mask | (1 << d)] + dp[i][mask]) % p;
        System.out.println(dp[n][(1 << 10) - 1]);
    }
}
