#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
  int hp, atk, def, thp, tatk, tdef, th, ta, td;
  cin >> hp >> atk >> def >> thp >> tatk >> tdef >> th >> ta >> td;
  thp -= th * max(0, tatk - def);
  tdef += td * max(0, atk - tdef);
  cout << ta * max(0, tatk - tdef) + td * max(0, atk - tdef) << endl;
  return 0;
}
