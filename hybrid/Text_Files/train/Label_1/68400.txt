n = int(input().strip())
a = list(map(int, input().strip().split()))

if sum(a) == 1:
    print(-1)
else:
    first = a.index(1)
    last = n - 1 - a[::-1].index(1)
    moves = 0
    for i in range(first, last + 1):
        if a[i] == 0:
            moves += min(i - first, last - i)
    print(moves)
