#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) { cin >> a[i]; }
        int ans = 0, j = 0;
        for (int i = 1; i < n; ++i) {
            if (a[i] < a[j]) {
                ++ans;
                j = i;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
