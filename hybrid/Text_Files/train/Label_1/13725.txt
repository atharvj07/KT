#include <iostream>
#include <string>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        string ans = s;
        int k = 1;
        for (int i = 2; i <= n; i++) {
            string t = s;
            for (int j = 0; j < n - i + 1; j++) {
                reverse(t.begin() + j, t.begin() + j + i);
            }
            if (t < ans) {
                ans = t;
                k = i;
            }
        }
        cout << ans << endl << k << endl;
    }
}
