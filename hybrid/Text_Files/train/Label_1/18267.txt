#include <bits/stdc++.h>

using namespace std;

#define ll long long

int main()
{
    ll t;
    cin >> t;
    while(t--)
    {
        ll a, b;
        cin >> a >> b;
        if(a == 1)
        {
            cout << (b == 0 ? 3 : 2) << "\n";
        }
        else
        {
            cout << 3 << "\n";
        }
    }
    return 0;
}
