#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb emplace_back
const ll N=2e5+10;
ll n,m,s,ans;
vector<ll>adj[N];
unordered_map<ll,ll>ds;
ll get(ll u,ll p)
{
    if(adj[u].size()==1&&u!=1)
    return 0;
    ll sum=0;
    for(ll v:adj[u])
    {
        if(v==p)
        continue;
        sum+=get(v,u)+1;
    }
    if(adj[u].size()>2)
    return sum;
    return min(sum,1LL);
}
void dfs1(ll u,ll p,ll d)
{
    ds[d]=u;
    for(ll v:adj[u])
    {
        if(v!=p)
        dfs1(v,u,d+1);
    }
}
void dfs2(ll u,ll p,ll d)
{
    if(adj[u].size()==1&&u!=1)
    cout<<p<<" "<<u<<" "<<u<<"\n";
    else
    {
        for(ll v:adj[u])
        {
            if(v!=p)
            {
                cout<<u<<" "<<v<<" "<<u<<"\n";
                dfs2(v,u,d+1);
            }
        }
    }
}
int main()
{
    ll u,v;
    cin>>n;
    for(ll i=0;i<n-1;i++)
    {
        cin>>u>>v;
        adj[u].pb(v);
        adj[v].pb(u);
    }
    ans=get(1,-1);
    dfs1(1,-1,0);
    dfs2(1,-1,0);
    cout<<ans<<"\n";
    return 0;
}
