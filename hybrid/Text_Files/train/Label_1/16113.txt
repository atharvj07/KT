#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
const int INF = 1e9;
int n, m, k, s, a[N], b[N];
pair<int, long long> c[N];

bool cmp(const pair<int, long long>& l, const pair<int, long long>& r) { return l.first < r.first; }

int main() {
    cin >> n >> m >> k >> s;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < m; i++) {
        int t, cos;
        cin >> t >> cos;
        c[i] = {t, cos};
    }
    sort(c, c + m, cmp);
    vector<pair<long long, int>> need;
    for (int i = 0; i < m; i++) {
        if (c[i].first == 1)
            need.push_back({c[i].second, i});
        else
            need.push_back({c[i].second * 1LL * b[0], i});
    }
    sort(need.begin(), need.end());
    vector<int> ans;
    long long sum = 0;
    for (auto p : need) {
        if (sum + p.first <= s) {
            sum += p.first;
            ans.push_back(p.second);
        }
        if (ans.size() == k) break;
    }
    if (ans.size() < k) {
        cout << -1 << endl;
    } else {
        cout << ans.size() << endl;
        for (int x : ans) {
            if (c[x].first == 1)
                cout << x + 1 << " " << 1 << endl;
            else
                cout << x + 1 << " " << 2 << endl;
        }
    }
}
