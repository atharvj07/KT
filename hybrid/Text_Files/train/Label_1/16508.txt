#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,m;
set<int> x[1000001];
int32_t main()
{
    cin>>n>>m;
    while(m--)
    {
        int a,b,c,d;
        cin>>a>>b>>c>>d;
        for(int i=a;i<=c;i++)
            x[i].insert(b),x[i].insert(d+1);
    }
    int ans=0;
    for(int i=1;i<=n;i++)
    {
        set<int> :: iterator it=x[i].begin();
        if(it!=x[i].end())
        {
            int prev=*it;
            it++;
            ans+=*it-prev-1;
            for(;it!=x[i].end();it++)
                ans+=*it-prev-1,prev=*it;
        }
    }
    cout<<ans<<endl;
    return 0;
}
