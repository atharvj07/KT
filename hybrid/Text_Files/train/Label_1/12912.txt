#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  long long a, b, c, n;
  cin >> a >> b >> c;
  cin >> n;
  vector<long long> x(n);
  for (long long i = 0; i < n; ++i) cin >> x[i];
  sort(x.begin(), x.end());
  long long ans = 0;
  long long l = b + 1, r = c - 1, p = 0;
  while (p < n) {
    if (l <= x[p] && x[p] <= r) {
      ++p;
      continue;
    }
    if (abs(a - l) < abs(a - r)) {
      if (l <= x[p]) {
        ans += x[p];
        ++p;
      }
      ++l;
    } else {
      if (r >= x[p]) {
        ans += x[p];
        ++p;
      }
      --r;
    }
  }
  cout << ans << endl;
  return 0;
}
