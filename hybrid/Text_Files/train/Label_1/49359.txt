import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    static int n, m;
    static List<List<Integer>> graph = new ArrayList<>();
    static char[] edgeColors;
    static boolean[] visited;
    static List<Integer> result = new ArrayList<>();
    static boolean impossible = false;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        n = Integer.parseInt(tokens[0]);
        m = Integer.parseInt(tokens[1]);
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        edgeColors = new char[m + 1];
        visited = new boolean[n + 1];
        for (int i = 1; i <= m; i++) {
            tokens = br.readLine().split(" ");
            int u = Integer.parseInt(tokens[0]);
            int v = Integer.parseInt(tokens[1]);
            char color = tokens[2].charAt(0);
            graph.get(u).add(i);
            graph.get(v).add(i);
            edgeColors[i] = color;
        }
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                dfs(i, -1);
            }
        }
        if (impossible) {
            System.out.println(-1);
        } else {
            System.out.println(result.size());
            for (int vertex : result) {
                System.out.print(vertex + " ");
            }
        }
    }

    static void dfs(int vertex, int parentEdge) {
        visited[vertex] = true;
        int[] edgeCounts = new int[2];
        List<Integer> incidentEdges = graph.get(vertex).stream().filter(e -> e != parentEdge).collect(Collectors.toList());
        for (int edge : incidentEdges) {
            edgeCounts[edgeColors[edge] == 'R' ? 0 : 1]++;
        }
        if (edgeCounts[0] != edgeCounts[1]) {
            impossible = true;
            return;
        }
        result.add(vertex);
        for (int edge : incidentEdges) {
            if (visited[graph.get(vertex).get(0) ^ graph.get(vertex).get(1) ^ edge]) {
                continue;
            }
            dfs(graph.get(vertex).get(0) ^ graph.get(vertex).get(1) ^ edge, edge);
        }
    }
}
