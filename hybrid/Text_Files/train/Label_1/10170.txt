#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int cnt = count(a.begin(), a.end(), 3);
    for (int i = 0; i < n; i++) {
      if (a[i] % 3 == 1) cnt++;
      if (a[i] % 3 == 2) {
        cnt++;
        if (cnt % 3 == 0) cnt++;
      }
    }
    cout << cnt << '\n';
  }
}
