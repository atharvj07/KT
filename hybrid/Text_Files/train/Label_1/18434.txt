#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m, ta, tb, k;
  cin >> n >> m >> ta >> tb >> k;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int j = 0; j < m; j++) cin >> b[j];
  sort(b.begin(), b.end());
  int j = 0, ans = 1e9;
  for (int i = 0; i < n; i++) {
    while (j < m && b[j] < a[i] + ta) j++;
    if (j == m) break;
    ans = min(ans, b[j] + tb);
    k--;
    if (k == 0) break;
  }
  if (k > 0) ans = -1;
  cout << ans << '\n';
  return 0;
}
