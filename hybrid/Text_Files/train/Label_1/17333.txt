#include <cstdio>
#include <vector>
#include <algorithm>
#include <map>

using namespace std;

int main() {
  int n, m;
  scanf("%d %d", &n, &m);

  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }

  map<int, int> freq;
  for (int i = 0; i < n; i++) {
    freq[a[i]]++;
  }

  int changes = 0;
  int max_min = 0;
  int min_freq = n;
  for (auto it = freq.begin(); it != freq.end(); it++) {
    if (it->first <= m) {
      min_freq = min(min_freq, it->second);
      max_min = max(max_min, it->second);
    }
  }

  vector<int> b(n);
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] <= m) {
      b[i] = a[i];
      j++;
    } else {
      b[i] = freq[a[i]] == 1 ? m + 1 : a[i];
      changes++;
    }
  }

  while (j < n) {
    for (int i = 1; i <= m; i++) {
      if (freq[i] > max_min) {
        b[j++] = i;
      }
    }
  }

  printf("%d %d\n", max_min, changes);
  for (int i = 0; i < n; i++) {
    printf("%d ", b[i]);
  }
  printf("\n");

  return 0;
}
