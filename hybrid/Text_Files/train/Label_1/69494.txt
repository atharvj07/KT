n = int(input())
a = list(map(int, input().split()))
pref = [0] * n
pref[0] = a[0]
for i in range(1, n):
    pref[i] = pref[i - 1] + a[i]
ans = 0
l = 0
for r in range(n):
    if a[r] == 0:
        if r - l > 0 and pref[r - 1] - (pref[l - 1] if l > 0 else 0) == r - l:
            ans += 1
        l = r + 1
print(ans)
