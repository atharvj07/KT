from collections import defaultdict

n = int(input())
degrees = [tuple(map(int, input().split())) for _ in range(n)]
degrees.sort(key=lambda x: (x[0], x[1]))
degrees.reverse()

adj = defaultdict(list)
edges = []
for degree, s in degrees:
    if degree == 0:
        continue
    elif degree < 3:
        adj[s]
    else:
        for u in adj[s]:
            if degree == 3:
                edges.append((u, n))
                edges.append((n, u))
            adj[u].remove(n)
            adj[n].append(u)
    adj[n].append(s)

print(len(edges))
for edge in edges:
    print(*edge)
