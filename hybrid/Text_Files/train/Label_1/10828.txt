#include <iostream>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  int ans = 0;
  bool ok = true;
  for (int i = 0; i < n; i++) {
    if (s[i] == '_') {
      ok = false;
      if (i == n - 1) ans += 9;
      else ans *= 10;
    } else if (s[i] == 'X') {
      ok = false;
      if (i == n - 1) ans += 9;
    } else if (s[i] == '0' && i != n - 1) {
      ok = false;
    }
  }
  if (ok && s[n - 1] == '0') ok = false;
  if (ok && (s[n - 1] - '0') % 5 != 0) ok = false;
  if (ok) ans++;
  cout << ans << endl;
}
