import java.util.Scanner;

public class RyoukosMemoryNote {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        long minPages = solve(n, m, a);
        System.out.println(minPages);
    }

    private static long solve(int n, int m, int[] a) {
        long[] prefixSum = new long[n + 1];
        for (int i = 0; i < m; i++) {
            prefixSum[a[i]]++;
        }
        for (int i = 1; i <= n; i++) {
            prefixSum[i] += prefixSum[i - 1];
        }

        long minPages = Long.MAX_VALUE;
        for (int mergeIndex = 1; mergeIndex <= n; mergeIndex++) {
            long pagesTurnedBeforeMerge = prefixSum[mergeIndex - 1] * (mergeIndex - 1);
            long pagesTurnedAfterMerge = (n - mergeIndex) * (prefixSum[n] - prefixSum[mergeIndex - 1]);
            minPages = Math.min(minPages, pagesTurnedBeforeMerge + pagesTurnedAfterMerge);
        }

        return minPages;
    }
}
