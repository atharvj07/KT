#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    long long ans = 0;
    for (int i = 0, j = n - 1; i < n && k1; i++, k1--) {
        ans += (long long)(a[i] - b[j]) * (a[i] - b[j]);
        j--;
    }
    for (int i = 0, j = 0; i < n && k2; i++, k2--) {
        ans += (long long)(a[i] - b[j]) * (a[i] - b[j]);
        j++;
    }
    cout << ans << endl;
    return 0;
}
