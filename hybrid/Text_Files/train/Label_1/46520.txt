import java.util.*;

class Main {
    static class Road {
        int from, to, type;

        Road(int from, int to, int type) {
            this.from = from;
            this.to = to;
            this.type = type;
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        List<Road> roads = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            roads.add(new Road(input.nextInt(), input.nextInt(), input.next().charAt(0)));
        }
        Collections.sort(roads, new Comparator<Road>() {
            @Override
            public int compare(Road r1, Road r2) {
                return r1.type - r2.type;
            }
        });
        int ans = 0;
        int[] path = new int[n + 1];
        Arrays.fill(path, -1);
        for (Road road : roads) {
            if (path[road.from] == -1 && path[road.to] == -1) {
                ans++;
                path[road.from] = road.to;
                path[road.to] = road.from;
            }
        }
        for (int i = 1; i <= n; i++) {
            if (path[i] == -1) {
                ans = -1;
                break;
            }
        }
        if (ans > 0) {
            System.out.println(ans);
            for (int i = 0; i < m; i++) {
                Road road = roads.get(i);
                if (path[road.from] == road.to) {
                    System.out.print((i + 1) + " ");
                }
            }
            System.out.println();
        } else {
            System.out.println(-1);
        }
    }
}
