#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;

int main() {
    ll n, x, ans = 0;
    vector<int> a(100002);
    cin >> n;
    for (int i = 0; i < n - 1; ++i) { cin >> a[i]; }
    for (int i = 0; i < n - 1; ++i) {
        ans += a[i] - i - 1;
    }
    cout << ans << endl;
    return 0;
}
