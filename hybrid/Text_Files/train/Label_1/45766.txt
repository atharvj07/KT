import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();

        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; ++i) {
            a[i] = scanner.nextLong();
            b[i] = scanner.nextLong();
        }

        long answer = 0;
        long current = 0;
        for (int i = 0; i < n; ++i) {
            if (current >= a[i]) {
                current -= a[i];
            } else {
                answer += (a[i] - current) * 2;
                current = b[i] - (a[i] - current);
            }
        }

        System.out.println(answer);
    }
}
