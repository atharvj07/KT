#include <iostream>

using namespace std;

int min_commands(int x, int y) {
  if (x == y) {
    return 0;
  }
  int dx = abs(x - y);
  int dy = abs(x + y);
  if (dx == dy) {
    return dx + 2;
  }
  if (dx > dy) {
    return dx * 2 + 1;
  }
  return dy * 2 + 1;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int x, y;
    cin >> x >> y;
    cout << min_commands(x, y) << endl;
  }
  return 0;
}
