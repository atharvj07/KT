#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  vector<int> t(k);
  int i = 0, j = n - 1, sum = 0;
  for (int l = k - 1; l >= 0; l--) {
    while (i < j && sum + a[j] >= a[i] + sum + a[i + 1]) {
      sum += a[j];
      j--;
    }
    t[l] = j - i + 1;
    sum += a[i];
    i++;
  }
  cout << sum << endl;
  for (int x : t) {
    cout << x << " ";
  }
  cout << endl;
  return 0;
}
