#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  int ans = 0;
  deque<int> a;
  for (int i = 1; i <= n; i++) {
    int u;
    cin >> u;
    while (!a.empty() && a.back() > u) {
      a.pop_back();
    }
    a.push_back(u);
    ans += a.size() - 1;
  }
  cout << ans << endl;
}
