#include <vector>
#include <iostream>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> pos;
  int a;
  while (std::cin >> a) { pos.push_back(a); }
  
  sort(pos.begin(), pos.end());
  int left = 0;
  for (int x: pos) { if (x > n / 2) left++; }
  
  int ans = 0;
  int now = 0;
  if (2 * left > n) now = left - (n - left);
  ans += n / 2 - now;
  int res = n - now;
  for (int x: pos) {
    if (x <= n / 2 && res > 0) {
      if (x != now + 1) ans++;
      res--;
    }
    else if (res == 0) {
      break;
    }
    now++;
  }
  
  std::cout << ans << "\n";
}
