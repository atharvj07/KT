#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    long long n, h, a, b, k, ta, fa, tb, fb;
    cin >> n >> h >> a >> b >> k;
    while (k--) {
        cin >> ta >> fa >> tb >> fb;
        long long ans = abs(fa - fb);
        if (ta == tb)
            cout << ans << "\n";
        else
            cout << ans + min(fa - 1, h - fa) + min(fb - 1, h - fb) + 2 << "\n";
    }

    return 0;
}
