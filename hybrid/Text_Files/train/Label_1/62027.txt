t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    map_mines = input().strip()
    cost = 0
    i = 0
    while i < len(map_mines):
        # Skip the safe buildings
        while i < len(map_mines) and map_mines[i] == '0':
            i += 1
        if i < len(map_mines) and map_mines[i] == '1':
            cost += a  # Activate the first mine in a segment
            i += 1
            # Go through the rest of the segment
            while i < len(map_mines) and map_mines[i] == '1':
                i += 1
            # Check if it's possible and cheaper to place mines to connect segments
            gap = 0
            while i < len(map_mines) and map_mines[i] == '0':
                gap += 1
                i += 1
            if i < len(map_mines) and map_mines[i] == '1':
                cost += min(a, gap * b)
    print(cost)
