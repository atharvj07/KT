import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    int t = in.nextInt();
    while (t-- > 0) {
      int n = in.nextInt();
      int m = in.nextInt();
      int a = in.nextInt() - 1;
      int b = in.nextInt() - 1;

      Map<Integer, Integer> distA = new HashMap<>();
      Map<Integer, Integer> distB = new HashMap<>();

      distA.put(a, 0);
      distB.put(b, 0);

      boolean[] vis = new boolean[n];

      for (int i = 0; i < m; i++) {
        int u = in.nextInt() - 1;
        int v = in.nextInt() - 1;
        if (!vis[u] && !vis[v]) {
          vis[u] = true;
          vis[v] = true;
          distA.put(v, distA.getOrDefault(u, 0) + 1);
          distB.put(u, distB.getOrDefault(v, 0) + 1);
        }
      }

      int ans = 0;
      for (int i = 0; i < n; i++) {
        if (i == a || i == b) continue;
        if (distA.containsKey(i) && distB.containsKey(i)) {
          ans++;
        }
      }

      System.out.println(ans);
    }
  }
}
