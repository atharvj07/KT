#include <bits/stdc++.h>
using namespace std;
const int maxn=1e5+5;
int n,p[maxn];
string f[maxn],s[maxn];
int main(){
	cin>>n;
	for(int i=1;i<=n;++i) cin>>f[i]>>s[i];
	for(int i=1;i<=n;++i) cin>>p[i];
	int sgn=1;
	for(int i=1;i<n;++i) if(p[i]<p[i+1]) sgn=-1;
	bool t=0;
	for(int i=1;i<=n;++i){
		if(sgn==1){
			if(f[p[i]]>f[p[i+1]]) t=1;
		} else {
			if(s[p[i]]>s[p[i+1]]) t=1;
		}
	}
	if(t) cout<<"NO\n";
	else cout<<"YES\n";
}
