#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, int>> v(n);
        for (int i = 0; i < n; i++) {
            cin >> v[i].first;
        }
        for (int i = 0; i < n; i++) {
            cin >> v[i].second;
        }
        sort(v.begin(), v.end(),
             [](const pair<int, int>& a, const pair<int, int>& b) {
                 return a.second < b.second;
             });
        int time = 0;
        for (auto el : v) {
            time += el.second;
            if (time > el.first) {
                time = el.first;
            }
        }
        cout << time << endl;
    }
    return 0;
}
