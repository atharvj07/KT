#include <bits/stdc++.h>
using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int n = s.size();
  vector<int> pos(26, -1);
  for (int i = 0; i < n; i++) {
    if (s[i] == t[i]) continue;
    if (pos[s[i] - 'a'] == -1) {
      pos[s[i] - 'a'] = i;
    } else {
      cout << 1 << endl;
      cout << s[i] << " " << t[i] << endl;
      return 0;
    }
  }
  int cnt = 0;
  for (int i = 0; i < 26; i++) {
    if (pos[i] != -1) cnt++;
  }
  if (cnt % 2 == 0) {
    cout << cnt / 2 << endl;
    for (int i = 0; i < 26; i++) {
      if (pos[i] != -1) {
        cout << char(i + 'a') << " ";
        cout << char(t[pos[i]] - 'a') << endl;
      }
    }
  } else {
    cout << -1 << endl;
  }
  return 0;
}
