import sys
input = sys.stdin.readline

def main():
    n = int(input())
    arr = [list(map(int, input().split())) for _ in range(n)]
    arr.sort(key=lambda x: x[1])
    color = [0] * n
    for i in range(n):
        x1, y1, x2, y2 = arr[i]
        if y1 > 0:
            continue
        j = i
        while j < n and arr[j][1] == 0:
            color[j] = 1
            for k in range(i, j):
                if color[k] == color[j] and (arr[k][2] == arr[j][2] or arr[k][3] == arr[j][3]):
                    print("NO")
                    return
            j += 1
        for k in range(i, j):
            if color[k] == 0:
                color[k] = 2
        i = j - 1
    print("YES")
    for c in color: print(c)

main()
