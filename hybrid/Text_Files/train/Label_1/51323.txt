import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int d = in.nextInt();
        int s = in.nextInt();
        int ans = -1;
        int mn = (int) Math.ceil((double) s / 9) + 1;
        int mx = (int) Math.floor((double) s / d) + 1;
        if (mn <= mx) {
            int num = d * mn;
            int sum = num;
            while (sum > s) {
                num -= d;
                sum = num;
            }
            while (sum < s) {
                num += d;
                sum = num;
            }
            ans = num;
        }
        System.out.println(ans);
    }
}
