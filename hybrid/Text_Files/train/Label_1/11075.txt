#include <cstdio>
#include <vector>

int h, w, q, r1, c1, r2, c2, ans, tmp;
char grid[502][502];
bool ok[502][502];
int sum[502][502];
int dx[] = {0, 1};
int dy[] = {1, 0};

int main() {
    scanf("%d%d\n", &h, &w);
    for (int i = 1; i <= h; i++) {
        scanf("%s", grid[i] + 1);
    }
    for (int i = 1; i <= h; i++)
        for (int j = 1; j <= w; j++) ok[i][j] = (grid[i][j] == '.');
    for (int i = 1; i <= h; i++) {
        for (int j = 1; j <= w; j++) {
            if (ok[i][j]) {
                tmp = 0;
                for (int dir = 0; dir < 2; dir++) {
                    int ni = i + dx[dir];
                    int nj = j + dy[dir];
                    if (1 <= ni && ni <= h && 1 <= nj && nj <= w && ok[ni][nj])
                        ++tmp;
                }
                sum[i][j] = tmp;
            }
        }
    }
    scanf("%d", &q);
    while (q--) {
        scanf("%d%d%d%d\n", &r1, &c1, &r2, &c2);
        ans = 0;
        for (int i = r1; i <= r2; i++)
            for (int j = c1; j <= c2; j++) ans += sum[i][j];
        printf("%d\n", ans);
    }
    return 0;
}
