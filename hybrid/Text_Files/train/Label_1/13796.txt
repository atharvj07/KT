#include <iostream>
#include <vector>
#include <map>
using namespace std;
int main() {
    int n, A;
    cin >> n >> A;
    vector<int> c(n);
    for (int i = 0; i < n; i++) cin >> c[i];
    map<int, int> cnt;
    map<int, bool> can;
    for (auto &x : c) {
        cnt[x]++;
        if (x != A) can[x] = true;
    }
    cnt[A]++;
    for (auto &x : c) {
        cnt[x]--;
        if (cnt[x] < cnt[A]) can[x] = false;
        if (cnt[A] > cnt[x] && can[x]) { cout << x << endl; return 0; }
    }
    cout << -1 << endl;
    return 0;
}
