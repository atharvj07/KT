#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int l = 0, r = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(')
      ++l;
    else
      --l;
    if (l == -1) {
      ++r;
      l = 0;
    }
    if (l + r == k / 2) break;
  }
  for (int i = 0; i < n; i++) {
    if (r) {
      if (s[i] == '(')
        --r;
      else
        ++r;
    }
    if (l) {
      if (s[i] == ')')
        --l;
      else
        ++l;
    }
    if (l == -1) {
      ++r;
      l = 0;
      cout << ')';
    }
    if (l == 0 && r == 0) {
      cout << s[i];
    }
    if (l + r == k / 2) break;
  }
  cout << endl;
}
