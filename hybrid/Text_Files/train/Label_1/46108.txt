import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            board[i] = sc.nextLine().toCharArray();
        }
        sc.close();
        boolean[][] visited = new boolean[n][n];
        System.out.println(dfs(board, visited, 0, 0) ? "YES" : "NO");
    }

    private static boolean dfs(char[][] board, boolean[][] visited, int x, int y) {
        if (x < 0 || x >= board.length || y < 0 || y >= board[0].length || visited[x][y] || board[x][y] == '#') {
            return false;
        }
        visited[x][y] = true;
        boolean result = true;
        result &= dfs(board, visited, x - 1, y);
        result &= dfs(board, visited, x + 1, y);
        result &= dfs(board, visited, x, y - 1);
        result &= dfs(board, visited, x, y + 1);
        if (result) {
            board[x][y] = '#';
        } else {
            visited[x][y] = false;
        }
        return result;
    }
}
