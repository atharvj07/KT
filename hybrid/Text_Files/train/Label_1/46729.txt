import java.util.*;

public class VitalyaAndAlarmClocks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] alarms = new int[n];
        for (int i = 0; i < n; i++) {
            alarms[i] = scanner.nextInt();
        }

        Arrays.sort(alarms);

        int minAlarmsToTurnOff = 0;
        for (int i = 0; i <= n - m; i++) {
            int count = 0;
            for (int j = i; j < i + m && j < n; j++) {
                if (alarms[j] - alarms[i] < m) {
                    count++;
                }
            }
            minAlarmsToTurnOff = Math.max(minAlarmsToTurnOff, count - k);
        }

        System.out.println(minAlarmsToTurnOff);
    }
}
