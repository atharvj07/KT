#include <vector>
#include <map>
#include <queue>
#include <iostream>

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;

ll dis(pi a, pi b) {
    return abs(a.first - b.first) + abs(a.second - b.second);
}

int solve() {
    ll n, x, y;
    cin >> n;
    vector<pi> points;
    while (n--) {
        cin >> x >> y;
        points.push_back({x, y});
    }
    map<ll, vector<pi> > level;
    for (auto &p : points) {
        level[max(p.first, p.second)].push_back(p);
    }
    ll ans = 0, curx = 0, cury = 0;
    for (auto &p : level) {
        vector<pi> &cur = p.second;
        pi prv = {0, 0};
        for (auto &q : cur) {
            ans += dis(q, prv);
            prv = q;
        }
        curx = prv.first, cury = prv.second;
    }

    return ans;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);
    int c = solve();
    cout << c << "\n";
}
