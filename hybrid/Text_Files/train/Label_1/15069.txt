#include <bits/stdc++.h>
using namespace std;
long long mod = 1e9 + 7, inf = 1e18;
#define pb push_back
#define ll long long
struct edge {
    long long to, p1, p2;
};
vector<edge> graph[100005];
long long ans = inf, way, vis[100005];
void dfs(long long u, long long par, long long sum, long long cnt) {
    vis[u] = 1;
    if (sum > ans) return;
    if (sum == ans) way = (way + cnt) % mod;
    if (sum < ans) {
        ans = sum;
        way = cnt;
    }
    for (auto i : graph[u]) {
        if (vis[i.to]) continue;
        if (i.to != par)
            dfs(i.to, u, sum + i.p1 + i.p2, cnt * 2);
        else
            dfs(i.to, u, sum + (i.p1 + i.p2) / 2, cnt * 2);
    }
    vis[u] = 0;
}
int main() {
    long long n, m, u, v, x, y;
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> x, graph[i].pb({i, x, 0});
    cin >> m;
    while (m--) {
        cin >> u >> v;
        graph[u].pb({v, 0, 0});
        graph[v].pb({u, 0, 0});
    }
    dfs(1, 1, 0, 1);
    cout << ans << " " << way << "\n";
    return 0;
}
