import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int a = in.nextInt();
            int b = in.nextInt();
            int c = in.nextInt();
            int d = in.nextInt();
            int k = in.nextInt();
            int x = (a + c - 1) / c;
            int y = (b + d - 1) / d;
            if (x + y <= k) {
                System.out.println(x + " " + y);
            } else {
                System.out.println(-1);
            }
        }
    }
}
