#include <iostream>
#include <vector>

using namespace std;

typedef int64_t i64;

int main() {
    i64 n, k;
    cin >> n >> k;

    vector<i64> c(n);
    for (i64& ci : c) { cin >> ci; }

    vector<i64> id(k);
    for (i64& idi : id) { cin >> idi; }

    i64 sum = 0;
    for (i64 i = 0; i < n; ++i) {
        for (i64 j = 0; j < n; ++j) {
            if (i == j) { continue; }

            bool is_capital = false;
            for (i64 idi : id) {
                if (idi == i + 1 || idi == j + 1) {
                    is_capital = true;
                    break;
                }
            }

            if (is_capital) {
                sum += c[i] * c[j];
            }
        }
    }

    cout << sum << endl;

    return 0;
}
