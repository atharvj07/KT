import bisect

for _ in range(int(input())):
    n, k, m = map(int, input().split())
    b = list(map(int, input().split()))
    if k == 3:
        print("YES" if len(b) == n - 1 and 2 * b[0] + k == 2 * b[-1] else "NO")
    else:
        a = [1] * (n + 1)
        for i in range(m):
            a[b[i]] = k // 2
        a[b[0] - 1] = k - a[b[0]]
        a[b[-1] + 1] = k - a[b[-1]]
        a[1] = 0
        l, r = b[0] - 1, b[-1] + 1
        while l >= 1 or r <= n:
            if l >= 1:
                if a[l] != k // 2:
                    l -= 1
                    continue
                if b[0] - l + 1 != a[l] or b[-1] - l + 1 != k - a[l]:
                    break
            if r <= n:
                if a[r] != k // 2:
                    r += 1
                    continue
                if r - b[0] != a[r] or r - b[-1] != k - a[r]:
                    break
            l -= 1
            r += 1
        print("YES" if l == 0 and r == n + 1 else "NO")
