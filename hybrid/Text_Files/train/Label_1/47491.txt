import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int n = in.nextInt();
        Set<String> stones = new HashSet<>();
        for (int i = 0; i < k; i++) {
            int ax = in.nextInt();
            int ay = in.nextInt();
            stones.add(ax + "," + ay);
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            int mx = in.nextInt();
            int my = in.nextInt();
            for (String stone : stones) {
                int ax = Integer.parseInt(stone.split(",")[0]);
                int ay = Integer.parseInt(stone.split(",")[1]);
                if ((ay - my) * (ax - mx) > 0) {
                    count++;
                    break;
                }
            }
        }
        System.out.println(count);
    }
}
