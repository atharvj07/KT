#include <iostream>
#include <vector>
using namespace std;
const int maxn = 3e4 + 5;
long long a[maxn];
bool exist[maxn], ans;
vector<int> v;

void calc(int sum, int l, int r) {
  if (exist[sum + maxn]) {
    ans = 1;
    return;
  }
  if (l < 0 || r >= maxn) return;
  exist[sum + maxn] = 1;
  calc(sum + a[l], l - 1, r);
  calc(sum - a[r], l, r + 1);
}
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, sum = 0;
    cin >> n;
    v.clear();
    for (int i = 0; i <maxn; i++) exist[i] = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
      v.push_back(sum);
    }
    ans = 0;
    for (int i = 0; i < n; i++) {
      calc(0, i - 1, i);
      if (ans) break;
    }
    cout << (ans ? "YES" : "NO") << endl;
  }
  return 0;
}
