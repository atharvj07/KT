import sys
for _ in range(int(input())):
    _=input()
    A=list(map(int,input().split()))
    A.sort()
    B=set(A)
    C=[]
    for n in B:
        C+=[n,A.count(n)]
    C.sort()
    if len(C)==1:print(-1)
    else:
        n=len(A)
        if n%len(C)!=0:print(-1)
        else:
            x=C[len(C)-1][0]-C[len(C)-2][0]
            y=C[len(C)-1][1]
            z=n//y
            if x*z>C[len(C)-2][0]:print(-1)
            else:print(x)
