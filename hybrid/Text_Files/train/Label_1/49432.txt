import java.io.*;
import java.util.*;

public class Main {
    static int[] parent;
    static int[][] vectors;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt();
        int d = in.nextInt();
        parent = new int[m + 1];
        vectors = new int[m + 1][d];
        for (int i = 1; i <= m; i++) {
            parent[i] = i;
        }
        for (int i = 1; i <= m; i++) {
            int k = in.nextInt();
            for (int j = 0; j < k; j++) {
                for (int l = 0; l < d; l++) {
                    vectors[i][l] ^= in.nextInt();
                }
            }
            link(i, find(i));
        }
        for (int i = 1; i <= m; i++) {
            parent[i] = parent[parent[i]];
        }
        int[] groups = new int[m + 1];
        int count = 0;
        for (int i = 1; i <= m; i++) {
            if (parent[i] == i) {
                groups[i] = ++count;
            }
        }
        for (int i = 1; i <= m; i++) {
            System.out.print(groups[parent[i]] + " ");
        }
        System.out.println();
    }

    static int find(int x) {
        if (parent[x] == x) {
            return x;
        }
        return find(parent[x]);
    }

    static void link(int x, int y) {
        parent[x] = y;
    }
}
