n, t = map(int, input().split())
s = input()
pos = [0] * 26
for i in range(26):
    pos[ord(chr(ord('a') + i)) - ord('a')] = i
def f(l, r):
    if r - l == 1:
        return 1 << pos[ord(s[l]) - ord('a')]
    m = l + 1
    return -f(l, m) + f(m, r)
if f(0, n) == t:
    print("Yes")
else:
    print("No")
