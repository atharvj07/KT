s = input()
g = input()
k = int(input())
c = [0]*len(s)
ans = 0
for i in range(1,len(s)):
    c[i] = c[i-1] + 1 if g[ord(s[i])-ord('a')] == '0' else c[i-1] - 1
    if c[i] <= k: ans += 1
    else:
        while i > 0 and c[i] > k:
            c[i] -= 1 if g[ord(s[i])-ord('a')] == '0' else c[i] + 1
            i -= 1
        if c[i] <= k: ans += 1
print(ans)
