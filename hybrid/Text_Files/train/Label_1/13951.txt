#include <bits/stdc++.h>
#define N 400010
#define ll long long
using namespace std;
ll n, k, w[N], f[N][205], pw[N];
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> w[i];
  pw[0] = 1;
  for (int i = 1; i <= n; i++) pw[i] = (pw[i - 1] * i) % 1000000007;
  f[n][1] = pw[n];
  for (int i = n - 1; i >= 1; i--) {
    for (int j = 1; j <= k; j++) {
      f[i][j] = (f[i + 1][j] + f[i + 1][j - 1] * w[i] % 1000000007) % 1000000007;
    }
  }
  cout << f[1][k];
}
