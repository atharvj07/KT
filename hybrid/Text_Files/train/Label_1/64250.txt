n = int(input())
a = list(map(int, input().split()))

ones = a.count(1)
twos = a.count(2)
threes = a.count(3)
fours = a.count(4)

swaps = 0

# Pairing ones with threes
min_ones_threes = min(ones, threes)
ones -= min_ones_threes
threes -= min_ones_threes

# Pairing twos with twos
swaps += twos // 2
twos %= 2

# If there are still twos, try to make a group of four with ones
if twos:
    if ones >= 2:
        twos -= 1
        ones -= 2
        swaps += 1
    elif ones == 1:
        twos -= 1
        ones -= 1
        swaps += 1

if ones > 0:
    swaps += (2*ones + 3) // 4

if twos > 0:
    swaps = -1

print(swaps)
