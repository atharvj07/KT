#include <iostream>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  long long a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int t = m, l, r;
  for (int i = m - 1; i >= 0; i--) {
    if (a[i] <= k) {
      l = i;
      break;
    }
  }
  for (int i = m + 1; i < n; i++) {
    if (a[i] <= k) {
      r = i;
      break;
    }
  }
  if (l == 0) {
    cout << abs(r - m) * 10;
  } else if (r == n - 1) {
    cout << abs(l - m) * 10;
  } else
    cout << min(abs(l - m), abs(r - m)) * 10;
  return 0;
}
