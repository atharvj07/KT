import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int s = sc.nextInt();
        int r = sc.nextInt();
        System.out.println(power(2, p - 1, BigInteger.valueOf(998244353)).multiply(BigInteger.valueOf(s - r)).mod(BigInteger.valueOf(998244353)));
    }
    public static BigInteger power(long x, long y, BigInteger m) {
        if (y == 0) return BigInteger.valueOf(1);
        if (y == 1) return BigInteger.valueOf(x);
        BigInteger val = power(x, y / 2, m);
        val = val.multiply(val).mod(m);
        if (y % 2 == 1) val = val.multiply(BigInteger.valueOf(x)).mod(m);
        return val;
    }
}
