#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;

    std::vector<std::vector<int>> adj(n);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        std::cin >> a >> b;
        adj[a - 1].push_back(b - 1);
        adj[b - 1].push_back(a - 1);
    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
        ans += adj[i].size() - 1;
    }

    std::cout << ans / 2 << '\n';

    return 0;
}
