#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
uniform_int_distribution<int> dis[210];
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

int adj[500010];
vector<pair<int, int>> e;

bool dead;
int b[210];
int bc;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, t, k;
  cin >> n >> t >> k;
  int a[t + 1];
  int sum = 0;
  for (int i = 1; i <= t; i++) {
    cin >> a[i];
    sum += a[i];
  }
  if (sum != n - 1) {
    cout << "-1" << endl;
    return 0;
  }
  if (k > n - 1 - (t - 1)) {
    cout << "-1" << endl;
    return 0;
  }
  for (int i = 1; i <= t; i++) {
    if (i != 1) dis[i] = uniform_int_distribution<int>(1, n - 1 - a[i]);
    for (int j = 0; j < a[i]; j++) {
      int x;
      if (i == 1)
        x = j + 1;
      else
        x = dis[i](rng) + 1;
      adj[x] = adj[x] * 10 + i;
    }
  }
  for (int i = 2; i <= n - 1; i++) {
    int aa = adj[i];
    vector<int> v;
    while (aa > 0) {
      v.push_back(aa % 10);
      aa = aa / 10;
    }
    sort(v.begin(), v.end());
    if (v.size() == 1) {
      b[v[0]]++;
      if (b[v[0]] == 2) bc++;
    } else if (v.size() > 1) b[v[1]]++;
  }
  bc = min(bc, k);
  if (bc != k) {
    cout << "-1" << endl;
    return 0;
  }
  cout << n << endl;
  for (int i = 1; i < n; i++) {
    if (adj[i] > 0) {
      int aa = adj[i];
      vector<int> v;
      while (aa > 0) {
        v.push_back(aa % 10);
        aa = aa / 10;
      }
      sort(v.begin(), v.end());
      int l = v.size() - 1;
      if (b[v[l]] == 1) {
        if (dead == true) {
          l--;
        }
      }
      if (v[l] != 1)
        e.push_back(make_pair(i, 1));
      else
        dead = true, e.push_back(make_pair(i, v[l + 1]));
    }
  }
  for (auto x : e) cout << x.first << ' ' << x.second << endl;
  return 0;
}
