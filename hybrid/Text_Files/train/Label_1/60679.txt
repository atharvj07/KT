n=int(input())
a=[[]for i in range(n)]
for i in range(n):
    a[i]=list(map(int,input().split()))
    a[i].sort()

    a[i][1]=a[i][2]
    del a[i][2]
for i in range(n):
    for j in range(i+1,n):
        if a[i]==a[j]:

            print(str(i+1)+" "+str(j+1))
