import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 1; i <= t; i++) {
            long n = sc.nextLong();
            long k = sc.nextLong();
            long b = (k - 1) % (2 * n);
            long ans;
            if (b < n) {
                ans = b + 1;
            } else {
                ans = b - n + 1;
            }
            System.out.println(ans);
        }
        sc.close();
    }
}
