#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    if (a[n - 1] <= k) {
      cout << 1 << "\n";
      continue;
    }
    int ans = 0;
    int cur = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (a[i] <= cur) {
        cout << -1 << "\n";
        goto nxt;
      }
      if (cur + k >= a[i]) {
        cur += a[i];
        ans++;
      } else {
        cout << -1 << "\n";
        goto nxt;
      }
    }
  nxt:;
  }
  return 0;
}
