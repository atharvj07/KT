def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        ans = 0
        for i in range(2, int(n ** 0.5) + 1):
            if n % i == 0:
                a = i
                b = n // i
                carry = (a % 10 + b % 10) // 10
                if a >= 10 and ((a // 10) % 10 + carry) >= 10:
                    continue
                if b >= 10 and ((b // 10) % 10 + carry) >= 10:
                    continue
                ans += 1
        print(ans)

solve()
