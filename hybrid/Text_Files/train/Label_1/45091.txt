import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        Set<Integer>[] edges = new Set[n + 1];
        for (int i = 1; i <= n; i++) {
            edges[i] = new HashSet<>();
        }
        Map<Integer, Integer> colorCount = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            int c = in.nextInt();
            edges[a].add(b);
            edges[b].add(a);
            colorCount.put(c, colorCount.getOrDefault(c, 0) + 1);
        }

        int components = 0;
        boolean[] visited = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                dfs(i, edges, visited);
                components++;
            }
        }

        int maxColors = 0;
        for (int color : colorCount.keySet()) {
            int count = colorCount.get(color);
            maxColors += Math.min(count, components);
        }

        System.out.println(maxColors);
    }

    private static void dfs(int node, Set<Integer>[] edges, boolean[] visited) {
        if (visited[node]) {
            return;
        }

        visited[node] = true;

        for (int neighbor : edges[node]) {
            dfs(neighbor, edges, visited);
        }
    }
}
