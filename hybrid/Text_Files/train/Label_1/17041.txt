#include <iostream>
int main() {
  std::ios::sync_with_stdio(false);
  std::cin.tie(0);
  std::string s;
  std::cin >> s;
  bool letters[26] = {false};
  for (char c : s) {
    int pos = c - 'a';
    if (letters[pos] == false) { letters[pos] = true; }
    else {
      std::cout << "NO" << std::endl;
      return 0;
    }
  }
  std::cout << "YES" << std::endl;
  return 0;
}
