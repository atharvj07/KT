import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int k = Integer.parseInt(input[1]);
        char[] left = br.readLine().toCharArray();
        char[] right = br.readLine().toCharArray();

        boolean[][] canReach = new boolean[2][n + 1];
        canReach[0][1] = true;
        for (int i = 2; i <= n; i++) {
            if (left[i - 1] != 'X') {
                canReach[0][i] = canReach[0][i - 1] || canReach[1][i - 1];
            }
            if (right[i - 1] != 'X') {
                canReach[1][i] = canReach[1][i - 1] || canReach[0][i - 1];
            }
        }

        System.out.println(canReach[0][n] || canReach[1][n] ? "YES" : "NO");
    }
}
