import sys
input = sys.stdin.readline

n, m, k, w = map(int,input().split())
grid = [[0]*m for _ in range(n)]
for i in range(n):
    string = input().strip()
    for j in range(m):
        grid[i][j] = string[j]

def get_diff(a, b):
    diff = 0
    for i in range(n):
        for j in range(m):
            if a[i][j] != b[i][j]:
                diff += 1
    return diff

dp = [[1000000000]*(k+1) for _ in range(k+1)]
dp[0][0] = 0
for i in range(k):
    for j in range(i):
        dp[i+1][j+1] = min(dp[i+1][j+1], dp[i][j] + get_diff(grid[i], grid[j])*w)
    dp[i+1][0] = min(dp[i+1][0], dp[i][0] + n*m)

print(dp[k][0])
for i in range(k-1,-1,-1):
    if dp[i+1][0] < dp[i][i+1]*w + n*m:
        print(i+1, 0)
    else:
        for j in range(i+1):
            if dp[i+1][j+1] == dp[i][j] + get_diff(grid[i], grid[j])*w:
                print(i+1, j+1)
                break
