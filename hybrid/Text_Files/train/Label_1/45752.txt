import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][][] cube = new int[n][n][n];
        if (n % 2 == 1) {
            System.out.println("-1");
            return;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (i % 2 == 0 && j % 2 == 0 && k % 2 == 0) {
                        cube[i][j][k] = 1;
                    } else if (i % 2 == 1 && j % 2 == 1 && k % 2 == 1) {
                        cube[i][j][k] = 1;
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    System.out.print(cube[i][j][k] == 1 ? "w" : "b");
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
