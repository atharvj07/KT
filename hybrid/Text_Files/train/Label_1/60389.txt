t = int(input())
for _ in range(t):
    x1, y1, z1 = map(int, input().split())
    x2, y2, z2 = map(int, input().split())

    sum = 0

    # Minimize negative impact by matching a's 2s with b's 1s
    t = min(z1, y2)
    sum += t * 2
    z1 -= t
    y2 -= t

    # Match a's 2s with b's 2s, neutral impact so no addition to sum

    # Minimize negative impact by using a's remaining 2s or 1s against b's 0s
    t = min(z1 + y1, x2)
    x2 -= t

    # Negative impact by matching a's 1s with b's 2s
    t = min(y1, z2)
    sum -= t * 2

    print(sum)
