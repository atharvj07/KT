#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
    int n;
    cin >> n;
    vector<int> s(n);
    int mx = 0, g1 = 0, cnt1 = 0;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
        if (g1 == 0) g1 = s[i];
        else g1 = gcd(g1, s[i]);
    }
    for (int i = 0; i < n; i++) {
        if (gcd(g1, s[i]) == 1) {
            cnt1++;
            continue;
        }
        mx = max(mx, cnt1);
        cnt1 = 1;
        g1 = s[i];
    }
    mx = max(mx, cnt1);
    cout << mx;
}
