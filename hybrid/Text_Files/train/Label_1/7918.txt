#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
      if (s[i - 1] == j - 1 + 'a') { dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + 1); }
    }
  }
  cout << (1LL << (n - dp[n][m])) << endl;
  return 0;
}
