#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
int n, a[MAXN];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a, a + n);
        if (a[0] != a[n - 1])
            cout << -1 << endl;
        else if ((n - 1) % 2 == 0)
            cout << n / 2 << endl;
        else
            cout << (n - 1) / 2 << endl;
    }
    return 0;
}
