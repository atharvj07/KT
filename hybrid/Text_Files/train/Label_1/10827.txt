#include<bits/stdc++.h>
using namespace std;
int main(){int a,x,l,b[10],c;
    string s;
    cin>>s;
    l=s.length();
    if(s.find('_')!=string::npos)a=9;
    else a=1;
    if(s.find('X')!=string::npos)x=9;
    else x=1;
    memset(b,0,sizeof(b));
    if(l>1){
    for(int i=0;i<l;i++){
    if(s[i]=='0'){
        if(i==0)c=0;
        else b[s[i-1]-'0']++;
    }
    else if(s[i]=='5'){
        if(s[i-1]=='2')b[s[i]-'0']++;
        else{
            c=0;
            break;
        }
    }
    else if(s[i]=='2'){
        if(s[i-1]=='1')b[s[i]-'0']++;
        else{
            c=0;
            break;
        }
    }
}}
    else{
        if(s[l-1]=='0'||s[l-1]=='5'||s[l-1]=='2')c=1;
        else c=0;
    }
    cout<<a*x*c<<endl;
    return 0;
}
