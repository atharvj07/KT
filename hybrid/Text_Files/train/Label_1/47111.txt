import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        List<Student> students = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int ri = sc.nextInt();
            int wi = sc.nextInt();
            students.add(new Student(ri - c, wi - d));
        }
        Collections.sort(students, (s1, s2) -> s1.wi * s2.ri - s1.ri * s2.wi);
        int res = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (students.get(i).wi * students.get(j).ri > students.get(i).ri * students.get(j).wi) {
                    for (int k = j + 1; k < n; k++) {
                        if (students.get(j).wi * students.get(k).ri > students.get(j).ri * students.get(k).wi &&
                                students.get(k).wi * students.get(i).ri > students.get(k).ri * students.get(i).wi) {
                            res++;
                        }
                    }
                }
            }
        }
        System.out.println(res);
    }

    static class Student {
        int ri;
        int wi;

        Student(int ri, int wi) {
            this.ri = ri;
            this.wi = wi;
        }
    }
}
