import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] p = new int[m];
        for (int i = 0; i < m; i++) {
            p[i] = sc.nextInt();
        }
        Arrays.sort(a);
        for (int i = 0; i < m; i++) {
            int l = 0;
            int r = n - 1;
            int ans = 0;
            while (l < r) {
                int area = a[l] * a[r];
                if (area >= p[i]) {
                    ans += r - l;
                    r--;
                } else {
                    l++;
                }
            }
            System.out.println(ans);
        }
    }
}
