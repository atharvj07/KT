#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> meat(n);
  for (int i = 0; i < n; i++) {
    cin >> meat[i].first >> meat[i].second;
  }
  sort(meat.begin(), meat.end(), [](const pair<int, int>& a, const pair<int, int>& b) { return a.second < b.second; });
  int ans = 0, sum = 0;
  for (const auto& m : meat) {
    int need = max(0, m.first - sum);
    ans += need * m.second;
    sum += m.first;
  }
  cout << ans << endl;
  return 0;
}
