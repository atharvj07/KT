#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> p(n);
        for (int i = 0; i < n; i++) p[i] = i + 1;
        k--;
        while (k > 0) {
            int i = n - 1;
            while (i > 0 && p[i - 1] >= p[i]) i--;
            if (i == 0) {
                k = -1;
                break;
            }
            int j = n - 1;
            while (p[j] <= p[i - 1]) j--;
            swap(p[i - 1], p[j]);
            reverse(p.begin() + i, p.end());
            k--;
        }
        if (k == 0) {
            for (int i = 0; i < n; i++) cout << p[i] << " ";
            cout << endl;
        } else cout << -1 << endl;
    }
    return 0;
}
