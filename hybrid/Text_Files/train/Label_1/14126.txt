#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    int w, h, n;
    cin >> w >> h >> n;

    while (w > 1 && h > 1) {
      if (w % 2 == 0 && w >= n) {
        w /= 2;
      } else if (h % 2 == 0 && h >= n) {
        h /= 2;
      } else {
        break;
      }
    }

    cout << (w == 1 && h == 1 ? "YES" : "NO") << endl;
  }

  return 0;
}
