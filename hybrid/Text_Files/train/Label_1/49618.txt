import java.util.Scanner;

public class Main {
    static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        long[] fact = new long[2 * n + 1];
        fact[0] = 1;
        for (int i = 1; i <= 2 * n; i++) {
            fact[i] = fact[i - 1] * i % MOD;
        }
        long[] invfact = new long[2 * n + 1];
        invfact[2 * n] = powmod(fact[2 * n], MOD - 2);
        for (int i = 2 * n - 1; i >= 0; i--) {
            invfact[i] = invfact[i + 1] * (i + 1) % MOD;
        }
        int[] cnt = new int[100001];
        for (int h : a) cnt[h]++;
        long ans = 0;
        for (int i = 0; i < 100001; i++) {
            if (cnt[i] < 2) continue;
            long tmp = fact[cnt[i]];
            tmp = tmp * invfact[cnt[i] / 2] % MOD;
            tmp = tmp * invfact[cnt[i] - cnt[i] / 2] % MOD;
            ans = (ans + tmp) % MOD;
        }
        System.out.println(ans);
    }

    static long powmod(long base, long e) {
        long result = 1;
        base = base % MOD;
        while (e > 0) {
            if ((e & 1) > 0) result = (result * base) % MOD;
            base = (base * base) % MOD;
            e >>= 1;
        }
        return result;
    }
}
