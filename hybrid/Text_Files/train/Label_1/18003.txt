#include <cstdio>
#include <vector>

const int MAXN=5001;
int n, a[MAXN], b[MAXN], ans;
std::vector<int>adj[MAXN];
bool mark[MAXN];

bool dfs(int u){
  for(int i=0; i<adj[u].size(); i++){
    int v=adj[u][i];
    if(mark[v])continue;
    mark[v]=true;
    if(dfs(v)){
      ans+=2;
      return true;
    }
  }
  return false;
}

int main(){
  scanf("%d", &n);
  for(int i=1; i<=n; i++){
    char c; scanf(" %c", &c);
    scanf("%d %d", &a[i], &b[i]);
    if(c=='M'){
      for(int j=1; j<=n; j++){
        if(c=='F' && a[i]<=b[j] && a[j]<=b[i]){
          adj[i].push_back(j);
          adj[j].push_back(i);
        }
      }
    }
  }
  for(int i=1; i<=n; i++){
    if(!mark[i]){
      mark[i]=true;
      dfs(i);
    }
  }
  printf("%d\n", ans);
  return 0;
}
