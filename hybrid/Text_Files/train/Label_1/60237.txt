from collections import defaultdict, deque

n = int(input())
s = set(map(int, input().split()))
dp = [-1] * (1 << 20)
p = [-1] * (1 << 20)
vis = [0] * (1 << 20)
mn = 0
dp[0] = 0
q = deque([(0, 0)])
while q:
    u, x = q.popleft()
    if u == (1 << 20) - 1:
        if x > mn:
            mn = x
            p[0] = 0
            for i in range(1, n):
                p[1 << i] = p[x & ((1 << (i + 1)) - 1)]
            break
    for i in s:
        if not vis[x ^ i]:
            vis[x ^ i] = 1
            dp[x ^ i] = x
            q.append((x ^ i, x + 1))
if mn == 0:
    print(0)
    print(0)
else:
    print(mn)
    ans = []
    for i in range(1 << mn):
        ans.append(p[i])
    ans.reverse()
    print(*ans)
