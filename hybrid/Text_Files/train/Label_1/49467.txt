import java.io.*;
import java.util.*;

public class Main {

    public static class Stone implements Comparable<Stone> {
        int a, b, c;

        public Stone(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }

        @Override
        public int compareTo(Stone o) {
            if (a != o.a) {
                return Integer.compare(a, o.a);
            } else if (b != o.b) {
                return Integer.compare(b, o.b);
            } else {
                return Integer.compare(c, o.c);
            }
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof Stone)) return false;
            Stone stone = (Stone) o;
            return a == stone.a && b == stone.b && c == stone.c;
        }

        @Override
        public int hashCode() {
            return Objects.hash(a, b, c);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        Set<Stone> stones = new HashSet<>();
        for (int i = 0; i < n; i++) {
            StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            int a = Integer.parseInt(tokenizer.nextToken());
            int b = Integer.parseInt(tokenizer.nextToken());
            int c = Integer.parseInt(tokenizer.nextToken());
            stones.add(new Stone(a, b, c));
        }
        List<Stone> list = new ArrayList<>(stones);
        Collections.sort(list);
        int maxVolume = 0;
        int[] stoneIndices = new int[2];
        for (int i = 0; i < list.size(); i++) {
            for (int j = i + 1; j < list.size(); j++) {
                Stone stone1 = list.get(i);
                Stone stone2 = list.get(j);
                int a = Math.max(stone1.a, stone2.a);
                int b = Math.max(stone1.b, stone2.b);
                int c = Math.max(stone1.c, stone2.c);
                int volume = a * b * c;
                if (volume > maxVolume) {
                    maxVolume = volume;
                    stoneIndices[0] = i + 1;
                    stoneIndices[1] = j + 1;
                }
            }
        }
        System.out.println(stoneIndices[0] != 0 || stoneIndices[1] != 0 ? 2 : 1);
        System.out.println(stoneIndices[0] + " " + stoneIndices[1]);
    }
}
