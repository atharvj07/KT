#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<ll,ll>
#define rep(i,x) for(ll i=0;i<x;i++)
#define repp(i,x) for(ll i=1;i<=x;i++)

const ll N=1e5+5,mod=1e9+7;

char g[55][55];
bool vis[55][55],flag;
int n,m;
void dfs(int x,int y)
{
    if(x<1||x>n||y<1||y>m){
        flag=0;
        return;
    }
    if(g[x][y]=='W'){
        flag=0;
        return;
    }
    if(vis[x][y]==1){
        flag=0;
        return;
    }
    vis[x][y]=1;
    dfs(x-1,y);
    dfs(x,y-1);
    dfs(x+1,y);
    dfs(x,y+1);
}

int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin>>n>>m;
    rep(i,n)
    {
        rep(j,m)
        {
            cin>>g[i+1][j+1];
        }
    }
    flag=1;
    bool f=0;
    rep(i,n)
    {
        rep(j,m)
        {
            if(g[i+1][j+1]=='B')
            {
                memset(vis,0,sizeof vis);
                dfs(i+1,j+1);
                if(flag==1)f=1;
            }
        }
    }
    if(f==1)cout<<"YES"<<endl;
    else cout<<"NO"<<endl;
    return 0;
}
