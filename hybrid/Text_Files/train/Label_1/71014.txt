MOD = int(1e9) + 7

def comb(n, m):
    if m > n: return 0
    num = 1
    den = 1
    for i in range(m):
        num = (num * (n - i)) % MOD
        den = (den * (i + 1)) % MOD
    return (num * pow(den, MOD - 2, MOD)) % MOD

for _ in range(int(input())):
    n, m, k = map(int, input().split())
    a = sorted([int(x) for x in input().split()])
    ans = 0
    l, r = 0, 0
    for i in range(n):
        while l < n and a[l] < a[i] - k: l += 1
        while r < n and a[r] <= a[i] + k: r += 1
        if r > l: ans = (ans + comb(r - l - 1, m - 1)) % MOD
    print(ans)
