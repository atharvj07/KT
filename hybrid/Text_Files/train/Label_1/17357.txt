#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a, q = 0;
  cin >> n;
  map<int, int> mp;
  for (int i = 0; i < n; i++) {
    cin >> a;
    mp[a]++;
  }
  for (auto it : mp) {
    if (it.second > 1) q += it.second - 1;
  }
  cout << q << endl;
  vector<int> v;
  for (int i = 1; i <= n; i++) {
    if (mp[i] == 0) {
      v.push_back(i);
    } else if (mp[i] == 1) {
      v.insert(v.begin(), i);
    } else {
      while (mp[i] > 1) {
        v.push_back(i);
        mp[i]--;
      }
    }
  }
  for (auto it : v) cout << it << ' ';
  cout << endl;
}
