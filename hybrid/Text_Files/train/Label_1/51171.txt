import java.util.Scanner;
public class Main {
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long t=0,a[]=new long[n],b[]=new long[n];
        for(int i=0;i<n;++i){
            a[i]=sc.nextLong();
            b[i]=a[i];
        }
        int f=1,g,h=1;
        while(h!=0){
            h=0;
            for(int i=0;i<n-f;++i){
                if(f%2==1)b[i]=a[i]-a[i+1];
                else b[i]=a[i]+a[i+1];
                h+=Math.abs(b[i]);
            }
            for(int i=0;i<n-f;++i){
                a[i]=b[i];
            }
            f+=2;
            g=f/2;
            n-=g;
        }
        System.out.println(Math.abs(b[0])%1000000007);
    }
}
