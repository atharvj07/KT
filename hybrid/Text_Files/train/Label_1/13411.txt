#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int M = 1e9+7;
const int MAXN = 1e6+1;
int fact[MAXN];
int main(){
    ll t,a,b;
    cin >> t >> a >> b;
    if(b == 0 || b == 1 || t == 1) cout << "inf\n";
    else{
        if(a%t == 0 && a/t <= b) cout << fact[(int) b] << '\n';
        else cout << 0 << '\n';
    }
    return 0;
}
