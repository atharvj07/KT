import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] offers = new int[n][3];
        for (int i = 0; i < n; i++) {
            offers[i][0] = sc.nextInt();
            offers[i][1] = sc.nextInt();
            offers[i][2] = sc.nextInt();
        }
        sc.close();

        int[][] dp = new int[n+1][2001];
        for (int i = 1; i <= n; i++) {
            int[] offer = offers[i - 1];
            for (int j = 0; j <= 2000; j++) {
                dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
                if (j >= offer[1] && dp[i - 1][j - offer[1]] + offer[0] >= 0)
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - offer[1]] + offer[0] + offer[0] * Math.max(0, offer[2] - 1));
            }
        }
        int max = 0;
        for (int i = 0; i <= 2000; i++) {
            max = Math.max(max, dp[n][i]);
        }
        System.out.println(max);
    }
}
