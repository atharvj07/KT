#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) p[a[i - 1]] = i;
  vector<int> cnt(n);
  for (int i = b[n - 1]; i > 0; i--) {
    cnt[n - 1 + p[i]]++;
  }
  for (int i = n - 2; i >= 0; i--) {
    cnt[i + p[b[i]]] += cnt[i + 1];
  }
  for (int i = 0; i < n; i++) cout << n - cnt[i] << ' ';
  cout << endl;
}
