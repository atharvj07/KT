#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    set<ll> s;
    bool ok = 1;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      if (s.find(x) != s.end() || !s.size()) {
        s.insert(x);
        continue;
      }
      ll mx = *s.rbegin();
      if (mx < x && !s.count(x - mx)) {
        if (s.find(-x - mx) == s.end()) {
          ok = 0;
          break;
        }
      } else if (mx > x && !s.count(-x + mx)) {
        if (s.find(-x + mx) == s.end()) {
          ok = 0;
          break;
        }
      }
      s.insert(x);
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
