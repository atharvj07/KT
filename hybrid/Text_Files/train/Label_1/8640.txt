#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,q;cin>>n>>q;string s;cin>>s;
    for(int tc=1;tc<=q;tc++)
    {
        int l,r;cin>>l>>r;
        int fr[10]={};
        char c='<';bool d=0;
        for(int i=l-1;i<=r-1;i++)
        {
            if(isdigit(s[i]))
            {
                if(d)
                {
                    fr[s[i]-'0']++;
                    if(s[i]=='0')
                    {
                        d=0;s.erase(i,1);--i;--r;
                    }
                    else s[i]--;
                }
                else
                {
                    d=1;
                    fr[s[i]-'0']++;
                    if(s[i]=='0')
                    {
                        s.erase(i,1);--i;--r;
                    }
                    else s[i]--;
                }
            }
            else if(s[i]=='<')
            {
                if(d)d=0;
                else c='<';
            }
            else if(s[i]=='>')
            {
                if(d)d=0;
                else c='>';
            }
            if(c=='<')i--;
        }
        for(int i=0;i<10;i++)
        cout<<fr[i]<<" ";
        cout<<"\n";
    }
}
