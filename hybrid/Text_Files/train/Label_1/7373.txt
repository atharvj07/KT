#include <bits/stdc++.h>

using namespace std;

int m, dp[(1 << 15) + 5];
string s;

int main() {
    cin >> m;
    dp[(1 << 4) - 1] = 1;
    while (m--) {
        int c;
        cin >> c;
        s += to_string(c);
        for (int i = 0; i < (1 << 4) - 1; i++) {
            int x = (i << 1) | c;
            if ((x & (x >> 1)) == 0) dp[x] = dp[i];
            dp[x] += dp[i];
            dp[x] %= 1000000007;
        }
        cout << dp[(1 << 4) - 1] << endl;
    }
    return 0;
}
