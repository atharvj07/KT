import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long C = sc.nextLong();
        sushi[] s = new sushi[N];
        for (int i = 0; i < N; i++) {
            long x = sc.nextLong();
            int v = sc.nextInt();
            s[i] = new sushi(x, v);
        }
        long ans = solve(N, C, s);
        System.out.println(ans);
    }

    public static long solve(int N, long C, sushi[] s) {
        long[][] dp = new long[N + 1][2];
        dp[0][0] = 0;
        dp[0][1] = -Long.MAX_VALUE;
        for (int i = 1; i <= N; i++) {
            dp[i][0] = Math.max(dp[i - 1][0], dp[i - 1][1]);
            dp[i][1] = Math.max(dp[i - 1][1], dp[i - 1][0] + s[i - 1].v - s[i - 1].x);
        }
        long ans = Math.max(dp[N][0], dp[N][1]);
        return ans;
    }

    public static class sushi {
        long x;
        int v;

        public sushi(long x, int v) {
            this.x = x;
            this.v = v;
        }
    }
}
