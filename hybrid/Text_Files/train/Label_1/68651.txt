import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, q = map(int, input().split())
    a = input().strip()
    prefix = [0] * n
    for i in range(n):
        if i > 0:
            prefix[i] = prefix[i - 1]
        if a[i] == '+':
            prefix[i] += 1
        else:
            prefix[i] -= 1
    for _ in range(q):
        l, r = map(int, input().split())
        l -= 1
        pre = 0
        if l > 0:
            pre = prefix[l - 1]
        ans = (prefix[r - 1] - pre) // 2 + (prefix[r - 1] - pre) % 2
        print(ans)
