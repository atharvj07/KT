import sys

n, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))

if k % 2 != n % 2:
    print(0)
    sys.exit()

den = 1
for i in range(1, n + 1):
    den *= i
den = den % (10 ** 9 + 7)

num = 0
for i in range(n - 1):
    if a[i] == 1:
        for j in range(i + 1, n):
            if a[j] == 0:
                num += 1
num = num % (10 ** 9 + 7)

print((num * pow(den, 10 ** 9 + 5, 10 ** 9 + 7)) % (10 ** 9 + 7))
