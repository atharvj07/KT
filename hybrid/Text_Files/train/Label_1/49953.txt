import java.util.Scanner;

public class UnimodalArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        boolean isUnimodal = true;
        int increasingIndex = -1;
        int decreasingIndex = -1;

        for (int i = 1; i < n; i++) {
            if (arr[i] > arr[i - 1]) {
                if (increasingIndex != -1) {
                    isUnimodal = false;
                    break;
                }
                increasingIndex = i;
            } else if (arr[i] < arr[i - 1]) {
                if (decreasingIndex != -1) {
                    isUnimodal = false;
                    break;
                }
                decreasingIndex = i;
            }
        }

        System.out.println(isUnimodal ? "YES" : "NO");
    }
}
