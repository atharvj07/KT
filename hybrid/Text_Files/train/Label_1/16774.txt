#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    vector<int> slices = {6, 8, 10};
    vector<int> times = {15, 20, 25};
    long long min_time = 1e18;
    for (int i = 0; i < (1 << 3); i++) {
      long long total_slices = 0;
      long long total_time = 0;
      for (int j = 0; j < 3; j++) {
        if (i & (1 << j)) {
          total_slices += slices[j];
          total_time += times[j];
        }
      }
      if (total_slices >= n) {
        min_time = min(min_time, total_time);
      }
    }
    cout << min_time << endl;
  }
  return 0;
}
