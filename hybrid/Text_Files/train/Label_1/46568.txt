import java.io.*;
import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long a00 = scanner.nextLong();
        long a01 = scanner.nextLong();
        long a10 = scanner.nextLong();
        long a11 = scanner.nextLong();
        if (a01 == 0 && a10 == 0) {
            for (int i = 0; i < a00; i++) {
                System.out.print("0");
            }
            for (int i = 0; i < a11; i++) {
                System.out.print("1");
            }
        } else if (a00 == 0 && a11 == 0) {
            for (int i = 0; i < a01; i++) {
                System.out.print("01");
            }
            for (int i = 0; i < a10; i++) {
                System.out.print("10");
            }
        } else if (a01 == a10 && a00 == a11) {
            for (int i = 0; i < a00; i++) {
                System.out.print("01");
            }
            for (int i = 0; i < a01; i++) {
                System.out.print("10");
            }
        } else {
            System.out.print("Impossible");
        }
    }
}
