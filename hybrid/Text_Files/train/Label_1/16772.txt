#include<vector>
#include<cstdio>

int main() {
	int l, r, ans = -1;
	scanf("%d%d", &l, &r);
	std::vector<int> v(10, 0);
	for (int i = l; i <= r; ++i) {
		v.assign(10, 0);
		int x = i;
		bool valid = true;
		while (x) {
			v[x % 10]++;
			if (v[x % 10] > 1) {
				valid = false;
				break;
			}
			x /= 10;
		}
		if (valid) {
			ans = i;
			break;
		}
	}
	printf("%d\n", ans);
}
