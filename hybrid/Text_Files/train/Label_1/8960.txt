#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, a;
    scanf("%d", &n);
    vector<int> v;
    for (int i = 0; i < n; i++) {
        scanf("%d", &a);
        v.push_back(a);
    }
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (v[i] < v[i - 1]) ans++;
    }
    printf("%d\n", ans);
    return 0;
}
