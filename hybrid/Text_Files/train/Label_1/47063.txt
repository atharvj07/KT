import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int S = sc.nextInt();
        int[] s = new int[N];
        int[] a = new int[N];
        int[] b = new int[N];
        for (int i = 0; i < N; i++) {
            s[i] = sc.nextInt();
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[][] dp = new int[N + 1][S + 1];
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= S; j++) {
                if (s[i - 1] <= j) {
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i - 1][j - s[i - 1]] + a[i - 1]);
                }
                if (s[i - 1] <= j) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - s[i - 1]] + b[i - 1]);
                } else {
                    dp[i][j] = dp[i - 1][j];
                }
            }
        }
        System.out.println(dp[N][S]);
    }
}
