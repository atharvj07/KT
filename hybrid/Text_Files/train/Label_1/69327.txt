import calendar

n = int(input())
dates = []
for _ in range(n):
    m, d, p, t = map(int, input().split())
    dates.append(((m-1)*31+d-t, m-1, d-t, p))

dates.sort()

need = 0
current = []
for s, m, d, p in dates:
    while current and current[0][0] <= s:
        _, end, _, _ = current.pop(0)
        if end == d:
            p -= 1
    while p > 0:
        current.append((s, -1, -1, p))
        p -= current[-1][3]
        need += 1

print(need)
