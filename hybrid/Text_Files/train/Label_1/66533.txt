def can_construct_beautiful_sequence(a, b, c, d):
    if abs(b - c) > 1 or abs(a - d) > 1:
        return "NO", []
    
    if b < c:
        if a > 0 and d > 0:
            return "NO", []
        sequence = [1] * b + [2] * c
        if a > 0:
            sequence = [0] * a + sequence
        else:
            sequence = sequence + [3] * d
    elif b > c:
        if a > 0 and d > 0:
            return "NO", []
        sequence = [1] * b + [2] * c
        if d > 0:
            sequence = sequence[::-1] + [3] * d
        else:
            sequence = [0] * a + sequence[::-1]
    else:  # b == c
        sequence = [1] * b + [2] * c
        if a > 0:
            sequence = [0] + sequence
        if d > 0:
            sequence = sequence + [3]
    
    return "YES", sequence

a, b, c, d = map(int, input().split())
result, sequence = can_construct_beautiful_sequence(a, b, c, d)
print(result)
if result == "YES":
    print(*sequence)
