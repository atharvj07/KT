#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vb = vector<bool>;
using vi = vector<int>;
using vl = vector<ll>;
const ll M = 1e9 + 7;

int main()
{
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int t;
    cin >> t;
    while (t--)
    {
        int n, k, m;
        cin >> n >> k >> m;
        vi a(m);
        for (auto &i : a)
            cin >> i;
        set<int> b(a.begin(), a.end());
        int st = 0;
        for (int i = 1; i <= n; i += k)
        {
            if (st == m)
                break;
            int l = i + k / 2 - 1, h = i + k - 1;
            auto it1 = b.lower_bound(l);
            auto it2 = b.lower_bound(h);
            if (it1 != b.end())
            {
                st++;
                b.erase(it1);
            }
            else if (it2 != b.end())
            {
                st++;
                b.erase(it2);
            }
        }
        cout << (st == m ? "YES\n" : "NO\n");
    }
    return 0;
}
