#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, a[100001], b[100001], pos[100001], sum[100001];
int main() {
  cin >> n;
  for (ll i = 1; i <= n; i++) { cin >> a[i] >> b[i]; }
  sort(a + 1, a + n + 1);
  sort(b + 1, b + n + 1);
  for (ll i = 1; i <= n; i++) {
    pos[a[i]] = i;
    pos[b[i] + n] = i;
  }
  for (ll i = 1; i <= 2 * n; i++) { pos[i] = pos[i] - 1; }
  sort(pos + 1, pos + 2 * n + 1);
  for (ll i = 1; i <= 2 * n; i++) { sum[i] = sum[i - 1] + pos[i]; }
  for (ll i = 1; i <= n; i++) {
    ll ta = pos[i], tb = pos[i + n];
    ll ra = sum[ta - 1], rb = sum[2 * n] - sum[tb];
    cout << a[i] * ra + b[i] * rb << endl;
  }
  return 0;
}
