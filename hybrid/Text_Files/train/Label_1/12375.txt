#include <bits/stdc++.h>
using namespace std;
#define MAX 200005
vector<int>primes;
bitset<MAX>bs;
void sieve(int n){
    bs.set();
    bs[0]=bs[1]=0;
    primes.push_back(2);
    for(int i=4;i<=n;i+=2) bs[i]=0;
    int sq=sqrt(n);
    for(int i=3;i<=sq;i+=2){
        if(bs[i]==1){
            primes.push_back(i);
            for(int j=i*i;j<=n;j+=i) bs[j]=0;
        }
    }
}
int main(){
    sieve(MAX-1);
    int t,n,e;
    cin>>t;
    while(t--){
        cin>>n>>e;
        vector<int>v(n);
        for(int i=0;i<n;i++) cin>>v[i];
        const int sz=primes.size();
        int ans=0;
        for(int i=0;i<n;i++){
            int cur=1;
            for(int j=i,cnt=0;j<n;j+=e,cnt++){
                cur*=v[j];
                if(cur==1) break;
                if(binary_search(primes.begin(),primes.end(),cur)) ans++;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
