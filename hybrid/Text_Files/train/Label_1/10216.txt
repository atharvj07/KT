#include <vector>
#include <iostream>
#include <cmath>

int main() {
    int n, k;
    std::cin >> n >> k;
    int d[n], s[n];
    for (int i = 0; i < n - 1; i++) std::cin >> d[i];
    for (int i = 0; i < n - 1; i++) std::cin >> s[i];

    std::vector<double> dp(n);
    double ans = 0;
    dp[0] = 0;
    for (int i = 1; i < n; i++) {
        dp[i] = dp[i - 1] + d[i - 1];
        for (int j = i - k - 1; j >= 0; j--) {
            double t = dp[j] + d[j];
            if (s[j] >= t - dp[j]) dp[i] = std::min(dp[i], t);
            else {
                double t_fuel = round((t - dp[j] - s[j]) / d[j] + 0.5) * d[j];
                t += t_fuel;
                dp[i] = std::min(dp[i], t);
            }
        }
    }
    std::cout << (int)round(dp[n - 1]);
    return 0;
}
