#include<bits/stdc++.h>
using namespace std;
const int INF = 1e9;
int n, x;
vector<int> v, pre;

int main() {
	cin >> n;
	v.resize(n);
	pre.resize(n);
	for(int i = 0; i < n; i++) {
		cin >> x;
		v[i] = x, pre[i] = x;
	}
	sort(v.rbegin(), v.rend()); // decreasing order
	for(int i = 1; i < n; i++) {
		if(v[i] < v[i - 1])
			v[i] = v[i - 1];
	}
	for(int i = 0; i < n; i++) {
		pre[i] = v[i];
		if(i > 0) pre[i] += pre[i - 1];
	}
	int ltime = 0, ans = INF;
	for(int i = 0; i < n; i++) {
		ltime += v[i] + 1;
		int rtime = (i == 0 ? 0 : pre[i - 1] - v[i]);
		ans = min(ans, ltime + rtime);
		ltime++; // eat
	}
	cout << ans << "\n";
	return 0;
}
