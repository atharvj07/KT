#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, t;
    cin >> n >> m >> t;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        int h, mi, s;
        char c;
        cin >> h >> c >> mi >> c >> s;
        a[i] = h * 3600 + mi * 60 + s;
    }
    vector<int> b(n);
    int r = 0;
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j < n && a[i] - a[j] <= t) { j++; }
        if (j - i > m) {
            cout << "No solution" << endl;
            return 0;
        }
        b[i] = r + 1;
        r++;
    }
    cout << r << endl;
    for (int i = 0; i < n; i++) { cout << b[i] << endl; }
    return 0;
}
