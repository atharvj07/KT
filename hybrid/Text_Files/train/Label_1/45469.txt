import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();

        Arrays.sort(arr);

        int days = 0, problems = 0;
        for (int i = n - 1; i >= 0; i--) {
            problems += arr[i];
            days++;
            if (problems < days) {
                break;
            }
        }

        System.out.println(days);
    }
}
