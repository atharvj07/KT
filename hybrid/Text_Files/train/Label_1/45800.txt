import java.util.*;

public class Main{
    public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int N = in.nextInt();
		int[][] f = new int[N][2];
		for (int i = 0; i < N; i++) {
			f[i][0] = in.nextInt();
			f[i][1] = in.nextInt();
		}
		Arrays.sort(f, new Comparator<int[]>() {
			public int compare(int[] a, int[] b) {
				return a[1] - b[1];
			}
		});
		int ans = 0;
		int r = 0;
		for (int[] a : f) {
			if (r >= a[0]) {
				ans++;
				r = a[1];
			}
		}
		System.out.println(ans);
    }
}
