from sys import stdin

x, y, x0, y0 = map(int, stdin.readline().split())
s = stdin.readline().strip()
ans = [0] * (len(s) + 1)

for i in range(x * y):
    x_, y_ = x0, y0
    cnt = 0
    for j in range(len(s)):
        if x_ == x0 and y_ == y0 and i % (x * y) == 0:
            cnt += 1
        if x_ == x0 and y_ == y0 and i % (x * y) != 0:
            break
        if s[j] == 'L' and y_ > 1:
            y_ -= 1
        if s[j] == 'R' and y_ < y:
            y_ += 1
        if s[j] == 'D' and x_ < x:
            x_ += 1
        if s[j] == 'U' and x_ > 1:
            x_ -= 1
    ans[cnt] += 1

for i in ans:
    print(i)
