x, y, x0, y0 = map(int, input().split())
s = input()
xr, yr, xm, ym, d = x0, y0, 0, 0, {'D': 1, 'L': -1, 'R': 1, 'U': -1}
ans = [1] * (len(s) + 1)
for c in s:
    x0 += d[c]
    if xm == x0 and ym == y0:
        print(' '.join(map(str, ans)))
        exit()
    y0 += d[c]
    if xm == x0 and ym == y0:
        print(' '.join(map(str, ans)))
        exit()
    if c == 'D': ym += 1
    if c == 'L': xm += 1
    if c == 'R': xm -= 1
    if c == 'U': ym -= 1
    ans[xr - x0 + yr - y0] += 1
print(' '.join(map(str, ans)))
