board = [input() for i in range(8)]
weight = {'q':9, 'r':5, 'b':3, 'n':3, 'p':1}
W,B = 0,0
for i in board:
    for j in i:
        if j.isupper():
            W += weight.get(j.lower(),0)
        else:
            B += weight.get(j.lower(),0)
if W > B:
    print('White')
elif W < B:
    print('Black')
else:
    print('Draw')
