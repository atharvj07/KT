from collections import defaultdict
n=int(input())
gr=defaultdict(list)
for i in range(n-1):
    a,b=map(int,input().split())
    a-=1
    b-=1
    gr[a].append(b)
    gr[b].append(a)
q=[]
for i in range(n):
    q.append((len(gr[i]),i))
q.sort()
ans=[]
cur=[0]*n
t=0
while q:
    t+=1
    l=len(q)
    for i in range(l):
        x=q.pop(0)
        v=x[1]
        if cur[v]!=0:
            continue
        ans.append([])
        for u in gr[v]:
            if cur[u]==0:
                ans[-1].append(u+1)
                cur[u]=t
                cur[v]=t
    for i in range(n):
        if cur[i]==t:
            q.append((len(gr[i]),i))
    q.sort()
print(t)
for a in ans:
    print(len(a),*a)
