#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    string ans;
    bool flag = false;
    for (int i = 0; i < n; i++) {
        if (a[i].first == 500) {
            ans += "A";
            flag = true;
        } else if (a[i].second == 500) {
            ans += "G";
            flag = true;
        } else {
            ans += "A";
        }
    }
    if (!flag) {
        cout << "-1" << endl;
        return 0;
    }
    int sumA = 0, sumG = 0;
    for (int i = 0; i < n; i++) {
        if (ans[i] == 'A') {
            sumA += a[i].first;
        } else {
            sumG += a[i].second;
        }
    }
    if (abs(sumA - sumG) > 500) {
        cout << "-1" << endl;
        return 0;
    }
    cout << ans << endl;
    return 0;
}
