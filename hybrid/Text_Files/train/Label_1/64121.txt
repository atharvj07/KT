import sys
input = sys.stdin.readline

def main():
    n = int(input())
    arr = [0]
    tot = 0.0
    for i in range(n):
        t, *a = map(int, input().split())
        if t == 1:
            l, x = a
            arr[:l] = [x + y for y in arr[:l]]
            tot += x * l
        elif t == 2:
            k, = a
            arr.append(k)
            tot += k
        else:
            tot -= arr.pop()
        print(tot / len(arr))

main()
