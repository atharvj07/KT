#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k, m, t;
  cin >> n >> k >> m >> t;
  vector<int> v(n + 1, 1);
  while (t--) {
    int x, y;
    cin >> x >> y;
    if (x == 1) {
      v[y] = 1;
      n++;
      if (y <= k) k++;
    } else {
      if (k > y + 1) k--;
      v[y] = 0;
      n--;
    }
    cout << n << ' ' << k << '\n';
  }
  return 0;
}
