import sys

input = sys.stdin.readline

n, k = map(int, input().split())

videos = []
for _ in range(n):
    s, m = map(int, input().split())
    videos.append((s, m))

videos.sort()

servers = [0] * k
for s, m in videos:
    servers[0] = max(servers[0], s)
    servers[0] += m
    servers.sort()
 
print(*servers)
