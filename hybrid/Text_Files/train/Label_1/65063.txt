import sys

# read the input
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))

# find the maximum aesthetic appeal and the number of trees with the same maximum appeal
max_appeal = max(arr)
max_count = arr.count(max_appeal)

# find the indices of the trees with the maximum aesthetic appeal
max_indices = [i for i, x in enumerate(arr) if x == max_appeal]

# check if there are at least two trees with the maximum aesthetic appeal
if max_count < 2:
    print(-1)
    exit()

# find the indices of the first and last trees with the maximum aesthetic appeal
first_max_idx = max_indices[0]
last_max_idx = max_indices[-1]

# calculate the total aesthetic appeal of the woodland belt after the Smart Beaver's intervention
total_appeal = max_appeal * 2 + sum(arr) - max_appeal * 2

# calculate the number of trees the Beaver needs to cut down
num_cut = n - 2

# print the results
print(total_appeal, num_cut)
print(*list(range(first_max_idx + 1, last_max_idx)))
