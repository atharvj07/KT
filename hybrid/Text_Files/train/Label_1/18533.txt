#include <vector>
#include <cmath>
#include <cstdio>
#include <algorithm>
using namespace std;

int n;
vector<int> a(3), b(3);

int gcd(int x, int y) {
  if (!x || !y) return 1;
  return gcd(y, x % y);
}

int main() {
  scanf("%d%d%d%d", &a[0], &a[1], &a[2], &b[0]);
  for (int i = 0; i < 3; ++i) {
    if (a[0] == a[1] && b[i] <= 1) {
      if (b[i]) puts("0");
      if (a[0]) puts("1");
      return 0;
    }
  }
  for (int i = 0; i < 3; ++i) {
    if (a[1] == a[2] && b[i] <= 1) {
      if (b[i]) puts("1");
      if (a[1]) puts("0");
      return 0;
    }
  }
  int c = gcd(gcd(a[0], a[1]), a[2]);
  if (c == 1 || a[1] - a[0] < c || a[2] - a[1] < c) {
    puts("Impossible");
    return 0;
  }
  for (int i = 0; i < 3; ++i) b[i] -= c;
  for (int i = 0; i < 3; ++i) {
    if (b[i]) {
      if (i == 1 && b[i] != c) {
        puts("Impossible");
        return 0;
      }
      for (int j = 0; j < 3; ++j) {
        int s = ceil(double(b[i]) / (a[i] - a[j]));
        for (int k = 0; k < s; ++k) {
          if (i == 0 && j == 1) putchar('0');
          if (i == 0 && j == 2) putchar('1');
          if (i == 1 && j == 2) putchar('0');
          if (i == 1 && j == 0) putchar('1');
        }
        for (int k = 0; k < b[i]; ++k) putchar('1');
        for (int k = 0; k < a[i] - a[j] - b[i]; ++k) {
          if (i == 0 && j == 1) putchar('1');
          if (i == 0 && j == 2) putchar('0');
          if (i == 1 && j == 2) putchar('1');
          if (i == 1 && j == 0) putchar('0');
        }
      }
      return 0;
    }
  }
  for (int i = 0; i < 3; ++i) {
    for (int j = 0; j < a[i]; ++j) {
      if (i == 0) putchar('1');
      if (i == 1) putchar('0');
      if (i == 2) putchar('1');
    }
  }
  return 0;
}
