import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int f = scanner.nextInt();
            int[] a = new int[8];
            int[] b = new int[8];
            for (int i = 0; i < 8; i++) {
                a[i] = scanner.nextInt();
            }
            for (int i = 0; i < 8; i++) {
                b[i] = scanner.nextInt();
            }
            boolean win = false;
            for (int i = 0; i < 8; i++) {
                for (int j = 0; j < 8; j++) {
                    if (a[i] * b[j] != 0) {
                        win = true;
                        break;
                    }
                }
            }
            if (!win) {
                System.out.println("Deal");
                continue;
            }
            boolean alice = false;
            for (int i = 0; i < 8; i++) {
                if (a[i] != 0) {
                    alice = true;
                    break;
                }
            }
            for (int i = 0; i < 8; i++) {
                if (b[i] != 0) {
                    alice = false;
                    break;
                }
            }
            if (f == 0) {
                alice = !alice;
            }
            System.out.println(alice ? "Alice" : "Bob");
        }
    }
}
