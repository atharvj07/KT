import sys
input=sys.stdin.readline

n,m,k=map(int,input().split())
g=[[]for i in range(n+1)]
for i in range(m):
    a,b=map(int,input().split())
    g[a].append(b)
    g[b].append(a)
vis=[False for i in range(n+1)]
ans=[]

def dfs(v,u):
    vis[v]=True
    ans.append(v)
    if len(ans)>k:
        return True

    for i in g[v]:
        if i!=u and not vis[i]:
            if dfs(i,v):
                return True
    ans.pop()
    vis[v]=False
    return False

for i in range(1,n+1):
    if len(g[i])<k:
        continue
    if dfs(i,i):
        break

print(len(ans))
for i in ans:
    print(i,end=' ')
