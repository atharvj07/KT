t = int(input().strip())
for _ in range(t):
    k = int(input().strip())
    first_row = input().strip().split()
    second_row = input().strip().split()

    # Locate E
    e_position = first_row.index('E') if 'E' in first_row else (second_row.index('E') + (2 * k + 1))

    # Early check for impossibility
    if e_position not in [k, 3*k+1]:
        print("SURGERY FAILED")
        continue

    # Shortcut definitions
    print("SURGERY COMPLETE")
    if e_position == k:
        # Moves to get E from the center of the first row to the correct position
        print("A r" + "d"*(2*k) + "l"*(2*k) + "u"*(2*k-1) + "r"*(2*k-1))
    else:
        # Moves to get E from the center of the second row to the correct position
        print("B l" + "u"*(2*k) + "r"*(2*k) + "d"*(2*k-1) + "l"*(2*k-1))
    print("DONE")
