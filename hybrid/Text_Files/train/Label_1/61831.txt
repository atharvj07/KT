import sys

n, m, x, y, vx, vy = map(int, sys.stdin.readline().split())

if vx == 0:
    if x == 0:
        print(0, y)
    else:
        print(n, y)
elif vy == 0:
    if y == 0:
        print(x, 0)
    else:
        print(x, m)
elif vx + vy == 0 or vx + vy == 2:
    print(-1)
elif vx * vy == 0:
    if vx == 1:
        print(n, y)
    elif vx == -1:
        print(0, y)
    elif vy == 1:
        print(x, m)
    elif vy == -1:
        print(x, 0)
else:
    if (n - x) / (-vy) == (m - y) / vx:
        print(n, m)
    elif (n - x) / (-vy) == y / vx:
        print(n, 0)
    elif x / (-vy) == (m - y) / vx:
        print(0, m)
    elif x / (-vy) == y / vx:
        print(0, 0)
    else:
        print(-1)
