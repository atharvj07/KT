#include <bits/stdc++.h>
using namespace std;
#define maxn 300010
vector<int> G[maxn], ans;
int deg[maxn], a[maxn];
bool dfs(int nod, int t, int p){
    if(deg[nod] > 0 && a[nod] != t){
        return 0;
    }
    if(deg[nod] < 0 && a[nod] == t){
        return 0;
    }
    a[nod] = t;
    for(auto x : G[nod]){
        if(x == p)continue;
        if(!dfs(x, 1 - t, nod)){
            return 0;
        }
    }
    return 1;
}
int main() {
    int n, m;
    cin >> n >> m;
    for(int i = 1; i <= n; i++){
        int x;
        cin >> x;
        deg[i] = x;
    }
    for(int i = 1; i <= m; i++){
        int x, y;
        cin >> x >> y;
        G[x].push_back(y);
        G[y].push_back(x);
    }
    if(!dfs(1, 0, 0)){
        cout << -1;
        return 0;
    }
    int anss = 0;
    for(int i = 1; i <= n; i++){
        for(auto x : G[i]){
            if(a[i] != a[x]){
                ans.push_back(anss);
            }
            anss++;
        }
    }
    cout << ans.size() << '\n';
    for(auto x : ans){
        cout << x + 1 << '\n';
    }
    return 0;
}
