from collections import defaultdict, deque
import sys

s = input()
t = input()
st = s + "#" + t
n = len(st)
z = [0] * n
l = 0
r = 0
for i in range(1, n):
    if i <= r:
        z[i] = min(r - i + 1, z[i - l])
    while i + z[i] < n and st[z[i]] == st[i + z[i]]:
        z[i] += 1
    if i + z[i] - 1 > r:
        l = i
        r = i + z[i] - 1
p = [0] * (len(t) + 1)
for i in range(len(t) + 1):
    if i == 0:
        p[i] = (0, -1)
    for j in range(len(z)):
        if z[j] >= len(t) - i:
            if j + 1 < len(s):
                p[i] = (j + 1, 0)
            else:
                p[i] = (j + 1, 1)
            break
if p[0][0] == 0:
    print(-1)
else:
    ans = []
    i = 0
    while i < len(t):
        x, y = p[i]
        i = i + len(t) - z[x] + 1
        if y == 0:
            ans.append((x, x + len(t) - 1))
        else:
            ans.append((x + len(t) - 1, x))
    print(len(ans))
    for x, y in ans:
        print(x + 1, y + 1)
