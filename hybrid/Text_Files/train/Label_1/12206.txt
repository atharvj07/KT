#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    unordered_map<char, int> mp;
    while (n--) {
        string s;
        cin >> s;
        mp[s[0]]++, mp[s[s.size() - 1]]++;
        for (int i = 1; i < s.size() - 1; i++) {
            mp[s[i]] += 2;
        }
    }
    bool flag = 1;
    for (auto p : mp) {
        if (p.second != 6) {
            flag = 0;
        }
    }
    cout << (flag ? "YES" : "NO") << "\n";
    return 0;
}
