#include <cmath>
#include <iostream>
#include <numeric>
#include <map>
using namespace std;

int main() {
  int n, a, r, m;
  cin >> n >> a >> r >> m;
  map<int, int> mp;
  int sum = 0;
  for (int i = 0; i < n; ++i) {
    int h;
    cin >> h;
    mp[h]++;
    sum += h;
  }
  int avg = llround((double)sum / n);
  int opt = 1e9;
  for (int i = avg - 1; i <= avg + 1; i++)
    if (mp.count(i) > 0) opt = min(opt, mp[i] * (a + m) + sum - i * n);
  cout << opt << '\n';
}
