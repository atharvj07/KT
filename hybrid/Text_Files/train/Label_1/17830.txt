#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); ++i)
typedef long long ll;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int max_cnt = 0, pos1 = 1, l, r;
  rep(i, n) {
    int cnt = 0;
    rep(j, n) {
      if (s[(i + j) % n] == ')') {
        if (cnt > 0) cnt--;
        else if (j != 0) break;
      } else {
        cnt++;
      }
    }
    if (cnt == 0 && max_cnt < cnt) {
      max_cnt = cnt;
      pos1 = i + 1;
    }
  }
  int pos2 = pos1 % n + 1;
  l = pos1, r = pos2;
  if (pos1 < pos2) l++;
  else pos2++;
  if (pos1 > pos2) {
    l = pos2;
    r = pos1;
  }
  cout << max_cnt * 2 << endl;
  cout << l << " " << r << endl;
  return 0;
}
