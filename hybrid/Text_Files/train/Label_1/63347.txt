import sys
input = sys.stdin.readline

n, m = map(int, input().split())

f = list(map(int, input().split()))
b = list(map(int, input().split()))

a = [0] * (n + 1)

for i in range(m):
  if a[b[i]]: return print("Ambiguity")
  a[b[i]] = f[i]

for i in range(1, n + 1):
  if not a[i]: return print("Impossible")

print("Possible")
print(*a[1:])
