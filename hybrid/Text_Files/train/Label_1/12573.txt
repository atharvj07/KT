#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> a(n + 1);
        for (int i = 0; i < n; i++) { a[i] = s[i] - '0'; }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1]) { ans = -1; }
        }
        if (ans == -1) {
            cout << "-" << endl;
        } else {
            for (int i = 0; i < n; i++) { cout << "1"; }
            cout << endl;
        }
    }
    return 0;
}
