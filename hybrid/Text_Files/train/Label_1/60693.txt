from collections import deque
from typing import Tuple, List

def solve(s: str, p: int, m: int) -> int:
    n = len(s)

    stack: List[Tuple[int, int]] = []
    ops: List[int] = []

    for i in range(n):
        if s[i].isdigit():
            stack.append((int(s[i]), 0))  # 0 -> digit
        elif s[i] == '(':
            ops.append(-1)  # -1 -> '('
        elif s[i] == ')':
            L = []
            while ops and ops[-1] != -1:
                L.append(ops.pop())
            assert ops and ops[-1] == -1
            ops.pop()
            ops += L
            if len(L) > 2:
                # no need to consider if there is only 1/0 digit
                stack.append((max(stack.pop()[0], stack.pop()[0]), 1))
        elif s[i] == '?':
            ops.append(0)

    for i in ops:
        if i == 0:
            if stack[-1][1] == 0:
                ops.pop()
                stack[-1] = (stack[-1][0] + stack.pop()[0], 0)
            else:
                if p:
                    ops[i] = 1
                    p -= 1
                elif m:
                    ops[i] = -1
                    m -= 1
                else:
                    ops.pop()
    
    # perform the calculation
    v = stack[0][0]
    for op in ops:
        if op == 1:
            v += stack.pop(0)[0]
        elif op == -1:
            v -= stack.pop(0)[0]
    
    return v

# Read Input Data
s = input()
p, m = map(int, input().split())

# Calculate the answer
ans = solve(s, p, m)

# Print the answer
print(ans)
