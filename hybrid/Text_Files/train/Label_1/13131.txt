#include <bits/stdc++.h>
#define int long long
using namespace std;

int n, m, s, ans;
int w[200005], c[200005], vis[200005];
vector<int> g[200005];

void dfs(int u, int p) {
  ans = max(ans, w[u]);
  vis[u] = 1;
  for (int v : g[u]) {
    if (v != p && !vis[v]) {
      dfs(v, u);
    }
  }
  vis[u] = 0;
  c[u] = 1;
  for (int v : g[u]) {
    if (v != p) {
      c[u] += c[v];
    }
  }
  ans = max(ans, w[u] + c[u] - 1);
}

signed main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> w[i]; }
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  cin >> s;
  dfs(s, s);
  cout << ans;
  return 0;
}
