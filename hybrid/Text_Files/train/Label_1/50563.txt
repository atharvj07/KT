import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] circles = new int[n][3];
        for (int i = 0; i < n; i++) {
            circles[i][0] = scanner.nextInt();
            circles[i][1] = scanner.nextInt();
            circles[i][2] = scanner.nextInt();
        }

        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int dx = circles[i][0] - circles[j][0];
                int dy = circles[i][1] - circles[j][1];
                int dist = dx * dx + dy * dy;
                if (dist > (circles[i][2] + circles[j][2]) * (circles[i][2] + circles[j][2])) {
                    count++;
                } else if (dist == (circles[i][2] + circles[j][2]) * (circles[i][2] + circles[j][2])) {
                    count += 2;
                }
            }
        }

        System.out.println(count + 1);
    }
}
