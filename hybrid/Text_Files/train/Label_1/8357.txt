#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<pair<int, int>> guests(k);
  vector<int> cnt(n + 1, 0);
  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    guests[i] = {x, y};
    cnt[x]++;
    cnt[y]++;
  }
  sort(guests.begin(), guests.end());
  int ans = 0;
  for (int i = 0; i < k; i++) {
    int x = guests[i].first;
    int y = guests[i].second;
    if (cnt[x] == 1 || cnt[y] == 1) ans++;
  }
  cout << ans << endl;
  return 0;
}
