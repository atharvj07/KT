import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int w1 = 0, w2 = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 100) {
                w1 += 1;
            } else {
                w2 += 1;
            }
        }
        if ( w1%2==0 && w2%2==0 || (w1>=2 && w2>=1)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
