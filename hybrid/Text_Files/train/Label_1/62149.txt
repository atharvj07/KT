def construct_array(X, d):
    # Initialize an empty list to store the final array
    final_array = []
    # Initialize the current value to be inserted into the final array
    current_val = 1
    # While there are still subsequences left to account for
    while X > 0:
        # Find the highest power of 2 such that 2^power <= X + 1
        power = 0
        while (1 << (power + 1)) <= X + 1:
            power += 1
        # Add 'power' number of current_val to the final array
        final_array += [current_val] * power
        # Decrease X by the number of new subsequences formed, which is 2^power - 1
        X -= (1 << power) - 1
        # Increment the current_val by d + 1 to ensure the difference condition is met
        current_val += d + 1
    # Return the final array
    return final_array

# Read the input values
X, d = map(int, input().split())

# Construct the array
result_array = construct_array(X, d)

# Output the results
print(len(result_array))
print(" ".join(map(str, result_array)))
