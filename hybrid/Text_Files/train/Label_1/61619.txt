import sys
input = sys.stdin.readline

def intersect(l1, r1, l2, r2):
    x1 = max(l1[0], l2[0])
    y1 = max(l1[1], l2[1])
    x2 = min(r1[0], r2[0])
    y2 = min(r1[1], r2[1])
    if x2 < x1 or y2 < y1:
        return 0
    return (x2 - x1) * (y2 - y1)

n, k = map(int, input().split())
a = []
p = []
for _ in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    a.append(((x1, y1), (x2, y2)))
    p.append(((x1 + x2) / 2, (y1 + y2) / 2))

for i in range(k):
    idx = -1
    mx_int = -1
    for j in range(n):
        if (j not in p) and intersect(a[j][0], a[j][1], a[0][0], a[0][1]) > mx_int:
            mx_int = intersect(a[j][0], a[j][1], a[0][0], a[0][1])
            idx = j
    if idx == -1:
        break
    p.append(p[idx])
    a.remove(a[idx])

ans = 0
for i in range(len(a)):
    for j in range(len(a)):
        if i == j: continue
        x1, y1 = p[i]
        x2, y2 = p[j]
        ans = max(ans, (abs(x1 - x2) + 1) * (abs(y1 - y2) + 1))

print(ans)
