#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, q; cin >> n >> q;
    vector<long long> a(n), pre(n + 1, 0), suf(n + 1, 0);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i];
        suf[i + 1] = suf[i];
        if (i != n - 1) { pre[i + 1] = max(pre[i + 1], a[i + 1] - a[i]); }
        if (i) { suf[i + 1] = max(suf[i + 1], a[i] - a[i - 1]); }
    }
    for (int i = 0; i < q; i++) {
        int l, r; cin >> l >> r;
        cout << pre[r] - pre[l - 1] + suf[r - 1] - suf[l] << endl;
    }
}
