import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] plates = new int[n];
        for (int i = 0; i < n; i++) {
            plates[i] = scanner.nextInt();
        }
        int[] friend1 = new int[m];
        int[] friend2 = new int[m];
        for (int i = 0; i < m; i++) {
            friend1[i] = scanner.nextInt() - 1;
            friend2[i] = scanner.nextInt() - 1;
        }
        if (canSurvive(plates, friend1, friend2)) {
            System.out.println("ALIVE");
            System.out.println(getSequence(plates, friend1, friend2));
        } else {
            System.out.println("DEAD");
        }
    }

    private static boolean canSurvive(int[] plates, int[] friend1, int[] friend2) {
        boolean[] eaten = new boolean[plates.length];
        for (int i = 0; i < friend1.length; i++) {
            if (plates[friend1[i]] > 0 || plates[friend2[i]] > 0) {
                eaten[friend1[i]] = true;
                eaten[friend2[i]] = true;
            }
        }
        return IntStream.range(0, plates.length).filter(i -> plates[i] > 0 && !eaten[i]).count() == 0;
    }

    private static String getSequence(int[] plates, int[] friend1, int[] friend2) {
        List<Integer> sequence = new ArrayList<>();
        boolean[] eaten = new boolean[plates.length];
        for (int i = 0; i < friend1.length; i++) {
            if (plates[friend1[i]] > 0) {
                eaten[friend1[i]] = true;
                sequence.add(i + 1);
                plates[friend1[i]]--;
            }
            if (plates[friend2[i]] > 0) {
                eaten[friend2[i]] = true;
                sequence.add(i + 1);
                plates[friend2[i]]--;
            }
        }
        for (int i = 0; i < plates.length; i++) {
            while (plates[i] > 0) {
                if (!eaten[i]) {
                    sequence.add(i + 1);
                    eaten[i] = true;
                }
                plates[i]--;
            }
        }
        return sequence.stream().map(i -> Integer.toString(i)).reduce((a, b) -> a + " " + b).get();
    }
}
