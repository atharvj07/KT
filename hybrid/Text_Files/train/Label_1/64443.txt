import sys

n, T = list(map(int, sys.stdin.readline().split()))
a = list(map(int, sys.stdin.readline().split()))
t = list(map(int, sys.stdin.readline().split()))

nA = [a[i]*t[i] for i in range(n)]

sA = sum(nA)
s = sum(a)

ans = 0.0
for i in range(n):
    ans = max(ans, (sA-nA[i])/(s-a[i])*1.0*T/(t[i]*1.0))

print(ans)
