import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

    private static int[] dist;
    private static int[][] graph;
    private static int[][] min;
    private static boolean[] visited;
    private static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = reader.readLine().split(" ");
        n = Integer.parseInt(firstLine[0]);
        int m = Integer.parseInt(firstLine[1]);
        String[] secondLine = reader.readLine().split(" ");
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = Integer.parseInt(secondLine[i]);
        }
        graph = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            String[] edge = reader.readLine().split(" ");
            graph[Integer.parseInt(edge[0])][Integer.parseInt(edge[1])] = 1;
            graph[Integer.parseInt(edge[1])][Integer.parseInt(edge[0])] = 1;
        }
        min = new int[n + 1][n + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(min[i], 100000000);
        }
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            dist = new int[n + 1];
            visited = new boolean[n + 1];
            dfs(i, i, 0, arr);
            for (int j = 1; j <= n; j++) {
                if (i == j) {
                    continue;
                }
                min[i][j] = Math.min(min[i][j], dist[j]);
                min[j][i] = min[i][j];
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                sum += min[i][j];
            }
        }
        long allPairs = ((long) n * (n - 1)) / 2;
        System.out.printf("%.6f\n", (double) sum / allPairs);
    }

    private static void dfs(int u, int start, int d, int[] arr) {
        visited[u] = true;
        dist[u] = d;
        List<Integer> neighbors = getNeighbors(u);
        for (int neighbor : neighbors) {
            if (!visited[neighbor]) {
                dfs(neighbor, start, d + arr[neighbor - 1], arr);
            } else if (neighbor == start) {
                min[u][start] = Math.min(min[u][start], dist[u]);
            }
        }
    }

    private static List<Integer> getNeighbors(int u) {
        List<Integer> neighbors = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (graph[u][i] == 1) {
                neighbors.add(i);
            }
        }
        return neighbors;
    }
}
