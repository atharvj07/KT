#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    std::string s;
    std::cin >> s;
    int ans = 0;
    bool inZero = false;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == '0') {
        inZero = true;
        continue;
      }
      if (inZero) ans++;
    }
    std::cout << ans << '\n';
  }
  return 0;
}
