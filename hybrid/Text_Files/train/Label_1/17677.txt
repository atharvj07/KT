#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x, a, b;
    cin >> n >> x >> a >> b;
    if (a > b) swap(a, b);
    int ans = b - a - 1;
    if (x & 1) ans++;
    x /= 2;
    ans += min(x, min(a - 1, n - b));
    cout << ans << endl;
  }
  return 0;
}
