#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  // cnt[i] = number of times s[i] = s[i + k] = s[i + 2k]
  // for some k
  vector<int> cnt(n, 0);
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (s[i] == s[j]) {
        cnt[i]++;
        cnt[j]++;
      }
    }
  }
  // prefix sum
  for (int i = 1; i < n; i++) {
    cnt[i] += cnt[i - 1];
  }
  // count pairs of integers l, r
  int ans = 0;
  for (int l = 0; l < n; l++) {
    for (int r = l + 2; r < n; r++) {
      if (l == 0) {
        ans += cnt[r];
      } else {
        ans += cnt[r] - cnt[l - 1];
      }
    }
  }
  cout << ans << endl;
  return 0;
}
