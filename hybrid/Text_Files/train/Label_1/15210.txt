#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> grid(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> grid[i][j];
    }
  }
  // count number of 'D's
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == 'D') cnt++;
    }
  }
  // output result
  if (cnt == 0)
    cout << "Poor Dima!" << endl;
  else {
    int ans = cnt;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (grid[i][j] == 'D') {
          // check if possible to go through 'DIMA' from (i, j)
          int ii = i, jj = j;
          int cur = 0;
          while (cur < 4) {
            if (ii < 0 || ii >= n || jj < 0 || jj >= m) break;
            if (grid[ii][jj] == "DIMA"[cur])
              cur++;
            else
              break;
            if (cur == 0) ii++;
            else if (cur == 1) jj--;
            else if (cur == 2) ii--;
            else if (cur == 3) jj++;
          }
          if (cur == 4) ans--;
        }
      }
    }
    if (ans == 0)
      cout << "Poor Inna!" << endl;
    else
      cout << cnt - ans << endl;
  }
  return 0;
}
