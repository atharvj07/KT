#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

pair<ll, ll> count(vector<ll> &a, ll l, ll r, ll x) {
    ll cnt = 0, mx = 0;
    for (ll i = l; i <= r; ++i) {
        if (a[i] == x) ++cnt;
        mx = max(mx, cnt);
    }
    return {cnt, mx};
}

int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        vector<ll> a(n);
        for (ll i = 0; i < n; ++i) cin >> a[i];
        ll ans = 2e9;
        for (ll i = 0; i < n; ++i) {
            for (ll j = i; j < n; ++j) {
                auto p = count(a, i, j, a[i]);
                if (p.second > p.first) ans = min(ans, j - i + 1);
            }
        }
        cout << (ans == 2e9 ? -1 : ans) << endl;
    }
    return 0;
}
