#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 1;
int n, k;
constexpr int mod = 1e9 + 7;
pair<int, int> a[maxn];
int sum[maxn], cnt[maxn];

int32_t main() {
  cin >> n >> k;
  for (int i = 1; i <= n; ++i) cin >> a[i].first >> a[i].second;
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; ++i) {
    sum[i] = sum[i - 1] + a[i].second - a[i].first + 1;
    cnt[i] = cnt[i - 1];
    int l = a[i].first, r = a[i].second;
    if (i > 1 && a[i].first == a[i - 1].second + 1) cnt[i] = cnt[i - 1] + 1;
    if (i > 1 && a[i].second == a[i - 1].first - 1) cnt[i] = cnt[i - 1] - 1;
    if (i > 1 && a[i].first == a[i - 1].first && a[i].second == a[i - 1].second)
      cnt[i] = cnt[i - 1];
  }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    for (int j = i + k - 1; j <= n; j += k) {
      ans = (ans + sum[j] - sum[i - 1] + mod) % mod;
      if (i != 1 && cnt[j] == cnt[i - 1] + 1) ans = (ans - a[i].second - a[i].first + 1 + mod) % mod;
      if (i != 1 && cnt[j] == cnt[i - 1] - 1) ans = (ans - a[i].second - a[i].first + 1 + mod) % mod;
    }
  }
  cout << ans << endl;
  return 0;
}
