import java.util.*;
import java.io.*;

class Main {

    static class Data implements Comparable<Data> {
        String key;
        int value;
        int index;

        public Data(String key, int value, int index) {
            this.key = key;
            this.value = value;
            this.index = index;
        }

        @Override
        public int compareTo(Data other) {
            if (key.equals(other.key)) {
                return index - other.index;
            }
            return key.compareTo(other.key);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(br.readLine());
        Map<String, List<Data>> map = new HashMap<>();
        List<Data> list = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            switch (type) {
                case 0:
                    String key = st.nextToken();
                    int value = Integer.parseInt(st.nextToken());
                    List<Data> values = map.getOrDefault(key, new ArrayList<>());
                    values.add(new Data(key, value, list.size()));
                    map.put(key, values);
                    list.add(new Data(key, value, list.size()));
                    break;
                case 1:
                    key = st.nextToken();
                    values = map.getOrDefault(key, new ArrayList<>());
                    for (Data data : values) {
                        System.out.println(data.value);
                    }
                    break;
                case 2:
                    key = st.nextToken();
                    values = map.remove(key);
                    if (values != null) {
                        for (Data data : values) {
                            list.remove(data);
                        }
                    }
                    break;
                case 3:
                    String l = st.nextToken();
                    String r = st.nextToken();
                    values = new ArrayList<>();
                    for (Data data : list) {
                        if (data.key.compareTo(l) >= 0 && data.key.compareTo(r) <= 0) {
                            values.add(data);
                        }
                    }
                    Collections.sort(values);
                    for (Data data : values) {
                        System.out.println(data.key + " " + data.value);
                    }
                    break;
            }
        }
    }
}
