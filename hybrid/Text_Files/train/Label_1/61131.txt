def almost_sorted(n: int, k: int) -> str:
    if k > n * (n + 1) // 2:
        return "-1"
    arr = list(range(1, n + 1))
    k -= 1
    for i in range(n - 1, -1, -1):
        j = k // i
        arr[i], arr[i - j - 1] = arr[i - j - 1], arr[i]
        k %= i
    return " ".join(map(str, arr))

for _ in range(int(input())):
    n, k = map(int, input().split())
    print(almost_sorted(n, k))
