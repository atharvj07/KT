n=int(input())
a=list(map(int,input().split()))
lis=[1]*n
lds=[1]*n
for i in range(1,n):
        for j in range(i):
                if(a[j]<a[i]):
                        lis[i]=max(lis[i],lds[j]+1)
for i in range(n-2,-1,-1):
        for j in range(n-1,i,-1):
                if(a[j]<a[i]):
                        lds[i]=max(lds[i],lis[j]+1)
ans=[0]*n
for i in range(n):
        if(lis[i]==lds[i]):ans[i]=3
        elif(lis[i]>lds[i]):ans[i]=2
        else:ans[i]=1
for i in ans:print(i,end='')
