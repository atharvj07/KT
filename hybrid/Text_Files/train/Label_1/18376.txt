#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n;
    cin >> n;
    long long ans = 0;
    for (long long d = 1; d <= n; ++d) {
        string s = to_string(d);
        bool good = 1;
        for (int i = 0; i < (int)s.length(); ++i) {
            good &= (s[i] == '4' || s[i] == '7');
        }
        if (good) ans++;
    }
    cout << ans << '\n';
    return 0;
}
