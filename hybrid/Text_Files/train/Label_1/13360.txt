#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> h(n);
  for (int i = 0; i < n; i++) { cin >> h[i]; }

  int ans = 0;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    ans += abs(h[i] - cur);
    cur = h[i];
    ans++;
  }
  cout << ans << endl;
  return 0;
}
