#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

int main() {
  string s;
  cin >> s;

  unordered_map<char, int> freq;
  for (auto c : s) freq[c]++;

  int even = 0, odd = 0;
  for (auto it : freq) {
    int freq = it.second;
    even += freq / 2;
    odd += (freq + 1) / 2;
  }

  cout << even << ' ' << odd << '\n';
}
