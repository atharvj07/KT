#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int N = 2e5 + 5;

int n;
int a[N], b[N], c[N];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) c[a[i]] = i;
    int ans = 0;
    for (int i = 0; i < n; i++) if (a[i]) {
        int x = c[b[i]];
        ans += x / n * (n - 1) + x % n;
    }
    cout << ans << endl;
}
