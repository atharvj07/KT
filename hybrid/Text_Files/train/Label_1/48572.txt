import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }

        int m = sc.nextInt();
        long[] b = new long[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextLong();
        }

        long sumA = Arrays.stream(a).sum(), sumB = Arrays.stream(b).sum();
        if (sumA != sumB) {
            System.out.println(-1);
            return;
        }
        
        int ans = n;
        boolean canSwap = true;
        for (int i = 1; i < n; i++) {
            if (a[i - 1] > a[i]) {
                canSwap = false;
            }
        }
        if (!canSwap) {
            ans = 1;
        }
        
        System.out.println(ans);
    }
}
