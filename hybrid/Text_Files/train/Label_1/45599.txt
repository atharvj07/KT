import java.util.*;

public class Main {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            System.out.println(Math.max(a - b, -b));
            if (a > b) {
                for (int i = 0; i < b; i++) {
                    System.out.print("ox");
                }
                for (int i = 0; i < a - b; i++) {
                    System.out.print("o");
                }
            } else if (a < b) {
                for (int i = 0; i < a; i++) {
                    System.out.print("xo");
                }
                for (int i = 0; i < b - a; i++) {
                    System.out.print("x");
                }
            } else {
                for (int i = 0; i < a; i++) {
                    System.out.print("o");
                }
            }
            System.out.println();
        }
    }
}
