#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m, points = 0, last = 0;
    cin >> n >> m;
    vector<int> v(n);
    while (m--) {
        int c;
        cin >> c;
        v[c - 1]++;
    }
    for (int i = 0; i < n; i++) {
        if (v[i] > 0)
            last = i;
        else
            points++;
    }
    cout << last + 1 - points << endl;
}
