import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] r = new int[n];
        String[] s = in.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            r[i] = Integer.parseInt(s[i]);
        }
        int[][] e = new int[n - 1][2];
        for (int i = 0; i < n - 1; i++) {
            s = in.readLine().split(" ");
            e[i][0] = Integer.parseInt(s[0]) - 1;
            e[i][1] = Integer.parseInt(s[1]) - 1;
        }
        int[] d = new int[n];
        int[] p = new int[n];
        Arrays.fill(d, -1);
        d[0] = 0;
        p[0] = -1;
        Queue<Integer> q = new LinkedList<>();
        q.add(0);
        while (!q.isEmpty()) {
            int x = q.poll();
            for (int[] edge : e) {
                int y = edge[0] == x ? edge[1] : edge[0];
                if (d[y] == -1 || d[y] > d[x] + 1) {
                    d[y] = d[x] + 1;
                    p[y] = x;
                    q.add(y);
                }
            }
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            if (p[i] != -1 && r[i] > r[p[i]]) {
                ans[i] = 1;
            }
        }
        int sum = 0;
        for (int x : ans) {
            sum += x;
        }
        System.out.println(sum);
    }
}
