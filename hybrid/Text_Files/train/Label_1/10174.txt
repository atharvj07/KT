#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        int sum = 0, ans = 0;
        for (int x : a) {
            sum += x % 3;
            ans += x % 3 == 0;
        }
        if (sum % 3 == 0) ans = n;
        cout << ans << '\n';
    }
    return 0;
}
