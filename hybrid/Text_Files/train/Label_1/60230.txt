from collections import Counter

for _ in range(int(input())):
  n, k = map(int, input().split())
  a = list(map(int, input().split()))
  cnt = Counter(a)

  ans = 0
  for x in cnt:
    if x % k == 0:
      continue
    rem = k - (x % k)
    need = cnt[x] * rem
    need -= cnt.get(x + rem, 0)
    ans += need // k + 1
  
  print(ans)
