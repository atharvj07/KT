n,m=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
a.sort()
b.sort()
ans=-1
for i in a:
    lb=0
    ub=len(b)-1
    j=-1
    while lb<=ub:
        mid=(lb+ub)//2
        if b[mid]>=2*i:
            j=mid
            ub=mid-1
        else:
            lb=mid+1
    if j!=-1:
        ans=b[j]//2
        break
print(ans)
