#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long a, b;
  cin >> a >> b;

  vector<pair<long long, long long>> factors;

  for (long long i = 1; i * i <= a; i++) {
    if (a % i == 0) {
      factors.push_back({i, a / i});
      if (i != a / i) factors.push_back({a / i, i});
    }
  }

  long long ans = 2e18;

  for (auto [x, y] : factors) {
    for (long long i = 1; i * i <= b; i++) {
      if (b % i == 0) {
        long long u = i, v = b / i;
        long long p = 2 * (x * u + x * v + y * u + y * v);
        ans = min(ans, p);
      }
    }
  }

  cout << ans << "\n";

  return 0;
}
