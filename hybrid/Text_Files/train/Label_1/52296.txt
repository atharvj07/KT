import java.util.Scanner;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        HashSet<Integer> a = new HashSet<>();
        for (int i = 0; i < p; i++) {
            a.add(sc.nextInt());
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            a.add(sc.nextInt());
        }
        System.out.println(a.size() == n ? "I become the guy." : "Oh, my keyboard!");
    }
}
