import sys

for _ in range(int(input())):
    n = int(input())
    ans = 0
    while n != 1:
        if n % 3 == 0:
            n = 2 * n // 3
        elif n % 2 == 0:
            n = n // 2
        elif n % 5 == 0:
            n = 4 * n // 5
        else:
            ans = -1
            break
        ans += 1
    print(ans)
