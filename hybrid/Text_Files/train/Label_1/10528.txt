#include<bits/stdc++.h>
using namespace std;
int main(){
    int t,x,y,p,q;
    cin>>t;
    while(t--){
        cin>>x>>y>>p>>q;
        if((long long)x*q==(long long)p*y) cout<<0<<endl;
        else if((long long)x*q>(long long)p*y) cout<<-1<<endl;
        else{
            cout<<(long long)(ceil((long double)(p*y)/(long double)(q*(x+1))))-x<<endl;
        }
    }
    return 0;
}
