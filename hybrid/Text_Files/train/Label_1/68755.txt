mod = 10**9 + 7
n, m = map(int, input().split())
arr = [[0] * (m + 2) for _ in range(n + 2)]
for i in range(1, n + 1):
    s = input()
    for j in range(1, m + 1):
        arr[i][j] = 1 if s[j - 1] == 'R' else 0
dp = [[0] * (m + 2) for _ in range(n + 2)]
dp[1][1] = 1
for i in range(1, n + 1):
    for j in range(1, m + 1):
        if arr[i][j] == 0:
            dp[i][j] += dp[i][j - 1] + dp[i - 1][j]
            if i < n and arr[i][j + 1] == 0:
                dp[i][j] += dp[i + 1][j - 1]
            if j < m and arr[i + 1][j] == 0:
                dp[i][j] += dp[i - 1][j + 1]
print(dp[n][m] % mod)
