n = int(input())
t = [int(x) for x in input().split()]
c = [20] * n

for i in range(1, n):
    ct = [c[i - 1]]
    for j in range(i, -1, -1):
        if j == 0:
            ct.append(20)
        elif t[i] - t[j] + 1 <= 90:
            ct.append(c[j] + 50 - c[i - 1])
        elif t[i] - t[j] + 1 <= 1440:
            ct.append(c[j] + 120 - c[i - 1])
        else:
            break
    c[i] = min(ct)

print(*c)
