import sys

n = int(input())
a = [int(x) for x in input().split()]
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    p, c = map(int, input().split())
    adj[p].append((c, len(adj[p])))
    adj[len(adj[p])].append((-c, len(adj[len(adj[p])])-1))

mn = 0
for i in range(n, 0, -1):
    mx = 0
    for c, j in adj[i]:
        mx = max(mx, a[i-1]+c)
    if mx > a[i-1]:
        a[i-1] = mx
        mn += 1

print(mn)
