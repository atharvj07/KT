#include <iostream>
#include <vector>
#include <climits>

using namespace std;

// Global variables
vector<bool> state;
int dist[101][101], n, m;
vector<int> p, ans;

void bfs(int x) {
  vector<bool> visited(n, false);
  vector<int> dist(n, INT_MAX);
  visited[x] = true;
  dist[x] = 0;
  for (int i = 1; i <= n; i++)
    if (state[i] && !visited[i]) {
      visited[i] = true;
      dist[i] = 1;
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i != j && !visited[j]) {
        visited[j] = true;
        dist[j] = min(dist[j], dist[i] + 1);
      }
  for (int i = 1; i <= n; i++) ::dist[x][i] = dist[i];
}

void dfs(int x, int cnt) {
  if (x == p.back()) {
    if (cnt < ans.size() || ans.empty()) { ans.clear(); ans = p; }
    return;
  }
  for (int i = 1; i <= n; i++)
    if (state[i] && p[cnt - 1] != i && ::dist[x][i] == ::dist[x][p[cnt - 1]] + 1) {
      p.push_back(i);
      dfs(i, cnt + 1);
      p.pop_back();
    }
}

int main() {
  ios_base::sync_with_stdio(0);
  cin >> n;
  state.resize(n + 1, true);
  for (int i = 1; i <= n; i++) {
    string t;
    cin >> t;
    for (int j = 0; j < n; j++)
      if (t[j] == '1') { state[i + 1] = false; state[j + 1] = false; }
  }
  for (int i = 1; i <= n; i++) bfs(i);
  cin >> m;
  p.resize(m);
  for (auto &el : p) cin >> el;
  ans.resize(100000);
  dfs(p[0], 1);
  cout << ans.size() << '\n';
  for (auto &el : ans) cout << el << ' ';
  return 0;
}
