#include <vector>
#include <iostream>

void heapSort(std::vector<int64_t> &a, int L, int R, int k) {
    for (int j = (R - L + 1) / k / 2; j >= 1; --j) {
        int i = L + (j - 1) * k;
        int idx = i;
        int v = a[i];
        while (i < L + (j - 1) * k + k) {
            if (a[i] > v) {
                idx = i;
                v = a[i];
            }
            ++i;
        }
        std::swap(a[L + (j - 1) * k], a[idx]);
    }
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);
    std::cout.tie(nullptr);
    int n;
    std::cin >> n;
    std::vector<int64_t> a(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
    }
    std::vector<int64_t> r(n - 1);
    for (int i = 1; i < n; ++i) {
        heapSort(a, 0, n - 1, i + 1);
        int violations = 0;
        for (int j = 1; j < n; ++j) {
            if (a[j] < a[j / (i + 1)]) {
                ++violations;
            }
        }
        r[i - 1] = violations;
    }
    for (auto &i : r) {
        std::cout << i << " ";
    }
    std::cout << std::endl;
    return 0;
}
