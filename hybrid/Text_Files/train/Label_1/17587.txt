#include <iostream>
using namespace std;

int main() {
    int n, c;
    cin >> n >> c;
    int last = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        int t;
        cin >> t;
        if (t - last > c)
            ans = 1;
        else
            ans++;
        last = t;
    }
    cout << ans << endl;
    return 0;
}
