def find_parent(parent, i):
    if parent[i] == i:
        return i
    return find_parent(parent, parent[i])

def union(parent, rank, x, y):
    xroot = find_parent(parent, x)
    yroot = find_parent(parent, y)
    
    if rank[xroot] < rank[yroot]:
        parent[xroot] = yroot
    elif rank[xroot] > rank[yroot]:
        parent[yroot] = xroot
    else:
        parent[yroot] = xroot
        rank[xroot] += 1

def minimum_gold(n, m, c, friendships):
    parent = [i for i in range(n)]
    rank = [0] * n
    
    for x, y in friendships:
        union(parent, rank, x - 1, y - 1)
    
    min_cost = [float('inf')] * n
    for i in range(n):
        root = find_parent(parent, i)
        min_cost[root] = min(min_cost[root], c[i])
    
    return sum([cost for cost in min_cost if cost != float('inf')])

n, m = map(int, input().split())
c = list(map(int, input().split()))
friendships = [list(map(int, input().split())) for _ in range(m)]

print(minimum_gold(n, m, c, friendships))
