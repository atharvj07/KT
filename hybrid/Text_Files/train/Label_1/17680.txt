#include <cstdio>
#include <cstring>
#define ll long long
using namespace std;
const ll MAX=505;
ll h, w, i, j;
char a[MAX][MAX];
bool f1, f2, flag;

int main(){
  scanf("%lld %lld",&h,&w);
  for(i=1;i<=h;i++)
    scanf("%s",a[i]+1);
  for(i=1;i<=h;i++){
    for(j=1;j<=w;j++){
      if(a[i][j]=='*'){
        if(f1 || f2) flag=1;
        f1=f2=1;
      }
      if(a[i][j]=='.'){
        f1=f2=0;
      }
    }
  }
  f1=f2=0;
  for(j=1;j<=w;j++){
    for(i=1;i<=h;i++){
      if(a[i][j]=='*'){
        if(f1 || f2) flag=1;
        f1=f2=1;
      }
      if(a[i][j]=='.'){
        f1=f2=0;
      }
    }
  }
  if(flag) printf("NO");
  else printf("YES");
	return 0;
}
