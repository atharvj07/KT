#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<string> s(d);
  for (int i = 0; i < d; i++) {
    cin >> s[i];
  }
  int ans = 0;
  for (int i = 0; i < d; i++) {
    int cnt = 0;
    for (int j = 0; j < n; j++) {
      if (s[i][j] == '0') cnt++;
    }
    if (cnt > 0) ans++;
    else
      break;
  }
  cout << ans << endl;
  return 0;
}
