from math import sqrt, ceil

n, s = list(map(int, input().split()))
circles = [(0, 0, sqrt(s))]
for i in range(n):
    x, y, k = list(map(int, input().split()))
    dist = sqrt(x**2 + y**2)
    radius = (sqrt((k + s - 1) / k) * dist if k + s > 0 else float('inf'))
    circles.append((x, y, radius))
circles.sort(key=lambda x: x[2])

ans = circles[0][2]
for i in range(1, len(circles)):
    x1, y1, r1 = circles[i - 1]
    x2, y2, r2 = circles[i]
    if (x1 - x2)**2 + (y1 - y2)**2 <= (r1 + r2)**2:
        ans = max(ans, (r1 + r2) / 2)
    else:
        ans = max(ans, r2)

if ans == float('inf'):
    print(-1)
else:
    print(ceil(ans * 1000000) / 1000000)
