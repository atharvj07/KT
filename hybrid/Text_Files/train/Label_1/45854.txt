import java.util.Scanner;
import java.util.TreeSet;
import java.util.SortedSet;
import java.util.Iterator;
import java.util.Map;
import java.util.HashMap;
import java.util.List;
import java.util.ArrayList;
import java.util.logging.Level;
import java.util.logging.Logger;

public class Main {
    private static final int MAX_SQUARE = 1000000000;
    private static final int MIN_COORD = -1000000000;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        if (n > 100000) {
            System.out.println(-1);
            return;
        }
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            int x = input.nextInt();
            int y = input.nextInt();
            points[i] = new Point(x, y);
        }
        TreeSet<Point> goodPoints = new TreeSet<Point>();
        for (int i = 0; i < n; i++) {
            goodPoints.add(points[i]);
            boolean xTaken = points[i].x == points[0].x;
            boolean yTaken = points[i].y == points[0].y;
            if (xTaken && yTaken)
                continue;
            Point toAdd = null;
            if (!xTaken)
                toAdd = points[i].minus(points[0]);
            if (toAdd != null && goodPoints.add(toAdd))
                continue;
            toAdd = points[0].minus(points[i]);
            if (goodPoints.add(toAdd))
                continue;
            System.out.println(-1);
            return;
        }
        SortedSet<Point> bridge = new TreeSet<Point>();
        for (Iterator<Point> i = goodPoints.iterator(); i.hasNext();) {
            Point p = i.next();
            if (p.x * p.x + p.y * p.y > MAX_SQUARE) {
                i.remove();
                bridge.add(p);
            }
        }
        SortedSet<Point> pointsToAdd = new TreeSet<Point>();
        for (Point p : bridge) {
            pointsToAdd.add(p.minus(points[0]));
            pointsToAdd.add(points[0].minus(p));
        }
        List<Point> pointsToAddList = new ArrayList<Point>(pointsToAdd);
        goodPoints.addAll(pointsToAddList);
        System.out.println(goodPoints.size());
        for (Point p : goodPoints)
            System.out.println(p.x + " " + p.y);
    }
}

class Point implements Comparable<Point> {
    public int x, y;

    public Point(int x, int y) {
        this.x = x;
        this.y = y;
    }

    public Point minus(Point o) {
        return new Point(x - o.x, y - o.y);
    }

    @Override
    public int compareTo(Point o) {
        int ans = x - o.x;
        if (ans != 0)
            return ans;
        return y - o.y;
    }
}
