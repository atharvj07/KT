#include <bits/stdc++.h>
#define pb push_back
using namespace std;
vector<int> a;
vector<long long> b;
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a.pb(x);
  }
  for (int i = 0; i < n; i++) {
    if (i != n - 1) {
      if (a[i] == 1)
        b.pb(2);
      else if (a[i] == 2)
        b.pb(1);
      else if (a[i] == 3)
        b.pb(4);
      else if (a[i] == 4)
        b.pb(3);
      else if (a[i] == 5)
        b.pb(6);
      else if (a[i] == 6)
        b.pb(5);
      else if (a[i] == 7)
        b.pb(8);
      else if (a[i] == 8)
        b.pb(7);
      else if (a[i] == 9)
        b.pb(10);
      else if (a[i] == 10)
        b.pb(9);
      else
        b.pb(a[i]);
    } else {
      b.pb(a[i]);
    }
  }
  for (int i = 0; i < n; i++) {
    cout << b[i] << " ";
  }
}
