#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, ans = 0, cnt = 0;
    cin >> n;
    while (n--) {
        char c;
        int x;
        cin >> c >> x;
        if (c == '+')
            cnt++;
        else
            cnt--;
        ans = max(ans, cnt);
    }
    cout << ans << endl;
}
