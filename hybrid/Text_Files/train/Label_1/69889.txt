n, f = map(int, input().split())
days = []
for _ in range(n):
    a, b = map(int, input().split())
    days.append([a, b, a])
days.sort(key=lambda x: x[1])
while f and days:
    f -= 1
    idx = days[-1][0] - 1
    days[-1][0] = 2 * idx
    days.sort(key=lambda x: x[1])
ans = 0
for x, y, z in days:
    if x <= y:
        ans += x
    else:
        ans += y
print(ans)
