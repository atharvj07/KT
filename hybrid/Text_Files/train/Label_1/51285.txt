import java.util.*;
import java.lang.*;

class Main {
    static int N, M;
    static int[] m;
    static List<Integer>[] adj;
    static int[] ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        m = new int[M];
        adj = new List[N + 1];
        for (int i = 0; i <= N; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < M; i++) {
            m[i] = sc.nextInt();
            adj[m[i] - 1].add(m[i]);
            adj[m[i]].add(m[i] - 1);
        }
        int Q = sc.nextInt();
        ans = new int[Q];
        for (int i = 0; i < Q; i++) {
            int l = sc.nextInt();
            ans[i] = solve(l);
        }
        for (int i = 0; i < Q; i++) {
            System.out.println(ans[i]);
        }
    }

    static int solve(int l) {
        int[] vis = new int[N + 1];
        int cnt = 0;
        for (int i = 0; i < N; i++) {
            if (vis[i] == 0) {
                if (dfs(i, i, l, vis) == true) {
                    cnt++;
                }
            }
        }
        if (cnt <= l) {
            return 2;
        } else {
            return -1;
        }
    }

    static boolean dfs(int u, int p, int l, int[] vis) {
        vis[u] = 1;
        boolean flag = false;
        for (int v : adj[u]) {
            if (v == p) {
                continue;
            }
            if (vis[v] == 0) {
                flag |= dfs(v, u, l, vis);
            }
        }
        if (flag == true) {
            return false;
        } else {
            return true;
        }
    }
}
