from collections import Counter

n, m = map(int, input().split())
A = [list(map(int, input().split())) for i in range(n)]
isPrime = [1] * 100001

def sieve():
    isPrime[0] = isPrime[1] = 0
    for i in range(2, 100001):
        for j in range(i * i, 100001, i):
            isPrime[j] = 0

sieve()
ans = 1000000000
for i in range(n):
    cnt = Counter(A[i])
    if cnt[1] == m:
        ans = 0
    else:
        for num in cnt:
            if isPrime[num + 1]:
                ans = min(ans, cnt[num])
for i in range(m):
    cnt = Counter()
    for j in range(n):
        cnt[A[j][i]] += 1
    if cnt[1] == n:
        ans = 0
    else:
        for num in cnt:
            if isPrime[num + 1]:
                ans = min(ans, cnt[num])
print(ans)
