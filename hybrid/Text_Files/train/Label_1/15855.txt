#include <bits/stdc++.h>
using namespace std;
int main() {
    int x, k;
    cin >> x >> k;
    vector<pair<int, int>> v(k);
    for (int i = 0; i < k; i++) {
        int t, y;
        cin >> t >> y;
        if (t == 1) v[i] = {y, y - 1};
        if (t == 2) v[i] = {y, y};
    }
    sort(v.begin(), v.end());
    int cur = 2, ans = 0;
    for (int i = 0; i < k; i++) {
        if (v[i].first == cur) cur++;
    }
    cur = 2;
    for (int i = 0; i < k; i++) {
        if (v[i].second == cur) cur++;
        else ans++;
    }
    cout << ans << " " << x - (k + ans) - 1 << endl;
}
