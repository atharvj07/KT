#include<bits/stdc++.h>
using namespace std;
#define ll long long int
#define pb push_back
#define mk make_pair

ll n, m, par[100001], sz[100001];    //DSU related variables

ll findpar(ll x) {
    if(par[x] == x) return x;
    return par[x] = findpar(par[x]);
}

void connect(ll x, ll y) {
    ll u = findpar(x), v = findpar(y);
    if(u == v) return;
    if(sz[u] < sz[v]) swap(u, v);
    sz[u] += sz[v];
    par[v] = u;
}

int main() {
    cin >> n >> m;
    for(int i=1; i<=n; i++) par[i] = i, sz[i] = 1;
    for(int i=0; i<m; i++) {
        int x, y;
        cin >> x >> y;
        connect(x, y);
    }

    ll ans = 0;
    for(int i=1; i<=n; i++) {
        if(par[i] == i) ans += sz[i]-1;
    }

    cout << ans << endl;
    return 0;
}
