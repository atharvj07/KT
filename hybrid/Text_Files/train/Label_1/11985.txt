#include<iostream>
#include<string>
using namespace std;
int main() {
  int n, c=0;
  cin>>n;
  cout<<n*n<<endl;
  while (c<n) {
    for (int i=0; i<n; i++) {
      if (i%2==0) {
        for (int j=0; j<n; j++) {
          if (j%2==0) cout<<"C";
          else cout<<".";
        }
      }
      else {
        for (int j=0; j<n; j++) {
          if (j%2==0) cout<<".";
          else cout<<"C";
        }
      }
    }
    c++;
    cout<<endl;
  }
  return 0;
}
