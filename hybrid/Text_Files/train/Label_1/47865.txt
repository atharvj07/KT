import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] a;
    static int[] ans;
    static List<Integer>[] edges;
    static int[] size;
    static int[] cnt;
    static int mx;

    static void dfs(int u, int p) {
        size[u] = 1;
        cnt[u] = a[u];
        for (int v : edges[u]) {
            if (v == p)
                continue;
            dfs(v, u);
            size[u] += size[v];
            cnt[u] += cnt[v];
            ans[u] = Math.max(ans[u], ans[v]);
        }
        // cnt[u] -= a[u];
        ans[u] = Math.max(ans[u], n - size[u] - cnt[u] + a[u]);
        ans[u] = Math.max(ans[u], cnt[u] - (n - size[u]));
        mx = Math.max(mx, ans[u]);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        a = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        edges = new List[n];
        for (int i = 0; i < n; i++)
            edges[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            edges[u].add(v);
            edges[v].add(u);
        }
        size = new int[n];
        cnt = new int[n];
        ans = new int[n];
        dfs(0, -1);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
