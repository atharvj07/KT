import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        ArrayList<Integer> l = new ArrayList<>();
        l.add(0);
        int[] ll = new int[n + 1];
        int[] rr = new int[n + 1];
        int[][] ans = new int[n + 1][2];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
            ll[a[i]] = i;
            rr[a[i]] = i;
        }
        for (int i = 1; i <= n; i++) {
            if (rr[i] != 0) {
                l.add(rr[i]);
            }
            if (ll[i] != 0) {
                l.add(ll[i]);
            }
        }
        Collections.sort(l);
        for (int i = 0; i < n; i++) {
            int x = a[i];
            int p1 = ll[x];
            int p2 = rr[x];
            int l1 = l.indexOf(p1);
            int r1 = l.indexOf(p2);
            int m1 = (l1 + r1 + 1) / 2;
            int mid = l.get(m1);
            ans[i][0] = Math.max(mid - p1, p2 - mid);
            int l2 = r1;
            int r2 = l.size() - 1;
            m1 = (l2 + r2 + 1) / 2;
            while (l2 < r2) {
                mid = l.get(m1);
                if (mid < p2) {
                    l2 = m1 + 1;
                } else {
                    r2 = m1 - 1;
                }
                m1 = (l2 + r2 + 1) / 2;
            }
            mid = l.get(m1);
            if (p2 != n && mid == p2) {
                l2++;
                mid = l.get(m1);
            }
            ans[i][1] = mid - p2;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(Math.max(ans[i][0], ans[i][1]) + " ");
        }
        System.out.println();
    }
}
