import sys

read = sys.stdin.readline

n, m = map(int, read().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, read().split())
    g[a].append(b)
    g[b].append(a)

q = int(read())
for _ in range(q):
    t, *query = map(int, read().split())
    if t == 1:
        # Add
        a, b = query
        g[a].append(b)
        g[b].append(a)
    elif t == 2:
        # Remove
        a, b = query
        g[a].remove(b)
        g[b].remove(a)
    elif t == 3:
        # Process
        vis = [False] * (n + 1)
        kill = []
        for i in range(1, n + 1):
            if g[i] and max(g[i]) > i:
                kill.append(i)
                vis[i] = True
                for j in g[i]:
                    vis[j] = True

        while kill:
            nxt = []
            for i in kill:
                for j in g[i]:
                    if not vis[j]:
                        vis[j] = True
                        nxt.append(j)
            kill = nxt
        cnt = 0
        for i in range(1, n + 1):
            if not vis[i]:
                cnt += 1
        print(cnt)
