#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, x1, x2;
    cin >> n >> x1 >> x2;
    vector<tuple<int, int>> lines(n);
    for (int i = 0; i < n; i++) {
        int k, b;
        cin >> k >> b;
        lines[i] = {k, b};
    }
    bool found = 0;
    for (int i = 0; i < n; i++) {
        int k1, b1;
        tie(k1, b1) = lines[i];
        for (int j = i + 1; j < n; j++) {
            int k2, b2;
            tie(k2, b2) = lines[j];
            if (k1 == k2) continue;
            int x = (b2 - b1) / (k1 - k2);
            if (x < x1 || x > x2) continue;
            cout << "YES" << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
    return 0;
}
