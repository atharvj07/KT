#include <bits/stdc++.h>
using namespace std;

const int N = 3e5 + 10;
const long long INF = 1e18;
int n;
long long a[N], red[N], blu[N], dis[N];
vector<pair<long long, long long>> adj[N];
priority_queue<pair<long long, long long>, vector<pair<long long, long long>>, greater<pair<long long, long long>>> pq;

void dijkstra(long long source, long long skip[N], long long d[N]) {
	pq.push(make_pair(0, source));
	for(long long i = 1; i <= n; i++) {
		d[i] = INF;
	}
	d[source] = 0;
	while(!pq.empty()) {
		auto u = pq.top();
		pq.pop();
		if(source == u.second && skip[u.second]) continue;
		for(auto x : adj[u.second]) {
			long long v = x.first, w = x.second;
			if(d[v] > d[u.second] + w) {
				d[v] = d[u.second] + w;
				pq.push({d[v], v});
			}
		}
	}
}

int main() {
	cin >> n;
	for(long long i = 1; i <= n; i++) {
		cin >> a[i];
	}
	char ch;
	for(long long i = 1; i <= n; i++) {
		cin >> ch;
		if(ch == 'R') {
			red[i] = 1;
		} else if(ch == 'B') {
			blu[i] = 1;
		}
	}
	for(long long i = 1; i < n; i++) {
		adj[i].push_back(make_pair(i + 1, a[i + 1] - a[i]));
		adj[i + 1].push_back(make_pair(i, a[i + 1] - a[i]));
	}
	dijkstra(1, red, dis);
	long long answer = 0;
	for(long long i = 1; i <= n; i++) {
		if(blu[i] == 0) {
			answer += dis[i];
		}
	}
	dijkstra(1, blu, dis);
	for(long long i = 1; i <= n; i++) {
		if(red[i] == 0) {
			answer += dis[i];
		}
	}
	cout << answer << endl;
}
