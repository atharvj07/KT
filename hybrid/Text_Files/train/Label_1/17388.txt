#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, x, y;
    cin >> n >> m >> x >> y;
    vector<string> square(n);
    int cnt = 0;
    for (string &s : square) {
      cin >> s;
      for (char c : s) { cnt += c == '.'; }
    }
    cout << (cnt % 2 ? cnt / 2 * y + x : min(cnt / 2 * y, cnt * x)) << '\n';
  }
  return 0;
}
