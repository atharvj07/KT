import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int w = scanner.nextInt();
        int d = scanner.nextInt();
        int n = scanner.nextInt();

        int[][] altitudes = new int[w][d];

        // Read measured altitudes.
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            int z = scanner.nextInt();
            altitudes[x][y] = z;
        }

        // Initialize the minimum altitude total to Integer.MAX_VALUE.
        int minTotal = Integer.MAX_VALUE;

        // Try all possible altitude assignments for unmeasured areas.
        for (int i = 0; i < (1 << (w * d - n)); i++) {
            int total = 0;
            boolean valid = true;

            for (int x = 0; x < w; x++) {
                for (int y = 0; y < d; y++) {
                    if (altitudes[x][y] != 0) {
                        // Measured altitude.
                        total += altitudes[x][y];
                    } else {
                        // Unmeasured altitude.
                        int altitude = (i >> (x * d + y)) & 1;
                        total += altitude;

                        // Check altitude differences with neighboring areas.
                        if (x > 0 && Math.abs(altitude - altitudes[x - 1][y]) > 1) {
                            valid = false;
                        }
                        if (y > 0 && Math.abs(altitude - altitudes[x][y - 1]) > 1) {
                            valid = false;
                        }
                    }
                }
            }

            // Update the minimum altitude total if the assignment is valid.
            if (valid && total < minTotal) {
                minTotal = total;
            }
        }

        // Output the result.
        if (minTotal != Integer.MAX_VALUE) {
            System.out.println(minTotal);
        } else {
            System.out.println("No");
        }
    }
}
