import sys
input = sys.stdin.readline

n,k = map(int,input().split())
e = [0]*n
for _ in range(n-1):
    x,y = map(int,input().split())
    e[x-1]+=1
    e[y-1]+=1

e.sort(reverse=1)
a = []
ans = 0
for i in e:
    if i==1: break
    if len(a)>k: ans+=1; a.clear()
    a.append(i)
if len(a)!=0: ans+=1
print(ans)
for i in e[n-len(a)-1:]:
    if i==1: break
    for j in range(i): print(j+1,end=' ')
    print()
