#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, k;
        cin >> a >> b >> k;
        vector<pair<int, int>> v(k);
        for (int i = 0; i < k; i++) { cin >> v[i].first; }
        for (int i = 0; i < k; i++) { cin >> v[i].second; }
        long long ans = 0;
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                if (v[i].first != v[j].first && v[i].second != v[j].second) ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
