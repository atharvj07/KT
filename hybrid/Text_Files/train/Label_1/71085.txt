s=input()
l=len(s)
a=['('for i in range(l)]
b=['('for i in range(l)]
q=[0 for i in range(l)]
for i in range(l):
    if s[i]=='(':
        a[i]='('
        b[i]='('
    elif s[i]==')':
        a[i]=')'
        if i>0:
            b[i]=b[i-1]
        else:
            b[i]=')'
    else:
        q[i]=1
        if i>0:
            a[i]=b[i-1]
        else:
            a[i]='('
        b[i]=')'
if b[l-1]=='(':
    for i in range(l-1,-1,-1):
        if not q[i]:
            b[i]=')'
            break
ans=''.join(a)+''.join(b)
if ans.count('(')==ans.count(')'):
    print(ans)
else:
    print(':(')
