#include<iostream>
#include<vector>
#include<queue>
#include<cstring>
using namespace std;
int main()
{
    int n,a,b,i;
    cin>>n;
    vector<int> pre(n+1),nxt(n+1);
    for(i=0;i<n;++i)
    {
        cin>>a>>b;
        pre[i+1]=a;
        nxt[i+1]=b;
    }
    vector<bool> vis(n+1,false);
    priority_queue<int,vector<int>,greater<int>> q;
    for(i=1;i<=n;++i)
        if(!pre[i])
            q.push(i);
    while(!q.empty())
    {
        a=q.top();
        q.pop();
        if(vis[a])
            continue;
        vis[a]=true;
        cout<<a<<" ";
        if(nxt[a])
            q.push(nxt[a]);
    }
    cout<<endl;
    return 0;
}
