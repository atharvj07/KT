#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
struct data{
    int a, b, p;
    bool operator<(const data &a) const{
        return a.a > a.b;
    }
}v[MAXN];
int main() {
    int n, t, ans = 0;
    scanf("%d", &n);
    for(int i = 1; i <= n; i++){
        scanf("%d %d", &v[i].a, &v[i].b);
        v[i].p = i;
    }
    sort(v + 1, v + 1 + n);
    for(int i = 1; i <= n; i++){
        ans += (i - 1) * v[i].a + (n - i) * v[i].b;
    }
    printf("%d\n", ans);
    return 0;
}
