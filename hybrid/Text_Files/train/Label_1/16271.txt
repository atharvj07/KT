#include <bits/stdc++.h>
using namespace std;

struct event {
    long long x;
    int y, id;
    bool open;
    event(long long x, int y, bool open, int id) : x(x), y(y), open(open), id(id) {}
    bool operator<(const event& e) const {
        if (x != e.x) return x < e.x;
        return open < e.open;
    }
};

int main() {
    int n;
    cin >> n;
    vector<event> events;
    vector<long long> ans(n + 1);
    for (int i = 0; i < n; i++) {
        long long l, r;
        cin >> l >> r;
        events.push_back(event(l, i, true, i));
        events.push_back(event(r + 1, i, false, i));
    }
    sort(events.begin(), events.end());
    set<int> active;
    for (auto& e : events) {
        if (e.open) {
            active.insert(e.id);
        } else {
            active.erase(e.id);
        }
        ans[active.size()]++;
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
    cout << endl;
    return 0;
}
