t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    v = list(map(int, input().split()))
    t = list(map(int, input().split()))
    
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    
    # Calculate the difference between current and target values for each node
    diff = [tv - vv for vv, tv in zip(v, t)]
    
    if sum(diff) % 2 != 0:
        print("NO")
        continue
    
    # If the graph is a tree (n-1 edges), it's always possible
    if m == n - 1:
        print("YES")
        continue
    
    # For non-tree graphs, we need to check if there's an odd cycle
    # Create adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)
    
    visited = [False] * n
    has_odd_cycle = False
    
    def dfs(node, parent, depth):
        nonlocal has_odd_cycle
        visited[node] = True
        for neighbour in adj[node]:
            if not visited[neighbour]:
                dfs(neighbour, node, depth + 1)
            elif neighbour != parent and depth % 2 == 0:
                has_odd_cycle = True
    
    # Check all components (although there should be just one since the graph is connected)
    for i in range(n):
        if not visited[i]:
            dfs(i, -1, 0)
    
    if has_odd_cycle:
        # If there's an odd cycle, any configuration is possible
        print("YES")
    else:
        # For even cycles, check if the sum of differences for any cycle is 0
        # This is equivalent to checking if the sum of all differences is 0
        # Because in an even cycle or a tree, we can always balance changes across the cycle
        if sum(diff) == 0:
            print("YES")
        else:
            print("NO")
