import math
y1, y2, yw, xb, yb, r = map(int, input().split())
if yb <= 0 or yb + r >= yw: print('-1'); exit()
p = (y2-yb)/(yb-y1)
a = r*(math.pow(p,2)-1)
b = -2*xb*(math.pow(p,2)-1)
c = math.pow(xb,2)*(math.pow(p,2)-1) + math.pow(yw,2)
D = b*b - 4*a*c
if D<0: print('-1')
else:
    x1 = (-b - math.sqrt(D))/(2*a)
    x2 = (-b + math.sqrt(D))/(2*a)
    if (1e-8 < x1 < xb) or (1e-8 < x2 < xb): print('-1')
    else:
        ans = x1 if x2<0 else x2
        print(ans)
