n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
q = int(input())

for _ in range(q):
    query = list(map(int, input().split()))
    if query[0] == 1:
        a[query[1]-1] = query[2]
    else:
        b[query[1]-1] = query[2]
        
    available_dishes = sorted(a, reverse=True)
    pupils_money = sorted(b, reverse=True)
    
    dish_pointer, pupil_pointer = 0, 0
    while dish_pointer < len(available_dishes) and pupil_pointer < len(pupils_money):
        if available_dishes[dish_pointer] <= pupils_money[pupil_pointer]:
            pupil_pointer += 1
            dish_pointer += 1
        else:
            dish_pointer += 1
    
    if dish_pointer == len(available_dishes):
        print(-1)
    else:
        print(available_dishes[dish_pointer])
