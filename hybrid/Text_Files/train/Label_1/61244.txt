from sys import stdin
input = stdin.readline

t = int(input())
while t > 0:
    n = int(input())
    l = list(map(int, input().split()))
    if len(set(l)) == 1: print("YES")
    elif len(set(l)) == 2 and (l[0] != l[-1] or l.count(l[0]) == 1 or l.count(l[-1]) == 1):
        print("NO")
    else: print("YES")
    t -= 1
