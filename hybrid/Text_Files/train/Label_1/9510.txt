#include <iostream>

int main() {
  int n, k;
  std::cin >> n >> k;
  long long a[200000], b[200000];
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
    b[i] = k / a[i];
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) ans = std::min(ans, b[i]);
  if (ans == 0) std::cout << ans;
  if (ans != 0) std::cout << ans << std::endl;
  return 0;
}
