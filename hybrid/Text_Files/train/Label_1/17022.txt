#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  int w[2 * n];
  for (int i = 0; i < 2 * n; ++i)
    cin >> w[i];
  sort(w, w + 2 * n);
  int res = 0;
  for (int i = 0; i < n - 1; ++i)
    res += w[2 * i + 1] - w[2 * i];
  cout << res << endl;
  return 0;
}
