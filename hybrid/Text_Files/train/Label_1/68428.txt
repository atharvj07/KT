import sys
input = sys.stdin.readline

n, m = map(int, input().split())

names = []
for i in range(n):
    names.append(input().strip())

deleted = [names[i - 1] for i in map(int, input().split())]

pattern = names[0]
for name in names:
    for i in range(min(len(name), len(pattern))):
        if name[i] != pattern[i] and pattern[i] != '?':
            pattern = pattern[:i] + '?' + pattern[i + 1:]

if pattern in deleted:
    found = False
    for name in names:
        if name != pattern and pattern in name:
            found = True
            break
    if found:
        print("No")
    else:
        print("Yes")
        print(pattern)
else:
    print("No")
