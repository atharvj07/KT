#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<string> v(2 * n - 2);
    for (string& x : v) cin >> x;

    sort(v.begin(), v.end());
    for (int len = 1; len < n; len++) {
        for (auto& x : v) if (x.size() == len)
            cout << ((x == v[len - 1] ? 'S' : 'P'));
    }
    cout << endl;
}
