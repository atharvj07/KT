#include <bits/stdc++.h>
typedef long long ll;
const ll mod = 998244353;
const ll p = 1000000007;
const int maxn = 200010;
ll powp[maxn];
ll invp[maxn];
using namespace std;
 
ll bpow(ll base, ll e) {
    if (e == 0) return 1;
    if (e == 1) return base;
    ll ret = bpow(base, e / 2);
    ret = (ret * ret) % mod;
    if (e & 1) ret = (ret * base) % mod;
    return ret;
}
 
int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    powp[0] = 1;
    for (int i = 1; i <= n; i++) powp[i] = (powp[i - 1] * p) % mod;
    invp[n] = bpow(powp[n], mod - 2);
    for (int i = n - 1; i >= 0; i--) invp[i] = (invp[i + 1] * p) % mod;
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        ll cur = (n - a[i] + (a[i] <= i));
        cur = (cur * invp[i]) % mod;
        ans = (ans + (cur * powp[i])) % mod;
    }
    cout << ans << endl;
    return 0;
}
