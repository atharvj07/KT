import sys
input = sys.stdin.readline

s = input().strip()
t = input().strip()

n, m = len(s), len(t)
pos = []
for i in range(n-m+1):
    if s[i:i+m] == t:
        pos.append(i)

mod = 10**9 + 7

# dp[i][j] = number of ways to extract j non-overlapping substrings from s[i] such that each of them contains string t as a substring
dp = [[0] * (len(pos) + 1) for _ in range(n + 1)]
dp[n][0] = 1
for i in range(n-1,-1,-1):
    for j in range(len(pos)):
        dp[i][j] = (dp[i+1][j] + (dp[pos[j]][j-1] if i <= pos[j] else 0)) % mod

print(dp[0][len(pos)-1])
