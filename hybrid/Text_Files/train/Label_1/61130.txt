import sys

t = int(input())

for _ in range(t):
    n, k = map(int, input().split())

    if k > n * (n+1) // 2:
        print(-1)
    else:
        p = [i for i in range(1, n+1)]
        k -= 1
        for j in range(n-1, 0, -1):
            i = j - 1
            while i >= 0 and p[i] < p[j]:
                i -= 1
            i += 1
            if i == j:
                continue
            while i < j:
                p[i], p[j] = p[j], p[i]
                k -= 1
                i += 1
                j -= 1
            if k == 0:
                break
        print(*p)
