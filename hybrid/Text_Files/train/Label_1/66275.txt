import sys
input = sys.stdin.readline
n, k, d = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
i, c = 0, 0
while i < n:
    j, cc = i+1, a[i]
    while j < n and a[i] - d <= a[j] <= a[i] + d:
        j, cc = j+1, cc+1
    if cc < k:
        break
    c, i = c+1, j
if c < n:
    print("NO")
else:
    print("YES")
