t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    total_dominoes = n * m // 2
    vertical_dominoes_possible = (n // 2) * m
    horizontal_dominoes_possible = (m // 2) * n
    
    if k <= horizontal_dominoes_possible and (total_dominoes - k) % 2 == 0 and (total_dominoes - k) <= vertical_dominoes_possible:
        print("YES")
    else:
        print("NO")
