#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    unordered_map<int, int> freq;
    for (int x : a) freq[x]++;
    for (int x : b) freq[x]++;
    int odd = 0;
    for (auto [k, v] : freq) odd += v % 2;
    if (odd % 2) cout << "Koyomi";
    else cout << "Karen";
    return 0;
}
