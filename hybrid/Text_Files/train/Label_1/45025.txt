import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int q = sc.nextInt();

        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }

        long[][] queries = new long[q][2];
        for (int i = 0; i < q; i++) {
            queries[i][0] = sc.nextLong();
            queries[i][1] = sc.nextLong();
        }

        long[] ans = new long[q];
        for (int i = 0; i < q; i++) {
            long l = queries[i][0];
            long r = queries[i][1];

            int cnt = 0;
            for (long x : a) {
                if (l <= x && x <= r) {
                    cnt++;
                }
            }

            ans[i] = cnt;
        }

        for (long x : ans) {
            System.out.println(x);
        }
    }
}
