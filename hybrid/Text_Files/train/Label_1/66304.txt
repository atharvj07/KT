def can_be_obtained(p, h):
    p_count = [0] * 26
    h_count = [0] * 26
    for char in p:
        p_count[ord(char) - ord('a')] += 1
    len_p = len(p)
    len_h = len(h)
    for i in range(len_h):
        h_count[ord(h[i]) - ord('a')] += 1
        if i >= len_p:
            h_count[ord(h[i-len_p]) - ord('a')] -= 1
        if p_count == h_count:
            return "YES"
    return "NO"

t = int(input().strip())
for _ in range(t):
    p = input().strip()
    h = input().strip()
    print(can_be_obtained(p, h))
