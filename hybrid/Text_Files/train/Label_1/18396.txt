#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef pair<int, int> pi;

const int MAXN = 1e6 + 5;

int n, m;
int a[MAXN], dp[MAXN];
int p[MAXN], pos[MAXN];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    dp[0] = 1;
    int ans = -1;
    for (int i = 1; i <= n; i++) {
        dp[i] = dp[i - 1] * a[i];
        if (dp[i] > m) {
            dp[i] = 0;
            p[i] = i - 2;
            pos[i] = i;
        } else {
            p[i] = i - 1;
            pos[i] = i;
        }
        if (dp[i] >= ans) {
            ans = dp[i];
        }
    }
    vector<int> v;
    int cur = n;
    while (cur > 0) {
        if (dp[cur] == ans) {
            v.push_back(pos[cur]);
            cur = p[cur];
        } else {
            cur--;
        }
    }
    cout << ans << " " << v.size() << "\n";
    for (auto &u : v) {
        cout << u << " ";
    }
    cout << "\n";
    return 0;
}
