#include <algorithm>
#include <cstddef>
#include <iostream>
#include <utility>
#include <vector>

int main() {
  int T;
  std::cin >> T;
  for (int C = 0; C < T; ++C) {
    int N, M;
    std::cin >> N >> M;
    std::vector<std::vector<int>> Vec(N + 1);
    for (int i = 1; i <= N; ++i) {
      Vec[i].resize(M);
      for (auto& X : Vec[i]) std::cin >> X;
      std::sort(Vec[i].begin(), Vec[i].end());
    }
    for (int i = 1; i <= N; ++i) {
      for (auto X : Vec[i]) std::cout << X << ' ';
      std::cout << '\n';
    }
  }
  return 0;
}
