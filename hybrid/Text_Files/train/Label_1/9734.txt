#include<cstdio>
bool done[55][55];
bool check(int n, int m){
    bool f=1;
    for(int i=1; i<=n; ++i){
        f=1;
        for(int j=1; j<=m; ++j){
            if(done[i][j]){
                f=0; break;
            }
        }
        if(f) return 0;
    }
    for(int j=1; j<=m; ++j){
        f=1;
        for(int i=1; i<=n; ++i){
            if(done[i][j]){
                f=0; break;
            }
        }
        if(f) return 0;
    }
    return 1;
}
int main(){
    int n,m; scanf("%d%d",&n,&m);
    for(int i=1; i<=n; ++i){
        char s[55]; scanf("%s",s+1);
        for(int j=1; j<=m; ++j){
            if(s[j]=='#') done[i][j]=1;
        }
    }
    puts(check(n,m)?"Yes":"No");
}
