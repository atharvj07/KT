#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    bool ok = true;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1' && ((i == 0 || s[i - 1] == '0') && (i == n - 1 || s[i + 1] == '0'))) continue;
        ok = false;
        break;
    }
    cout << (ok ? "Yes" : "No") << endl;
    return 0;
}
