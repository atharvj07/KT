import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            char[] chr = sc.nextLine().toCharArray();
            int maxFreq = 0;
            int rot = 0;
            for (int i = 0; i < 26; i++) {
                int freq = 0;
                for (char c : chr) {
                    if (c >= 'a' && c <= 'z') {
                        char d = (char) ((c - i + 26) % 26 + 'a');
                        if (d == 't' || d == 'h' || d == 'i' || d == 's') freq++;
                    }
                }
                if (freq > maxFreq) {
                    maxFreq = freq;
                    rot = i;
                }
            }
            for (char c : chr) {
                if (c >= 'a' && c <= 'z') {
                    char d = (char) ((c - rot + 26) % 26 + 'a');
                    System.out.print(d);
                } else System.out.print(c);
            }
            System.out.println();
        }
    }
}
