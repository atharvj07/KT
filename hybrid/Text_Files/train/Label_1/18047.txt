#include <bits/stdc++.h>

using namespace std;

struct Point {
    int x, y;
    double slope;
};

bool operator < (const Point &a, const Point &b) {
    return a.slope < b.slope;
}

int main() {
    int R, B;
    cin >> R >> B;
    vector <Point> RPoint(R), BPoint(B);
    for (int i = 0; i < R; i++) {
        cin >> RPoint[i].x >> RPoint[i].y;
    }
    for (int i = 0; i < B; i++) {
        cin >> BPoint[i].x >> BPoint[i].y;
    }
    for (int i = 0; i < R; i++) {
        for (int j = 0; j < B; j++) {
            double dx = RPoint[i].x - BPoint[j].x;
            double dy = RPoint[i].y - BPoint[j].y;
            if (dx) {
                RPoint[i].slope = dy / dx;
            } else {
                RPoint[i].slope = dy * INFINITY;
            }
        }
    }
    sort(RPoint.begin(), RPoint.end());
    bool ok = true;
    for (int i = 0; i < R - 1; i++) {
        if (RPoint[i].slope == RPoint[i + 1].slope) {
            ok = false;
            break;
        }
    }
    cout << (ok ? "Yes" : "No") << endl;
    return 0;
}
