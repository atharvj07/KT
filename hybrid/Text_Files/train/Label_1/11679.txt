#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, h, w, n, ans = INT_MAX;
  cin >> a >> b >> h >> w >> n;
  vector<int> mult(n);
  for (auto &x : mult) cin >> x;
  sort(mult.begin(), mult.end());
  if (w < b) swap(w, b);
  if (h < a) swap(h, a);
  if (w < a || h < b) ans = -1;
  if (ans != -1) ans = min(ans, (int)(ceil(1.0 * a / h) + ceil(1.0 * b / w)));
  for (int i = 0; i < n && ans != -1; ++i) {
    if (h < a && mult[i] > 1) {
      if (w * mult[i] >= a) {
        if (max(w * mult[i], h) * max(w * mult[i], h) >= b * a) ans = min(ans, i + 1);
      }
    } else if (w < b) {
      if (h * mult[i] >= b) {
        if (max(h * mult[i], w) * max(h * mult[i], w) >= b * a) ans = min(ans, i + 1);
      }
    }
    if (h >= a && w >= b) break;
  }
  cout << (ans == INT_MAX ? -1 : ans) << endl;
  return 0;
}
