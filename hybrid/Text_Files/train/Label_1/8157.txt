#include<bits/stdc++.h>
#define vi vector<int>
using namespace std;
int32_t main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        vi p(n),c(n),vis(n,0);
        for(int &x:p)cin>>x;
        for(int &x:c)cin>>x;
        map<int,vector<int>>mp;
        int ans=1e9;
        set<int> s;
        for(int i=0;i<n;++i)
        {
            int x=i;
            vi v;
            while(x!=i||v.empty())
            {
                if(vis[x])
                {
                    ans=min(ans,int(v.size()));
                    break;
                }
                v.push_back(x);
                vis[x]=1;
                x=p[x]-1;
            }
            for(int &y:v)vis[y]=0;
            mp[c[i]].push_back(int(v.size()));
        }
        for(auto x:mp){
            s.insert(x.second[0]);
            for(int i=1;i<x.second.size();++i)
            {
                int g=__gcd(x.second[i],x.second[0]);
                s.insert(g);
            }
        }
        if(s.size()==1)ans=1e9;
        for(int x:s)ans=min(ans,x);
        cout<<(ans==1e9?-1:ans)<<endl;
    }
}
