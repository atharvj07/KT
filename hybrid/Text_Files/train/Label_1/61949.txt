import sys
rl=sys.stdin.readline

n,m=map(int,rl().split())
a=[rl().rstrip() for i in range(n)]
q=int(rl())
for x1,x2 in [map(int,rl().split()) for i in range(q)]:
  pos=[(i,j) for i in range(n) for j in range(x1-1,x2) if a[i][j]=='.' and (i==0 or a[i-1][j]=='X') and (j==0 or a[i][j-1]=='X')];
  vis=[[0]*m for i in range(n)]
  vis[n-1][x2-1]=1
  for i,j in pos:
    if vis[i][j]==1:continue
    vis[i][j]=1
    if a[i][j]=='.':
      if i>0:vis[i-1][j]=1
      if j>0:vis[i][j-1]=1
  print('YES'if vis[0][x1-1]==1 else 'NO')
