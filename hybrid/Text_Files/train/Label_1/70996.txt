import sys
input=sys.stdin.readline

for _ in range(int(input())):
    n,m=map(int,input().split())
    v=list(map(int,input().split()))
    t=list(map(int,input().split()))
    p=[[x,y] for x in range(n) for y in range(x+1,n) if x!=y]
    e=[[-1]*n for x in range(n)]
    for x,y in p:
        e[x][y]=e[y][x]=0
    for _ in range(m):
        x,y=map(lambda x:int(x)-1,input().split())
        e[x][y]=e[y][x]=1
    ok=1
    for x in range(n):
        for y in range(n):
            if e[x][y]==-1:
                ok=0
    if ok==0:
        print("NO")
        continue
    for x in range(n):
        if (t[x]-v[x])%2==1:
            ok=0
    if ok==0:
        print("NO")
    else:
        print("YES")
