import java.util.*;

public class Main {
    static int n, m;
    static char[][] map;
    static int[][] dist;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};
    static boolean[][] vis;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        map = new char[n][m];
        dist = new int[n][m];
        vis = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < m; j++) {
                map[i][j] = s.charAt(j);
            }
        }
        int ans = bfs();
        System.out.println(ans);
    }

    static int bfs() {
        Queue<int[]> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] != '.' && map[i][j] != '#') {
                    q.offer(new int[]{i, j});
                    vis[i][j] = true;
                }
            }
        }
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m || vis[nx][ny] || map[nx][ny] == '#')
                    continue;
                dist[nx][ny] = dist[x][y] + 1;
                q.offer(new int[]{nx, ny});
                vis[nx][ny] = true;
            }
        }
        int res = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] != '.' && map[i][j] != '#') {
                    res = Math.max(res, dist[i][j]);
                }
            }
        }
        return res;
    }
}
