x = int(input())
ans = 0; sum = 0
for i in str(x)[::-1]:
    if int(i) < 9:
        ans = x
        break
    elif sum + int(i) > 9 and sum == 0:
        ans = x
    elif sum + int(i) > 9:
        x -= 1
        i = str(x)[-1]
        sum = int(i)
        ans = x
    else: sum += int(i)

print(ans)
