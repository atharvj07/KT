#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, w;
    cin >> n >> w;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    vector<int> ans(n);
    int rem = w;
    for (int i = 0; i < n; i++) {
        int cur = a[i] / 2;
        if (rem >= cur) {
            rem -= cur;
            ans[i] = cur;
        } else {
            ans[i] = rem;
            rem = 0;
        }
    }
    if (rem > 0)
        cout << -1 << endl;
    else {
        for (int x : ans) cout << x << ' ';
        cout << endl;
    }
}
