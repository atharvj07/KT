#include <cstdio>
int main() {
    long long up, down, n;
    scanf("%lld %lld", &up, &down);
    n = up + down + 1;
    for (long long i = 1, j = 1; i <= down + 1; ) {
        if (up == 0 || j <= i) {
            printf("%lld ", n--);
            down--;
        } else {
            printf("%lld ", i++);
            up--;
        }
    }
    return 0;
}
