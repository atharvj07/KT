#include <vector>
#include <iostream>
#include <queue>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n);
    vector<int> degree(n, 0);
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        if (a < n - 1) {
            cout << "NO" << endl;
            return 0;
        }
        degree[i] = a;
        for (int j = 0; j < a; j++) {
            int v;
            cin >> v;
            adj[i].push_back(v - 1);
            degree[v - 1]++;
        }
    }
    int diameter = 0;
    vector<int> dist(n, -1);
    queue<int> q;
    for (int i = 0; i < n; i++) {
        if (degree[i] == 1) {
            q.push(i);
            dist[i] = 0;
        }
    }
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int u : adj[v]) {
            if (dist[u] == -1) {
                dist[u] = dist[v] + 1;
                q.push(u);
                diameter = dist[u];
            }
        }
    }
    cout << "YES" << endl;
    cout << diameter << endl;
    int m = 0;
    for (int i = 0; i < n; i++) {
        for (int j : adj[i]) {
            if (dist[i] == dist[j] - 1) {
                cout << i + 1 << " " << j + 1 << endl;
                m++;
            }
        }
    }
    cout << m << endl;
    return 0;
}
