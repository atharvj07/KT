#include <iostream>
#include <cmath>
#include <vector>
#include <algorithm>
#define endl '\n'
using namespace std;
const double eps=1e-9;
int n,m;
double X[100010],Y[100010],D[100010];
typedef pair<double,double> pt;
pt operator+(pt a,pt b){return pt(a.first+b.first,a.second+b.second);}
pt operator-(pt a,pt b){return pt(a.first-b.first,a.second-b.second);}
double operator*(pt a,pt b){return a.first*b.first+a.second*b.second;}
double norm(pt a){return a*a;}
bool colinear(pt a,pt b,pt c){return abs((b-a)*(c-b))<eps;}
double dist(pt a,pt b){return sqrt(norm(a-b));}
vector<pt> circle(double x,double y,double r){
	vector<pt> ans;
	double z=sqrt(r*r-(x*x+y*y));
	if(z>0){
		ans.push_back(pt(x+z,y));
		ans.push_back(pt(x,y+z));
		ans.push_back(pt(x-z,y));
		ans.push_back(pt(x,y-z));
	}
	return ans;
}
bool ueq(double a,double b){return abs(a-b)<eps;}
bool comp(pt a,pt b){return a.first<b.first;}
bool check(pt p,vector<double>& d){
	for(int i=0;i<n;i++)
		if(!ueq(norm(p-pt(X[i],Y[i])),d[i]))
			return false;
	return true;
}
int main(){
	cin>>n>>m;
	for(int i=0;i<n;i++)
		cin>>X[i]>>Y[i];
	for(int i=0;i<m;i++){
		vector<double> d;
		for(int j=0;j<n;j++){
			cin>>D[j];
			d.push_back(D[j]);
		}
		sort(d.begin(),d.end());
		vector<pt> ans;
		for(pt c:circle(X[0],Y[0],d[0])){
			bool ok=true;
			for(int j=1;j<n;j++){
				double r=sqrt(d[j]),x=c.first-X[j],y=c.second-Y[j];
				if(ueq(x*x+y*y,r*r))
					continue;
				ok=false;
				break;
			}
			if(ok)
				ans.push_back(c);
		}
		sort(ans.begin(),ans.end(),comp);
		int k=0;
		for(pt p:ans)
			k+=check(p,d);
		cout<<k<<' ';
		for(pt p:ans)
			if(check(p,d))
				cout<<p.first<<' '<<p.second<<' ';
		cout<<endl;
	}
}
