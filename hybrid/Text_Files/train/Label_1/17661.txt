#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (i < n - 1 && s[i] == 'V' && s[i + 1] == 'K') ans++;
    else if ((i > 0 && s[i - 1] == 'V' && s[i] == 'K') || (i < n - 1 && s[i] == 'V' && s[i + 1] == 'K')) ans++;
  }
  cout << ans << endl;
  return 0;
}
