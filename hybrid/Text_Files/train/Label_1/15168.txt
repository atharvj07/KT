#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int y;
  std::cin >> y;
  while (true) {
    y++;
    std::vector<int> digits;
    int x = y;
    while (x > 0) {
      digits.push_back(x % 10);
      x /= 10;
    }
    std::sort(digits.begin(), digits.end());
    if (std::unique(digits.begin(), digits.end()) == digits.end()) {
      std::cout << y << std::endl;
      return 0;
    }
  }
}
