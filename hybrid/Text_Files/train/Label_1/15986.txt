#include <bits/stdc++.h>

using namespace std;

long long n, a[55], b[55];

int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n; ++i) cin >> b[i];
  sort(a, a + n);
  reverse(a, a + n);
  long long sum = a[0] + a[1];
  long long ans = sum / (b[0] + b[1]);
  ans *= 1000;
  long long p = ans % 1000;
  if (p != 0) ans += 1000 - p;
  cout << ans << endl;
  return 0;
}
