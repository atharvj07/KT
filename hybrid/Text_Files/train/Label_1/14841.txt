#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int maxn = 2e5 + 10;

int n, m, a[maxn];
map<int, int> mp;
set<int> st;
vector<pair<int, int>> b;

int main() {
    int t;
    cin >> t;
    while (t--) {
        m = b.size();
        cin >> n;
        st.clear();
        for (int i = 1; i <= n; i++) a[i] = i, st.insert(i);
        while (!st.empty()) {
            int u = *st.begin();
            if (u > 1) {
                st.erase(u);
                for (int v : st) {
                    if (u % v == 0) {
                        mp[u]++;
                        mp[v]++;
                        b.push_back({v, u});
                        a[u] = (a[u] + v - 1) / v;
                        if (a[u] == 1) { st.erase(u); break; }
                    }
                }
            } else st.erase(u);
        }
        for (auto it = mp.rbegin();; it--) {
            if (it == mp.rend()) break;
            if (it->second % 2 == 0) continue;
            int u = it->first;
            for (int v : st) {
                if (u % v == 0) {
                    mp[u]++;
                    mp[v]++;
                    b.push_back({v, u});
                    a[u] = (a[u] + v - 1) / v;
                    if (a[u] == 1) { st.erase(u); break; }
                }
            }
        }
        while (st.size() > 1) {
            int u = *st.rbegin(); st.erase(u);
            int v = *st.rbegin(); st.erase(v);
            b.push_back({v, u});
            mp[u]++;
            mp[v]++;
            a[u] = (a[u] + v - 1) / v;
        }
        cout << b.size() << endl;
        for (auto x : b) cout << x.first << ' ' << x.second << endl;
        b.clear();
        mp.clear();
    }
    return 0;
}
