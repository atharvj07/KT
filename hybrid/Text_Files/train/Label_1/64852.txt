n = int(input())
group_a = list(map(int, input().split()))
group_b = list(map(int, input().split()))

performance_counts_a = [0] * 6
performance_counts_b = [0] * 6

for performance in group_a:
    performance_counts_a[performance] += 1

for performance in group_b:
    performance_counts_b[performance] += 1

exchanges = 0
for i in range(1, 6):
    if (performance_counts_a[i] + performance_counts_b[i]) % 2 != 0:
        print(-1)
        break
    exchanges += abs(performance_counts_a[i] - performance_counts_b[i]) // 2
else:
    print(exchanges // 2)
