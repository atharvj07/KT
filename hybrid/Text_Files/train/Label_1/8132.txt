#include <iostream>
#include <queue>
#include <algorithm>
#include <cassert>
using namespace std;

int n, m, rep[100005], degree[100005];
struct Edge {
  int to, id;
};
vector<Edge> graph[100005];
vector<int> ans1, ans2;

inline bool cmp(const Edge &e1, const Edge &e2) {
  return degree[e1.to] > degree[e2.to];
}

int find(int a) {
  if(a != rep[a]) rep[a] = find(rep[a]);
  return rep[a];
}

int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  cin >> n >> m;
  for(int i = 1;i <= n;++i) rep[i] = i;
  for(int i = 1;i <= m;++i) {
    int u, v;
    cin >> u >> v;
    graph[u].push_back({v, i});
    graph[v].push_back({u, i});
    ++degree[u], ++degree[v];
  }
  // greedy to maintain vertex cover
  queue<int> q;
  for(int i = 1;i <= n;++i) if(degree[i] == 1) q.push(i);
  vector<int> remove;
  while(!q.empty()) {
    int u = q.front(); q.pop();
    for(auto edge : graph[u]) {
      int v = edge.to, id = edge.id;
      --degree[v];
      if(degree[v] == 1) q.push(v);
      remove.push_back(id);
    }
  }
  sort(remove.begin(), remove.end());
  for(auto i : remove) {
    int u, v;
    for(auto edge : graph[i]) {
      u = i, v = edge.to;
      break;
    }
    int p1 = find(u), p2 = find(v);
    if(p1 == p2) {
      cout << "-1";
      return 0;
    }
    rep[p2] = p1; // merge connected components
    ans1.push_back(u), ans2.push_back(v);
  }
  assert((int)ans1.size() == (int)ans2.size());
  cout << ans1.size() << endl;
  for(int i = 0;i < ans1.size();++i) {
    cout << ans1[i];
    if(i + 1 != ans1.size()) cout << ' ';
    else cout << endl;
  }
  cout << ans2.size() << endl;
  for(int i = 0;i < ans2.size();++i) {
    cout << ans2[i];
    if(i + 1 != ans2.size()) cout << ' ';
    else cout << endl;
  }
  return 0;
}
