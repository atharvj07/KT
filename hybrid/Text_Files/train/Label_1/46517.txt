import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Long> seq = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            seq.add(sc.nextLong());
        }
        Collections.sort(seq);
        for (Long v : seq) {
            System.out.print(v + " ");
        }
    }
}
