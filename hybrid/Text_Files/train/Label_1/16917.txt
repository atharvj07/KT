#include <iostream>
#include <vector>
#include <algorithm>
#include <cstring>

using namespace std;

int main() {
    int n, a, b;
    cin >> n >> a >> b;
    string s;
    cin >> s;
    vector<int> dp(n + 1, 1e9);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < i; j++) {
            if (s.substr(j, i - j) == s.substr(0, i - j)) {
                dp[i] = min(dp[i], dp[j] + b);
            }
        }
        dp[i] = min(dp[i], dp[i - 1] + a);
    }
    cout << dp[n] << endl;
    return 0;
}
