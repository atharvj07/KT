#include <bits/stdc++.h>
using namespace std;
#define ll long long

vector<ll> fre;
int main() {
    ll n, h, m, k;
    cin >> n >> h >> m >> k;
    for (ll i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        fre.push_back(x * m + y);
    }
    sort(fre.begin(), fre.end());
    ll ans = LLONG_MAX;
    ll res = -1;
    for (ll t = 0; t <= m / 2 - k; t++) {
        ll cnt = 0;
        for (ll i = 0; i < n; i++) {
            ll tmp = fre[i] - t;
            if (tmp % (m / 2) >= k) cnt++;
        }
        if (cnt < ans) {
            ans = cnt;
            res = t;
        }
    }
    cout << ans << ' ' << res << endl;
    for (ll i = 0; i < n; i++) {
        ll tmp = fre[i] - res;
        if (tmp % (m / 2) >= k) cout << fre[i] / m << ' ' << fre[i] % m << endl;
    }
    return 0;
}
