#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5+1, inf = 1e9+7;
int l[maxn], r[maxn], ans[maxn];
int main() {
    int n; cin >> n;
    for(int i = 0; i < n; i++) {
        cin >> l[i] >> r[i];
    }
    for(int i = n-1; i >= 0; i--) {
        ans[i] = inf;
        if(i == n-1) ans[i] = r[i];
        else ans[i] = min(ans[i+1]+2, r[i]);
    }
    for(int i = 0; i < n; i++) {
        if(i) ans[i] = max(ans[i], ans[i-1]+2);
        ans[i] += l[i];
    }
    cout << *min_element(ans, ans+n) << endl;
}
