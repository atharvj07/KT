from collections import deque, defaultdict
n=int(input())
Rick=deque([int(x) for x in input().split()[1:]])
Morty=deque([int(x) for x in input().split()[1:]])
ans=[[''] * (n-1) for _ in range(2)]
blackhole=set()
for i in range(n+1):
    blackhole.add(i)
for player in range(2):
    sets=[Rick,Morty][player]
    for i in range(1,n):
        state=i
        vis=defaultdict(int)
        while True:
            vis[state]+=1
            if vis[state]>1:
                ans[player][i-1]='Loop'
                break
            if state in blackhole:
                ans[player][i-1]='Win'
                break
            state+=sets.popleft()
            state%=n
            sets.append(state)
for x in ans:print(*x)
