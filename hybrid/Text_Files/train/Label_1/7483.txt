#include <bits/stdc++.h>
using namespace std;

const int MAXN = 4e5 + 5;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    partial_sum(a.begin(), a.end(), a.begin());
    multiset<int> cur;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (!cur.empty() && *cur.begin() != a[i]) {
            ans += cur.size() - 1;
            cur.clear();
        }
        cur.insert(a[i]);
    }
    cout << ans << endl;
    return 0;
}
