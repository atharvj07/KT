#include <iostream>
#include <algorithm>
#include <string>
#include <vector>

using namespace std;

int main() {
	int n;
	cin >> n;
	string s;
	cin >> s;
	vector<int> b(n);
	for (int i = 0; i < n; i++) {
		cin >> b[i];
	}
	sort(s.begin(), s.end());
	int ans = 0;
	for (int i = 0; i < n; i++) {
		if (s[i] != s[n - i - 1]) {
			ans += b[i];
		}
	}
	cout << ans << endl;
	return 0;
}
