import sys

n, m = map(int, input().split())
s = input()
d = {}
ans = []
for query in sys.stdin:
    x, y, l = map(int, query.split())
    flag = True
    sx, sy = x, y
    for i in range(l):
        if sx not in d or sy not in d or d[sx] != d[sy]:
            flag = False
            break
        sx += 1
        sy += 1
    if not flag:
        for i in range(x, x + l):
            if i not in d:
                d[i] = len(d) + 1
    if flag:
        ans.append('YES')
    else:
        ans.append('NO')
print(*ans, sep='\n')
