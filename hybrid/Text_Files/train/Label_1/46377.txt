import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long x1 = sc.nextInt();
            long y1 = sc.nextInt();
            long z1 = sc.nextInt();
            long x2 = sc.nextInt();
            long y2 = sc.nextInt();
            long z2 = sc.nextInt();
            ans += (x2 - x1) * (y2 - y1) * (z2 - z1);
        }
        if (k > 1) {
            ans /= k;
        }
        System.out.println(ans);
    }
}
