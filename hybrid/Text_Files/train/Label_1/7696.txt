#include <bits/stdc++.h>
using namespace std;

const int N = 105;

vector<int>g[N];
int dp[N][N];

int dfs(int s, int p, int t) {
    if(t == 0) return s == p;
    int &ret = dp[s][t];
    if(ret != -1) return ret;
    ret = 0;
    for(auto u: g[s]) {
        if(u != p) {
            ret |= dfs(u, s, t - 1);
        }
    }
    return ret;
}

int32_t main() {
    int n;
    cin >> n;
    memset(dp, -1, sizeof dp);
    for(int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        g[i].push_back(x);
        g[x].push_back(i);
    }
    
    int ans = -1;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++) {
            if(dfs(i, j, j)) {
                ans = j;
                break;
            }
        }
        if(ans != -1) break;
    }
    cout << ans << "\n";
}
