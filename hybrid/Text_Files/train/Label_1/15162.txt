#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    int a = -2, b = -1, c = 0;
    for (int i = 0; i < n; i++)
      if (s[i] == '?') {
        if (b == -1)
          b = i;
        else if (a == -2)
          a = i;
        else if (c == 0)
          c = i;
      } else if (s[i] == 'a') {
        a = -2;
        b = -1;
        c = 0;
      }
    if (a != -2 && b != -1 && c != 0 && b - a == 2 && c - b == 3) {
      s[a] = s[b] = s[c] = 'a';
      s[a + 1] = s[b + 1] = s[c - 1] = 'b';
      s[a + 2] = s[b + 2] = s[c - 2] = 'a';
      s[b + 3] = s[c - 3] = 'c';
      cout << "Yes\n" << s << endl;
    } else
      cout << "No\n";
  }
}
