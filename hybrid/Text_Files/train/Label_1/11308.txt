#include <bits/stdc++.h>
using namespace std;
int main() {
  int l;
  cin >> l;
  string s;
  cin >> s;
  string a = "", b = "";
  for (int i = 0; i < l; i++) {
    if (a == "" || a[a.length() - 1] >= s[i])
      a += s[i];
    else
      b += s[i];
  }
  stringstream sa(a), sb(b);
  int x, y;
  sa >> x;
  sb >> y;
  cout << x + y << endl;
  return 0;
}
