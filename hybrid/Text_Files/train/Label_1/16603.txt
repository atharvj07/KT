#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    int w, h;
    bool flag = true;
    while (n--) {
        cin >> w >> h;
        if (w > h) {
            swap(w, h);
            flag = false;
        }
    }
    cout << (flag ? "YES" : "NO") << '\n';
    return 0;
}
