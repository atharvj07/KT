def is_valid(requests, m, t, s):
    if len(requests) < m: return False
    requests.sort()
    for i in range(len(requests) - m):
        if requests[i + m - 1] - requests[i] + 1 > t:
            return False
    return True

def dfs(requests, m, t, s, ids, i, j):
    if i == len(requests): return j == len(ids)

    if requests[i] - s >= t:
        if dfs(requests, m, t, requests[i], ids, i+1, j+1):
            ids.append(j+1)
            return True
        if j > 0 and dfs(requests, m, t, s, ids, i+1, j):
            return True
    else:
        if dfs(requests, m, t, s, ids, i+1, j):
            return True

    return False

n, m, t = map(int, input().split())
requests = list()
s = None
for _ in range(n):
    h, m, s_ = map(int, input().split(':'))
    if s == None: s = 60 * 60 * h + 60 * m + s_
    requests.append(60 * 60 * h + 60 * m + s_)
if not is_valid(requests, m, t, s): print("No solution")
else:
    ids = list()
    if not dfs(requests, m, t, s, ids, 0, 0): print("No solution")
    else: print(len(ids))
    for id in ids: print(id)
