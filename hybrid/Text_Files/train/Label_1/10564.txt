#include <iostream>
#include <set>
using namespace std;

int main() {
  int n;
  cin >> n;
  set<pair<int, int>> s;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    s.insert({x, y});
  }
  if (s.size() != 4)
    cout << -1 << endl;
  else {
    pair<int, int> p1, p2, p3, p4;
    for (auto it = s.begin(); it != s.end(); it++) {
      if (it == s.begin())
        p1 = *it;
      else if (it == next(s.begin()))
        p2 = *it;
      else if (it == next(s.begin(), 2))
        p3 = *it;
      else
        p4 = *it;
    }
    int area = (p2.first - p1.first) * (p3.second - p1.second);
    cout << area << endl;
  }
  return 0;
}
