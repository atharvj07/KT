import sys

def solve(n):
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    odd = 0
    vis = [False] * (n+1)
    def dfs(u):
        nonlocal odd
        vis[u] = True
        for v in adj[u]:
            if not vis[v]:
                dfs(v)
        if len(adj[u]) % 2 == 1:
            odd += 1

    dfs(1)
    if odd > 2:
        return -1
    else:
        return n - 1 - odd // 2

n = int(input())
res = solve(n)
sys.stdout.write(str(res) + "\n")
