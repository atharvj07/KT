#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		string x;
		cin>>x;
		int ans=0;
		for(int i=1;i<=12;i++)
		{
			if(x[i-1]=='O')
				continue;
			int j=12/i;
			if(j*i==12)
				ans++;
		}
		cout<<ans<<" ";
		for(int i=1;i<=12;i++)
		{
			if(x[i-1]=='O')
				continue;
			int j=12/i;
			if(j*i==12)
				cout<<i<<"x"<<j<<" ";
		}
		cout<<endl;
	}
}
