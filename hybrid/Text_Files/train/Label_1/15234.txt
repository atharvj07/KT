#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    bool flag=1;
    int n=s.size();
    for(int i=0;i<n;i++){
        for(char j='a';j<='z';j++){
            string str=s;
            str.insert(i,1,j);
            string t=str;
            reverse(t.begin(),t.end());
            if(t==str){
                cout<<str<<endl;
                flag=0;
                break;
            }
        }
        if(!flag)break;
    }
    if(flag)cout<<"NA"<<endl;
}
