def mod_pow(base, p, mod):
    if p == 0:
        return 1
    if p == 1:
        return base
    val = mod_pow(base, int(p / 2), mod)
    val = (val * val) % mod
    if p % 2 != 0:
        val = (val * base) % mod
    return val

mod = 10 ** 9 + 7
n = int(input())
print(mod_pow(6, 2 ** n - 1, mod))
