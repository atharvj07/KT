n, k1, k2 = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
a.sort()
b.sort()
s = sum((x - y) ** 2 for x, y in zip(a, b))
i = 0
j = 0
while k1 != 0 and i != n - 1 and j != n - 1:
    if abs(a[i + 1] - b[j]) < abs(a[i] - b[j + 1]):
        s -= (a[i + 1] - b[j]) ** 2 - (a[i] - b[j]) ** 2; i += 1
    else:
        s -= (a[i] - b[j + 1]) ** 2 - (a[i] - b[j]) ** 2; j += 1
    k1 -= 1
while k2 != 0 and i != n - 1 and j != n - 1:
    if abs(a[i + 1] - b[j]) < abs(a[i] - b[j + 1]):
        s -= (a[i + 1] - b[j]) ** 2 - (a[i] - b[j]) ** 2; i += 1
    else:
        s -= (a[i] - b[j + 1]) ** 2 - (a[i] - b[j]) ** 2; j += 1
    k2 -= 1
print(s)
