import java.util.Arrays;
import java.util.Scanner;

public class VasyaQueue {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
        }

        int[] result = new int[n];
        int[] position = new int[n + 1];
        Arrays.fill(position, -1);

        for (int i = n - 1; i >= 0; i--) {
            position[a[i]] = i;
        }

        for (int i = n - 1; i >= 0; i--) {
            int current = i;
            for (int j = 0; j < c[i] && current > 0 && a[current - 1] < a[current]; j++) {
                swap(a, position, current, current - 1);
                current--;
            }
            result[position[a[i]]] = i + 1;
        }

        for (int i = 0; i < n; i++) {
            System.out.print(result[i] + " ");
        }
    }

    private static void swap(int[] a, int[] position, int i, int j) {
        int temp = a[i];
        a[i] = a[j];
        a[j] = temp;
        position[a[i]] = i;
        position[a[j]] = j;
    }
}
