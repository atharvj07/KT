#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e6 + 7;
int n, len;
int fail[MAXN], ans = 0;
string s, t;
int main() {
  getline(cin, s), getline(cin, t);
  n = s.size(), len = t.size();
  fail[0] = -1;
  for (int i = 0, j = -1; i < len; i++) {
    while (j >= 0 && t[i] != t[j]) j = fail[j];
    fail[i + 1] = ++j;
  }
  for (int i = 0, j = 0; i < n; i++) {
    while (j >= 0 && s[i] != t[j]) j = fail[j];
    if (j == len - 1) {
      j = fail[j], ans++;
    } else {
      j++;
    }
  }
  cout << ans << endl;
}
