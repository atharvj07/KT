for _ in range(int(input())):
    n = int(input())
    a = sorted(map(int, input().split()))
    m = int(input())
    h = []
    for i in range(m):
        p, s = map(int, input().split())
        h.append((p, s))
    h.sort(reverse=True)
    ans = -1
    cur = 0
    for i in h:
        nxt = cur
        for j in range(i[1]):
            if cur < n and a[cur] <= i[0]:
                cur += 1
            nxt += 1
        if cur == n:
            ans = nxt
            break
    print(ans)
