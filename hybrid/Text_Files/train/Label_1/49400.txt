import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int min = x[0];
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (x[i] < min) {
                ans += min - x[i];
            } else {
                min = x[i];
            }
        }
        System.out.println(ans);
    }
}
