#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
  ll n, m, i, j, k, l, idx[100005], x, y;
  char c;
  string s[100005], t[100005];

  cin >> n >> m;
  for (i = 1; i <= n; i++) {
    cin >> s[i];
    t[i] = s[i];
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      if (j % 2 == 1)
        t[i][j - 1] = 'z' - (s[i][j - 1] - 'a');
      else
        t[i][j - 1] = s[i][j - 1];
    }
  }

  sort(t + 1, t + n + 1);
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      if (j % 2 == 1)
        t[i][j - 1] = 'z' - (t[i][j - 1] - 'a');
      else
        t[i][j - 1] = t[i][j - 1];
    }
  }

  for (i = 1; i <= n; i++) {
    for (j = 1; j <= n; j++) {
      if (t[i] == s[j]) {
        idx[i] = j;
        break;
      }
    }
  }

  for (i = 1; i <= n; i++) cout << idx[i] << endl;
  return 0;
}
