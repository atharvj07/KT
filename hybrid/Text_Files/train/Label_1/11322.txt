#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

int main() {
    int n;
    string s;
    cin >> n >> s;
    unordered_map<string, int> m;
    for (int i = 0; i < n - 1; i++) {
        m[s.substr(i, 2)]++;
    }
    int max = 0;
    string ans;
    for (const auto &p : m) {
        if (p.second > max) {
            max = p.second;
            ans = p.first;
        }
    }
    cout << ans << endl;
    return 0;
}
