#include <algorithm>
#include <cstdio>
#include <vector>

int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> a(n), b(n), c(n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
  }
  for (int i = 0; i < n; ++i) {
    scanf("%d", &b[i]);
  }
  for (int i = 0; i < n; ++i) {
    c[i] = (a[i] + b[i]) % n;
  }
  std::sort(c.begin(), c.end());
  std::sort(b.begin(), b.end());
  for (int i = 0; i < n; ++i) {
    b[i] = (n - (c[i] - b[i]) % n) % n;
  }
  for (int i = 0; i < n; ++i) {
    printf("%d ", b[i]);
  }
  printf("\n");
  return 0;
}
