import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] f = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            f[i] = sc.nextInt();
        }
        int[] g = new int[n + 1];
        int[] h = new int[n + 1];
        boolean[] vis = new boolean[n + 1];
        int m = 0;
        Set<Integer> set = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            if (!vis[i]) {
                m++;
                int j = i;
                while (!vis[j]) {
                    g[j] = m;
                    h[m] = j;
                    set.add(j);
                    vis[j] = true;
                    j = f[j];
                }
            }
        }
        if (set.size() != n) {
            System.out.println(-1);
        } else {
            System.out.println(m);
            for (int i = 1; i <= n; i++) {
                System.out.print(g[i] + " ");
            }
            System.out.println();
            for (int i = 1; i <= m; i++) {
                System.out.print(h[i] + " ");
            }
        }
        sc.close();
    }
}
