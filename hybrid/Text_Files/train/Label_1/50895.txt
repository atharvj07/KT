import java.util.*;

public class Main {
    static int[] dx = {1, -1, 0, 0};
    static int[] dy = {0, 0, 1, -1};
    static int n, m;
    static int[][] a;
    static int[][][] dp;
    static int[][] cost;

    public static int dijkstra(int x, int y, int l) {
        for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) dp[i][j][l] = Integer.MAX_VALUE;
        dp[x][y][l] = 0;
        PriorityQueue<Data> q = new PriorityQueue<>();
        q.add(new Data(x, y, l, 0));
        while (!q.isEmpty()) {
            Data data = q.poll();
            int nx = data.x;
            int ny = data.y;
            int nl = data.l;
            int w = data.w;
            if (dp[nx][ny][nl] < w) continue;
            for (int k = 0; k < 4; k++) {
                int nnx = nx + dx[k];
                int nny = ny + dy[k];
                if (nnx < 0 || nnx >= n || nny < 0 || nny >= m) continue;
                int nw = w;
                if (a[nnx][nny] != -1 && a[nnx][nny] != l) nw += cost[l][a[nnx][nny]];
                if (dp[nnx][nny][nl] > nw) {
                    dp[nnx][nny][nl] = nw;
                    q.add(new Data(nnx, nny, nl, nw));
                }
            }
        }
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) ans = Math.min(ans, dp[i][j][l]);
        return ans;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        cost = new int[10][10];
        for (int i = 0; i < 10; i++) {
            for (int j = 0; j < 10; j++) {
                cost[i][j] = scanner.nextInt();
            }
        }
        a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = scanner.nextInt();
            }
        }
        dp = new int[n][m][10];
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] != -1) {
                    ans = Math.min(ans, dijkstra(i, j, a[i][j]));
                }
            }
        }
        System.out.println(ans);
    }

    static class Data implements Comparable<Data> {
        int x;
        int y;
        int l;
        int w;

        Data(int x, int y, int l, int w) {
            this.x = x;
            this.y = y;
            this.l = l;
            this.w = w;
        }

        @Override
        public int compareTo(Data o) {
            return w - o.w;
        }
    }
}
