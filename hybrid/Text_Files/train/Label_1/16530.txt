#include <bits/stdc++.h>
using namespace std;
const int N=2e5+10,M=1<<30;
int n,a[N];
int32_t main(){
    cin>>n;
    for(int i=0;i<n;i++) cin>>a[i];
    sort(a,a+n);
    int mx=0;
    for(int i=0,j=0,msk=0;i<n;i++){
        int l=i;
        for(;j<n&&!(msk&a[j]);j++)
            msk|=a[j];
        if(msk==M) mx=max(mx,j-i);
        else mx=max(mx,j-i-1);
        while(l<j&&a[l]==a[j]) l++;
        msk&=~a[l++];
    }
    cout<<(mx>0?mx:-1);
}
