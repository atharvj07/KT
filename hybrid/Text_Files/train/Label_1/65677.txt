import sys
input=sys.stdin.readline
m=int(input())
wor=list(input().split())
n=int(input())
syno={}
for x in range(n):
    a,b=input().split()
    syno[a]=[b,wor.count(a)]
    syno[b]=[a,wor.count(b)]
for x in range(m):
    if wor[x] in syno:
        wor[x],syno[wor[x]][1]=syno[wor[x]][0],syno[wor[x]][1]+wor.count(syno[wor[x]][0])
print(wor.count('r')+wor.count('R'),sum(len(x)for x in wor))
