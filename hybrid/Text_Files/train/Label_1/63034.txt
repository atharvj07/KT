n = int(input())
a = list(map(int,input().split()))
d,l = [],[]
for i in range(n):
    if a[i]==0:
        d.append(i)
    if a[i]==n:
        l.append(i)
for x in d:
    l.remove(x)
if len(d)>1 or len(l)>1:
    print(-1)
else:
    if len(d)==1 and len(l)==1:
        print(n-1)
    else:
        ans = []
        for x in a:
            if x!=n:
                ans.append(n-x-1)
        ans.sort()
        for x in ans:
            print(x,end=' ')
