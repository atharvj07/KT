#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, x, ans = 0, cnt = 0;
    unordered_map<int, int> mp;
    vector<int> arr;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> x;
        mp[x] = 1;
    }
    for (int i = 1;; i++) {
        if (!mp[i] && m - i >= 0) {
            m -= i;
            ans++;
            arr.push_back(i);
        } else
            break;
    }
    cout << ans << endl;
    for (auto &it : arr) cout << it << " ";
    cout << endl;
    return 0;
}
