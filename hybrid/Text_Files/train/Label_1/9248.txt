#include <bits/stdc++.h>
using namespace std;
const int maxN = 505;
int n, a[maxN], dp[maxN][maxN];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  for (int len = 1; len <= n; ++len) {
    for (int i = 1; i + len - 1 <= n; ++i) {
      if (~a[i] && ~a[i + len - 1] && a[i] == a[i + len - 1]) {
        dp[i][i + len - 1] = dp[i + 1][i + len - 2];
      } else {
        dp[i][i + len - 1] = 1 + min(dp[i][i + len - 2], dp[i + 1][i + len - 1]);
      }
    }
  }
  cout << dp[1][n] << '\n';
  return 0;
}
