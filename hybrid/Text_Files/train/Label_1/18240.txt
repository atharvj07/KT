#include <bits/stdc++.h>
#define ll long long
#define MOD 1000000007
using namespace std;

ll fact[2001];

ll fastpow(ll x, ll p) {
    if (p == 0) {
        return 1;
    }
    if (p == 1) {
        return x;
    }
    if (p % 2 == 0) {
        return fastpow(x * x % MOD, p / 2) % MOD;
    }
    return (x * fastpow(x * x % MOD, p / 2) % MOD) % MOD;
}

ll nCr(ll n, ll r) {
    if (n < r || n < 0 || r < 0) {
        return 0;
    }
    return (fact[n] * fastpow(fact[n - r] * fact[r] % MOD, MOD - 2)) % MOD;
}

int main() {
    ll n;
    cin >> n;
    fact[0] = 1;
    for (ll i = 1; i <= n; i++) {
        fact[i] = (fact[i - 1] * i) % MOD;
    }
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }
    ll ans = 1;
    ll neg = 0;
    for (ll i = 0; i < n; i++) {
        if (a[i] == -1) {
            neg++;
        }
    }
    for (ll i = 1; i <= n; i++) {
        ll cnt = 0;
        for (ll j = 0; j < n; j++) {
            if (a[j] == i) {
                cnt++;
            }
        }
        if (cnt > 0) {
            ans = (ans * nCr(neg + cnt - 1, cnt - 1)) % MOD;
            neg -= cnt;
        }
    }
    cout << ans << endl;
    return 0;
}
