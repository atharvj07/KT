import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    private static boolean[] bullets;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] l = br.readLine().split(" ");
        int length = Integer.valueOf(l[0]), numBullets = Integer.valueOf(l[1]);
        bullets = new boolean[length];
        int[] queries = new int[Integer.valueOf(l[2])];
        for (int i = 0; i < queries.length; i++) {
            queries[i] = Integer.valueOf(br.readLine()) - 1;
        }
        int pos = 0, bulletsLeft = numBullets;
        for (int i = 0; i < length; i++) {
            if (pos == queries[i]) {
                bullets[i] = true;
            } else {
                bullets[i] = false;
                bulletsLeft--;
            }
            pos++;
            if (pos == length) {
                pos = 0;
            }
            if (bulletsLeft == 0) {
                break;
            }
        }
        for (boolean b : bullets) {
            System.out.print(b ? 'X' : '.');
        }
    }
}
