#include<bits/stdc++.h>
using namespace std;
#define ll long long
typedef pair<ll,ll> pi;
#define f first
#define s second
const ll mod=1e9+7;
ll gcd(ll a,ll b)
{
    if(b==0) return a;
    return gcd(b,a%b);
}
int main()
{
    ll n,k;
    cin>>n>>k;
    ll ar[n];
    map<ll,ll>m;
    for(ll i=0;i<n;i++) cin>>ar[i];
    for(ll i=0;i<n;i++)
    {
        ll val=k-ar[i]%k;
        val%=k;
        m[val]++;
    }
    ll sum=0;
    for(auto x:m)
    {
        ll val=x.s;
        ll rem=k-x.f;
        if(rem==x.f) val/=2;
        sum+=val*m[rem];
    }
    cout<<sum<<endl;
    return 0;
}
