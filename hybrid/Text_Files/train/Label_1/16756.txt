#include<bits/stdc++.h>

using namespace std;

int main(){
    string s;
    cin>>s;
    int n = s.size(),cnt=0;
    bool vis[26]={false};
    for(int i=0;i<n;i++){
        if(!vis[s[i]-'a']) cnt++;
        vis[s[i]-'a']=true;
    }
    if(cnt&1) cout<<"IGNORE HIM!";
    else cout<<"CHAT WITH HER!";

    return 0;
}
