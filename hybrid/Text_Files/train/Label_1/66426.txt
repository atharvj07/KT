import sys

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [list(map(int, input().split())) for _ in range(n)]

flag = True
for i in range(1, n - 1):
    for j in range(1, m - 1):
        if a[i][j] != b[i][j]:
            if a[i - 1][j] == a[i + 1][j] or a[i][j - 1] == a[i][j + 1]:
                flag = False
                break

if flag:
    print("Yes")
else:
    print("No")
