#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  ll n;
  cin >> n;
  string s;
  cin >> s;
  s = " " + s;
  vector<ll> ans(n + 1, -1);
  ll dir = 0;
  for (ll i = 1; i <= n; i++) {
    if (s[i] == 'D') dir--;
    if (s[i] == 'U') dir++;
    if ((i == 1 && dir == -1) || (i == n && dir == 1)) {
      ans[i] = 1;
      break;
    }
    if (dir == 0) ans[i] = i;
  }
  if (dir == 0) {
    ll up = 0;
    for (ll i = n; i >= 1; i--) {
      if (s[i] == 'U') up++;
      if (s[i] == 'D') up--;
      if (ans[i] != -1) {
        if (up == 0) ans[i] = n - i;
        else ans[i] = n - i + 2 * up;
      }
    }
  }
  for (ll i = 1; i <= n; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
