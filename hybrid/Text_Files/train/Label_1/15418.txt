#include <iostream>
#include <vector>
#define modulo 1000000007

using namespace std;

typedef long long ll;

ll mpow(ll a, ll e, ll m) {
    a %= m;
    ll r = 1LL;
    while (e > 0) {
        if (e & 1LL) {
            r = (r * a) % m;
        }
        a = (a * a) % m;
        e >>= 1LL;
    }
    return r;
}

int main() {
    int n, m;
    cin >> n >> m;

    ll ans = (((2LL * n) + 1LL) * n) / 2LL;
    ans = mpow(ans, m, modulo);

    cout << ans << endl;

    return 0;
}
