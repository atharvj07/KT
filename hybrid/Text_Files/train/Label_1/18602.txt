#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    long long n, m, a[200005];
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) {
        long long l, r;
        cin >> l >> r;
        l--, r--;
        long long sum = 0;
        for (int j = l; j <= r; j++) { sum += a[j]; }
        if (sum == 0 || l == r && a[l] == 0)
            cout << 1 << "\n";
        else
            cout << 0 << "\n";
    }
}
