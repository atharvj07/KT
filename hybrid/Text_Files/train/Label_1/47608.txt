import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        while (true) {
            int a = input.nextInt();
            int d = input.nextInt();
            int n = input.nextInt();

            if (a == 0 && d == 0 && n == 0) {
                break;
            }

            BigInteger[] primes = new BigInteger[n];
            Arrays.fill(primes, BigInteger.ONE);

            int currentIndex = 0;
            int currentNumber = a;

            while (currentIndex < n) {
                if (isPrime(currentNumber)) {
                    primes[currentIndex] = BigInteger.valueOf(currentNumber);
                    currentIndex++;
                }

                currentNumber += d;
            }

            System.out.println(primes[n - 1]);
        }
    }

    private static boolean isPrime(int n) {
        if (n <= 1) {
            return false;
        }

        if (n <= 3) {
            return true;
        }

        if (n % 2 == 0 || n % 3 == 0) {
            return false;
        }

        int i = 5;

        while (i * i <= n) {
            if (n % i == 0 || n % (i + 2) == 0) {
                return false;
            }

            i += 6;
        }

        return true;
    }
}
