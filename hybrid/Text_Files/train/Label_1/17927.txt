#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    cin.tie(nullptr);
    ios_base::sync_with_stdio(false);

    int n, a, b;
    cin >> n >> a >> b;

    if (a == b) {
        cout << -1 << "\n";
        return 0;
    }

    vector<int> p(n + 1);

    for (int i = 1; i <= n; i++) {
        p[i] = i;
    }

    if (a < b) {
        swap(a, b);
    }

    int ca = 0, cb = 0;

    for (int i = 1; i <= n; i++) {
        if (ca < a) {
            p[i] = i;
            ca++;
        } else if (cb < b) {
            p[i] = i;
            cb++;
        } else {
            p[i] = p[p[i]];
        }
    }

    if (ca == a && cb == b) {
        for (int i = 1; i <= n; i++) {
            cout << p[i] << " ";
        }
    } else {
        cout << -1 << "\n";
    }

    return 0;
}
