#include <algorithm>
#include <iostream>
#include <vector>
int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> t(n);
  for (auto &v : t) {
    std::cin >> v;
  }
  std::vector<int> ans(n);
  std::vector<int> cnt(m + 1);
  for (int i = n - 1, sum = 0; i >= 0; --i) {
    sum += cnt[t[i]];
    ans[i] = n - 1 - i - sum;
    ++cnt[t[i]];
  }
  for (auto &v : ans) {
    std::cout << v << std::endl;
  }
}
