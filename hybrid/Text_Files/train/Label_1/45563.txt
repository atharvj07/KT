import java.util.*;
import java.io.*;

class Main {

    static long solve(long a, long b, long f, long k) {
        if (2 * f > a) return -1;
        if (b >= a) return 0;
        long rounds = (a + f - 1) / f;
        long toGas = a - f;
        long times = (k + rounds - 1) / rounds;
        if (b >= toGas) times--;
        return times;
    }

    public static void main(String[] args) throws Exception {
        Scanner input = new Scanner(System.in);
        long a = input.nextLong(), b = input.nextLong(), f = input.nextLong(), k = input.nextLong();
        input.close();
        System.out.println(solve(a, b, f, k));
    }
}
