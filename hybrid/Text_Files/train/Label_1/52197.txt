import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int x2 = input.nextInt();
        for (int i = 2; i * i <= x2; i++) {
            if (x2 % i == 0) {
                System.out.println(x2 / i);
                break;
            }
        }
    }
}
