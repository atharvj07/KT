#include <iostream>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;

ll modpow(ll a, ll b) {
    ll res = 1;
    while (b > 0) {
        if (b % 2 == 1) res = (res * a) % mod;
        a = (a * a) % mod;
        b /= 2;
    }
    return res;
}

int main() {
    int n, h;
    cin >> n >> h;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == h) ans = (ans + 1) % mod;
        else ans = (ans + modpow(2, n - 1 - i)) % mod;
    }
    cout << ans << endl;
    return 0;
}
