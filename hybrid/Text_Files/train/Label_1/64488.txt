u,v=map(int,input().split())
if u==v and v==0:
	print(0)
elif (u&1) and (v&1):
	print(-1)
elif (u&1) and (v%2==0):
	print(-1)
elif (v&1) and (u%2==0):
	print(-1)
else:
	if u&1==0:
		v-=1
	ans=[]
	while u>=1:
		x=1
		while x*2<=u:
			x*=2
		ans.append(x)
		u-=x
	ans.append(v)
	print(len(ans))
	print(*ans)
