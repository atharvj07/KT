#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int K;
  cin >> K;
  vector<int> ranks(K);
  for (int i = 0; i < K; i++) {
    cin >> ranks[i];
  }
  sort(ranks.begin(), ranks.end());
  int max_rank = ranks[K - 1];
  int declined = 25 - K;
  for (int i = K; i < 25; i++) {
    if (ranks[i] - i <= declined) {
      declined--;
    } else {
      break;
    }
  }
  cout << declined << endl;
  return 0;
}
