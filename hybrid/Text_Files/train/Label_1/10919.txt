#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 105;
int n, m;
pair<int, int> tiles[MAXN][2];

bool can(int x, int y, int rot) {
  if (rot == 0) return tiles[x][0].first == tiles[y][0].second && tiles[x][0].second == tiles[y][1].first;
  return tiles[x][0].second == tiles[y][0].first && tiles[x][1].first == tiles[y][0].second;
}

bool solve() {
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      if (can(i, j, 0) || can(i, j, 1)) return true;
    }
  }
  return false;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 2; j++) {
        cin >> tiles[i][j].first >> tiles[i][j].second;
      }
    }
    cout << (solve() ? "YES\n" : "NO\n");
  }
  return 0;
}
