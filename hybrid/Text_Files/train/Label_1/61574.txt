from sys import stdin

n, k = map(int, stdin.readline().split())
a = list(map(int, stdin.readline().split()))
a.sort()
ans = 0
for i in range(n - 1, -1, -1):
    if k == 0:
        break
    if a[i] < 0:
        a[i], a[0] = a[0], a[i]
        ans -= a[0] * 2
        k -= 1
ans += sum(a)
print(ans)
