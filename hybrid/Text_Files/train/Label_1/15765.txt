#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    long long sum = 0;
    for (int i = 0; i < k; i++) sum += a[i];
    long double avg = sum;
    for (int i = k; i < n; i++) {
        sum += a[i];
        sum -= a[i - k];
        avg += sum;
    }
    avg /= (long double)(n - k + 1);
    cout.precision(9);
    cout << avg << endl;
    return 0;
}
