#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for(int i = a; i < b; i++)

int main() {
    string s;
    cin >> s;
    int m;
    cin >> m;
    rep(_, 0, m) {
        int l, r;
        cin >> l >> r;
        string t = s.substr(l - 1, r - l + 1);
        rep(i, 1, t.size() - 1) {
            if(t[i] == t[i - 1] || t[i] == t[i + 1]) continue;
            cout << "YES" << endl;
            goto fin;
        }
        cout << "NO" << endl;
fin:
        ;
    }
}
