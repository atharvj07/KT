#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N = 200005;
struct edge {
    int to, id;
};
vector<edge> g[N];
int n, m, q, d[N], ans[N], cnt;
bool vis[N];
void dfs(int u, int p, int w) {
    vis[u] = 1;
    d[u] = w;
    for (edge e : g[u]) {
        if (e.to == p) continue;
        if (vis[e.to] == 0)
            dfs(e.to, u, w + (e.id != cnt));
        else
            ans[cnt] = min(ans[cnt], w + (e.id != cnt));
    }
}
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    cin >> q;
    for (int i = 0; i < q; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back({b, cnt});
        g[b].push_back({a, cnt});
    }
    for (int i = 1; i <= n; i++)
        if (!vis[i]) {
            ans[++cnt] = LLONG_MAX;
            dfs(i, i, 1);
        }
    int mx = 0;
    for (int i = 1; i <= cnt; i++) mx = max(mx, ans[i]);
    cout << mx << endl;
}
