import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Vector<Integer> v = new Vector<>();
        for (int i = 0; i < n; i++)
            v.add(in.nextInt());
        int q = in.nextInt();
        Vector<Integer> v2 = new Vector<>();
        for (int i = 0; i < q; i++)
            v2.add(in.nextInt());
        int c = 0;
        for (Integer num : v2)
            c += (v.contains(num) ? 1 : 0);
        System.out.println(c);
    }
}
