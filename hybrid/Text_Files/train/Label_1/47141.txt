import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] h = new int[n][4];
        for (int i = 0; i < n; i++) {
            h[i][0] = sc.nextInt();
            h[i][1] = sc.nextInt();
            h[i][2] = sc.nextInt();
            h[i][3] = sc.nextInt();
        }
        int[] degree = new int[360 + n * 2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 361; j++) {
                int s = j - ((j - h[i][1] - h[i][2] < 0) ? -1 : h[i][1] + h[i][2]);
                degree[j] = Math.max(degree[j], Math.max(0,h[i][3] - s));
            }
        } 
        System.out.println(Arrays.stream(degree).min().getAsInt());
    }
}
