n, q = map(int, input().split())
s = input()
r1, r2, r3 = '', '', ''
for query in range(q):
    x, *cmd = input().split()
    if cmd:
        if x == '1':
            r1 += cmd[0]
        elif x == '2':
            r2 += cmd[0]
        else:
            r3 += cmd[0]
    else:
        if x == '1':
            r1 = r1[:-1]
        elif x == '2':
            r2 = r2[:-1]
        else:
            r3 = r3[:-1]
    print(['NO', 'YES'][all(c not in (r1+r2+r3) for c in s)])
