#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, d, ai, si;
  cin >> n >> d;
  priority_queue<long long> pq;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> ai >> si;
    if (pq.size() && ai - pq.top() >= d)
      continue;
    ans += si;
    pq.push(ai);
  }
  cout << ans << endl;
  return 0;
}
