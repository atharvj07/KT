from math import sqrt

ax, ay, bx, by, tx, ty = [int(i) for i in input().split()]
n = int(input())

d = [(x-ax)**2+(y-ay)**2 for x, y in [tuple(int(i) for i in input().split()) for _ in range(n)]]

d.append(bx**2+by**2)

mn = 0
for s in d:
    mn = max(mn, (tx**2+ty**2-s)/2)

d.sort()

for s in d:
    mn = max(mn, sqrt(s/2.))

for s in d:
    mn = max(mn, (s+tx**2+ty**2)/2.)

print(mn**0.5)
