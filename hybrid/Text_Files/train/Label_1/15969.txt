#include <iostream>
#include <vector>

int main() {
    int n, x1, x2;
    std::cin >> n >> x1 >> x2;
    std::vector<std::pair<double, double>> lines(n);
    for (auto& line : lines)
        std::cin >> line.first >> line.second;

    for (int i = 0; i < n; ++i) {
        const auto& line1 = lines[i];
        for (int j = i + 1; j < n; ++j) {
            const auto& line2 = lines[j];
            if (line1.first == line2.first)
                continue;
            const double x = (line2.second - line1.second) / (line1.first - line2.first);
            if (x1 < x && x < x2) {
                std::cout << "YES" << std::endl;
                return 0;
            }
        }
    }
    std::cout << "NO" << std::endl;
    return 0;
}
