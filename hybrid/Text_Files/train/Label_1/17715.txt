#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef double ld;
const int mod=1e9+7;
const ll inf=1e18;
const int maxn=1e5+10;
const ld eps=1e-8;
int n,a;
int main(){
    cin>>n>>a;
    int ans[3]={0,0,0};
    int k=180/n;
    int d=(180-a)/2;
    d=d/k;
    ans[1]=d+1;
    if(a!=180)ans[2]=d;
    ans[3]=d+2;
    if(ans[2]==0)ans[2]=n;
    if(ans[3]>n)ans[3]-=n;
    cout<<ans[1]<<' '<<ans[2]<<' '<<ans[3]<<endl;
    return 0;
}
