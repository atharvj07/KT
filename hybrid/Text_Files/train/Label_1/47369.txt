import java.util.*;
import java.io.*;
import java.nio.file.*;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(
            new InputStreamReader(System.in));

        int numSubnets = Integer.parseInt(reader.readLine());
        boolean bothMatch = false; // Flag to check if there is a subnet in both whitelist & blacklist

        List<BigInteger> blacklist = new ArrayList<>();
        List<BigInteger> whitelist = new ArrayList<>();
        for (int i = 0; i < numSubnets; i++) {
            String input = reader.readLine();
            String[] split = input.substring(1).split("/");
            BigInteger subnet = new BigInteger(split[0].replaceAll("\\.", ""));
            int mask = split.length == 2 ? Integer.parseInt(split[1]) : 32;
            if (input.charAt(0) == '-') {
                blacklist.add(subnet.shiftRight(32 - mask));
            } else if (input.charAt(0) == '+') {
                whitelist.add(subnet.shiftRight(32 - mask));
            } else {
                throw new IllegalArgumentException("Invalid input");
            }
        }
        reader.close();

        TreeSet<BigInteger> mergedBlacklist = new TreeSet<>(blacklist);
        for (BigInteger subnet : whitelist) {
            if (mergedBlacklist.remove(subnet)) {
                bothMatch = true;
            }
        }
        if (bothMatch) {
            System.out.println(-1);
        } else {
            System.out.println(mergedBlacklist.size());
            for (BigInteger subnet : mergedBlacklist) {
                System.out.println(subnet.shiftLeft(32 - 32).toString(10));
            }
        }
    }
}
