#include <bits/stdc++.h>
using namespace std;
int main() {
    int a[14], ans = 1e9, sum = 0;
    for (int &x : a) cin >> x, sum += x;
    sum -= sum % 2;
    for (int i = 0; i <= 13; ++i) {
        int s = 0;
        if (a[i]) {
            for (int j = i; j <= 13; ++j) {
                s++;
                a[j]--;
                s += a[j];
            }
            a[i % 14] += s;
            ans = min(ans, sum - s);
            for (int j = i; j <= 13; ++j) {
                a[j] = 0;
                a[(j + 1) % 14]++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
