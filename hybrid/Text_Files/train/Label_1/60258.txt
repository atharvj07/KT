c=input().split(',')
d=0
for i in c:
    if i.isdigit():d=max(d,len(i))
print(d)
for j in range(d):
    p=[]
    for i in c:
        if i.isdigit():
            if len(i)==j+1:p.append(', '.join(p[len(p)-int(i):]))
            else:p.append(i)
        else:p.append(i)
    print(' '.join(filter(None,p)))
