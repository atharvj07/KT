import sys

n, m, x, y, vx, vy = map(int, sys.stdin.readline().split())

if vx == 0:
    if vy == -1:
        print(0, y)
elif vy == 0:
    if vx == -1:
        print(x, 0)
elif vx * vy < 0:
    if (vx == 1 and x == n) or (vx == -1 and x == 0):
        print(x, vy * (m - 1 - y))
    elif (vy == 1 and y == m) or (vy == -1 and y == 0):
        print(vx * (n - 1 - x), y)
else:
    print(-1)
