import sys

input = sys.stdin.readline

n, m = map(int, input().strip().split())
a = [input().strip() for _ in range(n)]

valid = True
for i in range(n):
    for j in range(m):
        if a[i][j] == 'X':
            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nx, ny = i + dx, j + dy
                if 0 <= nx < n and 0 <= ny < m:
                    valid &= a[nx][ny] == 'X'

print(['YES', 'NO'][valid])
