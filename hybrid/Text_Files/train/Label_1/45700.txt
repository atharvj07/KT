import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int d = sc.nextInt();
            int a = d * 2 + 1;
            while (divisors(a) < 4) {
                a += d;
            }
            System.out.println(a);
        }
    }

    public static int divisors(int n) {
        int cnt = 0;
        for (int i = 1; i * i <= n; i++) {
            if (n % i == 0) {
                cnt++;
                if (n / i != i) {
                    cnt++;
                }
            }
        }
        return cnt;
    }
}
