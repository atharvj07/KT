#include <iostream>
#include <utility>
#include <algorithm>
using namespace std;

int n, t;

int main() {
    int x;
    pair<int, int> p[301];

    cin >> n >> t;
    for (int i = 0; i < n; i++) {
        cin >> x;
        p[x].first++;
        p[x + n].second++;
    }
    for (int i = 1; i <= 300; i++) {
        p[i].first += p[i - 1].first;
    }
    int ans = 0;
    for (int i = 0; i <= 300; i++) {
        ans = max(ans, p[i].first + p[i].second);
    }
    cout << ans << "\n";
    return 0;
}
