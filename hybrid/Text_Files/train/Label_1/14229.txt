#include <bits/stdc++.h>
using namespace std;
map<char,char>mp;
int main()
{
    string str;
    cin>>str;
    int n=str.size();
    int ans=0,op=0,cl=0;
    for(int i=0;i<n;i++)
    {
        if(str[i]=='<'||str[i]=='{'||str[i]=='['||str[i]=='(')
        {
            op++;
        }
        else
        {
            if(op==0)
            {
                ans++;
            }
            else
            {
                if(str[i]=='>')
                {
                    if(mp['<'])
                        mp['<']--;
                    else
                        ans++;
                }
                else if(str[i]=='}')
                {
                    if(mp['{'])
                        mp['{']--;
                    else
                        ans++;
                }
                else if(str[i]==']')
                {
                    if(mp['['])
                        mp['[']--;
                    else
                        ans++;
                }
                else
                {
                    if(mp['('])
                        mp['(']--;
                    else
                        ans++;
                }
            }
        }
        if(str[i]=='<'||str[i]=='{'||str[i]=='['||str[i]=='(')
        {
            mp[str[i]]++;
        }
    }
    if(op!=0)
        cout<<"Impossible"<<endl;
    else
        cout<<ans<<endl;
    return 0;
}
