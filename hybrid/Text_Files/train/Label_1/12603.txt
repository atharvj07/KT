#include <iostream>
#include <cstdio>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::string s;
        std::cin >> s;
        int a = 0, b = 0, c = 0;
        for (char v : s) {
            if (v == 'T') {
                a++;
            } else if (v == 'M') {
                b += a;
            } else {
                c += b;
            }
        }
        std::cout << ((a == b && b == c) ? "YES" : "NO") << "\n";
    }
    return 0;
}
