#include<bits/stdc++.h>

#define ll long long
#define endl "\n"

using namespace std;

const ll mod = 1e9 + 7;
const double PI = acos(-1.0);

ll bigmod(ll b, ll p, ll m = mod) {
    if(!p) return 1;
    ll x = (__uint128_t)bigmod(b, p / 2, m);
    x = (x * x) % m;
    if(p & 1) x = (x * b) % m;
    return x;
}

ll modinv(ll b, ll m = mod) {
     return bigmod(b, m - 2, m);
}

int main() {
    ll n, l, r;
    cin >> n;
    vector< pair<ll, ll> > v;
    for(ll i = 0; i < n; i++) {
        cin >> l >> r;
        v.push_back({l, r});
    }

    sort(v.begin(), v.end());
    string s = "";

    for(ll i = 0; i < n; i++) {
        s += '(';
    }
    for(ll i = n - 1; i >= 0; i--) {
        s += ')';
    }

    bool flag = true;
    for(ll i = 0; i < n; i++) {
        ll x = v[i].first + i;

        if(s[x] == '(') {
            s[v[i].second + i] = '(';
            s[x] = ')';
        } else {
            s[x] = '(';
            s[v[i].second + i] = ')';
        }
        bool is_correct = true;
        ll bal = 0;

        for(ll j = 0; j < s.size(); j++) {
            if(s[j] == '(') bal++;
            else if(s[j] == ')') bal--;
            if(bal < 0) is_correct = false;
        }
        if(bal) is_correct = false;
        if(!is_correct) {
            flag = false;
            break;
        }
    }

    if(flag)
        cout << s << endl;
    else
        cout << "IMPOSSIBLE" << endl;
    return 0;
}
