import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[sc.nextInt()] = i;
        }

        List<Pair> swaps = new ArrayList<>();
        for (int i = 1; i <= n / 2; i++) {
            if (p[i] > n / 2) {
                for (int j = n / 2 + 1; j <= n; j++) {
                    if (p[j] == i) {
                        swaps.add(new Pair(i, j));
                        p[i] = j;
                        p[j] = i;
                        break;
                    }
                }
            }
        }

        System.out.println(swaps.size());
        for (Pair pair : swaps) {
            System.out.println(pair.x + " " + pair.y);
        }
    }
}

class Pair {
    int x, y;

    public Pair(int x, int y) {
        this.x = x;
        this.y = y;
    }
}
