#include <bits/stdc++.h>
using namespace std;
const int N = 5e4+10;
int n,a[N],cur[N],pre[N],suf[N];
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int T; cin >> T;
  while(T--) {
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= n; i++) cur[i] = a[i];
    for(int i = 1; i <= n; i++) {
      if(cur[i] >= cur[i+1]) cur[i+1] = cur[i] + 1;
    }
    for(int i = n; i >= 1; i--) {
      if(cur[i] >= cur[i-1]) cur[i-1] = cur[i] + 1;
    }
    for(int i = 1; i <= n; i++) {
      pre[i] = cur[i] + a[i];
    }
    for(int i = n; i >= 1; i--) {
      suf[i] = cur[i] + a[i];
    }
    int ok = 0, id1, id2, id3;
    for(int i = 2; i <= n-1; i++) {
      if(pre[i]+1 < suf[i+1]) { ok = 1, id1 = i; break; }
    }
    for(int i = 1; i <= n-2; i++) {
      if(pre[i+1]+1 < suf[i+2]) { ok = 1, id2 = i, id3 = i+1; if(id2 == id1) id2++; }
    }
    if(ok) cout << id1 << ' ' << id2 << ' ' << id3 << '\n';
    else cout << "-1\n";
  }
}
