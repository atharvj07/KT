def min_operations(n, a):
    dp = [0] * (n + 1)
    for i in range(n):
        dp[i + 1] = dp[i] + a[i]
    for l in range(n):
        min_val = float('inf')
        for r in range(l, n):
            min_val = min(min_val, a[r])
            dp[r + 1] = min(dp[r + 1], dp[l] + min_val * (r - l + 1))
    return dp[n]

n = int(input())
a = list(map(int, input().split()))
print(min_operations(n, a))
