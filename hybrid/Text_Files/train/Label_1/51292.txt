import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int ans = n;
        for (int i = 0; i < n; i++) {
            int j = i + 1;
            while (j < n && a[j] - a[i] <= k) {
                ans--;
                j++;
            }
        }
        System.out.println(ans);
    }
}
