#include<iostream>
#include<vector>
#include<string>
using namespace std;

int main() {
  string table, hand;
  cin >> table >> hand;
  vector<int> rank(10), suit(4);
  for (char i: hand) {
    if (i >= '2' && i <= '9') rank[i - '0']++;
    else if (i == 'T') rank[10]++;
    else if (i == 'J') rank[11]++;
    else if (i == 'Q') rank[12]++;
    else if (i == 'K') rank[13]++;
    else if (i == 'A') rank[14]++;
    else suit[i - 'D']++;
  }
  if (rank[table[0] - '0'] || suit[table[1] - 'D']) cout << "YES" << endl;
  else cout << "NO" << endl;
  return 0;
}
