import java.util.*;

public class Main {
    static final int N = 3000;
    static final int M = 26;
    static char[] s = new char[N];
    static int[] cnt = new int[M];

    static boolean solve(int k, int n, Scanner in) {
        Arrays.fill(cnt, 0);
        for (int i = 0; i < k; i++) {
            String si = in.nextLine();
            for (int j = 0; j < n; j++) {
                cnt[si.charAt(j) - 'a']++;
            }
        }
        int odd = 0;
        for (int i = 0; i < M; i++) {
            if (cnt[i] % 2 == 1) odd++;
        }
        if (odd > 2 || odd == 1) return false;
        for (int i = 0; i < M; i++) {
            if (cnt[i] % 2 == 1) {
                for (int j = 0; j < N; j++) {
                    if (s[j] == 0) {
                        s[j] = (char) (i + 'a');
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < M; i++) {
            if (cnt[i] % 2 == 0) {
                for (int j = 0; j < N; j++) {
                    if (s[j] == 0) {
                        s[j] = (char) (i + 'a');
                        break;
                    }
                }
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int n = in.nextInt();
        in.nextLine();
        if (solve(k, n, in)) {
            System.out.println(new String(s));
        } else {
            System.out.println(-1);
        }
    }
}
