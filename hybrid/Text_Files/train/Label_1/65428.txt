def max_cost(n, m, souvenirs):
    dp = [[[0 for _ in range(4)] for __ in range(4)] for ___ in range(m+1)]
    souvenirs.sort(key=lambda x: -x[1])

    for i in range(1, m+1):
        for j in range(4):
            for k in range(4):
                dp[i][j][k] = dp[i-1][j][k]
                if j > 0 and i >= 1:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j-1][k] + souvenirs[j+k-1][1])
                if k > 0 and i >= souvenirs[j+k-1][0]:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-souvenirs[j+k-1][0]][j][k-1] + souvenirs[j+k-1][1])
    
    result = 0
    for i in range(4):
        for j in range(4):
            result = max(result, dp[m][i][j])
    return result

n, m = map(int, input().split())

print(max_cost(n, m, souvenirs))
