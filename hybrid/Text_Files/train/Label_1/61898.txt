from bisect import bisect_left, bisect_right
n,m,ta,tb,k=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
b.sort()
for i in range(1,n):a[i]-=a[i-1]
for i in range(1,m):b[i]-=b[i-1]
l=[ta+a[i] for i in range(n)]
r=[b[i]+tb for i in range(m)]
sz=len(l)
d={0:0}
j=0
for i in range(1,sz):
    if j+1<=k and i+1<=m:
        while j+1<=k and j+1<=i and b[j]+tb<ta+a[i]:j+=1
        d[i]=max(d[i-1],r[j]-ta-a[i])
    else:d[i]=d[i-1]
#
i=bisect_left(l,[r[j]-d[sz-1] for j in range(m)])
print(-1 if i>=sz else max(l[i-1]+d[sz-1],r[j]-d[sz-1]))
