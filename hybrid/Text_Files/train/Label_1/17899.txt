#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int d;
    cin >> d;
    int a = d + 1;
    while (true) {
      vector<int> divs;
      for (int i = 1; i * i <= a; i++) {
        if (a % i == 0) {
          divs.push_back(i);
          if (i != a / i) divs.push_back(a / i);
        }
      }
      if (divs.size() >= 4) {
        int mi = divs[0], ma = divs[0];
        for (auto div : divs) {
          if (div < mi) mi = div;
          if (div > ma) ma = div;
        }
        if (ma - mi >= d) break;
      }
      a++;
    }
    cout << a << endl;
  }
  return 0;
}
