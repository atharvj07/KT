import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int[] c = new int[n - 1];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            c[i] = sc.nextInt();
        }
        int[] dp = new int[n + 1];
        dp[a[0]] = b[0];
        for (int i = 1; i < n; i++) {
            for (int j = i; j >= 0; j--) {
                dp[a[i]] = Math.max(dp[a[i]], dp[j] + b[i] + (j != 0 ? c[j - 1] : 0));
            }
        }
        System.out.println(dp[a[n - 1]]);
    }
}
