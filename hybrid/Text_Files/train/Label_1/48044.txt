import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int m = arr.length / 2;
        double median = (arr.length % 2 == 0) ? (arr[m] + arr[m - 1]) / 2.0 : arr[m];
        double mean = 0;
        for (int x : arr) {
            mean += x;
        }
        mean /= n;
        double skewness = mean - median;
        int[] subset = new int[n];
        int k = 0;
        for (int x : arr) {
            if (mean - x >= skewness) {
                subset[k++] = x;
            }
        }
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(subset[i] + " ");
        }
    }
}
