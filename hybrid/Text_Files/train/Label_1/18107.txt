#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  vector<int> ans(n);
  int ind = 0;
  for (int i = 0; i < n; i++) {
    if (v[i] == n - ind) {
      ans[ind++] = v[i];
      while (ind < n && !ans[ind]) {
        ind++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (ans[i])
      cout << ans[i] << " ";
    cout << endl;
  }
  return 0;
}
