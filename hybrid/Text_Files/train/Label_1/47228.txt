import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String[] line1 = br.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int m = Integer.parseInt(line1[1]);

        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }

        for (int i = 1; i <= n; i++) {
            String[] line2 = br.readLine().split(" ");
            int degree = Integer.parseInt(line2[0]);
            for (int j = 1; j <= degree; j++) {
                int adjVertex = Integer.parseInt(line2[j]);
                adjList.get(i).add(adjVertex);
            }
        }

        int start = Integer.parseInt(br.readLine());

        boolean[] visited = new boolean[n + 1];
        boolean[] visitedTwice = new boolean[n + 1];
        visited[start] = true;
        List<Integer> path = new ArrayList<>();
        path.add(start);
        if (dfs(start, adjList, visited, visitedTwice, path)) {
            System.out.println("Win");
            for (int vertex : path) {
                System.out.print(vertex + " ");
            }
        } else if (path.size() >= 106) {
            System.out.println("Draw");
        } else {
            System.out.println("Lose");
        }
    }

    private static boolean dfs(int curr, List<List<Integer>> adjList, boolean[] visited, boolean[] visitedTwice, List<Integer> path) {
        for (int adj : adjList.get(curr)) {
            if (!visited[adj]) {
                visited[adj] = true;
                path.add(adj);
                if (dfs(adj, adjList, visited, visitedTwice, path)) {
                    return true;
                }
                path.remove(path.size() - 1);
                visited[adj] = false;
            } else if (!visitedTwice[adj]) {
                visitedTwice[adj] = true;
                path.add(adj);
                return true;
            }
        }
        return false;
    }
}
