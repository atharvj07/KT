import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        int ans = 0;
        for (int i = 0; i < 26; i++) {
            if (cnt[i] % 2 == 1) {
                ans++;
            }
        }
        if (ans <= 1) {
            ans++;
        }
        System.out.println(ans);
        sc.close();
    }
}
