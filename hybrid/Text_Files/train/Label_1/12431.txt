#include <cstdio>
#include <vector>
#include <stack>
using namespace std;
#define maxn 200000
#define ll long long
int n, m, a[maxn + 1];
vector<vector<int>> adj(maxn + 1);
ll sum[maxn + 1]; stack<int> st; ll ans[maxn + 1];
int main() {
  int type, x, val;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  for (int i = 1; i < n; i++) {
    int v, u; scanf("%d%d", &v, &u); adj[v].push_back(u);
  }
  while (m--) {
    scanf("%d%d", &type, &x);
    if (type == 1) {
      scanf("%d", &val);
      sum[x] += val;
      for (auto t : adj[x]) st.push(t);
      while (!st.empty()) {
        int t = st.top();
        st.pop();
        sum[t] -= val;
        for (auto tt : adj[t]) st.push(tt);
      }
    } else
      printf("%lld\n", sum[x] + a[x]);
  }
  return 0;
}
