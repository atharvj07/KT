#include <bits/stdc++.h>

using namespace std;

const int N = 1005;

int n;
int x[N];

bool check(int i, int j, int k) {
    if (i < k && j < k) {
        return (x[k] - x[i]) * (x[j] - x[i]) > (x[j] - x[k]) * (x[i] - x[k]);
    }
    if (i < k && j > k) {
        return (x[j] - x[i]) * (x[k] - x[i]) < (x[k] - x[j]) * (x[i] - x[k]);
    }
    if (i > k && j < k) {
        return (x[k] - x[j]) * (x[i] - x[j]) > (x[i] - x[k]) * (x[j] - x[k]);
    }
    if (i > k && j > k) {
        return (x[j] - x[k]) * (x[i] - x[k]) < (x[i] - x[j]) * (x[k] - x[j]);
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 2; j < n; j++) {
            for (int k = i + 1; k < j; k++) {
                if (check(i, j, k)) {
                    cout << "yes" << endl;
                    return 0;
                }
            }
        }
    }
    cout << "no" << endl;
    return 0;
}
