import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] h = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = scanner.nextLong();
        }
        System.out.println(solve(h, n));
    }

    private static long solve(long[] h, int n) {
        long cnt = 0;
        while (true) {
            boolean flag = false;
            for (int i = 1; i < n - 1; i++) {
                if (h[i] < h[i - 1] && h[i] < h[i + 1]) {
                    h[i]--;
                    flag = true;
                }
            }
            cnt++;
            if (!flag) {
                return cnt;
            }
        }
    }
}
