n = int(input())
a = list(map(int, input().split()))

alice_seats = a[0]
total_seats = sum(a)
coalition = [1]
needed_majority = total_seats // 2 + 1

if alice_seats >= needed_majority:
    print(1)
    print(1)
else:
    a_sorted = sorted(enumerate(a[1:], start=2), key=lambda x: x[1])
    current_seats = alice_seats
    for party in a_sorted:
        if party[1] * 2 <= alice_seats and current_seats < needed_majority:
            coalition.append(party[0])
            current_seats += party[1]
    if current_seats >= needed_majority:
        print(len(coalition))
        print(' '.join(map(str, coalition)))
    else:
        print(0)
