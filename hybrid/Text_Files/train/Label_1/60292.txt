import sys
input=sys.stdin.readline

n,q,m=map(int,input().split())
a=list(map(int,input().split()))
for qi in range(q):
    t,l,r=[int(x) for x in input().split()]
    if(t==1):
        a[l-1:r]=a[l-1:r][::-1]
    else:
        a[l-1:r]=a[l-1:r][-(r-l):]+a[l-1:r][:-(r-l)]
b=[int(x) for x in input().split()]
for bi in b:
    print(a[bi-1])
