#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, k, x;
    cin >> n >> m >> k;
    unordered_map<int, int> um;
    while (n--) {
        cin >> x;
        um[x]++;
    }
    unordered_set<int> s;
    while (m--) {
        cin >> x;
        s.insert(x);
    }
    for (unordered_map<int, int>::iterator it = um.begin(); it != um.end(); it++) {
        if (it->second > s.count(it->first)) {
            cout << "YES";
            return 0;
        }
    }
    cout << "NO";
    return 0;
}
