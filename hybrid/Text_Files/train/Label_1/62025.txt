import sys
t=int(input())
for _ in range(t):
    a,b=map(int,input().split())
    x=input()
    n = len(x)
    x='0'+x+'0'
    dp = [[[0 for i in range(2)] for j in range(n+1)] for k in range(n+1)]
    for i in range(n-1,-1,-1):
        for j in range(i+1):
            for k in range(2):
                if x[j]!=x[i+1]:
                    dp[i][j][k]=min(dp[i][j][k],dp[i+1][i+1][1])
                if k or x[j]!=x[i]:
                    dp[i][j][k]=min(dp[i][j][k],a+dp[i+1][i][k])
                dp[i][j][k]=min(dp[i][j][k],b+dp[i+1][j][0])
    print(dp[0][0][0])
