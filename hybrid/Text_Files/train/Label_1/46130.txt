import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;

class Main {

    static class Scanner {
        BufferedReader br;
        StringTokenizer strtok;
 
        public Scanner() throws IOException {
            br = new BufferedReader(new InputStreamReader(System.in));
            strtok = null;
        }

        public String next() throws IOException {
            while(strtok == null || !strtok.hasMoreTokens()) {
                strtok = new StringTokenizer(br.readLine());
            }
            return strtok.nextToken();
        }

        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }

    public static boolean solve(int H, int W, int N, int s_r, int s_c, String S, String T) {
        int[] U, R, D, L;
        U = new int[N];
        R = new int[N];
        D = new int[N];
        L = new int[N];

        int sum_U = 0;
        int sum_R = 0;
        int sum_D = 0;
        int sum_L = 0;

        for (int i = 0; i < N; i++) {
            if (T.charAt(i) == 'U') {
                U[sum_U++] = i;
            } else if (T.charAt(i) == 'R') {
                R[sum_R++] = i;
            } else if (T.charAt(i) == 'D') {
                D[sum_D++] = i;
            } else if (T.charAt(i) == 'L') {
                L[sum_L++] = i;
            }
        }

        if (s_r == 1) {
            for (int i = 0; i < sum_U; i++) {
                if (S.charAt(U[i]) == 'U') {
                    return false;
                }
            }
        }
        if (s_r == H) {
            for (int i = 0; i < sum_D; i++) {
                if (S.charAt(D[i]) == 'D') {
                    return false;
                }
            }
        }
        if (s_c == 1) {
            for (int i = 0; i < sum_L; i++) {
                if (S.charAt(L[i]) == 'L') {
                    return false;
                }
            }
        }
        if (s_c == W) {
            for (int i = 0; i < sum_R; i++) {
                if (S.charAt(R[i]) == 'R') {
                    return false;
                }
            }
        }

        return true;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner();

        int H = sc.nextInt();
        int W = sc.nextInt();
        int N = sc.nextInt();

        int s_r = sc.nextInt();
        int s_c = sc.nextInt();

        String S = sc.next();
        String T = sc.next();

        System.out.println(solve(H, W, N, s_r, s_c, S, T) ? "YES" : "NO");
    }
}
