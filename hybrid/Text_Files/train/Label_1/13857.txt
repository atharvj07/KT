#include <bits/stdc++.h>
using namespace std;
const int N = 5005;
int n,dp[N],a[N];
int main() {
    cin >> n;
    for (int i = 1;i <= n;++i)
        cin >> a[i];
    dp[0]=0;
    for(int i=1;i<=n;++i)dp[i]=1000000000;
    for(int i=1;i<=n;++i){
        for(int j=i;j>=1;--j){
            dp[i]=min(dp[i],dp[j-1]+a[i]-a[j]);
        }
    }
    for(int i=1;i<=n/2;++i)cout<<dp[i]<<" ";
    cout<<endl;
    return 0;
}
