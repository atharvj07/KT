#include<bits/stdc++.h>
using namespace std;
map<int,int>mp;
vector<int>v;
int main()
{
    int n,i,x,c=0;
    cin>>n;
    for(i=1; i<=n; i++)
    {
        cin>>x;
        v.push_back(x);
        mp[x]++;
    }
    sort(v.begin(),v.end());
    for(i=0; i<n; i++)
    {
        if(mp[v[i]]>=2)
        {
            v[i]=-1;
            v[i+1]+=v[i];
            mp[v[i+1]]++;
            mp[v[i]]--;
            i++;
        }
    }
    sort(v.begin(),v.end());
    for(i=0; i<n; i++)
    {
        if(v[i]==-1)continue;
        c++;
    }
    cout<<c<<endl;
    for(i=0; i<n; i++)
    {
        if(v[i]==-1)continue;
        cout<<v[i]<<" ";
    }
    return 0;
}
