#include <algorithm>
#include <iostream>
#include <set>
#include <vector>
using namespace std;

bool intersect(int x1, int x2, int x3, int x4) {
  // x1 < x2, x3 < x4
  //        x1    x2
  //          x3   x4
  return (x2 >= x3 && x2 <= x4) || (x4 >= x1 && x4 <= x2);
}

int main() {
  int n;
  cin >> n;
  vector<int> x(n);
  for (int i = 0; i < n; i++) {
    cin >> x[i];
  }
  sort(x.begin(), x.end());
  set<int> s;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (intersect(x[i], x[j], s.empty() ? -1e9 : *s.begin(), s.empty() ? 1e9 : *s.rbegin())) {
        cout << "yes" << endl;
        return 0;
      }
    }
    s.insert(x[i]);
  }
  cout << "no" << endl;
  return 0;
}
