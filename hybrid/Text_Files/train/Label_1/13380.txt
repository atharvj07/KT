#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e2 + 1, MAXM = 1e2 + 1, INF = 1e9;
int n, m, a[MAXN][MAXM];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> a[i][j];
            }
            sort(a[i] + 1, a[i] + m + 1);
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) { cout << a[i][j] << " "; }
            cout << "\n";
        }
    }
    return 0;
}
