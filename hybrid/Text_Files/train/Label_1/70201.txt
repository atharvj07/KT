import sys
input = sys.stdin.readline

n = int(input())
x = [0] + list(map(int, input().split()))
for i in range(1, n):
    x[i] += x[i - 1]
ans = n
for i in range(1, n + 1):
    for j in range(i, n + 1):
        if x[j] - (x[i - 1] if i > 1 else 0) <= 0:
            continue
        ans = min(ans, (j - i) * 2 + (1 if i > 1 else 0))
print(ans)
