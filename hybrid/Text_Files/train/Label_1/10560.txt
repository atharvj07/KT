#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, q;
  cin >> n >> m >> q;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  for (int i = 0; i < m; ++i) { b[i] = m + 1; }
  for (int i = n - 1; i >= 0; --i) { b[a[i] - 1] = i; }
  for (int i = 0; i < q; ++i) {
    long long x;
    cin >> x;
    cout << *upper_bound(b.begin(), b.end(), x - n) + 1 << '\n';
  }
  return 0;
}
