#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, ans = 4;
  cin >> n >> m;
  bool f = false;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int x;
      cin >> x;
      if (x) {
        if (i <= 0 || j <= 0 || i == n - 1 || j == m - 1) {
          f = true;
        }
        ans = min(ans, (min(i + 1, n - i) + min(j + 1, m - j)));
      }
    }
  }
  if (f) {
    ans = 1;
  }
  cout << ans;
  return 0;
}
