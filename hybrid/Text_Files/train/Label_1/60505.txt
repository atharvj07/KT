def pow(base: int, exp: int) -> int:
    if exp == 0:
        return 1
    elif exp == 1:
        return base
    half = pow(base, exp // 2)
    if exp % 2 == 0:
        return half * half % 1000000007
    else:
        return half * half % 1000000007 * base % 1000000007


mod = int(1e9) + 7
a, b, n, x = map(int, input().split())
m = pow(a, n)
print((m * x + b * (m - 1) * pow(a, mod - 2)) % mod)
