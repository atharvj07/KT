import sys
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = [a[0]]
    for el in a[1:]:
        b.append(max(b[-1], el))
    c = [a[-1]]
    for el in reversed(a[:-1]):
        c.append(min(c[-1], el))
    c.reverse()
    print(max([x * y for x, y in zip(b, c)]))
