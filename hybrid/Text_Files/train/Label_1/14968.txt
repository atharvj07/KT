#include <iostream>
using namespace std;
int gcd (int a, int b) {
    while (b) {
        swap (a, b);
        b %= a;
    }
    return a;
}
int main () {
    int x, y, n;
    cin >> x >> y >> n;
    int a = x / y, b = y;
    while (n --) {
        int x1 = x / b * a + 1, y1 = b;
        if (abs ((double) x / y - (double) x1 / y1) < abs ((double) x / y - (double) a / b)) {
            a = x1;
            b = y1;
        }
    }
    cout << a << "/" << b << endl;
}
