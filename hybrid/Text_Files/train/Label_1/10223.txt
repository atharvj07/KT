#include <iostream>
using namespace std;

int main() {
  long long t;
  cin >> t;
  for (long long i = 0; i < t; i++) {
    long long p, q;
    cin >> p >> q;
    if (p % q == 0)
      cout << p << endl;
    else if (p % 2 == 0 && q % 2 == 0)
      cout << p / 2 << endl;
    else if (p % 2 == 0)
      cout << p << endl;
    else {
      int flag = 0;
      for (long long j = 3; p > 1; j += 2) {
        if (p % j == 0) {
          while (p % j == 0) {
            p = p / j;
            if (j == q) {
              flag = 1;
              break;
            }
          }
        }
        if (flag == 1) break;
      }
      if (p > 1 && q != p) cout << p << endl;
      else
        cout << 1 << endl;
    }
  }
  return 0;
}
