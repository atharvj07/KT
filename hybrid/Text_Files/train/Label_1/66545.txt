def solve(n, k):
    answer = []
    for i in range((k+1)//2, k):
        answer.append(i)
    for i in range(k+1, n+1):
        answer.append(i)
    return answer

T = int(input())
for _ in range(T):
    n, k = map(int, input().split())
    result = solve(n, k)
    print(len(result))
    print(' '.join(map(str, result)))
