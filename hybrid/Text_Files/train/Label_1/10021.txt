#include <iostream>
#include <set>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  set<pair<long long, long long>> bills;
  vector<bool> ans;
  for (int i = 0; i < n; i++) {
    char type;
    long long x, y;
    cin >> type >> x >> y;
    if (type == '+')
      bills.insert({x, y});
    else {
      bool ok = true;
      for (auto bill : bills) {
        if (!(bill.first <= x && bill.second <= y) &&
            !(bill.second <= x && bill.first <= y)) {
          ok = false;
          break;
        }
      }
      ans.push_back(ok);
    }
  }
  for (auto a : ans) {
    cout << (a ? "YES" : "NO") << endl;
  }
  return 0;
}
