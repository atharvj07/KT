import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.IOException;

public class Main {
    public static void main(String[] args) throws IOException {
        InputStream inputStream = System.in;
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(inputStream));
        String[] boardLine = bufferedReader.readLine().split(" ");
        int b1 = Integer.parseInt(boardLine[0]);
        int a1 = Integer.parseInt(boardLine[1]);
        String[] pictureLine1 = bufferedReader.readLine().split(" ");
        int b2 = Integer.parseInt(pictureLine1[0]);
        int a2 = Integer.parseInt(pictureLine1[1]);
        String[] pictureLine2 = bufferedReader.readLine().split(" ");
        int b3 = Integer.parseInt(pictureLine2[0]);
        int a3 = Integer.parseInt(pictureLine2[1]);
        System.out.println(
            (b1 >= a2 + a3 && a1 >= Math.max(b2, b3)) ||
            (b1 >= b2 + b3 && a1 >= Math.max(a2, a3)) ? "YES" : "NO"
        );
    }
}
