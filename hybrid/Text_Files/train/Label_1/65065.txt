n = int(input())
lst = list(map(int, input().split()))
lst_sum = sum(lst)
lf = lst[0]
if lf == lst[-1]:
    print(lst_sum, 0)
else:
    lst.sort()
    lst.reverse()
    sum_ = 0
    cnt = 0
    new_lst = []
    for i in lst:
        if i != lf:
            sum_ += i
            cnt += 1
            new_lst.append(cnt)
        else:
            break
    print(lst_sum - sum_, cnt + 1)
    print(*new_lst)
