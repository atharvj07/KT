#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    vector<vector<int>>a(n+1);
    for(int i=1;i<=n*k;i++){
        int x;
        cin>>x;
        a[x].push_back(i);
    }
    for(int col=1;col<=n;col++){
        vector<int>&cur=a[col];
        vector<int>ans;
        int l=1,r=l+k-1;
        while(l<=n*k){
            if((int)cur.size()<=1){
                l+=k;
                r=l+k-1;
                continue;
            }
            bool found=false;
            for(int i=0;i+1<(int)cur.size();i++){
                if(cur[i]<=l&&r<=cur[i+1]){
                    found=true;
                    break;
                }
            }
            if(!found){
                l+=k;
                r=l+k-1;
                continue;
            }
            ans.push_back(l);
            ans.push_back(r);
            cur.erase(cur.begin());
            cur.erase(cur.begin());
            l=r+1;
            r=l+k-1;
        }
        for(int i=0;i<(int)ans.size();i+=2){
            cout<<ans[i]<<' '<<ans[i+1]<<endl;
        }
    }
}
