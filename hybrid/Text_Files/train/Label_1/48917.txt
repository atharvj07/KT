import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        ArrayList<Integer>[] tree = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            tree[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            String[] edge = br.readLine().split(" ");
            int a = Integer.parseInt(edge[0]);
            int b = Integer.parseInt(edge[1]);
            tree[a].add(b);
            tree[b].add(a);
        }
        int[] depth = new int[n + 1];
        boolean[] visited = new boolean[n + 1];
        dfs(1, 0, depth, visited, tree);
        int[] maxDepth = new int[n + 1];
        Arrays.fill(maxDepth, -1);
        for (int i = 1; i <= n; i++) {
            dfs2(i, -1, maxDepth, depth, tree);
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = Math.max(ans, maxDepth[i]);
        }
        System.out.println(ans);
    }

    public static void dfs(int node, int parent, int[] depth, boolean[] visited, ArrayList<Integer>[] tree) {
        visited[node] = true;
        depth[node] = depth[parent] + 1;
        for (int next : tree[node]) {
            if (!visited[next]) {
                dfs(next, node, depth, visited, tree);
            }
        }
    }

    public static void dfs2(int node, int parent, int[] maxDepth, int[] depth, ArrayList<Integer>[] tree) {
        maxDepth[node] = depth[node];
        for (int next : tree[node]) {
            if (next != parent) {
                dfs2(next, node, maxDepth, depth, tree);
                maxDepth[node] = Math.max(maxDepth[node], maxDepth[next]);
            }
        }
    }
}
