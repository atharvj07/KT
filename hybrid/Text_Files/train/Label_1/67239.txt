from bisect import bisect_right
n,k,x=map(int,input().split())
a=list(map(int,input().split()))
a.append(0)
s=[0]
for i in a:s.append(i+s[-1])
ans=-1
for i in range(n-k+1):
    if x<=i or x>n-i:continue
    p=bisect_right(s,s[i+k]-s[i],lo=i+1,hi=n-x+i+1)
    if p-i-1>=x:ans=max(ans,s[p]-s[i])
print(ans)
