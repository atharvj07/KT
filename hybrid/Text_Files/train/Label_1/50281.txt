import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        sc.close();
        System.out.println(isPrime(n) && isPrime(m) && n < m && isNextPrime(n, m) ? "YES" : "NO");
    }

    private static boolean isNextPrime(int n, int m) {
        boolean[] sieve = new boolean[m + 1];
        for (int i = 2; i <= m; i++) {
            sieve[i] = true;
        }
        for (int i = 2; i * i <= m; i++) {
            if (sieve[i]) {
                for (int j = i + i; j <= m; j += i) {
                    sieve[j] = false;
                }
            }
        }
        for (int i = n + 1;; i++) {
            if (sieve[i]) {
                return i == m;
            }
        }
    }

    private static boolean isPrime(int n) {
        if (n <= 1) {
            return false;
        }
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
}
