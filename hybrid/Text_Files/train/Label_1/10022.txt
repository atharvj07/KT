#include <bits/stdc++.h>
using namespace std;

const int N = 1e5+5;
int n, m;
unordered_map<string, string> mp;
unordered_map<string, int> dist;
queue<string> q;
vector<string> s;
vector<int> rcnt;

int main(){
    cin >> m;
    for(int i=0; i<m; i++){
        string x;
        cin >> x;
        for(char &ch : x) ch = tolower(ch);
        s.push_back(x);
        rcnt.push_back(count(x.begin(), x.end(), 'r'));
    }
    cin >> n;
    for(int i=0; i<n; i++){
        string a, b;
        cin >> a >> b;
        for(char &ch : a) ch = tolower(ch);
        for(char &ch : b) ch = tolower(ch);
        mp[a] = b;
    }
    int ans1 = 1e9;
    int ans2 = 1e9;
    q.push(s[0]);
    dist[s[0]] = rcnt[0];
    while(!q.empty()){
        string u = q.front();
        q.pop();
        ans1 = min(ans1, dist[u]);
        ans2 = min(ans2, dist[u]+int(s.size())-1);
        for(int i=0; i<m; i++){
            if(s[i] == u) continue;
            string v = s[i];
            if(dist.find(v) == dist.end()){
                q.push(v);
                if(mp.find(u) != mp.end() && mp[u] == v){
                    dist[v] = dist[u];
                }
                else dist[v] = dist[u]+rcnt[i];
            }
        }
    }
    cout << ans1 << " " << ans2;
    return 0;
}
