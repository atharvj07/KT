#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int32_t main() {
  int n;
  cin >> n;
  int q[N], pref[N], ans[N], cnt[N];
  for (int i = 0; i < n; i++) cnt[i] = 0;
  for (int i = 0; i < n - 1; i++) { cin >> q[i]; }
  int mn = n, mx = -1;
  for (int i = 0; i < n - 1; i++) {
    mn = min(mn, q[i]);
    mx = max(mx, q[i]);
  }
  if (abs(mn) > 1 || mx > n || mn < -n) {
    cout << -1 << endl;
    return 0;
  }
  for (int i = 0; i < n - 1; i++) {
    cnt[q[i] + n]++;
    cnt[-q[i] + n]++;
  }
  if (*max_element(cnt + n, cnt + n + 2 * n) > 2) {
    cout << -1 << endl;
    return 0;
  }
  set<int> st;
  for (int i = 0; i < 2 * n; i++) {
    if (cnt[i] == 1) st.insert(i - n);
  }
  int oth = 0;
  for (int i = 1; i <= n; i++) {
    if (st.find(i) != st.end()) {
      oth = i;
      st.erase(i);
      break;
    }
  }
  pref[0] = oth;
  for (int i = 1; i < n; i++) {
    pref[i] = pref[i - 1] + q[i - 1];
  }
  if (*min_element(pref, pref + n) < 1 || *max_element(pref, pref + n) > n) {
    cout << -1 << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) { ans[i] = pref[i]; }
  sort(ans, ans + n);
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
