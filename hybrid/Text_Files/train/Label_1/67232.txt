n = int(input())
a = list(map(int, input().split()))
even = [-1] * n
odd = [-1] * n
for i in range(n):
    if i - a[i] >= 0:
        if a[i] % 2 == a[i - a[i]] % 2:
            even[i] = 1
            odd[i] = 1
        else:
            if a[i] % 2 == 0:
                even[i] = 1
            else:
                odd[i] = 1
    if i + a[i] < n:
        if a[i] % 2 == a[i + a[i]] % 2:
            even[i] = 1
            odd[i] = 1
        else:
            if a[i] % 2 == 0:
                even[i] = 1
            else:
                odd[i] = 1
for i in range(1, n):
    if a[i] % 2 == 0:
        if odd[i - 1] != -1:
            odd[i] = odd[i - 1] + 1
        if even[i - a[i]] != -1:
            even[i] = even[i - a[i]] + 1
    else:
        if even[i - 1] != -1:
            even[i] = even[i - 1] + 1
        if odd[i - a[i]] != -1:
            odd[i] = odd[i - a[i]] + 1
print(*even)
