#include <bits/stdc++.h>
using namespace std ;
const int N = 100005 ;

int32_t main() {
    int n ; cin >> n ;
    if(n == 2) return cout << -1, 0 ;
    vector<int> a, b, c ;
    for(int i = n - 1 ; i >= 0 ; --i) {
        c.push_back(i) ;
    }
    int half = (n + 1) / 2 ;
    for(int i = 0 ; i < half ; ++i) {
        a.push_back(2 * i) ;
        b.push_back(2 * i + 1) ;
    }
    if(n % 2) {
        b.push_back(n - 1) ;
    }
    for(auto x : c) cout << x << ' ' ; cout << '\n' ;
    for(auto x : a) cout << x << ' ' ; cout << '\n' ;
    for(auto x : b) cout << x << ' ' ; cout << '\n' ;
    return 0 ;
}
