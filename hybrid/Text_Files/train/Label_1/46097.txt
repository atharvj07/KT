import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        int[] count = new int[n];
        int[] last = new int[26];
        Arrays.fill(last, -1);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int c = s.charAt(i) - 'A';
            if (i > 0 && s.charAt(i) == s.charAt(i - 1)) {
                count[i] = count[i - 1] + 1;
            }
            ans += count[i] + (last[c] >= 0 ? i - last[c] : 1);
            last[c] = i;
        }
        System.out.println(ans);
    }
}
