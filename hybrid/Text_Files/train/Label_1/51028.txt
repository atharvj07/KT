import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.next();
    int ans = 0;
    for (int i = 0; i < s.length(); i++) {
      if (s.charAt(i) >= '0' && s.charAt(i) <= '9') {
        ans += s.charAt(i) - '0';
      } else if (s.charAt(i) >= 'a' && s.charAt(i) <= 'z') {
        ans -= s.charAt(i) - 'a' + 1;
      } else if (s.charAt(i) >= 'A' && s.charAt(i) <= 'Z') {
        ans += s.charAt(i) - 'A' + 1;
      }
    }
    System.out.println(ans);
  }
}
