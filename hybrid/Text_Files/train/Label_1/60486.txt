from collections import Counter

n = int(input())
s = input()
c = Counter(s)
odd = []
even = []
for k, v in c.items():
    if v % 2 == 1:
        odd.append(k)
    else:
        even.extend([k] * (v // 2))
ans = len(odd)
if len(odd) > 0:
    even.append(odd.pop())
print(ans)
print(' '.join(''.join(x) for x in zip(*[even] * ans)))
