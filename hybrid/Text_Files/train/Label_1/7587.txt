#include <iostream>
#include <queue>

using namespace std;

struct Video {
  long long timestamp;
  long long duration;
  bool operator<(const Video& other) const {
    return timestamp < other.timestamp;
  }
};

int main() {
  long long n, k;
  cin >> n >> k;

  priority_queue<Video> videos;
  for (long long i = 0; i < n; i++) {
    long long timestamp, duration;
    cin >> timestamp >> duration;
    videos.push({timestamp, duration});
  }

  long long current_timestamp = 0;
  priority_queue<long long, vector<long long>, greater<long long>> servers;
  for (long long i = 0; i < k; i++) {
    servers.push(0);
  }

  while (!videos.empty()) {
    Video video = videos.top();
    videos.pop();

    while (!servers.empty() && servers.top() <= video.timestamp) {
      servers.pop();
    }

    if (servers.empty()) {
      servers.push(video.timestamp);
    } else {
      long long server_timestamp = servers.top();
      servers.pop();
      servers.push(server_timestamp + video.duration);
    }

    current_timestamp = max(current_timestamp, video.timestamp + video.duration);
  }

  while (!servers.empty()) {
    current_timestamp = max(current_timestamp, servers.top() + 1);
    servers.pop();
  }

  cout << current_timestamp << endl;

  return 0;
}
