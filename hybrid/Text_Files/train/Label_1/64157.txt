m, t, r = map(int, input().split())
w = list(map(int, input().split()))
ans = -1
for i in range(w[0]-1-t+1):
  lst = [i+1]
  cnt = 0
  for j in w:
    cnt+=j < i+1 or (i+1<=j<i+1+t)
  if cnt>=r:
    ans = max(ans,len(lst))
  nxt=i+1
  while nxt < w[-1]:
    for j in range(nxt, nxt+t-1):
      cnt-=j < i+1 or (i+1<=j<i+1+t)
    for j in w:
      cnt+=j < nxt+1 or (nxt+1<=j<nxt+1+t)
    if cnt>=r:
      ans = max(ans,len(lst))
      lst.append(nxt+1)
    nxt+=1
print(ans)
