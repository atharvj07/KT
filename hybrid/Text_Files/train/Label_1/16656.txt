#include<bits/stdc++.h>
using namespace std;
const int MAXN = 3e3 + 10;
const int INF = 2e9;
int a[MAXN];
int b[MAXN];
int f[MAXN][MAXN];
int main() {
    int n, v;
    cin >> n >> v;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= v; j++) {
            f[i][j] = -INF;
        }
    }
    f[0][0] = 0;
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= v; j++) {
            if (f[i][j] == -INF) continue;
            // not collect
            f[i + 1][j] = max(f[i + 1][j], f[i][j]);

            // collect in day a[i]
            f[i + 1][min(v, j + b[i])] = max(f[i + 1][min(v, j + b[i])], f[i][j] + min(v, b[i]));

            // collect in day a[i] + 1
            if (i + 1 <= n) f[i + 2][min(v, j + b[i])] = max(f[i + 2][min(v, j + b[i])], f[i][j] + min(v, b[i]));
        }
    }
    int ans = 0;
    for (int i = n; i <= n + 1; i++) {
        for (int j = 0; j <= v; j++) {
            ans = max(ans, f[i][j]);
        }
    }
    cout << ans << endl;
    return 0;
}
