#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 4 == 0) {
      cout << "YES" << endl;
      for (int i = 1; i <= n / 2; i += 2) {
        cout << i + n / 2 - 1 << ' ' << i << ' ';
      }
      cout << endl;
    } else {
      cout << "NO" << endl;
    }
  }
}
