#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    int ans=0,pos=0;
    for(int i=0;i<s.size();i++){
        if(s[i]=='a'){
            pos=0;
            continue;
        }
        ans+=min(abs(s[i]-pos),26-abs(s[i]-pos));
        pos=s[i]-'a';
    }
    cout<<ans<<endl;
}
