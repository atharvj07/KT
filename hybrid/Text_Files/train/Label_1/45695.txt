import java.util.Arrays;
import java.util.Scanner;

public class VovaAndTests {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[k];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        Arrays.sort(marks);

        int sum = Arrays.stream(marks).sum();
        int remainingTests = n - k;
        int minSum = x - sum;
        int minMedian = y * remainingTests / 2;

        if (minSum > p * remainingTests || minMedian > p) {
            System.out.println("-1");
            return;
        }

        int[] result = new int[remainingTests];
        int j = 0;
        for (int i = 0; i < remainingTests; i++) {
            if (j < k && marks[j] < minMedian) {
                result[i] = marks[j];
                j++;
            } else {
                result[i] = minMedian;
            }
            minMedian++;
        }

        System.out.println(Arrays.toString(result).replace("[", "").replace("]", ""));
    }
}
