#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const double pi = 3.14159265358979323846;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    for (auto& [x, y] : a) {
        cin >> x >> y;
    }
    pair<int, int> ans = {-1, -1};
    double min_angle = 2 * pi;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double angle = acos(
                (a[i].first * a[j].first + a[i].second * a[j].second) /
                sqrt(pow(a[i].first, 2) + pow(a[i].second, 2)) /
                sqrt(pow(a[j].first, 2) + pow(a[j].second, 2)));
            if (angle < min_angle) {
                min_angle = angle;
                ans = {i + 1, j + 1};
            }
        }
    }
    cout << ans.first << " " << ans.second << endl;
    return 0;
}
