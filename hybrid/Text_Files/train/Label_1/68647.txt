import sys

t=int(input())

for _ in range(t):
    n,q=map(int,input().split())

    a=[1 if c=='+' else -1 for c in input()]
    s=[0]*n

    for i in range(n):
        s[i]=a[i]+(s[i-1] if i>0 else 0)

    for _ in range(q):
        l,r=map(int,input().split())
        l-=1
        r-=1
        ans=abs(s[r]-s[l-1] if l>0 else s[r])
        print(ans if ans else 0)
