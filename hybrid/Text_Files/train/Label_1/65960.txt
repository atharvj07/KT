import sys

input = sys.stdin.readline
for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    b=[0]*27
    ans=0
    for i in a:
        b[i]+=1
    for i in range(1,27):
        if b[i]:
            ans+=1
    x=ans
    for i in range(1,27):
        for j in range(1,27):
            if i==j:
                if b[i]>=2:
                    ans=max(ans,x+b[i]-1)
            else:
                if b[i] and b[j]:
                    ans=max(ans,x+1)
    for i in range(1,27):
        if b[i]:
            x=0
            for j in range(1,27):
                if i==j:
                    if b[i]>=2:
                        x=max(x,b[i]-1)
                else:
                    if b[j]:
                        x=max(x,b[j])
            ans=max(ans,x+1)
    print(ans)
