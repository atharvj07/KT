#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  long long n, a, b, c;
  cin >> n >> a >> b >> c;
  long long ans = n / 4 * a + min({n % 4 * a, b, c});
  if (n % 4 != 0) {
    ans = min(ans, 2 * a + min(c, 2 * b));
  }
  cout << ans << endl;
  return 0;
}
