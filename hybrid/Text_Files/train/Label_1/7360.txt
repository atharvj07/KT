#include <iostream>
using namespace std;
 
int main() {
  long long n, sx, sy, dx, dy, t, x = 0, y = 0;
  cin >> n >> sx >> sy >> dx >> dy >> t;
  for (; t > 0; t--, dx += x, dy += y) {
    x += dx, y += dy;
    x = (x % n + n) % n;
    y = (y % n + n) % n;
  }
  cout << (x + 1) << ' ' << (y + 1) << endl;
}
