#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.rbegin(), a.rend());
    int sum = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (i < k) {
            sum += a[i];
            ans = max(ans, sum);
        } else
            sum += a[i] / 2;
    }
    cout << ans << endl;
    return 0;
}
