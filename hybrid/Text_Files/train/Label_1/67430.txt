n, m, k = map(int, input().split())
a = sorted(list(map(int, input().split())))
b, c = [0 for _ in range(k)], a[-k:]
for i in range(k - 1, -1, -1):
	b[i] = c[i] + (c[i + 1] if i + 1 < k else 0)
	if i > 0:
		c[i - 1] = max(c[i - 1], b[i] + a[-k + i])
ans = b[0]
for i in range(n - k):
	ans += a[i]
	ans = max(ans, c[k - 1] + sum(a[i: i + k]))
print(ans)
