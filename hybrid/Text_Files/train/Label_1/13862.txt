#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ll n;
    cin >> n;
    vector<ll> v(n), ans;
    ll sum = 0;
    for (ll i = 0; i < n; i++) {
        cin >> v[i];
        sum += v[i];
    }
    for (ll i = 0; i < n; i++) {
        if (v[i] * (n - 1) == sum - v[i]) ans.push_back(i + 1);
    }
    cout << ans.size() << endl;
    for (auto x : ans) cout << x << " ";
}
