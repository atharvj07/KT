#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;

  int cnt0 = count(s.begin(), s.end(), '0');
  int cnt1 = count(s.begin(), s.end(), '1');

  if (cnt0 + cnt1 != t.length()) {
    cout << 0 << endl;
    return 0;
  }

  int res = 0;
  for (char c = 'a'; c <= 'z'; c++) {
    string t0 = "", t1 = "";
    int cnt00 = 0, cnt11 = 0;
    for (char d : t) {
      if (cnt00 < cnt0 && d == c) t0 += c, cnt00++;
      else t0 += ' ';
      if (cnt11 < cnt1 && d != c) t1 += d, cnt11++;
      else t1 += ' ';
    }
    if (cnt00 == cnt0 && cnt11 == cnt1) {
      bool ok = true;
      for (char d : t) {
        if (ok && d == c) continue;
        else if (ok && d != c) ok = false;
        else if (!ok && d != c) continue;
        else if (!ok && d == c) {
          ok = false;
          break;
        }
      }
      if (ok) res++;
    }
  }

  cout << res << endl;

  return 0;
}
