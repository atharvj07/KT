def can_form_identity(n, m, table):
    # Check if without any operation, the table is already in the desired state.
    if all(row == list(range(1, m + 1)) for row in table):
        return "YES"
    
    # If we need to swap columns, we need to identify which columns to swap.
    # We do this by checking if there's exactly one pair of columns that are out of place across all rows.
    for i in range(m):
        for j in range(i + 1, m):
            # Try swapping columns i and j
            valid_swap = True
            for row in table:
                # Check if swapping columns i and j in this row would sort the row
                if not (row[i] == row[j] or (row[:i] + [row[j]] + row[i+1:j] + [row[i]] + row[j+1:] == sorted(row))):
                    valid_swap = False
                    break
            if valid_swap:
                return "YES"
    return "NO"

n, m = map(int, input().split())
table = [list(map(int, input().split())) for _ in range(n)]
print(can_form_identity(n, m, table))
