import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[] t = new int[q];
        int c = 0;
        for (int i = 0; i < q; i++) {
            t[i] = sc.nextInt();
            for (int j = 0; j < n; j++) {
                if (t[i] == s[j]) {
                    c++;
                    break;
                }
            }
        }
        System.out.println(c);
    }
}
