def gcd(x, y):
    while(y):
        x, y = y, x % y
    return x

def lcm(x, y):
    return x * y // gcd(x, y)

def find_lcm(s, t):
    if s == t:
        return s
    lcm_length = lcm(len(s), len(t))
    lcm_s = s * (lcm_length // len(s))
    lcm_t = t * (lcm_length // len(t))
    if lcm_s == lcm_t:
        return lcm_s
    else:
        return -1

q = int(input().strip())
for _ in range(q):
    s = input().strip()
    t = input().strip()
    print(find_lcm(s, t))
