import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = {x for x in a}
    for i in range(n):
        for j in range(i + 1, n):
            if (a[i] % a[j]) not in b:
                print(a[i], a[j])
                b.add(a[i] % a[j])
                break
