import java.util.*;

public class Main {
    static int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int r = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] left = new int[n];
        int[] right = new int[n];
        for (int i = 0; i < n; i++) {
            left[i] = Math.max(0, i - r);
            right[i] = Math.min(n - 1, i + r);
        }
        long[][] dp = new long[n][k + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = -1;
            }
        }
        System.out.println(solve(0, k, a, left, right, dp));
    }

    static long solve(int idx, int k, int[] a, int[] left, int[] right, long[][] dp) {
        if (idx == a.length) {
            return 0;
        }
        if (dp[idx][k] != -1) {
            return dp[idx][k];
        }
        long ans = Long.MAX_VALUE;
        for (int i = 0; i <= k; i++) {
            long sum = a[idx] + i;
            for (int j = left[idx]; j <= right[idx]; j++) {
                sum += Math.min(i, a[j]);
            }
            ans = Math.min(ans, sum + solve(idx + 1, k - i, a, left, right, dp));
        }
        dp[idx][k] = ans;
        return ans;
    }
}
