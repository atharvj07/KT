import java.util.Scanner;

public class BipartitePerfectMatching {

    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[][] probabilities = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                probabilities[i][j] = scanner.nextInt() * 10000;
            }
        }

        long[][] dp = new long[1 << n][n];
        for (int mask = 0; mask < (1 << n); mask++) {
            for (int last = 0; last < n; last++) {
                if ((mask & (1 << last)) == 0) {
                    continue;
                }
                if (mask == (1 << last)) {
                    dp[mask][last] = 1;
                } else {
                    for (int prev = 0; prev < n; prev++) {
                        if ((mask & (1 << prev)) != 0) {
                            dp[mask][last] = (dp[mask][last] + dp[mask ^ (1 << last)][prev] * probabilities[prev][last] / 10000) % MOD;
                        }
                    }
                }
            }
        }

        long numerator = 0;
        for (int last = 0; last < n; last++) {
            numerator = (numerator + dp[(1 << n) - 1][last]) % MOD;
        }
        long denominator = 1;
        for (int i = 1; i <= n; i++) {
            denominator = (denominator * i) % MOD;
        }
        long inverseDenominator = modInverse(denominator, MOD);
        long answer = (numerator * inverseDenominator) % MOD;
        System.out.println(answer);
    }

    static long modInverse(long a, long m) {
        long m0 = m;
        long y = 0, x = 1;
        if (m == 1) return 0;
        while (a > 1) {
            long q = a / m;
            long t = m;
            m = a % m;
            a = t;
            t = y;
            y = x - q * y;
            x = t;
        }
        if (x < 0) x += m0;
        return x;
    }
}
