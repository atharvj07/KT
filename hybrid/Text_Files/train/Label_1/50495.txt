import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        
        String[] s = br.readLine().split("\\s");
        int n = Integer.parseInt(s[0]), m = Integer.parseInt(s[1]), x = Integer.parseInt(s[2]);
        
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            s = br.readLine().split("");
            for (int j = 0; j < m; j++) {
                grid[i][j] = s[j].charAt(0);
            }
        }
        
        int dist[][][] = new int[n][m][x + 1];
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k <= x; k++) {
                    dist[i][j][k] = -1;
                }
            }
        }
        
        int ans = solve(0, 0, 0, x, grid, dist);
        System.out.println(ans < 0 ? "-1" : ans);
    }
    
    static int solve(int i, int j, int cnt, int x, char[][] grid, int[][][] dist) {
        if (i < 0 || i >= grid.length || j < 0 || j >= grid[0].length) {
            return -1;
        }
        
        if (dist[i][j][cnt] != -1) {
            return dist[i][j][cnt];
        }
        
        char ch = grid[i][j];
        if (ch != 'S' && ch != Character.toLowerCase(ch) && cnt == 0) {
            return -1;
        }
        
        int res = Integer.MAX_VALUE;
        
        if (cnt > 0) {
            if (ch >= 'a' && ch <= 'z') {
                res = Math.min(res, solve(i, j, cnt - 1, x, grid, dist));
            } else {
                res = Math.min(res, solve(i, j, 0, x, grid, dist));
            }
        }
        
        for (int dx = -x; dx <= x; dx++) {
            for (int dy = -x; dy <= x; dy++) {
                if (dx == 0 && dy == 0) {
                    continue;
                }
                
                int nx = i + dx, ny = j + dy;
                if (nx >= 0 && nx < grid.length && ny >= 0 && ny < grid[0].length) {
                    char nch = grid[nx][ny];
                    
                    if (Character.toLowerCase(nch) == Character.toLowerCase(ch) && nch != 'S') {
                        res = Math.min(res, 1 + solve(nx, ny, cnt, x, grid, dist));
                    } else if (nch == 'S') {
                        res = Math.min(res, solve(nx, ny, cnt, x, grid, dist));
                    }
                }
            }
        }
        
        if (res == Integer.MAX_VALUE) {
            res = -1;
        }
        
        return dist[i][j][cnt] = res;
    }
}
