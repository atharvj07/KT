#include <cstdio>
#include <vector>
#include <algorithm>
#define ll long long
constexpr ll mod = 998244353;

ll qpow(ll base, ll pow) {
  ll ans = 1;
  while (pow) {
    if (pow & 1) ans = ans * base % mod;
    base = base * base % mod;
    pow >>= 1;
  }
  return ans;
}

int main() {
  ll n, m, l, r;
  scanf("%lld%lld%lld%lld", &n, &m, &l, &r);
  if (n == 1 && m == 1) {
    puts("1");
    return 0;
  }
  printf("%lld\n", ((qpow(2, n + m - 2) - 2 + mod) % mod) * ((qpow(2, n + m - 2) + 2 * (r - l) % mod + mod) % mod) % mod);
  return 0;
}
