#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ll n, k;
  cin >> n >> k;
  vector<int> c(n * k);
  for (auto &x : c) { cin >> x; }
  vector<ll> f(n);
  for (auto &x : f) { cin >> x; }
  vector<ll> h(k);
  for (auto &x : h) { cin >> x; }
  vector<map<ll, ll>> cd(n);
  for (ll i = 0; i < n * k; i++) { cd[i % n][c[i]]++; }
  ll ans = 0;
  for (auto &fav : f) {
    vector<ll> cnt(k + 1, 0);
    for (auto &p : cd) {
      if (p.count(fav)) {
        for (ll i = 0; i <= k; i++) {
          if (p[fav] >= i) { cnt[i]++; }
        }
      }
    }
    for (ll i = k; i >= 0; i--) {
      if (cnt[i] <= 0) { cnt[i] = 0; }
      else { cnt[i] = min(cnt[i], i); }
      ans += cnt[i] * h[i];
    }
  }
  cout << ans << endl;
  return 0;
}
