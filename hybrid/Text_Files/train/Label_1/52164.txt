import java.util.Arrays;
import java.util.Scanner;

public class Coke {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        int minVolume = -1;
        for (int i = 0; i < k; i++) {
            if (a[i] >= n) {
                minVolume = 1;
                break;
            }
            for (int j = i + 1; j < k; j++) {
                if (a[i] + a[j] == n) {
                    minVolume = 2;
                    break;
                } else if (a[i] + a[j] > n) {
                    break;
                }
            }
        }

        System.out.println(minVolume);
    }
}
