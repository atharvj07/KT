n = int(input())
s = input()
x, y = map(int, input().split())

def min_length(n, s, x, y):
    if abs(x) + abs(y) > n or (n - abs(x) - abs(y)) % 2:
        return -1
    dx = dy = 0
    for c in s:
        if c == 'U': dy += 1
        elif c == 'D': dy -= 1
        elif c == 'L': dx -= 1
        elif c == 'R': dx += 1
    if dx == x and dy == y:
        return 0
    
    ans = n
    l = 0
    for r in range(n):
        if s[r] == 'U': dy -= 1
        elif s[r] == 'D': dy += 1
        elif s[r] == 'L': dx += 1
        elif s[r] == 'R': dx -= 1
        while abs(x - dx) + abs(y - dy) <= r - l:
            if s[l] == 'U': dy += 1
            elif s[l] == 'D': dy -= 1
            elif s[l] == 'L': dx -= 1
            elif s[l] == 'R': dx += 1
            l += 1
        if abs(x - dx) + abs(y - dy) <= r - l + 1:
            ans = min(ans, r - l + 1)
    return ans

print(min_length(n, s, x, y))
