#include <iostream>
#include <vector>

using namespace std;

vector<int> solve(vector<int> const& a, vector<int> const& l) {
    int n = a.size();
    int m = l.size();
    vector<int> ans(m);
    vector<int> cnt(100001, 0);
    int distinct_count = 0;
    for (int i = n - 1; i >= 0; i--) {
        int cur = a[i];
        if (cnt[cur] == 0) {
            distinct_count++;
        }
        cnt[cur]++;
        for (int j = 0; j < m; j++) {
            if (l[j] <= i + 1) {
                ans[j] = distinct_count;
            }
        }
    }
    return ans;
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int& x : a) {
        cin >> x;
    }
    vector<int> l(m);
    for (int& x : l) {
        cin >> x;
        x--;
    }
    vector<int> ans = solve(a, l);
    for (int x : ans) {
        cout << x << endl;
    }
    return 0;
}
