import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        long ans = 0;
        int l = 1;
        int r = n;
        for (int i = 0; i < m; i++) {
            if (Math.abs(a[i] - l) < Math.abs(a[i] - r)) {
                ans += Math.abs(a[i] - l);
                l = a[i];
            } else {
                ans += Math.abs(a[i] - r);
                r = a[i];
            }
        }
        System.out.println(ans);
    }
}
