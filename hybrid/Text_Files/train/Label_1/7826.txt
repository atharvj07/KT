#include<bits/stdc++.h>
using namespace std;
int main()
{
    string a,b;
    cin>>a>>b;
    int n=a.size(),m=b.size(),i,j,k=0,ans=0;
    for(i=0,j=0;i<n;++i)
    {
        if(a[i]==b[j])
        {
            ++j,++k;
            if(j==m)
            {
                j=0;
                ans=max(ans,m+i+1-k);
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
