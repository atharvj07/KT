import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long a[] = new long[n + 1];
        long b[] = new long[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += (b[i] - a[i]);
        }
        if (sum % 2 == 0)
            System.out.println("YES");
        else
            System.out.println("NO");
    }
}
