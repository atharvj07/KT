import sys
input = sys.stdin.readline

n = int(input())
a = [20] * n
for i in range(1, n):
    for j in range(i - 1, -1, -1):
        if a[i] > a[j] + 20:
            a[i] = a[j] + 20
        if a[i] > a[j] + 50 and a[i] - a[j] >= 90:
            a[i] = a[j] + 50
        if a[i] > a[j] + 120 and a[i] - a[j] >= 1440:
            a[i] = a[j] + 120
for i in range(n):
    b = int(input())
