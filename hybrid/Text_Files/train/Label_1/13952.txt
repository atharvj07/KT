#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007

using namespace std;

ll binpow(ll x, ll p) {
    if (!p) return 1;
    if (p & 1) return (x * binpow(x, p - 1)) % mod;
    ll t = binpow(x, p / 2);
    return (t * t) % mod;
}

int main() {
    ll n, k;
    cin >> n >> k;
    ll arr[n], sum = 0, ans = 0;
    for (int i = 0; i < n; i++) cin >> arr[i], sum += arr[i];
    ans = (sum * binpow(n - 1, k - 1)) % mod;
    cout << ans << endl;
    return 0;
}
