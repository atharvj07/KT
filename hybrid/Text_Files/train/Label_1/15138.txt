#include <bits/stdc++.h>
#define ll long long

using namespace std;

ll nines(char x[], int n){
    ll ans = 0;
    for (int i=0; i<n; i++){
        int cur = x[i]-'0';
        x[i] = (cur+x[i+1]-'0'-10+'0')/2+'0';
        x[i+1] = '0';
        if (x[i]>='9')x[i]=x[i]-10, ans++;
    }
    return ans;
}

int main(){
    ll ans = 0;
    char x[100000];
    int n = 0;
    while (cin>>x[n]) n++;
    for (int i=0; i<n; i++){
        ll cnt = nines(x, n);
        ans = max(ans, cnt);
        reverse(x, x+n);
    }
    cout<<ans<<endl;
    return 0;
}
