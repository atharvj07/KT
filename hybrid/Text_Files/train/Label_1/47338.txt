import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    char[] s = scanner.next().toCharArray();

    int l = -1, r = -1;
    for (int i = 0; i < n - 1; i++) {
      if (s[i] > s[i + 1]) {
        l = i;
        r = i + 1;
        break;
      }
    }

    if (l == -1) {
      System.out.println("NO");
    } else {
      System.out.println("YES");
      System.out.println((l + 1) + " " + (r + 1));
    }
  }
}
