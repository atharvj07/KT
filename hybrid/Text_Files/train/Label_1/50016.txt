import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int h = in.nextInt();
        int w = in.nextInt();
        int[][] p = new int[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                p[i][j] = in.nextInt();
            }
        }
        int[][] c = new int[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if (p[i][j] == 0) {
                    c[i][j] = 1;
                }
                if (i > 0) {
                    c[i][j] += c[i - 1][j];
                }
            }
        }
        int max = 0;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                int min = c[i][j];
                for (int k = j; k >= 0; k--) {
                    min = Math.min(min, c[i][k]);
                    max = Math.max(max, min * (j - k + 1));
                }
            }
        }
        System.out.println(max);
    }
}
