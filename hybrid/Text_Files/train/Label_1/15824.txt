#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n, k;
  cin >> n >> k;
  if (k > 2 * n - 2) cout << -1 << endl;
  else {
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    vector<long long> id1(n + 1, 0);
    vector<long long> id2(n + 1, 0);
    vector<long long> v, u;
    v.push_back(a);
    u.push_back(c);
    id1[a] = id2[c] = 1;
    int pos;
    if (a == c) {
      pos = 1;
      for (long long i = 2; i <= n; i++) {
        if (i != a && i != b && id1[i] == 0) {
          v.push_back(i);
          u.push_back(i);
          id1[i] = id2[i] = 1;
          pos = i;
          break;
        }
      }
    }
    for (long long i = 1; i <= n; i++) {
      if (i != a && i != b && id1[i] == 0) {
        if (pos == n)
          u.push_back(i);
        else {
          v.push_back(i);
          u.insert(u.begin() + pos, i);
        }
        id1[i] = 1;
      }
    }
    for (long long i = 1; i <= n; i++) {
      if (i != c && i != d && id2[i] == 0) {
        u.push_back(i);
        id2[i] = 1;
      }
    }
    for (long long i = 0; i < n; i++) cout << v[i] << " ";
    cout << endl;
    for (long long i = 0; i < n; i++) cout << u[i] << " ";
    cout << endl;
  }
  return 0;
}
