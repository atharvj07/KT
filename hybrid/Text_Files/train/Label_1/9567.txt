#include <bits/stdc++.h>
using namespace std;
#define int long long

const int M = 1e9 + 7;

int power(int b, int p) {
    int res = 1;
    while (p) {
        if (p & 1) res = res * b % M;
        b = b * b % M;
        p >>= 1;
    }
    return res;
}

void solve() {
    int n;
    cin >> n;
    string s, ss;
    cin >> s;
    for (int i = 1; i < n; ++i) ss += s;
    int ones = 0;
    for (char c : s) ones += c == 'A' || c == 'C';
    int res = power(2, n);
    if (ones % 2 == 0) res = (res + power(2, n - 1)) % M;
    cout << res << '\n';
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int t;
    t = 1;
    while (t--) solve();
    return 0;
}
