#include <algorithm>
#include <iostream>
#include <iterator>
#include <vector>

int main()
{
    const int MAXN = 1e5 + 11;
    int n, a[MAXN];

    std::cin >> n;

    for (int i = 1; i <= n; ++i)
        std::cin >> a[i];

    std::sort(a + 1, a + 1 + n);

    bool ok = false;

    if (n > 1)
    {
        ok = true;
        for (int i = 2; i <= n; ++i)
        {
            if (a[i] - a[i - 1] != a[2] - a[1])
            {
                ok = false;
                break;
            }
        }
    }

    if (ok)
    {
        std::cout << -1 << std::endl;
        return 0;
    }

    long long res[3];
    res[0] = a[1] - a[2];
    res[1] = a[2] - a[1];
    res[2] = a[3] - a[2];

    std::sort(res, res + 3);

    int m = 0;
    for (int i = 1; i <= 2; ++i)
    {
        if (res[i] != res[i - 1])
            ++m;
    }

    if (m == 1)
    {
        std::cout << 0 << std::endl;
        return 0;
    }

    long long diff = 0;

    if (m == 2)
    {
        if (res[2] - res[1] == res[1] - res[0])
            diff = res[1] - res[0];
        else
            diff = a[2] - a[1];
    }
    else
        diff = a[2] - a[1];

    if (diff != 0)
    {
        std::cout << 1 << std::endl;
        std::cout << a[2] - diff << std::endl;
    }
    else
    {
        std::cout << 3 << std::endl;
        std::cout << a[1] - diff << " " << a[1] << " " << a[1] + diff << std::endl;
    }

    return 0;
}
