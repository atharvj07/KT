#include <algorithm>
#include <iostream>
#include <string>

using namespace std;

auto main() -> int {
  string ai, phone;
  cin >> ai >> phone;
  for (long long i = 0; i + phone.size() - 1 < ai.size(); ++i) {
    if (ai.substr(i, phone.size()) == phone) {
      int cnt = 0;
      for (long long j = i; j < i + phone.size(); ++j) {
        ai[j] == phone[j - i] ? ++cnt : ai[j] = '#';
      }
      cout << cnt << '\n';
      return 0;
    }
  }
  cout << "0\n";
}
