#include <bits/stdc++.h>

typedef long long ll;
typedef long double ld;

#define rep(i, n) for (int i = 0; i < n; i++)
#define rep1(i, n) for (int i = 1; i <= n; i++)

using namespace std;

const int MAXN = 1e9 + 7;

int n, x, y, c;

int main()
{
    cin >> n >> x >> y >> c;
    x--, y--;
    int t = 0;
    --c;
    while (c)
    {
        vector<pair<int, int>> e = {{x - 1, y}, {x + 1, y}, {x, y - 1}, {x, y + 1}};
        for (auto p : e)
        {
            if (p.first < 0 || p.first >= n || p.second < 0 || p.second >= n)
                continue;
            --c;
            x = p.first;
            y = p.second;
            break;
        }
        t++;
    }
    cout << t << endl;
    return 0;
}
