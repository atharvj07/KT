#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll n, m, k;
  cin >> n >> m >> k;
  set<ll> h;
  for (int i = 0; i < m; ++i) {
    ll d;
    cin >> d;
    h.insert(d);
  }
  ll b = 1;
  for (int i = 0; i < k; ++i) {
    ll u, v;
    cin >> u >> v;
    if (u == b) b = v;
    else if (v == b) b = u;
    if (h.count(b)) b = 1;
  }
  cout << b << '\n';
}
