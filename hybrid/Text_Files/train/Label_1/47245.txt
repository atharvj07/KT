import java.util.Scanner;

public class RegularBracketSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine();
        for (int i = 0; i < t; i++) {
            String s = scanner.nextLine();
            int result = maxMoves(s);
            System.out.println(result);
        }
        scanner.close();
    }

    private static int maxMoves(String s) {
        int n = s.length();
        int[] dp = new int[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            char c = s.charAt(i);
            if (c == '(' || c == '[') {
                int j = i + 1;
                while (j < n && s.charAt(j) != ')' && s.charAt(j) != ']') {
                    j++;
                }
                if (j < n && (s.charAt(j) == ')' && c == '(' || s.charAt(j) == ']' && c == '[')) {
                    dp[i] = dp[j + 1] + 1;
                }
            }
            dp[i] = Math.max(dp[i], dp[i + 1]);
        }
        return dp[0];
    }
}
