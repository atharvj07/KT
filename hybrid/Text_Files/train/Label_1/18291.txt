#include <iostream>
#include <vector>

using namespace std;

int64_t query_type_1(int64_t i, int64_t val, vector<int64_t> &a) {
    a[i - 1] = val;
    return 0;
}
int64_t query_type_2(int64_t li, int64_t ri, vector<int64_t> &a) {
    int64_t result = 0;
    while (li <= ri) {
        result += (li * a[li - 1]) % 1000000000;
        li++;
    }
    return result % 1000000000;
}
int64_t query_type_3(int64_t li, int64_t ri, int64_t val, vector<int64_t> &a) {
    while (li <= ri) {
        a[li - 1] = (a[li - 1] + val % 1000000000) % 1000000000;
        li++;
    }
    return 0;
}

int main() {
    int64_t n, m;
    cin >> n >> m;
    vector<int64_t> a(n);
    for (int64_t i = 0; i < n; i++) { cin >> a[i]; }
    while (m--) {
        int64_t t, x, l, r, v, d;
        cin >> t;
        if (t == 1) {
            cin >> x >> v;
            query_type_1(x, v, a);
        } else if (t == 2) {
            cin >> l >> r;
            cout << query_type_2(l, r, a) << "\n";
        } else if (t == 3) {
            cin >> l >> r >> d;
            query_type_3(l, r, d, a);
        }
    }
    return 0;
}
