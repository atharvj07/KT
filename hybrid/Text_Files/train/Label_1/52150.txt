import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int[] par=new int[n+1];
        int[] c=new int[n+1];
        for(int i=1;i<=n;i++){
            par[i]=sc.nextInt();
            c[i]=sc.nextInt();
        }
        int[] a=new int[n+1];
        boolean flag=true;
        for(int i=1;i<=n;i++){
            int num=1;
            for(int j=1;j<=n;j++){
                if(j!=i&&par[j]==i){
                    num+=a[j];
                    if(a[j]>a[i]){
                        flag=false;
                    }
                }
            }
            if(num<=c[i]+1){
                a[i]=num;
            }else{
                flag=false;
            }
        }
        if(flag){
            System.out.println("YES");
            for(int i=1;i<=n;i++){
                System.out.print(a[i]+" ");
            }
        }else{
            System.out.println("NO");
        }
    }
}
