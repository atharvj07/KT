#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    int ans = n / 4;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            if (n / i / 4 == ans) {
                ans = n / i;
            } else {
                ans = 1;
                break;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < n; j++) {
            int x = (s[j / 4] >> (j % 4)) & 1;
            if ((i % ans != 0 && x != 0) || (j % ans != 0 && x != 1)) {
                ans = 1;
                break;
            }
        }
        if (ans == 1) {
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
