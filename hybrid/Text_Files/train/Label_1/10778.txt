#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<int,int>> v(n+1);
    int mx1 = -1e9, mx2 = -1e9, mn1 = 1e9, mn2 = 1e9;
    for (int i = 1; i <= n; i++) {
      int x, y;
      cin >> x >> y;
      v[i] = {x, y};
      mx1 = max(mx1, x);
      mx2 = max(mx2, y);
      mn1 = min(mn1, x);
      mn2 = min(mn2, y);
    }
    if (mx1 == mn1) {
      cout << 0 << '\n';
      continue;
    }
    if (mn2 == mx2) {
      cout << 0 << '\n';
      continue;
    }
    int ans = 1e9;
    for (int i = mn1; i < mx1; i++) {
      int cur = mx2 - mn2;
      for (int j = 1; j <= n; j++) {
        if (v[j].first > i || v[j].second < i) cur++;
      }
      ans = min(ans, cur);
    }
    for (int i = mn2; i < mx2; i++) {
      int cur = mx1 - mn1;
      for (int j = 1; j <= n; j++) {
        if (v[j].second < i || v[j].first > i) cur++;
      }
      ans = min(ans, cur);
    }
    cout << ans << '\n';
  }
}
