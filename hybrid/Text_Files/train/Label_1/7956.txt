#include <bits/stdc++.h>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    int a[4 * n];
    for (int i = 0; i < 4 * n; i++) cin >> a[i];
    sort(a, a + 4 * n);
    if (n & 1)
      cout << "NO" << endl;
    else {
      bool ok = true;
      for (int i = 0; i < 4 * n; i += 4) {
        if (a[i] != a[i + 1] || a[i + 2] != a[i + 3] ||
            a[i] + a[i + 1] != a[i + 2] + a[i + 3]) {
          ok = false;
          break;
        }
      }
      if (ok)
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
    }
  }
  return 0;
}
