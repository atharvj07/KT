#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
    int l;
    cin >> l;
    string n;
    cin >> n;
    vector<int> v;

    for (int i = 0; i < l; i++)
    {
        int x = n[i] - '0';
        v.push_back(x);
    }
    sort(v.begin(), v.end());
    int sum = 0;
    for (int i = 0; i < l - 1; i++)
    {
        sum += v[i];
    }
    cout << sum + v[l - 1] << endl;
    return 0;
}
