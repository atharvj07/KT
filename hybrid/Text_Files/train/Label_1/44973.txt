import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t > 0) {
            int x = scanner.nextInt();
            int a = 1;
            int b = x - 1;
            System.out.println(a + " " + b);
            t--;
        }
    }
}
