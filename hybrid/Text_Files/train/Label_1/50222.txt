import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> w = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            int event = scanner.nextInt();

            if (event == -1) {
                int k = 0;
                while (k < m && a[k] <= w.size()) {
                    k++;
                }
                for (int j = 0; j < k; j++) {
                    w.remove(w.size() - 1);
                }
            } else {
                w.add(event);
            }
        }

        if (w.isEmpty()) {
            System.out.println("Poor stack!");
        } else {
            for (int i : w) {
                System.out.print(i);
            }
        }
    }
}
