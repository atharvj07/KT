import java.util.*;

public class Main {
    static final int INF = 2000000000;
    static class Rect {
        // [l, r) [b, t)
        public int l, r, b, t;
        public Rect() {}
        public Rect(int l, int r, int b, int t) { this.l = l; this.r = r; this.b = b; this.t = t; }
        public boolean pointInRect(int x, int y) {
            return (l <= x && x < r) && (b <= y && y < t);
        }
        public Rect intersect(Rect other) {
            int nl = Math.max(l, other.l);
            int nr = Math.min(r, other.r);
            int nb = Math.max(b, other.b);
            int nt = Math.min(t, other.t);
            return new Rect(nl, nr, nb, nt);
        }
        public int getArea() { return (r - l) * (t - b); }
    }
    static class Point {
        public int x, y;
        public Point() {}
        public Point(int x, int y) { this.x = x; this.y = y; }
        public boolean equals(Object o) {
            if (o instanceof Point) {
                Point p = (Point)o;
                return x == p.x && y == p.y;
            } else {
                return false;
            }
        }
        public int hashCode() {
            return Objects.hash(x, y);
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Point, Rect> pt2rect = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            Point pt = new Point(x, y);
            pt2rect.put(pt, new Rect(x, x + 1, y, y + 1));
        }
        while (true) {
            Rect r = new Rect(-INF, INF, -INF, INF);
            int cnt = 0, area = 0;
            for (Map.Entry<Point, Rect> entry : pt2rect.entrySet()) {
                Rect rect = entry.getValue();
                r = r.intersect(rect);
                cnt++;
                area += rect.getArea();
            }
            int rArea = r.getArea();
            if (rArea >= area - cnt) {
                break;
            }
            Set<Point> toRemove = new HashSet<>();
            for (Map.Entry<Point, Rect> entry : pt2rect.entrySet()) {
                Point pt = entry.getKey();
                Rect rect = entry.getValue();
                if (r.pointInRect(pt.x, pt.y)) {
                    toRemove.add(pt);
                }
            }
            for (Point pt : toRemove) {
                pt2rect.remove(pt);
            }
            Set<Point> toAdd = new HashSet<>();
            for (Map.Entry<Point, Rect> entry1 : pt2rect.entrySet()) {
                Rect rect1 = entry1.getValue();
                for (Map.Entry<Point, Rect> entry2 : pt2rect.entrySet()) {
                    Point pt2 = entry2.getKey();
                    Rect rect2 = entry2.getValue();
                    Rect intersectRect = rect1.intersect(rect2);
                    if (intersectRect.getArea() > 0 && !r.pointInRect(pt2.x, pt2.y)) {
                        toAdd.add(pt2);
                    }
                }
            }
            for (Point pt : toAdd) {
                pt2rect.put(pt, new Rect(pt.x, pt.x + 1, pt.y, pt.y + 1));
            }
        }
        System.out.println(pt2rect.size());
        for (Point pt : pt2rect.keySet()) {
            System.out.println(pt.x + " " + pt.y);
        }
    }
}
