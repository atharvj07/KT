#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> a(n);
  for (auto& x : a) std::cin >> x;
  int ans = 0, last = 0;
  for (auto x : a) {
    if (x < last) ans += m - (last - x);
    last = x;
  }
  std::cout << ans << std::endl;
  return 0;
}
