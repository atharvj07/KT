import sys
input = sys.stdin.readline

n = int(input())
sherlock = input().strip()
moriarty = input().strip()

cnt = {}
for c in sherlock:
    if c not in cnt:
        cnt[c] = 0
    cnt[c] += 1

for c in moriarty:
    if c in cnt:
        cnt[c] -= 1

flicksMin = 0
for c in cnt:
    if cnt[c] > 0:
        flicksMin += cnt[c]

flicksMax = 0
for c in cnt:
    if cnt[c] < 0:
        flicksMax -= cnt[c]

print(flicksMin, flicksMax)
