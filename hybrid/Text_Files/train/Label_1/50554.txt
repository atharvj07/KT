import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int[] dp = new int[n + 1];
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1] + 1;
            for (int j = 1; j <= i / 2; j++) {
                if (s.substring(0, j).equals(s.substring(i - j, i))) {
                    dp[i] = Math.min(dp[i], dp[i - j] + (int)Math.log10(i / j) + 1);
                }
            }
        }
        System.out.println(dp[n]);
    }
}
