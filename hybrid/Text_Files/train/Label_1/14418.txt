#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;
typedef long long ll;

const ll MOD = 1e9 + 7;

ll add(ll a, ll b) {
    return ((a + b) % MOD + MOD) % MOD;
}

ll sub(ll a, ll b) {
    return (((a - b) % MOD + MOD) % MOD);
}

ll power(ll a, ll b) {
    if (b == 0) {
        return 1;
    }
    if (b == 1) {
        return a;
    }
    if (b % 2 == 0) {
        ll res = power(a, b / 2);
        return (res * res) % MOD;
    }
    return (a * power(a, b - 1)) % MOD;
}

ll inv(ll a) {
    return power(a, MOD - 2);
}

ll ncr(ll n, ll r) {
    if (n < r) {
        return 0;
    }
    ll a = 1;
    for (ll i = n; i > n - r; i--) {
        a = (a * i) % MOD;
    }
    ll b = 1;
    for (ll i = 1; i <= r; i++) {
        b = (b * i) % MOD;
    }
    b = inv(b);
    return (a * b) % MOD;
}

struct node {
    ll cnt;
    ll pref[2];
};

class segment_tree {
private:
    vector<node> tree;
    ll size;

public:
    segment_tree(ll n) {
        size = 1;
        while (size < n) {
            size *= 2;
        }
        tree.resize(2 * size);
    }

    void build(ll idx, ll l, ll r, vector<ll> &arr) {
        if (l == r) {
            tree[idx].cnt = 1;
            tree[idx].pref[0] = arr[l];
            tree[idx].pref[1] = arr[l];
            return;
        }
        ll mid = (l + r) / 2;
        build(2 * idx, l, mid, arr);
        build(2 * idx + 1, mid + 1, r, arr);
        tree[idx].cnt = tree[2 * idx].cnt + tree[2 * idx + 1].cnt;
        tree[idx].pref[0] = tree[2 * idx].pref[0];
        tree[idx].pref[1] = tree[2 * idx + 1].pref[1];
    }

    void update(ll idx, ll pos, ll val, ll l, ll r) {
        if (l == r) {
            tree[idx].pref[0] = val;
            tree[idx].pref[1] = val;
            return;
        }
        ll mid = (l + r) / 2;
        if (pos <= mid) {
            update(2 * idx, pos, val, l, mid);
        } else {
            update(2 * idx + 1, pos, val, mid + 1, r);
        }
        tree[idx].pref[0] = tree[2 * idx].pref[0];
        tree[idx].pref[1] = tree[2 * idx + 1].pref[1];
    }

    ll get(ll idx, ll l, ll r, ll ql, ll qr, ll flag) {
        if (l > qr || r < ql) {
            return 0;
        }
        if (l >= ql && r <= qr) {
            return tree[idx].pref[flag];
        }
        ll mid = (l + r) / 2;
        return get(2 * idx, l, mid, ql, qr, flag) + get(2 * idx + 1, mid + 1, r, ql, qr, flag);
    }

    ll count(ll idx, ll l, ll r, ll ql, ll qr) {
        if (l > qr || r < ql) {
            return 0;
        }
        if (l >= ql && r <= qr) {
            return tree[idx].cnt;
        }
        ll mid = (l + r) / 2;
        return count(2 * idx, l, mid, ql, qr) + count(2 * idx + 1, mid + 1, r, ql, qr);
    }
};

int main() {
    ll n, q, k;
    cin >> n >> q >> k;
    vector<ll> arr(n);
    for (ll i = 0; i < n; i++) {
        cin >> arr[i];
    }
    segment_tree st(n);
    st.build(1, 0, n - 1, arr);
    while (q--) {
        ll l, r;
        cin >> l >> r;
        ll l_diff = st.get(1, 0, n - 1, l - 1, l - 1, 0);
        ll r_diff = st.get(1, 0, n - 1, r, r, 1);
        ll sum = st.count(1, 0, n - 1, l, r);
        if (r_diff <= l_diff) {
            cout << sum << "\n";
        } else {
            cout << add(sum, ncr(sum, 1)) << "\n";
        }
    }
    return 0;
}
