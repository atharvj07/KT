#include <bits/stdc++.h>
using namespace std;
int solve(int n, int k, int d, vector<int> &a) {
    map<int, int> cnt;
    int ans = k;
    for (int l = 0, r = 0; l < n; l++) {
        while (r < n && cnt.size() <= k) {
            cnt[a[r]]++;
            r++;
        }
        if (r - l == d) ans = min(ans, (int)cnt.size());
        cnt[a[l]]--;
        if (cnt[a[l]] == 0) cnt.erase(a[l]);
    }
    return ans;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k, d;
        cin >> n >> k >> d;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        cout << solve(n, k, d, a) << endl;
    }
    return 0;
}
