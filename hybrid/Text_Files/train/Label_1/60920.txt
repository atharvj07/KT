n, k = map(int, input().split())
coupons = []
for i in range(n):
    l, r = map(int, input().split())
    coupons.append((l, r, i + 1))

coupons.sort(key=lambda x: x[1], reverse=True)

max_products = 0
chosen_coupons = []
for i in range(n - k + 1):
    min_r = coupons[i][1]
    sorted_coupons = sorted(coupons[i:i + k], key=lambda x: x[0])
    max_l = sorted_coupons[-1][0]
    if max_l <= min_r:
        if min_r - max_l + 1 > max_products:
            max_products = min_r - max_l + 1
            chosen_coupons = [x[2] for x in sorted_coupons]

print(max_products)
print(' '.join(map(str, chosen_coupons)))
