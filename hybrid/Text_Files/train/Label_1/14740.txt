#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    int petya = 0, gena = 0;
    int ans = 0;
    map<int, int> mp;

    for (int i = 0; i < n; ++i) {
        if (a[i] == 1)
            petya++;
        else
            gena++;
        if (petya == gena)
            mp[petya + gena]++;

        if (petya > gena) {
            ans++;
            cout << petya + gena << " " << petya << endl;
        } else if (gena > petya) {
            if (mp.find(gena + petya) != mp.end()) {
                for (int j = 0; j < mp[gena + petya]; ++j) {
                    ans++;
                    cout << gena + petya << " " << petya + gena - 1 << endl;
                }
            }
        }
    }

    cout << ans << endl;
}
