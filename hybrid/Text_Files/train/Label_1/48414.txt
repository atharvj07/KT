import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int m = in.nextInt();
        int[] arr = new int[k];
        for (int i = 0; i < k; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int i = 0;
        int ans = 0;
        while (m >= arr[i] && i < k) {
            m -= arr[i++];
            ans++;
        }
        ans += Math.min(m / arr[i], n);
        System.out.println(ans);
    }
}
