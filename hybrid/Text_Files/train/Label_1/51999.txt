import java.util.Arrays;
import java.util.Scanner;

public class TeaParty {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int w = scanner.nextInt();
        int[] cups = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            cups[i] = scanner.nextInt();
        }

        Arrays.sort(cups);

        double maxWater = 0;
        for (int i = 0; i < n; i++) {
            double girlWater = cups[i];
            double boyWater = 2 * girlWater;
            int boysToFill = n - i - 1;
            int remainingWater = w - (int) (girlWater * n + boyWater * boysToFill);
            if (remainingWater >= 0) {
                maxWater = Math.max(maxWater, girlWater * n + boyWater * boysToFill + remainingWater / (boysToFill + 1));
            }
        }

        System.out.println(maxWater);
    }
}
