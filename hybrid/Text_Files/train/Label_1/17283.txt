#include <iostream>
#include <vector>
using namespace std;
int a[200005], ans[200005];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int sum = 0;
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (a[i] > 0)
      ++cnt;
    else if (a[i] < 0)
      --cnt;
    sum += abs(a[i]);
    if (cnt != 0 || a[i] != 0) {
      ans[i] = 1;
      ans[i] += ans[i - 1];
    }
  }
  cout << sum - ans[n] << endl;
  return 0;
}
