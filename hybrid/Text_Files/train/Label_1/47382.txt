import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] stones = new int[14];

        for (int i = 0; i < 14; i++) {
            stones[i] = scanner.nextInt();
        }

        int maxScore = 0;
        for (int i = 0; i < 14; i++) {
            if (stones[i] > 0) {
                int score = simulateMove(stones, i);
                maxScore = Math.max(maxScore, score);
            }
        }

        System.out.println(maxScore);
    }

    private static int simulateMove(int[] stones, int startHole) {
        int score = 0;
        int stonesToMove = stones[startHole];
        stones[startHole] = 0;
        int currentHole = startHole;

        while (stonesToMove > 0) {
            currentHole = (currentHole + 1) % 14;
            stones[currentHole]++;
            stonesToMove--;

            if (stones[currentHole] % 2 == 0) {
                score += stones[currentHole];
                stones[currentHole] = 0;
            }
        }

        return score;
    }
}
