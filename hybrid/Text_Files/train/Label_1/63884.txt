from collections import defaultdict

def main():
    n = int(input())
    points = defaultdict(list)
    for i in range(n):
        x, y = map(int, input().split())
        points[(x, y)].append(i)
    if len(set(points.keys())) == 1 or len(set(points.values())) == 1:
        print("YES")
    else:
        print("NO")

main()
