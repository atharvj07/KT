n, k = map(int, input().split())
total_red = total_blue = total_baskets = 0
remainder = [0] * k

for _ in range(n):
    a, b = map(int, input().split())
    total_red += a
    total_blue += b
    baskets = (a + b) // k
    total_baskets += baskets
    red_remainder = a % k
    blue_remainder = b % k
    remainder_sum = (red_remainder + blue_remainder) // k
    total_baskets += remainder_sum
    remainder[red_remainder] += 1
    remainder[blue_remainder] += 1

for i in range(1, k):
    min_pairs = min(remainder[i], remainder[k-i])
    total_baskets += min_pairs
    remainder[i] -= min_pairs
    remainder[k-i] -= min_pairs

print(min(total_baskets, (total_red + total_blue) // k))
