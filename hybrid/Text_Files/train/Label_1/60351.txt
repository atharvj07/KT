n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
pos = [0] * (n + 1)
for i in range(n):
    pos[a[i]] = i
ans = 0
sc = 1
l = 1
r = k
for x in b:
    if pos[x] < l:
        l = 1
        r = k
    while l <= r:
        m = (l + r) // 2
        if pos[x] <= m:
            r = m - 1
        else:
            l = m + 1
    ans += l-1
    tmp = pos[x]
    for i in range(pos[x], l - 1, -1):
        a[i] = a[i - 1]
        pos[a[i]] = i
    a[l - 1] = x
    pos[x] = l - 1
    sc += 1
    l = max(l-k, 1)
    r = sc*k
print(ans)
