#include <iostream>
using namespace std;
int main() {
  int t, a, b, x, y, n;
  cin >> t;
  while (t--) {
    cin >> a >> b >> x >> y >> n;
    for (int i = 0; i < n; i++) {
      if (a <= x and b <= y) break;
      if (a > b)
        swap(a, b);
      if (a > x) {
        a--, b++;
      } else if (b > y)
        b--;
    }
    cout << a * b << endl;
  }
}
