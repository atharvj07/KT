#include <iostream>
#include <algorithm>

int main() {
  int n, sx, sy;
  std::cin >> n >> sx >> sy;
  int x[n], y[n];
  for (int i = 0; i < n; i++) { std::cin >> x[i] >> y[i]; }
  int mx = 0, my = 0, maxcnt = 0;
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    if (x[i] == sx) {
      if (y[i] < sy) {
        cnt += std::upper_bound(y, y + n, sy - 1) - y;
      } else {
        cnt += n - (std::lower_bound(y, y + n, sy + 1) - y);
      }
    } else if (y[i] == sy) {
      if (x[i] < sx) {
        cnt += std::upper_bound(x, x + n, sx - 1) - x;
      } else {
        cnt += n - (std::lower_bound(x, x + n, sx + 1) - x);
      }
    } else {
      if (x[i] < sx && y[i] < sy) {
        cnt += std::upper_bound(y, y + n, sy - 1) - y;
        cnt += std::upper_bound(x, x + n, sx - 1) - x;
        cnt--;
      } else if (x[i] < sx && y[i] > sy) {
        cnt += n - (std::lower_bound(y, y + n, sy + 1) - y);
        cnt += std::upper_bound(x, x + n, sx - 1) - x;
        cnt--;
      } else if (x[i] > sx && y[i] < sy) {
        cnt += std::upper_bound(y, y + n, sy - 1) - y;
        cnt += n - (std::lower_bound(x, x + n, sx + 1) - x);
        cnt--;
      } else {
        cnt += n - (std::lower_bound(y, y + n, sy + 1) - y);
        cnt += n - (std::lower_bound(x, x + n, sx + 1) - x);
        cnt--;
      }
    }
    if (cnt > maxcnt) {
      maxcnt = cnt;
      mx = x[i];
      my = y[i];
    }
  }
  std::cout << maxcnt << "\n" << mx << " " << my << "\n";
  return 0;
}
