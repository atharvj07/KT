#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, a[55][55];
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) { cin >> a[i][j]; }
    }
    int p[55];
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (!p[i] && !p[j] && a[i][j] == 0) { p[i] = j, p[j] = i; }
        }
    }
    for (int i = 1; i <= n; i++) {
        if (!p[i]) {
            for (int j = 1; j <= n; j++) {
                if (!p[j]) { p[i] = j, p[j] = i; }
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << p[i] << " ";
    cout << endl;
}
