#include<bits/stdc++.h>
using namespace std;

int main()
{
    long long n,k,x = 0, y = 0, ans = 0;
    cin >> n >> k;
    x += k;
    y += k;
    for (long long i = 1; ; i++)
    {
        if (x == y) {
            ans = i - 1;
            break;
        }
        if (i % 2) {
           x += i;
           ans++;
        }
        else {
            y += i;
        }
    }
    cout << ans << endl;
    return 0;
}
