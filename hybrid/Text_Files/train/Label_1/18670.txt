#include <vector>
#include <cstdio>
#include <cmath>
using namespace std;

struct Time {
    long long s, d;
    long long cycle, remain;

    Time() : s(0), d(0), cycle(0), remain(0) { }
    Time(long long _s, long long _d) : s(_s), d(_d) {
        cycle = ceil(1.0 * s / d);
        remain = ceil(1.0 * (s - d * cycle) / d);
    }
    long long toDay() const {
        return cycle * d + remain;
    }
};

int main() {
    int n;
    scanf("%d", &n);
    vector<Time> times(n);
    for (int i = 0; i < n; i++) {
        long long s, d;
        scanf("%lld %lld", &s, &d);
        times[i] = Time(s, d);
    }
    long long ans = times[0].toDay();
    for (int i = 1; i < n; i++) {
        ans = max(ans, times[i].toDay());
    }
    printf("%lld\n", ans);
    return 0;
}
