#include <bits/stdc++.h>
#define int long long
#define endl '\n'
using namespace std;
const int mod = 1e9 + 7;
int f[100007][102];
int32_t main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int ans = 0;
    f[n][0] = 1;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j <= k; j++) {
            f[i][j] = f[i + 1][j];
            if (j < k) { f[i][j] = f[i][j] + f[i + 1][j + 1] * (s[i] - '0'); }
        }
    }
    for (int i = 0; i <= k; i++) { ans = (ans + f[0][i] * (s[0] - '0')) % mod; }
    cout << ans << endl;
}
