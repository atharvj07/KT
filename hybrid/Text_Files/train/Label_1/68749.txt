import sys
def is_valid(x: int, y: int, a: int, b: int) -> int:
    if x >= y:
        return -1
    delta = y - x
    speed = a + b
    if delta % speed:
        return -1
    return delta // speed

input = sys.stdin.readline
for _ in range(int(input())):
    x, y, a, b = map(int, input().split())
    print(is_valid(x, y, a, b))
