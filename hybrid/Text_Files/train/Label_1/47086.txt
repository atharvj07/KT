import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt(), m = sc.nextInt(), k = sc.nextLong();
        long r = k / m + 1, c = k % m + 1;
        System.out.printf("%d %d", r % n == 0 ? n : r % n, c);
    }
}
