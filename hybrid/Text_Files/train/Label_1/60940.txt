import sys
input=sys.stdin.readline
n,m=[int(x)for x in input().split()]
x=[int(x)for x in input().split()]
y=[int(x)for x in input().split()]
b=[{}for i in range(300001)]
for i in range(m):
 a,b=map(int,input().split())
 b[a][b]=1
 b[b][a]=1
L=[[]]
for i in range(2):
 for j in range(n):
  if i==0:
   a=x[j]
  else:
   a=y[j]
  p=0
  while L[p] and L[p][-1][0]>a:
   p+=1
  L[p].append([a,j+1])
 S=[0]*n
for j in L:
 for a,i in j:
  S[i-1]+=a
  for k in b[i]:
   S[k-1]+=a
print(*S)
