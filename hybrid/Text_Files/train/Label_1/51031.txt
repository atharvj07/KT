import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
  private static final int INF = Integer.MAX_VALUE;

  private static int n, q;
  private static long[] a, b;

  public static void main(String[] args) throws IOException {
    try (var reader = new BufferedReader(new InputStreamReader(System.in))) {
      var line = reader.readLine().split(" ");
      n = Integer.parseInt(line[0]);
      q = Integer.parseInt(line[1]);

      a = Arrays.stream(reader.readLine().split(" ")).mapToLong(Long::parseLong).toArray();
      b = Arrays.stream(reader.readLine().split(" ")).mapToLong(Long::parseLong).toArray();

      var dp = new long[2 * n + 1][2];
      Arrays.fill(dp[0], -INF);
      dp[0][0] = 0;

      for (int i = 0; i < 2 * n; i++) {
        for (int j = 0; j < 2; j++) {
          long cost = j == 0 ? a[i] : b[i];
          dp[i + 1][0] = Math.max(dp[i + 1][0], dp[i][j]);
          dp[i + 1][1] = Math.max(dp[i + 1][1], dp[i][j] + cost);
        }
      }

      while (q-- > 0) {
        line = reader.readLine().split(" ");
        var p = Integer.parseInt(line[0]);
        var x = Long.parseLong(line[1]);
        var y = Long.parseLong(line[2]);

        p--;
        a[p] = x;
        b[p] = y;

        dp = new long[2 * n + 1][2];
        Arrays.fill(dp[0], -INF);
        dp[0][0] = 0;

        for (int i = 0; i < 2 * n; i++) {
          for (int j = 0; j < 2; j++) {
            long cost = j == 0 ? a[i] : b[i];
            dp[i + 1][0] = Math.max(dp[i + 1][0], dp[i][j]);
            dp[i + 1][1] = Math.max(dp[i + 1][1], dp[i][j] + cost);
          }
        }

        System.out.println(dp[2 * n][1]);
      }
    }
  }
}
