#include <iostream>
#include <vector>
using namespace std;

int ext_gcd(int a, int b, int& x, int& y) {
    if (b == 0) {
        x = 1;
        y = 0;
        return a;
    }
    int gcd = ext_gcd(b, a % b, x, y);
    int tmp = x;
    x = y;
    y = tmp - (a / b) * y;
    return gcd;
}

int main() {
    int m, h1, a1, x1, y1, h2, a2, x2, y2;
    cin >> m >> h1 >> a1 >> x1 >> y1 >> h2 >> a2 >> x2 >> y2;
    int x, y, gcd;
    gcd = ext_gcd(x1, m, x, y);
    if ((a1 - h1) % gcd != 0) {
        cout << -1 << endl;
        return 0;
    }
    x = (a1 - h1) * x % m;
    x = (x % m + m) % m;
    gcd = ext_gcd(x2, m, x, y);
    if ((a2 - h2) % gcd != 0) {
        cout << -1 << endl;
        return 0;
    }
    x = (a2 - h2) * x % m;
    x = (x % m + m) % m;
    cout << x << endl;
    return 0;
}
