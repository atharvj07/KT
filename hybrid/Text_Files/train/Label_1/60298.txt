def max_tandem_repeat(s, k):
    n = len(s)
    max_length = 0
    for length in range(1, n + 1):
        for start in range(n - length + 1):
            segment = s[start:start + length]
            repetitions = s.count(segment)
            total_length = length * repetitions
            extra_length = min(k, length - (n - total_length))
            max_length = max(max_length, total_length + extra_length)
    return min(max_length * 2, 2 * (n + k))

s = input().strip()
k = int(input())
print(max_tandem_repeat(s, k))
