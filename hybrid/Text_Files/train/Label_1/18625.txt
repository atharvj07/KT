const int N=1005;

    ios::sync_with_stdio(0),cin.tie(0);
    cin>>n>>m;
    REP(i,m){
        cin>>a[i]>>b[i];

            ans[i]='0';

        else{
            cnt1++;
            ans[i]='1';
        }
    }
