#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
vector<ll> a, c;
vector<int> b;
ll dp[101][1000001], ans;
int main() {
    int n, m;
    cin >> n >> m;
    a.resize(n + 1), c.resize(n + 1), b.resize(m);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 0; i < m; i++) { cin >> b[i]; }
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= 1000000; j++) dp[i][j] = -1;
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= 1000000; j++) {
            int f = 0;
            for (int k = 0; k < m; k++) {
                if (b[k] == i) { f = 1; }
            }
            dp[i][j] = max(dp[i][j], dp[i - 1][j]);
            if (j >= a[i]) { dp[i][j] = max(dp[i][j], dp[i - 1][j - a[i]] + a[i]); }
            if (f == 0 && j + a[i] <= 1000000) { dp[i][j] = max(dp[i][j], dp[i - 1][j + a[i]] + a[i]); }
        }
    }
    for (int i = 0; i <= 1000000; i++) { ans = max(ans, dp[n][i]); }
    cout << ans;
    return 0;
}
