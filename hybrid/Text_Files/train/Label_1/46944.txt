import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

class Query {
    int x0, y0, x1, y1;

    public Query(int x0, int y0, int x1, int y1) {
        this.x0 = x0;
        this.y0 = y0;
        this.x1 = x1;
        this.y1 = y1;
    }
}

class SegmentTree {
    int n;
    long[] tree;
    int[] lazy;

    public SegmentTree(int n) {
        this.n = n;
        this.tree = new long[4 * n];
        this.lazy = new int[4 * n];
    }

    public void update(int l, int r, int v, int node, int sl, int sr) {
        if (lazy[node] != 0) {
            tree[node] ^= (long) lazy[node] * (sr - sl + 1);
            if (sl != sr) {
                lazy[2 * node + 1] ^= lazy[node];
                lazy[2 * node + 2] ^= lazy[node];
            }
            lazy[node] = 0;
        }

        if (sl > r || sr < l) {
            return;
        }

        if (l <= sl && sr <= r) {
            tree[node] ^= (long) v * (sr - sl + 1);
            if (sl != sr) {
                lazy[2 * node + 1] ^= v;
                lazy[2 * node + 2] ^= v;
            }
            return;
        }

        int mid = (sl + sr) / 2;
        update(l, r, v, 2 * node + 1, sl, mid);
        update(l, r, v, 2 * node + 2, mid + 1, sr);
        tree[node] = tree[2 * node + 1] ^ tree[2 * node + 2];
    }

    public long query(int l, int r, int node, int sl, int sr) {
        if (lazy[node] != 0) {
            tree[node] ^= (long) lazy[node] * (sr - sl + 1);
            if (sl != sr) {
                lazy[2 * node + 1] ^= lazy[node];
                lazy[2 * node + 2] ^= lazy[node];
            }
            lazy[node] = 0;
        }

        if (sl > r || sr < l) {
            return 0;
        }

        if (l <= sl && sr <= r) {
            return tree[node];
        }

        int mid = (sl + sr) / 2;
        return query(l, r, 2 * node + 1, sl, mid) ^ query(l, r, 2 * node + 2, mid + 1, sr);
    }
}

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        SegmentTree tree = new SegmentTree(n);

        List<Query> queries = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            if (type == 1) {
                int x0 = Integer.parseInt(line[1]) - 1;
                int y0 = Integer.parseInt(line[2]) - 1;
                int x1 = Integer.parseInt(line[3]) - 1;
                int y1 = Integer.parseInt(line[4]) - 1;
                queries.add(new Query(x0, y0, x1, y1));
            } else if (type == 2) {
                int x0 = Integer.parseInt(line[1]) - 1;
                int y0 = Integer.parseInt(line[2]) - 1;
                int x1 = Integer.parseInt(line[3]) - 1;
                int y1 = Integer.parseInt(line[4]) - 1;
                int v = Integer.parseInt(line[5]);

                tree.update(x0, x1, v, 0, 0, n - 1);
                tree.update(y0, y1, v, 0, 0, n - 1);
            }
        }

        for (Query query : queries) {
            long result = tree.query(query.x0, query.x1, 0, 0, n - 1) ^ tree.query(query.y0, query.y1, 0, 0, n - 1);
            System.out.println(result);
        }
    }
}
