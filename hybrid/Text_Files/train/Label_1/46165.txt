import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        List<List<Integer>> graph = new ArrayList<>();
        List<List<Integer>> weight = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
            weight.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt() - 1;
            int b = scanner.nextInt() - 1;
            int w1 = scanner.nextInt();
            int w2 = scanner.nextInt();
            graph.get(a).add(b);
            graph.get(b).add(a);
            weight.get(a).add(w1);
            weight.get(b).add(w2);
        }
        int[] dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[0] = 0;
        boolean[] visited = new boolean[n];
        int[] prev = new int[n];
        int minCycle = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            if (visited[i]) {
                continue;
            }
            if (findNegativeCycle(i, graph, weight, dist, visited, prev)) {
                int v = i;
                int cycleLength = 1;
                while (prev[v] != v) {
                    v = prev[v];
                    cycleLength++;
                }
                minCycle = Math.min(minCycle, cycleLength);
            }
        }
        System.out.println(minCycle == Integer.MAX_VALUE ? 0 : minCycle);
    }

    private static boolean findNegativeCycle(int start, List<List<Integer>> graph, List<List<Integer>> weight, int[] dist, boolean[] visited, int[] prev) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(start);
        while (!queue.isEmpty()) {
            int u = queue.poll();
            visited[u] = true;
            for (int i = 0; i < graph.get(u).size(); i++) {
                int v = graph.get(u).get(i);
                int w = weight.get(u).get(i);
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    prev[v] = u;
                    if (visited[v]) {
                        return true;
                    } else {
                        queue.add(v);
                    }
                }
            }
        }
        return false;
    }
}
