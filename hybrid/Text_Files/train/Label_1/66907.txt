for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=list(map(int,input().split()))
    if len(set(a)) != len(set(b)):
        print("NO")
        continue
    ina=[a.count(x) for x in set(a)]
    inb=[b.count(x) for x in set(a)]
    if len(set(ina))!=len(set(inb)) or max(ina)!=max(inb):
        print("NO")
        continue
    sq=int((max(ina))**0.5)
    if sq**2!=max(ina):
        print("NO")
        continue
    l1=[b.index(x) for x in set(a)]
    l2=[a.index(x) for x in set(a)]
    l=max(set(l1).intersection(set(l2)))
    r=max(l1)+min(ina)-1
    if r-l+1==max(ina) and ina[l1.index(l)]==max(ina):
        print("YES")
    else:
        print("NO")
