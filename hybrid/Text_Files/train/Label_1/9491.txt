#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> a(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> a[i][j];
    }
  }

  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    p[i] = i + 1;
  }

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (a[i][j] > 0) {
        if (p[i] > p[j]) {
          swap(p[i], p[j]);
        }
      }
    }
  }

  for (int i = 0; i < n; i++) {
    cout << p[i] << " ";
  }
  cout << endl;

  return 0;
}
