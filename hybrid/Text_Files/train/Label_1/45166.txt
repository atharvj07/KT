import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        StringBuilder sb = new StringBuilder(cin.next());
        int len = sb.length();
        int[] arr = new int[len];
        for (int i = 0; i < len; i++) {
            arr[i] = sb.charAt(i) - '0';
        }
        Arrays.sort(arr);
        for (int i = len - 1; i >= 0; i--) {
            System.out.print(arr[i]);
        }
        System.out.println();
        for (int n : arr) {
            System.out.print(n);
        }
        System.out.println();
    }
}
