#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int n;
long long a[300005], pre[300005];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pre[i] = pre[i - 1] ^ a[i];
    }
    long long ans = 0;
    for (int i = 2; i + 1 <= n; i += 2) {
        if ((pre[i - 1] ^ pre[i / 2]) == (pre[n] ^ pre[i / 2])) { ans++; }
    }
    cout << ans;
    return 0;
}
