from collections import Counter

def find_array_t(n, k, s):
    freq = Counter(s)
    left, right = 1, n // k + 1
    
    while left < right:
        mid = (left + right) // 2
        if sum(min(freq[num], mid) for num in freq) >= mid * k:
            left = mid + 1
        else:
            right = mid
    
    result_freq = (left - 1) * k
    result = []
    
    for num, count in freq.items():
        times = min(count, left - 1)
        result += [num] * times
        result_freq -= times
        if result_freq == 0:
            break
    
    return result

n, k = map(int, input().split())
s = list(map(int, input().split()))

print(*find_array_t(n, k, s))
