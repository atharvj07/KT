import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String a = sc.nextLine();
        String b = sc.nextLine();

        int n = a.length();
        int ops = 0;

        for (int i = 0; i < n; i++) {
            if (a.charAt(i) != b.charAt(i)) {
                if (a.charAt(i) == '4') {
                    ops++;
                } else {
                    ops += 3;
                }
            }
        }

        System.out.println(ops);
    }
}
