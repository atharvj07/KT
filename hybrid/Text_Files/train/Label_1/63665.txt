import sys

n = int(sys.stdin.readline().strip())
names = []
for i in range(n):
    names.append(sys.stdin.readline().strip())

def is_possible(names):
    for i in range(1, n):
        if names[i-1] > names[i]:
            return False
    return True

def find_order(names):
    order = []
    for i in range(26):
        order.append(chr(ord('a') + i))
    for i in range(n):
        for j in range(n):
            if names[i][0] == order[j]:
                order = order[j:] + order[:j]
                break
    return order

if is_possible(names):
    print(''.join(find_order(names)))
else:
    print("Impossible")
