import sys
input = sys.stdin.readline
n, m = map(int, input().split())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]

a.sort()
b.sort()
score = 0

for i in b:
    if score >= a[i - 1]:
        score += a[i - 1]
    else:
        score = a[i - 1]

for i in range(m, n):
    score += a[i]

print(score)
