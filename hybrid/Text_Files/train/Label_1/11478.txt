#include<bits/stdc++.h>
using namespace std;
int main() {
    long long n, a[100005];
    int ans1, ans2;
    cin >> n;
    for(int i=1; i<=n; i++) cin >> a[i];
    for(int i=1; i<=n; i++) {
        if(a[i] > a[i+1]) {
            ans1 = i;
            break;
        }
    }
    for(int i=n; i>=1; i--) {
        if(a[i] < a[i-1]) {
            ans2 = i;
            break;
        }
    }
    if(ans1 == 0 or ans2 == 0) cout << -1;
    else cout << ans1 << " " << ans2 << endl;
}
