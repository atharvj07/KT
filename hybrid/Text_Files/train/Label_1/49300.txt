import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();

            int[] l = new int[n];
            int[] h = new int[n];
            int[] t = new int[n];
            for (int j = 0; j < n; j++) {
                t[j] = scanner.nextInt();
                l[j] = scanner.nextInt();
                h[j] = scanner.nextInt();
            }

            int[] dp = new int[10000001];
            for (int j = 0; j < n; j++) {
                int start = Math.max(1, t[j] - 2000000);
                int end = Math.min(10000000, t[j] + 2000000);
                for (int k = start; k <= end; k++) {
                    dp[k] = Math.max(dp[k], dp[k - 1] + h[j] - l[j] + 1);
                }
            }

            int ans = 0;
            for (int j = 0; j < 10000001; j++) {
                ans = Math.max(ans, dp[j]);
            }

            if (ans >= 0) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
