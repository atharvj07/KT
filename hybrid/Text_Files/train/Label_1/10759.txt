#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  ll n;
  cin >> n;
  unordered_map<ll, vector<ll>> m;
  for (ll i = 0; i < n; i++) {
    ll a;
    cin >> a;
    m[a].push_back(i);
  }
  vector<pair<ll, ll>> ans;
  for (const auto &[k, v] : m) {
    if (v.size() == 1) {
      ans.push_back({k, 0});
      continue;
    }
    ll d = v[1] - v[0];
    bool ok = true;
    for (ll i = 2; i < v.size(); i++) {
      if (v[i] - v[i - 1] != d) {
        ok = false;
        break;
      }
    }
    if (ok) ans.push_back({k, d});
  }
  cout << ans.size() << '\n';
  for (const auto &[x, d] : ans) {
    cout << x << ' ' << d << '\n';
  }
  return 0;
}
