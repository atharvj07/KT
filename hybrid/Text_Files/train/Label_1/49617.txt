import java.util.*;
import java.io.*;

class Main {
  static long mod = 998244353;
  static int n;
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    int[] a = new int[2 * n + 1];
    for (int i = 0; i < 2 * n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    long ans = 1;
    for (int i = 0; i < n; i++) {
      ans = (ans * (a[2 * i + 1] - a[2 * i])) % mod;
    }
    System.out.println(ans);
  }
}
