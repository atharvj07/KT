
def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        fac = []
        for i in range(2, int(n ** 0.5) + 1):
            while n % i == 0:
                n //= i
                fac.append(i)
        if n > 1:
            fac.append(n)
        k = len(fac)
        print(k)
        print(*fac)

solve()
