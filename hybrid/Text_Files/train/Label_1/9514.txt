#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e6 + 5;
string s[mxN];
int32_t main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> s[i]; }
    sort(s + 1, s + n + 1, [&](string x, string y) {
        for (int j = 0; j < m; j++) {
            if (j & 1) {
                if (x[j] > y[j]) { return false; }
                if (x[j] < y[j]) { return true; }
            } else {
                if (x[j] < y[j]) { return false; }
                if (x[j] > y[j]) { return true; }
            }
        }
        return false;
    });
    for (int i = 1; i <= n; i++) { cout << i << ' '; }
    return 0;
}
