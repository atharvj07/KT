n, m = map(int, input().split())
a = list(map(int, input().split()))

# Count the current songs for bands 1 to m
band_counts = [0] * m
for song in a:
    if 1 <= song <= m:
        band_counts[song-1] += 1

# Calculate the minimum songs each band should perform
min_songs = n // m

# Calculate changes and prepare the new playlist
changes = 0
new_playlist = a.copy()
for i in range(n):
    if band_counts[a[i]-1] > min_songs and a[i] <= m:
        # Find a band with less than min_songs
        for j in range(m):
            if band_counts[j] < min_songs:
                new_playlist[i] = j + 1
                band_counts[j] += 1
                band_counts[a[i]-1] -= 1
                changes += 1
                break
    elif a[i] > m:
        # Replace an unknown band with a band that needs more songs
        for j in range(m):
            if band_counts[j] < min_songs:
                new_playlist[i] = j + 1
                band_counts[j] += 1
                changes += 1
                break

# Calculate the real minimum among the bands 1 to m after changes
real_min = min(band_counts)

print(real_min, changes)
print(' '.join(map(str, new_playlist)))
