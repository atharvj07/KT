#include <iostream>
#include <set>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::set<int>> rows(n), cols(n);
  for (int i = 0, row; i < n; i++) {
    for (int j = 0; j < n; j++) {
      std::cin >> row;
      rows[i].insert(row);
      cols[j].insert(row);
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      int num = *(rows[i].begin()) + *(cols[j].begin());
      if (*rows[i].find(num) == *rows[i].end() ||
          *cols[j].find(num) == *cols[j].end()) {
        std::cout << "No" << '\n';
        return 0;
      }
    }
  }
  std::cout << "Yes" << '\n';
  return 0;
}
