#include<bits/stdc++.h>

#define ll long long
#define pii pair<ll,ll>

int mod = 998244353;

ll f(ll n,vector<ll>&a,ll k){
    ll ans=0;
    for(ll i=2;i<=n;++i){
        ll par=(i-1+k)/k;
        if(a[i]<a[par])ans++;
    }
    return ans;
}
int main() {
    ll n;cin>>n;
    vector<ll> a(n+1);
    for(ll i=1;i<=n;++i)cin>>a[i];
    for(ll k=1;k<n;++k){
        cout<<f(n,a,k)<<" ";
    }
    return 0;
}
