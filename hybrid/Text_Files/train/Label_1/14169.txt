#include <iostream>
#include <vector>
#include <sstream>

using namespace std;

int main() {
  string s;
  cin >> s;
  vector<int> nums;
  vector<char> ops;
  int i = 0, n = s.length();
  stringstream ss;
  while (i < n) {
    if (s[i] != '+' && s[i] != '-') {
      ss.clear();
      ss << s[i];
      int x;
      ss >> x;
      nums.push_back(x);
    } else {
      ops.push_back(s[i]);
    }
    i++;
  }
  int ans = nums[0];
  for (i = 0; i < ops.size(); i++) {
    if (ops[i] == '+') {
      ans += nums[i + 1];
    } else {
      ans -= nums[i + 1];
    }
  }
  cout << ans << endl;
  return 0;
}
