#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> w(2 * n);
    for (int i = 0; i < 2 * n; i++) cin >> w[i];
    sort(w.begin(), w.end());
    int ans = 0;
    for (int i = 0; i < n - 1; i++) ans += w[2 * i + 1] - w[2 * i];
    cout << ans << endl;
    return 0;
}
