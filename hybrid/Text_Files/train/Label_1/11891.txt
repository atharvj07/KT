#include<bits/stdc++.h>
using namespace std;
int main()
    {
    int n;
        cin >> n;
        string s;
        cin >> s;
        int ans = 0;
        unordered_map<int, int> x;
        unordered_map<int, int> y;

        for (char c : s)
            {
            if (c == 'R')
                ++x[ans];
            else if (c == 'L')
                --x[ans];
            else if (c == 'U')
                ++y[ans];
            else
                --y[ans];
            }
        int mx1 = 0, mx2 = 0;
        int sum = 0;
        for (auto it : x)
            {
            mx1 = max(mx1, it.second);
            sum += it.second;
            }
        for (auto it : y)
            {
            mx2 = max(mx2, it.second);
            sum += it.second;
            }
        cout << sum - min(abs(mx1), abs(mx2)) << "\n";
        return 0;
    }
