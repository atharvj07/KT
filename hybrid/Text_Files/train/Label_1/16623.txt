#include <iostream>

int main() {
    long long a, b, c;
    std::cin >> a >> b >> c;
    long long num = a * 10;
    long long pos = 0;
    while (num < b * 10) {
        pos++;
        num = num * 10 + a;
        num %= b;
        if (num % 10 == c) {
            std::cout << pos << std::endl;
            return 0;
        }
    }
    std::cout << -1 << std::endl;
    return 0;
}
