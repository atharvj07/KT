n, T = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
dp = [[[(0, 0) for j in range(4)] for i in range(T + 1)] for k in range(n + 1)]
dp[n][0][0] = (1, 0)
for i in range(n - 1, -1, -1):
    for t in range(T + 1):
        for l in range(4):
            dp[i][t][l] = dp[i + 1][t][l]
            if l != a[i][1] and (t + a[i][0] <= T):
                x, y = dp[i + 1][t + a[i][0]][a[i][1]]
                dp[i][t][l] = ((dp[i][t][l][0] + x) % 1000000007, (dp[i][t][l][1] + y) % 1000000007)
print(dp[0][0][0][1])
