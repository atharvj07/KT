import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s1 = in.nextLine(), s2 = in.nextLine();
        if (s1.equals(s2)) {
            System.out.println("YES");
            return;
        }
        int n = s1.length();
        if (n % 2 == 1) {
            System.out.println("NO");
            return;
        }
        String s1a = s1.substring(0, n / 2), s1b = s1.substring(n / 2);
        String s2a = s2.substring(0, n / 2), s2b = s2.substring(n / 2);
        if (s1a.equals(s2a) && s1b.equals(s2b) || s1a.equals(s2b) && s1b.equals(s2a)) {
            System.out.println("YES");
            return;
        }
        System.out.println("NO");
    }
}
