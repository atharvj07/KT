import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long f = in.nextLong(), t = in.nextLong(), t0 = in.nextLong();
        long a1 = in.nextLong(), t1 = in.nextLong(), p1 = in.nextLong();
        long a2 = in.nextLong(), t2 = in.nextLong(), p2 = in.nextLong();
        in.close();

        long ans = -1;

        for (int i = 0; ; i++) {
            long remainder = f - i * a1 - i * a2;
            long time = i * t1 + i * t2 + remainder / t0 * t0;
            if (time <= t) {
                ans = i * p1 + i * p2;
            } else {
                break;
            }
        }

        System.out.println(ans);
    }
}
