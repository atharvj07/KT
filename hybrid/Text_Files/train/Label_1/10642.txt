#include <bits/stdc++.h>
using namespace std;
#define int long long
void solve() {
  string s;
  cin >> s;
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  reverse(a.begin(), a.end());
  for (int i = 0; i < n; i++) {
    int l = a[i] - 1, r = s.size() - a[i];
    reverse(s.begin() + l, s.begin() + r);
  }
  cout << s << endl;
}
int32_t main() { solve(); }
