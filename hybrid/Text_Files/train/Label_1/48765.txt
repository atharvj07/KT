import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a[] = new int[n];
        int b[] = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        for (int i = 0; i < n; i++)
            b[i] = sc.nextInt();
        long sum1 = 0, sum2 = 0;
        for (int i = 0; i < n; i++) {
            sum1 += a[i];
            sum2 += b[i] - a[i];
        }
        if (sum1 > sum2)
            System.out.println("NO");
        else
            System.out.println("YES");
    }
}
