#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a, cnt = 0;
  cin >> n;
  vector<bool> v(n);
  for (int i = 0; i < n; i++) {
    cin >> a;
    v[i] = a;
  }
  for (int i = 0; i < n - 1; i++) {
    if (v[i] != v[i + 1]) cnt++;
  }
  if (cnt == 0)
    cout << "NO";
  else if (cnt == 2)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
