import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int b = sc.nextInt();
        int[][] arrows = new int[n][4];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                arrows[i][j] = sc.nextInt();
            }
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            char d = sc.next().charAt(0);
            long t = sc.nextLong();
            long[] v = new long[2];
            if (d == 'U') {
                v[1] = 1;
            } else if (d == 'D') {
                v[1] = -1;
            } else if (d == 'L') {
                v[0] = -1;
            } else if (d == 'R') {
                v[0] = 1;
            }
            for (int j = 0; j < t; j++) {
                x += v[0];
                y += v[1];
                if (x < 0 || x > b || y < 0 || y > b) {
                    break;
                }
                for (int[] arrow : arrows) {
                    if (x == arrow[0] && y == arrow[1]) {
                        v[0] = arrow[2] - x;
                        v[1] = arrow[3] - y;
                        break;
                    }
                }
            }
            System.out.println(x + " " + y);
        }
    }
}
