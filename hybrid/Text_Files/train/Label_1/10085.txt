#include <cstdio>
#include <algorithm>

using namespace std;

const int MAXN = 1e3 + 10;

int a[MAXN];
int n;

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
    int ans = 0;
    for (int i = 2; i < n; ++i) {
        if ((a[i] > a[i - 1] && a[i] > a[i + 1]) || (a[i] < a[i - 1] && a[i] < a[i + 1])) ++ans;
    }
    printf("%d\n", ans);
    return 0;
}
