#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size(), ans = 0;
    int bal = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '0') {
            if (bal < 0) {
                ans++;
                bal += (1 << (n - 1 - i));
            }
        } else {
            bal -= (1 << (n - 1 - i));
            if (bal >= 0) ans++;
        }
    }
    cout << ans << endl;
}
