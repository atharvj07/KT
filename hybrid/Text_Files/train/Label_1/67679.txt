def f(n, r):
    s = n // r
    e = n % r
    if s % 2 == e % 2:
        ans = (s + 1) // 2 * (e + 1) // 2
    else:
        ans = s // 2 * (e + 1) // 2 + (s + 1) // 2 * e // 2
    if ans % 2 != n % 2:
        ans += 1
    return ans

t = int(input())
for _ in range(t):
    n, r = map(int, input().split())
    print(f(n, r))
