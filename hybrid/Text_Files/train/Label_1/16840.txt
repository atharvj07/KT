#include <bits/stdc++.h>
using namespace std;
#define int long long

const int M = 998244353;

int fastpow(int x, int p) {
    int ret = 1;
    while (p > 0) {
        if (p % 2 == 0) {
            x = (x * x) % M;
            p /= 2;
        } else {
            ret = (ret * x) % M;
            p--;
        }
    }
    return ret;
}

int32_t main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    map<int, int> cnt;
    int distinct = 0;
    for (auto x : v) {
        cnt[x]++;
        distinct += cnt[x] == 1;
    }
    int ans = 1;
    for (auto x : cnt) ans = (ans * fastpow(2, x.second - 1)) % M;
    cout << ans << "\n";
    return 0;
}
