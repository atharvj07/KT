#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    string s; cin >> s;
    int n; cin >> n;
    int cnt[26] = {};
    for (char c: s) cnt[c-'a']++;
    int sheets = 0, mx = 0;
    for (int i = 0; i < 26; i++) {
        sheets += cnt[i] / n;
        mx = max(mx, cnt[i] % n);
    }
    if (mx) sheets++;
    cout << sheets << endl;
    if (mx) {
        map<int, char> mp;
        int cur = 1;
        for (char c = 'a'; c <= 'z'; c++) {
            while (cnt[c-'a'] >= n * cur) {
                for (int i = 0; i < n; i++) { cout << c; cnt[c-'a']--; }
                mp[cur++] = c;
            }
        }
        for (char c: s) {
            if (mp.find(cnt[c-'a'] / n) != mp.end()) cout << c;
            cnt[c-'a']--;
        }
    } else {
        for (char c: s) cout << c;
    }
    cout << endl;
    return 0;
}
