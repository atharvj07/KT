import sys
n = int(sys.stdin.readline())
l = list(map(int, sys.stdin.readline().split()))
d = list(map(int, sys.stdin.readline().split()))
l.sort(reverse=True)
d.sort()
ans = 0
sum_l = 0
sum_d = 0
for i in range(n):
    if i * 2 >= n:  break
    sum_l += l[i]
    sum_d += d[i]
    if sum_l * 2 >= sum_d:  break
    ans += d[i]
print(ans)
