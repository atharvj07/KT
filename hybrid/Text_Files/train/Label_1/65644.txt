n, m = map(int, input().split())
a = list(map(int, input().split()))
for i in range(m):
    l, r = map(int, input().split())
    if any(a[j + 1] - a[j] < 0 for j in range(l - 1, r)):
        print("No")
    elif any(a[j + 1] - a[j] > 0 for j in range(l - 1, r - 1)):
        print("No")
    else:
        print("Yes")
