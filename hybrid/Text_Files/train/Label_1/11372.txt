#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;
int par[N], sz[N];

int root(int u)
{
    if (u == par[u])
        return u;
    return par[u] = root(par[u]);
}

void unite(int u, int v)
{
    u = root(u);
    v = root(v);
    if (u == v)
        return;
    if (sz[u] > sz[v])
        swap(u, v);
    par[u] = v;
    sz[v] += sz[u];
}

int main()
{
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        par[i] = i;
        sz[i] = 1;
    }
    for (int i = 1; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        unite(u, v);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++)
        if (root(i) == i)
            ans++;
    ans--;
    cout << ans << endl;
    return 0;
}
