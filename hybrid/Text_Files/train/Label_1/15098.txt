#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> x(m);
  for (int &i : x) cin >> i;
  vector<int> p(n);
  for (int i = 0; i < n; i++) p[i] = i + 1;
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m - 1; j++) ans[i] += abs(find(p.begin(), p.end(), x[j]) - find(p.begin(), p.end(), x[j + 1]));
    rotate(p.begin(), p.begin() + 1, p.end());
  }
  for (int i : ans) cout << i << " ";
}
