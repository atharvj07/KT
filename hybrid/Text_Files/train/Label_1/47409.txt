import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long p = sc.nextLong();
            long q = sc.nextLong();
            long x = p;
            while (x % q == 0) {
                x /= q;
            }
            System.out.println(x);
        }
    }
}
