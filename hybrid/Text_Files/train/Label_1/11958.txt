#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<int, int>

int main() {
    int n, k, m, t;
    cin >> n >> k >> m >> t;
    int l = n, pos = k;
    while (t--) {
        int x, y;
        cin >> x >> y;
        if (x == 1) {
            l++;
            if (y <= pos) { pos++; }
        } else {
            if (y == pos) {
                if (pos == 1) {
                    pos++;
                }
                l--;
                continue;
            }
            if (y == pos - 1) {
                pos--;
                l--;
                continue;
            }
            l--;
        }
        cout << l << " " << pos << endl;
    }
    return 0;
}
