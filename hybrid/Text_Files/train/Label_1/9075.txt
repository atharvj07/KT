#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int &i : a) cin >> i;
        while (k--) {
            int idx = 0;
            for (int i = 1; i < n; i++) {
                if (a[i] > a[idx]) idx = i;
            }
            if (a[idx] == 0) break;
            a[idx]++;
            a[0] = max(a[0] - 1, 0);
        }
        for (int i : a) cout << i << " ";
        cout << "\n";
    }
    return 0;
}
