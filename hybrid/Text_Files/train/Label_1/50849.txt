import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        int ans = 0;
        int[] order = new int[n];
        for (int i = 0; i < n; i++) {
            if (d >= a * x[i] + b * y[i]) {
                d -= a * x[i] + b * y[i];
                ans++;
                order[ans - 1] = i + 1;
            }
        }
        System.out.println(ans);
        for (int i = 0; i < ans; i++) {
            System.out.print(order[i] + " ");
        }
        System.out.println();
    }
}
