#include <iostream>
#include <vector>

int main()
{
    long n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    std::vector<bool> v(n, false);
    v[0] = v[n - 1] = true;
    for (long i = 0; i < n; ++i)
    {
        if (s[i] == '<')
            v[std::max(0L, i - 1)] = true;
        if (s[i] == '>')
            v[std::min(n - 1L, i + 1)] = true;
    }
    long ans = 0;
    for (bool x : v)
        if (x)
            ++ans;
    std::cout << ans << std::endl;
}
