import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        String s = in.next();
        int n = s.length(), sum = 0, temp = 0;
        boolean flag = true;
        char c;
        for(int i=0; i<n; i++){
            c = s.charAt(i);
            if(c>='a' && c<='z'){
                c -= 32;
            }
            int num = (int)c;
            if(num>=48 && num<=57){
                temp *= 10;
                temp += (int)(num-'0');
                flag = false;
            }
            else if(c=='.'){
                continue;
            }
            else{
                if(flag){
                    if(temp==0){
                        temp = 1;
                    }
                    sum += temp;
                }
                temp = 0;
                flag = true;
            }
        }
        if(flag==false){
            sum += temp;
        }
        System.out.println(sum);
    }
}
