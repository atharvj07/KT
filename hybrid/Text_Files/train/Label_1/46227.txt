import java.util.Scanner;

public class NikolayAndStaircases {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            scanner.nextLine();
            String staircases = scanner.nextLine();

            boolean[] visited = new boolean[2 * n];
            int maxRooms = 0;

            for (int j = 0; j < 2 * n; j++) {
                if (!visited[j]) {
                    int currentRoom = j;
                    visited[j] = true;
                    int roomsVisited = 1;

                    while (true) {
                        int nextRoom = getNextRoom(currentRoom, n, staircases, visited);
                        if (nextRoom == -1) {
                            break;
                        }
                        currentRoom = nextRoom;
                        visited[nextRoom] = true;
                        roomsVisited++;
                    }

                    maxRooms = Math.max(maxRooms, roomsVisited);
                }
            }

            System.out.println(maxRooms);
        }
    }

    private static int getNextRoom(int currentRoom, int n, String staircases, boolean[] visited) {
        int floor = currentRoom / n;
        int room = currentRoom % n;

        // Try to move to the right on the same floor
        if (room < n - 1 && !visited[floor * n + room + 1]) {
            return floor * n + room + 1;
        }

        // Try to move to the left on the same floor
        if (room > 0 && !visited[floor * n + room - 1]) {
            return floor * n + room - 1;
        }

        // Check for a staircase to the other floor
        if (staircases.charAt(room) == '1' && !visited[(floor + 1) * n + room]) {
            return (floor + 1) * n + room;
        }

        // No possible moves
        return -1;
    }
}
