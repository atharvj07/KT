n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
x = int(input())

# Precompute prefix sums for a and b
prefix_a = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_a[i] = prefix_a[i - 1] + a[i - 1]

prefix_b = [0] * (m + 1)
for i in range(1, m + 1):
    prefix_b[i] = prefix_b[i - 1] + b[i - 1]

# Function to compute all possible sums for an array using its prefix sums
def compute_sums(prefix):
    sums = set()
    length = len(prefix)
    for i in range(length):
        for j in range(i + 1, length):
            sums.add(prefix[j] - prefix[i])
    return sums

# Compute all possible sums for a and b
sums_a = compute_sums(prefix_a)
sums_b = compute_sums(prefix_b)

# For every sum in sums_a, find the maximum possible sum in sums_b that, 
# when multiplied, does not exceed x. This is to maximize the area.
max_area = 0
for sum_a in sums_a:
    for sum_b in sums_b:
        if sum_a * sum_b <= x:
            # Calculate the area (number of elements) of the subrectangle
            area = (prefix_a.index(prefix_a[prefix_a.index(sum_a) + 1]) - prefix_a.index(sum_a)) * \
                   (prefix_b.index(prefix_b[prefix_b.index(sum_b) + 1]) - prefix_b.index(sum_b))
            max_area = max(max_area, area)
print(max_area)
