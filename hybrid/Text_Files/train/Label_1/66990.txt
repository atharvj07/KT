t = int(input())
for _ in range(t):
    n, T = map(int, input().split())
    a = list(map(int, input().split()))
    painted = [0] * n
    if T % 2 == 0:
        half_T = T // 2
        count_half_T = 0
        for i in range(n):
            if a[i] < half_T:
                painted[i] = 0
            elif a[i] > half_T:
                painted[i] = 1
            else:
                painted[i] = count_half_T % 2
                count_half_T += 1
    else:
        for i in range(n):
            if a[i] < (T + 1) // 2:
                painted[i] = 0
            else:
                painted[i] = 1
    print(' '.join(map(str, painted)))
