#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n; string a, b;
int main() {
    cin >> n >> a >> b;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if ((a[i] != b[i] && a[j] != b[j] && (a[i] == '1' || a[j] == '1')) || (a[i] == b[i] && a[j] == b[j] && a[i] == '0')) ans++;
        }
    }
    cout << ans << endl;
}
