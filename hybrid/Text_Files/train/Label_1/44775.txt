import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        Map<String, Integer> points = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String name = sc.next();
            int p = sc.nextInt();

            points.put(name, Math.max(points.getOrDefault(name, 0), p));
        }

        int m = points.size();
        System.out.println(m);

        List<Map.Entry<String, Integer>> sorted = new ArrayList<>(points.entrySet());
        sorted.sort((a, b) -> b.getValue() - a.getValue());

        for (Map.Entry<String, Integer> entry : sorted) {
            String name = entry.getKey();
            int p = entry.getValue();

            int rank = 0;

            for (Map.Entry<String, Integer> other : sorted) {
                if (p < other.getValue()) {
                    rank++;
                }
            }

            String category;

            if (rank >= m * 0.5) {
                category = "noob";
            } else if (rank >= m * 0.2) {
                category = "random";
            } else if (rank >= m * 0.1) {
                category = "average";
            } else if (rank >= m * 0.01) {
                category = "hardcore";
            } else {
                category = "pro";
            }

            System.out.println(name + " " + category);
        }
    }
}
