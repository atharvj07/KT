#include <bits/stdc++.h>

using namespace std;

#define ll long long

ll n, k, ans, a[15], b[15], x[150005], y[150005], s[15], p[15], c[15], ss;
bool d[150005];

int main() {
  cin >> k;
  for (int i = 1; i <= k; i++) {
    cin >> n;
    for (int j = 1; j <= n; j++) {
      cin >> x[a[i] = b[++ans]];
      y[a[i]] = i;
      d[a[i]] = 1;
    }
    s[i] = s[i - 1];
    for (int j = a[i]; j >= a[i] - n + 1; j--) { p[s[i]] = y[j]; c[s[i]++] = j; }
    ss += n;
  }
  if (s[k] != ss || ss % k != 0) {
    cout << "No";
    return 0;
  }
  cout << "Yes" << endl;
  for (int i = 0; i < s[k]; i++) cout << c[i] << ' ' << p[i] << endl;
  return 0;
}
