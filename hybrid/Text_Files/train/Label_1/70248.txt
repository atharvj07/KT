import sys
input  = sys.stdin.readline

def solve():
    n, a, b, da, db = map(int, input().split())
    edges = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().split())
        edges[u].append(v)
        edges[v].append(u)
    vis = [False] * (n+1)
    dis_a = [-1] * (n+1)
    dis_b = [-1] * (n+1)
    def bfs(start, dis):
        q = [(start, 0)]
        while q:
            node, d = q.pop(0)
            dis[node] = d
            vis[node] = True
            for adj in edges[node]:
                if not vis[adj]:
                    q.append((adj, d+1))
    bfs(a, dis_a)
    vis = [False] * (n+1)
    bfs(b, dis_b)
    if dis_a[b] <= da: return "Alice"
    for adj in edges[b]:
        if dis_b[adj] > db:
            if dis_a[adj] <= da: return "Alice"
    return "Bob"

for _ in range(int(input())):
    print(solve())
