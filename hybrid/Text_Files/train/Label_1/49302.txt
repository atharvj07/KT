import java.util.Scanner;

public class Main {
  private static final int HEATING = 1;
  private static final int COOLING = -1;
  private static final int OFF = 0;

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int numTestCases = scanner.nextInt();
    for (int i = 0; i < numTestCases; i++) {
      int numCustomers = scanner.nextInt();
      int initialTemp = scanner.nextInt();
      Customer[] customers = new Customer[numCustomers];
      for (int j = 0; j < numCustomers; j++) {
        int visitTime = scanner.nextInt();
        int lowerBound = scanner.nextInt();
        int upperBound = scanner.nextInt();
        customers[j] = new Customer(visitTime, lowerBound, upperBound);
      }
      System.out.println(isPossible(initialTemp, customers) ? "YES" : "NO");
    }
  }

  private static boolean isPossible(int initialTemp, Customer[] customers) {
    int currentTime = 0;
    int currentTemp = initialTemp;
    int state = OFF;
    for (Customer customer : customers) {
      // Adjust the temperature to the customer's preferred range
      while (currentTemp < customer.lowerBound) {
        if (currentTime >= customer.visitTime) {
          return false;
        }
        currentTemp++;
        currentTime++;
        state = HEATING;
      }
      while (currentTemp > customer.upperBound) {
        if (currentTime >= customer.visitTime) {
          return false;
        }
        currentTemp--;
        currentTime++;
        state = COOLING;
      }

      // Move to the customer's visit time
      while (currentTime < customer.visitTime) {
        if (state == HEATING) {
          currentTemp++;
        } else if (state == COOLING) {
          currentTemp--;
        }
        currentTime++;
      }

      // Check if the customer is satisfied
      if (currentTemp < customer.lowerBound || currentTemp > customer.upperBound) {
        return false;
      }
    }
    return true;
  }

  private static class Customer {
    int visitTime;
    int lowerBound;
    int upperBound;

    Customer(int visitTime, int lowerBound, int upperBound) {
      this.visitTime = visitTime;
      this.lowerBound = lowerBound;
      this.upperBound = upperBound;
    }
  }
}
