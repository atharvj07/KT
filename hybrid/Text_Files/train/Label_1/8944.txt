#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 10;

int n, arr[MAXN];

int main() {
  int t;
  cin >> t;

  while (t--) {
    cin >> n;
    vector<int> a(2 * n);
    for (auto &x : a) { cin >> x; }

    sort(a.begin(), a.end());
    a.resize(unique(a.begin(), a.end()) - a.begin());

    int l = 0, r = a.size() - 1, ans = 1e9;
    while (l < r) {
      ans = min(ans, a[r] - a[l]);
      l++, r--;
    }

    cout << ans << '\n';
  }
  return 0;
}
