n, k = list(map(int, input().split()))
a = list(map(int, input().split()))
a.sort()

dif = 0
ans = 0

for i in range(n-1):
    dif = a[i+1] - a[i]
    if dif<=k:
        a[i+1] -= dif
        k -= dif
        ans += dif
    else:
        a[i+1] -= k
        ans += k
        k = 0
        break

print(ans)
