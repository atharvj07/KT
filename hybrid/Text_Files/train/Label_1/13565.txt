#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += a[i]; }
  if (sum % k != 0) {
    cout << "No" << endl;
    return 0;
  }
  int avg = sum / k;
  int curr = 0;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    curr += a[i];
    if (curr == avg) {
      ans.push_back(i - curr + 1);
      curr = 0;
    }
  }
  cout << "Yes" << endl;
  for (int i = 0; i < k; i++) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
