#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll N = 1e6;
ll n, a[N], l[N], r[N];
ll inver[N], inv = 0;

int main() {
  cin >> n;
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  iota(inver, inver + n, 0);
  sort(inver, inver + n, [&](const ll& i, const ll& j) { return a[i] < a[j]; });
  for (ll i = 0; i < n - 1; i++) {
    if (a[inver[i]] != a[inver[i + 1]]) l[inver[i + 1]] = inver[i] + 1, r[inver[i]] = inver[i + 1] - 1;
  }
  for (ll i = 0; i < n; i++) {
    if (!l[i]) cout << fixed << setprecision(9) << (double)a[i] << "\n";
    else {
      ll sum = 0, cnt = 0;
      for (ll j = l[i]; j <= r[i]; j++) { sum += a[j], cnt++; }
      cout << fixed << setprecision(9) << (double)sum / cnt << "\n";
    }
  }
  return 0;
}
