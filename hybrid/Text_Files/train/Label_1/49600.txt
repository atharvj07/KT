import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[] p = new int[n + 1], s = new int[n + 1], c = new int[k + 1];
        for (int i = 1; i <= n; i++) p[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) s[i] = sc.nextInt();
        for (int i = 1; i <= k; i++) c[i] = sc.nextInt();
        
        int ans = 0;
        int[] r = new int[m + 1];
        for (int i = 1; i <= n; i++) {
            if (r[s[i]] == 0 || p[i] > p[r[s[i]]]) r[s[i]] = i;
        }
        
        for (int i = 1; i <= k; i++) {
            if (r[s[c[i]]] != c[i]) ans++;
        }
        
        System.out.print(ans);
    }
}
