import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int hh = sc.nextInt();
        int mm = sc.nextInt();

        int count = 0;
        while (true) {
            if (hh < 10 && hh % 10 == 7 || hh >= 10 && hh / 10 == 7 || mm < 10 && mm % 10 == 7 || mm >= 10 && mm / 10 == 7) {
                System.out.println(count);
                break;
            }
            mm -= x;
            count++;
            if (mm < 0) {
                mm += 60;
                hh--;
                if (hh < 0) {
                    hh = 23;
                }
            }
        }
    }
}
