#include<bits/stdc++.h>

using namespace std ;

typedef pair < int , int > pii ;

const int N = 55 ;
char grid [ N ][ N ] ;
bool vis [ N ][ N ] ;
int n , m ;
int dx [] = { -1 , 0 , 1 , 0 } ;
int dy [] = { 0 , 1 , 0 , -1 } ;

bool is_valid ( int i , int j ) {
  return ( i >= 0 && i < n && j >= 0 && j < m && grid [ i ][ j ] == 'B' ) ;
}

bool dfs ( int i , int j ) {
  if ( vis [ i ][ j ] )
    return false ;
  vis [ i ][ j ] = true ;
  for ( int d = 0 ; d < 4 ; d ++ ) {
    int ni = i + dx [ d ] , nj = j + dy [ d ] ;
    if ( is_valid ( ni , nj ) ) {
      if ( !dfs ( ni , nj ) )
        return false ;
    }
  }
  return true ;
}

int main () {
  cin >> n >> m ;
  for ( int i = 0 ; i < n ; i ++ ) cin >> grid [ i ] ;
  bool ok = true ;
  for ( int i = 0 ; i < n ; i ++ ) {
    for ( int j = 0 ; j < m ; j ++ ) {
      if ( !vis [ i ][ j ] && grid [ i ][ j ] == 'B' ) {
        if ( !dfs ( i , j ) ) {
          ok = false ;
          break ;
        }
      }
    }
    if ( !ok )
      break ;
  }
  if ( ok ) cout << "YES" << endl ;
  else cout << "NO" << endl ;
  return 0 ;
}
