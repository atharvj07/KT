#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll dp[10005][101][2];
ll solve(string &s,ll pos,ll plus,bool lastop){
	if(pos==s.size()){
		if(plus==0)return 0;
		else return LLONG_MIN;
	}
	if(dp[pos][plus][lastop]!=-1)return dp[pos][plus][lastop];
	ll ans=LLONG_MIN;
	ll val=0;ll neg=0;
	for(ll i=pos;i<s.size();i++){
		if(s[i]=='('){
			neg++;
		}
		else if(s[i]==')'){
			neg--;
		}
		else{
			val=val*10+(s[i]-'0');
		}
		if(!neg){
			if(lastop==0){
				ans=max(ans,solve(s,i+1,plus,0)+val);
				if(plus>0)
				ans=max(ans,solve(s,i+1,plus-1,1)-val);
			}
			else{
				ans=max(ans,solve(s,i+1,plus,1)-val);
				if(plus>0)
				ans=max(ans,solve(s,i+1,plus-1,0)+val);
			}
		}
	}
	return dp[pos][plus][lastop]=ans;
}
int main(){
	string s;cin>>s;
	ll p,m;cin>>p>>m;
	memset(dp,-1,sizeof(dp));
	cout<<solve(s,0,p,0)<<endl;
    return 0;
}
