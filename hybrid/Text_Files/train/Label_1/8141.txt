#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10, maxbit = 32;
int n, m, k, a[maxn], dp[maxbit][maxn];
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  sort(a + 1, a + n + 1);
  for (int i = 1; i <= n; i++)
    for (int j = 0; j < maxbit; j++)
      dp[j][i] = dp[j][i - 1];
  for (int i = 1; i <= n; i++) {
    for (int j = maxbit - 1; j >= 0; j--) {
      if ((a[i] >> j) & 1) { dp[j][i] += 1; }
      if (i >= m && dp[j][i] - dp[j][i - m] >= k) {
        cout << i - m << endl;
        return 0;
      }
    }
  }
  cout << n << endl;
  return 0;
}
