import sys

t = int(input())

for _ in range(t):
    n, m, x, y = map(int, input().split())
    s = ''.join(input() for _ in range(n))
    l = 0
    for i in range(n):
        for j in range(m):
            if s[i * m + j] != '*':
                if j == m - 1 or s[i * m + j + 1] == '*':
                    l += x
                elif i == n - 1 or s[(i + 1) * m + j] == '*':
                    l += x
                else:
                    l += y
    sys.stdout.write(str(l))
    sys.stdout.write('\n')
