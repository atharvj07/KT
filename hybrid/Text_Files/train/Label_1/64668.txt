import bisect

n = int(input())
a = list(map(int, input().split()))
l = [a[0]]
pos = [0] * n
for i in range(1, n):
    if a[i] > l[-1]:
        l.append(a[i])
        pos[i] = len(l) - 1
    else:
        index = bisect.bisect_left(l, a[i])
        l[index] = a[i]
        pos[i] = index
sub = [0] * n
for i in range(n - 1, -1, -1):
    if pos[i] == len(l) - 1:
        sub[i] = 3
    else:
        sub[i] = sub[i + 1]
for x in sub:
    print(x, end = '')
