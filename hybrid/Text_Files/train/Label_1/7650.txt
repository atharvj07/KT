#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5; const long long mod = 998244353;
long long a[N], b[N], f[N], g[N], inv[N];
int main() {
  int n, m; cin >> n >> m;
  for(int i = 1; i <= n; i++) cin >> a[i];
  for(int i = 1; i <= m; i++) cin >> b[i];
  f[0] = g[n + 1] = inv[0] = inv[1] = 1;
  for(int i = 1; i <= n; i++) f[i] = f[i - 1] * i % mod;
  for(int i = 2; i <= n; i++) inv[i] = (mod - mod / i) * inv[mod % i] % mod;
  for(int i = n - 1; i >= 0; i--) g[i] = g[i + 1] * i % mod;
  int l = 1, r = 1, ans = 1, cnt = 1;
  for(int i = 1; i <= m; i++) {
    while(r <= n && a[l] <= a[r] && (l < r || a[r] <= b[i])) r++;
    while(l > 1 && a[l - 1] <= a[l]) l--;
    if(l > r || a[r] > b[i]) { ans = 0; break; }
    ans = ans * f[r - l + cnt] % mod * inv[cnt] % mod * inv[r - l] % mod;
    l = r + 1; cnt++;
  }
  cout << ans << endl;
  return 0;
}
