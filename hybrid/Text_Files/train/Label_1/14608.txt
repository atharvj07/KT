#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    set<int> s;
    while (n--) {
        string c;
        int p;
        cin >> c >> p;
        if (c == "-") {
            s.erase(p);
        } else {
            s.insert(p);
        }
        cout << s.size() << endl;
    }
}
