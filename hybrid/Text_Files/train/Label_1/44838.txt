import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

class Main {
    static class Pair {
        int x, y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Pair[] intervals = new Pair[n];
        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            intervals[i] = new Pair(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
        }
        int q = Integer.parseInt(br.readLine());
        int[][] queries = new int[q][2];
        for (int i = 0; i < q; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            queries[i][0] = Integer.parseInt(st.nextToken());
            queries[i][1] = Integer.parseInt(st.nextToken());
        }
        Arrays.sort(queries, (a, b) -> Integer.compare(a[1], b[1]));
        int[] answer = new int[q];
        int maxRight = 0;
        for (int i = 0; i < q; i++) {
            int[] query = queries[i];
            int left = query[0];
            int right = query[1];
            while (maxRight < right) {
                maxRight++;
                int leftInterval = -1;
                int rightInterval = -1;
                for (int j = 0; j < n; j++) {
                    int curLeft = intervals[j].x;
                    int curRight = intervals[j].x + intervals[j].y;
                    if (curLeft <= left && curRight >= left) {
                        leftInterval = j;
                        rightInterval = j;
                        break;
                    }
                    if (curLeft <= right && curRight >= right) {
                        rightInterval = j;
                        break;
                    }
                }
                if (leftInterval != -1 && rightInterval != -1) {
                    int leftLength = intervals[leftInterval].y;
                    int rightLength = intervals[rightInterval].y;
                    int minLeftLength = left - intervals[leftInterval].x;
                    int minRightLength = right - intervals[rightInterval].x;
                    if (leftLength < minLeftLength) {
                        answer[i] += minLeftLength - leftLength;
                    }
                    if (rightLength < minRightLength) {
                        answer[i] += minRightLength - rightLength;
                    }
                } else {
                    int leftLength = left - intervals[leftInterval].x;
                    int rightLength = right - intervals[rightInterval].x;
                    if (leftLength < rightLength) {
                        answer[i] += rightLength - leftLength;
                    }
                }
            }
        }
        for (int i = 0; i < q; i++) {
            System.out.println(answer[i]);
        }
    }
}
