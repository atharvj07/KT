n = int(input())
a = list(map(int, input().split()))
a.sort()

def game_winner(n, a):
    # Check for initial conditions where CSL wins directly
    if a[0] == 0 and a[1] == 0:
        return "cslnb"
    for i in range(1, n):
        if a[i] == a[i-1]:
            if i > 1 and a[i-2] == a[i] - 1:
                return "cslnb"
            if a.count(a[i]) > 2:
                return "cslnb"
            break
    # Calculate the sum of stones and determine the winner based on parity
    sum_of_stones = sum(a) - (n * (n - 1) // 2)
    if sum_of_stones % 2 == 0:
        return "cslnb"
    else:
        return "sjfnb"

print(game_winner(n, a))
