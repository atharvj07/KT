#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;

int n, m, k;
vector<int> sp;
vector<int> adj[MAXN];
int dis1[MAXN], disn[MAXN];
bool vis[MAXN];

void bfs(int st, int dis[]) {
    queue<int> q;
    q.push(st);
    vis[st] = 1;
    while (!q.empty()) {
        int cur = q.front();
        q.pop();
        for (int i = 0; i < adj[cur].size(); i++) {
            int nxt = adj[cur][i];
            if (!vis[nxt]) {
                vis[nxt] = 1;
                dis[nxt] = dis[cur] + 1;
                q.push(nxt);
            }
        }
    }
}

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < k; i++) {
        int x;
        cin >> x;
        sp.push_back(x);
    }
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    bfs(1, dis1);
    memset(vis, 0, sizeof(vis));
    bfs(n, disn);
    int ans = 0;
    for (int i = 0; i < sp.size(); i++) {
        for (int j = i + 1; j < sp.size(); j++) {
            int x = sp[i], y = sp[j];
            if (x != y) ans = max(ans, dis1[x] + disn[y] + 1);
        }
    }
    cout << ans << endl;
    return 0;
}
