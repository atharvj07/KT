#include <iostream>
#include <cmath>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, s;
    cin >> n >> s;
    long long ans = 0;
    long long sum = 0;
    while (sum <= s) {
      long long cur = n;
      sum = 0;
      while (cur > 0) {
        sum += cur % 10;
        cur /= 10;
      }
      n++;
      ans++;
    }
    cout << ans - 1 << '\n';
  }
}
