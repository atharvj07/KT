#include <bits/stdc++.h>
using namespace std;
const long long inf = 1e18;
int main() {
    long long n, x, y = 0, l = 0, r = inf;
    set<long long> s;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        if (s.insert(x).second) {
            y |= x;
            if (y & (y + 1))
                r = min(r, l);
            l = (y & -y);
        }
    }
    if (r == inf)
        cout << -1;
    else {
        for (long long i = l; i <= r; i <<= 1) {
            cout << __builtin_popcountll(i) << " ";
        }
    }
    return 0;
}
