#include<bits/stdc++.h>
using namespace std;
int main(){
    int n, k; cin >> n >> k;
    string s; cin >> s;
    cout << "PRINT " << s[k - 1] << endl;
    for(int i = k; i != 1; i--) cout << "LEFT" << endl;
    for(int i = k; i != n; i++){
        cout << "RIGHT" << endl;
        cout << "PRINT " << s[i] << endl;
    }
}
