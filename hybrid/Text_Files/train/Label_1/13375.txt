#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    int j = i;
    while (j < n && a[j] != b[i]) { j++; }
    if (j < n) { ans[i] = j - i + 1; }
  }
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
}
