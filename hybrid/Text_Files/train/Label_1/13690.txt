#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define sc(x) scanf("%lld",&x)
int main()
{
	ll n,i,x,ans,mn=1e9;
	sc(n);
	vector<ll>v;
	v.push_back(0);
	for(i=0;i<n;++i)
	{
		sc(x);
		v.push_back(x);
	}
	for(i=2;i<n;++i)
	{
		mn=min(v[i]-v[i-2],mn);
	}
	printf("%lld\n",mn);
	
	return 0;
}
