for t in range(int(input())):
    n = int(input())
    a = [list(map(int,input().split())) for i in range(n)]
    a.sort()
    j = -1
    for i in range(n):
        if j+1<i:
            print(0,end=' ')
        else:
            if a[i][0]>=a[j+1][1]:
                j=j+1
                print(a[i][0],end=' ')
            else:
                print(0,end=' ')
    print()
