n, k = map(int, input().split())
powers = list(map(int, input().split()))

current_winner = powers[0]
win_count = 0

for i in range(1, n):
    if win_count == k:
        break
    if current_winner > powers[i]:
        win_count += 1
    else:
        current_winner = powers[i]
        win_count = 1

print(current_winner)
