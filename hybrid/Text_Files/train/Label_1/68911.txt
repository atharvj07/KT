k=int(input())
s=input()
l=[]
for x in s.split():
 l+=x.split('-')
i=1
w=0
while i<k:
 t=0
 for y in l:
  if t+len(y)+1<=i:t+=len(y)+1
  else:break
 i+=1
 w=max(w,t)
 t=0
 for y in l:
  if t+len(y)+1<=k:t+=len(y)+1
  else:break
 i+=1
 w=max(w,t)
 l.reverse()
i=1
t=0
for y in l:
 if t+len(y)+1<=i:t+=len(y)+1
 else:break
 i+=1
 w=max(w,t)
t=0
for y in l:
 if t+len(y)+1<=k:t+=len(y)+1
 else:break
 i+=1
 w=max(w,t)
l.reverse()
print(w)
