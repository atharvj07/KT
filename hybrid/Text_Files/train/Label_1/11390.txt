#include <iostream>
#include <unordered_set>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  unordered_set<int> s;
  for (int i = 0; i < n; i++) {
    if (s.count(a[i])) {
      cout << "NO" << endl;
      return 0;
    }
    s.insert(a[i]);
  }
  cout << "YES" << endl;
  return 0;
}
