#include <iostream>
#include <vector>

using namespace std;

int main() {
  int f, i, t;
  cin >> f >> i >> t;
  vector<int> likes(i);
  for (int j = 0; j < f; ++j) {
    string s;
    cin >> s;
    for (int k = 0; k < i; ++k) likes[k] += s[k] == 'Y';
  }
  int ans = 0;
  for (int j = 0; j < i; ++j) ans += likes[j] >= t;
  cout << ans << endl;
  return 0;
}
