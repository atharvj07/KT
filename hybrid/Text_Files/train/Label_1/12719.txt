#include <bits/stdc++.h>
using namespace std;
int main() {
    int x[3], y[3];
    for (int i = 0; i < 3; ++i) cin >> x[i] >> y[i];
    int p[3] = {0, 1, 2};
    int a = x[0] * y[0], b = x[1] * y[1], c = x[2] * y[2], ab = a * b, bc = b * c, ac = a * c;
    do {
        int s = sqrt(bc + ac + ab);
        if (s * s == bc + ac + ab) {
            cout << s << "\n";
            for (int i = 0; i < s; ++i) {
                for (int j = 0; j < s; ++j) {
                    if (i < y[p[0]] && j < x[p[0]])
                        cout << "A";
                    else if (i < y[p[1]] && j >= x[p[0]] && j < x[p[0]] + x[p[1]])
                        cout << "B";
                    else if (i < y[p[2]] && j >= x[p[0]] + x[p[1]] && j < x[p[0]] + x[p[1]] + x[p[2]])
                        cout << "C";
                    else
                        cout << ".";
                }
                cout << "\n";
            }
            return 0;
        }
    } while (next_permutation(p, p + 3));
    cout << -1 << "\n";
    return 0;
}
