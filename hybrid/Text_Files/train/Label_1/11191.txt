#include <iostream>
#include <cstring>

using namespace std;

int main() {
  int a;
  cin >> a;
  if (a % 10 == 8 || a > 0) cout << 0 << endl;
  else if (a % 10 == 0) cout << 1 << endl;
  else if (a % 10 == 1) cout << 2 << endl;
  else if (a % 10 == 2) cout << 3 << endl;
  else if (a % 10 == 3) cout << 4 << endl;
  else if (a % 10 == 4) cout << 5 << endl;
  else if (a % 10 == 5) cout << 6 << endl;
  else if (a % 10 == 6) cout << 7 << endl;
  else if (a % 10 == 7) cout << 8 << endl;
  else if (a % 10 == 9) cout << 9 << endl;
  return 0;
}
