import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    i = 0
    j = n - 1
    a_cnt = 0
    b_cnt = 0
    tot = 0
    while i <= j:
        if a_cnt <= b_cnt:
            tot += a[i]
            a_cnt += a[i]
            i += 1
        else:
            tot += a[j]
            b_cnt += a[j]
            j -= 1
    print(i + j, a_cnt, b_cnt)
