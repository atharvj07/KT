#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
  int n,k;
  cin>>n>>k;
  vector<long long>v(n);
  for(long long &x:v)cin>>x;
  sort(v.begin(),v.end());
  vector<vector<long long>>a(n/k);
  int j=0;
  for(int i=0;i<n;++i){
    a[j++].push_back(v[i]);
    if(j==n/k){j=0;}
  }
  long long ans=0;
  for(int i=0;i<n/k;++i){
    ans=max(ans,(a[i][k-1])-(a[i][0]));
  }
  cout<<ans<<'\n';
  return 0;
}
