def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    rootX = find(parent, x)
    rootY = find(parent, y)
    if rootX != rootY:
        if rank[rootX] > rank[rootY]:
            parent[rootY] = rootX
        elif rank[rootX] < rank[rootY]:
            parent[rootX] = rootY
        else:
            parent[rootY] = rootX
            rank[rootX] += 1

def kruskal(edges, n):
    parent = [i for i in range(n)]
    rank = [0] * n
    cost = 0
    for edge in edges:
        w, x, y = edge
        if find(parent, x) != find(parent, y):
            cost += w
            union(parent, rank, x, y)
    return cost

n, m = map(int, input().split())
a = list(map(int, input().split()))

edges = []
for i in range(1, n + 1):
    edges.append((a[i - 1], 0, i))
for _ in range(m):
    x, y, w = map(int, input().split())
    edges.append((w, x, y))

edges.sort()
print(kruskal(edges, n + 1))
