import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double d = sc.nextInt();
        int count = 0;
        for (int i = 0; i < n; i++) {
            double x = sc.nextInt();
            double y = sc.nextInt();
            double dist = Math.sqrt(x * x + y * y);
            if (dist <= d) {
                count++;
            }
        }
        System.out.println(count);
    }
}
