#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n, e;
    cin >> n >> e;
    vi a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    for (int i = 0; i < n; i++) {
      bool ok = 1;
      ll v = 1;
      for (int j = 0; j <= e; j++) {
        int pos = i + j * e;
        if (pos >= n) {
          ok = 0;
          break;
        }
        v *= a[pos];
      }
      if (!ok) continue;
      for (ll j = 2; j * j <= v; j++)
        if (v % j == 0) {
          ok = 0;
          break;
        }
      if (ok) ans++;
    }
    cout << ans << '\n';
  }
  return 0;
}
