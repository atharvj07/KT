import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        char[] arr = input.next().toCharArray();
        int count = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 'X') {
                ans += i % 2 == count % 2 ? 0 : 1;
                count++;
            }
        }
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            System.out.print(i % 2 == count % 2 ? 'X' : 'x');
        }
    }
}
