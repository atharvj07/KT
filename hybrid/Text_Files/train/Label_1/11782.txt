#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int A, B, n;
    cin >> A >> B >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    long long hero = B;
    bool ok = true;
    for (int i = 0; i < n; i++) {
      if (ok) {
        while (b[i] > 0) {
          hero -= a[i];
          b[i] -= A;
          if (hero <= 0) {
            ok = false;
            break;
          }
        }
      }
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
  return 0;
}
