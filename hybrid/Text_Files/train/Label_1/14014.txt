#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    ll n,s,l;
    cin>>n>>s>>l;
    vector<ll>a(n);
    for(ll i=0;i<n;i++){
        cin>>a[i];
    }
    sort(a.begin(),a.end());
    ll ans=LLONG_MAX;
    for(ll i=0;i<=n-l;i++){
        if(a[i+l-1]-a[i]<=s){
            ans=min(ans,1+(n-i-l)/l);
        }
    }
    if(ans==LLONG_MAX) ans=-1;
    cout<<ans<<endl;
    return 0;
}
