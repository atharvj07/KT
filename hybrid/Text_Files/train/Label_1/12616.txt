#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> segments(n);
  for (int i = 0; i < n; i++) {
    cin >> segments[i].first >> segments[i].second;
  }
  sort(segments.begin(), segments.end());
  int ans = -1;
  int max_r = -1;
  for (int i = 0; i < n; i++) {
    if (segments[i].first <= max_r) {
      ans = i + 1;
    } else {
      max_r = segments[i].second;
    }
  }
  cout << ans << endl;
  return 0;
}
