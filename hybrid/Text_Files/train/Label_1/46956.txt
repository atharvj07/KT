import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = sc.nextInt();
            arr[i][1] = sc.nextInt();
        }
        Arrays.sort(arr, (a, b) -> a[1] - b[1]);
        int cnt = 0, last = 0;
        for (int[] a : arr) {
            if (a[0] >= last) {
                cnt++;
                last = a[1];
            }
        }
        System.out.println(cnt);
    }
}
