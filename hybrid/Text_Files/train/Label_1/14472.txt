#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main(){
    int n;
    cin>>n;
    vector<int>a;
    a.resize(n);
    for (int i=0;i<n;i++) cin>>a[i];
    vector<int>dp(n,1),id(n,0);
    int mx=1;
    for (int i=1;i<n;i++){
        for (int j=i-1;j>=0;j--){
            if (a[i]>a[j] and dp[i]<dp[j]+1){
                dp[i]=dp[j]+1;
                id[i]=j;
                if (dp[i]>mx) mx=dp[i];
            }
        }
    }
    cout<<mx<<endl;
    int st=0,ind=0;
    for (int i=0;i<n;i++){
        if (dp[i]>dp[ind]) ind=i;
    }
    while (ind!=0){
        cout<<ind+1<<" ";
        ind=id[ind];
    }
    cout<<ind+1;

    return 0;
}
