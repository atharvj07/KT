#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(2 * n);
    vector<int> pos(2 * n);
    for (int i = 0; i < 2 * n; i++) {
        cin >> a[i];
        pos[a[i]] = i;
    }
    int ans = 0;
    for (int i = 0; i < 2 * n;) {
        int j = i + 1;
        while (j < 2 * n && a[j] == a[i]) j++;
        if (j - i > 2) ans += (j - i) / 2 - 1;
        i = j;
    }
    cout << ans << endl;
    return 0;
}
