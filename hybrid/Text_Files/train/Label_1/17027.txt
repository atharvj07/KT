#include <iostream>
#include <vector>
using namespace std;

int n, m;
vector<bool> prime;

void sieve(int n){
    prime.resize(n + 1, true);
    for(int p = 2; p * p <= n; p++)
        if(prime[p])
            for(int i = p * p; i <= n; i += p)
                prime[i] = false;
}

int main(){
    cin >> n;
    sieve(1003);
    for(m = 1; m <= 1000; m++)
        if(!prime[n * m + 1])
            break;
    cout << m << endl;
}
