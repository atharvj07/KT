import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int k = 0;
        boolean[][] points = new boolean[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                if (i == 0 || j == 0 || i == j) {
                    points[i][j] = true;
                    k++;
                }
            }
        }

        if (n > 1 && m > 1) {
            points[1][m] = true;
            points[n][m] = true;
            points[n][1] = true;
            k += 3;
        }

        System.out.println(k);
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                if (points[i][j]) {
                    System.out.println(i + " " + j);
                }
            }
        }
    }
}
