import java.util.Scanner;

public class RailroadDoorways {

    private static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Line {
        Point start, end;
        int owner, elevation;

        public Line(Point start, Point end, int owner, int elevation) {
            this.start = start;
            this.end = end;
            this.owner = owner;
            this.elevation = elevation;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int datasets = scanner.nextInt();

        for (int i = 0; i < datasets; i++) {
            Point a = new Point(scanner.nextInt(), scanner.nextInt());
            Point b = new Point(scanner.nextInt(), scanner.nextInt());
            int lines = scanner.nextInt();
            Line[] existingLines = new Line[lines];

            for (int j = 0; j < lines; j++) {
                Point start = new Point(scanner.nextInt(), scanner.nextInt());
                Point end = new Point(scanner.nextInt(), scanner.nextInt());
                int owner = scanner.nextInt();
                int elevation = scanner.nextInt();
                existingLines[j] = new Line(start, end, owner, elevation);
            }

            int doorways = findMinimumDoorways(a, b, existingLines);
            System.out.println(doorways);
        }
    }

    private static int findMinimumDoorways(Point a, Point b, Line[] existingLines) {
        int doorways = 0;

        // Check intersections with existing lines
        for (Line line : existingLines) {
            if (doIntersect(a, b, line.start, line.end)) {
                // Determine elevation of the new line at the intersection
                int newElevation = line.owner == 1 ? line.elevation : 1 - line.elevation;

                // If the elevation changes, add a doorway
                if (newElevation != a.y) {
                    doorways++;
                }
            }
        }

        return doorways;
    }

    // Line intersection check using the determinant method
    private static boolean doIntersect(Point p1, Point p2, Point p3, Point p4) {
        int d1 = determinant(p1, p2, p3);
        int d2 = determinant(p1, p2, p4);
        int d3 = determinant(p3, p4, p1);
        int d4 = determinant(p3, p4, p2);

        return (d1 * d2 <= 0) && (d3 * d4 <= 0);
    }

    private static int determinant(Point p1, Point p2, Point p3) {
        return (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
    }
}
