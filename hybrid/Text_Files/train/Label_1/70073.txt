n = int(input())
l = list(map(int, input().split()))
m = int(input())
a = list(map(int, input().split()))

if a[0] != 1 or a[-1] != n:
    print(-1)
else:
    ans = l[n - 1] - l[0]
    for i in range(len(a) - 1):
        if a[i] == a[i + 1]:
            ans -= l[a[i] - 1] - l[a[i] - 2]
    print(ans)
