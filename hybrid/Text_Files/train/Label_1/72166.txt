n = int(input().strip())
s = input().strip()
t = input().strip()

def count_possible_words(s, t, n):
    diff = []
    for i in range(n):
        if s[i] != t[i]:
            diff.append(i)
    if len(diff) > 1:
        return 0
    if len(diff) == 1:
        return 2
    return 1

if n == 1:
    print(2)
else:
    if s[1:] == t[1:]:
        print(3)
    elif s[:-1] == t[:-1]:
        print(3)
    elif s[1:] == t[:-1] or s[:-1] == t[1:]:
        print(2)
    else:
        print(count_possible_words(s, t, n))
