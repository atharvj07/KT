#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    int mx = INT_MIN, mn = INT_MAX;
    while (n--) {
        char c; int x; cin >> c >> x >> c;
        switch (c) {
            case '<': mx = min(mx, x - 1); break;
            case '>': mn = max(mn, x + 1); break;
            case '<=': mx = min(mx, x); break;
            case '>=': mn = max(mn, x); break;
        }
    }
    if (mn > mx) cout << "Impossible" << endl;
    else cout << mn << endl;
}
