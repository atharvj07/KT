import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] count = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            count[i] = a[i];
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (count[i] > 0) {
                ans++;
                count[i]--;
                if (i > 1 && count[i - 1] > 0) {
                    count[i - 1]--;
                }
                if (i < n && count[i + 1] > 0) {
                    count[i + 1]--;
                }
            }
        }

        System.out.println(ans);
    }
}
