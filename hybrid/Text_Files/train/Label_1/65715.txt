from sys import stdin
n = int(stdin.readline())
a, b = [], []
for _ in range(n):
    f, i, j = stdin.readline().split()
    a.append((i, int(f == 'F')))
    b.append((j, int(f == 'F')))
a.sort()
b.sort(reverse=True)
ans = 0
l, r = 0, 0
for aa, bb in zip(a, b):
    if aa[1] == bb[1]:
        if aa[0] <= bb[0]:
            l += 1
        else:
            r += 1
    elif aa[1] != bb[1]:
        if aa[0] >= bb[0]:
            l += 1
        else:
            r += 1
    ans = max(ans, min(l, r) * 2)
print(ans)
