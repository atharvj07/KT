import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] nums = new int[N];
        for (int i = 0; i < N; i++) nums[i] = sc.nextInt();
        boolean pcp = true, scp = true;
        int g = nums[0];
        for (int i = 1; i < N; i++) {
            g = gcd(g, nums[i]);
            if (g == 1) pcp = false;
        }
        if (!pcp) {
            for (int i = 1; i < N; i++) {
                if (gcd(g, nums[i]) != 1) scp = false;
            }
        }
        System.out.println(pcp ? "pairwise coprime" : scp ? "setwise coprime" : "not coprime");
    }

    private static int gcd(int a, int b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
