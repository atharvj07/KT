#include <iostream>
#include <vector>
using namespace std;
int main() {
  string s;
  int nb, ns, nc, pb, ps, pc, r;
  cin >> s >> nb >> ns >> nc >> pb >> ps >> pc >> r;
  vector<int> cnt(4, 0);
  cnt[0] = r / pb;
  for (char c : s) {
    cnt[c - 'A' + 1] = min(cnt[c - 'A' + 1], cnt[c - 'A']);
  }
  cout << cnt[s.front() - 'A' + 1] << endl;
  return 0;
}
