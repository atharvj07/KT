#include<bits/stdc++.h>
using namespace std;
#define endl '\n'

void solve(){
    int n, x, y;
    cin >> n >> x >> y;
    int dif = y - x;
    cout << x << " ";
    bool fl = false;
    for(int i = x + dif; i <= y; i += dif){
        if(i == y){
            fl = true;
        }
        cout << i << " ";
    }
    for(int i = x - dif; i >= 1; i -= dif){
        if(n == 1 && fl){
            break;
        }
        cout << i << " ";
        n--;
    }
    cout << endl;
}

int main(){
    int t;
    cin >> t;
    while(t--){
        solve();
    }
    return 0;
}
