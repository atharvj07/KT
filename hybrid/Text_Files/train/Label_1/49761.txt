import java.util.Scanner;

public class BeautifulInteger {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        String xStr = scanner.next();
        char[] x = xStr.toCharArray();

        // Find the first index where the pattern breaks
        int breakIndex = -1;
        for (int i = 0; i < n - k; i++) {
            if (x[i] != x[i + k]) {
                breakIndex = i;
                break;
            }
        }

        // If the pattern never breaks, the answer is x
        if (breakIndex == -1) {
            System.out.println(n);
            System.out.println(xStr);
            return;
        }

        // Find the next digit in the pattern
        char nextDigit = x[breakIndex + k];

        // Create a new string with the next digit repeated k times
        StringBuilder yStr = new StringBuilder();
        for (int i = 0; i < k; i++) {
            yStr.append(nextDigit);
        }

        // If the new string is greater than or equal to x, it is the answer
        if (yStr.toString().compareTo(xStr) >= 0) {
            System.out.println(k);
            System.out.println(yStr);
            return;
        }

        // Otherwise, increment the last digit of the new string and repeat the process
        yStr.setCharAt(k - 1, (char) (yStr.charAt(k - 1) + 1));
        while (yStr.toString().compareTo(xStr) < 0) {
            yStr.setCharAt(k - 1, (char) (yStr.charAt(k - 1) + 1));
        }

        System.out.println(k);
        System.out.println(yStr);
    }
}
