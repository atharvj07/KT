#include <iostream>
#include <vector>
#include <cstdio>

const int MOD = 1e9 + 7;
typedef long long ll;

inline int add(ll a, ll b) { return (a += b) < MOD ? a : a - MOD; }
inline int sub(ll a, ll b) { return (a -= b) < 0 ? a + MOD : a; }
inline int mul(ll a, ll b) { return (a * b) % MOD; }
inline int pow(ll a, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) ans = mul(ans, a);
        a = mul(a, a);
        b >>= 1;
    }
    return ans;
}

inline int inv(int a) { return pow(a, MOD - 2); }

int main() {
    int k;
    scanf("%d", &k);
    ll odd = 0, pair = 1;
    for (int i = 0; i < k; ++i) {
        ll a;
        scanf("%lld", &a);
        if (a & 1) ++odd;
        pair = mul(pair, a);
    }
    int den = pair;
    if (odd & 1) pair = sub(pair, 1);
    den = inv(den);
    pair = mul(pair, den);
    printf("%d/%d\n", pair, den);
    return 0;
}
