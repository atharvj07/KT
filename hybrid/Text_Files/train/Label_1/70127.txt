def can_meet(board):
    knight_positions = [(i, j) for i in range(8) for j in range(8) if board[i][j] == 'K']
    return (knight_positions[0][0] + knight_positions[0][1]) % 2 == (knight_positions[1][0] + knight_positions[1][1]) % 2

t = int(input().strip())
for _ in range(t):
    board = [input().strip() for _ in range(8)]
    if _ < t - 1:
        input()  # Skip empty line
    print("YES" if can_meet(board) else "NO")
