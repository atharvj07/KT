import java.util.*;

public class Main {
    static int n, m;
    static int[] a;
    static int[][] pairs;

    static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        a = new int[n + 1];
        pairs = new int[m][2];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            pairs[i][0] = in.nextInt();
            pairs[i][1] = in.nextInt();
        }

        int ans = 0;
        for (int[] pair : pairs) {
            int x = pair[0];
            int y = pair[1];
            int g = gcd(a[x], a[y]);
            while (g > 1) {
                a[x] /= g;
                a[y] /= g;
                ans++;
                g = gcd(a[x], a[y]);
            }
        }

        System.out.println(ans);
    }
}
