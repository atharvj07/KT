import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long[][] dp = new long[n + 2][2];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 2; j++) {
                dp[i][j] = Math.max(dp[i + 1][0], dp[i + 1][1]);
                if (j == 0) {
                    dp[i][j] = Math.max(dp[i][j], a[i] + dp[i + 1][1]);
                } else {
                    dp[i][j] = Math.max(dp[i][j], a[i] + dp[i + 1][0]);
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
