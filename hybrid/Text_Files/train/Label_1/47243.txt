import java.util.Scanner;

public class RBSMoves {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            String s = scanner.nextLine();
            int maxMoves = findMaxMoves(s);
            System.out.println(maxMoves);
        }
    }

    private static int findMaxMoves(String s) {
        int n = s.length();
        int[][] dp = new int[n + 1][n + 1]; // dp[i][j] = maximum moves for substring s[i:j]

        // Base case: empty string
        for (int i = 0; i <= n; i++) {
            dp[i][i] = 0;
        }

        // Build the DP table
        for (int len = 1; len <= n; len++) {
            for (int i = 0; i <= n - len; i++) {
                int j = i + len - 1;

                // Check if current substring is an RBS
                if (isRBS(s, i, j)) {
                    dp[i][j] = 1 + Math.max(dp[i + 1][j], dp[i][j - 1]); // Remove current RBS or consider splitting
                } else {
                    // Consider all possible split points
                    for (int k = i; k < j; k++) {
                        dp[i][j] = Math.max(dp[i][j], dp[i][k] + dp[k + 1][j]);
                    }
                }
            }
        }

        return dp[0][n - 1];
    }

    private static boolean isRBS(String s, int i, int j) {
        if (i == j) {
            return false; // Single character cannot be an RBS
        }

        if (s.charAt(i) == '(' && s.charAt(j) == ')') {
            return true;
        }

        if (s.charAt(i) == '[' && s.charAt(j) == ']') {
            return true;
        }

        // Check if middle part is an RBS (for case '(' + RBS + ')')
        if (s.charAt(i) == '(' && isRBS(s, i + 1, j - 1)) {
            return true;
        }

        // Check if middle part is an RBS (for case '[' + RBS + ']')
        if (s.charAt(i) == '[' && isRBS(s, i + 1, j - 1)) {
            return true;
        }

        return false;
    }
}
