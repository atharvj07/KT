#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> a;
bool check() {
  sort(a.begin(), a.end());
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (a[i] + a[j] > a[(i + j + 1) % n]) return true;
    }
  }
  return false;
}
int main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  cout << (check() ? "YES" : "NO") << endl;
  return 0;
}
