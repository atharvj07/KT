#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

const int INF = (int)1e9;

int32_t main() {
    int n, m, q;
    cin >> n >> m >> q;
    vector<vector<int>> graph(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    vector<int> power(n + 1);
    vector<int> deg(n + 1);
    vector<bool> killed(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> power[i];
    }
    while (q--) {
        int type;
        cin >> type;
        if (type == 1) {
            int u, v;
            cin >> u >> v;
            graph[u].push_back(v);
            graph[v].push_back(u);
            deg[u]++;
            deg[v]++;
        } else if (type == 2) {
            int u, v;
            cin >> u >> v;
            graph[u].erase(remove(graph[u].begin(), graph[u].end(), v), graph[u].end());
            graph[v].erase(remove(graph[v].begin(), graph[v].end(), u), graph[v].end());
            deg[u]--;
            deg[v]--;
        } else {
            for (int i = 1; i <= n; ++i) {
                killed[i] = deg[i] > 0 && *max_element(graph[i].begin(), graph[i].end()) > power[i];
            }
            int ans = 0;
            while (true) {
                bool ok = false;
                for (int i = 1; i <= n; ++i) {
                    if (!killed[i]) continue;
                    if (all_of(graph[i].begin(), graph[i].end(), [&](int j) { return !killed[j]; })) {
                        ok = true;
                        killed[i] = false;
                        ans++;
                    }
                }
                if (!ok) break;
            }
            cout << n - ans << '\n';
        }
    }
}
