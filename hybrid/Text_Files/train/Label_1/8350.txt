#include <bits/stdc++.h>

const int MOD = 1e9 + 7;

int main() {
    int n, x;
    std::cin >> n >> x;
    std::vector<int> v(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> v[i];
    }
    std::sort(v.begin(), v.end());
    int ans = 1;
    for (int i = 0; i < n && v[i] <= x; ++i) {
        ans = (1LL * ans * (x - v[i] + 1)) % MOD;
    }
    std::cout << ans << std::endl;
}
