#include <iostream>
#include <string>
#include <vector>
#include <map>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> v(n, -1);
  map<char, int> m;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      cnt++;
    } else if (s[i] == ')') {
      cnt--;
    } else {
      if (m.find(s[i]) == m.end()) {
        m[s[i]] = cnt;
      } else if (m[s[i]] != cnt) {
        cout << "false" << endl;
        return 0;
      }
    }
  }
  if (cnt != 0) {
    cout << "false" << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      v[i] = 0;
    } else if (s[i] == ')') {
      v[i] = 9;
    } else {
      v[i] = m[s[i]];
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '>') {
      int j = i + 1;
      while (s[j] != '<') {
        if (v[j] == -1) {
          v[j] = v[i];
        } else if (v[j] > v[i]) {
          v[j] = v[i];
        }
        j++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '<') {
      int j = i - 1;
      while (s[j] != '>') {
        if (v[j] == -1) {
          v[j] = v[i];
        } else if (v[j] < v[i]) {
          v[j] = v[i];
        }
        j--;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (v[i] == -1) {
      cout << "false" << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    cout << v[i];
  }
  cout << endl;
  return 0;
}
