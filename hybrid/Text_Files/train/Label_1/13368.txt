#include <bits/stdc++.h>

using namespace std;

const double PI = acos(-1);

int n;
double r[100005], h[100005];
double dp[100005];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> r[i] >> h[i]; }
  sort(r + 1, r + 1 + n);
  sort(h + 1, h + 1 + n);
  dp[n] = PI * r[n] * r[n] * h[n];
  for (int i = n - 1; i >= 1; i--) {
    double max_vol = 0;
    int pos = i + 1;
    while (pos <= n && r[pos] <= r[i]) { pos++; }
    pos--;
    if (pos != i) { max_vol = dp[pos + 1]; }
    dp[i] = max(dp[i + 1], PI * r[i] * r[i] * h[i] + max_vol);
  }
  cout << fixed << setprecision(12) << dp[1] << endl;
}
