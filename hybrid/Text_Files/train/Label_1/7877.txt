#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  ll n, M, sum, ans;
  cin >> n >> M;
  vector<ll> a(n), pre(n);
  ans = sum = 0;
  for (auto &i : a) cin >> i, sum += i, pre.push_back(sum);
  for (ll i = 0; i < n; ++i) {
    if (i > 0) ans = max(ans, n - ((lower_bound(pre.begin(), pre.begin() + i, pre.back() - M)) - pre.begin() + 1));
    if (M - sum >= a[i])
      ans = 0;
    else
      sum -= a[i];
    cout << ans << " ";
  }
}
