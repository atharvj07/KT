n = int(input())
coin_values = list(map(int, input().split()))
coin_values.sort()
used_pockets = 0
current_pocket = -1
pockets = []
for i in coin_values:
    if i != current_pocket:
        current_pocket = i
        used_pockets += 1
        pockets.append([i])
    else:
        pockets[-1].append(i)
print(used_pockets)
