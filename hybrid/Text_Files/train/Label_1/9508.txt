#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, k, a[100005], b[100005];
int main() {
  cin >> n >> k;
  for (ll i = 1; i <= n; i++) cin >> a[i];
  for (ll i = 1; i <= n; i++) cin >> b[i];
  ll res = 0;
  for (int i = 1; i <= n; i++) {
    ll left = b[i] / a[i];
    k -= left;
    res += left;
    if (k <= 0) break;
  }
  if (k > 0) res++;
  cout << res << endl;
  return 0;
}
