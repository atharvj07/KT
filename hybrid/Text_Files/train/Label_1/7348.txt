#include <bits/stdc++.h>

using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<pair<int, int>> v(n);
        for (int i = 0; i < n; i++) {
            cin >> v[i].first >> v[i].second;
        }
        sort(v.begin(), v.end());
        long long ans = 0;
        for (int i = 1; i < n; i++) {
            if (v[i].first == v[i - 1].first) {
                long long diff = v[i].second - v[i - 1].second;
                if (diff < 0) {
                    ans -= diff;
                } else {
                    ans += diff;
                    v[i].first++;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
