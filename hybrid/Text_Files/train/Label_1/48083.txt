import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[4];
        for (int i = 0; i < 4; i++) arr[i] = sc.nextInt();
        Arrays.sort(arr);
        if (arr[0] == 1 && arr[1] == 9 && arr[2] == 7 && arr[3] == 4) System.out.println("YES");
        else System.out.println("NO");
    }
}
