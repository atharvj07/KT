import java.util.Scanner;
import java.util.ArrayList;
import java.util.Collections;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        BigInteger n = input.nextBigInteger();
        int p = input.nextInt();
        BigInteger t = BigInteger.ONE;
        ArrayList<BigInteger> b = new ArrayList<>();
        if (p > 0) {
            while (t.compareTo(n) <= 0) {
                b.add(t);
                t = t.multiply(BigInteger.valueOf(2));
            }
        } else if (p < 0) {
            ArrayList<BigInteger> o = new ArrayList<>();
            boolean c = true;
            while (c) {
                if (t.mod(BigInteger.valueOf(2)).compareTo(BigInteger.ONE) == 0) {
                    o.add(t.subtract(BigInteger.valueOf(p)));
                    c = false;
                }
                t = t.multiply(BigInteger.valueOf(2));
            }
            while (t.compareTo(n) <= 0) {
                b.add(t.subtract(BigInteger.valueOf(p)).multiply(BigInteger.valueOf(-1)));
                t = t.multiply(BigInteger.valueOf(2));
            }
            b.addAll(o);
        } else {
            System.out.println(-1);
            return;
        }
        Collections.sort(b);
        int ans = 0;
        t = BigInteger.ZERO;
        while (t.compareTo(n) < 0) {
            ans++;
            t = t.add(b.get(b.size() - ans));
        }
        if (t.equals(n)) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
