n = int(input())
v = list(map(int, input().split()))
t = list(map(int, input().split()))
if n <= 0:
    exit()
melted = [0 for i in range(n)]
for i in range(n):
    if v[i] > t[i]:
        melted[i] = t[i]
        v[i] -= t[i]
    else:
        melted[i] = v[i]
        v[i] = 0
    if v[i] > 0:
        j = i + 1
        while j < n and v[i] > t[j]:
            melted[j] += v[i] - t[j]
            v[i] -= t[j]
            v[j] = 0
            j += 1
        if j < n:
            melted[j] += v[i]
print(*melted)
