#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 2;
int diff[N];
map<int, int> cnt;
int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n);
  for (int &x : a) { cin >> x; }
  for (int i = 0; i < n; i++) {
    cnt[a[i]]++;
  }
  int mn = n;
  for (auto [k, v] : cnt) { mn = min(mn, n - v); }
  cout << mn << " ";
  while (q--) {
    int x, y;
    cin >> x >> y;
    x--;
    mn -= cnt[a[x]];
    cnt[a[x]]--;
    cnt[y]++;
    mn += cnt[y];
    a[x] = y;
    cout << mn << " ";
  }
  return 0;
}
