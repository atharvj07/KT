import java.util.*;
import java.io.*;

public class Main {
    static long[] a;
    static long k;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        k = sc.nextLong();
        a = new long[(int)n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }

        long ans = 0;
        long cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            if (cur > k) {
                ans += (cur + k - 1) / k;
                cur = cur % k;
            }
        }
        if (cur > 0) {
            ans++;
        }
        System.out.println(ans);
    }
}
