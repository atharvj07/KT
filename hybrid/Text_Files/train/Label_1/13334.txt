#include <iostream>
#include <algorithm>

using namespace std;

long long a[200000];

int main() {
    long long n;
    cin >> n;
    for(int i = 0; i < n; i++) {
        cin >> a[i];
    }
    long long ans = 0;
    for(int i = 1; i < n; i++) {
        if(a[i] * a[i - 1] < 0) {
            a[i] = -a[i];
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
