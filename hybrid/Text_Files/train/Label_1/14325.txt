#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  ll n, x;
  cin >> n >> x;
  if ((n & 1 && x == 0) || (n & 1 && (x & 1) == 0)) {
    cout << "NO";
    return 0;
  }
  cout << "YES\n";
  for (ll i = 0; i < n; i++) {
    if (n % 2) {
      if (x || i == n - 1)
        x = 1;
      else
        x = 0;
      cout << x << ' ';
    } else {
      if (i == n - 1) {
        if (x % 2)
          cout << 1 << ' ';
        else
          cout << 2 << ' ';
      } else {
        if (i % 2)
          cout << 1 << ' ';
        else
          cout << 2 << ' ';
      }
    }
    x /= 2;
  }
}
