#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=998244353;
int main(){
    int n;cin>>n;
    vector<pair<ll,int>> a(n);
    vector<int> b(n);
    for(int i=0;i<n;i++){
        ll x;cin>>x;
        a[i]={x,i};
    }
    for(int i=0;i<n;i++){
        cin>>b[i];
    }
    sort(a.begin(),a.end());
    sort(b.begin(),b.end());
    ll ans=0;
    for(int i=0;i<n;i++){
        ans+=1LL*a[i].first*b[i]%mod;
        ans%=mod;
    }
    cout<<ans<<endl;
}
