#include <iostream>
using namespace std;
typedef long long ll;
const ll max_x = 5e18;

int main() {
  ll x, y, l, r;
  cin >> x >> y >> l >> r;
  if (l == r)
    return cout << "1\n", 0;
  if (r == max_x)
    return cout << "0\n", 0;
  ll ans = 0, a = 1, b = 1;
  while (1) {
    if (a > r || b > r)
      break;
    if (a >= l && b >= l)
      ans++;
    a *= x;
    b *= y;
  }
  cout << ans << endl;
  return 0;
}
