import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String level = in.next();
        boolean good = false;
        for (int i = 0; i < n - 4; i++) {
            if (level.charAt(i) == '*' && level.charAt(i + 1) == '*' && level.charAt(i + 2) == '*' && level.charAt(i + 3) == '*') {
                good = true;
                break;
            }
        }
        System.out.println(good ? "yes" : "no");
    }
}
