import sys
rd = sys.stdin.readline

h,w = map(int,rd().split())
g = [list(rd()[:-1]) for _ in range(h)]
dp = [[[0]*3 for _ in range(w+1)] for _ in range(h+1)]

for i in range(h-1,-1,-1):
    for j in range(w-1,-1,-1):
        if g[i][j] == '.':
            dp[i][j][0] = dp[i][j+1][0] + dp[i+1][j][1]
            if g[i][j+1] == '.' and g[i+1][j] == '.':
                dp[i][j][0] += dp[i+1][j+1][2]
            if i+1<h and j+1<w and g[i+1][j+1] == '.':
                dp[i][j][0] += 1
            dp[i][j][1] = dp[i+1][j][0] + dp[i][j+1][2]
            if g[i+1][j] == '.' and g[i][j+1] == '.':
                dp[i][j][1] += dp[i+1][j+1][2]
            if i+1<h and j+1<w and g[i+1][j+1] == '.':
                dp[i][j][1] += 1
            dp[i][j][2] = dp[i+1][j][2] + dp[i][j+1][2]
            if g[i+1][j] == '.' and g[i][j+1] == '.':
                dp[i][j][2] += dp[i+1][j+1][2]
            if i+1<h and j+1<w and g[i+1][j+1] == '.':
                dp[i][j][2] += 1

q = int(rd())
for _ in range(q):
    r1,c1,r2,c2 = map(int,rd().split())
    r1-=1; c1-=1; r2-=1; c2-=1
    print(dp[r1][c1][0]-dp[r1][c2+1][0]-dp[r2+1][c1][1]+dp[r2+1][c2+1][1])
