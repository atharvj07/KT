#include <iostream>
#include <cmath>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        unsigned long long x, k;
        cin >> x >> k;
        unsigned long long ans = 0;
        while (x > 0)
        {
            if (x % k == 0)
                x /= k, ans++;
            else
                x--, ans++;
        }
        cout << ans << endl;
    }
    return 0;
}
