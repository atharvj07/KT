#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a;
  cin >> n >> a;
  int t[n];
  for (int i = 0; i < n; ++i) { cin >> t[i]; }
  int ans = 0;
  if (t[a - 1] == 1) ans++;
  for (int i = 1; i <= min(n - a, a - 1); ++i) {
    if (t[a - 1 - i] == 0 && t[a - 1 + i] == 0) ans++;
  }
  cout << ans << endl;
  return 0;
}
