#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int MM = 2e5+5;
int n, m, d, l[MM], r[MM], u[MM], dwn[MM];
int32_t main() {
    cin >> d;
    cin >> n >> m;
    memset(l, -1, sizeof(l));
    memset(r, -1, sizeof(r));
    memset(u, -1, sizeof(u));
    memset(dwn, -1, sizeof(dwn));
    while(d--) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        if(x1 == x2) {
            if(y1 > y2)
                swap(y1, y2);
            dwn[y1] = y2; u[y2] = y1;
        }
        else {
            if(x1 > x2)
                swap(x1, x2);
            r[x1] = x2; l[x2] = x1;
        }
    }
    int cntl, cntr, cntt, cntb;
    cin >> cntl >> cntr >> cntt >> cntb;
    for(int i = 1; i <= m; i++) {
        if(l[i]==-1 || r[i]==-1){
            if(l[i]==-1 || i-1-l[i]==cntl){
                if(r[i]==-1 || r[i]-i-1==cntr){
                    int id = i-1;
                    if(id<=n && id>=1){
                        if(dwn[i]==-1 || dwn[i]-i-1==cntt){
                            if(u[i]==-1 || i-1-u[i]==cntb){
                                cout << id << endl;
                                return 0;
                            }
                        }
                    }
                }
            }
        }
    }
    cout << "-1\n";
    return 0;
}
