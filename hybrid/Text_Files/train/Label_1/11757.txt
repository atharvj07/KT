#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mx=1e6+50;
int p[mx], ok[mx];
int main(){
    int a, b, k;
    cin>>a>>b>>k;
    bool has_sol=0;
    for(int i=2;i*i<=b;++i){
        if(!p[i]){
            for(int j=i*i;j<=b;++j+=i) p[j]=i;
        }
    }
    for(int i=a;i<=b;++i){
        bool sol=1;
        if(p[i]) continue;
        for(int j=0;j<k;++j){
            if(p[i+j]){
                sol=0;
                break;
            }
        }
        if(sol){
            has_sol=1;
            cout<<(i-a+1)<<"\n";
            return 0;
        }
    }
    cout<<"-1\n";
}
