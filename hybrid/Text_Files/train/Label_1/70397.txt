def can_coexist(word, religions):
    n = len(word)
    dp = [[[None for _ in range(n+1)] for __ in range(n+1)] for ___ in range(n+1)]
    dp[0][0][0] = 0

    for i in range(len(religions[0]) + 1):
        for j in range(len(religions[1]) + 1):
            for k in range(len(religions[2]) + 1):
                if i > 0 and dp[i-1][j][k] is not None and religions[0][i-1] in word[dp[i-1][j][k]:]:
                    dp[i][j][k] = min(dp[i][j][k] if dp[i][j][k] is not None else float('inf'), word.index(religions[0][i-1], dp[i-1][j][k]) + 1)
                if j > 0 and dp[i][j-1][k] is not None and religions[1][j-1] in word[dp[i][j-1][k]:]:
                    dp[i][j][k] = min(dp[i][j][k] if dp[i][j][k] is not None else float('inf'), word.index(religions[1][j-1], dp[i][j-1][k]) + 1)
                if k > 0 and dp[i][j][k-1] is not None and religions[2][k-1] in word[dp[i][j][k-1]:]:
                    dp[i][j][k] = min(dp[i][j][k] if dp[i][j][k] is not None else float('inf'), word.index(religions[2][k-1], dp[i][j][k-1]) + 1)
                
    return dp[len(religions[0])][len(religions[1])][len(religions[2])] is not None

n, q = map(int, input().split())
word = input()
religions = ['', '', '']

for _ in range(q):
    operation = input().split()
    if operation[0] == '+':
        religions[int(operation[1])-1] += operation[2]
    else:
        religions[int(operation[1])-1] = religions[int(operation[1])-1][:-1]
    
    if can_coexist(word, religions):
        print("YES")
    else:
        print("NO")
