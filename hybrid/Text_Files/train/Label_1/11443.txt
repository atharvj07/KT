#include<bits/stdc++.h>
using namespace std;
#define LL long long
#define pb push_back
const int maxn=2e5+10;
vector<vector<int>>G,RG;
vector<bool>vis;
vector<int>order;
int S[maxn],T[maxn],cnt;
void dfs1(int u)
{
    vis[u]=true;
    for(int i:G[u])
    {
        if(!vis[i])
        {
            dfs1(i);
        }
    }
    order.pb(u);
}
void dfs2(int u)
{
    vis[u]=true;
    T[u]=cnt;
    for(int i:RG[u])
    {
        if(!vis[i])
        {
            dfs2(i);
        }
    }
}
bool ok=true;
int main()
{
    int n,m,x,y,i;
    cin>>n>>m;
    G.resize(n+1);
    RG.resize(n+1);
    vis.resize(n+1);
    while(m--)
    {
        cin>>x>>y;
        G[x].pb(y);
        RG[y].pb(x);
    }
    for(i=1;i<=n;i++)
    {
        if(!vis[i])
        {
            dfs1(i);
        }
    }
    vis.assign(n+1,false);
    reverse(order.begin(),order.end());
    for(int u:order)
    {
        if(!vis[u])
        {
            dfs2(u);
            cnt++;
        }
    }
    for(i=1;i<=n;i++)
    {
        for(int j:G[i])
        {
            if(T[i]==T[j])
            {
                ok=false;
            }
        }
    }
    if(ok)cout<<0<<endl;
    else cout<<1<<endl;
    return 0;
}
