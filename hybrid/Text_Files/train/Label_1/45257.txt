import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		int n, cnt = 0;
		Scanner sc = new Scanner(System.in);
		while ((n = sc.nextInt()) != 0) {
			cnt = 0;
			while (n != 1) {
				if (n % 2 == 0) {
					n /= 2;
				} else {
					n = n * 3 + 1;
				}
				cnt++;
			}
			System.out.println(cnt);
		}
	}
}
