from collections import Counter
from math import gcd
from functools import reduce

def find_gcd(list):
    x = reduce(gcd, list)
    return x

n = int(input())
beads = list(map(int, input().split()))

beads_count = Counter(beads)
gcd_val = find_gcd(beads)

odd_count = sum(1 for bead in beads if bead % 2 != 0)

if odd_count > 1:
    print(0)
    necklace = ''.join(chr(97 + i) * bead for i, bead in enumerate(beads))
else:
    print(gcd_val)
    if odd_count == 1:
        odd_bead = ''
        for i, bead in enumerate(beads):
            if bead % 2 != 0:
                odd_bead = chr(97 + i) * bead
                beads[i] -= bead
                break
        half_necklace = ''.join(chr(97 + i) * (bead // 2) for i, bead in enumerate(beads))
        necklace = half_necklace + odd_bead + half_necklace[::-1]
    else:
        half_necklace = ''.join(chr(97 + i) * (bead // gcd_val) for i, bead in enumerate(beads))
        necklace = half_necklace * gcd_val
print(necklace)
