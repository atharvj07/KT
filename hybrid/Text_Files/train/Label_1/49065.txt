import java.util.*;
import java.lang.*;

public class Main {
    private static final long INF = 1L << 62;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long f = sc.nextLong(), T = sc.nextLong(), t0 = sc.nextLong();
        long a1 = sc.nextLong(), t1 = sc.nextLong(), p1 = sc.nextLong();
        long a2 = sc.nextLong(), t2 = sc.nextLong(), p2 = sc.nextLong();
        long res = INF;
        for (long i = 0; i <= Math.min(T, f / t0); i++) {
            long remain = f - t0 * i;
            long tmp = i * (T / t0) * p1 + ((T % t0 == 0) ? 0 : p1);
            if (remain <= a1) tmp += p1;
            else if (remain % a1 == 0) tmp += (remain / a1) * p1;
            else tmp += (remain / a1 + 1) * p1;
            res = Math.min(res, tmp);
            tmp = i * (T / t0) * p2 + ((T % t0 == 0) ? 0 : p2);
            if (remain <= a2) tmp += p2;
            else if (remain % a2 == 0) tmp += (remain / a2) * p2;
            else tmp += (remain / a2 + 1) * p2;
            res = Math.min(res, tmp);
            tmp = i * (T / t0) * p1 + ((T % t0 == 0) ? 0 : p1);
            tmp += (T / t1) * p1 + ((T % t1 == 0) ? 0 : p1);
            if (remain <= a1) tmp += p1;
            else if (remain % a1 == 0) tmp += (remain / a1) * p1;
            else tmp += (remain / a1 + 1) * p1;
            res = Math.min(res, tmp);
            tmp = i * (T / t0) * p1 + ((T % t0 == 0) ? 0 : p1);
            tmp += (T / t2) * p2 + ((T % t2 == 0) ? 0 : p2);
            if (remain <= a2) tmp += p2;
            else if (remain % a2 == 0) tmp += (remain / a2) * p2;
            else tmp += (remain / a2 + 1) * p2;
            res = Math.min(res, tmp);
        }
        System.out.println(res == INF ? -1 : res);
    }
}
