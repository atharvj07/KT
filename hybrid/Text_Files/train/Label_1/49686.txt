import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            PriorityQueue<int[]> q = new PriorityQueue<>((a, b) -> a[0] - b[0]);
            for (int i = 0; i < n; i++) {
                q.add(new int[]{sc.nextInt(), sc.nextInt()});
            }
            int time = 0;
            int[] curr = q.poll();
            while (!q.isEmpty()) {
                int[] next = q.poll();
                time += Math.max(curr[0] * 2 * curr[1], next[0] * 2 * next[1]);
                curr = next;
            }
            System.out.println(time);
        }
    }
}
