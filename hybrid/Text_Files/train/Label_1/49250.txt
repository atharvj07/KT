import java.util.*;
import java.lang.*;

class Main {
public static void main(String[] args) {
		long mod=1000000007;
		Scanner sc = new Scanner(System.in);
		int n=sc.nextInt();
		int[] a=new int[n];
		HashMap<Integer,Integer> hm=new HashMap<>();
		for(int i=0;i<n;++i) {
			a[i]=sc.nextInt();
			hm.put(a[i],hm.getOrDefault(a[i],0)+1);
		}
		long ans=0;
		for(int i=0;i<n;++i) {
			int cnt=0;
			for(int val:hm.keySet()) {
				if(Math.abs(val-a[i])<=1) cnt++;
			}
			ans=(ans+cnt)%mod;
		}
		System.out.println(ans);
}
}
