#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;
const int MAXN = 1e5 + 5;

int main() {
  int m, n;
  cin >> m;
  vector<int> q(m);
  for (int i = 0; i < m; i++) cin >> q[i];
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  sort(a.begin(), a.end());
  vector<vector<int>> dp(m + 1, vector<int>(n + 1, 1e9));
  dp[0][0] = 0;

  for (int i = 0; i < m; i++) {
    for (int j = 0; j <= n; j++) {
      dp[i + 1][j] = min(dp[i + 1][j], dp[i][j]);
      if (j + q[i] <= n) { dp[i + 1][j + q[i]] = min(dp[i + 1][j + q[i]], dp[i][j] + a[j]); }
      if (j + q[i] + 1 <= n) { dp[i + 1][j + q[i] + 1] = min(dp[i + 1][j + q[i] + 1], dp[i][j] + a[j]); }
      if (j + q[i] + 2 <= n) { dp[i + 1][j + q[i] + 2] = min(dp[i + 1][j + q[i] + 2], dp[i][j] + a[j]); }
    }
  }

  cout << dp[m][n] << "\n";

  return 0;
}
