#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;

  while (t--) {
    long long a, b, c;
    std::cin >> a >> b >> c;

    if (((a + b - c) % 2 == 0 && (a + b - c) > 0) || ((a + c - b) % 2 == 0 && (a + c - b) > 0) || ((b + c - a) % 2 == 0 && (b + c - a) > 0)) {
      std::cout << (a + b - c + 1000000007) % 1000000007 << std::endl;
    } else
      std::cout << -1 << std::endl;
  }

  return 0;
}
