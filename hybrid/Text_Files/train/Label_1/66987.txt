import sys

def get_prefix_sum(arr: list, n: int, t: int) -> list:
    pre = [0] * n
    for i in range(n):
        if i > 0:
            pre[i] = pre[i - 1]
        if arr[i] <= t // 2:
            pre[i] += 1
    return pre

def get_min(pre: list, n: int, t: int) -> int:
    mini = n
    for i in range(n):
        a = pre[i]
        b = i + 1 - a
        c = n - i - 1 - (pre[n - 1] - pre[i])
        d = pre[n - 1] - pre[i]
        mini = min(mini, (a * c) + (b * d))
    return mini

for _ in range(int(input())):
    n, t = map(int, input().split())
    arr = list(map(int, input().split()))
    pre = get_prefix_sum(arr, n, t)
    print(*[int(x <= t // 2) for x in arr], sep=' ')
