#include <bits/stdc++.h>
using namespace std;
int ans;
int main(){
    int n, m, x;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; ++i){
        string s;
        cin >> s;
        for(int j = 0; j < m; ++j)
            a[i][j] = s[j] - 'A';
    }
    vector<int> b(m);
    for(int i = 0; i < m; ++i){
        cin >> x;
        b[i] = x;
        ans = max(ans, x);
    }
    for (int j = 0; j < m; ++j){
        vector<int> cnt(5);
        for (int i = 0; i < n; ++i){
            ++cnt[a[i][j]];
        }
        for(int i = 0; i < 5; ++i){
            if(cnt[i] == 1)
                ans += b[j];
        }
    }
    cout << ans << endl;
    return 0;
}
