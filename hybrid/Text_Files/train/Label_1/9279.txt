#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 1;
#define fs first
#define sc second

int n, a[N];
pair<int, int> b[N];
vector<int> g[N];

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        b[i] = make_pair(a[i], i);
    }
    sort(b + 1, b + n + 1);
    for (int i = 1; i <= n; ++i) {
        g[a[i]].push_back(b[i].sc);
    }
    for (int i = 1; i <= n; ++i) {
        if (g[i].size() == 0) {
            cout << "NO" << endl;
            return 0;
        }
        int j = (int)g[i].size() - 1;
        for (; j >= 1; --j) {
            if (g[i][j] < g[i][j - 1]) {
                cout << "NO" << endl;
                return 0;
            }
        }
    }
    cout << "YES" << endl;
    return 0;
}
