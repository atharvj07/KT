MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, f1, f2, f3, c):
    e0 = 2 * (n - 3)
    e1 = n - 3
    e2 = e1 - 1
    e3 = e2 - 1
    f1_mod = f1 % MOD
    f2_mod = f2 % MOD
    f3_mod = f3 % MOD
    c_mod = c % MOD
    ans = (power(f1_mod, e1, MOD) * power(f2_mod, e2, MOD) % MOD * power(f3_mod, e3, MOD) % MOD * power(c_mod, e0, MOD) % MOD) % MOD
    return ans

n, f1, f2, f3, c = map(int, input().split())
print(solve(n, f1, f2, f3, c))
