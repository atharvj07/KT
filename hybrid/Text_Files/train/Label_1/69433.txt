n = int(input())
hand_preference = input()

left = [i+1 for i in range(n) if hand_preference[i] == 'L']
right = [i+1 for i in range(n) if hand_preference[i] == 'R']

left_pointer, right_pointer = 0, 0
while left_pointer < len(left) or right_pointer < len(right):
    if right_pointer < len(right):
        print(right[right_pointer], end=' ')
        right_pointer += 1
    else:
        print(left[left_pointer+1], end=' ')
        left_pointer += 2
    if left_pointer < len(left):
        print(left[left_pointer])
        left_pointer += 1
    else:
        print(right[right_pointer-1])
