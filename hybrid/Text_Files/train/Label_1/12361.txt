#include <bits/stdc++.h>
const int MOD = 1000000007;

using namespace std;

int main() {
  int n, k, d;
  cin >> n >> k >> d;
  vector<vector<long long>> dp(n + 1, vector<long long>(k + 1, 0));
  dp[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      for (int l = 0; l <= k; l++) {
        if (l >= d) {
          dp[i][j] = (dp[i][j] + dp[i - 1][j - l]) % MOD;
        }
        if (j > 0) {
          dp[i][j] = (dp[i][j] + dp[i - 1][j - 1]) % MOD;
        }
      }
    }
  }
  cout << dp[n][k];
  return 0;
}
