#include <iostream>
#include <vector>
using namespace std;

int main()
{
    long long n, x, y;
    cin >> n >> x >> y;
    vector<bool> dp(n + 1, true);
    for (long long i = x; i <= n; i += x)
        dp[i] = false;
    for (long long i = y; i <= n; i += y)
        dp[i] = false;
    int ans = 0;
    for (long long i = 1; i <= n; i++)
        if (dp[i])
            ans++;
    cout << ans << endl;
    return 0;
}
