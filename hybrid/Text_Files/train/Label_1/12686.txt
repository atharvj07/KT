#include <iostream>
#include <vector>
#include <stack>
using namespace std;
const long long int INF = 1e18;
int main() {
  int n;
  cin >> n;
  vector<long long int> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  vector<pair<int, char>> sol;
  long long int pref[n + 1], suff[n + 1];
  pref[0] = -INF;
  for (int i = 0; i < n; i++) { pref[i + 1] = max(pref[i], a[i]); }
  suff[n] = INF;
  for (int i = n - 1; i >= 0; i--) { suff[i] = min(suff[i + 1], b[i]); }
  int p = 0, q = n - 1;
  while (p < q) {
    if (pref[p + 1] < suff[q]) {
      sol.push_back({p + 1, 'R'});
      suff[q - 1] = b[p];
      p++;
    } else {
      sol.push_back({q, 'L'});
      pref[p + 1] = b[q];
      q--;
    }
  }
  if (p == q && pref[p + 1] == suff[q]) {
    cout << "YES" << endl;
    for (int i = 0; i < sol.size(); i++) { cout << sol[i].first << " " << sol[i].second << endl; }
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
