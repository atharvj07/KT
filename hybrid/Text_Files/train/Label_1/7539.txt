#include <iostream>
using namespace std;
typedef long long ll;
int main() {
    ll n, k, A, B;
    cin >> n >> k >> A >> B;
    ll ans = 0;
    while (n > 1) {
        if (n % k == 0 && A + B < n % k * B) {
            n /= k;
            ans += B;
        } else {
            n--;
            ans += A;
        }
    }
    cout << ans << endl;
}
