import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt();
        int yv = sc.nextInt();
        int xp = sc.nextInt();
        int yp = sc.nextInt();
        int xw1 = sc.nextInt();
        int yw1 = sc.nextInt();
        int xw2 = sc.nextInt();
        int yw2 = sc.nextInt();
        int xm1 = sc.nextInt();
        int ym1 = sc.nextInt();
        int xm2 = sc.nextInt();
        int ym2 = sc.nextInt();

        boolean canSee = true;
        if (xw1 <= Math.max(xv, xp) && xw1 >= Math.min(xv, xp)) {
            if (yw1 <= Math.max(yv, yp) && yw1 >= Math.min(yv, yp)) {
                canSee = false;
            } else if (yw2 <= Math.max(yv, yp) && yw2 >= Math.min(yv, yp)) {
                canSee = false;
            }
        } else if (xw2 <= Math.max(xv, xp) && xw2 >= Math.min(xv, xp)) {
            if (yw1 <= Math.max(yv, yp) && yw1 >= Math.min(yv, yp)) {
                canSee = false;
            } else if (yw2 <= Math.max(yv, yp) && yw2 >= Math.min(yv, yp)) {
                canSee = false;
            }
        }

        if (canSee) {
            if (xm1 <= Math.max(xv, xp) && xm1 >= Math.min(xv, xp)) {
                if (ym1 <= Math.max(yv, yp) && ym1 >= Math.min(yv, yp)) {
                    canSee = false;
                } else if (ym2 <= Math.max(yv, yp) && ym2 >= Math.min(yv, yp)) {
                    canSee = false;
                }
            } else if (xm2 <= Math.max(xv, xp) && xm2 >= Math.min(xv, xp)) {
                if (ym1 <= Math.max(yv, yp) && ym1 >= Math.min(yv, yp)) {
                    canSee = false;
                } else if (ym2 <= Math.max(yv, yp) && ym2 >= Math.min(yv, yp)) {
                    canSee = false;
                }
            }
        }

        if (canSee) {
            if (((xv == xp) && ((yv > ym1) || (yv < ym2))) || ((yv == yp) && ((xv > xm1) || (xv < xm2)))) {
                canSee = false;
            }
        }

        System.out.println(canSee ? "YES" : "NO");
    }
}
