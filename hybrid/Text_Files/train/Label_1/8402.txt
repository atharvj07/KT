#include <iostream>
#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> s(n);
  for (int i = 0; i < n; i++) { cin >> s[i]; }
  string ans = s[0];
  for (int i = 1; i < n; i++) {
    string new_ans = "";
    int overlap = 0;
    for (int j = s[i].size() - 1; j >= 0; j--) {
      if (ans[ans.size() - overlap - 1] == s[i][j]) {
        overlap++;
      } else {
        break;
      }
    }
    new_ans.append(ans, 0, ans.size() - overlap);
    new_ans.append(s[i], overlap, s[i].size() - overlap);
    ans = new_ans;
  }
  cout << ans << endl;
  return 0;
}
