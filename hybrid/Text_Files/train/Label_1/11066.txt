#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  string s = "";
  int k = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 1) {
      k++;
      s += "(";
    } else {
      s += "0";
    }
  }
  if (k == n) {
    cout << "NO" << endl;
    return 0;
  }
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] == 1) {
      s += ")";
    } else {
      s += "->1";
    }
  }
  cout << "YES" << endl;
  cout << s << endl;
  return 0;
}
