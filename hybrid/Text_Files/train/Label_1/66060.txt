import sys
input = sys.stdin.readline

n, m = map(int, input().split())
friends = [list(map(int, input().split())) for _ in range(n)]
pizzas = [list(map(int, input().split())) for _ in range(m)]

pizzas.sort(key=lambda x: x[0])

ans = []
best = 0

for i in range(m):
    for j in range(i + 1, m):
        cnt = 0
        for a, b in zip(pizzas[i][1:], pizzas[j][1:]):
            for ingr in friends:
                if ingr[0] <= cnt:
                    continue
                if a in ingr[1:] or b in ingr[1:]:
                    cnt += 1
        if best < cnt:
            best = cnt
            ans = [i + 1, j + 1]

print(*ans)
