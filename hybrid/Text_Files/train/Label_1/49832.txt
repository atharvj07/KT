import java.util.LinkedList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++)
            grid[i] = sc.next().toCharArray();
        String[] ans = new String[n];
        for (int i = 0, j; i < n; i++) {
            ans[i] = new String();
            for (j = 0; j < m; j++) {
                if (grid[i][j] == '*') {
                    boolean[][] vis = new boolean[n][m];
                    LinkedList<Integer> queue = new LinkedList<>();
                    queue.add(i);
                    queue.add(j);
                    int cnt = 0;
                    while (!queue.isEmpty()) {
                        int x = queue.poll(), y = queue.poll();
                        if (vis[x][y] || grid[x][y] == '*')
                            continue;
                        cnt++;
                        vis[x][y] = true;
                        if (x - 1 >= 0) {
                            queue.add(x - 1);
                            queue.add(y);
                        }
                        if (x + 1 < n) {
                            queue.add(x + 1);
                            queue.add(y);
                        }
                        if (y - 1 >= 0) {
                            queue.add(x);
                            queue.add(y - 1);
                        }
                        if (y + 1 < m) {
                            queue.add(x);
                            queue.add(y + 1);
                        }
                    }
                    ans[i] += cnt % 10;
                } else
                    ans[i] += grid[i][j];
            }
        }
        for (String s : ans)
            System.out.println(s);
    }
}
