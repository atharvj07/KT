import sys
from collections import Counter

k, n = map(int, input().split())
s = [input() for i in range(k)]
chars = Counter([c for sl in s for c in sl])
odd = []
for c, f in chars.items():
    if f & 1: odd += c * (f // 2)
if len(odd) > 2: print(-1); sys.exit()
if len(odd) == 2: print(''.join([c * (f // 2) for c, f in chars.items()]) + ''.join(odd))
else: print(''.join([c * (f // 2) for c, f in chars.items()]) + odd[0])
