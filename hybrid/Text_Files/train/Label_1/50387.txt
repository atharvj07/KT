import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  static Map<Integer, Integer> indegree;
  static int m, n;
  static List<Integer> [] adj;
  static int [][] ans;

  public static void solve() {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    try {
      StringTokenizer st = new StringTokenizer(br.readLine());
      n = Integer.parseInt(st.nextToken());
      m = Integer.parseInt(st.nextToken());
      adj = new List[n];
      for (int i = 0; i < n; i++) {
        adj[i] = new ArrayList<>();
      }
      ans = new int[m][3];
      for (int i = 0; i < m; i++) {
        st = new StringTokenizer(br.readLine());
        int a = Integer.parseInt(st.nextToken());
        int b = Integer.parseInt(st.nextToken());
        a--; b--;
        adj[a].add(b);
        adj[b].add(a);
      }
      indegree = new HashMap<>();
      for (int i = 0; i < n; i++) indegree.put(i, 0);
      for (int i = 0; i < n; i++) {
        for (int v: adj[i]) {
          indegree.put(v, indegree.get(v) + 1);
        }
      }
      Queue<Integer> q = new LinkedList<>();
      for (int i = 0; i < n; i++) {
        if (indegree.get(i) == 1) q.add(i);
      }
      int ptr = 0;
      while (!q.isEmpty()) {
        int u = q.poll();
        for (int v: adj[u]) {
          indegree.put(v, indegree.get(v) - 1);
          if (indegree.get(v) == 1) {
            ans[ptr][0] = u + 1;
            ans[ptr][1] = v + 1;
            for (int w: adj[v]) {
              if (w != u) {
                ans[ptr][2] = w + 1;
                break;
              }
            }
            ptr++;
            q.add(v);
          }
        }
      }
      if (ptr != m / 2) {
        System.out.println("No solution");
        return;
      }
      for (int i = 0; i < m / 2; i++) {
        System.out.println(ans[i][0] + " " + ans[i][1] + " " + ans[i][2]);
      }
    } catch (IOException e) {
      e.printStackTrace();
      System.exit(0);
    }
  }

  public static void main(String[] args) {
    solve();
  }
}
