for _ in range(int(input())):
    n, m = map(int, input().split())
    grid = [list(map(int, input().split())) for _ in range(n)]
    l = 1 >> 62
    for i in range(n):
        for j in range(m):
            if i == n - 1 and j == m - 1:
                continue
            if grid[i][j] == grid[n - 1 - i][m - 1 - j]:
                continue
            if not (grid[i][j] or grid[n - 1 - i][m - 1 - j]):
                grid[i][j] = 1
                l += 1
            elif grid[i][j]:
                grid[n - 1 - i][m - 1 - j] = 1
                l += 1
    print(l)
