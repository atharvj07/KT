import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    private static int n, v, e;
    private static int[] a, b;
    private static List<List<Integer>> adjList;
    private static int[][] maxFlow;
    private static int[] revEdge;
    private static boolean[] used;
    private static int totalFlow;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        v = Integer.parseInt(line[1]);
        e = Integer.parseInt(line[2]);
        a = new int[n + 1];
        b = new int[n + 1];
        adjList = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }
        line = br.readLine().split(" ");
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(line[i - 1]);
        }
        line = br.readLine().split(" ");
        for (int i = 1; i <= n; i++) {
            b[i] = Integer.parseInt(line[i - 1]);
        }
        for (int i = 0; i < e; i++) {
            line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            adjList.get(u).add(v);
            adjList.get(v).add(u);
        }
        maxFlow = new int[n + 1][n + 1];
        revEdge = new int[n + 1];
        used = new boolean[n + 1];
        totalFlow = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] > b[i]) {
                System.out.println("NO");
                return;
            }
            totalFlow += b[i] - a[i];
        }
        for (int i = 1; i <= n; i++) {
            for (int j : adjList.get(i)) {
                maxFlow[i][j] = v;
            }
        }
        int t = maxFlow();
        if (t != totalFlow) {
            System.out.println("NO");
            return;
        }
        List<String> ans = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (maxFlow[i][j] > 0) {
                    ans.add(i + " " + j + " " + maxFlow[i][j]);
                }
            }
        }
        System.out.println(ans.size());
        for (String s : ans) {
            System.out.println(s);
        }
    }

    private static int maxFlow() {
        int flow = 0;
        while (true) {
            int[] path = new int[n + 1];
            int[] minFlow = new int[n + 1];
            if (!findAugmentingPath(1, n, path, minFlow)) {
                break;
            }
            int min = Integer.MAX_VALUE;
            for (int i = 2; i <= n; i++) {
                if (path[i] != -1) {
                    min = Math.min(min, minFlow[i]);
                }
            }
            flow += min;
            for (int i = 2; i <= n; i++) {
                if (path[i] != -1) {
                    maxFlow[path[i]][i] -= min;
                    maxFlow[i][path[i]] += min;
                }
            }
        }
        return flow;
    }

    private static boolean findAugmentingPath(int s, int t, int[] path, int[] minFlow) {
        for (int i = 1; i <= n; i++) {
            used[i] = false;
        }
        used[s] = true;
        path[s] = -1;
        minFlow[s] = Integer.MAX_VALUE;
        boolean found = false;
        List<Integer> queue = new ArrayList<>();
        queue.add(s);
        while (!queue.isEmpty()) {
            int u = queue.remove(0);
            for (int v : adjList.get(u)) {
                if (!used[v] && maxFlow[u][v] > 0) {
                    used[v] = true;
                    path[v] = u;
                    minFlow[v] = Math.min(minFlow[u], maxFlow[u][v]);
                    queue.add(v);
                    if (v == t) {
                        found = true;
                        break;
                    }
                }
            }
        }
        return found;
    }
}
