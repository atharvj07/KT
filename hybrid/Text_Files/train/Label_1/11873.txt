#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i].first >> arr[i].second;
    }

    int max_beauty = 0;
    int max_beauty_idx = -1;

    for (int i = 0; i < n; i++) {
        int beauty = abs(arr[i].first - arr[i].second) - abs(arr[i].second - arr[i].first);
        if (beauty > max_beauty) {
            max_beauty = beauty;
            max_beauty_idx = i;
        }
    }

    cout << max_beauty_idx + 1 << endl;
    return 0;
}
