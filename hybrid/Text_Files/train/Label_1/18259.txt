#include <bits/stdc++.h>
using namespace std;
bool ok(string s) {
    bool v = 0, c = 0;
    for (int i = 0; i < (int)s.size(); i++) {
        if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u' || s[i] == 'y')
            v = 1;
        else
            c = 1;
    }
    return v and c;
}
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &p : a) cin >> p;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        istringstream buf(s);
        int syll = 0;
        string t;
        while (buf >> t) {
            if (ok(t)) { syll++; }
        }
        ans += syll;
    }
    cout << (ans == accumulate(a.begin(), a.end(), 0) ? "YES" : "NO") << '\n';
}
