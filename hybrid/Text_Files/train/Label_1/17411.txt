#include<bits/stdc++.h>
using namespace std;
int n,a[200002],depth[200002],sz[200002],mxdepth[200002],mxa[200002],ans;
vector<int>adj[200002];
void dfs1(int node,int p)
{
    depth[node]=depth[p]+1;
    sz[node]=1;
    mxdepth[node]=depth[node];
    mxa[node]=a[node];
    for(int adjacent:adj[node])
    {
        if(adjacent==p) continue;
        dfs1(adjacent,node);
        sz[node]+=sz[adjacent];
        if(depth[adjacent]>mxdepth[node])
        {
            mxdepth[node]=depth[adjacent];
            mxa[node]=a[adjacent];
        }
        if(depth[adjacent]==mxdepth[node] && a[adjacent]>mxa[node])
        {
            mxa[node]=a[adjacent];
        }
    }
}
void dfs2(int node,int p)
{
    for(int adjacent:adj[node])
    {
        if(adjacent==p) continue;
        if(mxdepth[node]==depth[adjacent] && mxa[node]==a[adjacent])
        {
            dfs2(adjacent,node);
            continue;
        }
        ans+=(sz[n]-sz[adjacent])*depth[node];
        ans+=a[node]*(sz[adjacent]);
        dfs2(adjacent,node);
    }
}
int main()
{
    int i,x,y;
    cin>>n;
    for(i=1;i<=n;i++) cin>>a[i];
    for(i=1;i<n;i++)
    {
        cin>>x>>y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    dfs1(1,0);
    dfs2(1,0);
    cout<<ans<<endl;
    return 0;
}
