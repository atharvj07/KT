#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  int v[n], l = 0;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    l += v[i] < 0;
  }
  if (l < 3)
    cout << 1 << endl << n << endl;
  else
    cout << 2 << endl << (n - 1) / 3 + 1 << ' ' << (n - 1) / 3 + 1 << endl;
  return 0;
}
