from collections import Counter
n = int(input())
nums = list(map(int, input().split()))
cnt = Counter(nums)
team = min(cnt[i] for i in range(1, 4))
print(team)
for i in range(1, 4):
    for j in range(1, 4):
        for k in range(1, 4):
            if i != j and i != k and j != k and cnt[i] and cnt[j] and cnt[k]:
                cnt[i] -= 1
                cnt[j] -= 1
                cnt[k] -= 1
                print(nums.index(i) + 1, nums.index(j) + 1, nums.index(k) + 1)
                team -= 1
                if team == 0:
                    exit()
