#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int w, h, n;
    cin >> w >> h >> n;
    if (w % 2 == 0 && (w / 2 >= n || h >= n)) {
      cout << "YES" << endl;
    } else if (h % 2 == 0 && (h / 2 >= n || w >= n)) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
