for _ in range(int(input())):
    n = int(input())
    t = [0] * (n + 1)
    x = [0] * (n + 1)
    for i in range(n):
        ti, xi = map(int, input().split())
        minimum = -10 ** 9
        for j in range(i + 1):
            if t[j] <= ti:
                minimum = max(minimum, x[j])
        if minimum == xi:
            t[i + 1] = ti
            x[i + 1] = xi
    print(i + 1)
