#include <iostream>
#include <string>
#include <vector>
using namespace std;

int sumHammingDistance(const string &a, const string &b) {
  int n = a.size(), m = b.size();
  vector<int> cnt(1 << n);
  for (int i = 0; i < m; i++) {
    int mask = 0;
    for (int j = i; j < i + n && j < m; j++) {
      mask |= (1 << (b[j] - '0'));
    }
    cnt[mask]++;
  }
  int ans = 0;
  for (int i = 0; i < (1 << n); i++) {
    int c = 0;
    for (int j = 0; j < n; j++) {
      if (((i >> j) & 1) != (a[j] - '0')) {
        c++;
      }
    }
    ans += c * cnt[i];
  }
  return ans;
}

int main() {
  string a, b;
  cin >> a >> b;
  cout << sumHammingDistance(a, b) << endl;
  return 0;
}
