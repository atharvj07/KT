#include<bits/stdc++.h>
using namespace std;
int n,m;
vector<int>adj[10005];
int v[10005][15],ind[10005][15];
bool vis[10005];
int mx;
void dfs(int node,int par,int pre){
    vis[node]=true;
    for(int i=0;i<=10;i++){
        if(pre!=i){
            v[node][i]=v[par][pre];
        }
    }
    for(int i=0;i<adj[node].size();i++){
        int x=adj[node][i];
        if(vis[x]==false){
            int c=0;
            for(int j=1;j<=10;j++){
                if(v[node][j]>=1){
                    v[x][j]=v[node][j]-1;
                    ind[x][j]=j;
                    c++;
                }
            }
            if(c>0){
                dfs(x,node,ind[node][c]);
            }
            else{
                dfs(x,node,pre);
            }
        }
    }
    int ans=0;
    for(int i=1;i<=10;i++){
        if(v[node][i]>0)ans++;
    }
    mx=max(mx,ans);
}
int main(){
    scanf("%d %d",&n,&m);
    memset(v,0,sizeof v);
    for(int i=1;i<=7;i++){
        v[0][i]=3;
    }
    while(m--){
        int x,y;
        scanf("%d %d",&x,&y);
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    dfs(1,0,0);
    printf("%d\n",mx);
    return 0;
}
