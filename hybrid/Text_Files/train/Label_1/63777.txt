import sys
from collections import defaultdict

n = int(input())
d = defaultdict(list)
for i in range(n):
    a, b = map(int, sys.stdin.readline().split())
    d[a].append(b)
    d[b].append(a)

a = []
b = []
for i in range(1, 2 * n + 1):
    if len(d[i]) == 1:
        a.append(i)
        b.append(d[i][0])
    elif len(d[i]) == 2:
        x, y = d[i]
        if x < y:
            a.append(x)
            b.append(y)
        else:
            b.append(x)
            a.append(y)
    else:
        print(-1)
        exit()

if sorted(a) != a or sorted(b, reverse=True) != b:
    print(-1)
else:
    print(sum(x > y for x, y in zip(a, b)))
