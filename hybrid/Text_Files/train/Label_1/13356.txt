#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define F first
#define S second

const ll maxn = 1e5 + 5;
ll h[maxn], dp[maxn];
int main(){
    ll n, i;
    cin >> n;
    for (i = 1; i <= n; ++i){
        cin >> h[i];
    }
    dp[1] = h[1] + 1; // Init dp state
    for (i = 2; i <= n; ++i){
        dp[i] = min(dp[i - 1], h[i - 1] + 1) + h[i] + 1;
    }
    cout << dp[n] << '\n';
    return 0;
}
