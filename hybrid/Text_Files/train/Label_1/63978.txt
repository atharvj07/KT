import sys

input = sys.stdin.readline
n, q = map(int, input().split())
grid = [[0] * (n + 1) for _ in range(3)]
for _ in range(q):
    x, y = map(int, input().split())
    grid[x][y] ^= 1
    can = True
    for i in range(1, n + 1):
        if not grid[1][i] and not grid[2][i]:
            can = False
            break
    print(['No', 'Yes'][can])
