import sys
input = sys.stdin.readline
n = int(input())
x = list(map(int, input().split()))
x.sort()
inc = True
dec = False
for i in range(1, len(x)):
    if x[i] == x[i - 1]:
        print("yes")
        exit()
    if x[i] < x[i - 1]:
        inc = False
    if x[i] > x[i - 1]:
        dec = False
if inc and dec:
    print("yes")
else:
    print("no")
