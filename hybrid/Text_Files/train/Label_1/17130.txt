#include<bits/stdc++.h>
#define nl "\n"
using namespace std;
char s[10][10];
int dx[]={0,0,-1,1};
int dy[]={-1,1,0,0};
int main(){
    for(int i=1;i<=6;i++){
        for(int j=1;j<=8;j++){
            cin>>s[i][j];
        }
    }
    int ans=0,x=-1,y=-1;
    for(int i=1;i<=6;i++){
        for(int j=1;j<=8;j+=2){
            int cnt=0;
            for(int k=0;k<4;k++){
                int xx=i+dx[k];
                int yy=j+dy[k];
                if(s[xx][yy]=='.') cnt++;
            }
            if(cnt>ans){
                ans=cnt;
                x=i;
                y=j;
            }
        }
    }
    s[x][y]='P';
    for(int i=1;i<=6;i++){
        for(int j=1;j<=8;j++){
            cout<<s[i][j];
        }
        cout<<nl;
    }
    return 0;
}
