import java.util.Scanner;

public class FloodRisk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int w = scanner.nextInt();
        int d = scanner.nextInt();
        int n = scanner.nextInt();

        int[][] grid = new int[w][d];
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            int z = scanner.nextInt();
            grid[x][y] = z;
        }

        int totalAltitude = 0;
        for (int i = 0; i < w; i++) {
            for (int j = 0; j < d; j++) {
                if (grid[i][j] == 0) {
                    // Find the minimum and maximum possible altitudes for the current cell.
                    int minAltitude = 0;
                    int maxAltitude = 0;
                    if (i > 0) {
                        minAltitude = Math.max(minAltitude, grid[i - 1][j] - 1);
                        maxAltitude = Math.min(maxAltitude, grid[i - 1][j] + 1);
                    }
                    if (i < w - 1) {
                        minAltitude = Math.max(minAltitude, grid[i + 1][j] - 1);
                        maxAltitude = Math.min(maxAltitude, grid[i + 1][j] + 1);
                    }
                    if (j > 0) {
                        minAltitude = Math.max(minAltitude, grid[i][j - 1] - 1);
                        maxAltitude = Math.min(maxAltitude, grid[i][j - 1] + 1);
                    }
                    if (j < d - 1) {
                        minAltitude = Math.max(minAltitude, grid[i][j + 1] - 1);
                        maxAltitude = Math.min(maxAltitude, grid[i][j + 1] + 1);
                    }

                    // If there is no valid altitude range, output "No".
                    if (minAltitude > maxAltitude) {
                        System.out.println("No");
                        return;
                    }

                    // Choose the minimum possible altitude for the current cell.
                    grid[i][j] = minAltitude;
                }
                totalAltitude += grid[i][j];
            }
        }

        System.out.println(totalAltitude);
    }
}
