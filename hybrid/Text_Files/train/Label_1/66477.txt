import sys
readline = sys.stdin.readline

t = int(readline())

for _ in range(t):
    s = input()
    n = len(s)
    e = [0] * n
    for i in range(n):
        if s[i] == 'E':
            e[i] = 1
            e[(i + 1) % n] = 1
    ok = True
    for i in range(n):
        if e[i] and e[(i + 1) % n]:
            ok = False
    if ok:
        print('YES')
    else:
        print('NO')
