import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Edge[] edges = new Edge[m];
        for (int i = 0; i < m; i++) {
            edges[i] = new Edge(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        int[] dist = dijkstra(n, edges, 1);
        int ans = 0;
        for (Edge edge : edges) {
            int u = edge.u;
            int v = edge.v;
            int w = edge.w;
            if (dist[u] + w == dist[v]) ans++;
            if (dist[v] + w == dist[u]) ans++;
        }
        System.out.println(ans);
    }

    static int[] dijkstra(int n, Edge[] edges, int start) {
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[start] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>(Comparator.comparingInt(e -> e.w));
        pq.add(new Edge(start, start, 0));
        while (!pq.isEmpty()) {
            Edge edge = pq.poll();
            int u = edge.u;
            int v = edge.v;
            int w = edge.w;
            if (dist[u] + w > dist[v]) continue;
            dist[v] = dist[u] + w;
            for (Edge next : edges) {
                if (next.u == v) pq.add(new Edge(v, next.v, next.w));
                if (next.v == v) pq.add(new Edge(v, next.u, next.w));
            }
        }
        return dist;
    }

    static class Edge {
        int u;
        int v;
        int w;

        public Edge(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }
}
