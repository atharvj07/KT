#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, a[1005];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0), cout.tie(0);
    cin >> n;
    while (n--) {
        int x;
        cin >> x;
        int lz = __builtin_clzll(x);
        cout << ((x & (1 << (lz - 1))) ? ((1 << lz) / 2) : 1) << "\n";
    }
    return 0;
}
