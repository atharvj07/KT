import sys

n = int(input())
B = sorted(int(x) for x in input().split())
ans = [0] * n
B.append(n)
for i in range(n):
    b = B[i] + 1
    e = B[i + 1]
    f = (b + e) // 2
    if (b + e) % 2 == 0:
        ans[i * 2 + 1] = f
    if b < f:
        ans[i * 2] = b
    if f < e:
        ans[i * 2] = f
print(*ans)
