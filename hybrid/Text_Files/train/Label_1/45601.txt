import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] A = new int[M];
        int[] B = new int[M];
        for (int i = 0; i < M; i++) {
            A[i] = sc.nextInt();
            B[i] = sc.nextInt();
        }
        sc.close();

        List<Set<Integer>> G = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            G.add(new HashSet<>());
        }
        for (int i = 0; i < M; i++) {
            G.get(A[i] - 1).add(B[i] - 1);
            G.get(B[i] - 1).add(A[i] - 1);
        }

        int[] ans = new int[2];
        int[] vis = new int[N];
        for (int i = 0; i < N; i++) {
            if (vis[i] == 0) {
                dfs(i, G, vis, ans);
            }
        }

        if (ans[0] == 0 || ans[1] == 0) {
            System.out.println(-1);
        } else {
            System.out.println(M - ans[2]);
        }
    }

    static void dfs(int u, List<Set<Integer>> G, int[] vis, int[] ans) {
        vis[u] = 1;
        ans[0]++;
        for (int v : G.get(u)) {
            if (vis[v] == 0) {
                dfs(v, G, vis, ans);
                ans[2]++;
            } else if (vis[v] == 1) {
                ans[1]--;
            }
        }
        vis[u] = 2;
    }
}
