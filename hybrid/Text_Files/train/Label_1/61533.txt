from sys import stdin
input = stdin.readline
 
n, m = map(int, input().strip().split())
p = list(map(int, input().strip().split()))
foe = [list(map(int, input().strip().split())) for _ in range(m)]
idx = [*range(n+1)]
for a, b in foe:
    idx[p[a-1]], idx[p[b-1]] = idx[p[b-1]], idx[p[a-1]]
 
ans = 0
for i in idx:
    if i:
        ans += i - idx[i-1]
print(ans)
