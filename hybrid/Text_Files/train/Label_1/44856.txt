import java.io.*;
import java.util.*;

public class Main {
    static final int[] dx = {1, -1};
    static final int[] dp = {1, -1};

    static class Node {
        int idx, v;
        public Node(int idx, int v) { this.idx = idx; this.v = v; }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        String s = br.readLine();
        int[] cnt = new int[10];
        for (int i = 0; i + 1 < n; i++) {
            if (Character.isDigit(s.charAt(i)) && Character.isDigit(s.charAt(i + 1))) {
                cnt[s.charAt(i) - '0']++;
            }
        }
        while (q-- > 0) {
            st = new StringTokenizer(br.readLine());
            int l = Integer.parseInt(st.nextToken()) - 1;
            int r = Integer.parseInt(st.nextToken()) - 1;
            ArrayList<Node> list = new ArrayList<>();
            int d = 1;
            for (int i = l; i <= r; i++) {
                if (i != l && s.charAt(i) != s.charAt(i - 1)) {
                    d = -d;
                }
                if (Character.isDigit(s.charAt(i))) {
                    list.add(new Node(i, s.charAt(i) - '0'));
                }
            }
            int p = l;
            for (int i = 0; i < list.size(); ) {
                int j = i;
                while (j < list.size() && list.get(j).idx <= p) {
                    j++;
                }
                p++;
                if (i < j) {
                    cnt[list.get(i).v]--;
                    d = -d;
                    i = j;
                }
            }
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < 10; i++) {
                sb.append(cnt[i]).append(" ");
            }
            sb.append("\n");
            bw.write(sb.toString());
        }
        bw.close();
    }
}
