import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

class Main {
    static int[][] g;
    static int[][] d;
    static int[] dx = {1, -1, 0, 0};
    static int[] dy = {0, 0, 1, -1};
    static int w, h, n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        w = sc.nextInt();
        h = sc.nextInt();
        g = new int[w + 2][h + 2];
        d = new int[w + 2][h + 2];
        for (int i = 0; i < w + 2; i++) {
            g[i][0] = g[i][h + 1] = -1;
        }
        for (int j = 0; j < h + 2; j++) {
            g[0][j] = g[w + 1][j] = -1;
        }
        for (int i = 1; i <= n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            g[x][y] = i;
        }

        int ans = bfs();
        System.out.println(ans);
    }

    static int bfs() {
        Queue<Integer> qx = new LinkedList<>();
        Queue<Integer> qy = new LinkedList<>();
        Queue<Integer> qs = new LinkedList<>();
        for (int x = 1; x <= w; x++) {
            for (int y = 1; y <= h; y++) {
                if (g[x][y] != 0) {
                    qx.add(x);
                    qy.add(y);
                    qs.add(0);
                }
            }
        }

        int ret = 0;
        while (!qx.isEmpty()) {
            int x = qx.poll();
            int y = qy.poll();
            int s = qs.poll();
            ret = s;

            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (g[nx][ny] == 0) {
                    qx.add(nx);
                    qy.add(ny);
                    qs.add(s + 1);
                } else if (g[nx][ny] != g[x][y]) {
                    return s + 1;
                }
            }
        }
        return ret;
    }
}
