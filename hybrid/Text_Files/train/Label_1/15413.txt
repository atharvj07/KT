#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, a;
    cin >> n;
    priority_queue<long long, vector<long long>, greater<long long> > pq;
    for (long long i = 0; i < n; i++) {
        cin >> a;
        pq.push(a);
    }
    long long ans = pq.top();
    while (!pq.empty() && ans == pq.top()) {
        pq.pop();
        ans++;
    }
    cout << ans << endl;
    return 0;
}
