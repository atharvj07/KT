#include <bits/stdc++.h>
using namespace std;
int n;
vector <int> a;
int main() {
    cin >> n;
    a.resize(n + 1);
    for (int i = 1; i <= n; ++i) {
        int x;
        cin >> x;
        a[x] = i;
    }
    int cnt = 0;
    for (int i = 1; i <= n; ++i) {
        if (a[i] == i) {
            continue;
        }
        if ((a[i] - i) % 2 == 0) {
            swap(a[i], a[a[i]]);
            printf("%d %d\n", i, a[i]);
            cnt++;
        } else {
            for (int j = i + 1; j <= n; ++j) {
                if ((j - a[i]) % 2 != 0 && (j - i) % 2 != 0) {
                    swap(a[i], a[j]);
                    printf("%d %d\n", i, j);
                    swap(a[i], a[a[i]]);
                    printf("%d %d\n", i, a[i]);
                    cnt += 2;
                    break;
                }
            }
        }
    }
    cout << cnt << '\n';
    return 0;
}
