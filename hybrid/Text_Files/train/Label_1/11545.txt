#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, a[205], sum[205];
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  sort(a + 1, a + n + 1);
  sum[0] = 0;
  for (int i = 1; i <= n; i++) { sum[i] = sum[i - 1] + a[i]; }
  int ans = sum[n];
  int l = n, r = n + 1;
  for (int i = 1; i <= k; i++) {
    while (r <= n && a[r] < 0) { r++; }
    l = min(l, r);
    r++;
    if (l > n) { break; }
    int x = a[l], y = a[r - 1];
    sum[0] += x + y;
    sum[l] -= x + y;
    sum[r] += x + y;
    ans = max(ans, sum[l] + sum[n] - sum[r]);
  }
  cout << ans << endl;
  return 0;
}
