#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) { cin >> x; }
    sort(a.begin(), a.end());
    int days = 0;
    int j = 0;
    for (int i = 1;; i++) {
        while (j < n && a[j] < i) j++;
        if (j == n) break;
        days++;
        j++;
    }
    cout << days << endl;
    return 0;
}
