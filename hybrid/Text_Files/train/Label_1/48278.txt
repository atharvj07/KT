import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        in.close();

        long ans = 0;
        int[] b = new int[n + 2];
        System.arraycopy(a, 0, b, 1, n);
        b[0] = Integer.MAX_VALUE;
        b[n + 1] = Integer.MAX_VALUE;
        for (int i = 0; i < k; i++) {
            if (b[1] > 0) {
                ans += b[1];
                b[1] = Integer.MAX_VALUE;
                for (int j = 1; j <= n; j++) if (b[j] != Integer.MAX_VALUE) { b[j - 1] = b[j]; b[j] = Integer.MAX_VALUE; break; }
            } else {
                if (b[n] > 0) {
                    ans += b[n];
                    b[n] = Integer.MAX_VALUE;
                    for (int j = n; j >= 1; j--) if (b[j] != Integer.MAX_VALUE) { b[j + 1] = b[j]; b[j] = Integer.MAX_VALUE; break; }
                } else {
                    ans += 0;
                }
            }
        }
        System.out.println(ans);
    }
}
