from sys import stdin

input()
a = list(map(int, stdin.readline().split()))
cnt = 0
for i in range(len(a)):
    for j in range(i + 1, len(a)):
        c = 0
        if (i % 2 == j % 2): c += 1
        if (a[i] < a[j] and i % 2 == 0): c += 1
        if (a[i] > a[j] and i % 2 == 1): c += 1
        cnt += (c == 2)
print(cnt)
