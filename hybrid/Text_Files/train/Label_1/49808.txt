import java.util.Scanner;

public class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt(), t = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) a[i] = sc.nextInt();
        int[] deg = new int[n + 1];
        for (int i = 0; i < q; i++) {
            int u = sc.nextInt(), v = sc.nextInt();
            deg[u]++; deg[v]--;
        }

        long[][] dp = new long[2][t + 1];
        dp[0][0] = 1;
        int cur = 0, nxt = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= t; j++) dp[nxt][j] = (dp[nxt][j] + dp[cur][j]) % MOD;
            for (int j = a[i]; j <= t; j++) dp[nxt][j] = (dp[nxt][j] + dp[cur][j - a[i]]) % MOD;
            cur ^= 1; nxt ^= 1;

            // Handle degree
            if (deg[i] > 0) {
                for (int j = 0; j <= t; j++) dp[nxt][j] = 0;
                for (int j = a[i]; j <= t; j++) dp[nxt][j] = (dp[nxt][j] + dp[cur][j - a[i]]) % MOD;
            } else if (deg[i] < 0) {
                for (int j = 0; j <= t; j++) dp[nxt][j] = 0;
                for (int j = 0; j <= t - a[i]; j++) dp[nxt][j + a[i]] = (dp[nxt][j + a[i]] + dp[cur][j]) % MOD;
            }
            cur ^= 1; nxt ^= 1;
        }

        System.out.println(dp[cur][t]);
    }
}
