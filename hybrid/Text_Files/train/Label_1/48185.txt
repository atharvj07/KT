import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    private static class Bank {
        int no;
        int rate;
        int type;
        double interest;

        Bank(int no, int rate, int type) {
            this.no = no;
            this.rate = rate;
            this.type = type;
            this.interest = 0;
        }

        void calculateInterest(double principal, int years) {
            double tmp = principal * rate / 100.0;
            if (type == 2) {
                tmp *= Math.pow(1 + rate / 100.0, years) - 1;
            }
            interest += tmp;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n, y, no, rate, type;
        double principal;

        while (true) {
            n = Integer.parseInt(br.readLine());
            if (n == 0) {
                return;
            }
            y = Integer.parseInt(br.readLine());
            Bank[] banks = new Bank[n];
            for (int i = 0; i < n; i++) {
                StringTokenizer st = new StringTokenizer(br.readLine(), " ");
                no = Integer.parseInt(st.nextToken());
                rate = Integer.parseInt(st.nextToken());
                type = Integer.parseInt(st.nextToken());
                banks[i] = new Bank(no, rate, type);
            }
            principal = 1000000;
            for (Bank bank : banks) {
                bank.calculateInterest(principal, y);
            }
            Bank maxBank = null;
            for (Bank bank : banks) {
                if (maxBank == null || bank.interest > maxBank.interest) {
                    maxBank = bank;
                }
            }
            System.out.println(maxBank.no);
        }
    }
}
