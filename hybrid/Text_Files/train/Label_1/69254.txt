import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    a = []
    b = []
    ans = 0
    for _ in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)
    
    for i in range(1, n):
        if a[i] == a[i - 1]:
            ans += min(b[i], b[i - 1])
            a[i] += 1 if b[i] < b[i - 1] else 0
            
    print(ans)

for _ in range(int(input())):
    solve()
