#include <iostream>
#include <string>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.size(), i, j, l = 0, r = n - 1;
  for (; l < r; l++, r--) if (s[l] != s[r]) break;
  string t = s;
  if (l == r)
    t.insert(l, 1, s[l]);
  else
    t.insert(l + 1, 1, s[l]);
  if (t == string(t.rbegin(), t.rend()))
    cout << t;
  else
    cout << "NA";
}
