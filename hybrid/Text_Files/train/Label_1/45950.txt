import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        List<Long> snukeNumbers = new ArrayList<>();
        snukeNumbers.add(1L);
        long prevSum = 1L;
        int l = 1;
        while (l < k) {
            long sn = snukeNumbers.get(l - 1);
            long sum = sn * 10L + (sn % 10L + 1L) - 1L;
            if (sn * sum >= prevSum * snukeNumbers.get(l - 1)) {
                snukeNumbers.add(sn + 1L);
                l++;
            }
            prevSum = sum;
        }
        for (long sn : snukeNumbers) {
            System.out.println(sn);
        }
    }
}
