#include <algorithm>
#include <iostream>

using namespace std;

int main() {
  int n, k, ans = 0, a[1001] = {}, b[1001] = {};
  cin >> n >> k;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    b[a[i]]++;
  }
  for (int i = 1; i <= k; ++i) ans = max(ans, min(n / 2, b[i] + b[i + k] + b[i + k + k]));
  cout << ans * 2;
  return 0;
}
