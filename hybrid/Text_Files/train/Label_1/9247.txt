#include <iostream>

const int MAXN = 500;
int n, c[MAXN];
bool pal[MAXN][MAXN];

int main() {
  std::cin >> n;
  for (int i = 0; i < n; i++) {
    std::cin >> c[i];
    c[i]--;
    pal[i][i] = true;
  }
  for (int len = 2; len <= n; len++) {
    for (int i = 0; i + len - 1 < n; i++) {
      pal[i][i + len - 1] = c[i] == c[i + len - 1] && pal[i + 1][i + len - 2];
    }
  }
  int ans = n;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      if (pal[i][j]) {
        ans = std::min(ans, (j - i + 1 + 1) / 2);
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
