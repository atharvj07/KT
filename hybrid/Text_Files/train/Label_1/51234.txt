import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine();

        List<Operation> operations = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] line = scanner.nextLine().split(" ");
            int type = Integer.parseInt(line[0]);
            long xi = Long.parseLong(line[1]);
            int yi = Integer.parseInt(line[2]);
            operations.add(new Operation(type, xi, yi));
        }

        int[] earliestTimeSteps = new int[m + 1];
        Arrays.fill(earliestTimeSteps, -1);
        earliestTimeSteps[0] = 0;

        int[] bananas = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            Operation currentOperation = operations.get(i - 1);
            if (currentOperation.type == 1) {
                for (int j = m; j >= 0; j--) {
                    if (earliestTimeSteps[j] != -1) {
                        int newBananas = (int) Math.ceil(bananas[i - 1] + (double) currentOperation.xi / 100000);
                        if (newBananas <= j && (earliestTimeSteps[newBananas] == -1 || earliestTimeSteps[newBananas] > i)) {
                            earliestTimeSteps[newBananas] = i;
                        }
                    }
                }
            } else {
                for (int j = m; j >= 0; j--) {
                    if (earliestTimeSteps[j] != -1) {
                        int newBananas = (int) Math.ceil(bananas[i - 1] * (double) currentOperation.xi / 100000);
                        if (newBananas <= j && (earliestTimeSteps[newBananas] == -1 || earliestTimeSteps[newBananas] > i)) {
                            earliestTimeSteps[newBananas] = i;
                        }
                    }
                }
            }
            bananas[i] = (int) Math.ceil(bananas[i - 1] + (double) currentOperation.xi / 100000);
        }

        for (int i = 1; i <= m; i++) {
            System.out.print(earliestTimeSteps[i] + " ");
        }
    }

    private static class Operation {
        int type;
        long xi;
        int yi;

        public Operation(int type, long xi, int yi) {
            this.type = type;
            this.xi = xi;
            this.yi = yi;
        }
    }
}
