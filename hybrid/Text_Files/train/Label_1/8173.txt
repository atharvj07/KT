#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int mxN = 1e5;
int n;
vector<int> adj[mxN + 1];

void add_edge(int u, int v) { adj[u].push_back(v); adj[v].push_back(u); }

int main() {
    cin >> n;
    for (int i = 0; i < n - 2; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        add_edge(a, b);
        add_edge(a, c);
        add_edge(b, c);
    }

    vector<bool> vis(n + 1, 0);
    vector<int> ans;
    function<void(int)> dfs = [&](int u) {
        vis[u] = 1;
        ans.push_back(u);

        for (int v : adj[u]) {
            if (!vis[v]) dfs(v);
        }
    };

    for (int i = 1; i <= n; i++) {
        if (!vis[i]) dfs(i);
    }

    for (int x : ans) cout << x << ' ';
    cout << '\n';
}
