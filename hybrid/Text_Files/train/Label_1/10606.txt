#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  for (int i = 0; i < n; i++) {
    int sumX = 0, mxY = 0;
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      sumX += v[j].first;
      mxY = max(mxY, v[j].second);
    }
    cout << sumX * mxY << " ";
  }
  cout << endl;
  return 0;
}
