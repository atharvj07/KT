#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
const int maxN=510;
int dp[(1<<maxN)][maxN];
int a[maxN][maxN];
bool f(bitset<maxN> b, int last, int n){
    if(b.count()==n)return 1;
    int& ans=dp[b.to_ullong() ][last];
    if(~ans)return ans;
    ans=0;
    for(int i=0;i<n;i++){
        if(b.test(i) || i==last || a[last][i]==0)continue;
        ans+=f(b|bitset<maxN>(1<<i),i,n);
        ans%=mod;
    }
    return ans;
}
int main(){
    int n;cin>>n;
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++)
            cin>>a[i][j];
    memset(dp,-1,sizeof dp);
    cout<<f(bitset<maxN>(1<<0),0,n)<<'\n';
}
