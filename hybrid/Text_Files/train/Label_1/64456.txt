def get_minimum_tiles_needed(tiles):
  count = dict()
  for tile in tiles:
    if tile[1] not in count:
      count[tile[1]] = 0
    count[tile[1]] += 1
  ans = 0
  for suit in count:
    if count[suit] >= 3:
      ans += 0
    elif count[suit] >= 1:
      ans += 2
    else:
      ans += 3
  return ans

tiles = input().split()
print(get_minimum_tiles_needed(tiles))
