#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  if (n == 1)
    cout << "-1" << endl;
  else if (n == 2)
    cout << "2 1" << endl;
  else {
    vector<int> p(n + 1);
    p[1] = 2;
    for (int i = 2; i <= n; i += 2)
      p[i] = i + 1, p[i + 1] = i;
    for (int i = 1; i <= n; i++)
      cout << p[i] << " ";
    cout << endl;
  }
  return 0;
}
