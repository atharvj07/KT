import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int flag = 0;
        for (int i = 1; i < n; i++) {
            if (arr[i] * arr[i - 1] <= 0) {
                flag = 2;
                break;
            }
            if (arr[i] % arr[i - 1] != 0) {
                flag = 1;
            }
        }
        double r = 0;
        if (flag == 1) {
            for (int i = 1; i < n; i++) {
                double tmp = (double) arr[i] / arr[i - 1];
                if (i == 1) {
                    r = tmp;
                }
                if (r != tmp) {
                    flag = 2;
                    break;
                }
            }
        }
        System.out.println(flag);
    }
}
