#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int MOD = 1e9 + 7;

vector<int> divisors;

void precalc(int n) {
    for (int i = 1; i * i <= n; ++i) {
        if (n % i == 0) {
            divisors.push_back(i);
            if (i * i != n) divisors.push_back(n / i);
        }
    }
    sort(divisors.begin(), divisors.end());
}

bool is_adjacent(int x, int y) {
    int g = __gcd(x, y);
    for (int d : divisors) {
        if ((x / g) % d == 0 && (y / g) % d == 0) return true;
    }
    return false;
}

vector<int> adj[300005];

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        int q;
        cin >> q;
        int mx_beauty = 0;
        for (int i = 0; i < n; ++i) {
            adj[a[i]].push_back(i);
            mx_beauty = max(mx_beauty, (int)adj[a[i]].size());
        }
        while (q--) {
            ll w;
            cin >> w;
            vector<int> na(n);
            for (int i = 0; i < n; ++i) {
                ll prod = 1;
                for (int j : adj[a[i]]) {
                    if (is_adjacent(a[i], a[j])) {
                        prod = (prod * a[j]) % MOD;
                    }
                }
                na[i] = (prod * a[i]) % MOD;
            }
            for (int i = 0; i < n; ++i) {
                a[i] = na[i];
                adj[a[i]].clear();
                adj[a[i]].push_back(i);
            }
            cout << mx_beauty << '\n';
        }
        for (int i = 0; i < 300005; ++i) adj[i].clear();
        divisors.clear();
    }
    return 0;
}
