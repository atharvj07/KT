#include <vector>
#include <iostream>
#include <tuple>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n-1);
  for (int i = 0; i < n-1; ++i) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = 1;
  for (int i = 0; i < n-1; ++i) {
    if (a[i] == ans) ++ans;
  }
  cout << ans << endl;
  return 0;
}
