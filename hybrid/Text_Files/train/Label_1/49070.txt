import java.util.*;
import java.io.*;

public class Main {
    static long f(long[] arr) {
        long ans = 0;
        for (long x : arr) ans ^= x;
        return ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long l = in.nextLong(), r = in.nextLong(), k = in.nextLong();
        long[] arr = new long[100];
        long cnt = 0;
        while (l <= r && cnt < k) {
            arr[(int)cnt] = l;
            cnt++;
            l += 1;
        }
        System.out.println(f(arr));
        System.out.println(cnt);
        if (cnt > 0) System.out.print(arr[0]);
        for (int i = 1; i < cnt; i++) System.out.print(" " + arr[i]);
    }
}
