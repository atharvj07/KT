import java.util.Scanner;

public class CaracalAndMonster {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long monsterHealth = scanner.nextLong();

        int attacks = 0;
        while (monsterHealth > 0) {
            monsterHealth = attack(monsterHealth);
            attacks++;
        }

        System.out.println(attacks);
    }

    private static long attack(long monsterHealth) {
        if (monsterHealth == 1) {
            return 0;
        } else {
            return (monsterHealth - 1) / 2;
        }
    }
}
