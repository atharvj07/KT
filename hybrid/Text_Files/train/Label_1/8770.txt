#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        vector<int> v(4);
        for (auto &x : v) cin >> x;
        sort(v.begin(), v.end());
        if (v[2] == v[0] + v[1])
            cout << "YES\n";
        else
            cout << "NO\n";
    }
}
