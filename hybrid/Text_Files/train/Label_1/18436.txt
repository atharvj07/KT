#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n, k, x, ans = 0, cur_max = 0, cur_min = 1e18;
    cin >> n >> k >> x;
    map<int, int> f;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        f[a[i]]++;
        cur_max = max(cur_max, a[i]);
        cur_min = min(cur_min, a[i]);
    }
    int diff = cur_max - cur_min;
    if (diff <= x) ans = 1;
    else {
        int num_gaps = (diff + x) / x;
        int g = floor((cur_min + x - 1) / x);
        int l = g * x;
        int r = l + x - 1;
        for (int i = 0; i < n; i++) {
            if (l <= a[i] && a[i] <= r) f[r]--;
        }
        k -= f[r];
        f.erase(r);
        auto it = f.begin();
        while (it != f.end()) {
            if (it->second > k) {
                break;
            }
            k -= it->second;
            f.erase(it);
            ans++;
            it = f.begin();
            if (it == f.end()) break;
            if (it->first <= r) {
                k -= f[r];
                f.erase(r);
                it = f.begin();
            }
        }
        if (f.size()) ans++;
    }
    cout << ans << endl;
    return 0;
}
