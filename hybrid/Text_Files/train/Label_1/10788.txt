#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
#define all(x) begin(x), end(x)
const ll mod = 1e9 + 7;

int main() {
  ll n;
  string s;
  cin >> n >> s;
  ll i, j, mask = 0;
  for (i = 0; i < n; i++) {
    if (s[i] == '0') mask |= (1LL << (i));
  }
  ll t = 1LL << n, ans = 0;
  for (i = 0; i < t; i++) {
    if ((i & mask) == mask) {
      bool ok = 1;
      for (j = 0; j < n; j++) {
        if (((i & (1LL << j)) > 0) != (s[j] == '1')) {
          ok = 0;
          break;
        }
      }
      if (ok) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
