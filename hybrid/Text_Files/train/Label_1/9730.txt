#include <bits/stdc++.h>
using namespace std;
int a[27];
int main(){
    string s;cin>>s;
    for (char x:s) a[x-'A']++;
    if (a[0]<2||a[25]<2) {cout<<"Impossible";return 0;}
    for (int i=0;i<2;++i) {
        a[13+i]-=2;
        for (int j=0;j<13;++j,i+=!i) {
            if (a[12+j]) --a[12+j],cout<<'A'+12+j;
            else if (a[j]) --a[j],cout<<'A'+j;
            else if (a[25+i]) --a[25+i],cout<<'A'+25+i;
        }
        cout<<endl;
    }
}
