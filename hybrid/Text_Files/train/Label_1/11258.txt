#include<bits/stdc++.h>
using namespace std;
int main(){
    string t, s;
    cin >> t >> s;
    int sz1 = s.size(), sz2 = t.size(), ans = 0;
    while (sz1 || sz2) {
        if (!sz1) {
            ans +=  sz2; break;
        }
        if (!sz2) {
            ans += sz1; break;
        }
        if (s[sz1 - 1] == t[sz2 - 1]) {
            --sz1; --sz2;
        } else {
            ++ans; --sz1;
        }
    }
    cout << ans << endl;
    return 0;
}
