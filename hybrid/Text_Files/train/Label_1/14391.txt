#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a, b;
  while (cin >> n >> a >> b) {
    while (n > 2) {
      if (abs(a - b) <= 1) {
        cout << log2(n / 2) + 1 << endl;
        return 0;
      }
      n /= 2;
      a = (a + n / 2 - 1) % n + 1;
      b = (b + n / 2 - 1) % n + 1;
    }
    cout << "Final!" << endl;
  }
  return 0;
}
