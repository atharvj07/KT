#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define vi vector<int>
#define pb push_back
#define rsz resize

int n, ans, cnt;
vi adj[200001];
bool vis[200001];

void dfs(int u, int p, int dep) {
    vis[u] = true;
    if (dep > 2) ans++;
    for (auto v : adj[u]) {
        if (v == p) continue;
        if (!vis[v]) {
            dfs(v, u, dep + 1);
            if (vis[v]) cnt++;
        }
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].pb(v);
        adj[v].pb(u);
    }
    dfs(1, 0, 0);
    cout << ans - cnt / 2 << endl;
    return 0;
}
