from collections import deque
import sys

n,m=map(int,input().split())
sx,sy,fx,fy=map(int,input().split())
x,y=[],[]
for i in range(m):
    a,b=map(int,input().split())
    if a==sx or b==sy:
        x.append((a,b))
    if a==fx or b==fy:
        y.append((a,b))
#
x.sort()
y.sort()
#
if not x and not y:
    print(abs(fx-sx)+abs(fy-sy))
elif not x:
    print(min(abs(fx-a)+abs(fy-b)+abs(a-sx)+abs(b-sy) for a,b in y))
elif not y:
    print(min(abs(fx-a)+abs(fy-b)+abs(a-sx)+abs(b-sy) for a,b in x))
else:
    if x[0][0]==x[-1][0]: a=x[0][0]
    elif x[0][1]==x[-1][1]: a=x[0][1]
    if y[0][0]==y[-1][0]: b=y[0][0]
    elif y[0][1]==y[-1][1]: b=y[0][1]
    #
    ans=abs(fx-a)+abs(fy-b)+abs(a-sx)+abs(b-sy)
    #
    if a==sx:
        ans=min(ans,abs(fx-a)+abs(fy-b)+abs(b-sy))
    if b==sy:
        ans=min(ans,abs(fx-a)+abs(fy-b)+abs(a-sx))
    if a==fx:
        ans=min(ans,abs(fx-a)+abs(b-sy)+abs(a-sx))
    if b==fy:
        ans=min(ans,abs(b-fy)+abs(fx-a)+abs(a-sx))
    #
    print(ans)
