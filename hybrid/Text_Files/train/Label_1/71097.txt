import sys
input=sys.stdin.readline
n,k=map(int,input().split())
adj=[[]for i in range(n+1)]
for i in range(n-1):
    a,b=map(int,input().split())
    adj[a].append(b)
    adj[b].append(a)
dp=[-1]*(n+1)
def dfs(cur,par,d,sum):
    dp[cur]=d
    sum+=d
    for nxt in adj[cur]:
        if nxt!=par:
            sum=dfs(nxt,cur,min(k,d+1),sum)
    return sum
print(dfs(1,1,0,0))
