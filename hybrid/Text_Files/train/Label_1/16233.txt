#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;
int main() {
    int n,x;
    long long ans=0;
    cin >> n;
    vector<int>v(n);
    for(int i=0; i<n; i++) {
        cin >> v[i];
    }
    if(v[n-1] >= 63) {
        cout << 0 << endl;
        return 0;
    }
    x=max(v[0],63-v[n-1]);
    for(int i=0; i<n; i++) {
        if(x<=v[i]) {
            break;
        }
        ans++;
    }
    cout << ans << endl;
}
