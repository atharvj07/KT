#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<vector<int>> dp(10, vector<int>(10, -1));
  for (int x = 0; x < 10; x++) {
    for (int y = 0; y < 10; y++) {
      int last = 0;
      int ans = 0;
      for (int i = 0; i < n; i++) {
        int d = s[i] - '0';
        if (d != last) {
          if (d == x || d == y) {
            ans++;
            last = d;
          } else {
            ans = -1;
            break;
          }
        }
      }
      dp[x][y] = ans;
    }
  }
  for (int x = 0; x < 10; x++) {
    for (int y = 0; y < 10; y++) {
      cout << dp[x][y] << " ";
    }
    cout << endl;
  }
  return 0;
}
