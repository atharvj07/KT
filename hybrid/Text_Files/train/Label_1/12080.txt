#include<bits/stdc++.h>
using namespace std;
int main(){
    int a[6],b[100005],n,mn=1e9+7;
    for(int i=0;i<6;i++) cin>>a[i];
    cin>>n;
    for(int i=0;i<n;i++) cin>>b[i];
    for(int i=0;i<n;i++){
        mn=min(mn,b[i]-a[0]);
        for(int j=1;j<6;j++){
            mn=min(mn,b[i]-a[j]);
        }
    }
    cout<<mn<<endl;
    return 0;
}
