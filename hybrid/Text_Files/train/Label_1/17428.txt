#include <vector>
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<int> freq(1e5 + 5, 0);
  vector<pair<int, int>> factors;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    int pf = 2;
    while (x > 1) {
      if (x % pf == 0) {
        factors.push_back({pf, 0});
        while (x % pf == 0) { x /= pf; factors.back().second++; }
      }
      pf++;
    }
  }
  sort(factors.begin(), factors.end());
  int ans = 1;
  for (auto p : factors) {
    if (freq[p.first] > 0) {
      ans = (1LL * ans * (p.second + freq[p.first])) % mod;
    } else {
      ans = (1LL * ans * (p.second + 1)) % mod;
    }
    freq[p.first] = p.second + 1;
  }
  cout << ans << '\n';
}
