#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, m, x, y;
    std::cin >> n >> m >> x >> y;
    std::vector<std::vector<char>> grid(n, std::vector<char>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) { std::cin >> grid[i][j]; }
    }
    long long cost = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (grid[i][j] == '.') {
          if ((j + 1 < m && grid[i][j + 1] == '.') || (i + 1 < n && grid[i + 1][j] == '.'))
            cost += y;
          else
            cost += x;
        }
      }
    }
    std::cout << cost << std::endl;
  }
  return 0;
}
