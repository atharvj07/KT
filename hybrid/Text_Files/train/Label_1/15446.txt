#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    int g = 2;
    for (int i = 0; i < n; i++) {
        g = gcd(g, v[i].first);
        g = gcd(g, v[i].second);
    }
    if (g == 1) cout << -1;
    else cout << g;
    return 0;
}
