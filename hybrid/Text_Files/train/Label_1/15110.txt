#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    bool f = true;
    int x, y, px = 0, py = 0;
    for (int i = 0; i < n; i++) {
        cin >> x >> y;
        if (!px && !py) {
            px = x, py = y;
            continue;
        }
        if (x * py != y * px) f = false;
    }
    cout << (f ? "Yes" : "No") << endl;
    return 0;
}
