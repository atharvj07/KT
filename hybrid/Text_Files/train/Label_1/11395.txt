#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    string s;
    cin >> s;
    ll n = s.size(), ans = 0;
    for (ll i = 0; i < n; i++) {
        ll cnt = 1;
        for (ll j = i + 1; j < n; j++) {
            if (s[j] == s[j - 1])
                cnt++;
            else
                break;
        }
        ans += cnt * (cnt + 1) / 2;
    }
    cout << ans << '\n';
    return 0;
}
