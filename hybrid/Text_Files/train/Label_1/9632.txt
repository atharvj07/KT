#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 998244353;
struct node{
    ll sum, prv;
    node(){}
    node(node l, node r){
        sum = (l.sum + r.sum) % mod;
        prv = (l.prv + r.prv) % mod;
    }
    node(ll _sum, ll _prv){
        sum = _sum;
        prv = _prv;
    }
};
struct segTree{
    vector<node> seg;
    segTree(vector<ll> &v){
        seg.resize(v.size() * 4);
        build(1, 0, v.size() - 1, v);
    }
    void build(ll index, ll l, ll r, vector<ll> &v){
        if(l == r){
            seg[index] = node(v[l], v[l] % 3);
            return;
        }
        ll mid = (l + r) / 2;
        build(index * 2, l, mid, v);
        build(index * 2 + 1, mid + 1, r, v);
        seg[index] = node(seg[index * 2], seg[index * 2 + 1]);
    }
    node query(ll index, ll l, ll r, ll ql, ll qr){
        if(l >= ql && r <= qr){
            return seg[index];
        }
        if(l > qr || r < ql){
            return node(0, 0);
        }
        ll mid = (l + r) / 2;
        return node(query(index * 2, l, mid, ql, qr), query(index * 2 + 1, mid + 1, r, ql, qr));
    }
    void update(ll index, ll l, ll r, ll pos, ll val){
        if(l == r){
            seg[index] = node(val, val % 3);
            return;
        }
        ll mid = (l + r) / 2;
        if(pos <= mid){
            update(index * 2, l, mid, pos, val);
        }else{
            update(index * 2 + 1, mid + 1, r, pos, val);
        }
        seg[index] = node(seg[index * 2], seg[index * 2 + 1]);
    }
};
int main(){
    string s; cin >> s;
    ll n = s.size();
    vector<ll> v(n, 0);
    v[0] = s[0] - '0';
    for(ll i = 1; i < n; i++){
        v[i] = (v[i - 1] * 10) % 3;
        v[i] = (v[i] + (s[i] - '0')) % 3;
    }
    segTree t(v);
    vector<ll> dp(n, 0);
    dp[0] = 1;
    for(ll i = 1; i < n; i++){
        node q = t.query(1, 0, n - 1, 0, i - 1);
        if(q.prv == 0){
            dp[i] = (dp[i] + (q.sum + 1)) % mod;
        }else{
            dp[i] = (dp[i] + q.sum) % mod;
        }
        dp[i] = (dp[i] + dp[i - 1]) % mod;
        t.update(1, 0, n - 1, i, 0);
    }
    cout << dp[n - 1] << "\n";
    return 0;
}
