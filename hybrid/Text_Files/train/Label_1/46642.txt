import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] on = new int[m + 1];
        for (int i = 0; i < n; ++i) {
            int k = sc.nextInt();
            for (int j = 0; j < k; ++j) {
                on[sc.nextInt()] = 1;
            }
        }
        System.out.println(Arrays.stream(on).sum() == m ? "YES" : "NO");
    }
}
