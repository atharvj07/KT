#include<bits/stdc++.h>

using namespace std;

int main()
{
    int n, m;
    cin >> n >> m;

    vector<int> v(n);
    for(int i = 0; i < n; i++) cin >> v[i];

    int sum = 0;
    for(int i = 0; i < n; i++){
        if(sum > m) sum = m;
        sum+=v[i];
        cout << (sum > m) << " ";
    }

    return 0;
}
