import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int H = sc.nextInt(), W = sc.nextInt();
        int[][] matrix = new int[H + 1][W + 1];
        int[][][] dp = new int[H + 1][W + 1][3];

        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                matrix[i][j] = sc.nextInt();
            }
        }

        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                if (matrix[i][j] == 1) {
                    continue;
                }

                // Four Squares
                dp[i][j][0] = Math.min(dp[i - 1][j - 1][0], Math.min(dp[i - 1][j - 1][1], dp[i - 1][j - 1][2])) + 1;
                // Three Squares (exclude top-left)
                dp[i][j][1] = Math.min(dp[i - 1][j][0], dp[i][j - 1][0]);
                dp[i][j][2] = Math.min(dp[i][j - 1][0], dp[i - 1][j - 1][0]);
            }
        }

        int maxSquare = 0;
        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                for (int k = 0; k < 3; k++) {
                    maxSquare = Math.max(maxSquare, dp[i][j][k]);
                }
            }
        }

        System.out.println(maxSquare * maxSquare);
    }
}
