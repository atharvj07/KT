#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
const int MAXN = 1e9 + 69;
const int MAXM = 1e5 + 69;

int main() {
    ll n, m;
    cin >> n >> m;
    ll sx, sy, fx, fy;
    cin >> sx >> sy >> fx >> fy;
    vector<pi> loc(m);
    for (int i = 0; i < m; i++) {
        cin >> loc[i].first >> loc[i].second;
    }
    // Time to reach home without instant movements
    ll base = abs(sx - fx) + abs(sy - fy);
    // If instant movements are impossible
    if (sx != fx && sy != fy) {
        cout << base << endl;
        return 0;
    }
    // Sorting instant movement locations
    if (sx == fx) {
        sort(loc.begin(), loc.end(), [](const pi &a, const pi &b) -> bool {
            return a.second < b.second;
        });
    } else {
        sort(loc.begin(), loc.end(), [](const pi &a, const pi &b) -> bool {
            return a.first < b.first;
        });
    }
    ll ans = base;
    // Considering vertical instant movements separately
    if (sx == fx) {
        ll x = loc[0].second;
        for (int i = 1; i < m; i++) {
            x += abs(loc[i].second - loc[i - 1].second);
        }
        ans = min(ans, x + sy + fy);
    }
    // Considering horizontal instant movements separately
    if (sy == fy) {
        ll x = loc[0].first;
        for (int i = 1; i < m; i++) {
            x += abs(loc[i].first - loc[i - 1].first);
        }
        ans = min(ans, x + sx + fx);
    }
    // Considering a combination of vertical and horizontal instant movements
    for (int i = 0; i < m; i++) {
        ll x, y;
        if (sx == fx) {
            x = sx;
            y = loc[i].second;
        } else {
            x = loc[i].first;
            y = sy;
        }
        for (int j = i + 1; j < m; j++) {
            if (sx == fx) {
                y = loc[j].second;
            } else {
                x = loc[j].first;
            }
            ans = min(ans, abs(x - sx) + abs(y - fy) + abs(x - fx) + abs(y - sy));
        }
    }
    cout << ans << endl;
    return 0;
}
