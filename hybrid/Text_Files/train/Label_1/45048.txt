import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = 0;
        int cur = -1;
        int prev = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] == -1) {
                continue;
            }
            if (cur == -1) {
                cur = a[i];
            } else if (prev == -1) {
                prev = a[i];
            } else {
                if (prev + cur != 2 * a[i]) {
                    ans++;
                    cur = prev;
                    prev = a[i];
                } else {
                    prev = a[i];
                }
            }
        }
        if (cur != -1) {
            ans++;
        }
        System.out.println(ans);
    }
}
