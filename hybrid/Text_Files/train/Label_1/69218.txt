import os

for _ in range(int(input())):
    s = input()
    x = int(input())
    if len(s)%2 != x%2:
        print(-1)
        continue

    n = len(s)
    ans = ['0'] * n
    for i in range(n):
        if s[i] == '0':
            continue
        if i-x >= 0 and i+x < n:
            if ans[i-x] == '1' or ans[i+x] == '1':
                print(-1)
                break
        elif i-x >= 0:
            if ans[i-x] == '1':
                print(-1)
                break
        elif i+x < n:
            if ans[i+x] == '1':
                print(-1)
                break
        ans[i] = '1'
    else:
        print(''.join(ans))
