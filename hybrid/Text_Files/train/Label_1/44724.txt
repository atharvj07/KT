import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[][] teams = new int[n][2];
        for (int i = 0; i < n; i++) {
            teams[i][0] = sc.nextInt();
            teams[i][1] = sc.nextInt();
        }
        Arrays.sort(teams, (a, b) -> a[0] != b[0] ? b[0] - a[0] : a[1] - b[1]);
        int curPlace = 1, curCnt = 1, cnt = 0;
        for (int i = 1; i < n; i++) {
            if (teams[i][0] != teams[i - 1][0] || teams[i][1] != teams[i - 1][1]) {
                curCnt = 1;
                curPlace = i + 1;
            } else {
                curCnt++;
            }
            if (curPlace == k) {
                cnt += curCnt;
            }
        }
        System.out.println(cnt);
    }
}
