#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> b(n / 2);
  for (int i = 0; i < n / 2; ++i) {
    cin >> b[i];
  }
  vector<long long> a(n, 0);
  for (int i = 0; i < n / 2; ++i) {
    a[i] = b[i];
    a[n - 1 - i] = b[i];
  }
  for (int i = 0; i < n; ++i) {
    cout << a[i] << " ";
  }
  cout << endl;
  return 0;
}
