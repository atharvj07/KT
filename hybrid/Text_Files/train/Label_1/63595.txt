T = int(input())
for _ in range(T):
    n, l, r = map(int, input().split())
    ans = []
    total_edges = n * (n - 1)
    for i in range(l, r + 1):
        if i == total_edges + 1:
            ans.append(1)
        else:
            i -= 1
            from_vertex = i // (n - 1) + 1
            to_vertex_index = i % (n - 1)
            if to_vertex_index >= from_vertex - 1:
                to_vertex_index += 1
            ans.append(from_vertex if i % 2 == 0 else to_vertex_index)
    print(' '.join(map(str, ans)))
