#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, d, ans = 0, cur = 0;
  cin >> n >> d;
  vector<long long> a(n);
  for (auto &x : a) cin >> x;
  for (long long i = 0; i < n; i++) {
    cur += a[i];
    if (cur < 0) cur = 0;
    if (cur > d) {
      ans = -1;
      break;
    }
    if (a[i] == 0) {
      if (cur < 0) {
        ans++;
        cur += d;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
