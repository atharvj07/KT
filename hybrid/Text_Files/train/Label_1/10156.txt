#include <cstdio>
#include <cmath>
#include <algorithm>
#include <iostream>
#include <cstring>
#include <ctime>

using namespace std;

const int maxn = 2e4 + 10;

struct query {
  int t;
  int id;
} q[maxn];

int main() {
  int n, m, T, id = 0;
  scanf("%d %d %d", &n, &m, &T);
  for (int i = 0; i < n; ++i) {
    char ti[10];
    scanf("%s", ti);
    int h, mm, s;
    sscanf(ti, "%d:%d:%d", &h, &mm, &s);
    q[i].t = h * 3600 + mm * 60 + s;
  }
  sort(q, q + n, [](const query &x, const query &y) { return x.t < y.t; });
  int last[maxn], mp[maxn];
  memset(last, -1, sizeof(last));
  for (int i = 0; i < n; ++i) {
    int users = 0, back = 0, front = 0;
    for (int j = i; j < n; ++j) {
      if (q[j].t - last[q[j].id] <= T) {
        if (!mp[q[j].id]) back = max(q[j].t - last[q[j].id] - 1, back);
      } else {
        users++;
        mp[q[j].id] = 1;
        last[q[j].id] = q[j].t;
        front = max(front, q[j].t + T);
      }
      if (users > m) break;
    }
    int match = 0, matched = 0, prev = -1;
    for (int j = i; j >= 0; --j) {
      if (match && mp[q[j].id] == 0 && prev != q[j].t) {
        if (matched < front - back - 1) {
          mp[q[j].id] = 1;
          last[q[j].id] = q[j].t;
          matched = max(matched, q[j].t + T);
        }
      }
      if (mp[q[j].id] && match == 0) ++match;
      prev = q[j].t;
    }
    for (int j = i; j < n; ++j) {
      if (mp[q[j].id]) {
        if (matched < front - back - 1) {
          mp[q[j].id] = 0;
          matched = max(matched, q[j].t + T);
          if (back && match < front - back - 1) {
            mp[q[j].id] = 1;
            last[q[j].id] = q[j].t + back + 1;
            matched = max(matched, last[q[j].id] + T);
          }
        }
      }
    }
    if (matched < front - back - 1) {
      cout << "No solution" << endl;
      return 0;
    }
  }
  int cnt = 0;
  for (int i = 0; i < maxn; ++i) if (mp[i]) ++cnt;
  cout << cnt << endl;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < maxn; ++j)
      if (mp[j] && last[j] == q[i].t) {
        ++id;
        cout << id << endl;
        break;
      }
  return 0;
}
