n = int(input())
l = input()
r = input()

# Dictionaries to count occurrences of each color in left and right boots.
left_counts = {}
right_counts = {}
# Lists to track indexes of '?' in left and right strings.
left_question_marks = []
right_question_marks = []
pairs = []

# Count occurrences and store indexes of '?'.
for i in range(n):
    if l[i] == '?':
        left_question_marks.append(i + 1)
    else:
        if l[i] in left_counts:
            left_counts[l[i]].append(i + 1)
        else:
            left_counts[l[i]] = [i + 1]
            
    if r[i] == '?':
        right_question_marks.append(i + 1)
    else:
        if r[i] in right_counts:
            right_counts[r[i]].append(i + 1)
        else:
            right_counts[r[i]] = [i + 1]

# Matching exact colors.
for color in left_counts:
    if color in right_counts:
        while left_counts[color] and right_counts[color]:
            pairs.append((left_counts[color].pop(), right_counts[color].pop()))

# Matching left '?' with right colors.
while left_question_marks and right_counts:
    color, indexes = right_counts.popitem()
    while indexes and left_question_marks:
        pairs.append((left_question_marks.pop(), indexes.pop()))

# Matching right '?' with left colors.
for indexes in left_counts.values():
    while indexes and right_question_marks:
        pairs.append((indexes.pop(), right_question_marks.pop()))

# Matching '?' with '?' if any left.
while left_question_marks and right_question_marks:
    pairs.append((left_question_marks.pop(), right_question_marks.pop()))

print(len(pairs))
for pair in pairs:
    print(pair[0], pair[1])
