#include <cstdio>

int main()
{
    int t;
    scanf("%d", &t);
    while (t--)
    {
        long long k, n, a, b;
        scanf("%lld %lld %lld %lld", &k, &n, &a, &b);
        if (k < n * b)
            printf("-1\n");
        else
            printf("%lld\n", (k - n * b) / (a - b));
    }
    return 0;
}
