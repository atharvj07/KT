#include <bits/stdc++.h>
using namespace std;

long long broken(int n, int m, int k, vector<long long> &v)
{
    long long ans = 0, prv = 1;
    for (int i = 0; i <= k && i < n; i++)
    {
        if (i == k)
        {
            ans += v[n - 1] - prv + 1;
            break;
        }
        if (v[i] > m)
        {
            ans += m - prv + 1;
            prv = m + 1;
            continue;
        }
        ans += v[i] - prv + 1;
        prv = v[i] + 1;
    }
    return ans;
}
int main()
{
    int n, m, k;
    cin >> n >> m >> k;
    vector<long long> v(n);
    for (int i = 0; i < n; i++)
        cin >> v[i];
    cout << broken(n, m, k, v);
    return 0;
}
