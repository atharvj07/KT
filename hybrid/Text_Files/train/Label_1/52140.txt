import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String a = sc.nextLine();
        String b = sc.nextLine();
        int[] freq = new int[26];
        for (char c : a.toCharArray()) {
            freq[c - 'a']++;
        }
        int res = 0;
        for (char c : b.toCharArray()) {
            if (freq[c - 'a'] > 0) {
                res++;
            }
        }
        System.out.println(res);
    }
}
