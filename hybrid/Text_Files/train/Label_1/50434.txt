import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        if ((n + m - 1) % m != 0) {
            System.out.println(-1);
        } else {
            System.out.println((n + m - 1) / m);
        }
    }
}
