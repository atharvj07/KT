n, q = map(int, input().split())
a = [0] * n
for i in range(q):
    t, x = map(int, input().split())
    if t == 1:
        a[x - 1] += 1
    elif t == 2:
        a[x - 1] = 0
    else:
        for j in range(n):
            if x <= 0:
                break
            a[j] -= 1
            x -= 1
    print(sum(a))
