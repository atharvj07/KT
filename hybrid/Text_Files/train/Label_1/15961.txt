#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, p, k;
    cin >> n >> p >> k;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    while (p >= a[ans] && ans < n) {
      p -= a[ans];
      ++ans;
      if (ans < n && p >= a[ans]) {
        for (int j = ans + 1; j < n; ++j) {
          if (a[j] > a[ans]) {
            break;
          }
          p -= a[j];
          ++ans;
          if (ans == n || p < a[ans]) {
            break;
          }
        }
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
