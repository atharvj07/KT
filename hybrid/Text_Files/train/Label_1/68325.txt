from collections import Counter, deque
n=int(input())
a=deque(map(int,input().split()))
lc=[0]*n
rc=[0]*n
cnt=Counter(a)
lc[0]=(a[0]>a[-1])
for i in range(1,n):
    lc[i]=lc[i-1]
    if a[i]>a[i-1]:lc[i]+=1
rc[-1]=(a[-1]>a[0])
for i in range(n-2,-1,-1):
    rc[i]=rc[i+1]
    if a[i]>a[i+1]:rc[i]+=1
m=0
for i in range(n):m=max(m,lc[i]+rc[i])
s=''
l,r=0,n-1
for i in range(m):
    if lc[l]+rc[r]>lc[r]+rc[l]:
        s+='L'
        if lc[l]>0:lc[l]-=1
        else:l+=1
    else:
        s+='R'
        if rc[r]>0:rc[r]-=1
        else:r-=1
print(m)
print(s)
