import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class JaberSuperhero {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt() - 1;
            }
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;

            int minTime = bfs(grid, r1, c1, r2, c2);
            System.out.println(minTime);
        }
    }

    private static int bfs(int[][] grid, int r1, int c1, int r2, int c2) {
        Queue<int[]> queue = new LinkedList<>();
        boolean[][] visited = new boolean[grid.length][grid[0].length];

        queue.offer(new int[]{r1, c1, 0});
        visited[r1][c1] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int row = current[0];
            int col = current[1];
            int time = current[2];

            if (row == r2 && col == c2) {
                return time;
            }

            // Check adjacent cells
            for (int i = -1; i <= 1; i++) {
                for (int j = -1; j <= 1; j++) {
                    int newRow = row + i;
                    int newCol = col + j;

                    if (newRow >= 0 && newRow < grid.length && newCol >= 0 && newCol < grid[0].length && !visited[newRow][newCol] && (grid[newRow][newCol] == grid[row][col] || Math.abs(i) + Math.abs(j) == 1)) {
                        queue.offer(new int[]{newRow, newCol, time + 1});
                        visited[newRow][newCol] = true;
                    }
                }
            }
        }

        return -1;
    }
}
