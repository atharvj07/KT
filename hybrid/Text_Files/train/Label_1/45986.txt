import java.util.*;

class Main {
    public static void main(String[] agrs) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] min = new long[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                min[i] = Math.max(min[i], sc.nextLong());
            }
        }
        long max = 0;
        int ans1 = 0, ans2 = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long cur = min[i] + min[j];
                if (cur > max) {
                    max = cur;
                    ans1 = i + 1;
                    ans2 = j + 1;
                }
            }
        }
        System.out.println(ans1 + " " + ans2);
    }
}
