n, k = map(int, input().split())
s = input()
l = [0] * n
r = [0] * n
if s[0] == "a":
    l[0] = 1
if s[n - 1] == "a":
    r[n - 1] = 1
for i in range(1, n):
    if s[i] == "a":
        l[i] = l[i - 1] + 1
for i in range(n - 2, -1, -1):
    if s[i] == "a":
        r[i] = r[i + 1] + 1
ans = 0
for i in range(n):
    if s[i] == "a":
        continue
    if i == 0 or i == n - 1:
        ans = max(ans, min(k, l[i] + r[i] - 1))
    else:
        ans = max(ans, min(k, l[i - 1] + r[i] + 1))
print(n if ans == k else ans + 2)
