
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] q = new int[n];
        for (int i = 0; i < n; i++) {
            q[i] = scanner.nextInt();
        }
        long ans = 0;
        int[] cnt = new int[1000001];
        for (int i = 0; i < n; i++) {
            int p = q[i];
            while (p % 2 == 0) {
                p /= 2;
                cnt[2]++;
            }
            for (int j = 3; j * j <= p; j += 2) {
                while (p % j == 0) {
                    p /= j;
                    cnt[j]++;
                }
            }
            if (p > 1) {
                cnt[p]++;
            }
        }
        for (int i = 2; i <= 1000000; i++) {
            if (cnt[i] > 0) {
                ans += (long) Math.pow(i, cnt[i]);
            }
        }
        System.out.println(ans % (long) (1e9 + 7));
    }
}
