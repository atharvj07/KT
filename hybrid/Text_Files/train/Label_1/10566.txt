#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    int x[n], y[n];
    set<pair<int, int>> s;
    for (int i = 0; i < n; ++i) {
        cin >> x[i] >> y[i];
        s.insert({x[i], y[i]});
    }
    if (s.size() == 4) {
        int mx = *max_element(x, x + n), mn = *min_element(x, x + n), my = *max_element(y, y + n), my2 = *min_element(y, y + n);
        cout << (mx - mn) * (my - my2) << endl;
        return 0;
    }
    cout << -1 << endl;
}
