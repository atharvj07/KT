import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long a = in.nextLong();
        long b = in.nextLong();
        long c = in.nextLong();
        in.close();
        long ans = Math.min(n / a, n / b);
        long rem = n % b;
        if (rem >= c) {
            ans += rem / (b - c);
        }
        System.out.println(ans);
    }
}
