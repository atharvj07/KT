_ = input()
h = list(map(int, input().split()))
dp = [[[0 for _ in range(11)] for _ in range(2001)] for _ in range(2)]
dp[0][len(h)][h[len(h) - 1]] = 1
for i in range(len(h) - 2, -1, -1):
    for j in range(2001):
        for last in range(11):
            dp[0][j][last] = (dp[0][j][last] + dp[1][j][last] * max(last - h[i], 0)) % 998244353
            dp[1][j][h[i]] = (dp[1][j][h[i]] + dp[0][j][last] * max(h[i] - last, 0)) % 998244353
            dp[1][j + 1][last] = (dp[1][j + 1][last] + dp[1][j][last]) % 998244353
print(dp[0][0][0])
