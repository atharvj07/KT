first_layout = input()
second_layout = input()
text = input()

layout_map = {first_layout[i]: second_layout[i] for i in range(26)}
layout_map_upper = {first_layout[i].upper(): second_layout[i].upper() for i in range(26)}

translated_text = ""

for char in text:
    if char.islower():
        translated_text += layout_map.get(char, char)
    elif char.isupper():
        translated_text += layout_map_upper.get(char, char)
    else:
        translated_text += char

print(translated_text)
