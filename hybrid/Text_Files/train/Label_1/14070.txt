#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k, m;
  cin >> n >> k >> m;
  vector<int> t(k);
  for (int i = 0; i < k; ++i) cin >> t[i];
  sort(t.begin(), t.end());
  long long ans = 0;
  for (int i = 0; i < k && m >= t[i]; ++i) {
    m -= t[i];
    ans++;
  }
  if (m >= 0) ans += m / k + (m % k > 0);
  cout << ans << endl;
  return 0;
}
