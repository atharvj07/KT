#include <iostream>
using namespace std;
int main() {
  int k;
  cin >> k;
  if (k == 2)
    cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    cout << k + 1 << " " << k * (k + 1) / 2 << endl;
    for (int i = 1; i <= k; i++) {
      for (int j = 1; j <= k; j++) {
        if (i == j)
          continue;
        cout << i << " " << j << endl;
      }
      cout << i << " " << k + 1 << endl;
    }
  }
  return 0;
}
