from collections import deque
import sys

def solve():
    t = int(input())
    for _ in range(t):
        n, m = map(int, input().split())
        a = list(map(int, input().split()))
        x = list(map(int, input().split()))
        sums = [0] * n
        sams = [0] * n
        sums[0] = a[0]
        sams[0] = a[0]
        for i in range(1, n):
            sums[i] = sums[i - 1] + a[i]
            sams[i] = sams[i - 1] + max(a[i], 0)
        for qi in x:
            if qi > sams[-1]:
                print(-1)
            else:
                idx = 0
                low = 0
                high = n - 1
                while low <= high:
                    mid = (low + high) // 2
                    if sums[mid] >= qi:
                        idx = mid
                        high = mid - 1
                    else:
                        low = mid + 1
                if idx == 0:
                    ans = sys.maxsize
                else:
                    ans = idx + (qi - sams[idx]) // a[idx]
                if ans > n:
                    print(-1)
                else:
                    print(ans)

solve()
