import java.util.Scanner;

public class Main {
    static int N, P, Q;
    static int[] C;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        P = sc.nextInt();
        Q = sc.nextInt();
        C = new int[N];
        for (int i = 0; i < N; i++) {
            C[i] = sc.nextInt();
        }
        sc.close();

        // dp[i][j] = the maximum happiness at day i with self-catering power j
        int[][] dp = new int[N + 1][2];
        dp[0][0] = 0;
        dp[0][1] = -P;
        for (int i = 1; i <= N; i++) {
            // If go to the cafeteria
            dp[i][0] = dp[i - 1][0] + C[i - 1];
            // If cook for himself
            int happiness = Q * P;
            dp[i][1] = Math.max(dp[i - 1][1], dp[i - 1][0]) + happiness;
            // Update Q
            Q = Q > 0 ? 1 : -1;
        }
        System.out.println(Math.max(dp[N][0], dp[N][1]));
    }
}
