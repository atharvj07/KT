import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        boolean isPrime = true;
        for (int i = 2; i <= m / 2; ++i) {
            if (m % i == 0) {
                isPrime = false;
                break;
            }
        }
        if (isPrime && m == n + 1) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
