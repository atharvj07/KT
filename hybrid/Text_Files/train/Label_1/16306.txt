#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (auto &&x : a) { std::cin >> x; }
    int ans = -1;
    for (int i = 0; i < n; i++) {
      if (a[i] % 2 == 0) {
        ans = 1;
        std::cout << 1 << '\n' << i + 1 << '\n';
        break;
      }
    }
    if (ans == -1) {
      for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
          if ((a[i] + a[j]) % 2 == 0) {
            ans = 2;
            std::cout << 2 << '\n' << i + 1 << ' ' << j + 1 << '\n';
            break;
          }
        }
        if (ans != -1) { break; }
      }
    }
    if (ans == -1) { std::cout << -1 << '\n'; }
  }
  return 0;
}
