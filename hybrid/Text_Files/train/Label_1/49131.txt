import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        boolean isGeometric = isGeometric(arr);
        if (isGeometric) {
            System.out.println(0);
        } else if (canBeGeometric(arr)) {
            System.out.println(1);
        } else {
            System.out.println(2);
        }
    }

    private static boolean isGeometric(int[] arr) {
        if (arr.length <= 2) {
            return true;
        }
        double commonRatio = (double) arr[1] / arr[0];
        for (int i = 2; i < arr.length; i++) {
            if ((double) arr[i] / arr[i - 1] != commonRatio) {
                return false;
            }
        }
        return true;
    }

    private static boolean canBeGeometric(int[] arr) {
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            int[] newArr = new int[n - 1];
            for (int j = 0; j < i; j++) {
                newArr[j] = arr[j];
            }
            for (int j = i + 1; j < n; j++) {
                newArr[j - 1] = arr[j];
            }
            if (isGeometric(newArr)) {
                return true;
            }
        }
        return false;
    }
}
