n,k=map(int,input().split())
s,ans=input(),-1
if k<=n:
    x=[0]*26
    for i in s:
        x[ord(i)-97]+=1
    ans=0
    for i in range(26):
        if x[i]>0:
            if x[i]>k:
                ans=-1
                break
            else:
                k-=x[i]
                ans+=x[i]*(n-x[i])
            if k<=0:
                break
print(ans)
