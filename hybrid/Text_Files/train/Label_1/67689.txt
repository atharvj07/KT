n, m = list(map(int,input().split()))
a = list(map(int,input().split()))
b = list(map(int,input().split()))
a.sort()
b.sort()
ans = 0
for x in a:
    l, r = 0, len(b)-1
    while l<=r:
        mid = (l+r)>>1
        if (x+b[mid])&1:
            ans+=1
            l = mid+1
        else:
            r = mid-1
print(ans)
