MOD = 998244353

def count_blocks(n):
    # Initialize the result list with zeros
    result = [0] * n
    
    # Calculate the count for each block length
    for i in range(1, n + 1):
        if i == 1:
            # For block length 1, the count is 10 times the number of positions
            result[i - 1] = 10
        else:
            # For block length i, calculate the count using the formula
            result[i - 1] = (10 * 9 * pow(10, n - i, MOD) * (n - i + 1)) % MOD
    
    # Adjust counts for overlap and incorrect inclusion
    for i in range(n - 1):
        result[i] = (result[i] - result[i + 1]) % MOD
    
    return result

n = int(input().strip())
print(*count_blocks(n))
