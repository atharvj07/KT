#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll n, w;
    cin >> n >> w;
    vector<ll> arr(2 * n);
    for (auto &&i : arr) cin >> i;
    sort(arr.begin(), arr.end());
    reverse(arr.begin(), arr.end());
    ll sum = 0;
    for (ll i = 0, j = n; i < n; i++, j++) {
        sum += arr[i];
        j < 2 * n ? sum += arr[j] / 2 : 0;
    }
    cout << fixed << setprecision(6) << (w < sum ? w : sum) << endl;
}
