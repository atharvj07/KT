import sys

def minbits(x):
  ans = 0
  while x:
    ans += (x & 1)
    x //= 2
  return ans

inputs = sys.stdin.readlines()
n, k = map(int, inputs[0].split())
ip = []
for i in range(1, n+1):
  ip.append(list(map(int, inputs[i].split('.')[:4])))
mx = [0 for i in range(4)]
for x in ip:
  for j in range(4):
    mx[j] = max(mx[j], x[j])
mn = [255 for i in range(4)]
for x in ip:
  for j in range(4):
    mn[j] = min(mn[j], x[j])

ok = False
ans = []
for a in range(0, 1<<4):
  net = [0 for i in range(4)]
  for j in range(4):
    if a & (1<<j):
      net[j] = mx[j]
    else:
      net[j] = mn[j]
  msk = [0 for i in range(4)]
  for j in range(4):
    if j == 0:
      msk[j] = net[j]
    else:
      msk[j] = net[j-1] - net[j]
  if minbits(msk[0]) <= k:
    ok = True
    ans = msk
    break

print(*ans, sep='.') if ok else print(-1)
