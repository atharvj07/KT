import collections, sys

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    arr = list(map(int, input().split()))
    num_freq = [0] * (n + 1)
    for i in arr:
        num_freq[i] += 1
    queries = int(input())
    for _ in range(queries):
        x, k = map(int, input().split())
        print(num_freq[arr[x - 1]] if k == 0 else num_freq[num_freq[arr[x - 1]] if k == 1 else arr[x - 1]])
