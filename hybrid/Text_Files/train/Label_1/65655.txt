import sys
n, x = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
mx = [0] * n
mn = [0] * n
s = 0
for i in range(n):
    s += a[i]
    mn[i] = s
    if s < 0:
        s = 0
mx[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
    mx[i] = max(a[i], mx[i + 1] + a[i])
ans = 0
for i in range(n):
    m = mx[i]
    if i > 0:
        m = max(m, s + mn[i - 1] + mx[i])
    ans = max(ans, m * x)
print(ans)
