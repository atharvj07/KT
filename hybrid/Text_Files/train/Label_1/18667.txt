#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  int s[n], d[n];
  for (int i = 0; i < n; ++i) {
    cin >> s[i] >> d[i];
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    int x = s[i];
    while (x <= ans) x += d[i];
    ans = x;
  }
  cout << ans << endl;
  return 0;
}
