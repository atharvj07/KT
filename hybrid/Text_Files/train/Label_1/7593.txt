#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,ans=-1;
        cin>>n;
        vector<int> a(n);
        for(int& x:a) cin>>x;
        sort(a.begin(),a.end());
        if(a[1]==a[0]){
            for(int i=2;i<n;i++){
                if(a[i]-a[0]%2==0){
                    ans=max(ans,a[i]-a[0]);
                }
                else{
                    break;
                }
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
