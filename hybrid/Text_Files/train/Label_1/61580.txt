n = int(input())
arr = list(map(int, input().split()))
freq = {}
for i in arr:
    if i in freq:
        freq[i] += 1
    else:
        freq[i] = 1
max_freq = max(freq.values())
winner = -1
for i in freq:
    if freq[i] == max_freq:
        if winner == -1:
            winner = i
        else:
            if arr.index(i) < arr.index(winner):
                winner = i
print(winner)
