import sys

a, n, m = map(int, input().split())
seg = [(0, 0)] * n
for i in range(n):
    l, r = map(int, input().split())
    seg[i] = (l, r)
umbrellas = [(0, 0)] * m
for i in range(m):
    x, p = map(int, input().split())
    umbrellas[i] = (x, p)
dp = {}

def f(pos, mask):
    if (pos, mask) in dp:
         return dp[(pos, mask)]
    if pos == a:
        dp[(pos, mask)] = 0
        return 0
    ans = 1e9
    for j in range(m):
        if mask & (1 << j) == 0:
            x, w = umbrellas[j]
            if x > pos:
                ans = min(ans, f(pos, mask | (1 << j)) + w)
    for i in range(n):
        l, r = seg[i]
        if l <= pos <= r:
            ans = min(ans, f(r + 1, mask) + 1)
    dp[(pos, mask)] = ans
    return ans

print(-1 if f(0, 0) == 1e9 else f(0, 0))
