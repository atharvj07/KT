def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def find_prime_distance(n):
    for i in range(n, 1, -1):
        if is_prime(i):
            return i
    return 2

def solve(a):
    n = len(a)
    operations = []
    for i in range(n):
        min_pos = i
        for j in range(i+1, n):
            if a[j] < a[min_pos]:
                min_pos = j
        if min_pos != i:
            dist = find_prime_distance(min_pos - i + 1)
            while min_pos - i + 1 != dist:
                if min_pos - dist + 1 >= i:
                    operations.append((min_pos - dist + 1 + 1, min_pos + 1))
                    a[min_pos - dist + 1], a[min_pos] = a[min_pos], a[min_pos - dist + 1]
                    min_pos -= dist - 1
                else:
                    break
            if min_pos != i:
                operations.append((i + 1, min_pos + 1))
                a[i], a[min_pos] = a[min_pos], a[i]
    return operations

n = int(input().strip())
a = list(map(int, input().strip().split()))

operations = solve(a)
print(len(operations))
for op in operations:
    print(*op)
