#include <bits/stdc++.h>
using namespace std;
#define int long long
void solve() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<vector<int>> g(n + 1);
    vector<int> deg(n + 1);
    while (m--) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
        deg[u]++, deg[v]++;
    }
    vector<int> cycle;
    function<void(int, int)> dfs = [&](int u, int p) {
        cycle.push_back(u);
        for (auto v : g[u]) {
            if (v == p) continue;
            if (cycle.size() <= k)
                dfs(v, u);
            else
                break;
        }
    };
    for (int i = 1; i <= n; i++) {
        if (deg[i] >= k) {
            dfs(i, i);
            break;
        }
    }
    cout << cycle.size() << '\n';
    for (auto v : cycle) cout << v << ' ';
    cout << '\n';
}
int32_t main() {
    solve();
}
