import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            int x = input.nextInt();
            int y = input.nextInt();
            points[i] = new Point(x, y);
        }

        List<Point> hull = grahamScan(points);
        System.out.println(hull.size());
        for (Point point : hull) {
            System.out.println(point.x + " " + point.y);
        }
    }

    private static List<Point> grahamScan(Point[] points) {
        int n = points.length;
        if (n < 3) {
            return new ArrayList<>(Arrays.asList(points));
        }

        // Sort the points by their y-coordinates, in ascending order
        Arrays.sort(points, (a, b) -> a.y - b.y);

        // Find the point with the minimum y-coordinate
        Point minYPoint = points[0];

        // Sort the points by their x-coordinates, relative to the point with the minimum y-coordinate
        Arrays.sort(points, (a, b) -> {
            int orientation = orientation(minYPoint, a, b);
            if (orientation == 0) {
                return a.distance(minYPoint) - b.distance(minYPoint);
            } else {
                return orientation;
            }
        });

        // Create the stack that will hold the points on the convex hull
        Stack<Point> hull = new Stack<>();

        // Add the first two points to the stack
        hull.push(minYPoint);
        hull.push(points[1]);

        // Iterate over the remaining points
        for (int i = 2; i < n; i++) {
            // While the stack has at least two points and the current point is not to the left of the line defined by the last two points on the stack, pop the last point from the stack
            while (hull.size() >= 2 && orientation(hull.peek(), hull.get(hull.size() - 2), points[i]) != 1) {
                hull.pop();
            }

            // Add the current point to the stack
            hull.push(points[i]);
        }

        // Return the points on the convex hull
        return new ArrayList<>(hull);
    }

    private static int orientation(Point p, Point q, Point r) {
        int val = (q.y - p.y) * (r.x - q.x) - (q.x - p.x) * (r.y - q.y);
        if (val == 0) {
            return 0; // colinear
        } else if (val > 0) {
            return 1; // clockwise
        } else {
            return -1; // counter-clockwise
        }
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public int distance(Point other) {
            return (x - other.x) * (x - other.x) + (y - other.y) * (y - other.y);
        }
    }
}
