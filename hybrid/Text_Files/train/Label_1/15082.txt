#include <bits/stdc++.h>
#include <iostream>
#define ll long long
#define MOD 1000000007

using namespace std;

vector<ll> ans;

int main() {
  ll n;
  cin >> n;
  ans.resize(n + 1);
  ans[0] = 1;
  ans[1] = 1;
  ans[2] = 3;
  for (ll i = 3; i <= n; i++) { ans[i] = (ans[i - 2] * ans[i - 1] % MOD) % MOD; }
  cout << ans[n];
  return 0;
}
