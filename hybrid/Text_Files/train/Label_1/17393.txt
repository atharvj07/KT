#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define dbg(x) cout<<#x<<":"<<x<<endl;
int main(){
	int tcs;
	cin>>tcs;
	while(tcs--){
		int n, x, cnt = 0;
		cin>>n;
		vector<int> v(n);
		for(int i=0; i<n; i++){
			cin>>x;
			v[x-1]=i+1;
		}
		for(int i=0; i<n; i++){
			if(v[i]) cnt+=v[i]-1-i;
		}
		cout<<cnt<<endl;
	}
	return 0;
}
