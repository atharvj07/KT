#include<bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size() - 1, cnt = 0;
    while (n >= 0 && s[n--] == '0') cnt++;
    int f = s[n + 1] - '0';
    vector<int> g(10, 0);
    for (int i = n + 1; i >= 0; i--) {
      g[s[i] - '0'] = 1;
    }
    int ans = g[f];
    for (int i = f - 1; i >= 0; i--) {
      if (g[i] == 0) ans++;
    }
    cout << ans << "\n";
  }
  return 0;
}
