import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

dp = [[[-1] * 3 for _ in range(n + 1)] for _ in range(m + 1)]

def f(i, j, last):
    if j == m:
        return 0
    if dp[j][i][last] != -1:
        return dp[j][i][last]

    ans = f(i, j + 1, last)
    if i + 1 <= n:
        if last == 0 or last == 1:
            ans = max(ans, f(i + 1, j, 0) + a[i - 1])
        else:
            ans = max(ans, f(i + 1, j, 2) + min(a[i - 1], a[b[j] - 1]))
    dp[j][i][last] = ans
    return ans

print(f(1, 0, 0))
