#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

const int MAXN = 1000;

int n, m, k;
vector<vector<char> > grid;
vector<vector<pair<int, string> > > dp;

string shortestCycle() {
  for (int len = 1; len <= k; ++len) {
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        if (grid[i][j] == '.') {
          for (int d = 0; d < 4; ++d) {
            int x = i + "DLUR"[d] - 'D';
            int y = j + "LRUD"[d] - 'L';
            if (x >= 0 && x < n && y >= 0 && y < m && grid[x][y] != '*') {
              string path = dp[x][y].second + (char)('D' + d);
              if (len == k && x == i && y == j) {
                path.erase(unique(path.begin(), path.end()), path.end());
                return path;
              }
              if (dp[i][j].second.empty() || path < dp[i][j].second) {
                dp[i][j] = {len, path};
              }
            }
          }
        }
      }
    }
  }
  return "IMPOSSIBLE";
}

int main() {
  cin >> n >> m >> k;
  grid.assign(n, vector<char>(m));
  dp.assign(n, vector<pair<int, string> >(m));
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      cin >> grid[i][j];
    }
  }
  cout << shortestCycle() << endl;
  return 0;
}
