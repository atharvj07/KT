#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
LL vp[300005];
int main() {
  LL n, m, x, y, u, v;
  cin >> n >> m;
  for (LL i = 1; i <= n; i++) {
    cin >> x >> y;
    vp[i] = x;
    vp[i + n] = y;
  }
  vector<int> g[300005];
  for (LL i = 1; i <= m; i++) {
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  sort(vp + 1, vp + 2 * n + 1);
  LL ans[300005];
  memset(ans, 0, sizeof(ans));
  for (LL i = 1; i <= n; i++) {
    for (int j = 0; j < g[i].size(); j++) {
      if (g[i][j] >= i) {
        ans[i] += vp[i];
        ans[g[i][j]] += vp[i + n];
      }
    }
  }
  for (LL i = 1; i <= n; i++) {
    cout << ans[i] << " ";
  }
  return 0;
}
