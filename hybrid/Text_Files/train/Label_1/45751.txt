import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  public static void main(String [] args) {
    Scanner sc = new Scanner(System.in);
    int k = sc.nextInt();
    if (k % 2 == 1 || k < 3) {
      System.out.println(-1);
    } else {
      for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
          if ((i + j) % 2 == 0) {
            System.out.printf("bw");
          } else {
            System.out.printf("wb");
          }
        }
        System.out.println();
      }
      for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
          if (i == 0 || j == 0 || i == k - 1 || j == k - 1) {
            System.out.printf("bw");
          } else {
            System.out.printf("wb");
          }
        }
        System.out.println();
      }
      for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
          if ((i + j) % 2 == 0) {
            System.out.printf("ww");
          } else {
            System.out.printf("bb");
          }
        }
        System.out.println();
      }
    }
  }
}
