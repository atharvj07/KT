#include<bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define forn(i, a, b) for(int i = a; i < b; i++)
using namespace std;
typedef long long ll;

const int mxn = 1e5 + 10;
bool can[mxn];

int main() {
	string s;
	int n, q;
	cin >> n >> q;
	cin >> s;
	forn(i, 0, n) can[i] = 0;
	forn(i, 0, q) {
		int x;
		char c;
		cin >> x;
		if(x > 0) {
			cin >> c;
			can[i] = 1;
			int L = 0;
			forn(j, 0, n) {
				if(can[j] == 0) {
					if(s[j] == c) {
						can[j] = 1;
						L = j;
					}
				}
			}
			forn(j, L + 1, n) {
				if(s[j] == c) can[j] = 1;
			}
		}
		else can[i] = 0;
	}
	forn(i, 0, q) {
		if(can[i]) cout << "YES" << endl;
		else cout << "NO" << endl;
	}
	return 0;
}
