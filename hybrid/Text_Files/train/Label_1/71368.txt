MOD = 998244353

def nCr_mod(n, r, MOD):
    if n < r:
        return 0
    if r == 0 or r == n:
        return 1
    num = den = 1
    for i in range(r):
        num = (num * (n - i)) % MOD
        den = (den * (i + 1)) % MOD
    return (num * pow(den, MOD - 2, MOD)) % MOD

n, k = map(int, input().split())
events = []
for _ in range(n):
    l, r = map(int, input().split())
    events.append((l, 1))
    events.append((r + 1, -1))

events.sort()
count = 0
result = 0
for _, event in events:
    if event == 1:
        result = (result + nCr_mod(count, k - 1, MOD)) % MOD
        count += 1
    else:
        count -= 1

print(result)
