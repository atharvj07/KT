#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
typedef pair<int,int> pi;
#define pb push_back

int main() {
    int n; cin >> n;
    vector<vector<pi>> g(n+1), sz(n+1, vector<pi>());
    vector<int> deg(n+1,0);
    vector<pi> e;
    for (int i=0; i<n-1; i++) {
        int a,b; cin >> a >> b;
        g[a].pb({b,i});
        g[b].pb({a,i});
        deg[a]++; deg[b]++;
        e.pb({a,b});
    }
    int p;
    for (int i=1; i<=n; i++) if (deg[i]==1) { p=i; break; }
    if (deg[p]!=1) { cout << "NO" << endl; return 0; }
    int root=p;
    vector<int> ans(n-1,0);
    while (sz[root].size()!=n-1) {
        vector<int> cur, nex;
        for (auto [u,i]: sz[root]) {
            if (g[u].size()!=2) { cur.pb(u); continue; }
            int v1, v2;
            for (auto [v,j]: g[u]) {
                if (v==root) continue;
                if (j==i) v1=v; else v2=v;
            }
            sz[v1].pb({u,i}); sz[u].pb({v1,i});
            sz[v2].pb({u,i}); sz[u].pb({v2,i});
            ans[i]=v1; ans[e.size()]=v2;
            g[u].clear(); g[v1].clear(); g[v2].clear();
            e.pop_back();
            nex.pb(v1); nex.pb(v2);
        }
        for (auto x: cur) {
            sz[root].pb({x,sz[root].size()});
            sz[x].pb({root,sz[root].size()});
        }
        for (auto x: nex) sz[root].pb({x,sz[root].size()});
    }
    cout << "YES" << endl;
    for (int i=0; i<n-1; i++) {
        cout << e[i].first << ' ' << ans[i] << endl;
    }
}
