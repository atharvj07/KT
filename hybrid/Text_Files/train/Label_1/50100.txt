import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        int[] stack = new int[n];
        int[] queue = new int[n];
        int[] deck = new int[n];

        int stackSize = 0;
        int queueSize = 0;
        int deckSize = 0;
        int kisses = 0;

        for (int i = 0; i < n; i++) {
            int command = Integer.parseInt(br.readLine());

            if (command == 0) {
                kisses += (stackSize > 0 ? stack[stackSize - 1] : 0)
                        + (queueSize > 0 ? queue[queueSize - 1] : 0)
                        + (deckSize > 0 ? deck[deckSize - 1] : 0);

                stackSize = 0;
                queueSize = 0;
                deckSize = 0;
            } else {
                int num = command;
                if (stackSize == 0) {
                    stack[stackSize++] = num;
                    System.out.println("pushStack");
                } else if (queueSize == 0) {
                    queue[queueSize++] = num;
                    System.out.println("pushQueue");
                } else if (deckSize == 0) {
                    deck[deckSize++] = num;
                    System.out.println("pushFront");
                } else if (deck[deckSize - 1] < queue[queueSize - 1]) {
                    deck[deckSize++] = num;
                    System.out.println("pushBack");
                } else {
                    stack[stackSize++] = num;
                    System.out.println("pushStack");
                }
            }
        }

        System.out.println(kisses);
    }
}
