#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 1e6 + 5;
const int64_t P = 31;
int64_t pw[MAX_N], hsh[MAX_N];

vector<int> L[MAX_N];
int mxl[MAX_N];

int get_gcd(int a, int b) {
  return (b == 0 ? a : get_gcd(b, a % b));
}

void pre(int x) {
  pw[0] = 1;
  for (int i = 1; i < MAX_N; i++) pw[i] = pw[i - 1] * P;
  for (int y = x + 1; y < MAX_N; y++) {
    if (get_gcd(y, x) == 1) L[x].push_back(y);
  }
}

void solve() {
  int t;
  cin >> t;
  while (t--) {
    int x, p, k;
    cin >> x >> p >> k;
    if (mxl[x] < k) {
      if (L[x].empty()) pre(x);
      while (mxl[x] < k) {
        int y = L[x][mxl[x]];
        hsh[y] = hsh[x] * pw[y] + p;
        mxl[x]++;
      }
    }
    cout << L[x][k - 1] << '\n';
  }
}

int main() {
  ios_base :: sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  solve();
  return 0;
}
