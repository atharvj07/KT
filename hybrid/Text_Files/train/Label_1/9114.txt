#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long x, y, a, b;
    cin >> x >> y >> a >> b;
    if (x == 0 && y == 0) {
      cout << 0 << endl;
    } else {
      long long diff = abs(x - y);
      long long ans = (diff + a - b) / 2;
      if (diff % 2 == 1 || (diff + a - b) % 2 == 1) {
        ans++;
      }
      ans *= a;
      cout << ans << endl;
    }
  }
}
