MOD = 998244353

def count_ways(n, m, a, b):
    pos = {}
    for i, x in enumerate(a):
        if x in b:
            if x not in pos:
                pos[x] = [i, i]
            else:
                pos[x][1] = i
                
    if b[-1] not in pos:
        return 0
    
    last_pos = pos[b[-1]][1]
    for x in reversed(b):
        if x not in pos or pos[x][1] > last_pos:
            return 0
        last_pos = pos[x][0]
        
    ways = 1
    for i in range(m-1):
        ways = (ways * (pos[b[i+1]][0] - pos[b[i]][1])) % MOD
        
    return ways

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

print(count_ways(n, m, a, b))
