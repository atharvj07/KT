#include <iostream>
#include <cstdio>

using namespace std;

int a[1001][8], b[8];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= 7; ++j) {
            cin >> a[i][j];
        }
    }
    int ans = 0;
    for (int j = 1; j <= 7; ++j) {
        for (int i = 1; i <= n; ++i) {
            b[j] += a[i][j];
        }
        ans = max(ans, b[j]);
    }
    cout << ans;
    return 0;
}
