import java.util.*;

class Main {
    static int[] cave;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        cave = new int[n + 2];
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            cave[x] |= 1 << (y - 1);
            cave[y] |= 1 << (x - 1);
        }
        boolean[] vis = new boolean[n + 1];
        Queue<Integer> q = new ArrayDeque<>();
        q.add(1);
        int step = 0;
        while (!q.isEmpty()) {
            int len = q.size();
            step++;
            while (len-- > 0) {
                int u = q.poll();
                vis[u] = true;
                for (int i = 1; i <= n; i++) {
                    if (i == u) continue;
                    if ((cave[u] & (1 << (i - 1))) != 0 && !vis[i]) {
                        q.add(i);
                    }
                }
            }
        }
        if (vis[n]) {
            StringBuilder sb = new StringBuilder(String.format("%d\n1 ", step));
            int x = 1;
            vis = new boolean[n + 1];
            vis[1] = true;
            q.add(1);
            while (step-- > 0) {
                int len = q.size();
                while (len-- > 0) {
                    int u = q.poll();
                    for (int i = 1; i <= n; i++) {
                        if (i == u) continue;
                        if ((cave[u] & (1 << (i - 1))) != 0 && !vis[i]) {
                            x = i;
                            vis[x] = true;
                            q.add(x);
                            sb.append(String.format("%d ", x));
                            break;
                        }
                    }
                }
            }
            System.out.println(sb);
        } else {
            System.out.println(-1);
        }
    }
}
