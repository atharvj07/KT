#include<bits/stdc++.h>
using namespace std;
const int N = 110;
int a[N][N], b[N][N];
int main() {
	int m, n;
	cin >> m >> n;
	for (int i = 0; i < m; i++)
		for (int j = 0; j < n; j++)
			cin >> b[i][j];
	for (int i = 0; i < m; i++)
		for (int j = 0; j < n; j++) {
			a[i][j] = b[i][j];
			for (int k = 0; k < m; k++)
				if (b[k][j] == 1) a[i][j] = 1;
			for (int k = 0; k < n; k++)
				if (b[i][k] == 1) a[i][j] = 1;
			if (a[i][j] != b[i][j]) {
				puts("NO");
				return 0;
			}
		}
	puts("YES");
	for (int i = 0; i < m; i++)
		for (int j = 0; j < n; j++)
			cout << a[i][j] << ' ';
}
