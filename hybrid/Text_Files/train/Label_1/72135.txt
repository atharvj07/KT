stepans_integer = input().strip()
m = int(input())

min_remainder = m
for i in range(len(stepans_integer)):
    if stepans_integer[i] != '0':
        shifted_integer = int(stepans_integer[i:] + stepans_integer[:i])
        min_remainder = min(min_remainder, shifted_integer % m)

print(min_remainder)
