from collections import deque, defaultdict
from heapq import heappop, heappush

def dijkstra(n, a, b, d):
    adj = defaultdict(list)
    for u, v in zip(a, b):
        adj[u].append((v, d))
        adj[v].append((u, d))

    dist = [10**10]*n

    q = [(0,0)]
    while q:
        dis, u = heappop(q)
        if dis > dist[u]:
            continue

        dist[u] = dis

        for v, d in adj[u]:
            if dis + d < dist[v]:
                heappush(q, (dis+d, v))

    return dist
