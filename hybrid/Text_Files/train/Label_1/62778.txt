MOD = 1000000007

def count_ways(a, k):
    n = len(a)
    ans = 0
    p = 1
    for i in range(n):
        if a[n - 1 - i] in '05':
            ans = (ans + pow(2, i, MOD) * k) % MOD
        p = p * 2 % MOD
    ans = ans * pow((pow(2, n, MOD) - 1), k, MOD) % MOD
    return ans

a = input()
k = int(input())
print(count_ways(a, k))
