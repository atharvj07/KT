import heapq

n, k = map(int, input().split())
videos = []
for i in range(n):
    s, m = map(int, input().split())
    videos.append((s, m))

current_time = 1
servers = []
for s, m in videos:
    while servers and servers[0] <= current_time:
        heapq.heappop(servers)
    if len(servers) < k:
        heapq.heappush(servers, current_time)
    else:
        current_time = servers[0] + 1
    current_time += m
    print(current_time)
