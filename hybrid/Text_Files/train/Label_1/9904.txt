#include <bits/stdc++.h>
using namespace std;
const int N = 2e5+3;
int n;
char s[N];
int l[N], r[N], vis[N];
vector<int> adj[N];
void dfs(int u, int p) {
    l[u] = n+1;
    r[u] = 0;
    vis[u] = 1;
    for(auto v: adj[u]) {
        if(v != p) dfs(v, u);
        l[u] = min(l[u], l[v]);
        r[u] = max(r[u], r[v]);
    }
}
int32_t main() {
    cin >> n;
    cin >> s;
    iota(l, l+n+1, 0);
    for(int i = 1; i < n; ++i) {
        if(s[i] <= s[i-1]) continue;
        adj[i].push_back(i+1);
        adj[i+1].push_back(i);
    }
    int res = 1;
    for(int i = 1; i <= n; ++i) {
        if(!vis[i]) dfs(i, i);
        if(l[i] == n+1) continue;
        res = max(res, r[i] - l[i] + 1);
    }
    cout << res << '\n';
    for(int i = 1; i <= n; ++i) {
        if(l[i] == n+1) cout << 1 << ' ';
        else cout << r[i] - l[i] + 1 << ' ';
    }
}
