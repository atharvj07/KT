from fractions import Fraction as F
n = int(input())
L, R = [], []
for _ in range(n):
    l, r = map(int, input().split())
    L.append(l)
    R.append(r)
ans = 0
for l, r in zip(L, R):
    ans += (l + r) / 4
for l1, r1 in zip(L, R):
    for l2, r2 in zip(L, R):
        if l1 != l2 or r1 != r2:
            ans += (r1 - max(l1, l2) + r2 - max(l1, l2)) / 8
print(ans / F(2, n - 1))
