#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int N = 3005;
int n, m, ind, ans;
vector<int> adj[N], adjT[N];
int vis[N], id[N], low[N], timer;
stack<int> st;

void dfs(int node) {
    vis[node] = 1;
    id[node] = low[node] = ++timer;
    st.push(node);
    for (auto i : adj[node]) {
        if (!vis[i]) { dfs(i); }
        if (vis[i] == 1) { low[node] = min(low[node], low[i]); }
    }
    if (id[node] == low[node]) {
        while (true) {
            int x = st.top();
            st.pop();
            vis[x] = 2;
            if (x == node) { break; }
        }
        ++ind;
    }
}
void dfs2(int node) {
    vis[node] = 1;
    for (auto i : adjT[node]) {
        if (!vis[i]) { dfs2(i); }
    }
}
int main() {
    scanf("%d%d", &n, &m);
    while (m--) {
        int a, b;
        scanf("%d%d", &a, &b);
        adj[a].push_back(b);
        adjT[b].push_back(a);
    }
    for (int i = 1; i <= n; ++i) {
        if (!vis[i]) { dfs(i); }
    }
    memset(vis, 0, sizeof vis);
    for (int i = 1; i <= n; ++i) {
        if (!vis[i]) { dfs2(i); }
    }
    for (int i = 1; i <= n; ++i) {
        for (auto j : adj[i]) {
            if (id[i] != id[j]) { ans++; }
        }
    }
    printf("%d\n", ans / 2);
}
