#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int> v(n), t(n);
    for (int i = 0; i < n; i++) {
      cin >> v[i];
    }
    for (int i = 0; i < n; i++) {
      cin >> t[i];
    }
    vector<vector<int>> adj(n);
    for (int i = 0; i < m; i++) {
      int a, b;
      cin >> a >> b;
      a--, b--;
      adj[a].push_back(b);
      adj[b].push_back(a);
    }
    bool ok = true;
    for (int i = 0; i < n; i++) {
      ok &= (v[i] == t[i]);
    }
    if (!ok) {
      cout << "NO\n";
      continue;
    }
    int c = 0;
    for (int i = 0; i < n; i++) {
      if (v[i] != t[i]) {
        ok &= (t[i] - v[i]) % n == 0;
        c += (t[i] - v[i]) / n;
      }
    }
    if (!ok) {
      cout << "NO\n";
      continue;
    }
    for (int i = 0; i < n; i++) {
      for (int j : adj[i]) {
        if (v[i] > t[i]) {
          ok &= (v[i] - v[j] <= c);
        } else {
          ok &= (v[j] - v[i] <= c);
        }
      }
    }
    cout << (ok ? "YES\n" : "NO\n");
  }
  return 0;
}
