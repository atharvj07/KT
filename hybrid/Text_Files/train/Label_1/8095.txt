#include<bits/stdc++.h>
using namespace std;
int n,t,a[150005];
int main(){
    cin>>n>>t;
    string s;
    cin>>s;
    int nw=0,sum=0;
    for(int i=0;i<n;i++){
        int x=s[i]-'a';
        if(x==nw){
            sum+=pow(2,nw);
        }
        else{
            if(nw)a[nw]=sum;
            nw=x;
            sum=pow(2,nw);
        }
    }
    if(nw)a[nw]=sum;
    int frst=a[s[0]-'a'];
    int lst=a[s[n-1]-'a'];
    if(t==0){
        if(frst==0&&lst==0)cout<<"Yes"<<endl;
        else cout<<"No"<<endl;
    }
    else{
        if(lst==t||lst==0)cout<<"Yes"<<endl;
        else cout<<"No"<<endl;
    }
    return 0;
}
