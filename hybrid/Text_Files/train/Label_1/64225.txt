from collections import defaultdict
import sys

n, m = map(int, input().split())
adj = defaultdict(list)
for i in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

visited = [False] * (n + 1)
deg  = []
for i in adj: deg.append(len(adj[i]));
dp = [[-1]* (1<<19) for i in range(n+1)]

def solve(u, mask):
  if (mask&(1<<(u-1))): return 0
  if dp[u][mask] != -1: return dp[u][mask]

  t1 = 0 
  for v in adj[u]:
    if mask & (1<<(v-1)): continue
    t1 += 1 + solve(v, mask|(1<<(u-1))|(1<<(v-1)))
  
  t2 = solve(u, mask|(1<<(u-1)))

  dp[u][mask] = max(t1, t2)
  return dp[u][mask]

print(solve(1, 0))
