def min_index(t, s, i):
    j = 0
    while i < len(t):
        if t[i] == s[j]:
            j += 1
            if j == len(s):
                return i
        i += 1
    return -1

def count(t, s):
    n = len(s)
    f = [0 for i in range(len(t) + 1)]
    L = min_index(t, s, 0)
    if L != -1:
        f[L + 1] = 1
    for i in range(L + 1, len(t) + 1):
        L = min_index(t, s, i)
        if L != -1:
            f[L + 1] += f[i]
    return f[len(t)]
