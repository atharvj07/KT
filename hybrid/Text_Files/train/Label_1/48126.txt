import java.util.Scanner;

public class GrindProfit {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        int[] weaponAttack = new int[n];
        int[] weaponCost = new int[n];
        for (int i = 0; i < n; i++) {
            weaponAttack[i] = scanner.nextInt();
            weaponCost[i] = scanner.nextInt();
        }

        int[] armorDefense = new int[m];
        int[] armorCost = new int[m];
        for (int i = 0; i < m; i++) {
            armorDefense[i] = scanner.nextInt();
            armorCost[i] = scanner.nextInt();
        }

        int[][] monsterData = new int[p][3];
        for (int i = 0; i < p; i++) {
            monsterData[i][0] = scanner.nextInt(); // defense
            monsterData[i][1] = scanner.nextInt(); // attack
            monsterData[i][2] = scanner.nextInt(); // coins
        }

        int maxProfit = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int profit = 0;
                for (int k = 0; k < p; k++) {
                    if (weaponAttack[i] > monsterData[k][0] && armorDefense[j] > monsterData[k][1]) {
                        profit += monsterData[k][2];
                    }
                }
                profit -= weaponCost[i] + armorCost[j];
                maxProfit = Math.max(maxProfit, profit);
            }
        }

        System.out.println(maxProfit);
    }
}
