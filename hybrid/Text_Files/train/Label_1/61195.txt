import sys

n = int(input())
v = list(map(int, input().split()))
m = int(input())

s = [0] * n
s[0] = v[0]
for i in range(1, n):
    s[i] = s[i - 1] + v[i]

u = v.copy()
u.sort()

for _ in range(m):
    t, l, r = map(int, input().split())
    if t == 1:
        print(s[r - 1] - s[l - 2] if l > 1 else s[r - 1])
    else:
        print(u[r - 1] - u[l - 2] if l > 1 else u[r - 1])
