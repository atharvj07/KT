#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;

    int min_dis = INT_MAX;
    vector<int> x(n);
    for (int i = 0; i < n; i++) {
        cin >> x[i];
        if (i != 0) {
            min_dis = min(min_dis, x[i] - x[i - 1]);
        }
    }

    int ans = 0;
    if (min_dis % 2 == 0) {
        for (int i = 1; i < n; i++) {
            if ((x[i] - x[i - 1]) % min_dis != 0) {
                ans++;
            }
        }
    } else {
        for (int i = 1; i < n; i++) {
            if ((x[i] - x[i - 1]) % min_dis != min_dis / 2) {
                ans++;
            }
        }
    }

    cout << ans;
    return 0;
}
