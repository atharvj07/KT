#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long

int32_t main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    string s; cin >> s;
    int x = 0, y = 0;
    int mx = -2e9, id = 0;
    int mx2 = -2e9, id2 = 0;
    pair<int, int> first = {0, 0}, second = {0, 0};
    for (int i = 0; i < n; i++) {
      if (s[i] == 'R') x++;
      else if (s[i] == 'D') y--;
      else if (s[i] == 'L') x--;
      else y++;
      if (x == 0 && y == 0) {
        if (i - id > mx - mx2) {
          first = {id + 1, i + 1};
          mx = i;
        }
        id = i + 1;
      } else {
        if (y * mx2 > x * mx) {
          second = {id, i + 1};
          mx2 = y, id2 = id;
        }
      }
    }
    if (x == 0 && y == 0) {
      cout << first.first << " " << first.second << endl;
    } else if (id2 != 0 && mx2 != -2e9) {
      cout << second.first << " " << second.second << endl;
    } else {
      cout << -1 << endl;
    }
  }
}
