import java.util.Scanner;

public class IntegerK {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long N = scanner.nextLong();
        long count = 0;
        for (long K = 2; K * K <= N; K++) {
            if (N % K == 0) {
                count++;
                while (N % K == 0) {
                    N /= K;
                }
            }
        }
        if (N > 1) {
            count++;
        }
        System.out.println(count);
    }
}
