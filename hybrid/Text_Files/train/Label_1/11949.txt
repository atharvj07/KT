#include <bits/stdc++.h>

using namespace std;

int main(){
    int t; cin >> t;
    while(t--){
        int n; cin >> n;
        vector<int> a(n), b(n), c(n);
        for(int i = 0; i < n; i++) cin >> a[i];
        for(int i = 0; i < n; i++) cin >> b[i];
        for(int i = 0; i < n; i++) c.push_back(a[n - 1 - i]);
        bool exist = false;
        for(int i = n / 2; i >= 1; i--){
            if(a != b) reverse(a.begin(), a.begin() + i);
            if(a == b) exist = true;
        }
        if(exist && a == c) cout << "Yes\n";
        else cout << "No\n";
    }
}
