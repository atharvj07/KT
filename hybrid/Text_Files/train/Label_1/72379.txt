import sys
input = sys.stdin.readline

n, q = map(int, input().split())
servers = list(range(1, n + 1))
tasks = []
for _ in range(q):
    tasks.append(tuple(map(int, input().split())))
tasks = sorted(tasks, key=lambda task: task[0])
for t, k, d in tasks:
    free = [server for server in servers if server + d - 1 <= t]
    if len(free) < k:
        print(-1)
    else:
        ans = sum(free[:k])
        servers = [server for server in servers if server not in free[:k]]
        print(ans)
