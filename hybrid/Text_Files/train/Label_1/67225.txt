T = int(input().strip())
for _ in range(T):
    n = int(input().strip())
    segments = [list(map(int, input().split())) for _ in range(n)]
    
    # Sort segments by their left endpoint
    segments_sorted = sorted(segments, key=lambda x: x[0])
    
    # Try to find a point that can act as a divider between the two groups
    rightmost_of_left_group = segments_sorted[0][1]
    possible = True
    for i in range(1, n):
        if segments_sorted[i][0] > rightmost_of_left_group:
            break
        rightmost_of_left_group = max(rightmost_of_left_group, segments_sorted[i][1])
    else:
        possible = False
    
    if possible:
        for segment in segments:
            if segment[1] <= rightmost_of_left_group:
                print(1, end=' ')
            else:
                print(2, end=' ')
        print()
    else:
        print(-1)
