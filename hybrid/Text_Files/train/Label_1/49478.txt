import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++)
            b[i] = sc.nextInt();
        int min = Integer.MAX_VALUE;
        for (int i = 0, j = 0; i < n; i++) {
            while (j < m && b[j] < a[i])
                j++;
            min = Math.min(min, j == m ? Integer.MAX_VALUE : Math.abs(b[j] - a[i]));
        }
        System.out.println(min);
    }
}
