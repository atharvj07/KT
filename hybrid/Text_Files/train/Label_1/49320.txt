import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int w = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }

        int[] dp = new int[n];
        Arrays.fill(dp, Integer.MIN_VALUE);
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            for (int j = Math.max(0, i - w); j < i; j++) {
                dp[i] = Math.max(dp[i], dp[j] + (a[i] + (i - j) % 2 * b[i]));
            }
        }

        int max = 0;
        int idx = -1;
        for (int i = 0; i < n; i++) {
            if (dp[i] > max) {
                max = dp[i];
                idx = i;
            }
        }

        StringBuilder sb = new StringBuilder();
        while (idx >= 0) {
            sb.append(idx % 2);
            idx = idx - w;
        }

        System.out.println(sb.reverse().toString());
    }
}
