import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] fib = new int[n + 1];
        fib[0] = 0;
        fib[1] = 1;
        for (int i = 2; i <= n; i++) {
            fib[i] = fib[i - 1] + fib[i - 2];
        }
        StringBuilder name = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            if (fib[i] % 2 == 1) {
                name.append('O');
            } else {
                name.append('o');
            }
        }
        System.out.println(name);
    }
}
