import java.util.Scanner;

public class DominoTiling {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        char[][] grid = new char[h][w];
        for (int i = 0; i < h; i++) {
            grid[i] = scanner.next().toCharArray();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;
            System.out.println(countDominoes(grid, r1, c1, r2, c2));
        }
    }

    private static int countDominoes(char[][] grid, int r1, int c1, int r2, int c2) {
        int count = 0;
        for (int i = r1; i <= r2; i++) {
            for (int j = c1; j <= c2 - 1; j++) {
                if (grid[i][j] == '.' && grid[i][j + 1] == '.') {
                    count++;
                }
            }
        }
        for (int i = r1; i <= r2 - 1; i++) {
            for (int j = c1; j <= c2; j++) {
                if (grid[i][j] == '.' && grid[i + 1][j] == '.') {
                    count++;
                }
            }
        }
        return count;
    }
}
