#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    map<int, int> mp;
    bool flag = false;
    while (n--) {
      int x;
      cin >> x;
      mp[x]++;
      if (x == 2048) flag = true;
    }
    if (flag) {
      cout << "YES" << endl;
      continue;
    }
    for (auto it : mp) {
      if (it.first % 2 == 0 && mp.count(it.first * 2) != 0) {
        cout << "YES" << endl;
        flag = true;
        break;
      }
    }
    if (!flag) { cout << "NO" << endl; }
  }
  return 0;
}
