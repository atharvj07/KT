#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[4];
  for (int i = 0; i < 4; i++) cin >> a[i];
  sort(a, a + 4);
  int sm = 0;
  for (int i = 0; i < 4; i++) sm += a[i];
  if (sm % 2 == 0 && a[0] == a[1] && a[2] == a[3])
    cout << "YES"
         << "\n";
  else
    cout << "NO"
         << "\n";
  return 0;
}
