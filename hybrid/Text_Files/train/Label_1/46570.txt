import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = in.nextInt();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum * 2 >= sumOfArray(a)) {
                System.out.println(i + 1);
                break;
            }
        }
    }

    private static int sumOfArray(int[] a) {
        int sum = 0;
        for (int i : a)
            sum += i;
        return sum;
    }
}
