import sys
in_ = sys.stdin.readline
n, m = map(int, in_().split())
p = [[] for i in range(n)]
for i in range(m):
    k, *a = map(int, in_().split())
    for t in a:
        t = abs(t)
        p[t-1] += [i]
check = [0 for i in range(n)]
for s in p:
    for i in s:
        check[i] |= 1
if check.count(1) < m:
    print('NO')
else: print('YES')
