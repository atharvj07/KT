#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll MOD = 1e9 + 7;
ll power(ll x, ll y, ll p) {
    ll res = 1;
    x = x % p;
    while (y > 0) {
        if (y & 1) res = (res * x) % p;
        y = y >> 1;
        x = (x * x) % p;
    }
    return res;
}
int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; ++i) cin >> a[i];
    sort(a.begin(), a.end());
    ll ans = 0;
    for (ll i = 0; i < n; ++i) {
        vector<ll> v;
        for (ll j = i + 1; j < n; ++j) { v.push_back(a[j]); }
        ll even_sum = 0, odd_sum = 0;
        for (ll i = 0; i < v.size(); ++i) {
            if (i % 2 == 0)
                even_sum += v[i];
            else
                odd_sum += v[i];
        }
        if (even_sum == odd_sum) ans++;
    }
    cout << ans << endl;
}
