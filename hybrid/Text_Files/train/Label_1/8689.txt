#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> cnt(3, 0);
  for (char c : s) {
    cnt[c - '0']++;
  }
  sort(cnt.begin(), cnt.end());
  int need0 = cnt[1] - cnt[0], need1 = cnt[2] - cnt[0];
  for (char& c : s) {
    if (need0 > 0 && c == '1') {
      c = '0';
      need0--;
    } else if (need1 > 0 && c == '2') {
      c = '0';
      need1--;
    }
  }
  cout << s << endl;
  return 0;
}
