import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        int min = 1000000000;
        int index = -1;
        for (int i = 0; i < n; i++) {
            int days = (arr[i] - 1) / 7 * 7 + 1;
            if (days < min) {
                min = days;
                index = i + 1;
            }
        }
        System.out.println(index);
    }
}
