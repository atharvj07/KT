#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt0 = 0, cnt1 = 0;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0')
      cnt0++;
    else
      cnt1++;
    if (cnt0 == cnt1) {
      ans.push_back(cnt0 + cnt1);
      cnt0 = 0;
      cnt1 = 0;
    }
  }
  if (cnt0 != 0 || cnt1 != 0) ans.push_back(cnt0 + cnt1);
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
