#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> s(n);
  for (int i = 0; i < n; i++) cin >> s[i];

  int l = s[n - 1], r = s[n - 1] * 2 + 1;
  while (l < r) {
    int m = (l + r) / 2;
    int cnt = 0, boxes = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (cnt + s[i] <= m)
        cnt += s[i];
      else {
        cnt = s[i];
        boxes++;
      }
    }
    if (boxes > k)
      l = m + 1;
    else
      r = m;
  }
  cout << l << endl;
}
