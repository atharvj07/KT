import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        List<Integer> hh = new ArrayList<>();
        List<Integer> mm = new ArrayList<>();
        for (int i = 0; i < 24; i += x) hh.add(i);
        for (int i = 0; i < 60; i += x) mm.add(i);
        int h = sc.nextInt();
        int m = sc.nextInt();
        int ans = Integer.MAX_VALUE;
        for (int i : hh) {
            for (int j : mm) {
                if ((i < 10 ? "0" + i : "" + i).concat(j < 10 ? "0" + j : "" + j).contains("7")) {
                    ans = Math.min(ans, (h - i + 24) % 24 * 60 + (m - j + 60) % 60);
                }
            }
        }
        System.out.println(ans / x);
    }
}
