depth=0
line=[]
co=[]
for c in input().split(','):
    try: c=int(c)
    except:
        if c not in line: line.append(c)
        if len(line)>=depth+1: line[depth]=line[depth]+' '+c
        if len(co)>depth+1: co[depth+1]=co[depth+1]+1
        else: co.append(1)
    else: depth=max(depth,c)
print(depth)
for i in range(depth+1):
    print(*line[i].split(' '),sep=' ')
