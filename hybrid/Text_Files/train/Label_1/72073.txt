k = int(input())
for _ in range(k):
    n = int(input())
    planks = list(map(int, input().split()))
    count = [0] * (n + 1)
    for plank in planks:
        count[plank] += 1
    potential = 0
    for i in range(n, 0, -1):
        potential += count[i]
        if potential >= i:
            print(i)
            break
