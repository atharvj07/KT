#include <bits/stdc++.h>
using namespace std;
int main() {
    long long int n, t, k;
    cin >> n >> t >> k;
    vector<long long int> a(t);
    for (int i = 0; i < t; i++)
        cin >> a[i];
    if (t < 2 || a[t - 1] != 1 || a[t - 2] < k || n - k - 1 < a[t - 2]) {
        cout << -1 << endl;
        return 0;
    }
    vector<long long int> ans;
    ans.push_back(n);
    long long int cnt = 0;
    for (long long int i = t - 3; i >= 0; i--) {
        for (long long int j = 1; j <= a[i]; j++) {
            ans.push_back(n - cnt - j);
            ans.push_back(n - cnt - j + 1);
        }
        cnt += a[i];
    }
    cnt = 0;
    for (long long int i = t - 2; i >= 0; i--) {
        for (long long int j = 1; j <= a[i] - (i == t - 2 ? k : 0); j++) {
            ans.push_back(n - cnt - j);
            ans.push_back(ans[ans.size() - 3]);
        }
        cnt += a[i];
    }
    cout << n << endl;
    for (int i = 1; i < ans.size(); i += 2) {
        cout << ans[i - 1] << ' ' << ans[i] << endl;
    }
}
