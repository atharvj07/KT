#include <iostream>
#include <vector>

int main() {
    std::string a;
    std::cin >> a;

    std::vector<int> dp(a.size() + 1);
    dp[a.size()] = 0;

    for (int i = a.size() - 1; i >= 0; i--) {
        dp[i] = std::max(dp[i + 1], a[i] - '0');
    }

    std::string ans;
    int i = 0;
    while (i < a.size()) {
        if (dp[i] == dp[i + 1]) {
            ans += a[i];
            i++;
        } else {
            i++;
        }
    }

    std::cout << ans << std::endl;

    return 0;
}
