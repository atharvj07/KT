import sys

def median(lst):
    lst.sort()
    return lst[1]

n = int(input())
arr = [int(i) for i in input().split()]

stable = False
steps = 0

while not stable:
    stable = True

    for i in range(1, n-1):
        if arr[i] != median([arr[i-1], arr[i], arr[i+1]]):
            stable = False
            break

    if not stable:
        for i in range(1, n-1):
            arr[i] = median([arr[i-1], arr[i], arr[i+1]])

        steps += 1

if steps > 0:
    print(steps)
    print(' '.join(map(str, arr)))
else:
    print(1)
