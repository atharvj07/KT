import sys

n, m = map(int, sys.stdin.readline().split())
users = [0] * (n + 1)
leaders = []

for _ in range(m):
    line = sys.stdin.readline().split()
    if line[0] == '+':
        users[int(line[1])] = 1
    else:
        users[int(line[1])] = 0

    if sum(users):
        leaders.append(1)
    else:
        leaders.append(0)

print(sum(leaders))
for i, leader in enumerate(leaders):
    if leader:
        print(i + 1, end=' ')
