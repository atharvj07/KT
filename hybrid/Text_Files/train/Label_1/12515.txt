#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int i;
    for (i = 0; i < s.length() - 1; i++) {
        if (s[i] != 'n' && s[i + 1] != 'a' && s[i + 1] != 'o' && s[i + 1] != 'u' && s[i + 1] != 'i' && s[i + 1] != 'e') {
            cout << "NO" << endl;
            return 0;
        }
    }
    if (s[i] != 'n' && s[i] != 'a' && s[i] != 'o' && s[i] != 'u' && s[i] != 'i' && s[i] != 'e') {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    return 0;
}
