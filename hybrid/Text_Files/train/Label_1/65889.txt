n, m = map(int, input().split())
a = []
b = []
for i in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)
vis = [False] * (m + 1)
vis[0] = True
for i in range(n):
    for j in range(a[i], b[i] + 1):
        vis[j] = True
if vis[m]:
    print('YES')
else:
    print('NO')
