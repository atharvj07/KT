#include <iostream>
#include <vector>
using namespace std;
const int MOD = 7;
int n;
int monster[3];
vector<int> v;
typedef long long ll;
ll gcd(ll a, ll b){
  if(b==0) return a;
  return gcd(b,a%b);
}
int main() {
  cin >> n;
  while (n--) {
    v.clear();
    int flag=1;
    for(int i=0; i<3; i++){
      cin >> monster[i];
      v.push_back(monster[i]);
      if(monster[i]<=0) flag=0;
    }
    if(flag==0){
      cout << "NO" << endl;
      continue;
    }
    ll l=monster[0],r=monster[0];
    for(int i=1; i<3; i++){
      l=gcd(l,monster[i]);
      r=r*monster[i]/gcd(r,monster[i]);
    }
    if(l>=MOD){
      cout << "NO" << endl;
      continue;
    }
    if(r>1e18){
      cout << "NO" << endl;
      continue;
    }
    cout << "YES" << endl;
  }
  return 0;
}
