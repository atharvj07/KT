import sys

input = sys.stdin.readline

n, k = map(int, input().split())
favs = [[] for _ in range(n + 1)]
for _ in range(k):
    a, b = map(int, input().split())
    favs[a].append(b)
    favs[b].append(a)

vis = [False] * (n + 1)
ans = 0

for i in range(1, n + 1):
    if vis[i]:
        continue

    q = [i]
    vis[i] = True
    cnt = 0
    while q:
        cur = q.pop()
        cnt += 1
        for j in favs[cur]:
            if not vis[j]:
                vis[j] = True
                q.append(j)
    ans += cnt // 2

print(k - ans)
