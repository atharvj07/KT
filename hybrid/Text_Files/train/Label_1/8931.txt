#include<bits/stdc++.h>
using namespace std;
 
int main() {
  int n, s, x;
  cin >> n >> s;
  priority_queue<int, vector<int>, greater<int>> q;
 
  while (n--) {
    cin >> x;
    q.push(x);
  }
 
  while (!q.empty()) {
    x = q.top();
    q.pop();
    if (s < x) {
      cout << "NO";
      return 0;
    }
    s -= x;
    s += x / 2;
  }
  cout << "YES";
}
