#include <bits/stdc++.h>

using namespace std;

int n;
string s[105];
map<char, int> mp;
int deg[28];
bool vis[28];
char ans[28];

bool dfs(int x) {
    if (x == 26) {
        for (int i = 0; i < n; ++i) {
            string t;
            for (int j = 0; j < s[i].size(); ++j) {
                t.push_back(ans[mp[s[i][j]]]);
            }
            if (i && t <= s[i - 1]) return false;
            s[i] = t;
        }
        return true;
    }
    for (int i = 0; i < 26; ++i) {
        if (vis[i] || (x && deg[i] != x - 1)) continue;
        ans[x] = i + 'a';
        vis[i] = true;
        if (dfs(x + 1)) return true;
        vis[i] = false;
    }
    return false;
}

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> s[i];
        for (char c : s[i]) {
            mp[c]++;
        }
    }
    for (char c : s[0]) {
        for (int i = 0; i < n; ++i) {
            if (s[i].find(c) != string::npos) {
                deg[mp[c]]++;
            }
        }
    }
    if (dfs(0)) {
        for (int i = 0; i < 26; ++i) {
            cout << ans[i];
        }
    } else {
        cout << "Impossible";
    }
    return 0;
}
