import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] day = new int[n];
        int cur = 1;
        int last = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] - last >= d) {
                day[i] = cur;
                last = a[i];
            } else {
                cur++;
                day[i] = cur;
                last = a[i];
            }
        }
        System.out.println(cur);
        for (int i = 0; i < n; i++) {
            System.out.print(day[i] + " ");
        }
        System.out.println();
    }
}
