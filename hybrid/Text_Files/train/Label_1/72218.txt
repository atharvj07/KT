from sys import stdin

D = int(stdin.readline())
N, M = map(int, stdin.readline().split())

ans = -1
for i in range(D):
    x1, y1, x2, y2 = map(int, stdin.readline().split())
    x1, y1, x2, y2 = x1 - 1, y1 - 1, x2 - 1, y2 - 1
    cnt = list(map(int, stdin.readline().split()))
    cnt[0] = cnt[2] = y1
    cnt[1] = cnt[3] = N - y2 - 1
    cnt[3] += x1
    cnt[1] += M - x2 - 1
    for j in range(4):
        if cnt[j] != cnt[(j+1)%4]:
            break
    else:
        ans = i + 1

print(ans)
