#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) {
    cin >> words[i];
  }
  int min_length = n;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      int length = 0;
      bool equal = true;
      for (int k = 0; k < words[i].length(); k++) {
        if (k == 0) {
          length += 1;
        } else if (words[i][k] != words[j][k]) {
          equal = false;
          length += words[i][k] - 'a' + 1;
        }
      }
      if (equal) {
        length += words[i].length() - 1;
        min_length = min(min_length, length);
      }
    }
  }
  cout << min_length << endl;
  return 0;
}
