import sys

def fft(a):
    n = len(a)
    if n == 1:
        return
    a0 = [a[2*i] for i in range(n//2)]
    a1 = [a[2*i+1] for i in range(n//2)]
    fft(a0)
    fft(a1)
    angle = 2 * math.pi / n
    w = complex(1)
    wn = complex(math.cos(angle), math.sin(angle))
    for i in range(n//2):
        a[i] = a0[i] + w * a1[i]
        a[i + n//2] = a0[i] - w * a1[i]
        w *= wn

def ifft(a):
    n = len(a)
    if n == 1:
        return
    for i in range(n):
        a[i] = complex(a[i].real/n, a[i].imag/n)
    a.reverse()
    a[0], a[n-1] = a[n-1], a[0]
    fft(a)
    a.reverse()
    for i in range(1, n//2):
        a[i], a[n-i] = a[n-i], a[i]

def solve(n, s):
    a = [complex(int(c), 0) for c in s]
    a += [complex(0, 0)] * (2**((len(bin(len(a)-1))-2)) - len(a))
    fft(a)
    for i in range(len(a)):
        if a[i].real != 0 or a[i].imag != 0:
            return "NO"
    return "YES"
