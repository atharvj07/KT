import sys

n, t = map(int, sys.stdin.readline().split())
tasks = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]

dp = [[0 for _ in range(t+1)] for _ in range(n+1)]

for i in range(n-1,-1,-1):
    ti, qi = tasks[i]
    for j in range(t+1):
        if j+ti <= t:
            dp[i][j] = max(dp[i+1][j], dp[i+1][j+ti]+qi)
        else:
            dp[i][j] = dp[i+1][j]

print(dp[0][0])
