#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    for (ll k = 0;; k++) {
      if (to_string(1ll << k).find(to_string(a[i])) != string::npos) {
        cout << k << endl;
        break;
      }
    }
  }
  return 0;
}
