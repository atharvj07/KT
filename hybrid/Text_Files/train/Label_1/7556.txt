#include <iostream>
#include <vector>
using namespace std;

int main() {
    int m;
    cin >> m;

    if (m > 24) cout << 0;
    else {
        int k = 0;
        for (int i = 1; i <= m; i++) {
            int j = i;
            while (j % 5 == 0) {
                j /= 5;
                k++;
            }
            if (k < i) break;
        }

        cout << k << endl;
        for (int i = 1; i <= m; i++) {
            int j = i;
            while (j % 5 == 0) {
                j /= 5;
                k--;
            }
            if (k == i) cout << i << " ";
        }
    }
    return 0;
}
