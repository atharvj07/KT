def min_operations(x, y, m):
    if max(x, y) >= m:
        return 0
    if max(x, y) <= 0:
        return -1
    x, y = min(x, y), max(x, y)
    operations = 0
    if x < 0:
        operations = (-x + y - 1) // y
        x += operations * y
    while max(x, y) < m:
        x, y = y, x + y
        operations += 1
    return operations

x, y, m = map(int, input().split())
print(min_operations(x, y, m))
