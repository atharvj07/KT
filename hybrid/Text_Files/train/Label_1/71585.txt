from collections import defaultdict, deque

def bfs(graph, n):
    depth = [-1] * (n + 1)
    parent = [-1] * (n + 1)
    queue = deque([1])
    depth[1] = 0
    while queue:
        vertex = queue.popleft()
        for neighbor in graph[vertex]:
            if depth[neighbor] == -1:
                queue.append(neighbor)
                depth[neighbor] = depth[vertex] + 1
                parent[neighbor] = vertex
    return depth, parent

def solve(n, m, edges, queries):
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    depth, parent = bfs(graph, n)
    
    for query in queries:
        k = query[0]
        vertices = query[1:]
        vertices.sort(key=lambda x: depth[x], reverse=True)
        
        is_possible = True
        for i in range(k - 1):
            if depth[vertices[i]] == depth[vertices[i + 1]] + 1:
                continue
            elif parent[vertices[i]] != vertices[i + 1] and parent[vertices[i]] != parent[vertices[i + 1]]:
                is_possible = False
                break
        
        if is_possible:
            print("YES")
        else:
            print("NO")

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(n - 1)]
queries = [list(map(int, input().split())) for _ in range(m)]
solve(n, m, edges, queries)
