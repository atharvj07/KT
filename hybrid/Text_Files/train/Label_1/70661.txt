MOD = 998244353

def solve(n, k, h):
    same, diff = 0, 0
    for i in range(n):
        if h[i] == h[i-1]:
            same += 1
        else:
            diff += 1
    
    # Total combinations
    total_combinations = pow(k-1, diff, MOD) * pow(k, same, MOD) % MOD
    return total_combinations

n, k = map(int, input().split())
h = list(map(int, input().split()))

print(solve(n, k, h))
