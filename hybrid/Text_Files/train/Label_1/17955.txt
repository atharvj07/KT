#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;
const ll oo = ll(1e15);

ll n, k, m, a, g[123];
vector<ll> V[123];

int main() {
    cin >> n >> k >> m >> a;
    for (ll i = 0; i < a; i++) { cin >> g[i]; V[g[i]].push_back(i); }

    for (ll i = 1; i <= n; i++) {
        if (V[i].size() > (k - (n - i + 1)))
            cout << 1;
        else {
            if (V[i].empty())
                cout << 3;
            else {
                cout << ((V[i].size() + a >= k) ? 2 : 3);
            }
        }
    }
    cout << endl;
}
