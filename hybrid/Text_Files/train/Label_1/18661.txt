#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll a[200005];
int n, m, k;

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n, greater<ll>());
    ll l = (m + (k + 1) / 2) / (k + 1);
    ll ans = 0;
    for (int i = 0; i < l && i < n; i++) ans += a[i] * (m / (k + 1));
    for (int i = l; i < n; i++) ans += a[i];
    cout << ans << endl;
    return 0;
}
