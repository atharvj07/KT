n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
w1=[0]*(n-1)
w2=[0]*(n-1)
for i in range(n-1):
    X=i+1
    for x in a[1:]:
        X=(X+x)%n
        if X==0: X=n
        if X==1: break
    if X!=1:
        for x in b[1:]:
            X=(X+x)%n
            if X==0: X=n
            if X==1: break
    w1[i]="Win" if X==1 else "Lose" if X==0 else "Loop"
print(*w1)
