def max_significance(n, m, d, physics_cups, informatics_cups):
    physics_cups.sort(reverse=True)
    informatics_cups.sort(reverse=True)

    max_total_significance = 0
    for i in range(n+1):
        for j in range(m+1):
            if i == 0 or j == 0:
                continue
            total_width = sum([w for _, w in physics_cups[:i]]) + sum([w for _, w in informatics_cups[:j]])
            if total_width <= d:
                total_significance = sum([c for c, _ in physics_cups[:i]]) + sum([c for c, _ in informatics_cups[:j]])
                max_total_significance = max(max_total_significance, total_significance)
    return max_total_significance

n, m, d = map(int, input().split())
physics_cups = [tuple(map(int, input().split())) for _ in range(n)]
informatics_cups = [tuple(map(int, input().split())) for _ in range(m)]

print(max_significance(n, m, d, physics_cups, informatics_cups))
