import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] x = new int[n];
            int[] y = new int[n];
            int[] z = new int[n];
            for (int j = 0; j < n; j++) {
                x[j] = sc.nextInt();
                y[j] = sc.nextInt();
                z[j] = sc.nextInt();
            }
            Arrays.sort(x);
            Arrays.sort(y);
            Arrays.sort(z);
            System.out.println(x[n / 2] + " " + y[n / 2] + " " + z[n / 2]);
        }
    }
}
