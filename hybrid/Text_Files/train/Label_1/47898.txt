import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            String s = sc.nextLine();
            int n = s.length();
            int[] dp = new int[n + 1];
            dp[0] = 0;
            for (int i = 1; i <= n; i++) {
                if (s.charAt(i - 1) == '1') {
                    if (i > 1) {
                        dp[i] = dp[i - 1] + a;
                    }
                    dp[i] = Math.min(dp[i], dp[Math.max(0, i - 2)] + b + a);
                } else {
                    dp[i] = dp[i - 1];
                }
            }
            System.out.println(dp[n]);
        }
    }
}
