#include <algorithm>
#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<int> a(n);
    for (auto& el : a) std::cin >> el;
    int ans = 0;
    for (int i = 1; i < n; ++i) { ans += std::max(0, a[i - 1] - a[i] + m); }
    std::cout << ans << std::endl;
    return 0;
}
