import copy
import sys

n = int(input())
a = list(map(int, input().split()))

per = set(a)
per = sorted(per)
mx = max(a)
mn = min(a)
ans = len(per)
b = []
pos = 0
for i in per:
    cnt = 0
    for j in range(n):
        if a[j] == i:
            cnt += 1
    for j in range(cnt):
        b.append(i)
    if pos + cnt < n:
        b.append(mn if i != mx else mx)
    pos += cnt
print(ans)
print(*b)
