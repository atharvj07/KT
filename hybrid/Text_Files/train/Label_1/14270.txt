#include <bits/stdc++.h>
using namespace std;
#define ll long long
const double EPS=1e-9;

int main() {
    int n; cin >> n;
    vector<double>v(2*n);
    for (int i = 0; i < 2*n; i++) cin >> v[i];
    sort(v.begin(), v.end());
    for (int i = 0; i < n; i++) cout << fixed << setprecision(3) << v[i] + v[2*n-i-1] - floor(v[i]) - ceil(v[2*n-i-1]) << ' ';
}
