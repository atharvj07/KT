n = int(input().strip())
s = input().strip()
q = int(input().strip())

for _ in range(q):
    m, c = input().split()
    m = int(m)

    max_koyomity = 0
    count = 0
    left = 0

    for right in range(n):
        if s[right] != c:
            count += 1
        while count > m:
            if s[left] != c:
                count -= 1
            left += 1
        max_koyomity = max(max_koyomity, right - left + 1)
    
    print(max_koyomity)
