import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            p[i] = sc.nextInt();
        }
        int[] id = new int[n];
        for (int i = 0; i < n; i++) {
            id[i] = i;
        }
        int[] ans = new int[n - 1];
        int cnt = 0;
        for (int i = n - 2; i >= 1; i--) {
            if (p[i] != p[p[i]]) {
                ans[cnt++] = i;
                id[i] = id[p[i]];
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(id[i] + " ");
        }
        System.out.println();
        System.out.println(cnt);
        for (int i = 0; i < cnt; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
