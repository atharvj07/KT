#include <bits/stdc++.h>
 
using namespace std;
 
const int MAXN = 1e5 + 5;
 
int n, m;
 
vector<int> adj[MAXN];
vector<int> sign[MAXN];
bool vis[MAXN];
 
int main() {
    cin >> n >> m;
    while (m--) {
        int t, x, y;
        cin >> t;
        if (t == 1) {
            cin >> x >> y;
            adj[x].push_back(y);
        } else if (t == 2) {
            cin >> x;
            sign[x].push_back(sign[x].size() + 1);
        } else {
            cin >> x >> y;
            vis[x] = true;
            cout << (binary_search(sign[x].begin(), sign[x].end(), y) ? "YES\n" : "NO\n");
        }
    }
    return 0;
}
