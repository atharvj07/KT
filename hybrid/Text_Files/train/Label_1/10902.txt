#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int dp[2002][2002];

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  a[0][0] = '.';
  dp[0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] != 'R') {
        if (i > 0) dp[i][j] = (dp[i][j] + dp[i - 1][j]) % mod;
        if (j > 0) dp[i][j] = (dp[i][j] + dp[i][j - 1]) % mod;
      }
    }
  }
  cout << dp[n - 1][m - 1] << endl;
  return 0;
}
