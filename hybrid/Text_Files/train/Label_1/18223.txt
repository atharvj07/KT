#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    long long arr[n], sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
        sum += arr[i];
    }
    sort(arr, arr + n);
    long long ans = 0;
    for (int i = 0; i < k; i++) {
        ans += (arr[i] * arr[i]);
    }
    cout << ans << endl;
    return 0;
}
