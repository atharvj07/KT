#include <bits/stdc++.h>
using namespace std;
using vi = vector<int>;
vector<vi> me(300000);
int l[300000], c[300000], x[300000];
int main() {
  int m, n;
  cin >> m;
  for (int i = 0; i < m; i++) {
    int a, b, c;
    cin >> a;
    if (a == 1)
      cin >> b, x[i] = b, me[i] = {1, b};
    else
      cin >> b >> c, l[i] = b, ::c[i] = c, me[i] = {2, b, c};
  }
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x, x--;
    for (int j = m - 1; j >= 0; j--) {
      if (me[j][0] == 1 && x >= me[j].size() - 1)
        x -= me[j].size() - 1;
      else if (me[j][0] == 2) {
        if (x >= l[j] * ::c[j])
          x -= l[j] * ::c[j];
        else
          x = x % l[j];
      }
    }
    cout << x + 1 << ' ';
  }
  cout << endl;
  return 0;
}
