#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  set<pair<long long, long long>> s;
  for (long long i = 1; i <= n; i++) {
    long long a;
    cin >> a;
    s.insert({a, 0});
    long long y = a;
    while (y < 1e9) {
      s.insert({y, 1});
      y *= k;
    }
  }
  cout << (long long)s.size() << endl;
  return 0;
}
