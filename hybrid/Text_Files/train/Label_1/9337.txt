#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

const int MAXN = 100005;

int main() {
  char S[MAXN], T[MAXN];
  scanf("%s%s", S, T);
  int n = strlen(S), m = strlen(T);
  vector<int> f(n + 1);
  f[n] = m + 1; // initialize last element
  for (int i = n - 1; i >= 0; i--) {
    // calculate f[i] for all i
    f[i] = f[i + 1];
    if (S[i] == T[f[i] - 1]) f[i] = f[i] - 1;
  }
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (S[i] == '?' || S[i] == T[j]) j++;
    if (j == m) {
      j = f[i + 1]; // next start of match
    }
  }
  printf("%d\n", j / m);
  return 0;
}
