#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<string> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<int> p(n, 0), q(m, 0), r(n + m), head = q;
    int tot = 0;
    function<bool(int)> relax = [&](int i) {
        int j = r[i];
        if (j == -2) return false;
        if (j == -1 and ++tot != n + m) return false;
        if (j < 0) {
            j = r[i] = -j;
            for (int k = 0; k < n + m; k++) if (k != i and r[k] > 0 and v[i / n][i % m] != v[k / n][k % m]) {
                if (!relax(k)) return false;
                if (r[k] <= j) r[k] = -r[k];
            }
            if (n + m == tot) head = r;
            return true;
        }
        if (v[i / n][i % m] == v[j / n][j % m]) return r[i] != -r[j];
        if (v[i / n][i % m] == '<') return r[i] <= r[j];
        return r[i] >= r[j];
    };
    bool ok = true;
    for (int i = 0; i < n + m; i++) r[i] = -2;
    for (int i = 0; i < n + m; i++) {
        if (r[i] == -2) ok &= relax(i);
    }
    if (!ok) {
        cout << "No" << endl;
    } else {
        cout << "Yes" << endl;
        for (int i = 0; i < n; i++) cout << head[i] << ' ';
        cout << endl;
        for (int i = n; i < n + m; i++) cout << head[i] << ' ';
        cout << endl;
    }
    return 0;
}
