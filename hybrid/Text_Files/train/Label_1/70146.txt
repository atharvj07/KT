m=int(input())
h1,a1=map(int,input().split())
x1,y1=map(int,input().split())
h2,a2=map(int,input().split())
x2,y2=map(int,input().split())
X=(a1-h1)%m
Y=(a2-h2)%m
if X==Y:
    if x1*x2%m==0: print(m)
    else: print(-1)
elif (x1*y2-y1*x2)%m==0: print(-1)
else:
    a=((Y-X)*(x1*y2-y1*x2)%m)%m
    b=(x1*y2-y1*x2)%m
    print(int((a/b+1)%m))
