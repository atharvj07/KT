#include <cstdio>
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

#define ll long long
#define vi vector<int>
#define pb push_back

const int mxN = 1005;

int n, x0, y0;
int par[mxN], sz[mxN], x[mxN], y[mxN];

int findpar(int u) {
    if(u == par[u]) return u;
    return par[u] = findpar(par[u]);
}

void merge(int a, int b) {
    a = findpar(a), b = findpar(b);
    if(a != b) {
        if(sz[a] < sz[b]) swap(a, b);
        par[b] = a;
        sz[a] += sz[b];
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> x0 >> y0;

    for(int i = 0; i < n; ++i) {
        cin >> x[i] >> y[i];
        par[i] = i;
        sz[i] = 1;
    }

    for(int i = 0; i < n; ++i) {
        for(int j = i + 1; j < n; ++j) {
            if(x[i] * y[j] == x[j] * y[i]) {
                merge(i, j);
            }
        }
    }

    int ans = n;
    for(int i = 0; i < n; ++i) {
        ans = min(ans, sz[findpar(i)]);
    }

    cout << ans << "\n";
    return 0;
}
