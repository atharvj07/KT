#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (cur < a[i])
      cur = a[i] * 2, ans++;
  }
  cout << ans << endl;
}
