import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int m = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        int count = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != a[(i + k - 1) % n]) {
                count++;
            }
        }

        System.out.println(count);
    }
}
