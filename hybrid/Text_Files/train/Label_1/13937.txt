#include <bits/stdc++.h>
using namespace std;
#define ff first
#define ss second
typedef vector<int> vi;
typedef long long ll;

const int N = 2e5 + 10;
int p[N], a[N], id[N], en[N];
int timer = 0;

struct DSU {
    vi par, sz;
    DSU(int n) : par(n), sz(n, 1) { iota(par.begin(), par.end(), 0); }
    int find(int v) {
        if (par[v] == v) return v;
        return par[v] = find(par[v]);
    }
    void merge(int a, int b) {
        a = find(a), b = find(b);
        if (a != b) {
            if (sz[a] < sz[b]) swap(a, b);
            par[b] = a;
            sz[a] += sz[b];
        }
    }
};

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    for (int i = 1; i <= n; i++) cin >> p[i];
    for (int i = 1; i <= m; i++) cin >> a[i], id[a[i]] = i;
    for (int i = 1; i <= n; i++) en[p[i]] = id[p[i]];
    DSU dsu(m + 1);
    for (int i = 1; i <= m; i++) {
        if (i == 1 || a[i] != a[i - 1]) dsu.merge(id[a[i]], id[a[i - 1]]);
        if (en[a[i]]) dsu.merge(en[a[i]], id[a[i]]);
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        if (dsu.find(id[a[l]]) == dsu.find(id[a[r]]))
            cout << 1;
        else
            cout << 0;
    }
    cout << '\n';
    return 0;
}
