#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int ab = 0, ba = 0;
    for (int i = 0; i < n - 1; i++) {
      if (s[i] == 'a' && s[i + 1] == 'b') ab++;
      if (s[i] == 'b' && s[i + 1] == 'a') ba++;
    }
    while (ab != ba) {
      if (ab > ba) {
        for (int i = 0; i < n - 1; i++) {
          if (s[i] == 'a' && s[i + 1] == 'b') {
            s[i] = 'b';
            ab--;
            ba++;
            break;
          }
        }
      } else {
        for (int i = 0; i < n - 1; i++) {
          if (s[i] == 'b' && s[i + 1] == 'a') {
            s[i] = 'a';
            ab++;
            ba--;
            break;
          }
        }
      }
    }
    cout << s << endl;
  }
  return 0;
}
