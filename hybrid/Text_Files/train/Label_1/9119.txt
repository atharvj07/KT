#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        int sum_even = 0;
        for (int i = 0; i < n; i += 2) {
            sum_even += a[i];
        }
        int sum_odd = 0;
        for (int i = 1; i < n; i += 2) {
            sum_odd += a[i];
        }
        int ans = max(sum_even, sum_odd);
        for (int i = 0, j = n - 1; i < j; ++i, --j) {
            sum_even -= a[i];
            sum_even += a[j];
            sum_odd -= a[j];
            sum_odd += a[i];
            ans = max({ans, sum_even, sum_odd});
        }
        cout << ans << '\n';
    }
    return 0;
}
