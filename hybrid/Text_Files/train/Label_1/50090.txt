import java.util.Scanner;
import java.util.Set;
import java.util.HashSet;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int b1 = in.nextInt();
        int q = in.nextInt();
        int l = in.nextInt();
        int m = in.nextInt();
        if (b1 == 0 && q == 0) {
            System.out.println("0");
        } else if (q == 0) {
            System.out.println("inf");
        } else {
            Set<Integer> bads = new HashSet<>();
            for (int i = 0; i < m; i++) {
                bads.add(in.nextInt());
            }
            long c = b1, ct = 0;
            while (true) {
                if (ct % 2 == 0) {
                    if (c > l) {
                        System.out.println("inf");
                        return;
                    } else if (c < -l) {
                        System.out.println("0");
                        return;
                    }
                }
                if (bads.contains(c)) {
                    ct++;
                } else {
                    System.out.println(ct + 1);
                    return;
                }
                c *= q;
            }
        }
    }
}
