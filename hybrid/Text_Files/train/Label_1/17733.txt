#include <iostream>
#include <vector>
#include <bitset>

int main() {
  int n, m, k;
  std::cin >> n >> m >> k;
  std::vector<std::vector<int>> a(n, std::vector<int>(m));
  std::bitset<63> path;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      std::cin >> a[i][j];
    }
  }
  std::vector<std::vector<std::bitset<63>>> dp(n + 1, std::vector<std::bitset<63>>(m + 1));
  dp[n][m][a[n - 1][m - 1]] = true;
  for (int i = n - 1; i >= 0; i--) {
    for (int j = m - 1; j >= 0; j--) {
      dp[i][j][a[i][j]] = true;
      dp[i][j + 1] |= dp[i][j] << a[i][j];
      dp[i + 1][j] |= dp[i][j] << a[i][j];
    }
  }
  std::cout << dp[0][0][k] << std::endl;
  return 0;
}
