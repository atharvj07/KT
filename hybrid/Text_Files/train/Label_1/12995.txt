#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
ll dp[55][55][55][2];
int n, p;
int col[55];
int main() {
  cin >> n >> p;
  for (int i = 1; i <= n; ++i) cin >> col[i];
  dp[n + 1][n + 1][0][0] = 1;
  for (int i = n; i >= 1; --i) {
    for (int j = i + 1; j <= n + 1; ++j) {
      for (int k = 0; k <= n + 1; ++k) {
        for (int z = 0; z <= 1; ++z) {
          if (!dp[i][j][k][z]) continue;
          if (col[i] == -1) {
            (dp[i - 1][j][k][z] += dp[i][j][k][z]) %= mod;
            (dp[i - 1][j][k][!z] += dp[i][j][k][z]) %= mod;
          } else if (col[i] == 1 - z) {
            (dp[i - 1][j][k][z] += dp[i][j][k][z]) %= mod;
          } else {
            (dp[i - 1][i][j][1 - z] += dp[i][j][k][z]) %= mod;
          }
          if (col[j] == -1) {
            (dp[i][j - 1][k][z] += dp[i][j][k][z]) %= mod;
            (dp[i][j - 1][k][!z] += dp[i][j][k][z]) %= mod;
          } else if (col[j] == 1 - z) {
            (dp[i][j - 1][k][z] += dp[i][j][k][z]) %= mod;
          } else {
            (dp[i][j - 1][i][1 - z] += dp[i][j][k][z]) %= mod;
          }
          if (col[k] == -1) {
            (dp[i][j][k - 1][z] += dp[i][j][k][z]) %= mod;
            (dp[i][j][k - 1][!z] += dp[i][j][k][z]) %= mod;
          } else if (col[k] == 1 - z) {
            (dp[i][j][k - 1][z] += dp[i][j][k][z]) %= mod;
          } else {
            (dp[i][j][k - 1][1 - z] += dp[i][j][k][z]) %= mod;
          }
        }
      }
    }
  }
  cout << dp[1][n + 1][1][p] << endl;
  return 0;
}
