#include <algorithm>
#include <iostream>
#include <utility>
#include <vector>

using namespace std;

int maxAreaRect(int n, int a, int b, vector<pair<int, int>> v) {
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      vector<pair<int, int>> c = {v[i], v[j]};
      sort(c.begin(), c.end());
      do {
        if (max(c[0].first, c[1].first) <= a && max(c[0].second, c[1].second) <= b) {
          ans = max(ans, c[0].first * c[1].first + c[0].second * c[1].second);
        }
        if (max(c[0].second, c[1].first) <= a && max(c[0].first, c[1].second) <= b) {
          ans = max(ans, c[0].second * c[1].first + c[0].first * c[1].second);
        }
      } while (next_permutation(c.begin(), c.end()));
    }
  }
  return ans;
}

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  cout << maxAreaRect(n, a, b, v) << endl;
  return 0;
}
