n=int(input());a=list(map(int,input().split()));b=list(map(int,input().split()));v=[False]*n;f=0
for i in range(n):
    for j in range(i+1,n):
        if a.index(b[i])>a.index(b[j]) and b.index(b[i])<b.index(b[j]):
            v[a.index(b[i])]=True;v[a.index(b[j])]=True
for i in range(n):
    if v[i]:f+=1
print(f)
