#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    bool ok = true;
    for (int i = 0; i < n; i++) ok &= (a[i] <= d);
    if (ok)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
}
