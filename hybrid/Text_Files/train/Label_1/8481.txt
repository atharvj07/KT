#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  vector<long long> a;
  cin >> n;
  long long sum = 0;
  long long maxn = -1e9;
  for (int i = 0; i < n; ++i) {
    long long x;
    cin >> x;
    sum += abs(x);
    maxn = max(maxn, abs(x));
    a.push_back(x);
  }
  cout << (sum + maxn - 1) / 2 << endl;
  return 0;
}
