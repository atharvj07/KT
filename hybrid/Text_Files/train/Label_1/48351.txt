import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int c = Integer.parseInt(line[1]);
        int[][] d = new int[c][c];
        for (int i = 0; i < c; i++) {
            line = br.readLine().split(" ");
            for (int j = 0; j < c; j++) {
                d[i][j] = Integer.parseInt(line[j]);
            }
        }
        int[][] c0 = new int[n][n];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            for (int j = 0; j < n; j++) {
                c0[i][j] = Integer.parseInt(line[j]) - 1;
            }
        }
        int[][][] dp = new int[n][n][c];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = n - 1; j >= 0; j--) {
                for (int k = 0; k < c; k++) {
                    dp[i][j][k] = Integer.MAX_VALUE;
                    int cur = (i + j) % 3;
                    for (int l = 0; l < c; l++) {
                        int nxt = (i + j + 1) % 3;
                        if (cur == nxt) {
                            dp[i][j][k] = Math.min(dp[i][j][k], dp[i][j][l]);
                        } else {
                            dp[i][j][k] = Math.min(dp[i][j][k], dp[i][j][l] + d[l][k]);
                        }
                    }
                    if (i + 1 < n) {
                        dp[i][j][k] = Math.min(dp[i][j][k], dp[i + 1][j][c0[i][j]]);
                    }
                    if (j + 1 < n) {
                        dp[i][j][k] = Math.min(dp[i][j][k], dp[i][j + 1][c0[i][j]]);
                    }
                }
            }
        }
        System.out.println(dp[0][0][0]);
    }
}
