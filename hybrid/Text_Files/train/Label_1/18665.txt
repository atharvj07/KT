#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, m, k;
    std::cin >> n >> m >> k;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    std::reverse(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i < n && m; i++) {
        if (i % k == 0) m--;
        ans += a[i];
    }
    std::cout << ans << std::endl;
    return 0;
}
