n = int(input())
directions = input()
positions = list(map(int, input().split()))

min_time = float('inf')

for i in range(n-1):
    if directions[i] == 'R' and directions[i+1] == 'L':
        min_time = min(min_time, (positions[i+1] - positions[i]) // 2)

if min_time == float('inf'):
    print(-1)
else:
    print(min_time)
