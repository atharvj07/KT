#include<bits/stdc++.h>
using namespace std;

const int N = 2 * 1e5 + 5;
int a[N];
int32_t main() {
    int n, k; cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    nth_element(a, a + n / 2, a + n);
    int ans = a[n / 2];
    while (k > 0) {
        bool found = false;
        for (int i = n / 2; i < n; i++) {
            if (a[i] < ans) {
                ans++;
                a[i]++;
                k--;
                nth_element(a, a + n / 2, a + n);
                found = true;
                break;
            }
        }
        if (!found) break;
    }
    cout << ans << "\n";
    return 0;
}
