import math

def coprime_generator(x: int, p: int) -> int:
    i = x
    while True:
        i += 1
        if math.gcd(i, p) == 1:
            yield i

def solve(x: int, p: int, k: int) -> int:
    coprime_gen = coprime_generator(x, p)
    for _ in range(k-1):
        next(coprime_gen)
    return next(coprime_gen)

for _ in range(int(input())):
    x, p, k = map(int, input().split())
    print(solve(x, p, k))
