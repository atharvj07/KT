#include <iostream>
#include <vector>
#include <limits>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<vector<int>> graph(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            int u;
            cin >> u;
            graph[i][j] = u;
        }
    }

    int max_distance = 0;
    for (int k = 0; k < n; k++) {
        vector<int> distances(n, numeric_limits<int>::max());
        distances[k] = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n; j++) {
                if (graph[i][j] != 0 && distances[j] > distances[i] + graph[i][j]) {
                    distances[j] = distances[i] + graph[i][j];
                }
            }
        }
        for (int i = 0; i < n; i++) {
            max_distance = max(max_distance, distances[i]);
        }
    }

    cout << max_distance << endl;

    return 0;
}
