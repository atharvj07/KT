import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Stack;

class Main {
    private static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    private static Stack<Integer> stack;

    public static void main(String[] args) throws IOException {
        int tasks = Integer.parseInt(br.readLine());
        for (int i = 0; i < tasks; i++) {
            int days = Integer.parseInt(br.readLine());
            int[] prices = parseIntArray(br.readLine());
            System.out.println(countBadPrices(days, prices));
        }
    }

    private static int countBadPrices(int days, int[] prices) {
        stack = new Stack<>();
        int[] lastDayWithSmallerPrice = new int[days];
        Arrays.fill(lastDayWithSmallerPrice, -1);

        for (int i = days - 1; i >= 0; i--) {
            while (!stack.isEmpty() && prices[i] <= prices[stack.peek()]) {
                stack.pop();
            }
            if (!stack.isEmpty()) {
                lastDayWithSmallerPrice[i] = stack.peek();
            }
            stack.push(i);
        }
        int res = 0;
        for (int i = 0; i < days; i++) {
            if (lastDayWithSmallerPrice[i] != -1) {
                res++;
            }
        }
        return res;
    }

    private static int[] parseIntArray(String line) {
        return Arrays.stream(line.split(" ")).mapToInt(Integer::parseInt).toArray();
    }
}
