import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int L = scanner.nextInt();
        int N = scanner.nextInt();

        Map<Integer, Integer> weightMap = new HashMap<>();
        for (int i = 0; i < N; i++) {
            int x = scanner.nextInt();
            int w = scanner.nextInt();
            weightMap.put(x, weightMap.getOrDefault(x, 0) + w);
        }

        List<Integer> keys = new ArrayList<>(weightMap.keySet());
        Collections.sort(keys);

        int leftSum = 0;
        int rightSum = 0;
        for (int i = 0; i < keys.size() / 2; i++) {
            leftSum += keys.get(i) * weightMap.get(keys.get(i));
        }
        for (int i = keys.size() / 2; i < keys.size(); i++) {
            rightSum += keys.get(i) * weightMap.get(keys.get(i));
        }

        int diff = rightSum - leftSum;
        int x = diff > 0 ? -diff / 2 : diff / 2;
        int w = Math.abs(diff) / 2;

        System.out.println("1");
        System.out.println(x + " " + w);
    }
}
