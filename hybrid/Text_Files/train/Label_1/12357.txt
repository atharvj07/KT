#include <bits/stdc++.h>

using namespace std;

const int MOD = 1e9 + 7;

int main() {
  int n, k, d;
  cin >> n >> k >> d;
  vector<vector<int>> dp(n + 1, vector<int>(k + 1, 0));
  dp[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      if (j >= d) {
        dp[i][j] += dp[i - 1][j - d];
        dp[i][j] %= MOD;
      }
      dp[i][j] += dp[i - 1][j] * k;
      dp[i][j] %= MOD;
    }
  }
  cout << dp[n][k] << endl;
  return 0;
}
