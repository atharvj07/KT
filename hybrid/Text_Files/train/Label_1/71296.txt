from sys import stdin
read=lambda:stdin.readline().strip().split()
T=int(read()[0])
def sqr(x):
 for i in range(2,int(x**0.5)+1):
  if x%i==0:
   while x%i==0:
    x//=i
   if x==1:return True
   return False
  else:continue
 return True
for i in range(T):
 n,a,q=int(read()[0]),[int(x) for x in read()],int(read()[0])
 d=[1]*n
 for j in range(q):
  w=int(read()[0])
  if w&1:
   for j in range(n):
    a[j]=a[j]*a[j]
    if sqr(a[j]):d[j]+=1
  else:
   for j in range(n):
    a[j]=a[j]*a[j]
   for j in range(n):
    if sqr(a[j]):d[j]+=1
  print(max(d))
