#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  vector<int> p;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == '+')
      p.push_back(i);
  }
  int cnt = 0;
  for (int i = 0; i < p.size(); i++) {
    cnt += p[i] - i;
  }
  if (cnt % 2 == 0)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
  return 0;
}
