#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    unordered_map<int, int> mp;
    vector<int> taken(2 * n + 1, 0);
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        mp[a] = b;
        mp[b] = a;
    }
    ll ans = 1;
    for (int i = 1; i <= 2 * n; i++) {
        if (!taken[i] && !taken[mp[i]]) {
            taken[i] = 1;
            taken[mp[i]] = 1;
            ans = (ans * 2) % mod;
        }
    }
    cout << ans << endl;
    return 0;
}
