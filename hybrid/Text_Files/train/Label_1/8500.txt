#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7;
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &i : a) { cin >> i; }
    int ans = 0, ans1 = 0;
    for (int i : a) {
        if (i <= k) {
            ans += i / 50;
            ans1 = ans;
            if ((k - i) / 50 < ans) {
                ans = (k - i) / 50 + 1;
            }
        } else {
            ans += 2;
            ans1 = 2;
        }
    }
    if (ans1 == 0) {
        cout << -1 << endl << 0 << endl;
    } else {
        cout << ans << endl << ans1 % mod << endl;
    }
    return 0;
}
