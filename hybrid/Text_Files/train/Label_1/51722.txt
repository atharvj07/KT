import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), max = 0;
        int isize = (int) (Math.log(4e9) / Math.log(2)) + 1;
        long ibytes = sc.nextLong();
        int[] arr = new int[n];
        int[][] list = new int[isize + 1][2];
        int[] sum = new int[isize + 1];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            max = Math.max(max, arr[i]);
        }
        for (int i = 0; i <= isize; i++) {
            list[i][0] = (int) Math.pow(2, i);
            list[i][1] = (int) Math.min((int) Math.pow(2, i + 1) - 1, max);
        }
        for (int i = 0; i <= isize; i++) {
            for (int j = 0; j < n; j++) {
                if (list[i][0] <= arr[j] && arr[j] <= list[i][1]) {
                    sum[i]++;
                }
            }
        }
        Arrays.sort(sum);
        Arrays.sort(list, (a, b) -> a[1] - b[1]);
        int ans = 1 << 30, pos = -1;
        for (int i = isize; i >= 0; i--) {
            if (ibytes >= sum[i] * (list[i][1] - list[i][0] + 1)) {
                ibytes -= sum[i] * (list[i][1] - list[i][0] + 1);
                ans = list[i][1] - list[i][0] + 1;
                pos = i;
                break;
            }
        }
        System.out.println(n + ibytes / ans - sum[pos]);
    }
}
