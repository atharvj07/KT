n=int(input());a=list(map(int,input().split()));x=n//2;b=[0]*n
for i in range(1,x):b[i]=(a[i-1]+a[i]+a[i+1])%2;b[x+i]=b[x-i]
for i in range(1,x,2):b[x-i]=b[x+i]
for i in range(x-2,-1,-1):b[i]=(b[i]+b[i+1]+b[i+2])%2
for i in range(x-1,-1,-1):b[i]=b[x+i]
for i in range(1,x):b[i]=min(1,a[i]+b[i-1]+b[i+1])
for i in range(x-2,-1,-1):b[i]=min(1,a[i]+b[i+1]+b[i+2])
if b!=a:
    print(1)
else:
    print(0);print(*b)
