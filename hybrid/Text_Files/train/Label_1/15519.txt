#include <bits/stdc++.h>
using namespace std ;
typedef long long ll ;
int main() {
    int n ; scanf("%d", &n) ; ll ma = 0LL ; int a, b, c, ia = -1, ib = -1 ;
    while (n--) {
        scanf("%d %d %d", &a, &b, &c) ; ll d = min(a, min(b, c)) ;
        if (d * d * d > ma) { ma = d * d * d ; ia = a, ib = b ; }
    }
    printf("%d\n", ia == ib ? 1 : 2) ;
    printf("%d%s", min(ia, ib), ia == ib ? "" : " " + to_string(max(ia, ib))) ;
}
