import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
        cnt0 = a.count(0)
        cnt1 = a.count(1)
        cntm1 = a.count(-1)
        if cnt1 + cnt0 == len(a) and cnt0 + cntm1 == len(b):
            print("YES")
        else:
            print("NO")

solve()
