#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int k;
  cin >> k;
  int cnt = 0;
  vector<int> freq(26, 0);
  for (char c : s) freq[c - 'a']++;
  for (int i = 0; i < 26; i++) {
    if (freq[i] == 0) cnt++;
  }
  if (cnt > k) cout << "impossible";
  else
    cout << max(0, cnt - k);
}
