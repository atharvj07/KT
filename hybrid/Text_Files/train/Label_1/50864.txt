import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        long sum = 0;
        boolean odd = false;
        for (int i = 0; i < n; i++) {
            if (odd) {
                if (a[i] % 2 == 0) {
                    sum += a[i];
                }
            } else {
                if (a[i] % 2 == 1) {
                    sum += a[i];
                }
            }
            odd = !odd;
        }
        System.out.println(sum);
    }
}
