import sys
input = sys.stdin.readline

q = int(input())
for _ in range(q):
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]
    a.sort(key=lambda x: x[0])
    ans = 0
    for i in range(n - 1):
        if a[i][0] == a[i + 1][0]:
            ans += min(a[i][1], a[i + 1][1])
            a[i + 1] = [a[i + 1][0] + 1, a[i + 1][1]]
    print(ans)
