n = int(input())
a = list(map(int,input().split()))
dp = [0] * n
dp[0] = 1
for i in range(1, n):
    if a[i] > a[i-1]:
        dp[i] = dp[i-1] + 1
    else:
        dp[i] = 1
mx = 1
for i in range(n):
    if i-1 >= 0 and a[i-1] > a[i]:
        mx = max(mx, dp[i-1]+1)
    mx = max(mx, dp[i])
print(mx)
