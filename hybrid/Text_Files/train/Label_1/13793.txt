#include <cstdio>
#include <unordered_set>
using namespace std;

int main(void) {
  int n, a;
  scanf("%d%d", &n, &a);
  unordered_set<int> s;
  for (int i = 1, x; i <= n; i++) {
    scanf("%d", &x);
    if (x != a) s.insert(x);
  }
  if (s.empty())
    printf("-1");
  else
    printf("%d", *s.begin());
  return 0;
}
