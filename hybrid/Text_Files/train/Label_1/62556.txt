import heapq
INF=1<<63

n,m=map(int,input().split())
a=list(map(int,input().split()))
adj=[[] for i in range(n)]
for i in range(m):
    u,v,w=map(int,input().split());u-=1;v-=1
    adj[u].append((w,v))
    adj[v].append((w,u))

dist=[INF]*n
def dijkstra(u,x):
    hq=[(0,u)]
    dist[u]=0
    while hq:
        d,v=heapq.heappop(hq)
        if d>dist[v]:
            continue
        for w,nxt in adj[v]:
            nd=d+w
            if nd<dist[nxt]:
                dist[nxt]=nd
                heapq.heappush(hq,(nd,nxt))
    return dist[x] if dist[x]!=INF else -1

ans=[]
for i in range(n):
    ans.append(2*dijkstra(i,i)+a[i])
print(*ans)
