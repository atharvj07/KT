#include<bits/stdc++.h>
using namespace std;
using ll=long long;
int32_t main(){
    ll n,ans=0,bb=0;cin>>n;vector<pair<ll,ll>>a(n);
    for(auto&[x,type]:a)cin>>x>>type;
    for(int i=0;i<n;++i){
        if(a[i].second=='B')bb+=a[i].first;
        else{
            if(bb)ans+=min(bb,a[i].first-bb),bb=0;
            bb=a[i].first;
        }
    }
    bb=0;
    for(int i=n-1;i>=0;--i){
        if(a[i].second=='R')bb+=a[i].first;
        else{
            if(bb)ans+=min(bb,bb-a[i].first),bb=0;
            bb=a[i].first;
        }
    }
    cout<<ans<<endl;
}
