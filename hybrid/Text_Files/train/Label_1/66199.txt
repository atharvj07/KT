import sys
for l in sys.stdin:
    n, s=map(int,l.split())
    a=[]
    for i in range(n):
        x, v, t = map(int,sys.stdin.readline().split())
        a.append((x,v,t))
    a.sort(key = lambda x: x[0])
    mn=s*1e30
    for x, v, t in a:
        if t==1:
            mn=min(mn,x/v)
        else:
            mn=min(mn,(10**6-x)/v)
    print(mn)
