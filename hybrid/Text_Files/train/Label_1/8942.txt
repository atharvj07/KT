#include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>

using namespace std;

long long solve(vector<long long> &a, long long n) {
  sort(a.begin(), a.end());
  long long ans = 1e18;
  int m = a.size();
  for (int i = 0; i + n - 1 < m; ++i) {
    ans = min(ans, a[i + n - 1] - a[i]);
  }
  return ans;
}

int main() {
  long long t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    vector<long long> a(2 * n);
    for (long long &x : a) cin >> x;
    cout << solve(a, n) << endl;
  }
  return 0;
}
