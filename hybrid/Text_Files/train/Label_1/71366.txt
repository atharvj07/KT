import sys

input = sys.stdin.readline

n, k = map(int, input().split())
lamps = []
for _ in range(n):
    l, r = map(int, input().split())
    lamps.append((l, r))
lamps.sort()

if k > n:
    print(0)
    exit()

dp = [[0 for j in range(k + 1)] for i in range(n + 1)]
dp[n][0] = 1
for i in range(n - 1, -1, -1):
    l, r = lamps[i]
    for j in range(k):
        dp[i][j] = (dp[i + 1][j] + dp[i + 1][j + 1]) % 998244353
    dp[i][k] = dp[i + 1][k]

print(dp[0][k])
