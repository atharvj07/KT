#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int c = a[i] + b[j];
            bool x = false, y = false;
            for (int k = 0; k < n; k++) if (c == a[k]) x = true;
            for (int k = 0; k < m; k++) if (c == b[k]) y = true;
            if (!x && !y) {
                cout << a[i] << " " << b[j] << endl;
                return 0;
            }
        }
    }
    return 0;
}
