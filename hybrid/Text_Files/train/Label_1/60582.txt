def fillomino(n, p):
    grid = [[0] * n for _ in range(n)]
    for i in range(n):
        grid[i][i] = p[i]
    for l in range(1, n):
        for i in range(n - l):
            j = i + l
            if grid[i][j] == 0:
                for k in range(i, j + 1):
                    if grid[k][j] != 0:
                        break
                else:
                    for k in range(i, j + 1):
                        grid[k][j] = l + 1
            if grid[j][i] == 0:
                for k in range(j, i - 1, -1):
                    if grid[j][k] != 0:
                        break
                else:
                    for k in range(j, i - 1, -1):
                        grid[j][k] = l + 1
    for row in grid:
        print(*row)
    return

n = int(input())
p = list(map(int, input().split()))
fillomino(n, p)
