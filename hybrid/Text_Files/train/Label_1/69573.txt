n, m = list(map(int, input().split()))
arr = list(map(int, input().split()))
pos = [0] * n
for i in range(n):
  pos[arr[i] - 1] = i

for _ in range(m):
  l, r, x = list(map(int, input().split()))
  l, r, x = l - 1, r - 1, x - 1
  if pos[x] < l or pos[x] > r:
    print("Yes")
  elif arr[l:r + 1].count(x + 1) != 0:
    print("No")
  else:
    print("Yes")
