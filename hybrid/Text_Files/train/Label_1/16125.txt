#include <bits/stdc++.h>

#define ll long long

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 1; i < n; i++) {
    if (i % 2) {
      for (int j = 0; j < n - i; j++) { a[j] = (a[j] - a[j + 1] + (ll)1e9 + 7) % (ll)1e9 + 7; }
    } else {
      for (int j = 0; j < n - i; j++) { a[j] = (a[j] + a[j + 1]) % (ll)1e9 + 7; }
    }
  }
  cout << a[0] << endl;
  return 0;
}
