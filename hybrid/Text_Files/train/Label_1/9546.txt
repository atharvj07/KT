#include <iostream>
#include <cmath>

using namespace std;

int main() {
  long long a, b;
  cin >> a >> b;
  long long ab = a + b;
  long long ans = 2 * ab;
  for (long long p = 1; p <= sqrt(ab); p++) {
    if (ab % p == 0) {
      long long t = 2 * (p + ab / p);
      if (t < ans) ans = t;
    }
  }
  cout << ans << endl;
  return 0;
}
