#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll P = 1e9 + 9;

ll powMod(ll a, ll b) {
    ll res = 1;
    while (b > 0) {
        if (b % 2 == 1) {
            res = (res * a) % P;
        }
        a = (a * a) % P;
        b /= 2;
    }
    return res;
}

int main() {
    ll n, a, b, k;
    cin >> n >> a >> b >> k;
    string s;
    cin >> s;
    vector<ll> powA(n + 1), powB(n + 1);
    powA[0] = 1;
    for (ll i = 1; i <= n; i++) {
        powA[i] = (powA[i - 1] * a) % P;
    }
    powB[0] = 1;
    for (ll i = 1; i <= n; i++) {
        powB[i] = (powB[i - 1] * b) % P;
    }
    ll ans = 0;
    for (ll i = 0; i < k; i++) {
        if (s[i] == '-') {
            ans = (ans - powMod(powA[n - i], b - 1) * powMod(powB[i], b - 1)) % P;
        } else {
            ans = (ans + powMod(powA[n - i], b - 1) * powMod(powB[i], b - 1)) % P;
        }
    }
    ans += P;
    cout << ans % P << "\n";
    return 0;
}
