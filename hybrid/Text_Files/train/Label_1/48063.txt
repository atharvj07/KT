import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt(), n;
        for (int i = 0; i < t; i++) {
            n = sc.nextInt();
            ArrayList<Integer> chairs = new ArrayList<>();
            int min = 2, max = 4 * n;
            for (int j = 1; j <= n; j++) {
                int a = min, b = max;
                while (a != b) {
                    if (a < b) a += n;
                    else b += n;
                }
                chairs.add(a);
            }
            for (int chair : chairs) System.out.print(chair + " ");
            System.out.println();
        }
    }
}
