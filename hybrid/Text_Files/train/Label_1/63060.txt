n, m = map(int, input().split())
A = []
for i in range(n):
    A.append(list(map(int, input().split())))
ans = [[0] * m for i in range(n)]
for i in range(n - 1):
    for j in range(m - 1):
        if (A[i][j] + A[i + 1][j] + A[i][j + 1] + A[i + 1][j + 1]) % 2 != ans[i][j]:
            ans[i][j] ^= 1
            ans[i][j + 1] ^= 1
            ans[i + 1][j] ^= 1
            ans[i + 1][j + 1] ^= 1
if ans == A:
    print(sum(sum(i) for i in ans))
    for i in range(n):
        for j in range(m):
            if ans[i][j]:
                print(i + 1, j + 1)
else:
    print(-1)
