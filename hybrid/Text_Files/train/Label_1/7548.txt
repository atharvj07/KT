#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
#define endl "\n"
const int MAX = 1e5 + 1;
vector<int> adj[MAX];
bool vis[MAX];
int par[MAX];
void dfs(int node, int p)
{
    vis[node] = 1;
    par[node] = p;
    for (auto u : adj[node])
    {
        if (u != p)
            dfs(u, node);
    }
}
int main()
{
    int n;
    cin >> n;
    for (int i = 0; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 0);
    for (int i = 1; i <= n; i++)
    {
        if (!vis[i])
        {
            cout << "NO" << endl;
            return 0;
        }
    }
    vector<int> ans;
    int u = 1;
    while (true)
    {
        ans.push_back(u);
        u = par[u];
        if (u == 1)
            break;
    }
    cout << "YES" << endl;
    for (auto X : ans)
        cout << X << " ";
    cout << endl;
    return 0;
}
