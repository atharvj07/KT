n, m, h = map(int,input().split())
f, l, t = [], [], [[int(x) for x in input().split()] for y in range(n)]
f = [int(x) for x in input().split()]
for x in range(n):
    l.extend(map(int,input().split()))
for i in range(n):
    for j in range(m):
        if not f[j]:
            t[i][j] = 0
        elif not l[j*n+i]:
            t[i][j] = 0
        elif not t[i][j]:
            for k in range(min(f[j],l[j*n+i])):
                if t[i+k][j]==t[i+k][j+1]==t[i+k+1][j]==t[i+k+1][j+1]:
                    t[i+k][j] = t[i+k][j+1] = t[i+k+1][j] = t[i+k+1][j+1] = 1
                    break
print('\n'.join([' '.join(str(x) for x in y) for y in t]))
