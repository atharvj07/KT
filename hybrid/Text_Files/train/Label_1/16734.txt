#include <bits/stdc++.h>
using namespace std;
const int N = 16;
typedef long long ll;
ll a[N];
int main() {
  for (int i = 1; i < N; i++) cin >> a[i];
  ll ans = 0;
  for (int i = 1; i < N; i++) {
    ll stones = a[i];
    a[i] = 0;
    int j = i + 1;
    while (stones > 0) {
      stones--;
      a[j] = (a[j] + 1) % 2;
      j = j % N + 1;
    }
    ll score = 0;
    for (int i = 1; i < N; i++) { if (a[i] % 2 == 0) score += a[i]; }
    ans = max(ans, score);
    a[i] = stones;
  }
  cout << ans << endl;
  return 0;
}
