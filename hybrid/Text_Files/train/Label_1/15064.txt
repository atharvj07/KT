#include <bits/stdc++.h>
using namespace std;
int n, m, a[55][55];
int vis[55][55], c;
pair<int, int> dir[4] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
bool valid(int x, int y) {
 return 1 <= x && x <= n && 1 <= y && y <= m;
}
void dfs(int x, int y) {
 vis[x][y] = c;
 for (auto d : dir) {
  int nx = x + d.first, ny = y + d.second;
  if (valid(nx, ny) && a[nx][ny] == a[x][y] && vis[nx][ny] != c) {
   dfs(nx, ny);
  }
 }
}
int main() {
 cin >> n >> m;
 for (int i = 1; i <= n; i++) {
  for (int j = 1; j <= m; j++) {
   cin >> a[i][j];
  }
 }
 for (int i = 1; i <= n; i++) {
  for (int j = 1; j <= m; j++) {
   if (vis[i][j] == 0) {
    c++;
    dfs(i, j);
   }
  }
 }
 cout << (c == 1 ? "YES" : "NO");
 return 0;
}
