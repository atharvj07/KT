#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int pos = 0, neg = 0;
  long long prod = 1;
  for (int l = 0, r = 0; r < n; r++) {
    prod *= a[r];
    while (l <= r and prod < 0) prod /= a[l++];
    if (l <= r) {
      pos += r - l + 1;
      neg += r - l + 1;
    }
  }
  cout << neg << " " << pos << endl;
  return 0;
}
