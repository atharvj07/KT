t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    if k in a:
        if n == 1:
            print("yes")
        else:
            possible = "no"
            for i in range(n):
                if a[i] == k:
                    if (i > 0 and a[i-1] <= k) or (i < n-1 and a[i+1] <= k):
                        possible = "yes"
                        break
                    if (i > 1 and a[i-2] <= k) or (i < n-2 and a[i+2] <= k):
                        possible = "yes"
                        break
            print(possible)
    else:
        print("no")
