#include <bits/stdc++.h>
using namespace std;
int n, p;
string s, ans;
int main() {
  cin >> n >> p;
  cin >> s;
  ans = s;
  for (int i = n - 1, j; i >= 0; i--) {
    for (j = (int)s[i] - 97 + 1; j < p; j++) {
      char c = char(97 + j);
      if (i < 1 || s[i - 1] != c) {
        ans[i] = c;
        for (int k = i + 1; k < n; k++) {
          ans[k] = char((j + k) % p + 97);
        }
        break;
      }
    }
    if (j < p) break;
  }
  if (ans == s) {
    cout << "NO";
  } else {
    cout << ans;
  }
}
