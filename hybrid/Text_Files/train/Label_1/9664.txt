#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    
    vector<pair<long long, long long>> v;
    for (int i = 0; i < n; i++) {
        long long a, b;
        cin >> a >> b;
        v.push_back(make_pair(b, a));
    }
    
    sort(v.begin(), v.end());
    
    long long ans = 0;
    long long cnt = 0;
    for (int i = 0; i < n; i++) {
        long long b = v[i].first;
        long long a = v[i].second;
        
        if (cnt >= a) continue;
        
        ans += (a - cnt) * 2;
        cnt = a + (b - 1);
    }
    
    cout << ans << endl;
    
    return 0;
}
