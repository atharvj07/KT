n,m,k,s=map(int,input().split())
a=[[0]*m for i in range(n)]
for i in range(n): a[i]=list(map(int,input().split()))
q=list(map(int,input().split()))
h=[[(i+1+j-1)//2 for j in range(m)] for i in range(n)]
res=0
now_x=1
now_y=[0]*k
for i in q:
    for j in range(1,n+1):
        if a[j-1][h[j-1][now_y[i-1]]-1]==i:
            res=max(res,abs(now_y[i-1]-h[j-1][now_y[i-1]-1])+abs(j-now_x))
            now_x=j
            now_y[i-1]=h[j-1][now_y[i-1]-1]
print(res)
