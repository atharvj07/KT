def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def sieve(n):
    prime = [True for _ in range(n+1)]
    p = 2
    while (p * p <= n):
        if (prime[p] == True):
            for i in range(p * p, n+1, p):
                prime[i] = False
        p += 1
    prime[0]= False
    prime[1]= False
    return prime

def game(n):
    primes = sieve(int(n**0.5)+1)
    count = 0
    for i in range(2, int(n**0.5)+1):
        if primes[i]:
            temp = n
            j = 1
            while temp >= i**j:
                if primes[i]:
                    count += 1
                j += 1
    if (n-count) % 2 == 0:
        return "Petya"
    else:
        return "Vasya"

n = int(input())
print(game(n))
