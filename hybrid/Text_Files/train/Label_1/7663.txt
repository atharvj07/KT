#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    while (q--) {
        int l, r;
        cin >> l >> r;
        int ans = 0, cur = 0;
        for (int i = l - 1; i < r; i += 2) {
            cur = (cur + a[i] + a[i + 1]) % 10;
            ans += (cur >= 5);
        }
        cout << ans << "\n";
    }
    return 0;
}
