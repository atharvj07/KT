import sys

for _ in range(int(input())):
    n = int(input())
    skills = list(map(int, input().split()))
    cnt = {}
    for x in skills:
        if x in cnt:
            cnt[x] += 1
        else:
            cnt[x] = 1
    a = [cnt[x] for x in cnt if cnt[x] > 1]
    a.sort(reverse=True)
    ans = 0
    for x in a:
        if ans + x > n // 2:
            break
        ans += x
    ans = min(ans, n // 2)
    print(ans)
