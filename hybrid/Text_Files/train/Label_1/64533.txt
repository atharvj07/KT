import sys
from collections import deque

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
op = deque()

for r in range(n - 1, 1, -1):
    for l in range(r):
        if (r - l + 1) <= 5 and n % (r - l + 1) == 0:
            a[l], a[r] = a[r], a[l]
            op.appendleft('%d %d' % (l + 1, r + 1))

print(len(op))
for i in op:
    print(i)
