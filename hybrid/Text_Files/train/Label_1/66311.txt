import sys

n, m = map(int, input().split())
house = [input() for _ in range(n)]

def solve(r, c, d, seen):
    if r == n - 1 and c == 0:
        return 0
    
    if (r, c, d) in seen:
        return -1
    
    seen.add((r, c, d))
    
    if house[r][c] == '.':
        if d == 1:
            if solve(r + 1, c, d, seen) != -1:
                return solve(r + 1, c, d, seen) + 1
        else:
            if solve(r + 1, c - 1, d, seen) != -1:
                return solve(r + 1, c - 1, d, seen) + 1
    else:
        if d == 1:
            if house[r][c] == '+' and solve(r, c + 1, -d, seen) != -1:
                return solve(r, c + 1, -d, seen) + 1
            elif house[r][c] == '#' and solve(r, c + 1, d, seen) != -1:
                return solve(r, c + 1, d, seen) + 1
        else:
            if house[r][c] == '+' and solve(r, c - 1, -d, seen) != -1:
                return solve(r, c - 1, -d, seen) + 1
            elif house[r][c] == '#' and solve(r, c - 1, d, seen) != -1:
                return solve(r, c - 1, d, seen) + 1
    
    return -1

if solve(0, 0, 1, set()) != -1:
    print(solve(0, 0, 1, set()))
else:
    print('Never')
