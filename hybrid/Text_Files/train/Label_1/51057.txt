import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] cups = new int[n];
        
        for (int i = 0; i < n; i++) {
            cups[i] = input.nextInt();
        }
        
        for(int i = 0; i < n; i++){
            for(int j = i+1; j < n; j++){
                int poured = cups[i] + cups[j] - cups[i]%2 + cups[j]%2 - cups[i]/2 % 2 + cups[j]/2 % 2 - cups[i]/2 - cups[j]/2;
                if (poured <= 0 || poured % 2 == 1) {
                    continue;
                }

                poured /= 2;
                if (cups[i] - poured >= 0 && cups[j] + poured <= 1000) {
                    System.out.println(poured + " ml. from cup #" + (i + 1) + " to cup #" + (j + 1) + ".");
                    return;
                }
            }
        }
        System.out.println("Unrecoverable configuration.");
    }
}
