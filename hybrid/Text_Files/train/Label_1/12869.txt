#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N = 2e5+5;
vector<int>g[N];
int x[N],n,k,vis[N];
bool ok;
void dfs(int u,int p){
	if(ok)return;
	if(vis[u])ok=1;
	vis[u]=1;
	for(auto v:g[u]){
		if(v==p)continue;
		dfs(v,u);
	}
}
int main(){
	int t;cin>>t;
	while(t--){
		cin>>n>>k;
		for(int i=1;i<=n;i++)g[i].clear(),vis[i]=0;
		ok=0;
		for(int i=1;i<=k;i++)cin>>x[i];
		for(int i=1;i<n;i++){
			int u,v;cin>>u>>v;
			g[u].push_back(v);
			g[v].push_back(u);
		}
		for(int i=1;i<=k;i++){
			if(ok)continue;
			dfs(x[i],x[i]);
		}
		cout<<(ok?"YES":"NO")<<endl;
	}
}
