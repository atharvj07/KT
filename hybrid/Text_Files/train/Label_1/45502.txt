import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        boolean[] remain = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            remain[i] = true;
        }

        for (int i = 0; i < q; i++) {
            char t = sc.next().charAt(0);
            char d = sc.next().charAt(0);

            for (int j = 1; j <= n; j++) {
                if (s.charAt(j - 1) == t) {
                    if ((d == 'L' && j == 1) || (d == 'R' && j == n)) {
                        remain[j] = false;
                    } else if (d == 'L') {
                        remain[j] = remain[j - 1];
                    } else if (d == 'R') {
                        remain[j] = remain[j + 1];
                    }
                }
            }
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (remain[i]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
