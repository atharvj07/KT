#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int s;
        cin >> s;
        if (s == 1)
            cout << 1 << endl;
        else if (s == 2)
            cout << 2 << endl;
        else {
            int ans = s / 3;
            if (s % 3 != 0)
                ans++;
            cout << ans << endl;
        }
    }
    return 0;
}
