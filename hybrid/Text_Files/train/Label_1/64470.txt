n,m = map(int,input().split())
a = [list(input()) for i in range(n)]
vis = [[0 for i in range(m)] for j in range(n)]

def dfs1(x,y):
    if(x<0 or y<0 or x>=n or y>=m or a[x][y]=='W' or vis[x][y]==1):
        return
    vis[x][y]=1
    dfs1(x+1,y)
    dfs1(x-1,y)    
    dfs1(x,y+1)
    dfs1(x,y-1)
    
def dfs2(x,y):
    if(x<0 or y<0 or x>=n or y>=m or a[x][y]=='W' or vis[x][y]==2):
        return
    vis[x][y]=2
    dfs2(x+1,y)
    dfs2(x-1,y)    
    dfs2(x,y+1)
    dfs2(x,y-1)
    
dfs1(0,0)
x=-1;y=-1
for i in range(n):
    for j in range(m):
        if(vis[i][j]==1):
            x=i;y=j
            break
    if(x!=-1):
        break
dfs2(x,y)
f=0
for i in range(n):
    for j in range(m):
        if(vis[i][j]==1):
            f=1
            break
if(f==0):
    print("YES")
else:
    print("NO")
