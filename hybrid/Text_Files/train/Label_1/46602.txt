import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        Arrays.sort(a);

        int ans = 0;
        for (int i = 0; i < n; i++) {
            int j = n - 1;
            while (i < j && a[i] + a[j] > 10) {
                j--;
            }
            if (i < j) {
                ans++;
                i++;
                j--;
            }
        }

        System.out.println(ans);
    }
}
