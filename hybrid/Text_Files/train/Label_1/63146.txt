MOD = 998244353

def modinv(a, m):
    return pow(a, m - 2, m)

def factorial(n):
    res = [1] * (n + 1)
    for i in range(2, n + 1):
        res[i] = res[i - 1] * i % MOD
    return res

n = int(input())
a = list(map(int, input().split()))
a.sort()

fact = factorial(2 * n)
inv = [modinv(fact[i], MOD) for i in range(2 * n + 1)]

res = 0
for i in range(n):
    res = (res + (a[n + i] - a[i]) * fact[2 * n] % MOD * inv[n] % MOD * inv[n]) % MOD

print(res)
