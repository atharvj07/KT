import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] winners = new int[n];
        for (int i = 0; i < n; i++) {
            int[] powers = new int[k];
            for (int j = 0; j < k; j++) {
                powers[j] = scanner.nextInt();
            }
            int maxPower = -1;
            int numWinners = 0;
            for (int j = 0; j < k; j++) {
                if (powers[j] > maxPower) {
                    maxPower = powers[j];
                    numWinners = 1;
                } else if (powers[j] == maxPower) {
                    numWinners++;
                }
            }
            winners[i] = numWinners;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(winners[i]);
        }
    }
}
