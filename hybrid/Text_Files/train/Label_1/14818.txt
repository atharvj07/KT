#include <iostream>
#include <vector>

using namespace std;

int main() {
  int k;
  cin >> k;
  vector<vector<int>> ans(1 << k, vector<int>(1 << k));
  for (int i = 0; i < (1 << k); i++) {
    for (int j = i + 1; j < (1 << k); j++) {
      int cnt = 0;
      for (int l = 0; l < k; l++) {
        if (((i >> l) & 1) != ((j >> l) & 1)) cnt++;
      }
      if (cnt % 2 == 1) ans[i][j] = ans[j][i] = 1;
    }
  }
  for (int i = 0; i < (1 << k); i++) {
    for (int j = 0; j < (1 << k); j++) {
      if (ans[i][j] == 1) cout << "*"; else cout << "+";
    }
    cout << endl;
  }
}
