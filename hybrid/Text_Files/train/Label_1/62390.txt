n, q = map(int, input().split())
colors = list(map(int, input().split()))
queries = list(map(int, input().split()))

positions = {}
for i, color in enumerate(colors):
    if color not in positions:
        positions[color] = i + 1

for t in queries:
    pos = positions[t]
    print(pos, end=' ')
    for c in positions:
        if positions[c] < pos:
            positions[c] += 1
    positions[t] = 1
