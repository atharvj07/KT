import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    l = [int(x) for x in input().split()]
    unlocked = [i for i, x in enumerate(l) if x == 0]
    a_unlocked = [a[i] for i in unlocked]
    a_unlocked.sort()
    for i in unlocked:
        a[i] = a_unlocked.pop(0)
    print(*a)
