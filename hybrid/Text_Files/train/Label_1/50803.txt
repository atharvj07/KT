import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.PriorityQueue;

public class ShapursTravel {

    private static class Edge implements Comparable<Edge> {
        int to;
        int weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }

        @Override
        public int compareTo(Edge other) {
            return Integer.compare(weight, other.weight);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        List<List<Edge>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            String[] parts = reader.readLine().split(" ");
            int from = Integer.parseInt(parts[0]) - 1;
            int to = Integer.parseInt(parts[1]) - 1;
            int weight = Integer.parseInt(parts[2]);
            graph.get(from).add(new Edge(to, weight));
            graph.get(to).add(new Edge(from, weight));
        }

        int[] distances = new int[n];
        boolean[] visited = new boolean[n];
        PriorityQueue<Edge> queue = new PriorityQueue<>();
        queue.offer(new Edge(0, 0));
        while (!queue.isEmpty()) {
            Edge edge = queue.poll();
            if (visited[edge.to]) {
                continue;
            }
            visited[edge.to] = true;
            distances[edge.to] = edge.weight;
            for (Edge nextEdge : graph.get(edge.to)) {
                if (!visited[nextEdge.to]) {
                    queue.offer(new Edge(nextEdge.to, edge.weight + nextEdge.weight));
                }
            }
        }

        int result = 0;
        for (int distance : distances) {
            result += distance;
        }
        System.out.println(result);
    }
}
