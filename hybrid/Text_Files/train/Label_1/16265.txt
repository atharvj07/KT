#include <bits/stdc++.h>
using namespace std;
const int N = 3e4 + 5;
int a[N], b[N], c[N], deg[N];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= 4 * n; i++) {
    cin >> a[i] >> b[i] >> c[i];
    ++deg[a[i]];
  }
  for (int i = 1; i <= n; i++) {
    if (deg[i] > 4) {
      cout << "NO";
      return 0;
    }
  }
  cout << "YES\n";
  for (int i = 1; i <= 4 * n; i++) cout << i << ' ';
  return 0;
}
