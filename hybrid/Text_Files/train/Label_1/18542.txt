#include <iostream>
#include <cmath>

using namespace std;

bool isLeapYear(int yr) {
  if (yr % 400 == 0) return true;
  else if (yr % 100 == 0) return false;
  else if (yr % 4 == 0) return true;
  return false;
}

int main(int argc, char** argv) {
  string d1, d2;
  cin >> d1 >> d2;
  int y1, y2, m1, m2, d3, d4;
  y1 = stoi(d1.substr(0, 4));
  m1 = stoi(d1.substr(5, 2));
  d3 = stoi(d1.substr(8, 2));
  y2 = stoi(d2.substr(0, 4));
  m2 = stoi(d2.substr(5, 2));
  d4 = stoi(d2.substr(8, 2));

  int days[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

  int totalDays = 0;

  for (int i = y1; i <= y2; i++) {
    if (i == y1) {
      for (int j = m1; j <= 12; j++) {
        if (j == 2 && isLeapYear(i)) totalDays += 29;
        else totalDays += days[j - 1];
      }
    } else if (i == y2) {
      for (int j = 1; j < m2; j++) {
        if (j == 2 && isLeapYear(i)) totalDays += 29;
        else totalDays += days[j - 1];
      }
      totalDays += d4;
    } else {
      if (isLeapYear(i)) totalDays += 366;
      else totalDays += 365;
    }
  }

  totalDays -= d3;

  cout << totalDays << endl;

  return 0;
}
