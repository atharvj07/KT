#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
typedef long long ll;
vector<int> g[100001];
int c[100001];
int p[100001];
bool vis[100001];
vector<ll> dp(100001, -1);
ll dfs(int cur, int par, bool is_black) {
  if (dp[cur] != -1) return dp[cur];
  dp[cur] = 1;
  if (c[cur] == 1) is_black = true;
  for (int child : g[cur]) {
    if (child == par) continue;
    dp[cur] *= dfs(child, cur, is_black);
    dp[cur] %= mod;
  }
  if (!is_black) dp[cur] = 0;
  return dp[cur];
}
int main() {
  int n;
  cin >> n;
  for (int i = 1; i < n; i++) {
    cin >> p[i];
    g[p[i]].push_back(i);
    g[i].push_back(p[i]);
  }
  for (int i = 0; i < n; i++) cin >> c[i];
  cout << dfs(0, 0, false) << endl;
  return 0;
}
