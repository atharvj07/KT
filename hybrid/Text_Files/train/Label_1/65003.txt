import collections

n, m = map(int, input().split())
d = collections.defaultdict(list)
for _ in range(m):
    op, user = input().split()
    op = -1 if op == '-' else 1
    d[user].append(op)

ans = []
for user, state in d.items():
    if not sum(state): continue
    flag = True
    for i in range(len(state)):
        if flag and not state[i]:
            ans.append(int(user))
            flag = False
        elif not flag and state[i]:
            break
print(len(ans))
print(*ans)
