s = "What are you doing at the end of the world? Are you busy? Will you save us?"
Q = int(input())
a = []
for i in range(Q):
    n,k = map(int,input().split())
    if n==0:
        if k<len(s):
            a.append(s[k-1])
        else:
            a.append('.')
    else:
        if k<71:
            a.append(s[k-1])
        else:
            a.append(a[(k-71)%len(a)])
print(''.join(a))
