import java.util.Scanner;

public class Main {
    private static class Position {
        int rowIndex;
        int columnIndex;

        public Position(int rowIndex, int columnIndex) {
            this.rowIndex = rowIndex;
            this.columnIndex = columnIndex;
        }
    }

    private static int calculateMinimumCost(int numberOfRows, int numberOfColumns, int tile1Cost, int tile2Cost, char[][] grid) {
        int[][] dp = new int[numberOfRows + 1][numberOfColumns + 1];

        for (int rowIndex = numberOfRows; rowIndex >= 1; rowIndex--) {
            for (int columnIndex = numberOfColumns; columnIndex >= 1; columnIndex--) {
                if (grid[rowIndex - 1][columnIndex - 1] == '*') {
                    dp[rowIndex - 1][columnIndex - 1] = dp[rowIndex][columnIndex];
                } else {
                    dp[rowIndex - 1][columnIndex - 1] = Math.min(dp[rowIndex][columnIndex] + tile1Cost, dp[rowIndex][columnIndex - 1] + tile2Cost);
                }
            }
        }

        return dp[0][0];
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int numberOfTestCases = scanner.nextInt();

        for (int testCase = 0; testCase < numberOfTestCases; testCase++) {
            int numberOfRows = scanner.nextInt();
            int numberOfColumns = scanner.nextInt();
            int tile1Cost = scanner.nextInt();
            int tile2Cost = scanner.nextInt();
            char[][] grid = new char[numberOfRows][numberOfColumns];

            for (int rowIndex = 0; rowIndex < numberOfRows; rowIndex++) {
                String row = scanner.next();
                for (int columnIndex = 0; columnIndex < numberOfColumns; columnIndex++) {
                    grid[rowIndex][columnIndex] = row.charAt(columnIndex);
                }
            }

            int minimumCost = calculateMinimumCost(numberOfRows, numberOfColumns, tile1Cost, tile2Cost, grid);
            System.out.println(minimumCost);
        }
    }
}
