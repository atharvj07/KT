import sys
input = sys.stdin.readline

n = int(input())
products = []
for _ in range(n):
    (a, b) = map(int, input().split())
    products.append((a, b))

products.sort(key=lambda x: x[0]+x[1])

total = 0
item_count = 0

for (a, b) in products:
    if item_count >= a + b:
        continue
    total += (a + b - item_count)
    item_count += (a + b)

print(total)
