def max_seal_area(n, a, b, seals):
    max_area = 0
    for i in range(n):
        for j in range(i + 1, n):
            for (w1, h1) in [(seals[i][0], seals[i][1]), (seals[i][1], seals[i][0])]:
                for (w2, h2) in [(seals[j][0], seals[j][1]), (seals[j][1], seals[j][0])]:
                    if ((w1 + w2) <= a and max(h1, h2) <= b) or ((w1 + w2) <= b and max(h1, h2) <= a) or \
                       ((h1 + h2) <= b and max(w1, w2) <= a) or ((h1 + h2) <= a and max(w1, w2) <= b):
                        max_area = max(max_area, w1 * h1 + w2 * h2)
    return max_area

n, a, b = map(int, input().split())
seals = [list(map(int, input().split())) for _ in range(n)]
print(max_seal_area(n, a, b, seals))
