#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    int ans = a[(n - 1) / 2];
    for (int i = 0; i < (n - 1) / 2; ++i) {
        ans -= a[i];
    }
    for (int i = (n + 1) / 2; i < n; ++i) {
        ans += a[i] - a[(n - 1) / 2];
    }
    std::cout << ans << '\n';
    return 0;
}
