import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt();
            int y = sc.nextInt();
            map.computeIfAbsent(p, k -> new ArrayList<>()).add(y);
        }
        for (int i = 1; i <= n; i++) {
            List<Integer> years = map.getOrDefault(i, new ArrayList<>());
            Collections.sort(years);
            for (int j = 0; j < years.size(); j++) {
                String id = String.format("%06d%06d", i, j + 1);
                System.out.println(id);
            }
        }
    }
}
