for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    ans = 0
    for i in range(n):
        for j in range(i + 1, n):
            if a[i] == a[j]:
                for k in range(j + 1, n):
                    for l in range(k + 1, n):
                        if a[k] == a[l]:
                            ans += 1
    print(ans)
