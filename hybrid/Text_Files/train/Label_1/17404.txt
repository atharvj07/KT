#include <iostream>
#include <queue>
using namespace std;

int main() {
  int n, x, crimes = 0;
  queue<int> police_force;
  cin >> n;
  while (n--) {
    cin >> x;
    if (x > 0) {
      police_force.push(x);
    } else {
      if (police_force.empty())
        crimes++;
      else
        police_force.pop();
    }
  }
  cout << crimes << endl;
  return 0;
}
