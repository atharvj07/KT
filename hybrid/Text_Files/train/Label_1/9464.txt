#include <bits/stdc++.h>
using namespace std;
int main() {
  string a, b;
  cin >> a >> b;
  int n = a.size(), m = b.size();
  vector<int> cnt(n, 0);
  for (int i = 0; i + m - 1 < n; i++) {
    int dif = 0;
    for (int j = 0; j < m; j++) {
      if (a[i + j] != b[j]) dif++;
    }
    if (dif % 2 == 0) cnt[i] = 1;
    cnt[i] += cnt[i - 1];
  }
  cout << cnt[n - m] << "\n";
  return 0;
}
