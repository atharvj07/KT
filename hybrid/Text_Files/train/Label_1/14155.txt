#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> divs(n);
        for (int i = 0; i < n; i++) {
            cin >> divs[i];
        }
        sort(divs.begin(), divs.end());
        if (divs.size() == 1) {
            cout << -1 << endl;
            continue;
        }
        long long ans = 1;
        for (int i = 1; i < divs.size(); i++) {
            if (divs[i] % divs[i - 1] != 0) {
                ans = -1;
                break;
            }
            ans *= divs[i];
        }
        cout << ans << endl;
    }
    return 0;
}
