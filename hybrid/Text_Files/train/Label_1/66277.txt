n, k, d = map(int, input().split())
a = sorted(list(map(int, input().split())))
ans = 'NO'
for i in range(n):
    l = i
    r = n - 1
    while l <= r:
        m = (l + r) // 2
        if a[m] - a[i] <= d:
            l = m + 1
        else:
            r = m - 1
    if l - i >= k:
        ans = 'YES'
        break
print(ans)
