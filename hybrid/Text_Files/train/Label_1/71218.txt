n, B = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
even = [i for i in a if i % 2 == 0]
odd = [i for i in a if i % 2 == 1]
cur = 0
ans = 0
for i in even:
    if cur + i <= B:
        ans += 1
        cur += i
    else:
        break
e = len(even) - 1
o = len(odd) - 1
while e >= 0 and o >= 0:
    if cur + even[e] - odd[o] <= B:
        ans += 1
        cur += even[e] - odd[o]
        e -= 1
        o -= 1
    else:
        break
print(ans)
