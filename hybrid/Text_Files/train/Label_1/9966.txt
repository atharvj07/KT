#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ll a, b, f, k;
  cin >> a >> b >> f >> k;
  if (b >= a) {
    cout << 0;
    return 0;
  }
  if (2 * f < a) {
    cout << -1;
    return 0;
  }
  k--;
  ll x = (a + f) / b;
  if ((a + f) % b == 0) x--;
  if (k > x)
    cout << x + 1;
  else
    cout << k;
}
