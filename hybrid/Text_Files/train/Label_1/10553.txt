#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> c(n, vector<int>(m));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) cin >> c[i][j];
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (c[i][j] == -1) continue;
            bool ok = true;
            for (int k = 0; k < n; ++k) {
                if (c[k][j] != c[i][j] && c[k][j] != -1) {
                    ok = false;
                }
            }
            for (int k = 0; k < m; ++k) {
                if (c[i][k] != c[i][j] && c[i][k] != -1) {
                    ok = false;
                }
            }
            if (ok) {
                ans++;
                for (int k = 0; k < n; ++k) {
                    c[k][j] = -1;
                }
                for (int k = 0; k < m; ++k) {
                    c[i][k] = -1;
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
