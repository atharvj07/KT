#include <bits/stdc++.h>

using namespace std;

int main()
{
    long long n, m, k;
    cin >> n >> m >> k;
    vector<pair<int, int>> p(k);
    for (int i = 0; i < k; i++) {
        cin >> p[i].first >> p[i].second;
    }
    set<pair<int, int>> s;
    for (int i = 0; i < k; i++) {
        s.insert(p[i]);
    }
    int ans = 0;
    while ((int)s.size() != n * m) {
        ans++;
        set<pair<int, int>> next;
        for (auto u : s) {
            if (!(1 <= u.first && u.first <= n && 1 <= u.second && u.second <= m)) {
                continue;
            }
            if (s.count(make_pair(u.first - 1, u.second)) == 0 && 1 <= u.first - 1 && u.first - 1 <= n && 1 <= u.second && u.second <= m) {
                next.insert(make_pair(u.first - 1, u.second));
            }
            if (s.count(make_pair(u.first + 1, u.second)) == 0 && 1 <= u.first + 1 && u.first + 1 <= n && 1 <= u.second && u.second <= m) {
                next.insert(make_pair(u.first + 1, u.second));
            }
            if (s.count(make_pair(u.first, u.second - 1)) == 0 && 1 <= u.first && u.first <= n && 1 <= u.second - 1 && u.second - 1 <= m) {
                next.insert(make_pair(u.first, u.second - 1));
            }
            if (s.count(make_pair(u.first, u.second + 1)) == 0 && 1 <= u.first && u.first <= n && 1 <= u.second + 1 && u.second + 1 <= m) {
                next.insert(make_pair(u.first, u.second + 1));
            }
        }
        for (auto u : next) {
            s.insert(u);
        }
    }
    cout << ans << endl;
    return 0;
}
