import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        PriorityQueue<int[]> lanes = new PriorityQueue<>((a, b) -> a[1] != b[1] ? Integer.compare(a[1], b[1]) : Integer.compare(a[0], b[0]));
        for (int i = 1; i <= n; i++) {
            lanes.offer(new int[]{i, 0});
        }
        for (int i = 0; i < m; i++) {
            int type = scanner.nextInt();
            if (type == 0) {
                int lane = scanner.nextInt();
                lanes.remove(new int[]{lane, lanes.peek()[1]});
                System.out.println(lanes.peek()[0]);
            } else {
                int car = scanner.nextInt();
                int[] lane = lanes.poll();
                lane[1]++;
                lanes.offer(lane);
            }
        }
        scanner.close();
    }
}
