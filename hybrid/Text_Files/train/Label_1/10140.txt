#include <iostream>
#include <vector>
using namespace std;

bool solveCase(int n, int m, int k, vector<int>& hs) {
    int bag = m;
    for (int i = 0; i + 1 < n; i++) {
        int diff = hs[i + 1] - hs[i];
        if (diff <= 0) {
            continue;
        }
        if (diff <= bag) {
            bag -= diff;
        } else if (k == 0) {
            return false;
        } else {
            int d = (diff + k - 1) / k;
            if (bag >= d) {
                bag -= d;
                bag += hs[i];
            } else {
                return false;
            }
        }
    }
    return true;
}

int main() {
    int t;
    cin >> t;
    for (int i = 0; i < t; i++) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<int> hs(n);
        for (int j = 0; j < n; j++) {
            cin >> hs[j];
        }
        if (solveCase(n, m, k, hs)) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
}
