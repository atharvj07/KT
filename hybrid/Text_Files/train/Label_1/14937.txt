#include <cstdio>
#include <vector>
#define ll long long
using namespace std;
const int MAXN = 100005;
ll a[MAXN] = {0}, b[MAXN] = {0};
vector<pair<ll, pair<ll, ll> > > event;
int main() {
    int n, q;
    scanf("%d%d", &n, &q);
    for (int i = 1; i <= q; i++) {
        int t, k, d;
        scanf("%d%d%d", &t, &k, &d);
        event.push_back(make_pair(t, make_pair(k, d)));
    }
    ll ans = -1;
    int type = 0;
    for (int i = 1; i <= q; i++) {
        ll e = event[i - 1].first;
        ll k = event[i - 1].second.first;
        ll d = event[i - 1].second.second;
        while (a[type] != 0 && a[type] <= e) {
            type++;
        }
        if (type + k - 1 <= n) {
            ans = 0;
            for (int j = 0; j < k; j++) {
                ans += type + j + 1;
                b[type + j + 1] = e + d;
            }
        }
        else
            ans = -1;
        printf("%lld\n", ans);
        for (int j = 1; j <= n; j++) {
            if (b[j] != 0 && b[j] <= e)
                b[j] = 0;
        }
    }
    return 0;
}
