#include <bits/stdc++.h>
using namespace std;
int n, x[100005], y[100005], ans;
int main()
{
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> x[i];
    }
    sort(x + 1, x + 1 + n);
    for (int i = 1; i + 1 <= n; i++)
    {
        if (x[i] == x[i + 1])
            x[i] = -1, ans++;
    }
    sort(x + 1, x + 1 + n);
    int t = (n - ans) / 2;
    for (int i = 1; i <= n; i++)
    {
        if (x[i] >= 0)
            ans++;
        if (t > 0 and ans <= t)
            y[i] = x[i], t--;
        else
            y[i] = x[n - ans + 1];
    }
    cout << ans << "\n";
    for (int i = 1; i <= n; i++)
        cout << y[i] << " ";
}
