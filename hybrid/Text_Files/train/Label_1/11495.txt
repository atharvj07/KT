#include <bits/stdc++.h>
using namespace std;
int main() {
  string x;
  cin >> x;
  int exp = 0, len = x.length();
  for (int i = 0; i < len; ++i) {
    if (x[i] == '.') {
      exp += i + 1;
      x.erase(i, 1);
      len--;
      i--;
    } else if (x[i] == '0' && exp == 0 && i != len - 1) {
      x.erase(i, 1);
      len--;
      i--;
    } else if (x[i] == '0' && exp != 0 && (i == 0 || x[i - 1] == '.')) {
      x.erase(i, 1);
      len--;
      i--;
      exp++;
    }
  }
  if (x[len - 1] == '.') x.erase(len - 1, 1), len--;
  if (len == 1 && x[0] == '0') {
    x = "0";
    exp = 0;
  }
  cout << x << (exp ? "E" + to_string(exp) : "");
  return 0;
}
