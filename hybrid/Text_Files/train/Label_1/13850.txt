#include <bits/stdc++.h>

using namespace std;

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  string s, t;
  cin >> s >> t;
  int p = s.find(t[0]);
  int q = 0;
  for (char c : t) {
    int w = p;
    for (int r = p + 1; r < s.size(); r++) {
      if (s[r] == c) {
        p = r;
        break;
      }
    }
    if (p == w) {
      cout << "-1\n";
      return 0;
    }
  }
  cout << s.size() - t.size() << "\n";
}
