from collections import defaultdict
from heapq import *

def dijkstra(n, graph, edge):
    pq, dist, visited = [(0, 1)], [0] * (n+1), [False] * (n+1)
    while pq:
        w, u = heappop(pq)
        if visited[u]: continue
        visited[u] = True
        for v, w_ in graph[u]:
            if not visited[v]:
                heappush(pq, (w + w_, v))
        edge[1] = max(edge[1], w)

n = int(input())
graph = defaultdict(list)
for _ in range(n-1):
    a, b = map(int, input().split())
    graph[a].append((b,1))
    graph[b].append((a,1))

edge = [0,0]; dijkstra(n, graph, edge)
print(n-1-edge[1])
