import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = s.charAt(i) - '0';
            int[] dp = new int[n + 1];
            dp[n] = 1;
            for (int i = n - 1; i >= 0; i--) {
                dp[i] = dp[i + 1];
                if (dp[i] == 0) continue;
                if (i > 0 && a[i - 1] > a[i]) dp[i] = 0;
                if (a[i] == 0 && i > 0 && a[i - 1] == 0) dp[i] = 0;
            }
            if (dp[0] == 0) {
                System.out.println('-');
            } else {
                int i = 0, j = 0;
                char[] ans = new char[n];
                while (i < n) {
                    if (dp[i + 1] == 0) {
                        ans[i] = '2';
                        i++;
                    } else if (i == 0 || a[i] != a[i - 1]) {
                        ans[i] = '1';
                        dp[i] = 0;
                        i++;
                    } else {
                        j = i;
                        while (j < n && dp[j + 1] != 0 && a[j] == a[i]) j++;
                        if (j == n) {
                            ans[i] = '1';
                            i++;
                        } else {
                            ans[i] = '2';
                            dp[j] = 0;
                            i = j + 1;
                        }
                    }
                }
                System.out.println(ans);
            }
        }
    }
}
