#include <bits/stdc++.h>
using namespace std;

const int N = 3e5 + 5;
vector<int> adj[N];
int n, x, y, sub[N], bad[N]; // subtree for each node, bad value for each node (if true, we can't choose this node as start nor end)

void dfs(int s, int p) {
    sub[s] = 1;
    for (auto u : adj[s]) {
        if (u == p) continue;
        dfs(u, s);
        sub[s] += sub[u];
    }
}

void dfs2(int s, int p, int state) {
    bad[s] = state;
    for (auto u : adj[s]) {
        if (u == p) continue;
        dfs2(u, s, state or (u == x and sub[u] > sub[s]));
    }
}

int main() {
    cin >> n >> x >> y;
    for (int i = 1; i < n; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    // root the tree at flowrisa so that all bad nodes are in the subtree of beetopia
    dfs(x, x);
    dfs2(x, x, 0);
    if (bad[y]) dfs2(y, y, 0);

    long long ans = 1ll * n * (n - 1) / 2;
    for (int i = 1; i <= n; i++) {
        ans -= 1ll * bad[i] * (sub[i] - bad[i] - 1);
    }
    cout << ans << endl;
    return 0;
}
