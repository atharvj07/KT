#include <bits/stdc++.h>
using namespace std;

long long n, t, a[200'005];

int main() {
  cin >> n >> t;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  long long ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (t >= a[i]) {
      t -= a[i];
      ans++;
    } else
      break;
  }
  cout << ans << endl;
}
