from collections import deque

n, k = map(int, input().split())
s = input()

if n < k: print(-1); exit()

left, right = deque(), deque()
for c in s:
    while left and left[-1] > c: left.pop()
    while right and right[-1] < c: right.pop()
    left.append(c)
    right.append(c)
    if len(left) == k:
        print(n - len(right))
        break
else: print(-1)
