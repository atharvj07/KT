from collections import deque

def dfs(v):
    for u in graph[v]:
        if u != par[v]:
            par[u] = v
            dfs(u)
            xs[u] = xs[v]
            if v % 2 == 0:
                xs[u] += 1
            else:
                xs[u] -= 1
            ys[u] = ys[v]
            if par[v] % 2 == 0:
                ys[u] += 1
            else:
                ys[u] -= 1

n = int(input())
graph = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)
que = deque([1])
par = [0] * (n + 1)
vis = [False] * (n + 1)
while que:
    v = que.popleft()
    if vis[v]:
        continue
    vis[v] = True
    for u in graph[v]:
        if not vis[u]:
            que.append(u)
if sum([len(u) for u in graph]) % 2 == 1:
    print("NO")
else:
    print("YES")
    xs = [0] * (n + 1)
    ys = [0] * (n + 1)
    dfs(1)
    for i in range(1, n + 1):
        print(xs[i], ys[i])
