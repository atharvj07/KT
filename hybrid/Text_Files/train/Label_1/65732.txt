import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m, k = map(int, input().split())
    data = [[] for i in range(n+1)]
    for i in range(m):
        x, y, s = map(int, input().split())
        data[x].append((y, s))
        data[y].append((x, s))
    vis = [False] * (n+1)
    vis[1] = True
    ans = 0
    def dfs(x, speed):
        nonlocal ans
        for y, s in data[x]:
            if not vis[y]:
                vis[y] = True
                if speed == k: ans += 1
                dfs(y, min(speed, s))
    dfs(1, 1<<30)
    print(ans)
