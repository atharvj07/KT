import sys

# read input
s = list(sys.stdin.readline().strip())

# count bombs and numbers
bombs = s.count('*')
numbers = sum(int(c) for c in s if c in '12')

# initialize dp table
dp = [0] * (bombs + numbers + 1)
dp[0] = 1

# calculate dp table
for c in s:
    if c == '?' or c == '0':
        for i in range(bombs, -1, -1):
            dp[i + 1] += dp[i]

# print result
print(dp[bombs] % 1000000007)
