#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  ll n, m, a[100005], b[100005], c[100005] = {0}, ans = 0;
  cin >> n >> m;
  for (ll i = 0; i < n; i++) cin >> a[i];
  for (ll i = 0; i < m; i++) cin >> b[i];
  sort(a, a + n);
  sort(b, b + m);
  for (ll i = 0; i < n; i++) c[a[i] % 2]++;
  for (ll i = 0; i < m; i++) c[b[i] % 2]++;
  ans = min(c[0], c[1]);
  cout << ans << endl;
  return 0;
}
