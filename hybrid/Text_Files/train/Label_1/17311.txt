#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9+7;
#define ll long long
#define pb push_back
#define mp make_pair
int n,k,a[100005][5],ans;
ll x,y;
int32_t main()
{
    cin>>n>>k;
    for(int i=1;i<=n;i++)
    {
        cin>>a[i][1]>>a[i][2]>>a[i][3]>>a[i][4];
        x+=(a[i][3]-a[i][1]+1);
        y+=(a[i][4]-a[i][2]+1);
    }
    if(n-k>1) ans=1;
    else ans=x*y;
    cout<<ans<<endl;
}
