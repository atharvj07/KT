import math 
n, s = map(int, input().split()) 
a = [[0, 0, 0] for i in range(n)] 
for i in range(n): 
    a[i] = list(map(int, input().split())) 
    s += a[i][2] 
if s < 1e6: 
    print(-1) 
    exit() 
l, r = 0, 0 
for i in range(n): 
    l = max(l, a[i][0] ** 2 + a[i][1] ** 2) 
r = 1e10 
while l + 1e-6 < r: 
    m = (l + r) / 2 
    pop = s 
    for i in range(n): 
        if a[i][0] ** 2 + a[i][1] ** 2 <= m: 
            pop += a[i][2] 
    if pop >= 1e6: 
        r = m 
    else: 
        l = m 
print(math.sqrt(r))
