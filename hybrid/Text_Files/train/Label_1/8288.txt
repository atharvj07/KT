#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> v;
    for (int i = 0; i < n; i++) {
        v.push_back(s[i] - '0');
    }
    for (int i = 0; i < n; i++) {
        if (k == 0) {
            break;
        }
        if (v[i] != 0) {
            continue;
        }
        int j = i + 1;
        while (j < n && v[j] == 0) {
            j++;
        }
        if (j == n) {
            break;
        }
        swap(v[i], v[j]);
        k--;
    }
    for (int i = 0; i < n; i++) {
        if (v[i] != 0) {
            cout << v[i];
        }
    }
    for (int i = n - 1; i >= 0; i--) {
        if (v[i] == 0) {
            cout << 0;
        }
    }
    cout << endl;
    return 0;
}
