import sys

n = int(sys.stdin.readline())
s1 = set(list(map(int, sys.stdin.readline().split())) + [1])
s2 = set(list(map(int, sys.stdin.readline().split())) + [1])
print(' '.join([['Lose', 'Win', 'Loop'][((n - i) % (max(s1) + 1) not in s1)] for i in range(1, n)]))
print(' '.join([['Lose', 'Win', 'Loop'][((n - i) % (max(s2) + 1) not in s2)] for i in range(1, n)]))
