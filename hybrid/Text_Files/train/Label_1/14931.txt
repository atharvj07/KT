#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> c(n), a(m);
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  for (int i = 0; i < m; i++) { cin >> a[i]; }
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (j == m) {
      break;
    }
    if (a[j] < c[i]) {
      continue;
    }
    j++;
  }
  cout << j << endl;
  return 0;
}
