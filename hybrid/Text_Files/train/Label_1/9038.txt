#include <bits/stdc++.h>
using namespace std;

const int mod = 998244353;
const int maxn = 2e5 + 5;

long long fastpow(long long a, long long b) {
  long long ans = 1ll;
  while (b) {
    if (b & 1ll) { ans = (ans * a) % mod; }
    a = (a * a) % mod;
    b >>= 1;
  }
  return ans;
}

int n, m;
long long f[maxn], invf[maxn];

int main() {
  cin >> n >> m;
  f[0] = invf[0] = 1;
  for (int i = 1; i <= m; i++) {
    f[i] = (f[i - 1] * i) % mod;
    invf[i] = fastpow(f[i], mod - 2);
  }
  cout << (((f[m] * invf[m - n + 1]) % mod) * invf[n - 1]) % mod << '\n';
  return 0;
}
