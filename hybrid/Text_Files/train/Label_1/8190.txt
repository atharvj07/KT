#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> v;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') v.push_back(i);
  }
  sort(v.begin(), v.end());
  int ans = -1;
  for (int i = 0; i < v.size() - k; i++) {
    int dist = v[i + k] - v[i];
    ans = max(ans, dist / 2);
  }
  cout << ans << endl;
  return 0;
}
