def min_time_to_eat_all_asterisks():
    n = int(input().strip())
    field = input().strip()

    # Find positions of Packmen and asterisks
    packmen_positions = []
    asterisks_positions = []
    for i, cell in enumerate(field):
        if cell == 'P':
            packmen_positions.append(i)
        elif cell == '*':
            asterisks_positions.append(i)

    # Binary search helper function
    def is_possible(time):
        asterisk_index = 0  # Index to track the current asterisk
        for packman in packmen_positions:
            # Find the farthest left asterisk that can be eaten
            left_most = packman
            while left_most > 0 and packman - left_most <= time:
                if packman - 2 * (left_most - asterisks_positions[asterisk_index]) <= time:
                    break
                left_most -= 1
            # Eat asterisks from left to right
            while asterisk_index < len(asterisks_positions) and asterisks_positions[asterisk_index] <= packman:
                asterisk_index += 1
            right_limit = packman + (time - (packman - left_most))
            while asterisk_index < len(asterisks_positions) and asterisks_positions[asterisk_index] <= right_limit:
                if asterisks_positions[asterisk_index] - packman > time:
                    break
                asterisk_index += 1
            if asterisk_index >= len(asterisks_positions):
                return True
        return False

    # Binary search to find the minimum time
    low, high = 0, n
    while low < high:
        mid = (low + high) // 2
        if is_possible(mid):
            high = mid
        else:
            low = mid + 1
    return low

print(min_time_to_eat_all_asterisks())
