def card_war(p1_cards, p2_cards):
  fights = 0
  while p1_cards and p2_cards:
    if p1_cards[0] > p2_cards[0]:
      p1_cards.extend(p2_cards[:2])
      p2_cards = p2_cards[2:]
    else:
      p2_cards.extend(p1_cards[:2])
      p1_cards = p1_cards[2:]
    fights += 1
  if p1_cards:
    return fights, 1
  elif p2_cards:
    return fights, 2
  else:
    return -1

n = int(input())
k1, *p1_cards = map(int, input().split())
k2, *p2_cards = map(int, input().split())
print(*card_war(p1_cards, p2_cards))
