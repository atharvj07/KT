n = int(input())
a = list(map(int, input().split()))
a.sort()
dp = [[0 for i in range(n)] for j in range(n)]
for i in range(n):
    dp[i][i] = a[i]
for d in range(1, n):
    for i in range(n - d):
        j = i + d
        if d & 1:
            dp[i][j] = max(a[i] + dp[i + 1][j], a[j] + dp[i][j - 1])
        else:
            dp[i][j] = min(a[i] + dp[i + 1][j], a[j] + dp[i][j - 1])
print(*dp[0])
