#include <cstdio>
#include <cstring>
#include <vector>
typedef long long ll;
const int MaxN=1e5+5;
int n;
int a[MaxN], pos[MaxN];
std::vector<int> rm;
void solve(){
  for(int i = 0; i < n; ++i){
    if(pos[a[i]] > pos[a[i+1]]){
      rm.emplace_back(a[i]);
    }
  }
}
int main(){
  scanf("%d", &n);
  for(int i = 0; i < n; ++i){
    scanf("%d", &a[i]);
    pos[a[i]] = i;
  }
  if(n <= 2){
    printf("0");
    return 0;
  }
  solve();
  memset(pos, 0, sizeof(pos));
  for(int i = 0; i < n; ++i){
    pos[a[i]] = i;
  }
  solve();
  printf("%d", rm.size());
  return 0;
}
