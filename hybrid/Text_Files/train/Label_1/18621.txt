#include<bits/stdc++.h>
using namespace std;
long long mod = 998244353;
long long power(long long base, long long p) {
    if (p == 0) return 1;
    if (p == 1) return base;
    long long ans = power(base, p / 2);
    ans = (ans * ans) % mod;
    if (p % 2) ans = (ans * base) % mod;
    return ans;
}
long long fact[1002];
long long choose(long long a, long long b) {
    return (fact[a] * power((fact[b] * fact[a - b]) % mod, mod - 2)) % mod;
}
int main() {
    long long n, k;
    cin >> n >> k;
    fact[0] = 1;
    for (int i = 1; i <= 1000; i++) {
        fact[i] = (i * fact[i - 1]) % mod;
    }
    long long ans = choose(2 * n, k);
    ans = (ans * power(2, n)) % mod;
    cout << ans << endl;
    return 0;
}
