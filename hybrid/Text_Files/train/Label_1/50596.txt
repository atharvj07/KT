import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            String[] nk = br.readLine().split(" ");
            int n = Integer.parseInt(nk[0]);
            int k = Integer.parseInt(nk[1]);
            char[] s = br.readLine().toCharArray();
            Arrays.sort(s);
            int[] freq = new int[26];
            for (int i = 0; i < n; i++) {
                freq[s[i] - 'a']++;
            }
            boolean found = false;
            int left = 0;
            int right = n - 1;
            while (left <= right && freq[s[left] - 'a'] == 0) {
                left++;
            }
            while (right >= 0 && freq[s[right] - 'a'] == k) {
                right--;
            }
            while (left <= right) {
                if (freq[s[left] - 'a'] == k) {
                    left++;
                } else if (freq[s[right] - 'a'] == 0) {
                    right--;
                } else if (s[left] == s[right]) {
                    int d = Math.min(k - freq[s[left] - 'a'], freq[s[right] - 'a']);
                    freq[s[left] - 'a'] += d;
                    freq[s[right] - 'a'] -= d;
                    left++;
                    right--;
                } else {
                    if (s[left] < s[right]) {
                        int d = Math.min(k - freq[s[left] - 'a'], k - freq[s[right] - 'a']);
                        freq[s[left] - 'a'] += d;
                        freq[s[right] - 'a'] -= d;
                        left++;
                    } else {
                        int d = Math.min(k - freq[s[right] - 'a'], freq[s[left] - 'a']);
                        freq[s[left] - 'a'] -= d;
                        freq[s[right] - 'a'] += d;
                        right--;
                    }
                }
            }
            boolean valid = true;
            for (int i = 0; i < 26; i++) {
                if (freq[i] % k != 0) {
                    valid = false;
                    break;
                }
            }
            if (valid) {
                StringBuilder sb = new StringBuilder();
                for (int i = 0; i < 26; i++) {
                    for (int j = 0; j < freq[i] / k; j++) {
                        sb.append((char) (i + 'a'));
                    }
                }
                System.out.println(sb.toString());
            } else {
                System.out.println(-1);
            }
        }
    }
}
