import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int crimes = 0, police = 0;

        while (n-- > 0) {
            int event = input.nextInt();
            if (event == -1) {
                if (police > 0) police--;
                else crimes++;
            } else {
                police += event;
            }
        }
        System.out.println(crimes);
    }
}
