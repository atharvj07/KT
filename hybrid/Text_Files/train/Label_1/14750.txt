#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 2003;

int n, k, row[MAXN], col[MAXN], ans;
vector<vector<int>> grid;

int main() {
  cin >> n >> k;
  grid.resize(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < n; j++) { grid[i][j] = (s[j] == 'B'); }
  }
  for (int i = 0; i + k - 1 < n; i++) {
    for (int j = 0; j + k - 1 < n; j++) {
      int cnt = 0;
      for (int ii = 0; ii < k; ii++) {
        for (int jj = 0; jj < k; jj++) {
          cnt += grid[i + ii][j + jj];
        }
      }
      if (cnt == 0) ans += 2;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      row[i] += grid[i][j];
      col[j] += grid[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    if (row[i] == 0) ans++;
    if (col[i] == 0) ans++;
  }
  cout << ans << endl;
  return 0;
}
