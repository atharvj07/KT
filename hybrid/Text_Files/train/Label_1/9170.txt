#include <iostream>
using namespace std;

int n, k;
int a[1002], b[1002];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    for (int i = 1; i <= n; i++)
        cin >> b[i];

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans += min(k, max(0, b[i]));
        k -= max(0, b[i]);
        b[i] = max(0, b[i] - a[i]);
    }

    k += ans; // remaining
    for (int i = 1; i <= n; i++)
        ans += min(k, b[i] / a[i]);
    cout << ans << endl;
    return 0;
}
