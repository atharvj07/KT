import collections

k, n = map(int, input().split())
S = [input() for i in range(k)]

c = collections.Counter()
for s in S:
    for i in range(n):
        for j in range(i+1, n):
            if s[i] != s[j]:
                c[(i, j, s[i], s[j])] += 1
                c[(i, j, s[j], s[i])] += 1

ans = ['a'] * n
for (i, j, a, b), f in c.items():
    if f % 2 == 1:
        print(-1)
        exit()
    if i != j or a != b:
        ans[i] = a
        ans[j] = b

print(''.join(ans))
