#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, s; cin >> n >> s;
  vector<pair<int, int>> a(n);
  for (auto& p : a) cin >> p.first >> p.second;
  sort(a.rbegin(), a.rend());
  int ans = 0;
  for (auto& p : a) {
    int t = max(ans, p.second);
    ans = t + s - p.first;
  }
  cout << ans << endl;
  return 0;
}
