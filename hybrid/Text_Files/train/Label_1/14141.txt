#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  if (s[0] != s[n - 1]) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> pos0, pos1;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      pos0.push_back(i);
    } else {
      pos1.push_back(i);
    }
  }
  int k = 0;
  vector<int> ans;
  for (int i = 0; i < pos0.size(); i++) {
    int j = i;
    while (j < pos0.size() && pos0[i] + pos0[j] == pos1[k] + pos0[k]) {
      j++;
      k++;
    }
    if (j > i) {
      ans.push_back(j - i + 1);
      for (int l = i; l <= j; l++) {
        cout << pos0[l] + 1 << " ";
      }
      cout << endl;
    }
  }
  if (ans.size() > 0) {
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
      cout << ans[i] << " ";
    }
    cout << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
