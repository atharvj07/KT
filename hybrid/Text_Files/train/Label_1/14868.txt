#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mxN = 1e3 + 5;
const ll inf = 1e9;

ll n, m, a[mxN], p[mxN], v[mxN], dp[20][1 << 10], g[mxN][mxN];
int main() {
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) cin >> v[i];
    for (ll i = 1, u, v; i <= m; i++) cin >> u >> v, g[u][v] = g[v][u] = 1;
    fill(p + 1, p + n + 1, inf);
    for (ll i = 1; i <= n; i++) {
        queue<ll> q;
        q.push(i);
        while (!q.empty()) {
            ll u = q.front();
            q.pop();
            if (p[u] != inf) continue;
            p[u] = 0;
            for (ll j = 1; j <= n; j++) {
                if (g[u][j] != 0) {
                    if (p[j] == inf) q.push(j);
                    p[u] += v[j];
                }
            }
        }
    }
    for (ll i = 0; i <= m; i++)
        for (ll j = 0; j < (1 << 10); j++) dp[i][j] = inf;
    dp[0][1] = 0;
    for (ll i = 0; i < m; i++) {
        for (ll j = 0; j < (1 << 10); j++) {
            if (dp[i][j] == inf) continue;
            for (ll k = 1; k <= n; k++) {
                if ((j & (1 << (k - 1))) == 0) {
                    dp[i + 1][j | (1 << (k - 1))] = min(dp[i + 1][j | (1 << (k - 1))], dp[i][j] + p[k]);
                }
            }
        }
    }
    cout << dp[m][(1 << n) - 1] << '\n';
}
