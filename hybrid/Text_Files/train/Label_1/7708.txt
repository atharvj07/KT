#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
int m, n, k, a[110][110], vis[110][110];
int dfs(int x, int y, int val) {
    if (x < 1 or x > n or y < 1 or y > m or vis[x][y] or a[x][y] != val) return 0;
    vis[x][y] = 1;
    return 1 + dfs(x + 1, y, val) + dfs(x, y + 1, val) + dfs(x - 1, y, val) + dfs(x, y - 1, val);
}
int32_t main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            cin >> a[i][j];
        }
    }
    int mn = 1e18;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            if (!vis[i][j]) {
                int area = dfs(i, j, a[i][j]);
                int cost = abs(area - a[i][j]);
                if (cost > k) mn = -1;
                else mn = min(mn, cost);
            }
        }
    }
    cout << mn << endl;
    return 0;
}
