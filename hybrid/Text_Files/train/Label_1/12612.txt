#include <iostream>
#include <string>
#include <vector>
using namespace std;
bool found(const string& s) {
    vector<int> loc(26, -1);
    for (int i = 0; i < s.size(); i++) {
        loc[s[i] - 'a'] = i;
        if (s[i] == 'd') {
            if (loc['e'] > 0 && loc['i'] > 0 && loc['h'] > 0) return true;
        }
    }
    return false;
}
int main() {
    string s;
    cin >> s;
    cout << (found(s) ? "YES" : "NO") << endl;
}
