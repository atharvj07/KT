#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> d(n);
  for (ll i = 0; i < n; i++) { cin >> d[i]; }
  vector<pair<ll, ll>> e(m);
  for (ll i = 0; i < m; i++) {
    ll u, v;
    cin >> u >> v;
    u--;
    v--;
    e[i] = {u, v};
  }
  vector<ll> deg(n, 0);
  for (auto x : e) { deg[x.first]++; }
  for (auto x : e) { deg[x.second]++; }
  for (ll i = 0; i < n; i++) {
    if (d[i] == -1 && deg[i] % 2 == 1) { cout << -1; return 0; }
    if (d[i] == 1 && deg[i] % 2 == 0) { cout << -1; return 0; }
  }
  vector<ll> ans;
  auto dfs = [&](auto &dfs, ll u, ll p = -1) -> void {
    deg[u]--;
    if (p != -1) deg[p]--;
    if (d[u] == -1 && deg[u] % 2 == 1) { ans.push_back(p); }
    for (auto x : e) {
      if (x.first == u && x.second != p) { dfs(dfs, x.second, u); }
      if (x.second == u && x.first != p) { dfs(dfs, x.first, u); }
    }
  };
  dfs(dfs, 0);
  if (ans.size() == 0) { cout << -1; } else {
    cout << ans.size() << "\n";
    for (ll x : ans) { cout << x + 1 << "\n"; }
  }
}
