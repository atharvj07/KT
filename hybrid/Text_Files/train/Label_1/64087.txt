n = int(input())
a = list(map(int, input().split()))

potions = []
health = 0
count = 0

for potion in a:
    if health + potion >= 0:
        health += potion
        count += 1
        potions.append(potion)
    else:
        potions.append(potion)
        potions.sort()
        min_potion = potions[0]
        if health + potion - min_potion >= 0:
            health += potion - min_potion
            potions.remove(min_potion)
        else:
            potions.remove(potion)

print(count)
