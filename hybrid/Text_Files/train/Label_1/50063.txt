import java.util.*;
public class Main {
    static int n;
    static List<Integer>[]g;
    static int[]sz;
    static int[]ans;
    static void dfs(int u,int p){
        sz[u]=1;
        int mx=-1,nxt=-1;
        for(int v:g[u]){
            if(v==p)continue;
            dfs(v, u);
            sz[u]+=sz[v];
            if(sz[v]>mx){
                mx=sz[v];
                nxt=v;
            }
        }
        if(n-sz[u]>mx){
            mx=n-sz[u];
            nxt=p;
        }
        ans[u]=nxt;
    }
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        n=sc.nextInt();
        g=new ArrayList[n+1];
        for(int i=0;i<=n;++i)g[i]=new ArrayList<>();
        for(int i=0;i<n-1;++i){
            int u=sc.nextInt();
            int v=sc.nextInt();
            g[u].add(v);
            g[v].add(u);
        }
        sz=new int[n+1];
        ans=new int[n+1];
        dfs(1, 0);
        for(int i=1;i<=n;++i){
            if(ans[i]==0)System.out.println(i);
        }
    }
}
