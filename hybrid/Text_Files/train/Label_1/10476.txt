#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> v(n + 1);
    while (m--) {
      int a, b;
      string s;
      cin >> a >> b >> s;
      v[a].push_back(b);
      v[b].push_back(a);
    }
    int ans = 0;
    vector<int> vis(n + 1, -1);
    for (int i = 1; i <= n; i++) {
      if (vis[i] != -1) continue;
      vis[i] = 0;
      int cnt0 = 1, cnt1 = 0;
      for (auto j : v[i]) {
        if (vis[j] == -1) {
          vis[j] = vis[i] ^ 1;
          if (vis[j] == 0)
            cnt0++;
          else
            cnt1++;
        } else if (vis[j] == vis[i]) {
          ans = -1;
          break;
        }
      }
      if (ans == -1) break;
      ans += max(cnt0, cnt1);
    }
    cout << ans << endl;
  }
  return 0;
}
