#include <bits/stdc++.h>
using namespace std;
pair<int, int> edge[1000005];
bool ok(int i, int j) {
    return edge[i].first > edge[j].second || edge[j].first > edge[i].second;
}
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        char c;
        cin >> c;
        if (c == '?') {
            int x, y;
            cin >> x >> y;
            cout << (edge[x].first < y || edge[x].second < y ? "NO" : "YES") << endl;
            continue;
        }
        int x, y;
        cin >> x >> y;
        edge[x] = make_pair(x, y);
        for (int j = 1; j <= 1000000; j++) {
            if (i > 0 && ok(j, x)) edge[j] = make_pair(max(edge[j].first, edge[x].first), max(edge[j].second, edge[x].second));
            if (i > 0 && ok(x, j)) edge[j] = make_pair(max(edge[j].first, edge[x].first), max(edge[j].second, edge[x].second));
        }
    }
}
