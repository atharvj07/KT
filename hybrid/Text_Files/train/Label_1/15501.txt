#include <bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  int a[n], b[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 1; i <= n; i++) {
    bool ok = 1;
    for (int j = 0; j < n; j++) {
      if (a[j] != i && b[j] != i) { ok = 0; }
    }
    if (ok) { cout << i << " "; }
  }
  cout << endl;
}
