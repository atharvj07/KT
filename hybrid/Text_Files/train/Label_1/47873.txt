import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        boolean[] lucky = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            int x = s.nextInt();
            if (x == 4 || x == 7) {
                lucky[i] = true;
            }
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (!lucky[i]) {
                for (int j = n - 1; j > i; j--) {
                    if (!lucky[j]) {
                        ans += j - i;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
