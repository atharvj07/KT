import sys
N = int(input().strip())
lst = []
rlst = []
for i in range(N):
    lst.append(list(map(int, input().strip().split())))
for i in range(N):
    x1 = min(lst[i][0], lst[i][2])
    y1 = min(lst[i][1], lst[i][3])
    x2 = max(lst[i][0], lst[i][2])
    y2 = max(lst[i][1], lst[i][3])
    for j in range(i+1, N):
        jx1 = min(lst[j][0], lst[j][2])
        jy1 = min(lst[j][1], lst[j][3])
        jx2 = max(lst[j][0], lst[j][2])
        jy2 = max(lst[j][1], lst[j][3])
        if jx1 <= x2 and jx2 >= x1 and jy1 <= y2 and jy2 >= y1:
            rlst.append(i)
            rlst.append(j)
#print(rlst)
if len(rlst) < N-1:
    print('YES')
    for i in range(N):
        if i not in rlst:
            print(1)
        else:
            if i%2 == 0:
                print(2)
            else:
                print(3)
else:
    print('NO')
