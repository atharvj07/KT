import queue
T = int(input())
while T:
    T -= 1
    N, M, R, C = map(int, input().split())
    vis = [[0] * (M + 2) for i in range(N + 2)]
    dis = [[0] * (M + 2) for i in range(N + 2)]
    que = queue.Queue()
    # enqueue (x, y)
    que.put((R, C))
    vis[R][C] = 1
    while not que.empty():
        x, y = que.get()
        # move right
        if y + 1 <= M and not vis[x][y + 1]:
            dis[x][y + 1] = dis[x][y] + 1
            vis[x][y + 1] = 1
            que.put((x, y + 1))
        # move down
        if x + 1 <= N and not vis[x + 1][y]:
            dis[x + 1][y] = dis[x][y] + 1
            vis[x + 1][y] = 1
            que.put((x + 1, y))
        # move up
        if x - 1 >= 1 and not vis[x - 1][y]:
            dis[x - 1][y] = dis[x][y] + 1
            vis[x - 1][y] = 1
            que.put((x - 1, y))
        # move left
        if y - 1 >= 1 and not vis[x][y - 1]:
            dis[x][y - 1] = dis[x][y] + 1
            vis[x][y - 1] = 1
            que.put((x, y - 1))
    ans = 0
    for i in range(1, N + 1):
        for j in range(1, M + 1):
            ans = max(ans, dis[i][j])
    print(ans)
