#include <iostream>
#include <vector>
#include <string>

using namespace std;

bool dfs(int pos, int k, const string &s, vector<bool> &used, string &ans) {
    if (pos == s.size()) {
        return ans.size() == k;
    }
    if (s[pos] != '?' && s[pos] != '*') {
        ans += s[pos];
        if (dfs(pos + 1, k, s, used, ans)) return true;
        ans.pop_back();
    }
    if (s[pos] == '?') {
        if (pos > 0 && s[pos - 1] != '?' && s[pos - 1] != '*') {
            ans += s[pos - 1];
            if (dfs(pos + 1, k, s, used, ans)) return true;
            ans.pop_back();
        }
        if (dfs(pos + 1, k, s, used, ans)) return true;
        if (pos > 0 && s[pos - 1] != '?' && s[pos - 1] != '*') {
            ans += s[pos - 1];
            ans += s[pos - 1];
            if (dfs(pos + 1, k, s, used, ans)) return true;
            ans.pop_back();
            ans.pop_back();
        }
    }
    if (s[pos] == '*') {
        if (pos > 0 && s[pos - 1] != '?' && s[pos - 1] != '*') {
            if (dfs(pos + 1, k, s, used, ans)) return true;
            ans += s[pos - 1];
            if (dfs(pos + 1, k, s, used, ans)) return true;
            ans.pop_back();
        } else {
            if (dfs(pos + 1, k, s, used, ans)) return true;
        }
    }
    return false;
}

int main() {
    string s;
    int k;
    cin >> s >> k;
    vector<bool> used(s.size(), false);
    string ans = "";
    if (dfs(0, k, s, used, ans)) {
        cout << ans << endl;
    } else {
        cout << "Impossible" << endl;
    }
    return 0;
}
