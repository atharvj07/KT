n, k = map(int, input().split())
coefficients = list(map(int, input().split()))

# Calculate P(2)
P_2 = sum(coef * (2 ** i) for i, coef in enumerate(coefficients))

# Count the number of ways to change one coefficient to make Q(2) = 0
ways = 0
for i, coef in enumerate(coefficients):
    # Calculate the difference needed to make Q(2) = 0 if we change this coefficient
    diff = P_2 - coef * (2 ** i)
    # Check if the new coefficient is valid
    if diff % (2 ** i) == 0:
        new_coef = -diff // (2 ** i)
        if abs(new_coef) <= k:
            ways += 1

print(ways)
