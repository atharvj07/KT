#include <iostream>
using namespace std;
int main() {
    int q;
    bool flag;
    cin >> q;
    while (q--) {
        string s, t;
        cin >> s >> t;
        flag = true;
        for (int i = 0; i < s.size(); i++) {
            if (s[i] != t[i]) flag = !flag;
        }
        cout << (flag ? "YES" : "NO") << '\n';
    }
}
