import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		char[] s = sc.nextLine().toCharArray();
		for (int i = 0; i < s.length; i++) {
			char c = s[i];
			if (c >= 'a' && c <= 'z') {
				c -= 32;
			} else if (c >= 'A' && c <= 'Z') {
				c += 32;
			}
			s[i] = c;
		}
		System.out.println(String.valueOf(s));
	}
}
