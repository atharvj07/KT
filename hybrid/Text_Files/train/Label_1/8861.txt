#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  ll t;
  cin >> t;
  for (ll q = 0; q < t; q++) {
    ll n;
    cin >> n;
    vector<ll> v(n);
    ll sum = 0;
    for (ll i = 0; i < n; i++) {
      cin >> v[i];
      sum += v[i];
    }
    ll k = sum / n;
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
      ans += ((sum - v[i]) % (n - 1) == 0 and (sum - v[i]) / (n - 1) == k);
    }
    cout << ans << endl;
  }
  return 0;
}
