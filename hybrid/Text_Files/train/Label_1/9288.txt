#include <cstdio>
#include <cstdlib>

int main() {
  int n, m, k;
  scanf("%d %d %d", &n, &m, &k);
  char a[n][m + 1];
  for (int i = 0; i < n; i++) scanf("%s", a[i]);
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '.')
        cnt++;
      else
        cnt = 0;
      if (cnt == k) ans++;
    }
  }
  for (int j = 0; j < m; j++) {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (a[i][j] == '.')
        cnt++;
      else
        cnt = 0;
      if (cnt == k) ans++;
    }
  }
  printf("%lld\n", ans);
  return 0;
}
