#include<bits/stdc++.h>
using namespace std;
#define F first
#define S second
#define mp make_pair

int a[200001],b[200001],n,k;
map<int,int> m;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	cin>>n;
	for(int i=0;i<n;i++){
		cin>>a[i];
		m[a[i]]++;
	}
	int p=0;
	for(auto x: m){
		int l=x.S/2,v=x.F;
		a[p++]=v;
		if(x.S%2) b[k++]=v;
		for(int i=0;i<l;i++) b[k++]=v+1;
	}
	cout<<k<<'\n';
	for(int i=0;i<k;i++) cout<<b[i]<<' ';
}
