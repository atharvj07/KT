import math

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [[i for i in range(1, m + 1)] for _ in range(n)]
ok = True

for i in range(n):
    for j in range(m):
        if a[i][j] != b[i][j]:
            print("NO")
            ok = False
            break
    if not ok:
        break

for i in range(m):
    col = [a[j][i] for j in range(n)]
    if col != b[0][:]:
        print("NO")
        ok = False
        break
    if not ok:
        break

if ok:
    print("YES")
