def time_to_seconds(t):
    return int(t[:2]) * 3600 + int(t[3:5]) * 60 + int(t[6:])

def solve(n, M, T, queries):
    times = [time_to_seconds(q) for q in queries]
    users = [0] * n
    current_users = []
    user_id = 0
    max_users = 0
    
    for i, start in enumerate(times):
        # Remove users whose online period has ended
        current_users = [u for u in current_users if u[1] >= start]
        if len(current_users) < M:
            # If max users online not reached or no users online, assign new user ID
            user_id += 1
            users[i] = user_id
            current_users.append((user_id, start + T - 1))
            max_users = max(max_users, len(current_users))
        else:
            # Assign ID of a user whose online period is closest to ending
            closest_end_user = min(current_users, key=lambda x: x[1])
            users[i] = closest_end_user[0]
            # Update the end time for this user
            for j in range(len(current_users)):
                if current_users[j][0] == closest_end_user[0]:
                    current_users[j] = (closest_end_user[0], start + T - 1)
                    break
    if max_users == M:
        return [str(user_id)] + [str(u) for u in users]
    else:
        return ["No solution"]

n, M, T = map(int, input().split())
queries = [input().strip() for _ in range(n)]
result = solve(n, M, T, queries)
print("\n".join(result))
