#include<bits/stdc++.h>
using namespace std;

int n;
int x[1005], y[1005];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> x[i] >> y[i];
    int minx = 1e9, miny = 1e9, maxx = -1e9, maxy = -1e9;
    for (int i = 1; i <= n; i++) {
        minx = min(minx, x[i]);
        maxx = max(maxx, x[i]);
        miny = min(miny, y[i]);
        maxy = max(maxy, y[i]);
    }
    cout << max((maxx - minx), (maxy - miny)) * max((maxx - minx), (maxy - miny)) << endl;
    return 0;
}
