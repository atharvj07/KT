#include <bits/stdc++.h>
using namespace std;
const int mxn = 110;
char s[mxn];
struct operation {
    int l, r, f, t;
};
vector<operation> op;
int main() {
    int n, m;
    cin >> n >> m;
    cin >> s;
    for (int i = 0; i < m; i++) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        op.push_back({a, b, c, d});
    }
    for (int i = 0; i < m; i++) {
        for (int j = op[i].l; j <= op[i].r; j++) {
            if (s[j] == op[i].f) s[j] = op[i].t;
        }
    }
    cout << s << endl;
    return 0;
}
