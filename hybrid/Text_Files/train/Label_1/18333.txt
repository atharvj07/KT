#include <bits/stdc++.h>
using namespace std;
const int N = 15, M = 5005;
int n, a[N][M], b[N][M], c[N], s[N];
long long tau[N], sum;
bool found;
void dfs(int ix, long long sum1) {
  if (ix > n) {
    if (sum1 == sum * n) {
      found = 1;
      for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
          if (a[i][c[j]] == b[j][c[j]]) { cout << a[i][c[j]] << " " << j << "\n"; }
      exit(0);
    }
    return;
  }
  for (int i = 1; i <= s[ix]; i++) {
    c[ix] = i;
    dfs(ix + 1, sum1 + tau[ix] - a[ix][i]);
  }
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> s[i];
    for (int j = 1; j <= s[i]; j++) cin >> a[i][j], b[i][j] = a[i][j];
    sort(b[i] + 1, b[i] + s[i] + 1);
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= s[i]; j++) { tau[i] += b[i][j]; }
  sum = tau[1] / s[1];
  dfs(1, 0);
  if (!found) puts("No");
  return 0;
}
