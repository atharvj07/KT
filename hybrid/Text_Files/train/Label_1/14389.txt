#include <iostream>
#include <vector>
#include <queue>

using namespace std;

struct cell {
    int x, y, d;
    cell(int x_, int y_, int d_): x(x_), y(y_), d(d_) {}
};

const int dx[] = {0, 1, 0, -1};
const int dy[] = {1, 0, -1, 0};

int main() {
    int n, m, p;
    cin >> n >> m >> p;
    vector<int> s(p);
    for (int &i: s) cin >> i;
    vector<vector<char>> a(n, vector<char>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    vector<vector<int>> dist(n, vector<int>(m, 1e9));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] != '.' && a[i][j] != '#') {
                dist[i][j] = 0;
                queue<cell> q;
                q.push(cell(i, j, 0));
                while (!q.empty()) {
                    cell u = q.front(); q.pop();
                    for (int k = 0; k < 4; k++) {
                        int x = u.x + dx[k];
                        int y = u.y + dy[k];
                        int d = u.d + 1;
                        if (x >= 0 && y >= 0 && x < n && y < m && a[x][y] != '#' && d <= s[a[x][y] - '1'] && dist[x][y] > d) {
                            dist[x][y] = d;
                            q.push(cell(x, y, d));
                        }
                    }
                }
            }
        }
    }
    vector<int> ans(p, 0);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (dist[i][j] < 1e9) ans[a[i][j] - '1']++;
        }
    }
    for (int i: ans) cout << i << ' ';
    cout << '\n';
    return 0;
}
