import sys
input = sys.stdin.readline

n = int(input())
a = [0] * n
b = list(map(int, input().split()))
ans = 0

for i in range(n):
    if a[i] != b[i]:
        if b[i] > a[i]:
            ans += b[i] - a[i]
            for j in range(i, n):
                a[j] += b[i] - a[i]
        else:
            ans += a[i] - b[i]
            for j in range(i, n):
                a[j] -= a[i] - b[i]

print(ans)
