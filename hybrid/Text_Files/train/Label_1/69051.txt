import sys
input=sys.stdin.readline

n,m,q=map(int,input().split())
s,t=input(),input()
z=[0]*n
for l in range(1,n):
    x=z[l-1]
    while x>0 and s[x]!=s[l]:
        x=z[x-1]
    if s[x]==s[l]:
        x+=1
    z[l]=x
for i in range(q):
    l,r=map(int,input().split())
    l-=1;r-=1
    x=z[r]
    while x>0 and s[x]!=t[0]:
        x=z[x-1]
    if s[x]==t[0]:
        x+=1
    cnt=0
    if l>0:
        cnt=z[l-1]
    print(max(0,x-(l>0)*cnt))
