#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long ans = 0;
    vector<int> order;
    for (int i = 0; i < n; i++) {
        ans += 1LL * (a[i] * i);
        order.push_back(i + 1);
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        cout << order[i] << " ";
    }
    cout << endl;
    return 0;
}
