n, k = map(int, input().split())
absurdities = list(map(int, input().split()))

# Calculate prefix sums to easily compute sums of segments
prefix_sums = [0]
for absurdity in absurdities:
    prefix_sums.append(prefix_sums[-1] + absurdity)

# Function to compute the sum of absurdities in a segment
def segment_sum(start, end):
    return prefix_sums[end] - prefix_sums[start]

# First, find the maximum segment sum for each possible starting position
max_segment_sums = []
for i in range(n - k + 1):
    max_segment_sums.append((segment_sum(i, i + k), i + 1))

# Compute the maximum pairs of segments
max_total = 0
best_pair = (0, 0)
for i in range(len(max_segment_sums) - k):
    for j in range(i + k, len(max_segment_sums)):
        total = max_segment_sums[i][0] + max_segment_sums[j][0]
        if total > max_total:
            max_total = total
            best_pair = (max_segment_sums[i][1], max_segment_sums[j][1])

print(best_pair[0], best_pair[1])
