import java.util.*;

public class Main {
	static long n, I, ans = Long.MAX_VALUE;
	static long[] a, cnt, arr;
	
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		n = scan.nextLong();
		I = scan.nextLong();
		a = new long[(int) n];
		cnt = new long[(int) n];
		arr = new long[(int) n];
		for (int i = 0; i < n; i++) { a[i] = scan.nextLong(); }
		Arrays.sort(a);
		int K = 0;
		for (int i = 0; i < n; i++) {
			if (i > 0 && a[i] != a[i - 1]) K++;
			cnt[K]++;
		}
		for (int i = 0; i < K; i++) { a[i] = cnt[i]; }
		for (int i = 0; i < K; i++) { arr[i] = a[i]; }
		for (int i = 1; i < K; i++) { arr[i] += arr[i - 1]; }
		for (int i = 0; i < K; i++) {
			long low = i == 0 ? 0 : arr[i - 1];
			long high = arr[i];
			long cur = low + high - cnt[i];
			if (cur >= I) {
				ans = Math.min(ans, low);
				continue;
			}
			long mid = 0;
			long L = low, R = high;
			while (L <= R) {
				mid = (L + R) / 2;
				long cur2 = cnt[i] - mid * (i + 1);
				if (cur2 >= I) {
					R = mid - 1;
					ans = Math.min(ans, mid);
				} else { L = mid + 1; }
			}
		}
		System.out.println(ans);
	}
}
