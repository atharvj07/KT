def main():
    t = int(input())
    for _ in range(t):
        px, py = map(int, input().split())
        s = str(input())
        x, y = 0, 0
        pos = True
        for c in s:
            if c == 'U': y += 1
            elif c == 'D': y -= 1
            elif c == 'R': x += 1
            elif c == 'L': x -= 1
            if x == px and y == py:
                pos = False
                break
        if pos: print("YES")
        else: print("NO")
main()
