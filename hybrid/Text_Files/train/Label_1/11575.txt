#include <iostream>

using namespace std;

const int MAXN = 110;

int main() {
  int n;
  cin >> n;

  bool chessboard[MAXN][MAXN];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { chessboard[i][j] = false; }
  }

  int cnt = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if ((i + j) % 2 == 0) {
        chessboard[i][j] = true;
        cnt++;
      }
    }
  }

  if (cnt == n * n / 2) {
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) { cout << "W"; }
      cout << endl;
    }
  } else {
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        if ((i + j) % 2 == 0)
          cout << "W";
        else
          cout << "B";
      }
      cout << endl;
    }
  }
  return 0;
}
