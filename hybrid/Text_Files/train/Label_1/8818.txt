#include <iostream>
#include <vector>
using namespace std;
#define MOD 1000000007LL

int pow(int a, int b){
    int ans = 1;
    for(int i = 0; i < b; i++){
        ans = (1LL * ans * a) % MOD;
    }
    return ans;
}

int main(){
    int n, k;
    cin >> n >> k;
    
    int ans = pow(n - 1, k);
    ans = (1LL * ans * (n - 1)) % MOD;
    ans = (1LL * ans * pow(n - 1, n - 1 - k)) % MOD;

    cout << ans << endl;
    
    return 0;
}
