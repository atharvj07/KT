import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int w = scanner.nextInt();
        int h = scanner.nextInt();
        int n = scanner.nextInt();

        int cableX1 = scanner.nextInt();
        int cableY1 = scanner.nextInt();
        int cableX2 = scanner.nextInt();
        int cableY2 = scanner.nextInt();

        int[][] table = new int[h + 1][w + 1];

        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();

            int t = scanner.nextInt();
            String pattern = scanner.nextLine().trim();

            for (int j = 0; j < t % pattern.length(); j++) {
                table[y][x]++;

                switch (pattern.charAt(j)) {
                    case 'U':
                        y--;
                        break;
                    case 'D':
                        y++;
                        break;
                    case 'L':
                        x--;
                        break;
                    case 'R':
                        x++;
                        break;
                }
            }
        }

        int min = Integer.MAX_VALUE;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if ((i == cableY1 && j == cableX1) || (i == cableY2 && j == cableX2)) {
                    continue;
                }

                int count = 0;
                if (i == cableY1) {
                    count += table[i][j];
                }
                if (i == cableY2) {
                    count += table[i][j];
                }
                if (j == cableX1) {
                    count += table[i][j];
                }
                if (j == cableX2) {
                    count += table[i][j];
                }

                if (count < min) {
                    min = count;
                }
            }
        }

        System.out.println(min);
    }
}
