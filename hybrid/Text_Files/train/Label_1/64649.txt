def max_sensors(n, y1, sensors1, m, y2, sensors2):
    def gcd(a, b):
        while b:
            a, b = b, a % b
        return a
    
    max_count = 1
    for base in range(2, 33):
        period = (2 ** base) * (y2 - y1)
        groups = {}
        for x in sensors1 + sensors2:
            g = x % period
            if g in groups:
                groups[g] += 1
            else:
                groups[g] = 1
        for _, count in groups.items():
            max_count = max(max_count, count)
    return max_count

# Input read
n, y1 = map(int, input().split())
sensors1 = list(map(int, input().split()))
m, y2 = map(int, input().split())
sensors2 = list(map(int, input().split()))

# Function call and output
print(max_sensors(n, y1, sensors1, m, y2, sensors2))
