#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  int n;
  cin >> n;
  vector<pair<long long, long long>> c(n);
  for (auto &e : c) { cin >> e.first; }
  for (auto &e : c) { cin >> e.second; }
  sort(c.begin(), c.end());
  long long ans = 0;
  for (int i = 0; i < n; ++i) {
    auto [a, b] = c[i];
    int64_t l = i, r = n - 1, pos = -1;
    while (l <= r) {
      int mid = (l + r) / 2;
      if (a + c[mid].first <= b + c[mid].second) {
        pos = mid;
        l = mid + 1;
      } else
        r = mid - 1;
    }
    if (pos == -1)
      ans += n - 1;
    else
      ans += n - 1 - pos;
  }
  cout << ans << endl;
}
