#include <bits/stdc++.h>
using namespace std;
int n;
char m[501][501];
bool vis[501][501];
int dx[] = {0, 0, -1, 1, 1, -1, 1, -1};
int dy[] = {-1, 1, 0, 0, 1, -1, -1, 1};
inline bool ok(int x, int y) { return x > 0 && x <= n && y > 0 && y <= n && m[x][y] == 'X'; }
int dfs(int x, int y) {
  vis[x][y] = true;
  int cnt = ok(x - 1, y - 1) + ok(x - 1, y + 1) + ok(x + 1, y - 1) + ok(x + 1, y + 1);
  for (int i = 0; i < 8; i++) {
    int nx = x + dx[i], ny = y + dy[i];
    if (ok(nx, ny) && !vis[nx][ny]) cnt += dfs(nx, ny);
  }
  return cnt;
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> m[i] + 1;
  int ans = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (!vis[i][j] && m[i][j] == 'X') ans += dfs(i, j) != 0;
    }
  cout << ans << endl;
  return 0;
}
