import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int round = 1;
        while (n > 1) {
            if (a % 2 == 1 ^ b % 2 == 1) {
                round++;
                break;
            }
            if (a % 2 == 1) {
                a++;
            }
            if (b % 2 == 1) {
                b++;
            }
            n /= 2;
            a /= 2;
            b /= 2;
            round++;
        }
        System.out.println(n == 1 ? "Final!" : round);
    }
}
