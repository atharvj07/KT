def find_numbers_with_m_zeroes(m):
    def count_trailing_zeroes(n):
        count = 0
        while n >= 5:
            n //= 5
            count += n
        return count

    left, right = 1, 5*m
    while left < right:
        mid = (left + right) // 2
        if count_trailing_zeroes(mid) < m:
            left = mid + 1
        else:
            right = mid

    first = left
    right = 5*m
    while left < right:
        mid = (left + right) // 2
        if count_trailing_zeroes(mid) <= m:
            left = mid + 1
        else:
            right = mid

    last = left
    if count_trailing_zeroes(first) != m:
        return [0]
    else:
        return [last-first] + list(range(first, last))

m = int(input())
result = find_numbers_with_m_zeroes(m)
print(result[0])
if result[0] != 0:
    print(' '.join(map(str, result[1:])))
