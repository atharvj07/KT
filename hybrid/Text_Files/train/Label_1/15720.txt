#include <cstdio>

int main() {
    int n, a[101][101];
    scanf("%d", &n);
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= n; ++j) scanf("%d", &a[i][j]);
    int ans = 0;
    for (int i = 1; i <= n; ++i) {
        bool good = true;
        for (int j = 1; j <= n; ++j)
            if (i != j && a[i][j] != 0) good = false;
        if (good) ++ans;
    }
    printf("%d\n", ans);
    for (int i = 1; i <= n; ++i)
        if (a[i][i] == -1) printf("%d ", i);
    return 0;
}
