#include <cstdio>
#include <cstring>

char inp[51];
char err[]="No\n";
char ok[]="Yes\n";
char s[51];

int chk(int pos) {
  int fs[26];
  memset(fs, 0, sizeof(fs));
  int sum=0;
  for (int i=pos; i>=0; ) {
    if (s[i]=='?') ++i;
    else if (fs[s[i]-'a']) return 0;
    else fs[s[i]-'a']=1, ++sum, --i;
    if (sum==7) return 1;
  }
  return 0;
}

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, i, j;
    scanf("%d", &n);
    scanf("%s", inp);
    for (i=n-1, j=0; i>=0; --i) {
      s[j++]=inp[i];
    }
    s[n]=0;
    int flag=0;
    for (i=0; i+6<n; ++i) {
      if (s[i]=='?' && chk(i)) {
        flag=1;
        s[i+0]='a', s[i+1]='b', s[i+2]='a', s[i+3]='c', s[i+4]='a', s[i+5]='b', s[i+6]='a';
        break;
      }
    }
    if (flag) {
      printf("%s%s", ok, s);
    } else {
      printf("%s", err);
    }
  }
  return 0;
}
