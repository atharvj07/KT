n,m,k=map(int,input().split())
l=[list(input())for i in range(n)for j in range(m)]
ans=0
for i in range(n):
 for j in range(m):
  for h in range(k):
   if l[i][j][h]=='1':
    flag=0
    if i>0:
     x=[l[i-1][j][l:l+3]for l in range(k-2)]
     for t in x:
      if t[0]=='1'or t[1]=='1':flag=1
    if j>0:
     x=[l[i][j-1][l:l+3]for l in range(k-2)]
     for t in x:
      if t[0]=='1'or t[2]=='1':flag=1
    if h>0:
     x=[[l[i][j][l],l[i][j+1][l],l[i][j+2][l]]for l in range(k-2)]
     for t in x:
      if t[0]=='1'or t[2]=='1':flag=1
    if not flag:ans+=1
print(ans)
