def area(x1, y1, x2, y2, x3, y3):
    return abs((x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2)) / 2.0)

n = int(input())
points = [list(map(int, input().split())) for _ in range(n)]

max_area = 0

for i in range(n):
    for j in range(i+1, n):
        for k in range(j+1, n):
            for l in range(k+1, n):
                a, b, c, d = points[i], points[j], points[k], points[l]
                area1 = area(a[0], a[1], b[0], b[1], c[0], c[1])
                area2 = area(a[0], a[1], c[0], c[1], d[0], d[1])
                area3 = area(a[0], a[1], b[0], b[1], d[0], d[1])
                area4 = area(b[0], b[1], c[0], c[1], d[0], d[1])
                max_area = max(max_area, area1 + area2, area2 + area3, area3 + area4, area1 + area4, area1 + area3, area2 + area4)

print(f'{max_area:.9f}')
