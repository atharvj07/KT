#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, x;
  cin >> n;
  vector<int> a(n + 1);
  vector<int> dp(n + 1), ans(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    dp[i] = 1;
  }
  int mx = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      if (a[i] < a[j]) {
        dp[j] = max(dp[j], dp[i] + 1);
      }
    }
    mx = max(mx, dp[i]);
  }
  for (int i = n; i >= 1; i--) {
    if (dp[i] == mx) {
      ans[i] = 3;
      mx--;
    } else {
      for (int j = i - 1; j >= 1; j--) {
        if (a[i] > a[j] && dp[j] + 1 == dp[i]) {
          if (ans[j] == 3) {
            ans[i] = 2;
          } else {
            ans[i] = 1;
          }
          break;
        }
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    if (ans[i] == 0) {
      ans[i] = 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << ans[i];
  }
  cout << endl;
  return 0;
}
