import heapq

n,m,k=map(int,input().split())
x=[int(i) for i in input().split()]

adj=[[] for i in range(n+1)]
for i in range(m):
    u,v,w=map(int,input().split())
    adj[u].append((w,v))
    adj[v].append((w,u))

def dijkstra(s,dis):
    q=[(0,s)]
    dis[s]=0
    while q:
        cnt,a=heapq.heappop(q)
        if cnt>dis[a]: continue
        for w,b in adj[a]:
            if dis[a]+w<dis[b]:
                dis[b]=dis[a]+w
                heapq.heappush(q,(dis[b],b))

ans=[0 for i in range(k)]
for i in range(k):
    dis=[int(1e9)]*(n+1)
    dijkstra(x[i],dis)
    mx=-1
    for j in range(k):
        if i==j: continue
        mx=max(mx,dis[x[j]])
    ans[i]=mx

print(*ans)
