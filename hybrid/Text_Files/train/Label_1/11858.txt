#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, cur = 0, nxt, pre = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    nxt = i;
    cin >> cur;
    if (cur < pre) {
      cout << "NO" << endl;
      return 0;
    }
    if (nxt != 0 and cur == pre) cout << "NO" << endl, exit(0);
    pre = cur;
  }
  cout << "YES" << endl;
  return 0;
}
