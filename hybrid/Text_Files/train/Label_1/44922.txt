import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong(), a = sc.nextLong(), b = sc.nextLong();
        long[] arr = new long[(int) n];
        long g = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
            g = gcd(g, arr[i]);
        }
        if (g != 1) {
            System.out.println(0);
            return;
        }
        long ans = b;
        for (int i = 0; i < n; i++) {
            for (long d = 2; (d * d) <= arr[i]; d++) {
                if (arr[i] % d == 0) {
                    int cnt = 0;
                    while (arr[i] % d == 0) {
                        arr[i] /= d;
                        cnt++;
                    }
                    if (cnt % 2 != 0) {
                        ans += Math.min(cnt * b, a + (cnt - 1) * b);
                        break;
                    }
                }
            }
            if (arr[i] > 1) {
                ans += Math.min(b, a);
            }
        }

        System.out.println(ans);
    }

    public static long gcd(long a, long b) {
        while (b != 0) {
            long t = a % b;
            a = b;
            b = t;
        }
        return a;
    }
}
