def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, x, a):
    MOD = 1000000007
    min_a = min(a)
    t = sum(pow(x, ai - min_a, MOD) for ai in a) % MOD
    s = pow(x, min_a, MOD) * n % MOD
    return gcd(s, t) % MOD

n, x = map(int, input().split())
a = list(map(int, input().split()))
print(solve(n, x, a))
