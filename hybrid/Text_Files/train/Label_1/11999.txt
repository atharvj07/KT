#include <iostream>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        long long a, b;
        cin >> a >> b;
        if(a > b)
        {
            cout << (a - b + 4) / 5 << endl;
        }
        else
        {
            cout << (b - a + 4) / 5 << endl;
        }
    }
    return 0;
}
