import java.util.*;
class Main {
    static class Pair<A, B> {
        A first;
        B second;
        public Pair(A first, B second) {
            this.first = first;
            this.second = second;
        }
    }
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int A = input.nextInt();
        int B = input.nextInt();
        char[][] pic = new char[A][B];
        for (int i = 0; i < A; i++)
            pic[i] = input.next().toCharArray();
        ArrayList<Pair<Integer, Integer>> ans = new ArrayList<>();
        for (int X = 1; X <= A; X++) {
            for (int Y = 1; Y <= B; Y++) {
                if (A % X == 0 && B % Y == 0) {
                    boolean ok = true;
                    for (int x = 0; x + X <= A; x++) {
                        for (int y = 0; y + Y <= B; y++) {
                            for (int dx = 0; dx + X <= A; dx++) {
                                for (int dy = 0; dy + Y <= B; dy++) {
                                    if (x != dx || y != dy) {
                                        char[][] f = new char[X][Y];
                                        char[][] s = new char[X][Y];
                                        for (int i = 0; i < X; i++)
                                            for (int j = 0; j < Y; j++) {
                                                f[i][j] = pic[x + i][y + j];
                                                s[i][j] = pic[dx + i][dy + j];
                                            }
                                        if (Arrays.deepEquals(f, s))
                                            ok = false;
                                    }
                                }
                            }
                        }
                    }
                    if (ok)
                        ans.add(new Pair<>(X, Y));
                }
            }
        }
        System.out.println(ans.size());
        int minX = Integer.MAX_VALUE;
        int minY = Integer.MAX_VALUE;
        int minA = A * B;
        for (Pair<Integer, Integer> pr : ans) {
            int X = pr.first;
            int Y = pr.second;
            if (X * Y < minA) {
                minA = X * Y;
                minX = X;
                minY = Y;
            } else if (X * Y == minA) {
                if (X < minX) {
                    minX = X;
                    minY = Y;
                }
            }
        }
        System.out.println(minX + " " + minY);
    }
}
