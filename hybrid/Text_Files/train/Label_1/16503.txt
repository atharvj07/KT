#include <bits/stdc++.h>
using namespace std;  
int main() {
    int q;
    cin >> q;
    while (q--) {
        string s, t;
        cin >> s >> t;
        if (s == t) cout << "YES\n";
        else if (s[0] != t[0]) cout << "NO\n";
        else {
            bool ok = 1;
            for (int i = 1; i < s.size(); ++i) if (s[i] == t[i]) ok = 0;
            if (ok) cout << "NO\n";
            else cout << "YES\n";
        }
    }
}
