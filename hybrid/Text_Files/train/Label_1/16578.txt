#include <bits/stdc++.h>
using namespace std;
const int maxn = 100000 + 5;
int par[maxn], sz[maxn];
int get(int x) { return (par[x] == x ? x : par[x] = get(par[x])); }
void merge(int x, int y) {
  x = get(x), y = get(y);
  if (x != y) { par[x] = y, sz[y] += sz[x]; }
}
int main() {
  int n;
  cin >> n;
  iota(par, par + n, 0), fill(sz, sz + n, 1);
  int lst = 0;
  map<int, vector<int>> pos;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    pos[x].push_back(i);
  }
  int ans = 0;
  for (auto i : pos) {
    for (int j : i.second) {
      if (j > 0) merge(j, j - 1);
      ans += sz[get(j)];
    }
    lst = (i.second).back();
  }
  cout << ans << '\n';
}
