n,m=map(int,input().split())
s=input()
x=[]
for i in range(m):
    x.append(int(s[i]==')'))
x.append(0)
stack=[]
for i in range(m-1,-1,-1):
    if(x[i]==0 and len(stack)>0):
        stack.pop()
    else:
        stack.append(i)
ans=(2**(n-m)-1)%1000000007
for i in range(len(stack)):
    idx=stack[i]
    ans=(ans*(i+1))%1000000007
    ans=(ans*(n-idx))%1000000007
print(ans)
