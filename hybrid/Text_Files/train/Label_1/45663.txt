import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong(), m = scanner.nextLong(), k = scanner.nextLong();
        long row = (k + m - 1) / m;
        long col = k % m;
        if (col == 0) {
            col = m;
        }
        System.out.println(row * col);
    }
}
