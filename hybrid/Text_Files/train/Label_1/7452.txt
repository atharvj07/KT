#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
const int MX = 2005;
ll bpow(ll b, ll e) {
  if (!e) return 1;
  ll r = bpow(b, e / 2);
  r = (r * r) % MOD;
  if (e & 1) r = (r * b) % MOD;
  return r;
}
int32_t main() {
  ll m, d, a, b;
  cin >> m >> d >> a >> b;
  ll ans = 0;
  string s = to_string(m);
  for (char c : s) {
    ll cur = c - '0';
    if (cur == d || cur == 0) {
      ans = (ans + (b - a + 1) / m) % MOD;
    }
  }
  cout << ans << "\n";
}
