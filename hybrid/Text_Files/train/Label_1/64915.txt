n=int(input())
a=[int(x) for x in input().split()]
tree=list([0]*n for i in range(n))
for i in range(n-1):
 tree[int(input().split()[0])-1][int(input().split()[1])-1]=1
 tree[int(input().split()[1])-1][int(input().split()[0])-1]=1
for i in range(n):
 for j in range(n):
  if tree[i][j]==1:
   tree[i][j]=tree[j][i]=2-min(i,j)%2
m=0
for i in range(n):
 s=0
 for j in range(n):
  s+=tree[i][j]*a[j]
 if s>m:
  m=s
print(m)
