MOD = 1000000007

def dp(n, k, d, memo, is_limit):
    if n == 0:
        return 1 if is_limit else 0
    if n < 0:
        return 0
    if memo[n][is_limit] != -1:
        return memo[n][is_limit]
    
    ways = 0
    for i in range(1, k+1):
        ways += dp(n-i, k, d, memo, is_limit or i >= d)
        ways %= MOD
    
    memo[n][is_limit] = ways
    return ways

def solve(n, k, d):
    memo = [[-1 for _ in range(2)] for _ in range(n+1)]
    return dp(n, k, d, memo, False)

n, k, d = map(int, input().split())
print(solve(n, k, d))
