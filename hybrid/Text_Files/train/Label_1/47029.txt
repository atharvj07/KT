import java.util.Scanner;

public class LemmingDefense {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] shields = new int[n];
        for (int i = 0; i < n; i++) {
            shields[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][n * (n - 1) / 2 + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= n * (n - 1) / 2; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j > 0 && i > 1 && shields[i - 1] == 0 && shields[i - 2] == 0) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 2][j - 1] + 1);
                }
                if (j > 0 && i < n && shields[i - 1] == 0 && shields[i] == 0) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - 1] + 1);
                }
            }
        }

        for (int i = 0; i <= n * (n - 1) / 2; i++) {
            System.out.print(dp[n][i] + " ");
        }
    }
}
