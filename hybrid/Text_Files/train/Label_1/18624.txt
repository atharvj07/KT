#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    int l[m], r[m];
    for (int i = 0; i < m; i++) cin >> l[i] >> r[i];
    vector<int> pos[n + 1];
    for (int i = 0; i < m; i++) {
        pos[l[i]].push_back(r[i]);
    }
    int ans[n + 1];
    for (int i = n; i > 0; i--) {
        if (pos[i].size() % 2 == 0) {
            ans[i] = 0;
        } else {
            ans[i] = 1;
            for (int j = 0; j < pos[i].size(); j += 2) {
                ans[pos[i][j]] ^= 1;
                ans[pos[i][j + 1]] ^= 1;
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << ans[i];
    return 0;
}
