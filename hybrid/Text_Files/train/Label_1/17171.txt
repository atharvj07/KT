#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
using namespace std;

int solve() {
    int n; cin >> n;
    vector<long long> a(n);
    int odd = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        odd += (int)(sqrt(a[i]) != (long long)(sqrt(a[i])));
    }
    sort(a.begin(), a.end());
    int ans = odd;
    if (odd % 2 == 0) {
        for (int i = 0; i < n && odd > 0; i++) {
            int ns = (int)(sqrt(a[i] + 1) != (long long)(sqrt(a[i] + 1)));
            odd -= (ns - (int)(sqrt(a[i]) != (long long)(sqrt(a[i]))));
            ans -= (int)(sqrt(a[i]) != (long long)(sqrt(a[i])));
        }
    } else {
        for (int i = 0; i < n && odd > 1; i++) {
            int ns = (int)(sqrt(a[i] + 1) != (long long)(sqrt(a[i] + 1)));
            odd -= (ns - (int)(sqrt(a[i]) != (long long)(sqrt(a[i]))));
            ans -= (int)(sqrt(a[i]) != (long long)(sqrt(a[i])));
        }
    }
    return ans;
}

int main() {
    cout << solve() << endl;
    return 0;
}
