#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  vector<bool> A(n + 1), B(n + 1);
  for (int i = 0; i < a; ++i) { int x; cin >> x; A[x] = true; }
  for (int i = 0; i < b; ++i) { int x; cin >> x; B[x] = true; }
  for (int i = 1; i <= n; ++i)
    if (A[i] && B[i])
      cout << 1;
    else if (!A[i] && B[i])
      cout << 2;
    else
      cout << 1;
  return 0;
}
