#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int MAXN = 3e5+5;
ll n, h[MAXN], b[MAXN];
ll dp[MAXN][3];

int main() {
  cin >> n;
  for (ll i = 1; i <= n; i++) {
    cin >> h[i] >> b[i];
  }
  sort(h + 1, h + n + 1);
  ll p = 1;
  for (ll i = 1; i <= n; i++) {
    if (i > 1 && h[i] == h[i - 1]) continue;
    dp[p][1] = b[i];
    dp[p][2] = b[i];
    p++;
  }
  for (ll j = 2; j < p; j++) {
    for (ll t = 1; t <= 2; t++) {
      dp[j][t] = max({dp[j][t], dp[j - 1][t], dp[j - 1][3 - t] + b[h[j]]});
    }
    dp[j][3] = dp[j][2];
  }
  cout << max(dp[p - 1][1], dp[p - 1][2]) << endl;
  return 0;
}
