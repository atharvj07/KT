def max_chessmen(n, m):
    if n > m:
        n, m = m, n
    if n == 1:
        return (m // 4) * 4 + min(2, m % 4) * 2
    if n == 2:
        if m == 2:
            return 0
        if m == 3 or m == 7:
            return n * m - 2
        return n * m
    return (n * m + 1) // 2 * 2

n, m = map(int, input().split())
print(max_chessmen(n, m))
