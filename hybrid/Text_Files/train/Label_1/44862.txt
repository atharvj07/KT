import java.util.*;

class Main {
    static boolean isPrime(int x) {
        if (x <= 1) {
            return false;
        }
        for (int i = 2; i * i <= x; i++) {
            if (x % i == 0) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        ArrayList<Integer> primes = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (isPrime(a[i])) {
                primes.add(a[i]);
            }
        }
        if (primes.size() < 3) {
            System.out.println("Impossible");
            return;
        }
        int m = n - primes.size();
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            ArrayList<Integer> b = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                if (!primes.contains(a[j])) {
                    b.add(a[j]);
                    primes.add(a[j]);
                    if (b.size() >= 3) {
                        break;
                    }
                }
            }
            System.out.print(b.size() + " ");
            for (int x : b) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }
}
