import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Map<Integer, TreeMap<Integer, String>> regions = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String name = sc.next();
            int reg = sc.nextInt();
            int score = sc.nextInt();
            regions.computeIfAbsent(reg, k -> new TreeMap<>()).put(score, name);
        }
        for (int i = 1; i <= m; i++) {
            TreeMap<Integer, String> region = regions.get(i);
            if (region.size() < 2) {
                System.out.println("?");
            } else {
                String[] team = new String[2];
                team[0] = region.lastEntry().getValue();
                region.remove(region.lastKey());
                team[1] = region.lastEntry().getValue();
                Arrays.sort(team);
                System.out.println(team[0] + " " + team[1]);
            }
        }
    }
}
