import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        char[] s = scanner.nextLine().toCharArray();
        int[] ans = new int[s.length];
        int n = s.length;
        for (int i = 0; i < n; ) {
            int j = i;
            while (j < n && s[j] == s[i]) {
                j++;
            }
            if ((j - i + 1) % 2 == 0) {
                i = j;
                continue;
            }
            if (j - i == 1) {
                ans[i] = 1;
            } else {
                int k = 0;
                for (; k * 2 + 1 < j - i; k++) {
                    if (s[i + k] != s[j - 1 - k]) {
                        break;
                    }
                }
                if (k * 2 + 1 < j - i) {
                    if (s[i + k] < s[j - 1 - k]) {
                        ans[i + k] = 1;
                        i = i + k + 1;
                    } else {
                        ans[j - 1 - k] = 1;
                        i = j - k;
                    }
                } else {
                    ans[i + k] = 1;
                    i = i + k + 1;
                }
            }
        }
        for (int a : ans) {
            System.out.print(a + " ");
        }
        System.out.println();
    }
}
