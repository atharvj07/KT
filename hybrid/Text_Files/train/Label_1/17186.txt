#include <bits/stdc++.h>

using namespace std;
using ll = long long;

inline char min_char(string &s, int i, int k) {
    for (int j = i + 1; j < i + 1 + k; ++j) {
        if (j >= s.size()) break;
        if (s[j] < s[i]) return s[j];
    }
    return s[i];
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n, k;
        cin >> n >> k;
        string s;
        cin >> s;
        for (int i = 0; i < n; ++i) {
            s[i] = min_char(s, i, k);
            k -= s[i] < s[i - 1];
            if (k <= 0) break;
        }
        cout << s << '\n';
    }
}
