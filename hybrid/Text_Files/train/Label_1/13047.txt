#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  bool has[3] = {false}, pet[3] = {false};
  for (char c : s) {
    if (c == 'A') { has[0] = true; }
    if (c == 'B') { has[1] = true; }
    if (c == 'C') { has[2] = true; }
  }
  for (char c : s) {
    if (c == '.') continue;
    if (c == 'A') { pet[0] = true; pet[1] = true; }
    if (c == 'B') { pet[1] = true; pet[2] = true; }
    if (c == 'C') { pet[0] = true; pet[2] = true; }
  }
  bool ok = false;
  for (int i = 0; i < 3; i++) { ok |= has[i] & pet[i]; }
  cout << (ok ? "Yes" : "No") << endl;
  return 0;
}
