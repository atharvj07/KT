#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
int main(){
    int n; cin>>n;
    vector<ll> a(n);
    for(int i=0; i<n; i++) cin>>a[i];
    sort(a.rbegin(), a.rend());
    vector<bool> vis(n);
    vector<ll> ans;
    ans.push_back(a[0]); vis[0]=1;
    for(int i=1; i<n; i++){
        for(int j=0; j<n; j++){
            if(vis[j]||!a[i]) continue;
            if(a[i]==a[j]){
                ans.push_back(a[i]); vis[i]=1; vis[j]=1;
                break;
            }
            else if(a[j]%3==0 && a[j]/3==a[i]){
                ans.push_back(a[i]); vis[i]=1; vis[j]=1;
                break;
            }
            else if(a[i]%2==0 && a[i]/2==a[j]){
                ans.push_back(a[i]); vis[i]=1; vis[j]=1;
                break;
            }
        }
    }
    for(auto x: ans) cout<<x<<" ";
    cout<<endl;
}
