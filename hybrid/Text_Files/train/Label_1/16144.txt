#include <iostream>
#include <set>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  set<char> have;
  for (char ch : s) have.insert(ch);
  if (have.size() > 10 || have.find('0') == have.end() || have.find('8') == have.end())
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
