import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        int C = sc.nextInt();
        int X = sc.nextInt();
        int Y = sc.nextInt();

        int minCost = Math.min(A * X + B * Y, (X + Y) % 2 == 0 ? (X + Y) / 2 * C : (X + Y) / 2 * C + A);
        System.out.println(minCost);
    }
}
