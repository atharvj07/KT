#include <bits/stdc++.h>
using namespace std;
#define ll long long

vector<int> loops = {0, 1, 2, 0, 1, 0, 1, 0, 2, 1};
ll n = 0, t = 0;
char arr[20];

int main() {
  int k;
  cin >> k;
  if (k == 1)
    cout << 4;
  else {
    if (!(k & 1)) k /= 2;
    k -= 1;
    for (int i = 0; i < 18; i++) {
      while (k >= 0) {
        if (loops[(t % 10)] + loops[(t / 10) % 10] > 0) {
          t += 8;
          if (i % 2 == 0) {
            n += (t * pow(10, i));
            t = 0;
          }
          k -= loops[(t % 10)] + loops[(t / 10) % 10];
        } else {
          t++;
          if (i % 2 == 0) {
            n += (t * pow(10, i));
            t = 0;
          }
        }
      }
      if (i % 2 == 0) n += (t * pow(10, i));
      if (!k && i % 2) {
        sprintf(arr, "%lld", n);
        int len = strlen(arr);
        len++;
        cout << n / pow(10, len / 2 - 1);
        return 0;
      }
      t /= 10;
      k++;
    }
    t /= 10;
    cout << -1;
  }
  return 0;
}
