import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[12];
        for (int i = 0; i < 12; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        for (int i = 0; i < 6; i++) {
            if (arr[i] != arr[6 - i - 1]) {
                System.out.println("no");
                return;
            }
        }
        System.out.println("yes");
    }
}
