#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;

int poww(int a,int b){
    int ans = 1;
    while(b){
        if(b&1){
            ans*=a;
            ans%=mod;
        }
        a*=a;
        a%=mod;
        b>>=1;
    }
    return ans;
}

int32_t main(){
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int n; cin>>n;
    string s; cin>>s;
    int m = s.size();
    int j = 0;
    int ans = 0;
    for(int i=0;i<m;i++){
        while(j<n && s[j%m] == s[(i+j)%m]) j++;
        if(j == n){
            ans += poww(2,(n-m));
            ans %= mod;
        }
    }
    cout<<ans<<endl;
    return 0;
}
