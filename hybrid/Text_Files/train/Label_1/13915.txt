#include <bits/stdc++.h>

#define ll long long
#define pii pair<int, int>
#define F first
#define S second
using namespace std;

const int MAXN = 1e5 + 5;
vector<int> G[MAXN], ans;
int sub[MAXN], tin[MAXN], tout[MAXN], timer, par[MAXN], dep[MAXN];
bool vis[MAXN];

void dfs_init(int u, int p) {
    par[u] = p;
    dep[u] = dep[p] + 1;
    tin[u] = ++timer;
    for (auto v : G[u]) {
        if (v != p) dfs_init(v, u);
    }
    tout[u] = ++timer;
}

bool is_anc(int u, int v) { return tin[u] <= tin[v] && tout[v] <= tout[u]; }

int lca(int u, int v) {
    if (is_anc(u, v)) return u;
    if (is_anc(v, u)) return v;
    int pu = par[u], pv = par[v];
    while (pu != pv) {
        if (dep[pu] > dep[pv])
            pu = par[pu];
        else
            pv = par[pv];
    }
    return pu;
}

unordered_set<int> path[MAXN];

int mex(int x, int y) {
    int l = lca(x, y);
    unordered_set<int> s = path[x];
    while (l != x) {
        for (auto p : path[par[x]]) s.insert(p);
        x = par[x];
    }
    int i = 0;
    while (s.count(i)) i++;
    return i;
}

void dfs(int u, int p, int par_edge_label) {
    path[u].insert(par_edge_label);
    for (auto v : G[u]) {
        if (v != p) {
            if (par[v] == u)
                ans.push_back(mex(u, v));
            else
                dfs(v, u, mex(u, v));
        }
    }
}

int main() {
    int n;
    cin >> n;
    for (int i = 1, u, v; i < n; i++) {
        cin >> u >> v;
        G[u].push_back(v);
        G[v].push_back(u);
    }
    dfs_init(1, 1);
    dfs(1, 1, -1);
    for (auto x : ans) cout << x << "\n";
    return 0;
}
