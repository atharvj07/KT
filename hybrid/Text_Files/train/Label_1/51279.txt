import java.util.*;
import java.nio.charset.StandardCharsets;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in, StandardCharsets.UTF_8);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        int min_shift = m;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int shift = (b[j] - a[i] + m) % m;
                if (shift < min_shift) {
                    min_shift = shift;
                }
            }
        }
        System.out.println(min_shift);
    }
}
