import java.util.Arrays;
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int[] sides = new int[12];
		for (int i = 0; i < 12; i++) {
			sides[i] = sc.nextInt();
		}
		sc.close();
		Arrays.sort(sides);
		int s1 = sides[0] + sides[1] + sides[2];
		int s2 = sides[3] + sides[4] + sides[5];
		int s3 = sides[6] + sides[7] + sides[8];
		if (s1 == s2 && s2 == s3) {
			System.out.println("yes");
		} else {
			System.out.println("no");
		}
	}
}
