import sys

n = int(sys.stdin.readline())
schedule = []
for _ in range(n):
    t, c = map(int, sys.stdin.readline().split())
    schedule.append([t, c])

time = 1
queue = 0
max_queue = 0
for task in schedule:
    if time <= task[0]:
        time = task[0]

    queue += task[1]
    max_queue = max(max_queue, queue)
    queue -= 1
    time += 1

print(time, max_queue)
