import sys

t = int(sys.stdin.readline())

for _ in range(t):
    n, w = map(int, sys.stdin.readline().split())
    arr = list(map(int, sys.stdin.readline().split()))
    arr.sort(reverse=True)
    h = 0
    cur = w
    for x in arr:
        if x <= cur:
            cur -= x
        else:
            cur = w - x
            h += 1
    print(h + 1)
