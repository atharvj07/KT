from collections import deque, defaultdict
from sys import stdin

n, m, p = map(int, stdin.readline().split())
s = list(map(int, stdin.readline().split()))
grid = [stdin.readline().strip() for _ in range(n)]
dist = [[n**2]*m for _ in range(n)]
vis = [[False]*m for _ in range(n)]

players = defaultdict(int)

for i in range(n):
    for j in range(m):
        if grid[i][j] != '.':
            players[grid[i][j]] += 1
            q = deque([(i,j,0)])
            vis[i][j] = True
            while q:
                x, y, d = q.popleft()
                if d > s[int(grid[i][j])-1]:
                    continue
                dist[x][y] = d
                for dx, dy in [(1,0),(-1,0),(0,1),(0,-1)]:
                    nx, ny = x+dx, y+dy
                    if 0<=nx<n and 0<=ny<m and not vis[nx][ny] and grid[nx][ny]=='.':
                        q.append((nx,ny,d+1))
                        vis[nx][ny] = True

for i in range(n):
    for j in range(m):
        if grid[i][j] != '.':
            players[grid[i][j]] -= dist[i][j]**2

for k in sorted(players.keys()):
    print(players[k])
