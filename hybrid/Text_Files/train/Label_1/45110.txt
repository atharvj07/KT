import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        int[] cnt = new int[26];
        long ans = 0;
        for (int i = 0; i < n; i++) {
            cnt[s.charAt(i) - 'A']++;
            if (cnt[s.charAt(i) - 'A'] == 3) ans += i;
        }
        System.out.println(ans);
    }
}
