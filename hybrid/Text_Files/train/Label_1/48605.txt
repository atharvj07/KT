import java.util.*;
import java.lang.*;

public class Main
{
    private static int n, m;
    private static int[] a, b, c;

    public static void main(String[] Args) throws Exception
    {
        Scanner cin = new Scanner(System.in);
        n = cin.nextInt(); m = cin.nextInt();
        a = new int[n]; b = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = cin.nextInt();
        for (int i = 0; i < n; i++)
            b[i] = cin.nextInt();

        c = new int[n];
        c[n - 1] = b[n - 1];
        for (int i = n - 2; i >= 0; i--)
            c[i] = Math.min(a[i], b[i] + c[i + 1]);

        long ans = 0;
        for (int i = 1; i < m; i++)
            ans += Math.min(a[(i - 1) % n] + c[i % n], b[(i - 1) % n]);
        System.out.println(ans);
    }
}
