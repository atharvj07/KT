import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n];
            boolean[] vis = new boolean[101];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                vis[a[i]] = true;
            }
            Arrays.sort(a);
            boolean f = true;
            for (int i = 1; i < n; i++) {
                if (a[i] - a[i - 1] > 1) {
                    f = false;
                    break;
                }
            }
            System.out.println(f || vis[a[0]] && vis[a[n - 1]] ? "YES" : "NO");
        }
    }
}
