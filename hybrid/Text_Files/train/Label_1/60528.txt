t = int(input())
for _ in range(t):
    n = int(input())
    grid = [list(input()) for _ in range(n)]
    counts = [0, 0, 0]
    for i in range(n):
        for j in range(n):
            if grid[i][j] == 'X':
                counts[(i+j)%3] += 1
    min_count_index = counts.index(min(counts))
    for i in range(n):
        for j in range(n):
            if grid[i][j] == 'X' and (i+j)%3 == min_count_index:
                grid[i][j] = 'O'
    for row in grid:
        print(''.join(row))
