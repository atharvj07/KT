#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int MAX_N = 2e5;

vector<int> adj[MAX_N + 1];
vector<int> order[MAX_N + 1];
int sz[MAX_N + 1];

void dfs(int u, int p) {
  order[u].push_back(u);
  for (int v : adj[u]) {
    if (v != p) {
      dfs(v, u);
      for (int x : order[v]) order[u].push_back(x);
    }
  }
}

int main() {
  int n, q;
  cin >> n >> q;
  for (int i = 2; i <= n; i++) {
    int p;
    cin >> p;
    adj[p].push_back(i);
  }
  dfs(1, 0);
  while (q--) {
    int u, k;
    cin >> u >> k;
    if (order[u].size() < k) {
      cout << "-1\n";
      continue;
    }
    cout << order[u][k - 1] << "\n";
  }
  return 0;
}
