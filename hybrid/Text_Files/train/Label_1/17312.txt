#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5 + 5, oo = 1e18;
struct edge { int to, w; };
int n, m, k, p[N], ans;
vector<edge> g[N];
bool vis[N];
void dfs(int u, int par, int &x, int &mx) {
  vis[u] = true;
  p[u] = par;
  for (edge e : g[u]) {
    int v = e.to;
    if (v == par) continue;
    if (!vis[v]) {
      dfs(v, u, x, mx);
    }
    if (mx < e.w)
      mx = e.w, x = u;
  }
}
main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) g[i].clear(), p[i] = 0, vis[i] = false;
    ans = 0;
    for (int i = 0, x, y, w; i < m; i++) {
      cin >> x >> y >> w;
      g[x].push_back({y, w});
      g[y].push_back({x, w});
      if (w > k) ans++;
    }
    int x = 0, mx = -oo;
    dfs(1, 1, x, mx);
    dfs(x, x, x, mx);
    if (mx > k) {
      int cur = x, d = 0;
      while (p[cur] != cur) d += mx - k, cur = p[cur];
      d -= mx - k;
      ans += d / 2;
    }
    cout << ans << '\n';
  }
}
