#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 1e5 + 5;
const int MAX_COLOR = 1e5 + 5;

vector<int> g[MAXN];
int color[MAXN];
vector<int> sizes;

void dfs(int u, int p, int c, int& t) {
    sizes[u] = 1;
    for (int j: g[u]) {
        if (j != p) {
            dfs(j, u, c, t);
            sizes[u] += sizes[j];
        }
    }
    t++;
    color[t] = c;
}

int main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n; i++) {
        int p;
        cin >> p;
        g[p].push_back(i);
        g[i].push_back(p);
    }

    sizes.resize(n + 1);

    int t = 0;
    dfs(1, 1, 0, t);

    sort(color + 1, color + t + 1);
    vector<int> cnt(MAX_COLOR, 0);

    int ans[MAXN];
    for (int i = 1; i <= n; i++) {
        cnt[color[i]]++;
        int cur = 0;
        for (int j = 1; j <= MAX_COLOR; j++) {
            cur += cnt[j] == 1;
        }

        ans[i] = t - cur;
    }

    for (int i = 1; i <= n; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
