#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int n;
vector<string> ans;

bool valid(int x, int y) {
    return x >= 0 && x < n && y >= 0 && y < n;
}

int main() {
    cin >> n;

    ans.resize(n, "");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            char c;
            if ((i + j) % 2 == 0)
                c = 'W';
            else
                c = 'B';
            ans[i] += c;
        }
    }

    if (n % 2 == 1) {
        if (ans[n - 1][n - 2] == ans[n - 2][n - 1]) {
            ans[n - 1][n - 2] = (ans[n - 1][n - 2] == 'W' ? 'B' : 'W');
            ans[n - 2][n - 1] = (ans[n - 2][n - 1] == 'W' ? 'B' : 'W');
        }
    }

    for (int i = 0; i < n; i++)
        cout << ans[i] << endl;

    return 0;
}
