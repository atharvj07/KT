import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String x = sc.next();
        int n = x.length();
        long[] pow2 = new long[n];
        pow2[0] = 1;
        for (int i = 1; i < n; i++) {
            pow2[i] = pow2[i - 1] * 2 % 1000000007;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (x.charAt(i) == '0') {
                continue;
            }
            long cnt = (1L << (n - 1 - i)) % 1000000007;
            cnt = (cnt * (cnt + 1)) % 1000000007;
            cnt = (cnt * pow2[i]) % 1000000007;
            ans += cnt;
            ans %= 1000000007;
        }
        System.out.println(ans);
    }
}
