import java.util.Scanner;

public class Mancala {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];
        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        int maxScore = 0;
        for (int i = 0; i < 14; i++) {
            if (holes[i] > 0) {
                int score = simulateMove(holes, i);
                maxScore = Math.max(maxScore, score);
            }
        }

        System.out.println(maxScore);
    }

    private static int simulateMove(int[] holes, int startHole) {
        int score = 0;
        int stones = holes[startHole];
        holes[startHole] = 0;
        int currentHole = startHole + 1;
        while (stones > 0) {
            holes[currentHole % 14]++;
            stones--;
            currentHole++;
        }

        for (int i = 0; i < 14; i++) {
            if (holes[i] % 2 == 0) {
                score += holes[i];
                holes[i] = 0;
            }
        }

        return score;
    }
}
