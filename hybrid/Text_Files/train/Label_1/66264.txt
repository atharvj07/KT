def check_minesweeper(n, m, field):
    for i in range(n):
        for j in range(m):
            if field[i][j] != '*':
                count = 0
                for x in range(max(0, i-1), min(n, i+2)):
                    for y in range(max(0, j-1), min(m, j+2)):
                        if field[x][y] == '*':
                            count += 1
                if count > 0 and field[i][j] == '.':
                    return "NO"
                elif field[i][j].isdigit() and count != int(field[i][j]):
                    return "NO"
    return "YES"

n, m = map(int, input().split())
field = [input() for _ in range(n)]
print(check_minesweeper(n, m, field))
