import java.util.Scanner;

public class AdjacentNumbersMatrix {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int[][] matrix = new int[n][n];

            if (!constructMatrix(matrix, n)) {
                System.out.println(-1);
            } else {
                printMatrix(matrix, n);
            }
        }
    }

    private static boolean constructMatrix(int[][] matrix, int n) {
        int num = 1;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (isSafe(matrix, i, j, n, num)) {
                    matrix[i][j] = num;
                    num++;
                } else {
                    return false;
                }
            }
        }

        return true;
    }

    private static boolean isSafe(int[][] matrix, int row, int col, int n, int num) {
        if (row < 0 || row >= n || col < 0 || col >= n || matrix[row][col] != 0) {
            return false;
        }

        if (row > 0 && Math.abs(matrix[row - 1][col] - num) == 1) {
            return false;
        }

        if (row < n - 1 && Math.abs(matrix[row + 1][col] - num) == 1) {
            return false;
        }

        if (col > 0 && Math.abs(matrix[row][col - 1] - num) == 1) {
            return false;
        }

        if (col < n - 1 && Math.abs(matrix[row][col + 1] - num) == 1) {
            return false;
        }

        return true;
    }

    private static void printMatrix(int[][] matrix, int n) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(matrix[i][j] + " ");
            }
            System.out.println();
        }
    }
}
