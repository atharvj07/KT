#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    int flag=0;
    vector<vector<int>> a(n,vector<int>(m));
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            cin>>a[i][j];
            if(a[i][j]) flag=1;
        }
    }
    if(a[0][0]==1 && flag){
        cout<<"NO"<<endl;
        return 0;
    }
    for(int i=1;i<n;i++)
        if(a[i][0]==1) a[i][0]=a[i-1][0]+1;
    for(int j=1;j<m;j++)
        if(a[0][j]==1) a[0][j]=a[0][j-1]+1;
    for(int i=1;i<n;i++){
        for(int j=1;j<m;j++){
            int x=min(a[i][0],a[0][j]);
            if(a[i][j]) x=min(x,a[i][j]);
            if((a[i][j] && a[i-1][j]==0 && a[i][j-1]==0) || x==0) {
                cout<<"NO"<<endl;
                return 0;
            }
            else if(a[i][j]==0) a[i][j]=x+1;
            else a[i][j]=x;
        }
    }
    cout<<"YES"<<endl;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            cout<<a[i][j]<<" ";
        }
        cout<<endl;
    }
}
