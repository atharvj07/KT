import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];
        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        int maxScore = 0;
        for (int i = 0; i < 14; i++) {
            if (holes[i] > 0) {
                int[] newHoles = new int[14];
                System.arraycopy(holes, 0, newHoles, 0, 14);
                int stones = newHoles[i];
                newHoles[i] = 0;
                int j = i + 1;
                while (stones > 0) {
                    newHoles[j % 14]++;
                    stones--;
                    j++;
                }
                int score = calculateScore(newHoles);
                maxScore = Math.max(maxScore, score);
            }
        }

        System.out.println(maxScore);
    }

    private static int calculateScore(int[] holes) {
        int score = 0;
        for (int i = 0; i < 14; i++) {
            if (holes[i] % 2 == 0) {
                score += holes[i];
            }
        }
        return score;
    }
}
