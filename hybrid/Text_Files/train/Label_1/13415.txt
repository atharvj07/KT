#include <iostream>

using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;

ll binpow(ll x, ll y) {
  ll res = 1;
  while (y > 0) {
    if (y % 2 == 1) res = res * x % mod;
    x = x * x % mod;
    y /= 2;
  }
  return res;
}

int main() {
  ll t, a, b;
  cin >> t >> a >> b;
  if (a == b)
    cout << "inf" << endl;
  else {
    cout << (binpow(a, t) - binpow(b, t) + mod) % mod << endl;
  }
  return 0;
}
