#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m, d;
    cin >> n >> m >> d;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    vector<int> ans(n);
    int cur = 0, day = 1;
    for (int i = 0; i < n; ++i) {
        if (a[i] - cur >= d) {
            cur = a[i];
            ans[i] = day++;
        } else {
            ans[i] = day;
        }
    }
    cout << day - 1 << endl;
    for (int i = 0; i < n; ++i) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
