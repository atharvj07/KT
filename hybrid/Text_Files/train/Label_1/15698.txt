#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        bool sorted = true;
        for (int i = 0; i < n - 1; i++)
            if (a[i] > a[i + 1]) {
                sorted = false;
                break;
            }
        long long ans = (1LL * n * (n - 1)) / 2 - 1;
        if (sorted || ans <= 0)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
