#include <iostream>
#include <vector>
#include <set>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    set<int> s;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        s.insert(a[i]);
    }
    if (s.size() == n) {
        cout << 0 << endl;
    } else {
        cout << n - 1 << endl;
    }
    return 0;
}
