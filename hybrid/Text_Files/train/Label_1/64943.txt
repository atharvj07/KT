import sys
n = int(input())
a = list(map(int, input().split()))
a.sort()
sum = a[n-1]
for i in range(n-2,-1,-1):
    sum += a[i]
    if sum > a[i]*2 and sum > a[n-1]:
        print(i+2)
        print(1, end=' ')
        for j in range(i,n):
            print(j+1, end=' ')
        sys.exit(0)
print(0)
