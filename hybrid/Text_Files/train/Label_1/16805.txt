#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 5;
const int MAXM = 5 + 5;
const int INF = 1e9 + 9;

using ll = long long;

int n, m, k;
int a[MAXM];
ll dp[MAXN][MAXM][2];

int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[j]; }
    dp[i][0][0] = 0;
    for (int j = 1; j <= m; j++) dp[i][j][0] = a[j];
    for (int j = 1; j <= m; j++) {
      for (int l = 0; l <= 1; l++) {
        dp[i][j][l] = min(dp[i - 1][j][l], dp[i - 1][j - 1][l] + a[j]);
        if (dp[i - 1][j - 1][l] < INF) dp[i][j][l] = min(dp[i][j][l], dp[i - 1][j - 1][l]);
      }
    }
  }

  ll ans = dp[n][m][1];
  if (ans >= INF) ans = -1;
  if (ans <= k) {
    vector<int> shot(m + 1, 0);
    int i = n, j = m, l = 1;
    while (i > 0 && j > 0 && l > 0) {
      if (dp[i - 1][j][l] == dp[i][j][l])
        i--;
      else if (dp[i][j - 1][l] == dp[i][j][l]) {
        j--;
        shot[j]++;
      } else if (dp[i][j - 1][l] < dp[i - 1][j][l]) {
        j--;
        if (dp[i][j][l] == dp[i][j][0]) l = 0;
        shot[j]++;
      }
    }
    for (int i = 1; i <= m; i++) { cout << shot[i] << " "; }
    cout << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
