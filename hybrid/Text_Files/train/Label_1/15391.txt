#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll N = 150005;
vector<ll> adj[N];

int main() {
    ll n, u, v;
    cin >> n;
    vector<ll> ans(n + 1, -1);
    for (int i = 0; i < n - 1; i++) {
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<bool> vis(n + 1);
    for (int i = 1; i <= n; i++) {
        if (vis[i]) continue;
        ll cnt = 1;
        vector<ll> curr;
        curr.push_back(i);
        vis[i] = true;
        for (int j = 0; j < curr.size(); j++) {
            ll node = curr[j];
            for (ll v : adj[node]) {
                if (!vis[v]) {
                    curr.push_back(v);
                    vis[v] = true;
                    cnt++;
                }
            }
        }
        for (ll x : curr) {
            if (ans[cnt] == -1)
                ans[cnt] = x;
            else
                ans[n - cnt + 1] = x;
        }
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << "\n";
    return 0;
}
