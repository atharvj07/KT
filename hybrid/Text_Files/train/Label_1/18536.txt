#include<bits/stdc++.h>
using namespace std;
int32_t main(){
  int n, d, m;
  cin >> n >> d >> m;
  vector<pair<int, int>> grass(m);
  for(int i=0; i<m; i++){
    cin >> grass[i].first >> grass[i].second;
    if(grass[i].first >=d && grass[i].first <=n-d && grass[i].second >=d && grass[i].second <=n-d){
      cout << "YES\n";
    }
    else cout << "NO\n";
  }
  return 0;
}
