import sys

def solve(n, l, r, a, p):
	# Initialize b with the minimum possible values
	b = [l+1] * n

	# Iterate over the compressed sequence
	for i in range(n):
		# Update the ith element of b to satisfy the compressed sequence
		b[i] += p[i] - 1

	# Check if the compressed sequence of the resulting c is valid
	c = [b[i] - a[i] for i in range(n)]
	p2 = [0] * n
	for x in c:
		for j in range(n):
			if x >= a[j]:
				p2[j] += 1
	if p2 != p:
		return -1

	# Return the resulting b
	return b

# Read the input
n, l, r = map(int, input().split())
a = list(map(int, input().split()))
p = list(map(int, input().split()))

# Solve the problem
b = solve(n, l, r, a, p)

# Print the result
if b == -1:
	print(-1)
else:
	print(*b)
