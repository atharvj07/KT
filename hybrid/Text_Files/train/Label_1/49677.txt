import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            String p = scanner.next();
            String h = scanner.next();
            boolean found = false;
            char[] shuffled = p.toCharArray();
            Arrays.sort(shuffled);
            for (int j = 0; j < h.length(); j++) {
                char c = h.charAt(j);
                int start = j;
                int end = j;
                while (end + 1 < h.length() && h.charAt(end + 1) == c) {
                    end++;
                }
                if (Arrays.binarySearch(shuffled, start, end + 1, c) < 0) {
                    continue;
                }
                if (j == 0 || h.charAt(j - 1) != c) {
                    start = j;
                }
                if (j == h.length() - 1 || h.charAt(j + 1) != c) {
                    end = j;
                }
                StringBuilder sb = new StringBuilder();
                sb.append(h.substring(0, start));
                sb.append(p);
                sb.append(h.substring(end + 1));
                if (sb.toString().equals(h)) {
                    found = true;
                    break;
                }
                j = end;
            }
            System.out.println(found ? "YES" : "NO");
        }
    }
}
