import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int k = in.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = in.nextInt();
            }
        }
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (i % 2 == 1) {
                    for (int l = Math.max(1, j - k); l <= Math.min(m, j + k); l++) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][l] + arr[i - 1][j - 1]);
                    }
                } else {
                    for (int l = Math.max(1, j - k); l <= Math.min(m, j + k); l++) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][l] + arr[i - 1][j - 1]);
                    }
                }
            }
        }
        int max = 0;
        for (int i = 1; i <= m; i++) {
            max = Math.max(max, dp[n][i]);
        }
        System.out.println(max);
    }
}
