a, b = map(int, input().split())

if a == b:
    print("infinity")
else:
    if a < b:
        print(0)
    else:
        a_minus_b = a - b
        solutions = 0
        for x in range(1, int(a_minus_b**0.5) + 1):
            if a_minus_b % x == 0:
                if x > b:
                    solutions += 1
                if a_minus_b // x != x and a_minus_b // x > b:
                    solutions += 1
        print(solutions)
