#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    long long ans1=0,ans2=0;
    for(int i=0;i<(int)s.size();i++){
        if(s[i]==s[i+1]){
            ans1++;ans2++;
            while(i<(int)s.size()-1 and s[i]==s[i+1]){i++;ans2++;}
        }
        else{
            ans2++;while(i<(int)s.size()-2 and s[i]==s[i+2]){i++;ans2++;}
        }
    }
    cout<<ans1<<" "<<ans2<<endl;
    return 0;
}
