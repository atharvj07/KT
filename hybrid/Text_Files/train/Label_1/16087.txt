#include <bits/stdc++.h>
using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int n = s.size() - t.size();
  int z = 0;
  for (int i = 0; i < t.size(); i++) {
    for (int j = 0; j < s.size(); j++) {
      if (t[i] == s[j]) {
        z = max(z, j - i);
        break;
      }
    }
  }
  while (z--) cout << '0';
  cout << t << '\n';
}
