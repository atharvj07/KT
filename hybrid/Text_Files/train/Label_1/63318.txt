x1,y1 = map(int, input().split())
x2,y2 = map(int, input().split())
n = int(input())
s = input() * 3
dx = [0, 0, -1, 1]
dy = [1, -1, 0, 0]
for k in range(4):
    nx = x1 + dx[k] * (n+1)
    ny = y1 + dy[k] * (n+1)
    if x2 == nx and y2 == ny or nx<0 or ny<0:
        continue
    for i in range(n):
        nx += dx[(s[i+k] == 'R') + (s[i+k] == 'L') * 2]
        ny += dy[(s[i+k] == 'U') - (s[i+k] == 'D') * 2]
        if x2 == nx and y2 == ny:
            print(i+k+1)
            exit()
print(-1)
