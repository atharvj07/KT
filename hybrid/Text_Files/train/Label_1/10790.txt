#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;

int n, k;
string s;
int dp[40][40][40][2];

int solve(int pos, int rem, int lst, bool already) {
  if (pos == n) {
    if (rem == 0) {
      return 1;
    }
    return 0;
  }
  if (dp[pos][rem][lst][already] != -1) return dp[pos][rem][lst][already];
  int ans = 0;
  int l = already ? s[pos] - '0' : 0;
  int r = already ? s[pos] - '0' : 1;
  for (int i = l; i <= r; i++) {
    ans = (ans + solve(pos + 1, rem - i, i, i == (s[pos] - '0'))) % mod;
  }
  return dp[pos][rem][lst][already] = ans;
}

int main() {
  cin >> n >> s;
  k = s.size();
  memset(dp, -1, sizeof(dp));
  cout << solve(0, n, 0, false) << "\n";
  return 0;
}
