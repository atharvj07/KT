#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) b[i] = i;
    sort(b.begin(), b.end(), [&](int x, int y) { return a[x] < a[y]; });
    for (int i = 0; i < n; ++i) cout << a[b[i]] << ' ';
    cout << '\n';
  }
}
