import java.util.Scanner;

public class MarekAndThePerfectMatching {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int[][] probabilities = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                probabilities[i][j] = scanner.nextInt();
            }
        }

        long numerator = 1;
        long denominator = 1;
        for (int i = 0; i < n; i++) {
            int row = findNonZeroRow(probabilities, i);
            if (row == -1) {
                // No perfect matching exists.
                System.out.println(0);
                return;
            }

            // Swap rows i and row.
            swapRows(probabilities, i, row);

            // Divide all elements in row i by probabilities[i][i].
            long inverse = modInverse(probabilities[i][i]);
            for (int j = i; j < n; j++) {
                probabilities[i][j] = (int) (((long) probabilities[i][j] * inverse) % MOD);
            }

            // Eliminate elements in column i below probabilities[i][i].
            for (int j = i + 1; j < n; j++) {
                int factor = probabilities[j][i];
                for (int k = i; k < n; k++) {
                    probabilities[j][k] = (int) ((probabilities[j][k] - (long) factor * probabilities[i][k]) % MOD);
                }
            }

            numerator = (numerator * probabilities[i][i]) % MOD;
            denominator = (denominator * findDeterminant(probabilities, i + 1, n)) % MOD;
        }

        long answer = (numerator * modInverse(denominator)) % MOD;
        System.out.println(answer);
    }

    private static int findNonZeroRow(int[][] matrix, int startRow) {
        for (int i = startRow; i < matrix.length; i++) {
            for (int j = startRow; j < matrix.length; j++) {
                if (matrix[i][j] != 0) {
                    return i;
                }
            }
        }
        return -1;
    }

    private static void swapRows(int[][] matrix, int row1, int row2) {
        for (int i = 0; i < matrix.length; i++) {
            int temp = matrix[row1][i];
            matrix[row1][i] = matrix[row2][i];
            matrix[row2][i] = temp;
        }
    }

    private static long modInverse(long a) {
        return power(a, MOD - 2);
    }

    private static long power(long a, long p) {
        if (p == 0) {
            return 1;
        }
        long halfPower = power(a, p / 2);
        if (p % 2 == 0) {
            return halfPower * halfPower % MOD;
        } else {
            return a * halfPower * halfPower % MOD;
        }
    }

    private static long findDeterminant(int[][] matrix, int startRow, int endRow) {
        if (startRow == endRow - 1) {
            return matrix[startRow][startRow];
        }
        long det = 0;
        for (int i = startRow; i < endRow; i++) {
            int sign = (i - startRow) % 2 == 0 ? 1 : -1;
            det = (det + (long) sign * matrix[startRow][i] * findDeterminant(matrix, startRow + 1, endRow)) % MOD;
        }
        return det;
    }
}
