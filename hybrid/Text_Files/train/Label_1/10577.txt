#include <cstring>
#include <iostream>
#include <vector>
using namespace std;

int n, m;
char a[510][510];
int dx[] = {0, 0, 1, -1};
int dy[] = {1, -1, 0, 0};

bool valid(int x, int y) {
    if(x < 0 || x >= n || y < 0 || y >= m) return false;
    return true;
}

int main() {
    cin >> n >> m;
    for(int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<pair<int,int> >ds;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(a[i][j] == 'S') {
                bool ok = false;
                for(int k = 0; k < 4; k++) {
                    int x = i + dx[k];
                    int y = j + dy[k];
                    if(valid(x, y) && a[x][y] == '.') {
                        ds.push_back(make_pair(x, y));
                        ok = true;
                    }
                }
                if(!ok) {
                    cout << "No";
                    return 0;
                }
            }
        }
    }
    cout << "Yes" << endl;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(a[i][j] == 'S') {
                for(int k = 0; k < ds.size(); k++) {
                    if(ds[k].first == i && ds[k].second == j) {
                        a[i][j] = 'D';
                    }
                }
            }
        }
    }
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            cout << a[i][j];
        }
        cout << endl;
    }
}
