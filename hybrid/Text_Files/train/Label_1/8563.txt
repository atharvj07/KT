#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n, s, ans = -1;
  cin >> n >> s;
  vector<int> t(n);
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    t[i] = 100*x + y;
  }
  sort(t.begin(), t.end());
  for (int i = 0; i < n; i++) {
    if (t[i] <= s) {
      ans = t[i] % 100;
      s -= t[i];
    }
  }
  cout << ans << endl;
  return 0;
}
