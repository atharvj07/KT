import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);
        int n = stdin.nextInt();
        int k = stdin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = stdin.nextInt();
        }
        Arrays.sort(a);
        int m = 0;
        for (int i = 0; i < n; i++) {
            if (k >= a[i]) {
                m++;
                k -= a[i];
            } else {
                break;
            }
        }
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            System.out.print(i + 1 + " ");
        }
        System.out.println();
    }
}
