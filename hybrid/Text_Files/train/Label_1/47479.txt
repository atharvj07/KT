import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), c = sc.nextInt(), x[] = new int[n], dp[] = new int[n];
        for (int i = 0; i < n; i++) x[i] = sc.nextInt();
        for (int i = 1; i < n; i++) {
            dp[i] = x[i];
            for (int j = 0; j < i; j++) {
                if (x[i] >= x[j] && dp[j] + x[i] - x[j] - c > dp[i])
                    dp[i] = dp[j] + x[i] - x[j] - c;
            }
        }
        Arrays.sort(dp);
        System.out.println(dp[n - 1]);
    }
}
