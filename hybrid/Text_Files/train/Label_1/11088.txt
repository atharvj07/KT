#include<iostream>
#include <algorithm>
#include <tuple>
#include <vector>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  vector<tuple<int, int, string>> lsh;
  for (int i = 1; i <= n; i++) {
    string s;
    cin >> s;
    int len = s.length() - 1;
    lsh.push_back(make_tuple(len, i, s));
  }
  sort(lsh.begin(), lsh.end());
  int pref[n + 1];
  pref[0] = 0;
  for (int i = 0; i < n; i++) {
    int len;
    int idx;
    string s;
    tie(len, idx, s) = lsh[i];
    pref[i + 1] = pref[i] + len;
  }
  int suf[n + 1];
  suf[n] = 0;
  for (int i = n - 1; i >= 0; i--) {
    int len;
    int idx;
    string s;
    tie(len, idx, s) = lsh[i];
    suf[i] = suf[i + 1] + len;
  }
  int ans = n * n;
  int idx = -1;
  for (int i = 0; i < n; i++) {
    int j = i;
    int cur = pref[i] + suf[i + 1];
    while (j < n && get<2>(lsh[i]) == get<2>(lsh[j])) {
      j++;
      cur -= get<0>(lsh[j]);
    }
    if (cur < ans) {
      ans = cur;
      idx = i;
    }
  }
  for (int i = 0; i <= idx; i++) {
    int len;
    int idx2;
    string s;
    tie(len, idx2, s) = lsh[i];
    cout << s.substr(0, len + 1) << '\n';
  }
  for (int i = idx + 1; i < n; i++) {
    int len;
    int idx2;
    string s;
    tie(len, idx2, s) = lsh[i];
    cout << s << '\n';
  }
  return 0;
}
