#include <algorithm>
#include <iostream>

using namespace std;

int main() {
    int n, b, d;
    cin >> n >> b >> d;
    long long sum = 0;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        sum += a;
        if (sum > d) {
            ans++;
            sum = a;
        }
    }
    cout << ans << endl;
    return 0;
}
