import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int count = 0;
        while (a > 0) {
            count++;
            a--;
            if (count % b == 0) {
                a++;
            }
        }
        System.out.println(count);
    }
}
