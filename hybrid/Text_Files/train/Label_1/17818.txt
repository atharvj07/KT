#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    long long b = max(a[i], sum + 1);
    cout << b << " ";
    sum += b;
  }
  cout << endl;
  return 0;
}
