import java.util.Scanner;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int max = (int) Math.sqrt(n * 10);
        int[][] board = new int[max][max];
        int[] dx = {1, 1, 2, 2, -1, -1, -2, -2};
        int[] dy = {2, -2, 1, -1, 2, -2, 1, -1};
        int cnt = 0;
        int x = 0, y = 0;
        board[x][y] = 1;
        while (cnt < n) {
            boolean moved = false;
            for (int i = 0; i < max; i++) {
                for (int j = 0; j < max; j++) {
                    if (board[i][j] == 0) {
                        int attacks = 0;
                        for (int k = 0; k < 8; k++) {
                            int nx = i + dx[k];
                            int ny = j + dy[k];
                            if (nx >= 0 && nx < max && ny >= 0 && ny < max && board[nx][ny] == 1) {
                                attacks++;
                            }
                        }
                        if (attacks >= 4) {
                            board[i][j] = 1;
                            x = i;
                            y = j;
                            cnt++;
                            moved = true;
                            break;
                        }
                    }
                }
                if (moved) {
                    break;
                }
            }
            if (!moved) {
                break;
            }
        }
        for (int i = 0; i < max; i++) {
            for (int j = 0; j < max; j++) {
                if (board[i][j] == 1) {
                    System.out.println(i + " " + j);
                }
            }
        }
    }
}
