#include <iostream>
#include <vector>
using namespace std;
int main() {
  double s, x1, x2, t1, t2, p, d;
  cin >> s >> x1 >> x2 >> t1 >> t2 >> p >> d;
  double ans = s / t1;
  if (x1 > x2) swap(x1, x2);
  if (p == x1) {
    if (d == -1) ans = max(ans, (p - x1) / t2 + (s - x2) / t1);
    else ans = max(ans, (s - p) / t1 + (x2 - x1) / t2);
  } else if (p == x2) {
    if (d == 1) ans = max(ans, (s - p) / t1 + (p - x1) / t2);
    else ans = max(ans, (p - x1) / t2 + (x2 - s) / t1);
  } else if (p > x1) {
    if (d == 1) ans = max(ans, (p - x1) / t2 + (s - x2) / t1);
    else ans = max(ans, (s - p) / t1 + (x2 - x1) / t2);
  } else {
    if (d == 1) ans = max(ans, (s - p) / t1 + (x2 - x1) / t2);
    else ans = max(ans, (p - x1) / t2 + (s - x2) / t1);
  }
  cout << ans << endl;
}
