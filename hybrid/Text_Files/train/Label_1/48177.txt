import java.util.Scanner;

class Main {
    static long mod = (long)998244353;
    static long pw(long a, long b) {
        long r = 1;
        for(; b > 0; b >>>= 1) {
            if((b & 1) == 1) r = (r * a) % mod;
            a = (a * a) % mod;
        }
        return r;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int s = sc.nextInt();
        int r = sc.nextInt();
        long inv = pw(s - r + 1, mod - 2);
        System.out.println((p + mod - 1) * inv % mod);
    }
}
