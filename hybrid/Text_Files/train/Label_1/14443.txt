#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll M=1e9;
int main(){
	ll t; cin>>t;
	while(t--){
		ll n; cin>>n;
		vector<ll>v(n);
		for(ll i=0; i<n; i++) cin>>v[i];
		bool ok=0;
		ll m=v[0];
		for(ll i=1; i<n; i++){
			if(v[i]>m){
				cout<<i+1<<endl;
				ok=1;
				break;
			}
			m=max(m,v[i]);
		}
		if(!ok) cout<<-1<<endl;
	}
}
