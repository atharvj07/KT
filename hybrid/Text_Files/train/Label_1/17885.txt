#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    char a = 'a';
    for (int i = 0; i < n / 2; i++) {
      cout << a;
      a++;
      if (a > 'c') a = 'a';
    }
    for (int i = n / 2 - 1; i >= 0; i--) {
      cout << a;
      a++;
      if (a > 'c') a = 'a';
    }
    cout << endl;
  }
  return 0;
}
