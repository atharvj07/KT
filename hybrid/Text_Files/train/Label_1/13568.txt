#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  if (n % k != 0) {
    std::cout << "No" << std::endl;
    return 0;
  }
  int p = n / k;
  std::vector<int> s(k, p);
  for (int i = 0; i < n % k; i++) { s[i]++; }
  std::cout << "Yes" << std::endl;
  for (int i = 0; i < k; i++) { std::cout << s[i] << " "; }
  std::cout << std::endl;
  return 0;
}
