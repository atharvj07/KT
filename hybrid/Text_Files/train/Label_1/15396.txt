#include <bits/stdc++.h>
#define vi vector<int>
#define f first
#define s second
#define pb push_back
using namespace std;

const int mxN = 3e5 + 5;
int n, m, dp[mxN], pre[mxN];
vi arr;

int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    cin >> n >> m;
    arr.resize(n);
    for (int &el : arr) cin >> el;
    for (int i = 0; i < n; i++) {
        dp[i] = mxN;
        for (int j = 0; j < i; j++) {
            if (arr[j] <= arr[i]) {
                dp[i] = min(dp[i], dp[j] + 1);
            }
        }
    }
    if (dp[n - 1] == mxN)
        dp[n - 1] = 0;
    cout << dp[n - 1];
    return 0;
}
