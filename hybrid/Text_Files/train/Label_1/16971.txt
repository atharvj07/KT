#include <bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >> t;
    while(t--){
        int n; cin >> n;
        string s; cin >> s;
        int x = -1, y = -1;
        int pre[n + 1];
        pre[0] = 0;
        for(int i = 0; i < n; i++){
            if(s[i] == 'a')
                pre[i + 1] = pre[i] + 1;
            else
                pre[i + 1] = pre[i] - 1;
        }
        map<int, int> mp;
        for(int i = 0; i <= n; i++){
            if(pre[i] == 0){
                x = i;
                break;
            }
        }
        for(int i = n; i >= 0; i--){
            if(pre[i] == 0){
                y = i;
                break;
            }
        }
        for(int i = 0; i <= n; i++){
            if(mp[pre[i]] == 0){
                mp[pre[i]] = i;
            }
            else{
                if(s[(mp[pre[i]] - 1) % n] != s[i % n])
                    continue;
                int z = 1;
                while(pre[(mp[pre[i]] - z + n) % n] != pre[i])
                    z++;
                if((mp[pre[i]] - z + n) % n < i){
                    x = (mp[pre[i]] - z + n) % n + 1;
                    y = i;
                }
                break;
            }
        }
        cout << x << " " << y << endl;
    }
    return 0;
}
