import sys

input = sys.stdin.readline

n, m = map(int, input().split())

v = set()
h = {}

ans = 0

for _ in range(n):
    x = int(input())
    if x in v:
        ans += 1
        continue
    if x - 1 in v:
        ans += 2
        continue
    v.add(x)

for _ in range(m):
    x1, x2, y = map(int, input().split())
    if y in h:
        if x1 <= h[y][0] <= x2 or x1 <= h[y][1] <= x2:
            ans += 1
        continue
    h[y] = [x1, x2]
print(ans)
