#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i] * (i + 1);
  }
  int ans = sum;
  for (int i = 1; i < n; i++) {
    sum -= a[i] * (i + 1);
    sum += a[i] * (i - 1);
    ans = min(ans, sum);
  }
  cout << ans << endl;
  return 0;
}
