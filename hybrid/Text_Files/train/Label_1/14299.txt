#include <bits/stdc++.h>
using namespace std;
int n, p[200005], q[200005], ord[200005], pos[200005];
vector<int> ans;
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> p[i]; }
    for (int i = 0; i < n; i++) { cin >> q[i]; }
    iota(ord, ord + n, 0);
    sort(ord, ord + n, [&](int i, int j) { return p[i] < p[j]; });
    for (int i = 0; i < n; i++) pos[p[i]] = i;
    int curp = 0, curq = 0;
    while (ans.size() < n) {
        if (curp < n && curq < n && ord[curp] == q[curq]) {
            ans.push_back(p[curp]);
            curp++;
            curq++;
            continue;
        }
        if (curp < n && ord[curp] != q[curq]) {
            ans.push_back(p[curp]);
            curp++;
            continue;
        }
        if (curq < n && ord[curp] != q[curq]) {
            ans.push_back(q[curq]);
            curq++;
        }
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << endl;
}
