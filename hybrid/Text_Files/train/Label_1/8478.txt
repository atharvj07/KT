#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    long long a[n], b[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    copy(a, a + n, b);
    sort(b, b + n);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] > b[i]) {
            ans += a[i] - b[i];
        } else {
            ans += b[i] - a[i];
        }
    }
    cout << ans << endl;
    return 0;
}
