#include<bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
int n,m;
int main(){
    cin>>n>>m;
    vector<vector<ll>> a(n+2,vector<ll> (m+2));
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            cin>>a[i][j];
        }
    }
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            a[i][m+1]=max(a[i][m+1],a[i][j]);
            a[n+1][j]=max(a[n+1][j],a[i][j]);
        }
    }
    for(int i=n;i>=1;--i){
        for(int j=m;j>=1;--j){
            a[i][j]=max(a[i+1][j],a[i][j+1]);
        }
    }
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            cout<<a[i][j]<<" ";
        }
        cout<<endl;
    }
}
