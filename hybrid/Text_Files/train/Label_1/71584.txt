import sys
import pdb

n_nodes, n_queries = list(map(int, input().split()))
n_queries_list = [0] * n_queries

edges = []
for _ in range(n_nodes - 1):
    edges.append(tuple(map(int, input().split())))

for i in range(n_queries):
    n_queries_list[i] = list(map(int, input().split()))[1:]

adj_list = [[] for _ in range(n_nodes + 1)]
for x, y in edges:
    adj_list[x].append(y)
    adj_list[y].append(x)

parent = [1] * (n_nodes + 1)
depth = [0] * (n_nodes + 1)

def dfs(curr_node, par):
    for adj_node in adj_list[curr_node]:
        if adj_node == par: continue
        parent[adj_node] = curr_node
        depth[adj_node] = depth[curr_node] + 1
        dfs(adj_node, curr_node)

dfs(1, 1)

for query in n_queries_list:
    query_depth = set()
    status = "YES"
    for node in query:
        if depth[node] > depth[query[0]]:
            status = "NO"
            break
        query_depth.add(depth[node])
    if len(query_depth) == 1 or len(query_depth) == 2:
        print(status)
    else:
        print("NO")
