import math

polycarp_list = [1]
for i in range(2, 1001):
    polycarp_list.append(i * i)
    polycarp_list.append(i * i * i)

def solve(n):
    ans = 0
    for num in polycarp_list:
        if num > n:
            break
        ans += n // num
    return ans

test_cases = int(input())
for _ in range(test_cases):
    n = int(input())
    print(solve(n))
