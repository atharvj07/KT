#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int a1, a2, a3, a4;
        cin >> a1 >> a2 >> a3 >> a4;
        vector<int> a = {a1, a2, a3, a4};
        sort(a.begin(), a.end());
        cout << a[0] * a[3] << endl;
    }
    return 0;
}
