#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll gcd(ll x, ll y) { return y ? gcd(y, x % y) : x; }
int main() {
    string s;
    cin >> s;
    int a, b;
    cin >> a >> b;
    int n = s.size(), ia = 0, ib = n - 1;
    while (ia < ib && s[ia] % a == 0) ia++;
    while (ib >= ia && s[ib] % b == 0) ib--;
    if (ia < ib) {
        cout << "YES\n";
        cout << s.substr(0, ia) << '\n';
        cout << s.substr(ia) << '\n';
    } else
        cout << "NO\n";
}
