w = list(map(int, input().split()))
x = list(map(int, input()))
y = [0] * len(x)
q = [0] * len(x)
h, t = 0, 0
for i, wi in enumerate(w):
    if x[i] == 0:
        while h < t and w[q[t - 1]] > wi:
            t -= 1
        q[t] = i
        t += 1
        y[i] = q[h] + 1
    else:
        while h < t and w[q[t - 1]] < wi:
            t -= 1
        q[t] = i
        t += 1
        y[i] = q[h] + 1
        h += 1
print(*y)
