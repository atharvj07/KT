t = int(input())
for _ in range(t):
    n, u, v = map(int, input().split())
    obstacles = list(map(int, input().split()))
    ans = float('inf')
    for i in range(n - 1):
        if obstacles[i] == obstacles[i + 1]:
            ans = min(ans, min(u, v) + v)
        elif abs(obstacles[i] - obstacles[i + 1]) == 1:
            ans = min(ans, min(u, v))
        else:
            ans = 0
            break
    print(ans)
