#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, p[N], c[N], deg[N], vis[N], ans, cycle[N];
vector<int> adj[N];
void dfs(int u, int cc, int d) {
    if (vis[u] == cc) {
        ans = min(ans, d - cycle[u]);
        return;
    }
    vis[u] = cc;
    cycle[u] = d;
    dfs(p[u], cc, d + 1);
}
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> p[i];
        for (int i = 1; i <= n; i++) cin >> c[i];
        for (int i = 1; i <= n; i++) adj[i].clear(), vis[i] = 0, cycle[i] = 0, deg[i] = 0;
        for (int i = 1; i <= n; i++) adj[c[i]].push_back(i), deg[c[i]]++;
        ans = INT_MAX;
        for (int i = 1; i <= n; i++) {
            if (deg[i] == 1) {
                dfs(adj[i][0], i, 1);
            }
        }
        cout << (ans == INT_MAX ? -1 : ans) << "\n";
    }
    return 0;
}
