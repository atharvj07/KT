#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    long long int a[n]={0};
    long long int b[n];
    for(int i=0;i<n;i++){
        cin>>b[i];
    }
    int count=0;
    for(int i =0;i<n;i++){
        if(a[i]>b[i]){
            a[i]=a[i+1]-1;
            a[i+1]=a[i+1]-1;
            count++;
        }
        else if(a[i]<b[i]){
            a[i]=a[i]+1;
            a[i+1]=a[i+1]+1;
            count++;
        }
    }
    cout<<count<<endl;
    return 0;
}
