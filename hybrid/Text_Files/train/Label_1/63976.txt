import sys
n = int(sys.stdin.readline())
code=[] 
for i in range(n):
    code.append(sys.stdin.readline())
ans = -1
for i in range(5,-1,-1):
    g = {}
    for j in code:
        k = j[:i+1]
        if k in g: g[k]+=1
        else: g[k]=1
    chk = True
    for j in g.values():
        if j>1: 
            chk = False
            break
    if chk==True:
        ans = i+1
        break
print(ans)
