import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        // Calculate sum of ai * i for the original array
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i] * i;
        }

        // Sort the array in non-decreasing order
        Arrays.sort(a);

        // Calculate the maximum sum of ai * i after the move
        long maxSum = sum;
        for (int i = 0; i < n; i++) {
            maxSum += (n - 1 - 2 * i) * a[i];
        }

        System.out.println(maxSum);
    }
}
