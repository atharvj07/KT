import java.util.Scanner;

public class ChoutiAndTheSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] evenElements = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            evenElements[i] = scanner.nextInt();
        }

        long[] oddElements = new long[n / 2];
        oddElements[0] = (long) Math.sqrt(evenElements[0]) - 1;
        for (int i = 1; i < n / 2; i++) {
            oddElements[i] = (long) Math.sqrt(evenElements[i] + evenElements[i - 1]) - (long) Math.sqrt(evenElements[i - 1]);
        }

        boolean possible = true;
        for (long element : oddElements) {
            if (element <= 0 || element > 1000000000000L) {
                possible = false;
                break;
            }
        }

        if (possible) {
            System.out.println("Yes");
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    System.out.print(evenElements[i / 2] + " ");
                } else {
                    System.out.print(oddElements[(i - 1) / 2] + " ");
                }
            }
        } else {
            System.out.println("No");
        }
    }
}
