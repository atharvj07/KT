import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        StringBuilder sb = new StringBuilder();
        int[] stack = new int[n];
        int top = -1;
        for (int i = 0; i < n; i++) {
            int j = arr[i];
            if (j == top + 1) {
                sb.append(')');
                top--;
            } else {
                while (top >= 0 && arr[top] > j) {
                    sb.append(')');
                    top--;
                }
                if (top == -1) {
                    sb = new StringBuilder(":(");
                    break;
                } else {
                    top++;
                    stack[top] = j;
                    sb.append('(');
                }
            }
        }
        while (top >= 0) {
            sb.append(')');
            top--;
        }
        System.out.println(sb);
    }
}
