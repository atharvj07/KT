#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    if (k > n || (k - n % 2) % 2 != n % 2 || (n % 2 == 0 && k % 2) || (n % 2 == 1 && k % 2 == 0)) {
      cout << "NO\n";
    } else {
      cout << "YES\n";
      for (int i = 0; i < k - n % 2; i++) cout << 1 << " ";
      if (n % 2)
        cout << n - k + 1 << endl;
      else
        cout << n - k << endl;
    }
  }
  return 0;
}
