#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  pair<int, int> v[n + 1];
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  v[n] = {0, 0};
  int ans = 0, k = 0;
  for (int i = 0; i < n + 1; i++) {
    if (v[i].first == v[i].second) {
      k++;
    } else {
      ans = max(ans, k);
      k = 0;
    }
  }
  cout << max(ans, k) << "\n";
}
