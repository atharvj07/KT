#include <iostream>
#include <vector>
#include <set>
using namespace std;

long long gcd(long long a, long long b) {
    while (b != 0) {
        long long t = a % b;
        a = b;
        b = t;
    }
    return a;
}

int main() {
    int n;
    cin >> n;
    vector<long long> w(n), h(n), c(n);
    for (int i = 0; i < n; i++) {
        cin >> w[i] >> h[i] >> c[i];
    }
    set<pair<long long, long long>> se;
    for (int i = 0; i < n; i++) {
        long long p = gcd(w[i], h[i]);
        if (p != w[i]) {
            se.insert({w[i] / p, h[i] * p});
        }
        if (p != h[i]) {
            se.insert({w[i] * p, h[i] / p});
        }
    }
    cout << se.size() << endl;
    return 0;
}
