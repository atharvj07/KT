#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
const int maxn = 1005, INF = 1e9;
struct Req {
    int c, p;
    bool operator<(const Req &rhs) const {
        return p > rhs.p;
    }
} a[maxn];
int dp[maxn][maxn], n, k, ans[maxn];
vector<int> t[maxn];
int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; ++i) scanf("%d%d", &a[i].c, &a[i].p);
    scanf("%d", &k);
    for (int i = 1; i <= k; ++i) {
        int x;
        scanf("%d", &x);
        t[x].emplace_back(i);
    }
    sort(a + 1, a + n + 1);
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= k; ++j) dp[i][j] = dp[i - 1][j];
    for (int i = 1; i <= n; ++i)
        for (auto &j : t[a[i].c]) dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + a[i].p);
    int mxv = 0, x;
    for (int i = 1; i <= k; ++i)
        if (dp[n][i] > mxv) mxv = dp[n][i], x = i;
    int j = x, m = 0, s = 0;
    for (int i = n; i; --i) {
        if (mxv == 0) break;
        if (dp[i][j] != dp[i - 1][j]) {
            ans[++m] = i;
            s += a[i].p;
            mxv -= a[i].p;
            for (auto &j : t[a[i].c]) if (dp[i][j] > dp[i][j - 1]) { j--; break; }
        }
    }
    printf("%d %d\n", m, s);
    for (int i = 1; i <= m; ++i) printf("%d %d\n", ans[i], x);
}
