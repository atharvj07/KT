def check(i, j, ans):
    if (i, j) == (2, n):
        return True
    if (i, j) in ans:
        return False
    ans.add((i, j))
    if (i, j + 1) in adj_map and adj_map[(i, j + 1)] == (i, j):
        if check(i, j + 1, ans):
            return True
    if (i, j - 1) in adj_map and adj_map[(i, j - 1)] == (i, j):
        if check(i, j - 1, ans):
            return True
    if (i + 1, j) in adj_map and adj_map[(i + 1, j)] == (i, j):
        if check(i + 1, j, ans):
            return True
    if (i - 1, j) in adj_map and adj_map[(i - 1, j)] == (i, j):
        if check(i - 1, j, ans):
            return True
    return False
