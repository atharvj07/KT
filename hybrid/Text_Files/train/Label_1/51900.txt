import java.util.*;

public class PowerOfTwoPairs {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }

        Map<Integer, Integer> count = new HashMap<>();
        for (int a : A) {
            count.put(a, count.getOrDefault(a, 0) + 1);
        }

        int pairs = 0;
        for (int a : count.keySet()) {
            for (int b = a + 1; b <= 1000000000; b *= 2) {
                if (count.containsKey(b - a)) {
                    pairs = Math.max(pairs, count.get(a) * count.get(b - a));
                }
            }
        }

        System.out.println(pairs);
    }
}
