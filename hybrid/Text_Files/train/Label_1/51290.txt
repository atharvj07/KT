import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    int count = 0;
    int j = 0;
    for (int i = 0; i < n; i++) {
      while (j < n && a[j] <= a[i] + k) {
        j++;
      }
      count++;
      i = j - 1;
    }
    System.out.println(count);
  }
}
