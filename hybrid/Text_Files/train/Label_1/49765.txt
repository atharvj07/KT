import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++)
            w[i] = scan.nextInt();
        int sum = 0;
        for (int i = 0; i < n; i++)
            sum += w[i];
        System.out.println(sum % 2 == 0 ? "YES" : "NO");
    }
}
