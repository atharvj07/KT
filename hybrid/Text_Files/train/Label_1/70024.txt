for _ in range(int(input())):
    n, m, k = map(int, input().split())
    isFull = n % m == 0
    full_len = n//m
    empty_len = full_len+1
    for i in range(k):
        for j in range(m):
            if isFull:
                print(full_len, *range(j*full_len+1,j*full_len+full_len+1))
            else:
                full = (j+1)*full_len <= n
                print((full_len if full else empty_len), *range(j*full_len+1,(j+1)*full_len+1 if full else (j+1)*empty_len+1))
