import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int n = scan.nextInt();
    int[] a = new int[n];
    int sum = 0;

    for (int i = 0; i < n; i++) {
      a[i] = scan.nextInt();
      sum += a[i];
    }

    int ans = 1;
    int cur = a[0];

    for (int i = 1; i < n; i++) {
      if (cur >= (sum + 1) / 2) {
        break;
      }
      ans++;
      cur += a[i];
    }

    System.out.println(ans);
  }
}
