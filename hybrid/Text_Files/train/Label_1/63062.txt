n,m=map(int,input().split())
a=[list(map(int,input().split())) for _ in range(n)]
b=[[0]*m for _ in range(n)]
chk=False
for i in range(n-1):
    for j in range(m-1):
        if a[i][j]!=a[i+1][j+1]:
            if a[i][j+1]==a[i+1][j]==1:
                b[i][j]=b[i+1][j]=b[i+1][j+1]=b[i][j+1]=1
                chk=True
            else: chk=False; break
        else: chk=True
if chk:
    k=0
    for i in range(n):
        for j in range(m):
            if a[i][j]!=b[i][j]:
                print(-1); exit()
            if b[i][j]==1: k+=1
    print(k)
    for i in range(n):
        for j in range(m):
            if b[i][j]==1: print(i+1,j+1)
else: print(-1)
