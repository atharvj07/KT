import sys
input = sys.stdin.readline

def solve():
    k = int(input())
    sequences = []
    sums = [0]
    for _ in range(k):
        n = int(input())
        a = list(map(int, input().split()))
        sequences.append(a)
        sums.append(sums[-1] + sum(a))
    for i in range(1, k):
        for j in range(i):
            if sums[i] - sequences[i][0] == sums[j] - sequences[j][0]:
                print("YES")
                print(i + 1, 1)
                print(j + 1, 1)
                return
    print("NO")

solve()
