for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if any(x % 2 == 1 for x in a):
        print(1)
        print(a.index(min(x for x in a if x % 2 == 1)) + 1)
    elif all(x % 2 == 0 for x in a):
        print(1)
        print(a.index(0) + 1 if 0 in a else a.index(min(a)) + 1)
    else:
        print(-1)
