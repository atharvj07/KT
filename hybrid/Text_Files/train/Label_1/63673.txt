import sys
read = sys.stdin.readline

n, m = map(int, read().split())
a = list(map(int, read().split()))
l = [0] * (m + 1)
r = [0] * (m + 1)
for i in range(m):
    x, y = map(int, read().split())
    l[i] = x
    r[i] = y

a.sort()
l.sort()
r.sort()
ans = 0
ans_l = []
for i in range(m):
    idx = a.index(l[i])
    if a[idx + 1] == l[i]:
        ans += (a[idx + 1] - a[idx]) * (r[i] - l[i] + 1)
        ans_l.append(i + 1)
        a.pop(idx)
        a.pop(idx)
    else:
        ans += (a[idx + 1] - a[idx]) * (r[i] - l[i])
        ans_l.append(i + 1)
        a.pop(idx)

print(ans)
print(len(ans_l))
print(*ans_l)
