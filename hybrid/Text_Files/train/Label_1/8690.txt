#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt0 = 0, cnt1 = 0, cnt2 = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') cnt0++;
    if (s[i] == '1') cnt1++;
    if (s[i] == '2') cnt2++;
  }
  int need0 = (cnt1 + cnt2 - cnt0) / 3;
  int need1 = (cnt2 + cnt0 - cnt1) / 3;
  int need2 = (cnt0 + cnt1 - cnt2) / 3;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0' && need0 > 0) {
      s[i] = '1';
      need0--;
    } else if (s[i] == '1' && need1 > 0) {
      s[i] = '2';
      need1--;
    } else if (s[i] == '2' && need2 > 0) {
      s[i] = '0';
      need2--;
    }
  }
  cout << s << endl;
  return 0;
}
