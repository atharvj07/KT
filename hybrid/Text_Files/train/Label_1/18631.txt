#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<ll,ll>
#define inf 1e9+7
const long long mod = 1e9 + 7;

void f()
{
	string s;
	cin>>s;
	ll n = (ll)s.size();
	stack<pii>st;
	for(ll i = 0 ; i < n ; i++ )
	{
		if(s[i] == '(')
		{
			st.push({i,1});
		}
		else if(s[i] == ')')
		{
			if(!st.empty())
			{
				pii x = st.top();
				if(x.second > 0)
				{
					x.first = i;
					x.second--;
					st.pop();
					st.push(x);
				}
			}
		}
		else
		{
			st.push({i,0});
		}
	}
	ll ans = 0;
	while(!st.empty())
	{
		pii x = st.top();
		if(x.second > 0)
		{
			st.pop();
			if(x.second %2 == 1)
			{
				ans += (x.first/2);
			}
			else ans += x.first+1;
		}
		else
		{
			st.pop();
			ans += (x.first/2+1);
		}
	}
	cout<<ans;
	cout<<'\n';
}

int32_t main()
{
	int t;
	t = 1;
	// cin>>t;
	for(int cs = 1 ; cs <= t ; cs++)
	{
		f();
	}
	return 0;
}
