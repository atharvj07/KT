import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String recipe = sc.nextLine();
        int[] count = new int[3];
        for (char c : recipe.toCharArray()) {
            if (c == 'B') {
                count[0]++;
            } else if (c == 'S') {
                count[1]++;
            } else {
                count[2]++;
            }
        }
        int[] have = new int[3];
        for (int i = 0; i < 3; i++) {
            have[i] = sc.nextInt();
        }
        int[] price = new int[3];
        for (int i = 0; i < 3; i++) {
            price[i] = sc.nextInt();
        }
        int money = sc.nextInt();
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < 3; i++) {
            if (count[i] != 0) {
                min = Math.min(min, (have[i] + money / price[i]) / count[i]);
            }
        }
        System.out.println(min);
    }
}
