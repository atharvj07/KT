import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prizeCodingContest = 0;
        switch (x) {
            case 1: prizeCodingContest = 300000; break;
            case 2: prizeCodingContest = 200000; break;
            case 3: prizeCodingContest = 100000; break;
        }

        int prizeRobotManeuver = 0;
        switch (y) {
            case 1: prizeRobotManeuver = 300000; break;
            case 2: prizeRobotManeuver = 200000; break;
            case 3: prizeRobotManeuver = 100000; break;
        }

        int additionalPrize = (x == 1 && y == 1) ? 400000 : 0;

        int totalPrize = prizeCodingContest + prizeRobotManeuver + additionalPrize;

        System.out.println(totalPrize);
    }
}
