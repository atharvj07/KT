import java.util.*;
import java.util.concurrent.atomic.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long[]a=new long[n];
        long sum=0;
        for(int i=0;i<n;i++) {
            a[i]=sc.nextInt();
            sum+=a[i];
        }
        long x=sum/2;
        long cur=0;
        int ans=-1;
        for(int i=0;i<n;i++) {
            cur+=a[i];
            if(cur>=x) {
                ans=i+1;
                break;
            }
        }
        System.out.println(ans);
    }
}
