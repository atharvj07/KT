import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        if (isGeometric(a)) {
            System.out.println(0);
        } else if (isGeometricPossible(a)) {
            System.out.println(1);
        } else {
            System.out.println(2);
        }
    }

    private static boolean isGeometric(long[] a) {
        if (a.length <= 2) {
            return true;
        }
        long b = a[1] / a[0];
        for (int i = 1; i < a.length; i++) {
            if (a[i] != a[i - 1] * b) {
                return false;
            }
        }
        return true;
    }

    private static boolean isGeometricPossible(long[] a) {
        if (a.length <= 2) {
            return true;
        }
        boolean[] deleted = new boolean[a.length];
        for (int i = 1; i < a.length - 1; i++) {
            long b = a[i + 1] / a[i];
            long c = a[i - 1] / a[i];
            if (b != c) {
                deleted[i] = true;
                if (isGeometric(remove(a, i))) {
                    return true;
                }
                deleted[i] = false;
            }
        }
        return false;
    }

    private static long[] remove(long[] a, int index) {
        long[] b = new long[a.length - 1];
        int j = 0;
        for (int i = 0; i < a.length; i++) {
            if (i != index) {
                b[j++] = a[i];
            }
        }
        return b;
    }
}
