import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String l = input.next();
        String r = input.next();
        
        int maxXor = 0;
        for (int i = 0; i < n; i++) {
            if (l.charAt(i) != r.charAt(i)) {
                maxXor |= (1 << i);
            }
        }
        
        System.out.println(Integer.toBinaryString(maxXor));
    }
}
