#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n, x, p[2], q[2], ans = 0;
    cin >> t;
    while (t--) {
        cin >> n;
        map<int, int> mp;
        for (int i = 0; i < n; i++) {
            cin >> x;
            mp[i % 2 ^ x % 2]++;
        }
        if (mp[0] + mp[1] > n) {
            cout << "-1\n";
            continue;
        }
        x = n - mp[0] - mp[1];
        if (x % 2) {
            cout << "-1\n";
            continue;
        }
        cout << x / 2 << endl;
    }
    return 0;
}
