import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Map<String, Integer> results = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String name = scanner.next();
            int points = scanner.nextInt();
            results.put(name, results.getOrDefault(name, 0) + points);
        }
        List<String> players = new ArrayList<>(results.keySet());
        Collections.sort(players, (a, b) -> Integer.compare(results.get(b), results.get(a)));
        int m = players.size();
        System.out.println(m);
        for (String player : players) {
            int points = results.get(player);
            int rank = 100 * (m - players.indexOf(player)) / m;
            String category = "noob";
            if (rank <= 50) {
                category = "noob";
            } else if (rank > 50 && rank <= 80) {
                category = "random";
            } else if (rank > 80 && rank <= 90) {
                category = "average";
            } else if (rank > 90 && rank <= 99) {
                category = "hardcore";
            } else {
                category = "pro";
            }
            System.out.println(player + " " + category);
        }
    }
}
