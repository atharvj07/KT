#include <bits/stdc++.h>
using namespace std;
#define ll long long

const int N = 1e5 + 3;
int n, a[N], b[N], ans[N];

int bfs(int x) {
    queue<int> q;
    q.push(x);
    while (q.size()) {
        int u = q.front();
        q.pop();
        if (u == b[0]) return 1;
        int lst = u % 10;
        u /= 10;
        int cur = u % 10;
        if (lst != 9) {
            int v = u;
            v = v * 10 + lst + 1;
            if (!a[v]) {
                a[v] = a[u] + 1;
                ans[a[v]] = -1;
                q.push(v);
            }
        }
        if (cur != 0) {
            int v = u;
            v = v * 10 + lst - 1;
            if (!a[v]) {
                a[v] = a[u] + 1;
                ans[a[v]] = 1;
                q.push(v);
            }
        }
        if (!cur && !lst) {
            int v = u + 1;
            if (!a[v]) {
                a[v] = a[u] + 1;
                ans[a[v]] = 2;
                q.push(v);
            }
        }
    }
    return -1;
}

int main() {
    cin >> n;
    string s, t;
    cin >> s >> t;
    for (int i = 0; i < n; i++) {
        a[i] = s[i] - '0';
        b[i] = t[i] - '0';
    }
    while (s.size() < n) s = '0' + s;
    while (t.size() < n) t = '0' + t;
    reverse(s.begin(), s.end());
    reverse(t.begin(), t.end());
    for (int i = 0; i < n; i++) {
        a[i] = s[i] - '0';
        b[i] = t[i] - '0';
    }
    int res = bfs(a[0]);
    if (res == -1) cout << -1;
    else {
        cout << a[b[0]] << endl;
        for (int i = 1; i <= a[b[0]]; i++) {
            if (ans[i] == -1)
                cout << 1 << " " << -1 << endl;
            else if (ans[i] == 1)
                cout << 1 << " " << 1 << endl;
            else
                cout << 2 << " " << 1 << endl;
        }
    }
    return 0;
}
