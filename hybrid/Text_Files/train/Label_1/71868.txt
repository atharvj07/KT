import sys
import math

[n] = [int(i) for i in input().split()]
if n < 4:
    print(0); sys.exit()
coords = [[int(j) for j in input().split()] for i in range(n)]

ans = 0
for i in range(n):
    for j in range(i+1, n):
        if i == j: continue
        for k in range(j+1, n):
            if j == k: continue
            for l in range(k+1, n):
                if l == i or l == j or l == k: continue
                x1, y1 = coords[i]
                x2, y2 = coords[j]
                x3, y3 = coords[k]
                x4, y4 = coords[l]
                if (x2 - x1) * (y3 - y4) == (x3 - x4) * (y2 - y1): continue
                ans += 1

print(ans)
