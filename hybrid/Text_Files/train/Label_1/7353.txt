#include <bits/stdc++.h>
using namespace std;

vector<pair<int, int>> v;
vector<int> ans;

bool tryy(int x, int y, int z) {
  if (z == v.size()) return true;
  if (x > v[z].first || y > v[z].second) return false;
  if (x == v[z].first && y == v[z].second) return tryy(x + 1, y, z + 1);
  if (tryy(x + 1, y, z)) {
    ans.push_back(2);
    return true;
  }
  if (tryy(x, y + 1, z)) {
    ans.push_back(1);
    return true;
  }
  return false;
}

int main() {
  int t, n, x, y;
  cin >> t;
  while (t--) {
    cin >> n;
    v.clear();
    for (int i = 0; i < n; i++) {
      cin >> x >> y;
      v.push_back({x, y});
    }
    sort(v.begin(), v.end());
    ans.clear();
    if (tryy(0, 0, 0)) {
      cout << "YES\n";
      for (int i = ans.size() - 1; i >= 0; i--) {
        if (ans[i] == 2)
          cout << 'R';
        else
          cout << 'U';
      }
      cout << "\n";
    } else
      cout << "NO\n";
  }
}
