#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n; cin >> n;
    int a[100005], b[100005];
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i] >> b[i];
        sum += a[i];
    }
    sort(a, a + n);
    sort(b, b + n);
    int ans = ((a[n-1] + a[n-2]) <= b[n-1]) || ((a[n-1] + a[n-2]) <= b[n-2]);
    cout << (ans ? "YES" : "NO") << endl;
    return 0;
}
