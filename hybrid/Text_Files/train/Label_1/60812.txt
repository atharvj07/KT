from bisect import bisect_left, bisect_right
def main():
    n = int(input())
    a = list(map(int,input().split()))
    psum = [0]
    for ai in a:psum.append(ai+psum[-1])
    msum = 0
    for l in range(n+1):
        r = bisect_right(psum, psum[l]//2)-1
        if r>=l:
            msum = max(msum, psum[r]-psum[l])
    print(msum)
main()
