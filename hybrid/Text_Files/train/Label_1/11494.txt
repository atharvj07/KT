#include <bits/stdc++.h>

using namespace std;

int main() {
  string x;
  cin >> x;
  int e = 0, cnt = 0;
  for (int i = 0; i < x.length(); i++) {
    if (x[i] == '.') {
      e = x.length() - i - 1;
      x.erase(i, 1);
      break;
    }
  }
  for (int i = x.length() - 1; i >= 0 && x[i] == '0'; i--) {
    x.pop_back();
    e++;
  }
  while (x[0] == '0') {
    x.erase(0, 1);
    cnt++;
  }
  e -= cnt;
  if (e == 0)
    cout << x;
  else if (x.length() > 1)
    cout << x[0] << "." << x.substr(1) << "E" << e;
  else
    cout << x << "E" << e;
}
