import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] enemies = new int[n][4];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1, b = sc.nextInt() - 1;
            enemies[a][0]++;
            enemies[b][0]++;
            enemies[a][enemies[a][0]] = b;
            enemies[b][enemies[b][0]] = a;
        }
        int[] party = new int[n];
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            if (party[i] != 0)
                continue;
            ArrayList<Integer> q = new ArrayList<>();
            q.add(i);
            party[i] = 1;
            while (!q.isEmpty()) {
                int cur = q.remove(0);
                for (int j = 1; j <= enemies[cur][0]; j++) {
                    int next = enemies[cur][j];
                    if (party[next] == party[cur]) {
                        possible = false;
                        break;
                    } else if (party[next] == 0) {
                        q.add(next);
                        party[next] = party[cur] ^ 3;
                    }
                }
            }
        }
        System.out.println(possible ? partyString(party) : -1);
    }

    static String partyString(int[] party) {
        StringBuilder sb = new StringBuilder();
        for (int i : party) {
            sb.append(i == 1 ? "0" : "1");
        }
        return sb.toString();
    }
}
