#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n,x,neg=0,pos=0;
    cin>>n;
    vector<int> a,b,c;
    for(int i=0;i<n;i++){
        cin>>x;
        if(x>0) pos++,a.push_back(x);
        else if(x<0) neg++,b.push_back(x);
        else c.push_back(x);
    }
    cout<<neg<<" ";
    for(int x:b) cout<<x<<" ";
    cout<<"\n"<<pos<<" ";
    for(int x:a) cout<<x<<" ";
    cout<<"\n"<<c.size()<<" ";
    for(int x:c) cout<<x<<" ";
    cout<<"\n";
}

int main(){
    solve();
    return 0;
}
