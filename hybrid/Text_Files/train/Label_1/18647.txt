#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;

    if (n % 2 == 0) {
        cout << n << endl;
        for (int i = 1; i <= n; i += 2)
            cout << i << " " << i + 1 << " ";
    } else {
        cout << n - 1 << endl;
        cout << "1 3 ";
        for (int i = 4; i <= n; i += 2)
            cout << i << " " << i + 1 << " ";
    }
    return 0;
}
