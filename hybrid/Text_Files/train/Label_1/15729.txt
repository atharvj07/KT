#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    long long mx = a[n - 1];
    int ans = 1;
    for (int i = n - 2; i >= 0; i--) {
      if (mx - a[i] >= mx) ans++;
      else break;
    }
    cout << ans << '\n';
  }
}
