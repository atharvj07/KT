import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    if n == 1 or a[n-1] <= 0:
        print(1)
        continue
    l, r = 0, n-1
    while l < r and -a[l] <= a[r]:
        l += 1
    while l < r and a[l] <= a[r]:
        r -= 1
    print(r-l+1)
