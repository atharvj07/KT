#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mx = 1e9 + 7;
int main() {
    ll n;
    cin >> n;
    ll a = 4608, b = 4096, c;
    if (n >= 4608) {
        a = 9216;
        b = 8192;
    }
    while (!(a - b == n)) {
        if (a - b < n) {
            c = a + b;
            a = c;
        } else {
            c = a - b;
            b = c;
        }
    }
    cout << a << " " << b << endl;
    return 0;
}
