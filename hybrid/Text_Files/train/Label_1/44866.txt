import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int[][] a = new int[4][4];
        for (int i = 0; i < 4; i++) {
            String line = sc.next();
            for (int j = 0; j < 4; j++) {
                a[i][j] = line.charAt(j) == '.' ? -1 : line.charAt(j) - '0';
            }
        }
        boolean[][][] dp = new boolean[5][4][4];
        dp[0][0][0] = true;
        for (int t = 0; t < 5; t++) {
            for (int i = 0; i < 4; i++) {
                for (int j = 0; j < 4; j++) {
                    if (dp[t][i][j]) {
                        if (a[i][j] == t) {
                            dp[t + 1][i][j] = true;
                        }
                        if (i + 1 < 4 && a[i + 1][j] == t) {
                            dp[t + 1][i + 1][j] = true;
                        }
                        if (i - 1 >= 0 && a[i - 1][j] == t) {
                            dp[t + 1][i - 1][j] = true;
                        }
                        if (j + 1 < 4 && a[i][j + 1] == t) {
                            dp[t + 1][i][j + 1] = true;
                        }
                        if (j - 1 >= 0 && a[i][j - 1] == t) {
                            dp[t + 1][i][j - 1] = true;
                        }
                        if (i + 1 < 4 && j + 1 < 4 && a[i + 1][j + 1] == t && t <= k) {
                            dp[t + 1][i + 1][j + 1] = true;
                        }
                        if (i + 1 < 4 && j - 1 >= 0 && a[i + 1][j - 1] == t && t <= k) {
                            dp[t + 1][i + 1][j - 1] = true;
                        }
                        if (i - 1 >= 0 && j + 1 < 4 && a[i - 1][j + 1] == t && t <= k) {
                            dp[t + 1][i - 1][j + 1] = true;
                        }
                        if (i - 1 >= 0 && j - 1 >= 0 && a[i - 1][j - 1] == t && t <= k) {
                            dp[t + 1][i - 1][j - 1] = true;
                        }
                    }
                }
            }
        }
        System.out.println(dp[4][3][3] ? "YES" : "NO");
    }
}
