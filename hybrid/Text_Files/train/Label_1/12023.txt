#include <iostream>
#include <vector>
#define ll long long
using namespace std;

int main() {
  ll n;
  cin >> n;
  ll a[n], c[n];
  for (int i = 0; i < n; i++) { cin >> a[i] >> c[i]; }
  vector<ll> v({a[0] - c[0]});
  v.push_back(a[0]);
  for (int i = 1; i < n; i++) {
    v.push_back(max(v.back() + c[i], max(a[i], a[0] - c[0] + c[i])));
  }
  cout << v.back() << endl;
  return 0;
}
