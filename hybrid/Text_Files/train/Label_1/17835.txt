#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    if (n % 2 == 0 && m % 2 == 1) {
        cout << -1 << endl;
        return 0;
    } else if (n % 2 == 1 && m % 2 == 0) {
        if (n == 1) {
            cout << -1 << endl;
            return 0;
        } else if (m % 2 == 0 && n % 2 == 1) {
            if (m <= n) {
                cout << m << endl;
            } else {
                cout << n << endl;
            }
            return 0;
        }
    } else if (n % 2 == 1 && m % 2 == 1) {
        if (n < m) {
            cout << -1 << endl;
        } else if (n % m == 0) {
            cout << n << endl;
        } else {
            if ((n - m) % 2 == 0) {
                cout << n - m + 1 << endl;
            } else {
                cout << n - m << endl;
            }
        }
        return 0;
    } else if (n % 2 == 0 && m % 2 == 0) {
        if (n <= m) {
            cout << n << endl;
        } else if (n % m == 0) {
            cout << n << endl;
        } else {
            if ((n - m) % 2 == 0) {
                cout << n - m << endl;
            } else {
                cout << n - m + 1 << endl;
            }
        }
        return 0;
    }
    return 0;
}
