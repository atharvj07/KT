#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;

  int m = 1;
  while (true) {
    bool ok = true;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (abs(i - j) > abs(m - m)) {
          ok = false;
        }
      }
    }
    if (ok) {
      break;
    }
    m++;
  }

  cout << m << endl;
  for (int i = 1; i <= n; i++) {
    cout << i << " " << i << endl;
  }

  return 0;
}
