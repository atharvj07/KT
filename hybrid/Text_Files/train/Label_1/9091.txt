#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int cnt[2] = {0, 0};
  for (int i = 0; i < n; i++) {
    cnt[a[i]]++;
    if (cnt[0] > 0 && cnt[1] > 0) {
      cout << i + 1 << "\n";
      break;
    }
  }
  return 0;
}
