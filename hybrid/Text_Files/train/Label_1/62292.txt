T = int(input())
for _ in range(T):
    x = input().rstrip()
    y = input().rstrip()

    # Find the rightmost '1' in y
    rightmost_one_y = y.rfind('1')
    
    # The distance from the rightmost '1' in y to the end of y string
    distance_y = len(y) - 1 - rightmost_one_y
    
    # Find the corresponding '1' in x starting from the position found in y
    corresponding_one_x = x.rfind('1', 0, len(x) - distance_y)
    
    # The distance from the found '1' in x to the end of the place where '1' in y should be placed
    k = len(x) - 1 - corresponding_one_x - distance_y
    print(k)
