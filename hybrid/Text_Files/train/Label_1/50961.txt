import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int a = input.nextInt();
    int b = input.nextInt();
    int res = 0;
    for (int i = a; i <= b; i++) {
      res += i % 3 == 0 ? 1 : 2;
    }
    System.out.println(res);
  }
}
