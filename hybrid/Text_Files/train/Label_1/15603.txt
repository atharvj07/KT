#include<bits/stdc++.h>
using namespace std;
int t, n, m, c, x, a[105][105];
bool vis[105][105];

int dx[] = {1, 0, -1, 0};
int dy[] = {0, 1, 0, -1};

void dfs(int i, int j) {
    if (i < 0 || i >= n || j<0 || j>=m || vis[i][j] || a[i][j] != c)
        return;
    vis[i][j] = 1;
    for (int k = 0; k < 4; k++)
        dfs(i + dx[k], j + dy[k]);
}

int main() {
    cin >> n >> m;
    int mx = n * m;
    for (c = 'A'; c <= 'Z'; c++) {
        memset(vis, 0, sizeof(vis));
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!vis[i][j]) {
                    a[i][j] = c;
                    dfs(i, j);
                    cnt++;
                }
            }
        }
        if (cnt == mx)
            break;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++)
            cout << (char) a[i][j];
        cout << endl;
    }
}
