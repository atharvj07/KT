#include <iostream>
#include <string>

int main() {
  std::string s;
  std::cin >> s;
  bool ab = false, ba = false;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == 'A' && s[i + 1] == 'B') ab = true;
    if (s[i] == 'B' && s[i + 1] == 'A') ba = true;
  }
  if (ab && ba) std::cout << "YES" << '\n';
  else std::cout << "NO" << '\n';
  return 0;
}
