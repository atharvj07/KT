#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int &x : p) cin >> x;
    vector<int> ans = p;
    for (int i = 0; i < n - 1; i++) {
      int min_idx = i;
      for (int j = i + 1; j < n; j++) {
        if (p[j] < p[min_idx]) min_idx = j;
      }
      if (min_idx != i) {
        swap(p[i], p[min_idx]);
        ans = p;
      }
    }
    for (int x : ans) cout << x << ' ';
    cout << '\n';
  }
  return 0;
}
