import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();

        int[] sequence = new int[n];
        for (int i = 0; i < n; i++) {
            sequence[i] = a + i * d;
        }

        int m = scanner.nextInt();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt() - 1;
            int z = scanner.nextInt() - 1;

            if (x == 0) {
                reverse(sequence, y, z);
            } else if (x == 1) {
                increment(sequence, y, z);
            } else {
                halve(sequence, y, z);
            }
        }

        int k = scanner.nextInt() - 1;
        System.out.println(sequence[k]);
    }

    private static void reverse(int[] sequence, int start, int end) {
        for (int i = start, j = end; i < j; i++, j--) {
            int temp = sequence[i];
            sequence[i] = sequence[j];
            sequence[j] = temp;
        }
    }

    private static void increment(int[] sequence, int start, int end) {
        for (int i = start; i <= end; i++) {
            sequence[i]++;
        }
    }

    private static void halve(int[] sequence, int start, int end) {
        for (int i = start; i <= end; i++) {
            sequence[i] /= 2;
        }
    }
}
