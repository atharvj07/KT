import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int n = sc.nextInt(), r = sc.nextInt();
            int[] x = new int[n];
            for (int i = 0; i < n; i++) {
                x[i] = sc.nextInt();
            }
            Arrays.sort(x);
            int ans = 0;
            int cur = 0;
            for (int i = 0; i < n; i++) {
                if (cur < x[i]) {
                    cur = x[i] + r;
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
