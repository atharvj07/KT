#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x: a) cin >> x;
    vector<int> b = a;
    sort(b.begin(), b.end());
    int ans = 0;
    for (int i = 0; i < n; i++)
      if (a[i] != b[i]) ans++;
    cout << ans / 2 << endl;
  }
  return 0;
}
