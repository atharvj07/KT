rcp = input()
bc, ws, ch = map(int, input().split())
pb, ps, pc = map(int, input().split())
m = int(input())
for i in rcp:
	if i == 'B': bc -= 1
	elif i == 'S': ws -= 1
	elif i == 'C': ch -= 1
	if bc < 0 or ws < 0 or ch < 0: break
	m -= 1
	bc += 1; ws += 1; ch += 1
	bc += pb; ws += ps; ch += pc
print(m)
