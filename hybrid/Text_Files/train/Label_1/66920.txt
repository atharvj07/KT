h1, a1, c1 = map(int, input().split())
h2, a2 = map(int, input().split())

n = (h2 + a2 - 1) // a1
if h1 - a2 > c1:
    n += (h1 - h2 + a2 - c1 - 1) // (c1 - a2)

print(n)
for i in range(n):
    if i == n - 1 or h1 - a2 > c1:
        print("STRIKE")
        h1 -= a2
        h2 -= a1
    else:
        print("HEAL")
        h1 = min(h1 + c1, h1 - a2 + c1)
