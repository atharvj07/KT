#include <bits/stdc++.h>
#define pb push_back

using namespace std;

typedef long long ll;
typedef pair <int, int> pii;

const int N = 1e6 + 10;
const int M = 1e12 + 10;

vector <pii> spells;
ll m;
int j;

void init() {
}

int main() {
    int q;
    cin >> q >> m;

    j = 0;
    init();
    while(q--) {
        int type, a, b;
        cin >> type >> a >> b;

        if(type == 1) {
            a = (a + j) % N + 1;
            b = (b + j) % N + 1;
            spells.pb({a, b});
        }
        else {
            a = (a + j) % M + 1;
            b = (b + j) % M + 1;

            ll left = m;
            ll t = a;
            bool ok = false;
            for(auto spell : spells) {
                ll dmg = spell.first;
                ll mana_cost = spell.second;
                ll cur = left / mana_cost;
                if(dmg * cur >= b) {
                    ok = true;
                    break;
                }
                left -= cur * mana_cost;
                t -= cur * dmg;
            }

            if(left >= a) {
                ok = true;
            }

            j = ok;
            cout << (ok ? "YES" : "NO") << endl;
        }
    }

    return 0;
}
