#include <bits/stdc++.h>
using namespace std;

int n;
vector<pair<int, int>> v;
unordered_map<int, int> m;

int main() {
    cin >> n;
    v.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    for (int i = 0; i < n; i++) {
        m[v[i].first]++;
        m[v[i].second]++;
    }
    for (int i = 0; i < n; i++) {
        cout << m[v[i].first] - 1 << " " << m[v[i].second] - 1 << "\n";
    }
}
