import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long[] a = new long[6];
        for (int i = 0; i < 6; i++) {
            a[i] = sc.nextLong();
        }
        int n = sc.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        long min = Long.MAX_VALUE;
        long max = Long.MIN_VALUE;
        for (long i : b) {
            long fret = Long.MAX_VALUE;
            for (int j = 0; j < 6; j++) {
                long f = i - a[j];
                if (f >= 0 && f < fret) {
                    fret = f;
                }
            }
            min = Math.min(min, fret);
            max = Math.max(max, fret);
        }
        System.out.println(max - min);
    }
}
