import sys,math

n=int(sys.stdin.readline())
lst=[int(i) for i in sys.stdin.readline().split()]
graph={i:[] for i in range(1,n+1)}
for i in range(int(sys.stdin.readline())):
    a,b=map(int,sys.stdin.readline().split())
    graph[a].append(b)
    graph[b].append(a)

mn=1e9
ans=0
for i in range(1,n+1):
    mx=0
    cnt=0
    st=[(i,0)]
    vis=[False for i in range(n+1)]
    while st:
        node,cost=st.pop()
        if vis[node]==True:continue
        vis[node]=True
        if cost>mx:
            mx=cost
            cnt=1
        elif cost==mx:cnt+=1
        for j in graph[node]:
            if vis[j]==False:
                st.append((j,cost+lst[j-1]))
    if mx<mn:
        mn=mx
        ans=cnt
    elif mx==mn:ans+=cnt
print(mn,ans%1000000007)
