#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int x = 0, y = 0, ans = 0;
        map<pair<int, int>, bool> mp;
        string s;
        cin >> s;
        for (auto& i : s) {
            if (i == 'N') y++;
            if (i == 'S') y--;
            if (i == 'E') x++;
            if (i == 'W') x--;
            if (!mp[{x, y}]) mp[{x, y}] = 1, ans += 5;
            else ans += 1;
        }
        cout << ans << endl;
    }
}
