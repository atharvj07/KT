#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) p[i] = i + 2;
  vector<int> color(n);
  int c = 0;
  for (int i = 0; i < n; i++) {
    if (color[i] != 0) continue;
    for (int j = 0; j < n; j++) {
      bool f = 0;
      if (color[j] != 0) continue;
      for (int k = 2; k <= sqrt(p[j]); k++)
        if (p[j] % k == 0) {
          if (color[i] == color[j] || color[i] == 0) color[i] = ++c;
          if (color[j] == color[i] || color[j] == 0) color[j] = ++c;
          f = 1;
          break;
        }
      if (f) continue;
      if (color[i] == 0) color[i] = ++c;
      if (color[j] == 0) color[j] = ++c;
    }
  }
  cout << c << endl;
  for (int i = 0; i < n; i++) cout << color[i] << " ";
}
