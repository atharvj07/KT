import sys
input = sys.stdin.readline

def ccw(a, b, c):
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]) > 0

def isBetween(a, b, c):
    return (min(a[0], b[0]) <= c[0] <= max(a[0], b[0]) and 
            min(a[1], b[1]) <= c[1] <= max(a[1], b[1]))

for t in range(int(input())):
    n = int(input())
    a = [(0, 0)] * n
    b = [(0, 0)] * n
    for i in range(n):
        a[i] = tuple(map(int, input().split()))
        b[i] = tuple(map(int, input().split()))
    ans = 0
    for i in range(n):
        for j in range(n):
            if i == j: continue
            if ccw(a[i], b[i], a[j]) ^ ccw(a[i], b[i], b[j]):
                if not isBetween(a[i], b[i], a[j]): ans += 1
    print(ans)
