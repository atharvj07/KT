import sys
n = int(input())
a = list(map(int, input().split()))
c = [-1] * n
vis = [False] * n
mx = 0
for i in range(n):
    if vis[i]: continue
    mn = 1e18
    cur = i
    cnt = 0
    while not vis[cur]:
        vis[cur] = True
        mn = min(mn, a[cur])
        mx = max(mx, a[cur])
        cnt += 1
        cur = (cur + 1) % n
    if mn < mx / 2:
        for j in range(i, (i + cnt) % n):
            c[j] = cnt
print(*c)
