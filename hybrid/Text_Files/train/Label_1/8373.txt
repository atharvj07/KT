#include <bits/stdc++.h>
using namespace std;

#define Inf 2000000000  // Infinity

int n, m, s, b, k, h;
int dist[105][105];

int main() {
  cin >> n >> m;
  memset(dist, -1, sizeof(dist));
  for (int i = 1; i <= n; i++) dist[i][i] = 0;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    dist[u][v] = dist[v][u] = 1;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      for (int k = 1; k <= n; k++) {
        if (dist[j][k] != -1 && dist[k][i] != -1 &&
            (dist[j][i] == -1 || dist[j][i] > dist[j][k] + dist[k][i]))
          dist[j][i] = dist[j][k] + dist[k][i];
      }
    }
  }
  cin >> s >> b >> k >> h;
  vector<tuple<int, int, int>> ships;  // ship(loc, atk, fuel)
  for (int i = 0; i < s; i++) {
    int l, a, f;
    cin >> l >> a >> f;
    ships.emplace_back(l, a, f);
  }
  vector<tuple<int, int>> bases;  // base(loc, def)
  for (int i = 0; i < b; i++) {
    int l, d;
    cin >> l >> d;
    bases.emplace_back(l, d);
  }
  sort(ships.begin(), ships.end(),
       [](tuple<int, int, int> &l, tuple<int, int, int> &r) {
         return get<2>(l) > get<2>(r);
       });
  sort(bases.begin(), bases.end(),
       [](tuple<int, int> &l, tuple<int, int> &r) {
         return get<1>(l) > get<1>(r);
       });
  int ans = 0;
  for (auto base : bases) {
    ans += k;
    for (auto ship : ships) {
      int l1 = get<0>(ship);
      int a1 = get<1>(ship);
      int f1 = get<2>(ship);
      int l2 = get<0>(base);
      int d2 = get<1>(base);
      if (a1 >= d2 && f1 >= dist[l1][l2]) {
        ans -= k;
        break;
      }
    }
  }
  int dummy_cost = (s - b) * h;
  ans = min(ans, dummy_cost);
  cout << ans << endl;
  return 0;
}
