#include<iostream>
#include<algorithm>
using namespace std;
int n;
int a[400000+10];
int m=1000000000;
int b[400000+10];
int main()
{
    cin>>n;
    for(int i=1;i<=n;++i)
    {
        cin>>a[i];
        b[i]=a[i];
    }
    sort(a+1,a+n+1);
    int ans=0,p=0,q=0;
    for(int i=1;i<=n;++i)
    {
        int cnt=0;
        for(int j=1;j<=n;++j)
        if(a[i]==b[j]) ++cnt,++j;
        if(cnt*cnt>ans) ans=cnt*cnt,p=cnt,q=cnt;
    }
    cout<<ans<<endl;
    cout<<p<<' '<<q<<endl;
    for(int i=1;i<=p;++i)
    {
        for(int j=1;j<=q;++j)
        {
            cout<<a[i*(j-1)+1]<<' ';
        }
        cout<<endl;
    }
}
