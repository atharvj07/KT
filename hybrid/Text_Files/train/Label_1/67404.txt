n, c = map(int, input().split())
a = list(map(int, input().split()))

l, r = (0, n - 1)
ans = n

while l <= r:
    if a[l] < c and a[r] > c:
        left = a[l]
        right = a[r]
        if right - c < c - left:
            k = c - right
            while l <= r and a[r] > c:
                a[r] -= k
                r -= 1
                if a[r] == c:
                    ans -= 1
            break
        else:
            k = c - left
            while l <= r and a[l] < c:
                a[l] += k
                l += 1
                if a[l] == c:
                    ans -= 1
            break
    elif a[l] > c and a[r] < c:
        l += 1
        r -= 1
        ans -= 2
    else:
        break

print(ans)
