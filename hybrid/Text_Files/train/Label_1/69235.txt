def possible_rating_increments(n):
    increments = set()
    for k in range(1, int(n**0.5) + 1):
        increments.add(n // k)
        if n // k != 0:
            increments.add(k)
    increments.add(0)
    return sorted(list(increments))

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    result = possible_rating_increments(n)
    print(len(result))
    print(*result)
