import sys, math 

def lcm(a, b):
    return a * b // math.gcd(a, b)

def solve():
    t = int(input())
    INF = int(1e9)
    for _ in range(t):
        n, k = map(int, input().split())
        if k % 2 == n % 2:
            for i in range(k-1):
                print(1, end=' ')
            print(n - (k-1))
        else:
            if n % 2 == 1: n -= 1
            n //= 2
            for i in range(k-1):
                print(2, end=' ')
            print(n)
    print()
solve()
