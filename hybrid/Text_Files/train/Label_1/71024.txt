import sys
h, q = map(int, input().split())
ans = [False] * (1 << (h + 1))
for _ in range(q):
    i, l, r, a = map(int, input().split())
    if a == 0 and l <= ans[1 << (i + 1)] <= r:
        print('Game cheated!'); sys.exit()
    if a == 1:
        for j in range(l, r + 1):
            ans[j] = True
c = 0; k = 0
for i in range(2 ** h, 2 ** (h + 1)):
    if ans[i]:
        c += 1; k = i
if c > 1: print('Data not sufficient!'); sys.exit()
print(k)
