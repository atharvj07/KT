#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n), aa(MAXN, 0), bb(MAXN, 0);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        aa[a[i]]++;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
        bb[b[i]]++;
    }
    long long ans = 0;
    for (int i = 0; i < MAXN; i++) {
        if (aa[i] < bb[i]) {
            ans -= (bb[i] - aa[i]);
        } else {
            ans += (aa[i] - bb[i]);
        }
    }
    cout << ans << "\n";
    return 0;
}
