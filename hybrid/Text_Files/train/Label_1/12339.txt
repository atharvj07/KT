#include <bits/stdc++.h>
#define ll long long
int main() {
  ll n, q;
  std::cin >> n >> q;
  std::vector<ll> a(n + 1);
  for (ll i = 1, j = n * n, k = (n * n + 1) / 2; i <= n; ++i) {
    if ((i + j) % 2) {
      a[i] = k;
      --k;
    } else {
      a[i] = j;
      --j;
    }
  }
  while (q--) {
    ll x, y;
    std::cin >> x >> y;
    std::cout << a[x] + y - 1 << "\n";
  }
}
