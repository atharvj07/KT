import sys

# Read input
n, k = map(int, sys.stdin.readline().split())
cards = [sys.stdin.readline().strip() for _ in range(n)]

# Initialize set count
set_count = 0

# Iterate over all triples of cards
for i in range(n):
    for j in range(i+1, n):
        for m in range(j+1, n):
            # Check if the three cards form a set
            is_set = True
            for l in range(k):
                if cards[i][l] != cards[j][l] and cards[i][l] != cards[m][l] and cards[j][l] != cards[m][l]:
                    is_set = False
                    break
            if is_set:
                set_count += 1

# Print the number of sets
print(set_count)
