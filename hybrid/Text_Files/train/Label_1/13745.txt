#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  string t;
  cin >> t;
  string s = "";
  bool a_found = false;
  for (char c : t) {
    if (c == 'a') {
      if (a_found) continue;
      a_found = true;
    } else {
      a_found = false;
    }
    s += c;
  }
  if (a_found) s.pop_back();
  if (s.size() * 2 == t.size())
    cout << s;
  else
    cout << ":(";
  return 0;
}
