#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int a[n], b[n];
        int ua = 0, da = 0, ub = 0, db = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            if (a[i] == 1) ua++;
            if (a[i] == 2) da++;
        }
        for (int i = 0; i < n; i++) {
            cin >> b[i];
            if (b[i] == 1) ub++;
            if (b[i] == 2) db++;
        }
        int ans = max(ua + max(db, ub), ub + max(da, ua));
        cout << ans << endl;
    }
    return 0;
}
