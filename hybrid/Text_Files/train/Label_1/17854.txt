#include<bits/stdc++.h>
using namespace std;
int main()
{
	long long n;
	cin>>n;
	if(n%3==0)
	{
		cout<<n/3<<" "<<n/3<<" "<<n/3;
	}
	else
	{
		long long a,b,c;
		if(n%3==1)
		{
			a=n/3;
			b=n/3;
			c=a+2;
		}
		else if(n%3==2)
		{
			a=n/3+1;
			b=n/3+1;
			c=a+1;
		}
		cout<<a<<" "<<b<<" "<<c;
	}
}
