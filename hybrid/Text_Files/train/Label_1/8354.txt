#include<cstdio>
#include<vector>
#define rep(i,a,n) for(int i=a;i<n;++i)
using namespace std;
int main(){
    int n;
    scanf("%d",&n);
    if(n%3==1){
        puts("NO");
        return 0;
    }
    if(n%3==0)
        printf("YES\n%d ",2*n-1);
    rep(i,1,2*n){
        if(i%3==2)
            printf("%d ",i-1);
        else
            printf("%d ",i);
    }
    puts("");
    return 0;
}
