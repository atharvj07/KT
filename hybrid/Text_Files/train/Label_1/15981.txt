#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<long long, long long>> songs(n);
    for (int i = 0; i < n; i++) {
        cin >> songs[i].first >> songs[i].second;
    }

    vector<long long> time(m);
    for (int i = 0; i < m; i++) {
        cin >> time[i];
    }

    int cur_song = 0;
    long long cur_time = 0;

    for (int i = 0; i < m; i++) {
        while (cur_time + songs[cur_song].second * songs[cur_song].first <= time[i]) {
            cur_time += songs[cur_song].second * songs[cur_song].first;
            cur_song++;
        }
        cout << cur_song + 1 << endl;
    }

    return 0;
}
