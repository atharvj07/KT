#define fastio()                      \
    ios_base::sync_with_stdio(false); \

typedef long long ll;
typedef vector<int> vi;
typedef vector<ll> vll;
typedef pair<int, int> pii;
#define all(x) (x).begin(), (x).end()

int main() {
    fastio();
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vi a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }

        int ans = 0;
        for (int i = 1; i < n - 1; i++) {
            if ((a[i] > a[i - 1] && a[i] > a[i + 1]) ||
                (a[i] < a[i - 1] && a[i] < a[i + 1])) {
                ans++;
            }
        }

        vi b = a;
        sort(all(b));
        int dif = (b[n - 1] - b[0]);
        for (int i = 1; i < n - 1; i++) {
            if ((a[i] > a[i - 1] && a[i] > a[i + 1]) ||
                (a[i] < a[i - 1] && a[i] < a[i + 1])) {
                int temp = a[i];
                a[i] = b[0];
                if ((a[i] > a[i - 1] && a[i] > a[i + 1]) ||
                    (a[i] < a[i - 1] && a[i] < a[i + 1])) {
                    ans--;
                }
                a[i] = b[n - 1];
                if ((a[i] > a[i - 1] && a[i] > a[i + 1]) ||
                    (a[i] < a[i - 1] && a[i] < a[i + 1])) {
                    ans--;
                }
                a[i] = temp;
            }
        }

        cout << ans << endl;
    }

    return 0;
}
