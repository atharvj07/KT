for t in range(int(input())):
    n = int(input())
    s = list(map(int, input().split()))
    ans = 0
    x = -1
    for i in range(n - 1):
        if s[i] <= 1:
            continue
        if x == -1 or x == i + s[i] - 1:
            x = i + s[i] - 1
            ans += 1
        s[i + s[i] - 1] = 1
    print(ans)
