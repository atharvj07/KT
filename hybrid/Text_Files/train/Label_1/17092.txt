#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1000 + 10, mod = 1e9 + 7;
int n, a[maxn], p[maxn], ans;
vector<int> v;
int gcd(int a, int b) {
  if (!b) return a;
  return gcd(b, a % b);
}
int32_t main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  p[1] = a[1];
  for (int i = 2; i <= n; i++) {
    if (gcd(p[i - 1], a[i]) == 1) continue;
    int nc = a[i];
    int last = p[i - 1];
    while (gcd(last, nc) != 1) {
      nc++;
      if (nc > 1e9) {
        cout << -1 << endl;
        return 0;
      }
    }
    p[i] = nc;
    ans++;
    v.push_back(nc);
  }
  cout << ans << endl;
  for (int i = 1; i <= n; i++) { cout << a[i] << " "; }
  for (int i = 0; i < v.size(); i++) { cout << v[i] << " "; }
  return 0;
}
