from collections import defaultdict

n, m = map(int, input().split())
g = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

vis = [0] * (n + 1)
ans = []
for i in range(1, n + 1):
    if vis[i] == 0:
        q = [i]
        while q:
            t = q.pop(0)
            vis[t] = 1
            for u in g[t]:
                if vis[u] == 0:
                    q.append(u)
        ans.append(i)

print(len(ans) - 1)
print(*ans[1:])
