#include <bits/stdc++.h>
using namespace std;
vector<int> lps(string s){
	int n=s.size();
	vector<int> fail(n,0);
	for(int i=1; i<n; i++){
		int j=fail[i-1];
		while(j>0 && s[i]!=s[j])
			j=fail[j-1];
		if(s[i]==s[j])
			fail[i]=j+1;
	}
	return fail;
}
int KMP(string p,string t){
	int ans=0,n=p.size(),m=t.size();
	vector<int> lps_p=lps(p);
	for(int i=0,j=0; i<m; i++){
		while(j>0 && t[i]!=p[j])
			j=lps_p[j-1];
		if(t[i]==p[j]){
			j++;
			if(j==n)
				ans++,j=lps_p[j-1];
		}
	}
	return ans;
}
int main(){
	string p,t; cin>>p>>t;
	cout<<p.size()+t.size()-2*KMP(p,t)<<endl;
}
