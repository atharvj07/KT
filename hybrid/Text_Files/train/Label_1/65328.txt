n, m = map(int, input().split())
a = [[int(j) for j in input().split()] for i in range(n)]
def p(x):
    return x > 1 and all(x % i for i in range(2, int(x**0.5) + 1))
for i in a:
    if all(map(p, i)):
        print(0); exit()
for j in range(m):
    col = [a[i][j] for i in range(n)]
    if all(map(p, col)):
        print(0); exit()
ans = 10**9
for i in range(n):
    for j in range(m):
        if p(a[i][j] + 1):
            ans = min(ans, a[i][j])
print(ans)
