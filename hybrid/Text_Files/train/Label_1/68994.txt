lever_description = input().strip()
pivot_index = lever_description.index("^")

left_moment = 0
right_moment = 0

for i in range(pivot_index):
    if lever_description[i].isdigit():
        left_moment += (pivot_index - i) * int(lever_description[i])

for i in range(pivot_index + 1, len(lever_description)):
    if lever_description[i].isdigit():
        right_moment += (i - pivot_index) * int(lever_description[i])

if left_moment > right_moment:
    print("left")
elif left_moment < right_moment:
    print("right")
else:
    print("balance")
