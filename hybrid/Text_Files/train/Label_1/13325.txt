#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+10;
vector<int>gr[maxn], col(maxn, -1), vis(maxn);
int mx = 0;
bool dfs(int src, int p, int c) {
    vis[src] = 1;
    col[src] = c;
    for(auto &u : gr[src]) {
        if(u != p) {
          if(vis[u] == 1) {
              if(col[u] == col[src]) return false;
          } else {
              if(!dfs(u, src, c^1)) return false;
          }
        }
    }
    return true;
}
int main() {
    int n; cin >> n;
    int p;
    for(int i=1; i<=n; i++) {
        cin >> p;
        if(p) mx++;
    }
    for(int i=1; i<n; i++) {
        int u, v; cin >> u >> v;
        gr[u].push_back(v);
        gr[v].push_back(u);
    }
    if(dfs(1, 1, 0) == false) {
        cout << -1 << endl;
        return 0;
    }
    cout << n - 1 - mx << endl;
    return 0;
}
