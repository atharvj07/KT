import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length();
        int[][] dp = new int[n][2];

        if (s.charAt(0) == '?') {
            dp[0][0] = dp[0][1] = 0;
        } else {
            dp[0][s.charAt(0) - '0'] = 0;
            dp[0][1 - (s.charAt(0) - '0')] = 1;
        }

        for (int i = 1; i < n; i++) {
            char c = s.charAt(i);
            if (c == '?') {
                dp[i][0] = Math.min(dp[i - 1][0], dp[i - 1][1]) + 1;
                dp[i][1] = Math.min(dp[i - 1][0], dp[i - 1][1]) + 1;
            } else {
                dp[i][c - '0'] = dp[i - 1][c - '0'];
                dp[i][1 - (c - '0')] = dp[i - 1][c - '0'] + 1;
            }
        }
        System.out.println(dp[n - 1][0]);
    }
}
