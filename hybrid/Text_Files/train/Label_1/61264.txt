from math import sqrt

def is_vowelly_possible(k):
    for n in range(5, int(sqrt(k)) + 1):
        if k % n == 0 and k // n >= 5:
            return n, k // n
    return -1, -1

def generate_vowelly_word(n, m):
    vowels = ['a', 'e', 'i', 'o', 'u']
    word = ''
    for i in range(n):
        for j in range(m):
            word += vowels[(i + j) % 5]
    return word

k = int(input())
n, m = is_vowelly_possible(k)
if n == -1:
    print(-1)
else:
    print(generate_vowelly_word(n, m))
