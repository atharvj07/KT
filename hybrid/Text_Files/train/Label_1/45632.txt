import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        long k = sc.nextLong();

        while (k-- > 0) {
            long tmp = a;
            a = b + c;
            b = c + tmp;
            c = tmp + a;
        }

        long ans = a - b;
        if (Math.abs(ans) > 1e18) {
            System.out.println("Unfair");
        } else {
            System.out.println(ans);
        }
    }
}
