import sys
def gcd(a, b):
    if(a == 0):
        return b
    return gcd(b % a, a)
n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
ans = []
for i in a:
    if len(ans) == 0 or gcd(i, ans[-1]) == 1:
        ans.append(i)
    else:
        for j in range(2, int(i ** 0.5) + 1):
            if i % j == 0:
                ans.append(j)
                ans.append(i // j)
                break
if len(ans) == n:
    print(0)
    print(*ans)
else:
    print(len(a) - len(ans))
    print(*ans)
