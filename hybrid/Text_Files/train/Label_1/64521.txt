t = int(input())
for _ in range(t):
    m = int(input())
    top_row = list(map(int, input().split()))
    bottom_row = list(map(int, input().split()))

    for i in range(1, m):
        top_row[i] += top_row[i-1]
        bottom_row[i] += bottom_row[i-1]

    max_score = 0
    for i in range(m):
        score = max(top_row[m-1] - top_row[i], bottom_row[i-1] if i > 0 else 0)
        if max_score == 0 or score < max_score:
            max_score = score

    print(max_score)
