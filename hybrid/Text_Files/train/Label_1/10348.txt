#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        if (abs(n) % 2 == k % 2)
            cout << abs(n - k) / 2 << endl;
        else
            cout << abs(n - k) / 2 + 1 << endl;
    }
    return 0;
}
