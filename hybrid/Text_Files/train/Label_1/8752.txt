#include <iostream>

using namespace std;

int main() {
  string n;
  cin >> n;
  int sum = 0, ans = 0;
  for (auto c : n) {
    int d = c - '0';
    sum += d;
    ans = max(ans, d);
  }
  if (ans == 0) {
    ans = 9;
    n[n.size() - 1] = '9';
  }
  while (sum > ans) {
    ans *= 10;
    ans += 9;
  }
  cout << ans << endl;
  return 0;
}
