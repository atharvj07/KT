def chk(l, r):
    for i in range(m):
        if a[l:r + 1].count(i + 1) != k[i]:
            return False
    return True
    

n, m = list(map(int, input().strip().split()))
a = list(map(int, input().strip().split()))
k = list(map(int, input().strip().split()))
flag = False
for i in range(n):
    for j in range(i, n):
        if chk(i, j):
            print('YES')
            flag = True
            break
    if flag:
        break
if not flag:
    print('NO')
