#include<iostream>
#include<cmath>
using namespace std;
int main() {
    int m, t, r;
    cin >> m >> t >> r;
    int w[m];
    for (int i = 0; i < m; i++) { cin >> w[i]; }
    int ans = ceil((w[0] + t - 1) / t); // first ghost
    int last = w[0]; // last ghost
    for (int i = 1; i < m; i++) {
        if (w[i] - last > t) { // last candle is off
            ans += ceil((w[i] + t - 1) / t);
        }
        last = w[i];
    }
    cout << (ans >= r ? ans : -1);
    return 0;
}
