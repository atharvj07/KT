#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        m[a] = b;
    }
    bool flag = false;
    for (auto it = m.begin(); it != m.end(); it++) {
        auto jt = it;
        jt++;
        if (jt != m.end() && it->first < jt->first && it->second > jt->second) {
            flag = true;
        }
    }
    if (flag)
        cout << "Happy Alex" << endl;
    else
        cout << "Poor Alex" << endl;
    return 0;
}
