#include <bits/stdc++.h>
#define maxn 200010
using namespace std;
typedef long long ll;

int n, k, par[maxn], sum[maxn], anc[maxn][6], dis[maxn], ans;
vector<int> adj[maxn];

void dfs(int u, int p) {
    par[u] = p;
    dis[u] = (p ? dis[p] + 1 : 0);
    sum[u] = 1;
    for (int i = 0; i < adj[u].size(); i++) {
        int v = adj[u][i];
        if (v == p) continue;
        dfs(v, u);
        sum[u] += sum[v];
    }
}

void calc_dis() {
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= n; j++) {
            anc[j][i] = anc[anc[j][i - 1]][i - 1];
        }
    }
}

int lca(int a, int b) {
    if (dis[a] < dis[b]) {
        swap(a, b);
    }
    int diff = dis[a] - dis[b];
    for (int i = 0; i < k; i++) {
        if ((1 << i) & diff) {
            a = anc[a][i];
        }
    }
    if (a == b) return a;
    for (int i = k - 1; i >= 0; i--) {
        if (anc[a][i] != anc[b][i]) {
            a = anc[a][i], b = anc[b][i];
        }
    }
    return anc[a][0];
}

void get_ans(int u, int p) {
    for (int i = 0; i < adj[u].size(); i++) {
        int v = adj[u][i];
        if (v == p) continue;
        get_ans(v, u);
        ans += sum[v] * (n - sum[v]) - sum[lca(u, v)] * (n - sum[lca(u, v)]);
    }
}

int main() {
    int a, b;
    scanf("%d%d", &n, &k);
    for (int i = 0; i < n - 1; i++) {
        scanf("%d%d", &a, &b);
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    dfs(1, 0);
    for (int i = 1; i <= n; i++) {
        anc[i][0] = par[i];
    }
    calc_dis();
    get_ans(1, 0);
    printf("%d", ans);
    return 0;
}
