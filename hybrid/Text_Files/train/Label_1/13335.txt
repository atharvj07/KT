#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i < n; i++) { cin >> a[i]; }
    a[n] = 0;
    int ans = 0;
    for (int i = 1; i < n; i++) {
        int t = -a[i + 1];
        if (t != 0) {
            ans++;
            if (a[i] > 0) {
                a[i + 1] += (a[i] < t) ? a[i] : t;
                a[i] -= (a[i] < t) ? a[i] : t;
            } else {
                a[i] += (a[i] > t) ? a[i] : t;
                a[i + 1] -= (a[i] > t) ? a[i] : t;
            }
        }
    }
    cout << ans << "\n";
    return 0;
}
