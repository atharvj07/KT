#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k, p, ans = 0, id = 0, pos = 0;
    vector<int> a(1001), b(2001);
    cin >> n >> k >> p;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= k; i++) {
        cin >> b[i];
    }
    sort(a.begin() + 1, a.begin() + n + 1);
    sort(b.begin() + 1, b.begin() + k + 1);
    for (int i = 1; i <= n; i++) {
        while (id < k && b[id + 1] < a[i]) id++;
        if (id >= k || a[i] == b[id + 1]) pos = max(pos, a[i]);
        else {
            ans += b[id + 1] - a[i];
            pos = b[id + 1];
            id++;
        }
    }
    ans += p - pos;
    cout << ans << endl;
    return 0;
}
