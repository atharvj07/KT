#include <iostream>
#include <string>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  for (int i = 0; i < k; i++) {
    for (char c = 'a'; c <= 'z'; c++) {
      if (s.find(c) != -1) {
        s.erase(s.find(c), 1);
        break;
      }
    }
  }
  cout << s;
  return 0;
}
