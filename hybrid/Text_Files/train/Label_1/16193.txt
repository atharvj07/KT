#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m; cin >> n >> m;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  sort(a.begin(), a.end()); sort(b.begin(), b.end());
  int ans = (b[0] - a[0] + m) % m;
  for (int i = 0; i < n; i++) {
    if ((b[i] - a[i] + m) % m != ans) { ans = -1; break; }
  }
  cout << ans << endl;
  return 0;
}
