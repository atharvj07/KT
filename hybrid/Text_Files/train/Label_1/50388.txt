import java.util.*;

public class Main {
    public static void main(String[] args) {
        long start = System.currentTimeMillis();
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        ArrayList<Integer>[] edges = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            edges[a].add(b);
            edges[b].add(a);
        }
        boolean[] visited = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                if (!dfs(i, -1, edges, visited, new ArrayList<>())) {
                    System.out.println("No solution");
                    return;
                }
            }
        }
        long end = System.currentTimeMillis();
        System.out.println((end - start) + "ms");
    }

    private static boolean dfs(int node, int parent, ArrayList<Integer>[] edges, boolean[] visited, ArrayList<Integer> path) {
        visited[node] = true;
        path.add(node);
        if (path.size() == 3) {
            System.out.println(path.get(0) + " " + path.get(1) + " " + path.get(2));
            return true;
        }
        for (int neighbor : edges[node]) {
            if (neighbor != parent) {
                if (dfs(neighbor, node, edges, visited, path)) {
                    return true;
                }
            }
        }
        path.remove(path.size() - 1);
        return false;
    }
}
