#include <bits/stdc++.h>
using i64 = long long;
using namespace std;
i64 dp[2010][20][2][2][2];
inline i64 add(i64 a, i64 b) { return (a + b) % 1000000007; }
inline i64 mul(i64 a, i64 b) { return (a * b) % 1000000007; }
i64 go(int pos, int prv, bool st, bool tk, bool leadzero, vector<int> &a) {
    if (pos == (int)a.size()) {
        if (!tk) return 0;
        return 1;
    }
    i64 &ans = dp[pos][prv][st][tk][leadzero];
    if (~ans) return ans;
    ans = 0;
    int lastd = (prv == 0 ? 9 : prv - 1);
    int stt = (st ? 0 : a[pos]);
    for (int i = stt; i <= lastd; i++) {
        if (leadzero && i == 0) continue;
        bool nw_tk = tk;
        if (i == prv) nw_tk = 1;
        ans = add(ans, go(pos + 1, i, st && i == a[pos], nw_tk, leadzero && i == 0, a));
    }
    return ans;
}
i64 solve(i64 b, i64 m, i64 d) {
    vector<int> A, B;
    while (b) {
        A.push_back(b % 10);
        b /= 10;
    }
    reverse(A.begin(), A.end());
    while (m) {
        B.push_back(m % 10);
        m /= 10;
    }
    reverse(B.begin(), B.end());
    return (go(0, 0, 1, 0, 1, B) - go(0, 0, 1, 0, 1, A) + 1000000007) % 1000000007;
}
int main() {
    i64 m, d;
    cin >> m >> d;
    string a, b;
    cin >> a >> b;
    a = '0' + a;
    b = '0' + b;
    i64 ans = solve(stoll(b), m, d);
    ans = (ans - solve(stoll(a) - 1, m, d) + 1000000007) % 1000000007;
    cout << ans << endl;
    return 0;
}
