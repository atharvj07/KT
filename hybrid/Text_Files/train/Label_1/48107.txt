import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            sc.nextInt();
            char[] s = sc.next().toCharArray();
            int ans = 0;
            for (int j = 0; j < s.length; j += 2) {
                if (s[j] == s[j + 1]) ans++;
            }
            System.out.println(ans);
        }
    }
}
