#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, r;
  cin >> n >> r;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0, dist = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) {
      dist++;
      if (dist > r) {
        ans = -1;
        break;
      }
    } else
      dist = 0;
  }
  cout << ans << endl;
}
