MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def modInverse(a, p):
    return power(a, p - 2, p)

def nCrModPFermat(n, r, p):
    if n < r:
        return 0
    if r == 0:
        return 1
    fac = [1 for i in range(n + 1)]
    for i in range(2, n + 1):
        fac[i] = (fac[i - 1] * i) % p
    return (fac[n] * modInverse(fac[r], p) % p * modInverse(fac[n - r], p) % p) % p

def solve(f, w, h):
    if w == 0:
        return 1
    if f == 0:
        return 1 if w > h else 0
    total, like = 0, 0
    for i in range(f + 1):
        for j in range(1, w + 1):
            total += (nCrModPFermat(f - 1, i - 1, MOD) * nCrModPFermat(w - 1, j - 1, MOD)) % MOD
            total %= MOD
            if j > h:
                like += (nCrModPFermat(f - 1, i - 1, MOD) * nCrModPFermat(w - h - 1, j - h - 1, MOD)) % MOD
                like %= MOD
    return like * modInverse(total, MOD) % MOD

f, w, h = map(int, input().split())
print(solve(f, w, h))
