s = input()
t = input()
if t[0] not in s:
    print(-1)
else:
    st = set(s)
    cnt = {}
    for i in st:
        cnt[i] = [j for j in range(len(s)) if s[j] == i]
    last = {}
    cur, ans = [], []
    for j in t:
        if j not in st:
            print(-1)
            exit()
        if not cur or j != s[cur[-1]]:
            cur.append(cnt[j][0])
        else:
            cur.append(-cnt[j][0] - 1)
        if j in last and last[j] == cur[-1]:
            ans.pop()
            cur.pop()
        last[j] = cur[-1]
    print(len(ans))
    for i in ans: print(i // -2 if i < 0 else i, i % 2 == 1)
