import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        for (int i = 1; i < n; i++) {
            s = s + sc.next();
        }
        int max = 0;
        int cur = 0;
        char prv = s.charAt(0);
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == prv) {
                cur++;
            } else {
                prv = s.charAt(i);
                max = Math.max(max, cur);
                cur = 1;
            }
        }
        max = Math.max(max, cur);
        System.out.println(max);
    }
}
