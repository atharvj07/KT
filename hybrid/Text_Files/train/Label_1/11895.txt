#include <bits/stdc++.h>
using namespace std;
int main(){
    int n,m,mn[101],mx[101];
    cin>>n>>m;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            int x; cin>>x;
            mn[i]=min(mn[i],x),mx[j]=max(mx[j],x);
        }
    }
    int ans=1e9;
    for(int i=1;i<=n;++i)for(int j=1;j<=m;++j)ans=min(ans,mn[i]+mx[j]);
    cout<<ans<<endl;
    return 0;
}
