import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] minX = new int[n];
    int[] maxX = new int[n];
    for (int i = 0; i < n; i++) {
      minX[i] = sc.nextInt();
      maxX[i] = sc.nextInt();
    }
    Arrays.sort(minX);
    Arrays.sort(maxX);
    int min = Integer.MAX_VALUE;
    int mi = 0, mj = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i; j < n; j++) {
        if (minX[j] >= maxX[i]) {
          int cnt = j - i + 1;
          if (cnt < min) {
            min = cnt;
            mi = i;
            mj = j;
          }
          break;
        }
      }
    }
    System.out.println(min - 1);
  }
}
