import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt(), k = input.nextInt();
        long[][] a = new long[n + 2][m + 2];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                a[i][j] = input.nextInt();
            }
        }
        long ans = 0;
        int x = 0, y = 0;
        for (int i = k; i <= n - k + 1; i++) {
            for (int j = k; j <= m - k + 1; j++) {
                long val = a[i][j];
                for (int p = i - k; p <= i + k; p++) {
                    for (int q = j - k; q <= j + k; q++) {
                        if (p != i || q != j) {
                            val = Math.max(val, a[p][q]);
                        }
                    }
                }
                if (val > ans) {
                    ans = val;
                    x = i;
                    y = j;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
