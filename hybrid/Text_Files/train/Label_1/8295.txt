#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
typedef long long ll;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    ll nq;
    cin >> nq;
    while (nq--) {
        ll nx, ny, k;
        cin >> nx >> ny >> k;
        ll d = abs(nx - ny);
        if (k - d < 0 || (k - d) % 2 != 0) {
            cout << "-1" << "\n";
        } else {
            cout << (k - d) / 2 << "\n";
        }
    }
    return 0;
}
