n, m, k = map(int, input().split())
app_order = list(map(int, input().split()))
launch_order = list(map(int, input().split()))

app_positions = {app_order[i]: i for i in range(n)}
gestures = 0

for app in launch_order:
    pos = app_positions[app]
    gestures += pos // k + 1
    if pos > 0:
        prev_app = app_order[pos - 1]
        app_order[pos], app_order[pos - 1] = app_order[pos - 1], app_order[pos]
        app_positions[app], app_positions[prev_app] = app_positions[prev_app], app_positions[app]

print(gestures)
