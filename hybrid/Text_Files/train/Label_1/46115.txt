import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        char[][] board = new char[n][];
        for (int i = 0; i < n; i++) {
            board[i] = input.next().toCharArray();
        }
        boolean[][] visited = new boolean[n][n];
        boolean possible = false;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !visited[i][j]) {
                    possible = dfs(board, visited, i, j);
                    if (possible) {
                        break;
                    }
                }
            }
            if (possible) {
                break;
            }
        }
        System.out.println(possible ? "YES" : "NO");
    }

    private static boolean dfs(char[][] board, boolean[][] visited, int i, int j) {
        if (i < 0 || i >= board.length || j < 0 || j >= board[0].length || board[i][j] == '#' || visited[i][j]) {
            return false;
        }
        visited[i][j] = true;
        boolean result = dfs(board, visited, i - 1, j) || dfs(board, visited, i + 1, j) || dfs(board, visited, i, j - 1) || dfs(board, visited, i, j + 1);
        if (result) {
            board[i][j] = '#';
        }
        return result;
    }
}
