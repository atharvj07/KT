import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            ans += a[i];
        }
        int max = 0;
        int l = 0;
        int r = 0;
        int c = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                c++;
            }
            while (c > k) {
                if (a[l] == 0) {
                    c--;
                }
                l++;
            }
            max = Math.max(max, i - l + 1);
            r = i;
        }
        ans = Math.max(ans, max);
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            if (i < l || i > r) {
                System.out.print(a[i] + " ");
            } else {
                System.out.print(1 + " ");
            }
        }
    }
}
