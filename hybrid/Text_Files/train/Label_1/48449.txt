import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int t = in.nextInt();
    while (t-- > 0) {
      String s = in.next();
      if (s.length() % 2 == 1 || s.charAt(0) != s.charAt(s.length() - 1)) {
        System.out.println("DA");
      } else {
        System.out.println("NET");
      }
    }
  }
}
