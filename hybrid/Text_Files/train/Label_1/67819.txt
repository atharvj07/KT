n = int(input())
a = list(map(int, input().split()))

ans = []
a.sort()

for i in range(n - 1):
    if a[i] < 0:
        if a[i + 1] >= 0:
            ans.append([1, i + 2, i + 1])
        else:
            ans.append([1, i + 1, i + 2])
    elif a[i] == 0:
        ans.append([2, i + 1])
        break
    else:
        if a[i + 1] >= 0:
            ans.append([1, i + 1, i + 2])
        else:
            ans.append([1, i + 2, i + 1])

for i in ans:
    print(*i)
