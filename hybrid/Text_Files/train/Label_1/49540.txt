import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = i;
        }
        for (int i = 0; i < n - 1; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            int px = find(p, x);
            int py = find(p, y);
            if (px != py) {
                p[px] = py;
            }
        }
        HashSet<Integer> set = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            int px = find(p, i);
            if (!set.contains(px)) {
                set.add(px);
            }
        }
        int[] arr = new int[n + 1];
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            int px = find(p, i);
            if (!set.contains(px)) {
                arr[++cnt] = i;
            }
        }
        for (int i = 1; i <= n; i++) {
            int px = find(p, i);
            if (set.contains(px)) {
                arr[++cnt] = i;
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(arr[i] + " ");
        }
    }

    static int find(int[] p, int x) {
        if (p[x] == x) {
            return x;
        }
        return p[x] = find(p, p[x]);
    }
}
