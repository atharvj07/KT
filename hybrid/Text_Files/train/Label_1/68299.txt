n = int(input())
a = list(map(int, input().split()))

divisors = {}

for x in a:
    for d in range(1, x + 1):
        if x % d == 0:
            if d not in divisors:
                divisors[d] = []
            divisors[d].append(x)

colors = 0
used = set()

for x in sorted(a, reverse=True):
    if x not in used:
        min_div = min([d for d in divisors if x in divisors[d]])
        for item in divisors[min_div]:
            used.add(item)
        colors += 1

print(colors)
