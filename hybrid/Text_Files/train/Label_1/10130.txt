#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

int main() {
  int n;
  ll a = 0, b = 0;
  cin >> n;
  vector<ll> va(n), vb(n);
  for (int i = 0; i < n; i++) { cin >> va[i]; }
  for (int i = 0; i < n; i++) { cin >> vb[i]; }
  sort(va.begin(), va.end());
  sort(vb.begin(), vb.end());
  reverse(va.begin(), va.end());
  reverse(vb.begin(), vb.end());
  int ia = n - 1, ib = n - 1;
  while (ia >= 0 && ib >= 0) {
    if (va[ia] > vb[ib]) {
      a += va[ia];
      ia--;
    } else {
      ib--;
    }
    if (ia >= 0 && ib >= 0) {
      if (va[ia] < vb[ib]) {
        b += vb[ib];
        ib--;
      } else {
        ia--;
      }
    }
  }
  cout << a - b << endl;
}
