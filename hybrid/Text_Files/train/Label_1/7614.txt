#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
    ll n; cin >> n;
    string s = to_string(n);
    ll cnt = 0, last = -1, pre = -1, len = s.size();
    for (ll i = 0; i < len; i++) {
        if (s[i] > '0') {
            cnt++;
            if (last == -1) {
                last = s[i] - '0';
            } else {
                pre = last, last = s[i] - '0';
            }
        }
    }
    if (last % 10 == 0 || ((last % 10 == 2 || last % 10 == 7) && pre > 0)) {
        if ((last % 10 == 2 || last % 10 == 7) && pre > 0) {
            cout << cnt - 1 << endl;
        } else {
            cout << cnt << endl;
        }
    } else {
        cout << "-1" << endl;
    }
    return 0;
}
