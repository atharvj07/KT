#include <bits/stdc++.h>

using namespace std;
const int N = 105;
int mat[N][N], dir[N][N], vis[N][N];
bool valid = true;
int n, dr[]{-1, 1, 0, 0}, dc[]{0, 0, -1, 1};

bool check(int r, int c) {
    int c1 = (r + n - 1) % n + 1;
    int c2 = (c + n - 1) % n + 1;
    return !vis[r][c] && vis[r][c1] && vis[c2][c] && mat[r][c] != mat[r][c1] && mat[r][c] != mat[c2][c];
}

void dfs(int r, int c, int d) {
    vis[r][c] = vis[r][(c + n - 1) % n + 1] = vis[(r + n - 1) % n + 1][c] = 1;
    mat[r][c] = d;
    for (int i = 0; i < 4; i++) {
        int nr = r + dr[i], nc = c + dc[i];
        if (nr >= 0 && nr < n && nc >= 0 && nc < n && check(nr, nc)) {
            dfs(nr, nc, d + 1);
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        valid = true;
        memset(vis, 0, sizeof vis);
        memset(dir, 0, sizeof dir);
        memset(mat, 0, sizeof mat);
        if (n % 2) {
            int mid = (n + 1) / 2;
            dfs(mid - 1, mid - 1, 1);
        } else {
            int m = n / 2;
            for (int r = 0; r < m; r++) {
                for (int c = 0; c < m; c++) {
                    if (check(r, c)) {
                        dfs(r, c, 1);
                        check(r, c + m);
                        dfs(r, c + m, m * m + 1);
                        check(r + m, c);
                        dfs(r + m, c, 2 * m * m + 1);
                    } else {
                        valid = false;
                    }
                }
            }
        }
        if (valid) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    cout << mat[i][j] << " ";
                }
                cout << '\n';
            }
        } else {
            cout << -1 << '\n';
        }
    }
    return 0;
}
