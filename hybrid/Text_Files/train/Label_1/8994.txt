#include <iostream>
using namespace std;

int main() {
  long long p, x, ans = 0;
  cin >> p >> x;
  if (x == 1) {
    cout << "Impossible" << endl;
    return 0;
  }
  for (long long i = 1; i <= 9 && p > 0; i++) {
    if (i * x < 10) {
      ans = ans * 10 + i;
      p--;
    }
  }
  for (long long i = 1; i <= 9 && p > 0; i++) {
    ans = ans * 10 + x;
    p--;
  }
  if (p > 0) {
    cout << "Impossible" << endl;
  } else
    cout << ans << endl;
  return 0;
}
