import java.util.Scanner;

public class Dryer {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Read student segments
        int[][] segments = new int[n][2];
        for (int i = 0; i < n; i++) {
            segments[i][0] = scanner.nextInt();
            segments[i][1] = scanner.nextInt();
        }

        // Sort segments by left endpoint
        sortSegments(segments);

        // Find Alexey's segment
        int alexeyLeft = 0;
        int alexeyRight = 100;
        for (int i = 0; i < n; i++) {
            if (segments[i][0] <= alexeyLeft && segments[i][1] >= alexeyRight) {
                alexeyLeft = segments[i][0];
                alexeyRight = segments[i][1];
            }
        }

        // Calculate length of unused segments
        int unusedLength = 0;
        int currentRight = 0;
        for (int i = 0; i < n; i++) {
            if (segments[i][1] <= alexeyLeft || segments[i][0] >= alexeyRight) {
                unusedLength += segments[i][1] - segments[i][0];
            } else if (segments[i][0] > currentRight) {
                unusedLength += segments[i][0] - currentRight;
                currentRight = segments[i][1];
            }
        }

        // Print the result
        System.out.println(unusedLength);
    }

    // Sort segments by left endpoint
    private static void sortSegments(int[][] segments) {
        for (int i = 0; i < segments.length - 1; i++) {
            for (int j = i + 1; j < segments.length; j++) {
                if (segments[i][0] > segments[j][0]) {
                    int[] temp = segments[i];
                    segments[i] = segments[j];
                    segments[j] = temp;
                }
            }
        }
    }
}
