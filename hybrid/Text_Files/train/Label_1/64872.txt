import sys
import collections
sys.setrecursionlimit(10**9)

n = int(input())
adj = collections.defaultdict(list)
for _ in range(n - 1):
  u, v = map(int, input().split())
  adj[u].append(v)
  adj[v].append(u)

costs, b, c = [], [], []
for _ in range(n):
  tmp = list(map(int, input().split()))
  costs.append(tmp[0])
  b.append(tmp[1])
  c.append(tmp[2])

dp = collections.defaultdict(int)

def dfs1(u, p):
  dp[(u, -1)] = 0
  for v in adj[u]:
    if v != p:
      dfs1(v, u)
      dp[(u, 0)] += min(dp[(v, 0)], dp[(v, 1)] + costs[v - 1] * (c[v - 1] != b[v - 1]))
      dp[(u, 1)] += min(dp[(v, 0)] + costs[v - 1] * (c[v - 1] != b[v - 1]), dp[(v, 1)])

dfs1(1, 1)

def dfs2(u, p, par):
  for v in adj[u]:
    if v != p:
      if par == 0:
        dp[(v, 0)] = dp[(u, 0)]
        dp[(v, 1)] = dp[(u, 0)] + costs[v - 1] * (c[v - 1] != b[v - 1])
      elif par == 1:
        dp[(v, 0)] = dp[(u, 0)] + costs[v - 1] * (c[v - 1] != b[v - 1])
        dp[(v, 1)] = dp[(u, 1)]
      else:
        dp[(v, 0)] = min(dp[(u, 0)], dp[(u, 1)] + costs[v - 1] * (c[v - 1] != b[v - 1]))
        dp[(v, 1)] = min(dp[(u, 0)] + costs[v - 1] * (c[v - 1] != b[v - 1]), dp[(u, 1)])
      dfs2(v, u, dp[(v, 0)] <= dp[(v, 1)])

dfs2(1, 1, -1)

print(min(dp[(1, 0)], dp[(1, 1)]) if min(dp[(1, 0)], dp[(1, 1)]) != float('inf') else -1)
