#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  int x = -1, y = -1;
  for (int i = 0; i < n; ++i) {
    if (s[i] == '(')
      ++cnt;
    else {
      --cnt;
      if (cnt < 0 && x == -1) {
        x = i;
        y = i + 1;
      } else if (cnt < 0) {
        y = i + 1;
      }
    }
  }
  if (cnt == 0)
    cout << "Yes" << endl;
  else if (cnt == -1 && x != -1 && y != -1)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
  return 0;
}
