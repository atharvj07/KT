#include<bits/stdc++.h>
using namespace std;
#define ll long long
int32_t main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll t;
    cin>>t;
    while(t--)
    {
        ll n;
        cin>>n;
        vector<ll>v;
        for(ll i=1;i<=2*n;i++)
        {
            ll x;
            cin>>x;
            v.push_back(x);
        }
        sort(v.begin(),v.end());
        ll median=v[n];
        ll ans=LLONG_MAX;
        for(ll i=0;i<=n;i++)
        {
            if((n-i)%2==1)
            {
                ll x=v[i+(n-i)/2];
                ans=min(ans,abs(x-median));
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
