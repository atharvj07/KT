import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    ans = []
    for i in range(n):
        if a[i] < b[i]:
            ans.append(a[i])
        else:
            ans.append(b[i])
    print(max(ans))
