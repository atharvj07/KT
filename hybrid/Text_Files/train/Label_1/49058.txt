import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long f = sc.nextLong(), t = sc.nextLong(), t0 = sc.nextLong();
        long a1 = sc.nextLong(), t1 = sc.nextLong(), p1 = sc.nextLong();
        long a2 = sc.nextLong(), t2 = sc.nextLong(), p2 = sc.nextLong();

        if (t0 * f <= t) {
            System.out.println(0);
            return;
        }

        long ans = Long.MAX_VALUE;
        for (long i = 0; i * t0 <= t; i++) {
            long r = f - i;
            long r1 = (r % a1 == 0) ? r / a1 : (r % a1 == 0) ? r / a1 : r / a1 + 1;
            long r2 = (r % a2 == 0) ? r / a2 : (r % a2 == 0) ? r / a2 : r / a2 + 1;
            ans = Math.min(ans, i * p1 + r1 * t1 + r2 * t2);
        }
        System.out.println(ans == Long.MAX_VALUE ? -1 : ans);
    }
}
