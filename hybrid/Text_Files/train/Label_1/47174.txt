import java.util.*;
class Main {
    static int[] Deg;
    static ArrayList<Integer>[] adj;
    static HashSet<Integer> edge;
    static int[] col;
    static boolean dfs(int u,int p) {
        if (Deg[u] % 2 != col[u]) return false;
        boolean ans = true;
        for(int v : adj[u]) {
            if (v == p) continue;
            ans &= dfs(v,u);
        }
        return ans;
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        Deg = new int[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) adj[i] = new ArrayList<>();
        col = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            col[i] = in.nextInt();
        }
        edge = new HashSet<>();
        for (int i = 1; i <= m; i++) {
            int u = in.nextInt(), v = in.nextInt();
            adj[u].add(v);
            adj[v].add(u);
            edge.add(i);
            Deg[u]++;
            Deg[v]++;
        }
        boolean flag = false;
        for(int i=1;i<=n;i++) {
            if (Deg[i] % 2 != col[i]) {
                if (col[i] == -1) continue;
                else {
                    flag = true;
                    break;
                }
            }
        }
        if (flag) System.out.println(-1);
        else {
            System.out.println(0);
        }
    }
}
