from array import array

def answer(a: list) -> int:
    odd = array('i', (i for i in range(len(a)) if a[i]&1))
    even = array('i', (i for i in range(len(a)) if not a[i]&1))
    if len(odd) == len(even): return 0
    if len(odd) == len(even) + 1: return len(even)
    return -1

for _ in range(int(input())): print(answer(list(map(int, input().split()))))
