#include <bits/stdc++.h>
using namespace std;
#define FOR(i, a, b) for (int i = (a); i < (b); i++)
#define REP(i, n) FOR(i, 0, n)
typedef long long ll;
typedef pair<int, int> ii;
typedef vector<ii> vii;
const int MAXN = 2e5 + 10;
int n;
vii g[MAXN];
int deg[MAXN];

bool ok() {
  queue<int> q;
  q.push(0);
  deg[0] = 1;
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    for (ii e : g[u]) {
      int v = e.first;
      if (deg[v] == 0) {
        q.push(v);
        deg[v] = deg[u] + 1;
      } else if (deg[v] == deg[u]) return false;
    }
  }

  FOR(i, 0, n) if (deg[i] == 0) return false;
  return true;
}

void solve() {
  cin >> n;
  int ans = n;
  map<int, int> m;
  REP(i, n) {
    int x;
    cin >> x;
    int msb = 30;
    while (msb >= 0 && ((1 << msb) & x) == 0) msb--;
    if (msb == -1) {
      m[0]++;
      continue;
    }
    int y = 1 << msb;
    if (!m.count(y)) m[y] = 0;
    if (m[y] == 0) {
      m[y] = 1;
      m[0]++;
      continue;
    }
    if ((y ^ x) > x) {
      m[0]++;
    } else {
      m[y ^ x]++;
    }
  }
  for (auto it : m) { ans = min(ans, n - it.second - 1); }
  cout << ans << endl;
}

int main() { solve(); }
