def find_stars(n, m, grid):
    def is_valid_star(x, y):
        if grid[x][y] != '*':
            return False
        lengths = []
        for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            length = 0
            nx, ny = x + dx, y + dy
            while 0 <= nx < n and 0 <= ny < m and grid[nx][ny] == '*':
                length += 1
                nx += dx
                ny += dy
            lengths.append(length)
        star_size = min(lengths)
        if star_size == 0:
            return False
        # Mark the star
        for dx, dy in [(0, 0), (0, 1), (1, 0), (0, -1), (-1, 0)]:
            nx, ny = x, y
            for _ in range(star_size + 1 if (dx, dy) == (0, 0) else star_size):
                grid[nx][ny] = '.'
                nx += dx
                ny += dy
        stars.append((x + 1, y + 1, star_size))
        return True

    stars = []
    # Copy grid to work on
    grid = [list(row) for row in grid]
    for i in range(n):
        for j in range(m):
            is_valid_star(i, j)

    # Check if any '*' left
    for row in grid:
        if '*' in row:
            return -1

    return stars

n, m = map(int, input().split())
grid = [input() for _ in range(n)]

result = find_stars(n, m, grid)
if result == -1:
    print(result)
else:
    print(len(result))
    for star in result:
        print(*star)
