#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll n, x, y;
vector<vector<ll>>adj;
vector<ll>dis;
void bfs(ll root) {
    dis[root] = 0;
    queue<ll>q;
    q.push(root);
    while (!q.empty()) {
        ll node = q.front();
        q.pop();
        for (auto u : adj[node]) {
            if (dis[u] > dis[node] + 1) {
                dis[u] = dis[node] + 1;
                q.push(u);
            }
        }
    }
}
int main() {
    cin >> n >> x >> y;
    adj.resize(n + 1, vector<ll>());
    dis.resize(n + 1, LLONG_MAX);
    ll mx_cost = 0;
    for (ll i = 1; i < n; i++) {
        ll a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    bfs(1);
    for (ll i = 1; i <= n; i++) {
        mx_cost = max(mx_cost, dis[i]);
    }
    cout << min(x, y) * (mx_cost - 1) + max(x, y);
    return 0;
}
