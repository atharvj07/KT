import sys
input = sys.stdin.readline

def solve(n, m, k):
    adj = [set() for _ in range(n+1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    dp = [0] * (m+2)
    for i in range(m, 0, -1):
        dp[i] = dp[i+1]
        for u in range(1, n+1):
            cnt = 0
            for v in adj[u]:
                cnt += (dp[i+1] >> v & 1)
            if cnt >= k:
                dp[i] |= (1 << u)
    return dp[1:]

n, m, k = map(int, input().split())
print(*solve(n, m, k), sep='\n')
