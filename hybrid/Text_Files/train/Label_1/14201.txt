#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int k; cin >> k;
    string s, t; cin >> s >> t;
    for (int i = 0; i < k; i++) {
        if (s[i] != t[i]) {
            int m1 = (s[i] - 'a' + t[i] - 'a') / 2;
            char c = m1 + 'a';
            s[i] = c;
            t[i] = c;
            break;
        }
    }
    cout << s << endl;
    return 0;
}
