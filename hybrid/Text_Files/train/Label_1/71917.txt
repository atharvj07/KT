import sys
input = sys.stdin.readline
n,m=[int(x) for x in input().split()]
vis=[]
for i in range(n+1):
    vis+=[[]]
for i in range(m):
    a,b=[int(x) for x in input().split()]
    vis[a]+=[b]
    vis[b]+=[a]
ans=0
for i in range(1,n+1):
    mn=n+1
    for j in vis[i]:
        if j<mn:mn=j
    if mn!=n+1 and mn>i:
        print(i,mn)
        ans+=1
print(ans)
