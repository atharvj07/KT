import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        boolean found = false;
        for (long d = -1000; d <= 1000; d++) {
            if (d == 0) {
                continue;
            }
            int pos = 0;
            for (long x : a) {
                if (x * d > 0) {
                    pos++;
                }
            }
            if (pos >= (n + 1) / 2) {
                System.out.println(d);
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println(0);
        }
    }
}
