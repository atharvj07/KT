n, k = map(int, input().split())
s = input()
if k == 1:
    print(s[0])
    exit()
cnt = [0 for i in range(n)]
mn = ord('z')
for i in range(k - 1):
    for j in range(n):
        if ord(s[j]) < mn and cnt[j] < k - i:
            mn = ord(s[j])
            t = j
    cnt[t] += 1
    print(chr(mn), end='')
    mn = ord('z')
for j in range(n):
    if cnt[j] < k:
        mn = min(mn, ord(s[j]))
print(chr(mn))
