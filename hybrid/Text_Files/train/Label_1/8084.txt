#include <bits/stdc++.h>
using namespace std;
const int64_t N = 1e9;
int64_t a1, b1, a2, b2, ans;

int64_t gcd(int64_t a, int64_t b) {
    if (!b)
        return a;
    return gcd(b, a % b);
}

int64_t get(int64_t a, int64_t b) {
    if (a < b)
        swap(a, b);
    int64_t g = gcd(a, b);
    return a * b / g / g;
}

int main() {
    cin >> a1 >> b1 >> a2 >> b2;
    int64_t f = get(a1, b1), s = get(a2, b2);
    if (f != s)
        ans = -1;
    else {
        int64_t x = 0, y = 0;
        while (a1 % 3 == 0 && b1 % 3 == 0)
            x++, a1 /= 3, b1 /= 3;
        while (a1 % 2 == 0 && b1 % 2 == 0)
            y++, a1 /= 2, b1 /= 2;
        while (a2 % 3 == 0 && b2 % 3 == 0)
            x++, a2 /= 3, b2 /= 3;
        while (a2 % 2 == 0 && b2 % 2 == 0)
            y++, a2 /= 2, b2 /= 2;
        ans = x + y;
        if (x & 1)
            a1 = b1;
        if (y & 1)
            a2 = b2;
        if (a1 != a2)
            ans = -1;
    }
    cout << ans << endl;
    if (ans != -1)
        cout << a1 << ' ' << b1 << endl << a2 << ' ' << b2 << endl;
}
