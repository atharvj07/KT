#include <iostream>
#include <vector>

using namespace std;

const int kMaxK = 9;

int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n, k;
    cin >> n >> k;
    if (k == 0) {
      cout << n << endl;
    } else {
      int digit_sum[k + 1];
      for (int j = 0; j <= k; j++) {
        digit_sum[j] = j;
      }
      for (int j = 1; j <= k; j++) {
        for (int l = 0; l <= 9; l++) {
          digit_sum[j] += l * (j + 1);
        }
      }
      int prefix_sum[k + 1];
      for (int j = 1; j <= k; j++) {
        prefix_sum[j] = prefix_sum[j - 1] + digit_sum[j];
      }
      int upper_bound = prefix_sum[k] + 9 * (k);
      if (n > upper_bound) {
        cout << -1 << endl;
      } else {
        vector<int> nums;
        int cur_sum = n;
        for (int j = k; j >= 0; j--) {
          int bound = prefix_sum[j] + 9 * (j + 1);
          while (cur_sum >= bound && j > 0) {
            j--;
            bound = prefix_sum[j] + 9 * (j + 1);
          }
          int num = (cur_sum - prefix_sum[j]) / (j + 1);
          cur_sum -= num * (j + 1);
          nums.push_back(num);
        }
        cout << nums[0];
        for (int j = 1; j < nums.size(); j++) {
          cout << nums[j] + 1;
        }
        cout << endl;
      }
    }
  }
  return 0;
}
