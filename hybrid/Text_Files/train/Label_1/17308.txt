#include<bits/stdc++.h>
using namespace std;

#define ll long long int
ll p[1000005];
ll prefix[1000005];
int main() {
    ll a, b;
    cin >> a >> b;
    p[1] = 1;
    for (int i = 2; i * i <= a + b; i++) {
        if (p[i] == 0) {
            for (int j = i * i; j <= a + b; j += i) {
                p[j] = 1;
            }
        }
    }
    ll ans = 0;
    for (int i = 1; i <= a + b; i++) {
        if (p[i] == 0) {
            prefix[i] = 1;
            ans++;
        }
        prefix[i] += prefix[i - 1];
    }
    cout << prefix[a + b] << "\n";
    return 0;
}
