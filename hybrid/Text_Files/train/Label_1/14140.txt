#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cur = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'L')
            cur--;
        else
            cur++;
        ans = max(ans, abs(cur));
    }
    cout << ans * 2 + 1 << endl;
    return 0;
}
