#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> l_and_r(n);
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    l_and_r[i] = {l, r};
  }
  int l = 0, r = 0;
  int max_beauty = 0, max_ind = -1;
  for (int i = 0; i < n; i++) {
    int beauty = abs(l - r + l_and_r[i].second - l_and_r[i].first);
    if (beauty > max_beauty) {
      max_beauty = beauty;
      max_ind = i;
    }
    l += l_and_r[i].first;
    r += l_and_r[i].second;
  }
  cout << max_ind + 1 << endl;
  return 0;
}
