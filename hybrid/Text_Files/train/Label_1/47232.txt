import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        int ans = 0;
        for (long i = 2; i * i <= n; i++) {
            while (n % i == 0) {
                n /= i;
                ans++;
            }
        }
        if (n > 1) ans++;
        System.out.println(ans);
    }
}
