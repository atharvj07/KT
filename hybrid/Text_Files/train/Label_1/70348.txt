n = int(input())
mishka_score = 0
chris_score = 0

for _ in range(n):
    m, c = map(int, input().split())
    if m > c:
        mishka_score += 1
    elif c > m:
        chris_score += 1

if mishka_score > chris_score:
    print("Mishka")
elif chris_score > mishka_score:
    print("Chris")
else:
    print("Friendship is magic!^^")
