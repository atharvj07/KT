n,q,c=map(int,input().split())
stx=[0]*n
sty=[0]*n
sts=[0]*n
vtx1=[0]*q
vty1=[0]*q
vtx2=[0]*q
vty2=[0]*q
vtt=[0]*q
for i in range(n):
    stx[i],sty[i],sts[i]=map(int,input().split())
for i in range(q):
    vtt[i],vtx1[i],vty1[i],vtx2[i],vty2[i]=map(int,input().split())
for i in range(q):
    ans=0
    for j in range(n):
        if vtx1[i]<=stx[j]<=vtx2[i] and vty1[i]<=sty[j]<=vty2[i]:
            ans+=(min(vtt[i]+1,sts[j]+1))
    print(ans)
