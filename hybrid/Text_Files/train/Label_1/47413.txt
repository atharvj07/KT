import java.io.*;
import java.util.*;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    int[][] w = new int[n + 1][4];
    for (int i = 1; i <= n; i++) {
      String[] s = br.readLine().split(" ");
      for (int j = 1; j <= 3; j++) {
        w[i][j] = Integer.parseInt(s[j - 1]);
      }
    }
    for (int i = 1; i <= n; i++) {
      String[] s = br.readLine().split(" ");
      for (int j = 1; j <= 3; j++) {
        if (w[i][j] != 0 && Integer.parseInt(s[j - 1]) == 0) {
          System.out.println("Impossible");
          return;
        }
      }
    }
    List<Integer> ct = new ArrayList<>();
    for (int i = 1; i <= n; i++) {
      ct.add(0);
    }
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= 3; j++) {
        if (w[i][j] != 0 && ct.get(i) == 0) {
          ct.set(i, 1);
          break;
        }
      }
    }
    System.out.println("Possible");
    for (int i = 1; i <= n; i++) {
      System.out.print(ct.get(i) + " ");
    }
    System.out.println();
    int m = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (w[j][1] == 0 || w[j][2] == 0 || w[j][3] == 0) {
          if (ct.get(i) != 0) {
            m++;
            System.out.println(i + " " + j);
          }
        }
      }
    }
    System.out.println(m);
  }
}
