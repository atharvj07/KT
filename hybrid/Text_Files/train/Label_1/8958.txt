#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define endl "\n"

int main() {
	int n;
	cin >> n;
	vi p(n);
	for (int i = 0; i < n; i++) {
		cin >> p[i];
	}
	int cnt = 0;
	for (int i = 0; i < n; i++) {
		if (p[i] < 0) cnt++;
	}
	cout << cnt * (cnt + 1) / 2 + (n - cnt) * (n - cnt - 1) / 2 << endl;
}
