#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

const int MXN = 2e5 + 5;
const ll linf = 1e16;

struct pt {
    ll x, p;
    bool operator<(const pt &other) {
        return x < other.x;
    }
};

pt a[MXN];
vector<ll> c[2];
int main() {
    ll n, i, j, ans = linf;
    scanf("%lld", &n);
    for (i = 0; i < n;++i){
        scanf("%lld %c", &a[i].x, &a[i].p);
        if (a[i].p == 'B') a[i].p = 0;
        else if (a[i].p == 'R') a[i].p = 1;
        else a[i].p = -1;
    }
    sort(a, a + n);
    for (i = 0; i < n;++i)
        if (a[i].p != -1) c[a[i].p].push_back(a[i].x);
    for (int mask = 0; mask < 4;++mask) {
        ll sum = 0;
        j = -linf;
        for (auto &x : c[mask & 1]) {
            sum += x - j;
            j = x;
        }
        j = linf;
        for (auto &x : c[mask & 2]) {
            sum += j - x;
            j = x;
        }
        ans = min(ans, sum);
    }
    printf("%lld\n", ans);
    return 0;
}
