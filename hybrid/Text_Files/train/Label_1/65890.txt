n, m = map(int, input().split())
t = [[int(j) for j in input().split()] for _ in range(n)]
t.sort()
if t[0][0] != 0 or t[0][1] < m:
    print('NO')
else:
    ans = False
    for i in t[1:]:
        if i[0] <= t[-1][1] and i[1] >= m:
            ans = True
            break
    if ans:
        print('YES')
    else:
        print('NO')
