import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        Arrays.sort(a);

        int sum = 0;
        for (int i = 0; i < n; i += 2) {
            sum += a[i] * a[i] + a[i + 1] * a[i + 1];
        }

        System.out.println(sum);
    }
}
