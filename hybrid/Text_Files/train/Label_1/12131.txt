#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n + 5];
  for (int i = 1; i <= n; i++) cin >> a[i];
  int cnt = 0;
  for (int i = 1; i * 2 <= n; i++) {
    int l = i, r = n - i + 1;
    while (l <= r) swap(a[l++], a[r--]);
  }
  for (int i = 1; i <= n; i++) cout << a[i] << " ";
  return 0;
}
