#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int wrong[4 * maxn];
int lower[4 * maxn], upper[4 * maxn];
void build(int id, int s, int e) {
    if(s == e) {
        lower[id] = 0;
        upper[id] = 0;
        wrong[id] = 0;
        return;
    }
    int m = (s + e) / 2;
    build(2 * id, s, m);
    build(2 * id + 1, m + 1, e);
    lower[id] = lower[2 * id] + lower[2 * id + 1];
    upper[id] = upper[2 * id] + upper[2 * id + 1];
    wrong[id] = wrong[2 * id] + wrong[2 * id + 1];
}
void update(int id, int s, int e, int l, int r, char c) {
    if(s > r || e < l) return;
    if(l <= s && r >= e) {
        if(islower(c)) lower[id]++;
        else upper[id]++;
        wrong[id]--;
        return;
    }
    int m = (s + e) / 2;
    update(2 * id, s, m, l, r, c);
    update(2 * id + 1, m + 1, e, l, r, c);
    lower[id] = lower[2 * id] + lower[2 * id + 1];
    upper[id] = upper[2 * id] + upper[2 * id + 1];
    wrong[id] = wrong[2 * id] + wrong[2 * id + 1];
}
int query(int id, int s, int e, int l, int r) {
    if(s > r || e < l) return 0;
    if(l <= s && r >= e) {
        return lower[id] + upper[id] - wrong[id];
    }
    int m = (s + e) / 2;
    int p1 = query(2 * id, s, m, l, r);
    int p2 = query(2 * id + 1, m + 1, e, l, r);
    return p1 + p2;
}
int main() {
    string s, t;
    cin >> s >> t;
    int n = (int) s.size();
    build(1, 0, n - 1);
    int ans1 = 0, ans2 = 0;
    for(int i = 0;i < n;++i) {
        int low = 0, high = n - 1, pos = -1, ans = -1;
        while(low <= high) {
            int mid = (low + high) / 2;
            int now = query(1, 0, n - 1, 0, mid);
            if(now + i >= n) {
                high = mid - 1;
                ans = mid;
            }
            else low = mid + 1;
        }
        pos = ans;
        if(pos == -1) continue;
        int now = query(1, 0, n - 1, 0, pos);
        int cnt = pos - now;
        if(s[i] == t[now]) {
            update(1, 0, n - 1, now, now, s[i]);
            if(s[i] == (t[now] - 32)) wrong[1]++;
            ans1++;
        }
        else if(s[i] == (t[now] - 32) || s[i] == (t[now] + 32)) {
            update(1, 0, n - 1, now, now, s[i]);
            ans2++;
        }
        else {
            if(pos + 1 < n && s[i] == t[pos + 1]) {
                update(1, 0, n - 1, pos + 1, pos + 1, s[i]);
                if(s[i] == (t[pos + 1] - 32)) wrong[1]++;
                ans1++;
            }
            else if(pos + 1 < n && s[i] == (t[pos + 1] - 32) || s[i] == (t[pos + 1] + 32)) {
                update(1, 0, n - 1, pos + 1, pos + 1, s[i]);
                ans2++;
            }
            else if(pos - 1 >= 0 && s[i] == t[pos - 1]) {
                update(1, 0, n - 1, pos - 1, pos - 1, s[i]);
                if(s[i] == (t[pos - 1] - 32)) wrong[1]++;
                ans1++;
            }
            else if(pos - 1 >= 0 && s[i] == (t[pos - 1] - 32) || s[i] == (t[pos - 1] + 32)) {
                update(1, 0, n - 1, pos - 1, pos - 1, s[i]);
                ans2++;
            }
            else {
                update(1, 0, n - 1, i, i, s[i]);
                wrong[1]++;
                ans2++;
            }
        }
    }
    cout << ans1 << ' ' << ans2 << endl;
    return 0;
}
