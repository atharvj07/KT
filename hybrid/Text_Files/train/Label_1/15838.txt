#include <bits/stdc++.h>
using namespace std;
int n, m, k, cn, a[1001][1001], f;
int main() {
  cin >> n >> m >> k;
  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    a[x][y] = 1;
    for (int j = x - 1; j <= x; j++)
      for (int k = y - 1; k <= y; k++) {
        if (j >= 1 && j <= n && k >= 1 && k <= m && a[j][k]) cn++;
      }
    if (cn > 3) {
      f = i + 1;
      break;
    }
    cn = 0;
  }
  cout << f << '\n';
}
