import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int n = sc.nextInt();
      if (n == 0) break;
      int y = sc.nextInt();
      double[][] banks = new double[n][3];
      for (int i = 0; i < n; i++) {
        int b = sc.nextInt();
        double r = sc.nextDouble();
        int t = sc.nextInt();
        banks[i] = new double[] { b, r, t };
      }
      double max = 0;
      int maxBank = 0;
      for (int i = 0; i < n; i++) {
        double b = banks[i][2];
        double p = 100;
        double a = banks[i][1] / 100;
        if (b == 1.0) p = p + y * a * p;
        else p = p * Math.pow(1 + a, y);
        if (p > max) {
          max = p;
          maxBank = (int) banks[i][0];
        }
      }
      System.out.println(maxBank);
    }
  }
}
