#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 1;
    for (int i = 0, j = 1; j < n; j++) {
        if (a[j] <= a[j - 1]) {
            i = j;
        }
        ans = max(ans, j - i + 1);
    }
    cout << ans << endl;
    return 0;
}
