import java.util.*;
import java.io.*;

public class Main {
    static int[] solve(int n, int q, String init, int[] p, char[] c) {
        char[] a = init.toCharArray();
        int[][] trans = {{2, 0, 1}, {0, 1, 2}, {1, 2, 0}};
        for (int i = 0; i < q; i++) {
            int j = p[i] - 1;
            a[j] = c[i];
        }
        int[] nxt = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                if (a[i] == a[j]) {
                    nxt[i] = -1;
                    break;
                }
                int t = trans[a[i] - 'R'][a[j] - 'R'];
                if (t == 0) {
                    nxt[i] = -1;
                    break;
                }
                if (nxt[i] != 0) {
                    nxt[i] = j;
                }
            }
        }
        int[] cnt = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (nxt[i] != -1 && nxt[j] != -1) {
                    if (a[nxt[i]] == a[j]) {
                        cnt[i]++;
                        cnt[j]++;
                    }
                    else {
                        if (trans[a[nxt[i]] - 'R'][a[j] - 'R'] == 0) cnt[i]++;
                        else cnt[j]++;
                    }
                }
            }
        }
        int[] res = new int[q + 1];
        int ans = n;
        res[q] = ans;
        for (int i = q - 1; i >= 0; i--) {
            int j = p[i] - 1;
            ans += cnt[j];
            ans -= (a[j] == a[nxt[j]] ? 1 : 0);
            nxt[j] = -1;
            for (int k = j + 1; k < n; k++) {
                if (nxt[k] == -1) continue;
                int t = trans[a[k] - 'R'][a[nxt[k]] - 'R'];
                if (t == 0) nxt[k] = -1;
                else if (nxt[k] == j) nxt[k] = nxt[j];
            }
            res[i] = ans;
        }
        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        String init = in.readLine();
        int[] p = new int[q];
        char[] c = new char[q];
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(in.readLine());
            p[i] = Integer.parseInt(st.nextToken());
            c[i] = st.nextToken().charAt(0);
        }
        int[] ans = solve(n, q, init, p, c);
        for (int i = 0; i <= q; i++) {
            System.out.println(ans[i]);
        }
    }
}
