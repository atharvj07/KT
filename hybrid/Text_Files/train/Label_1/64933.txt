import sys

for t in range(int(input())):
    n, m, k = map(int, input().split())
    a = [list(input()) for i in range(n)]
    for i in range(n):
        for j in range(m):
            if a[i][j] != '.':
                if i - k < 0 or j - k < 0 or j + k >= m or a[i - k][j - k] == '.':
                    print('NO')
                    sys.exit()
                for di in range(k):
                    if a[i - di][j - di + k] == '.':
                        print('NO')
                        sys.exit()
                    if a[i - di][j + di - k] == '.':
                        print('NO')
                        sys.exit()
    print('YES')
