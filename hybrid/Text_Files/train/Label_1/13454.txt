#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    while(cin>>n){
        vector<vector<int>>arr(n+1,vector<int>(n+1,0));
        vector<vector<int>>dp(n+1,vector<int>(n+1,0));
        for(int i=1;i<=n;++i){
            for(int j=1;j<=n;++j){
                int x;
                cin>>x;
                arr[i][j]=x;
                dp[i][j] = x + dp[i-1][j] + dp[i][j-1] - dp[i-1][j-1];
            }
        }
        bool flag = true;
        for(int i=1; i<=n;++i){
            for(int j=1; j<=n;++j){
                for(int k=i; k<=n;++k){
                    for(int l=j; l<=n;++l){
                        if((dp[k][l] - dp[k][j-1]-dp[i-1][l] + dp[i-1][j-1])==0) continue;
                        else if((dp[k][l] - dp[k][j-1]-dp[i-1][l] + dp[i-1][j-1])==arr[i][j]){
                            if((i-1)*l+(n-k)*(n-j) + (k-i)*(n-j)+(n-i)*(j-1) == arr[i][j] * n * n){
                                flag = true;
                            }
                            else{
                                flag = false;
                                break;
                            }
                        }
                        else{
                            flag = false;
                            break;
                        }
                    }
                }
            }
        }
        if(flag) cout << "Yes\n";
        else cout << "No\n";
    }
    return 0;
}
