#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define int ll
int32_t main() {
  int q;
  cin >> q;
  while (q--) {
    int n, m, k;
    cin >> n >> m >> k;
    if (abs(n - m) % 2 != k % 2) {
      cout << -1 << "\n";
      continue;
    }
    if (k < abs(n) + abs(m)) {
      cout << -1 << "\n";
      continue;
    }
    int diag = min(k / 2, min(n, m));
    cout << 2 * diag << "\n";
  }
  return 0;
}
