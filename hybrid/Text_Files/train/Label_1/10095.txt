#include <iostream>
#include <vector>
using namespace std;
int main(){
    int n,a;
    cin>>n;
    vector<int>b(n+1,0);
    for(int i=0;i<n-1;++i){
        cin>>a;
        b[a]=1;
    }
    for(int i=1;i<=n;++i){
        if(!b[i]){
            cout<<i<<endl;
            break;
        }
    }
    return 0;
}
