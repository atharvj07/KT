#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
#define endl '\n'
const ll mod = 1e9 + 7;
const double PI = 3.14159265358979323846;

void solve() {
  ll n, m, x, y, z, p;
  cin >> n >> m >> x >> y >> z >> p;
  vector<pair<ll, ll>> candies(p);
  for (ll i = 0; i < p; ++i) cin >> candies[i].first >> candies[i].second;
  for (auto& candy : candies) {
    ll newX, newY;
    newX = (candy.first + z) % n + 1;
    newY = (candy.second - (x + y + z)) % m + 1;
    cout << newX << " " << newY << endl;
  }
}

int main() {
  solve();
  return 0;
}
