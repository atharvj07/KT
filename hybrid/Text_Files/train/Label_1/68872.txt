from collections import Counter

def find_max_rectangle(n, numbers):
    count = Counter(numbers)
    freq = sorted(list(count.values()), reverse=True)

    max_area = 0
    best_p, best_q = 0, 0
    for p in range(1, n + 1):
        q = 0
        for f in freq:
            q += min(f, p)
        q //= p
        if p * q > max_area:
            max_area = p * q
            best_p, best_q = p, q
    
    rectangle = [[0] * best_q for _ in range(best_p)]
    num_freq_pairs = sorted([(num, freq) for num, freq in count.items()], key=lambda x: -x[1])
    
    rows, cols = best_p, best_q
    idx = 0
    for num, freq in num_freq_pairs:
        for _ in range(freq):
            rectangle[idx // cols][idx % cols] = num
            idx += 1
            if idx >= rows * cols:
                break
        if idx >= rows * cols:
            break
    
    return max_area, best_p, best_q, rectangle

n = int(input().strip())
numbers = list(map(int, input().strip().split()))

max_area, p, q, rectangle = find_max_rectangle(n, numbers)

print(max_area)
print(p, q)
for row in rectangle:
    print(*row)
