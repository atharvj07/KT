#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
vector<int32_t> a1(N), a2(N);
int32_t n, m, a[N], t, l, r, x, ans;
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int32_t i = 1; i <= n; i++) {
        cin >> a[i];
        a1[i] = a[i];
        a2[i] = a[i];
    }
    sort(a1.begin() + 1, a1.begin() + 1 + n);
    sort(a2.begin() + 1, a2.begin() + 1 + n);
    while (m--) {
        cin >> l >> r >> x;
        t = lower_bound(a1.begin() + l, a1.begin() + r + 1, x) - a1.begin();
        ans = t;
        if (a[t] == x) ans = -1;
        if (ans == n + 1) ans = -1;
        t = lower_bound(a2.begin() + l, a2.begin() + r + 1, x) - a2.begin();
        if (ans == -1 && a[t] == x)  ans = -1;
        cout << ans << '\n';
    }
}
