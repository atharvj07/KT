n = int(input())
board, rows, cols = [], [0] * n, [0] * n
for _ in range(n):
    line = input()
    board.append(line)
    for i in range(n):
        if line[i] == 'C':
            rows[_] += 1
            cols[i] += 1
ans = 0
for i in rows:
    ans += (i * (i - 1)) // 2
for i in cols:
    ans += (i * (i - 1)) // 2
print(ans)
