#include <bits/stdc++.h>
using namespace std;

const int dx[] = {-1, 0, 1, 0};
const int dy[] = {0, 1, 0, -1};
set<vector<int>> st;
int n, m;
vector<vector<char>> maze;

bool valid(int x, int y) {
  return x >= 0 && x < n && y >= 0 && y < m && maze[x][y] != '#';
}

void dfs(int x, int y, string& s, vector<int>& ord) {
  if (maze[x][y] == 'E') {
    st.insert(ord);
    return;
  }
  for (int i = 0; i < 4; i++) {
    int nx = x + dx[i];
    int ny = y + dy[i];
    int ind = ord[i];
    if (valid(nx, ny)) {
      ord[i] = ord[s[ind] - '0'];
      maze[x][y] = '#';
      dfs(nx, ny, s, ord);
      maze[x][y] = '.';
      ord[i] = ind;
    }
  }
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> m;
  maze.resize(n, vector<char>(m));
  int sx, sy;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> maze[i][j];
      if (maze[i][j] == 'S') {
        sx = i, sy = j;
        maze[i][j] = '.';
      }
    }
  }
  string s;
  cin >> s;
  vector<int> ord(4);
  iota(ord.begin(), ord.end(), 0);
  dfs(sx, sy, s, ord);
  cout << st.size() << '\n';
  return 0;
}
