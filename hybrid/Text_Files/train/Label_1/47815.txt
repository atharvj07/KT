import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int queries = scanner.nextInt();

        for (int i = 0; i < queries; i++) {
            int x0 = scanner.nextInt();
            int y0 = scanner.nextInt();
            int x1 = scanner.nextInt();
            int y1 = scanner.nextInt();
            int x2 = scanner.nextInt();
            int y2 = scanner.nextInt();
            int x3 = scanner.nextInt();
            int y3 = scanner.nextInt();

            if (isIntersect(x0, y0, x1, y1, x2, y2, x3, y3)) {
                System.out.println(1);
            } else {
                System.out.println(0);
            }
        }
    }

    private static boolean isIntersect(int x0, int y0, int x1, int y1, int x2, int y2, int x3, int y3) {
        int area0 = (x1 - x0) * (y1 - y0);
        int area1 = (x2 - x0) * (y2 - y0);
        int area2 = (x3 - x0) * (y3 - y0);
        int area3 = (x1 - x2) * (y1 - y2);
        int area4 = (x3 - x2) * (y3 - y2);
        int area5 = (x1 - x3) * (y1 - y3);

        return area0 * area1 > 0 && area3 * area4 > 0 && area2 * area5 > 0;
    }
}
