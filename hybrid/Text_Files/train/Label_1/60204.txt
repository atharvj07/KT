from sys import stdin

n, m = map(int, stdin.readline().split())
grid = [stdin.readline().strip() for _ in range(n)]

mp = [[(1, 1) for _ in range(m)] for _ in range(n)]
for i in range(n):
    for j in range(m):
        if grid[i][j] == '*':
            l = j
            while l >= 0 and grid[i][l] == '*':
                l -= 1
            l += 1
            r = j
            while r < m and grid[i][r] == '*':
                r += 1
            r -= 1
            u = i
            while u >= 0 and grid[u][j] == '*':
                u -= 1
            u += 1
            d = i
            while d < n and grid[d][j] == '*':
                d += 1
            d -= 1
            mp[i][j] = (min(l, r) - 1, min(u, d) - 1)

# print(mp)
stars = []
for i in range(n):
    for j in range(m):
        if grid[i][j] == '*':
            x, y = mp[i][j]
            if x < y:
                stars.append((i, j, y - 1))
                mp[i - y + 1 : i + y, j - x + 1 : j + x] = [(-1, -1)] * (2 * y - 1) * (2 * x - 1)
            else:
                stars.append((i, j, x - 1))
                mp[i - y + 1 : i + y, j - x + 1 : j + x] = [(-1, -1)] * (2 * y - 1) * (2 * x - 1)

print(len(stars))
for i, j, s in stars:
    print(i, j, s)
