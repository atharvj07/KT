#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;

  vector<bool> a(n), b(m);
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a[i] = x;
  }
  for (int i = 0; i < m; i++) {
    int x;
    cin >> x;
    b[i] = x;
  }

  vector<vector<int>> dp(n, vector<int>(m, 0));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i] && b[j]) {
        if (i == 0 || j == 0) {
          dp[i][j] = 1;
        } else {
          dp[i][j] = dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1] + 1;
        }
      }
    }
  }

  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      for (int x = i; x < n; x++) {
        for (int y = j; y < m; y++) {
          int cur = dp[x][y];
          if (i > 0) { cur -= dp[i - 1][y]; }
          if (j > 0) { cur -= dp[x][j - 1]; }
          if (i > 0 && j > 0) { cur += dp[i - 1][j - 1]; }
          if (cur == k) { ans++; }
        }
      }
    }
  }

  cout << ans << endl;

  return 0;
}
