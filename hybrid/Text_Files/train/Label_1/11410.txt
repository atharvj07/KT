#include<bits/stdc++.h>
using namespace std;
int a[1<<16],n;
int main(){
	cin>>n;
	for(int i=0;i<n;++i)
		cin>>a[i];
    vector<int> mx(n,1<<30);
    for(int i=n-2;i>=0;--i)
		mx[i]=min(mx[i+1],a[i+1]);
    vector<int> mn(n,0);
    for(int i=1;i<n;++i)
		mn[i]=max(mn[i-1],a[i-1]);
    int ans=1;
    for(int i=1;i<n;++i){
		if(a[i]>=mn[i-1]&&a[i]<=mx[i])
			ans++;
		else
			break;
    }
    cout<<ans<<endl;
    return 0;
}
