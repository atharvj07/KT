import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String l = "0000000" + in.next(), r = in.next();
        l = l.substring(l.length() - n);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (l.charAt(i) != r.charAt(i)) {
                ans |= (1 << i);
            }
        }
        System.out.println(Integer.toBinaryString(ans));
    }
}
