MOD = 998244353

def ways_to_paint(n, colors):
    dp = [[0 for _ in range(n + 2)] for __ in range(n + 2)]
    for i in range(n + 1):
        dp[i][i] = 1
        if i < n:
            dp[i][i + 1] = 1

    for length in range(2, n + 1):
        for start in range(n - length + 1):
            end = start + length
            min_pos = start
            for i in range(start + 1, end):
                if colors[i] < colors[min_pos]:
                    min_pos = i
            dp[start][end] = dp[start][min_pos] * dp[min_pos + 1][end]
            for i in range(start, end):
                if i != min_pos:
                    dp[start][end] = dp[start][end] * (dp[start][i] * dp[i + 1][end] % MOD) % MOD
            dp[start][end] %= MOD
    return dp[0][n]

n, m = map(int, input().split())
colors = list(map(int, input().split()))
colors.insert(0, 0)
print(ways_to_paint(n, colors))
