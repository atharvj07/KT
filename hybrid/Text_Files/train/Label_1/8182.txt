#include<bits/stdc++.h>
using namespace std;
 
#define endl "\n"
#define ll long long
#define pb push_back
#define mp make_pair
#define f first
#define s second
typedef pair<ll,ll> pi;
 
const ll mod = 1e9 + 7;
const ll INF = LLONG_MAX;
 
int main()
{
    ll n, k, i, x, y, u, v, c0, ans, vis[200005], now, c_count;
    vector<ll> adj[200005], color, ans_color;
    queue<ll> q;
 
    cin >> n >> k;
 
    for(i = 0; i < n - 1; i++)
    {
        cin >> x >> y;
        adj[x].pb(y);
        adj[y].pb(x);
    }
 
    memset(vis, 0, sizeof(vis));
    memset(color.data(), 0, color.size() * sizeof(ll));
    c_count = 0;
 
    q.push(1);
    vis[1] = 1;
    c_count++;
    color.pb(1);
    while(!q.empty())
    {
        c0 = q.front();
        q.pop();
        for(i = 0; i < adj[c0].size(); i++)
        {
            u = adj[c0][i];
            if(vis[u] == 0)
            {
                vis[u] = vis[c0] + 1;
                q.push(u);
                ans = vis[u] / 3;
                if(vis[u] % 3 == 0)
                {
                    ans--;
                }
                color.pb(ans + 1);
                c_count = max(c_count, ans + 1);
            }
        }
    }
 
    cout << c_count << endl;
 
    for(i = 0; i < n - 1; i++)
    {
        cout << color[i] << " ";
    }
    cout << endl;
    return 0;
}
