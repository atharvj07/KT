import sys

def solve(n, x, w):
    w.sort()
    for i in range(n):
        if w[i] == x:
            return "NO"
        for j in range(i + 1):
            if w[i] + w[j] == x:
                return "NO"
    return "YES", w

t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    w = list(map(int, input().split()))
    ans = solve(n, x, w)
    print(ans[0])
    if ans[0] == "YES":
        print(*ans[1])
