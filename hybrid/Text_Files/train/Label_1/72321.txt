t = int(input().strip())
for _ in range(t):
    a, b, c = map(int, input().split())
    dishes = sorted([a, b, c])
    visitors = 0
    for i in range(3):
        if dishes[i] > 0:
            visitors += 1
            dishes[i] -= 1
    if dishes[1] > 0 and dishes[2] > 0:
        visitors += 1
        dishes[1] -= 1
        dishes[2] -= 1
    if dishes[0] > 0 and dishes[2] > 0:
        visitors += 1
        dishes[0] -= 1
        dishes[2] -= 1
    if dishes[0] > 0 and dishes[1] > 0:
        visitors += 1
        dishes[0] -= 1
        dishes[1] -= 1
    if all(d >= 1 for d in dishes):
        visitors += 1
    print(visitors)
