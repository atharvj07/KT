import java.util.Scanner;

class Main {

  private static int maxProfit(int n, int m, int[] l, int[] s, int[] c) {
    int[] dp = new int[m + 1];
    int ans = 0;

    for (int i = 1; i <= n; i++) {
      int maxC = c[l[i - 1]];
      for (int j = m; j >= 1; j--) {
        if (j < l[i - 1]) {
          continue;
        }
        maxC = Math.max(maxC, c[j]);
        dp[j] = Math.max(dp[j], dp[j - l[i - 1]] + maxC);
        ans = Math.max(ans, dp[j] - s[i - 1]);
      }
    }

    return ans;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int[] l = new int[n];
    int[] s = new int[n];
    int[] c = new int[m + 1];
    for (int i = 0; i < n; i++) {
      l[i] = in.nextInt();
    }
    for (int i = 0; i < n; i++) {
      s[i] = in.nextInt();
    }
    for (int i = 1; i <= m; i++) {
      c[i] = in.nextInt();
    }
    System.out.println(maxProfit(n, m, l, s, c));
  }
}
