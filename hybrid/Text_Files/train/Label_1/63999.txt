n = int(input().strip())
graph = [input().strip() for _ in range(n)]
m = int(input().strip())
path = list(map(lambda x: int(x) - 1, input().strip().split()))

subsequence = [path[0]]
for i in range(1, m - 1):
    if graph[path[i - 1]][path[i + 1]] == '0':
        subsequence.append(path[i])
subsequence.append(path[-1])

print(len(subsequence))
print(' '.join(map(lambda x: str(x + 1), subsequence)))
