#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int d = -1000; d <= 1000; d++) {
    if (d == 0) continue;
    int cnt = 0;
    for (int i = 0; i < n; i++) if (double(a[i]) / d > 0) cnt++;
    if (cnt >= (n + 1) / 2) {
      cout << d << '\n';
      return 0;
    }
  }
  cout << 0 << '\n';
}
