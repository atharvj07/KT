import sys

input = sys.stdin.readline

n, m, q = map(int, input().split())
p = list(map(int, input().split()))
a = list(map(int, input().split()))

cnt = [0] * (n + 1)

for x in a:
    cnt[x] ^= 1

ok = True

for i in range(1, n + 1):
    if cnt[i] == 0:
        ok = False
        break

for _ in range(q):
    l, r = map(int, input().split())
    L = r - l + 1

    if L < n:
        ok = False
        break

    cur = cnt[p[0]]

    for i in range(1, n):
        cur ^= cnt[p[i]]
        if cur == 0:
            ok = True
            break

if ok:
    print('1' * q)
else:
    print('0' * q)
