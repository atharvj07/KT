#include <cstdio>
#include <vector>

using namespace std;
vector<int> ans;

int main() {
  int t, n;
  scanf("%d", &t);
  while (t--) {
    ans.assign(5005, 0);
    scanf("%d", &n);
    for (int i = 0, l, r; i < n; i++) {
      scanf("%d %d", &l, &r);
      if (ans[l - 1] == 0) ans[l] = l;
      if ((++ans[l - 1]) == r) ans.back() = l + 1;
    }
    for (int i = 0; i < n; i++) printf("%d ", ans[i]);
    printf("\n");
  }
}
