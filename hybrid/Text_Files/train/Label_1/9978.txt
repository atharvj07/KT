#include <iostream>

using namespace std;

int main() {
  int x;
  cin >> x;
  int hh, mm;
  cin >> hh >> mm;
  for (int y = 0;; y++) {
    int h = hh - (x * y / 60);
    int m = mm - (x * y % 60);
    if (m < 0) {
      m += 60;
      h--;
    }
    if (h < 0) {
      h += 24;
    }
    if (to_string(h).find('7') != string::npos || to_string(m).find('7') != string::npos) {
      cout << y << endl;
      return 0;
    }
  }
  return 0;
}
