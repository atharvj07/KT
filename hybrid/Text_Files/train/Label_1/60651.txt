import sys
input = lambda: sys.stdin.readline().strip()

def solve():
    n = int(input())
    s = input()
    t = input()
    shifts = 0
    for i in range(n):
        if s[i] != t[i]:
            shifts += 1
            s = s[1:] + s[0]
    if s == t:
        return shifts
    else:
        return -1

print(solve())
