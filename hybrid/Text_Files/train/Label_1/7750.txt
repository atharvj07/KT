#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int& x : a) cin >> x;
    sort(a.begin(), a.end());
    ll ans = 0;
    for (int i = 1; i < n; i++) {
        int d = (a[i] - a[i - 1] + 1) / 2;
        a[i] -= d;
        a[i - 1] += d;
        ans += d;
    }
    cout << ans << '\n';
    cout << a[0] << ' ' << a[1] - a[0] << '\n';
    return 0;
}
