import sys

t = int(sys.stdin.readline())
for _ in range(t):
    n, k = map(int, sys.stdin.readline().split())
    c = [list(map(int, sys.stdin.readline().split())) for _ in range(k)]
    e = [0] * (2 * n + 1)
    for x, y in c:
        e[x] = y
        e[y] = x
    ans = 0
    for i in range(1, 2 * n + 1):
        if e[i] > i:
            ans += (e[i] - i - 1) // 2
    print(ans)
