#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, w;
    cin >> n >> w;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if(a[i]%2) b[i] = a[i] / 2 + 1;
        else b[i] = a[i] / 2;
    }
    int sum = 0;
    for(int i: b) sum += i;
    if(sum > w) cout << -1 << endl;
    else {
        int cnt = w - sum;
        for(int& i: b) {
            if(cnt) {
                i++;
                cnt--;
            }
        }
        for(int i: b) cout << i << " ";
        cout << endl;
    }
}

int main() {
    int t;
    cin >> t;
    while(t--) solve();
    return 0;
}
