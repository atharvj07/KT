import sys
input = sys.stdin.readline

n, m = map(int, input().split())
V, H = [], []
for _ in range(n):
    V.append(int(input()))
for _ in range(m):
    a, b, c = map(int, input().split())
    H.append((a, b, c))
H.sort()
V.sort()
ans = 0; i = 0; j = 0
while i < n or j < m:
    if i < n and V[i] < (H[j][0] if j < m else +float('inf')):
        if i+1 < n and V[i+1] <= H[j][0]:
            i += 2
            ans += 1
        else:
            i += 1
            ans += 1
    elif j < m and H[j][0] <= V[i] if i < n else True:
        if j+1 < m and H[j+1][0] <= V[i]:
            j += 2
            ans += 1
        else:
            j += 1
            ans += 1
print(ans)
