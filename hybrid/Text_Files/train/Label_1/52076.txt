import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String l = input.next();
        String r = input.next();
        int[] p = new int[n];
        p[n - 1] = Integer.parseInt(r.charAt(n - 1) + "");
        for (int i = n - 2; i >= 0; --i) {
            p[i] = p[i + 1] + Integer.parseInt(r.charAt(i) + "");
        }
        int ans = Integer.parseInt(p[0] + "", 2);
        for (int i = 0; i < n; ++i) {
            int t = p[i];
            if (i > 0) {
                t -= Integer.parseInt(l.charAt(i - 1) + "");
            }
            ans = Math.max(ans, t);
        }
        System.out.println(Integer.toBinaryString(ans));
    }
}
