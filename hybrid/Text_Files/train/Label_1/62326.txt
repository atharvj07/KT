def calculate_f(a, b, n, m):
    min_f = float('inf')
    for j in range(m - n + 1):
        f = 0
        for i in range(n):
            f += (-1) ** i * (a[i] - b[i + j])
        min_f = min(min_f, abs(f))
    return min_f

n, m, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

print(calculate_f(a, b, n, m))

for _ in range(q):
    l, r, x = map(int, input().split())
    for i in range(l-1, r):
        a[i] += x
    print(calculate_f(a, b, n, m))
