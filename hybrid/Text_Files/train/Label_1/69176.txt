from collections import deque

def find_component(adj, start):
    visited = {start}
    queue = deque([start])
    component = [start]
    
    while queue:
        node = queue.popleft()
        for neighbor in adj[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
                component.append(neighbor)
    
    return component

n = int(input())
p = list(map(int, input().split()))
adj = [[] for _ in range(n)]

for i in range(n):
    row = input()
    for j in range(n):
        if row[j] == '1':
            adj[i].append(j)

for i in range(n):
    component = find_component(adj, i)
    if component:
        values = [p[j] for j in component]
        values.sort()
        for index, pos in enumerate(component):
            p[pos] = values[index]

print(" ".join(map(str, p)))
