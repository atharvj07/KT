import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
  public static void main(String[] args) throws NumberFormatException, IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    Rectangle[] arr = new Rectangle[6];

    for (int i = 0; i < 6; i++) {
      String[] line = br.readLine().split(" ");

      int w = Integer.parseInt(line[0]);
      int h = Integer.parseInt(line[1]);

      arr[i] = new Rectangle(w, h);
    }

    Arrays.sort(arr, (a, b) -> a.w - b.w);

    RearrangeablePair first = null;
    RearrangeablePair second = null;
    RearrangeablePair third = null;

    for (int i = 0; i < 6; i++) {
      for (int j = i + 1; j < 6; j++) {
        if (arr[i].w == arr[j].w && arr[i].h == arr[j].h) {
          continue;
        }

        for (int k = j + 1; k < 6; k++) {
          if (arr[i].w == arr[k].w && arr[i].h == arr[k].h) {
            continue;
          }

          if (arr[j].w == arr[k].w && arr[j].h == arr[k].h) {
            continue;
          }

          if (first == null) {
            first = new RearrangeablePair(arr[i], arr[j], arr[k]);
          } else if (second == null) {
            second = new RearrangeablePair(arr[i], arr[j], arr[k]);
          } else if (third == null) {
            third = new RearrangeablePair(arr[i], arr[j], arr[k]);
          }
        }
      }
    }

    if (first == null || second == null || third == null) {
      System.out.println("no");
      return;
    }

    boolean possible = first.canMakeParallelepiped(second, third);

    if (possible) {
      System.out.println("yes");
    } else {
      System.out.println("no");
    }
  }
}

class Rectangle {
  int w;
  int h;

  public Rectangle(int w, int h) {
    this.w = w;
    this.h = h;
  }
}

class RearrangeablePair {
  Rectangle a;
  Rectangle b;
  Rectangle c;

  public RearrangeablePair(Rectangle a, Rectangle b, Rectangle c) {
    this.a = a;
    this.b = b;
    this.c = c;
  }

  public boolean canMakeParallelepiped(RearrangeablePair second, RearrangeablePair third) {
    boolean firstFits = this.a.w == second.b.w && this.a.h == second.b.h && this.b.w == third.a.w && this.b.h == third.a.h
        && this.c.w == second.c.w && this.c.h == second.c.h;

    boolean secondFits = this.b.w == second.c.w && this.b.h == second.c.h && this.a.w == third.b.w && this.a.h == third.b.h
        && this.c.w == second.a.w && this.c.h == second.a.h;

    boolean thirdFits = this.c.w == second.a.w && this.c.h == second.a.h && this.b.w == third.a.w && this.b.h == third.a.h
        && this.a.w == second.b.w && this.a.h == second.b.h;

    return firstFits || secondFits || thirdFits;
  }
}
