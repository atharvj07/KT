import sys
input=sys.stdin.readline

mod=998244353
n,m=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))

# Find sum of all elements including the last element of minimum in b
sumb=0
for x in b:
    sumb+=x

# Find all indexes where the minimum in b is present in a
idx=[-1]*(m+1)
for i in range(n):
    for j in range(m):
        if a[i]==b[j]:
            idx[j]=i
            break

ans=1
#Contribution for each block
for i in range(m):
    #If minimum is not present , invalid partition
    if idx[i]==-1:
        ans=0
        break
    #contribution
    ans=(ans*(idx[i]+1-i))%mod

print(ans)
