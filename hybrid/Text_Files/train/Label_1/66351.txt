import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, s, k = map(int, input().split())
    closed = set(map(int, input().split()))
    if not closed or s in closed:
        print(0)
    else:
        l, r = min(s - 1 for s in closed), max(s + 1 for s in closed)
        print(min(abs(s-l), abs(s-r)) if l != r else 0)
