import java.util.*;

public class Main {
    static class Domino implements Comparable<Domino> {
        int a, b;

        Domino(int a, int b) {
            this.a = a;
            this.b = b;
        }

        public int compareTo(Domino o) {
            return a + b - (o.a + o.b);
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        List<Domino> dominoes = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String line = input.next();
            for (int j = 0; j < m; j++) {
                dominoes.add(new Domino(line.charAt(2 * j) - '0', line.charAt(2 * j + 1) - '0'));
            }
        }
        Collections.sort(dominoes);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(dominoes.get(i * m + j).a + "" + dominoes.get(i * m + j).b + " ");
            }
            System.out.println();
        }
    }
}
