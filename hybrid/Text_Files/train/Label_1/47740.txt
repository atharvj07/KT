import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static int[] table = new int[10];

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] input = reader.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int q = Integer.parseInt(input[1]);
        char[] seq = reader.readLine().toCharArray();
        System.out.println(solve(seq, n, q, reader));
    }

    static String solve(char[] seq, int n, int q, BufferedReader reader) throws IOException {
        int start;
        StringBuilder ans = new StringBuilder();
        while (q-- > 0) {
            Arrays.fill(table, 0);
            String[] query = reader.readLine().split(" ");
            start = Integer.parseInt(query[0]) - 1;
            int[] count = getCounts(seq, n, start, Integer.parseInt(query[1]) - 1);
            for (int i = 0; i < 10; i++) {
                ans.append(count[i]).append(" ");
            }
        }
        return ans.toString();
    }

    static int[] getCounts(char[] seq, int n, int start, int end) {
        int cp = start;
        int dp = 1;
        while (cp >= 0 && cp < n) {
            char c = seq[cp];
            if (c <= '9' && c >= '0') {
                table[c - '0']++;
                seq[cp] = (char) (c - 1);
                if (seq[cp] == '0') {
                    if (dp == 1) {
                        cp++;
                    } else {
                        cp--;
                    }
                }
                dp *= -1;
            } else if (c == '<') {
                dp = -1;
                if (dp == 1) {
                    cp++;
                } else {
                    cp--;
                }
            } else {
                dp = 1;
                if (dp == 1) {
                    cp++;
                } else {
                    cp--;
                }
            }
            if (cp == end) {
                break;
            }
        }
        return table;
    }
}
