n, q, c = map(int, input().split())
stars = [tuple(map(int, input().split())) for _ in range(n)]
views = [tuple(map(int, input().split())) for _ in range(q)]

for t, x1, y1, x2, y2 in views:
    total_brightness = 0
    for x, y, s in stars:
        if x1 <= x <= x2 and y1 <= y <= y2:
            total_brightness += (s + t) % (c + 1)
    print(total_brightness)
