import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[4];
        int[] b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        int[] attack = {a[0], b[0], a[2], b[2]};
        int[] defence = {b[0], a[0], b[2], a[2]};
        sort(attack);
        sort(defence);
        if (attack[1] > defence[0] && attack[2] > defence[1]) {
            System.out.println("Team 1");
        } else if (attack[0] > defence[1] && attack[1] > defence[2]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }

    private static void sort(int[] arr) {
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (arr[i] > arr[j]) {
                    int tmp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = tmp;
                }
            }
        }
    }
}
