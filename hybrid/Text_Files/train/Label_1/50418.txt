import java.util.*;
 
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
 
        Stack<Integer>[] stack = (Stack<Integer>[]) new Stack[n];
        for (int i = 0; i < n; i++) stack[i] = new Stack<>();
 
        while (q-- > 0) {
            int query = sc.nextInt();
            switch (query) {
                case 0:
                    stack[sc.nextInt()].push(sc.nextInt());
                    break;
                case 1:
                    Stack<Integer> s = stack[sc.nextInt()];
                    if (!s.empty()) System.out.println(s.peek());
                    break;
                case 2:
                    stack[sc.nextInt()].pop();
            }
        }
    }
}
