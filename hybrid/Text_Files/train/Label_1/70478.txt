def solve(n):
    if n == 2: return 2
    base = 2
    for _ in range(20):
      if base * base - base + 2 == n:
        return base
      base += 1
    return -1

k = int(input())
n = solve(k + 1)
print(n)
for x in range(1, n + 1):
  for y in range(1, n + 1):
    if k and x == y: print('N', end='')
    elif x == 1 and y == 2 or x == 2 and y == 1: print('Y', end='')
    elif k and x * y == n * n - n + 2: 
      k -= 1
      print('Y', end='')
    else: print('N', end='')
  print()
