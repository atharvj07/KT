import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static class Student {
        int friend;
        int gender;

        public Student(int friend, int gender) {
            this.friend = friend;
            this.gender = gender;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        Map<Integer, Student> students = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            int friend = Integer.parseInt(line[0]);
            int gender = Integer.parseInt(line[1]);
            students.put(i + 1, new Student(friend, gender));
        }
        List<Pair> pairs = new ArrayList<>();
        int maxPairs = 0;
        int maxBoyGirlPairs = 0;
        for (int i = 1; i <= n; i++) {
            if (!students.containsKey(i)) {
                continue;
            }
            Student student = students.get(i);
            if (student.friend == i) {
                continue;
            }
            if (students.containsKey(student.friend) && students.get(student.friend).friend == i) {
                pairs.add(new Pair(i, student.friend));
                if (student.gender != students.get(student.friend).gender) {
                    maxBoyGirlPairs++;
                }
                maxPairs++;
                students.remove(i);
                students.remove(student.friend);
            }
        }
        System.out.println(maxPairs + " " + maxBoyGirlPairs);
        for (Pair pair : pairs) {
            System.out.println(pair.a + " " + pair.b);
        }
    }

    private static class Pair {
        int a;
        int b;

        public Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }
}
