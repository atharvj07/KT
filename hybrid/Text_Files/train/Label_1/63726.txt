import sys

n, k = map(int, input().split())
passwords = {p: 1 for p in [input() for _ in range(n)]}
vanya = input()

acc = 0
for p in passwords:
    acc += 1
    if p == vanya:
        break
time = acc + (acc - 1)//k * 5

acc = 0
for p in passwords:
    acc += 1
    if acc > k:
        break
    if p == vanya:
        break
time2 = acc + (acc - 1)//k * 5
print(time, time2)
