def main():
    n, m = map(int, input().split())
    leaders = list(map(int, input().split()))
    a = [0] * n
    b = [0] * n
    for i in range(m):
        for j in range(b[leaders[i] - 1], b[leaders[i] - 1] + a[leaders[i] - 1] if a[leaders[i] - 1] else n):
            if not b[j % n]:
                a[leaders[i] - 1] = j % n - b[leaders[i] - 1] + 1
                b[j % n] = 1
                break
    if any(not i for i in a):
        print(-1)
    else:
        print(*a)

main()
