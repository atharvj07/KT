#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
typedef long long ll;
int a[N], b[N];
ll sumA[N], sumB[N];
int main() {
	int m, n;
	cin >> m >> n;
	for (int i = 1; i <= m; i++) cin >> a[i];
	for (int i = 1; i <= n; i++) cin >> b[i];
	sort(a + 1, a + m + 1);
	sort(b + 1, b + n + 1);
	sumA[m] = a[m];
	for (int i = m - 1; i >= 1; i--) sumA[i] = sumA[i + 1] + a[i];
	for (int j = 1; j <= n; j++) sumB[j] = sumB[j - 1] + b[j];
	ll ans = 0;
	int pre = -1;
	for (int i = 1; i <= m; i++) {
		if (a[i] == pre) continue;
		pre = a[i];
		// find j sao cho b[j] <= a[i]
		int l = 1, r = n, res = n + 1;
		while (l <= r) {
			int mid = (l + r) / 2;
			if (b[mid] <= a[i])  res = mid, l = mid + 1;
			else r = mid - 1;
		}
		ans += min(sumA[i], sumB[res]);
	}
	cout << ans << endl;
	return 0;
}
