#include <bits/stdc++.h>
using namespace std;
map<int, int> mp;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    mp[v[i]]++;
  }
  int ans = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    mp[v[i]]--;
    cnt = 0;
    for (auto it : mp) cnt = max(cnt, it.second);
    if (cnt == 1) ans = i + 1;
  }
  cout << ans << endl;
  return 0;
}
