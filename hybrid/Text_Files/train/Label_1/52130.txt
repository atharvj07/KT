import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] s = sc.next().toCharArray();
        char[] t = sc.next().toCharArray();
        int a = 0, b = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'a') {
                a++;
            } else {
                b++;
            }
        }
        if (a != b) {
            System.out.println(-1);
            return;
        }
        List<int[]> swaps = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                boolean found = false;
                for (int j = i + 1; j < n; j++) {
                    if (s[j] != t[j] && s[j] == t[i] && s[i] == t[j]) {
                        swaps.add(new int[]{i + 1, j + 1});
                        char temp = s[i];
                        s[i] = s[j];
                        s[j] = temp;
                        found = true;
                        break;
                    }
                }
                if (!found) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        System.out.println(swaps.size());
        for (int[] swap : swaps) {
            System.out.println(swap[0] + " " + swap[1]);
        }
    }
}
