import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] a = new int[4];
        int[] b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        if (a[3] > b[1] && b[3] > a[1]) {
            System.out.println("Team 1");
        } else if (b[3] > a[1] && a[3] > b[1]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
