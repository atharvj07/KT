n, q, c = map(int, input().split())
stars = [[*map(int, input().split())] for _ in range(n)]
views = [[*map(int, input().split())] for _ in range(q)]
for t, x1, y1, x2, y2 in views:
    ans = 0
    for x, y, s in stars:
        if x1<=x<=x2 and y1<=y<=y2:
            ans += min(c, s+t)
    print(ans)
