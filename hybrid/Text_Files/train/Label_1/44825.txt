import java.io.*;
import java.util.*;

public class AllenAndParkingLot {

    static int n, k;
    static int[][] grid;
    static int[] targetRow;
    static int[] targetCol;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        k = Integer.parseInt(line[1]);

        grid = new int[4][n];
        targetRow = new int[k + 1];
        targetCol = new int[k + 1];

        for (int i = 0; i < 4; i++) {
            line = br.readLine().split(" ");
            for (int j = 0; j < n; j++) {
                grid[i][j] = Integer.parseInt(line[j]);
                if (grid[i][j] > 0) {
                    targetRow[grid[i][j]] = i;
                    targetCol[grid[i][j]] = j;
                }
            }
        }

        int moves = solve();
        if (moves <= 20000) {
            System.out.println(moves);
            for (int i = 0; i < moves; i++) {
                System.out.println(moveList.get(i));
            }
        } else {
            System.out.println(-1);
        }
    }

    static List<String> moveList = new ArrayList<>();

    static int solve() {
        int moves = 0;
        for (int i = 1; i <= k; i++) {
            int carRow = getCarRow(i);
            int carCol = getCarCol(i);
            if (carRow == targetRow[i] && carCol == targetCol[i]) {
                continue;
            }
            if (!moveCarToTarget(i, carRow, carCol)) {
                return -1;
            }
            moves++;
        }
        return moves;
    }

    static boolean moveCarToTarget(int car, int carRow, int carCol) {
        if (carRow == targetRow[car]) {
            return moveCarHorizontally(car, carCol, targetCol[car]);
        } else if (carCol == targetCol[car]) {
            return moveCarVertically(car, carRow, targetRow[car]);
        } else {
            int targetRow1 = carRow, targetCol1 = targetCol[car];
            if (!moveCarVertically(car, carRow, targetRow1)) {
                return false;
            }
            if (!moveCarHorizontally(car, targetCol1, targetCol[car])) {
                return false;
            }
            return true;
        }
    }

    static boolean moveCarHorizontally(int car, int startCol, int endCol) {
        int direction = endCol > startCol ? 1 : -1;
        for (int col = startCol; col != endCol; col += direction) {
            if (grid[targetRow[car]][col] != 0) {
                return false;
            }
            moveCar(car, targetRow[car], col);
        }
        return true;
    }

    static boolean moveCarVertically(int car, int startRow, int endRow) {
        int direction = endRow > startRow ? 1 : -1;
        for (int row = startRow; row != endRow; row += direction) {
            if (grid[row][targetCol[car]] != 0) {
                return false;
            }
            moveCar(car, row, targetCol[car]);
        }
        return true;
    }

    static void moveCar(int car, int row, int col) {
        grid[getCarRow(car)][getCarCol(car)] = 0;
        grid[row][col] = car;
        moveList.add(car + " " + row + " " + col);
    }

    static int getCarRow(int car) {
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == car) {
                    return i;
                }
            }
        }
        return -1;
    }

    static int getCarCol(int car) {
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == car) {
                    return j;
                }
            }
        }
        return -1;
    }
}
