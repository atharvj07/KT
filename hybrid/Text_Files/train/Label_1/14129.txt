#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int N = 10005;
const ll MOD = 1e9 + 7;

int main() {
    ll x, d;
    cin >> x >> d;
    if (d == 1) {
        cout << -1 << endl;
        return 0;
    }
    ll n = (x + d - 1) / d;
    if (n > N) {
        cout << -1 << endl;
        return 0;
    }
    cout << n << endl;
    for (ll i = 0; i < n; i++) {
        cout << d * i + 1 << " ";
    }
    cout << endl;
    return 0;
}
