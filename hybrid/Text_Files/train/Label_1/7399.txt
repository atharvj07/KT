#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> colors(n * k);
    for (int &c : colors) cin >> c;

    vector<vector<int>> intervals(n);
    int current = 0;
    for (int i = 0; i < n; ++i) {
        int c = i + 1;
        intervals[i] = {0, 0};
        for (int j = current; j < n * k; ++j) {
            if (colors[j] == c) {
                if (intervals[i][0] == 0) intervals[i][0] = j + 1;
                intervals[i][1] = j + 1;
                ++current;
            }
        }
    }

    for (const auto &interval : intervals) {
        cout << interval[0] << ' ' << interval[1] << '\n';
    }

    return 0;
}
