mod = int(1e9) + 7

n, k = map(int, input().split())
fac = [1] * (n + 1)
for i in range(2, n + 1):
    fac[i] = i * fac[i - 1] % mod

# pow mod
def powm(x : int, pow : int) -> int:
    if pow == 0:
        return 1
    if pow == 1:
        return x
    val = powm(x, pow // 2)
    val = val * val % mod
    if pow % 2 != 0:
        return val * x % mod
    return val

res = []
for i in range(1, k + 1):
    res.append((fac[n] * powm(fac[n - i] * fac[i], mod - 2)) % mod)
print(*res)
