#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define INF 1e18

const int maxn = 2e5 + 10;
ll dp[maxn][3];

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, m, a, b;
        cin >> n >> m >> a >> b;
        vector<ll> s(m);
        for (int i = 0; i < m; ++i) {
            cin >> s[i];
        }
        sort(s.begin(), s.end());
        s.push_back(INF);

        // dp[i][0] = i-th second, didn't move
        // dp[i][1] = i-th second, moved left
        // dp[i][2] = i-th second, moved right
        memset(dp, -1, sizeof(dp));
        dp[0][0] = 0;
        dp[0][1] = -INF;
        dp[0][2] = -INF;

        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < 3; ++j) {
                if (dp[i][j] == -1) continue;

                // Don't move
                ll &cur = dp[i + 1][0];
                cur = max(cur, dp[i][j]);

                // Move left
                ll &cur1 = dp[i + 1][1];
                if (j == 0 || a > 1) {
                    cur1 = max(cur1, dp[i][j] + (a > 1));
                }

                // Move right
                ll &cur2 = dp[i + 1][2];
                if (j == 0 || a < n) {
                    cur2 = max(cur2, dp[i][j] + (a < n));
                }

                // Drop firecracker
                if (j == 0) {
                    cur = max(cur, dp[i][j] + 1);
                    if (a > 1) cur1 = max(cur1, dp[i][j] + 1);
                    if (a < n) cur2 = max(cur2, dp[i][j] + 1);
                }
            }

            // Explode firecracker
            while (s[i] <= i + 1) {
                for (int j = 0; j < 3; ++j) {
                    if (dp[i][j] == -1) continue;
                    dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + 1);
                }
                ++i;
            }
        }

        // Guard caught hooligan
        ll ans = 0;
        for (int i = 0; i < 3; ++i) {
            ans = max(ans, dp[m][i]);
        }
        cout << ans << endl;
    }

    return 0;
}
