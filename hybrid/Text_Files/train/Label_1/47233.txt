import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int ans = 0;
        for (long i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                ans++;
                while (n % i == 0) n /= i;
            }
        }
        if (n > 1) ans++; // n is prime
        System.out.println(ans);
    }
}
