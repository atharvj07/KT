import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		long x = sc.nextLong();
		long y = sc.nextLong();
		long gcd = 0;
		for (int i = 0; i < n; i++) {
			gcd = gcd(gcd, sc.nextLong());
		}
		System.out.println(gcd == 1 ? 0 : Math.min(x, y));
	}

	public static long gcd(long a, long b) {
		if (b == 0)
			return a;
		return gcd(b, a % b);
	}

}
