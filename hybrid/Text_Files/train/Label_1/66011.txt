def solve(n):
    primes = []
    for i in range(2, n + 1):
        is_prime = True
        for prime in primes:
            if i % prime == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(i)
    
    questions = []
    for prime in primes:
        power = prime
        while power <= n:
            questions.append(power)
            power *= prime

    print(len(questions))
    print(*sorted(questions))

n = int(input())
solve(n)
