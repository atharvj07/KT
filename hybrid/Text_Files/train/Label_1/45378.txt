import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int d1 = in.nextInt();
        int d2 = in.nextInt();
        int[][] points = new int[n * n][2];
        int idx = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                points[idx][0] = i;
                points[idx][1] = j;
                idx++;
            }
        }
        for (int i = 0; i < n * n; i++) {
            for (int j = i + 1; j < n * n; j++) {
                int dist = (int)Math.sqrt(Math.pow(points[i][0] - points[j][0], 2) + Math.pow(points[i][1] - points[j][1], 2));
                if (dist == d1 || dist == d2) {
                    points[j][0] = -1;
                    points[j][1] = -1;
                }
            }
        }
        for (int i = 0; i < n * n; i++) {
            if (points[i][0] != -1 && points[i][1] != -1) {
                System.out.println(points[i][0] + " " + points[i][1]);
            }
        }
    }
}
