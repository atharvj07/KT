n = int(input())
board = [input() for _ in range(n)]
def check(board, i, j, start):
    if i < 0 or i >= n or j < 0 or j >= n or board[i][j] == '.': return False
    if (i, j) in start: return False
    start.add((i, j))
    return check(board, i - 1, j, start) and check(board, i + 1, j, start) and check(board, i, j - 1, start) and check(board, i, j + 1, start)
cnt = 0
for i in range(n):
    for j in range(n):
        if board[i][j] == '#':
            if check(board, i, j, set((i, j))):
                cnt += 1
                board[i][j] = 'x'
if cnt * 5 >= sum(i.count('#') for i in board): print("YES")
else: print("NO")
