def find_min_buns_and_progression(n, heights):
    heights.sort()
    min_buns = float('inf')
    best_a, best_d = 0, 0

    for d in range(max(heights) - min(heights) + 1):
        low = heights[0]
        high = heights[0] + d * (n - 1)
        total_buns = 0
        possible = True

        for i, h in enumerate(heights):
            expected_height = low + d * i
            buns_needed = abs(h - expected_height)

            if h > high:
                possible = False
                break

            total_buns = max(total_buns, buns_needed)

        if possible and total_buns < min_buns:
            min_buns = total_buns
            best_a, best_d = low, d

    print(min_buns)
    print(best_a, best_d)

n = int(input().strip())
heights = list(map(int, input().split()))
find_min_buns_and_progression(n, heights)
