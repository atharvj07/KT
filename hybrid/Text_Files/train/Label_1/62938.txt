n, k = map(int, input().split())
tabs = list(map(int, input().split()))

max_diff = 0

for b in range(1, n + 1):
    e = s = 0
    for i in range(1, n + 1):
        if i != b and (i - b) % k == 0:
            continue
        if tabs[i - 1] == 1:
            e += 1
        else:
            s += 1
    max_diff = max(max_diff, abs(e - s))

print(max_diff)
