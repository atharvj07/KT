#include <bits/stdc++.h>
using namespace std;
#define FOR(a, b) for(int a = 0; a < b; ++a)
#define print(x) cout << x << endl
int32_t main() {
    long long int m, n, k, t;
    cin >> m >> n >> k >> t;
    vector<long long int> a(m), dp(n + 1, -1);
    FOR(i, m) cin >> a[i];
    FOR(i, k) {
        long long int l, r, d;
        cin >> l >> r >> d;
        if(r <= t) {
            FOR(j, n + 1) {
                if(dp[j] != -1 && dp[j] >= d) dp[r] = d;
            }
        }
    }
    dp[0] = LLONG_MAX;
    FOR(i, m) {
        FOR(j, n + 1) {
            if(j + 1 <= t) dp[j + 1] = max(dp[j + 1], min(dp[j], a[i]));
            if(j > 0 && j + 1 <= t) dp[j + 1] = max(dp[j + 1], min(dp[j - 1], a[i]));
        }
    }
    print(dp[n]);
}
