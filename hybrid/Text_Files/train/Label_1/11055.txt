#include <bits/stdc++.h>

using namespace std;

int main()
{
    string str;
    int n;
    cin >> str >> n;
    for (int i = 0; i < str.length(); i++)
    {
        if (str[i] <= 90)
        {
            if (str[i] + n > 90)
                str[i] = 'A' + (str[i] + n - 91);
            else
                str[i] += n;
        }
        else
        {
            if (str[i] + n > 122)
                str[i] = 'a' + (str[i] + n - 123);
            else
                str[i] += n;
        }
    }
    cout << str;
}
