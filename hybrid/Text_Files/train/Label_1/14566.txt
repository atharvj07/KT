#include <bits/stdc++.h>
using namespace std;
bool f(string s) {
  int n = s.size();
  int l = 0;
  int r = 0;
  int idx = 1;
  for (char& c : s) {
    if (c == 'a' + l) l++;
    else if (c == 'a' + r) r++;
    else return false;
    if (l + r == n) return true;
  }
  return false;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    cout << (f(s) ? "YES" : "NO") << endl;
  }
}
