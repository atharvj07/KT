#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    int ans = 0;
    int idx = -1;
    for (int i = 0; i < n; i++) {
        if (i > 0 && v[i].first > v[i - 1].second) {
            idx = i;
        }
        int cur = min(v[i].second, v[min(i + k - 1, n - 1)].second);
        if (ans < cur - v[i].first + 1) {
            ans = cur - v[i].first + 1;
            idx = i;
        }
    }
    cout << ans << endl;
    for (int i = idx; i < idx + k; i++) {
        cout << i + 1 << " ";
    }
    cout << endl;
    return 0;
}
