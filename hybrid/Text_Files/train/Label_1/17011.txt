int main()
{
    ll n, k, p;
    cin >> n >> k >> p;
    if (p > n - k + p || (n - k + p) % 2 != p % 2)
        cout << "NO" << endl;
    else
    {
        cout << "YES" << endl;
        ll odds = (n - k + p) / 2;
        for (ll i = 0; i < k; i++)
        {
            cout << (i < p ? odds : n - 2 * odds);
            for (ll j = 0; j < (i < p ? odds : n - 2 * odds); j++)
            {
                cout << " " << (i < p ? 2 * i + 1 : 2 * i + 2);
            }
            cout << endl;
        }
    }
}
