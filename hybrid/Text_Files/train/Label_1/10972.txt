#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, c, A, B, C;
        cin >> a >> b >> c;
        if (a < b) A = b - a + 1;
        else A = 0;
        if (a < c) A = max(A, c - a + 1);
        if (b < a) B = a - b + 1;
        else B = 0;
        if (b < c) B = max(B, c - b + 1);
        if (c < a) C = a - c + 1;
        else C = 0;
        if (c < b) C = max(C, b - c + 1);
        cout << A << " " << B << " " << C << endl;
    }
    return 0;
}
