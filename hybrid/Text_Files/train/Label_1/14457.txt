#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll INF = 1e9;
const int MAXN = 101;

ll dp[MAXN][MAXN], a[MAXN], sum[MAXN];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    
    string s;
    cin >> s;
    
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        sum[i] = sum[i - 1] + a[i];
    }
    
    for (int l = 1; l <= n; l++)
        for (int i = 1, j = i + l - 1; j <= n; i++, j++) {
            if (l == 1) {
                dp[i][j] = a[1];
            } else {
                if (s[i - 1] == s[j]) {
                    dp[i][j] = max(sum[j] - sum[i - 1], dp[i + 1][j]);
                } else {
                    int need = 1;
                    for (int k = i + 1; k <= j; k++) {
                        need += (s[i - 1] == s[k]);
                        dp[i][j] = max(dp[i][j], dp[i][k - 1] + dp[k + 1][j] + a[need]);
                    }
                }
            }
        }
    
    cout << dp[1][n] << "\n";
    return 0;
}
