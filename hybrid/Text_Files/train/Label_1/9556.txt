#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while(t--) {
        long long a, b, c;
        int fl = 0;
        cin >> a >> b >> c;
        long long x = min(min(a, b), c);
        while(x > 0) {
            a -= x; b -= x; c -= x;
            if(a <= 0 && b <= 0 && c <= 0) {
                cout << "YES" << endl;
                fl = 1;
                break;
            }
            x = min(min(a, b), c);
        }
        if(fl == 0)
            cout << "NO" << endl;
    }
    return 0;
}
