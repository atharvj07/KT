n, k = map(int, input().split())
ratings = list(map(int, input().split()))

unique_ratings = {}
indices = []

for i in range(n):
    if ratings[i] not in unique_ratings:
        unique_ratings[ratings[i]] = i + 1
        indices.append(i + 1)

if len(indices) < k:
    print("NO")
else:
    print("YES")
    print(*indices[:k])
