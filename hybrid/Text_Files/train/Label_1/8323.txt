#include <bits/stdc++.h>

inline bool overlap(int x, int y, int r, int x1, int y1, int r1) {
  return ((x - x1) * (x - x1) + (y - y1) * (y - y1) <= r * r &&
          (x - x1) * (x - x1) + (y - y1) * (y - y1) >= (r - r1) * (r - r1));
}
int main() {
  int r, d, n;
  std::cin >> r >> d >> n;
  int ans = 0;
  while (n--) {
    int x, y, s;
    std::cin >> x >> y >> s;
    if (overlap(x, y, r - d, x, y, s)) ans++;
  }
  std::cout << ans << std::endl;
  return 0;
}
