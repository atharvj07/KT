import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = sorted(map(int, input().split()))

g = [[]] * (n + 1)
for _ in range(n - 1):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

dis = [0] * (n + 1)
vis = [False] * (n + 1)


def bfs(x):
    q = [(x, 0)]
    while q:
        cx, cd = q.pop(0)
        vis[cx] = True
        dis[cx] = cd
        for nx in g[cx]:
            if vis[nx] == False:
                q.append((nx, cd + 1))


bfs(a[0])
a_max = 0
for x in a:
    a_max = max(a_max, dis[x])

vis = [False] * (n + 1)
bfs(a[-1])
b_max = 0
for x in a:
    b_max = max(b_max, dis[x])

print(a_max + b_max)
