#include <iostream>
#include <vector>

const int MAX_N = 2e5 + 10;

int n, m;
std::vector<int> a, p;
std::vector<std::vector<int>> g;

int find(int x) { return p[x] = (p[x] == x ? x : find(p[x])); }
void connect(int x, int y) { p[find(x)] = find(y); }

int main() {
  std::cin >> n >> m;
  a.resize(n + 1);
  p.resize(n + 1);
  g.resize(n + 1);
  for (int i = 1; i <= n; ++i) { std::cin >> a[i]; p[i] = i; }
  for (int i = 1; i < n; ++i) {
    int x, y;
    std::cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  while (m--) {
    int type, x, val;
    std::cin >> type >> x;
    switch (type) {
      case 1:
        std::cin >> val;
        for (int i = 0; i < g[x].size(); ++i) connect(x, g[x][i]);
        a[find(x)] += val;
        break;
      case 2: std::cout << a[find(x)] << std::endl; break;
    }
  }
  return 0;
}
