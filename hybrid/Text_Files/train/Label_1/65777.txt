from collections import defaultdict

for _ in range(int(input())):
    n, H = map(int, input().split())
    a = defaultdict(int)
    for ai in map(int, input().split()): a[ai] += 1

    ans = []
    while H > 0:
        found = False
        for ai in sorted(a, reverse=True):
            if a[ai]:
                H -= ai
                a[ai] -= 1
                if ai not in ans: ans.append(ai)
                found = True
                break
        if not found: break
    print(len(ans))
