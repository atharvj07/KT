#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 10, MAXM = 2e5 + 10;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> w(n + 1), c(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    vector<pair<int, int>> p(m);
    for (int i = 0; i < m; i++) {
        cin >> p[i].first >> p[i].second;
        c[p[i].first]++;
        c[p[i].second]++;
    }
    vector<int> ans;
    for (int i = 1; i <= n; i++) {
        if (c[i] > 0) {
            if (w[i] >= 2) {
                ans.push_back(i);
                ans.push_back(i);
                w[i] -= 2;
            } else if (w[i] >= 1) {
                ans.push_back(i);
                w[i] -= 1;
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        if (w[i] >= 1) {
            ans.push_back(i);
            w[i] -= 1;
        }
    }
    for (int i = 0; i < m; i++) {
        if (w[p[i].first] >= 1) {
            ans.push_back(p[i].first);
            w[p[i].first] -= 1;
        } else if (w[p[i].second] >= 1) {
            ans.push_back(p[i].second);
            w[p[i].second] -= 1;
        } else {
            cout << "DEAD" << endl;
            return 0;
        }
    }
    cout << "ALIVE" << endl;
    for (auto x : ans) {
        cout << x << " ";
    }
    cout << endl;
    return 0;
}
