def solve_molecule(a, b, c):
    total_valence = a + b + c
    if total_valence % 2 != 0:
        return "Impossible"
    ab = (b + a - c) // 2
    bc = (b + c - a) // 2
    ca = (c + a - b) // 2
    if ab < 0 or bc < 0 or ca < 0:
        return "Impossible"
    return f"{ab} {bc} {ca}"

a, b, c = map(int, input().split())
print(solve_molecule(a, b, c))
