import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        int n = scanner.nextInt();
        int[] extensions = new int[n];
        for (int i = 0; i < n; i++) {
            extensions[i] = scanner.nextInt();
        }
        Arrays.sort(extensions);
        int result = -2;
        int min = Math.min(a, b);
        int max = Math.max(a, b);
        int mult = 1;
        for (int i = n - 1; i >= 0; i--) {
            if (min > h && extensions[i] * h < min) continue;
            if (max > w && extensions[i] * w < max) continue;
            int newH = h;
            int newW = w;
            int count = 0;
            while (newH < min || newW < max) {
                if (newH < min) {
                    newH *= extensions[i];
                    count++;
                } else {
                    newW *= extensions[i];
                    count++;
                }
            }
            if (result == -2) result = count;
            else result = Math.min(result, count);
        }
        if (result == -2) result = -1;
        System.out.println(result);
    }
}
