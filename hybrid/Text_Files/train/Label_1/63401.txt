def is_possible_tree(n, c):
    c.sort(reverse=True)
    for i in range(n):
        if c[i] < i + 1:
            return "NO"
        for j in range(i + 1, min(n, i + c[i])):
            c[j] -= 1
        c = c[:i+1] + sorted(c[i+1:], reverse=True)
    return "YES"

n = int(input())
c = list(map(int, input().split()))
print(is_possible_tree(n, c))
