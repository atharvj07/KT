import sys, heapq
input = sys.stdin.readline

n = int(input())
x, y, c, k = [], [], [], []
for i in range(n):
    a, b = map(int, input().split())
    x.append(a); y.append(b)
for i in range(n): c.append(int(input()))
for i in range(n): k.append(int(input()))
ans = []
class DSU:
    def __init__(self, n: int):
        self.par = list(range(n))
        self.siz = [1] * n
    def find(self, x: int) -> int:
        if self.par[x] == x: return x
        self.par[x] = self.find(self.par[x])
        return self.par[x]
    def unite(self, x: int, y: int):
        x = self.find(x); y = self.find(y)
        if x == y: return
        if self.siz[x] > self.siz[y]: x, y = y, x
        self.par[x] = y
        self.siz[y] += self.siz[x]
dsu = DSU(n)
edges = []
for i in range(n):
    for j in range(i + 1, n):
        edges.append((abs(x[i] - x[j]) + abs(y[i] - y[j])) * (k[i] + k[j]), i, j)
edges.sort()
tot = 0
for e, i, j in edges:
    if dsu.find(i) != dsu.find(j):
        tot += e; dsu.unite(i, j)
print(tot + sum(c))
print(n)
for i in range(n):
    if dsu.find(i) == i:
        ans.append(i + 1)
        print(i + 1, end = ' ')
print()
print(len(edges) - n + 1)
for e, i, j in edges:
    if dsu.find(i) != dsu.find(j):
        print(i + 1, j + 1)
        dsu.unite(i, j)
