n,m=map(int,input().split())
a=[[] for i in range(m+1)]
for i in range(n):
    x, *y=map(int,input().split())
    a[x]+=y
bulbs=[False for i in range(m+1)]
for i in range(1,m+1):
    for bulb in a[i]:
        if not bulbs[bulb]:
            bulbs[bulb]=True
            break
print('YES' if all(bulbs) else 'NO')
