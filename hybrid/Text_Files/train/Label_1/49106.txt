import java.util.Scanner;

public class Main {
    private static int[] beads, preFirstIdx, preLastIdx;

    private static void update(int p, int x) {
        beads[p] = x;
        preFirstIdx[x] = p;
        if (preLastIdx[x] == -1) {
            preLastIdx[x] = p;
        }
    }

    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int n = sc.nextInt();
            int m = sc.nextInt();

            // Initialize
            beads = new int[n + 1];
            preFirstIdx = new int[n + 1];
            preLastIdx = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                beads[i] = sc.nextInt();
            }
            for (int i = 1; i <= n; i++) {
                if (preFirstIdx[beads[i]] == 0) {
                    preFirstIdx[beads[i]] = i;
                }
                preLastIdx[beads[i]] = i;
            }

            // Process queries
            while (m-- > 0) {
                int type = sc.nextInt();
                if (type == 1) {
                    int p = sc.nextInt();
                    int x = sc.nextInt();
                    update(p, x);
                } else {
                    int l = sc.nextInt();
                    int r = sc.nextInt();
                    int mem = 0;
                    for (int i = 1; i <= n; i++) {
                        if (preFirstIdx[i] >= l && preLastIdx[i] <= r) {
                            mem += preLastIdx[i] - preFirstIdx[i];
                        }
                    }
                    System.out.println(mem);
                }
            }
        }
    }
}
