#include <bits/stdc++.h>
using namespace std;

#define ll long long
const ll MOD = 1e9 + 7;

void solve() {
  ll n;
  cin >> n;
  vector<pair<int, int>> al, ar, ab, aa;
  for (int i = 0; i < n; i++) {
    string s;
    int a;
    cin >> s >> a;
    if (s == "00") {
      ab.push_back({a, i});
    } else if (s == "10") {
      al.push_back({a, i});
    } else if (s == "11") {
      aa.push_back({a, i});
    } else if (s == "01") {
      ar.push_back({a, i});
    }
  }
  sort(al.begin(), al.end(), greater<pair<int, int>>());
  sort(ar.begin(), ar.end(), greater<pair<int, int>>());
  ll ans = 0;
  ll i = 0, lcnt = 0, rcnt = 0;
  while (i < al.size() && lcnt < (n + 1) / 2) {
    ans += al[i].first;
    lcnt++;
    i++;
  }
  i = 0;
  while (i < ar.size() && rcnt < (n + 1) / 2) {
    ans += ar[i].first;
    rcnt++;
    i++;
  }
  if (lcnt < (n + 1) / 2 || rcnt < (n + 1) / 2) {
    cout << 0 << endl;
    return;
  }
  sort(ab.begin(), ab.end(), greater<pair<int, int>>());
  i = 0;
  while (lcnt + rcnt < n && i < ab.size()) {
    ans += ab[i].first;
    if (al[lcnt - 1].first > ar[rcnt - 1].first) {
      rcnt++;
    } else {
      lcnt++;
    }
    i++;
  }
  while (lcnt + rcnt < n && i < aa.size()) {
    ans += aa[i].first;
    if (al[lcnt - 1].first > ar[rcnt - 1].first) {
      rcnt++;
    } else {
      lcnt++;
    }
    i++;
  }
  cout << ans << endl;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
