import sys
readline = sys.stdin.readline

n, a, b = map(int, readline().strip().split())
ghosts = []
for i in range(n):
    x, vx, vy = map(int, readline().strip().split())
    ghosts.append((x, vx, vy))

ghosts.sort(key=lambda x: x[0])

ex = 0
for i in range(1, n):
    x1, vx1, vy1 = ghosts[i - 1]
    x2, vx2, vy2 = ghosts[i]
    if (vy2 - vy1) * (x2 - x1) > (vx2 - vx1) * (a * x2 + b - a * x1 - b):
        ex += 2
print(ex)
