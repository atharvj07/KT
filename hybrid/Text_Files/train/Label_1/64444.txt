import sys
from fractions import Fraction

n, T = map(int, input().split())
a = list(map(int, input().split()))
t = list(map(int, input().split()))

a_t = [a[i] * t[i] for i in range(n)]

lo, hi = 0, 2e18 + 10
error = 1e-6

def check(mid):
	res = 0
	for i in range(n):
		if mid * a[i] < a_t[i] + error:
			continue
		res += a[i]
	return res

while abs(hi - lo) > error:
	mid = (hi + lo) / 2
	water = check(mid)
	if water == 0 or water * T < sum(a_t) + error:
		hi = mid
	else:
		lo = mid

water = lo
res = 0
for i in range(n):
	if water * a[i] < a_t[i] + error:
		continue

	res += a[i] * Fraction(T, t[i])

print(res)
