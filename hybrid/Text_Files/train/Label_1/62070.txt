n = int(input())
a = [input() for i in range(n)]
x = False
for i in range(n):
    for j in range(0, 4, 2):
        if a[i][j] == 'O' and a[i][j+1] == 'O':
            a[i] = a[i][:j]+'+'*2+a[i][j+2:]
            x = True
            break
if not x: print("NO")
else:
    print("YES")
    for i in a: print(i)
