def count_good_pairs(n, a):
    count = 0
    for i in range(n):
        for j in range(i+1, n):
            if a[i] & a[j] >= a[i] ^ a[j]:
                count += 1
    return count

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(count_good_pairs(n, a))
