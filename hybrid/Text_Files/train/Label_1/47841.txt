import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int q = in.nextInt();
        boolean[] rows = new boolean[n + 1];
        boolean[] cols = new boolean[n + 1];
        for (int i = 0; i < q; i++) {
            int x1 = in.nextInt();
            int y1 = in.nextInt();
            int x2 = in.nextInt();
            int y2 = in.nextInt();
            for (int j = x1; j <= x2; j++) {
                rows[j] = true;
            }
            for (int j = y1; j <= y2; j++) {
                cols[j] = true;
            }
        }
        int count = 0;
        for (int i = 1; i <= n; i++) {
            if (!rows[i] && !cols[i]) {
                count++;
            }
        }
        System.out.println(count);
    }
}
