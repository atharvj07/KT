#include <stdio.h>
int d[100001][6];
int a[100001][6];
int s[100001];
int main() {
  int n, m, k, i, j, l, r, mx, x, y;
  scanf("%d %d %d", &n, &m, &k);
  l = r = mx = x = y = 0;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) scanf("%d", &a[i][j]);
  }
  for (i = 1; i <= m; i++) {
    s[0] = 0;
    for (j = 1; j <= n; j++) {
      s[j] = s[j - 1] + a[j][i];
      d[j][i] = d[j - 1][i];
      if ((s[j] <= k) && (j - s[j] + a[j][i] > d[j - 1][i])) {
        d[j][i] = j - s[j] + a[j][i];
        if (d[j][i] > mx) {
          mx = d[j][i];
          x = i;
          y = j;
        }
      }
    }
  }
  for (; a[y][x] && k; y--) {
    k--;
    a[y][x]--;
    if (r - l + 1 < mx) {
      l = y;
      r = l + d[y][x] - 1;
    }
  }
  for (i = 1; i <= m; i++) printf("%d ", (l ? mx - s[r] : 0));
  return 0;
}
