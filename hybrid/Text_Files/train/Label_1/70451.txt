from sys import stdin
from heapq import heappop, heappush

def read_ints():
    return list(map(int, stdin.readline().split()))

def dijkstra(adj, start):
    n = len(adj)
    dist = [float('inf')] * n
    dist[start] = 0
    pq = [(0, start)]
    while pq:
        d, u = heappop(pq)
        if d > dist[u]:
            continue
        for v, w in adj[u]:
            if dist[v] > max(d, w):
                dist[v] = max(d, w)
                heappush(pq, (dist[v], v))
    return dist

n, m, k = read_ints()
special = read_ints()
edges = [read_ints() for _ in range(m)]

adj = [[] for _ in range(n + 1)]
for u, v, w in edges:
    adj[u].append((v, w))
    adj[v].append((u, w))

distances = [float('-inf')] * k
for i, x in enumerate(special):
    d = dijkstra(adj, x)
    for j, y in enumerate(special):
        if i != j:
            distances[i] = max(distances[i], d[y])

print(' '.join(map(str, distances)))
