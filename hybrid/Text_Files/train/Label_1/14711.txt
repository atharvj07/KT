#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    string n;
    cin >> n;
    bool odd = n.back() & 1;
    int ans = INT_MAX;
    for (int len = 1; len <= n.size(); len++) {
      if (len % 2 == odd) ans = min(ans, (len + 1) / 2);
    }
    cout << (ans == INT_MAX ? -1 : ans) << '\n';
  }
}
