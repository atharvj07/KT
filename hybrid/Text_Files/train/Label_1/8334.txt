#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
char s[N];
int main() {
  cin >> s;
  int n = strlen(s), flag = 0;
  for (int i = 0; i < n; ++i) {
    if (s[i] == '"') {
      flag ^= 1;
      continue;
    }
    if (!flag && isspace(s[i])) {
      continue;
    }
    cout << "<";
    while (i < n && (!isspace(s[i]) || flag)) {
      cout << s[i];
      ++i;
    }
    --i;
    cout << ">" << '\n';
  }
}
