#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll maxn = 2e5 + 5;
const ll mod = 1e9 + 7;
ll n, m, k, a[maxn];
ll dp[maxn][25];
ll f(ll pos, ll c) {
    if (pos > n) return 0;
    if (dp[pos][c] != -1) return dp[pos][c];
    ll ret = f(pos + 1, c);
    if (c < k) { ret = max(ret, f(pos + 1, c + 1) + a[pos]); }
    return dp[pos][c] = ret;
}
int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + 1 + n, greater<int>());
    memset(dp, -1, sizeof(dp));
    ll mx = f(1, 0);
    cout << mx << endl;
    return 0;
}
