import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long a = 0, b = 0;
        for (int i = 0; i < n; i++) {
            a += sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            b += sc.nextLong();
        }
        if (a > b) System.out.println("NO");
        else System.out.println("YES");
    }
}
