#include <bits/stdc++.h>

#define endl '\n'
#define ll long long

using namespace std;
const ll maxN = 5e5 + 5;
ll minDist[maxN], lastPos[maxN], n, m, l, r, a[maxN];

int main() {
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        if (lastPos[a[i]]) minDist[i] = min(minDist[i], i - lastPos[a[i]]);
        lastPos[a[i]] = i;
    }

    for (ll i = 1; i <= n; i++) {
        if (lastPos[a[i]]) minDist[i] = min(minDist[i], i - lastPos[a[i]]);
    }

    while (m--) {
        cin >> l >> r;
        if (minDist[r] <= r - l) cout << minDist[r] << endl;
        else cout << -1 << endl;
    }

    return 0;
}
