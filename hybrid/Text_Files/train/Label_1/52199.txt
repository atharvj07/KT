import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long[][] islands = new long[n][2];
        for (int i = 0; i < n; i++) {
            islands[i][0] = scanner.nextLong();
            islands[i][1] = scanner.nextLong();
        }

        TreeSet<Long> bridges = new TreeSet<>();
        for (int i = 0; i < m; i++) {
            bridges.add(scanner.nextLong());
        }

        boolean possible = true;
        long[] answer = new long[n - 1];
        for (int i = 1; i < n; i++) {
            long diff = islands[i][0] - islands[i - 1][1];
            if (diff > 0) {
                if (!bridges.contains(diff)) {
                    possible = false;
                    break;
                }
                answer[i - 1] = bridges.pollFirst();
            } else {
                diff = -diff;
                if (!bridges.contains(diff)) {
                    diff++;
                    if (!bridges.contains(diff)) {
                        possible = false;
                        break;
                    }
                }
                answer[i - 1] = bridges.pollLast();
            }
        }

        if (possible) {
            System.out.println("Yes");
            for (long a : answer) {
                System.out.print(a + " ");
            }
            System.out.println();
        } else {
            System.out.println("No");
        }
    }
}
