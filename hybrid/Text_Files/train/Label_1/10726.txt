#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  ll max_time = 0;
  for (ll i = 0; i <= n; i++) {
    ll time = 0;
    if (i > 0) time += a[i - 1];
    if (i < n) time += m - a[i];
    for (ll j = i; j < n - 1; j++) time += a[j + 1] - a[j] - 1;
    max_time = max(max_time, time);
  }
  cout << max_time << endl;
  return 0;
}
