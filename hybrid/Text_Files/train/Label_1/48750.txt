import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int total = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            total += a[i];
        }
        boolean found = false;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] + a[j] <= b[i] && a[i] + a[j] <= b[j]) {
                    found = true;
                    break;
                }
            }
        }
        if (found || total <= b[0]) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
