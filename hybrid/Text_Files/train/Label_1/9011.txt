#include <iostream>
#include <string>

using namespace std;

int main() {
  string a;
  cin >> a;
  int n = a.size();
  char del = '0';
  int pos = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == del) {
      if (pos == 0) pos = i;
    } else {
      del = a[i];
      if (i > pos) pos = i;
    }
  }
  a.erase(pos, 1);
  cout << a << endl;
  return 0;
}
