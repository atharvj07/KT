from sys import stdin
input = stdin.readline

n = int(input())
L = [0] * n
R = [0] * n
for i in range(n):
    l, r = map(int, input().split())
    L[i] = l
    R[i] = r
L.sort(reverse=True)
R.sort(reverse=True)
ans = 0
while L and R:
    if L[0] > R[0]:
        ans += L[0]
        L.pop(0)
    else:
        ans += R[0]
        R.pop(0)
print(ans)
