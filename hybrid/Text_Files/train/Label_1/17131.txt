#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
  string s[6];
  for (int i = 0; i < 6; i++) cin >> s[i];
  int ans = 1, idx = 0;
  for (int i = 0; i < 6; i++) {
    int cnt = count(s[i].begin(), s[i].end(), '.');
    if (cnt > ans) {
      ans = cnt, idx = i;
    }
  }
  for (int i = 0; i < 8; i++) {
    if (s[idx][i] == '.') {
      s[idx][i] = 'P';
      break;
    }
  }
  for (int i = 0; i < 6; i++) cout << s[i] << '\n';
  return 0;
}
