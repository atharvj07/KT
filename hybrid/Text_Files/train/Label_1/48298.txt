import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long[] b = a.clone();
        Arrays.sort(b);
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                ok = false;
                break;
            }
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        List<Pair> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) {
                    list.add(new Pair(i + 1, j + 1));
                }
            }
        }
        System.out.println(list.size());
        for (Pair p : list) {
            System.out.println(p.x + " " + p.y);
        }
    }

    static class Pair {
        int x, y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
