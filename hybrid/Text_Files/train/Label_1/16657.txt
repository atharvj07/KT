#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> a(n), b(n);

  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }

  long long ans = 0;

  for (int i = 0; i < n; i++) {
    if (a[i] + b[i] < 2 * a[i]) {
      ans -= 2 * a[i];
    } else {
      ans += (a[i] * (b[i] - a[i]));
    }
  }

  cout << ans << endl;

  return 0;
}
