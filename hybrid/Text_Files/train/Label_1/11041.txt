#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  int xorval = 0;
  for (int i : a) xorval ^= i;

  if (xorval == 0) {
    cout << "YES\n0";
  } else {
    bool found = false;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if ((a[i] ^ a[j]) == xorval) {
          cout << "YES\n1\n" << i + 1 << " " << j + 1 << " " << n << endl;
          found = true;
          break;
        }
      }
      if (found) break;
    }
    if (!found) cout << "NO";
  }
}
