import math

for _ in range(int(input())):
    n, m, k = map(int, input().split())
    arr = [
        str((math.ceil(n / m) if i % 2 else math.floor(n / m)))
        + " "
        + " ".join(map(str, range(1 + i * math.ceil(n / m), 1 + (i + 1) * math.ceil(n / m))))
        for i in range(m)
    ]
    for _ in range(k):
        print(*arr, sep="\n")
        print()
