#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    set<pair<char, int>> se;
    int ans = 0;
    for (int i = 0; i < s.size(); i++) {
      if (se.find(make_pair(s[i], i)) != se.end())
        ans++;
      else {
        ans += 5;
        se.insert(make_pair(s[i], i));
      }
    }
    cout << ans << "\n";
  }
  return 0;
}
