import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static int[][][] dp;
    static int n, m;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, -1, 0, 1};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            String[] parts = br.readLine().split(" ");
            n = Integer.parseInt(parts[0]);
            m = Integer.parseInt(parts[1]);
            dp = new int[n][m][(1 << 16) + 1];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    Arrays.fill(dp[i][j], -1);
                }
            }
            int[][] board = new int[n][m];
            for (int i = 0; i < n; i++) {
                parts = br.readLine().split(" ");
                for (int j = 0; j < m; j++) {
                    board[i][j] = Integer.parseInt(parts[j]);
                }
            }
            int winner = solve(0, 0, board);
            if (winner == 1) {
                System.out.println("Ashish");
            } else {
                System.out.println("Vivek");
            }
        }
    }

    public static int solve(int x, int y, int[][] board) {
        if (x == n) {
            x = 0;
            y++;
        }
        if (y == m) {
            return 0;
        }
        if (dp[x][y][board[x][y]] != -1) {
            return dp[x][y][board[x][y]];
        }
        int ans = 0;
        if (board[x][y] == 0) {
            int mask = 0;
            int count = 0;
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (isInBound(nx, ny) && board[nx][ny] > 0) {
                    mask |= (1 << count);
                }
                count++;
            }
            if ((mask & board[x][y]) == 0) {
                ans = 1 ^ solve(x + 1, y, board);
            }
        }
        return dp[x][y][board[x][y]] = ans;
    }

    public static boolean isInBound(int x, int y) {
        return x >= 0 && x < n && y >= 0 && y < m;
    }
}
