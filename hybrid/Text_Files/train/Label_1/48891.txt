import java.util.*;

class Main {
    static final int INF = 0x3f3f3f3f;
    static int dij(int N, List<List<Integer>> adj, int s, int t, int[] min) {
        int[][] dis = new int[N + 1][N + 1];
        for (int i = 0; i <= N; i++)
            Arrays.fill(dis[i], INF);
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        pq.offer(new int[]{0, s});
        dis[s][s] = 0;
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            if (cur[0] > min[cur[1]]) continue;
            for (int v : adj.get(cur[1])) {
                if (cur[0] + 1 < dis[cur[1]][v]) {
                    dis[cur[1]][v] = dis[v][cur[1]] = cur[0] + 1;
                    pq.offer(new int[]{cur[0] + 1, v});
                }
            }
        }
        int cnt = 0;
        for (int i = 1; i <= N; i++) {
            if (dis[s][i] != INF && dis[t][i] != INF) cnt++;
        }
        return cnt;
    }

    static int query(int N, List<List<Integer>> adj, int s, int t) {
        int[] min = new int[N + 1];
        Arrays.fill(min, INF);
        int[][] dis = new int[N + 1][N + 1];
        for (int i = 0; i <= N; i++)
            Arrays.fill(dis[i], INF);
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        pq.offer(new int[]{0, s});
        dis[s][s] = 0;
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            if (cur[0] > min[cur[1]]) continue;
            for (int v : adj.get(cur[1])) {
                if (cur[0] + 1 < dis[cur[1]][v]) {
                    dis[cur[1]][v] = dis[v][cur[1]] = cur[0] + 1;
                    pq.offer(new int[]{cur[0] + 1, v});
                }
            }
            min[cur[1]] = cur[0];
        }
        if (dis[s][t] == INF) return -1;
        return dij(N, adj, s, t, min);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt(), s = sc.nextInt(), t = sc.nextInt();
        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 0; i <= N; i++)
            adj.add(new ArrayList<>());
        for (int i = 0; i < M; i++) {
            int u = sc.nextInt(), v = sc.nextInt();
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        System.out.println(query(N, adj, s, t));
    }
}
