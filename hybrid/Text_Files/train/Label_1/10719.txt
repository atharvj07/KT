#include <iostream>
#include <vector>
using std::cin;
using std::cout;
using std::vector;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) { cin >> p[i]; }
  int l = 0, r = n - 1;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] == i + 1) continue;
    if (p[i] == l + 1) {
      l++;
    } else if (p[i] == r + 1) {
      r--;
    } else {
      ans++;
      l++;
    }
  }
  cout << ans << '\n';
}
