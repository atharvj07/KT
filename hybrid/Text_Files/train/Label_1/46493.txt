import java.util.Scanner;

public class Main {

    private static String original, target;
    private static int[][] dp;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        original = in.nextLine();
        target = in.nextLine();
        dp = new int[original.length() + 1][target.length() + 1];
        for (int i = 0; i <= original.length(); i++) {
            for (int j = 0; j <= target.length(); j++) {
                dp[i][j] = -1;
            }
        }
        int result = solve(0, 0);
        if (result != Integer.MAX_VALUE) {
            System.out.println(result);
        } else {
            System.out.println(-1);
        }
    }

    private static int solve(int i, int j) {
        if (i == original.length() && j == target.length()) {
            return 0;
        }
        if (i == original.length() || j == target.length()) {
            return Integer.MAX_VALUE;
        }
        if (dp[i][j] != -1) {
            return dp[i][j];
        }
        int result = Integer.MAX_VALUE;
        if (original.charAt(i) == target.charAt(j)) {
            result = solve(i + 1, j + 1);
        } else {
            result = Math.min(result, solve(i + 1, j + 1) + 1);
            if (original.charAt(i) == '4') {
                result = Math.min(result, solve(i + 1, j + 1) + 1);
            }
            if (target.charAt(j) == '4') {
                result = Math.min(result, solve(i, j + 1) + 1);
            }
        }
        dp[i][j] = result;
        return result;
    }
}
