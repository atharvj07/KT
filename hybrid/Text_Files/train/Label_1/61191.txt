x, y, l, r = map(int, input().split())

unlucky_years = set()
for a in range(60):
    for b in range(60):
        unlucky_year = x**a + y**b
        if l <= unlucky_year <= r:
            unlucky_years.add(unlucky_year)
        elif unlucky_year > r:
            break

unlucky_years = sorted(list(unlucky_years))
unlucky_years = [l-1] + unlucky_years + [r+1]

max_length = 0
for i in range(1, len(unlucky_years)):
    max_length = max(max_length, unlucky_years[i] - unlucky_years[i-1] - 1)

print(max_length)
