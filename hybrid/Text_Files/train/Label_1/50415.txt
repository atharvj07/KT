import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        for (int i = 0; i < n; i++) {
            long timeVanya = (long) Math.ceil((double) a[i] / x);
            long timeVova = (long) Math.ceil((double) a[i] / y);
            if (timeVanya == timeVova) {
                System.out.println("Both");
            } else if (timeVanya > timeVova) {
                System.out.println("Vova");
            } else {
                System.out.println("Vanya");
            }
        }
    }
}
