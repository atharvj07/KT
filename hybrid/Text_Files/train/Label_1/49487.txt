import java.io.*;
import java.util.*;

public class Main {

    static int mod = 998244353;

    static class Node {
        int u, v, i;

        public Node(int u, int v, int i) {
            this.u = u;
            this.v = v;
            this.i = i;
        }
    }

    static int dfs1(int u, int p, int[] in, int[] out, int time, List<Node> edges) {
        in[u] = time++;
        for (Node edge : edges) {
            if (edge.u == u && edge.v != p) {
                time = dfs1(edge.v, u, in, out, time, edges);
            } else if (edge.v == u && edge.u != p) {
                time = dfs1(edge.u, u, in, out, time, edges);
            }
        }
        out[u] = time;
        return time;
    }

    static int dfs2(int u, int p, int par, int count, int[] in, int[] out, List<Node> edges, Map<Integer, Integer> freq) {
        int cur = count;
        for (Node edge : edges) {
            if ((edge.u == u && edge.v != p) || (edge.v == u && edge.u != p)) {
                if (in[edge.u] < in[edge.v]) {
                    cur = dfs2(edge.v, u, edge.i, count + 1, in, out, edges, freq);
                } else {
                    cur = dfs2(edge.u, u, edge.i, count + 1, in, out, edges, freq);
                }
            }
        }
        int val = freq.getOrDefault(cur, 0);
        freq.put(cur, val + 1);
        return cur;
    }

    static int calculate(int n, List<Node> edges) {
        int[] in = new int[n + 1];
        int[] out = new int[n + 1];
        dfs1(1, 1, in, out, 1, edges);
        Map<Integer, Integer> freq = new HashMap<>();
        dfs2(1, 1, 0, 0, in, out, edges, freq);
        long ans = 1;
        for (int val : freq.values()) {
            ans = (ans * val) % mod;
        }
        return (int) ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Node> edges = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            String[] s = br.readLine().split(" ");
            int u = Integer.parseInt(s[0]);
            int v = Integer.parseInt(s[1]);
            edges.add(new Node(u, v, i + 1));
        }
        int res = calculate(n, edges);
        System.out.println(res);
    }
}
