#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> room(n);
    for (int i = 0; i < n; i++) {
        cin >> room[i];
    }
    int ans = 0;
    for (int j = 0; j < n; j++) {
        bool clean = true;
        for (int i = 0; i < n; i++) {
            if (room[i][j] == '0') {
                clean = false;
                break;
            }
        }
        if (clean) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
