#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;

    stack<char> st;
    for (char c : s) {
      if (st.empty()) {
        st.push(c);
        continue;
      }
      if (st.top() == c && c == 'A') {
        st.push(c);
      } else if (st.top() != c) {
        st.push(c);
      } else {
        st.pop();
      }
    }
    cout << st.size() << endl;
  }
  return 0;
}
