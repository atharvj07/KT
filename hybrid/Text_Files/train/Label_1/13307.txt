#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n,p,k;
    cin>>n>>p>>k;
    if(p==1) cout<<"<< ";
    for(long long i = max(1ll, p-k); i<=p-1;i++) cout<<i<<" ";
    cout<<"("<<p<<") ";
    for(long long i = p+1; i<=min(n,p+k-1);i++) cout<<i<<" ";
    if(p==n) cout<<" >>";
    cout<<endl;
    return 0;
}
