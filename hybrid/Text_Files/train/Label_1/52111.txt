import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] t = new long[n][2];
        for (int i = 0; i < n; i++) {
            t[i][0] = sc.nextInt();
            t[i][1] = sc.nextInt();
        }
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = t[i][0] - i;
            b[i] = t[i][1] + i;
        }
        long p = Long.MAX_VALUE;
        long s = Long.MIN_VALUE;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > b[i]) {
                if (a[i] < p) {
                    p = a[i];
                    cnt = i;
                }
            } else {
                if (b[i] > s) {
                    s = b[i];
                    cnt = i;
                }
            }
        }
        if (p > s) {
            System.out.println(1);
            System.out.println(cnt + 1);
        } else {
            System.out.println(0);
        }
    }
}
