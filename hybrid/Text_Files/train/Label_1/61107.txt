fac = [1 for _ in range(65)]
for i in range(2, 65):
    fac[i] = (fac[i - 1] * i) % (10**9 + 7)
mod = 10**9 + 7
def binpow(base, p):
    if p == 0: return 1
    if p == 1: return base
    val = binpow(base, int(p / 2))
    val = (val * val) % mod
    if p % 2 != 0: val = (val * base) % mod
    return val % mod
def calc(n):
    if n == 0: return 1
    if n == 1: return 6
    val = 6 * (calc(n - 1) % mod)
    val = (val * fac[2 * (n - 1)])
    val = (val * binpow(fac[n - 1], mod - 2))
    val = (val * binpow(fac[n], mod - 2))
    return val % mod
n = int(input())
print(calc(n))
