import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        in.nextLine();
        char[] s = in.nextLine().toCharArray();
        in.close();

        boolean found = false;
        char[] next = null;
        for (int i = n - 1; i >= 0; i--) {
            char cur = s[i];
            if (cur == 'a' + p - 1) {
                continue;
            }
            next = Arrays.copyOf(s, n);
            next[i]++;
            if (next[i] > 'a' + p - 1) {
                next[i] = 'a';
                for (int j = i + 1; j < n; j++) {
                    next[j] = 'a';
                }
            }
            found = true;
            break;
        }

        if (found) {
            System.out.println(new String(next));
        } else {
            System.out.println("NO");
        }
    }
}
