#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e5 + 5;
int32_t main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    int start = -1e9, end = 1e9;
    while (start <= end) {
        int mid = (start + end) / 2;
        int ans = 0;
        for (int i = 0; i < n; i++) ans += abs(v[i] - mid);
        int temp = 0;
        for (int i = 0; i < n; i++) { temp += abs(v[i] - (mid + 1)); }
        if (ans < temp)
            end = mid - 1;
        else
            start = mid + 1;
    }
    cout << start << '\n';
}
