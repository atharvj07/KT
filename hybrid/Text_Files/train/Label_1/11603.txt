#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; ++i) { cin >> a[i]; }
        sort(a.begin(), a.end());
        int cnt = 0, pos = 1;
        for (int i = 0; i < n; ++i) {
            for (int j = i + 1; j < n; ++j) {
                if (a[j] - a[i] == pos) pos++;
            }
            if (pos == 1000001) {
                cout << "NO" << endl;
                goto end;
            }
            b[i] = pos++;
        }
        cout << "YES" << endl;
        for (int i = 0; i < n; ++i) { cout << b[i] << " "; }
        cout << endl;
    end:;
    }
}
