#include <iostream>
#include <vector>
using namespace std;

int n, k;
vector<int> a;
long long ans = 0;

void go(int i, long long sum) {
    if (i == n) {
        if (sum % k == 0) ans++;
        return;
    }
    go(i + 1, sum + a[i]);
    go(i + 1, sum);
}

int main() { ios_base::sync_with_stdio(false);
    cin >> n >> k;
    a.resize(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    go(0, 0);
    cout << ans << endl;
    return 0;
}
