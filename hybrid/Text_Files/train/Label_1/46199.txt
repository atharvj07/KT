import java.util.*;
 
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt(), x = sc.nextInt();
        int[] freq = new int[1000001], ans = new int[q + 1];
        for (int i = 0; i <= q; i++)
            ans[i] = 0;
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            freq[a]++;
            while (freq[ans[i]] != 0)
                ans[i]++;
            for (int j = 0; j < ans[i]; j++)
                if (j + x <= 1000000 && freq[j + x] != 0)
                    ans[i] = j + x;
            System.out.println(ans[i]);
        }
    }
}
