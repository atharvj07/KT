from bisect import bisect_right

l = input
n = int(l())
s = l().rstrip()
ans = []
for i in range(n-1):
    if s[i]>s[i+1]:
        r = bisect_right(s, s[i], i+1, n)-1
        if r==i+1:
            continue
        else:
            ans.append([i+1,r])
if len(ans)>0:
    print('YES')
    print(*ans[0])
else:
    print('NO')
