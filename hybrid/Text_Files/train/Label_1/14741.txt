#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int>> fact(int a) {
    vector<pair<int, int>> ret;
    for (int i = 1; i * i <= a; i += 1) {
        if (a % i == 0) {
            ret.push_back({i, a / i});
        }
    }
    return ret;
}
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int p = 0, g = 0, s = 0;
    for (int i = 0; i < n; i++) {
        p += a[i] == 1;
        g += a[i] == 2;
        if (p >= g and (i != n - 1 or p != g)) {
            s++;
            p = g = 0;
        }
    }
    vector<pair<int, int>> Q = fact(s);
    vector<pair<int, int>> ans;
    for (auto u : Q) {
        int t = (n - 1) / u.first;
        if ((n - 1) % u.first == 0 and 0 < t and t % u.second == 0) {
            ans.push_back({u.first, t});
        }
    }
    cout << ans.size() << '\n';
    for (auto u : ans) { cout << u.first << ' ' << u.second << '\n'; }
    return 0;
}
