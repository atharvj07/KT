#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'M') cnt++;
        else {
            cnt--;
            if (cnt <= 0) cout << i + 1 << endl, exit(0);
        }
    }
    cout << 0 << endl;
    return 0;
}
