#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;

const int N = 101;
int dp[N][N][N];

int solve(int sum, int k, int d) {
  if (sum < d) return 0;
  int &ans = dp[sum][k][d];
  if (~ans) return ans;
  ans = 0;
  for (int i = d; i <= sum; i++) {
    if (i == sum) ans = (ans + 1) % mod;
    if (k == 1) continue;
    ans = (ans + solve(sum - i, k - 1, d)) % mod;
  }
  return ans;
}

int main() {
  memset(dp, -1, sizeof dp);
  int sum, k, d;
  cin >> sum >> k >> d;
  cout << solve(sum, k, d) << endl;
  return 0;
}
