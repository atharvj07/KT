#include <iostream>
#include <vector>
#define MOD 1000000007

using namespace std;

int main() {
  long long l, r, res = 1;
  cin >> l >> r;
  int m = l % 4;
  if (m == 0) {
    if ((r - l + 1) <= 8) {
      for (int i = 0; i < (r - l + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    } else {
      int k = (r - l + 1) % 8, m = (r - l + 1) / 8;
      for (int i = 0; i < k / 2; i++) res *= 2, res %= MOD;
      res--;
      res *= m, res %= MOD;
      for (int i = 0; i < (k + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    }
  } else if (m == 1) {
    res = 2, res %= MOD;
    if ((r - l + 1) <= 10) {
      for (int i = 0; i < (r - l + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    } else {
      int k = (r - l + 1) % 10, m = (r - l + 1) / 10;
      for (int i = 0; i < k / 2; i++) res *= 2, res %= MOD;
      res--;
      res *= m, res %= MOD;
      for (int i = 0; i < (k + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    }
  } else if (m == 2) {
    res = 5, res %= MOD;
    if ((r - l + 1) <= 12) {
      for (int i = 0; i < (r - l + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    } else {
      int k = (r - l + 1) % 12, m = (r - l + 1) / 12;
      for (int i = 0; i < k / 2; i++) res *= 2, res %= MOD;
      res--;
      res *= m, res %= MOD;
      for (int i = 0; i < (k + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    }
  } else if (m == 3) {
    res = 11, res %= MOD;
    if ((r - l + 1) <= 14) {
      for (int i = 0; i < (r - l + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    } else {
      int k = (r - l + 1) % 14, m = (r - l + 1) / 14;
      for (int i = 0; i < k / 2; i++) res *= 2, res %= MOD;
      res--;
      res *= m, res %= MOD;
      for (int i = 0; i < (k + 1) / 2; i++) res *= 2, res %= MOD;
      res--;
      res %= MOD;
    }
  }
  cout << res << endl;
  return 0;
}
