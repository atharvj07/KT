#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin>>n;
    int arr[n+1],dp[n+1][3];
    for(int i=1;i<=n;++i) cin>>arr[i];
    dp[1][0]=0;
    dp[1][1]=0;
    dp[1][2]=1;
    for(int i=2;i<=n;++i){
        dp[i][0]=min(dp[i-1][1],dp[i-1][2]);
        dp[i][1]=(arr[i]==1||arr[i]==3)?min(dp[i-1][2],1+dp[i-1][0]):1e9;
        dp[i][2]=(arr[i]==2||arr[i]==3)?min(dp[i-1][1],1+dp[i-1][0]):1e9;
    }
    cout<<dp[n][0]<<endl;
    return 0;
}
