import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static int[] z, ac;
    static List<Integer> len;
    static char[] arr;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        arr = br.readLine().toCharArray();
        int n = arr.length;
        z = new int[n];
        ac = new int[n];
        len = new ArrayList<>();
        Arrays.fill(ac, 0);
        Arrays.fill(z, 0);
        z[0] = n;
        int L = 0, R = 0, r, l;
        for (int i = 1; i < n; i++) {
            if (i < R) {
                z[i] = Math.min(R - i, z[i - L]);
            }
            while (i + z[i] < n && arr[z[i]] == arr[i + z[i]]) {
                z[i]++;
            }
            if (i + z[i] - 1 > R) {
                L = i;
                R = i + z[i] - 1;
            }
        }
        for (int i = 0; i < n; i++) {
            if (z[i] == 1)
                len.add(0);
            else if (i + z[i] - 1 < n && arr[i + z[i] - 1] == arr[z[i] - 1])
                len.add(i);
            else
                len.add(0);
        }
        for (int i = 0; i < n; i++) {
            if (len.get(i) == 0)
                ac[i] = 0;
            else
                ac[i] = 1 + (i > len.get(i) ? ac[i - len.get(i)] : 0);
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                l = i;
                r = j;
                int inc = 0;
                while (l <= r) {
                    if (l >= len.get(l)) {
                        if (len.get(l) == 0)
                            break;
                        l = l - len.get(l);
                        inc++;
                    } else
                        l++;
                    if (r >= len.get(r)) r -= len.get(r);
                    else break;
                }
                int len = r - l + 1;
                if (l <= r) ans += ac[r] - (l > 0 ? ac[l - 1] : 0);
            }
        }
        System.out.println(ans);
    }
}
