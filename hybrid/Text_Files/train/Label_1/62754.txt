def solve():
    n, x = map(int, input().split())
    a = list(map(int, input().split()))

    if len(a) != len(set(a)):
        print(0)
        return

    b = set()
    c = set()
    for num in a:
        if num in b:
            print(1)
            return
        b.add(num)
        if num & x in b or num & x in c:
            print(1)
            return
        c.add(num & x)
    
    if len(c) != len(set(c)):
        print(2)
    else:
        print(-1)

solve()
