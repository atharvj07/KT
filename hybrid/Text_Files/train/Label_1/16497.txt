#include <bits/stdc++.h>
using namespace std;

char grid[55][55];
int n, m;
bool vis[55][55];

bool valid(int x, int y) {
    return x >= 0 && x < n && y >= 0 && y < m && !vis[x][y] && grid[x][y] != '.';
}

bool dfs(int x, int y, int px, int py, char c, int cnt) {
    if (cnt >= 4 && x == px && y == py) return true;
    vis[x][y] = true;
    bool ret = false;
    if (valid(x + 1, y) && grid[x + 1][y] == c) ret |= dfs(x + 1, y, x, y, c, cnt + 1);
    if (valid(x - 1, y) && grid[x - 1][y] == c) ret |= dfs(x - 1, y, x, y, c, cnt + 1);
    if (valid(x, y + 1) && grid[x][y + 1] == c) ret |= dfs(x, y + 1, x, y, c, cnt + 1);
    if (valid(x, y - 1) && grid[x][y - 1] == c) ret |= dfs(x, y - 1, x, y, c, cnt + 1);
    vis[x][y] = false;
    return ret;
}

bool solve() {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (!vis[i][j] && grid[i][j] != '.') {
                if (dfs(i, j, i, j, grid[i][j], 1)) return true;
            }
        }
    }
    return false;
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> grid[i];
    }
    cout << (solve() ? "Yes" : "No") << endl;
    return 0;
}
