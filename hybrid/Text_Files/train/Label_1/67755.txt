n, k = map(int, input().split())
s = input()

def can_be_exactly_k(s, k):
    max_n = 0
    cur_n = 0
    for char in s:
        if char == 'N':
            cur_n += 1
            max_n = max(max_n, cur_n)
        else:
            cur_n = 0
    if max_n == k:
        return "YES"
    for i in range(n):
        if s[i] == '?':
            left_N = 'N' in s[max(0, i-k):i]
            right_N = 'N' in s[i+1:min(n, i+k+1)]
            if not left_N and not right_N:
                return "YES"
            if (i > 0 and s[i-1] == 'N' and (i == 1 or s[i-2] != 'N')) or (i < n-1 and s[i+1] == 'N' and (i == n-2 or s[i+2] != 'N')):
                if k == 1:
                    return "YES"
    return "NO"

print(can_be_exactly_k(s, k))
