from sys import stdin

n = int(stdin.readline())
p = [int(x) for x in stdin.readline().split()]
v = [[0 for j in range(n+1)] for i in range(n+1)]
r = [[0 for j in range(n+1)] for i in range(n+1)]
x = 0
for i in range(1,n+1):
    for j in range(1,n+1):
        if i==j:
            v[i][j] = 1
            r[i][j] = p[i-1]
            x^=p[i-1]
        elif i>j:
            v[i][j] = 1
            r[i][j] = x
print(-1 if x else '\n'.join([' '.join(map(str, r[i][1:i+1])) for i in range(1,n+1)]))
