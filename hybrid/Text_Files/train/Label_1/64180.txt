from sys import stdin
read = lambda: stdin.readline().strip()
write = lambda x: print(x)

for _ in range(int(read())):
    n = int(read())
    adj = [set() for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, read().split())
        adj[u].add(v)
        adj[v].add(u)
    ans = []
    vis = set()
    def dfs(u):
        vis.add(u)
        ans.append(u)
        for v in adj[u]:
            if v not in vis:
                dfs(v)
    for i in range(1, n+1):
        if i not in vis:
            dfs(i)
    ans.sort()
    write(' '.join(map(str, ans)))
