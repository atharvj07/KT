#include <iostream>
#include <cstdint>

const int64_t MOD = 1000000007;

int64_t binpow(int64_t a, int64_t b) {
    int64_t res = 1;
    while (b > 0) {
        if (b & 1) {
            res *= a;
            res %= MOD;
        }
        a *= a;
        a %= MOD;
        b >>= 1;
    }
    return res;
}

int64_t solve(int64_t A, int64_t B, int64_t n, int64_t x) {
    if (n == 0) {
        return x;
    }
    int64_t res = solve(A, B, n >> 1, (A * x + B) % MOD);
    res = (res * res) % MOD;
    if (n & 1) {
        res = (res * A) % MOD;
        res = (res + B) % MOD;
    }
    return res;
}

int main() {
    int64_t A, B, n, x;
    std::cin >> A >> B >> n >> x;
    std::cout << solve(A, B, n, x) << std::endl;
    return 0;
}
