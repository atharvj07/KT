#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll md=998244353;
const ll inf=1e18;
const ll mod=1e9+7;
ll n,c1,c2,ans;
ll a[200005];
int main()
{
    cin>>n>>c1>>c2;
    for(ll i=1;i<=n;i++)
    {
        cin>>a[i];
        if(a[i])ans+=c1;
    }
    sort(a+1,a+1+n);
    ll j=n;
    for(ll i=1;i<=n;i++)
    {
        if(a[i])break;
        ans+=c1+c2*(j-i)*(j-i);
        j--;
    }
    cout<<ans<<endl;
}
