import java.util.*;
import java.io.*;

public class Main {

    // 0-based index
    static long modAdd(long a, long b, long mod) {
        return (a + b) % mod;
    }

    static long[] initializeArray(long[] arr, int val) {
        for (int i = 0; i < arr.length; i++) {
            arr[i] = val;
        }
        return arr;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[] a = new long[n];
        long[] b = new long[n];
        String[] parts = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(parts[i]);
        }
        parts = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            b[i] = Long.parseLong(parts[i]);
        }
        long sumA = 0, sumB = 0;
        for (int i = 0; i < n; i++) {
            sumA = modAdd(sumA, a[i], 1000000007);
            sumB = modAdd(sumB, b[i], 1000000007);
        }
        if (sumA != sumB) {
            System.out.println(-1);
            return;
        }
        long[] prefSumA = new long[n], prefSumB = new long[n], suffSumA = new long[n], suffSumB = new long[n];
        prefSumA[0] = a[0];
        prefSumB[0] = b[0];
        suffSumA[n - 1] = a[n - 1];
        suffSumB[n - 1] = b[n - 1];
        for (int i = 1; i < n; i++) {
            prefSumA[i] = modAdd(prefSumA[i - 1], a[i], 1000000007);
            prefSumB[i] = modAdd(prefSumB[i - 1], b[i], 1000000007);
        }
        for (int i = n - 2; i >= 0; i--) {
            suffSumA[i] = modAdd(suffSumA[i + 1], a[i], 1000000007);
            suffSumB[i] = modAdd(suffSumB[i + 1], b[i], 1000000007);
        }
        long ans = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            long l = modAdd(prefSumA[i], suffSumA[(i + 1) % n], 1000000007);
            long r = modAdd(prefSumB[i], suffSumB[(i + 1) % n], 1000000007);
            l = modAdd(l, -b[i], 1000000007);
            ans = Math.min(ans, Math.abs(l - r));
        }
        System.out.println(ans);
    }
}
