import sys

def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    b = [a[0]]
    for i in range(1, n):
        if a[i] != a[i-1]:
            b.append(a[i])
    if len(b) == 1:
        return 0
    if k == 1:
        return len(b) - 1
    return (len(b) - 1 + k - 1) // k

t = int(input())
for i in range(t):
    print(solve())
