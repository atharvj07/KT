#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 998244353;
const int maxN = 3e5 + 5;
ll fact[maxN], inv[maxN];
ll fastpow(ll a, ll b) {
    if (b <= 0)
        return 1;
    ll res = fastpow(a, b / 2);
    res = (res * res) % mod;
    if (b % 2 == 1)
        res = (res * a) % mod;
    return res;
}
void pre() {
    fact[0] = 1, inv[0] = 1;
    for (int i = 1; i < maxN; i++) {
        fact[i] = (fact[i - 1] * i) % mod;
        inv[i] = fastpow(fact[i], mod - 2);
    }
}
ll com(int n, int k) {
    return (((fact[n] * inv[k]) % mod) * inv[n - k]) % mod;
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    pre();
    int n, k;
    cin >> n >> k;
    int st[n], en[n], cnt = 0;
    for (int i = 0; i < n; i++) {
        cin >> st[i] >> en[i];
        cnt += (st[i] <= k && k <= en[i]);
    }
    if (cnt < k)
        cout << 0;
    else
        cout << ((com(cnt, k) * fastpow(2, n - k)) % mod);
}
