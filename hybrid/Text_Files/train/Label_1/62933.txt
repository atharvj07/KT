import sys
input = sys.stdin.readline

n,m = map(int,input().split())
d = [[0]*n for i in range(n)]
for i in range(m):
    u,v,w = map(int,input().split())
    d[u-1][v-1] = w
for k in range(n):
    for i in range(n):
        for j in range(n):
            if i == j or i == k or j == k: continue
            x = min(d[i][j],d[j][k])
            d[i][j] -= x
            d[j][k] -= x
            d[i][k] += x
            d[j][i] += x
ans = 0
a = []
for i in range(n):
    if d[i][i]:
        ans += 1
        a.append([i+1,i+1,d[i][i]])
        d[i][i] = 0
for i in range(n):
    for j in range(i+1,n):
        if d[i][j]:
            ans += 1
            a.append([i+1,j+1,d[i][j]])
            d[j][i] = 0
print(ans)
for l in a:
    print(*l)
