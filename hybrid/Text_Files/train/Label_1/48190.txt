import java.util.Scanner;

public class Main {

    private static final double MAX_INTEREST = 100.0;
    private static final String LINE_SEPARATOR = "\n";

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int next;
        while((next = sc.nextInt()) > 0) {
            int years = sc.nextInt();
            Bank[] banks = new Bank[next];
            for(int i = 0; i < next; i++) {
                banks[i] = new Bank(sc.nextInt(), sc.nextInt(), sc.nextInt());
            }
            int bestBank = findBestBank(banks, years);
            System.out.println(bestBank);
        }
    }

    private static int findBestBank(Bank[] banks, int years) {
        double maxInterest = 0.0;
        int bestBank = -1;
        for(int i = 0; i < banks.length; i++) {
            double interest = calculateInterest(banks[i], years);
            if(interest > maxInterest) {
                maxInterest = interest;
                bestBank = banks[i].number;
            }
        }
        return bestBank;
    }

    private static double calculateInterest(Bank bank, int years) {
        double interest = bank.principal;
        double rate = bank.rate / MAX_INTEREST;
        for(int i = 0; i < years; i++) {
            interest += bank.interestType == 1 ? interest * rate : bank.principal * rate;
        }
        return interest;
    }

    private static class Bank {

        private int number;
        private int rate;
        private int interestType;
        private double principal = 100.0;

        public Bank(int number, int rate, int interestType) {
            this.number = number;
            this.rate = rate;
            this.interestType = interestType;
        }
    }
}
