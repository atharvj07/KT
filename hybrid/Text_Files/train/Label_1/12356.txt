#include <bits/stdc++.h>
using namespace std;
#define ll long long
bool b[305];
int main() {
  ll n, p;
  cin >> p >> n;
  for (ll i = 1; i <= n; i++) {
    ll x;
    cin >> x;
    if (b[x % p]) {
      cout << i << endl;
      return 0;
    }
    b[x % p] = true;
  }
  cout << -1 << endl;
  return 0;
}
