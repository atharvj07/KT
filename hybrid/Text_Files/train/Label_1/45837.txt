import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] s = new int[n], t = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            t[i] = in.nextInt();
        }
        Map<Integer, Integer> st = new HashMap<>(), tt = new HashMap<>();
        for (int i = 0; i < n; i++) {
            st.put(s[i], st.getOrDefault(s[i], 0) + 1);
            tt.put(t[i], tt.getOrDefault(t[i], 0) + 1);
        }
        if (!st.equals(tt)) {
            System.out.println("NO");
            return;
        }
        List<int[]> ans = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = s[i], y = t[i];
            if (x != y) {
                while (x < y) {
                    ans.add(new int[]{i + 1, i + 1, (y - x) / 2});
                    x += (y - x) / 2;
                }
                while (x > y) {
                    ans.add(new int[]{i + 1, i + 2, (x - y) / 2});
                    x -= (x - y) / 2;
                }
            }
        }
        System.out.println("YES");
        System.out.println(ans.size());
        for (int[] a : ans) {
            for (int x : a) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }
}
