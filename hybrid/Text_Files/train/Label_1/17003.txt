#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, i, j, a, b;
    cin >> n >> m >> i >> j >> a >> b;
    int dx[] = {a, a, -a, -a};
    int dy[] = {-b, b, -b, b};
    int res = 1e9;
    for (int k = 0; k < 4; ++k) {
        for (int x = min(j, a); x <= m; x += a) {
            for (int y = max(1, i - (x - j) / b * b); y <= n; y += max(1, b)) {
                if ((y - i) % b == 0 && (x - j) % a == 0) {
                    res = min(res, (y - i) / b + (x - j) / a);
                }
            }
        }
    }
    if (res == 1e9) cout << "Poor Inna and pony!" << endl;
    else cout << res << endl;
    return 0;
}
