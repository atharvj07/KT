#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    vector<pair<int, int>> diff;
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            diff.push_back({a[i] - b[i], i});
        }
    }
    int ans = 0;
    for (int i = 0; i < diff.size(); i++) {
        if (diff[i].first == -2) {
            ans += diff[i].second;
        } else if (diff[i].first == 2) {
            ans += n - 1 - diff[i].second;
        } else {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
