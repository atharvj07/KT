def solve_tree_problem(n, d, h):
    if d > 2 * h or (d == 1 and n > 2):
        print(-1)
        return
    for i in range(1, h + 1):
        print(i, i + 1)
    if d > h:
        print(1, h + 2)
        for i in range(h + 2, d + 1):
            print(i, i + 1)
        for i in range(d + 1, n):
            print(1 if h == d else h + 2, i + 1)
    else:
        for i in range(h + 1, n):
            print(2, i + 1)

n, d, h = map(int, input().split())
solve_tree_problem(n, d, h)
