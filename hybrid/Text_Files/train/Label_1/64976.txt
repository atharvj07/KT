from typing import List, Tuple

def rot(point: Tuple[int, int]):
    return -point[1], point[0]

def dot(point1: Tuple[int, int], point2: Tuple[int, int]):
    return point1[0] * point2[0] + point1[1] * point2[1]

def cross(point1: Tuple[int, int], point2: Tuple[int, int]):
    return point1[0] * point2[1] - point1[1] * point2[0]

def check(people: List[int]):
    n = len(people)
    is_ok = [[False for _ in range(n)] for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            x = people[i]
            y = people[j]
            if not x or not y:
                continue
            if ((x[2] - x[0]) * (y[2] - y[0]) >= 0 and (x[3] - x[1]) * (y[3] - y[1]) >= 0) or ((x[2] - x[0]) * (y[2] - y[0]) <= 0 and (x[3] - x[1]) * (y[3] - y[1]) <= 0):
                is_ok[i][j] = is_ok[j][i] = True
    return is_ok

for _ in range(int(input())):
    n = int(input())
    people = []
    for _ in range(n):
        x, y, u, v = map(int, input().split())
        people.append((x - u, y - v, x - u, y - v))
        people[-1] = rot(people[-1])
    is_ok = check(people)
    count = 0
    for i in range(n):
        cur = people[i]
        for j in range(n):
            if i == j:
                continue
            nxt = people[j]
            if cross(cur, nxt) > 0:
                count += is_ok[i][j]
    print(count)
