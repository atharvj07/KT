#include<bits/stdc++.h>
using namespace std;
int main()
{
    int a[4],p,q,r;
    for(int i=0;i<4;i++)
        cin>>a[i];
    p=a[0]-a[1];
    q=a[1]-a[2];
    r=a[2]-a[3];
    if(p==q && q==r)
        cout<<a[3]-r<<endl;
    else if(a[0]*a[0]==a[1]*a[2] && a[1]*a[1]==a[2]*a[3])
        cout<<a[3]*a[3]/a[2]<<endl;
    else
        cout<<42<<endl;
    return 0;
}
