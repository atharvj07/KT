import java.util.*;
import java.io.*;

class Main {
    static int n;
    static int[][] color;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        color = new int[n + 1][2];
        for(int i = 1; i <= n; ++i) {
            color[i][0] = in.nextInt();
            color[i][1] = in.nextInt();
        }
        int ans = 0;
        for(int i = 1; i <= n; ++i) {
            for(int j = 1; j <= n; ++j) {
                if(i == j) continue;
                if(color[i][0] == color[j][1])
                    ans++;
            }
        }
        System.out.println(ans);
    }
}
