from heapq import heappop, heappush

def min_jumps(n, a, b):
    dp = [float('inf')] * (n + 1)
    dp[n] = 0
    pq = [(0, n)]
    path = [-1] * (n + 1)

    while pq:
        jumps, pos = heappop(pq)
        if pos == 0:
            break
        for next_pos in range(max(0, pos - a[pos]), pos):
            slip_pos = min(n, next_pos + b[next_pos])
            if dp[next_pos] > jumps + 1:
                dp[next_pos] = jumps + 1
                path[next_pos] = pos
                heappush(pq, (jumps + 1, next_pos))

    if dp[0] == float('inf'):
        return -1, []

    sequence = []
    pos = 0
    while pos != n:
        pos = path[pos]
        sequence.append(pos)

    return dp[0], sequence[::-1]

n = int(input().strip())
a = [0] + list(map(int, input().strip().split()))
b = [0] + list(map(int, input().strip().split()))

jumps, sequence = min_jumps(n, a, b)
if jumps == -1:
    print(-1)
else:
    print(jumps)
    print(*sequence)
