#include <bits/stdc++.h>
#define mod 1000000007
using namespace std;
typedef long long ll;
const int N = 1e6 + 5;
int p[N];

int main() {
    ll n, m;
    cin >> n >> m;
    ll ans = (1ll << n) % mod;
    ans = (ans - 1 + mod) % mod;
    ans = (ans * m) % mod;
    cout << ans << "\n";
    return 0;
}
