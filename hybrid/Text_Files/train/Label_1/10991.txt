#include <iostream>
#include <complex>

using namespace std;

const int base = 1000000000;

int64_t pow10(int p) {
    int64_t res = 1;
    while (p--) res *= 10;
    return res;
}

int main() {
    int64_t n;
    cin >> n;
    if (n < 3) { cout << 0 << ' ' << 0 << endl; return 0; }
    if (n < 7) { cout << -2 << ' ' << 0 << endl; return 0; }
    int64_t c = 0, x = 0, y = 0;
    int p = 1, r = 1;
    for (int64_t i = n - 7; i > 0; i /= r * 6) {
        c += i % r*6;
        int64_t d = i / r * 6;
        if (p == 1) x += d, y -= d;
        else if (p == 2) x -= d, y -= d;
        else if (p == 3) x -= d, y += d;
        else x += d, y += d;
        r *= 6;
        p = (p + 1) % 4;
    }
    cout << x / pow10(c) % base << ' ' << y % base << endl;
}
