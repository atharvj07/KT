import sys

phy, inf, d = map(int, input().split())
p = sorted([(a, b) for a, b in (map(int, line.split()) for line in sys.stdin)][:phy], key=lambda x: x[0])
i = sorted([(a, b) for a, b in (map(int, line.split()) for line in sys.stdin)][:inf], key=lambda x: x[0])
pp = [0] * phy
ii = [0] * inf
for a, b in p:
    pp[a - 1] = b
for a, b in i:
    ii[a - 1] = b
s = 0
j = 0
for a in range(phy):
    if s + pp[a] <= d:
        s += pp[a]
    else:
        break
    j = a + 1
k = 0
for a in range(inf):
    if s + ii[a] <= d:
        s += ii[a]
    else:
        break
    k = a + 1
if j == 0 or k == 0:
    print(0)
else:
    print(sum(map(lambda x: x[0], p[:j])) + sum(map(lambda x: x[0], i[:k])))
