#include<bits/stdc++.h>
using namespace std;
int a[200005];
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        for(int i=0;i<2*n;i++) cin>>a[i];
        sort(a,a+2*n);
        int ans=1e9;
        for(int i=0;i<n;i++){
            int j=2*n-i-1;
            if(i==j) continue;
            ans=min(ans,a[j]-a[i]);
        }
        cout<<ans<<'\n';
    }
    return 0;
}
