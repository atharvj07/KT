#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < n / 2; i++) sum += a[i];
    cout << 2 * sum << '\n';
    return 0;
}
