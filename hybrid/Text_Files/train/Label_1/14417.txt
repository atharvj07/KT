#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e5;
const int inf = 1e9;
int n, q, k, a[mxN], b[mxN];
int main() {
    cin >> n >> q >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (a[j] - a[i] > 1) b[i] = b[j] = 1;
        }
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        --l, --r;
        int ans = 0;
        for (int i = l; i <= r; i++) ans += b[i];
        if (ans > 0)
            cout << ans - 1 << '\n';
        else
            cout << k - 1 << '\n';
    }
}
