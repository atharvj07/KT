#include <iostream>

using namespace std;

const long long MAXN = 1e18, DIGIT_NUM = 3;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long l, r;
        cin >> l >> r;
        int ans = 0;
        for (long long i = l; i <= r; ++i) {
            int digit_cnt = 0;
            for (long long j = i; j; j /= 10) digit_cnt += (j % 10 != 0);
            ans += (digit_cnt <= DIGIT_NUM);
        }
        cout << ans << endl;
    }
    return 0;
}
