import sys
sys.stdin = open('input.txt', 'r')

def isNearlyPrime(n):
    i = 2
    while(i * i <= n):
        if(n % i == 0):
           if(n / i > i):
               return True
        i = i + 1

    return False

t = int(input())
while(t > 0):
    n = int(input())
    
    if(n <= 5 or n % 2 != 0):
        print("NO")
    else:
        if(isNearlyPrime(n - 4) == True and n - 4 > 1):
            print("YES")
            print("2", "2", end = " ")
            n = n - 4
            
        if(n % 3 == 1):
            print("1", "1", end = " ")
            n = n - 2
        
        if(n % 3 == 0):
            print("3", end = " ")
            n = n - 3
          
        print(n)
        
    t = t - 1
