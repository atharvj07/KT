#include<bits/stdc++.h>
using namespace std;

int main()
{
	int n, y1, m, y2;
	cin >> n >> y1 >> m >> y2;
	vector<int> a(n), b(m);
	for(int &x: a) cin >> x;
	for(int &x: b) cin >> x;
	rotate(b.begin(), b.begin() + (lower_bound(b.begin(), b.end(), 1ll*y2*(y1 - y2)/(y2-y1)) - b.begin()), b.end());
	int ans = 0;
	for(int i = 0; i < n; i++)
	{
		int idx = lower_bound(b.begin(), b.end(), a[i] - 1ll*y2*(y1 - y2)/(y2-y1)) - b.begin();
		ans += idx == m || b[idx] == a[i] - 1ll*y2*(y1 - y2)/(y2-y1);
	}
	cout << ans << endl;
	return 0;
}
