from collections import Counter

# Read input
n_colors = input()
m_colors = input()

# Count each color in both sequences
n_count = Counter(n_colors)
m_count = Counter(m_colors)

# Calculate the maximum area
total_area = 0
for color in m_count:
    if color in n_count:
        total_area += min(n_count[color], m_count[color])
    else:
        total_area = -1
        break

print(total_area)
