#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> w(n);
  for (int i = 0; i < n; i++) cin >> w[i];
  vector<pair<int, int>> f(m);
  for (int i = 0; i < m; i++) cin >> f[i].first >> f[i].second;
  int idx = -1;
  for (int i = 0; i < m; i++) {
    if (w[f[i].first - 1] && w[f[i].second - 1]) {
      idx = i;
      break;
    }
  }
  if (idx != -1) {
    vector<int> ans;
    w[f[idx].first - 1]--;
    w[f[idx].second - 1]--;
    ans.push_back(idx + 1);
    for (int i = 0; i < m; i++) {
      if (w[f[i].first - 1] && w[f[i].second - 1]) {
        w[f[i].first - 1]--;
        w[f[i].second - 1]--;
        ans.push_back(i + 1);
      }
    }
    bool ok = true;
    for (int i = 0; i < n; i++) {
      ok &= (w[i] == 0);
    }
    if (ok) {
      cout << "ALIVE" << endl;
      for (int x : ans) cout << x << " ";
      cout << endl;
    } else {
      cout << "DEAD" << endl;
    }
  } else {
    cout << "DEAD" << endl;
  }
  return 0;
}
