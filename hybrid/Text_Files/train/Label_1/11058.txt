#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] >= 'a' && s[i] <= 'z') {
      s[i] = (s[i] - 'a' + k) % 26 + 'a';
    } else {
      s[i] = (s[i] - 'A' + k) % 26 + 'A';
    }
  }
  cout << s << endl;
  return 0;
}
