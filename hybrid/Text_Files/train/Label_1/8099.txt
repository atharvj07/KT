#include<bits/stdc++.h>

using namespace std;

int main(){

    int a[4];
    int d;
    int g;

    cin >> a[0] >> a[1] >> a[2] >> a[3];

    d = a[1] - a[0];
    g = a[1] / a[0];

    if((a[2]-a[1]) == d && (a[2]-a[1]) % d == 0 && (a[3]-a[2]) == d && (a[3]-a[2]) % d == 0){
        cout << a[3] + d << endl;
        return 0;
    }

    if((a[2]/a[1]) == g && (a[2]/a[1]) % g == 0 && (a[3]/a[2]) == g && (a[3]/a[2]) % g == 0){
        cout << a[3] * g << endl;
        return 0;
    }

    cout << "42" << endl;

    return 0;
}
