#include <iostream>

using namespace std;

int main() {
    char direction;
    cin >> direction;
    string text;
    cin >> text;
    for (int i = 0; i < text.size(); i++) {
        if (direction == 'L') {
            if (text[i] == 'q' || text[i] == 'a' || text[i] == 'z') {
                text[i]++;
            } else {
                text[i] -= 2;
            }
        } else if (direction == 'R') {
            if (text[i] == 'p' || text[i] == 'l' || text[i] == 'm') {
                text[i]--;
            } else {
                text[i] += 2;
            }
        }
    }
    cout << text << endl;
    return 0;
}
