#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
#define pb push_back
ll n,ans,a[2010],b[2010],p[2010];
vector<pair<ll,ll> >op;
int main()
{
    cin>>n;
    for(ll i=1;i<=n;i++)
    {
        cin>>a[i];
        p[a[i]]=i;
    }
    for(ll i=1;i<=n;i++)
    {
        cin>>b[i];
        if(a[i]!=b[i])
        {
            op.pb({p[b[i]],i});
            swap(a[p[b[i]]],a[i]);
            p[a[i]]=i;
            ans++;
        }
    }
    cout<<ans<<"\n"<<op.size()<<"\n";
    for(auto x:op)
        cout<<x.first<<" "<<x.second<<"\n";
    return 0;
}
