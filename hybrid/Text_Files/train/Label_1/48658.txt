import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] l = new long[n], r = new long[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        Arrays.sort(l);
        Arrays.sort(r);
        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (r[i] < l[i + 1]) {
                ans += r[i] - l[i + 1];
            }
        }
        System.out.println(ans);
    }
}
