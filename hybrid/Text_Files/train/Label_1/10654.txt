#include<bits/stdc++.h>
using namespace std;
int q,n,a[105];
int main(){
	cin>>q;
	while(q--){
		cin>>n;
		for(int i=0;i<n;i++)cin>>a[i];
		for(int i=0;i<n;i++){
			int pos=i;
			for(int j=i+1;j<n;j++)if(a[j]<a[pos])pos=j;
			if(pos!=i)a[i]^=a[pos]^=a[i]^=a[pos];
		}
		for(int i=0;i<n;i++)cout<<a[i]<<" ";
		cout<<endl;
	}
}
