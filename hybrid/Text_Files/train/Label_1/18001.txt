#include <bits/stdc++.h>

using namespace std;
bool vis[5005][367];
int dp[5005][367];
int n, a[5005], b[5005], g[750];
int f(int pos, int day) {
  if (pos == n + 1) return 0;
  if (vis[pos][day] == 1) return dp[pos][day];
  vis[pos][day] = 1;
  int ans = f(pos + 1, day);
  if (day >= a[pos] && day <= b[pos]) ans = max(ans, 1 + f(pos + 1, day + 1));
  return dp[pos][day] = ans;
}
int main() {
  cin >> n;
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    char c;
    cin >> c >> a[i] >> b[i];
    if (c == 'F') g[++cnt] = a[i];
    if (c == 'M') g[++cnt] = b[i];
  }
  sort(g + 1, g + 1 + cnt);
  int ans = 0;
  for (int i = 1; i <= cnt; i++) {
    memset(vis, false, sizeof vis);
    ans = max(ans, f(1, g[i]));
  }
  cout << ans << endl;
  return 0;
}
