import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    if (isPrime(m) && isNextPrime(n, m)) {
      System.out.println("YES");
    } else {
      System.out.println("NO");
    }
  }

  private static boolean isPrime(int n) {
    if (n == 1) {
      return false;
    }
    for (int i = 2; i * i <= n; i++) {
      if (n % i == 0) {
        return false;
      }
    }
    return true;
  }

  private static boolean isNextPrime(int n, int m) {
    if (n + 1 == m && isPrime(m)) {
      return true;
    }
    return false;
  }
}
