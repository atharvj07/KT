import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int p = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[k];
        for (int i = 0; i < k; i++) {
            b[i] = sc.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        int j = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            while (j < k && b[j] < a[i]) {
                j++;
            }
            if (j < k) {
                ans = Math.max(ans, Math.abs(p - b[j]));
                j++;
            } else {
                ans = Math.max(ans, Math.abs(p - a[i]));
            }
        }
        System.out.println(ans);
    }
}
