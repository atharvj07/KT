#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int M = 1e9 + 7;
const int hi = 1e5 + 1;
int f[hi];
int fun(int sum){
    int c=0,t=0;
    while(sum){
        t=(sum%10)+t;
        sum/=10;
        c++;
        if(c==2){
            if(t>=10) f[c-2]+=1;
            t/=10;
            sum+=t;
            t=0;
            c=0;
        }
    }
}
void solve(){
    int n,q; cin>>n; string s; cin>>s; cin>>q;
    for(int i=0;i<n/2;i+=2){ fun(s[i]-'0'+s[i+1]-'0'); }
    for(int i=0;i<hi;i++){ f[i+1]+=f[i]; }
    while(q--){
        int l,r; cin>>l>>r;
        cout<<f[r/2]-f[l/2-1]<<endl;
    }
}
int main(){ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);solve();return 0;}
