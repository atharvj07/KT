import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] d = new int[n];
    for (int i = 0; i < n; i++) {
      d[i] = sc.nextInt();
    }
    boolean[][] edge = new boolean[n][n];
    for (int i = 0; i < m; i++) {
      int a = sc.nextInt() - 1;
      int b = sc.nextInt() - 1;
      edge[a][b] = true;
      edge[b][a] = true;
    }
    int[] deg = new int[n];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        if (edge[i][j]) {
          deg[i]++;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      if (d[i] == 1 && deg[i] % 2 == 0) {
        System.out.println(-1);
        return;
      }
      if (d[i] == -1 && deg[i] % 2 == 1) {
        System.out.println(-1);
        return;
      }
    }
    System.out.println(m);
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (edge[i][j] && deg[i] % 2 == 0 && deg[j] % 2 == 0) {
          System.out.println(i + 1 + " " + (j + 1));
          edge[i][j] = false;
          edge[j][i] = false;
          deg[i]--;
          deg[j]--;
        }
      }
    }
  }
}
