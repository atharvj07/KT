import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    char[] s = br.readLine().toCharArray();
    char[] t = br.readLine().toCharArray();

    int[] diffs = new int[n];
    for (int i = 0; i < n; i++) {
      if (s[i] != t[i]) {
        ++diffs[i];
        ++diffs[(i + 1) % n];
      }
    }
    int numDiffs = 0;
    int i1 = -1;
    int i2 = -1;
    for (int i = 0; i < n; i++) {
      if (diffs[i] > 0) {
        ++numDiffs;
        if (i1 == -1) {
          i1 = i;
        } else {
          i2 = i;
        }
      }
    }
    if (numDiffs % 2 == 1 || n % 2 == 1 && numDiffs > 0) {
      System.out.println(-1);
    } else {
      System.out.println(numDiffs / 2);
      for (int i = 0; i < numDiffs / 2; i++) {
        System.out.println((i1 + 1) + " " + (i2 + 1));
        i1 = i1 + 1;
        i2 = i2 - 1;
        if (i1 == n) {
          i1 = 0;
        }
        if (i2 == -1) {
          i2 = n - 1;
        }
      }
    }
  }
}
