#include <bits/stdc++.h>

using namespace std;

const int maxn = 110;

int n, m, k[maxn], pos[maxn], pre[ maxn * maxn ];

int main() {
    cin >> n >> m;
    for( int i = 1; i <= n; i++ ) {
        int x; cin >> x;
        pos[x] = i;
    }

    int cnt = 0;
    for( int i = 1; i <= m; i++ ) {
        cin >> k[i];
        cnt += k[i];
        pos[i + m] = pos[i] + cnt;
    }
    int l = 1, r = cnt;

    while( l <= r ) {
        int mid = (l + r) >> 1;
        bool ok = 1;
        for( int i = 1; i <= m; i++ ) {
            if( pre[pos[i] + mid - 1] - pre[pos[i] - 1] != k[i] )
                ok = 0;
        }
        if( ok ) r = mid - 1;
        else l = mid + 1;
    }

    if( l <= cnt ) puts("YES");
    else puts("NO");

    return 0;
}
