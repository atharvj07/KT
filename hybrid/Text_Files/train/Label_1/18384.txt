#include <bits/stdc++.h>
#define MAXM 505
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
bool pn[200005];
void sieve() {
  for (ll i = 2; i * i <= 200000; i++) {
    if (pn[i] == 0) {
      for (ll j = i * i; j <= 200000; j += i) { pn[j] = 1; }
    }
  }
}
int n, m;
int a[MAXM][MAXM];
int main() {
  sieve();
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  int ans = INT_MAX;
  for (int i = 1; i <= n; i++) {
    int sum = 0;
    for (int j = 1; j <= m; j++) { sum += pn[a[i][j]]; }
    if (sum == 0) { ans = 0; }
    ans = min(ans, m - sum);
  }
  for (int j = 1; j <= m; j++) {
    int sum = 0;
    for (int i = 1; i <= n; i++) { sum += pn[a[i][j]]; }
    if (sum == 0) { ans = 0; }
    ans = min(ans, n - sum);
  }
  cout << ans << endl;
  return 0;
}
