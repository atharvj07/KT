#include <bits/stdc++.h>
#define int long long
using namespace std;

vector<int> a, b;

int32_t main() {
  int n, m;
  cin >> n;
  a.assign(n, 0);
  b.assign(n, 0);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  cin >> m;
  int x = 1, y = 1;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (y <= b[i]) {
      ans += x * (n - i);
      ans %= m;
    }
    if (x <= a[i]) {
      ans += y * (n - i);
      ans %= m;
    }
    if (x < a[i]) x = a[i];
    if (y < b[i]) y = b[i];
  }
  cout << ans << '\n';
  return 0;
}
