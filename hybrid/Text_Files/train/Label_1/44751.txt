import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            int countA = 0, countB = 0;
            for (char c : s.toCharArray()) {
                if (c == 'A') countA++;
                else countB++;
            }
            System.out.println(Math.abs(countA - countB));
        }
    }
}
