#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
ll ans, mod = 1e9 + 7;
ll bigmod(ll a, ll p) {
    if (p == 0) return 1;
    if (p == 1) return a;
    if (p % 2 == 1) return (a * bigmod(a, p - 1)) % mod;
    ll ret = bigmod(a, p / 2) % mod;
    return (ret * ret) % mod;
}
int main() {
    ll n, m;
    cin >> n >> m;
    ans = (bigmod(2, n - 1) * bigmod(2, m - 1) % mod) % mod;
    cout << ans << endl;
    return 0;
}
