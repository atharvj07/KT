import java.io.*;
import java.util.*;

public class Main {
  static int[] a;
  static boolean[][] b, c;
  static ArrayList<Pair> ans = new ArrayList<>();
  static void swap(int i, int j) {
    int tmp = a[i];
    a[i] = a[j];
    a[j] = tmp;
    b[i][j] = b[j][i] = false;
    c[i][j] = c[j][i] = false;
  }
  static boolean dfs(int l, int r, boolean[] u, boolean reverse) {
    if (l > r) return true;
    if (u[l] && u[r]) return false;
    if (u[l]) return dfs(l + 1, r, u, reverse);
    if (u[r]) return dfs(l, r - 1, u, reverse);
    if (!b[l][r] && !c[l][r]) {
      if (reverse) {
        u[l] = u[r] = true;
      }
      return dfs(l + 1, r - 1, u, reverse);
    }
    if (b[l][r]) {
      for (int i = l; i <= r; i++) {
        if (a[i] > a[r]) {
          ans.add(new Pair(i, r));
          swap(i, r);
          u[r] = true;
          if (!dfs(l, r - 1, u, reverse)) {
            u[r] = false;
            swap(i, r);
            ans.remove(ans.size() - 1);
          } else {
            return true;
          }
        }
      }
    } else {
      for (int i = r; i >= l; i--) {
        if (a[i] < a[l]) {
          ans.add(new Pair(l, i));
          swap(l, i);
          u[l] = true;
          if (!dfs(l + 1, r, u, reverse)) {
            u[l] = false;
            swap(l, i);
            ans.remove(ans.size() - 1);
          } else {
            return true;
          }
        }
      }
    }
    return false;
  }
  public static void main(String[] args) throws IOException {
    BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(input.readLine());
    a = new int[n + 1];
    StringTokenizer st = new StringTokenizer(input.readLine());
    for (int i = 1; i <= n; i++) {
      a[i] = Integer.parseInt(st.nextToken());
    }
    b = new boolean[n + 1][n + 1];
    c = new boolean[n + 1][n + 1];
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (a[i] < a[j]) {
          c[i][j] = true;
        } else {
          b[i][j] = true;
        }
      }
    }
    if (!dfs(1, n, new boolean[n + 1], false)) {
      System.out.println(-1);
    } else {
      System.out.println(ans.size());
      for (Pair x: ans) {
        System.out.println(x.a + " " + x.b);
      }
    }
  }
  static class Pair {
    int a, b;
    Pair(int a, int b) { this.a = a; this.b = b; }
  }
}
