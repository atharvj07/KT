def min_fatigue(a, n, m, rain_segments, umbrellas):
    INF = float('inf')
    dp = [[INF] * (m + 2) for _ in range(a + 1)]
    rain = [0] * (a + 1)
    for l, r in rain_segments:
        for i in range(l, r):
            rain[i] = 1

    umbrellas.sort()
    umbrellas = [(0, 0)] + umbrellas + [(a, 0)]

    for i in range(m + 2):
        if umbrellas[i][0] == 0:
            dp[0][i] = 0

    for i in range(1, a + 1):
        for j in range(m + 2):
            if dp[i - 1][j] < INF:
                dp[i][j] += dp[i - 1][j]
                if rain[i - 1]:
                    dp[i][j] += umbrellas[j][1]
            if j > 0 and umbrellas[j][0] == i:
                for k in range(j):
                    dp[i][j] = min(dp[i][j], dp[i][k])

    answer = INF
    for i in range(m + 2):
        answer = min(answer, dp[a][i])

    return -1 if answer == INF else answer

# Read input
a, n, m = map(int, input().split())
rain_segments = [tuple(map(int, input().split())) for _ in range(n)]
umbrellas = [tuple(map(int, input().split())) for _ in range(m)]

# Solve and print the answer
print(min_fatigue(a, n, m, rain_segments, umbrellas))
