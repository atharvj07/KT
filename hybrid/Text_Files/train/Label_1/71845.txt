import math

t = int(input())
def calc(a):
    n = len(a)
    for m in range(21):
        s = a[0] % (2**m)
        c = (a[1] - s) % (2**m)
        if (a[2:] == [((x+c)%2**m) for x in a[:-2]]):
            return 2**m
    return 0

for i in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    res = calc(a)
    print(res if res else -1)
