#include <bits/stdc++.h>

using namespace std;

int main()
{
	int x, y, ans;
	cin >> x >> y;
	ans = 0;
	if (x == 0 && y == 0) cout << 0 << endl;
	else
	{
		if (x > 0 && y >= 0) ans = y;
		if (x <= 0 && y > 0) ans = y + x;
		if (x < 0 && y <= 0) ans = -x + y;
		if (x >= 0 && y < 0) ans = -y + 2 * x;
		cout << ans << endl;
	}

	return 0;
}
