#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  vector<int> cnt(k);
  for (int i = 0; i < n; ++i) { cin >> a[i]; cnt[a[i] - 1]++; }
  int ans = 0;
  for (int i = 0; i < k; ++i) { ans += min(n / 2, cnt[i]); }
  if (n % 2 == 1) ans += min(1, cnt[a[n / 2]]);
  cout << ans << endl;
  return 0;
}
