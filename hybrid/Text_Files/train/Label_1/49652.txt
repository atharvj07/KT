import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    int ans = -1;
    for (int i = 0; i < n; i++) {
      int cur = a[i];
      for (int j = 0; j < i; j++) {
        cur -= a[j];
      }
      if (cur <= 0) {
        ans = i + 1;
        break;
      }
    }
    System.out.println(ans);
  }
}
