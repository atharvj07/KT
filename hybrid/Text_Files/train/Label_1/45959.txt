import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        // Read input
        Scanner in = new Scanner(System.in);
        int x = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();

        // Compute distance to each store
        int distA = Math.abs(x - a);
        int distB = Math.abs(x - b);

        // Print result
        System.out.println(distA < distB ? "A" : "B");
    }
}
