#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define vb vector<bool>
void solve() {
  ll n, k, ans = 0;
  cin >> n >> k;
  vi a(n), b(n), c(n);
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] == i + 1) {
      ans++, b[i] = 1, c[i] = 1;
    }
  }
  if (ans >= k) {
    cout << 0 << endl;
    return;
  }
  ll idx = -1, cnt = 0;
  for (ll i = 0; i < n; i++) {
    if (c[i] != 1 && cnt < k - ans) {
      if (b[i] == 0) b[i] = 1, cnt++, idx = i;
      if (i + 1 < n && b[i + 1] == 0) b[i + 1] = 1, cnt++, idx = i;
    }
  }
  if (k - ans > cnt)
    cout << "-1\n";
  else
    cout << n - cnt << endl;
}
int32_t main() {
  ll t;
  cin >> t;
  while (t--) solve();
}
