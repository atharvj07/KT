def solve():
    n = int(input())
    last_winner = int(input())
    spectator = [0, 0, 0]
    for i in range(1, n):
        winner = int(input())
        if winner == last_winner:
            return "NO"
        if winner == spectator[last_winner]:
            return "NO"
        spectator[last_winner] = winner
        last_winner = winner
    return "YES"

print(solve())
