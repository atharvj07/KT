#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, t, x, y;
  cin >> n;
  int a = 0, b = 0;
  for (int i = 0; i < n; i++) {
    cin >> t >> x >> y;
    if (t == 1) a += x;
    else b += x;
  }
  cout << (a >= n / 2 * 10 ? "LIVE" : "DEAD") << endl;
  cout << (b >= n / 2 * 10 ? "LIVE" : "DEAD") << endl;
}
