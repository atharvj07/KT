import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
            pos[i] = i;
        }
        sort(a, pos);
        for (int i = 0; i < n - 1; i++) {
            if (gcd(a[i], a[i + 1]) == 1) {
                swap(a, pos, i, i + 1);
            }
        }
        for (long e : a) {
            System.out.print(e + " ");
        }
        System.out.println();
    }

    static void sort(long[] a, int[] pos) {
        int n = a.length;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j] || a[i] == a[j] && pos[i] > pos[j]) {
                    long t = a[i];
                    a[i] = a[j];
                    a[j] = t;
                    int p = pos[i];
                    pos[i] = pos[j];
                    pos[j] = p;
                }
            }
        }
    }

    static void swap(long[] a, int[] pos, int i, int j) {
        long t = a[i];
        a[i] = a[j];
        a[j] = t;
        int p = pos[i];
        pos[i] = pos[j];
        pos[j] = p;
    }

    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
