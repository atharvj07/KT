#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0, j = n - 1; i < j;) {
        if (a[i] + a[j] < a[i + 1] + a[j - 1]) {
            j--;
        } else {
            i++;
            j--;
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
