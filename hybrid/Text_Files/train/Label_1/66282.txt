import sys

n, k = map(int, input().split())
a = list(map(int, input().split()))
freq = [0] * 1000001
l, r = 0, 0
max_len, max_l, max_r = 0, 0, 0
for i in range(n):
    freq[a[i]] += 1
    while freq[a[i]] > k:
        freq[a[l]] -= 1
        l += 1
    if i - l + 1 > max_len:
        max_len = i - l + 1
        max_l = l
        max_r = i
print(max_l + 1, max_r + 1)
