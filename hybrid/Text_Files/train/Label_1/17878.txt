#include <bits/stdc++.h>

using namespace std;
const int N = 55;
bool vis[N];
int arr[N];

int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    memset(vis, 0, sizeof vis);
    int n;
    cin >> n;
    for (int i = 1; i <= 2 * n; i++) { cin >> arr[i]; }
    vector<int> res;
    for (int i = 1; i <= 2 * n; i++) {
      if (!vis[arr[i]]) {
        for (int j = i + 1; j <= 2 * n; j++) {
          if (arr[i] == arr[j]) {
            res.push_back(arr[i]);
            vis[arr[i]] = 1;
            break;
          }
        }
      }
    }
    for (int i = 0; i < n; i++) { cout << res[i] << " "; }
    cout << endl;
  }
}
