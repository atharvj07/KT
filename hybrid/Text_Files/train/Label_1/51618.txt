import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int y = in.nextInt();
        int k = in.nextInt();
        int n = in.nextInt();
        int x = n - y;
        if (x >= 1 && x % k == 0) {
            while (x % k == 0) {
                System.out.print(x + " ");
                x -= k;
            }
        } else {
            System.out.println(-1);
        }
    }
}
