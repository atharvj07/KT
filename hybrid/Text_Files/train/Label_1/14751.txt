#include <bits/stdc++.h>
#define int long long
#define vi vector<int>
#define vii vector<vi>
using namespace std;

int n, k;
char c[2010][2010];
string s;

int32_t main() {
    cin >> n >> k;
    for(int i = 0; i < n; i++) {
        cin >> s;
        for(int j = 0; j < n; j++)
            c[i][j] = s[j];
    }
    int ans = 0;
    for(int i = 0; i < n; i++) {
        bool fl = 1;
        for(int j = 0; j < k; j++)
            if(c[i][j] == 'B') fl = 0;
        if(fl) ans++;
    }
    for(int j = 0; j < n; j++) {
        bool fl = 1;
        for(int i = 0; i < k; i++)
            if(c[i][j] == 'B') fl = 0;
        if(fl) ans++;
    }
    cout << ans << "\n";
    return 0;
}
