import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] arr = new int[n];
		for(int i=0; i<n; i++){
			arr[i] = sc.nextInt();
		}
		int[] cnt = new int[n];
		for(int i=0; i<n; i++){
			cnt[arr[i] -1] = i;
		}
		int ans = 0;
		int idx = 0;
		for(int i=0; i<n; i++){
			ans += Math.abs(cnt[i] - idx);
			idx = cnt[i];
		}
		System.out.println(ans);
	}
}
