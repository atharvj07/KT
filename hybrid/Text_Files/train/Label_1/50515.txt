import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[6];
        int i;
        for (i = 0; i < 6; i++) a[i] = in.nextInt();
        int n = in.nextInt();
        int[] b = new int[n];
        for (i = 0; i < n; i++) b[i] = in.nextInt();
        Arrays.sort(a);
        Arrays.sort(b);
        int max = b[n - 1];
        int min = b[0];
        for (i = 0; i < 6; i++) {
            if (min <= a[i]) {
                min = a[i];
                break;
            }
        }
        System.out.println(max - min);
    }
}
