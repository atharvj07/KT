import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int ans = 0;
        int[] prices = {10000, 5000, 1000, 500};
        for (int price : prices) {
            while (p >= price) {
                ans += price;
                p -= price;
            }
        }
        System.out.println(ans);
    }
}
