n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

left, right = 0, 2000

while left < right:
    mid = (left + right + 1) // 2
    need = sum(max(0, mid * a[i] - b[i]) for i in range(n))
    if need <= k:
        left = mid
    else:
        right = mid - 1

print(left)
