#include <iostream>
#include <iomanip>
#include <cmath>

using namespace std;

int main() {
    double x1, y1, r1, x2, y2, r2;
    cin >> x1 >> y1 >> r1 >> x2 >> y2 >> r2;
    double d = sqrt((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1));
    if (d > r1 + r2) {
        cout << "0.00000000000000000000" << endl;
    } else if (d <= fabs(r1 - r2)) {
        cout << min(r1, r2) * min(r1, r2) * M_PI << endl;
    } else {
        double h = ((r1 * r1 - r2 * r2) + (d * d)) / (2 * d);
        double s1 = sqrt(r1 * r1 - h * h);
        double s2 = sqrt(r2 * r2 - h * h);
        double A = s1 * s2;
        double a = asin(s2 / r1);
        double b = asin(s1 / r2);
        A += r1 * r1 * a - r1 * s1 * h;
        A += r2 * r2 * b - r2 * s2 * h;
        cout << fixed << setprecision(24) << A << endl;
    }
    return 0;
}
