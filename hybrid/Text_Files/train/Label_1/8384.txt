#include <iostream>

using namespace std;

int main() {
  string s;
  getline(cin, s);
  int x, ans = 0;
  bool isWeek = false;
  if (s.find("week") != -1) {
    isWeek = true;
  }
  x = stoi(s.substr(0, s.find(" ")));
  for (int i = 1; i <= 12; i++) {
    int y = 0;
    if (i == 4 || i == 6 || i == 9 || i == 11) y = 30;
    else if (i == 2) y = 29;
    else y = 31;
    int z = 0;
    for (int j = 1; j <= y; j++) {
      int day = isWeek ? ((j + x - 1) % 7) + 1 : j;
      if (day == x) z++;
    }
    ans += z;
  }
  cout << ans << endl;
  return 0;
}
