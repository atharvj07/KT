import java.util.*;
import java.io.*;

public class Main {
    static BufferedReader br;
    static StringTokenizer st;
    static int[][] a, b;
    static int na, ma, nb, mb;
    static long ans, ansx, ansy;

    public static void main(String[] args) throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        st = new StringTokenizer(br.readLine());
        na = Integer.parseInt(st.nextToken());
        ma = Integer.parseInt(st.nextToken());
        a = new int[na][ma];
        for (int i = 0; i < na; i++) {
            String s = br.readLine();
            for (int j = 0; j < ma; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        st = new StringTokenizer(br.readLine());
        nb = Integer.parseInt(st.nextToken());
        mb = Integer.parseInt(st.nextToken());
        b = new int[nb][mb];
        for (int i = 0; i < nb; i++) {
            String s = br.readLine();
            for (int j = 0; j < mb; j++) {
                b[i][j] = s.charAt(j) - '0';
            }
        }
        for (int x = -nb + 1; x <= na - 1; x++) {
            for (int y = -mb + 1; y <= ma - 1; y++) {
                long cur = 0;
                for (int i = Math.max(1, -x + 1); i <= Math.min(na, nb - x); i++) {
                    for (int j = Math.max(1, -y + 1); j <= Math.min(ma, mb - y); j++) {
                        cur += a[i - 1][j - 1] * b[i + x - 1][j + y - 1];
                    }
                }
                if (cur > ans) {
                    ans = cur;
                    ansx = x;
                    ansy = y;
                }
            }
        }
        System.out.println(ansx + " " + ansy);
    }
}
