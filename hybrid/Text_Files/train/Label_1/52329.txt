import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int k = cin.nextInt();
        for (int i = 0; i < n; i++) {
            int m = cin.nextInt();
            if (m > k) {
                System.out.println(-1);
            } else {
                int x = (k + 1) / 2;
                int y = x - m / 2;
                System.out.println(x + " " + y + " " + (y + m - 1));
            }
        }
    }
}
