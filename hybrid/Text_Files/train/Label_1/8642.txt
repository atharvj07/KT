#include<cstdio>
using namespace std;
struct ele{
 int ch;
 char dr;
}e[100];
int main(){
 int n,q,l,r,sz=-1;
 char s[105];
 scanf("%d %d %s",&n,&q,s);
 for(int i=0;i<n;i++) e[++sz].dr=(s[i]=='<')?0:1,e[sz].ch=s[i]-'0';
 for(int i=0;i<q;i++){
  int ans[11]={},now=0,z=0;
  scanf("%d %d",&l,&r);
  while(now<l-1) now++,z=e[now].ch;
  while(now<=r){
   if(e[now].dr) now--;
   else{
    if(z) z--,ans[z]++;
    else e[now].ch=0;
    now++;
   }
  }
  for(int j=0;j<10;j++) printf("%d ",ans[j]);
  puts("");
 }
 return 0;
}
