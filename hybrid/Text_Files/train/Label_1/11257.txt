#include<bits/stdc++.h>
using namespace std;

const int MAX_N = 2e5 + 5;

int n, m;
int a[MAX_N], b[MAX_N];

int main() {
    string s, t;
    cin >> s >> t;
    n = s.size(), m = t.size();
    int l = 0, r = n - 1;
    while(l < n && s[l] == t[l]) ++l;
    while(r >= 0 && s[r] == t[r]) --r;
    cout << n + m - 2 * l << '\n';
    return 0;
}
