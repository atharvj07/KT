#include<bits/stdc++.h>
using namespace std;
#define ll long long

ll const mod=1e9+7;

int main() {
    int n; cin>>n;
    map<int,int>x,y;
    while(n--){
        int a,b; cin>>a>>b;
        x[a]++;
        y[b]++;
    }
    ll ans=1;
    for(auto i:x){
        ans=(ans*i.second)%mod;
    }
    for(auto i:y){
        ans=(ans*i.second)%mod;
    }
    cout<<ans<<endl;
}
