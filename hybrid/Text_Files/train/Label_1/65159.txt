def solve(n: int, rects: list[list[int]]) -> str:
    xmin, ymin, xmax, ymax = 2**31, 2**31, 0, 0
    for x1, y1, x2, y2 in rects:
        xmin = min(xmin, x1)
        xmax = max(xmax, x2)
        ymin = min(ymin, y1)
        ymax = max(ymax, y2)
    length = max(xmax - xmin, ymax - ymin)
    return 'YES' if all(x2 - x1 == length and y2 - y1 == length for x1, y1, x2, y2 in rects) else 'NO'

n = int(input())
rects = [list(map(int, input().split())) for _ in range(n)]
print(solve(n, rects))
