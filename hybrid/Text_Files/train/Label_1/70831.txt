import sys
def solve():
    # Read input
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = [int(x) for x in input().split()]

        # Apply the shuffling process
        vis = [False for _ in range(n)]
        for i in range(n):
            k = (i + a[i % n] + n) % n
            if vis[k]:
                print("NO")
                return
            vis[k] = True

        # Check if there is one guest assigned to each room
        print("YES" if all(vis) else "NO")

solve()
