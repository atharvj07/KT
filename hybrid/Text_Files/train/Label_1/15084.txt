#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<int(n);++i)
#define ll long long
using namespace std;

int main() {
    int t; cin >> t;
    while(t--) {
        int n, k1, k2; cin >> n >> k1 >> k2;
        vector<int> a(k1);
        vector<int> b(k2);
        rep(i,k1) cin >> a[i];
        rep(i,k2) cin >> b[i];
        int k = 0;
        int l = 0;
        sort(a.begin(),a.end());
        sort(b.begin(),b.end());
        int ans = 0;
        rep(i,k1) {
            if(k==k2) break;
            if(a[i]>b[k]) {
                ans++;
                k++;
            }
            else {
                l++;
            }
        }

        if(ans>l) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
