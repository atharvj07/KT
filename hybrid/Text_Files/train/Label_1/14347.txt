#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl "\n"

struct DSU {
    vector<int> parent, sz;
    DSU(int n) : parent(n), sz(n, 1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) { return parent[x] == x ? x : parent[x] = find(parent[x]); }
    bool unite(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return false;
        if (sz[x] > sz[y]) swap(x, y);
        parent[x] = parent[y], sz[y] += sz[x];
        return true;
    }
};

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n;
        DSU dsu(2 * n + 1);
        vector<vector<pair<int, ll>>> adj(2 * n + 1);
        for (int i = 0; i < 2 * n - 1; i++) {
            int a, b, w;
            cin >> a >> b >> w;
             adj[a].emplace_back(b, w);
            adj[b].emplace_back(a, w);
        }
        vector<tuple<ll, int, int>> edges;
        for (int i = 1; i <= 2 * n; i++) {
            for (auto& [j, w] : adj[i]) {
                if (dsu.find(i) != dsu.find(j)) {
                    edges.emplace_back(w, i, j);
                }
            }
        }
        sort(begin(edges), end(edges));
        ll g = 0, b = 0, sum = 0;
        for (auto& [w, a, b] : edges) {
            if (dsu.unite(a, b)) {
                sum += w;
                g += 2 * w;
                b += w;
            }
        }
        cout << sum - g << " " << sum + b << endl;
    }
}
