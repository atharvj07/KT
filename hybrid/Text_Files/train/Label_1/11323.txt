#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 4 == 0) {
      cout << "YES" << endl;
      for (int i = 0; i < n / 2; i++) {
        cout << 4 * i + 2 << " ";
      }
      for (int i = 0; i < n / 2; i++) {
        cout << 4 * i + 3 << " ";
      }
      cout << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
