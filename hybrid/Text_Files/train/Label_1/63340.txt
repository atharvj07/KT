import sys
input = sys.stdin.readline

mod = int(1e9) + 7
n = int(input())
x = {}
y = {}
for _ in range(n):
    a, b = map(int, input().split())
    if a not in x: x[a] = 0
    if b not in y: y[b] = 0
    x[a] += 1
    y[b] += 1
ans = 1
for i in x.values(): ans = (ans * i) % mod
for i in y.values(): ans = (ans * i) % mod
print(ans)
