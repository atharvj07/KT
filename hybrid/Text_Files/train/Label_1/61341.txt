chatlist=[]
counttalk=[]
n=int(input())
for i in range(n):
    chat=input()
    if chat in chatlist:
        ind=chatlist.index(chat)
        chatlist.pop(ind)
        counttalk[ind]-=1
    chatlist.append(chat)
    counttalk.append(1)
chatlist.sort(reverse=True)
counttalk.sort(reverse=True)
for chat, count in zip(chatlist, counttalk):
    for j in range(count): print(chat)
