#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), vis(n + 2, 0);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> p(n + 2);
    for (int i = 0; i < n; i++)
      if (!a[i]) p[i + 1] = n + 1, vis[n + 1] = 1;
    for (int i = n - 1; i >= 0; i--) {
      if (a[i]) p[p[n + 1]] = i + 1, vis[i + 1] = 1;
    }
    if (vis[1] == 0)
      for (int i = n; i >= 1; i--) {
        if (!vis[i]) {
          p[p[n + 1]] = i;
          vis[i] = 1;
          break;
        }
      }
    if (!vis[1] || !p[1]) {
      cout << -1 << endl;
      continue;
    }
    int cur = 1;
    vector<int> ans;
    while (cur != n + 1) {
      ans.push_back(cur);
      cur = p[cur];
    }
    ans.push_back(cur);
    for (int x : ans) cout << x << " ";
    cout << endl;
  }
}
