#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<vector<int> > a(n, vector<int> (n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      a[i][j] = 0;
    }
  }
  int l = 0, r = n - 1, up = 0, dn = n - 1;
  while (k > 0) {
    while (a[up][l] == 1 && l < r) l++;
    while (a[up][r] == 1 && l < r) r--;
    if (l >= r) {
      l = 0;
      up++;
    }
    while (a[dn][l] == 1 && l < r) l++;
    while (a[dn][r] == 1 && l < r) r--;
    if (l >= r) {
      l = 0;
      dn--;
    }
    if (up > dn || l > r) {
      cout << -1 << endl;
      return 0;
    }
    if (up == dn && l == r) {
      a[up][l] = 1;
      k--;
    } else {
      a[up][l] = a[dn][r] = 1;
      k -= 2;
    }
    l++;
    r--;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cout << a[i][j] << " ";
    }
    cout << endl;
  }
  return 0;
}
