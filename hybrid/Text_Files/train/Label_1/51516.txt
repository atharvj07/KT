import java.util.*;

public class Main {
    static int[][] s, map;
    static int N, M;
    static List<Integer> good = new ArrayList<>();
    static int minArea = Integer.MAX_VALUE;
    static int minX = 0;
    static int minY = 0;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        N = in.nextInt();
        M = in.nextInt();
        s = new int[N][M];
        map = new int[N][M];
        for (int i = 0; i < N; i++) {
            String line = in.next();
            for (int j = 0; j < M; j++) {
                s[i][j] = line.charAt(j) - 'A';
            }
        }
        for (int x = 1; x <= N; x++) {
            for (int y = 1; y <= M; y++) {
                int cnt = 0;
                for (int i = 0; i <= N - x; i++) {
                    for (int j = 0; j <= M - y; j++) {
                        int t = s[i][j];
                        for (int k = 0; k < x; k++) {
                            for (int l = 0; l < y; l++) {
                                if (t != s[i + k][j + l]) {
                                    t = -1;
                                    break;
                                }
                            }
                        }
                        if (t != -1) {
                            cnt++;
                            map[i][j] = t;
                        }
                    }
                }
                if (cnt > 0) {
                    int area = x * y;
                    good.add(area);
                    if (area < minArea || area == minArea && x < minX) {
                        minArea = area;
                        minX = x;
                        minY = y;
                    }
                }
            }
        }
        Collections.sort(good);
        System.out.println(good.size());
        System.out.println(minX + " " + minY);
    }
}
