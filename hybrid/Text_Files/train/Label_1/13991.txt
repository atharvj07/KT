#include<bits/stdc++.h>
#define ll long long
using namespace std;

ll n, m, ans = LLONG_MAX, vis[1005][1005];
string a[1005];

bool valid(ll x, ll y){
    return x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] != '#';
}

void dfs(ll x, ll y, ll cnt){
    if(vis[x][y] == 1) return;
    vis[x][y] = 1;
    if(x == n && y == m){
        ans = min(ans, cnt);
        vis[x][y] = 0;
        return;
    }
    if(valid(x+1, y)) dfs(x+1, y, cnt+1);
    if(valid(x-1, y)) dfs(x-1, y, cnt+1);
    if(valid(x, y+1)) dfs(x, y+1, cnt+1);
    if(valid(x, y-1)) dfs(x, y-1, cnt+1);
    vis[x][y] = 0;
}

int main(){
    cin >> n >> m;
    for(ll i = 1; i <= n; i++) cin >> a[i];
    dfs(1, 1, 0);
    if(ans == LLONG_MAX) ans = -1;
    cout << ans << endl;
    return 0;
}
