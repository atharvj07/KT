n, m = map(int, input().split())
c = list(map(int, input().split()))
g = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)
ans = 0
mx = 0
for i in range(1, n+1):
    s = set()
    for j in g[i]:
        s.add(c[j-1])
    if len(s) > mx:
        mx = len(s)
        ans = c[i-1]
print(ans)
