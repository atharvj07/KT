n, a, b, k = map(int, input().split())
s = input()

segments = []
start = 0

# Find segments where ships can be placed
for i, c in enumerate(s + '1'):  # Append '1' to include the last segment
    if c == '1':
        if start < i:
            segments.append((start, i))
        start = i + 1

shots = []
for seg in segments:
    start, end = seg
    while start + b <= end:  # Place shots in each segment to guarantee a hit
        shots.append(str(start + b))  # +1 for 1-based indexing
        start += b

# Output
print(min(len(shots), a))
print(' '.join(shots[:a]))
