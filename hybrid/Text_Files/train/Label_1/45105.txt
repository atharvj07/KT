import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int s = Integer.parseInt(st.nextToken());
        List<Instruction> instructions = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            String op = st.nextToken();
            if (op.equals("set")) {
                int x = Integer.parseInt(st.nextToken());
                int v = Integer.parseInt(st.nextToken());
                instructions.add(new SetInstruction(x, v));
            } else if (op.equals("if")) {
                int x = Integer.parseInt(st.nextToken());
                instructions.add(new IfInstruction(x));
            } else if (op.equals("end")) {
                instructions.add(new EndInstruction());
            }
        }
        int minCost = Integer.MAX_VALUE;
        for (int i = 0; i < instructions.size(); i++) {
            if (instructions.get(i) instanceof SetInstruction) {
                SetInstruction setInstruction = (SetInstruction) instructions.get(i);
                if (setInstruction.x == s) {
                    instructions.remove(i);
                    minCost = Math.min(minCost, setInstruction.v);
                    i--;
                }
            }
        }
        System.out.println(minCost == Integer.MAX_VALUE ? 0 : minCost);
    }

    private static class Instruction {
    }

    private static class SetInstruction extends Instruction {
        int x;
        int v;

        public SetInstruction(int x, int v) {
            this.x = x;
            this.v = v;
        }
    }

    private static class IfInstruction extends Instruction {
        int x;

        public IfInstruction(int x) {
            this.x = x;
        }
    }

    private static class EndInstruction extends Instruction {
    }
}
