import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] count = new int[n];
        int[][] presents = new int[n][];
        for (int i = 0; i < n; i++) {
            int s = sc.nextInt();
            count[i] = s;
            presents[i] = new int[s];
            for (int j = 0; j < s; j++) {
                presents[i][j] = sc.nextInt();
            }
        }
        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            min = Math.min(min, count[i]);
            max = Math.max(max, count[i]);
        }
        int diff = max - min;
        if (diff <= 1) {
            System.out.println(0);
            return;
        }
        int[][][] dp = new int[n][max + 1][m + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= max; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }
        int[][][] next = new int[n][max + 1][m + 1];
        dp[0][count[0]][presents[0][0]] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = min; j <= max; j++) {
                for (int k = 1; k <= m; k++) {
                    if (dp[i][j][k] != -1) {
                        if (j < max) {
                            dp[i][j + 1][k] = Math.max(dp[i][j + 1][k], dp[i][j][k]);
                            next[i][j + 1][k] = next[i][j][k];
                        }
                        int idx = Arrays.binarySearch(presents[i], k);
                        if (idx >= 0 && j > 0) {
                            dp[i][j - 1][k] = Math.max(dp[i][j - 1][k], dp[i][j][k]);
                            next[i][j - 1][k] = next[i][j][k];
                            if (i + 1 < n) {
                                dp[i + 1][j - 1][k] = Math.max(dp[i + 1][j - 1][k], dp[i][j][k] + 1);
                                next[i + 1][j - 1][k] = next[i][j][k] + 1;
                            }
                        }
                    }
                }
            }
        }
        int ans = Integer.MAX_VALUE;
        int from = -1;
        int to = -1;
        int kind = -1;
        for (int j = min; j <= max; j++) {
            for (int k = 1; k <= m; k++) {
                if (dp[n - 1][j][k] != -1 && dp[n - 1][j][k] < ans) {
                    ans = dp[n - 1][j][k];
                    from = n - 1;
                    to = -1;
                    kind = k;
                }
            }
        }
        System.out.println(ans);
        while (from != -1) {
            System.out.println(from + 1 + " " + (to + 1) + " " + kind);
            int tmp = next[from][count[from]][kind];
            if (to != -1) {
                count[from]++;
                count[to]--;
            }
            from = to;
            kind = tmp;
            if (to != -1) {
                tmp = Arrays.binarySearch(presents[to], kind);
                if (tmp >= 0) {
                    to = -1;
                }
            }
        }
    }
}
