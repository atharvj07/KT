#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
int f[N];
int a[N], ans[N];
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < 9; i++) cin >> f[i];
  for (int i = 0; i < n; i++) a[i] = s[i] - '0';
  for (int i = 0; i < n; i++) ans[i] = a[i];
  int x = -1;
  for (int i = 0; i < n; i++) {
    if (a[i] != f[a[i] - 1]) {
      if (x == -1 || a[i] > a[x]) x = i;
    }
  }
  if (x != -1) {
    int y = -1;
    for (int i = x + 1; i < n; i++) {
      if (a[i] != f[a[i] - 1]) {
        if (y == -1 || a[i] > a[y]) y = i;
      }
    }
    if (y == -1) {
      a[x] = f[a[x] - 1];
    } else {
      for (int i = x; i <= y; i++) {
        a[i] = f[a[i] - 1];
      }
    }
  }
  for (int i = 0; i < n; i++) ans[i] = max(a[i], ans[i]);
  for (int i = 0; i < n; i++) cout << ans[i];
  cout << "\n";
  return 0;
}
