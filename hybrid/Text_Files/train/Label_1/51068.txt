import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int d = input.nextInt();
        int[] s = new int[n];
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            p[i] = input.nextInt();
        }
        int[] points = new int[n];
        for (int i = 0; i < n; i++) {
            points[i] = s[i] + p[i];
        }
        Arrays.sort(points);
        int rank = 1;
        for (int i = n - 1; i >= 0; i--) {
            if (points[i] == points[n - 1]) {
                rank++;
            }
            if (points[i] == points[d - 1]) {
                System.out.println(rank);
                break;
            }
        }
    }
}
