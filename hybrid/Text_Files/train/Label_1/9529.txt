#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<bool>> has(n, vector<bool>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int x;
            cin >> x;
            has[i][j] = x;
        }
    }
    int ans = n * m * 4;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (has[i][j]) {
                ans -= 4;
                if (i > 0) ans--;
                if (j > 0) ans--;
                if (i + 1 < n) ans--;
                if (j + 1 < m) ans--;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
