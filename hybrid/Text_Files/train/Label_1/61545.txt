import sys

def possible(p, n):
    if n % 4 == 1:
        return p <= (5 * n // 2)
    elif n % 4 == 3:
        return p <= (5 * n // 2 - 1)
    else:
        return False

def sort(a):
    n = len(a)
    p = 0
    for i in range(n):
        if a[i] == i + 1:
            continue
        j = i
        while a[j] != i + 1:
            j += 1
        l = j - i + 1
        if l % 2 == 0:
            return -1
        p += l
        a[i:j+1] = reversed(a[i:j+1])
    return p

for t in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    if not possible(sort(a), n):
        print(-1)
    else:
        print(sort(a))
