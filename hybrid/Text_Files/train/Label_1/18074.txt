#include <iostream>
#include <cmath>
using namespace std;

long long getCell(long long n, long long m, long long k) {
    long long turns = k / (n + m - 2);
    k %= n + m - 2;
    long long row = (k % (n - 1)) + 1;
    long long col = ((k / (n - 1)) + 2 + turns) % m + 1;
    cout << row << " " << col << endl;
    return 0;
}

int main() {
    long long n, m, k;
    cin >> n >> m >> k;
    getCell(n, m, k);
    return 0;
}
