import sys

n = int(input())
g = [[] for i in range(n+1)]

for i in range(n-1):
    a, b = map(int, sys.stdin.readline().split())
    g[a].append(b)
    g[b].append(a)

cnt = [0] * (n+1)

def dfs(cur, par):
    cnt[cur] = 1
    for u in g[cur]:
        if u == par: continue
        dfs(u, cur)
        cnt[cur] += cnt[u]

dfs(1, 1)

ans = max(cnt)
if ans%2==0:
    print(ans//2)
else:
    print(-1)
