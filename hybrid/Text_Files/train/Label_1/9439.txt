#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  char a;
  while (n--) {
    cin >> a;
    if (a == 'a' || a == 'e' || a == 'i' || a == 'o' || a == 'u' || a == 'y') {
      cout << a;
      char b = a;
      cin >> a;
      while (a == b) {
        cin >> a;
      }
      cout << a;
    }
  }
  return 0;
}
