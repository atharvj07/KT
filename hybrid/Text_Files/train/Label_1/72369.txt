n = int(input())
points = [list(map(int, input().split())) for _ in range(n)]

def manhattan_distance(p1, p2):
    return abs(p1[0] - p2[0]) + abs(p1[1] - p2[1])

def max_perimeter(k):
    # For k points, the maximum perimeter will be formed by taking the points
    # that are furthest apart in x and y coordinates, due to the nature of
    # Manhattan distance. This means taking the two points with the minimum and
    # maximum x coordinates, and the two points with the minimum and maximum
    # y coordinates, to form a non-self-intersecting polygon.

    min_x = min(points, key=lambda x: x[0])
    max_x = max(points, key=lambda x: x[0])
    min_y = min(points, key=lambda x: x[1])
    max_y = max(points, key=lambda x: x[1])

    # The perimeter of the rectangle formed by these points
    rectangle_perimeter = 2 * (manhattan_distance(min_x, max_x) + manhattan_distance(min_y, max_y))

    # For k = 3, we cannot form a rectangle, so we take the largest three sides.
    if k == 3:
        return rectangle_perimeter - min(manhattan_distance(min_x, min_y), manhattan_distance(min_x, max_y), manhattan_distance(max_x, min_y), manhattan_distance(max_x, max_y))
    else:
        # For k > 3, the maximum perimeter increases linearly with the number of points,
        # as we can place additional points on the sides of the rectangle without
        # changing the perimeter.
        return rectangle_perimeter + 2 * (k - 4)

for k in range(3, n + 1):
    print(max_perimeter(k), end=' ')
