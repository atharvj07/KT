import java.util.*;
import java.io.*;

public class Main {
    static char[][] grid;
    static int[][] red;
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new char[n + 2][m + 2];
        red = new int[n + 2][m + 2];
        for (int i = 1; i <= n; i++) {
            String line = sc.next();
            for (int j = 1; j <= m; j++) {
                grid[i][j] = line.charAt(j - 1);
            }
        }

        int ans = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (grid[i][j] != '#') {
                    if (j - 1 >= 1 && grid[i][j - 1] == '.' || i - 1 >= 1 && grid[i - 1][j] == '.') {
                        ans++;
                        sb.append("R " + i + " " + j + "\n");
                    } else {
                        ans++;
                        sb.append("B " + i + " " + j + "\n");
                    }
                }
            }
        }
        System.out.println(ans);
        System.out.print(sb);
    }
}
