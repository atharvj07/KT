void solve() {
    int n, k;
    cin >> n >> k;
    int a[n + 1];
    bool flag = 0;
    for (int i = n; i >= 0; i--) {
        string s;
        cin >> s;
        if (s == "?") continue;
        a[i] = stoi(s);
        if (i != 0) flag = 1;
    }
    if (!flag) {
        cout << "Yes";
        return;
    }
    if (a[0] % k != 0) {
        cout << "Yes";
        return;
    }
    cout << "No";
}

int main() {
    solve();
    return 0;
