from collections import defaultdict

def dfs(v, u):
    vis[v] = True
    for c in graph[v]:
        if c != u:
            if not vis[c]:
                if dfs(c, v):
                    return True
            else:
                return True
    return False

n, m = map(int, input().split())
graph = defaultdict(list)
for e in range(m):
    x, y = map(int, input().split())
    x -= 1
    y -= 1
    graph[x].append(y)
    graph[y].append(x)
vis = [False for _ in range(n)]
if dfs(0, 0):
    print('NO')
else:
    print('YES')
    for u in graph:
        for v in graph[u]:
            if u < v:
                print(0, end='')
            else:
                print(1, end='')
