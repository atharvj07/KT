from sys import stdin
input = stdin.readline

n, k = map(int, input().split())
cards = [input().strip() for _ in range(n)]

def same(l1, l2, l3):
	for i in range(k):
		if l1[i] != l2[i] and l1[i] != l3[i] and l2[i] != l3[i]:
			return False
	return True

def diff(l1, l2, l3):
	for i in range(k):
		if l1[i] == l2[i] and l1[i] == l3[i]:
			return False
	return True

ans = 0
for i in range(n):
	for j in range(i + 1, n):
		for l in range(j + 1, n):
			if same(cards[i], cards[j], cards[l]) or diff(cards[i], cards[j], cards[l]):
				ans += 1
print(ans)
