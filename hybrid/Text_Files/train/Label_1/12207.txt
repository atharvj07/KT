#include <bits/stdc++.h>
using namespace std;
#define int long long
#define eb emplace_back
#define vi vector<int>
const int M = 1<<17;
int n, m, dp[M][3], ans = 1e18;
string s, t;
int32_t main() {
    cin >> n >> s >> m >> t;
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < (1 << m); j++) {
            dp[i][j] = dp[i - 1][j];
            if ((j & 1) == (s[i - 1] == t[0])) {
                int f = 1;
                for (int k = 1; k < m; k++) {
                    if ((j >> k & 1) != (t[k] == s[i - k - 1])) {
                        f = 0;
                        break;
                    }
                }
                if (f) dp[i][j] = max(dp[i][j], dp[i - m][0] + 1);
            }
            if (s[i - 1] != '?') {
                if (((j >> 1) & 1) == (s[i - 1] == t[0])) dp[i][j] = max(dp[i][j], dp[i - 1][j >> 1] + 1);
                else dp[i][j] = max(dp[i][j], dp[i - 1][(j >> 1) | (1 << (m - 1))]);
            } else {
                dp[i][j] = max(dp[i][j], dp[i - 1][j >> 1] + 1);
                dp[i][j] = max(dp[i][j], dp[i - 1][(j >> 1) | (1 << (m - 1))]);
            }
        }
    }
    for (int i = 0; i < (1 << m); i++) ans = min(ans, n - dp[n][i] * m);
    cout << ans << endl;
    return 0;
}
