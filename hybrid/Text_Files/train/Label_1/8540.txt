#include <iostream>
#include <bitset>

using namespace std;

int main() {
    int n;
    cin >> n;
    bitset<7> times[n];
    for (int i = 0; i < n; i++) {
        int t;
        cin >> t;
        times[i] = t;
    }
    int ans = 0;
    for (int i = 0; i < 7; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) {
            cnt += times[j][i];
        }
        ans = max(ans, cnt);
    }
    cout << ans << endl;
    return 0;
}
