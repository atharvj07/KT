#include <algorithm>
#include <cstdio>
#include <vector>

using namespace std;

struct Edge {
  int to, len;
};

const int MAXN = 1005;
const int MAXM = 2005;
const int INT_INF = 0x3f3f3f3f;

// u is the start point, all dist refer to it
int dist[MAXN];
int parent[MAXN];

vector<Edge> adj[MAXN];

// Calculate the original shortest path tree of graph
// (start from edge u) using dijkstra algorithm
void dikjstra(int n, int u) {
  for (int i = 1; i <= n; i++) dist[i] = INT_INF;
  dist[u] = 0;
  for (int i = 1; i <= n; i++) {
    // Find the vertex with the smallest dist that hasn't been visited
    int mindist = INT_INF;
    int next = 0;
    for (int j = 1; j <= n; j++) {
      if (dist[j] < mindist && !parent[j]) {
        next = j;
        mindist = dist[j];
      }
    }
    if (next == 0) break;
    for (auto e : adj[next]) {
      int newdist = dist[next] + e.len;
      if (newdist < dist[e.to]) {
        dist[e.to] = newdist;
        parent[e.to] = next;
      }
    }
  }
}

int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) scanf("%d", &v[i]);
  for (int i = 1; i <= m; i++) {
    int a, b;
    scanf("%d%d", &a, &b);
    adj[a].push_back(Edge{b, v[b]});
    adj[b].push_back(Edge{a, v[a]});
  }

  long long ans = 0;
  // Calculate the shortest path tree of each vertex
  for (int i = 1; i <= n; i++) {
    dikjstra(n, i);
    for (int j = 1; j <= n; j++) {
      if (i == j) continue;
      ans += dist[j];
    }
  }

  printf("%lld\n", ans);
  return 0;
}
