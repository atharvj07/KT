n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
a.sort()
b.sort()
k=0; t=0
for i in range(n):
    if a[i]>0:
        if b[k]==a[i]:
            k+=1; continue
        elif b[k]>a[i]:
            b[k]-=a[i]; t+=a[i]; continue
        else:
            t+=b[k]; a[i]-=b[k]; k+=1
            if b[k]==a[i]:
                k+=1; continue
            elif b[k]>a[i]:
                b[k]-=a[i]; t+=a[i]; continue
            else:
                t+=b[k]; a[i]-=b[k]; k+=1
print(k,t)
