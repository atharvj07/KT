#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string p, h;
    cin >> p >> h;
    int i, j, k;
    for (i = 0, j = 0, k = 0; i < h.length(); i++) {
      if (h[i] == p[j] && j < p.length()) j++;
      else if (k == 0 && (j == p.length() || h[i] > p[j])) k = i;
      else if (k > 0 && h[i] < p[j]) break;
    }
    if (j == p.length() && k == 0) cout << "YES\n";
    else cout << "NO\n";
  }
}
