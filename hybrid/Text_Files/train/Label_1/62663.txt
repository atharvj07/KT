t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    sorted_a = sorted([(val, index) for index, val in enumerate(a)])
    dp = [1] * n
    for i in range(n):
        for j in range(i):
            if sorted_a[i][1] > sorted_a[j][1]:
                dp[i] = max(dp[i], dp[j] + 1)
    print(n - max(dp))
