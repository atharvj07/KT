#include <cinttypes>
#include <iostream>
#include <vector>

int main() {
    int x, y, x0, y0;
    std::cin >> x >> y >> x0 >> y0;
    std::string s;
    std::cin >> s;
    std::vector<int> ans(s.size() + 1);
    int dx[] = {-1, 1, 0, 0};
    int dy[] = {0, 0, -1, 1};
    for (size_t i = 0; i < s.size() + 1; i++) {
        std::vector<std::vector<int8_t>> was(x + 1, std::vector<int8_t>(y + 1, 0));
        int xx = x0, yy = y0, cnt = 0;
        for (size_t j = 0; j < i; j++) {
            int ndx = xx + dx[s[j] - 'L'], ndy = yy + dy[s[j] - 'L'];
            if ((ndx > 0 && ndx <= x) && (ndy > 0 && ndy <= y) && !was[ndx][ndy]) {
                cnt++;
                xx = ndx;
                yy = ndy;
            }
            was[xx][yy] = 1;
        }
        for (int j = 1; j <= x; j++)
            for (int k = 1; k <= y; k++)
                if (!was[j][k]) cnt++;
        ans[i] = cnt;
    }
    for (size_t i = 0; i < s.size() + 1; i++) { std::cout << ans[i] << " "; }
}
