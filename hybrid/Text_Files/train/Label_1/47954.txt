import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] maxL = new int[n];
        int[] maxR = new int[n];
        Arrays.fill(maxL, -1);
        Arrays.fill(maxR, -1);
        maxL[0] = a[0];
        maxR[n - 1] = a[n - 1];
        for (int i = 1; i < n; i++) {
            maxL[i] = Math.max(maxL[i - 1], a[i]);
        }
        for (int i = n - 2; i >= 0; i--) {
            maxR[i] = Math.max(maxR[i + 1], a[i]);
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += Math.max(maxL[i], maxR[i]) > a[i] ? 1 : 0;
        }
        System.out.println(ans);
    }
}
