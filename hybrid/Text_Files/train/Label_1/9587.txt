#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll inf = 4e18;
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (auto &i : a) cin >> i;
    int t = n / k;
    vector<int> b(t, 0);
    for (int i = 0; i < t; i++) b[i] += (a[i] != a[i + t]);
    ll ans = inf;
    for (int i = 0; i < t; i++) {
        int cnt = 0, num = a[i];
        for (int j = 0; j < t; j++) {
            if (a[i + j] != a[i + j % t]) cnt++;
            if (a[i + j % t] != num) num = 3;
        }
        if (num != 3) ans = min(ans, 1ll * (cnt + b[i] * k));
    }
    if (ans != inf) cout << ans << endl;
    else
        cout << 0 << endl;
}
