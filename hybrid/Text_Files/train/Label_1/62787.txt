from collections import deque

n, k = map(int, input().split())
d = list(map(int, input().split()))
if max(d) > n-1:
    print(-1)
else:
    adj = [set() for i in range(n)]
    for i, di in enumerate(d):
        for j in range(i+1, n):
            if di == d[j] + 1:
                adj[i].add(j)
                adj[j].add(i)
    edges = []
    for i in range(n):
        for j in adj[i]:
            if j > i:
                edges.append((i+1, j+1))
    print(len(edges))
    for e in edges:
        print(*e)
