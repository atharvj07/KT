import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.next();
        boolean isBerlanese = true;
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == 'n') {
                continue;
            } else if (!isVowel(s.charAt(i))) {
                if (i + 1 >= s.length() || !isVowel(s.charAt(i + 1))) {
                    isBerlanese = false;
                    break;
                } else {
                    i++;
                }
            }
        }
        System.out.println(isBerlanese ? "YES" : "NO");
    }

    private static boolean isVowel(char c) {
        return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
    }
}
