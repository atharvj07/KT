#include <iostream>
#include <vector>
#include <set>
#include <algorithm>
using namespace std;
const int maxn = 1001;
int ans, n;
vector<pair<pair<int, int>, pair<int, int>>> s;
vector<int> lx, ly, rx, ry;
int x[maxn * maxn], y[maxn * maxn], z[maxn * maxn];
int main() {
    cin >> n;
    s.resize(n);
    for (int i = 0; i < n; i++) {
        int ax, ay, bx, by;
        cin >> ax >> ay >> bx >> by;
        if (ax > bx) { swap(ax, bx); }
        if (ay > by) { swap(ay, by); }
        s[i] = {{ax, ay}, {bx, by}};
    }
    for (int i = 0; i < n; i++) {
        lx.push_back(s[i].first.first);
        ly.push_back(s[i].first.second);
        rx.push_back(s[i].second.first);
        ry.push_back(s[i].second.second);
    }
    sort(lx.begin(), lx.end());
    sort(ly.begin(), ly.end());
    sort(rx.begin(), rx.end());
    sort(ry.begin(), ry.end());
    lx.erase(unique(lx.begin(), lx.end()), lx.end());
    ly.erase(unique(ly.begin(), ly.end()), ly.end());
    rx.erase(unique(rx.begin(), rx.end()), rx.end());
    ry.erase(unique(ry.begin(), ry.end()), ry.end());
    int m = lx.size() * ly.size() + rx.size() * ry.size();
    for (int i = 0; i < n; i++) {
        int ax = s[i].first.first, ay = s[i].first.second;
        int bx = s[i].second.first, by = s[i].second.second;
        int lx = lower_bound(::lx.begin(), ::lx.end(), ax) - ::lx.begin();
        int ly = lower_bound(::ly.begin(), ::ly.end(), ay) - ::ly.begin();
        int rx = lower_bound(::rx.begin(), ::rx.end(), bx) - ::rx.begin();
        int ry = lower_bound(::ry.begin(), ::ry.end(), by) - ::ry.begin();
        x[lx * ::ly.size() + ly] = 1;
        x[rx * ::ry.size() + ry] = -1;
    }
    for (int i = 0; i < m; i++) {
        z[i] = x[i] + z[i - 1];
        if (z[i]) { ans++; }
    }
    cout << ans;
}
