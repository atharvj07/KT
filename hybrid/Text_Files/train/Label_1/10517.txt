#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    int cnt = 0, last = -1;
    for (int i = 0; i < n; i++) {
        if (v[i] != last) {
            cnt++;
            last = v[i];
        }
    }
    if (cnt <= 2) cout << "YES" << endl;
    else cout << "NO" << endl;
}
