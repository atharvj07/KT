#include <bits/stdc++.h>
#define N 200000

using namespace std;

pair<int, int> tree[N << 2];
int arr[N];

void build(int node, int s, int e) {
    if (s == e) {
        tree[node] = {arr[s], arr[s]};
        return;
    }

    int mid = (s + e) >> 1;
    build(node << 1, s, mid);
    build(node << 1 | 1, mid + 1, e);

    if (tree[node << 1].first < tree[node << 1 | 1].first) {
        tree[node].first = tree[node << 1].first;
        tree[node].second = tree[node << 1 | 1].second;
    } else if (tree[node << 1].first == tree[node << 1 | 1].first) {
        tree[node].first = tree[node << 1].first;
        tree[node].second = min(tree[node << 1].second, tree[node << 1 | 1].second);
    } else {
        tree[node].first = tree[node << 1 | 1].first;
        tree[node].second = tree[node << 1].second;
    }
}

pair<int, int> query(int node, int s, int e, int l, int r) {
    if (l > e || r < s) return {INT_MAX, INT_MAX};
    if (l <= s && e <= r) return tree[node];

    int mid = (s + e) >> 1;
    pair<int, int> left = query(node << 1, s, mid, l, r);
    pair<int, int> right = query(node << 1 | 1, mid + 1, e, l, r);

    if (left.first < right.first) return left;
    if (left.first == right.first) return {left.first, min(left.second, right.second)};
    return right;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        for (int i = 1; i <= n; i++) { cin >> arr[i]; }

        build(1, 1, n);

        pair<int, int> ans = query(1, 1, n, 1, n);
        if (ans.first == ans.second) cout << 0 << endl;
        else cout << 1 << endl;
    }
    return 0;
}
