import sys

n = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
ok = False

for i in range(n-1):
    if b[i] > c[i]:
        ok = True
        break

if ok:
    print(-1)
    sys.exit()

a = [0] * n
a[0] = b[0]
a[1] = c[0]
for i in range(1, n - 1):
    if b[i] == a[i]:
        a[i+1] = c[i]
    else:
        a[i+1] = b[i]

print(*a)
