#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        long long sum = 0;
        for (int x : a) sum += x;
        long long ans = 0;
        for (int x : a) {
            if (x > (sum - x) / (n - 1)) ans++;
        }
        cout << ans << '\n';
    }
}
