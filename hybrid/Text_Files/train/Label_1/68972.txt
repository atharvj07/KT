n=int(input())
s=input()
t=s.replace('S','0').replace('G','1')
z=[0,0]
for i in t:
    z[int(i)]=z[int(i)]+1
ans=0
if z[1]%2==0:
    ans=min(z[1],n)
else:
    ans=z[1]-1
for i in range(n):
    if i>0:
        x=list(t)
        x[i]=x[i-1]
        if x[i-1]=='0':
            x[i-1]='1'
        else:
            x[i-1]='0'
        pref=[]
        suff=[]
        cn=0
        for j in x:
            if j=='0':
                cn=0
            else:
                cn=cn+1
            pref.append(cn)
        cn=0
        for j in range(n-1,-1,-1):
            if x[j]=='0':
                cn=0
            else:
                cn=cn+1
            suff.append(cn)
        for j in range(n-1):
            ans=max(ans,pref[j]+suff[n-j-2])
    if i<n-1:
        x=list(t)
        x[i]=x[i+1]
        if x[i+1]=='0':
            x[i+1]='1'
        else:
            x[i+1]='0'
        pref=[]
        suff=[]
        cn=0
        for j in x:
            if j=='0':
                cn=0
            else:
                cn=cn+1
            pref.append(cn)
        cn=0
        for j in range(n-1,-1,-1):
            if x[j]=='0':
                cn=0
            else:
                cn=cn+1
            suff.append(cn)
        for j in range(n-1):
            ans=max(ans,pref[j]+suff[n-j-2])
print(ans)
