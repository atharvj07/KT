import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String t = in.nextLine();
        int n = t.length();
        for (int i = 0; i <= n / 2; i++) {
            if (t.substring(0, i).repeat(2).equals(t)) {
                System.out.println(t.substring(0, i));
                return;
            }
        }
        System.out.println(":(");
    }
}
