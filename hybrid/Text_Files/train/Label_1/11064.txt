#include <iostream>
#include <vector>
#include <set>
#include <algorithm>

using namespace std;

const int N = 3e5 + 5;
int bit[N], a[N];
vector<int> adj[N];

void update(int x, int val) {
    while(x < N) {
        bit[x] += val;
        x += x & -x;
    }
}

int get(int x) {
    int sum = 0;
    while(x > 0) {
        sum += bit[x];
        x -= x & -x;
    }
    return sum;
}

int get(int l, int r) {
    return get(r) - get(l - 1);
}

int main() {
    int n, q; cin >> n >> q;
    for(int i = 0; i < n; i++) {
        int x; cin >> x;
        a[i] = x;
        update(i + 1, x);
    }
    for(int i = 0; i < q; i++) {
        int x, y; cin >> x >> y;
        int ans = get(x + 1, y);
        if(ans > 0) cout << "Shi" << endl;
        else cout << "Fou" << endl;
    }
    return 0;
}
