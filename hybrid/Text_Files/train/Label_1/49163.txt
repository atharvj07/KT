import java.util.*;
import java.io.*;

public class Main {
   public static void main(String[] args) {
      Scanner cin = new Scanner(System.in);
      int n = cin.nextInt(), p = cin.nextInt();
      if (p == 0) {
         int cnt = 0;
         while (n > 0) {
            if (n % 2 == 1) {
               cnt++;
               n--;
            }
            n /= 2;
         }
         System.out.println(cnt);
      } else {
         int cnt = 0;
         while (n > 0) {
            if (n % 2 == 1) {
               n += p;
            }
            if (n % 2 == 0) {
               int curr = 0;
               while (n % (p + 2) == 0) {
                  n /= (p + 2);
                  curr++;
               }
               cnt++;
               if (curr > 0)
                  n /= 2;
            }
            if (n < 0) {
               System.out.println(-1);
               return;
            }
         }
         System.out.println(cnt);
      }
   }
}
