#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 5005;
int n, vis[N];
vector<int> edges[N];
void dfs(int u) {
    vis[u] = 1;
    for(int v: edges[u]) {
        if(vis[v]) continue;
        dfs(v);
    }
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for(int i = 2;i <= n;i++) {
        int x;
        cin >> x;
        edges[x].push_back(i);
    }
    int cnt = 0;
    for(int i = 1;i <= n;i++) {
        if(!vis[i]) {
            ++cnt;
            dfs(i);
        }
    }
    cout << cnt << "\n";
    return 0;
}
