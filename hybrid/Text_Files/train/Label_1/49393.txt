import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[x[i] - 1] = i;
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += Math.abs(p[i] - p[i + 1]);
        }
        System.out.println(ans);
    }
}
