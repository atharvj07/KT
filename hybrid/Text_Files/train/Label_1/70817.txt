import sys
n, m = map(int, sys.stdin.readline().split())
d = list(map(int, sys.stdin.readline().split()))
a = list(map(int, sys.stdin.readline().split()))

if any(day == 0 for day in d): 
    print(-1)
else: 
    days = 0
    while True:
        for i, day in enumerate(d):
            if not a[day -1]: 
                continue
            days += 1 
            a[day -1] -= 1
        if all(a[day -1] == 0 for day in d): 
            break
    print(days)
