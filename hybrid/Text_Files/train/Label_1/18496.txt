#include<iostream>
#include<string>
using namespace std;
int main()
{
 string s; cin>>s;
 for(int i=0; i<(int)s.size()-1; i++)
  {
   if(s[i]=='y'&&s[i+1]=='x')
    {
      swap(s[i],s[i+1]);
      i++;
    }
   else if(s[i]=='x'&&s[i+1]=='y')
    {
      s.erase(i,2);
      i--;
    }
  }
 cout<<s<<endl;
 return 0;
}
