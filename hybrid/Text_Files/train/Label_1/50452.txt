import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            if (max >= n - 1) {
                System.out.println("yes");
                break;
            }
            max = Math.max(max, i + d[i]);
        }
        if (max < n - 1) {
            System.out.println("no");
        }
    }
}
