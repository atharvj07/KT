#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;

int a[maxn];
int sum[maxn];
int n, m;
int q[maxn];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    sum[i] = sum[i - 1] + a[i];
  }
  cin >> m;
  for (int i = 1; i <= m; i++) cin >> q[i];
  for (int i = 1; i <= m; i++) {
    int l = 1, r = n, ans = 0;
    while (l <= r) {
      int mid = (l + r) >> 1;
      if (sum[mid] < q[i]) {
        l = mid + 1;
        ans = mid;
      } else {
        r = mid - 1;
      }
    }
    cout << ans + 1 << endl;
  }
  return 0;
}
