for _ in range(int(input())):
    n, m = map(int, input().split())
    tile = []
    for i in range(n):
        a = []
        for j in range(2):
            a += [*map(int, input().split())]
        tile.append(a)
    a = [[0] * m for i in range(m)]
    for i in tile:
        for j in tile:
            if i == j:
                for x in range(m - 1, -1, -1):
                    for y in range(m - 1, -1, -1):
                        if a[x][y] == a[m - x - 1][m - y - 1] == 0:
                            a[x][y] = a[m - x - 1][m - y - 1] = i[0]
                            a[x][m - y - 1] = a[m - x - 1][y] = i[1]
    c = 0
    for i in range(m):
        for j in range(m):
            if a[i][j] == 0:
                c = 1
                break
    if c == 1:
        print('NO')
    else:
        print('YES')
