#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mxN = 100005;
ll n, ara[mxN], cnt[mxN], mx;
int main()
{
    ll t;
    cin >> t;
    while(t--)
    {
        cin >> n;
        memset(cnt,0,sizeof(cnt));
        mx = 0;
        for(ll i=1;i<=n;i++) cin >> ara[i],cnt[ara[i]]++;
        for(ll i=1;i<=n;i++) mx = max(mx,cnt[i]);

        cout << mx-1 << endl;
    }
}
