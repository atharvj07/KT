import sys
from collections import deque

def solve():
    n, m = map(int, input().split())
    x, k, y = map(int, input().split())
    a = [int(x) for x in input().split()]
    b = [int(x) for x in input().split()]
    dq = deque()
    dq.append((0, a[0]))
    for i in range(1, n):
        while dq and dq[-1][1] > a[i]:
            dq.pop()
        dq.append((i, a[i]))
    mn = 1e9
    for i in range(m):
        while dq and dq[0][0] + k - 1 < i:
            dq.popleft()
        if dq and dq[0][1] == b[i]:
            mn = min(mn, dq[0][0] * y + x)
    if mn == 1e9:
        mn = -1
    return mn
print(solve())
