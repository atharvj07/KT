import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        int neg = 0, pos = 0;
        for (int i = 0; i < n; i++) {
            int sign = 1;
            for (int j = i; j < n; j++) {
                sign *= arr[j];
                if (sign == 0) {
                    break;
                }
                if (sign == 1) {
                    pos++;
                } else {
                    neg++;
                }
            }
        }
        System.out.println(neg + " " + pos);
    }
}
