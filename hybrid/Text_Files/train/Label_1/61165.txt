import sys
input = sys.stdin.readline
t = int(input())
for i in range(t):
  n, q = map(int, input().split())
  a = list(map(int, input().split()))
  for j in range(q):
    s = int(input())
    mn = 1e9
    mx = -1e9
    for k in a:
      mn = min(mn, k)
      mx = max(mx, k)
    L = []
    R = []
    if mn == mx:
      if s == mx * n:
        print('Yes')
      else:
        print('No')
      continue
    for k in a:
      if k <= (mx + mn) / 2:
        L.append(k)
      else:
        R.append(k)
    if (mx + mn) / 2 in a:
      R.append((mx + mn) / 2)
    ok = False
    for k in range(len(L) + len(R) + 1):
      if len(L) >= k and len(R) >= len(L) - k:
        if sum(L) + sum(R[:len(L) - k]) == s:
          ok = True
          break
    if ok:
      print('Yes')
    else:
      print('No')
