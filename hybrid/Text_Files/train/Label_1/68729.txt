import math

for _ in range(int(input())):
    n, m, x, y = map(int, input().split())
    a = [input() for _ in range(n)]
    ans = math.inf
    for _ in range(2):
        cost = (m+1)//2 * y
        for i in range(n):
            for j in range(m):
                if j > 0 and a[i][j] == '.' and a[i][j-1] == '.':
                    cost -= y-x
                if a[i][j] == '.':
                    cost += x
        ans = min(ans, cost)
        for i in range(n):
            a[i] = a[i][::-1]
    print(ans)
