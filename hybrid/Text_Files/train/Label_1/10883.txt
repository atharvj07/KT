#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    long long aa[n], bb[n];
    int acnt = 0, bcnt = 0;
    for (int i = 0; i < n; i++) {
        cin >> aa[i];
        if (aa[i] == 0) acnt++;
    }
    for (int i = 0; i < n; i++) {
        cin >> bb[i];
        if (bb[i] == 0) bcnt++;
    }
    long long suma = 0, sumb = 0;
    for (int i = 0; i < n; i++) {
        suma += aa[i];
        sumb += bb[i];
    }
    if (suma == 0 && sumb == 0) {
        cout << n << endl;
        return 0;
    } else if (suma == 0) {
        cout << bcnt << endl;
        return 0;
    } else if (sumb == 0) {
        cout << acnt << endl;
        return 0;
    } else if (suma == -sumb) {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (aa[i] + bb[i] == 0) cnt++;
        }
        cout << cnt << endl;
        return 0;
    } else {
        cout << 0 << endl;
        return 0;
    }
}
