def solve():
    n, k = map(int, input().strip().split())
    cards = list(map(int, input().strip().split()))
    favs = list(map(int, input().strip().split()))
    joys = list(map(int, input().strip().split()))
    fav_dict = {}
    for i in range(n):
        fav_dict[favs[i]] = fav_dict.get(favs[i], 0) + k

    cards.sort()
    
    ans = 0
    for card in cards:
        if fav_dict.get(card, 0) > 0:
            ans += joys[fav_dict[card] - 1]
            fav_dict[card] -= 1
    
    print(ans)

solve()
