#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n,a,ans=0;
  cin >> n;
  vector<long long>v(n+1);
  for(int i=1;i<=n;++i) {
    cin >> a;
    v[i]=v[i-1]+a;
  }
  vector<long long>mx(n+1),mn(n+1);
  mx[n]=v[n]; mn[n]=v[n];
  for(int i=n-1;i>=1;--i) {
    mx[i]=max(mx[i+1],v[i]);
    mn[i]=min(mn[i+1],v[i]);
  }
  for(int i=1;i<=n;++i) {
    if(v[i]<=mn[i] || v[i]>=mx[i]) {
      ans+=min(mx[i]-v[i],v[i]-mn[i]);
    }
  }
  cout << ans << endl;
  return 0;
}
