#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> P;
int n;
vector<vector<P>> adj;
int v[1000001];
int dx[] = {0,0,1,-1};
int dy[] = {1,-1,0,0};
vector<P> pos;
bool vis[1000001];
void dfs(int c, int d) {
  vis[c] = true;
  pos.push_back(P(d,c));
  for (int i=0;i<adj[c].size();i++) {
    int e = adj[c][i].first;
    if (vis[e]) continue;
    dfs(e,adj[c][i].second);
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n;
  adj.resize(n+1);
  for (int i=1;i<=n;i++) {
    int x; cin >> x;
    v[x] = i;
  }
  queue<P> q;
  for (int i=0;i<=n;i++) {
    if (v[i] == 0) q.push(P(0,i));
  }
  while (!q.empty()) {
    P p = q.front(); q.pop();
    if (vis[p.first]) continue;
    dfs(p.first,p.second);
    int nx = p.first + dx[p.second];
    int ny = p.first + dy[p.second];
    if (nx >= 0 && ny >= 0 && nx <= n && ny <= n) {
      if (!vis[nx]) {
        adj[p.first].push_back(P(nx,p.second));
        q.push(P(nx,p.second));
      }
    }
  }
  int mn = abs(pos[0].first - pos[n-1].first) + abs(pos[0].second - pos[n-1].second);
  P p1 = pos[0];
  P p2 = pos[n-1];
  if (abs(pos[0].first - pos[n-1].first) > abs(pos[0].second - pos[n-1].second)) {
    p1.first = pos[n-1].first;
    p2 = pos[0];
  }
  if (mn % 2 != 0) {
    cout << -1 << '\n';
    return 0;
  }
  mn /= 2;
  if (p1.first+mn <= n && p1.second+mn <= n && p2.first+mn <= n && p2.second+mn <= n) {
    cout << mn*2+1 << " " << mn*2+1 << '\n';
    cout << p1.first+mn+1 << " " << p1.second+mn+1 << '\n';
  } else {
    cout << -1 << '\n';
  }
  return 0;
}
