import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            int sum = 0;
            for (int j : a) {
                sum += j;
            }
            sum = (sum + n - 1) / n;
            System.out.println(sum);
        }
    }
}
