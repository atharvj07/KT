import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }

        int[] left = new int[n];
        int[] right = new int[n];

        left[0] = d[0];
        for (int i = 1; i < n; i++) {
            left[i] = Math.max(left[i - 1], d[i]);
        }

        right[n - 1] = d[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            right[i] = Math.max(right[i + 1], d[i]);
        }

        boolean canJump = true;
        for (int i = 0; i < n; i++) {
            if (i + d[i] < n - 1 && left[i + d[i]] <= 0) {
                canJump = false;
                break;
            }
            if (i - d[i] >= 0 && right[i - d[i]] <= 0) {
                canJump = false;
                break;
            }
        }

        if (canJump) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
