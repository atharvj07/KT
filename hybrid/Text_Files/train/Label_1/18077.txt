#include <bits/stdc++.h> 

using namespace std;

typedef long long ll;
typedef vector<ll> vi;

vi factor(ll n) {
	vi ans;
	for (ll i = 2; i * i <= n; ++i) {
		if (n % i == 0) {
			ans.push_back(i);
			while (n % i == 0) {
				n /= i;
			}
		}
	}
	if (n > 1) {
		ans.push_back(n);
	}
	return ans;
}

int main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	int t;
	cin >> t;
	while (t--) {
		ll n;
		cin >> n;
		vi f = factor(n);
		int len = f.size();
		cout << len << "\n";
		for (int i = 0; i < len; ++i) {
			cout << f[i] << ' ';
		}
		cout << "\n";
	}
	return 0;
}
