#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mxn = 1e5 + 10;
const int mod = 1e9 + 7;
const int inf = INT_MAX;
const ll linf = LLONG_MAX;
#define endl '\n'

int good[mxn]; bool vis[mxn];
int32_t main()
{
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);

    int n; cin >> n;
    for (int i=0; i<n; i++) { cin >> good[i]; vis[good[i]] = 1; }

    int dp[mxn]; dp[0] = 0;
    for (int i=1; i<mxn; i++)
    {
        dp[i] = 0;
        for (int j=0; j<n; j++)
        {
            if (!vis[i-good[j]]) continue;
            dp[i] = max(dp[i], dp[i-good[j]] + 1);
        }
    }

    int ans = 0;
    for (int i=0; i<mxn; i++) ans = max(ans, dp[i]);
    cout << ans << endl;
}
