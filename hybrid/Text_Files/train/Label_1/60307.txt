def solve():
    text = input()
    n = int(text)
    words = text.split()
    m = {}
    last = {}
    for i in range(n):
        if words[i] not in m:
            m[words[i]] = [(i, 1)]
            last[words[i]] = i
        else:
            m[words[i]].append((i, len(words[i])))
    for w in words:
        if len(m[w]) > 1:
            mn = len(w)
            pos = -1
            for (i, l) in m[w]:
                if i == last[w] or i != 0 and i - 1 not in last[w]:
                    mn = min(l, mn)
            for (i, l) in m[w]:
                if l == mn:
                    if i == last[w] or i != 0 and i - 1 not in last[w]:
                        pos = i
            if pos != -1:
                last[w] = pos - 1
                print(n - len(w) + 1)
                return

    print(n)

solve()
