import sys

n = int(input())
S = set(map(int, input().split()))
if 0 in S or 1 in S: x = 1
else: x = 0
for s in S:
    if x > 0 and s < 2**x: x = 0
    elif s < 2**x+2 and 2**x not in S: x = x+1
print(x)
if x > 0:
    p = []
    for i in range(2**x):
        if i > 0 and i^p[-1] not in S:
            print(0); sys.exit()
        p.append(i^p[-1])
    print(*p)
