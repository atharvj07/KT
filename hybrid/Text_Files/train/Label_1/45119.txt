import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        boolean ok = true;
        for (int i = 1; i <= n; i++) {
            int sum = 0;
            for (int j = i; j <= n; j += i) {
                sum += a[j];
            }
            if (sum % 2 != 0) {
                ok = false;
            }
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        int m = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] == 1) {
                m++;
            }
        }
        System.out.println(m);
        for (int i = 1; i <= n; i++) {
            if (a[i] == 1) {
                System.out.print(i + " ");
            }
        }
        System.out.println();
    }
}
