def max_cookies(n, k, a, b):
    left, right = 0, 2 * 10**9
    while left < right:
        mid = (left + right + 1) // 2
        need = sum(max(0, mid * ai - bi) for ai, bi in zip(a, b))
        if need <= k:
            left = mid
        else:
            right = mid - 1
    return left

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

print(max_cookies(n, k, a, b))
