import java.util.Scanner;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] forest = new int[n + 2][m + 2];
        int[][] dist = new int[n + 2][m + 2];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                forest[i][j] = 1;
                dist[i][j] = -1;
            }
        }
        int k = sc.nextInt();
        Queue<int[]> q = new LinkedList<>();
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            q.add(new int[]{x, y});
            dist[x][y] = 0;
        }
        while (!q.isEmpty()) {
            int[] curr = q.remove();
            int x = curr[0];
            int y = curr[1];
            if (x - 1 >= 1 && forest[x - 1][y] == 1 && dist[x - 1][y] == -1) {
                dist[x - 1][y] = dist[x][y] + 1;
                q.add(new int[]{x - 1, y});
            }
            if (x + 1 <= n && forest[x + 1][y] == 1 && dist[x + 1][y] == -1) {
                dist[x + 1][y] = dist[x][y] + 1;
                q.add(new int[]{x + 1, y});
            }
            if (y - 1 >= 1 && forest[x][y - 1] == 1 && dist[x][y - 1] == -1) {
                dist[x][y - 1] = dist[x][y] + 1;
                q.add(new int[]{x, y - 1});
            }
            if (y + 1 <= m && forest[x][y + 1] == 1 && dist[x][y + 1] == -1) {
                dist[x][y + 1] = dist[x][y] + 1;
                q.add(new int[]{x, y + 1});
            }
        }
        int maxDist = 0;
        int x = 0;
        int y = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (dist[i][j] > maxDist) {
                    maxDist = dist[i][j];
                    x = i;
                    y = j;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
