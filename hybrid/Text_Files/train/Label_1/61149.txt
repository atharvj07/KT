n = int(input().strip())
c = list(map(int, input().strip().split()))
t = list(map(int, input().strip().split()))

if c[0] != t[0] or c[-1] != t[-1]:
    print("No")
else:
    diff_c = [c[i+1] - c[i] for i in range(n-1)]
    diff_t = [t[i+1] - t[i] for i in range(n-1)]
    if diff_c == diff_t:
        print("Yes")
    else:
        print("No")
