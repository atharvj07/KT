#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;

  vector<int> army(m + 1);
  for (int i = 0; i <= m; i++) {
    cin >> army[i];
  }

  int ans = 0;
  for (int i = 0; i < m; i++) {
    int diff = army[i] ^ army[m];
    int cnt = 0;
    while (diff > 0) {
      if (diff % 2 == 1) cnt++;
      diff /= 2;
    }
    if (cnt <= k) ans++;
  }

  cout << ans << endl;

  return 0;
}
