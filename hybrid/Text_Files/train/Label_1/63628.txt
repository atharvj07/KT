for _ in range(int(input())):
    n = int(input());a = list(map(int, input().split()));
    odd, odd_idx = [], []
    for i in range(n):
        if a[i]%2:
            odd.append(a[i])
            odd_idx.append(i)
    ok = True
    if n%2==0:
        if len(odd)%2:
            ok = False
    else:
        if len(odd)%2==0:
            ok = False
    if not ok:
        print(-1)
        continue
    ans = 0
    if len(odd)%2:
        ans+=1
        a[odd_idx.pop(0)], a[n-1] = a[n-1], a[odd_idx.pop(0)]
    while odd:
        ans+=1
        a[odd_idx.pop(0)], a[odd_idx.pop(0)] = a[odd_idx.pop(0)], a[odd_idx.pop(0)]
    print(ans)
