#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
vector<vector<ll>> dp;
ll pw[11];
ll n, k, s;
vector<ll> v;
ll solve(ll i, ll rem, ll cur) {
  if(i == n) { return cur == s; }
  if(dp[i][rem] != -1) { return dp[i][rem]; }
  dp[i][rem] = solve(i + 1, rem, cur) % mod;
  if(rem) {
    dp[i][rem] += solve(i + 1, rem - 1, cur + pw[v[i]]) % mod;
    dp[i][rem] %= mod;
  }
  return dp[i][rem];
}
int main() {
  cin >> n >> k >> s;
  v.resize(n);
  dp.resize(n, vector<ll>(k + 1, -1));
  for(ll &i : v) cin >> i;
  pw[0] = 1;
  for(ll i = 1; i <= 10; i++) {
    pw[i] = (pw[i - 1] * i) % mod;
  }
  cout << solve(0, k, 0) << '\n';
  return 0;
}
