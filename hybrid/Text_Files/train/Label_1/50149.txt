import java.util.*;

public class Main {
    static int[][] dist;
    static int[] path;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Integer>[] adj = new List[n+1];
        for (int i = 0; i <= n; i++)
            adj[i] = new ArrayList<Integer>();
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        int[] targets = new int[m];
        for (int i = 0; i < m; i++)
            targets[i] = sc.nextInt();
        dist = new int[n+1][n+1];
        path = new int[n+1];
        for (int i = 1; i <= n; i++) {
            Arrays.fill(path, -1);
            bfs(i, adj);
        }
        int minDist = Integer.MAX_VALUE;
        int start = 0;
        for (int i = 1; i <= n; i++) {
            int d = 0;
            for (int j : targets)
                d += dist[i][j];
            if (d < minDist) {
                minDist = d;
                start = i;
            }
        }
        System.out.println(start);
        System.out.println(minDist);
    }

    static void bfs(int s, List<Integer>[] adj) {
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        dist[s][s] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj[u]) {
                if (dist[s][v] == 0) {
                    dist[s][v] = dist[s][u] + 1;
                    path[v] = u;
                    q.add(v);
                }
            }
        }
    }
}
