#include <bits/stdc++.h>
using namespace std;

const int N = 2005, M = 505;
int n, m;
string s, p;
int f[N][M], g[N][M];

int main() {
    cin >> s >> p;
    n = s.size();
    m = p.size();
    for (int i = n; i >= 0; --i)
        for (int j = m; j >= 0; --j) {
            if (i == n)
                f[i][j] = 0;
            else if (j == m)
                f[i][j] = n - i;
            else
                f[i][j] = max(f[i + 1][j], f[i + 1][j + 1] + (s[i] == p[j]));
        }
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            if (s[i - 1] == p[j - 1])
                g[i][j] = g[i - 1][j - 1] + 1;
            g[i][j] = max(g[i][j], max(g[i - 1][j], g[i][j - 1]));
        }
    }
    for (int x = 0; x <= n; ++x)
        cout << max(f[x][0], g[n - x][m]) << ' ';
    cout << '\n';
    return 0;
}
