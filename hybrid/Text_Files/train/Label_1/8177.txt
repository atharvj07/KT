#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::string s;
  std::cin >> s;
  int g, t;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 'G') g = i;
    if (s[i] == 'T') t = i;
  }
  if ((t - g) % k == 0)
    std::cout << "YES" << std::endl;
  else
    std::cout << "NO" << std::endl;
  return 0;
}
