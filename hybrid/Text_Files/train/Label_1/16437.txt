#include <bits/stdc++.h>
using namespace std;

const int N = 300000 + 5;

int a[N];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  int p = 0;
  for (int i = 0; i < n; i++) {
    while (p < n && a[i] > p) {
      p++;
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
