#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi(x) vector<x>
#define pb push_back
#define all(x) begin(x), end(x)
#define sz(x) (int)(x).size()
#define endl '\n'
int main() {
  ll n;
  cin >> n;
  vi(ll) a(n), b(n + 1);
  for (ll& x : a) cin >> x;
  ll ans = LLONG_MAX;
  ll pos = -1;
  for (ll i = 0; i < n; i++) {
    for (ll j = 0; j < n; j++) { b[(j + i) % n] = a[(j + i) % n]; }
    ll sum = 0;
    for (ll j = 1; j < n; j++) sum += abs(b[j] - b[j - 1]);
    if (sum < ans) {
      ans = sum;
      pos = i;
    }
  }
  for (ll j = 0; j < n; j++) { cout << a[(j + pos) % n] << " "; }
}
