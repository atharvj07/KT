import sys

n, m = map(int, input().split())
participants = {}
for _ in range(n):
    surname, region, points = input().split()
    if region not in participants:
        participants[region] = []
    participants[region].append((surname, int(points)))

for region in sorted(participants):
    participants[region].sort(key=lambda x: x[1], reverse=True)
    if len(participants[region]) < 2 or participants[region][0][1] == participants[region][1][1]:
        print('?')
    else:
        print(' '.join(map(lambda x: x[0], participants[region][:2])))
