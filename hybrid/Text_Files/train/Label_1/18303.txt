#include <bits/stdc++.h>
#define nl "\n"
using namespace std;
typedef long long ll;
int a[1005], b[1005], n, t;
string s;
int main() {
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i <= 26; i++) a[i] = 0, b[i] = 0;
        bool f = 1;
        for (int i = 0; i < n; i++) {
            cin >> s;
            for (int j = 0; j < s.length(); j++) a[s[j] - 'a']++;
        }
        for (int i = 0; i < n; i++) {
            cin >> s;
            for (int j = 0; j < s.length(); j++) b[s[j] - 'a']++;
        }
        for (int i = 0; i <= 26; i++)
            if (a[i] != b[i]) f = 0;
        f ? cout << "YES" << nl : cout << "NO" << nl;
    }
    return 0;
}
