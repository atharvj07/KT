import sys
input = sys.stdin.readline

def modpow(base, p, MOD):
    if p == 0:
        return 1
    if p == 1:
        return base
    val = modpow(base, int(p / 2), MOD)
    val = (val * val) % MOD
    if p % 2 != 0:
        val = (val * base) % MOD
    return val

t, k = list(map(int, input().split()))
for _ in range(t):
    a, b = list(map(int, input().split()))
    if k == 1:
        print(b - a + 1)
        continue
    ways = modpow(2, (b - a + k) // k, 1000000007)
    ways = (ways - 1) % 1000000007
    if k % 2 == 0:
        ways = (ways + 1) % 1000000007
    ways = (ways * modpow(k, 1000000005, 1000000007)) % 1000000007
    print(ways)
