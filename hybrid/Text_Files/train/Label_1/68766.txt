import sys
sys.setrecursionlimit(100000)
n = int(input())
e = [[] for _ in range(n)]
for _ in range(n - 1):
    a, b = tuple(map(int, input().split()))
    a -= 1; b -= 1
    e[a].append(b)
    e[b].append(a)
vis = [False] * n
ans = [0] * n
dep = [0] * n
l = []
m = []
def dfs(x, p):
    vis[x] = True
    l.append(x)
    for i in e[x]:
        if vis[i] == False:
            dfs(i, x)
    m.append(x)
dfs(0, -1)
#print(l)
#print(m)
dif = []
for i in range(n):
    dif.append([l[i], m[i]])
dif.sort()
l = dif
ans[l[0][0]] = (0, 0)
#print(l)
for i in range(1, n):
    x, y = ans[l[i][0]]
    #print(x, y)
    if l[i][1] - l[i][0] == 1:
        ans[l[i][0]] = (x, y + 1)
        ans[l[i][1]] = (x + 1, y + 1)
    else:
        ans[l[i][0]] = (x + 1, y)
        ans[l[i][1]] = (x + 1, y - 1)
print("YES")
for i in range(n):
    print(*ans[i])
