import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), L = sc.nextInt(), a = sc.nextInt();
        int[] t = new int[n], l = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
            l[i] = sc.nextInt();
        }
        sc.close();
        int tb = 0, i = 0;
        while (i < n) {
            if (tb + a <= t[i]) {
                tb += a;
            } else if (tb < t[i]) {
                tb = t[i];
            }
            tb += l[i];
            i++;
        }
        int ans = tb <= L ? tb / a : tb / a - 1;
        System.out.println(ans);
    }
}
