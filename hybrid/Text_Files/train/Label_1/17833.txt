#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> dp(n + 1, -1);
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    if (dp[i - 1] != -1 && (dp[i - 1] + 1) % m == 0) dp[i] = dp[i - 1] + 1;
    if (i - 2 >= 0 && dp[i - 2] != -1 && (dp[i - 2] + 2) % m == 0) dp[i] = dp[i - 2] + 2;
  }
  cout << dp[n] << endl;
  return 0;
}
