#include <iostream>
#include <vector>

int main()
{
    int n, p, m;
    std::cin >> n >> p >> m;
    std::vector<std::pair<int, int>> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i].first >> a[i].second;
    }
    int cnt = 0, cur = 0;
    for (int i = 0, j = 0; i < m; i++) {
        cur -= p;
        while (j < n && a[j].first == i) {
            cur += a[j].second;
            j++;
        }
        if (cur < 0) {
            cnt++;
        }
    }
    std::cout << cnt << std::endl;
    return 0;
}
