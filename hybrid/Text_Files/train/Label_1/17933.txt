#include <bits/stdc++.h>
using namespace std;
int a, b, n;
vector<int> ans;

int main() {
  cin >> n >> a >> b;
  int x = (n + 1) / 2;
  int y = n - x;
  if ((a * b) / 2 < x || (a * b) / 2 < y) return puts("-1"), 0;
  for (int i = 0; i < a; i++) {
    for (int j = 0; j < b; j++) {
      if (i % 2 == 0 && j % 2 == 0 && x) {
        ans.push_back(x--);
      } else if (i % 2 == 1 && j % 2 == 1 && y) {
        ans.push_back(y--);
      } else {
        ans.push_back(0);
      }
    }
  }
  for (int i = 0; i < a * b; i++) { cout << ans[i] << " "; }
  return 0;
}
