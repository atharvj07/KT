#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
const int mxN = 1e5;
int a[mxN + 5], dp[mxN + 5];
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr); cout.tie(nullptr);
  int n, k; cin >> n >> k;
  int sum = 0;
  for(int i = 1; i <= n; i++)
    cin >> a[i], sum += a[i];
  dp[0] = dp[1] = 0;
  for(int i = 2; i <= n; i++) {
    dp[i] = LLONG_MAX;
    for(int j = max(1LL * 1, i - k); j < i; j++)
      dp[i] = min(dp[i], dp[j] + a[i]);
  }
  cout << dp[n] << '\n';
}
