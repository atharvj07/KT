import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0 ; i < n ; i++) a[i] = sc.nextInt();
        Arrays.sort(a);
        int[] dp = new int[n]; //Stores the length of the longest increasing subsequence ending with any index i.
        for(int i = 0 ; i < n ; i++) dp[i] = 1;
        String res = ""; //Stores the characters L or R which denotes that we choose left or right element while constructing the increasing subsequence.
        int max_index = 0; //Stores the index of the last element of the longest increasing subsequence.
        for(int i = 1 ; i < n ; i++) {
            for(int j = 0 ; j < i ; j++) {
                if(a[i] > a[j] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    if(dp[i] > dp[max_index]) max_index = i;
                }
            }
        }
        int last = max_index;
        int len = dp[max_index];
        for(int i = n - 1 ; i >= 0 ; i--) {
            if(len == dp[i] && a[i] < a[last]) {
                last = i;
                len--;
            }
        }
        for(int i = 0 ; i < n ; i++) {
            if(a[i] == a[last]) {
                if(i < n / 2) res += "L";
                else res += "R";
                last++;
            }
        }
        System.out.println(dp[max_index]);
        System.out.println(res);
    }
}
