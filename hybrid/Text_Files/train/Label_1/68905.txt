import sys
input=sys.stdin.readline
t=int(input())
for _ in range(t):
    n,p=map(int,input().split())
    a=list(map(int,input().split()))
    g=dict()
    for i in a:
        g.setdefault(i,[]).append(len(g))
    mn=[0]*(n+2)
    for i in range(n-1,-1,-1):
        for x in g[a[i]]:
            if x+1<n:mn[i]=max(mn[i],mn[x+1])
        mn[i]=max(mn[i],a[i])
    ans=0
    mn[n]=mn[n-1]
    for i in range(n-1):
        ans+=min(mn[i+1],p)
    print(ans)
