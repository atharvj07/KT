import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d1 = sc.nextInt();
        int d2 = sc.nextInt();
        int[][] points = new int[n*n][2];
        int index = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == 0 && j == 0) {
                    continue;
                }
                if (i == 0 || j == 0) {
                    points[index][0] = i + j;
                    points[index][1] = 0;
                    index++;
                } else if (i == n - 1 || j == n - 1) {
                    points[index][0] = i + j - 1;
                    points[index][1] = 2 * n - 2;
                    index++;
                } else {
                    points[index][0] = i + j;
                    points[index][1] = i + j - 1;
                    index++;
                }
            }
        }
        for (int[] point : points) {
            System.out.println(point[0] + " " + point[1]);
        }
    }
}
