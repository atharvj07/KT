#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef long long ll;

vector<int> a;
int n;
const int maxn = 2e5 + 5, inf = 1e18;
int dp[maxn][4];

int f(int pos, int mask) {
  if (pos == n + 1) { return mask == 3 ? 0 : inf; }
  int &ret = dp[pos][mask];
  if (ret != -1) return ret;
  ret = inf;
  ret = min(ret, f(pos + 1, mask));
  if (mask == 0) { ret = min(ret, a[pos] + f(pos + 1, 1)); }
  if (mask == 1) { ret = min(ret, a[pos] + f(pos + 1, 2)); }
  if (mask == 2) { ret = min(ret, a[pos] + f(pos + 1, 3)); }
  return ret;
}

main() {
  cin >> n;
  a.resize(n + 1);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  memset(dp, -1, sizeof dp);
  cout << f(1, 0) << '\n';
}
