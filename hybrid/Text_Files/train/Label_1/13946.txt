#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;

int a[N], b[N], pos[N];

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= k; i++) cin >> a[i];
  for (int i = 1; i <= k; i++) pos[a[i]] = i;
  int ans = 0;
  for (int q = 1; q <= n; q++) {
    int sum = 0;
    for (int j = 1; j <= m; j++) {
      cin >> b[j];
      sum += pos[b[j]];
      for (int i = pos[b[j]]; i > 1; i--) a[i] = a[i - 1], pos[a[i]] = i;
      a[1] = b[j], pos[b[j]] = 1;
    }
    ans += sum;
  }
  cout << ans << endl;
}
