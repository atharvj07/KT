import sys
INF = sys.maxsize

def Dijkstra(s, n, edges):
    dist = [INF] * (n + 1)
    dist[s] = 0
    pq = [(0, s)]
    par = [0] * (n + 1)
    while pq:
        cost, curr = pq.pop(0)
        for u, w in edges[curr]:
            if cost + w < dist[u]:
                dist[u] = cost + w
                par[u] = curr
                pq.append((cost + w, u))
    return dist, par

def dfs(u, d, par, mx):
    if u == 1:
        ans.add(mx)
        return
    dfs(par[u], d, par, max(mx, d[u]))

n, m, k = map(int, input().split())
edges = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b, c = map(int, input().split())
    edges[a].append((b, c))
    edges[b].append((a, c))

d, par = Dijkstra(1, n, edges)
ans = set()
if k == 0:
    print(k)
else:
    for i in range(2, n + 1):
        if d[i] != INF:
            dfs(i, d, par, d[i])
    
    print(len(ans))
    for i in ans:
        print(i, end = ' ')
