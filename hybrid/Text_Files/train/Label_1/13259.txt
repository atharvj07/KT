#include <iostream>
#include <string>
using namespace std;
int main() {
  int a, b, x, t = 1, r = 1;
  cin >> a >> b >> x;
  string ans(a + b, '1');
  if (x) {
    ans[t] = '0';
    t += 2;
  } else {
    ans[b] = '0';
  }
  if (x >= a) {
    ans[t] = '1';
    t += 2;
    x -= a;
  } else {
    t += 2 * x;
  }
  for (int i = b; i >= 0; i--) {
    if (x == 0) break;
    ans[r] = '0';
    r += 2;
    x--;
  }
  if (x) ans[r] = '0';
  cout << ans << endl;
}
