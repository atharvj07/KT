import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long L = sc.nextLong();
            long v = sc.nextLong();
            long l = sc.nextLong();
            long r = sc.nextLong();
            long ans = (L / v) - (l - 1) / v - (L - r) / v;
            System.out.println(ans);
        }
    }
}
