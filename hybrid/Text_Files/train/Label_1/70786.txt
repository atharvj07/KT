def check_sequence(p, a, l, r):
    cnt = [0] * len(p)
    for i in range(l - 1, r):
        cnt[a[i] - 1] += 1
    for i in range(len(p)):
        if cnt[i] == 0:
            return False
    return True

n, m, q = map(int, input().split())
p = list(map(int, input().split()))
a = list(map(int, input().split()))
for i in range(q):
    l, r = map(int, input().split())
    print(int(check_sequence(p, a, l, r)))
