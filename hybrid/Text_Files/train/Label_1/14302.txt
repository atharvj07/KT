#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    if (n % 2 == 0) {
      cout << "NO" << endl;
    } else if (n <= k && n % 2 == 1) {
      cout << "YES" << endl;
    } else if (n > k * 2) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
    }
  }
}
