import sys
for t in range(int(input())):
    n = int(input())
    b = sorted([int(x) for x in input().split()])
    a1, a2 = [], []
    for x in b:
        if not a1 or a1[-1] >= x: a1.append(x)
        else: a2.append(x)
    if len(a1) == len(b):
        for x in a1[::-1]: a2.append(x)
    if len(a2) > len(a1): print(-1)
    else:
        for x in b:
            if a1 and a1[0] == x: a1.pop(0)
            elif a2 and a2[0] == x: a2.pop(0)
        print(*[x for x in a1] + [x for x in a2])
