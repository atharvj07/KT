import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] dist = new int[n];
        for (int i = 0; i < n; i++) {
            dist[i] = input.nextInt();
        }
        int maxLeft = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (dist[i] >= maxLeft) {
                maxLeft = dist[i];
            } else {
                System.out.println("no");
                return;
            }
        }
        int maxRight = 0;
        for (int i = 0; i < n; i++) {
            if (dist[i] >= maxRight) {
                maxRight = dist[i];
            } else {
                System.out.println("no");
                return;
            }
        }
        System.out.println("yes");
    }
}
