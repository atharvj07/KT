import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int k = scan.nextInt();
        int[] c = new int[k * n];
        for (int i = 0; i < c.length; i++) {
            c[i] = scan.nextInt();
        }
        int[] f = new int[n];
        for (int i = 0; i < f.length; i++) {
            f[i] = scan.nextInt();
        }
        int[] h = new int[k];
        for (int i = 0; i < h.length; i++) {
            h[i] = scan.nextInt();
        }
        int[][] dp = new int[n + 1][k + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= k; j++) {
                dp[i][j] = dp[i - 1][j];
                for (int l = 0; l < c.length; l++) {
                    if (f[i - 1] == c[l] && dp[i - 1][j - 1] + h[j - 1] > dp[i][j]) {
                        dp[i][j] = dp[i - 1][j - 1] + h[j - 1];
                    }
                }
            }
        }
        System.out.println(dp[n][k]);
    }
}
