n = int(input())
a = list(map(int, input().split()))

def min_arithmetic_progressions(n, a):
    # Function to check if three points can form an arithmetic progression
    def is_progression(x, y, z):
        return (y - x == z - y) or (x == -1 and z == -1) or (x == -1 and y == -1) or (y == -1 and z == -1) or (x == -1 and (z - y) % 2 == 0) or (z == -1 and (y - x) % 2 == 0) or (y == -1 and (z - x) % 3 == 0)
    
    # Count of arithmetic progressions
    count = 1
    i = 0
    while i < n - 1:
        if a[i] != -1 and a[i + 1] != -1 and a[i] == a[i + 1]:
            count += 1
            i += 1
        elif i + 2 < n and not is_progression(a[i], a[i + 1], a[i + 2]):
            count += 1
            i += 1
        else:
            i += 1
    return count

print(min_arithmetic_progressions(n, a))
