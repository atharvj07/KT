import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        long[] diff = new long[n];
        for (int i = 0; i < n; i++) {
            diff[i] = b[i] - a[i];
        }
        long gcd = diff[0];
        for (int i = 1; i < n; i++) {
            gcd = gcd(gcd, diff[i]);
        }
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            if (diff[i] % gcd != 0) {
                possible = false;
            }
        }
        if (!possible) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += diff[i] / gcd;
        }
        System.out.println(ans);
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
