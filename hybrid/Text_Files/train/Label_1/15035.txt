#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
#include <cmath>
using namespace std;
#define ll long long
const int MAXP = 1000006;
vector<int> pr;
map<int, int> c;
bool isPrime(int n) {
	if (n <= 1) return false;
	for (ll i = 2; i * i <= n; i++) {
		if (n % i == 0) return false;
	}
	return true;
}
int main() {
	ll n, p; cin >> n >> p;
	bool flag = isPrime(p);
	if (!flag) {
		cout << -1;
		return 0;
	}
	map<int, int> cou;
	ll ans = 0;
	for (int i = 0; i < n; ++i) {
		ll x; cin >> x;
		if (isPrime(x) && x != p) { pr.push_back(x); c[x]++; ans++; }
		if (!isPrime(x)) {
			for (int j = 2; j * j <= x; ++j) {
				while (x % j == 0) {
					cou[j]++;
					x /= j;
					if (j == p) flag = false;
				}
			}
			if (x != 1) {
				cou[x]++;
				if (x == p) flag = false;
			}
		}
	}
	if (!flag) ans = 0;
	else {
		for (int i = 0; i < pr.size(); ++i) {
			if (cou[pr[i]] == 0) ans++;
		}
	}
	cout << ans << endl;
	map<int, int> ::iterator it;
	for (it = cou.begin(); it != cou.end(); ++it) {
		if (it->second == 0) cout << it->first << " ";
	}
	for (int i = 0; i < pr.size(); ++i) {
		if (cou[pr[i]] == 0) cout << pr[i] << " ";
	}
}
