#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 1e5 + 10;
bitset<N> vis;
int a[N];
int32_t main() {
  int n, x;
  cin >> n >> x;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a, a + n);
  int ans = n;
  for (int i = 0; i < n; i++) {
    if (vis[a[i]]) continue;
    vis[a[i]] = true;
    ans--;
    if (a[i] >= x) break;
  }

  if (vis[x] == false) ans++;

  cout << ans << endl;

  return 0;
}
