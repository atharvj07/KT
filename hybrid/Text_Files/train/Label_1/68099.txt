def solve_magic_square(notes):
    # Calculate the sum of the side (non-diagonal) elements of the middle row
    # This is twice the sum that each row, column, and diagonal must have
    target_sum = notes[0][1] + notes[0][2] + notes[1][0] + notes[1][2] + notes[2][0] + notes[2][1]
    # Since each number appears twice in the sum, divide by 2 to get the target sum for each row, column, and diagonal
    target_sum //= 2

    # Calculate the missing diagonal elements
    notes[0][0] = target_sum - (notes[0][1] + notes[0][2])
    notes[1][1] = target_sum - (notes[1][0] + notes[1][2])
    notes[2][2] = target_sum - (notes[2][0] + notes[2][1])

    return notes

notes = [list(map(int, input().split())) for _ in range(3)]
solution = solve_magic_square(notes)
for row in solution:
    print(' '.join(map(str, row)))
