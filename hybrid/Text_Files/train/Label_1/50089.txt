import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int b1 = sc.nextInt(), q = sc.nextInt(), l = sc.nextInt(), m = sc.nextInt();
    int[] a = new int[m];
    for (int i = 0; i < m; i++) a[i] = sc.nextInt();
    int ans = 0;
    boolean inf = false;
    for (int bi = b1; !inf && (Math.abs(bi) <= l); bi *= q) {
      boolean bad = false;
      for (int i = 0; i < m; i++) if (bi == a[i]) bad = true;
      if (!bad) ans++;
      else inf = true;
    }
    System.out.println(inf ? "inf" : ans);
  }
}
