def minLength(n, text):
    words = text.split()
    dp = [len(text)] * (n + 1)
    for i in range(1, n + 1):
        for j in range(i - 1, -1, -1):
            if words[i - 1] == words[j]:
                dp[i] = min(dp[i], dp[j] + i - j - 1)
            else: dp[i] = min(dp[i], dp[i-1])
    return dp[n]

n = int(input())
text = input()
print(minLength(n, text))
