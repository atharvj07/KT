import java.util.Scanner;

public class StaircaseBoxes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] stairs = new int[n];
        for (int i = 0; i < n; i++) {
            stairs[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        for (int i = 0; i < m; i++) {
            int width = scanner.nextInt();
            int height = scanner.nextInt();

            int landingHeight = findLandingHeight(stairs, width, height);
            System.out.println(landingHeight);
        }
    }

    private static int findLandingHeight(int[] stairs, int width, int height) {
        int landingHeight = 0;
        for (int i = 0; i < width; i++) {
            landingHeight = Math.max(landingHeight, stairs[i]);
        }
        return landingHeight + height;
    }
}
