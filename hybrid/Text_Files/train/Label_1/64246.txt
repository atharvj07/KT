import sys

for _ in range(int(input())):
    a = list(input())
    b = list(input())
    c = list(input())
    for i in range(len(a)):
        need = a[i] if a[i] == b[i] else c[i]
        if need != c[i]:
            a[i], c[i] = c[i], a[i]
    if a == b:
        print("YES")
    else:
        print("NO")
