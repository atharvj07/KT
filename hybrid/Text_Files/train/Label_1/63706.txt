import sys

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    x, y, z = map(int, input().split())

    if x == y and y == z and x == z:
        print("YES")
        print(x, x, x)
    elif x == y:
        if x < z:
            print("NO")
        else:
            print("YES")
            print(x, z, z)
    elif y == z:
        if y < x:
            print("NO")
        else:
            print("YES")
            print(x, x, y)
    else:
        if x < z or y < z:
            print("NO")
        else:
            print("YES")
            print(x, y, z)
