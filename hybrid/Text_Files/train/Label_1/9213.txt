#include <vector>
#include <iostream>
#include <limits>
#include <algorithm>
using namespace std;

const int INF = numeric_limits<int>::max() / 2;

struct Edge {
    Edge(int to, int w) : to(to), w(w) {}
    int to, w;
};
int n, c;
vector<vector<Edge>> G;
vector<int> d;

void Dijkstry() {
    d[1] = 0;
    for(int i = 2; i <= n; ++i) {
        d[i] = INF;
    }
    vector<bool> used(n + 1, false);
    for(int i = 1; i <= n; ++i) {
        int v = -1;
        for(int j = 1; j <= n; ++j) {
            if(!used[j] && (v == -1 || d[v] > d[j])) {
                v = j;
            }
        }
        if(v == -1) break;
        used[v] = true;
        for(auto [to, w] : G[v]) {
            d[to] = min(d[to], d[v] + w);
        }
    }
}

int main() {
    cin >> n >> c;
    d.resize(n + 1, INF);
    G.resize(n + 1);
    vector<int> a(n), b(n);
    for(int i = 1; i <= n-1; ++i) {
        cin >> a[i];
    }
    for(int i = 1; i <= n-1; ++i) {
        cin >> b[i];
        G[i].emplace_back(i+1, a[i]);
        G[i+1].emplace_back(i, a[i]);
        G[i].emplace_back(i+1, b[i] + c);
        G[i+1].emplace_back(i, b[i] + c);
    }
    
    Dijkstry();
    for(int i = 1; i <= n; ++i) {
        cout << d[i] << ' ';
    }
    return 0;
}
