import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) throws java.lang.Exception {
        Scanner in = new Scanner(System.in);
        long f = in.nextLong(), T = in.nextLong(), t0 = in.nextLong();
        long a1 = in.nextLong(), t1 = in.nextLong(), p1 = in.nextLong();
        long a2 = in.nextLong(), t2 = in.nextLong(), p2 = in.nextLong();
        in.close();
        long minTime1 = f * t0, minTime2 = f * t1, minTime3 = f * t2;
        long minCost1 = (minTime1 + t0 - 1) / t0, minCost2 = (minTime2 + t1 - 1) / t1, minCost3 = (minTime3 + t2 - 1) / t2;
        long pack1 = p1 / a1, pack2 = p2 / a2;
        long res;

        if (a1 >= f) {
            res = p1;
        } else {
            long rem = f - a1, cost = p1;
            long tm1 = rem * t1;
            long tm2 = rem * t0;
            long extra1 = (tm1 + t1 - 1) / t1, extra2 = (tm2 + t0 - 1) / t0;
            long extraCost1 = extra1 * pack1, extraCost2 = extra2 * pack2;
            res = cost + Math.min(extraCost1, extraCost2);
        }
        T -= minTime1;
        if (T >= 0) {
            res = Math.min(res, minCost1 * p1);
        }
        T -= minTime2;
        if (T >= 0) {
            res = Math.min(res, minCost2 * p1);
        }
        T -= minTime3;
        if (T >= 0) {
            res = Math.min(res, minCost3 * p2);
        }
        if (res >= 1e9) {
            res = -1;
        }
        System.out.println(res);
    }
}
