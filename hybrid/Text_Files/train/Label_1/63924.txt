n = int(input())
a = list(map(int, input().split()))

# Count the occurrence of each height
height_counts = {}
for height in a:
    if height in height_counts:
        height_counts[height] += 1
    else:
        height_counts[height] = 1

# Sort heights to process in ascending order
sorted_heights = sorted(height_counts.keys())

# Dynamic programming approach to find the maximum number of people in a balanced circle
dp = [0] * (max(sorted_heights) + 1)
for height in sorted_heights:
    dp[height] = height_counts[height]
    if height > 1:
        # Choose the maximum between keeping the current sequence or appending to the previous height's sequence
        dp[height] = max(dp[height], dp[height - 1] + height_counts[height])

# Backtrack to find the actual sequence
result = []
current_height = max(dp)
while current_height > 0:
    if current_height in sorted_heights and dp[current_height] - height_counts[current_height] == dp[current_height - 1]:
        result.extend([current_height] * height_counts[current_height])
        current_height -= 1
    else:
        current_height -= 1

# Adjusting the result for correct output
result.sort()
print(len(result))
print(' '.join(map(str, result)))
