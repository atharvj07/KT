#include <iostream>

using namespace std;

int main() {
  int a, b, c, l;
  cin >> a >> b >> c >> l;
  int ans = 0;
  for (int i = 0; i <= l; i++) {
    for (int j = 0; j <= l - i; j++) {
      int k = l - i - j;
      if (a + i > 0 && b + j > 0 && c + k > 0) {
        if (a + i < b + j + c + k && b + j < a + i + c + k && c + k < a + i + b + j)
          ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
