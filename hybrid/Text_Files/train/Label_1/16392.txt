#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int k;
    cin >> k;
    string n;
    cin >> n;
    vector<int> ans;
    int cnt = 0;
    for (auto &c : n) {
      int x = (c - '0');
      if (cnt == 0 && x == 1) {
        continue;
      } else {
        if (x != 0) {
          ans.push_back(x);
          cnt++;
        }
      }
    }
    cout << ans.size() << endl;
    for (auto &it : ans) {
      cout << it;
    }
    cout << endl;
  }
  return 0;
}
