#include <bits/stdc++.h>
using namespace std;
#define MAXN 105

int n, a[MAXN], sum[MAXN], vis[MAXN];
bool flag;

void dfs(int x){
    if(flag) return;
    vis[x] = 1;
    for(int i = 0; i < n; i++){
        if(vis[abs(x - a[i])] == 0){
            if(i%2 == n%2){
                dfs(abs(x - a[i]));
            }
            else flag = true;
        }
    }
}

int main(){
    cin >> n;
    for(int i = 0; i < n; i++){
        cin >> a[i];
        sum[i] = a[i];
    }
    sort(sum, sum + n);
    for(int i = 1; i < n; i++){
        if(sum[i] == sum[i - 1]){
            cout << "Bob" << endl;
            return 0;
        }
    }
    dfs(a[0]);
    if(flag) cout << "Bob" << endl;
    else cout << "Alice" << endl;
    return 0;
}
