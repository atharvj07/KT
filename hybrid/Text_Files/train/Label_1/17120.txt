#include <bits/stdc++.h>
using namespace std;
bool ok[26], found;
int main() {
    int k;
    cin >> k;
    string q;
    cin >> q;
    vector<string> ans;
    for (int i = 0, j = 0; i < (int)q.size(); i = j) {
        string s;
        bool f[26] = {};
        for (j = i; j < (int)q.size() && !f[q[j] - 'a']; j++) {
            f[q[j] - 'a'] = 1;
            s += q[j];
        }
        if (k == 0 || !f[q[i] - 'a']) {
            found = 1;
            break;
        }
        ans.push_back(s);
        ok[q[i] - 'a'] = 1;
        k--;
    }
    if (found) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (auto x : ans) cout << x << endl;
    }
}
