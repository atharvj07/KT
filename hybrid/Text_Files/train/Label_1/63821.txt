import sys
input=lambda:sys.stdin.readline().rstrip()
sys.setrecursionlimit(10000)

def dfs(x,y,pre):
    global ans,vis
    for dx,dy in [[0,1],[0,-1],[1,0],[-1,0]]:
        nx,ny=x+dx,y+dy
        if 1<=nx<=n and 1<=ny<=m and not vis[nx][ny][pre] and not a[nx][ny]:
            vis[nx][ny][pre]=True
            dfs(nx,ny,(pre+1)%4)
            ans+=list('LRUD')[pre]
            vis[nx][ny][(pre+1)%4]=False
n,m,k=map(int,input().split())
a=[list(input())for _ in range(n)]
for i in range(n):
    for j in range(m):
        if a[i][j]=='X':
            x,y=i,j
vis=[[[False]*4 for _ in range(m)]for _ in range(n)]
ans=""
dfs(x,y,0)
print(ans if len(ans)==k else "IMPOSSIBLE")
