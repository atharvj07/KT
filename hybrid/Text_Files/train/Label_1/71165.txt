import sys

input = sys.stdin.readlines()
n, m = [int(x) for x in input[0].split()]
w = [int(x) for x in input[1].split()]
roads = [[] for i in range(n+1)]
for i in range(m):
    u, v = [int(x) for x in input[i+2].split()]
    roads[u].append(v)
    roads[v].append(u)
s = int(input[m+2])

dp = [0] * (n+1)
vis = [False] * (n+1)

def dfs(u):
    vis[u] = True
    dp[u] = w[u-1]
    for v in roads[u]:
        if vis[v] == False:
            dfs(v)
            dp[u] = max(dp[u], dp[v])

dfs(s)
print(max(dp))
