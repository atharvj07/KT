#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 2e5 + 5;

vector<pair<long long, int>> adj[MAXN];
int subtree[MAXN];
long long dist[MAXN];

void dfs(int v, int p, long long d) {
    subtree[v] = 1;
    dist[v] = d;
    for (auto u : adj[v]) {
        if (u.first != p) {
            dfs(u.first, v, d + u.second);
            subtree[v] += subtree[u.first];
        }
    }
}

long long dfs2(int v, int p, long long d, long long q) {
    long long ans = 0;
    for (auto u : adj[v]) {
        if (u.first != p) {
            if (d + u.second <= q) {
                ans += subtree[u.first];
            }
            ans += dfs2(u.first, v, d + u.second, q);
        }
    }
    return ans;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n - 1; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    dfs(1, 1, 0);
    for (int i = 0; i < m; i++) {
        long long q;
        cin >> q;
        cout << dfs2(1, 1, 0, q) << " ";
    }
}
