import sys

n, m = map(int, input().split())
p = list(map(int, input().split()))
foe = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, input().split())
    foe[a].append(b)
    foe[b].append(a)
vis = [0] * (n+1)
tot = 0
for i in range(1, n+1):
    if vis[i]: continue
    vis[i] = 1
    mx = i
    q = [i]
    while q:
        x = q.pop()
        tot += x-i+1
        for y in foe[x]:
            if vis[y] == 0:
                q.append(y)
                mx = max(mx, y)
                vis[y] = 1
    for j in range(i+1, mx+1):
        vis[j] = 2
print(tot)
