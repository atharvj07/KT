import java.util.*;
import java.lang.*;

class Main {
    static class Point {
        int x, y;
        Point(int x, int y) { this.x = x; this.y = y; }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Point[] points = new Point[n];
        int[] dp = new int[n];
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            points[i] = new Point(x, y);
        }

        Arrays.sort(points, (Point p1, Point p2) -> p1.x - p2.x);
        int minCost = Integer.MAX_VALUE;
        for (int endpoint = 0; endpoint < n; endpoint++) {
            // dp[i] = minimum cost to connect the i-th point to the tree
            Arrays.fill(dp, Integer.MAX_VALUE);
            dp[endpoint] = 0;

            for (int i = endpoint + 1; i < n; i++) {
                for (int j = i - 1; j >= endpoint; j--) {
                    if (points[j].x < points[i].x && points[j].y > points[i].y) {
                        dp[i] = Math.min(dp[i], dp[j] + points[i].x - points[j].x + points[j].y - points[i].y);
                    }
                }
            }

            for (int i = endpoint; i < n; i++) {
                minCost = Math.min(minCost, dp[i]);
            }
        }

        System.out.println(minCost);
    }
}
