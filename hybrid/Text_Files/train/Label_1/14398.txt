#include <bits/stdc++.h>
#define LL long long
#define pb push_back
using namespace std;

LL arr[ 200000 ];
bool isPrime[ 200000 ];
pair<bool, pair<LL, LL> > primeFactors[ 200000 ];
vector<pair<LL, LL> > primeFactorList;

void sieve() {
    isPrime[ 0 ] = isPrime[ 1 ] = false;
    for( LL i = 2; i <= 199999; ++i ) {
        if( isPrime[ i ] == false ) {
            if( i * i > 199999 ) continue;
            for( LL j = i * i; j <= 199999; j += i ) isPrime[ j ] = false;
        }
    }
}

LL getPrime( LL x ) {
    for( LL i = 2; i * i <= x; ++i ) {
        if( x % i == 0 ) {
            if( isPrime[ i ] ) {
                if( isPrime[ x / i ] ) {
                    return x / i;
                }
            }
            return i;
        }
    }
    return -1;
}

void getPrimeFactors( LL x ) {
    while( x > 1 ) {
        LL now = getPrime( x );
        arr[ now ]++;
        primeFactorList.pb( { now, arr[ now ] } );
        x /= now;
    }
}

int main() {
    memset( isPrime, true, sizeof isPrime );
    sieve();
    LL n, t; cin >> t;
    while( t-- ) {
        cin >> n;
        bool hasPrime = false;
        primeFactorList.clear();
        fill( arr, arr + 200000, 0 );
        getPrimeFactors( n );
        for( pair<LL, LL> pr : primeFactorList ) {
            if( pr.second > 1 ) hasPrime = true;
        }
        if( primeFactorList.size() >= 3 || hasPrime ) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
