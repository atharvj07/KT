#include<bits/stdc++.h>
using namespace std;
int n,q,a[300005],tr[51],ans[300005];
int main()
{
    cin>>n>>q;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=q;i++)cin>>tr[i];
    for(int i=q;i>=1;i--)
    {
        for(int j=1;j<=n;j++)
        {
            if(a[j]==tr[i])
            {
                ans[i]=j;
                swap(a[j],a[1]);
                break;
            }
        }
    }
    for(int i=1;i<=q;i++)cout<<ans[i]<<"\n";
    return 0;
}
