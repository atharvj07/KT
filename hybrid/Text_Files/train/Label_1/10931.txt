#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m, k;
  std::cin >> n >> m >> k;
  std::vector<int> blocked(m);
  for (int i = 0; i < m; ++i) {
    std::cin >> blocked[i];
  }
  std::vector<int> costs(k);
  for (int i = 0; i < k; ++i) {
    std::cin >> costs[i];
  }
  std::sort(blocked.begin(), blocked.end());
  std::sort(costs.begin(), costs.end());
  int ans = 0, cur = 0;
  for (int i = 0; i < m; ++i) {
    int need = blocked[i] - cur - 1;
    if (need > 0) {
      int need_cost = need * costs[0];
      if (ans == 0 || need_cost < ans) {
        ans = need_cost;
      }
    }
    cur = blocked[i];
  }
  int need = n - cur - 1;
  if (need > 0) {
    int need_cost = need * costs[0];
    if (ans == 0 || need_cost < ans) {
      ans = need_cost;
    }
  }
  if (ans == 0) {
    ans = -1;
  }
  std::cout << ans << std::endl;
  return 0;
}
