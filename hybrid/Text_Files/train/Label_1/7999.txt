#include<bits/stdc++.h>
typedef long long LL;
using namespace std;
LL read(){
	LL a=0,f=1;
	char c=getchar();
	while(!isdigit(c)){
		if(c=='-') f=-1;
		c=getchar();
	}
	while(isdigit(c)){
		a=a*10+c-'0';
		c=getchar();
	}
	return a*f;
}
int n,k;
struct data{
	int a[4];
	data(int aa=0,int bb=0,int cc=0,int dd=0){
		a[0]=aa;a[1]=bb;a[2]=cc;a[3]=dd;
	}
	data operator&(const data &b)const{
		data c;
		for(int i=0;i<=3;i++) c.a[i]=a[i]&b.a[i];
		return c;
	}
}p[101000];
char s[101000];
int cnt[30];
int main(){
	n=read();k=read();
	for(int i=1;i<=n;i++){
		scanf("%s",s);
		int aa,bb,cc,dd;
		sscanf(s,"%d.%d.%d.%d",&aa,&bb,&cc,&dd);
		p[i]=data(aa,bb,cc,dd);
	}
	data ans;
	int flag=0;
	for(int i=0;i<=29;i++){
		memset(cnt,0,sizeof(cnt));
		for(int j=1;j<=n;j++) cnt[p[j].a[i>>2]>>(i&3)]++;
		if(cnt[k]!=0){
			int aa=cnt[k];
			for(int j=0;j<=255;j++) if(cnt[j]==aa) ans.a[i>>2]|=(1<<((i&3)))*j;
			flag=1;
		}
	}
	int aa,bb,cc,dd;
	if(flag) printf("%d.%d.%d.%d\n",ans.a[0],ans.a[1],ans.a[2],ans.a[3]);
	else puts("-1");
	return 0;
}
