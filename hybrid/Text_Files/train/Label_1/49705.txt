import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        if (a[0] == a[n - 1]) {
            System.out.println(0 + " " + a[0]);
        } else {
            System.out.println(1 + " " + (a[n - 1] - a[0]));
        }
    }
}
