import java.util.*;

public class Main {

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();
    int[][] teams = new int[n][2];
    for (int i = 0; i < n; i++) {
      teams[i][0] = input.nextInt();
      teams[i][1] = input.nextInt();
    }
    Arrays.sort(teams, (a, b) -> {
      int cmp = Integer.compare(b[0], a[0]);
      return cmp != 0 ? cmp : Integer.compare(a[1], b[1]);
    });

    int count = 0;
    int prevScore = -1;
    for (int i = 0; i < n; i++) {
      if (prevScore != teams[i][0] * 1000 + teams[i][1]) {
        prevScore = teams[i][0] * 1000 + teams[i][1];
        count++;
      }
      if (count == k) {
        System.out.println(i + 1);
        return;
      }
    }
  }
}
