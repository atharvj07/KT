import sys
input = sys.stdin.readline

m, k = map(int, input().split())
d = list(map(int, input().split()))
s = list(map(int, input().split()))
sum = 0
ans = 0
for i in range(0, m):
    if (sum + s[i]) >= d[i]:
        sum = sum + s[i] - d[i]
    else:
        ans += k
        sum = sum + s[i] - d[i]
    ans += d[i]
print(ans)
