import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String s = sc.next();
    StringBuilder sb = new StringBuilder();
    int x = 0;
    int y = 0;
    for (int i = 0; i < n; i++) {
      char c = s.charAt(i);
      if (c >= 'a' && c <= 'm') {
        x++;
      } else if (c >= 'n' && c <= 'z') {
        x--;
      } else if (c >= 'A' && c <= 'M') {
        y++;
      } else if (c >= 'N' && c <= 'Z') {
        y--;
      }
    }
    if (x == 0 && y == 0) {
      sb.append('0');
    } else if (x > 0) {
      for (int i = 0; i < x; i++) {
        sb.append('a');
      }
    } else if (x < 0) {
      for (int i = 0; i < -x; i++) {
        sb.append('n');
      }
    } else if (y > 0) {
      for (int i = 0; i < y; i++) {
        sb.append('A');
      }
    } else if (y < 0) {
      for (int i = 0; i < -y; i++) {
        sb.append('N');
      }
    }
    System.out.println(sb.length());
    System.out.println(sb.toString());
  }
}
