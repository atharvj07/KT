#include <iostream>
#include <string>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    bool ok = true, ok2 = true;
    int open = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '(')
            open++;
        else {
            if (open == 0) {
                ok = false;
                if (ok2)
                    ok2 = false;
            } else
                open--;
        }
    }
    if (open != 0)
        ok = false;
    if (ok)
        cout << "Yes" << endl;
    else if (ok2)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;
}
