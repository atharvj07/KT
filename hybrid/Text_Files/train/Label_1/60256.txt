n, k = map(int, input().split())
s = input()
mn = 999999999999999999
mn2 = 999999999999999999
ls = {}
for i in range(26):
    ls[chr(ord('a') + i)] = []
for i in range(n):
    ls[s[i]].append(i)
if k > n * (n + 1) // 2:
    mn = -1
for i in range(26):
    a = ls[chr(ord('a') + i)]
    mn2 = min(mn2, n - len(a))
    for j in range(len(a)):
        for d in range(j + 1, len(a)):
            mn = min(mn, n - (d - j + 1))
if mn == 999999999999999999:
    mn = mn2
if mn == 999999999999999999:
    mn = -1
print(mn)
