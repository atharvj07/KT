#include <bits/stdc++.h>
using namespace std;
vector<vector<pair<int, int>>> adj;
vector<int> d, p;
const int inf = 1<<30;
void dijkstra(int s) {
  d.assign(adj.size(), inf);
  p.assign(adj.size(), -1);
  d[s] = 0;
  priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> q;
  q.push({0, s});
  while (!q.empty()) {
    int dist = q.top().first;
    int v = q.top().second;
    q.pop();
    if (dist > d[v]) continue;
    for (auto edge : adj[v]) {
      int to = edge.first;
      int len = edge.second;
      if (d[v] + len < d[to]) {
        d[to] = d[v] + len;
        p[to] = v;
        q.push({d[to], to});
      }
    }
  }
}
int main() {
  int n, m, t;
  cin >> n >> m >> t;
  adj.resize(n);
  for (int i = 0; i < m; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    u--, v--;
    adj[u].push_back({v, w});
  }
  dijkstra(0);
  if (d[n - 1] > t) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> path;
  for (int v = n - 1; v != -1; v = p[v]) path.push_back(v + 1);
  reverse(path.begin(), path.end());
  cout << path.size() << endl;
  for (auto u : path) cout << u << " ";
  cout << endl;
  return 0;
}
