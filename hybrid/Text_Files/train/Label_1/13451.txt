#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, l, r, s;
    cin >> n >> l >> r >> s;
    int sum = 0;
    for (int i = 1; i <= r; i++) { sum += i; }
    if (s < sum || (s - sum) % (r - l + 1) != 0 || (l > 1 && s == sum)) {
      cout << -1 << '\n';
      continue;
    }
    int x = (s - sum) / (r - l + 1);
    vector<int> p(n + 1);
    for (int i = l; i <= r; i++) { p[i] = x; }
    sum = 0;
    for (int i = 1; i <= n; i++) {
      if (!p[i]) {
        if (sum == 0) {
          p[i] = x + 1;
          sum += p[i];
          continue;
        }
        int y = r - s + sum;
        if (y <= 0 || y > n) {
          cout << -1 << '\n';
          goto end;
        }
        p[i] = y;
        sum += p[i];
      }
    }
    for (int i = 1; i <= n; i++) cout << p[i] << ' ';
    cout << '\n';
  end:;
  }
}
