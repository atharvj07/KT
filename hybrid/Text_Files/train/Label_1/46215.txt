import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static class Student {
        int bestFriendId;
        int gender;
    }

    public static class Pair {
        int a;
        int b;

        Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        Student[] students = new Student[n + 1];
        for (int i = 1; i <= n; i++) {
            Student student = new Student();
            String[] input = reader.readLine().split(" ");
            student.bestFriendId = Integer.parseInt(input[0]);
            student.gender = Integer.parseInt(input[1]);
            students[i] = student;
        }

        Map<Integer, Integer> bestFriends = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            bestFriends.put(i, students[i].bestFriendId);
        }

        int maxPairs = 0;
        int maxBoyGirlPairs = 0;
        List<Pair> pairs = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (bestFriends.containsKey(i)) {
                int bestFriendId = bestFriends.get(i);
                if (bestFriends.containsKey(bestFriendId) && bestFriends.get(bestFriendId) == i) {
                    bestFriends.remove(i);
                    bestFriends.remove(bestFriendId);
                    pairs.add(new Pair(i, bestFriendId));
                    if (students[i].gender != students[bestFriendId].gender) {
                        maxBoyGirlPairs++;
                    }
                    maxPairs++;
                }
            }
        }

        System.out.println(maxPairs + " " + maxBoyGirlPairs);
        for (Pair pair : pairs) {
            System.out.println(pair.a + " " + pair.b);
        }
    }
}
