import queue

n = int(input())
a = [int(x) for x in input().split()]
q = queue.PriorityQueue()
for i in range(n):
    q.put((a[i], i))
ans = 0
while q.qsize() > 1:
    top1 = q.get()
    top2 = q.get()
    if top1[0] > 1:
        top1 = (top1[0] - 1, top1[1])
        q.put(top1)
    if top2[0] > 1:
        top2 = (top2[0] - 1, top2[1])
        q.put(top2)
    ans += 1
    if top1[1] > top2[1]:
        top1, top2 = top2, top1
    a[top1[1]] += a[top2[1]]
    a[top2[1]] = 0
    q.put((a[top1[1]], top1[1]))
print(ans)
