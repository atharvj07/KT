#include <iostream>
#include <vector>

using namespace std;

struct point {
    int x, y;
};

int main() {
    int n;
    cin >> n;
    vector<char> d(n);
    for (int i = 0; i < n; ++i) cin >> d[i];
    point p;
    cin >> p.x >> p.y;
    vector<point> um(2);
    for (char c : d) {
        if (c == 'U') ++um[1].y;
        if (c == 'D') --um[1].y;
        if (c == 'L') --um[1].x;
        if (c == 'R') ++um[1].x;
    }
    int ans = abs(um[1].x - p.x) + abs(um[1].y - p.y);
    int ans_idx = -1;
    for (int i = 0; i < n; ++i) {
        if (d[i] == 'U') um[0].y++;
        if (d[i] == 'D') um[0].y--;
        if (d[i] == 'L') um[0].x--;
        if (d[i] == 'R') um[0].x++;
        int cur = abs(um[0].x - p.x) + abs(um[0].y - p.y);
        if (cur < ans) {
            ans_idx = i;
            ans = cur;
        }
    }
    cout << (ans + 1) / 2 << '\n';
    if (ans_idx != -1) {
        for (int i = 0; i < n; ++i) {
            if (ans_idx != i) cout << d[i];
            else {
                if (um[0].x < p.x) {
                    cout << 'R';
                    ++um[0].x;
                } else if (um[0].x > p.x) {
                    cout << 'L';
                    --um[0].x;
                } else if (um[0].y < p.y) {
                    cout << 'U';
                    ++um[0].y;
                } else if (um[0].y > p.y) {
                    cout << 'D';
                    --um[0].y;
                }
            }
        }
    }
    return 0;
}
