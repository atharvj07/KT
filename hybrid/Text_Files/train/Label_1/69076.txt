def solve(n0, n1, n2):
    if n1 == 0:
        if n0 > 0:
            return '0' * (n0 + 1)
        else:
            return '1' * (n2 + 1)
    else:
        result = '0' * (n0 + 1) + '1' * (n2 + 1)
        n1 -= 1
        for _ in range(n1):
            if _ % 2 == 0:
                result += '0'
            else:
                result += '1'
        return result

t = int(input().strip())
for _ in range(t):
    n0, n1, n2 = map(int, input().strip().split())
    print(solve(n0, n1, n2))
