#include <bits/stdc++.h>

#define ll long long
#define mod 1000000007
#define pb push_back
using namespace std;
vector<ll>a, adj[100001];
bool vis[100001];
ll ans;

void dfs(ll nd)
{
	if (vis[nd])
		return;
	vis[nd] = true;
	ans++;
	for (auto x : adj[nd])
		if (!vis[x])
			dfs(x);
}

int main()
{
	ll n, x, mx = 0;
	cin >> n;
	for (ll i = 0; i < n; i++)
	{
		cin >> x;
		a.pb(x);
	}
	for (ll i = 0; i < n; i++)
	{
		for (ll j = i + 1; j < n; j++)
		{
			if (a[i] > a[j])
			{
				swap(a[i], a[j]);
				adj[a[j]].pb(a[i]);
				adj[a[i]].pb(a[j]);
			}
		}
	}
	for (ll i = 1; i <= n; i++)
	{
		if (!vis[i])
		{
			ans = 0;
			dfs(i);
			mx = max(ans, mx);
		}
	}
	cout << mx << endl;
	return 0;
}
