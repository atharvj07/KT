#include <bits/stdc++.h>

using namespace std;

struct Table {
    int id, cap;
    bool operator<(const Table &a) { return cap > a.cap; }
};

struct Request {
    int id, num, pay;
    bool operator<(const Request &a) { return num > a.num; }
};

int main() {
    int n, k;
    cin >> n;
    vector<Request> req(n);
    for (int i = 0; i < n; i++) {
        cin >> req[i].num >> req[i].pay;
        req[i].id = i + 1;
    }
    cin >> k;
    vector<Table> tab(k);
    for (int i = 0; i < k; i++) {
        cin >> tab[i].cap;
        tab[i].id = i + 1;
    }
    sort(req.begin(), req.end());
    sort(tab.begin(), tab.end());
    int ans = 0, cnt = 0;
    vector<pair<int, int>> answ;
    for (auto &a : req) {
        for (auto &b : tab) {
            if (a.num <= b.cap) {
                ans += a.pay;
                cnt++;
                answ.push_back({a.id, b.id});
                b.cap -= a.num;
                break;
            }
        }
    }
    cout << cnt << ' ' << ans << endl;
    for (auto &a : answ) cout << a.first << ' ' << a.second << endl;
    return 0;
}
