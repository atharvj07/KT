import sys
input = sys.stdin.readline

m, k, n, s = map(int, input().strip().split())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split())) + [0]
cnt = [0] * (5 * 10 ** 5 + 10)
f = 0
for i in a:
    cnt[i] += 1
    if i in b and cnt[i] == b[0]:
        b = b[1:]
        if b == []:
            f = 1
if f == 0:
    print(-1)
else:
    ans = []
    for i in range(1, m + 1):
        if cnt[a[i - 1]] > 0:
            cnt[a[i - 1]] -= 1
            if f == 1 and all(i for i in cnt[1:] if i == 0):
                ans = []
                f = 0
            if cnt[a[i - 1]] < 0:
                ans.append(i)
    print(len(ans))
    for an in ans: print(an)
