#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7, B = 233;
int qpow(int x, int y) {
    int res = 1;
    while (y) {
        if (y & 1) res = res * x % mod;
        x = x * x % mod;
        y >>= 1;
    }
    return res;
}
int n, p, fac[2001], inv[2001];
bool f[2001];
vector<int> v;
set<int> ans;
signed main() {
    cin >> n >> p;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    fac[0] = fac[1] = inv[0] = inv[1] = 1;
    for (int i = 2; i <= n; i++) {
        fac[i] = fac[i - 1] * i % p;
        inv[i] = qpow(fac[i], p - 2);
    }
    for (int x = 1; x <= 1 << n; x++) {
        int num = 0;
        for (int i = 0; i < n; i++) {
            if (x >> i & 1) num += a[i];
        }
        f[num] = true;
    }
    for (int i = 1; i <= n; i++) {
        int sum = 0;
        for (int j = 0; j < n; j++) {
            if (!f[sum + a[j]]) sum += a[j];
        }
        v.push_back(sum);
    }
    sort(v.begin(), v.end());
    int j = 0;
    for (int i = 1; i <= 2000; i++) {
        while (j < n && v[j] <= i) j++;
        if (j < n && v[j] == i) continue;
        if (!f[i]) {
            int tmp = i;
            while (tmp > 1) {
                tmp = (tmp + p - 1) / p;
                if (f[tmp]) break;
            }
            if (tmp == 1) ans.insert(i);
        }
    }
    cout << ans.size() << endl;
    for (auto x : ans) cout << x << ' ';
    return 0;
}
