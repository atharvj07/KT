#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
ll a[200005], b[200005], n, m, ans = 0;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  cin >> a[n];
  cin >> b[m];
  while (m) {
    ans = (ans + (a[n] & b[m])) % 998244353;
    for (ll i = m; i >= 0; i--) {
      b[i] /= 2;
      if (b[i] == 0 && i > 0) m--;
    }
  }
  cout << ans << '\n';
  return 0;
}
