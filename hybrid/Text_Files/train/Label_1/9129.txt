#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1505;
const ll inf = 1ll << 60;
int n, s, m, k;
int a[N];
int l[N], r[N];
int d[N];
vector<int> v[N];
int p[N][N];
signed main() {
    cin >> n >> s >> m >> k;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= s; i++) {
        cin >> l[i] >> r[i];
        d[l[i]]++, d[r[i] + 1]--;
    }
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        sum += d[i];
        if (sum >= k) {
            v[i - sum + 1].push_back(i);
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= s; j++) {
            if (l[j] <= i && i <= r[j]) { p[j][i] = 1; }
        }
    }
    ll ans = inf;
    for (int t = 1; t <= s - m + 1; t++) {
        if (v[t].size() >= k) {
            vector<int> p1;
            for (int x : v[t]) { p1.push_back(a[x]); }
            sort(p1.begin(), p1.end());
            ans = min(ans, 1ll * p1[k - 1]);
        } else {
            ll sum = inf;
            for (int x : v[t]) {
                vector<ll> p2;
                for (int j = 1; j <= s; j++) {
                    if (p[j][x]) {
                        for (int x : v[t + j - 1]) { p2.push_back(a[x]); }
                    }
                }
                sort(p2.begin(), p2.end());
                if (p2.size() >= k) { sum = min(sum, p2[k - 1]); }
            }
            ans = min(ans, sum);
        }
    }
    if (ans == inf) {
        cout << -1;
    } else {
        cout << ans;
    }
}
