n, m, k = map(int, input().split())
p = list(map(int, input().split()))
cum = [p[0]]
for i in p[1:]:
    cum.append(cum[-1]+i)
ans = 0
l = 0
for r in range(m*k):
    if r>=m:
        if r-m>=0:
            l = r-m+1
        ans = max(ans, cum[r]-cum[l-1])
    else:
        ans = max(ans, cum[r])
print(ans)
