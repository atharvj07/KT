#include <bits/stdc++.h>
using namespace std;

const int MX = 1e5;

int main() {
    int t, n;
    vector<int> q(MX);
    cin >> t;
    while(t--) {
        cin >> n;
        for(int i=0; i<n; i++) cin >> q[i];
        if(n>1) {
            bool f[n+1];
            memset(f, 0, sizeof(f));
            f[q[n-1]]=1;
            for(int i=n-2; i>=0; i--) {
                int mx=0;
                for(int j=q[i]; j<=q[i+1]; j++)
                    if(f[j] == 0) 
                        mx=j;
                if(mx==0) {
                    cout << -1 << endl;
                    goto END;
                }
                f[mx]=1;
            }
            vector<int> p(n+1);
            int cur=0;
            for(int i=1; i<=n; i++) {
                if(f[i]) {
                    p[cur]=q[cur]=i;
                    cur++;
                }
            }
            for(int i=1; i<=n; i++) {
                if(!f[i]) {
                    q[cur]=i;
                    p[cur]=i;
                    cur++;
                }
            }
            for(int i=1; i<=n; i++)
                cout << p[i] << " ";
            cout << endl;
        }
        else cout << 1 << endl;
        END:;
    }
    return 0;
}
