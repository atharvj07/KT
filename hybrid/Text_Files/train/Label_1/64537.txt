import sys
from collections import Counter
input=sys.stdin.readline

n=int(input())
a=list(map(int,input().split()))
b=[0]*n
c=[*Counter([i-j for i,j in zip(a,b)]).keys()]
ans=[]
for i in range(n):
    for j in range(i+1,n):
        if (j-i+1)in c and a[j]<a[i]:
            a[i],a[j]=a[j],a[i]
            ans.append([i+1,j+1])
print(len(ans))
for i in ans:
    print(*i)
