#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, c1, c2;
    cin >> n >> c1 >> c2;
    string s;
    cin >> s;
    ll ans = 0, ad = 0, ch = 0;
    for (ll i = 0; i < n; i++) {
        if (s[i] == '1') { ad++; }
    }
    for (ll i = 0; i < n; i++) {
        if (s[i] == '1') { ch--; }
        else if (s[i] == '0' && ch < 0) { ch = 0; }
        ch++;
        if (ch == 1) { ans += c1; }
        else { ans += c2 + (ch - 1ll) * (ch - 1ll) * c2; }
    }
    cout << ans << endl;
    return 0;
}
