import java.util.Scanner;

public class CountryNeuralNetwork {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] interLayerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            interLayerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        int[][] dp = new int[l][n];

        // Base case for the first layer
        for (int i = 0; i < n; i++) {
            if (entryCosts[i] % m == 0) {
                dp[0][i] = 1;
            }
        }

        // Iterate through the remaining layers
        for (int i = 1; i < l; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if ((dp[i - 1][k] % m != 0) || (interLayerCosts[k] % m != 0)) {
                        continue;
                    }
                    dp[i][j] = (dp[i][j] + dp[i - 1][k]) % MOD;
                }
            }
        }

        int totalPaths = 0;
        for (int i = 0; i < n; i++) {
            if ((dp[l - 1][i] % m == 0) && (exitCosts[i] % m == 0)) {
                totalPaths = (totalPaths + dp[l - 1][i]) % MOD;
            }
        }

        System.out.println(totalPaths);
    }
}
