def sum_of_digits(x):
    return sum(int(digit) for digit in str(x))

def find_x(n):
    results = []
    for x in range(max(1, n - 81), n):
        if x + sum_of_digits(x) == n:
            results.append(x)
    return results

n = int(input())
results = find_x(n)
print(len(results))
for x in results:
    print(x)
