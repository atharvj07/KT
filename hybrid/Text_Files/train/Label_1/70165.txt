import heapq
n = int(input())
a = [0 for i in range(n)]
b = [0 for i in range(n)]
for i in range(n):
    a[i], b[i] = map(int, input().split())
d = []
heapq.heapify(d)
for i in range(n):
    while d:
        if d[0] < a[i]:
            heapq.heappop(d)
        else:
            break
    heapq.heappush(d, a[i] + b[i])
ans = len(d)
print(ans)
