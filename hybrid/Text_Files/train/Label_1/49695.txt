import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (true) {
            String p = in.nextLine();
            if (p.equals("#")) break;
            if (p.matches("^(?:0101)*0100$")) System.out.println("Yes");
            else System.out.println("No");
        }
    }
}
