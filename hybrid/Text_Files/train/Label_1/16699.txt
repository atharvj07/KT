#include<bits/stdc++.h>
#define ll long long
#define INF 1e15
using namespace std;

ll n, k, l[100005], r[100005], pref[100005], ans;

int main() {
    cin >> n >> k;
    for(int i = 1;i <= n;i++) {
        cin >> l[i] >> r[i];
        pref[r[i] + 1] -= 1;
        pref[l[i]] += 1;
    }
    ll cover = 0;
    for(int i = 1;i <= 1e5;i++) {
        cover += pref[i];
        if(cover % k)
            ans++;
    }
    cout << ans << endl;
    return 0;
}
