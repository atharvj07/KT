#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<char>> photo(n, std::vector<char>(m));
    bool is_colored = false;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            std::cin >> photo[i][j];
            if (photo[i][j] == 'C' || photo[i][j] == 'M' || photo[i][j] == 'Y') {
                is_colored = true;
                break;
            }
        }
        if (is_colored) {
            break;
        }
    }
    std::cout << (is_colored ? "#Color" : "#Black&White") << std::endl;
    return 0;
}
