#include<iostream>
using namespace std;
typedef long long ll; const int mod = 1e9 + 7;
int main() {
    ll n, k; cin >> n >> k;
    ll dp[n + 1][k + 1]; for (int i = 0; i <= n; i++) for (int j = 0; j <= k; j++) dp[i][j] = 0;
    dp[1][0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            dp[i][j] += dp[i - 1][j];
            if (j > 0) dp[i][j] += dp[i - 1][j - 1];
            dp[i][j] %= mod;
        }
    }
    cout << (k < n ? dp[n][k] : 0) << '\n';
}
