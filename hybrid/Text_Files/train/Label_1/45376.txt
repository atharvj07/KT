import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Set<String> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            set.add(s);
        }
        int minLen = Integer.MAX_VALUE;
        String minStr = "";
        for (char c = 'a'; c <= 'z'; c++) {
            String s = String.valueOf(c);
            if (!set.contains(s)) {
                if (s.length() < minLen || (s.length() == minLen && s.compareTo(minStr) < 0)) {
                    minLen = s.length();
                    minStr = s;
                }
            }
        }
        for (int i = 2; i <= 20; i++) {
            for (char c1 = 'a'; c1 <= 'z'; c1++) {
                for (char c2 = c1; c2 <= 'z'; c2++) {
                    String s = String.valueOf(c1) + String.valueOf(c2);
                    if (!set.contains(s)) {
                        if (s.length() < minLen || (s.length() == minLen && s.compareTo(minStr) < 0)) {
                            minLen = s.length();
                            minStr = s;
                        }
                    }
                }
            }
        }
        System.out.println(minStr);
    }
}
