#include <iostream>
#include <vector>
using namespace std;
int main() {
    long long p, q;
    cin >> p >> q;
    long long s1 = p, s2 = q;
    long long s3 = 0, s4 = 1;
    int n;
    cin >> n;
    vector<long long> a(n);
    for (auto &i : a) cin >> i;
    for (int i = n - 1; i >= 0; i--) {
        s3 = a[i] * s4 + s3;
        s4 = a[i] * s3 + s4;
    }
    long long x = s1 * s4, y = s2 * s3;
    if (x == y)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
