#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> odd, even;
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            if (odd.empty() || odd.back() != i) {
                odd.push_back(i);
            } else {
                odd.pop_back();
            }
        } else {
            if (even.empty() || even.back() != i) {
                even.push_back(i);
            } else {
                even.pop_back();
            }
        }
    }
    int k = (odd.size() + 1) / 2;
    cout << k << endl;
    for (int i = 0; i < k; i++) {
        int m = odd.size();
        cout << m << endl;
        for (int j = 0; j < m; j++) {
            cout << odd[j] + 1 << " ";
        }
        odd.clear();
        m = even.size();
        cout << m << endl;
        for (int j = 0; j < m; j++) {
            cout << even[j] + 1 << " ";
        }
        even.clear();
    }
    return 0;
}
