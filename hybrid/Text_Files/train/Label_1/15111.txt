#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5+10;
typedef long long ll;
int main() {
    ll n; cin >> n;
    bool has_x = 0;
    long double minx = 1e18, maxx = -1e18;
    long double miny = 1e18, maxy = -1e18;
    for (ll i = 0; i < n; i++) {
        ll x, y; cin >> x >> y;
        has_x |= (x!=0);
        minx = min(minx, (long double)x);
        maxx = max(maxx, (long double)x);
        miny = min(miny, (long double)y);
        maxy = max(maxy, (long double)y);
    }
    if (has_x) cout << "Yes" << endl;
    else if (minx*maxx < 0 || miny*maxy < 0) cout << "Yes" << endl;
    else cout << "No" << endl;
}
