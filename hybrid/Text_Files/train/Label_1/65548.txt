from string import ascii_lowercase
good_letters = set(input())
pattern = input()
if '*' in pattern:
    pattern = pattern.replace('*', '\\w*')
pattern = pattern.replace('?', '\\w')
for s in [input() for _ in range(int(input()))]:
    print('YES' if s.count('?') <= len(good_letters) and s.count('?') + s.count('*') <= len(ascii_lowercase) else 'NO')
