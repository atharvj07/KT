import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = scanner.nextInt();
        }
        Arrays.sort(d);
        int s = 0;
        for (int i = 1; i < n; i++) {
            s = Math.max(s, Math.min(d[i], 24 - d[i]));
        }
        System.out.println(s);
    }
}
