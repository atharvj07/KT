import sys

input = sys.stdin.readline

n, T = map(int, input().split())

a = list(map(int, input().split()))
t = list(map(int, input().split()))

# Calculate the total amount of water and the total heat
total_a = 0
total_heat = 0

for i in range(n):
    total_a += a[i]
    total_heat += a[i] * t[i]

# Calculate the maximum amount of water with the desired temperature
max_water = 0

for i in range(n):
    water = total_a * (T - total_heat + a[i] * t[i]) / (a[i] * (T - t[i]))
    if water > max_water:
        max_water = water

print(max_water)
