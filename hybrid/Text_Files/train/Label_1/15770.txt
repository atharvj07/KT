#include <iostream>
#include <string>
#include <set>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    set<char> res;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == s[i + 1]) {
        res.erase(s[i]);
      } else {
        res.insert(s[i]);
      }
      i++;
    }
    for (auto c : res) { cout << c; }
    cout << '\n';
  }
  return 0;
}
