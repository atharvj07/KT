import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        char[] s = sc.next().toCharArray();
        int g = 0, t = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'G') g = i;
            if (s[i] == 'T') t = i;
        }
        if ((t - g) % k == 0) System.out.println("YES");
        else System.out.println("NO");
    }
}
