#include <algorithm>
#include <iostream>
#include <string>

int main() {
  std::string a;
  std::cin >> a;
  int n = a.length();
  int index = n - 1;
  int ma = 0;
  for (int i = 0; i < n; i++) {
    int cur = 0;
    int t = n - 1 - i;
    for (int j = i; j < n; j++) {
      cur += (a[j] - '0') << t;
      t--;
    }
    if (cur > ma) {
      ma = cur;
      index = i;
    }
  }
  std::string b = a.substr(0, index);
  if (index + 1 < n) b += a.substr(index + 1);
  std::cout << b << std::endl;
}
