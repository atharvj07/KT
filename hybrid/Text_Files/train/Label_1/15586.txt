#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
int n, h, m, k;
vector<pair<ll, ll>> a;
bool cmp(const pair<ll, ll>& a, const pair<ll, ll>& b) {
  if (a.second == b.second) {
    return a.first > b.first;
  }
  return a.second > b.second;
}
int main() {
  cin >> n >> h >> m >> k;
  for (int i = 0; i < n; i++) {
    ll hi, mi;
    cin >> hi >> mi;
    a.push_back(make_pair(hi, mi));
  }
  sort(a.begin(), a.end(), cmp);
  ll pre = ll(1e18);
  ll ans = 0, ansl, ansr;
  int cnt = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (a[i].first * m + a[i].second > pre - k) {
      ans++;
      a[i].first = ll(1e18);
    }
    ansl = a[i].second - k;
    if (ansl < 0) ansl += m;
    ansr = a[i].second + m / 2 - k;
    if (ansr >= m) ansr -= m;
    if (ansl <= ansr) cnt++;
    pre = a[i].first * m + a[i].second;
  }
  if (cnt == n) {
    cout << 0 << " " << 0 << "\n";
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (a[i].first == ll(1e18)) {
      cout << 1 << " " << a[i].second << "\n";
      for (int j = i + 1; j < n; j++) {
        cout << a[j].first << " " << a[j].second << "\n";
      }
      return 0;
    }
  }
  cout << 0 << " " << 0 << "\n";
  return 0;
}
