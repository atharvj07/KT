n = int(input())
total_faces = 0
for _ in range(n):
    polyhedron = input()
    if polyhedron == "Tetrahedron":
        total_faces += 4
    elif polyhedron == "Cube":
        total_faces += 6
    elif polyhedron == "Octahedron":
        total_faces += 8
    elif polyhedron == "Dodecahedron":
        total_faces += 12
    elif polyhedron == "Icosahedron":
        total_faces += 20
print(total_faces)
