#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back

const int N = 1e3 + 5;
vector<int> adj[N];
int dis[N];
bool vis[N];
void bfs(int start){
    queue<int> q;
    q.push(start);
    vis[start] = true;
    while(!q.empty()){
        int cur = q.front();
        q.pop();
        for(int u : adj[cur]){
            if(!vis[u]){
                vis[u] = true;
                dis[u] = dis[cur] + 1;
                q.push(u);
            }
        }
    }
}
int main(){
    int n, m, s, t; cin >> n >> m >> s >> t;
    for(int i = 0; i < m; i ++){
        int u, v; cin >> u >> v;
        adj[u].pb(v);
        adj[v].pb(u);
    }
    bfs(s);
    int maximum = -1;
    for(int i = 1; i <= n; i++) maximum = max(maximum, dis[i]);
    int ans = 0;
    for(int i = 1; i <= n; i++){
        for(int u : adj[i]){
            if(dis[i] + dis[u] == maximum){
                ans ++;
            }
        }
    }
    ans /= 2;
    cout << ans << endl;
}
