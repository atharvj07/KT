#include <iostream>
#include <vector>

using namespace std;

int a[101];

int main()
{
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            int t = 0;
            for (int k = i; k <= j; k++)
            {
                t ^= a[k];
            }
            ans = max(ans, t);
        }
    }
    cout << ans << endl;
    return 0;
}
