#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> groups(n);
  for (int &x : groups) cin >> x;

  int teams = 0, threes = 0, twos = 0;
  for (int x : groups) {
    if (x == 3) threes++;
    if (x == 2) twos++;
  }
  teams += threes;
  if (twos % 2 == 0) teams += twos / 2;
  else {
    teams += twos / 2;
    if (threes > 0) teams++, threes--;
    else twos -= 2;
  }
  teams += min(threes, twos);
  cout << teams << endl;
}
