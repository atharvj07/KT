#include <iostream>
#include <vector>
using namespace std;

vector<int> ans;
void u(int n)
{
    if(ans.size()>n)
    {
        ans.erase(ans.begin()+n);
        return;
    }
    while(ans.size()<=n)ans.push_back(0);
}
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        ans.clear();
        int n,r;
        cin>>n>>r;
        ans.push_back(0);
        for(int i=0; i<n-1; ++i)
        {
            int x;
            cin>>x;
            u(x);
            if(ans[x]==0)ans[x]=x;
            else ans[x]=ans[x]==x?0:x;
        }
        int c=0;
        for(int i=1; i<ans.size(); ++i)if(ans[i])++c;
        if(!c)
        {
            cout<<'*';
            continue;
        }
        int x=1;
        while(ans[x])++x;
        if(ans[x]==0)cout<<x<<endl;
        else cout<<"*"<<endl;
    }
    return 0;
}
