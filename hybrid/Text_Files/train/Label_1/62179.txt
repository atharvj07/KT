def min_seconds_to_modify_levels(test_cases):
    answers = []
    for n, p, k, a, x, y in test_cases:
        a = '0' + a  # Adjust indexing.
        dp = [0] * (n + 1)
        cost = float('inf')
        
        for start in range(p, min(p+k, n+1)):  # Loop through all possible starts within one period.
            # Calculate the initial cost for this start.
            if start == p:
                dp[start] = sum(1 for i in range(start, n+1, k) if a[i] == '0') * x
            else:
                # Reduce the calculation by removing the farthest element and adding the new start if necessary.
                dp[start] = dp[start-1] - (0 if a[start-1+(n-start+1)//k*k] == '1' else x) + (0 if a[start] == '1' else x)
            
            current_cost = dp[start] + (start-p) * y
            cost = min(cost, current_cost)
        
        answers.append(cost)
    
    return answers

# Read input
t = int(input())
test_cases = []
for _ in range(t):
    n, p, k = map(int, input().split())
    a = input().strip()
    x, y = map(int, input().split())
    test_cases.append((n, p, k, a, x, y))

# Solve and output
answers = min_seconds_to_modify_levels(test_cases)
for answer in answers:
    print(answer)
