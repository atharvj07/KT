#include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

int h, w, q, ans;
char grid[505][505];
vector<int> horiz[505], vert[505];

int main() {
  scanf("%d %d", &h, &w);
  for (int i = 1; i <= h; i++) {
    scanf(" %s", grid[i] + 1);
  }
  for (int i = 1; i <= h; i++) {
    for (int j = 1; j < w; j++) {
      if (grid[i][j] != '#' && grid[i][j + 1] != '#') horiz[i].push_back(j);
    }
  }
  for (int j = 1; j <= w; j++) {
    for (int i = 1; i < h; i++) {
      if (grid[i][j] != '#' && grid[i + 1][j] != '#') vert[j].push_back(i);
    }
  }
  scanf("%d", &q);
  for (int i = 1; i <= q; i++) {
    int y1, x1, y2, x2;
    scanf("%d %d %d %d", &y1, &x1, &y2, &x2);
    ans = 0;
    for (int j = x1; j <= x2; j++) {
      const vector<int> &a = horiz[y1];
      const vector<int> &b = horiz[y2];
      int l = 0, r = a.size() - 1;
      while (l <= r) {
        int m = (l + r) / 2;
        if (a[m] < j)
          l = m + 1;
        else
          r = m - 1;
      }
      if (l < a.size() && a[l] == j) ans += b.size();
    }
    for (int j = y1; j <= y2; j++) {
      const vector<int> &a = vert[x1];
      const vector<int> &b = vert[x2];
      int l = 0, r = a.size() - 1;
      while (l <= r) {
        int m = (l + r) / 2;
        if (a[m] < j)
          l = m + 1;
        else
          r = m - 1;
      }
      if (l < a.size() && a[l] == j) ans += b.size();
    }
    printf("%d\n", ans);
  }
  return 0;
}
