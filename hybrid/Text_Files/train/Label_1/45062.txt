import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] pixels = new int[n][n];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < n; j++) {
                pixels[i][j] = line.charAt(j) - '0';
            }
        }
        int minCommands = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (pixels[i][j] == 1) {
                    int commands = 0;
                    for (int k = 0; k <= i; k++) {
                        commands += pixels[k][j];
                    }
                    for (int k = 0; k <= j; k++) {
                        commands += pixels[i][k];
                    }
                    minCommands = Math.min(minCommands, commands);
                }
            }
        }
        System.out.println(minCommands);
    }
}
