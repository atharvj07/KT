import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long m = 2;

        // Loop until we find a value of m that makes nCm even
        while((n-m+1) % 2 == 1){
            m++;
        }

        System.out.println(m);
    }
}
