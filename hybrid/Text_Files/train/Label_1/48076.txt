import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] coins = new int[5];
        for (int i = 0; i < 5; i++) {
            coins[i] = in.nextInt();
        }
        int sum = 0;
        for (int coin : coins) {
            sum += coin;
        }
        if (sum % 2 == 0) {
            System.out.println(sum / 2);
        } else {
            System.out.println(-1);
        }
    }
}
