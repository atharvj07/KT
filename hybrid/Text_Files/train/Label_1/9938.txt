#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.size(), cnt_n = 0, cnt_t = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'n')
      ++cnt_n;
    if (s[i] == 't')
      ++cnt_t;
    if (s[i] == 'e') {
      if (cnt_n > cnt_t)
        --cnt_n;
      else
        --cnt_t;
    }
  }
  cout << cnt_n + cnt_t << endl;
  return 0;
}
