n,v=[int(x)for x in input().split()]
a=[0]*n
b=[0]*n
for i in range(n):
    a[i],b[i]=[int(x)for x in input().split()]
s=sorted([(a[i],i)for i in range(n)])
ans,cur=0,0
for x in s:
    if cur+b[x[1]]<=v:
        ans+=b[x[1]]
        cur+=b[x[1]]
    else:
        ans+=v-cur
        cur=v
        if cur+b[x[1]]<=v:
            ans+=b[x[1]]
            cur+=b[x[1]]
        else:
            break
print(ans)
