#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string o, i;
  cin >> o >> i;
  int n = o.size();
  vector<char> ans(n, '?');
  for (int j = 0, k = 0; j < n; j++) {
    if (k == n || o[j] < i[k]) ans[j] = o[j];
    else ans[j] = i[k++];
  }
  for (char c : ans) cout << c;
  cout << '\n';
}
