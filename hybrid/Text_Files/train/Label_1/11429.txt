#include <bits/stdc++.h>
#define ios ios_base::sync_with_stdio(false),cin.tie(0),cout.tie(0)
using namespace std;

const int maxe = 1e6 + 7, maxt = 1e7;
int n, q;
int a[maxe], b[maxe];
map<int, int> m;

int main() {
    ios;
    cin >> n >> q;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        ++m[a[i]];
    }
    cout << m.size() << endl;
    for (int i = 1; i <= q; ++i) {
        int p, x;
        cin >> p >> x;
        --m[a[p]];
        if (m[a[p]] == 0)
            m.erase(a[p]);
        ++m[x];
        a[p] = x;
        cout << m.size() << endl;
    }
    return 0;
}
