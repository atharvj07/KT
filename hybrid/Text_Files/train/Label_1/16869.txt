#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    if (n % 2 == 0) {
        for (int i = 0; i < n; i++) {
            if (v[i] % 2 == 1) {
                cout << "NO" << endl;
                return 0;
            }
        }
    } else {
        int odd = 0;
        for (int i = 0; i < n; i++) {
            if (v[i] % 2 == 1) {
                odd++;
            }
        }
        if (odd > 1) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    return 0;
}
