#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    long long a[n+5];
    for(int i = 1; i <= n; i++){
        cin >> a[i];
    }
    for(int i = 1; i <= n; i+=2){
        if(i+1<=n){
            a[i] = (a[i]%2) + (a[i+1]%2);
            if(a[i]==2) a[i]++;
        }
    }
    for(int i = 1; i <= n; i++){
        cout << a[i] << " ";
    }
    return 0;
}
