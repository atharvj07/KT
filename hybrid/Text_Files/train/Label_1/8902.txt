#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll M = 1e9 + 7;
ll n, x, m;
unordered_map<ll, ll> f;
bool check(vector<ll> &a, ll l, ll r) {
  for (ll i = l; i <= r; i++) {
    if (a[i] == x) return false;
  }
  return true;
}
int main() {
  ll t;
  cin >> t;
  while (t--) {
    cin >> n >> x;
    vector<ll> a(n);
    f.clear();
    for (ll &i : a) cin >> i;
    sort(a.begin(), a.end());
    ll ans = 0;
    m = n;
    for (ll i = 0; i < n; i++) {
      if (a[i] > x) continue;
      ll l = i, r = n - 1;
      while (l < r) {
        ll mid = (l + r) / 2;
        if (check(a, i, mid))
          l = mid + 1;
        else
          r = mid;
      }
      for (ll j = i; j < l; j++) {
        if (a[j] != a[i] && a[j] != x) {
          m--;
        }
      }
      if (m == 0) break;
    }
    if (m == 0)
      ans = 1;
    else if (m == n)
      ans = 0;
    else
      ans = 2;
    cout << ans << endl;
  }
}
