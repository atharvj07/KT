#include<bits/stdc++.h>
using namespace std;
long long solve(vector<long long> &a, long long k){
    const long long n = a.size();
    priority_queue<long long> pq;
    long long sum = 0;
    for(long long i=0; i<n; i++){
        if(a[i]%10ll) pq.push((10ll-a[i]%10ll));
    }
    while(!pq.empty()&&k){
        sum+=pq.top();
        pq.pop();
        k--;
    }
    return sum;
}
int main(){
    long long n, k;
    cin>>n>>k;
    vector<long long> a(n);
    for(long long i=0; i<n; i++) cin>>a[i];
    cout<<solve(a, k)<<"\n";
    return 0;
}
