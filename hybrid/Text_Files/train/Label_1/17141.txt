#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, k, ans=0, r=0, b=0;
    cin >> n >> k;
    vector<pair<int, int>> v(n);
    for(auto& p : v) {
        cin >> p.first >> p.second;
        r += p.first;
        b += p.second;
    }
    sort(v.rbegin(), v.rend());
    for(auto p : v) {
        ans += min({p.first, p.second, k});
        k -= min({p.first, p.second, k});
    }
    ans += min({r, b, k});
    cout << ans << endl;
}
