import sys

t = int(input())
for _ in range(t):
    n, h = map(int, input().split())
    a = [int(x) for x in input().split()]
    k = 10**9
    l, r = 1, h
    while l <= r:
        m = (l + r) // 2
        cur = 0
        prv = 0
        for v in a:
            if prv + m - 1 < v:
                prv = v
                cur += m
        if cur >= h:
            k = min(k, m)
            r = m - 1
        else:
            l = m + 1
    print(k)
