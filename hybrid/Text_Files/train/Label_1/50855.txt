import java.util.Scanner;

public class Main {
    static long[] fac;
    static long mod = (long) Math.pow(10, 9) + 7;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        fac = new long[n + 1];
        fac[0] = 1;
        for (int i = 1; i <= n; i++) {
            fac[i] = fac[i - 1] * i % mod;
        }
        long ans = (long) Math.pow(n, n) % mod * (fac[n - 1] * fac[n - 1] % mod) % mod;
        System.out.println(ans);
    }
}
