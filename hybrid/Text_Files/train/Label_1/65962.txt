def max_three_blocks_palindrome(t, test_cases):
    results = []

    for n, a in test_cases:
        count = [[0] * 27 for _ in range(n + 1)]
        for i in range(1, n + 1):
            for j in range(1, 27):
                count[i][j] = count[i - 1][j] + (1 if a[i - 1] == j else 0)
        
        answer = 0
        for i in range(1, 27):
            answer = max(answer, count[n][i])
            for j in range(1, 27):
                if i != j:
                    prefix = 0
                    suffix = n + 1
                    while prefix < suffix - 1:
                        mid = (prefix + suffix) // 2
                        if count[mid][i] < count[n][i] - count[mid][i]:
                            prefix = mid
                        else:
                            suffix = mid
                    mid = prefix
                    temp = 2 * count[mid][i] + count[n][j] - count[mid][j]
                    answer = max(answer, temp)
                    
        results.append(answer)

    return results

# Example usage
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    test_cases.append((n, a))

results = max_three_blocks_palindrome(t, test_cases)
for res in results:
    print(res)
