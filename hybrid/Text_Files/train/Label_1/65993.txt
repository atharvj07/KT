import sys

input = sys.stdin.readline

n = int(input())
a = []
b = []
for i in range(n):
  sa, ea, sb, eb = map(int, input().split())
  a.append((sa, ea))
  b.append((sb, eb))

a.sort()
b.sort()

i, j = 0, 0
while i < n and j < n:
  if a[i][1] > b[j][1]:
    j += 1
  elif a[i][1] == b[j][1]:
    i += 1
    j += 1
  else:
    print("NO")
    exit()

print("YES")
