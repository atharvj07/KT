n = int(input())
a = list(map(int, input().split()))
r1, c1, r2, c2 = map(int, input().split())

if r1 == r2:
    print(abs(c1 - c2))
else:
    ans = abs(r1 - r2) + min(abs(c1 - c2),
                             abs(c1 - 1) + abs(c2 - 1),
                             abs(c1 - a[r1-1] - 1) + abs(c2 - a[r2-1] - 1),
                             abs(c1 - a[r1-1] - 1) + abs(c2 - 1))
    for i in range(min(r1, r2) - 1, max(r1, r2)):
        tmp = abs(i + 1 - r1) + abs(i + 1 - r2) + min(abs(c1 - 1) + abs(c2 - 1),
                                                      abs(c1 - a[i] - 1) + abs(c2 - 1),
                                                      abs(c1 - 1) + abs(c2 - a[i] - 1),
                                                      abs(c1 - a[i] - 1) + abs(c2 - a[i] - 1))
        ans = min(ans, tmp)
    print(ans)
