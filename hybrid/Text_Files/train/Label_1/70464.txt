import sys

n, m = map(int, input().split())

l = []
r = []
for _ in range(n):
    li, ri = map(int, input().split())
    l.append(li)
    r.append(ri)

a = sorted([int(x) for x in input().split()])

if a[-1] < r[-1] - l[0]:
    print("No")
    sys.exit()

ans = []
for i in range(n - 1):
    for j in range(m):
        if r[i] + 1 <= l[i + 1] - a[j]:
            ans.append(j + 1)
            a.pop(j)
            break

if len(ans) == n - 1:
    print("Yes")
    print(*ans)
else:
    print("No")
