#include<bits/stdc++.h>
using namespace std;
int f[1000006],g[1000006];
int main(){
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n;
  cin >> n;
  f[0]=1,f[1]=1,f[2]=1;
  for(int i=3;i<=n;i++){
    f[i]=i;
    for(int j=i-1;j>=1;j--)
      g[i]=__gcd(g[i],__gcd(f[i],f[j]));
    f[i]=g[i];
  }
  for(int i=1;i<=n;i++)
    cout<<f[i]<<" ";
  return 0;
}
