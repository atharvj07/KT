import java.util.*;

public class Main {
    static int[] parent;
    static int[] size;

    static void makeSet(int n) {
        parent = new int[n];
        size = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
    }

    static int findSet(int a) {
        if (parent[a] == a) {
            return a;
        }
        return parent[a] = findSet(parent[a]);
    }

    static void unionSet(int a, int b) {
        a = findSet(a);
        b = findSet(b);
        if (a != b) {
            if (size[a] < size[b]) {
                int temp = a;
                a = b;
                b = temp;
            }
            parent[b] = a;
            size[a] += size[b];
        }
    }

    static boolean isSameSet(int a, int b) {
        return findSet(a) == findSet(b);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        makeSet(n);
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt() - 1;
            int b = scanner.nextInt() - 1;
            unionSet(a, b);
        }
        int[] ans = new int[n - 1];
        int ind = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (!isSameSet(i, j)) {
                    ans[ind++] = i + 1;
                    unionSet(i, j);
                }
            }
        }
        System.out.println(ind);
        for (int i = 0; i < ind; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
