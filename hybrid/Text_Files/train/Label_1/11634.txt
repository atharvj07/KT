#include <cstdio>
#include <map>
#include <vector>

using namespace std;

struct point {
  long long x, y;
};

long long dist2(point a, point b) {
  long long dx = a.x - b.x, dy = a.y - b.y;
  return dx * dx + dy * dy;
}

map<long long, vector<point> > groups;

int main() {
  int n;
  scanf("%d", &n);
  vector<point> antennas(n);
  for (int i = 0; i < n; i++) {
    scanf("%lld%lld", &antennas[i].x, &antennas[i].y);
  }
  int m;
  scanf("%d", &m);
  while (m--) {
    vector<long long> distances(n);
    for (int i = 0; i < n; i++) { scanf("%lld", &distances[i]); }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        groups[distances[i] - distances[j]].push_back(point{antennas[i].x - antennas[j].x, antennas[i].y - antennas[j].y});
      }
    }
    bool first = true;
    for (map<long long, vector<point>>::iterator it = groups.begin(); it != groups.end(); it++) {
      vector<point> &group = it->second;
      if (group.size() >= 2) {
        point p0 = group[0], p1 = group[1];
        long long dx = p0.x - p1.x, dy = p0.y - p1.y, len = dx * dx + dy * dy;
        if (first) {
          first = false;
          printf("%d", (int)group.size());
          for (int i = 0; i < (int)group.size(); i++) {
            printf(" %lld %lld", p0.x + group[i].x, p0.y + group[i].y);
          }
        } else {
          for (int i = 0; i < (int)group.size(); i++) {
            printf(" %lld %lld", p0.x + group[i].x, p0.y + group[i].y);
          }
        }
      }
    }
    if (first)
      printf("0");
    puts("");
    groups.clear();
  }
  return 0;
}
