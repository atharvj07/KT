import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] w = new int[n];

        for (int i = 0; i < n; i++) {
            w[i] = in.nextInt();
        }
        int total = 0;
        for (int i = 0; i < n; i++) {
            total += w[i];
        }

        if (total % 2 == 1) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
        }
    }
}
