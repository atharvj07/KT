import sys

def dfs(u, v):
    vis[u] = True
    ans.append(u)
    if u == v:
        return 1
    for x in g[u]:
        if not vis[x]:
            if dfs(x, v):
                return 1
    return 0

n, m = map(int, input().split())
g = [[]] * n
for i in range(m):
    x = input().split()
    if x[0] == '+':
        g[int(x[1]) - 1].append(int(x[2]) - 1)
        g[int(x[2]) - 1].append(int(x[1]) - 1)
    elif x[0] == '-':
        g[int(x[1]) - 1].remove(int(x[2]) - 1)
        g[int(x[2]) - 1].remove(int(x[1]) - 1)
    else:
        ans = []
        vis = [False] * n
        k = int(x[1])
        for i in range(n):
            if dfs(i, i):
                if len(ans) == k:
                    print('YES')
                    break
                else:
                    vis = [False] * n
        else:
            print('NO')
