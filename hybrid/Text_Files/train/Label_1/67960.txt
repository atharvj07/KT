n, l, r, ql, qr = map(int, input().split())
weights = list(map(int, input().split()))

prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sum[i] = prefix_sum[i - 1] + weights[i - 1]

min_energy = float('inf')
for i in range(n + 1):
    left = prefix_sum[i] * l
    right = (prefix_sum[n] - prefix_sum[i]) * r
    if i > n - i:
        left += (i - (n - i) - 1) * ql
    elif i < n - i:
        right += ((n - i) - i - 1) * qr
    min_energy = min(min_energy, left + right)

print(min_energy)
