import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), t = in.nextInt();
        char[] s = in.next().toCharArray();
        char[] a = s.clone();
        for (int i = 0; i < t; i++)
            for (int j = 0; j < n - 1; j++)
                if (s[j] == 'B' && s[j + 1] == 'G') {
                    a[j + 1] = 'B';
                    a[j] = 'G';
                    j++;
                }
        System.out.println(a);
    }
}
