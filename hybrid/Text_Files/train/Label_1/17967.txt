#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, p, q;
  cin >> n >> p >> q;
  string s;
  cin >> s;
  int k = (n + q - 1) / q;
  if (k * p != n) {
    cout << "-1\n";
  } else {
    vector<string> ans;
    for (int i = 0; i < k; i++) {
      if (i + 1 != k) {
        ans.push_back(s.substr(i * p, q));
      } else {
        ans.push_back(s.substr(i * p, n - i * p));
      }
    }
    cout << k << "\n";
    for (auto x : ans) {
      cout << x << "\n";
    }
  }
  return 0;
}
