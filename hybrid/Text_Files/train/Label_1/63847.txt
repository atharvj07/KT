MOD = 1000000007

def solve(n, s, w):
    total_ways = 0
    for i in range(n):
        if s[i] == '?' and w[i] == '?':
            total_ways = (total_ways + 81 * pow(10, n - i - 1, MOD) * (i + 1)) % MOD
        elif s[i] == '?':
            total_ways = (total_ways + int(w[i]) * pow(10, n - i - 1, MOD) * (i + 1) + (9 - int(w[i])) * pow(10, n - i - 1, MOD) * i) % MOD
        elif w[i] == '?':
            total_ways = (total_ways + int(s[i]) * pow(10, n - i - 1, MOD) * i + (9 - int(s[i])) * pow(10, n - i - 1, MOD) * (i + 1)) % MOD
        elif s[i] != w[i]:
            break
    return total_ways

n = int(input().strip())
s = input().strip()
w = input().strip()

print(solve(n, s, w))
