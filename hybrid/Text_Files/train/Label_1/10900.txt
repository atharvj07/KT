#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;

const int max_n = 150005;
const ll mod = 1e9 + 7;

int n;
ll ans[max_n];
map<ll, int> cnt;

void go(ll x, int l, int r) {
  int a = l, b = -1;
  for (int i = l; i <= r; i++) {
    if (ans[i] == x) {
      if (a == -1) {
        a = i;
      } else {
        b = i;
        break;
      }
    }
  }
  ans[a] = 2 * x;
  cnt[x] -= 2;
  ll nx = -1;
  for (auto [x, y] : cnt) {
    if (y >= 2) {
      nx = x;
      break;
    }
  }
  if (nx != -1) {
    go(nx, b + 1, r);
  }
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    ans[i] = x;
    cnt[x]++;
  }
  ll tmp = -1;
  for (auto [x, y] : cnt) {
    if (y >= 2) {
      tmp = x;
      break;
    }
  }
  if (tmp != -1) {
    go(tmp, 1, n);
    int k = 0;
    for (int i = 1; i <= n; i++) {
      if (ans[i] != 0) {
        k++;
      }
    }
    cout << k << endl;
    for (int i = 1; i <= n; i++) {
      if (ans[i] != 0) {
        cout << ans[i] << ' ';
      }
    }
    cout << endl;
  } else {
    cout << n << endl;
    for (int i = 1; i <= n; i++) {
      cout << ans[i] << ' ';
    }
  }
  return 0;
}
