n,k = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(n-1):
    x,y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)

vis = [False for i in range(n+1)]
p = [0 for i in range(n+1)]
color = [-1 for i in range(n+1)]
c = 1
q = [1]
vis[1] = True

while q:
    x = q.pop(0)
    for i in adj[x]:
        if vis[i] == True and i !=  p[x]:
            color[i] = color[x]
        elif vis[i] == False:
            p[i] = x
            q.append(i)
            vis[i] = True
            color[i] = c
            c+=1

# print(color)
print(c-1)
for i in range(2, n+1):
    print(color[i], end = " ")
