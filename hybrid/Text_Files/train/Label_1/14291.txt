#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int inf = 1e9;
const int mxN = 305;
int vis[mxN];
int main() {
    memset(vis, 0, sizeof vis);
    int m, t, r;
    cin >> m >> t >> r;
    vector<int> arr(m);
    for (int i = 0; i < m; i++) cin >> arr[i];
    int ans = 0;
    for (int i = 0, j = 0; i <= 300; i++) {
        if (j < m && i == arr[j]) {
            j++;
            if (vis[i - 1] < r) ans++;
        }
        vis[i] = vis[i - 1];
        if (i >= t) vis[i] = max(vis[i], vis[i - t + 1] + 1);
    }
    if (vis[300] < r) ans = -1;
    cout << ans << endl;
    return 0;
}
