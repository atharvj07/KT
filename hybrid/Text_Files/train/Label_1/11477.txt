#include<bits/stdc++.h>
using namespace std;
#define ms(x, y) memset(x, y, sizeof(x))
#define ull unsigned long long
#define ll long long
#define pi acos(-1)
#define mx 1000005
#define eps 1e-9
#define ff first
#define ss second
#define inf LONG_MAX
int n, a[mx];

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < n - 1; ++i) {
        if ((a[i] > a[i + 1])) {
            cout << i + 1 << " " << i + 2 << endl;
            return 0;
        }
    }
    cout << -1 << endl;
    return 0;
}
