#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, I;
  cin >> n >> I;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  sort(a.begin(), a.end());
  int ans = INT_MAX;
  for (int l = 0; l < n; l++) {
    int sum = 0;
    for (int i = l; i < n; i++) {
      if (a[i] < a[l]) sum += a[l] - a[i];
      else break;
    }
    int r = l;
    while (r + 1 < n && sum + a[r + 1] - a[l] <= I * 8 - (n - l)) {
      sum += a[r + 1] - a[l];
      r++;
    }
    if (r + 1 < n) ans = min(ans, l + n - r - 1);
  }
  cout << ans << "\n";
  return 0;
}
