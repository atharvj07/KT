t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    total_sum = sum(a)
    for ops in range(n):
        if total_sum % (n - ops) == 0:
            target = total_sum // (n - ops)
            curr_sum, parts = 0, 0
            for j in range(n):
                curr_sum += a[j]
                if curr_sum > target:
                    break
                if curr_sum == target:
                    curr_sum = 0
                    parts += 1
            if parts == n - ops:
                print(ops)
                break
