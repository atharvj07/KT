import sys,math

for T in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = sorted(a)
    if a == b:
        print("YES")
    else:
        if{a[0], a[-1]} <= {b[0], b[-1]}:
            print("YES")
        else:
            found = 0
            for i in range(n):
                if a[i] != b[i]:
                    for j in range(i+1,n):
                        if a[i] == b[j] and a[j] == b[i]:
                            found = 1
                            print("YES")
                            break
                    if found:
                        break
            if not found:
                print("NO")
