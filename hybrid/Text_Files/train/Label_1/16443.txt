const int mod = 998244353;
struct segtree {
    vector<int> tree;
    vector<int> arr;
    int n;
    segtree() {}
    segtree(vector<int>& arr_) {
        arr = arr_;
        n = arr.size();
        tree.resize(n * 4);
        build(1, 0, n - 1);
    }
    void build(int node, int start, int end) {
        if (start == end)
            tree[node] = arr[start];
        else {
            int mid = (start + end) / 2;
            build(node * 2, start, mid);
            build(node * 2 + 1, mid + 1, end);
            tree[node] = tree[node * 2] + tree[node * 2 + 1];
        }
    }
    void update(int index, int val) {
        update(1, index, 0, n - 1, val);
    }
    void update(int node, int index, int start, int end, int val) {
        if (start == end) {
            arr[index] = val;
            tree[node] = val;
            return;
        }
        int mid = (start + end) / 2;
        if (index <= mid)
            update(node * 2, index, start, mid, val);
        else
            update(node * 2 + 1, index, mid + 1, end, val);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    int query(int l, int r) {
        return query(1, 0, n - 1, l, r);
    }
    int query(int node, int start, int end, int l, int r) {
        if (l > end or r < start)
            return 0;
        if (l <= start and r >= end)
            return tree[node];
        int mid = (start + end) / 2;
        return query(node * 2, start, mid, l, r) + query(node * 2 + 1, mid + 1, end, l, r);
    }
};
int binpow(int a, int b) {
    int res = 1;
    while (b > 0) {
        if (b & 1)
            res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

    for (int i = 0; i < n; ++i) {

        if (a[i] == -1) a[i] = 1e9;

    segtree tr(a);
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        ans = (ans + i * (tr.query(i + 1, n - 1))) % mod;
        tr.update(i, i + 1);
    }
    int inv = binpow(n, mod - 2);
    cout << (ans * inv) % mod;
}
