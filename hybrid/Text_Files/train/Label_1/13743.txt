#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  bool ok = true;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a') {
      if (i % 2 == n % 2) {
        s.erase(i, 1);
        i--;
        n--;
      } else {
        ok = false;
        break;
      }
    }
  }
  if (!ok) {
    cout << ":(" << endl;
  } else {
    cout << s << endl;
  }
  return 0;
}
