#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, w; cin >> n >> w;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int ans = 0;
    for (int x : a) {
        int y = min(x / 2, w);
        ans += y; w -= y;
        if (w == 0) break;
    }
    if (w == 0) {
        for (int x : a) cout << max(w, (x + 1) / 2) << endl, w -= max(w, (x + 1) / 2);
    } else cout << -1 << endl;
    return 0;
}
