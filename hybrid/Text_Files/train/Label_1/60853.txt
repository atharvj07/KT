import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    l = list(map(int,input().split()))
    unlocked = []
    for i in range(n):
        if not l[i]:
            unlocked.append(a[i])
    unlocked.sort()
    idx = 0
    for i in range(n):
        if not l[i]:
            a[i] = unlocked[idx]
            idx += 1
    print(*a)
