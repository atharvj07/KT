#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int i = 0, n = s.size();
  while (i < n) {
    if (s[i] == 'y' && s[i + 1] == 'x') {
      char tmp = s[i];
      s[i] = s[i + 1];
      s[i + 1] = tmp;
      i += 2;
    } else if (s[i] == 'x' && s[i + 1] == 'y') {
      s.erase(i, 2);
      n -= 2;
    } else {
      i++;
    }
  }
  cout << s << endl;
  return 0;
}
