import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        boolean found = false;
        for (int i = 1; i < n; i++) {
            if (s.charAt(i - 1) > s.charAt(i)) {
                System.out.println("YES\n" + (i) + " " + (i + 1));
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("NO");
        }
    }
}
