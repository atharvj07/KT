import sys
input = sys.stdin.readline
mod = 998244353
def mi(x, p):
  return pow(x, p, mod)
n, m = map(int, input().split())
dist = []
for i in range(n):
  dist.append([int(x) for x in input().split()])
dp = [[0] * (n + 1) for _ in range(m + 1)]
dp[0][0] = 1
for j in range(m):
  for i in range(n):
    for k in range(dist[i][j], n + 1):
      dp[j + 1][k] += dp[j][k - dist[i][j]]
num = dp[m][n]
den = 1
for i in range(1, n + 1):
  den *= i
den = mi(den, mod - 2)
print((num * den) % mod)
