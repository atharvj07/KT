#include <cstdio>
#include <string>
using namespace std;

int main()
{
    char s[1000001];
    scanf("%s", s);
    int i;
    for(i = 0; s[i] != '\0'; i++)
        if(s[i] != '0')
            break;
    string a;
    while(s[i] != '\0')
        a.push_back(s[i++]);

    scanf("%s", s);
    for(i = 0; s[i] != '\0'; i++)
        if(s[i] != '0')
            break;
    string b;
    while(s[i] != '\0')
        b.push_back(s[i++]);

    int m = a.length(), n = b.length();
    if(m < n) printf("<");
    else if(m > n) printf(">");
    else
    {
        int flag = 0;
        while(m--)
        {
            if(a[m] > b[m])
            {
                flag = 1;
                break;
            }
            else if(a[m] < b[m])
            {
                flag = -1;
                break;
            }
        }
        if(flag == 0) printf("=");
        else if(flag == 1) printf(">");
        else printf("<");
    }

    return 0;
}
