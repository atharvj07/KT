def check_algorithm_works(s, l, r):
    if r - l < 2:
        return "YES"
    x = y = z = 0
    for c in s[l-1:r]:
        if c == 'x': x += 1
        elif c == 'y': y += 1
        elif c == 'z': z += 1
    if max(x, y, z) - min(x, y, z) > 1:
        return "NO"
    return "YES"

s = input().strip()
m = int(input().strip())
for _ in range(m):
    l, r = map(int, input().split())
    print(check_algorithm_works(s, l, r))
