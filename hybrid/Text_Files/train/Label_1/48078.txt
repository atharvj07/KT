import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] coins = new int[5];
        for (int i = 0; i < 5; i++) {
            coins[i] = scanner.nextInt();
        }

        int sum = 0;
        int min = Integer.MAX_VALUE;
        for (int coin : coins) {
            sum += coin;
            min = Math.min(min, coin);
        }

        if (sum % 4 != 0 || min < 0) {
            System.out.println(-1);
        } else {
            System.out.println(sum / 4);
        }
    }
}
