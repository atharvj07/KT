#include<bits/stdc++.h>
#define ll long long
#define pb emplace_back
using namespace std;
const ll maxn=1e5+5;
pair<ll,ll>v[maxn];
ll n,ans,m,t,h;
int main(){
    cin>>n>>m;
    while(n--){
        ll k,a,b,mx=0;
        cin>>k>>a>>b;
        if(k==1){
            v[++ans]={((a+ans-1)%1000000)+1,((b+ans-1)%1000000)+1};
            sort(v+1,v+ans+1);
        }else{
            t=((a+ans-1)%1000000)+1,h=((b+ans-1)%1000000)+1;
            for(ll i=ans;i;i--){
                if(t*(v[i].second)>m) break;
                if(t*(v[i].first)>=h) {
                    cout<<"YES\n";
                    goto XX;
                }
                m-=(t*(v[i].second));
                t-=v[i].first;
            }
            cout<<"NO\n";
XX:
            ;
        }
    }
    return 0;
}
