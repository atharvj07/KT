import java.util.Scanner;

public class Main {
    public static int cal(int n, int m, int d, int v, int s) {
        int start = (m - 1) * 30 + d;
        int end = start + v - 1;
        return Math.max(s - (360 - start), s - end);
    }

    public static void main(String[] args) {
        int min = 360;
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] m = new int[n];
        int[] d = new int[n];
        int[] v = new int[n];
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
            d[i] = sc.nextInt();
            v[i] = sc.nextInt();
            s[i] = sc.nextInt();
            if (m[i] == 12 && d[i] == 30) {
                min = 0;
            }
        }
        for (int i = 0; i < 360; i++) {
            int S = 0;
            for (int j = 0; j < n; j++) {
                S = Math.max(S, cal(n, m[j], d[j], v[j], s[j]));
            }
            min = Math.min(min, S);
        }
        System.out.println(min);
        sc.close();
    }
}
