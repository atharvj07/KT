#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
using ll = long long;

int main() {
    ll n, m, l, r;
    cin >> n >> m;
    vector<pair<ll, ll>> is(n);
    for (ll i = 0; i < n; ++i) {
        cin >> l >> r;
        is[i] = {l, r};
    }

    vector<ll> br(m);
    for (ll i = 0; i < m; ++i) cin >> br[i];
    sort(br.begin(), br.end());
    sort(is.begin(), is.end(), [&](pair<ll, ll>& a, pair<ll, ll>& b) {
        return a.second < b.second;
    });

    vector<ll> ans(n - 1);
    for (ll i = 0, j = 0; i < n - 1; ++i) {
        while (j < m && is[i].second - is[i].first + 1 > br[j]) ++j;
        if (j == m) { cout << "No\n"; return 0; }
        ans[i] = j + 1;
        ++j;
    }

    cout << "Yes\n";
    for (auto x : ans) cout << x << " ";
    cout << endl;

    return 0;
}
