import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
            String program = reader.readLine();
            String result = solve(program);
            System.out.println(result);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static String solve(String program) {
        int[] stack = new int[1024];
        int stackPointer = 0;
        char[] commands = program.toCharArray();
        int commandPointer = 0;
        int[] direction = new int[]{1, 0};
        int[][] grid = new int[1024][1024];
        int gridX = 0;
        int gridY = 0;
        List<Integer> result = new ArrayList<>();

        while (commandPointer < commands.length) {
            char command = commands[commandPointer++];
            switch (command) {
                case '<':
                    direction = new int[]{-1, 0};
                    break;
                case '>':
                    direction = new int[]{1, 0};
                    break;
                case '^':
                    direction = new int[]{0, -1};
                    break;
                case 'v':
                    direction = new int[]{0, 1};
                    break;
                case '_':
                    if (stackPointer > 0) {
                        stackPointer--;
                        grid[gridX][gridY] = stack[stackPointer];
                    }
                    break;
                case '*':
                    if (stackPointer > 1) {
                        int a = stack[stackPointer - 1];
                        int b = stack[stackPointer - 2];
                        stack[stackPointer - 2] = a * b;
                        stackPointer--;
                    }
                    break;
                case '/':
                    if (stackPointer > 1) {
                        int a = stack[stackPointer - 1];
                        int b = stack[stackPointer - 2];
                        stack[stackPointer - 2] = b / a;
                        stackPointer--;
                    }
                    break;
                case '+':
                    if (stackPointer > 1) {
                        int a = stack[stackPointer - 1];
                        int b = stack[stackPointer - 2];
                        stack[stackPointer - 2] = a + b;
                        stackPointer--;
                    }
                    break;
                case '-':
                    if (stackPointer > 1) {
                        int a = stack[stackPointer - 1];
                        int b = stack[stackPointer - 2];
                        stack[stackPointer - 2] = b - a;
                        stackPointer--;
                    }
                    break;
                case '0':
                case '1':
                case '2':
                case '3':
                case '4':
                case '5':
                case '6':
                case '7':
                case '8':
                case '9':
                    stack[stackPointer++] = command - '0';
                    break;
                case '?':
                    int value = grid[gridX][gridY];
                    if (value == 0) {
                        return "false";
                    }
                    result.add(value);
                    break;
            }
            gridX += direction[0];
            gridY += direction[1];
        }
        StringBuilder resultBuilder = new StringBuilder();
        for (int num : result) {
            resultBuilder.append(num);
        }
        return resultBuilder.toString();
    }
}
