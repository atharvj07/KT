chessboard = []
for i in range(8):
    line = input()
    chessboard.append(line)

white_weight = 0
black_weight = 0

for i in range(8):
    for j in range(8):
        c = chessboard[i][j]
        if c == 'Q':
            white_weight += 9
        elif c == 'R':
            white_weight += 5
        elif c == 'B':
            white_weight += 3
        elif c == 'N':
            white_weight += 3
        elif c == 'P':
            white_weight += 1
        elif c == 'q':
            black_weight += 9
        elif c == 'r':
            black_weight += 5
        elif c == 'b':
            black_weight += 3
        elif c == 'n':
            black_weight += 3
        elif c == 'p':
            black_weight += 1

if white_weight > black_weight:
    print("White")
elif black_weight > white_weight:
    print("Black")
else:
    print("Draw")
