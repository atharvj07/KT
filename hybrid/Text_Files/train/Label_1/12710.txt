#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> ii;
const int mx = 31400 + 10;
bitset<mx> R[mx];
int main() {
    int n, x[4];
    cin >> n;
    while(n--) {
        for(int i = 0; i < 4; i++) {
            cin >> x[i];
        }
        R[x[0]][x[1]] = true;
        R[x[0]][x[3]] = false;
        R[x[2]][x[1]] = false;
        R[x[2]][x[3]] = true;
    }
    for(int i = 0; i < mx; i++) R[i] ^= R[i - 1];
    bool ok = true;
    for(int x1 = 0, x2 = 0; x1 < mx; x1++) {
        if(x2 < x1) x2 = x1;
        while(x2 < mx && R[x2] == R[x1]) x2++;
        ok &= (x2 - x1 >= 3);
    }
    cout << (ok ? "YES" : "NO");
}
