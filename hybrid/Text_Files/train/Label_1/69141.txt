from collections import Counter
n, k = map(int, input().split())
favs = Counter()
for _ in range(k):
    a, b = map(int, input().split())
    favs[a] += 1
    favs[b] += 1
vis = [0] * (n + 1)
ans = 0
for i in favs:
    if vis[i]: continue
    vis[i] = 1
    if favs[i] % 2 == 0:
        ans += int(favs[i] / 2)
    else:
        ans += int(favs[i] / 2)
        favs[i] -= 1
        x = []
        for j in favs:
            if not vis[j] and favs[j] % 2 == 1 and favs[j] > 0:
                favs[j] -= 1
                x.append(j)
            if favs[j] == 0:
                vis[j] = 1
        ans += len(x) // 2
print(ans)
