n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
b = [False] * (100001)
b[0] = b[1] = False
b[2] = True
for i in range(4, 100001, 2):
    b[i] = False
for i in range(3, 100001, 2):
    if b[i]:
        for j in range(i * i, 100001, i):
            b[j] = False
ans = 100000000000000
for i in range(n):
    tmp = -1
    for j in range(m):
        if b[a[i][j]]:
            tmp = 0
        else:
            tmp = min(a[i][j] - 1, tmp) if tmp >= 0 else a[i][j] - 1
    ans = min(ans, tmp)
for j in range(m):
    tmp = -1
    for i in range(n):
        if b[a[i][j]]:
            tmp = 0
        else:
            tmp = min(a[i][j] - 1, tmp) if tmp >= 0 else a[i][j] - 1
    ans = min(ans, tmp)
print(ans)
