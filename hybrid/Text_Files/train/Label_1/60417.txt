def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, p, c):
    p = [x-1 for x in p]  # Convert to 0-based indexing
    ans = n  # Set ans to the maximum possible value initially
    
    for i in range(n):
        cycle_length = 0
        j = i
        # Find the length of the cycle
        while True:
            j = p[j]
            cycle_length += 1
            if j == i:
                break
        
        for start in range(cycle_length):
            g = gcd(cycle_length, start)
            flag = True
            for k in range(g):
                j = i
                color = c[i]
                
                for _ in range(start):
                    j = p[j]
                
                for _ in range(cycle_length // g):
                    if c[j] != color:
                        flag = False
                        break
                    j = p[j]
                    for _ in range(g-1):
                        j = p[j]
                if flag:
                    ans = min(ans, g)
                    break
    return ans

T = int(input())
for _ in range(T):
    n = int(input())
    p = list(map(int, input().split()))
    c = list(map(int, input().split()))
    print(solve(n, p, c))
