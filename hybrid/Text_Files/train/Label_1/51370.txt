import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int n = sc.nextInt();
      if (n == 0) {
        break;
      }
      long[] tres = new long[n];
      long[] max = new long[n];
      for (int i = 0; i < n; i++) {
        tres[i] = sc.nextInt();
        max[i] = sc.nextInt();
      }
      System.out.println(canCollect(tres, max) ? "Yes" : "No");
    }
    sc.close();
  }

  public static boolean canCollect(long[] tres, long[] max) {
    long[] dp = new long[1 << tres.length];
    int all = (1 << tres.length) - 1;
    dp[all] = 0;
    for (int i = all - 1; i >= 0; i--) {
      for (int j = 0; j < tres.length; j++) {
        if ((i & (1 << j)) == 0) {
          long bal = dp[i | (1 << j)] - tres[j];
          if (bal >= 0 && bal <= max[j]) {
            dp[i] = Math.max(dp[i], bal);
          }
        }
      }
    }
    return dp[0] >= 0;
  }
}
