#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    int mx = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            mx = __gcd(mx, (v[i] * v[j]) / __gcd(v[i], v[j]));
        }
    }
    cout << mx << endl;
}
