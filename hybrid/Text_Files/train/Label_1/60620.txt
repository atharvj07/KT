import sys

def check_chronological(data):
    prev_x = -1
    prev_k = -1
    for x, k in data:
        if x < prev_x or (x == prev_x and k <= prev_k):
            return "NO"
        prev_x = x
        prev_k = k
    return "YES"

n = int(sys.stdin.readline())
data = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]
print(check_chronological(data))
