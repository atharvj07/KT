import sys
input = sys.stdin.readline
n, k, a, b, q = (int(x) for x in input().split())
d = [0] * (n + 1)
for _ in range(q):
    cmd, *nums = (int(x) for x in input().split())
    if cmd == 1:
        di, ai = nums
        d[di] += ai
    elif cmd == 2 and n-k+1<=nums[0]<=n:
        pi = nums[0]
        ans, rem = 0, 0
        for i in range(pi, pi+k-1):
            rem += min(b, d[i])
        for i in range(pi+k, n+1):
            rem = min(a, rem+b)
            ans += min(rem, d[i])
        print(ans)
