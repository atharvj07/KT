import sys

n = int(sys.stdin.readline())
p = [int(x) for x in sys.stdin.readline().split()]
adj = {i: set() for i in range(1, n+1)}
for i in range(2, n+1):
    adj[p[i-1]].add(i)
    adj[i].add(p[i-1])

subtreesize = [1] * (n+1)
def dfs(u):
    for v in adj[u]:
        dfs(v)
        subtreesize[u] += subtreesize[v]

dfs(1)

result = [1] * (n+1)
happy = [1] * (n+1)
def dfs2(u, need):
    curcolor = 1
    for v in adj[u]:
        if happy[v] == -1:
            dfs2(v, need)
        curcolor = max(curcolor, result[v]+1)
    if subtreesize[u] == 1:
        happy[u] = 1
    else:
        if curcolor <= need:
            happy[u] = 1
            result[u] = curcolor
        else:
            happy[u] = -1
            result[u] = curcolor+1

for k in range(1, n+1):
    dfs2(1, k)
    print(result[1])
