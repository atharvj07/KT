#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  int mid = (n + 1) / 2;
  bool isS = true;
  for (int i = 0; i < mid; i++) {
    if (s[i] != s[n - 1 - i]) {
      isS = false;
      break;
    }
  }
  if (isS)
    cout << "TAK" << endl;
  else
    cout << "NIE" << endl;
  return 0;
}
