def count_paving_ways(A, B, C):
    ways = 0
    for a in range(1, A+1):
        for b in range(a, B+1):
            for c in range(b, C+1):
                if A % a == 0 and B % b == 0 and C % c == 0:
                    ways += 1
    return ways

t = int(input())
for _ in range(t):
    A, B, C = map(int, input().split())
    print(count_paving_ways(A, B, C))
