import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        Song[] songs = new Song[n];
        for (int i = 0; i < n; i++) {
            songs[i] = new Song(in.nextInt(), in.nextInt());
        }
        Arrays.sort(songs);
        long ans = 0;
        for (int i = 0; i < k; i++) {
            ans += songs[i].length;
        }
        ans *= songs[k - 1].beauty;
        System.out.println(ans);
    }

    static class Song implements Comparable<Song> {
        int length;
        int beauty;

        public Song(int length, int beauty) {
            this.length = length;
            this.beauty = beauty;
        }

        @Override
        public int compareTo(Song o) {
            return o.beauty - this.beauty;
        }
    }
}
