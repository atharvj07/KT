n, m = map(int, input().split())
debts = {}
for i in range(m):
    a, b, c = map(int, input().split())
    debts[(a, b)] = c
total = 0
for i in range(1, n + 1):
    for j in range(i + 1, n + 1):
        if (i, j) in debts:
            total += debts[(i, j)]
        elif (j, i) in debts:
            total -= debts[(j, i)]
print(abs(total))
