#include <bits/stdc++.h>
using namespace std;
#define FOR(i, x, y) for (int i = (x), i##_ = (y); i <= i##_; ++i)
#define ROF(i, x, y) for (int i = (x), i##_ = (y); i >= i##_; --i)
const int N = 2e5 + 5;
vector<vector<int>> g;
vector<int> sub, ord;
void dfs(int u, int p) {
  sub[u] = 1;
  for (auto& v : g[u]) {
    if (v == p) continue;
    dfs(v, u);
    sub[u] += sub[v];
  }
  ord.push_back(u);
}
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int n, q;
  cin >> n >> q;
  g.resize(n + 1);
  sub.resize(n + 1);
  FOR(i, 2, n) {
    int p;
    cin >> p;
    g[i].push_back(p), g[p].push_back(i);
  }
  dfs(1, 1);
  while (q--) {
    int u, k;
    cin >> u >> k;
    cout << ((sub[u] >= k) ? ord[k - 1] : -1) << "\n";
  }
}
