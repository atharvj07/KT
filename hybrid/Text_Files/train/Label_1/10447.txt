#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (auto& e : v) {
    cin >> e.first >> e.second;
  }
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = 0; i < n / 2; ++i) {
    ans += (v[i].first + v[n - i - 1].second) * (v[i].second + v[n - i - 1].first);
  }
  cout << ans << endl;
  return 0;
}
