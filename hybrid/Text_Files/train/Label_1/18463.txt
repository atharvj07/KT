#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n, k, temp; cin >> n >> k;
        vector<int> taps;
        for (int i = 0; i < k; i++) { cin >> temp; taps.push_back(temp); }
        sort(taps.begin(), taps.end());
        int t = 0;
        for (int i = 1; i < k; i++) { t = max(t, (taps[i] - taps[i - 1] + 1) / 2); }
        t = max(t, n - taps[k - 1]);
        cout << t << endl;
    }
    return 0;
}
