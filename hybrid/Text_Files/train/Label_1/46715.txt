import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        for (int i = 0; i < t; i++) {
            String[] s = br.readLine().split(" ");
            int n = Integer.parseInt(s[0]);
            int x = Integer.parseInt(s[1]);
            s = br.readLine().split(" ");
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = Integer.parseInt(s[j]);
            }
            boolean[] infected = new boolean[n];
            infected[0] = true;
            int cnt = 0;
            while (true) {
                boolean allInfected = true;
                for (int j = 0; j < n; j++) {
                    if (!infected[j]) {
                        allInfected = false;
                        break;
                    }
                }
                if (allInfected) {
                    break;
                }
                cnt++;
                int[] change = new int[n];
                for (int j = 0; j < n; j++) {
                    if (infected[j]) {
                        continue;
                    }
                    change[j] = x - a[j];
                }
                int sum = 0;
                for (int j = 0; j < n; j++) {
                    sum += change[j];
                }
                for (int j = 0; j < n; j++) {
                    change[j] -= sum / n;
                }
                for (int j = 0; j < n; j++) {
                    a[j] += change[j];
                    if (!infected[j] && a[j] == x) {
                        infected[j] = true;
                    }
                }
            }
            System.out.println(cnt);
        }
    }
}
