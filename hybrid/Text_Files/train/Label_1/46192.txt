import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean flag = true;
        Set<Integer> set = new HashSet<>();
        while (n-- > 0) {
            set.clear();
            for (int i = 0; i < 6; i++) {
                set.add(sc.nextInt());
            }
            if (set.size() != 6) {
                flag = false;
            }
        }
        System.out.println(flag ? "Yes" : "No");
    }
}
