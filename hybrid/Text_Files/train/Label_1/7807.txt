#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int p, f, cnt_s, cnt_w, s, w;
        cin >> p >> f >> cnt_s >> cnt_w >> s >> w;
        int ans = 0;
        while (p > 0 && cnt_s > 0) {
            ans++;
            cnt_s--;
            p -= s;
        }
        while (f > 0 && cnt_w > 0) {
            ans++;
            cnt_w--;
            f -= w;
        }
        while (cnt_s > 0 && cnt_w > 0 && p > 0 && f > 0) {
            ans += 2;
            cnt_s--;
            cnt_w--;
            f -= w;
            p -= s;
        }
        cout << ans << endl;
    }
    return 0;
}
