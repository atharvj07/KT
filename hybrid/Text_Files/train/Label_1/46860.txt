import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long p = sc.nextLong();
        long x = sc.nextLong();

        long ans = 0;
        long cur = 1;
        for (long i = 1; i <= x; i++) {
            if (cur % p == b) {
                ans++;
            }
            cur = (cur * a) % p;
        }

        System.out.println(ans);
    }
}
