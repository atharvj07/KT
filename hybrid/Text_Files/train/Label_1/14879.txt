#include <bits/stdc++.h>
using namespace std;
vector<int> arr;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  string s;
  cin >> s;
  s = ' ' + s;
  int n = s.size();
  for (int i = 1; i < n; i++) {
    if (s[i] == s[i + 1]) { arr.push_back(i); }
  }
  long long int ans = 0;
  for (int i = 0; i < arr.size(); i++) {
    ans += n - arr[i] - i;
  }
  cout << ans << "\n";
}
