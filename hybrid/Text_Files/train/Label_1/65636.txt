import sys

def modpow(base, exp, mod):
    if exp == 0:
        return 1
    if exp == 1:
        return base % mod
    val = modpow(base, exp // 2, mod)
    val = (val * val) % mod
    if exp % 2 != 0:
        val = (val * base) % mod
    return val

n = int(input())
a, b, c, d = [0] * 4
for _ in range(n):
    s, p = input().split()
    if s == 'ADD':
        if p not in [a, c]:
            if b < int(p) < d:
                b = int(p)
            elif int(p) < b:
                a = int(p)
            else:
                c = int(p)
    else:
        if int(p) == a:
            a = b
        elif int(p) == b:
            b = c
        elif int(p) == c:
            c = d
        else:
            d = int(p)

ans = 1
if a == 0:
    ans = 0
elif b == 0:
    ans = 2
elif c == 0:
    ans = 1
elif d == 0:
    ans = 2
print(ans)
