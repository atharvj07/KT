#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            continue;
        }
        if (ans == -1) {
            ans = 0;
        }
        int g = __gcd(a[i], ans);
        if (g == 1) {
            ans = -1;
            break;
        }
        ans = ans / g * a[i];
    }
    cout << ans << endl;
    return 0;
}
