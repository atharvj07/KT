s = input()
l = len(s)
pref = [0] * l
for i in range(1, l):
    if s[i] == s[i - 1]:
        pref[i] = pref[i - 1] + 1
d = dict()
ans = 0
for i in range(l):
    j = i
    cnt1 = 0
    cnt2 = 0
    while j < l:
        if s[j] not in d:
            d[s[j]] = 0
        if d[s[j]] == cnt1:
            cnt2 += 1
        else:
            cnt1 += 1
        if cnt2 > 1:
            break
        j += 1
    j = i
    while j >= 0:
        if s[j] not in d:
            d[s[j]] = 0
        if d[s[j]] == cnt1:
            cnt2 += 1
        else:
            cnt1 += 1
        if cnt2 > 1:
            break
        j -= 1
    if cnt2 == 1:
        ans += 1
print(ans / l)
