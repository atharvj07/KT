#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;

    string s;
    cin >> s;

    vector<int> digits(n);
    for (int i = 0; i < n; i++) {
        digits[i] = s[i] - '0';
    }

    for (int i = 0; i < n && k > 0; i++) {
        if (digits[i] == 0) {
            int j;
            for (j = i + 1; j < n && digits[j] == 0; j++) {}
            if (j < n) {
                digits[i] = digits[j];
                digits[j] = 0;
                k--;
            }
        }
        if (digits[i] != 0) {
            digits[i] = min(digits[i], 1);
        }
    }

    for (int i = 0; i < n; i++) {
        cout << digits[i];
    }
    cout << endl;

    return 0;
}
