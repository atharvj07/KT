#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct event {
    int c, w, type;
    bool operator<(const event& e) const {
        return c < e.c;
    }
};

int main() {
    int n, m, d;
    cin >> n >> m >> d;
    vector<event> events;
    for (int i = 0; i < n; i++) {
        int c, w;
        cin >> c >> w;
        events.push_back({c, w, 1});
    }
    for (int i = 0; i < m; i++) {
        int c, w;
        cin >> c >> w;
        events.push_back({c, w, 2});
    }
    sort(events.begin(), events.end());
    int ans = 0, cur = 0;
    for (auto& e : events) {
        if (e.type == 1 && cur + e.w <= d) {
            ans += e.c;
            cur += e.w;
        } else if (e.type == 2 && cur + e.w <= d) {
            ans += e.c;
            cur += e.w;
        }
    }
    if (cur == 0) ans = 0;
    cout << ans << endl;
    return 0;
}
