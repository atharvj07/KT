#include <iostream>
#include <vector>
using namespace std;

int n;
vector<int> seq[200001];
int seen[200001];
int ans[200001];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int a, b;
    cin >> a >> b;
    seq[i].push_back(a);
    seq[i].push_back(b);
  }
  int idx = -1;
  for (int i = 1; i <= n; i++) {
    if (!seen[i]) {
      idx = i;
      break;
    }
  }
  int cnt = 0;
  while (!seen[idx]) {
    seen[idx] = true;
    ans[cnt++] = idx;
    idx = seq[idx][0] == ans[cnt - 2] ? seq[idx][1] : seq[idx][0];
  }
  for (int i = 0; i < n; i++) cout << ans[i] << ' ';
  return 0;
}
