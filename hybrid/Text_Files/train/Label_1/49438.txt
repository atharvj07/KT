import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();

        int m = scanner.nextInt();
        int[][] statements = new int[m][3];
        for (int i = 0; i < m; i++) {
            statements[i][0] = scanner.nextInt();
            statements[i][1] = scanner.nextInt();
            statements[i][2] = scanner.nextInt();
        }

        int k = scanner.nextInt();

        int[] sequence = new int[n];
        for (int i = 0; i < n; i++) {
            sequence[i] = a + d * i;
        }

        for (int[] statement : statements) {
            int type = statement[0];
            int start = statement[1] - 1;
            int end = statement[2] - 1;

            if (type == 0) {
                reverse(sequence, start, end);
            } else if (type == 1) {
                increment(sequence, start, end);
            } else if (type == 2) {
                halve(sequence, start, end);
            }
        }

        System.out.println(sequence[k - 1]);
    }

    private static void reverse(int[] sequence, int start, int end) {
        for (int i = start, j = end; i < j; i++, j--) {
            int temp = sequence[i];
            sequence[i] = sequence[j];
            sequence[j] = temp;
        }
    }

    private static void increment(int[] sequence, int start, int end) {
        for (int i = start; i <= end; i++) {
            sequence[i]++;
        }
    }

    private static void halve(int[] sequence, int start, int end) {
        for (int i = start; i <= end; i++) {
            sequence[i] /= 2;
        }
    }
}
