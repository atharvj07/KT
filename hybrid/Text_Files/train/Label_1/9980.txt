#include <iostream>
#include <vector>

int main() {
    int x, h, m;
    std::cin >> x >> h >> m;
    std::vector<int> h7, m7;
    int a = h / 10, b = h % 10, c = m / 10, d = m % 10;
    if (a == 7 || b == 7 || c == 7 || d == 7) {
        std::cout << 0 << std::endl;
        return 0;
    }
    for (int i = 0; i < 60; i++) {
        if (i / 10 == 7 || i % 10 == 7) h7.push_back(i);
        if ((h + i) / 10 == 7 || (h + i) % 10 == 7) h7.push_back(h + i);
    }
    for (int i = 0; i < 60; i++) {
        if (i / 10 == 7 || i % 10 == 7) m7.push_back(i);
        if ((m + i) / 10 == 7 || (m + i) % 10 == 7) m7.push_back(m + i);
    }
    int ans = 0;
    while (true) {
        bool found = false;
        for (int i = 0; i < h7.size(); i++) {
            for (int j = 0; j < m7.size(); j++) {
                if ((h + ans * x) % 60 == h7[i] && (m + ans * x) % 60 == m7[j]) {
                    found = true;
                    std::cout << ans << std::endl;
                    return 0;
                }
            }
        }
        if (!found) ans++;
    }
    return 0;
}
