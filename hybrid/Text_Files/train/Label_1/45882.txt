import java.util.Scanner;

public class BubbleCupFinals {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int rows = scanner.nextInt();
        int cols = scanner.nextInt();

        int[][] field = new int[rows][cols];
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                field[i][j] = scanner.nextInt();
            }
        }

        long maxBubbles = findMaxBubbles(field, rows, cols);
        System.out.println(maxBubbles);
    }

    private static long findMaxBubbles(int[][] field, int rows, int cols) {
        long[][] dp = new long[rows + 1][cols + 1];

        for (int i = 1; i <= rows; i++) {
            for (int j = 1; j <= cols; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]) + field[i - 1][j - 1];
            }
        }

        return dp[rows][cols];
    }
}
