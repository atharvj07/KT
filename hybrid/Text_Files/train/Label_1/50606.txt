import java.util.*;
import java.time.*;
import java.time.temporal.ChronoUnit;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        PriorityQueue<LogEntry> pq = new PriorityQueue<>((a, b) -> a.time.compareTo(b.time));
        List<LogEntry> logs = new ArrayList<>();
        while (sc.hasNext()) {
            LogEntry entry = new LogEntry(sc.nextLine(), n, m);
            if (entry.valid) {
                logs.add(entry);
            }
        }
        Collections.sort(logs, (a, b) -> a.time.compareTo(b.time));
        for (LogEntry entry : logs) {
            while (!pq.isEmpty() && pq.peek().time.plus(n, ChronoUnit.SECONDS).compareTo(entry.time) <= 0) {
                pq.poll();
            }
            pq.offer(entry);
            if (pq.size() >= m) {
                System.out.println(entry.time);
                return;
            }
        }
        System.out.println(-1);
    }

    static class LogEntry {
        LocalDateTime time;
        boolean valid;

        public LogEntry(String s, int n, int m) {
            try {
                time = LocalDateTime.parse(s.substring(0, 19));
                valid = true;
            } catch (Exception e) {
                valid = false;
            }
        }
    }
}
