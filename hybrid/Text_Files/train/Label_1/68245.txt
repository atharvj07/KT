n, k = map(int, input().split())
a = list(map(int, input().split()))
s, p = [a[0]], [a[0]]
for x in a[1:]:
    s.append(s[-1] + x)
    p.append(p[-1] * x)
ans = 0
for l in range(n):
    for r in range(l, n):
        if p[r] * (s[l] if l else 1) == k * p[l - 1] * (s[r + 1] if r + 1 < n else 1):
            ans += 1
print(ans)
