#include <bits/stdc++.h>
using namespace std;
const int p=1e9+7;
const int N=1e6+1;
long long smallestPrime[N];
void sieve()
{
    smallestPrime[1]=-1;
    for(long long i=2; i<N; i++)
    {
        if(smallestPrime[i]==0)
        {
            smallestPrime[i]=i;
            for(long long j=i*i; j<N; j+=i)
            {
                if(smallestPrime[j]==0)
                {
                    smallestPrime[j]=i;
                }
            }
        }
    }
}
int main()
{
    sieve();
    long long t;
    cin>>t;
    while(t--)
    {
        long long n;
        cin>>n;
        int ans=0;
        while(n>1)
        {
            ans++;
            n/=smallestPrime[n];
        }
        cout<<ans<<endl;
    }
    return 0;
}
