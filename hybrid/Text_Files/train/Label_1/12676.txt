#include <bits/stdc++.h>
using namespace std;
int a[110], b[110];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i] >> b[i];
    bool f = 1;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++)
            if (a[i] > b[j]) f = 0;
    if (b[0] >= m)
        f = 1;
    else
        for (int i = 0; i < n; i++)
            if (b[i] >= m) f = 1;
    if (f)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
}
