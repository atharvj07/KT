from collections import defaultdict
n, m, q = map(int, input().split())
p = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
dic = defaultdict(list)
for i in range(m):
    dic[a[i]].append(i)
ans = ['0'] * q
for i in range(q):
    l, r = map(int, input().split())
    if len(dic[p[0]]) == 0:
        ans[i] = '0'
        continue
    l -= 1
    r -= 1
    if dic[p[0]][-1] < l:
        ans[i] = '0'
        continue
    ind = bisect.bisect_left(dic[p[0]], l)
    if len(dic[p[0]]) == ind or dic[p[0]][ind] > r:
        ans[i] = '0'
    else:
        ans[i] = '1'
print(''.join(ans))
