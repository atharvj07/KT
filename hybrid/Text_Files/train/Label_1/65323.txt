def solve(n):
    mi, ma = -2e9, 2e9
    for _ in range(n):
        op, x, res = input().split()
        v = int(x)
        if res == 'Y':
            if op == '>':
                mi = max(mi, v + 1)
            elif op == '<':
                ma = min(ma, v - 1)
            elif op == '>=':
                mi = max(mi, v)
            else:
                ma = min(ma, v)
        else:
            if op == '>':
                ma = min(ma, v - 1)
            elif op == '<':
                mi = max(mi, v + 1)
            elif op == '>=':
                ma = min(ma, v - 1)
            else:
                mi = max(mi, v + 1)
    if mi <= ma:
        return int((mi + ma) / 2)
    else:
        return "Impossible"

n = int(input())
print(solve(n))
