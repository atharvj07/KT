#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    while(cin >> t) {
        while(t--) {
            string s;
            cin >> s;
            int a[26] = {0}, odd = 0, even = 0;
            for(int i = 0; i < s.length(); i++) {
                a[s[i] - 'a']++;
            }
            for(int i = 0; i < 26; i++) {
                if(a[i]%2 == 0) ++even;
                else ++odd;
            }
            if(odd > 1 || even == 26) cout << -1 << endl;
            else {
                string ans = "";
                for(int i = 0; i < 26; i++) {
                    if(a[i] == 0) continue;
                    if(a[i]%2 == 0) {
                        for(int j = 0; j < a[i]/2; j++) {
                            ans += (i + 'a');
                        }
                    } else {
                        ans += (i + 'a');
                        for(int j = 0; j < a[i]/2; j++) {
                            ans += (i + 'a');
                        }
                    }
                }
                cout << ans << endl;
            }
        }
    }
    return 0;
}
