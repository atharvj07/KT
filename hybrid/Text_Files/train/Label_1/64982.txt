a, b, n = map(int, input().split())
MOD = 1000000007

def power(base, exp):
    if exp == 0:
        return 1
    if exp == 1:
        return base
    val = power(base, exp // 2)
    val = (val * val) % MOD
    if exp % 2 != 0:
        val = (val * base) % MOD
    return val

def good_number(n, a, b):
    if n == 0:
        return 1
    if n == 1:
        if a <= b:
            return 2
        else:
            return 1
    val = good_number(n - 1, a, b)
    val = (val * (b - a + 1)) % MOD
    return val

def excellent_number(n, a, b):
    if n == 0:
        return 1
    if n == 1:
        if a == b:
            return 1
        else:
            return 2
    val = excellent_number(n - 1, a, b)
    val = (val * (good_number(n - 1, a, b) - 2)) % MOD
    val = (val + 2) % MOD
    return val

print(excellent_number(n, a, b))
