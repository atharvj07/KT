import java.util.*;
import java.io.*;

class Main {
    static int[] gasStations;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // number of cars
        int k = sc.nextInt(); // number of gas stations
        int s = sc.nextInt(); // length of the road
        int t = sc.nextInt(); // time in which the film starts

        int[] cars = new int[n]; // price of each car
        int[] capacities = new int[n]; // fuel tank capacity of each car
        for (int i = 0; i < n; i++) {
            cars[i] = sc.nextInt();
            capacities[i] = sc.nextInt();
        }

        gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = sc.nextInt();
        }

        long result = -1;
        for (int i = 0; i < n; i++) {
            long time = getTimeToReachCinema(i, cars[i], capacities[i], s, t);
            if (time != -1 && (result == -1 || time < result)) {
                result = time;
            }
        }

        System.out.println(result);
    }

    // Returns the time taken by the ith car to reach the cinema with the given parameters
    public static long getTimeToReachCinema(int carIndex, int price, int capacity, int s, int t) {
        int speed = 1; // 1 km/min in accelerated mode
        int fuelConsumption = 2; // 2 liters/km in accelerated mode
        int distanceCovered = 0;
        int timeTaken = 0;
        int fuelLeft = capacity;

        // Loop through the gas stations
        for (int i = 0; i < gasStations.length; i++) {
            // Check if the car has enough fuel to reach the next gas station
            int distanceToNextStation = gasStations[i] - distanceCovered;
            if (distanceToNextStation > fuelLeft * speed) {
                // If not, switch to normal mode and drive to the next gas station
                speed = 2;
                fuelConsumption = 1;
                timeTaken += distanceToNextStation / speed;
                fuelLeft = capacity;
            }

            // Fill up the tank at the gas station
            fuelLeft = capacity;

            // Check if the car has reached the cinema
            if (distanceCovered + speed >= s) {
                // If so, return the time taken
                timeTaken += (s - distanceCovered) / speed;
                return timeTaken;
            }

            // Otherwise, drive to the next gas station in accelerated mode
            distanceCovered += speed;
            timeTaken++;
            fuelLeft -= fuelConsumption;
        }

        // If the car has not reached the cinema, return -1
        return -1;
    }
}
