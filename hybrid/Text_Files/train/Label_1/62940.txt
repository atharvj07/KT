import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
        if sum(a) != sum(b):
            print(-1)
            continue
        m = 0
        while a != b:
            for i in range(n):
                for j in range(n):
                    if a[i] > b[i] and a[j] < b[j]:
                        a[i] -= 1
                        a[j] += 1
                        print(i + 1, j + 1)
                        m += 1
        print(m)

solve()
