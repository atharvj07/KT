#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int l = 0, r = n - 1; l < r; ++l) {
        ans += (a[l] + a[r]) * (a[l] + a[r]);
        --r;
    }
    cout << ans << endl;
    return 0;
}
