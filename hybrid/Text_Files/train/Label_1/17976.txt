#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;

vector<int> adj[N];
int vis[N], s, t, m;

bool dfs(int u){
    vis[u] = 1;
    if(u == t)
        return true;
    for(auto v : adj[u]){
        if(!vis[v]){
            if(dfs(v))
                return true;
        }
    }
    return false;
}

int main(){
    int n, u, v;
    cin >> n >> m;
    for(int i = 0; i < m; i++){
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    long long ans = 1LL * n * (n - 1) / 2;
    for(int i = 1; i <= n; i++){
        if(!vis[i]){
            s = i;
            for(int j = i + 1; j <= n; j++)
                if(!vis[j]){
                    t = j;
                    if(dfs(s))
                        ans -= 1LL * (n - i) * (n - j) / 2;
                }
        }
    }
    cout << ans << '\n';
    return 0;
}
