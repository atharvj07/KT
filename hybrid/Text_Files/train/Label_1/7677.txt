#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main()
{
    int t, n;
    cin >> t;
    while (t--)
    {
        cin >> n;
        vector<int> v(n);
        for (int i = 0; i < n; ++i)
            cin >> v[i];
        sort(v.begin(), v.end());
        int sum = 0;
        for (int i = 0; i < n; ++i)
            sum += v[i];
        if (sum % 2 == 0)
            cout << "Second\n";
        else
            cout << "First\n";
    }
    return 0;
}
