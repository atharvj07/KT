n, m = map(int, input().split())
f = [list(input()) for _ in range(n)]
vis = [[0 for j in range(m)] for i in range(n)]
dx = [1, -1, 0, 0]
dy = [0, 0, 1, -1]

for x in range(n):
    for y in range(m):
        if vis[x][y]: continue
        for dd in range(4):
            nx, ny = x + dx[dd], y + dy[dd]
            if 0 <= nx < n and 0 <= ny < m and f[nx][ny] == f[x][y]:
                vis[x][y] = 1
                if vis[nx][ny] and (nx != x or ny != y): print("Yes"); exit()
                vis[nx][ny] = 1
print("No")
