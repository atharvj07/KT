#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
typedef long long ll;

int main()
{
	string a;
	cin >> a;
	int n = a.length(), i, c = 0, x=0;
	vector<int> no;
	for (i = 0; i < n; i++)
	{
		if (a[i] == 'a')	x++;
		if (a[i] != a[c])
		{
			no.push_back(i - c);
			c = i;
		}
	}
	no.push_back(i - c);
	if (no.size() < 3 || x == 0 || x == n)
		cout << "No" << endl;
	else if (no.size() == 3)
	{
		if (no[1] > 1)	cout << "Yes" << endl;
		else	cout << "No" << endl;
	}
	else
	{
		sort(no.begin(), no.end());
		if (no[0] > 1 && no[1] > 1)	cout << "Yes" << endl;
		else	cout << "No" << endl;
	}
	return 0;
}
