#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<int> b(n);
    int min_b = n + 1;
    for (int &x : b) { cin >> x; min_b = min(min_b, x); }
    int cnt = 0;
    for (int x : b) { if (x == min_b) cnt++; }
    if (cnt > 1) { cout << -1 << endl; continue; }

    vector<bool> used(2 * n + 1, false);
    vector<int> a(2 * n, 0);
    for (int i = 0; i < n; i++) {
      int val = (b[i] != min_b) ? b[i] : 2 * n;
      if (used[val]) { cout << -1 << endl; return 0; }
      used[val] = true;
      a[2 * i] = val;
      a[2 * i + 1] = 1;
    }
    for (int i = 1; i <= 2 * n; i++) { if (!used[i]) { a[2 * n - 1] = i; break; } }
    for (int x : a) cout << x << ' ';
    cout << endl;
  }
  return 0;
}
