#include <iostream>
#include <vector>
using namespace std;
int n, h, m, l[55], r[55], x[55], dp[55][55];
int get(int, int);
int main() {
  cin >> n >> h >> m;
  for (int i = 1; i <= m; ++i) cin >> l[i] >> r[i] >> x[i];
  cout << get(1, 0);
  return 0;
}
int get(int u, int mx) {
  if (u == n + 1) return 0;
  int &ret = dp[u][mx];
  if (ret != -1) return ret;
  ret = get(u + 1, mx);
  for (int i = 0; i <= h && i > mx; ++i)
    ret = max(ret, get(u + 1, i) + (min(i, x[l[1]]) == mx ? 0 : i * i));
  return ret;
}
