import java.util.*;

public class Main {
    static int n, m;
    static boolean[] vis;
    static ArrayList<Integer>[] adj;
    static int[] ans;
    static int[] len;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        adj = new ArrayList[n + 1];
        vis = new boolean[n + 1];
        ans = new int[n + 1];
        len = new int[n + 1];
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        len[1] = 0;
        vis[1] = true;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int next : adj[cur]) {
                if (!vis[next]) {
                    q.add(next);
                    len[next] = len[cur] + 1;
                    ans[next] = cur;
                    vis[next] = true;
                }
            }
        }
        if (!vis[n]) {
            System.out.println(-1);
        } else {
            int cnt = 0;
            int cur = n;
            while (cur != 1) {
                cnt++;
                cur = ans[cur];
            }
            System.out.println(cnt);
            cur = n;
            while (cur != 1) {
                System.out.print(cur + " ");
                cur = ans[cur];
            }
            System.out.println(1);
        }
    }
}
