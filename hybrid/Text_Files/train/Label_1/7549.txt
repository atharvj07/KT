#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
#define pb push_back
const ll mod=1e9+7;
const char nl = '\n';
#define all(x) x.begin(), x.end()

int main()
{
  ll n;cin>>n;
  vi v(n);
  for(ll i=0;i<n;i++) cin>>v[i];
  sort(all(v));
  cout<<(v[0]<v.back()?"YES":"NO")<<nl;
  if(v[0]<v.back())
  {
    cout<<v.back()<<' ';
    for(ll i=0;i<n;i++)
    {
      if(v[i]==v.back()) continue;
      cout<<v[i]<<' ';
    }
  }
  return 0;
}
