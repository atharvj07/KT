#include <bits/stdc++.h>
using namespace std;
string a,b;
int ans;
int main() {
    cin>>a>>b;
    int l1=a.size();
    int l2=b.size();
    int cnta1=0,cnta2=0,cntb1=0,cntb2=0;
    for(int i=0;i<l1;i++){
        if(a[i]=='0'){
            cnta1++;
        }
        else{
            cnta2++;
        }
    }
    for(int i=0;i<l2;i++){
        if(b[i]=='0'){
            cntb1++;
        }
        else{
            cntb2++;
        }
    }
    if((l1!=l2) || (cnta1!=cntb1) || (cnta2!=cntb2)) ans=0;
    else{
        ans=1;
        for(int i=0;i<l1;i++){
            if(a[i]!=b[i]){
                if(a[0]==b[i]){
                    a.erase(a.begin());
                }
                else{
                    a.push_back(b[i]);
                }
            }
        }
        if(a==b){
            ans=1;
        }
        else{
            ans=0;
        }
    }
    if(ans==0){
        cout<<"NO"<<endl;
    }
    else{
        cout<<"YES"<<endl;
    }
    return 0;
}
