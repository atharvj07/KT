import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        long sum = 0;
        for (int i = 0; i < n; i++) {
            if ((i + 1) % 2 == a[i] % 2) {
                sum += a[i];
            }
        }
        System.out.println(sum);
    }
}
