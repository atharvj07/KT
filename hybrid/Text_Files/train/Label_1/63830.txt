n, m = map(int, input().split())
a = sorted([int(x.split()[1]) for x in [input() for _ in range(n)]], reverse=True)
b = sorted([int(input()) for _ in range(m)], reverse=True)
i, j, ans = 0, 0, 0
while i < n and j < m and b[j] >= a[i]: j += 1
while j < m: ans += b[j] - a[i]; j += 1; i += 1
print(ans)
