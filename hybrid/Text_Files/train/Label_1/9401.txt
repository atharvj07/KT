#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll gcd(ll a, ll b) {return !b ? a : gcd(b, a % b);}
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll p, a, b, c;
        cin >> p >> a >> b >> c;
        ll g = gcd(a, b);
        g = gcd(g, c);
        cout << (g - p % g + g) % g << '\n';
    }
}
