#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
char str[maxn];
int pos[30], pre[30];
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int n;
  cin >> n;
  for (int i = 0; i < 26; i++) pos[i] = pre[i] = -1;
  for (int i = 1; i <= n; i++) {
    cin >> str[i];
    for (int j = 0; j < 26; j++) pre[j] = max(pre[j], pos[j]);
    pos[str[i] - 'a'] = i;
  }
  int ans = 0;
  for (int i = 0; i < 26; i++)
    if (pos[i] != -1) ans = max(ans, pre[i] + 1);
  cout << (ans == n ? 0 : ans) << endl;
  return 0;
}
