#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;
typedef tuple<int, int, int> tp;

const ll mod = 1e9 + 7;
const int MX = 2e5 + 5;

ll binpow(ll b, ll p) {
    ll ret = 1;
    while (p) {
        if (p & 1) ret = (ret * b) % mod;
        b = (b * b) % mod;
        p >>= 1;
    }
    return ret;
}

ll get_ans(vector<int>& a, int n, int m, int k) {
    ll ans = binpow(n, m - 1);
    map<tp, int> freq;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (abs(a[i] - a[j]) <= k) {
                int mi = min(a[i], a[j]), mx = max(a[i], a[j]);
                tp t = make_tuple(mi, mx, mx - mi);
                ++freq[t];
            }
        }
    }
    for (auto& [t, f] : freq) {
        int mi, mx, dif;
        tie(mi, mx, dif) = t;
        if (dif <= k) {
            ll cnt1 = f, cnt2 = 0;
            if (mi == mx) cnt2 = f * (f - 1) / 2;
            else cnt2 = f * (f - 1);
            cnt2 %= mod;
            ans = (ans + (cnt1 - cnt2 + mod) % mod * binpow(n, m - 2)) % mod;
        }
    }
    return ans;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<int> a(n);
        for (auto& x : a) cin >> x;
        cout << get_ans(a, n, m, k) << '\n';
    }
    return 0;
}
