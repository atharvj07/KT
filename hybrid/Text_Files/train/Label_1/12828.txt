#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int k = 0;
  vector<int> ans;
  for (int i = 1; i <= 1000; i++) {
    if (n >= i) {
      ans.push_back(i);
      n -= i;
      k++;
    } else
      break;
  }
  cout << k << endl;
  for (int i = 0; i < k; i++) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
