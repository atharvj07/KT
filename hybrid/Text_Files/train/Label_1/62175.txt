import sys

def main():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n, m = map(int, sys.stdin.readline().split())
        p = list(map(int, sys.stdin.readline().split()))
        k0 = [0] * n
        for i in range(n):
            k0[p[i] - 1] = i
        k = set()
        for i in range(n):
            if k0[i] >= m:
                for j in range(m):
                    k.add((k0[i] - j) % n)
        print(len(k), *sorted(k))

main()
