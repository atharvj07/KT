#include <bits/stdc++.h>
using namespace std;
int mul(const vector<int>&v){
    int ans=1;
    for(int x:v)ans*=x;
    return ans;
}
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        vector<int>v(n);
        for(int i=0;i<n;++i){
            cin>>v[i];
        }
        sort(v.begin(),v.end());
        int g=v[0];
        for(int i=1;i<n;++i){
            g=__gcd(g,v[i]);
        }
        if(g==1)cout<<-1<<endl;
        else{
            int ans=1;
            for(int i=0;i<n;++i){
                ans=ans*v[i]/__gcd(ans,v[i]);
            }
            cout<<ans<<endl;
        }
    }
}
