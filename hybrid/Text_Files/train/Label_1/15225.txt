#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, a[210000];
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a, a + n);
  long long pre[2];
  for (int i = 0; i <= 1; ++i) {
    pre[i] = 0;
    for (int j = i; j < n; j += 2) pre[i] += a[j];
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (pre[0] - a[i] == pre[1]) ans++;
  }
  cout << ans << endl;
  return 0;
}
