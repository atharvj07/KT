#include <bits/stdc++.h>
#define ll long long
using namespace std;

pair<ll, ll> solve(vector<ll> a, ll n) {
  sort(a.begin(), a.end());
  ll y = n - 1, z;
  for (ll i = 1; i < n; i++) {
    if (a[i] == a[i - 1]) {
      y--;
    } else {
      z = a[i] - a[0];
      break;
    }
  }
  return {y, z};
}

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  pair<ll, ll> ans = solve(a, n);
  cout << ans.first << " " << ans.second << endl;
  return 0;
}
