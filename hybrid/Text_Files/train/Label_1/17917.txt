#include <bits/stdc++.h>
using namespace std;
vector<char> v[3];
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    v[s.size() - 1].push_back(s[0]);
  }
  for (int i = 0; i < m; i++) {
    string s;
    cin >> s;
    bool ok = 0;
    int ch = s[0];
    for (int j = 0; j < v[s.size() - 1].size(); j++) {
      int cnt = 0;
      for (int k = 0; k < s.size(); k++) {
        if (s[k] != v[s.size() - 1][j] && ch == v[s.size() - 1][j])
          cnt++;
      }
      if (cnt == 1) { ok = 1; }
    }
    if (ok)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
