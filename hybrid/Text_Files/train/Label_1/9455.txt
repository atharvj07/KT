#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> s(n), c(n);
  for (int i = 0; i < n; i++) cin >> s[i];
  for (int i = 0; i < n; i++) cin >> c[i];
  vector<long long> ss(n, 1e9), cc(n, 1e9);
  for (int i = n - 2; i >= 0; i--) {
    if (s[i] < s[i + 1]) {
      ss[i] = s[i + 1];
      cc[i] = c[i + 1];
    }
  }
  ss[n - 1] = c[n - 1];
  long long ans = 1e18;
  for (int i = 0; i < n; i++) {
    int idx1 = -1, idx2 = -1;
    for (int j = 0; j < i; j++) {
      if (s[j] < s[i]) { idx1 = j; }
    }
    for (int j = i + 1; j < n; j++) {
      if (ss[j] > s[i]) { idx2 = j; }
    }
    if (idx1 != -1 && idx2 != -1) { ans = min(ans, c[i] + ss[idx2] + cc[idx1]); }
  }
  if (ans == 1e18)
    ans = -1;
  cout << ans << endl;
  return 0;
}
