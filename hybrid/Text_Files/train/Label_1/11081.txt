#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> flowers(n);
    for (int i = 0; i < n; ++i) cin >> flowers[i];
    sort(flowers.begin(), flowers.end());
    cout << flowers[n - 1] - flowers[0] << " "
         << (flowers[0] == flowers[n - 1] ? 1 : n * (n - 1)) << endl;
    return 0;
}
