import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[][] want = new int[n][3];
        int[][] have = new int[n][3];
        for (int i = 0; i < n; i++) {
            String[] line = in.readLine().split(" ");
            for (int j = 0; j < 3; j++) {
                want[i][j] = Integer.parseInt(line[j]);
            }
        }
        for (int i = 0; i < n; i++) {
            String[] line = in.readLine().split(" ");
            for (int j = 0; j < 3; j++) {
                have[i][j] = Integer.parseInt(line[j]);
            }
        }
        if (!checkPossible(n, want, have)) {
            System.out.println("Impossible");
            return;
        }
        List<Integer> ct = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            ct.add(0);
        }
        List<Edge> cd = new ArrayList<>();
        dfs(1, ct, cd, want, have);
        System.out.println("Possible");
        System.out.println(ct);
        System.out.println(cd.size());
        for (Edge edge : cd) {
            System.out.println(edge.from + " " + edge.to);
        }
    }

    static boolean checkPossible(int n, int[][] want, int[][] have) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                if (want[i][j] == 1 && have[i][j] == 0) {
                    return false;
                }
            }
        }
        return true;
    }

    static void dfs(int node, List<Integer> ct, List<Edge> cd, int[][] want, int[][] have) {
        if (node == 1) {
            ct.set(node - 1, 1);
        }
        for (int i = 1; i < want.length; i++) {
            if (i == node) {
                continue;
            }
            boolean canReach = false;
            for (Edge edge : cd) {
                if (edge.to == node) {
                    canReach = true;
                    break;
                }
            }
            if (!canReach) {
                continue;
            }
            boolean needCt = false;
            for (int j = 0; j < 3; j++) {
                if (want[i][j] == 1 && have[i][j] == 0) {
                    needCt = true;
                    break;
                }
            }
            if (needCt) {
                if (ct.get(i - 1) == 0) {
                    ct.set(i - 1, 1);
                    dfs(i, ct, cd, want, have);
                }
            } else {
                cd.add(new Edge(node, i));
                dfs(i, ct, cd, want, have);
            }
        }
    }

    static class Edge {
        int from;
        int to;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }
}
