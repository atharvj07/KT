import java.util.*;

public class Main {
    private static void solve() {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = in.nextInt();
        }
        int[] pos = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            pos[p[i]] = i;
        }
        int[] ans = new int[5 * n + 1];
        int cnt = 0;
        for (int i = 1; i <= n / 2; i++) {
            if (pos[i] > n / 2) {
                int j = pos[i];
                ans[++cnt] = i;
                ans[++cnt] = j;
                swap(p, i, j);
                pos[p[i]] = i;
                pos[p[j]] = j;
            }
        }
        for (int i = n / 2 + 1; i <= n; i++) {
            if (pos[i] <= n / 2) {
                int j = pos[i];
                ans[++cnt] = i;
                ans[++cnt] = j;
                swap(p, i, j);
                pos[p[i]] = i;
                pos[p[j]] = j;
            }
        }
        System.out.println(cnt);
        for (int i = 1; i <= cnt; i++) {
            System.out.println(ans[i] + " " + ans[i + 1]);
            i++;
        }
    }

    private static void swap(int[] p, int i, int j) {
        int t = p[i];
        p[i] = p[j];
        p[j] = t;
    }

    public static void main(String[] args) {
        solve();
    }
}
