#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while (t--)
    {
        int n,m;
        cin>>n>>m;
        string s;
        cin>>s;
        int p[m];
        for(int i=0;i<m;i++)
        {
            cin>>p[i];
        }
        long long count[26]={0};
        long long sum=n+1;
        for(int i=0;i<m;i++)
        {
            count[s[p[i]-1]-'a']++;
            sum-=(p[i]);
        }
        for(int i=0;i<n;i++)
        {
            count[s[i]-'a']+=sum;
        }
        for(int i=0;i<26;i++)
        {
            cout<<count[i]<<" ";
        }
        cout<<endl;
    }
    return 0;
}
