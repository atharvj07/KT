#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;

    int a = 0, b = 0;

    for (int i = 0; i < n; i++) {
        int t, x, y;
        cin >> t >> x >> y;

        if (t == 1)
            a += x;
        else
            b += x;
    }

    if (a > n / 2)
        cout << "LIVE" << endl;
    else
        cout << "DEAD" << endl;

    if (b > n / 2)
        cout << "LIVE" << endl;
    else
        cout << "DEAD" << endl;

    return 0;
}
