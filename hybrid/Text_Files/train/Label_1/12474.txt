#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int N = 110;

int a[N];

int f(int x) {
  if (x < 8) return x;
  return x % 8 + (x / 8) * 10;
}

int main() {
  string s;
  cin >> s;
  int n = s.size();
  for (int i = 0; i < n; i++) a[i] = s[i] - '0';
  vector<int> ans;
  int carry = 0;
  carry = f(a[n - 1]);
  for (int i = n - 2; i >= 0; i--) {
    carry = f(carry * 10 + a[i]);
    if (carry % 8 == 0 && ans.empty() && a[i] == 0) continue;
    ans.push_back(carry % 8);
    carry /= 8;
  }
  if (ans.empty()) cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    reverse(ans.begin(), ans.end());
    for (int x : ans) cout << x;
    cout << endl;
  }
  return 0;
}
