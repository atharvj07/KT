import sys

n, k, m = list(map(int, sys.stdin.readline().split()))
a = list(map(int, sys.stdin.readline().split()))
d = {}
for e in a:
    d[e % m] = d.get(e % m, 0) + 1
if d.get(0, 0) >= k:
    print('Yes')
    for i in range(k):
        print(0, end=' ')
else:
    ans = []
    for i in range(1, m):
        if i in d and d[i] + d.get(0, 0) >= k:
            for j in range(d[i]):
                ans.append(i)
            for j in range(k - d[i]):
                ans.append(0)
            print('Yes')
            print(*ans)
            sys.exit()
print('No')
