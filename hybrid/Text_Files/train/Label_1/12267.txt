#include <bits/stdc++.h>
using namespace std;

const int INF = 2e9;
struct edge {
    int to, w;
    edge(int to, int w) : to(to), w(w) {}
};
vector<vector<edge>> g;
vector<int> dist, par;

void dijkstra(int s) {
    dist[s] = 0;
    priority_queue<pair<int, int>> q;
    q.push({0, s});
    while (!q.empty()) {
        int v = q.top().second;
        int d = -q.top().first;
        q.pop();
        if (d > dist[v]) continue;
        for (auto& e : g[v]) {
            if (dist[e.to] > dist[v] + e.w) {
                dist[e.to] = dist[v] + e.w;
                q.push({-dist[e.to], e.to});
                par[e.to] = v;
            }
        }
    }
}

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    g.resize(n + 1);
    dist.assign(n + 1, INF);
    par.assign(n + 1, -1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }
    vector<int> storages;
    if (k > 0) {
        for (int i = 0; i < k; i++) {
            int a;
            cin >> a;
            storages.push_back(a);
        }
    }
    int ans = INF;
    for (int i = 1; i <= n; i++) {
        bool ok = true;
        for (int s : storages) {
            if (s == i) {
                ok = false;
                break;
            }
        }
        if (!ok) continue;
        dijkstra(i);
        int mn = INF;
        for (int s : storages) {
            if (dist[s] < mn) mn = dist[s];
        }
        if (mn < INF) ans = min(ans, mn);
    }
    if (ans == INF) ans = -1;
    cout << ans << '\n';
    return 0;
}
