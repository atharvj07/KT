import sys
from collections import defaultdict
from math import sqrt

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n, e = map(int, input().split())
    arr = list(map(int, input().split()))

    pf = [[] for _ in range(n)]
    for i in range(n):
        for j in range(2, int(sqrt(arr[i])) + 1):
            if arr[i] % j == 0:
                while arr[i] % j == 0:
                    arr[i] //= j
                pf[i].append(j)
        if arr[i] > 1:
            pf[i].append(arr[i])

    cnt = 0
    for i in range(n):
        d = defaultdict(int)
        for j in pf[i]:
            d[j] += 1
        for j in range(i + e, n, e):
            fl = True
            for k in d:
                if k in pf[j] and d[k] == pf[j].count(k):
                    continue
                else:
                    fl = False
                    break
            if fl:
                cnt += 1
    print(cnt)
