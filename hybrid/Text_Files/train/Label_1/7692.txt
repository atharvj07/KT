#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pll;
#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()
#define rep(i, a, b) for (ll i = a; i < b; i++)
#define rrep(i, a, b) for (ll i = a; i >= b; i--)
#define dbg(x) cerr << #x << " = " << x << endl

const ll INF = 1e18;
const ll MOD = 1e9 + 7;

ll n, m;
vi(pll) p;
ll sumx, sumd;

ll get(pair<ll, ll> x) {
    ll sum = 0;
    rep(i, 1, n + 1) {
        sum += x.first + x.second * (abs(i - (n / 2 + 1)));
    }
    return sum;
}

void solve() {
    cin >> n >> m;
    sumx = 0, sumd = 0;
    rep(i, 0, m) {
        ll x, d;
        cin >> x >> d;
        sumx += x, sumd += d;
        p.pb({x, d});
    }
    sort(all(p));
    ll ans = get({sumx, sumd});
    rrep(i, m - 1, 0) {
        ll x = p[i].first, d = p[i].second;
        sumx -= x, sumd -= d;
        ans = max(ans, get({sumx, sumd}));
    }
    cout << fixed << setprecision(9) << (double)ans / n << endl;
}
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    solve();
    return 0;
}
