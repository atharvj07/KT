#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, m, x, k, y;
int a[N], b[N], pos[N];
vector<int> inc[N];

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> m >> x >> k >> y;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    pos[a[i]] = i;
  }
  for (int i = 1; i <= m; i++) cin >> b[i], inc[b[i]].push_back(i);
  int ans = 0, l = 1;
  for (int i = 1; i <= n; i++) {
    if (inc[a[i]].empty())
      continue;
    int ch = inc[a[i]].back();
    inc[a[i]].pop_back();
    ans += min(1LL * (pos[a[i]] - l + 1) * x, 1LL * y);
    l = pos[a[i]] + k;
    if (ch == m)
      break;
  }
  cout << (l > n ? -1 : ans);
  return 0;
}
