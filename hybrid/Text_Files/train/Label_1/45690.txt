import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        sc.close();
        int max = Arrays.stream(a).max().getAsInt();
        for (int i = max + 1;; i++) {
            int sum = 0;
            for (int x : a) if (x < i) sum += i - x;
            if (sum > 0) {
                System.out.println(i);
                return;
            }
        }
    }
}
