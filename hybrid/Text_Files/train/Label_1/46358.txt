import java.util.*;

public class Main {
    public static int g[][];
    public static int d[][];
    public static int n, k, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        m = sc.nextInt();
        g = new int[n + 1][n + 1];
        d = new int[n + 1][1 << k];
        for (int i = 1; i <= m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int c = sc.nextInt();
            g[u][v] = g[v][u] = c;
        }
        for (int i = 0; i <= k; i++) {
            Arrays.fill(d[i], (int) 1e9);
        }
        d[1][1 << 0] = 0;
        for (int mask = 1; mask < (1 << k); mask++) {
            for (int i = 1; i <= n; i++) {
                if ((mask & (1 << (i - 1))) == 0) {
                    continue;
                }
                for (int j = 1; j <= n; j++) {
                    if (g[i][j] == 0) {
                        continue;
                    }
                    d[i][mask] = Math.min(d[i][mask], d[j][mask ^ (1 << (i - 1))] + g[i][j]);
                }
            }
        }
        int ans = (int) 1e9;
        for (int i = 1; i <= n; i++) {
            ans = Math.min(ans, d[i][(1 << k) - 1]);
        }
        if (ans == (int) 1e9) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
        sc.close();
    }
}
