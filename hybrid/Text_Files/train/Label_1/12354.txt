#include <iostream>
using namespace std;

int main() {
    int p, n;
    cin >> p >> n;
    int a[p];
    for (int i = 0; i < p; ++i) a[i] = -1;
    int ans = -1;
    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        int h = x % p;
        if (a[h] != -1) {
            ans = i + 1;
            break;
        }
        a[h] = x;
    }
    cout << ans << endl;
    return 0;
}
