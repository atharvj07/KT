import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int[] lcp = new int[n];
        for (int i = 1; i < n; i++) {
            int j = lcp[i - 1];
            while (j > 0 && s.charAt(i) != s.charAt(j)) j = lcp[j - 1];
            lcp[i] = j + 1;
        }
        int[] c = new int[n];
        int ans = 0;
        int k = 0;
        c[n - 1] = k++;
        for (int i = n - 2; i >= 0; i--) {
            if (lcp[i] == 0 || s.charAt(i) != s.charAt(lcp[i] - 1)) {
                c[i] = k++;
            } else {
                c[i] = c[lcp[i] - 1];
            }
            ans = Math.max(ans, c[i]);
        }
        System.out.println(ans + 1);
        for (int i = 0; i < n; i++) {
            System.out.print(c[i] + 1 + " ");
        }
    }
}
