#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 1, cnt = 0;
    for (int i = 0; i < n; i++) {
        if (i > 0 && a[i] > a[i - 1]) {
            cnt++;
            ans = max(ans, cnt);
        } else
            cnt = 1;
    }
    cout << ans << endl;
}
