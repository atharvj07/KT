import java.util.Scanner;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] piles = new int[n];
        for (int i = 0; i < n; i++) {
            piles[i] = scanner.nextInt();
        }

        boolean kevinWins = true;
        for (int pile : piles) {
            if (pile % 2 == 0) {
                kevinWins = false;
                break;
            }
        }

        System.out.println(kevinWins ? "Kevin" : "Nicky");
    }
}
