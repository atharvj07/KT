#include <bits/stdc++.h>

using namespace std;
const int N = 150'000;

int64_t a[N + 2], b[N + 2], fr[N + 2];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int cnt = n;
  for (int i = 0; i < n; i++) {
    if (!fr[a[i]]) { b[cnt++] = a[i]; }
    fr[a[i]]++;
  }
  sort(b, b + cnt);
  int ans = n;
  for (int i = 0; i < cnt; i++) {
    if (fr[b[i]] < 2) continue;
    ans -= fr[b[i]] - 1;
    int tmp = 0;
    for (int j = 0; j < n; j++) {
      if (a[j] == b[i]) {
        if (tmp == 0) {
          a[j] = 2 * b[i];
          tmp++;
        } else
          a[j] = 0;
        tmp++;
        if (tmp == fr[b[i]]) break;
      }
    }
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    if (a[i]) cout << a[i] << " ";
  }
}
