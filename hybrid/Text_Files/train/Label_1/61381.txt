n, m = map(int, input().split())
matrix = [list(map(int, input().split())) for _ in range(n)]

def invertRow(row):
    return [1 - val for val in row]

def invertColumn(matrix, col):
    for row in matrix:
        row[col] = 1 - row[col]

# Inverting rows if necessary to make first column all zeros if possible
row_inversion = ['0'] * n
for i in range(n):
    if matrix[i][0] == 1:
        matrix[i] = invertRow(matrix[i])
        row_inversion[i] = '1'

# Checking if it's possible to sort matrix by column operations
for j in range(1, m):
    col = [matrix[i][j] for i in range(n)]
    if col != sorted(col):
        invertColumn(matrix, j)
        if [matrix[i][j] for i in range(n)] != sorted(col): # After one inversion, it's still not sorted
            print("NO")
            break
else: # This else belongs to the for-loop and is executed if the loop completes normally (no break)
    print("YES")
    print(''.join(row_inversion))
    col_inversion = ['1' if matrix[0][j] == 1 else '0' for j in range(m)]
    print(''.join(col_inversion))
