import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        long a = Long.parseLong(line[1]);
        long b = Long.parseLong(line[2]);
        PriorityQueue<Long> hp = new PriorityQueue<>();
        for(int i = 0; i < n; i++) hp.add(Long.parseLong(br.readLine()));
        long cnt = 0;
        while(!hp.isEmpty()) {
            long monster = hp.poll();
            long explode = monster / a;
            cnt += explode;
            monster %= a;
            if(monster > 0) hp.add(monster - b);
        }
        System.out.println(cnt);
    }
}
