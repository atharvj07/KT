#include <bits/stdc++.h>
using namespace std;
const int M = 2e5 + 10;
vector<int> Solve(vector<int>b) {
    int n = (int)b.size() - 2, sum = 0;
    for (int i = 0; i < n + 1; i++)
        sum += b[i];
    int x = b[n + 1], y = b[n + 2];
    if (y < x)return vector<int>{-1};
    vector<int>a(n);
    a[n - 1] = y - x;
    b[n] -= a[n - 1];
    for (int i = n - 2; ~i; i--) {
        if (b[i] <= b[i + 1])return vector<int>{-1};
        a[i] = b[i + 1] - b[i];
        sum -= a[i];
    }
    if (sum == x)return a;
    return vector<int>{-1};
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int>b(n + 2);
        for (int i = 0; i < n + 2; i++)
            cin >> b[i];
        auto ans = Solve(b);
        if (ans.empty())cout << "-1\n";
        else {
            for (int i : ans)cout << i << " ";
            cout << "\n";
        }
    }
}
