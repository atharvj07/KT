#include<bits/stdc++.h>
using namespace std;

vector<int>adj[505];
int vis[505]={0};

bool dfs(int cur, int par, int flag)
{
    if(vis[cur]==2)return 0;
    vis[cur]=2;
    for(int it:adj[cur])
    {
        //back edge
        if(vis[it]==1)
        {
            //first back edge
            if(it==par)continue;
            else if(flag){return 1;}
            else{return dfs(it,cur,1);}
        }
        //forward edge
        else{
            if(flag){return dfs(it,cur,1);}
            if(dfs(it,cur,0)){return 1;}
        }
    }
    vis[cur]=1;
    return 0;
}

int main()
{
    int n,m;
    cin>>n>>m;

    for(int i=0;i<m;i++)
    {
        int u,v;
        cin>>u>>v;
        adj[u].push_back(v);
    }

    if(dfs(1,-1,0)){cout<<"YES"<<endl;}
    else{cout<<"NO"<<endl;}
    return 0;
}
