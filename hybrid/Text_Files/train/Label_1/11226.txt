#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2e5 + 10;
const double eps = 1e-6;
int type[N], a[N], b[N];
ll sum[N]; double ans[N];

int main() {
  int n; cin >> n;
  a[0] = b[0] = 0; ans[0] = sum[0] = 0.0;
  for (int i = 1, op; i <= n; ++i) {
    cin >> op;
    if (op == 1) {
      cin >> a[i] >> b[i];
      sum[i] = sum[i - 1] + 1ll * a[i] * b[i];
    } else if (op == 2) {
      cin >> b[i];
      sum[i] = sum[i - 1] + b[i];
    } else {
      sum[i] = sum[i - 1] - b[i - 1];
    }
    ans[i] = sum[i] / double(i);
    printf("%.6f\n", ans[i]);
  }
  return 0;
}
