#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  ll m, n;
  cin >> m >> n;
  ll am[m], bm[n];
  for (ll i = 0; i < m; i++) { cin >> am[i]; }
  for (ll i = 0; i < n; i++) { cin >> bm[i]; }
  ll ans = 0;
  sort(am, am + m);
  sort(bm, bm + n);
  ll l = 0, r = n - 1;
  for (ll i = m - 1; i >= 0; i--) {
    while (l <= r && bm[r] + am[i] <= bm[r]) { r--; }
    if (l <= r) { ans += r; }
    ans += am[i];
  }
  cout << ans << endl;
  return 0;
}
