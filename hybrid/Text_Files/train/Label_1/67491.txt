n, T = map(int, input().split())
t = list(map(int, input().split()))

max_bowls = 0
for start in range(n):
    time = 1
    bowls_eaten = 0
    for i in range(start, n):
        if time <= T:
            if time >= t[i]:
                bowls_eaten += 1
            time += 1
        else:
            break
    max_bowls = max(max_bowls, bowls_eaten)

print(max_bowls)
