#include <bits/stdc++.h>
using namespace std;
int main() {
  string t;
  cin >> t;
  int n = t.size();
  for (int i = 0; i < n; i++) {
    if (t[i] != t[n - 1 - i]) {
      cout << ":(";
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    if (t[i] != 'a') {
      cout << t[i];
    }
  }
  return 0;
}
