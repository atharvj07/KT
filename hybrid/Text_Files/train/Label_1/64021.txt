mod = int(1e9)+7

n, k = list(map(int,input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
d = [[0]*11 for i in range(k+1)]
for i in range(1,10):
	for j in range(k+1):
		d[j][i-b[j]%i] += 1
ans = 1
for i in range(1,n+1,k):
	temp = 1
	for j in range(k):
		if a[i//k-1]%d[j][a[i//k-1]%10] == 0:
			temp = -1
			break
	ans = (ans*temp)%mod
print(ans)
