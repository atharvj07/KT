t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    zero_groups = one_groups = 0
    i = 0
    while i < n:
        j = i
        while j < n and s[j] == s[i]:
            j += 1
        if s[i] == '0':
            zero_groups += 1
        else:
            one_groups += 1
        i = j
    print(max(zero_groups, one_groups) - 1)
