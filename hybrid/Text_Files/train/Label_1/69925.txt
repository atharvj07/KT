n=input()
l=len(n)
a=[]
for i in n:
  a.append(int(i))
for i in range(l):
  for j in range(l):
    if i!=j and a[i]%2==0 and a[j]%2==0 and a[i]<a[j]:
      a[i],a[j]=a[j],a[i]
      b=[]
      for x in a:
        b.append(str(x))
      print("".join(b))
      exit()
for i in range(l):
  for j in range(l):
    if i!=j and a[i]%2==1 and a[j]%2==1 and a[i]<a[j]:
      a[i],a[j]=a[j],a[i]
      b=[]
      for x in a:
        b.append(str(x))
      print("".join(b))
      exit()
print(-1)
