import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [[0] * m for _ in range(n)]
for i in range(n):
    for j in range(m):
        if j == 0: b[i][j] = a[i][j]
        else: b[i][j] = a[i][j] + b[i][j - 1]
for i in range(m):
    ma = -1
    ans = 0
    for j in range(n):
        if j > 0: cur = b[j][i] - b[j - 1][i]
        else: cur = b[j][i]
        if cur > ma:
            ma = cur
            ans = j
    print(ma, end = " ")
