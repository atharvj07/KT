import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = input.nextInt();
        }

        for (int i = 0; i < n; i++) {
            if (m[i] > k) {
                System.out.println(-1);
                continue;
            }

            int x = (k + 1) / 2;
            int y = (k + 1 - m[i]) / 2;

            System.out.println(x + " " + (y + 1) + " " + (y + m[i]));
        }
    }
}
