#include<bits/stdc++.h>
using namespace std;
int main() {
    int t; cin >> t;
    while(t--) {
        int n; cin >> n;
        int arr[n + 1];
        for(int i = 1; i <= n; i++) { arr[i] = i; }
        int p = 1, q = n;
        for(int i = 1; i <= n / 2; i++) {
            swap(arr[p], arr[q]);
            p++, q--;
        }
        for(int i = 1; i <= n; i++) { cout << arr[i] << " "; }
        cout << endl;
    }
    return 0;
}
