import java.util.Scanner;

class Main {
    static int n, h, ans = 0;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        h = input.nextInt();
        dfs(1, 0);
        System.out.println(ans);
    }

    public static void dfs(int idx, int rem) {
        if (rem == h) {
            ans++;
            return;
        }
        if (idx > n) {
            return;
        }
        dfs(idx + 1, rem + 1);
        dfs(idx + 2, rem + 1);
    }
}
