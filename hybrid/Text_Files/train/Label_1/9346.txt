#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<pair<char, int>> runs;
  char now = s[0];
  int cnt = 1;
  for (int i = 1; i < n; ++i) {
    if (s[i] != now) {
      runs.push_back({now, cnt});
      cnt = 1, now = s[i];
    } else
      ++cnt;
  }
  runs.push_back({now, cnt});
  int ans = 0;
  for (auto i : runs) {
    if (i.second >= 4) continue;
    if (i.first == 'G' && i.second >= 2) {
      ans += (4 - i.second) / 2;
      continue;
    }
    int prv = 0, nxt = 0;
    if (i.first != 'A') nxt = i.second;
    if (i.first != 'Z' && i.second > 0) prv = i.second;
    ans += min(nxt, prv);
  }
  cout << ans << endl;
  return 0;
}
