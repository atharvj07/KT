import sys
from array import array
n,m=map(int,sys.stdin.readline().split())
a=[[int(c) for c in sys.stdin.readline().strip()]for _ in range (n)]
ans=0
for i in range(0,n-1):
    for j in range(0,m-1):
        tot=a[i][j]
        tot+=a[i+1][j]
        tot+=a[i][j+1]
        tot+=a[i+1][j+1]
        if tot%2==0:ans+=1
print(ans)
