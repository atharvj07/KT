import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    Max = [0] * n
    Min = [0] * n
    Max[n - 1] = a[n - 1]
    Min[n - 1] = a[n - 1]
    for i in range(n - 2, -1, -1):
        Max[i] = max(a[i], Max[i + 1])
        Min[i] = min(a[i], Min[i + 1])
    ans = 0
    for i in range(n - 1):
        ans = max(ans, Max[i] * Min[i + 1])
    print(ans)
