import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class Jigsaw {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = reader.readLine().split(" ");
        int N = Integer.parseInt(firstLine[0]);
        int H = Integer.parseInt(firstLine[1]);

        List<Piece> pieces = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            String[] line = reader.readLine().split(" ");
            int A = Integer.parseInt(line[0]);
            int B = Integer.parseInt(line[1]);
            int C = Integer.parseInt(line[2]);
            int D = Integer.parseInt(line[3]);
            pieces.add(new Piece(i, A, B, C, D));
        }

        Collections.sort(pieces, Comparator.comparingInt(a -> a.height));

        int[] bottom = new int[H + 1];
        for (Piece piece : pieces) {
            if (piece.height + piece.a + piece.b > H || piece.c > piece.a || piece.d > piece.b) {
                System.out.println("NO");
                return;
            }
            bottom[piece.height] = Math.max(bottom[piece.height], piece.a + piece.b);
            for (int i = piece.height - piece.a; i < piece.height; i++) {
                bottom[i] = Math.max(bottom[i], piece.a);
            }
            for (int i = piece.height + 1; i <= piece.height + piece.b; i++) {
                bottom[i] = Math.max(bottom[i], piece.b);
            }
        }

        for (int i = 0; i < H; i++) {
            if (bottom[i] > bottom[i + 1]) {
                System.out.println("NO");
                return;
            }
        }

        System.out.println("YES");
    }

    private static class Piece {
        int id;
        int height;
        int a;
        int b;
        int c;
        int d;

        public Piece(int id, int a, int b, int c, int d) {
            this.id = id;
            this.height = a + b;
            this.a = a;
            this.b = b;
            this.c = c;
            this.d = d;
        }
    }
}
