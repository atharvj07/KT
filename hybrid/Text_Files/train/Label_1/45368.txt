import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		int[][] a = new int[n][];
		for (int i = 0; i < n; i++) {
			a[i] = new int[m];
			String s = sc.next();
			for (int j = 0; j < m; j++) {
				a[i][j] = s.charAt(j) - '0';
			}
		}
		int ans = -1;
		for (int j = 0; j < m; j++) {
			int cnt = 0;
			for (int i = 0; i < n; i++) {
				cnt += a[i][j];
			}
			if (cnt == n) {
				ans = 0;
			} else if (cnt > n / 2) {
				ans = Math.min(ans == -1 ? Integer.MAX_VALUE : ans, n - cnt);
			} else {
				ans = Math.min(ans == -1 ? Integer.MAX_VALUE : ans, cnt);
			}
		}
		System.out.println(ans);
	}
}
