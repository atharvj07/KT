#include <iostream>
#include <vector>

int main() {
  std::vector<int> coins(5);
  for (int &coin : coins) std::cin >> coin;
  int sum = coins[0] + coins[1] + coins[2] + coins[3] + coins[4];
  if (sum % 2 == 1) {
    std::cout << -1 << std::endl;
  } else {
    std::cout << sum / 2 << std::endl;
  }
  return 0;
}
