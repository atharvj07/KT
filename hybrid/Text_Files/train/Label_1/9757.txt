#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define all(v) (v).begin(), (v).end()
#define rall(v) (v).begin(), (v).end()
#define Unique(x) sort(all(x)), x.erase(unique(all(x)), x.end())
using namespace std;
const int N = 5e3 + 5, M = 3e6 + 5, mod = 998244353;
typedef long long ll;
typedef pair<ll, ll> pi;

int n, m, t, l[M], r[M], a[M], b[M], c[N], in[N];
vector<int> g[N];

bool valid() {
	for (int i = 1; i <= m; i++) if (l[i] > r[i]) return 0;
	for (int i = 1; i <= n; i++) if (in[i] ^ g[i].size()) return 0;
	return 1;
}

int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1; i <= m; i++) {
		cin >> t >> l[i] >> r[i];
		if (t == 1) { cin >> a[i], g[r[i]].pb(i), in[l[i]]++; }
		else cin >> b[i];
	}
	for (int i = 1; i <= n; i++) {
		for (int j : g[i]) c[i] += a[j];
		c[i] += b[in[i]];
	}
	(valid()) ? cout << "YES\n" : cout << "NO", exit(0);
	for (int i = 1; i <= n; i++) cout << c[i] << ' ';
	return 0;
}
