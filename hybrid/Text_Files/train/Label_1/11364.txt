#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e6+3;

inline int add(int a,int b){a+=b;return a>=mod?a-mod:a;}

int main(){
	int n,c;
	cin>>n>>c;
	vector<int> dp(n+1);
	dp[1]=c;
	for(int i=2;i<=n;++i)
		for(int j=1;j<=c;++j)
			dp[i]=add(dp[i],dp[i-j]);
	cout<<dp[n]<<endl;
}
