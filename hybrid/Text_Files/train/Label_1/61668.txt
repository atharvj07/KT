n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]
ops = 0
for x1 in range(n):
    for y1 in range(m):
        for x2 in range(x1+1):
            for y2 in range(y1+1):
                summ = 0
                for xb in range(x1, x2+1):
                    for yb in range(y1, y2+1):
                        if grid[xb][yb] == 'W':
                            summ += 1
                summ -= summ*2
                if summ < 0: summ = 0
                ops += 1 + bool(summ)
print(ops)
