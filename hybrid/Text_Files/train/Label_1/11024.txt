#include<iostream>

using namespace std;

int gcd(int a, int b) {
  int t;
  while (b != 0) {
    t = a % b;
    a = b;
    b = t;
  }
  return a;
}

int f(int a, int b) {
  if (b == 0) return 0;
  return 1 + f(a, b - gcd(a, b));
}

int main() {
  int x, y;
  cin >> x >> y;
  cout << f(x, y) << endl;
}
