T = int(input().strip())
for _ in range(T):
    n = int(input().strip())
    monsters = [list(map(int, input().strip().split())) for _ in range(n)]
    total_bullets = 0
    min_extra_damage = float('inf')
    for i in range(n):
        next_i = (i + 1) % n
        damage_to_next = min(monsters[next_i][0], monsters[i][1])
        total_bullets += monsters[next_i][0] - damage_to_next
        monsters[next_i][0] -= damage_to_next
        min_extra_damage = min(min_extra_damage, damage_to_next)
    total_bullets += min_extra_damage
    print(total_bullets)
