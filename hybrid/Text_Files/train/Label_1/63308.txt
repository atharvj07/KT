board = [input() for i in range(8)]
weight = {'Q': 9, 'R': 5, 'B': 3, 'N': 3, 'P': 1, 'q': -9, 'r': -5, 'b': -3, 'n': -3, 'p': -1}
wb, bb = 0, 0
for row in board:
    for piece in row:
        if piece in weight:
            if piece.isupper(): wb += weight[piece]
            else: bb += weight[piece]
if wb == bb: print("Draw")
elif wb > bb: print("White")
else: print("Black")
