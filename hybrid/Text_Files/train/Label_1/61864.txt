n, m = map(int, input().split())
a = list(map(int, input().split()))
if 0 in a:
    print('YES')
elif sum(a) % m == 0:
    print('YES')
else:
    s = 0
    count = 0
    for i in range(n):
        s += a[i]
        s %= m
        if s == 0:
            print('YES')
            exit()
        if s in count:
            print('YES')
            exit()
        count.add(s)
    print('NO')
