import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt();
    int[] a = new int[n * 2];
    for (int i = 0; i < n * 2; i++) a[i] = cin.nextInt();
    int[] ans = new int[n * 2];
    Arrays.sort(a);
    int cnt1 = 0, cnt2 = 0;
    for (int i = 0; i < n * 2; i++) {
      if (a[i] % 10 == 0) {
        ans[i] = 1;
        cnt1++;
      } else {
        ans[i] = 2;
        cnt2++;
      }
    }
    System.out.printf("%d\n", Math.min(cnt1, cnt2));
    for (int i = 0; i < n * 2; i++) System.out.printf("%d ", ans[i]);
  }
}
