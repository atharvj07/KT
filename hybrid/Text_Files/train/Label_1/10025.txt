#include <iostream>
#include <algorithm>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    int m, n, r, len;
    cin >> m;
    string s;
    vector<bool> taken(m);
    unordered_map<string, vector<string>> words;
    for (int i = 0; i < m; i++) {
        cin >> s;
        for (int j = 0; j < s.size(); j++) s[j] = tolower(s[j]);
        sort(s.begin(), s.end());
        words[s].push_back(words[s].empty() ? s : "");
    }
    cin >> n;
    for (int i = 0; i < n; i++) {
        string a, b;
        cin >> a >> b;
        a = b + a;
        for (int j = 0; j < a.size(); j++) a[j] = tolower(a[j]);
        sort(a.begin(), a.end());
        words[a].push_back(words[a].empty() ? a : "");
    }
    for (auto &p: words) {
        sort(p.second.begin(), p.second.end());
        p.second.erase(unique(p.second.begin(), p.second.end()), p.second.end());
    }
    r = len = 1e9;
    for (int mask = 0; mask < (1 << m); mask++) {
        int rr = 0, ll = 0;
        vector<string> word(m);
        for (int j = 0; j < m; j++) {
            if ((mask >> j) & 1) {
                int id = m - 1;
                for (int i = 0; i < words[s].size(); i++) {
                    if (taken[id]) { id--; continue; }
                    if (words[s][i] == words[words[s].back()][words[s].size() - 1 - id]) {
                        word[j] = words[s][i];
                        taken[id] = true;
                        ll += words[s][i].size();
                        for (int k = 0; k < words[s][i].size(); k++) {
                            if (words[s][i][k] == 'r') rr++;
                        }
                        id--;
                        break;
                    }
                }
            } else {
                int id = 0;
                for (int i = 0; i < words[s].size(); i++) {
                    if (taken[id]) { id++; continue; }
                    if (words[s][i] == words[words[s].back()][id]) {
                        word[j] = words[s][i];
                        taken[id] = true;
                        ll += words[s][i].size();
                        for (int k = 0; k < words[s][i].size(); k++) {
                            if (words[s][i][k] == 'r') rr++;
                        }
                        id++;
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < m; i++) taken[i] = false;
        if (rr < r || (rr == r && ll < len)) {
            r = rr;
            len = ll;
        }
    }
    cout << r << ' ' << len << endl;
    return 0;
}
