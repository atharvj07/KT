#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = 0, l = 0, r = 0;
  while (r < n) {
    if (a[r] - a[l] >= m) {
      if (r - l + 1 >= k) {
        ans = max(ans, r - l - (k - 1));
      }
      l++;
    } else {
      r++;
    }
  }
  cout << n - ans << endl;
}
