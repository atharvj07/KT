#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    p[i] = i + 1;
  }

  if (n % 2 == 0) {
    for (int i = 0; i < n / 2; i++) {
      swap(p[i], p[n - i - 1]);
    }
  } else {
    for (int i = 0; i < n / 2; i++) {
      swap(p[i], p[n - i - 2]);
      i++;
      swap(p[i], p[n - i - 1]);
    }
  }

  for (int i = 0; i < n; i++) {
    cout << p[i] << " ";
  }

  return 0;
}
