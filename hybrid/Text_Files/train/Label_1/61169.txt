import sys

def check(arr, s):
    return ((s == sum(arr)) or (len(arr) == 1))

isNegative = lambda x: x < 0

def solve(n, q, a, s):
    ans = []
    for i in range(q):
        if isNegative(s[i] - sum(a)):
            ans.append("No")
        elif check(a, s[i]):
            ans.append("Yes")
        else:
            a.sort()
            r = []; l = []; m = (max(a) + min(a)) // 2
            for x in a:
                if x <= m: l.append(x)
                else: r.append(x)
            if check(l, s[i]):
                ans.append("Yes")
            elif check(r, s[i]):
                ans.append("Yes")
            else:
                ans.append("No")
    return ans

t = int(input())
for tc in range(t):
    n, q = map(int, input().split())
    a = list(map(int, input().split()))
    s = list(map(int, [input() for i in range(q)]))
    print(*solve(n, q, a, s), sep = "\n")
