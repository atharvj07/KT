#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
int main() {
  int n, k, ans = 0;
  cin >> n >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++)
    cin >> a[i];
  long long pre[n + 2], suf[n + 2];
  pre[0] = suf[n + 1] = 1;
  for (int i = 0; i < n; i++)
    pre[i + 1] = pre[i] * a[i];
  for (int i = n - 1; i >= 0; i--)
    suf[i] = suf[i + 1] * a[i];
  for (int i = 0; i < n; i++)
    if (pre[i] % k == 0)
      ans += int((suf[i + 1] % k == 0));
  cout << ans << endl;
  return 0;
}
