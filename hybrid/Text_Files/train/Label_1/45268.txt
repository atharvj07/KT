import java.io.*;
import java.util.*;

public class Main {
    static long mod = (long) 1e9 + 7;
    static long[] a;
    static long[] sa;
    static List<Integer>[] g;

    static void dfs1(int v, int p) {
        a[v] += sa[v - 1];
        a[v] %= mod;
        for (int u : g[v]) {
            if (u == p) {
                continue;
            }
            dfs1(u, v);
        }
    }

    static void dfs2(int v, int p) {
        for (int u : g[v]) {
            if (u == p) {
                continue;
            }
            sa[u - 1] = (sa[v - 1] + a[v] - a[u]) % mod;
            dfs2(u, v);
        }
    }

    public static void main(String[] args) throws IOException {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] p = new int[n - 1];
        for (int i = 1; i < n; i++) {
            p[i - 1] = input.nextInt();
        }
        int q = input.nextInt();
        a = new long[n + 1];
        sa = new long[n + 1];
        g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            g[i + 1].add(p[i - 1]);
            g[p[i - 1]].add(i + 1);
        }
        for (int i = 0; i < q; i++) {
            int type = input.nextInt();
            if (type == 1) {
                int v = input.nextInt();
                int x = input.nextInt();
                int k = input.nextInt();
                a[v] += x;
                for (int u : g[v]) {
                    sa[u - 1] += x - k;
                    sa[u - 1] %= mod;
                }
            } else {
                int v = input.nextInt();
                dfs1(v, 0);
                dfs2(v, 0);
                System.out.println(a[v] % mod);
            }
        }
    }
}
