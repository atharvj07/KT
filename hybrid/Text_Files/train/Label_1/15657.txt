#include <bits/stdc++.h>

using namespace std;

using i64 = long long;
const i64 INF = 1e18;

i64 solve() {
  i64 n, m, k;
  cin >> n >> m >> k;
  vector<i64> a(n);
  multiset<i64> cur_sum, s;
  for (i64& ai : a) {
    cin >> ai;
    cur_sum.insert(ai);
    s.insert(ai);
  }
  i64 ans = 0, sum = 0;
  vector<i64> pos;
  while (k--) {
    auto it = cur_sum.end();
    --it;
    sum += *it;
    pos.push_back(s.count(*it));
    s.erase(s.find(*it));
    cur_sum.erase(it);
    for (i64 i = 0; i < m - 1; i++) {
      it = cur_sum.end();
      --it;
      sum += *it;
      cur_sum.erase(it);
    }
  }
  ans = sum;
  sort(pos.begin(), pos.end());
  for (i64 i = 0; i + 1 < pos.size(); i++) { pos[i] += pos[i + 1]; }
  pos.pop_back();
  cout << ans << endl;
  for (auto p : pos) { cout << p << " "; }
  cout << endl;
  return 0;
}

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
  return solve();
}
