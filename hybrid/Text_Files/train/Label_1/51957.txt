import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        while (q-- > 0) {
            int n = in.nextInt();
            String s = in.next();
            String t = in.next();
            if (s.equals(t)) {
                System.out.println("YES");
                continue;
            }
            boolean flag = true;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) != t.charAt(i)) {
                    if (flag) {
                        flag = false;
                    } else {
                        System.out.println("NO");
                        break;
                    }
                }
            }
            if (flag) {
                System.out.println("YES");
            }
        }
    }
}
