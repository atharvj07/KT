#include <iostream>
#include <vector>
#include <numeric>

const long long mod = 1e9 + 7;
long long qkpow(long long base, long long pow) {
  if (pow == 0) {
    return 1;
  } else if (pow == 1) {
    return base;
  } else if (pow % 2 == 0) {
    long long tmp = qkpow(base, pow / 2);
    return (tmp * tmp) % mod;
  } else {
    long long tmp = qkpow(base, pow - 1);
    return (base * tmp) % mod;
  }
}

int main() {
  long long n;
  std::cin >> n;
  std::vector<long long> nums(n);
  for (long long i = 0; i < n; ++i) std::cin >> nums[i];

  auto sum = std::accumulate(nums.begin(), nums.end(), 0LL,
                            [](long long a, long long b) -> long long {
                              return (a + b) % mod;
                            });
  auto sub = std::accumulate(nums.begin(), nums.end(), 0LL,
                            [](long long a, long long b) -> long long {
                              return (a - b + mod) % mod;
                            });
  long long ans = qkpow(sub, n - 1) * sum % mod;
  std::cout << ans << std::endl;

  return 0;
}
