import java.util.*;

public class Main {
  static Set<Integer> del;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt();
    int[] l = new int[m], r = new int[m];
    for (int i = 0; i < m; i++) {
      l[i] = sc.nextInt();
      r[i] = l[i] + sc.nextInt() - 1;
    }
    Arrays.sort(l);
    Arrays.sort(r);
    del = new HashSet<>();
    for (int cur = -1, i = 0, j = 0; i < m; i++) {
      while (j < m && r[j] <= l[i]) {
        add(j++);
      }
      if (l[i] <= cur) {
        add(i);
      } else {
        cur = r[i];
      }
    }
    System.out.println(del.size());
    for (int d : del) {
      System.out.print((d + 1) + " ");
    }
  }

  static void add(int idx) {
    if (!del.add(idx)) {
      del.remove(idx);
    }
  }
}
