import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        for (int i = 0; i < d; i++) {
            int n = sc.nextInt();
            for (int j = 1; j <= n; j++) {
                for (int k = 1; k <= n; k++) {
                    if (j + k < n + 1 || j - k > n - 1 || k - j > n - 1 || j + k > 2 * n - 1) {
                        System.out.print("#");
                    } else {
                        System.out.print(" ");
                    }
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
