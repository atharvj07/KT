#include<bits/stdc++.h>
using namespace std;
int n,m;
char a[2001][2001];
bool b[2001];
bool check(int x)
{
    int c[m];
    memset(c,0,sizeof(c));
    for(int i=1;i<=n;i++)
    {
        if(i==x) continue;
        for(int j=0;j<m;j++)
        {
            c[j]=c[j]|a[i][j]-'0';
        }
    }
    for(int i=0;i<m;i++)
    {
        if(c[i]==0) return 0;
    }
    return 1;
}
int main()
{
    cin>>n>>m;
    for(int i=1;i<=n;i++)
    {
        for(int j=0;j<m;j++)
        {
            cin>>a[i][j];
        }
    }
    for(int i=1;i<=n;i++)
    {
        if(check(i))
        {
            cout<<"YES"<<endl;
            return 0;
        }
    }
    cout<<"NO"<<endl;
    return 0;
}
