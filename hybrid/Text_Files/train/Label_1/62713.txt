n, p = map(int, input().split())
a = list(map(int, input().split()))

prefix_sums = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sums[i] = (prefix_sums[i - 1] + a[i - 1]) % p

max_score = 0
for i in range(1, n):
    first_part_score = prefix_sums[i] % p
    second_part_score = (prefix_sums[n] - prefix_sums[i] + p) % p
    total_score = (first_part_score + second_part_score) % p
    max_score = max(max_score, total_score)

print(max_score)
