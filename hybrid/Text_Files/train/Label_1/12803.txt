#include <iostream>
#include <vector>
using namespace std;
#define linf 1000000000000000000ll
vector<int> a;
int main() {
  int n;
  cin >> n;
  long long ans = linf;
  for (int i = 0, x; i < n; i++) { cin >> x, a.emplace_back(x), ans = min(ans, 1LL * x); }
  int m;
  cin >> m;
  for (int i = 0, x, y; i < m; i++) {
    cin >> x >> y;
    ans = linf;
    for (int j = 0; j < n; j++) {
      if (a[j] >= x) ans = min(ans, 1LL * y - (a[j] - x));
    }
    cout << (ans > 1e18 ? -1 : (ans + 1) / 2) << '\n';
  }
}
