#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1), p(n + 1);
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (int i = 2; i <= n; ++i) { cin >> p[i]; }
  vector<vector<int>> g(n + 1);
  vector<int> leaf, depth(n + 1), subTreeOdd(n + 1), parDepthOdd(n + 1);
  for (int i = 2; i <= n; ++i) g[p[i]].push_back(i);
  function<void(int, int, int, bool)> dfs = [&](int u, int pr, int dep, bool odd) {
    if (g[u].empty()) {
      leaf.push_back(u);
      depth[u] = dep;
      subTreeOdd[u] = odd;
      return;
    }
    for (auto v : g[u])
      if (v != pr) dfs(v, u, dep + 1, !odd);
    depth[u] = dep;
    subTreeOdd[u] = odd;
  };
  dfs(1, 1, 0, 0);
  int ans = 0;
  for (auto u : leaf) {
    int d = depth[u], odd = subTreeOdd[u];
    for (auto v : leaf) {
      if (u == v) continue;
      parDepthOdd[u] = odd;
      int otherOdd = subTreeOdd[v];
      if (d % 2 == otherOdd) continue;
      int curOdd = !(d % 2 == 0 and otherOdd == 0);
      if (curOdd != parDepthOdd[u] and curOdd != parDepthOdd[v])
        ans += 2;
    }
  }
  cout << ans / 2 << "\n";
}
