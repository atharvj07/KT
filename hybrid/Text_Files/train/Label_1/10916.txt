#include<cstdio>
#include<cstring>
#include<algorithm>
using namespace std;
const int M=2e5+10;
int n,a[M],b[M];
int main(){
    int t;
    scanf("%d",&t);
    while(t--){
        memset(b,0,sizeof(b));
        scanf("%d",&n);
        for(int i=1;i<=n;++i) scanf("%d",&a[i]);
        for(int i=1;i<=n;++i){
            if(!a[i]) continue;
            b[max(i-a[i]+1,1)]+=1;
            b[i+1]-=1;
        }
        for(int i=1;i<=n;++i) b[i]+=b[i-1];
        for(int i=1;i<=n;++i) printf("%d ",b[i]>0);
        printf("\n");
    }
    return 0;
}
