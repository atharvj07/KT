s=input()
ans=0
if len(s)!=1:
    for x in range(len(s)):
        if x==0:
            if s[x]!=s[x+1]:
                ans+=1
        elif x==len(s)-1:
            if s[x]!=s[x-1]:
                ans+=1
        else:
            if s[x]!=s[x-1] and s[x]!=s[x+1]:
                ans+=1
    print(ans)
else:
    print(1)
