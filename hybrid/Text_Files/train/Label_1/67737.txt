n = int(input())
a = [int(x) for x in input().split()]
ans = 1
ans_i = 0
ans_j = 0
for i in range(0, n):
    for j in range(i + 1, n):
        if a[j - 1] > a[j]:
            if j - i + 1 > ans:
                ans = j - i + 1
                ans_i = i
                ans_j = j
        else:
            break
if ans == 1:
    print(1)
else:
    b = a[ans_i:ans_j + 1]
    for i in range(0, len(b) - 1):
        if b[i] > b[i + 1]:
            if b[i] + 1 == b[i + 1]:
                b[i] += 1
            else:
                b[i + 1] = b[i] + 1
            break
    print(len(b))
