#include<bits/stdc++.h>
using namespace std;

int main()
{
    int m, c[105], x, y, csum=0, mx=0;
    cin>>m;
    for(int i=1; i<=m; i++){ cin>>c[i]; csum+=c[i]; mx=max(mx, c[i]); }
    cin>>x>>y;
    if(csum<=x || c[mx]<x || max(c[1], m-c[mx])<x){ cout<<0;}
    else for(int k=1; k<=m; k++){
        csum-=c[k];
        m-=c[k];
        if(csum>=x && m>=x){ cout<<k<<endl; return 0; }
    }
}
