import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            BigInteger p = sc.nextBigInteger();
            int q = sc.nextInt();
            for (int j = 2; j * j <= q; j++) {
                while (q % j == 0) {
                    q /= j;
                }
            }
            if (q > 1) {
                BigInteger gcd = p.gcd(BigInteger.valueOf(q));
                System.out.println(p.divide(gcd));
            } else {
                System.out.println(p);
            }
        }
    }
}
