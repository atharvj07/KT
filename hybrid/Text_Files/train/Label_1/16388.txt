#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;
const int maxn = 1e5 + 5;

vector<int> adj[maxn];
bool vis[maxn];
int color[maxn];
ll dp[maxn][2]; // 0 - no split, 1 - split at parent

int dfs(int u, int p) {
    vis[u] = true;
    int cnt = (color[u] == 1);
    for (int v : adj[u]) {
        if (v == p) continue;
        cnt += dfs(v, u);
    }
    return cnt;
}

ll solve(int u, int p) {
    if (~dp[u][p]) return dp[u][p];
    ll &ans = dp[u][p];
    if (p == 0) {
        ans = 1;
        return ans;
    }
    ll ret = 0;
    ret = (ret + solve(u, 0)) % mod;
    if (color[u] == 1) {
        int sz = dfs(u, p);
        if (sz > 1) ret = (ret + solve(p, u)) % mod;
    }
    return ans = ret;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;
    cin >> n;

    for (int i = 1; i < n; ++i) {
        int p;
        cin >> p;
        adj[p].push_back(i);
        adj[i].push_back(p);
    }

    for (int i = 0; i < n; ++i) {
        cin >> color[i];
    }

    memset(dp, -1, sizeof dp);
    cout << solve(0, 0) << '\n';

    return 0;
}
