#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10, mod = 1e9 + 7;
int n, a[N], suf[N], pre[N];
int32_t main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  pre[n] = a[n];
  suf[1] = a[1];
  for (int i = n - 1; i >= 1; i--) { pre[i] = pre[i + 1] + a[i]; }
  for (int i = 2; i <= n; i++) { suf[i] = suf[i - 1] + a[i]; }
  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    int sum = a[i];
    int r = i, l = i;
    if (i > 1) {
      int low = i - 1, high = n, pos = -1;
      while (low <= high) {
        int mid = (low + high) / 2;
        if (suf[mid] > pre[i - 1]) {
          high = mid - 1;
        } else {
          low = mid + 1;
          pos = mid;
        }
      }
      if (pos != -1) { sum += suf[pos]; }
    }
    if (i < n) {
      int low = i + 1, high = n, pos = -1;
      while (low <= high) {
        int mid = (low + high) / 2;
        if (pre[mid] < suf[i]) {
          low = mid + 1;
        } else {
          high = mid - 1;
          pos = mid;
        }
      }
      if (pos != -1) { sum += pre[pos]; }
    }
    if (sum != 0) { ans++; }
  }
  cout << ans;
}
