import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.StringTokenizer;

public class Main {
    static int n, k, a, b, q;
    static HashMap<Integer, Integer> orders;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        a = Integer.parseInt(st.nextToken());
        b = Integer.parseInt(st.nextToken());
        q = Integer.parseInt(st.nextToken());
        orders = new HashMap<>();
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            if (type == 1) {
                int day = Integer.parseInt(st.nextToken());
                int numOrders = Integer.parseInt(st.nextToken());
                orders.put(day, orders.getOrDefault(day, 0) + numOrders);
            } else {
                int day = Integer.parseInt(st.nextToken()) - 1;
                int[] prod = new int[n + 1]; // [day][produced]
                int[] fill = new int[n + 1]; // [day][filled]
                int days = k;
                prod[0] = a;
                while (days > 0 && day >= days) {
                    prod[day - days] = Math.min(a, prod[day - days] + (days + 1) * b);
                    days--;
                }
                int[] prefixSum = new int[n + 1];
                prefixSum[0] = orders.getOrDefault(0, 0);
                for (int j = 1; j <= n; j++) prefixSum[j] = prefixSum[j - 1] + orders.getOrDefault(j, 0);
                for (int j = 0; j <= n; j++) fill[j] = Math.min(prod[j], prefixSum[j]);
                for (int j = n - 1; j > 0; j--) fill[j] = Math.max(fill[j], fill[j + 1]);
                System.out.println(fill[day]);
            }
        }
    }
}
