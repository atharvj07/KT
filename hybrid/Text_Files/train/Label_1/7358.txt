#include <iostream>
using namespace std;
const int MOD = 1e9 + 7;

int main() {
  long long n, sx, sy, dx, dy, t;
  cin >> n >> sx >> sy >> dx >> dy >> t;
  auto add = [](long long a, long long b) { return (a + b) % MOD; };
  auto mul = [](long long a, long long b) { return (a * b) % MOD; };
  long long cdx = (dx * t + sx - 1) % n + 1;
  long long cdy = (dy * t + sy - 1) % n + 1;
  cout << cdx << ' ' << cdy << endl;
  return 0;
}
