import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    b = [x for x in a if x & 1]
    b.sort()
    if len(b) == 0 or b[0] > 1:
        print(1)
    elif len(b) == len(a):
        print(0)
    else:
        mn = 10**18
        for x in b:
            c = 0
            for y in a:
                if y % x == 0:
                    c += 1
            if c == len(a) - len(b):
                mn = min(mn, x)
        print(mn)
