#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100005], mx = -1, smx = -1, ans = -1;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        if (a[i] > mx) smx = mx, mx = a[i];
        else if (a[i] > smx) smx = a[i];
    }
    if (mx != -1) ans = (mx ^ smx);
    cout << ans;
}
