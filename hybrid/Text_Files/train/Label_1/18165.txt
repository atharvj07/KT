#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long ans = 0;
    for (long long i = 1; i <= n; i++) {
      long long j = i;
      bool ok = true;
      while (j > 0) {
        if (j % 10 != i % 10) {
          ok = false;
          break;
        }
        j = j / 10;
      }
      if (ok) ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
