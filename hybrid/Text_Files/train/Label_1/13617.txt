#include <bits/stdc++.h>
using namespace std;

#define ll long long

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (ll &x : a) { cin >> x; }
  int k = 0;
  ll orr = 0;
  for (int i = n - 1; i >= 0; i--) {
    if ((a[i] | orr) == orr) { continue; }
    orr |= a[i];
    a[k++] = a[i];
  }
  cout << k << endl;
  for (int i = 0; i < k; i++) { cout << a[i] << " "; }
  cout << endl;
}
