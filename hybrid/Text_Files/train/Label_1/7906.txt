#include <bits/stdc++.h>
using namespace std;
#define ll long long

ll m, n, a[35], b[35];
string s;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  cin >> s;
  m = s.length();
  for (int i = 0; i < m;) {
    ll x = 0, y = 0;
    if (a[s[i] - 'a'] >= 0) {
      while (i < m && s[i] == s[i + 1]) {
        x++;
        i++;
      }
      while (i < m && a[s[i] - 'a'] >= 0) {
        y++;
        i++;
      }
      if (x + y > 1) b[s[i - 1] - 'a'] += (x * y);
    } else
      i++;
  }
  for (int i = 0; i < n; i++) cout << b[i] << " ";
  return 0;
}
