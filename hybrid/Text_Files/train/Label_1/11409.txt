#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    vector<int> v(n);
    for (int& x : v) cin >> x;
    int mx = 1;
    int prv = n + 1;
    for (int x : v) {
        if (prv > x) prv = -1;
        if (~prv) mx++;
    }
    cout << mx << endl;
}
