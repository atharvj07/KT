n = int(input())
l = []
for i in range(n):
    l.append(input())
total = 0
for i in range(n):
    isClean = l[0][i] == '1'
    isAllClean = True
    for j in range(1, n):
        if l[j][i] != isClean:
            isAllClean = False
            break
    if isAllClean:
        total += 1
print(total)
