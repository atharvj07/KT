q = int(input().strip())
for _ in range(q):
    n, m = map(int, input().strip().split())
    total_pages = n // m
    cycle_sum = sum(int(str(m*i)[-1]) for i in range(1, 11))
    total_sum = cycle_sum * (total_pages // 10)
    remaining_pages = total_pages % 10
    total_sum += sum(int(str(m*i)[-1]) for i in range(1, remaining_pages + 1))
    print(total_sum)
