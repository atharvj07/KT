#include <bits/stdc++.h>
using namespace std;
int main() {
    int tt;
    cin >> tt;
    while (tt--) {
        string s;
        cin >> s;
        int len = s.length();
        int l0 = 0, l1 = 0;
        for (int i = 0; i < len; i++) {
            int x = s[i] - '0';
            l0 += (x == 0);
            l1 += (x == 1);
        }
        if (l0 == len || l1 == len) {
            if (len % 2 == 0) {
                int sum = 0;
                if (l1 == len)
                    sum = 1;
                else
                    sum = 0;
                sum += (len / 2);
                cout << sum << endl;
                continue;
            } else {
                int sum = 0;
                if (l1 == len)
                    sum = 1;
                else
                    sum = 0;
                sum += (len - 1) / 2;
                sum += 2;
                cout << sum << endl;
                continue;
            }
        }
        cout << 1 << endl;
    }
    return 0;
}
