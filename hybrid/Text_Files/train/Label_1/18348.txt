#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main()
{
    ll n;
    cin>>n;
    vector<ll>a(n);
    vector<ll>b(n);
    for(auto &idx: a)    cin>>idx;
    for(auto &idx: b)    cin>>idx;
    vector<pair<ll,ll>>v;
    for(ll i=0 ;i<n ;i++)    v.push_back({a[i]+b[i],i});
    sort(v.begin(),v.end());
    for(auto x:v) cout<<x.second<<" ";
    return 0;
}
