#include <algorithm>
#include <iostream>

using namespace std;

long long mod = 1e9 + 7;
long long quickpow(long long a, long long b) {
  long long ans = 1;
  while (b) {
    if (b & 1) { ans = ans * a % mod; }
    a = a * a % mod;
    b >>= 1;
  }
  return ans;
}
int main() {
  long long m;
  cin >> m;
  long long ans = 1;
  for (long long i = 1; i <= m; i++) {
    long long x;
    cin >> x;
    long long cnt = 0;
    long long cur = x;
    while (cur % x == 0) {
      cnt++;
      cur /= x;
    }
    long long t = quickpow(x, cnt);
    ans = ans * t % mod;
  }
  cout << ans << endl;
}
