import sys

n, m = map(int, input().split())
v = list(map(int, input().split()))
e = [0 for i in range(m)]
deg = [0 for i in range(n)]
for i in range(m):
    a, b, c = map(int, input().split())
    deg[a-1]+=1
    deg[b-1]+=1
    e[i] = c

start=-1
for i in range(n):
    if deg[i]==1:
        start=i
        break
q=[start]
ans=[v[start]]
vis=[False for i in range(n)]
vis[start]=True
while len(q)!=0:
    cur=q.pop(0)
    for i in range(m):
        if vis[(e[i]-1)//2]==False and ((e[i]-1)//2!=cur or (e[i]%2)==0):
            x, y = (cur+1, (e[i]-1)//2+1)
            if x>y:
                x, y = y, x
            vis[y-1]=True
            ans.append(v[y-1])
            q.append(y-1)
    
print("{0:.15f}".format(sum(ans)/len(ans)))
