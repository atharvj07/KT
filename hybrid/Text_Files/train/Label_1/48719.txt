import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        String s = sc.next(), t = sc.next();
        int lcm = lcm(n, m);
        if (lcm % n != 0 || lcm % m != 0) {
            System.out.println(-1);
        } else {
            int r = lcm / n, c = lcm / m;
            String[] ss = new String[r], tt = new String[c];
            for (int i = 0; i < r; i++) {
                for (int j = 0; j < n; j++) {
                    ss[i] = ss[i] == null ? String.valueOf(s.charAt(j)) : ss[i] + s.charAt(j);
                }
            }
            for (int i = 0; i < c; i++) {
                for (int j = 0; j < m; j++) {
                    tt[i] = tt[i] == null ? String.valueOf(t.charAt(j)) : tt[i] + t.charAt(j);
                }
            }
            boolean ok = true;
            for (int i = 0; i < r; i++) {
                for (int j = 0; j < c; j++) {
                    if (!ss[i].equals(tt[j])) {
                        ok = false;
                    }
                }
            }
            System.out.println(ok ? lcm : -1);
        }
    }

    public static int gcd(int a, int b) {
        if (a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }

    public static int lcm(int a, int b) {
        return (a * b) / gcd(a, b);
    }
}
