#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;
const ll MOD = 1e9 + 7;

int main() {
  string s;
  cin >> s;
  int q;
  cin >> q;
  vector<pair<char, string>> queries(q);
  for (int i = 0; i < q; i++) { cin >> queries[i].first >> queries[i].second; }
  for (int i = q - 1; i >= 0; i--) {
    char d = queries[i].first;
    string t = queries[i].second;
    for (int j = 0; j < s.length(); j++) { if (s[j] == d) s.replace(j, 1, t); }
  }
  ll ans = 0;
  ll base = 1;
  for (int i = s.length() - 1; i >= 0; i--) {
    ans = (ans + (s[i] - '0') * base) % MOD;
    base = (base * 10) % MOD;
  }
  cout << ans << endl;
  return 0;
}
