#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll t;
    cin>>t;
    while(t--) {
        ll a,b;
        cin>>a>>b;
        if(b==a) {
            cout<<0<<endl;
            continue;
        }
        if(b%2==1&&a%2==0) {
            cout<<-1<<endl;
            continue;
        }
        if(b<a) {
            if(b*8<=a) {
                cout<<1<<endl;
            }
            else {
                cout<<2<<endl;
            }
            continue;
        }
        ll cnt=0;
        while(a*2<=b) {
            a*=2;
            cnt++;
        }
        while(a*4<=b) {
            a*=4;
            cnt++;
        }
        while(a*8<=b) {
            a*=8;
            cnt++;
        }
        if(a==b) {
            cout<<cnt<<endl;
        }
        else {
            cout<<-1<<endl;
        }
    }
    return 0;
}
