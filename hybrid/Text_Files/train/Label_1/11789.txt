#include<bits/stdc++.h>
using namespace std;typedef long long ll;
const ll MOD = 1e9 + 7;
const ll MAXN = 1e5 + 5;

int main(){
    int n, m; 
    cin >> n >> m;
    vector<ll> b(n), g(m);
    for(int i=0; i<n; i++){
        cin >> b[i];
    }
    for(int i=0; i<m; i++){
        cin >> g[i];
    }
    sort(b.begin(), b.end());
    sort(g.rbegin(), g.rend());
    ll ans = 0;
    for(int i=0; i<min(n, m); i++){
        ans += max(b[i], g[i]);
    }
    if(n > m){
        for(int i=m; i<n; i++){
            ans += b[i];
        }
    }
    else if(n < m){
        for(int i=n; i<m; i++){
            ans += g[i];
        }
    }
    cout << ans << endl;   
    return 0;
}
