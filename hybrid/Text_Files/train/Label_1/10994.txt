#include <iostream>
#include <cmath>

int main() {
  long n;
  std::cin >> n;
  long layer = std::floor(std::sqrt(n) / 2);
  long m = (2*layer+1)*(2*layer+1);
  n -= m;
  m /= 2;
  long dir = 6 * layer;
  if (n <= m) {
    dir += -1;
  } else if (n <= 2 * m) {
    dir += -2;
  } else if (n <= 3 * m) {
    dir += -3;
  } else if (n <= 4 * m) {
    dir += -4;
  } else if (n <= 5 * m) {
    dir += -5;
  }
  n -= m;
  long x = 2*layer*(1 + (dir == 3 || dir == -3));
  long y = 2*layer*(1 + (dir == 1 || dir == -5));
  if (dir == -1) {
    x -= n;
  } else if (dir == -2) {
    x += n;
  } else if (dir == -3) {
    y += n;
  } else if (dir == -4) {
    x -= n;
    y -= n;
  } else if (dir == -5) {
    y -= n;
  }
  std::cout << x << " " << y << std::endl;
  return 0;
}
