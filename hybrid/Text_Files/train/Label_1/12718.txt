#include <bits/stdc++.h>
using namespace std;

int main()
{
    vector<pair<int, int>> logos;
    for (int i = 0, x, y; i < 3; ++i) {
        cin >> x >> y;
        logos.push_back({min(x, y), max(x, y)});
    }
    sort(logos.begin(), logos.end(), [](pair<int, int> a, pair<int, int> b) { return a.first + a.second < b.first + b.second; });
    const int side = logos[0].first + logos[1].first + logos[2].first;
    for (int i = (logos[0].first + logos[1].first + logos[2].first) % 2 == 0 ? logos[0].first + logos[1].first + logos[2].first - 2 : logos[0].first + logos[1].first + logos[2].first - 1; i >= logos[0].first + logos[1].first; i -= 2) {
        if (i >= logos[0].first && i >= logos[1].first && i >= logos[2].first && (i - logos[0].first - logos[1].first) % logos[2].first == 0 && i - logos[0].first - logos[1].first >= 0) {
            cout << i << '\n';
            for (int j = 0; j < i; ++j) {
                for (int k = 0; k < i; ++k) {
                    if (j < logos[0].first) {
                        if (k < logos[1].first) {
                            cout << 'A';
                        } else if (k >= logos[1].first + i - logos[0].first - logos[2].first && k < i) {
                            cout << 'C';
                        } else {
                            cout << 'B';
                        }
                    } else if (j >= logos[0].first * 2 - i && j < logos[0].first + i && k < logos[1].first + logos[2].first) {
                        cout << 'B';
                    } else {
                        cout << 'C';
                    }
                }
                cout << '\n';
            }
            return 0;
        }
    }
    cout << "-1";
}
