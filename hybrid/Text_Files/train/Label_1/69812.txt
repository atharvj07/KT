from sys import stdin
n = int(stdin.readline())
a = list(map(int, stdin.readline().split()))
vis = [False] * (n+1)
ans = -1
for i in range(1, n+1):
    if vis[i]:
        continue
    tmp = []
    cur = i
    while True:
        if vis[cur]:
            break
        vis[cur] = True
        tmp.append(cur)
        cur = a[cur-1]
    if len(tmp) % 2 == 1:
        ans = 1
        break
    else:
        x = len(tmp)
        y = (x+1) // 2
        for j in range(y):
            if tmp[j] != tmp[x-j-1]:
                ans = y+1
                break
if ans == -1:
    print(-1)
else:
    print(ans)
