#include <iostream>
using namespace std;

int main() {
	int t; cin >> t;
	while (t--) {
		string s; cin >> s;
		int n = s.size();
		int eCnt = 0; bool bad = false;
		for (int i = 0; i < n; i++) {
			if (s[i] == 'E') eCnt++;
			else if (i == 0 || s[i - 1] == 'N' || s[i] == s[(i + 1) % n]) bad = true;
		}
		cout << (bad || eCnt == n ? "NO\n" : "YES\n");
	}
}
