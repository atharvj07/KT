#include <bits/stdc++.h>
#define nl '\n'
#define sp ' '
#define ti tuple<int, short, short>
using namespace std;
vector<vector<int>> dp;
vector<ti> a;
int main() {
    int n, m;
    cin >> n >> m;
    dp.resize(n + 1, vector<int>(m + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { int k; cin >> k; a.push_back({k, i, j}); }
    }
    sort(a.rbegin(), a.rend());
    for (auto [k, i, j] : a) {
        int& p = dp[i][j];
        p = k;
        int& q = dp[i - 1][j], &w = dp[i][j - 1];
        p = max(p, max(q, w) + k);
    }
    cout << dp[n][m] << nl;
}
