n,k=map(int,input().split())
c=list(map(int,input().split()))
f=sorted(list(map(int,input().split())))
h=list(map(int,input().split()))
c.sort()
ans=0
for i in range(n):
    lc=len(c)-1
    lo=0
    hi=lc
    while lo<=hi:
        mi=(lo+hi)//2
        if c[mi]>f[i]: hi=mi-1
        else: lo=mi+1
    #print(hi)
    if hi>=0 and c[hi]==f[i]:
        for j in range(k):
            ans+=h[(hi-j)%k]
            c.pop(hi-j)
            if len(c)==0: break
            lc=len(c)-1
            lo=0
            hi=lc
            while lo<=hi:
                mi=(lo+hi)//2
                if c[mi]>f[i]: hi=mi-1
                else: lo=mi+1
            if hi>=0 and c[hi]==f[i]: pass
            else: break
print(ans)
