#include <bits/stdc++.h>
#define int long long

using namespace std;

vector<int> g[100005];

signed main() {
  int n, m;
  cin >> n >> m;
  if (m > n * (n - 1) / 2) {
    cout << "Impossible";
    return 0;
  }
  cout << "Possible\n";
  for (int i = 2; i <= n; i++) {
    g[i].push_back(1);
    g[1].push_back(i);
  }
  int remain = m - n + 1;
  for (int i = 3; i <= n && remain; i++) {
    for (int j = i + 1; j <= n && remain; j++) {
      g[i].push_back(j);
      g[j].push_back(i);
      remain--;
    }
  }
  for (int u = 1; u <= n; u++) {
    for (int v : g[u]) {
      cout << u << " " << v << "\n";
    }
  }
}
