import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        long ans = 0, cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            ans += (cur + k - 1) / k;
            cur %= k;
        }
        System.out.println(ans);
    }
}
