#include <iostream>
#include <cstring>
using namespace std;
int t[105][105], d[105], a[105];

int main() {
  int n, i, j;
  cin >> n >> i;
  while (i--) {
    for (j = 0; j < n; j++) { cin >> a[j]; }
    for (j = 0; j < n; j++)
      if (a[j])
        for (int k = 0; k < j; k++)
          t[j][k] = 0, t[k][j] = 0;
  }
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++) d[i] += t[i][j];
  for (i = 0, j = 0; i < n; i++)
    if (d[i] == n - 1) j++;
  cout << j;
}
