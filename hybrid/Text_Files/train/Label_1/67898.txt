t = int(input())
for _ in range(t):
    n, k, d1, d2 = map(int, input().split())
    possible = False
    for x1, x2 in [(d1, d2), (d1, -d2), (-d1, d2), (-d1, -d2)]:
        w1 = k + x1 + x2
        w2 = k - x1 + x2
        w3 = k - x2
        max_w = max(w1, w2, w3)
        total = 3 * max_w - w1 - w2 - w3
        if w1 >= 0 and w2 >= 0 and w3 >= 0 and total <= n - k and (n - k - total) % 3 == 0:
            possible = True
            break
    print("yes" if possible else "no")
