#include<bits/stdc++.h>
using namespace std;
int n,m,a[51][51],vis[51][51],ans;
int dx[]={1,-1,0,0},dy[]={0,0,1,-1};
void dfs(int x,int y,int c){
    if(x<0||x>=n||y<0||y>=m||a[x][y]!=c||vis[x][y])return;
    vis[x][y]=1;
    ans++;
    for(int i=0;i<4;++i)dfs(x+dx[i],y+dy[i],c);
}
int main(){
    cin>>n>>m;
    for(int i=0;i<n;++i){
	for(int j=0;j<m;++j)cin>>a[i][j];
    }
    for(int i=0;i<n;++i){
	for(int j=0;j<m;++j){
	    if(vis[i][j]==0){
		ans=0;
		dfs(i,j,a[i][j]);
		if(ans>1)ans--;
	    }
	}
    }
    cout<<ans<<'\n';
}
