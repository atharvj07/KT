#include <vector>
#include <iostream>
#include <algorithm>
#define ll long long

using namespace std;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) cin >> a[i];
    ll ans = a[0];
    for (ll l = 1, r = 0; l < n; l++) {
        while (r + 1 < n && a[r + 1] >= 0) r++;
        if (l <= r) ans += a[l];
        else break;
    }
    for (ll l = 0, r = n - 1; r >= 0; r--) {
        while (l - 1 >= 0 && a[l - 1] <= 0) l--;
        if (l <= r) ans += a[r];
        else break;
    }
    cout << ans << '\n';
    return 0;
}
