#include <bits/stdc++.h>
#define nl '\n'
using namespace std;

struct node {
    int v, from;
};

int k, sum[17], win;
long long ans[17][5009];
vector<int> box[17];

int main() {
    cin >> k;
    for (int i = 1; i <= k; i++) {
        int n;
        cin >> n;
        for (int j = 1; j <= n; j++) {
            int x;
            cin >> x;
            sum[i] += x;
            box[i].push_back(x);
        }
    }

    for (int i = 1; i <= k; i++) {
        sort(box[i].begin(), box[i].end());
        win += box[i].size() % 2;
    }
    if (win > 1) {
        cout << "No" << nl;
        return 0;
    }

    bool fn = 0;
    for (int i = 1; i <= k; i++) {
        bool f = 0;
        for (int j = 1; j <= k; j++) {
            if (i == j) continue;
            if (sum[i] + box[i].back() == sum[j]) {
                f = 1;
                break;
            }
        }
        if (!f) {
            fn = 1;
            break;
        }
    }
    if (fn) {
        cout << "No" << nl;
        return 0;
    }

    cout << "Yes" << nl;
    for (int i = 1; i <= k; i++) {
        int mx = 0, idx = 0;
        for (int j = 1; j <= k; j++) {
            if (sum[i] + box[i].back() == sum[j] && mx <= box[i].size()) {
                mx = box[i].size();
                idx = j;
            }
        }
        cout << box[i].back() << ' ' << idx << nl;
        sum[idx] += box[i].back();
        box[i].pop_back();
    }
    return 0;
}
