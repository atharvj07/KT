#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, b;
  cin >> n >> b;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n - 1; i++) {
    if (abs(a[i] - a[i + 1]) <= b) {
      ans++;
      b -= abs(a[i] - a[i + 1]);
    }
  }
  cout << ans << endl;
  return 0;
}
