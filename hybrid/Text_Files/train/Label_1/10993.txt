#include <iostream>

using namespace std;

int main() {
  long long n;
  cin >> n;
  long long x = 0, y = 0, d = 0;
  for (; 6 * d * d < n; ++d) {
    n -= 6 * d * d;
    y += d + 1;
  }
  for (; n > 0; --n) {
    if (d % 2) {
      x--;
    } else {
      y--;
    }
  }
  cout << x << ' ' << y << '\n';
}
