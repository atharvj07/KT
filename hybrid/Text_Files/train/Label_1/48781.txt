import java.util.Scanner;

public class Main {
    static long gcd(long a, long b) {
        if(a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] ans = new long[n];
        long g = -1;
        for(int i = 0; i < n; i++) {
            ans[i] = in.nextLong();
            ans[i] *= in.nextInt();
        }
        for(int i = 0; i < n; i++) {
            if(g == -1) {
                g = ans[i];
            } else {
                g = gcd(g, ans[i]);
            }
        }
        if(g == 1) {
            System.out.println(-1);
        } else {
            System.out.println(g);
        }
    }
}
