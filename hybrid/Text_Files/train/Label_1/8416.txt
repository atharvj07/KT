#include <algorithm>
#include <iostream>
using namespace std;
int n, a[2005], b[2005], f[2005][2005];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  copy(a + 1, a + n + 1, b + 1);
  for (int i = n; i >= 1; i--) {
    for (int j = i; j <= n; j++) {
      f[i][j] = max(f[i][j - 1], f[i + 1][j]);
      if (a[i] <= a[j]) { f[i][j] = max(f[i][j], f[i + 1][j - 1] + 1); }
    }
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      if (b[i] <= b[j]) { ans = max(ans, f[i][j]); }
      swap(b[i], b[j]);
      if (b[i] <= b[j]) { ans = max(ans, f[i][j]); }
      swap(b[i], b[j]);
    }
  }
  cout << ans << endl;
  return 0;
}
