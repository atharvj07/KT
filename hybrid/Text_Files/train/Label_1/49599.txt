import java.util.*;
import java.io.*;

public class Main {
    static int n, m, k;
    static int[] p, s, c;
    static int[][] a;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        m = input.nextInt();
        k = input.nextInt();

        p = new int[n + 1];
        s = new int[n + 1];
        c = new int[k + 1];

        for (int i = 1; i <= n; i++)
            p[i] = input.nextInt();

        for (int i = 1; i <= n; i++)
            s[i] = input.nextInt();

        for (int i = 1; i <= k; i++)
            c[i] = input.nextInt();

        a = new int[m + 1][n + 1];

        for (int i = 1; i <= m; i++)
            for (int j = 1; j <= n; j++)
                if (i == s[j])
                    a[i][j] = p[j];

        int ans = 0;

        for (int i = 1; i <= k; i++)
            if (p[c[i]] != get(s[c[i]]))
                ans++;

        System.out.println(ans);
    }

    static int get(int x) {
        int sum = 0;

        for (int i = 1; i <= m; i++)
            sum = Math.max(sum, a[i][x]);

        return sum;
    }
}
