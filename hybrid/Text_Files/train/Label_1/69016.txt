n, r = map(int, input().split())
a = list(map(int, input().split()))

heaters_on = 0
current_pos = 0

while current_pos < n:
    # Find the farthest heater to the right that can warm up the current_pos
    best_heater = -1
    for i in range(max(0, current_pos - r + 1), min(n, current_pos + r)):
        if a[i] == 1 and i + r - 1 >= current_pos:
            best_heater = i

    if best_heater == -1:
        # No heater can warm up the current position
        print(-1)
        exit(0)
    else:
        # Turn on the best heater found and update the current position
        heaters_on += 1
        current_pos = best_heater + r

print(heaters_on)
