x, y, x_0, y_0 = map(int, input().split())
s = input()

visited = set([(x_0, y_0)])
results = [1]

for command in s:
    if command == 'U':
        x_0 = max(1, x_0 - 1)
    elif command == 'D':
        x_0 = min(x, x_0 + 1)
    elif command == 'L':
        y_0 = max(1, y_0 - 1)
    elif command == 'R':
        y_0 = min(y, y_0 + 1)
    
    if (x_0, y_0) in visited:
        results.append(0)
    else:
        visited.add((x_0, y_0))
        results.append(1)

results[-1] += x * y - len(visited)
print(' '.join(map(str, results)))
