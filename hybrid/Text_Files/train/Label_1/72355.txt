import queue

n = int(input())
q = queue.Queue()
t = 0
max_size = 0

for _ in range(n):
    ti, ci = map(int, input().split())
    t = max(t, ti)
    q.put([ti, ci])
    if q.qsize() > max_size: max_size = q.qsize()
    while not q.empty() and t - q.queue[0][0] >= 1: q.get()
    t += min(ci, q.qsize())
    
print(t, max_size)
