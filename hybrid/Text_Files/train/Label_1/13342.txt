#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int mx = 3e5;

int main(){
    ll t, n, m, a[mx], b[mx], c[mx];
    cin >> t;
    while(t--){
        cin >> n >> m;
        for(int i = 0; i < n; ++i) cin >> a[i];
        for(int i = 0; i < m; ++i) cin >> b[i];
        for(int i = 0; i < n; ++i) c[a[i]] += b[i];
        sort(c, c + m + 1);
        ll ans = 0;
        for(int i = 0; i < n; ++i) ans += min(c[a[i]], c[a[i] - 1]);
        cout << ans << "\n";
        fill(c, c + m + 1, 0);
    }
    return 0;
}
