#include<bits/stdc++.h>
using namespace std;

int main(){
    int n,m;
    cin>>n>>m;
    int odd=0;
    for(int i=1;i<=n;i++){
        int cnt=0;
        for(int j=1;j<=m;j++){
            int x; cin>>x;
            cnt+=x;
        }
        if(cnt%2!=0) odd++;
    }
    if(odd%2==0) cout<<n;
    else cout<<n+1;
    return 0;
}
