import java.util.Scanner;

public class JabberChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String jabberId = scanner.nextLine();

        if (isValidJabberId(jabberId)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isValidJabberId(String jabberId) {
        String[] parts = jabberId.split("@");
        if (parts.length != 2) {
            return false;
        }

        String username = parts[0];
        String hostname = parts[1];

        if (!isValidUsername(username)) {
            return false;
        }

        if (!isValidHostname(hostname)) {
            return false;
        }

        if (jabberId.contains("[") && !jabberId.contains("]")) {
            return false;
        }

        if (jabberId.contains("]") && !jabberId.contains("[")) {
            return false;
        }

        if (jabberId.contains("[") && jabberId.contains("]")) {
            String resource = jabberId.substring(jabberId.indexOf("[") + 1, jabberId.indexOf("]"));
            if (!isValidResource(resource)) {
                return false;
            }
        }

        return true;
    }

    private static boolean isValidUsername(String username) {
        if (username.length() < 1 || username.length() > 16) {
            return false;
        }

        for (char c : username.toCharArray()) {
            if (!Character.isLetterOrDigit(c) && c != '_') {
                return false;
            }
        }

        return true;
    }

    private static boolean isValidHostname(String hostname) {
        if (hostname.length() < 1 || hostname.length() > 32) {
            return false;
        }

        String[] words = hostname.split("\\.");
        for (String word : words) {
            if (!isValidUsername(word)) {
                return false;
            }
        }

        return true;
    }

    private static boolean isValidResource(String resource) {
        if (resource.length() < 1 || resource.length() > 16) {
            return false;
        }

        for (char c : resource.toCharArray()) {
            if (!Character.isLetterOrDigit(c) && c != '_') {
                return false;
            }
        }

        return true;
    }
}
