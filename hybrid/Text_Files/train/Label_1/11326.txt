#include <iostream>
#include <utility>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 4 != 0) {
      cout << "NO\n";
      continue;
    }
    cout << "YES\n";
    vector<pair<int, int>> v;
    v.push_back(make_pair(n / 2 + 1, n % 2));
    int j = n % 2 + 2;
    for (int i = 2; i <= n / 2; i++) {
      v.push_back(make_pair(i, j));
      j += 2;
    }
    for (int i = 0; i < (int)v.size(); i++) {
      cout << v[i].first << " ";
    }
    for (int i = 0; i < (int)v.size(); i++) {
      cout << v[i].second << " ";
    }
    cout << "\n";
  }
  return 0;
}
