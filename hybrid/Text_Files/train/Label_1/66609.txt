n, m = map(int, input().split())
edges = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v = map(int, input().split())
    if u < v:
        edges[u].append(v)
    else:
        edges[v].append(u)

dp = [1] * (n + 1)
for i in range(1, n + 1):
    for j in edges[i]:
        if i < j:
            dp[j] = max(dp[j], dp[i] + 1)

max_beauty = 0
for i in range(1, n + 1):
    max_beauty = max(max_beauty, dp[i] * len(edges[i]))

print(max_beauty)
