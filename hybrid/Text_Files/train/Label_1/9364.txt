#include <iostream>
#include <vector>
#include <deque>
using namespace std;

int main() {
    string t, p, a;
    deque<int> a_;
    cin >> t >> p >> a;
    a_.resize(a.size());
    for (int i = 0; i < a.size(); ++i) {
        a_[i] = a[i] - '1';
    }
    int j = 0;
    for (int i = 0; i < t.size(); ++i) {
        if (j == p.size()) break;
        if (a_[i] >= j && t[a_[i]] == p[j]) {
            ++j;
        }
    }
    cout << j << endl;
    return 0;
}
