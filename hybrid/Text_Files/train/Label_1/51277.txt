import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[][] t = new char[n][n];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < n; j++) {
                t[i][j] = s.charAt(j);
            }
        }

        int[][][] dp = new int[n][n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 2; k++) {
                    dp[i][j][k] = -1;
                }
            }
        }

        int ans = solve(t, 0, 0, 0, dp);
        if (ans == 1) System.out.println("FIRST");
        else if (ans == -1) System.out.println("SECOND");
        else System.out.println("DRAW");
    }

    private static int solve(char[][] t, int i, int j, int turn, int[][][] dp) {
        if (i == t.length - 1 && j == t[0].length - 1) {
            return 0;
        }
        if (dp[i][j][turn] != -1) return dp[i][j][turn];
        int ans = 0;
        int aCount = 0;
        int bCount = 0;
        if (i + 1 < t.length) {
            int next = solve(t, i + 1, j, 1 - turn, dp);
            if (next == 0 || (next == 1 && turn == 0)) {
                aCount++;
            } else if (next == -1 && turn == 1) {
                bCount++;
            }
        }
        if (j + 1 < t[0].length) {
            int next = solve(t, i, j + 1, 1 - turn, dp);
            if (next == 0 || (next == 1 && turn == 0)) {
                aCount++;
            } else if (next == -1 && turn == 1) {
                bCount++;
            }
        }
        if (turn == 0) {
            if (aCount > 0) ans = 1;
            else if (bCount > 0) ans = -1;
            else ans = 0;
        } else {
            if (aCount > bCount) ans = 1;
            else if (bCount > aCount) ans = -1;
            else ans = 0;
        }
        dp[i][j][turn] = ans;
        return ans;
    }
}
