#include <iostream>
#include <vector>

using namespace std;

int main() {
    // Variables
    long long a, s, b;
    int t;

    // Input
    cin >> t;
    while (t--) {
        cin >> a >> s;

        // Check if Tanya's algorithm will result in s
        b = s - a;
        if (a == 0 || (a * 2 < s)) {
            cout << -1 << endl;

            if (b % 10 == 0 && (b / 10) % 10 > 0) b /= 10;
            cout << b << endl;
        }
    }

    return 0;
}
