for _ in range(int(input())):
    n = int(input())
    s = input()
    mint = s
    for k in range(1, n):
        t = ''
        for i in range(n-k+1):
            t += s[i:i+k][::-1]
        if t < mint:
            mint = t
    for k in range(1, n):
        t = s[0:n-k] + s[n-k:n]
        if t <= mint:
            mink = k
            break
    print(mint)
    print(mink)
