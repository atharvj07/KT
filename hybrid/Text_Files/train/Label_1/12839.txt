#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, i, s;
  long long int j;
  cin >> n >> j;
  vector<long long int> a(n);
  for (int k = 0; k < n; k++) {
    cin >> a[k];
  }
  sort(a.begin(), a.end());
  s = a[0];
  for (int k = 1; k < n; k++) {
    if (a[k] != a[k - 1]) {
      a[i++] = a[k];
    }
  }
  a.resize(i);
  if (8 * j < n) {
    return cout << -1, 0;
  }
  long long int ans = 1e18;
  if (j < a.size()) {
    for (int k = 0; k < (int)a.size() - j + 1; k++) {
      long long int sum = 0;
      for (int l = 0; l < j - 1; l++) {
        sum +=
            max({0ll, a[k] - a[l + 1], a[k + j - 1] - a[l + 1], a[k] - a[l], a[k + j - 1] - a[l]});
      }
      ans = min(ans, sum / 2);
    }
  }
  cout << ans << endl;
  return 0;
}
