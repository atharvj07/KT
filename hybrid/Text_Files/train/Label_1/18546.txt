#include <bits/stdc++.h>
using namespace std;
const int mod=998244353,MAXN=2001;
int dp[MAXN][MAXN];
int main(){
    int n,m,k;cin>>n>>m>>k;
    dp[0][0]=1;
    for(int i=1;i<=n;i++){
        dp[i][0]=(dp[i-1][0]*(m-1))%mod;
        dp[i][1]=(dp[i-1][0]*max(0,m-1))%mod;
        for(int j=2;j<=i;j++){
            dp[i][j]=(dp[i-1][j]*max(0,m-k))%mod;
            dp[i][j]+=((dp[i-1][j-1]*max(0,m-k+1))%mod+dp[i-1][j-2]*max(0,m-k))%mod;
            dp[i][j]%=mod;
        }
    }
    cout<<dp[n][k]<<'\n';
}
