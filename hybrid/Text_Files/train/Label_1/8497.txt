#include <bits/stdc++.h>
#define int long long
const int MOD = 1e9 + 7;
using namespace std;
main() {
  int n, k, c, d, sum = 0;
  vector<int> a(n);
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i], sum += a[i];
  sort(a.begin(), a.end());
  c = n / 2, d = n - c;
  if (sum > k) cout << -1 << endl, exit(0);
  int ans = (d - 1) / 2 + ceil(sum / 2.0) / k + 1;
  int ans2 = 1;
  for (int i = 1; i <= c; i++) ans2 = (ans2 * i) % MOD;
  for (int i = 1; i <= d; i++) ans2 = (ans2 * i) % MOD;
  cout << ans << endl;
  cout << ans2 << endl;
}
