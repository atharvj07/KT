#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
const ll mod = 998244353;
ll f[N], ifac[N], pw[N];
ll qpow(ll x, ll y) {
    ll ans = 1;
    x %= mod;
    while (y) {
        if (y & 1) ans = ans * x % mod;
        x = x * x % mod;
        y >>= 1;
    }
    return ans;
}
void pre() {
    f[0] = 1;
    for (int i = 1; i < N; i++) f[i] = f[i - 1] * i % mod;
    ifac[N - 1] = qpow(f[N - 1], mod - 2);
    for (int i = N - 2; i >= 0; i--) ifac[i] = ifac[i + 1] * (i + 1) % mod;
    pw[0] = 1;
    for (int i = 1; i < N; i++) pw[i] = pw[i - 1] * 10 % mod;
}
ll C(int n, int m) {
    if (n < m) return 0;
    return f[n] * ifac[n - m] % mod * ifac[m] % mod;
}
int main() {
    pre();
    int n;
    cin >> n;
    int up = 10 * n + 1, cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (i == 1) {
            cout << qpow(10, n) * pw[n - 1] % mod << endl;
            continue;
        }
        if (i == n) {
            cout << (qpow(10, n) - 9) * pw[n - 1] % mod << endl;
            continue;
        }
        int l = 1, r = n - i, ans = 0, tot = n + i - 1, k = i - 1;
        while (l <= r) {
            int mid = (l + r) >> 1;
            if (C(tot, mid) >= up) {
                r = mid - 1;
                k = mid;
            } else
                l = mid + 1;
        }
        ans = (C(tot, k) - C(tot - 1, k - 1) + mod) % mod;
        cnt = (cnt + ans) % mod;
        cout << cnt << endl;
    }
    return 0;
}
