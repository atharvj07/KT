#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'

const int MX = 1e4 + 5;
char grid[MX][MX];
int dp[MX][MX];
int32_t main() {
	ios_base::sync_with_stdio(0); cin.tie(0);
	int n, m;
	cin >> n >> m;
	int ans = 0;
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= m; j++) {
			cin >> grid[i][j];
		}
	}
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= m; j++) {
			if (i == 1) dp[i][j] = 1;
			else {
				if (grid[i][j] != grid[i - 1][j]) dp[i][j] = dp[i - 1][j] + 1;
				else dp[i][j] = 1;
			}
		}
	}
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= m; j++) {
			if (dp[i][j] >= 3) ans += dp[i][j] - 2;
		}
	}
	cout << ans << endl;
	return 0;
}
