#include <iostream>
#include <string>

using namespace std;

const int MAXN = 20;

bool is_divisible(string s, string t) {
    int n = s.size();
    int m = t.size();
    if (n % m != 0) return false;
    for (int i = 0; i < n; i++) {
        if (s[i] != t[i % m]) return false;
    }
    return true;
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        string s, t;
        cin >> s >> t;
        if (is_divisible(s, t)) {
            cout << s << endl;
        } else if (is_divisible(t, s)) {
            cout << t << endl;
        } else {
            cout << -1 << endl;
        }
    }
    return 0;
}
