import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read input
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int m = scanner.nextInt();

        int[] x = new int[m];
        int[] y = new int[m];
        int[] z = new int[m];
        for (int i = 0; i < m; i++) {
            x[i] = scanner.nextInt();
            y[i] = scanner.nextInt();
            z[i] = scanner.nextInt();
        }
        int k = scanner.nextInt();

        // Solve the problem
        int[] sequence = new int[n];
        for (int i = 0; i < n; i++) {
            sequence[i] = a + i * d;
        }

        for (int i = 0; i < m; i++) {
            if (x[i] == 0) {
                reverse(sequence, y[i] - 1, z[i] - 1);
            } else if (x[i] == 1) {
                increment(sequence, y[i] - 1, z[i] - 1);
            } else {
                halve(sequence, y[i] - 1, z[i] - 1);
            }
        }

        // Output the result
        System.out.println(sequence[k - 1]);
    }

    // Reverse the subarray from index start to index end (inclusive)
    private static void reverse(int[] arr, int start, int end) {
        while (start < end) {
            int temp = arr[start];
            arr[start] = arr[end];
            arr[end] = temp;
            start++;
            end--;
        }
    }

    // Increment each element in the subarray from index start to index end (inclusive) by 1
    private static void increment(int[] arr, int start, int end) {
        for (int i = start; i <= end; i++) {
            arr[i]++;
        }
    }

    // Halve each element in the subarray from index start to index end (inclusive), rounding down
    private static void halve(int[] arr, int start, int end) {
        for (int i = start; i <= end; i++) {
            arr[i] /= 2;
        }
    }
}
