def is_sharpenable(a):
  n = len(a)
  b = [a[i + 1] - a[i] for i in range(n - 1)]
  for i in range(1, n - 1):
    if b[i] * b[i - 1] < 0:
      return True
  if a[n - 1] > a[n - 2]:
    return True
  return False

t = int(input())
for _ in range(t):
  n = int(input())
  a = list(map(int, input().split()))
  print("Yes" if is_sharpenable(a) else "No")
