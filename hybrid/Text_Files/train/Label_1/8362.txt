#include<bits/stdc++.h>
using namespace std;
int p,q,l,r;
const int maxn=55;
struct node{
    int c,d;
};
vector<node>a,b;
bool ok(int x){
    for(int i=0;i<p;++i){
        for(int j=0;j<q;++j){
            if(x+a[i].c<=b[j].d && x+a[i].d>=b[j].c)
                return true;
        }
    }
    return false;
}
int main(){
    cin>>p>>q>>l>>r;
    for(int i=0;i<p;++i){
        int x,y;
        cin>>x>>y;
        a.push_back(node{x,y});
    }
    for(int i=0;i<q;++i){
        int x,y;
        cin>>x>>y;
        b.push_back(node{x,y});
    }
    int ans=0;
    for(int i=l;i<=r;++i){
        if(ok(i))
            ++ans;
    }
    cout<<ans<<'\n';
    return 0;
}
