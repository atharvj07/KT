import sys
input=sys.stdin.readline

n,m=map(int,input().split())
a=[list(map(int,input().split())) for i in range(m)]
adj=[[] for i in range(n+1)]
for u,v in a:
    adj[u].append(v)
    adj[v].append(u)
vis=[0]*(n+1)

def dfs(i,g):
    vis[i]=1
    ans.append(i)
    for j in adj[i]:
        if vis[j]!=g: dfs(j,g)

ans=[]
for i in range(1,n+1):
    if not vis[i]:
        dfs(i,1)
    if len(ans)>n//2: break
print(len(ans)*(n-len(ans)))
