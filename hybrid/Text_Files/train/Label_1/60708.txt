import sys

input = sys.stdin.readline

n, k, s, t = map(int, input().split())

cars = []
for _ in range(n):
    cars.append(list(map(int, input().split())))

gas = [int(input()) for _ in range(k)]
gas.sort()

cars.sort(key=lambda x: x[0])

best = -1
for c, v in cars:
    time = 0
    fuel = v
    for g in gas:
        if time + (g - time) / 2 + s - g <= t and fuel >= g - time:
            fuel -= g - time
            time = g
        else:
            break
    if time + (s - time) / 2 <= t and fuel >= s - time:
        best = c
        break

print(best)
