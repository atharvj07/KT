import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        char[] s = br.readLine().toCharArray();
        int len = s.length;
        int i, j;
        for (i = 0; i < len - 1; i++) {
            if (s[i] == 'y' && s[i + 1] == 'x') {
                char t = s[i];
                s[i] = s[i + 1];
                s[i + 1] = t;
                break;
            }
        }
        if (i == len - 1) {
            for (i = 0; i < len - 1; i++) {
                if (s[i] == 'x' && s[i + 1] == 'y') {
                    for (j = i; j + 1 < len; j++) {
                        s[j] = s[j + 1];
                    }
                    len--;
                    break;
                }
            }
        }
        for (i = 0; i < len; i++) {
            System.out.print(s[i]);
        }
    }
}
