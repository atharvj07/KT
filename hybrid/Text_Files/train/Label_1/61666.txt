n,m=list(map(int,input().split()))
a=[]
for i in range(n):
	a.append(input().replace('W',"1").replace('B',"-1"))
b=[[0 for j in range(m)]for i in range(n)]
for i in range (n):
	for j in range(m):
		if i-1>=0: b[i][j]=int(a[i][j])+int(b[i-1][j])
		else: b[i][j]=int(a[i][j])
for j in range(1,m):
	for i in range(n):
		b[i][j]+=b[i][j-1]
for i in range(1,n):
	for j in range(1,m):
		b[i][j]-=b[i-1][j-1]
mn=n*m*5
for i in range(n):
	for j in range(m):
		for x in range(i,n):
			for y in range(j,m):
				v=b[x][y]
				if i-1>=0: v-=b[i-1][y]
				if j-1>=0: v-=b[x][j-1]
				if i-1>=0 and j-1>=0: v+=b[i-1][j-1]
				mn=min(mn,abs(v))
print(mn)
