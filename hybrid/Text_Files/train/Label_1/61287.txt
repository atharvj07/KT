from sys import stdin
input=stdin.readline
n,k=map(int,input().split())
a=list(map(int,input().split()))
b=[bool(x!=-1)for x in a]
l=[-1]*(n+1);r=[-1]*(n+1)
for i in range (n):
 if b[i]==0:l[i+1]=i
for i in range (n-1,-1,-1):
 if b[i]==0:r[i]=i
sum=0
for i in range(n):
 if b[i]==0:sum+=r[i]-l[i]-1
print(pow(k,sum,998244353))
