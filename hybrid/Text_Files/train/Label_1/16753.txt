#include <iostream>
#include <set>
using namespace std;
int main() {
  string s;
  cin >> s;
  int c = 0;
  set<char> ss;
  for (auto x : s) {
    if (ss.count(x))
      c--;
    else {
      c++;
      ss.insert(x);
    }
  }
  if (c % 2 == 0)
    cout << "CHAT WITH HER!" << endl;
  else
    cout << "IGNORE HIM!" << endl;
  return 0;
}
