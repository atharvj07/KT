def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return a * b // gcd(a, b)

n = int(input())
crushes = list(map(int, input().split()))

visited = [False] * n
result = 1

for i in range(n):
    if not visited[i]:
        cycle_length = 0
        current = i
        while not visited[current]:
            visited[current] = True
            current = crushes[current] - 1
            cycle_length += 1
        if current != i:
            result = -1
            break
        else:
            result = lcm(result, cycle_length)

print(result)
