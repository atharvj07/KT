from collections import defaultdict

n = int(input())
d = defaultdict(lambda: defaultdict(int))
modulo = 10**9 + 7
for _ in range(n):
    x, y = map(int, input().split())
    d[x][y] += 1
ans = 1
for dx in d:
    dy = d[dx]
    options = 0
    for dyy in dy.values():
        options += dyy
    ans *= pow(pow(3, dyy, modulo), options, modulo)
    ans %= modulo
print(ans)
