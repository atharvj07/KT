#include <bits/stdc++.h>
#define REP(i, n) for (int i = 0; i < (n); ++i)
using namespace std;
const int N = 201;
char s[N];
int a[N];
int main() {
    int n;
    cin >> n;
    cin >> s;
    int mx = 0;
    REP(i, n) { a[i] = s[i] - 'a', mx = max(mx, a[i]); }
    int mn = n;
    REP(i, n) {
        if (s[i] > s[i + 1]) mn = i;
    }
    if (mn == n) {
        puts("YES");
        REP(i, n) cout << 0;
        return 0;
    }
    if (mn == 0) {
        puts("NO");
        return 0;
    }
    printf("YES\n");
    REP(i, n) { cout << (a[i] == mx ? 0 : 1); }
    return 0;
}
