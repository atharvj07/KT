import java.util.*;
import java.lang.*;

class Main{
    static boolean[] visited;
    static int[] t, a;
    static int n, q, k;
    static void dfs(int node)
    {
        if(visited[node])
            return;
        TreeSet<Integer> s = new TreeSet<>((x, y) -> Integer.compare(a[x], a[y]));
        visited[node] = true;
        s.add(node);
        for(int i=1; i<=n; ++i)
        {
            if(!visited[i] && t[node]==t[i])
            {
                s.add(i);
                dfs(i);
            }
        }
        int cnt = s.size();
        int cur[] = new int[cnt+1];
        cur[0] = -1;
        for(Integer u: s)
        {
            cur[cnt--] = u;
        }
        for(Integer u: s)
        {
            a[u] = cur[visited[cur[cnt+1]]? 0 : 1];
        }
    }
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        n= sc.nextInt();
        k= sc.nextInt();
        q= sc.nextInt();
        t= new int[n+1];
        a= new int[n+1];
        visited = new boolean[n+1];
        for(int i=1; i<=n; ++i)
        {
            t[i] = sc.nextInt();
        }
        dfs(1);
        while(q-->0)
        {
            int type = sc.nextInt();
            int id = sc.nextInt();
            if(type==1)
            {
                visited[id] = true;
            }
            else
            {
                System.out.println(a[id]==1 ? "YES":"NO");
            }
        }
    }
}
