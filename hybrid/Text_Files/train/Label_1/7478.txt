#include <bits/stdc++.h>
using namespace std;

const int MAXN = 5005;

int n, a[MAXN], vis[MAXN], dp[MAXN][MAXN], lg[MAXN], T[8 * MAXN], L[8 * MAXN];
vector<int> g[MAXN];

int get(int l, int r) {
    int p = lg[r - l + 1];
    return max(T[L[p] + l], T[L[p] + r - (1 << p) + 1]);
}

void upd(int p, int v) {
    T[p] = v;
    if (p > 1) upd(p / 2, max(T[p / 2], v));
}

int dfs(int u, int fa) {
    dp[u][u] = a[u];
    for (int v : g[u])
        if (v != fa && !vis[v]) {
            dp[u][v] = dp[u][u] ^ a[v];
            dfs(v, u);
            for (int i = v; i != u; i = dp[u][i])
                dp[u][i] = dp[u][u] ^ dp[i][v];
        }
}

void build(int p, int l, int r) {
    if (l == r) {
        T[p] = a[l];
        L[p] = l;
    } else {
        int mid = (l + r) / 2;
        build(p * 2, l, mid);
        build(p * 2 + 1, mid + 1, r);
        L[p] = get(l, mid) > get(mid + 1, r) ? L[p * 2] : L[p * 2 + 1];
        T[p] = max(T[p * 2], T[p * 2 + 1]);
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 2; i <= n; i++) {
        int x, y;
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    dfs(1, 1);
    build(1, 1, n);
    for (int i = 1; i <= n; i++) {
        int l = 1, r = i, ans = 0;
        while (l <= r) {
            int mid = (l + r) / 2;
            if (get(mid, i) > a[i]) {
                ans = mid;
                r = mid - 1;
            } else
                l = mid + 1;
        }
        for (int j = 1; j <= n; j++) {
            if (dp[i][j] == dp[i][ans]) {
                vis[j] = 1;
                upd(1, dp[i][j]);
                ans = j;
            }
        }
    }
    cout << T[1] << endl;
    return 0;
}
