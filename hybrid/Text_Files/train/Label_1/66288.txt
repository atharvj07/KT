t = int(input())
for _ in range(t):
    n = int(input())
    for x in range(n//3 + 1):
        for y in range(n//5 + 1):
            z = (n - 3*x - 5*y) / 7
            if z.is_integer() and z >= 0:
                print(f"{x} {y} {int(z)}")
                break
        else:
            continue
        break
    else:
        print(-1)
