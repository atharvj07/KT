#include <bits/stdc++.h>
using namespace std;

const int MAXN = 200005;

int v[MAXN], d[MAXN];

int main() {
  int n; cin >> n;
  for (int i = 1; i <= n; ++i) { cin >> v[i]; d[i] = v[i] - v[i - 1]; }
  int ans = 0;
  int cur = 0;
  for (int i = 1; i <= n; ++i) {
    cur += d[i];
    if (cur < 0) ans -= cur, cur = 0;
  }
  cout << ans << endl;
  return 0;
}
