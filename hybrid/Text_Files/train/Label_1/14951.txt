#include <iostream>
#include <map>
#include <string>

using namespace std;

map<string, string> l1_l2, l2_l1;
int main() {
  int n, m;
  cin >> n >> m;
  while (m--) {
    string a, b;
    cin >> a >> b;
    if (a.size() <= b.size()) {
      l1_l2[a] = b;
      l2_l1[b] = a;
    } else {
      l1_l2[b] = a;
      l2_l1[a] = b;
    }
  }
  while (n--) {
    string word;
    cin >> word;
    if (l1_l2.count(word))
      cout << l1_l2[word] << ' ';
    else
      cout << l2_l1[word] << ' ';
  }
}
