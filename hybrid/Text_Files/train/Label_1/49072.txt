import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        int k = sc.nextInt();
        long res = r, cnt = 0;
        for (long i = 0; i < 63; i++) {
            if ((l & (1 << i)) != (r & (1 << i))) {
                res &= ~(1 << i);
                cnt++;
            }
            if (cnt == k) break;
        }
        System.out.println(res);
        System.out.println(cnt);
        for (long i = l; i <= r; i++) {
            if ((i ^ res) == 0) {
                System.out.print(i + " ");
            }
        }
    }
}
