import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj.get(a).add(b);
            adj.get(b).add(a);
        }
        int[] vis = new int[n + 1];
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (vis[i] != 0) continue;
            cnt++;
            ArrayList<Integer> q = new ArrayList<>();
            q.add(i);
            vis[i] = 1;
            while (!q.isEmpty()) {
                ArrayList<Integer> nq = new ArrayList<>();
                for (int u : q) {
                    for (int v : adj.get(u)) {
                        if (vis[v] == 0) {
                            vis[v] = vis[u] == 1 ? 2 : 1;
                            nq.add(v);
                        } else if (vis[v] == vis[u]) {
                            System.out.println(-1);
                            return;
                        }
                    }
                }
                q = nq;
            }
        }
        System.out.println(cnt - 1);
    }
}
