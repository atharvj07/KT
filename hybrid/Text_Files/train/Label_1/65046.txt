from sys import stdin

n, h, m = map(int, stdin.readline().split())
heights = [h for _ in range(n+1)]

for _ in range(m):
    l, r, x = map(int, stdin.readline().split())
    for i in range(l, r+1):
        heights[i] = min(heights[i], x)

dp = [[0 for _ in range(h+1)] for _ in range(n+1)]

for i in range(1, n+1):
    for j in range(h+1):
        dp[i][j] = max(dp[i-1][j], dp[i-1][j-1] + (min(j, heights[i]))**2)

print(dp[n][h])
