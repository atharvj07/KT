#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  ll sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
  }
  if (sum % 2 == 1) {
    cout << "NO" << endl;
    return 0;
  }
  sum /= 2;
  ll cur = 0;
  for (int i = 0; i < n; i++) {
    cur += a[i];
    if (cur == sum) {
      cout << "YES" << endl;
      return 0;
    }
    if (cur > sum) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
  return 0;
}
