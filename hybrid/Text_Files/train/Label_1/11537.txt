#include <cstdio>

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int a, b, c, ans = 0;
        scanf("%d %d %d", &a, &b, &c);
        while (a > 0 && b > 1) {
            b -= 2;
            a -= 1;
            ans += 3;
        }
        while (b > 0 && c > 1) {
            c -= 2;
            b -= 1;
            ans += 3;
        }
        printf("%d\n", ans);
    }
}
