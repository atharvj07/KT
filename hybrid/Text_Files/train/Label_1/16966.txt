#include <iostream>
#include <vector>
#include <algorithm>

typedef long long ll;

using namespace std;

ll a[110], dp[110][110], p[110][110];

int main() {
    int n, m;
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    cin >> m;
    for (int sum = 1; sum <= n; sum++) {
        for (int l = n; l >= 1; l--) {
            dp[l][sum] = dp[l + 1][sum];
            p[l][sum] = p[l + 1][sum];
            if (sum - a[l] >= 0 && dp[l + 1][sum - a[l]] + a[l] > dp[l][sum]) {
                dp[l][sum] = dp[l + 1][sum - a[l]] + a[l];
                p[l][sum] = l;
            }
        }
    }
    while (m--) {
        int l, r;
        cin >> l >> r;
        cout << a[p[r][l]] << '\n';
    }
    return 0;
}
