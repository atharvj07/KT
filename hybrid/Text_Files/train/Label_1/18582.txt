#include<bits/stdc++.h>
using namespace std;
using ll=long long;
#define pb push_back
#define mp make_pair
const ll mod=1e9+7;
#define rep(i,a,n) for(ll i=a;i<n;i++)
#define per(i,a,n) for(ll i=n-1;i>=a;i--)
#define all(x) begin(x),end(x)
bool comp(const pair<string,string> &a,const pair<string,string> &b){
    if(a.first==b.first)return a.second<b.second;
    return a.first<b.first;
}
int main() {
    ll n;
    cin>>n;
    vector<pair<string,string>>a(n);
    map<string,ll>m;
    rep(i,0,n){
        string f,s;cin>>f>>s;a[i]={f,s};
    }
    vector<ll>v(n);
    rep(i,0,n){cin>>v[i];}
    sort(all(a),comp);
    for(const auto &i:a){
        m[i.first]+=1;
        m[i.second]+=1;
    }
    sort(all(v));
    for(const auto &i:v){
        if(m[a[i-1].first]>1 and m[a[i-1].second]==1){
            m[a[i-1].first]--;
            m[a[i-1].second]++;
        }
        else if(m[a[i-1].second]>1 and m[a[i-1].first]==1){
            m[a[i-1].second]--;
            m[a[i-1].first]++;
        }
        if(a[i-1].first!=a[i-1].second){cout<<"NO";return 0;}
    }
    cout<<"YES";
    return 0;
}
