import numpy as np
n, k, m = map(int, input().split())
a = np.array(list(map(int, input().split())))

a = np.mod(a, m)
a.sort()
b = [a[0]]
ans = True
while len(b) < k:
    if len(b) == 1 and a[-1] == a[0]:
        ans = False
        break
    l = 0
    r = len(a) - 1
    need = (m - a[r] + a[0]) % m
    pos = -1
    while l <= r:
        mid = (l + r) // 2
        if a[mid] >= need:
            pos = mid
            r = mid - 1
        else:
            l = mid + 1
    if pos != -1:
        b.append(a[pos])
        a = np.delete(a, pos)
    else:
        ans = False
        break

if ans:
    print("Yes")
    print(*b)
else:
    print("No")
