#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
int main() {
	string s;
	cin >> s;
	int k;
	cin >> k;
	int n = s.size();
	int cnt = 0;
	ll ans = 0;
	for (int i = 0; i < n; i++) {
		if (s[i] == '0') cnt = 0;
		else {
			cnt++;
			if (cnt == k) ans++;
		}
	}
	cout << ans << endl;
	return 0;
}
