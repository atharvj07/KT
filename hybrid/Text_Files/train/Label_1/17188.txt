#include <algorithm>
#include <cassert>
#include <iostream>
#include <vector>
struct edge {int from, to;};
int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::vector<int>> adj(n);
  std::vector<bool> printed(n, false);
  for (int i = 0; i < m; ++i) {
    int a, b;
    std::cin >> a >> b;
    a--;
    b--;
    adj[a].push_back(b);
  }
  std::vector<int> ans;
  for (int i = 0; i < n; ++i) {
    if (printed[i]) continue;
    std::vector<int> cur;
    for (int j = i;; j = adj[j].front()) {
      cur.push_back(j);
      printed[j] = true;
      if (j == i) break;
      adj[j].erase(std::remove(adj[j].begin(), adj[j].end(), i), adj[j].end());
      i = adj[j].back();
    }
    std::reverse(cur.begin(), cur.end());
    ans.insert(ans.end(), cur.begin(), cur.end());
  }
  bool ok = true;
  for (int i = 0; i < n; ++i) ok &= printed[i];
  if (ok)
    for (int i : ans) std::cout << i + 1 << ' ';
  else
    std::cout << -1;
}
