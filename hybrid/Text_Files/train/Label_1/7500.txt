#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int,int> pi;
int main(){
    int n;
    cin>>n;
    string s,t;
    cin>>s>>t;
    vector<pi> v;
    for(int i=0;i<n;){
        if(s[i]==t[i]) i++;
        else{
            int last=-1;
            for(int j=i+1;j<n;j++){
                if(s[j]==s[i]){
                    last=j;
                }
            }
            if(last==-1){
                cout<<-1;
                return 0;
            }
            v.push_back({last,i});
            swap(s[i],t[last]);
            i++;
        }
    }
    cout<<v.size()<<endl;
    for(pi pp:v){
        cout<<pp.first+1<<" "<<pp.second+1<<endl;
    }
}
