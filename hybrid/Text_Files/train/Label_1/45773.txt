import java.io.*;
import java.util.*;

public class Main {

    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));

    public static void main(String[] args) throws IOException {
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        long t = Long.parseLong(st.nextToken());
        String s = br.readLine();
        st = new StringTokenizer(br.readLine());
        long[] tArr = new long[k + 1];
        for (int i = 0; i < k; i++) {
            tArr[i + 1] = Long.parseLong(st.nextToken());
        }
        long[][] a = new long[k + 1][k + 1];
        for (int i = 1; i <= k; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 1; j <= k; j++) {
                a[i][j] = Long.parseLong(st.nextToken());
            }
        }
        long[] dp = new long[1 << (k + 1)];
        long[] cur = new long[1 << (k + 1)];
        Arrays.fill(dp, Long.MAX_VALUE);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            int c = s.charAt(i) - 'A' + 1;
            for (int j = 0; j < (1 << (k + 1)); j++) {
                cur[j] = dp[j];
                if ((j & (1 << c)) == 0) {
                    cur[j | (1 << c)] = Math.min(cur[j | (1 << c)], dp[j] + tArr[c]);
                    for (int l = 1; l <= k; l++) {
                        if (l != c && (j & (1 << l)) == 0) {
                            cur[j | (1 << c)] = Math.min(cur[j | (1 << c)], dp[j | (1 << l)] + a[c][l]);
                            cur[j | (1 << c) | (1 << l)] = Math.min(cur[j | (1 << c) | (1 << l)], dp[j | (1 << l)] + a[c][l] + tArr[c]);
                        }
                    }
                }
            }
            dp = cur;
        }
        long ans = 0;
        for (long l : dp) {
            if (l <= t) {
                ans++;
            }
        }
        bw.write(ans + "\n");
        bw.flush();
        bw.close();
    }
}
