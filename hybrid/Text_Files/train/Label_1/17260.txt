#include <cstdio>
typedef long long LL;
LL qmi(LL base, LL exp, LL mod) {
  LL res=1;
  base=base%mod;
  while(exp>0) {
    if(exp&1) { res=res*base; res%=mod; }
    exp>>=1;
    base=base*base; base%=mod;
  }
  res%=mod;
  return res;
}
LL count(LL n, LL b) {
  LL l=0, r=n/5;
  while(l<=r) {
    LL mid=(l+r)>>1;
    if(qmi(b,mid,1000000007)<n) l=mid+1;
    else r=mid-1;
  }
  return l;
}
int main() {
  LL n, m;
  scanf("%lld%lld", &n, &m);
  printf("%lld\n", count(n,m));
}
