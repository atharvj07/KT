t = int(input())
for _ in range(t):
    n = int(input())
    grid = [list(input()) for _ in range(n)]
    counts = [0, 0, 0]
    for i in range(n):
        for j in range(n):
            if grid[i][j] in ['X', 'O']:
                counts[(i + j) % 3] += 1
    min_count = min(counts)
    change = counts.index(min_count)
    for i in range(n):
        for j in range(n):
            if (i + j) % 3 == change and grid[i][j] == 'X':
                grid[i][j] = 'O'
    for row in grid:
        print(''.join(row))
