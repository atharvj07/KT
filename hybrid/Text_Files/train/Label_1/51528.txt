import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int ans = 0;
            for (int i = 0; i < n; i++) {
                char c = s.charAt(i);
                if (c == '-' || c == s.charAt((i + 1) % n)) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
