import sys

n = int(input())
b = list(map(int, input().split()))

a = [-1] * n
for i in range(n - 1, -1, -1):
    for j in range(1, max(b[i], 1) + 1):
        if j % b[i] == b[(i + 1) % n]:
            a[i] = j
            break
    if a[i] == -1:
        print("NO")
        sys.exit(0)

print("YES")
print(*a)
