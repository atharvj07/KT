#include <iostream>
#include <string>

using namespace std;

int main() {
  long long L, A;
  cin >> L >> A;
  string s = to_string(A + 1);
  long long N = s.length() / L;
  if (s.length() % L != 0) N++;
  string t = "";
  for (long long i = 0; i < N; i++) t += s.substr(0, L);
  cout << t << endl;
  return 0;
}
