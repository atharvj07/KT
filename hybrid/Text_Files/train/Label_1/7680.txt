#include <iostream>
#include <vector>
#include <algorithm>
typedef long long ll;
using namespace std;

int main() {
    int t;
    cin >> t;
    for (int i = 0; i < t; i++) {
        ll a, b, c, d;
        cin >> a >> b >> c >> d;
        ll ans = a * c - (c * (c - 1) / 2) * b;
        if (ans > 0) cout << ans << endl;
        else cout << -1 << endl;
    }
    return 0;
}
