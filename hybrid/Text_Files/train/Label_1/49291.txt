import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String level = input.next();
        input.close();

        char[] segments = level.toCharArray();
        boolean isGood = false;
        for (int i = 2; i < n - 2; i++) {
            if (segments[i] == '*' && segments[i - 1] == '*' && segments[i - 2] == '*' && segments[i + 1] == '*' && segments[i + 2] == '*') {
                isGood = true;
                break;
            }
        }

        System.out.println(isGood ? "yes" : "no");
    }
}
