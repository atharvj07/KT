#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 2e5 + 10;

ll n, k, a[N];

int main() {
    int t; cin >> t;
    while (t--) {
        cin >> n >> k;
        ll sum = 0;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            sum += a[i];
        }
        // apply all the available pourings
        sort(a + 1, a + n + 1);
        ll diff = 0;
        for (int i = n; i >= n - k; i--) {
            diff += a[i];
        }
        // maximize the difference
        ll mn = a[n - k];
        for (int i = 1; i <= k; i++) {
            if (a[i] < mn) {
                diff += mn - a[i];
            }
        }
        cout << diff << '\n';
    }
    return 0;
}
