#include<iostream>
#include<vector>
#include<sstream>
using namespace std;

int main(){
    long long n,k,a,b,x,y; cin>>n;
    vector<long long> va,vb;
    vector<pair<long long,long long>> act;

    for(long long i=0;i<n;i++){
        cin>>a;
        va.push_back(a);
    }
    cin>>k;
    for(long long i=0;i<k;i++){
        cin>>b;
        vb.push_back(b);
    }
    if(n==k){
        for(long long i=0;i<n;i++){
            if(va[i]!=vb[i]) goto end;
        }
        cout<<"YES";
        return 0;
    }
    end:
    for(long long i=0;i<n;i++){
        for(long long j=i+1;j<n;j++){
            if(va[i]>va[j]){
                act.push_back(make_pair(i,va[i]));
                va.erase(va.begin()+j);
                va[i]+=act.back().second;
                act.back().second=vb[i]-act.back().second;
                if(va[i]==vb[i]) goto done;
                j=i;
            }
            if(j==n-1){
                if(va[i]==vb[i]) goto done;
                else{
                    cout<<"NO";
                    return 0;
                }
            }
        }
    }
    done:
    for(long long i=0;i<k;i++){
        if(va[i]!=vb[i]){
            cout<<"NO";
            return 0;
        }
    }
    cout<<"YES"<<endl;
    for(long long i=0;i<act.size();i++){
        if(act[i].second>vb[act[i].first]) act[i].second=vb[act[i].first];
        if(vb[act[i].first]==act[i].second){
            x=vb[act[i].first]-act[i].second;
            y=act[i].first-n+k-1;
            x>y?cout<<act[i].first+1<<" R":cout<<act[i].first+1<<" L";
        }
    }
    return 0;
}
