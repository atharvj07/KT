#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x = a[i];
    while (x % 2 == 0) { x /= 2; }
    for (int j = i + 1; j < n; j++) {
      int y = a[j];
      while (y % 2 == 0) { y /= 2; }
      if (x == y) { ans++; }
    }
  }
  cout << ans << endl;
  return 0;
}
