n = int(input())
friends = [list(map(int, input().split())) for _ in range(n)]

max_height = max([max(w, h) for w, h in friends])
min_area = float('inf')

for max_h in range(1, max_height + 1):
    total_width = 0
    valid = True
    for w, h in friends:
        if h > max_h and w > max_h:
            valid = False
            break
        elif h <= max_h and (w <= max_h or h >= w):
            total_width += w
        else:
            total_width += h
    if valid:
        min_area = min(min_area, total_width * max_h)

print(min_area)
