#define ll long long
#define INF 1000000007
using namespace std;

int MOD = 1000000007;
int main() {
    ll t, a, b;
    cin >> t >> a >> b;
    if (t == 1) {
        cout << "inf" << endl;
        return 0;
    }
    if (a == 0) {
        if (b == 0) {
            cout << "1" << endl;
            return 0;
        } else {
            cout << "0" << endl;
            return 0;
        }
    }
    ll cur = b % MOD;
    ll inv = 1;
    ll exp = t - 2;
    while (exp) {
        if (exp % 2 == 1) {
            inv = (inv * t) % MOD;
        }
        t = (t * t) % MOD;
        exp /= 2;
    }
    inv = (inv % MOD + MOD) % MOD;
    cur = (cur * inv) % MOD;
    ll ans = (a * cur) % MOD;
    cout << ans << endl;
    return 0;
}
