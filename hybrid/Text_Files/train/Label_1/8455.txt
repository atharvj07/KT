#include <bits/stdc++.h>

#define ll long long
using namespace std;

struct edge {
  ll w, u, v;
  bool operator<(const edge& o) const {
    return w < o.w;
  }
};

const ll inf = 1ll << 62;
const int maxn = 1e5 + 5;

typedef pair<ll, int> pli;
vector<pli> g[maxn];

ll prim(int n) {
  vector<int> vis(n + 1, 0);
  vector<ll> dis(n + 1, inf);
  ll sum = 0;
  dis[1] = 0;
  for (int i = 1; i <= n; i++) {
    ll v = -1;
    for (int j = 1; j <= n; j++) {
      if (!vis[j] && (v == -1 || dis[j] < dis[v])) {
        v = j;
      }
    }
    if (v == -1) {
      break;
    }
    sum += dis[v];
    vis[v] = 1;
    for (auto u : g[v]) {
      int to = u.second;
      ll w = u.first;
      if (!vis[to] && w < dis[to]) {
        dis[to] = w;
      }
    }
  }
  return sum;
}

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    g[u].push_back({w, v});
    g[v].push_back({w, u});
  }
  ll ans = prim(n);
  cout << ans << "\n";
  return 0;
}
