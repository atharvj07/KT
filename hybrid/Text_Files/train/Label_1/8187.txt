#include <bits/stdc++.h>
#define ll long long
#define pb push_back
using namespace std;
const int MAXN = 2e5 + 5;
vector<int> adj[MAXN];
int depth[MAXN], anc[MAXN][20], lg[MAXN];
int ans = 0;
void dfs(int node, int d, int p) {
    depth[node] = d;
    anc[node][0] = p;
    lg[node] = lg[p] + 1;
    for (int u : adj[node]) {
        if (u != p) dfs(u, d + 1, node);
    }
}
void preprocess(int n) {
    dfs(1, 0, 1);
    for (int j = 1; j < 20; j++) {
        for (int i = 1; i <= n; i++) { anc[i][j] = anc[anc[i][j - 1]][j - 1]; }
    }
}
int lca(int a, int b) {
    if (depth[a] < depth[b]) swap(a, b);
    int diff = depth[a] - depth[b];
    for (int i = 19; i >= 0; i--) {
        if ((1 << i) & diff) { a = anc[a][i]; }
    }
    if (a == b) return a;
    for (int i = 19; i >= 0; i--) {
        if (anc[a][i] != anc[b][i]) {
            a = anc[a][i];
            b = anc[b][i];
        }
    }
    return anc[a][0];
}
int dist(int a, int b) { return depth[a] + depth[b] - 2 * depth[lca(a, b)]; }
void solve(int node, int p) {
    ans += dist(node, p);
    for (int u : adj[node]) {
        if (u != p) solve(u, node);
    }
}
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].pb(b);
        adj[b].pb(a);
    }
    preprocess(n);
    solve(1, 1);
    cout << ans << '\n';
    return 0;
}
