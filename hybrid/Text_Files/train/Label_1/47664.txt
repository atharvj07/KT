import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] dp = new int[n + 1];
        for (int i = 2; i <= n; i++) {
            int max = 0;
            for (int j = 2; j * j <= i; j++) {
                if (i % j == 0) {
                    max = Math.max(max, dp[i / j] + j);
                    max = Math.max(max, dp[j] + i / j);
                }
            }
            dp[i] = max;
        }
        int ans = 0;
        for (int i = 2; i <= n; i++) {
            ans = Math.max(ans, dp[i]);
        }
        System.out.println(ans);
    }
}
