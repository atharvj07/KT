a, b = map(int, input().split())

def solve(a, b):
    operations = []
    while b > a:
        operations.append(b)
        if b % 10 == 1:
            b //= 10
        elif b % 2 == 0:
            b //= 2
        else:
            break
    if b == a:
        operations.append(a)
        operations.reverse()
        print("YES")
        print(len(operations))
        print(' '.join(map(str, operations)))
    else:
        print("NO")

solve(a, b)
