#include<iostream>
#include<vector>
using namespace std;
int main()
{
    long long n,a,sum=0,mx=0;
    cin>>n;
    vector<long long> x(n+1,0);
    for(long long i=1;i<=n;i++){cin>>a;x[i]=x[i-1]+a;mx+=a;}
    for(long long i=1;i<=n;i++)
    {
        for(long long j=i;j<=n;j++)
        {
            sum=max(sum,mx-(x[j]-x[i-1]));
        }
    }
    cout<<sum<<endl;
    return 0;
}
