#include <cstring>
#include <iostream>
#include <memory.h>
#include <vector>
using namespace std;
typedef long long ll;
const int MAXN = 200005;
ll a[MAXN], f[MAXN][2];
int n;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  memset(f, -1, sizeof f);
  f[0][0] = f[0][1] = a[0];
  f[1][0] = max(a[0], a[1]);
  f[1][1] = f[0][0] + a[1];
  for (int i = 2; i < n; i++) {
    f[i][0] = max(f[i - 1][0], f[i - 2][1] + a[i]);
    f[i][1] = max(f[i - 1][1], f[i - 2][0] + a[i]);
  }
  cout << f[n - 1][0] << "\n";
  return 0;
}
