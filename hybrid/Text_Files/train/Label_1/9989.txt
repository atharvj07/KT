#include <iostream>

using namespace std;

int main() {
  double y1, y2, yw, xb, yb, r;
  cin >> y1 >> y2 >> yw >> xb >> yb >> r;
  double d1 = yw - yb;
  double d2 = y2 - y1;
  double k = -d1 / d2;
  double b = yw - k * xb;
  if (k < 0 || k > 1 || b < 0)
    cout << "-1";
  else
    cout << (y1 - b) / (k - 1);
  return 0;
}
