import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int k = Integer.parseInt(line1[1]);
        String[] line2 = reader.readLine().split(" ");
        double[] p = new double[n];
        for (int i = 0; i < n; i++) {
            p[i] = Double.parseDouble(line2[i]);
        }

        Map<Integer, Double> cache = new LinkedHashMap<>();
        Map<Integer, Integer> lastAccess = new HashMap<>();
        for (int i = 0; i < 10100; i++) {
            double r = Math.random();
            int video = 0;
            while (r > 0) {
                r -= p[video];
                video++;
            }
            video--;
            if (cache.size() == k && !cache.containsKey(video)) {
                int lru = Integer.MAX_VALUE;
                for (int key : cache.keySet()) {
                    if (lastAccess.get(key) < lru) {
                        lru = lastAccess.get(key);
                    }
                }
                cache.remove(lru);
                lastAccess.remove(lru);
            }
            cache.put(video, p[video]);
            lastAccess.put(video, i);
        }

        for (int key : cache.keySet()) {
            System.out.printf("%.6f ", cache.get(key));
        }
        System.out.println();
    }
}
