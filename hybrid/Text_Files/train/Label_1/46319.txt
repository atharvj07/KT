import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        List<long[]>[] adj = new List[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            long c = sc.nextInt();
            adj[u].add(new long[]{v, c});
            adj[v].add(new long[]{u, c});
        }
        long[] mx = new long[n];
        dfs(0, -1, mx, w, adj);
        System.out.println(mx[0]);
    }

    public static void dfs(int u, int p, long[] mx, long[] w, List<long[]>[] adj) {
        for (long[] v : adj[u]) {
            if (v[0] != p) {
                dfs((int) v[0], u, mx, w, adj);
                mx[u] = Math.max(mx[u], mx[(int) v[0]] + w[(int) v[0]] - v[1]);
            }
        }
    }
}
