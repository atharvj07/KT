import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] abcd = new int[4];
        for (int i = 0; i < 4; i++) abcd[i] = sc.nextInt();

        if (k > n - 1) {
            System.out.println(-1);
            return;
        }

        Map<Integer, Integer> deg = new HashMap<>();
        for (int i = 0; i < 4; i++) deg.put(abcd[i], 0);

        List<int[]> ans = new ArrayList<>();

        int[] path1 = new int[n];
        int[] path2 = new int[n];
        path1[0] = abcd[0];
        path2[0] = abcd[2];

        for (int i = 0; i < n - 1; i++) {
            int cur1 = path1[i], cur2 = path2[i];
            if (cur1 != abcd[1] && deg.getOrDefault(cur1, 0) < k) {
                deg.put(cur1, deg.getOrDefault(cur1, 0) + 1);
                deg.put(abcd[1], deg.getOrDefault(abcd[1], 0) + 1);
                ans.add(new int[]{cur1, abcd[1]});
                path1[i + 1] = abcd[1];
            } else if (cur2 != abcd[3] && deg.getOrDefault(cur2, 0) < k) {
                deg.put(cur2, deg.getOrDefault(cur2, 0) + 1);
                deg.put(abcd[3], deg.getOrDefault(abcd[3], 0) + 1);
                ans.add(new int[]{cur2, abcd[3]});
                path2[i + 1] = abcd[3];
            } else {
                System.out.println(-1);
                return;
            }
        }

        path1[n - 1] = abcd[1];
        path2[n - 1] = abcd[3];
        for (int i = 0; i < n; i++) System.out.print(path1[i] + " ");
        System.out.println();
        for (int i = 0; i < n; i++) System.out.print(path2[i] + " ");
    }
}
