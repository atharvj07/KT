#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m, k, ord[100010], b[100010];
int main() {
    cin >> n >> m >> k;
    for (ll i = 1; i <= n; i++) {
        ll x; cin >> x; ord[x] = i;
    }
    ll ans = 0, last = 1;
    for (ll i = 1; i <= m; i++) {
        ll x; cin >> x;
        ll now = ord[x];
        ans += (now - 1) / k;
        if (now % k == 0) now = k;
        if (now > last) ans += now - last;
        else if (now < last) ans += k - last + now;
        last = now;
    }
    cout << ans << '\n';
    return 0;
}
