#include <iostream>

using namespace std;

bool classy(long long x) {
  int cnt = 0;
  while (x) {
    if (x % 10) cnt++;
    if (cnt > 3) return false;
    x /= 10;
  }
  return true;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long l, r;
    cin >> l >> r;
    long long ans = 0;
    for (long long x = l; x <= r; x++) {
      if (classy(x)) ans++;
    }
    cout << ans << '\n';
  }
  return 0;
}
