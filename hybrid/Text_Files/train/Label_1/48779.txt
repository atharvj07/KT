import java.util.*;

public class Main {
    static int[] materials, priors;
    static List<Integer>[] pos = new List[1001];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        materials = new int[n];
        priors = new int[n];
        for (int i = 0; i < n; i++) {
            String material = sc.next();
            int price = sc.nextInt();
            materials[i] = i;
            priors[i] = price;
            pos[i] = new ArrayList<>();
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            String from = sc.next(), to = sc.next();
            pos[from.charAt(0) - 'a'].add(to.charAt(0) - 'a');
            materials[from.charAt(0) - 'a'] = -1;
        }

        for (int i = 0; i < n; i++) {
            if (materials[i] == -1) continue;
            int min = priors[i];
            Queue<Integer> q = new LinkedList<>();
            q.offer(i);
            while (!q.isEmpty()) {
                int mat = q.poll();
                int prior = priors[mat];
                for (int next : pos[mat]) {
                    if (min > prior + priors[next]) {
                        min = prior + priors[next];
                        materials[next] = materials[mat];
                        priors[next] = min;
                        q.offer(next);
                    }
                }
            }
        }

        int ans = 0;
        for (int mat : materials) {
            if (mat != -1) ans += priors[mat];
        }
        System.out.println(ans);
    }
}
