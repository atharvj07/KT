#include <bits/stdc++.h>

using namespace std;

typedef unsigned int uint32_t;

const int MAXN = 200010;
int N;
typedef pair<pair<uint32_t, int>, bool> ipblock;
ipblock blocks[MAXN];

map<uint32_t, int> blkm;

bool compare(const ipblock &a, const ipblock &b) {
    return a.first.second > b.first.second;
}

int main() {
    cin >> N;
    for (int i = 0; i < N; i++) {
        char op;
        cin >> op;
        uint32_t ip = 0;
        int mask = 0;
        int pos = 32;
        if (op == '-') {
            blocks[i].second = true;
            char c;
            while ((c = getchar()) != '/') { ip = ip * 256 + (c - '0'); }
            cin >> mask;
            while (pos >= mask) { ip >>= 1; pos--; }
        } else {
            blocks[i].second = false;
            char c;
            while ((c = getchar()) != '/') { ip = ip * 256 + (c - '0'); }
            cin >> mask;
            while (pos >= mask) { ip >>= 1; pos--; }
            ip &= (1 << pos) - 1;
        }
        blocks[i].first = make_pair(ip, pos);
    }

    sort(blocks, blocks + N, compare);

    int cnt = 0;
    for (int i = 0; i < N; i++) {
        if (blkm.count(blocks[i].first.first)) {
            if (blkm[blocks[i].first.first] == blocks[i].second) { continue; }
            cnt = -1;
            break;
        }
        if (!blocks[i].second) { blkm[blocks[i].first.first] = false; }
    }

    if (cnt == -1) { cout << -1 << endl; } else {
        set<pair<uint32_t, int>> s;
        for (int i = 0; i < N; i++) {
            if (blkm.count(blocks[i].first.first) || blocks[i].second) {
                s.insert(make_pair(blocks[i].first.first, blocks[i].first.second));
            }
        }
        cout << s.size() << endl;
        for (auto i : s) {
            if (i.second == 32) { cout << i.first << endl; } else {
                cout << (i.first << i.second) << '/' << i.second << endl;
            }
        }
    }
    return 0;
}
