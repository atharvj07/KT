import sys

board = [input() for _ in range(int(input()))]
free = [list(line) for line in board]

pieces = [
    [(0, 0), (0, 1), (0, 2), (1, 1), (2, 1)],
    [(0, 0), (1, 0), (1, 1), (1, 2), (2, 1)],
    [(0, 0), (1, 0), (1, 1), (1, 2), (2, 1)],
    [(0, 0), (0, 1), (0, 2), (1, 1), (1, 2)],
    [(0, 0), (0, 1), (0, 2), (1, 1), (2, 1)],
]

def fits(x, y, p):
    for dx, dy in p:
        if x + dx < 0 or x + dx >= len(board) or y + dy < 0 or y + dy >= len(board):
            return False
        if free[x + dx][y + dy] == '#':
            return False
    return True

for i in range(len(board)):
    for j in range(len(board)):
        if free[i][j] == '.':
            for p in pieces:
                if fits(i, j, p):
                    for dx, dy in p:
                        free[i + dx][j + dy] = '#'
                    break
            else:
                sys.stdout.write('NO')
                sys.exit()

sys.stdout.write('YES')
