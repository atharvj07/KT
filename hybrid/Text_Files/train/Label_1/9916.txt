#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int odd = 0;
    for (int i = 0; i < n; i++) odd += a[i] % 2;
    cout << (odd > 0 ? "YES" : "NO") << endl;
  }
  return 0;
}
