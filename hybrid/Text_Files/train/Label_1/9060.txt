#include <bits/stdc++.h>
using namespace std;
const int N=1e5+5;
int main() {
    int n,t=0; cin>>n;
    vector<int>a(N,-1);
    for (int i=2;i<=n;i++) {
        if (~a[i]) continue;
        for (int j=i;j<=n;j+=i) {
            if (~a[j]) {
                a[j]=-a[j];
                if (a[j]>t) t=a[j];
                continue;
            }
            a[j]=++t;
        }
    }
    for (int i=2;i<=n;i++) cout<<a[i]-t+1<<" ";
    return 0;
}
