#include <bits/stdc++.h>
#define EPS 1e-9
using namespace std;
typedef long long Int;
const Int INF = 1LL<<60;
typedef pair<Int,Int> pi;
vector<pi> pt;
double cross(pi a, pi b) {return a.first*b.second - a.second*b.first;}
double area() {
    double sum = 0;
    for(int i=0;i<pt.size();i++) {
        int j = (i + 1) % pt.size();
        sum += cross(pt[i],pt[j]);
    }
    return abs(sum) / 2.0;
}
int main() {
    int n;
    double ans = 0;
    cin >> n;
    pt.resize(n);
    for(int i=0;i<n;i++) cin >> pt[i].first >> pt[i].second;
    for(int i=0;i<n;i++) {
        int j1 = (i + 1) % n;
        int j2 = (i - 1 + n) % n;
        pi v1 = {pt[i].first-pt[j1].first,pt[i].second-pt[j1].second};
        pi v2 = {pt[i].first-pt[j2].first,pt[i].second-pt[j2].second};
        double theta = acos(cross(v1,v2) / (sqrt(cross(v1,v1) * cross(v2,v2)) + EPS));
        double side1 = sqrt(cross(v1,v1));
        double side2 = sqrt(cross(v2,v2));
        ans = max(ans,side1 * sin(theta / 2) + side2 * sin(theta / 2));
    }
    double totArea = area();
    ans = min(ans,totArea / n);
    cout << fixed << setprecision(10) << ans << endl;
    return 0;
}
