#include <iostream>
#include <string>
using namespace std;

int main() {
    int n;
    string s;
    cin >> n >> s;
    int last0 = n;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '0') {
            last0 = i;
            break;
        }
    }
    int cnt = 0;
    for (int i = 0; i < last0; i++) {
        if (s[i] == '0') cnt++;
    }
    cout << cnt << endl;
    return 0;
}
