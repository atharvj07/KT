#include <bits/stdc++.h>
#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()
using namespace std;

typedef long long ll;
typedef string str;
typedef pair<ll, ll> pll;

const ll mod = 1e9 + 7;

bool check(str s, ll x, ll y) {
    ll cur = 0;
    for (auto &c : s) {
        if (cur % 2 == 0) {
            if (c - '0' != cur % 10) return false;
        } else {
            if (y == 0 && c - '0' > x) return false;
            if (x == 0 && c - '0' != y) return false;
            if (c - '0' == x) cur += y;
            if (c - '0' == y) cur += x;
        }
        cur /= 10;
    }
    return true;
}

int main() {
    str s;
    cin >> s;
    s = "0" + s;
    ll n = s.size();
    vi(vi(ll)) ans(10, vi(ll)(10, -1));
    for (ll i = 0; i < 10; i++) {
        for (ll j = 0; j < 10; j++) {
            ll sum = 0;
            for (ll k = 2; k < n; k++) {
                bool ok = 0;
                ll num = 0;
                for (ll idx = 0; idx < 5; idx++) {
                    if (k + idx >= n) break;
                    num = num * 10 + (s[k + idx] - '0');
                    if (num < j) ok = 1;
                }
                if (!ok) sum++;
            }
            if (check(s, i, j)) ans[i][j] = sum;
        }
    }

    for (auto &v : ans) {
        for (auto &x : v) cout << x << " ";
        cout << endl;
    }
}
