#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Event {
    int start, end, num;
};

int main() {
    int n;
    cin >> n;
    vector<Event> events;
    while (n--) {
        int m, d, p, t;
        cin >> m >> d >> p >> t;
        events.push_back({d + (m - 1) * 30 - t, d + (m - 1) * 30, p});
    }
    sort(events.begin(), events.end(), [](const Event& a, const Event& b) {
        return a.start < b.start;
    });
    int ans = 0, cnt = 0;
    for (auto& event : events) {
        if (event.start >= cnt) {
            cnt = event.end;
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
