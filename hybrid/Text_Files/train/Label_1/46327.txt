import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static class Road {
        int to;
        int len;

        Road(int to, int len) {
            this.to = to;
            this.len = len;
        }
    }

    static List<Road>[] adj;
    static int[] gas;
    static int[] maxGas;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        gas = new int[n + 1];
        maxGas = new int[n + 1];

        String[] split = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            gas[i + 1] = Integer.parseInt(split[i]);
        }

        adj = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) {
            split = br.readLine().split(" ");
            int u = Integer.parseInt(split[0]);
            int v = Integer.parseInt(split[1]);
            int len = Integer.parseInt(split[2]);

            adj[u].add(new Road(v, len));
            adj[v].add(new Road(u, len));
        }

        dfs(1, 1, 0);
        dfs2(1, 1, 0);

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = Math.max(ans, maxGas[i]);
        }
        System.out.println(ans);
    }

    static void dfs(int cur, int pre, int curGas) {
        maxGas[cur] = curGas;
        for (Road road : adj[cur]) {
            if (road.to == pre) continue;
            dfs(road.to, cur, Math.max(curGas, road.len));
        }
    }

    static void dfs2(int cur, int pre, int curGas) {
        maxGas[cur] = Math.max(maxGas[cur], curGas);
        for (Road road : adj[cur]) {
            if (road.to == pre) continue;
            dfs2(road.to, cur, Math.max(curGas, road.len) + gas[road.to]);
        }
    }
}
