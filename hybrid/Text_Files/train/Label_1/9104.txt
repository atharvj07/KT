#include <iostream>
using namespace std;
typedef long long ll;
const int mxn = 1e12;
int main() {
  ll n, k, ans = 0, cer = 0, no = 0;
  cin >> n >> k;
  if (k == 1) {
    ans = 0;
    cer = 0;
    no = n;
  } else if (n % (k + 1) * 2 == 0) {
    ans = n / (k + 1) * 2;
    cer = ans * k;
    no = n - ans - cer;
  } else {
    ll baki = ans % 2;
    ll ans1 = ans / 2;
    ans = ans1 + baki;
    cer = ans * k;
    no = n - ans - cer;
  }
  cout << ans << " " << cer << " " << no << '\n';
}
