t = int(input())
for _ in range(t):
    n, l, r = map(int, input().split())
    colors = list(map(int, input().split()))

    # Count colors for left and right socks
    left_colors = {}
    right_colors = {}
    for i in range(l):
        left_colors[colors[i]] = left_colors.get(colors[i], 0) + 1
    for i in range(l, n):
        right_colors[colors[i]] = right_colors.get(colors[i], 0) + 1

    # Matching process
    # Step 1: Find pairs and remove them
    for color in left_colors.keys():
        if color in right_colors:
            min_pairs = min(left_colors[color], right_colors[color])
            left_colors[color] -= min_pairs
            right_colors[color] -= min_pairs

    # Step 2: Calculate the remaining socks after pairing
    remaining_left = sum(left_colors.values())
    remaining_right = sum(right_colors.values())

    # Step 3: Make pairs by converting the excess side socks to the other side
    # We only need to convert the minimum number of excess socks from the larger side to make the sides even
    conversions_needed = abs(remaining_left - remaining_right) // 2

    # Step 4: Recolors or additional conversions needed
    # After making sides even, the minimum extra steps needed is the number of conversions
    # because we can either recolor or convert the evened out sides' socks to make pairs
    extra_steps = conversions_needed + max(remaining_left, remaining_right) // 2

    print(extra_steps)
