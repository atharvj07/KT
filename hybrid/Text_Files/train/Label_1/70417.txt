import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
x = int(input())

ans = 0

for i in range(n):
    for j in range(m):
        cur = a[i] * b[j]
        if cur <= x:
            ans = max(ans, 1)
        for k in range(i + 1, n):
            for l in range(j + 1, m):
                cur += a[k] * b[l]
                if cur <= x:
                    ans = max(ans, (k - i + 1) * (l - j + 1))
                else:
                    break
            else:
                continue
            break

print(ans)
