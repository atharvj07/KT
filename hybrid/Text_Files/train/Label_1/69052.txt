n,m,q=map(int, input().split())
s=input()
t=input()
st=[]
for i in s:
    st+=i
for i in t:
    st+=i
sp=[0 for x in range(len(st))]
for i in range(1,len(st)):
    if st[i]==st[i-1]:
        sp[i]=sp[i-1]+1
ans=[]
for i in range(q):
    l,r=map(int, input().split())
    l-=1
    r-=1
    ans.append(sp[r]-sp[l])
for i in ans:
    print(i)
