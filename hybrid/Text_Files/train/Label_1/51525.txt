import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            String s = in.next();
            int ret = 0;
            for (int i = 0; i < n; i++) {
                char c = s.charAt(i);
                if (c == '-' || c == s.charAt((i + 1) % n)) {
                    ret++;
                }
            }
            System.out.println(ret);
        }
    }
}
