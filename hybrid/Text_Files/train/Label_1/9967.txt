#include <iostream>
#include <algorithm>
using namespace std;

string v[8];

int main() {
    for (int i = 0; i < 8; i++) {
        cin >> v[i];
    }
    int flag = v[0][0] == 'W';
    for (int i = 0; i < 8; i++) {
        int odd = 0;
        int even = 0;
        for (int j = 0; j < 8; j++) {
            if ((j % 2 == 0 && v[i][j] == 'W') || (j % 2 == 1 && v[i][j] == 'B')) {
                odd++;
            }
            if ((j % 2 == 1 && v[i][j] == 'W') || (j % 2 == 0 && v[i][j] == 'B')) {
                even++;
            }
        }
        if (odd != 4 && even != 4) {
            flag = 0;
        }
    }
    if (flag) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
