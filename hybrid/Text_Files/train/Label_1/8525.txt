#include<bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 10;
struct node
{
    long long val;
    int ty;
    bool operator<(const node& a) { return val > a.val; }
}p[maxn];
int main() {
    int a, b, c, m;
    cin >> a >> b >> c >> m;
    for (int i = 1; i <= m; i++)
        cin >> p[i].val >> p[i].ty;
    sort(p + 1, p + m + 1);
    int ans = 0, sum = 0, cnt[3] = {0};
    for (int i = 1; i <= m; i++)
    {
        if (cnt[0] + cnt[2] < a && p[i].ty == 0) cnt[0]++, ans++, sum += p[i].val;
        else if (cnt[1] + cnt[2] < b && p[i].ty == 1) cnt[1]++, ans++, sum += p[i].val;
        else if (cnt[2] < c) cnt[2]++, ans++, sum += p[i].val;
    }
    cout << ans << ' ' << sum << endl;
    return 0;
}
