from collections import deque

n = int(input())
g = [list(map(int, input())) for i in range(n)]
m = int(input())
p = list(map(int, input().split()))

sd = [set() for i in range(n+1)]
for i in range(1, n+1):
    q = deque()
    q.append((i, [i]))
    while q:
        v, path = q.popleft()
        if v in sd[i]: continue
        sd[i].add(v)
        for j in range(1, n+1):
            if g[v-1][j-1]:
                q.append((j, path+[j]))

s = {}
for i in range(1, n+1):
    path = sd[i]
    for a in path:
        for b in path:
            s[tuple([a, b])] = min(s.get(tuple([a, b]), m), len(p))

for i in range(2, m):
    for j in range(1, m-i+1):
        a, b = p[j-1], p[j+i-1]
        if a == b: continue
        s[tuple([a, b])] = min(s[tuple([a, b])], s[tuple([a, p[j]])] + s[tuple([p[j+i-1], b])]-1)

ans = []
for i in range(2, m):
    if p[i-1] == p[i]: continue
    ans.append(p[i-1])
    if s[tuple([p[i-1], p[i]])] == s[tuple([p[i-2], p[i]])] + s[tuple([p[i-1], p[i+1]])]-1:
        break

ans.insert(0, p[0])
ans.append(p[-1])
print(len(ans))
print(*ans)
