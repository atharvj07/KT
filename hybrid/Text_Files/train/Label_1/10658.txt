#include <bits/stdc++.h>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int odd = 0;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      odd += x % 2;
    }
    if (odd == 0 || odd == n)
      cout << "NO" << endl;
    else
      cout << "YES" << endl;
  }
}
