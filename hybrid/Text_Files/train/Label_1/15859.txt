#include <bits/stdc++.h>
#define int long long int
#define rep(i, a, b) for (int i = a; i <= b; ++i)
#define repr(i, a, b) for (int i = a; i >= b; --i)
#define pb push_back
#define eb emplace_back
#define all(x) x.begin(), x.end()
using namespace std;
signed main() {
  int t;
  cin >> t;
  while (t--) {
    int n, ok = 1;
    cin >> n;
    vector<int> v(n);
    map<int, int> mp;
    for (int &x : v) cin >> x;
    for (int x : v) mp[x]++;
    for (auto p : mp) {
      if (p.first) {
        ok = 0;
        break;
      }
    }
    if (n & 1) {
      if (mp[1] & 1) ok = 0;
    } else {
      if (mp[1]) ok = 0;
    }
    if (!ok) {
      cout << "NO" << '\n';
      continue;
    }
    cout << "YES" << '\n';
    if ((n & 1) && mp[1]) {
      cout << 1 << '\n';
      cout << (n + 1) / 2 << '\n';
    } else {
      cout << 0 << '\n';
    }
  }
  return 0;
}
