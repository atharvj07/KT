grid = [
    list(map(int, input().split())),
    list(map(int, input().split())),
    list(map(int, input().split()))
]

for i in range(3):
    for j in range(3):
        grid[i][j] = (grid[i][j] + grid[i][(j + 1) % 3] + grid[i][(j - 1) % 3] +
                      grid[(i + 1) % 3][j] + grid[(i - 1) % 3][j]) % 2
        print('1' if grid[i][j] else '0', end='')
    print()
