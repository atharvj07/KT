#include <bits/stdc++.h>
#define N 4

using namespace std;

int n, x[N], y[N], t[N], sum;
bool fl;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> x[i];
    sum += x[i];
  }
  y[1] = x[1], y[2] = x[2], y[3] = x[3], y[4] = x[4];
  sort(y + 1, y + n + 1);
  t[1] = y[1], t[2] = y[2], t[3] = y[3], t[4] = y[4];
  sort(t + 1, t + 5);
  for (int i = 1; i <= 3; i++) sum -= t[i];
  int k = 4 - n;
  if (sum % k != 0 || t[1] > sum / k) {
    fl = 0;
  } else {
    fl = 1;
    int p = sum / k;
    for (int i = 1; i <= n; i++) x[i] -= p;
    sort(x + 1, x + n + 1);
    for (int i = n + 1; i <= n + k; i++) x[i] = p;
    sort(x + 1, x + n + k + 1);
    for (int i = 1; i <= n + k; i++) cout << x[i] << endl;
  }
  if (!fl) cout << "NO\n";
  return 0;
}
