import math
import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    days = [list(map(int, input().split())) for _ in range(m)]

    occurrences = [0 for _ in range(n+1)]
    chosen = []

    for day in days:
        friend = None
        for f in day[1:]:
            if occurrences[f] < math.ceil(m/2): 
                friend = f
                break
        if friend: 
            chosen.append(friend)
            occurrences[friend] += 1
    if len(chosen) == m:
        print("YES")
        print(*chosen)
    else:
        print("NO")
