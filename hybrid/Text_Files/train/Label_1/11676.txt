#include <bits/stdc++.h>
using namespace std;

int main()
{
    int a, b, h, w, n;
    cin >> a >> b >> h >> w >> n;
    vector<int> v;
    for (int i = 0; i < n; ++i)
    {
        int x;
        cin >> x;
        v.push_back(x);
    }
    sort(v.begin(), v.end());
    int ans = -1;
    int h1 = h, w1 = w;
    for (int i = 0; i < n; ++i)
    {
        a /= v[i];
        b /= v[i];
        if (a <= 0 || b <= 0)
        {
            ans = i;
            break;
        }
    }
    if (ans == -1)
    {
        while (h1 >= a && w1 >= b)
        {
            ++ans;
            h1 /= v[ans % n];
            w1 /= v[ans % n];
        }
    }
    if (h >= a && w >= b)
    {
        ans = 0;
    }
    cout << ans << endl;
    return 0;
}
