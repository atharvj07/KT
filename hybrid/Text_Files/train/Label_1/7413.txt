#include <bits/stdc++.h>
using namespace std;

struct edge {
  int a, b, w;
  bool operator<(const edge &rhs) const { return w < rhs.w; }
};

const int maxn = 505;

long long sum_v[maxn], sum_e;
int fa[maxn], sz[maxn];

int getf(int x) {
  if (fa[x] == x) return x;
  return fa[x] = getf(fa[x]);
}

void merge(int x, int y) {
  x = getf(x), y = getf(y);
  if (x == y) return;
  if (sz[x] > sz[y]) {
    fa[y] = x;
    sz[x] += sz[y];
    sum_v[x] += sum_v[y];
  } else {
    fa[x] = y;
    sz[y] += sz[x];
    sum_v[y] += sum_v[x];
  }
}

int main() {
  int n, m;
  cin >> n >> m;
  vector<edge> edge_list;
  for (int i = 1; i <= n; i++) {
    cin >> sum_v[i];
    fa[i] = i;
    sz[i] = 1;
  }
  for (int i = 0; i < m; i++) {
    int a, b, w;
    cin >> a >> b >> w;
    edge_list.push_back({a, b, w});
    sum_e += w;
  }
  sort(edge_list.begin(), edge_list.end());
  long double max_dens = 0;
  for (auto &e : edge_list) {
    int fa_x = getf(e.a), fa_y = getf(e.b);
    if (fa_x == fa_y) continue;
    max_dens = max(max_dens,
                   (long double)(sum_v[fa_x] + sum_v[fa_y]) /
                       (long double)(sum_e - e.w));
    merge(fa_x, fa_y);
  }
  cout << fixed << setprecision(15) << max_dens;
  return 0;
}
