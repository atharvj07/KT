for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=[0 for i in range(101)]
    for i in a:
        b[i]+=1
    ans=0
    for i in range(101):
        if b[i]==0:
            ans=i
            break
    if ans==0:
        for i in range(1,101):
            if b[i]==1:
                ans=i
                break
    for i in range(101):
        if b[i]>1:
            ans+=i
    print(ans)
