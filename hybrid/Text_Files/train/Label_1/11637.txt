#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, j;
        cin >> n;
        vector<pair<int, int>> a(n);
        for (auto &[h, w] : a) cin >> h >> w;
        for (int i = 0; i < n; i++) {
            j = -1;
            for (int x = 0; x < n; x++)
                if (x != i && (a[x].first < a[i].first && a[x].second < a[i].second ||
                                a[x].second < a[i].first && a[x].first < a[i].second)) {
                    j = x + 1;
                    break;
                }
            cout << j << ' ';
        }
        cout << '\n';
    }
}
