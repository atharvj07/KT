from collections import deque

def play_war(n, soldier1, soldier2):
    soldier1 = deque(soldier1)
    soldier2 = deque(soldier2)
    fights = 0
    while soldier1 and soldier2:
        fights += 1
        if fights > 1000:  # Assuming game is infinite if it doesn't end in 1000 moves
            return -1
        card1 = soldier1.popleft()
        card2 = soldier2.popleft()
        if card1 > card2:
            soldier1.append(card2)
            soldier1.append(card1)
        else:
            soldier2.append(card1)
            soldier2.append(card2)
    if not soldier1:
        return (fights, 2)
    else:
        return (fights, 1)

n = int(input())
k1, *soldier1 = map(int, input().split())
k2, *soldier2 = map(int, input().split())
result = play_war(n, soldier1, soldier2)
if result == -1:
    print(result)
else:
    print(result[0], result[1])
