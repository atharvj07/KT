import java.util.*;
import java.io.*;

public class Main {
    static long[][] posS, posC;

    public static void main(String[] args) throws IOException {
        InputStreamReader isr = new InputStreamReader(System.in);
        BufferedReader br = new BufferedReader(isr);
        StringTokenizer st = new StringTokenizer(br.readLine());
        int N = Integer.parseInt(st.nextToken()), M = Integer.parseInt(st.nextToken());
        posS = new long[N][2];
        posC = new long[M][2];
        for (int i = 0; i < N; i++) {
            st = new StringTokenizer(br.readLine());
            posS[i][0] = Integer.parseInt(st.nextToken());
            posS[i][1] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < M; i++) {
            st = new StringTokenizer(br.readLine());
            posC[i][0] = Integer.parseInt(st.nextToken());
            posC[i][1] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < N; i++) {
            long min = Long.MAX_VALUE;
            int idx = -1;
            for (int j = 0; j < M; j++) {
                long dist = Math.abs(posS[i][0] - posC[j][0]) + Math.abs(posS[i][1] - posC[j][1]);
                if (dist < min || (dist == min && idx > j)) {
                    min = dist;
                    idx = j;
                }
            }
            System.out.println(idx + 1);
        }
    }
}
