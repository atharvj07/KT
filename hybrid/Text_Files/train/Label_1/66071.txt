n, c = map(int, input().split())
p = list(map(int, input().split()))
t = list(map(int, input().split()))
limaks = sum(max(0, p[i] - c * sum(t[:i])) for i in range(n))
radewooshs = sum(max(0, p[i] - c * sum(t[i+1:])) for i in range(n))
if limaks == radewooshs:
    print("Tie")
elif limaks > radewooshs:
    print("Limak")
else:
    print("Radewoosh")
