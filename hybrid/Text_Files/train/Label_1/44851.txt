import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        int[][] ans = new int[q][10];
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int[] cnt = new int[10];
            char[] seq = s.substring(l, r + 1).toCharArray();
            int cp = 0;
            int dp = 1;
            while (cp >= 0 && cp < seq.length) {
                if (seq[cp] >= '0' && seq[cp] <= '9') {
                    cnt[seq[cp] - '0']++;
                    seq[cp] = (char) (seq[cp] - 1);
                    if (seq[cp] == '0') {
                        seq[cp] = '_';
                        for (int j = cp + dp; j >= 0 && j < seq.length; j += dp) {
                            seq[j] = '_';
                        }
                    }
                    cp += dp;
                } else {
                    dp *= -1;
                    cp += dp;
                    if (cp >= 0 && cp < seq.length && seq[cp] == seq[cp - dp]) {
                        seq[cp - dp] = '_';
                    }
                }
            }
            for (int j = 0; j < 10; j++) {
                ans[i][j] = cnt[j];
            }
        }
        for (int[] a : ans) {
            for (int i : a) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
