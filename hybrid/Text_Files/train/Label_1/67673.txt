def correct_typos(s):
    result = []
    for c in s:
        if len(result) >= 2 and result[-1] == c and result[-2] == c:
            continue
        if len(result) >= 3 and result[-1] == c and result[-2] == result[-3]:
            continue
        result.append(c)
    return ''.join(result)

s = input().strip()
print(correct_typos(s))
