#include <iostream>
#include <vector>

using namespace std;

int main() {
    int c1, c2, c3, c4;
    cin >> c1 >> c2 >> c3 >> c4;
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    vector<int> b(m);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < m; i++) { cin >> b[i]; }
    int ans = 1e9;
    ans = min(ans, c1 * (n + m));
    ans = min(ans, c2 * n + c4);
    ans = min(ans, c2 * m + c4);
    for (int i = 0; i < n; i++) { ans = min(ans, max(a[i], 1) * c1 + c3); }
    for (int i = 0; i < m; i++) { ans = min(ans, max(b[i], 1) * c1 + c3); }
    cout << ans << endl;
    return 0;
}
