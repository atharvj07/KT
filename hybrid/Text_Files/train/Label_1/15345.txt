#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    vector<int >v(n),v1,v2,v3;
    for(int i=0; i<n; i++)
    {
        cin>>v[i];
    }
    int cnt=0;
    for(int i=0; i<n; i++)
    {
        if(v[i]<0)
            cnt++;
    }
    cout<<cnt<<' ';
    for(int i=0; i<n; i++)
    {
        if(v[i]<0)
        {
            v1.push_back(v[i]);
        }
    }
    for(int x:v1)
        cout<<x<<' ';
        cout<<endl;
    int cnt1=0,cnt2=0;
    for(int i=0; i<n; i++)
    {
        if(v[i]==0)
            cnt2++;
        else if(v[i]>0)
            cnt1++;
    }
    cout<<cnt1<<' ';
    for(int i=0; i<n; i++)
    {
        if(v[i]>0)
        {
            v2.push_back(v[i]);
        }
    }
    for(int x:v2)
        cout<<x<<' ';

   for(int i=0; i<n; i++)
