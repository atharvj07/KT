import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] res = new int[n + 1];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || a[i] != a[i - 1]) {
                res[cnt++] = a[i];
            } else {
                res[cnt] = a[i] + 1;
                cnt++;
                i++;
                while (i < n && a[i] == a[i - 1]) {
                    i++;
                }
                i--;
            }
        }
        System.out.println(cnt);
        for (int i = 0; i < cnt; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
