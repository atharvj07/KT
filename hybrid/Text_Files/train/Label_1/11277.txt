#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll MAXN=100005;
const ll inf=100000000000000000;
struct Point{
    ll x,y,dir;
};
Point dancer[MAXN];
int main(){
    ll tc,n,w,h,d,x,y,t,i;
    while(cin>>tc>>w>>h){
        map<pair<ll,ll>,bool>vis;
        while(tc--){
            cin>>d>>x>>y>>t;
            if(d==1){
                dancer[x+y].x=x;
                dancer[x+y].y=y;
                dancer[x+y].dir=1;
                if(vis[{x,y}]==false)vis[{x,y}]=true;
                else{
                    dancer[x+y].dir=0;
                    vis[{dancer[x+y].x,dancer[x+y].y}]=true;
                }
            }
            else{
                dancer[x+y].x=x;
                dancer[x+y].y=y;
                dancer[x+y].dir=2;
                if(vis[{x,y}]==false)vis[{x,y}]=true;
                else{
                    dancer[x+y].dir=0;
                    vis[{dancer[x+y].x,dancer[x+y].y}]=true;
                }
            }
        }
        for(i=1;i<=w+h-2;i++){
            if(dancer[i].dir==0)continue;
            if(dancer[i].dir==1){
                while(dancer[i].y<h&&vis[{dancer[i].x,dancer[i].y+1}]==false){
                    dancer[i].y++;
                }
                if(dancer[i].y==h){
                    cout<<dancer[i].x<<" "<<dancer[i].y<<endl;
                    continue;
                }
                dancer[i].dir=0;
                vis[{dancer[i].x,dancer[i].y}]=false;
                vis[{dancer[i].x,dancer[i].y+1}]=true;
            }
            else{
                while(dancer[i].x<w&&vis[{dancer[i].x+1,dancer[i].y}]==false){
                    dancer[i].x++;
                }
                if(dancer[i].x==w){
                    cout<<dancer[i].x<<" "<<dancer[i].y<<endl;
                    continue;
                }
                dancer[i].dir=0;
                vis[{dancer[i].x,dancer[i].y}]=false;
                vis[{dancer[i].x+1,dancer[i].y}]=true;
            }
            cout<<dancer[i].x<<" "<<dancer[i].y<<endl;
        }
    }
    return 0;
}
