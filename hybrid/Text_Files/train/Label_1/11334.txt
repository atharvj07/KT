#include <bits/stdc++.h>
using namespace std;
const int  MAX = 5010;

int a[MAX][MAX];
int main(){
    int n, m, k;
    cin >> n >> m >> k;
    for(int i = 1; i <= k; i++){
        int t;
        cin >> t;
        int p, c;
        cin >> p >> c;
        if(t == 1){
            for(int j = 1; j <= m; j++){
                a[p][j] = c;
            }
        }
        else{
            for(int j = 1; j <= n; j++){
                a[j][p] = c;
            }
        }
    }
    for(int i = 1; i <= n; i++){
        for(int j = 1; j <= m; j++){
            cout << a[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
