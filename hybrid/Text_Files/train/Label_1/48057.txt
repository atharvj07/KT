import java.io.*;
import java.util.*;

class Main {
    static long[] dpLeft;
    static long[] dpRight;
    static int[] arr;
    static int n, l, r, Ql, Qr;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        n = Integer.parseInt(s[0]);
        l = Integer.parseInt(s[1]);
        r = Integer.parseInt(s[2]);
        Ql = Integer.parseInt(s[3]);
        Qr = Integer.parseInt(s[4]);
        s = br.readLine().split(" ");
        
        arr = new int[n+2];
        dpLeft = new long[n+2];
        dpRight = new long[n+2];
        
        for(int i=0; i<n; i++){
            arr[i+1] = Integer.parseInt(s[i]);
        }
        
        dpLeft[1] = l * arr[1];
        dpRight[n] = r * arr[n];
        
        for(int i=2; i<=n; i++){
            dpLeft[i] = dpLeft[i-1] + l * arr[i] + Ql * l;
        }
        
        for(int i=n-1; i>0; i--){
            dpRight[i] = dpRight[i+1] + r * arr[i] + Qr * r;
        }
        
        long ans = Long.MAX_VALUE;
        
        for(int i=1; i<=n; i++){
            ans = Math.min(ans, dpLeft[i] + dpRight[i+1]);
        }
        
        System.out.println(ans);
    }
}
