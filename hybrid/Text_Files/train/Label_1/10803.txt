#include<bits/stdc++.h>
using namespace std;
struct edge {
    int v, w;
    edge(int _v = 0, int _w = 0) : v(_v), w(_w) {}
};
vector<edge> e[5005];
int f[5005], vis[5005];
queue<int> q;
void bfs(int s, int id) {
    q.push(s); vis[s] = 1; f[s] = id;
    while(!q.empty()) {
        int u = q.front(); q.pop();
        for(edge i : e[u]) {
            if(vis[i.v]) continue;
            q.push(i.v); f[i.v] = id; vis[i.v] = 1;
        }
    }
}
int g[5005], cnt;
void dfs(int u, int fa) {
    g[u] = cnt;
    for(edge i : e[u]) {
        if(i.v == fa) continue;
        if(!g[i.v]) dfs(i.v, u);
    }
}
int32_t main() {
    int n, m; cin >> n >> m;
    for(int i = 1; i <= m; i++) {
        int u, v; cin >> u >> v;
        e[u].emplace_back(v, i);
    }
    for(int i = 1; i <= n; i++) if(!vis[i]) bfs(i, i);
    for(int i = 1; i <= n; i++) if(!g[i]) {
        cnt++;
        dfs(i, i);
    }
    cout << cnt << endl;
    for(int i = 1; i <= m; i++) {
        for(edge j : e[i]) {
            if(f[i] == f[j.v]) cout << g[f[i]] << ' ';
            else cout << g[f[j.v]] << ' ';
        }
    }
}
