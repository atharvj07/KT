import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = -1;
        }
        for (int i = 0; i < n; i++) {
            if (d[i] != -1) continue;
            if (dfs(i, a[i], -1, d, a)) {
                System.out.println("Possible");
                return;
            }
        }
        System.out.println("Impossible");
    }

    public static boolean dfs(int u, int dist, int p, int[] d, int[] a) {
        d[u] = dist;
        for (int v = 0; v < d.length; v++) {
            if (v == u || v == p) continue;
            if (d[v] == -1) {
                if (!dfs(v, dist + 1, u, d, a)) return false;
            } else if (dist + 1 + d[v] != a[u] || dist + 1 + d[u] != a[v]) {
                return false;
            }
        }
        return true;
    }
}
