import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static class Pair {
        int i;
        int j;

        Pair(int i, int j) {
            this.i = i;
            this.j = j;
        }
    }

    static int[][] d;
    static int[][] cost;
    static int[] di = {-1, 1, 0, 0};
    static int[] dj = {0, 0, -1, 1};
    static char[][] S;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int H = sc.nextInt();
        int W = sc.nextInt();
        sc.nextLine();
        S = new char[H][W];
        for (int i = 0; i < H; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < W; j++) {
                S[i][j] = line.charAt(j);
            }
        }

        d = new int[H][W];
        for (int i = 0; i < H; i++) {
            Arrays.fill(d[i], -1);
        }

        int maxCost = 0;
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (S[i][j] == '.') {
                    cost = new int[H][W];
                    for (int ii = 0; ii < H; ii++) {
                        Arrays.fill(cost[ii], -1);
                    }
                    bfs(i, j);
                    maxCost = Math.max(maxCost, getMaxCost());
                }
            }
        }
        System.out.println(maxCost);
    }

    static void bfs(int i, int j) {
        Queue<Pair> que = new LinkedList<>();
        que.add(new Pair(i, j));
        cost[i][j] = 0;

        while (!que.isEmpty()) {
            Pair p = que.poll();
            for (int k = 0; k < 4; k++) {
                int ni = p.i + di[k];
                int nj = p.j + dj[k];
                if (ni >= 0 && ni < cost.length && nj >= 0 && nj < cost[0].length && cost[ni][nj] == -1 && S[ni][nj] == '.') {
                    cost[ni][nj] = cost[p.i][p.j] + 1;
                    que.add(new Pair(ni, nj));
                }
            }
        }
    }

    static int getMaxCost() {
        int maxCost = 0;
        for (int i = 0; i < cost.length; i++) {
            for (int j = 0; j < cost[0].length; j++) {
                if (S[i][j] == '.') {
                    maxCost = Math.max(maxCost, cost[i][j]);
                }
            }
        }
        return maxCost;
    }
}
