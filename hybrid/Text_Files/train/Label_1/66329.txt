import sys, collections
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    g = collections.defaultdict(list)
    for i, ai in enumerate(a, 1):
        g[ai].append(i)
    vis = [False] * (n + 1)
    ans = []
    q = [1]
    vis[1] = True
    while q:
        v = q.pop(0)
        for u in g[a[v - 1]]:
            if vis[u]: continue
            vis[u] = True
            ans += [v, u]
            q += [u]
    if len(ans) == n - 1: print('YES'); print(*ans, sep='\n')
    else: print('NO')
