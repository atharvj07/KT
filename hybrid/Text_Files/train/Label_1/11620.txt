#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, A, B, C, T;
  cin >> n >> A >> B >> C >> T;
  vector<int> t(n);
  for (int i = 0; i < n; i++) cin >> t[i];
  sort(t.begin(), t.end());
  long long ans = 0, cur = 0, unread = 0;
  for (int i = 0; i < T + 1; i++) {
    while (unread < n && t[unread] <= i) {
      ans += A;
      cur += A;
      unread++;
    }
    ans += min(cur, max(0ll, unread - 1) * C);
    cur = max(0ll, cur - B);
  }
  cout << ans << endl;
}
