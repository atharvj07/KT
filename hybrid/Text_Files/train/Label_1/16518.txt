#include<bits/stdc++.h>
using namespace std;
#define pii pair<int, int>
const int N = 1e5 + 10;
const int inf = 1e9;
int a[N], b[N], c[N], stud[N];
pii p[N];
int main() {
    int n, m, s; cin >> n >> m >> s;
    for(int i = 1; i <= m; i++) cin >> a[i];
    for(int i = 1; i <= n; i++) cin >> b[i];
    for(int i = 1; i <= n; i++) cin >> c[i];
    for(int i = 1; i <= n; i++) p[i] = {c[i], i};
    sort(p + 1, p + 1 + n);
    for(int i = 1; i <= m; i++) {
        int x = -1, mn = inf;
        for(int j = 1; j <= n; j++) {
            if(b[j] >= a[i] && c[j] < mn) {
                mn = c[j];
                x = j;
            }
        }
        if(x == -1) {
            cout << "NO" << endl;
            return 0;
        }
        stud[i] = x;
        s -= c[x];
        if(s < 0) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    for(int i = 1; i <= m; i++) cout << stud[i] << " ";
    cout << endl;
    return 0;
}
