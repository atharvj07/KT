#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> v(n);
    long long s1 = 0, s2 = 0;
    for (int i = 0; i < n; i++) {
        int s;
        cin >> s;
        v[i].resize(s);
        for (int j = s - 1; j >= 0; j--) {
            cin >> v[i][j];
            if (s % 2 == 0)
                s2 += v[i][j];
            else
                s1 += v[i][j];
        }
    }
    cout << s1 << " " << s2 << endl;
    return 0;
}
