w, h = map(int, input().split())

a = [input() for _ in range(h)]

b = [['.' for _ in range(2*w)] for _ in range(2*h)]

for i in range(h):
    for j in range(w):
        b[i*2][j*2] = a[i][j]

for i in range(h):
    for j in range(w):
        b[i*2+1][(w-1-j)*2] = a[i][j]

for i in range(2*h):
    print(''.join(b[i]))
