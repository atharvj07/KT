#include<bits/stdc++.h>
using namespace std;
int main()
{
    int x;
    cin>>x;
    for(;;x++)
    {
        if((x%400==0)||((x%4==0)&&(x%100!=0)))
        {
            if((x+2)%7==1)
            {
                cout<<x+2<<endl;
                break;
            }
        }
        else if((x+1)%7==1)
        {
            cout<<x+1<<endl;
            break;
        }
    }
    return 0;
}
