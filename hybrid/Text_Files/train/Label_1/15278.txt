#include <iostream>
#include <vector>
using namespace std;

struct Wave {
    int l, r, a;
};

int main() {
    int n, k;
    cin >> n >> k;
    vector<Wave> waves(n);
    for (Wave &w : waves) {
        cin >> w.l >> w.r >> w.a;
    }

    long long ans = 0;
    long long bullets = k;
    long long last_end = 0;
    for (Wave w : waves) {
        ans += (w.l - last_end + k - 1) / k * k;
        bullets = max(bullets - (w.l - last_end), 0LL);
        bullets += w.a;
        last_end = w.r;
    }
    if (bullets < 0) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    return 0;
}
