h, w = map(int, input().split())
grid = [input() for _ in range(h)]

# Precompute prefix sums for horizontal and vertical placements
horizontal = [[0] * (w + 1) for _ in range(h + 1)]
vertical = [[0] * (w + 1) for _ in range(h + 1)]

for i in range(1, h + 1):
    for j in range(1, w + 1):
        horizontal[i][j] = horizontal[i][j - 1]
        vertical[i][j] = vertical[i - 1][j]
        if grid[i - 1][j - 1] == '.':
            if j < w and grid[i - 1][j] == '.':
                horizontal[i][j] += 1
            if i < h and grid[i][j - 1] == '.':
                vertical[i][j] += 1

# Precompute 2D prefix sums
for i in range(1, h + 1):
    for j in range(1, w + 1):
        horizontal[i][j] += horizontal[i - 1][j]
        vertical[i][j] += vertical[i][j - 1]

for i in range(1, h + 1):
    for j in range(1, w + 1):
        horizontal[i][j] += horizontal[i][j - 1]
        vertical[i][j] += vertical[i - 1][j]

q = int(input())
for _ in range(q):
    r1, c1, r2, c2 = map(int, input().split())
    h_ways = horizontal[r2][c2] - horizontal[r2][c1 - 1] - horizontal[r1 - 1][c2] + horizontal[r1 - 1][c1 - 1]
    v_ways = vertical[r2][c2] - vertical[r1 - 1][c2] - vertical[r2][c1 - 1] + vertical[r1 - 1][c1 - 1]
    print(h_ways + v_ways)
