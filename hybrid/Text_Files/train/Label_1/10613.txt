#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define pb push_back

const ll N = 2e5 + 5;
const ll mod = 1e9 + 7;
const ll INF = 1e12;

int a[N];
int main() {
  ll n, k;
  cin >> n >> k;
  for (ll i = 1; i <= n; i++) { cin >> a[i]; }
  ll ans = 0, pre = 0, suf = 0;
  for (ll i = n; i >= 1; i--) {
    suf += (a[i] ^ pre);
    if (suf != 0) { ans = i; }
    pre ^= a[i];
  }
  cout << ans << "\n";
}
