import sys
input = sys.stdin.readline

n = int(input())
board = []
for i in range(n):
    board.append(list(input().strip()))

# Check if it's possible to place a piece at (x, y)
def check(x, y):
    if x < 0 or x >= n or y < 0 or y >= n or board[x][y] == '#':
        return False
    for i in range(5):
        for j in range(5):
            if i == 2 and j == 2:
                continue
            if x+i < 0 or x+i >= n or y+j < 0 or y+j >= n or board[x+i][y+j] == '#':
                return False
    return True

# Flood fill to fill the board with pieces
def flood_fill(x, y):
    if not check(x, y):
        return False
    for i in range(5):
        for j in range(5):
            if i == 2 and j == 2:
                continue
            board[x+i][y+j] = '#'
    return True

# Check if the board can be tiled
def can_tile():
    for x in range(n):
        for y in range(n):
            if board[x][y] == '.':
                if flood_fill(x, y):
                    return True
    return False

if can_tile():
    print("YES")
else:
    print("NO")
