import sys

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split())) * n
    ans = 1
    lis = [0] * n
    lis[0] = 1
    for i in range(1, n):
        lis[i] = max([lis[j] for j in range(i) if a[j] < a[i]] or [0]) + 1
        ans = max(ans, lis[i])
    print(ans)
