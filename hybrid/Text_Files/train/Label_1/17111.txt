#include <iostream>

int main() {
  unsigned long long n, m, q;
  std::cin >> n >> m >> q;
  for (unsigned long long i = 0; i < q; ++i) {
    unsigned long long s_x, s_y, e_x, e_y;
    std::cin >> s_x >> s_y >> e_x >> e_y;
    if (s_x != e_x) {
      std::cout << "NO" << std::endl;
    } else {
      if (s_x == 1 and e_x == 1) {
        if (s_y > e_y) { std::cout << "NO" << std::endl; } else { std::cout << "YES" << std::endl; }
      } else {
        if (s_y < e_y) { std::cout << "NO" << std::endl; } else { std::cout << "YES" << std::endl; }
      }
    }
  }
  return 0;
}
