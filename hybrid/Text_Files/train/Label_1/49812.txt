import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        int time = -1;
        for (int i = 0; i < 1000000; i++) {
            int rickTime = b + a * i;
            int mortyTime = d + c * i;
            if (rickTime == mortyTime) {
                time = rickTime;
                break;
            }
        }
        System.out.println(time);
    }
}
