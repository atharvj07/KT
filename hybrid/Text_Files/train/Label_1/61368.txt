import sys
input = sys.stdin.readline
n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
ans1 = 0
ans2 = 0
for j in range(m):
    mn = n + 1
    found = False
    for i in range(n):
        if a[i][j] != 0:
            mn = i
            found = True
            break
    if not found:
        continue
    ans1 += min(k, n - mn + 1)
    for i in range(mn, min(mn + k, n)):
        if a[i][j] == 0:
            a[i][j] = 1
            ans2 += 1
print(ans1, ans2)
