#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;
const int INF = 1e9;

struct Edge {
    int v, w;
    Edge(int v, int w) : v(v), w(w) {}
};

vector<Edge> g[N];
int dist[N];

void bfs(int u) {
    memset(dist, -1, sizeof dist);
    queue<int> q;
    q.push(u);
    dist[u] = 0;
    while (!q.empty()) {
        u = q.front();
        q.pop();
        for (auto e : g[u]) {
            int v = e.v, w = e.w;
            if (dist[v] == -1) {
                q.push(v);
                dist[v] = dist[u] + w;
            }
        }
    }
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(Edge(v, 1));
        g[v].push_back(Edge(u, 1));
    }
    bfs(1);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > 2) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
