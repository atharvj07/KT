#include <bits/stdc++.h>
#define int long long
using namespace std;
const int MOD = 1e9 + 7;
const int MAXN = 1005;
vector<int>adj[MAXN];
int sz[MAXN];
vector<int>p;
vector<pair<int,int>>e;
void dfs(int node,int par){
    p.push_back(node);
    for(auto u:adj[node]){
        if(u==par)continue;
        dfs(u,node);
    }
}
int32_t main(){
    int n,m;
    cin>>n>>m;
    for(int i=1;i<=m;i++){
        int u,v;cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        e.push_back({u,v});
    }
    if(m==n-1){
        cout<<0<<endl;
    }
    else{
        dfs(1,0);
        for(int i=1;i<p.size();i++){
            int u=p[i],v=p[i-1];
            if(adj[u].size()==1)adj[v].erase(find(adj[v].begin(),adj[v].end(),u));
            else adj[u].erase(find(adj[u].begin(),adj[u].end(),v));
        }
        int ans=0,id=-1;
        vector<int>ansv;
        for(auto u:e){
            int s=adj[u.first].size()+adj[u.second].size()-2;
            if(s>ans){
                ans=s;
                id=u.first;
            }
        }
        for(auto u:adj[id])ansv.push_back(u);
        sort(ansv.begin(),ansv.end());
        cout<<ans<<endl;
        for(auto u:ansv)cout<<id<<' '<<u<<endl;
    }
    return 0;
}
