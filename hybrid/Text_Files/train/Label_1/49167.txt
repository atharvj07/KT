import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int m = in.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }
        int max = 0, cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int g = gcd(a[i], b[j]);
                if (a[i] / g == b[j] / g && a[i] / g > max) {
                    max = a[i] / g;
                    cnt = 1;
                } else if (a[i] / g == b[j] / g && a[i] / g == max) {
                    cnt++;
                }
            }
        }
        System.out.println(cnt);
    }

    public static int gcd(int a, int b) {
        while (b != 0) {
            int t = a;
            a = b;
            b = t % b;
        }
        return a;
    }
}
