import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        // Input
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt(), w = sc.nextInt();
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        // Solve
        int[][] c = new int[h][w];
        int[][] cnt = new int[h][w];
        int[][] dir = {{0, 1}, {1, 0}}; // right, down
        for (int i = 0, x = 0, y = 0; i < h * w; i++) {
            c[x][y] = i % n + 1;
            cnt[x][y]++;
            if (cnt[x][y] == a[i % n]) {
                for (int[] d : dir) {
                    int nx = x + d[0], ny = y + d[1];
                    if (0 <= nx && nx < h && 0 <= ny && ny < w) {
                        x = nx;
                        y = ny;
                        break;
                    }
                }
            }
        }
        // Output
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                System.out.print(c[i][j] + " ");
            }
            System.out.println();
        }
        sc.close();
    }
}
