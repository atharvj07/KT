#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        cin>>s;
        string ans;
        int k=1;
        for(int j=2; j<=n; j++){
            string temp=s;
            for(int i=1; i<=n-j+1; i++){
                reverse(temp.begin()+(i-1), temp.begin()+i+j-1);
            }
            if(temp<ans || ans.length()==0) {
                ans=temp;
                k=j;
            }
        }
        cout<<ans<<"\n"<<k<<"\n";
    }
}
