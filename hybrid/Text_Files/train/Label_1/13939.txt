#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, l, r, ansy = 1000000000, ansk = 0;
    vector<pair<int, int>> interval;
    scanf("%d", &n);
    while (n--) {
        scanf("%d%d", &l, &r);
        interval.push_back({l, 1});
        interval.push_back({r, -1});
    }
    int cnt = 0;
    sort(interval.begin(), interval.end());
    for (auto i : interval) {
        cnt += i.second;
        if (cnt > ansk) {
            ansk = cnt;
            ansy = i.first;
        }
    }
    printf("%d %d", ansy, ansk);
}
