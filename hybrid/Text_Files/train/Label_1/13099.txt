#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    double sum = s;
    bool flag = false;
    for (int i = 0; i < n; i++) {
        int x, y, k;
        cin >> x >> y >> k;
        sum += k;
        double dist = sqrt(x * x + y * y);
        if (dist < 1e-6) flag = true;
    }
    if (sum < 1e6 || flag) cout << -1;
    else {
        double l = 0, r = 1e10;
        for (int iter = 0; iter < 100; iter++) {
            double mid = (l + r) / 2;
            double need = 1e6 - sum;
            double covered = 0;
            for (int i = 0; i < n; i++) {
                int x, y, k;
                cin >> x >> y >> k;
                double dist = sqrt(x * x + y * y);
                if (dist <= mid) covered += k;
            }
            if (covered < need) l = mid;
            else r = mid;
        }
        cout << l << endl;
    }
    return 0;
}
