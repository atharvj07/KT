import sys

n, m = map(int, input().split())
mx1, mx2 = 0, 0
a = []
for i in range(n):
    si, ai, bi = map(int, input().split())
    a.append([(ai, bi), si])
    mx1 = max(mx1, ai)
    mx2 = max(mx2, bi)
a.sort(key = lambda x: x[0][0] - x[0][1])
ans = 0
n1 = n2 = p = 0
for i in range(n):
    ai, bi, si = a[i][0][0], a[i][0][1], a[i][1]
    mx = max(ai, bi)
    if p < si:
        ans += (si - p) * (mx1 + mx2)
        p = si
    if ai >= bi:
        if n1 < si:
            ans += (si - n1) * (mx - bi)
            n1 = si
    else:
        if n2 < si:
            ans += (si - n2) * (mx - ai)
            n2 = si
print(ans)
