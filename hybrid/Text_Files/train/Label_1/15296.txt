#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 2e5 + 5;
int32_t main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end(), greater<int>());
    int ans = (m + n - 1) / n;
    if (ans * n < m) {
        ans++;
    }
    if (ans * n < m) {
        cout << -1 << '\n';
        return 0;
    }
    cout << ans << '\n';
    return 0;
}
