#include <iostream>

using namespace std;

int main() {
  long long n, x, y;
  cin >> n >> x >> y;
  for (long long i = 0; i < n; i++) {
    long long a;
    cin >> a;
    long double v = 1.0 * a / x;
    long double vo = 1.0 * a / y;
    if (v < vo) {
      cout << "Vanya" << endl;
    } else if (v > vo) {
      cout << "Vova" << endl;
    } else {
      cout << "Both" << endl;
    }
  }
  return 0;
}
