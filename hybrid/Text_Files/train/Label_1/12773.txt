#include <cstdio>
int main() {
  int T, n;
  scanf("%d", &T);
  while (T--) {
    scanf("%d", &n);
    if (!(n % 3))
      printf("%d %d %d", n / 3, 0, 0);
    else if (!(n % 5))
      printf("0 %d %d", 0, n / 5, 0);
    else if (!(n % 7))
      printf("0 %d %d", 0, 0, n / 7);
    else {
      int x = (n / 3) * 3, y = (n / 5) * 5, z = (n / 7) * 7;
      if (n == x)
        printf("%d %d %d", n / 3, 0, 0);
      else if (n == y)
        printf("0 %d %d", 0, n / 5, 0);
      else if (n == z)
        printf("0 %d %d", 0, 0, n / 7);
      else if (n < x + 2)
        printf("%d %d %d", n / 3, 0, 0);
      else if (n < y + 4)
        printf("0 %d %d", 0, n / 5, 0);
      else if (n < z + 6)
        printf("0 %d %d", 0, 0, n / 7);
      else
        printf("-1");
    }
    printf("\n");
  }
  return 0;
}
