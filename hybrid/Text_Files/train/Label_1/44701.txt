import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        int C = sc.nextInt();
        if(A >= C) System.out.println(1);
        else {
            if(B == 0) System.out.println(Math.ceil((double)(C - A) / A));
            else {
                int i = 1;
                int count = A * i;
                while(count < C) {
                    count += B;
                    i++;
                }
                System.out.println(i);
            }
        }
    }
}
