#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> sugar(n);
    int ans = -1;
    for (int i = 0; i < n; i++) {
        cin >> sugar[i].first >> sugar[i].second;
        if (sugar[i].first <= s && sugar[i].second > ans) {
            ans = sugar[i].second;
        }
    }
    cout << ans << endl;
    return 0;
}
