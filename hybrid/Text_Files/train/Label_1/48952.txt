import java.util.Arrays;
import java.util.HashMap;
import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
  private static HashMap<Integer, Integer> friendMap = new HashMap<>();
  private static PriorityQueue<Integer> onlineFriendQueue = new PriorityQueue<>((a, b) -> friendMap.get(b) - friendMap.get(a));

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);

    int n = scanner.nextInt();
    int k = scanner.nextInt();
    int q = scanner.nextInt();

    int[] friends = new int[n];
    for (int i = 0; i < n; i++) {
      friends[i] = scanner.nextInt();
    }

    for (int i = 0; i < n; i++) {
      friendMap.put(i + 1, friends[i]);
    }

    while (q-- > 0) {
      int type = scanner.nextInt();
      int id = scanner.nextInt();

      if (type == 1) {
        handleOnlineFriend(id);
      } else if (type == 2) {
        checkOnlineFriend(id);
      }
    }
  }

  private static void handleOnlineFriend(int id) {
    onlineFriendQueue.add(id);
  }

  private static void checkOnlineFriend(int id) {
    while (!onlineFriendQueue.isEmpty()) {
      if (onlineFriendQueue.peek() == id) {
        System.out.println("YES");
        return;
      } else {
        onlineFriendQueue.poll();
      }
    }
    System.out.println("NO");
  }
}
