#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a1, k;
    cin >> a1 >> k;
    while (k--) {
      string s = to_string(a1);
      int mn = *min_element(s.begin(), s.end()) - '0';
      int mx = *max_element(s.begin(), s.end()) - '0';
      a1 += mn * mx;
    }
    cout << a1 << endl;
  }
}
