#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k,a,b;
    cin>>n>>k>>a>>b;
    if(a*k+(a+b)%k<n){
        cout<<"NO";
        return 0;
    }
    string s="";
    for(int i=0;i<k;++i){
        for(int j=0;j<a;++j){
            s+='G';
        }
        if(b){
            s+='B';
            --b;
        }
    }
    int len=s.size();
    if(len<n){
        int rep=n-len;
        int times=rep/k;
        string p="";
        for(int i=0;i<k;++i){
            for(int j=0;j<a;++j){
                p+='G';
            }
            if(times){
                p+='B';
                --times;
            }
        }
        s+=p;
    }
    cout<<s;
}
