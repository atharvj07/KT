#include<bits/stdc++.h>
using namespace std;
#define ll long long
int ca[200010],cb[200010],a[200010],idx[200010];
int n,t,k,ansl1,ansl2;
int main(){
    while(cin >> t){
        while(t--){
            cin >> n;
            set<int>st;
            for(int i = 1;i <= n;i++){
                if(i == n)cin >> a[i];
                if(i < n)cin >> a[i],st.insert(a[i]),idx[a[i]] = i;
            }
            int l = 0,r = 0;
            for(int i = 1;i <= n;i++){
                if(i <= idx[i])l++;
                else if(i > idx[i])r++;
            }
            if(st.size() == n){
                cout << 1 << endl;
                cout << l << " " << n - l << endl;
            }
            else cout << 0 << endl;
        }
    }
}
