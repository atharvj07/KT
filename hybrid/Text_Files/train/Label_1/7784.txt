#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<ll> d(2 * n);
    for (int i = 0; i < 2 * n; ++i) { cin >> d[i]; }
    ll sum = 0;
    for (int i = 0; i < 2 * n; ++i) { sum += d[i]; }
    bool ok = (sum % (2 * n) == 0);
    if (!ok) {
      cout << "NO" << endl;
      continue;
    }
    ll avg = sum / (2 * n);
    ok = true;
    for (int i = 0; i < 2 * n; ++i) {
      if (avg - d[i] < 0 || (avg - d[i]) % 2 != 0) {
        ok = false;
        break;
      }
    }
    if (ok) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
