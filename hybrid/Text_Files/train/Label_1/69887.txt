n, f = map(int, input().split())
dp = [[None] * (f + 1) for i in range(n + 1)]
arr = []
for i in range(n):
    a, b = map(int, input().split())
    arr.append((a, b))
dp[n][0] = 0
for i in range(n - 1, -1, -1):
    for j in range(f + 1):
        dp[i][j] = max(dp[i + 1][j], dp[i + 1][j - 1] + min(arr[i][0] * (2 ** (j > 0)), arr[i][1]))
print(dp[0][f])
