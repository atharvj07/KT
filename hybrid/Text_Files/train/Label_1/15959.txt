#include <iostream>
#include <algorithm>
#include <vector>

int main() {
    long long n, c;
    std::cin >> n >> c;
    std::vector<long long> a(n);
    for (long long i = 0; i < n; i++)
        std::cin >> a[i];

    std::sort(a.rbegin(), a.rend());

    long long ans = 0;
    long long block_sum = 0;
    long long block_size = 0;

    for (long long i = 0; i < n; i++) {
        ans += a[i];
        block_sum += a[i];
        block_size++;

        if (block_size % c == 0) {
            ans -= block_sum;
            block_sum = 0;
            block_size = 0;
        }
    }

    std::cout << ans << std::endl;
    return 0;
}
