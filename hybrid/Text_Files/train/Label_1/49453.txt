import java.util.Scanner;

public class Main {
    static char[][] board;
    static int n, m;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        int q = in.nextInt();
        board = new char[2 * n + 1][2 * m + 1];
        for (int i = 1; i <= 2 * n; i++) {
            for (int j = 1; j <= 2 * m; j++) {
                board[i][j] = (i + j) % 2 == 0 ? 'w' : 'b';
            }
        }
        StringBuilder sb = new StringBuilder();
        while (q-- > 0) {
            int i = in.nextInt();
            int j = in.nextInt();
            board[i][j] = board[i][j] == 'w' ? 'b' : 'w';
            sb.append(check() ? "YES\n" : "NO\n");
        }
        System.out.print(sb.toString());
    }

    static boolean check() {
        boolean[][] used = new boolean[2 * n + 1][2 * m + 1];
        for (int i = 1; i <= 2 * n; i++) {
            for (int j = 1; j <= 2 * m; j++) {
                if (board[i][j] == 'w' && !used[i][j]) {
                    if (!dfs(i, j, used)) {
                        return false;
                    }
                }
            }
        }
        return true;
    }

    static boolean dfs(int i, int j, boolean[][] used) {
        if (i < 1 || i > 2 * n || j < 1 || j > 2 * m || board[i][j] == 'b' || used[i][j]) {
            return true;
        }
        used[i][j] = true;
        return dfs(i + 1, j, used) && dfs(i - 1, j, used) && dfs(i, j + 1, used) && dfs(i, j - 1, used) && dfs(i + 1, j + 1, used) && dfs(i - 1, j - 1, used) && dfs(i + 1, j - 1, used) && dfs(i - 1, j + 1, used);
    }
}
