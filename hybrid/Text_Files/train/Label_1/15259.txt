#include <bits/stdc++.h>
using namespace std;
vector<int> a;
bool f(int n, const vector<int>&b, const vector<int>&c) {
    if (n == 1) return true;
    vector<int>nb, nc;
    int NN = (n + 1) / 2;
    int ab = 0, ac = 0;
    for (int i = 0, p = NN - 1; i < NN; i++, p--) {
        if (b[i] == b[p]) { nb.push_back(b[i]); }
        else if (b[i] > b[p]) { nb.push_back(b[i]); ab = 1; }
        else { nb.push_back(b[p]); ab = 1; }
        if (c[i] == c[p]) { nc.push_back(c[i]); }
        else if (c[i] > c[p]) { nc.push_back(c[i]); ac = 1; }
        else { nc.push_back(c[p]); ac = 1; }
    }
    int x = ab ? *max_element(nc.begin(), nc.end()) : 0;
    int y = ac ? *max_element(nb.begin(), nb.end()) : 0;
    if (x != y) return false;
    a.push_back(x);
    return f(NN, nb, nc);
}
int main() {
    int n;
    cin >> n;
    vector<int>b(n - 1), c(n - 1);
    for (int i = 0; i < n - 1; i++) cin >> b[i];
    for (int i = 0; i < n - 1; i++) cin >> c[i];
    if (!f(n - 1, b, c)) { cout << -1; return 0; }
    reverse(a.begin(), a.end());
    for (int i = 1; i < a.size(); i++) cout << a[i] << ' ';
}
