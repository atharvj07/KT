import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] requirements = new int[k][3];
        for (int i = 0; i < k; i++) {
            requirements[i][0] = scanner.nextInt() - 1;
            char sign = scanner.next().charAt(0);
            requirements[i][1] = sign == '=' ? 0 : sign == '<' ? -1 : 1;
            requirements[i][2] = scanner.nextInt() - 1;
        }

        int[] heights = new int[2 * n];
        for (int i = 0; i < n; i++) {
            heights[i] = i + 1;
            heights[2 * n - i - 1] = i + 1;
        }
        long mod = (long) 1e9 + 7;
        long[][] dp = new long[1 << k][2 * n];
        dp[0][0] = 1;
        for (int mask = 0; mask < 1 << k; mask++) {
            for (int i = 0; i < 2 * n; i++) {
                if (dp[mask][i] == 0) continue;
                for (int j = i + 1; j < 2 * n; j++) {
                    boolean canSwap = true;
                    for (int[] requirement : requirements) {
                        if (((mask >> requirement[0]) & 1) != 0) continue;
                        int sign = requirement[1];
                        int a = heights[i];
                        int b = heights[j];
                        if (sign == 0 && a != b) {
                            canSwap = false;
                            break;
                        }
                        if (sign == -1 && a > b) {
                            canSwap = false;
                            break;
                        }
                        if (sign == 1 && a < b) {
                            canSwap = false;
                            break;
                        }
                    }
                    if (!canSwap) continue;
                    int newMask = mask;
                    for (int[] requirement : requirements) {
                        if (requirement[0] == i && requirement[2] == j) {
                            newMask |= 1 << requirement[0];
                        }
                    }
                    dp[newMask][j] = (dp[newMask][j] + dp[mask][i]) % mod;
                }
            }
        }

        System.out.println(dp[(1 << k) - 1][2 * n - 1]);
    }
}
