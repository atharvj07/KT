n = int(input())
s = input()

operations = 0
result = []

for i in range(0, n, 2):
    a_count = s[i:i+2].count('a')
    b_count = s[i:i+2].count('b')
    if a_count != b_count:
        operations += 1
        if a_count > b_count:
            result.append('ab')
        else:
            result.append('ba')
    else:
        result.append(s[i:i+2])

print(operations)
print(''.join(result))
