#include <bits/stdc++.h>
using namespace std;

int main() {
  unordered_set<char> seen;
  int need = 0;
  string s;
  cin >> s;
  for (int i = 0; i < s.length(); i++) {
    char c = tolower(s[i]);
    if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
      if (seen.count(c) == 0) {
        seen.insert(c);
        if ((s[i - 1] - '0') % 2) need++;
      }
    }
  }
  cout << need << endl;
}
