#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int mn = 2e9, mx = -2e9;
    for (int i = 0; i < n - 1; i++) {
        mn = min(mn, a[i + 1] - a[i]);
        mx = max(mx, a[i + 1] - a[i]);
    }
    bool f = false;
    for (int i = 0; i < n; i++) {
        for (long long d = mn; d <= mx; d++) {
            vector<int> b = a;
            b.erase(b.begin() + i);
            bool ok = true;
            for (int j = 0; j + 1 < b.size(); j++) {
                if (b[j + 1] - b[j] != d) ok = false;
            }
            if (ok) {
                cout << i + 1 << endl;
                return 0;
            }
        }
    }
    cout << -1 << endl;
    return 0;
}
