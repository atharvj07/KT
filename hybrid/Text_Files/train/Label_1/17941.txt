#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n / 2; i++) {
      for (int j = i + 1; j < n; j++) {
        if ((a[i] % a[j]) == 0 || (a[j] % a[i]) == 0) continue;
        cout << a[i] << " " << a[j] << "\n";
      }
    }
  }
  return 0;
}
