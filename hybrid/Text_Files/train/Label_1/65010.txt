import sys
from collections import defaultdict

n, m, k = map(int, input().split())
p = list(map(int, input().split()))
s = list(map(int, input().split()))
c = list(map(int, input().split()))

# create a dictionary of school to student with maximum power
school_max = {}
for i in range(n):
    if s[i] not in school_max or p[i] > school_max[s[i]][0]:
        school_max[s[i]] = [p[i], i]

# sort the schools by maximum power of students
schools = sorted(school_max.keys(), key=lambda x: school_max[x][0], reverse=True)

# count the number of chosen ones that are not the maximum power in their school
cnt = 0
for i in c:
    if p[i - 1] != school_max[s[i - 1]][0]:
        cnt += 1

# output the minimum number of schools to be made up
print(max(0, cnt - (m - k)))
