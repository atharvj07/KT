n = int(input())
arr = list(map(int, input().split()))
m = min(arr); M = max(arr)
if(2*m<=M):
    print(0)
else:
    arr.sort()
    l = 0; r = n-1
    ans = n
    while(l<=r):
        mid = (l+r)//2;
        if(2*arr[mid]<=M):
            ans = r-mid;
            r = mid-1
        else: l = mid+1
    print(ans)
