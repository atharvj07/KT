#include<bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<string> v(n);
    for (auto &x : v) cin >> x;
    set<int> del;
    for (int i = 0; i < m; ++i) {
        int x;
        cin >> x;
        del.insert(x - 1);
    }
    vector<char> vv(v[0].size(), '?');
    for (int j = 0; j < v[0].size(); ++j) {
        bool ok = false;
        for (int i = 0; i < n; ++i)
            if (!del.count(i) && v[i][j] != vv[j]) {
                vv[j] = '?';
                ok = true;
                break;
            }
        if (!ok)
            for (int i = 0; i < n; ++i)
                vv[j] = max(vv[j], v[i][j]);
    }
    for (int i = 0; i < n; ++i) {
        bool ok = true;
        for (int j = 0; j < v[0].size(); ++j)
            if (vv[j] != '?' && vv[j] != v[i][j]) {
                ok = false;
                break;
            }
        if (!ok) {
            cout << "No\n";
            return 0;
        }
    }
    cout << "Yes\n";
    for (auto x : vv) cout << x;
    cout << "\n";
}
