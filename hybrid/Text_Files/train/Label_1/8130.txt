#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> points(n);
  for (int i = 0; i < n; i++) { cin >> points[i]; }
  sort(points.begin(), points.end());
  cout << (unique(points.begin(), points.end()) == points.end() ? "No" : "Yes")
       << endl;
  return 0;
}
