h1, a1, c1 = map(int, input().split())
h2, a2 = map(int, input().split())

actions = []

while h2 > 0:
    if h2 - a1 <= 0:
        actions.append("STRIKE")
        break
    if h1 - a2 > 0 or c1 > a2:
        actions.append("STRIKE")
        h1 -= a2
        h2 -= a1
    else:
        actions.append("HEAL")
        h1 += c1 - a2
print(len(actions))
for action in actions:
    print(action)
