import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] t = new int[200005];
        int[] s = new int[200005];
        char[] tc = new char[200005];
        char[] sc = new char[200005];
        int n = in.nextInt(), m = in.nextInt();
        for (int i = 0; i < n; i++) {
            String[] a = in.next().split("-");
            t[i] = Integer.parseInt(a[0]);
            tc[i] = a[1].charAt(0);
        }
        for (int i = 0; i < m; i++) {
            String[] a = in.next().split("-");
            s[i] = Integer.parseInt(a[0]);
            sc[i] = a[1].charAt(0);
        }
        long ans = 0, pp = 0, pp2 = 0;
        int j = 0, k = 0;
        for (int i = 0; i < n; i++) {
            if (k < m && s[k] <= t[i]) {
                while (k < m && s[k] <= t[i]) {
                    if (s[k] == t[i] && sc[k] == tc[i]) {
                        if (pp2 == pp) {
                            ans += t[i] - s[k] + 1;
                        } else {
                            ans += pp - pp2 + 1;
                        }
                        pp2 = pp;
                    }
                    pp += t[i];
                    k++;
                }
            } else {
                pp2 = pp;
            }
            pp += t[i];
        }
        System.out.println(ans);
    }
}
