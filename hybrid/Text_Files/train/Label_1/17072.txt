#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e6 + 5;
int n, a[MAXN], sum, cntOdd;
bool visited[MAXN];

int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    sum += a[i];
    cntOdd += (a[i] & 1);
  }
  if (cntOdd == 1) {
    cout << "NO";
    return 0;
  }
  cout << (sum & 1 ? "YES" : "NO");
  return 0;
}
