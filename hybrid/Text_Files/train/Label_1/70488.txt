n = input().strip()
i = 0
magic = True
while i < len(n):
    if n[i] == '1':
        if i + 1 < len(n) and n[i+1] == '4':
            if i + 2 < len(n) and n[i+2] == '4':
                i += 3
            else:
                i += 2
        else:
            i += 1
    else:
        magic = False
        break
if magic:
    print("YES")
else:
    print("NO")
