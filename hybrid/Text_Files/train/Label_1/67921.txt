import sys
input = sys.stdin.readline

n = int(input())
events = []
for i in range(n):
    l, r = map(int, input().split())
    events.append((l, 1))
    events.append((r, -1))
events.sort()

watched = 0
for e in events:
    t, d = e
    watched += d
    if watched > 2:
        print("NO")
        exit()
print("YES")
