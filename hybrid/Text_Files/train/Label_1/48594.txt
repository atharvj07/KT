import java.util.Scanner;

public class Jigsaw {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int h = scanner.nextInt();
        int[][] pieces = new int[n][4];
        for (int i = 0; i < n; i++) {
            pieces[i][0] = scanner.nextInt(); // A_i
            pieces[i][1] = scanner.nextInt(); // B_i
            pieces[i][2] = scanner.nextInt(); // C_i
            pieces[i][3] = scanner.nextInt(); // D_i
        }

        boolean possible = true;
        int maxHeight = 0;
        for (int i = 0; i < n; i++) {
            maxHeight = Math.max(maxHeight, pieces[i][0] + pieces[i][1] + pieces[i][2] + pieces[i][3]);
        }

        if (maxHeight > h) {
            possible = false;
        } else {
            int[] heights = new int[h + 1];
            for (int i = 0; i < n; i++) {
                heights[pieces[i][2]]++;
                heights[pieces[i][2] + pieces[i][0]]--;
                heights[pieces[i][2] + pieces[i][0] + pieces[i][1]]++;
                heights[pieces[i][2] + pieces[i][0] + pieces[i][1] + pieces[i][3]]--;
            }

            for (int i = 1; i <= h; i++) {
                heights[i] += heights[i - 1];
                if (heights[i] > 1) {
                    possible = false;
                    break;
                }
            }
        }

        System.out.println(possible ? "YES" : "NO");
    }
}
