#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, a, sum = 0;
  cin >> n >> a;
  vector<int> v(n);
  for (auto &i : v) cin >> i;
  sort(v.begin(), v.end());
  for (int i = 0; i < n - 1; i++) sum += abs(v[i] - a);
  a = v[n - 1];
  for (int i = n - 2; i >= 0; i--) sum += abs(v[i] - a);
  cout << sum << endl;
}
