import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long sum = 0;
        long maxSum = 0;
        for (int i = 0; i < n; i++) {
            long shift = (long) i * a[i];
            sum += shift;
            maxSum = Math.max(maxSum, sum);
        }
        for (int i = n - 1; i >= 0; i--) {
            long shift = (long) (n - 1 - i) * a[i];
            sum += shift;
            maxSum = Math.max(maxSum, sum);
        }
        System.out.println(maxSum);
    }
}
