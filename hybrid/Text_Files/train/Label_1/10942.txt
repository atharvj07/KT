#include <iostream>
#include <vector>
using namespace std;
const int N = 100;
int l[N],r[N];
int main() {
    int n;
    cin >> n;
    vector<int> head;
    vector<int> tail;
    for (int i = 1; i <= n; i++) {
        cin >> l[i] >> r[i];
        if (l[i] == 0) head.push_back(i);
        if (r[i] == 0) tail.push_back(i);
    }
    for (int i : head) {
        for (int j : tail) {
            r[i] = j;
            l[j] = i;
            break;
        }
    }
    for (int i = 1; i <= n; i++)
        cout << l[i] << ' ' << r[i] << endl;
}
