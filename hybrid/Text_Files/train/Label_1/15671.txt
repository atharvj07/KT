#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m), ans(n + 1, -1);
  for (int i = 0; i < m; i++) cin >> a[i];
  for (int i = m - 1; i >= 0; i--) {
    int button = a[i];
    for (int j = button; j <= n; j++) {
      if (ans[j] == -1) ans[j] = button;
    }
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
}
