import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) a[i] = input.nextInt();
    int[] dp = new int[n];
    int[] lg = new int[n];
    for (int i = 0; i < n; i++) {
      dp[i] = 1;
      lg[i] = 0;
      for (int j = 0; j < i; j++) {
        if ((a[i] & a[j]) == 0 && dp[j] + 1 > dp[i]) {
          dp[i] = dp[j] + 1;
          lg[i] = j;
        }
      }
    }
    int max = 0;
    int last = -1;
    for (int i = 0; i < n; i++) {
      if (dp[i] > max) {
        max = dp[i];
        last = i;
      }
    }
    System.out.println(max);
  }
}
