#include <cstdio>
using ll = long long;
const ll mod = 1e9 + 7;
int main() {
  int n;
  scanf("%d", &n);
  char s[n + 1], t[n + 1];
  scanf("%s%s", s, t);
  ll ans = 1;
  for (int i = 0; i < n; i++)
    if (s[i] == '?' || t[i] == '?')
      ans = (ans * 10) % mod;
    else if (s[i] == t[i])
      return puts("0"), 0;
  printf("%lld", ans);
}
