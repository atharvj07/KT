import sys
input = sys.stdin.readline
inf = int(1e20)
a, n, m = map(int, input().split())
rain = []
for _ in range(n):
    l, r = map(int, input().split())
    rain.append((l, r))
u = []
for _ in range(m):
    x, p = map(int, input().split())
    u.append((x, p))
rain.sort(key=lambda x: x[1])
u.sort(key=lambda x: x[0])
ans = inf
last = [(-1, 0)] * (a + 1)
cur疲勞 = 0
for j in range(len(rain)):
    l, r = rain[j]
    while cur疲勞 < len(u) and u[cur疲勞][0] <= l:
        x, p = u[cur疲勞]
        if x >= l and x < r:
            last[x] = (cur疲勞, p)
        cur疲勞 += 1
    if last[l][0] == -1:
        ans = -1
        break
    else:
        aus, p = last[l]
        ans = min(ans, aus + r - l)
if ans == inf:
    ans = -1
print(ans)
