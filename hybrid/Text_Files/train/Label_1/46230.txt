import java.util.Scanner;

public class NikolayAndRooms {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int testCases = scanner.nextInt();

        for (int i = 0; i < testCases; i++) {
            int n = scanner.nextInt();
            String staircases = scanner.next();

            int[][] dp = new int[2][n + 1];
            for (int j = 1; j <= n; j++) {
                dp[0][j] = 1;
                dp[1][j] = 1;
            }

            for (int j = 1; j <= n; j++) {
                if (staircases.charAt(j - 1) == '1') {
                    dp[1][j] = Math.max(dp[1][j], dp[0][j] + 1);
                    dp[0][j] = Math.max(dp[0][j], dp[1][j] + 1);
                }

                dp[0][j] = Math.max(dp[0][j], dp[0][j - 1] + 1);
                dp[1][j] = Math.max(dp[1][j], dp[1][j - 1] + 1);
            }

            System.out.println(Math.max(dp[0][n], dp[1][n]));
        }
    }
}
