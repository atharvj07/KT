#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    vector<int> a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; i++) { cin >> b[i]; }
    vector<int> v;
    for (int i = 1; i <= n; i++) { v.push_back(a[i] - b[i]); }
    sort(v.begin(), v.end());
    ll ans = 0;
    for (int i = 0; i < n && k1 > 0 && k2 > 0; i++) {
        if (-v[i] > v[n - 1 - i]) {
            if (k1 < k2) {
                k1--;
                ans -= v[i];
            } else {
                k2--;
                ans += v[i];
            }
        } else {
            if (k1 > k2) {
                k1--;
                ans -= v[i];
            } else {
                k2--;
                ans += v[i];
            }
        }
    }
    cout << ans << endl;
    return 0;
}
