#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i] >> b[i];
  }
  sort(a.begin(), a.end());
  sort(b.rbegin(), b.rend());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] >= ans) {
      ans = a[i];
    } else {
      ans = b[i];
    }
    ans++;
  }
  cout << ans << endl;
  return 0;
}
