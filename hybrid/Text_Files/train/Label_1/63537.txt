for i in range(int(input())):
    px,py=map(int,input().split())
    s=input()
    x,y=0,0
    ans="YES"
    for j in s:
        if j=='U':
            y+=1
        if j=='D':
            y-=1
        if j=='R':
            x+=1
        if j=='L':
            x-=1
        if x==px and y==py:
            ans="NO"
            break
    print(ans)
