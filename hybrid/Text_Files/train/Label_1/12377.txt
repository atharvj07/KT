#include <iostream>
#include <vector>

using namespace std;

bool is_prime(int n) {
    if (n < 2) {
        return false;
    }
    for (int i = 2; i * i <= n; ++i) {
        if (n % i == 0) {
            return false;
        }
    }
    return true;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, e;
        cin >> n >> e;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            int prod = a[i];
            for (int k = 1; i + k * e <= n; ++k) {
                prod *= a[i + k * e];
                if (is_prime(prod)) {
                    ++ans;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
