n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = [1 for i in range(n)]
for i in range(m):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    for i in range(l, (l + r) // 2 + 1):
        if a[l] < a[r]:
            b[a[l] - 1] = -b[a[l] - 1]
        l += 1
        r -= 1
    c = 0
    for i in b:
        c += i
    if c < 0:
        print('odd')
    else:
        print('even')
