#include <bits/stdc++.h>
#define ll long long

using namespace std;

struct pts {
    ll x, y, idx;
    pts(ll x, ll y, ll i) : x(x), y(y), idx(i) {}
    bool operator<(const pts &p) const { return y < p.y; }
};

ll n, a, b;
vector<pts> v;

int main() {
    cin >> n >> a >> b;
    for (ll i = 0; i < n; i++) {
        ll x, vx, vy;
        cin >> x >> vx >> vy;
        v.push_back(pts(x, a * x + b, i));
    }
    sort(v.begin(), v.end());
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        for (ll j = i + 1; j < n; j++) {
            if (v[i].x > v[j].x) {
                break;
            } else {
                ll cross = v[i].y - v[j].y;
                ll xn = v[i].x - v[j].x;
                if (cross % xn == 0) {
                    ll tn = cross / xn;
                    ans += (tn >= 0);
                }
            }
        }
    }
    cout << ans * 2 << '\n';
    return 0;
}
