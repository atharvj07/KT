#include <bits/stdc++.h>
using namespace std;
 
#define int long long
#define ar array
 
const int mod = 1e9 + 7;
const int mxN = 2e5;

int n, k;

// store prefix frequency of each number
int pref[mxN + 1];
// store suffix frequency of each number
int suff[mxN + 1];

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int t, m;
    cin >> t;
    while (t--) {
        cin >> n >> m >> k;
        memset(pref, 0, sizeof(pref));
        memset(suff, 0, sizeof(suff));
        vector<int> a(n);
        for (int& x : a) {
            cin >> x;
            pref[x]++;
        }
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            suff[a[i]]++;
            // the maximum number minus the minimum number 
            // is at most k
            ans += suff[a[i] - k] * pref[a[i] + k];
            // the maximum number minus the minimum number 
            // is exactly k
            ans += (pref[a[i] - k] - suff[a[i] - k]) * 
                   (suff[a[i] + k] - pref[a[i] + k]) % mod;
            if (k != 0) {
                // this condition only applies when k > 0
                // the maximum number minus the minimum number
                // is at most k
                ans += suff[a[i]] * pref[a[i]];
                // the maximum number minus the minimum number
                // is exactly 0
                ans += (pref[a[i]] - suff[a[i]]) * 
                       (suff[a[i]] - pref[a[i]]) % mod;
            }
        }
        ans %= mod;
        if (ans < 0) ans += mod;
        cout << ans << "\n";
    }
    return 0;
}
