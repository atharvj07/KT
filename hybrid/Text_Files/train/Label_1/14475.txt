#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> a;

map<pair<int, int>, int> has;
vector<int> ans;

int main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    has[{a[i], i}] = 1;
  }
  for (int i = 0; i < n; i++) {
    if (has.count({a[i] - 1, i - 1}) == 0) {
      int cur = a[i], p = i;
      vector<int> res;
      while (has.count({cur, p}) == 1) {
        res.push_back(p);
        p++;
        cur++;
      }
      if (res.size() > ans.size()) {
        ans = res;
      }
    }
  }
  cout << ans.size() << "\n";
  for (int i : ans) {
    cout << i + 1 << " ";
  }
  cout << "\n";
}
