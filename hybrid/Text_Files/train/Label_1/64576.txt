import sys
x = input()
a = []
for i in range(int(x)):
    y = input()
    t = 0
    for j in a:
        if j ^ int(y) in a:t+=1;break
    if t==0: a.append(0)
    else:
        b = []
        for j in a:
            if j ^ int(y) in a:b.append(a.index(j))
            if len(b)>1:break
        a.append(len(b))
        a.extend(b)
    print(a[i])
