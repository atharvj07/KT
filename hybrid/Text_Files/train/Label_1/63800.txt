import sys

n = int(input())
a = [int(i) for i in input().split()]
last = 0
if a[0] == 3: last = 1
for i in range(1, n):
    if a[i] == a[i - 1]: continue
    if last == 0:
        if a[i] == 1: last = 1
        if a[i] == 2: last = 2
    elif last == 1:
        if a[i] == 2: last = 3
    elif last == 2:
        if a[i] == 3: last = 2
if last != 0: print("Finite")
else: print("Infinite")
if last != 0: print(last)
