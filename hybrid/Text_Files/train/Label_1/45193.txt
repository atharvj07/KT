import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), m = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = sc.nextInt();
            int[] x = new int[m];
            for (int i = 0; i < m; i++) x[i] = sc.nextInt();
            long sum = 0;
            int[] ans = new int[m];
            Arrays.fill(ans, -1);
            for (int i = 0; i < n; i++) {
                sum += a[i];
                for (int j = 0; j < m; j++) {
                    if (ans[j] != -1) continue;
                    if (sum >= x[j]) ans[j] = i + 1;
                    else if (sum + a[(i + 1) % n] >= x[j]) {
                        ans[j] = i + 2;
                        sum += a[(i + 1) % n];
                    } else sum += a[(i + 1) % n];
                }
            }
            for (int j = 0; j < m; j++) System.out.print(ans[j] + " ");
            System.out.println();
        }
    }
}
