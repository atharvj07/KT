import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int ans = a[n - 1];
        for (int i = n - 1; i >= 0; i--) {
            int x = (int) Math.sqrt(a[i]);
            if (x * x != a[i]) {
                ans = a[i];
                break;
            }
        }
        System.out.println(ans);
    }
}
