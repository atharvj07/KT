import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int k = sc.nextInt();
		if (k % 4 != 0) {
			System.out.println(-1);
			return;
		}
		char[][][] cube = new char[k][k][k];
		for (int i = 0; i < k; i++) {
			for (int j = 0; j < k; j++) {
				for (int l = 0; l < k; l++) {
					if (i % 2 == 0) {
						if (j % 2 == 0) {
							if (l % 2 == 0) {
								cube[i][j][l] = 'w';
							} else {
								cube[i][j][l] = 'b';
							}
						} else {
							if (l % 2 == 0) {
								cube[i][j][l] = 'b';
							} else {
								cube[i][j][l] = 'w';
							}
						}
					} else {
						if (j % 2 == 0) {
							if (l % 2 == 0) {
								cube[i][j][l] = 'b';
							} else {
								cube[i][j][l] = 'w';
							}
						} else {
							if (l % 2 == 0) {
								cube[i][j][l] = 'w';
							} else {
								cube[i][j][l] = 'b';
							}
						}
					}
				}
			}
		}
		for (int i = 0; i < k; i++) {
			for (int j = 0; j < k; j++) {
				for (int l = 0; l < k; l++) {
					System.out.print(cube[i][j][l]);
				}
				System.out.println();
			}
			System.out.println();
		}
	}
}
