#include <iostream>
#include <cmath>
using namespace std;
int main() {
    int v1, v2, t, d;
    cin >> v1 >> v2 >> t >> d;
    double ans = 0;
    int c = abs(v2 - v1);
    if (c <= 2 * d) {
        int e = t - 1;
        int f = t - e;
        ans = e * v1 + f * v2;
    } else
        ans = v1 * t + 0.5 * d * (t - 1) * (t - 2) + 0.5 * d * (t - 1);
    cout << (int)ans << endl;
    return 0;
}
