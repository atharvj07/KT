import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        if (a[0] == 1) {
            System.out.println(-1);
            return;
        }

        int min = 2;
        int max = 2;
        for (int i = 0; i < k; i++) {
            if (max % a[i] == 0) {
                max += a[i];
            } else {
                min = -1;
                max = -1;
                break;
            }
        }

        System.out.println(min + " " + max);
    }
}
