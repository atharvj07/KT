#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0; i<(n); i++)

int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        vector<pair<int,int>> v(n);
        rep(i,n) cin >> v[i].first >> v[i].second;
        vector<int> ans(n,0);
        int t = 0;
        int cur = 0;
        rep(i,n) {
            if(v[i].first >= cur) {
                ans[i] = cur+1;
                cur++;
            } else if(v[i].second >= cur) {
                ans[i] = cur;
            }
            cur = min(cur+1, v[i].second);
        }
        rep(i,n) cout << ans[i] << " ";
        cout << "\n";
    }
}
