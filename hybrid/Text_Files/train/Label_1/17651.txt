#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
#define cmax(a, b) (a = max((a), (b)))
#define cmin(a, b) (a = min((a), (b)))
#define s(x) ((int)x.size())
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()
#define vl vector<ll>
#define vi vector<int>
#define vb vector<bool>
#define vs vector<string>
#define vc vector<char>
#define FIO ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define FOR(i, a, b) for(int i = a; i <= b; ++i)
#define ROF(i, a, b) for(int i = a; i >= b; --i)
#define print(x) cout << x << " "
#define printl(x) cout << x << endl
#define dbg(x) cout << __LINE__ << ": " << #x << " = " << x << endl
#define endl "\n"

const int MAX_N = 2e5 + 10;
int n, a[MAX_N], b[MAX_N], ans[MAX_N];
vector<int> hand[MAX_N], pile[MAX_N], frec;

signed main() {
  FIO;
  cin >> n;
  FOR(i, 1, n) {
    cin >> a[i];
    if(a[i] != 0) hand[a[i]].push_back(i);
  }
  FOR(i, 1, n) {
    cin >> b[i];
    if(b[i] != 0) pile[b[i]].push_back(i);
  }
  FOR(i, 1, n) {
    if(a[i] == 0) frec.push_back(i);
    if(b[i] == 0) frec.push_back(i);
  }
  sort(all(frec));
  int j = 0;
  FOR(i, 1, n) {
    if(s(hand[i])) {
      ans[hand[i].back()] = frec[j++];
      hand[i].pop_back();
    } else if(s(pile[i])) {
      ans[pile[i].back()] = frec[j++];
      pile[i].pop_back();
    }
    print(ans[i]);
  }
  return 0;
}
