#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 3005;
int n;
struct dis {
  ll s, c;
  bool operator<(const dis &a) const { return s < a.s; }
};
vector<dis> p;
ll dp[N][N];
int main() {
  cin >> n;
  p.resize(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> p[i].s;
  }
  for (int i = 1; i <= n; i++) {
    cin >> p[i].c;
  }
  sort(p.begin() + 1, p.end());
  for (int i = n; i >= 1; i--) {
    for (int j = i - 1; j >= 1; j--) {
      dp[i][j] = dp[i + 1][j];
      for (int k = j - 1; k >= 1; k--) {
        if (p[k].s < p[j].s && p[j].s < p[i].s) { dp[i][j] = max(dp[i][j], dp[j][k] + p[i].c + p[j].c + p[k].c); }
      }
    }
  }
  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) { ans = max(ans, dp[i][j]); }
  }
  if (ans == 0) {
    cout << -1 << endl;
  } else {
    cout << ans << endl;
  }
  return 0;
}
