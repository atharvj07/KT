#include <bits/stdc++.h>
using namespace std;

vector<pair<int, int>> ans;
const int LIM = 1e5;
const int INF = 1<<30;

int main() {
  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  reverse(a.begin(), a.end());
  reverse(b.begin(), b.end());
  queue<pair<string, int>> q;
  map<string, int> dist;
  q.push({a, 0});
  dist[a] = 0;
  while (!q.empty()) {
    auto u = q.front();
    q.pop();
    if (u.first == b) {
      cout << u.second << endl;
      int sz = ans.size();
      for (int i = 0; i < min(sz, LIM); i++) {
        cout << ans[i].first << ' ' << ans[i].second << endl;
      }
      return 0;
    }
    for (int i = 0; i < n - 1; i++) {
      int x = u.first[i] - '0';
      int y = u.first[i + 1] - '0';
      if (x > 0 || y < 9) {
        string v = u.first;
        v[i] = (x + 1) + '0';
        v[i + 1] = (y - 1) + '0';
        if (!dist.count(v) || dist[v] > u.second + 1) {
          q.push({v, u.second + 1});
          dist[v] = u.second + 1;
          ans.push_back({i + 1, 1});
        }
      }
      if (x > 1 || y < 8) {
        string v = u.first;
        v[i] = (x - 1) + '0';
        v[i + 1] = (y + 1) + '0';
        if (!dist.count(v) || dist[v] > u.second + 1) {
          q.push({v, u.second + 1});
          dist[v] = u.second + 1;
          ans.push_back({i + 1, -1});
        }
      }
    }
  }
  cout << -1 << endl;
  return 0;
}
