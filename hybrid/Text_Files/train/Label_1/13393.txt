#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t, n;
    while (scanf("%d", &t) == 1) {
        while (t--) {
            scanf("%d", &n);
            vector<int> arr(n);
            for (int i = 0; i < n; i++)
                scanf("%d", &arr[i]);
            sort(arr.begin(), arr.end());
            long long ans = 1LL * arr[0] * arr[n - 1];
            for (int i = 0, j = n - 1; i < j; i++, j--)
                ans = max(ans, 1LL * arr[i] * arr[j]);
            printf("%lld\n", ans);
        }
    }
    return 0;
}
