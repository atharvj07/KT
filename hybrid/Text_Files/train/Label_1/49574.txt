import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int m = input.nextInt();
        int[] p = new int[m];
        for (int i = 0; i < m; i++) {
            p[i] = input.nextInt();
        }
        Arrays.sort(a);
        for (int i = 0; i < m; i++) {
            int low = 0, high = n - 1, ans = 0;
            while (low < high) {
                if (a[low] * a[high] >= p[i]) {
                    ans = high - low;
                    high--;
                } else {
                    low++;
                }
            }
            System.out.println(ans * ans);
        }
        input.close();
    }
}
