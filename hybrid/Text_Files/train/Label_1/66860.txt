class Solve:
    def __init__(self):
        self.n, self.q = [int(x) for x in input().split()]
        self.s = input()

    def solve(self):
        for _ in range(self.q):
            l, r = [int(x) for x in input().split()]
            ans = [0] * 10
            for i in range(l - 1, r):
                ch = self.s[i]
                if ch == "<":
                    self.ch = -1
                elif ch == ">":
                    self.ch = 1
                else:
                    ans[int(ch)] += 1
                    if int(ch) == 0:
                        self.s = self.s[:i] + self.s[i + 1:]
                    else:
                        self.s = self.s[:i] + str(int(ch) - 1) + self.s[i + 1:]
            print(*ans)

solve = Solve()
solve.solve()
