import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();

        int a[][] = new int[n][2];

        for (int i = 0; i < n; i++) {
            a[i][0] = sc.nextInt() - c;
            a[i][1] = sc.nextInt() - d;
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (i != j && j != k && k != i) {
                        if (a[i][0] * a[j][1] > a[j][0] * a[i][1]) {
                            if (a[j][0] * a[k][1] > a[k][0] * a[j][1]) {
                                if (a[k][0] * a[i][1] > a[i][0] * a[k][1]) {
                                    ans++;
                                }
                            }
                        }
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
