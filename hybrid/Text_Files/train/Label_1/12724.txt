#include <iostream>
#include <numeric>
using namespace std;
int main() {
    int n, a[200001], sum = 0;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) {
        sum += a[i];
        if (sum >= (accumulate(a, a + n, 0))) {
            cout << i + 1;
            return 0;
        };
    }
    return 0;
}
