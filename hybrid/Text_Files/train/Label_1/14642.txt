#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string s, t;
  cin >> s >> t;
  vector<int> s0, s1;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '0') s0.push_back(i);
    if (s[i] == '1') s1.push_back(i);
  }
  unordered_map<char, int> m;
  for (char c : t) m[c]++;
  if (m.size() != 2) cout << 0 << "\n";
  else
    cout << s0.size() * s1.size() << "\n";
  return 0;
}
