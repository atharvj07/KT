from sys import stdin

n, p, k = map(int, stdin.readline().split())
a = [int(x) for x in stdin.readline().split()]
b, c = {}, {}
for ai in a:
    if (ai ** 2) % p not in b:
        b[ai ** 2 % p] = 0
    b[ai ** 2 % p] += 1
for ai in a:
    if (p - (ai ** 2) % p) % p in b:
        c[ai] = b[(p - (ai ** 2) % p) % p]
print(sum([c.get(ai, 0) if (k - (ai + aj) % p) % p in b else 0 for aj in a if aj > ai]))
