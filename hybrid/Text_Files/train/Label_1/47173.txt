import java.io.*;
import java.util.*;

class Main {
    static int n, m;
    static int[] a, deg;
    static int[] p;
    static ArrayList<Integer>[] edge;
    static ArrayList<Edge> ans;

    static class Edge {
        int u, v, id;

        Edge(int u, int v, int id) {
            this.u = u;
            this.v = v;
            this.id = id;
        }
    }

    static int find(int x) {
        if (p[x] != x) {
            p[x] = find(p[x]);
        }
        return p[x];
    }

    static void merge(int x, int y) {
        if (x != y) {
            p[y] = x;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader sc = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(sc.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        a = new int[n];
        deg = new int[n];
        p = new int[n];
        edge = new ArrayList[n];
        ans = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            p[i] = i;
        }
        st = new StringTokenizer(sc.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < n; i++) {
            edge[i] = new ArrayList<>();
        }
        int ansCnt = 0;
        for (int i = 1; i <= m; i++) {
            st = new StringTokenizer(sc.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            deg[u]++;
            deg[v]++;
            edge[u].add(i);
            edge[v].add(i);
            if (a[u] == a[v]) {
                ansCnt++;
            }
            if (a[u] == 0 || a[v] == 0) {
                ans.add(new Edge(u, v, i));
            }
            int x = find(u);
            int y = find(v);
            if (x != y) {
                merge(x, y);
            }
        }
        if (ansCnt == m) {
            if (a[0] != 0) {
                System.out.println("-1");
            } else {
                System.out.println("0");
            }
            return;
        }
        boolean flag = false;
        for (int i = 0; i < n; i++) {
            if (a[find(i)] != a[i] && a[i] != 0) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println("-1");
            return;
        }
        int sdeg = deg[0];
        for (int i = 1; i < n; i++) {
            sdeg = gcd(sdeg, deg[i]);
        }
        if (sdeg % 2 == 0 && a[0] != 0) {
            System.out.println("-1");
            return;
        }
        for (int i = 0; i < n; i++) {
            if (a[i] < 0) {
                continue;
            }
            if (deg[i] % 2 == 0 && a[find(i)] == 1) {
                for (int j : edge[i]) {
                    ans.add(new Edge(i, (i + 1) % n, j));
                }
                deg[i] = 1;
                merge(0, find(i));
                a[0] = 1;
            }
        }
        for (int i = 0; i < n; i++) {
            if (a[i] == -1) {
                continue;
            }
            if (deg[i] % 2 == 0 && a[find(i)] == -1) {
                for (int j : edge[i]) {
                    ans.add(new Edge(i, (i + 1) % n, j));
                }
                deg[i] = 1;
                merge(0, find(i));
            }
        }
        for (int i = 0; i < n; i++) {
            if (deg[i] % 2 == 1) {
                for (int j : edge[i]) {
                    if (i == 0) {
                        ans.add(new Edge(i, i, j));
                    } else {
                        ans.add(new Edge(i, 0, j));
                    }
                }
            }
        }
        System.out.println(ans.size());
        for (Edge e : ans) {
            System.out.println(e.id);
        }
    }

    static int gcd(int x, int y) {
        while (y != 0) {
            int temp = y;
            y = x % y;
            x = temp;
        }
        return x;
    }
}
