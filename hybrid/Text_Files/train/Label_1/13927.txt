#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k, x;
  cin >> n >> k >> x;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0;
  for (int i = 0; i < k; i++) {
    ans += x;
    a[i] = 0;
  }
  sort(a.begin(), a.end());
  for (int i = 0; i < n; i++) ans += a[i];
  cout << ans << endl;
  return 0;
}
