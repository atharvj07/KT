#include <bits/stdc++.h>
using namespace std;
const int N = 210, inf = 1e9;
int a[N], b[N], pre[N], suf[N];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i <= n; ++i) b[i] = a[i];
    sort(b + 1, b + n + 1);
    for (int i = 1; i <= n; ++i) {
        pre[i] = pre[i - 1] + a[i];
        suf[n - i + 1] = suf[n - i + 2] + a[n - i + 1];
    }
    int ans = -inf;
    for (int i = 1; i <= n; ++i) {
        for (int j = max(1, i - k); j <= min(n, i + k); ++j) {
            ans = max(ans, pre[i - 1] + suf[j + 1] + b[j]);
        }
    }
    cout << ans << endl;
    return 0;
}
