#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<pair<int, int>> times(n);
  for (int i = 0; i < n; i++) {
    int h, m;
    cin >> h >> m;
    times[i] = {h * 60 + m, i};
  }
  int ansH = 24 * 60, ansM = 0;
  for (int i = 0; i < n; i++) {
    int curH = times[i].first, curM = times[i].first;
    if (i > 0) curM = max(curM, times[i - 1].first + s);
    if (i < n - 1) curH = min(curH, times[i + 1].first - s);
    if (curH - curM >= s) {
      ansH = min(ansH, curH);
      ansM = min(ansM, curM);
    }
  }
  cout << ansH / 60 << " " << ansH % 60 << endl;
  return 0;
}
