#include <iostream>

using namespace std;

int main() {
    int m;
    cin >> m;
    int h1, a1;
    cin >> h1 >> a1;
    int x1, y1;
    cin >> x1 >> y1;
    int h2, a2;
    cin >> h2 >> a2;
    int x2, y2;
    cin >> x2 >> y2;

    int cnt1 = 0, cnt2 = 0;
    while (h1 != a1 || h2 != a2) {
        cnt1++;
        h1 = (1LL * x1 * h1 + y1) % m;
        h2 = (1LL * x2 * h2 + y2) % m;
        if (cnt1 > 1e9) {
            cout << -1 << endl;
            return 0;
        }
    }

    cout << cnt1 << endl;
    return 0;
}
