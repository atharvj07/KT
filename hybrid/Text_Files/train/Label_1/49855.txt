import java.util.*;
import java.lang.*;
public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        while(cin.hasNext()){
            int H = cin.nextInt();
            int W = cin.nextInt();
            int X = cin.nextInt();
            int Y = cin.nextInt();
           if(H*W%2==1&& (X+Y)%2==1){
               System.out.println("No");
           }else {
               System.out.println("Yes");
           }
        }
    }
}
