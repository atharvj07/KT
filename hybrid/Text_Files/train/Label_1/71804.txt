n = int(input().strip())

# To minimize the number of bombs, Slava should follow a strategy
# where he first damages all tanks (except those in the first and last cell),
# making them move to the neighboring cells. Then, he should bomb
# the first and last cell to damage the tanks that have moved there.
# Finally, bombing the middle cells again will destroy all tanks.

# Minimum number of bombs required is always n + 1 for n > 2,
# and for n = 2, it's 3.
if n == 2:
    print(3)
    print("2 1 2")
else:
    print(n + 1)
    for i in range(2, n):
        print(i, end=" ")
    print("1", n, "2")
