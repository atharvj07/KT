#include<bits/stdc++.h>
using namespace std;

int a[100005], f[100005], ans[100005];

int main() {
    int n, m, l;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        f[a[i]] = i;
    }
    for (int i = 1; i <= m; i++) {
        cin >> l;
        ans[i] = max(ans[i - 1], a[l]);
        if (l < n) ans[i] = max(ans[i], f[a[l + 1]]);
    }
    for (int i = 1; i <= m; i++) {
        cout << 1 + ans[i] - f[ans[i]] << '\n';
    }
    return 0;
}
