#include <iostream>
using namespace std;

int main() {
    long long a, b;
    cin >> a >> b;
    if (a == b) {
        cout << 0 << endl;
        return 0;
    }
    if (a % 5 == 0 && b % 3 == 0) {
        cout << 1 << endl;
        return 0;
    }
    if (a % 3 == 0 && b % 5 == 0) {
        cout << 1 << endl;
        return 0;
    }
    if (a % 2 == 0 && b % 2 == 0) {
        cout << 1 << endl;
        return 0;
    }
    cout << -1 << endl;
    return 0;
}
