#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<long long> b(n / 2);
    for (auto &bi : b) { cin >> bi; }
    sort(begin(b), end(b));
    for (int i = 0; i < n / 2; ++i) {
        cout << max(0ll, b[i] - b[n / 2 - 1 - i]) << " ";
    }
    for (int i = n / 2 - 1; i >= 0; --i) {
        cout << b[i] - max(0ll, b[i] - b[n / 2 - 1 - i]) << " ";
    }
    cout << "\n";
    return 0;
}
