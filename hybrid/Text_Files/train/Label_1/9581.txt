#include<bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5+5;
int32_t main()
{
    int n;
    cin>>n;
    multiset<pair<int,int>> arr;
    for(int i=1;i<=n;i++)
    {
        int a,b;
        cin>>a>>b;
        arr.insert({a,i});
        arr.insert({a+b,-i});
    }
    int st=0,cnt=0;
    map<int,pair<int,int>> ans;
    for(auto x:arr)
    {
        if(x.second>0)
        {
            cnt++;
            if(cnt==1)
                st=x.first;
        }
        else
        {
            cnt--;
            ans[abs(x.second)]={st,st+abs(x.second)-1};
            st=st+abs(x.second);
        }
    }
    for(int i=1;i<=n;i++)
        cout<<ans[i].first<<" "<<ans[i].second<<endl;
}
