#include <bits/stdc++.h>
using namespace std;
char grid1[20][20], grid2[20][20];
int cost[20][20];
int main() {
  int n, m, k, w;
  cin >> n >> m >> k >> w;
  for (int i = 1; i <= k; i++) {
    for (int j = 1; j <= n; j++) {
      for (int l = 1; l <= m; l++) { cin >> grid1[j][l]; }
    }
    for (int j = 1; j <= n; j++) {
      for (int l = 1; l <= m; l++) { cin >> grid2[j][l]; }
    }
    int mn = 1e9;
    for (int j = 1; j <= k; j++) {
      int c = 0;
      for (int l = 1; l <= n; l++) {
        for (int r = 1; r <= m; r++) { c += (grid1[l][r] != grid2[l][r]); }
      }
      mn = min(mn, c * w);
      cost[i][j] = c * w;
    }
    mn = min(mn, n * m);
    cout << mn << endl;
    int last = 0, cur = 1;
    if (mn == n * m) {
      cout << i << " 0\n";
    } else {
      for (int j = 1; j <= k; j++) {
        if (j != i && cost[i][j] == mn) {
          cout << i << " " << j << "\n";
          last = j;
          break;
        }
      }
      for (int j = 1; j <= k; j++) {
        if (j != i && j != last && cost[last][j] == mn) {
          cout << i << " " << j << "\n";
        }
      }
    }
  }
  return 0;
}
