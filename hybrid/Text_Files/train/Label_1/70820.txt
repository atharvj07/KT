n, m = map(int, input().split())
d = list(map(int, input().split()))
a = list(map(int, input().split()))
days = [[] for i in range(m+1)]
for i in range(n):
    days[d[i]].append(i)
dp = [10**9] * (n+1)
dp[0] = 0
for i, day in enumerate(days):
    if day:
        dp[day[0]] = 1
        for j in day:
            for k in range(1, j+1):
                dp[j] = min(dp[j], dp[k-1] + a[d[k]-1])
if dp[n] != 10**9:
    print(dp[n])
else:
    print(-1)
