#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll MOD = 1e9+7;

ll f[505][505][505];
ll dp[505][505][505];
ll pre[505];

ll solve(int rem,int cnt,int cur,int a[],int n,int m,int b){
    if(cur>b) return 0;
    if(cnt>n) return 0;
    if(rem==0 && cnt==n){
        return 1;
    }
    if(rem==0){
        return 0;
    }
    ll &ans = dp[rem][cnt][cur];
    if(ans!=-1){
        return ans;
    }
    ll op1 = solve(rem,cnt+1,cur,a,n,m,b);
    ll op2 = solve(rem-a[cnt],cnt+1,cur+a[cnt],a,n,m,b);
    return ans = (op1+op2)%MOD;
}

int main(){
    int n,m,b,mod;
    cin>>n>>m>>b>>mod;
    int a[n];
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    f[0][0][0] = 1;
    for(int i=1;i<=m;i++){
        for(int j=0;j<=n;j++){
            for(int k=0;k<=b;k++){
                if(f[i][j][k]==0) continue;
                // don't chose 
                f[i+1][j][k] = (f[i+1][j][k]+f[i][j][k])%mod;
                // chose 
                if(k+a[j-1]<=b){
                    f[i+1][j+1][k+a[j-1]] = (f[i+1][j+1][k+a[j-1]]+f[i][j][k])%mod;
                }
            }
        }
    }
    for(int i=0;i<=b;i++){
        pre[i] = f[m][n][i];
    }
    for(int i=1;i<=b;i++){
        pre[i] = (pre[i]+pre[i-1])%mod;
    }
    ll ans = 0;
    for(int i=0;i<=b;i++){
        ans = (ans+((pre[b]-pre[i-1]+mod)%mod)*solve(m,0,i,a,n,m,b))%mod;
    }
    cout<<ans<<endl;
    return 0;
}
