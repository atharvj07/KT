#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;

int main() {
    int n; cin >> n;
    string s; cin >> s;

    int ans = 0;
    int l[26], r[26]; // count of occurrences of each alphabet
    memset(l, 0, sizeof l);
    memset(r, 0, sizeof r);

    int cur_l = 0;
    for (int i = 0; i < n; i++) {
        int c = s[i] - 'a';
        l[c]++;
        cur_l = max(cur_l, l[c]);
    }

    int cur_r = 0;
    for (int i = n - 1; i >= 0; i--) {
        int c = s[i] - 'a';
        r[c]++;
        cur_r = max(cur_r, r[c]);
    }

    for (int i = 0; i < 26; i++) {
        if (l[i] == cur_l) {
            ans = (ans + 1LL * l[i] * (l[i] - 1)) % MOD;
        }
        if (r[i] == cur_r) {
            ans = (ans + 1LL * r[i] * (r[i] - 1)) % MOD;
        }
    }

    cout << ans << endl;
    return 0;
}
