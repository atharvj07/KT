import sys

n, k = map(int, input().split())
coupons = []
for _ in range(n):
    l, r = map(int, input().split())
    coupons.append((l, r))
coupons.sort(key=lambda x: x[1])

ans = 0
chosen = []
i = 0
while i < k and coupons:
    l, r = coupons[0]
    if l <= r:
        ans += r - l + 1
        chosen.append(l)
        coupons = [(x, y) for x, y in coupons if y < l or x > r]
    else:
        coupons.pop(0)
    i += 1
print(ans)
print(*chosen)
