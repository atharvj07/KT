#include <iostream>
using namespace std;
typedef long long ll;
const ll maxn = 200000L;

ll occ = 0;
ll n, m;

ll t[maxn], s[maxn];
char ct[maxn], cs[maxn];
ll p1, p2;

int main() {
  cin >> n >> m;
  for (ll i = 0; i < n; i++) {
    cin >> t[i] >> ct[i];
  }
  for (ll i = 0; i < m; i++) {
    cin >> s[i] >> cs[i];
  }

  while (p1 < n && p2 < m) {
    if (ct[p1] == cs[p2]) {
      if (t[p1] == s[p2]) {
        occ++;
        p1 += t[p1];
      }
      p2 += s[p2];
    } else {
      if (t[p1] > s[p2]) {
        p1 += t[p1] - s[p2];
      } else {
        p2 += s[p2] - t[p1];
      }
    }
  }

  cout << occ << endl;
  return 0;
}
