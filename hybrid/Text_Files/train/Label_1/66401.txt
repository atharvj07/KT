n = int(input())
letters = list(map(int, input().split()))

operations = 0
if 1 in letters:
    first_unread = letters.index(1)
    last_unread = n - 1 - letters[::-1].index(1)
    operations = last_unread - first_unread + 1
    for i in range(first_unread, last_unread):
        if letters[i] == 0 and letters[i + 1] == 1:
            operations += 1

print(operations)
