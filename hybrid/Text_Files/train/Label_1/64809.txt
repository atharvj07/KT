n=int(input())
a=[int(x) for x in input().split()]
for i in range(n):
    if i%a[i]==0:print('A',end='');continue
    j=i-a[i]
    while j>=0:
        if j%a[j]==0:break
        j-=a[j]
    if j>=0:print('A',end='');continue
    j=i+a[i]
    while j<n:
        if j%a[j]==0:break
        j+=a[j]
    if j<n:print('A',end='');continue
    print('B',end='')
