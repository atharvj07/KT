for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    if sorted(a) == a:
        print("Yes")
    else:
        for i in range(n - 1):
            if a[i] > a[i + 1] and b[i] != b[i + 1]:
                print("Yes")
                break
        else:
            print("No")
