#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll M = 1e9 + 7, N = 1e5 + 5;

ll a[N], b[N];
bool vis[N];
vector<pair<ll, ll>> ops[N];
vector<pair<ll, ll>> queries[N];

int main() {
    ll n, q;
    cin >> n >> q;
    for (ll i = 1; i <= n; ++i) cin >> a[i];
    for (ll i = 1; i <= n; ++i) cin >> b[i];
    for (ll i = 0; i < q; ++i) {
        ll l, r;
        cin >> l >> r;
        queries[r].emplace_back(l, i);
    }
    ll op = 0;
    for (ll i = 1; i <= n; ++i) {
        if (a[i] != b[i] && !vis[i]) {
            op++;
            vis[i] = true;
            for (auto p : queries[i]) {
                ops[p.second].emplace_back(p.first, op);
            }
        }
    }
    for (ll i = 0; i < q; ++i) {
        if (ops[i].empty()) cout << "-1\n";
        else {
            ll res = -1;
            set<ll> s;
            for (auto p : ops[i]) {
                if (!s.count(p.second)) {
                    s.insert(p.second);
                    if (res == -1) res = 0;
                    res++;
                }
            }
            cout << res << "\n";
        }
    }
    return 0;
}
