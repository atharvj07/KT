#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  if (n * (n - 1) / 2 < k * n) {
    cout << -1 << endl;
  } else {
    cout << k * n << endl;
    for (int i = 1; i <= n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (k) {
          cout << i << ' ' << j << endl;
          --k;
        } else {
          cout << j << ' ' << i << endl;
          ++k;
        }
      }
    }
  }
}
