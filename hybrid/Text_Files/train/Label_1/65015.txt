p,q,l,r=map(int,input().split())
a,b,aa,bb=({},{},{},{})
for i in range(p):
 y,z=map(int,input().split())
 a[y]=1
 b[z]=1
for i in range(q):
 y,z=map(int,input().split())
 aa[y]=1
 bb[z]=1
ans=0
for i in range(l,r+1):
  chk=0
  for j in range(p):
   if ((i>=j+a[j])and(i<=j+b[j])):
    chk=1
   if ((i+j>=a[j])and(i+j<=b[j])):
    chk=1
  if chk==0:
   continue
  chk=0
  for j in range(q):
   if ((i>=j+aa[j])and(i<=j+bb[j])):
    chk=1
   if ((i+j>=aa[j])and(i+j<=bb[j])):
    chk=1
  if chk==1:
   ans+=1
print(ans)
