import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextLong();
        }
        Arrays.sort(a);
        long count = 0;
        int j = 0;
        for (int i = 0; i < n; i++) {
            while (j < n && a[i] * 2 >= a[j]) {
                j++;
            }
            count += Math.max(0, j - i - 1);
        }
        System.out.println(count);
    }
}
