import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        Set<Character> set = new HashSet<>();
        char cur = 'a';
        for (char c : s.toCharArray()) {
            if (!set.add(c)) continue;
            char old = c;
            while (c != '\0') {
                if (c == old) c = cur;
                c++;
            }
            cur++;
        }
        System.out.println(set.size() <= 26 ? "YES" : "NO");
    }
}
