#include <iostream>
#include <vector>
using namespace std;
int n, m, ans = 2e9, a[2505][2505];
int32_t main() {
  cin >> n >> m;
  vector<int> b(n, (m + 1) / 2), c((n + 1) / 2, (m + 1) / 2);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < m; j++) { a[i][j] = s[j] - '0'; }
  }
  for (int k = 1; k * k <= n * m; k++) {
    int tot = 0;
    if (n % k || m % k) continue;
    int nx = n / k, ny = m / k;
    for (int i = 0; i * k < n; i++)
      for (int j = 0; j * k < m; j++) {
        int p = 0;
        for (int ii = i * k; ii < (i + 1) * k; ii++)
          for (int jj = j * k; jj < (j + 1) * k; jj++) { p += a[ii][jj]; }
        tot += (p + k * k / 2) / (k * k);
      }
    if (tot < ans) ans = tot;
  }
  cout << ans << endl;
}
