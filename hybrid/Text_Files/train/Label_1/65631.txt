for _ in range(int(input())):
    n, x = map(int, input().split())
    a = [(i+1)%x for i in map(int, input().split())]
    dp = [0]*n
    dp[0] = 1
    last = [0]*x
    for i in range(n):
        last[a[i]] = i+1
        if a[i]==0:
            dp[i] = i+1
            continue
        dp[i] = max(dp[i-1], i-last[a[i]]+1)
    print(dp[-1] if dp[-1]>1 else -1)
