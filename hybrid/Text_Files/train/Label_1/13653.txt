#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

ll pshmak(vector<ll> &a, ll n, ll x) {
  ll ans = 0, r = n - 1, cnt = 0;
  for (ll l = n - 1; l >= 0; l--) {
    while (r >= 0 && a[r] >= x) {
      if (a[r] == x) cnt++;
      r--;
    }
    if (r < 0) break;
    if (l > r) continue;
    ans += cnt * (cnt - 1) / 2;
  }
  return ans;
}
int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    for (ll j = i + 1; j < n; j++) {
      ll x = a[i], y = a[j];
      ans += pshmak(a, i, x) > pshmak(a, n, y);
      ans -= pshmak(a, j, x) > pshmak(a, n, y);
    }
  }
  cout << ans << endl;
  return 0;
}
