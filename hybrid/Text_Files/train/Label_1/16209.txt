#include<bits/stdc++.h>
using namespace std;

int par[300005], par2[300005];

int findp(int x, int par[]) { return par[x] == x ? x : par[x] = findp(par[x], par); }

int main() {
    int n, x, y, p, q, c = 0;
    cin >> n >> x >> y;
    for (int i = 1; i < n; i++) { cin >> p >> q; par[p] = p; par2[q] = q; }
    for (int i = 1; i < n; i++) {
        int p = findp(par2[i], par), q = findp(par[i], par2);
        if (p != q) {
            c++;
            par[p] = q;
            par2[q] = p;
        }
    }
    cout << (c - 2) * (c - 1) / 2 << '\n';
    return 0;
}
