#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<int> a(n + 1), b(n + 1);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> b[i]; }
  for (int i = 1; i <= n; i++) { a[i] ^= b[i]; }
  cout << ((a[s] == 0) ? "YES" : "NO") << endl;
  return 0;
}
