#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.rbegin(), a.rend());
    int ans = 0;
    vector<int> order;
    for (int i = 0; i < n; i++) {
        ans += (a[i] * i + 1);
        order.push_back(i + 1);
    }
    cout << ans << endl;
    for (int u : order) {
        cout << u << ' ';
    }
    cout << endl;
    return 0;
}
