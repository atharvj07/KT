#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define fast ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);

int deg[100001];
vector<int> adj[100001];

void dfs(int u, int p, vector<int>& path) {
    path.push_back(u);
    for(auto v: adj[u]) {
        if(v == p) continue;
        dfs(v, u, path);
    }
}

int main() {
    fast;
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        for(int i=1; i<=n+1; i++) {
            adj[i].clear();
            deg[i] = 0;
        }
        for(int i=1; i<=n; i++) {
            int a;
            cin >> a;
            if(!a) {
                deg[i]++;
                deg[n+1]++;
                adj[i].push_back(n+1);
                adj[n+1].push_back(i);
            }
            else {
                deg[i]++;
                deg[n+1]++;
                adj[n+1].push_back(i);
                adj[i].push_back(n+1);
            }
        }
        int start = -1;
        for(int i=1; i<=n+1; i++) {
            if(deg[i] % 2 == 1) {
                start = i;
                break;
            }
        }
        if(start == -1) {
            cout << -1 << endl;
            continue;
        }
        vector<int> path;
        dfs(start, -1, path);
        for(auto x: path) cout << x << ' ';
        cout << endl;
    }
    return 0;
}
