#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 2), suf(n + 2);
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
    suf[i] = i;
  }
  p[0] = p[n + 1] = n + 1;
  suf[0] = suf[n + 1] = 0;
  for (int i = n; i >= 1; --i) {
    if (p[i] > p[suf[i + 1]]) suf[i] = suf[i + 1];
  }
  long long ans = 0;
  for (int i = 1; i <= n; ++i) {
    ans += suf[i - 1] - i + 1;
  }
  cout << ans << endl;
  return 0;
}
