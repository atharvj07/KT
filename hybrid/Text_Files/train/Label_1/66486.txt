import sys

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for i in range(n + 2)]  # 1-based indexing

for i in range(n - 1):
    u, v, w = map(int, input().split())
    adj[u].append((v, w))
    adj[v].append((u, w))

q = list(map(int, input().split()))

sub_tree_max = [0] * (n + 1)
ans = {}

def dfs(u, p, w):
    sub_tree_max[u] = w
    for v, weight in adj[u]:
        if v == p:
            continue
        dfs(v, u, max(w, weight))

dfs(1, 0, 0)

for weight in q:
    cnt = 0
    for i in range(1, n + 1):
        for j, w in adj[i]:
            if j > i and sub_tree_max[i] > weight and sub_tree_max[j] > weight:
                cnt += 1
    ans[weight] = cnt

for weight in q:
    print(ans[weight])
