#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <set>

int main() {
  int n;
  std::cin >> n;

  std::set<std::string> name_set;
  std::vector<std::string> name_list;

  for (int i = 0; i < n; ++i) {
    std::string name;
    std::cin >> name;
    name_set.insert(name);
    name_list.push_back(name);
  }

  bool is_lexicographical = true;
  std::string new_alphabet = "abcdefghijklmnopqrstuvwxyz";

  for (char c = 'a'; c <= 'z'; ++c) {
    std::vector<std::string> modified_name_list;

    for (const std::string &name : name_list) {
      std::string modified_name;

      for (char d : name) {
        int pos = d - 'a';
        modified_name += new_alphabet[pos];
      }

      modified_name_list.push_back(modified_name);
    }

    std::sort(modified_name_list.begin(), modified_name_list.end());

    for (int i = 0; i < n; ++i) {
      if (modified_name_list[i] != name_list[i]) {
        is_lexicographical = false;
        break;
      }
    }

    if (is_lexicographical) {
      std::cout << new_alphabet << std::endl;
      return 0;
    }

    std::next_permutation(new_alphabet.begin(), new_alphabet.end());
  }

  std::cout << "Impossible" << std::endl;

  return 0;
}
