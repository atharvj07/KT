import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int[] ai = new int[n];
        for (int i = 0; i < n; i++) {
            ai[i] = scanner.nextInt();
        }
        scanner.close();
        long res = Long.MAX_VALUE;
        int g = 0;
        for (int i = 0; i < n; i++) {
            g = gcd(g, ai[i]);
        }
        if (g > 1) {
            res = 0;
        } else {
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    int ng = gcd(g, ai[i] - 1);
                    ng = gcd(ng, ai[j] - 1);
                    if (ng > 1) {
                        res = Math.min(res, (long) a * (j - i + 1) + (long) b * 2);
                    }
                    ng = gcd(g, ai[i] + 1);
                    ng = gcd(ng, ai[j] + 1);
                    if (ng > 1) {
                        res = Math.min(res, (long) a * (j - i + 1) + (long) b * 2);
                    }
                }
            }
        }
        System.out.println(res);
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        } else {
            return gcd(b, a % b);
        }
    }
}
