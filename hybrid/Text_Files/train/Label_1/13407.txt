#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const ll inf = 1e9;
ll n, k, a[1000001];
vector<ll> dp;
int main() {
    scanf("%lld %lld", &n, &k);
    dp.resize(n + 1, inf);
    for (ll i = 1; i <= k; i++) { scanf("%lld", &a[i]); }
    dp[0] = 0;
    for (ll i = 1; i <= n; i++) {
        for (ll j = 1; j <= k; j++) {
            if (i - a[j] >= 0) dp[i] = min(dp[i], 1 + dp[i - a[j]]);
        }
    }
    if (dp[n] == inf) printf("-1");
    else
        printf("%lld", dp[n]);
    return 0;
}
