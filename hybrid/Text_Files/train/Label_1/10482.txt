#include <iostream>
#include <vector>
#include <queue>
using namespace std;

#define ll long long
#define mod 1000000007

int dirx[] = {-1, 0, 1, 0};
int diry[] = {0, 1, 0, -1};

bool valid(int x, int y, int r, int c) {
  return x >= 0 && y >= 0 && x < r && y < c;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int r, c;
    cin >> r >> c;
    vector<vector<char>> grid(r, vector<char>(c));
    for (int i = 0; i < r; i++) {
      for (int j = 0; j < c; j++) {
        cin >> grid[i][j];
      }
    }

    int ans = 0;
    bool flag = true;
    for (int i = 0; i < r; i++) {
      if (grid[i][0] == 'P') {
        flag = false;
        ans++;
        for (int j = 0; j < c; j++) {
          grid[i][j] = 'A';
        }
      }
    }
    for (int i = 0; i < c; i++) {
      if (grid[0][i] == 'P') {
        flag = false;
        ans++;
        for (int j = 0; j < r; j++) {
          grid[j][i] = 'A';
        }
      }
    }
    if (flag) {
      cout << 0 << endl;
      continue;
    }

    flag = true;
    for (int i = r - 1; i >= 0; i--) {
      if (grid[i][c - 1] == 'P') {
        flag = false;
        ans++;
        for (int j = 0; j < c; j++) {
          grid[i][j] = 'A';
        }
      }
    }
    for (int i = c - 1; i >= 0; i--) {
      if (grid[r - 1][i] == 'P') {
        flag = false;
        ans++;
        for (int j = 0; j < r; j++) {
          grid[j][i] = 'A';
        }
      }
    }
    if (flag) {
      cout << "MORTAL" << endl;
      continue;
    }

    cout << ans << endl;
  }
  return 0;
}
