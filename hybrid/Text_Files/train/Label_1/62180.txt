import sys

def solve():
    n, p, k = map(int, input().split())
    a = input()
    x, y = map(int, input().split())
    
    ans = 0
    for i in range(p, n + 1, k):
        if a[i - 1] == '0':
            if a[i - 2] == '1':
                ans += y
            else:
                ans += x
    print(ans)

for _ in range(int(input())):
    solve()
