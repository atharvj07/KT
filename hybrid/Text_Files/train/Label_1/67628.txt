import sys
n=int(input())
a=list(map(int,input().split()))
for i in range(n):
    if a[i]==-1:
        a[i]=n+1-i;
ans=n*(n-1)//4
b=[0]*n
for x in a:
    if x==-1:
        continue
    b[x-1]+=1
    ans-=b[x-1]
ans%=998244353
den=1
for x in b:
    den*=x
den%=998244353
inv=pow(den,998244352,998244353)
print((ans*inv)%998244353)
