t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    weights = list(map(int, input().strip().split()))
    total_weight = sum(weights)
    if total_weight % 2 != 0:
        print("NO")
    else:
        ones = weights.count(1)
        twos = weights.count(2)
        if twos % 2 == 0 or (twos % 2 == 1 and ones >= 2):
            print("YES")
        else:
            print("NO")
