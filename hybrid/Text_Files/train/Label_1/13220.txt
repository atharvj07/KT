#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    k = min(k, 26);
    vector<char> v(n);
    for (int i = 0; i < n; i++) { v[i] = 'a' + i % k; }
    for (char c : v) { cout << c; }
    cout << '\n';
  }
}
