s = input()
t = input()
times = t.count(s)
if times == 0:
    print(0)
else:
    x = [0] * len(t)
    n = len(s)
    suf = {}
    for i in range(len(t)-1,-1,-1):
        if t[i] == s[n-1]:
            if n > 1:
                suf[i] = x[i+1]
            else:
                suf[i] = 1
        else:
            suf[i] = 0
    cnt = 0
    for i in range(len(t)):
        if t[i] == s[0]:
            if n > 1:
                cnt = cnt + x[i+1]
            else:
                cnt = cnt + 1
        x[i] = cnt
    for i in range(len(t) - len(s) + 1):
        if i+n-1 <= len(t) - 1:
            cnt = cnt + suf[i+n-1]
    print(cnt)
