import sys

t = int(input())
for _ in range(t):
    s = input()
    sl = list(s)
    qcount = sl.count('?')
    if qcount == 0:
        print(s)
    elif qcount == 1:
        for i in range(len(sl)):
            if sl[i] == '?':
                if i == 0:
                    sl[i] = 'a'
                    break
                elif sl[i-1] != sl[i+1]:
                    sl[i] = sl[i+1]
                    break
                else:
                    sl[i] = 'a'
                    break
        if sl.count('?') == 0:
            print("".join(sl))
        else:
            print(-1)
    else:
        print(-1)
