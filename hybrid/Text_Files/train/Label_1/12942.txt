#include <iostream>
#include <vector>

long long mod = 1000000007;

int main() {
  int n;
  std::cin >> n;
  std::vector<int> p(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> p[i];
  }
  std::vector<long long> dp(n + 1, 1);
  for (int i = 1; i <= n; ++i) {
    if (p[i - 1] == 1) {
      dp[i] = 2 * dp[i - 1] % mod;
    } else {
      dp[i] = (dp[i - 1] + dp[p[i - 1] - 1]) % mod;
    }
  }
  std::cout << dp[n] << std::endl;
  return 0;
}
