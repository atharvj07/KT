#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 998244353;
int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  ll ans = 0;
  ll prv = LLONG_MAX, pre = 0;
  for (ll i = 0; i < n; ++i) {
    if (a[i] > 0 && a[i] - 1 == prv) {
      ans += (i - pre - 1) % mod;
      pre = i;
    }
    prv = a[i];
  }
  ans = (ans + (n - pre - 1)) % mod;
  if (ans < 0) ans += mod;
  cout << ans << "\n";
  return 0;
}
