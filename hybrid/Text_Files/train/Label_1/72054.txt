import sys
import math

n, m, k = list(map(int, input().strip().split()))
if n == 1 or m == 1:
    if n == 1 and m == 1:
        if k == 0:
            print("YES\n2\n1 U\n1 D")
        else:
            print("NO")
    else:
        if (n == 1 and k % 2 == 1) or (m == 1 and k % 2 == 0):
            print("NO")
        elif m == 1:
            if k % 2 == 1:
                print("YES\n1 D\n", (k - 1) // 2, " R\n1 D")
            else:
                print("YES\n", (k - 2) // 2, " R\n1 D\n1 D")
        else:
            if k % 2 == 0:
                print("YES\n1 R\n", (k - 2) // 2, " D\n1 L")
            else:
                print("YES\n1 R\n1 D\n", (k - 3) // 2, " D\n1 L")
else:
    if k % 2 == 0:
        print("YES\n1 R\n1 D\n", (k - 2) // 2, " L")
