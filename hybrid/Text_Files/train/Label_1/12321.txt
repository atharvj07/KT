#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> l(n), r(n);
    for (int i = 0; i < n; i++) cin >> l[i];
    for (int i = 0; i < n; i++) cin >> r[i];
    vector<int> a(n, 0);
    bool ok = true;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j < n; j++) {
            if (a[j] && a[j] <= n) {
                if (l[i] > 0) a[j + 1] = a[j] + 1;
                if (r[i] > 0) a[j - 1] = a[j] + 1;
                l[i] = l[i] - 1;
                r[i] = r[i] - 1;
            }
        }
        if (l[i] > 0 || r[i] > 0) {
            ok = false;
            break;
        }
    }
    if (!ok) {
        cout << "NO";
    } else {
        cout << "YES\n";
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) a[i] = 1;
            cout << a[i] << " ";
        }
    }
    return 0;
}
