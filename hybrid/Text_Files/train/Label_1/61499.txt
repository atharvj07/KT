import sys
mod = 998244353
n, m = map(int, input().split())
arr = [input() for _ in range(n)]
ww, wb = [], []
for i in range(n):
    for j in range(m):
        if arr[i][j] == 'o':
            if i%2==j%2: ww.append((i, j))
            else: wb.append((i, j))
w = len(ww)
b = len(wb)
ans = (1<<w)*(1<<b)
print(ans%mod)
