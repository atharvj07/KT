#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = (int)1e9 + 7;

int main() {
    ll n, f = 0, ans = 1;
    cin >> n;
    string s;
    while (n--) {
        cin >> s;
        if (s[0] == 's') ans = (ans * f) % mod;
        else f++;
    }
    cout << ans << endl;
    return 0;
}
