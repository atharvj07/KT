t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    players = sorted([(max(a[i], b[i]), min(a[i], b[i]), i) for i in range(n)], reverse=True)
    
    winners = ['0'] * n
    max_second = -1
    for _, second, idx in players:
        if second > max_second:
            winners[idx] = '1'
            max_second = second
            
    print(''.join(winners))
