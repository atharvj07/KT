import java.util.Scanner;
import java.util.HashSet;
import java.util.Set;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    String[] codes = new String[n];
    for (int i = 0; i < n; i++) {
      codes[i] = input.next();
    }
    Set<String> ambiguous = new HashSet<>();
    int minLength = Integer.MAX_VALUE;
    for (int i = 0; i < n; i++) {
      String code = codes[i];
      for (int j = i + 1; j < n; j++) {
        String otherCode = codes[j];
        if (code.length() + otherCode.length() < minLength) {
          if (code.endsWith(otherCode) || otherCode.endsWith(code)) {
            ambiguous.add(code + otherCode);
            ambiguous.add(otherCode + code);
            minLength = code.length() + otherCode.length();
          }
        }
      }
    }
    System.out.println(ambiguous.isEmpty() ? 0 : minLength);
  }
}
