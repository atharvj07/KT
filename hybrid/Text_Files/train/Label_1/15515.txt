#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  string a, b, c;
  cin >> a >> b;
  int n, flag;
  cin >> n;
  vector<string> v = {a, b};
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 2; j++) {
      cin >> c;
      if (c == a || c == b) {
        cout << a << " " << b << endl;
        a = v[flag];
      } else {
        cout << a << " " << c << endl;
        v[flag] = c;
        a = v[j];
        flag = j;
      }
    }
  }
}
