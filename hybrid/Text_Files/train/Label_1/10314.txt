#include <vector>
#include <iostream>

using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n, k; cin >> n >> k;
    vector<long long> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    long long ans = 0, cur = p[0];
    for (int i = 1; i < n; i++) {
      if (p[i] * 100 > k * (cur + p[i])) {
        ans += p[i] - (cur + p[i]) * k / 100;
        p[i] = (cur + p[i]) * k / 100;
      }
      cur += p[i];
    }
    cout << ans << endl;
  }
}
