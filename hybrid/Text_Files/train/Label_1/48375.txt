import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N, S;
		N = sc.nextInt();
		S = sc.nextInt();
		int[] A = new int[N];
		for (int i = 0; i < N; i++) A[i] = sc.nextInt();
		int[] cnt = new int[3001];
		for (int i = 0; i < N; i++) cnt[A[i]]++;
		long ans = 0;
		for (int i = 1; i <= S; i++) {
			for (int j = 1; j <= S; j++) {
				if (i + j <= S) ans += cnt[i] * cnt[j] * cnt[S - i - j];
			}
		}
		System.out.println(ans % 998244353);
	}
}
