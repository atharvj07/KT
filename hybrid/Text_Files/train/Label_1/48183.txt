import java.util.Scanner;
class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int days;

        while ((days=s.nextInt())>0) {
            int[][] a = new int[4][4];
            int[][] b = new int[4][4];

            for (int i=0; i<7; i++) a[i/2][i%2]=1;
            for (int i=0; i<days; i++) {
                for (int j=0; j<4; j++) {
                    for (int k=0; k<4; k++) b[j][k]=s.nextInt();
                }
                for (int j=0; j<4; j++) {
                    for (int k=0; k<4; k++) {
                        if (b[j][k]==1) a[j][k]=1;
                    }
                }
            }
            boolean ok=true;
            for (int i=0; i<4; i++) {
                for (int j=0; j<4; j++) {
                    if (a[i][j]==0) ok=false;
                }
            }
            if (ok) System.out.println(1); else System.out.println(0);
        }
    }
}
