from sys import stdin
n,m=map(int,stdin.readline().split())
a=[[] for i in range(n+1)]
for i in range(m):
 a[int(stdin.readline())].append(int(stdin.readline()))
s=-1
for i in range(1,n+1):
 if len(a[i])==2:
  for j in a[a[i][0]]:
   for k in a[a[i][1]]:
    if j==k and j!=i:
     s=i+a[i][0]+a[i][1]
     break
   if s!=-1:break
  if s!=-1:break
print(s)
