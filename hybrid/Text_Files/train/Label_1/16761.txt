#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> cnt(26);
    string s;
    cin >> s;
    for (char c : s) {
        cnt[c - 'A']++;
    }
    sort(cnt.rbegin(), cnt.rend());
    int ans = 0;
    for (int i = 0; i < min(k, n); i++) {
        ans += cnt[i];
    }
    cout << ans << endl;
}
