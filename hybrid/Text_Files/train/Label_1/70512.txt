t = int(input())
for _ in range(t):
    a, b, c, d = map(int, input().split())
    x, y, x1, y1, x2, y2 = map(int, input().split())
    print(["NO", "YES"][
        x1 <= x <= x2 and y1 <= y <= y2 and a + b - c - d >= 0 and a - b + c - d >= 0 and
        -a - b + c + d >= 0 and -a + b - c + d >= 0
    ])
