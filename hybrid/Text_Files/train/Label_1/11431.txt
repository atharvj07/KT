#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i=a;i<b;++i)
const int maxN=1e5;
const int mod=998244353;
int n;
string s;
int a[maxN];
int l[maxN];
int r[maxN];
int cur;
int main()
{
 cin>>n;
 cin>>s;
 rep(i,0,n){cin>>a[i];}
 for(int i=n-1;i>=0;--i){r[i]=max(r[i+1],i);if(s[i]=='d')r[i]=max(r[i],i+1);}
 for(int i=0;i<n;++i){l[i]=min(l[i-1],i);if(s[i]=='h')l[i]=min(l[i],i-1);}
 rep(i,0,n)
 {
  if(l[i]==-1 || r[i]==n)continue;
  if(s[i]=='r')cur+=a[l[i]];
  else if(s[i]=='a')cur+=a[r[i]];
  else if(s[i]=='d')cur+=a[r[i]]+a[l[i]];
 }
 cout<<cur;
 return 0;
}
