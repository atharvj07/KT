import java.util.*;
import java.io.*;
class Main {
  static int n, m;
  static char[][] arr;
  static int[][] vis;
  static int[] dx = {0, 0, -1, 1};
  static int[] dy = {-1, 1, 0, 0};
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    arr = new char[n][m];
    vis = new int[n][m];
    for (int i = 0; i < n; i++) {
      String s = sc.next();
      for (int j = 0; j < m; j++) {
        arr[i][j] = s.charAt(j);
      }
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (arr[i][j] == '#' && vis[i][j] == 0) {
          dfs(i, j);
          cnt++;
        }
      }
    }
    if (cnt > 1) {
      System.out.println(cnt - 1);
    } else {
      System.out.println(-1);
    }
  }
  public static void dfs(int x, int y) {
    vis[x][y] = 1;
    for (int i = 0; i < 4; i++) {
      int nx = x + dx[i];
      int ny = y + dy[i];
      if (nx >= 0 && nx < n && ny >= 0 && ny < m && arr[nx][ny] == '#' && vis[nx][ny] == 0) {
        dfs(nx, ny);
      }
    }
  }
}
