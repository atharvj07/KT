#include<iostream>
#include<vector>
#include<queue>
using namespace std;
int n, p[310], a[310][310];
vector<int> ans;
bool check(int x) {
    for (int i = 1; i <= n; i++) {
        if (a[x][i] == 1) {
            if (p[i] < p[x]) {
                return 1;
            }
        }
    }
    return 0;
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            char c;
            cin >> c;
            a[i][j] = c - '0';
        }
    }
    for (int i = 1; i <= n; i++) {
        if (check(i) == 0) {
            ans.push_back(p[i]);
        }
    }
    for (int i = 1; i <= n; i++) {
        if (check(i) == 1) {
            ans.push_back(p[i]);
        }
    }
    for (int x : ans) {
        cout << x << ' ';
    }
    return 0;
}
