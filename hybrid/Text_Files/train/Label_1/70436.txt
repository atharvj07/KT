import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    s = input().strip()
    ans = 0
    while s:
        i = s.find('0')
        if i == -1:
            i = s.find('1')
        ans += 1
        s = s[i+1:]
        if s:
            i = s.find('0') if '0' in s else len(s)
            s = s[i:]
    print(ans)
