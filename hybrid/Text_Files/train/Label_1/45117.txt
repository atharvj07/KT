import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    static ArrayList<Integer>[] add;
    static int[] pref;
    static char[] s;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String t = in.next();
        s = t.toCharArray();
        int q = in.nextInt();
        add = new ArrayList[100001];
        for (int i = 0; i < add.length; i++) add[i] = new ArrayList<>();
        pref = new int[t.length() + 1];
        int j = 0;
        for (int i = 1; i <= t.length(); i++) {
            if (i % 5000 == 0) j++;
            pref[i] = pref[i - 1];
            pref[i] += (t.charAt(i - 1) == 'a' ? add[j].size() : 0);
            add[j].add(i);
        }
        for (int i = 0; i < q; i++) {
            int type = in.nextInt();
            if (type == 1) {
                int pos = in.nextInt() - 1;
                char ch = in.next().charAt(0);
                if (s[pos] == 'a' && ch != 'a') {
                    int ind = lowerbound(pos, pos + 1);
                    for (int k = ind; k < add[ind].size(); k++) pref[add[ind].get(k)]--;
                } else if (s[pos] != 'a' && ch == 'a') {
                    int ind = lowerbound(pos, pos + 1);
                    for (int k = ind; k < add[ind].size(); k++) pref[add[ind].get(k)]++;
                }
                s[pos] = ch;
            } else {
                int l = in.nextInt() - 1;
                int r = in.nextInt();
                String str = in.next();
                int ind = lowerbound(l, r + 1);
                System.out.println(pref[r] - (l - 1 < 0 ? 0 : pref[l - 1]));
            }
        }
    }

    public static int lowerbound(int l, int r) {
        int res = 0;
        int a = l + 1;
        int b = s.length;
        while (a <= b) {
            int mid = (a + b) >> 1;
            if (s[mid - 1] >= s[l]) {
                a = mid + 1;
            } else {
                b = mid - 1;
                res = mid;
            }
        }
        return res;
    }
}
