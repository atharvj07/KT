import java.util.*;

class Main {
    static int n, m, d;
    static int[] a;
    static int[][] days;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt(); m = sc.nextInt(); d = sc.nextInt();
        a = new int[n]; for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        Arrays.sort(a);
        days = new int[n][2];
        int day = 0;
        for (int i = 0, j = 0; i < n; i++) {
            if (j == day) day++;
            while (j < n && a[j] - a[i] <= d) j++;
            days[i][0] = day;
            days[i][1] = a[i];
        }
        System.out.println(day);
        for (int i = 0; i < n; i++) System.out.print(days[i][0] + " ");

        sc.close();
    }
}
