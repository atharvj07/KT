import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class RomaGrind {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        Weapon[] weapons = new Weapon[n];
        for (int i = 0; i < n; i++) {
            weapons[i] = new Weapon(scanner.nextInt(), scanner.nextInt());
        }

        ArmorSet[] armorSets = new ArmorSet[m];
        for (int i = 0; i < m; i++) {
            armorSets[i] = new ArmorSet(scanner.nextInt(), scanner.nextInt());
        }

        Monster[] monsters = new Monster[p];
        for (int i = 0; i < p; i++) {
            monsters[i] = new Monster(scanner.nextInt(), scanner.nextInt(), scanner.nextInt());
        }

        Arrays.sort(weapons, Comparator.comparingInt(Weapon::getAttackModifier).reversed());
        Arrays.sort(armorSets, Comparator.comparingInt(ArmorSet::getDefenseModifier).reversed());
        Arrays.sort(monsters, Comparator.comparingInt(Monster::getCoins).reversed());

        int maxProfit = 0;
        for (Weapon weapon : weapons) {
            for (ArmorSet armorSet : armorSets) {
                int profit = 0;
                for (Monster monster : monsters) {
                    if (weapon.getAttackModifier() > monster.getDefense() && armorSet.getDefenseModifier() > monster.getAttack()) {
                        profit += monster.getCoins();
                    } else {
                        break;
                    }
                }
                profit -= weapon.getCost() + armorSet.getCost();
                maxProfit = Math.max(maxProfit, profit);
            }
        }

        System.out.println(maxProfit);
    }

    static class Weapon {
        private final int attackModifier;
        private final int cost;

        public Weapon(int attackModifier, int cost) {
            this.attackModifier = attackModifier;
            this.cost = cost;
        }

        public int getAttackModifier() {
            return attackModifier;
        }

        public int getCost() {
            return cost;
        }
    }

    static class ArmorSet {
        private final int defenseModifier;
        private final int cost;

        public ArmorSet(int defenseModifier, int cost) {
            this.defenseModifier = defenseModifier;
            this.cost = cost;
        }

        public int getDefenseModifier() {
            return defenseModifier;
        }

        public int getCost() {
            return cost;
        }
    }

    static class Monster {
        private final int defense;
        private final int attack;
        private final int coins;

        public Monster(int defense, int attack, int coins) {
            this.defense = defense;
            this.attack = attack;
            this.coins = coins;
        }

        public int getDefense() {
            return defense;
        }

        public int getAttack() {
            return attack;
        }

        public int getCoins() {
            return coins;
        }
    }
}
