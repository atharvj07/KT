#include <bits/stdc++.h>

using namespace std;

const int mod = 998244353;

int inv(int base) {
  int p = mod - 2;
  int ret = 1;
  while (p > 0) {
    if (p & 1) { ret = 1LL * ret * base % mod; }
    base = 1LL * base * base % mod;
    p >>= 1;
  }
  return ret;
}

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> d(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> d[i][j]; }
  }
  vector<int> sum(n + 1, 0);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { sum[d[i][j]]++; }
  }
  int nw = 1, de = n;
  for (int i = 1; i <= n + 1; i++) {
    nw = 1LL * nw * i % mod;
    de = 1LL * de * inv(i) % mod;
  }
  int ans = 1LL * nw * de % mod;
  for (int i = 2; i <= n; i++) {
    ans = 1LL * ans * i % mod;
  }
  int sum1 = 0;
  for (int k = 1; k <= n + 1; k++) {
    int cnt = sum[k];
    if (k != n + 1) { cnt -= sum[k + 1]; }
    sum1 = 1LL * sum1 + 1LL * cnt * k % mod;
    sum1 %= mod;
  }
  ans = 1LL * ans * inv(sum1) % mod;
  cout << ans << endl;
  return 0;
}
