#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += a[i];
        if (i + 1 < n) ans += a[i] * (n - 1 - i);
    }
    cout << ans << endl;
    return 0;
}
