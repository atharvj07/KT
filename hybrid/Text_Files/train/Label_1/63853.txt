from typing import List, Tuple

n = int(input())
students: List[Tuple[int, int]] = []
for _ in range(n):
    a, b = map(int, input().split())
    students.append((a, b))


students.sort(key=lambda x: x[0]-x[1])


ans = 0
pref = [0]*n
pref[0] = students[0][1]
for i in range(1, n):
    pref[i] = pref[i - 1] + students[i][1]

suff = [0]*n
suff[n - 1] = students[n - 1][0]
for i in range(n - 2, -1, -1):
    suff[i] = suff[i + 1] + students[i][0]

for i in range(n):
    left = 0
    right = n - 1
    if i - 1 >= 0:
        left = pref[i - 1]
    if i + 1 < n:
        right = suff[i + 1]
    ans += left*students[i][0] + right*students[i][1]

print(ans)
