#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(m);
    for (int i = 0; i < m; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0; i < m; i++) {
        if (i + n - 1 < m && a[i] == a[i + n - 1]) {
            ans++;
            i += n - 1;
        }
    }
    cout << ans << endl;
}
