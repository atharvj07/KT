#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n), ans(n);
    unordered_map<int, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        mp[a[i]]++;
    }
    for (int i = 0; i < n; i++) {
        ans[i] = mp[a[i]];
        if (mp.count(a[i] + 1)) {
            ans[i] += mp[a[i] + 1];
        }
        if (mp.count(a[i] - 1)) {
            ans[i] += mp[a[i] - 1];
        }
    }
    int res = 0;
    for (int i = 0; i < n; i++) {
        res = max(res, ans[i]);
    }
    cout << res << endl;
}
