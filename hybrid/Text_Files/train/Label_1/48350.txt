import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), c = in.nextInt();
    int[][] d = new int[c][c];
    for (int[] r : d)
      for (int i = 0; i < c; i++) r[i] = in.nextInt();
    int[][] g = new int[n][n];
    for (int[] r : g)
      for (int i = 0; i < n; i++) r[i] = in.nextInt();
    int[][][] dp = new int[n + 1][n + 1][c];
    for (int[][] a : dp)
      for (int[] r : a)
        Arrays.fill(r, Integer.MAX_VALUE);
    dp[n][n][0] = 0;
    for (int i = n - 1; i >= 0; i--)
      for (int j = n - 1; j >= 0; j--) {
        for (int k = 0; k < c; k++) {
          int cost = d[g[i][j] - 1][k];
          if (cost == 0) cost = d[k][g[i][j] - 1];
          dp[i][j][k] =
              Math.min(
                  dp[i][j][k],
                  Math.min(dp[i + 1][j][k], dp[i][j + 1][k]) + cost);
          if ((i + j) % 3 != (i + j + 1) % 3)
            dp[i][j][k] =
                Math.min(dp[i][j][k], dp[i + 1][j + 1][k] + cost);
        }
      }
    System.out.println(dp[0][0][0]);
  }
}
