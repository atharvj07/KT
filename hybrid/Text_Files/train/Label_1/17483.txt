#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = a[(n - 1) / 2];
  int i = (n - 1) / 2;
  while (k > 0 && i < n) {
    if (ans >= a[i]) break;
    ans++;
    k--;
    i++;
  }
  cout << ans << endl;
  return 0;
}
