import java.util.Scanner;

public class UnimodalArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        boolean isUnimodal = isUnimodalArray(arr);
        System.out.println(isUnimodal ? "YES" : "NO");
    }

    private static boolean isUnimodalArray(int[] arr) {
        int increasingIndex = 0;
        while (increasingIndex < arr.length - 1 && arr[increasingIndex] < arr[increasingIndex + 1]) {
            increasingIndex++;
        }

        int decreasingIndex = arr.length - 1;
        while (decreasingIndex > 0 && arr[decreasingIndex] < arr[decreasingIndex - 1]) {
            decreasingIndex--;
        }

        for (int i = increasingIndex + 1; i < decreasingIndex; i++) {
            if (arr[i] != arr[increasingIndex]) {
                return false;
            }
        }

        return true;
    }
}
