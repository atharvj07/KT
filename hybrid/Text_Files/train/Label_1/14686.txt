#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
int main() {
  ll n, m, k;
  cin >> n >> m >> k;
  vector<ll> a(k);
  for (auto& x : a) cin >> x;

  ll g = a[0];
  for (int i = 0; i < k; i++) {
    g = gcd(g, a[i]);
    if (!g) {
      cout << "NO" << endl;
      return 0;
    }
  }
  if (m % g != 0) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  return 0;
}
