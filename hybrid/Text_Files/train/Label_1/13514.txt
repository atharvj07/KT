#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;

    std::vector<int> permutation(n + 1);
    std::vector<bool> visited(n + 1);

    std::vector<int> leaders(m);
    for (int i = 0; i < m; i++) {
        std::cin >> leaders[i];
    }

    int current_leader = leaders[0];
    int current_position = current_leader;
    for (int i = 0; i < n; i++) {
        int steps = 0;
        while (!visited[current_position]) {
            steps++;
            visited[current_position] = true;
            current_position = (current_position + 1) % (n + 1);
            if (current_position == 0) {
                current_position = n;
            }
        }
        permutation[i + 1] = steps;
        current_position = (current_position + steps) % (n + 1);
        if (current_position == 0) {
            current_position = n;
        }
        current_leader = leaders[(i + 1) % m];
        while (permutation[current_leader] == 0) {
            current_leader = leaders[(i + 2) % m];
        }
    }

    for (int i = 1; i <= n; i++) {
        std::cout << permutation[i] << " ";
    }
    std::cout << std::endl;

    return 0;
}
