#include <bits/stdc++.h>
using namespace std;
const int md=1e6+7,MAXSERVERS=107;
typedef pair<int,int> pii;
int servers[MAXSERVERS];
priority_queue<pii,vector<pii>,greater<pii>> pq;//first:time when free,second:server
int main(){
    int n,q;
    cin>>n>>q;
    while(q--){
        int t,k,d;
        cin>>t>>k>>d;
        while(!pq.empty()&&pq.top().first<=t){
            servers[pq.top().second]=0;
            pq.pop();
        }
        int sum=0;
        for(int i=1;i<=k&&i<=n;i++){
            if(!servers[i]){
                pq.push({t+d,i});
                servers[i]=1;
                sum+=i;
            }
        }
        if(sum==0)cout<<-1<<'\n';
        else cout<<sum<<'\n';
    }
    return 0;
}
