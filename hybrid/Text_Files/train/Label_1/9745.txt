#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> votes(n);
    for (int& i : votes) cin >> i;
    sort(votes.begin(), votes.end(), greater<>());
    int ans = 0, sum = votes[0];
    for (int i = 1; i < n; i++) {
        if (sum > votes[i]) break;
        ans += votes[i] - sum + 1;
        sum = votes[i] + 1;
    }
    cout << ans << endl;
    return 0;
}
