#include <iostream>
void solve(){
  int n;
  std::cin>>n;
  if(n==1){
    std::cout<<1<<std::endl;
    return;
  }
  if(n&1){
    std::cout<<-1<<std::endl;
    return;
  }
  int mat[n][n];
  int k=1;
  for(int i=0; i<n; i++)
    for(int j=0; j<n; j++)
      mat[i][j] = 0;
  for(int j=n-1; j>=0; j--){
    for(int i=0; i<n; i++){
      if(!mat[i][j]){
        mat[i][j] = k++;
        mat[i][n-1-j] = k++;
      }
    }
  }
  for(int i=0; i<n; i++)
    for(int j=0; j<n; j++)
      std::cout<<mat[i][j]<<" ";
  std::cout<<std::endl;
}
int main(){
  int t;
  std::cin>>t;
  while(t--)
    solve();
}
