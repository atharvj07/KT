import sys
input = sys.stdin.readline

n, k1, k2 = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
a.sort(); b.sort(reverse=True)
ans = 0
for i in range(min(k1, k2)):
    ans += (a[i] - b[i])**2
    k1 -= 1; k2 -= 1
for i in range(k1):
    ans += (a[i] - a[i+1])**2
for i in range(k2):
    ans += (b[i] - b[i+1])**2
print(ans)
