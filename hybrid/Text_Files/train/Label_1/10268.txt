#include <bits/stdc++.h>
using namespace std;
bool board[101][101];
int32_t main() {
  int n;
  cin >> n;
  char c;
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      cin >> c;
      board[i][j] = (c == '#');
      if (!board[i][j]) continue;
      if (board[i - 1][j] && board[i][j - 1] && board[i + 1][j] && board[i][j + 1])
        board[i][j] = false, cnt++;
    }
  }
  puts(cnt * 5 == n * n && n * n % 5 == 0 ? "YES" : "NO");
}
