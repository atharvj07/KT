import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int na = input.nextInt();
        int nb = input.nextInt();
        int w = input.nextInt();
        long[] ma = new long[na];
        int[] ca = new int[na];
        for (int i = 0; i < na; i++) {
            ma[i] = input.nextInt();
            ca[i] = input.nextInt();
        }
        long[] mb = new long[nb];
        int[] cb = new int[nb];
        for (int i = 0; i < nb; i++) {
            mb[i] = input.nextInt();
            cb[i] = input.nextInt();
        }

        long ans = Long.MAX_VALUE;
        for (int i = 0; i < (1 << na); i++) {
            for (int j = 0; j < (1 << nb); j++) {
                long w1 = 0, w2 = 0, cmax = 0, cmin = Long.MAX_VALUE;
                for (int k = 0; k < na; k++) {
                    if ((i & (1 << k)) > 0) {
                        w1 += ma[k];
                        cmax = Math.max(cmax, ca[k]);
                        cmin = Math.min(cmin, ca[k]);
                    }
                }
                for (int k = 0; k < nb; k++) {
                    if ((j & (1 << k)) > 0) {
                        w2 += mb[k];
                        cmax = Math.max(cmax, cb[k]);
                        cmin = Math.min(cmin, cb[k]);
                    }
                }
                if (w1 <= w && w2 <= w) {
                    ans = Math.min(ans, Math.abs(w1 - w2) + Math.abs(cmax - cmin));
                }
            }
        }
        System.out.println(ans);
    }
}
