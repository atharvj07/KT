#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<std::string> grid(n);
    for (int i = 0; i < n; i++) {
        std::cin >> grid[i];
    }
    int max_clean = 0;
    for (int j = 0; j < n; j++) {
        int clean_count = 0;
        for (int i = 0; i < n; i++) {
            clean_count += grid[i][j] == '1';
        }
        max_clean = std::max(max_clean, clean_count);
    }
    std::cout << max_clean << std::endl;
    return 0;
}
