import sys

def solve(n: int, p: int, l: list[int], r: list[int]):
    # Get the mod inverse of p using Fermat's Little Theorem
    def get_mod_inv(a, m):
        return pow(a, m - 2, m)

    mod_inv_p = get_mod_inv(p, 10**9 + 7)
    
    dp = [[[0 for _ in range(2)] for _ in range(2)] for _ in range(n + 1)] 
    
    for i in range(n - 1, -1, -1):
        for j in range(2):
            for k in range(2):
                dp[i][j][k] = (
                    (dp[i + 1][j][k] * (r[i] - l[i] + 1) % (10**9 + 7)) + 
                    (dp[i + 1][(j * (l[i] % p == 0)) % 2][(k * (r[i] % p == 0)) % 2] * l[i] % (10**9 + 7)) + 
                    (dp[i + 1][(j * (r[i] % p == 0)) % 2][(k * (l[i] % p == 0)) % 2] * r[i] % (10**9 + 7))
                ) % (10**9 + 7)
    
    ans = dp[0][1][1] * mod_inv_p % (10**9 + 7)
    return ans

n, p = map(int, sys.stdin.readline().split())
l = []
r = []
for i in range(n):
    x, y = map(int, sys.stdin.readline().split())
    l.append(x)
    r.append(y)

ans = solve(n, p, l, r)
print(ans)
