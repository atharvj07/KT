#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
const ll mod = 1e9 + 7;
const char nl = '\n';

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ll n, a, b, k;
  cin >> n >> a >> b >> k;
  vi h(n);
  for (int i = 0; i < n; i++)
    cin >> h[i];
  k = min(k, n);
  sort(h.begin(), h.end());
  ll sum = 0;
  for (int i = 0; i < k; i++) {
    if (h[i] > a) {
      sum += h[i] / a * b <= h[i] % a;
      h[i] = h[i] % a;
    }
  }
  for (int i = 0; i < n; i++)
    sum += h[i] / a;
  cout << sum << nl;
  return 0;
}
