#include <bits/stdc++.h>
using namespace std;

#define MAXN 10005
typedef bool boolean;

char input[10];
int query[MAXN][2];
boolean ans[MAXN];
int n;

boolean solve(int l, int r) {
    if (l > r) {
        return false;
    }
    int mid = (l + r) / 2;
    for (int i = 1; i <= n; ++i) {
        if ((query[i][0] == 1 && query[i][1] > mid) ||
            (query[i][0] == 2 && query[i][1] < mid) ||
            (query[i][0] == 3 && query[i][1] >= mid) ||
            (query[i][0] == 4 && query[i][1] <= mid)) {
            if (ans[i] == 'N') {
                return false;
            }
        } else if ((ans[i] == 'Y')) {
            return false;
        }
    }
    cout << mid << endl;
    return true;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        scanf("%s", input);
        if (input[0] == '>') {
            query[i][0] = 1;
        } else if (input[0] == '<') {
            query[i][0] = 2;
        } else if (input[0] == '>') {
            query[i][0] = 3;
        } else if (input[0] == '<') {
            query[i][0] = 4;
        }
        cin >> query[i][1];
        cin >> ans[i];
    }
    boolean r = solve(-2e9, 2e9);
    if (!r) cout << "Impossible" << endl;
    return 0;
}
