import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        String[] tokens = reader.readLine().split(" ");
        List<Coordinate> coordinates = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            coordinates.add(new Coordinate(Integer.parseInt(tokens[i]), i));
        }
        tokens = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            coordinates.get(i).v = Integer.parseInt(tokens[i]);
        }
        coordinates.sort(Comparator.comparingInt(cr -> cr.x));
        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long diff = coordinates.get(j).x - coordinates.get(i).x;
                long d = Math.abs(coordinates.get(j).v - coordinates.get(i).v);
                if (d == 0) {
                    sum += diff;
                } else {
                    long t = diff / d;
                    if (t * d == diff) {
                        sum += diff;
                    } else {
                        sum +=  Math.min(t * d + coordinates.get(i).v, (t + 1) * d + coordinates.get(i).v) - coordinates.get(i).x;
                    }
                }
            }
        }
        System.out.println(sum);
    }

    static class Coordinate {
        int x;
        int v;
        int index;

        public Coordinate(int x, int index) {
            this.x = x;
            this.index = index;
        }
    }
}
