#include <iostream>
#include <map>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::map<int, int> cnt;
    int most = 0;
    int ans = 0;
    int last = 0;
    for (int i = 0; i < n; ++i) {
        int x;
        std::cin >> x;
        cnt[x]++;
        if (cnt[x] > most) {
            most = cnt[x];
            ans = x;
            last = i;
        } else if (cnt[x] == most) {
            last = std::min(last, i);
        }
    }
    std::cout << ans;
    return 0;
}
