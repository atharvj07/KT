import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    private static class Node {
        int value;
        List<Node> children;

        public Node(int value) {
            this.value = value;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int s = Integer.parseInt(line[1]);

        Node root = new Node(0);
        constructTree(root, reader, n);

        int minBurles = removeDangerousPaths(root, s, 0);
        System.out.println(minBurles);
    }

    private static int removeDangerousPaths(Node root, int s, int currentBurles) {
        if (root.value == s) {
            // Remove the current node and all its children to prevent assigning x to s
            return currentBurles + root.children.stream().mapToInt(n -> removeDangerousPaths(n, s, 0)).sum();
        } else {
            // Recursively remove dangerous paths from all children
            for (Node child : root.children) {
                currentBurles += removeDangerousPaths(child, s, currentBurles);
            }
            return currentBurles;
        }
    }

    private static void constructTree(Node root, BufferedReader reader, int n) throws IOException {
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            switch (line[0]) {
                case "set":
                    int value = Integer.parseInt(line[1]);
                    int burles = Integer.parseInt(line[2]);
                    addSetNode(root, value, burles);
                    break;
                case "if":
                    value = Integer.parseInt(line[1]);
                    addNode(root, value);
                    break;
                case "end":
                    break;
            }
        }
    }

    private static void addSetNode(Node root, int value, int burles) {
        Node node = new Node(value);
        node.children.add(new Node(0));
        node.children.add(new Node(burles));
        root.children.add(node);
    }

    private static void addNode(Node root, int value) {
        Node node = new Node(value);
        root.children.add(node);
    }
}
