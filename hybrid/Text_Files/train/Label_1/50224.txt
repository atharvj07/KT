import java.util.*;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; ++i) {
            a[i] = scanner.nextInt();
        }
        int[] w = new int[n];
        int length = 0;
        int k = 0;
        for (int i = 0; i < n; ++i) {
            int event = scanner.nextInt();
            if (event == -1) {
                k = 0;
                while (k < m && a[k] <= length) {
                    k++;
                }
                length -= k;
            } else {
                w[length++] = event;
            }
        }
        if (length == 0) {
            System.out.println("Poor stack!");
        } else {
            for (int i = 0; i < length; ++i) {
                System.out.print(w[i]);
            }
        }
    }
}
