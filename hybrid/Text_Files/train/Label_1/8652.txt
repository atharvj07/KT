#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

typedef long long ll;
const int MOD = 1000000007;

ll gcd(ll a, ll b) {
    while (b != 0) {
        a %= b;
        swap(a, b);
    }
    return a;
}

int main() {
    int n;
    cin >> n;
    vector<ll> arr(n);
    for (ll &x : arr) cin >> x;
    ll g = arr[0];
    for (int i = 1; i < n; i++) g = gcd(g, arr[i]);
    ll ans = 0;
    for (ll x : arr) ans += x / g;
    ans %= MOD;
    ans = ans * g % MOD;
    cout << ans << '\n';
}
