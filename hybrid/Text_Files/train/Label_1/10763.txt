#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
	int t;
	cin >> t;
	while (t--) {
		string s;
		cin >> s;
		int ans = 0;
		int last = -1;
		for (char c : s) {
			int d = c - '0';
			if (d != last) {
				ans += d;
				last = d;
			}
		}
		cout << ans << endl;
	}
	return 0;
}
