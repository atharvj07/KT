import sys

def solve(n: int) -> int:
    a = [3]
    while a[-1] < n:
        a.append(a[-1]*3)
    ans = 0
    for i in range(len(a)-1, -1, -1):
        if n >= a[i]:
            ans = ans + a[i]
            n = n - a[i]
    return ans

a = int(sys.stdin.readline())
for _ in range(a):
    n = int(sys.stdin.readline())
    print(solve(n))
