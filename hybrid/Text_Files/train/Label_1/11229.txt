#include <bits/stdc++.h>
#define ll long long

using namespace std;

void ADD(vector<ll>&arr, const vector<ll>&inc){
    ll n = arr.size();
    ll m = inc.size();
    ll denominator = m;
    ll numerator = 0;
    for(ll i=0; i<inc.size(); i++){
        if(i < m){
            numerator += arr[i];
        }
        numerator += inc[i];
    }
    ll average = (__int128)numerator / (__int128)denominator;
    printf("%3.6lf\n", (double)(average)/100000.0);
}

void REMOVE(vector<ll>&arr){
    arr.pop_back();
    ll n = arr.size();
    long double sum = 0;
    for(int i=0; i<arr.size(); i++){
        sum += arr[i];
    }
    long double average = sum / n;
    printf("%3.6lf\n", average);
}

void APPEND(vector<ll>&arr, ll val){
    arr.push_back(val);
    ll n = arr.size();
    long double sum = 0;
    for(int i=0; i<arr.size(); i++){
        sum += arr[i];
    }
    long double average = sum / n;
    printf("%3.6lf\n", average);
}

int main() {
    ll n;
    cin>>n;
    vector<ll>arr;
    arr.push_back(0);
    for(ll i=0; i<n; i++){
        ll t;
        cin>>t;
        if(t==1){
            ll a, x;
            cin>>a>>x;
            vector<ll>inc;
            for(ll j=0; j<a; j++){
                inc.push_back(x);
            }
            ADD(arr, inc);
        }
        else if(t==2){
            ll k;
            cin>>k;
            APPEND(arr, k);
        }
        else if(t==3){
            REMOVE(arr);
        }
    }
    return 0;
}
