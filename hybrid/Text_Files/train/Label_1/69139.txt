import sys
input = sys.stdin.readline
from collections import defaultdict, deque

n,k = map(int,input().split())
fav = defaultdict(list)
indeg = {}
for i in range(1,n+1):
    indeg[i] = 0
for i in range(k):
    a,b = map(int,input().split())
    fav[a].append(b)
    fav[b].append(a)
    indeg[a] += 1
    indeg[b] += 1
q = deque()
for i in range(1,n+1):
    if indeg[i] == 1:
        q.append(i)
ans = 0
while len(q)>0:
    a = q.popleft()
    ans += 1
    for b in fav[a]:
        indeg[b] -= 1
        if indeg[b] == 1:
            q.append(b)
print(k-ans)
