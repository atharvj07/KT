#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    int l = 0, r = -1;
    for (int i = 0; i < n; i++) {
        if (s[i] != 'a' && (r == -1 || s[i] < s[r])) {
            r = i;
        }
    }
    for (int i = r; i >= 0 && s[i] > s[l]; i--) {
        l = i;
    }
    for (int i = l; i <= r; i++) {
        if (s[i] == 'a') {
            s[i] = 'z';
        } else {
            s[i]--;
        }
    }
    cout << s << "\n";
    return 0;
}
