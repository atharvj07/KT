// Custom imports
import java.util.Scanner;

// Main class
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int c = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt() % c;
        }
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt() % c;
        }
        for (int i = 0; i < n - m + 1; i++) {
            for (int j = i; j < i + m; j++) {
                a[j] = (a[j] + b[j - i]) % c;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
