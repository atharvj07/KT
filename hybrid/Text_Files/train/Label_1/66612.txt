n,m=map(int,input().split())
G=[[]for _ in range(n+1)]
for i in range(m):
    u,v=map(int,input().split())
    G[u].append(v)
    G[v].append(u)
S=[[-1]*(n+1)for _ in range(n+1)]
for i in range(1,n+1):
    if S[i][i]==-1:
        S[i][i]=0
        Q=[i]
        while Q:
            j=Q.pop(0)
            for k in G[j]:
                if S[i][k]==-1:
                    S[i][k]=S[i][j]+1
                    Q.append(k)
Max=0
for i in range(1,n+1):
    for j in range(1,n+1):
        if S[i][j]>Max: Max=S[i][j]
print(Max*(Max-1))
