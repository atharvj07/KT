n, d = [int(i) for i in input().split()]
s = sorted([(int(i), j) for j, i in enumerate(input().split())], reverse=1)
a = [int(i) for i in input().split()]
a.sort(reverse=1)
p = 0
for i in range(len(a)):
  if s[d - 1][0] + a[i] == s[i][0]:
    p = s[i][1]
    break
  elif s[d - 1][0] + a[i] > s[i][0]:
    p = i + 1
    break
print(p)
