#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define mp(a,b) make_pair(a,b)
#define pb push_back
#define rep(i,a,b) for(int i=(a);i<(b);i++)
int n, ans;
vector<pair<ll,ll>> v;
bool cmp(pair<ll,ll> x, pair<ll,ll> y) { return x.second > y.second; }
int main() {
    cin >> n;
    rep(i,0,n) {
        ll a, b;
        cin >> a >> b;
        v.push_back(mp(a, b));
    }
    sort(v.begin(), v.end());
    for (auto x : v) {
        while (ans > 0 && v[ans - 1].first + v[ans - 1].second <= x.first) ans--;
        ans++;
    }
    cout << ans << endl;
    return 0;
}
