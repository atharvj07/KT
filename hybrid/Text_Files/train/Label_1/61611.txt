n, x = map(int, input().split())
a = list(map(int, input().split()))

total_balls = sum(a)
min_balls = min(a)
min_index = a.index(min_balls)

# Find the total rounds completed and adjust the balls in the boxes
rounds = total_balls // n
a = [balls - rounds for balls in a]

# Determine the starting box where Vasya began distributing the balls
start = (x - (total_balls % n) - 1) % n
if start < 0:
    start += n

# Distribute the remaining balls
remaining_balls = total_balls % n
for _ in range(remaining_balls):
    a[start] += 1
    start = (start + 1) % n

# Adjust the box from where balls were taken
a[min_index] += rounds * n - sum(a)

print(*a)
