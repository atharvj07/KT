#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    char gr[300][300];
    for (int i = 0; i < n; i++) cin >> gr[i];
    char d1 = gr[0][0], d2 = gr[0][n - 1];
    for (int i = 0; i < n; i++)
        if (gr[i][i] != d1 || gr[i][n - 1 - i] != d2) {
            cout << "NO";
            return 0;
        }
    char m = gr[n / 2][n / 2];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (i == n / 2 && j == n / 2) continue;
            else if (i == j || i + j == n - 1) continue;
            else if (gr[i][j] != m) {
                cout << "NO";
                return 0;
            }
    cout << "YES";
}
