#include <bits/stdc++.h>
using namespace std;
#define MAX_N 200005
#define ll long long
map<ll, pair<int, int>> sum;
int n, k;
int main() {
  cin >> k;
  while (k--) {
    int cnt;
    cin >> cnt;
    ll s = 0;
    for (int i = 0, x; i < cnt; i++) {
      cin >> x;
      s += x;
      sum[s].first++;
      sum[s].second = i;
    }
  }
  for (auto &kv : sum) {
    auto &p = kv.second;
    if (p.first == 1) continue;
    cout << "YES" << endl;
    cout << p.second + 1 << ' ' << p.first << endl;
    cout << sum[0].second + 1 << ' ' << 0 << endl;
    return 0;
  }
  cout << "NO" << endl;
  return 0;
}
