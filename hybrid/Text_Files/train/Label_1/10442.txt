#include <iostream>
#include <cstring>
#include <vector>
using namespace std;
const int N = 500;
int n, p[N], ans[N][N];

int dx[] = {1, 0, 1};
int dy[] = {0, 1, 1};

void rec(int x, int y, int cnt, int id) {
    ans[x][y] = id;
    for (int i = 0; i < 3; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (nx < n && ny < n && p[nx] == cnt + 1 && !ans[nx][ny]) {
            rec(nx, ny, cnt + 1, id);
        }
    }
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        memset(ans, 0, sizeof(ans));
        cin >> n;
        for (int i = 0; i < n; i++) cin >> p[i];
        int id = 0;
        for (int i = 0; i < n; i++) {
            if (!ans[i][i]) {
                rec(i, i, p[i], ++id);
            }
        }
        if (id != n)
            cout << -1 << endl;
        else {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) cout << ans[i][j] << " ";
                cout << endl;
            }
        }
    }
    return 0;
}
