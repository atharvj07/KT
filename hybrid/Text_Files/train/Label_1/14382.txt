#include <cstdio>
#include <vector>

int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &v[i]);
  }
  int cnt = 0;
  bool flg = 1;
  for (int i = 1; flg; i++) {
    flg = 0;
    for (int j = 1; j <= n; j++) {
      if (v[j] >= i && v[j] != -1) {
        cnt++;
        v[j] = -1;
        flg = 1;
        break;
      }
    }
  }
  printf("%d\n", cnt);
}
