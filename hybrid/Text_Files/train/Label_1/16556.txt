#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; ++i)
    cin >> a[i];
  sort(a.begin(), a.end());
  int ans = n;
  for (int i = 0; i < n - 1; ++i)
    if (a[i + 1] <= a[i] + k) {
      --ans;
      ++i;
    }
  cout << ans << endl;
  return 0;
}
