import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        char[] s = input.nextLine().toCharArray();
        int n = s.length;
        int[] ans = new int[n];
        int l = 0, r = n - 1;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'R') {
                ans[i] = ++r;
            } else {
                ans[i] = --l;
                cnt++;
            }
        }
        System.out.println(cnt);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
