import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    a = sorted(list(map(int, input().split())))
    if len(set(a)) == n:
        print("NO")
    else:
        if a[0] == a[1]:
            print("NO")
        else:
            b = a[::]
            for i in range(n):
                for j in range(i, n):
                    if b[i] == b[j]:
                        b[i], b[j] = b[j], b[i]
                        if b[i] != b[i-1]:
                            break
            if b[n-1] == b[n-2]:
                print("NO")
            else:
                print("YES")
                print(*b)

for _ in range(int(input())):
    solve()
