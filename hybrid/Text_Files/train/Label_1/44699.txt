import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        c -= b * (c / (a + b)); // b*7の倍数の部分を取り除く
        if (a >= c) System.out.println(1);
        else System.out.println((c - 1 + a) / a);
    }
}
