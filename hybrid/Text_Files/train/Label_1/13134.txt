#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    bool pt[30];
    while (n--) {
        int x;
        cin >> x;
        for (int i = 0;; i++) {
            if ((1 << i) > x) {
                break;
            }
            pt[i] |= x & (1 << i);
        }
    }
    int ans = 0;
    for (int i = 0; i < 30; i++) {
        if (pt[i]) {
            ans += pt[i] * (pt[i] - 1) / 2;
        }
    }
    cout << ans << endl;
    return 0;
}
