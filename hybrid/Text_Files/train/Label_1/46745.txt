import java.util.Scanner;

public class ArtemAndMasha {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int count = 0;
        int lastGift = 0;

        while (n > 0) {
            int gift = Math.min(n, lastGift + 1);
            n -= gift;
            lastGift = gift;
            count++;
        }

        System.out.println(count);
    }
}
