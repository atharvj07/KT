import sys

mod = 10 ** 9 + 7

for _ in range(int(input())):
    n, m, k = map(int, input().split())
    arr = [int(x) for x in input().split()]

    freq = [0] * (n + 1)
    for x in arr:
        freq[x] += 1

    dp = [[[[0 for l in range(k + 2)] for j in range(m + 1)] for i in range(n + 1)] for z in range(n + 1)]
    dp[n][m][n][0] = 1
    for i in range(n - 1, -1, -1):
        for j in range(m):
            for z in range(n + 1):
                for l in range(k + 1):
                    dp[i][j][z][l] += dp[i + 1][j][z][l]
                    dp[i][j][z][l] %= mod
                    if l == 0:
                        dp[i][j][z][0] += freq[i + 1]
                    elif i + 1 < z or l <= k:
                        dp[i][j][z][l] += freq[i + 1] * dp[i + 1][j + 1][i + 1][l - 1]
                    dp[i][j][z][l] %= mod

    print(dp[0][0][n][k])
