#include <iostream>
#include <vector>
#include <stack>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    stack<int> s;
    for (int i = 0; i < n; i++) {
        while (!s.empty() && s.top() <= a[i]) {
            s.pop();
        }
        if (!s.empty() && s.top() == a[i]) {
            s.pop();
        } else {
            s.push(a[i]);
            cout << a[i] << endl;
        }
    }
    return 0;
}
