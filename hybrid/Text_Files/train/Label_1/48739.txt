import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    int ans = 0;
    for (int d = -1000; d <= 1000; d++) {
      if (d == 0) {
        continue;
      }
      int count = 0;
      for (int x : a) {
        if ((double) x / d > 0) {
          count++;
        }
      }
      if (count >= (n + 1) / 2) {
        ans = d;
        break;
      }
    }
    System.out.println(ans);
  }
}
