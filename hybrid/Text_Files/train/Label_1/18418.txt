#include <iostream>

using namespace std;


int main() {
    int t;
    cin >> t;

    while (t--) {
        unsigned long long k;
        cin >> k;
        unsigned long long row = 0, col = 0;
        unsigned long long step = 0;

        while (k > step) {
            ++row;
            step += row;
        }

        if (k > step - row) {
            ++col;
            step += 2 * col + 1;
        } else
            col = step - k;

        cout << row << ' ' << col << endl;
    }

    return 0;
}
