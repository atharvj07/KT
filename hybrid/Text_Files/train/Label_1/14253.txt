#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string n;
    int s;
    cin >> n >> s;
    int sum = 0;
    for (char c : n) sum += c - '0';
    int ans = 0;
    while (sum > s) {
      int d = n.back() - '0' - 1;
      sum -= d;
      n.back()--;
      while (n.back() == '0' && n.size() > 1) {
        n.pop_back();
        sum -= 10;
        ans++;
      }
      ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
