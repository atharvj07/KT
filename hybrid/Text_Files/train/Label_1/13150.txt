#include <iostream>
using namespace std;

int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    double misha = max(3.0 * a / 10, (double)a - (double)a / 250 * c);
    double vasya = max(3.0 * b / 10, (double)b - (double)b / 250 * d);
    if (misha > vasya) {
        cout << "Misha" << endl;
    } else if (vasya > misha) {
        cout << "Vasya" << endl;
    } else {
        cout << "Tie" << endl;
    }
    return 0;
}
