import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] l = new long[n];
        long[] r = new long[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextLong();
            r[i] = sc.nextLong();
        }
        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += Math.max(0, r[i] - l[i + 1]);
        }
        System.out.println(ans);
    }
}
