#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int& x: a) cin >> x;
    sort(a.rbegin(), a.rend());
    int ans = 0, now = 0;
    for (int x: a) {
        if (x > now) ans++, now = x;
    }
    cout << ans << endl;
}
