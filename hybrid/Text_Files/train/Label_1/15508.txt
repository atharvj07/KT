#include<bits/stdc++.h>
using namespace std;
int n,m,a[25][25],b[25];
bool ok(){
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            if(a[i][j]!=j)return 0;
        }
    }
    return 1;
}
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            cin>>a[i][j];
            b[j]=a[i][j];
        }
        sort(b+1,b+m+1);
        for(int j=1;j<=m;j++){
            if(b[j]!=j)return cout<<"NO",0;
        }
    }
    sort(b+1,b+m+1);
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            if(a[i][j]!=b[j])return cout<<"NO",0;
        }
    }
    cout<<"YES";
    return 0;
}
