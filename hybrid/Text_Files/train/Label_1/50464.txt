import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class DynamicArrays {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int q = Integer.parseInt(line[1]);

        List<ArrayList<Integer>> arrays = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            arrays.add(new ArrayList<>());
        }

        for (int i = 0; i < q; i++) {
            line = reader.readLine().split(" ");
            int operation = Integer.parseInt(line[0]);
            int arrayIndex = Integer.parseInt(line[1]);

            switch (operation) {
                case 0:
                    int element = Integer.parseInt(line[2]);
                    arrays.get(arrayIndex).add(element);
                    break;
                case 1:
                    ArrayList<Integer> array = arrays.get(arrayIndex);
                    if (array.isEmpty()) {
                        System.out.println();
                    } else {
                        for (int j = 0; j < array.size(); j++) {
                            System.out.print(array.get(j) + " ");
                        }
                        System.out.println();
                    }
                    break;
                case 2:
                    arrays.get(arrayIndex).clear();
                    break;
            }
        }
    }
}
