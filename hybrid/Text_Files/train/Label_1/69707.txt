n = int(input())
grid = [list(input()) for _ in range(n)]
ans = 0
for col in range(n):
    dirty = 0
    for i in range(n):
        if grid[i][col] == '0':
            dirty += 1
    if dirty == n or dirty == 0:
        ans += 1
        for i in range(n):
            grid[i][col] = '1'
            grid[i][col] = '0' if grid[i][col] == '1' else '1'
print(ans)
