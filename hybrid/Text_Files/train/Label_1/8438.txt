#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  long long int w, l;
  std::cin >> w >> l;
  std::vector<long long int> a(w - 1);
  for (long long int i = 0; i < w - 1; i++) std::cin >> a[i];
  long long int ans = 0, pos = 0;
  while (pos < w) {
    if (pos + l >= w) {
      ans++;
      break;
    } else {
      pos += l;
      if (a[pos] > 0) {
        pos++;
        a[pos]--;
      } else {
        break;
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
