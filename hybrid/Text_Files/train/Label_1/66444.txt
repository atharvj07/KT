from collections import deque

def reachable_cells():
    n, m = map(int, input().split())
    r, c = map(int, input().split())
    x, y = map(int, input().split())
    maze = [input() for _ in range(n)]
    
    visited = [[False] * m for _ in range(n)]
    visited[r-1][c-1] = True
    
    q = deque([(r-1, c-1, 0, 0)])
    count = 1
    
    while q:
        i, j, left, right = q.popleft()
        
        for di, dj, dl, dr in ((0, 1, 0, 1), (0, -1, 1, 0), (1, 0, 0, 0), (-1, 0, 0, 0)):
            ni, nj, nleft, nright = i + di, j + dj, left + dl, right + dr
            
            if 0 <= ni < n and 0 <= nj < m and not visited[ni][nj] and maze[ni][nj] == '.':
                if nleft <= x and nright <= y:
                    visited[ni][nj] = True
                    q.append((ni, nj, nleft, nright))
                    count += 1
    
    return count

print(reachable_cells())
