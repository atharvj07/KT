#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> grid(n);
  for (int i = 0; i < n; i++) {
    cin >> grid[i];
  }

  int ans = -1;
  for (int i = 0; i < n; i++) {
    bool found = false;
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == 'G') {
        if (j == m - 1 || grid[i][j + 1] == 'S') {
          found = true;
          break;
        }
        cnt++;
      }
    }
    if (!found) {
      if (ans == -1) ans = cnt;
      else ans = min(ans, cnt);
    }
  }
  cout << ans << endl;
  return 0;
}
