import java.util.*;
import java.io.*;
public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int[] h = new int[n];
    for (int i = 0; i < n; i++) {
      h[i] = input.nextInt();
    }
    int min = Integer.MAX_VALUE;
    for (int i = 1; i < n; i++) {
      int j = i - 1;
      while (j >= 0 && h[j] >= h[i]) {
        j--;
      }
      if (j >= 0) {
        min = Math.min(min, i - j);
      }
    }
    System.out.println(min);
  }
}
