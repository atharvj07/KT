#include <algorithm>
#include <cstring>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<unsigned long long> b(n);
  for (int i = 0; i < n; i++) {
    std::cin >> b[i];
  }
  unsigned long long tmp = 0;
  std::vector<unsigned long long> ans;
  for (int i = n - 1; i >= 0; i--) {
    if (tmp > b[i]) {
      std::cout << "No" << std::endl;
      return 0;
    }
    ans.push_back(b[i] ^ tmp);
    tmp = b[i];
  }
  if (ans[ans.size() - 1] != 0) {
    std::cout << "No" << std::endl;
    return 0;
  }
  reverse(ans.begin(), ans.end());
  std::cout << "Yes" << std::endl;
  for (unsigned long long x : ans) std::cout << x << " ";
  std::cout << std::endl;
  return 0;
}
