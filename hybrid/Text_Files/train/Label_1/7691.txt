#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;
const double EPS = 1e-6;

int main() {
    int n, m;
    cin >> n >> m;

    vector<pair<int, int>> changes(m);
    for (int i = 0; i < m; i++) {
        cin >> changes[i].first >> changes[i].second;
    }

    vector<double> pref(n + 1);
    for (int i = 0; i < m; i++) {
        int x = changes[i].first, d = changes[i].second;
        pref[0] += (double)x;
        pref[1] += (double)d;
    }

    for (int i = 1; i <= n; i++) {
        pref[i] += pref[i - 1];
    }

    double max_avg = -1e9;
    for (int i = 1; i <= n; i++) {
        max_avg = max(max_avg, pref[i] / i);
    }

    cout << fixed << setprecision(9) << max_avg << endl;

    return 0;
}
