#include <bits/stdc++.h>

using namespace std;

#define MAX 100010
#define ll long long

int n, p;
char s[MAX];
ll dp[MAX][2][2]; // 0/1 - not a palindrome/palindrome; 0/1 - current char was changed/not
int cc[MAX];

inline ll get(char lc, char rc) {
    if(lc == rc) return 0;
    if(cc[rc] - cc[lc - 1] <= cc[rc] - cc[rc - 1]) return cc[rc] - cc[lc - 1];
    return cc[rc] - cc[rc - 1] + 1;
}

inline ll solve(int idx, bool pal, bool ch) {
    if(idx == n) return pal;
    ll &ret = dp[idx][pal][ch];
    if(ret != -1) return ret;
    ret = LLONG_MAX;
    if(!pal) {
        ret = min(solve(idx + 1, false, false), solve(idx + 1, false, true) + get(s[idx], s[idx]));
    }
    if(!ch) {
        ret = min(ret, solve(idx, true, false) + get(s[idx], s[p]));
        ret = min(ret, solve(idx, false, true) + get(s[idx], s[idx]) + get(s[idx], s[p]));
    }
    return ret;
}

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    cin >> n >> p;
    cin >> s;
    for(int i = 0; i < 26; ++i) cc[i + 'a'] = 0;
    for(int i = 0; i < n; ++i) ++cc[s[i]];
    memset(dp, -1, sizeof(dp));
    cout << solve(0, true, false) << "\n";
    return 0;
}
