#include <bits/stdc++.h>
#define mod 1000000007
using namespace std;
typedef long long ll;
int main() { ll n, a[200005], sum[200005], prv; cin >> n; for (ll i = 1; i <= n; ++i) cin >> a[i]; for (ll i = 1, j = 0; i <= n; ++i) { sum[i] = prv; if (j % 2 == 0) prv = ((prv + a[i]) % mod + mod) % mod; else prv = ((prv - a[i]) % mod + mod) % mod; ++j; } cout << sum[n] << "\n"; return 0; }
