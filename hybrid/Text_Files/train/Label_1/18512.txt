#include <bits/stdc++.h>
#define ll long long
using namespace std;
struct edge {
    ll v, u;
    bool operator<(const edge &other) { return v != other.v ? v < other.v : u < other.u; }
};
bool dead[200001], calc[200001];
vector<edge> edges[200001];
vector<ll> no_frnd;
void process(ll n) {
    vector<ll> tmp;
    for (const auto &i : no_frnd) if (!dead[i]) tmp.push_back(i);
    no_frnd.swap(tmp);
    memset(dead, 0, sizeof(dead));
    for (const auto &i : no_frnd) {
        for (const auto &j : edges[i]) {
            if (!dead[j.u]) {
                dead[j.v] = 1;
                break;
            }
        }
    }
}
int main() {
    memset(calc, 0, sizeof(calc));
    ll n, m, q;
    cin >> n >> m;
    while (m--) {
        ll u, v;
        cin >> u >> v;
        edges[u].push_back({v, u});
        edges[v].push_back({u, v});
    }
    for (ll i = 1; i <= n; i++) {
        if (edges[i].empty()) no_frnd.push_back(i);
    }
    cin >> q;
    while (q--) {
        ll tp;
        cin >> tp;
        if (tp == 3) {
            process(n);
            ll ans = 0;
            for (ll i = 1; i <= n; i++) { if (!dead[i]) ans++; }
            cout << ans << endl;
        } else {
            ll u, v;
            cin >> u >> v;
            if (tp == 1) {
                edges[u].push_back({v, u});
                edges[v].push_back({u, v});
            } else {
                for (ll i = 0; i < edges[u].size(); i++) {
                    if (edges[u][i].v == v) { edges[u].erase(edges[u].begin() + i); }
                }
                for (ll i = 0; i < edges[v].size(); i++) {
                    if (edges[v][i].v == u) { edges[v].erase(edges[v].begin() + i); }
                }
            }
            memset(dead, 0, sizeof(dead));
            process(n);
            for (const auto &i : no_frnd) {
                if (!calc[i]) {
                    calc[i] = 1;
                    ll c = 0;
                    for (const auto &j : edges[i]) {
                        if (!dead[j.u]) c++;
                    }
                    if (c == edges[i].size()) no_frnd.push_back(i);
                }
            }
        }
    }
}
