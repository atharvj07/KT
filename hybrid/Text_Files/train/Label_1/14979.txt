#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> pre, dp;

int main() {
  int n;
  cin >> n;
  pre.resize(n + 1);
  dp.resize(n + 1);
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    pre[i + 1] = pre[i] + a;
  }
  dp[n] = 0;
  for (int i = n - 1; i >= 1; i--) {
    for (int j = i; j <= n; j++) {
      dp[i] = max(dp[i], pre[j] - pre[i - 1] - dp[j]);
    }
  }
  cout << dp[1] << "\n";
  return 0;
}
