import math
x1,y1,r1 = map(int, input().split())
x2,y2,r2 = map(int, input().split())
d = math.sqrt((x1-x2)**2+(y1-y2)**2)
if d>r1+r2:
    ans = 0
elif d<=abs(r1-r2):
    ans = min(r1,r2)**2*math.pi
else:
    a = (r1**2-r2**2+d**2)/(2*d)
    h = math.sqrt(r1**2-a**2)
    ans = (a**2*math.acos((r1-a)/r1) + (r2-a)**2*math.acos((r2-a)/r2) - a*h)
print(ans)
