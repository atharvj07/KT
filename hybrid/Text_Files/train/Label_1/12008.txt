#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
  int n, m, t;
  cin >> n >> m;
  vector<int> d(n), a(m), indeg(m);
  priority_queue<int, vector<int>, greater<int>> q;
  vector<vector<int>> g(m);
  for (int i = 0; i < n; i++) {
    cin >> d[i];
    d[i]--;  // to index from 0
    if (d[i] != -1) {
      indeg[d[i]]++;
      g[d[i]].push_back(i);
    }
  }
  for (int i = 0; i < m; i++) {
    cin >> a[i];
    if (!indeg[i]) q.push(i);
  }
  int ans = 0;
  while (!q.empty()) {
    t = q.top();
    q.pop();
    ans += a[t];
    for (int u : g[t]) {
      if (--indeg[d[u]] == 0) q.push(d[u]);
    }
  }
  if (ans > n)
    cout << -1 << endl;
  else
    cout << ans << endl;
  return 0;
}
