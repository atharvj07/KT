#include <iostream>
#include <vector>
using namespace std;
const int mod = 1e9 + 7;
vector<int>dp(60, 0);
int32_t main() {
    int n; cin >> n;
    dp[1] = 6;
    for (int i = 2; i <= n; ++i) {
        dp[i] = (4ll * dp[i - 1] % mod * dp[i - 1] % mod) % mod;
    }
    cout << dp[n] << endl;
    return 0;
}
