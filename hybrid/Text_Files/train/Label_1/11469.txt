#include<bits/stdc++.h>
using namespace std;
int main(){
    int x,y;
    cin>>x>>y;
    long long q=sqrt(y);
    long long ans=0;
    for(int i=1;i<=q;++i){
        if(y%i==0){
            if(i%2==0||i%3==0)continue;
            if(i>=x&&y/i>=x){ans+=((y/i-x)/i+1);}
            if((y/i)>=x&&i>=x){ans+=((y/i-x)/i+1);}
        }
    }
    cout<<ans;
    return 0;
}
