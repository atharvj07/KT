import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        Arrays.sort(x);
        int m = 1;
        int[] ans = new int[n];
        ans[0] = x[0];
        for (int i = 1; i < n; i++) {
            if ((x[i] ^ x[i - 1]) % 2 == 0) {
                ans[m++] = x[i];
            }
        }
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
