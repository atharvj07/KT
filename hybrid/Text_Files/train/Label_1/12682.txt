#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int a, b, c;
    cin >> a >> b >> c;
    vector<int> arr = {a, b, c};
    sort(arr.begin(), arr.end());
    int ans = max(arr[0] * arr[1] * arr[2], arr[0] + arr[1] + arr[2]);
    cout << ans << endl;
    return 0;
}
