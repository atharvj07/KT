import sys
input=sys.stdin.readline
n=int(input())
a=[[]for _ in range(n)]
for _ in range(n-1):
    x,y=map(int,input().split())
    a[x-1].append(y-1)
    a[y-1].append(x-1)
def dfs1(i,p):
    for x in a[i]:
        if x==p:continue
        dfs1(x,i)
        f[i]=max(f[i],f[x]+1)
def dfs2(i,p):
    mf=ms=0
    for x in a[i]:
        if x==p:continue
        if f[i]==f[x]+1:
            mf=x
        else:
            ms=max(ms,f[x]+1)
    for x in a[i]:
        if x==p:continue
        if x==mf:dfs2(x,i)
        else:g[i]=max(g[i],ms+f[x]+1)
f=[0]*n
dfs1(0,-1)
g=[0]*n
dfs2(0,-1)
print(max(f+g))
