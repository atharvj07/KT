#include <iostream>
#include <vector>
#include <bitset>

int main() {
    int n; std::cin >> n;
    std::vector<std::bitset<4>> rows(n);
    for (auto& row : rows) {
        char c;
        for (int i = 0; i < n / 4; ++i) {
            std::cin >> c;
            row |= std::bitset<4>(c) << (i * 4);
        }
    }

    bool can_compress = true;
    for (int x = 2; x * x <= n && can_compress; ++x) {
        if (n % x || x == n) continue;
        for (int i = 0; i + x <= n && can_compress; ++i) {
            for (int j = 0; j + x <= n && can_compress; ++j) {
                auto ref = rows[i] >> (4 * j);
                for (int a = 0; a + x <= n && can_compress; ++a) {
                    for (int b = 0; b + x <= n && can_compress; ++b) {
                        if ((rows[i + a] >> (4 * b)) != ref) can_compress = false;
                    }
                }
            }
        }
    }

    std::cout << (can_compress ? n / 2 : 1) << std::endl;
}
