import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      long n = sc.nextInt();
      long k = sc.nextInt();
      System.out.println((k - 1) * n + k);
    }
    sc.close();
  }
}
