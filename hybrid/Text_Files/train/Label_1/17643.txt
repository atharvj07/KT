#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> k(n);
  for (int i = 0; i < n; ++i) std::cin >> k[i];
  std::vector<int> v(n, -1), l(n, -1);
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (k[i] == 0) {
      ++ans;
    } else {
      ++ans;
      int pv = -1, pl = k[i];
      for (int j = i + 1; j < n; ++j) {
        if (pv == -1 && k[j] > k[i] && k[j] <= pl) {
          pv = i;
          pl = k[j];
          v[j] = i;
        } else if (pv != -1 && k[j] > pv && k[j] <= pl) {
          pl = k[j];
          v[j] = pv;
        }
        if (l[j] != -1 && k[l[j]] < k[i]) {
          ans--;
          break;
        }
        if (j == (i + 1)) l[i] = j;
        if (v[j] != -1) l[v[j]] = j;
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
