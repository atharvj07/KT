#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  long long ans = 0, a[n], sum = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  long long s = 0;
  for (int i = 0; i < k - 1; i++) {
    s = s + (a[i] * (k - i - 1));
    s = s - (a[i + n - k] * (n - i - k));
  }
  ans = ans + s;
  for (int i = k - 1; i < n; i++) {
    s = s + (a[i] * (k - 1));
    s = s - (a[i - k + 1]);
    ans = ans + s;
  }
  cout << ans << endl;
  return 0;
}
