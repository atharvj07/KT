import java.util.*;
import java.lang.*;

class Solution {
    public static void main(String[] args) throws java.lang.Exception {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int j = i + m - 1;
            while (j < n && a[j] - a[i] < m) {
                j++;
            }
            if (j - i >= k) {
                ans = Math.max(ans, j - i - k + 1);
            }
        }
        System.out.println(ans);
    }
}
