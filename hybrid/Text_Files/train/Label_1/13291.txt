#include <bits/stdc++.h>

using namespace std;
using ll = long long;
const int maxn = 1e5 + 10;
int n;
int a[maxn];
int pfx[maxn];

int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int j = 1; j <= 100000; j++) {
    for (int i = 1; i <= n; i++) {
      if (a[i] == j) pfx[i] = pfx[i - 1] + 1;
      else
        pfx[i] = pfx[i - 1];
    }
    int x = 0;
    for (int i = 1; i <= n; i++) {
      int l = i, r = n, ans = -1;
      while (l <= r) {
        int m = (l + r) / 2;
        int cnt = pfx[m] - pfx[i - 1];
        if (cnt * 2 <= m - i + 1)
          ans = m, l = m + 1;
        else
          r = m - 1;
      }
      x = max(x, ans - i + 1);
    }
    if (x >= 3) { cout << x << endl; return 0; }
  }
  cout << -1 << '\n';
  return 0;
}
