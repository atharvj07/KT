n = int(input())
a = list(map(int, input().split()))
b = [0] * n
for i in range(1, n):
    a[i] += a[i - 1]
for i in range(n - 1, -1, -1):
    if a[i] < 25:
        print("NO")
        exit()
    b[i] = a[i] - 25
    for j in range(i + 1, n):
        if b[j] < b[i]:
            print("NO")
            exit()
print("YES")
