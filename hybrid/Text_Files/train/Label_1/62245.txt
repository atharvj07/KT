n,k=map(int,input().split())
a=[int(i) for i in input().split()]
x,y=0,0
ans=[]
for i in range(n):
    if i==0:
        ans.append(a[i])
        if a[i]==1:y+=1
        else:x+=1
        continue
    if k<=0:
        ans.append(0)
        x+=1
        continue
    if a[i]==1:y+=1
    else:
        if y+1<=k:
            ans.append(1)
            y+=1
            k-=1
        else:
            ans.append(0)
            x+=1
print(x+y)
print(*ans)
