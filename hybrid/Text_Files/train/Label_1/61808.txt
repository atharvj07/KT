from sys import stdin
from math import sqrt, gcd

n = int(stdin.readline())
arr = [int(i) for i in stdin.readline().split()]
mini = min(arr)
if mini == 0:
    print(-1)
else:
    ans = 0
    for i in arr:
        if i % mini > 0:
            x = mini - i % mini
            ans += x // gcd(mini, x)
    print(ans)
