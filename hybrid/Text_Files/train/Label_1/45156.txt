import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int r = input.nextInt();
        int c = input.nextInt();
        char[][] map = new char[r][];
        int sr = 0, sc = 0;
        for (int i = 0; i < r; i++) {
            map[i] = input.next().toCharArray();
            for (int j = 0; j < c; j++) {
                if (map[i][j] == 'S') {
                    sr = i;
                    sc = j;
                }
            }
        }
        int[][] dist = new int[r][c];
        for (int i = 0; i < r; i++) {
            Arrays.fill(dist[i], -1);
        }
        Queue<int[]> queue = new LinkedList<>();
        queue.add(new int[] {sr, sc, 0});
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int x = cur[0];
            int y = cur[1];
            int d = cur[2];
            if (dist[x][y] != -1) continue;
            dist[x][y] = d;
            if (map[x][y] == 'E') {
                System.out.println(d);
                return;
            }
            if (x > 0 && map[x - 1][y] != 'T' && dist[x - 1][y] == -1) {
                queue.add(new int[] {x - 1, y, d});
            }
            if (x < r - 1 && map[x + 1][y] != 'T' && dist[x + 1][y] == -1) {
                queue.add(new int[] {x + 1, y, d});
            }
            if (y > 0 && map[x][y - 1] != 'T' && dist[x][y - 1] == -1) {
                queue.add(new int[] {x, y - 1, d});
            }
            if (y < c - 1 && map[x][y + 1] != 'T' && dist[x][y + 1] == -1) {
                queue.add(new int[] {x, y + 1, d});
            }
        }
        System.out.println(-1);
    }
}
