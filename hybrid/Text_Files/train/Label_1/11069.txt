#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  if (a[0] || !a[n - 1]) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    cout << '(';
    for (int i = 0; i + 1 < n; i++) {
      cout << a[i] << "->";
      if (i == 0 || !a[i]) {
        cout << '(';
      }
    }
    for (int i = n - 1; i >= 0; i--) {
      if (i == 0 || !a[i]) {
        cout << ')';
      }
      cout << a[i];
      if (i != 0) {
        cout << "->";
      }
    }
    cout << endl;
  }
  return 0;
}
