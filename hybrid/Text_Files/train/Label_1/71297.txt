from math import gcd, lcm

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    ans = list(map(int, input().split()))

    for i in range(n):
        lst = []
        for j in range(n):
            if i != j and gcd(a[i], a[j]) * lcm(a[i], a[j]) == a[i] * a[j]:
                lst.append(j)
        ans.append(len(lst) + 1)
    q = int(input())
    for i in map(int, input().split()):
        print(ans[i])
