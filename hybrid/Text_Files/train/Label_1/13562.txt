#include <vector>
#include <iostream>

using namespace std;

int main() {
  long long q, x, ans = 0, last = -1;
  cin >> q >> x;
  while (q--) {
    long long y;
    cin >> y;
    if (~last) ans = max(ans, y / x + 1);
    if (y % x == ans) ans++;
    if (y % x <= last) ans++;
    last = y % x;
    cout << ans << endl;
  }
  return 0;
}
