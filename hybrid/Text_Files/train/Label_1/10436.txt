#include <iostream>
#include <algorithm>
#include <vector>

long long n, k, ans;
std::vector<long long> a;

int main() {
  std::cin >> n >> k;
  a.resize(n);
  for (long long i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  std::sort(a.begin(), a.end());
  for (long long i = 0; i < n; ++i) {
    if (a[i] * ans > k) {
      break;
    }
    ++ans;
  }
  std::cout << ans << std::endl;
  return 0;
}
