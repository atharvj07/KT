import sys

readline = lambda: sys.stdin.readline().strip()
readline_ints = lambda: [int(i) for i in readline().split()]

n = int(readline())
a, b = [0] * n, [0] * n
for i in range(n):
    ai, bi = readline_ints()
    a[i], b[i] = ai, bi

ok = [True] * (2 * n + 3)
for x in a + b:
    ok[x] = False

inc = 0
for x in a:
    if ok[x - 1]:
        inc += 1
    ok[x] = True

dec = 0
for x in b:
    if ok[x + 1]:
        dec += 1
    ok[x] = True

if inc + dec > n:
    print(-1)
else:
    print(inc + dec)
