#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k; cin >> n >> k;
    if (!k) return cout << 0, 0;
    string s; cin >> s;
    static array<bool, 26> o;
    for (int i = 0; i < k; ++i) { char c; cin >> c; o[c - 'a'] = true; }
    int ans = k;
    for (char c : s) {
        if (!o[c - 'a']) ans += k;
        else ans += k - 1;
    }
    cout << ans;
}
