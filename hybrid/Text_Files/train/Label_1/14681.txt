#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

signed main() {
    int n, p, m; cin >> n >> p >> m;
    vector<pair<int, long long>> v(n);
    for (auto& x : v) { cin >> x.first >> x.second; }
    long long ans = 0;
    long long rem = 0;
    for (int i = 1; i <= m; i++) {
        rem -= p;
        while (n > 0 && v.front().first <= i) {
            rem += v.front().second;
            v.erase(v.begin());
            n--;
        } 
        if (rem < 0) ans++;
    }
    cout << ans << endl;
    return 0;
}
