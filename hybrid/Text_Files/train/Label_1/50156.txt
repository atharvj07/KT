import java.util.Scanner;

public class SmallestIntegerWithNDivisors {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int smallestInteger = n;
        for (int i = 2; i * i <= n; i++) {
            int count = 0;
            while (n % i == 0) {
                n /= i;
                count++;
            }
            if (count % 2 == 1) {
                smallestInteger *= i;
            }
        }
        if (n > 1) {
            smallestInteger *= n;
        }

        System.out.println(smallestInteger);
    }
}
