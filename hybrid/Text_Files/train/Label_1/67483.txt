n, m = map(int, input().split())
a = [input() for i in range(n)]
f = lambda l: l.index('D')
g = lambda l: l.index('A')
ans = 0
for i in range(n):
    for j in range(m):
        if a[i][j] == 'D':
            for k in range(i, n):
                if a[k][j] != a[(i + k) % n][j]: break
                for l in range(j, m):
                    if a[(i + k) % n][l] != a[(i + k) % n][(j + l) % m]: break
                    if f(a[k]) == j and g(a[(i + k) % n]) == l:
                        for t in range(1, k + 1):
                            if a[(i + t) % n][j] != a[(i + t) % n][(j + t) % m]: break
                        else:
                            ans = max(ans, (1 + k) // 4)
print(-~ans or "Poor Inna!" if ans else "Poor Dima!")
