#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while(t--) {
        long long n;
        cin >> n;
        if(n % 2 == 0) {
            cout << "YES" << endl;
            int a = 2, b = 2, c = n / 4;
            cout << a << " " << b << " " << c << endl;
        } else if(n % 4 == 0) {
            cout << "YES" << endl;
            int a = 2, b = 4, c = n / 8;
            cout << a << " " << b << " " << c << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
