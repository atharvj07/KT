  int n;

  vector<int> a(n), b(n);
  for (int &x : a) cin >> x;
  for (int &x : b) cin >> x;
  int m;
  cin >> m;
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int pos = lower_bound(b.begin(), b.end(), a[i]) - b.begin();
    ans += (long long)(i - pos) * (n - i - 1);
    ans += (long long)(n - pos - 1) * (i - 0);
  }
  cout << ans % m << endl;
}
