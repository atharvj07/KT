#include<bits/stdc++.h>

#define ll long long
#define vi vector<ll>
#define pii pair<ll, ll>
#define pb push_back

using namespace std;

const ll inf = 1e18;

int main() {
    ll n, x;
    cin >> n >> x;

    vi arr(n);

    for (ll i = 0; i < n; i++)
        cin >> arr[i];

    ll mex = *max_element(arr.begin(), arr.end());

    for (ll i = 0; i < n; i++)
        mex &= arr[i];

    if (mex == 0) {
        cout << -1 << endl;
        return 0;
    }

    ll ans = 0;

    for (ll i = 0; i < n; i++)
        if (arr[i] != mex)
            ans++;

    cout << ans << endl;

    return 0;
}
