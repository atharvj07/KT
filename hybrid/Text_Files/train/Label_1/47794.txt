import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = Integer.parseInt(in.nextLine());
        List<int[]> tiles = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int[] tile = new int[4];
            tile[0] = in.nextInt();
            tile[1] = in.nextInt();
            tile[2] = in.nextInt();
            tile[3] = in.nextInt();
            tiles.add(tile);
        }
        Collections.sort(tiles, (a, b) -> a[2] - b[2]);
        int res = 0;
        int last = -1;
        for (int[] tile : tiles) {
            if (tile[2] > last) {
                res++;
                last = tile[3];
            }
        }
        System.out.println(res);
    }
}
