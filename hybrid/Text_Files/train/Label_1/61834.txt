from collections import defaultdict
from sys import stdin, stdout

readl = lambda: stdin.readline()[:-1]

s = readl()
t = readl()

d = defaultdict(set)

for c in t:
    d[c].add(0)
for i, c in enumerate(s, 1):
    d[c].add(i)

ans = []

for c in d.keys():
    if len(d[c]) == 2:
        ans.append(c)
        i, j = sorted(d[c])
        s = s[:i] + t + s[i:]
        break
s = s[j:]

for c in d.keys():
    if len(d[c]) == 1:
        ans.append(c)

print(''.join(ans), end='')
stdout.write(s)
