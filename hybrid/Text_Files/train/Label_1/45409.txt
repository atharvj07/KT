import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        String t = scanner.nextLine();
        int[] cnt1 = new int[26];
        int[] cnt2 = new int[26];
        int p = -1;
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == '0') {
                cnt1[t.charAt(i) - 'a']++;
            } else {
                cnt2[t.charAt(i) - 'a']++;
            }
            if (cnt1[t.charAt(i) - 'a'] > 0 && cnt2[t.charAt(i) - 'a'] > 0) {
                p = t.charAt(i) - 'a';
                break;
            }
        }
        if (p == -1) {
            System.out.println(0);
        } else {
            int ans = 0;
            for (int i = 0; i < 26; i++) {
                if (i != p && cnt1[i] > 0) {
                    ans += cnt1[i];
                }
                if (i != p && cnt2[i] > 0) {
                    ans += cnt2[i];
                }
            }
            System.out.println(ans);
        }
    }
}
