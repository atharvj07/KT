#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int n, m, ans, u[200005], v[200005], f[100005], g[100005], id[100005], deg[100005], cnt, sz[100005];
vector<int> a[100005];
bool cmp(int x, int y) {return u[x] < u[y];}
int find(int x) {return f[x] == x ? x : f[x] = find(f[x]);}
int main() {
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= n; i++) f[i] = i;
  for (int i = 1; i <= m; i++) {
    scanf("%d %d", &u[i], &v[i]);
    if (u[i] > v[i]) swap(u[i], v[i]);
    if (find(u[i]) != find(v[i])) {
      f[find(u[i])] = find(v[i]);
      g[u[i]] |= (1 << i);
      g[v[i]] |= (1 << i);
    }
  }
  for (int i = 1; i <= n; i++) {
    id[i] = find(i);
    a[id[i]].push_back(i);
  }
  for (int i = 1; i <= n; i++) {
    if (a[i].size() == 0) continue;
    int x = *a[i].begin(), y = *(a[i].begin() + 1 % a[i].size());
    if (g[x] == g[y]) {
      sz[i] = a[i].size();
      for (int j = 1; j < a[i].size(); j++)
        if ((g[a[i][j]] ^ g[x]) == (g[a[i][(j - 1 + a[i].size()) % a[i].size()]] ^ g[x]))
          sz[i]++;
      continue;
    }
    deg[i] = 2;
    for (int j = 1; j < a[i].size(); j++) {
      if (g[a[i][j]] == g[x] || g[a[i][j]] == g[y]) continue;
      if ((g[a[i][j]] ^ g[x]) != (g[a[i][(j - 1 + a[i].size()) % a[i].size()]] ^ g[x])) deg[i]++;
    }
    if (deg[i] <= 2) {
      if (deg[i] == 2) {
        int c = 0;
        for (int j = 1; j < a[i].size(); j++)
          if ((g[a[i][j]] ^ g[x]) != (g[a[i][(j - 1 + a[i].size()) % a[i].size()]] ^ g[x])) c = j;
        sz[i] = c + a[i].size() - 1 - c;
      } else
        sz[i] = a[i].size();
      continue;
    }
    sz[i] = -1;
  }
  for (int i = 1; i <= n; i++)
    if (sz[id[i]] != -1) {
      ans += sz[id[i]];
      cnt++;
    }
  printf("%s\n", ans == n && cnt == 1 ? "Yes" : "No");
}
