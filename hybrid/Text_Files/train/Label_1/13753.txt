#include <iostream>
using namespace std;

int main() {
    int l[5] = {0};
    for (int i = 0; i < 4; ++i) {
        string s;
        getline(cin, s);
        l[i + 1] = s.length();
    }
    int mn = 1e9, mx = 0, sum = 0;
    for (int i = 1; i < 5; ++i) sum += l[i];
    for (int i = 1; i < 5; ++i) {
        if (l[i] * 2 < sum - l[i]) mx = i;
        if (l[i] * 2 > sum - l[i]) mn = i;
    }
    if (mn == mx) cout << 'C';
    else if (mn) cout << char('A' + mn - 1);
    else cout << char('A' + mx - 1);
    cout << endl;
    return 0;
}
