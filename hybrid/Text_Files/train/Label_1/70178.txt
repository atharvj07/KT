n,m=map(int,input().split())
a=[list(map(int,input().split()))for i in range(n)]
ans=0
for i in range(n):
    for j in range(m):
        for k in range(i,n):
            for l in range(m):
                if(i==k and j>l or j==l and i<k):continue
                if(a[i][j]==a[k][l]):ans+=1
print(ans)
