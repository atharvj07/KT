from math import sqrt, atan2, cos, sin
import sys

input = sys.stdin.readline

x1, y1, x2, y2 = map(int, input().split())
v, t = map(int, input().split())
v1x, v1y = map(int, input().split())
v2x, v2y = map(int, input().split())

def length(v):
    return sqrt(v[0] ** 2 + v[1] ** 2)

v1 = length((v1x, v1y))
v2 = length((v2x, v2y))

angle1 = atan2(v1y, v1x)
angle2 = atan2(v2y, v2x)

if v > v1 and v > v2:
    dist = sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2)
    if angle1 == angle2:
        time = dist / v
    else:
        time = t * v1 / v + (dist - v1 * t / v) / v2
else:
    if v1 > v2:
        angle = angle1
        vv = v1
    else:
        angle = angle2
        vv = v2
    dx = x2 - x1 - v * t * cos(angle)
    dy = y2 - y1 - v * t * sin(angle)
    time = t + sqrt(dx ** 2 + dy ** 2) / vv

print(time)
