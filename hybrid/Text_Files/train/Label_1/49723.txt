import java.util.*;
import java.io.*;

public class Main {
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};

    static int n, m;
    static int[][] grid;
    static boolean[][] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new int[n + 1][m + 1];
        visited = new boolean[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
        int count = 0;
        for (int l = 1; l <= n * m; l++) {
            for (int r = l; r <= n * m; r++) {
                for (int i = 1; i <= n; i++) {
                    for (int j = 1; j <= m; j++) {
                        if (l <= grid[i][j] && grid[i][j] <= r && !visited[i][j]) {
                            if (dfs(i, j, l, r)) {
                                count++;
                            }
                        }
                    }
                }
            }
        }
        System.out.println(count);
    }

    static boolean dfs(int x, int y, int l, int r) {
        if (visited[x][y]) {
            return false;
        }
        visited[x][y] = true;
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && l <= grid[nx][ny] && grid[nx][ny] <= r) {
                if (!dfs(nx, ny, l, r)) {
                    return false;
                }
            }
        }
        return true;
    }
}
