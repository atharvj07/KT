import java.util.*;

public class Main {
  
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        int[] next = new int[n];
        int[] prev = new int[n];
        for (int i = 0; i < n; i++) {
            next[i] = i + 1;
            prev[i] = i - 1;
        }
        for (int i = 0; i < n; i++) {
            int j = i + k;
            if (j < n && Math.abs(p[i] - p[j]) == 1) {
                int tmp = p[i];
                p[i] = p[j];
                p[j] = tmp;
                int tmp2 = next[i];
                next[i] = next[j];
                next[j] = tmp2;
                int tmp3 = prev[i];
                prev[i] = prev[j];
                prev[j] = tmp3;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(p[i]);
        }
    }
}
