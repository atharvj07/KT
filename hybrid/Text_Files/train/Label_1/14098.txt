#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
const int N = 2e5 + 5;

ll n, dp[N], pos[N];
vector<ll> adj[N];

struct srt { ll out, in; };
vector<srt> v;

bool comp(srt a, srt b) {
  if (a.in != b.in) return a.in > b.in;
  return a.out > b.out;
}

ll f(ll x) {
  if (x == -1) return 0;
  ll &ret = dp[x];
  if (ret != -1) return ret;
  ret = 1;
  for (ll i : adj[x]) { ret += f(i); ret %= mod; }
  return ret;
}

int main() {
  memset(dp, -1, sizeof dp);
  cin >> n;
  for (ll i = 0, a, b; i < n; i++) {
    cin >> a >> b;
    v.push_back({a, b});
  }
  sort(v.begin(), v.end(), comp);

  for (ll i = 0; i < n; i++) {
    ll out = v[i].out;
    ll posx = i;
    for (ll j = i + 1; j < n; j++) {
      if (v[j].out >= out) { pos[i] = j; break; }
    }
    if (pos[i] != i) adj[pos[i]].push_back(i);
  }

  ll ans = f(pos[0]);
  cout << ans << '\n';
}
