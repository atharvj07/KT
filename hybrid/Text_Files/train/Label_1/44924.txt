import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        if (gcd(arr) > 1) {
            System.out.println(0);
            return;
        }
        Arrays.sort(arr);
        int minCost = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                continue;
            }
            int cost = 0;
            if (arr[i] % 2 == 1) {
                cost += b;
                arr[i] = arr[i] - 1;
            }
            int g = gcd(arr);
            if (g > 1) {
                cost += a + (n - i) * b;
            } else {
                int nextOdd = i;
                while (nextOdd < n && arr[nextOdd] % 2 == 0) {
                    nextOdd++;
                }
                if (nextOdd < n) {
                    cost += a + (nextOdd - i) * b;
                    arr[nextOdd] = arr[nextOdd] - 1;
                } else {
                    cost += a + (n - i) * b;
                }
            }
            minCost = Math.min(minCost, cost);
        }
        System.out.println(minCost);
    }

    private static int gcd(int[] arr) {
        int g = 0;
        for (int x : arr) {
            g = gcd(g, x);
        }
        return g;
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
