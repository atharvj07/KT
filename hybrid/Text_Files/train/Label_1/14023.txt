#include<iostream>
#include<numeric>
#include<vector>
#include<algorithm>
using namespace std;
int main() {
    int n,w;cin>>n>>w;
    vector<long long> a(2*n);
    for (auto &x:a) cin>>x;
    sort(a.rbegin(),a.rend());
    double ans=0;
    for (int i=0;i<n;++i) ans+=a[2*i];
    for (int i=1;i<n;++i) ans+=a[2*i-1]/2.0;
    ans=min(ans,(double)w);
    cout<<ans<<endl;
}
