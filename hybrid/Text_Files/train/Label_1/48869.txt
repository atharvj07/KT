import java.util.*;

public class Main {
    static int[][] adj;
    static boolean[] used;
    static char[] res = new char[27];
    static int idx = 0;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        adj = new int[27][27];
        used = new boolean[27];
        int a = 'A';
        for (int i = 0; i < 13; i++) {
            for (int j = 0; j < 2; j++) {
                int x = i + j * 13;
                int y = s.charAt(x) - a;
                adj[x][y] = 1;
                adj[y][x] = 1;
            }
        }
        dfs(0);
        if (idx == 26) {
            for (int i = 0; i < 13; i++) {
                for (int j = 0; j < 2; j++) {
                    int x = i + j * 13;
                    System.out.print((char) (res[x] + a));
                }
                System.out.println();
            }
        } else {
            System.out.println("Impossible");
        }
    }

    static void dfs(int v) {
        if (v == 26) {
            idx = 26;
            return;
        }
        used[v] = true;
        res[idx++] = (char) (v + 'A');
        for (int u = 0; u < 27; u++) {
            if (adj[v][u] == 1 && !used[u]) {
                dfs(u);
            }
        }
        used[v] = false;
        idx--;
    }
}
