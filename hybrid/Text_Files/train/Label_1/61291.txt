from sys import stdin
from functools import reduce

MOD = 998244353
n, k = list(map(int, stdin.readline().split()))
a = list(map(int, stdin.readline().split()))

s1 = 0
s2 = 0
for ai in a:
    if ai == -1:
        s1 = 2 * s1
        s2 = (2 * s2) % MOD
    elif ai != 0 and ai % 2 == 0:
        s1 = (s1 + 1) % MOD
    elif ai != 0 and ai % 2:
        s2 = (s2 + 1) % MOD

print(pow(k, s1, MOD) * pow(k - 1, s2, MOD) % MOD)
