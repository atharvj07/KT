#include <bits/stdc++.h>
#define rep(i,a,b) for(int i=a; i<=b; i++)

using namespace std;

int n, a, b;
vector<vector<int>> adj;
vector<int> path;

int vis[100005];

bool euler_path() {
    int odd=0;
    rep(i,1,n) {
        if(adj[i].size()%2==1) odd++;
    }
    if(odd>2) return false;

    int start=0;
    rep(i,1,n) {
        if(adj[i].size()%2==1) {
            if(start==0) start=i;
            else {
                a=start; b=i;
            }
        }
    }

    stack<int> s;
    s.push(start);
    while(!s.empty()) {
        int x=s.top();
        s.pop();
        path.push_back(x);
        for(int y: adj[x]) {
            if(vis[y]) continue;
            vis[y]=true;
            s.push(y);
        }
    }

    if(path.size()!=2*n) return false;

    path.erase(path.begin());
    path.pop_back();
    return true;
}

int main() {
    cin>>n;
    adj.resize(n+1);
    while(cin>>a>>b) {
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    if(!euler_path()) cout<<-1<<endl;
    else {
        for(int x: path) cout<<x<<" ";
        cout<<endl;
    }

    return 0;
}
