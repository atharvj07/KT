#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<string, int>> p(n);
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        p[i].first = s;
        p[i].second = i + 1;
    }
    sort(p.begin(), p.end(), [](const pair<string, int> &a, const pair<string, int> &b) {
        for (int i = 0; i < min(a.first.size(), b.first.size()); i++) {
            if ((i % 2) == 0) {
                if (a.first[i] != b.first[i]) { return a.first[i] > b.first[i]; }
            } else {
                if (a.first[i] != b.first[i]) { return a.first[i] < b.first[i]; }
            }
        }
        return a.first.size() < b.first.size();
    });
    for (int i = 0; i < n; i++) { cout << p[i].second << ' '; }
}
