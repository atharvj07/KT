#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 1;
int a[N], p[N], dp[N];
int32_t main()
{
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i] >> p[i];
    }
    dp[0] = 0;
    for (int i = 1; i <= n; i++)
    {
        dp[i] = 1e18;
        for (int j = 1; j <= i; j++)
        {
            dp[i] = min(dp[i], dp[j - 1] + (i - j + 1) * p[j]);
        }
    }
    cout << dp[n] << endl;
    return 0;
}
