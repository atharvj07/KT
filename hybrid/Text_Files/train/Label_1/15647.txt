#include <bits/stdc++.h>

using namespace std;

const int MAX_ID = 1e9+7;
const int MAX_FRIEND = 100;
const int MAX_PERCENT = 100;

int main() {
    int m, k;
    cin >> m >> k;
    
    map<int, vector<int>> adj;
    
    int a, b;
    for (int i = 0; i < m; ++i) {
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    
    for (auto &[id, friends] : adj) {
        sort(friends.begin(), friends.end());
        vector<int> potentialFriends;
        
        int numFriends = friends.size();
        int threshold = numFriends * k / MAX_PERCENT;
        
        for (int j = 0; j < numFriends; ++j) {
            int friendId = friends[j];
            if (id == friendId) continue;
            
            int mutualFriends = 0;
            
            for (int friendId2 : adj[friendId]) {
                if (binary_search(friends.begin(), friends.end(), friendId2)) {
                    ++mutualFriends;
                }
            }
            
            if (mutualFriends >= threshold) {
                potentialFriends.push_back(friendId);
            }
        }
        
        sort(potentialFriends.begin(), potentialFriends.end());
        
        int numPotentialFriends = potentialFriends.size();
        cout << id << ": " << numPotentialFriends << " ";
        for (int friendId : potentialFriends) {
            cout << friendId << " ";
        }
        cout << endl;
    }
    
    return 0;
}
