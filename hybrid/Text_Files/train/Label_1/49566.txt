import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        boolean[][] m = new boolean[n][n];
        for (int i = 0; i < (n * (n - 1)) / 2 - 1; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            m[x][y] = true;
            m[y][x] = false;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j || m[i][j]) {
                    continue;
                }
                System.out.println((i + 1) + " " + (j + 1));
                return;
            }
        }
    }
}
