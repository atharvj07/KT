import sys

n = int(sys.stdin.readline())
s = list(sys.stdin.readline().strip())
t = list(sys.stdin.readline().strip())

min_dist = n
min_i = -1
min_j = -1

for i in range(n):
    for j in range(i + 1, n):
        if s[i] != t[j] and s[j] != t[i]:
            dist = 0
            for k in range(n):
                if s[k] != t[k]:
                    dist += 1
            if dist < min_dist:
                min_dist = dist
                min_i = i + 1
                min_j = j + 1

print(min_dist)
if min_i == -1:
    print("-1 -1")
else:
    print(min_i, min_j)
