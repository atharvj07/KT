#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  map<int, vector<int>> m;
  for (int i = 0; i < n; i++) m[a[i]].push_back(i);
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      int sum = a[i] + a[j];
      if (m.find(sum) == m.end()) continue;
      auto &v = m[sum];
      for (int k = 0; k < v.size(); k++) {
        if (v[k] != i && v[k] != j) {
          cout << "YES" << endl;
          cout << i + 1 << " " << j + 1 << " " << v[k] + 1 << " " << (v.back() == v[k] ? v[k - 1] : v.back()) + 1 << endl;
          return 0;
        }
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
