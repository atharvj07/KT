#include <iostream>
using namespace std;
int main() {
    int n, m, p = 0;
    cin >> n >> m;
    int a[n][m], b[n][m] = {1, 1, 1, 1};
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) cin >> a[i][j];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) {
            if (!a[i][j]) {
                for (int k = 0; k < 4; k++)
                    if (b[i][k] && !a[(i + (k & 1) * (k == 1 ? 1 : -1)) % n][(j + (k & 2) * (k == 3 ? 1 : -1)) % m])
                        b[i][k] = 0;
                p += b[i][0] + b[i][1] + b[i][2] + b[i][3];
            }
        }
    cout << p;
    return 0;
}
