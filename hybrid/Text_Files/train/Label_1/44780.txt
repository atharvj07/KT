import java.util.*;
import java.io.*;

public class Jaber {
    static class City {
        int row, col, color;

        City(int row, int col, int color) {
            this.row = row;
            this.col = col;
            this.color = color;
        }
    }

    static class Pair {
        int row, col;

        Pair(int row, int col) {
            this.row = row;
            this.col = col;
        }
    }

    static int n, m, k, q;
    static int[][] grid;
    static List<City>[] colorToCities;
    static int[][] dist;

    static int dx[] = {-1, 1, 0, 0};
    static int dy[] = {0, 0, -1, 1};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());

        grid = new int[n][m];
        colorToCities = new ArrayList[k + 1];
        for (int i = 0; i <= k; i++) {
            colorToCities[i] = new ArrayList<>();
        }

        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < m; j++) {
                int color = Integer.parseInt(st.nextToken());
                grid[i][j] = color;
                colorToCities[color].add(new City(i, j, color));
            }
        }

        q = Integer.parseInt(br.readLine());
        dist = new int[n][m];

        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int r1 = Integer.parseInt(st.nextToken()) - 1;
            int c1 = Integer.parseInt(st.nextToken()) - 1;
            int r2 = Integer.parseInt(st.nextToken()) - 1;
            int c2 = Integer.parseInt(st.nextToken()) - 1;

            bfs(r1, c1, r2, c2);
        }
    }

    static void bfs(int r1, int c1, int r2, int c2) {
        Queue<Pair> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }

        queue.offer(new Pair(r1, c1));
        dist[r1][c1] = 0;

        while (!queue.isEmpty()) {
            Pair cur = queue.poll();

            if (cur.row == r2 && cur.col == c2) {
                System.out.println(dist[cur.row][cur.col]);
                return;
            }

            for (int i = 0; i < 4; i++) {
                int nr = cur.row + dx[i];
                int nc = cur.col + dy[i];

                if (nr >= 0 && nr < n && nc >= 0 && nc < m && dist[nr][nc] == -1) {
                    if (grid[nr][nc] == grid[cur.row][cur.col]) {
                        dist[nr][nc] = dist[cur.row][cur.col];
                        queue.offer(new Pair(nr, nc));
                    }
                }
            }

            for (City next : colorToCities[grid[cur.row][cur.col]]) {
                if (dist[next.row][next.col] == -1) {
                    dist[next.row][next.col] = dist[cur.row][cur.col] + 1;
                    queue.offer(new Pair(next.row, next.col));
                }
            }
        }
    }
}
