n, m, p = map(int,input().split())
s = input()
ss = list(s)
for _ in range(m):
    c = input()
    if c=='L': p-=1
    elif c=='R': p+=1
    else:
        i=p-1
        j=i
        while s[i]!=')': i-=1
        while s[j]!='(': j+=1
        del ss[i:j+1]
        if p>i: p-=j-i-1
s = ''.join(ss)
print(s)
