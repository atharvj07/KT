#include <bits/stdc++.h>
using namespace std;
vector<int>z, o;
int main() {
  string s;
  cin >> s;
  int n = s.size();
  for (int i = 0; i < n; i++) {
    if (s[i] == '1') {
      z.push_back(i);
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      o.push_back(i);
    }
  }
  for (int i = 0; i < n; i++) { cout << z[i % z.size()]; }
  return 0;
}
