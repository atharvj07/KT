import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static long[] x, y;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        x = new long[n + 1];
        y = new long[n + 1];
        for (int i = 0; i < n; i++) {
            String[] input = reader.readLine().split(" ");
            x[i] = Integer.parseInt(input[0]);
            y[i] = Integer.parseInt(input[1]);
        }
        x[n] = x[0];
        y[n] = y[0];
        long cX = 0, cY = 0;
        for (int i = 0; i < n; i++) {
            cX += x[i];
            cY += y[i];
        }
        cX /= n;
        cY /= n;
        boolean found = false;
        for (int i = 0; i < n; i++) {
            long a = 0, b = 0, c = 0, d = 0;
            for (int j = 0; j < n; j++) {
                long x1 = x[i], y1 = y[i], x2 = x[(i + j + 1) % n], y2 = y[(i + j + 1) % n];
                a += x1 * y2 - x2 * y1;
                b += y1 - y2;
                c += x2 - x1;
                d += x1 * y1 - x2 * y2;
            }
            long detA = b * c - a * d;
            if (detA == 0) {
                continue;
            }
            long detX = cY * c - cX * d;
            long detY = -b * cX + a * cY;
            if (detA < 0) {
                detA = -detA;
                detX = -detX;
                detY = -detY;
            }
            double xP = (double) detX / detA;
            double yP = (double) detY / detA;
            if (max(Math.abs(xP - cX), Math.abs(yP - cY)) <= 0.0001) {
                System.out.printf("%.5f %.5f\n", xP, yP);
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("NA");
        }
    }

    static double max(double a, double b) {
        return a > b ? a : b;
    }
}
