#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,f=0;
        cin>>n;
        long long sum=0,mul=1;
        for(int i=0;i<n;i++){
            int x;
            cin>>x;
            sum+=x;
            mul*=x;
            if(x)f=1;
        }
        if(f){
            if(mul==0)cout<<2<<"\n";
            else cout<<0<<"\n";
        }
        else if(sum==0)cout<<1<<"\n";
        else cout<<2<<"\n";
    }
}
