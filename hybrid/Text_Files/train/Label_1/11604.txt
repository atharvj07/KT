#include <iostream>
#include <map>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    map<int, int> a, b;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      a[x] = i;
    }
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      b[x] = i;
    }
    string s(n, '0');
    for (const auto& p : a) s[p.second] = '1';
    for (const auto& p : b) if (s[p.second] == '0') s[p.second] = '1';
    cout << s << '\n';
  }
}
