import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        boolean gp = true;
        long c = arr[1] * 1L;
        long b = 1 * 1L;
        for (int i = 2; i < n; i++) {
            long cb = (c * b) / arr[i - 1];
            if (arr[i] != cb) {
                gp = false;
                break;
            }
            b = cb;
        }
        if (gp) {
            System.out.println(0);
            return;
        }
        int ans = 2;
        for (int i = 0; i < n; i++) {
            int[] newArr = new int[n - 1];
            for (int j = 0, k = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                newArr[k++] = arr[j];
            }
            boolean isGP = true;
            c = newArr[1] * 1L;
            b = 1 * 1L;
            for (int j = 2; j < n - 1; j++) {
                long cb = (c * b) / newArr[j - 1];
                if (newArr[j] != cb) {
                    isGP = false;
                    break;
                }
                b = cb;
            }
            if (isGP) {
                ans = 1;
                break;
            }
        }
        System.out.println(ans);
    }
}
