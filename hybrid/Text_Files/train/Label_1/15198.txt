#include<bits/stdc++.h>
using namespace std;
int32_t main(){
    int n,m;
    cin>>n>>m;
    vector<int> a(n,0);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    vector<pair<int,int>> p(m);
    for(int i=0;i<m;i++){
        int x,y;
        cin>>x>>y;
        p[i]={x,y};
    }
    sort(p.begin(),p.end());
    int day=0,ans=0,cost=0;
    for(int i=0;i<n;i++){
        cost+=(2*a[i]);
    }
    while(ans<n){
        cost=0;
        int j=0,day1=day,x=0;
        while(j<m && day1<=day){
            if(p[j].first<=day){
                if(p[j].second<=x){
                    j++;
                    continue;
                }
                x=p[j].second;
                for(int y=x;y<n;y++){
                    if(a[y]>0){
                        a[y]--;
                        cost+=1;
                    }
                }
            }
            else break;
            j++;
        }
        for(int i=0;i<n;i++){
            cost+=(2*a[i]);
        }
        if(cost<=day1){
            ans=n;
            break;
        }
        else{
            day++;
        }
    }
    cout<<day<<endl;
    return 0;
}
