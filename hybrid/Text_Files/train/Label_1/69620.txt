for _ in range(int(input())):
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]
    a.sort(key=lambda x: x[0])
    p = tmp = ans = 0
    for x, y in a:
        if x <= tmp:
            p += y
        else:
            ans += min(p, y)
            tmp = x
    print(ans)
