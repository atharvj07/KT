from collections import defaultdict
from functools import reduce
from operator import mul
import sys

def sieve(n: int) -> list[int]:
    primes = [True for _ in range(n+1)]
    p = 2
    while p * p <= n:
        if primes[p]:
            for i in range(p * p, n+1, p):
                primes[i] = False
        p += 1
    primes[0] = primes[1] = False
    return [p for p in range(n+1) if primes[p]]

def modpow(base: int, exp: int, mod: int) -> int:
    if exp == 0:
        return 1
    if exp == 1:
        return base
    t = modpow(base, exp // 2, mod)
    t = (t * t) % mod
    if exp % 2 == 1:
        t = (t * base) % mod
    return t

n, p = map(int, input().split())
a = list(map(int, input().split()))

primes = sieve(p)

a.sort()
dp = defaultdict(int)
for el in a:
    for pr in primes:
        if pr > el:
            break
        if el % pr == 0:
            dp[pr] += 1
            while el % pr == 0:
                el //= pr

nums = list(dp.keys())
nums.sort()

def solve(idx: int, prv: int, cur: int, cnt: int) -> int:
    if idx == len(nums):
        if cnt > 0:
            return dp[prv] // cnt
        else:
            return 0
    if cur > n:
        return 0
    res = solve(idx + 1, prv, cur, cnt)
    if cur * nums[idx] <= n:
        res += solve(idx, nums[idx], cur * nums[idx], cnt + dp[nums[idx]])
    return res

res = []
for x in range(3, n + 1):
    if solve(0,1,1,0) % p == 0:
        continue
    res.append(x)
    
print(len(res))
print(*res)
