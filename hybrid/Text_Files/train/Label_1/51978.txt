import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long ans = -1;
        for (int i = 0; i < n; i++) {
            long a = in.nextLong();
            long b = in.nextLong();
            if (ans == -1) {
                ans = a;
            } else {
                while (ans % b != 0) {
                    ans += a;
                }
            }
        }
        System.out.println(ans % 1000000007);
    }
}
