#include <bits/stdc++.h>
using namespace std;

const int N = 3e5 + 5;
int n, a[N], l[N], r[N];
long long dp[N];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  l[0] = -1e9;
  r[n + 1] = -1e9;
  for (int i = 1; i <= n; i++) l[i] = max(l[i - 1], a[i]);
  for (int i = n; i >= 1; i--) r[i] = max(r[i + 1], a[i]);
  dp[n] = a[n];
  for (int i = n - 1; i >= 1; i--) {
    if (a[i] >= l[i - 1])
      dp[i] = r[i + 1];
    else
      dp[i] = l[i - 1];
  }
  for (int i = 0; i < n; i++) {
    if (i) cout << " ";
    cout << dp[i + 1];
  }
  cout << endl;
}
