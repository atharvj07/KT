n = int(input())
p = list(map(int, input().split()))
vowels = 'eyuioa'

def main():
    invalid = False
    for line in range(n):
        words = input().split()
        cnt = 0
        for word in words:
            has_v = False
            for c in word:
                if not has_v and c in vowels:
                    has_v = True
                    cnt += 1
        if cnt != p[line]:
            return False
    return True

if main(): print('YES')
else: print('NO')
