n = int(input().strip())
h = list(map(int, input().strip().split()))

left_operations = [0] * n
right_operations = [0] * n
operations = 0

for i in range(1, n):
    left_operations[i] = min(left_operations[i - 1] + 1, h[i])

for i in range(n - 2, -1, -1):
    right_operations[i] = min(right_operations[i + 1] + 1, h[i])

for i in range(n):
    operations = max(operations, min(left_operations[i], right_operations[i]))

print(operations + 1)
