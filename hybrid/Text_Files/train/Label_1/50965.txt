import java.util.*;

class Main {
    static void swap(int[] a, int[] b, int i, int j) {
        int t = a[i];
        a[i] = b[j];
        b[j] = t;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int m = in.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }
        int sa = 0;
        for (int x : a) {
            sa += x;
        }
        int sb = 0;
        for (int x : b) {
            sb += x;
        }
        int v = Math.abs(sa - sb);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int d = a[i] - b[j];
                if (Math.abs(v + d) < Math.abs(v)) {
                    v = v + d;
                    sa = sa + d;
                    sb = sb - d;
                    swap(a, b, i, j);
                }
            }
        }
        int k = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int d = a[i] - b[j];
                if (Math.abs(v + d) < Math.abs(v)) {
                    v = v + d;
                    sa = sa + d;
                    sb = sb - d;
                    k++;
                    swap(a, b, i, j);
                }
            }
        }
        System.out.println(v);
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                for (int l = 0; l < m; l++) {
                    if (a[j] == b[l]) {
                        System.out.println((j + 1) + " " + (l + 1));
                        swap(a, b, j, l);
                    }
                }
            }
        }
    }
}
