import math
for _ in range(int(input())):
    n = str(int(input()))
    l = len(n)
    if n[0] != '1':
        print(1)
    else:
        ans = 0
        for i in range(l):
            if n[i] == '0':
                if i > 0 and n[i - 1] == '1':
                    ans += i
                    break
            else:
                if i > 0 and n[i - 1] == '0':
                    ans += i
                    break
        k = int(math.log2(int(n)))
        ans += l - 1 - k
        print(ans)
