def min_insertions_to_make_functional_protein(dna):
    insertions = 0
    i = 0
    while i < len(dna):
        count = 1
        while i + 1 < len(dna) and dna[i] == dna[i + 1]:
            i += 1
            count += 1
        if count % 2 == 0:
            insertions += 1
        i += 1
    return insertions

# Read input
s = input().strip()
# Compute and print result
print(min_insertions_to_make_functional_protein(s))
