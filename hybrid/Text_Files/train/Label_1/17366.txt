#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> y1(n), y2(m);
  for (int i = 0; i < n; i++) cin >> y1[i];
  for (int i = 0; i < m; i++) cin >> y2[i];
  sort(y1.begin(), y1.end());
  sort(y2.begin(), y2.end());
  int ans = n + m;
  if (y1.size() > 0 && y2.size() > 0) {
    int mi = min(y1[0], y2[0]);
    if (mi > 0) ans -= min(mi, n + m - 1);
  }
  if (y1.size() > 0 && y2.size() > 0) {
    int ma = max(y1[n - 1], y2[m - 1]);
    if (ma < 0) ans -= min(-ma, n + m - 1);
  }
  cout << ans << endl;
  return 0;
}
