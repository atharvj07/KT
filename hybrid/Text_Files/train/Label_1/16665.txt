#include <iostream>
#include <deque>
using namespace std;
typedef long long ll;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    ll n, q;
    cin >> n >> q;
    deque<ll> dq;
    for (ll i = 0; i < n; i++) {
        ll tmp;
        cin >> tmp;
        dq.push_back(tmp);
    }
    while (q--) {
        ll m;
        cin >> m;
        m %= n;
        if (!m)
            m = n;
        ll a, b;
        for (ll i = 0; i < m; i++) {
            a = dq.front();
            dq.pop_front();
            b = dq.front();
            dq.pop_front();
            if (a > b) {
                dq.push_front(b);
                dq.push_back(a);
            } else {
                dq.push_front(a);
                dq.push_back(b);
            }
        }
        cout << a << " " << b << "\n";
    }
    return 0;
}
