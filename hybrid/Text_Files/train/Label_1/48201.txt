import java.util.Scanner;

public class MartianDollar {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int b = scanner.nextInt();
        int[] prices = new int[n];
        for (int i = 0; i < n; i++) {
            prices[i] = scanner.nextInt();
        }

        int maxProfit = 0;
        int currentProfit = 0;
        int boughtAt = -1; // Day when we bought dollars

        for (int i = 0; i < n; i++) {
            int currentPrice = prices[i];

            // Sell dollars if we have some and the price is higher than the buying price
            if (boughtAt != -1 && currentPrice > prices[boughtAt]) {
                currentProfit += currentPrice - prices[boughtAt];
                boughtAt = -1; // Reset buying day
            }

            // Buy dollars if we have enough money and the price is lower than the selling price
            if (boughtAt == -1 && b >= currentPrice) {
                b -= currentPrice;
                boughtAt = i;
            }

            maxProfit = Math.max(maxProfit, currentProfit + b); // Update max profit
        }

        // Sell dollars on the last day if we have some
        if (boughtAt != -1) {
            maxProfit += prices[n - 1] - prices[boughtAt];
        }

        System.out.println(maxProfit);
    }
}
