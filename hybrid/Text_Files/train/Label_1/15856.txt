#include <bits/stdc++.h>

#define rep(i,a,b) for(int i=a;i<b;i++)
#define REP(i,n) rep(i,0,n)
#define each(x,i) for(auto& x: i)

using namespace std;

int main() {
  int x, k; cin >> x >> k;
  vector<int> v;
  REP(i, k) {
    int a, b;
    cin >> a >> b;
    if (a == 1) v.push_back(b - 1);
    else v.push_back(b);
  }

  sort(v.begin(), v.end());
  v.push_back(x);

  int mx = 0, mn = 0;
  REP(i, v.size() - 1) {
    if (v[i] < x) {
      mn++;
      if (v[i + 1] != v[i] + 1) mx++;
    }
  }
  
  cout << mn << " " << mx << endl;

  return 0;
}
