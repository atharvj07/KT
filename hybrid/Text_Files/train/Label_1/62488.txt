import sys
input = sys.stdin.readline

N = int(input())
chair = [0] * (2 * N + 1)
for _ in range(N):
    a, b = map(int, input().split())
    chair[a] = b
    chair[b] = a

ans = [-1] * N
vis = [0] * (2 * N + 1)
for i in range(1, 2 * N + 1):
    if vis[i]:
        continue
    if vis[chair[i]]:
        ans = [-1] * N
        break
    vis[i] = vis[chair[i]] = 1
    vis[i] = vis[chair[i]] = 2
    ans[i // 2 - 1] = (i % 2) + 1
    ans[chair[i] // 2 - 1] = 3 - ans[i // 2 - 1]

print(*ans)
