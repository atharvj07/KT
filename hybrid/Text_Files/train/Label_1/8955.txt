#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		int n,k;
		cin>>n>>k;
		string s;
		cin>>s;
		int ans=0;
		if(k%2==1)
		{
			for(int i=0;i<n-1;i++)
			{
				if(s.substr(i,k)==s.substr(0,k))
					break;
				else {
					if(s[i]!=s.substr(0,k)[i%k])
						ans++;				
				}
			}
		}
		else
		{
			int x=0,y=0;
			for(int i=0;i<k;i++)
			{
				if(s[i]=='R')
					x++;
				if(s[i]=='B')
					y++;
				if(x>=1&&y>=1)
					break;
			}
			if(x==0||y==0)
				ans=-1;
			else
			{
				for(int i=k;i<n;i++)
				{
					if(s.substr(i,k)==s.substr(0,k))
						break;
					else {
						if(s[i]!=s.substr(0,k)[i%k])
						{
							if(s[i]=='R'&&s.substr(0,k)[i%k]=='B')
							{
								if(--x) ans++;
								else ans+=2;
							}
							else if(s[i]=='B'&&s.substr(0,k)[i%k]=='R')
							{
								if(--y) ans++;
								else ans+=2;
							}
							else ans++;		
						}
					}
				}
			}
		}
		cout<<ans<<endl;
	}
	return 0;
}
