n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for i in range(len(a)):
    for j in range(len(b)):
        if (a[i] + b[j]) % 2 == 1:
            a[i] = -1
            b[j] = -1
            break
print(len(list(filter(lambda x: x != -1, a))))
