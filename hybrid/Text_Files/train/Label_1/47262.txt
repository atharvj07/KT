import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();

        for (int i = 0; i < t; i++) {
            long a = in.nextLong();
            long b = in.nextLong();

            if (a == b) {
                System.out.println(0);
                continue;
            }

            if (b % a != 0) {
                System.out.println(-1);
                continue;
            }

            int ans = 0;
            while (b > a) {
                if (b % 8 == 0) {
                    b /= 8;
                    ans++;
                } else if (b % 4 == 0) {
                    b /= 4;
                    ans++;
                } else if (b % 2 == 0) {
                    b /= 2;
                    ans++;
                } else {
                    ans = -1;
                    break;
                }
            }

            if (ans == -1) {
                System.out.println(-1);
                continue;
            }

            while (a > b) {
                if (a % 2 == 0) {
                    a /= 2;
                    ans++;
                } else {
                    ans = -1;
                    break;
                }
            }

            if (ans == -1) {
                System.out.println(-1);
                continue;
            }

            System.out.println(ans);
        }
    }
}
