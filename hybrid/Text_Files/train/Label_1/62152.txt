import sys
input = sys.stdin.readline

n = int(input())
s = input()
l = len(s)
for i in range(l):
    s += s[i]
p = [0] * (l * 2)
c = 0
for i in range(1, l * 2):
    while c > 0 and s[i] != s[c]: c = p[c - 1]
    if s[i] == s[c]: c += 1
    p[i] = c
ans = (1 << n) - l + 1
for i in range(l, l * 2):
    if p[i] == l: ans -= 1
print(ans)
