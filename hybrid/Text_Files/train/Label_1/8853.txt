#include <iostream>
#include <vector>

using namespace std;

struct Node {
    vector<Node*> children;
    Node* parent;
    bool signed_docs;
};

int main() {
    int n, m;
    cin >> n >> m;

    vector<Node> nodes(n + 1);

    for (int i = 0; i < m; i++) {
        int t;
        cin >> t;

        if (t == 1) {
            int x, y;
            cin >> x >> y;

            nodes[y].children.push_back(&nodes[x]);
            nodes[x].parent = &nodes[y];
        } else if (t == 2) {
            int x;
            cin >> x;

            nodes[x].signed_docs = true;
        } else if (t == 3) {
            int x, i;
            cin >> x >> i;

            Node* cur = &nodes[x];
            while (cur != nullptr) {
                if (cur->signed_docs) {
                    cout << "YES" << endl;
                    break;
                }

                cur = cur->parent;
            }

            if (cur == nullptr) {
                cout << "NO" << endl;
            }
        }
    }

    return 0;
}
