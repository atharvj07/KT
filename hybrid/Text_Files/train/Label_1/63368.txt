n = int(input())
a = list(map(int, input().split()))
b = input()

l, r = -10**9, 10**9

for i in range(4, n):
    if b[i] == '0':
        if b[i-1:i+1] == '1111':
            r = min(r, a[i]-1, a[i-1]-1, a[i-2]-1, a[i-3]-1, a[i-4]-1)
    else:
        if b[i-1:i+1] == '0000':
            l = max(l, a[i]+1, a[i-1]+1, a[i-2]+1, a[i-3]+1, a[i-4]+1)

print(l, r)
