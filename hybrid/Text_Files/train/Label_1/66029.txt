n,k=map(int,input().split())
a=list(map(int,input().split()))
b=[0]*n
c=[0]*n
tot=0
for i in range(n):
    if b[a[i]-1]==0:
        tot+=1
        if tot>k:
            for j in range(n):
                if c[j]==1:
                    c[j]=0
                    tot-=1
                    break
        b[a[i]-1]=1
        c[a[i]-1]=1
print(tot)
