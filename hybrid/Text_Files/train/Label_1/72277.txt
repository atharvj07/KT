def min_cost_to_catch_mouse(n, c, a):
    visited = [False] * n
    cost = 0

    for i in range(n):
        if not visited[i]:
            cycle_min_cost = float('inf')
            j = i
            while not visited[j]:
                visited[j] = True
                cycle_min_cost = min(cycle_min_cost, c[j])
                j = a[j] - 1
            cost += cycle_min_cost
    return cost

n = int(input())
c = list(map(int, input().split()))
a = list(map(int, input().split()))

print(min_cost_to_catch_mouse(n, c, a))
