#include <iostream>
#include <cstdio>

constexpr int64_t MOD = 1e9 + 7;

int64_t pow(int64_t x, int64_t n, int64_t mod) {
  int64_t ret = 1;
  while (n) {
    if (n & 1) ret = ret * x % mod;
    n >>= 1;
    x = x * x % mod;
  }
  return ret;
}

int64_t gcd(int64_t x, int64_t y) { return (!y ? x : gcd(y, x % y)); }

void solve() {
  int64_t n;
  std::cin >> n;
  int64_t ans = 0;
  for (int64_t i = 1; i <= n; ++i) {
    if (gcd(i * i % MOD, n) == 1) ans = (ans + 1) % MOD;
  }
  std::cout << ans % MOD << '\n';
}

int main() {
  int t;
  scanf("%d", &t);
  while (t--) solve();
  return 0;
}
