#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    vector<int> ord(s.size());
    iota(ord.begin(), ord.end(), 1);
    for (int i = 0, j = 0; i < s.size(); i++) {
        if (s[i] == 'l') {
            swap(ord[j], ord[j + 1]);
            j++;
        }
    }
    for (auto x : ord) cout << x << "\n";
}
