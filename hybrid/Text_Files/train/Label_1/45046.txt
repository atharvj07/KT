import java.util.*;

class Main {
    static int n;
    static int[] a, d, p;
    static ArrayList<List<Integer>> prog;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n]; d = new int[n]; p = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        prog = new ArrayList<>();
        int l = -1, r = -1, prev = 0, cur, step = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == -1) {
                if (l == -1) l = i;
            } else {
                if (l != -1) {
                    r = i-1;
                    if (r > l) {
                        cur = a[l];
                        if (step == 0) {
                            d[l] = a[l+1] - a[l];
                            step++;
                        } else if (d[l] != a[l+1] - a[l]) {
                            l = -1;
                            continue;
                        }
                        prog.add(new ArrayList<>());
                        for (int j = l; j <= r; j++) {
                            prog.get(prog.size()-1).add(a[j]);
                            p[j] = prog.size();
                        }
                        prev = r;
                        l = -1;
                        step = 0;
                    }
                }
            }
        }

        if (l != -1) prog.add(new ArrayList<>());

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] == 0) {
                ans++;
            }
        }

        System.out.println(ans);
    }
}
