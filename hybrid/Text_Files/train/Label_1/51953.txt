import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            char[] s = sc.next().toCharArray();
            char[] t = sc.next().toCharArray();
            boolean equal = true;
            for (int j = 0; j < n; j++) {
                if (s[j] != t[j]) {
                    equal = false;
                    break;
                }
            }
            if (equal) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
