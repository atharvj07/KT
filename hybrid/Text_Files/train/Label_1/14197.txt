#include <cstdio>
#include <vector>
#include <memory.h>
struct house {
    int n, m;
    char *arr[105];
    house(int _n, int _m) : n(_n), m(_m) {
        for (int i = 0; i <= n; i++) {
            arr[i] = new char[m + 2];
            memset(arr[i], '+', m + 2);
        }
    }
    ~house() {
        for (int i = 0; i <= n; i++) { delete[] arr[i]; }
    }
    void print() {
        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= m; j++) { printf("%c", arr[i][j]); }
            printf("\n");
        }
    }
};
int main() {
    int n, m; scanf("%d %d", &n, &m);
    house a(n, m);
    for (int i = 1; i <= n; i++)
        scanf("%s", a.arr[i] + 1);
    int x = 1, y = 1, d = 1, t = 0;
    if (a.arr[1][1] == '.') return printf("Never"), 0;
    while (x > 1) {
        if (y == 1 || y == m || a.arr[x - 1][y] == '.') x--;
        else {
            if (a.arr[x][y] == '+') d *= -1, a.arr[x][y] = '.', t++;
            y += d;
            if (y == 1 || y == m || a.arr[x][y] == '+' ||
                (a.arr[x - 1][y] == '.') == (a.arr[x][y] == '+'))
                d *= -1;
        }
    }
    printf("%d\n", t);
    return 0;
}
