import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int m, d;
        while (cin.hasNextInt()) {
            m = cin.nextInt();
            d = cin.nextInt();
            if (m == 0) break;
            int[] a = {0, 31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30};
            int tot = 0;
            for (int i = 0; i < m; ++i) tot += a[i];
            tot += d - 1;
            tot += 3 % 7;
            tot %= 7;
            String[] w = {"Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"};
            System.out.println(w[tot]);
        }
    }
}
