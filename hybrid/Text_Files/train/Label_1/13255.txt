#include <iostream>
#include <vector>
#include <queue>
#include <utility>

using namespace std;
#define MAXN 101

int n, m;
int grid[MAXN][MAXN];
int target[MAXN][MAXN];
int visited[MAXN][MAXN];
int ans[MAXN * MAXN];
int cnt;

int dir[4][2] = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};

struct State {
    int x, y;
    int step;
    int type; // 0: row, 1: col
};

bool check() {
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (grid[i][j] != target[i][j]) {
                return false;
            }
        }
    }
    return true;
}

bool bfs() {
    queue<State> q;
    q.push({1, 1, 0, 0});
    visited[1][1] = 1;

    while (!q.empty()) {
        State cur = q.front();
        q.pop();

        if (cur.type == 0) {  // row
            for (int j = 1; j <= m; j++) {
                grid[cur.x][j]++;
                if (visited[cur.x][j] == 0 && check()) {
                    ans[++cnt] = cur.step + 1;
                    ans[++cnt] = cur.x;
                    return true;
                } else if (visited[cur.x][j] == 0) {
                    visited[cur.x][j] = 1;
                    q.push({cur.x, j, cur.step + 1, cur.type});
                }
                grid[cur.x][j]--;
            }
        } else {  // col
            for (int i = 1; i <= n; i++) {
                grid[i][cur.y]++;
                if (visited[i][cur.y] == 0 && check()) {
                    ans[++cnt] = cur.step + 1;
                    ans[++cnt] = cur.y + n;
                    return true;
                } else if (visited[i][cur.y] == 0) {
                    visited[i][cur.y] = 1;
                    q.push({i, cur.y, cur.step + 1, cur.type});
                }
                grid[i][cur.y]--;
            }
        }
    }
    return false;
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> target[i][j];
        }
    }

    if (bfs()) {
        cout << cnt / 2 << endl;
        for (int i = 1; i <= cnt; i += 2) {
            cout << ans[i] << ' ';
        }
        cout << endl;
    } else {
        cout << -1 << endl;
    }

    return 0;
}
