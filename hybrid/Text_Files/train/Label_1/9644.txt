#include<bits/stdc++.h>
using namespace std;
const int MX=200005,mod=1e9+7;
int N,val[MX];
long long ans;
int main(){
    cin>>N;
    for(int i=0;i<N;++i)cin>>val[i];
    sort(val,val+N);
    for(int i=0;i<N;++i){
        int l=i+1;
        int r=N-1;
        int ind=-1;
        while(l<=r){
            int mid=(l+r)>>1;
            if(val[mid]>=2*val[i]){
                ind=mid;
                r=mid-1;
            }
            else l=mid+1;
        }
        if(ind==-1)ind=N;
        ans+=1LL*(ind-i)*(N-ind)%mod;
        ans%=mod;
    }
    cout<<ans<<endl;
    return 0;
}
