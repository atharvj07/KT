#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vi v(n);
  for (auto &i : v) cin >> i;
  int l = -1, r = -1, ans = 1e9;
  for (int i = 0; i < n; i++) {
    if (v[i] == 0 || v[i] <= k) {
      if (i < m) {
        l = i;
      } else {
        r = i;
      }
      if (l != -1 && r != -1) ans = min(ans, abs(r - l) * 10);
    }
  }
  cout << ans;
}
