#include <bits/stdc++.h>
using namespace std;
const int MX = 1e5 + 5;
const int lg = 20;
char s[MX];
struct info {
    int cnt[26];
    info operator+(const info &other) {
        info temp;
        for (int i = 0; i < 26; i++) { temp.cnt[i] = cnt[i] + other.cnt[i]; }
        return temp;
    }
};
info st[MX][lg];
int lg2[MX];
void build_st(int n) {
    for (int i = 0; i < n; i++) { st[i][0].cnt[(int)(s[i] - 'a')] = 1; }
    for (int lvl = 1; lvl < lg; lvl++) {
        for (int i = 0; i + (1 << lvl) - 1 < n; i++) {
            st[i][lvl] = st[i][lvl - 1] + st[i + (1 << (lvl - 1))][lvl - 1];
        }
    }
}
info query(int l, int r) {
    int len = r - l + 1;
    int lvl = lg2[len];
    return st[l][lvl] + st[r - (1 << lvl) + 1][lvl];
}
int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n, q;
    cin >> s >> q;
    n = strlen(s);
    for (int i = 2; i <= n; i++) { lg2[i] = lg2[i / 2] + 1; }
    build_st(n);
    while (q--) {
        int t;
        cin >> t;
        if (t == 1) {
            int x;
            char c;
            cin >> x >> c;
            s[x - 1] = c;
            st[x - 1][0].cnt[(int)(s[x - 1] - 'a')] = 1;
            for (int lvl = 1; lvl < lg; lvl++) {
                if (x - (1 << lvl) + 1 >= 0)
                    st[x - (1 << lvl) + 1][lvl] = st[x - (1 << lvl) + 1][lvl - 1] + st[x - (1 << (lvl - 1))][lvl - 1];
            }
        } else {
            int x, y;
            cin >> x >> y;
            int ans = 0;
            info res = query(x - 1, y - 1);
            for (int i = 0; i < 26; i++) { ans += (res.cnt[i] > 0); }
            cout << ans << endl;
        }
    }
}
