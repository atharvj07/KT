#include <bits/stdc++.h>
using namespace std;
const int MN = 2e5 + 5;
int n, q;
vector<int> adj[MN];
int sz[MN], sub[MN], par[MN], lvl[MN];
int tin[MN], tout[MN], timer;
void dfs1(int u, int p = 1) {
    sz[u] = 1;
    sub[u] = tin[u] = ++timer;
    lvl[u] = lvl[par[u]] + 1;
    for(auto v : adj[u]) if(v != p) {
        par[v] = u;
        dfs1(v, u);
        sz[u] += sz[v];
    }
    tout[u] = timer;
}
void dfs2(int u, int p) {
    for(auto v : adj[u]) if(v != p) {
        sub[u] = max(sub[u], sub[v] + 1);
        dfs2(v, u);
    }
}
int getKth(int u, int k) {
    int cur = u;
    while(k > sz[cur]) {
        int cnt = 0;
        for(auto v : adj[cur]) if(v != par[cur]) {
            if(sub[v] >= k) {
                cur = v;
                break;
            }
            cnt++;
            k -= sub[v] + 1;
        }
        if(cnt == adj[cur].size() - 1) return -1;
    }
    return cur;
}
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> q;
    for(int i = 2; i <= n; ++i) {
        int p;
        cin >> p;
        adj[i].push_back(p);
        adj[p].push_back(i);
    }
    dfs1(1); dfs2(1, 1);
    while(q--) {
        int u, k;
        cin >> u >> k;
        cout << getKth(u, k) << endl;
    }
}
