#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    int a[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int sum = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (sum + a[i] <= t) {
            sum += a[i];
            ans++;
        } else {
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
