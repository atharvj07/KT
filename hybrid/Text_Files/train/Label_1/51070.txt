import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt(), d = scan.nextInt(), s[] = new int[n], p[] = new int[n];
        for (int i = 0; i < n; i++) s[i] = scan.nextInt();
        for (int i = 0; i < n; i++) p[i] = scan.nextInt();
        int score = s[d - 1] + p[0], end = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] + p[i] == score) end++;
            else if (s[i] + p[i] > score) break;
        }
        System.out.println(end + d);
    }
}
