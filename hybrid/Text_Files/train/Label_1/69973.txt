for _ in range(int(input())):
    n = int(input())
    a = [0] + list(map(int, input().split()))
    step = [0 for _ in range(n+1)]
    for i in range(1, n+1):
        step[a[a[i]]] += 1
    for i in range(2, n+1):
        a[i] = step[a[i]]
    q = int(input())
    for _ in range(q):
        x, k = map(int, input().split())
        print(a[a[x]] if k else a[x])
