#include <iostream>
const int MOD = 1000000007;

int main() {
  int n, k, d;
  std::cin >> n >> k >> d;
  int dp[n + 1][k + 1];
  for (int i = 0; i <= n; i++)
    for (int j = 0; j <= k; j++) dp[i][j] = 0;
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      if (i == 0 && j == 0) dp[i][j] = 1;
      else if (j == 0)
        dp[i][j] = 0;
      else if (i < j)
        dp[i][j] = 0;
      else {
        dp[i][j] = (dp[i - j][j] + dp[i - j][j - 1]) % MOD;
        if (j >= d) dp[i][j] = (dp[i][j] + dp[i - j][j - d]) % MOD;
      }
    }
  }
  std::cout << dp[n][k] << std::endl;
  return 0;
}
