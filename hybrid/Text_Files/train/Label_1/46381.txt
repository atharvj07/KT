import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt();
        long[] h = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextLong();
        }

        long ans = 0;
        for (long x : h) {
            ans += (x + (a - 1)) / a;
        }
        System.out.println(ans);
    }
}
