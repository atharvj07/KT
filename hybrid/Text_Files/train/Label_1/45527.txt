import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in =  new Scanner(System.in);
        int xv = in.nextInt(), yv = in.nextInt();
        int xp = in.nextInt(), yp = in.nextInt();
        int x1 = in.nextInt(), y1 = in.nextInt(), x2 = in.nextInt(), y2 = in.nextInt();
        int mx1 = in.nextInt(), my1 = in.nextInt(), mx2 = in.nextInt(), my2 = in.nextInt();
        if ((xv - xp) * (my1 - my2) - (yv - yp) * (mx1 - mx2) == 0) {
            System.out.println("NO");
            return;
        }
        if(xv != xp || yv != yp){
            if (x1 <= Math.min(xv, xp) || x2 >= Math.max(xv, xp) || y1 <= Math.min(yv, yp) || y2 >= Math.max(yv, yp)) {
                System.out.println("NO");
                return;
            }
        }
        int xw = in.nextInt(), yw = in.nextInt();
        if(xv != xp || yv != yp){
            if (xw <= Math.min(xv, xp) || xw >= Math.max(xv, xp)) {
                System.out.println("NO");
                return;
            }
            if (((yv - yp) * (xw - xv) == (yw - yv) * (xv - xp)) == ((yv - my1) * (xw - xv) == (yw - my1) * (xv - mx1))) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
