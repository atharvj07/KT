import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int q = Integer.parseInt(input[1]);

        String word = br.readLine();

        StringBuilder[] religions = new StringBuilder[3];
        for (int i = 0; i < 3; i++) {
            religions[i] = new StringBuilder();
        }

        for (int i = 0; i < q; i++) {
            String[] query = br.readLine().split(" ");
            int religion = Integer.parseInt(query[1]) - 1;

            if (query[0].charAt(0) == '+') {
                religions[religion].append(query[2].charAt(0));
            } else {
                religions[religion].deleteCharAt(religions[religion].length() - 1);
            }

            boolean[] used = new boolean[n];
            for (StringBuilder religionDescription : religions) {
                Arrays.fill(used, false);
                for (int j = 0; j < religionDescription.length(); j++) {
                    for (int k = 0; k < n; k++) {
                        if (word.charAt(k) == religionDescription.charAt(j) && !used[k]) {
                            used[k] = true;
                            break;
                        }
                    }
                }
                boolean isUsed = false;
                for (boolean b : used) {
                    isUsed |= b;
                }
                if (!isUsed) {
                    System.out.println("NO");
                    return;
                }
            }
            System.out.println("YES");
        }
    }
}
