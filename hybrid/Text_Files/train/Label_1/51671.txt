import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    Target[] targets = new Target[n];
    for (int i = 0; i < n; i++) {
      int x = sc.nextInt();
      int r = sc.nextInt();
      targets[i] = new Target(x, r);
    }
    int m = sc.nextInt();
    int[] hits = new int[n];
    for (int i = 0; i < m; i++) {
      int x = sc.nextInt();
      int y = sc.nextInt();
      for (int j = 0; j < n; j++) {
        if (targets[j].isInside(x, y) && hits[j] == 0) {
          hits[j] = i + 1;
          break;
        }
      }
    }
    int count = 0;
    for (int h : hits) {
      if (h > 0) count++;
    }
    System.out.println(count);
    for (int h : hits) {
      System.out.print(h + " ");
    }
    System.out.println();
  }

  static class Target {
    int x, r;

    Target(int x, int r) {
      this.x = x;
      this.r = r;
    }

    boolean isInside(int x, int y) {
      return (x - this.x) * (x - this.x) + y * y <= r * r;
    }
  }
}
