#include <bits/stdc++.h>
using namespace std;
int main() {
    set<pair<int, int>> s;
    for (int i = 0; i < 8; i++) {
        int x, y;
        cin >> x >> y;
        s.insert(make_pair(x, y));
    }
    int cntx = 0, cnty = 0;
    for (auto i : s) {
        cntx++;
        cnty++;
        for (auto j : s) {
            cntx += (i.first == j.first);
            cnty += (i.second == j.second);
        }
    }
    cout << (cntx == 9 && cnty == 9 ? "respectable" : "ugly") << endl;
}
