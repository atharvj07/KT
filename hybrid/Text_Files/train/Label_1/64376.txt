from queue import Queue

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

def bfs(x, y):
    visited = [[False for _ in range(m + 2)] for _ in range(n)]
    q = Queue()
    q.put((x, y, 0))
    visited[x][y] = True
    ans = 0

    while not q.empty():
        x, y, t = q.get()
        ans = max(ans, t)

        for dx, dy in [[1, 0], [-1, 0], [0, 1], [0, -1]]:
            nx, ny = x + dx, y + dy
            if 0 <= nx < n and 0 <= ny < m + 2 and not visited[nx][ny] and grid[nx][ny] != '0':
                q.put((nx, ny, t + 1))
                visited[nx][ny] = True

    return ans

print(bfs(0, 0))
