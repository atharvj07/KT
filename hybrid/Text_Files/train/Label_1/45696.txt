import java.util.Arrays;
import java.util.Scanner;

public class VovaAndTests {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[k];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        int remainingTests = n - k;
        int remainingSum = x - Arrays.stream(marks).sum();
        int minMedian = (k + 1) / 2 * y;
        int maxMedian = (n + 1) / 2 * y;

        if (remainingSum < 0 || remainingTests * p < remainingSum || maxMedian < minMedian) {
            System.out.println("-1");
            return;
        }

        int[] remainingMarks = new int[remainingTests];
        int remainingMedian = minMedian - Arrays.stream(marks).sum();
        for (int i = 0; i < remainingTests; i++) {
            remainingMarks[i] = remainingMedian / (remainingTests - i);
            remainingMedian -= remainingMarks[i];
        }

        for (int mark : remainingMarks) {
            System.out.print(mark + " ");
        }
    }
}
