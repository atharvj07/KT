#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;

struct edge {
	int u, v, w;
	bool operator < (const edge& o) const {
		return w > o.w;
	}
};
vector<edge> e;
int n, m, fa[MAXN];

int getfa(int x) {
	if (x == fa[x]) {
		return x;
	}
	return fa[x] = getfa(fa[x]);
}

void merge(int x, int y) {
	x = getfa(x);
	y = getfa(y);
	fa[x] = y;
}

int main() {
	cin >> n >> m;
	for (int i = 0; i < m; ++i) {
		int u, v, w;
		cin >> u >> v >> w;
		e.push_back({u, v, w});
	}
	iota(fa, fa + n + 1, 0);
	sort(e.begin(), e.end());
	int ans = 0, last = 0;
	for (auto& ec : e) {
		if (getfa(ec.u) == getfa(ec.v)) {
			if (++ans) {
				last = ec.w;
			}
			continue;
		}
		merge(ec.u, ec.v);
	}
	cout << ans << endl;
	if (ans > 1) {
		cout << (last - 1) << endl;
	}
	return 0;
}
