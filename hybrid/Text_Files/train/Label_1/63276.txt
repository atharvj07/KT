from collections import defaultdict

n = int(input())
words = set()
replacements = defaultdict(set)
for i in range(n):
    word = input()
    words.add(word)
    for j in range(len(word)):
        if word[j] in {'h', 'u'}:
            if word[j] == 'h':
                replacements[i].add('kh')
            else:
                replacements[i].add('oo')
            if word[j] in word[:j]:
                continue
            for k in range(i):
                if i in replacements[k]:
                    replacements[k].add(word[j])

result = 0
for i in range(n):
    if i in replacements:
        continue
    result += 1
    for j in range(i + 1, n):
        if i in replacements[j] or j in replacements[i]:
            replacements[i].update(replacements[j])
            replacements[j].clear()

print(result)
