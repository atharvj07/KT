import sys
input = sys.stdin.readline

n = int(input())
x = list(map(int, input().split()))
v = list(map(int, input().split()))

x.sort()
v.sort()

ans = 0

for i in range(n - 1):
    for j in range(i + 1, n):
        if x[j] - x[i] > v[j] - v[i]:
            ans += x[j] - x[i] - v[j] + v[i]
        else:
            ans += x[j] - x[i]

print(ans)
