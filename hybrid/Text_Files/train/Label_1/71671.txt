from collections import defaultdict

n, m, s = map(int, input().split())
r = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    r[u].append(v)
vis = [False] * (n + 1)
vis[s] = True
ans = 0
q = [s]
while q:
    u = q.pop(0)
    for v in r[u]:
        if not vis[v]:
            vis[v] = True
            q.append(v)
for i in range(1, n + 1):
    if not vis[i]: ans += 1
print(ans)
