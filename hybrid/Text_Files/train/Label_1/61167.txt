def can_make_sum(n, a, s):
    total_sum = sum(a)
    min_a, max_a = min(a), max(a)
    # If the sum s is either less than the minimum element or greater than the total sum, it's not possible.
    if s > total_sum or s < min_a:
        return False
    # If the sum s is equal to the total sum or equal to the minimum element, it's possible.
    if s == total_sum or s == min_a:
        return True
    # The sum s is possible if it's between the min element and the total sum excluding the max element.
    if min_a <= s <= total_sum - max_a:
        return True
    # If s is equal to the max element, it's possible.
    if s == max_a:
        return True
    return False

t = int(input())
for _ in range(t):
    n, q = map(int, input().split())
    a = list(map(int, input().split()))
    for _ in range(q):
        s = int(input())
        print("Yes" if can_make_sum(n, a, s) else "No")
