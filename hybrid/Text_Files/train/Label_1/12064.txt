#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<ll> v(n);
  for (ll &i : v) cin >> i;
  while (true) {
    ll mn = *min_element(v.begin(), v.end());
    if (mn * n <= v.size())
      break;
    for (ll &i : v) {
      if (i > mn) {
        v.push_back(i - mn);
        i = mn;
      }
    }
  }
  if (v.size() % 2)
    cout << "BitAryo";
  else
    cout << "BitLGM";
}
