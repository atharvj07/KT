#include <bits/stdc++.h>
using namespace std;
const int N = 1050;
int n, m, d, dp[N], c[N];
vector<int> a;
int main() {
    cin >> n >> m >> d;
    for (int i = 0; i < m; i++) {
        cin >> c[i];
    }
    dp[n] = dp[0] = 1;
    for (int i = n - 1; i >= 0; i--)
        for (int j = i + 1; j <= i + d && j <= n; j++) dp[i] |= dp[j];
    if (!dp[1]) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    int pos = 0;
    for (int i = 1; i <= n; i++) {
        bool f = 0;
        for (int j = pos; j <= n; j++) {
            if (dp[j] && j <= d + i) {
                a.push_back(i == j ? 0 : 1);
                f = 1;
            } else {
                a.push_back(0);
            }
            if (!dp[j]) {
                pos = j;
                break;
            }
        }
        if (!f) pos = i + d;
    }
    for (int i = 0; i < n; i++) {
        if (i) cout << ' ';
        cout << a[i];
    }
    cout << endl;
    return 0;
}
