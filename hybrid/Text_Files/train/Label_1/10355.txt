#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll t[1000005];
ll n,c,ans=0,mx=-1,ti[1000005];
typedef pair<ll,ll> point;
#define fi first
#define se second
int main()
{
    cin>>n;
    for (int i=1;i<=n;i++)
    {
        cin>>t[i]>>c;
        ti[i]=ti[i-1]+c;
    }
    mx=max(mx,ti[n]);
    ll tr=0;
    for (int i=1;i<=n;i++)
    {
        if (tr<t[i]) tr=t[i];
        tr+=c;
        ans=max(ans,tr-t[i]);
    }
    cout<<tr<<' '<<ans<<endl;
}
