#include <bits/stdc++.h>

const int MOD = 998244353;

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::string> b(n);
  for (int i = 0; i < n; i++) std::cin >> b[i];
  int w = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { w += b[i][j] == 'o'; }
  }
  int h = 0;
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (b[i][j] == 'o') cnt++;
      if (b[i][j] == '*' || j == m - 1) {
        h = (h + (cnt - 1) / 2 % MOD) % MOD;
        cnt = 0;
      }
    }
  }
  int v = 0;
  for (int j = 0; j < m; j++) {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (b[i][j] == 'o') cnt++;
      if (b[i][j] == '*' || i == n - 1) {
        v = (v + (cnt - 1) / 2 % MOD) % MOD;
        cnt = 0;
      }
    }
  }
  int ans = 1LL * (h + v) % MOD * w % MOD;
  std::cout << ans << std::endl;
  return 0;
}
