#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> ids(n + 1, 0);
  vector<int> res;
  int last = 0;
  for (int i = 1; i <= n; i++) {
    cin >> ids[i];
    auto it = find(res.begin(), res.end(), ids[i]);
    if (it != res.end()) {
      res.erase(it);
      res.push_back(ids[i]);
    } else {
      if (res.size() == k) res.erase(res.begin());
      res.push_back(ids[i]);
    }
    last = res.size();
  }
  cout << last << endl;
  for (int i = last - 1; i >= 0; i--) cout << res[i] << ' ';
  return 0;
}
