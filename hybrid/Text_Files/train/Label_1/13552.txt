#include <cstdio>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
  int q, n;
  scanf("%d", &q);
  while (q--) {
    scanf("%d", &n);
    vector<int> p(n + 1);
    vector<int> ans(n + 1);
    for (int i = 1; i <= n; i++) {
      scanf("%d", &p[i]);
      ans[i] = -1;
    }
    int cnt = 1;
    for (int i = 1; i <= n; i++) {
      if (ans[i] == -1) {
        int j = i;
        while (ans[j] == -1) {
          ans[j] = cnt;
          j = p[j];
        }
        cnt++;
      }
    }
    ans.erase(ans.begin());
    for (int e : ans) printf("%d ", e);
    printf("\n");
  }
}
