import java.util.Scanner;

public class Main {
    static void solve(int n, String digits) {
        char[] ans = new char[n];
        ans[n - 1] = '1';
        for (int i = n - 2; i >= 0; i--) {
            if (digits.charAt(i) > digits.charAt(i + 1)) {
                ans[i] = '2';
            } else {
                ans[i] = ans[i + 1];
            }
        }
        for (char c : ans) {
            System.out.print(c);
        }
        System.out.println();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String digits = sc.next();
            if (n == 1) {
                System.out.println('1');
                continue;
            }
            if (digits.charAt(0) > digits.charAt(1)) {
                System.out.println('-');
                continue;
            }
            solve(n, digits);
        }
    }
}
