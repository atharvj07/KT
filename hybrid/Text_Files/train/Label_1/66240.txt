import sys
input = sys.stdin.readline

n, v = map(int, input().split())
a = [0] * n
b = [0] * n
for i in range(n):
    ai, bi = map(int, input().split())
    a[i] = ai
    b[i] = bi
a.sort()
ans = 0
for ai, bi in zip(a, b):
    if v >= bi:
        ans += bi
        v -= bi
    else:
        ans += v
        break
print(ans)
