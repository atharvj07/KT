import java.util.*;

public class Main {
    static int n, m;
    static List<Edge>[] edges;
    static int ans;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        edges = new List[n + 1];
        for (int i = 1; i <= n; i++) edges[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            edges[a].add(new Edge(b, c));
            edges[b].add(new Edge(a, c));
        }
        vis = new boolean[n + 1];
        dfs(1, 0);
        System.out.println(ans);
    }

    static void dfs(int now, int cost) {
        if (now == n) {
            ans = Math.min(ans, cost);
            return;
        }
        vis[now] = true;
        for (Edge next : edges[now]) {
            if (vis[next.node]) continue;
            dfs(next.node, cost + next.cost);
        }
        vis[now] = false;
    }

    static class Edge {
        int node, cost;

        Edge(int node, int cost) {
            this.node = node;
            this.cost = cost;
        }
    }
}
