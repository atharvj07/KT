#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> h(n);
    for (int& e : h) cin >> e;

    vector<long long> pref(n + 1);
    for (int i = 0; i < n; i++) pref[i + 1] = pref[i] + h[i];

    long long ans = pref[k];
    int idx = 1;
    for (int i = k; i <= n; i++) {
        long long cur = pref[i] - pref[i - k];
        if (cur < ans) {
            ans = cur;
            idx = i - k + 1;
        }
    }

    cout << idx << endl;
}
