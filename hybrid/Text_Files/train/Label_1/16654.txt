#include <iostream>
#include <utility>
#include <algorithm>
using namespace std;
const int mxN = 3e3 + 5;
int n, v, ans, a[mxN], b[mxN];
bool cmp(pair<int, int>a, pair<int, int>b) {
    return a.first < b.first;
}

int main() {
    cin >> n >> v;
    pair<int, int>p[n];
    for (int i = 0;i < n;i++) {
        cin >> a[i] >> b[i];
        p[i] = {a[i], i};
    }
    sort(p, p + n, cmp);
    for (int i = 0, j, k, cur;i < n;) {
        j = i, cur = 0, k = 0;
        while (j < n && p[j].first - p[i].first <= 1) {
            if (b[p[j].second] + cur <= v) {
                cur += b[p[j].second], k += b[p[j].second];
            }
            else {
                k += v - cur;
                break;
            }
            j++;
        }
        i = j;
        ans += k;
    }
    cout << ans << endl;
    return 0;
}
