#include <iostream>
#include <set>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) { cin >> words[i]; }
  vector<int> cnt(200, 0);
  for (auto &word : words) {
    set<char> s;
    for (auto &c : word) { s.insert(c); }
    cnt[s.size()] += word.size();
  }
  int ans = 0;
  for (int i = 1; i <= 2; i++) {
    while (cnt[i]) {
      ans += i;
      cnt[i]--;
    }
  }
  cout << ans << endl;
}
