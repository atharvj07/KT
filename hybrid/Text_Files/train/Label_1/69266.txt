import sys

for _ in range(int(input())):
    na = input(); n, a = int(na), list(map(int, input().split()))
    Alice, Bob = 0, 0; Turn = 1; score = 0
    for i in range(n):
        if score + a[i] <= Bob: score += a[i]
        else:
            if Turn: Alice += score; Turn = 0
            else: Bob += score; Turn = 1; score = 0
            score += a[i]
    if score > 0:
        if Turn: Alice += score; Turn = 0
        else: Bob += score; Turn = 1
    print(i + 1, Alice, Bob)
