#include <iostream>
#include <string>
using namespace std;

int main() {
  int a, b, x;
  cin >> a >> b >> x;
  string s;
  for (int i = 0; i < a; i++) { s += '0'; }
  for (int i = 0; i < b; i++) { s += '1'; }
  for (int i = 0; i < x; i++) {
    if (s[i] == '0') { s[i] = '1'; s[i + 1] = '0'; }
    else { s[i] = '0'; s[i + 1] = '1'; }
  }
  cout << s << endl;
  return 0;
}
