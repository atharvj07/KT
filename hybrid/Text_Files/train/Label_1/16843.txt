#include <iostream>
using namespace std;
int main() {
  long long n, m, k, q, x, y, t, mx = 0;
  cin >> n >> m >> k >> q;
  bool a[505][505] = {0};
  while (q--) {
    cin >> x >> y >> t;
    a[x][y] = 1;
    if (t > mx) mx = t;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (i < k || j < k) continue;
      bool ok = 1;
      for (int t = i - k + 1; t <= i; t++) {
        for (int l = j - k + 1; l <= j; l++) { ok &= a[t][l]; }
      }
      if (ok) { cout << mx << endl; return 0; }
    }
  }
  cout << -1 << endl;
  return 0;
}
