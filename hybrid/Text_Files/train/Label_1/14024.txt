#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, w, ans = 0;
  cin >> n >> w;
  vector<long long> a(2 * n);
  for (int i = 0; i < 2 * n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  long long sumg = 0, sumb = 0;
  for (int i = 0; i < n; i++) { sumg += a[i]; }
  for (int i = n; i < 2 * n; i++) { sumb += a[i]; }
  long long x = min(sumg, w / 3);
  ans += x;
  ans += 2 * min(sumb, (w - x) / 2);
  cout << fixed << setprecision(9) << ans << endl;
  return 0;
}
