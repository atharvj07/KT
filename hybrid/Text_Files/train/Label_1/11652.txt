#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int l = 0, r = 0, q = 0;
    for (char c : s) {
      if (c == '(') {
        l++;
      } else if (c == ')') {
        r++;
        if (l > 0)
          l--;
        else
          q++;
      } else
        q++;
    }
    cout << ((q <= r && l == 0) ? "YES" : "NO") << endl;
  }
  return 0;
}
