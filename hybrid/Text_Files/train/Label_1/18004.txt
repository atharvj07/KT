#include <iostream>
#include <vector>
#include <algorithm>
#include <set>
using namespace std;
int main() {
    int n; cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    set<long long> st;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (!st.insert(a[i]).second) ans = i + 1;
    }
    cout << ans << endl;
    return 0;
}
