#include <bits/stdc++.h>
using namespace std;
int main() {
    string str;
    getline(cin, str);
    int n = str.size();
    int ans = 1;
    for (char c = 'a'; c <= 'z'; c++) {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            cnt += (str[i] == c);
            if(cnt > i) ans = max(ans, i + 1);
        }
    }
    cout << ans << endl;
}
