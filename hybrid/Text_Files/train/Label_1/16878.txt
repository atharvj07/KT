#include <bits/stdc++.h>
using namespace std;

int v[200010], n, k, mn;

int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    mn = min(mn, v[i]);
  }
  int l = 0, r = 30;
  while (l < r) {
    int md = (l + r) / 2;
    int sum = 0;
    for (int i = 0; i < n; i++) sum += (v[i] + (1 << md) - 1) / (1 << md);
    if (sum >= k)
      r = md;
    else
      l = md + 1;
  }
  cout << l << endl;
  return 0;
}
