#include <vector>
#include <set>
#include <iostream>
using namespace std;

#define ll long long
const ll limit = 1e12;

int main() {
  ll n;
  cin >> n;
  if (n == 1) {
    cout << 1 << '\n';
    return 0;
  }

  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  set<ll> p;
  for (ll i = 0; i < n; i++) {
    ll x = 1, sum = 0;
    while (sum + x <= a[i]) {
      sum += x;
      p.insert(x);
      x *= 2;
    }
  }
  p.erase(1);
  if (p.empty()) {
    cout << -1 << '\n';
    return 0;
  }

  ll x = *p.begin();
  int ans = 1;
  p.erase(p.begin());
  while (!p.empty()) {
    ll y = *p.begin();
    if (y <= x * 2 + 1) {
      p.erase(p.begin());
      x += y;
    } else {
      ans++;
      x = y;
      p.erase(p.begin());
    }
  }
  cout << ans << '\n';
  return 0;
}
