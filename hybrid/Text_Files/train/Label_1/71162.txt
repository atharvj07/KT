import sys

n, a, b = map(int, sys.stdin.readline().split())
if a * b - a - b > n * (n - 1) / 2:
    print("NO")
else:
    print("YES")
    for i in range(n):
        for j in range(n):
            if i == j:
                print('0', end='')
            elif a > 0 and b > 0 and i < j:
                a -= 1
                b -= 1
                print('1', end='')
            else:
                print('0', end='')
        print()
