import sys

t = int(input())

for _ in range(t):
    n, m = map(int, input().split())
    s = input()
    p = list(map(int, input().split()))
    a = [0] * 26

    for i in range(m):
        a[ord(s[p[i]-1]) - ord('a')] += 1

    a[ord(s[n-1]) - ord('a')] += 1

    for i in range(n-2, -1, -1):
        a[ord(s[i]) - ord('a')] += a[ord(s[i+1]) - ord('a')]

    for i in range(26):
        print(a[i] % (10**9 + 7), end=' ')
    print()
