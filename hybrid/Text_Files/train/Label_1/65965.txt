for i in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    ans=1
    for i in a:
        if a.count(i)>=3:
            ans=max(ans,a.count(i))
    for i in a:
        if a.count(i)>=2:
            for j in a:
                if i==j:
                    continue
                if a.count(j)>=1:
                    ans=max(ans,a.count(i)+a.count(j))
    print(ans)
