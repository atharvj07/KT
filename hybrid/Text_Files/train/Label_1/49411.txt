import java.util.*;
public class Main {
    static int n;
    static int[] size;
    static boolean[] vis;
    static int[] par;
    static List<Integer>[] adj;
    static int dfs(int v, int prv){
        size[v] = 1;
        for(int u : adj[v]){
            if(u != prv && !vis[u])
                size[v] += dfs(u, v);
        }
        return size[v];
    }
    static boolean centroid(int v, int prv, int tot){
        boolean flag = true;
        for(int u : adj[v]){
            if(u != prv && !vis[u]){
                if(2 * size[u] > tot)
                    flag = false;
                if(centroid(u, v, tot))
                    return true;
            }
        }
        return flag;
    }
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        size = new int[n + 1];
        vis = new boolean[n + 1];
        par = new int[n + 1];
        adj = new List[n + 1];
        for(int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for(int i = 1; i < n; i++){
            int u = in.nextInt(), v = in.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        dfs(1, 1);
        int tot = dfs(1, 1);
        List<Integer> answer = new ArrayList<>();
        for(int i = 1; i <= n; i++){
            vis[i] = true;
            if(centroid(i, par[i], tot)){
                answer.add(i);
            }
            vis[i] = false;
        }
        Collections.sort(answer);
        for(int u : answer)
            System.out.print("1 ");
        for(int i = 1; i <= n; i++)
            if(!answer.contains(i))
                System.out.print("0 ");
    }
}
