import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        String t = sc.next();
        int ans = 0;
        for (char c0 = 'a'; c0 <= 'z'; c0++) {
            for (char c1 = 'a'; c1 <= 'z'; c1++) {
                if (c0 == c1) {
                    continue;
                }
                boolean ok = true;
                for (int i = 0; i < s.length(); i++) {
                    if (s.charAt(i) == '0' && t.charAt(i) != c0) {
                        ok = false;
                        break;
                    }
                    if (s.charAt(i) == '1' && t.charAt(i) != c1) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
