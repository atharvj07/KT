#include <iostream>
#include <vector>

int main() {
  int m;
  std::cin >> m;
  int ans = 0;
  int tmp = 0;
  for (int i = 1; i <= 100000; i++) {
    int tmp2 = i;
    while (tmp2 % 5 == 0) {
      tmp2 /= 5;
      tmp++;
    }
    int tmp3 = tmp;
    if (tmp3 >= m) {
      ans++;
      if (tmp3 == m) {
        std::cout << i << '\n';
      }
    }
    tmp = tmp3;
  }
  std::cout << ans << '\n';
}
