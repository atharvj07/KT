import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int xv = in.nextInt(), yv = in.nextInt(), xp = in.nextInt(), yp = in.nextInt();
        int xw1 = in.nextInt(), yw1 = in.nextInt(), xw2 = in.nextInt(), yw2 = in.nextInt();
        int xm1 = in.nextInt(), ym1 = in.nextInt(), xm2 = in.nextInt(), ym2 = in.nextInt();
        boolean onSameSide = (xv-xm1)*(ym1-ym2) >= (yv-ym1)*(xm1-xm2);
        if(xw1==xw2 && Math.min(yw1, yw2)<=yv && yv<=Math.max(yw1, yw2) || xw1>xw2 && yv>=yw1 && yv<=yw2 || xw1<xw2 && yv>=yw2 && yv<=yw1) System.out.println("NO");
        else if(onSameSide && Math.min(xm1, xm2)<=xv && xv<=Math.max(xm1, xm2) || !onSameSide && Math.min(xm1, xm2)>=xv && xv>=Math.max(xm1, xm2) || onSameSide && Math.min(xm1, xm2)<=xp && xp<=Math.max(xm1, xm2) || !onSameSide && Math.min(xm1, xm2)>=xp && xp>=Math.max(xm1, xm2)) System.out.println("NO");
        else System.out.println("YES");
    }
}
