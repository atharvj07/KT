#include <bits/stdc++.h>

using namespace std;

const int maxn = 1005;
int n, a[maxn], b[maxn], ans[maxn], diff1, diff2;

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            if (!diff1)
                diff1 = i + 1;
            else
                diff2 = i + 1;
        }
    }
    for (int i = 1; i <= n; i++) { ans[i] = i; }
    swap(ans[diff1], ans[diff2]);
    for (int i = 1; i <= n; i++) { cout << ans[i] << " "; }
    cout << endl;
    return 0;
}
