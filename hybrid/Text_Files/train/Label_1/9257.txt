#include<bits/stdc++.h>
using namespace std;
const int maxn=1e3+5;
string s,t;
int n,m,q,st[maxn][maxn],en[maxn][maxn],f[maxn],g[maxn],fr[maxn];
int main(){
    cin>>n>>m>>q;
    cin>>s>>t;
    int c=0;
    for(int i=0;i<m;i++){
        if(i)f[i]=f[i-1];
        if(t[i]==t[c])f[i]=++c;
    }
    c=0;
    for(int i=m-1;~i;i--){
        if(i<m-1)g[i]=g[i+1];
        if(t[i]==t[m-1-c])g[i]=++c;
    }
    for(int i=0;i<n;i++){
        for(int j=i;j<n;j++){
            if(!i)st[i][j]=0;
            else if(s[j]==t[0]&&j-i<m){
                if(j-i==m-1)st[i][j]=1;
                else st[i][j]=st[i][j-1];
            }
            else if(s[i]==t[0]&&j-i<m){
                if(j-i==m-1)st[i][j]=1;
                else st[i][j]=st[i][j-1]+1;
            }
            else st[i][j]=st[i][j-1];
        }
    }
    for(int i=0;i<n;i++){
        for(int j=i;j<n;j++){
            if(j==n-1)en[i][j]=0;
            else if(s[j]==t[m-1]&&j-i<m){
                if(j-i==m-1)en[i][j]=1;
                else en[i][j]=en[i+1][j];
            }
            else if(s[i]==t[m-1]&&j-i<m){
                if(j-i==m-1)en[i][j]=1;
                else en[i][j]=en[i][j+1]+1;
            }
            else en[i][j]=en[i][j+1];
        }
    }
    while(q--){
        int l,r;
        cin>>l>>r;
        --l,--r;
        int cnt=0;
        if(r-l+1<m)cout<<0<<endl;
        else{
            if(st[l][r-m+1]==f[m-1])cnt++;
            if(en[l+m-1][r]==g[m-1])cnt++;
            if(cnt==2){
                fr[0]=fr[1]=1;
                for(int i=2;i<m-1;i++)fr[i]=fr[i-1]+(f[i]==g[i]);
                cout<<fr[m-2]<<endl;
            }
            else cout<<cnt<<endl;
        }
    }
    return 0;
}
