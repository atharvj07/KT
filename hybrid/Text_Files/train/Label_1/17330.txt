#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int n,m,a[505][505],b[505],ans[505];
int main() {
    cin>>n>>m;
    memset(b,0,sizeof b);
    for (int i=1;i<=n;i++) {for (int j=1;j<=m;j++) cin>>a[i][j],b[i]|=a[i][j];}
    bool pd=0;
    for (int i=1;i<=n;i++) if (!b[i]) pd=1;
    if (pd){puts("NIE");return 0;}
    puts("TAK");
    for (int i=1;i<=n;i++) {for (int j=1;j<=m;j++) if ((b[i]^a[i][j])>b[i]){ans[i]=j;break;}}
    for (int i=1;i<=n;i++) cout<<ans[i]<<" ";cout<<endl;
    return 0;
}
