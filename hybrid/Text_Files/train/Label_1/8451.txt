#include<bits/stdc++.h>
using namespace std;
string s;
bool canwin(int k)
{
    int n=s.size();
    s='|'+s;
    for(int i=k+1;i<=n;i++)
    {
        if(s[i]<=s[k])
        {
            return true;
        }
    }
    return false;
}
int main()
{
    cin>>s;
    int n=s.size();
    for(int i=0;i<n;i++)
    {
        if(canwin(i))
        {
            cout<<"Mike\n";
        }
        else
        {
            cout<<"Ann\n";
        }
    }
    return 0;
}
