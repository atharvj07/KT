#include <bits/stdc++.h>

using namespace std;
using ll = long long;
using ld = long double;

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n), b;
  for (ll i = 0; i < n; ++i) {
    cin >> a[i];
  }
  b = a;
  sort(b.begin(), b.end());
  if (a == b) {
    cout << 0 << endl;
    return 0;
  }
  vector<pair<ll, ll>> ans;
  for (ll i = 0; i < n; ++i) {
    for (ll j = i + 1; j < n; ++j) {
      if (a[i] > a[j] && b[i] < b[j]) {
        ans.push_back({i + 1, j + 1});
      }
    }
  }
  cout << ans.size() << endl;
  for (auto it : ans) {
    cout << it.first << " " << it.second << endl;
  }
  return 0;
}
