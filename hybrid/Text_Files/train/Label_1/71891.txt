color_names = ['Power', 'Time', 'Space', 'Soul', 'Reality', 'Mind']
color_map = {'purple': 'Power',
           'green': 'Time',
           'blue': 'Space',
           'orange': 'Soul',
           'red': 'Reality',
           'yellow': 'Mind'}
gems = []
n = int(input())
for i in range(n):
  gems.append(color_map[input()])

absence = []
for name in color_names:
  if name not in gems:
    absence.append(name)

print(len(absence))
for name in absence:
  print(name)
