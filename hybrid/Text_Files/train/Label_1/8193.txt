#include <bits/stdc++.h>
using namespace std;

int main() {
    int a, b;
    cin >> a >> b;
    string str;
    cin >> str;
    vector<int> v;
    for (int i = 0; i < a; i++) {
        if (str[i] == '0')
            v.push_back(i);
    }
    sort(v.begin(), v.end());
    int ans = *max_element(v.begin() + b, v.end()) - *min_element(v.begin(), v.end() - b);
    cout << ans << endl;
}
