import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[] p = new int[n], s = new int[n], need = new int[m], found = new int[k];
        for (int i = 0; i < n; i++) p[i] = sc.nextInt();
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
            need[s[i] - 1]++;
        }
        for (int i = 0; i < k; i++) {
            int id = sc.nextInt() - 1;
            found[i] = s[id];
            need[s[id] - 1]--;
        }
        int ans = 0;
        for (int i = 0; i < m; i++) {
            if (need[i] > 0) {
                Arrays.sort(p);
                int last = -1;
                for (int j = p.length - 1; j >= 0; j--) {
                    if (s[p[j] - 1] == i + 1) {
                        last = p[j] - 1;
                        break;
                    }
                }
                if (last == -1) ans += need[i];
                else ans += Math.max(0, need[i] - 1);
            }
        }
        System.out.println(ans);
    }
}
