#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
vector<int> adj[maxn];
int n, m, k, a[maxn];

struct offer {
    int x, y;
};

offer offers[maxn];

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) {
        cin >> offers[i].x >> offers[i].y;
        adj[offers[i].y].push_back(i);
    }
    sort(a + 1, a + 1 + n);
    int ans = 0;
    while (k > 0) {
        int cur = 0;
        for (int i = 1; i <= n; i++) {
            int l = 1, r = adj[i].size(), idx = -1;
            while (l <= r) {
                int mid = (l + r) / 2;
                if (offers[adj[i][mid]].x <= k) {
                    idx = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            if (idx == -1) {
                if (k >= 1) {
                    k--;
                    ans += a[i];
                }
            } else {
                int y = offers[adj[i][idx]].y;
                if (k >= offers[adj[i][idx]].x) {
                    k -= offers[adj[i][idx]].x;
                    ans += a[i] * (offers[adj[i][idx]].x - y);
                } else {
                    k = 0;
                    ans += a[i] * k;
                }
            }
        }
    }
    cout << ans << endl;
}
