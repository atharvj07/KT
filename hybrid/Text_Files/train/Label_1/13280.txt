#include <bits/stdc++.h>
#define fo(i, a, b) for (int i = (a); i <= (b); ++i)
#define of(i, a, b) for (int i = (a); i >= (b); --i)
using namespace std;
typedef long long int ll;
typedef pair<ll, ll> pll;
const ll mod = 1e9 + 7ll;
ll powmod(ll a, ll b) {
  ll res = 1ll;
  a %= mod;
  if (a < 0) a += mod;
  for (; b; b >>= 1ll) {
    if (b & 1ll) res = (res * a) % mod;
    a = (a * a) % mod;
  }
  return res;
}
const int maxn = 3e5 + 10;
int n;
ll h[maxn], b[maxn], f[maxn], g[maxn];
int main() {
  cin >> n;
  fo(i, 1, n) cin >> h[i];
  fo(i, 1, n) cin >> b[i];
  memset(f, -0x3f, sizeof f);
  vector<pair<ll, int>> v;
  fo(i, 1, n) {
    int p = lower_bound(h + 1, h + n + 1, h[i]) - h;
    f[p] = max(f[p], b[i]);
    if (i == 1 || b[i] != b[i - 1]) v.push_back({b[i], p});
  }
  memset(g, -0x3f, sizeof g);
  g[0] = 0ll;
  for (int j = 1, k = v.size(); j <= k; ++j) {
    int p = v[j - 1].second;
    fo(i, 1, n) {
      g[i] = max(g[i], g[i - 1]);
      if (i >= p) g[i] = max(g[i], f[i] + g[p - 1]);
    }
  }
  cout << g[n] << endl;
  return 0;
}
