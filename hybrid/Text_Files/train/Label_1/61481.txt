n, m = map(int, input().split())
c = list(map(int, input().split()))
count = 0
res = []
for i in range(n):
    if c[i] not in c[:i]:
        count += 1
        res.append((c[i], 0))
for i in range(n):
    if count == n:
        break
    if c[i] not in [x[0] for x in res] and c[i] not in c[:i]:
        res[i % count] = (res[i % count][0], c[i])
        count += 1
print(count)
for x in res:
    print(*x)
