import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        boolean res = true;
        int[] seq = new int[n];
        for (int i = 0; i < n; i++) {
            seq[arr[i]]++;
            if (seq[arr[i]] > 1) {
                res = false;
            }
        }
        for (int i = 0; i < n - 1; i++) {
            if (arr[(i + 1) % n] != (arr[i] + 1) % n) {
                res = false;
            }
        }
        System.out.println(res ? "Yes" : "No");
    }
}
