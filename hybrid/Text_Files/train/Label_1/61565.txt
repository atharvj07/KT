for t in range(int(input())):
    n=int(input())
    ans=0
    for i in range(1,round((n**0.5))):
        if not n%i:
            y=n//i
            if i%10+y%10+((i+y>=10))==n%10:   ans+=1
            if (i%100//10+y%100//10+((i+y>=100)))==n%100 and (i%100%10+y%100%10+((i+y>=100))==n%10): ans+=1
            if (i%1000//100+y%1000//100+((i+y>=1000)))==n%1000 and (i%1000%100//10+y%1000%100//10+((i+y>=1000))==n%100) and (i%1000%100%10+y%1000%100%10+((i+y)>=1000))==n%10: ans+=1
            if (i%10000//1000+y%10000//1000+((i+y)>=10000))==n%10000 and (i%10000%1000//100+y%10000%1000//100+((i+y)>=10000))==n%1000 and (i%10000%1000%100//10+y%10000%1000%100//10+((i+y)>=10000))==n%100 and (i%10000%1000%100%10+y%10000%1000%100%10+((i+y)>=10000)==n%10): ans+=1
    print(ans)
