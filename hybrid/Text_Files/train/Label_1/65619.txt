n,k=map(int,input().split())
g=[input()for _ in range(n)]
mx=0
mxcell=(0,0)
for i in range(n):
    for j in range(n):
        if g[i][j]=='.':
            cnt=0
            if i+k-1<n:
                flag=1
                for X in range(i,i+k):
                    if g[X][j]=='#':flag=0
                cnt+=flag
            if i-k+1>=0:
                flag=1
                for X in range(i,i-k,-1):
                    if g[X][j]=='#':flag=0
                cnt+=flag
            if j+k-1<n:
                flag=1
                for X in range(j,j+k):
                    if g[i][X]=='#':flag=0
                cnt+=flag
            if j-k+1>=0:
                flag=1
                for X in range(j,j-k,-1):
                    if g[i][X]=='#':flag=0
                cnt+=flag
            if cnt>mx:
                mx=cnt
                mxcell=(i+1,j+1)
print(*mxcell)
