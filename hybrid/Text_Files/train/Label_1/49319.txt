import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int w = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] dp = new int[n + 1];
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            dp[i] = Math.max(dp[i - 1], dp[Math.max(0, i - 2 * w - 1)] + a[i - 1]);
            for (int j = i - 2 * w; j >= 1; j--) {
                if ((i - j) % 2 == 1) {
                    dp[i] = Math.max(dp[i], dp[j - 1] + b[i - 1]);
                }
            }
        }
        StringBuilder sb = new StringBuilder();
        int i = n;
        while (i > 0) {
            if (dp[i] == dp[i - 1]) {
                sb.append('0');
                i--;
            } else {
                if (dp[i] == dp[Math.max(0, i - 2 * w - 1)] + a[i - 1]) {
                    sb.append('1');
                    i--;
                } else {
                    boolean found = false;
                    for (int j = i - 2 * w; j >= 1; j--) {
                        if ((i - j) % 2 == 1 && dp[i] == dp[j - 1] + b[i - 1]) {
                            sb.append('1');
                            i = j - 1;
                            found = true;
                            break;
                        }
                    }
                    if (!found) {
                        sb.append('0');
                        i--;
                    }
                }
            }
        }
        System.out.println(sb.reverse());
    }
}
