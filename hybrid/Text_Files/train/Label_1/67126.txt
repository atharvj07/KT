from collections import defaultdict

def solve_queries(queries):
    results = []
    for n, candies in queries:
        type_count = defaultdict(int)
        fav_count = defaultdict(int)
        for a, f in candies:
            type_count[a] += 1
            if f == 1:
                fav_count[a] += 1
        
        types_sorted = sorted(type_count.values(), reverse=True)
        fav_sorted = [fav_count[i] for i in sorted(fav_count, key=lambda x: type_count[x], reverse=True)]
        
        max_candies, max_fav = 0, 0
        current_count = float('inf')
        
        for i in range(len(types_sorted)):
            if types_sorted[i] < current_count:
                current_count = types_sorted[i]
                max_candies += current_count
                max_fav += min(current_count, fav_sorted[i])
            else:
                if current_count > 1:
                    current_count -= 1
                    max_candies += current_count
                    max_fav += min(current_count, fav_sorted[i])
        
        results.append((max_candies, max_fav))
    return results

# Reading input and writing output
q = int(input())
queries = []
for _ in range(q):
    n = int(input())
    candies = [tuple(map(int, input().split())) for _ in range(n)]
    queries.append((n, candies))

results = solve_queries(queries)
for res in results:
    print(*res)
