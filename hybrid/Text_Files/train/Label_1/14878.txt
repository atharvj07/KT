#include <bits/stdc++.h>
using namespace std;
int main() {
    string s; cin >> s;
    int n = s.size(), ans = 0;
    vector<int> pos[2][2];
    for (int i = 0, j; i < n; i = j) {
        j = i;
        while (j < n && s[j] == s[i]) j++;
        pos[s[i] - '0'][i % 2].push_back(i);
        pos[s[i] - '0'][1 - i % 2].push_back(j);
    }
    for (auto &x : pos) {
        for (auto &y : x) sort(y.begin(), y.end());
    }
    for (int l = 0; l < n; l++) {
        for (int r = l; r < n; r++) {
            if (pos[0][0].empty() || pos[0][1].empty() || pos[1][0].empty() || pos[1][1].empty()) ans++;
            else ans += lower_bound(pos[0][1].begin(), pos[0][1].end(), l) != pos[0][1].end() && lower_bound(pos[1][0].begin(), pos[1][0].end(), r) != pos[1][0].end();
        }
    }
    cout << ans;
    return 0;
}
