import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        int[][] d = new int[c + 1][c + 1];
        int[][] g = new int[n + 1][n + 1];
        for (int i = 1; i <= c; i++) {
            for (int j = 1; j <= c; j++) {
                d[i][j] = sc.nextInt();
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                g[i][j] = sc.nextInt();
            }
        }
        int[][][] dp = new int[n + 1][n + 1][c + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                int mod = (i + j) % 3;
                for (int k = 1; k <= c; k++) {
                    dp[i][j][k] = 1 << 29;
                    if (mod == 0) {
                        for (int l = 1; l <= c; l++) {
                            dp[i][j][k] = Math.min(dp[i][j][k], dp[i - 1][j - 1][l] + d[l][k]);
                        }
                    } else if (mod == 1) {
                        for (int l = 1; l <= c; l++) {
                            if (l == k) continue;
                            dp[i][j][k] = Math.min(dp[i][j][k], dp[i - 1][j - 1][l] + d[l][k]);
                        }
                    } else {
                        for (int l = 1; l <= c; l++) {
                            if (l == k) continue;
                            dp[i][j][k] = Math.min(dp[i][j][k], dp[i - 1][j - 1][l] + d[l][k]);
                        }
                    }
                    if (i > 1) {
                        dp[i][j][k] = Math.min(dp[i][j][k], dp[i - 1][j][k]);
                    }
                    if (j > 1) {
                        dp[i][j][k] = Math.min(dp[i][j][k], dp[i][j - 1][k]);
                    }
                }
            }
        }
        int ans = 1 << 29;
        for (int i = 1; i <= c; i++) {
            ans = Math.min(ans, dp[n][n][i]);
        }
        System.out.println(ans);
    }
}
