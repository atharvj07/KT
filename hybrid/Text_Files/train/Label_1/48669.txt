import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    private static int n, m;
    private static long[] x;
    private static Map<Integer, List<Edge>> graph = new HashMap<>();
    private static long[] size;
    private static boolean[] visited;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        line = br.readLine().split(" ");
        x = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            x[i] = Long.parseLong(line[i - 1]);
        }

        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            long y = Long.parseLong(line[2]);
            graph.computeIfAbsent(a, k -> new ArrayList<>()).add(new Edge(b, y));
            graph.computeIfAbsent(b, k -> new ArrayList<>()).add(new Edge(a, y));
        }

        size = new long[n + 1];
        visited = new boolean[n + 1];
        Arrays.fill(size, -1);

        long answer = dfs(1, 0);
        System.out.println(answer);
    }

    private static long dfs(int u, long p) {
        if (visited[u]) {
            return 0;
        }
        visited[u] = true;
        long sum = x[u];
        long min = Long.MAX_VALUE;
        for (Edge edge : graph.getOrDefault(u, List.of())) {
            int v = edge.v;
            if (v == p) {
                continue;
            }
            sum += dfs(v, u);
            min = Math.min(min, edge.y);
        }
        size[u] = sum;
        if (sum < min) {
            return m;
        }
        return min == Long.MAX_VALUE ? 0 : 1;
    }

    private static class Edge {
        int v;
        long y;

        public Edge(int v, long y) {
            this.v = v;
            this.y = y;
        }
    }
}
