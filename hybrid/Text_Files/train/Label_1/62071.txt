n=int(input())
s=[]
for i in range(n):s.append(input())
for i in range(n):
 for j in range(4):
  if s[i][j]=='O':
   if (j&1==0 and s[i][j+1]=='O') or (j&1 and s[i-1][3]==s[i][j+1]=='O'):
    s[i]=s[i][:j]+'+'+s[i][j+1:]
    s[i-1 if j&1 else i+1]=s[i-1 if j&1 else i+1][:3]+'+'+s[i-1 if j&1 else i+1][4:]
    break
if any('O' in i for i in s):print('NO')
else:
 print('YES')
 for i in s:print(i)
