import sys
input = sys.stdin.readline

n,q = map(int,input().split())
a = [[0 for _ in range(n+1)]for _ in range(3)]
for _ in range(q):
    r,c = map(int,input().split())
    a[r][c] ^= 1
    can = 1
    for i in range(1,n+1):
        if a[1][i]:
            can = 0
            break
    if can == 0:
        print("No")
        continue
    can = 1
    for i in range(1,n+1):
        if a[2][i]:
            can = 0
            break
    if can == 0:
        print("No")
        continue
    can = 1
    for i in range(2,n+1):
        if a[1][i] and a[2][i-1]:
            can = 0
            break
    if can == 0:
        print("No")
        continue
    print("Yes")
