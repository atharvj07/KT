#include <iostream>
#include <vector>

using namespace std;

vector<long long> pref;
int  a[1000001];

int main() {
  int  n, i, j;
  long long c = 0;
  cin >> n;
  for (i = 0; i < n; i++) {
    cin >> a[i];
    if (i) pref.push_back(pref[i - 1] + a[i]); else pref.push_back(a[i]);
  }
  for (i = 0; i < n; i++) {
    for (j = i + 1; j < n; j++) {
      if (i)
        if (pref[j] - pref[i - 1] == pref[i - 1]) c++;
      else
        if (pref[j] == 0) c++;
    }
  }
  cout << c << endl;
  return 0;
}
