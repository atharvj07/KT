import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static int[][] dist;
    static int[][] graph;
    static String[] keyb;
    static boolean[] used;
    static int n, m, x, ans;

    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] data = reader.readLine().split(" ");
        n = Integer.parseInt(data[0]);
        m = Integer.parseInt(data[1]);
        x = Integer.parseInt(data[2]);
        keyb = new String[n];

        for (int i = 0; i < n; i++) {
            keyb[i] = reader.readLine();
        }
        int q = Integer.parseInt(reader.readLine());
        String text = reader.readLine();
        char[] ch = new char[text.length()];
        for (int i = 0; i < text.length(); i++) {
            ch[i] = text.charAt(i);
        }
        boolean[][] conn = new boolean[m * n][m * n];
        dist = new int[m * n][m * n];
        for (int i = 0; i < m * n; i++) {
            Arrays.fill(dist[i], 1 << 20);
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (keyb[i].charAt(j) != 'S') {
                    for (int u = i - x; u <= i + x; u++) {
                        if (u >= 0 && u < n) {
                            for (int v = j - x; v <= j + x; v++) {
                                if (v >= 0 && v < m) {
                                    if (keyb[u].charAt(v) == 'S') {
                                        conn[i * m + j][u * m + v] = true;
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        for (int i = 0; i < m * n; i++) {
            dist[i][i] = 0;
        }
        for (int k = 0; k < m * n; k++) {
            for (int i = 0; i < m * n; i++) {
                for (int j = 0; j < m * n; j++) {
                    if (conn[i][k] && conn[k][j]) {
                        dist[i][j] = Math.min(dist[i][j], dist[i][k] + 1);
                    }
                }
            }
        }
        used = new boolean[m * n];
        graph = new int[m * n][m * n];
        for (int i = 0; i < m * n; i++) {
            for (int j = 0; j < m * n; j++) {
                if (dist[i][j] != 1 << 20) {
                    graph[i][j] = 1;
                }
            }
        }

        ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (ch[0] == keyb[i].charAt(j) && !used[i * m + j]) {
                    ans++;
                    used[i * m + j] = true;
                    dfs(i * m + j);
                }
            }
        }
        if (ch.length == ans) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }

    static void dfs(int v) {
        for (int i = 0; i < m * n; i++) {
            if (graph[v][i] == 1 && !used[i]) {
                ans++;
                used[i] = true;
                dfs(i);
            }
        }
    }
}
