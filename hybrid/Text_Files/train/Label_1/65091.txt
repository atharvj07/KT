import sys

n, m = map(int, input().split())
a = [0] * n
op = [[]] * m
for i in range(m):
    op[i] = list(map(int, input().split()))
    if op[i][0] == 2:
        a[op[i][1] - 1] = op[i][3]
q = []
for i in range(m):
    if op[i][0] == 1:
        for j in range(op[i][1] - 1, op[i][2]):
            q.append((a[j] - op[i][3], j))
    else:
        if len(q) == 0:
            print("NO")
            sys.exit()
        ma = q[-1][0]
        for j in range(op[i][1] - 1, op[i][2]):
            ma = max(ma, q[-1][0])
            while len(q) > 0 and q[-1][1] < j:
                q.pop()
            if len(q) > 0 and q[-1][0] >= ma:
                print("NO")
                sys.exit()
            a[j] = ma
q.sort()
if len(q) != len(set(q)):
    print("NO")
else:
    print("YES")
    print(*a)
