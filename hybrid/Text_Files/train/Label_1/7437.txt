bool pos(string a, string b){
    int n = a.size();
    if(n%2){
        for(int i = 0; i < n; i++){
            if(a[i] != b[i]) return false;
        }
    }
    else{
        int start = 0, end = n-1;
        while(start < end){
            if(a[start] != b[start] || a[end] != b[end]) return false;
            start++;
            end--;
        }
    }
    return true;
}
int main(){
    FIO
    string a, b;
    cin >> a >> b;
    int n = a.size();
    int ans = INT_MAX;
    for(int i = 0; i < n; i++){
        for(char ch = 'a'; ch <= 'z'; ch++){
            if(ch == a[i]) continue;
            if(pos(a, b) and pos(string(1, ch)+a.substr(1), b)){
                ans = min(ans, 1);
            }
            else{
                int cnt = 0;
                bool ok = true;
                for(int j = 0; j < n; j++){
                    if(a[j] != b[j]){
                        if(ok){
                            ok = false;
                            if(ch == b[j]) cnt++;
                            else cnt += 2;
                        }
                        else{
                            cnt += 2;
                        }
                    }
                }
                ans = min(ans, cnt);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
