#include <bits/stdc++.h>

#define ll long long

using namespace std;

ll n, d;
ll amt, mx;
map<ll, ll> have;
vector<ll> t;

int main() {
    cin >> n >> d;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        amt += x;
        mx = max(mx, amt);
        have[amt]++;
        t.push_back(amt);
    }
    if (mx > d) {
        cout << -1 << endl;
        return 0;
    }
    ll ans = 0;
    ll last = -d;
    for (ll i = mx; i >= 0; i--) {
        if (have[i] == 0 && last < i) {
            ans += 1;
            last = i;
        }
        last = max(last, i - (d - t[last] + 1));
    }
    if (last != 0) {
        ans += 1;
    }
    cout << ans << endl;
    return 0;
}
