#include <iostream>
#include <cstring>
using namespace std;
string s;

int main() {
    int k;
    cin >> k >> s;
    int n = s.length();
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += s[i] - '0';
    }
    if (sum >= k) {
        cout << 0 << endl;
    } else {
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int need = k - sum + s[i] - '0';
            if (need >= 10) {
                ans = -1;
                break;
            }
            if (need > 0) {
                sum += need;
                s[i] = need + '0';
                ans++;
            }
        }
        cout << ans << endl;
    }
}
