import sys
input = sys.stdin.readline

n = int(input())
a = [] 
b = [] 
for i in range(n):
    x, y = map(int, input().split())
    a.append((x, i + 1))
    b.append((y, i + 1))

a.sort()  
b.sort()

t = 0
seq = []
front = True
while t < n and (front or len(seq) > 1):
    if front:
        if a[t][0] < seq[-1] if seq else True:
            seq.append(a[t][1])
            front = False
        else: t += 1
    else:
        if b[t][0] > seq[-1]:
            seq.append(b[t][1])
            front = True
        else: t += 1

print(len(seq))
print(*seq)
