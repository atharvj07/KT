#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MX = 1e18;
int main(){
    ll n, k, m, d;
    cin >> n >> k >> m >> d;
    ll lo = (n + k - 1) / k, hi = n / d + 1;
    while(lo < hi){
        ll mid = lo + (hi - lo + 1) / 2;
        if(mid <= m && mid * k <= n){
            lo = mid;
        }
        else{
            hi = mid - 1;
        }
    }
    cout << lo << "\n";
    return 0;
}
