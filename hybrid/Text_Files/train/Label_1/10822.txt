#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    vector<int> pos[26];
    for (int i = 0; i < s.size(); i++) { pos[s[i] - 'a'].push_back(i); }
    bool ok = true;
    for (int i = 0; i < 26; i++) {
      if (pos[i].size() > 2) {
        ok = false;
        break;
      }
      if (pos[i].size() == 2 && pos[i][1] - pos[i][0] == 1) {
        ok = false;
        break;
      }
    }
    if (!ok) {
      cout << "No answer" << endl;
      continue;
    }
    for (int i = 0; i < 26; i++) {
      if (pos[i].size() == 1) { cout << s[pos[i][0]]; }
    }
    for (int i = 0; i < 26; i++) {
      if (pos[i].size() == 2) { cout << s[pos[i][1]] << s[pos[i][0]]; }
    }
    cout << endl;
  }
  return 0;
}
