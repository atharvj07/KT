import sys
input = sys.stdin.readline

k = int(input())

if k == 1:
    print(2)
    print("NY\nYN")
else:
    n = 3
    g = [["N" for _ in range(n)] for _ in range(n)]
    for i in range(n):
        g[i][i] = "N"
        g[0][1] = "Y"
        g[1][0] = "Y"
        g[1][2] = "Y"
        g[2][1] = "Y"
    for _ in range(k - 1):
        n += 2
        g += [["N" for _ in range(n)]] * 2
        for i in range(n - 2):
            for j in range(n - 2):
                g[i][j] = g[j][i]
        g[n - 2][n - 1] = "Y"
        g[n - 1][n - 2] = "Y"
    print(n)
    for row in g:
        print(*row)
