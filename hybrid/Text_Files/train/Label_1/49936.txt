import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int[][] forest;
    static Queue<int[]> burningTrees = new LinkedList<>();
    static int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        forest = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                forest[i][j] = -1;
            }
        }
        int k = sc.nextInt();
        for (int i = 0; i < k; i++) {
            int xi = sc.nextInt();
            int yi = sc.nextInt();
            burningTrees.add(new int[]{xi, yi});
            forest[xi][yi] = 0;
        }
        bfs();
        int maxTime = 0;
        int lastBurningXi = 0;
        int lastBurningYi = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (forest[i][j] > maxTime) {
                    maxTime = forest[i][j];
                    lastBurningXi = i;
                    lastBurningYi = j;
                }
            }
        }
        System.out.println(lastBurningXi + " " + lastBurningYi);
    }

    static void bfs() {
        while (!burningTrees.isEmpty()) {
            int[] burningTree = burningTrees.poll();
            int x = burningTree[0];
            int y = burningTree[1];
            for (int[] dir : dirs) {
                int nextX = x + dir[0];
                int nextY = y + dir[1];
                if (nextX >= 1 && nextX <= forest.length - 1 && nextY >= 1 && nextY <= forest[0].length - 1 && forest[nextX][nextY] == -1) {
                    forest[nextX][nextY] = forest[x][y] + 1;
                    burningTrees.add(new int[]{nextX, nextY});
                }
            }
        }
    }
}
