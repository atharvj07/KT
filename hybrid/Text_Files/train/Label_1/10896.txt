#include <bits/stdc++.h>
using namespace std;
const int maxn = 150001;
pair<int, int> f[maxn];
int a[maxn], ans[maxn];
bool vis[maxn];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    f[i].first = a[i];
    f[i].second = i;
  }
  sort(f + 1, f + 1 + n);
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (vis[i] == 1) continue;
    int l1 = 0, r1 = 0;
    for (int j = i; j <= n; j++) {
      if (f[i].first == f[j].first && vis[j] == 0) {
        if (l1 == 0)
          l1 = f[j].second;
        else {
          r1 = f[j].second;
          vis[r1] = 1;
          ans[++cnt] = 2 * f[i].first;
          break;
        }
      }
    }
  }
  cout << cnt << endl;
  for (int i = 1; i <= cnt; i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
