import sys
input = sys.stdin.readline

n, q = map(int, input().split())
a = list(map(int, input().split()))
reach = [False] * (n + 1)
for i in range(n):
    if a[i]:
        reach[i] = True
        for j in range(i + 1, n):
            if reach[i] and reach[j]:
                a[j] &= a[i]
                reach[j] = True
for _ in range(q):
    x, y = map(int, input().split())
    print(['Fou', 'Shi'][reach[y] and (a[x] & a[y]) > 0])
