#include <iostream>
#include <vector>

using namespace std;
struct post {
  int a[4];
} posts[4];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < 4; i++)
    for (int j = 0; j < 4; j++) cin >> posts[i].a[j];
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++)
      for (int k = 0; k < 4; k++)
        if (posts[i].a[j] + posts[i].a[k] == n) {
          cout << i + 1 << ' ' << posts[i].a[j] << ' ' << posts[i].a[k]
               << endl;
          return 0;
        }
  }
  cout << -1 << endl;
  return 0;
}
