#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int n;
    cin >> n;
    vector<vector<int>> g(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < n; j++) { g[i][j] = (s[j] == 'C'); }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (g[i][j] + g[j][i] == 1) { ans++; }
        }
    }
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) { cnt += g[i][j]; }
        ans += (cnt * (cnt - 1) / 2);
    }
    cout << ans << endl;
    return 0;
}
