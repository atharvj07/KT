#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int& ai : a) cin >> ai;
        sort(a.begin(), a.end());
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] == a[i - 1]) ans += 1;
        }
        cout << ans << endl;
    }
    return 0;
}
