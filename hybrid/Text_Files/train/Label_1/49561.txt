import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] t = new int[3][3];
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                t[i][j] = sc.nextInt();
        int n = sc.nextInt(), d[][] = new int[n][3];
        for (int i = 0; i < 3; i++)
            d[0][i] = t[0][i];
        for (int i = 1; i < n; i++)
            for (int j = 0; j < 3; j++)
                for (int k = 0; k < 3; k++)
                    if (j != k && d[i - 1][j] + t[j][k] < d[i][k])
                        d[i][k] = d[i - 1][j] + t[j][k];
        System.out.println(d[n - 1][2]);
    }
}
