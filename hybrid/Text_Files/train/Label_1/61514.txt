import sys

n = int(input())
s = input()

if s.count('?') == 0:
    a = list(s)
    n = len(a)
    sum1 = 0
    sum2 = 0
    for i in range(n):
        if i < n//2:
            sum1 += int(a[i])
        else:
            sum2 += int(a[i])
    if sum1 == sum2:
        print("Bicarp")
    else:
        print("Monocarp")
else:
    print("Bicarp")
