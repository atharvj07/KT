#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
int main()
{
    ll n,m,d,i,j,a,b,c,ans=0,p,pp,np;
    vector<pair<ll,ll>>ar,br;
    cin>>n>>m>>d;
    for(i=0; i<n; i++)
    {
        cin>>a>>b;
        ar.pb({a,b});
    }
    for(i=0; i<m; i++)
    {
        cin>>a>>b;
        br.pb({a,b});
    }
    sort(ar.begin(),ar.end());
    sort(br.begin(),br.end());
    i=0,j=0,p=0;
    while(i<n&&j<m)
    {
        if(p+ar[i].second<=d)
        {
            ans+=ar[i].first,p+=ar[i].second,i++;
            if(p==d)break;
            while(j<m&&p+br[j].second<=d)
            {
                ans+=br[j].first,p+=br[j].second,j++;
                if(p==d)break;
            }
        }
        else break;
    }
    cout<<ans<<endl;
}
