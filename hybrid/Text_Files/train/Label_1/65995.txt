import sys
input = sys.stdin.readline

n = int(input())
a = []
b = []
for _ in range(n):
    sa, ea, sb, eb = map(int, input().split())
    a.append((sa, ea))
    b.append((sb, eb))

a.sort()
b.sort()

impossible = False
for i in range(n):
    sa, ea = a[i]
    sb, eb = b[i]
    if sa > sb or ea < eb:
        impossible = True
        break

print("YES" if not impossible else "NO")
