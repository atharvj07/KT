import java.util.*;
import java.lang.*;
 
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
 
        int L = scanner.nextInt();
        int a = scanner.nextInt();
        long[] t = new long[n + 1];
        t[0] = 0;
        long[] l = new long[n + 1];
 
        for (int i = 0; i < n; i++) {
            t[i + 1] = scanner.nextLong();
            l[i + 1] = scanner.nextLong();
        }
 
        long time = 0;
        long ans = 0;
 
        for (int i = 0; i <= n; i++) {
 
            if ((time - t[i] + a) < l[i]) {
                continue;
            }
            ans++;
            time = t[i] + l[i];
        }
        System.out.println(ans);
    }
}
