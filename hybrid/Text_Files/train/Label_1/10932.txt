#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1000000007;
const ll inf = 1e9 + 7;
const char nl = '\n';
int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    vector<ll> blocked(m);
    for (ll i = 0; i < m; i++) { cin >> blocked[i]; }
    vector<ll> costs(k + 2);
    for (ll i = 1; i <= k; i++) { cin >> costs[i]; }
    vector<ll> pref(k + 2);
    for (ll i = 1; i <= k; i++) { pref[i] = pref[i - 1] + costs[i]; }
    vector<ll> dp(n + 2, inf);
    dp[0] = 0;
    for (ll i = 1; i <= n; i++) {
        bool flag = false;
        for (ll j = 0; j < m; j++) {
            if (blocked[j] == i) {
                flag = true;
                break;
            }
        }
        if (flag) continue;
        for (ll j = min(i, k); j >= 1; j--) {
            if (pref[j] > dp[i - j]) continue;
            if (i + j - 1 >= n) {
                dp[i] = min(dp[i], pref[j]);
            } else {
                dp[i] = min(dp[i], pref[j] + dp[i + j]);
            }
        }
    }
    if (dp[n] != inf) {
        cout << dp[n] << nl;
    } else {
        cout << -1 << nl;
    }
    return 0;
}
