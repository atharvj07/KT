#include <algorithm>
#include <array>
#include <cmath>
#include <iostream>
#include <vector>

using namespace std;

const int MAXV = 504;

char board[MAXV][MAXV];
bool visited[MAXV][MAXV][4];
vector<pair<int, string>> steps;

int di[] = {0, 0, 1, -1};
int dj[] = {1, -1, 0, 0};

int n, m, k;

bool ok(int i, int j, int dir) {
    return i >= 1 && i <= n && j >= 1 && j <= m && !visited[i][j][dir];
}

void dfs(int i, int j, int dir, int km) {
    visited[i][j][dir] = true;
    km++;
    cout << ((km + 3) / 4) << " ";
    if (km % 4 == 0) {
        string s;
        if (dir == 0)
            s = "R";
        else if (dir == 1)
            s = "L";
        else if (dir == 2)
            s = "D";
        else if (dir == 3)
            s = "U";
        steps.emplace_back(km / 4, s);
    }
    for (int k = 0; k < 4; k++) {
        int ni = i + di[k];
        int nj = j + dj[k];
        int ndir = (k + dir) % 4;
        if (ok(ni, nj, ndir)) dfs(ni, nj, ndir, km);
    }
}

int main() {
    cin >> n >> m >> k;
    if (k > (n - 1) * m + (m - 1) * n) {
        cout << "NO";
    } else {
        cout << "YES\n";
        dfs(1, 1, 0, 0);
        for (pair<int, string> x : steps) cout << x.first << " " << x.second << endl;
    }
    return 0;
}
