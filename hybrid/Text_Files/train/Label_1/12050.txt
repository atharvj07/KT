#include <iostream>
using namespace std;

bool solve(int n) {
  int v, p1, p2, a[701];
  cin >> v;
  for(int i = 1; i <= v; ++i) cin >> a[i];
  p1 = v;
  p2 = v - 1;
  for(int i = p1; i > 2; --i) {
    if(a[i] % a[i - 1] <= 1) return 0;
    while(p2 != 0 && a[i] % a[p2] <= 1) p2--;
  }
  if(a[2] % a[1] <= 1) return 0;
  else return 1;
}

int main() {
  int n;
  cin >> n;
  if(solve(n)) cout << "Yes" << endl;
  else cout << "No" << endl;
  return 0;
}
