t = int(input())
for _ in range(t):
    n = int(input())
    s = list(map(int, input().split()))
    jumps_needed = 0
    # To avoid multiple passes on the same trampoline, start from the last one and move backwards
    for i in range(n - 2, -1, -1):
        while s[i] > 1:
            jump_to = i + s[i]
            if jump_to < n:
                s[jump_to] = max(s[jump_to] - 1, 1)  # Reduce the strength of the landing trampoline
                s[i] = max(s[i] - 1, 1)  # Reduce the strength of the current trampoline if possible
            else:
                s[i] = max(s[i] - 1, 1)  # If jumping outside, just reduce the strength
            jumps_needed += 1
    print(jumps_needed)
