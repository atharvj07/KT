#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100001], s, f, t[100001]{}, ans;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        t[i % n + 1] += a[i];
    }
    cin >> s >> f;
    for (int i = s; i <= f; i++) {
        if (t[(i % n + n) % n + 1] > t[ans]) ans = (i % n + n) % n + 1;
    }
    cout << ans << endl;
    return 0;
}
