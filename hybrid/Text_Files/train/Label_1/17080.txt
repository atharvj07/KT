#include <bits/stdc++.h>
 
using namespace std;
 
typedef long long ll;
#define mod 1000000007
 
ll power(ll base, ll exp) {
    if(exp == 0) return 1;
    if(exp == 1) return base;
    ll temp = power(base, exp/2);
    if(exp % 2 == 0) return (temp * temp) % mod;
    else return (((temp * temp) % mod) * base) % mod;
}
 
int main() {
    ll n, a[1000001], fact[1000001], inv[1000001], sum = 0;
    cin >> n;
    for(ll i = 1; i <= n; i++) cin >> a[i], sum += a[i];
    fact[0] = 1;
    for(ll i = 1; i <= sum; i++) fact[i] = (fact[i - 1] * i) % mod;
    inv[0] = 1;
    for(ll i = 1; i <= sum; i++) inv[i] = power(fact[i], mod - 2);
    ll temp = 1;
    for(ll i = 1; i <= n; i++) temp = (temp * inv[a[i]]) % mod;
    ll ans = (fact[sum] * temp) % mod;
    cout << ans << endl;
    return 0;
}
