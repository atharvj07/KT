import sys
input=sys.stdin.readline

n=int(input())
p=[0]*n; b=[0]*n
vis=[0]*n
for i in range (n):
    x=int(input())-1
    p[i]=x; vis[x]=i
for i in range (n):
    y=int(input())
    if p[i]!=vis[i]: y^=1
    b[i]=y
#
cur=[0]*2*n; ans=0
for i in range (n):
    if vis[p[i]-1]<i: ans+=1; vis[p[i]-1]=i
    if b[p[i]-1]==0: ans+=1; b[p[i]-1]=1
    cur[i+vis[p[i]-1]]+=1
#
k=-1
for i in range (n):
    if cur[n+i]<2:
        k=i
        break
if k==-1: k=n-1
#
for i in range (n):
    if cur[i+vis[k]]==0: ans+=1
    if b[k]==0: ans+=1
#
print(ans)
