import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] adj = new int[n+1][n+1];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a][b] = 1;
            adj[b][a] = 1;
        }
        int[] s = new int[n+1];
        boolean ok = dfs(1, 1, s, adj);
        if (!ok) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i+1; j <= n; j++) {
                if (adj[i][j] == 1 && s[i] == s[j]) ans++;
            }
        }
        System.out.println(ans);
    }
    public static boolean dfs(int v, int state, int[] s, int[][] adj) {
        s[v] = state;
        for (int i = 1; i <= adj.length-1; i++) {
            if (adj[v][i] == 1 && s[i] == 0) {
                if (dfs(i, 3-state, s, adj)) return true;
            } else if (adj[v][i] == 1 && s[i] == state) return true;
        }
        return false;
    }
}
