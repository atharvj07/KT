n,m=map(int,input().split())
l1=list(map(str,input().split()))
l2=list(map(str,input().split()))
t=[0]*n
s=[0]*m
j=0
for i in l1:
    t[j]=int(i[:i.find('-')])
    j+=1
j=0
for i in l2:
    s[j]=int(i[:i.find('-')])
    j+=1
j=0
k=0
cnt=0
while j<n:
    if t[j]==s[k]:
        k+=1
        if k==m:
            cnt+=1
            k=0
    j+=1
print(cnt)
