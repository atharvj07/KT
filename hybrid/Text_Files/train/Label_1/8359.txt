#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define rep(i, a, b) for (int i = (a); i < (b); ++i)
typedef long long ll;
const int N = 1e5 + 10;
int n, k, arr[N], occ[N], gr[N], vis[N];
vector<int> g[N];
void dfs1(int u) {
    vis[u] = 1;
    occ[u] = 1;
    for (auto x : g[u]) {
        if (!vis[x]) { dfs1(x); }
        occ[u] += occ[x];
    }
}
void dfs2(int u, int fa) {
    for (auto x : g[u]) {
        if (x == fa) { continue; }
        gr[x] = gr[u] + occ[u] - occ[x];
        dfs2(x, u);
    }
}
int main() {
    cin >> n >> k;
    rep(i, 1, k + 1) {
        int a, b;
        cin >> a >> b;
        if (a > b) { swap(a, b); }
        g[a].pb(b);
    }
    dfs1(1);
    dfs2(1, 1);
    int ans = k;
    rep(i, 1, n + 1) {
        if (arr[i]) { continue; }
        if (occ[i] == 1) { continue; }
        ans -= occ[i] - gr[i] - 1;
        arr[i] = 1;
    }
    cout << ans << endl;
    return 0;
}
