#include<bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
vector<pair<int,int>> G[maxn];
int dist[maxn], vis[maxn];

void Dijkstra(int start){
    for(int i=1; i<=maxn; i++){
        dist[i] = INT_MAX;
        vis[i] = 0;
    }
    dist[start] = 0;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, start});
    while(!pq.empty()){
        int u = pq.top().second;
        int c = pq.top().first;
        pq.pop();
        if(vis[u]) continue;
        vis[u] = 1;
        for(auto e: G[u]){
            int v = e.first;
            int w = e.second;
            if(dist[u] + w < dist[v]){
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
}
int main() {
    int n, m, k; cin>>n>>m>>k;
    int special[k+10];
    for(int i=0; i<k; i++){
        int x; cin>>x;
        special[i] = x;
    }
    for(int i=0; i<m; i++){
        int u, v, w; cin>>u>>v>>w;
        G[u].push_back({v, w});
        G[v].push_back({u, w});
    }
    int mx_dist[k+10];
    for(int i=0; i<k; i++){
        int x = special[i];
        Dijkstra(x);
        int mx = 0;
        for(int j=1; j<=n; j++){
            mx = max(mx, dist[j]);
        }
        mx_dist[i] = mx;
    }
    for(int i=0; i<k; i++){
        cout<<mx_dist[i]<<" ";
    }
    cout<<endl;
    return 0;
}
