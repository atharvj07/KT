#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

const ll N = 1e5 + 1, M = 2e5 + 1;
vector<ll> graph[N], ans1, ans2;
bool used[N];

void dfs1(ll v)
{
    if (used[v]) return;
    used[v] = true;
    ans1.push_back(v);
    for (ll to : graph[v]) dfs1(to);
}

void dfs2(ll v)
{
    if (used[v]) return;
    used[v] = true;
    ans2.push_back(v);
    for (ll to : graph[v]) dfs2(to);
}

int main()
{
    ll n, m;
    cin >> n >> m;
    for (ll i = 0; i < m; i++)
    {
        ll x, y;
        cin >> x >> y;
        graph[x].push_back(y);
        graph[y].push_back(x);
    }
    for (ll i = 1; i <= n; i++)
    {
        if (used[i]) continue;
        dfs1(i);
        break;
    }
    for (ll i = 1; i <= n; i++)
    {
        if (used[i]) continue;
        dfs2(i);
        break;
    }
    if (ans1.empty() || ans2.empty())
        cout << "-1\n";
    else
    {
        cout << ans1.size() << endl;
        for (ll v : ans1) cout << v << ' ';
        cout << endl << ans2.size() << endl;
        for (ll v : ans2) cout << v << ' ';
        cout << endl;
    }
    return 0;
}
