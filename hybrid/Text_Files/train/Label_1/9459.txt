#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    long long num, min_cost;
    cin >> num;
    vector<long long> size(num + 2), cost(num + 2);
    cin >> size[1] >> cost[1];
    for (long long i = 2; i <= num; i++) {
        cin >> size[i] >> cost[i];
        size[i] += size[i - 1];
        cost[i] += cost[i - 1];
    }
    min_cost = -1;
    for (long long i = 1; i <= num - 2; i++) {
        long long l = i + 1;
        long long r = num;
        long long ll = -1, rr = -1;
        while (l <= r) {
            long long mid = (l + r) / 2;
            if (size[mid] > size[i - 1]) {
                ll = mid;
                l = mid + 1;
            } else
                r = mid - 1;
        }
        l = i + 1;
        r = num;
        while (l <= r) {
            long long mid = (l + r) / 2;
            if (size[mid] > size[i]) {
                rr = mid;
                l = mid + 1;
            } else
                r = mid - 1;
        }
        if (ll > -1 && rr > -1) {
            long long sum = cost[rr] - cost[i - 1];
            if (min_cost == -1 || sum < min_cost) min_cost = sum;
        }
    }
    cout << min_cost << endl;
    return 0;
}
