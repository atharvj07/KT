#include <bits/stdc++.h>
using namespace std;
int t, n, x, sum1, sum2;
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    sum1 = sum2 = 0;
    for (int i = 1; i <= n; i++) {
      cin >> x;
      if (x == 1)
        sum1++;
      else if (x == 2)
        sum2--;
      else
        (sum1 > 0) ? sum1++ : sum2--;
    }
    cout << sum1 + max(0, sum2) << endl;
  }
}
