n = int(input())
a = list(map(int, input().split()))
ans = 0
st = 0                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                 
for i in range(n):
    if a[i] == 0:
        if st > 0:
            st -= 1
    else:
        st += 1
        ans += 1
print(ans)
