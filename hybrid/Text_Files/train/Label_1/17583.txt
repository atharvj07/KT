#include <iostream>
#include <vector>

using namespace std;

int main() {
	int q;
	cin >> q;

	int arr[500001];
	int cnt = 0;

	while (q--) {
		int type, x, y;
		cin >> type >> x;

		if (type == 1) {
			arr[cnt] = x;
			cnt++;
		} else {
			cin >> y;
			for (int i = 0; i < cnt; i++) {
				if (arr[i] == x) {
					arr[i] = y;
				}
			}
		}
	}

	for (int i = 0; i < cnt; i++) {
		cout << arr[i] << ' ';
	}

	return 0;
}
