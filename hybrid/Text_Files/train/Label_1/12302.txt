#include <iostream>
#include <stack>
using namespace std;
int main() {
  int n;
  stack<int> s;
  cin >> n;
  char c;
  int x;
  bool valid = true;
  for (int i = 0; i < 2 * n; i++) {
    cin >> c;
    if (c == '+') {
      cin >> x;
      s.push(x);
    } else if (c == '-') {
      cin >> x;
      if (s.empty() || s.top() != x) {
        valid = false;
        break;
      } else {
        s.pop();
      }
    }
  }
  if (s.empty() && valid) {
    cout << "YES" << endl;
    while (!s.empty()) {
      cout << s.top() << ' ';
      s.pop();
    }
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
