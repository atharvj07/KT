#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  string s;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    cin >> n;
    cin >> s;
    bool ok = false;
    for (int j = 0; j < s.size() - 6; j++) {
      if (s.substr(j, 7) == "abacaba") {
        cout << "Yes" << endl;
        cout << s << endl;
        ok = true;
        break;
      }
    }
    if (!ok) {
      for (int j = 0; j < s.size() - 6; j++) {
        if (s[j] == '?' && s[j + 1] == 'a' && s[j + 2] == 'b' && s[j + 4] == 'a' &&
            s[j + 5] == 'c' && s[j + 6] == 'a') {
          s[j] = 'b';
          cout << "Yes" << endl;
          cout << s << endl;
          ok = true;
          break;
        }
      }
    }
    if (!ok) {
      cout << "No" << endl;
    }
  }
}
