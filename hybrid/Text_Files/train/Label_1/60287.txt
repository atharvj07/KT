import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(input().rstrip())
    b = list(input().rstrip())

    if a == b:
        print("YES")
        continue

    for i in range(n - k + 1):
        if a[i] != b[i]:
            for j in range(k):
                if ord(a[i + j]) == ord(b[i]) + j:
                    continue
                else:
                    break
            else:
                print("YES")
                break
    else:
        print("NO")
