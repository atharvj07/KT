import sys

for _ in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    price = list(map(int, sys.stdin.readline().split()))
    count = 0
    mn = price[-1]  
    for i in range(n - 2, -1, -1):
        mn = min(mn, price[i])
        if price[i] > mn:
            count += 1
    print(count)
