#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  getline(cin, s);
  vector<int> p;
  for (int i = 0, j = 0; i < s.size(); i++) {
    if (s[i] == 'M') continue;
    p.push_back(i - j++);
  }
  long long ans = 0;
  for (int i = 0; i + 1 < p.size(); i++) ans += (p[i] + p[i + 1]) / 2;
  cout << ans << endl;
}
