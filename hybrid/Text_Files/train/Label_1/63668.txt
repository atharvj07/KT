import sys

readline = sys.stdin.readline

n, m = map(int, readline().split())
a = [int(x) for x in readline().split()]
b = []
for _ in range(m):
    l, r = map(int, readline().split())
    b.append((l, r))
b.sort(key=lambda x: x[1] - x[0], reverse=True)
assert 1 <= n <= 1e5
assert 0 <= m <= 300
assert -1e6 <= min(a) <= max(a) <= 1e6
assert all(1 <= l <= r <= n for l, r in b)

sum_b = 0
mn = sum_b
for l, r in b:
    sum_b += a[r - 1] - a[l - 1]
    mn = min(mn, sum_b)
    if mn == sum_b:
        q = 0
        c = []
    else:
        q += 1
        c.append(l, r)
print(sum_b - mn, q)
for l, r in c:
    print(l, r)
