n, p = map(int, input().split())
a = list(map(int, input().split()))
dp = [0] * (n + 1)
for i in range(n):
    dp[i + 1] = (dp[i] + a[i]) % p
pre = [0] * (n + 1)
for j in range(n - 1, -1, -1):
    pre[j] = (pre[j + 1] + a[j]) % p
ms = 0
for i in range(n):
    ms = max(ms, (dp[i] + pre[i + 1]) % p)
print(ms)
