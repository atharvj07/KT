n = int(input())

matrix = [[0]*n for _ in range(n)]

odd, even = 1, 2

for i in range(n):
    for j in range(n):
        if (i+j) % 2 == 0:
            matrix[i][j] = odd
            odd += 2
        else:
            matrix[i][j] = even
            even += 2

for row in matrix:
    print(' '.join(map(str, row)))
