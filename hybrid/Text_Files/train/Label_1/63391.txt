import sys
n = int(sys.stdin.readline())
a = [[int(j) for j in sys.stdin.readline().split()] for i in range(n)]
good = []
for i in range(n):
    f = 0
    for j in range(n):
        if a[i][j] == 0:
            f = 1
        elif a[i][j] == 1:
            f = f or (a[j][i] == 2)
        elif a[i][j] == 2:
            f = f or (a[j][i] == 1)
        elif a[i][j] == 3:
            f = f or (a[j][i] == 3)
    if f:
        good.append(i+1)
print(len(good))
print(*good)
