#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, k, i, j, dp[55][55], a[55];
  cin >> n >> k;
  for(i = 1; i <= n; i++) cin >> a[i];
  for(i = 1; i <= n; i++) dp[i][1] = a[i];
  for(j = 2; j <= k; j++){
    for(i = j; i <= n; i++){
      dp[i][j] = dp[i-1][j-1]&a[i];
      for(int l = i-2; l >= j-1; l--)
        dp[i][j] = max(dp[i][j], dp[i-1][j-1]&dp[l][j-1]);
    }
  }
  cout << dp[n][k] << endl;
}
