#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long solve() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> v(n);
    for (int i = 0; i < n; i++) {
        long long a, b;
        cin >> a >> b;
        v[i].first = a;
        v[i].second = b;
    }
    long long ans = 0;
    for (int i = 1; i < n; i++) {
        if (v[i].first == v[i - 1].first) {
            long long diff = min(v[i].second, v[i - 1].second);
            ans += diff;
            v[i].first += diff;
            v[i - 1].first += diff;
        }
    }
    return ans;
}

int main() {
    int q;
    cin >> q;
    for (int i = 0; i < q; i++) {
        cout << solve() << endl;
    }
    return 0;
}
