import sys
for _ in range(int(sys.stdin.readline())):
    n=int(sys.stdin.readline())
    s=list(sys.stdin.readline().strip())
    for i in range(n-6):
        if s[i]=='?' and s[i+1]=='a' and s[i+2]=='b' and s[i+3]=='a' and s[i+4]=='c' and s[i+5]=='a' and s[i+6]=='b' and s[i+7]=='a':
            for j in range(i,i+8):
                if s[j]=='?': s[j]='?'
                else: s[j]=s[j].lower()
            break
    if '?' in s: print("No")
    else:
        print("Yes")
        print(''.join(s))
