import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;
import java.util.stream.Collectors;

class Main {
    static class City implements Comparable<City> {
        int id;
        ArrayList<City> next;
        long tax;
        long minTax;

        public City(int id) {
            this.id = id;
            next = new ArrayList<>();
            tax = Long.MAX_VALUE;
            minTax = Long.MAX_VALUE;
        }

        @Override
        public int compareTo(City o) {
            return Long.compare(tax, o.tax);
        }

        public String toString() {
            return String.format("[id=%d, tax=%d]", id, tax);
        }
    }

    public static void main(String[] args) throws IOException {
        long start = System.currentTimeMillis();
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);
        StringTokenizer st = new StringTokenizer(input.readLine());

        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        List<City> cities = new ArrayList<>(n);
        for (int i = 0; i < n; i++)
            cities.add(new City(i));

        st = new StringTokenizer(input.readLine());
        for (int i = 0; i < n; i++)
            cities.get(i).tax = Long.parseLong(st.nextToken());

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(input.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            cities.get(a).next.add(cities.get(b));
        }

        boolean canWin = true;
        for (City city : cities) {
            Queue<City> queue = new ArrayDeque<>();
            city.minTax = city.tax;
            queue.add(city);
            while (!queue.isEmpty()) {
                City current = queue.poll();
                for (City next : current.next) {
                    if (next.minTax < Math.min(current.minTax, current.tax))
                        continue;
                    next.minTax = Math.min(current.minTax, current.tax);
                    queue.add(next);
                }
            }
            if (city.minTax == city.tax)
                canWin = false;
        }

        if (canWin) {
            out.println("WIN");
            Collections.sort(cities);
            for (City city : cities)
                out.print(Math.min(city.tax, 2 * city.minTax) + " ");
        } else {
            out.println("LOSE");
        }

        out.flush();
        System.err.flush();
        long end = System.currentTimeMillis();
        System.err.printf("TIME: %.3f\n", (end - start) / 1000.0);
    }
}
