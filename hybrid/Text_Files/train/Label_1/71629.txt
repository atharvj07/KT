def find_permutation_q(n, p):
    q = [0] * n
    for i in range(n):
        if p[p[i] - 1] != i + 1:
            return [-1]
        q[p[i] - 1] = i + 1
    return q

n = int(input().strip())
p = list(map(int, input().strip().split()))
result = find_permutation_q(n, p)
print(' '.join(map(str, result)) if result[0] != -1 else "-1")
