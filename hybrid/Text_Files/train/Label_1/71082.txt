n = int(input())
s = input()

if n % 2 != 0:
    print(":(")
else:
    open_brackets, close_brackets = 0, 0
    for char in s:
        if char == "(":
            open_brackets += 1
        elif char == ")":
            close_brackets += 1

    if open_brackets > n // 2 or close_brackets > n // 2:
        print(":(")
    else:
        result = []
        question_marks = n // 2 - open_brackets
        balance = 0
        for char in s:
            if char == "?":
                if question_marks > 0:
                    result.append("(")
                    balance += 1
                    question_marks -= 1
                else:
                    result.append(")")
                    balance -= 1
            else:
                result.append(char)
                if char == "(":
                    balance += 1
                else:
                    balance -= 1

            if balance < 0 or (balance == 0 and char == "?"):
                print(":(")
                break
        else:
            if balance == 0:
                print("".join(result))
            else:
                print(":(")
