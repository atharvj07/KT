#include<bits/stdc++.h>
using namespace std;
const int maxn=1e5+1;
typedef long long ll;
ll n,q,sum=0;
ll t[maxn],k[maxn],d[maxn];

struct process{
    ll a,b,c;
}p[maxn];

bool cmp(const process &x,const process &y)
{
    return x.a<y.a;
}

int main()
{
    ll x,y,z;
    cin>>n>>q;
    for(ll i=0;i<q;i++)
    {
        cin>>t[i]>>k[i]>>d[i];
        p[i].a=t[i];p[i].b=i;p[i].c=k[i];
    }
    sort(t,t+q);
    sort(p,p+q,cmp);

    for(ll i=0,j=0;i<q;i++)
    {
        if(j<q&&t[i]==p[j].a)
        {
            if(sum>=p[j].c)
            {
                x=sum-p[j].c;
                cout<<sum<<endl;
                y=x+1;
                for(;y<=x+p[j].c;++y)
                sum-=k[p[j].b];
            }
            else cout<<-1<<endl;
            j++;
        }
        else
        {
            sum+=n;
            if(sum>=p[j].c)
            {
                x=sum-p[j].c;
                cout<<sum<<endl;
                y=x+1;
                for(;y<=x+p[j].c;++y)
                sum-=k[p[j].b];
            }
            else cout<<-1<<endl;
            j++;
        }
    }
    return 0;
}
