#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> s(n), m(n);
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> m[i];
    }
    sort(s.begin(), s.end());
    sort(m.begin(), m.end());
    int minFlicks = 0, maxFlicks = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] > m[i]) {
            minFlicks++;
        }
        if (s[i] < m[i]) {
            maxFlicks++;
        }
    }
    cout << minFlicks << endl;
    cout << maxFlicks << endl;
    return 0;
}
