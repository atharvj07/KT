import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
for _ in range(m):
    i, j = map(int, input().split())
    i, j = i - 1, j - 1
    while a[i] % 2 == 0 and a[j] % 2 == 0:
        a[i] //= 2
        a[j] //= 2
res = 0
for i in range(n):
    if a[i] % 2 == 0: res += 1
print(res)
