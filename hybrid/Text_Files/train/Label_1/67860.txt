from bisect import bisect
rl = lambda: list(map(int, input().split()))
n = int(input()); a = rl()
a.sort(reverse=1); s = sum(a); s0 = 0
for i in a: s0 += i
ans = [a[-1]]
for i in range(1, n): ans.append(a[-i-1] if i%2 else a[-1])
for k in range(n-1):
    if k%2: ans[k] = max(ans[k], s - a[ bisect(a, s0 - ans[k]) ] )
    else:   ans[k] = max(ans[k], s - s0 + a[ bisect(a, s - s0 + ans[k]) ] )
print(*ans)
