n, b, a = map(int, input().split())
s = list(map(int, input().split()))
cur, ans = b, 0
for i in range(n):
    if cur==0 and a==0: break
    if s[i]==1:
        if a:
            ans += 1
            a -= 1
        elif cur: cur -= 1
    else:
        if cur:
            ans += 1
            cur -= 1
        elif a: a -= 1
print(ans)
