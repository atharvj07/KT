#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string p, h;
        cin >> p >> h;
        map<char, int> mp;
        for (char c : p) mp[c]++;
        bool found = 1;
        int st = 0, en = h.size() - 1;
        while (st <= en) {
            if (st != en && h[st] == h[en] && mp[h[st]] * 2 > p.size()) {
                found = 0;
                break;
            }
            if (!mp[h[st]]) {
                found = 0;
                break;
            } else
                mp[h[st]]--;
            if (!mp[h[en]]) {
                found = 0;
                break;
            } else
                mp[h[en]]--;
            st++, en--;
        }
        for (auto i : mp)
            if (i.second > 0) found = 0;
        cout << (found ? "YES\n" : "NO\n");
    }
}
