import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int q = scan.nextInt();
        while(q-->0){
            int xp0 = scan.nextInt();
            int yp0 = scan.nextInt();
            int xp1 = scan.nextInt();
            int yp1 = scan.nextInt();
            int xp2 = scan.nextInt();
            int yp2 = scan.nextInt();
            int xp3 = scan.nextInt();
            int yp3 = scan.nextInt();
            boolean isIntersect = ((xp0>xp2 && xp0<xp3) || (xp1>xp2 && xp1<xp3) || (xp2>xp0 && xp2<xp1) || (xp3>xp0 && xp3<xp1)) && ((yp0>yp2 && yp0<yp3) || (yp1>yp2 && yp1<yp3) || (yp2>yp0 && yp2<yp1) || (yp3>yp0 && yp3<yp1));
            System.out.println(isIntersect?"1":"0");
        }
    }
}
