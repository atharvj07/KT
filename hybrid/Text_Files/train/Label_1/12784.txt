#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;
char s[MAXN];

void solve() {
  stack<char> st;
  int n = strlen(s);
  for (int i = 0; i < n; i++) {
    if (st.empty() || s[i] > st.top()) {
      st.push(s[i]);
    } else {
      while (!st.empty() && s[i] < st.top()) {
        cout << st.top();
        st.pop();
      }
      st.push(s[i]);
    }
  }
  while (!st.empty()) {
    cout << st.top();
    st.pop();
  }
}

int main() {
  cin >> s;
  solve();
  return 0;
}
