import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    private static final long MOD = 998244353L;

    private static class Edge {
        int u, v;

        public Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    private static long fastpow(long base, long exp) {
        if (exp == 0) {
            return 1L;
        }
        if (exp == 1) {
            return base;
        }
        long halfpow = fastpow(base, exp / 2);
        halfpow = (halfpow * halfpow) % MOD;
        if (exp % 2 == 1) {
            halfpow = (halfpow * base) % MOD;
        }
        return halfpow;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        int[][] lr = new int[n + 1][2];
        for (int i = 1; i <= n; i++) {
            line = br.readLine().split(" ");
            lr[i][0] = Integer.parseInt(line[0]);
            lr[i][1] = Integer.parseInt(line[1]);
        }

        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            edges.add(new Edge(u, v));
            edges.add(new Edge(v, u));
        }

        int[] color = new int[n + 1];

        int numBipartiteComponents = 0;
        int maxBipartiteComponentSize = 0;
        int[] bipartiteComponentSizes = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (color[i] == 0) {
                Queue<Integer> queue = new LinkedList<>();
                queue.offer(i);
                color[i] = 1;
                int bipartiteComponentSize = 0;
                while (!queue.isEmpty()) {
                    int u = queue.poll();
                    bipartiteComponentSize++;
                    for (Edge edge : edges) {
                        if (edge.u == u && color[edge.v] == 0) {
                            queue.offer(edge.v);
                            color[edge.v] = -color[u];
                        } else if (edge.v == u && color[edge.u] == 0) {
                            queue.offer(edge.u);
                            color[edge.u] = -color[u];
                        }
                    }
                }
                bipartiteComponentSizes[numBipartiteComponents] = bipartiteComponentSize;
                numBipartiteComponents++;
                maxBipartiteComponentSize = Math.max(maxBipartiteComponentSize, bipartiteComponentSize);
            }
        }

        long ans = fastpow(2, n - maxBipartiteComponentSize);
        for (int i = 1; i <= n; i++) {
            int l = lr[i][0];
            int r = lr[i][1];
            if (l > maxBipartiteComponentSize) {
                ans = 0;
                break;
            }
            ans = (ans * ((fastpow(2, r - l + 1) - 1 + MOD) % MOD)) % MOD;
        }

        System.out.println(ans);
    }
}
