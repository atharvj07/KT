def min_price(n, c1, c2, visitors):
    adults = visitors.count('1')
    groups = range(1, adults + 1)
    min_cost = float('inf')
    for x in groups:
        pupils_per_group = (n - x + (x - 1)) // x
        cost_per_group = c1 + c2 * (pupils_per_group ** 2)
        total_cost = x * cost_per_group
        min_cost = min(min_cost, total_cost)
    return min_cost

n, c1, c2 = map(int, input().split())
visitors = input().strip()
print(min_price(n, c1, c2, visitors))
