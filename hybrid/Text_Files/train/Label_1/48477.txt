import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        BigInteger a, b;
        if (scan.hasNext()) {
            a = scan.nextBigInteger();
            b = scan.nextBigInteger();
            System.out.println(a.add(b));
        }
    }
}
