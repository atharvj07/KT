#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, k, l = 0, a[1001] = {0};
    cin >> n >> k;
    while(1) {
        l++;
        for(int i = 1; i <= n; i++) {
            if(a[i] == 0) {
                a[i] = 1;
                for(int j = max(i - k, 1); j <= min(i + k, n); j++) {
                    a[j] = 1;
                }
                break;
            }
        }
        int ok = 1;
        for(int i = 1; i <= n; i++) if(a[i] == 0) ok = 0;
        if(ok) break;
    }
    cout << l << endl;
    for(int i = 1; i <= n; i++) for(int j = 1; j <= n; j++) if(a[j] && i == j) cout << j << " ";
    cout << endl;
    return 0;
}
