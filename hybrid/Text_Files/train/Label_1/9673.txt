#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, q;
  cin >> n >> q;
  vector<vector<int>> grid(3, vector<int>(n + 1));
  while (q--) {
    int r, c;
    cin >> r >> c;
    grid[r][c] ^= 1;
    cout << (grid[1][c] || grid[2][c - 1] || grid[2][c + 1] ? "No\n" : "Yes\n");
  }
}
