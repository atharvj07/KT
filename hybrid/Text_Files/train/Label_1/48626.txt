import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        City[] cities = new City[n];
        for (int i = 0; i < n; i++) {
            cities[i] = new City(input.nextInt(), input.next().charAt(0));
        }
        Queue<City> byteQueue = new LinkedList<>();
        Queue<City> berQueue = new LinkedList<>();
        long cost = 0;
        for (City city : cities) {
            if (city.type == 'B') {
                if (!byteQueue.isEmpty()) {
                    cost += city.x - byteQueue.peek().x;
                    byteQueue.poll();
                }
                byteQueue.add(city);
            } else if (city.type == 'R') {
                if (!berQueue.isEmpty()) {
                    cost += city.x - berQueue.peek().x;
                    berQueue.poll();
                }
                berQueue.add(city);
            }
        }
        System.out.println(cost);
    }

    static class City {
        int x;
        char type;

        public City(int x, char type) {
            this.x = x;
            this.type = type;
        }
    }
}
