#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9+7;
vector<int>adj[123456];
ll dp[123456][650];
int32_t main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    string s;
    cin>>s;
    for(int i=s.size()-1;i>=0;i--) 
    {
        int c=s[i]-'a';
        for(int j=0;j<26;++j) 
        {
            dp[i][j]=dp[i+1][j];
        }
        dp[i][c]++;
        for(int j=1;j<26;++j) 
        {
            dp[i][j]+=dp[i+1][j-1];
        }
        adj[c].push_back(i);
    }
    ll ans=0;
    for(int i=0;i<26;++i) 
    {
        for(int id=0;id<adj[i].size();++id) 
        {
            int st=adj[i][id];
            ll len=1,cnt=dp[st][i]-dp[st+1][i-1];
            for(int j=i+1;j<26;++j)
            {
                int idx=lower_bound(adj[j].begin(),adj[j].end(),st+len)-adj[j].begin();
                if(idx==adj[j].size()) break;
                int pos=adj[j][idx];
                cnt=cnt*(dp[pos][j]-dp[pos+1][j-1])%mod;
                len++;
            }
            ans=(ans+cnt)%mod;
        }
    }
    cout<<ans<<endl;
}
