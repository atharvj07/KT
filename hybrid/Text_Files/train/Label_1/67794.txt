n, m, k = map(int, input().split())
grid = [[0] * m for _ in range(n)]
for _ in range(k):
    x, y = map(int, input().split())
    x, y = x - 1, y - 1
    grid[x][y] ^= 1
    if x + 1 < n and y + 1 < m and grid[x + 1][y] and grid[x][y + 1] and grid[x + 1][y + 1]:
        print(_) + 1
        exit()
print(0)
