s = input()

n_count = s.count('n')
i_count = s.count('i')
e_count = s.count('e')
t_count = s.count('t')

# Since 'n' occurs twice in "nineteen", we divide by 2 but since we need at least 3 'n's to form one "nineteen", we adjust the formula.
# For 'e', it occurs thrice in "nineteen", so we divide by 3.
nineteen_count = min(n_count // 3, i_count, e_count // 3, t_count)

print(nineteen_count)
