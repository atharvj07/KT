import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int T1 = scanner.nextInt();
        int T2 = scanner.nextInt();
        int R1 = scanner.nextInt();
        int R2 = scanner.nextInt();

        // Check if both players have AOJ rates
        if (R1 == -1 && R2 == -1) {
            // Compare 50m sprint times
            if (T1 < T2) {
                System.out.println("Alice");
            } else if (T1 > T2) {
                System.out.println("Bob");
            } else {
                System.out.println("Draw");
            }
        } else if (R1 == -1) {
            // If only Bob has an AOJ rate, he wins
            System.out.println("Bob");
        } else if (R2 == -1) {
            // If only Alice has an AOJ rate, she wins
            System.out.println("Alice");
        } else {
            // Compare AOJ rates
            if (R1 > R2) {
                System.out.println("Alice");
            } else if (R1 < R2) {
                System.out.println("Bob");
            } else {
                // If AOJ rates are the same, compare 50m sprint times
                if (T1 < T2) {
                    System.out.println("Alice");
                } else if (T1 > T2) {
                    System.out.println("Bob");
                } else {
                    System.out.println("Draw");
                }
            }
        }
    }
}
