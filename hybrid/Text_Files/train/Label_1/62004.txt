n=int(input())
a=list(map(int,input().split()))
m=int(input())
b=list(map(int,input().split()))
c=list(map(int,input().split()))
d={x:0 for x in a}
for x,y in zip(b,c):
    d[x]+=1
    d[y]+=1
movie1=max(d.values())
movies=[i for i in d.values() if i==movie1]
mp1=0
ind1=0
for i in range(len(movies)):
    if movies[i]==movie1:
        k=[key for key,value in d.items() if value==movies[i]]
        mp=0
        for x in k:
            mp+=a.count(x)
        if mp>mp1:
            mp1=mp
            ind1=i
print(ind1+1)
