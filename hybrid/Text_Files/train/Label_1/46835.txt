import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = a[i - 1];
            if(s.charAt(i - 1) == '0') {
                a[i]++;
            }
        }
        int ans = n;
        for (int i = 2; i <= n; i++) {
            if(s.charAt(i - 2) != s.charAt(i - 1)) {
                ans = Math.min(ans, n - i + a[i - 1] + a[n] - a[i]);
            }
        }
        System.out.println(ans);
    }
}
