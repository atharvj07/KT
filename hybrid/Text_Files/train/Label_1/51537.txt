import java.util.Scanner;
import java.util.LinkedHashSet;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String cipher = scanner.next().toLowerCase();
        int m = scanner.nextInt();
        Set<String> words = new LinkedHashSet<>();
        for (int i = 0; i < m; i++) {
            words.add(scanner.next().toLowerCase());
        }
        StringBuilder sb = new StringBuilder();
        int start = 0;
        int end = 0;
        while (end < cipher.length()) {
            while (end < cipher.length() && !Character.isLetter(cipher.charAt(end))) {
                end++;
            }
            if (end == cipher.length()) {
                break;
            }
            start = end;
            while (end < cipher.length() && Character.isLetter(cipher.charAt(end))) {
                end++;
            }
            String word = new StringBuilder(cipher.substring(start, end)).reverse().toString();
            if (words.contains(word)) {
                sb.append(word);
            } else {
                sb = new StringBuilder();
                break;
            }
            end++;
        }
        System.out.print(sb.toString());
    }
}
