#include <bits/stdc++.h>
using namespace std;
const int nax = 2e5 + 5;
long long a[nax], b[nax];
long long k,  aa, bb, cc, dd , ans;
int main() {
    cin >> k >> aa >> bb >> cc >> dd;
    if (aa > cc) swap(aa, cc), swap(bb, dd);
    if (4 * aa != cc + dd) {
        cout << 0 << endl;
        return 0;
    }
    long long x = (cc - dd) / 4;
    b[2] = x;
    a[2] = aa - x;
    if (dd % 2 == 1) {
        bb = (dd + 1) / 2;
        dd = (dd - 1) / 2;
    } else {
        bb = dd / 2;
    }
    a[1] = bb;
    b[3] = bb;
    a[3] = cc - bb;
    b[1] = dd;
    if (k <= 3 || a[1] <= 0 || b[1] <= 0 || a[2] <= 0 || b[2] <= 0 || a[3] <= 0 || b[3] <= 0) {
        cout << 0 << endl;
    } else {
        cout << 1 << endl;
    }
    return 0;
}
