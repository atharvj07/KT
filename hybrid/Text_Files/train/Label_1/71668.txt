n, m, s = map(int, input().split())
a = [[] for i in range(n+1)]
for i in range(m):
    x, y = map(int, input().split())
    a[x].append(y)
vis = [False] * (n+1)
vis[s] = True
ans = 0
for i in range(1, n+1):
    if vis[i]:
        for j in a[i]:
            if not vis[j]:
                ans += 1
                vis[j] = True
print(ans)
