#include <iostream>
#include <vector>
using namespace std;

int main() {
    int c[4], n, m, tmp, ans = 1e9;
    vector<int> a, b;
    cin >> c[0] >> c[1] >> c[2] >> c[3];
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> tmp;
        a.push_back(tmp);
    }
    for (int i = 0; i < m; i++) {
        cin >> tmp;
        b.push_back(tmp);
    }
    for (int i = 0; i < (1 << 2); i++) {
        int sum = 0;
        for (int j = 0; j < n; j++)
            sum += (i & 1) ? min(a[j], c[1]) : min(c[0] * a[j], c[2]);
        for (int j = 0; j < m; j++)
            sum += (i >> 1 & 1) ? min(b[j], c[1]) : min(c[0] * b[j], c[2]);
        ans = min(ans, sum);
    }
    cout << ans << endl;
}
