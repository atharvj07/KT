import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        long[] t = new long[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextLong();
        }
        Arrays.sort(t);
        long res = 0;
        for (int i = 0; i < k; i++) {
            res += t[2 * i + 1] - t[2 * i];
        }
        System.out.println(res);
    }
}
