for _ in range(int(input())):
    n, k = map(int, input().split())
    h = list(map(int, input().split()))
    for i in range(1, n):
        if k <= 0:
            break
        if h[i] > h[i - 1]:
            k -= 1
            if k <= 0:
                print(i)
                break
        else:
            h[i - 1] += 1
    else:
        print(-1)
