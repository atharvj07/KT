#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {

	string commandLine;
	getline(cin, commandLine);

	vector<string> lexemes;
	bool inQuotes = false;
	string currentLexeme;
	
	for (char c : commandLine) {
		if (c == '"') {
			inQuotes = !inQuotes;
			if (!inQuotes) {
				lexemes.push_back(currentLexeme);
				currentLexeme = "";
			}
		} else if (inQuotes) {
			currentLexeme += c;
		} else if (isspace(c)) {
			lexemes.push_back(currentLexeme);
			currentLexeme = "";
		} else {
			currentLexeme += c;
		}
	}
	if (!currentLexeme.empty()) {
		lexemes.push_back(currentLexeme);
	}

	for (string lexeme : lexemes) {
		cout << "<" << lexeme << ">" << endl;
	}

	return 0;
}
