#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
int arr[maxn];
int32_t main() {
  int n;
  cin >> n;
  for (int i = 1; i <= 2 * n; ++i) {
    int x;
    cin >> x;
    ++arr[x];
  }
  int ans = 0;
  for (int i = 0; i <= n; ++i) {
    ans += (arr[i] + 1) / 2;
  }
  cout << ans;
  return 0;
}
