#include <bits/stdc++.h>
typedef long long ll;
#define all(c) c.begin(),c.end()
#define pb push_back
#define fi first
#define se second
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    vector<ll> a(n);
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    vector<int> ind(k);
    for(int i=0;i<k;i++){
        cin>>ind[i];
        ind[i]--;
    }
    sort(all(a));
    sort(all(ind));
    ind.pb(0);
    ind.pb(n-1);
    ll ans=0;
    int j=0;
    for(int i=n-1;i>=0;i--){
        while(j+1<k and ind[j+1]<i){
            j++;
        }
        ans+=a[i]*a[ind[j]];
    }
    cout<<ans<<'\n';
    return 0;
}
