#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<long long> a(n);
  unordered_map<long long, pair<int, int>> mp;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (mp.find(a[i]) != mp.end()) {
      cout << "YES" << endl;
      cout << mp[a[i]].first + 1 << " " << mp[a[i]].second + 1 << " " << i + 1 << " ";
      int t = -1;
      for (int j = 0; j < n; j++) {
        if (i == j)
          continue;
        if (a[i] + a[j] == a[i] + a[i]) {
          t = j;
          break;
        }
      }
      cout << t + 1 << endl;
      return 0;
    }
    mp[a[i]] = make_pair(i, 0);
    for (int j = 0; j < i; j++) {
      mp[a[i]] = make_pair(
          min(mp[a[i]].first, j), max(mp[a[i]].second, i));
      mp[a[i] + a[j]] = make_pair(
          min(mp[a[i] + a[j]].first, j),
          max(mp[a[i] + a[j]].second, i));
    }
  }
  cout << "NO" << endl;
  return 0;
}
