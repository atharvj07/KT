#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<n;i++)
using namespace std;
const int INF=1e9;
typedef long long ll;
typedef pair<ll,int> P;
const ll MOD=1e9+7;
typedef vector<int> vi;
typedef vector<ll> vll;
const ll MAX=1e5+10;
vi G[MAX];
ll dp[MAX][4];
bool ok(int c1,int c2,int c3){
	return ((c1==c2 and c1==c3) or (c1==c2 or c2==c3 or c1==c3));
}
void dfs(int cur,int par){
	dp[cur][1]=G[cur].size()>1?INF:0;
	dp[cur][2]=G[cur].size()>1?INF:0;
	dp[cur][3]=G[cur].size()>1?INF:0;
	for(int nxt:G[cur]){
		if(nxt==par)continue;
		dfs(nxt,cur);
		for(int i=1;i<=3;i++){
			for(int j=1;j<=3;j++){
				if(ok(i,j,dp[nxt][3-i-j]))dp[cur][i]=min(dp[cur][i],dp[cur][3-i-j]+dp[nxt][j]);
			}
		}
	}
	for(int i=1;i<=3;i++)dp[cur][i]+=i;
}
int main(){
	int n;cin>>n;
	rep(i,n){
		ll c1,c2,c3;cin>>c1>>c2>>c3;
		rep(j,G[i].size())dp[i][j]=INF;
		dp[i][1]=c1;
		dp[i][2]=c2;
		dp[i][3]=c3;
	}
	rep(i,n-1){
		int a,b;cin>>a>>b;a--;b--;
		G[a].push_back(b);
		G[b].push_back(a);
	}
	dfs(0,-1);
	if(dp[0][1]<INF){
		cout<<dp[0][1]<<endl;
		vector<int> color(n,0);
		int cur=0,c=0;
		while(1){
			int mi=INF,id=-1;
			rep(i,3){
				if(mi>dp[cur][i+1] and dp[cur][i+1]!=INF){
					mi=dp[cur][i+1];
					id=i+1;
				}
			}
			if(id!=-1){
				color[cur]=id;
				c+=id;
				for(int nxt:G[cur]){
					if(color[nxt]==0)
						dp[nxt][id]=INF;
				}
			}
			int ncnt=0;
			rep(i,G[cur].size()){
				if(color[G[cur][i]]==0)ncnt++;
			}
			if(ncnt==0)break;
			for(int nxt:G[cur]){
				if(color[nxt]==0){
					cur=nxt;
					break;
				}
			}
		}
		rep(i,n)cout<<color[i]<<" ";
		cout<<endl;
	}
	else{
		cout<<"-1"<<endl;
	}
}
