#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, a[3][200005];
    cin >> n;
    for (int i = 0; i < 2; ++i)
      for (int j = 0; j < n; ++j) cin >> a[i][j];
    int ok = 1;
    for (int j = 0; j < n; ++j) ok &= (a[0][j] == 3) || (a[0][j] == 4);
    ok &= (a[1][0] == 5) || (a[1][0] == 6);
    ok &= (a[1][n - 1] == 1) || (a[1][n - 1] == 2);
    for (int j = 0; j < n; ++j) ok &= (a[1][j] == 1) || (a[1][j] == 4) || (a[1][j] == 6);
    if (ok) cout << "YES\n"; else cout << "NO\n";
  }
}
