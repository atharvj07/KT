#include <bits/stdc++.h>
using namespace std;

int n, k, a[505], b[505], vis[505];
vector<pair<int, char>> ans;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> k;
    for (int i = 1; i <= k; i++) cin >> b[i];
    for (int i = 1; i <= k; i++) {
        if (b[i] != a[i]) {
            bool f = 0;
            for (int j = 1; j <= n; j++) {
                if (vis[j] || (vis[j - 1] && a[j] != b[i])) continue;
                if (a[j] > a[j - 1]) {
                    a[j] += a[j - 1];
                    a[j - 1] = 0;
                    vis[j - 1] = 1;
                    ans.push_back({j, 'L'});
                    f = 1;
                    break;
                }
            }
            if (!f) {
                for (int j = n; j >= 1; j--) {
                    if (vis[j] || (vis[j + 1] && a[j] != b[i])) continue;
                    if (a[j] > a[j + 1]) {
                        a[j] += a[j + 1];
                        a[j + 1] = 0;
                        vis[j + 1] = 1;
                        ans.push_back({j, 'R'});
                        f = 1;
                        break;
                    }
                }
            }
            if (!f) {
                cout << "NO\n";
                return 0;
            }
        }
    }
    cout << "YES\n";
    for (auto i : ans) {
        cout << i.first << ' ' << i.second << '\n';
    }
}
