#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n, vector<int>(m)), b(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> a[i][j]; }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> b[i][j]; }
  }
  for (int x1 = 0; x1 < n - 1; x1++) {
    for (int y1 = 0; y1 < m - 1; y1++) {
      for (int x2 = x1 + 1; x2 < n; x2++) {
        for (int y2 = y1 + 1; y2 < m; y2++) {
          int sum = 0;
          sum += a[x1][y1] ^ a[x1][y2] ^ a[x2][y1] ^ a[x2][y2];
          sum += b[x1][y1] ^ b[x1][y2] ^ b[x2][y1] ^ b[x2][y2];
          if (sum % 2 == 1) { cout << "No" << endl; return 0; }
        }
      }
    }
  }
  cout << "Yes" << endl;
}
