import java.util.*;
import java.io.*;

public class Main {
    // Read input and create an array a
    static int[] readInput() throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] input = br.readLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(input[i]);
        }
        return a;
    }

    // Perform the operation and return the final sequence
    static int[] solve(int[] a) {
        int n = a.length;
        // Create a stack to store the final sequence
        Stack<Integer> stack = new Stack<>();

        // Iterate over the array and perform the operation
        for (int i = 0; i < n; i++) {
            // If the current element is not equal to the top of the stack, push it onto the stack
            if (stack.isEmpty() || stack.peek() != a[i]) {
                stack.push(a[i]);
            }
            // Otherwise, pop the top of the stack and increment it by 1
            else {
                stack.pop();
                stack.push(a[i] + 1);
            }
        }

        // Create an array to store the final sequence
        int[] result = new int[stack.size()];

        // Pop all the elements from the stack and store them in the array
        for (int i = result.length - 1; i >= 0; i--) {
            result[i] = stack.pop();
        }

        return result;
    }

    // Print the output
    static void printOutput(int[] result) {
        System.out.println(result.length);
        for (int x : result) {
            System.out.print(x + " ");
        }
        System.out.println();
    }

    public static void main(String[] args) throws IOException {
        int[] a = readInput();
        int[] result = solve(a);
        printOutput(result);
    }
}
