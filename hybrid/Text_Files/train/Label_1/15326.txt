#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main() {
    int n, x, ans = -1;
    cin >> n >> x;
    --x;
    vector<int> v(n);
    for (int i = 0; i < n; ++i) { cin >> v[i]; v[i] &= x; ans = max(ans, v[i]); }
    if (v[0] < ans) ans = -1;
    else {
        int inv = 0;
        for (int i = 1; i < n; ++i) if (v[i] != ans) ++inv;
        ans = inv;
    }
    cout << ans << '\n';
    return 0;
}
