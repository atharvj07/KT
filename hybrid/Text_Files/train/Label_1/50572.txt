import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String []args){
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-->0){
            int n = sc.nextInt();
            int[] a = new int[n];
            for(int i=0;i<n;i++) a[i] = sc.nextInt();
            boolean flag = true;
            for(int i=0;i<n;i++){
                for(int j=0;j<n;j++){
                    if(i==j) continue;
                    if(Math.abs(a[i]-a[j])>1) flag=false;
                }
            }
            System.out.println(flag?"YES":"NO");
        }
    }
}
