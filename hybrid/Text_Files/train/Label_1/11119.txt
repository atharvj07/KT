#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    vector<pair<int, int>> ans;
    for (int a = 1; a <= 12; a++) {
      for (int b = 1; b <= 12; b++) {
        if (a * b == 12) {
          bool ok = true;
          for (int i = 0; i < a; i++) {
            bool hasX = false;
            for (int j = 0; j < b; j++) { hasX |= (s[i * b + j] == 'X'); }
            ok &= hasX;
          }
          if (ok) ans.push_back({a, b});
        }
      }
    }
    cout << ans.size() << ' ';
    for (auto p : ans) cout << p.first << 'x' << p.second << ' ';
    cout << '\n';
  }
  return 0;
}
