from collections import defaultdict

for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    c = list(map(int, input().split()))
    mp = defaultdict(list)
    for i in range(n):
        mp[c[i]].append(i)
    ans = n + 1
    for i in range(1, n):
        cur = []
        for x in mp[i]:
            cur.append(p[x])
        mp[i] = cur
        if len(set(cur)) == 1:
            ans = i
            break
    print(ans)
