#include<bits/stdc++.h>

using namespace std;

int main() {
    int t;  cin >> t;
    while (t--) {
        int x, y, z;  cin >> x >> y >> z;
        if (x + y == z) { cout << "NO" << endl; continue; }
        if (y + z == x) { cout << "NO" << endl; continue; }
        if (z + x == y) { cout << "NO" << endl; continue; }
        cout << "YES" << endl;
        vector<int> v = {x, y, z};
        sort(v.begin(), v.end());
        cout << v[2] << " " << v[1] << " " << v[0] << endl;
    }
}
