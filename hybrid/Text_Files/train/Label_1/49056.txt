import java.util.*;
public class Main {
     public static void main(String[] args) {
          Scanner in=new Scanner(System.in);
          long f=in.nextInt(),t=in.nextInt(),t0=in.nextInt(),t1=in.nextInt(),p1=in.nextInt(),a1=in.nextInt(),t2=in.nextInt(),p2=in.nextInt(),a2=in.nextInt(),ans=Long.MAX_VALUE;
          for(long i=0;i<=f/a1;++i){
               long sum=p1*i;
               long rem=f-i*a1;
               long rem1=rem/a2;
               rem=rem%a2;
               long rem2=rem/t0;
               rem=rem%t0;
               sum+=rem1*p2;
               sum+=Math.ceil(rem2+1e-9)*t0;
               if(sum<ans)ans=sum;
          }
          for(long j=0;j<=f/a2;++j){
               long sum=p2*j;
               long rem=f-j*a2;
               long rem1=rem/a1;
               rem=rem%a1;
               long rem2=rem/t0;
               rem=rem%t0;
               sum+=rem1*p1;
               sum+=Math.ceil(rem2+1e-9)*t0;
               if(sum<ans)ans=sum;
          }
          long rem=f/t0;
          rem=f%t0;
          ans=Math.min(ans,rem*t0);
          if(ans==Long.MAX_VALUE)System.out.println(-1);
          else System.out.println(ans);
    }
}
