import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] dp = new int[n + 1];
        int max = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                dp[i + 1] = dp[i] + 1;
                max = Math.max(max, dp[i + 1]);
            } else {
                dp[i + 1] = dp[i];
            }
        }

        int[] ans = new int[n];
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] == 0 && k > 0 && dp[i + 1] + 1 > max) {
                a[i] = 1;
                max++;
                k--;
            }
            ans[i] = a[i];
        }

        System.out.println(max);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
