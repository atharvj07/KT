n = int(input())
k1, *a = map(int, input().split())
k2, *b = map(int, input().split())
c1, c2 = 0, 0
for i in range(n):
  if a:
    x = a.pop(0)
    c1 += 1
  else:
    x = b.pop(0)
    c2 += 1
  if not b:
    print(c1 + c2, 1)
    break
  if not a:
    print(c1 + c2, 2)
    break
  if x > b[0]:
    a.append(b.pop(0))
    a.append(x)
  else:
    b.append(a.pop(0))
    b.append(x)
else:
  print(-1)
