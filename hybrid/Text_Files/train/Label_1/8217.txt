#include <iostream>
using namespace std;
long long a,b;
int main(){
    cin>>a>>b;
    if(a==b){
        cout<<"YES"<<endl<<"1"<<endl<<a<<endl;
        return 0;
    }
    if((b-a)%2){
        cout<<"NO"<<endl;
        return 0;
    }
    a*=5;
    b*=5;
    cout<<"YES"<<endl<<"3"<<endl<<a<<endl<<b<<endl<<b+2<<endl;
    return 0;
}
