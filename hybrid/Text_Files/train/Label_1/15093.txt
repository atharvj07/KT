#include <iostream>

using namespace std;

int main() {
    long long n, h, a, b, k;
    cin >> n >> h >> a >> b >> k;
    for (long long i = 0; i < k; i++) {
        long long ta, fa, tb, fb;
        cin >> ta >> fa >> tb >> fb;
        if (ta == tb) { cout << abs(fa - fb) << endl; }
        else {
            if (fa < a) fa = a;
            if (fa > b) fa = b;
            if (fb < a) fb = a;
            if (fb > b) fb = b;
            long long dist = abs(fa - fb) + abs(ta - tb);
            cout << dist << endl;
        }
    }
    return 0;
}
