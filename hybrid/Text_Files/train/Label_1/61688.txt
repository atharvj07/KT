from fractions import Fraction

a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

if a1 * b1 != a2 * b2:
    print(-1)
else:
    p = a1 * b1
    if p % 2 == 0:
        a1 //= 2
        b1 //= 2
    if p % 3 == 0:
        a1 //= 3
        b1 //= 3
        a2 //= 3
        b2 //= 3
    g = gcd(a1, b1)
    a1 //= g
    b1 //= g
    if a2 % a1 == 0 and b2 % b1 == 0:
        a2 //= a1
        b2 //= b1
        print(1)
        print(f"{a1} {b1}")
        print(f"{a2} {b2}")
    else:
        print(-1)
