import sys
q=int(input());
for __ in range(q):
	s,t=input(),input()
	if len(s)>len(t):
		s,t=t,s
	l1,l2=len(s),len(t)
	if s+s==t or t+t==s:
		print(s)
		continue
	if l2%l1 and l1%l2:
		print(-1)
		continue
	L=l1 if l2%l1==0 else l2
	r=''
	for i in range(L):
		r+=s[i%l1]
	if r+r==t:
		print(r)
		continue
	print(-1)
