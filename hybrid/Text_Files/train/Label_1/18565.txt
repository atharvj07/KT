#include<iostream>
using namespace std;
#define mo 1000000009
int n, a, b, k;
string s;
int main() {
    cin >> n >> a >> b >> k;
    cin >> s;
    long long ans = 0;
    for (int i = 0; i < k; ++i) {
        if (s[i] == '+') ans = ((ans * b % mo) + a) % mo;
        else ans = ((ans * b % mo) - a + mo) % mo;
    }
    for (int i = k; i <= n; ++i) {
        if (s[i % k] == '+') ans = ((ans * b % mo) + a) % mo;
        else ans = ((ans * b % mo) - a + mo) % mo;
        if ((ans % 10 == 0) && (ans / 10 % 10 == 0)) ans = ans * 9 / 10 % mo, i = n + 1;
    }
    cout << ans << endl;
    return 0;
}
