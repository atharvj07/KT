import sys

n = int(input())

def isprime(a):
    if a < 2: return False
    if any(a%x == 0 for x in range(2,round(a**0.5)+1)): return False
    else: return True

x, c = 2, 0
for i in range(2,1000000):
    if n%i==0 and isprime(i):
        c = i
        break

x0 = n
while True:
    x0 -=1
    if x0%c == 0: break

print(x0)
