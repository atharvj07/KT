import java.util.Scanner;
import java.math.BigInteger;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            BigInteger a = in.nextBigInteger();
            BigInteger b = in.nextBigInteger();
            if (b.compareTo(a) < 0) {
                System.out.println(-1);
            } else {
                int operations = 0;
                while (a.compareTo(b) < 0) {
                    if (b.mod(BigInteger.valueOf(8)).compareTo(BigInteger.ZERO) == 0) {
                        b = b.divide(BigInteger.valueOf(8));
                        operations++;
                    } else if (b.mod(BigInteger.valueOf(4)).compareTo(BigInteger.ZERO) == 0) {
                        b = b.divide(BigInteger.valueOf(4));
                        operations++;
                    } else if (b.mod(BigInteger.valueOf(2)).compareTo(BigInteger.ZERO) == 0) {
                        b = b.divide(BigInteger.valueOf(2));
                        operations++;
                    } else {
                        operations = -1;
                        break;
                    }
                }
                System.out.println(operations);
            }
        }
    }
}
