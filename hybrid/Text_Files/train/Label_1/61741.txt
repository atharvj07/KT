import sys
input=sys.stdin.readline
n=int(input())
g=[[]for i in range(n)]
for i in range(n-1):
    a,b=map(int,input().split())
    a-=1
    b-=1
    g[a].append(b)
    g[b].append(a)
h=[0]*n
ans=[0]*(n-1)
def dfs(at,la):
    for to in g[at]:
        if la==to:continue
        ans[h[at]]=h[to]=h[at]+1
        dfs(to,at)
h[0]=1
dfs(0,-1)
print(*ans)
