def solve(s, t):
    n, m = len(s), len(t)
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + (s[i] == '?')
    dp = [-1] * (n + 1)
    dp[0] = 0
    for i in range(1, n + 1):
        dp[i] = dp[i - 1]
        if i >= m:
            matches = True
            cost = 0
            for j in range(m):
                if s[i - j - 1] != '?' and s[i - j - 1] != t[m - j - 1]:
                    matches = False
                    break
                if s[i - j - 1] == '?':
                    cost += 1
            if matches and dp[i - m] >= 0:
                dp[i] = max(dp[i], dp[i - m] + 1 - (prefix[i] - prefix[i - m] - cost))
    return max(dp)

s = input().strip()
t = input().strip()
print(solve(s, t))
