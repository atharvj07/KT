#include <bits/stdc++.h>
#define nl '\n'
using namespace std;

bool marked[55][55];
char a[55], b[55];
int n, cnt;
vector<pair<int, int>>ans;

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int t; cin >> t;
    while(t--) {
        cin >> n;
        cin >> a >> b;
        bool flag = true;
        cnt = 0;
        memset(marked, false, sizeof marked);
        ans.clear();

        for(int i = 0; i < n; i++) {
            if(a[i] == b[i]) {
                continue;
            }
            bool found = false;
            for(int j = i+1; j < n; j++) {
                if(a[i] == b[j] && !marked[i][j]) {
                    ans.push_back({i+1, j+1});
                    marked[i][j] = true;
                    cnt++;
                    found = true;
                    break;
                }
                if(b[i] == a[j] && !marked[j][i]) {
                    ans.push_back({j+1, i+1});
                    marked[j][i] = true;
                    cnt++;
                    found = true;
                    break;
                }
            }
            if(!found) {
                flag = false;
                break;
            }
        }
        if(flag) cout << "Yes" << nl;
        else cout << "No" << nl;
        if(flag) {
            cout << cnt << nl;
            for(auto x: ans) {
                cout << x.first << ' ' << x.second << nl;
            }
        }
    }
    return 0;
}
