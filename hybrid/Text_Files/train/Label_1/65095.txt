import sys

n, m = list(map(int, input().split()))
a = [0] * n
for _ in range(m):
    t, l, r, x = list(map(int, input().split()))
    l -= 1
    r -= 1
    if t == 1: a[l] += x
    else:
        if not l:
            if a[r] < x: a[r] = x
            continue
        if a[l - 1] + x < x:  a[l - 1] += x
if a[n - 1] >= 0: sys.stdout.write('YES\n')
else: sys.stdout.write('NO\n')
for i in range(1, n): a[i] = max(a[i], a[i - 1])
for i in a: sys.stdout.write(f'{i} ')
