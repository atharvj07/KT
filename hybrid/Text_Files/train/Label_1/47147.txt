import java.util.Scanner;

public class CupcakeShop {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            long[] a = new long[n];

            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextLong();
            }

            long prefixSum = 0;
            long maxSuffixSum = Integer.MIN_VALUE;

            for (int i = n - 1; i >= 0; i--) {
                prefixSum += a[i];
                maxSuffixSum = Math.max(maxSuffixSum, prefixSum);
            }

            System.out.println(maxSuffixSum > 0 ? "YES" : "NO");
        }
    }
}
