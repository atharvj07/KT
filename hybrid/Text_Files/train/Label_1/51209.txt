import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int m = sc.nextInt();
        int y = sc.nextInt();
        int ans = n + m + 4;
        if (x == y)
            ans -= 2;
        System.out.println(ans);
    }
}
