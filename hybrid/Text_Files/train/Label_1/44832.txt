import java.util.*;

class Main {
    static class Portal {
        int u;
        int v;

        Portal(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Castle {
        int a;
        int b;
        int c;

        Castle(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        Castle[] castles = new Castle[n];
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            int c = scanner.nextInt();
            castles[i] = new Castle(a, b, c);
        }
        Portal[] portals = new Portal[m];
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            portals[i] = new Portal(u, v);
        }
        int[] dp = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dp[i] = -1;
        }
        dp[1] = k;
        for (int i = 1; i < n; i++) {
            if (dp[i] == -1) {
                continue;
            }
            int army = dp[i];
            if (castles[i - 1].a > army) {
                continue;
            }
            army -= castles[i - 1].a;
            army += castles[i - 1].b;
            for (Portal portal : portals) {
                if (portal.u == i) {
                    if (dp[portal.v] == -1 || dp[portal.v] < army) {
                        dp[portal.v] = army;
                    }
                }
            }
            if (dp[i + 1] == -1 || dp[i + 1] < army) {
                dp[i + 1] = army;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (dp[i] == -1) {
                ans = -1;
                break;
            }
            ans += castles[i - 1].c;
        }
        System.out.println(ans);
    }
}
