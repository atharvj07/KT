import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt();
        int[] l = new int[N], r = new int[N];
        for (int i = 0; i < N; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        int[] ans = new int[M];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < M; j++) {
                if (l[i] <= j + 1 && j + 1 <= r[i]) ans[j]++;
            }
        }
        for (int i = 0; i < M; i++) {
            System.out.println(ans[i]);
        }
    }
}
