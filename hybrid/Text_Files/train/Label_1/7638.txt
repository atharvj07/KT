#include <iostream>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int ans = 0;
        while (!s.empty()) {
            ++ans;
            s.erase(s.begin());
            int i = 0;
            while (i < (int)s.length() && s[i] == s[0]) ++i;
            s.erase(s.begin(), s.begin() + i);
        }
        cout << ans << endl;
    }
    return 0;
}
