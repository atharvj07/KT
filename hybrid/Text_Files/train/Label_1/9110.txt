#include <iostream>
#include <string>
using namespace std;

int solve() {
  string s;
  cin >> s;
  int res = 0;
  bool prev = false;
  for (char c : s) {
    if (!prev) {
      res++;
      prev = true;
    }
    if (c != s[0]) prev = false;
  }
  return res;
}
int main() { cout << solve() << endl; }
