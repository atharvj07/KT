#include <bits/stdc++.h>
using namespace std;
#define REP(i,n) for(int i = 0; i < n; ++i)
#define FOR(i,a,b) for(int i = a; i < b; ++i)
#define endl "\n"

int solve() {
  int n; cin >> n;
  string s, t; cin >> s >> t;
  vector<pair<int, int>> ans;
  int cnt = 0;
  for(int i = 0; i < n; ++i) {
    if(s[i] == t[i]) continue;
    if(cnt % 2 == 0) {
      for(int j = i + 1; j < n; ++j) {
        if(s[j] == s[i] && t[j] == t[i]) {
          ans.push_back({j, i});
          swap(s[j], t[j]);
          ++cnt;
          break;
        }
      }
    } else {
      for(int j = i + 1; j < n; ++j) {
        if(s[j] == t[i] && t[j] == s[i]) {
          ans.push_back({i, j});
          swap(s[i], t[i]);
          ++cnt;
          break;
        }
      }
    }
    if(cnt == 2 * n) break;
  }
  bool flag = 1;
  for(int i = 0; i < n && flag; ++i) {
    if(s[i] != t[i]) flag = 0;
  }
  if(!flag) {
    cout << "No\n";
    return 0;
  }
  cout << "Yes" << endl;
  cout << ans.size() << endl;
  for(auto &p: ans) {
    cout << p.first + 1 << " " << p.second + 1 << endl;
  }
  return 1;
}

int main() {
  int t; cin >> t;
  while(t--) {
    solve();
  }
  return 0;
}
