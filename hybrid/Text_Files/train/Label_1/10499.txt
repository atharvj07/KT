#include <iostream>

#define ll long long
#define mod 1000000007

using namespace std;

int main() {
  int t, k;
  cin >> t >> k;
  while (t--) {
    int a, b;
    cin >> a >> b;
    ll ans = 0;
    for (int i = a; i <= b; i++) {
      ll dp[i + 1][k + 1];
      for (int j = 0; j <= i; j++) {
        for (int l = 0; l <= k; l++) {
          dp[j][l] = 0;
        }
      }
      dp[0][0] = 1;
      for (int j = 1; j <= i; j++) {
        for (int l = 0; l <= k; l++) {
          dp[j][l] = (dp[j - 1][l] * 2) % mod;
          if (l > 0) dp[j][l] = (dp[j][l] + dp[j - 1][l - 1]) % mod;
        }
      }
      ans = (ans + dp[i][k]) % mod;
    }
    cout << ans << endl;
  }
  return 0;
}
