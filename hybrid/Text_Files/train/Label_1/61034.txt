def max_removals(t, p, a):
    left, right = 0, len(t)
    while left < right:
        mid = (left + right + 1) // 2
        temp_t = list(t)
        for i in range(mid):
            temp_t[a[i] - 1] = ""
        temp_str = "".join(temp_t)
        j = 0
        for i in range(len(temp_str)):
            if j < len(p) and temp_str[i] == p[j]:
                j += 1
        if j == len(p):
            left = mid
        else:
            right = mid - 1
    return left

# Reading input
t = input().strip()
p = input().strip()
a = list(map(int, input().split()))

# Output the result
print(max_removals(t, p, a))
