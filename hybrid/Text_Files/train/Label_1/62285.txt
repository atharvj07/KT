L = "abcdefghijklmnopqrstuvwxyz"
s = input()
t = [0]*len(s)
for i in range(len(s)-1,-1,-1):
    if s[i]==s[i-1]:
        t[i-1]=1
ans = []
cur = []
for i in range(len(s)):
    if t[i]==1:
        ans.append(cur+[s[i]])
        cur=[]
    else:
        cur.append(s[i])
for i in range(len(ans)):
    if len(ans[i])>10:
        print(len(ans[i]), L[L.index(ans[i][0]):L.index(ans[i][-1])+1]+'...'+ans[i][-2:])
    else:
        print(len(ans[i]), ''.join(ans[i]))
