n, m = map(int, input().split())
graph = {}
rev_graph = {}

for _ in range(m):
    query = input().split()
    if query[0] == "+":
        u, v, c = int(query[1]), int(query[2]), query[3]
        if u not in graph:
            graph[u] = {}
        if v not in rev_graph:
            rev_graph[v] = {}
        graph[u][v] = c
        rev_graph[v][u] = c
    elif query[0] == "-":
        u, v = int(query[1]), int(query[2])
        del graph[u][v]
        del rev_graph[v][u]
    else:
        k = int(query[1])
        if k % 2 == 1:
            print("YES")
        else:
            for u in graph:
                for v in graph[u]:
                    if v in graph and u in graph[v]:
                        print("YES")
                        break
                else:
                    continue
                break
            else:
                print("NO")
