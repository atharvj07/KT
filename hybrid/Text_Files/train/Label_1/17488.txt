#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
ll cal(const string &s, int l, int r, vector<vector<ll>> &dp, vector<vector<int>> &op) {
    if (l > r) return 0;
    if (l == r) return s[l] - '0';
    ll &ret = dp[l][r];
    if (~ret) return ret;
    int &operation = op[l][r];
    if (s[l + 1] == '*') {
        ret = max(ret, cal(s, l + 2, r, dp, op) * (s[l] - '0'));
    }
    else {
        ret = max(ret, cal(s, l + 2, r, dp, op) + (s[l] - '0'));
    }
    for (int i = l + 3; i <= r; i += 2) {
        ll left, right;
        if (s[i] == '*') {
            left = cal(s, l, i - 1, dp, op);
            right = cal(s, i + 1, r, dp, op);
            if (operation == 1) ret = max(ret, left * right);
            else ret = max(ret, left + right);
        }
        else {
            left = cal(s, l, i - 1, dp, op);
            right = cal(s, i + 1, r, dp, op);
            if (operation == 1) ret = max(ret, left * right);
            else ret = max(ret, left + right);
        }
    }
    return ret;
}
int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<vector<int>> op(n + 1, vector<int>(n + 1, -1));
    vector<vector<ll>> dp(n + 1, vector<ll>(n + 1, -1));
    ll ans = cal(s, 0, n - 1, dp, op);
    cout << ans << endl;
    return 0;
}
