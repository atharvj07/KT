#include <bits/stdc++.h>
using namespace std;
#define rep(i, j, k) for (int i = (int)j; i < (int)k; i++)

struct name {
  string name;
  int id;
};

bool compare(name n1, name n2) {
  return n1.name < n2.name;
}

int main() {
  int n;
  cin >> n;
  vector<name> arr1(n), arr2(n);
  rep(i, 0, n) {
    string s;
    cin >> s;
    arr1[i] = name{s, i};
  }
  rep(i, 0, n) {
    string s;
    cin >> s;
    arr2[i] = name{s, i};
  }
  sort(arr1.begin(), arr1.end(), compare);
  sort(arr2.begin(), arr2.end(), compare);
  int ans = 0;
  rep(i, 0, n) {
    int mn = arr1[i].name.size() + arr2[i].name.size() - 2;
    int l = 0, r = min(arr1[i].name.size(), arr2[i].name.size());
    while (l < r) {
      int mid = (l + r) / 2;
      if (arr1[i].name.substr(0, mid) == arr2[i].name.substr(0, mid)) {
        l = mid + 1;
      } else {
        r = mid;
      }
    }
    mn = min(mn, l - 1);
    ans += mn;
  }
  cout << ans << endl;
  rep(i, 0, n) { cout << arr1[i].id + 1 << " " << arr2[i].id + 1 << endl; }
  return 0;
}
