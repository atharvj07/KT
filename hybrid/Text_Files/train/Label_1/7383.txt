#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

ll mod = 1e9 + 7;
ll pwr[200001];
ll inv[200001];

ll bigmod(ll x, ll p) {
    if(p == 0) return 1;
    if(p == 1) return x;
    if(p % 2 == 0) {
        ll ret = bigmod(x, p/2);
        ret = (ret * ret) % mod;
        return ret;
    }
    return (x * bigmod(x, p-1)) % mod;
}

void pre() {
    pwr[0] = 1;
    inv[0] = 1;
    for(int i = 1; i < 200001; i++) {
        pwr[i] = (pwr[i-1] * 2) % mod;
        inv[i] = bigmod(pwr[i], mod - 2);
    }
}

int main() {
    pre();
    ll n, k;
    cin >> n >> k;
    vector<ll> v(n+1);
    for(int i = 0; i <= n; i++) cin >> v[i];
    
    ll ans = 0;
    for(int j = 0; j <= n; j++) {
        if(v[j] % 2 == 1) {
            ans = (ans + pwr[n]) % mod;
            ans = (ans - inv[j + 1]) % mod;
            if(ans < 0) ans += mod;
            ans = (ans - inv[n - j]) % mod;
            if(ans < 0) ans += mod;
            ans = (ans + 1) % mod;
        }
    }
    cout << ans << "\n";
    return 0;
}
