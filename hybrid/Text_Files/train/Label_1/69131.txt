MOD = 998244353

def power(base, exp, mod):
    result = 1
    while exp > 0:
        if exp % 2 == 1:
            result = (result * base) % mod
        base = (base * base) % mod
        exp //= 2
    return result

def solve(n, dominoes):
    fully_unknown = 0
    partially_unknown = 0
    has_conflict = False
    last_color = None
    first_color = None

    for domino in dominoes:
        if domino == "??":
            fully_unknown += 1
        elif "?" in domino:
            partially_unknown += 1
            if domino[0] == "?":
                color = domino[1]
            else:
                color = domino[0]
            if last_color is not None and last_color == color:
                has_conflict = True
            last_color = color
            if first_color is None:
                first_color = color
        else:
            if domino[0] == domino[1]:
                return 0
            if last_color is not None and last_color == domino[0]:
                has_conflict = True
            last_color = domino[1]
            if first_color is None:
                first_color = domino[0]

    if has_conflict:
        return 0

    if fully_unknown == n:
        return (2 * power(2, fully_unknown - 1, MOD)) % MOD

    if partially_unknown == 0 and (first_color is None or first_color != last_color):
        return 1

    return power(2, fully_unknown, MOD)

n = int(input().strip())
dominoes = [input().strip() for _ in range(n)]

print(solve(n, dominoes))
