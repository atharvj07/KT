import sys

n, h, a, b, k = map(int, input().split())
ans = []
for _ in range(k):
    ta, fa, tb, fb = map(int, input().split())
    if ta == tb:
        ans.append(abs(fa - fb))
    elif ta + 1 == tb:
        if a <= fa <= b or a <= fb <= b:
            ans.append(abs(fa - fb) + 1)
        else:
            ans.append(abs(fa - fb) + 2)
    else:
        ans.append(abs(fa - fb) + 2)

print(*ans)
