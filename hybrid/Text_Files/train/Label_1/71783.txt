s = input().strip()
n = len(s)

# Count the number of 'v's before each position
prefix_v = [0] * n
count = 0
for i in range(n):
    if s[i] == 'v':
        count += 1
    prefix_v[i] = count

# Count the number of 'v's after each position
suffix_v = [0] * n
count = 0
for i in range(n-1, -1, -1):
    if s[i] == 'v':
        count += 1
    suffix_v[i] = count

wow_factor = 0
for i in range(1, n-1):
    if s[i] == 'o':
        # Count the number of 'w's that can be formed with 'v's before and after the 'o'
        wow_factor += (prefix_v[i] - prefix_v[i-1]) * (suffix_v[i+1] - suffix_v[i])

print(wow_factor)
