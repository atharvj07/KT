#include <iostream>
#include <vector>
using namespace std;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int N, R;
    cin >> N >> R;
    vector<int> v(N - 1);
    for (int i = 0; i < N - 1; i++) cin >> v[i];
    if (N == 2)
      cout << v[0] << endl;
    else if (N == 3)
      cout << v[0] + 1 << endl;
    else
      cout << 1 << endl;
  }
}
