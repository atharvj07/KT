#include<bits/stdc++.h>
using namespace std;
const int maxn = 110;
const int inf = 0x3f3f3f3f;
int n, m, a[maxn], b[maxn];
int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    sort(a, a + n);
    sort(b, b + m);
    int tl = -1;
    for (int i = 0; i < m; i++) {
        if (b[i] < a[0]) tl = b[i];
    }
    for (int i = 0; i < m; i++) {
        if (b[i] == a[0] && (tl == -1 || tl > a[0] * 2)) tl = a[0] * 2;
    }
    for (int i = 0; i < n; i++) {
        if (a[i] > tl) tl = -1;
        if (a[i] == tl && (tl == -1 || tl > a[i] * 2)) tl = a[i] * 2;
    }
    cout << tl << endl;
    return 0;
}
