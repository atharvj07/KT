#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, a[N];
int32_t main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> R, L;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) {
      if (L.size() > 0) {
        ans += L.back();
        L.pop_back();
      }
      L.push_back(i);
    } else {
      if (R.size() > 0) {
        ans += R.back();
        R.pop_back();
      }
      R.push_back(i);
    }
  }
  cout << ans << endl;
  return 0;
}
