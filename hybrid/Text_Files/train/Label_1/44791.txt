import java.util.*;

public class Main {
    static int n;
    static int[] w;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        w = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            w[i] = in.nextInt();
        }
        Arrays.sort(w);
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += w[2 * i + 1] - w[2 * i];
        }
        System.out.println(ans);
    }
}
