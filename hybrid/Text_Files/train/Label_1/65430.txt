n, m = map(int, input().split())
w = []
c = []
for _ in range(n):
    a, b = map(int, input().split())
    w.append(a)
    c.append(b)
dp = [[0] * (m + 1) for _ in range(n + 1)]
for i in range(n):
    for j in range(m + 1):
        if j + w[i] <= m:
            dp[i + 1][j + w[i]] = max(dp[i + 1][j + w[i]], dp[i][j] + c[i])
        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j])
print(dp[n][m])
