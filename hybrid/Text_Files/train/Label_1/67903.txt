n, m, h = map(int, input().split())
u = list(map(int, input().split()))
clients = [list(map(int, input().split())) for _ in range(m)]

# Find maintenance times that would cause conflicts if shifted
conflict_times = set()
for c1, c2 in clients:
    if u[c1-1] == (u[c2-1] + 1) % h or u[c2-1] == (u[c1-1] + 1) % h:
        conflict_times.add(u[c1-1])
        conflict_times.add(u[c2-1])

# Find data centers that can safely shift their maintenance time
safe_shift = [i+1 for i in range(n) if u[i] not in conflict_times]

print(len(safe_shift))
print(*safe_shift)
