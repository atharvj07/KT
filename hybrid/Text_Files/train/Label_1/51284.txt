import java.util.Arrays;
import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[] l = new int[q];
        for (int i = 0; i < q; i++) {
            l[i] = sc.nextInt();
        }
        sc.close();

        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < m; j++) {
                if (i < a[j]) {
                    continue;
                }
                if (dp[i - a[j]] != -1) {
                    dp[i] = Math.min(dp[i], dp[i - a[j]] + 1);
                }
            }
        }

        for (int i = 0; i < q; i++) {
            if (dp[n] == -1 || dp[n] > l[i]) {
                System.out.println(-1);
            } else {
                System.out.println(n);
            }
        }
    }
}
