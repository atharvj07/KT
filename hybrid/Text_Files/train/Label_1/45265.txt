import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    private static final int MAX_N = 1000000;
    private static final Map<Integer, Integer> memo = new HashMap<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            System.out.println(getNumOperations(n));
        }
        sc.close();
    }

    private static int getNumOperations(int n) {
        if (n == 1) {
            return 0;
        }
        if (n <= MAX_N && memo.containsKey(n)) {
            return memo.get(n);
        }
        int count;
        if (n % 2 == 0) {
            count = 1 + getNumOperations(n / 2);
        } else {
            count = 1 + getNumOperations(3 * n + 1);
        }
        if (n <= MAX_N) {
            memo.put(n, count);
        }
        return count;
    }
}
