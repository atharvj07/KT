#include <bits/stdc++.h>
using namespace std;
int main(){
	int n;cin>>n;
	string s;cin>>s;
	vector<int> ans;
	int cnt=1;
	char last=s[0];
	for(int i=1;i<n;i++){
		if(s[i]==last){
			continue;
		}
		cnt++;
		last=s[i];
	}
	cout<<cnt<<endl;
	for(int i=0;i<n;i++){
		if(i==0||s[i]!=s[i-1]){
			ans.push_back(cnt);
			cnt--;
		}
		else{
			ans.push_back(ans[i-1]);
		}
	}
	for(int i=0;i<n;i++)cout<<ans[i]<<" ";
	cout<<endl;
}
