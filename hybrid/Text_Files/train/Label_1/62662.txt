import sys

def main():
    input = sys.stdin.readline
    for _ in range(int(input())):
        n = int(input())
        a = list(map(int, input().split()))
        b = sorted(a)
        i, ans = 0, 0
        for j in range(n):
            if a[j] == b[i]:
                i += 1
            else:
                ans += 1
        print(ans)

main()
