#include <bits/stdc++.h>
using namespace std;

const int mxN = 2e5;
int x[mxN], y[mxN];
int n, m, p;

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        int a; cin >> a;
        if (i == 0) x[m++] = a;
        else if (a > x[m - 1]) x[m++] = a;
        else y[p++] = a;
    }
    if (m == 0 || p == 0) {
        cout << "NO";
        return 0;
    }
    cout << "YES\n" << m << '\n'; 
    for (int i = 0; i < m; i++)
        cout << x[i] << ' ';
    cout << '\n' << p << '\n';
    for (int i = 0; i < p; i++)
        cout << y[i] << ' ';
    cout << '\n';
}
