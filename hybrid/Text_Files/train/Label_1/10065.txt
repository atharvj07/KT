#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    auto it = max_element(a.begin(), a.end());
    cout << distance(a.begin(), it) + 1 << ' '
         << (*it == a[0] ? *next(it) : a[0]) << endl;
}
