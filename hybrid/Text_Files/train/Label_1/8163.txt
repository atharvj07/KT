#include<iostream>
using namespace std;
bool get(long long n,int d){
    while(n){
        int tmp=n%10;
        if(tmp==d) return 1;
        n/=10;
    }
    return 0;
}
int main(){
    int t; cin>>t;
    while(t--){
        int q,d; cin>>q>>d;
        for(int i=0,x;i<q;++i){
            cin>>x;
            if(get(x,d)) cout<<"YES"<<endl;
            else cout<<"NO"<<endl;
        }
    }
}
