def dfs1(x, fa, dist, a):
    for i in a[x]:
        if i != fa:
            dist[i] = dist[x] + 1
            dfs1(i, x, dist, a)

def dfs2(x, fa, dist, a):
    for i in a[x]:
        if i != fa:
            dist[i] = dist[x] + 1
            dfs2(i, x, dist, a)

n = int(input())
a = [[] for i in range(n+1)]
for i in range(n-1):
    x, y = map(int, input().split())
    a[x].append(y)
    a[y].append(x)
dist = [0 for i in range(n+1)]
dfs1(1, 0, dist, a)
m = 0
for i in range(1, n+1):
    if dist[i] > dist[m]:
        m = i
dist = [0 for i in range(n+1)]
dfs2(m, 0, dist, a)
ans = 0
for i in dist:
    ans += i
print(ans)
