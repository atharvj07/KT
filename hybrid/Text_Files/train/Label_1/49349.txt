import java.util.*;
import static java.lang.Math.*;

public class Main {

    private static int a[];
    private static int b[];

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        a = new int[4];
        b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = input.nextInt();
            b[i] = input.nextInt();
        }
        int def1 = a[0];
        for (int i = 1; i < 4; i++) {
            if (a[i] > def1) {
                def1 = a[i];
            }
        }
        int def2 = b[0];
        for (int i = 2; i < 4; i++) {
            if (b[i] > def2) {
                def2 = b[i];
            }
        }
        int atk1 = b[0];
        for (int i = 2; i < 4; i++) {
            if (b[i] < atk1) {
                atk1 = b[i];
            }
        }
        int atk2 = a[0];
        for (int i = 1; i < 4; i++) {
            if (a[i] < atk2) {
                atk2 = a[i];
            }
        }
        if (def1 > atk2 && atk1 > def2) {
            System.out.println("Team 1");
        } else if (atk2 > def1 && def2 > atk1) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
