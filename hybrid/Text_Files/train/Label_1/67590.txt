m,s=map(int,input().split())
if 9*(m-1)<=s or s>9*m:
    print(-1,-1)
else:
    dig=[0]*m
    dig[m-1]=s%9
    s-=dig[m-1]
    for i in range(m-2,-1,-1):
        dig[i]=min(9,s)
        s-=dig[i]
    dig[0]=s
    mm=[0]*m
    mm[0]=1
    for i in range(1,m):
        mm[i]=mm[i-1]*10
    mx=[0]*m
    for i in range(m):
        if i==0:
            mx[i]=9
        else:
            mx[i]=mx[i-1]%9+1
    print(*dig,sep='',end=' ')
    print(*mx,sep='',end=' ')
