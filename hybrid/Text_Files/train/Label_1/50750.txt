import java.util.ArrayList;
import java.util.HashMap;
import java.util.Scanner;

public class OgudarOlok {

    private static int n, m;
    private static ArrayList<ArrayList<String>> warehouse;
    private static HashMap<String, Integer> boxLocations;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine();

        warehouse = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            warehouse.add(new ArrayList<>());
            for (int j = 0; j < m; j++) {
                warehouse.get(i).add(null);
            }
        }

        boxLocations = new HashMap<>();

        for (int i = 0; i < k; i++) {
            String query = scanner.nextLine();
            String[] parts = query.split(" ");
            if (parts[0].equals("+1")) {
                addBox(Integer.parseInt(parts[1]), Integer.parseInt(parts[2]), parts[3]);
            } else if (parts[0].equals("-1")) {
                removeBox(parts[1]);
            }
        }
    }

    private static void addBox(int shelf, int section, String id) {
        if (boxLocations.containsKey(id)) {
            return;
        }

        for (int i = shelf - 1; i >= 0; i--) {
            for (int j = section - 1; j >= 0; j--) {
                if (warehouse.get(i).get(j) == null) {
                    warehouse.get(i).set(j, id);
                    boxLocations.put(id, i * m + j);
                    return;
                }
            }
        }
    }

    private static void removeBox(String id) {
        if (!boxLocations.containsKey(id)) {
            System.out.println("-1 -1");
            return;
        }

        int location = boxLocations.get(id);
        int shelf = location / m;
        int section = location % m;
        warehouse.get(shelf).set(section, null);
        boxLocations.remove(id);
        System.out.println(shelf + 1 + " " + (section + 1));
    }
}
