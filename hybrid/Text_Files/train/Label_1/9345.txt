#include <iostream>
#include <string>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 2 * n;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n - 3; j++) {
      if (s.substr(j, 4) == "ACTG") ans = min(ans, j + n - 1 - j - 4);
    }
  }
  cout << ans << endl;
  return 0;
}
