import java.util.Scanner;

public class ChoutiAndTheSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] squares = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            squares[i] = scanner.nextInt();
        }

        if (!isPossible(squares)) {
            System.out.println("No");
            return;
        }

        System.out.println("Yes");
        int[] result = new int[n];
        for (int i = 0; i < n / 2; i++) {
            result[i * 2] = squares[i] - result[i * 2 - 2];
            result[i * 2 + 1] = squares[i] - result[i * 2];
        }

        for (int num : result) {
            System.out.print(num + " ");
        }
    }

    private static boolean isPossible(int[] squares) {
        for (int i = 1; i < squares.length; i++) {
            if (squares[i] < squares[i - 1]) {
                return false;
            }
        }
        return true;
    }
}
