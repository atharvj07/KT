from bisect import insort

def solve_queries(queries, m):
    spells = []
    last_positive = 0
    answers = []

    for query in queries:
        k, a, b = query
        a = (a + last_positive) % 10**6 + 1
        b = (b + last_positive) % 10**6 + 1

        if k == 1:
            insort(spells, (b / a, a, b))
        else:
            t, h = a, b
            mana_needed, dmg_done = 0, 0

            for spell in spells:
                if dmg_done >= h:
                    break
                mana_ratio, dmg_per_sec, mana_per_sec = spell
                max_dmg_this_spell = dmg_per_sec * t
                if dmg_done + max_dmg_this_spell > h:
                    time_needed = (h - dmg_done) / dmg_per_sec
                    mana_needed += time_needed * mana_per_sec
                    dmg_done = h
                else:
                    mana_needed += t * mana_per_sec
                    dmg_done += max_dmg_this_spell

            if dmg_done >= h and mana_needed <= m:
                last_positive = t
                answers.append("YES")
            else:
                answers.append("NO")

    return answers

# Example usage
q, m = map(int, input().split())
queries = [list(map(int, input().split())) for _ in range(q)]
result = solve_queries(queries, m)
print('\n'.join(result))
