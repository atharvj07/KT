k = int(input())
g = []
for i in range(4):
    g.append(input())
ans = "YES"
for i in range(4):
    for j in range(4):
        if g[i][j] != '.' and int(g[i][j]) > k:
            ans = "NO"
if ans == "YES":
    for i in range(1, 10):
        x = 0
        for j in range(4):
            for k in range(4):
                if g[j][k] == str(i):
                    x += 1
        if x > k * 2:
            ans = "NO"
print(ans)
