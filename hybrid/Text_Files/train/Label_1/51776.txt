import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        String str = s.nextLine();
        int price = 700;
        for (char c : str.toCharArray()) {
            if (c == 'o') price += 100;
        }
        System.out.println(price);
    }
}
