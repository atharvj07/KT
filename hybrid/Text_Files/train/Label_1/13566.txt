#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    if (n % k != 0) {
        cout << "No" << endl;
        return 0;
    }
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
    }
    int avg = sum / k;
    cout << "Yes" << endl;
    for (int i = 0; i < k; i++) {
        int cnt = 0;
        int temp = 0;
        for (int j = i * n / k; j < (i + 1) * n / k; j++) {
            temp += a[j];
            cnt++;
            if (temp == avg) {
                break;
            }
        }
        cout << cnt << " ";
    }
    cout << endl;
    return 0;
}
