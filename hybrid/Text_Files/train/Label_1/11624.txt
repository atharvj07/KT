#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    bool p[n];
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'o') {
            p[i] = true;
            cnt++;
        } else {
            p[i] = false;
        }
    }
    if (cnt == 0 || cnt == n) {
        cout << "NO" << endl;
        return 0;
    }
    if (cnt % 2 == 0) {
        cout << "YES" << endl;
        return 0;
    }
    for (int i = 0; i < n; i++) {
        if (p[i] && p[(i + 1) % n] && !p[(i + 2) % n]) {
            cout << "YES" << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
    return 0;
}
