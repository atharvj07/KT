#include<bits/stdc++.h>
using namespace std;
long long t,a[1000005],b[1000005],c[1000005],d[1000005];
long long solve(long long a,long long b,long long c,long long d){
    long long ans=-1;
    long long l=0,r=1e9;
    while(l<=r){
        long long mid=(l+r)>>1;
        long long tot=(mid/c)*c+(mid/d)*d;
        if(tot>=a){
            ans=mid;
            r=mid-1;
        }
        else{
            l=mid+1;
        }
    }
    if(ans==-1) return ans;
    ans+=(b/c)*c+(b/d)*d;
    if(ans>=a) return ans;
    return -1;
}
int main(){
    cin>>t;
    for(int i=1;i<=t;i++){
        cin>>a[i]>>b[i]>>c[i]>>d[i];
    }
    for(int i=1;i<=t;i++){
        long long ans=solve(a[i],b[i],c[i],d[i]);
        cout<<ans<<endl;
    }
    return 0;
}
