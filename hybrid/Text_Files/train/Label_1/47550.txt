import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        boolean isEven = true;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            if (a[i] % 2 != 0) isEven = false;
        }
        int cnt = 0;
        while (isEven) {
            for (int i = 0; i < n; i++) a[i] /= 2;
            cnt++;
            isEven = true;
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 != 0) isEven = false;
            }
        }
        System.out.println(cnt);
    }
}
