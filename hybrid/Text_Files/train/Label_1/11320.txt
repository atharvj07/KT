#include <iostream>
#include <string>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    map<string, int> m;
    for (int i = 0; i + 1 < n; i++) {
        m[s.substr(i, 2)]++;
    }
    int mx = 0;
    string ans = "";
    for (auto it : m) {
        if (it.second > mx) {
            mx = it.second;
            ans = it.first;
        }
    }
    cout << ans << endl;
    return 0;
}
