import java.util.Scanner;

public class LexicographicallyMaximumSubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        System.out.println(findLexicographicallyMaximumSubsequence(s));
    }

    public static String findLexicographicallyMaximumSubsequence(String s) {
        int n = s.length();
        int[] dp = new int[n];
        dp[n - 1] = 1;
        int maxIndex = n - 1;

        for (int i = n - 2; i >= 0; i--) {
            if (s.charAt(i) >= s.charAt(maxIndex)) {
                maxIndex = i;
                dp[i] = 1;
            } else {
                dp[i] = 0;
            }
        }

        StringBuilder result = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (dp[i] == 1) {
                result.append(s.charAt(i));
            }
        }

        return result.toString();
    }
}
