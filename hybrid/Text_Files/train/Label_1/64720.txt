import sys, math

def dist(a, b):
    return a+b-1 if a > b else b-a

n = int(input())
roads = list()
for i in range(n-1):
    a, b = map(int, input().split())
    roads.append((a, b, i+1))

roads.sort(key=lambda road: dist(road[0], road[1]))

days = math.ceil((n-1)/2)
ans = [[] for i in range(days)]
j = 0
for r in roads:
    ans[j].append(r[2])
    j = (j+1)%days

print(days)
for i in range(days):
    print(len(ans[i]), *ans[i])
