import sys
from collections import Counter

n, m = map(int, sys.stdin.readline().split())
a = Counter()
for p in sys.stdin.readline().split():
    a[int(p)] += 1
b = Counter()
for p in sys.stdin.readline().split():
    b[int(p)] += 1

for v in a.keys() & b.keys():
    if a[v] == 1 and b[v] == 1:
        print(v)
        exit()

for v in a.keys():
    if a[v] == 2:
        if v in b:
            print(0)
            exit()

print(-1)
