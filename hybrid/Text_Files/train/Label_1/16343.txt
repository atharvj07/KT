#include <bits/stdc++.h>
using namespace std;
string s;
int main() {
  cin >> s;
  int i = 1, j = s.length() / 2;
  while (i < s.length() / 2 && s[i] == s[i - 1]) i++;
  while (j < s.length() && s[j] == s[j - 1]) j++;
  for (; i < s.length() / 2 && j < s.length() && s[i] == s[j]; i++, j++)
    ;
  cout << (s[i] < s[j] ? s.substr(0, i + 1) : s.substr(0, j + 1));
  return 0;
}
