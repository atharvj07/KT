#include <iostream>
#include <unordered_map>
using namespace std;
int main() {
  unordered_map<string, int> cnt;
  int soft = 0;
  string name, type;
  int n;
  cin >> n;
  while (n--) {
    cin >> name >> type;
    cnt[type]++;
    if (type == "soft") soft++;
  }
  int hard = cnt["hard"];
  if (soft > hard)
    cout << hard;
  else
    cout << soft;
}
