import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            int[] x = new int[m];
            for (int i = 0; i < m; i++) {
                x[i] = sc.nextInt();
            }
            for (int i = 0; i < m; i++) {
                long sum = 0;
                int ans = -1;
                for (int j = 0; j < n; j++) {
                    sum += a[j];
                    if (sum >= x[i]) {
                        ans = j + 1;
                        break;
                    }
                }
                System.out.print(ans + " ");
            }
            System.out.println();
        }
    }
}
