import sys,math
input=sys.stdin.readline

t=int(input())
for j in range(t):
  n=int(input())
  if n==2:
    print(-1)
  else:
    a=[0]*n
    if n%2==0:
      p=math.ceil(n/2)
      for i in range(0,n,2):
        a[i]=p
        p-=1
      p=math.ceil(n/2)+1
      for i in range(1,n,2):
        a[i]=p
        p+=1
    else:
      p=n//2
      for i in range(0,n,2):
        a[i]=p
        p-=1
      p=n//2+1
      for i in range(1,n,2):
        a[i]=p
        p+=1
    a[n-1]=n
    print(*a)
