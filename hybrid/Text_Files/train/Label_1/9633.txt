#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const int maxn = 2e5 + 10;
int dp[maxn];
int32_t main() {
  string s;
  cin >> s;
  int len = s.size();
  int sum = 0;
  dp[len - 1] = s[len - 1] - '0';
  sum = dp[len - 1];
  for (int i = len - 2; i >= 0; i--) {
    int tmp = ((s[i] - '0') * 10 + (s[i + 1] - '0')) % 3;
    dp[i] = dp[i + 1] + tmp;
    sum += tmp;
  }
  int ans = 0;
  for (int i = 0; i < len - 1; i++) {
    ans = max(ans, dp[i + 1] + ((sum - dp[i + 1]) % 3 == 0));
  }
  cout << ans << endl;
  return 0;
}
