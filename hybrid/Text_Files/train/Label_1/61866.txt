n, m = map(int, input().split())
a = list(map(int, input().split()))

if n > m:
    print("YES")
else:
    dp = [0] * m
    for i in range(n):
        dp[a[i] % m] = 1
    for i in range(1, (m + 1) // 2):
        if dp[i] and dp[m - i]:
            print("YES")
            break
    else:
        if dp[0]:
            print("YES")
        else:
            print("NO")
