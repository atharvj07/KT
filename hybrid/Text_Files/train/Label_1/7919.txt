#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n);
    for (int i = 0; i < n; i++) cin >> b[i];
    int cnt = 0;
    for (int i = 0; i < n; i++)
      if (b[i] != i + 1) cnt++;
    cout << cnt << '\n';
  }
  return 0;
}
