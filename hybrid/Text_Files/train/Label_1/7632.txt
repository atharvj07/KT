#include <iostream>
#include <map>
using namespace std;

char neighbor(char c, char dir) {
    static map<char, char> keyboard = {
        {'q', 'w'}, {'w', 'e'}, {'e', 'r'}, {'r', 't'}, {'t', 'y'}, {'y', 'u'},
        {'u', 'i'}, {'i', 'o'}, {'o', 'p'},
        {'a', 's'}, {'s', 'd'}, {'d', 'f'}, {'f', 'g'}, {'g', 'h'}, {'h', 'j'},
        {'j', 'k'}, {'k', 'l'},
        {'z', 'x'}, {'x', 'c'}, {'c', 'v'}, {'v', 'b'}, {'b', 'n'}, {'n', 'm'},
    };
    char next = keyboard[c];
    if (dir == 'L') return next == c ? c : next;
    return next == c ? c : keyboard[next];
}

int main() {
    char dir;
    cin >> dir;
    string s;
    cin >> s;
    for (char& c : s) {
        c = neighbor(c, dir);
    }
    cout << s << endl;
    return 0;
}
