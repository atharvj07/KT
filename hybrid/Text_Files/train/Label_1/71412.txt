n = int(input())
s = input()
aa = []
for c in s:
    if c == '(':
        aa.append(1)
    else:
        if len(aa) > 0 and aa[-1] == 1:
            aa.pop()
        else:
            aa.append(-1)
if len(aa) != 0:
    print(-1)
else:
    balan = 0
    ans = 0
    for c in s:
        if c == '(':
            balan += 1
        else:
            balan -= 1
        if balan < 0:
            ans += abs(balan) * 2
            balan = 0
    print(ans)
