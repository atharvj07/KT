import sys

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = []
    for x in a:
        for y in a:
            if x != y and x % y == 0:
                b.append(y)
    if len(set(b)) == len(a) - 1:
        num = max(a) * max(a)
        for x in b:
            num = num // x * x
        print(num)
    else:
        print(-1)
