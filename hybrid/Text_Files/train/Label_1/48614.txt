import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in=new Scanner(System.in);
        int[] ay=new int[3],am=new int[3],p=new int[3];
        for(int i=0;i<3;++i){
            ay[i]=in.nextInt();
        }
        for(int i=0;i<3;++i){
            am[i]=in.nextInt();
        }
        for(int i=0;i<3;++i){
            p[i]=in.nextInt();
        }
        long ans=0;
        for(int i=0;i<3;++i){
            int yp=am[i]-ay[i],rp=-yp/p[i];
            if(yp>=0&&-yp%p[i]!=0){
                ++rp;
            }
            ans+=rp*p[i];
        }
        System.out.println(ans);
    }
}
