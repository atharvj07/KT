import heapq
import sys

input=sys.stdin.readline
n,m,k=map(int,input().split())
graph=[[] for _ in range(n+1)]
q=[]
vis=[False]*(n+1)

for _ in range(m):
    a,b,w=map(int,input().split())
    graph[a].append((w,b))
    graph[b].append((w,a))
    
for x in graph[1]:
    heapq.heappush(q,(x[0],x[1]))
vis[1]=True

while q and k>0:
    w,v=heapq.heappop(q)
    if vis[v]==True:
        continue
    k-=1
    vis[v]=True
    for x in graph[v]:
        if vis[x[1]]==False:
            heapq.heappush(q,(x[0],x[1]))

e=0
rem=[]
for x in graph[1]:
    if vis[x[1]]==True:
        e+=1
        rem.append(x[0])

print(e)
print(*rem)
