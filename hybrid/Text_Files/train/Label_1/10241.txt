#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int pos = -1;
    for (int i = 1; i < n; i++) {
        if (s[i] < s[i - 1]) {
            pos = i - 1;
            break;
        }
    }
    if (pos == -1) {
        s.pop_back();
    } else {
        string l = s.substr(0, pos);
        string r = s.substr(pos + 1);
        if (l < r) {
            s = l;
        } else {
            s = r;
        }
    }
    cout << s << endl;
    return 0;
}
