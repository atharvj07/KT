import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long k = in.nextLong();
        long l = in.nextLong();
        long m = in.nextLong();
        System.out.println((modPow(2, l, m) - 1 + m) % m * modPow(2, n - k, m) % m);
    }
    static long modPow(long base, long exp, long mod) {
        if (exp == 0) return 1;
        long halfpow = modPow(base, exp / 2, mod);
        long fullpow = halfpow * halfpow % mod;
        if (exp % 2 == 0) return fullpow;
        return fullpow * base % mod;
    }
}
