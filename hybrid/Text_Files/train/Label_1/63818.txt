def find_robot_cycle(n, m, k, maze):
    for i in range(n):
        for j in range(m):
            if maze[i][j] == "X":
                start = (i, j)
                break
    
    if k % 2 != 0:
        return "IMPOSSIBLE"
    
    moves = []
    for _ in range(k // 4):
        if start[1] + 1 < m and maze[start[0]][start[1] + 1] == ".":
            moves.extend(["R", "D", "L", "U"])
        elif start[0] + 1 < n and maze[start[0] + 1][start[1]] == ".":
            moves.extend(["D", "L", "U", "R"])
        else:
            return "IMPOSSIBLE"
    
    return "".join(moves)

n, m, k = map(int, input().split())
maze = [input().strip() for _ in range(n)]
print(find_robot_cycle(n, m, k, maze))
