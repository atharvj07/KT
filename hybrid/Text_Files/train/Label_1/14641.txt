#include <iostream>
#include <vector>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    int n0 = 0, n1 = 0;
    for (char c : s) {
        if (c == '0') n0++;
        else n1++;
    }
    int ans = 0;
    for (char c1 : t) {
        for (char c2 : t) {
            if (c1 == c2) continue;
            int cnt1 = 0, cnt2 = 0;
            for (char c : s) {
                if (c == '0') {
                    if (c1 == t[cnt1]) cnt1++;
                    if (c2 == t[cnt1]) cnt1++;
                } else {
                    if (c1 == t[cnt2]) cnt2++;
                    if (c2 == t[cnt2]) cnt2++;
                }
            }
            if (cnt1 == n0 && cnt2 == n1) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
