n, k = map(int, input().split())
c  = [int(a) for a in input().split()]
f  = [int(a) for a in input().split()]
h  = [int(a) for a in input().split()]

dp = [[[-1] * (k + 1) for _ in range(k + 1)] for _ in range(n + 1)]

def dfs(i, j, cnt):
	if dp[i][j][cnt] != -1:
		return dp[i][j][cnt];

	if i == n:
		if j == k:
			dp[i][j][cnt] = h[cnt] 
		else:
			dp[i][j][cnt] = 0 
	else:
		dp[i][j][cnt] = dfs(i + 1, j + 1 if c[i] == f[i] else j, cnt + (c[i] == f[i]));

		if j > 0:
			dp[i][j][cnt] = max(dp[i][j][cnt], dfs(i + 1, j - 1, cnt));

	return dp[i][j][cnt];

print(dfs(0, 0, 0));
