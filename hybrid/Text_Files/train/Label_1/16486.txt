#include<bits/stdc++.h>
#define vi vector<int>
using namespace std;
int main(){
    int t;  cin >>t;
    while(t--){
        int n;  cin >>n;
        vi a(n),b(n);
        for(int &x:a)  cin >>x;
        for(int &x:b)  cin >>x;
        if(accumulate(a.begin(),a.end(),0)!=accumulate(b.begin(),b.end(),0)){
            cout <<"-1\n";
        }
        else{
            int ans=0;
            for(int i=0; i<n; i++){
                if(a[i]>b[i]){
                    ans+=a[i]-b[i];
                    a[i]-=a[i]-b[i];
                    a[0]+=a[i]-b[i];
                }
            }
            cout <<ans <<"\n";
            for(int i=0; i<n; i++){
                if(a[i]!=b[i]){
                    cout <<i+1 <<" " <<1 <<"\n";
                }
            }
        }
    }
    return 0;
}
