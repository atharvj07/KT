#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll inf = 1e18;

int main() {
    int n, m;
    cin >> n >> m;
    vector<ll> b(n), g(m);
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < m; i++) { cin >> g[i]; }
    sort(b.begin(), b.end());
    sort(g.begin(), g.end());
    ll ans = 0;
    for (int i = 0; i < n && i < m; i++) { ans += max(b[i], g[i]); }
    int lb = n < m ? n : m;
    for (int i = lb; i < n || i < m; i++) { ans += max(0LL, i < n ? b[i] : inf); }
    for (int i = lb; i < n || i < m; i++) { ans += max(0LL, i < m ? g[i] : inf); }
    if (ans >= inf) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    return 0;
}
