#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> diffs(n);
  while (m--) {
    int a, b, c;
    cin >> a >> b >> c;
    diffs[a - 1].first -= c;
    diffs[b - 1].second += c;
  }
  bool negative = false;
  for (const auto& diff : diffs) {
    if (diff.first > 0 && diff.second < 0) {
      negative = true;
      break;
    }
  }
  long long sum = 0;
  if (negative) {
    for (const auto& diff : diffs) {
      if (diff.first > 0) { sum += diff.first; }
    }
  } else {
    for (const auto& diff : diffs) {
      if (diff.second > 0) { sum += diff.second; }
    }
  }
  cout << sum << endl;
  return 0;
}
