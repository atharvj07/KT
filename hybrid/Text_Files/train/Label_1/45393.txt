import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] a = new int[51][51];
        int[][] b = new int[51][51];
        int na = sc.nextInt();
        int ma = sc.nextInt();
        for (int i = 1; i <= na; i++) {
            String s = sc.next();
            for (int j = 1; j <= ma; j++) {
                a[i][j] = s.charAt(j - 1) - '0';
            }
        }
        int nb = sc.nextInt();
        int mb = sc.nextInt();
        int ansx = 0, ansy = 0;
        long ans = 0;
        for (int x = -100; x <= 100; x++) {
            for (int y = -100; y <= 100; y++) {
                long cur = 0;
                for (int i = 1; i <= na; i++) {
                    int nx = i + x;
                    if (nx < 1 || nx > nb) {
                        continue;
                    }
                    for (int j = 1; j <= ma; j++) {
                        int ny = j + y;
                        if (ny < 1 || ny > mb) {
                            continue;
                        }
                        cur += a[i][j] * b[nx][ny];
                    }
                }
                if (cur > ans) {
                    ans = cur;
                    ansx = x;
                    ansy = y;
                }
            }
        }
        for (int i = 1; i <= nb; i++) {
            String s = sc.next();
            for (int j = 1; j <= mb; j++) {
                b[i][j] = s.charAt(j - 1) - '0';
            }
        }
        System.out.println(ansx + " " + ansy);
    }
}
