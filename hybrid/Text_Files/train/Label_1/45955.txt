import java.util.*;
class Main{
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        char[]ar=(sc.nextLine()).toCharArray();
        int ans=0,cnt=0,last=-1;
        for(int i=0;i<ar.length;i++){
            if(ar[i]>=97){
                if(cnt>ans)ans=cnt;
                cnt=1;
                last=i;
            }
            else cnt++;
        }
        if(cnt>ans)ans=cnt;
        if(ans<3)ans=0;
        System.out.println(ans);
    }
}
