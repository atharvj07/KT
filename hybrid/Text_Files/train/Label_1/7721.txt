#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Point {
    int x, y;
};

int main() {
    int n;
    cin >> n;

    vector<Point> points(n + 1);
    for (int i = 0; i <= n; i++) {
        cin >> points[i].x >> points[i].y;
    }

    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (points[i - 1].x != points[i].x && points[i].x != points[i + 1].x) {
            cnt++;
        }
        if (points[i - 1].y != points[i].y && points[i].y != points[i + 1].y) {
            cnt++;
        }
    }

    cout << cnt << endl;

    return 0;
}
