#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, a, r, m;
    cin >> n >> a >> r >> m;
    vector<long long> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    sort(h.begin(), h.end());
    long long ans = 0;
    for (int i = 1; i < n; i++) ans += min({a * (h[i] - h[0]), r * (h[0] - h[i]), m * (h[i] - h[0] + h[0] - h[i - 1])});
    cout << ans << endl;
    return 0;
}
