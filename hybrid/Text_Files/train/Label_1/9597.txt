#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s[n],temp;
        unordered_map<string,int> m;
        for(int i=0;i<n;i++){
            cin>>s[i];
            temp = s[i];
            reverse(temp.begin(),temp.end());
            m[temp]++;
        }
        int ans=0;
        for(int i=0;i<n;i++){
            if(m.find(s[i])!=m.end()){
                m[s[i]]--;
                if(m[s[i]]==0){
                    m.erase(s[i]);
                }
                ans++;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
