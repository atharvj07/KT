#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define maxn 200010
vector<ll>G[maxn];
ll n,ans,sub[maxn],a[maxn],sum=0;
void dfs(ll u,ll par)
{
    sub[u]=a[u];
    for(auto v:G[u])
    {
        if(v!=par)
        {
            dfs(v,u);
            sub[u]+=sub[v];
        }
    }
    ans=max(ans,sub[u]);
}
int main()
{
    ll u,v;
    cin>>n;
    for(ll i=1; i<=n; i++)
        cin>>a[i],sum+=a[i];
    for(ll i=1; i<n; i++)
    {
        cin>>u>>v;
        G[u].push_back(v);
        G[v].push_back(u);
    }
    dfs(1,1);
    cout<<sum-ans<<endl;
    return 0;
}
