from sys import stdin

def get_lcs(s: str, t: str) -> int:
    for i in range(len(s)):
        for j in range(len(t)):
            if s[i] == t[j] and (s == t or s[(i + 1) % len(s):] == t[j + 1:]):
                return i + 1

def get_lcm(s: str, t: str) -> str:
    lcs = get_lcs(s, t)
    if lcs == None:
        return -1
    return s[:lcs] * ((len(t) - 1) // lcs + 1)

for _ in range(int(stdin.readline())):
    s = stdin.readline().strip()
    t = stdin.readline().strip()
    print(get_lcm(s, t))
