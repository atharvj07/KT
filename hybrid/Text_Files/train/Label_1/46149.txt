import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();
        int[] affected = new int[m];
        for (int i = 0; i < m; i++) {
            affected[i] = sc.nextInt();
        }
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            edges.add(new int[]{a, b});
        }
        int[] dist = new int[n + 1];
        for (int i = 0; i < m; i++) {
            bfs(edges, affected[i], dist);
        }
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] <= d) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }

    private static void bfs(List<int[]> edges, int start, int[] dist) {
        Queue<Integer> q = new LinkedList<>();
        q.add(start);
        dist[start] = 0;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int[] edge : edges) {
                if (edge[0] == cur) {
                    if (dist[edge[1]] == 0) {
                        q.add(edge[1]);
                        dist[edge[1]] = dist[edge[0]] + 1;
                    }
                } else if (edge[1] == cur) {
                    if (dist[edge[0]] == 0) {
                        q.add(edge[0]);
                        dist[edge[0]] = dist[edge[1]] + 1;
                    }
                }
            }
        }
    }
}
