import sys

n, m = map(int, input().split())
g = [[0] * (m + 2) for i in range(n)]
for i in range(n):
    g[i] = [int(j) for j in input()]

si = {i: False for i in range(n)}
ri = {i: False for i in range(n)}
ans = 0
for i in range(n - 1, -1, -1):
    for j in range(1, m + 1):
        if g[i][j] == 0: continue
        if not si[i]: ans += 1; si[i] = True
        if not ri[i]: ans += 1; ri[i] = True
print(ans)
