import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    private static final int MAX_SIZE = 100005;
    private static int n, m, maxNeighborColorDiversity = 0, answer = -1;
    private static List<List<Integer>> graph = new ArrayList<>();
    private static int[] colors, neighborColorDiversity;

    public static void main(String[] args) throws IOException {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
            String[] line = reader.readLine().split(" ");
            n = Integer.parseInt(line[0]);
            m = Integer.parseInt(line[1]);
            colors = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            neighborColorDiversity = new int[MAX_SIZE];
            Arrays.fill(neighborColorDiversity, -1);
            for (int i = 0; i < n; i++) {
                graph.add(new ArrayList<>());
            }
            for (int i = 0; i < m; i++) {
                line = reader.readLine().split(" ");
                int u = Integer.parseInt(line[0]);
                int v = Integer.parseInt(line[1]);
                graph.get(u - 1).add(v - 1);
                graph.get(v - 1).add(u - 1);
            }
            for (int i = 0; i < n; i++) {
                if (neighborColorDiversity[colors[i]] == -1) {
                    neighborColorDiversity[colors[i]] = getNeighborColorDiversity(i);
                    if (neighborColorDiversity[colors[i]] > maxNeighborColorDiversity) {
                        maxNeighborColorDiversity = neighborColorDiversity[colors[i]];
                        answer = colors[i];
                    } else if (neighborColorDiversity[colors[i]] == maxNeighborColorDiversity) {
                        answer = Math.min(answer, colors[i]);
                    }
                }
            }
            System.out.println(answer);
        }
    }

    private static int getNeighborColorDiversity(int vertex) {
        int colorDiversity = 0;
        boolean[] visited = new boolean[MAX_SIZE];
        List<Integer> queue = new ArrayList<>();
        queue.add(vertex);
        visited[vertex] = true;
        while (!queue.isEmpty()) {
            int currentVertex = queue.remove(0);
            colorDiversity = Math.max(colorDiversity, visited[colors[currentVertex]] ? 0 : 1);
            visited[colors[currentVertex]] = true;
            for (int neighbor : graph.get(currentVertex)) {
                if (!visited[neighbor]) {
                    queue.add(neighbor);
                    visited[neighbor] = true;
                }
            }
        }
        return colorDiversity;
    }
}
