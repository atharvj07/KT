#include <bits/stdc++.h>

using namespace std;

const int maxn = 50 + 1;
int n, c, p[maxn], t[maxn];
int dp[2][maxn][maxn];

void solve() {
  cin >> n >> c;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
    cin >> t[i];
  }
  memset(dp, 0, sizeof(dp));
  dp[1][1][0] = p[1];
  for (int i = 2; i <= n; i++) {
    for (int j = 0; j <= t[i - 1]; j++) {
      dp[0][i][j] = max(dp[1][i - 1][j], dp[0][i - 1][j]);
      if (j + t[i] <= c * t[i]) {
        dp[1][i][j + t[i]] = max(dp[1][i][j + t[i]], dp[0][i - 1][j] + p[i] - j - t[i] * c);
      }
    }
  }
  int ans = 0;
  for (int i = 0; i <= c * t[n]; i++) {
    ans = max(ans, dp[1][n][i]);
  }
  memset(dp, 0, sizeof(dp));
  dp[1][n][0] = p[n];
  for (int i = n - 1; i >= 1; i--) {
    for (int j = 0; j <= t[i + 1]; j++) {
      dp[0][i][j] = max(dp[1][i + 1][j], dp[0][i + 1][j]);
      if (j + t[i] <= c * t[i]) {
        dp[1][i][j + t[i]] = max(dp[1][i][j + t[i]], dp[0][i + 1][j] + p[i] - j - t[i] * c);
      }
    }
  }
  int ans2 = 0;
  for (int i = 0; i <= c * t[1]; i++) {
    ans2 = max(ans2, dp[1][1][i]);
  }
  if (ans > ans2) {
    cout << "Limak";
  } else if (ans < ans2) {
    cout << "Radewoosh";
  } else {
    cout << "Tie";
  }
}

int main() {
  solve();
  return 0;
}
