#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e6 + 5;

int n;
int a[MAXN], b[MAXN];

int main() {
    cin >> n;
    int ans = 0;
    for (int i = 0; i <= n; i++) {
        a[i] = i;
        ans += (i ^ 0);
    }
    int j = n;
    for (int i = 1; i <= n; i += 2) {
        swap(a[i], a[j]);
        ans += (a[i] ^ i) - (a[j] ^ i);
        j--;
    }
    cout << ans << endl;
    for (int i = 0; i <= n; i++) {
        cout << a[i] << ' ';
    }
    return 0;
}
