def can_pass_test(grid):
    for i in range(3):
        for j in range(3):
            count_black = 0
            count_black += grid[i][j] == "#"
            count_black += grid[i][j+1] == "#"
            count_black += grid[i+1][j] == "#"
            count_black += grid[i+1][j+1] == "#"
            if count_black != 2:
                return "YES"
    return "NO"

grid = [input() for _ in range(4)]
print(can_pass_test(grid))
