#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) std::cin >> a[i];
    int ans = 0;
    bool flag = false;
    while (!flag) {
      flag = true;
      for (int i = 0; i < n; i++) {
        if (a[i] > k) continue;
        for (int j = 0; j < n; j++) {
          if (a[i] == 0 || a[j] == k) continue;
          flag = false;
          a[j] += a[i];
          a[i] = 0;
          ans++;
          break;
        }
      }
    }
    std::cout << ans << '\n';
  }
  return 0;
}
