import java.util.Scanner;

import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[][] arr = new int[n][6];

        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < 6; j++)
            {
                arr[i][j] = sc.nextInt();
            }
        }

        boolean[] canMake = new boolean[10];
        int max = 0;

        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < 6; j++)
            {
                int num = arr[i][j];

                if (num != 6 && num != 9)
                {
                    canMake[num] = true;
                }
            }
        }

        for (int i = 1; i <= 1000; i++)
        {
            boolean canMakeThis = true;
            String numStr = String.valueOf(i);

            for (int j = 0; j < numStr.length(); j++)
            {
                int digit = Character.getNumericValue(numStr.charAt(j));

                if (!canMake[digit])
                {
                    canMakeThis = false;
                }
            }

            if (!canMakeThis)
            {
                max = i - 1;

                break;
            }
        }

        System.out.println(max);
    }
}
