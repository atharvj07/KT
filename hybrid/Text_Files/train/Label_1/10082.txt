#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
int _, n, m, arr[2505][2505], t[2505][2505], dp[2505][2505];
int32_t main() {
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { string s; cin >> s; for (int j = 0; j < m; j++) arr[i][j] = s[j] == '1'; }
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < m; j++) { t[i][j] = t[i][j - 1] + arr[i][j]; }
  }
  for (int i = 0; i <= n + 1; i++) dp[i][0] = dp[i][m + 1] = 1e9;
  for (int i = 1; i <= m + 1; i++) dp[0][i] = dp[n + 1][i] = 1e9;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      dp[i][j] = min(dp[i][j - 1], dp[i - 1][j]) + arr[i][j - 1];
      if (i + m - j >= j + i - 1) dp[i][j] = min(dp[i][j], dp[i][j + 1] + t[i][j - 1]);
    }
  }
  cout << dp[n][m] << "\n";
  return 0;
}
