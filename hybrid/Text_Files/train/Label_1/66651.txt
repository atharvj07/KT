import math

def circle_intersection_area(x1, y1, r1, x2, y2, r2):
    d = math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
    if d >= r1 + r2:
        return 0.0
    elif d <= abs(r2 - r1) and r1 >= r2:
        return math.pi * r2 ** 2
    elif d <= abs(r2 - r1) and r2 >= r1:
        return math.pi * r1 ** 2
    else:
        a1 = r1 ** 2 * math.acos((d ** 2 + r1 ** 2 - r2 ** 2) / (2 * d * r1))
        a2 = r2 ** 2 * math.acos((d ** 2 + r2 ** 2 - r1 ** 2) / (2 * d * r2))
        a3 = 0.5 * math.sqrt((-d + r1 + r2) * (d + r1 - r2) * (d - r1 + r2) * (d + r1 + r2))
        return a1 + a2 - a3

x1, y1, r1 = map(int, input().split())
x2, y2, r2 = map(int, input().split())

print(circle_intersection_area(x1, y1, r1, x2, y2, r2))
