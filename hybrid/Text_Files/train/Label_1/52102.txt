import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][][] a = new int[n][n][n];
        int dx = 1, dy = 0, dz = 0;
        int sx = 0, sy = 0, sz = 0;
        for (int i = 1; i <= n * n * n;) {
            a[sx][sy][sz] = i++;
            sx += dx;
            sy += dy;
            sz += dz;
            if (sx < 0 || sx >= n || sy < 0 || sy >= n || sz < 0 || sz >= n) {
                sx -= dx;
                sy -= dy;
                sz -= dz;
                dx = -dx;
                dy = -dy;
                dz = -dz;
                sx += dx;
                sy += dy;
                sz += dz;
            }
            if (dx != 0 && sx == 0) {
                dx = 0;
                dy = 1;
                dz = 0;
                sx += dx;
                sy += dy;
                sz += dz;
            }
            if (dy != 0 && sy == n - 1) {
                dx = 0;
                dy = -1;
                dz = 0;
                sx += dx;
                sy += dy;
                sz += dz;
            }
            if (dz != 0 && sz == 0) {
                dx = 0;
                dy = 0;
                dz = 1;
                sx += dx;
                sy += dy;
                sz += dz;
            }
            if (dz != 0 && sz == n - 1) {
                dx = 0;
                dy = 0;
                dz = -1;
                sx += dx;
                sy += dy;
                sz += dz;
            }
        }
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    System.out.print(a[i][j][k] + " ");
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
