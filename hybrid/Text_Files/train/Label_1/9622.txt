#include <vector>
#include <iostream>
#include <algorithm>

int main() {
    int n, m, d;
    std::cin >> n >> m >> d;
    std::vector<int> a(n);
    for (auto& x : a) std::cin >> x;
    std::sort(a.begin(), a.end());
    int days = 0;
    int cur = 0;
    for (auto x : a) {
        if (x < cur) continue;
        else {
            ++days;
            cur = x + d;
        }
    }
    std::cout << days << '\n';
    int day = 1;
    cur = 0;
    for (auto x : a) {
        if (x < cur) std::cout << day << ' ';
        else {
            while (x >= cur) cur += d;
            std::cout << day << ' ';
            ++day;
        }
    }
    return 0;
}
