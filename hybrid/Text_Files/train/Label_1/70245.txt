import sys
n, m = list(map(int, sys.stdin.readline().split()))
s = sys.stdin.readline().rstrip()
t = sys.stdin.readline().rstrip()
if m == n:
    print(n - 1)
else:
    s = '#' + s
    t = '#' + t
    l = [-1] * (n + 1)
    ll = [-1] * (n + 1)
    p = dict()
    maxl = 0
    for i in range(1, n + 1):
        if s[i] in p:
            ll[i] = p[s[i]]
        p[s[i]] = i
    for i in range(1, n + 1):
        l[i] = ll[i]
        if ll[i] != -1:
            ll[i] = l[ll[i]]
    last = -1
    for i in range(m, n + 1):
        while last != -1 and t[i - last] != s[i]:
            last = l[last]
        if t[i - last] == s[i]:
            last += 1
        if last > maxl:
            maxl = last
    print(maxl)
