def merge_words(word1, word2):
    max_prefix_len = min(len(word1), len(word2))
    for i in range(max_prefix_len, 0, -1):
        if word1.endswith(word2[:i]):
            return word1 + word2[i:]
    return word1 + word2

n = int(input())
words = input().split()

result = words[0]

for i in range(1, n):
    result = merge_words(result, words[i])

print(result)
