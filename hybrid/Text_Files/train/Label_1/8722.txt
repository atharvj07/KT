#include <iostream>
#include <vector>
#include <utility>
using namespace std;

int main() {
  int n, ans = 0;
  cin >> n;
  vector<int> a(2 * n);
  vector<pair<int, int>> pos(n + 1);
  for (int i = 0; i < 2 * n; i++) {
    cin >> a[i];
    if (!pos[a[i]].first) {
      pos[a[i]] = {i + 1, 0};
    } else {
      pos[a[i]].second = i + 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    ans += pos[i].second - pos[i].first;
  }
  cout << ans << '\n';
}
