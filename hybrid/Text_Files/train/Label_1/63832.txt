def calculate_max_damage(n, m, jiro_cards, ciel_cards):
    jiro_atk = sorted([strength for position, strength in jiro_cards if position == "ATK"])
    jiro_def = sorted([strength for position, strength in jiro_cards if position == "DEF"])
    ciel_atk = sorted(ciel_cards)

    damage = 0
    while jiro_atk and ciel_atk and ciel_atk[-1] >= jiro_atk[0]:
        ciel_card = ciel_atk.pop()
        jiro_card = jiro_atk.pop(0)
        damage += ciel_card - jiro_card

    if not jiro_atk:
        for card in ciel_atk:
            damage += card
        for def_card in jiro_def:
            if ciel_atk and ciel_atk[-1] > def_card:
                ciel_atk.pop()
            else:
                damage -= sum(ciel_atk)
                break

    all_ciel = sorted(ciel_cards, reverse=True)
    all_jiro = sorted([strength for _, strength in jiro_cards if _ == "ATK"], reverse=True)
    sum_damage = 0
    if len(all_ciel) >= len(all_jiro):
        for i in range(len(all_jiro)):
            if all_ciel[i] < all_jiro[i]:
                break
            sum_damage += all_ciel[i] - all_jiro[i]
        else:
            for i in range(len(all_jiro), len(all_ciel)):
                sum_damage += all_ciel[i]

    return max(damage, sum_damage)

n, m = map(int, input().split())
jiro_cards = [input().split() for _ in range(n)]
jiro_cards = [(position, int(strength)) for position, strength in jiro_cards]
ciel_cards = [int(input()) for _ in range(m)]

print(calculate_max_damage(n, m, jiro_cards, ciel_cards))
