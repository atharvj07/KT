#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

int main() {
    ll n;
    cin >> n;
    ll a[2 * n], cnt = 0, pos[n + 1];
    memset(pos, -1, sizeof(pos));
    for (ll i = 0; i < 2 * n; i++) {
        cin >> a[i];
        if (pos[a[i]] == -1) pos[a[i]] = i;
        else cnt += abs(pos[a[i]] - i), pos[a[i]] = -1;
    }
    cout << cnt << endl;
    return 0;
}
