import sys
input = sys.stdin.readline

n, m = map(int, input().split())
board = [list(input().strip()) for _ in range(n)]

def dfs(x, y, dx, dy, cnt):
    if not (0 <= x < n and 0 <= y < m and board[x][y] == '.' and cnt < 4):
        return 0
    if cnt == 4:
        if x == 0 or x == n - 1 or y == 0 or y == m - 1:
            return 1
        else:
            return 0
    return dfs(x + dx, y + dy, dx, dy, cnt + 1) + dfs(x + dx, y + dy, -dx, -dy, cnt + 1)

ans = 0
for i in range(n):
    for j in range(m):
        if board[i][j] == '.':
            for dx, dy in [(1, 0), (0, 1), (-1, 0), (0, -1)]:
                ans += dfs(i, j, dx, dy, 1)
print(ans)
