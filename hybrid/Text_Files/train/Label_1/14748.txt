#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<string> g(n);
  for (int i = 0; i < n; ++i) cin >> g[i];
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    int cnt = 0;
    for (int j = 0; j < n; ++j) {
      if (g[i][j] == 'W') ++cnt;
      else cnt = 0;
      if (cnt >= k) ++ans;
    }
  }
  for (int j = 0; j < n; ++j) {
    int cnt = 0;
    for (int i = 0; i < n; ++i) {
      if (g[i][j] == 'W') ++cnt;
      else cnt = 0;
      if (cnt >= k) ++ans;
    }
  }
  cout << ans << endl;
  return 0;
}
