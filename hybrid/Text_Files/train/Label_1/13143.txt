#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, L;
    cin >> n >> L;
    vector<long long> c(n);
    for (auto &i : c) { cin >> i; }
    long long res = 9e18;
    for (int mask = 0; mask < (1 << n); mask++) {
        long long sum = 0, cost = 0;
        for (int i = 0; i < n; i++) {
            if ((mask >> i) & 1 == 1) {
                cost += c[i];
                sum += (1LL << i);
            }
        }
        if (sum >= L) { res = min(res, cost); }
    }
    cout << res << endl;
}
