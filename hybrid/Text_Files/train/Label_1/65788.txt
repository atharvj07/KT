from sys import stdin

n,m = map(int,stdin.readline().split())
a = list(map(int,stdin.readline().split()))
adj = [[] for _ in range(n+1)]
for _ in range(m):
  u,v,w = map(int,stdin.readline().split())
  adj[u].append((v,w))
  adj[v].append((u,w))

dsu = [i for i in range(n+1)]

def find(x):
    if x != dsu[x]:
        dsu[x] = find(dsu[x])
    return dsu[x]

def union(x,y):
    x = find(x); y = find(y)
    if x == y:
        return
    if a[x-1] < a[y-1]:
        dsu[y] = x
    else:
        dsu[x] = y

ans = 0
for i in range(n):
    v,w = adj[i+1][0]
    ans += min(a[i],w)
    union(i+1,v)

print(ans)
