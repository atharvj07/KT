import java.util.*;
import java.math.*;

class Solution {

    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] interLayerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            interLayerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        // dp[i][j] represents the number of paths from the entry point to the j-th city in the i-th layer (0-indexed) such that the total cost is divisible by m.
        BigInteger[][] dp = new BigInteger[l + 1][n];

        // Initialize the base case.
        for (int i = 0; i < n; i++) {
            if (entryCosts[i] % m == 0) {
                dp[0][i] = BigInteger.ONE;
            }
        }

        // Iterate over the layers.
        for (int i = 1; i <= l; i++) {
            // Iterate over the cities in the current layer.
            for (int j = 0; j < n; j++) {
                // Iterate over the cities in the previous layer.
                for (int k = 0; k < n; k++) {
                    if (dp[i - 1][k] != null && interLayerCosts[k] % m == 0) {
                        // Add the number of paths from the previous layer to the current layer.
                        dp[i][j] = (dp[i][j] == null ? BigInteger.ZERO : dp[i][j]).add(dp[i - 1][k].multiply(BigInteger.valueOf(interLayerCosts[k])));
                    }
                }
            }
        }

        // Initialize the number of valid paths to 0.
        BigInteger validPaths = BigInteger.ZERO;

        // Iterate over the cities in the last layer.
        for (int i = 0; i < n; i++) {
            if (dp[l][i] != null && exitCosts[i] % m == 0) {
                // Add the number of paths from the last layer to the exit point.
                validPaths = validPaths.add(dp[l][i].multiply(BigInteger.valueOf(exitCosts[i])));
            }
        }

        // Print the number of valid paths modulo 10^9 + 7.
        System.out.println(validPaths.mod(BigInteger.valueOf(MOD)));
    }
}
