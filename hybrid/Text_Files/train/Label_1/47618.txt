import java.util.*;

public class BerlandRoads {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        List<Integer>[] graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            graph[u].add(v);
            graph[v].add(u);
        }

        int k = scanner.nextInt();
        int[] fools = new int[2 * k];
        for (int i = 0; i < k; i++) {
            fools[2 * i] = scanner.nextInt() - 1;
            fools[2 * i + 1] = scanner.nextInt() - 1;
        }

        int[] roadCount = new int[n - 1];
        for (int i = 0; i < k; i++) {
            int start = fools[2 * i];
            int end = fools[2 * i + 1];
            findPaths(graph, start, end, new boolean[n], roadCount);
        }

        for (int count : roadCount) {
            System.out.print(count + " ");
        }
    }

    private static void findPaths(List<Integer>[] graph, int start, int end, boolean[] visited, int[] roadCount) {
        if (start == end) {
            return;
        }

        visited[start] = true;
        for (int neighbor : graph[start]) {
            if (!visited[neighbor]) {
                findPaths(graph, neighbor, end, visited, roadCount);
                roadCount[neighbor - 1]++;
            }
        }
        visited[start] = false;
    }
}
