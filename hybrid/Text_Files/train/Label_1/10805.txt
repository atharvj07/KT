#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n, a, b;
    cin >> n >> a >> b;
    if (a > n - 2 || b > n - 2 || a + b > n - 1)
      cout << "-1\n";
    else {
      vector<int> p(n);
      iota(begin(p), end(p), 1);
      a = n - 2 - a;
      b = n - 2 - b;
      if (a > b) {
        for (int i = 0; i < a - b; i++) {
          swap(p[i + 1], p[i + 2]);
          i++;
        }
        for (int i = n - 1; i >= n - 2 - b; i--) {
          swap(p[i - 1], p[i]);
          i--;
        }
      } else if (b) {
        for (int i = n - 1; i >= n - 1 - (b - a); i--) {
          swap(p[i - 1], p[i]);
          i--;
        }
        for (int i = 0; i < b - a; i++) {
          swap(p[i + 1], p[i + 2]);
          i++;
        }
      }
      for (auto x : p) cout << x << ' ';
      cout << '\n';
    }
  }
  return 0;
}
