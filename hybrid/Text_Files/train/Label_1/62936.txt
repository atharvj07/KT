n, m = map(int, input().split())
debt = [0] * (n + 1)

for _ in range(m):
    u, v, d = map(int, input().split())
    debt[u] -= d
    debt[v] += d

new_debts = [(i, debt[i]) for i in range(1, n + 1) if debt[i] != 0]
new_debts.sort(key=lambda x: x[1])

i, j = 0, len(new_debts) - 1
result = []

while i < j:
    debtor, debt_amount = new_debts[i]
    creditor, credit_amount = new_debts[j]
    
    transaction = min(-debt_amount, credit_amount)
    
    result.append((debtor, creditor, transaction))
    
    new_debts[i] = (debtor, debt_amount + transaction)
    new_debts[j] = (creditor, credit_amount - transaction)
    
    if new_debts[i][1] == 0:
        i += 1
    if new_debts[j][1] == 0:
        j -= 1

print(len(result))
for debtor, creditor, amount in result:
    print(debtor, creditor, amount)
