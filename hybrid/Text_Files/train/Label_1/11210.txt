#include<bits/stdc++.h>
using namespace std;
const int N=4001;
int main(){
  int m,s[N],g[1<<10],c=0,i,last,x,y,q;
  memset(g,-1,sizeof(g));
  cin>>m;
  for(i=0;i<m;i++) cin>>s[i];
  for(i=0;i<m;i++){
    for(x=i+1;x<m;x++){
      if(g[s[x]]==-1){
        g[s[x]]=1;
        q=s[x];
        while(q!=1){
          for(y=1;y<=9;y++){
            if(q%y==0){
              g[q/y]=1;
              q/=y;
            }
          }
        }
      }
    }
  }
  for(i=0;i<s[m-1];i++){
    if(g[i+1]!=-1) c++;
  }
  x=s[m-1];
  while(x!=1){
    for(y=1;y<=9;y++){
      if(x%y==0){
        g[x/y]=1;
        x/=y;
      }
    }
  }
  for(i=2;i<=s[m-1];i++){
    if(g[i]!=-1) c++;
  }
  if(c==m){
    for(i=1;i<=s[m-1];i++) if(g[i]!=-1) last=i;
    for(i=last;i>=1;i--){
      if(g[i]!=-1) s[m-1]/=i;
    }
    cout<<1<<endl;
    cout<<s[m-1]<<endl;
    return 0;
  }
  if(c==m-1){
    x=s[m-1];
    for(i=2;i<=sqrt(x);i++){
      if(x%i==0){
        while(x%i==0){
          x/=i;
          g[i]=1;
        }
      }
    }
    if(x>1) g[x]=1;
    for(i=1,c=-1;i<=s[m-1];i++){
      if(g[i]!=-1) c++;
    }
    if(c==m){
      cout<<2<<endl;
      cout<<s[m-1]<<" 1"<<endl;
      return 0;
    }
  }
  cout<<-1<<endl;
  return 0;
}
