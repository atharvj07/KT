#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int sum = 0;
    for (int i : a) { sum += i % 3; }
    if (sum % 3 == 0)
      cout << n << "\n";
    else
      cout << n - 1 << "\n";
  }
}
