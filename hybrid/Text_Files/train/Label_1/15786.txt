#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  if ((n == 1 && s == "0") || (n > 1 && s == string(n, '0'))) {
    cout << "Yes";
  } else
    if ((s[0] != '0' && s[1] != '0') || (s[n - 1] != '0' && s[n - 2] != '0')) {
      cout << "No";
    } else {
      int cnt = 0;
      for (char c : s) {
        cnt += c == '0';
      }
      cout << (cnt <= 2 ? "Yes" : "No");
    }
  return 0;
}
