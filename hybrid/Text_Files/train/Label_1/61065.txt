t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    s = input()
    count = 0
    i = 0
    while i < n:
        if s[i] == '0':
            # Check if there's a '1' in the next k positions or in the previous k positions
            left = max(0, i - k)
            right = min(n, i + k + 1)
            if '1' not in s[left:right]:
                count += 1
                i += k
        i += 1
    print(count)
