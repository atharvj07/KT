import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int sereja = 0, dima = 0;
        boolean serejaTurn = true;
        for (int i = 0; i < n; i++) {
            if (serejaTurn) {
                sereja += arr[i];
            } else {
                dima += arr[i];
            }
            serejaTurn = !serejaTurn;
        }
        System.out.println(sereja + " " + dima);
    }
}
