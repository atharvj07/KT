#include <bits/stdc++.h>
using namespace std;
#define int long long

#define vi(x) vector<x>
#define pb push_back
#define all(x) begin(x), end(x)
#define sz(x) (int)(x).size()

int32_t main() {
  int n;
  cin >> n;
  vi(int) a(n), b(n), c(n);
  vi(vi(int)) g(n);
  for (int i = 0; i < n; ++i)
    cin >> a[i] >> b[i] >> c[i];
  for (int i = 1; i < n; ++i) {
    int u, v;
    cin >> u >> v;
    --u, --v;
    g[u].pb(v);
    g[v].pb(u);
  }
  int ans = LLONG_MAX;
  vi(int) f(n), tot(n);
  function<void(int, int)> dfs1 = [&](int u, int p) {
    f[u] = -1;
    tot[u] = b[u];
    for (int v : g[u])
      if (v != p) {
        dfs1(v, u);
        if (f[u] == -1)
          f[u] = v, tot[u] += tot[v];
        else {
          if ((b[v] ^ b[u]) || (b[v] ^ c[u])) { f[u] = 0; }
          tot[u] += tot[v];
        }
      }
  };
  dfs1(0, -1);
  function<void(int, int, int)> dfs2 = [&](int u, int p, int v) {
    if (u != p) {
      tot[u] = tot[p] - tot[u] + c[u];
      if (f[u] != -1) dfs2(f[u], u, v);
    } else {
      if (tot[u] == n) ans = min(ans, v * a[u]);
    }
  };
  for (int u = 0; u < n; ++u)
    if (f[u] != -1) dfs2(f[u], u, a[u]);
  cout << (ans == LLONG_MAX ? -1 : ans) << '\n';
}
