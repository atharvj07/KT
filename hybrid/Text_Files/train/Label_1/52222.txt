import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length();
        int[] ans = new int[n];
        int open = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                open++;
            } else if (s.charAt(i) == ')') {
                open--;
            } else {
                ans[i] = -1;
            }
        }
        if (open < 0) {
            System.out.println("-1");
            return;
        }
        int close = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (s.charAt(i) == ')') {
                close++;
            } else if (s.charAt(i) == '(') {
                close--;
            } else {
                ans[i] = close;
            }
        }
        for (int i = 0; i < n; i++) {
            if (ans[i] < 0) {
                System.out.println("-1");
                return;
            }
        }
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '#') {
                System.out.println(ans[i]);
            }
        }
    }
}
