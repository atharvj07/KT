#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  int w;
  cin >> w;
  vector<int> cnt(9);
  for (int &x : cnt) { cin >> x; }
  for (int j = 8; j >= 2; j--)
    for (int i = j; i <= 8; i += j)
      cnt[j] += cnt[i];
  ll ans = 0;
  if (w >= 8) { ans += 1LL * w / 8 * cnt[8]; }
  if (w >= 7) { ans += 1LL * w / 7 * cnt[7]; }
  if (w >= 6) { ans += 1LL * w / 6 * cnt[6]; }
  if (w >= 4) { ans += 1LL * w / 4 * cnt[4]; }
  if (w >= 3) { ans += 1LL * w / 3 * cnt[3]; }
  if (w >= 2) { ans += 1LL * w / 2 * cnt[2]; }
  if (w >= 1) { ans += min(1LL * w, 1LL * cnt[1]); }
  cout << ans << endl;
  return 0;
}
