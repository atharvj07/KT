#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll n, a, cf, cm, m;
vector<ll>v;

ll f(ll x){
    ll ans = 0, count = 0, mn = 1e9;
    for(ll i = 0; i < n; i++){
        if(v[i] >= x){
            count++;
            ans += cf;
        }
        mn = min(mn, v[i]);
    }
    ans += cm*mn;
    return ans;
}

int main() {
    cin >> n >> a >> cf >> cm >> m;
    v.resize(n);
    for(auto &x : v) cin >> x;
    ll low = 0, high = a;
    ll ans = 0;
    vector<ll>res(n);
    while(low <= high){
        ll mid = low + (high-low)/2;
        if(f(mid) <= m){
            ans = mid;
            for(ll i = 0; i < n; i++){
                res[i] = max(v[i], mid);
            }
            low = mid + 1;
        }
        else{
            high = mid - 1;
        }
    }
    cout << ans << endl;
    for(auto &x : res) cout << x << " ";
}
