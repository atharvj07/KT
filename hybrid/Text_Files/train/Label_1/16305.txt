#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'
#define fi first
#define se second
#define pb push_back
#define mp make_pair
#define all(x) (x).begin(),(x).end()
#define rall(x) (x).rbegin(),(x).rend()
#define sz(x) (x).size()
#define rep(i,n) for(i=0;i<n;i++)
ll i,n,m,q,a,b;
vector<pair<ll,ll>>v;
int main()
{
    cin>>n>>m>>q;
    while(q--)
    {
        cin>>a>>b;
        v.pb(mp(a,b));
    }
    ll p=0;
    for(i=0;i<sz(v);i++)
    {
        if(p+1==v[i].fi+v[i].se-1)
        {
            p++;
        }
        else if(p+1<v[i].fi+v[i].se-1)
        {
            p=v[i].fi+v[i].se-1;
        }
    }
    if(p+1==n+m-1)
        cout<<2<<endl;
    else if(p+1<n+m-1)
        cout<<3<<endl;
    return 0;
}
