import java.util.Scanner;

public class ChemicalSubstance {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int ma = scanner.nextInt();
        int mb = scanner.nextInt();

        int[][] chemicals = new int[n][3];
        for (int i = 0; i < n; i++) {
            chemicals[i][0] = scanner.nextInt(); // a_i
            chemicals[i][1] = scanner.nextInt(); // b_i
            chemicals[i][2] = scanner.nextInt(); // c_i
        }

        int result = solve(chemicals, ma, mb);
        System.out.println(result);
    }

    static int solve(int[][] chemicals, int ma, int mb) {
        int gcd = gcd(ma, mb);
        int targetA = ma / gcd;
        int targetB = mb / gcd;

        int[][] dp = new int[targetA + 1][targetB + 1];
        for (int i = 0; i <= targetA; i++) {
            for (int j = 0; j <= targetB; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }
        dp[0][0] = 0;

        for (int[] chemical : chemicals) {
            int a = chemical[0];
            int b = chemical[1];
            int c = chemical[2];

            for (int i = targetA; i >= a; i--) {
                for (int j = targetB; j >= b; j--) {
                    dp[i][j] = Math.min(dp[i][j], dp[i - a][j - b] + c);
                }
            }
        }

        return dp[targetA][targetB] == Integer.MAX_VALUE ? -1 : dp[targetA][targetB];
    }

    static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
