import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = (int) Math.pow(10, n - 1);
        for (int i = 0; i < m; i++) {
            int s = sc.nextInt() - 1;
            int c = sc.nextInt();
            d = Math.min(c * d / 10, d);
        }
        if (d > 0) {
            System.out.println(d);
        } else {
            System.out.println(-1);
        }
    }
}
