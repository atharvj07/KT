#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<string> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<vector<int>> dp(n, vector<int>(m + 2, 1e9));
    dp[0][0] = 0;
    for (int i = 0; i < n; i++)
        for (int j = 0; j <= m + 1; j++) {
            if (j > 0) dp[i][j] = min(dp[i][j], dp[i][j - 1] + (a[i][j - 1] == '1'));
            if (j < m + 1) dp[i][j] = min(dp[i][j], dp[i][j + 1] + (a[i][j + 1] == '1'));
            if (i + 1 < n) dp[i + 1][j] = min(dp[i + 1][j], dp[i][j] + (a[i][j] == '1'));
        }
    cout << dp[n - 1][m + 1] << endl;
}
