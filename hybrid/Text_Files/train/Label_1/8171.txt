#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef unsigned long long ll;

int main() {
    ll n, m, w; cin >> n >> m >> w;
    vector<ll> arr(n);
    for (ll i = 0; i < n; i++) cin >> arr[i];
    sort(arr.begin(), arr.end());

    ll add = m / w;
    ll rem = m % w;
    ll ans = arr[0] + add * arr[w];

    if (w <= rem) ans += arr[w-1];
    else ans += arr[w-1] + arr[w-2] - arr[rem-1];
    cout << ans << '\n';
    return 0;
}
