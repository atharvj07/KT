#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> b = a;
    sort(b.begin(), b.end());
    bool ok = true;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) {
        if (i > 0 && a[i - 1] > a[i]) ok = false;
        if (i < n - 1 && a[i + 1] < a[i]) ok = false;
      }
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
  return 0;
}
