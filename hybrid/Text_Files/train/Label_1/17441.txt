#include<bits/stdc++.h>
using namespace std;
int main() {
    string s, t;
    cin >> s >> t;
    int n = s.size(), m = t.size();
    vector<int> p(n + 1, -1);
    vector<int> lp(n, 1), rp(n, 1);
    int last = -n - 1;
    for (int i = 0; i < n; ++i) {
        if (s[i] == '?') {
            p[i] = i;
            last = i;
        }
        else if (s[i] == t[0]) {
            p[i] = last;
        }
    }
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] == '?') last = i;
        else if (s[i] == t.back()) rp[i] = last;
    }
    string ans;
    for (int i = 0; i < n; ++i) {
        if (s[i] == '?') {
            if (p[i] != -1) {
                bool ok = true;
                for (int j = p[i]; j < i; ++j) {
                    if (s[j] != '?' && s[j] != t[j - p[i]]) ok = false;
                }
                if (ok) ans += t[i - p[i]];
                else ans += 'a';
            }
            else if (rp[i] != -1) {
                bool ok = true;
                for (int j = rp[i]; j > i; --j) {
                    if (s[j] != '?' && s[j] != t[rp[i] - j]) ok = false;
                }
                if (ok) ans += t[rp[i] - i];
                else ans += 'a';
            }
            else ans += 'a';
        }
        else ans += s[i];
    }
    cout << ans << endl;
    return 0;
}
