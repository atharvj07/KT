#include <cstdio>
#include <vector>
#define rep(i, a, n) for (i = a; i < n; i++)
int ans1 = -1, ans2 = -1, a[105][105], c, d, i, j, n, k, mx, temp;
std::vector<int> v;
int main() {
  scanf("%d %d", &n, &k);
  rep(i, 0, n) rep(j, 0, n) scanf(" %c", &a[i][j]);
  rep(i, 0, n) {
    rep(j, 0, n) {
      temp = 0;
      if (i + k - 1 < n) {
        rep(d, 0, k) temp += (a[i + d][j] != '#');
      }
      if (j + k - 1 < n) {
        rep(d, 0, k) temp += (a[i][j + d] != '#');
      }
      if (temp > mx) {
        mx = temp;
        ans1 = i + 1, ans2 = j + 1;
        while (!v.empty()) v.pop_back();
      }
      if (temp == mx) v.push_back(temp);
      if ((int)v.size() > 1) {
        ans1 = ans2 = -1;
      }
    }
  }
  printf("%d %d", ans1, ans2);
  return 0;
}
