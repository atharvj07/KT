#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    int cur = 0, last = -1;
    for (int i = 0; i <= n; i++) {
        if (i == n || s[i] != s[cur]) {
            if (cur != last) ans++;
            last = cur;
            cur = i;
        }
    }
    cout << ans << '\n';
    cur = 0, last = -1;
    for (int i = 0; i <= n; i++) {
        if (i == n || s[i] != s[cur]) {
            if (cur != last) cout << s.substr(cur, i - cur) << ' ';
            last = cur;
            cur = i;
        }
    }
    return 0;
}
