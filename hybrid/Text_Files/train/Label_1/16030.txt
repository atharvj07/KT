#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 1e9 + 7;
const int N = 1e3 + 5;
char s[N], t[N];
int dp[N][N][11];
int n, m, k;
int solve(int i, int j, int rem) {
  if (i >= n || j >= m || rem == 0) return 0;
  int &ret = dp[i][j][rem];
  if (ret != -1) return ret;
  ret = 0;
  for (int l = 1; i + l - 1 < n && j + l - 1 < m; l++) {
    if (s[i + l - 1] != t[j + l - 1]) continue;
    ret = max(ret, l + solve(i + l, j + l, rem - 1));
  }
  ret = max(ret, solve(i + 1, j, rem));
  ret = max(ret, solve(i, j + 1, rem));
  return ret;
}
int main() {
  memset(dp, -1, sizeof dp);
  cin >> n >> m >> k;
  cin >> s >> t;
  cout << solve(0, 0, k) << '\n';
  return 0;
}
