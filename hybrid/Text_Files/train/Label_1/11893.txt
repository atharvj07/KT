#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m; cin >> n >> m;
    long long ans = 1e18;
    vector<vector<long long>> c(n, vector<long long>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> c[i][j];
        }
        sort(c[i].begin(), c[i].end());
    }
    for (int i = 0; i < n; i++) {
        ans = min(ans, c[i][0]);
    }
    cout << ans << endl;
    return 0;
}
