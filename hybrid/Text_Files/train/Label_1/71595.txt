import sys

n, m = map(int, input().split())
dances = [list(map(int, input().split())) for _ in range(m)]

colors = [0] * (n + 1)
used = set()

for dance in dances:
    for dancer in dance:
        if dancer not in used:
            colors[dancer] = colors[dance[0]] % 3 + 1
            used.add(dancer)

print(*colors[1:])
