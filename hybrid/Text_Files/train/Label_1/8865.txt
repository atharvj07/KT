#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    int a[n][m];
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) cin >> a[i][j];
    int ans = 1e9;
    for (int i = n - 1; i >= 0; i--) {
        bool fl = true;
        for (int j = 0; j < m; ++j) {
            if (a[i][j] != a[n - 1 - i][j]) fl = false;
        }
        if (fl) ans = min(ans, i + 1);
    }
    cout << ans << endl;
}
