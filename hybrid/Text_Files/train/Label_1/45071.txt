import java.util.*;

public class Main {

    public static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
            b[i] = sc.nextLong();
        }

        long[][][] dp = new long[n][n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }

        long ans = solve(0, 0, 1, a, b, dp);
        if (ans == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }

    public static long solve(int i, int j, int f, long[] a, long[] b, long[][][] dp) {
        if (dp[i][j][f] != -1) {
            return dp[i][j][f];
        }

        long ans = -1;
        if (i == j) {
            if (a[i] == b[i]) {
                ans = 0;
            } else {
                ans = -1;
            }
        } else {
            long x = a[i];
            long y = b[j];
            if (f == 1) {
                x += a[(i + 1) % a.length] + a[(i - 1 + a.length) % a.length];
            } else {
                y += b[(j + 1) % b.length] + b[(j - 1 + b.length) % b.length];
            }

            if (x == y) {
                ans = 0;
            } else if (x > y) {
                long t = x - y;
                if (t % 3 == 0) {
                    ans = t / 3;
                }
            } else {
                long t = y - x;
                if (t % 3 == 0) {
                    ans = t / 3;
                }
            }
        }

        if (f == 1) {
            j = (j + 1) % a.length;
        } else {
            i = (i + 1) % b.length;
        }

        if (ans != -1) {
            long next = solve(i, j, f ^ 1, a, b, dp);
            if (next != -1) {
                ans += next;
            }
        }

        dp[i][j][f] = ans;
        return ans;
    }
}
