import sys

N = int(sys.stdin.readline())
cnt1 = {}
cnt2 = {}
s1 = []
s2 = []
for i in range(N):
    s1.append(sys.stdin.readline().strip())
    cnt1[s1[-1]] = cnt1.get(s1[-1], 0) + 1
    
for i in range(N):
    s2.append(sys.stdin.readline().strip())
    cnt2[s2[-1]] = cnt2.get(s2[-1], 0) + 1

if cnt1 == cnt2:
    print(0)
else:
    sum = 0
    for i in range(N):
        if s1[i]!=s2[i]:
            sum += 1
            cnt1[s1[i]] -= 1
            cnt2[s2[i]] -= 1
            if cnt1[s1[i]]==0:
                del cnt1[s1[i]]
            if cnt2[s2[i]]==0:
                del cnt2[s2[i]]

    print(sum)
