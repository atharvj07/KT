#include<iostream>
#include<vector>
#include<map>
using namespace std;
map<int,vector<int>>M;
int main(){
    int n;
    cin>>n;
    for(int i=1;i<=n;++i){
        int x;
        cin>>x;
        M[x].push_back(i);
    }
    bool f=0;
    for(auto i:M){
        if(i.second.size()!=1)f=1;
        for(int j=1;j<i.second.size();++j){
            if(i.second[j]!=i.second[j-1]+1)f=1;
        }
    }
    if(f)cout<<"NO"<<endl;
    else cout<<"YES\n"<<M.begin()->second.size()<<' '<<M.begin()->second.back()<<endl;
    return 0;
}
