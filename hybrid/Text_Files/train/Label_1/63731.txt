t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    a = [2**x for x in b]
    prefix_sum = {0: -1}
    current_sum = 0
    answer = "NO"
    for i in range(n):
        current_sum += a[i]
        if current_sum in prefix_sum:
            if prefix_sum[current_sum] + 1 < i:  # ensure non-intersecting
                answer = "YES"
                break
        else:
            prefix_sum[current_sum] = i
    print(answer)
