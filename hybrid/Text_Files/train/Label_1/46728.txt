import java.util.Arrays;
import java.util.Scanner;

public class VitalyaAndNight {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);

        int minAlarmsToTurnOff = n;
        for (int i = 0; i <= n - m; i++) {
            int count = 0;
            for (int j = i; j < i + m && j < n; j++) {
                if (a[j] - a[i] + 1 >= k) {
                    count++;
                }
            }
            minAlarmsToTurnOff = Math.min(minAlarmsToTurnOff, n - count);
        }

        System.out.println(minAlarmsToTurnOff);
    }
}
