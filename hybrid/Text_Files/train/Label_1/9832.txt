#include <iostream>
#include <vector>
using namespace std;

string s;
int l, r;
int n;

int main() {
    cin >> s;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> l >> r;
        if (l + 2 > r) {
            cout << "YES" << endl;
            continue;
        }
        vector<int> cnt(3);
        for (int j = l + 2; j <= r; j++) {
            cnt[s[j - 1] - 'x']++;
        }
        bool flag = 0;
        for (int j = 0; j < 3; j++)
            flag |= cnt[j] > 0;
        for (int j = 0; j < 3; j++)
            for (int k = j + 1; k < 3; k++)
                if (cnt[j] == cnt[k]) flag = 1;
        if (flag)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
