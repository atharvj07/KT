#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> bal(n);
  for (int i = 0; i < n; i++) { cin >> bal[i]; }
  int q;
  cin >> q;
  for (int i = 0; i < q; i++) {
    int t;
    cin >> t;
    if (t == 1) {
      int p, x;
      cin >> p >> x;
      p--;
      bal[p] = x;
    } else {
      int x;
      cin >> x;
      for (int j = 0; j < n; j++) {
        if (bal[j] < x) { bal[j] = x; }
      }
    }
  }
  for (int i = 0; i < n; i++) { cout << bal[i] << ' '; }
  cout << '\n';
  return 0;
}
