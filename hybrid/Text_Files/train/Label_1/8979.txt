#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int a[n];
        int cnt[3] = {};
        for (int i = 0; i < n; i++) {
            cin >> a[i], cnt[a[i] % 3]++;
        }
        sort(a, a + n);
        int ans = 0;
        for (int i = 0; cnt[0] < cnt[1] && i < n; i++) {
            if (a[i] % 3 == 1) cnt[0]++, cnt[1]--, ans++;
        }
        for (int i = 0; cnt[0] > cnt[2] && i < n; i++) {
            if (a[i] % 3 == 2) cnt[0]--, cnt[2]++, ans++;
        }
        cout << ans << endl;
    }
}
