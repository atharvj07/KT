#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<long long> a(n), k(q);
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < q; ++i) cin >> k[i];
  for (int i = 0, l = 0; i < q; ++i) {
    l += k[i];
    while (l >= 0 && i < q) {
      l -= a[0];
      a.erase(a.begin());
      ++i;
      if (a.empty()) {
        l = 0;
        break;
      }
    }
    cout << a.size() << '\n';
  }
  return 0;
}
