import collections

def main():
    t = int(input())
    for _ in range(t):
        s = []
        for _ in range(8):
            s.append(list(input()))
        sx, sy = 0, 0
        for i in range(8):
            for j in range(8):
                if s[i][j] == 'K':
                    if sx == 0:
                        sx, sy = i, j
                    else:
                        x, y = i, j

        ans = "NO"
        d = collections.deque([(sx, sy)])
        vis = [[False] * 8 for _ in range(8)]
        while d:
            ix, iy = d.popleft()
            if x == ix and y == iy:
                ans = "YES"
                break
            for dx, dy in [(-2, -2), (-2, 2), (2, -2), (2, 2)]:
                nx, ny = ix + dx, iy + dy
                if 0 <= nx < 8 and 0 <= ny < 8 and not vis[nx][ny] and s[nx][ny] != '#':
                    vis[nx][ny] = True
                    d.append((nx, ny))

        print(ans)

main()
