#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 2e5 + 5;
int n, ans;
vector<int> g[maxn];
vector<tuple<int, int, int>> ops;

void dfs(int u, int p, int d) {
    if (g[u].size() == 1) {
        ops.emplace_back(p, u, u);
        ans += d;
        return;
    }
    for (int v : g[u]) {
        if (v != p) dfs(v, u, d + 1);
    }
}

int main() {
    cin >> n;
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 1, 0);
    cout << ans << '\n';
    for (auto op : ops) {
        cout << get<0>(op) << ' ' << get<1>(op) << ' ' << get<2>(op) << '\n';
    }
}
