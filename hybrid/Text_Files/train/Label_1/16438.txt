#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a)
        cin >> x;
    sort(a.begin(), a.end());
    int ans = 1;
    vector<int> res = {a[0]};
    for (int i = 1; i < n; i++) {
        if (abs(a[i] - res.back()) <= 1) {
            res.push_back(a[i]);
            ans++;
        }
    }
    cout << ans << endl;
    for (int x : res)
        cout << x << " ";
    cout << endl;
    return 0;
}
