#include <bits/stdc++.h>
using namespace std;
long long n, x[200005];
int main() {
  cin >> n;
  for (int i = 0; i < n / 2; i++) { cin >> x[2 * (i + 1)]; }
  if (x[n] != n * n || x[2] * n / 2 != x[n]) {
    cout << "No" << endl;
    return 0;
  }
  cout << "Yes" << endl;
  long long ans = sqrt(x[n]);
  int q = sqrt(x[2]);
  cout << q * ans - x[2] << " ";
  for (int i = 0; i < n / 2; i++) {
    cout << x[2 * (i + 1)] << " ";
    if (i < n / 2 - 1) cout << (ans - x[2 * (i + 1)] - x[2 * (i + 2)]) << " ";
  }
  return 0;
}
