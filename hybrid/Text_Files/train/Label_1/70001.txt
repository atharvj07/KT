n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
f = [0] * (m + 1)
for i in range(n):
    b = []
    for j in range(1, a[i][0] + 1):
        b.append(a[i][j])
    b.sort()
    for j in range(len(b) - 1):
        if b[j] == b[j + 1]:
            f[b[j]] += 1
            break
print(pow(2, len(b) - f[b[0]] - 1, 1000000007))
