#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    partial_sum(a.begin(), a.end(), a.begin());
    vector<long long> dp(k + 1, LONG_MIN);
    dp[0] = 0;
    for (int i = 1; i <= k; i++) {
        for (int j = 0; j < n && j <= i - 1; j++) { dp[i] = max(dp[i], a[j] * i - dp[j]); }
    }
    cout << dp[k] << endl;
    return 0;
}
