import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] h = new int[6];
        int[] w = new int[6];
        for (int i = 0; i < 6; i++) {
            h[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        Arrays.sort(h);
        Arrays.sort(w);
        if (h[0] == h[1] && h[1] == h[2] && w[0] == w[1] && w[1] == w[2]) {
            System.out.println("yes");
        } else if (h[0] == h[1] && h[1] == h[2] && w[3] == w[4] && w[4] == w[5]) {
            System.out.println("yes");
        } else if (h[3] == h[4] && h[4] == h[5] && w[0] == w[1] && w[1] == w[2]) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
