#include<bits/stdc++.h>
using namespace std;
int n, a[101], b[101], ans[101];
int main() {
	int t; cin >> t;
	while (t--) {
		cin >> n;
		for (int i = 0; i < n; i++)
			cin >> a[i];
		int mx = 0;
		for (int i = 0; i < n; i++) {
			mx = max(mx, a[i]);
		}
		for (int i = 0; i <= mx; i++)
			b[i] = 0;
		for (int i = 0; i < n; i++)
			b[a[i]] = 1;
		int mex = 0;
		for (int i = 0; i <= mx; i++) {
			if (!b[i])
				mex = i;
			ans[i] = mex;
		}
		for (int i = 0; i < n; i++)
			cout << ans[a[i]] << " ";
		cout << "\n";
	}
}
