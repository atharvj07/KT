#include <bits/stdc++.h>
using namespace std;
const int maxn = 110;
int dp[maxn][maxn][maxn];

int lcs(string &s1, string &s2, string &t) {
    int n = s1.size(), m = s2.size(), k = t.size();
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= m; j++)
            for (int p = 0; p <= k; p++) dp[i][j][p] = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            for (int p = 1; p <= k; p++) {
                if (s1[i - 1] == s2[j - 1])
                    dp[i][j][p] = max({dp[i - 1][j - 1][p], dp[i - 1][j][p], dp[i][j - 1][p]});
                else if (s1[i - 1] != t[p - 1] && s2[j - 1] != t[p - 1])
                    dp[i][j][p] = max({dp[i - 1][j][p], dp[i][j - 1][p], dp[i - 1][j - 1][p] + 1});
            }
    int ans = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++) ans = max(ans, dp[i][j][k]);
    return ans;
}

int main() {
    string s1, s2, t;
    cin >> s1 >> s2 >> t;
    cout << lcs(s1, s2, t) << endl;
    return 0;
}
