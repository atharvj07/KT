n, m, k = map(int, input().split())
p = list(map(int, input().split()))
s = list(map(int, input().split()))
c = list(map(int, input().split()))
c = [p[c[i]-1] for i in range(k)]
s1 = []
for i in range(m):
    s1.append([p[i], i])
s1.sort(reverse=True)
s2 = []
for j in range(k):
    s2.append(s1[j])
s2.sort(key=lambda x: x[1])
ans = 0
for i in range(k):
    if s2[i][0] == c[i]:
        continue
    ans += 1
print(ans)
