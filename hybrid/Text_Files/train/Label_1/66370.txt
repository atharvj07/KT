import sys
inp, pr = list(map(int, sys.stdin.readline().split()))
l = []
r = []
for i in range(inp):
    a, b = list(map(int, sys.stdin.readline().split()))
    l.append(a)
    r.append(b)
ex = 0
for i in range(inp):
    for j in range(inp):
        if i != j:
            ex += (r[i] - l[i] + 1) * (r[j] - l[j] + 1) / (r[i] + r[j] - l[i] - l[j] + 1) * (r[i] * r[j] % pr == 0)
print(ex / 4.0)
