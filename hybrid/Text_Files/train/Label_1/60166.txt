from collections import defaultdict
n = int(input())
times = defaultdict(list)
for _ in range(n):
	s, d = map(int, input().split())
	times[s].append(d + s - 1)
curr = 0
ans = []
for s in sorted(times):
	times[s].sort()
	for f in times[s]:
		if curr <= s:
			ans.append((s, f))
			curr = f + 1
		else:
			ans.append((curr, curr + f - s))
			curr += f - s + 1
for a in ans:
	print(*a)
