for t in range(int(input())):
    s, c = input().split()
    if len(s) > len(c):
        print('---')
    else:
        s = list(s)
        for i, ch in enumerate(s):
            for j in range(i+1, len(s)):
                if ch > s[j]:
                    s[i], s[j] = s[j], ch
                    if ''.join(s) < c:
                        print(''.join(s))
                        s[i], s[j] = s[j], ch
                        break
            if ''.join(s) < c:
                break
        else:
            print('---')
