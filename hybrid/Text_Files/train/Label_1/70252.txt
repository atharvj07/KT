from collections import deque

def read_ints():
    return list(map(int, input().split()))

def bfs(u, graph, n):
    dist = [-1] * (n + 1)
    dist[u] = 0
    queue = deque([u])
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                queue.append(v)
    return dist

def find_max_dist_vertex(dist):
    return dist.index(max(dist)), max(dist)

def solve():
    t = int(input())
    for _ in range(t):
        n, a, b, da, db = read_ints()
        graph = [[] for _ in range(n + 1)]
        for _ in range(n - 1):
            u, v = read_ints()
            graph[u].append(v)
            graph[v].append(u)
        
        dist_from_a = bfs(a, graph, n)
        if dist_from_a[b] <= da:
            print("Alice")
            continue

        u, max_dist = find_max_dist_vertex(dist_from_a)
        dist_from_u = bfs(u, graph, n)
        _, tree_diameter = find_max_dist_vertex(dist_from_u)

        if da * 2 >= tree_diameter or da * 2 >= db:
            print("Alice")
        else:
            print("Bob")

solve()
