#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  if (n < k || n - k > 1)
    cout << "-1";
  else {
    char a[n];
    int p = 1, q = 1;
    for (int i = 0; i < n; i++) {
      if (q == k && p < 26) {
        p++;
        q = 0;
      }
      a[i] = p + 96;
      q++;
    }
    for (int i = 0; i < n; i++) cout << a[i];
  }
}
