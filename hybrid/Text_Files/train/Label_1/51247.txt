import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int k = sc.nextInt();

        int ans = 0;
        for (int i = 0; i < s.length(); i++) {
            if (i + k < s.length() && s.charAt(i) == s.charAt(i + k)) {
                ans = k + 1;
            }
        }

        System.out.println(ans);
    }
}
