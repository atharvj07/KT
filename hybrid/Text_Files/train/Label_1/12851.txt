void solve(){
  int n,k;
  cin>>n>>k;
  if(n%2 && k==n-1) cout<<"-1"<<nl;
  else{
    int a[n];
    if(k==n-1){
      for(int i=0;i<n;++i){
        if(i&1) a[i]=0;
        else a[i]=1;
      }
    }
    else{
      for(int i=0;i<n;++i){
        if(i<k) a[i]=i;
        else a[i]=i-1;
      }
    }
    for(int i=0;i<n;++i) cout<<char('a'+a[i]);
    cout<<nl;
  }
}
