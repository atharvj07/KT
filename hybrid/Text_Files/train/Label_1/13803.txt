#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n), perm(n * 2);
    for (int &i : b) { cin >> i; }
    iota(perm.begin(), perm.end(), 1);
    for (int i = 0; i < n; i++) {
      for (int j = i * 2, k = 0; j <= i * 2 + 1; j++, k++) {
        if (perm[j + 1] > perm[j] && b[i] == perm[j]) { swap(perm[j], perm[j + 1]); }
      }
    }
    bool ok = false;
    for (int i = 0; i < n; i++) {
      if (b[i] != min(perm[i * 2], perm[i * 2 + 1])) { ok = true; }
    }
    if (ok) { cout << -1 << '\n'; } else { for (int i : perm) { cout << i << ' '; } }
    cout << '\n';
  }
}
