import java.util.*;

public class Main {
    static long[] tree;
    static long[] lazy;
    static int N;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        int M = sc.nextInt();
        tree = new long[N * 4 + 1];
        lazy = new long[N * 4 + 1];
        while (M-- > 0) {
            int type = sc.nextInt();
            if (type == 1) {
                int l = sc.nextInt() - 1, r = sc.nextInt(), d = sc.nextInt();
                updateRange(1, N, 1, l, r, d);
            } else {
                int l = sc.nextInt() - 1, r = sc.nextInt(), m = sc.nextInt();
                long max = query(1, N, 1, l, r);
                if (max > m) {
                    System.out.println("NO");
                    return;
                }
            }
        }
        System.out.println("YES");
        for (int i = 1; i <= N; i++) {
            System.out.print(query(1, N, 1, i, i) + " ");
        }
        System.out.println();
        sc.close();
    }

    static long query(int s, int e, int node, int l, int r) {
        propagate(node, s, e);
        if (s > r || e < l) return 0;
        if (s >= l && e <= r) return tree[node];
        int mid = (s + e) / 2;
        return Math.max(query(s, mid, node * 2, l, r), query(mid + 1, e, node * 2 + 1, l, r));
    }

    static void updateRange(int s, int e, int node, int l, int r, long v) {
        propagate(node, s, e);
        if (s > r || e < l) return;
        if (s >= l && e <= r) {
            lazy[node] += v;
            propagate(node, s, e);
            return;
        }
        int mid = (s + e) / 2;
        updateRange(s, mid, node * 2, l, r, v);
        updateRange(mid + 1, e, node * 2 + 1, l, r, v);
        tree[node] = Math.max(tree[node * 2], tree[node * 2 + 1]);
    }

    static void propagate(int node, int s, int e) {
        tree[node] += lazy[node];
        if (s != e) {
            lazy[node * 2] += lazy[node];
            lazy[node * 2 + 1] += lazy[node];
        }
        lazy[node] = 0;
    }
}
