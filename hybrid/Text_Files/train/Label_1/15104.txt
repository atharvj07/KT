#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, k;
        cin >> a >> b >> k;
        vector<set<int>> boys(a + 1), girls(b + 1);
        for (int i = 0; i < k; i++) {
            int boy, girl;
            cin >> boy >> girl;
            boys[boy].insert(i);
            girls[girl].insert(i);
        }
        long long ans = 0;
        for (int i = 1; i <= a; i++) {
            for (int j = i + 1; j <= a; j++) {
                for (int p : boys[i]) {
                    for (int q : boys[j]) {
                        if (p != q) {
                            ans += girls[p].size() * girls[q].size();
                        }
                    }
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
