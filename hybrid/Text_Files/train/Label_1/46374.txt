import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int K = sc.nextInt();

    long[][] ranges = new long[N][6];
    for (int i = 0; i < N; i++) {
      for (int j = 0; j < 6; j++) {
        ranges[i][j] = sc.nextInt();
      }
    }

    long ansXmin = Long.MAX_VALUE;
    long ansXmax = Long.MIN_VALUE;
    long ansYmin = Long.MAX_VALUE;
    long ansYmax = Long.MIN_VALUE;
    long ansDmin = Long.MAX_VALUE;
    long ansDmax = Long.MIN_VALUE;

    for (int i = 0; i < N; i++) {
      long xmin = ranges[i][0];
      long xmax = ranges[i][1];
      long ymin = ranges[i][2];
      long ymax = ranges[i][3];
      long dmin = ranges[i][4];
      long dmax = ranges[i][5];
      if (xmax < ansXmin) {
        ansXmin = xmax;
      }
      if (ansXmax < xmin) {
        ansXmax = xmin;
      }
      if (ymax < ansYmin) {
        ansYmin = ymax;
      }
      if (ansYmax < ymin) {
        ansYmax = ymin;
      }
      if (dmax < ansDmin) {
        ansDmin = dmax;
      }
      if (ansDmax < dmin) {
        ansDmax = dmin;
      }
    }

    long ans = 0;
    if (ansXmax >= ansXmin && ansYmax >= ansYmin && ansDmax >= ansDmin) {
      ans = (ansXmax - ansXmin) * (ansYmax - ansYmin) * (ansDmax - ansDmin);
    }
    System.out.println(ans);
  }
}
