#include <iostream>
#include <vector>

const int MOD = 1e9 + 7;

int main() {
    int n;
    std::cin >> n;
    std::vector<int> p(n);
    for (int i = 0; i < n; ++i) std::cin >> p[i];
    std::vector<int> dp(n + 1, -1);
    dp[1] = 1;
    for (int i = 1; i <= n; ++i) {
        if (dp[i] == -1) continue;
        dp[(i + 1) % MOD] = (2 * dp[i]) % MOD;
        dp[p[i - 1]] = (dp[p[i - 1]] + dp[i]) % MOD;
    }
    std::cout << dp[n] << std::endl;
    return 0;
}
