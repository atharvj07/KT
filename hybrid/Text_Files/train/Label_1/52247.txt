import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;

public class UniverseCounter {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Planet> planets = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            planets.add(new Planet(x, y));
        }
        Set<Universe> universes = splitIntoUniverses(planets);
        System.out.println(universes.size());
    }

    private static Set<Universe> splitIntoUniverses(List<Planet> planets) {
        Set<Universe> universes = new HashSet<>();
        List<Planet> unallocatedPlanets = new ArrayList<>(planets);
        while (!unallocatedPlanets.isEmpty()) {
            Planet planet = unallocatedPlanets.remove(0);
            Universe universe = new Universe(planet);
            universes.add(universe);
            expandUniverse(universe, unallocatedPlanets);
        }
        return universes;
    }

    private static void expandUniverse(Universe universe, List<Planet> unallocatedPlanets) {
        List<Planet> planetsToRemove = new ArrayList<>();
        for (Planet planet : unallocatedPlanets) {
            if (universe.canAddPlanet(planet)) {
                universe.addPlanet(planet);
                planetsToRemove.add(planet);
            }
        }
        unallocatedPlanets.removeAll(planetsToRemove);
    }

    private static class Planet {
        int x;
        int y;

        public Planet(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Universe {
        Set<Planet> planets;

        public Universe(Planet planet) {
            this.planets = new HashSet<>();
            this.planets.add(planet);
        }

        public boolean canAddPlanet(Planet planet) {
            for (Planet existingPlanet : planets) {
                if (existingPlanet.x == planet.x || existingPlanet.y == planet.y) {
                    return false;
                }
            }
            return true;
        }

        public void addPlanet(Planet planet) {
            planets.add(planet);
        }

        @Override
        public int hashCode() {
            return planets.hashCode();
        }

        @Override
        public boolean equals(Object obj) {
            if (!(obj instanceof Universe)) {
                return false;
            }
            Universe other = (Universe) obj;
            return planets.equals(other.planets);
        }
    }
}
