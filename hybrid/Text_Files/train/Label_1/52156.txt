import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class ColorfulRibbon {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = reader.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        long[] colors = new long[n + 1];
        long[] colorfulness = new long[n + 1];

        for (int i = 1; i <= n; i++) {
            colors[i] = i;
        }

        for (int i = 0; i < m; i++) {
            String[] line = reader.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            if (type == 1) {
                int l = Integer.parseInt(line[1]);
                int r = Integer.parseInt(line[2]);
                long x = Long.parseLong(line[3]);
                update(colors, colorfulness, l, r, x);
            } else {
                int l = Integer.parseInt(line[1]);
                int r = Integer.parseInt(line[2]);
                System.out.println(query(colors, colorfulness, l, r));
            }
        }
    }

    private static void update(long[] colors, long[] colorfulness, int l, int r, long x) {
        long[] delta = new long[colors.length];
        for (int i = l; i <= r; i++) {
            delta[i] = x - colors[i];
            colors[i] = x;
            colorfulness[i] += Math.abs(delta[i]);
        }
        for (int i = l - 1; i >= 1; i--) {
            colorfulness[i] += delta[i + 1];
        }
        for (int i = r + 1; i <= colors.length - 1; i++) {
            colorfulness[i] += delta[i - 1];
        }
    }

    private static long query(long[] colors, long[] colorfulness, int l, int r) {
        return colorfulness[r] - colorfulness[l - 1];
    }
}
