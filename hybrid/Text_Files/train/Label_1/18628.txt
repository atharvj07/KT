#include <bits/stdc++.h>
    using namespace std;
    const int N=5005;
    int n,ans,l[N],r[N],mn[N];
    string s;
    int main(){
        cin >> s ;
        n=s.size();
        for (int i=n-1;i>=0;i--) {
            if (s[i]==')') r[i]=r[i+1];
            else l[i]=l[i+1]+1;
        }
        for (int i=n-1;i>=0;i--) {
            if (s[i]==')') r[i]=r[i+1];
            if (s[i]=='?') mn[i]=mn[i+1]+1;
            else mn[i]=0;
        }
        for (int i=n-1;i>=0;i--) {
            int x=0;
            if (s[i]==')') x=l[i];
            if (s[i]=='?') x=min(l[i],mn[i]);
            ans+=r[i]>x;
        }
        cout << ans << "\n" ;
        return 0;
    }
