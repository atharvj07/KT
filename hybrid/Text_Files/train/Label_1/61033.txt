n = int(input().strip())
a = list(map(int, input().strip().split()))

def check_zebra(a):
    stripe_width = 1
    for i in range(1, len(a)):
        if a[i] == a[i-1]:
            stripe_width += 1
        else:
            if i == stripe_width:
                expected_width = stripe_width
            elif stripe_width != expected_width:
                return "NO"
            stripe_width = 1
    return "YES" if stripe_width == expected_width else "NO"

print(check_zebra(a))
