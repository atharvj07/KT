#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++)
            cin >> a[i];
        vector<int> even, odd;
        for (int i = 0; i < n; i++)
        {
            if (i % 2 == a[i] % 2)
                continue;
            else if (i % 2)
                odd.push_back(i);
            else
                even.push_back(i);
        }
        if (odd.size() != even.size())
        {
            cout << "-1\n";
            continue;
        }
        sort(odd.begin(), odd.end());
        sort(even.begin(), even.end());
        int ans = 0;
        for (int i = 0; i < odd.size(); i++)
        {
            ans += odd[i] > even[i];
        }
        cout << ans << "\n";
    }
    return 0;
}
