n, x = map(int, input().split())

if n == 1:
    print("YES")
    print(x)
elif n == 2 and x == 0:
    print("NO")
else:
    print("YES")
    if n == 2:
        print(0, x)
    else:
        # Generate the first n-3 numbers as 0, 1, 2, ..., n-4
        a = list(range(n-3))
        # Calculate the xor of these numbers
        xor_a = 0
        for num in a:
            xor_a ^= num
        # The last three numbers will be the next power of 2 larger than n-3, 0 and the result to achieve x
        last_three = [2**(len(bin(n-3))-2), 0, 2**(len(bin(n-3))-2) ^ xor_a ^ x]
        print(*a, *last_three)
