#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mx=1e18+7;
ll bigMod(ll base, ll pow){
    if(pow==0) return 1;
    ll ret=bigMod(base, pow/2);
    ret=(ret*ret)%mx;
    if(pow%2==1) ret=(ret*base)%mx;
    return ret;
}
int main(){
    ll n, k;
    cin>>n>>k;
    ll low=0, high=n+1;
    ll ans=0;
    while(low<=high){
        ll mid=(low+high)>>1;
        ll temp=(mid*(mid+1))/2;
        if(temp+n>=k){
            ans=mid;
            high=mid-1;
        }
        else{
            low=mid+1;
        }
    }
    cout<<(k-ans-n)*bigMod(2, ans-1)%mx<<endl;
}
