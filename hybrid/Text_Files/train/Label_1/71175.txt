class DisjointSetUnion:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        xr, yr = self.find(x), self.find(y)
        if xr == yr:
            return False
        if self.rank[xr] < self.rank[yr]:
            self.parent[xr] = yr
        elif self.rank[xr] > self.rank[yr]:
            self.parent[yr] = xr
        else:
            self.parent[yr] = xr
            self.rank[xr] += 1
        return True

def kruskal(edges, n):
    dsu = DisjointSetUnion(n)
    mst_edges = []
    for w, u, v in sorted(edges):
        if dsu.find(u) != dsu.find(v):
            dsu.union(u, v)
            mst_edges.append((w, u, v))
    return mst_edges

def min_operations_to_unify_mst(n, m, edges):
    edges_grouped_by_weight = {}
    for u, v, w in edges:
        if w not in edges_grouped_by_weight:
            edges_grouped_by_weight[w] = []
        edges_grouped_by_weight[w].append((u-1, v-1))
    
    mst_edges = kruskal([(w, u-1, v-1) for u, v, w in edges], n)
    mst_weight_set = set([w for w, u, v in mst_edges])
    
    operations = 0
    for weight, grouped_edges in edges_grouped_by_weight.items():
        if weight in mst_weight_set:
            dsu = DisjointSetUnion(n)
            for w, u, v in mst_edges:
                if w < weight:
                    dsu.union(u, v)
            extra_edges = 0
            for u, v in grouped_edges:
                if dsu.find(u) != dsu.find(v):
                    dsu.union(u, v)
                    extra_edges += 1
                else:
                    extra_edges -= 1
            operations += max(0, extra_edges)
    return operations

n, m = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]
print(min_operations_to_unify_mst(n, m, edges))
