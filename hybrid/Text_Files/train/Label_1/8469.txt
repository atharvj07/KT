#include <iostream>
#include <string>
using namespace std;
string a, b;
int main() {
  string s;
  getline(cin, s);
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == ',' || s[i] == ';') {
      if (!a.empty() && b.empty() && (a[a.length() - 1] < '0' || a[a.length() - 1] > '9')) a.clear();
      if (!b.empty() && a.empty() && (b[b.length() - 1] >= '0' && b[b.length() - 1] <= '9')) b.clear();
      if (!a.empty() && !b.empty()) {
        if (a[a.length() - 1] < '0' || a[a.length() - 1] > '9') {
          a.clear();
          b += s[i];
        } else if (b[b.length() - 1] < '0' || b[b.length() - 1] > '9') {
          b.clear();
          a += s[i];
        } else {
          a += s[i];
        }
      } else {
        a += s[i];
      }
    } else {
      if (s[i] < '0' || s[i] > '9')
        b += s[i];
      else
        a += s[i];
    }
  }
  if (a.empty())
    cout << "-" << endl;
  else
    cout << '"' << a << '"' << endl;
  if (b.empty())
    cout << "-" << endl;
  else
    cout << '"' << b << '"' << endl;
  return 0;
}
