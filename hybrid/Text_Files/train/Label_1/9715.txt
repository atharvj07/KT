#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int LMT = 2 * 1e5 + 10;
int has[LMT];
string s[LMT];

int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);

  int n;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> s[i];
    has[s[i].size() * 100 + s[i][0]] = 1;
  }
  ll ans = 0;
  for (int i = 1; i <= n; ++i) {
    int m = s[i].size();
    int st = 0, en = m - 1;
    while (s[i][st] == ')') st++;
    while (s[i][en] == '(') en--;
    m -= st + en;
    if (m > 0) {
      ans += has[(m - 1) * 100 + ')'];
    }
    if (en > 0) {
      ans += has[(m + 1) * 100 + '('];
    }
    if (m == 0) ans--;
  }
  cout << ans / 2 << endl;
}
