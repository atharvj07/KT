#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    vector<int> col(n);
    for (int i = 0; i < n; i++){
        cin >> col[i];
    }
    int ans = 0, save = col[0];
    for (int i = 1; i < n; i++){
        if (save != col[i]){
            ans++;
        }
        save = col[i];
    }
    cout << ans << '\n';
    return 0;
}
