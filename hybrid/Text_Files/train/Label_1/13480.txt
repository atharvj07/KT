#include <iostream>
#include <algorithm>
#include <vector>
#include <cmath>

#define EPS (1e-9)

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<vector<double>> points(n, vector<double>(5));

  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < 5; ++j) {
      cin >> points[i][j];
    }
  }

  vector<bool> is_good(n, true);

  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      double dot_product = 0;
      double norm_i = 0;
      double norm_j = 0;
      for (int k = 0; k < 5; ++k) {
        dot_product += points[i][k] * points[j][k];
        norm_i += points[i][k] * points[i][k];
        norm_j += points[j][k] * points[j][k];
      }
      norm_i = sqrt(norm_i);
      norm_j = sqrt(norm_j);
      double angle = acos(dot_product / (norm_i * norm_j));
      if (angle > EPS) {
        is_good[i] = false;
        is_good[j] = false;
      }
    }
  }

  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (is_good[i]) {
      ++cnt;
    }
  }

  cout << cnt << endl;

  for (int i = 0; i < n; ++i) {
    if (is_good[i]) {
      cout << i + 1 << endl;
    }
  }

  return 0;
}
