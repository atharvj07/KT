from collections import defaultdict
n=int(input())
e=[tuple(map(int,input().split()))for _ in range(n-1)]
q=[1]
a=list(map(int,input().split()))
d=defaultdict(int)
for i in e:
    d[i[0]]^=1<<i[1]
    d[i[1]]^=1<<i[0]
for i in a:
    if i in q:
        q+=list(t for t in range(1,n+1)if(t not in q)and(d[i]&(1<<t)))
        q.remove(i)
    if not q:break
print('Yes' if not q else 'No')
