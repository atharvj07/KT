n = int(input())
skills = list(map(int, input().split()))

# Count the number of students good at each subject
programmers = [i + 1 for i, skill in enumerate(skills) if skill == 1]
mathematicians = [i + 1 for i, skill in enumerate(skills) if skill == 2]
sportsmen = [i + 1 for i, skill in enumerate(skills) if skill == 3]

# Find the maximum number of teams that can be formed
teams = min(len(programmers), len(mathematicians), len(sportsmen))

print(teams)
for i in range(teams):
    print(programmers[i], mathematicians[i], sportsmen[i])
