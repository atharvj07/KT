import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    long[] cards = new long[n];
    long maxSum = 0, prefixSum = 0, suffixSum = 0, maxPrefix = 0, maxSuffix = 0, ans = 0;
    for (int i = 0; i < n; i++) {
      cards[i] = input.nextInt();
      prefixSum += cards[i];
      maxPrefix = Math.max(maxPrefix, prefixSum);
    }
    for (int i = n - 1; i > 0; i--) {
      suffixSum += cards[i];
      maxSuffix = Math.max(maxSuffix, suffixSum);
      ans = Math.max(ans, maxPrefix + maxSuffix);
    }
    System.out.println(ans);
  }
}
