#include <iostream>
#include <algorithm>
#include <vector>

const int MOD = 1e9 + 7;
using ll = long long;

ll powmod(ll base, ll exp) {
    ll res = 1;
    while(exp > 0) {
        if (exp & 1) {
            res = (res * base) % MOD;
        }
        exp >>= 1;
        base = (base * base) % MOD;
    }
    return res;
}

int main() {
    ll n, f1, f2, f3, c;
    std::cin >> n >> f1 >> f2 >> f3 >> c;
    ll f4 = (f1 * f2) % MOD * f3 % MOD;
    if (n < 10) {
        ll ans = f4;
        for (ll i = 5; i <= n; ++i) {
            ans = (ans * f4) % MOD * c % MOD;
        }
        std::cout << ans << '\n';
        return 0;
    }
    ll ans = powmod((f4 * c) % MOD, n - 4);
    ll cnt = std::min(n % 4, 3LL);
    if (cnt > 0) {
        ll val = powmod(f4, cnt);
        ans = (val * ans) % MOD * c % MOD;
    }
    std::cout << ans << '\n';
    return 0;
}
