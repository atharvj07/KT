def read_int():
    return int(input())

def read_ints():
    return [int(x) for x in input().split()]
    
n, m, k = read_ints()

a = [[0] * m for i in range(n)]

for i in range(k):
    t, r, c = read_ints()
    if t == 1:
        for j in range(m):
            a[r-1][j] = c
    elif t == 2:
        for j in range(n):
            a[j][c-1] = c

for i in range(n):
    print(*a[i])
