import java.util.*;
import java.lang.*;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n], b = new long[n], c = new long[n];
        long side = Long.MIN_VALUE;
        for(int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
            c[i] = in.nextInt();
            side = Math.max(side, Math.max(a[i], Math.max(b[i], c[i])));
        }
        Arrays.sort(a);
        Arrays.sort(b);
        Arrays.sort(c);
        if(a[n-1] == b[n-1] && b[n-1] == c[n-1]) {
            System.out.println(1);
            System.out.println(1);
        } else if(a[n-1] == b[n-1]) {
            System.out.println(2);
            System.out.println(1 + " " + 2);
        } else if(a[n-1] == c[n-1]) {
            System.out.println(2);
            System.out.println(1 + " " + 3);
        } else if(b[n-1] == c[n-1]) {
            System.out.println(2);
            System.out.println(2 + " " + 3);
        } else {
            System.out.println(3);
            System.out.println(1 + " " + 2 + " " + 3);
        }
    }
}
