import java.util.*;

class Main {
    static int n, m;
    static char[][] map;
    static int[][] dis;
    static int[] dx = {1, -1, 0, 0};
    static int[] dy = {0, 0, 1, -1};
    static boolean[][] vis;

    static int bfs(int sx, int sy) {
        LinkedList<Integer> qx = new LinkedList<>();
        LinkedList<Integer> qy = new LinkedList<>();
        qx.add(sx);
        qy.add(sy);
        vis[sx][sy] = true;
        int cnt = 0;
        while (!qx.isEmpty()) {
            int x = qx.poll();
            int y = qy.poll();
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                if (vis[nx][ny] || map[nx][ny] == '#') continue;
                if (map[nx][ny] != '.' && map[nx][ny] != map[sx][sy]) return cnt;
                qx.add(nx);
                qy.add(ny);
                vis[nx][ny] = true;
            }
            cnt++;
        }
        return -1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        map = new char[n][m];
        dis = new int[n][m];
        vis = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                map[i][j] = s.charAt(j);
            }
        }
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!vis[i][j] && map[i][j] != '.') {
                    int tmp = bfs(i, j);
                    if (tmp == -1) {
                        System.out.println(-1);
                        return;
                    } else ans = Math.min(ans, tmp);
                }
            }
        }
        System.out.println(ans);
    }
}
