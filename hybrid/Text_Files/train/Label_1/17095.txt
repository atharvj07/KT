#include <bits/stdc++.h>
using namespace std;
string A,B;
int dp[1005][1005];
int fun(int i,int j)
{
    if(i==A.size())
    {
        if(j==B.size()) return 0;
        else return -1e9;
    }
    if(j==B.size()) return 0;
    int &ret=dp[i][j];
    if(~ret) return ret;
    ret=0;
    if(A[i]==B[j])
    {
        ret=max(ret,1+fun(i+1,j+1));
    }
    ret=max(ret,fun(i+1,j));
    ret=max(ret,fun(i,j+1));
    return ret;
}
int main()
{
    memset(dp,-1,sizeof(dp));
    cin>>A>>B;
    int ans=fun(0,0);
    if(ans<=0) cout<<-1<<endl;
    else cout<<ans<<endl;
    return 0;
}
