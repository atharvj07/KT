#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;

ll fpow(ll a, ll b) {
  ll ans = 1;
  while (b != 0) {
    if (b % 2 == 1) {
      ans = (ans * a) % mod;
    }
    a = (a * a) % mod;
    b /= 2;
  }
  return ans;
}

int main() {
  ll a, b, n, x;
  cin >> a >> b >> n >> x;
  if (n == 0) {
    cout << x % mod << endl;
    return 0;
  }
  ll y = fpow(a, n) % mod;
  ll z = fpow(y - 1, mod - 2) % mod;
  ll ans = ((y - 1) * z % mod) * b % mod;
  ans += x;
  cout << ans % mod << endl;
  return 0;
}
