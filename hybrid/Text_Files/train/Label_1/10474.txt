#include<bits/stdc++.h>
using namespace std;

int main() {
    int n, m, d;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    sort(a.rbegin(), a.rend());
    cin >> m;
    vector<int> b(m);
    for (int i = 0; i < m; ++i) {
        cin >> b[i];
    }
    sort(b.rbegin(), b.rend());
    int a_score = 0, b_score = 0;
    for (int i = 0; i < n; ++i) {
        for (d = 0; d < 2000000000; ++d) {
            if (a[i] <= d) {
                break;
            }
        }
        a_score += (d + 1);
    }
    for (int i = 0; i < m; ++i) {
        for (d = 0; d < 2000000000; ++d) {
            if (b[i] <= d) {
                break;
            }
        }
        b_score += (d + 1);
    }
    cout << a_score << ":" << b_score << endl;
}
