import sys

n, m = map(int, input().split())
colors = [0] + list(map(int, input().split()))
graph = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

cnt = [0] * (n + 1)
for i in range(1, n + 1):
    color = colors[i]
    for j in graph[i]:
        if colors[j] != color:
            cnt[color] += 1
color = 1
result = cnt[1]
for i in range(2, n + 1):
    if cnt[i] > result:
        result = cnt[i]
        color = i

print(color)
