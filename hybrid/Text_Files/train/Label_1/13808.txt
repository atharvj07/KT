#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> grades(n);
  for (int i = 0; i < n; ++i) {
    cin >> grades[i];
  }
  sort(grades.begin(), grades.end());
  int sum = 0;
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    sum += grades[i];
    if (sum * 1.0 / (i + 1) < 4.5) {
      ++cnt;
    }
  }
  cout << cnt << endl;
  return 0;
}
