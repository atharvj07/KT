#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n; cin >> n;
    string s; cin >> s;

    vector<int> color(n);

    vector<int> l, r;
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            l.push_back(i);
        } else {
            if (l.empty()) {
                color[i] = 1;
            } else {
                r.push_back(i);
            }
            l.pop_back();
        }
    }

    while (!l.empty()) {
        color[l.back()] = 1;
        l.pop_back();
    }

    for (int i = 0; i < n; i++) {
        if (color[i] == 0) cout << '0';
        else cout << '1';
    }

    cout << endl;

    return 0;
}
