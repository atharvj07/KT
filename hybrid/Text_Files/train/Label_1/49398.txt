import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            int min = x[i];
            int minIndex = i;
            for (int j = i + 1; j < n; j++) {
                if (x[j] < min) {
                    min = x[j];
                    minIndex = j;
                }
            }
            if (minIndex != i) {
                ans += min;
                x[minIndex] = x[i];
                x[i] = min;
            }
        }
        System.out.println(ans);
    }
}
