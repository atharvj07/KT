import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    p = list(map(int, input().split()))
    foe = [[] for i in range(n + 1)]
    for _ in range(m):
        x, y = map(int, input().split())
        foe[x].append(y)
        foe[y].append(x)
    ans = 0
    for i in range(1, n + 1):
        st = []
        ok = True
        for j in range(1, n + 1):
            if p[j - 1] == i:
                st.append(j)
            elif len(st) > 0 and p[j - 1] in foe[st[-1]]:
                ok = False
                break
            elif len(st) > 0:
                ans += len(st)
                st = []
                st.append(j)
        if ok and len(st) > 0:
            ans += len(st)
    print(ans)

solve()
