#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, t, k;
    cin >> n >> t >> k;
    vector<int> a(t);
    for (int i = 0; i < t; ++i) {
        cin >> a[i];
    }
    if (a.back() < k) {
        cout << "-1" << endl;
        return 0;
    }
    vector<int> id(n + 1);
    for (int i = 1; i <= n; ++i) {
        id[i] = i;
    }
    a.back() -= k;
    int cnt = 0;
    for (int i = t - 1; i >= 0; --i) {
        while (a[i] > 0) {
            if (cnt == k) {
                break;
            }
            id[id[1]] = id[n];
            --a[i];
            ++cnt;
        }
    }
    vector<int> p(n);
    for (int i = 1; i <= n; ++i) {
        p[i - 1] = id[i];
    }
    sort(p.begin(), p.end());
    cout << n << endl;
    for (int i = 1; i < n; ++i) {
        cout << p[i] << " " << p[i - 1] << endl;
    }
    return 0;
}
