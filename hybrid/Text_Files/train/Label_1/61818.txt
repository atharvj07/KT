import math

def angle(x, y):
    return math.atan2(y, x) % (2 * math.pi)

n = int(input().strip())
vectors = [tuple(map(int, input().split())) for _ in range(n)]

angles = [(angle(x, y), i) for i, (x, y) in enumerate(vectors, start=1)]
angles.sort()

min_angle = 2 * math.pi
answer = None

for i in range(n):
    diff = (angles[(i + 1) % n][0] - angles[i][0]) % (2 * math.pi)
    if diff < min_angle:
        min_angle = diff
        answer = (angles[i][1], angles[(i + 1) % n][1])

print(*answer)
