#include <bits/stdc++.h>
#define mod 998244353
typedef long long ll;
using namespace std;

ll pow_mod(ll b, ll p) {
    ll ans = 1;
    while (p) {
        if (p & 1) ans = ans * b % mod;
        b = b * b % mod;
        p /= 2;
    }
    return ans % mod;
}

int main() {
    int n, k, len;
    cin >> n >> k >> len;
    vector<int> v(n);
    for (int &el : v) { cin >> el; }
    int cnt = 0;
    vector<int> last(k + 2);
    for (int i = 0; i <= k + 1; i++) { last[i] = -n - 1; }
    for (int i = 0; i < n; i++) {
        if (v[i] == -1) cnt++;
        else {
            cnt = 0;
            last[v[i]] = i;
        }
    }
    ll ans = pow_mod(k, cnt);
    for (int i = 1; i <= k; i++) {
        ans = (ans - pow_mod(k - 1, last[i] - last[i - 1] - 1) + mod) % mod;
    }
    cout << ans << "\n";
}
