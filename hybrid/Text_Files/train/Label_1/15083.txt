#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll t;
int main() {
    cin>>t;
    while(t--) {
        ll n,k1,k2,i,j;
        cin>>n>>k1>>k2;
        vector<ll>v1(k1),v2(k2);
        for(i=0; i<k1; i++) cin>>v1[i];
        for(i=0; i<k2; i++) cin>>v2[i];
        sort(v1.begin(),v1.end());
        sort(v2.begin(),v2.end());
        if((v1[0]%2==0 && v2[0]%2==1) || (v1[0]%2==1 && v2[0]%2==0)) cout<<"YES\n";
        else cout<<"NO\n";
    }
    return 0;
}
