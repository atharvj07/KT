#include <iostream>
using namespace std;
typedef long long ll;
const ll M=1000000009LL;
int n,k,a,b,s[100010];
int main(){
    cin>>n>>a>>b>>k;
    for(int i=0;i<k;i++){
        char c;
        cin>>c;
        s[i]=(c=='+'?1:-1);
    }
    ll ans=0;
    for(int i=0;i<k;i++){
        ll aa=a,bb=b;
        aa%=M;bb%=M;
        for(int j=0;j<n;j++){
            ans=(ans+(ll)s[i]*aa%M*bb%M)%M;
            aa=aa*a%M;
            bb=bb*b%M;
        }
    }
    cout<<ans<<endl;
}
