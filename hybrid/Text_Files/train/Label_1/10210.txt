#include <vector>
#include <algorithm>
#include <iostream>

using std::vector;
using std::min;
using std::max;

vector<long long> get_dp(long long n) {
    vector<long long> dp(n + 1, n + 1);
    dp[0] = 0;
    for (long long i = 1; i <= n; ++i) {
        if (i % 3 == 0) {
            dp[i] = min(dp[i], dp[i / 3] + 1);
        }
        if (i % 2 == 0) {
            dp[i] = min(dp[i], dp[i / 2] + 1);
        }
        dp[i] = min(dp[i], dp[i - 1] + 1);
    }
    return dp;
}

int main() {
    long long n;
    std::cin >> n;

    vector<long long> dp = get_dp(n);
    std::cout << dp[n] << std::endl;
}
