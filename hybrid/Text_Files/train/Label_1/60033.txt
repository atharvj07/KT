n, m, k = map(int, input().split())
a = [[0] * m for _ in range(n)]
for i in range(n):
    s = input()
    for j in range(m):
        a[i][j] = int(s[j])

ans = 0
for i in range(n):
    cnt = 0
    for j in range(m):
        if a[i][j] == 1:
            cnt += 1
        if cnt <= k + 1:
            ans += 1
        else:
            break

print(ans)
