import java.util.*;

class Main {
  static int n, m, k, ans = 0;
  static int[][] a;

  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    n = scan.nextInt();
    m = scan.nextInt();
    k = scan.nextInt();
    a = new int[n][m];

    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        a[i][j] = scan.nextInt();
      }
    }

    for (int x1 = k - 1; x1 < n - k + 1; x1++) {
      for (int y1 = k - 1; y1 < m - k + 1; y1++) {
        int cur = a[x1][y1];
        for (int x2 = x1; x2 < n - k + 1; x2++) {
          for (int y2 = y1; y2 < m - k + 1; y2++) {
            cur = Math.max(cur, a[x2][y2]);
          }
        }
        if (cur > ans) {
          ans = cur;
          x1 = x1 - k + 1;
          y1 = y1 - k + 1;
        }
      }
    }

    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        int cur = a[i][j];
        for (int x = i; x < n - k + 1; x++) {
          for (int y = j; y < m - k + 1; y++) {
            cur = Math.max(cur, a[x][y]);
          }
        }
        if (cur == ans) {
          System.out.println((i + 1) + " " + (j + 1));
          return;
        }
      }
    }
  }
}
