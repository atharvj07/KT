import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong(), m = sc.nextLong(), k = sc.nextLong();
        long row = k % n + 1, col = (k / n) % m + 1;
        if (row == n) {
            row = n - (k % n) % (n - 1) - 1;
            if (row <= 0) {
                row = 1;
            }
        }
        System.out.println(row + " " + col);
    }
}
