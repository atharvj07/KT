import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextInt();
        long b = in.nextInt();

        long ans = 0;
        StringBuilder sb = new StringBuilder();

        while (a > 0 && b > 0) {
            if (ans + a * a - b * b > ans + b * b - a * a) {
                ans += a * a - b * b;
                sb.append("o".repeat((int) a));
                a = 0;
            } else {
                ans += b * b - a * a;
                sb.append("x".repeat((int) b));
                b = 0;
            }
        }

        while (a > 0) {
            ans += a * a;
            sb.append("o".repeat((int) a));
            a = 0;
        }

        while (b > 0) {
            ans -= b * b;
            sb.append("x".repeat((int) b));
            b = 0;
        }

        System.out.println(ans);
        System.out.println(sb);
    }
}
