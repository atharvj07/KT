import sys

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    ans = 0
    for i in range(n - 1):
        if a[i] > b[i]:
            ans += 1
            a[i], a[i + 1] = a[i + 1], a[i]
    print(ans)
