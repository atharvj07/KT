#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a, b = 0, c = 0, d = 0;
  cin >> n;
  while (n--) {
    cin >> a;
    if (a % 2)
      c++;
    else if (a % 3 == 0)
      b++;
    else
      d++;
  }
  if (d || ((b != 0 || c != 0) && b != c))
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
}
