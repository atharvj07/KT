#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  if (n < k)
    cout << -1 << endl;
  else if (k == 1)
    cout << string(n, 'a') << endl;
  else if (n / 2 < k)
    cout << -1 << endl;
  else {
    string s(n, 'a');
    int p = 1;
    for (; p * 2 <= n; ++p) {
      for (int i = p; i < n; i = (i % (2 * p)) + p) s[i] = 'a' + p;
    }
    if (n % 2 == 1) {
      int d = 'a' + p - 1;
      for (; p < k; ++p) {
        s[n - 1 - p] = d;
        d--;
      }
    } else {
      int d = 'a' + p;
      for (; p < k; ++p) {
        s[n - 1 - p] = d;
        s[n - 2 - p] = d - 1;
        d += 2;
      }
    }
    cout << s << endl;
  }
  return 0;
}
