#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    int ans = a[n - 1] - a[0];
    for (int i = 1; i < n - 1; i++) {
        ans = std::min(ans, a[i + 1] - a[i - 1]);
    }
    std::cout << ans << std::endl;
    return 0;
}
