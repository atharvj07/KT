#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'

int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n,m; cin>>n>>m;
    ll p[n+5];
    for(int i=1; i<=n; i++) cin>>p[i];
    map<pair<int,int>, ll> mp;
    for(int i=1; i<=m; i++)
    {
        int x,y; cin>>x>>y;
        mp[{x,y}]++, mp[{y,x}]++;
    }
    ll ans=0;
    for(int i=1; i<=n; i++)
    {
        ll l=i-1;
        ll r=n-i;
        ans+=l*r;
        for(int j=i+1; j<=n; j++)
        {
            if(mp.count({p[i],p[j]}))
               ans-=((j-i)*r*l)/mp[{p[i],p[j]}];
        }
    }
    cout<<ans<<endl;

    return 0;
}
