n,m,k,q=map(int,input().split())
a=[[0]*m for i in range(n)]
for i in range(q):
    x,y,t=map(int,input().split())
    if t>a[x-1][y-1]:
        x-=1;y-=1
        for i in range(x,x+k):
            for j in range(y,y+k):
                a[i][j]=t
        if i+1>=k and j+1>=k:
            print(t)
            exit()
print(-1)
