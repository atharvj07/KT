n = int(input())
p = list(map(int, input().split()))

def min_inversions(arr):
    count = 0
    for i in range(len(arr)):
        for j in range(i+1, len(arr)):
            if arr[i] > arr[j]:
                count += 1
    return count

print(min(min_inversions(p), min_inversions([-x for x in p])))
