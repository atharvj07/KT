#include <bits/stdc++.h>
using namespace std;

long long cx[100005], cy[100005], n, k, ans = LLONG_MAX;
bool used[100005];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) { cin >> cx[i] >> cy[i]; }
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (cx[i] == cx[j]) {
                ans = min(ans, (cy[j] - cy[i]) * (cy[j] - cy[i]));
            }
            if (cy[i] == cy[j]) {
                ans = min(ans, (cx[j] - cx[i]) * (cx[j] - cx[i]));
            }
        }
    }
    cout << ans << endl;
    return 0;
}
