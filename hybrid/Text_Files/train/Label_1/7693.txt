#include<bits/stdc++.h>
#define ll long long
#define PI 3.14159265358979323846
using namespace std;
int main(){
  ll a,b,c,d,x,y;
  cin>>a>>b;
  ll ans=0,sum1=0,sum2=0,e=0;
  for(int i=0;i<b;i++){
    cin>>x>>y;
    sum1+=x+y*a;
    if(y>0)
      sum2+=y;
    if(y<0)
      e+=1;
  }
  if(e==b){
    cout<<fixed<<setprecision(9)<<(double)sum1/(double)a<<endl;
    return 0;
  }
  ans=sum1-sum2;
  int p;
  if(e){
    p=b-e;
    ans+=p*a;
  }
  else{
    p=b-1;
    ans+=p*a;
  }
  cout<<fixed<<setprecision(9)<<(double)ans/(double)a<<endl;
  return 0;
}
