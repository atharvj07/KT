#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, k, t, ans = 0;
  cin >> n >> k;
  vector<long long> a(n);
  for (auto &ai : a) cin >> ai;
  for (long long i = 0; i < n; i++) {
    t = a[i];
    while (t != 0) {
      if (t % 2 == 1) { ans += (k - __builtin_popcountll(a[i] ^ (1LL << k) - 1)); break; }
      t /= 2;
    }
  }
  cout << ans << endl;
}
