import sys
import collections

n, m = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))

count = collections.Counter(a)
freq = sorted([(count[i], i) for i in range(1, m + 1)], reverse=True)
val = 0
for cnt, i in freq:
  val = cnt
  break

ans = []
for ai in a:
  if count[ai] > val:
    ans.append(ai)
    count[ai] -= 1
  else:
    ok = False
    for j in range(1, m + 1):
      if count[j] > 0:
        count[j] -= 1
        ans.append(j)
        ok = True
        break
    if not ok:
      ans.append(ai)

print(val, len(a) - len(ans))
print(*ans)
