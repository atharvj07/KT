#include <bits/stdc++.h>
using namespace std;
int kmp[1000005];
const int MAXN = 1000005;
string t, p;
vector<long long> dp(MAXN, -1);
long long solve(int pos) {
    if (pos == t.length()) return dp[pos] = 1;
    if (~dp[pos]) return dp[pos];
    dp[pos] = 0;
    for (int i = pos; i < t.length(); i++) {
        for (int j = 0; j < p.length(); j++) {
            if (p[j] == t[i]) {
                dp[pos] += solve(i + 1);
                break;
            }
        }
    }
    return dp[pos];
}
void build() {
    int n = p.length();
    for (int i = 1; i < n; i++) {
        int j = kmp[i - 1];
        while (j > 0 && p[i] != p[j]) j = kmp[j - 1];
        kmp[i] = j + (p[i] == p[j]);
    }
}
int main() {
    cin >> p;
    cin >> t;
    build();
    cout << solve(0) << endl;
    return 0;
}
