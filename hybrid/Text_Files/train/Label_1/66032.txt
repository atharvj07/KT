n, k = map(int, input().split())
a = list(map(int, input().split()))
need = []
b = [0] * (n + 1)
for i in a:
    if b[i] == 0:
        if len(need) < k:
            need.append(i)
            b[i] = 1
        else:
            j = need.index(a[i - 1])
            b[need.pop(j)] = 0
            need.insert(j, i)
            b[i] = 1
print(len(need))
