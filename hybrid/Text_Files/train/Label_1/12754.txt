#include <iostream>
#include <string>
#include <vector>

int main() {
  std::string s;
  std::cin >> s;
  int n = s.size();
  std::vector<int> pref(n);
  std::vector<int> suff(n);
  for (int i = 0; i < n; ++i) {
    if (i > 0) pref[i] = pref[i - 1];
    if (s[i] != '=') pref[i] += s[i] - '0';
  }
  for (int i = n - 1; i >= 0; --i) {
    if (i < n - 1) suff[i] = suff[i + 1];
    if (s[i] != '=') suff[i] += s[i] - '0';
  }
  for (int i = 0; i < n; ++i) {
    if (s[i] == '^') {
      if (pref[i] == suff[i]) {
        std::cout << "balance" << std::endl;
      } else if (pref[i] > suff[i]) {
        std::cout << "left" << std::endl;
      } else {
        std::cout << "right" << std::endl;
      }
      return 0;
    }
  }
  return 0;
}
