import sys

n, m = map(int, input().split())
files = [input() for _ in range(n)]
inds = [int(idx)-1 for idx in input().split()]
filelens = set(len(filename) for filename in files)
if len(filelens) != 1:
    print('No')
    sys.exit()
pat = files[0]
for i in range(len(pat)):
    if not all(filename[i] == pat[i] or pat[i] == '?' for filename in files):
        print('No')
        sys.exit()
    if not all(filename[i] == pat[i] for i in inds):
        print('No')
        sys.exit()
print('Yes')
print(pat.replace('?', '['))
