#include <bits/stdc++.h>
using namespace std; typedef long long ll;
ll gcd(ll a, ll b) { if (a == 0) return b; return gcd(b % a, a); }
int main() {
    ll a, b, c, d; cin >> a >> b >> c >> d;
    if (c * b == d * a) cout << "0/1\n";
    else if (c * b < d * a) cout << (d * a - c * b) / gcd(c * b, d * a) << '/' << d * a / gcd(c * b, d * a) << '\n';
    else cout << c * b / gcd(c * b, d * a) << '/' << (d * a - c * b) / gcd(c * b, d * a) << '\n';
    return 0;
}
