from collections import defaultdict

def solve(n, edges):
    graph = defaultdict(list)
    for i, (u, v) in enumerate(edges, start=1):
        graph[u].append((v, i))
        graph[v].append((u, i))
    
    max_degree = max(len(graph[node]) for node in graph)
    result = [[] for _ in range(max_degree)]
    visited = set()
    
    def dfs(node, parent=-1, day=0):
        visited.add(node)
        current_day = day
        for neighbor, edge_number in graph[node]:
            if neighbor not in visited:
                result[current_day].append(edge_number)
                current_day = (current_day + 1) % max_degree
                dfs(neighbor, node, current_day)
                
    dfs(1)
    print(max_degree)
    for day in result:
        print(len(day), *day)

n = int(input().strip())
edges = [tuple(map(int, input().strip().split())) for _ in range(n - 1)]
solve(n, edges)
