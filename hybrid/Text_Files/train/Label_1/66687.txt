import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
w = list(map(int, input().split()))
a = []
for i in range(m):
    ai, bi = map(lambda x: int(x) - 1, input().split())
    a.append((ai, bi, w[ai] + w[bi]))
a.sort(key=lambda x: x[2])
ans = {}
for i in range(n):
    ans[i] = [0] * k
for i in range(m):
    ai, bi, wi = a[i]
    ans[ai][ans[ai].index(max(ans[ai]))] += 1
    ans[ai][ans[ai].index(min(ans[ai]))] -= 1
    ans[bi][ans[bi].index(max(ans[bi]))] += 1
    ans[bi][ans[bi].index(min(ans[bi]))] -= 1
for i in range(n):
    ans[i].sort()
    if ans[i][0] < ans[i][1] or ans[i][-1] - ans[i][-2] >= 3:
        for s in range(k):
            ans[i][s] = 0
for ai, bi in a:
    if ans[ai].count(ans[ai][0]) == ans[ai].count(ans[ai][1]) == 0:
        x = ans[ai].pop(ans[ai].index(ans[ai][0]))
        y = ans[bi].pop(ans[bi].index(ans[bi][0]))
        ans[ai].extend([x, y])
        ans[bi].extend([x, y])
    else:
        ans[ai].pop(ans[ai].index(ans[ai][0]))
        ans[bi].pop(ans[bi].index(ans[bi][0]))
for _, s in a:
    print(1 + ans[ai] + ans[bi])
