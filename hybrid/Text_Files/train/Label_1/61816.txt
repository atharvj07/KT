import math

n = int(input())
a = []
for i in range(n):
    x, y = map(int, input().split())
    a.append((math.atan2(y, x), x, y))
a.sort()
ans = [n, 0]
for i in range(n):
    for j in range(i + 1, n):
        ang = abs(a[i][0] - a[j][0])
        if ang > math.pi: ang = math.pi * 2 - ang
        if ang < ans[0]: ans = [ang, a[i][1], a[j][1]]
print(ans[1], ans[2])
