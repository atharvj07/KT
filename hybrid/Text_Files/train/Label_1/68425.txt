n, m = map(int, input().split())
files = [input() for _ in range(n)]
to_delete = [files[int(i)-1] for i in input().split()]

mask = ['?' for _ in files[0]]
for i in range(len(files[0])):
    for f in to_delete:
        if files[0][i] != f[i] and mask[i] != '?':
            mask[i] = '?'
            break

if all(mask[i] == '?' or mask[i] == files[0][i] for i in range(len(files[0])) for f in files if f not in to_delete):
    print('Yes')
    print(''.join(mask))
else:
    print('No')
