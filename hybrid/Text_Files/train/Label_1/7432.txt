#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<bitset<10>> v(n);
    for (auto &x : v) {
        for (int i = 0; i < 6; i++) {
            int a;
            cin >> a;
            x.set(a);
        }
    }
    int ans = 0;
    for (int i = 1; i <= 9; i++) {
        bool ok = 1;
        for (int j = i + 1; j <= 9; j++) {
            ok &= !((v[0][i] & v[1][j]) | (v[1][i] & v[2][j]) | (v[2][i] & v[0][j]));
        }
        if (ok) ans = i;
    }
    cout << ans << endl;
}
