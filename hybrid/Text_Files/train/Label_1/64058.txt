import math

n = int(input())
a = list(map(int, input().split()))

g = [0] * 1000001
for x in a:
    for i in range(1, int(math.sqrt(x)) + 1):
        if x % i == 0:
            g[x] = math.gcd(g[x], i)
            g[x] = math.gcd(g[x], x // i)

ans = 0
for x in a:
    for i in range(1, int(math.sqrt(x)) + 1):
        if x % i == 0:
            ans += (g[x] * i) % 1000000007
            ans += (g[x] * (x // i)) % 1000000007
    if int(math.sqrt(x)) * int(math.sqrt(x)) == x:
        ans -= g[x]
print(ans % 1000000007)
