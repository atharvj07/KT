#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;cin>>n;
    int w[n],h[n];
    int sum=0, mx=0;
    for(int i=0;i<n;i++){
        cin>>w[i]>>h[i];
        sum+=w[i];
        mx=max(mx,h[i]);
    }
    for(int i=0;i<n;i++){
        cout<<(sum-w[i])*mx<<' ';
    }
    cout<<endl;
    return 0;
}
