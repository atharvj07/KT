n, m = map(int, input().split())
x = list(map(int, input().split()))

# Precompute the positions for each p_i(n)
pos = [[i for i in range(1, n + 1)] for _ in range(n)]
for i in range(n):
    pos[i] = [i + 1] + pos[i][:i] + pos[i][i + 1:]

# Function to calculate f(p)
def f(p, x):
    total_distance = 0
    for i in range(len(x) - 1):
        total_distance += abs(p[x[i] - 1] - p[x[i + 1] - 1])
    return total_distance

# Calculate f(p_i(n)) for all i
results = []
for i in range(n):
    results.append(f(pos[i], x))

print(*results)
