import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] s = new int[M + 1];
        int[] c = new int[M + 1];
        for (int i = 1; i <= M; i++) {
            s[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        int[] ans = new int[N + 1];
        for (int i = 1; i <= M; i++) {
            if (s[i] == 1 && c[i] > 0) {
                System.out.println(-1);
                return;
            }
            ans[s[i]] = c[i];
        }
        if (ans[1] == 0) {
            System.out.println(-1);
            return;
        }
        for (int i = 1; i <= N; i++) {
            System.out.print(ans[i]);
        }
        System.out.println();
    }
}
