import sys
input = sys.stdin.readline
n,s = map(int,input().split())
p=[list(map(int,input().split())) for _ in range(n)]
p.sort(key=lambda x:-x[0])
a,b,f=1e12,1e12,1
for x,v,t in p:
    v,t=min(s-1,v*t),min(s-1,v*t+1)
    if t==1:a=min(a,max(1,(1000000-x)//v))
    if t==2:b=min(b,(x+1)//v)
    f=min(f,a+b)
print(f'{f:.15f}')
