import sys
input = sys.stdin.readline

n = int(input())
s = input()
l = []
r = []
for i in range(n):
    if s[i] == '?':
        l.append(i)
    elif s[i] == '(':
        r.append(i)
    else:
        if len(r):
            r.pop()
        else:
            l.append(i)
for i in l:
    if i < r[0]:
        print(':)')
        break
    s = s[:i] + ')' + s[i+1:]
print(s)
