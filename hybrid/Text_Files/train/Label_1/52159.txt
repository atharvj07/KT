import java.util.Scanner;

public class ColorfulRibbon {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] colors = new int[n + 1];
        long[] colorfulness = new long[n + 1];

        for (int i = 1; i <= n; i++) {
            colors[i] = i;
        }

        for (int i = 0; i < m; i++) {
            int type = scanner.nextInt();
            if (type == 1) {
                int l = scanner.nextInt();
                int r = scanner.nextInt();
                int x = scanner.nextInt();
                paint(colors, colorfulness, l, r, x);
            } else {
                int l = scanner.nextInt();
                int r = scanner.nextInt();
                long sum = query(colors, colorfulness, l, r);
                System.out.println(sum);
            }
        }
    }

    private static void paint(int[] colors, long[] colorfulness, int l, int r, int x) {
        for (int i = l; i <= r; i++) {
            colorfulness[i] += Math.abs(x - colors[i]);
            colors[i] = x;
        }
    }

    private static long query(int[] colors, long[] colorfulness, int l, int r) {
        long sum = 0;
        for (int i = l; i <= r; i++) {
            sum += colorfulness[i];
        }
        return sum;
    }
}
