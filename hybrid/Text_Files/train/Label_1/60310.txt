from math import inf
n=int(input())
l=input().split()
dp=[inf]*n
m=len(l)
dp[n-1]=len(l[n-1])
for i in range(n-2,-1,-1):
    l[i]=l[i]+'.'
    li=len(l[i])
    mn=inf
    dp[i]=min(dp[i],mn+li)
    for j in range(i+1,n):
        if len(l[i])==len(l[j]) and l[i]==l[j]:
            mn=min(mn,dp[j]-li+1)
        dp[i]=min(dp[i],mn+li)
print(dp[0])
