import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        char[] balls = scanner.nextLine().toCharArray();
        int n = balls.length;
        int[] left = new int[n + 1];
        int[] right = new int[n + 1];
        int ans = 0;

        for (int i = 1; i <= n; i++) {
            if (i == 1 || balls[i - 1] != balls[i - 2]) {
                left[i] = i - 1;
            } else {
                left[i] = left[i - 1];
            }
        }

        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1 || balls[i] != balls[i + 1]) {
                right[i] = i + 1;
            } else {
                right[i] = right[i + 1];
            }
        }

        for (int i = 1; i < n; i++) {
            char color = balls[i - 1];
            int l = left[i];
            int r = right[i];
            if (l > 0 && balls[l - 1] == color) {
                ans += r - l + 1;
            }
        }

        System.out.println(ans);
    }
}
