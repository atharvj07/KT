#include <bits/stdc++.h>
using namespace std;
#define fast ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0)
int n, m, ans = 0;
char arr[2505][2505];
void compress(int r, int c) {
  int a = 0, b = 0, c1 = 0, d1 = 0;
  for (int i = r; i < r + 2; i++) {
    for (int j = c; j < c + 2; j++) {
      if ((i + j) % 2) {
        b += arr[i][j] == '0';
        d1 += arr[i][j] == '1';
      } else {
        a += arr[i][j] == '0';
        c1 += arr[i][j] == '1';
      }
    }
  }
  ans += min(a + d1, b + c1);
}
int main() {
  fast;
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> arr[i][j];
  n = ((n + 1) / 2) * 2, m = ((m + 1) / 2) * 2;
  for (int i = 0; i < n; i += 2)
    for (int j = 0; j < m; j += 2) compress(i, j);
  cout << ans << endl;
  return 0;
}
