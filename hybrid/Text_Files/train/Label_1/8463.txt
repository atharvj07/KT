#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const ll mod = 998244353;

int main() {
    int n;
    cin >> n;
    vector<int> c(n);
    for (int &i : c) cin >> i;
    vector<ll> dp(n + 1, 0);
    dp[1] = 1;
    for (int i = 2; i <= n; i++) {
        ll sum = 0;
        for (int j = 0; j < i; j++) {
            if (c[i - 1] == c[j]) {
                sum += dp[j] * dp[i - 1 - j];
                sum %= mod;
            }
        }
        dp[i] = sum;
    }
    cout << dp[n] << endl;
}
