from sys import stdin
from collections import deque

def bfs(adj_list, values):
    queue = deque([1])
    visited = [False] * (len(adj_list) + 1)
    visited[1] = True
    increase_ops = 0
    decrease_ops = 0

    while queue:
        vertex = queue.popleft()
        for child in adj_list[vertex]:
            if not visited[child]:
                visited[child] = True
                queue.append(child)
                # Calculate the operation needed to make the child's value 0
                required_ops = values[vertex] - values[child]
                if required_ops > 0:
                    increase_ops += required_ops
                else:
                    decrease_ops -= required_ops
                # Update the child's value to make it zero in the next steps
                values[child] = values[vertex]

    return max(increase_ops, decrease_ops)

def solve():
    n = int(stdin.readline())
    adj_list = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        a, b = map(int, stdin.readline().split())
        adj_list[a].append(b)
        adj_list[b].append(a)
    values = [0] + list(map(int, stdin.readline().split()))
    print(bfs(adj_list, values))

solve()
