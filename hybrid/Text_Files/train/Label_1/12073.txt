#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        int x = a[n - 1];
        int cnt = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (a[i] > x) cnt++;
            else {
                a[i] = x + 1;
                cnt = 0;
            }
        }
        cout << cnt << endl;
    }
    return 0;
}
