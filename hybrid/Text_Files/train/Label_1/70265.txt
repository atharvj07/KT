import math

for _ in range(int(input())):
    n, k = map(int, input().split())
    if n % k > (k - 1):
        print(*[math.ceil(n / k)] * k)
    else:
        ans = [(n // k)] * k
        a = n % k
        b = n // k
        while a:
            ans[a - 1] += 1
            a -= 1
        print(*ans)
