from collections import deque

def bfs(graph, n):
    dist = [float('inf')] * n
    dist[0] = 0
    queue = deque([0])
    paths = [1] * n

    while queue:
        v = queue.popleft()
        for u in graph[v]:
            if dist[u] == float('inf'):
                dist[u] = dist[v] + 1
                queue.append(u)
                paths[u] = paths[v]
            elif dist[u] == dist[v] + 1:
                paths[u] += paths[v]

    return dist, paths

n, m, k = map(int, input().split())
graph = [[] for _ in range(n)]
edges = []

for _ in range(m):
    a, b = map(int, input().split())
    graph[a - 1].append(b - 1)
    graph[b - 1].append(a - 1)
    edges.append((a - 1, b - 1))

dist, paths = bfs(graph, n)

options = []
for edge in edges:
    if dist[edge[0]] + 1 == dist[edge[1]] or dist[edge[1]] + 1 == dist[edge[0]]:
        options.append('1')
    else:
        options.append('0')

print(min(k, paths[-1]))
for i in range(min(k, paths[-1])):
    print(''.join(options))
