def expected_races(c, m, p, v):
    def calc_exp(c, m, p, v, depth):
        # Base case: if Pink Slip is drawn, return the depth
        if p == 1:
            return depth
        exp = p * depth  # Expectation value when Pink Slip is drawn
        if c > 0:
            # If Cash Slip is drawn
            new_v = min(c, v)
            distributed = new_v / (2 if m > 0 else 1)
            exp += calc_exp(max(c - new_v, 0), m + distributed * (m > 0), p + distributed, v, depth + 1) * c
        if m > 0:
            # If Impound Strike Release Marker is drawn
            new_v = min(m, v)
            distributed = new_v / (2 if c > 0 else 1)
            exp += calc_exp(c + distributed * (c > 0), max(m - new_v, 0), p + distributed, v, depth + 1) * m
        return exp

    return calc_exp(c, m, p, v, 1)

t = int(input().strip())
for _ in range(t):
    c, m, p, v = map(float, input().strip().split())
    print(f"{expected_races(c, m, p, v):.12f}")
