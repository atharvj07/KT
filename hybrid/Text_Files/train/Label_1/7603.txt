#include <bits/stdc++.h>
using namespace std;

#define pb push_back
#define mp make_pair

typedef long long ll;
typedef vector<int> vi;
typedef vector<ll> vll;
typedef pair<int,int> ii;

const int mod = 1e9 + 7;
const double PI = acos(-1.0);

#define FOR(i,a,b) for(int i=a;i<b;++i)
#define REP(i,n) FOR(i,0,n)

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int n;
	cin >> n;
	vll a(n);
	REP(i,n) {
		cin >> a[i];
	}
	vll dp(n,0);
	vll sum(n,0);
	sum[n-1] = a[n-1];
	for(int i=n-2; i>=0; i--)
		sum[i] = sum[i+1] + a[i];
	dp[n-1] = a[n-1];
	for(int i=n-2; i>=0; i--) {
		for(int j=2; j<=min(i+1,n-i); j++) {
			dp[i] = max(dp[i],dp[i+j] + sum[i]);
		}
	}
	cout << dp[0] << endl;
    return 0;
}
