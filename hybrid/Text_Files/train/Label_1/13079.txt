#include <bits/stdc++.h>
using namespace std;
int n, hamm;
string s, t;
int main() {
  cin >> n;
  cin >> s >> t;
  for (int i = 0; i < n; i++) { hamm += (s[i] != t[i]); }
  swap(s[0], s[1]);
  int cha = (s[0] != t[0]) + (s[1] != t[1]);
  if (hamm > cha) {
    hamm = cha;
    s[0] = s[0] + s[1];
    s[1] = s[0] - s[1];
    s[0] = s[0] - s[1];
    cout << hamm << endl;
    cout << "1 2" << endl;
    return 0;
  }
  for (int i = 0; i < n - 1; i++) {
    swap(s[i], s[i + 1]);
    int cha = 0;
    for (int j = 0; j < n; j++) { cha += (s[j] != t[j]); }
    if (hamm > cha) {
      hamm = cha;
      cout << hamm << endl;
      cout << i + 1 << " " << i + 2 << endl;
      return 0;
    }
    swap(s[i], s[i + 1]);
  }
  cout << hamm << endl;
  cout << "-1 -1" << endl;
}
