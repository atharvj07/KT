from collections import defaultdict

def find_two_arrays(n, m, comparisons):
    adj_list = defaultdict(list)
    for a, b in comparisons:
        adj_list[a].append(b)
        adj_list[b].append(a)
    
    distinct_array = [-1] * (n + 1)
    equal_array = [-1] * (n + 1)
    current_number = 1
    
    for i in range(1, n + 1):
        if distinct_array[i] == -1:
            distinct_array[i] = current_number
            equal_array[i] = current_number
            queue = [i]
            while queue:
                node = queue.pop(0)
                for neighbor in adj_list[node]:
                    if distinct_array[neighbor] == -1:
                        distinct_array[neighbor] = current_number + 1
                        equal_array[neighbor] = current_number
                        queue.append(neighbor)
                current_number += 1
    
    for a, b in comparisons:
        if distinct_array[a] == distinct_array[b]:
            return "NO"
    
    return ["YES", distinct_array[1:], equal_array[1:]]

n, m = map(int, input().split())
comparisons = [list(map(int, input().split())) for _ in range(m)]

result = find_two_arrays(n, m, comparisons)
if result == "NO":
    print("NO")
else:
    print(result[0])
    print(*result[1])
    print(*result[2])
