#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> showcase;
  int cur = 0;
  bool consistent = true;
  for (int i = 0; i < 2 * n; ++i) {
    char op;
    std::cin >> op;
    if (op == '+') {
      ++cur;
      showcase.push_back(0);
    } else {
      int x;
      std::cin >> x;
      if (x <= cur) { showcase[cur - x] = x; } else { consistent = false; }
    }
  }
  if (consistent) {
    std::cout << "YES\n";
    for (int x : showcase) std::cout << x << ' ';
    std::cout << '\n';
  } else {
    std::cout << "NO\n";
  }
  return 0;
}
