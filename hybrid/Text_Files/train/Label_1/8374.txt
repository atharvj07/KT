#include <bits/stdc++.h>
using namespace std;

const int mxN = 3000;

struct Edge {
    int u, v, w;
    Edge(int u, int v, int w) : u(u), v(v), w(w) {}
};

int n, m, s, b, k, h, dist[3][mxN], par[mxN], id[mxN], base[mxN], ship[mxN];

vector<Edge> adj[mxN];

void dijkstra(int src, int p) {
    for (int i = 0; i <= n; i++) { dist[p][i] = LLONG_MAX; }
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    dist[p][src] = 0;
    pq.push({0, src});

    while (!pq.empty()) {
        int u = pq.top().second;
        pq.pop();
        if (dist[p][u] < pq.top().first) continue;

        for (auto e : adj[u]) {
            int v = e.v, w = e.w;
            if (dist[p][u] + w < dist[p][v]) {
                dist[p][v] = dist[p][u] + w;
                par[v] = u;
                pq.push({dist[p][v], v});
            }
        }
    }
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(Edge(u, v, 1));
        adj[v].push_back(Edge(v, u, 1));
    }

    cin >> s >> b >> k >> h;
    for (int i = s + 1; i <= s + b; i++) { base[i] = 1; }

    for (int i = 1; i <= s; i++) {
        int a, f, x;
        cin >> x >> a >> f;
        ship[x] = i;
    }

    for (int i = 1; i <= b; i++) {
        int x, d;
        cin >> x >> d;
        id[x] = s + i;
    }

    dijkstra(1, 0);
    dijkstra(2, 1);
    dijkstra(3, 2);

    int ans = 0;

    for (int i = 1; i <= s; i++) {
        if (ship[i] == 0) continue;

        for (int j = 1; j <= n; j++) {
            if (base[id[j]]) {
                if (dist[0][ship[i]] >= dist[1][j] && dist[0][ship[i]] <= dist[2][j]) {
                    ans += k;
                    base[id[j]] = 0;
                    break;
                }
            }
        }
    }

    for (int i = 1; i <= n; i++) {
        if (base[id[i]]) ans += h;
    }

    cout << ans << endl;
    return 0;
}
