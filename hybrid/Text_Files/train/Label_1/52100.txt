import java.util.*;

public class Main {
  static int N;
  static int[][][] A;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    N = sc.nextInt();
    A = new int[N][N][N];

    int[] Dx = {0, 1, 0, -1};
    int[] Dy = {1, 0, -1, 0};
    int x = (N - 1) / 2;
    int y = 0;
    int d = 0;
    int num = 1;
    A[x][y][0] = num++;
    while (num <= N * N * N) {
      int nx = x + Dx[d];
      int ny = y + Dy[d];
      if (0 <= nx && nx < N && 0 <= ny && ny < N && A[nx][ny][0] == 0) {
        x = nx;
        y = ny;
        A[x][y][0] = num++;
      } else {
        d = (d + 1) % 4;
      }
    }

    for (int z = 0; z < N; z++) {
      for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
          for (int k = 0; k < N - z; k++) {
            A[i][j][z + k] = A[i][j][z] + k * N * N;
          }
        }
      }
    }

    for (int z = 0; z < N; z++) {
      for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
          System.out.print(A[i][j][z] + " ");
        }
        System.out.println();
      }
      System.out.println();
    }
  }
}
