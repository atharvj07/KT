for _ in range(int(input())):
    s = input()
    odd = '0' in s
    for i in range(len(s)-1):
        if s[i] == s[i+1]: continue
        if (s[i] == '8' and s[i+1] != '6') or (s[i] != '4' and s[i+1] == '2'):
            odd = 0
            break
    if s[-1] == '1' or s[-1] == '3' or s[-1] == '5' or s[-1] == '7' or s[-1] == '9': odd = 1
    if not odd: print(0)
    elif odd: print(1)
    else: print(-1)
