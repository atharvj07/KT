import sys
def is_prime(n):
    """
    Check if a number is prime.

    Args:
        n: The number to check.

    Returns:
        True if n is prime, False otherwise.
    """
    if n == 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

inputs = sys.stdin.readlines()
t = int(inputs[0])
for i in range(1, t + 1):
    n, e = map(int, inputs[i].split())
    a = list(map(int, inputs[i + 1].split()))
    count = 0
    for j in range(0, n - e*1 + 1):
        product = 1
        for k in range(0, e*1):
            product *= a[j + k]
        if is_prime(product):
            count += 1
    print(count)
