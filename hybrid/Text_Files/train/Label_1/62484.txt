import sys

n = int(sys.stdin.readline())
l = list(map(int, sys.stdin.readline().split()))
d = list(map(int, sys.stdin.readline().split()))
l.sort()
d.sort()
ans = 0
cnt = 0
for i in range(n - 1, -1, -1):
    if l[i] != l[n - 1]:
        ans += d[i]
        cnt += 1
    if cnt > n // 2:
        break
print(ans)
