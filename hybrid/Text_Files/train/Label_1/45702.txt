import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int d = scanner.nextInt();
            int a = d + 4;
            if (d % 2 == 0) {
                a = d + 2;
            }
            if (d == 1) {
                a = 6;
            }
            System.out.println(a);
        }
    }
}
