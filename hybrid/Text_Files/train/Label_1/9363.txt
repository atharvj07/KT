#include <iostream>
#include <unordered_set>
int main() {
  std::string s, t;
  std::cin >> s >> t;
  std::unordered_set<char> sett(s.begin(), s.end());
  for (auto &j : t) sett.erase(j);
  std::string a;
  std::cin >> a;
  for (auto &i : a) {
    if (sett.count(s[i - 1])) {
      sett.erase(s[i - 1]);
    } else
      break;
  }
  std::cout << sett.size() << '\n';
}
