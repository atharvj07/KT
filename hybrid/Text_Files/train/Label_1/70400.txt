def solve_circle_problem(n):
    if n % 2 == 1:
        print("YES")
        ans = [0] * (2 * n)
        for i in range(1, n + 1):
            if i % 2 == 1:
                ans[i - 1] = 2 * i - 1
                ans[i + n - 1] = 2 * i
            else:
                ans[i - 1] = 2 * i
                ans[i + n - 1] = 2 * i - 1
        print(' '.join(map(str, ans)))
    else:
        print("NO")

n = int(input())
solve_circle_problem(n)
