t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    freq = {}
    for num in a:
        if num in freq:
            freq[num] += 1
        else:
            freq[num] = 1
    count_freq = {}
    for value in freq.values():
        if value in count_freq:
            count_freq[value] += 1
        else:
            count_freq[value] = 1
    min_remove = n
    total_elements = sum(freq.values())
    for c, times in count_freq.items():
        to_remove = total_elements - c * times
        min_remove = min(min_remove, to_remove)
    print(min_remove)
