n,x,k=map(int,input().split())
a=list(map(int,input().split()))
ans=0
a.sort()
for i,j in enumerate(a):
    l=i
    r=n-1
    pos=-1
    while l<=r:
        m=(l+r)//2
        if a[m]>=j+x:
            pos=m
            r=m-1
        else: l=m+1
    if pos==-1: continue
    ans+=(pos-i)*(pos-i+1)//2
    if pos-i>k: ans-=pos-i-k
print(ans)
