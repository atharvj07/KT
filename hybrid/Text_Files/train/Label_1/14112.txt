#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, y;
        cin >> n >> x >> y;
        vector<int> ans;
        int dif = (y - x) / (n - 1);
        int val = x;
        for (int i = 0; i < n; i++) {
            ans.push_back(val);
            val += dif;
        }
        for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
        cout << endl;
    }
    return 0;
}
