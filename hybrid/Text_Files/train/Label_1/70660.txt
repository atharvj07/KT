from sys import stdin

def fastpow(a, b):
    ans = 1
    while b >= 1:
        if b % 2 == 0:
            a *= a
            a %= mod
            b /= 2
        else:
            ans *= a;
            ans %= mod
            b -= 1
    return ans

h = list(map(int, stdin.readline().split()))
k = h[1]
h = h[2:]
l = len(h)
lastans = []
ans = 0
mod = 998244353
for i in range(l):
    curans = 0
    for j in range(l):
        if h[(j - i + l) % l] == h[j]:
            curans += 1
    lst = curans
    if lastans:
        if lastans > curans:
            lastans = curans
            ans -= fastpow(k, l)
            ans %= mod
    else:
        lastans = curans
    ans += (fastpow(k, l) - fastpow(k, l - curans))
    ans %= mod
    lastans = curans
print(ans)
