import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int[][] grid = new int[n][m];

            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    grid[i][j] = scanner.nextInt();
                }
            }

            boolean ashishWins = canAshishWin(grid, n, m);
            System.out.println(ashishWins ? "Ashish" : "Vivek");
        }
    }

    public static boolean canAshishWin(int[][] grid, int n, int m) {
        int[][] dp = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 0) {
                    if (i == 0 || j == 0) {
                        dp[i][j] = 1;
                    } else if (dp[i - 1][j] == 1 || dp[i][j - 1] == 1) {
                        dp[i][j] = 1;
                    }
                }
            }
        }

        return dp[n - 1][m - 1] == 1;
    }
}
