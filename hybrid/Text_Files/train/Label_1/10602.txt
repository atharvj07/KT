#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, t;
    cin >> t;
    while(t--) {
        cin >> n;
        long long ans = 0;
        for(int i=1; i*i<=n; i++) {
            for(int j=i; i*i+j*j<=n; j++) {
                int c = i*i - j;
                if(c>1 && c*c+j*j==i*i) ans++;
            }
        }
        cout << ans << endl;
    }
}
