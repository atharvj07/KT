import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line;
        while (!(line = br.readLine()).equals("0")) {
            int N = Integer.parseInt(line);
            int[] prizes = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int minChallenges = Integer.MAX_VALUE;
            for (int i = 0; i < N; i++) {
                if (prizes[i] == 0) continue;
                for (int j = i + 1; j < N; j++) {
                    if (prizes[j] == 0) continue;
                    minChallenges = Math.min(minChallenges, prizes[i] + prizes[j]);
                }
            }
            if (minChallenges == Integer.MAX_VALUE) {
                System.out.println("NA");
            } else {
                System.out.println(minChallenges);
            }
        }
    }
}
