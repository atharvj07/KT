import java.util.Scanner;

public class Main {
    public static Scanner stdin = new Scanner(System.in);
    public static int[][] net = new int[5][5];
    public static boolean opposite(int a, int b) {
        return a != b && (a + b == 7);
    }
    public static boolean proper() {
        boolean ok = true;
        for (int i = 0; i < 5; ++i)
            for (int j = 0; j < 5; ++j) {
                if (net[i][j] <= 0 || net[i][j] > 6)
                    return false;
                if (j > 0 && net[i][j] == net[i][j - 1]
                    || (j < 4 && net[i][j] == net[i][j + 1])
                    || (i > 0 && net[i][j] == net[i - 1][j])
                    || (i < 4 && net[i][j] == net[i + 1][j]))
                    return false;
                if (i == 0 || i == 4 || j == 0 || j == 4)
                    continue;
                if (!opposite(net[i][j], net[i][4 - j]))
                    ok = false;
            }
        return ok;
    }
    public static void main(String[] args) {
        int N = stdin.nextInt();
        for (int i = 0; i < N; ++i) {
            for (int j = 0; j < 5; ++j) {
                for (int k = 0; k < 5; ++k) {
                    net[j][k] = stdin.nextInt();
                }
            }
            System.out.println(proper() ? "true" : "false");
        }
    }
}
