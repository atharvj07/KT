#include<bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<long long>v(n), c(n);
        for (int i = 0; i < n; i++)cin >> v[i];
        long long ans = 0;
        for (int i = 1; i < n; i++) {
            if (v[i] < v[i - 1]) {
                ans += (v[i - 1] - v[i] + 1) / 2;
                c[i] = ans;
            }
            else if (c[i - 1]) {
                ans += (v[i] - v[i - 1] + c[i - 1] + 1) / 2;
                c[i] = ans;
            }
        }
        cout << ans << endl;
    }
}
