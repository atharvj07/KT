#include <bits/stdc++.h>
using namespace std;
#define INF (1<<30)
int main() {
    long long n, m, a;
    cin >> n >> m >> a;
    vector<long long> b(n), p(m);
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < m; i++) cin >> p[i];
    sort(b.begin(), b.end());
    sort(p.begin(), p.end());
    long long r = 0, s = 0;
    for (int i = 0, j = 0; i < n; i++) {
        while (j < m && p[j] <= a) a -= p[j++], r++;
        if (j >= m) break;
        if (a + b[i] >= p[j]) {
            a += b[i] - (p[j] - a);
            j++, r++;
        }
    }
    for (int i = 0; i < r; i++) s += p[i];
    cout << r << " " << s << endl;
}
