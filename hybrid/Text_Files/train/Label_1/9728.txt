#include<iostream>
#include<vector>
#include<queue>
using namespace std;
int main(){
    string s;
    cin>>s;
    int l[26]={0};
    for(int i=0;i<s.length();i++){
        l[s[i]-65]++;
    }
    if(l[25]==0 || l[24]==0){cout<<"Impossible"; return 0;}
    char arr[2][13]={ 
        {'Y','X','W','V','U','T','S','R','Q','P','O','N','M'},
        {'Z','A','D','B','C','F','G','H','J','K','L','I','E'} 
    };
    for(int i=0,j=0;i<26;i++){
        while(l[i]>0){
            if(arr[j][0]==s[i]){arr[j][0]=0; l[i]--;}
            else if(arr[j][12]==s[i]){arr[j][12]=0; l[i]--;}
            else if(arr[j][1]==s[i] && arr[j][2]==0){arr[j][1]=0;  arr[j][2]=s[i]; l[i]--;}
            else if(arr[j][11]==s[i] && arr[j][10]==0){arr[j][11]=0; arr[j][10]=s[i]; l[i]--;}
            else{cout<<"Impossible"; return 0;}
            j=(j+1)%2;
        }
    }
    for(int i=0;i<2;i++){
        for(int j=0;j<13;j++){
            if(arr[i][j]==0){cout<<' '; continue;}
            cout<<arr[i][j];
        }
        cout<<endl;
    }
}
