import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        int[][] cnt = new int[q][10];
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            for (int j = l; j <= r; j++) {
                char ch = s.charAt(j);
                if (ch >= '0' && ch <= '9') {
                    cnt[i][ch - '0']++;
                    s = s.substring(0, j) + (ch - 1) + s.substring(j + 1);
                }
            }
        }
        for (int[] x : cnt) {
            for (int y : x) {
                System.out.print(y + " ");
            }
            System.out.println();
        }
    }
}
