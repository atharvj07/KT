import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Holiday[] h = new Holiday[n];
        for (int i = 0; i < n; i++) {
            h[i] = new Holiday(sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        int min = Integer.MAX_VALUE;
        for (int i = 1; i <= 360; i++) {
            int tmp = 0;
            for (Holiday h1 : h) {
                tmp = Math.max(tmp, h1.calc(i));
            }
            min = Math.min(min, tmp);
        }
        System.out.println(min);
    }
}

class Holiday {
    int m, d, v, s;
    Holiday(int m, int d, int v, int s) {
        this.m = m;
        this.d = d;
        this.v = v;
        this.s = s;
    }
    int calc(int day) {
        int m = this.m, d = this.d, v = this.v, s = this.s;
        int day0 = (m - 1) * 30 + d;
        int day1 = day0 + v - 1;
        int dist = Math.abs(day - day0) + Math.abs(day - day1);
        return Math.max(0, s - Math.min(dist, s));
    }
}
