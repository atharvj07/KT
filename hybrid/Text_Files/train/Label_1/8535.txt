#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;cin>>n>>k;
    vector<int> perm(n);
    for(int i=0;i<n;i++)perm[i]=i+1;
    for(int i=n-2;i>=0;i--){
        if(k==0)break;
        if(perm[i]>perm[i+1]){
            swap(perm[i],perm[i+1]);
            k--;
        }
    }
    for(int x:perm)cout<<x<<" ";
    return 0;
}
