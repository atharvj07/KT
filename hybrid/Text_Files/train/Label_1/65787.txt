import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
adj = [set() for i in range(n)]
for i in range(m):
    x, y, w = map(int, input().split())
    x -= 1; y -= 1
    if w < a[x] + a[y]:
        adj[x].add(y)
        adj[y].add(x)

vis = [False for i in range(n)]
def dfs(u):
    vis[u] = True
    for v in adj[u]:
        if not vis[v]:
            dfs(v)

dfs(0)
if all(vis):
    print(0)
else:
    mn1 = 10**18
    for i in range(n):
        if not vis[i]:
            mn2 = 10**18
            for j in range(n):
                if vis[j]:
                    mn2 = min(mn2, a[i] + a[j])
            mn1 = min(mn1, mn2)
    print(mn1)
