#include <bits/stdc++.h>
using namespace std;
pair<int, int> a[200005], b[200005];
int main() {
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i].first; }
  cin >> m;
  for (int i = 1; i <= m; i++) {
    cin >> b[i].first >> b[i].second;
    if (b[i].first > b[i].second) { swap(b[i].first, b[i].second); }
    a[n + i] = b[i];
  }
  sort(a + 1, a + n + m + 1);
  int ans = n, cnt = 0;
  for (int i = 2; i <= n + m; i++) {
    if (a[i] == a[i - 1]) {
      cnt++;
      if (cnt > ans + 1) { ans = i - 1; }
    } else {
      cnt = 0;
    }
  }
  if (ans <= n) return cout << ans, 0;
  return cout << ans - n, 0;
}
