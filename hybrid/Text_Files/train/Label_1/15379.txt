#include <bits/stdc++.h>
using namespace std;

int32_t main() {
    int t; cin >> t; while(t--) {
        int n, k; cin >> n >> k;
        vector<int> a(n), b(n);
        for(auto &it : a) cin >> it;
        for(auto &it : b) cin >> it;
        sort(a.begin(), a.end());
        sort(b.rbegin(), b.rend());
        for(int i = 0; i < min(n, k); i++) {
            if(a[i] < b[i]) swap(a[i], b[i]);
            else break;
        }
        int ans = 0;
        for(auto &it : a) ans += it;
        cout << ans << endl;
    }
    return 0;
}
