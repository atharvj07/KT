#include <iostream>
const int MOD = 1000000007;
int main() {
  int a, b, k, t;
  std::cin >> a >> b >> k >> t;
  int dp[102][102][202];
  dp[a][b][k] = 1;
  for (int turn = 1; turn <= t; turn++) {
    for (int i = 1; i <= 100; i++) {
      for (int j = 1; j <= 100; j++) {
        for (int score = 0; score <= 200; score++) {
          if (i > j) {
            dp[i][j][score] = (dp[i][j][score] + dp[i - 1][j][score - 1] + dp[i - 1][j][score] +
                                 dp[i - 1][j][score + 1]) %
                          MOD;
          } else {
            dp[i][j][score] = (dp[i][j][score] + dp[i][j - 1][score - 1] + dp[i][j - 1][score] +
                                 dp[i][j - 1][score + 1]) %
                          MOD;
          }
        }
      }
    }
  }
  int ans = 0;
  for (int i = 1; i <= 100; i++) {
    for (int j = 1; j <= 100; j++) {
      for (int score = 0; score <= 200; score++) {
        if (i > j && score >= 1) ans = (ans + dp[i][j][score]) % MOD;
      }
    }
  }
  std::cout << ans << "\n";
  return 0;
}
