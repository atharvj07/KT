#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
int32_t main() {
    int n, k;
    cin >> n >> k;
    ll l[n + 5], r[n + 5];
    for (int i = 1; i <= n; i++) cin >> l[i] >> r[i];
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (l[j] >= l[i] && r[j] <= r[i]) { l[j] = l[i], r[j] = r[i]; }
            if (l[i] >= l[j] && r[i] <= r[j]) { l[i] = l[j], r[i] = r[j]; }
        }
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) ans = (ans + (r[i] - l[i] + 1) % mod) % mod;
    cout << ans << endl;
    return 0;
}
