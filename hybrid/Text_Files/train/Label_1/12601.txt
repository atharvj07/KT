#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    int n;
    cin >> n;

    string s;
    cin >> s;

    int cntT = 0, cntM = 0;
    for (char c : s) {
      if (c == 'T') ++cntT;
      else if (c == 'M') ++cntM;
    }

    if (cntT % 3 || cntM % 3) cout << "NO" << endl;
    else cout << "YES" << endl;
  }

  return 0;
}
