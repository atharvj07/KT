#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  int bal = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(')
      bal++;
    else {
      if (bal == 0) {
        ans = -1;
        break;
      }
      bal--;
    }
  }
  if (bal != 0) ans = -1;
  cout << ans << endl;
  return 0;
}
