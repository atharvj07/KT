#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> c(n);
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  sort(c.begin(), c.end());
  long long ans = 0;
  for (int i = 0; i < n; i++) { ans += 1LL * (i + 1) * c[i]; }
  vector<int> t(n);
  for (int i = 0; i < n; i++) { t[i] = k + i + 1; }
  cout << ans << endl;
  for (int i = 0; i < n; i++) { cout << t[i] << " "; }
  cout << endl;
  return 0;
}
