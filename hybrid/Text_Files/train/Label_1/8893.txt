#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> p(n), a(n), ans;
  for (int i = 0; i < k; i++) {
    std::cin >> p[i];
    --p[i];
    a[p[i]] = 1;
  }
  for (int i = 0; i < n; i++) if (!a[i]) ans.push_back(i);
  sort(ans.begin(), ans.end());
  for (int i = 0; i < k; i++) ans.insert(ans.begin() + i, p[i]);
  for (int i = 0; i < n; i++) ++ans[i];
  if (std::is_sorted(ans.begin(), ans.end())) {
    for (int x : ans) std::cout << x << ' ';
  } else {
    std::cout << "-1";
  }
  return 0;
}
