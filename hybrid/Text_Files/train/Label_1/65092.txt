n, m = map(int, input().split())

# Initialize the array with very small values, assuming no operation has been performed
array = [-10**9] * n
# To keep track of the maximum value in the range for "type 2" operations
max_in_range = [-10**9] * n

for _ in range(m):
    t, l, r, x = map(int, input().split())
    l -= 1  # Adjusting index to fit Python's 0-based indexing
    r -= 1
    if t == 1:
        # Increase the range by x
        for i in range(l, r + 1):
            array[i] += x
    else:
        # Set the maximum for this range, if it is larger than the current
        for i in range(l, r + 1):
            max_in_range[i] = max(max_in_range[i], x)

# Reconstruct the array based on the maximum values found
# Start with the minimum possible value that ensures all elements are <= 10^9
reconstructed_array = [10**9] * n

for _ in range(m):
    t, l, r, x = map(int, input().split())
    l -= 1  # Adjusting index to fit Python's 0-based indexing
    r -= 1
    if t == 2:
        # Ensure that the reconstructed array meets the maximum requirements
        for i in range(l, r + 1):
            reconstructed_array[i] = min(reconstructed_array[i], x)

# Apply the increments in reverse to ensure the solution meets the "type 1" operations
for _ in range(m):
    t, l, r, x = map(int, input().split())
    l -= 1  # Adjusting index to fit Python's 0-based indexing
    r -= 1
    if t == 1:
        # Decrease the range by x to counter the initial increment
        for i in range(l, r + 1):
            reconstructed_array[i] -= x

# Check if the reconstructed array meets all the requirements
is_valid = True
for i in range(n):
    if reconstructed_array[i] < -10**9 or reconstructed_array[i] > 10**9:
        is_valid = False
        break
    if max_in_range[i] != -10**9 and reconstructed_array[i] != max_in_range[i]:
        is_valid = False
        break

if is_valid:
    print("YES")
    print(' '.join(map(str, reconstructed_array)))
else:
    print("NO")
