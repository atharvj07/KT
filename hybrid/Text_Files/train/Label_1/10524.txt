#include <cstdio>
#include <algorithm>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    long long x, y, p, q;
    scanf("%lld%lld%lld%lld", &x, &y, &p, &q);
    if (p * y % q == x) puts("0");
    else if (p * y < x * q) puts("-1");
    else printf("%lld\n", (x * q - p * y + q - 1) / q);
  }
  return 0;
}
