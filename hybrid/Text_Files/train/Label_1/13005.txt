#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int k;
    cin >> k;
    vector<vector<int>> grid(2, vector<int>(2 * k + 1));
    for (int i = 0; i < 2; i++) {
      for (int j = 0; j < 2 * k + 1; j++) {
        string s;
        cin >> s;
        if (s == "E") {
          grid[i][j] = -1;
        } else {
          grid[i][j] = stoi(s);
        }
      }
    }
    vector<vector<int>> target(2, vector<int>(2 * k + 1));
    for (int i = 0; i < 2 * k + 1; i++) {
      target[0][i] = i + 1;
    }
    for (int i = 0; i < 2 * k; i++) {
      target[1][i] = i + 2 * k + 2;
    }
    target[1][2 * k] = -1;
    int empty_x = -1, empty_y = -1;
    for (int i = 0; i < 2; i++) {
      for (int j = 0; j < 2 * k + 1; j++) {
        if (grid[i][j] == -1) {
          empty_x = j;
          empty_y = i;
          break;
        }
      }
      if (empty_x != -1) break;
    }
    vector<pair<int, int>> moves;
    bool failed = false;
    for (int i = 0; i < 2; i++) {
      for (int j = 0; j < 2 * k + 1; j++) {
        if (grid[i][j] == target[i][j]) continue;
        bool found = false;
        for (int dx = -1; dx <= 1; dx++) {
          for (int dy = -1; dy <= 1; dy++) {
            if (abs(dx) + abs(dy) == 1) {
              int nx = j + dx;
              int ny = i + dy;
              if (nx >= 0 && nx < 2 * k + 1 && ny >= 0 && ny < 2) {
                if (grid[ny][nx] == target[i][j]) {
                  if (empty_x == j && empty_y == i) {
                    swap(grid[i][j], grid[ny][nx]);
                    moves.push_back(make_pair(dx, dy));
                    found = true;
                    break;
                  } else if (ny == empty_y && nx == empty_x) {
                    swap(grid[i][j], grid[ny][nx]);
                    moves.push_back(make_pair(-dx, -dy));
                    found = true;
                    break;
                  }
                }
              }
            }
          }
          if (found) break;
        }
        if (!found) {
          failed = true;
          break;
        }
      }
      if (failed) break;
    }
    if (failed) {
      cout << "SURGERY FAILED" << endl;
    } else {
      cout << "SURGERY COMPLETE" << endl;
      unordered_map<string, string> shortcuts;
      string s;
      for (int i = 0; i < moves.size(); i++) {
        if (i != 0) s += " ";
        if (moves[i].first == -1) {
          s += "l";
        } else if (moves[i].first == 1) {
          s += "r";
        } else if (moves[i].second == -1) {
          s += "u";
        } else if (moves[i].second == 1) {
          s += "d";
        }
      }
      int n = s.size();
      int l = 0;
      while (l < n) {
        int r = l + 1;
        while (r < n && s[r] == s[l]) r++;
        if (r - l > 2) {
          char ch = s[l];
          string sub = s.substr(l, r - l);
          if (shortcuts.count(sub)) {
            cout << shortcuts[sub] << " ";
          } else {
            string key = "";
            key += ch;
            shortcuts[sub] = key;
            cout << key << " " << sub << endl;
          }
        } else {
          cout << s[l];
        }
        l = r;
      }
      cout << endl << "DONE" << endl;
    }
  }
  return 0;
}
