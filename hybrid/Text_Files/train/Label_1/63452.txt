import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
p = [0] * n
t = [[]] * n
for i in range(1, n):
    x, y = map(int, input().split())
    p[i] = x - 1
    t[x - 1].append(i)
q = [i for i in range(n) if len(t[i]) > 1 or i == 0]
v = [0] * n
while q:
    x = q.pop()
    v[x] = 1
    for y in t[x]:
        if v[y] == 0:
            q.append(y)
rem = []
for i in range(n):
    if len(t[i]) == 0 and v[i] == 1:
        rem.append(i)
print(len(rem))
