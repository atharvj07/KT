n, m = map(int, input().split())
a = []
b = []
for i in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
ans = []
for i in range(n):
    if a[i] > m:
        ans.append(-1)
    elif a[i] - b[i] > m:
        ans.append(-1)
    elif a[i] <= m:
        ans.append(0)
    else:
        ans.append(1)
print(min(ans))
