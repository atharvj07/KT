import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong(), v = sc.nextLong();
        long b = sc.nextLong(), w = sc.nextLong();
        long t = sc.nextLong();
        sc.close();
        long x = v - w;
        if (x == 0) System.out.println("NO");
        else if (a > b && x > 0) System.out.println("NO");
        else if (a < b && x < 0) System.out.println("NO");
        else if (Math.abs(a - b) % Math.abs(x) <= t) System.out.println("YES");
        else System.out.println("NO");
    }
}
