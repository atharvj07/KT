#include <bits/stdc++.h>
#define ll long long
#define M 200005
using namespace std;

int n;
ll a[M];
ll gcd(ll A, ll B) {
    if (B == 0) return A;
    return gcd(B, A % B);
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a, a + n);
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        if (i == 0)
            ans = a[i];
        else
            ans = gcd(ans, a[i]);
    }
    cout << ans << '\n';
    return 0;
}
