import java.util.*;

class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        ArrayList<Integer>[] arrays = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            arrays[i] = new ArrayList<>();
        }
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            int t = sc.nextInt();
            if (type == 0) {
                int x = sc.nextInt();
                arrays[t].add(x);
            } else if (type == 1) {
                for (int j : arrays[t]) {
                    System.out.print(j + " ");
                }
                System.out.println();
            } else if (type == 2) {
                arrays[t].clear();
            }
        }
        sc.close();
    }
}
