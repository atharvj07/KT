import sys
import heapq

n, m = map(int, input().split())
graph = [[] for i in range(n+1)]
for i in range(n-1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)
cities = list(map(int, input().split()))

dist = [-1]*(n+1)
vis = [False]*(n+1)
pq = [(0, 1)]
while pq:
    dis, node = heapq.heappop(pq)
    if vis[node]:
        continue
    vis[node] = True
    dist[node] = dis
    for u in graph[node]:
        if not vis[u]:
            heapq.heappush(pq, (dis+1, u))

ans = 0
for city in cities:
    ans = max(ans, dist[city])

idx = -1
for i in range(1, n+1):
    if dist[i] == ans:
        idx = i
        break

print(idx)
print(ans)
