import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 1; i < n; i++) {
            p[i] = sc.nextInt() - 1;
        }
        int q = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                int v = sc.nextInt() - 1;
                long x = sc.nextLong();
                long k = sc.nextLong();
                a[v] += x;
                while (v != 0) {
                    v = p[v];
                    a[v] += (x - k);
                    k += k;
                }
            } else {
                int v = sc.nextInt() - 1;
                System.out.println(a[v] % 1000000007);
            }
        }
        sc.close();
    }
}
