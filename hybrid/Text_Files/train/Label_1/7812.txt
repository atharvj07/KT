#include <bits/stdc++.h>
using namespace std;
pair<int, int> all[200005];
int32_t main() {
  int n, k;
  cin >> n >> k;
  if ((n * 2) % k) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) { all[i * 2] = {i + 1, (i * 2) % k + 1}; }
  for (int i = 0; i < n; i++) { all[(i * 2) + 1] = {n + i + 1, (i * 2 + 1) % k + 1}; }
  random_shuffle(all, all + n * 2);
  for (int i = 0; i < n; i++) { cout << all[i * 2].second << " " << all[(i * 2) + 1].second << endl; }
  return 0;
}
