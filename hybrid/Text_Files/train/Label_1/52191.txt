import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int y = input.nextInt();
        int w = input.nextInt();
        if (y == w) {
            System.out.println("0/1");
        } else if (Math.min(y, w) == 1) {
            System.out.println("1/1");
        } else {
            System.out.println("1/2");
        }
    }
}
