import sys
import math
ans=1
N=int(input())
lst=[]
for i in range(N):
    a,p=sys.stdin.readline().split()
    p=int(p)
    if a=="ADD":
        lst.append(p)
    else:
        if len(lst)==0 or lst[-1]!=p:
            ans=0
            break
        lst.pop()
    ans=ans*2%1000000007
if len(lst): ans=0
print(ans)
