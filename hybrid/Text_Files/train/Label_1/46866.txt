import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] years = new int[2 * n];
        for (int i = 0; i < n; i++) {
            years[2 * i] = scanner.nextInt();
            years[2 * i + 1] = scanner.nextInt();
        }
        Arrays.sort(years);
        int alive = 0, maxAlive = 0, maxAliveYear = 0;
        for (int i = 0; i < 2 * n; i++) {
            if (i > 0 && years[i] == years[i - 1]) {
                continue;
            }
            if (i % 2 == 0) {
                alive++;
            } else {
                alive--;
            }
            if (alive > maxAlive) {
                maxAlive = alive;
                maxAliveYear = years[i];
            }
        }
        System.out.println(maxAliveYear + " " + maxAlive);
    }
}
