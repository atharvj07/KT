#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;

int n, a[N], b[N];
pair<int, int> c[N];
bool ok[N];

int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    if (a[i]) { c[a[i]] = {i, 1}; }
  }
  for (int i = 1; i <= n; ++i) {
    cin >> b[i];
    if (b[i]) { c[b[i]] = {i, 2}; }
  }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (!c[i].first) { continue; }
    if (c[i].second == 1) {
      if (!ok[i]) {
        ans++;
        ok[i] = true;
      }
      if (i != n) { ok[i + 1] = true; }
    } else {
      if (i < n && !ok[i]) { ans++; }
    }
  }
  cout << ans << endl;
  return 0;
}
