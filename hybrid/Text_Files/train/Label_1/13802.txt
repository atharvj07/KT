#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    sort(v.begin(), v.end());
    bool ok = true;
    for (int i = 1; i < n; i++) {
        int a = v[i].size(), b = v[i - 1].size();
        if (a < b) {
            ok = false;
            break;
        }
        bool found = false;
        for (int j = b; j > 0; j--) {
            if (v[i].substr(a - b, b) == v[i - 1].substr(b - j, j)) {
                found = true;
                break;
            }
        }
        if (!found) {
            ok = false;
            break;
        }
    }
    if (ok) {
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) { cout << v[i] << endl; }
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
