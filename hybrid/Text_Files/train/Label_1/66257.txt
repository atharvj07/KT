from collections import defaultdict

n, m = map(int, input().split())
p = list(map(int, input().split()))
friends = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    friends[u].append(v)
    friends[v].append(u)

pos = {x: i for i, x in enumerate(p)}
vis = [False] * (n + 1)
ans = 0

for i in p[1:]:
    if not vis[i]:
        vis[i] = True
        j = i
        while friends[i] and friends[i][0] == j:
            j = friends[i].pop(0)
            vis[j] = True
        if friends[i]:
            ans += 1

print(ans)
