import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] players = new int[4][2];
        for (int i = 0; i < 4; i++) {
            players[i][0] = sc.nextInt();
            players[i][1] = sc.nextInt();
        }
        int[] team1 = new int[2];
        int[] team2 = new int[2];
        if (players[0][0] > players[2][1] && players[0][1] > players[2][0]) {
            team1[0] = players[0][0];
            team1[1] = players[0][1];
        } else {
            team1[0] = players[1][0];
            team1[1] = players[1][1];
        }
        if (players[2][0] > players[0][1] && players[2][1] > players[0][0]) {
            team2[0] = players[2][0];
            team2[1] = players[2][1];
        } else {
            team2[0] = players[3][0];
            team2[1] = players[3][1];
        }
        if (team1[0] > team2[1] && team1[1] > team2[0]) {
            System.out.println("Team 1");
        } else if (team2[0] > team1[1] && team2[1] > team1[0]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
