#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
    ll t;
    cin >> t;

    while (t--) {
        string s;
        cin >> s;

        ll n = s.size();

        ll cnt[2][3];
        for (ll i = 0; i < 2; i++) {
            for (ll j = 0; j < 3; j++) {
                cnt[i][j] = 0;
            }
        }

        for (ll i = 0; i < n; i++) {
            if (s[i] == 'R') {
                cnt[0][0]++;
                cnt[1][2]++;
            } else if (s[i] == 'S') {
                cnt[0][1]++;
                cnt[1][0]++;
            } else if (s[i] == 'P') {
                cnt[0][2]++;
                cnt[1][1]++;
            }
        }

        string c;
        for (ll i = 0; i < n; i++) {
            if (cnt[0][0] >= cnt[0][1] && cnt[0][0] >= cnt[0][2]) {
                c += 'P';
            } else if (cnt[0][1] >= cnt[0][0] && cnt[0][1] >= cnt[0][2]) {
                c += 'S';
            } else {
                c += 'R';
            }
        }

        cout << c << '\n';
    }

    return 0;
}
