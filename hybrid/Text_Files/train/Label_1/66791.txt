from math import gcd
from functools import reduce

def find_divine_digits(n, k, banknotes):
    g = reduce(lambda x, y: gcd(x, y), banknotes)
    g = gcd(g, k)
    res = []
    for d in range(k):
        if d % g == 0:
            res.append(d)
    return len(res), res

n, k = map(int, input().split())
banknotes = list(map(int, input().split()))

count, digits = find_divine_digits(n, k, banknotes)
print(count)
print(*digits)
