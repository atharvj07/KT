import sys
for i in range(int(input())):
    n=int(input());t=list(map(int,input().split()))
    if n<3:print(-1);continue
    l=r=ans=0;f=1
    mp={}
    for i in range(n):
        if t[i] not in mp:mp[t[i]]=0
        mp[t[i]]+=1
        if mp[t[i]]>1:
            f=0
        while l<=r and mp[t[l]]!=max({*mp.values()})*f:
            mp[t[l]]-=1
            l+=1
        if r-l+1<ans or ans==0:ans=r-l+1
        r+=1
    if ans!=0:print(ans)
    else:print(-1)
