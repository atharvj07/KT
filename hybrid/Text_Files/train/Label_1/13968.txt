typedef long long int ll;
int n, t, ans[4097], L[2097], R[2097], col[2097], CL[2097], CR[2097];
bool ok;
void dfs(int x) {
  if (x >= n && !ok) {
    ok = true;
    for (int i = 0; i < n; i++) {
      if (ans[i] == ans[i + n]) printf("("); else printf(")");
    }
    printf("\n");
  } else {
    if (!col[x]) {
      col[x] = col[CL[x]] = col[CR[x]] = 1;
      ans[x] = 1;
      dfs(x + 1);
      ans[x] = 0;
      col[x] = col[CL[x]] = col[CR[x]] = 0;
    }
    if (!col[CL[x]]) {
      col[CL[x]] = col[x] = col[CR[CL[x]]] = 1;
      ans[x] = 2;
      dfs(x + 1);
      ans[x] = 0;
      col[CL[x]] = col[x] = col[CR[CL[x]]] = 0;
    }
    if (!col[CR[x]]) {
      col[CR[x]] = col[x] = col[CL[CR[x]]] = 1;
      ans[x] = 2;
      dfs(x + 1);
      ans[x] = 0;
      col[CR[x]] = col[x] = col[CL[CR[x]]] = 0;
    }
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d%d", &L[i], &R[i]);
    CR[L[i]] = i;
    CL[R[i]] = i;
  }
  dfs(0);
  if (!ok) printf("IMPOSSIBLE\n");
  return 0;
}
