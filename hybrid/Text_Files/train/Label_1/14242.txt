#include <bits/stdc++.h>

using namespace std;

const int MAXN = 3000 + 1;

int n, m;
int c[MAXN], party[MAXN], cnt[MAXN];

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> party[i] >> c[i];
    cnt[party[i]]++;
  }
  int mn = INT_MAX;
  for (int i = 2; i <= m; i++) {
    mn = min(mn, cnt[1] + cnt[i] - 1);
  }
  cout << *min_element(c + 1, c + n + 1) * mn << endl;
  return 0;
}
