#include <bits/stdc++.h>
using namespace std;
const int N = 1E5 + 5;
int a[N], pos[N], lst[N];
bool f[N], g[N];
vector<int>adj[N];
main() {
    int n, k; cin >> n >> k;
    if(k > n * (n - 1) / 2) { cout << -1 << endl; return 0; }
    int l = 2, r = n;
    for(int i = 1; i < n; i++) {
        if(k <= 0) {
            for(int i = 1; i <= n; i++) {
                if(!f[i]) { cout << i << ' '; f[i] = 1; }
            }
            return 0;
        }
        f[i] = f[n - i + 1] = 1;
        k -= n - i - 1;
        adj[i].emplace_back(n - i + 1);
        adj[n - i + 1].emplace_back(i);
    }
    vector<int>ans;
    int tot = 0;
    for(int i = 1; i <= n; i++) {
        if(f[i]) continue;
        ans.emplace_back(i);
        g[i] = 1;
        f[i] = 1;
        for(auto j : adj[i]) {
            if(g[j]) tot++;
        }
        if(tot >= k) {
            for(int j = 1; j <= n; j++) {
                if(f[j]) continue;
                ans.emplace_back(j);
            }
            for(auto j : ans) cout << j << ' ';
            return 0;
        }
        tot = 0;
    }
    return 0;
}
