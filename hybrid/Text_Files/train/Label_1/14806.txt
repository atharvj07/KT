#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n1,n2,n3,n4,m1,m2,m3;
    cin>>n1>>n2>>n3>>n4;
    vector<int>a(n1+5),b(n2+5),c(n3+5),d(n4+5);
    for(int i=1;i<=n1;i++) cin>>a[i];
    for(int i=1;i<=n2;i++) cin>>b[i];
    for(int i=1;i<=n3;i++) cin>>c[i];
    for(int i=1;i<=n4;i++) cin>>d[i];
    cin>>m1;
    vector<pair<int,int> >adj1[n1+3];
    for(int i=1;i<=m1;i++){
        int u,v;
        cin>>u>>v;
        adj1[u].push_back({v,i});
    }
    cin>>m2;
    vector<pair<int,int> >adj2[n2+3];
    for(int i=1;i<=m2;i++){
        int u,v;
        cin>>u>>v;
        adj2[u].push_back({v,i});
    }
    cin>>m3;
    vector<pair<int,int> >adj3[n3+3];
    for(int i=1;i<=m3;i++){
        int u,v;
        cin>>u>>v;
        adj3[u].push_back({v,i});
    }
    vector<int>col(2*(m1+m2+m3),0),vis(2*(m1+m2+m3),0);
    function<int(int)>dfs=[&](int u)->int{
        if(vis[u]) return 0;
        vis[u]=1;
        for(auto x:adj1[u]){
            if(!col[x.second]){
                col[x.second]=1;
                if(!dfs(x.first)){
                    col[x.second]=0;
                    return 0;
                }
            }
        }
        for(auto x:adj2[u]){
            if(!col[m1+x.second]){
                col[m1+x.second]=1;
                if(!dfs(x.first)){
                    col[m1+x.second]=0;
                    return 0;
                }
            }
        }
        for(auto x:adj3[u]){
            if(!col[m1+m2+x.second]){
                col[m1+m2+x.second]=1;
                if(!dfs(x.first)){
                    col[m1+m2+x.second]=0;
                    return 0;
                }
            }
        }
        return 1;
    };
    int ans=-1;
    for(int i=1;i<=n1;i++){
        for(int j=1;j<=n2;j++){
            for(int k=1;k<=n3;k++){
                for(int l=1;l<=n4;l++){
                    bool ok=1;
                    for(auto x:adj1[i]){
                        if(x.first==j) ok=0;
                    }
                    for(auto x:adj2[j]){
                        if(x.first==k) ok=0;
                    }
                    for(auto x:adj3[k]){
                        if(x.first==l) ok=0;
                    }
                    if(ok){
                        fill(col.begin(),col.end(),0);
                        fill(vis.begin(),vis.end(),0);
                        if(dfs(i)) ans=min(ans,a[i]+b[j]+c[k]+d[l]);
                    }
                }
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
