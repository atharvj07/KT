#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    long long a1, k;
    std::cin >> a1 >> k;
    for (int i = 0; i < k; ++i) {
      long long min_digit = a1 % 10;
      long long max_digit = a1 % 10;
      long long temp = a1;
      while (temp > 0) {
        long long digit = temp % 10;
        min_digit = std::min(min_digit, digit);
        max_digit = std::max(max_digit, digit);
        temp /= 10;
      }
      a1 += min_digit * max_digit;
    }
    std::cout << a1 << '\n';
  }
  return 0;
}
