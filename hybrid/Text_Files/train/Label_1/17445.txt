#include <iostream>
#include <cmath>
using namespace std;
int main() {
  int x, y, n;
  cin >> x >> y >> n;
  double ans = 1e9;
  while (n--) {
    int Xi, Yi, Vi;
    cin >> Xi >> Yi >> Vi;
    double dist = sqrt((x - Xi) * (x - Xi) + (y - Yi) * (y - Yi));
    ans = min(ans, dist / Vi);
  }
  cout << ans << "\n";
  return 0;
}
