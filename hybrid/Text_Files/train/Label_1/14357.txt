#include<bits/stdc++.h>
using namespace std;
typedef pair<long long,long long> ii;
const int maxn=2e5+10;
ii v[maxn];
int n,ans;
int can[maxn];
map<int,int> mp;
int main(){
    int t;
    scanf("%d",&t);
    while(t--){
        mp.clear();
        scanf("%d",&n);
        long long l,r;
        ans=0;
        for(int i=1;i<=n;i++){
            scanf("%lld%lld",&l,&r);
            if(l==r){
                if(!mp[l]) ans++;
                mp[l]=1;
                continue;
            }
            v[++ans]={l,r};
            can[ans]=1;
        }
        sort(v+1,v+1+ans);
        for(int i=1;i<=ans;i++){
            if(can[i]&&mp[v[i].first]==0&&mp[v[i].second]==0){
                can[i]=0;
                ans--;
            }
            mp[v[i].first]=1;
            mp[v[i].second]=1;
        }
        printf("%d\n",ans);
    }
    return 0;
}
