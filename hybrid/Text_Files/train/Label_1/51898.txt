import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int j = n - 1;
        for (int i = 0; i < n; i++) {
            while (j >= 0 && a[i] + a[j] > (int) Math.pow(2, ans + 1)) {
                j--;
            }
            if (j >= 0 && a[i] + a[j] == (int) Math.pow(2, ans + 1)) {
                ans++;
                j--;
            }
        }
        System.out.println(ans);
    }
}
