#include <bits/stdc++.h>

typedef long long ll;

const ll MOD = 1e9 + 7;

bool is_odd(ll a) {
    return a & 1;
}

int main() {
    ll n, x, y;
    std::cin >> n >> x >> y;
    std::string ans;
    if (x == n && y == n) {
        ans = (n % 2 == 1) ? "Black" : "White";
    } else if (x == n || y == n || x == 1 || y == 1) {
        ans = (is_odd(n + x + y)) ? "White" : "Black";
    } else {
        ans = "Black";
    }
    std::cout << ans << std::endl;
    return 0;
}
