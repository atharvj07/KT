#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  vector<int> ans1(n), ans2(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
  }
  for (int i = 0; i < n; i++) {
    int cnt = 0, cnt2 = 0;
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      if (v[i].first == v[j].first || v[i].second == v[j].first) cnt++;
      if (v[i].first == v[j].second || v[i].second == v[j].second) cnt2++;
    }
    ans1[i] = n - 1 - cnt;
    ans2[i] = n - 1 - cnt2;
  }
  for (int val : ans1) { cout << val << " "; }
  cout << endl;
  for (int val : ans2) { cout << val << " "; }
  return 0;
}
