from collections import defaultdict
h, w, g = map(int, input().split())
fl = list(map(int, input().split()))
fr = list(map(int, input().split()))
tv = [list(map(int, input().split())) for i in range(h)]
th = [[0]*w for i in range(h)]
for i in range(h):
    for j in range(w):
        if not tv[i][j]:
            continue
        th[i][j] = min([g-fl[j], g-fr[i]])
for j in range(w):
    mxh = 0
    for i in range(h):
        mxh = max(mxh, th[i][j])
    for i in range(h):
        th[i][j] = mxh-th[i][j]
for i in range(h):
    for j in range(w):
        print(th[i][j], end=' ')
    print()
