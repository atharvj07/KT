n = int(input())
s = input()
k = s.count('0')
if 2 * k != n:
    print(0)
    exit()
cnt = [0] * n
cnt[0] = (1 if s[0] == '0' else -1)
for i in range(1, n):
    cnt[i] = cnt[i - 1] + (1 if s[i] == '0' else -1)
mx = 0
ans = 0
for i in range(n):
    if cnt[i] == 0:
        ans = max(ans, i + 1)
    else:
        if -cnt[i] in cnt[:i]:
            ans = max(ans, i - cnt.index(-cnt[i]))
    mx = max(mx, cnt[i])
if mx != 0:
    ans = max(ans, n - cnt.index(mx))
print(ans)
