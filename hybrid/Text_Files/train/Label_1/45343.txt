import java.util.*;
import java.lang.*;
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int M = scanner.nextInt();
        int[] D = new int[N];
        for (int i = 0; i < N; i++) {
            D[i] = scanner.nextInt();
        }
        int[] C = new int[M];
        for (int j = 0; j < M; j++) {
            C[j] = scanner.nextInt();
        }
        int[][] dp = new int[N + 1][M + 1];
        for (int i = 0; i <= N; i++) {
            for (int j = 0; j <= M; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }
        dp[0][0] = 0;
        for (int j = 1; j <= M; j++) {
            for (int i = 0; i <= N; i++) {
                if (i > 0) {
                    dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - 1] + D[i - 1] * C[j - 1]);
                }
                dp[i][j] = Math.min(dp[i][j], dp[i][j - 1]);
            }
        }
        System.out.println(dp[N][M]);
    }
}
