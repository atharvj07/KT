import sys

mod = int(1e9) + 7

n, p = map(int, input().split())
c = list(map(int, input().split()))

dp = [[[-1] * 3 for _ in range(3)] for _ in range(n + 1)]
def solve(i, prv, l):
    if i == n:
        if l == p: return 1
        return 0
    if dp[i][prv][l] != -1: return dp[i][prv][l]

    ans = 0
    if c[i] == -1:
        for nx in range(3):
            if not (prv == nx and l == p):
                ans = (ans + solve(i + 1, nx, (l + nx) % 2)) % mod
    else:
        nx = c[i]
        if not (prv == nx and l == p):
            ans = solve(i + 1, nx, (l + nx) % 2)

    dp[i][prv][l] = ans
    return ans

print(solve(0, -1, 0))
