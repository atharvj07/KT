#include <bits/stdc++.h>
using namespace std;

int gcd(long long a, long long b) {
  if (b == 0)
    return a;
  else
    return gcd(b, a % b);
}

int main() {
  long long a, b;
  cin >> a >> b;
  long long ans = 1;
  for (int i = 1; i <= min(a, b); i++) {
    ans = gcd(ans, (long long)i);
  }
  cout << ans << endl;
  return 0;
}
