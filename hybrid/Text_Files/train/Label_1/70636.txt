n,m=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
q=int(input())
while q>0:
    q-=1
    t,i,x=map(int,input().split())
    if t==1:
        a[i-1]=x
    else:
        b[i-1]=x
    c=[0]*n
    for i in b:
        for j in range(n):
            if a[j]<=i:
                c[j]+=1
    print(max(a) if max(c)<m else -1)
