import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        long[][] dp = new long[n + 1][m + 1];
        long[][] prefixSum = new long[n + 1][m + 1];

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                dp[i][j] = (dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1] + (i - j) * (i - j)) % p;
            }
        }

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                prefixSum[i][j] = (prefixSum[i - 1][j] + prefixSum[i][j - 1] - prefixSum[i - 1][j - 1] + dp[i][j]) % p;
            }
        }

        long answer = 0;
        for (int i = m; i <= n; i++) {
            answer = (answer + prefixSum[i][m] - prefixSum[i - m][m]) % p;
        }

        System.out.println(answer);
    }
}
