import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        long[][] b = new long[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                b[i][j] = scanner.nextInt();
            }
        }

        Arrays.sort(b, (x, y) -> Long.compare(x[2], y[2]));

        int[] ans = new int[m];
        int j = 0;
        int l = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (j == m) {
                break;
            }
            if (b[i][0] == 1 && a[(int) b[i][1] - 1] == b[i][2]) {
                l++;
                continue;
            }
            if (b[i][0] == 2 && a[(int) b[i][1] - 1] < b[i][2]) {
                ans[j++] = (int) b[i][1];
                a[(int) b[i][1] - 1] = b[i][2];
                l++;
            }
            if (b[i][0] == 3 && a[(int) b[i][1] - 1] > 1) {
                ans[j++] = (int) b[i][1];
                a[(int) b[i][1] - 1] /= b[i][2];
                l++;
            }
        }

        System.out.println(l);
        for (int i = 0; i < l; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
