#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
int main() {
  int n;
  cin >> n;
  if (n == 2) {
    int a, b;
    cin >> a >> b;
    cout << "1 1 2\n";
    cout << "2 1\n";
    return 0;
  }
  vector<int> a(n + 2);
  deque<int> que;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    while (!que.empty() && a[i] > a[que.back()]) {
      que.pop_back();
    }
    que.push_back(i);
  }
  for (int i = n - 1; i >= 1; i--) {
    int x = que.front();
    if (que.size() > 1 && -a[que[1]] < a[x]) {
      cout << "1 " << x << " " << que[1] << "\n";
      que.pop_front();
    } else {
      cout << "2 " << x << "\n";
      que.pop_front();
    }
  }
  return 0;
}
