#include <bits/stdc++.h>
using namespace std;

int n, m;
vector<int> a, b, f, pos;
bool ok;

int main() {
    cin >> n >> m;
    a.resize(m), b.resize(m), f.resize(n + 1), pos.resize(n + 1, -1);
    for (int i = 0; i < n; i++) { cin >> f[i + 1]; }
    for (int i = 0; i < m; i++) { cin >> b[i]; }
    for (int i = 0; i < m; i++) {
        if (pos[b[i]] != -1) {
            ok = 1;
            break;
        }
        pos[b[i]] = i;
        a[i] = f[b[i]];
    }
    if (ok) {
        cout << "Ambiguity\n";
    } else {
        for (int i = 0; i < m; i++) { cout << a[i] << " "; }
        cout << endl;
    }
    return 0;
}
