#include <iostream>
#include <cstdio>
#include <vector>
#include <queue>

using namespace std;

void dfs(int x, int y, const vector<string>& v, vector<vector<int>>& sz, vector<vector<bool>>& vis) {
    if (x >= 0 && x < v.size() && y >= 0 && y < v[0].size() && v[x][y] == '.' && !vis[x][y]) {
        vis[x][y] = true;
        sz[x][y]++;
        dfs(x + 1, y, v, sz, vis);
        dfs(x, y + 1, v, sz, vis);
        dfs(x - 1, y, v, sz, vis);
        dfs(x, y - 1, v, sz, vis);
    }
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<string> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i];
    }
    vector<vector<int>> sz(n, vector<int>(m, 0));
    vector<vector<bool>> vis(n, vector<bool>(m, false));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (v[i][j] == '*') {
                v[i][j] = '.';
                dfs(i, j, v, sz, vis);
                v[i][j] = '*';
            }
            sz[i][j] = sz[i][j] % 10;
        }
    }
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cout << sz[i][j];
        }
        cout << endl;
    }
    return 0;
}
