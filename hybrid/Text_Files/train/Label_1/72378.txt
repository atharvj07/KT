n, q = map(int, input().split())
tasks = []
for i in range(q):
    t, k, d = map(int, input().split())
    tasks.append((t, k, d, i))
tasks.sort()
busy = set()
for t, k, d, i in tasks:
    ava = [j for j in range(1, n+1) if j not in busy and j+d-1 >= t]
    if len(ava) < k:
        print(-1)
    else:
        print(sum(ava[:k]))
        busy.update(ava[:k])
