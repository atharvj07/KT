#include <bits/stdc++.h>
using namespace std;
#define ld long double
const ld eps = 1e-6;

int main() {
  ld R, x1, y1, x2, y2;
  cin >> R >> x1 >> y1 >> x2 >> y2;
  ld dx = x2 - x1, dy = y2 - y1;
  ld dist = hypot(dx, dy);
  if (dist > 2 * R)
    cout << x1 + (ld)dx * R / dist << " " << y1 + (ld)dy * R / dist << " " << R
         << "\n";
  else
    cout << x1 << " " << y1 << " " << dist / 2 << "\n";
  return 0;
}
