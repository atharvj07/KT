#include <iostream>
#include <vector>

using namespace std;

long long dp[56] = {0};

long long fib(int n) {
    if (n <= 2) {
        return n;
    }
    if (n > 2 && dp[n] != 0) {
        return dp[n];
    }
    return dp[n] = fib(n - 1) + fib(n - 2);
}

int main() {
    int n;
    cin >> n;
    cout << fib(n + 2) - 1;
    return 0;
}
