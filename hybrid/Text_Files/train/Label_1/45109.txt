import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length();
        int[] pre = new int[n];
        int[] suf = new int[n];
        Map<Character, Integer> last = new HashMap<>();
        for (int i = 0; i < n; i++) {
            pre[i] = last.getOrDefault(s.charAt(i), -1);
            last.put(s.charAt(i), i);
        }
        last = new HashMap<>();
        for (int i = n - 1; i >= 0; i--) {
            suf[i] = last.getOrDefault(s.charAt(i), n);
            last.put(s.charAt(i), i);
        }
        int ans = 0;
        Set<Character> colors = new HashSet<>();
        for (int i = 0; i < n; i++) {
            colors.add(s.charAt(i));
            for (char c : colors) {
                if (pre[i] < suf[i] - 2 && (i - pre[i] < 3 || suf[i] - i < 3)) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
