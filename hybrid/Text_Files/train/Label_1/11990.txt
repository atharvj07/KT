#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 1;
int n, a[maxn], l[maxn], r[maxn], ans;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    l[n + 1] = r[n + 1] = ans = 0;
    for (int i = n; i >= 1; i--) {
      if (a[i] > r[i + 1]) r[i] = a[i], ans++; else r[i] = r[i + 1];
      if (a[i] < l[i + 1]) l[i] = a[i], ans++; else l[i] = l[i + 1];
    }
    cout << ans << '\n';
  }
  return 0;
}
