#include <vector>
#include <bitset>
#include <iostream>
using namespace std;

int main() {
    long long n, a;
    cin >> n;
    bitset<32> xor_val;
    bool ok = false;
    for (long long i = 0; i < n; i++) {
        cin >> a;
        xor_val ^= a;
        if (xor_val == 0) {
            ok = true;
            break;
        }
    }
    if (!ok) {
        cout << n << endl;
    } else {
        cout << -1 << endl;
    }
    return 0;
}
