import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int y1 = sc.nextInt(), m1 = sc.nextInt(), d1 = sc.nextInt();
            int y2 = sc.nextInt(), m2 = sc.nextInt(), d2 = sc.nextInt();
            if (y1 == -1 && m1 == -1 && d1 == -1 && y2 == -1 && m2 == -1 && d2 == -1) {
                break;
            }
            int[] monthDays = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
            boolean leap1 = (y1 % 4 == 0 && (y1 % 100 != 0 || y1 % 400 == 0));
            boolean leap2 = (y2 % 4 == 0 && (y2 % 100 != 0 || y2 % 400 == 0));
            if (leap1) monthDays[1] = 29;
            if (leap2) monthDays[1] = 29;
            int days = 0;
            for (int i = y1; i < y2; i++) {
                days += 365;
                if ((i % 4 == 0 && (i % 100 != 0 || i % 400 == 0))) days++;
            }
            for (int i = 0; i < m1 - 1; i++) {
                days -= monthDays[i];
            }
            days -= d1;
            for (int i = 0; i < m2 - 1; i++) {
                days += monthDays[i];
            }
            days += d2;
            System.out.println(days);
        }
    }
}
