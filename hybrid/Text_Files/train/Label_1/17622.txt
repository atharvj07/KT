#include <bits/stdc++.h>
using namespace std;
int main() {
    int m;
    cin >> m;
    int c[101];
    for (int i = 0; i < m; ++i) cin >> c[i];
    int x, y;
    cin >> x >> y;
    int sum = 0;
    for (int i = 0; i < m; ++i) sum += c[i] * i;
    int cnt = 0;
    for (int i = 0; i < m; ++i) cnt += c[i];
    int l = (x + y + m - 1) / m, r = (sum + y) / cnt;
    if (l > r) {
        cout << 0 << endl;
        return 0;
    }
    int ans = r;
    while (ans >= l) {
        int sum1 = 0, cnt1 = 0, cur = 0;
        for (int i = 0; i < m; ++i) {
            int t = min(ans, i + 1);
            sum1 += c[i] * t;
            cnt1 += c[i];
            if (sum1 >= x * cur) {
                ++cur;
                cnt1 = 0;
                sum1 = 0;
            }
            if (cnt1 > y) {
                ans = -1;
                break;
            }
        }
        if (ans != -1 && cur >= 2) break;
        --ans;
    }
    cout << ans << endl;
}
