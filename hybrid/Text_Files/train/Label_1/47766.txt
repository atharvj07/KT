import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    Point[] p = new Point[n];
    for (int i = 0; i < n; i++) {
      p[i] = new Point(sc.nextInt(), sc.nextInt());
    }
    Arrays.sort(p, new Comparator<Point>() {
      @Override
      public int compare(Point o1, Point o2) {
        return o1.y - o2.y;
      }
    });

    int[] lis = new int[n];
    for (int i = 0; i < n; i++) lis[i] = 1;
    for (int i = 1; i < n; i++) {
      for (int j = 0; j < i; j++) {
        if (p[i].x > p[j].x && lis[i] < lis[j] + 1) {
          lis[i] = lis[j] + 1;
        }
      }
    }

    int ans = 0;
    for (int i = 0; i < n; i++) ans += lis[i] - 1;
    System.out.println(ans);
  }
}

class Point {
  int x, y;
  public Point(int x, int y) {
    this.x = x;
    this.y = y;
  }
}
