#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int t; cin >> t;
  while (t--) {
    string s; cin >> s;
    int n = s.size(), x = -1;
    unordered_map<char, int> m;
    for (char c : s) m[c]++;
    for (auto [c, f] : m) {
      if (f >= (n + 1) / 2) x = 0;
    }
    if (x == -1) cout << s;
    else cout << x;
    cout << endl;
  }
  return 0;
}
