import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		int a[] = new int[n];
		int b[] = new int[m];
		int[] cnt = new int[1 << 11];
		for (int i = 0; i < n; i++)
			a[i] = sc.nextInt();
		for (int i = 0; i < m; i++)
			b[i] = sc.nextInt();
		for (int i : a)
			cnt[i & 1023]++;
		for (int i : b) {
			cnt[i & 1023]--;
		}
		int ans = 0;
		for (int i = 0; i < (1 << 11); i++) {
			if (cnt[i] <= 0)
				ans += cnt[i];
		}
		System.out.println(-ans);
	}
}
