#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  int k = 0;
  for (int i = 1; i * i <= n; i++) {
    if (n % i == 0) k += 2;
    if (i * i == n) k--;
  }
  cout << k << endl;
  for (int i = 1; i * i <= n; i++) {
    if (n % i == 0) {
      cout << i << " ";
      if (i * i != n) { cout << n / i << " "; }
    }
  }
  return 0;
}
