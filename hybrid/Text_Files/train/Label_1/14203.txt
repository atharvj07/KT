#include <bits/stdc++.h>
using namespace std;

string s, t;

int main() {
  int n;
  cin >> n >> s >> t;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == t[i]) continue;
    char c = s[i];
    for (int j = i + 1; j < n; j++) s[j] = c;
    s[i] = c + 1;
    cout << s << endl;
    return 0;
  }
}
