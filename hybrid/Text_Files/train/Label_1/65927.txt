def solve():
    s = input()
    c = [0,0,0]
    for i in s:
        if i == '.':
            if c[0] + c[1] + c[2] > 0:
                return "Yes"
        elif i == 'A':
            c[0] += 1
        elif i == 'B':
            c[1] += 1
        else:
            c[2] += 1
    return "No"

print(solve())
