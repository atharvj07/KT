import java.util.*;

class Main {

  final static int MAX_VALUE = (1 << 60) - 1;

  public static void solve() {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] b = new int[n];
    for (int i = 0; i < n; i++) {
      b[i] = in.nextInt();
    }
    System.out.println(tryOver(b));
  }

  static String tryOver(int[] b) {
    ArrayList<Integer> bl = new ArrayList<>();
    for (int v : b)
      bl.add(v);
    Collections.sort(bl);
    int[] av = new int[bl.size()];
    for (int i = 0; i < av.length; i++)
      av[i] = bl.get(i);
    long[] sum = new long[av.length];
    sum[0] = av[0];
    for (int i = 1; i < av.length; i++)
      sum[i] = sum[i - 1] + av[i];
    if (sum[sum.length - 1] == MAX_VALUE)
      return "No";
    for (int j = 0; j < b.length; j++) {
      int k = bl.indexOf(b[j]);
      if (checkOK(sum, k, b.length - 1)) {
        bl.remove(k);
        b[j] = av[k];
      }
    }
    int a = 0;
    for (int j = 0; j < b.length; j++) {
      a ^= b[j];
      if (a <= b[j])
        return "No";
    }
    System.out.println("Yes");
    for (int j = 0; j < b.length; j++)
      System.out.print(b[j] + " ");
    System.out.print("\n");
    return "";
  }

  static boolean checkOK(long[] sum, int k, int right) {
    if (k == right)
      return true;
    long s = (sum[right] - sum[k] + MAX_VALUE) % MAX_VALUE;
    long s1 = (s + sum[k]) % MAX_VALUE;
    long s2 = (k > 0 ? (sum[k - 1] + MAX_VALUE) % MAX_VALUE : 0);
    return s1 >= s2;
  }

  public static void main(String[] args) {
    solve();
  }
}
