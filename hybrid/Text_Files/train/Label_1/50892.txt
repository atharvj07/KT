import java.util.*;

public class Main {
    public static void main(String[] args) {
        try (Scanner in = new Scanner(System.in)) {
            int n = in.nextInt();
            int p = in.nextInt();
            int[] t = new int[n];
            for (int i = 0; i < n; i++) {
                t[i] = in.nextInt();
            }
            int[] ans = new int[n];
            TreeSet<Integer> set = new TreeSet<>(); // stores indices of passengers who haven't got water yet
            for (int i = 0; i < n; i++) {
                set.add(i);
            }
            int time = 0;
            while (!set.isEmpty()) {
                int first = set.first();
                int end = Math.min(time + p, t[first]);
                while (set.lower(first) != null && t[set.lower(first)] <= end) {
                    first = set.lower(first);
                }
                set.remove(first);
                ans[first] = end;
                time = end;
            }
            for (int i : ans) {
                System.out.println(i);
            }
        }
    }
}
