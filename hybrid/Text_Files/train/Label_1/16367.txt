#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
const int mx = 1e5 + 7;
int bigmod(int n, int p) {
    if (p == 0) {
        return 1;
    } else if (p % 2 == 1) {
        return n * bigmod(n, p - 1) % mod;
    } else {
        int ret = bigmod(n, p / 2);
        return (ret * 1ll * ret) % mod;
    }
}
int pw(int x) { return bigmod(2, x); }
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int k, a[mx];
    cin >> k;
    ll ans = 1;
    for (int i = 0; i < k; i++) {
        cin >> a[i];
        ans = ans * 1ll * a[i] % mod;
    }
    if (ans % 2) {
        cout << ans / 2 << "/" << pw(ans / 2) << "\n";
    } else {
        cout << ans / 2 << "/" << pw(ans / 4) << "\n";
    }
    return 0;
}
