#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<long long> res;
  for (int i = 1; i * i <= n; i++) {
    if (n % i == 0) {
      long long tmp = n - i;
      if (tmp <= 9 * i) { res.push_back(i); }
      tmp = n / i;
      if (tmp <= 9 * i && tmp != i) { res.push_back(tmp); }
    }
  }
  cout << res.size() << '\n';
  for (auto i : res) { cout << i << '\n'; }
  return 0;
}
