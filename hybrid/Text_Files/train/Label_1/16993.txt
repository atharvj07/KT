#include<bits/stdc++.h>
using namespace std;
const int M=1e9+7,N=1e6+5;
int32_t main(){
    ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);
    int n,x;
    cin>>n>>x;
    vector<int>v(n);
    for(auto &i:v)cin>>i;
    long long ans=0;
    for(int i=0;i<n;i++){
        x=max(x,1);
        ans+=x*v[i];
        x--;
    }
    cout<<ans<<endl;
    return 0;
}
