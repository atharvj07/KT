#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,v[100005], sum[100005], l[100005], r[100005];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> v[i];
    sum[i] = sum[i - 1] + v[i];
  }
  int t;
  cin >> t;
  while (t--) {
    int type, a, b;
    cin >> type >> a >> b;
    if (type == 1) cout << sum[b] - sum[a - 1] << '\n';
    else cout << l[b] - l[a - 1] << '\n';
  }
}
