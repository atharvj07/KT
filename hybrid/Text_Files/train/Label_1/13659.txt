#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,k;
	cin>>n>>k;
	vector<int> arr(n);
	map<int,int> mp;
	for(int i=0;i<n;i++){
		cin>>arr[i];
		mp[arr[i]]++;
	}
	sort(arr.begin(),arr.end());
	int ans=0;
	for(int i=0;i<n;i++){
		while(arr[i]<=1e9 && mp[arr[i]]){
			mp[arr[i]]--;
			arr[i]*=k;
			ans++;
		}
	}
	cout<<ans<<endl;
}
