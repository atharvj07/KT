#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), p(32);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    int j = 0;
    while (a[i] > 0) {
      if (a[i] % 2) p[j]++;
      j++;
      a[i] /= 2;
    }
  }
  long long ans = 0;
  for (int i = 0; i < 32; i++) {
    ans += 1ll * p[i] * (p[i] - 1) / 2;
  }
  cout << ans << endl;
}
