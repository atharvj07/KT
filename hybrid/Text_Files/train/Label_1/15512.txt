#include <iostream>
#include <vector>

using namespace std;

bool check(vector<int> &a, vector<int> &b) {
    for (int i = 0; i < a.size(); i++) {
        if ((a[i] | b[i]) != a[i] || (a[i] & b[i]) != b[i]) return false;
    }
    return true;
}

bool solve(vector<int> &a, vector<int> &b, vector<int> &ans) {
    if (a.size() == 0) return true;
    int x = a[0], y = b[0];
    for (int i = 1; i <= 3; i++) {
        for (int j = 1; j <= 3; j++) {
            ans.push_back(i);
            if ((i | j) == x && (i & j) == y) {
                a.erase(a.begin());
                b.erase(b.begin());
                if (solve(a, b, ans)) return true;
                a.insert(a.begin(), x);
                b.insert(b.begin(), y);
            }
            ans.pop_back();
        }
    }
    return false;
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n - 1), b(n - 1);
    for (int i = 0; i < n - 1; i++) cin >> a[i];
    for (int i = 0; i < n - 1; i++) cin >> b[i];
    vector<int> ans;
    if (check(a, b) && solve(a, b, ans)) {
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) cout << ans[i] << " ";
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
