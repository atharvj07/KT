n, s = map(int, input().split())
a = [0] * n
b = [0] * n
for i in range(n):
    x, ai, bi = map(int, input().split())
    a[i] = ai * x
    b[i] = bi * x
    s -= x
l1 = []
l2 = []
for x in a:
    if x > 0:
        l1.append(x)
for x in b:
    if x > 0:
        l2.append(x)
l1.sort(reverse=True)
l2.sort(reverse=True)
ans = 0
for x in l1:
    if s < 0: break
    ans += x
    s -= 1
for x in l2:
    if s < 0: break
    ans += x
    s -= 1
print(ans)
