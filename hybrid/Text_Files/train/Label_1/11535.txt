#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b, c;
    cin >> a >> b >> c;
    int ans = 0;
    while (a > 0 && b >= 2) {
      a--;
      b -= 2;
      ans += 3;
    }
    while (b > 0 && c >= 2) {
      b--;
      c -= 2;
      ans += 3;
    }
    cout << ans << endl;
  }
  return 0;
}
