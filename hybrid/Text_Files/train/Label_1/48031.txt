import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        final int MAX = 100 + 7;
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = "";

        while (n > 0 && m > 0) {
            if (n > m) {
                s += "BG";
                n--;
            } else {
                s += "GB";
                m--;
            }
        }

        while (n > 0) {
            s += 'B';
            n--;
        }

        while (m > 0) {
            s += 'G';
            m--;
        }

        System.out.println(s);
    }
}
