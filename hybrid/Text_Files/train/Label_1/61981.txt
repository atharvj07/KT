def hex_to_bin(hex_num):
    return bin(int(hex_num, 16))[2:].zfill(4)

def is_compression_possible(matrix, x):
    n = len(matrix)
    for i in range(0, n, x):
        for j in range(0, n, x):
            value = matrix[i][j]
            for a in range(x):
                for b in range(x):
                    if matrix[i + a][j + b] != value:
                        return False
    return True

n = int(input().strip())
matrix = []

for _ in range(n):
    row = input().strip()
    bin_row = ''.join([hex_to_bin(ch) for ch in row])
    matrix.append(bin_row)

x = n
while x > 1:
    if n % x == 0 and is_compression_possible(matrix, x):
        break
    x -= 1

print(x)
