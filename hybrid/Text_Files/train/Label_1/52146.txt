import java.io.*;
import java.util.*;

public class Main {
    static int n, m;
    static int[][] a, b;
    static int[] c;
    static int[] d;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n][m];
        b = new int[n][m];
        c = new int[n];
        d = new int[m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                b[i][j] = a[i][j];
            }
        }
        for (int i = 0; i < n; i++) {
            Set<Integer> set = new HashSet<>();
            for (int j = 0; j < m; j++) {
                set.add(a[i][j]);
            }
            List<Integer> list = new ArrayList<>(set);
            Collections.sort(list);
            for (int j = 0; j < m; j++) {
                a[i][j] = Collections.binarySearch(list, a[i][j]) + 1;
            }
        }
        for (int j = 0; j < m; j++) {
            Set<Integer> set = new HashSet<>();
            for (int i = 0; i < n; i++) {
                set.add(a[i][j]);
            }
            List<Integer> list = new ArrayList<>(set);
            Collections.sort(list);
            for (int i = 0; i < n; i++) {
                a[i][j] = Collections.binarySearch(list, a[i][j]) + 1;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(a[i][j] + " ");
            }
            System.out.println();
        }
    }
}
