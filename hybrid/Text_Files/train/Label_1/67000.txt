for T in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    b = [0] * n
    for x in a:
        b[x - 1] += 1
    if sum(1 for x in b if x >= 1) < k:
        print(-1)
        continue
    ans = 0
    for i in range(n - 1, 0, -1):
        if b[i] > 0:
            b[i] -= 1
            b[i - 1] += 1
            ans += 1
        if b[i - 1] >= 1:
            break
    print(ans)
