#include <iostream>
#include <string>
using namespace std;

int main() {
  char shift; cin >> shift;
  string s; cin >> s;
  if (shift=='R') for (int i = 0; i < s.size(); i++) s[i] = (s[i] == 'z' ? 'y' : (s[i] == 'Z' ? 'Y' : (((s[i] >= 'a' && s[i] <= 'z') || (s[i] >= 'A' && s[i] <= 'Z')) ? s[i] - 1 : s[i])));
  else for (int i = 0; i < s.size(); i++) s[i] = (s[i] == 'a' ? 'z' : (s[i] == 'A' ? 'Z' : (((s[i] >= 'a' && s[i] <= 'z') || (s[i] >= 'A' && s[i] <= 'Z')) ? s[i] + 1 : s[i])));
  cout << s << '\n';
  return 0;
}
