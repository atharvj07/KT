import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt(), m = cin.nextInt(), t = cin.nextInt();
        int[][] dis = new int[m+1][m+1];
        String s = cin.next();
        int[] rem = new int[m];
        for (int i = 0; i < m; i++) {
            rem[i] = cin.nextInt();
        }
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= m; j++) {
                dis[i][j] = cin.nextInt();
            }
        }
        int[][][] dp = new int[n][1<<m][2];
        int inf = 0x3f3f3f3f;
        for (int i = 0; i < (1<<m); i++) {
            Arrays.fill(dp[n - 1][i], inf);
        }
        dp[n - 1][(1<<m) - 1][0] = 0;
        dp[n - 1][(1<<m) - 1][1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 0; j < (1<<m); j++) {
                for (int p = 0; p < 2; p++) {
                    int add = 0;
                    if (p == 0) {
                        add = rem[s.charAt(i) - 'A'];
                    }
                    dp[i][j][p] = Math.min(dp[i+1][j][p], dp[i+1][j|(1<< (s.charAt(i) - 'A'))][1] + dis[0][s.charAt(i) - 'A']);
                    dp[i][j][p] = Math.min(dp[i][j][p], dp[i+1][j][1] + add + dis[0][s.charAt(i) - 'A']);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < (1<<m); i++) {
            if (dp[0][i][0] <= t) ans++;
        }
        System.out.println(ans);
    }
}
