#include <bits/stdc++.h>
using namespace std;

const int maxn = 100000 + 5, maxw = 100000 + 5;

int main() {
  int a, b, h, w, n;
  cin >> a >> b >> h >> w >> n;
  int ansx = maxw, ansy = maxw;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    ansy = min(ansy, (a + b - 1) / x);
    ansx = min(ansx, (w + h - 1) / x);
  }
  if (ansy > h && ansx > w)
    cout << -1 << '\n';
  else
    cout << (ansy <= h ? ansx : ansy) << '\n';
  return 0;
}
