n, k = map(int, input().split())
s = input()
for i in range(n - 1):
    if s[i] != '0' or i != 0:
        break
s = s[i:]
n = len(s)
if k >= n: s = '0' + s
else:
    for i in range(n):
        if k == 0 or s[i] == '0':
            s = s[i:]
            n = len(s)
            break
        k -= 1
    if k != 0: s = '0' + s
    s = '1' + '0' * (n - 1 - k) + s[k + 1:]
print(s)
