#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1010;
int dp[maxn][maxn], p[maxn][maxn];
int w[maxn];
bool vis[maxn];
vector<int> G[maxn];
int n, m, s, t;
void dfs(int x){
	vis[x] = true;
	for(auto y: G[x]){
		if(vis[y]) continue;
		dp[x][y] = 1;
		dfs(y);
		for(int z = 1; z <= n; z++){
			if(dp[x][z] == n || dp[y][z] == n){
				p[x][z] = max(p[x][z], p[y][z]+1);
				p[z][x] = max(p[z][x], p[y][z]+1);
			}
			else if(dp[x][z] > 0){
				p[x][z] = max(p[x][z], dp[y][z]+1);
				p[z][x] = max(p[z][x], dp[y][z]+1);
			}
			else if(dp[y][z] > 0){
				p[x][z] = max(p[x][z], dp[y][z]+1);
				p[z][x] = max(p[z][x], dp[y][z]+1);
			}
		}
		dp[x][y] += dp[y][x];
		dp[y][x] = dp[x][y];
	}
}
int main(){
	cin >> n >> m >> s >> t;
	for(int i = 1; i <= n; i++){
		dp[i][i] = n;
	}
	for(int i = 0; i < m; i++){
		int u, v; cin >> u >> v;
		G[u].push_back(v);
		G[v].push_back(u);
	}
	dfs(s);
	int ans = 0;
	for(int i = 1; i <= n; i++){
		if(i != s && i != t){
			if(max(p[i][s], p[i][t]) <= p[s][t]) ans++;
		}
	}
	cout << ans << endl;
}
