import sys
input = sys.stdin.readline

n, U = map(int, input().split())
E = list(map(int, input().rstrip().split()))
ans = -1
for i in range(n):
    for j in range(i + 1, n):
        for k in range(j + 1, n):
            if E[k] - E[i] <= U:
                eta = (E[k] - E[j]) / (E[k] - E[i]);
                if ans < eta:
                    ans = eta
print(ans)
