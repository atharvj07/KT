import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class BullsAndCows {

    public static String guessNumber(String[] guesses) {
        List<int[]> constraints = new ArrayList<>();
        for (String guess : guesses) {
            String[] parts = guess.split(" ");
            int[] constraint = new int[5];
            constraint[0] = Integer.parseInt(parts[0]);
            constraint[1] = Integer.parseInt(parts[1]);
            constraint[2] = Integer.parseInt(parts[2]);
            constraints.add(constraint);
        }

        int[] digits = new int[10];
        Arrays.fill(digits, -1);

        for (int[] constraint : constraints) {
            int guessNumber = constraint[0];
            int bulls = constraint[1];
            int cows = constraint[2];

            for (int i = 0; i < 4; i++) {
                int digit = guessNumber % 10;
                if (i < bulls) {
                    digits[digit] = i;
                } else if (cows > 0) {
                    for (int j = 0; j < 10; j++) {
                        if (digits[j] == -1 && j != digit) {
                            digits[j] = -2;
                            cows--;
                            break;
                        }
                    }
                }
                guessNumber /= 10;
            }
        }

        StringBuilder result = new StringBuilder();
        for (int digit : digits) {
            if (digit != -1 && digit != -2) {
                result.append(digit);
            }
        }

        if (result.length() == 4) {
            return result.toString();
        } else if (result.length() < 4) {
            return "Need more data";
        } else {
            return "Incorrect data";
        }
    }

    public static void main(String[] args) {
        String[] guesses = {"1263 1 2", "8103 2 1"};
        String result = guessNumber(guesses);
        System.out.println(result);
    }
}
