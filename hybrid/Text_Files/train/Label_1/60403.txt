def check(n):
    if n % 8 == 0:
        return print('YES\n', n, sep='')
    s = str(n)
    while len(s) > 1:
        for i in range(len(s)):
            t = int(s[:i] + s[i + 1:])
            if t % 8 == 0:
                return print('YES\n', t, sep='')
        s = s[1:]
    print('NO')

check(int(input()))
