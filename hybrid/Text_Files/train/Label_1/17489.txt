#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    getline(cin, s);
    int n = s.size();
    int dp[n][n];
    memset(dp, 0, sizeof(dp));
    for (int i = 0; i < n; i++) dp[i][i] = s[i] - '0';
    for (int l = 3; l <= n; l += 2) {
        for (int i = 0, j = i + l - 1; j < n; i += 2, j += 2) {
            for (int k = i + 1; k < j; k += 2) {
                if (s[k] == '*') {
                    dp[i][j] = max(dp[i][j], dp[i][k - 1] * dp[k + 1][j]);
                } else {
                    dp[i][j] = max(dp[i][j], dp[i][k - 1] + dp[k + 1][j]);
                }
            }
        }
    }
    cout << dp[0][n - 1] << endl;
    return 0;
}
