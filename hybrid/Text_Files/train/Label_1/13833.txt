#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll MOD = 1e9+7;

ll pw(ll x, ll y) {
    ll ans = 1;
    while (y) {
        if (y % 2 == 0) {
            x = (x * x) % MOD;
            y /= 2;
        } else {
            ans = (ans * x) % MOD;
            y--;
        }
    }
    return ans;
}

int main() {
    ll n, k; cin >> n >> k;
    string s; cin >> s;
    if (k == 1 || k == n) {
        cout << "once again" << endl;
        return 0;
    }
    if (s[0] == s[n-1]) {
        ll ans = pw(2, (n-k)/2);
        ans = (ans + MOD - 1) % MOD;
        cout << (ans % 2 == 0 ? "once again" : "quailty") << endl;
    } else {
        ll ans = pw(2, (n-k-1)/2 + 1);
        ans = (ans + MOD - 1) % MOD;
        cout << (ans % 2 == 0 ? "once again" : "tokitsukaze") << endl;
    }
}
