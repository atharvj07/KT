import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int price = 700;
        for (char c : s.toCharArray()) {
            if (c == 'o') {
                price += 100;
            }
        }
        System.out.println(price);
    }
}
