#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    set<int> a;
    while(n--) {
        int x; cin >> x;
        a.insert(x);
    }
    vector<int> v;
    for(auto x: a) v.push_back(x);
    int k = v.size();
    if(k==1) k=0;
    else if(k==2) k=1;
    else if(k>2) {
        k = 2;
        if(v[0] > 1) v[1]=1;
    }
    cout << k << endl;
    for(int i=0; i<k; i++) {
        cout << v[i] << " ";
    }
    cout << endl;
}
