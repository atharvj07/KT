import sys

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    if len(set(a)) != n:
        print("NO")
    else:
        print("YES")
        l = []
        r = []
        i = 0
        j = n - 1
        while i <= j:
            if a[i] != a[j]:
                l.append(a[i])
                r.append(a[j])
            i += 1
            j -= 1
        print(len(l))
        print(*l)
        print(len(r))
        print(*r[::-1])

solve()
