#include <bits/stdc++.h>
using namespace std;

constexpr int maxn = 2e5 + 5;
vector<int> b;
int main() {
  int n;
  cin >> n;
  b.resize(n);
  for (int &x : b) cin >> x;
  set<int> s;
  for (int i = 0; i < n; i++) {
    if (i == n - 1 || b[i] + b[i + 1] > b[n - 1]) {
      s.insert(b[i]);
      if (s.size() == 2) {
        cout << 1 << endl;
        cout << b[i] << endl;
        return 0;
      }
    }
  }
  cout << 0 << endl;
}
