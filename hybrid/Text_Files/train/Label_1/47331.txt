import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int numTilesVertically = (h + a - 1) / a;
        int numTilesHorizontally = (w + b - 1) / b;
        int totalTilesArea = numTilesVertically * numTilesHorizontally * a * b;
        int wallArea = h * w;
        System.out.println(wallArea - totalTilesArea);
    }
}
