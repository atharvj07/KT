#include <bits/stdc++.h>
using namespace std;

int l, r;
const int maxn = 100005;
int vis[maxn];

int main() {
  cin >> l >> r;
  int x = l;
  while (x <= r) {
    int p = x;
    memset(vis, 0, sizeof(vis));
    bool flg = true;
    while (p) {
      int w = p % 10;
      p /= 10;
      if (vis[w]) {
        flg = false;
        break;
      }
      vis[w] = 1;
    }
    if (flg) {
      cout << x << endl;
      return 0;
    }
    x++;
  }
  cout << -1 << endl;

  return 0;
}
