from sys import stdin

for _ in range(int(stdin.readline())):
    n = int(stdin.readline())

    seen = [False] * 10000
    arr = [int(stdin.readline()) for _ in range(n)]
    for i in range(n):
        seen[arr[i]] = True
    ans = 0
    for i in range(n):
        for j in range(0, 10000):
            if not seen[j] and j != arr[i]:
                arr[i] = j
                seen[j] = True
                ans += 1
                break
    print(ans)
    for val in arr:
        print(val)
