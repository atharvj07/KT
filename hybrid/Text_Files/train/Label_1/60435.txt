def f(n, v):
    mod = 10 ** 9 + 7
    f = [1]
    for i in range(1, n + 1):
        f += [f[-1] * i % mod]
    I = [0] * (n + 3)
    for x in v:
        I[x] += 1
    r = 1
    for x in range(2, n + 1):
        r = r * f[I[x]] % mod
    print(r)

n = int(input())
v = list(map(int, input().split())) + [0]
f(n, v)
