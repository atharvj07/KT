#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, i, j, tmp, p[55][55];
    cin >> n >> k;
    vector<int> a(n + 1);
    for (i = 1; i <= n; i++) cin >> a[i];
    sort(a.begin() + 1, a.begin() + n + 1);
    for (i = k; i >= 1; i--) {
        if (a[i] != a[i + 1]) {
            j = i - 1;
            while (j >= 1) {
                if (a[j] != a[j + 1]) {
                    p[i][0] = 2;
                    p[i][1] = a[j + 1];
                    p[i][2] = a[j];
                    j--;
                    break;
                }
                j--;
            }
            if (j == 0) {
                if ((a[1] + a[2]) != a[i]) {
                    p[i][0] = 2;
                    p[i][1] = a[2];
                    p[i][2] = a[1];
                } else {
                    p[i][0] = 1;
                    p[i][1] = a[1];
                }
            }
            if (p[i][0] == 0) {
                p[i][0] = 2;
                p[i][1] = a[2];
                p[i][2] = a[1];
            }
            cout << p[i][0];
            for (j = 1; j <= p[i][0]; j++) cout << " " << p[i][j];
            cout << "\n";
        }
    }
    return 0;
}
