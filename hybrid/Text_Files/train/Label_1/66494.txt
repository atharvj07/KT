import bisect

n = int(input())
a = []
for i in range(n):
    c, p = map(int, input().split())
    a.append((c, p))
k = int(input())
b = list(map(int, input().split()))

a.sort(key=lambda x: x[1], reverse=True)
b.sort()

j = 0
ans = []
tot = 0
for c, p in a:
    idx = bisect.bisect_right(b, c)
    if idx == len(b):
        continue
    ans.append((j + 1, idx + 1))
    tot += p
    b[idx] -= c
    j += 1

print(len(ans), tot)
for x in ans:
    print(*x)
