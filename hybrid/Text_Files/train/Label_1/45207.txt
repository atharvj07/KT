import java.util.*;

class Main {
    public static void main(String[] args) {
        char[] s = new Scanner(System.in).next().toCharArray();
        for (int i = 0, j = 0, n = s.length; i < n;) {
            while (i < n && s[i] == 'x')
                i++;
            while (j < n && s[j] == 'y')
                j++;
            if (i == n)
                System.out.println(s);
            else if (j == n || i + 1 < j) {
                char c = s[i];
                s[i] = s[i + 1];
                s[i + 1] = c;
                i += 2;
            } else {
                s[i] = s[j];
                s[j] = 0;
                i++;
            }
            j = i;
        }
    }
}
