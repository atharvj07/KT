#include<iostream>
using namespace std;
int main() {
    int n, a[100], i, j, min, sum;
    cin >> n;
    for (i = 0; i < n; i++) cin >> a[i];
    while (true) {
        min = 300;
        for (i = 0; i < n; i++) min = min < a[i] ? min : a[i];
        if (min == 300) {
            cout << (n % 2 ? "BitLGM" : "BitAryo");
            return 0;
        }
        sum = 0;
        for (i = 0; i < n; i++) {
            if (a[i] == min) sum++;
            a[i] -= min;
        }
        if (n == 1) {
            if (sum % 2) cout << "BitLGM";
            else cout << "BitAryo";
            return 0;
        }
        if (sum % 2 == 0) {
            if (sum == n) {
                cout << (n % 2 ? "BitLGM" : "BitAryo");
                return 0;
            }
            else {
                if (n - sum == 1) {
                    cout << (n % 2 ? "BitAryo" : "BitLGM");
                    return 0;
                }
                else continue;
            }
        }
        else {
            if (sum == 1) {
                cout << (n % 2 ? "BitLGM" : "BitAryo");
                return 0;
            }
            else continue;
        }
    }
    return 0;
}
