#include<bits/stdc++.h>
using namespace std;
int n,k,a[103],b[103],vis[103];
int main()
{
    cin>>n>>k;
    for(int i=1;i<=n;i++)
    {
        cin>>a[i];
    }
    for(int i=1;i<=n;i++)
    {
        cin>>b[i];
    }
    if(b[a[1]]!=1)
    {
        cout<<"NO";
        return 0;
    }
    int cnt=1;
    for(int l=1;l<=k;l++)
    {
        vis[a[cnt]]=1;
        cnt=a[cnt];
        if(cnt==1)
        {
            for(int i=1;i<=n;i++)
            {
                if(!vis[i])
                {
                    if(i!=b[cnt])
                    {
                        cout<<"NO";
                        return 0;
                    }
                }
            }
            cout<<"YES";
            return 0;
        }
    }
    cout<<"NO";
    return 0;
}
