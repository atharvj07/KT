#include <bits/stdc++.h>
using namespace std;
char s[200], t[200], ss[2000], ans[200];
int n, k;
int main() {
  cin >> s >> k;
  n = strlen(s);
  int m = 0;
  for (int i = 0; i < n; i++) {
    char c = s[i];
    if ((c >= 'a' && c <= 'z') || c == '?') {
      ss[m++] = c;
    }
  }
  if (m < k)
    return puts("Impossible"), 0;
  int now = 0;
  for (int i = 0; i < m; i++) {
    char c = ss[i];
    if (c == '?') {
      now++;
      if (now > k) return puts("Impossible"), 0;
    }
    if ((c >= 'a' && c <= 'z') || (c == '?' && now == k)) {
      ans[now++] = c;
    }
  }
  if (now != k) return puts("Impossible"), 0;
  ans[k] = 0;
  cout << ans << endl;
  return 0;
}
