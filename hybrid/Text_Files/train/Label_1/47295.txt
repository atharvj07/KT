import java.util.Scanner;

public class RedundantTVSet {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] intervals = new int[n][2];
        for (int i = 0; i < n; i++) {
            intervals[i][0] = scanner.nextInt();
            intervals[i][1] = scanner.nextInt();
        }

        int leftmostEnd = intervals[0][1];
        int rightmostStart = intervals[0][0];
        for (int i = 1; i < n; i++) {
            leftmostEnd = Math.max(leftmostEnd, intervals[i][1]);
            rightmostStart = Math.min(rightmostStart, intervals[i][0]);
        }

        if (leftmostEnd < rightmostStart) {
            System.out.println(-1);
        } else {
            for (int i = 0; i < n; i++) {
                if (intervals[i][0] > rightmostStart && intervals[i][1] < leftmostEnd) {
                    System.out.println(i + 1);
                    return;
                }
            }
            System.out.println(-1);
        }
    }
}
