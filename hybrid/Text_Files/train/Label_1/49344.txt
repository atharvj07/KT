import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] a = new int[4];
        int[] b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }
        int d1 = Math.max(a[0], a[1]);
        int d2 = Math.max(a[2], a[3]);
        int a1 = Math.max(b[0], b[1]);
        int a2 = Math.max(b[2], b[3]);
        if (d1 > a2 && d2 > a1)
            System.out.println("Team 1");
        else if (d2 > a1 && d1 > a2)
            System.out.println("Team 2");
        else
            System.out.println("Draw");
    }
}
