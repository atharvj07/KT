n = int(input())
a = list(map(int, input().split()))
d = {}
for i in range(n):
    if a[i] not in d:
        d[a[i]] = [i]
    else:
        d[a[i]].append(i)
b = list(d.keys())
b.sort()
c = 0
for i in range(len(b) - 1):
    for j in range(len(d[b[i]])):
        a[d[b[i]][j]] = b[i]
        c += (b[i + 1] - 1 - b[i])
print(c)
