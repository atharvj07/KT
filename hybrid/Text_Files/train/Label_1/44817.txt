import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Main {
    public static class Dist {
        int d;
        int v;

        public Dist(int d, int v) {
            this.d = d;
            this.v = v;
        }

        @Override
        public int hashCode() {
            return d * 31 + v;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Dist dist = (Dist) o;
            return d == dist.d && v == dist.v;
        }
    }

    public static class Edge {
        int v;
        int w;

        public Edge(int v, int w) {
            this.v = v;
            this.w = w;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] dist = new int[n];
        List<Edge>[] list = new List[n];
        for (int i = 0; i < n; i++) {
            list[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            int u = sc.nextInt();
            int k = sc.nextInt();
            for (int j = 0; j < k; j++) {
                int v = sc.nextInt();
                int w = sc.nextInt();
                list[u].add(new Edge(v, w));
            }
        }
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[0] = 0;
        List<Dist> que = new ArrayList<>();
        que.add(new Dist(0, 0));
        while (!que.isEmpty()) {
            Dist d = que.remove(0);
            if (d.d > dist[d.v]) {
                continue;
            }
            for (Edge e : list[d.v]) {
                if (d.d + e.w < dist[e.v]) {
                    dist[e.v] = d.d + e.w;
                    que.add(new Dist(d.d + e.w, e.v));
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(i + " " + dist[i]);
        }
        sc.close();
    }
}
