from functools import reduce

n=int(input())
a=list(map(int,input().split()))
t=[[i,[]]for i in range(n)]
for i in range(n):
    if a[i]==-1:
        t[i][1]=[i]
    else:
        t[a[i]-1][1].append(i)
m=[0]*n
for i in range(n):
    s=set(t[i][1])
    for j in s:
        m[i]+=m[j]
    m[i]+=1
print(reduce(lambda x, y: (x*y)%1000000007, m))
