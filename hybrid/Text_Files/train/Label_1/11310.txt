#include <bits/stdc++.h>
using namespace std;
int main() {
    int l, a[100001], ans = 0;
    cin >> l;
    string s;
    cin >> s;
    int pos = 100001;
    for (int i = 0; i < l; i++) {
        if (s[i] != '0') ans = ans * 10 + s[i] - '0', pos = i;
    }
    for (int i = pos + 1; i < l; i++) a[i - pos] = s[i] - '0';
    sort(a, a + l - pos);
    int cur = 0;
    for (int i = 0; i < l - pos; i++) {
        cur = cur * 10 + a[i];
        if (cur > ans) {
            ans = cur;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
