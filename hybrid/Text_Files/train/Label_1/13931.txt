#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ll x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    if (x1 == x2) {
        cout << x1 + y2 - y1 << " " << y1 << " " << x2 + y2 - y1 << " " << y2 << endl;
    } else if (y1 == y2) {
        cout << x1 << " " << y1 + x2 - x1 << " " << x2 << " " << y1 + x2 - x1 << endl;
    } else {
        cout << -1 << endl;
    }
    return 0;
}
