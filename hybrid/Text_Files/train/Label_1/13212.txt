#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int ans = -1;
    for (int i = 0; i < n; i++) {
      if (a[i] <= x) continue;
      if (ans == -1)
        ans = i;
      else
        ans = -1;
    }
    cout << ans << endl;
  }
}
