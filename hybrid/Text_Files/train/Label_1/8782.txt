#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const int mx = 2e5;
vector<int> freq(mx, 0);
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        freq.clear();
        freq.resize(mx, 0);
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            freq[x]++;
        }
        while (k--) {
            int mexValue = -1;
            int mxValue = -1;
            for (int i = 0; i < mx; i++) {
                if (freq[i] > 0) {
                    mexValue = i;
                    break;
                }
            }
            for (int i = mx - 1; i >= 0; i--) {
                if (freq[i] > 0) {
                    mxValue = i;
                    break;
                }
            }
            int addValue = ceil((mxValue + 1.0) / 2.0);
            freq[addValue]++;
        }
        int cnt = 0;
        for (int i = 0; i < mx; i++) {
            if (freq[i] > 0) { cnt++; }
        }
        cout << cnt << endl;
    }
}
