import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] mesh = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    mesh[j][k] = sc.nextInt();
                }
            }
            boolean valid = true;
            for (int j = 0; j < 5; j++) {
                if (mesh[j][0] + mesh[j][4] != 7 || mesh[0][j] + mesh[4][j] != 7) {
                    valid = false;
                    break;
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (mesh[j][k] == 0) {
                        continue;
                    }
                    if (mesh[j][k] == mesh[j][4 - k] || mesh[j][k] == mesh[4 - j][k]) {
                        valid = false;
                        break;
                    }
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (mesh[j][k] == 0) {
                        continue;
                    }
                    if (mesh[j][k] == mesh[4 - j][4 - k] || mesh[j][k] == mesh[4 - j][k]) {
                        valid = false;
                        break;
                    }
                }
            }
            System.out.println(valid);
        }
    }
}
