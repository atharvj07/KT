#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    ll ans = 0, cur = 0;
    for (auto x : a) {
        if (x >= cur) cur++, ans++;
    }
    cout << ans << '\n';
    return 0;
}
