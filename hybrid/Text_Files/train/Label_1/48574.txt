import java.util.*;

public class Main {

    static long gcd(long a, long b) {
        while (b != 0) {
            long tmp = a;
            a = b;
            b = tmp % b;
        }
        return a;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        long[] b = new long[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }

        long g = a[0];
        for (int i = 1; i < n; i++) {
            g = gcd(g, a[i]);
        }

        boolean ok = true;
        for (int i = 0; i < m; i++) {
            if (g % b[i] != 0) {
                ok = false;
            }
        }

        if (!ok) {
            System.out.println(-1);
            return;
        }

        long max = 0;
        for (int i = 0; i < n; i++) {
            max += a[i] / g;
        }

        for (int i = 0; i < m; i++) {
            max = gcd(max, b[i] / g);
        }

        System.out.println(max);
    }
}
