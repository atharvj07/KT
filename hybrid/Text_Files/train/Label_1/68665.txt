import sys

input = sys.stdin.readlines()
n, m = map(int, input[0].split())
songs = []

for i in range(n):
  song = list(map(int, input[i+1].split()))
  songs.append(song)

duration = 0

for i in range(n):
  duration += songs[i][0] * songs[i][1]

times = list(map(int, input[n+1].split()))

for t in times:
  cur = 0

  for i in range(n):
    if cur + songs[i][0] * songs[i][1] >= t:
      print(i+1)
      break
    cur += songs[i][0] * songs[i][1]
