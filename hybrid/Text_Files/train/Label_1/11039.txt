#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a, b, c, d;
    cin >> n >> a >> b >> c >> d;
    bool possible = true;
    int max_grain = a + b;
    int min_grain = a - b;
    int max_pack = c + d;
    int min_pack = c - d;
    if (min_pack > n * max_grain || max_pack < n * min_grain) possible = false;
    cout << (possible ? "Yes" : "No") << endl;
  }
  return 0;
}
