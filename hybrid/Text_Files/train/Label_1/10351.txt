#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int h, c, T;
        cin >> h >> c >> T;
        long long num = (long long) (h - T) * c;
        long long den = (long long) (h - c) * T;
        if (T == c) {
            cout << 1 << endl;
            continue;
        }
        if (num % den == 0)
            cout << num / den << endl;
        else
            cout << num / den + 1 << endl;
    }
    return 0;
}
