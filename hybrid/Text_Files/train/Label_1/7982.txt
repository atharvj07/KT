#include <cstdio>
#include <cstring>
#include <iostream>
#include <string>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
    int n;
    cin >> n;
    vector<string> s(n);
    for (string &x : s) cin >> x;
    int sz = 20, ans = -1;
    vector<bool> vis(sz);
    for (string &x : s) {
        vis.assign(sz, false);
        for (char y : x) vis[y - 'a'] = true;
        for (int i = 0; i < sz; i++) if (!vis[i]) {
            if (ans == -1 || i < s[ans].size()) ans = i;
        }
    }
    cout << (char)(ans + 'a') << endl;
}
