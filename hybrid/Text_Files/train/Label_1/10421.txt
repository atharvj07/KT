#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, p;
        cin >> n >> p;
        vector<pair<int, int>> edges;
        for (int i = 1; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                edges.push_back({i, j});
            }
        }
        int e = 2 * n + p;
        int idx = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (e <= 0) break;
                edges.push_back({i, j});
                e--;
                idx++;
            }
        }
        for (int i = 0; i < idx; i++) {
            cout << edges[i].first << " " << edges[i].second << "\n";
        }
    }
    return 0;
}
