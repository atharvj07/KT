import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        b = list(map(int, input().split()))
        if (b[n] - b[n + 1]) % n != 0:
            print(-1)
            continue
        a = [(b[n] - b[n + 1]) // n] * n
        a[0] += b[n + 1]
        print(*a)

solve()
