MOD = 998244353

def modinv(x, mod=MOD):
    return pow(x, mod-2, mod)

def solve(n, p):
    missing = [i for i in range(1, n+1) if i not in p]
    inv_count = 0
    for i in range(n):
        if p[i] != -1:
            inv_count += len([x for x in missing if x < p[i]])
        else:
            for j in range(i+1, n):
                if p[j] != -1:
                    inv_count += 1 if missing[0] < p[j] else 0
                else:
                    inv_count += (len(missing) * (len(missing) - 1) // 2)
                    break
    
    total_missing = len(missing)
    total_perms = modinv(pow(total_missing, total_missing, MOD))
    
    result = (inv_count * total_perms) % MOD
    return result

n = int(input().strip())
p = list(map(int, input().strip().split()))

print(solve(n, p))
