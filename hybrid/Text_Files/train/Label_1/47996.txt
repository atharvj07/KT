import java.util.ArrayList;
import java.util.List;

public class DistanceSum {

    public static long solve(int n) {
        List<Integer>[] graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (gcd(i, j) == 1) {
                    graph[i].add(j);
                    graph[j].add(i);
                }
            }
        }

        boolean[] visited = new boolean[n + 1];
        long sum = 0;

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                dfs(graph, visited, i, 0, sum);
            }
        }

        return sum;
    }

    private static void dfs(List<Integer>[] graph, boolean[] visited, int node, int distance, long sum) {
        visited[node] = true;
        for (int neighbor : graph[node]) {
            if (!visited[neighbor]) {
                dfs(graph, visited, neighbor, distance + 1, sum);
                sum += distance;
            }
        }
    }

    private static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }

    public static void main(String[] args) {
        int n = 6;
        long result = solve(n);
        System.out.println(result); // Output: 8
    }
}
