#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
  int n;
  cin >> n;
  vector<vector<int>> G(n + 1);
  vector<int> deg(n + 1, 0), vis(n + 1, 0);
  for (int i = 0; i < n - 1; i++) {
    int x, y;
    cin >> x >> y;
    G[x].push_back(y), G[y].push_back(x);
    deg[x]++, deg[y]++;
  }
  vector<int> ans;
  function<void(int, int)> dfs = [&](int u, int p) {
    ans.push_back(u);
    vis[u] = 1;
    for (auto v : G[u]) {
      if (vis[v]) continue;
      dfs(v, u);
    }
  };
  for (int i = 1; i <= n; i++) {
    if (!vis[i] && deg[i] == 1) { dfs(i, i); }
  }
  for (int i : ans) { cout << i << " "; }
  cout << endl;
}
