#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<long long> prices(n);
  for (int i = 0; i < n; ++i) { cin >> prices[i]; }
  sort(prices.begin(), prices.end());
  for (int i = 0; i < q; ++i) {
    long long money;
    cin >> money;
    int left = lower_bound(prices.begin(), prices.end(), money) - prices.begin();
    cout << left << endl;
  }
  return 0;
}
