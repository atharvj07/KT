import java.util.Scanner;
class Main {
 public static void main(String[] args) {
  Scanner sc = new Scanner(System.in);
  int n = sc.nextInt();
  int a; long asum = 0;
  for(int i = 0; i < n; ++i) { a = sc.nextInt(); asum += a; }
  sc.nextLine();
  int b; long bmax = 0;
  for(int i = 0; i < n; ++i) { b = sc.nextInt(); bmax = Math.max(bmax, b); }
  System.out.println(asum <= bmax * 2 ? "YES" : "NO");
  sc.close();
 }
}
