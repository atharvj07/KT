import java.util.*;
import java.io.*;
import java.math.*;

class Main {
    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static boolean valid(int x1, int y1, int x2, int y2) {
        return Math.abs(x1 - x2) == 1 && Math.abs(y1 - y2) == 1;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            long n = scanner.nextInt();
            long m = scanner.nextInt();
            long k = scanner.nextInt();
            if ((n + m) % 2 != k % 2 || n + m > k) {
                System.out.println(-1);
            } else {
                int x = (int) Math.abs(n - m) / 2;
                int y = (int) (k - x) / 2;
                if (k - x - y != 0) {
                    System.out.println(-1);
                } else {
                    Point last = new Point(0, 0);
                    for (int i = 0; i < x; i++) {
                        if (last.x == 0 && last.y == 0 && valid(last.x, last.y, 1, 1)) {
                            last = new Point(1, 1);
                        } else if (last.x == 0 && last.y == 0 && valid(last.x, last.y, -1, 1)) {
                            last = new Point(-1, 1);
                        } else if (valid(last.x, last.y, last.x + 1, last.y)) {
                            last = new Point(last.x + 1, last.y);
                        } else if (valid(last.x, last.y, last.x - 1, last.y)) {
                            last = new Point(last.x - 1, last.y);
                        } else if (valid(last.x, last.y, last.x, last.y + 1)) {
                            last = new Point(last.x, last.y + 1);
                        } else if (valid(last.x, last.y, last.x, last.y - 1)) {
                            last = new Point(last.x, last.y - 1);
                        } else {
                            break;
                        }
                    }
                    if (last.x == n && last.y == m) {
                        System.out.println(x);
                    } else {
                        System.out.println(-1);
                    }
                }
            }
        }
    }
}
