#include <iostream>
#include <vector>
#include <utility>
#include <numeric>
#include <algorithm>
using namespace std;
int main() {
    int n; cin >> n;
    vector<pair<int, int>> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first;
        a[i].second = i;
    }
    sort(a.begin(), a.end());
    long long ans = 0;
    long long mod = 998244353;
    for (int i = 0; i < n; i++) {
        if (a[i].first == -1) {
            long long l = 0, r = n - 1;
            while (l + 1 < r) {
                long long mid = (l + r) / 2;
                if (a[mid].first != -1 && a[mid].first > a[i].second) l = mid;
                else r = mid;
            }
            ans += (l - i) % mod;
        }
    }
    cout << ans % mod <<endl;
    return 0;
}
