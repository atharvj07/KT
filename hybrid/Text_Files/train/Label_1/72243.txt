n, d = map(int, input().split())
friends = [list(map(int, input().split())) for _ in range(n)]
friends.sort(key=lambda x: x[0])

max_friendship = 0
current_friendship = 0
j = 0
for i in range(n):
    while friends[i][0] - friends[j][0] >= d:
        current_friendship -= friends[j][1]
        j += 1
    current_friendship += friends[i][1]
    max_friendship = max(max_friendship, current_friendship)

print(max_friendship)
