import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int E = in.nextInt();
        int T = in.nextInt();
        int[] x = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = in.nextInt();
        }
        int ans = 0;
        int pos = 0;
        for (int i = 0; i < N; i++) {
            ans += Math.abs(x[i] - pos);
            pos = x[i];
            ans += T;
        }
        ans += E - pos;
        System.out.println(ans);
    }
}
