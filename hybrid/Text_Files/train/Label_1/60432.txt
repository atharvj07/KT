for _ in range(int(input())):
    path = input()
    vis = [False] * len(path)
    cur, add = 0, 0
    for i, c in enumerate(path):
        if not vis[i]:
            vis[i] = True
            cur = 5
        else:
            cur = 1
        if i:
            add += cur
        if c == 'N':
            cur += add
    print(cur)
