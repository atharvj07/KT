#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int> > votes(m, vector<int>(n));
    vector<int> cities(n, 0);
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cin >> votes[i][j];
        }
    }
    for (int i = 0; i < m; i++) {
        int max = 0, idx = -1;
        for (int j = 0; j < n; j++) {
            if (votes[i][j] > max || (votes[i][j] == max && j < idx)) {
                max = votes[i][j];
                idx = j;
            }
        }
        cities[idx]++;
    }
    int max = 0, idx = -1;
    for (int i = 0; i < n; i++) {
        if (cities[i] > max || (cities[i] == max && i < idx)) {
            max = cities[i];
            idx = i;
        }
    }
    cout << idx + 1 << endl;
    return 0;
}
