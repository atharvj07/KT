#include <iostream>
#include <vector>
using namespace std;

const int N = 2e5+5;

int parent[N], tag[N], val[N], cur[N];
long long sz[N];

int find(int x) {
    if (x == parent[x]) return x;
    return parent[x] = find(parent[x]);
}

void unite(int a, int b) {
    a = find(a); b = find(b);
    if (a == b) return;
    if (sz[a] < sz[b]) swap(a, b);
    parent[b] = a;
    sz[a] += sz[b];
    tag[a] += tag[b];
    tag[a] %= 2;
}

int32_t main() {
    ios::sync_with_stdio(0); cin.tie(0);
    int n, q, m; cin >> n >> q >> m;
    for (int i = 1; i <= n; i++) parent[i] = i, sz[i] = 1;

    for (int i = 1; i <= n; i++) { cin >> val[i]; cur[i] = val[i]; }
    while (q--) {
        int t, l, r; cin >> t >> l >> r;
        if (t == 1) {
            for (int i = l; i <= r; i++) {
                int x = find(i);
                tag[x]++;
            }
        }
        if (t == 2) {
            for (int i = l; i <= r; i++) unite(i, i + 1);
        }
    }

    for (int i = 1; i <= n; i++) {
        int x = find(i);
        cur[i] = val[i];
        if (tag[x]) cur[i] = val[i + 1];
        cout << cur[i] << ' ';
    }
    cout << '\n';
    return 0;
}
