from sys import stdin
input = stdin.readline

def bfs(grid, n, x, y):
    q = [(x, y)]
    while q:
        x, y = q.pop(0)
        if grid[x][y] == 'F':
            return True
        if x > 1 and grid[x - 1][y] != '*':
            grid[x - 1][y] = '*'
            q.append((x - 1, y))
        if x < n and grid[x + 1][y] != '*':
            grid[x + 1][y] = '*'
            q.append((x + 1, y))
        if y > 1 and grid[x][y - 1] != '*':
            grid[x][y - 1] = '*'
            q.append((x, y - 1))
        if y < n and grid[x][y + 1] != '*':
            grid[x][y + 1] = '*'
            q.append((x, y + 1))
    return False

for _ in range(int(input())):
    n = int(input())
    grid = [['*'] * (n + 2) for _ in range(n + 2)]
    for i in range(1, n + 1):
        grid[i][1:n + 1] = ['S'] + list(input().strip()) + ['F']
    
    for x in range(1, n + 1):
        for y in range(1, n + 1):
            if grid[x][y] == '0':
                grid[x][y] = '*'
                if not bfs(grid, n, x, y):
                    print(1)
                    print(x, y)
                    break
                grid[x][y] = '0'
            elif grid[x][y] == '1':
                grid[x][y] = '*'
                if not bfs(grid, n, x, y):
                    print(1)
                    print(x, y)
                    break
                grid[x][y] = '1'
        else:
            continue
        break
    else:
        for x in range(1, n + 1):
            for y in range(1, n + 1):
                if grid[x][y] == '0':
                    grid[x][y] = '*'
                    if not bfs(grid, n, x, y):
                        print(2)
                        print(x, y)
                        print(x + 1, y)
                        break
                    grid[x][y] = '0'
                    break
            else:
                continue
            break
        else:
            print(0)
