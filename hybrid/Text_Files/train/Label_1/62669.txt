n, k = map(int, input().split())
a = ['0'] + list(input())
dp = {}

def get(i, j):
    if (i, j) in dp:
        return dp[(i, j)]
    if i == n:
        if j == k:
            return int(''.join(a))
        else:
            return 0
    if j == k:
        dp[(i, j)] = get(i + 1, j)
    else:
        dp[(i, j)] = (get(i + 1, j) + get(i + 1, j + 1) * 10 + int(a[i]) * pow(10, j, 1000000007)) % 1000000007
    return dp[(i, j)]

print(get(1, 0))
