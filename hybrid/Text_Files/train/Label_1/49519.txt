import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++){
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for(int i = 0; i < n; i++){
            b[i] = sc.nextInt();
        }
        Arrays.sort(b);
        StringBuilder sb = new StringBuilder();
        for(int i = 0; i < n; i++){
            sb.append((a[i] + b[i]) % n + " ");
        }
        System.out.println(sb);
    }
}
