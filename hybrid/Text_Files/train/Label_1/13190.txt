#include <algorithm>
#include <cmath>
#include <functional>
#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
typedef pair<double, double> dd;
typedef pair<ll, ll> ld;

const double EPS = 1e-9;
const double PI = acos(-1.0);
const double RAD_TO_DEG = 180.0 / PI;

int n;
vector<ld> v;

double dist(const ld &a, const ld &b) {
  ll dx = a.first - b.first;
  ll dy = a.second - b.second;
  return sqrt((double)(dx * dx + dy * dy));
}

bool ccw(const ld &a, const ld &b, const ld &c) {
  ll dx1 = b.first - a.first;
  ll dy1 = b.second - a.second;
  ll dx2 = c.first - b.first;
  ll dy2 = c.second - b.second;
  return dx1 * dy2 - dx2 * dy1 < 0;
}

double convexHull() {
  sort(v.begin(), v.end());
  vector<ld> up, down;
  v.push_back(v[0]);
  for (int i = 0; i < n; i++) {
    while (up.size() >= 2 && ccw(up[up.size() - 2], up[up.size() - 1], v[i])) up.pop_back();
    up.push_back(v[i]);
  }
  for (int i = n - 1; i >= 0; i--) {
    while (down.size() >= 2 && ccw(down[down.size() - 2], down[down.size() - 1], v[i])) down.pop_back();
    down.push_back(v[i]);
  }
  up.pop_back();
  down.pop_back();
  vector<ld> hull;
  merge(up.begin(), up.end(), down.begin(), down.end(), back_inserter(hull));
  n = hull.size();
  v = hull;
  double ans = 1e99;
  for (int i = 0; i < n; i++) {
    for (int j = 1; j < n; j++) {
      if (i == j || i == j - 1 || j == i - 1) continue;
      double r = dist(v[i], v[j]) / 2.0;
      double x = (v[i].first + v[j].first) / 2.0;
      double y = (v[i].second + v[j].second) / 2.0;
      bool ok = true;
      for (int k = 0; k < n; k++) {
        if (k == i || k == j) continue;
        double d = dist(ld(x, y), v[k]);
        if (d > r + EPS) {
          ok = false;
          break;
        }
      }
      if (ok) ans = min(ans, r);
    }
  }
  return ans;
}

int main() {
  cin >> n;
  v.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
    if (v[i].second < 0) {
      cout << -1.0 << endl;
      return 0;
    }
  }
  double ans = convexHull();
  if (ans < 1e98)
    cout << ans << endl;
  else
    cout << -1.0 << endl;
  return 0;
}
