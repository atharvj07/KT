#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a[i] = x;
    }
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        b[i] = x;
    }
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    reverse(b.begin(), b.end());
    int sum = 0;
    bool flag = false;
    for (int i = 0; i < n; i++) {
        if (!flag) {
            if (a[i] > b[i]) {
                sum += a[i];
                flag = true;
            } else {
                sum += b[i];
                flag = false;
            }
        } else {
            sum += b[i];
            flag = false;
        }
    }
    cout << sum << endl;
    return 0;
}
