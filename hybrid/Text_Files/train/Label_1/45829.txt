import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = scanner.nextInt();
        }
        scanner.close();

        // Map the target positions to their counts
        Map<Integer, Integer> target = new HashMap<>();
        for (int i = 0; i < n; i++) {
            target.put(t[i], target.getOrDefault(t[i], 0) + 1);
        }

        // Create a list of operations
        List<Operation> operations = new ArrayList<>();

        // Iterate over the stones and move them to their target positions
        for (int i = 0; i < n; i++) {
            // If the stone is already in its target position, continue
            if (s[i] == t[i]) {
                continue;
            }
            
            // Find the target position of the stone
            int targetPosition = t[i];

            // If the stone is to the left of its target position, move it right
            if (s[i] < targetPosition) {
                int d = (targetPosition - s[i]) / 2;
                operations.add(new Operation(i + 1, i + 2, d));
                s[i] += 2 * d;
                s[i + 1] -= 2 * d;
            }

            // If the stone is to the right of its target position, move it left
            else if (s[i] > targetPosition) {
                int d = (s[i] - targetPosition) / 2;
                operations.add(new Operation(i + 1, i + 2, d));
                s[i] -= 2 * d;
                s[i + 1] += 2 * d;
            }
        }

        // Check if the final positions match the target positions
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                System.out.println("NO");
                return;
            }
        }

        // Output the number of operations and the operations
        System.out.println("YES");
        System.out.println(operations.size());
        for (Operation operation : operations) {
            System.out.println(operation.i + " " + operation.j + " " + operation.d);
        }

    }
    
    private static class Operation {
    	int i, j, d;

   		public Operation(int i, int j, int d) {
       		this.i = i;
       		this.j = j;
       		this.d = d;
   		}
	}
}
