#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  vector<long long> dp(n + 1, 0);
  sort(a.begin(), a.end());
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1] + 1LL * i * a[i - 1];
  }
  for (int k = 1; k <= n; k++) {
    int s = max(k - m, 0);
    dp[k] = min(dp[k], dp[s] + 1LL * (k - s) * a[s]);
  }
  for (int i = 1; i <= n; i++) {
    cout << dp[i] << " ";
  }
  cout << endl;
  return 0;
}
