import java.util.*;

class Main {
    static final int INF = 1 << 29;

    static int dist(int[] d, int a, int b) {
        return d[a] == INF || d[b] == INF ? INF : d[a] + d[b];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int s = sc.nextInt() - 1;
        int t = sc.nextInt() - 1;
        List<Integer>[] g = new List[n];
        for (int i = 0; i < n; i++) {
            g[i] = new ArrayList<>();
        }
        int[] d = new int[n];
        Arrays.fill(d, INF);
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            g[a].add(b);
            g[b].add(a);
        }
        Queue<Integer> q = new LinkedList<>();
        d[s] = 0;
        q.add(s);
        while (!q.isEmpty()) {
            int v = q.poll();
            for (int u : g[v]) {
                if (d[u] > d[v] + 1) {
                    d[u] = d[v] + 1;
                    q.add(u);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (!g[i].contains(j) && dist(d, i, t) == dist(d, j, t) && dist(d, i, s) == dist(d, j, s)) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
