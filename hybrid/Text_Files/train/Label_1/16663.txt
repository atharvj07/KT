#include <iostream>
#include <deque>

using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    deque<long long> dq;
    for (int i = 0; i < n; i++) {
        long long a;
        cin >> a;
        dq.push_back(a);
    }
    while (q--) {
        long long m;
        cin >> m;
        for (int i = 0; i < m; i++) {
            long long a = dq.front();
            dq.pop_front();
            long long b = dq.front();
            dq.pop_front();
            if (a > b) {
                dq.push_front(b);
                dq.push_back(a);
            } else {
                dq.push_front(a);
                dq.push_back(b);
            }
        }
        cout << dq.front() << " " << dq.back() << endl;
    }
    return 0;
}
