#include<bits/stdc++.h>
using namespace std;
int main(){
    long long int t,n,i,ans;
    string s;
    cin>>t;
    while(t--){
        ans=0;
        cin>>n;
        cin>>s;
        for(i=0;i<n;i++){
            if(s[i]=='*' && (i+1<n && s[i+1]=='.') )
                ans++;
        }
        cout<<ans<<"\n";
    }
    return 0;
}
