#include <bits/stdc++.h>
#define vi  vector<int>
#define fi(i,x,y) for(int i=x;i<y;i++)
typedef long long ll;
using namespace std;

int f[505][2005];
int z[505];

int main() {
	string s, p; cin >> s >> p;
	int n = s.length(), m = p.length();
	fi(i,0,m) z[i] = -1;
    z[0] = 0;
    int l = 0, r = 0;
    fi(i,1,m) {
        if(i <= r)
            z[i] = min(z[i - l + r], r - i + 1);
        while(i + z[i] < m && p[z[i]] == p[z[i] + i]) z[i]++;
        if(i + z[i] - 1 > r)
            l = i, r = i + z[i] - 1;
    }
    fi(j,0,n) {
        fi(i,0,m) f[i][j] = -1;
    }
    f[0][0] = 0;
    fi(i,0,m) {
        fi(j,0,n) {
            f[i][j + 1] = max(f[i][j + 1], f[i][j]);
            int nw = j;
            if(i > 0) nw += z[i];
            if(nw < n && s[nw] == p[i]) f[i + 1][nw + 1] = max(f[i + 1][nw + 1], f[i][j] + 1);
        }
    }
	fi(i,0,n+1) cout << f[m][i] << ' ';
    return 0;
}
