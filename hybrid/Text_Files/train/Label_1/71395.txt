m = int(input())
p = list(map(int, input().split()))
a = [1] * m
b = [0] * m
for i in range(m):
    a[i] = p[i]
    b[i] = 1
for i in range(m):
    for j in range(i + 1, m):
        if p[j] % p[i] == 0:
            a[i] *= p[j]
            b[i] += 1
ans = 1
for i in range(m):
    ans *= pow(a[i], b[i], 1000000007)
print(ans)
