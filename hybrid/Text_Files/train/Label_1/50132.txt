import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

class Main {
    private static class UnionFind {
        private Map<Integer, Integer> parent;
        private Map<Integer, Integer> rank;

        public UnionFind(int size) {
            parent = new HashMap<>();
            rank = new HashMap<>();
            for (int i = 0; i < size; i++) {
                parent.put(i, i);
                rank.put(i, 0);
            }
        }

        public int find(int x) {
            if (parent.get(x) != x) {
                parent.put(x, find(parent.get(x)));
            }
            return parent.get(x);
        }

        public void union(int a, int b) {
            int rootA = find(a);
            int rootB = find(b);
            if (rootA != rootB) {
                if (rank.get(rootA) > rank.get(rootB)) {
                    parent.put(rootB, rootA);
                } else if (rank.get(rootA) < rank.get(rootB)) {
                    parent.put(rootA, rootB);
                } else {
                    parent.put(rootB, rootA);
                    rank.put(rootA, rank.get(rootA) + 1);
                }
            }
        }

        public boolean sameComponent(int a, int b) {
            return find(a) == find(b);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int q = Integer.parseInt(line[2]);
        UnionFind uf = new UnionFind(n);
        Map<Integer, List<Integer>> adjList = new HashMap<>();
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]) - 1;
            int b = Integer.parseInt(line[1]) - 1;
            uf.union(a, b);
            adjList.computeIfAbsent(a, k -> new ArrayList<>()).add(b);
            adjList.computeIfAbsent(b, k -> new ArrayList<>()).add(a);
        }
        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int a = Integer.parseInt(line[1]) - 1;
            if (type == 1) {
                System.out.println(dfs(a, -1, uf, adjList, new int[n]));
            } else {
                int b = Integer.parseInt(line[2]) - 1;
                uf.union(a, b);
                List<Integer> toVisit = new ArrayList<>();
                toVisit.add(a);
                toVisit.add(b);
                while (!toVisit.isEmpty()) {
                    int current = toVisit.remove(0);
                    for (int neighbor : adjList.getOrDefault(current, new ArrayList<>())) {
                        if (!uf.sameComponent(current, neighbor)) {
                            uf.union(current, neighbor);
                            toVisit.add(current);
                            toVisit.add(neighbor);
                        }
                    }
                }
            }
        }
    }

    private static int dfs(int current, int parent, UnionFind uf, Map<Integer, List<Integer>> adjList, int[] visited) {
        if (visited[current] != 0) {
            return visited[current];
        }
        visited[current] = 1;
        int max = 0;
        for (int neighbor : adjList.getOrDefault(current, new ArrayList<>())) {
            if (neighbor != parent) {
                int result = dfs(neighbor, current, uf, adjList, visited);
                if (result > max) {
                    max = result;
                }
            }
        }
        return visited[current] = max + 1;
    }
}
