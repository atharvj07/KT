import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.nextLine().trim();
        String t = sc.nextLine().trim();
        List<Integer> ans = solve(s, t);
        int k = ans.size() >> 1;
        if (k == 0)
            System.out.println(-1);
        else {
            System.out.println(k);
            for (int i = 0; i < ans.size(); i += 2) {
                System.out.println(ans.get(i) + " " + ans.get(i + 1));
            }
        }
    }

    static List<Integer> solve(String s, String t) {
        if (s.equals(t))
            return List.of();
        int n = s.length();
        List<Integer> idx0 = IntStream.range(0, n).filter(i -> s.charAt(i) != t.charAt(i)).filter(i -> s.charAt(i) == 'a').boxed().collect(Collectors.toList());
        List<Integer> idx1 = IntStream.range(0, n).filter(i -> s.charAt(i) != t.charAt(i)).filter(i -> s.charAt(i) == 'b').boxed().collect(Collectors.toList());
        if (idx0.size() != idx1.size())
            return List.of();
        List<Integer> ans = new LinkedList<>();
        for (int i = 0; i < idx0.size(); i++) {
            ans.add(idx0.get(i) + 1);
            ans.add(idx1.get(i) + 1);
        }
        return ans;
    }
}
