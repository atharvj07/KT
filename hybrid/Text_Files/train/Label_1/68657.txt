n=list(input())
mn=-1
for i in range(len(n)-1):
    if int(n[i])%2==0:
        ind=i
        break
if ind==-1:
    for i in range(len(n)-2):
        if int(n[i])%5==0:
            ind=i
            mn='1'
            break
if ind!=-1:
    for i in range(ind+1,len(n)):
        if int(n[i])%5==0:
            first=n[ind]
            second=n[i]
            n[ind]=second
            n[i]=first
            if int(''.join(n))%25==0:
                if i-ind-1+len(str(int(''.join(n))))-len(str(int(n)))+mn=='1':
                    ans=i-ind
                else:
                    ans=i-ind-1+len(str(int(''.join(n))))-len(str(int(n)))
            else:
                ans=i-ind-1
            break
print(ans if ans!='1' else mn)
