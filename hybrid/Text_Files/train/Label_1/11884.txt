#include <cstdio>
#include <vector>
#include <utility>
using namespace std;
int g[9][9],deg[9],d[40][10],ans;
int main(){
	int n,m,a,b,i,x,y,j,k;
	scanf("%d%d",&n,&m);
	while(m--){
		scanf("%d%d",&a,&b);
		g[a][b]=g[b][a]=1;
		deg[a]++;deg[b]++;
	}
	for(i=1;i<=6;i++)
		for(j=1;j<=6;j++)
			if(d[i][j]==0&&i!=j){
				for(k=1;k<=n;k++)
					if(g[k][i]&&g[k][j]){
						if(g[k][i]>g[k][j])x=i,y=j; else x=j,y=i;
						if(deg[x]%2==0&&deg[y]%2==0)d[i][j]=d[j][i]=-1;
						else if(deg[x]%2==1&&deg[y]%2==1){
							ans++;d[i][j]=1;
						}else d[i][j]=d[j][i]=-1;
					}
			}
	printf("%d\n",ans);
	return 0;
}
