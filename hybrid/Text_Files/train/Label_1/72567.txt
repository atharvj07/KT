for _ in range(int(input())):
    a = input()
    b = list(a)
    for i in b:
        if i != '0' and int(a) % int(i) != 0:
            for j in range(int(b[-1]) + 1, 10):
                if int(a) % j == 0:
                    b[-1] = str(j)
                    break
            break
    print(*b, sep='')
