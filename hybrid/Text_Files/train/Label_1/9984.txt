#include <iostream>
using namespace std;
int main(){
    int n, p=0, ans=0;
    cin >> n;
    int a[n];
    for(int i=0; i<n; i++){
        cin >> a[i];
        p+= (a[i]==1);
    }
    for(int i=0; i<n; i++){
        if(a[i]==1) p--;
        else p++;
        ans=max(ans, min(p, i%2?n-i:i+1));
    }
    cout << ans << endl;
    return 0;
}
