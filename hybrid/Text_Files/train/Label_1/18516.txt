#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll OO=1e18;
const int N=1e4+5;
ll a[N];
int main()
{
    ll n,i,ans,mx,mn,sum;
    cin>>n;
    for(i=1;i<=n;i++)cin>>a[i];
    sort(a+1,a+n+1);
    if(n==1){cout<<1<<endl;return 0;}
    if(n==2)
    {
        if(a[2]-a[1]<=1){cout<<1<<endl;return 0;}
        cout<<2<<endl;
        return 0;
    }
    mx=a[n];
    mn=a[1];
    sum=0;
    if(mx-mn<=1)
    {
        ans=0;
        for(i=1;i<=n;i++)
        {
            ans=max(a[i],ans);
        }
        cout<<ans<<endl;
        return 0;
    }
    ans=0;
    for(i=1;i<=n;i++)
    {
        if(sum+a[i]<=mx)
        {
            sum+=a[i];
            ans++;
        }
    }
    cout<<ans<<endl;
    return 0;
}
