import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[sc.nextInt()] = i;
        }
        long[] dp = new long[n + 2];
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            int j = i + 2;
            while (p[j - 2] < p[i] && p[j - 1] > p[i] && j <= n + 1) {
                ans += dp[j - 1];
                j++;
            }
            dp[i + 1] = ans;
        }
        System.out.println(ans);
    }
}
