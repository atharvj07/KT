#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans[n];
  int cur = 0;
  for (int i = n - 1; i >= 0; --i) {
    if (s[i] == 'U')
      cur++;
    else
      cur--;
    ans[i] = cur;
  }
  for (int i = 0; i < n; ++i) {
    if (ans[i] <= 0)
      ans[i] = -1;
  }
  for (int i = 0; i < n; ++i) {
    cout << ans[i] << ' ';
  }
  return 0;
}
