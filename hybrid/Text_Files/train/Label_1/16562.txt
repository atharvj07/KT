#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    int n,x,ans=0;
    cin>>n>>x;
    set<int>st;
    while(n--)
    {
        int p;
        cin>>p;
        st.insert(p);
    }
    auto it=st.begin();
    while(it!=st.end() && *it<=x)
    {
        ans+=1;
        it++;
    }
    cout<<ans<<endl;
}
