#include <iostream>
#include <string>
#include <vector>
using namespace std;
typedef long long ll;
vector<int> fact(21);
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    string s;
    cin >> s;
    int n = s.size();
    fact[0] = 1;
    for (int i = 1; i <= 20; i++) fact[i] = i * fact[i - 1];
    for (int i = 0; i < n;) {
        int c = 0;
        while (i < n && s[i] == '0') { ++i; ++c; }
        if (i == n) { cout << "red\n"; continue; }
        int cnt[10] = {};
        int sum = 0;
        for (; i < n && s[i] != '0'; i++) { ++sum; ++cnt[s[i] - '0']; }
        int ans = sum;
        for (int j = 2; j <= sum; j += 2) ans -= fact[j];
        for (int j = 0; j < 10; j++) ans -= fact[cnt[j]];
        if (ans % 60 == 0)
            cout << "red\n";
        else
            cout << "cyan\n";
    }
    return 0;
}
