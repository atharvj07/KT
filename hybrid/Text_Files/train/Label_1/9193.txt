#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> b = a;
    sort(b.begin(), b.end());
    for (int i = 1; i < n; i++) {
        if (a[i - 1] + a[i] % 2) {
            int p = i + 1;
            while (p < n && a[i - 1] + a[p] % 2) p++;
            reverse(a.begin() + i, a.begin() + p + 1);
            i = p;
        }
    }
    for (int i = 1; i < n; i++) {
        if (a[i - 1] < a[i]) continue;
        int l = i - 1;
        int r = i;
        while (r < n && a[r] != b[l]) { r++; }
        reverse(a.begin() + i, a.begin() + r + 1);
    }
    for (int i = 0; i < n; i++) { cout << a[i] << " "; }
}
