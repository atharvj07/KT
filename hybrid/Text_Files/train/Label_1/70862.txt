import queue
import sys

n, m, s, t = list(map(int, input().split()))
l = [sys.maxsize for i in range(n)]
l[s - 1] = 0
g = [[] for i in range(n)]
for i in range(m):
    a, b, d = list(map(int, input().split()))
    g[a - 1].append((b - 1, d))
q = queue.Queue()
q.put((0, s - 1))
while not q.empty():
    dist, node = q.get()
    for x, w in g[node]:
        if dist + w < l[x]:
            l[x] = dist + w
            q.put((dist + w, x))
def dfs(node, ans):
    if node == t - 1:
        return ans
    can = ans
    for x, w in g[node]:
        if l[node] + w == l[x]:
            can = dfs(x, min(can, w) - 1)
            if can != ans:
                return can
    return ans
for x, w in g[s - 1]:
    if l[s - 1] + w == l[x]:
        can = dfs(x, w - 1)
        if can != w - 1:
            print("CAN", can)
            continue
for x, w in g[t - 1]:
    if l[t - 1] + w == l[x]:
        print("CAN", w - 1)
        continue
for i in range(m):
    if l[s - 1] + l[t - 1] == l[i]:
        print("YES")
        continue
    if l[s - 1] + w + l[t - 1] == l[i]:
        print("CAN", w - 1)
        continue
    print("NO")
