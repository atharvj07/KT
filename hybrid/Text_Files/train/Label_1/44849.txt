import java.io.*;
import java.util.*;

class Main {
    static class Ptr {
        int pos;
        int dir;

        public Ptr(int pos, int dir) {
            this.pos = pos;
            this.dir = dir;
        }

        public void move() {
            pos += dir;
        }
    }

    static class State {
        Ptr ptr;
        char ch;

        public State(Ptr ptr, char ch) {
            this.ptr = ptr;
            this.ch = ch;
        }
    }

    static void simulate(char[] s, int l, int r, int[] cnt) {
        Stack<State> stack = new Stack<>();
        stack.push(new State(new Ptr(l - 1, 1), s[l - 1]));
        while (!stack.isEmpty()) {
            State state = stack.pop();
            Ptr ptr = state.ptr;
            char ch = state.ch;
            if (ch >= '0' && ch <= '9') {
                cnt[ch - '0'] += ptr.dir;
                ptr.move();
                if (ch > '0') {
                    s[ptr.pos] = (char) (ch - 1);
                } else {
                    s[ptr.pos] = ' ';
                    r--;
                }
            } else {
                if (ch == '<') {
                    ptr.dir = -1;
                } else {
                    ptr.dir = 1;
                }
                if (ptr.pos + ptr.dir >= l && ptr.pos + ptr.dir <= r) {
                    ptr.move();
                    stack.push(new State(ptr, s[ptr.pos]));
                }
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int q = Integer.parseInt(line[1]);
        char[] s = br.readLine().toCharArray();
        int[][] cnt = new int[q][10];
        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int l = Integer.parseInt(line[0]);
            int r = Integer.parseInt(line[1]);
            simulate(s, l, r, cnt[i]);
        }
        for (int i = 0; i < q; i++) {
            for (int j = 0; j < 10; j++) {
                System.out.print(cnt[i][j] + " ");
            }
            System.out.println();
        }
    }
}
