#include <bits/stdc++.h>
using namespace std;
vector<int>factors[101];
int main(){
    int n;cin>>n;
    for(int i=1;i<=100;i++){
        for(int j=i<<1;j<=100;j+=i)
            factors[j].push_back(i);
    }
    int a[n+1];
    long long ans=0;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        ans+=a[i];
    }
    for(int i=1;i<=100;i++){
        vector<int>ind;
        int sum=0;
        for(int j=1;j<=n;j++){
            if(a[j]%i==0){
                ind.push_back(j);
                sum+=a[j];
            }
        }
        int sz=ind.size();
        if(sz>1){
            sort(ind.begin(),ind.end());
            for(int j=0;j<sz-1;j++){
                ans-=a[ind[j]];
                ans+=i;
            }
        }
    }
    cout<<ans<<endl;
}
