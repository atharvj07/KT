#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, t;
  cin >> n >> t;
  long long time[n + 1];
  for (int i = 0; i < n; i++) cin >> time[i];
  long long last = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (last + 1 < time[i]) {
      last = max(last + 1, time[i] - 1);
    } else {
      last = time[i];
      ans++;
      if (last >= t) break;
    }
  }
  cout << ans << endl;
}
