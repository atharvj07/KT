import java.util.Arrays;
import java.util.Scanner;

public class ExcitingAtoms {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            d[i] = scanner.nextInt();
        }

        int[] e = new int[n];
        for (int i = 0; i < n - 1; i++) {
            e[i] = i + 1;
        }

        // Find the maximum energy gain for each possible change of E_i
        int[] maxEnergyGains = new int[n];
        for (int i = 0; i < n - 1; i++) {
            maxEnergyGains[i] = calculateMaxEnergyGain(a, d, e, i);
        }

        // Sort the energy gains in descending order
        Arrays.sort(maxEnergyGains);

        // Select the K largest energy gains
        int totalEnergyGain = 0;
        for (int i = 0; i < k; i++) {
            totalEnergyGain += maxEnergyGains[n - 1 - i];
        }

        System.out.println(totalEnergyGain);
    }

    // Calculate the maximum energy gain for a given change of E_i
    private static int calculateMaxEnergyGain(int[] a, int[] d, int[] e, int i) {
        // Create a copy of the E array
        int[] eCopy = Arrays.copyOf(e, e.length);

        // Change E_i to a different value
        eCopy[i] = (i + 2) % (e.length - 1);

        // Calculate the energy gain for exciting atom i and all its connected atoms
        int energyGain = 0;
        boolean[] visited = new boolean[e.length];
        dfs(i, eCopy, a, d, visited, energyGain);

        return energyGain;
    }

    // Depth-first search to calculate the energy gain for exciting atom i and its connected atoms
    private static void dfs(int i, int[] e, int[] a, int[] d, boolean[] visited, int energyGain) {
        if (visited[i]) {
            return;
        }

        visited[i] = true;
        energyGain += a[i] - d[i];

        int next = e[i];
        dfs(next, e, a, d, visited, energyGain);
    }
}
