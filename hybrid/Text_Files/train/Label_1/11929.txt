#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> occ(26, 0);
  string s;
  cin >> s;
  for (char c : s) { occ[c - 'a']++; }
  for (int i = 0; i < k; i++) {
    for (char c = 'a'; c <= 'z'; c++) {
      if (occ[c - 'a'] > 0) {
        cout << c;
        occ[c - 'a']--;
        break;
      }
    }
  }
  cout << endl;
  return 0;
}
