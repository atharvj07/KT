n = int(input())
x0, y0 = map(int, input().split())

closest_pieces = {'up': None, 'down': None, 'left': None, 'right': None, 'up_left': None, 'up_right': None, 'down_left': None, 'down_right': None}

for _ in range(n):
    piece, x, y = input().split()
    x, y = int(x), int(y)
    
    if x == x0:
        if y > y0:
            if closest_pieces['up'] is None or y < closest_pieces['up'][1]:
                closest_pieces['up'] = (piece, y)
        else:
            if closest_pieces['down'] is None or y > closest_pieces['down'][1]:
                closest_pieces['down'] = (piece, y)
    elif y == y0:
        if x > x0:
            if closest_pieces['right'] is None or x < closest_pieces['right'][0]:
                closest_pieces['right'] = (piece, x)
        else:
            if closest_pieces['left'] is None or x > closest_pieces['left'][0]:
                closest_pieces['left'] = (piece, x)
    elif abs(x - x0) == abs(y - y0):
        if x > x0 and y > y0:
            if closest_pieces['up_right'] is None or (x < closest_pieces['up_right'][0] and y < closest_pieces['up_right'][1]):
                closest_pieces['up_right'] = (piece, x, y)
        elif x > x0 and y < y0:
            if closest_pieces['down_right'] is None or (x < closest_pieces['down_right'][0] and y > closest_pieces['down_right'][1]):
                closest_pieces['down_right'] = (piece, x, y)
        elif x < x0 and y > y0:
            if closest_pieces['up_left'] is None or (x > closest_pieces['up_left'][0] and y < closest_pieces['up_left'][1]):
                closest_pieces['up_left'] = (piece, x, y)
        else:
            if closest_pieces['down_left'] is None or (x > closest_pieces['down_left'][0] and y > closest_pieces['down_left'][1]):
                closest_pieces['down_left'] = (piece, x, y)

in_check = False
for direction, piece_info in closest_pieces.items():
    if piece_info is not None:
        piece = piece_info[0]
        if direction in ['up', 'down', 'left', 'right'] and piece in ['R', 'Q']:
            in_check = True
            break
        elif direction in ['up_left', 'up_right', 'down_left', 'down_right'] and piece in ['B', 'Q']:
            in_check = True
            break

print("YES" if in_check else "NO")
