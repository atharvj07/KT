import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String sL = sc.next();
        String sR = sc.next();
        BigInteger L = new BigInteger(sL, 2);
        BigInteger R = new BigInteger(sR, 2);
        BigInteger X = R.xor(R.add(BigInteger.ONE)).and(R.subtract(L).add(BigInteger.ONE));
        System.out.println(X.toString(2));
    }
}
