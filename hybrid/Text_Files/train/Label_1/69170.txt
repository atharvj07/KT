n = int(input())
interesting_minutes = list(map(int, input().split()))

watch_time = 0
last_interesting_minute = 0

for minute in interesting_minutes:
    if minute - last_interesting_minute > 15:
        watch_time = last_interesting_minute + 15
        break
    else:
        watch_time = minute
    last_interesting_minute = minute

if 90 - last_interesting_minute > 15:
    watch_time = last_interesting_minute + 15

print(min(watch_time, 90))
