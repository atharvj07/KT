#include <bits/stdc++.h>
using namespace std;
#define mp make_pair
#define X first
#define Y second
#define pii pair<int,int>
#define pb push_back
typedef long long ll;
const int MAXN = 2000 + 5;
const ll mod = 1e9 + 7;
int n, k;
char a[MAXN][MAXN];
int main() {
    cin >> n >> k;
    for(int i = 1; i <= n; ++i) for(int j = 1; j <= n; ++j) cin >> a[i][j];
    int ans = n + n;
    for(int i = 1; i <= n - k + 1; ++i) {
        int cnt = 0;
        for(int j = 1; j <= n; ++j) if(a[i][j] == 'B') ++cnt;
        if(cnt == 0) ++ans, ans += (a[i][n] == 'B');
    }
    for(int j = 1; j <= n - k + 1; ++j) {
        int cnt = 0;
        for(int i = 1; i <= n; ++i) if(a[i][j] == 'B') ++cnt;
        if(cnt == 0) ++ans, ans += (a[n][j] == 'B');
    }
    cout << ans << endl;
    return 0;
}
