import java.util.*;
class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] w = new int[2*n];
        for(int i=0;i<2*n;i++) w[i] = sc.nextInt();
        Arrays.sort(w);
        int min = Integer.MAX_VALUE;
        for(int i=0;i<=n-1;i++){
            min = Math.min(min, w[2*n-2 - i] - w[i]);
        }
        System.out.println(min);
    }
}
