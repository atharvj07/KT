import sys

cases = int(sys.stdin.readline())
for _ in range(cases):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    if sum(a) == 0 or sum(a) == n:
        print(-1)
    else:
        idx = 0
        path = [n + 1]
        while idx < n:
            if a[idx] == 0:
                path.append(idx + 1)
            elif a[idx] == 1:
                path.insert(0, n + 1 - idx)
            idx += 1
        print(" ".join(str(x) for x in path))
