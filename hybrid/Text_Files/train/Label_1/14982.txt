#include <iostream>
#include <limits>

int main() {
    long long x, y, m;
    std::cin >> x >> y >> m;
    if (x < 0 && y < 0)
        std::cout << "-1" << std::endl;
    else if (m == 0 && (x | y) == 0)
        std::cout << "-1" << std::endl;
    else if (std::abs(x) + std::abs(y) < m)
        std::cout << "-1" << std::endl;
    else {
        int n = 0;
        while (x < m || y < m) {
            if (x < m)
                x += y;
            else
                y += x;
            ++n;
        }
        std::cout << n << std::endl;
    }
    return 0;
}
