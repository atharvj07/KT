#include <algorithm>
#include <vector>
#include <cmath>
#include <iostream>

using namespace std;

const long long MOD = 1e9 + 7;

long long power(long long base, long long exponent) {
    long long result = 1;
    while (exponent > 0) {
        if (exponent % 2 == 1) {
            result = (result * base) % MOD;
        }
        base = (base * base) % MOD;
        exponent /= 2;
    }
    return result;
}

int main() {
    int n;
    cin >> n;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<int> indegree(n);
    for (int i = 0; i < n; i++) {
        indegree[a[i] - 1]++;
    }

    long long num_cycles = 0;
    for (int i = 0; i < n; i++) {
        if (indegree[i] == 0) {
            num_cycles++;
        }
    }

    long long answer = power(2, n - num_cycles) % MOD;
    cout << answer << endl;

    return 0;
}
