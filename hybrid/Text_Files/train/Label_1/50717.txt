import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);
        int n = stdin.nextInt();
        int k = stdin.nextInt();
        String s = stdin.next();
        int gi = 0, ti = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'G') gi = i;
            else if (s.charAt(i) == 'T') ti = i;
        }
        System.out.println((gi + k) % n == ti || (ti + k) % n == gi ? "YES" : "NO");
    }
}
