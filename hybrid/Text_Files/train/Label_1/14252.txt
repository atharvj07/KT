#include <bits/stdc++.h>
using namespace std;
string s;
int sum() {
    int sum = 0;
    for (char c : s)
        sum += c - '0';
    return sum;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, s;
        cin >> n >> s;
        s = s - sum();
        int ans = s / 9;
        s = s % 9;
        if (s != 0)
            ans++;
        if (sum() > 0)
            ans++;
        cout << ans << endl;
    }
    return 0;
}
