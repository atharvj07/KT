import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] a = sc.nextLine().toCharArray();
        char[] b = sc.nextLine().toCharArray();
        int n = a.length;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                if (a[i] == '4') {
                    a[i] = '7';
                } else {
                    a[i] = '4';
                }
                ans++;
            }
        }
        System.out.println(ans);
    }
}
