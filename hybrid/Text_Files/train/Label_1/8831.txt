#include <iostream>
#include <limits>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> dp(123456, numeric_limits<int>::max());
        dp[0] = 0;
        bool found = false;
        int best = -1;
        for (int i = 0; i <= n; i++) {
            if (dp[i] == numeric_limits<int>::max()) continue;
            if (i + k + 1 <= n) dp[i + k + 1] = min(dp[i + k + 1], dp[i] + 1);
            int sum = i;
            for (int j = i + 1; j <= i + k; j++) {
                while (sum % 10 > 0) sum /= 10;
                sum += j;
            }
            if (sum <= n) {
                dp[sum] = min(dp[sum], dp[i] + 1);
                if (sum == n) found = true, best = i;
            }
        }
        if (found) cout << best << endl;
        else cout << -1 << endl;
    }
    return 0;
}
