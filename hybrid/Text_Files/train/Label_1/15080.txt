#include <iostream>
const int mod = 1e9 + 7;

long long power(long long base, long long exp) {
  long long res = 1;
  while (exp) {
    if (exp & 1) res = res * base % mod;
    base = base * base % mod;
    exp >>= 1;
  }
  return res % mod;
}

int main() {
  int n;
  std::cin >> n;
  std::cout << power(2, n * n - n) << std::endl;
}
