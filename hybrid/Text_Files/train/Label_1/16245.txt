#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            cnt++;
        }
    }
    if (cnt >= 2) {
        cout << "cslnb";
    } else {
        cout << "sjfnb";
    }
    return 0;
}
