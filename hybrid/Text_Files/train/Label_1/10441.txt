#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int N = 501;

int a[N][N], vis[N][N], ans[N][N];
int dx[8] = {0, 1, 1, 1, 0, -1, -1, -1};
int dy[8] = {1, 1, 0, -1, -1, -1, 0, 1};

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> p(n + 1);
    for (int i = 1; i <= n; ++i) {
        int x;
        cin >> x;
        p[x] = {i, i};
        a[i][i] = x;
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = i + 1; j <= n; ++j) {
            int x = p[i].first + j - i;
            int y = p[i].second + j - i;
            a[i][j] = a[j][i] = a[x][y];
        }
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (vis[i][j]) continue;
            vector<pair<int, int>> q = {{i, j}};
            int cur = a[i][j];
            while (!q.empty()) {
                int x = q.back().first;
                int y = q.back().second;
                q.pop_back();
                if (vis[x][y]) continue;
                vis[x][y] = 1;
                ans[x][y] = cur;
                for (int k = 0; k < 8; ++k) {
                    int nx = x + dx[k];
                    int ny = y + dy[k];
                    if (1 <= nx && nx <= n && 1 <= ny && ny <= n && a[nx][ny] == cur) {
                        q.push_back({nx, ny});
                    }
                }
            }
        }
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= i; ++j) {
            cout << ans[i][j] << " ";
        }
        cout << endl;
    }
}
