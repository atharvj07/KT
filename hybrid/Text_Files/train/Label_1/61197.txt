import sys
input = sys.stdin.readline

n = int(input())
v = [int(i) for i in input().split()]
u = sorted(v)
vv, uu = [0]*n, [0]*n
for i in range(n):
    vv[i] = v[i] + (vv[i-1] if i-1>=0 else 0)
    uu[i] = u[i] + (uu[i-1] if i-1>=0 else 0)
m = int(input())
for i in range(m):
    t, l, r = map(int, input().split())
    print(vv[r-1] if l==1 else vv[r-1]-vv[l-2], uu[r-1] if l==1 else uu[r-1]-uu[l-2])
