#include <bits/stdc++.h>
using namespace std;

int seen[1024];
int main() {
	int n;
	cin >> n;
	while (n--) {
		string c;
		cin >> c;
		int mask = 0;
		for (char x : c) {
			mask |= 1 << (x - 'a');
		}
		seen[mask] = 1;
	}
	int ans = 0;
	for (int i = 0; i < 1024; i++)
		ans += seen[i];
	cout << ans << endl;
}
