import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        boolean[] dp = new boolean[2001];
        dp[0] = true;

        for (int i = 0; i < n; i++) {
            for (int j = 2000; j >= a[i]; j--) {
                dp[j] |= dp[j - a[i]];
            }
        }

        if (!dp[1]) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        Stack<Pair> stack = new Stack<>();

        int sum = 0;
        for (int i = n - 1; i >= 0; i--) {
            sum += a[i];
            if (sum % k == 0) {
                stack.push(new Pair(i + 1, -1));
                sum /= k;
            }
        }

        while (!stack.isEmpty()) {
            Pair p = stack.pop();
            if (p.y == -1) {
                for (int i = p.x - 1; i >= 0; i--) {
                    if (dp[sum - a[i]]) {
                        stack.push(new Pair(i + 1, p.x));
                        sum -= a[i];
                        break;
                    }
                }
            } else {
                System.out.println(a[p.x] + " " + a[p.y]);
            }
        }
    }

    static class Pair {
        int x, y;

        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
