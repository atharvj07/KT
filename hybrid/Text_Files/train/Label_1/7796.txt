#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n];
    unordered_map<int, int> m;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      m[a[i]]++;
    }
    int ans = 0;
    for (int i = 0; i < 101; i++) {
      if (m[i] == 0) {
        ans += i;
        break;
      }
    }
    for (int i = ans + 1; i < 101; i++) {
      if (m[i] == 0) {
        ans += i;
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
