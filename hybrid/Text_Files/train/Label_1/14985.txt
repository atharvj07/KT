#include <iostream>
#include <cstdint>

int64_t minMoves(int64_t x, int64_t y, int64_t m) {
  int64_t cnt = 0;
  while (true) {
    if (x >= m || y >= m)
      return cnt;
    int64_t sum = x + y;
    if (x < y)
      std::swap(x, y);
    ++cnt;
    x = std::min(x, sum), y = sum;
  }
  return -1;
}

int main() {
  int64_t x, y, m;
  std::cin >> x >> y >> m;
  std::cout << minMoves(x, y, m) << std::endl;
  return 0;
}
