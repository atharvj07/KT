import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();   // size of the array
        int k = in.nextInt();   // number of subsegments
        int[] a = new int[n];  // the array
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] min = new int[k];  // minimum integer on each subsegment
        for (int i = 0; i < k; i++) {
            min[i] = Integer.MAX_VALUE;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                if (i >= j && i < j + (n - 1) % k + 1) {
                    min[j] = Math.min(min[j], a[i]);
                }
            }
        }
        int max = Integer.MIN_VALUE;  // maximum integer over the k obtained minimums
        for (int i = 0; i < k; i++) {
            max = Math.max(max, min[i]);
        }
        System.out.println(max);
    }
}
