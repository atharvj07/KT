#include <iostream>
#include <vector>

using namespace std;

struct rect {
    int x1, y1, x2, y2;
};

int main() {
    int n;
    cin >> n;

    vector<rect> rects(n);
    for (int i = 0; i < n; i++) {
        cin >> rects[i].x1 >> rects[i].y1 >> rects[i].x2 >> rects[i].y2;
    }

    vector<int> colors(n);
    for (int i = 0; i < n; i++) { colors[i] = -1; }

    for (int i = 0; i < n; i++) {
        int cx = -1, cy = -1;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            if (rects[i].x1 == rects[j].x2 && rects[i].y1 <= rects[j].y2 && rects[i].y2 >= rects[j].y1) {
                if (cx == -1) cx = colors[j];
                else if (cx != colors[j]) { cx = -2; break; }
            }
            if (rects[i].x2 == rects[j].x1 && rects[i].y1 <= rects[j].y2 && rects[i].y2 >= rects[j].y1) {
                if (cx == -1) cx = colors[j];
                else if (cx != colors[j]) { cx = -2; break; }
            }
            if (rects[i].y1 == rects[j].y2 && rects[i].x1 <= rects[j].x2 && rects[i].x2 >= rects[j].x1) {
                if (cy == -1) cy = colors[j];
                else if (cy != colors[j]) { cy = -2; break; }
            }
            if (rects[i].y2 == rects[j].y1 && rects[i].x1 <= rects[j].x2 && rects[i].x2 >= rects[j].x1) {
                if (cy == -1) cy = colors[j];
                else if (cy != colors[j]) { cy = -2; break; }
            }
        }
        if (cx == -2 || cy == -2) { cout << "NO" << endl; return 0; }
        if (cx == -1) cx = 1;
        if (cy == -1) cy = 1;
        if (cx == cy) cy++;
        colors[i] = cx;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            if (rects[i].x1 == rects[j].x2 && rects[i].y1 <= rects[j].y2 && rects[i].y2 >= rects[j].y1) {
                if (colors[j] == cx) colors[j] = cy;
            }
            if (rects[i].x2 == rects[j].x1 && rects[i].y1 <= rects[j].y2 && rects[i].y2 >= rects[j].y1) {
                if (colors[j] == cx) colors[j] = cy;
            }
            if (rects[i].y1 == rects[j].y2 && rects[i].x1 <= rects[j].x2 && rects[i].x2 >= rects[j].x1) {
                if (colors[j] == cx) colors[j] = cy;
            }
            if (rects[i].y2 == rects[j].y1 && rects[i].x1 <= rects[j].x2 && rects[i].x2 >= rects[j].x1) {
                if (colors[j] == cx) colors[j] = cy;
            }
        }
    }

    cout << "YES" << endl;
    for (int i = 0; i < n; i++) { cout << colors[i] << endl; }

    return 0;
}
