#include <bits/stdc++.h>
#define ll long long
#define endl '\n'
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        sort(a.begin(), a.end());
        ll alice = 0, bob = 0;
        int i = 0, j = n - 1;
        bool aliceTurn = true;
        while (i <= j) {
            if (aliceTurn) {
                if (a[i] % 2 == 0) alice += a[i];
                i++;
            } else {
                if (a[j] % 2 == 1) bob += a[j];
                j--;
            }
            aliceTurn = !aliceTurn;
        }
        if (alice > bob) cout << "Alice" << endl;
        else if (bob > alice) cout << "Bob" << endl;
        else cout << "Tie" << endl;
    }
    return 0;
}
