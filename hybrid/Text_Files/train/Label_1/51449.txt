import java.util.*;
import java.io.*;

public class Main {
    static boolean[] wt = new boolean[11];
    static int[] ans = new int[1010];
    static boolean dfs(int idx, int n, int sum, int prev) {
        if(idx == n) {
            for(int i = 1; i <= n; i++)
                System.out.print(ans[i]+ " ");
            System.out.println();
            return true;
        }
        for(int i = 1; i <= 10; i++) {
            if(!wt[i] || i == prev)
                continue;
            ans[idx] = i;
            if(sum + i > (idx - 1) * 10) {
                if(dfs(idx + 1, n, sum + i, i))
                    return true;
            }
            if(sum > (idx - 1) * 10) {
                if(dfs(idx + 1, n, sum, i))
                    return true;
            }
        }
        return false;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = sc.nextInt();
        int cnt = 0;
        for(int i = 0; i < 10; i++) {
            wt[i + 1] = s.charAt(i) == '1';
            if(wt[i + 1])
                cnt++;
        }
        if(cnt < 3) {
            System.out.println("NO");
            return;
        }
        if(dfs(1, n, 0, 0))
            System.out.println("YES");
        else
            System.out.println("NO");
    }
}
