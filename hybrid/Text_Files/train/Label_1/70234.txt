t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().split()))
    a.sort()
    if n % 2 == 0:
        for i in range(n//2):
            print(a[i], a[n-i-1], end=' ')
    else:
        for i in range(n//2):
            print(a[i], a[n-i-1], end=' ')
        print(a[n//2])
    print()
