#include <bits/stdc++.h>
using namespace std;
int q, sz[30], ts[30], tp[30];
int main() {
  cin >> q;
  while (q--) {
    memset(sz, 0, sizeof(sz));
    memset(ts, 0, sizeof(ts));
    memset(tp, 0, sizeof(tp));
    string s, t, p;
    cin >> s >> t >> p;
    for (char c : s) sz[c - 'a']++;
    for (char c : t) ts[c - 'a']++;
    for (char c : p) tp[c - 'a']++;
    bool ok = 1;
    for (int i = 0; i < 26; i++) {
      if (tp[i] < max(0, ts[i] - sz[i])) ok = 0;
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
