#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b;
    cin >> a >> b;
    while (b % 8 == 0) {
      b /= 8;
      a /= 2;
    }
    while (b % 4 == 0) {
      b /= 4;
      a /= 2;
    }
    while (b % 2 == 0) {
      b /= 2;
      a /= 2;
    }
    if (a == b)
      cout << __builtin_popcountll(a) - __builtin_popcountll(b) << "\n";
    else
      cout << -1 << "\n";
  }
  return 0;
}
