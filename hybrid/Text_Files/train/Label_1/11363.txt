#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MOD = 1000003;

int mod_pow(int a, int b) {
    int res = 1;
    while (b > 0) {
        if (b % 2 == 1) {
            res = 1LL * res * a % MOD;
        }
        a = 1LL * a * a % MOD;
        b /= 2;
    }
    return res;
}

int main() {
    int n, c;
    cin >> n >> c;

    vector<int> dp(n + 1);
    dp[0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= min(c, i); j++) {
            dp[i] = (dp[i] + dp[i - j]) % MOD;
        }
    }

    cout << dp[n] << "\n";

    return 0;
}
