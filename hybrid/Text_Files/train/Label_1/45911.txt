import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[][] animals = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                animals[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                int max = 0;
                for (int p = Math.max(1, j - k); p <= Math.min(j, m); p++) {
                    max = Math.max(max, dp[i - 1][p]);
                }
                dp[i][j] = max + animals[i][j];
            }
        }
        int ans = 0;
        for (int i = 1; i <= m; i++) {
            ans = Math.max(ans, dp[n][i]);
        }
        System.out.println(ans);
    }
}
