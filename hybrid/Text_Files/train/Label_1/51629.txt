import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(input.readLine());
        for (int i = 1; i <= t; i++) {
            Scanner scanner = new Scanner(input.readLine());
            int n = scanner.nextInt();
            int k = scanner.nextInt();
            scanner = new Scanner(input.readLine());
            char[] top = scanner.next().toCharArray();
            scanner = new Scanner(input.readLine());
            char[] middle = scanner.next().toCharArray();
            scanner = new Scanner(input.readLine());
            char[] bottom = scanner.next().toCharArray();
            int j;
            boolean canWin = true;
            outer:
            for (int m = 1; m <= n; m++) {
                int trainLeft = m;
                int trainTop = n + 1;
                int trainBottom = n + 1;
                for (j = 1; j <= k; j++) {
                    int trainLength = 1;
                    while (j <= k && top[m] == top[j]) {
                        trainLength++;
                        j++;
                    }
                    j--;
                    if (trainLeft > m - trainLength) {
                        trainLeft = m - trainLength;
                    }
                    if (trainBottom <= m + trainLength) {
                        trainBottom = m + trainLength;
                    }
                    j++;
                }
                for (j = 1; j <= k; j++) {
                    int trainLength = 1;
                    while (j <= k && middle[m] == middle[j]) {
                        trainLength++;
                        j++;
                    }
                    j--;
                    if (trainLeft > m - trainLength) {
                        trainLeft = m - trainLength;
                    }
                    if (trainBottom <= m + trainLength) {
                        trainBottom = m + trainLength;
                    }
                    j++;
                }
                for (j = 1; j <= k; j++) {
                    int trainLength = 1;
                    while (j <= k && bottom[m] == bottom[j]) {
                        trainLength++;
                        j++;
                    }
                    j--;
                    if (trainLeft > m - trainLength) {
                        trainLeft = m - trainLength;
                    }
                    if (trainBottom <= m + trainLength) {
                        trainBottom = m + trainLength;
                    }
                    if (trainTop >= m + trainLength) {
                        trainTop = m + trainLength;
                    }
                    j++;
                }
                if (trainLeft <= 0 || trainTop <= n || trainBottom > n) {
                    canWin = false;
                    break outer;
                }
            }
            System.out.println(canWin ? "YES" : "NO");
        }
    }
}
