n, m = map(int, input().split())
a = sorted([int(x) for x in input().split()])
b = sorted([int(x) for x in input().split()])
ans = 0
i = 0
j = 0
while i < n and j < m:
    if (a[i] + b[j]) % 2:
        ans += 1
        i += 1
        j += 1
    elif a[i] < b[j]:
        i += 1
    else:
        j += 1
print(ans)
