import sys

def solve():
    for _ in range(int(sys.stdin.readline())):
        n, k = map(int, sys.stdin.readline().split())
        graph = [[] for _ in range(n+1)]
        for _ in range(n-1):
            a, b = map(int, sys.stdin.readline().split())
            graph[a].append(b)
            graph[b].append(a)

        def dfs(u, p):
            sz = 1
            for v in graph[u]:
                if v != p:
                    sz += dfs(v, u)
            return sz

        def dfs2(u, p, dep):
            sz = 1
            cnt = 0
            for v in graph[u]:
                if v != p:
                    cur, lcnt = dfs2(v, u, dep+1)
                    if cur == 0: cnt += 1
                    sz += cur
            if dep != 0:
                cnt += max(0, sz-k)
            return sz - min(sz, k), cnt

        print(dfs2(1, 0, 0)[1])

solve()
