import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int q = input.nextInt();
        while (q-- > 0) {
            int n = input.nextInt();
            String s = input.next();
            String t = input.next();
            if (n % 2 == 0)
                System.out.println(s.equals(t) ? "YES" : "NO");
            else
                System.out.println(s.charAt(n / 2) == t.charAt(n / 2) ? "YES" : "NO");
        }
    }
}
