import java.util.*;

class Main {

    static class Edge {
        int to, color;

        Edge(int to, int color) {
            this.to = to;
            this.color = color;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] colors = new int[n + 1];
        ArrayList<Edge>[] edges = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            colors[i] = sc.nextInt();
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int from = sc.nextInt(), to = sc.nextInt();
            edges[from].add(new Edge(to, colors[to]));
            edges[to].add(new Edge(from, colors[from]));
        }
        int maxDiversity = 0, maxColor = 0;
        for (int i = 1; i <= n; i++) {
            Set<Integer> diversity = new HashSet<>();
            for (Edge edge : edges[i]) {
                diversity.add(edge.color);
            }
            diversity.remove(colors[i]);
            if (diversity.size() > maxDiversity) {
                maxDiversity = diversity.size();
                maxColor = colors[i];
            }
        }
        System.out.println(maxColor);
    }
}
