import queue
n, m = map(int, input().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    x, y, z = map(int, input().split())
    adj[x].append((y, z))
    adj[y].append((x, z))
vis = [False] * (n + 1)
vis[1] = True
q = queue.Queue()
q.put(1)
ans = []
while not q.empty():
    u = q.get()
    for v, w in adj[u]:
        if not vis[v]:
            vis[v] = True
            q.put(v)
            if not w:
                ans.append((u, v, 1))
            else:
                ans.append((u, v, 0))
print(len(ans))
for u, v, w in ans:
    print(u, v, w)
