import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextLong();
        }

        long min = 0;
        boolean oddDeleted = false;
        for (long num : arr) {
            if (num % 2 == 0) {
                if (!oddDeleted) {
                    oddDeleted = true;
                    continue;
                } else {
                    min += num;
                }
            }
        }

        for (long num : arr) {
            if (num % 2 != 0) {
                if (oddDeleted) {
                    oddDeleted = false;
                    continue;
                } else {
                    min += num;
                }
            }
        }

        System.out.println(min);
    }
}
