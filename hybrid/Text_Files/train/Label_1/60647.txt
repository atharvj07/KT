from collections import defaultdict

n = int(input())
s = input()
t = input()

if s == t:
    print(0)
    exit()

ops = defaultdict(list)

for i in range(n):
    if s[i] != t[i]:
        ops[i].append((i + 1, s[i] == '0'))

# print(ops)
c = 0
for i in range(n):
    if ops[i]:
        c += 1
        for j in ops[i]:
            if j[1]:
                s = s[:j[0]-1] + '1' + s[j[0]-1:]
            else:
                s = s[:j[0]-1] + '0' + s[j[0]-1:]
        ops[i] = []

if s == t:
    print(c)
else:
    print(-1)
