#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> pref(n + 1);
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        pref[i + 1] = pref[i] + a;
    }
    long long dp[200005][3];
    dp[n][0] = 0;
    dp[n][1] = pref[n];
    dp[n][2] = pref[n];
    for (int i = n - 1; i >= 1; i--) {
        dp[i][0] = max(dp[i + 1][0], dp[i + 1][1]);
        dp[i][1] = max(dp[i][0] - pref[i], dp[i + 1][2] - pref[i]);
        dp[i][2] = dp[i + 1][1] + pref[i];
    }
    cout << dp[1][0] << endl;
}
