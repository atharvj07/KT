#include <bits/stdc++.h>
#define sz(x) ((int)(x).size())

using namespace std;
typedef long long ll;

int main() {
  string s;
  cin >> s;
  int n = sz(s);
  string t = s;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      if (i < n - 1 && s[i + 1] == '1') {
        t[i] = '1';
        t[i + 1] = '0';
      } else if (i > 0 && s[i - 1] == '1') {
        t[i] = '1';
        t[i - 1] = '0';
      }
    }
  }
  for (int i = 0; i < n - 1; i++) {
    if (t[i] == '1' && t[i + 1] == '2') {
      t[i] = '2';
      t[i + 1] = '1';
    }
  }
  cout << t << endl;
  return 0;
}
