#include <iostream>
#include <vector>
using namespace std;
int main() {
    int a, b, c, sum;
    cin >> a >> b >> c;
    sum = a + b + c;
    if (sum & 1) {
        cout << "Impossible";
        return 0;
    }
    vector<int> out{a / 2, (sum - a) / 2 - c, c};
    if (out[1] < 0) {
        cout << "Impossible";
    } else {
        cout << out[0] << ' ' << out[1] << ' ' << out[2];
    }
    return 0;
}
