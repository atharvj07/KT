#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int MOD = 1e9 + 7;

int main() {
  ll n, k;
  cin >> n >> k;
  set<ll> s;
  for (int i = 0; i < n; i++) {
    ll l, r;
    cin >> l >> r;
    s.insert(l);
    s.insert(r + 1);
  }
  vector<ll> a(s.begin(), s.end());
  ll ans = 0;
  for (int i = 0; i < (1 << k); i++) {
    ll cnt = 0;
    for (int j = 0; j < k; j++) {
      if ((i >> j) & 1) cnt++;
    }
    ll cur = 1;
    for (int j = 1; j < a.size(); j++) {
      if (cnt % 2 == 0)
        cur = (cur * (a[j] - a[j - 1])) % MOD;
      cnt--;
    }
    ans = (ans + cur) % MOD;
  }
  cout << ans << endl;
  return 0;
}
