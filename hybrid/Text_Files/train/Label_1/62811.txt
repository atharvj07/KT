from sys import stdin
input = stdin.readline

n, m = map(int, input().split())
a = [list(input().strip()) for _ in range(n)]

ans = m*n
for K in range(2, min(n, m) + 1):
    x = (n + K - 1) // K
    y = (m + K - 1) // K
    b = [['0'] * y * K for _ in range(x * K)]
    for i in range(n):
        for j in range(m):
            b[i % K + (i // K) * K][j % K + (j // K) * K] = a[i][j]
    cnt = 0
    for i in range(x * K):
        for j in range(y * K):
            if b[i][j] != b[i % K][j % K]:
                cnt += 1
    ans = min(ans, cnt)

print(ans)
