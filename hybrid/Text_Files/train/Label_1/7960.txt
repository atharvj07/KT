#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define vll vector<ll>
#define pb push_back

const int mx = 200005;

vll v[mx];
ll n, a[mx], vis[mx], sz, an;

int cmp(ll a, ll b) { return a > b; }

void dfs(ll node) {
  if (vis[node]) return;
  vis[node] = 1;
  ++sz;
  an = node;
  for (auto child : v[node]) {
    dfs(child);
  }
}

int main() {
  ll q;
  cin >> q;
  while (q--) {
    cin >> n;
    for (ll i = 1; i <= 4 * n; i++) {
      vis[i] = 0;
      v[i].clear();
    }
    for (ll i = 1; i <= 4 * n; i++) {
      cin >> a[i];
    }
    sort(a + 1, a + 4 * n + 1, cmp);
    for (ll i = 1; i <= 4 * n; i += 2) {
      v[a[i]].pb(a[i + 1]);
      v[a[i + 1]].pb(a[i]);
    }
    sz = 0, an = 0;
    dfs(1);
    if (sz < n * 4)
      cout << "NO\n";
    else {
      dfs(an);
      if (sz == n * 4)
        cout << "YES\n";
      else
        cout << "NO\n";
    }
  }
  return 0;
}
