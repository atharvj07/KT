#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = m - k, cnt = 0;
    for (int x : a) { cnt += x - 1; }
    if (cnt >= ans) cout << ans + 1 << endl;
    else cout << -1 << endl;
    return 0;
}
