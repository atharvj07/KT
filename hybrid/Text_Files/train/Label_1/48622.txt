import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] cities = new int[n][2];
        for (int i = 0; i < n; i++) {
            cities[i][0] = scanner.nextInt();
            cities[i][1] = scanner.next().charAt(0);
        }
        scanner.close();
        long cost = 0;
        int lastB = -1, lastR = -1;
        for (int[] city : cities) {
            if (city[1] == 'B') {
                if (lastB != -1) {
                    cost += city[0] - lastB;
                }
                lastB = city[0];
            } else if (city[1] == 'R') {
                if (lastR != -1) {
                    cost += city[0] - lastR;
                }
                lastR = city[0];
            }
        }
        System.out.println(cost);
    }
}
