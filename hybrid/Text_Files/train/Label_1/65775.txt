t = int(input())
for _ in range(t):
    n, H = map(int, input().split())
    weapons = list(map(int, input().split()))
    max1, max2 = sorted(weapons, reverse=True)[:2]
    cycle_damage = max1 + max2
    cycles, rem = divmod(H, cycle_damage)
    moves = cycles * 2
    if rem > 0:
        moves += 1 if rem <= max1 else 2
    print(moves)
