#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
ll h[N];
int main() {
    ll n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> h[i];
    vector<ll> L(n + 2), R(n + 2);
    stack<ll> st;
    st.push(n + 1);
    for (int i = n; i; i--) {
        while (h[st.top()] >= h[i]) st.pop();
        R[i] = st.top();
        st.push(i);
    }
    while (!st.empty()) st.pop();
    st.push(0);
    for (int i = 1; i <= n; i++) {
        while (h[st.top()] >= h[i]) st.pop();
        L[i] = st.top();
        st.push(i);
    }
    for (int i = 1; i <= n; i++)
        cout << max({h[i], h[L[i]], h[R[i]]}) << " ";
    cout << endl;
}
