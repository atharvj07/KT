t = int(input())
for _ in range(t):
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    steps = 0
    while 1 in a:
        a_next = [a[(i + n - d) % n] for i in range(n)]
        if a_next == a:
            steps = -1
            break
        a = [a[i] & a_next[i] for i in range(n)]
        steps += 1
    print(steps)
