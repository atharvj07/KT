import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        int ans = 0;
        boolean prevBlack = false;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '#') {
                if (prevBlack) {
                    ans++;
                } else {
                    prevBlack = true;
                }
            } else {
                prevBlack = false;
            }
        }
        System.out.println(ans);
    }
}
