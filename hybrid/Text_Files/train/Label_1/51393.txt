import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    static long MOD = 1000000007;
    static long countMagicNums(int b, int l, int r) {
        long ans = 0;
        for (int i = l; i <= r; i++) {
            Map<Integer, Integer> freq = new HashMap<>();
            int n = i;
            boolean valid = true;
            while (n > 0) {
                int rem = n % b;
                n /= b;
                if (rem >= b) {
                    valid = false;
                    break;
                }
                freq.put(rem, freq.getOrDefault(rem, 0) + 1);
            }
            if (!valid)
                continue;

            boolean ok = true;
            for (int d = 0; d < b; d++) {
                if (freq.getOrDefault(d, 0) % 2 != 0) {
                    ok = false;
                    break;
                }
            }

            if (ok)
                ans++;
        }
        return ans;
    }
    public static void main(String[] args) {
        Scanner scn = new Scanner(System.in);
        int q = scn.nextInt();
        for (int i = 0; i < q; i++) {
            int b = scn.nextInt();
            int l = scn.nextInt();
            int r = scn.nextInt();
            System.out.println(countMagicNums(b, l, r));
        }
    }
}
