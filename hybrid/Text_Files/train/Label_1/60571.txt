def dfs(tree, node, parent, depth, subtree_sizes, paths_count):
    subtree_sizes[node] = 1
    paths_count[0] += depth
    for child in tree[node]:
        if child != parent:
            dfs(tree, child, node, depth + 1, subtree_sizes, paths_count)
            subtree_sizes[node] += subtree_sizes[child]

def solve(n, edges):
    tree = [[] for _ in range(n + 1)]
    for u, v in edges:
        tree[u].append(v)
        tree[v].append(u)

    subtree_sizes = [0] * (n + 1)
    paths_count = [0]
    dfs(tree, 1, -1, 0, subtree_sizes, paths_count)

    total = paths_count[0]
    for node in range(1, n + 1):
        total += subtree_sizes[node] * (n - subtree_sizes[node])

    return total // 2

n = int(input().strip())
edges = [list(map(int, input().strip().split())) for _ in range(n - 1)]
print(solve(n, edges))
