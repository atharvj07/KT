#include <bits/stdc++.h>
using namespace std;
int main() {
    int a, b, l1, l2, c1 = 0, c2 = 0, ans = 0;
    cin >> l1 >> l2;
    for (int i = 1; i <= l1; i++) {
        cin >> a;
        c1 |= (1 << (a - 1));
    }
    for (int i = 1; i <= l2; i++) {
        cin >> a;
        c2 |= (1 << (a - 1));
    }
    for (int i = 1; i <= 9; i++) {
        if (c1 & (1 << (i - 1)) && c2 & (1 << (i - 1))) {
            ans = i;
            break;
        }
    }
    cout << ans << endl;
}
