import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt(), q = in.nextInt();
    int[][] mat = new int[n][m];
    for (int i = 0; i < q; i++) {
      int r = in.nextInt() - 1, c = in.nextInt() - 1;
      mat[r][c] = -1;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (mat[i][j] == -1) continue;
        if (j + 1 < m && mat[i][j + 1] == -1 && i + 1 < n && mat[i + 1][j + 1] == -1 && mat[i + 1][j] == -1) continue;
        if (j - 1 >= 0 && mat[i][j - 1] == -1 && i + 1 < n && mat[i + 1][j - 1] == -1 && mat[i + 1][j] == -1) continue;
        ans++;
      }
    }
    System.out.println(ans);
  }
}
