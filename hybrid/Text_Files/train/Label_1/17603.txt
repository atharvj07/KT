#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
#define endl '\n'
#define vi vector<ll>
#define max 500001
vi graph[max];
ll dis[max],lft[max],rgt[max],color[max],par[max],cnt,vis[max];
unordered_map<ll,ll> f;
void dfs(ll node,ll p,ll dis){
	if(vis[node]) return;
	vis[node]=1;
	par[node]=p;
	f[color[node]]=1;
	lft[node]=cnt;
	cnt++;
	for(ll x:graph[node]){
		if(x!=p) dfs(x,node,dis+1);
	}
	rgt[node]=cnt-1;
}
int main(){
	ll n,m,a,b,c,q,u,v;
	cin>>n>>m;
	for(ll i=0;i<m;i++){
		cin>>a>>b>>c;
		graph[a].push_back(b);
		graph[b].push_back(a);
		color[i+1]=c;
	}
	dfs(1,0,0);
	cin>>q;
	while(q--){
		cin>>u>>v;
		ll ans=0;
		while(u!=v){
			if(lft[u]<=lft[v] && rgt[u]>=rgt[v]){
				ans+=f[color[par[v]]];
				v=par[v];
			}
			else{
				ans+=f[color[par[u]]];
				u=par[u];
			}
		}
		cout<<ans<<endl;
	}
	return 0;
}
