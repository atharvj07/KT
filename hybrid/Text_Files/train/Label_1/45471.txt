import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            int k = in.nextInt();
            if (k == 0 || n == k) {
                System.out.println(0);
            } else if (n < k) {
                System.out.println(Math.min(k - n, n + 1));
            } else if (n > k) {
                System.out.println(Math.min(n - k - 1, 1000000 - n));
            }
        }
    }
}
