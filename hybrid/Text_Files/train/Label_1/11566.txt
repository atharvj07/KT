#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
ll a[maxn], b[maxn], dp[2][maxn];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i];
    dp[0][0] = dp[1][0] = 0;
    for (int j = 1; j <= n; j++) {
        for (int i = 0; i < 2; i++) {
            dp[i][j] = max({dp[i][j - 1], dp[i ^ 1][j - 1], dp[i ^ 1][j - 2]});
            if (i == 0) dp[i][j] = max(dp[i][j], a[j]);
            else dp[i][j] = max(dp[i][j], b[j]);
        }
    }
    cout << dp[0][n] << "\n";
}
