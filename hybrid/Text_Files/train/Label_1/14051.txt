#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, gaps = 0, ast = -1;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      if (ast == -1 && a[i] == 1) {
        ast = i;
      }
      if (i > 0 && a[i] == 1 && a[i - 1] == 0) {
        gaps++;
      }
    }
    if (ast > 0) {
      gaps++;
    }
    cout << gaps << '\n';
  }
  return 0;
}
