import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = Arrays.stream(sc.nextLine().split(" ")).mapToInt(Integer::parseInt).sorted().toArray();
        sc.close();
        if (arr[0] + arr[1] <= arr[2]) System.out.println("IMPOSSIBLE");
        else if (arr[2] == arr[3]) System.out.println("SEGMENT");
        else System.out.println("TRIANGLE");
    }
}
