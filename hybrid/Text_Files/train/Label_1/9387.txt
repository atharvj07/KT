#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    while(cin>>t)
    {
        while(t--)
        {
            int n,x;
            cin>>n>>x;
            int a[n],b[n];
            for(int i=0;i<n;i++)
            {
                cin>>a[i];
            }
            for(int i=0;i<n;i++)
            {
                cin>>b[i];
            }
           bool ok=true;
           for(int i=0;i<n;i++)
           {
             if(a[i]+b[n-i-1]>x)
             {
                 ok=false;
                 break;
             }
           }
           if(ok) cout<<"YES\n";
           else cout<<"NO\n";
        }
    }
    return 0;
}
