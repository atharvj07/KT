#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(k);
    for (int i = 0; i < k; i++) { cin >> a[i]; }
    int l = 1, r = 1e9;
    while (l <= r) {
        int cnt = 0, cur = INT_MAX;
        int m = (l + r) / 2;
        for (int j = 0; j < k; j++) {
            int f = (m * n) / a[j];
            cnt += f;
            cur = min(f, cur);
        }
        cur = m * n - cur * a[0];
        if (cur <= 0 && cnt >= n) {
            r = m - 1;
        } else if (cur > 0 || cnt < n) {
            l = m + 1;
        }
    }
    if (l > 1e9)
        cout << -1;
    else
        cout << l << "\n";
}
