import sys

n, d = map(int, input().split())
friends = []
for i in range(n):
    m, s = map(int, input().split())
    friends.append((m, s))

friends.sort(key=lambda x: x[0])

max_friendship = 0
cur_max = -1
for m, s in friends:
    if m - cur_max > d:
        max_friendship += s
        cur_max = m

print(max_friendship)
