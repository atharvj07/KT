#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        string s;
        cin>>n>>s;
        bool f=false;
        for(int j=n-2;j>=0;j--){
            if(s[j]=='A'&&s[j+1]=='P'){
                f=true;
                cout<<j+1<<endl;
                break;
            }
        }
        if(!f)cout<<0<<endl;
    }
    return 0;
}
