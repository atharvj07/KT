import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int[] ans = new int[n];
            dfs(a, 0, n - 1, ans, 0);
            for (int i = 0; i < n; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        }
    }

    public static void dfs(int[] a, int l, int r, int[] ans, int d) {
        if (l > r) {
            return;
        }
        int maxIndex = l;
        for (int i = l + 1; i <= r; i++) {
            if (a[i] > a[maxIndex]) {
                maxIndex = i;
            }
        }
        ans[maxIndex] = d;
        dfs(a, l, maxIndex - 1, ans, d + 1);
        dfs(a, maxIndex + 1, r, ans, d + 1);
    }
}
