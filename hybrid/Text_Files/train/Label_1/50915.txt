import java.util.*;
class Main {
    static int n, k;
    static int[] x;
    static long[][] dp;
    static long MOD = 1000000007;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (in.hasNext()) {
            n = in.nextInt(); 
            k = in.nextInt(); 
            x = new int[n];
            for (int i = 0; i < n; i++)
                x[i] = in.nextInt();
            Arrays.sort(x);
            dp = new long[n + 1][k + 1];
            for (int i = 0; i <= n; i++)
                Arrays.fill(dp[i], -1);
            long ans = dfs(0, 0, 0);
            if (ans != -1) {
                ans%=MOD;
                System.out.println(ans);
                System.out.println(ans);
            } else {
                System.out.println(-1);
                System.out.println(0);
            }
        }
    }
    static long dfs(int i, int p, int rem) {
        if (i == n) {
            if (p == 0)
                return 1;
            else
                return -1;
        }
        if (dp[i][rem] != -1)
            return dp[i][rem];
        long res = 0;
        if (x[i] + rem <= k)
            res = (res + dfs(i + 1, p + x[i], rem + x[i])) % MOD;
        res = (res + dfs(i + 1, p, rem)) % MOD;
        return dp[i][rem] = res;
    }
}
