import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = input.nextInt();
        }
        Arrays.sort(heights);
        int minDiff = Integer.MAX_VALUE;
        int[] result = new int[n];
        for (int i = 0; i < n; i++) {
            int[] cur = new int[n];
            System.arraycopy(heights, i, cur, 0, n - i);
            System.arraycopy(heights, 0, cur, n - i, i);
            int curDiff = Math.abs(cur[n - 1] - cur[0]);
            for (int j = 0; j < n - 1; j++) {
                curDiff = Math.max(curDiff, Math.abs(cur[j] - cur[j + 1]));
            }
            if (curDiff < minDiff) {
                minDiff = curDiff;
                result = cur;
            }
        }
        for (int i : result) {
            System.out.print(i + " ");
        }
    }
}
