import java.util.Scanner;

public class StringProcessing {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();

        StringBuilder numbers = new StringBuilder();
        StringBuilder nonNumbers = new StringBuilder();

        for (String word : s.split("[,;]")) {
            if (isInteger(word)) {
                numbers.append(word).append(",");
            } else {
                nonNumbers.append(word).append(",");
            }
        }

        String resultNumbers = numbers.length() > 0 ? numbers.substring(0, numbers.length() - 1) : "-";
        String resultNonNumbers = nonNumbers.length() > 0 ? nonNumbers.substring(0, nonNumbers.length() - 1) : "-";

        System.out.println("\"" + resultNumbers + "\"");
        System.out.println("\"" + resultNonNumbers + "\"");
    }

    private static boolean isInteger(String word) {
        if (word.isEmpty()) {
            return false;
        }

        for (int i = 0; i < word.length(); i++) {
            if (!Character.isDigit(word.charAt(i))) {
                return false;
            }
        }

        return true;
    }
}
