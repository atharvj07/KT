import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        BigInteger[] arr = new BigInteger[n];
        for (int i = 0; i < n; i++) arr[i] = BigInteger.valueOf(sc.nextInt());
        BigInteger[] pref = new BigInteger[n + 1];
        pref[0] = BigInteger.ZERO;
        for (int i = 1; i <= n; i++) pref[i] = pref[i - 1].add(arr[i - 1]);
        BigInteger[] arrows = new BigInteger[q];
        for (int i = 0; i < q; i++) arrows[i] = BigInteger.valueOf(sc.nextInt());
        BigInteger cur = pref[n], ans = BigInteger.ZERO;
        for (int i = q - 1; i >= 0; i--) {
            if (pref[n].compareTo(arrows[i]) <= 0) {
                cur = pref[n];
                ans = BigInteger.ZERO;
            } else {
                int j = helper(pref, n, arrows[i]);
                cur = cur.subtract(pref[j]).add(arr[j]);
                ans = BigInteger.valueOf(n - j);
            }
            System.out.println(ans);
        }
    }

    public static int helper(BigInteger[] arr, int n, BigInteger x) {
        int low = 0, high = n, mid, ans = 0;
        while (low <= high) {
            mid = (low + high) >> 1;
            if (arr[mid].compareTo(x) <= 0) {
                low = mid + 1;
                ans = mid;
            } else high = mid - 1;
        }
        return ans;
    }
}
