#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N = 105;
bool vis[N], ok;
ll par[N], p[N], ans;
vector<ll> g[N], rG[N];

void addEdge(ll u, ll v) {g[u].push_back(v); rG[v].push_back(u);}

void dfs1(ll node) {
    vis[node] = 1;
    for (ll i : g[node]) if(!vis[i]) dfs1(i);
}

void dfs2(ll node, ll comp) {
    vis[node] = 1;
    par[node] = comp;
    for (ll i : rG[node]) if(!vis[i]) dfs2(i, comp);
}

int32_t main() {
    ll n, u, x, y, i;
    cin >> n;
    for (i = 1; i <= n; i++) {
        cin >> x;
        addEdge(x, i);
    }
    memset(vis, 0, sizeof(vis));
    for (i = 1; i <= n; i++)
        if(!vis[i]) dfs1(i);
    memset(vis, 0, sizeof(vis));
    ll cnt = 0;
    for (i = n; i >= 1; i--)
        if(!vis[i]) dfs2(i, ++cnt);
    ok = 1;
    for (i = 1; i <= n; i++)
        if(par[par[i]] == par[i]) ok &= 1;
        else ok &= 0;
    if(ok) {
        for (i = 1; i <= n; i++) p[i] = i;
        ans = 1e9;
        do {
            ll cn = 1;
            for (i = 1; i <= n; i++) {
                u = p[i];
                if(par[u] == par[p[x]]) cn++;
                x = u;
            }
            ans = min(ans, cn);
        } while(next_permutation(p + 1, p + n + 1));
    }
    if(ok) cout << ans << endl;
    else cout << -1 << endl;
    return 0;
}
