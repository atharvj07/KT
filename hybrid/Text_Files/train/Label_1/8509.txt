#include <bits/stdc++.h>
using namespace std;
int arr[2001], brr[2001], a[2001];
int main() {
  int n, i, j, cnt = 0, flag = 0;
  cin >> n;
  for (i = 1; i <= 2 * n; i++) {
    cin >> arr[i];
    brr[i] = arr[i];
  }
  sort(brr + 1, brr + 2 * n + 1);
  for (i = 1; i <= 2 * n; i++) {
    if (arr[i] != brr[i]) {
      a[++cnt] = arr[i];
      if (cnt > 2) flag = 1;
    }
  }
  if (flag)
    cout << "-1" << endl;
  else
    cout << cnt / 2 << endl;
  return 0;
}
