import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong(), p = sc.nextLong();
        long ans = 0;
        while (n > 0) {
            if ((n & 1) != (p & 1)) {
                ans++;
                n--;
            }
            n /= 2;
            p /= 2;
        }
        System.out.println(n == 0 || p == 0 ? ans : -1);
    }
}
