import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int r = sc.nextInt();
        List<int[]> ducks = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int head = sc.nextInt();
            int tail = sc.nextInt();
            ducks.add(new int[] {head, tail});
        }
        Collections.sort(ducks, (a, b) -> a[0] - b[0]);
        int maxHit = 0;
        int shots = 0;
        int time = 0;
        for (int[] duck : ducks) {
            if (time >= duck[0] && duck[1] - duck[0] >= r) {
                int newHit = Math.min(duck[1] - duck[0], time - duck[0]) / r + 1;
                shots += newHit;
                maxHit = Math.max(maxHit, shots);
            } else {
                maxHit = Math.max(maxHit, shots);
                shots = 0;
                time = duck[0] + r;
            }
        }
        maxHit = Math.max(maxHit, shots);
        System.out.println(maxHit);
    }
}
