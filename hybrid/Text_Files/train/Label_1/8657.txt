#include <bits/stdc++.h>
using namespace std;
#define int long long
const int MM = 2e5 + 5;
int n, sz[MM];
int par[MM];
vector<int> adj[MM];
vector<tuple<int, int, int>> ans;
void dfs1(int u, int p) {
    par[u] = p;
    sz[u] = 1;
    for (int v : adj[u]) {
        if (v == p)
            continue;
        dfs1(v, u);
        sz[u] += sz[v];
    }
}
int dfs2(int u, int p, int keep) {
    int mx = 0;
    for (int v : adj[u]) {
        if (v == p)
            continue;
        mx = max(mx, dfs2(v, u, keep));
    }
    return mx + sz[u] - (u == keep ? 0 : 2 * sz[u]);
}
void dfs3(int u, int p, int keep, int dist) {
    int mx = 0, vmx = 0, vmv = 0;
    for (int v : adj[u]) {
        if (v == p)
            continue;
        int cur = dfs2(v, u, keep);
        if (cur > mx)
            mx = cur, vmx = v;
    }
    for (int v : adj[u]) {
        if (v == p || v == vmx)
            continue;
        int cur = dfs2(v, u, keep);
        if (cur > vmx)
            vmx = cur, vmv = v;
    }
    if (u != keep)
        ans.push_back({u, vmx, vmx});
    else {
        if (vmx)
            ans.push_back({u, vmx, vmx});
        if (vmv)
            ans.push_back({u, vmv, vmv});
    }
    for (int v : adj[u]) {
        if (v == p || v == vmx || v == vmv)
            continue;
        dfs3(v, u, keep, dist + sz[u] - 2 * sz[v]);
    }
}
int32_t main() {
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    dfs1(1, 1);
    int s = 0;
    for (int i = 1; i <= n; i++)
        s += sz[i] * (sz[i] - 1) / 2;
    cout << s << endl;
    dfs3(1, 1, 1, 0);
    for (auto t : ans) {
        int u, v, w;
        tie(u, v, w) = t;
        cout << u << " " << v << " " << w << endl;
    }
}
