#include <iostream>
#include <unordered_set>

using namespace std;

int main() {
    int n, a, m, b;
    cin >> n;
    unordered_set<int> s1;
    for (int i = 0; i < n; i++) {
        cin >> a;
        s1.insert(a);
    }
    cin >> m;
    unordered_set<int> s2;
    for (int i = 0; i < m; i++) {
        cin >> b;
        s2.insert(b);
    }
    for (auto &x : s1) {
        for (auto &y : s2) {
            if (s1.count(x + y) == 0 && s2.count(x + y) == 0) {
                cout << x << " " << y << endl;
                return 0;
            }
        }
    }
    return 0;
}
