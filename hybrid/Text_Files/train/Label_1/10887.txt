#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, s;
    cin >> n >> s;
    vector<int> a(n);
    for (int& x : a) {
        cin >> x;
    }
    sort(a.begin(), a.end());
    int k = 0, sum = 0;
    for (int i = 0; i < n && sum + a[i] + i * (k + 1) <= s; ++i, ++k) {
        sum += a[i] + i * k;
    }
    cout << k << ' ' << sum << '\n';
    return 0;
}
