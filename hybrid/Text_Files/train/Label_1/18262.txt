#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int &pi : p) cin >> pi;
  int v = 0;
  for (int i = 0; i < n; i++) {
    string s;
    getline(cin, s);
    int w = 1, prv = -1;
    for (int j = 0; j < s.size(); j++) {
      if (s[j] == 'a' || s[j] == 'e' || s[j] == 'i' || s[j] == 'o' || s[j] == 'u' || s[j] == 'y') {
        if (prv != j - 1) w++;
        prv = j;
      }
    }
    if (i && w != p[i - 1]) {
      v = 1;
      break;
    }
    if (w != p[i]) {
      v = 1;
      break;
    }
  }
  cout << (v ? "NO" : "YES") << '\n';
  return 0;
}
