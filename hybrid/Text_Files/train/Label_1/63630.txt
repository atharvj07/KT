t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    odd_indices = [i for i in range(n) if i % 2 == 1]
    even_indices = [i for i in range(n) if i % 2 == 0]
    odd_values = [a[i] for i in range(n) if a[i] % 2 == 1]
    even_values = [a[i] for i in range(n) if a[i] % 2 == 0]
    if len(odd_indices) != len(odd_values) or len(even_indices) != len(even_values):
        print(-1)
    else:
        odd_mismatches = sum(1 for i in range(len(odd_indices)) if a[odd_indices[i]] % 2 != odd_indices[i] % 2)
        even_mismatches = sum(1 for i in range(len(even_indices)) if a[even_indices[i]] % 2 != even_indices[i] % 2)
        print(max(odd_mismatches, even_mismatches))
