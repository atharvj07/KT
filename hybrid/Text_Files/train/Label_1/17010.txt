#include <bits/stdc++.h>
using namespace std;
const int maxN = 1e5+1;
int n, k, p;
int a[maxN];
vector<int> sol[maxN];
bool vis[maxN];
bool dfs(int sum, int cnt, int ind, int type) {
  if(cnt == 0) {
    if(type == 1) p--;
    else k--;
    if(p == 0 && k == 0) return true;
    return false;
  }
  for(int i = ind+1; i <= n; i++) {
    if(vis[i]) continue;
    vis[i] = true;
    if(dfs(sum+a[i], cnt-1, i, type)) { sol[type].push_back(a[i]); return true; }
    if(dfs(sum+a[i], cnt-1, i, 1-type)) { sol[1-type].push_back(a[i]); return true; }
    vis[i] = false;
  }
  return false;
}
signed main() {
  cin >> n >> k >> p;
  for(int i = 1; i <= n; i++) cin >> a[i];
  if(dfs(0, p, 0, 0)) {
    cout << "YES" << '\n';
    for(int i = 0; i < 2; i++) {
      cout << sol[i].size() << ' ';
      for(auto it : sol[i]) cout << it << ' ';
      cout << '\n';
    }
  }
  else cout << "NO";
  return 0;
}
