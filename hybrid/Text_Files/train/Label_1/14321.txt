#include <bits/stdc++.h>
using namespace std;
int a[1000010];
int main() {
  int n, x;
  cin >> n >> x;
  if (n == 1) {
    if (x == 0) {
      cout << "YES\n0";
    } else {
      cout << "NO";
    }
    return 0;
  }
  if (n == 2) {
    if (x % 2 == 0) {
      cout << "YES\n" << 0 << ' ' << x;
    } else {
      cout << "NO";
    }
    return 0;
  }
  if (x == n - 1) {
    cout << "YES\n";
    for (int i = 0; i < n - 1; i++) {
      cout << i << ' ';
    }
    cout << x;
    return 0;
  }
  a[0] = 0;
  a[1] = x;
  for (int i = 2; i < n; i++) {
    a[i] = i;
  }
  cout << "YES\n";
  for (int i = 0; i < n; i++) {
    cout << a[i] << ' ';
  }
  return 0;
}
