import java.util.Scanner;
import java.io.PrintWriter;

public class Main {
    static int[] getAnswer(char[] s) {
        int n = s.length;
        int[] ans = new int[n];
        int balance = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') balance++;
            else if (s[i] == ')') balance--;
            else {
                ans[i] = -balance;
                balance++;
            }
        }
        if (balance == 0) return ans;
        else return new int[]{-1};
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        char[] s = in.next().toCharArray();
        int[] ans = getAnswer(s);
        if (ans.length == 1 && ans[0] == -1) out.println(-1);
        else {
            for (int x : ans) {
                out.println(x);
            }
        }

        in.close();
        out.close();
    }
}
