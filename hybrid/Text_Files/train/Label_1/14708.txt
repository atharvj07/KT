#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string n;
    cin >> n;
    int ans = -1;
    if (n.back() % 2 == 0) ans = 0;
    else if (n.front() % 2 == 0) ans = 1;
    else {
      int pref = 0;
      for (int i = 0; i < n.size(); i++) {
        if ((n[i] - '0') % 2 == 0) {
          pref = i;
          break;
        }
      }
      if (pref == 0) {
        ans = -1;
      } else {
        reverse(n.begin(), n.begin() + pref + 1);
        ans = 2;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
