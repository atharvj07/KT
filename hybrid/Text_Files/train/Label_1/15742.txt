#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int sum = 0;
    for (int i = 0; i < n; ++i) {
      int x;
      cin >> x;
      sum += x;
    }
    int diff = sum == 0 ? 1 : sum < 0 ? -sum : 1 - sum;
    cout << diff << endl;
  }
  return 0;
}
