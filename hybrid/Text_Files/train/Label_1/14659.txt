#include <bits/stdc++.h>
using namespace std;
int main(){
    long long int a, b, i=1;
    cin>>a>>b;
    while(a--){
        if(i%2==1) cout<<"B";
        else cout<<"G";
        i++;
    }
    while(b--){
        if(i%2==0) cout<<"B";
        else cout<<"G";
        i++;
    }
    cout<<endl;
    return 0;
}
