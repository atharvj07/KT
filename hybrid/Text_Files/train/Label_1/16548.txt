#include <algorithm>
#include <iostream>
#include <string>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string a;
    std::cin >> a;
    std::sort(a.begin(), a.end());
    a.erase(std::unique(a.begin(), a.end()), a.end());
    if (a.find("trygub") == std::string::npos) {
      std::cout << a << '\n';
    } else {
      a.erase(a.find('t'), 1);
      a.erase(a.find('r'), 1);
      a.erase(a.find('y'), 1);
      std::cout << a << '\n';
    }
  }
  return 0;
}
