#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mod=1e9+7;
#define endl "\n"
#define pb push_back

int main() {
    ll n;
    cin>>n;
    vector<pair<ll,ll>>v;
    for(ll i=0;i<n;++i) {
        ll a;
        cin>>a;
        v.pb({a,0});
    }
    for(ll i=0;i<n;++i) {
        ll a;
        cin>>a;
        v[i].second=a;
    }
    sort(v.begin(),v.end());
    ll ans=0;
    for(ll i=0;i<n;++i) {
        for(ll j=i+1;j<n;++j) {
            ll a=v[i].first;
            ll b=v[j].first;
            ll c=v[i].second;
            ll d=v[j].second;
            if((c-d)*(b-a)>0) {
                ans+=abs(d*a-c*b)/(d-c);
            }
        }
    }
    cout<<ans<<endl;
}
