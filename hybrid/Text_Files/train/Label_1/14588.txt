#include <iostream>
#include <vector>
using namespace std;
#define ll long long
const ll mod = 1e9+7;
ll doll[]={1,2,5,10,20,50,100}, eur[]={5,10,20,50,100,200};
ll dp[100100];
int main(){
  ll n,d,e; cin>>n>>d>>e;
  for(int i=30; i<=100; i++)
    dp[i]=1e18;
  dp[0]=0;
  for(int i=30; i<=n; i++){
    for(ll cur: doll){
      if(i-cur>=0)
        dp[i]=min(dp[i], 1+dp[i-cur]);
    }
    for(ll cur: eur){
      if(i-cur>=0)
        dp[i]=min(dp[i], 1+dp[i-cur]);
    }
  }
  cout<<dp[n]<<endl;
  return 0;
}
