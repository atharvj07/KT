import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt(), y = sc.nextInt(), z = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[x], b = new int[y], c = new int[z];
        for (int i = 0; i < x; i++) a[i] = sc.nextInt();
        for (int i = 0; i < y; i++) b[i] = sc.nextInt();
        for (int i = 0; i < z; i++) c[i] = sc.nextInt();

        List<Integer> list = new ArrayList<>();
        for (int i = 0; i < x; i++) {
            for (int j = 0; j < y; j++) {
                for (int l = 0; l < z; l++) {
                    list.add(a[i] + b[j] + c[l]);
                }
            }
        }
        Collections.sort(list, Collections.reverseOrder());
        for (int i = 0; i < Math.min(list.size(), k); i++) System.out.println(list.get(i));
    }
}
