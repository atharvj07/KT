#include <bits/stdc++.h>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    int n = t.size(), cnt = 0;
    map<char, int> mp;
    for (char c : s) mp[c]++;
    for (char c : t) cnt += mp[c];
    if (cnt < s.size())
        return cout << -1, 0;
    int mn = s.size() / n + (s.size() % n != 0);
    return cout << mn << endl << t, 0;
}
