#include <bits/stdc++.h>
using namespace std;
struct node {
  int val, pos;
  bool operator<(const node &a) const { return val > a.val; }
};
int main() {
  int n, q;
  cin >> n >> q;
  priority_queue<node> pq;
  for (int i = 0, x; i < n; i++) {
    cin >> x;
    pq.push({x, i});
  }
  while (q--) {
    int s;
    cin >> s;
    bool flag = false;
    vector<int> ans;
    while (s > 0 && !pq.empty()) {
      auto p = pq.top();
      pq.pop();
      if (s >= p.val) {
        s -= p.val;
        ans.push_back(p.pos + 1);
      }
    }
    if (s == 0) {
      sort(ans.begin(), ans.end());
      cout << ans.size() << "\n";
      for (int x : ans) {
        cout << x << " ";
      }
      cout << "\n";
    } else {
      cout << -1 << "\n";
    }
  }
}
