#include <cstdio>
#include <vector>
#include <cstring>
#include <algorithm>
#include <string>
using namespace std;
const int maxn = 110;
char mp[30];
char ord[maxn][maxn];
int main() {
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; ++i) scanf("%s", ord[i]);
    int len = strlen(ord[0]);
    for (int i = 0; i < 26; ++i) mp[i] = 'a' + i;
    bool flag = 0;
    for (int i = 0; i < 26 && !flag; ++i) {
        for (int j = i + 1; j < 26 && !flag; ++j) {
            for (int kase = 0; kase < n && !flag; ++kase) {
                for (int ll = 0; ll + 1 < len && !flag; ++ll) {
                    int x, y;
                    x = find(mp, mp + 26, *(ord[kase] + ll)) - mp;
                    y = find(mp, mp + 26, *(ord[kase] + ll + 1)) - mp;
                    if (x > y) {
                        flag = 1;
                        break;
                    }
                }
            }
            if (flag) continue;
            swap(mp[i], mp[j]);
            int tmp[maxn][maxn];
            for (int x = 0; x < n; ++x)
                for (int y = 0; y < len; ++y) tmp[x][y] = ord[x][y];
            for (int x = 0; x < n; ++x) {
                for (int y = 0; y < len; ++y) {
                    *(ord[x] + y) = tmp[x][y];
                    for (int z = 0; z < 26; ++z)
                        if (mp[z] == *(ord[x] + y)) { (*(ord[x] + y)) = 'a' + z; break; }
                }
            }
            for (int kase = 0; kase < n; ++kase) {
                for (int ll = 0; ll + 1 < len; ++ll) {
                    int x, y;
                    x = find(mp, mp + 26, *(ord[kase] + ll)) - mp;
                    y = find(mp, mp + 26, *(ord[kase] + ll + 1)) - mp;
                    if (x > y) { flag = 1; break; }
                }
                if (flag) break;
            }
        }
    }
    if (flag) printf("Impossible"); else {
        for (int i = 0; i < 26; ++i) printf("%c", mp[i]);
    }
    return 0;
}
