t = int(input())

for _ in range(t):
    a, b, c = map(int, input().split())
    if a == b == c:
        print("YES")
        print(a, b, c)
    else:
        if a <= b:
            a, b = b, a
        if a <= c:
            a, c = c, a
        if b <= c:
            b, c = c, b
        if a == b and b == c:
            print("YES")
            print(a, b, c)
        else:
            print("NO")
