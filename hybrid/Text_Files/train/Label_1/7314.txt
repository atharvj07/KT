#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<vector<int>> adj(n);
  vector<int> in(n);
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    a--;
    b--;
    adj[a].push_back(b);
    in[b]++;
  }

  vector<int> topo;
  for (int i = 0; i < n; i++) {
    if (in[i] == 0) {
      topo.push_back(i);
    }
  }

  for (int i = 0; i < n; i++) {
    int u = topo[i];
    for (int v : adj[u]) {
      in[v]--;
      if (in[v] == 0) {
        topo.push_back(v);
      }
    }
    if (topo.size() == n) {
      cout << i + 1 << endl;
      return 0;
    }
  }

  cout << -1 << endl;
  return 0;
}
