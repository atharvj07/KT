import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<String> strings = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            strings.add(sc.next());
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int k = sc.nextInt();
            String permutation = sc.next();
            Map<Character, Integer> permutationMap = new HashMap<>();
            for (int j = 0; j < permutation.length(); j++) {
                permutationMap.put(permutation.charAt(j), j);
            }
            Collections.sort(strings, (s1, s2) -> {
                for (int j = 0; j < Math.min(s1.length(), s2.length()); j++) {
                    int order1 = permutationMap.get(s1.charAt(j));
                    int order2 = permutationMap.get(s2.charAt(j));
                    if (order1 != order2) {
                        return order1 - order2;
                    }
                }
                return s1.length() - s2.length();
            });
            System.out.println(strings.indexOf(strings.get(k - 1)) + 1);
        }
    }
}
