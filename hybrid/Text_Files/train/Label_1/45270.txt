import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    private static final int MOD = 1000000007;
    private static int[] color;
    private static long[] val;
    private static long[][] dp;
    private static List<Integer>[] adj;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] p = new int[n + 1];
        String[] pLine = br.readLine().split(" ");
        for (int i = 2; i <= n; i++) {
            p[i] = Integer.parseInt(pLine[i - 2]);
        }
        int q = Integer.parseInt(br.readLine());
        adj = new List[n + 1];
        color = new int[n + 1];
        val = new long[n + 1];
        dp = new long[n + 1][60];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 2; i <= n; i++) {
            adj[i].add(p[i]);
            adj[p[i]].add(i);
        }
        for (int i = 1; i <= q; i++) {
            String[] qLine = br.readLine().split(" ");
            int type = Integer.parseInt(qLine[0]);
            if (type == 1) {
                int v = Integer.parseInt(qLine[1]);
                long x = Long.parseLong(qLine[2]);
                long k = Long.parseLong(qLine[3]);
                dfs(v, -1, 0, x, k);
            } else {
                int v = Integer.parseInt(qLine[1]);
                System.out.println(val[v]);
            }
        }
    }

    private static void dfs(int u, int parent, int depth, long x, long k) {
        color[u] = 1;
        val[u] = (val[u] + dp[depth][0] % MOD) % MOD;
        dp[depth + 1][0] = (dp[depth + 1][0] + x) % MOD;
        dp[depth + 1][1] = (dp[depth + 1][1] + x - k) % MOD;
        for (int v : adj[u]) {
            if (v == parent) {
                continue;
            }
            if (color[v] == 1) {
                continue;
            }
            dfs(v, u, depth + 1, x, k);
        }
        for (int i = 2; i < 60; i++) {
            dp[depth][i] = (dp[depth][i] + dp[depth][i - 1] - k * dp[depth][i - 1] % MOD + MOD) % MOD;
        }
        color[u] = 2;
        dp[depth][0] = 0;
        dp[depth][1] = 0;
    }
}
