#include <string>
#include <iostream>
using std::cin, std::cout, std::string;
int main() {
  string a, b, v;
  cin >> a >> b >> v;
  int la = a.length(), lb = b.length(), lv = v.length();
  int dp[la + 1][lb + 1];
  for (int i = 0; i <= la; i++) dp[i][0] = 0;
  for (int i = 1; i <= lb; i++) dp[0][i] = 0;
  for (int i = 1; i <= la; i++) {
    for (int j = 1; j <= lb; j++) {
      if (a[i - 1] == b[j - 1] && !v.empty() && v[0] == a[i - 1]) {
        for (int k = 1; k < lv && i + k - 1 < la && j + k - 1 < lb &&
                 v[k] == a[i + k - 1] && v[k] == b[j + k - 1];
             k++)
          dp[i][j] = std::max(dp[i][j], dp[i - k][j - k] + k);
      } else if (a[i - 1] == b[j - 1])
        dp[i][j] = dp[i - 1][j - 1] + 1;
      else
        dp[i][j] = std::max(dp[i - 1][j], dp[i][j - 1]);
    }
  }
  if (dp[la][lb] == 0) {
    cout << 0;
  } else {
    string ans;
    int i = la, j = lb;
    while (i && j) {
      if (a[i - 1] == b[j - 1]) {
        ans += a[i - 1];
        i--, j--;
      } else {
        if (dp[i][j] == dp[i - 1][j])
          i--;
        else
          j--;
      }
    }
    for (int i = ans.length() - 1; i >= 0; i--) cout << ans[i];
  }
  cout << "\n";
}
