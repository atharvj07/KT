#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    long long sum = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
    }
    vector<long long> b(n);
    for (int i = 0; i < n; i++) {
      b[i] = sum / n;
    }
    long long diff = sum % n;
    for (int i = 0; i < diff; i++) {
      b[i]++;
    }
    for (int i = 0; i < n; i++) {
      cout << b[i] << " ";
    }
    cout << endl;
  }
  return 0;
}
