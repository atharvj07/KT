#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        long long sum = 0;
        for (auto& x : a) {
            sum += x;
        }
        long long ans = 0;
        for (int mask = 0; mask < (1 << n); mask++) {
            long long cur = 0;
            for (int i = 0; i < n; i++) {
                if (mask & (1 << i)) {
                    cur += a[i];
                }
            }
            if (cur == sum - 1) {
                ans++;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
