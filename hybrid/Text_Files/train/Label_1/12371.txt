#include <bits/stdc++.h>
using namespace std;
#define MOD 998244353
#define ll long long
const int N = 1005;
ll dp[N][N], f[N], g[N];
int main()
{
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; ++i)
        cin >> dp[i + 1][1];
    for (int len = 2; len <= k; ++len)
        for (int i = 1; i + len - 1 <= n; ++i)
            dp[i][len] = min(dp[i][len - 1], dp[i + 1][len - 1]);
    f[0] = 1;
    for (int i = 1; i <= n; ++i)
        f[i] = f[i - 1] * (i - 1) % MOD;
    g[n] = 1;
    for (int i = n - 1; i >= 1; --i)
        g[i] = g[i + 1] * (n - i) % MOD;
    ll ans = 0;
    for (int i = 1; i + k - 1 <= n; ++i)
        ans = (ans + dp[i][k] * f[k - 1] % MOD * g[i + k] % MOD) % MOD;
    cout << ans << endl;
    return 0;
}
