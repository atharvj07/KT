#include<iostream>
#include<algorithm>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,x;
        cin>>n>>x;
        int a[n],b[n];
        for(int i=0;i<n;i++) cin>>a[i];
        for(int i=0;i<n;i++) cin>>b[i];
        sort(b,b+n);
        bool ok=1;
        for(int i=0;i<n;i++){
            if(a[i]+b[i]>x){
                ok=0;
                break;
            }
        }
        if(ok) cout<<"Yes\n";
        else cout<<"No\n";
    }
}
