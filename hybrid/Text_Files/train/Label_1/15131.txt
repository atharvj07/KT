#include <bits/stdc++.h>
using namespace std;

#define pb push_back
#define vi vector<int>
#define ll long long
#define vll vector<long long>

const int mod = 1e9 + 7;

vector<vi> g1, g2;
vector<char> e;
int n, m;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  cin >> n >> m;
  g1.resize(n + 1);
  g2.resize(n + 1);
  e.resize(m);
  while (m--) {
    char t;
    cin >> t;
    if (t == '+') {
      int u, v;
      char w;
      cin >> u >> v >> w;
      g1[u].pb(v);
      g2[v].pb(u);
      e.pb(w);
    } else if (t == '-') {
      int u, v;
      cin >> u >> v;
      for (int i = 0; i < g1[u].size(); i++) {
        if (g1[u][i] == v) {
          g1[u].erase(g1[u].begin() + i);
          break;
        }
      }
      for (int i = 0; i < g2[v].size(); i++) {
        if (g2[v][i] == u) {
          g2[v].erase(g2[v].begin() + i);
          break;
        }
      }
    } else if (t == '?') {
      int k;
      cin >> k;
      int fl = 0;
      for (int i = 1; i <= n; i++) {
        int sz = g2[i].size();
        int sz1 = g1[i].size();
        for (int j = 0; j < sz1; j++) {
          for (int l = 0; l < sz; l++) {
            if (i == g1[i][j] || i == g2[i][l]) continue;
            fl = 1;
            cout << "YES" << endl;
            break;
          }
          if (fl) break;
        }
        if (fl) break;
      }
      if (!fl) cout << "NO" << endl;
    }
  }
  return 0;
}
