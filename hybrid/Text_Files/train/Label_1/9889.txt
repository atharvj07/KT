#include <bits/stdc++.h>
using namespace std;
#define int long long
const long long MOD = 998244353;
int power(int base, int exponent) {
  base %= MOD;
  int ans = 1;
  while (exponent) {
    if (exponent & 1)
      ans = (ans * base) % MOD;
    base = (base * base) % MOD;
    exponent >>= 1;
  }
  return ans % MOD;
}
main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, m, p;
  cin >> n >> m >> p;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> b(m);
  for (int i = 0; i < m; i++) cin >> b[i];
  if (m < n) {
    m += n;
    b.resize(m);
    for (int i = n; i < m; i++) b[i] = 0;
  }
  vector<int> t(max(n, m)), s(max(n, m));
  for (int i = 0; i < n; i++) t[i] = a[i];
  for (int i = n; i < max(n, m); i++) t[i] = 0;
  for (int i = 0; i < m; i++) s[i] = b[i];
  for (int i = m; i < max(n, m); i++) s[i] = 0;
  int sum = 0;
  for (int i = 0; i < max(n, m); i++) sum += t[i] * s[i];
  if (sum % p) {
    cout << 0 << endl;
    return 0;
  }
  for (int i = 0; i < max(n, m); i++)
    if (t[i] % p || s[i] % p) {
      cout << i << endl;
      return 0;
    }
  int L = 1, R = max(n, m) - 1, po = 0;
  while (L <= R) {
    po++;
    int mid = (L + R) >> 1;
    vector<int> tt = t, ss = s;
    for (int i = mid + 1; i < max(n, m); i++) tt[i] = ss[i] = 0;
    int news = 0;
    for (int i = 0; i < max(n, m); i++) news += tt[i] * ss[i];
    if (news % p)
      L = mid + 1;
    else
      R = mid - 1;
  }
  cout << L << endl;
}
