import math
import sys

def read_int():
    return int(input())

def read_ints():
    return [int(x) for x in input().split()]

def read_f():
    return float(input())

def read_fs():
    return [float(x) for x in input().split()]

def read_str():
    return input()

def write(a):
    print(a)

def write_list(l, sep=" "):
    print(sep.join(map(str, l)))

def write_set(s, sep=" "):
    print(sep.join(map(str, s)))

def write_dict(d, sep=" "):
    print(sep.join(map(lambda x: str(x) + ":" + str(d[x]), d.keys())))

mod = int(1e9) + 7

m, n, k, t = read_ints()
a = read_ints()
traps = [read_ints() for _ in range(k)]

traps.sort()

idx = 0
ans = 0
cur = 0
for i in range(t):
    if idx < k and traps[idx][0] == i:
        while idx < k and traps[idx][0] == i:
            if cur <= traps[idx][1]:
                cur = traps[idx][1]
            idx += 1
    elif cur >= n:
        break
    else:
        agility = math.inf
        for ai in a:
            if ai > cur:
                agility = min(agility, ai)
        if agility == math.inf:
            break
        cur += 1
        ans += 1

write(ans)
