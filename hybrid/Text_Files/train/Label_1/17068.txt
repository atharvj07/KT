#include <bits/stdc++.h>
using namespace std;
#define ll long long
int n;
ll a[300001];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (ll i = 1; i <= n; i++) {
            cin >> a[i];
        }
        vector<int> ans(n + 2);
        int mx = 0;
        int mn = 1e9;
        for (ll i = 1; i <= n; i++) {
            mx = max(mx, (int)a[i]);
            mn = min(mn, (int)a[i]);
        }
        ans[n] = 1;
        if (mx != mn)
            ans[n - 1] = 0;
        else
            ans[n - 1] = 1;
        for (ll i = n - 2; i >= 1; i--) {
            ans[i] = ans[i + 1];
            if (a[i] > a[i + 1] || a[i] == a[i + 1] && a[i - 1] != a[i])
                ans[i] = 0;
        }
        for (int i = 1; i <= n; i++) {
            cout << ans[i];
        }
        cout << '\n';
        for (int i = 1; i <= n; i++) a[i] = 0;
    }
    return 0;
}
