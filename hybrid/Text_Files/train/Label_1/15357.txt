#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    int score = 0;
    for (int i = 0; i < s.size(); i++) {
        score += s[i] == s[(i + 1) % n];
        ans += score >= 4;
    }
    cout << ans << endl;
    return 0;
}
