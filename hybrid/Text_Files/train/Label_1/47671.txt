import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            int max = a[0];
            int ans = -1;
            for (int j = 1; j < n; j++) {
                if (a[j] > max) {
                    max = a[j];
                    ans = j + 1;
                }
            }
            System.out.println(ans);
        }
    }
}
