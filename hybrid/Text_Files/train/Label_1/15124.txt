#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n - 1; ++i) {
    cin >> a[i];
  }
  vector<int> next(n);
  for (int i = n - 2; i >= 0; --i) {
    next[i] = next[a[i] - 1] + 1;
  }
  long long ans = 0;
  for (int i = 0; i < n - 1; ++i) {
    ans += next[i];
  }
  cout << ans << endl;
  return 0;
}
