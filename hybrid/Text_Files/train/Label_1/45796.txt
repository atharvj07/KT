import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
        sc.close();

        Arrays.sort(arr);
        long cost = 0;
        for (int i = 1; i < n; i++) cost += arr[i];
        System.out.println(cost);
    }
}
