#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  long long ans=0;
  for (int i=0;i<n;++i) {
    cin>>v[i].first>>v[i].second;
    ans+=v[i].first*v[i].first+v[i].second*v[i].second;
  }
  double ans1=sqrt(ans);
  if (ans1>1500000) {
    for (int i=0;i<n;++i) {
      if ((long long)v[i].first*v[i].first+(long long)v[i].second*v[i].second>(long long)1500000*1500000) {
        if (v[i].first>0) v[i].first=-v[i].first;
        else v[i].second=-v[i].second;
        break;
      }
    }
  }
  for (int i=0;i<n;++i) {
    if (ans>1500000) cout<<-1<<" ";
    else {
      cout<<1<<" ";
      ans-=v[i].first*v[i].first+v[i].second*v[i].second;
    }
  }
}
