import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int[] sticks = new int[4];
    for (int i = 0; i < 4; i++) sticks[i] = s.nextInt();
    Arrays.sort(sticks);
    if (sticks[0] + sticks[1] <= sticks[2] || sticks[0] + sticks[2] <= sticks[3] || sticks[1] + sticks[2] <= sticks[3])
      System.out.println("IMPOSSIBLE");
    else if (sticks[0] == sticks[1] && sticks[1] == sticks[2] && sticks[2] == sticks[3])
      System.out.println("SEGMENT");
    else
      System.out.println("TRIANGLE");
  }
}
