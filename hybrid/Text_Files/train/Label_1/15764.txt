#include <bits/stdc++.h>
using namespace std;
int main(){
    long long n,k,l=0,r,m,sum=0,ans;
    cin>>n>>k;
    long long a[n];
    for(int i=0;i<n;i++){
        cin>>a[i];
        sum+=a[i];
    }
    if(k==1){
        cout<<fixed<<setprecision(15)<<(double)sum/n<<endl;
        return 0;
    }
    r=k-1;
    while(r<n){
        ans=sum;
        sum-=a[l];
        l++;
        r++;
        sum+=a[r];
        ans=(double)ans/(double)k;
        cout<<fixed<<setprecision(15)<<ans<<endl;
    }
}
