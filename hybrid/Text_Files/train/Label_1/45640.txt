import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt() - 1;
        }
        int[] l = new int[n];
        int[] r = new int[n];
        Arrays.fill(l, -1);
        Arrays.fill(r, -1);
        for (int i = 0; i < n; i++) {
            if (l[p[i]] == -1) {
                l[p[i]] = i;
            }
            r[p[i]] = i;
        }
        int ans = 0;
        for (int x = 0; x < n; x++) {
            int y = (l[x] == x ? r[x] : l[x]);
            if (y == -1) {
                continue;
            }
            if (p[x] < p[y]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
