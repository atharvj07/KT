#include <bits/stdc++.h>

using namespace std;

pair<vector<int>, vector<int>> query(vector<int>& v, int x, int y) {
  vector<int> u, w;
  int n = v.size();
  for (int i = 0; i < n; ++i) {
    if (i + 1 < x || y < i + 1) v[i] = 0;
    u.push_back(v[i]);
  }
  for (int i = n - 1, j = 0; i >= 0; --i, ++j) {
    if (i + 1 < x || y < i + 1) continue;
    if (!j || u[j - 1] != v[i]) w.push_back(v[i]);
  }
  reverse(begin(w), end(w));
  return {u, w};
}

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> v(n);
  iota(begin(v), end(v), 1);
  while (q--) {
    int t, p;
    cin >> t;
    if (t == 1) {
      cin >> p;
      tie(v, ignore) = query(v, p, n);
    } else {
      int l, r;
      cin >> l >> r;
      int sum = 0;
      tie(ignore, v) = query(v, l, r);
      for (int x : v) sum += x;
      cout << sum << '\n';
    }
  }
  return 0;
}
