#include <bits/stdc++.h>
#define MAX 405
using namespace std;
string arr[MAX];
bool same(int i, int j) {
  string s1 = arr[i], s2 = arr[j];
  for (int I = 0; s1[I]; I++)
    if (s1[I] == 'o' || s1[I] == 'u') s1[I] = 'a';
  for (int I = 0; s2[I]; I++)
    if (s2[I] == 'o' || s2[I] == 'u') s2[I] = 'a';
  for (int I = 0; I < s1.size(); I++) {
    if (s1[I] == 'h' || s1[I] == 'k') s1[I] = 'h';
    if (s2[I] == 'h' || s2[I] == 'k') s2[I] = 'h';
  }
  return s1 == s2;
}
int main() {
  int n, ans = 0;
  cin >> n;
  bool vis[MAX];
  for (int i = 0; i < MAX; i++) vis[i] = false;
  for (int i = 1; i <= n; i++) {
    cin >> arr[i];
    for (int j = 1; j < i; j++) {
      if (!vis[j] && same(i, j)) { vis[j] = true; ans++; break; }
    }
  }
  cout << n - ans << endl;
  return 0;
}
