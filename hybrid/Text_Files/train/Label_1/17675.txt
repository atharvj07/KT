#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, a, b;
        cin >> n >> x >> a >> b;
        if (a > b) swap(a, b);
        if (x == 0) {
            cout << b - a - 1 << '\n';
            continue;
        }
        int ans = min(x, b - a - 1);
        ans += max(a - 1, n - b);
        cout << ans << '\n';
    }
    return 0;
}
