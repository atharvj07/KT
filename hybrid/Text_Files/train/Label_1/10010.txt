#include <cstdio>
#include <iostream>
#include <string>
#include <vector>

using namespace std;

const int tri = 4, sq = 6, cube = 8, pen = 12, ico = 20;

int main() {
  int n;
  cin >> n;
  vector<int> cnt(5);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    if (s == "Tetrahedron") cnt[0]++;
    if (s == "Cube") cnt[1]++;
    if (s == "Octahedron") cnt[2]++;
    if (s == "Dodecahedron") cnt[3]++;
    if (s == "Icosahedron") cnt[4]++;
  }
  int ans = cnt[0] * tri + cnt[1] * sq + cnt[2] * cube + cnt[3] * pen + cnt[4] * ico;
  cout << ans << '\n';
  return 0;
}
