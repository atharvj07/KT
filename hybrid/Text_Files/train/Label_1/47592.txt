import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (sc.hasNext()) {
            int[] input = new int[3];
            for (int i = 0; i < 3; i++) {
                input[i] = sc.nextInt();
            }
            int maxx = input[0];
            int maxy = input[1];
            int w = input[2];

            long left = 0;
            long right = maxy * 1L * maxx;
            long bestSum = right + 1;
            long bestX = 0;
            long bestY = 0;
            while (left <= right) {
                long mid = (left + right) / 2;
                if (mid / maxy <= maxx && getLuckyNum(mid / maxy, mid % maxy) >= w) {
                    bestSum = mid;
                    bestX = mid / maxy;
                    bestY = mid % maxy;

                    right = mid - 1;
                } else {
                    left = mid + 1;
                }
            }
            if (bestSum > maxy * 1L * maxx) {
                System.out.println(-1);
            } else {
                System.out.println(bestX + " " + bestY);
            }
        }
    }

    private static long getLuckyNum(long a, long b) {
        if (a * b == reverse(a) * reverse(b)) {
            return 1;
        }
        return 0;
    }

    private static long reverse(long n) {
        long res = 0;
        while (n > 0) {
            res = res * 10 + n % 10;
            n /= 10;
        }
        return res;
    }
}
