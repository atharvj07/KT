import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int min = 0;
        for (char c : s.toCharArray()) {
            if (c == '-') {
                min--;
            } else {
                min++;
            }
        }
        System.out.println(min);
    }
}
