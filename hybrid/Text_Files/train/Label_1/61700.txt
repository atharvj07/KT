def func(n):
        a = [[None]*n for _ in range(n)]
        def pos(x,y):
                if x<0 or x>=n or y<0 or y>=n:
                        return False
                return True
        def solve(x,y,v):
                if not pos(x,y):
                        return False
                if a[x][y] != None:
                        return False
                a[x][y]=v
                if n*n==v:
                        return True
                return solve(x+1,y,v+1) or solve(x,y+1,v+1) or solve(x-1,y,v+1) or solve(x,y-1,v+1) or\
                        solve(x+1,y-1,v+1) or solve(x-1,y+1,v+1) or solve(x+1,y+1,v+1) or solve(x-1,y-1,v+1)
        solve(0,0,1)
        for i in range(n):
                for j in range(n):
                        print(a[i][j],end=" ")
                print()
n = int(input())
func(n)
