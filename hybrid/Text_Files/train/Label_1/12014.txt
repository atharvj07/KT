#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int l = 0, r = 0, ans = 1;
  while (r < n) {
    if ((r - l + 1 > ans) || ((r - l + 1 == ans) && (a[r] < a[l]))) {
      ans = r - l + 1;
    }
    if (l > 0 && (a[l - 1] >= a[r])) {
      l++;
    } else {
      r++;
    }
  }
  cout << ans << endl;
  return 0;
}
