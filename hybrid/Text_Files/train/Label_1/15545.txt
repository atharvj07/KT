#include <bits/stdc++.h>

using namespace std;

#define ll long long int

const ll INF = 1e18;

struct Edge {
  ll to, weight;
};

vector<vector<Edge>> adj_list;
vector<bool> vis;
vector<ll> max_dist;

void dfs(ll node, ll prev, ll dist) {
  vis[node] = true;
  max_dist[node] = dist;
  for (Edge &e : adj_list[node]) {
    if (!vis[e.to]) { dfs(e.to, node, max(dist, e.weight)); }
  }
}

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  ll n, m, k;
  cin >> n >> m >> k;
  vector<ll> special(k);
  for (ll i = 0; i < k; i++) { cin >> special[i]; }
  adj_list.resize(n + 1);
  vis.resize(n + 1, false);
  max_dist.resize(n + 1, 0);
  for (ll i = 0; i < m; i++) {
    ll u, v, w;
    cin >> u >> v >> w;
    adj_list[u].push_back({v, w});
    adj_list[v].push_back({u, w});
  }
  for (ll node : special) {
    vis.assign(n + 1, false);
    dfs(node, -1, 0);
    ll max_dist_node = -1;
    ll max_d = 0;
    for (ll i = 1; i <= n; i++) {
      if (max_dist[i] > max_d) {
        max_d = max_dist[i];
        max_dist_node = i;
      }
    }
    cout << max_d << " ";
  }
  cout << endl;
  return 0;
}
