import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int m = scanner.nextInt();

        List<Operation> operations = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            int z = scanner.nextInt();
            operations.add(new Operation(x, y, z));
        }

        int k = scanner.nextInt();

        long[] sequence = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            sequence[i] = a + (i - 1) * d;
        }

        for (Operation operation : operations) {
            if (operation.x == 0) {
                reverse(sequence, operation.y, operation.z);
            } else if (operation.x == 1) {
                increment(sequence, operation.y, operation.z);
            } else if (operation.x == 2) {
                halve(sequence, operation.y, operation.z);
            }
        }

        System.out.println(sequence[k]);
    }

    private static void reverse(long[] sequence, int y, int z) {
        for (int i = y, j = z; i < j; i++, j--) {
            long temp = sequence[i];
            sequence[i] = sequence[j];
            sequence[j] = temp;
        }
    }

    private static void increment(long[] sequence, int y, int z) {
        for (int i = y; i <= z; i++) {
            sequence[i]++;
        }
    }

    private static void halve(long[] sequence, int y, int z) {
        for (int i = y; i <= z; i++) {
            sequence[i] /= 2;
        }
    }

    private static class Operation {
        int x, y, z;

        public Operation(int x, int y, int z) {
            this.x = x;
            this.y = y;
            this.z = z;
        }
    }
}
