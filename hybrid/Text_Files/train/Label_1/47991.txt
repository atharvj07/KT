import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int s = scanner.nextInt();
        int[] li = new int[s];
        int[] ri = new int[s];
        int[] xi = new int[s];
        for (int i = 0; i < s; i++) {
            li[i] = scanner.nextInt();
            ri[i] = scanner.nextInt();
            xi[i] = scanner.nextInt();
        }
        int[] ans = new int[s];
        for (int i = 0; i < s; i++) {
            int l = li[i];
            int r = ri[i];
            int x = xi[i];
            int min = x;
            int max = x;
            for (int j = l; j <= r; j++) {
                if (a[j - 1] < min) {
                    min = a[j - 1];
                }
                if (a[j - 1] > max) {
                    max = a[j - 1];
                }
            }
            if (min < x) {
                ans[i] = r - l + 1;
            } else if (max > x) {
                ans[i] = 0;
            } else {
                ans[i] = r - l + 1 - (min - x);
            }
        }
        for (int i = 0; i < s; i++) {
            System.out.println(ans[i]);
        }
    }
}
