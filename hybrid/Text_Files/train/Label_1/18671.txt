#include <bits/stdc++.h>
using namespace std;
#define MAXN 100

int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    cin >> n;
    int a[MAXN];
    memset(a, 0, sizeof(a));
    for (int j = 0; j < n; j++) {
      cin >> a[j];
    }
    int sum = 0;
    for (int j = 0; j < n; j++) {
      if (a[j] % 3 == 1) {
        sum++;
      }
      if (a[j] % 3 == 2) {
        sum += 2;
      }
    }
    cout << sum << endl;
  }
  return 0;
}
