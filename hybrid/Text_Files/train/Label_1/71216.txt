import numpy as np
def main():
    n = int(input())
    oddmap = np.zeros((n,n),dtype=int)
    evenmap = np.zeros((n,n),dtype=int)
    for i in range(n):
        for j in range(n):
            if (i+j)%2==0:
                oddmap[i][j]=1
    board=['B' if oddmap[i][j]==1 else 'W' for i in range(n) for j in range(n)]
    print('\n'.join([''.join(x) for x in np.array(board).reshape(n,n)]))

main()
