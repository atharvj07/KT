#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, t, ans = 0, c = 0;
  cin >> n >> a;
  for (int i = 0; i < n; i++) { // read in input
    cin >> t;
    c += (t == 1);
  }
  t = 0;
  for (int i = 0; i < n; i++) { // solve input
    cin >> t;
    c += t;
    if (i < a) {
      ans += (c > 0);
    } else {
      ans += (c > 1);
      c -= (t == 1);
    }
  }
  cout << ans << endl;
  return 0;
}
