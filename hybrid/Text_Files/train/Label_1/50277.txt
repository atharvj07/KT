import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        if (n == 2 && m == 3) {
            System.out.println("YES");
            return;
        }
        if (n % 2 == 0) {
            n++;
        }
        while (true) {
            int d = 0;
            for (int i = 2; i <= Math.sqrt(n); i++) {
                if (n % i == 0) {
                    d++;
                    break;
                }
            }
            if (d == 0) {
                if (n == m) {
                    System.out.println("YES");
                    return;
                }
                n += 2;
            } else {
                n += 2;
            }
        }
    }
}
