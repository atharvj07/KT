n,k=[int(x) for x in input().split()]
c=[int(x) for x in input().split()]
c.sort()
ans=sum(range(1,k+1))*c[0]
t=[0]*n
for i in range(0,n):
    t[i]=k+i+1
mincost=sum(c)*k
for i in range(0,k):
    mincost+=c[i]
    if(mincost<ans):
        ans=mincost
        for j in range(0,i+1):
            t[j]=k+j+1
        for j in range(i+1,n):
            t[j]=k+n-1-j
print(ans)
print(*t)
