from collections import deque

def read_input():
    n, m, p = map(int, input().split())
    speeds = list(map(int, input().split()))
    grid = [list(input().strip()) for _ in range(n)]
    return n, m, p, speeds, grid

def solve(n, m, p, speeds, grid):
    directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]
    counts = [0] * p
    queues = [deque() for _ in range(p)]

    for i in range(n):
        for j in range(m):
            if grid[i][j].isdigit():
                player = int(grid[i][j]) - 1
                counts[player] += 1
                queues[player].append((i, j, speeds[player]))

    def valid(x, y):
        return 0 <= x < n and 0 <= y < m and grid[x][y] == '.'

    while any(queues):
        new_queues = [deque() for _ in range(p)]
        for player in range(p):
            while queues[player]:
                x, y, speed = queues[player].popleft()
                if speed == 0:
                    new_queues[player].append((x, y, speeds[player]))
                    continue
                for dx, dy in directions:
                    nx, ny = x + dx, y + dy
                    if valid(nx, ny):
                        grid[nx][ny] = str(player + 1)
                        counts[player] += 1
                        queues[player].append((nx, ny, speed - 1))
        queues = new_queues

    return counts

n, m, p, speeds, grid = read_input()
result = solve(n, m, p, speeds, grid)
print(" ".join(map(str, result)))
