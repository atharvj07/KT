#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 1e5 + 5;
int n, a[MAXN];
vector<int> l(MAXN, -1), r(MAXN, -1);

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    l[0] = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] == a[i - 1] - 1) l[i] = l[i - 1];
    }

    r[n - 1] = n - 1;
    for (int i = n - 2; i >= 0; i--) {
        if (a[i] == a[i + 1] - 1) r[i] = r[i + 1];
    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (l[i] != -1 && r[i] != -1) ans = max(ans, r[i] - l[i] + 1);
    }

    cout << ans << endl;

    return 0;
}
