#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<vector<int>> a(n, vector<int>(4));
    for (int i = 0; i < n; i++) {
        cin >> a[i][0] >> a[i][1] >> a[i][2];
    }
    sort(a.begin(), a.end());
    int x = n / 2;
    int ans = -1;
    for (int i = 0; i < x; i++) {
        int cnt1 = 0;
        int cnt2 = 0;
        int time = 0;
        for (int j = i; j < x + i; j++) {
            if (a[j][1] == 1) cnt1++;
            if (a[j][2] == 1) cnt2++;
            time += a[j][0];
        }
        if (cnt1 >= k && cnt2 >= k) {
            if (ans == -1) ans = time;
            else ans = min(ans, time);
        }
    }
    cout << ans << endl;
}
