#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7, inf = 1e18;
const int mxN = 3e5;
int n;
vector<int> adj[mxN];
vector<ll> a(mxN), dp(mxN, -1);
bool leaf[mxN], vis[mxN];
int f[mxN];
ll ans = 0;

void dfs(int u, int p) {
    leaf[u] = true;
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        leaf[u] &= leaf[v];
    }
    if (leaf[u]) a[u] = 1;
}

ll dfs2(int u, int p, bool mx) {
    if (dp[u] != -1) return dp[u];
    ll ret = 0;
    if (leaf[u]) return dp[u] = a[u];
    for (int v : adj[u]) {
        if (v == p) continue;
        if (mx)
            ret = max(ret, dfs2(v, u, mx));
        else
            ret = min(ret, dfs2(v, u, mx));
    }
    return dp[u] = a[u] + ret;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 2; i <= n; i++) { cin >> f[i]; }
    for (int i = 2; i <= n; i++) { adj[f[i]].push_back(i); }

    dfs(1, 1);
    ans = dfs2(1, 1, 1);
    ans = max(ans, dfs2(1, 1, 0));
    cout << ans << "\n";
}
