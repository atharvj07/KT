#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> h(n);
  for (int i = 0; i < n; i++) { cin >> h[i]; }
  sort(h.rbegin(), h.rend());
  int ans = 0;
  for (int i = 1; i < n; i++) { ans += max(h[i - 1] - h[i], 0); }
  if (ans > k) ans = -1;
  cout << ans << endl;
  return 0;
}
