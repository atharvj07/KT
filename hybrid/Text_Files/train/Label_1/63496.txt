def min_refuels(a, b, f, k):
    refuels = 0
    current_fuel = b
    for i in range(k):
        if i % 2 == 0:  # Going to a
            if current_fuel < f:  # Can't reach the gas station
                return -1
            current_fuel -= f
            if i == k - 1:  # Last journey, only need to reach a
                if current_fuel < a - f:
                    refuels += 1
            else:  # Need to go back to 0 as well
                if current_fuel < 2 * (a - f):
                    refuels += 1
                    current_fuel = b
                current_fuel -= (a - f)
        else:  # Going back to 0
            if current_fuel < a - f:  # Can't reach the gas station
                return -1
            current_fuel -= (a - f)
            if current_fuel < 2 * f:  # Need to be able to go to a and back to f
                refuels += 1
                current_fuel = b
            current_fuel -= f
    return refuels

a, b, f, k = map(int, input().split())
print(min_refuels(a, b, f, k))
