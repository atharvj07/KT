#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll MOD=998244353;
int main(){
  ll n; cin>>n;
  if(n==1) cout<<10<<endl;
  else{
    cout<<180<<" ";
    ll ans=288; ll cur=1;
    for(ll x=3;x<=n;x++){
      ans=(ans+cur%MOD*cur%MOD)%MOD;
      cur=(cur+cur)%MOD;
      cout<<ans<<" ";
    }
    cout<<endl;
  }
}
