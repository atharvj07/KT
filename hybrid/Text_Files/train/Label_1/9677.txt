#include <cstdio>
#include <vector>

int main() {
    int w, h, n;
    scanf("%d %d %d", &w, &h, &n);
    std::vector<int> hs(h), ws(w);
    while (n--) {
        char d;
        int p;
        scanf(" %c %d", &d, &p);
        if (d == 'H') hs[p - 1] = 1;
        if (d == 'V') ws[p - 1] = 1;
        int maxh = 0, maxw = 0;
        for (int i = 0; i < h; ++i) {
            if (!hs[i]) ++maxh;
            else maxh = 0;
        }
        for (int i = 0; i < w; ++i) {
            if (!ws[i]) ++maxw;
            else maxw = 0;
        }
        printf("%d\n", maxh * maxw);
    }
}
