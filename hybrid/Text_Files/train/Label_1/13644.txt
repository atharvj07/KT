#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '-') {
      ans--;
    } else {
      ans++;
    }
  }
  if (ans < 0) {
    ans = 0;
  }
  cout << ans << endl;
  return 0;
}
