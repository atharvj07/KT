#include <iostream>

using namespace std;

struct Seg {
    long long l, r;
};

long long n;
Seg s[100005];

int main() {
    cin >> n;
    for (long long i = 1; i <= n; i++) { cin >> s[i].l >> s[i].r; }

    long long ans = -1;
    for (long long i = 1; i <= n; i++) {
        long long ok = 1;
        for (long long j = 1; j <= n; j++) {
            if (i == j) continue;
            if (!(s[i].l <= s[j].l && s[j].r <= s[i].r)) { ok = 0; }
        }
        if (ok == 1) { ans = i; }
    }
    cout << ans << endl;
}
