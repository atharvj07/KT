n = int(input().strip())
s = input().strip()

# Create a list to keep track of domino states, initially all are vertical ('.')
states = ['.'] * n

# Simulate the falling of dominoes to the right
force = 0
for i in range(n):
    if s[i] == 'R':
        force = n  # Set force to a high value to ensure it reaches the end unless stopped
    elif s[i] == 'L':
        force = 0  # Domino falls to the left, stopping the rightward force
    else:
        force = max(force-1, 0)  # Decrease force by 1 for each step if not stopped
    if force:
        states[i] = 'R'  # Mark as falling right if there's any force

# Simulate the falling of dominoes to the left
force = 0
for i in range(n-1, -1, -1):
    if s[i] == 'L':
        force = n  # Set force to a high value to ensure it reaches the end unless stopped
    elif s[i] == 'R':
        force = 0  # Domino falls to the right, stopping the leftward force
    else:
        force = max(force-1, 0)  # Decrease force by 1 for each step if not stopped
    if force:
        if states[i] == 'R':  # If already marked as falling right, it becomes balanced
            states[i] = '.'
        else:
            states[i] = 'L'  # Mark as falling left if there's any force

# Count the number of dominoes that remain vertical
print(states.count('.'))
