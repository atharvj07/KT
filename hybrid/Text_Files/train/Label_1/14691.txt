#include <cstdio>
#include <cstring>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    long long a, b;
    int p, q;
    scanf("%lld %d %lld %d", &a, &p, &b, &q);
    while (p--) a *= 10;
    while (q--) b *= 10;
    if (a < b) {
      printf("<");
    } else if (a > b) {
      printf(">");
    } else {
      printf("=");
    }
  }
  return 0;
}
