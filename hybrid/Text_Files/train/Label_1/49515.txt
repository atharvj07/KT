import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        int[] pos = new int[n + 1];
        for (int i = 0; i < n; i++) {
            pos[p[i]] = i;
        }
        int[] sorted = p.clone();
        Arrays.sort(sorted);
        List<Pair> ans = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (sorted[i] != p[i]) {
                int j = pos[sorted[i]];
                ans.add(new Pair(i, j));
                swap(p, i, j);
                pos[p[i]] = i;
                pos[p[j]] = j;
            }
        }
        System.out.println(ans.size());
        for (Pair pair : ans) {
            System.out.println((pair.x + 1) + " " + (pair.y + 1));
        }
    }

    public static void swap(int[] p, int i, int j) {
        int temp = p[i];
        p[i] = p[j];
        p[j] = temp;
    }

    public static class Pair {
        public int x;
        public int y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
