nmq = list(map(int, input().split()))
n, m, q = nmq[0], nmq[1], nmq[2]
host = list(map(int, input().split()))
k = []
for i in range(q):
  k.append(int(input()))
khost = host[-1]
kcnt = [0] * m
kcnt[khost - 1] = 1
for i in range(n - 2, -1, -1):
  h = host[i]
  if kcnt[h - 1] <= kcnt[khost - 1]:
    khost = h
  kcnt[h - 1] += 1
for year, kh in zip(k, kcnt):
  kh -= 1
  for i in range(m):
    kh -= kcnt[i]
    if kh <= 0:
      print(i + 1)
      break
