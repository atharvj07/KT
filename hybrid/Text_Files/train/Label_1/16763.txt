#include <bits/stdc++.h>
using namespace std;

const int N = 2003;

int n, x;
int h[N], m[N], t[N];

vector<pair<int, int>> adj[N];

int dp[N][N];

int solve(int idx, int type) {
  if (idx > n) return 0;
  if (dp[idx][type] != -1) return dp[idx][type];
  int nx_type = 1 - type;
  int ret = solve(idx + 1, type);
  if (h[idx] <= x) {
    if (t[idx] != type) ret = max(ret, solve(idx + 1, nx_type) + m[idx]);
  }
  return dp[idx][type] = ret;
}

int main() {
  memset(dp, -1, sizeof(dp));
  cin >> n >> x;
  for (int i = 1; i <= n; i++) {
    cin >> t[i] >> h[i] >> m[i];
  }
  int ans = solve(1, 0);
  ans = max(ans, solve(1, 1));
  cout << ans << endl;
  return 0;
}
