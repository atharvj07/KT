#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long s;
    cin >> s;
    long long ans = 0;
    while (s >= 10) {
      long long x = s / 10 * 10;
      ans += x;
      s -= x;
      s += x / 10;
    }
    ans += s;
    cout << ans << endl;
  }
  return 0;
}
