n, k = map(int, input().split())
s = input().strip()

active_doors = set()
max_active_doors = 0
last_seen = {}

for i, door in enumerate(s):
    # Add door to active doors
    active_doors.add(door)
    # Update the last seen position of the door
    last_seen[door] = i
    # Check and remove doors that should be closed
    doors_to_remove = [d for d, pos in last_seen.items() if pos == i]
    for d in doors_to_remove:
        active_doors.remove(d)
    # Update max active doors
    max_active_doors = max(max_active_doors, len(active_doors))

print("YES" if max_active_doors > k else "NO")
