n, m = map(int, input().split())
black_cells = []

for i in range(n):
    row = input()
    for j in range(m):
        if row[j] == 'B':
            black_cells.append((i, j))

if not black_cells:
    print(1)
else:
    rows = [cell[0] for cell in black_cells]
    cols = [cell[1] for cell in black_cells]
    min_row, max_row = min(rows), max(rows)
    min_col, max_col = min(cols), max(cols)
    
    side_length = max(max_row - min_row + 1, max_col - min_col + 1)
    
    if side_length > n or side_length > m:
        print(-1)
    else:
        current_black_cells = len(black_cells)
        total_cells_in_square = side_length**2
        print(total_cells_in_square - current_black_cells)
