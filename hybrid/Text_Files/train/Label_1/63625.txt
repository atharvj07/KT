from sys import stdin
def gcd(a, b):
    return a if b == 0 else gcd(b, a%b)
def lcm(a, b):
    return a*b//gcd(a, b)
def myWCD(L, N):
    ret = L[0][0]
    for x, y in L:
        ret = lcm(ret, x)
        ret = lcm(ret, y)
    for x, y in L:
        if ret%x == 0 and ret%y == 0:
            continue
        return -1
    return ret
N = int(stdin.readline())
L = []
for _ in range(N):
    l = list(map(int, stdin.readline().split()))
    L.append(l)
print(myWCD(L, N))
