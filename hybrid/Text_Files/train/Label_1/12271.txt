#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<int> dp(n + 1);
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < i; j++) {
      dp[i] = (dp[i] + 1LL * dp[j] * dp[i - j - 1]) % mod;
    }
  }
  cout << dp[n] << endl;
  return 0;
}
