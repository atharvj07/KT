#include <bits/stdc++.h>
using namespace std;
const int MXN = 2e5, MXM = 2e5;
vector<int> adj[MXN];
int w[MXM];
int sub[MXN];
int tin[MXN], tout[MXN], timer = 0;
int dp[MXN][21], lg[MXN];
int ans[MXN], cur[MXN];
void dfs0(int v, int p, int e) {
    sub[v] = 1;
    tin[v] = timer++;
    dp[v][0] = p;
    lg[1] = 0;
    for (int i = 2; i <= timer; i++) lg[i] = lg[i >> 1] + 1;
    for (int i = 1; i <= lg[timer]; i++)
        for (int j = 0; j < timer; j++) dp[j][i] = dp[dp[j][i - 1]][i - 1];
    for (auto u : adj[v]) {
        if (u == p) continue;
        w[e] = e;
        dfs0(u, v, e + 1);
        sub[v] += sub[u];
    }
    tout[v] = timer++;
}
int get(int v, int e) {
    int mx = 0;
    for (int i = lg[timer]; i >= 0; i--) {
        if (dp[v][i] && w[dp[v][i]] <= e) {
            mx = max(mx, dp[v][i]);
            v = dp[v][i];
        }
    }
    return mx;
}
void dfs1(int v, int p, int e) {
    if (p) cur[e] = max(cur[e], cur[w[e]]);
    ans[v] = ans[p];
    for (auto u : adj[v]) {
        if (u == p) continue;
        if (get(u, e) == v) ans[v]++;
        else
            dfs1(u, v, e + 1);
    }
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n, m;
    cin >> n >> m;
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs0(1, 1, 1);
    vector<int> q(m);
    for (int i = 0; i < m; i++) {
        cin >> q[i];
    }
    for (int i = 0; i < m; i++) {
        dfs1(1, 1, 1);
        for (int j = 1; j <= n; j++) {
            cout << ans[j] << " ";
            ans[j] = cur[j] = 0;
        }
        cout << '\n';
    }
    return 0;
}
