#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> ans(n, 0);
  vector<int> last(n, -1);
  vector<vector<int>> dances(m, vector<int>(3));
  for (int i = 0; i < m; i++) {
    for (int &j : dances[i]) {
      cin >> j, j--;
    }
  }
  for (int i = 0; i < m; i++) {
    for (int j : dances[i]) {
      if (last[j] != -1) {
        ans[j] = -1;
      }
      last[j] = i;
    }
  }
  for (int i = 0; i < n; i++) {
    if (ans[i] != -1) {
      for (int j = 1; j <= 3; j++) {
        bool ok = true;
        for (int k : dances[last[i]]) {
          if (ans[k] == j) {
            ok = false;
          }
        }
        if (ok) {
          ans[i] = j;
          break;
        }
      }
    }
  }
  for (int i : ans) {
    cout << i << " ";
  }
  cout << endl;
}
