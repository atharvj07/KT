from heapq import heappop, heappush
from collections import defaultdict

def dijkstra(graph, start, n):
    dist = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    dist[start][1] = 0
    q = [(0, start, 1)]  # (distance, node, count)
    while q:
        d, node, count = heappop(q)
        if dist[node][count] < d:
            continue
        for nxt, time in graph[node]:
            if count + 1 <= n and d + time < dist[nxt][count + 1]:
                dist[nxt][count + 1] = d + time
                heappush(q, (d + time, nxt, count + 1))
    return dist

def reconstruct_path(parent, n, k):
    path = []
    current = n
    while current != 1:
        path.append(current)
        current = parent[current][k]
        k -= 1
    path.append(1)
    return path[::-1]

n, m, T = map(int, input().split())
graph = defaultdict(list)
for _ in range(m):
    u, v, t = map(int, input().split())
    graph[u].append((v, t))

dist = dijkstra(graph, 1, n)
max_count = 0
for i in range(n, 0, -1):
    if dist[n][i] <= T:
        max_count = i
        break

parent = [[0] * (n + 1) for _ in range(n + 1)]
for u in range(1, n + 1):
    for v, t in graph[u]:
        for count in range(1, n):
            if dist[u][count] + t == dist[v][count + 1]:
                parent[v][count + 1] = u

path = reconstruct_path(parent, n, max_count)
print(max_count)
print(' '.join(map(str, path)))
