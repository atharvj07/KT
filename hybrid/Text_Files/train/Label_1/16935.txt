#include <bits/stdc++.h>
using namespace std;

vector<int>g[200005];
vector<int>dis(200005, -1);
vector<bool>vis(200005, false);
vector<pair<int, int>>path;
void dfs(int node, int d, int f) {
  if (vis[node] == true) return;
  dis[node] = d;
  path.push_back(make_pair(node, d));
  vis[node] = true;
  for (auto u : g[node]) {
    if (u == f) continue;
    dfs(u, d + 1, node);
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, a, b;
    cin >> n >> m >> a >> b;
    for (int i = 1; i <= n; i++) {
      g[i].clear();
      dis[i] = -1;
      vis[i] = false;
    }
    for (int i = 0; i < m; i++) {
      int x, y;
      cin >> x >> y;
      g[x].push_back(y);
      g[y].push_back(x);
    }
    dfs(a, 0, a);
    for (int i = 1; i <= n; i++) vis[i] = false;
    dfs(b, 0, b);
    int ans = 0;
    for (auto p : path) {
      if (p.second != 0 && p.second >= dis[a] && p.second < dis[b]) ans++;
      if (p.second != 0 && p.second < dis[a] && p.second >= dis[b]) ans++;
    }
    cout << ans << endl;
    path.clear();
  }
  return 0;
}
