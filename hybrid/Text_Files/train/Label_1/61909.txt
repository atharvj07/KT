import heapq

n, m = map(int, input().split())
c = list(map(int, input().split()))
adj = [[] for _ in range(n+1)]
for _ in range(m):
    x, y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)

vis = [False for _ in range(n+1)]
vis[1] = True
q = [(0, 1)]
ans = 0

while q:
    cost, node = heapq.heappop(q)
    if vis[node]: continue
    vis[node] = True
    ans += c[node-1]
    for u in adj[node]:
        if not vis[u]:
            heapq.heappush(q, (cost+1, u))

print(ans)
