#include <vector>
#include <cstdio>
#include <algorithm>
using namespace std;

const int inf = 1<<30;

int rnk[200010], prnt[200010], sz[200010], bl[200010];
int find_set(int x) { return x == prnt[x] ? x : prnt[x] = find_set(prnt[x]); }
void union_set(int x, int y) {
    int rx = find_set(x), ry = find_set(y);
    if(rx == ry) return;
    if(rnk[rx] > rnk[ry]) {prnt[ry] = rx; sz[rx] += sz[ry];}
    else {prnt[rx] = ry; sz[ry] += sz[rx]; if(rnk[rx] == rnk[ry]) rnk[ry]++;}
}

int main(void) {
    int n, q; scanf("%d", &n);
    vector<int> a(n); for(int i=0; i<n; ++i) scanf("%d", &a[i]);
    scanf("%d", &q);

    vector<int> ev(q);
    vector<pair<int, int>> amt(q);
    for(int i=0; i<q; ++i) {
        scanf("%d%d", &ev[i], &amt[i].first);
        if(ev[i] == 2) amt[i].second = amt[i].first;
        else scanf("%d", &amt[i].second);
    }

    for(int i=0; i<200010; ++i) prnt[i] = i, sz[i] = 1;
    for(int i=0; i<q; ++i) {
        if(ev[i] == 1) bl[ev[i]-1] = max(bl[ev[i]-1], amt[i].second);
        else {
            int u = find_set(0);
            for(int j=0; j<n; ++j) if(a[j] < amt[i].first) union_set(j, u);
            for(int j=0; j<n; ++j) if(find_set(j) == u) a[j] = amt[i].first;
        }
    }

    for(int i=0; i<n; ++i) a[i] = max(a[i], bl[i]);
    for(int i=0; i<n; ++i) printf("%d ", a[i]);
    
    return 0;
}
