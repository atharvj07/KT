import sys
input = sys.stdin.readline
n, h, a, b, q = map(int, input().split())
ans = []
for _ in range(q):
    ta, fa, tb, fb = map(int, input().split())
    if ta == tb:
        ans.append(abs(fa - fb))
    else:
        if fa == a or fa == b:
            ans.append(fa - 1 + abs(fb - tb) + tb - 1)
        else:
            ans.append(fa + tb - 1 + abs(fb - ta))
for a in ans:
    print(a)
