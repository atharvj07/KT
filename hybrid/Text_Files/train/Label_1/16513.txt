#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll mod=1e9+7;
int main()
{
	int n,k;
	cin>>n>>k;
	ll a[n+1];
	for(int i=1;i<=n;i++)cin>>a[i];
	int pre[n+1];
	pre[0]=1;
	for(int i=1;i<=n;i++)pre[i]=pre[i-1]*(a[i]%k);
	pre[n]=pre[n]%k;
	unordered_map<int,int>mp;
	ll ans=0;
	for(int i=1;i<=n;i++)
	{
		int now=pre[i-1];
		now=(now*k)%k;
		now+=pre[i];
		now%=k;
		ans+=mp[now];
		mp[now]++;
	}
	cout<<ans<<endl;
	return 0;
}
