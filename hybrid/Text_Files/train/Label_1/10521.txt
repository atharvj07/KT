#include<bits/stdc++.h>
using namespace std;
int a[101][101],b[101][101],f=1,m,n;
int main(){
  cin>>m>>n;
  for(int i=0;i<m;i++){
   for(int j=0;j<n;j++){
    cin>>b[i][j];
   }
  }
  for(int i=0;i<m;i++){
   for(int j=0;j<n;j++){
    if(b[i][j]==0) a[i][j]=0;
    else{
     int cnt=0;
     for(int p=0;p<m;p++) if(b[p][j]) cnt++;
     for(int p=0;p<n;p++) if(b[i][p]) cnt++;
     if(cnt<=1) {f=0; break;}
     a[i][j]=1;
    }
   }
  }
  cout<<(f?"YES\n":"NO\n");
  for(int i=0;i<m;i++){
   for(int j=0;j<n;j++){
    cout<<a[i][j]<<" ";
   }
   cout<<endl;
  }
  return 0;
}
