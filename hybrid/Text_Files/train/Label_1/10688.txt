#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    map<string, char> mp;
    while(n--)
    {
        string s;
        cin>>s;
        if(s.size()>2) continue;
        int f1=0,f2=0;
        for(int i=0; i<s.size(); i++)
        {
            if(s[i]==s[0]) f1++;
            else f2++;
        }
        if(f1==s.size()) mp[s]='1';
        else if(f2==s.size()) mp[s]='2';
        else continue;
    }
    int mx=0;
    for(auto x:mp)
    {
        mx+=(x.first).size();
    }
    if(mp.size()>0) cout<<mx<<endl;
    else cout<<"0"<<endl;
}
