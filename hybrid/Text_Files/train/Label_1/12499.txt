#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll x1, y1, x2, y2, w, h, W, H;
        cin >> W >> H;
        cin >> x1 >> y1 >> x2 >> y2;
        cin >> w >> h;
        if (x2 - x1 + w > W || y2 - y1 + h > H)
            cout << "-1\n";
        else
            cout << min({x1, W - x2, y1, H - y2}) << "\n";
    }
    return 0;
}
