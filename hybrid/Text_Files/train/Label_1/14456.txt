#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<ll> a(n + 1);
  rep(i, n) cin >> a[i + 1];

  vector<ll> dp(n + 1);
  rep(i, n) {
    dp[i] = dp[i + 1] + a[1];
    rep(j, i) {
      if (s[j] == s[i]) dp[i] = max(dp[i], dp[j] + a[i - j]);
    }
  }
  cout << dp[n - 1] << endl;
  return 0;
}
