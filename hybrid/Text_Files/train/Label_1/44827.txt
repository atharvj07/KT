import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.StringTokenizer;

class Main {
    static class Castle {
        int warriorsNeeded;
        int warriorsAvailable;
        int importance;
    }

    static class Portal {
        int source;
        int destination;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        Castle[] castles = new Castle[n + 1];
        for (int i = 1; i <= n; i++) {
            st = new StringTokenizer(br.readLine());
            castles[i] = new Castle();
            castles[i].warriorsNeeded = Integer.parseInt(st.nextToken());
            castles[i].warriorsAvailable = Integer.parseInt(st.nextToken());
            castles[i].importance = Integer.parseInt(st.nextToken());
        }

        List<Portal>[] portals = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            portals[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            Portal portal = new Portal();
            portal.source = Integer.parseInt(st.nextToken());
            portal.destination = Integer.parseInt(st.nextToken());
            portals[portal.source].add(portal);
        }

        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[1] = k;

        for (int castle = 1; castle <= n; castle++) {
            if (dp[castle] == -1) {
                continue;
            }

            Castle currentCastle = castles[castle];
            // Check if it's possible to capture current castle
            if (dp[castle] < currentCastle.warriorsNeeded) {
                continue;
            }

            // Capture current castle
            dp[castle] += currentCastle.warriorsAvailable;

            // Check all portals from current castle
            for (Portal portal : portals[castle]) {
                int nextCastle = portal.destination;
                dp[nextCastle] = Math.max(dp[nextCastle], dp[castle] - 1);
            }

            // Try defending current castle
            dp[castle] = Math.max(dp[castle], dp[castle] - 1);
        }

        if (dp[n] == -1) {
            // It's impossible to capture all castles
            System.out.println(-1);
        } else {
            // Print the maximum sum of importance values of defended castles
            int maxImportance = 0;
            for (int castle = 1; castle <= n; castle++) {
                if (dp[castle] >= castles[castle].warriorsNeeded) {
                    maxImportance += castles[castle].importance;
                }
            }
            System.out.println(maxImportance);
        }
    }
}
