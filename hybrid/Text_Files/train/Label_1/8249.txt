#include <iostream>
#include <vector>
#include <map>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> planks(n);
    for (int i = 0; i < n; i++) {
        std::cin >> planks[i];
    }

    std::map<int, int> plank_counts;
    for (int plank : planks) {
        plank_counts[plank]++;
    }

    int q;
    std::cin >> q;

    for (int i = 0; i < q; i++) {
        char op;
        int x;
        std::cin >> op >> x;

        if (op == '+') {
            plank_counts[x]++;
        } else {
            plank_counts[x]--;
            if (plank_counts[x] == 0) {
                plank_counts.erase(x);
            }
        }

        bool can_build_square = false;
        bool can_build_rectangle = false;

        if (plank_counts.size() == 1) {
            can_build_square = true;
        } else {
            int num_pairs = plank_counts.size() / 2;
            if (plank_counts.size() % 2 == 0) {
                can_build_rectangle = true;
                for (auto it = plank_counts.begin(); it != plank_counts.end(); it++) {
                    if (it->second != num_pairs) {
                        can_build_rectangle = false;
                        break;
                    }
                }
            }
        }

        if (can_build_square && can_build_rectangle) {
            std::cout << "YES\n";
        } else {
            std::cout << "NO\n";
        }
    }

    return 0;
}
