#include <bits/stdc++.h>
using namespace std;

#define ll long long
const int MM = 1e5 + 5;
int n, m, s, f;
vector<tuple<ll, ll, ll>> v;

int main() {
    cin >> n >> m >> s >> f;
    for (int i = 0; i < m; i++) {
        ll a, b, c;
        cin >> a >> b >> c;
        v.push_back({a, b, c});
    }
    string ans = "";
    ll c = s;
    for (auto e : v) {
        ll tim, x, y;
        tie(tim, x, y) = e;
        if (tim > ans.size() || x > c || y < c)
            continue;
        if (c < f) {
            while (c < y && tim > ans.size()) {
                ans += 'R';
                c++;
            }
            if (c == f)
                break;
        } else {
            while (c > x && tim > ans.size()) {
                ans += 'L';
                c--;
            }
            if (c == f)
                break;
        }
    }
    for (ll i = ans.size(); i <= f - s - 1; i++) {
        if (c < f)
            ans += 'R', c++;
        else
            ans += 'L', c--;
    }
    cout << ans << '\n';
}
