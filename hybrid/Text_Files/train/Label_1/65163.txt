n = int(input())
strengths = []
for i in range(2*n-1):
    row = list(map(int, input().split()))
    for j, val in enumerate(row):
        strengths.append((val, i+2, j+1))
strengths.sort(reverse=True)

teammates = [0] * (2 * n + 1)

for strength, i, j in strengths:
    if teammates[i] == 0 and teammates[j] == 0:
        teammates[i] = j
        teammates[j] = i

print(*teammates[1:])
