import sys

n, m = map(int, input().split())
graph = [set() for i in range(n + 1)]
for i in range(m):
    u, v = map(int, input().split())
    graph[u].add(v)
    graph[v].add(u)

visited = [False for i in range(n + 1)]
par = [0 for i in range(n + 1)]

def dfs(node, p, cover):
    visited[node] = True
    par[node] = p
    cover.add(node)
    for u in graph[node]:
        if u == p:
            continue
        if not visited[u]:
            if dfs(u, node, cover):
                return True
        else:
            if par[node] != u:
                return True
    return False

result = []
for i in range(1, n + 1):
    if not visited[i]:
        cover = set()
        if dfs(i, 0, cover):
            result.append(cover)

if len(result) != 2:
    print(-1)
else:
    print(len(result[0]))
    for node in result[0]:
        print(node, end=" ")
    print()
    print(len(result[1]))
    for node in result[1]:
        print(node, end=" ")
