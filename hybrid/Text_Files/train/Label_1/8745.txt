#include <bits/stdc++.h>

int main() {
  int n, k, m;
  std::cin >> n >> k >> m;
  std::vector<int> a(n);
  for (auto &x : a) std::cin >> x;
  while (k <= n) {
    for (int i = 0; i + k <= n; i++) {
      int same_city = 1;
      for (int j = i, cnt = 0; j < i + k; j++, cnt++) {
        same_city = (same_city && a[j] == a[i]);
        if (same_city) m -= (cnt == k - 1);
        if (m <= 0) { n = j + 1; break; }
      }
    }
  }
  std::cout << n;
  return 0;
}
