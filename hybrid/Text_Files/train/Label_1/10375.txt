#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  int cnt = 0;  // current number of books at the library
  vector<int> b(k + 1, 0);  // last day when book i was bought
  for (int i = 0; i < n; i++) {
    if (b[a[i]] == 0) {
      if (cnt < k) {
        cnt++;
        ans++;
        b[a[i]] = i + 1;
      } else {
        int mx = 0;
        int r = -1;
        for (int j = 1; j <= k; j++) {
          if (b[j] > mx) {
            mx = b[j];
            r = j;
          }
        }
        ans -= (mx > i);
        cnt--;
        b[r] = 0;
        cnt++;
        ans++;
        b[a[i]] = i + 1;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
