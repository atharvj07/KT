#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n, m; cin >> n >> m;
        vector<int> b(n), l(m);
        for (int& x : b) cin >> x;
        for (int& x : l) cin >> x;
        int tot = 0;
        for (int x: b) for (int y: l) if (x == y) tot++;
        cout << max(0,tot - (n + m - tot)) << endl;
    }
    return 0;
}
