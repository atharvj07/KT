#include <iostream>
#include <algorithm>
#include <string>
using namespace std;
int main() {
  int n;
  cin >> n;
  int beauty = 0;
  string prv = "";
  for (int i = 0; i < n; i++) {
    string a;
    cin >> a;
    beauty = max(beauty, (int)a.size());
    if (prv + a != prv) {
      beauty = max(beauty, (int)prv.size());
      prv = a;
    } else
      prv += a;
  }
  cout << beauty << endl;
  return 0;
}
