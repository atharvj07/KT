#include <bits/stdc++.h>

using namespace std;

typedef long long LL;

const int MAXN = 300005;

int n, a[MAXN], f[MAXN], g[MAXN], ans[MAXN], lst[MAXN];
vector<pair<int, int> > v;

int main() {
  ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);
  cin >> n;
  set<int> s;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    if (i > 1) {
      g[i] = __gcd(g[i - 1], a[i]);
    } else {
      g[i] = a[i];
    }
    s.insert(a[i]);
  }
  int ansl = n + 1, ansr = -1, ansgcd = 0;
  for (int i = n; i > 0; --i) {
    f[i] = *s.lower_bound(g[i]);
    if (a[i] % f[i] == 0) {
      lst[f[i]] = i;
      if (lst[g[i]] >= i) {
        if (lst[g[i]] - i > ansr - ansl) {
          ansl = i, ansr = lst[g[i]];
          ansgcd = g[i];
        }
      }
    }
    s.erase(a[i]);
  }
  if (ansl <= n) {
    cout << 1 << " " << ansr - ansl << "\n";
    cout << ansl << "\n";
  } else {
    cout << 0 << "\n";
  }
  return 0;
}
