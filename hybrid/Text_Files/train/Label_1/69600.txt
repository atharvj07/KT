import sys
input = sys.stdin.readline

def solve(n):
    if n == 2:
        return 1, [(1, 2)]
    if n % 2 == 0:
        return n // 2, [(i + n // 2, i) for i in range(1, n // 2)]
    else:
        return (n + 1) // 2, [(i + (n + 1) // 2, i) for i in range(1, (n + 1) // 2)] + [(n, (n + 1) // 2)]

for _ in range(int(input())):
    n = int(input())
    result, operations = solve(n)
    print(result)
    for op in operations:
        print(*op)
