#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        vector<ll> a(n);
        for (ll i = 0; i < n; i++) {
            cin >> a[i];
        }
        ll cnt0 = 0;
        for (ll i = 0; i < n; i++) {
            if (a[i] == 0) {
                cnt0++;
            }
        }
        if (cnt0 == n) {
            cout << "YES" << endl;
            cout << 0 << endl;
            continue;
        }
        ll idx = -1;
        ll cnt1 = 0;
        for (ll i = 0; i < n - 2; i++) {
            if (a[i] == 1 && a[i + 1] == 1 && a[i + 2] == 1) {
                idx = i + 1;
                cnt1++;
            }
        }
        if (idx != -1) {
            if (cnt1 % 2 == 1 || cnt0 > 0) {
                cout << "YES" << endl;
                cout << cnt1 << endl;
                for (ll i = 0; i < cnt1; i++) {
                    cout << idx << " ";
                    idx += 2;
                }
                cout << endl;
            } else {
                cout << "NO" << endl;
            }
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
