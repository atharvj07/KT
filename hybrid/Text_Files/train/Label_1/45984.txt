import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int L = scanner.nextInt();
        int N = scanner.nextInt();
        int[] x = new int[N];
        int[] w = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = scanner.nextInt();
            w[i] = scanner.nextInt();
        }

        int sumX = 0, sumW = 0;
        for (int i = 0; i < N; i++) {
            sumX += x[i] * w[i];
            sumW += w[i];
        }

        int x0 = -sumX / sumW;
        int w0 = sumW;

        System.out.println(1);
        System.out.println(x0 + " " + w0);
    }
}
