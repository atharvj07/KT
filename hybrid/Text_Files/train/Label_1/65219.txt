def is_nearly_prime(n):
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return i, n // i
    return False

def solve(n):
    if n <= 30:
        return "NO"
    else:
        if n - 30 in [6, 10, 14]:
            return "YES", [6, 10, 15, n - 31]
        else:
            return "YES", [6, 10, 14, n - 30]

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    result = solve(n)
    if result[0] == "NO":
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, result[1])))
