#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    long long n, c;
    cin >> n >> c;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) { cin >> a[i]; }
    long long ans = 0;
    for (long long i = 0, j = 0, sum = 0; i < n; i++) {
        while (j < n && sum + a[j] <= c) { sum += a[j++]; }
        if (sum == c) { ans = max(ans, j - i); }
        sum -= a[i];
    }
    cout << ans << endl;
    return 0;
}
