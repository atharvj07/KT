import java.util.Scanner;

public class CountryWithLayers {

    private static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] interLayerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            interLayerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        long[][] dp = new long[l + 1][n];
        dp[0][0] = 1;

        for (int i = 1; i <= l; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if ((entryCosts[k] + interLayerCosts[k] * (i - 1) + exitCosts[j]) % m == 0) {
                        dp[i][j] = (dp[i][j] + dp[i - 1][k]) % MOD;
                    }
                }
            }
        }

        long result = 0;
        for (int i = 0; i < n; i++) {
            result = (result + dp[l][i]) % MOD;
        }

        System.out.println(result);
    }
}
