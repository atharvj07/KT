#include<bits/stdc++.h>
#include<cstring>
#include<iostream>
using namespace std;

int t, n, m, in[2005][2005], ans[2005][2005];
char s[2005][2005];

void solve() {
    cin >> n >> m;
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> s[i][j];
            if (s[i][j] == '.') ans[i][j] = -1;
        }
    }

    for (int j = 1; j <= m; j++) {
        for (int i = 1; i <= n; i++) {
            if (s[i][j] != '.') {
                int p = s[i][j] - 'a' + 1;
                if (cnt == p) continue;
                else if (cnt < p) {
                    int x = i, y = j;
                    while (s[x][y] != '.') {
                        ans[x][y] = cnt;
                        if (x + 1 <= n && ans[x + 1][y] == -1) x++;
                        else if (y + 1 <= m && ans[x][y + 1] == -1) y++;
                        else {
                            cout << "NO" << endl;
                            return;
                        }
                    }
                    ans[x][y] = cnt;
                    cnt++;
                } else {
                    cout << "NO" << endl;
                    return;
                }
            }
        }
    }

    cout << "YES" << endl;
    cout << cnt << endl;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (ans[i][j] != -1) {
                int p = ans[i][j] + 1;
                int x = i, y = j;
                while (ans[x][y] == p) {
                    if (x + 1 <= n && ans[x + 1][y] == p) x++;
                    else if (y + 1 <= m && ans[x][y + 1] == p) y++;
                    else {
                        cout << i << " " << j << " " << x << " " << y << endl;
                        break;
                    }
                }
            }
        }
    }
    return;
}

int main() {
    cin >> t;
    while (t--) {
        memset(ans, -1, sizeof ans);
        solve();
    }
    return 0;
}
