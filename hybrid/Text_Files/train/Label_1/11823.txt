#include <iostream>
#include <algorithm>
#include <vector>
#define MAX 100005
using namespace std;
int a[MAX], sz, b[MAX], ans;
int main() {
  cin >> sz;
  for (int tc = 1; tc <= sz; tc++) {
    int n, m = 0;
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      m = max(m, a[i]);
    }
    for (int i = 0; i < n; i++) {
      b[i] = m - a[i];
      if (i > 0) b[i] += b[i - 1];
    }
    ans = b[n - 1];
    int p = *max_element(a, a + n);
    for (int i = 0; i < n; i++) {
      int s = 0;
      if (i > 0) s = b[i - 1];
      if (i + 1 < n) s += b[n - 1] - b[i];
      if (s > ans) ans = s;
      if (a[i] != p) ans = min(ans, m - p);
    }
    cout << ans << endl;
  }
}
