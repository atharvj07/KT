import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long a = sc.nextLong();
        long b = sc.nextLong();
        long[] t = new long[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextLong();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (t[i] < a || b < t[i]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
