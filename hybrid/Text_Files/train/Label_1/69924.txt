def solve_exchange_rate_problem(n):
    n_list = list(n)
    last_digit = int(n_list[-1])
    has_even = False
    max_even_index = -1
    for i in range(len(n_list) - 2, -1, -1):
        if int(n_list[i]) % 2 == 0:
            has_even = True
            if int(n_list[i]) < last_digit:
                n_list[-1], n_list[i] = n_list[i], n_list[-1]
                return "".join(n_list)
            max_even_index = i
    if has_even:
        n_list[-1], n_list[max_even_index] = n_list[max_even_index], n_list[-1]
        return "".join(n_list)
    else:
        return "-1"

n = input().strip()
print(solve_exchange_rate_problem(n))
