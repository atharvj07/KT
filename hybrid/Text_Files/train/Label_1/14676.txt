#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long

int32_t main() {
    int n, m; cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (auto &v : a) {
        for (auto &i : v) { cin >> i; }
    }
    for (int i = 1; i < n - 1; i++) {
        for (int j = 1; j < m - 1; j++) {
            if (a[i][j] == 0) {
                if (a[i][j - 1] > a[i][j + 1] || a[i - 1][j] > a[i + 1][j]) {
                    cout << -1 << endl;
                    exit(0);
                }
            }
        }
    }
    int ans = 0;
    for (auto &v : a) {
        for (auto &i : v) { ans += i; }
    }
    for (int i = 1; i < n - 1; i++) {
        for (int j = 1; j < m - 1; j++) {
            if (a[i][j] == 0) {
                a[i][j] = min({a[i][j - 1] - 1, a[i][j + 1] - 1, a[i - 1][j] - 1, a[i + 1][j] - 1});
                ans += a[i][j];
            }
        }
    }
    cout << ans << endl;
}
