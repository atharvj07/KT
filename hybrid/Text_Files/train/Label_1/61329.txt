import sys

def solve():
    n, x = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    a.sort()
    i = 0
    ans = 0
    while i < n:
        j = i
        while j < n and a[j] * (j - i + 1) >= x:
            j += 1
        if j > i:
            ans += 1
            i = j
        else:
            i += 1
    return ans

t = int(sys.stdin.readline())
for _ in range(t):
    print(solve())
