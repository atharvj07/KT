#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    ll n,c;
    cin>>n>>c;
    vector<ll>v(n);
    for(int i=0;i<n;i++)cin>>v[i];
    sort(v.begin(),v.end());
    ll l=0, r=n-1, ans=0, sum=0;
    while(l<=r)
    {
        if(v[r]<=c)
        {
            ans+=n-r;
            sum+=v[r];
            r--;
        }
        else if(v[l]<=c)
        {
            ans+=n-l;
            sum+=v[l];
            l++;
        }
        else break;
    }
    sum+=(c-sum)/c*c;
    for(int i=0;i<n;i++)
    {
        if(sum>=c)
        {
            sum-=c;
            ans++;
        }
        else break;
    }
    cout<<ans<<endl;
    return 0;
}
