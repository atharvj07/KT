#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  int mn = INT_MAX, mx = INT_MIN;
  for (int &i : a) {
    cin >> i;
    mn = min(mn, i);
    mx = max(mx, i);
  }
  if (n % 2) {
    cout << mn << '\n';
  } else {
    cout << (mn == mx ? mn : min(mn, mx + 1)) << '\n';
  }
  return 0;
}
