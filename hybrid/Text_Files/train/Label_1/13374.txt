#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> v(m);
  for (int i = 0; i < m; i++) { cin >> v[i]; }
  sort(v.begin(), v.end());
  int cnt = 1, ans = 0;
  for (int i = 1; i < m; i++) {
    if (v[i] == v[i - 1])
      cnt++;
    else {
      ans = max(ans, cnt);
      cnt = 1;
    }
  }
  ans = max(ans, cnt);
  if (ans >= n)
    cout << m / n;
  else
    cout << 0;
}
