n=int(input());a=list(map(int,input().split()));b=[];ans=1;i=0
while i<n:
 j=i+1;k=i+1;f=0
 while j<n and a[j]>=0:j+=1
 while k<n and a[k]<0:k+=1
 if k-j>=3:
  ans+=1
  if f==0:b.append(i-j+1);f=1
  b.append(k-j);i=k
 else:
  j-=1
  if f==0:b.append(i-j+1)
  i=j+1
print(ans);print(' '.join(map(str,b)))
