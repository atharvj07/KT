#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, d, a[100005], b[100005];
int main() {
  cin >> n >> d;
  for (int i = 1; i <= n; i++) cin >> a[i];
  copy(a + 1, a + n + 1, b + 1);
  sort(b + 1, b + n + 1);
  ll ans = 1e18, cur = 0, cnt = 0;
  for (int i = 1; i <= n; i++) {
    cur += a[i];
    if (cur < 0 || cur + b[i] > d) {
      ans = -1;
      break;
    }
    cur += b[i];
    if (cur == 0) cnt++;
  }
  ans = ans == 1e18 ? 0 : cnt;
  cout << ans << endl;
  return 0;
}
