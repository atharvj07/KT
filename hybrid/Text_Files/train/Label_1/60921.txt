import sys
input = sys.stdin.readline

n, k = map(int, input().split())
coupons = []
for i in range(n):
    l, r = map(int, input().split())
    coupons.append([l, r])

# Sort coupons by their right endpoints
coupons.sort(key=lambda x: x[1])

# Find the maximum number of products that can be bought with any k coupons
cnt = 0
for i in range(k):
    cnt += coupons[i][1] - coupons[i][0] + 1

# Find the coupons that cover the maximum number of products
chosen = []
for i in range(k):
    chosen.append(i + 1)

print(cnt)
print(*chosen)
