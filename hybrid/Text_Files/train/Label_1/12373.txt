#include <iostream>
#include <vector>
using namespace std;

vector<bool> isPrime;

bool checkPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++)
        if (n % i == 0) return false;
    return true;
}

void pre() {
    isPrime.resize(1000001, true);
    isPrime[0] = isPrime[1] = false;
    for (int p = 2; p * p <= 1000000; p++) {
        if (!isPrime[p]) continue;
        for (int i = p * p; i <= 1000000; i += p)
            isPrime[i] = false;
    }
}

int main() {
    pre();
    int t;
    cin >> t;
    while (t--) {
        int n, e;
        cin >> n >> e;
        vector<int> a(n);
        for (int i = 0; i < n; i++)
            cin >> a[i];

        int ans = 0;
        for (int i = 0; i < n; i++) {
            int prod = a[i];
            for (int k = 1; i + k * e <= n; k++) {
                prod *= a[i + k * e];
                if (isPrime[prod]) ans++;
            }
        }
        cout << ans << endl;
    }
}
