#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int32_t main() {
  int n, m;
  cin >> n >> m;
  vector<pair<ll, char>> v(n), w(m);
  for (auto& p : v) cin >> p.first >> p.second;
  for (auto& p : w) cin >> p.first >> p.second;
  int ans = 0, p = n - 1, q = m - 1;
  ll l = 0;
  while (p >= 0 && q >= 0) {
    if (v[p].first > l) {
      if (w[q].first > 0 && w[q].second == v[p].second) {
        w[q].first--;
        p--;
        q--;
        ans++;
        l = 0;
      } else {
        l += v[p].first;
        p--;
      }
    } else {
      l -= v[p].first;
      p--;
    }
  }
  cout << ans;
}
