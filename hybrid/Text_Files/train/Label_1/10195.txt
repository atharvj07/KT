#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  long long sum = 0, ans = 0;
  for (int i = n-1; i >= 0; i--) {
    if (sum < a[i] * (n-i-1)) {
      ans = ans + (a[i] * (n-i-1) - sum);
      sum = sum + (a[i] * (n-i-1));
    }
  }
  cout << ans << endl;
  return 0;
}
