import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] widths = new int[m];
        int[] depths = new int[m];
        for (int i = 0; i < m; i++) {
            widths[i] = sc.nextInt();
            depths[i] = sc.nextInt();
        }
        int[] bottomHeights = new int[m];
        for (int i = 0; i < m; i++) {
            int width = widths[i];
            int depth = depths[i];
            int bottomHeight = 0;
            for (int j = 0; j < width; j++) {
                bottomHeight = Math.max(bottomHeight, heights[j]);
            }
            if (width != n) {
                bottomHeight = Math.max(bottomHeight, heights[width]);
            }
            bottomHeights[i] = bottomHeight + depth;
        }
        for (int bottomHeight : bottomHeights) {
            System.out.println(bottomHeight);
        }
    }
}
