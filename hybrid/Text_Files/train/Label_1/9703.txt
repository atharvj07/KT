#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    int cnt = 0;
    for (char c : s) cnt += (c == '0');
    string r = s;
    reverse(r.begin(), r.end());
    if (cnt & 1 || s == r)
      cout << "BOB" << endl;
    else
      cout << "ALICE" << endl;
  }
}
