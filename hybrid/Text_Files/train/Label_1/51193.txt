import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static class Graph {
        private ArrayList<Integer>[] adj;
        private int size;

        public Graph(int size) {
            this.size = size;
            adj = new ArrayList[size];

            for (int i = 0; i < size; i++) {
                adj[i] = new ArrayList<>();
            }
        }

        public void addEdge(int u, int v) {
            adj[u].add(v);
            adj[v].add(u);
        }

        public boolean[] minVertexCover() {
            int[] matching = new int[size];
            Arrays.fill(matching, -1);

            boolean[] visited = new boolean[size];
            int[] path = new int[size];
            int[] maxMatchingPermutations = new int[size];
            int maxMatching = 0;

            for (int i = 0; i < size; i++) {
                if (matching[i] == -1) {
                    int pathLength = findAugmentingPath(i, matching, visited, path);

                    if (pathLength > maxMatching) {
                        maxMatching = pathLength;
                        for (int j = 0; j < pathLength; j++) {
                            maxMatchingPermutations[j] = path[j];
                        }
                    }
                }
            }

            boolean[] vertexCover = new boolean[size];
            for (int i = 0; i < maxMatching; i++) {
                if (i % 2 == 0) {
                    vertexCover[maxMatchingPermutations[i]] = true;
                } else {
                    vertexCover[matching[maxMatchingPermutations[i]]] = true;
                }
            }

            return vertexCover;
        }

        private int findAugmentingPath(int start, int[] matching, boolean[] visited, int[] path) {
            if (visited[start]) {
                return 0;
            }

            visited[start] = true;
            path[0] = start;

            for (int neighbor : adj[start]) {
                if (matching[neighbor] == -1 || findAugmentingPath(matching[neighbor], matching, visited, path) > 0) {
                    path[1] = neighbor;
                    matching[neighbor] = start;
                    matching[start] = neighbor;
                    return 2;
                }
            }

            return 0;
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int m = input.nextInt();

        Graph graph = new Graph(n);

        for (int i = 0; i < m; i++) {
            int u = input.nextInt() - 1;
            int v = input.nextInt() - 1;

            graph.addEdge(u, v);
        }

        boolean[] vertexCover = graph.minVertexCover();

        if (!checkVertexCover(graph, vertexCover)) {
            System.out.println("-1");
        } else {
            int count = 0;
            ArrayList<Integer> a = new ArrayList<>();
            ArrayList<Integer> b = new ArrayList<>();

            for (int i = 0; i < n; i++) {
                if (vertexCover[i]) {
                    count++;
                    if (count % 2 == 1) {
                        a.add(i + 1);
                    } else {
                        b.add(i + 1);
                    }
                }
            }

            System.out.println(a.size());
            for (int vertex : a) {
                System.out.print(vertex + " ");
            }
            System.out.println();
            System.out.println(b.size());
            for (int vertex : b) {
                System.out.print(vertex + " ");
            }
            System.out.println();
        }
    }

    private static boolean checkVertexCover(Graph graph, boolean[] vertexCover) {
        for (int i = 0; i < graph.size; i++) {
            if (vertexCover[i]) {
                continue;
            }

            for (int neighbor : graph.adj[i]) {
                if (!vertexCover[neighbor]) {
                    return false;
                }
            }
        }

        return true;
    }
}
