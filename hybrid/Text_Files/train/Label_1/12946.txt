#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;

int n, m, k;
bool vis[502][502];

bool valid(ll x, ll y) { return 1 <= x && x <= n && 1 <= y && y <= m; }

bool dfs(ll x, ll y, ll dist, vector<pi>& path, unordered_map<ll, vector<vector<pi>>>& memo) {
  if (dist == k) { return 1; }
  if (dist > k) { return 0; }
  if (memo[x * 502 + y].size()) { return memo[x * 502 + y][k - dist].size(); }
  ll key = x * 502 + y;
  vector<vector<pi>>& moves = memo[key];
  if (valid(x - 1, y) && !vis[x - 1][y]) {
    path.push_back({x - 1, y});
    vis[x - 1][y] = 1;
    if (dfs(x - 1, y, dist + 1, path, memo)) { moves.push_back(path); }
    vis[x - 1][y] = 0;
    path.pop_back();
  }
  if (valid(x + 1, y) && !vis[x + 1][y]) {
    path.push_back({x + 1, y});
    vis[x + 1][y] = 1;
    if (dfs(x + 1, y, dist + 1, path, memo)) { moves.push_back(path); }
    vis[x + 1][y] = 0;
    path.pop_back();
  }
  if (valid(x, y - 1) && !vis[x][y - 1]) {
    path.push_back({x, y - 1});
    vis[x][y - 1] = 1;
    if (dfs(x, y - 1, dist + 1, path, memo)) { moves.push_back(path); }
    vis[x][y - 1] = 0;
    path.pop_back();
  }
  if (valid(x, y + 1) && !vis[x][y + 1]) {
    path.push_back({x, y + 1});
    vis[x][y + 1] = 1;
    if (dfs(x, y + 1, dist + 1, path, memo)) { moves.push_back(path); }
    vis[x][y + 1] = 0;
    path.pop_back();
  }
  return moves.size();
}

int main() {
  cin >> n >> m >> k;
  vector<pi> path;
  unordered_map<ll, vector<vector<pi>>> memo;
  if (dfs(1, 1, 0, path, memo)) {
    cout << "YES\n";
    vector<vector<pi>>& moves = memo[1 * 502 + 1];
    cout << moves.size() << "\n";
    for (auto& m : moves) {
      cout << m.size() << " ";
      for (auto& move : m) {
        cout << (move.first == 1 ? "U" : (move.first == n ? "D" : (move.second == 1 ? "L" : "R"))) << "";
      }
      cout << "\n";
    }
  } else {
    cout << "NO\n";
  }
  return 0;
}
