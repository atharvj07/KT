def solve(s):
    n = len(s)
    vis = [False] * 26
    ans = []
    for i in range(n):
        if ord(s[i]) - 97 != 0 and not vis[ord(s[i]) - 96] and not vis[ord(s[i]) - 98]:
            ans.append(s[i])
            vis[ord(s[i]) - 96] = True
            vis[ord(s[i]) - 98] = True
    for i in range(n):
        if not vis[ord(s[i]) - 97]:
            ans.append(s[i])
    if len(ans) == n:
        return ''.join(ans)
    else:
        return "No answer"


t = int(input())
for i in range(t):
    s = input()
    print(solve(s))
