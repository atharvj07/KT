#include <bits/stdc++.h>

using namespace std;
#define int long long
const int mod = 1e9 + 7;

int dp[16][226][4][4];

signed main() {
  int n, t;
  cin >> n >> t;
  vector<pair<int, int>> vec(n);
  for (int i = 0; i < n; i++) { cin >> vec[i].first >> vec[i].second; }
  dp[0][0][0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= t; j++) {
      for (int last = 0; last < 4; last++) {
        for (int pre = 0; pre < 4; pre++) {
          // don't pick
          dp[i + 1][j][last][pre] = (dp[i + 1][j][last][pre] + dp[i][j][last][pre]) % mod;
          // pick
          if (j + vec[i].first <= t) {
            if ((last != vec[i].second || pre != vec[i].second) && last != 0) {
              dp[i + 1][j + vec[i].first][vec[i].second][last] =
                  (dp[i + 1][j + vec[i].first][vec[i].second][last] + dp[i][j][last][pre]) % mod;
            }
          }
        }
      }
    }
  }
  cout << dp[n][t][0][0] << endl;
  return 0;
}
