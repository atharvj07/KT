import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] prices = new long[n];
        for (int i = 0; i < n; i++) {
            prices[i] = sc.nextLong();
        }
        Arrays.sort(prices);
        long ans = 0;
        long num = 1;
        for (int i = n - 1; i >= 0; i--) {
            while (prices[i] > num) {
                num *= 10;
                ans++;
            }
            num -= prices[i];
        }
        System.out.println(ans);
    }
}
