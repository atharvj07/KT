#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long ans = 0;
    // n <= 2, no Pythagorean triple exists.
    // n == 3, only one Pythagorean triple exists, (3, 4, 5), which doesn't satisfy c = a^2 - b
    if (n <= 3) cout << ans << endl;
    else {
      for (long long a = 2; a * a <= n; a++) {
        for (long long b = a + 1; b * b + a * a <= n; b++) {
          if (a * a + b * b > n) break;
          long long c = a * a + b * b;
          if (c == a * a - b) ans++;
        }
      }
      cout << ans << endl;
    }
  }
  return 0;
}
