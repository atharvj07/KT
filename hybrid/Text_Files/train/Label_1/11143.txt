#include <iostream>
#include <vector>
using namespace std;

int n, d;
vector<int> v;

int f() {
    vector<int> dp(n + 1, 1e9);
    dp[1] = 0;
    for (int i = 2; i <= n; i++) {
        for (int j = 1; j <= d; j++) {
            if (i - j >= 1 && v[i - j]) {
                dp[i] = min(dp[i], dp[i - j] + 1);
            }
        }
    }
    return dp[n] < 1e9 ? dp[n] : -1;
}

int main() {
    cin >> n >> d;
    string s;
    cin >> s;
    v.resize(n + 1);
    for (int i = 1; i <= n; i++) { v[i] = (s[i - 1] - '0'); }
    cout << f();
    return 0;
}
