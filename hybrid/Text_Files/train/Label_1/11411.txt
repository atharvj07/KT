#include <bits/stdc++.h>

using namespace std;

const int MAXN = 17;
int n, a[MAXN];

void solve(int l, int r)
{
    if (l + 1 == r)
        return;
    int mid = (l + r) / 2;
    solve(l, mid);
    solve(mid, r);
    int i = l, j = mid;
    vector<int> c;
    while (i < mid && j < r)
    {
        if (a[i] <= a[j])
            c.push_back(a[i++]);
        else
            c.push_back(a[j++]);
    }
    while (i < mid)
        c.push_back(a[i++]);
    while (j < r)
        c.push_back(a[j++]);
    for (i = l; i < r; i++)
        a[i] = c[i - l];
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    solve(0, n);
    for (int i = 0; i < n - 1; i++)
        if (a[i] > a[i + 1])
            return cout << i, 0;
    cout << n;
    return 0;
}
