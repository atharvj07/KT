#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,k,ans,l[100001],r[100001],b[100001];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
  cin>>n>>k;
  for (ll i=1; i<=n; i++) {
    ll l1,r1;
    cin>>l1>>r1;
    l[i]=l1; r[i]=r1;
    b[i]=l1*r1;
  }
  sort(b+1,b+1+n);
  ll last=0;
  for (ll i=n; i>=n-k+1; i--) {
    if (last==b[i]) continue;
    ans+=b[i]; last=b[i];
  }
  cout<<ans<<'\n';
  return 0;
}
