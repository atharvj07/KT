n = int(input())
s = input()
t = input()
mn, ans = n, (-1, -1)
for i in range(n):
  for j in range(n):
    if i != j and s[i] != t[j]:
      cnt = 0
      for k in range(n):
        if k != i and k != j and s[k] != t[k]:
          cnt += 1
      if cnt < mn:
        mn = cnt
        ans = i, j
print(mn)
print(*ans)
