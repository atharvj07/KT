r, c, n, k = map(int, input().split())
v = [[False]*c for i in range(r)]
for i in range(n):
    x, y = map(int, input().split())
    v[x-1][y-1] = True
ans = 0
for x1 in range(r):
    for y1 in range(c):
        for x2 in range(x1, r):
            for y2 in range(y1, c):
                if sum(sum(v[i][j] for j in range(y1, y2 + 1)) for i in range(x1, x2 + 1)) >= k:
                    ans += 1
print(ans)
