import sys, functools

input = lambda: sys.stdin.readline().strip()

memo = {}
def mex(s):
    if s not in memo:
        x = [0]*3
        for c in s:
            x[int(c)] = 1
        for i in range(3):
            if not x[i]:
                return i
        memo[s] = 3
    return memo[s]

for _ in range(int(input())):
    print(functools.reduce(lambda a,b: a+mex(b), [s for s in input()], 0))
