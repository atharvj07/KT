#include <cstdio>
#include <vector>
#include <algorithm>
using u32 = unsigned;
using i32 = int;
using v32 = std::vector<i32>;
using vv32 = std::vector<v32>;
const i32 MAXN{3000};

i32 solve(i32 n, i32 v, const vv32& t) {
    v32 dp(MAXN + 1, 0);
    for (i32 i = 0; i < n; i++) {
        i32 start{t[i][0]};
        i32 end{start + 1};
        for (i32 j = end; j <= MAXN; j++)
            dp[j] = std::max(dp[j], dp[j - 1]);
        for (i32 j = start; j <= MAXN; j++)
            dp[j] = std::max(dp[j], dp[std::max(0, j - 2)] + std::min(v, t[i][1]));
    }
    return dp[MAXN];
}

int main() {
    i32 n, v;
    scanf("%d%d", &n, &v);
    vv32 t(n, v32(2));
    for (i32 i = 0; i < n; i++) {
        scanf("%d%d", &t[i][0], &t[i][1]);
    }
    printf("%d\n", solve(n, v, t));
}
