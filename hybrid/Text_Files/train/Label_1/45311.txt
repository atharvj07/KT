import java.util.*;
import java.io.*;

public class SupersonicRocket {

    static class Point {
        double x, y;

        public Point(double x, double y) {
            this.x = x;
            this.y = y;
        }

        public Point rotate(double angle) {
            double newX = x * Math.cos(angle) - y * Math.sin(angle);
            double newY = x * Math.sin(angle) + y * Math.cos(angle);
            return new Point(newX, newY);
        }

        public Point translate(double dx, double dy) {
            return new Point(x + dx, y + dy);
        }

        public double distance(Point p) {
            return Math.sqrt((x - p.x) * (x - p.x) + (y - p.y) * (y - p.y));
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Point point = (Point) o;
            return Double.compare(point.x, x) == 0 && Double.compare(point.y, y) == 0;
        }

        @Override
        public int hashCode() {
            return Objects.hash(x, y);
        }

        @Override
        public String toString() {
            return "(" + x + ", " + y + ")";
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());

        Point[] engine1 = new Point[n];
        for (int i = 0; i < n; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int x = Integer.parseInt(tokenizer.nextToken());
            int y = Integer.parseInt(tokenizer.nextToken());
            engine1[i] = new Point(x, y);
        }

        Point[] engine2 = new Point[m];
        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int x = Integer.parseInt(tokenizer.nextToken());
            int y = Integer.parseInt(tokenizer.nextToken());
            engine2[i] = new Point(x, y);
        }

        boolean safe = isSafe(engine1, engine2);
        writer.write(safe ? "YES" : "NO");
        writer.newLine();
        writer.flush();
    }

    static boolean isSafe(Point[] engine1, Point[] engine2) {
        Set<Point> combinedPowerSources = new HashSet<>(Arrays.asList(engine1));
        combinedPowerSources.addAll(Arrays.asList(engine2));

        // Check if any point can be removed without changing the combined power field
        for (Point p : combinedPowerSources) {
            Set<Point> newCombinedPowerSources = new HashSet<>(combinedPowerSources);
            newCombinedPowerSources.remove(p);

            if (!isPowerFieldEqual(combinedPowerSources, newCombinedPowerSources)) {
                return false;
            }
        }

        return true;
    }

    static boolean isPowerFieldEqual(Set<Point> powerField1, Set<Point> powerField2) {
        for (Point p1 : powerField1) {
            boolean found = false;
            for (Point p2 : powerField2) {
                if (p1.distance(p2) < 1e-6) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                return false;
            }
        }
        return true;
    }
}
