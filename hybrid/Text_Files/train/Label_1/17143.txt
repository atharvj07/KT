#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 505;
int n, k, a[maxn], b[maxn], sum[maxn];
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        sum[i] = a[i] + b[i];
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            if (j <= a[i]) ans = max(ans, j);
            if (j <= b[i]) ans = max(ans, j);
            if (j <= sum[i]) ans = max(ans, j);
        }
    }
    cout << ans << endl;
    return 0;
}
