import sys

n = int(sys.stdin.readline())
edges = []
for _ in range(n):
    a, b = map(int, sys.stdin.readline().split())
    edges.append((a, b))

if len(set(edges)) != len(edges):
    print(-1)
    exit()

colors = [1, 2] * n
for a, b in edges:
    if colors[a - 1] == colors[b - 1]:
        print(-1)
        exit()

for a, b in edges:
    print(colors[a - 1], colors[b - 1])
