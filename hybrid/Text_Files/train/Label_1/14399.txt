#include <iostream>
#include <vector>
#include <set>

using namespace std;

bool isNearlyPrime(int n) {
    if (n == 1) {
        return false;
    }
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            while (n % i == 0) {
                n /= i;
            }
            if (n == 1) {
                return false;
            }
            return true;
        }
    }
    return true;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        if (n < 10 || n % 2 != 0) {
            cout << "NO" << endl;
            continue;
        }
        vector<int> ans;
        bool found = false;
        for (int a = 2; a <= n / 4; a++) {
            if (isNearlyPrime(a)) {
                for (int b = a + 1; b <= n / 3; b++) {
                    if (isNearlyPrime(b)) {
                        for (int c = b + 1; c <= n / 2; c++) {
                            if (isNearlyPrime(c)) {
                                ans = {a, b, c, n - a - b - c};
                                if (isNearlyPrime(ans.back())) {
                                    found = true;
                                    break;
                                }
                            }
                        }
                    }
                    if (found) {
                        break;
                    }
                }
            }
            if (found) {
                break;
            }
        }
        if (found) {
            cout << "YES" << endl;
            for (int x : ans) {
                cout << x << " ";
            }
            cout << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
