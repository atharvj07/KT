#include<iostream>
#include<vector>
using namespace std;
int main() {
    int n, m, i, ans = -1;
    vector<int> a, b;
    cin >> n >> m;
    for (i = 1; i <= n; i++) {
        int tmp;
        cin >> tmp;
        a.push_back(tmp);
    }
    for (i = 1; i <= m; i++) {
        int tmp;
        cin >> tmp;
        b.push_back(tmp);
    }
    for (i = 1; i <= 100; i++) {
        int ok1 = 1, ok2 = 0;
        for (auto p : a) {
            if (p > i)ok1 = 0;
            else if (p * 2 <= i)ok2 = 1;
        }
        for (auto p : b)if (p <= i)ok1 = 0;
        if (ok1 && ok2) {
            ans = i;
            break;
        }
    }
    cout << ans << "\n";
}
