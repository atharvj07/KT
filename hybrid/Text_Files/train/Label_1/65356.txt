q = (pow(5, 0.5) + 1) / 2

def golden_to_decimal(golden_str):
    decimal_value = 0
    n = len(golden_str)
    for i in range(n):
        if golden_str[i] == '1':
            decimal_value += q ** (n - i - 1)
    return decimal_value

first_number = input().rstrip('0')
second_number = input().rstrip('0')

first_decimal = golden_to_decimal(first_number)
second_decimal = golden_to_decimal(second_number)

if first_decimal > second_decimal:
    print(">")
elif first_decimal < second_decimal:
    print("<")
else:
    print("=")
