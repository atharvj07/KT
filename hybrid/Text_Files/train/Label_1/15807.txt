#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int x = 0, y = 0;
        vector<int> x_left(n), y_up(n);
        for (int i = 0; i < n; i++) {
            if (s[i] == 'L')
                x--;
            else if (s[i] == 'R')
                x++;
            else if (s[i] == 'U')
                y++;
            else
                y--;
            x_left[i] = x;
            y_up[i] = y;
        }
        int min_len = n + 1;
        int l = -1, r = -1;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int cur_len = j - i + 1;
                if (cur_len >= min_len)
                    break;
                int cur_x = x_left[j];
                int cur_y = y_up[j];
                if (i > 0) {
                    cur_x -= x_left[i - 1];
                    cur_y -= y_up[i - 1];
                }
                if (cur_x == 0 && cur_y == 0) {
                    min_len = cur_len;
                    l = i;
                    r = j;
                }
            }
        }
        if (min_len == n + 1)
            cout << -1 << endl;
        else
            cout << l + 1 << ' ' << r + 1 << endl;
    }
    return 0;
}
