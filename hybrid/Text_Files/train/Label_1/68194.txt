import math

n = int(input())
M = [list(map(int, input().split())) for _ in range(n)]

a = [0] * n  # Initialize array a with zeros

# Calculate gcd of all elements in the first row (excluding the first element)
gcd = M[0][1]
for i in range(2, n):
    gcd = math.gcd(gcd, M[0][i])

# Calculate each element of array a
a[0] = int(math.sqrt(M[0][1] * M[1][0] / gcd))
for i in range(1, n):
    a[i] = M[0][i] // a[0]

print(' '.join(map(str, a)))
