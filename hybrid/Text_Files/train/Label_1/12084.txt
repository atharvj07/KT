#include <vector>
#include <iostream>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    for (int g = 0; g < 64; g++) {
        vector<int> ans(n, 0);
        int cnt = 0, t = (1LL << g);
        for (int i = 0; i < n; i++) {
            if (a[i] & t) {
                ans[i] = 1;
                cnt++;
            }
        }
        if (cnt > 0 && cnt != n) {
            cout << "YES" << endl;
            for (int v : ans) cout << v << " ";
            cout << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
}
