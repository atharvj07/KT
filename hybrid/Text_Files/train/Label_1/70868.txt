def game_winner(n, k, strings):
    trie = {}
    for s in strings:
        node = trie
        for char in s:
            if char not in node:
                node[char] = {}
            node = node[char]
    
    def dfs(node):
        children = 0
        can_win, must_lose = False, True
        for child in node.values():
            children += 1
            child_win, child_lose = dfs(child)
            can_win |= child_lose
            must_lose &= child_win
        if children == 0:  # Leaf node
            return False, True
        return can_win, not must_lose

    can_win, must_lose = dfs(trie)
    if can_win and must_lose:
        return "First" if k % 2 == 1 else "Second"
    elif can_win:
        return "First"
    else:
        return "Second"

n, k = map(int, input().split())
strings = [input() for _ in range(n)]
print(game_winner(n, k, strings))
