import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt();
        int yv = sc.nextInt();
        int xp = sc.nextInt();
        int yp = sc.nextInt();
        int xw1 = sc.nextInt();
        int yw1 = sc.nextInt();
        int xw2 = sc.nextInt();
        int yw2 = sc.nextInt();
        int xm1 = sc.nextInt();
        int ym1 = sc.nextInt();
        int xm2 = sc.nextInt();
        int ym2 = sc.nextInt();
        if (xv == xp && yv == yp) {
            System.out.println("NO");
        } else if ((xw1 <= xv && xv <= xw2 && yw1 <= yv && yv <= yw2) || (xw1 <= xp && xp <= xw2 && yw1 <= yp && yp <= yw2)) {
            System.out.println("NO");
        } else if ((xm1 <= xv && xv <= xm2 && ym1 <= yv && yv <= ym2) || (xm1 <= xp && xp <= xm2 && ym1 <= yp && yp <= ym2)) {
            if (xv * (ym2 - ym1) == xp * (ym2 - ym1) + (xv - xp) * (xm2 - xm1)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        } else {
            System.out.println("YES");
        }
    }
}
