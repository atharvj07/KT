#include <iostream>
#include <vector>
using namespace std;

struct Ans { int cnt; vector<int> idx; };

int main()
{
    int m;
    cin >> m;
    vector<long long> a(m);
    vector<Ans> ans(m);
    for (int i = 0; i < m; ++i)
        cin >> a[i];
    for (int i = 0; i < m; ++i)
    {
        long long x = a[i];
        vector<int> idx1, idx2;
        for (int j = 0; j < i; ++j)
        {
            if (a[j] == x)
            {
                if (idx1.empty())
                    idx1.push_back(j);
                idx2.push_back(j);
            }
            else if (x > a[j])
            {
                long long diff = x ^ a[j];
                if (idx1.size() && diff == a[idx1[0]])
                    idx1.push_back(j);
            }
        }
        int si1 = idx1.size(), si2 = idx2.size();
        if (si1 >= 1)
            ans[i] = {si1 + si2, idx1};
        else
            ans[i] = {si2, idx2};
        if (si1 + si2)
            for (int j = 0; j < i; ++j)
                if (ans[j].cnt >= ans[i].cnt && ans[j].idx != idx1)
                    ans[i] = {ans[j].cnt, ans[j].idx};
    }
    for (int i = 0; i < m; ++i)
    {
        if (!ans[i].cnt)
            cout << 0 << '\n';
        else
        {
            cout << ans[i].cnt;
            for (auto x : ans[i].idx)
                cout << ' ' << x + 1;
            cout << '\n';
        }
    }
    return 0;
}
