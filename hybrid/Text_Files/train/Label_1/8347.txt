#include <bits/stdc++.h>
using namespace std;
struct p {
  int t;
  double x;
};
int main() {
  int n, m, ans = 0, i, lst = 0;
  vector<p> v;
  p a;
  cin >> n >> m;
  for (i = 0; i < n; i++) {
    cin >> a.t >> a.x;
    v.push_back(a);
  }
  for (i = 0; i < n; i++) {
    if (v[i].t != v[lst].t) {
      if (i - lst > m - (m - (v[i].t - v[lst].t)))
        ans += i - lst - (m - (v[i].t - v[lst].t));
      lst = i;
    }
  }
  cout << ans << endl;
  return 0;
}
