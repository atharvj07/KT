#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> cnt(6);
    for (int x : a) cnt[x / 4 - 1]++;
    int ans = 0;
    for (int i = 0; i < 6; i++) ans += max(0, cnt[i] - n / 6);
    cout << ans << '\n';
}
