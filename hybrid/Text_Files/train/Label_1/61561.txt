n, k = map(int, input().split())
a = list(map(int, input().split()))

stress_count = [0] * (k + 1)
for i in range(1, n):
    if a[i] != a[i - 1]:
        stress_count[a[i]] += 1
        stress_count[a[i - 1]] += 1

# Since each genre occurs at least once, the first and last movies can potentially reduce stress if their genres are skipped
stress_count[a[0]] -= 1
stress_count[a[-1]] -= 1

print(stress_count.index(min(stress_count[1:])))
