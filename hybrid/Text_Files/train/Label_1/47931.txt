import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        BigInteger[] heights = new BigInteger[n];
        for (int i = 0; i < n; i++) {
            heights[i] = input.nextBigInteger();
        }
        BigInteger min = heights[n - 1];
        int jumps = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (heights[i].compareTo(min) > 0) {
                jumps++;
            } else {
                min = heights[i];
            }
        }
        System.out.println(jumps);
    }
}
