from math import gcd
from collections import defaultdict

def is_perfect_square(n):
    return int(n**0.5) ** 2 == n

def lcm(x, y):
    return x * y // gcd(x, y)

def adjacent(x, y):
    return is_perfect_square(lcm(x, y) // gcd(x, y))

def process_array(a):
    adj_dict = defaultdict(set)
    for i, val in enumerate(a):
        for j, comp_val in enumerate(a):
            if adjacent(val, comp_val):
                adj_dict[i].add(j)
    return adj_dict

def beauty_of_array(a, adj_dict, w):
    if w == 0:
        return max(len(adj_dict[i]) for i in range(len(a)))
    if w >= 2:
        return len(a)
    new_a = [1] * len(a)
    for i in range(len(a)):
        for j in adj_dict[i]:
            new_a[i] *= a[j]
    adj_dict_new = process_array(new_a)
    return max(len(adj_dict_new[i]) for i in range(len(new_a)))

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    q = int(input().strip())
    adj_dict = process_array(a)
    for __ in range(q):
        w = int(input().strip())
        print(beauty_of_array(a, adj_dict, w))
