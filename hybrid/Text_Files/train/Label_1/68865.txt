n = int(input())
card_numbers = [int(input()) for _ in range(n)]
card_count = {}

for number in card_numbers:
    if number in card_count:
        card_count[number] += 1
    else:
        card_count[number] = 1

if len(card_count) == 2 and all(count == n // 2 for count in card_count.values()):
    print("YES")
    print(*card_count.keys())
else:
    print("NO")
