import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextInt();
        long x = in.nextInt();
        long y = in.nextInt();
        if (x < y) {
            System.out.println(((n + 1) / 2) * x + ((n - 1) / 2) * y);
        } else {
            System.out.println(n * x);
        }
    }
}
