import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;

public class Main {
    static long maxVolume;
    static int[] bestTwoStones;
    static List<long[]> stones;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        stones = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            long[] stone = Arrays.stream(reader.readLine().split(" ")).mapToLong(Long::parseLong).toArray();
            Arrays.sort(stone);
            stones.add(stone);
        }

        maxVolume = 0;
        Collections.sort(stones, (a, b) -> Long.compare(a[0] * a[1] * a[2], b[0] * b[1] * b[2]));

        for (int i = 0; i < n - 1; i++) {
            long[] firstStone = stones.get(i);
            for (int j = i + 1; j < n; j++) {
                long[] secondStone = stones.get(j);
                if (firstStone[1] == secondStone[0] && firstStone[2] == secondStone[1]) {
                    long newVolume = (firstStone[0] * firstStone[1] * firstStone[2]) + (secondStone[0] * secondStone[1] * secondStone[2]);
                    if (newVolume > maxVolume) {
                        maxVolume = newVolume;
                        bestTwoStones = new int[]{i + 1, j + 1};
                    }
                } else if (firstStone[1] == secondStone[2] && firstStone[2] == secondStone[0]) {
                    long newVolume = (firstStone[0] * firstStone[1] * firstStone[2]) + (secondStone[0] * secondStone[1] * secondStone[2]);
                    if (newVolume > maxVolume) {
                        maxVolume = newVolume;
                        bestTwoStones = new int[]{i + 1, j + 1};
                    }
                } else if (firstStone[0] == secondStone[1] && firstStone[2] == secondStone[2]) {
                    long newVolume = (firstStone[0] * firstStone[1] * firstStone[2]) + (secondStone[0] * secondStone[1] * secondStone[2]);
                    if (newVolume > maxVolume) {
                        maxVolume = newVolume;
                        bestTwoStones = new int[]{i + 1, j + 1};
                    }
                } else if (firstStone[0] == secondStone[2] && firstStone[1] == secondStone[1]) {
                    long newVolume = (firstStone[0] * firstStone[1] * firstStone[2]) + (secondStone[0] * secondStone[1] * secondStone[2]);
                    if (newVolume > maxVolume) {
                        maxVolume = newVolume;
                        bestTwoStones = new int[]{i + 1, j + 1};
                    }
                }
            }
        }

        long maxValue = 0;
        int bestOneStone = -1;
        for (int i = 0; i < n; i++) {
            long[] stone = stones.get(i);
            long volume = stone[0] * stone[1] * stone[2];
            if (volume > maxValue) {
                maxValue = volume;
                bestOneStone = i + 1;
            }
        }

        if (maxVolume > maxValue) {
            System.out.println(2);
            System.out.println(bestTwoStones[0] + " " + bestTwoStones[1]);
        } else {
            System.out.println(1);
            System.out.println(bestOneStone);
        }
    }
}
