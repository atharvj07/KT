#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
  string s;
  cin >> s;
  ll ans = 0, hash = 0, pw = 1;
  unordered_map<ll, ll> mp;
  mp[0] = 1;
  for (int i = s.size() - 1; i >= 0; i--) {
    if (s[i] >= '0' && s[i] <= '9')
      hash = (hash + (s[i] - '0') * pw) % mod;
    else if (s[i] >= 'a' && s[i] <= 'z')
      hash = (hash + (s[i] - 'a' + 10) * pw) % mod;
    else if (s[i] >= 'A' && s[i] <= 'Z')
      hash = (hash + (s[i] - 'A' + 36) * pw) % mod;
    else if (s[i] == '-')
      hash = (hash + 62 * pw) % mod;
    else
      hash = (hash + 63 * pw) % mod;
    pw = pw * 64LL % mod;
    ans += mp[ans & hash]++;
    mp[hash]++;
  }
  cout << ans << '\n';
  return 0;
}
