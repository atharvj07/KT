#include <iostream>
#include <vector>

int ones, twos;

void seat(int t) {
    if (ones) {
        ones--; return;
    } else if (twos) {
        twos--; return;
    } else if (t > 1) {
        twos++; return;
    } else {
        ones++; return;
    }
}

int main() {
    int n, a, b;
    std::cin >> n >> a >> b;
    std::vector<int> t(n);
    for (auto &i : t) std::cin >> i;
    ones = a; twos = b;
    for (int i = 0; i < n; i++) seat(t[i]);
    std::cout << ones;
    return 0;
}
