a, b, c = map(int, input().split())
x, y, z = map(int, input().split())

excess = 0
if a > x:
    excess += (a - x) // 2
if b > y:
    excess += (b - y) // 2
if c > z:
    excess += (c - z) // 2

needed = max(0, x - a) + max(0, y - b) + max(0, z - c)

if excess >= needed:
    print("Yes")
else:
    print("No")
