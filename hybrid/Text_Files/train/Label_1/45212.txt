import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        int x1 = sc.nextInt();
        int y1 = sc.nextInt();
        int dx = x1 - x;
        int dy = y1 - y;
        if (dx * dx + dy * dy <= r * r)
            System.out.println(0);
        else
            System.out.println(2);
    }
}
