#include <bits/stdc++.h>
using namespace std;

const int MX = 100001;
int dp[MX];
int X[MX], C[MX];

int solve(int i, int j, int sum, int n, int k, int a) {
  if (i == n) {
    if (sum >= k) return j;
    return -1;
  }
  int &ans = dp[i * MX + j];
  if (ans != -1) return ans;
  ans = -1;
  if (sum >= X[i]) ans = solve(i + 1, j, sum, n, k, a);
  if (ans == -1) {
    if (j + C[i] < k) {
      ans = solve(i + 1, j + C[i], sum + a, n, k, a);
    }
  }
  if (ans == -1) {
    ans = solve(i + 1, j, sum + a, n, k, a);
  }
  return ans;
}

int main() {
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> X[i];
  int a;
  cin >> a;
  for (int i = 0; i < n; i++) cin >> C[i];
  memset(dp, -1, sizeof dp);
  int ans = solve(0, 0, 0, n, k, a);
  cout << ans << '\n';
  return 0;
}
