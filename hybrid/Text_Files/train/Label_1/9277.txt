#include <deque>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  deque<int> dq;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] == 1) {
      dq.push_back(i);
    }
  }
  while (!dq.empty()) {
    int i = dq.front(), j = dq.back();
    if (i == j) break;
    dq.pop_front();
    dq.pop_back();
    if (a[i] > a[j]) swap(i, j);
    if (j - i != 1) {
      cout << "NO\n";
      return 0;
    }
    dq.push_front(i - 1);
    dq.push_back(j + 1);
  }
  cout << "YES\n";
}
