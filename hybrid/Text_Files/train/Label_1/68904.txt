import sys, math

def gcd(a, b):
    while b:
        a, b = b, a%b
    return a

for _ in range(int(input())):
    n, p = map(int, input().strip().split())
    a = [int(i) for i in input().strip().split()]
    m = 0
    g = [0] * n
    for i in range(n - 1):
        x = a[i]
        for j in range(i + 1, n):
            x = gcd(x, a[j])
            if x < m:
                break
        g[i + 1] = max(g[i + 1], x)
    ans = 1e18
    for i in range(n - 1):
        if not i or g[i + 1] < m:
            ans = min(ans, p)
        else:
            ans = min(ans, g[i + 1])
    for i in range(1, n):
        m = max(m, a[i - 1])
        ans = min(ans, m)
    print(ans)
