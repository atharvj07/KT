#include <bits/stdc++.h>

using namespace std;

#define int long long

const int MAXN = 100005;
string s;
int n, q, l, r, cnt[MAXN];

int32_t main() {
    cin >> s >> q;
    for (int i = 0; i < s.size(); i++)
        cnt[i + 1] = cnt[i] + (s[i] != s[i + 1]);
    while (q--) {
        int t;
        cin >> t;
        if (t == 1) {
            int pos;
            char c;
            cin >> pos >> c;
            int l = pos - 1, r = pos;
            while (l >= 1 && s[l] == s[l - 1])
                l--;
            while (r < n && s[r] == s[r + 1])
                r++;
            cnt[l] += (s[l] != s[l - 1]);
            cnt[r + 1] -= (s[r] != s[r + 1]);
            s[pos - 1] = c;
        } else {
            cin >> l >> r;
            cout << cnt[r] - cnt[l] + 1 << endl;
        }
    }
    return 0;
}
