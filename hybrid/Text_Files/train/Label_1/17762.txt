#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
vector<int> get(string s) {
    vector<int> v(30, 0);
    for (char c : s) { v[c - 'S']++; }
    return v;
}
int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    int n, k;
    cin >> n >> k;
    vector<vector<int>> a(n, vector<int>(k, 0));
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        a[i] = get(s);
    }
    ll ans = 0;
    for (int i = 0; i < 30; i++) {
        for (int j = i + 1; j < 30; j++) {
            for (int l = j + 1; l < 30; l++) {
                int cnt1 = 0, cnt2 = 0, cnt3 = 0;
                for (auto &x : a) {
                    bool t1 = x[i] == x[j] && x[j] == x[l];
                    bool t2 = x[i] != x[j] && x[j] != x[l] && x[i] != x[l];
                    cnt1 += t1;
                    cnt2 += !t1 && !t2;
                    cnt3 += t2;
                }
                ll cur = 1ll * cnt1 * (cnt1 - 1) * (cnt1 - 2) / 6;
                cur += 1ll * cnt2 * (cnt2 - 1) * (cnt2 - 2) / 6;
                cur += 1ll * cnt3 * (cnt3 - 1) * (cnt3 - 2) / 6;
                cur += 1ll * cnt1 * cnt2 * cnt3;
                ans += cur;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
