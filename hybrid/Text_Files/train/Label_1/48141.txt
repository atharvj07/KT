import java.util.*;
import java.io.*;

class Main {
    boolean[][] vis;
    int[][] a;
    int n, m;
    char[][] c;

    int[] dx = {-1, 0, 1, 0};
    int[] dy = {0, -1, 0, 1};

    public void dfs(int x, int y) {
        vis[x][y] = true;
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx < 0 || nx >= n || ny < 0 || ny >= m) {
                continue;
            }
            if (a[nx][ny] == 1 || vis[nx][ny]) {
                continue;
            }
            dfs(nx, ny);
        }
    }

    public void solve() {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        int k = in.nextInt();
        vis = new boolean[n][m];
        a = new int[n][m];
        c = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < m; j++) {
                if (s.charAt(j) == '#') {
                    a[i][j] = 1;
                }
                c[i][j] = s.charAt(j);
            }
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 0 && !vis[i][j]) {
                    cnt++;
                    dfs(i, j);
                }
            }
        }
        cnt -= k;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 0 && !vis[i][j] && cnt > 0) {
                    cnt--;
                    c[i][j] = 'X';
                    a[i][j] = 1;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(c[i][j]);
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        new Main().solve();
    }
}
