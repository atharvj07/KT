#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  string t;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'e' || s[i] == 'o') {
      if (t.size() >= 2 && s[i] == t[t.size() - 1] &&
          t[t.size() - 1] == t[t.size() - 2])
        continue;
    }
    t.push_back(s[i]);
  }

  string u;
  for (int i = 0; i < t.size(); i++) {
    if (i + 1 < t.size() && t[i] == t[i + 1])
      continue;
    u.push_back(t[i]);
  }

  cout << u << endl;

  return 0;
}
