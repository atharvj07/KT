import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = scanner.nextInt();
        }
        Arrays.sort(d);
        int s = d[0];
        for (int i = 1; i < n; i++) {
            s = Math.min(s, d[i] - d[i - 1]);
        }
        System.out.println(s);
    }
}
