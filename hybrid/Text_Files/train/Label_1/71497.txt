import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    t = min((m + 1) // 2, n)
    cnt = [0 for _ in range(n + 1)]
    ll = [[] for _ in range(m)]
    for i in range(m):
        _, *l = map(int, input().split())
        ll[i] = [*map(int, l)]
    ans = []
    for i in range(m):
        l = ll[i]
        ls = []
        for j in l:
            if cnt[j] < t:
                ls.append(j)
        if not ls:
            print('NO')
            break
        j = ls[0]
        ans.append(j)
        cnt[j] += 1
    else:
        print('YES')
        print(*ans)
