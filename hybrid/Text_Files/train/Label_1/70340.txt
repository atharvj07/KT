import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n,m,k = map(int,input().split())
    a = [0] + list(map(int,input().split()))
    a.sort()
    if k == 0: l = 1
    else: l = m - k
    r = min(n - k, m + k)
    ans = []
    for i in range(l,r+1):
        ans.append(min(a[i],a[n-i+1]))
    ans.sort()
    print(ans[-1])
