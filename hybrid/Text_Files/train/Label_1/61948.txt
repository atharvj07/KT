n, m = map(int, input().split())
grid = [input() for _ in range(n)]

# Precompute exitability grid
exitable = [['N'] * m for _ in range(n)]
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            exitable[i][j] = 'E'
            if i > 0:
                exitable[i][j] = exitable[i-1][j]
            if j > 0:
                exitable[i][j] = exitable[i][j-1]

q = int(input())
for _ in range(q):
    x1, x2 = map(int, input().split())
    x1 -= 1
    x2 -= 1
    determinable = True
    for i in range(1, n):
        # If there is a change in exitability from not exitable to exitable in the range, it's not determinable
        if exitable[i-1][x2] == 'E' and exitable[i][x2] == 'N':
            determinable = False
            break
    print("YES" if determinable else "NO")
