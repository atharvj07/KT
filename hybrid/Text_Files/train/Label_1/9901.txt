#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    string str;
    cin >> str;
    int n = str.size();
    int cost = 0;
    vector<int> mines;
    for (int i = 0; i < n; i++) {
      if (str[i] == '1') mines.push_back(i);
    }
    int len = mines.size();
    if (len == 0) {
      cost = 0;
    } else if (len == 1) {
      cost = a;
    } else {
      cost += a;
      for (int i = 1; i < len - 1; i++) { cost += min(a, b); }
      if (mines[0] == 0) cost += b;
      if (mines[len - 1] == n - 1) cost += b;
    }
    cout << cost << endl;
  }
}
