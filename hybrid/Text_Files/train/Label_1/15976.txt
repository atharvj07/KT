#include <vector>
#include <iostream>
#include <unordered_map>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v;
    unordered_map<int, bool> m;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        if (m[a] == false) {
            v.push_back(a);
        }
        m[a] = true;
        for (int j = 0; j < v.size(); j++) {
            m[a + v[j]] = true;
        }
    }
    cout << v.size() << "\n";
    for (auto& x : v) {
        cout << x << " ";
    }
    cout << endl;
}
