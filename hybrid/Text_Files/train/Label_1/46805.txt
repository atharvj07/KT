import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int k = Integer.parseInt(input[1]);
        int s = Integer.parseInt(input[2]);
        int t = Integer.parseInt(input[3]);

        input = new String[n];
        for (int i = 0; i < n; i++) {
            input[i] = br.readLine();
        }

        String[] gas = br.readLine().split(" ");
        Arrays.sort(gas);

        long ans = Long.MAX_VALUE;
        for (String car : input) {
            String[] carData = car.split(" ");
            int c = Integer.parseInt(carData[0]);
            int v = Integer.parseInt(carData[1]);

            int nGas = 0;
            int f = 0;
            long time = 0;

            for (int i = 0; i < k; i++) {
                int d = Integer.parseInt(gas[i]);
                int dist = i == 0 ? d : d - Integer.parseInt(gas[i - 1]);
                while (f + dist > v) {
                    f -= v - f;
                    nGas++;
                    time += 1;
                    f += v;
                }
                f -= dist;
                time += dist;
            }

            int d = s - Integer.parseInt(gas[k - 1]);
            while (f + d > v) {
                f -= v - f;
                nGas++;
                time += 1;
                f += v;
            }
            f -= d;
            time += d;

            if (f > 0) {
                time += (f + 1) / 2;
            }

            if (time <= t && nGas % 2 == 0) {
                ans = Math.min(ans, c);
            }
        }

        System.out.println(ans == Long.MAX_VALUE ? -1 : ans);
    }
}
