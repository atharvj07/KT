import math
x1, y1, r1 = map(int, input().split())
x2, y2, r2 = map(int, input().split())
dist = math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
if dist > r1 + r2:
    print(0)
elif dist + min(r1, r2) < max(r1, r2):
    print(0)
else:
    a = (r1 * r1 - r2 * r2 + dist * dist) / (2 * dist)
    h = math.sqrt(r1 * r1 - a * a)
    area1 = a * h
    print(area1)
