import java.util.Scanner;

public class Main {
    public static final int MOD = 100000000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n1 = sc.nextInt();
        int n2 = sc.nextInt();
        int k1 = sc.nextInt();
        int k2 = sc.nextInt();
        long[][] dp = new long[n1 + 1][n2 + 1];
        dp[0][0] = 1;
        for (int i = 0; i <= n1; i++) {
            for (int j = 0; j <= n2; j++) {
                if (i < n1) {
                    int next = i + k1 + 1;
                    if (next < n1 + 1) dp[next][j] = (dp[next][j] + dp[i][j]) % MOD;
                    next = i + 1;
                    if (next < n1 + 1) dp[next][j] = (dp[next][j] + dp[i][j]) % MOD;
                }
                if (j < n2) {
                    int next = j + k2 + 1;
                    if (next < n2 + 1) dp[i][next] = (dp[i][next] + dp[i][j]) % MOD;
                    next = j + 1;
                    if (next < n2 + 1) dp[i][next] = (dp[i][next] + dp[i][j]) % MOD;
                }
            }
        }
        System.out.println(dp[n1][n2]);
    }
}
