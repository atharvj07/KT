import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> w = new ArrayList<>();
        int currentLength = 0;
        int lastRemoved = 0;

        for (int i = 0; i < n; i++) {
            int event = scanner.nextInt();
            if (event == -1) {
                int k = 0;
                while (k < m && a[k] <= currentLength - lastRemoved) {
                    k++;
                }
                lastRemoved += k;
            } else {
                w.add(event);
                currentLength++;
            }
        }

        if (currentLength == 0) {
            System.out.println("Poor stack!");
        } else {
            for (int i = lastRemoved; i < currentLength; i++) {
                System.out.print(w.get(i));
            }
        }
    }
}
