#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll t[150005], lazy[150005], seg[4 * 150005], ans[150005];
ll n, k, q, val;
void relax(ll id) {
  seg[id * 2 + 1] += lazy[id];
  seg[id * 2 + 2] += lazy[id];
  lazy[id * 2 + 1] += lazy[id];
  lazy[id * 2 + 2] += lazy[id];
  lazy[id] = 0;
}
void upd(ll l, ll r, ll id, ll lq, ll rq, ll val) {
  if (l >= lq && r <= rq) {
    lazy[id] += val;
    seg[id] += val;
    return;
  }
  if (l > rq || r < lq) return;
  relax(id);
  upd(l, (l + r) / 2, id * 2 + 1, lq, rq, val);
  upd((l + r) / 2 + 1, r, id * 2 + 2, lq, rq, val);
  seg[id] = min(seg[id * 2 + 1], seg[id * 2 + 2]);
}
ll get(ll l, ll r, ll id, ll lq, ll rq) {
  if (l >= lq && r <= rq) {
    return seg[id];
  }
  if (l > rq || r < lq) return 1e9;
  relax(id);
  return min(get(l, (l + r) / 2, id * 2 + 1, lq, rq),
             get((l + r) / 2 + 1, r, id * 2 + 2, lq, rq));
}
pair<bool, ll> query(ll x) {
  x--;
  ll l = 0, r = k - 1;
  ll res = 1e9;
  while (l <= r) {
    ll mid = (l + r) / 2;
    if (get(0, n - 1, 0, mid, x) <= t[x])
      res = mid, l = mid + 1;
    else
      r = mid - 1;
  }
  if (res < k)
    return {true, res + 1};
  else
    return {false, -1};
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> k >> q;
  for (ll i = 0; i < n; i++) {
    cin >> t[i];
    upd(0, n - 1, 0, i, i, t[i]);
  }
  while (q--) {
    ll x, y;
    cin >> x >> y;
    if (x == 1)
      upd(0, n - 1, 0, y - 1, y - 1, 1e9);
    else if (x == 2) {
      pair<bool, ll> ans = query(y);
      if (ans.first)
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
    }
  }
}
