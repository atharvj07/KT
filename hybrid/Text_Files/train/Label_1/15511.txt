#include<bits/stdc++.h>
using namespace std;
int n;
vector<int>a;
vector<int>b;
int main(){
    cin>>n;
    a.resize(n-1),b.resize(n-1);
    for(int i=0;i<n-1;++i){
        cin>>a[i];
        cin>>b[i];
    }
    for(int i=0;i<(1<<4);++i){
        vector<int>t(n,i);
        //assert(t.size()==n);
        for(int j=1;j<n;++j){
            //assert(j<n);
            t[j]=t[j]|(t[j-1]&a[j-1]);
            t[j]=t[j]&((~t[j-1])|b[j-1]);
        }
        bool ok=1;
        for(int j=1;j<n;++j){
            if(a[j-1]!=(t[j]|t[j-1]))ok=0;
            if(b[j-1]!=(t[j]&t[j-1]))ok=0;
        }
        if(ok){
            cout<<"YES"<<endl;
            for(int j=0;j<n;++j){
                cout<<(t[j])<<" ";
            }
            cout<<endl;
            return 0;
        }
    }
    cout<<"NO"<<endl;
    return 0;
}
