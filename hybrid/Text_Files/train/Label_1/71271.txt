n = int(input())
a = list(map(int, input().split()))

odd = [x for x in a if x % 2 != 0]
even = [x for x in a if x % 2 == 0]

if len(odd) == len(even) or abs(len(odd) - len(even)) == 1:
    print(0)
else:
    odd.sort()
    even.sort()
    if len(odd) > len(even):
        diff = len(odd) - len(even) - 1
        print(sum(odd[:diff]))
    else:
        diff = len(even) - len(odd) - 1
        print(sum(even[:diff]))
