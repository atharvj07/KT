n, m = map(int, input().split())
grid = [input() for _ in range(n)]

# Find top-left and bottom-right corners of the puzzle piece
top_left = [n, m]
bottom_right = [0, 0]
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'X':
            top_left[0] = min(top_left[0], i)
            top_left[1] = min(top_left[1], j)
            bottom_right[0] = max(bottom_right[0], i)
            bottom_right[1] = max(bottom_right[1], j)

# Check if the piece between top-left and bottom-right corners forms a solid rectangle
can_form_rectangle = True
for i in range(top_left[0], bottom_right[0] + 1):
    for j in range(top_left[1], bottom_right[1] + 1):
        if grid[i][j] != 'X':
            can_form_rectangle = False
            break

print("YES" if can_form_rectangle else "NO")
