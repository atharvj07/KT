#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    int lastr = -1, lastl = -1;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'L') {
            if (lastl == -1) lastl = i;
        } else if (s[i] == 'R') {
            if (lastr == -1) lastr = i;
        } else {
            if (lastl > lastr) {
                if (lastl - lastr > 1) ans += lastl - lastr - 1;
                lastr = lastl = -1;
            } else {
                ans++;
                lastl = lastr = -1;
            }
        }
    }
    if (lastl > lastr) {
        if (lastl - lastr > 1) ans += lastl - lastr - 1;
    } else {
        ans++;
    }
    cout << ans << endl;
    return 0;
}
