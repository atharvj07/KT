import java.util.*;

public class Main {

    static class Edge {
        int to, weight;

        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static class State {
        int num;
        Set<Integer> cities;

        State(int num, Set<Integer> cities) {
            this.num = num;
            this.cities = cities;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt();
        List<List<Edge>> adjList = new ArrayList<>();
        for (int i = 0; i <= N; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            adjList.get(a).add(new Edge(b, 1));
            adjList.get(b).add(new Edge(a, 1));
        }
        int ans = -1;
        for (int i = 1; i <= N; i++) {
            Set<Integer> visited = new HashSet<>();
            if (dfs(i, i, adjList, visited, 0, new HashSet<>())) {
                ans = Math.min(ans == -1 ? Integer.MAX_VALUE : ans, visited.size());
            }
        }
        System.out.println(ans);
    }

    static boolean dfs(int cur, int start, List<List<Edge>> adjList, Set<Integer> visited, int num, Set<Integer> cities) {
        if (visited.contains(cur)) {
            return num == 1;
        }
        visited.add(cur);
        cities.add(cur);
        boolean ret = true;
        for (Edge next : adjList.get(cur)) {
            if (next.to == start) {
                continue;
            }
            ret &= dfs(next.to, start, adjList, visited, num + 1, cities);
        }
        return ret;
    }
}
