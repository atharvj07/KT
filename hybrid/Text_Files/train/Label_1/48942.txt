import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String[] words = new String[n];
        for (int i = 0; i < n; i++) {
            words[i] = input.next();
        }
        String message = input.next();
        String encoded = "<3" + String.join("<3", words) + "<3";
        System.out.println(message.contains(encoded) ? "yes" : "no");
    }
}
