import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        long[] s = new long[n], m = new long[n];
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextLong();
            m[i] = sc.nextLong();
        }
        for (long t = s[0]; n > 0; t++) {
            while (!pq.isEmpty() && pq.peek() <= t) {
                pq.poll();
            }
            while (n > 0 && pq.size() < k) {
                pq.add(t + m[n - 1]);
                n--;
            }
            if (n > 0) {
                pq.add(t + m[n - 1]);
                n--;
            }
            if (!pq.isEmpty()) {
                System.out.println(pq.peek());
            }
        }
    }
}
