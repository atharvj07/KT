def min_segments():
    points = [list(map(int, input().split())) for _ in range(3)]
    x = [p[0] for p in points]
    y = [p[1] for p in points]
    if len(set(x)) == 1 or len(set(y)) == 1:
        return 1
    elif (x[0] == x[1] and y[1] == y[2]) or (x[1] == x[2] and y[0] == y[1]) or (x[0] == x[2] and y[0] == y[2]):
        return 2
    elif (x[0] == x[1] or x[1] == x[2] or x[0] == x[2]) and (min(y) in y and max(y) in y):
        return 2
    elif (y[0] == y[1] or y[1] == y[2] or y[0] == y[2]) and (min(x) in x and max(x) in x):
        return 2
    else:
        return 3

print(min_segments())
