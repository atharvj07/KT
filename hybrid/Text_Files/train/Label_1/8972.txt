#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0, cur = 0;
  for (int i = 0; i < n / 2; i++) { cur += (s[i] == 'a' ? -1 : 1); }
  ans += abs(cur), cur = 0;
  for (int i = n / 2; i < n; i++) {
    cur += (s[i] == 'a' ? -1 : 1);
    if (cur == 0) continue;
    if ((n - i) % 2 == 0) {
      ans += abs(cur);
      cur = 0;
    }
  }
  cout << ans << endl;
  for (int i = 0; i < n / 2; i++) {
    if (cur != 0) s[i] = s[i] == 'a' ? 'b' : 'a', cur += (s[i] == 'a' ? -1 : 1);
    cout << s[i];
  }
  for (int i = n / 2; i < n; i++) cout << s[i];
}
