#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 5010;
const int MAX_K = 2500;

char s[MAX_N];
char t[MAX_K][MAX_N];
int p[MAX_N];

int f(int n, char *s) {
  for (int i = 0; i < n; ++i) { p[i] = i; }
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      if (s[i] > s[j]) { swap(p[i], p[j]); }
    }
  }
  return p[0];
}

int main() {
  int k, n;
  cin >> k >> n;
  for (int i = 0; i < k; ++i) { cin >> t[i]; }
  int pos = f(n, t[0]);
  for (int i = 0; i < k; ++i) {
    if (pos != f(n, t[i])) {
      cout << -1 << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < k; ++j) {
      if (p[i] == f(n, t[j])) { s[i] = t[j][i]; }
    }
  }
  cout << s << endl;
  return 0;
}
