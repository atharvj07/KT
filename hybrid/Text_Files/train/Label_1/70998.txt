from sys import stdin

for _ in range(int(stdin.readline())):

    n, m = map(int, stdin.readline().split())

    values = list(map(int, stdin.readline().split()))

    targets = list(map(int, stdin.readline().split()))

    graph = {}

    for _ in range(m):

        a, b = map(int, stdin.readline().split())

        if not a in graph:
            graph[a] = []

        if not b in graph:
            graph[b] = []

        graph[a].append(b)
        graph[b].append(a)

    sum_values = sum(values)
    sum_targets = sum(targets)

    if sum_values != sum_targets or sum_values % n != sum_targets % n:
        print('NO')
        continue

    value_change = (sum_targets - sum_values) // n

    for value, target in zip(values, targets):

        if value + value_change != target:
            print('NO')
            break
    else:
        print('YES')
