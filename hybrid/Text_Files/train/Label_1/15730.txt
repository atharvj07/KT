#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, n, a[100005];
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    int ans = 1, cur = a[n];
    for (int i = n - 1; i >= 1; i--) {
      if (cur - a[i] >= a[n]) ans++, cur = a[i];
    }
    cout << ans << "\n";
  }
  return 0;
}
