#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    for (int i = 0; i < n; i++) {
      if (s[i] == 'R') {
        cout << 'P';
      } else if (s[i] == 'P') {
        cout << 'S';
      } else {
        cout << 'R';
      }
    }
    cout << '\n';
  }
  return 0;
}
