#include <bits/stdc++.h>

using namespace std;

int main() {
  int k2, k3, k5, k6;
  cin >> k2 >> k3 >> k5 >> k6;
  int ans = 0;
  ans += min(k3, k6) * 32;
  k3 -= min(k3, k6);
  k6 -= min(k3, k6);
  int t = min(k3 + min(k2, k5 * 2), k6);
  ans += t * 32;
  k3 -= t;
  k6 -= t;
  k2 -= (t + k5) / 2;
  k5 -= (t + k2) / 2;
  ans += k5 * 16;
  cout << ans << endl;
  return 0;
}
