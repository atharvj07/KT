import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
        int k = sc.nextInt();
		int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
	    int[] test_time = new int[n + 1];
        for (int i = 1; i <= n; i++) {
		    if (i <= k) {
			    test_time[i] = a[i];
            } else {
			    test_time[i] = test_time[i - k] + a[i];
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            int j = test_time[i] - 1;
            if (j % 100 == 0 || j % 100 == a[i] % 100) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
