import java.util.Scanner;

public class ShopSellOut {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int f = scanner.nextInt();

        long[] products = new long[n];
        long[] clients = new long[n];

        for (int i = 0; i < n; i++) {
            products[i] = scanner.nextLong();
            clients[i] = scanner.nextLong();
        }

        long maxSold = 0;
        for (int mask = 0; mask < (1 << f); mask++) {
            long sold = 0;
            long remainingProducts = 0;

            for (int i = 0; i < n; i++) {
                long currentProducts = (mask & (1 << i)) != 0 ? products[i] * 2 : products[i];
                long soldToday = Math.min(currentProducts + remainingProducts, clients[i]);
                remainingProducts = currentProducts + remainingProducts - soldToday;
                sold += soldToday;
            }

            maxSold = Math.max(maxSold, sold);
        }

        System.out.println(maxSold);
    }
}
