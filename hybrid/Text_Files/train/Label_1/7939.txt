#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;getline(cin,s);
    for(int i=0;i<s.length();){
        if(s[i]+s[i+1]+s[i+2] == '+' and s[i+3]+s[i+4]+s[i+5] == '='){
            if(s[i] == s[i+3]){
                s.erase(i+3,1);
                cout<<s<<endl;
                return 0;
            }
            if(s[i+1] == s[i+4]){
                s.erase(i+1,1);
                cout<<s<<endl;
                return 0;
            }
            if(s[i+2] == s[i+5]){
                s.erase(i+5,1);
                cout<<s<<endl;
                return 0;
            }
        }
        if(s[i]+s[i+1]+s[i+2] == '=' and s[i+3]+s[i+4]+s[i+5] == '+'){
            if(s[i] == s[i+5]){
                s.erase(i+5,1);
                cout<<s<<endl;
                return 0;
            }
            if(s[i+1] == s[i+4]){
                s.erase(i+4,1);
                cout<<s<<endl;
                return 0;
            }
            if(s[i+2] == s[i+3]){
                s.erase(i+3,1);
                cout<<s<<endl;
                return 0;
            }
        }
        i+=3;
    }
    cout<<"Impossible"<<endl;
    return 0;
}
