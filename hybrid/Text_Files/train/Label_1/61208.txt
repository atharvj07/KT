n, m, k = map(int, input().split())

cells_per_tube = (n*m) // k
extra_cells = (n*m) % k

tubes = [cells_per_tube + (1 if i < extra_cells else 0) for i in range(k)]

x, y, dx = 1, 1, 1
for i in range(k):
    print(tubes[i], end=' ')
    for j in range(tubes[i]):
        print(x, y, end=' ')
        y += dx
        if y > m:
            dx = -1
            y = m
            x += 1
        elif y < 1:
            dx = 1
            y = 1
            x += 1
    print()
