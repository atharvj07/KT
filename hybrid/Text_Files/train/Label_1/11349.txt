#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    unordered_map<string, bool> m = {{"purple", false}, {"green", false}, {"blue", false}, {"orange", false}, {"red", false}, {"yellow", false}};
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        m[s] = true;
    }
    vector<string> ans;
    for (auto x : m) {
        if (!x.second) ans.push_back(x.first);
    }
    cout << ans.size() << "\n";
    for (auto x : ans) {
        if (x == "purple")
            cout << "Power\n";
        else if (x == "green")
            cout << "Time\n";
        else if (x == "blue")
            cout << "Space\n";
        else if (x == "orange")
            cout << "Soul\n";
        else if (x == "red")
            cout << "Reality\n";
        else if (x == "yellow")
            cout << "Mind\n";
    }
    return 0;
}
