#include <iostream>

const int MOD = 1e9 + 7;

long long mul(long long a, long long b) {
    return (a * b) % MOD;
}

long long pow(long long a, long long b) {
    if (b == 0)
        return 1;
    if (b == 1)
        return a;
    long long t = pow(a, b / 2);
    t = mul(t, t);
    if (b % 2 == 1)
        t = mul(t, a);
    return t;
}

int main() {
    long long n, m;
    std::cin >> n >> m;

    while (n < m) {
        if (m % 3 == 0) {
            m /= 3;
        } else if (m % 2 == 0) {
            m /= 2;
        } else {
            std::cout << -1 << std::endl;
            return 0;
        }
    }

    if (n == m) {
        std::cout << pow(3, 8) << std::endl;
    } else {
        std::cout << -1 << std::endl;
    }

    return 0;
}
