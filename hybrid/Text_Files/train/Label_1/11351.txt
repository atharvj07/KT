#include <bits/stdc++.h>
using namespace std;
int n;
int main() {
  cin >> n;
  vector<string> col = {"purple", "green", "blue", "orange", "red", "yellow"};
  vector<string> gem = {"Power",   "Time",   "Space",   "Soul",   "Reality",
                        "Mind"};
  vector<bool> ok(6);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < 6; j++) {
      if (col[j] == s) ok[j] = true;
    }
  }
  cout << 6 - n << endl;
  for (int i = 0; i < 6; i++) {
    if (!ok[i]) cout << gem[i] << endl;
  }
}
