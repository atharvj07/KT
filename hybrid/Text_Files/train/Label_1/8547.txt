#include <bits/stdc++.h>
#define ll long long
using namespace std;

vector<string> users;
map<string, bool> vis, exist;
vector<string> ans, chat;
void dfs(string cur, int id) {
  if (id == chat.size()) {
    for (auto &x : ans) { cout << x << '\n'; }
    exit(0);
  }
  for (auto &x : users) {
    if ((x != cur) && (!exist[x]) && (!vis[x])) {
      string s = chat[id], ss;
      bool ok = 1;
      for (auto &c : s) {
        if (!ok && isalpha(c)) ss.clear();
        ok = 0;
        ss += (char)c;
        if (ss == x) ok = 1;
      }
      if (!ok) {
        vis[x] = 1;
        exist[x] = 1;
        ans.push_back(x + ':' + chat[id]);
        dfs(x, id + 1);
        ans.pop_back();
        vis[x] = 0;
        exist[x] = 0;
      }
    }
  }
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n;
    users.resize(n);
    for (auto &x : users) cin >> x;
    cin >> m;
    chat.resize(m);
    for (auto &x : chat) {
      cin >> x;
      int poz = x.find(':');
      string s = x.substr(0, poz), ss = x.substr(poz + 1);
      if (s[0] == '?') {
        for (auto &c : ss)
          if (isalpha(c)) ss.clear();
        for (auto &y : users) {
          if (y != s && ss == y) {
            cout << "Impossible" << '\n';
            return 0;
          }
        }
      } else
        exist[s] = true;
    }
    dfs("start", 0);
    cout << "Impossible" << '\n';
  }
}
