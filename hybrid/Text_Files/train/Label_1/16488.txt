#include <bits/stdc++.h>
using namespace std;

const int MAXN = 105;
const int oo = 1e9;
int a[MAXN], b[MAXN];
int n, sumA, sumB;

bool check() {
    return sumA == sumB;
}

int main() {
    int t; cin >> t;
    while (t--) {
        cin >> n;
        sumA = sumB = 0;
        for (int i=1; i<=n; i++) {
            cin >> a[i];
            sumA += a[i];
        }
        for (int i=1; i<=n; i++) {
            cin >> b[i];
            sumB += b[i];
        }
        if (check()) {
            cout << "0\n";
            continue;
        }
        vector<pair<int, int>> res;
        while (sumA < sumB) {
            for (int i=1; i<=n; i++) {
                for (int j=1; j<=n; j++) {
                    if (i == j) continue;
                    if (a[i] > 0) {
                        sumA -= 1;
                        sumB += 1;
                        a[i] -= 1;
                        a[j] += 1;
                        res.push_back({i, j});
                        if (check()) break;
                    }
                }
                if (check()) break;
            }
            if (check()) break;
        }
        if (check()) {
            cout << res.size() << '\n';
            for (auto x: res) cout << x.first << ' ' << x.second << '\n';
        }
        else cout << "-1\n";
    }
    return 0;
}
