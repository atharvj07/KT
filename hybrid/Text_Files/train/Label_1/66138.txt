from math import factorial
from collections import Counter

def solve(n, p, a):
    a.sort()
    dp = [0] * (n + 1)
    dp[0] = 1
    for i in range(1, n + 1):
        dp[i] = dp[i - 1] * i % p

    good_integers = []
    for x in range(1, max(a) + 1):
        count = 0
        for ai in a:
            if x > ai:
                count += 1
            else:
                break
        if count == 0:
            break
        if dp[count] * dp[n - count] % p != 0:
            good_integers.append(x)

    print(len(good_integers))
    print(*good_integers)

n, p = map(int, input().split())
a = list(map(int, input().split()))
solve(n, p, a)
