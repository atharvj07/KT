#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int x = (k + i) / (i + 1);
    if (x > a[i]) break;
    ans += 1LL * x * x * (i + 1);
    k -= 1LL * x * (i + 1);
  }
  cout << ans << endl;
}
