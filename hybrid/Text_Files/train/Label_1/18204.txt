#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n), t(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> t[i];
  sort(a.begin(), a.end());
  int cnt = 0;
  for (int i = 0; i < n - 1; i++) cnt += (a[i] == a[i + 1]);
  if (cnt == n - 1)
    cout << 0 << endl;
  else
    cout << t[max_element(a.begin(), a.end()) - a.begin()] << endl;
  return 0;
}
