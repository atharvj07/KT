#include <bits/stdc++.h>
using namespace std;
const int N = 101;
int n, a[N], f[N], mark[N];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    sort(a + 1, a + n + 1);
    for (int i = 1; i <= n / 2; i++) {
        f[i] = a[2 * i] - a[2 * i - 1];
        mark[2 * i - 1] = mark[2 * i] = 1;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++)
        if (!mark[i]) { ans += a[i] - a[1]; }
    for (int i = 1; i <= n / 2; i++) ans += f[i];
    cout << ans << endl;
    return 0;
}
