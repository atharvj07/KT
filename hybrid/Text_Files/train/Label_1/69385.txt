from collections import defaultdict, deque

def find_min_traffic_controllers(n, m, roads):
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    for i, (u, v, c) in enumerate(roads):
        graph[u].append((v, c, i + 1))
        indegree[v] += 1

    # Topological sort to find the order
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)

    order = []
    while queue:
        node = queue.popleft()
        order.append(node)
        for v, _, _ in graph[node]:
            indegree[v] -= 1
            if indegree[v] == 0:
                queue.append(v)

    if len(order) != n:
        return -1, []

    # Assign ranks based on the topological order
    rank = [0] * (n + 1)
    for i, node in enumerate(order):
        rank[node] = i

    # Find minimum controllers and roads to be reversed
    min_controllers = 0
    roads_to_reverse = []
    for u, v, c, idx in [(u, v, c, idx) for u, adj in graph.items() for v, c, idx in adj]:
        if rank[u] >= rank[v]:  # If there is a cycle or wrong order
            min_controllers = max(min_controllers, c)
            roads_to_reverse.append(idx)

    return min_controllers, roads_to_reverse

# Read input
n, m = map(int, input().split())
roads = [list(map(int, input().split())) for _ in range(m)]

# Solve
min_controllers, roads_to_reverse = find_min_traffic_controllers(n, m, roads)

# Print output
if min_controllers == -1:
    print("Impossible")
else:
    print(min_controllers, len(roads_to_reverse))
    print(*roads_to_reverse)
