n, a, b, k, f = map(int, input().split())

previous_stop = None
costs = []
trips = {}

for _ in range(n):
    start, end = input().split()
    if start == previous_stop:
        trip_cost = b
    else:
        trip_cost = a
    previous_stop = end
    
    route = tuple(sorted([start, end]))
    if route in trips:
        trips[route] += trip_cost
    else:
        trips[route] = trip_cost
        
sorted_costs = sorted(trips.values(), reverse=True)

total_cost = sum(sorted_costs)

for i in range(min(k, len(sorted_costs))):
    if sorted_costs[i] > f:
        total_cost += f - sorted_costs[i]

print(total_cost)
