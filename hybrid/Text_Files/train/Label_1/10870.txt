#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
const int INF = 1e9;
int main() {
    int n;
    cin >> n;
    unordered_map<ll, ll> row;
    unordered_map<ll, ll> column;
    unordered_map<ll, ll> a_diag;
    unordered_map<ll, ll> b_diag;
    ll res = 0;
    for (int i = 0; i < n; i++) {
        int r, c;
        cin >> r >> c;
        ll sum1 = r + c;
        ll sum2 = r - c;
        res += row[sum1] + column[c] + a_diag[sum1] + b_diag[sum2];
        row[sum1]++;
        column[c]++;
        a_diag[sum1]++;
        b_diag[sum2]++;
    }
    cout << res << endl;
    return 0;
}
