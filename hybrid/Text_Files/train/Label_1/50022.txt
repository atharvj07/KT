import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Game {

    static class Cell {
        int row;
        int col;
        int player;

        public Cell(int row, int col, int player) {
            this.row = row;
            this.col = col;
            this.player = player;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();
        int[] speeds = new int[p];
        for (int i = 0; i < p; i++) {
            speeds[i] = scanner.nextInt();
        }
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        int[] result = new int[p];
        for (int i = 0; i < p; i++) {
            result[i] = countCells(grid, speeds[i], i);
        }

        for (int i = 0; i < p; i++) {
            System.out.print(result[i] + " ");
        }
    }

    static int countCells(char[][] grid, int speed, int player) {
        int n = grid.length;
        int m = grid[0].length;
        boolean[][] visited = new boolean[n][m];
        Queue<Cell> queue = new LinkedList<>();

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == (char) (player + '1')) {
                    queue.add(new Cell(i, j, player));
                    visited[i][j] = true;
                }
            }
        }

        int count = 0;
        while (!queue.isEmpty()) {
            Cell cell = queue.poll();
            count++;
            for (int dx = -speed; dx <= speed; dx++) {
                for (int dy = -speed; dy <= speed; dy++) {
                    int newRow = cell.row + dx;
                    int newCol = cell.col + dy;
                    if (newRow >= 0 && newRow < n && newCol >= 0 && newCol < m &&
                            !visited[newRow][newCol] && grid[newRow][newCol] == '.') {
                        queue.add(new Cell(newRow, newCol, player));
                        visited[newRow][newCol] = true;
                    }
                }
            }
        }

        return count;
    }
}
