#include <cstdio>
#include <vector>

int main() {
  long long n, m, tmp, cnt = 1;
  std::vector<long long> va, vb;

  scanf("%lld", &n);
  va.resize(n);
  vb.resize(n);
  for (long long i = 0; i < n; ++i) scanf("%lld", &va[i]);
  for (long long i = 0; i < n; ++i) scanf("%lld", &vb[i]);
  scanf("%lld", &m);
  for (long long i = 1; i < n; ++i) {
    tmp = va[i] - va[i - 1];
    cnt *= vb.end() - std::lower_bound(vb.begin(), vb.end(), tmp);
    cnt %= m;
  }
  printf("%lld\n", cnt);
  return 0;
}
