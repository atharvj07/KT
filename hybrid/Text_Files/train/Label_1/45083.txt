import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N];
        int[] B = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        for (int i = 0; i < N; i++) {
            B[i] = sc.nextInt();
        }
        sc.close();

        long[] sum = new long[N + 1];
        for (int i = 0; i < N; i++) {
            sum[i + 1] = sum[i] + A[i];
        }

        int ans = -1;
        for (int i = 0; i < N; i++) {
            long target = sum[i + 1] - sum[i] + B[i] - A[i];
            if (target % N != 0) {
                continue;
            }
            long x = target / N;
            boolean ok = true;
            for (int j = 0; j < N; j++) {
                if (A[(i + j) % N] + x != B[(i + j) % N]) {
                    ok = false;
                }
            }
            if (ok) {
                if (ans == -1) {
                    ans = 0;
                }
                ans += x;
            }
        }
        System.out.println(ans);
    }
}
