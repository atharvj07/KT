import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int r = sc.nextInt();
        int[] heads = new int[n];
        int[] tails = new int[n];
        for (int i = 0; i < n; i++) {
            heads[i] = sc.nextInt();
            tails[i] = sc.nextInt();
        }
        int[] dp = new int[n+1];
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            dp[i+1] = dp[i];
            for (int j = i+1; j < n; j++) {
                if (tails[j] > heads[i] || tails[i] > heads[j]) {
                    dp[i+1] = Math.max(dp[i+1], dp[j]+1);
                }
            }
        }
        System.out.println(dp[n]);
    }
}
