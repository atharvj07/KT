a = []
q = 0
for x in input():
    if x == '?': q += 1
    else: a += [x]
ans = (q * (q-1) * (q-2) // 6 * 3**q) % (10**9 + 7)
for i in range(len(a)):
    for j in range(i+1, len(a)):
        for k in range(j+1, len(a)):
            if a[i] == 'a' and a[j] == 'b' and a[k] == 'c': ans += 1
            if a[i] == 'a' and a[j] == 'c' and a[k] == 'b': ans += (q+1) * (q-1) * (q-2) // 6 * 3**q
print(ans % (10**9 + 7))
