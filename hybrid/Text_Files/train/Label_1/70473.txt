import sys
n, k = map(int, sys.stdin.readline().split())
board = []
for _ in range(4):
    board.append([int(x) for x in sys.stdin.readline().split()])
    
dx = [-1, 0, 1, 0]
dy = [0, 1, 0, -1]
ans = []
def can_move(car, x, y):
    for i in range(4):
        nx = x + dx[i]
        ny = y + dy[i]
        if 0 <= nx < 4 and 0 <= ny < n and board[nx][ny] == 0:
            return True
    return False
def solve(cnt):
    if cnt == k * 2:
        print(len(ans))
        for a in ans:
            print(*a)
        exit(0)
    for x in range(4):
        for y in range(n):
            if board[x][y] != 0 and can_move(board[x][y], x, y):
                for i in range(4):
                    nx = x + dx[i]
                    ny = y + dy[i]
                    if 0 <= nx < 4 and 0 <= ny < n and board[nx][ny] == 0:
                        board[nx][ny] = board[x][y]
                        board[x][y] = 0
                        ans.append([board[nx][ny], nx + 1, ny + 1])
                        solve(cnt + 1)
                        board[x][y] = board[nx][ny]
                        board[nx][ny] = 0
                        ans.pop()
solve(0)
print(-1)
