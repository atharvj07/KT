#include <iostream>
#include <vector>
using namespace std;

int n, s, cnt;
vector<int> a;
bool f[61][3001];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    a.clear();
    a.resize(n + 1);
    s = 0;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      s += a[i];
    }
    cnt = 0;
    f[0][0] = 1;
    for (int i = 1; i <= n; i++) {
      for (int j = 0; j <= s; j++) {
        f[i][j] = f[i - 1][j];
        if (j >= a[i]) f[i][j] = f[i][j] || f[i - 1][j - a[i]];
      }
      if (f[i][s - 1]) cnt++;
    }
    cout << cnt << '\n';
  }
  return 0;
}
