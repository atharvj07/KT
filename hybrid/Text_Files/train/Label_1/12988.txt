#include <cstdio>
#include <algorithm>
using namespace std;
const double EPS = 1e-9;
int main() {
	int n, m;
	scanf(" %d%d", &n, &m);
	double ans = 1e18;
	for (int i = 0, a, b; i < n; i++) {
		scanf(" %d%d", &a, &b);
		ans = min(ans, (double)a / b);
	}
	ans *= m;
	if (ans <= EPS) puts("0");
	else printf("%.10lf\n", ans);
	return 0;
}
