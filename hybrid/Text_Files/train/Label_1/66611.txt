import sys

inp = sys.stdin.readlines()
n,m = map(int, inp[0].split())
g = [[] for _ in range(n+1)]
for l in inp[1:]:
    a,b = map(int, l.split())
    g[a].append(b)
    g[b].append(a)
vis = [0]* (n+1)
def dfs(u, f, len):
    vis[u] = 1
    ans = len
    for v in g[u]:
        if v != f:
            if vis[v] == 0:
                ans = max(ans, dfs(v, u, len+1))
    return ans
ans = 0
for i in range(1,n+1):
    if vis[i] == 0:
        ans = max(ans, dfs(i, -1, 0))
print(ans*(sum([1 for x in vis if x>0])-1))
