def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return a * b // gcd(a, b)

def min_operations(a1, b1, a2, b2):
    lcm_ab1 = lcm(a1, b1)
    lcm_ab2 = lcm(a2, b2)
    lcm_both = gcd(lcm_ab1, lcm_ab2)
    
    if lcm_both == 0:
        return -1, (), ()
    
    def min_reduce(a, b, target):
        ops = 0
        while True:
            if a * b == target:
                return ops, (a, b)
            if a % 2 == 0 and (a // 2) * b >= target:
                a //= 2
                ops += 1
            elif b % 2 == 0 and a * (b // 2) >= target:
                b //= 2
                ops += 1
            elif a % 3 == 0 and (a // 3) * 2 * b >= target:
                a = (a // 3) * 2
                ops += 1
            elif b % 3 == 0 and a * (b // 3) * 2 >= target:
                b = (b // 3) * 2
                ops += 1
            else:
                return -1, ()
    
    ops1, size1 = min_reduce(a1, b1, lcm_both)
    ops2, size2 = min_reduce(a2, b2, lcm_both)
    
    if ops1 == -1 or ops2 == -1:
        return -1, (), ()
    else:
        return ops1 + ops2, size1, size2

a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())

m, bar1, bar2 = min_operations(a1, b1, a2, b2)
if m == -1:
    print(m)
else:
    print(m)
    print(*bar1)
    print(*bar2)
