n, m, k = map(int, input().split())
a = list(map(int, input().split()))

left, right = 0, n
while left < right:
    mid = (left + right + 1) // 2
    boxes, current_box = m, k
    for i in range(mid):
        if a[mid - i - 1] <= current_box:
            current_box -= a[mid - i - 1]
        else:
            boxes -= 1
            current_box = k - a[mid - i - 1]
        if boxes == 0 and i != mid - 1:
            break
    if boxes > 0 or current_box >= 0:
        left = mid
    else:
        right = mid - 1

print(left)
