for _ in range(int(input())):
    n = int(input())
    s = input()
    if s.count('a') != s.count('b'):
        print(-1, -1)
        continue
    l, r = [], []
    sa = sb = 0
    for i in range(n):
        if s[i] == 'a':
            sa += 1
        else:
            sb += 1
        if sa == sb:
            l.append(i + 1)
        if sa > sb:
            break
    sa = sb = 0
    for i in range(n - 1, -1, -1):
        if s[i] == 'a':
            sa += 1
        else:
            sb += 1
        if sa == sb:
            r.append(i + 1)
        if sb > sa:
            break
    if len(l) != len(r):
        print(-1, -1)
    else:
        print(l[len(l) // 2], r[len(r) // 2])
