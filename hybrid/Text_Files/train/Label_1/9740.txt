#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n; cin>>n;
    long long ans;
    long long root=sqrt(n);
    int fl=0;
    if(n==1) ans=1;
    else if(n==2) ans=2;
    else if(n==3) ans=2;
    else{
        for(long long i=2; i<=root; i++){
            if(n%i==0){
                fl=1;
                ans=i;
            }
        }
        if(fl==0) ans=n;
    }
    cout<<ans<<"\n";
    return 0;
}
