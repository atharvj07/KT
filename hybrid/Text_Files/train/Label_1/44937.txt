import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            if (!map.containsKey(a)) {
                map.put(a, new ArrayList<>());
            }
            map.get(a).add(b);
            if (!map.containsKey(b)) {
                map.put(b, new ArrayList<>());
            }
            map.get(b).add(a);
        }
        Set<Integer> set = new HashSet<>();
        for (int start : map.keySet()) {
            boolean isRotation = false;
            for (int i = 0; i < n; i++) {
                int cur = start + i;
                if (cur >= n) {
                    cur -= n;
                }
                int next = cur + 1;
                if (next >= n) {
                    next -= n;
                }
                if (map.containsKey(cur) && map.get(cur).contains(next)) {
                    isRotation = true;
                } else {
                    isRotation = false;
                    break;
                }
            }
            if (isRotation) {
                set.add(start);
            }
        }
        System.out.println(set.size() > 0 ? "Yes" : "No");
    }
}
