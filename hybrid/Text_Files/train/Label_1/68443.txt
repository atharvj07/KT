import sys
n = int(input())
a = []
lst = []
for i in range(2*n):
    x = input()
    if x[0] == '+': lst.append(1)
    else: lst.append(int(x[1:]))
if lst[0] == -1: print('NO'); exit()
for x in lst:
    if x == 1: a.append(-1)
    else: a.pop()
if len(a) == 0: print('YES'); print(*lst[1:n+1])
else: print('NO')
