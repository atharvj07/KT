n,k=map(int,input().split())
a=[[] for i in range(n+1)]
for i in range(n-1):
    u,v=map(int,input().split())
    a[u].append(v)
    a[v].append(u)
vis=[0]*(n+1)
que=[]
ans="Yes"
for i in range(1,n+1):
    if len(a[i])>1:que.append(i);vis[i]=1
    if len(a[i])>3:ans="No"
if len(que)>1:ans="No"
if n%k!=0:ans="No"
if k==1 and len(que)!=1:ans="No"
if k>1 and len(que)!=0:ans="No"
print(ans)
