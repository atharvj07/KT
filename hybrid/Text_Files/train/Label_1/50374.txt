import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] sticks = new int[4];
        for (int i = 0; i < 4; i++) {
            sticks[i] = sc.nextInt();
        }
        sc.close();
        int[] sortedSticks = sticks.clone();
        java.util.Arrays.sort(sortedSticks);
        if (sortedSticks[0] + sortedSticks[1] > sortedSticks[2] && sortedSticks[1] + sortedSticks[2] > sortedSticks[3]) {
            System.out.println("TRIANGLE");
        } else if (sortedSticks[0] + sortedSticks[1] == sortedSticks[2] || sortedSticks[1] + sortedSticks[2] == sortedSticks[3]) {
            System.out.println("SEGMENT");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
