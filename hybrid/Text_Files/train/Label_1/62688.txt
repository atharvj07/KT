from sys import stdin
n=int(stdin.readline())
a=list(map(int,stdin.readline().split()))
m=int(1e9+7)
dp=[[0]*2 for i in range(n+1)]
dp[0][1]=dp[0][0]=1
for i in range(n):
 for j in range(2):
  dp[i+1][j]=(dp[i+1][j]+dp[i][j])%m
  if j and a[i]!=1 and all(a[j]%a[i]!=0 for j in range(i)):
   dp[i+1][1]=(dp[i+1][1]+dp[i][0])%m
print(dp[n][1])
