#include <cstdio>
#include <iostream>
#include <vector>
#include <sstream>

using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  string s;
  cin >> s;
  int ans = 0, cnt = 0, prv = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '*') {
      cnt = 0;
      prv = i;
    } else {
      cnt++;
      if (i - prv >= 3) {
        if (ans % 2 == 0 && a > 0) {
          a--;
          ans++;
        } else if (ans % 2 == 1 && b > 0) {
          b--;
          ans++;
        }
      } else if ((i - prv == 2 && (s[prv] != s[prv + 1])) || cnt == 2) {
        if (ans % 2 == 0 && a > 0) {
          a--;
          ans++;
        } else if (ans % 2 == 1 && b > 0) {
          b--;
          ans++;
        }
      }
    }
  }
  if (ans % 2 == 0 && a > 0) {
    ans++;
  } else if (ans % 2 == 1 && b > 0) {
    ans++;
  }
  cout << ans << endl;
  return 0;
}
