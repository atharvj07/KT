t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    c = list(map(int, input().split()))

    last = None
    for i in range(n-1, -1, -1):
        if b[i] == c[-1]:
            last = i + 1
            break
    if last is None:
        print("NO")
        continue

    answer = [0] * m
    answer[-1] = last
    correct = True
    needs = {}
    for i in range(n):
        if a[i] != b[i]:
            if b[i] not in needs:
                needs[b[i]] = []
            needs[b[i]].append(i + 1)

    for i in range(m-1):
        color = c[i]
        if color in needs and needs[color]:
            answer[i] = needs[color].pop()
            if not needs[color]:
                del needs[color]
        else:
            answer[i] = last

    if needs:
        print("NO")
    else:
        print("YES")
        print(*answer)
