import sys

n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))

sa = sum(a)
sb = sum(b)
mini = abs(sa - sb)
ans = []

for i in range(n):
    for j in range(m):
        cur = abs(sa - a[i] + b[j] - sb + a[i] - b[j])
        if cur < mini:
            mini = cur
            ans = [(i + 1, j + 1), (i + 1, j + 1)]
        elif cur == mini:
            ans.append((i + 1, j + 1))

for i in range(n):
    for j in range(m):
        for k in range(n):
            for l in range(m):
                if i == k and j == l:
                    continue
                cur = abs(sa - a[i] + b[j] - sb + a[k] - b[l])
                if cur < mini:
                    mini = cur
                    ans = [(i + 1, j + 1), (k + 1, l + 1)]
                elif cur == mini:
                    ans.append((i + 1, j + 1))
                    ans.append((k + 1, l + 1))

print(mini)
print(len(ans) // 2)
for i in range(0, len(ans), 2):
    print(*ans[i])
