#include<bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;

    int ans = s.size();
    for(int k = 1; k <= s.size(); k++) {
        set<char> st;
        for(int i = 0; i < s.size(); i++) {
            if(i + k - 1 >= s.size()) {
                break;
            }
            st.insert(s[i + k - 1]);
        }
        
        if(st.size() == 1) {
            ans = k;
            break;
        }
    }
    
    cout << ans << endl;
    return 0;
}
