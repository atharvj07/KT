#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
const int mod = 1000000007;
const int maxn = 200005;
int fac[maxn];
int powmod(int a, int b) {
    return b == 0 ? 1 : ((b & 1) ? 1LL * a * powmod(1ll * a * a % mod, b >> 1) % mod : powmod(1ll * a * a % mod, b >> 1));
}
void make(int n) {
    fac[0] = 1;
    for (int i = 1; i <= n; i++) {
        fac[i] = 1ll * i * fac[i - 1] % mod;
    }
}
int main() {
    int q;
    cin >> q;
    make(200000);
    while (q--) {
        int n, f;
        cin >> n >> f;
        int ans = 1ll * fac[n] * powmod(1ll * fac[n - f] * fac[f] % mod, mod - 2) % mod;
        cout << ans << endl;
    }
}
