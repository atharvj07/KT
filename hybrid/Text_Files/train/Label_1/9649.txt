#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m, ans = 0, rem = 0;
    cin >> n >> m;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        rem += max(a[i] - m, 0);
    }
    for (int i = 1; i <= n && rem > 0; i++) {
        rem -= max(a[i] - m, 0);
        ans++;
    }
    cout << ans << "\n";
    return 0;
}
