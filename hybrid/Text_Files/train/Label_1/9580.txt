#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> requests(n);
    for (int i = 0; i < n; i++) {
        cin >> requests[i].first >> requests[i].second;
    }
    vector<pair<long long, long long>> scheduled(n);
    long long end = 0;
    for (int i = 0; i < n; i++) {
        if (requests[i].first >= end) {
            scheduled[i].first = requests[i].first;
            scheduled[i].second = requests[i].first + requests[i].second - 1;
            end = requests[i].first + requests[i].second;
        } else {
            long long j = i;
            while (j < n && requests[j].first < end) {
                j++;
            }
            if (j == n) {
                scheduled[i].first = requests[i].first;
                scheduled[i].second = requests[i].first + requests[i].second - 1;
                end = requests[i].first + requests[i].second;
            } else {
                scheduled[i].first = requests[j].first;
                scheduled[i].second = requests[j].first + requests[i].second - 1;
                end = requests[j].first + requests[i].second;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        cout << scheduled[i].first << " " << scheduled[i].second << endl;
    }
    return 0;
}
