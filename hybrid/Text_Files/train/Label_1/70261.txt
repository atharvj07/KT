MOD = 998244353

def modinv(x, p=MOD):
    return pow(x, p-2, p)

def solve(n, m, distances):
    # Calculate the probability of each point being conquered
    prob = [0] * m
    for j in range(m):
        for mask in range(1 << n):
            max_dist = 0
            for i in range(n):
                if mask & (1 << i):
                    max_dist = max(max_dist, distances[i][j])
            if max_dist <= bin(mask).count('1'):
                prob[j] += 1
        prob[j] *= modinv(1 << n)
        prob[j] %= MOD
    
    # Calculate the expected number of conquered points
    expected = sum(prob) % MOD
    return expected

n, m = map(int, input().split())
distances = [list(map(int, input().split())) for _ in range(n)]
print(solve(n, m, distances))
