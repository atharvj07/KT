import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        List<Pair> pairs = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int ik = in.nextInt() - 1, jk = in.nextInt() - 1;
            pairs.add(new Pair(ik, jk));
        }
        int ans = 0;
        for (Pair pair : pairs) {
            int ik = pair.ik, jk = pair.jk;
            if (a[ik] % 2 == 0 && a[jk] % 2 == 0) ans++;
        }
        System.out.println(ans);
    }
    static class Pair {
        int ik, jk;
        Pair(int ik, int jk) {
            this.ik = ik;
            this.jk = jk;
        }
    }
}
