#include <iostream>

using namespace std;

int main() {
  int a[3], b[3];
  int n;
  cin >> a[0] >> a[1] >> a[2];
  cin >> b[0] >> b[1] >> b[2];
  cin >> n;
  int cnt1 = 0, cnt2 = 0;
  for (int i = 0; i < 3; i++) {
    cnt1 += a[i];
    cnt2 += b[i];
  }
  if (cnt1 > n * 5 || cnt2 > n * 10)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
