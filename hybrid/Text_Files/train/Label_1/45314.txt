import java.util.*;
import java.io.*;

public class RocketSafety {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        Point[] engine1 = new Point[n];
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            engine1[i] = new Point(x, y);
        }

        Point[] engine2 = new Point[m];
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            engine2[i] = new Point(x, y);
        }

        if (isSafe(engine1, engine2)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static boolean isSafe(Point[] engine1, Point[] engine2) {
        // Check if any point in engine1 is collinear with any point in engine2
        for (int i = 0; i < engine1.length; i++) {
            for (int j = 0; j < engine2.length; j++) {
                if (isCollinear(engine1[i], engine2[j])) {
                    return false;
                }
            }
        }

        // Check if any point in engine1 is collinear with any point in the power field generated by engine2
        for (int i = 0; i < engine1.length; i++) {
            if (!isPointInPowerField(engine1[i], engine2)) {
                return false;
            }
        }

        // Check if any point in engine2 is collinear with any point in the power field generated by engine1
        for (int i = 0; i < engine2.length; i++) {
            if (!isPointInPowerField(engine2[i], engine1)) {
                return false;
            }
        }

        return true;
    }

    static boolean isCollinear(Point p1, Point p2) {
        return p1.x * p2.y - p1.y * p2.x == 0;
    }

    static boolean isPointInPowerField(Point p, Point[] engine) {
        for (int i = 0; i < engine.length; i++) {
            for (int j = i + 1; j < engine.length; j++) {
                Point p1 = engine[i];
                Point p2 = engine[j];
                Point p3 = new Point((p1.x + p2.x) / 2, (p1.y + p2.y) / 2);
                if (isCollinear(p, p1) && isCollinear(p, p2)) {
                    return true;
                } else if (isCollinear(p, p1) && isCollinear(p, p3)) {
                    return true;
                } else if (isCollinear(p, p2) && isCollinear(p, p3)) {
                    return true;
                }
            }
        }
        return false;
    }

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
