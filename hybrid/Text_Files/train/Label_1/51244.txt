import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int k = sc.nextInt();
        int l = s.length() + k;
        for (int i = l / 2; i >= 1; i--) {
            if (l % i == 0) {
                boolean isValid = true;
                for (int j = i; j < l; j++) {
                    if (s.charAt(j % s.length()) != s.charAt(j % i)) {
                        isValid = false;
                        break;
                    }
                }
                if (isValid) {
                    l = i;
                    break;
                }
            }
        }
        System.out.println(l);
    }
}
