import sys
from heapq import heappop, heappush
input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for _ in range(n + 1)]
dist = [[1e9] * (n + 1) for _ in range(n + 1)]
for i in range(m):
    a, b, c = map(int, input().split())
    graph[a].append([b, c])
    graph[b].append([a, c])
q = []
a = list(map(int, input().split()))
for i in range(1, n + 1):
    dist[i][i] = a[i - 1]
    heappush(q, [dist[i][i], i])
while q:
    cost, now = heappop(q)
    if cost > dist[now][now]:
        continue
    for next, weight in graph[now]:
        if cost + weight < dist[now][next]:
            dist[now][next] = dist[next][now] = cost + weight
            heappush(q, [cost + weight, next])
for i in range(1, n + 1):
    print(min(dist[i]))
