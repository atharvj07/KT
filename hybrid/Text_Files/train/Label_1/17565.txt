#include <bits/stdc++.h>
using namespace std;

int gcd(int a, int b) {
    if (!b)
        return a;
    return gcd(b, a % b);
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        unsigned long long int n;
        cin >> n;
        unsigned long long int sum = 0;
        bool flag = true;
        for (unsigned long long int i = n;; i++) {
            unsigned long long int temp = i, temp1 = 0;
            while (temp) {
                temp1 += temp % 10;
                temp /= 10;
            }
            if (gcd(i, temp1) > 1) {
                cout << i << endl;
                flag = false;
                break;
            }
        }
    }
    return 0;
}
