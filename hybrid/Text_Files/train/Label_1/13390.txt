#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        vector<int> a(n), mxa(n), mna(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        mna[n - 1] = a[n - 1];
        for (int i = n - 1; i > 0; i--) {
            mna[i - 1] = min(mna[i], a[i - 1]);
        }
        int mx = a[0];
        for (int i = 0; i < n; i++) {
            mxa[i] = mx, mx = max(mxa[i], a[i]);
        }
        long long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans = max(ans, 1LL * mxa[i] * mna[i + 1]);
        }
        cout << ans << '\n';
    }
}
