import java.util.*;
import java.lang.*;
import java.io.*;
class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] nkst = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int n = nkst[0], k = nkst[1], s = nkst[2], t = nkst[3];
        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            int[] civ = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            cars[i] = new Car(civ[0], civ[1]);
        }
        int[] gs = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        Arrays.sort(gs);
        int minPrice = -1;
        for (Car c : cars) {
            if (c.canReach(s, t, gs)) {
                minPrice = Math.min(minPrice == -1 ? Integer.MAX_VALUE : minPrice, c.price);
            }
        }
        System.out.println(minPrice);
    }

    static class Car {
        int price, fuelTankCapacity;

        Car(int price, int fuelTankCapacity) {
            this.price = price;
            this.fuelTankCapacity = fuelTankCapacity;
        }

        boolean canReach(int s, int t, int[] gs) {
            int speed = 1, dist = 0, fuel = fuelTankCapacity, time = 0;
            for (int g : gs) {
                if (fuel < (g - dist + speed - 1) / speed) {
                    speed = 2;
                }
                time += (g - dist + speed - 1) / speed;
                dist = g;
                fuel -= (g - dist + speed) / speed;
                fuel = Math.min(fuelTankCapacity, fuel + fuelTankCapacity);
            }
            time += (s - dist + speed - 1) / speed;
            return time <= t;
        }
    }
}
