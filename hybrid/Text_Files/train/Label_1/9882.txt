#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
vector<int> e[N];
int d[N], a[N];
void dfs(int u, int p) {
	for(int v : e[u]) {
		if(v ^ p) { d[v] = (d[u] ^ 1), dfs(v, u); }
	}
}
int main() {
	int n, m, D;
	cin >> n >> m >> D;
	if(m != n - 1) return puts("NO"), 0;
	for(int i = 0; i < m; ++i) {
		int u, v;
		cin >> u >> v;
		e[u].push_back(v);
		e[v].push_back(u);
	}
	for(int i = 1; i <= n; ++i) {
		if(e[i].size() > 2) {
			puts("NO");
			return 0;
		}
	}
	dfs(1, 1);
	if(d[1] ^ D) { puts("NO"); return 0; }
	for(int i = 1; i <= n; ++i) {
		for(int v : e[i]) {
			if(i < v) {
				a[(d[i] ^ d[v]) + 1]++;
			}
		}
	}
	int c = 0;
	while(a[1] > 0) {
		if(a[1] & 1) { c = 1; break; }
		a[1] >>= 1;
		a[2]++;
	}
	for(int i = 2; i <= D; ++i) {
		if(a[i] & 1) { c = 1; break; }
		a[i] >>= 1;
		a[i + 1]++;
	}
	if(c) return puts("NO"), 0;
	cout << "YES" << endl;
	for(int i = 1; i <= n; ++i) {
		for(int v : e[i]) {
			if(i < v && (d[i] ^ d[v]) + 1 == 1) {
				cout << i << " " << v << endl;
			}
		}
	}
	return 0;
}
