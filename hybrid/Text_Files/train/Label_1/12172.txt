#include <iostream>
#include <vector>
#include <cassert>
#include <algorithm>
using namespace std;

long long solve() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    long long b = a[i];
    long long c = 0;
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      if (a[j] % b == 0) c += a[j];
    }
    ans = max(ans, b + c);
  }
  return ans;
}

int main() {
  cout << solve() << '\n';
  return 0;
}
