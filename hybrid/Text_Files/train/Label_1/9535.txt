#include <bits/stdc++.h>
using namespace std;

string solve(string s) {
  vector<int> cnt(26);
  vector<bool> vis(26);
  for (char c: s) cnt[c-'a']++;
  string t;
  for (char c: s) {
    if (!vis[c-'a']) {
      while (!t.empty() && t.back() > c && cnt[t.back()-'a'] > 0) {
        vis[t.back()-'a'] = false;
        t.pop_back();
      }
      t += c;
      vis[c-'a'] = true;
    }
    cnt[c-'a']--;
  }
  return t;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    cout << solve(s) << endl;
  }
  return 0;
}
