p,q,l,r=map(int,input().split())
a=[0]*(p+1)
b=[0]*(p+1)
c=[0]*(q+1)
d=[0]*(q+1)
for i in range(1,p+1):
    a[i],b[i]=map(int,input().split())
for i in range(1,q+1):
    c[i],d[i]=map(int,input().split())
ans=0
for i in range(l,r+1):
    for j in range(1,p+1):
        x=max(a[j],i)
        y=min(b[j],i)
        if x<=y: ans+=1
for i in range(l,r+1):
    for j in range(1,q+1):
        x=max(c[j]+i,i)
        y=min(d[j]+i,i)
        if x<=y: ans+=1
print(ans)
