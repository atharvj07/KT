import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt()-1,    // Decrement for 0-based indexing
            s = sc.nextInt(),
            best = 0,
            type1 = 0,
            type2 = 0;
        while (n-- > 0) {
            int slices = sc.nextInt(),
                a = sc.nextInt(),
                b = sc.nextInt();
            if (best < slices) {
                best = slices;
                type1 = a;
                type2 = b;
            }
        }
        System.out.println((best / s) * (s * type1 + s * type2) + (best % s) * Math.max(type1, type2));
    }
}
