#include<bits/stdc++.h>
using namespace std;

int main(){
	string s; cin >> s;
	int n = s.length();
	vector<int> pre(n+2,0);
	for(int i = 1; i<=n; i++) pre[i] = pre[i-1] + (s[i-1] == '1');
	int ans = 0;
	for(int i = 1; i<=n; i++){
		for(int j = i+1; j<=n; j++){
			int tot = pre[j] - pre[i-1];
			if(tot > 0){
				ans += j - i;
			}
		}
	}
	cout << ans << endl;
}
