#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll const MOD = 1000000007;
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  ll n, a, b, c, d, p1 = 1, p2 = 1, p3 = 1, ans;
  cin >> n >> a >> b >> c >> d;
  for (--n; n; n >>= 1, p1 = (1ll * p1 * p1) % MOD,
               p2 = (1ll * p2 * p2) % MOD, p3 = (1ll * p3 * p3) % MOD)
    if (n & 1) p1 = (1ll * p1 * d) % MOD, p2 = (1ll * p2 * c) % MOD,
                p3 = (1ll * p3 * b) % MOD;
  ans = (1ll * a * p1) % MOD;
  ans = (1ll * ans * p2) % MOD;
  ans = (1ll * ans * p3) % MOD;
  cout << ans << '\n';
  return 0;
}
