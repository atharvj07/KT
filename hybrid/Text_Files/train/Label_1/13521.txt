#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int64_t M = 1e9 + 7;

int main() {
  int n, m; cin >> n >> m;
  vector<int> p(n);
  for (int i = 0; i < n; i++) { cin >> p[i]; p[i]--; }
  vector<vector<int>> foe(n);
  vector<bool> occ(m);
  for (int i = 0; i < m; i++) {
    int x, y; cin >> x >> y; x--; y--;
    foe[x].push_back(y); foe[y].push_back(x);
    occ[i] = 0;
  }
  vector<bool> vis(n);
  vector<int64_t> dp(n + 1);

  dp[n] = 1;
  for (int i = n - 1; i >= 0; i--) {
    if (vis[p[i]]) continue;
    vis[p[i]] = 1;
    dp[i] = 0;
    int64_t cur = 1;
    for (int j = 0; j < foe[p[i]].size(); j++) {
      int u = foe[p[i]][j];
      if (occ[u]) continue;
      cur = (cur * dp[u + 1]) % M;
      occ[u] = 1;
    }
    dp[i] = cur;
    for (int j = 0; j < foe[p[i]].size(); j++) {
      int u = foe[p[i]][j];
      occ[u] = 0;
    }
    dp[i] = (dp[i] + dp[i + 1]) % M;
  }

  cout << dp[0] << "\n";

  return 0;
}
