#include <bits/stdc++.h>
using namespace std;

int main() {
  long double p[110];
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    p[i] = 1 - p[i];
  }
  sort(p, p + n);
  long double prob = 1.0;
  for (int i = 0; i < n; i++) {
    prob *= p[i];
    if (prob < 1e-9) break;
  }
  prob = 1 - prob;
  cout << setprecision(12) << fixed << prob << endl;
  return 0;
}
