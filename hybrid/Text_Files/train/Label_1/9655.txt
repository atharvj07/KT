#include <cstdio>
int main() {
  unsigned long long n, q, l, x, i, j;
  scanf("%llu %llu", &n, &q);
  for (i = 0; i < q; i++) {
    scanf("%llu", &x);
    if (x % 2)
      printf("%llu ", (x / 2) * 2 + 1);
    else {
      j = x;
      while ((j /= 2)) {
        if ((j % 2) == 0) {
          printf("%llu ", x);
          break;
        }
      }
      if ((j % 2) == 1) printf("%llu ", n + x - 2 * j + 1);
    }
  }
}
