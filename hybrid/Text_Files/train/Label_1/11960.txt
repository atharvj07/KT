#include <iostream>
using namespace std;

int main() {
  int n, k, m, t;
  cin >> n >> k >> m >> t;
  while (t--) {
    int op, pos;
    cin >> op >> pos;
    if (op == 0) {
      n--;
      if (pos == k) { k = m + 1 - n - k; }
      if (pos < k) { k--; }
    } else {
      n++;
      if (pos <= k) { k++; }
      if (n > m) { n--; k = m + 1 - n - k; }
    }
    cout << n << ' ' << k << endl;
  }
}
