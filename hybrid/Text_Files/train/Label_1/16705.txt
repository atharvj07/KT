#include <bits/stdc++.h>
using namespace std;
#define fi first
#define se second
const int MAXN = 2001;
int n, k, ans[MAXN];
string s[MAXN];
bool avail[7];
map<string, int> m = {
    {"1110111", 0}, {"0010010", 1}, {"1011101", 2}, {"1011011", 3},
    {"0111010", 4}, {"1101011", 5}, {"1101111", 6}, {"1010010", 7},
    {"1111111", 8}, {"1111011", 9}};
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> s[i];
    }
    if (k == 0) {
        for (int i = 1; i <= n; i++) {
            ans[i] = m[s[i]];
        }
    } else {
        bool good = true;
        for (int i = 1; i <= n; i++) {
            int cnt = 0;
            for (int j = 0; j < 7; j++) {
                cnt += (s[i][j] != '1');
                avail[j] |= (s[i][j] != '1');
            }
            if (cnt > k) {
                good = false;
                break;
            }
        }
        if (!good) {
            cout << "-1\n";
        } else {
            for (int i = 1; i <= n; i++) {
                int cnt = 0;
                for (int j = 0; j < 7; j++) {
                    if (!avail[j]) {
                        s[i][j] = '1';
                        cnt += (s[i][j] != '0');
                    }
                }
                while (cnt < k) {
                    for (int j = 0; j < 7; j++) {
                        if (!avail[j] && s[i][j] == '0') {
                            s[i][j] = '1';
                            cnt++;
                            break;
                        }
                    }
                }
                ans[i] = m[s[i]];
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        cout << ans[i];
    }
    cout << '\n';
    return 0;
}
