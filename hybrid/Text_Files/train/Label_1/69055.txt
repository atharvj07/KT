def solve_circle(n, pairs):
    from collections import defaultdict

    # Create a graph from the pairs
    graph = defaultdict(list)
    for a, b in pairs:
        graph[a].append(b)
        graph[b].append(a)
    
    # Check if any vertex doesn't have exactly 4 neighbors
    for key in graph:
        if len(graph[key]) != 4:
            return [-1]
    
    # Find starting point (any vertex could be a starting point)
    start = next(iter(graph))
    # The order list which will contain the final sequence
    order = [start]
    # Current explores the immediate next vertex
    current = graph[start][0]
    
    # Loop to find the order of numbers in the circle
    while len(order) < n:
        order.append(current)
        # Find the next vertex that is not in the order list
        # The next vertex must be in the adjacency list of the current vertex
        # and must not be the previous vertex in the order list
        next_vertex = None
        for neighbor in graph[current]:
            if neighbor not in order:
                next_vertex = neighbor
                break
        current = next_vertex
    
    return order

# Reading input
n = int(input().strip())
pairs = [list(map(int, input().strip().split())) for _ in range(2*n)]

# Solving and printing the result
result = solve_circle(n, pairs)
if result == [-1]:
    print(-1)
else:
    print(' '.join(map(str, result)))
