import sys

tc = int(input())
for i in range(tc):
  n, x = [int(x) for x in input().split()]
  arr = [int(x) for x in input().split()]
  a = arr.copy()
  a.sort()
  if arr == a:
    print("YES")
  else:
    ok = True
    for j in range(n - 1):
      if abs(arr[j] - arr[j + 1]) > x:
        ok = False
    if ok:
      print("YES")
    else:
      print("NO")
