#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;
const ll INF = 0x3f3f3f3f;
const int MAXN = 1e3 + 7;

int n, m;
vector<int> t;
vector<pi> intervals;

bool ok(ll x) {
    vector<int> sends;
    vector<int> done(n);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (intervals[j].first <= x && intervals[j].second >= x + t[i]) {
                sends.push_back(x);
                done[i] = 1;
            }
        }
    }
    sort(sends.begin(), sends.end());
    for (int i = 0; i < n; i++) {
        if (!done[i]) {
            auto it = lower_bound(sends.begin(), sends.end(), x + t[i]);
            if (it == sends.end()) {
                return false;
            }
        }
    }
    return true;
}

int main() {
    cin >> n;
    t.resize(n);
    for (int i = 0; i < n; i++) { cin >> t[i]; }
    cin >> m;
    intervals.resize(m);
    for (int i = 0; i < m; i++) { cin >> intervals[i].first >> intervals[i].second; }
    sort(intervals.begin(), intervals.end());
    ll l = 0, r = INF;
    ll ans = -1;
    while (l <= r) {
        ll mid = (l + r) / 2;
        if (ok(mid)) {
            ans = mid;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    cout << ans << '\n';
    return 0;
}
