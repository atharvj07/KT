#include <iostream>
#include <vector>
using namespace std;
int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  if (a + c > b + d || (a + c + d) % 3 != 0)
    cout << "NO";
  else {
    cout << "YES" << endl;
    int cnt = (a + c + d) / 3;
    for (int i = 0; i < a; i++) cout << 0 << " ";
    for (int i = 0; i < b; i++) cout << 1 << " ";
    for (int i = 0; i < cnt; i++) cout << 2 << " ";
    for (int i = 0; i < cnt; i++) cout << 3 << " ";
    for (int i = 0; i < cnt; i++) cout << 2 << " ";
  }
  return 0;
}
