#include <bits/stdc++.h>
using namespace std;
using ll=long long;
bool isp[1000010];
vector<int>pr;
void sieve(){
for(int i=0;i<1000010;i++)isp[i]=1;
isp[0]=isp[1]=0;
for(int i=2;i<1000010;i++){
if(isp[i]){
pr.push_back(i);
for(int j=i+i;j<1000010;j+=i)isp[j]=0;
}
}
}
int main(){
ll x2;
cin>>x2;
sieve();
set<int>st;
for(auto p:pr){
if(x2%p==0){
st.insert(x2/p);
st.insert((x2+p-1)/p);
}
}
cout<<*st.rbegin()<<endl;
return 0;
}
