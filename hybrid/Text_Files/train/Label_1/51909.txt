import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] lines = new int[n][3];
        for (int i = 0; i < n; i++) {
            lines[i][0] = sc.nextInt();
            lines[i][1] = sc.nextInt();
            lines[i][2] = sc.nextInt();
        }
        Arrays.sort(lines, (a, b) -> {
            double r = a[1] - b[1];
            if (r == 0) {
                r = a[0] - b[0];
            }
            return r < 0 ? -1 : 1;
        });
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long ax = lines[i][0], ay = lines[i][1], bx = lines[j][0], by = lines[j][1];
                if (ax * by != ay * bx) {
                    for (int k = j + 1; k < n; k++) {
                        long cx = lines[k][0], cy = lines[k][1];
                        long a = ax * (bx * cy - by * cx) + bx * (ay * cx - ax * cy) + cx * (ay * bx - ax * by);
                        if (a != 0) {
                            ans++;
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
