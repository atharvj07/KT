#include <bits/stdc++.h>

using namespace std;

int32_t main() {
  long long n, d, h;
  cin >> n >> d >> h;
  if (d % 2 == h % 2) {
    if (h < d - h) {
      cout << "-1" << endl;
      return 0;
    }
  } else {
    if (h != d - h || d % 2 == 0) {
      cout << "-1" << endl;
      return 0;
    }
  }
  for (long long i = 2; i <= d; i++) { cout << 1 << " " << i << endl; }
  long long cnt = n - d;
  while (cnt--) { cout << d << " " << d + 1 << endl; }
  return 0;
}
