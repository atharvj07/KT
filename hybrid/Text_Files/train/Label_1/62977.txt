n = int(input())
a = [int(x) for x in input().split()]
b = sorted(a)
ans = 0
for i in range(n):
    aa = []
    bb = []
    for j in range(n):
        if j == i: continue
        if j%2 == 0: aa.append(a[j])
        else: bb.append(a[j])
    if len(aa) == len(bb):
        if sum(aa) == sum(bb): ans+=1
    else:
        if len(aa) == len(bb) + 1 and sum(aa) - a[i] == sum(bb): ans+=1
        elif len(aa) + 1 == len(bb) and sum(aa) == sum(bb) - a[i]: ans+=1
print(ans)
