import sys
input = sys.stdin.readline

def get_classy_num(num):
    num = str(num)
    cnt = 0
    for i in num:
        if i!='0':
            cnt+=1
            if cnt>3:
                return False
    return True

t = int(input())

for _ in range(t):
    l,r = map(int, input().split())
    ans = 0
    for i in range(l, r+1):
        if get_classy_num(i):
            ans+=1
    print(ans)
