#include <cstdio>
#include <algorithm>
int64_t n, I, a[400005], s[400005], pref[400005];
int main() {
    scanf("%lld %lld", &n, &I);
    I *= 8;
    for (int64_t i = 0; i < n; i++) scanf("%lld\n", &a[i]);
    std::sort(a, a + n);
    int64_t sz = std::unique(a, a + n) - a, l = 0, r = n - 1, ans = 0;
    while (l < r) {
        if (I - (r - l + 1) * a[r] >= (a[r] - a[l]) * l) {
            I -= (r - l + 1) * a[r];
            ans += r - l + 1;
            l++;
        } else break;
    }
    for (int64_t i = 0; i < n; i++) {
        s[i] = a[i];
        pref[i + 1] = pref[i] + (a[i] > a[l] || a[i] < a[r]);
    }
    int64_t ll = 0, rr = n - 1, q = 0;
    for (int64_t i = 0; i < n; i++) {
        if (a[i] > a[l]) q++;
        if (l < r) {
            while (I - pref[rr + 1] + pref[ll] >= (a[rr] - a[ll]) * ll) {
                I -= pref[rr + 1] - pref[ll];
                rr--;
                if (a[ll] == a[l]) q--;
                ll++;
            }
        }
        ans += q;
    }
    printf("%lld", ans);
    return 0;
}
