#include <iostream>
#include <vector>
using namespace std;
const int MOD = 1e9 + 7;
int n, sub[300005], sum[300005];
vector<int> G[300005];
int dfs(int x, int pre) {
  sub[x] = 1;
  sum[x] = 0;
  for (int y : G[x]) {
    if (y == pre) continue;
    dfs(y, x);
    sum[x] += sum[y];
    sub[x] += sub[y];
  }
  return sum[x];
}
int main() {
  cin >> n;
  int type, x;
  for (int i = 2; i <= n; i++) {
    cin >> type;
    sum[i] = type;
  }
  for (int i = 2; i <= n; i++) {
    cin >> x;
    G[x].push_back(i);
    G[i].push_back(x);
  }
  dfs(1, 1);
  int mx = 0;
  for (int i = 1; i <= n; i++) {
    if (G[i].size() == 1) {
      sum[i] = 1;
    }
    if (G[i].size() > 1 && sum[i] == 0) {
      int tmp = 0, cnt = sub[i];
      for (int y : G[i]) {
        if (y != 1) {
          if (tmp <= sum[y]) {
            tmp = sum[y];
            cnt = sub[y];
          }
        }
      }
      sum[i] = tmp + 1;
      mx = max(mx, sum[i]);
    }
  }
  cout << mx << endl;
}
