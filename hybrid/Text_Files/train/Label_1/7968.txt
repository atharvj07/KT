#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> f(k, 0);
  int ans = 0;
  int l = 0, r = 0;
  int minf = k;
  while (r < n) {
    f[s[r] - 'A']++;
    minf = min(minf, f[s[r] - 'A']);
    while (minf * (r - l + 1) > n) {
      f[s[l] - 'A']--;
      l++;
      minf = *min_element(f.begin(), f.end());
    }
    ans = max(ans, r - l + 1);
    r++;
  }
  cout << ans << endl;
}
