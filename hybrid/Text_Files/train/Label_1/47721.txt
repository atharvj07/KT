import java.io.*;
import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int b = in.nextInt();
    int[] x = new int[n];
    int[] k = new int[n];
    int[][] a = new int[n][m];
    for (int i = 0; i < n; i++) {
      x[i] = in.nextInt();
      k[i] = in.nextInt();
      int mi = in.nextInt();
      for (int j = 0; j < mi; j++) {
        a[i][in.nextInt() - 1]++;
      }
    }
    int[] d = new int[1 << m];
    Arrays.fill(d, (int) 1e9);
    d[0] = 0;
    for (int i = 0; i < (1 << m); i++) {
      for (int j = 0; j < n; j++) {
        int nw = i;
        for (int l = 0; l < m; l++) {
          if ((i & (1 << l)) != 0) {
            nw |= a[j][l];
          }
        }
        if (nw == i) {
          continue;
        }
        d[nw] = Math.min(d[nw], d[i] + x[j] + (k[j] - Integer.bitCount(nw & ((1 << m) - 1))) * b);
      }
    }
    if (d[(1 << m) - 1] < (int) 1e9) {
      System.out.println(d[(1 << m) - 1]);
    } else {
      System.out.println(-1);
    }
  }
}
