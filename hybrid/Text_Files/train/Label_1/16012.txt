#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, k, l;
    cin >> n >> k >> l;
    vector<int> a(n * k);
    for (int &i : a) cin >> i;
    sort(a.begin(), a.end());
    int ans = 0, cnt = 0, t;
    for (int i = 0; i < n * k;) {
        t = a[i];
        while (i < n * k && a[i] == t) cnt++, i++;
        if (cnt >= k) {
            ans += t;
            cnt -= k;
        } else break;
    }
    if (cnt) ans = 0;
    cout << ans << endl;
    return 0;
}
