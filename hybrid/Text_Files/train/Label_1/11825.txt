#include <bits/stdc++.h>
using namespace std;

#ifdef DEBUG
#define db(...) fprintf(stderr, __VA_ARGS__)
#else
#define db(...)
#endif

#define ll long long
#define ull unsigned long long
#define SZ(x) ((int)x.size())
#define all(x) x.begin(), x.end()
#define clr(x, i) memset(x, i, sizeof(x))
#define F(i, n) for (int i = 0; i < n; ++i)
#define FD(i, n) for (int i = n - 1; i >= 0; --i)
#define fori(i, x, n) for (int i = x; i <= n; ++i)

bool vis[3005];
vector<int> adj[3005];
int d1[3005], d2[3005];

int dist(int s, int t, int lim)
{
    clr(d1, 0x3f), clr(d2, 0x3f), d1[s] = 0;
    queue<pair<int, int>> q;
    q.push({s, 0});
    while (!q.empty())
    {
        pair<int, int> u = q.front();
        q.pop();
        if (u.second > lim)
            continue;
        if (u.first == t)
            return u.second;
        for (int v : adj[u.first])
            if (u.second + 1 <= lim && d1[v] > u.second + 1 && !vis[v])
                d1[v] = u.second + 1, q.push({v, u.second + 1});
    }
    return -1;
}

int main()
{
    int n, m;
    while (cin >> n >> m)
    {
        F(i, n + 1) adj[i].clear(), vis[i] = 0;
        F(i, m)
        {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        int s1, t1, l1, s2, t2, l2;
        cin >> s1 >> t1 >> l1;
        cin >> s2 >> t2 >> l2;
        if (l2 < (dist(s1, t1, l1) > -1 ? dist(s1, t1, l1) : 2e9) || l1 < (dist(s2, t2, l2) > -1 ? dist(s2, t2, l2) : 2e9))
            cout << -1 << endl;
        else
        {
            vector<pair<pair<int, int>, pair<int, int>>> v;
            F(i, n + 1)
            {
                for (int j : adj[i])
                {
                    if (i < j)
                        v.push_back({{i, j}, {dist(i, s1, l1 - 1) > -1 ? dist(i, s1, l1 - 1) : 2e9, dist(j, s1, l1 - 1) > -1 ? dist(j, s1, l1 - 1) : 2e9}});
                }
            }
            int lb = -1, rb = 2e9, ans = -1;
            while (lb <= rb)
            {
                int mid = (lb + rb) / 2;
                bool valid = 1;
                for (pair<pair<int, int>, pair<int, int>> w : v)
                {
                    int i = w.first.first;
                    int j = w.first.second;
                    int c1 = w.second.first;
                    int c2 = w.second.second;
                    vis[i] = 1;
                    vis[j] = 1;
                    if (c1 + c2 + 2 > mid)
                    {
                        valid = 0;
                        break;
                    }
                    if (dist(s1, t1, l1 - 1) > -1 && dist(s2, t2, l2 - 1) > -1)
                    {
                        int lft = l1 - 1 - c1 - c2;
                        if (dist(i, t1, lft) > -1 && dist(j, t2, lft) > -1)
                            continue;
                    }
                    vis[i] = 0;
                    vis[j] = 0;
                    valid = 0;
                    break;
                }
                if (valid)
                {
                    rb = mid - 1;
                    ans = mid;
                }
                else
                    lb = mid + 1;
            }
            cout << ans << endl;
        }
    }
    return 0;
}
