#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> f(n);
  for (int i = 0; i < n; ++i) {
    cin >> f[i];
    --f[i];
  }
  for (int i = 0; i < n; ++i) {
    if (f[f[f[i]]] == i) {
      cout << "YES" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
  return 0;
}
