import sys

def get_max(a):
  ans = -1
  for x in a:
    if x > ans:
      ans = x
  return ans

def get_min(a):
  ans = sys.maxsize
  for x in a:
    if x < ans:
      ans = x
  return ans

n = int(input())
a = []
for i in range(n):
  s = int(input())
  a.append(s)
a.sort()
m = get_max(a)
if n == 4:
  if a[1] == a[2] == a[3]:
    print("YES")
    print(a[1])
    print(a[1])
    print(a[1])
    print(a[1])
  else:
    print("NO")
elif n == 3:
  mn = get_min(a)
  mx = get_max(a)
  if a[1] - a[0] == a[2] - a[1] and a[1] - a[0] > 0 and a[2] - a[1] > 0:
    print("YES")
    print(a[0] - a[1] + mn)
    print(a[0] - a[1] + mn)
    print(a[2] - a[1] + mx)
    print(a[2] - a[1] + mx)
  else:
    print("NO")
elif n == 2:
  mn = get_min(a)
  mx = get_max(a)
  if mx - mn > 0 and mx - mn <= 2:
    print("YES")
    print(mn - 2 * (mx - mn))
    print(mn - 2 * (mx - mn))
    print(mn)
    print(mx)
  else:
    print("NO")
elif n == 1:
  print("YES")
  print(a[0])
  print(a[0])
  print(a[0])
  print(a[0])
else:
