import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int minIndex = 1;
        long minVolume = Long.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            long volume = in.nextInt();
            if (volume < minVolume) {
                minVolume = volume;
                minIndex = i;
            }
        }
        System.out.println(minIndex);
    }
}
