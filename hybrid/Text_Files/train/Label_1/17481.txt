#include<bits/stdc++.h>
using namespace std;
#define FOR(i, a, b) for (int i = (a); i < (b); ++i)
#define REP(i, n) FOR(i, 0, n)
typedef long long ll;
const int maxn = 2e5 + 5;
int n, m[maxn], s[maxn], ans;
vector<int> a[maxn];
int main() {
  cin >> n;
  REP(i, n) {
    cin >> m[i];
    a[i].resize(m[i]);
    REP(j, m[i]) cin >> a[i][j], s[i] = max(s[i], a[i][j]);
  }
  sort(s, s + n);
  int j = n - 1, sum = 0;
  REP(i, n) {
    ans += sum + s[i] - a[i][0];
    sum += s[i] - a[i][0];
    while (j >= 0 && s[j--] >= s[i]);
    sum -= (n - j - 1) * (s[i] - a[i][0]);
  }
  cout << ans << endl;
}
