n, s = map(int, input().split())
orders = {'B': {}, 'S': {}}

for _ in range(n):
    d, p, q = input().split()
    p, q = int(p), int(q)
    if p in orders[d]:
        orders[d][p] += q
    else:
        orders[d][p] = q

buy_orders = sorted(orders['B'].items(), key=lambda x: -x[0])[:s]
sell_orders = sorted(orders['S'].items(), key=lambda x: x[0])[:s]

for order in sell_orders:
    print('S', order[0], order[1])
for order in buy_orders:
    print('B', order[0], order[1])
