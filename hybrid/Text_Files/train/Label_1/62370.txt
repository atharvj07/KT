n, t = map(int, input().split())
a = list(map(int, input().split()))

left, right, sum_time, max_books = 0, 0, 0, 0

while right < n:
    if sum_time + a[right] <= t:
        sum_time += a[right]
        right += 1
        max_books = max(max_books, right - left)
    else:
        sum_time -= a[left]
        left += 1

print(max_books)
