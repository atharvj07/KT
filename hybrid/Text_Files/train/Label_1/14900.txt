#include<bits/stdc++.h>
using namespace std;

const int MAX_N = 1e5 + 5;
const int mod = 1e9 + 7;
vector<int> adj[MAX_N];
int dist[MAX_N];
int vis[MAX_N];
int n;
int ans;

void dfs(int node, int par, int cost) {
    if (vis[node]) {
        ans = min(ans, cost + dist[node]);
        return;
    }
    vis[node] = 1;
    dist[node] = cost;
    for (int child : adj[node]) {
        if (child != par) {
            dfs(child, node, (cost + node + child) % (n + 1));
        }
    }
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == j) {
                continue;
            }
            adj[i].push_back(j);
            adj[j].push_back(i);
        }
    }
    ans = mod;
    dfs(1, 1, 0);
    cout << ans << endl;
    return 0;
}
