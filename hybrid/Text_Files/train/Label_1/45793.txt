import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] constraints = new int[m][3];
        for (int i = 0; i < m; i++) {
            constraints[i][0] = scanner.nextInt();
            constraints[i][1] = scanner.nextInt();
            constraints[i][2] = scanner.nextInt();
        }

        int[] solution = solve(n, m, constraints);
        System.out.println(solution[n]);
    }

    private static int[] solve(int n, int m, int[][] constraints) {
        int[] solution = new int[n + 1];
        boolean[] visited = new boolean[n + 1];
        for (int i = 0; i < m; i++) {
            if (!visited[constraints[i][0]]) {
                solution[constraints[i][0]] = (constraints[i][2] - solution[constraints[i][1]]) % 2;
                visited[constraints[i][0]] = true;
            }
            if (!visited[constraints[i][1]]) {
                solution[constraints[i][1]] = (constraints[i][2] - solution[constraints[i][0]]) % 2;
                visited[constraints[i][1]] = true;
            }
        }

        int cost = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                solution[i] = 1;
                cost++;
            }
        }

        solution[n] = cost;
        return solution;
    }
}
