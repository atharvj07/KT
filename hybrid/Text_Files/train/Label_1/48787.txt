import java.util.Scanner;

public class SubnetMask {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] addresses = new int[n];
        for (int i = 0; i < n; i++) {
            addresses[i] = ipToInt(scanner.next());
        }

        int mask = findSubnetMask(addresses, k);
        if (mask != -1) {
            System.out.println(intToIp(mask));
        } else {
            System.out.println(-1);
        }
    }

    private static int findSubnetMask(int[] addresses, int k) {
        int mask = 0;
        for (int i = 31; i >= 0; i--) {
            int bitCount = 0;
            for (int address : addresses) {
                if ((address >> i & 1) == 1) {
                    bitCount++;
                }
            }

            if (bitCount > k) {
                mask |= (1 << i);
            } else if (bitCount < k) {
                return -1;
            }
        }

        return mask;
    }

    private static int ipToInt(String ip) {
        String[] parts = ip.split("\\.");
        return (Integer.parseInt(parts[0]) << 24) |
               (Integer.parseInt(parts[1]) << 16) |
               (Integer.parseInt(parts[2]) << 8) |
               Integer.parseInt(parts[3]);
    }

    private static String intToIp(int ip) {
        return String.format("%d.%d.%d.%d", (ip >> 24) & 0xFF, (ip >> 16) & 0xFF, (ip >> 8) & 0xFF, ip & 0xFF);
    }
}
