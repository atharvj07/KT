n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

ans = 0
for idx, val in enumerate(a):
    if val != 0:
        idxa = idx
        break

for idx, val in enumerate(b):
    if val != 0:
        idxb = idx
        break

diff = idxa - idxb
if diff < 0:
    diff += n

ans += diff

for x in a:
    if x != 0:
        ans += 1

print(ans)
