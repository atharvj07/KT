#include <bits/stdc++.h>
using ll = long long;

int main() {
    ll a;
    std::cin >> a;
    std::string s;
    std::cin >> s;
    int n = s.size();
    std::vector<ll> v;
    for (char c : s) v.push_back(c - '0');
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            ll x = v[i], y = 0;
            int l = (j - i + 1) / 2;
            for (int k = 0; k < l; k++) {
                x *= 10;
                y *= 10;
                x += v[i + k];
                y += v[j - k];
            }
            if (x * y == a) ans++;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
