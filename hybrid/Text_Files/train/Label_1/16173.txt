#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 5;

int n, q, a[MAXN], dp[MAXN];

inline void init() {
  for (int i = 0; i < n; i++) cin >> a[i];
}
inline void solve() {
  dp[0] = 0;
  for (int i = 1; i < MAXN; i++) dp[i] = -1;
  for (int i = 0; i < n; i++) {
    for (int j = MAXN - 1; j >= a[i]; j--) {
      dp[j] = max(dp[j], dp[j - a[i]] + 1);
    }
  }
  for (int i = 0; i < q; i++) {
    int x;
    cin >> x;
    cout << dp[x] << '\n';
  }
}

int main() {
  cin >> n >> q;
  init();
  solve();
  return 0;
}
