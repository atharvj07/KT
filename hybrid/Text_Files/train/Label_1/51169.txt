import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        var n = sc.nextInt();
        var arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        boolean isAdd = true;
        long ans = 0;
        for (long num : arr) {
            if (isAdd) {
                ans += num;
            } else {
                ans -= num;
            }
            isAdd = !isAdd;
        }
        System.out.println((ans + 1000000007) % 1000000007);
    }
}
