import sys
input = sys.stdin.readline

n = int(input())
m = list(map(int, input().split()))
l = [-1] * n
r = [-1] * n
for i in range(n):
    for j in range(n):
        if j < i and m[j] > m[i]:
            r[i] = j
        if j > i and m[j] > m[i]:
            l[i] = j
dp = [0] * n
dp[0] = m[0]
for i in range(1, n):
    if l[i] == -1:
        dp[i] = max(m[i], dp[i - 1])
    else:
        dp[i] = max(m[i], dp[l[i]])
ans = []
i = n - 1
while i >= 0:
    t = m[i]
    if l[i] != -1:
        t = max(t, dp[l[i]])
    if r[i] != -1:
        t = max(t, dp[r[i]])
    ans.append(t)
    i = l[i]
ans.reverse()
print(*ans)
