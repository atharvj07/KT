#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e3 + 10;
const int mod = 998244353;
int n, k, a[maxn];
int rnk[maxn], ord[maxn], p[maxn], cnt[maxn];
int sum[maxn][maxn], ex[maxn];
int cnt_p[maxn];

int Pow(int a, int b) {
    long long ans = 1;
    for (; b; b >>= 1, a = (long long)a * a % mod) {
        if (b & 1) ans = ans * a % mod;
    }
    return ans;
}

int inv(int x) { return Pow(x, mod - 2); }

// 判断 [L, R] 是否为幸运子段
bool is_lucky(int L, int R) {
    return ord[L] + k - 1 <= ord[R];
}

void add(int a, int c) {
    for (; a <= n; a += a & (-a)) { ex[a] += c; }
}

int que(int a) {
    int ans = 0;
    for (; a; a -= a & (-a)) { ans += ex[a]; }
    return ans;
}

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i], rnk[i] = a[i];
    sort(rnk + 1, rnk + n + 1);
    for (int i = 1; i <= n; i++) {
        ord[i] = lower_bound(rnk + 1, rnk + n + 1, a[i]) - rnk;
        p[i] = ord[i];
    }
    sort(p + 1, p + n + 1);
    for (int i = 1; i <= n; i++) {
        if (i > 1 && p[i] == p[i - 1]) cnt_p[p[i]]++;
    }
    for (int i = 1; i <= n; i++) {
        sum[ord[i]][i] += 1, add(ord[i], 1);
    }
    for (int j = 1; j <= n; j++) {
        for (int i = 1; i <= n; i++) {
            sum[i][j] += sum[i][j - 1];
        }
        for (int i = 1; i <= n - k + 1; i++) {
            if (!is_lucky(i, i + k - 1)) continue;
            int l = ord[i], r = ord[i + k - 1];
            int pp = que(r) - que(l - 1) + cnt_p[l];
            sum[l][j] = (sum[l][j] - 1ll * (pp - 1) % mod * (1ll * pp * inv(2) % mod) % mod + mod) % mod;
        }
        for (int i = 1; i <= n; i++) {
            add(ord[i], -1);
        }
    }
    int ans = 0;
    for (int i = 1; i <= n - k + 1; i++) {
        ans = (ans + 1ll * sum[i][n] * (1ll * sum[i][n] - 1) % mod * inv(2) % mod) % mod;
    }
    cout << ans << endl;
    return 0;
}
