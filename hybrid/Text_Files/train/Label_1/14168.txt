#include <iostream>

using namespace std;

int main() {
    int x = 0;
    int n;
    char op = '+';
    while (cin >> n) {
        switch (op) {
            case '+':
                x += n;
                break;
            case '-':
                x -= n;
                break;
            default:
                break;
        }
        cin >> op;
    }
    cout << x << endl;
    return 0;
}
