#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> d(n);
  for (int i = 0; i < n; i++) cin >> d[i];
  unordered_map<int, int> m;
  for (int i = 0; i < n; i++) m[d[i] % k]++;
  long long ans = 0;
  for (auto& [key, val] : m) {
    if (key == 0)
      ans += val / 2;
    else if (k % 2 == 0 && key == k / 2)
      ans += val / 2;
    else if (m.count(k - key))
      ans += min(m[key], m[k - key]);
  }
  cout << ans * 2 << endl;
}
