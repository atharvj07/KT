import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class ChoutiAndTheSequence {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Integer> evens = new ArrayList<>();
        for (int i = 0; i < n / 2; i++) {
            evens.add(scanner.nextInt());
        }

        if (!isValid(evens)) {
            System.out.println("No");
            return;
        }

        System.out.println("Yes");
        List<Integer> result = new ArrayList<>();
        result.addAll(reconstructSequence(evens));
        for (int num : evens) {
            result.add(num);
        }
        for (int num : result) {
            System.out.print(num + " ");
        }
    }

    private static boolean isValid(List<Integer> evens) {
        int sum = 0;
        for (int num : evens) {
            sum += num;
            if (!isPerfectSquare(sum)) {
                return false;
            }
        }
        return true;
    }

    private static boolean isPerfectSquare(int num) {
        return (int) Math.sqrt(num) * (int) Math.sqrt(num) == num;
    }

    private static List<Integer> reconstructSequence(List<Integer> evens) {
        List<Integer> result = new ArrayList<>();
        int currentSum = 0;
        for (int i = 0; i < evens.size() - 1; i++) {
            int nextSquare = getNextSquare(currentSum + evens.get(i));
            result.add(nextSquare - (currentSum + evens.get(i)));
            currentSum = nextSquare;
        }
        return result;
    }

    private static int getNextSquare(int num) {
        int sqrt = (int) Math.sqrt(num);
        while (!isPerfectSquare(sqrt * sqrt)) {
            sqrt++;
        }
        return sqrt * sqrt;
    }
}
