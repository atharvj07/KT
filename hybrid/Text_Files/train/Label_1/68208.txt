n,m=map(int,input().split())
if n>1 and m==1:
    print('NO')
else:
    print('YES')
    for i in range(n):
        if i%2==0:
            for j in range(0,m,2):
                print(i*m+j+1,end=' ')
            for j in range(1,m,2):
                print(i*m+j+1,end=' ')
        else:
            for j in range(1,m,2):
                print(i*m+j+1,end=' ')
            for j in range(0,m,2):
                print(i*m+j+1,end=' ')
