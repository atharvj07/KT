def solve(n, p):
    swaps = []
    
    def swap_positions(i, j):
        p[i], p[j] = p[j], p[i]
        swaps.append((i + 1, j + 1))  # Transform to 1-based index

    for i in range(n // 2):
        correct_pos = p[i] - 1
        if correct_pos != i:
            if abs(correct_pos - i) < n // 2:
                if correct_pos < n // 2:
                    swap_positions(correct_pos, n - 1)
                    swap_positions(i, n - 1)
                else:
                    swap_positions(i, 0)
                    swap_positions(0, correct_pos)
                    swap_positions(i, 0)
            else:
                swap_positions(i, correct_pos)

    for i in range(n - 1, n // 2 - 1, -1):
        correct_pos = p[i] - 1
        if correct_pos != i:
            if abs(correct_pos - i) < n // 2:
                if correct_pos >= n // 2:
                    swap_positions(correct_pos, 0)
                    swap_positions(i, 0)
                else:
                    swap_positions(i, n - 1)
                    swap_positions(n - 1, correct_pos)
                    swap_positions(i, n - 1)
            else:
                swap_positions(i, correct_pos)

    return swaps

n = int(input().strip())
p = list(map(int, input().strip().split()))

swaps = solve(n, p)
print(len(swaps))
for swap in swaps:
    print(swap[0], swap[1])
