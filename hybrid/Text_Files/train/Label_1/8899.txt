#include <bits/stdc++.h>
using namespace std;
main(int argc, char **argv) {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> seen(n, 0), ans(n, -1);
  queue<int> q;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'U') q.push(i);
    if (i == n - 1) ans[i] = 0;
    while (!q.empty()) {
      int f = q.front();
      q.pop();
      if (seen[f]) continue;
      seen[f] = 1;
      ans[f] = i - f + 1;
      if (f) {
        if (s[f - 1] != s[f]) q.push(f - 1);
      }
      if (f != n - 1) {
        if (s[f + 1] != s[f]) q.push(f + 1);
      }
    }
  }
  for (int x : ans) cout << x << ' ';
  cout << endl;
}
