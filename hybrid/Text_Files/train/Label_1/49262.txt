import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), m = sc.nextInt();
            long[] a = new long[n], b = new long[n];
            long sumA = 0, sumB = 0;
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                sumA += a[i];
            }
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
                sumB += b[i];
            }
            if (sumA < m) {
                System.out.println(-1);
            } else {
                Arrays.sort(a);
                Arrays.sort(b);
                long ans = 0;
                for (int i = 0; i < n; i++) {
                    ans += b[i];
                    sumA -= a[i];
                    if (sumA < m) break;
                }
                System.out.println(ans);
            }
        }
    }
}
