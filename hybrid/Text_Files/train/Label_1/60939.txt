import sys
input = sys.stdin.readline

N,M = map(int,input().split())
A,B = [],[]
for i in range(N):
    x,y = map(int,input().split())
    A.append(x)
    B.append(y)
D = {}
for i in range(M):
    x,y = map(int,input().split())
    x-=1
    y-=1
    if x>y: x,y=y,x
    if (x,y) not in D: D[(x,y)] = 1
    else: D[(x,y)] += 1
D2 = {}
for i in range(N):
    for j in range(i+1,N):
        if (i,j) not in D:
            if i not in D2: D2[i] = [j]
            else: D2[i].append(j)
            if j not in D2: D2[j] = [i]
            else: D2[j].append(i)
INF = 1<<60
ans = [INF for i in range(N)]
for i in range(N):
    id = i
    L = []
    while True:
        if id==i and len(L)>0: break
        if id not in D2: break
        L.append(id)
        mn = INF
        id2 = -1
        for j in D2[id]:
            if j not in L and A[id]+B[j]<mn:
                mn = A[id]+B[j]
                id2 = j
        if id2==-1: break
        id = id2
    for j in L: ans[j] = min(ans[j],mn)
for x in ans: print(x if x!=INF else 0)
