n, k = map(int, input().split())
s = input()
for i in range(n):
    if k <= 0: break
    if int(s[i]) > 0 or i == n - 1: continue
    s = '0' + s[1:]
    k -= 1
if k > 0:
    for i in range(n - 1, -1, -1):
        if k <= 0: break
        if int(s[i]) < 9:
            s = s[:i] + str(int(s[i]) + 1) + s[i + 1:]
            k -= 1
print(s)
