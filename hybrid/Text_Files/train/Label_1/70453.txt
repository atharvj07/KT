import heapq

n, m, k = map(int, input().split())
l = list(map(int, input().split()))
edges = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b, w = map(int, input().split())
    edges[a].append((b, w))
    edges[b].append((a, w))

dist = [[-1] * (k + 1) for _ in range(n + 1)]
for i in range(1, k + 1):
    q = [(0, l[i - 1])]
    while q:
        d, u = heapq.heappop(q)
        if dist[u][i] == -1:
            dist[u][i] = d
            for v, w in edges[u]:
                if dist[v][i] == -1:
                    heapq.heappush(q, (d + w, v))

ans = [0] * k
for i in range(1, k + 1):
    for j in range(1, k + 1):
        if i != j:
            ans[i - 1] = max(ans[i - 1], dist[l[i - 1]][j])

print(*ans)
