# -*- coding: utf-8 -*-

n, m = map(int, input().split())
c = list(map(int, input().split()))

a = [[0] * m for _ in range(m)]
for i in c:
    a[i - 1][i - 1] = 1

for i in range(m):
    for j in range(i + 1, m):
        a[i][j] = a[j][i] = 1

l = [0] * n
r = [0] * n

for i in range(n):
    for j in range(m):
        if a[c[i] - 1][j] and not l[i] and not r[i]:
            l[i] = c[i]
            r[i] = j + 1
            a[c[i] - 1][j] = a[j][c[i] - 1] = 0
            break

print(sum(x > 0 for x in l))
for i in range(n):
    print(l[i], r[i])
