#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 1;
int s[N], g[N], a[N], sum[N], ss[N], dp[N][2];
int32_t main() {
    int n;
    cin >> n;
    bool f = 0;
    for (int i = 1; i <= n; i++) {
        cin >> s[i] >> g[i];
        if (g[i] < a[i - 1]) f = 1;
        a[i] = g[i];
    }
    for (int i = n; i >= 1; i--) {
        dp[i][0] = min(dp[i + 1][0] + s[i], dp[i + 1][1] + max(s[i] - 1, 0));
        dp[i][1] = min(dp[i + 1][0] + max(s[i] + 1, 0), dp[i + 1][1] + s[i]);
        sum[i] = max(dp[i][0], dp[i][1]);
    }
    if (f) {
        cout << -1 << '\n';
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        if (dp[i][1] == sum[i])
            ss[i] = s[i];
        else if (dp[i + 1][0] + s[i] == sum[i])
            ss[i] = s[i] + 1;
        else
            ss[i] = s[i] - 1;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        cout << ss[i] << ' ';
        ans += ss[i] - s[i];
    }
    cout << '\n' << ans << '\n';
}
