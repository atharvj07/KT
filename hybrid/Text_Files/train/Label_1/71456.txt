t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    grid = [list(map(int, input().split())) for _ in range(n)]
    negatives = 0
    min_abs_value = float('inf')
    total_sum = 0
    for row in grid:
        for value in row:
            total_sum += abs(value)
            if value < 0:
                negatives += 1
            min_abs_value = min(min_abs_value, abs(value))
    if negatives % 2 == 0:
        print(total_sum)
    else:
        print(total_sum - 2 * min_abs_value)
