import java.util.*;
import java.io.*;

class Main {
    static int n, k;
    static char[] s;
    static ArrayList<Integer>[] adj;
    static boolean[] vis;
    static int[][] dp;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        s = br.readLine().toCharArray();
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        vis = new boolean[n + 1];
        dp = new int[n + 1][2];
        for (int i = 0; i <= n; i++) Arrays.fill(dp[i], -1);
        for (int i = 1; i <= k; i++) {
            int c = Integer.parseInt(br.readLine());
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < c; j++) {
                int u = Integer.parseInt(st.nextToken());
                adj[u].add(i);
            }
        }
        System.out.println(dfs(1, 0));
        for (int i = 2; i <= n; i++) {
            System.out.println(dfs(i, 0));
        }
    }

    static int dfs(int u, int state) {
        if (u > n) {
            if (state == 0) return 0;
            return 100000000;
        }
        if (dp[u][state] != -1) return dp[u][state];
        int ans = 100000000;
        if (state == 1) {
            // keep current state
            ans = Math.min(ans, dfs(u + 1, 1));
            // flip current state
            ans = Math.min(ans, 1 + dfs(u + 1, 0));
            for (int v : adj[u]) {
                if (vis[v]) continue;
                vis[v] = true;
                ans = Math.min(ans, 1 + dfs(u + 1, 1));
                vis[v] = false;
            }
        } else {
            // keep current state
            ans = Math.min(ans, dfs(u + 1, 0));
            for (int v : adj[u]) {
                if (vis[v]) continue;
                vis[v] = true;
                ans = Math.min(ans, 1 + dfs(u + 1, 1));
                vis[v] = false;
            }
        }
        return dp[u][state] = ans;
    }
}
