#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
    string n;
    cin >> n;
    int ans = 0;
    while (n != "0") {
        int m = n.size(), mn = 10;
        for (int i = 0; i < m; i++) {
            int dig = n[i] - '0';
            if (dig != 0)
                mn = min(mn, dig);
        }
        for (int i = 0; i < m; i++) {
            int dig = n[i] - '0';
            if (dig >= mn) {
                dig -= mn;
                n[i] = dig + '0';
            }
        }
        while (m > 1 && n[m - 1] == '0')
            m--;
        n = n.substr(0, m);
        ans++;
    }
    cout << ans << endl;
    return 0;
}
