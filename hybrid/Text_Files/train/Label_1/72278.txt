n = int(input())
a = list(map(float, input().split()))

# Calculate the difference between the original sum and the sum after rounding down all elements.
original_sum = sum(a)
rounded_down_sum = sum(int(x) for x in a)
diff = original_sum - rounded_down_sum

# Count how many elements have a fractional part.
fractional_parts = [x - int(x) for x in a]
non_zero_fractions = sum(1 for x in fractional_parts if x > 0)

# The optimal strategy is to round up as many elements as possible without exceeding the original sum.
# This is equivalent to finding the minimum absolute difference.
min_diff = abs(diff - min(n, non_zero_fractions))
print(f"{min_diff:.3f}")
