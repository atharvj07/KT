import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int b = scanner.nextInt();
        int d = scanner.nextInt();

        int[] divisors = {2, 3, 4, 6, 8, 9, 10, 11};

        for (int divisor : divisors) {
            if (d % divisor == 0) {
                switch (divisor) {
                    case 2:
                    case 4:
                    case 8:
                    case 10:
                        System.out.println("2-type");
                        System.out.println(getLeastLastDigits(b, divisor));
                        return;
                    case 3:
                    case 9:
                        System.out.println("3-type");
                        return;
                    case 11:
                        System.out.println("11-type");
                        return;
                    case 6:
                        if (b % 2 == 0 && getSumOfDigits(b, d) % divisor == 0) {
                            System.out.println("6-type");
                            return;
                        }
                        break;
                }
            }
        }

        System.out.println("7-type");
    }

    private static int getLeastLastDigits(int b, int divisor) {
        if (divisor == 10) {
            return 1;
        }
        int lastDigits = b;
        while (lastDigits % divisor != 0) {
            lastDigits *= b;
        }
        return lastDigits / divisor;
    }

    private static int getSumOfDigits(int b, int divisor) {
        int sum = 0;
        while (b > 0) {
            sum += b % divisor;
            b /= divisor;
        }
        return sum;
    }
}
