import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x1 = sc.nextInt();
        int y1 = sc.nextInt();
        int x2 = sc.nextInt();
        int y2 = sc.nextInt();
        int x3 = sc.nextInt();
        int y3 = sc.nextInt();
        int a = (x2 - x1) * (x3 - x2) + (y2 - y1) * (y3 - y2);
        int b = (x2 - x1) * (x3 - x1) + (y2 - y1) * (y3 - y1);
        int c = (x3 - x2) * (x1 - x3) + (y3 - y2) * (y1 - y3);
        if (a == 0 || b == 0 || c == 0) {
            System.out.println("RIGHT");
        } else if (a == b || b == c || c == a) {
            System.out.println("ALMOST");
        } else {
            System.out.println("NEITHER");
        }
    }
}
