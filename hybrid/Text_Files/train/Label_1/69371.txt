n = int(input())
positions = input()

standing = positions.count('X')
sitting = n // 2
minutes = 0
result = list(positions)

if standing > sitting:
    for i in range(n):
        if standing == sitting:
            break
        if result[i] == 'X':
            result[i] = 'x'
            standing -= 1
            minutes += 1
elif standing < sitting:
    for i in range(n):
        if standing == sitting:
            break
        if result[i] == 'x':
            result[i] = 'X'
            standing += 1
            minutes += 1

print(minutes)
print(''.join(result))
