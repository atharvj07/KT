#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    int ans = 0;
    vector<char> t(n);
    t[0] = s[0];
    for (int i = 1; i < n; i++) {
        if (s[i] == t[i - 1]) {
            ans++;
            if (t[i - 1] == 'R') {
                t[i] = 'G';
            } else if (t[i - 1] == 'G') {
                t[i] = 'B';
            } else {
                t[i] = 'R';
            }
        } else {
            t[i] = s[i];
        }
    }

    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        cout << t[i];
    }
    cout << endl;

    return 0;
}
