#include <iostream>
#include <vector>
using int64 = long long;
int main() {
    int64 n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int64>> gr(n + 1);
    for (int64 i = 0; i < m; ++i) {
        int64 a, b;
        std::cin >> a >> b;
        gr[a].push_back(b);
    }
    std::vector<bool> vis(n + 1);
    int64 ans = 0;
    for (int64 i = 1; i <= n; ++i) {
        if (vis[i] == true) continue;
        ans++;
        std::vector<int64> q;
        q.push_back(i);
        vis[i] = true;
        while (!q.empty()) {
            int64 u = q.back();
            q.pop_back();
            for (int64 v : gr[u]) {
                if (vis[v] == true) continue;
                q.push_back(v);
                vis[v] = true;
            }
        }
    }
    std::cout << ans - 1 << std::endl;
}
