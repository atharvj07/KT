#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;

long long powmod(long long x, long long e) {
  if (e == 0) return 1;
  if (e == 1) return x;
  long long res = powmod(x, e / 2);
  res = (res * res) % mod;
  if (e % 2) res = (res * x) % mod;
  return res;
}

int main() {
  int n;
  cin >> n;
  map<int, pair<bool, int>> a;
  long long ans = 1;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    int x;
    cin >> x;
    if (s == "ADD") {
      if (a.find(x) != a.end()) {
        cout << 0 << endl;
        return 0;
      }
      a[x].first = 0;
    } else {
      if (a.find(x) == a.end()) {
        cout << 0 << endl;
        return 0;
      }
      if (a[x].first == 0) {
        a[x].first = 1;
        a[x].second = i;
      } else {
        if (!a[x].second) {
          cout << 0 << endl;
          return 0;
        }
        if (i - a[x].second > 1) ans = (ans * 2) % mod;
      }
    }
  }
  for (auto p : a) {
    if (!p.second.first || !p.second.second) {
      cout << 0 << endl;
      return 0;
    }
  }
  cout << ans << endl;
  return 0;
}
