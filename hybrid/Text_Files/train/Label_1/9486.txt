#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    long long n, a, b, sum = 0, ans;
    cin >> n >> a >> b;
    vector<long long> s(n);
    for (int i = 0; i < n; i++) {
        cin >> s[i];
        sum += s[i];
    }
    sort(s.begin(), s.end());
    for (ans = 0; ans < n; ans++) {
        sum -= s[ans];
        if ((a * s[ans]) >= (b * sum)) break;
    }
    cout << ans << endl;
    return 0;
}
