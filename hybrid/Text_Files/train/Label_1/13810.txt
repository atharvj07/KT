#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> grades(n);
  for (int i = 0; i < n; i++) cin >> grades[i];
  int sum = 0;
  int worst = 2;
  for (int grade : grades) {
    sum += grade;
    worst = max(worst, grade);
  }
  int avg = (sum + n - 1) / n;
  if (avg < 5) {
    int diff = 5 - avg;
    if (worst < 5) {
      cout << diff << endl;
    } else {
      cout << diff - 1 << endl;
    }
  } else {
    cout << 0 << endl;
  }
  return 0;
}
