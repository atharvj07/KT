#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  for (int i = 0; i < 8; i++) {
    string s;
    cin >> s;
    if (i % 2 == 0) {
      if (s[0] == 'B') {
        cout << "NO";
        return 0;
      }
      for (int j = 1; j < 8; j++) {
        if (s[j] != s[j - 1]) {
          cout << "NO";
          return 0;
        }
      }
    } else {
      if (s[0] == 'W') {
        cout << "NO";
        return 0;
      }
      for (int j = 1; j < 8; j++) {
        if (s[j] != s[j - 1]) {
          cout << "NO";
          return 0;
        }
      }
    }
  }
  cout << "YES";
  return 0;
}
