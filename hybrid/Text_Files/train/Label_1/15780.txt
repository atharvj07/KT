#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;
int main() {
  int n;
  string t;
  cin >> n >> t;
  unordered_map<char, int> cnt;
  for (char c : t) { cnt[c]++; }
  string s;
  for (auto &p : cnt) {
    for (int i = 1; i <= n && i <= p.second; i++) {
      s.push_back(p.first);
    }
  }
  cout << s << endl;
}
