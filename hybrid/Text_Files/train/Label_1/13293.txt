#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const ll INF = 1e18;
struct Tshirt {
    ll price, color;
};
bool operator<(const Tshirt &a, const Tshirt &b) {
    return a.price < b.price;
}
int main() {
    int n, m;
    scanf("%d", &n);
    vector<ll> p(n);
    for(int i = 0;i < n;++i) scanf("%lld", &p[i]);
    vector<int> a(n), b(n);
    for(int i = 0;i < n;++i) scanf("%d", &a[i]);
    for(int i = 0;i < n;++i) scanf("%d", &b[i]);
    scanf("%d", &m);
    vector<int> c(m);
    for(int i = 0;i < m;++i) scanf("%d", &c[i]);
    vector<vector<Tshirt>> colorList(4);
    for(int i = 0;i < n;++i) {
        colorList[a[i]].push_back({p[i], b[i]});
        colorList[b[i]].push_back({p[i], a[i]});
    }
    for(auto &v : colorList) {
        sort(v.begin(), v.end());
    }
    for(auto &x : c) {
        ll best = INF;
        int ok = 0;
        for(int i = 0;i < colorList[x].size();++i) {
            if(colorList[x][i].price >= best) continue;
            ok = 1;
            best = colorList[x][i].price;
        }
        if(ok) printf("%lld ", best);
        else printf("-1 ");
    }
    printf("\n");
    return 0;
}
