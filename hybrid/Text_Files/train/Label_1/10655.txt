#include <bits/stdc++.h>
using namespace std;
 
int main() {
    int q;
    cin >> q;
    while(q--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for(int i = 0; i < n; i++) {
            cin >> a[i];
            b[a[i] - 1] = i;
        }
        for(int i = 0; i < n - 1; i++) {
            if(b[i] > b[i + 1]) {
                swap(b[i], b[i + 1]);
                swap(a[b[i]], a[b[i + 1]]);
            }
        }
        for(int i = 0; i < n; i++)
            cout << a[i] << ' ';
        cout << '\n';
    }
    return 0;
}
