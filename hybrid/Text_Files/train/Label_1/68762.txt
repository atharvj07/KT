import sys
input = sys.stdin.readline

n,a = map(int,input().split())
d,c = [],[]
for _ in range(n):
    di,ci = map(int,input().split())
    d.append(di)
    c.append(ci)

dp = [0] * n
dp[0] = a - c[0]
for i in range(1,n):
    dp[i] = dp[i-1] + a - c[i]
    if i > 0:
        dp[i] = max(dp[i], dp[i - 1] - (d[i] - d[i - 1]) ** 2)

print(max(dp))
