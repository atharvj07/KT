#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<string> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<vector<int>> dist(n, vector<int>(m, -1));
    queue<pair<int, int>> q;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (v[i][j] == 'S') q.push({i, j}), dist[i][j] = 0;
        }
    }
    while (!q.empty()) {
        int x = q.front().first, y = q.front().second;
        q.pop();
        if (v[x][y] == 'G') {
            cout << dist[x][y] << endl;
            return 0;
        }
        if (x > 0 && v[x - 1][y] == '*' && dist[x - 1][y] == -1)
            q.push({x - 1, y}), dist[x - 1][y] = dist[x][y] + 1;
        if (x < n - 1 && v[x + 1][y] == '*' && dist[x + 1][y] == -1)
            q.push({x + 1, y}), dist[x + 1][y] = dist[x][y] + 1;
        if (y + 1 < m && v[x][y + 1] == '*' && dist[x][y + 1] == -1)
            q.push({x, y + 1}), dist[x][y + 1] = dist[x][y] + 1;
    }
    cout << -1 << endl;
    return 0;
}
