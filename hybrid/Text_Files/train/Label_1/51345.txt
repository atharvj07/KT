import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int numSquares = scanner.nextInt();
            if (numSquares == 0) {
                break;
            }
            int[] left = new int[numSquares];
            int[] right = new int[numSquares];
            int[] top = new int[numSquares];
            int[] bottom = new int[numSquares];
            for (int i = 1; i < numSquares; i++) {
                int num = scanner.nextInt();
                int direction = scanner.nextInt();
                if (direction == 0) {
                    right[num] = i;
                    left[i] = num;
                } else if (direction == 1) {
                    bottom[num] = i;
                    top[i] = num;
                } else if (direction == 2) {
                    left[num] = i;
                    right[i] = num;
                } else if (direction == 3) {
                    top[num] = i;
                    bottom[i] = num;
                }
            }
            int minLeft = 0;
            int maxRight = 0;
            int minTop = 0;
            int maxBottom = 0;
            for (int i = 0; i < numSquares; i++) {
                if (left[i] == 0) {
                    minLeft = i;
                }
                if (right[i] != 0 && right[i] > maxRight) {
                    maxRight = right[i];
                }
                if (top[i] == 0) {
                    minTop = i;
                }
                if (bottom[i] != 0 && bottom[i] > maxBottom) {
                    maxBottom = bottom[i];
                }
            }
            System.out.println((maxRight - minLeft + 1) + " " + (maxBottom - minTop + 1));
        }
    }
}
