import java.io.*;
import java.util.*;

class Main {
    static int n, m;
    static boolean[][] C;
    static int[] par;
    static boolean[] vis;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());
        for (int t = 0; t < T; t++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            n = Integer.parseInt(st.nextToken());
            m = Integer.parseInt(st.nextToken());
            C = new boolean[m + 1][n + 1];
            par = new int[n + 1];
            vis = new boolean[n + 1];
            for (int i = 1; i <= m; i++) {
                String s = br.readLine();
                for (int j = 0; j < n; j++) {
                    C[i][j + 1] = (s.charAt(j) == '1');
                }
            }
            if (solve()) {
                System.out.println("YES");
                for (int i = 2; i <= n; i++) {
                    System.out.println(par[i] + " " + i);
                }
            } else {
                System.out.println("NO");
            }
        }
    }

    static boolean solve() {
        for (int i = 1; i <= n; i++) {
            if (!vis[i] && !dfs1(i, 0)) {
                return false;
            }
        }
        for (int i = 1; i <= n; i++) {
            vis[i] = false;
        }
        return dfs2(1, 0);
    }

    static boolean dfs1(int u, int p) {
        vis[u] = true;
        for (int i = 1; i <= m; i++) {
            if (C[i][u]) {
                for (int j = 1; j <= n; j++) {
                    if (C[i][j] && j != u && !vis[j]) {
                        if (!dfs1(j, u)) {
                            return false;
                        }
                    }
                }
            }
        }
        return true;
    }

    static boolean dfs2(int u, int p) {
        if (vis[u]) {
            return true;
        }
        vis[u] = true;
        boolean ok = true;
        for (int i = 1; i <= n; i++) {
            if (i != u && !vis[i]) {
                for (int j = 1; j <= m; j++) {
                    if (C[j][u] && C[j][i]) {
                        if (!dfs2(i, u)) {
                            ok = false;
                        }
                        par[i] = u;
                        break;
                    }
                }
                if (!ok) {
                    return false;
                }
            }
        }
        return ok;
    }
}
