#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<vector<int>> a(n, vector<int>(m));
  vector<vector<int>> b(n, vector<int>(m));

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> b[i][j];
    }
  }

  bool possible = true;

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] != b[i][j]) {
        if (i + 1 < n && j + 1 < m) {
          a[i][j] = 1 - a[i][j];
          a[i + 1][j] = 1 - a[i + 1][j];
          a[i][j + 1] = 1 - a[i][j + 1];
          a[i + 1][j + 1] = 1 - a[i + 1][j + 1];
        } else {
          possible = false;
          break;
        }
      }
    }
  }

  cout << (possible ? "Yes" : "No") << endl;

  return 0;
}
