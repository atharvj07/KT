#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;

int n, k;
int freq[MAXN];

int main() {
    cin >> n >> k;
    string s;
    cin >> s;
    for (int i = 0; i < n; i++) {
        int idx = s[i] - 'A';
        freq[idx]++;
    }
    int mx = 0;
    for (int i = 0; i < k; i++) {
        mx = max(mx, freq[i]);
    }
    cout << mx << endl;
}
