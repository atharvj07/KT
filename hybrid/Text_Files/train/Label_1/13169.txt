#include <iostream>
#include <algorithm>

int main() {
    int n, w;
    std::cin >> n >> w;
    int a[n];
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::sort(a, a + n);
    if (w <= a[0]) {
        std::cout << -1 << std::endl;
        return 0;
    }
    for (int i = 0; i < n; i++) {
        int extra = w - (n - i) * (a[i] / 2);
        if (extra >= 0) {
            w = extra;
        } else {
            std::cout << -1 << std::endl;
            return 0;
        }
    }
    for (int i = 0; i < n; i++) {
        std::cout << (a[i] / 2) + (w > 0 && i == n - 1 ? w : 0) << " ";
        w--;
    }
    std::cout << std::endl;
    return 0;
}
