const ll mod = 1e9 + 7;
ll powmod(ll a, ll b) {
  if (b == 0) return 1LL;
  if (b == 1LL) return a % mod;
  ll ans = powmod(a, b / 2LL) % mod;
  ans = (ans * ans) % mod;
  if (b % 2LL) ans = (ans * a) % mod;
  return ans;
}
int main() {
  cin.tie(0)->sync_with_stdio(0);

  vector<ll> vec(b);
  for (int i = 0; i < b; ++i) {
    cin >> vec[i];

  }
  if (a <= 2LL) {
    cout << 1 << "\n";
    return 0;
  }
  if (2LL * vec[b - 1LL] > n) {
    a++;
  }
  a /= 2LL;
  cout << (a * (a - 1LL)) % mod << "\n";
  return 0;
}
