from collections import defaultdict
from heapq import *
def Mex(lst):
    Mex = 0
    msk = defaultdict(int)
    for x in lst:
        if not msk[x]:
            msk[x]+=1
            if Mex == x: Mex+=1
    return Mex

n,m = map(int,input().split())
msk = defaultdict(int)
for i in range(m):
    l,r = map(int,input().split())
    msk[l]+=1
    msk[r+1]-=1
mskT = []
for i in range(1,n+1):
    msk[i]+=msk[i-1]
    mskT.append(msk[i])
heappush(mskT,2e9)
ans = []
for i in range(n):
    if mskT[0]<=i: heappop(mskT)
    ans.append(mskT[0]-i)
print(Mex(ans))
print(*ans)
