n=int(input())
a=list(map(int,input().split()))
for i in range(3):
    cur=[]
    now=a[0]
    l=1
    for j in range(n):
        if a[j]==now:
            cur+=[j-l+1]
        else:
            print(l,j,sep=' ')
            print(*cur)
            now=a[j]
            l=j+1
            cur=[j-l+1]
    print(l,n,sep=' ')
    print(*cur)
