#include <algorithm>
#include <cmath>
#include <iostream>
#include <vector>

struct tree {
    int x;
    int a;
};

int main() {
    int n;
    std::cin >> n;
    std::vector<tree> trees(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> trees[i].x >> trees[i].a;
    }
    std::sort(trees.begin(), trees.end(),
              [](const tree& a, const tree& b) { return a.x < b.x; });
    int ans = 0;
    int cur = 0;
    int last = 0;
    for (int i = 0; i < n; ++i) {
        if (trees[i].x < 0) {
            if (cur < -trees[i].x) {
                ans += last;
                cur = -trees[i].x + trees[i].a;
                last = 0;
            } else {
                last += trees[i].a;
            }
        } else {
            if (cur < trees[i].x) {
                ans += last;
                cur = trees[i].x + trees[i].a;
                last = 0;
            } else {
                last += trees[i].a;
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
