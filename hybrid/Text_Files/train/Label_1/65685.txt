import sys
inp = sys.stdin.readlines()
n, m = map(int, inp[0].split())
x, k, y = map(int, inp[1].split())
a = list(map(int, inp[2].split()))
b = list(map(int, inp[3].split()))
a.sort()
b.sort()
if len({a[i]: i for i in range(n)}) != n or len({b[i]: i for i in range(m)}) != m:
    print(-1)
else:
    j = 0
    cost = 0
    for i in range(n):
        if j < m and a[i] == b[j]:
            j += 1
        else:
            k = i
            while k < n and a[k] != b[j]:
                k += 1
            if k == n:
                print(-1)
                exit()
            if (k - i + 1) % 2 == 0:
                cost += (k - i + 1) // 2 * y
            else:
                cost += (k - i) // 2 * y + x
            i = k
    print(cost)
