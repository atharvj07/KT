n = int(input())
a = list(map(int, input().split()))

a.sort()

# Case 1: Remove smallest element, compare second smallest and largest
instability_remove_smallest = a[-1] - a[1]

# Case 2: Remove largest element, compare smallest and second largest
instability_remove_largest = a[-2] - a[0]

print(min(instability_remove_smallest, instability_remove_largest))
