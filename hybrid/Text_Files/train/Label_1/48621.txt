import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int ans = 0, cnt = 0;
        int L = 0, R = 0;
        int n = s.length();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                ++L;
                a[i] = 1;
            } else {
                R = i;
                if (L > 0) {
                    --L;
                    a[i] = 1;
                    ans = Math.max(ans, i - R + 1);
                    cnt++;
                } else {
                    break;
                }
            }
        }
        if (cnt == 0) {
            System.out.println("0 1");
        } else {
            System.out.println(ans + " " + cnt);
        }
        sc.close();
    }
}
