def find_permutation(N, A, B):
    if A == 1 and B == 1:
        if N == 1:
            return [1]
        else:
            return -1
    if A == 1 or B == 1:
        cycle_size = max(A, B)
        if N % cycle_size != 0:
            return -1
        permutation = [0] * N
        for i in range(N):
            permutation[i] = (i + cycle_size) % N + 1
        return permutation
    if A != B and abs(A - B) != 1:
        return -1
    cycle_size = A + B - 2
    if N % cycle_size not in {0, A, B}:
        return -1
    permutation = [0] * N
    for i in range(0, N, cycle_size):
        for j in range(cycle_size):
            if j < A - 1:
                permutation[i + j] = i + j + 2
            elif j == A - 1:
                permutation[i + j] = i + 1
            else:
                permutation[i + j] = i + j + 1
        if N - i == A or N - i == B:
            break
    return permutation

N, A, B = map(int, input().split())
permutation = find_permutation(N, A, B)
if permutation == -1:
    print(-1)
else:
    print(" ".join(map(str, permutation)))
