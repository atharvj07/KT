#include <bits/stdc++.h>
#define int long long
using namespace std;

const int MAXN = 2e5 + 5, mod = 998244353;

int n, m;
int b[MAXN], freq[MAXN];
vector<int> adj[MAXN], rev[MAXN];
vector<array<int, 2>> ans;

void dfs(int node, int p) {
  for (auto u : adj[node]) {
    if (u != p) {
      dfs(u, node);
      freq[node] += freq[u];
    }
  }
}

void dfs2(int node, int p, int rev_freq) {
  freq[node] += rev_freq;
  ans.push_back({freq[node], node});
  for (auto u : rev[node]) {
    if (u != p) {
      dfs2(u, node, rev_freq + 1);
    }
  }
}

int32_t main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    cin >> b[i];
    adj[i].push_back(i << 1), adj[i].push_back(i << 1 | 1);
    rev[i << 1].push_back(i), rev[i << 1 | 1].push_back(i);
  }
  dfs(1, 1);
  dfs2(1, 1, 0);
  sort(ans.begin(), ans.end());
  int cnt = 1, res = 1;
  for (int i = 1; i < ans.size(); i++) {
    if (ans[i][0] == ans[i - 1][0]) {
      cnt++;
    } else {
      res = (res * cnt) % mod;
      cnt = 1;
    }
  }
  res = (res * cnt) % mod;
  cout << res << '\n';
  return 0;
}
