n, v = map(int, input().split())
trees = {}
for _ in range(n):
    a, b = map(int, input().split())
    if a in trees:
        trees[a] += b
    else:
        trees[a] = b

max_day = max(trees.keys()) + 2
collected_fruits = 0
remaining_fruits = 0
for day in range(1, max_day):
    current_day_fruits = trees[day] if day in trees else 0
    to_collect = min(v, remaining_fruits + current_day_fruits)
    collected_fruits += to_collect
    if remaining_fruits < to_collect:
        remaining_fruits = current_day_fruits - (to_collect - remaining_fruits)
    else:
        remaining_fruits = current_day_fruits
    remaining_fruits = min(remaining_fruits, v)

print(collected_fruits)
