import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        BigInteger[] xs = new BigInteger[n];
        BigInteger[] ys = new BigInteger[n];
        BigInteger[] zs = new BigInteger[n];
        for (int i = 0; i < n; i++) {
            xs[i] = sc.nextBigInteger();
            ys[i] = sc.nextBigInteger();
            zs[i] = sc.nextBigInteger();
        }
        for (int i = 0; i < n; i++) {
            BigInteger xmax = sc.nextBigInteger();
            BigInteger ymax = sc.nextBigInteger();
            BigInteger zmax = sc.nextBigInteger();
            xs[i] = xmax.subtract(xs[i]);
            ys[i] = ymax.subtract(ys[i]);
            zs[i] = zmax.subtract(zs[i]);
        }
        int[][] l = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                l[i][j] = -1;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    l[i][j] = 0;
                } else if (l[i][j] == -1) {
                    l[i][j] = intersect(xs[i], ys[i], zs[i], xs[j], ys[j], zs[j]);
                }
            }
        }
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = 0;
            for (int j = 0; j < n; j++) {
                d[i] += l[i][j];
            }
        }
        BigInteger ans = BigInteger.ZERO;
        for (int i = 0; i < n; i++) {
            if (d[i] >= k) {
                ans = ans.add(xs[i].multiply(ys[i]).multiply(zs[i]));
            }
        }
        System.out.println(ans);
    }

    public static int intersect(BigInteger x1, BigInteger y1, BigInteger z1, BigInteger x2, BigInteger y2, BigInteger z2) {
        BigInteger x_min = x1.max(x2);
        BigInteger y_min = y1.max(y2);
        BigInteger z_min = z1.max(z2);
        BigInteger x_max = x1.min(x2);
        BigInteger y_max = y1.min(y2);
        BigInteger z_max = z1.min(z2);
        if (x_max.compareTo(x_min) >= 0 && y_max.compareTo(y_min) >= 0 && z_max.compareTo(z_min) >= 0) {
            return 1;
        }
        return 0;
    }
}
