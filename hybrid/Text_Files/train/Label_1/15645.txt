#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<ll,ll>
int main() {
    ll m, k;
    cin >> m >> k;
    map<ll, set<ll>> adj;
    for(int i = 0; i < m; i++) {
        ll u, v;
        cin >> u >> v;
        adj[u].insert(v);
        adj[v].insert(u);
    }
    for(auto it: adj) {
        ll u = it.first;
        cout << u << ": ";
        set<ll> st;
        for(auto ad: adj[u]) {
            ll cnt = 0;
            for(auto it2: adj[ad]) {
                if(it2 != u) cnt++;
            }
            double pp = (cnt*1.0/adj[u].size())*100.0;
            if(pp >= k) st.insert(ad);
        }
        cout << st.size() << " ";
        for(auto x: st) cout << x << " ";
        cout << "\n";
    }
}
