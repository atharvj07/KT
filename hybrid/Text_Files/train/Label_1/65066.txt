import sys

n = int(input())
a = list(map(int, input().split()))

ans = 0
mn = 1e9
for i in range(n):
    mn = min(mn, abs(a[i]))
    ans += a[i]

a.sort(key = lambda x: abs(x))
mn = 1e9
k = 0
b = []
for i in range(n):
    if abs(a[i]) == mn:
        k += 1
        b.append(i+1)
    else:
        break

print(ans-k*mn, k)
for x in b:
    print(x, end = ' ')
