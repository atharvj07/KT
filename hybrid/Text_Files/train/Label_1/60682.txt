import sys

n, k, q = map(int, sys.stdin.readline().split())
recipes = []
for i in range(n):
    l, r = map(int, sys.stdin.readline().split())
    recipes.append((l, r))

for i in range(q):
    a, b = map(int, sys.stdin.readline().split())
    cnt = 0
    for l, r in recipes:
        if l <= a and b <= r:
            cnt += 1
    if cnt >= k:
        print(b - a + 1)
    else:
        print(0)
