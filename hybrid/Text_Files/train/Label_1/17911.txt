#include <bits/stdc++.h>
using namespace std;
int main() {
  int format;
  cin >> format;
  string t;
  cin >> t;
  int h = stoi(t.substr(0, 2)), m = stoi(t.substr(3, 2));
  if (format == 24) cout << setfill('0') << setw(2) << h << ':' << setw(2) << m;
  else {
    if (h == 12) h = 0;
    if (h >= 13) h -= 12;
    if (m >= 60) m -= 60;
    cout << setfill('0') << setw(2) << h << ':' << setw(2) << m;
  }
  return 0;
}
