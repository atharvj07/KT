import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] s = new char[n];
        
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) s[i] = (char) ('a' + i / 2);
            else s[i] = (char) ('z' - i / 2);
        }
        
        System.out.println(new String(s));
    }
}
