#include <bits/stdc++.h>
typedef long long int ll;
const int mod = 1e9+7;
using namespace std;
struct edge { int v,w; };
int n, p[100005], c[100005], d[100005];
vector<edge> G[100005];

ll powmod(ll a, ll pow) {
  if (pow == 0) return 1;
  ll ret = powmod(a, pow/2);
  if (pow % 2) return (ret * ret % mod) * a % mod;
  return ret * ret % mod;
}

void dfs(int u, int p) {
  for (int i = 0; i < G[u].size(); i++) {
    int v = G[u][i].v;
    int w = G[u][i].w;
    if (v == p) continue;
    d[v] = d[u] + w;
    dfs(v,u);
  }
}

int main() {
  scanf("%d", &n);
  for (int i = 1; i < n; i++) {
    scanf("%d", &p[i]);
    G[i].push_back(edge{p[i], 1});
    G[p[i]].push_back(edge{i, 1});
  }
  for (int i = 0; i < n; i++) scanf("%d", &c[i]);
  dfs(0,0);
  
  ll ans = 0;
  for (int i = 0; i < n; i++) for (int u = i+1; u < n; u++) {
    for (int j = 0; j < G[i].size(); j++) {
      int v = G[i][j].v;
      if (v == p[i]) continue;
      for (int k = 0; k < G[u].size(); k++) {
        int w = G[u][k].v;
        if (w == p[u]) continue;
        if (d[v] == d[w]) {
          if (c[i] + c[u] + c[v] == 1) ans++;
          if (c[i] + c[u] + c[w] == 1) ans++;
        }
      }
    }
  }
  cout << ans % mod << endl;
  return 0;
}
