#include <iostream>
using namespace std;

const int lim = 1000001;
unsigned int v[lim];

void prv(unsigned int &x, int cnt, int *p, bool &skok) {
  if (skok == 1)
    return;
  x += cnt;
  if (x > lim - cnt)
    skok = 1;
  v[x] += 1;
  *p -= 1;
  if (*p == 0) {
    prv(x, cnt, p, skok);
  }
}

int main() {
  int l, p, cnt, n;
  unsigned int x;
  bool skok = 0;
  x = 0;
  cin >> l;
  while (l-- != 0) {
    string w;
    cin >> w;
    if (w == "for") {
      cin >> n;
      p = n;
    } else if (w == "add") {
      cnt = 1;
      prv(x, cnt, &p, skok);
    } else if (w == "end") {
      continue;
    }
  }
  if (skok == 1)
    cout << "OVERFLOW!!!" << endl;
  else
    cout << x << endl;
  return 0;
}
