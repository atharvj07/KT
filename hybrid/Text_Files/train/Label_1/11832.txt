#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxN = 1e5 + 5;
vector<int> g[maxN];
int dist[maxN][2], sub[maxN];
int dfs1(int u, int p) {
  sub[u] = 1;
  for (int v : g[u])
    if (v != p) {
      dist[v][0] = dist[u][0] + 1;
      sub[u] += dfs1(v, u);
    }
  return sub[u];
}
void dfs2(int u, int p) {
  for (int v : g[u]) {
    if (v != p) {
      dist[v][1] = max(dist[u][1], sub[u] - sub[v] + dist[u][0] - dist[v][0]);
      dfs2(v, u);
    }
  }
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a, b, da, db;
    cin >> n >> a >> b >> da >> db;
    for (int i = 1; i <= n; i++) g[i].clear();
    for (int i = 0; i < n - 1; i++) {
      int u, v;
      cin >> u >> v;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    dfs1(a, a);
    dfs2(a, a);
    bool ok = false;
    for (int i = 1; i <= n; i++)
      if (dist[i][1] <= db) ok = true;
    if (ok)
      cout << "Bob\n";
    else
      cout << "Alice\n";
  }
  return 0;
}
