def sum_of_digits(n):
    return sum([int(digit) for digit in str(n)])

def find_sequence(n, b):
    a = [0] * n
    a[0] = b[0]
    for i in range(1, n):
        candidate = a[i - 1] + 1
        while sum_of_digits(candidate) != b[i]:
            candidate += 1
        a[i] = candidate
    return a

n = int(input())
b = [int(input()) for _ in range(n)]
a = find_sequence(n, b)
for x in a:
    print(x)
