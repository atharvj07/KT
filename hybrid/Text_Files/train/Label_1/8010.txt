#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 1;
    int cur = a[0];
    for (int i = 1; i < n; i++) {
        if (a[i] % cur != 0) {
            ans++;
            cur = a[i];
        }
    }
    cout << ans << endl;
    return 0;
}
