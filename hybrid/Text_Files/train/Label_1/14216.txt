#include <bits/stdc++.h>
#define INF 0x3f3f3f3f
using namespace std;

int n, m, k, t, a[200005], l[200005], r[200005], d[200005];
bool alive[200005], vis[200005];
vector<vector<int> > g;

bool cmp(int x, int y){
    return x > y;
}

int main(){
    cin >> m >> n >> k >> t;
    for(int i = 1; i <= m; i++){
        cin >> a[i];
        alive[i] = true;
    }
    for(int i = 1; i <= k; i++){
        cin >> l[i] >> r[i] >> d[i];
    }
    sort(a + 1, a + 1 + m, cmp);
    for(int i = 1; i <= n + 2; i++){
        g.push_back(vector<int>());
    }
    for(int i = 1; i <= k; i++){
        if(r[i] <= n){
            g[l[i]].push_back(r[i]);
        }
    }
    int ans = 0;
    for(int i = 1; i <= m; i++){
        memset(vis, false, sizeof vis);
        queue<int> q;
        q.push(0);
        vis[0] = true;
        while(!q.empty()){
            int cur = q.front(); q.pop();
            if((cur == n + 1) && vis[n + 1]){
                ans++;
                break;
            }
            for(int j = 0; j < g[cur].size(); j++){
                int v = g[cur][j];
                if(a[i] >= d[v] && !vis[v]){
                    q.push(v);
                    vis[v] = true;
                }
            }
            if(!vis[cur + 1] && cur + 1 <= n + 1){
                q.push(cur + 1);
                vis[cur + 1] = true;
            }
            if(!vis[cur - 1] && cur - 1 >= 0){
                q.push(cur - 1);
                vis[cur - 1] = true;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
