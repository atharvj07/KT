import java.util.Scanner;

public class VasyaTheSleuth {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String question = scanner.nextLine().trim();

        // Check if the last character is a vowel or consonant
        char lastChar = question.charAt(question.length() - 1);
        boolean isVowel = "aeiouyAEIOUY".indexOf(lastChar) != -1;

        // Print the answer accordingly
        System.out.println(isVowel ? "YES" : "NO");
    }
}
