#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int MAXN = 1e9 + 1;

int main() {
    ll a, b;
    cin >> a >> b;
    ll gcd = __gcd(a, b);
    ll ans = 0;
    for (int period = 1; period * period <= gcd; period++) {
        if (gcd % period == 0) {
            ans += 2;
            if (gcd / period != period) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
