n = int(input())
a = [int(x) for x in input().split()]
a.append(1145141919810857)
inc = [0] * n
dec = [0] * n
for i in range(1, n):
    if a[i] > a[i - 1]:
        inc[i] = inc[i - 1] + 1
for i in range(n - 2, -1, -1):
    if a[i] < a[i + 1]:
        dec[i] = dec[i + 1] + 1
ans = 0
for i in range(n):
    if i == 0:
        ans = max(ans, inc[i] + 1)
    elif i == n - 1:
        ans = max(ans, dec[i] + 1)
    else:
        ans = max(ans, inc[i], dec[i] + 1, inc[i - 1] + dec[i + 1] + 1)
print(ans)
