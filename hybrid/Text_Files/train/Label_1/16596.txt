#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long n, m;
    cin >> n >> m;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) { cin >> a[i]; }
    long long ans = (m - 3) * (m - 2) * (m - 1) / 6;
    for (long long x = 1; x < m; x++) {
        long long y = x + 1, cnt = 0;
        for (long long i = 0; i < n; i++) {
            while (y <= m && a[i] > (y > 2 ? (y - 2) : (y - 1)) * x) { y++; }
            if (a[i] == x * (y - 1)) { cnt++; }
        }
        ans += cnt * (m - x) * (m - x - 1) / 2;
    }
    cout << ans << endl;
}
