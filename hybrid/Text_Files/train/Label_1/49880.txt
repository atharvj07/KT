import java.util.Scanner;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt(),
        p = 998244353;
    long r = 1;
    for (long i = n - a, j = n + 1 - b; i > 0 || j > 0; i--, j--) {
      r = (r * i) % p;
      r = (r * j) % p;
    }
    System.out.println(r);
  }
}
