#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n;
  cin >> n;
  set<int> s;
  while (n--) {
    int x;
    cin >> x;
    if (!s.insert(x).second) {
      cout << "NO" << endl;
      return 0;
    }
  } cout << "YES" << endl;
}
