#include<iostream>
#include<string>
using namespace std;
int main(){
    string a;
    cin>>a;
    if(a.find('8')!=-1){
        cout<<0<<endl;
        return 0;
    }
    int ans=10;
    for(int i=0;i<a.length();i++){
        if(a[i]=='1')continue;
        int tmp=a[i]-'0';
        for(int j=1;j<=9;j++){
            int t=tmp+j;
            if(t>9)t-=10;
            if(t==8){
                ans=min(ans,j);
                break;
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
