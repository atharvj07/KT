#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mxN = 2e5 + 1;
int main() {
	int n; cin >> n;
	vector<ll> a(n), dif(n, 0);
	for(int i = 0; i < n; ++i) cin >> a[i];
	for(int i = 1; i < n; ++i) dif[i] = a[i] - a[i - 1];
	int ex = 0; ll sum = 0;
	for(int i = 2; i < n; ++i) {
		if(dif[i] != dif[i - 1]) {
			if(ex) { ex = -1; break; }
			ex = i;
			sum = dif[i] - dif[i - 1];
		}
	}
	if(ex == 2 && n == 3) ex = -1;
	if(ex != -1 && sum == 0) ex = -1;
	if(ex != -1) ex = a[ex - 1];
	for(int i = 1; i < n; ++i) {
		if(dif[i] == 0) {
			if(i == ex) continue;
			ex = -1; break;
		}
	}
	cout << ex << endl;
}
