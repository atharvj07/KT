for _ in range(int(input())):
    n = int(input())
    s = input()
    a, b = sum(s[i] == '0' for i in range(n)), 0
    if s[::-1] == s:
        if a % 2:
            a, b = 0, 1
        else:
            b += 1
    if a <= b:
        print("ALICE")
    elif b == 0:  # special case when all characters are 1
        print("DRAW")
    else:
        print("BOB")
