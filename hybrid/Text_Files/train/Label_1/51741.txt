import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();

        int[] l = new int[N + 1];
        int[] r = new int[N + 1];

        for (int i = 1; i <= N; i++) {
            l[i] = in.nextInt();
            r[i] = in.nextInt();
        }

        if (!check(l, r, N)) {
            System.out.println("NO");
            int[] a = restore(l, r, N);
            int[] b = restore(r, l, N);
            for (int i = 1; i <= N; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
            for (int i = 1; i <= N; i++) {
                System.out.print(b[i] + " ");
            }
        } else {
            System.out.println("YES");
            int[] a = restore(l, r, N);
            for (int i = 1; i <= N; i++) {
                System.out.print(a[i] + " ");
            }
        }
    }

    static boolean check(int[] l, int[] r, int N) {
        for (int i = 1; i <= N; i++) {
            if (l[i] > r[i])
                return false;
        }

        int[] a = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            if (a[l[i]] == 0) {
                a[l[i]] = i;
            } else if (a[r[i]] == 0) {
                a[r[i]] = i;
            } else {
                return false;
            }
        }

        return true;
    }

    static int[] restore(int[] l, int[] r, int N) {
        int[] a = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            if (a[l[i]] == 0) {
                a[l[i]] = i;
            } else {
                a[r[i]] = i;
            }
        }

        return a;
    }
}
