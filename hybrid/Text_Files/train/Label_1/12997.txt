#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
#define mx 51
vector<int>v[mx], vf[mx], vb[mx];
int dp[mx][mx][2][2];
int f(int st, int cnt, int clr, int pr) {
    if(cnt == 0) return pr == 0;
    int &ret = dp[st][cnt][clr][pr];
    if(~ret) return ret;
    ret = 0;
    for(int i = st + 1; i < v[st].size(); i++) {
        int nx = v[st][i];
        if(clr) {
            if(vf[st][i] == 1) ret = (ret + f(nx, cnt - 1, 0, pr ^ 1)) % mod;
            if(vf[st][i] == 0) ret = (ret + f(nx, cnt - 1, 0, pr)) % mod;
        }
        else {
            if(vb[st][i] == 1) ret = (ret + f(nx, cnt - 1, 1, pr)) % mod;
            if(vb[st][i] == 0) {
                ret = (ret + f(nx, cnt - 1, 0, pr)) % mod;
                ret = (ret + f(nx, cnt - 1, 1, pr)) % mod;
            }
        }
    }
    return ret;
}
int main() {
    memset(dp, -1, sizeof dp);
    int n, par;
    cin >> n >> par;
    for(int i = 1; i <= n; i++) {
        int clr;
        cin >> clr;
        if(clr == -1) clr = 2;
        for(int j = 1; j <= n; j++) {
            if(i == j) continue;
            v[i].push_back(j);
            vf[i].push_back(clr);
            vb[i].push_back(clr);
        }
    }
    cout << f(1, n - 1, 1, par) << "\n";
    return 0;
}
