import sys
input = sys.stdin.readline

n = int(input())
words = input().split()
m = len(max(words, key=len))

ans = n * m
for i in range(1, n):
    for j in range(i - 1, -1, -1):
        if ''.join(words[i - (k - j):i + 1]) == ''.join(words[j:i + 1]):
            k = j
            ans = min(ans, (i - j + 1) * m + n - i - 1)
            break
print(ans)
