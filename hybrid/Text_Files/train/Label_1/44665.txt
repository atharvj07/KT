import java.util.Arrays;
import java.util.Scanner;

public class MaximumDistinctPrefixProducts {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] forbiddenProducts = new int[n];
        for (int i = 0; i < n; i++) {
            forbiddenProducts[i] = scanner.nextInt();
        }

        Arrays.sort(forbiddenProducts);

        int[] sequence = new int[m];
        int k = 0;
        int currentProduct = 1;

        for (int i = 0; i < m; i++) {
            if (Arrays.binarySearch(forbiddenProducts, currentProduct) < 0) {
                sequence[k++] = i;
                currentProduct = (currentProduct * i) % m;
            }
        }

        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(sequence[i] + " ");
        }
    }
}
