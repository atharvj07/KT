import java.util.Scanner;

public class Main {
    static long mod = 1_000_000_007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        long d = sc.nextLong();
        sc.close();

        long ans = (a - 1 - d) * (b - 1 - d) * (c - 1 - d) % mod;
        ans = (ans + (a - 1) * (b - 1) * (c + d) % mod + (a - 1) * (b - 1 + d) * (c) % mod) % mod;
        ans = (ans + (a - 1) * (b + d) * (c) % mod + (a - 1 + d) * (b) * (c) % mod) % mod;
        System.out.println(ans);
    }
}
