import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<List<Integer>> specialCases = new ArrayList<>();
        List<List<Integer>> parts = new ArrayList<>();

        for (int i = 0; i <= n; i++) {
            specialCases.add(new ArrayList<>());
            parts.add(new ArrayList<>());
        }

        for (int i = 1; i <= n; i++) {
            String[] inputs = br.readLine().split(" ");
            int parent = Integer.parseInt(inputs[0]);
            int type = Integer.parseInt(inputs[1]);

            if (type == -1) continue;

            if (type == 0) {
                specialCases.get(i).add(parent);
            } else {
                parts.get(i).add(parent);
            }
        }

        int q = Integer.parseInt(br.readLine());

        for (int i = 0; i < q; i++) {
            String[] inputs = br.readLine().split(" ");
            int type = Integer.parseInt(inputs[0]);
            int u = Integer.parseInt(inputs[1]);
            int v = Integer.parseInt(inputs[2]);

            if (isSpecialCase(u, v, specialCases)) System.out.println("YES");
            else if (isPart(u, v, parts)) System.out.println("YES");
            else System.out.println("NO");
        }
    }

    static boolean isSpecialCase(int u, int v, List<List<Integer>> specialCases) {
        if (u == v) return true;

        for (int parent : specialCases.get(v)) {
            if (isSpecialCase(u, parent, specialCases)) return true;
        }

        return false;
    }

    static boolean isPart(int u, int v, List<List<Integer>> parts) {
        if (u == v) return true;

        for (int parent : parts.get(v)) {
            if (isPart(u, parent, parts)) return true;
        }

        return false;
    }
}
