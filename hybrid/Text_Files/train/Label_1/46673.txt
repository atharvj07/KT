import java.util.Scanner;

public class Laundry {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] segments = new int[n][2];
        for (int i = 0; i < n; i++) {
            segments[i][0] = scanner.nextInt();
            segments[i][1] = scanner.nextInt();
        }
        int alexeyStart = segments[0][0];
        int alexeyEnd = segments[0][1];
        int totalLength = 0;
        for (int i = 0; i < n; i++) {
            int[] segment = segments[i];
            if (i == 0) {
                continue;
            }
            int start = Math.max(segment[0], alexeyStart);
            int end = Math.min(segment[1], alexeyEnd);
            if (start < end) {
                totalLength += end - start;
            }
        }
        System.out.println(alexeyEnd - alexeyStart - totalLength);
    }
}
