#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  int cnt_h = 0, cnt_v = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    for (int j = 0; j < n; j++) {
      if (i + 1 < n && a[i][j] == '#' && a[i + 1][j] == '#') cnt_v++;
      if (j + 1 < n && a[i][j] == '#' && a[i][j + 1] == '#') cnt_h++;
    }
  }
  cout << (cnt_v % 5 == 0 && cnt_v / 5 == cnt_h ? "YES" : "NO") << endl;
  return 0;
}
