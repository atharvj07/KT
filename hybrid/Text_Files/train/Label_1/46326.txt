import java.util.*;

public class Main {
    static int n;
    static int[] w;
    static List<Pair>[] adj;
    static long[] dp;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        adj = new List[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            int c = sc.nextInt();
            adj[u].add(new Pair(v, c));
            adj[v].add(new Pair(u, c));
        }
        dp = new long[n];
        vis = new boolean[n];
        dfs(0, -1);
        System.out.println(dp[0]);
    }

    static void dfs(int u, int p) {
        vis[u] = true;
        dp[u] = w[u];
        for (Pair v : adj[u]) {
            if (!vis[v.x]) {
                dfs(v.x, u);
                dp[u] = Math.max(dp[u], dp[v.x] + w[u] - v.y);
            }
        }
    }

    static class Pair {
        int x, y;

        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
