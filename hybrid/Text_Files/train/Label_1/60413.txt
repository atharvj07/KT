import sys
input = sys.stdin.readline

def dfs(u, p, lst):
    lst.append(u)
    for v in graph[u]:
        if v == p: continue
        dfs(v, u, lst)

n = int(input())
graph = {i: [] for i in range(1, n + 1)}
for _ in range(n - 1):
    u, v = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)

leaf = []
vis = [0] * (n + 1)
for i in range(1, n + 1):
    if len(graph[i]) == 1:
        leaf.append(i)
        vis[i] = 1

ans = []
for u in leaf:
    lst = []
    dfs(u, -1, lst)
    for v in lst: vis[v] = 1
    ans.append(lst)

ans.sort(key=lambda x: x[-1])
if n == 2: print(1, 1)
elif len(ans) > 1:
    print(ans[-1][-1], ans[0][-1])
else:
    for v in graph[1]:
        if not vis[v]:
            ans = dfs(v, 1, [])
            ans.sort(key=lambda x: x[-1])
            print(ans[-1][-1], ans[0][-1])
            exit()
