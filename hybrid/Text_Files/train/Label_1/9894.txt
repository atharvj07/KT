#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, T, mx = 0;
    cin >> n >> T;
    vector<int> t(n);
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    sort(t.begin(), t.end());
    for (int i = 0; i < n; i++) {
        int cnt = 1;
        for (int j = i + 1; j < n; j++) {
            if (t[j] > t[i] + T) break;
            cnt++;
        }
        mx = max(mx, cnt);
    }
    cout << mx << endl;
    return 0;
}
