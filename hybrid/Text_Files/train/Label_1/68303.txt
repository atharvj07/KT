n = int(input().strip())
v = list(map(int, input().strip().split()))
t = list(map(int, input().strip().split()))

melted = [0] * n
remaining_snow = []

for day in range(n):
    new_snow = v[day]
    total_melt = t[day]
    today_melt = 0

    remaining_snow.append(new_snow)
    for i in range(len(remaining_snow)):
        if remaining_snow[i] <= total_melt:
            today_melt += remaining_snow[i]
            remaining_snow[i] = 0
        else:
            today_melt += total_melt
            remaining_snow[i] -= total_melt

    remaining_snow = [snow for snow in remaining_snow if snow > 0]
    melted[day] = today_melt

print(' '.join(map(str, melted)))
