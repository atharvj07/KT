import java.util.ArrayDeque;
import java.util.Queue;
import java.util.Scanner;

public class Jaber {

    static class City {
        int row;
        int col;
        int color;

        public City(int row, int col, int color) {
            this.row = row;
            this.col = col;
            this.color = color;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;

            System.out.println(solve(grid, r1, c1, r2, c2));
        }
    }

    private static int solve(int[][] grid, int r1, int c1, int r2, int c2) {
        int n = grid.length;
        int m = grid[0].length;
        int k = 100; // Maximum possible number of colors

        Queue<City> queue = new ArrayDeque<>();
        queue.offer(new City(r1, c1, grid[r1][c1]));
        boolean[][][] visited = new boolean[n][m][k + 1];
        visited[r1][c1][grid[r1][c1]] = true;

        int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
        int time = 0;

        while (!queue.isEmpty()) {
            int size = queue.size();
            for (int i = 0; i < size; i++) {
                City current = queue.poll();

                if (current.row == r2 && current.col == c2) {
                    return time;
                }

                for (int[] direction : directions) {
                    int newRow = current.row + direction[0];
                    int newCol = current.col + direction[1];

                    if (newRow >= 0 && newRow < n && newCol >= 0 && newCol < m &&
                            !visited[newRow][newCol][current.color]) {
                        if (grid[newRow][newCol] == current.color) {
                            visited[newRow][newCol][current.color] = true;
                            queue.offer(new City(newRow, newCol, current.color));
                        } else {
                            visited[newRow][newCol][grid[newRow][newCol]] = true;
                            queue.offer(new City(newRow, newCol, grid[newRow][newCol]));
                        }
                    }
                }
            }

            time++;
        }

        return -1; // unreachable
    }
}
