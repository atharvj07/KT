import sys
n = int(sys.stdin.readline())
x0,y0 = map(int,sys.stdin.readline().split())
dic = []
for i in range(n):
    a,x,y = sys.stdin.readline().split()
    dic.append((a,int(x),int(y)))
for i in dic:
    if (i[0] == 'R' and (i[1] == x0 or i[2] == y0)) or (i[0] == 'B' and abs(i[1]-x0) == abs(i[2]-y0)) or (i[0] == 'Q' and (i[1] ==x0 or i[2] ==y0 or abs(i[1]-x0) == abs(i[2]-y0))):
        print('YES')
        exit()
print('NO')
