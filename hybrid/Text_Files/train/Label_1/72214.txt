import sys

n, m = tuple(map(int, sys.stdin.readline().split()))
set1 = set(map(int, sys.stdin.readline().split()))
set2 = set(map(int, sys.stdin.readline().split()))

if len(set1.intersection(set2)) == 1:
    print(next(iter(set1.intersection(set2))))
elif set1.issubset(set2) or set2.issubset(set1):
    print(0)
else:
    print(-1)
