#include<bits/stdc++.h>
using namespace std;
int vis[25], c[25];
int n, f;
void dfs(int u, int p, int sum) {
	if (vis[u] == 1) return;
	vis[u] = 1;
	if (sum + 1 != c[u]) f = 1;
	else {
		if (u != 1) {
			if (p < 2)f = 1; 
		}
		for (int i = 2; i <= n; i++) {
			if (u != i) {
				dfs(i, p + 1, sum + 1);
			}
		}
	}
}
int main() {
	int i;
	cin >> n;
	for (i = 1; i <= n; i++) { cin >> c[i]; }
	dfs(1, 0, 0);
	if (f == 0) cout << "YES";
	else cout << "NO";
	return 0;
}
