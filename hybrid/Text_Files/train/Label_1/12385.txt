#include <bits/stdc++.h>
using namespace std;

struct DSU {
  vector<int> fa;
  int n;
  DSU(int n) : n(n) { fa.resize(n + 1); iota(fa.begin(), fa.end(), 0); }
  int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
  void merge(int a, int b) { fa[find(b)] = find(a); }
  int query(int a, int b) { return find(a) == find(b); }
};

struct Edge {
  int u, v;
  Edge() {}
  Edge(int u, int v) : u(u), v(v) {}
};

const int N = 3e5 + 10;
vector<Edge> e;
vector<vector<int>> g;
vector<int> dep, ans;
int n, q;

void dfs(int u, int fa) {
  for (int i : g[u]) {
    int v = e[i].u + e[i].v - u;
    if (v == fa) continue;
    dep[v] = dep[u] + 1;
    dfs(v, u);
  }
}

void solve() {
  cin >> n;
  g.resize(n + 1);
  e.resize(n);
  dep.clear();
  dep.resize(n + 1);
  for (int i = 1; i < n; ++i) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(i);
    g[v].push_back(i);
    e[i] = Edge(u, v);
  }
  dfs(1, 0);
  int mx = 0, id = 0;
  for (int i = 1; i <= n; ++i) {
    if (dep[i] > mx) { mx = dep[i]; }
  }
  for (int i = 1; i <= n; ++i) {
    if (dep[i] == mx) id = i;
  }
  ans.push_back(mx);
  // cout << "id = " << id << ", dep = " << dep[id] << "\n";
  dfs(id, 0);
  mx = 0, id = 0;
  for (int i = 1; i <= n; ++i) {
    if (dep[i] > mx) { mx = dep[i]; }
  }
  for (int i = 1; i <= n; ++i) {
    if (dep[i] == mx) id = i;
  }
  ans.push_back(mx);
  cout << max(ans[0], ans[1]) << '\n';
}

int main() {
  cin >> q;
  while (q--) { solve(); }
  return 0;
}
