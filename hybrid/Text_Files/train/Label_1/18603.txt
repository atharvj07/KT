#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  while (m--) {
    int l, r;
    cin >> l >> r;
    int sum = 0;
    for (int i = l - 1; i < r; i++) { sum += a[i]; }
    cout << (sum ? "0" : "1") << endl;
  }
  return 0;
}
