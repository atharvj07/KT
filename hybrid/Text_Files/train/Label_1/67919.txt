import operator
n = int(input())
a = []
for i in range(n):
    a.append(list(map(int, input().split())))
a.sort(key=operator.itemgetter(1))
b = []
for i in a:
    if len(b) == 0 or b[-1][1] <= i[0]:
        b.append(i)
        continue
    if b[-1][1] < i[1]:
        b[-1] = i
if len(b) <= 2:
    print("YES")
else:
    print("NO")
