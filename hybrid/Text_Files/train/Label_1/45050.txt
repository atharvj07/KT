import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int cnt = 0;
        int last = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] != -1) {
                if (last == -1) {
                    cnt++;
                } else if (a[i] - a[last] != a[last] - a[i - 1]) {
                    cnt++;
                }
                last = i;
            }
        }
        System.out.println(cnt);
    }
}
