n, m = map(int, input().split())
g = [[0] * (m + 2) for _ in range(n + 2)]
for i in range(1, n + 1):
    g[i][1: m + 1] = list(input())
adj = lambda a, b, c, d: a == c and abs(b - d) == 1 or b == d and abs(a - c) == 1
for i in range(1, n + 1):
    for j in range(1, m + 1):
        for k in range(i - 1, i + 2):
            for l in range(j - 1, j + 2):
                if adj(i, j, k, l) and g[k][l] == g[i][j]:
                    if i == 1 and j == 1:
                        continue
                    if k == 1 and l == 1:
                        continue
                    if i == n and j == m:
                        continue
                    if k == n and l == m:
                        continue
                    if (k == i and l == j - 1) or (k == i and l == j + 1) or (k == i - 1 and l == j) or (k == i + 1 and l == j):
                        continue
                    print("Yes")
                    exit()
print("No")
