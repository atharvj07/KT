#include <bits/stdc++.h>
#define ll long long
#define mod (ll)(1e9 + 7)
using namespace std;

ll modpow(ll b, ll p) {
    if (!p) return 1;
    if (p & 1) return (modpow(b, p - 1) * b) % mod;
    return modpow((b * b) % mod, p >> 1);
}

int main() {
    ll inv2 = modpow(2, mod - 2);
    ll n;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    vector<vector<int>> g(n + 1);
    for (int i = 1; i <= n; i++) {
        if (a[i] > 0) { g[a[i]].push_back(i); }
    }
    ll ans = 0, cnt = n;
    for (int i = 1; i <= n; i++) {
        if (a[i] > 0) continue;
        for (int j : g[i]) {
            ans += (ll)(j - 1) * (cnt - 1);
            ans %= mod;
        }
        cnt--;
        cnt = (cnt * cnt) % mod;
    }
    ans = (ans * modpow(cnt, mod - 2)) % mod;
    ans = (ans * inv2) % mod;
    cout << ans << "\n";
    return 0;
}
