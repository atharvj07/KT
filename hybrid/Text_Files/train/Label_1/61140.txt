import datetime

n, m, t = map(int, input().split())
times = [datetime.datetime.strptime(input(), '%H:%M:%S') for _ in range(n)]

ans = [0] * n
u = 1

for i in range(n):
    if (times[i] - datetime.timedelta(seconds=t)).time() in times:
        u += 1
    else:
        for j in range(i):
            if times[j] >= times[i] - datetime.timedelta(seconds=t):
                break
            if ans[j] == ans[i]:
                ans[j] += 1
                break
    ans[i] = u

print(u)
for x in ans:
    print(x)
