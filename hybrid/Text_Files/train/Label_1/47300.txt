import java.util.Scanner;

public class RedundantTVSet {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] l = new int[n];
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = scanner.nextInt();
            r[i] = scanner.nextInt();
        }

        // Find the leftmost and rightmost working moments
        int left = l[0], right = r[0];
        for (int i = 1; i < n; i++) {
            left = Math.min(left, l[i]);
            right = Math.max(right, r[i]);
        }

        // Check if any TV set covers the entire working period
        for (int i = 0; i < n; i++) {
            if (l[i] <= left && right <= r[i]) {
                System.out.println(i + 1);
                return;
            }
        }

        // No redundant TV set found
        System.out.println(-1);
    }
}
