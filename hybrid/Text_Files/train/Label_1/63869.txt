n,m,k=[int(x) for x in input().split()]
ins=[list(map(int,input().split())) for i in range(n)]
lock=[0]*k
blocked=[]
for ci in range(m):
    for i in range(n):
        if not ins[i][ci] or ins[i][ci] in blocked: continue
        if lock[ins[i][ci]-1]>0:
            for t in blocked:
                if lock[t-1]>ci: lock[t-1]=ci
            lock[ins[i][ci]-1]=ci
            blocked.append(ins[i][ci])
print('\n'.join(map(str,[0 if lock[i]==0 else lock[i]+1 for i in range(n)])))
