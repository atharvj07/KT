import sys

n = int(input().strip())
s = input().strip()
x, y = map(int, input().strip().split(' '))
if x+y < 0 or x-y > n or (x+y)%2 != n%2:
    print(-1)
else:
    ops = {'U': 'D', 'D': 'U', 'L': 'R', 'R': 'L'}
    ans = []
    u, d = 0, 0
    l, r = 0, 0
    for c in s:
        if c == 'U':
            u += 1
            if u > x+y-d:
                ans.append(c)
                u -= 1
                d += 1
        elif c == 'D':
            d += 1
            if d > x-y-u:
                ans.append(c)
                d -= 1
                u += 1
        elif c == 'L':
            l += 1
            if l > (x+y-d)/2 - r:
                ans.append(c)
                l -= 1
                r += 1
        else:
            r += 1
            if r > (x-y-u)/2 - l:
                ans.append(c)
                r -= 1
                l += 1
    print(max(ans)-min(ans)+1 if ans else 0)
