#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<int>v(n);
    for(int i=0;i<n;i++)cin>>v[i];
    sort(v.begin(),v.end());
    int mn=2,mx=n;
    if(v[0]>2)mn=1;
    for(int i=0;i<n-1;i++){
        if(v[i+1]-v[i]>1){
            mn=1;
            break;
        }
    }
    if(v[n-1]<n-1)mx=n-1;
    cout<<mn<<" "<<mx<<endl;
}
