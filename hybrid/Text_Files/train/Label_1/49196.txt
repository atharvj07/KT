import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int k = 0;
        int[] newArr = new int[n];
        for (int i = 0; i < n; i++) {
            if (i + 1 < n && arr[i] == arr[i + 1]) {
                newArr[k++] = arr[i] + 1;
                i++;
            } else {
                newArr[k++] = arr[i];
            }
        }
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(newArr[i] + " ");
        }
        System.out.println();
    }
}
