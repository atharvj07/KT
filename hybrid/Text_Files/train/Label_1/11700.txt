#include <vector>
#include <cstdio>
using namespace std;
typedef long long LL;
const int N = 15;
const int Mod = 1e9 + 7;
int c[N], n, l, r, x;
int dp[(1 << N)][N];
int main() {
    scanf("%d%d%d%d", &n, &l, &r, &x);
    for (int i = 0; i < n; i++) scanf("%d", c + i);
    dp[(1 << n) - 1][0] = 1;
    for (int i = (1 << n) - 1; i; i--) {
        for (int j = 0; j <= n; j++) {
            if (x == 0) {
                dp[i & (i - 1)][j] += dp[i][j];
                dp[i & (i - 1)][j] %= Mod;
                continue;
            }
            int mn = c[0], mx = c[0];
            bool flag = true;
            for (int k = 0; k < n; k++) {
                if (!(i & (1 << k))) continue;
                mn = min(mn, c[k]);
                mx = max(mx, c[k]);
                if (j > 0 && c[k] - mn < x) { flag = false; }
                if (j < n - 1 && mx - c[k] < x) { flag = false; }
            }
            if (flag) {
                LL sum = 0;
                for (int k = 0; k < n; k++) {
                    if (!(i & (1 << k))) continue;
                    sum += c[k];
                }
                if (sum >= l && sum <= r) {
                    dp[i & (i - 1)][j + 1] += dp[i][j];
                    dp[i & (i - 1)][j + 1] %= Mod;
                }
            }
        }
    }
    printf("%d\n", dp[0][0]);
    return 0;
}
