#include<bits/stdc++.h>
using namespace std;
int main(){
  int t; cin>>t; while(t--){
    int n; cin>>n; vector<int> a(n),b(n); 
    for(int &x: a) cin>>x;
    for(int &x: b) cin>>x;
    map<int,int> mp;
    for(int &x: a) mp[x]++;
    for(int &x: b) mp[x]--;
    int ans=0;
    for(auto &x: mp) if(x.second!=0) ans=1;
    if(ans) cout<<"NO\n"; else cout<<"YES\n";
  }
}
