#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  cin >> m;
  vector<int> b(m);
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  for (int i = 0, j = 0; i < n; i++) {
    while (j < m && b[j] <= a[i]) j++;
    if (j < m && a[i] + b[j] > a[n - 1] && a[i] + b[j] > b[m - 1]) {
      cout << a[i] << ' ' << b[j] << '\n';
      return 0;
    }
  }
}
