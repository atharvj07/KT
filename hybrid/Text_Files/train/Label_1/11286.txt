#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, pair<int, int>>> v(n);
  for (int i = 0; i < n; i++) {
    int t, d, p;
    cin >> t >> d >> p;
    v[i] = {p, {d, t}};
  }
  sort(v.rbegin(), v.rend());
  int sum = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (v[i].second.first >= v[i].second.second) {
      sum += v[i].first;
      ans++;
    }
  }
  cout << sum << endl << ans << endl;
  for (int i = 0; i < n; i++) {
    if (v[i].second.first >= v[i].second.second) {
      cout << i + 1 << " ";
    }
  }
  cout << endl;
  return 0;
}
