#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    long long n;
    cin >> n;
    vector<int> divisors;
    for (long long i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) {
            divisors.push_back(i);
            if (n / i != i) divisors.push_back(n / i);
        }
    }
    cout << divisors.size() + 1 << endl;
    return 0;
}
