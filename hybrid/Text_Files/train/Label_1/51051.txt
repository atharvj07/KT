import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static final int MOD = 1000000007;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] dp = new int[n][n + 1];
        for (int[] i : dp) Arrays.fill(i, -1);
        System.out.println(solve(0, n, a, dp));
    }

    static int solve(int idx, int n, int[] a, int[][] dp) {
        if (idx == n) {
            return 1;
        }
        if (dp[idx][n] != -1) {
            return dp[idx][n];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (Math.abs(a[idx] - (idx - i)) == Math.abs(a[idx] - (n - i - 1))) {
                ans = (ans + solve(idx + 1, n, a, dp)) % MOD;
            }
        }
        dp[idx][n] = ans;
        return ans;
    }
}
