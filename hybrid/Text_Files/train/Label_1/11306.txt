#include <bits/stdc++.h>
const int mod = 1e9 + 9;
int mpow(int a, int b) {
    int ret = 1;
    while (b) {
        if (b & 1) ret = 1ll * ret * a % mod;
        a = 1ll * a * a % mod;
        b >>= 1;
    }
    return ret;
}
int32_t main() {
    int n, m, k;
    std::cin >> n >> m >> k;
    if (m <= k) std::cout << m << '\n';
    else {
        int ans = mpow(2, (m - k) / (k - 1) + 1);
        std::cout << 1ll * ans * mpow(2, m - 1) % mod << '\n';
    }
}
