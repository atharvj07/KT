#include <bits/stdc++.h>

using namespace std;

const int INF = 1e9;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> adj[n];
    for (int i = 1; i < n; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    int ans = 0;
    function<int(int, int)> dfs = [&](int u, int par) {
        int mx = -INF;
        for (auto &e : adj[u]) {
            if (e.first != par) {
                mx = max(mx, dfs(e.first, u) + e.second);
            }
        }
        if (mx == -INF) {
            return 0;
        }
        ans = max(ans, mx);
        return mx;
    };
    dfs(0, -1);
    cout << ans << endl;
    return 0;
}
