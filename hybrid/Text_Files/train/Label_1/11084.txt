#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  sort(v.begin(), v.end());

  long long maxDiff = v.back() - v.front();
  int ways = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (v[j] - v[i] == maxDiff) {
        ways++;
      }
    }
  }

  cout << maxDiff << " " << ways << endl;
  return 0;
}
