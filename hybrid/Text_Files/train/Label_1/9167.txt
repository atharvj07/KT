#include <bits/stdc++.h>
#define int long long
using namespace std;
int a[1001], dp[1001][1001], P = 1e9 + 7;
signed main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) dp[i][0] = 1;
    for (int j = 1; j <= k; j++) dp[0][j] = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= k; j++) {
        dp[i][j] = dp[i - 1][j];
        if (j >= a[i]) dp[i][j] += dp[i - 1][j - a[i]];
        dp[i][j] %= P;
      }
    }
    cout << dp[n][k] << '\n';
  }
}
