import sys

n, m = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
c = [0] * (m+1)
for x in a:
    c[x] += 1
add = 0
for i in range(1, m+1):
    if c[i] > 0:
        add += 1
ans = min(c[1:m+1])
if add == m:
    print(ans, 0)
    print(*a)
    sys.exit()
cur = 0
for i in range(1, m+1):
    if c[i] > 0:
        cur += 1
    if cur == m:
        break
print(ans, n-cur-c[0])
for x in a:
    if c[0] > 0 and x > m:
        print(i, end=" ")
        c[0] -= 1
    else:
        print(x, end=" ")
