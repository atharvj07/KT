import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        String s = in.nextLine();
        int stones = 0;
        int minStones = n;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '-') {
                stones--;
            } else {
                stones++;
            }
            minStones = Math.min(minStones, stones);
        }
        System.out.println(minStones);
    }
}
