#include <bits/stdc++.h>
using namespace std;
#define int long long
int p[100001], rnk[100001], c[100001];
vector<int> adj[100001], ans;
int find(int x) {
    if(p[x] == x) return p[x];
    return p[x] = find(p[x]);
}
void merge(int x, int y) {
    x = find(x);
    y = find(y);
    if(rnk[x] > rnk[y])
        p[y] = x;
    else if(rnk[x] < rnk[y])
        p[x] = y;
    else {
        rnk[x]++;
        p[y] = x;
    }
}
void dfs(int u, int par) {
    if(u != 1 and !c[u] and !c[par]) {
        ans.push_back(u);
        merge(u, par);
    }
    for(int x: adj[u]) {
        if(x != par)
            dfs(x, u);
    }
}
main() {
    int n;
    cin >> n;
    for(int i = 1; i <= n; i++)
        cin >> p[i] >> c[i];
    for(int i = 1; i <= n; i++) {
        if(p[i] != -1)
            adj[p[i]].push_back(i), adj[i].push_back(p[i]);
    }
    for(int i = 1; i <= n; i++)
        p[i] = i;
    dfs(1, 1);
    reverse(ans.begin(), ans.end());
    if(ans.empty())
        cout << "-1\n";
    else {
        for(int x: ans)
            cout << x << " ";
        cout << "\n";
    }
}
