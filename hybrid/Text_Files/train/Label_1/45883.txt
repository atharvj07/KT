import java.util.Scanner;

public class BubbleCupFinals {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] bubbleField = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                bubbleField[i][j] = scanner.nextInt();
            }
        }

        System.out.println(maxBubbles(bubbleField));
    }

    static int maxBubbles(int[][] bubbleField) {
        int n = bubbleField.length;
        int m = bubbleField[0].length;
        int[][] dp = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                // Choose the maximum of either harvesting the current row or column
                dp[i][j] = Math.max(harvestRow(bubbleField, i, j, dp), harvestColumn(bubbleField, i, j, dp));
            }
        }

        return dp[n - 1][m - 1];
    }

    static int harvestRow(int[][] bubbleField, int i, int j, int[][] dp) {
        if (j == 0) {
            return bubbleField[i][j];
        }
        return bubbleField[i][j] + dp[i][j - 1];
    }

    static int harvestColumn(int[][] bubbleField, int i, int j, int[][] dp) {
        if (i == 0) {
            return bubbleField[i][j];
        }
        return bubbleField[i][j] + dp[i - 1][j];
    }
}
