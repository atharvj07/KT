import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        var home = input.nextLine();
        var away = input.nextLine();
        var n = input.nextInt();
        var homePlayers = new HashMap<Integer, Integer>();
        var awayPlayers = new HashMap<Integer, Integer>();
        for (int i = 0; i < n; i++) {
            var minute = input.nextInt();
            var team = input.next().charAt(0);
            var player = input.nextInt();
            var card = input.next().charAt(0);
            if (card == 'y') {
                if (team == 'h') {
                    homePlayers.put(player, minute);
                } else {
                    awayPlayers.put(player, minute);
                }
            } else {
                if (team == 'h') {
                    if (homePlayers.containsKey(player)) {
                        System.out.println(home + " " + player + " " + homePlayers.get(player));
                    }
                } else {
                    if (awayPlayers.containsKey(player)) {
                        System.out.println(away + " " + player + " " + awayPlayers.get(player));
                    }
                }
            }
        }
    }
}
