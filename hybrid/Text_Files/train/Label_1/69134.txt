import sys
input = sys.stdin.readline

n = int(input())

adj = [set() for _ in range(n)]
color = [None] * n
for i in range(n):
    s = input().strip()
    for j in range(n):
        if (i + j) % n == 0 and s[1] == s[0]:
            print(0)
            exit()
        if s[1] != '?':
            adj[i].add(s[1])
    if s[0] != '?':
        color[i] = s[0]

vis = [False] * n
ans = 1
mod = 998244353

for i in range(n):
    if color[i] != None:
        continue
    if vis[i]:
        print(0)
        exit()
    stk = [i]
    vis[i] = True
    cur = {'W', 'B'}
    while stk:
        v = stk.pop()
        for u in adj[v]:
            if color[u] == None:
                stk.append(u)
                vis[u] = True
                cur.remove(u)
            else:
                cur.discard(color[u])
    color[i] = cur.pop()
    ans *= 2
    ans %= mod

print(ans)
