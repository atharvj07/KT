import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.close();
        int[] primes = {2, 3, 5, 7};
        StringBuilder number = new StringBuilder();
        int pos = 0;
        for (int i = 0; i < n; i++) {
            boolean found = false;
            for (int j = 0; j < 4; j++) {
                if (number.length() == 0) {
                    number.append(primes[j]);
                    primes[j]++;
                    pos = j;
                    found = true;
                    break;
                }
                if (primes[j] % 10 < 9 && Long.parseLong(number.toString()) % primes[j] == 0) {
                    number.append(primes[j] % 10 + 1);
                    primes[j]++;
                    pos = j;
                    found = true;
                    break;
                }
            }
            if (!found) {
                System.out.println(-1);
                return;
            }
            primes[pos] %= 10;
        }
        System.out.println(number);
    }
}
