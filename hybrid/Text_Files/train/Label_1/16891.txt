#include<bits/stdc++.h>
using namespace std ;

typedef unsigned long long ull ;

ull gcd( ull a, ull b ) {
    if( b == 0 ) return a ;
    else return gcd( b, a%b ) ;
}

int main() {
    ull a, b ;
    cin >> a >> b ;
    ull ans = 1, ma = max(a,b) ;
    for( ull i = 1; i <= ma; i++ ) {
        if( i<=a ) ans *= i ;
        if( i<=b ) ans = gcd( ans, i ) ;
    }
    cout << ans << endl ;
    return 0 ;
}
