#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    ll ans = 0;
    for (int i = 0; i < n - 1; i++) {
        int l = i + 1, r = n - 1, mid;
        while (l <= r) {
            mid = (l + r) / 2;
            if (a[mid] >= a[i]) {
                if (a[mid] == a[i]) ans += mid - i - 1;
                r = mid - 1;
            } else
                l = mid + 1;
        }
    }
    cout << ans << endl;
    return 0;
}
