#include <algorithm>
#include <vector>
#include <iostream>
using namespace std;

int n, m;
vector<vector<pair<int, char>>> adj;
vector<bool> vis;
vector<char> path;
vector<int> used;
bool ok = false;

void dfs1(int v, char c) {
    vis[v] = true;
    for (const auto& [u, w] : adj[v]) {
        if (w == c && !vis[u]) {
            path.push_back(c);
            dfs1(u, c);
        }
    }
}

void dfs2(int v, char c) {
    vis[v] = true;
    for (const auto& [u, w] : adj[v]) {
        if (w == c && !vis[u]) {
            used.push_back(path.size());
            dfs2(u, c);
        }
    }
}

int main() {
    cin >> n >> m;
    adj.resize(n + 1);
    vis.resize(n + 1, false);

    for (int i = 0; i < m; ++i) {
        char op;
        cin >> op;

        if (op == '+') {
            int u, v;
            char c;
            cin >> u >> v >> c;
            adj[u].emplace_back(v, c);
            adj[v].emplace_back(u, c);
        } else if (op == '-') {
            int u, v;
            cin >> u >> v;
            adj[u].erase(remove_if(begin(adj[u]), end(adj[u]),
                                    [v](const auto& e) {
                                        return e.first == v;
                                    }),
                         end(adj[u]));
            adj[v].erase(remove_if(begin(adj[v]), end(adj[v]),
                                    [u](const auto& e) {
                                        return e.first == u;
                                    }),
                         end(adj[v]));
        } else {
            int k;
            cin >> k;

            ok = false;
            vis.assign(n + 1, false);
            for (int v = 1; v <= n; ++v) {
                if (!vis[v]) {
                    path.clear();
                    dfs1(v, 'a');
                    if (path.size() >= k) {
                        vis.assign(n + 1, false);
                        used.clear();
                        dfs2(v, 'a');
                        if (used.size() >= k) {
                            ok = true;
                            break;
                        }
                    }
                }
            }

            cout << (ok ? "YES\n" : "NO\n");
        }
    }
    return 0;
}
