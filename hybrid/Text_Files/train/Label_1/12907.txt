#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007LL

typedef long long ll;

vector<pair<ll, ll>> g[100010];
bool vis[100010];

ll dfs(ll n, ll p, ll cnt) {
  if (vis[n]) {
    if (cnt % 2 == 0) return cnt;
    return -1;
  }
  vis[n] = true;
  ll ret = 0;
  for (auto [v, w] : g[n]) {
    if (v == p) continue;
    ret = dfs(v, n, cnt + 1);
    if (ret != -1) return ret;
  }
  return -1;
}

int main() {
  ll n, m;
  cin >> n >> m;
  for (ll i = 0, u, v; i < m; i++) {
    cin >> u >> v;
    g[u].push_back({v, 0});
    g[v].push_back({u, 0});
  }

  ll ans = LLONG_MAX;
  ll ans2 = 0;

  for (ll i = 1; i <= n; i++) {
    if (!vis[i]) {
      ll ans1 = dfs(i, -1, 0);
      if (ans1 != -1) {
        if (ans1 < ans) {
          ans = ans1;
          ans2 = 1;
        } else if (ans1 == ans)
          ans2++;
      }
    }
  }

  if (ans != LLONG_MAX)
    cout << (ans + 1) / 2 << " " << ans2 << endl;
  else
    cout << -1 << endl;
  return 0;
}
