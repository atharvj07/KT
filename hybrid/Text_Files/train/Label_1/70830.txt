from sys import stdin
input=stdin.readline

t=int(input())
while t>0:
    t-=1
    n=int(input())
    a=list(map(int,input().split()))
    check=[0 for i in range(n)]
    for i in range(n):
        p=(i+a[i%n])%n
        if(check[p]):
          print("NO")
          break
        check[p]=1
    else:
        print("YES")
