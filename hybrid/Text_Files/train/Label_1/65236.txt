n, m = map(int, input().split())
arr = [0] * (n + 1)
adj = [[]]
for _ in range(n + 1):
    adj.append([])
for _ in range(m):
    a, b = map(int, input().split())
    arr[a] = 1
    arr[b] = 1
    adj[a].append(b)
    adj[b].append(a)
eq = ans = 0
for i in range(1, n + 1):
    if arr[i] == 0:
        eq = 1
    elif len(adj[i]) > 2:
        ans = 1
if ans:
    print('NO')
else:
    print('YES')
    for i in range(1, n + 1):
        if adj[i]:
            print(i, end=' ')
            print(adj[i][0], end=' ')
            break
    for i in range(1, n + 1):
        if arr[i] == 0:
            print(i, end=' ')
            break
    print()
    for i in range(1, n + 1):
        if arr[i] == 0:
            print(i, end=' ')
            break
    for i in range(1, n + 1):
        if adj[i]:
            print(i, end=' ')
            break
