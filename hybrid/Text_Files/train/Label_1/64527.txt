s = input()
ans = ""
cnt = 0
for x in s:
    if x == '1':
        if cnt > 0:
            ans += '0'
            cnt -= 1
    else:
        ans += '0'
    if cnt == 0:
        for x in s:
            if x == '1':
                cnt += 1
            else:
                break
        for _ in range(cnt):
            ans += '1'
print(ans)
