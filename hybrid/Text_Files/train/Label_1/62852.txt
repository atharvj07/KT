n = int(input())
words = []
for i in range(n):
    words.append(input())
msg = input()

if '<3' in msg and '3>' in msg:
    open_flag = False
    for word in words:
        if '<3' + word + '<3' in msg:
            if not open_flag:
                open_flag = True
            if msg.find('<3' + word + '<3') < msg.find('3>' + word + '<3'):
                print("no")
                exit(0)
        else:
            print("no")
            exit(0)
    print("yes")
else:
    print("no")
