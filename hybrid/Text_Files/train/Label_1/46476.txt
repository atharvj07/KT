import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] grid = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    grid[j][k] = sc.nextInt();
                }
            }
            boolean oppositeSum = true;
            for (int j = 0; j < 5; j++) {
                if (grid[j][0] + grid[j][4] != 7) {
                    oppositeSum = false;
                    break;
                }
            }
            for (int j = 0; j < 5; j++) {
                if (grid[0][j] + grid[4][j] != 7) {
                    oppositeSum = false;
                    break;
                }
            }
            if (!oppositeSum) {
                System.out.println("false");
                continue;
            }
            boolean duplicate = false;
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    for (int l = 0; l < 5; l++) {
                        for (int m = 0; m < 5; m++) {
                            if (grid[j][k] == grid[l][m] && (j != l || k != m)) {
                                duplicate = true;
                            }
                        }
                    }
                }
            }
            if (duplicate) {
                System.out.println("false");
                continue;
            }
            boolean validCube = true;
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (grid[j][k] != 0) {
                        grid[j][k] = 1;
                    }
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (grid[j][k] == 0) {
                        int count = 0;
                        if (j != 0 && grid[j - 1][k] == 1) {
                            count++;
                        }
                        if (j != 4 && grid[j + 1][k] == 1) {
                            count++;
                        }
                        if (k != 0 && grid[j][k - 1] == 1) {
                            count++;
                        }
                        if (k != 4 && grid[j][k + 1] == 1) {
                            count++;
                        }
                        if (count != 2) {
                            validCube = false;
                            break;
                        }
                    }
                }
            }
            if (!validCube) {
                System.out.println("false");
                continue;
            }
            System.out.println("true");
        }
    }
}
