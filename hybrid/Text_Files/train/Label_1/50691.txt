import java.util.*;
import java.io.*;

class Main {
    static int[] p, s;
    static int n, m;

    static int find(int x) {
        return x == p[x] ? x : (p[x] = find(p[x]));
    }

    static void union(int a, int b) {
        p[find(b)] = find(a);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        p = new int[n + 1];
        s = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            s[i] = in.nextInt();
            p[i] = i;
        }
        for (int i = 1; i <= m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            union(a, b);
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (p[i] == i) {
                int sum = 0;
                for (int j = 1; j <= n; j++) {
                    if (p[j] == i) sum += s[j];
                }
                ans += sum - s[i];
            }
        }
        System.out.println(ans);
    }
}
