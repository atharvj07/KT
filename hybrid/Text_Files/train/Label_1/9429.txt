#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        for (auto &x : a) cin >> x;
        ll s1 = 0, s2 = 0;
        for (int i = 0; i < n; i += 2) s1 += max(0ll, a[i]);
        for (int i = 1; i < n; i += 2) s1 += max(0ll, -a[i]);
        for (int i = 0; i < n; i += 2) s2 += max(0ll, -a[i]);
        for (int i = 1; i < n; i += 2) s2 += max(0ll, a[i]);
        cout << max(s1, s2) << '\n';
    }
}
