#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b;
    cin >> n >> a >> b;
    set<int> A, B;
    for (int i = 0; i < a; i++) {
        int x;
        cin >> x;
        A.insert(x);
    }
    for (int i = 0; i < b; i++) {
        int x;
        cin >> x;
        B.insert(x);
    }
    for (int i = 1; i <= n; i++) {
        if (A.count(i) && !B.count(i))
            cout << 1;
        else if (B.count(i) && !A.count(i))
            cout << 2;
        else
            cout << ((i % 2 == 1) ? 1 : 2);
    }
    return 0;
}
