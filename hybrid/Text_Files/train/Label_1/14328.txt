#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int N = 4e5 + 5;
int n, a, b;
ll ans;
string s;

int main() {
    cin >> n;
    vector<tuple<string, int, int, int>>v;
    for (int i = 1; i <= n; i++) {
        cin >> s >> a;
        v.push_back({s, a, (s == "01" || s == "11"), (s == "10" || s == "11")});
    }
    sort(v.begin(), v.end());
    for (auto [x, inf, al, bo] : v) {
        if (al >= a / 2 && bo >= b / 2) ans += inf;
        else if (al < a / 2) a -= al, b += bo;
        else if (bo < b / 2) a -= al, b += bo;
        if (a < b / 2 || b < a / 2) break;
    }
    cout << ans << endl;
    return 0;
}
