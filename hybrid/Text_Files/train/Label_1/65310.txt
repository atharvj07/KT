def candies_from_sequence(seq):
    candies = 0
    while len(seq) > 1:
        new_seq = []
        for i in range(0, len(seq), 2):
            candies += (seq[i] + seq[i + 1]) // 10
            new_seq.append((seq[i] + seq[i + 1]) % 10)
        seq = new_seq
    return candies

n = int(input())
s = list(map(int, input().split()))
q = int(input())
queries = [list(map(int, input().split())) for _ in range(q)]

for l, r in queries:
    print(candies_from_sequence(s[l-1:r]))
