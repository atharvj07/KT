import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] a = new int[n], b = new int[m];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        for (int i = 0; i < m; i++) b[i] = in.nextInt();
        boolean ok = true;
        for (int i = 0; i < m; i++) {
            int cur = 0;
            for (int j = 0; j < n; j++) {
                cur ^= a[j];
            }
            if (cur != b[i]) ok = false;
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print((a[i] ^ b[j] + (i % 2 + j % 2 == 1 ? 1 : 0)) + " ");
            }
            System.out.println();
        }
    }
}
