for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for i in range(n - 1):
        if a[i] == a[i + 1]:
            cur = 2
            j = i + 2
            while j < n and a[j] == a[i]:
                cur += 1
                j += 1
            ans += cur * (cur - 1) // 2
            i += cur - 1
    print(ans)
