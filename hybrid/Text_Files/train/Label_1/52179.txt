import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] ans = new long[n];
        for (int i = 0; i < n; i++) {
            int k = scanner.nextInt();
            long[][] cards = new long[k][2];
            for (int j = 0; j < k; j++) {
                cards[j][0] = scanner.nextInt();
                cards[j][1] = scanner.nextInt();
            }
            Arrays.sort(cards, (a, b) -> Long.compare(a[1], b[1]));
            long sum = 0;
            int cnt = 0;
            for (int j = k - 1; j >= 0; j--) {
                if (sum + cards[j][0] <= 3) {
                    sum += cards[j][0];
                    if (cnt % 10 == 9) {
                        ans[i] += cards[j][1] * 2;
                    } else {
                        ans[i] += cards[j][1];
                    }
                    cnt++;
                }
            }
        }
        long max = 0;
        for (long x : ans) {
            max = Math.max(max, x);
        }
        System.out.println(max);
    }
}
