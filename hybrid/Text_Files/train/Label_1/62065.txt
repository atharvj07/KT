a, b, t, *s = input().strip().split()
c = [[1000000000, 1000000000] for _ in range(int(a) + 1)]
c[int(s[0] == 'w')][int(s[0] == 'w')] = 1
x = 1
y = int(s[0] == 'w')
while True:
    nx = (x + int(s[x % int(a)] == 'w')) % int(a) + 1
    if c[nx][y] > c[x][y] + int(y != int(s[x % int(a)] == 'w')) * int(b) + int(s[x % int(a)] == 'w'):
        c[nx][y] = c[x][y] + int(y != int(s[x % int(a)] == 'w')) * int(b) + int(s[x % int(a)] == 'w')
        x = nx
    ny = y ^ 1
    if c[x][ny] > c[x][y] + int(b) + 1:
        c[x][ny] = c[x][y] + int(b) + 1
        y = ny
    if c[x][y] >= int(t):
        break
print(x - 1)
