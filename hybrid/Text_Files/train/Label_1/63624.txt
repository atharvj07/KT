from math import gcd
from functools import reduce

def find_wcd(n, pairs):
    def prime_factors(n):
        i = 2
        factors = set()
        while i * i <= n:
            if n % i:
                i += 1
            else:
                n //= i
                factors.add(i)
        if n > 1:
            factors.add(n)
        return factors

    common_factors = prime_factors(pairs[0][0]) | prime_factors(pairs[0][1])
    
    for a, b in pairs[1:]:
        common_factors &= prime_factors(a) | prime_factors(b)
        if not common_factors:
            return -1
    
    return min(common_factors)

n = int(input())
pairs = [tuple(map(int, input().split())) for _ in range(n)]
print(find_wcd(n, pairs))
