import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int T = in.nextInt();
        for (int i = 0; i < T; ++i) {
            int n = in.nextInt();
            int[] arr = new int[n + 1];
            for (int j = 0; j <= n; ++j) {
                arr[j] = j;
            }
            int[] min = findMin(arr, 1, n);
            int[] max = findMax(arr, 1, n);
            System.out.println("! " + min[0] + " " + max[0]);
        }
    }

    public static int[] findMin(int[] arr, int left, int right) {
        if (left == right) {
            return new int[]{left};
        }
        int mid = (left + right) / 2;
        int[] min1 = findMin(arr, left, mid);
        int[] min2 = findMin(arr, mid + 1, right);
        int minIndex = arr[min1[0]] < arr[min2[0]] ? min1[0] : min2[0];
        return new int[]{minIndex};
    }

    public static int[] findMax(int[] arr, int left, int right) {
        if (left == right) {
            return new int[]{left};
        }
        int mid = (left + right) / 2;
        int[] max1 = findMax(arr, left, mid);
        int[] max2 = findMax(arr, mid + 1, right);
        int maxIndex = arr[max1[0]] > arr[max2[0]] ? max1[0] : max2[0];
        return new int[]{maxIndex};
    }
}
