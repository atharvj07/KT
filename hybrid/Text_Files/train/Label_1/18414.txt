#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    int pos = 1;
    for (int i = 0; i < k; i++) {
      if (pos == n) {
        pos = -1;
        break;
      }
      if (h[pos] >= h[pos + 1]) {
        pos++;
      } else {
        h[pos]++;
      }
    }
    cout << pos << endl;
  }
  return 0;
}
