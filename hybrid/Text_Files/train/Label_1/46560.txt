import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] max = new int[n];
        max[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            max[i] = Math.max(a[i], max[i + 1]);
        }
        int[] sum = new int[n];
        sum[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            sum[i] = sum[i + 1] + a[i];
        }
        int l = 0, r = 0, best = 0;
        while (r < n) {
            int cur = sum[r] - (l > 0 ? sum[l - 1] : 0);
            while (l < r && cur + k < r - l + 1) {
                cur -= a[l];
                l++;
            }
            best = Math.max(best, r - l + 1);
            r++;
        }
        l = 0;
        r = 0;
        while (r < n) {
            int cur = sum[r] - (l > 0 ? sum[l - 1] : 0);
            if (cur + k >= r - l + 1) {
                while (l < r && a[l] == 1) {
                    l++;
                }
                if (l < r) {
                    a[l] = 1;
                    k--;
                }
            }
            best = Math.max(best, r - l + 1);
            r++;
        }
        System.out.println(best);
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
