#include <bits/stdc++.h>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long a, b, c;
    cin >> a >> b >> c;
    long long x = abs(a - b) + abs(a - c) + abs(b - c);
    long long y = abs(a - b) + abs(a + b - 2 * c);
    long long z = abs(a - c) + abs(a + c - 2 * b);
    cout << min({x, y, z}) << endl;
  }
}
