for _ in range(int(input())):
  a, b, p = map(int, input().split())
  s = input()
  pref = [0] * len(s)
  for i in range(len(s)):
    pref[i] = (pref[i - 1] + (s[i] == "A")) if i else (s[i] == "A")
  l = 0
  r = len(pref)
  ans = len(pref)
  while l <= r:
    m = (l + r) // 2
    if pref[m - 1] <= (m + 1) * a - p:
      ans = m
      r = m - 1
    else:
      l = m + 1
  print(ans + 1)
