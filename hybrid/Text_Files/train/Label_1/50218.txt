import java.util.*;

class Main {
    static List<Integer>[] adj;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), t = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[t];
        for (int i = 0; i < t; i++) a[i] = sc.nextInt();
        int[] deg = new int[n];
        adj = new List[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < t; i++) {
            int sum = 0;
            for (int j = i + 1; j < t; j++) sum += a[j];
            if (a[i] > sum || a[i] > n - i - 1) {
                System.out.println(-1);
                return;
            }
            if (i + 1 != t) deg[a[i]]++;
        }
        if (deg[1] != 1 || deg[n - t - 1] != 1) {
            System.out.println(-1);
            return;
        }
        for (int i = t - 1; i > 0; i--) {
            for (int j = a[i - 1]; j > 0; j--) {
                if (deg[j] < 2) {
                    adj[i + a[i]].add(j);
                    deg[j]++;
                    deg[i + a[i]]++;
                    break;
                }
            }
        }
        if (k > 0) {
            for (int i = t + 1, j = n - 1; i < n && k > 0; i++, j--) {
                if (deg[i] < 2) {
                    adj[i].add(j);
                    deg[i]++;
                    deg[j]++;
                    k--;
                }
            }
        }
        if (k != 0) {
            System.out.println(-1);
            return;
        }
        System.out.println(n);
        for (int i = 0; i < n; i++) {
            for (int j : adj[i]) {
                System.out.println(i + " " + j);
                break;
            }
        }
    }
}
