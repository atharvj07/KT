t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    result = -1
    for i in range(n-1):
        if s[i] == 'a':
            if s[i+1] == 'a':
                result = 2
                break
            elif i < n-2 and s[i+2] == 'a':
                result = 3
                break
    print(result)
