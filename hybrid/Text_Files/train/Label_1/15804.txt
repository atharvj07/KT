#include <bits/stdc++.h>
using namespace std;

int a[365], n;

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a, a + n);
  int ans = 360;
  for (int i = 0; i < n; i++) {
    int tmp = a[i];
    int idx = lower_bound(a, a + n, 360 - tmp) - a;
    if (idx == n) idx = 0;
    ans = min(ans, 360 - (tmp + a[idx]));
    ans = min(ans, tmp + a[idx]);
  }
  cout << ans << endl;
}
