import sys

def solve():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n = int(sys.stdin.readline())
        l = [0] + [int(x) for x in sys.stdin.read().split()]
        r = l[1:]
        l = l[:-1]
        ans = []
        use_time = 1
        for i in range(n):
            if use_time < l[i]:
                use_time = l[i]
            if use_time > r[i]:
                ans.append(0)
            else:
                ans.append(use_time)
                use_time += 1
        print(*ans)

solve()
