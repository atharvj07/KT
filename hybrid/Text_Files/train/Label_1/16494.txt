#include <iostream>
using namespace std;
char a[51][51];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (j + 1 < m) {
                if (i > 0 && a[i][j] == a[i - 1][j + 1] && a[i][j] == a[i][j + 1] && a[i][j] == a[i - 1][j]) {
                    cout << "Yes";
                    return 0;
                }
                if (i + 1 < n && a[i][j] == a[i + 1][j + 1] && a[i][j] == a[i][j + 1] && a[i][j] == a[i + 1][j]) {
                    cout << "Yes";
                    return 0;
                }
            }
            if (i + 1 < n) {
                if (a[i][j] == a[i + 1][j] && a[i][j] == a[i + 1][j + 1] && a[i][j] == a[i][j + 1]) {
                    cout << "Yes";
                    return 0;
                }
            }
        }
    }
    cout << "No";
}
