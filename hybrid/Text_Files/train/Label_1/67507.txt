n=int(input())
a=[int(i) for i in input().split()]
ad=[0 for i in range(n+2)]
for x in range(n):
    ad[a[x]]+=1
for x in range(1,n+1):
    ad[x]+=ad[x-1]
vis=[False for i in range(n+2)]
ans=0
cur=0
while(1):
    ans+=1
    while(cur+1<=n and vis[cur+1]==False):
        cur+=1
        vis[cur]=True
    if(cur==n):
        break
    cur=a[cur]
    ad[cur]-=1
    if(ad[cur]==0):
        vis[cur]=True
print(ans)
