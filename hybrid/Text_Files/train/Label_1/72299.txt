import sys
input = sys.stdin.readline

n = int(input())
trees = [tuple(map(int,input().split())) for _ in range(n)]

trees.sort()

ans = 0
l = -10**9
r = -10**9

for x,h in trees:
    if x+h<=r or x-h>=l:
        continue
    
    ans += 1
    if x<l or x>r:
        l = x-h
        r = x+h

print(ans)
