import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int n = sc.nextInt();
            int[] p = new int[n];
            for (int i = 0; i < n; i++) {
                p[i] = sc.nextInt();
            }
            if (isRoundDancePossible(p, n)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }

    private static boolean isRoundDancePossible(int[] p, int n) {
        // Check if clockwise round dance is possible
        boolean clockwise = true;
        for (int i = 1; i < n; i++) {
            if (p[i] != p[i - 1] + 1) {
                clockwise = false;
                break;
            }
        }
        if (p[n - 1] != 1) {
            clockwise = false;
        }

        // Check if counterclockwise round dance is possible
        boolean counterclockwise = true;
        for (int i = 1; i < n; i++) {
            if (p[i] != p[i - 1] - 1) {
                counterclockwise = false;
                break;
            }
        }
        if (p[0] != n) {
            counterclockwise = false;
        }

        return clockwise || counterclockwise;
    }
}
