#include<iostream>
#include<vector>
#include<queue>
using namespace std;
vector<int> gr[5005], costs[5005], rev[5005];
vector<int> arr[5005], dep[5005];
int a[5005], b[5005], c[5005], k, sum, n, m, mn[5005][5005], cnt[5005], ans, by[5005], taken[5005];
void dfs1(int node){
    for (int i = 0; i < gr[node].size(); i++){
        if (dep[node].size() == 0 ){
            dep[node].push_back(costs[node][i]);
            mn[node][gr[node][i]] = costs[node][i];
        }
        else{
            if (mn[node][gr[node][i]] == -1){
                dep[node].push_back(costs[node][i] + dep[node].back());
                mn[node][gr[node][i]] = costs[node][i] + dep[node].back();
            }
        }
        dfs1(gr[node][i]);
    }
}
void dfs2(int node, int sum){
    //cout << node << " " << sum << endl;
    if (cnt[node] == 0){
        if (sum < a[node]){
            return;
        }
        sum -= a[node];
        ans += c[node];
        taken[node] = 1;
        cnt[node] = 1;
        return;
    }
    bool flag = 0;
    for (int i = 0; i < gr[node].size(); i++){
        int u = gr[node][i];
        dfs2(u, sum + dep[node][i] + b[node]);
        if (taken[u] == 1){
            flag = 1;
        }
    }
    if (flag == 1){
        ans += c[node];
    }
}
int main(){
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++){
        cin >> a[i] >> b[i] >> c[i];
    }
    for (int i = 1; i <= m; i++){
        int u, v;
        cin >> u >> v;
        gr[u].push_back(v);
        rev[v].push_back(u);
        costs[u].push_back(1);
    }
    dfs1(1);
    dfs2(1, k);
    if (ans == 0) ans = -1;
    cout << ans << endl;
    return 0;
}
