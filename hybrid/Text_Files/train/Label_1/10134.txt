#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> nums(n);
  for (auto &num : nums) cin >> num;
  int ans = 0;
  for (auto num : nums) {
    int cnt = 0;
    while (num) {
      if (num % 10 == 4 || num % 10 == 7) cnt++;
      num /= 10;
    }
    if (cnt <= k) ans++;
  }
  cout << ans << endl;
  return 0;
}
