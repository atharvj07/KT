import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] p0 = new int[]{sc.nextInt(), sc.nextInt()};
        int[] p1 = new int[]{sc.nextInt(), sc.nextInt()};
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int[] p2 = new int[]{sc.nextInt(), sc.nextInt()};
            int v = (p1[1] - p0[1]) * (p2[0] - p1[0]) - (p2[1] - p1[1]) * (p1[0] - p0[0]);
            if (v > 0) System.out.println("COUNTER_CLOCKWISE");
            else if (v < 0) System.out.println("CLOCKWISE");
            else {
                if ((p2[0] <= Math.max(p0[0], p1[0]) && p2[0] >= Math.min(p0[0], p1[0])) || (p2[1] <= Math.max(p0[1], p1[1]) && p2[1] >= Math.min(p0[1], p1[1]))) {
                    if ((p0[0] == p1[0] && p2[0] == p1[0]) && (p0[1] <= p2[1] && p2[1] <= p1[1]))
                        System.out.println("ONLINE_BACK");
                    else if ((p0[1] == p1[1] && p2[1] == p1[1]) && (p0[0] <= p2[0] && p2[0] <= p1[0]))
                        System.out.println("ONLINE_BACK");
                    else if ((p0[0] == p1[0] && p2[0] == p1[0]) && (p1[1] <= p2[1] && p2[1] <= p0[1]))
                        System.out.println("ONLINE_FRONT");
                    else if ((p0[1] == p1[1] && p2[1] == p1[1]) && (p1[0] <= p2[0] && p2[0] <= p0[0]))
                        System.out.println("ONLINE_FRONT");
                    else
                        System.out.println("ON_SEGMENT");
                } else System.out.println("ONLINE_BACK");
            }
        }
    }
}
