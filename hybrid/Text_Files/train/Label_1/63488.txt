from math import gcd
from functools import reduce

def find_min_people_and_swords(n, a):
    # Find the maximum value in a as it represents the original number of swords for some type
    x = max(a)
    # Calculate the differences which represent how many swords were taken of each type
    diffs = [x - ai for ai in a]
    # Calculate the GCD of all differences to find the minimum number of swords taken by one person
    z = reduce(gcd, diffs)
    # Calculate the sum of all differences to find the total number of swords taken
    total_swords_taken = sum(diffs)
    # The number of people y is the total number of swords taken divided by the number of swords taken by one person
    y = total_swords_taken // z
    return y, z

# Read input
n = int(input().strip())
a = list(map(int, input().strip().split()))

# Solve and print the result
y, z = find_min_people_and_swords(n, a)
print(y, z)
