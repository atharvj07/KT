#include <iostream>

const int MOD = 1e9 + 7;

int main() {
    long long x, k;
    std::cin >> x >> k;
    int ans = x;
    for (int i = 0; i < k; i++) {
        ans *= 2;
        ans %= MOD;
        if (i < k - 1) {
            ans /= 2;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
