import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n*m];
        int idx = 0;
        for (int i=1; i<=n; i++) {
            for (int j=1; j<=m; j++) {
                a[idx++] = i*j;
            }
        }
        Arrays.sort(a);
        System.out.println(a[k-1]);
    }
}
