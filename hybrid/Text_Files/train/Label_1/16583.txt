#include <iostream>
#include <vector>

using namespace std;

const int N = 100005;

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  vector<pair<int, int>> v(q);
  for (int i = 0; i < q; i++) { cin >> v[i].first >> v[i].second; }
  for (int i = 0; i < q; i++) {
    int l = v[i].first - 1, r = v[i].second - 1, mn = 1e9, mx = -1e9;
    for (int j = l; j <= r; j++) {
      mn = min(mn, a[j] - b[j]);
      mx = max(mx, a[j] - b[j]);
    }
    if (mn >= 0 && mx <= 0) cout << (mn == 0 ? 1 : (mn <= -mx ? abs(mn) : abs(mx) + 1)) << endl;
    else cout << "-1" << endl;
  }
  return 0;
}
