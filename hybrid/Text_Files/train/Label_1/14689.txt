#include <bits/stdc++.h>
using namespace std;

int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }

int main() {
  long long n, m, k;
  cin >> n >> m >> k;
  vector<long long> a(k);
  for (auto &x : a) cin >> x;
  long long g = a[0];
  for (int i = 1; i < k; i++) g = gcd(g, a[i]);
  for (long long i = 1; i <= n; i++) {
    long long cur = g;
    for (int j = 0; j < k; j++) {
      if (cur != a[j]) break;
      cur = gcd(cur, i + j + 1);
      if (j == k - 1) { cout << "YES" << endl; return 0; }
    }
  }
  cout << "NO" << endl;
}
