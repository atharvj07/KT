import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> scores = new HashMap<>();
        int maxScore = 0;
        String winner = "";
        for (int i = 0; i < n; i++) {
            String name = sc.next();
            int score = sc.nextInt();
            scores.put(name, scores.getOrDefault(name, 0) + score);
            if (scores.get(name) > maxScore) {
                maxScore = scores.get(name);
                winner = name;
            } else if (scores.get(name) == maxScore && scores.get(name) >= score && (winner.equals("") || scores.get(winner) < score)) {
                winner = name;
            }
        }
        System.out.println(winner);
    }
}
