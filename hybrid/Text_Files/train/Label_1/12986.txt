#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m, k;
struct chip {
  ll sx, sy, fx, fy;
  bool ok = false;
};
chip arr[205];
int main() {
  cin >> n >> m >> k;
  for (ll i = 0; i < k; i++)
    cin >> arr[i].sx >> arr[i].sy >> arr[i].fx >> arr[i].fy;
  ll ans = 0;
  string path = "";
  for (ll i = 0; i < k; i++) {
    if (!arr[i].ok) {
      ll x = arr[i].fx - arr[i].sx;
      ll y = arr[i].fy - arr[i].sy;
      if (x < 0) {
        path += string(-x, 'U');
        x = 0;
      }
      if (y < 0) {
        path += string(-y, 'L');
        y = 0;
      }
      if (x > 0) {
        path += string(x, 'D');
        y = 0;
      }
      if (y > 0) {
        path += string(y, 'R');
        y = 0;
      }
      arr[i].ok = true;
      ans += abs(x) + abs(y);
    }
  }
  if (ans <= 2 * n * m)
    cout << ans << "\n" << path << endl;
  else
    cout << -1 << "\n";
}
