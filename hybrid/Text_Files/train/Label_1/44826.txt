import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt(), k = scanner.nextInt();
        long[] a = new long[n], b = new long[n], c = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
        }

        long[][] g = new long[n][n];
        for(int i = 0; i < n; i++) {
            Arrays.fill(g[i], -1);
            g[i][i] = 0;
        }
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            g[u][v] = g[v][u] = 1;
        }
        long ans = dfs(0, 0, g, a, b, c, k);
        System.out.println(ans == -1 ? -1 : ans);
    }
    static long dfs(int i, long army, long[][] g, long[] a, long[] b, long[] c, long k) {
        if (i == a.length) {
            return 0L;
        }
        if (army < a[i]) {
            return -1L;
        }
        long ans = -1L;
        for (int j = i; j < a.length; j++) {
            if (j == i) army += b[i];
            else {
                if (g[i][j] == -1) continue;
                army -= g[i][j];
            }
            long temp = dfs(j+1, army, g, a, b, c, k);
            if (temp != -1L) ans = Math.max(ans, temp + c[i]);
            if (j != i) army += g[i][j];
        }
        return ans;
    }
}
