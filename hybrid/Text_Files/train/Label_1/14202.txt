#include <bits/stdc++.h>
using namespace std;
int main() {
  int k;
  cin >> k;
  string s, t;
  cin >> s >> t;
  string ans = s;
  for (int i = 0; i < k; i++) {
    if (s[i] != t[i]) {
      ans[i] = s[i];
      int j;
      for (j = i + 1; j < k; j++) { ans[j] = 'a'; }
      if (ans > s && ans < t) break;
      for (j = i + 1; j < k; j++) { ans[j] = 'z'; }
      if (ans > s && ans < t) break;
      ans[i] = t[i];
      for (j = i + 1; j < k; j++) { ans[j] = 'a'; }
      if (ans > s && ans < t) break;
      ans[i] = s[i];
    }
  }
  cout << ans << endl;
  return 0;
}
