import sys
input = sys.stdin.readline

def main():
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]
    for i in range(n):
        a[i][3], a[i][5] = a[i][5], a[i][3]
    
    ans = 0
    for i in range(10):
        for j in range(10):
            for k in range(6):
                if a[0][k] == i and a[1][k] == j:
                    ok = True
                    for l in range(2, n):
                        if a[l][k] != i and a[l][k] != j:
                            ok = False
                            break
                    if ok:
                        ans = max(ans, j*10+i)
    print(ans)

main()
