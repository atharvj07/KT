#include <bits/stdc++.h>
using namespace std;
bool canmake(string s) {
  int x = -1, y = -1, n = s.size();
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '?') {
      if (x == -1)
        x = i;
      else if (y == -1)
        y = i;
      else
        return false;
    } else if (s[i] != 'a')
      return false;
  }
  if (x != -1 && y == -1)
    y = x + 4;
  else if (y != -1 && x == -1)
    x = y - 4;
  else if (x != -1 && y != -1 && y - x < 5)
    return false;
  s[x] = 'b';
  s[x + 1] = 'a';
  s[x + 2] = 'c';
  s[x + 3] = 'a';
  s[x + 4] = 'b';
  s[y] = 'a';
  for (int i = 0; i < s.size(); i++)
    if (s[i] == '?') s[i] = 'b';
  return true;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (!canmake(s))
      cout << "No\n";
    else {
      cout << "Yes\n";
      cout << s << "\n";
    }
  }
}
