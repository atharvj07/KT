n = int(input())
directions = input().strip()
jumps = list(map(int, input().split()))

i = 0
visited = set()

while 0 <= i < n:
    if i in visited:
        print("INFINITE")
        break
    visited.add(i)
    
    if directions[i] == '>':
        i += jumps[i]
    else:
        i -= jumps[i]
else:
    print("FINITE")
