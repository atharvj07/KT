import bisect

n,m = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
x = int(input())
c = [[0]*m for i in range(n)]
for i in range(n):
    for j in range(m):
        c[i][j] = a[i]*b[j]
dp = [[-1]*m for i in range(n+1)]
for i in range(n):
    for j in range(m):
        dp[i+1][j] = dp[i][j]
        if i>0: dp[i+1][j] = max(dp[i+1][j],dp[i][j])
        if j>0: dp[i+1][j] = max(dp[i+1][j],dp[i+1][j-1])
        dp[i+1][j] = max(dp[i+1][j],dp[i][j]+c[i][j])
for j in range(m-1,-1,-1):
    for i in range(n):
        ok = False
        l = 0
        r = m-1
        while l<=r:
            mid = (l+r)//2
            if dp[i+1][mid]>=x:
                ok = True
                r = mid-1
            else: l = mid+1
        if ok: dp[i+1][j] = max(dp[i+1][j],dp[i+1][r+1])
ans = 0
for i in range(n):
    for j in range(m):
        l = i
        r = n-1
        while l<=r:
            mid = (l+r)//2
            if dp[mid+1][j]>=x:
                ans = max(ans,(mid-i+1)*(j-bisect.bisect_left(dp[i+1],dp[i+1][j]-c[mid][j]+c[i][j],lo=j+1)))
                r = mid-1
            else: l = mid+1
print(ans)
