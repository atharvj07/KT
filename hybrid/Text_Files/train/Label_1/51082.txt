import java.util.*;
import java.math.*;

public class Main {
  static int N, M, K;
  static List<List<Integer>> G = new ArrayList<>();
  static List<Integer> candis = new ArrayList<>();
  static int dist[];

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    N = sc.nextInt();
    M = sc.nextInt();
    K = sc.nextInt();

    for(int i = 0; i <= N; i++)
      G.add(new ArrayList<>());

    for(int i = 0; i < M; i++) {
      int a = sc.nextInt()-1, b = sc.nextInt()-1;
      int l = sc.nextInt();
      G.get(a).add(b); G.get(a).add(l);
      G.get(b).add(a); G.get(b).add(l);
    }

    for(int i = 0; i < K; i++)
      candis.add(sc.nextInt()-1);

    dist = new int[N];
    Arrays.fill(dist, Integer.MAX_VALUE);

    PriorityQueue<Node> que = new PriorityQueue<>();
    que.add(new Node(0, 0));
    dist[0] = 0;

    while(!que.isEmpty()) {
      Node cur = que.poll();
      if(dist[cur.v] < cur.d) continue;

      if(candis.contains(cur.v)) {
        for(int i = 0; i < N; i++)
          dist[i] = Math.min(dist[i], cur.d + dist[cur.v]);
      }

      for(int i = 0; i < G.get(cur.v).size(); i += 2) {
        int to = G.get(cur.v).get(i);
        int l = G.get(cur.v).get(i+1);
        if(dist[to] > dist[cur.v] + l) {
          dist[to] = dist[cur.v] + l;
          que.add(new Node(to, dist[to]));
        }
      }
    }

    int ans = (int)Math.round(Arrays.stream(dist).average().getAsDouble());
    System.out.println(ans);
  }

  static class Node implements Comparable<Node> {
    int v, d;

    Node(int v, int d) { this.v = v; this.d = d; }

    public int compareTo(Node o) {
      return this.d - o.d;
    }
  }
}
