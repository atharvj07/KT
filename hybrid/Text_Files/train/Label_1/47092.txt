import java.util.*;

public class Main {
  static int[] parent;

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int m = scanner.nextInt();
    parent = new int[n + 1];
    for (int i = 0; i <= n; i++) {
      parent[i] = i;
    }
    for (int i = 0; i < m; i++) {
      int x = scanner.nextInt();
      int y = scanner.nextInt();
      union(x, y);
    }
    scanner.close();

    Map<Integer, Integer> componentSize = new HashMap<>();
    for (int i = 1; i <= n; i++) {
      int root = findParent(i);
      componentSize.put(root, componentSize.getOrDefault(root, 0) + 1);
    }

    int[] componentSizes = new int[componentSize.size()];
    int index = 0;
    for (Integer size : componentSize.values()) {
      componentSizes[index++] = size;
    }
    Arrays.sort(componentSizes);

    System.out.println(componentSizes.length);
    for (int size : componentSizes) {
      System.out.print(size + " ");
    }
  }

  static int findParent(int x) {
    if (parent[x] == x) {
      return x;
    }
    return parent[x] = findParent(parent[x]);
  }

  static void union(int x, int y) {
    int a = findParent(x);
    int b = findParent(y);
    if (a != b) {
      parent[a] = b;
    }
  }
}
