#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
const int MAXN = 105;
int n, m, g[MAXN][MAXN], ans;
int dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};
void dfs(int x, int y, int cnt, int col) {
    if (x < 1 || x > n || y < 1 || y > m) return;
    if (g[x][y] == col) {
        ans = min(ans, cnt);
        return;
    }
    for (int k = 0; k < 4; k++) dfs(x + dx[k], y + dy[k], cnt + 1, col);
}
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; ++j) g[i][j] = (s[j - 1] == 'B');
    }
    ans = -1;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            if (!g[i][j]) continue;
            dfs(i, j, 0, g[i][j]);
        }
    }
    cout << ans << endl;
    return 0;
}
