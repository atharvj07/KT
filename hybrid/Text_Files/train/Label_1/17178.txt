#include<bits/stdc++.h>
using namespace std;
string revs(string a)
{
	reverse (a.begin(), a.end());
	return a;
}
string add(string a, string b)
	{
		while(a.length()<b.length()) a='0'+a;
		while(b.length()<a.length()) b='0'+b;
		int carry=0;
		string ans="";
		for(int i=a.length()-1;i>=0;i--)
		{
			int sum=(a[i]-'0')+(b[i]-'0')+carry;
			carry=sum/2;
			sum=sum%2;
			ans+=(char)(sum+'0');
		}
		if(carry) ans+=(char)(carry+'0');
		return ans;
	}
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		string a,b;
		cin>>a>>b;
		string temp=b;
		string ans=a;
		for(int i=0;;i++)
		{
			string c=add(a,revs(temp));
			if(c<ans) 
			{
				ans=c;
			}
			else break;
			temp=temp+'0';
		}
		cout<<ans.length()-1<<endl;
	}
	return 0;
	
}
