#include <bits/stdc++.h>
using namespace std;
#define ll long long
int n;
vector<ll> a, sz;
vector<vector<ll>> g;
ll ans = LLONG_MIN;
bool ok = true;
void dfs(ll u, ll p, ll s) {
    sz[u] = 1;
    for (auto &v : g[u]) {
        if (v == p) continue;
        dfs(v, u, s + a[v - 1]);
        if (sz[v] > (n >> 1) || sz[u] > (n >> 1)) ok = false;
        sz[u] += sz[v];
    }
    ans = max(ans, s);
}
int main() {
    cin >> n;
    a.resize(n);
    sz.resize(n + 1);
    g.resize(n + 1);
    for (auto &x : a) cin >> x;
    for (int i = 0, u, v; i < n - 1; i++) {
        cin >> u >> v;
        g[u].emplace_back(v), g[v].emplace_back(u);
    }
    dfs(1, 1, a[0]);
    if (ok) cout << ans << endl;
    else cout << "Impossible" << endl;
    return 0;
}
