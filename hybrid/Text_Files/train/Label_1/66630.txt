for _ in range(int(input())):
    n,m=map(int,input().split())
    a=list(map(int,input().split()))
    b=list(map(int,input().split()))
    d=a.copy()
    a+=b
    a.sort()
    ans=0
    for i in range(1,len(a)):
        if a[i]==a[i-1]:
            ans+=1
            if d[i-1]==a[i]:
                d.pop(i-1)
            else:
                d.pop(i)
    print(ans)
