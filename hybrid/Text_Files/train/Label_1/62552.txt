n,m=map(int,input().split())
d=[0]*n
for i in range(m):
    a,b,c=map(int,input().split())
    a-=1;b-=1
    d[a]-=c;d[b]+=c
p=list(enumerate(d));p.sort(key=lambda x:x[1])
ans=0
for i in range(n):
    if p[i][1]==0:continue
    a=p[i][0];b=p[i+1][0];c=p[i][1]-p[i+1][1]
    ans+=c
    d[a]+=c;d[b]-=c
print(ans)
