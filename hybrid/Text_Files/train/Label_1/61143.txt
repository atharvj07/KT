import sys

n, M, T = map(int, input().split())
queries = sorted(set(input() for _ in range(n)))
possible = set(queries)
for i in range(len(queries)):
    for j in range(i + 1, len(queries)):
        if int(queries[i][:2]) + T // 3600 >= int(queries[j][:2]):
            possible.discard(queries[j])
if len(possible) == M:
    for query in queries:
        if query in possible:
            print(1)
        else:
            print(2)
else:
    print('No solution')
