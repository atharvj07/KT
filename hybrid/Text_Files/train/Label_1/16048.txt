#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  int lw = -1, up = -1;
  for (int i = 0; i < n; i++) {
    if (s[i] >= 'a' && s[i] <= 'z') {
      ans = max(ans, i - up);
      lw = i;
    } else {
      up = i;
    }
  }
  ans = max(ans, n - up);
  cout << ans << endl;
  return 0;
}
