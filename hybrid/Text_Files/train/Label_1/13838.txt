#include<bits/stdc++.h>

using namespace std;

int main() {
    int t, n, ans = 0;
    string s;
    cin >> t >> s;
    n = s.size();
    for (int i = 0; i < n; i++) {
        if ((s[i] == '6' || s[i] == '9') || (s[i] == '2' || s[i] == '5')) {
            ans++;
        } else if (s[i] == '0' || s[i] == '8') {
            continue;
        } else {
            ans += 2;
        }
    }
    cout << ans / 2 << endl;
    return 0;
}
