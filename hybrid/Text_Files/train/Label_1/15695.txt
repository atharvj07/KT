#include <iostream>
#include <string>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  k = (k) % 2;
  n = (n) % 2;
  if (n == k) {
    string s;
    for (int i = 0; i < (n + 1) / 2; i++) s += '1';
    for (int i = 0; i < (n) / 2; i++) s += '0';
    cout << s << "\n";
  } else {
    string s;
    for (int i = 0; i < (n) / 2; i++) s += '1';
    for (int i = 0; i < (n) / 2; i++) s += '0';
    s += '1';
    cout << s << "\n";
  }
  return 0;
}
