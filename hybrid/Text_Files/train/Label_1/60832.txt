MOD = 10**9 + 7

def fast_pow(base, power):
    result = 1
    while power > 0:
        if power % 2 == 1:
            result = (result * base) % MOD
        base = (base * base) % MOD
        power //= 2
    return result

x, k = map(int, input().split())

if x == 0:
    print(0)
else:
    print(((x * fast_pow(2, k + 1) % MOD) - fast_pow(2, k) + 1 + MOD) % MOD)
