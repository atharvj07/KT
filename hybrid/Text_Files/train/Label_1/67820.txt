from sys import stdin
n = int(stdin.readline())
a = list(map(int, stdin.readline().split()))
for i in range(n - 1):
    a.sort()
    if a[-1] <= 0:
        print(2, a.index(a[-1]) + 1)
        a.pop()
    else:
        print(1, a.index(a[-2]) + 1, a.index(a[-1]) + 1)
        a[-2] = a[-2] * a[-1]
        a.pop()
