#include <cstdio>

int main() {
    int T, n, p;
    scanf("%d", &T);
    while (T--) {
        scanf("%d%d", &n, &p);
        int cnt = 0;
        for (int i = 1; i <= n; ++i) {
            for (int j = i + 1; j <= n; ++j, ++cnt) {
                if (cnt < 2 * n + p) printf("%d %d\n", i, j);
            }
        }
    }
    return 0;
}
