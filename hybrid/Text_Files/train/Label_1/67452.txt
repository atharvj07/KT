import sys
for _ in range(int(input())):
    s = input()
    m = int(input())
    b = [int(x) for x in input().split()]
    t = []
    fre = {}
    for c in s:
        if c not in fre: fre[c] = 0
        fre[c] += 1
    for i in range(m-1,-1,-1):
        ok = False
        for j in range(26):
            c = chr(ord('a') + j)
            if fre.get(c,0) > 0:
                fre[c] -= 1
                k = 0
                for d in range(i+1,len(b)):
                    k += fre.get(c,0)
                if k == b[i]:
                    t.append(c)
                    ok = True
                    break
        if not ok:
            print(-1)
            exit()
    print(''.join(t))
