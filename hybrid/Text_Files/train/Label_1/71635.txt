t = int(input())

for i in range(t):
    s = input()
    if s.count('?') == 1:
        s = s.replace('?', 'a')
    elif list(s).count(s[0]) == len(s) - 1:
        s = s.replace('?', s[0])
    elif list(s).count(s[-1]) == len(s) - 1:
        s = s.replace('?', s[-1])
    else:
        s = "-1"
    print(s)
