n, m = map(int, input().split())
g = [input() for _ in range(n)]
for x in range(n):
    for y in range(m):
        if g[x][y] == '*': continue
        if all(g[x][i] == '.' for i in range(m)) or all(g[i][y] == '.' for i in range(n)):
             print("YES")
             print(x + 1, y + 1)
             exit()
print("NO")
