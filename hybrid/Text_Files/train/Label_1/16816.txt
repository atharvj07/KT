#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    int sum = 0, seat = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
        seat = std::max(seat, a[i]);
    }
    if (sum / 2 <= seat || sum % 2 == 0) {
        std::cout << 0 << std::endl;
        return 0;
    }
    std::vector<int> ans;
    ans.push_back(1);
    for (int i = 2; i <= n; i++) {
        if (a[i - 1] <= seat / 2) {
            ans.push_back(i);
        }
    }
    std::cout << ans.size() << std::endl;
    for (int i = 0; i < ans.size(); i++) {
        std::cout << ans[i] << " ";
    }
    std::cout << std::endl;
    return 0;
}
