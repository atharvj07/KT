import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        boolean possible = true;
        for (int i = 1; i < n; i++) {
            possible &= a[i] >= a[i - 1];
        }

        if (!possible) {
            System.out.println(0);
            return;
        }

        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);

        int mod = 1000000007;
        System.out.println(solve(0, a, dp, mod));
    }

    public static int solve(int i, int[] a, int[] dp, int mod) {
        if (i == a.length - 1) {
            return 1;
        }

        if (dp[i] != -1) {
            return dp[i];
        }

        int ans = 0;
        int l = a[i] - 1;
        int r = a[i + 1] + 1;

        for (int j = l; j < r; j++) {
            ans = (ans + solve(i + 1, a, dp, mod)) % mod;
        }

        dp[i] = ans;
        return ans;
    }
}
