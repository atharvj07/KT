n, m = map(int, input().split())
variables = {}

for _ in range(n):
    name, _, expression = input().partition(' := ')
    variables[name] = expression

min_val = ['0'] * m
max_val = ['0'] * m


def evaluate(expression, minmax):
    if expression[0] in '01':
        return int(expression, 2)

    x, op, y = expression.split()

    if x == '?':
        x_val = int(''.join(minmax), 2)
    else:
        x_val = evaluate(variables[x], minmax)

    if y == '?':
        y_val = int(''.join(minmax), 2)
    else:
        y_val = evaluate(variables[y], minmax)

    if op == 'AND':
        return x_val & y_val
    elif op == 'OR':
        return x_val | y_val
    elif op == 'XOR':
        return x_val ^ y_val


for bit in range(m):
    min_val[bit] = '1'
    if evaluate(variables[list(variables.keys())[-1]], min_val) > evaluate(variables[list(variables.keys())[-1]], ['0'] * m):
        min_val[bit] = '0'

for bit in range(m):
    max_val[bit] = '1'
    if evaluate(variables[list(variables.keys())[-1]], ['1'] * m) < evaluate(variables[list(variables.keys())[-1]], max_val):
        max_val[bit] = '0'

print(''.join(min_val))
print(''.join(max_val))
