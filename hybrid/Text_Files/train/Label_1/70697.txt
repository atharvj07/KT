import sys

s = 'abcdefghijklmnopqrstuvwxyz'

for test in range(int(input())):
    n, m = map(int, input().split())
    t = input()
    p = list(map(int, input().split()))
    ans = [0] * 26
    for i in range(m + 1):
        for j in range(min(n, p[i - 1] if i > 0 else n)):
            ans[s.index(t[j])] += 1
    print(*ans)
