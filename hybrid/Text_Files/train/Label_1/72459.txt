n = int(input())
ds = [[[0] for x in range(n)] for y in range(2)]
for i in range(n):
    l,r=map(int,input().split())
    ds[0][i][l] = i
    ds[1][r][i] = i
cnt, l = 0, 0
for i in range(n):
    if not ds[0][i]:
        cnt+=1
        l = i
        break
r = l
if cnt == 1:
    while r:
        l, r = r, ds[1][r][l]
        if r:
            ds[1][r][l] = ds[1][l][r] = 0
else:
    while r:
        l, r = r, ds[1][r][l]
        if r:
            ds[1][r][l] = ds[1][l][r] = 0
    r = l
    while r:
        l, r = r, ds[1][r][l]
        if r:
            ds[1][r][l] = ds[1][l][r] = 0
for i in range(n):
    print(*ds[0][i])
