#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t, n, x, a;
  cin >> t;
  while (t--) {
    cin >> n >> x;
    priority_queue<int> q;
    ll sum = 0;
    for (int i = 0; i < n; i++) {
      cin >> a;
      sum += a;
      q.push(a);
    }
    if (sum < x * n)
      cout << 0 << endl;
    else {
      ll need = sum % n;
      sum -= need;
      int ans = 0;
      while (!q.empty()) {
        if (q.top() + need >= x) {
          need = 0;
          ans++;
        } else
          need -= x - q.top(), q.pop();
        if (need == 0) break;
      }
      cout << ans << endl;
    }
  }
  return 0;
}
