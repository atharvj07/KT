s=input()
a=[0]*26
for c in s:
    if c!='?':a[ord(c)-ord('A')]+=1
if s.count('?')+max(a)<26:print(-1)
else:
    ans=[chr(i+ord('A'))for i in a]
    for c in s:
        if c=='?':
            for i in range(26):
                if a[i]==0:
                    ans[-1]=chr(i+ord('A'))
                    a[i]+=1
                    break
    print(''.join(ans))
