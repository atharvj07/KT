#include <algorithm>
#include <iostream>
#include <vector>
#include <cmath>

using namespace std;
using ll = long long;
const ll MOD = 1000000007;

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n / 2);
  for (ll i = 0; i < n / 2; i++) cin >> a[i];
  vector<ll> b;
  bool ok = true;
  for (ll i = 0; i < n / 2; i++) {
    ll x = sqrt(a[i]);
    if (x * x == a[i]) {
      b.push_back(x);
      a[i] = x * x;
    } else {
      ok = false;
      break;
    }
  }
  if (!ok) {
    cout << "No" << endl;
    return 0;
  }
  cout << "Yes" << endl;
  reverse(a.begin(), a.end());
  for (ll i = 0; i < n / 2; i++) a[i] += b[i] * b[i];
  for (ll i = 0; i < n; i++) cout << a[i] << " ";
  cout << endl;
  return 0;
}
