#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    string s;
    cin >> n >> s;
    int ans = 0;
    int r = 0;
    for (int i = 0; i < n; i++)
        if (i == 0 || i == n - 1 || s[i] != s[i - 1] && s[i] != s[i + 1]) r = 0;
        else if (s[i] == s[i - 1])
            r++, ans = max(ans, r);
        else {
            ans = max(ans, r), r = 0;
            if (s[i] == s[i + 1]) r++;
        }
    cout << ans << '\n';
}
