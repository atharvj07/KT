#include<bits/stdc++.h>
using namespace std;
const int maxn=1e5+5,inf=0x3f3f3f3f;
int n,m,u[maxn],v[maxn],idx,ans,fa[maxn];
vector<int> e[maxn];
bool vis[maxn];
inline int find(int x){return fa[x]==x?x:fa[x]=find(fa[x]);}
inline void uni(int a,int b){fa[find(a)]=find(b);}
int main(){
    ios::sync_with_stdio(false);cin.tie(0);
    cin>>n>>m;
    while (m--) {cin>>u[idx]>>v[idx];idx++;}
    iota(fa+1,fa+n+1,1);
    for (int i=0;i<idx;++i){
        int f1=find(u[i]),f2=find(v[i]);
        if (f1!=f2){
            ans++;
            uni(f1,f2);
        }
    }
    for (int i=1;i<=n;++i){
        if (!vis[find(i)]) vis[find(i)]=1,e[ans].push_back(find(i));
    }
    if (e[ans].size()>1) cout<<-1;
    else cout<<ans;
    return 0;
}
