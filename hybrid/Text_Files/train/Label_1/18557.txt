#include <iostream>
#include <cmath>

int main() {
  int n, c;
  std::cin >> n >> c;
  int p[n], t[n];
  int limaks = 0, radewooshs = 0;
  for (int i = 0; i < n; i++) {
    std::cin >> p[i] >> t[i];
    limaks += std::max(0, p[i] - c * t[i]);
    radewooshs += std::max(0, p[n - 1 - i] - c * t[n - 1 - i]);
  }
  if (limaks > radewooshs)
    std::cout << "Limak" << std::endl;
  else if (radewooshs > limaks)
    std::cout << "Radewoosh" << std::endl;
  else
    std::cout << "Tie" << std::endl;
}
