#include <algorithm>
#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  sort(v.begin(), v.end());
  string s = v[0].substr(1);
  int len = s.size();
  for (int i = 1; i < n; i++) {
    string t = v[i].substr(1);
    int l = min(len, int(t.size()));
    while (l > 0 && s.substr(len - l) == t.substr(t.size() - l)) {
      l--;
    }
    len = l;
  }
  for (int i = 0; i < n; i++) {
    cout << "#" << s.substr(0, min(len, int(v[i].size()) - 1)) << endl;
  }
  return 0;
}
