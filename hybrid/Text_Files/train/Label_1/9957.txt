#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> handedness(n);
  for (int i = 0; i < n; i++) {
    cin >> handedness[i];
  }
  for (int i = 0; i < n / 2; i++) {
    bool found = false;
    for (int j = i + 1; j < n; j++) {
      if (handedness[i] == 'L' && handedness[j] == 'R' && i + 1 != j) {
        cout << i + 1 << " " << j + 1 << endl;
        found = true;
        break;
      } else if (handedness[i] == 'R' && handedness[j] == 'L' && i + 1 != j) {
        cout << j + 1 << " " << i + 1 << endl;
        found = true;
        break;
      }
    }
    if (!found) {
      for (int j = i + 1; j < n; j++) {
        if (handedness[i] == handedness[j] && i + 1 != j) {
          cout << i + 1 << " " << j + 1 << endl;
          found = true;
          break;
        }
      }
    }
  }
  return 0;
}
