import java.util.*;

public class Main {
    private static final int mod = (int) 1e9 + 7;

    private static class Point {
        long x;
        long y;

        Point(long x, long y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Vector {
        Point a;
        Point b;

        Vector(Point a, Point b) {
            this.a = a;
            this.b = b;
        }

        // Compute the cross product of two vectors
        long cross(Vector other) {
            return (b.x - a.x) * (other.b.y - other.a.y) - (b.y - a.y) * (other.b.x - other.a.x);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read the number of polygons
        int n = scanner.nextInt();

        // Read the polygons
        List<List<Point>> polygons = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int k = scanner.nextInt();
            List<Point> polygon = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                long x = scanner.nextLong();
                long y = scanner.nextLong();
                polygon.add(new Point(x, y));
            }
            polygons.add(polygon);
        }

        // Read the number of tasks
        int q = scanner.nextInt();

        // Process the tasks
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;

            // Compute the Minkowski sum of the polygons
            List<Point> minkowskiSum = minkowskiSum(polygons.subList(l, r + 1));

            // Compute the number of vertices in the Minkowski sum
            int numVertices = minkowskiSum.size();

            // Print the number of vertices
            System.out.println(numVertices);
        }
    }

    private static List<Point> minkowskiSum(List<List<Point>> polygons) {
        if (polygons.isEmpty()) {
            return new ArrayList<>();
        }

        // Initialize the Minkowski sum with the first polygon
        List<Point> minkowskiSum = new ArrayList<>(polygons.get(0));

        // Compute the Minkowski sum of the remaining polygons
        for (List<Point> polygon : polygons.subList(1, polygons.size())) {
            // Translate the polygon to the origin
            List<Point> translatedPolygon = new ArrayList<>();
            for (Point point : polygon) {
                translatedPolygon.add(new Point(point.x - polygons.get(0).get(0).x, point.y - polygons.get(0).get(0).y));
            }

            // Compute the Minkowski sum of the Minkowski sum and the translated polygon
            minkowskiSum = merge(minkowskiSum, translatedPolygon);
        }

        // Return the Minkowski sum
        return minkowskiSum;
    }

    private static List<Point> merge(List<Point> minkowskiSum, List<Point> polygon) {
        // Create a new list to store the merged polygon
        List<Point> mergedPolygon = new ArrayList<>();

        // Initialize the indices of the two polygons
        int i = 0;
        int j = 0;

        // Merge the two polygons
        while (i < minkowskiSum.size() && j < polygon.size()) {
            // Compute the cross product of the two vectors
            long crossProduct = new Vector(minkowskiSum.get(i), polygon.get(j)).cross(new Vector(minkowskiSum.get(i), minkowskiSum.get((i + 1) % minkowskiSum.size())));

            // If the cross product is positive, then the two vectors are pointing in the same direction and we can add the second vector to the merged polygon
            if (crossProduct > 0) {
                mergedPolygon.add(polygon.get(j));
                j++;
            }
            // Otherwise, we can add the first vector to the merged polygon
            else {
                mergedPolygon.add(minkowskiSum.get(i));
                i = (i + 1) % minkowskiSum.size();
            }
        }

        // Add the remaining vectors to the merged polygon
        while (i < minkowskiSum.size()) {
            mergedPolygon.add(minkowskiSum.get(i));
            i++;
        }
        while (j < polygon.size()) {
            mergedPolygon.add(polygon.get(j));
            j++;
        }

        // Return the merged polygon
        return mergedPolygon;
    }
}
