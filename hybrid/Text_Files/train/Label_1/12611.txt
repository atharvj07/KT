#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  vector<int> dp(s.size() + 1);
  dp[0] = 0;
  for (int i = 0; i < s.size(); i++) {
    dp[i + 1] = dp[i];
    if (s[i] == 'e') dp[i + 1] = max(dp[i + 1], dp[i] + 1);
    if (s[i] == 'd' && dp[i] > 0) dp[i + 1] = max(dp[i + 1], dp[i] + 1);
    if (s[i] == 'i' && dp[i] > 1) dp[i + 1] = max(dp[i + 1], dp[i] + 1);
    if (s[i] == 'h' && dp[i] > 2) dp[i + 1] = max(dp[i + 1], dp[i] + 1);
  }
  cout << (dp[s.size()] >= 5 ? "YES" : "NO") << endl;
  return 0;
}
