#include <iostream>

using namespace std;

int main() {
  long long n, sx, sy, dx, dy, t;
  cin >> n >> sx >> sy >> dx >> dy >> t;
  long long x = sx;
  long long y = sy;
  for (long long i = 0;; i++) {
    long long nx = (x + dx - 1) % n + 1;
    long long ny = (y + dy - 1) % n + 1;
    long long disx = (nx - x + n) % n;
    long long disy = (ny - y + n) % n;
    long long sum = disx + disy;
    if (t <= i * n) {
      if (t % n > sum) {
        x += dx * (t % n - sum);
        y += dy * (t % n - sum);
        dx += t % n - sum;
        dy += t % n - sum;
      }
      break;
    }
    t -= i * n;
    x += dx * n;
    y += dy * n;
    dx += n;
    dy += n;
  }
  cout << x << " " << y << endl;
  return 0;
}
