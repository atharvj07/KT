import queue

for _ in range(int(input())):
    n, u, v = map(int, input().split())
    a = list(map(int, input().split()))
    a.insert(0, 0)
    q = queue.Queue()
    q.put((1, 0, n, u, v))
    ans = -1
    while not q.empty():
        l, r, c, i, j = q.get()
        if l == r:
            ans = c
            break
        if l < r:
            x = a[l + 1]
            if x < i or x > j:
                q.put((l + 1, r, c + u, i - 1, j + 1))
            elif x >= i and x <= (i + j) // 2:
                q.put((l + 1, r, c + j - x + u, i, j))
            else:
                q.put((l + 1, r, c + x - i + u, i, j))
        if l > r:
            x = a[r + 1]
            if x < i or x > j:
                q.put((l, r + 1, c + v, i - 1, j + 1))
            elif x >= i and x <= (i + j) // 2:
                q.put((l, r + 1, c + i - x + v, i, j))
            else:
                q.put((l, r + 1, c + j - x + v, i, j))
    print(ans)
