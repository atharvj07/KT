import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        String[] str = s.nextLine().split(" ");
        if (str.length != 5 || !str[0].equals("?") || !str[3].equals("-") || str[4].length() > 6) {
            System.out.println("Impossible");
            return;
        }
        boolean exist = false;
        int n = Integer.parseInt(str[4]);
        for (int i = 1; i <= n / 2; i++) {
            for (int j = i + 1; j <= n; j++) {
                int k = n - i + j;
                if (k > 0 && k != j) {
                    System.out.println("Possible");
                    System.out.println(i + " + " + j + " - " + k + " + " + k + " = " + n);
                    exist = true;
                    break;
                }
            }
            if (exist) {
                break;
            }
        }
        if (!exist) {
            System.out.println("Impossible");
        }
    }
}
