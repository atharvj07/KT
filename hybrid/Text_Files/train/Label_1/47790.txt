import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Rect> rects = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            rects.add(new Rect(Integer.parseInt(s[0]), Integer.parseInt(s[1]), Integer.parseInt(s[2]), Integer.parseInt(s[3])));
        }
        System.out.println(minSpells(rects));
    }

    static int minSpells(List<Rect> rects) {
        rects.sort((a, b) -> a.x1 - b.x1);
        int ans = 0;
        int end = 0;
        for (Rect rect : rects) {
            if (rect.x1 > end) {
                ans++;
                end = rect.x2;
            } else {
                end = Math.max(end, rect.x2);
            }
        }
        return ans;
    }

    static class Rect {
        int x1, x2, y1, y2;

        Rect(int x1, int x2, int y1, int y2) {
            this.x1 = x1;
            this.x2 = x2;
            this.y1 = y1;
            this.y2 = y2;
        }
    }
}
