#include<bits/stdc++.h>
using namespace std;
int n;
string s;
int main(){
	cin>>n>>s;
	for(int i=0;i<n;i++){
		bool ok=1;
		for(int j=0;j<26;j++){
			int cnt=0;
			for(int k=i;k<n;k++)if(s[k]-'a'==j)cnt++;
			if(cnt>(n-i)/2){
				ok=0;
				break;
			}
		}
		if(ok){
			cout<<"YES"<<"\n";
			cout<<s.substr(i,n-i)<<'\n';
			return 0;
		}
	}
	cout<<"NO"<<"\n";
	return 0;
}
