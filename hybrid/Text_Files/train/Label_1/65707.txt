def solve_permutation_cases(q, test_cases):
    results = []
    for n, perm in test_cases:
        visited = [False] * n
        for i in range(n):
            if not visited[i]:
                min_pos = i
                for j in range(i, n):
                    if perm[j] < perm[min_pos]:
                        min_pos = j
                    visited[j] = True
                    if j < n - 1 and perm[j] < perm[j + 1]:
                        break
                perm[i:min_pos+1] = sorted(perm[i:min_pos+1])
        results.append(perm)
    return results

q = int(input())
test_cases = []
for _ in range(q):
    n = int(input())
    perm = list(map(int, input().split()))
    test_cases.append((n, perm))

results = solve_permutation_cases(q, test_cases)
for result in results:
    print(" ".join(map(str, result)))
