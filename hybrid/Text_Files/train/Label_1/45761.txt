import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        char[] op = new char[n - 1];
        for (int i = 0; i < n; i++) {
            if (i < n - 1) {
                a[i] = in.nextLong();
                op[i] = in.next().charAt(0);
            } else {
                a[i] = in.nextLong();
            }
        }

        long max = Long.MIN_VALUE;
        long sum = a[0];
        for (int i = 1; i < n; i++) {
            if (op[i - 1] == '+') {
                sum += a[i];
            } else {
                max = Math.max(max, sum);
                sum = a[i];
            }
        }
        max = Math.max(max, sum);
        System.out.println(max);
    }
}
