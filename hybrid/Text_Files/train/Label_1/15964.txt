#include <bits/stdc++.h>
using namespace std;
const int N = 205, K = 205;
int n, k, a[N], f[N][2][K], u[N], v[N];
vector<int> g[N];
void dfs1(int x, int p) {
    for (auto i : g[x]) {
        if (i != p) {
            dfs1(i, x);
            for (int st = k; st >= 1; st--) {
                f[x][1][st] = max({f[x][1][st], f[i][0][st - 1] + a[x], f[i][1][st - 1] + a[x]});
            }
            for (int st = k; st >= 1; st--) {
                f[x][0][st] = max({f[x][0][st], f[i][0][st], f[i][1][st]});
            }
        }
    }
}
void dfs2(int x, int p) {
    for (auto i : g[x]) {
        if (i != p) {
            for (int st = k; st >= 1; st--) {
                if (st == 1) {
                    f[i][1][st] = max({f[i][1][st], f[x][0][st - 1] + a[i], f[x][1][st - 1] + a[i]});
                } else {
                    f[i][1][st] = max({f[i][1][st], f[x][0][st - 1] + a[i], f[x][1][st - 1] + a[i], f[x][0][st] + a[i]});
                }
            }
            for (int st = k; st >= 1; st--) {
                f[i][0][st] = max({f[i][0][st], f[x][0][st], f[x][1][st]});
            }
            dfs2(i, x);
        }
    }
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        cin >> u[i] >> v[i];
        g[u[i]].push_back(v[i]);
        g[v[i]].push_back(u[i]);
    }
    dfs1(1, 1);
    dfs2(1, 1);
    int ans = 0;
    for (int x = 0; x <= k; x++)
        ans = max(ans, max(f[1][0][x], f[1][1][x]));
    cout << ans << endl;
    return 0;
}
