def check_winner(m, n, mat):
    for i in range(n):
        for j in range(m):
            if(mat[i][j] != 0):
                return 'Ashish' if (i + j) % 2 else 'Jeel'

    return 'Jeel'

for _ in range(int(input())):
    n, m = map(int, input().split())
    mat = [list(map(int, input().split())) for _ in range(n)]
    print(check_winner(m, n, mat))
