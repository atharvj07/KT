import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(reader.readLine());

        for (int i = 0; i < q; i++) {
            String[] parts = reader.readLine().split(" ");
            int n = Integer.parseInt(parts[0]);
            int m = Integer.parseInt(parts[1]);

            List<Customer> customers = new ArrayList<>();

            for (int j = 0; j < n; j++) {
                String[] customerParts = reader.readLine().split(" ");
                customers.add(new Customer(Integer.parseInt(customerParts[0]),
                        Integer.parseInt(customerParts[1]), Integer.parseInt(customerParts[2])));
            }

            System.out.println(isPossible(m, customers) ? "YES" : "NO");
        }
    }

    public static boolean isPossible(int m, List<Customer> customers) {
        int time = 0;
        int temp = m;

        for (Customer customer : customers) {
            if (customer.arrivalTime < time) {
                return false;
            }

            int steps = (int) Math.ceil((customer.minTemp - temp) / 2.0);
            time += steps;
            temp += steps;

            if (temp < customer.minTemp) {
                return false;
            }

            steps = (int) Math.ceil((temp - customer.maxTemp) / 2.0);
            time += steps;
            temp -= steps;

            if (temp > customer.maxTemp) {
                return false;
            }

            time = customer.arrivalTime;
            temp = customer.minTemp;
        }

        return true;
    }

    public static class Customer {
        int arrivalTime;
        int minTemp;
        int maxTemp;

        public Customer(int arrivalTime, int minTemp, int maxTemp) {
            this.arrivalTime = arrivalTime;
            this.minTemp = minTemp;
            this.maxTemp = maxTemp;
        }
    }
}
