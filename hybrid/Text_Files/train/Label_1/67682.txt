n = int(input())
a = list(map(int, input().split()))

coupon = 0
for i in range(n):
    if a[i] % 2 == 0:
        continue
    else:
        if coupon > 0:
            coupon -= 1
        else:
            if i < n - 1:
                a[i+1] -= 1
                if a[i+1] < 0:
                    print("NO")
                    break
            else:
                print("NO")
                break
        coupon += 1
else:
    print("YES" if coupon % 2 == 0 else "NO")
