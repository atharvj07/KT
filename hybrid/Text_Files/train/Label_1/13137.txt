#include <bits/stdc++.h>
using namespace std;
int main(){
    int n; cin>>n;
    vector<long long> a(n);
    for(int i=0;i<n;i++) cin>>a[i];
    vector<int> ans(n,-1);
    vector<long long> dp(n+1,LLONG_MIN); 
    for(int i=n-1;i>=0;i--){
        dp[i]=max(dp[i+1],a[i]);
        if(ans[i]==-1 && (i==0 || a[i]>=dp[i+1]/2)) ans[i]=1;
        else if(ans[i]==-1 && dp[i+1]!=LLONG_MIN) ans[i]=1+ans[i+1];
    }
    for(auto x:ans) cout<<x<<" ";
}
