#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, a, b;
    cin >> n >> a >> b;
    vector<pair<int, int>> v(n);
    for (auto& [x, y] : v) {
        cin >> x >> y;
        if (x > y) swap(x, y);
    }
    sort(v.begin(), v.end(), [](auto& x, auto& y) { return x.second < y.second; });
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (v[i].first <= a && v[j].first <= b && v[i].second + v[j].second <= b) {
                ans = max(ans, v[i].first * v[i].second + v[j].first * v[j].second);
            }
            swap(v[j].first, v[j].second);
            if (v[i].first <= a && v[j].first <= b && v[i].second + v[j].second <= b) {
                ans = max(ans, v[i].first * v[i].second + v[j].first * v[j].second);
            }
        }
    }
    cout << ans << '\n';
}
