import sys
input = sys.stdin.readline
n,m = map(int,input().split())
a = list(map(int,input().split()))
b = []
for i in range(m):
  x,y = map(int,input().split())
  b.append((x-1,y-1,x+y))
b.sort(key = lambda x: x[2])
ans = 0
vis = [0]*m
for x,y,z in b:
  f = 2
  while True:
    if z%f == 0 and a[x]%f == 0 and a[y]%f == 0:
      a[x] /= f
      a[y] /= f
      z /= f
      ans += 1
    else:
      break
print(ans)
