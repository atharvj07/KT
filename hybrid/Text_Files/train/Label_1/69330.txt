n=int(input())
a=[int(x) for x in input().split()]
ans,d={},{}
for x in a:
    if x in d: d[x]+=1
    else: d[x]=1
i=1
while d:
    for x in list(d):
        if x in d and d[x]>0:
            if x not in ans:
                ans[x]=i
            d[x]-=1
            if d[x]==0: del d[x]
    i+=1
print(i-1)
