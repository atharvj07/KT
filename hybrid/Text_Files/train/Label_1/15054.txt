#include <iostream>
#include <vector>
#include <cstdio>
#include <string>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    if (n == 1)
      std::cout << 8 << std::endl;
    else {
      std::cout << 9;
      for (int i = 0; i < n - 1; ++i)
        std::cout << 8;
      std::cout << std::endl;
    }
  }
  return 0;
}
