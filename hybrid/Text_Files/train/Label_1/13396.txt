#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  ll t;
  cin >> t;
  while (t--) {
    ll n, k;
    cin >> n >> k;
    --k;
    if (k >= (1LL << (2 * n - 2)))
      cout << "-1\n";
    else {
      vector<ll> perm(n + 1);
      iota(perm.begin(), perm.end(), 0);
      ll remain = k;
      for (ll i = n - 1; i >= 1; i--) {
        ll taken;
        if (i == n - 1)
          taken = remain / i;
        else
          taken = (remain + 1) / i;
        ll unsorted = n - i;
        ll ptr = min(unsorted, taken);
        for (ll j = 0; j < ptr; j++) swap(perm[i + j], perm[n - j]);
        remain -= taken * i;
      }
      for (ll i = 1; i <= n; i++) cout << perm[i];
      cout << "\n";
    }
  }
  return 0;
}
