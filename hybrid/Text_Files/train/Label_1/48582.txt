import java.util.*;
import java.lang.*;

class Main {
    static int h,w,n;
    static int[] a;
    static int[][] c;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        h = sc.nextInt();
        w = sc.nextInt();
        n = sc.nextInt();
        a = new int[n+1];
        for (int i=1; i<=n; i++) {
            a[i] = sc.nextInt();
        }
        c = new int[h+2][w+2];
        int x=1, y=1, color=1;
        for (int i=1; i<=n; i++) {
            for (int j=0; j<a[color]; j++) {
                c[x][y++] = color;
                if (y > w) { y=1; x++; }
            }
            color++;
        }
        for (int i=1; i<=h; i++) {
            for (int j=1; j<=w; j++) {
                System.out.print(c[i][j] + " ");
            }
            System.out.println();
        }
    }
}
