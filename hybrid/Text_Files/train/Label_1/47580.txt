import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int dist = Math.abs(a - b);
        long ans = dist * (dist + 1) / 2;
        System.out.println(ans);
    }
}
