#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, id, a[100000] = {};
  cin >> n >> m;
  while (m--) {
    cin >> id;
    if (id > 0) a[id - 1] = 1;
    else a[-id - 1] = 0;
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (a[i]) cnt++;
  }
  cout << cnt << "\n";
  for (int i = 0; i < n; i++) {
    if (a[i]) cout << i + 1 << " ";
  }
  return 0;
}
