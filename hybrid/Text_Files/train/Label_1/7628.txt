#include <iostream>
#include <string>

using namespace std;

int main() {
  char direction;
  string s;
  cin >> direction >> s;
  for (int i = 0; i < s.length(); i++) {
    if (direction == 'R') {
      if (s[i] == 'a')
        s[i] = 'z';
      else if (s[i] == ';')
        s[i] = 'p';
      else if (s[i] == ']')
        s[i] = 'm';
      else
        s[i]--;
    } else {
      if (s[i] == 'z')
        s[i] = 'a';
      else if (s[i] == 'p')
        s[i] = ';';
      else if (s[i] == 'm')
        s[i] = ']';
      else
        s[i]++;
    }
  }
  cout << s << endl;
  return 0;
}
