a, b, c, d = map(int, input().split())
if a + c + d > b or abs(a - b) > 1 or abs(c - (b - a)) > 1:
    print('NO')
else:
    print('YES')
    for _ in range(b - a):
        print(1, end=" ")
    for _ in range(a):
        print(0, end=" ")
    for _ in range(b - a):
        print(2, end=" ")
    for _ in range(c - (b - a) + 1):
        print(3, end=" ")
    for _ in range(d - (c - (b - a) + 1)):
        print(3, end=" ")
