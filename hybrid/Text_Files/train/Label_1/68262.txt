for _ in range(int(input())):
    s = input()
    stack = [s[0]]
    valid = 1
    for i in range(1, len(s)):
        if s[i] == s[0]: stack.append(s[i])
        else:
            if len(stack) == 0: valid = 0; break
            stack.pop()
    if len(stack) == 0 and valid == 1: print('YES')
    else: print('NO')
