#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<int> u;
    for (int i = 0; i < n; i++) {
        while (i < n - 1 && v[i] == v[i + 1]) { v[i]++; i++; }
        u.push_back(v[i]);
    }
    cout << u.size() << endl;
    for (int i = 0; i < u.size(); i++) cout << u[i] << " ";
    cout << endl;
}
