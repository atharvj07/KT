#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, a, b;
    cin >> n >> a >> b;
    vector<long long> x(n);
    for (int i = 0; i < n; i++) cin >> x[i];
    for (auto &i : x) {
        long long ans = x[i], c = a / b;
        if (a % b == 0)
            c--;
        cout << ans - max(0LL, x[i] - c) << " ";
    }
}
