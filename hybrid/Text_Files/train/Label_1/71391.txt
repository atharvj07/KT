from collections import Counter
from math import prod

MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def divisors_product(n, factors):
    factors_count = Counter(factors)
    total_divisors = prod([(2*count + 1) for count in factors_count.values()]) % (MOD - 1)
    result = 1
    for factor, count in factors_count.items():
        result = (result * power(factor, (count * total_divisors) // 2, MOD)) % MOD
    return result

m = int(input().strip())
factors = list(map(int, input().strip().split()))

print(divisors_product(m, factors))
