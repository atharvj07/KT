import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int maxValue = 0, ans_x = 0, ans_y = 0;
        for (int i = k - 1; i < n - k + 1; i++) {
            for (int j = k - 1; j < m - k + 1; j++) {
                int min = a[i][j];
                for (int l = -k + 1; l <= k - 1; l++) {
                    min = Math.min(min, a[i + l][j]);
                }
                for (int l = -k + 1; l <= k - 1; l++) {
                    min = Math.min(min, a[i][j + l]);
                }
                if (min > maxValue) {
                    maxValue = min;
                    ans_x = i + 1;
                    ans_y = j + 1;
                }
            }
        }
        System.out.println(ans_x + " " + ans_y);
    }
}
