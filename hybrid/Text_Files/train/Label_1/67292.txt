import sys

n = int(input())
days = set()
roads = [[False for _ in range(n+1)] for _ in range(n+1)]

for _ in range(n**2):
    h, v = map(int, sys.stdin.readline().split())
    if not roads[h][v]:
        days.add(len(days)+1)
        for x in range(n+1):
            roads[h][x] = True
            roads[x][v] = True

print(*days)
