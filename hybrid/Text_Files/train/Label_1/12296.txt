#include <bits/stdc++.h>
using namespace std;

struct data {  // count of ingredients (1-9)
    int a[10];
    void operator+=(const data &b) {
        for (int i = 1; i <= 9; ++i) { a[i] += b.a[i]; }
    }
} s[100005];

int main() {
    int n, m, p[2], t, x, tmp, ans = 0, sum[100005] = {0};
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; ++i) {
        scanf("%d", &t);
        for (int j = 1; j <= t; ++j) {
            scanf("%d", &x);
            s[i].a[x]++;
        }
    }
    for (int i = 1; i <= m; ++i) {
        scanf("%d%d", &x, &t);
        for (int j = 1; j <= t; ++j) {
            scanf("%d", &x);
            s[i + n].a[x]++;
        }
        sum[i + n] = x;
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = n + 1; j <= n + m; ++j) {
            tmp = 0;
            for (int k = 1; k <= 9; ++k) { tmp += min(s[i].a[k], s[j].a[k]); }
            if (tmp > ans) { ans = tmp; p[0] = i, p[1] = j; }
        }
    }
    printf("%d %d\n", p[0], p[1]);
    return 0;
}
