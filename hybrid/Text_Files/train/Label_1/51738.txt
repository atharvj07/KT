import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args)throws IOException{
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(br.readLine());
        int[][] p=new int[n][2];
        int[] ans=new int[n];
        for(int i=0;i<n;++i){
            String[] s=br.readLine().split(" ");
            p[i]=new int[]{Integer.parseInt(s[0]),Integer.parseInt(s[1])};
        }
        boolean ok=true;
        for(int i=0;i<n;++i){
            for(int j=i+1;j<n;++j){
                if(p[i][0]>p[j][0])ok=false;
                if(p[i][1]>p[j][1])ok=false;
            }
        }
        if(ok){
            System.out.println("YES");
            for(int i=0;i<n;++i){
                ans[p[i][0]-1]=i+1;
            }
            StringBuilder sb=new StringBuilder();
            for(int i:ans){
                sb.append(i+" ");
            }
            System.out.println(sb.toString());
        }else{
            System.out.println("NO");
            for(int i=0;i<n;++i){
                ans[p[i][0]-1]=i+1;
            }
            StringBuilder sb=new StringBuilder();
            for(int i:ans){
                sb.append(i+" ");
            }
            System.out.println(sb.toString());
            for(int i=0;i<n;++i){
                ans[p[i][1]-1]=i+1;
            }
            sb=new StringBuilder();
            for(int i:ans){
                sb.append(i+" ");
            }
            System.out.println(sb.toString());
        }
    }
}
