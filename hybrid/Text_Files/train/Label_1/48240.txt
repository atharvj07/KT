import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[][] a = new int[n][n];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        a[i][j] = sc.nextInt();
      }
    }
    int[][] dp = new int[n][n];
    dp[0][0] = a[0][0];
    for (int i = 1; i < n; i++) {
      dp[i][0] = a[i][0] + dp[i - 1][0];
    }
    for (int j = 1; j < n; j++) {
      dp[0][j] = a[0][j] + dp[0][j - 1];
    }
    for (int i = 1; i < n; i++) {
      for (int j = 1; j < n; j++) {
        dp[i][j] = a[i][j] + Math.max(dp[i - 1][j], dp[i][j - 1]);
      }
    }
    int max = dp[0][0];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        max = Math.max(max, dp[i][j]);
      }
    }
    System.out.println(max);
  }
}
