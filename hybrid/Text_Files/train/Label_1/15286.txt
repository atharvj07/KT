#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
#define long long long
typedef pair<long, pair<long, long>> p3l;
const long INF = 1e9;
const long MXN = 2e5 + 1;
vector<p3l> bucket[MXN];
vector<long> ans;
long n, m, a[MXN], pre[MXN], dp[MXN][2], choosen[MXN][2];
void input() {
  scanf("%lld", &n);
  for(long i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
  }
  scanf("%lld", &m);
  for(long i = 1; i <= m; i++) {
    long x, y;
    scanf("%lld %lld", &x, &y);
    ans.push_back(y);
    bucket[x].push_back({y, {i, y}});
  }
}
void preprocess() {
  long sz;
  for(long i = 1; i <= n; i++) {
    pre[i] = pre[i - 1] + a[i];
  }
  for(long i = 1; i <= n; i++) {
    sz = bucket[i].size();
    for(long j = 0; j < sz; j++) {
      auto [pos, id] = bucket[i][j].second;
      bucket[i][j].first = pre[pos];
    }
    sort(bucket[i].begin(), bucket[i].end());
  }
}
void solve() {
  preprocess();
  for(long i = 1; i <= n; i++) {
    for(long picked = 0; picked <= 1; picked++) {
      dp[i][picked] = -INF;
      if(picked) {
        dp[i][picked] = max(dp[i][picked], dp[i - 1][picked]);
      }
      if(!picked) {
        dp[i][picked] = max(dp[i][picked], a[i]);
        choosen[i][picked] = i;
      }
      for(auto [sum, id]: bucket[i]) {
        if(picked) {
          if(dp[i][picked] < dp[i - 1][0] + sum) {
            dp[i][picked] = dp[i - 1][0] + sum;
            choosen[i][picked] = id.second;
          }
        }
        if(!picked) {
          if(dp[i][picked] < sum) {
            dp[i][picked] = sum;
            choosen[i][picked] = id.second;
          }
        }
      }
    }
  }
  for(long i = n, which = 1; i >= 1; i--) {
    if(choosen[i][which] <= i) {
      printf("%lld ", a[choosen[i][which]]);
      which = !which;
    }
  }
}
int main() {
  input();
  solve();
  return 0;
}
