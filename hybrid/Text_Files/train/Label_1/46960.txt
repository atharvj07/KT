import java.util.*;
import java.io.*;
import java.lang.*;

public class Main {

    private static boolean possible(int[] path, int[] degree) {
        for (int i = 0; i < path.length; i++) {
            if (path[i] != -1) {
                if (path[i] == i || degree[i] != 2) {
                    return false;
                }
            }
        }
        return true;
    }

    public static void main(String[] argv) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] path = new int[n];
        int[] degree = new int[n];
        Arrays.fill(path, -1);
        for (int i = 0; i < m; i++) {
            int a = in.nextInt() - 1, b = in.nextInt() - 1;
            path[a] = b;
            path[b] = a;
            degree[a]++;
            degree[b]++;
        }
        if (!possible(path, degree)) {
            System.out.println("NO");
            return;
        }
        int k = 0;
        for (int i = 0; i < n; i++) {
            if (path[i] == -1) {
                k++;
            }
        }
        System.out.println("YES");
        System.out.println(k);
        for (int i = 0; i < n; i++) {
            if (path[i] == -1) {
                System.out.println((i + 1) + " " + (i + 1));
            }
        }
    }
}
