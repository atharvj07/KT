#include <bits/stdc++.h>
using namespace std;
int main() {
    int a, b, c, n, i, x;
    cin >> a >> b >> c >> n;
    vector<int> v(c + 5);
    for (i = 1; i <= n; i++) {
        cin >> x;
        v[x]++;
    }
    int ans = 0, cur = 0;
    for (i = 1; i < b; i++) cur += v[i];
    for (i = b; i <= c; i++) {
        cur += v[i] - v[i - b] + v[i + c - b];
        ans = max(ans, cur);
    }
    cout << ans << endl;
    return 0;
}
