#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> a, b, ans;

bool solve() {
  map<int, int> cnta, cntb;
  for (int x : a) cnta[x]++;
  for (int x : b) cntb[x]++;
  bool same = 1;
  for (int x = -100; x <= 100; x++) {
    if (cnta[x] != cntb[x]) {
      same = 0;
      break;
    }
  }
  return same;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    a.resize(n);
    b.resize(n);
    for (int &x : a) cin >> x;
    for (int &x : b) cin >> x;
    if (solve())
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
