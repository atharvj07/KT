#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll MOD = 1000000007;

int main() {
  int n, k;
  cin >> n >> k;

  vector<vector<ll>> dp(2, vector<ll>(n + 1));
  dp[0][1] = 1;
  for (int i = 2; i <= n; i++) {
    if (i <= k) {
      dp[0][i] = (i - 1) * dp[0][i - 1] % MOD;
    } else {
      dp[0][i] = (MOD + i - 1) * dp[0][i - 1] % MOD;
    }
  }

  dp[1][1] = 1;
  for (int i = 2; i <= n; i++) {
    dp[1][i] = dp[0][i];
    if (i - k - 1 >= 0) {
      dp[1][i] = (dp[1][i] - dp[0][i - k - 1] + MOD) % MOD;
    }
  }

  cout << dp[1][n] << '\n';

  return 0;
}
