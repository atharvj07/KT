#include<bits/stdc++.h>
using namespace std;

const int mxn = 100005;
int n, par[mxn];
vector<int>adj[mxn];

int main(){
    cin >> n;
    vector<int> hotel, mountain;
    for(int i = 0, x; i < n; i++){
        cin >> x;
        if(x) hotel.push_back(i + 1);
        else mountain.push_back(i + 1);
    }
    for(int i = 0, x; i < n; i++){
        cin >> x;
        if(x) adj[x].push_back(i + 1);
    }
    for(int x : hotel){
        int u = x;
        while(adj[u].size() == 1 and u != 1){
            par[u] = adj[u][0];
            u = adj[u][0];
        }
    }
    vector<int> path;
    for(int x : mountain) if(par[x]) path.push_back(x);
    reverse(path.begin(), path.end());
    cout << path.size() + 1 << endl;
    for(int x : path) cout << x << ' ';
    for(int x : hotel) cout << x << ' ';
    cout << endl;
    return 0;
}
