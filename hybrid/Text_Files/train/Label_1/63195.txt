import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]

c = [(a[i] + b[i]) % n for i in range(n)]
b.sort()

for i in range(n):
    c[i] = (c[i] - a[i]) % n

ans = [c[i] for i in range(n) if b[i] == c[i]]
c.sort()
for i in range(n):
    if b[i] != c[i]:
        ans.append(c[i])

print(*ans)
