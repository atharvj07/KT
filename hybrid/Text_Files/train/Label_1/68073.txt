import sys

n, x = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))

if sum(a) != x:
    print('NO')
    sys.exit()

for i in range(n-1):
    if a[i] > a[i+1]:
        print('NO')
        sys.exit()

for i in range(n-2):
    if a[i] == a[i+1] + a[i+2]:
        print('NO')
        sys.exit()

print('YES')
