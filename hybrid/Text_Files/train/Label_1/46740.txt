import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    private static int[] ans;
    private static List<Integer>[] graph;
    private static char[] letters;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        graph = new List[n];
        ans = new int[n];
        for (int i = 0; i < n; i++) graph[i] = new ArrayList<>();
        letters = new char[n - 1];
        StringTokenizer tokenizer;
        for (int i = 0; i < n - 1; i++) {
            tokenizer = new StringTokenizer(br.readLine());
            int p = Integer.parseInt(tokenizer.nextToken()) - 1;
            char c = tokenizer.nextToken().charAt(0);
            graph[i + 1].add(p);
            graph[p].add(i + 1);
            letters[i] = c;
        }
        dfs1(0, -1);
        dfs2(0, -1, 0);
        for (int i = 0; i < n; i++) System.out.print(ans[i] + " ");
    }

    private static int[] dfs1(int u, int p) {
        int[] dp = new int[26];
        for (int v : graph[u]) {
            if (v == p) continue;
            int[] dpV = dfs1(v, u);
            for (int i = 0; i < 26; i++) dp[i] = Math.max(dp[i], dpV[i]);
            if (letters[v - 1] != '.') dp[letters[v - 1] - 'a'] += 2;
        }
        return dp;
    }

    private static void dfs2(int u, int p, int dpU) {
        for (int v : graph[u]) {
            if (v == p) continue;
            int[] dpV = dfs1(v, u);
            for (int i = 0; i < 26; i++) dpV[i] = Math.max(dpV[i], dpU);
            if (letters[v - 1] != '.') dpV[letters[v - 1] - 'a'] += 2;
            int max = 0;
            for (int i = 0; i < 26; i++) max = Math.max(max, dpV[i]);
            ans[v] = max;
            dfs2(v, u, dpV[letters[v - 1] - 'a']);
        }
    }
}
