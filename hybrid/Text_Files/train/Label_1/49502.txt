import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        int ans = 0;
        if (n > 0) {
            int cur = 1;
            int friend = 106;
            for (int i = 0; i < n; i++) {
                int dist1 = Math.abs(cur - a[i]);
                int dist2 = Math.abs(friend - a[i]);
                if (dist1 < dist2) {
                    ans += dist1;
                    cur = a[i];
                } else {
                    ans += dist2;
                    friend = a[i];
                }
            }
        }

        System.out.println(ans);
    }
}
