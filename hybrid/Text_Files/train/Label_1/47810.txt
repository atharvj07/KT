import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int q = sc.nextInt();
    while (q-- > 0) {
      long xp0 = sc.nextLong(), yp0 = sc.nextLong(), xp1 = sc.nextLong(), yp1 = sc.nextLong(),
          xp2 = sc.nextLong(), yp2 = sc.nextLong(), xp3 = sc.nextLong(), yp3 = sc.nextLong();
      long a1 =
          (xp1 - xp0) * (xp2 - xp0) + (yp1 - yp0) * (yp2 - yp0),
          a2 =
              (xp1 - xp0) * (xp3 - xp0) + (yp1 - yp0) * (yp3 - yp0);
      long b1 =
          (xp3 - xp2) * (xp2 - xp0) + (yp3 - yp2) * (yp2 - yp0),
          b2 =
              (xp3 - xp2) * (xp1 - xp0) + (yp3 - yp2) * (yp1 - yp0);
      if (a1 * a2 < 0 && b1 * b2 < 0) System.out.println("1");
      else System.out.println("0");
    }
  }
}
