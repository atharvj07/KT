import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int w = sc.nextInt();
        int[] wList = new int[n];
        for (int i = 0; i < n; i++) {
            wList[i] = sc.nextInt();
        }
        long[][] dp = new long[n + 1][w + 1];
        dp[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= w; j++) {
                dp[i + 1][j] += dp[i][j];
                if (j + wList[i] <= w) {
                    dp[i + 1][j + wList[i]] += dp[i][j];
                }
                dp[i + 1][j] %= 1000000007;
            }
        }
        System.out.println(dp[n][w]);
    }
}
