import java.util.*;
public class Main {
    static int[][] pic;
    static int a, b;
    static TreeSet<Integer> ans = new TreeSet<>();
    static int minx, miny;
    static boolean ok(int x, int y) {
        int ax = a / x, bx = b / y;
        TreeSet<Integer> tmp = new TreeSet<>();
        for (int i = 0; i < ax; i++) {
            for (int j = 0; j < bx; j++) {
                int now = 0;
                for (int p = 0; p < x; p++) {
                    for (int q = 0; q < y; q++) {
                        int px = i * x + p;
                        int py = j * y + q;
                        if (pic[px][py] == 0) now *= 2; else now = now * 2 + 1;
                    }
                }
                tmp.add(now);
            }
        }
        if (tmp.size() == ax * bx) return true;
        return false;
    }
    static void dfs(int x) {
        if (x * x > a || x * x > b) {
            ans.add(x * x);
        } else {
            dfs(x + 1);
            if (a % x == 0 && ok(x, b / x)) {
                ans.add(x * (b / x));
                dfs(x + 1);
            }
            if (b % x == 0 && ok(a / x, x)) {
                ans.add(x * (a / x));
                dfs(x + 1);
            }
        }
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        a = in.nextInt();
        b = in.nextInt();
        pic = new int[a + 1][b + 1];
        for (int i = 1; i <= a; i++) {
            String s = in.next();
            for (int j = 1; j <= b; j++) {
                if (s.charAt(j - 1) >= 'A' && s.charAt(j - 1) <= 'Z') {
                    pic[i][j] = 1;
                }
            }
        }
        dfs(1);
        System.out.println(ans.size());
        int best = ans.first();
        int canx = (int)Math.sqrt(best), cany = best / canx;
        System.out.println(canx + " " + cany);
    }
}
