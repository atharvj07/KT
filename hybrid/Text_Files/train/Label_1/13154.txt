#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e5 + 5;

int n, m, a[MAXN], f[MAXN];

int main() {
  string s, t;
  cin >> s >> t;
  n = s.size();
  m = t.size();
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == t[0]) { f[i] = f[i - 1] + 1; }
    else { f[i] = f[i - 1]; }
  }
  int ans = n + 1;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == t[m - 1] && f[i - 1] == (m - 1)) {
      ans = min(ans, n - i);
    }
  }
  if (ans == n + 1) { ans = 0; }
  cout << ans << '\n';
  return 0;
}
