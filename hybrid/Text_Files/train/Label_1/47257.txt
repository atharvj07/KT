import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int min = 0;
        int max = 0;
        int[] color = new int[8];
        for (int i = 0; i < n; i++) {
            if (a[i] >= 3200) {
                max++;
            } else if (a[i] >= 2800) {
                color[7]++;
            } else if (a[i] >= 2400) {
                color[6]++;
            } else if (a[i] >= 2000) {
                color[5]++;
            } else if (a[i] >= 1600) {
                color[4]++;
            } else if (a[i] >= 1200) {
                color[3]++;
            } else if (a[i] >= 800) {
                color[2]++;
            } else if (a[i] >= 400) {
                color[1]++;
            } else {
                min++;
            }
        }

        if (max > 0) {
            min++;
            max += 7;
        } else {
            for (int i = 0; i < 8; i++) {
                if (color[i] > 0) {
                    min++;
                    max++;
                }
            }
        }

        System.out.println(min + " " + max);
    }
}
