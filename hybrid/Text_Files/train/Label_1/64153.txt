m, t, r = map(int, input().split())
w = list(map(int, input().split()))

if t < r:
    print(-1)
else:
    candles = 0
    burning = [0] * 305
    for i in range(m):
        required = r
        for j in range(w[i] - 1, w[i] - r - 1, -1):
            if burning[j]:
                required -= 1
        candles += required
        for j in range(required):
            for k in range(w[i] + j, w[i] + j + t):
                burning[k] = 1
    print(candles)
