import java.util.Scanner;

public class MagnetGroups {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        int groups = 1; // Start with one group for the first magnet
        char prevPole = scanner.nextLine().charAt(0); // Get the pole of the first magnet

        for (int i = 1; i < n; i++) {
            char currentPole = scanner.nextLine().charAt(0);

            if (prevPole != currentPole) {
                groups++; // Different poles, so a new group
            }

            prevPole = currentPole;
        }

        System.out.println(groups);
    }
}
