#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    string s;
    cin >> s;
    vector<int> pref(s.size());
    if (s[0] == '1') pref[0] = a + b;
    for (int i = 1; i < s.size(); i++) {
      pref[i] = pref[i - 1];
      if (s[i] == '1') pref[i] = min(pref[i], a + b);
      else pref[i] += a;
    }
    int mn = pref[0];
    for (int i = 1; i < s.size(); i++) {
      if (s[i - 1] == '0') pref[i] = min(pref[i], b);
      mn = min(mn, pref[i]);
    }
    cout << mn << endl;
  }
  return 0;
}
