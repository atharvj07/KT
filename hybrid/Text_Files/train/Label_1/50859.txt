import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long a = input.nextLong();
        long b = input.nextLong();
        long t = a / b;
        long a_mod_b = a % b;
        long ans;
        if (a_mod_b == 0)
            ans = t;
        else
            ans = t + 2;
        System.out.println(ans);
    }
}
