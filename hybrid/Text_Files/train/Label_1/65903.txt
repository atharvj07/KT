n,m,h=map(int,input().split())
f1=list(map(int,input().split()))
f2=list(map(int,input().split()))
t=[[0]*m for i in range(n)]
for i in range(n):
    t[i][:]=map(int,input().split())
l=[]
for i in range(n):
    temp1=[0]*m
    for j in range(m):
        temp1[j]=max(f1[j]-t[i][j],0)+max(f2[i]-t[i][j],0)
    l.append(temp1)
for tmp in l:
    print(" ".join(map(str,tmp)))
