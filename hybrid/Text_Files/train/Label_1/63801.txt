n = int(input())
a = list(map(int, input().split()))

def find_touch_points(sequence):
    if len(sequence) < 2:
        return "Finite", 0
    touch_points = 0
    for i in range(1, len(sequence)):
        if sequence[i-1] == 1 and sequence[i] == 3 or sequence[i-1] == 3 and sequence[i] == 1:
            return "Infinite", None
        if sequence[i-1] == 2 and sequence[i] == 3 or sequence[i-1] == 3 and sequence[i] == 2:
            touch_points += 4
        elif sequence[i-1] == 1 and sequence[i] == 2 or sequence[i-1] == 2 and sequence[i] == 1:
            touch_points += 3
    return "Finite", touch_points

result, points = find_touch_points(a)
print(result)
if points is not None:
    print(points)
