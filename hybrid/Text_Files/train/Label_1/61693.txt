#!/bin/python3

import sys
cards = input()
n = int(cards[0])
cards = cards[2:]
unique = set()

for c in cards:
    unique.add(c)

unique2 = set()
for c in cards:
    if c[0] in unique2:
        continue
    unique2.add(c[0])
    for i in unique:
        if i[0] == c[0]:
            unique.remove(i)

print(len(unique) + len(unique2))
