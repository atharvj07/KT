#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    for (int& x : a) {
        std::cin >> x;
        x = (x + 9) / 10;
    }
    std::sort(a.begin(), a.end());
    int ans = 0;
    for (int& x : a) {
        if (k >= x) {
            ans++;
            k -= x;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
