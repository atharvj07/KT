#include <bits/stdc++.h>

using namespace std;

const long long MOD = 1e9 + 7;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<long long> dp(n + 1, 0);
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < i; j++) {
      long long num = stoll(s.substr(j, i - j));
      if (j == 0 || s[j - 1] != '0') {
        dp[i] = (dp[i] + dp[j] * num) % MOD;
      }
    }
  }
  cout << dp[n] << endl;
  return 0;
}
