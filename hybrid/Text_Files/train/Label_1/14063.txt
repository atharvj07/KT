#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[2005], x = 0, y = 0;
    cin >> n;
    for (int i = 0; i < 2 * n; i++) { cin >> a[i]; }
    sort(a, a + 2 * n);
    bool f = 1;
    for (int i = 0; i < n; i++) { x += a[i]; }
    for (int i = n; i < 2 * n; i++) { y += a[i]; }
    if (x == y) {
        f = 0;
        cout << -1;
    }
    if (f) {
        for (int i = 0; i < 2 * n; i++) { cout << a[i] << " "; }
    }
    cout << endl;
}
