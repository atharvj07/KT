n, m = map(int, input().split())
r, c = map(int, input().split())
lx, rx = map(int, input().split())
grid = [list(input()) for _ in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            if abs(j - (c - 1)) <= lx and abs((i - 1) - (r - 1)) <= rx:
                ans += 1
print(ans)
