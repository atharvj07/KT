#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll M = 998244353;

int main() {
  int n;
  string s;
  cin >> n >> s;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<ll> t(n);
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a' || s[i] == 'e' || s[i] == 'h' || s[i] == 'r' || s[i] == 'd') {
      if (i)
        t[i] = t[i - 1];
      continue;
    }
    int j = i - 1;
    while (j >= 2 && t[j] >= a[j - 1]) { j -= 2; }
    if (s[j + 1] == s[i] && (j + 2 >= n || s[j + 2] != s[i])) {
      t[i] = t[j] + a[j + 1];
    }
  }
  cout << *min_element(t.begin(), t.end()) << endl;
}
