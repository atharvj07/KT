#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  if (a[n - 1] == 0 || a[n - 1] == 15) {
    cout << -1 << endl;
  } else if (a[n - 1] < 15) {
    cout << "UP" << endl;
  } else
    cout << "DOWN" << endl;
}
