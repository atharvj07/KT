import java.util.Scanner;

public class VasyaAndTheEasyProblem {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String s = scanner.nextLine();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][6];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 6; j++) {
                if (j == 0) {
                    dp[i][j] = dp[i + 1][j];
                } else if (s.charAt(i) == 'h') {
                    dp[i][j] = Math.min(dp[i + 1][j - 1], dp[i + 1][j] + a[i]);
                } else {
                    dp[i][j] = dp[i + 1][j];
                }
            }
        }

        System.out.println(dp[0][5]);
    }
}
