import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), p = sc.nextInt();
        int[] a = new int[n], b = new int[k];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < k; i++) {
            b[i] = sc.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        int ans = 0, j = 0;
        for (int i = 0; i < n; i++) {
            while (j < k && b[j] < a[i]) {
                j++;
            }
            if (j < k) {
                ans = Math.max(ans, b[j] - a[i]);
                j++;
            }
        }
        ans = Math.max(ans, Math.max(Math.abs(a[n - 1] - p), Math.abs(b[k - 1] - p)));
        System.out.println(ans);
    }
}
