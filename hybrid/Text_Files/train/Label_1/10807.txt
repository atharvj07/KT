#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t; cin >> t;
    for (int test = 1; test <= t; test++) {
        int n, a, b; cin >> n >> a >> b;
        if (a > (n - 2) / 2 || b > (n - 2) / 2) {
            cout << -1 << '\n';
        } else {
            vector<int> perm(n);
            for (int i = 0; i < n; i++) { perm[i] = i + 1; }
            int dir = 1;
            for (int i = 1; i + 1 < n && b > 0; i += 2) {
                swap(perm[i], perm[i + 1]);
                --b; dir = -dir;
            }
            bool flag = true;
            for (int i = 1; i + 1 < n && a > 0; i += 2) {
                if (dir == 1) {
                    if (i + 2 < n && perm[i] > perm[i + 2]) {
                        swap(perm[i], perm[i + 2]);
                        --a; dir = -dir;
                    } else { flag = false; }
                } else {
                    if (i > 1 && perm[i] > perm[i - 2]) {
                        swap(perm[i], perm[i - 2]);
                        --a; dir = -dir;
                    } else { flag = false; }
                }
            }
            if (flag) {
                for (int i : perm) { cout << i << ' '; }
            } else {
                cout << -1;
            }
            cout << '\n';
        }
    }
    return 0;
}
