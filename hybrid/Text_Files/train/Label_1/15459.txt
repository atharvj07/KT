#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    for (int i = 0; i < t; ++i) {
        int str, intl, exp;
        cin >> str >> intl >> exp;
        if (intl > exp)
            cout << 0 << endl;
        else
            cout << exp - intl + 1 << endl;
    }
    return 0;
}
