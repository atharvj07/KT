#include <bits/stdc++.h>
using namespace std;

int main() {
  int x, k;
  cin >> x >> k;
  vector<int> u2(x);
  while (k--) {
    int t, a, b;
    cin >> t >> a;
    if (t == 1) cin >> b, u2[a] = 1, u2[b] = 1;
    if (t == 2) u2[a] = 1;
  }
  int mn = 0, mx = 0;
  for (int i = 1; i < x; i++) {
    if (u2[i] == 0) mn++, mx++;
    else if (u2[i - 1] == 0) mx++;
    if (u2[i] == 1 && u2[i - 1] == 1) mx++;
  }
  cout << mn << " " << mx << endl;
  return 0;
}
