#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> a(n, vector<int>(n));
  int x = 1;
  for (int i = 0; i < n; i++) {
    for (int j = i % 2; j < n; j += 2) { a[i][j] = x++; }
  }
  for (int i = 0; i < n; i++) {
    for (int j = (i + 1) % 2; j < n; j += 2) { a[i][j] = x++; }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cout << a[i][j] << ' '; }
    cout << '\n';
  }
}
