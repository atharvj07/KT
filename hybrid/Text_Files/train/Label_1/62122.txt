n,m=map(int,input().split())
X=[]
for i in range(n):
 X+=list(input())
pos=[]
for i in range(n*m):
 if X[i]=='X':pos+=[i]
l=len(pos)
ans="NO"
for i in range(l):
 for j in range(i,l):
  if pos[i]%m+1==pos[j]%m and pos[i]%m!=0 and (pos[i]//m,pos[j]//m) not in X:ans="YES"
  if pos[i]//m+1==pos[j]//m and pos[i]%m!=0 and (pos[i]//m,pos[j]//m) not in X:ans="YES"
print(ans)
