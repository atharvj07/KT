t = int(input())
for _ in range(t):
    x = int(input())
    if x == 0:
        print("1 1")
    elif x == 1:
        print("-1")
    else:
        for m in range(2, int(x**0.5) + 2):
            n = ((x - 1) // (m - 1))**0.5
            if n == int(n) and n * (m - 1) + 1 == x:
                print(int(n), m)
                break
        else:
            print("-1")
