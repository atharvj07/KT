import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        int odd = 0;
        for (int i = 0; i < N; i++) {
            if (a[i] % 2 == 1) {
                odd++;
            }
        }
        if (N % 2 == odd) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
