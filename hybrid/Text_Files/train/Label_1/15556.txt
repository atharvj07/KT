#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e5 + 10;
long long dp[MAX];  // dp[i] stores min cost to satisfy needs up to and including day i
int32_t main() {
    int n;
    cin >> n;
    pair<int, int> p[n];  // store (need, cost) for each day
    for (int i = 0; i < n; i++) { cin >> p[i].first >> p[i].second; }
    sort(p, p + n);
    memset(dp, 0x3f, sizeof dp);
    dp[0] = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j <= p[i].first; j++) { dp[j] = min(dp[j], 1LL * j * p[i].second + dp[j - p[i].first]); }
    }
    cout << dp[p[n - 1].first] << endl;
    return 0;
}
