import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long k = sc.nextLong();
        String[] words = new String[n];
        for (int i = 0; i < n; ++i) {
            words[i] = sc.next();
        }
        Arrays.sort(words);
        long[] dp = new long[m + 1];
        for (int i = 1; i <= m; ++i) {
            for (String word : words) {
                if (i >= word.length()) {
                    dp[i] += dp[i - word.length()];
                }
            }
        }
        if (dp[m] < k) {
            System.out.println("-");
            return;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = m; i > 0; ) {
            for (String word : words) {
                if (i >= word.length() && dp[i - word.length()] < k) {
                    sb.append(word);
                    i -= word.length();
                    k -= dp[i - word.length()];
                    break;
                }
            }
        }
        System.out.println(sb);
    }
}
