import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int n = input.nextInt();
            long maxl = Long.MAX_VALUE, minr = Long.MAX_VALUE;
            while (n-- > 0) {
                long l = input.nextLong();
                long r = input.nextLong();
                maxl = Math.max(maxl, l);
                minr = Math.min(minr, r);
            }
            System.out.println(Math.max(0, minr - maxl));
        }
    }
}
