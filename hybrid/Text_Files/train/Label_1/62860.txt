n = int(input())
previous_year = [input() for _ in range(n)]
current_year = [input() for _ in range(n)]

previous_counter = {}
current_counter = {}

for size in previous_year:
    previous_counter[size] = previous_counter.get(size, 0) + 1

for size in current_year:
    current_counter[size] = current_counter.get(size, 0) + 1

seconds = 0
for size in current_counter.keys():
    if size in previous_counter:
        seconds += max(0, current_counter[size] - previous_counter.get(size, 0))
    else:
        seconds += current_counter[size]

print(seconds)
