#include <bits/stdc++.h>
#define FOR(i, a, b) for (int i = (a), i##_end_ = (b); i < i##_end_; ++i)
using namespace std;
typedef long long ll;
const int maxn = 51;
int a[maxn << 1], p[maxn];
int main() {
  int t;
  cin >> t;
  FOR(_, 0, t) {
    int n;
    cin >> n;
    FOR(i, 0, 2 * n) cin >> a[i];
    int idx = 0;
    set<int> vis;
    FOR(i, 0, 2 * n) {
      if (vis.count(a[i]) == 0) {
        p[idx++] = a[i];
        vis.insert(a[i]);
      }
    }
    FOR(i, 0, n) cout << p[i] << ' ';
    cout << '\n';
  }
  return 0;
}
