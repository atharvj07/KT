#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<char>> grid(n, vector<char>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> grid[i][j];
        }
    }

    int moves = 0;
    bool canMove = true;
    while (canMove) {
        canMove = false;
        for (int i = 0; i < n; i++) {
            bool hasDwarf = false;
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 'G') {
                    hasDwarf = true;
                    if (j == m - 1) {
                        canMove = true;
                    } else if (grid[i][j + 1] == 'S') {
                        canMove = true;
                    }
                }
            }
            if (!hasDwarf) {
                canMove = true;
            }
        }
        if (canMove) {
            moves++;
            for (int i = 0; i < n; i++) {
                for (int j = m - 1; j > 0; j--) {
                    if (grid[i][j] == 'G') {
                        grid[i][j] = '*';
                        grid[i][j - 1] = 'G';
                    }
                }
            }
        }
    }

    bool allCandy = true;
    for (int i = 0; i < n; i++) {
        bool hasCandy = false;
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == 'S') {
                hasCandy = true;
            }
        }
        if (!hasCandy) {
            allCandy = false;
            break;
        }
    }

    if (allCandy) {
        cout << moves << endl;
    } else {
        cout << -1 << endl;
    }

    return 0;
}
