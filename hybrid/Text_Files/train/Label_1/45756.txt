import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        boolean[][] add = new boolean[n + 1][n + 1];
        boolean[][] mx = new boolean[n + 1][n + 1];
        int[] ans = new int[n];
        boolean ok = true;
        for (int i = 0; i < m; i++) {
            int t = sc.nextInt();
            int l = sc.nextInt();
            int r = sc.nextInt();
            if (t == 1) {
                int d = sc.nextInt();
                add[l][r] = true;
                if (l > 1)
                    mx[l - 1][r] = true;
                mx[l][r - 1] = true;
            } else {
                int max = sc.nextInt();
                mx[l][r] = true;
                if (l > 1) {
                    mx[l - 1][r] = true;
                    add[l - 1][r] = true;
                }
                for (int j = l; j <= r; j++) {
                    if (ans[j - 1] > max || ans[j - 1] + 1 < max) {
                        ok = false;
                    }
                    ans[j - 1] = max;
                }
            }
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        for (int i = 1; i < n; i++) {
            for (int j = 1; j <= n; j++) {
                if (add[i][j])
                    ans[i] += ans[i - 1];
            }
            for (int j = 1; j <= n; j++) {
                if (mx[i][j])
                    ans[i] = Math.max(ans[i], ans[i - 1] + 1);
            }
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
