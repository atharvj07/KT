#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n = 5;
  vector<vector<int>> g(n, vector<int>(n, 0));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> g[i][j];
    }
  }

  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    p[i] = i;
  }

  int best_happiness = 0;
  do {
    int happiness = 0;
    for (int i = 0; i < n; i++) {
      if (i % 2 == 0) {
        happiness += g[p[i]][p[i + 1]];
      } else {
        happiness += g[p[i + 1]][p[i]];
      }
    }
    best_happiness = max(best_happiness, happiness);
  } while (next_permutation(p.begin(), p.end()));

  cout << best_happiness << endl;

  return 0;
}
