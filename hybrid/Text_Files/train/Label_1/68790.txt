scales = input().split('|')
weights = input()
left, right = scales[0], scales[1]

total_weights = len(left) + len(right) + len(weights)

if total_weights % 2 != 0:
    print("Impossible")
else:
    half = total_weights // 2
    left_needed = half - len(left)
    if left_needed < 0 or left_needed > len(weights):
        print("Impossible")
    else:
        left += weights[:left_needed]
        right += weights[left_needed:]
        print(left + '|' + right)
