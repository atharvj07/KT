import sys

def solve():
    n, x = map(int, input().split())
    w = sorted(list(map(int, input().split())))
    if w[0] >= x:
        print("NO")
    else:
        print("YES")
        for i in w:
            if i + w[0] >= x:
                w[0], i = i, w[0]
            print(i, end=" ")

for _ in range(int(input())):
    solve()
