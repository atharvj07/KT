#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int n;
int a[N];
int l[N], r[N];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    a[0] = -1;
    a[n + 1] = -1;

    for (int i = 2; i <= n; i++) {
        if (a[i] - a[i - 1] <= 1 && a[i] - a[i - 2] <= 2) {
            l[i] = l[i - 1] + 1;
        } else {
            l[i] = 1;
        }
    }
    for (int i = n - 1; i > 0; i--) {
        if (a[i + 1] - a[i] <= 1 && a[i + 2] - a[i] <= 2) {
            r[i] = r[i + 1] + 1;
        } else {
            r[i] = 1;
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) ans = max(ans, l[i] + r[i] - 1);
    cout << ans << endl;
    return 0;
}
