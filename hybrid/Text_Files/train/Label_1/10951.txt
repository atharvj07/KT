#include <bits/stdc++.h>
using namespace std;

const int N = 1505;

int n, q;
char s[N], t[N];
int l[27], r[27];

int main() {
    cin >> n >> s >> q;
    memset(l, 0x3f, sizeof(l));
    memset(r, 0, sizeof(r));
    for (int i = 0; i < n; i++) {
        l[s[i] - 'a'] = min(l[s[i] - 'a'], i);
        r[s[i] - 'a'] = max(r[s[i] - 'a'], i);
    }
    for (int i = 0; i < q; i++) {
        int m;
        char c;
        cin >> m >> c;
        for (int j = 0; j < 26; j++) {
            if (j == c - 'a') {
                l[j] = min(l[j], m);
                r[j] = max(r[j], m);
            } else {
                l[j] = 0x3f3f3f;
                r[j] = 0;
            }
        }
        int ans = 0;
        for (int j = 0; j < 26; j++) {
            int len = r[j] - l[j] + 1;
            if (len > 0) ans = max(ans, len);
        }
        cout << ans << endl;
    }
    return 0;
}
