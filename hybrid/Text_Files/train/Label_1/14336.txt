#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  for (string &s : a) cin >> s;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (a[i][j] == 'C') {
        ans++;
        for (int k = j + 1; k < n; k++) {
          if (a[i][k] == 'C') {
            ans++;
          }
        }
        for (int k = i + 1; k < n; k++) {
          if (a[k][j] == 'C') {
            ans++;
          }
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
