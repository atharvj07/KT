import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
sa = sum(a)
sb = sum(b)
ans = abs(sa-sb)
swaps = 0
swa=[]
for i in range(n):
    for j in range(m):
        if abs(sa-sb+a[i]-b[j])<ans:
            ans = abs(sa-sb+a[i]-b[j])
            swaps = 1
            swa=[i+1, j+1]
sa+=a[0]-b[0]
sb+=b[0]-a[0]
for i in range(1, n):
    for j in range(1, m):
        if abs(sa-sb+a[i]-b[j])<ans:
            ans = abs(sa-sb+a[i]-b[j])
            swaps = 2
            swa.append([i+1, j+1])
print(ans)
print(swaps)
for sw in swa:
    print(*sw)
