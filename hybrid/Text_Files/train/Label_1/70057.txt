n,k = map(int, input().split())
s = input()
pos = list(range(26))
for i,c in enumerate(sorted(s)):
    pos[ord(c)-ord('a')] = i

dp = [0]*17
dp[k] = 0
for i in range(k-1,-1,-1):
    dp[i] = 1e18
    for j in range(i,k+1):
        if pos[j]-pos[i]>=2:
            dp[i] = min(dp[i], dp[j]+j-i)
    if dp[i]<1e18:
        break
print(-1 if dp[i]==1e18 else dp[i])
