#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        string a;
        cin>>a;
        int cnt1=0;
        for(int i=0;i<n;++i)
        {
            if(a[i]=='?') ++cnt1;
        }
        int cnt=0;
        char prev='.';
        for(int i=0;i<n;++i)
        {
            if(a[i]=='?')
            {
                if(cnt%2==0 && prev!='R') ++cnt, a[i]='R';
                else if(cnt%2==1 && prev!='B') ++cnt, a[i]='B';
                else if(prev=='R') ++cnt, a[i]='B';
                else ++cnt, a[i]='R';
                prev=a[i];
            }
            else
            prev=a[i];
        }
        cout<<a<<'\n';
    }
    return 0;
}
