#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    while (true) {
      bool ok = true;
      for (int i = 0; i < n; i++) {
        if (a[i] > 1) {
          ok = false;
          a[i] = (a[i] > 1) ? a[i] - 1 : 1;
        }
      }
      if (ok) break;
      ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
