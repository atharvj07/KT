#include <iostream>
#include <string>
#include <vector>
#define endl "\n"
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    vector<pair<char, char>> cards;
    for (int i = 0; i < n; i++) {
        pair<char, char> card;
        card.first = s[i];
        if (s[i] >= '0' && s[i] <= '9') {
            card.second = s[i];
        } else {
            card.second = s[i] - 32;
        }
        cards.push_back(card);
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (cards[i].first == 'A' || cards[i].first == 'E' || cards[i].first == 'I' || cards[i].first == 'O' || cards[i].first == 'U') {
            if (cards[i].second % 2 != 0) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
