#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> v(n);
  const int lim = n * 10;
  std::vector<int> freq(2 * lim + 1, 0);
  for (int i = 0; i < n - 1; ++i) {
    int q;
    scanf("%d", &q);
    q += lim + 1;
    if (freq[q] > 0 || q < 1 || q > 2 * lim + 1)
      return puts("-1"), 0;
    ++freq[q];
    --freq[q + 1];
  }
  for (int i = 1, pre = lim + 1; i <= lim + 1; ++i) {
    freq[i] += freq[i - 1];
    if (freq[i] <= 0)
      return puts("-1"), 0;
    pre += freq[i];
    v[freq[i] - 1] = pre - freq[i];
  }
  for (auto x : v)
    printf("%d ", x);
  return 0;
}
