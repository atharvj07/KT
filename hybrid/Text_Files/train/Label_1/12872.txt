#include <iostream>
#include <vector>
#include <queue>
using namespace std;

class Graph {
public:
  Graph(int n) : _n(n), _adj(n) {}
  void addEdge(int u, int v) {
    _adj[u].push_back(v);
  }

  bool isAcyclic() const;

private:
  int _n;
  vector<vector<int>> _adj;
};

bool Graph::isAcyclic() const {
  vector<int> in_degree(_n);
  vector<int> top_order;
  for (int i = 0; i < _n; i++) {
    for (auto v : _adj[i]) {
      in_degree[v]++;
    }
  }

  queue<int> q;
  for (int i = 0; i < _n; i++) {
    if (in_degree[i] == 0) {
      q.push(i);
    }
  }

  while (!q.empty()) {
    int u = q.front();
    q.pop();
    top_order.push_back(u);
    for (auto v : _adj[u]) {
      in_degree[v]--;
      if (in_degree[v] == 0) {
        q.push(v);
      }
    }
  }

  return top_order.size() == _n;
}

int main() {
  int n, m;
  cin >> n >> m;

  Graph g(n);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    g.addEdge(u, v);
  }

  if (g.isAcyclic()) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
