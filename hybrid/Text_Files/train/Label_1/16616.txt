#include <cstdio>
#include <algorithm>
using namespace std;

const int maxN = 150005;

int n, k, q;
int v[maxN];
pair<int, int> f[maxN]; // {t, id}
int id[maxN];

int main() {
    scanf("%d %d %d", &n, &k, &q);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &f[i].first);
        f[i].second = i;
    }
    sort(f + 1, f + 1 + n);
    for (int i = 1; i <= n; i++) {
        id[f[i].second] = i;
    }
    while (q--) {
        int type, x;
        scanf("%d %d", &type, &x);
        if (type == 1) {
            v[id[x]] = 1;
        }
        else {
            if (v[id[x]]) {
                puts("YES");
            }
            else {
                puts("NO");
            }
        }
    }
}
