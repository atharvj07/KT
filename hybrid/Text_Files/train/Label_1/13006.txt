#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
const int lim = 2e5 + 5;
int main() {
	ios::sync_with_stdio(false);
	cin.tie(NULL);

	int t, n, m, p;
	cin >> t;
	while (t--) {
		cin >> n >> m;
		string s;
		cin >> s;
		ull cnt[26] = {};
		for (int i = 0; i < m; i++) {
			cin >> p;
			cnt[s[p - 1] - 'a']++;
		}
		for (int i = 0; i < n; i++) cnt[s[i] - 'a']++;
		for (int i = 0; i < 26; i++) cout << cnt[i] << ' ';
		cout << '\n';
	}

	return 0;
}
