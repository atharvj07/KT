from collections import deque

n=int(input())
w=[int(i) for i in input().split()]
g=[[] for i in range(n)]
for i in range(n-1):
    a,b,c=[int(i) for i in input().split()]
    g[a-1].append((b-1,c))
    g[b-1].append((a-1,c))
d=[[-1]*(w[i]+1) for i in range(n)]
q=deque()
q.append((0,w[0]))
d[0][w[0]]=0
while q:
    x,f=q.popleft()
    for y,c in g[x]:
        if d[y][f]!=-1:continue
        ok=0
        for z in range(max(0,f-c),f+1):
            if d[x][z]!=-1 and w[y]>=c-z:
                ok=1
                d[y][f]=d[x][z]+c-z
                q.append((y,f))
                break
        if ok==0:
            for z in range(f,w[y]+1):
                if d[x][z]!=-1:
                    ok=1
                    d[y][f]=d[x][z]+c-z
                    q.append((y,f))
                    break
        if ok==0:
            d[y][w[y]]=d[x][f]+c-w[y]
            q.append((y,w[y]))
ans=0
for i in d[n-1]:
    if i!=-1:ans=max(i,ans)
print(ans)
