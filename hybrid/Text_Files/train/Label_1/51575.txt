import java.util.*;
import java.lang.*;
public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int m = input.nextInt();
    int[][] videos = new int[n][2];
    for (int i = 0; i < n; i++) {
      videos[i][0] = input.nextInt();
      videos[i][1] = input.nextInt();
    }
    int[][] channels = new int[m][3];
    for (int i = 0; i < m; i++) {
      channels[i][0] = input.nextInt();
      channels[i][1] = input.nextInt();
      channels[i][2] = input.nextInt();
    }
    int maxEff = 0;
    int bestVideo = -1;
    int bestChannel = -1;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        int li = videos[i][0];
        int ri = videos[i][1];
        int aj = channels[j][0];
        int bj = channels[j][1];
        int eff = -1;
        if (li >= aj && li <= bj) {
          eff = (ri - li) * channels[j][2];
        } else if (ri >= aj && ri <= bj) {
          eff = (ri - aj) * channels[j][2];
        } else if (li <= aj && ri >= bj) {
          eff = (bj - aj) * channels[j][2];
        }
        if (eff > maxEff) {
          maxEff = eff;
          bestVideo = i + 1;
          bestChannel = j + 1;
        }
      }
    }
    System.out.println(maxEff);
    if (maxEff > 0) {
      System.out.println(bestVideo + " " + bestChannel);
    }
  }
}
