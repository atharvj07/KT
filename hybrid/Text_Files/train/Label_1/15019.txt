#include<bits/stdc++.h>
using namespace std;
int arr[6];
int main(){
    int n; cin >> n;
    for(int i = 0; i < n; i++){
        int x; cin >> x; arr[x]++;
    }
    for(int i = 0; i < n; i++){
        int x; cin >> x; arr[x]++;
    }
    int mx = 0;
    for(int i = 1; i <= 5; i++){
        mx = max(mx,abs(arr[i] - n));
    }
    if(mx != 0 && mx != n){
        cout << -1;
        return 0;
    }
    cout << mx / 2;
    return 0;
}
