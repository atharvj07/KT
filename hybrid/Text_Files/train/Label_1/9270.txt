#include <algorithm>
#include <cmath>
#include <iostream>
#include <numeric>
#include <vector>

int main() {
  int n, k;
  long long ans = 0;
  std::cin >> n >> k;
  std::vector<long long> c(n);
  for (auto &i : c) { std::cin >> i; }
  std::sort(c.begin(), c.end());
  std::partial_sum(c.begin(), c.end(), c.begin());
  ans += c[n - 1] * k;
  for (int i = 0; i < n - 1; ++i) {
    ans -= i * c[i];
  }
  std::cout << ans << '\n';
  for (int i = k + 1; i <= k + n; ++i) {
    std::cout << i << ' ';
  }
  std::cout << '\n';
  return 0;
}
