#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n, m;
    cin >> n >> m;
    vector<array<ll, 3>> a(n), b(m);
    for (auto& vi : a) cin >> vi[0] >> vi[1] >> vi[2];
    for (auto& vi : b) cin >> vi[0] >> vi[1] >> vi[2];
    ll sum = 0, cur = 0;
    for (int i = 1; i < n; i++) {
        ll x1 = a[i][0] - a[i - 1][0];
        ll y1 = a[i][1] - a[i - 1][1];
        ll z1 = a[i][2] - a[i - 1][2];
        ll x2 = a[i + 1][0] - a[i][0];
        ll y2 = a[i + 1][1] - a[i][1];
        ll z2 = a[i + 1][2] - a[i][2];
        ll area = x1 * y2 - x2 * y1;
        ll xx = y1 * z2 - y2 * z1;
        ll yy = z1 * x2 - z2 * x1;
        ll zz = x1 * y2 - x2 * y1;
        cur = area * cur - xx * a[i][0] - yy * a[i][1] - zz * a[i][2];
        sum += area;
    }
    a.push_back(a[0]);
    bool sign = (cur < 0);
    if (sign) cur = -cur;
    int cnt = 0;
    for (int i = 0; i < m; i++) {
        ll x1 = b[i][0] - b[i - 1][0];
        ll y1 = b[i][1] - b[i - 1][1];
        ll z1 = b[i][2] - b[i - 1][2];
        ll x2 = b[i + 1][0] - b[i][0];
        ll y2 = b[i + 1][1] - b[i][1];
        ll z2 = b[i + 1][2] - b[i][2];
        ll area = x1 * y2 - x2 * y1;
        ll xx = y1 * z2 - y2 * z1;
        ll yy = z1 * x2 - z2 * x1;
        ll zz = x1 * y2 - x2 * y1;
        ll cur2 = area * cur - xx * a[0][0] - yy * a[0][1] - zz * a[0][2];
        if (cur2 < 0) cnt++;
    }
    if (cnt == 0 || cnt == 2)
        cout << "YES";
    else
        cout << "NO";
    return 0;
}
