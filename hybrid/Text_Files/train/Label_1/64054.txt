import sys

mod = 1000000007

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

def lcm(a, b):
    return a * b // gcd(a, b)

def solve():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    
    a.sort()
    
    strength = 0
    
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                strength += lcm(lcm(a[i], a[j]), a[k])
    
    return strength % mod

print(solve())
