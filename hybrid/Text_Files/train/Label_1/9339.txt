#include <bits/stdc++.h>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    int n = s.length(), m = t.length();
    vector<int> lps(m + 1);
    lps[0] = 0;
    for (int i = 1; i <= m; i++) {
        int j = lps[i - 1];
        while (j > 0 && t[i - 1] != t[j]) { j = lps[j - 1]; }
        if (t[i - 1] == t[j]) { j++; }
        lps[i] = j;
    }
    int cnt = 0;
    for (int i = 0, j = 0; i < n; i++) {
        if (j < m && (s[i] == t[j] || s[i] == '?')) { j++; }
        if (j == m) { cnt++; j = lps[j - 1]; }
    }
    cout << cnt << endl;
    return 0;
}
