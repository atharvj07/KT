n = int(input())
games = [int(input()) for _ in range(n)]

# Initial players: Alex(1), Bob(2), Carl(3)
players = {1, 2, 3}

# Initial spectator is Carl(3), and the initial game is between Alex(1) and Bob(2)
spectator = 3

possible = True
for winner in games:
    # If the winner is the spectator, it's impossible
    if winner == spectator:
        possible = False
        break
    else:
        # The spectator becomes the loser of the game, and the winner stays
        players.remove(winner)
        spectator = players.pop()
        players.add(winner)
        players.add(spectator)

if possible:
    print("YES")
else:
    print("NO")
