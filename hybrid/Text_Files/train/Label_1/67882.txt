n = int(input())
s = input()
ans = l = 0
cur = [0] * n
for i in range(n):
    if 'a' <= s[i] <= 'z' or 'A' <= s[i] <= 'Z':
        cur[i] = cur[i - 1] + 1
    else:
        cur[i] = 0
for i in range(n):
    if s[i] == '_':
        if cur[i - 1] > l:
            l = cur[i - 1]
            ans = 1
        elif cur[i - 1] == l:
            ans += 1
    elif s[i] == '(':
        cur[i] = 0
        if cur[i - 1] > l:
            l = cur[i - 1]
            ans = 0
        elif cur[i - 1] == l:
            ans += 1
num = 0
cnt = 0
for i in range(n):
    if s[i] == '(':
        num += 1
        cnt = 0
        continue
    elif s[i] == ')':
        if cnt > 0:
            ans += 1
        num -= 1
        continue
    if num > 0:
        if 'a' <= s[i] <= 'z' or 'A' <= s[i] <= 'Z':
            cnt += 1
print(l, ans)
