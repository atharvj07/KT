#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
vector<vector<int>> adj(MAXN);
vector<int> deg(MAXN, 0);
vector<pair<int, int>> ans;

int main() {
    int n, m, D;
    cin >> n >> m >> D;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        deg[u] += 1;
        deg[v] += 1;
    }
    queue<int> q;
    vector<bool> vis(n + 1, false);
    q.push(1);
    vis[1] = true;
    bool found = false;
    int cnt = 0;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        cnt++;
        if (u == 1) {
            deg[u] = D;
        }
        if (deg[u] == D && u == 1) {
            found = true;
            break;
        }
        for (int v : adj[u]) {
            if (!vis[v]) {
                q.push(v);
                vis[v] = true;
            }
        }
    }
    if (!found) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) {
        for (int j : adj[i]) {
            if (deg[i] + deg[j] > D) {
                ans.push_back({i, j});
                deg[i]--;
                deg[j]--;
            }
        }
    }
    for (auto p : ans) {
        cout << p.first << " " << p.second << endl;
    }
    return 0;
}
