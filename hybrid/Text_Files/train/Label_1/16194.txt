#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,i;
    cin>>n>>m;
    vector<int>a(n),b(n);
    for(i=0;i<n;i++)cin>>a[i];
    for(i=0;i<n;i++)cin>>b[i];
    sort(a.begin(),a.end());
    sort(b.begin(),b.end());
    for(i=0;i<n;i++)if(a[i]!=b[i])break;
    cout<<(m+(b[i]-a[i]))%m;
}
