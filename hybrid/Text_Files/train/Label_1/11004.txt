#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n), b(m);
    for (int &i : a)
        cin >> i;
    for (int &i : b)
        cin >> i;
    unordered_map<int, int> mp;
    for (int i : a) {
        while (i) {
            int mn = i;
            for (int j : b)
                mn = min(mn, j);
            for (int &j : b)
                j -= mn;
            i -= mn;
            mp[mn]++;
        }
    }
    cout << mp.size() << '\n';
    return 0;
}
