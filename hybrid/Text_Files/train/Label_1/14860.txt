#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 5e5 + 5;
ll a[maxn];
ll t[maxn];
void upd(int i, ll x) {
  for (; i <= maxn; i += i & (-i)) t[i] = max(t[i], x);
}
ll get(int i) {
  ll res = LLONG_MIN;
  for (; i; i -= i & (-i)) res = max(res, t[i]);
  return res;
}
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  ll sum = 0;
  for (int i = 1; i <= n; i++) {
    sum += get(i - 1);
    upd(i, sum + a[i]);
  }
  cout << *max_element(a, a + n + 1);
}
