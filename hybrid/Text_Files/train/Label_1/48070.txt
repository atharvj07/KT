import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] t = new int[n];
        boolean found = false;
        for (int i = 0; i < (1 << 2); i++) {
            t[0] = i;
            boolean valid = true;
            for (int j = 1; j < n; j++) {
                t[j] = (a[j - 1] | t[j - 1]) & (b[j - 1] | t[j - 1]);
                if (t[j] > 3) {
                    valid = false;
                    break;
                }
            }
            if (valid) {
                System.out.println("YES");
                for (int j = 0; j < n; j++) {
                    System.out.print(t[j] + " ");
                }
                System.out.println();
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("NO");
        }
    }
}
