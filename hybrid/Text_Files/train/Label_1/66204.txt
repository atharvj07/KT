import heapq
a, n, m = map(int, input().split())
l = [list(map(int, input().split())) for _ in range(n)]
u = sorted([(0,0)]+[(x,p) for x,p in (map(int, input().split()) for _ in range(m))]+[(a,100000)])
dp = [[100000]*(i+2) for i in range(len(u))]
dp[0][1] = 0
for i in range(len(u)-1):
    if dp[i][0]==100000: continue
    x1, y1 = u[i]
    x2, y2 = u[i+1]
    for j in range(1, i+2):
        if l and x2<=l[0][0]: break
        if j==i+1 and y2: continue
        dp[i+1][j] = min(dp[i+1][j], dp[i][j]+(x2-x1)*j*y1)
        dp[i+1][j+1] = min(dp[i+1][j+1], dp[i][j]+(x2-x1)*j*y1)
    if l:
        while l and x2>=l[0][1]:
            l.pop(0)
print(dp[-1][0] if dp[-1][0]!=100000 else -1)
