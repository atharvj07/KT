n, m = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))
flag = True
for x in k:
    if k.count(x)>1:
        flag = False
for i in range(1,n+1):
    for j in range(i,n+1):
        b = a[i-1:j]
        b.sort()
        if b==k:
            flag = True
            break
if flag:
    print("YES")
else:
    print("NO")
