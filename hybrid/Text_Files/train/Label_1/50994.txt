import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args) throws IOException {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] arr = new int[n];
            long[] maxPrefix = new long[n];
            long[] minPrefix = new long[n];
            for (int i = 0; i < n; i++) {
                arr[i] = scanner.nextInt();
            }
            maxPrefix[0] = arr[0];
            minPrefix[0] = arr[0];
            for (int i = 1; i < n; i++) {
                maxPrefix[i] = Math.max(maxPrefix[i - 1], arr[i]);
                minPrefix[i] = Math.min(minPrefix[i - 1], arr[i]);
            }
            boolean found = false;
            for (int i = 0; i < n; i++) {
                if (maxPrefix[i] - minPrefix[i] >= i + 1) {
                    System.out.println("YES");
                    System.out.println((i + 1) + " " + (i + 1));
                    found = true;
                    break;
                }
            }
            if (!found) {
                System.out.println("NO");
            }
        }
    }
}
