import sys
from collections import deque

d, n, m = map(int,input().split())
stations = []
for _ in range(m):
    stations.append(list(map(int,input().split())))

stations.sort(key=lambda x: x[0])

if d <= n:
    print(0)
    sys.exit()

cost = 0
dq = deque()
dq.append((0,0))

for x, p in stations:
    while dq and dq[0][0] <= x - n:
        dq.popleft()
    if not dq:
        print(-1)
        sys.exit()
    cost += (x - dq[0][0] - dq[0][1]) * dq[0][2]
    while dq and dq[-1][2] >= p:
        dq.pop()
    dq.append((x,x-dq[0][0],p))

while dq and dq[0][0] <= d - n:
    dq.popleft()
if not dq:
    print(-1)
    sys.exit()
cost += (d - dq[0][0] - dq[0][1]) * dq[0][2]

print(cost)
