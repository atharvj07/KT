from collections import defaultdict
for _ in range(int(input())):
    n,x = map(int,input().split())
    a = list(map(int,input().split()))
    sum_pref = defaultdict(list)
    sum = 0
    sum_pref[0].append(-1)
    mx_len=-1
    for i in range(n):
        sum=(sum+a[i]%x)%x
        if sum in sum_pref:
            mx_len=max(mx_len,i-sum_pref[sum][0])
        else:
            sum_pref[sum].append(i)
    print(mx_len)
