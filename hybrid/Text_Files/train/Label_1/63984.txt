n, w = map(int, input().split())
a = list(map(int, input().split()))

a.sort()

i = 0
j = n - 1

ans = [-1] * n

while i <= j:
    if a[i] >= a[j] // 2:
        if w >= a[j]:
            ans[j] = a[j] // 2
            w -= ans[j]
        else:
            ans[j] = w
            w = 0
        j -= 1
    else:
        if w >= a[i] * 2:
            ans[i] = a[i]
            w -= a[i]
        else:
            ans[i] = w // 2
            w -= ans[i]
        i += 1

if w != 0:
    ans = [-1] * n

print(*ans)
