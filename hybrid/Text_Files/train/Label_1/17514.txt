#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    long long sum = 0;
    bool zero = false;
    for (long long i = 0, a; i < n; i++) {
      cin >> a;
      sum += a;
      if (!a) zero = true;
    }
    if (!zero)
      cout << "YES" << endl;
    else if (sum > 0)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
