#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  ll n, a, b, x;
  unordered_map<ll, ll> m1;
  unordered_map<ll, ll> m2;
  vector<ll> v;
  cin >> n;
  for (ll i = 0; i < n; i++) {
    cin >> x >> a;
    if (x == 1) {
      v.push_back(a);
      m1[a]++;
    } else {
      m2[a] = b;
      cin >> b;
    }
  }
  for (ll i = 0; i < v.size(); i++) {
    x = m2[v[i]];
    if (x == 0)
      cout << v[i] << " ";
    else
      cout << x << " ";
  }
  return 0;
}
