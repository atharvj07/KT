import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String s = input.next();
        int ans = 0;
        int pos = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'L') pos--;
            else pos++;
            ans = Math.max(ans, pos);
            ans = Math.max(ans, -pos);
        }
        System.out.println(ans * 2 + 1);
    }
}
