#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  int mx = 0, sum = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    sum += v[i];
    if (sum > 100 * (i + 1)) {
      sum -= v[i - mx];
      mx++;
    }
    if (i - mx + 1 > ans) ans = i - mx + 1;
  }
  cout << ans << endl;
  return 0;
}
