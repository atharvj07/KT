import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String a = scanner.nextLine();
        String b = scanner.nextLine();
        System.out.println(a.equals(b) ? "YES" : isEquivalent(a, b) ? "YES" : "NO");
    }

    private static boolean isEquivalent(String a, String b) {
        int n = a.length();
        for (int i = 1; i <= n / 2; i++) {
            String a1 = a.substring(0, i), a2 = a.substring(i);
            String b1 = b.substring(0, i), b2 = b.substring(i);
            if ((isEquivalent(a1, b1) && isEquivalent(a2, b2)) || (isEquivalent(a1, b2) && isEquivalent(a2, b1))) {
                return true;
            }
        }
        return false;
    }
}
