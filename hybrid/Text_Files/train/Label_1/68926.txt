import sys
input = sys.stdin.readline

n = int(input())
items = [list(map(int, input().split())) for _ in range(n)]
items.sort(key=lambda x: x[1])
dp = [[0] * 2000 for _ in range(n + 1)]
for i in range(n):
    ti, di, pi = items[i]
    for j in range(2000):
        if j + ti <= di:
            dp[i + 1][j + ti] = max(dp[i + 1][j + ti], dp[i][j] + pi)
        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j])
res = []
j = 1999
for i in range(n, 0, -1):
    if dp[i][j] > dp[i - 1][j]:
        res.append(i)
        j -= items[i - 1][0]
print(dp[n][1999])
print(len(res))
print(*res)
