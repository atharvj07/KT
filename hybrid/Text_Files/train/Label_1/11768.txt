#include<cstdio>
#include<cstring>
#include<vector>

#include<algorithm>
using namespace std;
#define pb push_back
#define mp make_pair
#define pii pair<int,int>
#define inf 100000000
int f[80][80], g[80][80];
char c[80][80], s[80];
int main(){
    int n, x, y, z, i, j, k, u, v, w;
    scanf("%d %d %d %d", &n, &x, &y, &z);
    for(i = 1; i <= n; ++i){
        scanf("%s", c + i + 1);
    }
    for(i = 1; i <= n; ++i)for(j = 1; j <= n; ++j)if(j != i)f[i][j] = inf;
    for(i = 1; i <= n; ++i){
        queue<int> q;
        q.push(i);
        while(!q.empty()){
            x = q.front();
            q.pop();
            for(j = 1; j <= n; ++j)if(i != j){
                if(c[i][j] == c[x][y] && c[j][y] == c[x][z]){
                    if(f[x][j] > f[i][x] + 1){
                        f[x][j] = f[i][x] + 1;
                        g[x][j] = i;
                        q.push(j);
                    }
                }
            }
        }
    }
    for(i = 1; i <= n; ++i)if(f[i][1] != inf && f[i][2] != inf && f[i][3] != inf){
        j = i;
        for(k = 0; k < f[i][1] + f[i][2] + f[i][3]; ++k){
            printf("%d %d\n", j, g[j][1]);
            j = g[j][1];
        }
        return 0;
    }
    puts("-1");
    return 0;
}
