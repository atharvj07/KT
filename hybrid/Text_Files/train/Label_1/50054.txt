import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt(), b = sc.nextInt(), h = sc.nextInt(), w = sc.nextInt(), n = sc.nextInt(), ans = -1;
        int[] as = new int[n];
        for (int i = 0; i < n; i++) as[i] = sc.nextInt();
        Arrays.sort(as);
        if (a <= h && b <= w) {
            ans = 0;
        } else if (a <= w && b <= h) {
            ans = 0;
        } else {
            for (int i = n - 1; i >= 0; i--) {
                if (Math.min(a, b) <= h && Math.max(a, b) <= w) {
                    ans = n - i;
                    break;
                }
                if (Math.min(a, b) <= w && Math.max(a, b) <= h) {
                    ans = n - i;
                    break;
                }
                if (a < h) {
                    int temp = a;
                    a = b;
                    b = temp;
                }
                int nh = h * as[i];
                if (nh >= a && w >= b) {
                    ans = n - i;
                    break;
                }
                h = nh;
            }
        }
        System.out.println(ans);
    }
}
