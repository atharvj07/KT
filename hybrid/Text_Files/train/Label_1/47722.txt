import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.math.RoundingMode;

public class Main {

    public static void main(String[] args) throws java.lang.Exception {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(bf.readLine());
        double m = Double.parseDouble(bf.readLine());
        double[] a = new double[n];
        String[] a_line = bf.readLine().split(" ");
        double[] b = new double[n];
        String[] b_line = bf.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Double.parseDouble(a_line[i]);
        }
        for (int i = 0; i < n; i++) {
            b[i] = Double.parseDouble(b_line[i]);
        }
        double low = 0, high = 1000000000;
        while (high - low > 1e-6) {
            double mid = (low + high) / 2;
            double[] fuel = new double[n + 1];
            boolean ok = true;
            for (int i = 1; i <= n; i++) {
                if ((i == n || m + fuel[i - 1] <= a[i] * mid) && (fuel[i - 1] >= b[i] * mid)) {
                    fuel[i] = (m + fuel[i - 1]) / a[i] - mid;
                } else {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                high = mid;
            } else {
                low = mid;
            }
        }
        if (high < 1e9) {
            System.out.println(String.format("%.9f", high));
        } else {
            System.out.println(-1);
        }
        bf.close();
    }
}
