import heapq
import sys

n, m, t = map(int, input().split())
edges = [[] for _ in range(n+1)]
for _ in range(m):
    u, v, w = map(int, input().split())
    edges[u].append((w, v))

q = [(0, 1)]
dist = [t for _ in range(n+1)]
ans = []
while q:
    d, u = heapq.heappop(q)
    if u == n:
        break
    if d <= dist[u]:
        dist[u] = d
        for w, v in edges[u]:
            if d+w <= dist[v]:
                ans.append(u)
                heapq.heappush(q, (d+w, v))

print(len(ans)+1)
print(*ans, n)
