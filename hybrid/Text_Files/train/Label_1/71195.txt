r, c = map(int, input().split())
a = [list(input()) for i in range(r)]
ans = 0
for i in range(r):
    t = set()
    for j in range(c):
        if a[i][j] == '.':
            t.add(j)
    ans += len(t)
    for j in t:
        for k in range(r):
            if a[k][j] == '.':
                a[k][j] = 'X'
for j in range(c):
    t = set()
    for i in range(r):
        if a[i][j] == '.':
            t.add(i)
    ans += len(t)
    for i in t:
        for k in range(c):
            if a[i][k] == '.':
                a[i][k] = 'X'
print(ans)
