from collections import defaultdict, deque
from sys import stdin

def bfs(G: defaultdict, visited: set[int], start: int, ans: list) -> None:
    q = deque([(start, 0)])
    visited.add(start)
    while q:
        node, dis = q.popleft()
        for i in G[node]:
            if i in visited:
                continue
            ans.append(i)
            visited.add(i)
            q.append((i, dis + 1))
    
input = stdin.readline
n, m = map(int, input().split())
G = defaultdict(list)
for i in range(m):
    u, v, c = map(str, input().split())
    u, v = int(u), int(v)
    G[u].append(v)
    G[v].append(u)

visited = set()
ans = []
bfs(G, visited, 1, ans)

if len(ans) != n - 1:
    ans = [-1]
print(len(ans))
print(*ans)
