#include <bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    int a[n][k];
    for(int i=0;i<n;++i){
        for(int j=0;j<k;++j){
            cin>>a[i][j];
        }
    }
    int ans=1;
    for(int i=0;i<n;++i){
        int c=0;
        for(int j=0;j<k;++j){
            if(a[i][j]==1) c+=1;
        }
        if(c>(k/2.0)) ans=0;
    }
    if(ans==1) cout<<"YES"<<"\n";
    else cout<<"NO"<<"\n";
}
