import sys
input = lambda: sys.stdin.readline().rstrip()

N = int(input())
A = [int(a) for a in input().split()]
A.sort()

res = 0
k = 0
teams = []
i = 0
while i < N:
    j = i
    while j < N and A[j] == A[i]:
        j += 1
    if j - i >= 3:
        k += 1
        res += A[j - 1] - A[i]
        teams += [k] * (j - i)
        i = j
    else:
        break

if i < N:
    k += 1
    teams += [k] * (N - i)
    res += A[-1] - A[i]

print(res, k)
print(*teams)
