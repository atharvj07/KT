#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  if (a == b) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) {
    if (i % a == 0 && i % b != 0) {
      p[i] = i - a;
    } else if (i % b == 0) {
      p[i] = i - b;
    } else {
      p[i] = i + 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << p[i] << " ";
  }
  cout << endl;
  return 0;
}
