import java.util.*;

public class Main {
    static int[] dijkstra(List<List<int[]>> adjList, int n) {
        int[] dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[0] = 0;
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        pq.add(new int[]{0, 0});
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            int u = cur[1];
            if (cur[0] > dist[u]) {
                continue;
            }
            for (int[] edge : adjList.get(u)) {
                int v = edge[0];
                int w = edge[1];
                if (dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    pq.add(new int[]{dist[v], v});
                }
            }
        }
        return dist;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<List<int[]>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int u = sc.nextInt();
            int k = sc.nextInt();
            List<int[]> edges = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                int v = sc.nextInt();
                int w = sc.nextInt();
                edges.add(new int[]{v, w});
            }
            adjList.add(edges);
        }
        sc.close();
        int[] dist = dijkstra(adjList, n);
        for (int i = 0; i < n; i++) {
            System.out.println(i + " " + dist[i]);
        }
    }
}
