import java.util.*;

public class Main {
    static final int MOD = 998244353;
    int n, k;
    int[][] segs;

    Main(int n, int k, int[][] segs) {
        this.n = n;
        this.k = k;
        this.segs = segs;
    }

    static long modpow(long a, long p) {
        if (p == 0)
            return 1;
        if (p == 1)
            return a;
        if (p % 2 == 0) {
            long half = modpow(a, p / 2);
            return half * half % MOD;
        }
        return a * modpow(a, p - 1) % MOD;
    }

    long solve() {
        long[] dp = new long[n + 1];
        dp[1] = 1;
        for (int i = 2; i <= n; i++) {
            for (int[] seg : segs) {
                if (seg[0] <= i && i <= seg[1]) {
                    dp[i] += dp[i - seg[0]];
                    dp[i] %= MOD;
                }
            }
        }
        return dp[n];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] segs = new int[k][2];
        for (int i = 0; i < k; i++) {
            segs[i][0] = sc.nextInt();
            segs[i][1] = sc.nextInt();
        }
        Main m = new Main(n, k, segs);
        System.out.println(m.solve());
    }
}
