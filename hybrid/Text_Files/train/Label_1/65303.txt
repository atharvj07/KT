from collections import Counter

n = int(input())
s = str(input())
t = str(input())
if Counter(s) != Counter(t):
    print(-1)
else:
    # find index of differing characters
    idx = [i for i in range(n) if s[i] != t[i]]
    swap = [[], []]
    for i in idx[::2]:
        swap[0].append(i)
    for i in idx[1::2]:
        swap[1].append(i)
    print(len(idx) // 2)
    for i in range(len(idx) // 2):
        print(swap[0][i] + 1, swap[1][i] + 1)
