#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, l, r, s;
        cin >> n >> l >> r >> s;
        int a[n];
        for (int i = 0; i < n; ++i) a[i] = i + 1;
        if (s > n * (n + 1) / 2 - n + 1) {
            cout << -1 << endl;
            continue;
        }
        swap(a[l - 1], a[r - 1]);
        int sum = 0;
        for (int i = l - 1; i < r; ++i) sum += a[i];
        int cnt = 1;
        for (int i = r - 1; i >= 0; --i) {
            sum -= a[i];
            sum += n - cnt;
            if (sum > s) {
                s -= a[i];
                a[i] = n - cnt;
                ++cnt;
            }
        }
        for (int i = 0; i < n; ++i) cout << a[i] << ' ';
        cout << endl;
    }
    return 0;
}
