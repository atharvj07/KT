#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<long long> c(n + 1), t(n + 1);
  for (int i = 1; i <= n; ++i) std::cin >> c[i] >> t[i];
  std::vector<long long> v(m + 1), prefix(n + 1);
  for (int i = 1; i <= m; ++i) std::cin >> v[i];
  for (int i = 1; i <= n; ++i) prefix[i] = prefix[i - 1] + c[i] * t[i];
  for (int i = 1; i <= m; ++i) {
    int l = 1, r = n, ans = -1;
    while (l <= r) {
      int mid = (l + r) / 2;
      if (prefix[mid] < v[i]) {
        l = mid + 1;
      } else {
        ans = mid;
        r = mid - 1;
      }
    }
    std::cout << ans << std::endl;
  }
  return 0;
}
