import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        long d = (long) Math.pow(10, k - 1);
        long ans = 0;
        for (long i = d; i > 0 && i <= n; i *= 10)
            ans += (n % i) / d;
        System.out.println(ans);
    }
}
