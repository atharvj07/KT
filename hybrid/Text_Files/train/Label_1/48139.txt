import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        char[][] a = new char[n][m];
        int[] dx = {-1, 0, 1, 0}, dy = {0, 1, 0, -1};
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j);
            }
        }
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '.') {
                    Queue<Integer> q = new LinkedList<>();
                    q.add(i * m + j);
                    dist[i][j] = 0;
                    while (!q.isEmpty()) {
                        int cur = q.remove();
                        int x = cur / m, y = cur % m;
                        for (int d = 0; d < 4; d++) {
                            int nx = x + dx[d], ny = y + dy[d];
                            if (nx >= 0 && nx < n && ny >= 0 && ny < m && a[nx][ny] == '.' && dist[nx][ny] == 0) {
                                dist[nx][ny] = dist[x][y] + 1;
                                q.add(nx * m + ny);
                            }
                        }
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '.' && dist[i][j] > k) {
                    a[i][j] = 'X';
                    k--;
                }
                if (k == 0) {
                    break;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(a[i][j]);
            }
            System.out.println();
        }
    }
}
