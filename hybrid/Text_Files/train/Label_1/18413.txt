#include <iostream>
#include <vector>
using namespace std;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, k;
    cin >> n >> k;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    int pos = 0;
    while (k && pos + 1 < n && h[pos] >= h[pos + 1]) k--, pos++;
    cout << (k ? -1 : pos + 1) << endl;
  }
  return 0;
}
