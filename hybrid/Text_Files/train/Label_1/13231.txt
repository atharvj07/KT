#include <iostream>
#include <vector>

using namespace std;

const long long mod = 1e18;
const int maxn = 100001;

long long a[maxn];
vector<int> ans;

void solve() {
    int n;
    cin >> n;

    a[1] = 2;
    for (int i = 2; i <= n + 1; ++i) {
        a[i] = a[i - 1] + (i - 1) * a[i - 1];
        ans.push_back(a[i] / i);
    }
    for (int i : ans) {
        cout << i << '\n';
    }
}

int main() {
    solve();
    return 0;
}
