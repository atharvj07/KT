#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n + 1], b[n + 1], c[n + 1];
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) { cin >> b[i]; }
    for (int i = 1; i <= n; i++) { cin >> c[i]; }

    int dp[n + 1][3]; // dp[i][j] means the maximum total joy when feeding the first i hares, and hare i is fed with j status (0: both hungry, 1: one full, 2: both full)
    dp[1][0] = a[1];
    dp[1][1] = b[1];
    dp[1][2] = c[1];
    for (int i = 2; i <= n; i++) {
        dp[i][0] = max(dp[i - 1][1], dp[i - 1][2]) + a[i];
        dp[i][1] = max(dp[i - 1][0], max(dp[i - 1][1], dp[i - 1][2])) + b[i];
        dp[i][2] = max(dp[i - 1][0], dp[i - 1][1]) + c[i];
    }

    cout << max(dp[n][0], max(dp[n][1], dp[n][2])) << endl;

    return 0;
}
