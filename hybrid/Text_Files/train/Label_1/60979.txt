import sys

n = int(input())
a = list(map(int,input().split()))
p = [0]*n
inv = []
for i in range(n):
    for j in range(i+1,n):
        if a[i]>a[j]:
            inv.append((i+1,j+1))
            p[i]+=1
            p[j]-=1
for i in range(n):
    if p[i]!=0:
        print(-1)
        sys.exit(0)
print(len(inv))
for x in inv:
    print(*x)
