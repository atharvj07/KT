import sys

m, n, k, t = map(int, input().split())
a = list(map(int, input().split()))
traps = []
for _ in range(k):
    l, r, d = map(int, input().split())
    traps.append((l, r, d))
traps.sort()

# dp[i][j][0] = max soldiers that can reach point i in j seconds without squad
# dp[i][j][1] = max soldiers that can reach point i in j seconds with squad
dp = [[[0 for _ in range(2)] for _ in range(t + 1)] for _ in range(n + 2)]
dp[0][0][0] = 0

for i in range(n + 1):
    for j in range(t + 1):
        if i > 0:
            # move without squad
            dp[i][j][0] = max(dp[i][j][0], dp[i - 1][j][0])
        if i + 1 <= n:
            # move without squad
            dp[i][j][0] = max(dp[i][j][0], dp[i + 1][j][0])
        if j > 0 and i > 0:
            # move with squad
            dp[i][j][1] = max(dp[i][j][1], dp[i - 1][j - 1][1])
        if j > 0 and i + 1 <= n:
            # move with squad
            dp[i][j][1] = max(dp[i][j][1], dp[i + 1][j - 1][1])
        # disarm trap
        for l, r, d in traps:
            if i == l and j > 0:
                dp[i][j][1] = max(dp[i][j][1], dp[r][j - 1][0])

for i in range(n + 1):
    for j in range(t + 1):
        # bring squad
        dp[i][j][1] = max(dp[i][j][1], dp[i][j][0] + 1)

ans = 0
for i in range(n + 1):
    for j in range(t + 1):
        if dp[i][j][1] >= m:
            ans = max(ans, i)

print(ans)
