#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> f(n);
  for (int &i : f) cin >> i;
  vector<bool> vis(n + 1, false);
  for (int i = 0; i < n; i++)
    if (f[i] != 0) vis[f[i]] = true;
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    if (f[i] == 0) {
      int j = 1;
      while (vis[j]) j++;
      vis[j] = true;
      ans[i] = j;
    } else
      ans[i] = f[i];
  }
  for (int i : ans) cout << i << ' ';
}
