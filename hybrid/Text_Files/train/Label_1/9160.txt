#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n;
vector<pair<ll,ll>>a;
int main(){
	cin>>n;
	a.resize(n);
	for(ll i=0;i<n;i++){
		ll x;
		cin>>x;
		a[i]={x,i+1};
	}
	sort(a.begin(),a.end());
	vector<vector<ll>>skr;
	vector<ll>team;
	for(ll i=0;i<n;i++){
		if(!team.empty() && team[0]!=a[i].first){
			skr.push_back(team);
			team.clear();
		}
		team.push_back(a[i].first);
	}
	skr.push_back(team);
	ll res=0;
	for(auto v:skr){
		res+=v[v.size()-1]-v[0];
	}
	cout<<res<<" "<<skr.size()<<endl;
	for(auto v:a){
		for(ll i=0;i<skr.size();i++){
			if(skr[i][0]==v.first){
				cout<<i+1<<" ";
				break;
			}
		}
	}
	cout<<endl;
	return 0;
}
