#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t1, t2, t3, t4, n, m;
  cin >> t1 >> t2 >> t3 >> t4;
  cin >> n >> m;
  vector<int> a(n), b(m);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    ans += min(t1 * a[i], t2);
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
    ans += min(t1 * b[i], t2);
  }
  ans = min(ans, t3);
  ans = min(ans, t4);
  cout << ans;
}
