n, m = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
x = int(input())

c = [[a[i - 1] * b[j - 1] for j in range(1, m + 1)] for i in range(1, n + 1)]
f = [[0 for _ in range(m + 1)] for _ in range(n + 1)]

for i in range(n):
    for j in range(m):
        f[i + 1][j + 1] = f[i + 1][j] + f[i][j + 1] - f[i][j] + c[i][j]

for ma in range(x, 0, -1):
    l = 0
    r = 0
    ans = 0
    for r in range(1, m + 1):
        if c[0][r - 1] <= ma:
            ans = 1
        for l in range(1, n + 1):
            cur = f[l][r] - f[l][r - 1] - f[l - 1][r] + f[l - 1][r - 1]
            if cur <= ma:
                ans = max(ans, (r - l + 1) * (l - r + 1))
    if ans > 0:
        print(ans)
        break
if ans == 0:
    print(0)
