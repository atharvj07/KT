import java.util.*;

class Main {
    static boolean[][] visit = new boolean[3000][3000];
    static int[][] ans = new int[1000][2];
    static int[] dx = {-2, -1, 1, 2, 2, 1, -1, -2, -2, -1, 1, 2};
    static int[] dy = {-1, -2, -2, -1, 1, 2, 2, 1, -1, -2, -2, -1};
    static int n, cnt = 1;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        int x = 0, y = 0;

        while (cnt <= n) {
            while (visit[x][y]) {
                if (++x > 2999) {
                    x = 0;
                    ++y;
                }
            }
            visit[x][y] = true;
            ans[cnt][0] = x;
            ans[cnt][1] = y;
            ++cnt;

            for (int i = 0; i < 12; ++i) {
                int a = x + dx[i];
                int b = y + dy[i];
                if (a >= 0 && a < 3000 && b >= 0 && b < 3000) {
                    visit[a][b] = true;
                }
            }
        }

        for (int i = 1; i <= n; ++i) {
            System.out.println(ans[i][0] + " " + ans[i][1]);
        }
    }
}
