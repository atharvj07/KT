#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  string a;
  cin >> a;
  int n = a.size();
  vector<int> l(n, 0);
  vector<int> r(n, 0);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == '[') {
      if (i > 0) {
        l[i] = l[i - 1] + 1;
      } else {
        l[i] = 1;
      }
    }
  }
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] == ']') {
      if (i < n - 1) {
        r[i] = r[i + 1] + 1;
      } else {
        r[i] = 1;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (a[i] == ':' && l[i] > 0 && r[i] > 0) {
      ans = max(ans, 2 * (l[i] + r[i]) + 2);
    }
  }
  if (ans == 0) {
    cout << "-1" << endl;
  } else {
    cout << ans << endl;
  }
}
