import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long d = sc.nextLong();
        long n = sc.nextLong();
        long m = sc.nextLong();

        long[] stations = new long[2 * (int) m];
        for (int i = 0; i < m; i++) {
            stations[i * 2] = sc.nextLong();
            stations[i * 2 + 1] = sc.nextLong();
        }
        Arrays.sort(stations);
        long cost = 0;
        long prev = 0;
        long curr = 0;
        for (int i = 0; i < stations.length; i += 2) {
            long pos = stations[i];
            long price = stations[i + 1];
            if (pos > prev + n) {
                if (curr == 0) {
                    cost = -1;
                    break;
                }
                cost += (pos - prev - n) * curr;
                prev += (pos - prev - n);
            }
            curr = Math.min(curr + n - (pos - prev), price);
            prev = pos;
        }
        if (prev + n < d) {
            if (curr == 0) {
                cost = -1;
            } else {
                cost += (d - prev - n) * curr;
            }
        }
        System.out.println(cost);
    }
}
