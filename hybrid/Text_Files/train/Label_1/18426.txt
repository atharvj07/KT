#include <bits/stdc++.h>
using namespace std;
#define ll long long
pair<ll, ll> p[1005];
int main() {
    ll n, ans = 0, sum1 = 0, sum2 = 0;
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        ll a, b, c, d;
        cin >> a >> b >> c >> d;
        p[i].first = a + b + c + d;
        p[i].second = i;
    }
    sort(p + 1, p + n + 1);
    reverse(p + 1, p + n + 1);
    for (ll i = 1; i <= n; i++) {
        if (p[i].first == p[i + 1].first)
            continue;
        ans++;
        if (p[i].second == 1) {
            sum1 = ans;
            break;
        }
    }
    ll idx = p[1].second;
    for (ll i = 1; i <= n; i++) {
        if (p[i].second == idx) {
            sum2 = ans;
        }
    }
    cout << sum1 - sum2 + 1 << endl;
    return 0;
}
