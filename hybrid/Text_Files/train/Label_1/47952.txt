import java.util.Scanner;

public class Main {
  public static void main(String[] agrs) {
    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = cin.nextInt();
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int or = 0;
      for (int j = i; j < n; j++) {
        or |= a[j];
        if (or > a[j]) ans++;
      }
    }
    System.out.println(ans);
  }
}
