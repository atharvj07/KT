#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  cin >> s;
  if (s.size() % 2 == 1)
    cout << "First" << endl;
  else {
    bool odd = false;
    int cnt[26] = {};
    for (char c : s) cnt[c - 'a']++;
    for (int i = 0; i < 26; i++) {
      if (cnt[i] % 2 == 1) {
        if (odd) {
          cout << "Second" << endl;
          return 0;
        }
        odd = true;
      }
    }
    cout << "First" << endl;
  }
}
