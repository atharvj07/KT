#include <iostream>
#include <algorithm>
using namespace std;
int main(){
    string s; cin >> s;
    int k; cin >> k;
    sort(s.begin(), s.end());
    int same = 0;
    for(int i = 1; i < s.length(); i++){
        if(s[i] == s[i - 1]) same++;
        else if(same > 0){
            same--;
            k--;
        }
    }
    if(k > 0) cout << "impossible";
    else cout << same;
}
