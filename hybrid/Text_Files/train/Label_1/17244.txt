#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    pair<long long,long long>ans={0,0},tmp;
    long long x,y;
    for(int i=0;i<n;++i){
        cin>>x>>y;
        if(x*x+y*y>ans.first*ans.first+ans.second*ans.second)
            ans={x,y};
    }
    tmp={ans.second,-ans.first};
    if(tmp.first*tmp.first+tmp.second*tmp.second>ans.first*ans.first+ans.second*ans.second)
        ans=tmp;
    long double sqr=sqrt(ans.first*ans.first+ans.second*ans.second);
    long double deg=acos(ans.first/sqr)*180/acos(-1.0);
    cout<<fixed<<setprecision(10)<<deg<<endl;
    return 0;
}
