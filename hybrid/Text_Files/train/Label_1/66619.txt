from math import gcd

a, b, c, d = map(int, input().split())

# Find the scaling factors for both dimensions and choose the minimum
scale = min(a*d, b*c)

# Calculate the empty space ratio
empty = a*b*d*d - scale*scale*c*b

# Calculate the total screen size
total = a*b*d*d

# Simplify the fraction
g = gcd(empty, total)
print(f"{empty//g}/{total//g}")
