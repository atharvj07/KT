from sys import stdin

n, m = map(int, stdin.readline().strip().split())
grid = [list(stdin.readline().strip()) for i in range(n)]
visited = [[False for j in range(m)] for i in range(n)]

def valid(i, j):
    return 0 <= i < n and 0 <= j < m and not visited[i][j] and grid[i][j] == "B"

def dfs(i, j, prev_i, prev_j):
    if not valid(i, j):
        return False
    visited[i][j] = True
    if i == prev_i and j == prev_j:
        return True
    if prev_i != -1 and prev_j != -1 and abs(i - prev_i) + abs(j - prev_j) > 1:
        return False
    return dfs(i + 1, j, i, j) or dfs(i - 1, j, i, j) or dfs(i, j + 1, i, j) or dfs(i, j - 1, i, j)

if dfs(0, 0, -1, -1):
    print("YES")
else:
    print("NO")
