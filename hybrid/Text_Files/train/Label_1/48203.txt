import java.util.Scanner;

public class MartianDollar {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int b = scanner.nextInt();
        int[] prices = new int[n];
        for (int i = 0; i < n; i++) {
            prices[i] = scanner.nextInt();
        }

        int maxMoney = b;
        int dollars = 0;
        for (int i = 0; i < n; i++) {
            // Buy dollars if the price is lower than the current max money
            if (prices[i] < maxMoney) {
                dollars = b / prices[i];
                b -= dollars * prices[i];
            }
            // Sell dollars if the price is higher than the buying price
            if (prices[i] > maxMoney) {
                maxMoney += dollars * prices[i];
                dollars = 0;
            }
        }

        System.out.println(maxMoney);
    }
}
