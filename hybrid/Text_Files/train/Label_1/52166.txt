import java.util.Scanner;

public class CokeCocktail {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        int minVolume = -1;
        for (int i = 0; i < k; i++) {
            for (int j = i; j < k; j++) {
                if (a[i] + a[j] >= n) {
                    int volume = (int) Math.ceil((double) n / (a[i] + a[j] - n));
                    if (minVolume == -1 || volume < minVolume) {
                        minVolume = volume;
                    }
                }
            }
        }

        System.out.println(minVolume);
    }
}
