import sys

input = sys.stdin

def solve():
    n = int(input.readline())
    a = list(map(int, input.readline().split()))

    if sum(a) % 2 == 1:
        print(-1)
    else:
        for i in range(n):
            if a[i] % 2 == 1:
                print(1)
                print(i + 1)
                return
        print(2)
        print(1, 2)

for _ in range(int(input.readline())):
    solve()
