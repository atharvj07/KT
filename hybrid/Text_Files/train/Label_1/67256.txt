t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    x1, y1, x2, y2 = map(int, input().split())
    x3, y3, x4, y4 = map(int, input().split())

    total_cells = n * m
    white_cells = (total_cells + 1) // 2
    black_cells = total_cells // 2

    def count_cells(x1, y1, x2, y2):
        cells = (x2 - x1 + 1) * (y2 - y1 + 1)
        if (x1 + y1) % 2 == 0:
            return (cells + 1) // 2, cells // 2
        else:
            return cells // 2, (cells + 1) // 2

    white1, black1 = count_cells(x1, y1, x2, y2)
    white_cells += black1
    black_cells -= black1

    overlap_x1, overlap_y1 = max(x1, x3), max(y1, y3)
    overlap_x2, overlap_y2 = min(x2, x4), min(y2, y4)

    if overlap_x1 <= overlap_x2 and overlap_y1 <= overlap_y2:
        overlap_white, overlap_black = count_cells(overlap_x1, overlap_y1, overlap_x2, overlap_y2)
        white_cells -= overlap_black
        black_cells += overlap_black

    white3, black3 = count_cells(x3, y3, x4, y4)
    white_cells -= white3
    black_cells += white3

    print(white_cells, black_cells)
