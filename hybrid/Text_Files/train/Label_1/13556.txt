#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  int s, f;
  cin >> s >> f;
  int prefix[n + 1];
  for (int i = 1; i <= n; i++) {
    prefix[i] = prefix[i - 1] + a[i - 1];
  }
  int ans = 0;
  for (int i = s; i <= f; i++) {
    if (n - prefix[i] + prefix[i - s] > prefix[ans]) ans = i;
  }
  cout << ans << '\n';
  return 0;
}
