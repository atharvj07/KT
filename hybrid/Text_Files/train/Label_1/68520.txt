w=input().replace('0',' 0')+' 0'
n=int(input())
v=[]
for i,j in enumerate(w.split(),1):
    if j==' 1':v.extend([i]*i)
ans=[]
i=0
while n and i<len(v):
    if i+1<len(v) and v[i+1]==v[-1]:i+=1
    elif not ans or ans[-1]!=v[i]:
        n-=1;ans.append(v[i])
    i+=1
print(['NO','YES'][ans!=[]],*ans)
