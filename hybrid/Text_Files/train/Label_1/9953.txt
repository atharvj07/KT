#include <cmath>
#include <iostream>
#include <vector>

using namespace std;

const double EPS = 1e-9;

void swap(int &a, int &b) {
  int tmp = a;
  a = b;
  b = tmp;
}

int main() {
  int n, m;
  double ans = 0;
  cin >> n >> m;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
  }
  vector<pair<int, int>> moves(m);
  for (int i = 0; i < m; ++i) {
    cin >> moves[i].first >> moves[i].second;
  }
  for (int mask = 0; mask < (1 << m); ++mask) {
    vector<int> perm = p;
    for (int i = 0; i < m; ++i) {
      if ((mask >> i) & 1) {
        swap(perm[moves[i].first], perm[moves[i].second]);
      }
    }
    int inv = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (perm[i] > perm[j]) {
          ++inv;
        }
      }
    }
    ans += inv * (1.0 / (1 << m));
  }
  cout.precision(9);
  cout << fixed << ans << endl;
  return 0;
}
