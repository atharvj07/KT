#include <iostream>
#include <vector>
#include <map>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  map<int, vector<string>> mp;
  const string vowels = "eyuioa";
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    int cnt = 0;
    for (char c : s) cnt += c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
    mp[cnt].push_back(s);
  }
  int ans = 0;
  vector<pair<string, string>> res;
  for (auto &p : mp) {
    if (p.second.size() < 4) continue;
    for (int i = 0; i < p.second.size(); i++) {
      for (int j = i + 1; j < p.second.size(); j++) {
        if (p.second[i].back() == p.second[j].back()) {
          ans++;
          res.push_back({p.second[i], p.second[j]});
        }
      }
    }
    p.second.clear();
  }
  for (auto &p : mp) {
    if (p.second.size() < 2) continue;
    for (int i = 0; i < p.second.size(); i++) {
      for (auto &v : vowels) {
        for (int j = i + 1; j < p.second.size(); j++) {
          if (p.second[i].back() == v && p.second[j].back() == v) {
            ans++;
            res.push_back({p.second[i], p.second[j]});
          }
        }
      }
    }
    p.second.clear();
  }
  for (int i = 1; i <= 5; i++) {
    if (mp[i].size() < 4) continue;
    for (int j = 0; j < mp[i].size(); j++) {
      for (int k = j + 1; k < mp[i].size(); k++) {
        if (mp[i][j].back() == mp[i][k].back()) {
          int cnt = 0;
          for (char c : mp[i][j]) cnt += c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
          for (char c : mp[i][k]) cnt += c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
          if (cnt % 2 == 0) {
            ans++;
            res.push_back({mp[i][j], mp[i][k]});
          }
        }
      }
    }
    mp[i].clear();
  }
  cout << ans << '\n';
  for (auto &p : res) {
    cout << p.first << ' ' << p.second << '\n';
  }
  return 0;
}
