import java.util.Scanner;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N + 1];
        int[] B = new int[N + 1];
        int[] C = new int[N + 1];
        int X = sc.nextInt();
        for (int i = 0; i < X; i++) {
            A[sc.nextInt()] = 1;
        }
        int Y = sc.nextInt();
        for (int i = 0; i < Y; i++) {
            B[sc.nextInt()] = 1;
        }
        int Z = sc.nextInt();
        for (int i = 0; i < Z; i++) {
            C[sc.nextInt()] = 1;
        }
        int ans = 0;
        for (int i = 1; i <= N; i++) {
            if ((A[i] == 0 && C[i] > 0) || (B[i] > 0 && C[i] > 0)) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
