import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int r = input.nextInt();
        int k = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int[] prefix = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            prefix[i] = prefix[i - 1] + a[i - 1];
        }
        int answer = 0;
        for (int i = 0; i <= n; i++) {
            int l = Math.max(i - r, 0);
            int h = Math.min(i + r, n);
            answer = Math.max(answer, prefix[h] - prefix[l] + Math.min(k, r));
        }
        System.out.println(answer);
    }
}
