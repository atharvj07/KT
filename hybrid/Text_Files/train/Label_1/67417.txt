import sys

n, m, k, q = map(int, input().split())
a = [[0 for j in range(m+1)] for i in range(n+1)]
ans = 1e9
for i in range(q):
    x, y, t = map(int, input().split())
    if a[x][y] == 0:
        a[x][y] = t
        if a[x][y+1] and a[x+k][y] and a[x+k][y+1] and a[x][y] < ans:
            ans = a[x][y]
if ans == 1e9:
    ans = -1
print(ans)
