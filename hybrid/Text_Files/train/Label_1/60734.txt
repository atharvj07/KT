n, x = map(int, input().split())
vouchers = [list(map(int, input().split())) for _ in range(n)]

# Calculate the duration of each voucher and store it with its cost
for voucher in vouchers:
    voucher.append(voucher[1] - voucher[0] + 1)

# Sort vouchers by their duration
vouchers.sort(key=lambda v: v[3])

# Create a dictionary to hold the minimum cost for each duration
min_cost_for_duration = {}

# Variable to hold the answer
min_cost = float('inf')

for l, r, cost, duration in vouchers:
    # Check if there is a voucher with the complementary duration (x - duration)
    if x - duration in min_cost_for_duration:
        # Update the minimum cost if the sum of costs is less than the current minimum
        min_cost = min(min_cost, cost + min_cost_for_duration[x - duration])
    
    # Update the minimum cost for the current duration if it's the lowest so far
    if duration not in min_cost_for_duration or cost < min_cost_for_duration[duration]:
        min_cost_for_duration[duration] = cost

# Print the answer
print(min_cost if min_cost != float('inf') else -1)
