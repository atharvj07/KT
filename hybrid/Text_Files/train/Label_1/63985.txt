n,w=map(int,input().split())
a=list(map(int,input().split()))
ans=[(a[i]+1)//2 for i in range(n)]
rem=w-sum(ans)
mn=101
for i in range(n):
    if rem==0:break
    if a[i]>ans[i]:
        add=min(rem,a[i]-ans[i])
        ans[i]+=add
        rem-=add
        mn=min(mn,ans[i])
if sum(ans)==w:
    for i in range(n):
        if ans[i]==mn:ans[i]-=1
        print(ans[i],end=" ")
else:print(-1)
