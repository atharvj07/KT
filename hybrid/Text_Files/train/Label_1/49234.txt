import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        char[] arr = s.toCharArray();
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            pos[i] = -1;
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (arr[i] > arr[j]) {
                    pos[j] = Math.max(pos[j], i);
                }
            }
        }
        int idx = 0;
        while (k > 0) {
            if (pos[idx] == -1) {
                k -= n - idx;
                if (k == 0) {
                    break;
                }
            } else {
                k--;
                idx = pos[idx];
                int len = (n - idx) / 2;
                for (int i = idx; i < idx + len; i++) {
                    char tmp = arr[i];
                    arr[i] = arr[n - i - 1];
                    arr[n - i - 1] = tmp;
                }
            }
        }
        System.out.println(new String(arr));
    }
}
