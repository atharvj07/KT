n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]

eastern_rank = [sorted(set(row)) for row in a]
southern_rank = [sorted(set(a[i][j] for i in range(n))) for j in range(m)]

result = [[0] * m for _ in range(n)]

for i in range(n):
    for j in range(m):
        east_rank = eastern_rank[i].index(a[i][j]) + 1
        south_rank = southern_rank[j].index(a[i][j]) + 1
        max_rank = max(east_rank, south_rank)
        result[i][j] = max_rank + max(len(eastern_rank[i]) - east_rank, len(southern_rank[j]) - south_rank)

for row in result:
    print(' '.join(map(str, row)))
