import sys

a, n, m = map(int, input().split())
segments = [[int(x) for x in input().split()] for _ in range(n)]
umbrellas = [[int(x) for x in input().split()] for _ in range(m)]

rain_map = [0] * (a + 1)
for l, r in segments:
    rain_map[l] = 1
    rain_map[r] = -1

for i in range(1, a + 1):
    rain_map[i] += rain_map[i - 1]

umbrella_map = dict()
for x, p in umbrellas:
    if x not in umbrella_map:
        umbrella_map[x] = p
    else:
        umbrella_map[x] = min(umbrella_map[x], p)

min_fatigue = sys.maxsize

for j in range(1, a + 1):
    if rain_map[j] > 0:
        min_fatigue = min(min_fatigue, umbrella_map.get(j, 0))

if min_fatigue == sys.maxsize:
    print(-1)
else:
    print(min_fatigue)
