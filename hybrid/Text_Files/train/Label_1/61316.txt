import math

n, k = map(int, input().split())
c = list(map(int, input().split()))
a = []
for i in range(n):
    color = i + 1
    for j in range(k):
        index = c.index(color)
        c[index] = 0
        if j == 0:
            a.append(index + 1)
        elif a[-1] < index:
            a.append(index + 1)
    if a[-1] < len(c):
        a.append(len(c) + 1)
    for j in range(len(a)):
        a[j] = math.ceil(a[j] / k)
for i in range(n):
    print(*a[i * 2 : i * 2 + 2])
