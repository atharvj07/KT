#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 1e5 + 5, inf = 1e18;
int a[maxn];
signed main() {
    int n; cin >> n;
    int ans = 0, cur = 0;
    for (int i = 1; i <= 2 * n; i++) {
        cin >> a[i];
        if (a[i] == cur + 1) cur++;
        else ans += i - cur;
    }
    cout << ans << endl;
}
