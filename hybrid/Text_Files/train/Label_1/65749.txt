n = int(input())
s = list(input())
t = list(input())
if s == t: print(0); exit()
ans = []
for i in range(n-1):
    if s[i] == t[i]: continue
    j = s.index(t[i], i+1)
    for k in range(j-1, i, -1):
        s[k+1], s[k] = s[k], s[k+1]
        ans += [k]
    break
for i in range(n-1):
    if s[i] == t[i]: continue
    j = i+1
    while s[j] != t[j]: j += 1
    for k in range(j, i, -1):
        s[k+1], s[k] = s[k], s[k+1]
        ans += [k]
    break
if s == t: print(len(ans)); print(*ans); exit()
print(-1)
