#include <bits/stdc++.h>
using namespace std;
#define ll long long
map<ll, pair<int, int>> mp;
int main() {
  int k;
  cin >> k;
  for (int i = 1; i <= k; i++) {
    int n;
    cin >> n;
    ll sum = 0;
    for (int j = 1; j <= n; j++) {
      int x;
      cin >> x;
      sum += x;
      if (mp.find(sum) != mp.end()) {
        cout << "YES" << endl;
        cout << mp[sum].first << " " << mp[sum].second << endl;
        cout << i << " " << j << endl;
        return 0;
      } else {
        mp[sum] = {i, j};
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
