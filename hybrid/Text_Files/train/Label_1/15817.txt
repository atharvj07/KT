#include <algorithm>
#include <iostream>

using namespace std;

const long long mod = 1000000000 + 7;

int main() {
  int n;
  cin >> n;
  long long a[n], b[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  int m;
  cin >> m;
  sort(a, a + n);
  sort(b, b + n);
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int idx = lower_bound(b, b + n, a[i]) - b;
    ans += (n - idx);
    ans %= mod;
  }
  cout << ans << endl;
  return 0;
}
