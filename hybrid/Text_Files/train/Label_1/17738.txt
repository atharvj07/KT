#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll dp[20][110],a[20][110];
ll n,m;
ll rec(ll floor,ll pos){
    if(floor==n+1)return 0;
    if(dp[floor][pos]!=-1)return dp[floor][pos];
    dp[floor][pos]=1e18;
    dp[floor][pos]=min(dp[floor][pos],rec(floor+1,pos+1)+a[floor][pos+1]);
    dp[floor][pos]=min(dp[floor][pos],rec(floor+1,pos-1)+a[floor][pos-1]);
    return dp[floor][pos];
}
int main(){
    cin>>n>>m;
    memset(dp,-1,sizeof(dp));
    for(ll i=1;i<=n;i++){
        for(ll j=1;j<=m;j++){
            char c;
            cin>>c;
            if(c=='1')a[i][j]=1;
        }
    }
    ll ans=rec(1,1);
    cout<<ans<<endl;
    return 0;
}
