import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    static int n, m, k;
    static int[] a, b, c;
    static List<Pair>[] adj;
    static long[][] dp;
    static boolean flag;

    static class Pair implements Comparable<Pair> {
        int to, wt;

        Pair(int to, int wt) {
            this.to = to;
            this.wt = wt;
        }

        @Override
        public int compareTo(Pair o) {
            return o.wt - this.wt;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        a = new int[n + 1];
        b = new int[n + 1];
        c = new int[n + 1];
        adj = new List[n + 1];
        for (int i = 1; i <= n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            st = new StringTokenizer(br.readLine());
            a[i] = Integer.parseInt(st.nextToken());
            b[i] = Integer.parseInt(st.nextToken());
            c[i] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(new Pair(v, 0));
        }
        dp = new long[n + 1][5001];
        for (int i = 0; i <= n; i++)
            Arrays.fill(dp[i], -1);

        long ans = solve(1, k);
        if (flag)
            System.out.println(-1);
        else
            System.out.println(ans);
    }

    static long solve(int x, int army) {
        if (x == n + 1)
            return 0;
        if (dp[x][army] != -1)
            return dp[x][army];
        long op1 = -1, op2 = -1, op3 = -1;
        // go to the castle
        if (army >= a[x])
            op1 = solve(x + 1, army - a[x] + b[x]);
        // skip the castle
        op2 = solve(x + 1, army);
        // go to the castle and hire army
        if (army >= a[x])
            for (Pair p : adj[x])
                op3 = Math.max(op3, c[x] + solve(p.to, army - a[x] + b[x]));
        dp[x][army] = Math.max(Math.max(op1, op2), op3);
        if (op1 == -1 && op2 == -1 && op3 == -1)
            flag = true;
        return dp[x][army];
    }
}
