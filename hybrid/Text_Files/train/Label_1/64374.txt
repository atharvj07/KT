n = int(input())
s = list(input())
m = list(input())
s_f = [0 for i in range(10)]
m_f = [0 for i in range(10)]
for i in range(n):
    s_f[int(s[i])] += 1
    m_f[int(m[i])] += 1

fn = 0
fx = 0
for i in range(1, 10):
    if s_f[i] < m_f[i]:
        fn += s_f[i]
    else:
        fx += m_f[i]

print(fn, fx)
