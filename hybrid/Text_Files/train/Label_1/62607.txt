n = int(input())
a = list(map(int, input().split()))
psum = [0]
for x in a:
    psum.append(x + psum[-1])
L = [0] * n
R = [0] * n
c = 0; mx = 0
for i in range(1, n):
    if a[i] > a[c]:
        c = i
    L[i] = c
for i in range(n - 2, -1, -1):
    if a[i] >= a[c]:
        c = i
    R[i] = c
ans = 0
for i in range(1, n):
    if L[i] != R[i]:
        ans = max(ans, R[i] - L[i] + 1)
    else:
        ans = max(ans, R[i] - L[i])
print(ans)
