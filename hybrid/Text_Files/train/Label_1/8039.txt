#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        bool add = false;
        int sum = 0, mul = 1;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            sum += a[i];
            mul *= a[i];
            if (a[i] < 0) add = true;
        }
        int ans = -1;
        if (sum == 0 || mul == 0) ans = 1;
        if (!add) ans = 0;
        if (ans == -1) {
            if (sum < 0) {
                if (a[0] < 0) {
                    ans = 1;
                } else {
                    ans = 2;
                }
            } else if (mul < 0) {
                int x = 0;
                for (int i = 0; i < n; i++) {
                    if (a[i] < 0) x++;
                }
                if (x % 2 == 0) ans = 2;
                else ans = 1;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
