import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        in.nextLine();
        char[] pos = in.nextLine().toCharArray();
        int ans = 0;
        int cnt = 0;
        for (char c : pos) {
            if (c == 'X') cnt++;
        }
        int needStand = n / 2;
        if (cnt == needStand) {
            System.out.println(0);
            System.out.println(new String(pos));
            return;
        } else if (cnt < needStand) {
            for (int i = 0; i < n; i++) {
                if (pos[i] == 'X') continue;
                pos[i] = 'X';
                ans++;
                cnt++;
                if (cnt == needStand) break;
            }
        } else {
            for (int i = 0; i < n; i++) {
                if (pos[i] == 'x') continue;
                pos[i] = 'x';
                ans++;
                cnt--;
                if (cnt == needStand) break;
            }
        }
        System.out.println(ans);
        System.out.println(new String(pos));
    }
}
