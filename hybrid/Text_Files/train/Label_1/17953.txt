#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 10;
ll a[MAXN];
bool vis[MAXN];
int main() {
  ll n, ans1 = 0, ans2 = 0, tmp = 1;
  cin >> n;
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    if (!vis[i]) {
      bool flag = 1;
      ll cnt = 0;
      while (i < n && !vis[i]) {
        vis[i] = 1;
        tmp *= a[i++];
        if (a[i - 1] == 0) {
          flag = 0;
          cnt = 0;
        }
        cnt++;
        if (tmp < 0) {
          if (flag) ans1 += cnt * (cnt + 1) / 2;
        } else if (flag)
          ans2 += cnt * (cnt + 1) / 2;
      }
      tmp = 1;
    }
  }
  cout << ans1 << " " << ans2 << endl;
  return 0;
}
