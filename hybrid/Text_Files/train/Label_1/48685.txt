import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.HashMap;
import java.util.Map;

public class Main {
	public static void main(String[] args) throws IOException {
		BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
		String[] nk = reader.readLine().split(" ");
		int n = Integer.parseInt(nk[0]);
		int k = Integer.parseInt(nk[1]);
		long[][] arr = new long[n][2];
		for (int i = 0; i < n; i++) {
			String[] lr = reader.readLine().split(" ");
			arr[i][0] = Long.parseLong(lr[0]);
			arr[i][1] = Long.parseLong(lr[1]);
		}
		Arrays.sort(arr, Comparator.comparingLong(a -> a[0]));
		Map<Long, Integer> dp = new HashMap<>();
		int mod = 998244353;
		long ans = 0;
		for (long[] interval : arr) {
			long start = interval[0];
			long end = interval[1];
			if (dp.containsKey(start)) {
				ans += dp.get(start);
				ans %= mod;
			}
			if (dp.containsKey(end + 1)) {
				dp.put(end + 1, dp.get(end + 1) - 1);
				dp.put(end + 1, dp.get(end + 1) + mod);
				dp.put(end + 1, dp.get(end + 1) % mod);
			}
			dp.put(start, dp.getOrDefault(start, 0) + 1);
		}
		ans %= mod;
		if (ans < 0) {
			ans += mod;
		}
		System.out.println(ans);
	}
}
