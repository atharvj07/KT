import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        int[] cnt = new int[10];
        char[] ss = s.toCharArray();
        int l = 0, r = 0, d = 1, ans = 0;
        for (int i = 0; i < n; i++) {
            if (d == 1) {
                r++;
            } else {
                l--;
            }
            if (ss[i] == '<') {
                d = -1;
            } else if (ss[i] == '>') {
                d = 1;
            } else {
                int v = ss[i] - '0';
                ans++;
                cnt[v]++;
                if (v == 0 && d == 1) {
                    ss[i] = ' ';
                    r--;
                } else if (v == 0 && d == -1) {
                    ss[i] = ' ';
                    l++;
                } else {
                    ss[i]--;
                }
            }
        }
        for (int i = 0; i < q; i++) {
            int li = sc.nextInt() - 1;
            int ri = sc.nextInt() - 1;
            for (int j = 0; j < 10; j++) {
                System.out.print(cnt[j] * (ri - li + 1 - (li <= l && r <= ri ? ans : 0)) / n + " ");
            }
            System.out.println();
        }
    }
}
