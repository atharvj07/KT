#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
ll gcd(ll a, ll b) {
    return b ? gcd(b, a % b) : a;
}
ll lcm(ll a, ll b) {
    return (a / gcd(a, b)) * b;
}
int main() {
    ll t, l, r;
    cin >> t;
    while (t--) {
        cin >> l >> r;
        if (l * 2 > r) {
            cout << -1 << " " << -1 << endl;
            continue;
        }
        if (l % 2 == 1) {
            cout << l << " " << l + 1 << endl;
        } else {
            cout << l + 1 << " " << l + 2 << endl;
        }
    }
    return 0;
}
