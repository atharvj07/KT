#include <iostream>
#include <string>

using namespace std;

int main() {
	string s;
	int k;
	cin >> s >> k;
	int freq[26] = {};
	for (char c : s)
		freq[c - 'a']++;
	int cnt = 0;
	for (int i = 0; i < 26; i++)
		if (freq[i] == 0)
			cnt++;
	if (cnt > k)
		cout << "impossible";
	else
		cout << cnt;
	return 0;
}
