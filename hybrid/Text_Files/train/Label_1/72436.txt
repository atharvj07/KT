import sys
def landslide(n, heights):
    ans = [0] * n
    l = 0
    for r in range(n):
        while l < r and heights[l] + 2 <= heights[l + 1]:
            ans[l] += 1
            ans[l + 1] -= 1
            l += 1
        ans[r] += heights[r] - heights[l]
    return ans
n = int(input())
heights = list(map(int, sys.stdin.readline().split()))
print(*landslide(n, heights))
