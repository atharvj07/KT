#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
unordered_map<ll, int> a, b, c;
int main() {
    ll n, m;
    cin >> n;
    while (n--) {
        ll x;
        cin >> x;
        a[x]++;
    }
    cin >> m;
    while (m--) {
        ll x, y;
        cin >> x >> y;
        b[x]++, c[y]++;
    }
    ll x = -1, y = -1;
    for (auto i : a) {
        if (i.second > x) x = i.second, y = i.first;
        else if (i.second == x) y = min(y, i.first);
    }
    for (auto i : a) {
        if (i.second > x) x = i.second, y = i.first;
        else if (i.second == x) {
            if (b[i.first] + c[i.first] > b[y] + c[y]) y = i.first;
            else if (b[i.first] + c[i.first] == b[y] + c[y]) y = min(y, i.first);
        }
    }
    cout << y;
}
