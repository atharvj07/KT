import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt();
        if ((a + b) % k == 0) {
            System.out.println((a + b) / k);
        } else {
            System.out.println(-1);
        }
    }
}
