import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] mesh = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    mesh[j][k] = sc.nextInt();
                }
            }
            boolean isValid = true;
            int[][] sum = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    sum[j][k] = 0;
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    int l = 4 - j;
                    int m = 4 - k;
                    sum[j][k] += mesh[l][m];
                    sum[l][m] += mesh[j][k];
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (sum[j][k] != 7) {
                        isValid = false;
                    }
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (mesh[j][k] == 0) {
                        isValid = false;
                    }
                }
            }
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    for (int l = j + 1; l < 5; l++) {
                        for (int m = k + 1; m < 5; m++) {
                            if (mesh[j][k] == mesh[l][m]) {
                                isValid = false;
                            }
                        }
                    }
                }
            }
            if (isValid) {
                System.out.println("true");
            } else {
                System.out.println("false");
            }
        }
    }
}
