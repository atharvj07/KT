import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        long[] b = new long[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }
        scanner.close();

        while (n > 1 && m > 1) {
            if (a[n - 1] != b[m - 1]) {
                System.out.println(-1);
                return;
            }
            n--;
            m--;
        }

        System.out.println(Math.max(n, m));
    }
}
