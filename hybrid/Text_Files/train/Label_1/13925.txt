#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int i = 0; i < k; ++i) { pq.push(a[i] - x); }
    int ans = a[k - 1];
    for (int i = k; i < n; ++i) {
        ans += a[i];
        int time = pq.top();
        pq.pop();
        ans += time;
        pq.push(a[i] + time - x);
    }
    cout << ans;
    return 0;
}
