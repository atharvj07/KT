#include<iostream>
#include<vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> f(n + 1);
    for (int i = 1; i <= n; i++) { cin >> f[i]; }
    bool found = false;
    for (int m = 1; m <= n; m++) {
        vector<bool> used(m + 1);
        vector<int> g(n + 1), h(m + 1);
        bool ok = true;
        for (int i = 1; i <= n && ok; i++) {
            if (used[f[i]]) { ok = false; continue; }
            g[i] = f[i];
            used[f[i]] = true;
        }
        if (ok) {
            for (int i = 1; i <= n; i++) { h[g[i]] = i; }
            for (int i = 1; i <= n; i++) {
                if (f[h[g[i]]] != i) { ok = false; break; }
            }
        }
        if (ok) {
            cout << m << endl;
            for (int i = 1; i <= n; i++) { cout << g[i] << " "; }
            cout << endl;
            for (int i = 1; i <= m; i++) { cout << h[i] << " "; }
            cout << endl;
            found = true;
            break;
        }
    }
    if (!found) { cout << "-1" << endl; }
    return 0;
}
