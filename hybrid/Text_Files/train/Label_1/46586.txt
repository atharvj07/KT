import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum * 2 >= sumTotal(a)) {
                System.out.println(i + 1);
                return;
            }
        }
    }

    public static int sumTotal(int[] a) {
        int sum = 0;
        for (int i : a) {
            sum += i;
        }
        return sum;
    }
}
