#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, ans = 0;
  long long sum = 0;
  cin >> n >> k;
  vector<long long> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  sort(v.rbegin(), v.rend());
  for (auto x : v) {
    if (sum + x <= k) {
      ans++;
      sum += x;
    }
  }
  cout << ans << endl;
  return 0;
}
