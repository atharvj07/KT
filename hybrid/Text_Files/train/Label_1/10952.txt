#include<bits/stdc++.h>
using namespace std;
const int N=200005;
int n,q,a[N],ans[N];
char s[N];
int main(){
    cin>>n>>s>>q;
    for(int i=1;i<=n;i++){
        if(s[i]!=s[i-1])a[i]=a[i-1]+1;
        else a[i]=a[i-1];
    }
    for(int i=1;i<=q;i++){
        int x;
        char y;
        cin>>x>>y;
        int mx=a[n];
        for(int j=x;j<=n;j++){
            if(s[j]==y)mx=max(mx,a[j]-a[j-x]);
        }
        ans[i]=mx;
    }
    for(int i=1;i<=q;i++){
        cout<<ans[i]<<"\n";
    }
    return 0;
}
