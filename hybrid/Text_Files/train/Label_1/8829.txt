#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> x(n);
  vector<int> y;
  long long sum_x = 0;
  for (int i = 0; i < n; i++) {
    cin >> x[i];
    sum_x += x[i];
  }
  sort(x.rbegin(), x.rend());
  long long sum_y = 0;
  int m = 0;
  for (int i = 1000000; i > 0; i--) {
    if (count(x.begin(), x.end(), i) == 0) {
      sum_y += i;
      y.push_back(i);
      m++;
      if (sum_y > sum_x) {
        break;
      }
    }
  }
  cout << m << endl;
  for (int i = 0; i < m; i++) {
    cout << y[i] << " ";
  }
  cout << endl;
  return 0;
}
