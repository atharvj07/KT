import sys

md = 10**9 + 7

def mul(a, b):
    return (a * b) % md

def binpow(a, b):
    if b == 0:
        return 1
    elif b % 2 == 0:
        c = binpow(a, b // 2)
        c = mul(c, c)
        return c
    else:
        return mul(binpow(a, b-1), a)

def solve(n):
    return binpow(3, n)

n = int(sys.stdin.readline())
print(solve(n))
