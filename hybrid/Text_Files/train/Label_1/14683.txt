#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, p, m;
  cin >> n >> p >> m;
  vector<pair<int, int>> deposits(n);
  for (int i = 0; i < n; i++) {
    cin >> deposits[i].first >> deposits[i].second;
  }
  int cur = 0, ans = 0;
  for (int i = 1; i <= m; i++) {
    int left = p;
    while (cur < n && deposits[cur].first == i) {
      left -= deposits[cur].second;
      cur++;
    }
    if (left > 0) ans++;
  }
  cout << ans << endl;
  return 0;
}
