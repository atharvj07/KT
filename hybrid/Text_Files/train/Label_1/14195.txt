#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 128;
const int MAXM = 10000;

char a[MAXN][MAXM];
int n, m;

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  int x = 0, y = 0, dir = 1, ans = 0;
  while (1) {
    if (x == n - 1 && y == 0) {
      break;
    }
    ++ans;
    if (a[x + dir][y] == '.' || a[x + dir][y] == '+') {
      x += dir;
    } else if (a[x + dir][y] == '#') {
      dir *= -1;
    } else {
      x -= 1;
      y = 0;
      dir = 1;
    }
  }
  if (ans > 1e9) {
    cout << "Never" << endl;
  } else {
    cout << ans << endl;
  }
  return 0;
}
