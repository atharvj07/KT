s = input()
p = input()
n = len(s)
m = len(p)
dp = [[0 for j in range(m + 1)] for i in range(n + 1)]
for i in range(n - 1, -1, -1):
    for j in range(m):
        if s[i] == p[j]:
            dp[i][j] = dp[i + 1][j + 1] + 1
        else:
            dp[i][j] = max(dp[i + 1][j], dp[i][j + 1])
ans = [dp[i][0] for i in range(n + 1)]
print(' '.join(map(str, ans)))
