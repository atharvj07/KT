#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  if (k > 2 * n - 2)
    cout << -1;
  else {
    if (k == 2 * n - 2) {
      if (n % 2 == 0)
        for (int j = 0; j < n; j++) {
          cout << (j + 1) % n + 1 << " ";
        }
      else {
        for (int j = 0; j < n - 5; j++) {
          cout << (j + 1) % n + 1 << " ";
        }
        cout << "1 3 7 6 5 2\n1 2 6 5 4 3";
      }
    } else {
      for (int j = 0; j < n - 3; j++) {
        cout << (j + 1) % n + 1 << " ";
      }
      cout << "1 3 2 4 ";
      if (n % 2 == 0) {
        cout << n << endl;
        for (int j = n - 1; j >= 0; j--) {
          cout << j + 1 << " ";
        }
      } else {
        cout << "5\n1 2 6 5 3 4";
      }
    }
  }
}
