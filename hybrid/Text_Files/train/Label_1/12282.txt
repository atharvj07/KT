#include <bits/stdc++.h>

using namespace std;

const int MOD = 998244353;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  sort(b.begin(), b.end(), [&](const long long &x, const long long &y) {
    if (x * y < 0) {
      return x > y;
    } else {
      return x < y;
    }
  });
  long long ans = 0, sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i] * b[i];
    ans += sum;
  }
  ans %= MOD;
  cout << ans << endl;
  return 0;
}
