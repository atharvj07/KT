import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
if sum(a) == 0:
    print("NO")
    sys.exit()
pref = [0] * (n + 2)
for i in range(n):
    pref[i + 1] = pref[i] + a[i]
ans = []
cur = 0
for i in range(n):
    if pref[i] == pref[n]:
        continue
    if pref[i] == pref[cur]:
        ans.append((cur + 1, i))
        cur = i + 1
if len(ans) == 0:
    print("NO")
else:
    print("YES")
    print(len(ans))
    for l, r in ans:
        print(l, r)
