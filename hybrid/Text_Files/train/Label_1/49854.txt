import java.util.Scanner;

public class ArrayError {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k1 = scanner.nextInt();
        int k2 = scanner.nextInt();

        int[] a = new int[n];
        int[] b = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        System.out.println(solve(a, b, k1, k2));
    }

    private static long solve(int[] a, int[] b, int k1, int k2) {
        long[][] dp = new long[k1 + 1][k2 + 1];

        for (int i = 0; i <= k1; i++) {
            for (int j = 0; j <= k2; j++) {
                if (i == 0 && j == 0) {
                    dp[i][j] = calculateError(a, b);
                } else if (i > 0) {
                    dp[i][j] = Math.min(dp[i - 1][j] + calculateErrorDiff(a, i - 1, 1), dp[i][j]);
                }
                if (j > 0) {
                    dp[i][j] = Math.min(dp[i][j - 1] + calculateErrorDiff(b, j - 1, -1), dp[i][j]);
                }
            }
        }

        return dp[k1][k2];
    }

    private static long calculateError(int[] a, int[] b) {
        long error = 0;
        for (int i = 0; i < a.length; i++) {
            error += (long) Math.pow(a[i] - b[i], 2);
        }
        return error;
    }

    private static long calculateErrorDiff(int[] arr, int index, int diff) {
        long errorDiff = 0;
        for (int i = 0; i < arr.length; i++) {
            if (i == index) {
                errorDiff += (long) Math.pow(arr[i] + diff - arr[i], 2) - (long) Math.pow(arr[i] - arr[i], 2);
            } else {
                errorDiff += (long) Math.pow(arr[i] - arr[i], 2);
            }
        }
        return errorDiff;
    }
}
