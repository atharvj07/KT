import sys
from collections import defaultdict
from queue import PriorityQueue

input = sys.stdin.readline

n = int(input())
adj = defaultdict(list)
for _ in range(n - 1):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    adj[a].append(b)
    adj[b].append(a)

cost = []
target = []
for _ in range(n):
    a, b, c = map(int, input().split())
    cost.append(a)
    target.append(c)

q = PriorityQueue()
q.put((cost[0], 0))
vis = [False for i in range(n)]
ans = 0

while not q.empty():
    cur, node = q.get()
    if vis[node]:
        continue
    vis[node] = True
    ans += cur

    if target[node] == 0 and cost[node] % 2 == 1:
        print(-1)
        exit()

    for u in adj[node]:
        if not vis[u]:
            q.put((cost[node], u))

print(ans)
