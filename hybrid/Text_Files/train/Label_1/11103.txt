#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;

  std::vector<int> times;
  for (int i = 0; i < n; i++) {
    std::string s;
    std::cin >> s;
    int h, m;
    std::sscanf(s.c_str(), "%d:%d", &h, &m);
    times.push_back(h * 60 + m);
  }

  std::sort(times.begin(), times.end());

  int longest_sleep = 0;
  int start = times[0];
  for (int i = 1; i < times.size(); i++) {
    if (times[i] - start > longest_sleep) {
      longest_sleep = times[i] - start;
    }
    start = times[i];
  }

  if (24 * 60 - start > longest_sleep) {
    longest_sleep = 24 * 60 - start;
  }

  int h = longest_sleep / 60;
  int m = longest_sleep % 60;
  std::printf("%02d:%02d\n", h, m);

  return 0;
}
