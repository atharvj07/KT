#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    int k1;
    cin >> k1;
    vector<int> v1(k1);
    for (int i = 0; i < k1; i++) {
        cin >> v1[i];
    }
    int k2 = n - k1;
    vector<int> v2(k2);
    for (int i = 0; i < k2; i++) {
        cin >> v2[i];
    }
    int cnt = 0, win = 0;
    while (!v1.empty() && !v2.empty()) {
        cnt++;
        if (v1.back() > v2.back()) {
            v1.insert(v1.end(), v2.back());
            v2.pop_back();
            v1.insert(v1.end(), v1.back());
            v1.pop_back();
            win = 1;
        } else {
            v2.insert(v2.end(), v1.back());
            v1.pop_back();
            v2.insert(v2.end(), v2.back());
            v2.pop_back();
            win = 2;
        }
    }
    if (!v1.empty()) {
        cout << cnt << " " << win << endl;
        return 0;
    }
    if (!v2.empty()) {
        cout << cnt << " " << win << endl;
        return 0;
    }
    cout << "-1" << endl;
    return 0;
}
