#include <bits/stdc++.h>
using namespace std;
#define ll long long int
int main() {
  ll n, a, b, ans = 0;
  vector<pair<ll, ll>> arr;
  cin >> n;
  for (ll i = 0; i < n; i++) {
    cin >> a >> b;
    arr.push_back({a, b});
  }
  sort(arr.begin(), arr.end());
  ans = arr[0].first * 2;
  ll ct = arr[0].first;
  for (ll i = 1; i < n; i++) {
    if (arr[i].second <= ct) {
      ans += arr[i].first;
    } else {
      ans += (arr[i].second - ct) + arr[i].first;
      ct = arr[i].second;
    }
  }
  cout << ans << endl;
  return 0;
}
