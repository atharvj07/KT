#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    long long a, b, x, y, n;
    std::cin >> a >> b >> x >> y >> n;
    while (a > x && b > y && n > 0) {
      if (a - x < b - y) {
        a--;
      } else {
        b--;
      }
      n--;
    }
    std::cout << a * b << std::endl;
  }
  return 0;
}
