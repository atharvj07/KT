import sys
n,sx,sy=map(int,sys.stdin.readline().split())
l=[]
dic={}
for i in range(n):
    a,b=map(int,sys.stdin.readline().split())
    l.append((a,b,i))
    dic[i]=-1
ans=0
ansp=[]
for x, y,i in l:
    if (x+1==sx and y==sy) or (x-1==sx and y==sy) or (x==sx and y+1==sy) or (x==sx and y-1==sy): dic[i]=i
    elif (x+1==sx and y==sy) and (x+1,y) not in dic: dic[i]=i
    elif (x-1==sx and y==sy) and (x-1,y) not in dic: dic[i]=i
    elif (x==sx and y+1==sy) and (x,y+1) not in dic: dic[i]=i
    elif (x==sx and y-1==sy) and (x,y-1) not in dic: dic[i]=i
    if dic[i]!=-1:
        ans+=1
        ansp.append(i+1)
print(ans)
if ansp:
    print(*l[min(ansp)-1][0:2])
