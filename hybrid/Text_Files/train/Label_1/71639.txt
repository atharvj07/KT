for _ in range(int(input())):
    s = input()
    last = ''
    for i in range(len(s)):
        if s[i] != '?' and last == s[i]:
            print(-1)
            break
        elif s[i] != '?':
            last = s[i]
    else:
        res = ''
        used = {last: 1}
        for c in s:
            if c == '?':
                for d in 'abc':
                    if not (used.get(d) or d == last):
                        res += d
                        used[d] = 1
                        last = d
                        break
            else:
                res += c
                used[c] = 1
                last = c
        print(res)
