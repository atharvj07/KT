#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    map<char, int> cnt;
    for (char c : s) {
        cnt[c]++;
    }
    if (cnt['z'] < cnt['o'] || cnt['z'] < cnt['r']) {
        cout << 0 << endl;
        return 0;
    }
    int numZeros = min(cnt['z'], min(cnt['e'], cnt['r']));
    for (int i = 0; i < numZeros; i++) {
        cout << 0 << " ";
    }
    cnt['z'] -= numZeros;
    cnt['e'] -= numZeros;
    cnt['r'] -= numZeros;
    int numOnes = 0;
    if (cnt['o'] > cnt['n']) {
        numOnes = cnt['n'];
    } else {
        numOnes = cnt['o'];
    }
    for (int i = 0; i < numOnes; i++) {
        cout << 1 << " ";
    }
    cout << endl;
    return 0;
}
