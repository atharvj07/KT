#include<bits/stdc++.h>
using namespace std;
typedef long long int lli;
typedef pair<lli,lli> pi;
map<lli,lli>dd;
set<lli>adj[1000000];
lli mark[1000000];

void dfs(lli s,lli p=-1){
	mark[s]=1;
	for(auto u:adj[s]){
		if(u!=p){
			dd[s]++;
			if(dd[s]<=2) dfs(u,s);
		}
	}
}

int main(){
	lli n,k;
	cin>>n>>k;
	for(lli i=0;i<n-1;i++){
		lli u,v;
		cin>>u>>v;
		adj[u].insert(v);
		adj[v].insert(u);
	}
	lli mx=-1,id=-1;
	for(lli i=1;i<=n;i++){
		if(adj[i].size()>mx){
			mx=adj[i].size();
			id=i;
		}
	}
	if(mx<3 || k==1){
		cout<<"No"<<endl;
		return 0;
	}
	dfs(id);
	lli c=0;
	for(lli i=1;i<=n;i++){
		if(mark[i]==0) c++;
	}
	if(c!=0){
		cout<<"No"<<endl;
		return 0;
	}
	cout<<"Yes"<<endl;
	return 0;
}
