#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll MAXN = 120;
vector<vector<ll>> G(MAXN);
vector<bool> vis(MAXN);
vector<ll> match(MAXN);
ll n, m, ans;

ll dfs(ll u) {
  for (auto v : G[u]) {
    if (vis[v]) continue;
    vis[v] = true;
    if (match[v] == 0 || dfs(match[v])) {
      match[v] = u;
      return true;
    }
  }
  return false;
}

int main() {
  cin >> n;
  for (ll i = 1; i <= 2 * n; i++) {
    ll x;
    cin >> x;
    G[x].push_back(i);
    G[i].push_back(x);
  }
  for (ll i = 1; i <= 2 * n; i++) {
    if (!vis[i]) {
      vis[i] = 1;
      if (dfs(i)) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
