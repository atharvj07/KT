import java.util.Scanner;

public class Main {

    private static int n, s;
    private static int[] a;
    private static int[] sup;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        s = in.nextInt();
        a = new int[n + 1];
        sup = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            if (i == s) {
                sup[i] = 0;
            } else {
                sup[i] = -1;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (sup[i] == -1) {
                if (dfs(i, 0)) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }

    private static boolean dfs(int u, int cnt) {
        if (cnt > a[u]) {
            return true;
        }
        if (sup[u] != -1) {
            return sup[u] == 1;
        }
        sup[u] = 0;
        if (dfs(a[u], cnt + 1)) {
            sup[u] = 1;
            return true;
        }
        return false;
    }
}
