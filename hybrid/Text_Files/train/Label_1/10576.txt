#include<bits/stdc++.h>
using namespace std;
main()
{
    int n,m,x,i,j,ok=0;
    string s[505];
    cin>>n>>m;
    for(i=0; i<n; i++)cin>>s[i];
    for(i=0; i<n; i++)
    {
        for(j=0; j<m; j++)
        {
            if(s[i][j]=='.')
            {
                ok=1;
                if((i==0||s[i-1][j]!='S')&&(j==0||s[i][j-1]!='S')&&(i==n-1||s[i+1][j]!='S')&&(j==m-1||s[i][j+1]!='S'))
                {
                    cout<<"Yes\n";
                    s[i][j]='D';
                    for(i=0; i<n; i++)cout<<s[i]<<'\n';
                    return 0;
                }
            }
        }
    }
    if(ok==0||(ok==1&&i==n))cout<<"No\n";
}
