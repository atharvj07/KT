#include<bits/stdc++.h>
using namespace std;
int a0[200005],sum[200005];
int suf[200005];
int main()
{
    int n,k,i,j,l,mn,sm,ans;
    cin>>n>>k;
    for(i=0; i<n; i++)
    {
        cin>>a0[i];
    }
    for(i=0; i<n; i++)
    {
        sum[i+1]=a0[i]+sum[i];
    }
    sm=0;
    for(i=n-1; i>=0; i--)
    {
        sm=sm+a0[i];
        suf[i]=sm;
    }
    mn=100000000;
    for(i=0; i<n; i++)
    {
        if(sum[i]>=k)
        {
            mn=0;
            break;
        }
        else
        {
            j=i;
            while(sum[i]+suf[j]<k)
            {
                if(j==n-1)
                {
                    break;
                }
                j++;
                i++;
            }
        }
        if(i+1<mn)
        {
            mn=i+1;
        }
    }
    cout<<mn<<endl;
    return 0;
}
