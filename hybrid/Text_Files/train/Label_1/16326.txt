#include <iostream>
const int base=1000;

int numdigit(int num){
  int ans=0;
  while(num>0){
    ans+=num%10!=0;
    num/=10;
  }
  return ans;
}

int main(){
  int t;
  std::cin>>t;
  while(t--){
    int l,r;
    std::cin>>l>>r;
    int ans=0;
    for(;l<=r;l++){
      ans+=numdigit(l)<=3;
    }
    std::cout<<ans<<'\n';
  }
}
