import sys, heapq
t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    adj = [[] for i in range(n + 1)]
    for i in range(m):
        x, y, z = map(int, input().split())
        adj[x].append((y, z))
        adj[y].append((x, z))
    vis = [False] * (n + 1)
    inc = sys.maxsize
    dec = sys.maxsize
    for i in range(1, n + 1):
        if not vis[i]:
            st = [i]
            vis[i] = True
            while st:
                node = st.pop(0)
                for j in adj[node]:
                    if not vis[j[0]]:
                        st.append(j[0])
                        vis[j[0]] = True
                        if j[1] < k:
                            inc = min(inc, k - j[1])
                        else:
                            dec = min(dec, j[1] - k)
    ans = inc + dec
    if dec == sys.maxsize and inc == sys.maxsize:
        print(-1)
    elif dec < inc:
        print(dec)
    else:
        print(inc)
