import java.util.Arrays;
import java.util.Scanner;
import java.util.HashMap;
import java.lang.Math;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // number of tiles in your hand
        int m = sc.nextInt(); // number of tiles types
        int[] a = new int[n]; // the number written on the i-th tile
        HashMap<Integer, Integer> tiles = new HashMap<>();
        for (int i = 0 ; i < n ; i++ ) {
            a[i] = sc.nextInt();
            tiles.put(a[i], tiles.getOrDefault(a[i], 0) + 1);
        }
        
        int ans = 0;
        for (int num=1 ; num<=m ; num++) {
            if (tiles.containsKey(num)) {
                int cnt = tiles.get(num);
                ans += (cnt / 3) *3;
                tiles.put(num, cnt%3);
            }
        }
        System.out.println(ans);
    }
}
