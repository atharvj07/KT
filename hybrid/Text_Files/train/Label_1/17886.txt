#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    if (k == 1) {
      for (int i = 0; i < n; i++) s += 'a';
    } else if (k == 2) {
      for (int i = 0; i < n / 2; i++) s += "ab";
    } else {
      for (int i = 0; i < n / 3; i++) s += "abc";
      for (int i = 0; i < n % 3; i++) s += "a";
    }
    cout << s << endl;
  }
}
