#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 2e5 + 5;

int n, a[N], pos[N];
ll ans;

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> a[i];
        memset(pos, 0, sizeof(pos));
        ans = 1LL * n * (n + 1) / 2;
        for (int i = 1; i <= n; i++) {
            int prev = pos[a[i]];
            if (prev) {
                ans -= 1LL * (i - prev) * (i - prev - 1) / 2;
                ans -= 1LL * (prev - 1) * (n - i) / 2;
            }
            pos[a[i]] = i;
        }
        cout << ans << '\n';
    }
    return 0;
}
