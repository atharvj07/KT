#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x;
    cin >> n;
    deque<int> a;
    stack<int> ok;
    for (int i = 0; i < n; i++) cin >> x, a.push_back(x);
    for (int i = 0; i < n; i++) {
        x = a.front();
        while (ok.size() && ok.top() > x) {
            cout << ok.top() << endl;
            ok.pop();
        }
        ok.push(a.front());
        a.pop_front();
    }
    while (ok.size()) {
        cout << ok.top() << endl;
        ok.pop();
    }
    return 0;
}
