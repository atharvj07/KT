import java.util.Scanner;

public class PerfectMatching {
    private static final int MODULO = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        long[][] probabilities = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                probabilities[i][j] = scanner.nextLong();
            }
        }

        long[][] dp = new long[1 << n][n + 1];
        dp[0][0] = 1;
        for (int mask = 0; mask < (1 << n); mask++) {
            for (int i = 0; i <= n; i++) {
                if (dp[mask][i] == 0) {
                    continue;
                }

                for (int j = 0; j < n; j++) {
                    if (((1 << j) & mask) != 0) {
                        continue;
                    }

                    long probability = probabilities[i][j] * dp[mask][i];
                    dp[mask | (1 << j)][i + 1] = (dp[mask | (1 << j)][i + 1] + probability) % MODULO;
                }
            }
        }

        long numerator = dp[(1 << n) - 1][n];
        long denominator = 1;
        for (int i = 1; i <= (2 * n); i++) {
            denominator = (denominator * i) % MODULO;
        }

        long inverseDenominator = modInverse(denominator, MODULO);
        long answer = (numerator * inverseDenominator) % MODULO;
        System.out.println(answer);
    }

    private static long modInverse(long a, long m) {
        long m0 = m;
        long y = 0, x = 1;

        if (m == 1) {
            return 0;
        }

        while (a > 1) {
            long q = a / m;
            long t = m;

            m = a % m;
            a = t;
            t = y;

            y = x - q * y;
            x = t;
        }

        if (x < 0) {
            x += m0;
        }

        return x;
    }
}
