#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> f(n), b(m);
  for (int &x : f) cin >> x;
  for (int &x : b) cin >> x;
  vector<int> perm(n);
  for (int i = 0; i < n; i++) {
    perm[f[i] - 1] = i + 1;
  }
  bool possible = true, ambiguous = false;
  for (int i = 0; i < m; i++) {
    int a = perm[b[i] - 1];
    if (a == 0)
      possible = false;
    else if (possible) {
      if (a != perm[f[b[i] - 1] - 1])
        ambiguous = true;
      perm[b[i] - 1] = 0;
    }
  }
  if (possible) {
    if (ambiguous)
      cout << "Ambiguity" << endl;
    else {
      cout << "Possible" << endl;
      for (int i = 0; i < m; i++) {
        int a = perm[b[i] - 1];
        cout << a << " ";
        perm[b[i] - 1] = 0;
      }
      cout << endl;
    }
  } else {
    cout << "Impossible" << endl;
  }
  return 0;
}
