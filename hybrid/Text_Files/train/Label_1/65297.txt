import sys

n = int(input())
players = {}
for _ in range(n):
    m, *arr = map(int, input().split())
    players[m] = players.get(m, 0) + 1
    for a in arr:
        players[a] = players.get(a, 0) + 1

win = False
for k, v in players.items():
    if v == n:
        win = True
        break

for i in range(1, 101):
    if players.get(i, 0) == n - 1:
        win = True
        break

print(*['YES' if players.get(i, 0) == n or win else 'NO' for i in range(1, n + 1)], sep='\n')
