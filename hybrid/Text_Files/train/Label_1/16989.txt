#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while(t--){
        int n, m; cin >> n >> m;
        vector<vector<int>>a(n, vector<int>(m));
        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                int x; cin >> x;
                a[i][j] = x;
            }
        }
        int ans = 0;
        for(int i = 0; i < n; i++){
            int mx = INT_MIN;
            for(int j = 0; j < m; j++){
                mx = max(mx, a[i][j]);
            }
            ans += mx;
        }
        cout << ans << '\n';
    }
    return 0;
}
