#include <bits/stdc++.h>
using namespace std;

#define forn(i, n) for (int i = 0; i < (int)(n); ++i)
#define for1(i, n) for (int i = 1; i <= (int)(n); ++i)

typedef long long ll;
typedef unsigned long long ull;

const int maxn = 1e5;
int a[maxn];

int main() {
    int n, k;
    cin >> n >> k;
    iota(a, a + n, 0);
    vector<vector<int>> ans;
    ans.resize(k);
    for (int day = 0; day < k; ++day) {
        int idx = day % n;
        auto &v = ans[day];
        v.push_back(a[idx]);
        for (int i = n - 1 - idx; i >= 0; --i) {
            if (a[idx] != a[i]) {
                v.push_back(a[i]);
                swap(a[idx], a[i]);
            }
        }
    }
    for (const auto &v : ans) {
        cout << v.size() << ' ';
        for (const auto &el : v) cout << el << ' ';
        cout << '\n';
    }
    return 0;
}
