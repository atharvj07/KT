n, k, p = map(int, input().split())
a = list(map(int, input().split()))
b = sorted(map(int, input().split()))
time = 0
a.sort()
for ai in a:
    li = 0
    ri = k - 1
    while li <= ri:
        mi = (li + ri) // 2
        if b[mi] >= ai:
            ri = mi - 1
        else:
            li = mi + 1
    if li >= k:
        time = -1
        break
    time = max(time, abs(p - b[li]) + abs(b[li] - ai))
print(time)
