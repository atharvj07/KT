import sys
input = sys.stdin.readline

n = int(input())
C = list(map(int, input().split()))
m = int(input())
edges = [[] for i in range(n)]
for i in range(m):
    u, v = map(int, input().split())
    edges[u - 1].append(v - 1)
    edges[v - 1].append(u - 1)

dp = [[-1] * 3 for i in range(n)]

def dfs(u, p, s):
    if dp[u][s] != -1:
        return dp[u][s]

    res = 10 ** 18
    cnt = 0
    skip = 0

    if s == 0:
        skip = 1
        res = C[u]
        cnt = 1

    for v in edges[u]:
        if v == p:
            continue

        if v == u:
            if s == 1:
                skip += 1
                res += C[u]
                cnt += 1
            elif s == 2:
                res += C[u]
                cnt += 1
        elif skip == 0:
            res = min(res, dfs(v, u, 0))
        elif skip == 1:
            res = min(res, dfs(v, u, 1))
        elif skip == 2:
            res = min(res, dfs(v, u, 2))

        cnt = (cnt + dfs(v, u, 0)) % (10 ** 9 + 7)


    dp[u][s] = (res, cnt)
    return dp[u][s]

res = dfs(0, -1, 0)
print(*res)
