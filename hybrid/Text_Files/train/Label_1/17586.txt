#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, c;
  cin >> n >> c;
  vector<int> v(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  int ans = 0, last = 0;
  for (int i = 0; i < n; i++) {
    if (v[i] - last > c) ans = 1;
    last = v[i];
    ans++;
  }
  cout << ans << endl;
  return 0;
}
