import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long c = sc.nextLong(), hr = sc.nextLong(), hb = sc.nextLong(), wr = sc.nextLong(), wb = sc.nextLong();
        System.out.println((long) Math.floor(c / wb) * hb + ((long) Math.min(hr * (c / wr), c % wr / wb)) * hb);
    }
}
