#include <iostream>
using namespace std;

int main() {
  long long S, ans = 1e18;
  cin >> S;
  string s;
  cin >> s;
  for (long long i = 0; i < s.size(); ++i) {
    if (s[i] - '0' < S) {
      ans = 0;
      break;
    }
    if (s[i] - '0' > S) {
      ans = min(ans, (long long)i + 1);
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
