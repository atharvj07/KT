#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    unordered_map<string, int> umap;
    for (int i = 0; i < n; i++) {
      string s; cin >> s;
      int cnt[26] = {0};
      for (char& c: s) cnt[c - 'a']++;
      string hashed = "";
      for (int i = 0; i < 26; i++) hashed += to_string(cnt[i]);
      umap[hashed]++;
    }
    int mx = 0;
    for (auto& p: umap) mx = max(mx, p.second);
    cout << (mx == n ? "YES\n" : "NO\n");
  }
  return 0;
}
