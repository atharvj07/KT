t = int(input())
for _ in range(t):
    n = int(input())
    divisors = list(map(int, input().split()))
    divisors.sort()
    x = divisors[0] * divisors[-1]
    actual_divisors = []
    for i in range(1, int(x**0.5) + 1):
        if x % i == 0:
            if i*i == x:
                actual_divisors.append(i)
            else:
                actual_divisors.append(i)
                actual_divisors.append(x // i)
    if sorted(actual_divisors) == sorted(divisors + [1, x]):
        print(x)
    else:
        print(-1)
