#include <bits/stdc++.h>
using namespace std;
int d[100],n,ans;vector<int> v;bool u[100];
int main(){
    int i,j;
    cin>>n;
    for(i=0;i<n;i++) cin>>d[i];
    sort(d,d+n);
    for(i=0;i<n;i++){
        if(d[i]%10!=0 && d[i]!=10){
            v.push_back(d[i]);
            ans++;
            u[i]=true;
        }
        if(d[i]==10){
            v.push_back(d[i]);
            ans++;
            for(j=i+1;j<n;j++) if(!u[j]){ v.push_back(d[j]),ans++,u[j]=true; }
            break;
        }
    }
    if(ans==0){
        cout<<"1\n10\n";
        return 0;
    }
    cout<<ans<<'\n';
    for(i=0;i<n;i++) if(u[i]) cout<<d[i]<<' ';
}
