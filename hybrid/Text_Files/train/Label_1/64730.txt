import math

x1, y1, x2, y2 = map(int, input().split())
vmax, t = map(int, input().split())
v = tuple(map(int, input().split()))
w = tuple(map(int, input().split()))

def dist(p1, p2):
    return math.sqrt((p1[0] - p2[0]) ** 2 + (p1[1] - p2[1]) ** 2)

def time(p):
    d = dist((x1, y1), (x2, y2))
    return d / max(vmax, p)

print(min(time(math.sqrt(v[0] ** 2 + v[1] ** 2)), time(t), time(math.sqrt(w[0] ** 2 + w[1] ** 2))))
