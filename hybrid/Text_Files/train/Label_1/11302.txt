#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    int n, sum = 0, a;
    cin >> n;
    unordered_map<int, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> a;
        if (a) mp[a]++;
    }
    for (auto x : mp) sum += x.second * (x.second - 1) / 2;
    if (mp.size() == n)
        cout << sum;
    else
        cout << -1;
}
