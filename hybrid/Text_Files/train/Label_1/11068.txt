#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e5 + 10;
char s[maxn];
int n;
bool a[maxn], b[maxn], ans[maxn];

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i], s[i] = a[i] + '0';
    if (n == 1 && a[1]) {
        cout << "NO" << endl;
        return 0;
    }
    if (n == 2) {
        if (a[1] == a[2]) s[2] = '0' + !a[1], b[2] = true;
        ans[1] = true;
    } else {
        for (int i = 3; i <= n; ++i) {
            if (a[i]) b[i] = true;
            if (a[i - 1] == a[i]) {
                if (a[i] == 0) s[i] = '0';
                if (a[i] == 1) s[i - 1] = '1', b[i - 1] = true;
                ans[i - 2] = true;
            }
        }
        if (a[1] == a[2]) s[2] = '0' + (a[1] ^ 1), b[2] = true, ans[1] = true;
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; ++i) cout << s[i];
    for (int i = 1; i < n; ++i) {
        if (ans[i]) cout << ") -> ";
        else if (b[i + 1]) cout << " ->";
    }
    cout << endl;
    return 0;
}
