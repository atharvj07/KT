#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;

ll modpow(ll x, ll y, ll m) {
    ll ans = 1;
    x %= m;
    while (y) {
        if (y & 1) ans = ans * x % m;
        x = x * x % m;
        y >>= 1;
    }
    return ans;
}

ll mul(ll a, ll b, ll m) { return ((a % m) * (b % m)) % m; }

ll add(ll a, ll b, ll m) { return ((a % m) + (b % m)) % m; }

ll inv(ll x, ll m) { return modpow(x, m - 2, m); }

const int mxN = 505;

array<int, mxN> g;
array<int, mxN> sz;

int uf_find(int i) {
    return g[i] == i ? i : (g[i] = uf_find(g[i]));
}

void uf_union(int i, int j) {
    i = uf_find(i), j = uf_find(j);
    if (i == j) return;
    g[j] = i;
    sz[i] += sz[j];
}

int main() {
    cin.tie(0)->sync_with_stdio(0);
    int n;
    cin >> n;
    iota(begin(g), begin(g) + n, 0);
    fill(begin(sz), begin(sz) + n, 1);
    for (int i = 0; i < n; i++) {
        int pi;
        cin >> pi;
        if (pi != -1) uf_union(pi - 1, i);
    }
    ll x = 0, y = 1;
    for (int i = 0; i < n; i++) {
        if (uf_find(i) == i) {
            x = add(x, modpow(sz[i], n - 2, mod), mod);
            y = mul(y, sz[i], mod);
        }
    }
    y = inv(y, mod);
    cout << mul(x, y, mod) << '\n';
    return 0;
}
