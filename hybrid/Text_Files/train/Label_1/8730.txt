#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<string> a(n);
    map<string, int> mp;
    for (int i = 0; i < n; i++) { cin >> a[i]; mp[a[i]]++; }
    vector<string> b;
    for (int i = 0; i < (1 << m); i++) {
      string s = "";
      for (int j = m - 1; j >= 0; j--) { s += (i & (1 << j)) ? '1' : '0'; }
      if (mp.find(s) == mp.end()) { b.push_back(s); }
    }
    sort(b.begin(), b.end());
    cout << b[(b.size() - 1) / 2] << endl;
  }
  return 0;
}
