import java.io.*;
import java.math.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            BigInteger x = new BigInteger(in.next(), 2);
            BigInteger y = new BigInteger(in.next(), 2);
            int k = 0;
            while (x.compareTo(y) >= 0) {
                x = x.subtract(y.shiftLeft(k));
                k++;
            }
            System.out.println(k - 1);
        }
    }
}
