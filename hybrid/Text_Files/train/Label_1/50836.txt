import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();

    if (k < n) {
      System.out.println(-1);
      return;
    }

    int[] p = new int[n];
    int[] q = new int[n];

    int sum = n * (n + 1) / 2;

    if (k == sum) {
      for (int i = 0; i < n; i++) {
        p[i] = i + 1;
        q[i] = n - i;
      }
    } else {
      int[] pCopy = Arrays.copyOf(p, p.length);
      int[] qCopy = Arrays.copyOf(q, q.length);

      int index = n - 1;
      while (k > sum) {
        sum -= pCopy[index] + qCopy[index];
        sum += n - pCopy[index] + n - qCopy[index];
        pCopy[index] = n - pCopy[index];
        qCopy[index] = n - qCopy[index];
        index--;
      }

      p = Arrays.copyOf(pCopy, pCopy.length);
      q = Arrays.copyOf(qCopy, qCopy.length);
    }

    System.out.println(sum);
    for (int i = 0; i < n; i++) {
      System.out.print(p[i] + " ");
    }
    System.out.println();
    for (int i = 0; i < n; i++) {
      System.out.print(q[i] + " ");
    }
    System.out.println();
  }
}
