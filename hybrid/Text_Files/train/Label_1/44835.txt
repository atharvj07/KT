import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    static int[] a, b, c;
    static int[] d;
    static List<Integer>[] e;
    static int n, m, k;
    static long ans = 0;

    static void dfs(int v, int p) {
        d[v] = Integer.MAX_VALUE;
        for (int to : e[v]) {
            if (to == p) continue;
            dfs(to, v);
            d[v] = Math.min(d[v], d[to] - 1);
        }
    }

    static boolean canCapture() {
        dfs(1, -1);
        for (int i = 1; i <= n; i++) {
            if (d[i] >= 0) return true;
        }
        return false;
    }

    static void go(int i, long sum, int cnt) {
        ans = Math.max(ans, sum);
        if (i == n + 1) return;
        // capture castle i
        if (cnt < a[i]) return;
        go(i + 1, sum + c[i], cnt - a[i] + b[i]);
        // don't capture castle i
        go(i + 1, sum, cnt + b[i]);
        // leave one warrior to defend castle i
        go(i + 1, sum + c[i], cnt - 1);
        // send a warrior to defend castle i
        for (int j : e[i]) {
            if (j != d[i] + 1) continue;
            go(i + 1, sum + c[i], cnt - 1);
        }
    }

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        a = new int[n + 1];
        b = new int[n + 1];
        c = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        e = new List[n + 1];
        for (int i = 1; i <= n; i++) e[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            e[u].add(v);
        }
        d = new int[n + 1];
        if (canCapture()) {
            go(1, 0, k);
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
