l, r, x, y = map(int, input().split())
if x > r or y < l:
  print(0)
else:
  if x == y:
    if x <= r:
      ans = (r - x + 1) // x
    else:
      ans = 0
  else:
    if l % x == 0:
      st = l
    else:
      st = l + x - l % x
    en = r - (r % x)
    if st <= en:
      ans = (en - st) // x + 1
    else:
      ans = 0
  print(ans)
