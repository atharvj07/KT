def solve(n, p, s, prog_skills, sports_skills):
    students = list(zip(prog_skills, sports_skills, range(1, n + 1)))
    students.sort(key=lambda x: (-x[0], -x[1]))

    prog_team = students[:p]
    remaining_students = students[p:]
    remaining_students.sort(key=lambda x: (-x[1], -x[0]))
    sports_team = remaining_students[:s]

    max_strength = sum([x[0] for x in prog_team]) + sum([x[1] for x in sports_team])
    prog_team_ids = [str(x[2]) for x in prog_team]
    sports_team_ids = [str(x[2]) for x in sports_team]

    return f"{max_strength}\n{' '.join(prog_team_ids)}\n{' '.join(sports_team_ids)}"

n, p, s = map(int, input().split())
prog_skills = list(map(int, input().split()))
sports_skills = list(map(int, input().split()))

print(solve(n, p, s, prog_skills, sports_skills))
