import java.util.Scanner;

public class Chemical {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int Ma = scanner.nextInt();
        int Mb = scanner.nextInt();
        int[] a = new int[N];
        int[] b = new int[N];
        int[] c = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
        }

        // Calculate the GCD of Ma and Mb
        int gcd = gcd(Ma, Mb);

        // Calculate the amount of substance A and B needed
        int targetA = Ma / gcd;
        int targetB = Mb / gcd;

        // Initialize the minimum cost to infinity
        int minCost = Integer.MAX_VALUE;

        // Iterate over all possible combinations of packages
        for (int mask = 0; mask < (1 << N); mask++) {
            int currentA = 0;
            int currentB = 0;
            int currentCost = 0;
            for (int i = 0; i < N; i++) {
                if ((mask & (1 << i)) != 0) {
                    currentA += a[i];
                    currentB += b[i];
                    currentCost += c[i];
                }
            }

            // Check if the current combination can generate the substance C
            if (currentA >= targetA && currentB >= targetB) {
                minCost = Math.min(minCost, currentCost);
            }
        }

        // If no combination can generate the substance C, print -1
        if (minCost == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minCost);
        }
    }

    // Function to calculate the greatest common divisor (GCD) of two numbers
    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
