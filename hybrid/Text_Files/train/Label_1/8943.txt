#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(2 * n);
    for (int &x : a) { cin >> x; }
    sort(a.begin(), a.end());
    cout << a[n] - a[n - 1] << '\n';
  }
  return 0;
}
