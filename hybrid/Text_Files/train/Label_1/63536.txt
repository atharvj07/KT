for _ in range(int(input())):
    px, py = [int(x) for x in input().split()]
    s = input()
    x, y = 0, 0
    for i in s:
        if i == 'U':
            y += 1
        elif i == 'D':
            y -= 1
        elif i == 'R':
            x += 1
        elif i == 'L':
            x -= 1
    if x == px and y == py:
        print("YES")
    else:
        print("NO")
