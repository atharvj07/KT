n = int(input())
min_1, max_1 = map(int, input().split())
min_2, max_2 = map(int, input().split())
min_3, max_3 = map(int, input().split())

# Distribute the diplomas starting from the first degree to the third
# Firstly, assign maximum possible to first degree within its limit
first_degree = min(max_1, n - min_2 - min_3)
n -= first_degree

# Secondly, assign maximum possible to second degree within its limit
second_degree = min(max_2, n - min_3)
n -= second_degree

# Lastly, all remaining go to third degree
third_degree = n

print(first_degree, second_degree, third_degree)
