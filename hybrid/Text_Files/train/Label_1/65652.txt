n, x = map(int, input().split())
arr = list(map(int, input().split()))
Max = 0
sum = 0
for i in arr:
    sum += i
    if sum < 0:
        sum = 0
    if Max < sum:
        Max = sum
sum = 0
for i in arr:
    sum += i * x
    if sum < 0:
        sum = 0
    if Max < sum:
        Max = sum
print(Max)
