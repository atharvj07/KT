import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] f = new int[n + 1];
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                f[i + 1] = Math.max(f[i + 1], f[i] + 1);
            }
        }
        for (int i = 0; i < n; i++) {
            if (a[i] == 0 && k > 0) {
                f[i + 1] = Math.max(f[i + 1], f[i] + 1);
                k--;
            }
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, f[i]);
        }
        System.out.println(max);
        for (int i = 0; i < n; i++) {
            if (f[i + 1] == max) {
                a[i] = 1;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
