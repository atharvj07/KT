import sys

n, k = map(int, sys.stdin.readline().split())
s = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]

cnt = [0] * 201

for l, r in s:
    cnt[l] += 1
    cnt[r + 1] -= 1

# compute the number of segments covering each point
co = 0
for i in range(1, 201):
    co += cnt[i]
    cnt[i] = co

# count # of bad points
bd = 0
for c in cnt:
    if c > k:
        bd += c - k

# sort segments by their right endpoint
s.sort(key=lambda x: x[1])

# remove segments
ans = []
cur = 0
for l, r in s:
    if cur + cnt[l] - cnt[r + 1] > k:
        ans.append(s.index([l, r]) + 1)
        cur += cnt[l] - cnt[r + 1]
        if cur <= k:
            break

# output
print(len(ans))
print(*ans)
