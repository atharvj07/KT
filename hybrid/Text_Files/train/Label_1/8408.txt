#include <iostream>

int main() {
  long long p, y;
  std::cin >> p >> y;
  for (; p <= y; p++) {
    bool flag = true;
    for (long long i = 2; i * p <= y; i++) {
      if ((i * p) % p == 0) {
        flag = false;
        break;
      }
    }
    if (flag) {
      std::cout << p << '\n';
      return 0;
    }
  }
  std::cout << -1 << '\n';
  return 0;
}
