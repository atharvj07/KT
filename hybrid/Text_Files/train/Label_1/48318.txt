import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[][] arr = new int[n][k];
        int[][] ans = new int[n][k];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                arr[i][j] = sc.nextInt();
            }
        }

        for (int i = 0; i < n; i++) {
            ans[i][0] = 1;
            for (int j = 1; j < k; j++) {
                ans[i][j] = 0;
                for (int l = 0; l < j; l++) {
                    if (arr[i][j] > arr[i][l]) {
                        ans[i][j] += ans[i][l];
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(ans[i][k - 1]);
        }
    }
}
