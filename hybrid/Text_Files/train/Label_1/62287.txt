def solve(s):
    n = len(s)
    ans = []
    stack = []
    for i in range(n):
        if stack and stack[-1] == s[i]:
            stack.pop()
        else:
            stack.append(s[i])
        cur_ans = ''.join(stack)
        if len(cur_ans) > 10:
            cur_ans = cur_ans[:5] + "..." + cur_ans[-2:]
        ans.append((len(cur_ans), cur_ans))
    for length, string in reversed(ans):
        print(length, string)

s = input().strip()
solve(s)
