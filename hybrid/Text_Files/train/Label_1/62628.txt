import sys

n = int(sys.stdin.readline())

def generate_permutation(n):
    p = [0 for _ in range(n + 1)]
    for i in range(1, n + 1):
        if i % 2 == 1:
            p[i] = i + 1
        else:
            p[i] = i - 1
    return p[1:]

def check_permutation(p, n):
    for i in range(1, n + 1):
        if p[i] == i or p[i] & i != 0:
            return False
    return True

if n & 1 == 0:
    p = generate_permutation(n)
    if check_permutation(p, n):
        print("YES")
        print(*p)
    else:
        print("NO")
else:
    print("NO")
    print("NO")
