#include <bits/stdc++.h>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  if (m > n * (n - 1) / 2 || m * 2 < n - 1) {
    cout << "Impossible";
    return 0;
  }
  cout << "Possible" << endl;
  m -= n - 1;
  for (int i = 2; i <= n; i++) cout << 1 << " " << i << endl;
  for (int j = n - 1; j > 1 && m > 0; j--) {
    for (int k = j + 1; k <= n && m > 0; k++) {
      cout << j << " " << k << endl;
      m--;
    }
  }
}
