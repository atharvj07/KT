#include <bits/stdc++.h>
using namespace std;
bool c[52];
int main() {
  int n, m, ans = 0;
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    string s;
    cin >> s;
    for (int j = 0; j < m; ++j) {
      if ((s[j] == 'a' && c[0]) || (s[j] == 'c' && c[1]) ||
          (s[j] == 'e' && c[2]) || (s[j] == 'f' && c[3]))
        ans = ans + 1;
      c[s[j] - 'a'] = true;
      if (j > 2) c[s[j - 3] - 'a'] = false;
    }
  }
  cout << ans;
}
