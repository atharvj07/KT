import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tmp = br.readLine().split(" ");
        int n = Integer.parseInt(tmp[0]);
        int m = Integer.parseInt(tmp[1]);
        int[][] dim = new int[n][2];
        long[] bridges = new long[m];
        Map<Long, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            tmp = br.readLine().split(" ");
            dim[i][0] = Integer.parseInt(tmp[0]);
            dim[i][1] = Integer.parseInt(tmp[1]);
        }
        tmp = br.readLine().split(" ");
        for (int i = 0; i < m; i++) {
            bridges[i] = Long.parseLong(tmp[i]);
            map.put(bridges[i], map.getOrDefault(bridges[i], 0) + 1);
        }
        Arrays.sort(bridges);
        Arrays.sort(dim, (a, b) -> a[0] - b[0]);
        long prev_end = Long.MIN_VALUE;
        int j = 0;
        for (int[] d : dim) {
            if (prev_end == Long.MIN_VALUE) {
                prev_end = d[1];
                continue;
            }
            prev_end = prev_end + 1;
            while (prev_end < d[0]) {
                if (j >= bridges.length || prev_end + bridges[j] > d[1]) {
                    System.out.println("No");
                    return;
                }
                prev_end += bridges[j];
                if (map.containsKey(bridges[j]))
                    map.put(bridges[j], map.get(bridges[j]) - 1);
                if (map.get(bridges[j]) < 0) {
                    System.out.println("No");
                    return;
                }
                j++;
            }
        }
        System.out.println("Yes");
        for (int i = 0; i < n - 1; i++) {
            System.out.print((j + i) + " ");
        }
        System.out.println();
    }
}
