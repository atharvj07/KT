import java.util.*;
import java.io.*;
import java.lang.*;

public class Main {

    static int[] ar;

    public static boolean isPossible(int k) {
        for (int a : ar) {
            if (a % k != 0) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        try (BufferedReader bf = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(bf.readLine());
            ar = new int[n];
            String[] line = bf.readLine().split(" ");
            for (int i = 0; i < n; i++) {
                ar[i] = Integer.parseInt(line[i]);
            }
            int ans = -1;
            for (int i = 1; i <= n; i++) {
                if (isPossible(i)) {
                    ans = i;
                    break;
                }
            }
            System.out.println(ans);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
