#include <cstdio>

const int dxy[4][2] = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};

int n, a[105][105], dx, dy, cur, x, y;

int main() {
    int T;
    scanf("%d", &T);
    while (T--) {
        scanf("%d", &n);
        if ((n & 1) || n < 3) {
            puts("-1");
            continue;
        }
        dx = 0, dy = 1, cur = 1, x = (n + 1) / 2, y = (n + 1) / 2;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                a[i][j] = cur++;
            }
        }
        a[x][y] = n * n;
        while ((x + dx >= 1 && x + dx <= n) && (y + dy >= 1 && y + dy <= n)) {
            a[x + dx][y + dy] = --cur;
            x += dx, y += dy;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) { printf("%d ", a[i][j]); }
            puts("");
        }
    }
}
