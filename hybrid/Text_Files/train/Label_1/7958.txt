#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 100005;
const int MAXQ = 500005;

int N, M, Q;
int arr[MAXQ];
vector<int> ans;

int main() {
    scanf("%d", &Q);
    while (Q--) {
        scanf("%d", &N);
        M = 4 * N;
        ans.clear();
        for (int i = 0; i < M; i++) {
            scanf("%d", &arr[i]);
        }
        sort(arr, arr + M);
        if (!N || M % 4) {
            printf("NO\n");
            continue;
        }
        for (int i = 0, j = M - 1; i < j; i += 2, j -= 2) {
            if (arr[i] != arr[i + 1] || arr[j] != arr[j - 1]) {
                printf("NO\n");
                goto end;
            }
            ans.push_back(arr[i]);
        }
        if (ans.size() == N)
            printf("YES\n");
        else
            printf("NO\n");
    end:;
    }
    return 0;
}
