n=int(input());t1,t2=input(),input();mod=1000000007
s1,s2=[0]*n,[0]*n
for i in range(n):s1[i]=int(t1[i])if t1[i]!='?'else-1;s2[i]=int(t2[i])if t2[i]!='?'else-1
c1=[0]*n;c2=[0]*n
for i in range(n):
    if s1[i]==-1:c1[i]=i
    if s2[i]==-1:c2[i]=i
p1,p2=[-1]*n,[-1]*n
for i in range(n):
    if s1[i]>=0:p1[i]=i
    if s2[i]>=0:p2[i]=i
s1sort,s2sort=[i for i in sorted(s1)],sorted(s2)
ans=1;lst,lst1=-1,-1
for i in range(n):
    if c1[i]!=0:
        if s1sort[lst+1] > s2sort[lst1]:lst=-1
        else:lst=lst1+1
        x=[a for a,b in enumerate(s2)if a>i and (b==-1 or b==s1sort[lst])]
        if len(x)>0:lst1=max(lst1,max(x))
    if c2[i]!=0:
        if s2sort[lst1+1] > s1sort[lst]:lst1=-1
        else:lst1=lst+1
        x=[a for a,b in enumerate(s1)if a>i and (b==-1 or b==s2sort[lst1])]
        if len(x)>0:lst=max(lst,max(x))
    if c1[i]!=0 and c2[i]!=0:
        if s1sort[lst+1] > s2sort[lst1] or s2sort[lst1+1] > s1sort[lst]:ans=0
    if c1[i]!=0:
        for j in range(i+1,n):
            if s2[j]==-1 and p1[j]>0:ans=0
    if c2[i]!=0:
        for j in range(i+1,n):
            if s1[j]==-1 and p2[j]>0:ans=0
ans=ans*pow(2,c1.count(-1)+c2.count(-1),mod)%mod
print(ans)
