n, k = map(int, input().split())
a = list(map(int, input().split()))

from collections import Counter
counts = Counter(a)

if max(counts.values()) > k:
    print("NO")
else:
    print("YES")
    colors = {}
    result = []
    color = 1
    for num in a:
        if num in colors and len(colors[num]) < k:
            result.append(colors[num][-1] + 1)
            colors[num].append(colors[num][-1] + 1)
        elif num not in colors:
            colors[num] = [color]
            result.append(color)
        else:
            color += 1
            if color > k:
                color = 1
            colors[num] = [color]
            result.append(color)
    print(*result)
