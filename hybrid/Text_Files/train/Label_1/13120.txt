#include <bits/stdc++.h>
using namespace std;
const int maxN = 1e5;
int n, k, freq[maxN];
int main()
{
    cin >> n >> k;
    if (k == 1)
        for (int i = 1; i <= n; i++)
            cout << i << " ";
    else
    {
        cout << "1";
        for (int i = n - k + 1; i <= n; i++)
            cout << " " << i;
        for (int i = 2; i <= n - k; i++)
            cout << " " << i;
    }
    return 0;
}
