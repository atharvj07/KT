def max_soldiers(m, n, k, t, agility, traps):
    agility.sort(reverse=True)
    left, right, ans = 0, m, 0
    while left <= right:
        mid = (left + right) // 2
        min_agility = agility[mid - 1] if mid > 0 else 0
        time_needed = n + 1
        disarm_points = []
        for trap in traps:
            if trap[2] >= min_agility:
                disarm_points.append((trap[0], trap[1]))
        disarm_points.sort()
        current_end = 0
        for dp in disarm_points:
            if dp[0] > current_end:
                time_needed += 2 * (dp[1] - dp[0])
                current_end = dp[1]
            elif dp[1] > current_end:
                time_needed += 2 * (dp[1] - current_end)
                current_end = dp[1]
        if time_needed <= t:
            ans = mid
            left = mid + 1
        else:
            right = mid - 1
    return ans

# Input reading
m, n, k, t = map(int, input().split())
agility = list(map(int, input().split()))
traps = [list(map(int, input().split())) for _ in range(k)]
# Function call
print(max_soldiers(m, n, k, t, agility, traps))
