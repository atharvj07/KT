#include <bits/stdc++.h>
using namespace std;
int main(){
    int q; cin >> q;
    while (q--) {
        string s, t; cin >> s >> t;
        int i = 0, j = 0;
        while (j < t.size()) {
            if (i < s.size() && s[i] == t[j]) { i++; j++; }
            else if (i > 0 && s[i - 1] == t[j]) j++;
            else break;
        }
        cout << (i == s.size() && j == t.size() ? "YES\n" : "NO\n");
    }
    return 0;
}
