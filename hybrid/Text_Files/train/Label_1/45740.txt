import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    long n = new Scanner(System.in).nextLong();
    System.out.println(solve(n));
  }

  static long solve(long n) {
    String str = String.valueOf(n);
    long sum = 0;
    for (char c : str.toCharArray()) {
      sum += c - '0';
    }
    return sum;
  }
}
