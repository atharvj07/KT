from collections import defaultdict
n, m = map(int, input().split())
a = defaultdict(list)
for _ in range(n):
    s, r, v = input().split()
    a[r].append((int(v), s))
for r in sorted(a.keys()):
    l = sorted(a[r], reverse = True)
    if len(l) == 2 and l[0][0] == l[1][0]:
        print('?')
    else:
        print(l[0][1], l[1][1])
