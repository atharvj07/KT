#include <bits/stdc++.h>
using namespace std;
const int maxl = 2e5 + 1, M = 1e9 + 8;
int hp[maxl], ht[maxl];
int main() {
    int n, m, p;
    cin >> n >> m >> p;
    vector<int> a(n), b(m);
    for (auto &x : a) cin >> x;
    for (auto &x : b) cin >> x;
    for (int i = 0; i < n; i++) { (hp[i] = hp[i - 1] + a[i]) %= M; }
    for (int i = 0; i < m; i++) { (ht[i] = ht[i - 1] + b[i]) %= M; }
    unordered_map<int, vector<int>> mp;
    for (int i = 0; i < n; i++) {
        int cur = hp[i] - (i == 0 ? 0 : hp[i - 1]);
        if ((i + p * (m - 1) + 1) <= n) mp[cur].push_back(i + 1);
    }
    int cnt = 0;
    for (int i = 0; i < m; i++) {
        int cur = ht[i] - (i == 0 ? 0 : ht[i - 1]);
        if (mp.count(cur)) {
            for (auto &x : mp[cur]) cnt++, cout << x << " ";
        }
    }
    cout << "\n" << cnt << "\n";
    return 0;
}
