#include<bits/stdc++.h>
using namespace std;
const int M = 1e5 + 5;
int n, t, a[M][2], q, ans;
int main() {
	cin >> q;
	while(q--) {
		cin >> n; ans = 0;
		for(int i = 0; i < n; i++) {
			cin >> a[i][0] >> a[i][1];
			if(i && a[i][0] <= a[i - 1][0] + abs(a[i][1] - a[i - 1][1]))ans++;
		}
		cout << ans << '\n';
	}
	return 0;
}
