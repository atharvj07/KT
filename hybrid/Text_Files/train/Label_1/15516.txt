#include<iostream>
#include<map>
#include<vector>
using namespace std;

int main()
{
    string a, b, c, d;
    cin >> a >> b;
    int n;
    cin >> n;
    map<string, string> mp;
    mp[a] = b;
    mp[b] = a;
    vector<string> ans;
    ans.push_back(a + " " + b);
    for (int i = 1; i <= n; i++)
    {
        cin >> c >> d;
        mp[mp[c]] = d;
        ans.push_back(mp[c] + " " + d);
    }
    for (auto it : ans)
        cout << it << endl;
    return 0;
}
