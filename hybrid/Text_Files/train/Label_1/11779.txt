#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n,k,x,a[200005],b[200005];
int main(){
    cin>>n>>k>>x;
    for(ll i=1;i<=n;i++){
        cin>>a[i];
        b[i]=a[i];
    }
    while(k--){
        for(ll i=1;i<=n;i++){
            if((a[i]>0&&a[i]%x!=0)||(a[i]<0&&-a[i]%x!=0)){
                if(a[i]>0) b[i]+=x;
                else b[i]-=x;
                a[i]+=x;
            }
        }
    }
    for(ll i=1;i<=n;i++) cout<<b[i]<<" ";
    return 0;
}
