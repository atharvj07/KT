n = int(input().strip())
a = list(map(int, input().split()))

if sum(a) == 0:
    print("NO")
else:
    print("YES")
    if sum(a) != 0:
        print(1)
        print(1, n)
    else:
        for i in range(n):
            if sum(a[:i+1]) != 0 and sum(a[i+1:]) != 0:
                print(2)
                print(1, i+1)
                print(i+2, n)
                break
