import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] arr = new int[n];
            int numX = 0;
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                if (s.equals("x")) {
                    arr[i] = Integer.MIN_VALUE;
                    numX++;
                } else {
                    arr[i] = Integer.parseInt(s);
                }
            }
            if (numX == 1) {
                int brokenVal = Integer.MIN_VALUE;
                for (int i = 0; i < n; i++) {
                    if (arr[i] == Integer.MIN_VALUE) {
                        brokenVal = (arr[i - 1] + arr[(i + 1) % n]) / 2;
                        break;
                    }
                }
                System.out.println(brokenVal);
            } else if (numX == 0) {
                System.out.println("none");
            } else {
                System.out.println("ambiguous");
            }
        }
    }
}
