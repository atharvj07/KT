#include <iostream>
#include <vector>
#define MOD 1000000007

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; ++i) cin >> p[i];

    vector<int> dp(n + 1);
    dp[1] = 1;
    for (int i = 1; i <= n; ++i) {
        dp[p[i]] = (dp[p[i]] + dp[i]) % MOD;
        dp[i + 1] = (dp[i + 1] + dp[i]) % MOD;
    }

    cout << dp[n + 1] << endl;
    return 0;
}
