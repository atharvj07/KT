import math

def isprime(a):
    if (a < 2):
        return False
    for i in range(2, int(math.sqrt(a)) + 1):
        if (a % i == 0):
            return False
    return True

t = int(input())
for i in range(t):
    n, e = map(int, input().split())
    a = list(map(int, input().split()))
    ans = 0
    for i in range(n):
        cur = a[i]
        for j in range(i + e, n, e):
            cur *= a[j]
            if isprime(cur):
                ans += 1
    print(ans)
