import sys
from typing import List
input = sys.stdin.readline

n = int(input())
x = []
y = []
for _ in range(n):
    xi, yi = map(int, input().split())
    x.append(xi)
    y.append(yi)

ans = 0
for i in range(n):
    for j in range(i+1, n):
        for k in range(j+1, n):
            for l in range(k+1, n):
                tmp = abs(
                    (x[i]-x[j])*(y[k]-y[l]) - (x[i]-x[k])*(y[j]-y[l])
                )
                ans = max(tmp, ans)

print(ans/2)
