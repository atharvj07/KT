import sys

def solve():
    input = sys.stdin.readline().split()
    n, x, y, d = int(input[0]), int(input[1]), int(input[2]), int(input[3])
    if x == y:
        return 0
    elif (y - x) % d == 0:
        return (y - x) // d
    else:
        return -1

test_cases = int(sys.stdin.readline())
for i in range(test_cases):
    print(solve())
