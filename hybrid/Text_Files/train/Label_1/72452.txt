import sys
a=list(map(int, sys.stdin.readline().split()))
n=a[0]
del a[0]
for i in range(n):
    for j in range (n):
        if i==j: continue
        sum=0
        if i<j:
            for k in a[i+1:j+1]:
                sum+=k
        else:
            for k in a[j+1:i+1]:
                sum+=k
        f=i-j
        f=f**2+sum**2
        if f==0: continue
        if i==0 and j==1: minans=f
        if f<minans:
           minans=f
print(minans)
