#include<iostream>
using namespace std;
int a[110],n,m,ans;
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        a[i]=a[i]-m;
    }
    while(1){
        int can=1;
        for(int i=1;i<=n;++i){
            if(a[i]>0){
                ++can;
                a[i]-=m;
            }
        }
        if(can==1){
            break;
        }
        ans=can;
    }
    cout<<ans<<endl;
    return 0;
}
