def schedule_repairs(n, requests):
    # Initialize an empty list to store the scheduled repairs
    scheduled = []
    for i in range(n):
        start, duration = requests[i]
        # Check if the requested start time and duration is available
        is_scheduled = False
        for s, _ in scheduled:
            if not (s + duration <= start or s >= start + duration):
                is_scheduled = True
                break
        if not is_scheduled:
            scheduled.append((start, start + duration - 1))
        else:
            # Find the first available day to schedule the repair
            x = 1
            while True:
                found = True
                for s, d in scheduled:
                    if not (s + duration <= x or s >= x + duration):
                        found = False
                        break
                if found:
                    break
                x += 1
            scheduled.append((x, x + duration - 1))
    return scheduled

n = int(input())
requests = [tuple(map(int, input().split())) for _ in range(n)]
for start, end in schedule_repairs(n, requests):
    print(start, end)
