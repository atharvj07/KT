import java.util.*;

class Main {
    static boolean isPossible(int n, char[][] board) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !blocked(board, i, j, n)) {
                    return true;
                }
            }
        }
        return false;
    }

    static boolean blocked(char[][] board, int x, int y, int n) {
        if (x + 1 < n && board[x + 1][y] == '.' &&
                (x + 2 < n && board[x + 2][y] == '.')) {
            return false;
        }

        if (y + 1 < n && board[x][y + 1] == '.' &&
                (y + 2 < n && board[x][y + 2] == '.')) {
            return false;
        }

        if (x + 1 < n && y + 1 < n && board[x + 1][y + 1] == '.' &&
                (x + 2 < n && y + 2 < n && board[x + 2][y + 2] == '.')) {
            return false;
        }

        if (x + 1 < n && y - 1 >= 0 && board[x + 1][y - 1] == '.' &&
                (x + 2 < n && y - 2 >= 0 && board[x + 2][y - 2] == '.')) {
            return false;
        }

        return true;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[][] board = new char[n][n];

        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            board[i] = line.toCharArray();
        }

        System.out.println(isPossible(n, board) ? "YES" : "NO");
    }
}
