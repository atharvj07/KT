import java.util.Scanner;
import java.util.Arrays;

class Main {
    static int n;
    static int[] a;
    static int[] l;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n];
        l = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            l[i] = sc.next().charAt(0);
        }
        for (int i = 1; i < n; i++) {
            a[i] -= a[i - 1];
        }
        Arrays.sort(a);
        Arrays.sort(l);
        int ans = 0;
        int j = 0;
        int k = 0;
        int flag = 0;
        for (int i = 0; i < n; i++) {
            if (l[i] == 'P') {
                if (j < n / 2) {
                    ans += a[j++];
                } else if (k < n / 2) {
                    ans += a[n / 2 + k++];
                } else {
                    flag = 1;
                    break;
                }
            } else {
                continue;
            }
        }
        if (flag == 0) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
