s = input()
t = input()

from collections import Counter

# Count the frequency of characters in t
t_count = Counter(t)
# Initialize a counter for s that doesn't count '?' (wildcards)
s_count = Counter([c for c in s if c != '?'])

# Find the number of '?' in s
wildcards = s.count('?')

# Distribute the wildcards to maximize suitability
# We try to even out the distribution of characters according to t in s
for char, freq in t_count.items():
    needed = freq - s_count.get(char, 0)
    to_add = min(needed, wildcards)
    s_count[char] += to_add
    wildcards -= to_add

# If there are still wildcards left, distribute them arbitrarily
if wildcards > 0:
    for char in 'abcdefghijklmnopqrstuvwxyz':
        if wildcards == 0:
            break
        s_count[char] += wildcards
        wildcards = 0

# Construct the final string
result = list(s)
for i, char in enumerate(result):
    if char == '?':
        for c, count in s_count.items():
            if count > 0:
                result[i] = c
                s_count[c] -= 1
                break

print(''.join(result))
