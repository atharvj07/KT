#include <bits/stdc++.h>

using namespace std;
#define mod 1000000007

typedef long long ll;
const int N = 2e5 + 5;
ll arr[N], phi[N], g[N], sum[N];

void sieve() {
    phi[1] = 1;
    for (ll i = 2; i < N; ++i) {
        if (!g[i]) {
            for (ll j = i; j < N; j += i) {
                g[j] = i;
                phi[j] = phi[j] * (i - 1) / i;
            }
        }
    }
}

int main() {
    sieve();
    ll n;
    cin >> n;
    for (ll i = 0; i < n; ++i) {
        cin >> arr[i];
    }
    for (ll i = 0; i < n; ++i) {
        ll num = arr[i];
        while (num != 1) {
            ll div = g[num];
            while (num % div == 0) { num /= div; }
            sum[div] += phi[num] * arr[i];
            sum[div] %= mod;
        }
    }
    ll ans = 0;
    for (ll i = 0; i < N; ++i) { ans += (i * sum[i]) % mod; }
    ans = (ans % mod + mod) % mod;
    cout << ans << endl;
}
