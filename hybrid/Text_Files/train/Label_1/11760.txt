#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> cnt(4);
  for (char c : s) {
    if (c != '?') cnt[c - 'A']++;
  }
  int sum = cnt[0] + cnt[1] + cnt[2] + cnt[3];
  if (sum % 4 != 0) {
    cout << "===";
    return 0;
  }
  int avg = sum / 4;
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      for (int j = 0; j < 4; j++) {
        if (cnt[j] < avg) {
          s[i] = 'A' + j;
          cnt[j]++;
          break;
        }
      }
    }
  }
  cout << s;
  return 0;
}
