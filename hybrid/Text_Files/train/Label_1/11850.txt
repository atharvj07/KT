#include <algorithm>
#include <ios>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  if (n == 1) {
    std::cout << "1\n";
  } else if (n == 2) {
    std::cout << "2 1\n";
  } else {
    std::vector<int> arr(n);
    for (int i = 0; i < n - 1; i++)
      arr[i] = i + 2;

    arr[n - 2] = n;
    arr[n - 1] = 1;
    for (int x : arr) std::cout << x << " ";
    std::cout << "\n";
  }
  return 0;
}
