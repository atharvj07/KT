#include <algorithm> 
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> g(n);
  vector<int> deg(n);
  while (m--) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    g[u].push_back(v);
    g[v].push_back(u);
    deg[u]++, deg[v]++;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int v : g[i]) {
      ans = max(ans, deg[i] + deg[v]);
    }
  }
  cout << ans << endl;
  return 0;
}
