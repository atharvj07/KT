#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 105;
const int MAXM = 105;
const int MAXQ = 105;
vector<int> adj[MAXN], color[MAXN];
int n, m, q, a[MAXM], b[MAXM], c[MAXM], u[MAXQ], v[MAXQ];
bool vis[MAXQ];
vector<int> ans;

void dfs(int k, int u, int par) {
    vis[u] = true;
    for (int i = 0; i < adj[u].size(); i++) {
        int v = adj[u][i];
        if (color[u][i] == k && !vis[v]) {
            dfs(k, v, u);
        }
    }
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i] >> c[i];
        adj[a[i]].push_back(b[i]);
        adj[b[i]].push_back(a[i]);
        color[a[i]].push_back(c[i]);
        color[b[i]].push_back(c[i]);
    }
    cin >> q;
    for (int i = 1; i <= q; i++) {
        cin >> u[i] >> v[i];
    }
    for (int k = 1; k <= m; k++) {
        int cnt = 0;
        for (int i = 1; i <= n; i++) vis[i] = false;
        dfs(k, u[1], 0);
        if (vis[v[1]]) cnt++;
        for (int i = 1; i <= n; i++) vis[i] = false;
        dfs(k, v[1], 0);
        if (vis[u[1]]) cnt++;
        if (cnt == 2) ans.push_back(k);
    }
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << endl;
    }
    return 0;
}
