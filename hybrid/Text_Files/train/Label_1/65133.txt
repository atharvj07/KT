s = input().strip()
n = len(s)

# Initialize counters for odd and even good substrings
odd, even = 0, 0

# Variables to track the count of continuous same characters
# and also to handle the case where characters change
a_count, b_count = 0, 0

for i in range(n):
    if s[i] == 'a':
        # Increase count for 'a', reset 'b' count
        a_count += 1
        b_count = 0
    else:
        # Increase count for 'b', reset 'a' count
        b_count += 1
        a_count = 0
    
    # If current character count is odd, it contributes to odd substrings
    # Else, it contributes to even substrings
    if (a_count % 2 == 0 and a_count > 0) or (b_count % 2 == 0 and b_count > 0):
        even += 1
    else:
        odd += 1

    # Each substring contributes to the previous substrings
    # Odd substrings remain odd, even becomes odd and vice versa
    odd += even
    even += odd - even - 1

print(even, odd)
