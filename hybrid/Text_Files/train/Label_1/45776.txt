import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] points = new int[n];
        for (int i = 0; i < n; i++) {
            points[i] = in.nextInt();
        }
        int count = 0;
        int max = points[0];
        int min = points[0];
        for (int i = 1; i < n; i++) {
            if (points[i] > max) {
                max = points[i];
                count++;
            } else if (points[i] < min) {
                min = points[i];
                count++;
            }
        }
        System.out.println(count);
    }
}
