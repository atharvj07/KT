#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

void solve() {
  long long n;
  cin >> n;
  vector<long long> factors;
  for (long long i = 2; i * i <= n; i++) {
    while (n % i == 0) {
      factors.push_back(i);
      n /= i;
    }
  }
  if (n > 1) {
    factors.push_back(n);
  }
  if (factors.size() < 3 || factors.front() == factors.back()) {
    cout << "NO\n";
    return;
  }
  cout << "YES\n";
  sort(factors.begin(), factors.end());
  for (int i = 0; i < 3; i++) {
    cout << factors[i] << " ";
  }
  cout << '\n';
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
