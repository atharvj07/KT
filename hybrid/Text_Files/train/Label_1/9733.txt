#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll inf = 9e18;
const int mxN = 55;
char g[mxN][mxN];
int visr[mxN], visc[mxN];
int n, m;
bool chk(int r, int c){
    if(r < 0 || r >= n || c < 0 || c >= m)
        return 0;
    return g[r][c] == '#';
}
bool check(){
    memset(visr, 0, sizeof(visr));
    memset(visc, 0, sizeof(visc));
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < m; ++j){
            if(!chk(i, j))
                continue;
            if(visr[i] and visc[j])
                return 0;
            if(!visr[i]){
                bool has = 0;
                for(int k = 0; k < m; ++k){
                    if(chk(i, k))
                        has = 1;
                }
                if(!has)
                    return 0;
                visr[i] = 1;
            }
            if(!visc[j]){
                bool has = 0;
                for(int k = 0; k < n; ++k){
                    if(chk(k, j))
                        has = 1;
                }
                if(!has)
                    return 0;
                visc[j] = 1;
            }
        }
    }
    return 1;
}
int main(){
    cin >> n >> m;
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < m; ++j){
            cin >> g[i][j];
        }
    }
    if(check())
        cout << "Yes" << endl;
    else
        cout << "No" << endl;
}
