#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    int n;cin>>n;
    string s;cin>>s;
    set<char>v;
    for(int i=0;i<n;i++) v.insert(s[i]);
    int len = v.size();
    int ans=len,i=0,j=0;
    unordered_map<char,int>mp;
    while(i<=j && i<n){
        if(mp[s[i]]==0) len--;
        mp[s[i]]++;
        i++;
        while(len==0){
            mp[s[j]]--;
            if(mp[s[j]]==0) len++;
            j++;
        }
        ans=min(ans,i-j+1);
    }
    cout<<ans;
    return 0;
}
