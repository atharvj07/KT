#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 205;
int a[MAXN], b[MAXN];
int f[MAXN][MAXN];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) cin >> b[i];
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            f[i][j] = (a[i] & b[j]) | f[i - 1][j] | f[i][j - 1];
        }
    }
    cout << f[n][m] << endl;
    return 0;
}
