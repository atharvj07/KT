MOD = int(1e9) + 7
n = int(input())
out = []
for _ in range(n):
 out += [int(x) for x in input().split()]
out.sort()
dp = [[-1] * 1001 for _ in range(n)]
def dfs(i, j):
    if i == n:
        return 1
    if dp[i][j] != -1:
        return dp[i][j]
    ans = 0
    for k in range(j, n):
        if out[i] <= out[k]:
            ans += dfs(i + 1, k + 1)
        else:
            break
    dp[i][j] = ans
    return ans
print(dfs(0, 0))
