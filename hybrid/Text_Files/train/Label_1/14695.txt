#include <iostream>
using namespace std;

int main() {
  long long n;
  cin >> n;
  while (n--) {
    int x1, p1;
    int x2, p2;
    cin >> x1 >> p1;
    cin >> x2 >> p2;
    long long after1 = x1;
    long long after2 = x2;
    long long multi = 10;
    while (p1) {
      after1 *= multi;
      p1--;
    }
    while (p2) {
      after2 *= multi;
      p2--;
    }
    if (after1 < after2)
      cout << "<" << endl;
    else if (after1 > after2)
      cout << ">" << endl;
    else
      cout << "=" << endl;
  }
  return 0;
}
