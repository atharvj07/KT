import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    m = int(input())
    h = []
    for _ in range(m):
        p, s = map(int, input().split())
        h.append((p, s))
    a.sort()
    h.sort(reverse=True)
    ans = -1
    cur = 0
    for i in range(m):
        p, s = h[i]
        if s > n - cur:
            s = n - cur
        cnt = 0
        for j in range(cur, cur + s):
            if j >= n:
                break
            if p >= a[j]:
                cnt += 1
            else:
                break
        if cnt == s:
            if cur + s == n:
                ans = i + 1
                break
            cur += s
    print(ans)
