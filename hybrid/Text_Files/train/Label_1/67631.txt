from collections import Counter
from operator import mul
from functools import reduce

n = int(input())
a = list(map(int,input().split()))

cnt = Counter(a)
inv = 0
sum = 0
for x in cnt.values():
  sum += x * (x - 1) // 2
  inv += x * (n - x)
  
for x in range(1,n+1):
  if not x in cnt:
    inv += x * sum
    sum += x
    
mod = 998244353
print(inv * pow(sum,mod-2,mod) % mod)
