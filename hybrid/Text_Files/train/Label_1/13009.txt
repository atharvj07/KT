#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 26;
ll a[N], b[N];
int main(){
  string s, t;
  getline(cin, s);
  getline(cin, t);
  for(int i = 0; i < s.length(); i++){
    if(s[i] == '|') continue;
    a[s[i] - 'A']++;
  }
  for(int i = 0; i < t.length(); i++){
    b[t[i] - 'A']++;
  }
  bool ok = 1;
  for(int i = 0; i < N; i++){
    if(a[i] + b[i] == 0) continue;
    if((a[i] + b[i]) & 1){
      ok = 0;
      break;
    }
    a[i] = (a[i] + b[i]) / 2;
  }
  string ans;
  for(int i = 0; i < N; i++){
    if(a[i] == 0) continue;
    for(int j = 0; j < a[i]; j++){
      ans += char(i + 'A');
    }
  }
  ans += '|';
  for(int i = 0; i < N; i++){
    if(b[i] == 0) continue;
    for(int j = 0; j < b[i]; j++){
      ans += char(i + 'A');
    }
  }
  if(!ok) cout << "Impossible";
  else cout << ans;
  cout << endl;
  return 0;
}
