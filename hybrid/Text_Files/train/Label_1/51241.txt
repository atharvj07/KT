import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] nq = Arrays.stream(sc.nextLine().split(" "))
                .mapToInt(Integer::parseInt)
                .toArray();
        int n = nq[0], q = nq[1];
        String s = sc.nextLine();
        int[] t = new int[4];
        for (int i = 0; i < q; i++) {
            String[] query = sc.nextLine().split(" ");
            int type = query[0].charAt(0);
            int religion = query[1].charAt(0) - '0';
            if (type == '+') {
                t[religion]++;
                if (s.charAt(t[religion] - 1) != t[0]) {
                    System.out.println("NO");
                    continue;
                }
                if (t[1] > t[0] && s.charAt(t[1] - 1) != t[2]) {
                    System.out.println("NO");
                    continue;
                }
                if (t[2] > t[0] && s.charAt(t[2] - 1) != t[3]) {
                    System.out.println("NO");
                    continue;
                }
                System.out.println("YES");
            } else {
                t[religion]--;
                System.out.println("YES");
            }
        }
    }
}
