n,t=map(int,input().split())
s1,s2=input(),input()
cnt0,cnt1=0,0
for i in range(n):
    if s1[i]!=s2[i]:
        if cnt0==t or cnt1==t:
            print(-1)
            exit()
        if s1[i]==s2[i]:cnt0+=1
        else:cnt1+=1
if cnt0+cnt1==t:
    for i in range(n):
        if s1[i]!=s2[i]:
            s1[i]=s2[i]
    print(s1)
else:print(-1)
