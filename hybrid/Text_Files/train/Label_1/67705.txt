from collections import deque

n, m = map(int, input().split())
maze = [input() for _ in range(n)]

sr, sc = None, None
for i in range(n):
    for j in range(m):
        if maze[i][j] == 'S':
            sr, sc = i, j
        elif maze[i][j] == 'E':
            er, ec = i, j

s = input()

ans = 0
def valid(r, c):
    return 0 <= r < n and 0 <= c < m and maze[r][c] != '#'

for dr in [-1, 0, 1, 0]:
    for dc in [0, 1, 0, -1]:
        for dr2 in [-1, 0, 1, 0]:
            for dc2 in [0, 1, 0, -1]:
                if dr == dr2 and dc == dc2:
                    continue
                ok = True
                r = sr
                c = sc
                for d in s:
                    nr, nc = r + dr * int(d) + dr2, c + dc * int(d) + dc2
                    if not valid(nr, nc):
                        ok = False
                        break
                    r, c = nr, nc
                if ok and r == er and c == ec:
                    ans += 1
print(ans)
