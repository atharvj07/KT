import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = sc.next().toCharArray();
        }
        int[][] dis = new int[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dis[i], -1);
        }
        Queue<int[]> q = new LinkedList<>();
        int[] dx = {0, 0, 1, -1};
        int[] dy = {1, -1, 0, 0};
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    q.add(new int[]{i, j});
                    dis[i][j] = 0;
                }
            }
        }
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m && dis[nx][ny] == -1) {
                    if (grid[nx][ny] == '.' || grid[nx][ny] == grid[x][y]) {
                        dis[nx][ny] = dis[x][y];
                        q.add(new int[]{nx, ny});
                    } else {
                        dis[nx][ny] = dis[x][y] + 1;
                        q.add(new int[]{nx, ny});
                    }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (dis[i][j] != -1) {
                    ans = Math.max(ans, dis[i][j]);
                }
            }
        }
        System.out.println(ans);
    }
}
