n,k=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
b.sort()
ai=0
bi=0
ok=True
while ai<n:
    if a[ai]==0:
        if bi<k and b[bi]>a[ai-1]:
            a[ai]=b[bi]
            bi+=1
        else:
            ok=False
            break
    ai+=1
if ok:
    for i in range(n-1):
        if a[i]>a[i+1]:
            ok=False
            break
if ok: print("Yes")
else: print("No")
