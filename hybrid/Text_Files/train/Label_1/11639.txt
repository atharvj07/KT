#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,t;
    cin>>t;
    while(t--){
        cin>>n;
        vector<pair<int,int> >v(n);
        for(int i=0;i<n;i++){
            int a,b;
            cin>>a>>b;
            v[i]={a,b};
        }
        vector<int>ans(n,-1);
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                if(v[i].first>v[j].first and v[i].second>v[j].second)
                    ans[i]=j+1;
                if(v[i].second>v[j].first and v[i].first>v[j].second)
                    ans[i]=j+1;
            }
        }
        for(int i=0;i<n;i++)
            cout<<ans[i]<<" ";
        cout<<endl;
    }
    return 0;
}
