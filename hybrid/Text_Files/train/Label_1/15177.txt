#include <bits/stdc++.h>
using namespace std;
const int MXN = 1e5+1;
int n, a[MXN];
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    if (n <= 2) {
        cout << 0 << endl;
        return 0;
    }
    sort(a, a + n);
    int l = 0, r = n - 1, ans = 0;
    while (l < r && a[l]*2 <= a[r]) {
        ans++;
        l++;
        r--;
    }
    cout << ans << endl;
}
