#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e5 + 5;
int f[MAXN], w[MAXN], s[MAXN], m[MAXN], vis[MAXN];
vector<int> adj[MAXN];
void dfs(int u, int len) {
    vis[u] = len;
    int next = f[u], wgt = w[u];
    if (vis[next] + 1 == len) {
        s[u] = s[next] + wgt;
        m[u] = min(m[next], wgt);
    }
    if (!vis[next]) dfs(next, len + 1);
}
int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> f[i];
    for (int i = 0; i < n; ++i) cin >> w[i];
    s[0] = w[0], m[0] = w[0];
    dfs(0, 1);
    for (int i = 0; i < n; ++i) cout << s[i] << " " << m[i] << endl;
}
