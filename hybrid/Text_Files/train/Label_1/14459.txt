#include <bits/stdc++.h>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::string s;
  std::cin >> s;
  std::set<char> st(s.begin(), s.end());
  while (st.size() < k) {
    for (char c = 'a'; c <= 'z'; c++) {
      if (st.find(c) == st.end()) {
        st.insert(c);
        break;
      }
    }
  }
  std::string t;
  for (auto c : st) { t += c; }
  std::cout << t << std::endl;
  return 0;
}
