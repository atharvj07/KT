import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long d = in.nextLong();
        int a = in.nextInt();
        int b = in.nextInt();
        List<Pair> clients = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            clients.add(new Pair(in.nextInt(), in.nextInt()));
        }
        Collections.sort(clients, (o1, o2) -> o1.low - o2.low);
        int ans = 0;
        long current = 0;
        List<Integer> indexes = new ArrayList<>();
        for (Pair client : clients) {
            long low = client.low * a;
            long high = client.high * b;
            if (current + low + high <= d) {
                ans++;
                current += low + high;
                indexes.add(client.index);
            }
        }
        System.out.println(ans);
        for (int index : indexes) {
            System.out.print(index + " ");
        }
    }

    static class Pair {
        int low;
        int high;
        int index;

        Pair(int low, int high) {
            this.low = low;
            this.high = high;
        }
    }
}
