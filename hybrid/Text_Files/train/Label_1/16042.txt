#include <bits/stdc++.h>
using namespace std;
const int N = 2005, inf = 1e9;
int c[N][N], dp[N][N];
int n, p[N], s[N], l[N], r[N];
vector<pair<int, int>> ans;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> p[i], l[p[i]] = i;
    for (int i = 1; i <= n; i++) cin >> s[i], r[s[i]] = i;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) {
            dp[i][j] = inf;
            if (l[s[j]] > l[s[j - 1]]) c[i][j] = c[i][j - 1] + 1;
            else
                c[i][j] = c[i][j - 1] - 1;
        }
    for (int i = n - 1; i >= 1; i--) {
        for (int j = 1; j <= n; j++) {
            int cur = c[i][j];
            if (cur < 0) cur--;
            dp[i][j] = min(dp[i][j], dp[i + 1][j] + abs(cur));
            if (r[s[j]] < r[s[j - 1]]) cur--;
            else
                cur++;
            dp[i][j] = min(dp[i][j], dp[i + 1][j] + abs(cur));
        }
    }
    for (int i = 1, j = 1;; i++) {
        int cur = c[i][j];
        if (cur < 0) cur--;
        if (dp[i][j] == dp[i + 1][j] + abs(cur)) { ans.push_back({i, j}); j++; }
        if (r[s[j]] < r[s[j - 1]]) cur--;
        else
            cur++;
        if (dp[i][j] == dp[i + 1][j] + abs(cur)) { ans.push_back({i, j}); j++; }
        if (j == n) break;
    }
    cout << ans.size() << "\n" << ans.size() << endl;
    for (auto x : ans) cout << x.first << " " << x.second << "\n";
}
