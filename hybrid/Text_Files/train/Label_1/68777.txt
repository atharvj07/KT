n,m=map(int,input().split())
a,b=[0]*m,[0]*m
for i in range(m):
    a[i],b[i]=map(int,input().split())
    a[i]-=1;b[i]-=1
f=0
for i in range(m):
    for j in range(i+1,m):
        if (b[j]-b[i]+n)%(a[j]-a[i])==0 and (a[j]-a[i]+n)%(b[j]-b[i])==0:
            f=1;break
    if f==1:break
print(['No','Yes'][f])
