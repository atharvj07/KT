import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            points[i] = new Point(sc.nextInt(), sc.nextInt());
        }
        boolean isConvex = true;
        for (int i = 0; i < n; i++) {
            double cross = (points[(i+1)%n].x - points[i].x) * (points[(i+2)%n].y - points[i].y) - (points[(i+1)%n].y - points[i].y) * (points[(i+2)%n].x - points[i].x);
            if (cross < 0) {
                isConvex = false;
                break;
            }
        }
        System.out.println(isConvex ? 1 : 0);
    }

    static class Point {
        final int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
