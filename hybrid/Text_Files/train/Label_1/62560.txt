import heapq
inf = 10 ** 18
n, m = map(int, input().split())
L = [[] for _ in range(n+1)]
for _ in range(m):
  a, b, w = map(int, input().split())
  L[a].append([w, b])
  L[b].append([w, a])
D = [inf] * (n+1)
prv = [0]*(n+1)
ans = [inf]*(n+1)
a = list(map(int, input().split()))
Q = [(0, 1)]
while Q:
  d, x = heapq.heappop(Q)
  if D[x] < d: continue
  D[x] = d
  for w, y in L[x]:
    if d + 2*w < D[y]:
      D[y] = d + 2*w
      prv[y] = x
      heapq.heappush(Q, (d+2*w, y))
for i in range(1, n+1):
  ans[i] = a[i-1]
  if i != 1:
    ans[i] = min(ans[i], ans[prv[i]])
print(*ans[1:])
