import heapq

m, n, k, t = map(int, input().split())
a = list(map(int, input().split()))
traps = []

for _ in range(k):
    l, r, d = map(int, input().split())
    traps.append((l - 1, r - 1, d))  # Convert to 0-based indexing

traps.sort()

# Dijkstra's algorithm

dist = [10**9] * (n + 1)
dist[0] = 0
pos = 0
# The priority queue stores (dist, pos) pairs
pq = [(0, 0)]

while pq:
    d, pos = heapq.heappop(pq)
    
    if pos == n:
        break
    
    # Move without squad
    if pos + 1 <= n and dist[pos + 1] > d + 1:
        dist[pos + 1] = d + 1
        heapq.heappush(pq, (d + 1, pos + 1))
    
    # Move with squad
    for i in range(pos + 1, n + 1):
        is_safe = True
        for l, r, d in traps:
            if l <= i <= r and d > a[0]:
                is_safe = False
                break
        
        if is_safe and dist[i] > d + 1:
            dist[i] = d + 1
            heapq.heappush(pq, (d + 1, i))
    
    # Disarm traps
    for l, r, d in traps:
        if pos == l:
            for i in range(l + 1, n + 1):
                is_safe = True
                for l2, r2, d2 in traps:
                    if l2 <= i <= r2 and d2 > a[0]:
                        is_safe = False
                        break
                
                if is_safe and dist[i] > d + 1:
                    dist[i] = d + 1
                    heapq.heappush(pq, (d + 1, i))
            break

ans = 0
for i in range(m):
    if dist[n] <= t:
        ans += 1
    a.pop(0)

print(ans)
