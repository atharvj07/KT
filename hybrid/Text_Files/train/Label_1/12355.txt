#include<bits/stdc++.h>
using namespace std;
int main()
{
    int p, n;
    cin >> p >> n;
    int a[p+5];
    memset(a, -1, sizeof(a));
    int ans = -1;
    for(int i = 0; i < n; i++)
    {
        int x;
        cin >> x;
        x = x % p;
        if(a[x] != -1)
        {
            ans = i+1;
            break;
        }
        a[x] = x;
    }
    cout << ans << endl;
}
