import java.util.Scanner;

public class Main {
    static int k, l, r;
    static long m = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        k = in.nextInt();
        l = in.nextInt();
        r = in.nextInt();
        System.out.println((modPow(2, r - l + 1) - modPow(2, l - 1)) % m);
    }

    static long modPow(long base, long p) {
        if (p == 0)
            return 1;
        if (p == 1)
            return base % m;
        long val = modPow(base, p / 2) % m;
        val = (val * val) % m;
        if (p % 2 != 0)
            val = (val * base) % m;
        return val % m;
    }
}
