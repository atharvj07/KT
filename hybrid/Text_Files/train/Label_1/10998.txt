#include<bits/stdc++.h>
using namespace std;
const int mxN=1000;
int main(){
    int n; cin>>n;
    vector<long long>arr(n);
    for(int i=0;i<n;i++) cin>>arr[i];
    sort(arr.rbegin(),arr.rend());
    long long ans=0;
    for(int i=0;i<n;i++)
        if(arr[i]>ans+1) ans+=arr[i];
    if(ans==0) cout<<-1<<endl;
    else cout<<ans+1<<endl;
}
