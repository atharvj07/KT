n, k = map(int, input().split())
p = list(map(int, input().split()))
q = list(map(int, input().split()))
used = [False] * (n + 1)
lst = {}
for i in range(n):
    if p[i] < q[i]:
        lst[p[i]] = q[i]
    else:
        lst[q[i]] = p[i]
char = 'a'
ans = [''] * n
for i in range(n):
    if not used[p[i]]:
        while used[ord(char) - ord('a') + 1]:
            char = chr(ord(char) + 1)
        ans[p[i] - 1] = char
        used[ord(char) - ord('a') + 1] = True
        if char in lst:
            ans[lst[char] - 1] = char
            used[ord(char) - ord('a') + 1] = True
    if not used[q[i]]:
        while used[ord(char) - ord('a') + 1]:
            char = chr(ord(char) + 1)
        ans[q[i] - 1] = char
        used[ord(char) - ord('a') + 1] = True
        if char in lst:
            ans[lst[char] - 1] = char
            used[ord(char) - ord('a') + 1] = True
if len(set(ans)) < k:
    print("NO")
else:
    print("YES")
    print(''.join(ans))
