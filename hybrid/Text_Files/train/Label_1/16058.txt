#include <iostream>
using namespace std;

const long long MAX_N = 1e5 + 5;
long long n, k, a[MAX_N], dp[MAX_N];

int main() {
  cin >> n >> k;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
  }
  long long ans = 0, min_ans = 1e18;
  for (long long i = 1; i <= n; i += k) {
    dp[i] = a[i];
  }
  for (long long i = 1; i <= n; i++) {
    for (long long j = min(i + k - 1, n); j >= i; j--) {
      dp[i] = min(dp[i], dp[j] + a[i]);
    }
  }
  for (long long i = 1; i <= n; i += k) {
    if (min_ans > dp[i]) {
      min_ans = dp[i];
      ans = i;
    }
  }
  cout << ans << endl;
  return 0;
}
