import bisect
n, k = map(int,input().split())
p = []
for i in range(n):
    l,r = map(int,input().split())
    p.append([l,r])
p.sort()
for i in range(n):
    l,r = p[i]
    v = (r//-k)*k+1
    if v<=l:
        ans = 0
        break
    ans = v-l
    if ans%k==0:
        ans-=1
    ans//=k
    ans+=i
    break
for l,r in p[i+1:]:
    v = (r//-k)*k+1
    if v<=l:
        ans+=2
        break
    ans+=1
print(ans)
