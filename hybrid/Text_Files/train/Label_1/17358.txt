#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int maxn = 2e5 + 5;
int n;
int a[maxn], pos[maxn], l[maxn], r[maxn];
vector<int> res;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    pos[a[i]] = i;
  }
  int q = 0;
  for (int i = 1; i <= n; i++) {
    if (a[i] == i) continue;
    q++;
    int j = pos[i];
    int sl = l[j];
    int sr = r[j];
    int len = sr - sl + 1;
    if (sl) {
      r[sl] = sr;
      l[sr] = sl;
      if (a[sl] == sl) q--;
    } else {
      l[j] = j;
      r[j] = j;
      if (a[j + 1] == j + 1) q--;
    }
    pos[a[j]] = pos[a[i]];
    swap(a[j], a[i]);
    pos[a[i]] = i;
  }
  for (int i = 1; i <= n; i++) {
    res.push_back(a[i]);
  }
  cout << q << endl;
  for (int x : res) { cout << x << " "; }
  cout << endl;
  return 0;
}
