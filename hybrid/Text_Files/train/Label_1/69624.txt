def find_coordinates(n):
    if n == 0:
        return (0, 0)
    # Determine the ring on which the point after n moves lies
    ring = int(((3 + (12*n - 3)**0.5) / 6)**0.5)
    start_of_ring_n = 3 * ring * (ring - 1) + 1
    position_in_ring = n - start_of_ring_n
    if position_in_ring < ring:
        return (ring, position_in_ring)
    position_in_ring -= ring
    if position_in_ring < ring:
        return (ring - position_in_ring, ring)
    position_in_ring -= ring
    if position_in_ring < ring:
        return (-position_in_ring, ring - position_in_ring)
    position_in_ring -= ring
    if position_in_ring < ring:
        return (-ring, -position_in_ring)
    position_in_ring -= ring
    if position_in_ring < ring:
        return (position_in_ring - ring, -ring)
    position_in_ring -= ring
    return (position_in_ring, position_in_ring - ring)

n = int(input().strip())
print(*find_coordinates(n))
