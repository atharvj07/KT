#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, ans = 0;
        cin >> n >> x;
        vector<int> v(101, 0);
        for (int i = 0; i < n; i++) {
            int a;
            cin >> a;
            v[a] = 1;
        }
        int now = n;
        now += x;
        for (int i = 1; i <= 100; i++) {
            if (!v[i]) {
                if (now > 0) {
                    now--;
                    ans++;
                } else
                    break;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
