#include <vector>
#include <iostream>
#include <algorithm>
#define ll long long

using namespace std;

vector<ll> adj[100000];
int vis[100000];
vector<ll> ans;

void dfs(ll node, ll par){
    vis[node]=1;
    for(ll child: adj[node]){
        if(!vis[child]){
            dfs(child,node);
        }
    }
    ans.push_back(node);
}

int main() {
    ll n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        ll v, u;
        cin >> v >> u;
        adj[v].push_back(u);
    }
    for (ll i = 1; i <= n; i++) {
        if (!vis[i]) {
            dfs(i, -1);
        }
    }
    reverse(ans.begin(), ans.end());
    for (ll i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
}
