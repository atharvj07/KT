#include <iostream>
using namespace std;
#define mod 1000000007

long long fastPower(long long base, long long p) {
  long long ans = 1;
  while (p > 0) {
    if (p % 2 == 0) {
      base = (base * base) % mod;
      p = p / 2;
    } else {
      ans = (ans * base) % mod;
      p = p - 1;
    }
  }
  return ans;
}

int main() {
  long long n, m;
  cin >> n >> m;
  long long ans = fastPower(m, n / m) % mod;
  if (n % m != 0) ans = (ans * m) % mod;
  cout << ans << endl;
}
