import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) b[i] = scanner.nextInt();
            int[] a = new int[n];
            int l = 0;
            int r = n - 1;
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    a[l] = b[i];
                    l++;
                } else {
                    a[r] = b[i];
                    r--;
                }
            }
            for (int ai : a) System.out.print(ai + " ");
            System.out.println();
        }
        scanner.close();
    }
}
