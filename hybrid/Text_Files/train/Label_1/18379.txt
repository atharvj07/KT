#include <bits/stdc++.h>
using namespace std;

int main() {
    string n;
    cin >> n;
    int ans = 0, base = 1;
    for (int i = n.size() - 1; i >= 0; i--) {
        if (n[i] == '4') ans += base;
        if (n[i] == '7') ans += base;
        base *= 2;
    }
    cout << ans + 1 << endl;
    return 0;
}
