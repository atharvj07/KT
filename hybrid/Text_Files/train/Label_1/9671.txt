#include <bits/stdc++.h>

using namespace std;

const int N = 1e3 + 5;

vector<int> g[N];
vector<int> govt;
bool vis[N];

void dfs(int u) {
  vis[u] = true;
  for (auto &v : g[u]) {
    if (!vis[v]) dfs(v);
  }
}

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= k; i++) {
    int x;
    cin >> x;
    govt.push_back(x);
  }
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  int comp = 0;
  for (auto &u : govt) {
    if (!vis[u]) {
      dfs(u);
      comp++;
    }
  }
  int ans = n - m - comp + 1;
  cout << ans << endl;
  return 0;
}
