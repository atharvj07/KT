import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        String l_str = sc.nextLine();
        String r_str = sc.nextLine();
        StringBuilder l_sb = new StringBuilder();
        int l_diff = n - l_str.length();
        for (int i = 0; i < l_diff; i++) {
            l_sb.append('0');
        }
        l_sb.append(l_str);
        l_str = l_sb.toString();
        int l = Integer.parseInt(l_str, 2);
        int r = Integer.parseInt(r_str, 2);
        int max_xor = 0;
        for (int i = l; i <= r; i++) {
            max_xor ^= i;
        }
        String ans = Integer.toBinaryString(max_xor);
        System.out.println(ans);
    }
}
