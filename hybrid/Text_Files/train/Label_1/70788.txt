n,m,q=[int(x)for x in input().split()]
a=[int(x)for x in input().split()]
b=[int(x)for x in input().split()]
for i in range(q):
    l,r=[int(x)for x in input().split()]
    ans=0
    for j in range(l-1,r):
        if(b[j]==a[ans]):
            ans+=1
            if(ans==n):
                print(1)
                break
    if(ans<n):
        print(0)
