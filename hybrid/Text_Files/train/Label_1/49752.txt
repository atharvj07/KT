import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String password = sc.nextLine();
        int n = sc.nextInt();
        boolean[] possible = new boolean[password.length() + 1];
        Arrays.fill(possible, 0, 2, true);
        while (n-- > 0) {
            String word = sc.nextLine();
            for (int i = password.length(); i >= word.length(); i--) {
                if (possible[i - word.length()] && password.substring(i - word.length(), i).equals(word)) {
                    possible[i] = true;
                }
            }
        }
        System.out.println(possible[password.length()] ? "YES" : "NO");
    }
}
