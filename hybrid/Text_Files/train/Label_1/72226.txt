from collections import deque

n = int(input())
a = input()
q = deque()
for i in range(n):
    if a[i] == '.': continue
    if a[i] == 'P':
        q.append((i, 0))
    else:
        q.append((i, 1))
b = [0] * n
ans = 0
while q:
    i, d = q.popleft()
    ans = max(ans, d)
    if i - 1 >= 0 and a[i - 1] != '.' and not b[i - 1]:
        b[i - 1] = 1
        q.append((i - 1, d + 1))
    if i + 1 < n and a[i + 1] != '.' and not b[i + 1]:
        b[i + 1] = 1
        q.append((i + 1, d + 1))
print(ans)
