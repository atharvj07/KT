import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<String> strings = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int si = scanner.nextInt();
            int fi = scanner.nextInt();

            StringBuilder sb = new StringBuilder();
            for (int j = 0; j < fi; j++) {
                sb.append(si);
            }
            for (int j = fi; j < n; j++) {
                sb.append(1 - si);
            }
            strings.add(sb.toString());
        }

        Map<String, Integer> frequencies = new HashMap<>();
        for (String string : strings) {
            frequencies.put(string, frequencies.getOrDefault(string, 0) + 1);
        }

        List<String> distinctStrings = new ArrayList<>(frequencies.keySet());
        Collections.sort(distinctStrings);

        int maxSum = 0;
        int count = 0;
        for (int i = 0; i < distinctStrings.size(); i++) {
            for (int j = i + 1; j < distinctStrings.size(); j++) {
                for (int k = j + 1; k < distinctStrings.size(); k++) {
                    int sum = calculateHammingDistance(distinctStrings.get(i), distinctStrings.get(j)) +
                            calculateHammingDistance(distinctStrings.get(j), distinctStrings.get(k)) +
                            calculateHammingDistance(distinctStrings.get(k), distinctStrings.get(i));

                    if (sum > maxSum) {
                        maxSum = sum;
                        count = 0;
                    }

                    if (sum == maxSum) {
                        count += frequencies.get(distinctStrings.get(i)) * frequencies.get(distinctStrings.get(j)) * frequencies.get(distinctStrings.get(k));
                    }
                }
            }
        }

        System.out.println(count);
    }

    private static int calculateHammingDistance(String string1, String string2) {
        int distance = 0;
        for (int i = 0; i < string1.length(); i++) {
            if (string1.charAt(i) != string2.charAt(i)) {
                distance++;
            }
        }
        return distance;
    }
}
