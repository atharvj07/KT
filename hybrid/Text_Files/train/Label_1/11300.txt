#include <iostream>
#include <set>

using namespace std;

int main() {
  int n;
  cin >> n;
  set<int> calls;
  int x;
  for (int i = 0; i < n; i++) {
    cin >> x;
    if (x != 0) calls.insert(x);
  }
  if (calls.size() % 2)
    cout << -1 << endl;
  else
    cout << calls.size() / 2 << endl;
  return 0;
}
