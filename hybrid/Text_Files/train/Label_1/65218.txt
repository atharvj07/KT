import sys

def is_nearly_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return True
    return False

def solve(n):
    if n & 1:
        if n >= 7:
            return "YES 2 3 6 " + str(n - 11)
        else:
            return "NO"
    else:
        if n % 4 != 0:
            if n >= 6:
                return "YES 2 3 " + str(n - 5)
            else:
                return "NO"
        else:
            m = n // 4
            if is_nearly_prime(m):
                return "YES " + str(m) + " " + str(m) + " " + str(m) + " " + str(n - 3*m)
            else:
                return "NO"

for line in sys.stdin:
    n = int(line)
    print(solve(n))
