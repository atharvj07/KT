def solve(n, a):
    result = []
    # Initial string made of 'a's, ensuring maximum length based on the largest prefix needed plus extra padding.
    str_base = 'a' * (max(a) + 1)  
    result.append(str_base)
    for i in range(n):
        # Generate the next string by changing the character just after the common prefix
        # to ensure the right length of the longest common prefix.
        prefix_length = a[i]
        if result[-1][prefix_length] == 'a':
            new_str = result[-1][:prefix_length] + 'b' + result[-1][prefix_length + 1:]
        else:
            new_str = result[-1][:prefix_length] + 'a' + result[-1][prefix_length + 1:]
        result.append(new_str)
    return result

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    answer = solve(n, a)
    for s in answer:
        print(s)
