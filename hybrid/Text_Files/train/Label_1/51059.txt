import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] cups = new int[n];
        for (int i = 0; i < n; i++) {
            cups[i] = sc.nextInt();
        }
        sc.close();
        int max = cups[0], min = cups[0], a = 0, b = 0;
        for (int i = 1; i < n; i++) {
            if (cups[i] > max) {
                max = cups[i];
                a = i;
            }
            if (cups[i] < min) {
                min = cups[i];
                b = i;
            }
        }
        int diff = max - min;
        int j;
        for (j = 0; j < n; j++) {
            if (cups[j] != min && cups[j] != max) {
                break;
            }
        }
        if (j == n) {
            System.out.println("Exemplary pages.");
        } else if (cups[j] + diff == max) {
            System.out.printf("%d ml. from cup #%d to cup #%d.", diff, b + 1, j + 1);
        } else {
            System.out.println("Unrecoverable configuration.");
        }
    }
}
