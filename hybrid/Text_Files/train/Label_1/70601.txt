n, x = map(int, input().split())
d = list(map(int, input().split()))

# Prepare prefix sums for days and hugs
days_prefix_sum = [0] * (n + 1)
hugs_prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    days_prefix_sum[i] = days_prefix_sum[i - 1] + d[i - 1]
    hugs_prefix_sum[i] = hugs_prefix_sum[i - 1] + d[i - 1] * (d[i - 1] + 1) // 2

max_hugs = 0
for i in range(1, n + 1):
    # Find the end of the vacation starting from i, using binary search
    left, right = 1, n
    while left < right:
        mid = (left + right + 1) // 2
        if days_prefix_sum[mid] - days_prefix_sum[i - 1] <= x:
            left = mid
        else:
            right = mid - 1
    end = left

    # Calculate remaining days if the vacation doesn't exactly fit into full months
    remaining_days = x - (days_prefix_sum[end] - days_prefix_sum[i - 1])

    # Calculate total hugs
    total_hugs = hugs_prefix_sum[end] - hugs_prefix_sum[i - 1]
    if remaining_days > 0:
        if end == n:
            end = 0
        total_hugs += remaining_days * (remaining_days + 1) // 2 + (d[end] * (d[end] + 1) // 2 - (d[end] - remaining_days) * (d[end] - remaining_days + 1) // 2)

    max_hugs = max(max_hugs, total_hugs)

print(max_hugs)
