#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
int main() {
    int n;
    cin >> n;
    vector<int> x(n), v(n);
    for (int i = 0; i < n; i++) cin >> x[i];
    for (int i = 0; i < n; i++) cin >> v[i];
    if (n >= 5000) {
        cout << -1;
        return 0;
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (v[j] == v[i]) continue;
            if (x[j] < x[i] && v[j] > v[i]) {
                ll t = (x[j] - x[i]);
                t *= mod;
                t /= (v[j] - v[i]);
                ans += abs(x[i] + t * v[i] - x[j] - t * v[j]);
                ans %= mod;
            }
            if (x[i] < x[j] && v[i] > v[j]) {
                ll t = (x[i] - x[j]);
                t *= mod;
                t /= (v[i] - v[j]);
                ans += abs(x[i] + t * v[i] - x[j] - t * v[j]);
                ans %= mod;
            }
        }
    }
    cout << ans;
    return 0;
}
