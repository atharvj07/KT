#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2005;
int n;
char s[MAXN];

bool check(int i, int t) {
    if (s[i] == s[i - t] && s[i] == s[i - 2 * t] && s[i] == s[i - 3 * t]) {
        return true;
    }
    return false;
}

int main() {
    cin >> n;
    cin >> s;
    int ans = 0;
    for (int i = 4; i < n; i++) {
        if (check(i, 1)) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
