import sys

n, m = map(int, sys.stdin.readline().split())
adj = [[] for _ in range(n)]
for i in range(m):
    a, b, c = map(int, sys.stdin.readline().split())
    a -= 1; b -= 1
    if c == 1:
        adj[a].append(b)
        adj[b].append(a)
    else:
        adj[a].append(-b)
        adj[b].append(-a)
ans = 0

def dfs(u, p, same):
    global ans
    if same == 1:
        for v in adj[u]:
            if v < 0:
                if dfs(abs(v), u, -1) == -1:
                    return -1
            else:
                if dfs(v, u, 1) == -1:
                    return -1
        ans += 1
        return 1
    elif same == -1:
        state = [0] * n
        cnt = 0
        for v in adj[u]:
            if v < 0:
                if state[abs(v)] == 0:
                    state[abs(v)] = -1
                    cnt += 1
            else:
                if state[v] == 0:
                    state[v] = 1
                    cnt += 1
        if cnt > 2:
            return -1
        for v in adj[u]:
            if v < 0:
                if state[abs(v)] == same:
                    if dfs(abs(v), u, -1) == -1:
                        return -1
            else:
                if state[v] == same:
                    if dfs(v, u, 1) == -1:
                        return -1
        ans += 1
        return 1
    else:
        for v in adj[u]:
            if v < 0:
                if dfs(abs(v), u, -1) == -1:
                    return -1
            else:
                if dfs(v, u, 1) == -1:
                    return -1
        ans += 1
        return 1
dfs(0, -1, 0)
print(ans % (10 ** 9 + 7))
