n, m = map(int, input().split())
f = list(map(int, input().split()))
b = list(map(int, input().split()))
a = [-1] * n
for i in range(m):
    if a[b[i]] == -1: a[b[i]] = f[b[i]]
    elif a[b[i]] != f[b[i]]: print("Ambiguity"); exit()
if -1 in a: print("Impossible")
else:
    print("Possible")
    for i in b: print(a[i], end=" ")
