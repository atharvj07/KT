import sys
input = sys.stdin.readline
n,m=map(int,input().split())
l=[]
r=[]
for i in range(n):
    p,q=map(int,input().split())
    l.append(p)
    r.append(q)
b=[0]*n
a=sorted(list(map(int,input().split())))
x,i=0,0
while x<n and i<m:
    x1=0
    while l[x]<l[x]+a[i] and x+1<n:
        if r[x]+1>l[x]+a[i]:
            x1=1
            b[x]=i+1
        x+=1
    if not x1: i+=1
    x+=1
if x==n and i==m:
    print("Yes")
    print(*b)
else: print("No")
