import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner cin = new Scanner(System.in);
		int n,i,curr = 1,old = 0,temp;
		n = cin.nextInt();
		for(i=1; i<=n; i++)
		{
			if((i == curr) || (i == old))
				System.out.print("O");
			else
				System.out.print("o");
			temp = curr;
			curr = curr + old;
			old = temp;
		}
	}
}
