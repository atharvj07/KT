#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, a, b, c, d, ab[10] = {}, ad[10] = {}, bc[10] = {}, bd[10] = {};
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a >> b;
        ab[a]++;
        ab[b]++;
    }
    for (int i = 0; i < m; i++) {
        cin >> c >> d;
        bc[c]++;
        bc[d]++;
    }
    for (int i = 1; i <= 9; i++) {
        ad[i] = ab[i] - bc[i];
        bd[i] = bc[i] - ab[i];
    }
    int cnt = 0, ans;
    for (int i = 1; i <= 9; i++) {
        if (ad[i] == 1 && bd[i] == -1)
            cnt++, ans = i;
    }
    if (cnt == 0)
        cout << -1;
    else if (cnt == 1)
        cout << ans;
    else
        cout << 0;
    return 0;
}
