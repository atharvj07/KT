import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] ids = new int[n];
        for (int i = 0; i < n; i++) {
            ids[i] = sc.nextInt();
        }
        int[] counts = new int[n];
        int count = 1;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (ids[i] == ids[j]) {
                    count++;
                }
            }
            counts[i] = count;
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (sum + counts[i] >= k) {
                System.out.println(ids[i]);
                return;
            }
            sum += counts[i];
        }
    }
}
