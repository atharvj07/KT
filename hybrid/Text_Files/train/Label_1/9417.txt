#include <cstdio>
int main() {
    long long int t;
    scanf("%lld", &t);
    while (t--) {
        long long int x1, y1, z1, x2, y2, z2, a[1000000], b[1000000], c[1000000], s1 = 0, s2 = 0, ans = 0;
        scanf("%lld%lld%lld", &x1, &y1, &z1);
        scanf("%lld%lld%lld", &x2, &y2, &z2);
        while (x1 > 0) { a[s1++] = 0; x1--; }
        while (y1 > 0) { a[s1++] = 1; y1--; }
        while (z1 > 0) { a[s1++] = 2; z1--; }
        while (x2 > 0) { b[s2++] = 0; x2--; }
        while (y2 > 0) { b[s2++] = 1; y2--; }
        while (z2 > 0) { b[s2++] = 2; z2--; }
        for (long long int i = 0, j = 0; i < s1 && j < s2; i++, j++) {
            if (a[i] > b[j]) { ans += a[i] * b[j]; } else if (a[i] == b[j]) { ans += 0; } else { ans -= a[i] * b[j]; }
        }
        printf("%lld\n", ans);
    }
    return 0;
}
