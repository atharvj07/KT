a = input().replace('0', 'a').replace('1', 'b')
b = input()
ans = 0
for i in range(26):
    for j in range(26):
        if chr(97 + i) in b and chr(97 + j) in b and chr(97 + i) != chr(97 + j):
            if a.replace('a', chr(97 + i)).replace('b', chr(97 + j)) == b:
                ans += 1
print(ans)
