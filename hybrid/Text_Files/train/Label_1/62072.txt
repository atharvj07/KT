n = int(input())
s = []
for i in range(n):
    s.append(list(input()))
exist = False
for i in range(n):
    for j in range(4):
        if s[i][j] == 'O' and (j % 2 == 0 or s[i][j + 1] == 'O'):
            s[i][j] = '+'
            s[i][j + 1] = '+'
            exist = True
            break
if exist:
    print('YES')
    for i in range(n):
        print(''.join(s[i]))
else:
    print('NO')
