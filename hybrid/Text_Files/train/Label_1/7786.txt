#include <iostream>
#include <string>
#include <unordered_map>

int main() {
  int q;
  std::cin >> q;
  while (q--) {
    std::string s, t, p;
    std::cin >> s >> t >> p;
    std::unordered_map<char, int> cnt;
    for (char c : s) ++cnt[c];
    for (char c : t) --cnt[c];
    bool ok = true;
    for (auto& [k, v] : cnt) {
      if (v != 0) ok = false;
    }
    if (p.size() < t.size() - s.size()) ok = false;
    std::cout << (ok ? "YES" : "NO") << '\n';
  }
  return 0;
}
