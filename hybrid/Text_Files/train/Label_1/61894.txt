def read_square():
    coords = list(map(int, input().split()))
    points = [(coords[i], coords[i + 1]) for i in range(0, 8, 2)]
    return points

def square_intersects(axial_square, angled_square):
    # Find the bounding box of the angled square
    min_x = min_y = float('inf')
    max_x = max_y = -float('inf')
    for x, y in angled_square:
        min_x = min(min_x, x)
        max_x = max(max_x, x)
        min_y = min(min_y, y)
        max_y = max(max_y, y)
    
    # Check if any of the vertices of the angled square are inside the axial square
    for x, y in angled_square:
        if axial_square[0][0] <= x <= axial_square[2][0] and axial_square[0][1] <= y <= axial_square[2][1]:
            return True
            
    # Check if any of the vertices of the axial square are inside the angled square's bounding box
    for x, y in axial_square:
        if min_x <= x <= max_x and min_y <= y <= max_y:
            return True
    
    # Check if edges intersect (cross product)
    def cross_product(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    def on_segment(p, q, r):
        return (q[0] <= max(p[0], r[0]) and q[0] >= min(p[0], r[0]) and
                q[1] <= max(p[1], r[1]) and q[1] >= min(p[1], r[1]))
    
    def segments_intersect(p1, q1, p2, q2):
        o1 = cross_product(p1, q1, p2)
        o2 = cross_product(p1, q1, q2)
        o3 = cross_product(p2, q2, p1)
        o4 = cross_product(p2, q2, q1)
        
        if o1 != o2 and o3 != o4:
            return True
        
        if o1 == 0 and on_segment(p1, p2, q1): return True
        if o2 == 0 and on_segment(p1, q2, q1): return True
        if o3 == 0 and on_segment(p2, p1, q2): return True
        if o4 == 0 and on_segment(p2, q1, q2): return True
        
        return False
    
    for i in range(4):
        if segments_intersect(axial_square[i], axial_square[(i + 1) % 4], angled_square[0], angled_square[2]) or \
           segments_intersect(axial_square[i], axial_square[(i + 1) % 4], angled_square[1], angled_square[3]):
            return True
            
    return False

axial_square = read_square()
angled_square = read_square()

if square_intersects(axial_square, angled_square):
    print("YES")
else:
    print("NO")
