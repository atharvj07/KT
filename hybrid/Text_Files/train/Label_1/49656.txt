import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] ink = new long[n];
        for (int i = 0; i < n; i++) {
            ink[i] = sc.nextLong();
        }
        long minInk = ink[0];
        int minPen = 1;
        for (int i = 1; i < n; i++) {
            if (ink[i] < minInk) {
                minInk = ink[i];
                minPen = i + 1;
            }
        }
        System.out.println(minPen);
    }
}
