n, p, q = map(int, input().split())
a, b = [], []
for i in range(n):
    tmp = [int(x) for x in input().split()]
    a.append(tmp[0])
    b.append(tmp[1])
ans = (max(0, p - a[0]) / (a[1] - a[0]) if a[1] != a[0] else float('inf'))
for i in range(1, n):
    for j in range(i):
        if b[j] < b[i]:
            tmp = (max(0, p - a[j]) / (a[i] - a[j]) if a[i] != a[j] else float('inf'))
            if tmp < ans:
                ans = tmp
        elif b[j] == b[i]:
            if a[j] > a[i]:
                ans = 0
print("{:.15f}".format(ans))
