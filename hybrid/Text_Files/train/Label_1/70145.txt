import math

m = int(input())
h1, a1 = map(int, input().split())
x1, y1 = map(int, input().split())
h2, a2 = map(int, input().split())
x2, y2 = map(int, input().split())

if (x1 * h1 + y1 - a1) % m != (x2 * h2 + y2 - a2) % m:
    print(-1)
else:
    lcm = math.lcm(m, x1)
    lcm = math.lcm(lcm, x2)
    time = ((x1 * h1 + y1 - a1) % m * int(lcm / x1)) % m
    time = (time + ((x2 * h2 + y2 - a2) % m * int(lcm / x2)) % m) % m
    time = time / m
    print(time)
