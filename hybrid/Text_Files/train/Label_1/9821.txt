#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, h;
  cin >> n >> m >> h;
  vector<int> u(n);
  for (int i = 0; i < n; i++) cin >> u[i];
  vector<pair<int, int>> c(m);
  for (int i = 0; i < m; i++) {
    cin >> c[i].first >> c[i].second;
    c[i].first--, c[i].second--;
  }
  vector<int> cnt(h);
  for (auto x : u) cnt[x]++;
  int ans = 0, mx = 0;
  for (int i = 0; i < h; i++) {
    if (cnt[i] > mx) {
      mx = cnt[i];
      ans = i;
    }
  }
  cout << n - mx << endl;
  for (int i = 0; i < n; i++) {
    if (u[i] != ans) cout << i + 1 << " ";
  }
  return 0;
}
