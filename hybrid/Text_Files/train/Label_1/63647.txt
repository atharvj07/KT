t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    instructions = input()
    prefix_sum = [0]
    min_prefix = [0]
    max_prefix = [0]
    for i in instructions:
        if i == '+':
            prefix_sum.append(prefix_sum[-1] + 1)
        else:
            prefix_sum.append(prefix_sum[-1] - 1)
        min_prefix.append(min(min_prefix[-1], prefix_sum[-1]))
        max_prefix.append(max(max_prefix[-1], prefix_sum[-1]))

    suffix_sum = [0]
    min_suffix = [0]
    max_suffix = [0]
    for i in instructions[::-1]:
        if i == '+':
            suffix_sum.append(suffix_sum[-1] - 1)
        else:
            suffix_sum.append(suffix_sum[-1] + 1)
        min_suffix.append(min(min_suffix[-1], suffix_sum[-1]))
        max_suffix.append(max(max_suffix[-1], suffix_sum[-1]))
    min_suffix.reverse()
    max_suffix.reverse()

    for _ in range(m):
        l, r = map(int, input().split())
        l -= 1
        min_val = min(min_prefix[l], prefix_sum[l] + min_suffix[r] - suffix_sum[n - r])
        max_val = max(max_prefix[l], prefix_sum[l] + max_suffix[r] - suffix_sum[n - r])
        print(max_val - min_val + 1)
