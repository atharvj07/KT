import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        input.nextLine();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            board[i] = input.nextLine().toCharArray();
        }
        boolean[][] visited = new boolean[n][n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !visited[i][j]) {
                    if (helper(board, i, j, visited)) {
                        count++;
                    }
                }
            }
        }
        if (count == n * n) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean helper(char[][] board, int i, int j, boolean[][] visited) {
        if (i < 0 || j < 0 || i >= board.length || j >= board[0].length || visited[i][j] || board[i][j] == '#') {
            return false;
        }
        visited[i][j] = true;
        boolean result = helper(board, i - 1, j - 1, visited) && helper(board, i, j - 1, visited) && helper(board, i + 1, j - 1, visited) && helper(board, i + 1, j, visited) && helper(board, i + 1, j + 1, visited);
        if (result) {
            board[i][j] = '#';
            board[i - 1][j - 1] = '#';
            board[i][j - 1] = '#';
            board[i + 1][j - 1] = '#';
            board[i + 1][j] = '#';
            board[i + 1][j + 1] = '#';
        }
        return result;
    }
}
