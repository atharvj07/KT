#include<bits/stdc++.h>
#include<ext/pb_ds/assoc_container.hpp>
using namespace __gnu_pbds;
using namespace std;
#define all(x) x.begin(), x.end()
void solve(){
    string str;
    cin>>str;
    int n=str.length();
    vector<vector<int>>dp(n, vector<int>(n));
    vector<int>ans(n);
    for(int len=0; len<n; len++){
        for(int l=0, r=len; r<n; l++, r++){
            if(len==0)
                dp[l][r]=1;
            else if(len==1)
                dp[l][r]=(str[l]==str[r]);
            else
                dp[l][r]=(str[l]==str[r] && dp[l+1][r-1]);
        }
    }
    for(int len=0; len<n; len++){
        for(int l=0, r=len; r<n; l++, r++){
            ans[len]+=dp[l][r];
        }
    }
    for(auto it: ans)
        cout<<it<<" ";
    cout<<endl;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t=1;
    //cin>>t;
    while(t--)
        solve();
    return 0;
}
