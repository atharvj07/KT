n = int(input())
a = list(map(int, input().split()))
b = [0] * n
c = [0] * n
b[n - 1] = a[n - 1]
c[0] = a[0]
for i in range(n - 2, -1, -1):
    b[i] = max(b[i + 1], a[i])
for i in range(1, n):
    c[i] = max(c[i - 1], a[i])
ans = 0
for i in range(n):
    if a[i] + a[i] == b[i] + c[i]:
        ans += 1
print(ans)
