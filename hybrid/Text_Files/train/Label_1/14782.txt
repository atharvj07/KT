#include <bits/stdc++.h>
using namespace std;
int main() {
    string s, t;
    cin >> s >> t;
    map<char, char> swaps;
    set<char> used;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] != t[i]) {
            if (swaps.find(s[i]) != swaps.end()) {
                if (swaps[s[i]] != t[i]) {
                    cout << "-1\n";
                    return 0;
                }
            } else {
                if (used.find(t[i]) != used.end()) {
                    cout << "-1\n";
                    return 0;
                }
                used.insert(s[i]);
                swaps[s[i]] = t[i];
            }
        }
    }
    cout << swaps.size() << '\n';
    for (auto p : swaps) {
        cout << p.first << ' ' << p.second << '\n';
    }
}
