import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        sc.close();

        int pos = 0;
        boolean canReach = true;
        for (int i = 0; i < n; i++) {
            if (pos + d[i] >= n - 1) {
                pos = n - 1;
                break;
            } else if (pos + d[i] >= i) {
                pos += d[i];
            } else {
                canReach = false;
                break;
            }
        }
        if (canReach) {
            pos = n - 1;
            for (int i = n - 2; i >= 0; i--) {
                if (pos - d[i] >= i) {
                    pos -= d[i];
                } else {
                    canReach = false;
                    break;
                }
            }
        }
        System.out.println(canReach ? "yes" : "no");
    }
}
