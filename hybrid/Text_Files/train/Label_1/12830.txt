#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> a[i][j]; }
  }
  int ans = 0, replace = 0;
  vector<bool> col(m, 0);
  for (int j = 0; j < m; j++) {
    int top = n + 1;
    for (int i = 0; i < n; i++) {
      if (a[i][j] == 1) {
        top = min(top, i);
      }
    }
    if (top < n + 1) {
      ans += min(k, n - top + 1);
      if (a[top][j] == 1) {
        replace++;
        col[j] = 1;
      }
    }
  }
  int add = 0;
  for (int j = 0; j < m; j++) {
    if (col[j] == 0) {
      for (int i = 0; i < n; i++) {
        if (a[i][j] == 1) { add++; }
      }
    }
  }
  ans += add;
  cout << ans << " " << replace << endl;
  return 0;
}
