import java.util.Arrays;
import java.util.Scanner;

class Main {
  static int minModular(int[] arr, int n, int k) {
    Arrays.sort(arr);
    int m = arr[0];
    for (int i = 1; i < n; i++) {
      if (i <= k) {
        k -= i;
        m = gcd(m, arr[i]);
        if (m == 1) {
          return 1;
        }
      } else {
        m = gcd(m, arr[i] - arr[i - 1]);
        if (m == 1) {
          return 1;
        }
      }
    }
    return m;
  }

  static int gcd(int a, int b) {
    while (b > 0) {
      int temp = a;
      a = b;
      b = temp % b;
    }
    return a;
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int k = scanner.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = scanner.nextInt();
    }
    System.out.println(minModular(arr, n, k));
  }
}
