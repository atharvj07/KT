import sys

n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
sa = sum(a)
sb = sum(b)
a.sort()
b.sort(reverse=True)
v = abs(sa - sb)
if v == 0:
    k = 0
else:
    v -= min(a[0] - b[0], b[0] - a[-1])
    if v == 0:
        k = 1
    else:
        v -= min(a[1] - b[0], b[1] - a[-1])
        k = 2
print(v)
print(k)
for i in range(k):
    x, y = map(int, input().split())
    print(x, y)
