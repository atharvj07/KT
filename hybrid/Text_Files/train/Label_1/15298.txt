#include <iostream>
#include <algorithm>
using namespace std;
int a[200005], n, m, x; long long cur, ans;
int main() {
    cin >> n >> m;
    for(int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    for(int i = n - 1; i >= 0; i--) {
        cur += a[i];
        x += 1;
        if(cur >= m) {
            cur -= m;
            x -= 1;
            ans += x;
            x = 0;
        }
    }
    if(m) ans = -1;
    cout << ans << endl;
}
