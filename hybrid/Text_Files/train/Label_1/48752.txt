import java.util.*;

class Main {
    private static int[] a, b;
    private static boolean valid;

    private static void dfs(int l, int r, int val) {
        if (valid || l > r) {
            return;
        }

        if (l == r) {
            if (val == 0)
                valid = true;
            return;
        }

        dfs(l + 1, r, val + a[l]);
        dfs(l + 1, r, val);
        dfs(l, r - 1, val + a[r]);
        dfs(l, r - 1, val);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        dfs(0, n - 1, 0);
        System.out.println(valid ? "YES" : "NO");
    }
}
