import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt(), y = sc.nextInt(), arr[] = new int[n], gcd = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            gcd = __gcd(gcd, arr[i]);
        }
        System.out.println((gcd == 1 ? 0 : (x + (y - x) * (n - 1))));
    }

    static int __gcd(int a, int b) {
        if (b == 0) return a;
        return __gcd(b, a % b);
    }
}
