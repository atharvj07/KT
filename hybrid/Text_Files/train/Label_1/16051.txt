#include <bits/stdc++.h>
using namespace std;
int n, a[205], ans;
bool f[205];
char s[205];
int main() {
  cin >> n >> s;
  for (int i = 1; i <= n; i++) { a[i] = s[n - i] >= 'a'; }
  for (int i = 1; i <= n; i++) {
    if (!a[i] && f[i]) {
      for (int j = i + 1; j <= n; j++) { f[j] = false; }
      continue;
    }
    f[i] = true;
    for (int j = i - 1; j; j--) { f[j] = f[j] && a[j]; }
    int tmp = 0;
    for (int j = i; j <= n; j++) { tmp += a[j] && f[j]; }
    ans = max(ans, tmp);
  }
  cout << ans << endl;
  return 0;
}
