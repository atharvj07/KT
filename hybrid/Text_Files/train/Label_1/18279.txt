#include <iostream>
#include <string>
using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  if (a.length() != b.length()) {
    cout << "No" << endl;
  } else {
    bool flag = true;
    for (int i = 0; i < a.length(); i++) {
      if (a[i] != b[i]) {
        if ((a[i] == 'a' || a[i] == 'e' || a[i] == 'i' || a[i] == 'o' || a[i] == 'u') &&
            (b[i] == 'a' || b[i] == 'e' || b[i] == 'i' || b[i] == 'o' || b[i] == 'u'))
          continue;
        else if (a[i] != b[i]) {
          flag = false;
          break;
        }
      }
    }
    flag ? cout << "Yes" << endl : cout << "No" << endl;
  }
  return 0;
}
