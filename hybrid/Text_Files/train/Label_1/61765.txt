for t in range(int(input())):
    n = int(input())
    L = []
    for _ in range(n):
        l, r = map(int, input().split())
        L.append((l, r))
    L.sort(key=lambda x: x[1])
    ans = [0] * n
    cur = 0
    for l, r in L:
        if cur >= l:
            cur += 1
        else:
            ans[L.index((l, r))] = cur + 1
            cur = cur + 1
    print(*ans)
