import heapq

n = int(input())
arr = []
for i in range(n):
    arr.append(list(map(int, input().split())))
arr.sort(key=lambda x: x[1])
ans = 0
q = []
for i in range(n):
    ans += arr[i][0] * 2
    heapq.heappush(q, arr[i][0])
    if len(q) >= arr[i][1]:
        ans -= 2 * heapq.heappop(q)
print(ans)
