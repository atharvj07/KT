from math import gcd

def lcm(x, y):
    return x * y // gcd(x, y)

n = int(input().strip())
a = list(map(int, input().strip().split()))

# Initialize minimum lcm to a high value and the pair indices
min_lcm = float('inf')
pair = (0, 0)

# Sort the array along with original indices
sorted_arr = sorted((val, idx) for idx, val in enumerate(a, start=1))

for i in range(1, n):
    # Calculate lcm for consecutive elements in sorted array, as they are most likely to have minimum lcm
    current_lcm = lcm(sorted_arr[i-1][0], sorted_arr[i][0])
    # Update min_lcm and pair if a new minimum is found
    if current_lcm < min_lcm:
        min_lcm = current_lcm
        pair = (sorted_arr[i-1][1], sorted_arr[i][1])

# Ensure the smaller index comes first
pair = tuple(sorted(pair))

print(pair[0], pair[1])
