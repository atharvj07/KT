import java.util.*;

public class Main {
    static int n, k;
    static int[] a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int maxCnt = 0;
        int[] ans = new int[n];
        for (int i = 0, j = 0, cnt = 0; i < n; i++) {
            cnt += 1 - a[i];
            while (cnt > k) {
                cnt -= 1 - a[j];
                j++;
            }
            if (i - j + 1 > maxCnt) {
                maxCnt = i - j + 1;
                for (int x = 0; x < n; x++) {
                    ans[x] = a[x];
                }
                for (int x = j; x <= i; x++) {
                    ans[x] = 1;
                }
            }
        }
        System.out.println(maxCnt);
        for (int x : ans) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
