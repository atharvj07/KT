import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int k = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int[] dp = new int[n + 1];
        int max = 0;
        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1];
            if (i > k) {
                dp[i] = Math.max(dp[i], dp[i - k - 1]);
            }
            if (i > 1 && a[i - 1] == a[i - 2]) {
                dp[i] = Math.max(dp[i], dp[i - 2] + 1);
            } else {
                dp[i] = Math.max(dp[i], 1);
            }
            max = Math.max(max, dp[i]);
        }
        System.out.println(max);
    }
}
