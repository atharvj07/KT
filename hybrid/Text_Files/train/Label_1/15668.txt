#include <iostream>
#include <vector>

using namespace std;

int main() {
    int uphillDays, downhillDays;
    cin >> uphillDays >> downhillDays;

    vector<int> heights;
    for (int i = 1; i <= uphillDays + 1; i++)
        heights.push_back(i);
    for (int i = downhillDays; i >= 1; i--)
        heights.push_back(i);

    for (int height : heights)
        cout << height << " ";
    cout << endl;

    return 0;
}
