#include<bits/stdc++.h>
using namespace std;
const long long mod=1e9+7;
int main()
{
    long long n,m,ans=1;
    cin>>n>>m;
    for(long long i=0; i<m*2; i++)
    {
        ans=(ans*(n-i))%mod;
    }
    for(long long i=1; i<=m; i++)
    {
        ans=(ans/i)%mod;
    }
    cout<<ans<<endl;
    return 0;
}
