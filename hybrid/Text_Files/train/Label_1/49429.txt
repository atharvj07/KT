import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int d = sc.nextInt();
        Map<Set<Integer>, Integer> map = new HashMap<>();
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            int k = sc.nextInt();
            Set<Integer> set = new HashSet<>();
            for (int j = 0; j < k; j++) {
                int h = 0;
                for (int l = 0; l < d; l++) {
                    h ^= sc.nextInt();
                }
                set.add(h);
            }
            if (!map.containsKey(set)) {
                map.put(set, map.size() + 1);
            }
            ans[i] = map.get(set);
        }
        for (int i = 0; i < m; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
