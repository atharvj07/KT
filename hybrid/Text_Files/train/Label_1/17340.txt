#include <bits/stdc++.h>
using namespace std;
int n, q, a[100005], b[100005];
int main() {
  cin >> n >> q;
  for (int i = 1; i <= n; i++) cin >> a[i];
  while (q--) {
    char c;
    int x;
    cin >> c >> x;
    for (int i = 1; i <= n; i++) b[i] = a[i];
    if (c == '>') {
      for (int i = 1; i <= n; i++) {
        if (a[i] > x) b[i] *= -1;
      }
    } else {
      for (int i = 1; i <= n; i++) {
        if (a[i] < x) b[i] *= -1;
      }
    }
    for (int i = 1; i <= n; i++) a[i] = b[i];
  }
  for (int i = 1; i <= n; i++) { cout << a[i] << ' '; }
}
