#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    int sum_a = 0, sum_b = 0;
    for (int i = 0; i < n; i++) { sum_a += a[i]; }
    for (int i = 0; i < n; i++) { sum_b += b[i]; }
    if (sum_a != sum_b) {
      cout << -1 << endl;
      continue;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] > b[i]) {
        cout << ans << endl;
        for (int j = 0; j < a[i] - b[i]; j++) { cout << i + 1 << " " << 1 << endl; }
        for (int j = 0; j < a[i] - b[i]; j++) { cout << n << " " << i + 1 << endl; }
        ans += a[i] - b[i];
      } else if (a[i] < b[i]) {
        cout << ans << endl;
        for (int j = 0; j < b[i] - a[i]; j++) { cout << n << " " << i + 1 << endl; }
        for (int j = 0; j < b[i] - a[i]; j++) { cout << i + 1 << " " << 1 << endl; }
        ans += b[i] - a[i];
      }
    }
  }
  return 0;
}
