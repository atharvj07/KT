#include <cstdio>
#include <vector>
using namespace std;

const int MAXN = 2e5 + 5;
const int MOD = 1e9 + 7;
const int SIEVE_SIZE = 1e5 + 5;

vector<int> primes;

void sieve() {
  vector<bool> is_prime(SIEVE_SIZE, true);
  is_prime[0] = is_prime[1] = false;
  for (int p = 2; p * p <= SIEVE_SIZE; p++) {
    if (is_prime[p]) {
      for (int i = p * p; i <= SIEVE_SIZE; i += p) {
        is_prime[i] = false;
      }
    }
  }
  primes.push_back(2);
  for (int p = 3; p < SIEVE_SIZE; p += 2) {
    if (is_prime[p]) primes.push_back(p);
  }
}

int get_num_div(int n) {
  int num_div = 1;
  for (int p : primes) {
    if (p * p > n) break;
    if (n % p == 0) {
      int cnt = 0;
      while (n % p == 0) {
        cnt++;
        n /= p;
      }
      num_div *= (cnt + 1);
    }
  }
  if (n > 1) num_div *= 2;
  return num_div;
}

int main() {
  sieve();
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, e;
    scanf("%d%d", &n, &e);
    vector<int> a(n);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    int ans = 0;
    for (int i = 0; i < n; i++) {
      long long prod = 1LL * a[i];
      int cnt = 1;
      while (i + e * cnt <= n) {
        prod *= 1LL * a[i + e * cnt];
        if (get_num_div(prod) == 2) ans++;
        cnt++;
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
