#include <bits/stdc++.h>
using namespace std;

#define ll long long

int a[500001];

int main() {
    int n;
    cin >> n;
    
    for (int i = 0; i < n; i++)
        cin >> a[i];
    
    int freq[7] = {0};
    for (int i = 0; i < n; i++)
        freq[a[i]]++;
    
    int rem = (freq[4] + freq[8] + freq[15] + freq[16] + freq[23] + freq[42]) % 6;
    
    int ans = 0;
    
    for (int i = 1; i <= 5; i++) {
        if (freq[i] < freq[6]) {
            rem -= freq[i];
            ans += freq[i];
            freq[6] -= freq[i];
            freq[i] = 0;
        }
        else if (freq[i] == freq[6]) {
            rem -= freq[i];
            ans += freq[i];
            freq[6] = 0;
            freq[i] = 0;
        }
    }
    
    if (rem <= freq[6]) {
        ans += rem;
        freq[6] -= rem;
    }
    else {
        ans += freq[6];
        rem -= freq[6];
        
        if (rem % 3 == 0) {
            ans += rem / 3;
            freq[6] = 0;
        }
        else if (rem % 3 == 1) {
            ans += (rem / 3) + 1;
            freq[6] = 1;
        }
        else if (rem % 3 == 2) {
            ans += (rem / 3) + 2;
            freq[6] = 2;
        }
    }
    
    if (freq[6] > 0)
        ans += freq[6];
    
    cout << ans << endl;
    
    return 0;
}
