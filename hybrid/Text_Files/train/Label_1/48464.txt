import java.util.*;
public class Main {
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            int a=sc.nextInt();
            int b=sc.nextInt();
            int k=sc.nextInt();
            if(k==0){System.out.println("NO");continue;}
            if(a==b){System.out.println("YES");continue;}
            if(a>b){System.out.println("NO");continue;}
            if(b%a==0){
                int x=b/a;
                if(k>=x && (k-x)%2==0){System.out.println("YES");}
                else {System.out.println("NO");}
            }
            else{System.out.println("NO");}
        }
    }
}
