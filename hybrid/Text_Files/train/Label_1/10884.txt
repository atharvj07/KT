#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

vector<long long> a, b;

int cal(double d) {
    vector<long long> c(a.size());
    int zero = 0;
    for (int i = 0; i < a.size(); i++) {
    	long long x = d * a[i] + b[i];
        c[i] = x;
        if (x == 0) zero++;
    }
    sort(c.begin(), c.end());
    for (int i = 0; i < c.size(); i++) {
        while (i + 1 < c.size() && c[i] == c[i + 1]) i++;
        if (c[i] > 0) break;
        zero++;
    }
    return zero;
}

int main() {
    int n;
    scanf("%d", &n);
    a.resize(n);
    b.resize(n);
    for (int i = 0; i < n; i++) scanf("%lld", &a[i]);
    for (int i = 0; i < n; i++) scanf("%lld", &b[i]);
    double l = -1e9, r = 1e9;
    int ans = 0;
    for (int i = 0; i < 100; i++) {
        double m1 = (l + r) / 3.0;
        double m2 = (l + r) / 3.0 * 2;
        int z1 = cal(m1);
        int z2 = cal(m2);
        if (z1 > z2) {
            r = m2;
        } else {
            l = m1;
        }
        ans = max(ans, z1);
        ans = max(ans, z2);
    }
    printf("%d\n", ans);
    return 0;
}
