#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;

#define F0R(i, a) for (int i = 0; i < (a); i++)
#define FOR(i, a, b) for (int i = (a); i < (b); i++)
#define F0Rd(i, a) for (int i = (a) - 1; i >= 0; i--)
#define FORd(i, a, b) for (int i = (b) - 1; i >= (a); i--)

#define pb push_back
#define mp make_pair
#define sz(x) (int)(x).size()
#define all(x) begin(x), end(x)
#define rall(x) (x).rbegin(), (x).rend()
#define ins insert

int n, t, pos = 0;
string s;

int get() {
    if (pos) return 1 << (s[pos - 1] - 'a');
    int ans = 0;
    F0R(i, n) {
        int j = n - i - 1;
        if (s[j] == '+') {
            if (i == 0) {
                if (!j) return 1 << (s[j] - 'a');
                else return -1;
            }
            auto x = get();
            if (x == -1) return -1;
            ans -= x;
            pos = j + 1;
        } else if (s[j] == '-') {
            if (i == 0) {
                if (j) return -1;
                else return 1 << (s[j] - 'a');
            }
            auto x = get();
            if (x == -1) return -1;
            ans += x;
            pos = j + 1;
        } else
            pos++;
    }
    if (pos != sz(s)) return -1;
    return ans;
}
int main() { cin >> n >> t >> s, cout << ((get() == t) ? "Yes" : "No") << '\n'; }
