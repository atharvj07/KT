#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long int a, b;
    cin >> a >> b;
    long long int ans = 0;
    for (long long int i = 1; i <= a; i++) {
      for (long long int j = 1; j <= b; j++) {
        string s1 = to_string(i);
        string s2 = to_string(j);
        string s3 = s1 + s2;
        long long int p = i * j + i + j;
        long long int q = stoi(s3);
        if (p == q) ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
