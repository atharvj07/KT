t = int(input())
for _ in range(t):
    n, x, m = map(int, input().split())
    l_bound, r_bound = x, x

    for _ in range(m):
        l, r = map(int, input().split())
        if r < l_bound or l > r_bound:
            continue
        l_bound = min(l_bound, l)
        r_bound = max(r_bound, r)

    print(r_bound - l_bound + 1)
