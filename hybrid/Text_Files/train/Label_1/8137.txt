#include <iostream>
#include <string>
#include <climits>
using namespace std;
string s, ans;
int main() {
    cin >> s;
    ans = "a";
    for (char c : s) {
        if (c <= ans.back() || c == 'z') continue;
        ans.push_back(c);
    }
    int len = ans.length();
    if (len == 26)
        cout << ans;
    else
        cout << -1;
    cout << endl;
}
