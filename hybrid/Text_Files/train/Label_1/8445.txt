#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<ll,ll> ii;
#define MAXN 500000 //5e5

ll n,c;
ll a[MAXN];

template<class T>
struct FenwickTree {
    T tree[MAXN+1];

	FenwickTree() { memset(tree,0,sizeof tree); }

    void update(ll idx, T val) {
        while(idx <= n){
            tree[idx] = max(tree[idx],val);
            idx += (idx & -idx);
        }
    }
    T query(ll idx) {
        T sum = 0;
        while(idx>0){
            sum = max(sum,tree[idx]);
            idx -= (idx & -idx);
        }
        return sum;
    }
    T query(ll l, ll r){ return query(r)-query(l-1); }
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL); cout.tie(NULL);

    FenwickTree<ll> ft;
    cin >> n >> c;
    for(int i=0;i<n;++i){
        cin >> a[i];
        ft.update(i+1,(c-a[i]));
    }

    ll ans = LLONG_MIN;
    for(int i=0;i<n;++i){
        ans = max(ans,c - a[i] + ft.query(i+1,n));
    }

    cout << ans << "\n";

    return 0;
}
