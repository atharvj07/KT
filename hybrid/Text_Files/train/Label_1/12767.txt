#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (v[i] != 0 && v[i] == v[i - 1]) {
      ans++;
    }
  }
  cout << ans << endl;
}
