import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] blowers = new int[m];
        for (int i = 0; i < m; i++) {
            blowers[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        int[] queries = new int[q];
        for (int i = 0; i < q; i++) {
            queries[i] = scanner.nextInt();
        }
        for (int query : queries) {
            int minLength = -1;
            int losses = 0;
            for (int arrowLength = 1; arrowLength <= n; arrowLength++) {
                losses = 0;
                for (int x = 1; x <= n; x++) {
                    boolean loss = true;
                    for (int blower : blowers) {
                        if (x >= blower && x < blower + arrowLength) {
                            loss = false;
                        }
                    }
                    if (loss) {
                        losses++;
                    }
                }
                if (losses <= query) {
                    minLength = arrowLength;
                    break;
                }
            }
            System.out.println(minLength);
        }
    }
}
