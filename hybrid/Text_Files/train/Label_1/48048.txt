import java.util.Scanner;
import java.util.Arrays;

public class Main{
    public static void main(String args[]){
        Scanner input = new Scanner(System.in);
        int x = input.nextInt();
        int n = input.nextInt();
        int []a = new int[n];
        for(int i = 0; i < n ; i++)
            a[i] = input.nextInt();
        Arrays.sort(a);
        int l = 0, r = n - 1;
        int ans = 101;
        while(l <= r){
            int mid = (l + r) / 2;
            if(a[mid] == x){
                ans = 0;
                break;
            }
            if(a[mid] < x){
                l = mid + 1;
                if(x - a[mid] < ans)
                    ans = x - a[mid];
            }
            else{
                r = mid - 1;
                if(a[mid] - x < ans)
                    ans = a[mid] - x;
            }
        }
        System.out.println(ans);
    }
}
