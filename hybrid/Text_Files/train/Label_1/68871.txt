import sys, operator as op
n=int(sys.stdin.readline())
a=[int(x) for x in sys.stdin.readline().split()]
a.sort()
if n==1:
    print(1);print(1,1);sys.exit(0)
c=1
last=a[0];
for i in range(1,n):
    if a[i]==last:
        c+=1
    else:
        c=1
    if c>=2:
        print(n);print(n,1);print(*a,sep='\n');sys.exit(0)
    last=a[i]
b=a[::-1];c=1;last=b[0]
for i in range(1,n):
    if b[i]==last:
        c+=1
    else:
        c=1
    if c>=2:
        print(n);print(1,n);print(*b,sep='\n');sys.exit(0)
    last=b[i]
ans=[[0]]
for i in range(n):
    for j in range(n):
        if a[i]==b[j] and i!=j:
            ans=[[a[i]]]
            break
l=len(ans[0])
for i in range(n):
    for j in range(n):
        if ans[0][0]==b[j] and i+1!=j:
            ans[0].append(b[j])
            if len(ans[0])>l:
                l=len(ans[0])
            break
print(l*(l+1));print(l,l+1)
for i in ans:
    for j in i:
        print(j,end=' ')
    print()
