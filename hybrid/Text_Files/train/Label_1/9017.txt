#include <iostream>
using namespace std;
typedef long long ll;
int main() {
  int q;
  cin >> q;
  while (q--) {
    ll k, n, a, b;
    cin >> k >> n >> a >> b;
    if ((n - 1) * b + 0 > k)
      cout << -1 << endl;
    else if (n - 1 > k / min(a, b))
      cout << -1 << endl;
    else
      cout << k / a << endl;
  }
}
