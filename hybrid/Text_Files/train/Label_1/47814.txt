import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q --> 0) {
            int xp0 = sc.nextInt(), yp0 = sc.nextInt(), xp1 = sc.nextInt(), yp1 = sc.nextInt(),
                xp2 = sc.nextInt(), yp2 = sc.nextInt(), xp3 = sc.nextInt(), yp3 = sc.nextInt();
            System.out.println(isIntersect(xp0, yp0, xp1, yp1, xp2, yp2, xp3, yp3) ? 1 : 0);
        }
    }

    private static boolean isIntersect(int xp0, int yp0, int xp1, int yp1, int xp2, int yp2, int xp3, int yp3) {
        int dx = xp3 - xp2, dy = yp3 - yp2, d1x = xp1 - xp0, d1y = yp1 - yp0;
        int d2x = xp0 - xp2, d2y = yp0 - yp2;
        float n1 = d1x * dy - d1y * dx, n2 = d2x * dy - d2y * dx;
        return n1 * n2 < 0 && (Math.abs(n1) > Math.abs(dx * d2y - dy * d2x) || Math.abs(n2) > Math.abs(dx * d1y - dy * d1x));
    }
}
