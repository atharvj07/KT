import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int,input().split())
    x = list(map(int,input().split()))
    y = list(map(int,input().split()))
    x.sort()
    y.sort()
    l = len(x)
    mx = 0
    for i in range(l):
        pos = x[i]
        cnt = 0
        for j in range(i,l):
            if x[j]-pos<=k:
                cnt+=1
            else: break
        for j in range(i-1,-1,-1):
            if pos-x[j]<=k:
                cnt+=1
            else: break
        mx = max(mx,cnt)
        pos = y[i]
        cnt = 0
        for j in range(i,l):
            if y[j]-pos<=k:
                cnt+=1
            else: break
        for j in range(i-1,-1,-1):
            if pos-y[j]<=k:
                cnt+=1
            else: break
        mx = max(mx,cnt)
    print(mx)
