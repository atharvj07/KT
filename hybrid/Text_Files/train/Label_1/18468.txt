#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n, m, ans = 0;
    cin >> n >> m;
    unordered_map<int, int> cnt;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        cnt[a] = -b;
    }
    priority_queue<int> pq;
    for (auto p : cnt) pq.push(p.second);
    for (int i = 0; i < n; i++) {
        if (pq.empty()) break;
        int now = -pq.top();
        pq.pop();
        ans += now;
    }
    cout << ans << '\n';
    return 0;
}
