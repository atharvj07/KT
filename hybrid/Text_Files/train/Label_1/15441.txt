#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (long long i = 0; i < n; i++) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long ans = 0;
  for (long long i = 0; i < k; i++) ans += a[i];
  cout << ans << endl;
  return 0;
}
