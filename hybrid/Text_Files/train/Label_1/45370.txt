import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        int ans = -1;
        for (int j = 0; j < m; j++) {
            int cnt0 = 0;
            for (int i = 0; i < n; i++) {
                cnt0 += a[i][j] == 0 ? 1 : 0;
            }
            if (cnt0 == 0 || cnt0 == n) {
                ans = 0;
                break;
            }
            int cnt1 = n - cnt0;
            if (cnt0 != cnt1) {
                ans = -1;
                break;
            }
            int cur = 0;
            for (int i = 0; i < n; i++) {
                if (a[i][j] == 0) {
                    cur++;
                }
            }
            if (ans == -1) {
                ans = cur;
            } else {
                ans = Math.min(ans, cur);
            }
        }
        System.out.println(ans);
    }
}
