import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static long mod = (long) (1e9 + 7);
    static int n, k, c, t;
    static List<Shuffle> shuffles;

    public static void main(String[] args) {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String[] line = br.readLine().split(" ");
            n = Integer.parseInt(line[0]);
            k = Integer.parseInt(line[1]);
            c = Integer.parseInt(line[2]);
            t = Integer.parseInt(line[3]);
            shuffles = new ArrayList<>(k);

            for (int i = 0; i < k; i++) {
                line = br.readLine().split(" ");
                int a_i = Integer.parseInt(line[0]);
                int b_i = Integer.parseInt(line[1]);
                int t_i = Integer.parseInt(line[2]);

                List<Integer> cards = new ArrayList<>();
                for (int j = a_i; j < a_i + b_i; j++) {
                    cards.add(j);
                }
                shuffles.add(new Shuffle(cards, t_i));
            }

            long[][] dp = new long[t + 1][n + 1];
            for (int i = 0; i <= t; i++) {
                Arrays.fill(dp[i], -1);
            }

            System.out.println(solve(c, t, dp));
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    static long solve(int pos, int remainingTime, long[][] dp) {
        if (pos == c) {
            return 1;
        }
        if (remainingTime <= 0) {
            return 0;
        }
        if (dp[remainingTime][pos] != -1) {
            return dp[remainingTime][pos];
        }

        long ans = 0;
        for (Shuffle shuffle : shuffles) {
            if (shuffle.time > remainingTime) {
                continue;
            }

            List<Integer> newPosList = new ArrayList<>();
            for (Integer card : shuffle.cards) {
                newPosList.add(card);
            }
            newPosList.remove(pos);

            ans = (ans + solve(newPosList.get(0), remainingTime - shuffle.time, dp)) % mod;
        }

        dp[remainingTime][pos] = ans;
        return ans;
    }

    static class Shuffle {
        List<Integer> cards;
        int time;

        public Shuffle(List<Integer> cards, int time) {
            this.cards = cards;
            this.time = time;
        }
    }
}
