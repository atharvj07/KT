from heapq import heappop, heappush

def solve():
    n, k = map(int, input().split())
    matrix = [input().strip() for _ in range(n)]

    dp = [[float('inf')] * n for _ in range(n)]
    dp[0][0] = 0 if matrix[0][0] == 'a' else 1
    pq = [(dp[0][0], 0, 0)]

    while pq:
        changes, x, y = heappop(pq)
        if x == n - 1 and y == n - 1:
            break
        for dx, dy in [(1, 0), (0, 1)]:
            nx, ny = x + dx, y + dy
            if 0 <= nx < n and 0 <= ny < n:
                n_changes = changes + (matrix[nx][ny] != 'a')
                if n_changes < dp[nx][ny]:
                    dp[nx][ny] = n_changes
                    heappush(pq, (n_changes, nx, ny))

    max_dist = max(x + y for x in range(n) for y in range(n) if dp[x][y] <= k)
    result = ['a'] * (max_dist + 1)

    for dist in range(max_dist, 2 * n - 2):
        smallest = 'z'
        for x in range(n):
            for y in range(n):
                if dp[x][y] <= k and x + y == dist:
                    if x + 1 < n:
                        smallest = min(smallest, matrix[x + 1][y])
                    if y + 1 < n:
                        smallest = min(smallest, matrix[x][y + 1])
        result.append(smallest)
        k -= 1

    return ''.join(result)

print(solve())
