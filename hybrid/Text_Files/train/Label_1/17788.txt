#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5+5;

int n;
vector<int> adj[MAXN];
int subtree[MAXN], path[MAXN], ans[2];

void dfs(int u, int p, int dep) {
    subtree[u] = 1;
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs(v, u, dep + 1);
        subtree[u] += subtree[v];
        if (subtree[v] > subtree[path[dep]]) {
            path[dep] = v;
        }
    }
}

void dfs2(int u, int p, int len, int mx) {
    if (len > ans[0]) {
        ans[0] = len;
        ans[1] = mx;
    }
    for (int v : adj[u]) {
        if (v == p) continue;
        if (v == path[len]) {
            dfs2(v, u, len + 1, max(mx, subtree[v]));
        } else {
            dfs2(v, u, len, mx);
        }
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1, 0);
    dfs2(1, 1, 0, 0);
    cout << ans[0] << " " << ans[1] << endl;
    return 0;
}
