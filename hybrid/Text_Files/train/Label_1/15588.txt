#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int MAXN = 1e5 + 10;
const ll INF = 1e18;

struct train {
    int h, m;
    train() {}
    train(int _h, int _m) : h(_h), m(_m) {}
    bool operator<(const train &o) const { return h < o.h || (h == o.h && m < o.m); }
};

pair<int, int> solve(int n, int h, int m, int k, vector<train> &t) {
    if (k >= m) return {t.size(), 0};
    int ans = n;
    int opt = 0;
    for (int i = 0; i < m / 2; i++) {
        int bad = 0;
        for (auto &tr : t) {
            int needLeft = tr.h * m + tr.m - m / 2 + i + k;
            needLeft = min(needLeft, h * m - 1);
            bad += (needLeft < 0);
        }
        if (bad < ans) {
            ans = bad;
            opt = i;
        }
    }
    return {ans, opt};
}

int main() {
    int n, h, m, k;
    cin >> n >> h >> m >> k;
    vector<train> t(n);
    for (auto &tr : t) { cin >> tr.h >> tr.m; }
    sort(t.begin(), t.end());
    auto res = solve(n, h, m, k, t);
    cout << res.first << " " << res.second << "\n";
    return 0;
}
