import java.util.Scanner;

public class Main {
    private static final long mod = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long x = in.nextLong();
        int[] d = new int[n];
        long[] pow = new long[n + 1];
        for (int i = 0; i < n; i++) {
            d[i] = in.nextInt();
            pow[i + 1] = pow[i] * n % mod;
        }
        long ans = 1;
        for (int i = 0; i <= x; i++) {
            long sum = 0;
            for (int j = 0; j < n; j++) {
                if (i >= d[j]) {
                    sum = (sum + pow[i - d[j]]) % mod;
                }
            }
            ans = (ans + sum) % mod;
        }
        System.out.println(ans);
    }
}
