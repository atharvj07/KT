from sys import stdin

n, m = map(int, stdin.readline().split())

g = [[] for i in range(n + 1)]
for i in range(m):
    u, v = map(int, stdin.readline().split())
    g[u].append(v)
    g[v].append(u)

vis = [False] * (n + 1)
ans = 0

def dfs(node, par, t):
    vis[node] = True
    for u in g[node]:
        if u == par:
            continue
        if vis[u] == False:
            dfs(u, node, t)
        else:
            if t == 0:
                ans += 1
            t ^= 1

for i in range(1, n + 1):
    if vis[i] == False:
        dfs(i, i, 0)

print(ans)
