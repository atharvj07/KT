import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] t = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            t[i] = in.nextInt();
        }
        int m = in.nextInt();
        for (int i = 0; i < m; i++) {
            int p = in.nextInt();
            int x = in.nextInt();
            t[p] = x;
        }
        long sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += t[i];
        }
        System.out.println(sum);
    }
}
