#include <iostream>
#include <algorithm>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    long long arr[n], sum = 0;
    for (int i = 0; i < n; i++) {
      std::cin >> arr[i];
      sum += arr[i];
    }
    std::sort(arr, arr + n);
    if (arr[n - 1] > 0 || (arr[n - 1] == 0 && arr[n - 2] > 0))
      std::cout << "YES" << std::endl;
    else
      std::cout << "NO" << std::endl;
  }
}
