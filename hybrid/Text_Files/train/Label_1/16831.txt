#include <iostream>
#include <vector>

using namespace std;
using ll = long long;

int main() {
  int t;
  cin >> t;
  while (t--) {
    ll a, b, c, d;
    cin >> a >> b >> c >> d;
    ll x = a + (b - a) / 2, y = b + (c - b) / 2, z = c + (d - c) / 2;
    cout << x << " " << y << " " << z << endl;
  }
  return 0;
}
