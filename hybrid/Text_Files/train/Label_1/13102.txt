#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e3 + 5;
int n, m, a[MAXN], sum, avg;
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        sum = 0;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            sum += a[i];
        }
        avg = sum / n;
        sum = sum - avg;
        a[1] = m;
        sum = sum + m - avg;
        cout << a[1] << endl;
    }
    return 0;
}
