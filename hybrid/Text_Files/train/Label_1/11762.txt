#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string a;
    cin >> a;
    int A = 0, C = 0, G = 0, T = 0;
    for(int i = 0; i < n; i++) {
        if(a[i] == 'A') A++;
        else if(a[i] == 'C') C++;
        else if(a[i] == 'G') G++;
        else if(a[i] == 'T') T++;
    }
    if(A == C && C == G && G == T && A == T) {
        cout << a << endl;
        return 0;
    }
    int m = max(A, max(C, max(G, T)));
    a.clear();
    for(int i = 0; i < n; i++) {
        if(a[i] != '?') a.push_back(a[i]);
    }
    int l = a.size();
    if(l + 3 > n || 1 + l < n) cout << "===" << endl;
    else {
        if(A < m) for(int i = 0; i < m - A; i++) a.push_back('A');
        else if(C < m) for(int i = 0; i < m - C; i++) a.push_back('C');
        else if(G < m) for(int i = 0; i < m - G; i++) a.push_back('G');
        else if(T < m) for(int i = 0; i < m - T; i++) a.push_back('T');
        cout << a << endl;
    }
    return 0;
}
