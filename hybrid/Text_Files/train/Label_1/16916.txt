#include <iostream>
#include <string>
#include <vector>

int main() {
  int n, a, b;
  std::string s;
  std::cin >> n >> a >> b >> s;
  std::vector<int> dp(n + 1);
  dp[0] = 0;
  for (int i = 1; i <= n; ++i) {
    dp[i] = dp[i - 1] + a;
    for (int j = 0; j < i; ++j) {
      if (s.substr(j, i - j) == s.substr(0, i - j)) {
        dp[i] = std::min(dp[i], dp[j] + b);
      }
    }
  }
  std::cout << dp[n] << '\n';
  return 0;
}
