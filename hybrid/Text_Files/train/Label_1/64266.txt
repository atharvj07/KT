n, k = map(int, input().split())

if k * 2 > n:
    print(-1)
else:
    ans = [0] * n
    num = 1
    for i in range(1, n + 1, 2):
        if num > k:
            num = 1
        ans[i - 1] = num
        if i + 1 <= n:
            ans[i] = num
        num += 1
    for a in ans:
        print(a, end=" ")
