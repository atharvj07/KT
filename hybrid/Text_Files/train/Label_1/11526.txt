#include<bits/stdc++.h>
using namespace std;
int main()
{
    double R, x1, y1, x2, y2;
    cin >> R >> x1 >> y1 >> x2 >> y2;
    double theta, dx, dy;
    if(x2 - x1 == 0)
    {
        theta = 1.5707963267948966;
        dx = x2 - R;
        dy = y1;
    }
    if(y2 - y1 == 0)
    {
        theta = 0;
        dx = x1;
        dy = y2;
    }
    if((x2-x1) != 0 && (y2-y1) != 0)
    {
        dx = x2 - R * (x2 - x1) / sqrt((x2-x1)*(x2-x1) + (y2-y1)*(y2-y1));
        dy = y2 - R * (y2 - y1) / sqrt((x2-x1)*(x2-x1) + (y2-y1)*(y2-y1));
        theta = atan((y2-dy)/(x2-dx));
    }
    cout << dx << " " << dy << " " << R << endl;
}
