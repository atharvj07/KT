#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

const int N=1e5+10;

vector<pair<int,int>> g[N];

int n,m;

int main() {
  cin>>n>>m;
  rep(i,m) {
    int a,b,w; cin>>a>>b>>w;
    g[a].emplace_back(b,w);
    g[b].emplace_back(a,w);
  }
  
  vector<int> vis(n+1);
  int cnt=0;
  
  function<void(int,int)> dfs=[&](int u,int p) {
    if(vis[u]) {
      cnt++;
      return;
    }
    vis[u]=1;
    for(auto& e:g[u]) {
      if(e.first==p) continue;
      dfs(e.first,u);
    }
  };
  
  vector<int> ans;
  
  rep(i,n+1) {
    if(vis[i]) continue;
    dfs(i,-1);
    if(cnt) {
      ans.push_back(i);
      cnt=0;
    }
  }
  
  cout<<ans.size()<<" "<<ans.size()<<endl;
  for(int x:ans) cout<<x<<" ";
  cout<<endl;
  
  return 0;
}
