import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            long l = sc.nextLong();
            long r = sc.nextLong();
            long d = sc.nextLong();
            if (l % d == 0) {
                l += d;
            }
            if (l > r) {
                System.out.println(l);
            } else {
                System.out.println((r / d + 1) * d);
            }
        }
    }
}
