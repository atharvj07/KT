import sys

mod = 998244353

def exgcd(a, b):
    if b == 0:
        return (1, 0)
    r, s = exgcd(b, a % b)
    return (s, r - (a // b) * s)

def mod_inv(a):
    r, s = exgcd(a, mod)
    return (r + mod) % mod

n, m = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
w = list(map(int, sys.stdin.readline().split()))
p, q = [w[i] for i in range(n)], [w[i] for i in range(n)]
for i in range(n):
    if a[i] == 1:
        q[i] += m
for i in range(n):
    inv_p = mod_inv(p[i])
    r = (q[i] * inv_p) % mod
    print(r)
