#include <iostream>
#include <vector>
using namespace std;

const int mod = 1e9 + 7;

int dp[505][505][505];

int solve(int n, int m, int b, vector<int> &a) {
    dp[n][0][0] = 1;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j <= m; j++) {
            for (int k = 0; k <= b; k++) {
                dp[i][j][k] = (dp[i][j][k] + 1LL * dp[i + 1][j][k] + (j + a[i] <= b) * dp[i + 1][j + a[i]][k]) % mod;
            }
        }
    }
    return dp[0][m][b];
}

int main() {
    int n, m, b, mod;
    cin >> n >> m >> b >> mod;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    cout << solve(n, m, b, a) << endl;
    return 0;
}
