import java.util.*;
import java.io.*;

public class Main {

    static void solve(int n, int m) throws Exception {
        ArrayList[] al = new ArrayList[m + 1];
        for (int i = 0; i <= m; i++) {
            al[i] = new ArrayList<>();
        }
        int[] arr = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            StringTokenizer st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            arr[i] = x;
            for (int j = 0; j < x; j++) {
                int y = Integer.parseInt(st.nextToken());
                al[y].add(i);
            }
        }
        boolean[] vis = new boolean[n + 1];
        for (int i = 1; i <= m; i++) {
            if (al[i].size() == 0) {
                System.out.println("NO");
                return;
            }
        }
        for (int i = 1; i <= m; i++) {
            if (vis[i]) {
                continue;
            }
            vis[i] = true;
            Queue<Integer> q = new LinkedList<>();
            q.add(i);
            while (q.size() != 0) {
                int cur = q.remove();
                for (int j = 0; j < al[cur].size(); j++) {
                    int nxt = (int) al[cur].get(j);
                    if (vis[nxt]) {
                        continue;
                    }
                    vis[nxt] = true;
                    q.add(nxt);
                }
            }
            boolean ok = true;
            for (int j = 1; j <= m; j++) {
                if (!vis[j]) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                System.out.println("YES");
                return;
            }
        }
        System.out.println("NO");
    }

    public static void main(String[] args) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());
            solve(n, m);
        } catch (Exception e) {
        }
    }
}
