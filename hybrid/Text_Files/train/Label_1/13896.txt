#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
#define ll long long
const int maxn=(1<<20)+1;
ll dp[maxn];
int cnt[65];
void pre(){
    dp[0]=1;
    int sum=1;
    for(int i=1;i<maxn;++i){
        for(int j=1;j<=64;++j){
            if((1<<j)&i){
                dp[i]|=dp[i^(1<<j)]<<j;
                sum+=dp[i]>>j;
            }
        }
    }
    for(int i=1;i<=64;++i){
        for(int j=0;j<maxn;++j){
            if((dp[j]>>i)&1){
                ++cnt[i];
            }
        }
    }
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    pre();
    ll m,k;
    cin>>m>>k;
    if(k==1&&m==1){
        cout<<1<<endl;
        return 0;
    }
    if(m>cnt[k]){
        cout<<-1<<endl;
        return 0;
    }
    ll sum=0;
    for(int i=1;;++i){
        sum+=dp[i]>>k;
        if(sum>=m){
            cout<<i-1<<endl;
            return 0;
        }
    }
    return 0;
}
