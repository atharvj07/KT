from math import gcd

n = int(input())
a = list(map(int, input().split()))
ans = 0
for i in range(1, n):
    if gcd(a[i], a[i - 1]) == 1:
        continue
    ans += 1
    for j in range(1, 1000000001):
        if gcd(a[i], j) == 1 and gcd(a[i - 1], j) == 1:
            a.insert(i, j)
            break
print(ans)
print(*a)
