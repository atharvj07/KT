#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 5e5 + 5;
map<int, ll> frq;
int a[N];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        frq[a[i]]++;
    }
    priority_queue<pair<ll, int>> pq;
    for (auto x : frq) { pq.push({x.second, x.first}); }
    ll sum = 0;
    while (!pq.empty()) {
        auto x = pq.top();
        pq.pop();
        sum += x.first;
        if (sum & 1) {
            pq.push({x.first - 1, x.second});
        }
    }
    cout << sum << endl;
}
