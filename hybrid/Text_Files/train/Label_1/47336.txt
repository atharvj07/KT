import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        StringBuilder sb = new StringBuilder(s.nextLine());
        int n = Integer.parseInt(sb.substring(sb.lastIndexOf("=") + 1).trim());
        s.close();

        boolean[] used = new boolean[n + 1];
        int prev = -1, op = 0;
        for (int i = 0; i < sb.length(); i++) {
            char c = sb.charAt(i);
            if (c == '?') {
                if (prev == -1) prev = 0;
                int num;
                switch (op) {
                    case 0:
                        num = nextNumber(used, prev, n);
                        if (num == -1) {
                            System.out.println("Impossible");
                            return;
                        }
                        sb.setCharAt(i, (char) (num + '0'));
                        prev = num;
                        break;
                    case 1:
                        num = prev + nextNumber(used, 1, n - prev);
                        if (num == -1) {
                            System.out.println("Impossible");
                            return;
                        }
                        sb.replace(i - 1, i + 1, String.valueOf(num));
                        prev = num;
                        break;
                    case 2:
                        num = prev - nextNumber(used, 1, prev - 1);
                        if (num == -1) {
                            System.out.println("Impossible");
                            return;
                        }
                        sb.replace(i - 1, i + 1, String.valueOf(num));
                        prev = num;
                        break;
                }
            } else if (c == '+') {
                op = 1;
            } else if (c == '-') {
                op = 2;
            } else if (c == '=') {
                if (prev == -1) {
                    System.out.println("Impossible");
                    return;
                }
                op = 0;
            }
        }

        System.out.println("Possible");
        System.out.println(sb);
    }

    private static int nextNumber(boolean[] used, int min, int max) {
        for (int i = min; i <= max; i++) {
            if (!used[i]) {
                used[i] = true;
                return i;
            }
        }
        return -1;
    }
}
