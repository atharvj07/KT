#include <bits/stdc++.h>

#define pb push_back
#define fi first
#define se second
#define mp make_pair

using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;

int n;
vector<int> a, d;

void add(int x) {
    if (d.empty()) {
        d.pb(1);
        a.pb(x);
        return;
    }
    if (d.back() == 1) {
        if (x <= a.back()) {
            d.pb(0);
            a.pb(x);
        }
        else {
            d.pb(1);
            a.pb(x);
        }
    }
    else {
        if (x >= a.back()) {
            d.pb(1);
            a.pb(x);
        }
        else {
            d.pb(0);
            a.pb(x);
        }
    }
}

void accept(int x) {
    if (a.empty() || x != a.back()) {
        return;
    }
    a.pop_back();
    d.pop_back();
}

int go(int id) {
    // if all directions restored
    if (id == n) {
        return 1;
    }
    // if can't restore current direction
    if (d.empty() || a[id] != a[id - 1]) {
        return 0;
    }
    return (go(id + 1) + go(id + 1) % mod) % mod;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 0; i < n; i++) {
        string s;
        int x;
        cin >> s >> x;
        if (s == "ADD") {
            add(x);
        }
        else {
            accept(x);
        }
    }
    cout << go(1) << '\n';
    return 0;
}
