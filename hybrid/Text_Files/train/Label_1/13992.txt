#include <bits/stdc++.h>
using namespace std;
int a[1001][1001], b[1001][1001], vis[1001][1001], dx[] = {1, 0, -1, 0}, dy[] = {0, 1, 0, -1}, n, m;
bool valid(int x, int y) {
    return 1 <= x && x <= n && 1 <= y && y <= m && vis[x][y] == 0 && (a[x][y] != 0 || b[x][y] == 0);
}
void dfs(int x, int y) {
    vis[x][y] = 1;
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (valid(nx, ny)) dfs(nx, ny);
    }
}
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < m; j++) {
            if (s[j] == '.')
                b[i][j + 1] = 1;
            else
                a[i][j + 1] = s[j] - '0';
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (vis[i][j] == 0 && a[i][j] == 0) {
                dfs(i, j);
                ans++;
            }
        }
    }
    bool ok = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] != 0 && vis[i][j] == 0) {
                ok = 0;
                break;
            }
        }
    }
    if (ok)
        cout << ans << endl;
    else
        cout << -1 << endl;
}
