import queue

n, k = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
ans = 0
depth = [-1] * (n+1)
bfs = queue.deque([(1, 0)])
while bfs:
    u, d = bfs.popleft()
    depth[u] = d
    ans += d
    for v in adj[u]:
        if depth[v] == -1:
            bfs.append((v, d+1))
print(2*ans)
