import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            Map<Integer, Integer> frequency = new HashMap<>();
            for (int j = 0; j < n; j++) {
                int a = scanner.nextInt();
                frequency.put(a, frequency.getOrDefault(a, 0) + 1);
            }
            int countEven = 0;
            int minOdd = Integer.MAX_VALUE;
            for (int frequencyValue : frequency.values()) {
                if (frequencyValue % 2 == 0) {
                    countEven++;
                } else {
                    minOdd = Math.min(minOdd, frequencyValue);
                }
            }
            if (countEven % 2 == 0) {
                // If even frequencies count is even, then all even frequencies can be paired.
                if (minOdd > 1) {
                    System.out.println(1);
                } else {
                    System.out.println(0);
                }
            } else {
                // If even frequencies count is odd, then one even frequency will remain unpaired.
                if (minOdd > 1) {
                    System.out.println(2);
                } else {
                    System.out.println(1);
                }
            }
        }
    }
}
