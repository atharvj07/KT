#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll mod = 1e9 + 7;
ll bp(ll x, ll y) {
    ll r = 1;
    while (y) {
        if (y % 2) r = r * x % mod;
        x = x * x % mod;
        y /= 2;
    }
    return r;
}

int main() {
    ll n, a, b;
    cin >> n >> a >> b;
    ll ans = bp(2, n) % mod * min(a, b) % mod;
    cout << ans << endl;
    return 0;
}
