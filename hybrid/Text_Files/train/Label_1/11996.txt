#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
vector<int>gr[200005];
bool vis[200005];
int main()
{
	int n;
	cin>>n;
	for(int i=1;i<=n;i++)
	{
		int x;
		cin>>x;
		gr[i].push_back(x);
	}
	long long ans=1;
	for(int i=1;i<=n;i++)
	{
		if(vis[i]) continue;
		int j=i;
		int cnt=0;
		while(!vis[j])
		{
			vis[j]=true;
			j=gr[j][0];
			cnt++;
			if(i==j)
			{
				ans=(ans*(cnt-1))%mod;
				ans=(ans*2)%mod;
				break;
			}
		}
	}
	cout<<ans%mod<<endl;
	return 0;
}
