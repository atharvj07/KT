def solve():
    s = input()
    t = input()
    swaps = []
    vis = [False for _ in range(26)]
    for i in range(len(s)):
        if s[i] != t[i]:
            if not vis[ord(s[i]) - ord('a')]:
                vis[ord(s[i]) - ord('a')] = True
                swaps.append([s[i], t[i]])
    print(len(swaps))
    for swap in swaps:
        print(*swap)

solve()
