import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        BigInteger a0 = BigInteger.valueOf(sc.nextInt());
        BigInteger b0 = BigInteger.valueOf(sc.nextInt());
        BigInteger gcd = a0;

        for (int i = 1; i < n; i++) {
            BigInteger a = BigInteger.valueOf(sc.nextInt());
            BigInteger b = BigInteger.valueOf(sc.nextInt());
            gcd = gcd.gcd(a0.subtract(a).mod(b));
        }

        if (gcd.compareTo(BigInteger.ONE) == 0) {
            System.out.println(-1);
        } else {
            System.out.println(a0.mod(gcd).mod(BigInteger.valueOf(1000000007)));
        }
    }
}
