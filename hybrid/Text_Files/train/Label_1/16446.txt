#include <iostream>
using namespace std;
const int m = 998244353;
int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  int n, ans = 0, a[200001], sum, inv;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    a[i] = (a[i] == -1 ? 0 : a[i]);
  }
  for (int i = n; i; i--) {
    sum = 0;
    for (int j = i + 1; j <= n; j++) { sum += (a[j] > a[i]); }
    inv = i - 1 - sum;
    ans = (ans + inv) % m;
    a[i] = n - i + 1;
  }
  cout << 1LL * ans * 499122177 % m << '\n';
}
