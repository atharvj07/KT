#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
int main() {
  int n, a, ans = 0, prv = 1, seq[400005], ind[400005];
  cin >> n;
  unordered_map<int, int> have;
  for (int i = 0; i < 2 * n; i++) cin >> a, have[a] = i + 1;
  for (int i = 1; i <= n; i++) {
    int x = have[i];
    if (x > prv) {
      if (ans == 0 || ind[ans] < x) {
        ind[++ans] = x;
        seq[ans] = i;
      }
    } else {
      if (ans == 0 || ind[ans] < x) {
        ind[++ans] = x;
        seq[ans] = i;
      } else {
        ind[ans] = x;
        seq[ans] = i;
      }
    }
    prv = x;
  }
  cout << ans << endl;
  for (int i = 1; i <= ans; i++) {
    cout << seq[i] << " ";
  }
}
