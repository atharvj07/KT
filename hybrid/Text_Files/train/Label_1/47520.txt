import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n + 2];
        int[] y = new int[n + 2];
        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        x[n + 1] = x[1];
        y[n + 1] = y[1];
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            if ((x[i] - x[i + 1]) * (y[i - 1] - y[i + 1]) > (x[i - 1] - x[i]) * (y[i] - y[i + 1])) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
