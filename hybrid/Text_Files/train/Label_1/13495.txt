#include<bits/stdc++.h>
using namespace std;
int a[100001], b[100001], c[100001];
int main() {
  int n, q, l, r, x, ans;
  string s;
  cin >> n >> q;
  cin >> s;
  for(int i = 0; i < n; i++) {
    if(s[i] == 'a') a[i + 1] = 1;
    if(s[i] == 'b') b[i + 1] = 1;
    if(s[i] == 'c') c[i + 1] = 1;
    a[i + 1] += a[i];
    b[i + 1] += b[i];
    c[i + 1] += c[i];
  }
  while(q--) {
    cin >> l >> s[l - 1];
    ans = 1e9;
    if(s[l - 1] == 'a') {
      ans = min(ans, b[l] + c[l]);
      ans = min(ans, c[l + 1] + b[n] - b[l]);
      ans = min(ans, a[l] + b[n] - b[l]);
    }
    if(s[l - 1] == 'b') {
      ans = min(ans, a[l] + c[l]);
      ans = min(ans, a[l] + c[l + 1] + b[n] - b[l]);
      ans = min(ans, b[l] + c[n] - c[l]);
    }
    if(s[l - 1] == 'c') {
      ans = min(ans, a[l] + b[l]);
      ans = min(ans, a[l] + b[l + 1] + c[n] - c[l]);
      ans = min(ans, c[l] + a[n] - a[l]);
    }
    cout << ans << "\n";
  }
  return 0;
}
