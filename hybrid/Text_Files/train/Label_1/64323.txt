import sys

x, y, x0, y0 = map(int, input().split())
s = input()
mx = x0-1
my = y0-1

ans = [0] * (len(s)+1)
for c in s:
    if c == 'L' and my > 0:
        my -= 1
    elif c == 'R' and my < y-1:
        my += 1
    elif c == 'U' and mx > 0:
        mx -= 1
    elif c == 'D' and mx < x-1:
        mx += 1
    else:
        ans[len(s)] += 1
        continue
    for k in range(len(ans)-1):
        if mx == k and my == len(s)-k-1:
            ans[k] += 1

print(*ans)
