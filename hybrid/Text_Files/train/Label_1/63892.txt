from sys import stdin

read = stdin.readline

def dfs1(node, par, col, ans):
    ans[node] = col
    for adj in g[node]:
        if adj != par:
            dfs1(adj, node, col, ans)

def dfs2(node, par, col, ans):
    for adj in g[node]:
        if adj != par:
            if col[node] != col[adj]:
                ans.add(col[adj])
            dfs2(adj, node, col, ans)

n = int(read())
g = [[] for i in range(n + 1)]
for i in range(2, n + 1):
    p = int(read())
    g[i].append(p)
    g[p].append(i)

ans = [1] * (n + 1)
dfs1(1, 0, 1, ans)
dfs2(1, 0, ans, set())
ans[0] = len(g) - 1
for i in ans: print(i)
