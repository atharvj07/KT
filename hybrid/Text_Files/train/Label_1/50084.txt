import java.util.Scanner;

public class MyonCoins {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextLong();
        }

        long max = 0;
        for (long a : A) {
            max = Math.max(max, a);
        }

        int count = 0;
        for (int i = 0; max > 0; i++) {
            long coin = (long) Math.pow(10, i);
            while (max >= coin) {
                max -= coin;
                count++;
            }
        }

        System.out.println(count);
    }
}
