import java.util.*;
public class Main {
  static ArrayList<Integer>[] adjList;
  static boolean[] visited;
  static boolean[] isRail;
  static int[] shortestTime;
  static int n;

  static int dfs(int time, int town) {
    if (shortestTime[town] < time) {
      return shortestTime[town];
    }
    shortestTime[town] = time;
    if (town == n) {
      return time;
    }
    int bestTime = Integer.MAX_VALUE;
    for (int nextTown : adjList[town]) {
      if (isRail[nextTown] == isRail[town]) {
        bestTime = Math.min(bestTime, dfs(time + 1, nextTown));
      } else {
        bestTime = Math.min(bestTime, dfs(time, nextTown));
      }
    }
    return bestTime;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    int m = sc.nextInt();
    adjList = new ArrayList[n + 1];
    for (int i = 1; i <= n; i++) {
      adjList[i] = new ArrayList<>();
    }

    isRail = new boolean[n + 1];

    for (int i = 0; i < m; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      adjList[a].add(b);
      adjList[b].add(a);
      isRail[a] = true;
      isRail[b] = true;
    }

    shortestTime = new int[n + 1];
    Arrays.fill(shortestTime, Integer.MAX_VALUE);

    int minTime = Math.min(dfs(0, 1), dfs(0, 1));
    if (minTime == Integer.MAX_VALUE) {
      System.out.println(-1);
    } else {
      System.out.println(minTime);
    }
  }
}
