#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  string s;
  while(t--) {
    cin >> s;
    int n = s.size(), ans = n;
    string st;
    for(int i = 0; i < n; i++) {
      if(s[i] == 'A') st += 'A';
      else if(st.size() && st.back() == 'A') st.pop_back();
      else st += 'B';
    }
    ans = st.size();
    cout << ans << '\n';
  }
}
