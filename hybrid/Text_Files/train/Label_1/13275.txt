#include <bits/stdc++.h>
using namespace std;

const int N = 110;
int a[N], ord[N];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    iota(ord, ord + n, 0);
    sort(ord, ord + n, [&](int x, int y) { return a[x] < a[y]; });
    for (int i = 0; i < n; ++i) cout << a[ord[i]] << ' ';
    cout << endl;
    return 0;
}
