import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur + a[i] <= k) {
                cur += a[i];
            } else {
                ans += (cur + k - 1) / k;
                cur = a[i] % k;
            }
        }
        if (cur > 0) {
            ans++;
        }
        System.out.println(ans);
    }
}
