import sys
input = sys.stdin.readline
def gcd(a: int, b: int) -> int:
    while b:
        a, b = b, a%b
    return a
def lcm(a: int, b: int) -> int:
    return (a*b)//gcd(a, b)
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    q = int(input())
    ans, d = [0 for _ in range(q)], [1 for _ in range(n+1)]
    for i in range(n):
        for j in range(n):
            if i==j:
                continue
            if gcd(a[i], a[j])**2 == lcm(a[i], a[j]):
                d[i+1]+=1
    for i in range(n):
        for j in range(n):
            if i==j:
                continue
            if gcd(a[i], a[j])**2 == lcm(a[i], a[j]):
                a[j]*=a[i]

    for i in range(q):
        w = int(input())
        if w%2==0:
            for j in range(n):
                if d[j+1]%2==0:
                    d[j+1]+=1
        ans[i] = max(d)
    print(*ans)
