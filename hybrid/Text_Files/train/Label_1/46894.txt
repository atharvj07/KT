import java.util.Scanner;

public class VasyaAndContest {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        String statement = scanner.next();
        int[] ambiguity = new int[n];

        for (int i = 0; i < n; i++) {
            ambiguity[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][5];

        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 5; j++) {
                dp[i][j] = Integer.MAX_VALUE;

                if (statement.charAt(i) == 'h' && j == 0) {
                    dp[i][j] = dp[i + 1][1];
                } else if (statement.charAt(i) == 'a' && j == 1) {
                    dp[i][j] = dp[i + 1][2];
                } else if (statement.charAt(i) == 'r' && j == 2) {
                    dp[i][j] = dp[i + 1][3];
                } else if (statement.charAt(i) == 'd' && j == 3) {
                    dp[i][j] = dp[i + 1][4];
                } else if (statement.charAt(i) == 'e' && j == 4) {
                    dp[i][j] = 0;
                } else {
                    dp[i][j] = Math.min(dp[i + 1][j], dp[i + 1][0] + ambiguity[i]);
                }
            }
        }

        System.out.println(dp[0][0]);
    }
}
