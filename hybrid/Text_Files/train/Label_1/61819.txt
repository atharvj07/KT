from math import *
n = int(input())
v = []
for i in range (n):
    v.append(input().split())
for i in range(n):
    v[i][0]=int(v[i][0]) 
    v[i][1]=int(v[i][1])
mnang=3.1415926535897932384626433832795
v.append(v[0])
for i in range(n):
    for j in range(i+1,n+1):
        ang=acos((v[i][0]*v[j][0]+v[i][1]*v[j][1])/(sqrt(v[i][0]**2+v[i][1]**2)*sqrt(v[j][0]**2+v[j][1]**2)))
        if ang<mnang:
            mnang=ang
            a=i+1
            b=j+1
print(a,b)
