#include <iostream>
#include <queue>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    int now = 1;
    for (int i = 0, j = 0; i < n || !pq.empty(); i++) {
      while (j < n && v[j].first <= now) {
        pq.push({v[j].second, j});
        j++;
      }
      if (pq.empty()) {
        now++;
        continue;
      }
      if (pq.top().first >= now) {
        cout << now << " ";
        pq.pop();
        now++;
      } else {
        cout << 0 << " ";
        pq.pop();
      }
    }
    cout << endl;
  }
}
