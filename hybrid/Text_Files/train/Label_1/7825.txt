#include <bits/stdc++.h>

using namespace std;

#define ll long long

bool is_intersect(pair<ll, ll> l1, pair<ll, ll> r1, pair<ll, ll> l2, pair<ll, ll> r2) {
  double c1 = (l2.second - r2.second) * (l1.first - l2.first) + (r2.first - l2.first) * (l1.second - l2.second);
  double c2 = (l2.second - r2.second) * (r1.first - l2.first) + (r2.first - l2.first) * (r1.second - l2.second);
  double d1 = (r1.second - l1.second) * (l2.first - l1.first) + (l1.first - r1.first) * (l2.second - l1.second);
  double d2 = (r1.second - l1.second) * (r2.first - l1.first) + (l1.first - r1.first) * (r2.second - l1.second);
  if (c1 * c2 < 0 && d1 * d2 < 0) return true;
  else if ((c1 == 0 && (l2.first <= l1.first <= r2.first || l2.first <= r1.first <= r2.first)) || (c2 == 0 && (l2.first <= l1.first <= r2.first || l2.first <= r1.first <= r2.first)))
    return true;
  else if ((d1 == 0 && (l1.first <= l2.first <= r1.first || l1.first <= r2.first <= r1.first)) || (d2 == 0 && (l1.first <= l2.first <= r1.first || l1.first <= r2.first <= r1.first)))
    return true;
  else
    return false;
}

int main() {
  int n;
  cin >> n;
  vector<pair<ll, ll>> points;
  for (int i = 0; i < n; i++) {
    ll x, y;
    cin >> x >> y;
    points.push_back({x, y});
  }
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      for (int k = j + 1; k < n; k++) {
        if (is_intersect(points[i], points[j], points[j], points[k])) ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
