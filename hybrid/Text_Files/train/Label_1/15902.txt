#include<bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
constexpr ll maxl = 100000;
constexpr ll limit = 1e9;
ll n, a[maxl], b[maxl], c[maxl], sa, sb, ca, cb;
int main() {
    cin >> n;
    cin >> sa >> sb;
    for (ll i = 0; i < n; i++) a[i] = (sa / (c[i] = pow(10, i))) % 10, b[i] = (sb / (c[i] = pow(10, i))) % 10;
    bool f = 0;
    a[n] = b[n] = 0;
    vector<pair<int, int>> ans;
    for (int i = 0; i < n || i < n - 1; i++) {
        if (a[i] != b[i]) {
            if (i == n - 1) { i--; continue; }
            if (a[i] > b[i]) {
                if (a[i + 1] == 9 && (a[i] - b[i] > 1 || (a[i] - b[i] == 1 && b[i + 1] == 0))) { f = 1; break; }
                a[i + 1] += a[i] - b[i]; a[i] = b[i]; ca++;
                ans.pb({i, a[i] - b[i]});
            }
            else {
                if (a[i] == 0 && a[i + 1] == 0 && (b[i] - a[i] > 1 || (b[i] - a[i] == 1 && b[i + 1] == 9))) { f = 1; break; }
                a[i + 1] -= b[i] - a[i]; a[i] = b[i]; ca++;
                ans.pb({i, b[i] - a[i]});
            }
            if (ca + cb > limit) { f = 1; break; }
            if (ca + cb >= 100000) { sb--, cb++; break; }
        }
        else if (i == n - 1 && a[i] == b[i] && ca + cb >= 100000) { sb--, cb++; break; }
        cb += a[i] != b[i];
    }
    if (!f && sb <= sa) { cout << ca + cb << endl; for (auto x : ans) cout << x.first + 1 << " " << x.second << endl; }
    else cout << -1 << endl;
    return 0;
}
