#include<bits/stdc++.h>
#define ll long long int
using namespace std;

ll num[1000001];

bool valid(ll x, ll l, ll u, ll sum) {

    if(x>0 && x<=u &&  num[x]>0 && x!=l)
        return false;

    if(x>l && num[x]==0)
        return false;

    sum-=num[x];
    num[x]=1;
    x++;

    if(sum%(x)<0 || (x*x)-sum/(x) != num[l])
        return false;

    ll p = sum/(x);
    if(p<=u && p>=l && num[p]==0)
        return true;

    return false;

}

int main()
{
    ll n, a, b, c, d, i, j, sum=0, l=0, u=0, cnt=0;
    cin>>n>>a>>b>>c>>d;
    sum=a+b+c+d;
    u=n;
    num[a]=num[b]=num[c]=num[d]=1;
    for(i=1; i<=n; i++)
    {
        if(valid(l, i, u, sum))
            cnt++;
        if(valid(u, l, i, sum))
            cnt++;
        l++;
    }
    cout<<cnt<<endl;
    return 0;
}
