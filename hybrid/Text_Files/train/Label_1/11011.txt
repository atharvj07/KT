#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    ll n, k;
    cin >> n >> k;
    vector<ll> v(n);
    for (auto &i : v) cin >> i;
    sort(v.begin(), v.end());
    if (k <= 0 || k > n)
        cout << "-1\n";
    else
        cout << v[k - 1] << "\n";
}
