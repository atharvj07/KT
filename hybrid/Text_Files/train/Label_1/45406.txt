import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        sc.close();
        long x = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == b[i]) continue;
            if (x == b[i]) {
                ans++;
            } else if (x == 0) {
                x = b[i] ^ a[i];
                ans++;
            } else {
                System.out.println(-1);
                return;
            }
        }
        System.out.println(ans);
    }
}
