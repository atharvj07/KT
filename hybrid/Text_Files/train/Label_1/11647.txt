#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, l = 0, r = 0, ans = 0;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<int> lmax(n), rmax(n);
    lmax[0] = v[0];
    for (int i = 1; i < n; i++) {
        lmax[i] = max(lmax[i - 1], v[i]);
    }
    rmax[n - 1] = v[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        rmax[i] = max(rmax[i + 1], v[i]);
    }
    for (int i = 0; i < n; i++) {
        while (l < i && v[i] >= lmax[l]) l++;
        while (r > i && v[i] >= rmax[r]) r--;
        ans += l * 1LL * (n - r);
    }
    cout << ans << endl;
}
