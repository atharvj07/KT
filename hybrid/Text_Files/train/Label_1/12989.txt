#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<double> prices(n);
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    prices[i] = (double)a / b;
  }
  sort(prices.begin(), prices.end());
  double ans = 0;
  for (int i = 0; i < n && m; i++) {
    if (m >= 1) {
      ans += prices[i];
      m--;
    } else {
      ans += prices[i] * m;
      m = 0;
    }
  }
  cout << ans << endl;
  return 0;
}
