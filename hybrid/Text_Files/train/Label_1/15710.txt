#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        vector<int> c(10);
        while (n || !c[0]) {
            int d = n % 10;
            c[d]++;
            if (c[d] == 1 && d > 0) k--;
            n /= 10;
            if (k == 0) {
                for (int i = 0; i < d; i++) {
                    while (c[i]) {
                        cout << i;
                        c[i]--;
                    }
                }
                while (c[d]) {
                    cout << d;
                    c[d]--;
                }
                for (int i = d + 1; i < 10; i++) {
                    while (c[i]) {
                        cout << i;
                        c[i]--;
                    }
                }
                cout << endl;
                break;
            }
        }
    }
    return 0;
}
