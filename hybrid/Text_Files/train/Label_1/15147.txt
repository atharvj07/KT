#include <iostream>

int main() {
    int n;
    std::cin >> n;
    char grid[301][301];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            std::cin >> grid[i][j];
    bool valid = true;
    char diag1 = grid[0][0], diag2 = grid[0][n - 1];
    for (int i = 0; i < n; i++) {
        if (grid[i][i] != diag1 || grid[i][n - 1 - i] != diag2)
            valid = false;
    }
    char other = grid[1][1];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if ((i == 0 || i == n - 1) && (j == 0 || j == n - 1))
                continue;
            if (grid[i][j] != other)
                valid = false;
        }
    }
    std::cout << (valid ? "YES" : "NO") << std::endl;
    return 0;
}
