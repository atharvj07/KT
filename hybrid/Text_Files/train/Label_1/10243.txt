#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  int p = min_element(s.begin(), s.end()) - s.begin();
  string t = s;
  if (p != 0) t[p - 1] = t[p];
  p = min_element(t.begin(), t.end()) - t.begin();
  cout << t.substr(0, p) + t.substr(p + 1) << endl;
}
