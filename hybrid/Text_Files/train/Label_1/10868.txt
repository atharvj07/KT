#include <bits/stdc++.h>
#define ll long long
using namespace std;
void solve() {
  int n;
  cin >> n;
  map<ll, ll> mp;
  ll ans = 0;
  while (n--) {
    ll x, y;
    cin >> x >> y;
    ans += mp[x + y] + mp[x - y];
    mp[x + y]++;
    mp[x - y]++;
  }
  cout << ans << '\n';
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) solve();
  return 0;
}
