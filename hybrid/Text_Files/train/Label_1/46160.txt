import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Arrays;

public class Main {
    private static final long MOD = 998244353L;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        String[] str = br.readLine().split(" ");
        BigInteger[] a = new BigInteger[n];

        for (int i = 0; i < n; i++) {
            a[i] = new BigInteger(str[i]).mod(BigInteger.valueOf(MOD));
        }

        Arrays.sort(a);

        BigInteger result = BigInteger.ZERO;

        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                result = result.add(f(a[i], a[j]).mod(BigInteger.valueOf(MOD)));
            }
        }

        System.out.println(result.mod(BigInteger.valueOf(MOD)));
    }

    private static BigInteger f(BigInteger a, BigInteger b) {
        StringBuilder sb = new StringBuilder();

        int i = 0;
        int j = 0;

        while (i < a.toString().length() || j < b.toString().length()) {
            if (i < a.toString().length()) {
                sb.append(a.toString().charAt(i));
                i++;
            }

            if (j < b.toString().length()) {
                sb.append(b.toString().charAt(j));
                j++;
            }
        }

        return new BigInteger(sb.toString());
    }
}
