from collections import deque

def find_course_order(n, k, main_courses, dependencies):
    graph = {i: [] for i in range(1, n + 1)}
    indegree = {i: 0 for i in range(1, n + 1)}

    # Build graph and indegree map
    for i in range(1, n + 1):
        for dep in dependencies[i]:
            graph[dep].append(i)
            indegree[i] += 1

    # Start with courses that have no dependencies
    queue = deque([i for i in range(1, n + 1) if indegree[i] == 0])
    order = []

    while queue:
        course = queue.popleft()
        order.append(course)

        for next_course in graph[course]:
            indegree[next_course] -= 1
            if indegree[next_course] == 0:
                queue.append(next_course)

    # Check if all main courses are in the order
    if not all(course in order for course in main_courses):
        return -1, []

    # Filter the order to contain only necessary courses
    necessary_courses = set(main_courses)
    for course in reversed(order):
        if course in necessary_courses:
            for dep in dependencies[course]:
                necessary_courses.add(dep)

    # Reconstruct the order for necessary courses only
    necessary_order = [course for course in order if course in necessary_courses]

    return len(necessary_order), necessary_order

# Read input
n, k = map(int, input().split())
main_courses = list(map(int, input().split()))
dependencies = {i: [] for i in range(1, n + 1)}
for i in range(1, n + 1):
    line = list(map(int, input().split()))
    dependencies[i] = line[1:]

# Solve
m, course_order = find_course_order(n, k, main_courses, dependencies)

# Output
if m == -1:
    print(m)
else:
    print(m)
    print(' '.join(map(str, course_order)))
