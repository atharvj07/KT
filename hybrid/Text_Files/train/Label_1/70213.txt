n = int(input())
line = list(map(int, input().split()))
swaps = 0

for i in range(2*n):
    if i % 2 == 0:
        if line[i] != line[i+1]:
            for j in range(i+2, 2*n):
                if line[j] == line[i]:
                    swaps += j - (i + 1)
                    line.pop(j)
                    line.insert(i+1, line[i])
                    break
print(swaps)
