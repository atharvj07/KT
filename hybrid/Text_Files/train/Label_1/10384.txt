#include <iostream>
#include <string>

int main() {
  std::string table, hand[5];
  std::cin >> table >> hand[0] >> hand[1] >> hand[2] >> hand[3] >> hand[4];
  for (int i = 0; i < 5; ++i) {
    if (table[0] == hand[i][0] || table[1] == hand[i][1]) {
      std::cout << "YES\n";
      return 0;
    }
  }
  std::cout << "NO\n";
  return 0;
}
