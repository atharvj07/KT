#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

void solve() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];

  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    ll need = (k - a[i] % k) % k;
    ans += need;
    a[i] += need;
  }
  cout << ans << endl;
}

int main() {
  ll t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
