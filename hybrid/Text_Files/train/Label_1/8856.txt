#include <cstdio>
#include <iostream>
using namespace std;
int r, c, ans;
const int dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};
int main() {
  for (int i = 1; i <= 5; ++i) {
    for (int j = 1; j <= 5; ++j) {
      cin >> r >> c;
      if (r == 1) {
        if (r == 3) ans += abs(3 - i);
        if (c == 3) ans += abs(3 - j);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
