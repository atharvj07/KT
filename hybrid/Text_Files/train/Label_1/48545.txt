import java.util.Scanner;
import java.util.Arrays;

class Main {
  static int n, m, q;
  static int[][] elements;
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    q = sc.nextInt();
    elements = new int[q][2];
    for (int i = 0; i < q; i++) {
      elements[i][0] = sc.nextInt() - 1;
      elements[i][1] = sc.nextInt() - 1;
    }
    Arrays.sort(elements, (a, b) -> (a[0] - b[0] != 0) ? a[0] - b[0] : a[1] - b[1]);
    int[] rows = new int[n];
    int[] cols = new int[m];
    int count = 0;
    for (int[] element : elements) {
      if (rows[element[0]] == 0 && cols[element[1]] == 0) {
        rows[element[0]]++;
        cols[element[1]]++;
      } else {
        count++;
      }
    }
    System.out.println(count);
  }
}
