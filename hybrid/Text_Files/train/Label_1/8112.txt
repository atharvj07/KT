#include <iostream>
#define _ % 1000000007LL
typedef long long ll;
const int mxn = 100005;
using namespace std;
int main() {
  ll n, k, dp[mxn], p10[mxn];
  p10[0] = 1LL;
  cin >> n >> k;
  for (int i = 1; i < mxn; i++) p10[i] = p10[i - 1] * 10LL _;
  dp[0] = 1LL;
  for (int i = 1; i < mxn; i++) dp[i] = 0LL;
  int a[mxn], b[mxn];
  for (int i = 0; i < n / k; i++) cin >> a[i];
  for (int i = 0; i < n / k; i++) cin >> b[i];
  ll pmx = 0;
  for (int i = 0; i + k - 1 < n / k; i++) {
    if (!a[i])
      dp[pmx] = 2 * dp[pmx] _;
    else if (pmx >= a[i])
      dp[pmx - a[i]] = 2 * dp[pmx - a[i]] _;
    if (b[i]) {
      pmx += p10[k - 1] * b[i] _;
      dp[pmx] *= 9LL _;
    } else
      dp[pmx] *= 10LL _;
    pmx += p10[k - 1] _;
  }
  cout << dp[0] << endl;
}
