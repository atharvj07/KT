import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] votes = new int[m+1];
        int[] cost = new int[n+1];
        int maxVotes = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            int party = sc.nextInt();
            int c = sc.nextInt();
            votes[party]++;
            cost[i] = c;
            maxVotes = Math.max(maxVotes, votes[party]);
        }
        Arrays.sort(cost);
        for (int i = 0; i < n; i++) {
            sum += cost[i];
            votes[1]++;
            if (votes[1] > maxVotes) {
                System.out.println(sum);
                return;
            }
        }
        System.out.println("-1");
    }
}
