#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n);
    for (int i = 0; i < m; i++) {
      int u, v;
      cin >> u >> v;
      u--, v--;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    vector<int> in(n), out(n);
    for (int i = 0; i < n; i++) {
      for (int j : adj[i]) {
        in[j]++;
        out[i]++;
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (in[i] == out[i]) ans++;
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) {
      for (int j : adj[i]) {
        if (in[j] > out[j]) {
          cout << i + 1 << " " << j + 1 << endl;
          in[j]--, out[i]++;
        }
      }
    }
  }
  return 0;
}
