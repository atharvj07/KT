#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3 + 5;
int n, c[maxn];
int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> c[i];
    int ans = 0;
    c[0] = c[n + 1] = 0;
    for (int l = 0, r = 0, sum = 0; l <= n; sum -= c[l], ++l) {
        while (l <= n && (sum += c[l]) > 0) ++l;
        while (r <= n && 0 >= (sum += c[r])) {
            if (sum == 0)
                ans += l > r;
            ++r;
        }
    }
    cout << ans << endl;
    return 0;
}
