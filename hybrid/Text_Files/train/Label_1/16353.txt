#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n, string(m, 0));
  for (int i = 0; i < n; ++i) cin >> a[i];
  int ans = 0, b[m];
  const string signs = "=<>";
  for (int i = 0, alone = 1; i < n; ++i) {
    bool fl = 0;
    for (int j = 0; j < m; ++j) {
      if (a[i][j] == '=') continue;
      fl = 1;
      if (a[i][j] == '<') {
        if (b[j] != 0) {
          cout << "No" << endl;
          return 0;
        }
        b[j] = -alone;
      } else {
        if (b[j] == -alone) {
          cout << "No" << endl;
          return 0;
        }
        b[j] = alone;
      }
    }
    if (!fl) ans++;
    alone++;
  }
  cout << "Yes" << endl;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (a[i][j] == '=') cout << ans << ' ';
      else if (a[i][j] == '<') cout << ans + b[j] << ' ';
      else cout << ans - b[j] << ' ';
    }
    cout << endl;
  }
  return 0;
}
