import sys

def solve():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n = int(sys.stdin.readline())
        a = list(map(int, sys.stdin.readline().split()))
        b = sorted(a)
        if abs(b[0]) == abs(b[-1]):
            ans = 0
        else:
            flag = True
            for i in range(1, n):
                if b[i] == b[i - 1]:
                    flag = False
            if flag:
                ans = 1
            else:
                ans = 2
        print(ans)

solve()
