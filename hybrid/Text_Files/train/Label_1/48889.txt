import java.util.*;
import java.io.*;

class Main {
  static int n, m, s, t;
  static boolean[][] adj;
  static int[] dist, path;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    m = in.nextInt();
    s = in.nextInt() - 1;
    t = in.nextInt() - 1;
    adj = new boolean[n][n];
    for (int i = 0; i < m; i++) {
      int a = in.nextInt() - 1, b = in.nextInt() - 1;
      adj[a][b] = adj[b][a] = true;
    }
    dijkstra();
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (i == s || i == t) continue;
      if (dist[i] + 1 == dist[s]) ans++;
    }
    System.out.println(ans);
  }

  static void dijkstra() {
    PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(o -> o[1]));
    pq.offer(new int[]{s, 0});
    dist = new int[n];
    path = new int[n];
    Arrays.fill(dist, Integer.MAX_VALUE);
    dist[s] = 0;
    while (!pq.isEmpty()) {
      int[] cur = pq.poll();
      int node = cur[0], d = cur[1];
      if (d > dist[node]) continue;
      for (int i = 0; i < n; i++) {
        if (i == node) continue;
        if (adj[node][i] && dist[node] + 1 < dist[i]) {
          dist[i] = dist[node] + 1;
          pq.offer(new int[]{i, dist[i]});
          path[i] = node;
        }
      }
    }
  }
}
