#include <iostream>

using namespace std;

int main() {
    int c, v0, v1, a, l;
    cin >> c >> v0 >> v1 >> a >> l;
    int days = 0, pages = 0;
    while (pages < c) {
        days++;
        if (pages + v0 + days * a <= v1) {
            pages += v0 + days * a;
        } else {
            pages += v1;
        }
        pages -= l;
    }
    cout << days << endl;
    return 0;
}
