#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define fast ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
typedef long long ll;
ll mod = 1000000007;
const int N = 100001;
vector<int> g[N], sz(N), freq(N);
vector<ll> beauty(N);
void dfs(int u, int p) {
    sz[u] = 1;
    for (int v : g[u]) {
        if (v != p) {
            dfs(v, u);
            sz[u] += sz[v];
        }
    }
}
ll gcd(ll a, ll b) {
    if (!b) return a;
    return gcd(b, a % b);
}
ll sum = 0;
void dfs2(int u, int p) {
    for (int v : g[u]) {
        if (v != p) {
            sum += gcd(beauty[u], beauty[v]);
            sum %= mod;
            dfs2(v, u);
            sum += gcd(beauty[u], beauty[v]);
            sum %= mod;
        }
    }
}
int main() {
    fast;
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> beauty[i + 1];
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    dfs(1, 1);
    dfs2(1, 1);
    cout << sum << endl;
    return 0;
}
