#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s;
  cin >> s;
  vector<int> freq(10, 0);
  for (auto c : s) freq[c - '0']++;
  int ans = 0;
  for (int i = 1; i < 10; i++) {
    if (freq[i]) {
      ans++;
      freq[i]--;
      for (int j = 0; j < 10; j++) {
        if (freq[j] > 0) {
          ans += freq[j];
        }
      }
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
