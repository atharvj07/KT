#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
  int ans = 0;
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] == 0) {
      ++cnt;
    } else {
      if (cnt >= 2) {
        ans += cnt - 1;
      }
      cnt = 0;
    }
  }
  if (cnt >= 2) {
    ans += cnt - 1;
  }
  std::cout << ans << std::endl;
  return 0;
}
