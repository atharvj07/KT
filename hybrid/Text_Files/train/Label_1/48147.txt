import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        // Get N, M, and the array A
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int M = scanner.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextInt();
        }
        scanner.close();

        // Sort the array A in descending order
        sortDescending(A);

        // Initialize the minimum amount of money required to buy all the items
        long minimumAmount = 0;

        // Use the discount tickets to buy the items
        int[] discountTickets = new int[M];
        for (int i = 0; i < M; i++) {
            discountTickets[i] = 1 << i;
        }
        for (int i = 0; i < N; i++) {
            // Use the maximum discount that is less than or equal to the current item price
            int maxDiscount = 0;
            for (int j = 0; j < M; j++) {
                if (discountTickets[j] <= A[i]) {
                    maxDiscount = discountTickets[j];
                }
            }
            // Add the discounted price of the current item to the minimum amount
            minimumAmount += A[i] / maxDiscount;
        }

        // Print the minimum amount of money required to buy all the items
        System.out.println(minimumAmount);
    }

    // Sort an array in descending order
    private static void sortDescending(int[] A) {
        for (int i = 0; i < A.length - 1; i++) {
            for (int j = i + 1; j < A.length; j++) {
                if (A[j] > A[i]) {
                    int temp = A[i];
                    A[i] = A[j];
                    A[j] = temp;
                }
            }
        }
    }
}
