n = int(input())
words = []
for i in range(n):
    words.append(input())

# sort the words by length in descending order
words.sort(key=lambda x: len(x), reverse=True)

# set of used letters
used = set()

# answer
ans = 0

# iterate over the words
for word in words:
    # check if the word can be used
    can_use = True
    for letter in word:
        if letter in used and letter != word[0]:
            can_use = False
    # if the word can be used, add its length to the answer
    if can_use:
        ans += len(word)
        # add the letters of the word to the set of used letters
        for letter in word:
            used.add(letter)

print(ans)
