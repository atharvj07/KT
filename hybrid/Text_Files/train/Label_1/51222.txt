import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] min = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            min[i] = Integer.MAX_VALUE;
        }
        for (int i = 1; i <= n; i++) {
            min[i] = Math.min(min[i], c[i - 1]);
            if (i > 1) {
                min[i] = Math.min(min[i], min[a[i - 1]]);
            }
        }
        int result = 0;
        for (int i = 1; i <= n; i++) {
            result += min[i];
        }
        System.out.println(result);
    }
}
