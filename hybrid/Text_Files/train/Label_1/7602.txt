#include <bits/stdc++.h>
using namespace std;

int q, n, k, a[200005];
int sum[200005];

int main() {
    cin >> q;
    while (q--) {
        cin >> n >> k;
        for (int i = 1; i <= n; i++) cin >> a[i];
        sum[0] = 0;
        for (int i = 1; i <= n; i++) { sum[i] = sum[i - 1] + a[i]; }
        if ((sum[n] ^ k) & 1) {
            cout << "NO\n";
            continue;
        }
        cout << "YES\n";
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (((sum[i] - sum[cnt]) ^ 1) & 1) {
                cout << i << ' ';
                cnt = i;
            }
        }
        cout << n << '\n';
    }
}
