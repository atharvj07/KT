import java.util.Scanner;

public class Solution {

    static boolean isTree(int[][] segments) {
        int n = segments.length;
        int[] parent = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            parent[i] = i;
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (intersect(segments[i], segments[j])) {
                    int p1 = find(parent, i);
                    int p2 = find(parent, j);
                    if (p1 == p2) {
                        return false;
                    }
                    parent[p1] = p2;
                }
            }
        }

        return true;
    }

    static boolean intersect(int[] a, int[] b) {
        return a[0] < b[1] && a[1] > b[0] && a[0] != b[0] && a[1] != b[1];
    }

    static int find(int[] parent, int x) {
        if (parent[x] != x) {
            parent[x] = find(parent, parent[x]);
        }
        return parent[x];
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] segments = new int[n][2];
        for (int i = 0; i < n; i++) {
            segments[i][0] = scanner.nextInt();
            segments[i][1] = scanner.nextInt();
        }
        scanner.close();

        if (isTree(segments)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
