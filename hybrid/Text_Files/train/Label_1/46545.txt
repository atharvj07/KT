import java.util.*;
import java.io.*;
import java.lang.*;

class Main {

    static int n;
    static int[] deg;
    static int[] par;
    static int[] sz;
    static ArrayList<Integer>[] g;
    static int u, v;

    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] ss = br.readLine().split(" ");
        n = Integer.parseInt(ss[0]);
        u = Integer.parseInt(ss[1]);
        v = Integer.parseInt(ss[2]);

        g = new ArrayList[n+1];
        for(int i = 1; i <= n; i ++) {
            g[i] = new ArrayList<Integer>();
        }
        deg = new int[n+1];
        par = new int[n+1];
        sz = new int[n+1];

        for(int i = 0; i < (n-1); i ++) {
            ss = br.readLine().split(" ");
            int a = Integer.parseInt(ss[0]);
            int b = Integer.parseInt(ss[1]);
            g[a].add(b);
            g[b].add(a);
            deg[a] ++;
            deg[b] ++;
        }
        dfs1(1, 1);
        dfs2(u, 1, 0);
        int ans = 0;
        while(u != v) {
            u = par[u];
            ans ++;
        }
        System.out.println(ans);
    }

    static void dfs1(int now, int pre) {
        par[now] = pre;
        sz[now] = 1;
        for(int i = 0; i < g[now].size(); i ++) {
            int next = g[now].get(i);
            if(next == pre) continue;
            dfs1(next, now);
            sz[now] += sz[next];
        }
    }

    static void dfs2(int now, int tot, int pre) {
        if(2*sz[now] >= tot) {
            v = now;
            return;
        }
        int mx = 0;
        int nxt = 0;
        for(int i = 0; i < g[now].size(); i ++) {
            int next = g[now].get(i);
            if(next == pre) continue;
            if(mx < sz[next]) {
                mx = sz[next];
                nxt = next;
            }
        }
        dfs2(nxt, tot, now);
    }

}
