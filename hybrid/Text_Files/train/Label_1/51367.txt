import java.util.*;

public class Main {

    static int n, max;
    static int[] t, b;
    static boolean[][] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            n = sc.nextInt();
            if (n == 0) break;
            t = new int[n];
            b = new int[n];
            g = new boolean[n][n];
            max = 0;
            for (int i = 0; i < n; ++i) {
                t[i] = sc.nextInt();
                b[i] = sc.nextInt();
                max += t[i];
                g[0][i] = g[i][0] = true;
            }
            for (int i = 1; i < n; ++i) {
                for (int j = i + 1; j < n; ++j) {
                    if (b[i] >= t[j] && b[j] >= t[i]) {
                        g[i][j] = g[j][i] = true;
                    }
                }
            }

            int[] d = dijkstra(0);
            System.out.println(d[n] == max ? "Yes" : "No");
        }
    }

    static int[] dijkstra(int s) {
        int[] d = new int[n + 1];
        Arrays.fill(d, Integer.MAX_VALUE);
        d[s] = 0;
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(p -> p[0]));
        pq.add(new int[] { 0, s });
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            if (cur[0] > d[cur[1]]) continue;
            for (int i = 0; i < n; ++i) {
                if (!g[cur[1]][i]) continue;
                int nd = cur[0] + t[i];
                if (nd < d[i]) {
                    pq.add(new int[] { nd, i });
                    d[i] = nd;
                }
            }
        }
        return d;
    }
}
