#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(0);
    int n;
    cin >> n;
    vector<long long> b(n / 2), a(n);
    for (int i = 0; i < n / 2; i++) { cin >> b[i]; }
    for (int i = 0; i < n / 2; i++) {
        a[i] = b[i] / 2;
        a[n - i - 1] = b[i] - a[i];
    }
    for (int i = 0; i < n; i++) { cout << a[i] << ' '; }
    cout << '\n';
}
