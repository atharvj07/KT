#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, a, b, c, d;
  cin >> n >> k >> a >> b >> c >> d;
  if (k < n - 1 || k > 2 * n - 2) {
    cout << -1;
    return 0;
  }
  vector<pair<int, int>> arr;
  for (int i = 1; i < n; i++) {
    if (i != a && i != b) {
      arr.emplace_back(a, i);
      a = i;
    } else if (i != c && i != d) {
      arr.emplace_back(c, i);
      c = i;
    }
  }
  arr.emplace_back(a, b);
  arr.emplace_back(c, d);
  if (arr.size() > k) {
    cout << -1;
    return 0;
  }
  for (auto u : arr) cout << u.first << ' ' << u.second << '\n';
  return 0;
}
