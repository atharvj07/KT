#include <bits/stdc++.h>
using namespace std;
 
int main() {
    int n, h, a, b, q;
    cin >> n >> h >> a >> b >> q;
    for (int i = 0; i < q; i++) {
        int xa, ya, xb, yb;
        cin >> xa >> ya >> xb >> yb;
        if (xa == xb) {
            cout << abs(ya - yb) << endl;
        } else {
            cout << abs(ya - a) + abs(xb - xa) + abs(yb - a) << endl;
        }
    }
}
