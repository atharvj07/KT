#include <bits/stdc++.h>
using namespace std;
const int N = 105, M = 100005;
int n, k, arr[N], tmp[N], c[M];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) { cin >> arr[i]; }
    if (arr[n] <= k) {
      cout << 1 << endl;
      continue;
    }
    int mx = 0, last = 0;
    for (int i = n; i >= 1; i--) {
      tmp[i] = arr[i] - last;
      last = arr[i];
      mx = max(mx, tmp[i]);
    }
    if (mx <= k) {
      cout << 2 << endl;
      continue;
    }
    memset(c, 0, sizeof c);
    for (int i = 1; i <= n; i++) { c[tmp[i]]++; }
    int ans = 0;
    for (int i = 100000; i >= 1; i--) {
      if (c[i] > 0) {
        ans += (c[i] + k - 1) / k;
        for (int j = i; j <= 100000; j += i) { c[j] = 0; }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
