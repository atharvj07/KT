import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] a = new int[n][5];
        int[][] b = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                a[i][j] = in.nextInt();
                if (a[i][j] == -1) {
                    a[i][j] = 119;
                }
                b[i][j] = a[i][j];
            }
        }
        int ans = -1;
        for (int i = 0; i < 5; i++) {
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (b[j][i] <= 20) {
                    cnt++;
                }
            }
            if (cnt * 250 < n) {
                int x = n - cnt * 250;
                int y = 0;
                for (int j = 0; j < n; j++) {
                    if (b[j][i] > 20) {
                        b[j][i] = Math.min(b[j][i], 120 - x * 250 / (n - cnt));
                        y++;
                        if (y == x) {
                            break;
                        }
                    }
                }
                if (y == x) {
                    ans = Math.max(ans, x);
                }
            }
        }
        System.out.println(ans);
    }
}
