#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mx = 2e5+2;
int a[mx];
ll bagsize;
void solve(){
  cin>>bagsize;
  int n; cin>>n;
  for(int i=0;i<n;++i) cin>>a[i];
  int mxb=0;
  for(int i=0;i<n;++i){
    int k=63;
    for(k=63;k>=0;--k)
      if(bagsize&((ll)1<<k) && a[i]&((ll)1<<k)) break;
    mxb=max(k,mxb);
  }
  if(bagsize%((ll)1<<mxb)){
    cout<<-1<<'\n'; return;
  }
  int ans=0;
  for(int i=mxb;i>=0;--i){
    if(!(bagsize&((ll)1<<i))) continue;
    ans++;
    if(i>=1)
      ans+=a[n-1]&(((ll)1<<i)-1)!=0;
  }
  cout<<ans<<'\n';
  return;
}
int main(){
  int t; cin>>t;
  while(t--){
    solve();
  }
  return 0;
}
