import java.util.*;

public class Main {
    static long mod = 998244353;
    static long bs(int p, long mult) {
        long r = 1;
        while (p > 0) {
            if ((p & 1) != 0) {
                r = (r * mult) % mod;
            }
            p /= 2;
            mult = (mult * mult) % mod;
        }
        return r;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] l = new int[n + 1];
        int[] r = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        Arrays.sort(l);
        Arrays.sort(r);
        long ans = 0;
        for (int i = n - k + 1; i >= 1; i--) {
            ans += l[i];
            ans %= mod;
            long rw = r[i + k - 1];
            ans += (rw - l[i]) * bs(k - 1, rw);
            ans %= mod;
        }
        System.out.println(ans);
    }
}
