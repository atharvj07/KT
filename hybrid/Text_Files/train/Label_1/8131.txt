#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
const ll inf = 9e18;
const double pi = 3.14159265358979323846;

int main() {
    ll n, m; cin >> n >> m;
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b; cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }

    vector<int> pa(n + 1), pb(n + 1);
    for (int i = 1; i <= n; i++) pa[i] = pb[i] = -1;

    vector<int> ans = {0, 0};
    for (int i = 1; i <= n; i++) {
        if (pa[i] != -1) continue;

        bool ok = false;
        vector<int> qa, qb;
        qa.push_back(i);
        pa[i] = 1;
        while (!qa.empty()) {
            int x = qa.back(); qa.pop_back();
            for (auto y : g[x]) {
                if (pa[y] == pb[y]) { ok = true; break; }
                if (pa[y] == 1) continue;
                pa[y] = 1;
                qa.push_back(y);
            }
        }
        if (ok) {
            for (int x = 1; x <= n; x++) {
                if (pa[x] == 1) ans[0]++, qa.push_back(x);
            }
            for (int x = 1; x <= n; x++) {
                if (pa[x] == -1) {
                    pb[x] = 1;
                    ans[1]++, qb.push_back(x);
                }
            }
            cout << ans[0] << "\n";
            for (auto x : qa) cout << x << " "; cout << "\n";
            cout << ans[1] << "\n";
            for (auto x : qb) cout << x << " "; cout << "\n";
            return 0;
        } else {
            for (int x = 1; x <= n; x++) {
                if (pa[x] == 1) pb[x] = 1;
            }
            ans[1]++;
            qb.push_back(i);
        }
    }

    cout << "-1\n";
    return 0;
}
