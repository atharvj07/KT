import java.util.*;

public class Main {
    static int n;
    static int[][] a;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        a = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                a[i][j + 1] = s.charAt(j) - '0';
            }
        }

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (a[i][j] == 0) {
                    if (a[j][i] == 0) {
                        System.out.println("-1");
                        return;
                    } else {
                        for (int k = 1; k <= n; k++) {
                            if (a[k][i] == 1 && a[j][k] == 1) {
                                System.out.println(i + " " + j + " " + k);
                                return;
                            }
                        }
                    }
                }
            }
        }
    }
}
