import java.util.*;
import java.lang.*;
class Main
{
    public static void main (String[] args) throws java.lang.Exception
    {
        Scanner s = new Scanner(System.in);

        int k = s.nextInt();
        int a = s.nextInt();
        int b = s.nextInt();

        if ((a + b) % k == 0) {
            System.out.println((a + b) / k);
        } else {
            System.out.println(-1);
        }
    }
}
