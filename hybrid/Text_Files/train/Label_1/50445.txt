import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n + 2];
        for (int i = 1; i <= n; i++) {
            d[i] = sc.nextInt();
        }
        d[0] = 0;
        d[n + 1] = 1000000000;
        boolean[] f = new boolean[n + 2];
        int l = 0;
        while (!f[n]) {
            l++;
            for (int i = 0; i <= n; i++) {
                if (f[i]) {
                    for (int j = i + 1; j <= n && j <= i + d[i]; j++) {
                        if (j == n) {
                            System.out.println("yes");
                            return;
                        }
                        f[j] = true;
                    }
                }
            }
            if (l > 100000) {
                break;
            }
        }
        System.out.println("no");
    }
}
