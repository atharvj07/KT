import sys

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def next_prime(L, cur):
    while True:
        cur += 1
        isPrime = True
        for i in range(2, int(cur**0.5) + 1):
            if cur % i == 0:
                isPrime = False
                break
        if isPrime:
            L.add(cur)
            return cur

def solve(x, p, k):
    L = set()
    for i in range(x + 1, int(1e6) + 1):
        if gcd(i, p) == 1:
            L.add(i)
            if len(L) == k:
                return i
    return -1

t = int(input())
for i in range(t):
    x, p, k = map(int, input().split())
    print(solve(x, p, k))
