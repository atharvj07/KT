import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), h = in.nextInt();
        int w = 0;
        for (int i = 0; i < n; i++) {
            int a = in.nextInt();
            w += a > h ? 2 : 1;
        }
        System.out.println(w);
    }
}
