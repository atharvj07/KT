#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
typedef long long ll;
int n, a[maxn];
ll sum;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  sort(a, a + n);
  ll ans = 0, now = 0;
  for (int i = 0; i < n; i++) {
    if (i == 0 || a[i] == a[i - 1])
      now++;
    else {
      ans += now * a[i];
      now = 1;
    }
  }
  ans += now * a[n - 1];
  cout << ans << endl;
}
