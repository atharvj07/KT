#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string a, b, c;
        cin >> a >> b >> c;
        vector<int> cnta(26, 0), cntb(26, 0), cntc(26, 0);
        for (char ch : a) { cnta[ch - 'a']++; }
        for (char ch : b) { cntb[ch - 'a']++; }
        for (char ch : c) { cntc[ch - 'a']++; }
        bool ok = true;
        for (int i = 0; i < 26; i++) {
            ok &= (cnta[i] + cntc[i]) == (cntb[i] + cntc[i]);
        }
        cout << (ok ? "YES" : "NO") << '\n';
    }
}
