#include <bits/stdc++.h>
using namespace std;
const int maxn = 1 << 18;
struct node {
  bool left, right, have;
} tree[maxn];
int n, m, a[maxn];
inline int lowbit(int x) { return x & -x; }
inline void add(int x, int v) {
  for (; x <= m + 2; x += lowbit(x)) tree[x].have |= v;
}
inline bool get(int x) {
  bool ans = 0;
  for (; x; x -= lowbit(x)) ans |= tree[x].have;
  return ans;
}
int main() {
  cin >> n;
  m = (n + 1) * 2;
  vector<int> ans;
  for (int i = 1; i <= n; i++) {
    a[i] = a[i + n] = i;
    add(i, 1);
    add(i + n, 1);
  }
  for (int i = 1; i <= n; i++) {
    tree[i + n].left = get(i + 1);
    tree[i].right = get(i - 1);
    if (!tree[i].right && !tree[i].left) {
      add(i, 0);
      add(i + n, 0);
      ans.push_back(i);
    } else {
      if (tree[i].right) { add(i, 0); }
      if (tree[i + n].left) { add(i + n, 0); }
    }
  }
  cout << ans.size() << '\n';
  for (int x : ans) cout << x << ' ';
  cout << '\n';
  return 0;
}
