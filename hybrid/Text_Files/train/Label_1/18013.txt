#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 1000 + 5;
const int MAXM = 1000 + 5;

struct Task {
    int time, period;
    bool operator<(const Task &b) const {
        if (time == b.time) {
            return period < b.period;
        }
        return time < b.time;
    }
};

int n, m;
int a[MAXN];
int l[MAXM], r[MAXM];
Task tasks[MAXN];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> l[i] >> r[i];
    }
    
    int cur_time = 0;
    int tasks_left = n;
    for (int i = 0; i < n; i++) {
        cur_time += a[i];
        int j = 0;
        while (j < m && r[j] < cur_time) {
            j++;
        }
        if (j < m) {
            tasks[i] = {cur_time, j};
            tasks_left--;
        } else {
            cout << -1 << endl;
            return 0;
        }
    }
    
    if (tasks_left > 0) {
        cout << -1 << endl;
        return 0;
    }
    
    sort(tasks, tasks + n);
    int ans = -1;
    for (int i = 0; i < n; i++) {
        if (l[tasks[i].period] <= tasks[i].time) {
            ans = tasks[i].time;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
