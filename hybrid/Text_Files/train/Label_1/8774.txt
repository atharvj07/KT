#include <iostream>
#include <vector>
#include <map>

using namespace std;

const int MX = 200005;
const long long MOD = 1e9 + 7;

int n;
vector<long long> a(MX);
map<long long, int> mp;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    for (int i = 0; i < n - 1; i++) {
        long long diff = a[i + 1] - a[i];
        mp[diff]++;
    }

    if (mp.size() <= 2) {
        cout << "YES\n";
        long long s = a[0] - 1;
        cout << s << " " << s + mp.begin()->first << "\n";
    } else {
        cout << "NO\n";
    }
    return 0;
}
