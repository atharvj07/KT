import java.util.Scanner;

public class ChoutiAndTheSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] even = new long[n / 2];
        for (int i = 0; i < n / 2; i++) {
            even[i] = scanner.nextLong();
        }

        long[] odd = new long[n / 2];
        long sum = 0;
        for (int i = 0; i < n / 2; i++) {
            sum += even[i];
            odd[i] = (long) Math.sqrt(sum) - (i == 0 ? 0 : odd[i - 1]);
            if (odd[i] * odd[i] != sum || odd[i] <= 0) {
                System.out.println("No");
                return;
            }
        }

        System.out.println("Yes");
        for (int i = 0; i < n; i++) {
            System.out.print((i % 2 == 0 ? even[i / 2] : odd[i / 2]) + " ");
        }
    }
}
