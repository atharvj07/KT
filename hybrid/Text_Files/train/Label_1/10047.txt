#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int mod = 1e9 + 7;
int dp[20];

int main() {
  int n, a;
  cin >> n >> a;
  int ans = a;
  while (n > 1) {
    n >>= 1;
    ans = max(ans, dp[n] + a);
    dp[n] = (dp[n] + a) % mod;
  }
  cout << ans << endl;
  return 0;
}
