#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    vector<int> dp(n, n);
    dp[0] = 0;
    for (int i = 1; i < n; ++i) {
        if (a[i] == 0) {
            if (dp[i - 1] != n) dp[i] = dp[i - 1];
        } else {
            for (int j = 0; j < i; ++j) {
                if (a[j] + 1 >= a[i]) {
                    if (dp[j] != n) dp[i] = min(dp[i], dp[j] + 1);
                }
            }
        }
    }
    cout << dp[n - 1] << endl;
    return 0;
}
