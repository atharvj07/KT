#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;
vector<int> up, dw;
int32_t main() {
    int n;
    cin >> n;
    up.resize(n + 1), dw.resize(n + 1);
    for (int i = 1; i <= n; i++) cin >> up[i];
    for (int i = 1; i <= n; i++) cin >> dw[i];
    if (up[1] == 0) {
        cout << -1;
        return 0;
    }
    vector<int> dis(n + 1, MOD), vis(n + 1, 0), par(n + 1, 0);
    queue<int> q;
    dis[1] = 0;
    vis[1] = 1;
    q.push(1);
    while (!q.empty()) {
        int cur = q.front();
        q.pop();
        for (int i = 1; i <= up[cur]; i++) {
            int nw = cur + i - dw[cur + i];
            if (nw <= n && vis[nw] == 0) {
                par[nw] = cur;
                dis[nw] = dis[cur] + 1;
                vis[nw] = 1;
                q.push(nw);
            }
        }
    }
    if (vis[n] == 0) {
        cout << -1;
        return 0;
    }
    cout << dis[n] << "\n";
    int tmp = n;
    vector<int> ans;
    while (tmp != 1) {
        ans.push_back(tmp);
        tmp = par[tmp];
    }
    ans.push_back(1);
    reverse(ans.begin(), ans.end());
    for (int i : ans) cout << i << " ";
    cout << "\n";
    return 0;
}
