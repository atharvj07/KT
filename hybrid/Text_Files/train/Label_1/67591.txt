import math

# Get the input
n = int(input())

# Calculate the maximum number of nines that can be achieved
max_nines = math.floor(math.log10(n) / math.log10(10))

# Calculate the number of pairs that end with the maximum number of nines
if max_nines == 0:
    num_pairs = n * (n - 1) // 2
else:
    num_pairs = (n - 10 ** max_nines + 1) * (n - 10 ** max_nines) // 2

# Print the result
print(num_pairs)
