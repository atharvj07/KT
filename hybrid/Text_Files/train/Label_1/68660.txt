import sys
input = sys.stdin.readline

n, m = map(int, input().split())
grid = [list(map(int, input().split())) for _ in range(n)]
k = int(input())
for l, r in [map(int, input().split()) for _ in range(k)]:
    flag = False
    for j in range(m):
        ok = True
        for i in range(l, r):
            if grid[i][j] > grid[i+1][j]:
                ok = False
        if ok:
            flag = True
            break
    print('Yes' if flag else 'No')
