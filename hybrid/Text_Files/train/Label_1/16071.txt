#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cin >> m;
    vector<int> b(m);
    for (int i = 0; i < m; i++) {
        cin >> b[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int sum = a[i] + b[j];
            if (!binary_search(a.begin(), a.end(), sum) &&
                !binary_search(b.begin(), b.end(), sum)) {
                cout << a[i] << ' ' << b[j] << endl;
                return 0;
            }
        }
    }
}
