#include <bits/stdc++.h>
using namespace std;
#define MAXN 202
#define ll long long

const ll INF = 1e18;
ll dp[MAXN][MAXN][2];
vector<ll> adj[MAXN];
ll w[MAXN], n, k;

ll dfs(ll u, ll par, ll remK){
    if(dp[u][remK][par != 0] != -1) return dp[u][remK][par != 0];

    ll &ans = dp[u][remK][par != 0];

    ans = w[u];
    for(ll v : adj[u]){
        if(v == par) continue;
        if(remK > 1) ans = max(ans, dfs(v, u, remK - 1) + w[u]);
        ans = max(ans, dfs(v, u, k) + w[u]);
    }

    return ans;
}

int main(){
    memset(dp, -1, sizeof dp);
    cin >> n >> k;
    for(ll i = 1; i <= n; ++i) cin >> w[i];
    for(ll i = 1; i < n; ++i){
        ll u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    cout << dfs(1, 1, k) << endl;
}
