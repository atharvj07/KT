#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int MAXN = 200;
bool adj[MAXN][MAXN];
int in_count[MAXN];
int out_count[MAXN];
int n, m;

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    memset(adj, 0, sizeof(adj));
    memset(in_count, 0, sizeof(in_count));
    memset(out_count, 0, sizeof(out_count));
    for (int i = 0; i < m; ++i) {
      int u, v;
      cin >> u >> v;
      --u, --v;
      adj[u][v] = adj[v][u] = true;
      ++in_count[v];
      ++out_count[u];
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
      if (in_count[i] == out_count[i]) ++ans;
    }
    cout << ans << endl;
    vector<pair<int, int>> oririented_edges;
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < n; ++j) {
        if (adj[i][j]) {
          if (in_count[i] < out_count[i]) {
            oririented_edges.emplace_back(i, j);
          } else if (in_count[j] < out_count[j]) {
            oririented_edges.emplace_back(j, i);
          }
        }
      }
    }
    for (const auto& edge : oririented_edges) {
      cout << edge.first + 1 << " " << edge.second + 1 << endl;
    }
  }
  return 0;
}
