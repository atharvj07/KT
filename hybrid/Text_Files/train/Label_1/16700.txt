#pragma GCC optimize ("O3")
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const int mod = 1e9 + 7;
const int maxn = int(1e5);
ll l[maxn], r[maxn], v[maxn];
int main() {
    ll n, k;
    cin >> n >> k;
    for (ll i = 0; i < n; ++i) {
        cin >> l[i] >> r[i];
        v[i] = r[i] - l[i] + 1;
    }
    ll sum = 0;
    for (ll i = 0; i < n; ++i) sum += v[i];
    sum %= k;
    for (ll i = 0; i < n; ++i) {
        if (v[i] % k == sum) {
            cout << 0;
            return 0;
        }
        if (k - v[i] % k <= r[i] - l[i]) {
            cout << 1;
            return 0;
        }
    }
    cout << 2;
    return 0;
}
