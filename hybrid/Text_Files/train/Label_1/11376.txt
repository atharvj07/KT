#include <bits/stdc++.h>
using namespace std;
// 0 -> not beautiful, 1 -> beautiful, 2 -> very beautiful
int dp[100005];
int main() {
  int t, n, a[100005];
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    dp[1] = 1;
    for (int i = 2; i <= n; i++) {
      dp[i] = 0;
      for (int j = 1; j < i; j++) {
        if (a[j] < a[i] && i % j == 0) {
          dp[i] = max(dp[i], dp[j] + 1);
        }
      }
      if (dp[i] > 1) dp[i] = 2;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) ans = max(ans, dp[i]);
    cout << ans << '\n';
  }
}
