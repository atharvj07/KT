import java.util.*;

public class Main {
    static int n, m, k;
    static int[] dist;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        List<List<Node>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            int l = in.nextInt();
            adj.get(a).add(new Node(b, l));
            adj.get(b).add(new Node(a, l));
        }
        dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        Queue<Node> q = new LinkedList<>();
        int[] mall = new int[k];
        for (int i = 0; i < k; i++) {
            mall[i] = in.nextInt();
            q.add(new Node(mall[i], 0));
            dist[mall[i]] = 0;
        }
        while (!q.isEmpty()) {
            Node current = q.poll();
            int u = current.node;
            int w = current.weight;
            for (Node next : adj.get(u)) {
                if (dist[next.node] > w + next.weight) {
                    dist[next.node] = w + next.weight;
                    q.add(new Node(next.node, w + next.weight));
                }
            }
        }
        int res = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] == Integer.MAX_VALUE) res++;
        }
        System.out.println(res);
    }
}

class Node {
    int node;
    int weight;

    public Node(int node, int weight) {
        this.node = node;
        this.weight = weight;
    }
}
