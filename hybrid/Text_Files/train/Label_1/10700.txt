#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int sum_s = 0, sum_d = 0;
  for (int i = n - 1; i >= 0; i -= 2) { sum_s += a[i]; }
  for (int i = n - 2; i >= 0; i -= 2) { sum_d += a[i]; }
  cout << sum_s << " " << sum_d << endl;
  return 0;
}
