#include <map>
#include <iostream>
#include <typeinfo>

using namespace std;

const string pokemons[8] = {"vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"};

int main() {
	int n;
	string s;
	cin >> n >> s;
	for (int i = 0; i < 8; i++) {
		bool matched = true;
		for (int j = 0; j < n; j++) {
			if (s[j] == '.' && pokemons[i][j] != '.') {
				matched = false;
				break;
			}
			if (s[j] != '.' && s[j] != pokemons[i][j]) {
				matched = false;
				break;
			}
		}
		if (matched) {
			cout << pokemons[i] << endl;
			return 0;
		}
	}
	return 0;
}
