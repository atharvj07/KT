#include <iostream>
using namespace std;

const int MAXN = 505;
int n, m, a[MAXN][MAXN], b[MAXN][MAXN];

bool ok(int x, int y) {
  if (x + 1 > n || y + 1 > m) return false;
  if (a[x][y] != b[x][y]) return false;
  if (a[x][y + 1] != b[x][y + 1]) return false;
  if (a[x + 1][y] != b[x + 1][y]) return false;
  if (a[x + 1][y + 1] != b[x + 1][y + 1]) return false;
  return true;
}

int main() {
  int x, y, ans = 1;
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      cin >> a[i][j];
      cin >> b[i][j];
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) { ans &= ok(i, j); }
  if (ans)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
  return 0;
}
