#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
vector<int> g[N], adj[N];
ll a[N], sum[N];
void dfs(int u, int f) {
    for (auto &v : g[u]) {
        if (v == f) continue;
        dfs(v, u);
        adj[u].push_back(v);
        sum[u] += sum[v];
    }
}
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v), g[v].push_back(u);
    }
    dfs(1, 1);
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        ll t = sum[i] + a[i];
        for (auto &v : adj[i]) {
            ans = max(ans, t - sum[v] + (sum[1] - sum[v] - t + a[i]));
        }
    }
    cout << ans << '\n';
    return 0;
}
