import sys
input = sys.stdin.readline

n, k = map(int, input().split())
c = list(map(int, input().split()))
ans = 0
t = [k + 1] * n
l = []
for i in range(n):
    for j in range(i + 1, n):
        if c[i] > c[j]:
            i, j = j, i
    l.append(i)
    ans += c[i] * (n - i)
for i in range(n):
    t[l[i]] = k + 1 + i
print(ans)
print(*t)
