#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m, a;
    cin >> n >> m >> a;
    vector<int> b(n), p(m);
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int j = 0; j < m; j++) cin >> p[j];
    sort(b.begin(), b.end());
    sort(p.begin(), p.end());
    int r = 0, s = 0;
    for (int i = 0, j = 0; i < n && j < m; ) {
        if (b[i] + p[j] <= a) {
            s += b[i];
            r++;
            i++;
            j++;
        } else {
            j++;
        }
    }
    cout << r << ' ' << s << endl;
}
