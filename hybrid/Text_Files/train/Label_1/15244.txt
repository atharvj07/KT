#include <iostream>
#include <cmath>
using namespace std;
int main() {
    long long int n, a, b, s, a1, b1;
    cin >> n >> a >> b;
    if (min(a, b) >= 6 * n) {
        if (a < b)
            swap(a, b);
        a1 = (6 * n) / b;
        if (a1 * b >= (6 * n))
            a1++;
        b1 = b;
    } else {
        s = ceil(6.0 * n / (a * b));
        a1 = s * a;
        b1 = s * b;
    }
    cout << a1 * b1 << endl << a1 << " " << b1 << endl;
    return 0;
}
