import math

# Reading Vasiliy's home coordinates
a, b = map(int, input().split())

# Reading the number of Beru-taxi cars
n = int(input())

min_time = float('inf')

# Reading each taxi's information and calculating the minimum time
for _ in range(n):
    x, y, v = map(int, input().split())
    distance = math.sqrt((x - a)**2 + (y - b)**2)
    time = distance / v
    min_time = min(min_time, time)

print(f"{min_time:.20f}")
