#include<bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int main() {
    int n, m;
    cin >> n >> m;
    string x, y;
    cin >> x >> y;
    while (y.size() < x.size()) y = '0' + y;
    while (x.size() < y.size()) x = '0' + x;
    reverse(x.begin(), x.end());
    reverse(y.begin(), y.end());
    long long ans = 0;
    for (int i = 0; i < x.size(); i++) {
        if (y[i] - '0') {
            int cur = 0;
            for (int j = i; j < x.size(); j++) {
                cur *= 2;
                cur += x[j] - '0';
                cur %= mod;
            }
            ans += cur;
            ans %= mod;
            string ny = y;
            ny.erase(ny.begin() + i);
            while (ny.size() < y.size()) ny = '0' + ny;
            y = ny;
        }
    }
    cout << ans << '\n';
    return 0;
}
