#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> dp(n, 1);
  int maxlen = 1;
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < i; j++) {
      if (s[i] != s[j] && s[i] >= 'a' && s[i] <= 'z') {
        dp[i] = max(dp[i], dp[j] + 1);
      }
    }
    maxlen = max(maxlen, dp[i]);
  }
  cout << maxlen << endl;
  return 0;
}
