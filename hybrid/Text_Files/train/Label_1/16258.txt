#include <bits/stdc++.h>
#define ll long long
using namespace std;

struct skill {
    ll cur;
    ll cost;
};

int main() {
    ll n, A, cf, cm, m;
    cin >> n >> A >> cf >> cm >> m;
    vector<skill> skillset(n + 2);
    ll minSkill = LLONG_MAX;
    for (ll i = 1; i <= n; i++) {
        cin >> skillset[i].cur;
        minSkill = min(skillset[i].cur, minSkill);
    }
    for (ll i = 1; i <= n; i++) {
        skillset[i].cost = A - skillset[i].cur;
    }
    sort(skillset.begin(), skillset.end(), [](skill a, skill b) { return a.cost > b.cost; });
    ll total = cm * minSkill;
    ll extra = 0;
    for (ll i = 1; i <= n; i++) {
        if (skillset[i].cost <= m) {
            m -= skillset[i].cost;
            extra += cf;
        } else {
            extra += cf * m / skillset[i].cost;
            m = 0;
            break;
        }
    }
    cout << total + extra << endl;
    for (ll i = 1; i <= n; i++) {
        cout << min(skillset[i].cur + m / n, A) << " ";
        m -= min(skillset[i].cur + m / n, A) - skillset[i].cur;
    }
    cout << endl;
}
