#include <iostream>
#include <cmath>

using namespace std;

typedef long long ll;

ll gcd( ll a, ll b ) {
    return b == 0 ? a : gcd( b, a % b );
}

int main() {
    ll n, m;
    while( cin >> n >> m ) {
        ll ans = 0;
        for( ll i = 1; i <= n; i++ ) {
            for( ll j = 1; j <= n; j++ ) {
                ll g = gcd( ( i * i + j * j ), m );
                if( g > 1 ) ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
