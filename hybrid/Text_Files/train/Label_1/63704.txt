t = int(input())
for _ in range(t):
    x, y, z = map(int, input().split())
    if x == y == z:
        print("YES")
        print(x, y, z)
    elif x == y and z < x:
        print("YES")
        print(x, z, 1)
    elif x == z and y < x:
        print("YES")
        print(x, y, 1)
    elif y == z and x < y:
        print("YES")
        print(y, x, 1)
    else:
        print("NO")
