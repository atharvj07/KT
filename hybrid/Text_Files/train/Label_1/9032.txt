#include <algorithm>
#include <iostream>
#include <map>
#include <string>
using namespace std;

int main() {
    int n;
    cin >> n;
    string a;
    cin >> a;

    int maps[10];
    for (int i = 0; i < 9; i++) {
        cin >> maps[i + 1];
    }

    int max_after_segment[n + 1];
    for (int i = n; i >= 1; i--) {
        max_after_segment[i] = maps[a[i - 1] - '0'];
        for (int j = i + 1; j <= n; j++) {
            max_after_segment[i] = max(max_after_segment[i], maps[a[j - 1] - '0']);
        }
    }

    string ans = a;
    for (int i = 1; i < n; i++) {
        if (maps[a[i - 1] - '0'] < maps[a[i] - '0']) {
            int candidate = max_after_segment[i + 1];
            if (candidate > maps[a[i] - '0']) {
                ans = a.substr(0, i) + to_string(candidate) + a.substr(i, n - i);
            }
        }
    }

    for (int i = 0; i < n; i++) {
        if (maps[a[i] - '0'] < maps[a[(i + 1) % n] - '0']) {
            int candidate = max_after_segment[(i + 1) % n + 1];
            if (candidate > maps[a[i] - '0']) {
                ans = a.substr(0, i) + to_string(candidate) + ans.substr(i, n - i);
            }
        }
    }

    cout << ans << endl;
    return 0;
}
