#include<bits/stdc++.h>
using namespace std;
const int mx=7010;
#define ll long long

vector<pair<int,int>>adj[mx];
int n,k1,k2;
int s1[mx],s2[mx];

bool vis[mx];

string res[2];

void dfs(int u,int p,bool f,int set){
    if(vis[u])   return;
    vis[u]=1;
    res[f]+=(set==1?"Win ":"Lose ");
    for(auto v:adj[u]){
        if(v.first==p)   continue;
        dfs(v.first,u,1-f,set);
    }
}

int main(){
    cin>>n;
    cin>>k1;
    for(int i=0;i<k1;++i)  cin>>s1[i];
    cin>>k2;
    for(int i=0;i<k2;++i)  cin>>s2[i];
    for(int i=0;i<k1;++i){
        int u=s1[i];
        int v=(s1[i]+1)%n;
        adj[u].push_back({v,i});
    }
    for(int i=0;i<k2;++i){
        int u=s2[i];
        int v=(s2[i]+1)%n;
        adj[u].push_back({v,i});
    }
    for(int i=1;i<n;++i){
        if(vis[i])  continue;
        dfs(i,i,0,1);
        dfs(i,i,1,2);
    }
    cout<<res[0]<<endl;
    cout<<res[1]<<endl;
}
