import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) s[i] = sc.nextInt();
        int q = sc.nextInt();
        int[] t = new int[q];
        for (int i = 0; i < q; i++) t[i] = sc.nextInt();

        int c = 0;
        for (int i = 0; i < q; i++) {
            if (n <= 0) break;
            if (t[i] < s[0] || t[i] > s[n - 1]) continue;
            if (s[(n - 1) / 2] == t[i]) c++;
            else if (s[(n - 1) / 2] < t[i]) n /= 2;
            else n -= n / 2;
        }
        System.out.println(c);
    }
}
