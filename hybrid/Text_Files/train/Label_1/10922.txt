#include <cstdio>

char board[105][105];
bool covered[105][105];
int tile[105][2][2];

bool check(int i, int j, int m, int t, int tp, bool ok) {
    if (i > m || j > m) return true;
    if (covered[i][j]) return false;
    if (ok) {
        int nt1 = tile[t][0][0], nt2 = tile[t][0][1];
        tile[t][0][0] = tile[t][1][0], tile[t][0][1] = tile[t][1][1];
        tile[t][1][0] = nt1, tile[t][1][1] = nt2;
    }
    if (tile[t][0][0] == board[i][j] && tile[t][0][1] == board[i][j + 1]) {
        if (!check(i + 1, j + 1, m, t, tp, false)) return false;
    }
    if (tile[t][1][0] == board[i][j] && tile[t][1][1] == board[i][j + 1]) {
        if (!check(i + 1, j + 1, m, t, tp, false)) return false;
    }
    if (!ok && tile[t][0][0] == board[i][j] && tile[t][1][0] == board[i + 1][j]) {
        if (!check(i + 1, j + 1, m, t, true, true)) return false;
    }
    if (!ok && tile[t][0][1] == board[i][j] && tile[t][1][1] == board[i + 1][j]) {
        if (!check(i + 1, j + 1, m, t, true, true)) return false;
    }
    covered[i][j] = covered[i][j + 1] = covered[i + 1][j] = covered[i + 1][j + 1] = true;
    if (board[i][j] == 0) {
        if (!check(i + 1, j, m, tp, 0, false)) return false;
        if (!check(i, j + 1, m, tp, 0, false)) return false;
    }
    return true;
}

bool check2(int m) {
    int i, j;
    for (i = 1; i <= m; i++) {
        for (j = 1; j <= m; j++) {
            if (covered[i][j]) continue;
            for (int t = 1; t <= m; t++) {
                if (tile[t][0][0] == board[i][j] && tile[t][0][1] == board[i][j + 1]) {
                    covered[i][j] = covered[i][j + 1] = covered[i + 1][j] = covered[i + 1][j + 1] = true;
                    if (check(i + 1, j + 1, m, t, t, false)) return true;
                }
                if (tile[t][1][0] == board[i][j] && tile[t][1][1] == board[i][j + 1]) {
                    covered[i][j] = covered[i][j + 1] = covered[i + 1][j] = covered[i + 1][j + 1] = true;
                    if (check(i + 1, j + 1, m, t, t, false)) return true;
                }
            }
            return false;
        }
    }
    return true;
}

int main() {
    int t, n, m, i, j;
    scanf("%d", &t);
    for (int cs = 1; cs <= t; cs++) {
        scanf("%d%d", &n, &m);
        for (i = 1; i <= n; i++) {
            scanf("%d %d", &tile[i][0][0], &tile[i][0][1]);
            scanf("%d %d", &tile[i][1][0], &tile[i][1][1]);
        }
        for (i = 1; i <= m; i++) {
            for (j = 1; j <= m; j++) {
                covered[i][j] = false;
                board[i][j] = 0;
            }
        }
        char ch;
        bool ok = true;
        for (i = 1; i <= m; i++) {
            for (j = 1; j <= m; j++) {
                scanf(" %c", &ch);
                if (ch >= 'A' && ch <= 'Z') board[i][j] = ch - 'A' + 1;
                if (ch >= 'a' && ch <= 'z') { board[i][j] = ch - 'a' + 1, ok = false; }
                if (board[i][j] == 0) ok = false;
                if (board[i][j] == board[m - i + 1][m - j + 1]) continue;
                ok = false;
            }
        }
        if (!ok) {
            puts("NO");
        } else if (check2(m)) {
            puts("YES");
        } else if (check(1, 1, m, 0, 0, false)) {
            puts("YES");
        } else {
            puts("NO");
        }
    }
    return 0;
}
