import java.util.*;

public class Main {
    static int n, m;
    static int[] color;
    static boolean[][] adj;

    static int solve() {
        int max = 0, ans = -1;
        int[] cnt = new int[100001];
        for (int i = 1; i <= n; i++) {
            cnt[color[i]]++;
        }
        for (int i = 1; i <= n; i++) {
            int cur = 0;
            for (int j = 1; j <= n; j++) {
                if (adj[i][j] && color[i] != color[j]) {
                    cur++;
                }
            }
            if (cur > max || (cur == max && color[i] < ans)) {
                max = cur;
                ans = color[i];
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        color = new int[n + 1];
        adj = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            color[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a][b] = adj[b][a] = true;
        }
        System.out.println(solve());
    }
}
