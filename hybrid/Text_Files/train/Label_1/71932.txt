from collections import defaultdict, Counter
class Team:
 def __init__(self, region_num):
  self.region_num = region_num
  self.members = set()

 def add_member(self, member):
  self.members.add(member)

 def is_valid(self):
  return len(self.members) == 2

 def __str__(self):
  return ' '.join(sorted(self.members))

class SchoolTeams:
 def __init__(self, n, m):
  self.n = n
  self.m = m
  self.students = []
  self.teams = {}

 def add_student(self, student):
  self.students.append(student)

 def build_teams(self):
  students_by_region = defaultdict(list)
  for student in self.students:
   students_by_region[student.region_num].append(student)

  for region_num in range(1, self.m + 1):
   students = students_by_region[region_num]
   if len(students) < 2:
    self.teams[region_num] = None
    continue

   students.sort(key=lambda s: s.score, reverse=True)
   team = Team(region_num)
   team.add_member(students[0].name)
   for student in students[1:]:
    if not team.is_valid():
     team.add_member(student.name)
    else:
     break

   self.teams[region_num] = team

 def output_teams(self):
  for region_num in range(1, self.m + 1):
   team = self.teams[region_num]
   if team is None or not team.is_valid():
    print('?')
   else:
    print(team)

class Student:
 def __init__(self, name, region_num, score):
  self.name = name
  self.region_num = region_num
  self.score = score

n, m = map(int, input().split())
teams = SchoolTeams(n, m)
for _ in range(n):
 line = input().split()
 student = Student(line[0], int(line[1]), int(line[2]))
 teams.add_student(student)

teams.build_teams()
teams.output_teams()
