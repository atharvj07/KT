def min_tickets_to_sell(q, queries):
    results = []
    for i in range(q):
        n, prices, xa, yb, k = queries[i]
        x, a = xa
        y, b = yb
        prices.sort(reverse=True)
        lcm = a * b // math.gcd(a, b)
        l, r = 1, n
        result = -1
        while l <= r:
            m = (l + r) // 2
            count_lcm = m // lcm
            count_a = m // a - count_lcm
            count_b = m // b - count_lcm
            total = sum(prices[:count_lcm]) * (x + y) // 100 + sum(prices[count_lcm:count_lcm + count_a]) * x // 100 + sum(prices[count_lcm + count_a:count_lcm + count_a + count_b]) * y // 100
            if total >= k:
                result = m
                r = m - 1
            else:
                l = m + 1
        results.append(result)
    return results

import math

q = int(input().strip())
queries = []
for _ in range(q):
    n = int(input().strip())
    prices = list(map(int, input().strip().split()))
    xa = list(map(int, input().strip().split()))
    yb = list(map(int, input().strip().split()))
    k = int(input().strip())
    queries.append((n, prices, xa, yb, k))

results = min_tickets_to_sell(q, queries)
for result in results:
    print(result)
