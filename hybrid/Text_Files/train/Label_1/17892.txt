#include <cstdio>

int main() {
    int n, k;
    scanf("%d %d", &n, &k);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            int x = (i < j) ? (k / (n - 1)) : ((k - ((n - 1) * (i == j))) % n);
            printf("%d%c", x, (j == n - 1) ? '\n' : ' ');
        }
    }
    return 0;
}
