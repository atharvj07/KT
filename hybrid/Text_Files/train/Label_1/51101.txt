import java.util.*;
import java.io.*;

class Main {
  static int[] solve(int n, int k, int d) {
    int[] ans = new int[n];
    for (int i = 0; i < n; i++) {
      Set<Integer> set = new HashSet<Integer>();
      for (int j = i; j < i + d && j < n; j++) {
        set.add(a[j]);
      }
      ans[i] = set.size();
    }
    return ans;
  }

  static int[] a;
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int n = sc.nextInt(), k = sc.nextInt(), d = sc.nextInt();
      a = new int[n];
      for (int i = 0; i < n; i++) {
        a[i] = sc.nextInt();
      }
      int[] r = solve(n, k, d);
      int res = r[0];
      for (int i = 1; i < n; i++) {
        res = Math.min(res, r[i]);
      }
      System.out.println(res);
    }
  }
}
