#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        s += s;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            if (s[i] == '<')
                a[i] = -1;
            else if (s[i] == '>')
                a[i] = 1;
            else
                a[i] = 0;
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[(i + j) % n];
                if (sum == 0)
                    cnt++;
                else if (sum < 0)
                    break;
            }
        }
        cout << cnt << endl;
    }
}
