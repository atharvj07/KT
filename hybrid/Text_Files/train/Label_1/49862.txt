import java.io.*;
import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        List<Polygon> polygons = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            List<Point> points = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                points.add(new Point(sc.nextLong(), sc.nextLong()));
            }
            polygons.add(new Polygon(points));
        }

        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            Polygon sum = polygons.get(l);
            for (int j = l + 1; j <= r; j++) {
                sum = sum.minkowskiSum(polygons.get(j));
            }
            System.out.println(sum.size());
        }
    }

    static class Point implements Comparable<Point> {
        long x, y;

        Point(long x, long y) {
            this.x = x;
            this.y = y;
        }

        // compare points on Y-axis, and if equal on X-axis
        @Override
        public int compareTo(Point o) {
            if (this.y != o.y) {
                return Long.compare(this.y, o.y);
            }
            return Long.compare(this.x, o.x);
        }
    }

    static class Polygon {
        List<Point> points;

        Polygon(List<Point> points) {
            Collections.sort(points);
            this.points = points;
        }

        // calculate Minkowski sum of two convex polygons
        Polygon minkowskiSum(Polygon other) {
            List<Point> sum = new ArrayList<>();
            int n = points.size();
            int m = other.points.size();
            int i = 0, j = 0;

            while (i < n && j < m) {
                if (points.get(i).compareTo(other.points.get(j)) <= 0) {
                    sum.add(points.get(i++));
                } else {
                    sum.add(other.points.get(j++));
                }
            }

            while (i < n) {
                sum.add(points.get(i++));
            }

            while (j < m) {
                sum.add(other.points.get(j++));
            }

            return new Polygon(sum);
        }

        int size() {
            return points.size();
        }
    }
}
