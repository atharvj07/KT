import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            String s = sc.next();
            char[] a = new char[26];
            int[] b = new int[26];
            for (int i = 0; i < n; i++) {
                int index = s.charAt(i) - 'a';
                a[index]++;
            }
            int need = 0;
            for (int i = 0; i < 26; i++) {
                b[i] = a[i] % k;
                if (b[i] != 0) {
                    need = 1;
                }
            }
            int min = 26;
            for (int i = 0; i < 26; i++) {
                if (b[i] != 0) {
                    min = i;
                }
            }
            char[] ans = new char[n];
            int index = 0;
            for (int i = 0; i < 26; i++) {
                int count = a[i] / k;
                while (count-- > 0) {
                    ans[index++] = (char) ('a' + i);
                }
            }
            if (need == 0) {
                System.out.println(s);
            } else {
                ans[index] = (char) ('a' + min);
                for (int i = 0; i < index; i++) {
                    System.out.print(ans[i]);
                }
                System.out.println();
            }
        }
    }
}
