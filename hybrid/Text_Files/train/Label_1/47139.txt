import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Day> days = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int m = Integer.parseInt(line[0]);
            int d = Integer.parseInt(line[1]);
            int v = Integer.parseInt(line[2]);
            int s = Integer.parseInt(line[3]);
            for (int j = 0; j < v; j++) {
                Day day = new Day(m, d + j, s);
                days.add(day);
            }
        }

        Collections.sort(days, (d1, d2) -> Integer.compare(d1.date, d2.date));

        int[] congestion = new int[366];
        for (Day day : days) {
            for (int i = 1; i <= 365; i++) {
                if (i < day.date - 1 || day.date + day.v <= i) {
                    continue;
                }
                congestion[i] = Math.max(congestion[i], day.s);
            }
        }

        int minCongestion = Arrays.stream(congestion).filter(c -> c > 0).min().orElse(0);
        System.out.println(minCongestion);
    }

    static class Day {
        int m;
        int d;
        int s;
        int v;
        int date;

        public Day(int m, int d, int s) {
            this.m = m;
            this.d = d;
            this.s = s;
            this.date = m * 30 + d;
        }
    }
}
