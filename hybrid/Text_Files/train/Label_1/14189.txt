#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
const int mxN=2e5+5;
#define FAST ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);

ll f[200001];

int main(){
    FAST;

    ll n; cin>>n;

    for(ll i=2; i*i<=200000; i++){
        if(f[i]) continue;
        for(ll j=i*i; j<=200000; j+=i){
            f[j]=i;
        }
    }

    vector<ll>primes, v;

    for(ll i=1; i<=200000; i++){
        if(f[i]==0) f[i]=i;
    }

    for(ll i=1; i<=n; i++){
        ll x; cin>>x;
        while(x>1){
            ll p=f[x]; v.push_back(p);
            while(x%p==0) x/=p;
        }
    }

    ll ans=1;
    map<ll, ll>freq;
    for(auto x:v) freq[x]++;

    for(auto x:freq){
        if(x.second&1){
            ans=ans*x.first;
        }
    }

    cout<<ans<<endl;
}
