#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    bool valid = true;
    if (s[0] != '8') valid = false;
    if (n != 11) valid = false;
    for (int i = 0; i < n; ++i) {
      if (s[i] < '0' || s[i] > '9') valid = false;
    }
    std::cout << (valid ? "YES" : "NO") << "\n";
  }
  return 0;
}
