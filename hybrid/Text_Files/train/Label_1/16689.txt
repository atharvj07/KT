#include <bits/stdc++.h>
using namespace std;
int ans, n, cnt;
char s[500005];
int main() {
  cin >> s;
  for (int i = 0; i < 500005; ++i) {
    if (!s[i]) {
      n = i;
      break;
    }
  }
  for (int i = 0; i < n; ++i) {
    if (s[i] == '[') { ++cnt; }
    if (s[i] == ']') { ++ans; --cnt; }
    if (cnt < 0) { break; }
  }
  if (cnt > 0) {
    cout << -1 << endl;
    return 0;
  }
  cout << ans * 2 << endl;
  return 0;
}
