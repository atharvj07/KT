import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int[][] max = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                max[i][j] = Math.max(max[i - 1][j], max[i][j - 1]);
                max[i][j] = Math.max(max[i][j], a[i - 1][j - 1]);
            }
        }
        int ansx = 0, ansy = 0;
        for (int x = k; x <= n - k + 1; x++) {
            for (int y = k; y <= m - k + 1; y++) {
                if (max[x][y] > max[ansx][ansy]) {
                    ansx = x;
                    ansy = y;
                }
            }
        }
        System.out.println(ansx + " " + ansy);
    }
}
