#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, f = 0;
  cin >> n;
  vector<string> b(n);
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= 2; j += 2) {
      if (b[i][j] == 'O' && b[i][j + 1] == 'O') {
        b[i][j] = b[i][j + 1] = '+';
        f = 1;
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) { cout << b[i] << endl; }
        return 0;
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
