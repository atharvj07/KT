import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int d = scanner.nextInt();
            int a;
            if (d == 1)
                a = 6;
            else
                a = d + d;
            System.out.println(a);
        }
    }
}
