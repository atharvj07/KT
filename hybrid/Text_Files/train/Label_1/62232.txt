n, m = map(int, input().split())
stage = [list(map(int, input().split())) for _ in range(n)]

def is_good_position(x, y, dx, dy):
    while 0 <= x < n and 0 <= y < m:
        if stage[x][y] == 1:
            return True
        x += dx
        y += dy
    return False

good_positions = 0
directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]  # Right, Left, Down, Up

for i in range(n):
    for j in range(m):
        if stage[i][j] == 0:
            for dx, dy in directions:
                if is_good_position(i + dx, j + dy, dx, dy):
                    good_positions += 1

print(good_positions)
