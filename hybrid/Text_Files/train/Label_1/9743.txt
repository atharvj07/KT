#include <iostream>
#include <algorithm>

int main() {
    int n; std::cin >> n;
    int arr[n];
    for (int i = 0; i < n; i++) std::cin >> arr[i];
    std::sort(arr, arr + n);
    int sum = 0;
    for (int i = 0; i < n - 1; i++) sum += arr[n - 1] - arr[i] - 1;
    std::cout << sum << std::endl;
    return 0;
}
