import sys
fin=sys.stdin
n=int(fin.readline())
a=[int(i) for i in fin.readline().split()]
e=[[]for i in range(n+1)]
for i in range(n-1):
 x,y=map(int,fin.readline().split())
 e[x].append(y)
 e[y].append(x)
b=sorted(a)
x=False
for i in b:
 c=a.count(i)
 if c>n-c:
  x=True
  break
if x:print(i)
else:print(b[n-1])
