#include <iostream>
#include <set>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> adj(n + 1);
  for (int i = 1; i < n; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  set<int> s;
  for (int i = 1; i <= n; i++) {
    if (adj[i].size() == 1) s.insert(i);
  }
  cout << s.size() << endl;
  return 0;
}
