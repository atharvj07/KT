#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 40010;
ll prefa[MAXN], prefb[MAXN], suffa[MAXN], suffb[MAXN];
ll ans;

int main(){
    int n, m, k;
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i ++){
        int x;
        cin >> x;
        prefa[i] = prefa[i - 1] + x;
    }
    for (int i = n; i >= 1; i --){
        suffa[i] = suffa[i + 1] + prefa[n] - prefa[i - 1];
    }
    for (int i = 1; i <= m; i ++){
        int x;
        cin >> x;
        prefb[i] = prefb[i - 1] + x;
    }
    for (int i = m; i >= 1; i --){
        suffb[i] = suffb[i + 1] + prefb[m] - prefb[i - 1];
    }
    for (int i = 1; i + k - 1 <= n; i ++){
        for (int j = 1; j + k - 1 <= m; j ++){
            if (prefa[i + k - 1] - prefa[i - 1] == k && prefb[j + k - 1] - prefb[j - 1] == k){
                ans += suffa[i + k] * suffb[j + k];
            }
        }
    }
    cout << ans << endl;
    return 0;
}
