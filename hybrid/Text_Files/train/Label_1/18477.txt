#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<long long> solve(vector<long long>& b) {
    long long n = b.size();
    vector<long long> a(2 * n);
    for (long long i = 0; i < n; i++) {
        long long l = i + 1;
        long long r = 2 * n - i;
        a[l] = b[i] / 2;
        a[r] = b[i] - a[l];
    }
    sort(a.begin() + 1, a.begin() + 2 * n);
    return a;
}

int main() {
    long long n;
    cin >> n;
    vector<long long> b(n / 2);
    for (long long& x : b) {
        cin >> x;
    }
    vector<long long> a = solve(b);
    for (long long x : a) {
        cout << x << " ";
    }
    cout << endl;
    return 0;
}
