n = int(input())
p = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
misp = dict()
misb = dict()
for i in range(n):
    if p[i] != i + 1:
        if p[i] not in misp:
            misp[p[i]] = []
        misp[p[i]].append(i)
    if p[i] != n - i:
        if i + 1 not in misp:
            misp[i + 1] = []
        misp[i + 1].append(i)
for i in range(n):
    if b[i] == 1:
        if p[i] not in misb:
            misb[p[i]] = []
        misb[p[i]].append(i)
    if p[i] != n - i:
        if i + 1 not in misb:
            misb[i + 1] = []
        misb[i + 1].append(i)
ans = 0
for i in misp:
    if i not in misb:
        ans += len(misp[i])
        continue
    if len(misp[i]) < len(misb[i]):
        ans += len(misp[i])
    else:
        ans += len(misb[i])
        for x in misb[i]:
            if x in misp[i]:
                misp[i].remove(x)
for i in misb:
    if i not in misp:
        ans += len(misb[i])
print(ans)
