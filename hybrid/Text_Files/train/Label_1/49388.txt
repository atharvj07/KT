import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] s = in.nextLine().toCharArray();
        int[] cnt = new int[s.length];
        for (int i = 0; i < s.length; i++) {
            if (i > 0 && s[i] == s[i - 1]) {
                cnt[i] = cnt[i - 1] + 1;
            } else {
                cnt[i] = 1;
            }
        }
        long even = 0;
        long odd = 0;
        for (int i = 0; i < s.length; i++) {
            even += cnt[i] / 2;
            odd += (cnt[i] + 1) / 2;
        }
        System.out.println(even + " " + odd);
    }
}
