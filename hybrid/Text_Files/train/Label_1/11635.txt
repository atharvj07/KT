#include <algorithm>
#include <cmath>
#include <cstdio>
#include <set>
#include <vector>
using namespace std;
const int MAXN = 1e5 + 5;
vector<pair<int, int>> points;
vector<double> dists;
int main() {
  int N, M;
  scanf("%d", &N);
  points.resize(N);
  dists.resize(MAXN);
  for (int i = 0; i < N; ++i) {
    scanf("%d%d", &points[i].first, &points[i].second);
  }
  scanf("%d", &M);
  for (int i = 0; i < M; ++i) {
    double d;
    set<double> uniqs;
    for (int j = 0; j < N; ++j) {
      scanf("%lf", &d);
      uniqs.insert(d);
    }
    int cnt = 0;
    for (set<double>::iterator it = uniqs.begin(); it != uniqs.end(); ++it) {
      dists[cnt++] = *it;
    }
    int L = 0, R = N - 1;
    int ans = 0;
    vector<pair<int, int>> res;
    while (L < R) {
      double d1 = dists[L] + dists[R];
      int x = (int)sqrt(d1);
      if (x * x == d1) {
        vector<pair<int, int>> newRes;
        int x1 = points[L].first - x, y1 = points[L].second - x;
        int x2 = points[L].first + x, y2 = points[L].second + x;
        int x3 = points[R].first - x, y3 = points[R].second - x;
        int x4 = points[R].first + x, y4 = points[R].second + x;
        if (x1 < 0 || y1 < 0 || x1 > 1e8 || y1 > 1e8) {
        } else if (x2 < 0 || y2 < 0 || x2 > 1e8 || y2 > 1e8) {
        } else if (x3 < 0 || y3 < 0 || x3 > 1e8 || y3 > 1e8) {
        } else if (x4 < 0 || y4 < 0 || x4 > 1e8 || y4 > 1e8) {
        } else {
          newRes.push_back(make_pair(x1, y1));
          newRes.push_back(make_pair(x2, y2));
          newRes.push_back(make_pair(x3, y3));
          newRes.push_back(make_pair(x4, y4));
        }
        L++, R--;
        ans += newRes.size();
        for (int j = 0; j < newRes.size(); ++j) { res.push_back(newRes[j]); }
      } else {
        double d2 = dists[L] + dists[R - 1];
        int y = (int)sqrt(d2);
        if (y * y == d2) {
          vector<pair<int, int>> newRes;
          int x1 = points[L].first - y, y1 = points[L].second - y;
          int x2 = points[L].first + y, y2 = points[L].second + y;
          int x3 = points[R - 1].first - y, y3 = points[R - 1].second - y;
          int x4 = points[R - 1].first + y, y4 = points[R - 1].second + y;
          if (x1 < 0 || y1 < 0 || x1 > 1e8 || y1 > 1e8) {
          } else if (x2 < 0 || y2 < 0 || x2 > 1e8 || y2 > 1e8) {
          } else if (x3 < 0 || y3 < 0 || x3 > 1e8 || y3 > 1e8) {
          } else if (x4 < 0 || y4 < 0 || x4 > 1e8 || y4 > 1e8) {
          } else {
            newRes.push_back(make_pair(x1, y1));
            newRes.push_back(make_pair(x2, y2));
            newRes.push_back(make_pair(x3, y3));
            newRes.push_back(make_pair(x4, y4));
          }
          L++, R -= 2;
          ans += newRes.size();
          for (int j = 0; j < newRes.size(); ++j) { res.push_back(newRes[j]); }
        } else {
          double d3 = dists[L + 1] + dists[R];
          int z = (int)sqrt(d3);
          if (z * z == d3) {
            vector<pair<int, int>> newRes;
            int x1 = points[L + 1].first - z, y1 = points[L + 1].second - z;
            int x2 = points[L + 1].first + z, y2 = points[L + 1].second + z;
            int x3 = points[R].first - z, y3 = points[R].second - z;
            int x4 = points[R].first + z, y4 = points[R].second + z;
            if (x1 < 0 || y1 < 0 || x1 > 1e8 || y1 > 1e8) {
            } else if (x2 < 0 || y2 < 0 || x2 > 1e8 || y2 > 1e8) {
            } else if (x3 < 0 || y3 < 0 || x3 > 1e8 || y3 > 1e8) {
            } else if (x4 < 0 || y4 < 0 || x4 > 1e8 || y4 > 1e8) {
            } else {
              newRes.push_back(make_pair(x1, y1));
              newRes.push_back(make_pair(x2, y2));
              newRes.push_back(make_pair(x3, y3));
              newRes.push_back(make_pair(x4, y4));
            }
            L += 2, R--;
            ans += newRes.size();
            for (int j = 0; j < newRes.size(); ++j) { res.push_back(newRes[j]); }
          } else {
            L++, R--;
          }
        }
      }
    }
    if (L == R) {
      double d1 = dists[L] + dists[L];
      int x = (int)sqrt(d1);
      if (x * x == d1) {
        vector<pair<int, int>> newRes;
        int x1 = points[L].first - x, y1 = points[L].second - x;
        int x2 = points[L].first + x, y2 = points[L].second + x;
        if (x1 < 0 || y1 < 0 || x1 > 1e8 || y1 > 1e8) {
        } else if (x2 < 0 || y2 < 0 || x2 > 1e8 || y2 > 1e8) {
        } else {
          newRes.push_back(make_pair(x1, y1));
          newRes.push_back(make_pair(x2, y2));
        }
        ans += newRes.size();
        for (int j = 0; j < newRes.size(); ++j) { res.push_back(newRes[j]); }
      }
    }
    printf("%d", ans);
    for (int j = 0; j < res.size(); ++j) {
      printf(" %d %d", res[j].first, res[j].second);
    }
    puts("");
  }
  return 0;
}
