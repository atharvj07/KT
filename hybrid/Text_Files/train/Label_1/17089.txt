#include <iostream>
#include <vector>
using namespace std;

const int MAX_SIZE = 1000;
const int MAX_VALUE = 1e9;
const int INF = 1e9;

int gcd(int a, int b) {
    if (a == 0)
        return b;
    return gcd(b % a, a);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int k = 0;
    for (int i = 1; i < n; i++) {
        if (gcd(a[i], a[i - 1]) == 1) continue;
        a.insert(a.begin() + i, 1);
        k++;
        i++;
    }

    cout << k << endl;
    for (int i = 0; i < n + k; i++) cout << a[i] << " ";
    cout << endl;

    return 0;
}
