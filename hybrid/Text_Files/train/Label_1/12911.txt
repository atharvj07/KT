#include <bits/stdc++.h>
using namespace std;
const int M = 1e9, MN = 1e5;
int n, a, b, c, f[MN + 10];
int main() {
    cin >> a >> b >> c >> n;
    vector<int> x(n);
    for (int i = 0; i < n; i++) { cin >> x[i]; }
    int lb = -M, rb = M;
    for (int i = 0; i < n; i++) {
        if (x[i] <= a) lb = max(lb, x[i]);
        if (x[i] >= a) rb = min(rb, x[i]);
    }
    if (lb > rb) {
        cout << 0 << endl;
    } else {
        for (int i = 0; i < n; i++) {
            if (x[i] >= lb && x[i] <= rb) f[x[i]]++;
        }
        int ans = 0;
        for (int i = 0; i <= a; i++) {
            if (i < lb || i > rb || (i >= b && i <= c))
                continue;
            ans += f[i];
        }
        cout << ans << endl;
    }
}
