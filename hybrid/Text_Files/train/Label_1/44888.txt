import java.util.*;
import java.io.*;

class Main {
    static int mod = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] x = new int[n+1];
        int[] y = new int[m+1];
        for (int i = 1; i <= n; i++) x[i] = in.nextInt();
        Arrays.sort(x);
        for (int i = 1; i <= m; i++) y[i] = in.nextInt();
        Arrays.sort(y);

        long ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i+1; j <= n; j++) {
                int dx = x[j] - x[i];
                for (int k = 1; k <= m; k++) {
                    for (int l = k+1; l <= m; l++) {
                        long dy = y[l] - y[k];
                        ans += dx * dy;
                        ans %= mod;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
