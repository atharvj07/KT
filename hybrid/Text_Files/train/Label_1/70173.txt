t = int(input())
for _ in range(t):
    n = int(input())
    s = input()
    i = 0
    j = n - 1
    ans = "YES"
    while i < j and ans == "YES":
        if s[i] != s[j]:
            if s[i] == chr(ord('a') - 1):
                ans = "NO"
            if s[j] == chr(ord('z') + 1):
                ans = "NO"
            if abs(ord(s[i]) - ord(s[j])) > 1:
                ans = "NO"
        i += 1
        j -= 1
    print(ans)
