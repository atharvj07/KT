import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.PriorityQueue;

public class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] s = br.readLine().split(" ");
    int d = Integer.parseInt(s[0]), n = Integer.parseInt(s[1]), m = Integer.parseInt(s[2]);
    PriorityQueue<long[]> pq = new PriorityQueue<>((a, b) -> (int) (a[1] - b[1]));
    for (int i = 0; i < m; i++) {
      s = br.readLine().split(" ");
      pq.offer(new long[] { Integer.parseInt(s[0]), Integer.parseInt(s[1]) });
    }
    long ans = 0, cur = 0, price = 0;
    while (!pq.isEmpty()) {
      long[] gas = pq.poll();
      if (cur + n >= d) {
        System.out.println(ans);
        return;
      } else if (cur >= gas[0]) {
        ans += (gas[0] - cur) * price;
        price = gas[1];
        cur = gas[0];
        n -= gas[0] - cur;
      } else {
        ans += (n - (cur + n - gas[0])) * price;
        cur = gas[0];
        price = gas[1];
        n = 0;
      }
    }
    if (cur + n >= d) System.out.println(ans);
    else System.out.println(-1);
  }
}
