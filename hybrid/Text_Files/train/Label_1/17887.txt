#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    string s;
    cin >> s;
    vector<int> freq(s.size() + 1, 0);
    freq[s.size()] = 1;
    for(int i = s.size() - 1; i >= 0; --i){
        freq[i] = freq[i + 1];
        if(s[i] == 'r') freq[i] += freq[i + 1];
    }
    ll ans = 0;
    for(int i = 0; i < s.size() - 3; ++i){
        if(s[i] == 'b') ans += freq[i + 4];
    }
    cout << ans << '\n';
    return 0;
}
