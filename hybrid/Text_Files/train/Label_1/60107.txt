import sys
input = lambda: sys.stdin.readline().rstrip()
mod = 998244353
inv2 = pow(2, mod-2, mod)

def fastpow(x, p):
    if p == 0:
        return 1
    if p == 1:
        return x
    val = fastpow(x, int(p/2))
    val = (val * val) % mod
    if p % 2 != 0:
        val = (val * x) % mod
    return val

n = int(input())
print(fastpow(inv2, n))
