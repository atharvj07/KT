#include <bits/stdc++.h>

using namespace std;

using ll = long long;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }
    ll mn = LLONG_MAX;
    pair<ll, ll> ans;
    for (ll i = 0; i < n - 1; i++) {
        for (ll j = i + 1; j < n; j++) {
            ll lcm = (a[i] * a[j]) / __gcd(a[i], a[j]);
            if (lcm < mn) {
                mn = lcm;
                ans = {i + 1, j + 1};
            }
        }
    }
    cout << ans.first << " " << ans.second << endl;
    return 0;
}
