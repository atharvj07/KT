import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long p = sc.nextLong();
        int ans = 0;
        while (n > 0) {
            if (p >= 0) {
                long max = 1L << (Long.numberOfLeadingZeros(n) + 2);
                if (p + max <= n) {
                    n -= p + max;
                    ans += 2;
                } else {
                    n -= p;
                    ans++;
                }
            } else {
                long max = 1L << (Long.numberOfLeadingZeros(n) + 1);
                if (p + max <= n) {
                    n -= p + max;
                    ans += 2;
                } else {
                    ans = -1;
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
