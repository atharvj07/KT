#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        if (n % 2 == 0)
            cout << "NO" << endl;
        else if (k * 2 < n)
            cout << "NO" << endl;
        else if (n % 2 != k % 2)
            cout << "NO" << endl;
        else
            cout << "YES" << endl;
    }
    return 0;
}
