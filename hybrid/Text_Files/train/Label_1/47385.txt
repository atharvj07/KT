import java.util.Scanner;

public class Mancala {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];

        // Read the number of stones in each hole
        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        int maxScore = 0;

        // Iterate over all holes
        for (int i = 0; i < 14; i++) {
            if (holes[i] > 0) {
                // Simulate the move and calculate the score
                int score = simulateMove(holes, i);

                // Update the maximum score
                maxScore = Math.max(maxScore, score);
            }
        }

        // Print the maximum score
        System.out.println(maxScore);
    }

    // Simulate the move and return the score
    private static int simulateMove(int[] holes, int startHole) {
        int score = 0;
        int stones = holes[startHole];
        holes[startHole] = 0;

        // Redistribute the stones
        int currentHole = startHole;
        while (stones > 0) {
            currentHole = (currentHole + 1) % 14;
            holes[currentHole]++;
            stones--;
        }

        // Collect stones from even holes
        for (int i = 0; i < 14; i++) {
            if (holes[i] % 2 == 0) {
                score += holes[i];
                holes[i] = 0;
            }
        }

        return score;
    }
}
