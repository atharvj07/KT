mod = int(1e9 + 7)

def fastpow(x: int, p: int) -> int:
    if p == 0:
        return 1
    if p == 1:
        return x

    val = fastpow(x, int(p / 2)) % mod
    val = (val * val) % mod

    if p % 2 != 0:
        return (val * x) % mod

    return val % mod

n = int(input())
a = list(map(int, input().split()))
dp = [[0 for i in range(n + 1)] for j in range(n + 1)]
dp[0][0] = 1

for i in range(1, n + 1):
    for j in range(i):
        dp[i][j] = dp[i - 1][j]
    if a[i - 1] % i == 0:
        dp[i][i] = (dp[i - 1][i - 1] + dp[i - 1][i]) % mod
    else:
        dp[i][i] = dp[i - 1][i]

ans = 0
for i in range(n + 1):
    ans = (ans + dp[n][i]) % mod

print(ans)
