import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long p = sc.nextLong();
        long ans = 1L;
        for (long i = 2L; i * i <= p; i++) {
            if (p % i == 0L) {
                ans = i;
                while (p % i == 0L) {
                    p /= i;
                }
            }
        }
        if (p > 1L) {
            ans = p;
        }
        System.out.println(ans);
    }
}
