n,q=map(int,input().split())
a=list(map(int,input().split()))
a.sort()
for i in range(1,n):a[i]+=a[i-1]
for i in range(q):
    b=int(input())
    l,r=0,n-1
    ans=-1
    while l<=r:
        mid=(l+r)//2
        if a[mid]>=b:
            ans=mid
            r=mid-1
        else:l=mid+1
    print(ans+1 if ans!=-1 else -1)
