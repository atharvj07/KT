#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n], b[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = i + 1; j < n; j++) {
      if (a[j] <= a[i]) cnt++;
    }
    b[i] = cnt;
  }
  for (int i = 0; i < n; i++) { cout << b[i] + 1 << " "; }
  cout << endl;
  return 0;
}
