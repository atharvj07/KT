import java.util.Arrays;
import java.util.Scanner;

public class DayPlan {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Job[] jobs = new Job[n];
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            jobs[i] = new Job(a, b);
        }

        scanner.close();

        Arrays.sort(jobs, (a, b) -> Integer.compare(b.reward, a.reward));

        int maxReward = 0;
        int daysLeft = m;
        for (Job job : jobs) {
            if (job.days <= daysLeft) {
                maxReward += job.reward;
                daysLeft -= job.days;
            } else {
                break;
            }
        }

        System.out.println(maxReward);
    }

    static class Job {
        int days;
        int reward;

        Job(int days, int reward) {
            this.days = days;
            this.reward = reward;
        }
    }
}
