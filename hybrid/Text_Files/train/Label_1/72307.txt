n, m = map(int, input().split())
table = [input() for _ in range(n)]

col_to_remove = 0
for j in range(m):
    remove_column = False
    for i in range(1, n):
        if table[i][j] < table[i-1][j] and all(table[i][k] == table[i-1][k] for k in range(j)):
            col_to_remove += 1
            remove_column = True
            break
    if remove_column:
        for i in range(n):
            table[i] = table[i][:j] + table[i][j+1:]
        break

print(col_to_remove)
