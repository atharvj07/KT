#include <iostream>
using namespace std;

int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  double mana = max(3 * a / 10.0, a - a / 250.0 * c);
  double vanb = max(3 * b / 10.0, b - b / 250.0 * d);
  if (mana > vanb)
    cout << "Misha" << endl;
  else if (mana < vanb)
    cout << "Vasya" << endl;
  else
    cout << "Tie" << endl;
  return 0;
}
