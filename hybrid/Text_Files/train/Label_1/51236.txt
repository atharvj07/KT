import java.util.Scanner;
import java.util.Stack;

class Main {
    private static class Pair {
        int id, type;
        String message;

        public Pair(int id, int type, String message) {
            this.id = id;
            this.type = type;
            this.message = message;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        Stack<Pair> tryStack = new Stack<>();
        Stack<Integer> throwStack = new Stack<>();
        for (int id = 1; id <= n; id++) {
            String line = in.nextLine().trim();
            if (line.startsWith("try")) tryStack.push(new Pair(id, 0, ""));
            else if (line.startsWith("catch(")) {
                String exceptionType = line.substring(6, line.indexOf(',')).trim();
                String message = line.substring(line.indexOf('"') + 1, line.lastIndexOf('"')).trim();
                int type = (exceptionType.charAt(0) >= 'a' ? 1 : 0);
                if (!tryStack.isEmpty() && tryStack.peek().type == type) {
                    System.out.println(message);
                    return;
                } else tryStack.push(new Pair(id, type, message));
            } else if (line.startsWith("throw")) {
                int type = (line.substring(6).trim().charAt(0) >= 'a' ? 1 : 0);
                throwStack.push(type);
            }
        }
        if (throwStack.pop() != (tryStack.isEmpty() ? 0 : tryStack.pop().type)) System.out.println("Unhandled Exception");
    }
}
