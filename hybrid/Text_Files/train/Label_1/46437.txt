import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = scanner.nextInt();
        }
        int cnt = 0;
        int max = h[0];
        for (int i = 1; i < n; i++) {
            if (h[i] >= max) {
                cnt++;
                max = h[i];
            }
        }
        System.out.println(cnt + 1);
    }
}
