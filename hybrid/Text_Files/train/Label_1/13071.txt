#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int q;
  cin >> q;
  while (q--) {
    ll h, n;
    cin >> h >> n;
    vector<ll> p(n + 1);
    for (int i = n; i; i--) cin >> p[i];
    if (n == h - 1 || (n == h && h != 1))
      cout << 0 << '\n';
    else if (n == h)
      cout << 1 << '\n';
    else if (p[n - 1] - p[n] > 2)
      cout << -1 << '\n';
    else
      cout << 2 << '\n';
  }
  return 0;
}
