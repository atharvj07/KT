#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

ll a[200005], XOR[200005], cnt[10000005];

int main() {
  ll n;
  cin >> n;
  for (ll i = 1; i <= n; i++) {
    cin >> a[i];
    XOR[i] = XOR[i - 1] ^ a[i];
  }
  ll ans = -1;
  for (ll i = 0; i <= n; i++) {
    if (cnt[XOR[i]]) {
      ans = max(ans, cnt[XOR[i]]);
    }
    cnt[XOR[i]] = i;
  }
  cout << ans << endl;
  return 0;
}
