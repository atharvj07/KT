#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  sort(a.begin(), a.end());
  long long ans = 0;
  for (int i = n - 1; i >= n - m * k; i--) {
    ans += a[i];
  }
  cout << ans << endl;
  for (int i = 0; i < k - 1; i++) {
    cout << n - m * (k - 1 - i) << " ";
  }
  cout << endl;

  return 0;
}
