#include <bits/stdc++.h>
using namespace std;

int query(int h, int n, vector<int> p) {
    vector<int> dp(h, -1); // dp[i] means the minimum crystals to safely come down to height i
    dp[h - 1] = 0;
    for (int i = h - 2; i >= 1; i--) {
        // try to fall from current height
        if (dp[i + 1] != -1 && i + 1 <= h - 1) { dp[i] = dp[i + 1]; }
        // try to land on any platform below
        for (auto x : p) {
            if (x > i && i >= x - 2 && (dp[x] != -1 || x == h - 1)) {
                if (dp[i] == -1) dp[i] = dp[x] + (x != h - 1);
                else dp[i] = min(dp[i], dp[x] + (x != h - 1));
            }
        }
    }
    return dp[1];
}

int main() {
    int q;
    cin >> q;
    for (int i = 1; i <= q; i++) {
        int h, n;
        cin >> h >> n;
        vector<int> p(n);
        for (int j = 0; j < n; j++) { cin >> p[j]; }
        cout << query(h, n, p) << endl;
    }
    return 0;
}
