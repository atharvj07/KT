#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<int> f(n + 1, 0);
    for (int i = 0; i < n; i++) {
        f[a[i]]++;
    }
    for (int i = 2; i <= n; i++) {
        f[i] += f[i - 1];
    }

    long long ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            ans += 1LL * (f[j] - f[i - 1]);
        }
    }

    cout << ans << endl;
}
