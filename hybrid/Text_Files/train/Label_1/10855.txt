#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const static int N = 1.5e5 + 10;
int a[N];
int main() {
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<ll> s(n);
  s[0] = a[0];
  for (int i = 1; i < n; i++) s[i] = s[i - 1] + a[i];
  ll ans = s[k - 1];
  int id = 0;
  for (int i = k; i < n; i++) {
    if (ans > s[i] - s[i - k]) {
      ans = s[i] - s[i - k];
      id = i - (k - 1);
    }
  }
  cout << id + 1 << endl;
  return 0;
}
