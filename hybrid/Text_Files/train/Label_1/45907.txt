import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[][] animals = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                animals[i][j] = scanner.nextInt();
            }
        }
        int[][] dp = new int[n][m * 2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m * 2; j++) {
                dp[i][j] = -1;
            }
        }
        int result = 0;
        for (int i = 0; i < m; i++) {
            result = Math.max(result, solve(i, i + k - 1, 0, n, m, k, animals, dp));
        }
        System.out.println(result);
    }

    private static int solve(int left, int right, int day, int n, int m, int k, int[][] animals, int[][] dp) {
        if (left < 0 || left >= m || right < 0 || right >= m || day >= n) {
            return 0;
        }
        if (dp[day][left] != -1) {
            return dp[day][left];
        }
        int result = 0;
        if (day % 2 == 0) {
            for (int i = left; i <= right; i++) {
                result = Math.max(result, solve(i, i + k - 1, day + 2, n, m, k, animals, dp) + animals[day][i]);
            }
            for (int i = left; i < m; i++) {
                result = Math.max(result, solve(i, Math.min(i + k - 1, m - 1), day + 1, n, m, k, animals, dp) + animals[day][i]);
            }
        } else {
            for (int i = left; i <= right; i++) {
                result = Math.max(result, solve(i, i + k - 1, day + 2, n, m, k, animals, dp) + animals[day][i]);
            }
            for (int i = 0; i <= right; i++) {
                result = Math.max(result, solve(Math.max(0, i - k + 1), i, day + 1, n, m, k, animals, dp) + animals[day][i]);
            }
        }
        dp[day][left] = result;
        return result;
    }
}
