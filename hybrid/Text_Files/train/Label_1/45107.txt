import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.util.List;
import java.util.ArrayList;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    int n = Integer.parseInt(st.nextToken());
    int forbidden = Integer.parseInt(st.nextToken());

    Node[] nodes = new Node[n];
    for (int i = 0; i < n; i++) {
      nodes[i] = new Node();
    }

    int[] stack = new int[n];
    int top = 0;

    for (int i = 0; i < n; i++) {
      st = new StringTokenizer(br.readLine());
      String type = st.nextToken();
      if (type.equals("set")) {
        int value = Integer.parseInt(st.nextToken());
        int cost = Integer.parseInt(st.nextToken());
        nodes[i].type = 's';
        nodes[i].cost = cost;
        nodes[i].value = value;
        if (value == forbidden) {
          nodes[i].forbidden = true;
        }
      } else if (type.equals("if")) {
        int value = Integer.parseInt(st.nextToken());
        nodes[i].type = 'i';
        nodes[i].value = value;
        stack[top++] = i;
      } else if (type.equals("end")) {
        top--;
        nodes[i].type = 'e';
        nodes[stack[top]].end = i;
      }
    }

    int result = dfs(nodes, 0, 0);
    System.out.println(result);
  }

  public static int dfs(Node[] nodes, int node, int cost) {
    if (nodes[node].type == 's') {
      if (nodes[node].forbidden) {
        return Integer.MAX_VALUE;
      } else {
        return nodes[node].cost;
      }
    } else if (nodes[node].type == 'i') {
      int left = dfs(nodes, node + 1, cost);
      if (nodes[node].value == nodes[nodes[node].end].value) {
        int right = dfs(nodes, nodes[node].end + 1, cost);
        return Math.min(left, right);
      } else {
        return left;
      }
    } else {
      return cost;
    }
  }

  public static class Node {
    public char type;
    public int value;
    public int cost;
    public boolean forbidden;
    public int end;

    public Node() {
      this.type = ' ';
      this.value = 0;
      this.cost = 0;
      this.forbidden = false;
      this.end = 0;
    }
  }
}
