def can_sort(n, a):
    sorted_a = sorted(a)
    pos = {v: i for i, v in enumerate(sorted_a)}
    cycles = 0
    visited = [False] * n

    for i in range(n):
        if visited[i] or a[i] == sorted_a[i]:
            continue
        cycle_size = 0
        j = i
        while not visited[j]:
            visited[j] = True
            j = pos[a[j]]
            cycle_size += 1
        if cycle_size > 0:
            cycles += cycle_size - 1

    return cycles % 2 == 0

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    print("YES" if can_sort(n, a) else "NO")
