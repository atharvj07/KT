#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <set>
#include <map>
#include <utility>
#include <stack>
#include <tuple>
#include <array>
#include <climits>
#include <sstream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, x;
    cin >> a >> b >> x;
    if (a == b) {
      cout << "YES" << '\n';
    } else if (x <= abs(a - b)) {
      cout << "YES" << '\n';
    } else if ((x - abs(a - b)) % 2 == 0) {
      cout << "YES" << '\n';
    } else {
      cout << "NO" << '\n';
    }
  }
  return 0;
}
