import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        if (d[0] < 10) {
            System.out.println("no");
        } else {
            int max = d[0] - 10;
            for (int i = 1; i < n; i++) {
                if (d[i] > max + 10) {
                    System.out.println("no");
                    return;
                }
                max = Math.max(max, d[i] - 10);
            }
            if (max < d[n - 1]) {
                System.out.println("no");
            } else {
                System.out.println("yes");
            }
        }
    }
}
