#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 4e5 + 5;
ll v[maxn], cnt[maxn];
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    int tot = n * 4;
    memset(cnt, 0, sizeof cnt);
    for (int i = 1; i <= tot; i++) cin >> v[i], cnt[v[i]]++;
    if (n % 2 == 0 && cnt[0] == 0) {
      cout << "NO\n";
      continue;
    }
    vector<int> len;
    for (int i = 0; i < maxn; i++) {
      if (cnt[i] % 4 == 0 && cnt[i]) {
        for (int j = 1; j <= cnt[i] / 4; j++) len.push_back(i);
      }
    }
    if (cnt[0] % 2 == 1) len.push_back(0);
    sort(len.begin(), len.end());
    if ((n % 2 == 0 && len.size() % 2 == 1) ||
        (n % 2 == 1 && len.size() % 2 == 0)) {
      cout << "NO\n";
      continue;
    }
    cout << "YES\n";
  }
  return 0;
}
