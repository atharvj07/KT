#include <iostream>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int a[n], b[m];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  int ans = (1 << 30);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { ans = min(ans, a[i] & b[j]); }
  }
  cout << ans;
}
