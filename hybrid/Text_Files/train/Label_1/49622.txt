import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int b = input.nextInt();
        int d = input.nextInt();
        String type = "7-type";
        int lastDigits = -1;
        if (d % 2 == 0) {
            type = "2-type";
            lastDigits = 1;
        }
        if (d % 3 == 0) {
            type = "3-type";
        }
        if (d % 11 == 0) {
            type = "11-type";
        }
        if (d % 6 == 0) {
            type = "6-type";
        }
        System.out.println(type);
        if (type.equals("2-type")) {
            System.out.println(lastDigits);
        }
    }
}
