import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        while (m-- > 0) {
            int t = sc.nextInt();
            if (t == 1) {
                int v = sc.nextInt() - 1, x = sc.nextInt();
                a[v] = x;
            } else if (t == 2) {
                int y = sc.nextInt();
                for (int i = 0; i < n; i++) a[i] += y;
            } else {
                System.out.println(a[sc.nextInt() - 1]);
            }
        }
    }
}
