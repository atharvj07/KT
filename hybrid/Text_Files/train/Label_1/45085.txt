import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        int C = sc.nextInt();
        int X = sc.nextInt();
        int Y = sc.nextInt();
        int minCost = Math.min(A * X + B * Y, 2 * C * Math.max(X, Y));
        System.out.println(minCost);
    }
}
