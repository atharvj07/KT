import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt();
    long[] ar = new long[n], ac = new long[m];
    for (int i = 0; i < n; i++) { ar[i] = sc.nextLong(); }
    for (int i = 0; i < m; i++) { ac[i] = sc.nextLong(); }
    long as = 0, bc = 0;
    for (int i = 0; i < n; i++) { as ^= ar[i]; }
    for (int i = 0; i < m; i++) { bc ^= ac[i]; }
    if (as != bc) { System.out.println("NO"); } else {
      System.out.println("YES");
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) { System.out.print((ar[i] ^ ac[j]) + " "); }
        System.out.println();
      }
    }
  }
}
