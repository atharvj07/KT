import java.util.Scanner;

public class ColorfulRibbon {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] colors = new int[n + 1];
        int[] colorfulness = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            colors[i] = i;
        }

        for (int i = 0; i < m; i++) {
            int type = scanner.nextInt();

            if (type == 1) {
                int l = scanner.nextInt();
                int r = scanner.nextInt();
                int x = scanner.nextInt();

                update(colors, colorfulness, l, r, x);
            } else {
                int l = scanner.nextInt();
                int r = scanner.nextInt();

                System.out.println(query(colors, colorfulness, l, r));
            }
        }
    }

    private static void update(int[] colors, int[] colorfulness, int l, int r, int x) {
        for (int i = l; i <= r; i++) {
            colorfulness[i] += Math.abs(x - colors[i]);
            colors[i] = x;
        }
    }

    private static int query(int[] colors, int[] colorfulness, int l, int r) {
        int sum = 0;
        for (int i = l; i <= r; i++) {
            sum += colorfulness[i];
        }
        return sum;
    }
}
