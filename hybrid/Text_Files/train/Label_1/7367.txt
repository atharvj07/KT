#include <iostream>
using namespace std;
int main() {
  long long n, a, b;
  cin >> n >> a >> b;
  if (n % (a + b) == 0) {
    cout << "YES\n" << n / (a + b) << " " << n / (a + b) << "\n";
    return 0;
  }
  if (n % a == 0 || n % b == 0) {
    cout << "YES\n" << n / a << " " << n / b << "\n";
    return 0;
  }
  cout << "NO\n";
  return 0;
}
