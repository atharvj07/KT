import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner stdIn = new Scanner(System.in);
        int N = stdIn.nextInt();
        int X = stdIn.nextInt();
        List<Integer> A = new ArrayList<>();
        for (int i = 0; i < X; ++i) { A.add(stdIn.nextInt()); }

        int Y = stdIn.nextInt();
        List<Integer> B = new ArrayList<>();
        for (int i = 0; i < Y; ++i) { B.add(stdIn.nextInt()); }

        int Z = stdIn.nextInt();
        List<Integer> C = new ArrayList<>();
        for (int i = 0; i < Z; ++i) { C.add(stdIn.nextInt()); }
        System.out.println(complements(A, C).size() + intersections(B, C).size());
    }

    public static List<Integer> complements(List<Integer> a, List<Integer> b) {
        List<Integer> ret = new ArrayList<>();
        for (Integer n : b) {
            if (!a.contains(n)) { ret.add(n); }
        }
        return ret;
    }

    public static List<Integer> intersections(List<Integer> a, List<Integer> b) {
        List<Integer> ret = new ArrayList<>();
        for (Integer n : a) {
            if (b.contains(n)) { ret.add(n); }
        }
        return ret;
    }
}
