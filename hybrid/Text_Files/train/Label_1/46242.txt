import java.util.Scanner;

public class YenWithdrawal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int operations = 0;
        while (N > 0) {
            int maxPowerOf6 = 0;
            while (Math.pow(6, maxPowerOf6) <= N) {
                maxPowerOf6++;
            }
            maxPowerOf6--;

            int maxPowerOf9 = 0;
            while (Math.pow(9, maxPowerOf9) <= N) {
                maxPowerOf9++;
            }
            maxPowerOf9--;

            if (Math.pow(6, maxPowerOf6) > Math.pow(9, maxPowerOf9)) {
                N -= Math.pow(6, maxPowerOf6);
            } else {
                N -= Math.pow(9, maxPowerOf9);
            }
            operations++;
        }

        System.out.println(operations);
    }
}
