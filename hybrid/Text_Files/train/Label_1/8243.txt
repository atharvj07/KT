#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long p[n + 2], pos[n + 2];
  for (int i = 1; i <= n; i++) {
    cin >> p[i] >> pos[i];
  }
  sort(p + 1, p + n + 1);
  for (int i = 1; i <= n; i++) {
    if (pos[i] != p[i]) {
      cout << "rated\n";
      return 0;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (p[i] < p[pos[i]]) {
      cout << "maybe\n";
      return 0;
    }
  }
  cout << "unrated\n";
  return 0;
}
