n,x,y=map(int,input().split())
a=[int(i) for i in input().split()]
a=[-1000000009]+a
l=[]
for i in range(x+y+1): l.append([i,a[i]])
l.sort(key=lambda s: s[1])
for i in range(n-x-y):
    if a[i+x+y+1]<l[0][1]: print(i+1);break
    for j in l:
        if a[i+x+y+1]<=j[1]:
            l.pop(0)
            l.append([i+x+y+1,a[i+x+y+1]])
            break
