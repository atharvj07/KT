#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> freq(26, 0);
    for (char c : s) freq[c - 'a']++;
    sort(freq.rbegin(), freq.rend());
    if (freq[0] <= n / 2) {
        cout << "YES\n";
        cout << s << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
