import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = input.nextInt();
            }
        }

        int[][] b = new int[n][m];
        for (int i = 0; i < n; i++) {
            int[] row = Arrays.copyOf(a[i], m);
            Arrays.sort(row);
            for (int j = 0; j < m; j++) {
                b[i][j] = Arrays.binarySearch(row, a[i][j]) + 1;
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(b[i][j] + " ");
            }
            System.out.println();
        }
    }
}
