import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        for (int i = 0; i < s.length(); i++) {
            System.out.println(s.charAt(i) >= s.charAt(Math.max(i - 1, 0)) ? "Mike" : "Ann");
        }
    }
}
