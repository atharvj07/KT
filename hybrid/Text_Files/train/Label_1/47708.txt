import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int x = input.nextInt();
        int a = 0;
        char b = 'A';
        if (x % 4 == 3) {
            a = 1;
            b = 'B';
        } else if (x % 4 == 2) {
            a = 2;
            b = 'C';
        } else if (x % 4 == 0) {
            b = 'D';
        }
        System.out.println(a + " " + b);
    }
}
