#include <bits/stdc++.h>

using namespace std;

long long a[100005], b[100005];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a, a + n);
  sort(b, b + m, greater<long long>());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] >= b[i]) {
      break;
    }
    ans += b[i] - a[i];
  }
  cout << ans << endl;
  return 0;
}
