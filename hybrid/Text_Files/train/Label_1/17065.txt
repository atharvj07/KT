#include <bits/stdc++.h>
using namespace std;
const int N = 1e6+1;
int dp[N];
string s;
int main() {
    int n; cin >> n >> s;
    int bal = 0;
    priority_queue<int> q; // max heap
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            bal++;
        } else {
            if (bal > 0) {
                q.push(bal * 2);
                bal--;
            } else {
                q.push(1);
            }
        }
    }
    int sum = 0, ans = -1;
    while (!q.empty()) {
        int x = q.top(); q.pop();
        sum += x;
        if (bal > 0) {
            x += q.top(); q.pop();
            bal--;
        } else break;
        if (bal == 0) ans = sum;
    }
    cout << ans << '\n';
    return 0;
}
