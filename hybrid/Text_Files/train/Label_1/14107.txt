#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int t;
    cin >> t;
    while (t--) {
        ll n, x;
        cin >> n >> x;
        vector<ll> a(n);
        for (ll i = 0; i < n; i++) cin >> a[i];
        ll ans = LLONG_MAX;
        for (int i = 0; i < n; i++) {
            ll l = 0, r = LLONG_MAX / a[i];
            while (l <= r) {
                ll mid = (l + r) / 2;
                if (mid * a[i] > x)
                    r = mid - 1;
                else
                    l = mid + 1;
            }
            ll val = 1LL * ((x + a[i] - 1) / a[i]);
            ans = min(ans, val);
        }
        cout << ans << '\n';
    }
    return 0;
}
