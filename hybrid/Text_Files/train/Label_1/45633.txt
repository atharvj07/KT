import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        long k = sc.nextLong();

        long t = (k / 2) % 3;
        long n = (k + 1) / 2;
        long res = 0;

        if (t == 1) res = b - c;
        if (t == 2) res = c - a;
        if (t == 0) res = a - b;
        res *= n;
        if (Math.abs(res) >= 1e18) {
            System.out.println("Unfair");
        } else {
            System.out.println(res);
        }
    }
}
