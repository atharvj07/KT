import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        String s = sc.next(), t = sc.next();
        int l = lcm(n, m);
        int[] ns = new int[l], nt = new int[l];
        for (int i = 0; i < n; i++) {
            ns[i * l / n]++;
        }
        for (int i = 0; i < m; i++) {
            nt[i * l / m]++;
        }
        boolean ok = true;
        for (int i = 0; i < l; i++) {
            if (ns[i] != 0 && nt[i] != 0) {
                ok = false;
                break;
            }
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dp[i], -1);
        }
        dp[0][0] = 0;
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                if (i > 0) {
                    if (dp[i - 1][j] != -1) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][j] + l / n);
                    }
                }
                if (j > 0) {
                    if (dp[i][j - 1] != -1) {
                        dp[i][j] = Math.max(dp[i][j], dp[i][j - 1] + l / m);
                    }
                }
            }
        }
        System.out.println(dp[n][m] == -1 ? -1 : dp[n][m]);
    }

    public static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    public static int lcm(int a, int b) {
        return a / gcd(a, b) * b;
    }
}
