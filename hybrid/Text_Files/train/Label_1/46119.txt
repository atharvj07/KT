import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++)
            board[i] = in.next().toCharArray();

        boolean[][] visited = new boolean[n][n];
        int[] dx = new int[]{-1, 0, 1, 0};
        int[] dy = new int[]{0, 1, 0, -1};
        int[] bx = new int[]{0, 0, 1, 1, 2};
        int[] by = new int[]{0, 1, 0, 1, 1};

        boolean ok = false;
        outer:
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !visited[i][j]) {
                    ok = true;
                    visited[i][j] = true;
                    Queue<int[]> q = new LinkedList<>();
                    q.offer(new int[]{i, j});
                    while (!q.isEmpty()) {
                        int[] curr = q.poll();
                        int x = curr[0], y = curr[1];
                        for (int k = 0; k < 4; k++) {
                            int nx = x + dx[k], ny = y + dy[k];
                            if (nx >= 0 && nx < n && ny >= 0 && ny < n && !visited[nx][ny] && board[nx][ny] == '.') {
                                visited[nx][ny] = true;
                                q.offer(new int[]{nx, ny});
                            }
                        }
                        boolean good = true;
                        for (int k = 0; k < 5; k++) {
                            int nx = x + bx[k], ny = y + by[k];
                            if (nx < 0 || nx >= n || ny < 0 || ny >= n || board[nx][ny] == '#') {
                                good = false;
                                break;
                            }
                        }
                        if (good) {
                            for (int k = 0; k < 5; k++) {
                                int nx = x + bx[k], ny = y + by[k];
                                visited[nx][ny] = true;
                            }
                            continue outer;
                        }
                    }
                    ok = false;
                    break outer;
                }
            }
        }

        System.out.println(ok ? "YES" : "NO");
    }
}
