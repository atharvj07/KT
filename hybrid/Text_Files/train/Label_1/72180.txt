for _ in range(int(input())):
    n = int(input())
    a = list(input())
    b = list(input())
    a0 = b0 = a1 = b1 = 0
    for i in range(n):
        a0 += a[i] == '0'
        b0 += b[i] == '0'
        a1 += a[i] == '1'
        b1 += b[i] == '1'
    print('YES' if a0 == b0 and a1 == b1 else 'NO')
