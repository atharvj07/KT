#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k, sum = 0;
    vector<int> a;
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a.push_back(x);
        sum += x;
    }
    if (sum < k) {
        cout << -1 << endl;
        return 0;
    }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int x = a[i];
        while (x > 0) {
            if (k > 0) {
                ans += x * x;
                x--;
                k--;
            } else {
                break;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
