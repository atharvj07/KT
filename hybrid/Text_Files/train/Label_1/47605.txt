import java.util.*;

class Main {
  static class CharStack {
    private char[] chars;
    private int top;

    public CharStack(int n) {
      this.chars = new char[n];
      this.top = -1;
    }

    public int size() { return top + 1; }

    public boolean isEmpty() { return top == -1; }

    public char pop() {
      if (isEmpty()) return 0;
      return chars[top--];
    }

    public void push(char c) { chars[++top] = c; }

    public char peek() { return chars[top]; }
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.nextLine();
    in.close();

    // Initialize the stack
    CharStack stack = new CharStack(s.length());

    // Push the opening brackets to the stack
    for (int i = 0; i < s.length(); i++) {
      char c = s.charAt(i);
      if (c == '(') stack.push(c);
    }

    // Add the closing brackets
    StringBuilder sb = new StringBuilder();
    while (!stack.isEmpty()) {
      char c = stack.pop();
      sb.append('(');
      sb.append(')');
    }

    // Shift the brackets to the beginning
    for (int i = 0; i < s.length() - sb.length(); i++) {
      sb.append(')');
      sb.append('(');
    }

    // Print the result
    System.out.println(sb.toString());
  }
}
