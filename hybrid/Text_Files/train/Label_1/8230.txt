#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, s, k;
        cin >> n >> s >> k;
        vector<bool> closed(n + 1);
        while (k--) {
            int x;
            cin >> x;
            closed[x] = true;
        }
        long long ans = 0;
        if (!closed[s])
            ans += 0;
        else {
            ans += min(s - 1, n - s);
            if (s == 1 || s == n)
                ans += 1;
        }
        cout << ans << endl;
    }
}
