#include<iostream>
#include<vector>
#include<functional>

void solve() {
    int n; std::cin >> n;
    std::vector<int> vec(n);
    for (int i = n - 1; i >= 0; --i) {
        vec[i] = 9;
        if (i + 1 == n) { vec[i] = 8; }
    }
    for (int x : vec) { std::cout << x; }
    std::cout << '\n';
}
int main() {
    int t; std::cin >> t;
    while (t--) { solve(); }
    return 0;
}
