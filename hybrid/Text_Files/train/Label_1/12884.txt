#include<iostream>
#include <algorithm>
using namespace std;
const int MX = 1e5;
int a[MX+5], freq[MX+5];
int main(){
    int n, last = -1, m = 0;
    cin >> n;
    for(int i=0;i<n;i++) cin >> a[i], freq[a[i]]++, m = max(m,a[i]);
    for(int i=0;i<=m;i++){
        if(last<i){
           if(freq[i]==0) { cout << i << endl; return 0;  }
           last = i + freq[i] - 1;
        }
        else{
            freq[i] = max(0,freq[i]-1);
        }
    }
    cout << -1 << endl;
    return 0;
}
