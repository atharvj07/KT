#include<bits/stdc++.h>
using namespace std;
#define ll long long int
const ll mod=1000000007;
ll POW(ll x,ll y)
{
    x=x%mod;
	ll res=1;
	while(y>0)
	{
	    if(y%2==1)
	    {
	        res=(res*x)%mod;
	    }
	    y/=2;
	    x=(x*x)%mod;
	}
	return res;
}
ll GCD(ll a, ll b)
{
    if (b == 0)
        return a;
    return GCD(b, a % b);
}
int main()
{
    ll n;
    cin>>n;
    vector<ll>v(n+1);
    for(int i=1;i<=n;i++)
    {
        cin>>v[i];
    }
    vector<ll>u;
    for(int i=1;i<=n;i++)
    {
        if(v[i]>=2){
            u.push_back(v[i]);
            continue;
        }
        ll w=0;
        for(ll j=2 ; j*j<=v[i] ; j++)
        {
            if(v[i]%j==0)
            {
                bool f=0;
                ll p=2;
                while(1)
                {
                    if(POW(j,p)>v[i])
                    {
                        f=1;
                        break;
                    }
                    else
                    {
                        if(v[i]%POW(j,p)==0)
                        {
                            v[i]=v[i]/POW(j,p);
                            p++;
                        }
                        else
                        {
                            break;
                        }
                    }
                }
                if(f==0)
                {
                    w=1;
                    u.push_back(j);
                    break;
                }
            }
        }
        if(v[i]>1)
        {
            u.push_back(v[i]);
        }
        else if(w==0)
        {
            u.push_back(2);
        }
    }
    ll g=u[0];
    for(ll i=1;i<u.size();i++)
    {
        g=GCD(g,u[i]);
    }
    if(g==1)
    {
        for(ll i=0;i<u.size();i++)
        {
            cout<<u[i]<<" ";
        }
        for(ll i=u.size();i<n;i++)
        {
            cout<<2<<" ";
        }
    }
    else
    {
        u.push_back(2);
        sort(u.begin(),u.end());
        for(ll i=0;i<u.size();i++)
        {
            cout<<u[i]<<" ";
        }
    }
    cout<<endl;
    return 0;
}
