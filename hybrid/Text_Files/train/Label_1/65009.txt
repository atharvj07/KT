n, m, k = map(int, input().split())
p = list(map(int, input().split()))
s = list(map(int, input().split()))
c = list(map(int, input().split()))
chosen = [False] * (n + 1)
for i in c:
    chosen[i] = True
max_p = [0] * (m + 1)
for i in range(1, n + 1):
    if p[i - 1] > max_p[s[i - 1]]:
        max_p[s[i - 1]] = p[i - 1]
made = k - sum(chosen)
ans = 0
for i in range(1, m + 1):
    if not chosen[max_p.index(max_p[i])]:
        made -= 1
        ans += 1
    if made == 0:
        break
print(ans)
