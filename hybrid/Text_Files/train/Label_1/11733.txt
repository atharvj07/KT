#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int fixed = 0;
    for (int i = 0; i < n; i++) if (a[i] == i) fixed++;
    int mx = fixed;
    int pos1 = -1, pos2 = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] == i) continue;
        if (a[a[i]] == i) {
            if (i < a[i]) mx = max(mx, fixed + 2);
            else mx = max(mx, fixed + 1);
            pos1 = i;
            pos2 = a[i];
            break;
        }
    }
    cout << mx << endl;
    if (mx > fixed) cout << pos1 << " " << pos2 << endl;
    return 0;
}
