import sys

input = sys.stdin.readline
n = int(input())
s = [input()[:-1] for i in range(n)]
cnt = [0] * n
for i in range(n):
    x, y = 0, 0
    for j in s[i]:
        if j == '(': x += 1
        if j == ')': y += 1
    cnt[i] = y - x
ans = 0
for i in range(n):
    for j in range(n):
        if cnt[i] == cnt[j]: ans += 1
print(ans)
