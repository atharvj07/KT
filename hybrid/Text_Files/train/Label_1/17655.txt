#include <iostream>
#include <vector>
#include <sstream>

using namespace std;

vector<double> split(const string& s) {
    vector<double> res;
    istringstream iss(s);
    string token;
    while (getline(iss, token, '.')) {
        if (!token.empty()) {
            double num;
            stringstream(token) >> num;
            res.push_back(num);
        }
    }
    return res;
}

int main() {
    string s;
    getline(cin, s);
    
    double sum = 0.0;
    for (int i = 0; i < s.length(); i += 11) {
        vector<double> prices = split(s.substr(i, 11));
        sum += prices[0] * 100 + prices[1];
    }
    
    int dollars = static_cast<int>(sum / 100);
    int cents = static_cast<int>(sum) % 100;
    
    cout << dollars << "." << (cents < 10 ? "0" : "") << cents << endl;
    
    return 0;
}
