import java.util.Scanner;

public class Nightmare {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[][] strings = new int[m][n];
        for (int i = 0; i < m; i++) {
            int si = scanner.nextInt();
            int fi = scanner.nextInt();
            for (int j = 0; j < fi; j++) {
                strings[i][j] = si;
            }
            for (int j = fi; j < n; j++) {
                strings[i][j] = 1 - si;
            }
        }

        int maxSum = 0;
        int count = 0;
        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                for (int k = j + 1; k < m; k++) {
                    int sum = hammingDistance(strings[i], strings[j]) +
                            hammingDistance(strings[j], strings[k]) +
                            hammingDistance(strings[k], strings[i]);
                    if (sum > maxSum) {
                        maxSum = sum;
                        count = 1;
                    } else if (sum == maxSum) {
                        count++;
                    }
                }
            }
        }

        System.out.println(count);
    }

    private static int hammingDistance(int[] a, int[] b) {
        int distance = 0;
        for (int i = 0; i < a.length; i++) {
            if (a[i] != b[i]) {
                distance++;
            }
        }
        return distance;
    }
}
