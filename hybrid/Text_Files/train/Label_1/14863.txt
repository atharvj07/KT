#include<bits/stdc++.h>
using namespace std;
int main(){
	int n; cin>>n;
	vector<int>a(n),b(n);
	for(int &i:a) cin>>i;
	for(int &i:b) cin>>i;
	
	vector<int>mn(n+1,INT_MAX),mx(n+1,INT_MIN);
	for(int i=n-1;i>=0;i--) mn[i]=min(mn[i+1],b[i]), mx[i]=max(mx[i+1],a[i]);
	
	int cnt=0;
	for(int i=0;i<n;i++) for(int j=i;j<n;j++) if(mx[i]==mn[j]) cnt++;
	cout<<cnt<<'\n';
}
