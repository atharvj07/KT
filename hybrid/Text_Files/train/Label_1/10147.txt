#include <bits/stdc++.h>
using namespace std;
string s[2510];
int n, k, a[2510][2510], mt[5100];
int main() {
    cin >> k >> n;
    for (int i = 1; i <= k; i++) {
        cin >> s[i];
        for (int j = 0; j < n; j++) {
            a[i][j] = s[i][j] - 'a';
        }
    }
    for (int i = 1; i <= k; i++) {
        for (int j = i + 1; j <= k; j++) {
            int cnt = 0, aa = -1, bb = -1;
            for (int p = 0; p < n; p++) {
                if (a[i][p] != a[j][p]) {
                    if (cnt == 1) {
                        aa = a[i][p];
                        bb = a[j][p];
                    }
                    cnt++;
                }
            }
            if (cnt == 2 && aa < bb) mt[aa] ^= (1 << j);
        }
    }
    for (int i = 0; i < 5100; i++) {
        if (mt[i] == 0) continue;
        for (int j = 0; j < 5100; j++) {
            if ((mt[i] >> j) & 1 == 0) continue;
            if (mt[j] == 0) continue;
            for (int p = 0; p < 5100; p++) {
                if ((mt[j] >> p) & 1 == 0) continue;
                if (mt[p] == 0) continue;
                cout << -1 << endl;
                return 0;
            }
        }
    }
    for (int i = 0; i < 26; i++) {
        if (mt[i] == 0) {
            for (int j = 0; j < n; j++) {
                cout << char(i + 'a');
            }
            cout << endl;
            return 0;
        }
    }
}
