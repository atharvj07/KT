#include<bits/stdc++.h>
using namespace std;
double pi = acos(-1.);
int main(){
    double R, x1, y1, x2, y2;
    cin >> R >> x1 >> y1 >> x2 >> y2;
    double dis = hypot(x1 - x2, y1 - y2);
    double a1 = pi * R * R;
    double r1 = R;
    if (dis <= 2 * R)
    {
        double a2 = 0., r2 = 0.;
        double low = 0., high = dis / 2.;

        while (fabs(r2 - r1) > 1e-6)
        {
            r1 = r2;
            double mid = (low + high) / 2.;
            double ang = acos(mid / r1);
            double seg = r1 * ang;
            a2 = 2. * mid * seg + r1 * r1 * (2 * pi - 2. * ang);
            if (a2 > a1) high = mid;
            else low = mid;
        }

        double x = (x1 + x2) / 2.;
        double y = (y1 + y2) / 2.;
        double theta = atan2(y2 - y, x2 - x);
        theta -= acos(r2 / r1);
        x = x + r2 * cos(theta);
        y = y + r2 * sin(theta);
        cout << fixed << setprecision(10) << x << " " << y << " " << r2 << endl;
    }
    else
    {
        cout << x1 << " " << y1 << " " << R << endl;
    }
    return 0;
}
