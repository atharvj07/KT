#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;

int solve(ll s, ll n) {
    if (n == 1) {
        return s;
    } else {
        if (s > 9) {
            return 9 * solve(s / 11, n - 1);
        } else {
            return 10 * solve(s / 11, n - 1);
        }
    }
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll s, n;
        cin >> s >> n;
        ll tmp = solve(s, n);
        vector<int> res;
        while (tmp > 0) {
            res.push_back(tmp % 11);
            tmp /= 11;
        }
        sort(res.begin(), res.end(), greater<int>());
        for (int i = 0; i < res.size(); i++) {
            cout << res[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
