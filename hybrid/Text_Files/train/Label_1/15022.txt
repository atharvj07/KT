#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,i,x,y,z,a[105]={0},b[105]={0},c[6]={0},ans=1e9;
    cin>>n;
    for(i=0;i<n;i++){
        cin>>x;
        a[i]=x;
        c[x]++;
    }
    for(i=0;i<n;i++){
        cin>>x;
        b[i]=x;
        c[x]++;
    }
    for(i=1;i<=5;i++){
        if(c[i]%2){
            cout<<-1;
            return 0;
        }
        c[i]/=2;
        ans=min(ans,c[i]);
    }
    cout<<ans;
    return 0;
}
