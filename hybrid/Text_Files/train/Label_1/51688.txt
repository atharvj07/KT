import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            String[] words = new String[n];
            for (int i = 0; i < n; i++) {
                words[i] = sc.next();
            }
            int start = 0;
            int end = n - 1;
            while (start < end) {
                int len = words[start].length() + words[end].length();
                if (len == 5) {
                    break;
                } else if (len < 5) {
                    start++;
                } else {
                    end--;
                }
            }
            System.out.println(start + 1);
        }
    }
}
