import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long d;
        int n, m;
        while ((d = sc.nextInt()) != 0) {
            n = sc.nextInt();
            m = sc.nextInt();
            long[] dis = new long[n];
            for (int i = 0; i < n - 1; i++) {
                dis[i] = sc.nextInt();
            }
            long ans = 0;
            for (int i = 0; i < m; i++) {
                int k = sc.nextInt();
                long min = d;
                for (int j = 0; j < n - 1; j++) {
                    if (dis[j] >= k) {
                        min = Math.min(min, dis[j] - k);
                    }
                    if (d - dis[j] >= k) {
                        min = Math.min(min, d - dis[j] + k);
                    }
                }
                ans += min;
            }
            System.out.println(ans);
        }
    }
}
