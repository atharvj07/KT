import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] net = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    net[j][k] = sc.nextInt();
                }
            }
            System.out.println(isValidNet(net) ? "true" : "false");
        }
    }

    private static boolean isValidNet(int[][] net) {
        // Check if the net is a cube
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                if (net[i][j] == 0) {
                    return false;
                }
            }
        }

        // Check if the sum of the opposite faces is always 7
        for (int i = 0; i < 5; i++) {
            if (net[i][0] + net[i][4] != 7) {
                return false;
            }
            if (net[0][i] + net[4][i] != 7) {
                return false;
            }
        }

        // Check if there are no duplicate numbers
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                for (int k = i + 1; k < 5; k++) {
                    for (int l = j + 1; l < 5; l++) {
                        if (net[i][j] == net[k][l]) {
                            return false;
                        }
                    }
                }
            }
        }

        return true;
    }
}
