#include<bits/stdc++.h>

#define vi(x) vector<x>

#define deb(x) cerr<<#x<<" = "<<x<<"\n"

#define tiii tuple<int,int,int>
typedef long long ll;
const int mod = 1e9+7;
const ll INF = 1e18L + 5;
int n;
int a[100002],b[100002],c[100002];

int main(){
    cin>>n;
    if(n<=2){cout<<-1;return 0;}
    for(int i=1;i<=n;++i) a[i]=i%3,b[i]=(3-a[i]%3)%3,c[i]=(a[i]+b[i])%3;
    if(n%3){
        int p=(n/3)+1;
        for(int i=1;i<=n;++i){
            if(i%3==1) a[i]=p,b[i]=p-1,c[i]=p+1;
            else if(i%3==2) a[i]=p+1,b[i]=p,c[i]=p-1;
            else a[i]=p-1,b[i]=p+1,c[i]=p;
        }
    }
    for(int i=1;i<=n;++i) cout<<a[i]<<" ";
    cout<<"\n";
    for(int i=1;i<=n;++i) cout<<b[i]<<" ";
    cout<<"\n";
    for(int i=1;i<=n;++i) cout<<c[i]<<" ";
    return 0;
}
