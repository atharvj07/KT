MOD = 10**9 + 7

n = int(input())
dolls = [tuple(map(int, input().split())) for _ in range(n)]

dolls.sort(key=lambda x: (x[1], -x[0]))

dp = [1] * n
min_extra_space = [float('inf')] * n
min_extra_space[0] = dolls[0][1]

for i in range(1, n):
    for j in range(i):
        if dolls[j][0] <= dolls[i][1]:
            extra_space = min_extra_space[j] + dolls[i][1] - dolls[j][0]
            if extra_space < min_extra_space[i]:
                min_extra_space[i] = extra_space
                dp[i] = dp[j]
            elif extra_space == min_extra_space[i]:
                dp[i] += dp[j]
                dp[i] %= MOD

print(sum(dp[i] for i in range(n) if min_extra_space[i] == min(min_extra_space)) % MOD)
