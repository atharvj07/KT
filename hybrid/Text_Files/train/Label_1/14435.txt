#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> ans(n);
    for (int i = 0; i < n; i++) {
        if (s[i] == '1' || (i > 0 && s[i - 1] == '0')) ans[i] = 1;
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i];
    }
    cout << endl;
    return 0;
}
