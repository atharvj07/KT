n, k, m = map(int, input().split())
pow = [1 for i in range(n + 1)]
for i in range(1, n + 1):
    pow[i] = pow[i - 1] * 10 % m
dp = [[[-1 for p in range(9)] for j in range(k)] for i in range(n + 1)]
def solve(pos, rem, last):
    if pos == n:
        return rem == 0
    if last == 0 and rem > 0:
        return 0
    if dp[pos][rem][last] != -1:
        return dp[pos][rem][last]
    ret = 0
    for i in range(0, 10):
        if pos == n - 1 and i == 0:
            continue
        if i == last and rem != 0:
            continue
        ret += solve(pos + 1, (10 * rem + i) % k, i) % m
        ret %= m
    dp[pos][rem][last] = ret
    return ret
print(solve(0, 0, 0))
