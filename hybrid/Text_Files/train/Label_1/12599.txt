#include <iostream>
#include <string>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        bool possible = true;
        int cntT = 0, cntM = 0;
        for (char c : s) {
            if (c == 'T') cntT++;
            else cntM++;
            if (cntT > cntM + 1 || cntM > cntT) {
                possible = false;
                break;
            }
        }
        if (cntT != cntM) possible = false;
        cout << (possible ? "YES" : "NO") << "\n";
    }
    return 0;
}
