#include<bits/stdc++.h>
#define ll long long
using namespace std;
int n;
ll a,ans;
int main()
{
    cin>>n>>a;
    priority_queue<ll> pq;
    for(int i = 0; i < n; i++)
    {
        ll x;
        cin>>x;
        pq.push(x);
    }
    ll b = pq.top();
    pq.pop();
    while(!pq.empty())
    {
        ll c = pq.top();
        pq.pop();
        ans+=abs(b - c);
        b = c;
    }
    ans+=abs(a - b);
    cout<<ans<<endl;
}
