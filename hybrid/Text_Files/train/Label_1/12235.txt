#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string ans;
    vector<string> arr(n);
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    sort(arr.begin(), arr.end());
    for (int i = 0; i < n; i++) { ans += arr[i]; }
    cout << ans << endl;
    return 0;
}
