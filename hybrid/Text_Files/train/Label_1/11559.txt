#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
#define vi vector<int>
#define vl vector<ll>
#define pb push_back
#define endl "\n"

const ll mod = 1e9 + 7;
const char nl = '\n';
const double pi = acos(-1);
const ll inf = 1e9;
const ll inf64 = 1e18;
const char *yes = "YES";
const char *no = "NO";

int isPerfectSquare(int x) {
  int sr = sqrt(x);
  return (sr * sr == x);
}

void solve() {
  int n;
  cin >> n;
  vi a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  vi pre(n, 0);
  pre[0] = a[0];
  for (int i = 1; i < n; i++) {
    pre[i] = pre[i - 1] * a[i];
  }
  vi suf(n, 0);
  suf[n - 1] = a[n - 1];
  for (int i = n - 2; i >= 0; i--) {
    suf[i] = suf[i + 1] * a[i];
  }
  vi ans(n, 0);
  for (int i = 0; i < n; i++) {
    ll p = 1;
    if (i > 0) {
      p = pre[i - 1];
    }
    ll s = 1;
    if (i < n - 1) {
      s = suf[i + 1];
    }
    if (isPerfectSquare(p * s)) {
      ans[i]++;
    }
    for (int j = i - 1; j >= 0; j--) {
      p /= a[j];
      if (isPerfectSquare(p * s)) {
        ans[i]++;
      }
    }
    for (int j = i + 1; j < n; j++) {
      s /= a[j];
      if (isPerfectSquare(p * s)) {
        ans[i]++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
}

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  solve();
  return 0;
}
