#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int mx = *max_element(a.begin(), a.end());
  int c = count(a.begin(), a.end(), mx);
  int l = -1, r = -1;
  for (int i = 0; i < n; i++) {
    if (a[i] == mx) {
      l = i;
      break;
    }
  }
  for (int i = l; i < n; i++) {
    if (a[i] == mx) {
      r = i;
    }
  }
  cout << l + 1 << " " << r + 1 << endl;
  return 0;
}
