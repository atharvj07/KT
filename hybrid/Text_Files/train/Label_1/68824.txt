n, k, a, b, q = map(int, input().split())
orders = [0] * (n + 1)

for _ in range(q):
    query = list(map(int, input().split()))
    if query[0] == 1:
        d, order = query[1], query[2]
        orders[d] = min(orders[d] + order, a)
    else:
        p = query[1]
        filled_orders = sum(min(b, orders[i]) for i in range(1, p))
        filled_orders += sum(orders[i] for i in range(p + k, n + 1))
        print(filled_orders)
