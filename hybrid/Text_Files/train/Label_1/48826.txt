import java.util.*;
import java.math.*;

class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        BigInteger N = new BigInteger(cin.next());
        int K = cin.nextInt();
        BigInteger L = BigInteger.ONE;
        while (L.compareTo(N) < 0) L = L.multiply(BigInteger.TEN);
        L = L.divide(BigInteger.TEN);
        BigInteger ans = BigInteger.ZERO;
        for (; K >= 0; K--) {
            BigInteger start = BigInteger.TEN.pow(K), end = start.multiply(BigInteger.TEN), cur = start;
            if (L.compareTo(end) < 0) end = L.add(BigInteger.ONE);
            while (cur.compareTo(end) < 0) {
                if (cur.compareTo(L) > 0) ans = ans.add(BigInteger.ONE);
                cur = cur.multiply(BigInteger.TEN);
            }
        }
        System.out.println(ans);
    }
}
