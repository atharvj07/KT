#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, d, e, dp[1000010], cost[1000010];
    dp[0] = 0, cost[0] = 0;
    cin >> n >> d >> e;
    for (int i = 1; i <= n; i++) cost[i] = 1e9;
    for (int i = d; i <= n; i++) cost[i] = min(cost[i], dp[i - d] + 1);
    for (int i = e; i <= n; i++) cost[i] = min(cost[i], dp[i - e] + 1);
    for (int i = 1; i <= n; i++) dp[i] = cost[i];
    for (int i = d; i <= n; i++) dp[i] = min(dp[i], dp[i - d]);
    for (int i = e; i <= n; i++) dp[i] = min(dp[i], dp[i - e]);
    cout << dp[n] << '\n';
}
