#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int ans = 0;
    if (a[0] * 2 <= a[n - 1]) {
      ans = a[n - 1] + a[n - 2] + a[n - 3];
    } else {
      ans = a[n - 1];
      for (int i = 0; i < n; i++) {
        if (a[i] * 2 > a[n - 1]) {
          ans += a[i];
          break;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
