import java.util.Arrays;
import java.util.HashSet;
import java.util.Scanner;

public class SubnetMask {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        String[] ipAddresses = new String[n];
        for (int i = 0; i < n; i++) {
            ipAddresses[i] = scanner.nextLine();
        }

        String subnetMask = findSubnetMask(ipAddresses, k);
        System.out.println(subnetMask);
    }

    private static String findSubnetMask(String[] ipAddresses, int k) {
        int[] bitCounts = new int[32];
        for (String ipAddress : ipAddresses) {
            String[] octets = ipAddress.split("\\.");
            for (int i = 0; i < 4; i++) {
                int octet = Integer.parseInt(octets[i]);
                for (int j = 0; j < 8; j++) {
                    if ((octet & (1 << (7 - j))) != 0) {
                        bitCounts[i * 8 + j]++;
                    }
                }
            }
        }

        StringBuilder subnetMaskBuilder = new StringBuilder();
        for (int i = 0; i < 32; i++) {
            if (bitCounts[i] == k) {
                subnetMaskBuilder.append("1");
            } else {
                subnetMaskBuilder.append("0");
            }
        }

        String subnetMask = subnetMaskBuilder.toString();
        if (isValidSubnetMask(subnetMask)) {
            return convertBinaryToIP(subnetMask);
        } else {
            return "-1";
        }
    }

    private static boolean isValidSubnetMask(String subnetMask) {
        int leadingOnes = 0;
        for (char c : subnetMask.toCharArray()) {
            if (c == '1') {
                leadingOnes++;
            } else {
                break;
            }
        }
        return leadingOnes > 0 && subnetMask.length() - leadingOnes > 0;
    }

    private static String convertBinaryToIP(String binaryString) {
        StringBuilder ipStringBuilder = new StringBuilder();
        for (int i = 0; i < 4; i++) {
            int octet = Integer.parseInt(binaryString.substring(i * 8, (i + 1) * 8), 2);
            ipStringBuilder.append(octet);
            if (i < 3) {
                ipStringBuilder.append(".");
            }
        }
        return ipStringBuilder.toString();
    }
}
