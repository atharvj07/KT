from sys import stdin
input = stdin.readline

n = int(input())
B = list(map(int,input().split()))

C = [0] * n
for i in range(n): B[i] += (i + 1) % n
for i in range(n): C[(B[i] - i - 1) % n] = 1

for i in range(n):
    if C[i]: continue
    for j in range(n):
        if (j + 1) % n == i: continue
        if B[(j + 1) % n] != B[(i + 1) % n]: continue
        B[(j + 1) % n] = j
        B[(i + 1) % n] = i
        break

for i in range(n):
    B[i] -= (i + 1) % n
    B[i] = B[i] % n

print(' '.join(map(str,B)))
