#include<bits/stdc++.h>
#define INF 1e9
using namespace std;
typedef pair<long long,long long>pii;
#define fi first
#define se second
const int maxN=1e5+7;
vector<pii>v;
int main(){
    int n;
    cin>>n;
    long long MaxX=0,MaxY=0,MinX=INF,MinY=INF;
    for(int i=0;i<n;i++){
        long long a,b;
        cin>>a>>b;
        v.push_back({a,b});
        MaxX=max(a,MaxX);
        MaxY=max(b,MaxY);
        MinX=min(a,MinX);
        MinY=min(b,MinY);
    }
    cout<<(MaxX-MinX+1)*(MaxY-MinY+1)/2<<'\n';
}
