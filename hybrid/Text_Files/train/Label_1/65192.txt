n, m, k = map(int, input().split())
a = list(map(int, input().split()))

# Step 1: Find the sum of m*k largest elements
a_sorted = sorted(a, reverse=True)
max_sum = sum(a_sorted[:m*k])

# Step 2: Find the partition
partitions = []
counts = 0
current_sum = 0
for i, num in enumerate(a):
    if num in a_sorted[:m*k]:
        current_sum += num
        a_sorted[a_sorted.index(num)] = -1  # Mark as used
        counts += 1
        if counts == m:
            partitions.append(i + 1)
            counts = 0
            if len(partitions) == k - 1:
                break

print(max_sum)
print(*partitions)
