#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    int l = 0, r = 0;
    for (auto &x : a) {
      l = min(l, x);
      r = max(r, x);
    }
    cout << r - l << endl;
  }
  return 0;
}
