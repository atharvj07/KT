import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int an = input.nextInt(), bn = input.nextInt(), r = input.nextInt();
            if (an == 0 && bn == 0 && r == 0) break;
            int[][] a = new int[an][2], b = new int[bn][2];
            for (int i = 0; i < an; i++) {
                a[i][0] = input.nextInt();
                a[i][1] = input.nextInt();
            }
            for (int i = 0; i < bn; i++) {
                b[i][0] = input.nextInt();
                b[i][1] = input.nextInt();
            }
            int ans = 0;
            for (int[] p : a) {
                int cnt = 0;
                for (int[] q : b) {
                    if (Math.sqrt((p[0] - q[0]) * (p[0] - q[0]) + (p[1] - q[1]) * (p[1] - q[1])) <= 2 * r) cnt++;
                }
                ans += cnt;
            }
            System.out.println(ans);
        }
    }
}
