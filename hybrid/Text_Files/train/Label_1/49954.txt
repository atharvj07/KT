import java.util.Scanner;

public class UnimodalArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        boolean isUnimodal = true;
        int increasingIndex = -1;
        int decreasingIndex = -1;

        // Check for increasing part
        for (int i = 0; i < n - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                if (increasingIndex != -1) {
                    isUnimodal = false;
                    break;
                } else {
                    increasingIndex = i;
                }
            }
        }

        // Check for decreasing part
        for (int i = n - 1; i > 0; i--) {
            if (arr[i] > arr[i - 1]) {
                if (decreasingIndex != -1) {
                    isUnimodal = false;
                    break;
                } else {
                    decreasingIndex = i;
                }
            }
        }

        if (isUnimodal && (increasingIndex == -1 || decreasingIndex == -1)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
