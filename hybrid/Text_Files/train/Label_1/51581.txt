import java.util.Scanner;

public class DiscoKun {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prizeCodingContest = getPrize(x);
        int prizeRobotManeuver = getPrize(y);
        int additionalPrize = (x == 1 && y == 1) ? 400000 : 0;

        System.out.println(prizeCodingContest + prizeRobotManeuver + additionalPrize);
    }

    private static int getPrize(int rank) {
        switch (rank) {
            case 1: return 300000;
            case 2: return 200000;
            case 3: return 100000;
            default: return 0;
        }
    }
}
