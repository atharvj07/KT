#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int q;
  cin >> q;
  vector<int> ans(q, 0);
  for (int i = 0; i < q; i++) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    a--, b--, c--, d--;
    vector<int> cnt(3, 0);
    for (int j = a; j <= b; j++) cnt[s[j] - 'A']++;
    for (int j = c; j <= d; j++) cnt[t[j] - 'A']--;
    int sum = 0;
    for (int j = 0; j < 3; j++) sum += abs(cnt[j]);
    ans[i] = (sum <= 6);
  }
  for (int i = 0; i < q; i++) cout << ans[i];
  return 0;
}
