n, k, l = map(int, input().split());
a = list(map(int, input().split())); a.sort();
if n * k > len(a) or (a[0] + l < a[-1]): print(0)
else:
    s = 0; cnt = 0; l = len(a) - 1; r = len(a) - 1 - n * k;
    while r >= 0:
        if a[l] - a[r] <= l:
            s += a[r]; cnt += 1; l -= k; r -= k;
        else: r -= 1;
    print(s)
