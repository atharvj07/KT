#include <bits/stdc++.h>
using namespace std;
#define MOD 1000000007

int n, m;
vector<int> r, c;
long long dp[1001][1001];

long long rec(int i, int j) {
    if (i == n) {
        int cnt = 0;
        for (int k = 0; k < m; k++)
            if (c[k] > 0) cnt++;
        return cnt == m;
    }
    if (j == m) {
        int cnt = 0;
        for (int k = 0; k < n; k++)
            if (r[k] > 0) cnt++;
        return cnt == n;
    }

    long long &ret = dp[i][j];
    if (ret != -1) return ret;

    ret = 0;
    if (r[i] <= j) {
        ret = rec(i + 1, j - r[i]) % MOD;
    }
    if (c[j] <= i) {
        ret += rec(i + c[j], j + 1) % MOD;
    }

    return ret %= MOD;
}

int main() {
    cin >> n >> m;
    r.resize(n);
    c.resize(m);

    for (int i = 0; i < n; i++) cin >> r[i];
    for (int i = 0; i < m; i++) cin >> c[i];
    memset(dp, -1, sizeof dp);

    cout << rec(0, 0) << "\n";
}
