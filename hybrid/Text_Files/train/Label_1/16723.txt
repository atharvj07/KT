#include<bits/stdc++.h>
using namespace std;
int wt[8]={0,1,3,3,5,9,0,0};
char ch[9][9];
int main(){
    int i,j,k,r,c,sw=0,sb=0;
    for(i=1;i<=8;++i){
        for(j=1;j<=8;++j){
            cin>>ch[i][j];
            if(ch[i][j]>='A' && ch[i][j]<='Z')
                sw+=wt[ch[i][j]-'A'+1];
            else if(ch[i][j]>='a'&& ch[i][j]<='z')
                sb+=wt[ch[i][j]-'a'+1];
        }
    }
    if(sw==sb)
        cout<<"Draw";
    else if(sw>sb)
        cout<<"White";
    else
        cout<<"Black";
    return 0;
}
