def can_sum(v1, v2):
    return all(d1 == '0' or d2 == '0' for d1, d2 in zip(f'{v1:010}', f'{v2:010}'))

k = int(input())
d = list(map(int, input().split()))

chosen = []
for i in range(k):
    can_choose = True
    for j in chosen:
        if not can_sum(d[i], d[j]):
            can_choose = False
            break
    if can_choose:
        chosen.append(d[i])

print(len(chosen))
print(' '.join(map(str, chosen)))
