import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            String s = sc.next();
            if (s.length() % 2 == 0 || s.charAt(0) == s.charAt(s.length() - 1)) {
                System.out.println("DA");
            } else {
                System.out.println("NET");
            }
        }
    }
}
