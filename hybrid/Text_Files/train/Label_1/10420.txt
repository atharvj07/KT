#include <bits/stdc++.h>
using namespace std;
const int maxn = 25;
int p[maxn];
void solve() {
  int n, t;
  cin >> n >> t;
  int m = n - 1, extra = 0;
  for (int i = 1; i <= n; i++) p[i] = 1;
  while (m--) {
    p[extra + 1] = t + 2;
    p[extra + 2] = t + 2;
    extra += 2;
    t -= 2;
  }
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++) {
      if (p[i] || p[j]) {
        --p[i];
        --p[j];
        cout << i << ' ' << j << '\n';
      }
    }
}
int main() {
  int t;
  cin >> t;
  while (t--) solve();
}
