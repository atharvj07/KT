n, m = map(int, input().split())
a = list(map(int, input().split()))
segments = [tuple(map(int, input().split())) for _ in range(m)]

def apply_segments(a, segments):
    b = a[:]
    for l, r in segments:
        for i in range(l-1, r):
            b[i] -= 1
    return b

max_diff = max(a) - min(a)
best_segments = []

for i in range(m):
    b = apply_segments(a, [segments[i]])
    diff = max(b) - min(b)
    if diff > max_diff:
        max_diff = diff
        best_segments = [i+1]

for i in range(m):
    for j in range(i+1, m):
        b = apply_segments(a, [segments[i], segments[j]])
        diff = max(b) - min(b)
        if diff > max_diff:
            max_diff = diff
            best_segments = [i+1, j+1]

print(max_diff)
print(len(best_segments))
print(*best_segments)
