n, m = map(int, input().split())
cards = list(map(int, input().split()))

odd = set()
even = set()
for card in cards:
    if card % 2 == 0:
        even.add(card)
    else:
        odd.add(card)

# Attempt to balance by exchanging the minimal amount of cards
needed_odd = n//2 - len(odd)
needed_even = n//2 - len(even)

if needed_odd < 0 or needed_even < 0:
    print(-1)
else:
    exchanged = 0
    new_cards = cards.copy()
    used_numbers = odd.union(even)
    next_odd = 1
    next_even = 2

    for i, card in enumerate(cards):
        if needed_odd > 0 and card in odd:
            while next_odd <= m and (next_odd in used_numbers or next_odd % 2 == 0):
                next_odd += 2
            if next_odd <= m:
                new_cards[i] = next_odd
                used_numbers.add(next_odd)
                needed_odd -= 1
                exchanged += 1
                next_odd += 2
        elif needed_even > 0 and card in even:
            while next_even <= m and (next_even in used_numbers or next_even % 2 != 0):
                next_even += 2
            if next_even <= m:
                new_cards[i] = next_even
                used_numbers.add(next_even)
                needed_even -= 1
                exchanged += 1
                next_even += 2

    if needed_odd > 0 or needed_even > 0:
        print(-1)
    else:
        print(exchanged)
        print(' '.join(map(str, new_cards)))
