n = input()
a = [int(x) for x in input().split()]
m = input()
q = [input().split() for i in range(int(m))]
w = [0 for i in range(int(n) + 1)]
for x in q:
    if x[0] == '1':
        i, j = int(x[1]) - 1, int(x[2])
        w[i] += j
        if w[i] > a[i]:
            w[i] -= a[i]
            w[i + 1] += a[i]
    else:
        i = int(x[1]) - 1
        print(w[i])
