#include <iostream>
#define MOD 998244353

using namespace std;

long long fastpow(long long base, long long p) {
    if (p == 0) return 1;
    long long half = fastpow(base, p / 2);
    if (p % 2 == 0) return half * half % MOD;
    return half * half % MOD * base % MOD;
}

int main() {
    long long n; cin >> n;
    long long ans[n + 1]; ans[0] = 1;
    for (long long i = 1; i <= n; ++i) {
        ans[i] = ans[i - 1] * (10 + (i % 2 == 0));
        ans[i] %= MOD;
    }
    for (long long i = 1; i <= n; ++i) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
