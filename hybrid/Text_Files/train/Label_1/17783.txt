#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string p, h;
        cin >> p >> h;
        int n = p.size();
        int m = h.size();
        if (n > m) {
            cout << "NO" << endl;
            continue;
        }
        vector<bool> vis(m, false);
        int j = 0;
        for (int i = 0; i < m && j < n;) {
            if (p[j] == h[i] && !vis[i]) {
                j++;
                vis[i] = true;
            }
            i++;
        }
        if (j == n)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
