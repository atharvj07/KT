#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 2 == 1)
      cout << -1 << '\n';
    else {
      for (int a = 0; a <= n / 3; ++a) {
        for (int b = 0; b <= (n - 3 * a) / 5; ++b) {
          int c = (n - 3 * a - 5 * b) / 7;
          if (3 * a + 5 * b + 7 * c == n) {
            cout << a << ' ' << b << ' ' << c << '\n';
            goto jmp;
          }
        }
      }
    jmp:;
    }
  }
  return 0;
}
