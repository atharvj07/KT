#include <algorithm>
#include <iostream>
#include <utility>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> tasks(n);
  for (int i = 0; i < n; i++) { cin >> tasks[i].first >> tasks[i].second; }
  sort(tasks.begin(), tasks.end());
  long long total_power = 0;
  int total_processors = 0;
  for (int i = 0; i < n; i++) {
    total_power += tasks[i].first;
    total_processors += tasks[i].second;
  }
  long long min_threshold = total_power / total_processors * 1000;
  if (total_power % total_processors) min_threshold++;
  cout << min_threshold << endl;
  return 0;
}
