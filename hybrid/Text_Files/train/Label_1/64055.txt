MOD = 1000000007

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, arr):
    max_val = max(arr)
    cnt = [0] * (max_val + 1)
    for val in arr:
        cnt[val] += 1
    
    ans = 0
    for i in range(2, max_val + 1):
        multiples = 0
        for j in range(i, max_val + 1, i):
            multiples += cnt[j]
        if multiples > 0:
            ans = (ans + pow(2, multiples - 1, MOD) * multiples * i) % MOD
    return ans

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(solve(n, arr))
