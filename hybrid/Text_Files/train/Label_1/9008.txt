#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    a.erase(unique(a.begin(), a.end()), a.end());
    for (int i = 0; i < k; i++) {
        int c = a.size();
        cout << c << " ";
        for (int j = i; j < c; j += k) cout << a[j] << " ";
        cout << endl;
    }
}
