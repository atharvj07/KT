#include <iostream>
#include <vector>
#include <unordered_map>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    std::unordered_map<int, int> m;
    for (int i = 0; i < n; ++i) {
      std::cin >> a[i];
      ++m[a[i]];
    }
    int l = 1;
    for (int i = 0; i < n; ++i) {
      for (int j = i + 1; j < n; ++j) {
        if (m[a[i]] > m[a[j]] && j - i + 1 < l) {
          l = j - i + 1;
        }
      }
    }
    if (l == 1)
      std::cout << -1 << std::endl;
    else
      std::cout << l << std::endl;
  }
}
