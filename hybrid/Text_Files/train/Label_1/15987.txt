#include <bits/stdc++.h>
using namespace std;
const int maxn = 110;
vector<int>adj[maxn];
char ch[maxn][maxn];
int dp[maxn][maxn][2], vis[maxn][maxn][2];
int n, m;
int dfs(int u, int v, int p) {
	if (vis[u][v][p])return dp[u][v][p];
	int &ans = dp[u][v][p]; vis[u][v][p] = 1;
	if (u == v)return ans = 1;
	for (auto to : adj[u]) {
		if (p == 0) {
			if (ch[u][to] >= ch[v][to])ans |= dfs(to, v, p);
		}
		else {
			if (ch[v][to] >= ch[u][to])ans |= dfs(u, to, p);
		}
	}
	return ans;
}
int main() {
	cin >> n >> m;
	for (int i = 0; i < m; i++) {
		int u, v; char c; cin >> u >> v >> c;
		ch[u][v] = c;
		adj[u].push_back(v);
	}
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= n; j++) {
			int x = dfs(i, j, 0), y = dfs(j, i, 1);
			cout << (x > y ? 'A' : 'B');
		}
		cout << endl;
	}
}
