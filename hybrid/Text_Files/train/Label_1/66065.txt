def solve(a, b, c):
    def s(x):
        ans = 0
        while x:
            ans += x % 10
            x //= 10
        return ans

    l, r = -1, 1000000000
    while r - l > 1:
        m = (l + r) // 2
        if b * s(m) ** a + c <= m:
            r = m
        else:
            l = m

    ans = []
    if 0 < r:
        t = b * s(r) ** a + c - r
        if t % b == 0 and t // b >= 0 and 0 < r + t // b < 1000000000:
            ans.append(t // b)

    if 0 < r - 1:
        t = b * s(r - 1) ** a + c - (r - 1)
        if t % b == 0 and t // b >= 0 and 0 < r - 1 + t // b < 1000000000:
            ans.append(t // b)

    print(len(ans))
    for x in ans:
        print(x, end=" ")

a, b, c = map(int, input().split())
solve(a, b, c)
