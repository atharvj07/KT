#include <cstdio>
#include <vector>
using namespace std;

int main() {
  int n, q, t, x, ans = 0;
  vector<int> a(300001);
  scanf("%d %d", &n, &q);
  while (q--) {
    scanf("%d", &t);
    if (t == 1) {
      scanf("%d", &x);
      if (!a[x]++) ans++;
    } else if (t == 2) {
      scanf("%d", &x);
      if (!--a[x]) ans--;
    } else {
      scanf("%d", &t);
      ans -= a[t];
      a[t] = 0;
    }
    printf("%d\n", ans);
  }
}
