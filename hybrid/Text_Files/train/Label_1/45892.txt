import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++)
            p[i] = in.nextInt();
        int[] s = new int[n + 2];
        int top = 0;
        StringBuilder ans = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            int x = p[i];
            while (top > 0 && s[top] >= x) {
                ans.append(')');
                top--;
            }
            if (top == 0 && x != 1) {
                System.out.println(":(");
                return;
            }
            s[++top] = x;
            ans.append('(');
        }
        while (top > 0) {
            ans.append(')');
            top--;
        }
        System.out.println(ans);
    }
}
