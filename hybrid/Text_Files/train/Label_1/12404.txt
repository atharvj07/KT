#include <iostream>
#include <string>
#include <set>

using namespace std;

int main() {
  string s;
  cin >> s;
  set<string> ans;
  int n = s.length();
  for (int i = 0; i <= n; i++) {
    for (char c = 'a'; c <= 'z'; c++) {
      string t = s;
      t.insert(i, 1, c);
      ans.insert(t);
    }
  }
  cout << ans.size() << endl;
  return 0;
}
