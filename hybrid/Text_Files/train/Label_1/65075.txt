t = int(input())
for _ in range(t):
    n, x, y, d = map(int, input().split())
    if abs(y - x) % d == 0:
        print(abs(y - x) // d)
    else:
        to_start = (x - 1) // d + 1 + (y - 1) // d
        to_end = (n - x) // d + 1 + (n - y) // d
        if (x - 1) % d != 0:
            to_start += 1
        if (n - y) % d != 0:
            to_end += 1
        if (y - 1) % d != 0 and (n - y) % d != 0:
            print(-1)
        else:
            print(min(to_start, to_end))
