#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int n;
ll a[200001], t[200001];

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i <= n; ++i) cin >> t[i];

    map<ll, ll> cnt;
    for (int i = 1; i <= n; ++i) cnt[a[i]]++;

    ll ans = 0;
    if (cnt.size() == n) goto L;

    for (auto it = cnt.begin(); it != cnt.end(); ++it) {
        auto itt = it;
        ++itt;
        if (it->first == itt->first) {
            ans += min(it->first, itt->first) * t[it->first];
            cnt.erase(itt);
        }
    }
    cnt.clear();
    for (int i = 1; i <= n; ++i) cnt[a[i]]++;

L:
    if (cnt.size() == n) {
        cout << ans << endl;
        return 0;
    }
    cout << -1 << endl;
    return 0;
}
