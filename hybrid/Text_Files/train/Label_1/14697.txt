#include <iostream>
#include <string>

using namespace std;

int main() {
  string n;
  cin >> n;
  int len = n.length();
  int ans[len * 2];
  for (int i = 0; i < len; i++) {
    ans[i] = n[i] - '0';
    ans[len * 2 - i - 1] = n[i] - '0';
  }
  for (int i = 0; i < len * 2; i++) cout << ans[i];
  cout << endl;
  return 0;
}
