import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int y1 = sc.nextInt();
            if (y1 == -1) {
                break;
            }
            int m1 = sc.nextInt();
            int d1 = sc.nextInt();
            int y2 = sc.nextInt();
            int m2 = sc.nextInt();
            int d2 = sc.nextInt();
            int days = 0;
            for (int y = y1; y < y2; y++) {
                days += isLeap(y) ? 366 : 365;
            }
            for (int m = m1; m < m2; m++) {
                days += daysInMonth(m, y2);
            }
            days += d2 - d1;
            System.out.println(days);
        }
    }

    private static int daysInMonth(int m, int y) {
        switch (m) {
            case 1:
            case 3:
            case 5:
            case 7:
            case 8:
            case 10:
            case 12:
                return 31;
            case 2:
                return isLeap(y) ? 29 : 28;
            default:
                return 30;
        }
    }

    private static boolean isLeap(int y) {
        return y % 4 == 0 && (y % 100 != 0 || y % 400 == 0);
    }
}
