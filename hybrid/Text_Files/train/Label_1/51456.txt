import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        int q = in.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        int[][] dp = new int[n + 1][2];
        dp[0][0] = 0;
        dp[0][1] = q;
        for (int i = 1; i <= n; i++) {
            dp[i][0] = Math.max(dp[i - 1][0], dp[i - 1][1]);
            dp[i][1] = dp[i - 1][0] + c[i - 1] + p * q;
            q = p * q;
        }
        int ans = dp[n][0];
        System.out.println(ans);
    }
}
