#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  cin.tie(0), ios::sync_with_stdio(0);
  int t;
  cin >> t;
  while (t--) {
    ll n, k, ans = 0;
    cin >> n >> k;
    vector<ll> a(n);
    for (ll &ai : a) cin >> ai;
    for (ll ai : a) {
      if (ai % k) { ans += (k - (ai % k)); }
    }
    cout << ans << '\n';
  }
}
