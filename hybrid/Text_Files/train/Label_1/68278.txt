import sys, heapq

def kruskal_min_span(n,edges):
    par = list(range(n))
    size = [1] * n
    ans = 0
    for w,u,v in edges:
        pu,pv = find_par(u,par),find_par(v,par)
        if pu != pv:
            if size[pu] > size[pv]:
                par[pv] = pu
                size[pu] += size[pv]
            else:
                par[pu] = pv
                size[pv] += size[pu]
            ans += w
    return ans

def find_par(node,par):
    if node == par[node]: return node
    return find_par(par[node],par)

def dijkstra(n,adj,src):
    dist = [float('inf')] * n
    dist[src] = 0
    pq = [(0,src)]
    while pq:
        cur_dist,cur_node = heapq.heappop(pq)
        if cur_dist > dist[cur_node]: continue
        for w,adj_node in adj[cur_node]:
            if dist[cur_node] + w < dist[adj_node]:
                dist[adj_node] = dist[cur_node] + w
                heapq.heappush(pq,(dist[adj_node],adj_node))
    return dist

n,m,a,b = map(int,sys.stdin.readline().split())

for i in range(m):

    adj[u].append((w,v))
