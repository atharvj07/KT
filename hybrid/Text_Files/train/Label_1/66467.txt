t = int(input())
for _ in range(t):
    n = int(input())
    h = list(map(int, input().split()))
    h.sort()
    if n % 2 == 0:
        for i in range(n // 2 - 1, -1, -1):
            print(h[i], end=' ')
        for i in range(n // 2, n):
            print(h[i], end=' ')
    else:
        for i in range(n // 2, -1, -1):
            print(h[i], end=' ')
        for i in range(n // 2 + 1, n):
            print(h[i], end=' ')
    print()
