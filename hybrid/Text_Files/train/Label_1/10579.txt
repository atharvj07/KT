#include <cstdio>
#include <utility>
#include <algorithm>
#include <vector>
#include <cstring>

using namespace std;

int n, d[11][11], mx;
bool v[11];

vector<int> g[11];

void bfs(int x) {
    memset(v, false, sizeof(v));
    v[x] = true;
    vector<int> q;
    q.push_back(x);
    while (!q.empty()) {
        int cur = q[0];
        q.erase(q.begin());
        for (int i = 0; i < g[cur].size(); i++) {
            if (!v[g[cur][i]]) {
                v[g[cur][i]] = true;
                q.push_back(g[cur][i]);
                d[x][g[cur][i]] = d[g[cur][i]][x] = d[x][cur] + 1;
                mx = max(mx, d[x][g[cur][i]]);
            }
        }
    }
}

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            int w;
            scanf("%d", &w);
            if (w) g[i].push_back(j), d[i][j] = w;
        }
    }
    for (int i = 1; i <= n; i++) bfs(i);
    printf("%d\n", mx);
    return 0;
}
