using namespace std;

typedef long long ll;

ll pmod(ll base, ll exponent, ll mod = 1000000007) {
    base %= mod;
    if (exponent == 0) {
        return 1;
    } else if (exponent == 1) {
        return base;
    } else if (exponent % 2 == 0) {
        ll t = pmod(base, exponent / 2, mod);
        return (t * t % mod);
    } else {
        return (base * pmod(base, exponent - 1, mod)) % mod;
    }
}

int main() {
    ll k;
    cin >> k;
    ll n = 1;
    for (ll i = 0; i < k; i++) {
        ll a;
        cin >> a;
        n *= a;
    }
    ll exp = pmod(2, n - 1);
    cout << 1 << "/" << exp;
    return 0;
}
