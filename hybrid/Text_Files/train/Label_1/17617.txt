#include <iostream>
using namespace std;
int main() {
  int n, k, ans = 1;
  cin >> n >> k;
  for (int i = 1; i < n && k > 1; ++i) {
    int temp = ans;
    for (int j = 1; j <= temp; ++j) {
      --k;
      if (k <= 0) {
        ans = j;
        break;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
