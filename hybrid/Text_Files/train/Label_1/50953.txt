import java.util.Scanner;

public class KseniaAndCombinatorics {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        // dp[i][j] is the number of trees with i vertices and j edges in the maximum matching
        long[][] dp = new long[n + 1][k + 1];
        dp[1][0] = 1;

        for (int i = 2; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                // Case 1: Vertex 1 is connected to two other vertices
                if (j >= 1) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - 1] * (i - 2)) % MOD;
                }

                // Case 2: Vertex 1 is connected to one other vertex
                if (j >= 1) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - 1] * (i - 1)) % MOD;
                }

                // Case 3: Vertex 1 is connected to zero other vertices
                dp[i][j] = (dp[i][j] + dp[i - 1][j] * (i - 1)) % MOD;
            }
        }

        System.out.println(dp[n][k]);
    }
}
