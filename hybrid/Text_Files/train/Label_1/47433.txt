import java.util.Scanner;

class Main { public static void main(String[] a) { var sc = new Scanner(System.in); var n = sc.nextInt(); var V = sc.nextInt(); var aa = new double[n]; var bb = new double[n]; for (int i = 0; i < n; i++) { aa[i] = sc.nextDouble(); bb[i] = sc.nextDouble(); } var x = 1e100; for (int i = 0; i < n; i++) if (aa[i] != 0) x = Math.min(x, V * bb[i] / aa[i]); System.out.println(x); } }
