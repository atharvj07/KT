t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    row1 = input().strip()
    row2 = input().strip()
    mex_sum = 0
    i = 0
    while i < n:
        if row1[i] == '0' and row2[i] == '0':
            if i + 1 < n and row1[i + 1] == '1' and row2[i + 1] == '1':
                mex_sum += 2
                i += 2
            else:
                mex_sum += 1
                i += 1
        elif row1[i] == '1' and row2[i] == '1':
            mex_sum += 0
            i += 1
        else:
            mex_sum += 2
            i += 1
    print(mex_sum)
