#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define vi vector<int>
int main() {
  ll n, a, b, k;
  cin >> n >> a >> b >> k;
  string s;
  cin >> s;
  ll l = -1, r = -1, ans = 2e9;
  for (ll i = 0; i < n; i++) {
    if (s[i] == '1') {
      if (l == -1)
        l = i;
      else
        r = i;
    }
  }
  if (a == 1) {
    cout << 1 << "\n" << l + 1 << "\n";
    return 0;
  }
  if (l == -1) {
    cout << b << "\n";
    for (ll i = 1; i <= b; i++) {
      cout << i << " ";
    }
    return 0;
  }
  if (r == -1) {
    cout << b << "\n";
    for (ll i = n - b + 1; i <= n; i++) {
      cout << i << " ";
    }
    return 0;
  }
  if (l - 1 + b <= r) {
    ans = min(ans, r - l + b);
    cout << ans << "\n";
    for (ll i = l - 1; i < r - b + 1; i++) {
      cout << i + 1 << " ";
    }
    return 0;
  }
  if (r + b - 1 >= n - l) {
    ans = min(ans, r - l + b);
    cout << ans << "\n";
    for (ll i = r + 1; i <= (r + b - 1) % n + 1; i++) {
      cout << i << " ";
    }
    return 0;
  }
  ans = min(ans, abs(l - r) + b);
  cout << ans << "\n";
  if (l < r) {
    for (ll i = l - 1; i <= l - 1 + b - 1; i++) {
      cout << i + 1 << " ";
    }
    for (ll i = r + 1; i <= r + b - 1; i++) {
      cout << i << " ";
    }
  } else {
    for (ll i = r - 1; i <= r - 1 + b - 1; i++) {
      cout << i + 1 << " ";
    }
    for (ll i = l + 1; i <= l + b - 1; i++) {
      cout << i << " ";
    }
  }
}
