#include<bits/stdc++.h>
using namespace std;
vector<long long> v;
long long n,x, heights[1000005];
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    cin>>n;
    for(long long i=1; i<=n; i++){
        cin>>x;
        heights[i] = x;
        v.push_back(x);
    }
    for(long long i=n-1; i>=0; i--){
        while(i>=0 && v[i]+2<=v[i+1]){
            v[i]+=1;
            v[i+1]-=1;
            i--;
        }
    }
    for(long long j=1; j<=n; j++){
        if(j>1)cout<<" ";
        cout<<v[j];
    }
    return 0;
}
