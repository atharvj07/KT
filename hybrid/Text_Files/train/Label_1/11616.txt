#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    bool ok = true;
    for (int i = 0; i < n - 1; i++) {
      if (s[i] == s[i + 1] && s[i] != '?') {
        ok = false;
        break;
      }
    }
    if (!ok) {
      cout << "-1" << endl;
      continue;
    }
    for (int i = 0; i < n; i++) {
      if (s[i] == '?') {
        if (i == 0) {
          if (n == 1) {
            s[i] = 'a';
          } else if (s[i + 1] != 'a') {
            s[i] = 'a';
          } else if (s[i + 1] == 'a' && (i + 2 == n || s[i + 2] != 'a')) {
            s[i] = 'b';
          } else {
            s[i] = 'c';
          }
        } else if (i == n - 1) {
          if (s[i - 1] != 'a') {
            s[i] = 'a';
          } else if (s[i - 1] == 'a' && (i - 2 == -1 || s[i - 2] != 'a')) {
            s[i] = 'b';
          } else {
            s[i] = 'c';
          }
        } else {
          if (s[i - 1] == s[i + 1] && s[i - 1] != '?') {
            s[i] = 'a' + 'b' + 'c' - s[i - 1] - s[i + 1];
          } else if (s[i - 1] != s[i + 1] && s[i - 1] != '?') {
            s[i] = s[i + 1];
          } else if (s[i - 1] != s[i + 1] && s[i + 1] != '?') {
            s[i] = s[i - 1];
          } else {
            if (s[i - 1] == 'a') {
              s[i] = 'b';
            } else if (s[i - 1] == 'b') {
              s[i] = 'c';
            } else {
              s[i] = 'a';
            }
          }
        }
      }
    }
    cout << s << endl;
  }
  return 0;
}
