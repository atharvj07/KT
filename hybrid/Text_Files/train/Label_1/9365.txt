#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string t, p;
  cin >> t >> p;
  int n = t.size();
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    a[i]--;
  }
  reverse(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int ind = a[i];
    if (t[ind] == p[ans]) {
      ans++;
      if (ans == p.size()) break;
    }
  }
  cout << ans << endl;
  return 0;
}
