#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;

int main() {
    int m;
    cin >> m;
    map<int, int> dp = {{0, 1}};
    while (m--) {
        int x;
        cin >> x;
        int y = dp[x ^ 3];
        dp[x] = (y + dp[x ^ 1] + dp[x ^ 2]) % mod;
        cout << dp[x] << "\n";
    }
}
