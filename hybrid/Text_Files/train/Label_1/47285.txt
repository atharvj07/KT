import java.util.Scanner;

public class Main {
	static int[] p;
	static int[] x;
	static int n;
	static boolean[] visited;
	static boolean[] black;
	static int[] weight;

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		n = sc.nextInt();
		p = new int[n];
		x = new int[n];
		visited = new boolean[n];
		black = new boolean[n];
		weight = new int[n];
		for (int i = 1; i < n; i++) {
			p[i] = sc.nextInt() - 1;
		}
		for (int i = 0; i < n; i++) {
			x[i] = sc.nextInt();
		}
		System.out.println(dfs(0) ? "POSSIBLE" : "IMPOSSIBLE");
	}

	static boolean dfs(int u) {
		visited[u] = true;
		int subBlack = 0;
		int subWhite = 0;
		for (int v = 0; v < n; v++) {
			if (v != u && p[v] == u) {
				if (!visited[v] && !dfs(v)) {
					return false;
				}
				if (black[v]) {
					subBlack += weight[v];
				} else {
					subWhite += weight[v];
				}
			}
		}
		if (subBlack != subWhite) {
			black[u] = true;
			weight[u] = x[u] - subWhite;
		} else {
			black[u] = false;
			weight[u] = x[u] - subBlack;
		}
		return weight[u] >= 0;
	}
}
