import sys
input=sys.stdin.readline
mod=int(1e9)+7
n=int(input())
a=[int(i) for i in input().split()]
dp=[0]*n
dp[0]=1
f=[0]*100001
for x in a:
    f[x]=1
g=[0]*100001
for i in range(1,n):
    t=a[i]
    while t>1:
        if f[t]==0: break
        f[t]=0
        if g[t]==0: dp[i]=(dp[i]+dp[g[t]])%mod
        g[t]=i
        t//=2
    dp[i]=(dp[i]+1)%mod
print(dp[n-1])
