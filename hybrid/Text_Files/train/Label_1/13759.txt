#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=1e5+1;
int n, m, a[N], b[N];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; ++i) {
        int x, y;
        cin >> x >> y;
        if (a[x] == 0 && b[y] == 0) {
            a[x] = 1;
            b[y] = 1;
            cout << n * n - (m - i + 1) << ' ';
        } else {
            cout << n * n - (m - i) << ' ';
        }
    }
}
