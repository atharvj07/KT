#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int N = 2e5 + 5;

int n, m, k;
int par[N], sz[N];
int f[N];

int find(int x) {
    if (par[x] != x) par[x] = find(par[x]);
    return par[x];
}

void merge(int x, int y) {
    x = find(x);
    y = find(y);
    if (x != y) {
        if (sz[x] < sz[y]) swap(x, y);
        par[y] = x;
        sz[x] += sz[y];
    }
}

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> m >> k;
    iota(par, par + n + 1, 0);
    fill(sz, sz + n + 1, 1);

    while (m--) {
        int x, y;
        cin >> x >> y;
        merge(x, y);
    }

    vector<int> ans;
    for (int i = 1; i <= n; i++) {
        int comp = find(i);
        if (sz[comp] >= k) ans.push_back(sz[comp]);
    }

    sort(ans.rbegin(), ans.rend());
    if (ans.empty()) ans.push_back(0);

    for (int i = 0; i < (int)ans.size(); i++) cout << ans[i] << '\n';

    return 0;
}
