#include <iostream>
#include <vector>

using std::cin;
using std::cout;
using std::vector;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int64_t> a(n, 0);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<vector<int64_t>> ops(m, vector<int64_t>(3, 0));
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < 3; j++) { cin >> ops[i][j]; }
  }
  vector<vector<int64_t>> qs(k, vector<int64_t>(2, 0));
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < 2; j++) { cin >> qs[i][j]; }
  }
  vector<int64_t> diffs(n, 0);
  for (int i = 0; i < k; i++) {
    int lx = qs[i][0], rx = qs[i][1];
    for (int j = lx - 1; j < rx; j++) {
      int l = ops[j][0] - 1, r = ops[j][1] - 1;
      int64_t d = ops[j][2];
      diffs[l] += d;
      if (r + 1 < n) { diffs[r + 1] -= d; }
    }
  }
  for (int i = 1; i < n; i++) { diffs[i] += diffs[i - 1]; }
  for (int i = 0; i < n; i++) { a[i] += diffs[i]; }
  for (int i = 0; i < n; i++) { cout << a[i] << " "; }
  cout << "\n";
}
