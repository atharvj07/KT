#include <cstdio>
#include <vector>

using namespace std;

int gcd(int a, int b) {
  if(b == 0) return a;
  return gcd(b, a%b);
}

int main() {
  int n, k=0;
  scanf("%d", &n);
  vector<int> a(n);
  for(int i=0; i<n; ++i) {
    scanf("%d", &a[i]);
  }
  for(int i=0; i+1<n; ++i) {
    if(gcd(a[i], a[i+1]) == 1) continue;
    k++;
    int x = 1;
    while(gcd(a[i], x) != 1) x++;
    a.insert(a.begin()+i+1, x);
  }
  printf("%d\n", k);
  for(int x : a) printf("%d ", x);
  puts("");
  return 0;
}
