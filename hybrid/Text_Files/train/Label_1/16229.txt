#include <iostream>
#include <queue>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int xa, ya, xb, yb, xf, yf;
    cin >> xa >> ya >> xb >> yb >> xf >> yf;
    int ans = abs(xa - xb) + abs(ya - yb);
    if (xa == xf && xb == xf && ya != yb) ans += 2;
    if (ya == yf && yb == yf && xa != xb) ans += 2;
    cout << ans << endl;
  }
  return 0;
}
