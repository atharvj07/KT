#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    bool found = false;
    for (int i = 1; i < n / 2; i++) {
        if (s.substr(0, i) == s.substr(n - i, i)) {
            cout << i - 1 << endl;
            found = true;
            break;
        }
    }
    if (!found) cout << "Impossible" << endl;
    return 0;
}
