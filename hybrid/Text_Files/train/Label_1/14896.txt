#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> c(n);
    int ans = 1e9;
    for (int i = 0; i < n; i++) {
        cin >> c[i];
    }
    sort(c.begin(), c.end());
    c.erase(unique(c.begin(), c.end()), c.end());
    for (int i = 0; i < c.size(); i++) {
        int cost = 0;
        int l = lower_bound(c.begin(), c.end(), c[i]) - c.begin();
        int r = upper_bound(c.begin(), c.end(), c[i]) - c.begin() - 1;
        cost += l * (c[l] != c[i]);
        cost += (n - 1 - r) * (c[r] != c[i]);
        ans = min(ans, cost);
    }
    cout << ans << endl;
}
