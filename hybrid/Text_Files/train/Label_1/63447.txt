import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = [int(x) for x in input().split()]
        ans = 0
        for i in range(n - 1):
            if a[i] > 0:
                c = min(a[i], -a[i + 1])
                a[i] -= c
                a[i + 1] += c
                ans += c
        print(ans)
        
solve()
