#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

int main() {
    int t;
    cin >> t;
    for (int i = 0; i < t; ++i) {
        int n;
        cin >> n;
        vector<long long> a(n);
        long long sum = 0;
        for (int j = 0; j < n; ++j) {
            cin >> a[j];
            sum += a[j];
        }
        long long k = sum / n;
        long long ans = 0;
        for (int l = 0; l < n; ++l) {
            if (sum - a[l] - a[l] == (n - 2) * k) {
                ++ans;
            }
            sum -= a[l];
        }
        ans /= 2;
        cout << ans << endl;
    }
}
