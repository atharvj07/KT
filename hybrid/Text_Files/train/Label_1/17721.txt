#include <bits/stdc++.h>
using namespace std;
char M[550][550];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) { cin >> M[i][j]; }
    }
    int cnt = 0;
    for (int i = 2; i < n; ++i) {
        for (int j = 2; j < n; ++j) {
            if (!(M[i][j] ^ M[i - 1][j - 1])) {
                if (!(M[i][j] ^ M[i - 1][j + 1])) {
                    if (!(M[i][j] ^ M[i + 1][j - 1])) {
                        if (!(M[i][j] ^ M[i + 1][j + 1])) cnt++;
                    }
                }
            }
        }
    }
    cout << cnt << endl;
    return 0;
}
