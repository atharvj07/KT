#include <bits/stdc++.h>
using namespace std;
int n,m,bit[5005][1005],val[5005],symbol[5005];
string name[5005];
unordered_map<string,int> mp;
int get(string nw){
    if(mp.count(nw))return mp[nw];
    mp[nw]=++n;
    name[n]=nw;
    return n;
}
int get_val(string nw){
    if(nw=="?")return -1;
    int tmp=get(nw);
    if(symbol[tmp]==-1)return -1;
    return val[tmp];
}
int get_bit(string nw){
    if(nw=="?")return -1;
    int tmp=get(nw);
    if(symbol[tmp]==-1)return -1;
    for(int i=m-1;i>=0;--i){
        if(bit[tmp][i])return i;
    }
}
int dfs(int pos){
    if(pos==n+1)return 0;
    if(symbol[pos]!=-1)return val[pos];
    string nw=name[pos];
    int &ret=val[pos];
    int x=get_bit(nw);
    if(x!=-1){
        ret=bit[pos][x];
        symbol[pos]=0;
        return ret;
    }
    x=get_val(nw);
    if(x!=-1){
        ret=x;
        symbol[pos]=1;
        return ret;
    }
    int l=dfs(pos+1),r=dfs(pos+1);
    if(nw[0]=='A' or nw[0]=='O'){
        ret=l|r;
    }else{
        ret=l^r;
    }
    symbol[pos]=2;
    return ret;
}
int main(){
    cin>>n>>m;
    vector<int>ans;
    for(int i=1;i<=n;++i){
        string nw,y;
        cin>>nw>>y;
        int x=get(nw);
        string w;
        int v;
        if(y[0]=='?'){
            symbol[x]=-1;
            continue;
        }
        int j=3;
        while(j<y.size()){
            w+=y[j];
            ++j;
        }
        if(y[j+1]=='A' or y[j+1]=='O'){
            j+=4;
        }else{
            j+=3;
        }
        ++j;
        while(j<y.size()){
            v+=y[j]-48;
            ++j;
        }
        if(y[j+1]=='A' or y[j+1]=='O'){
            val[x]|=1<<v;
            bit[x][v]=1;
        }else{
            val[x]^=1<<v;
            bit[x][v]=1;
        }
        symbol[x]=1;
    }
    int now=dfs(1);
    for(int i=m-1;i>=0;--i){
        if(now&(1<<i))ans.push_back(1);
        else ans.push_back(0);
    }
    for(int i=m-1;i>=0;--i){
        now^=1<<i;
        for(int i=1;i<=n;++i){
            string nw=name[i];
            int x=get_bit(nw);
            if(x==i){
                val[i]=now&(1<<i);
            }
        }
        dfs(1);
    }
    for(int i=m-1;i>=0;--i){
        if(now&(1<<i))ans.push_back(1);
        else ans.push_back(0);
    }
    for(int i=m-1;i>=0;--i){
        cout<<ans[i];
    }
    cout<<endl;
    for(int i=m-1;i>=0;--i){
        cout<<ans[i];
    }
    return 0;
}
