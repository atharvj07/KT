#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int l, r;
    cin >> l >> r;
    string s = to_string(l);
    sort(s.begin(), s.end());
    int n = s.length();
    if (r - l < n - 1)
        cout << "-1";
    else {
        for (int i = 1; i <= n; i++)
            cout << s[i - 1];
        for (int i = n + 1; i <= r - l; i++)
            cout << i;
    }
    cout << endl;
    return 0;
}
