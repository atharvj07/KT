import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        int min = Math.min(x[0], x[1]), max = Math.max(x[0], x[1]);
        int cnt = 0;
        for (int i = 2; i < n; i++) {
            if (x[i] >= min && x[i] <= max) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
