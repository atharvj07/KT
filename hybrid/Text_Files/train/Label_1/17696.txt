#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  map<int, int> m;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    m[v[i]]++;
  }
  int ans = 0;
  for (auto &x : m) {
    int y = x.second;
    if (y >= 2) {
      y -= 2;
      ans += 2;
      while (y >= 2) {
        y -= 2;
        ans += 2;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
