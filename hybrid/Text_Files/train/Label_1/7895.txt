#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int n;
    cin >> n;
    vector<int> v(n), t(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    for (int i = 0; i < n; i++) cin >> t[i];
    int sum = 0;
    for (int i = 0; i < n; i++) {
        int dif = v[i] - t[i];
        v[i] -= dif;
        sum += dif;
        cout << sum << " ";
    }
    cout << endl;
}
