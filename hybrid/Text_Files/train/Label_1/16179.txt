#include <cstdio>
#include <vector>

using namespace std;

const long long MAX = 1e6;
long long n, m;
vector<pair<long long, char> > t, s;

int main() {
  scanf("%lld %lld", &n, &m);
  t.resize(n);
  s.resize(m);
  for (int i = 0; i < n; ++i) {
    long long a;
    char c;
    scanf("%lld-%c", &a, &c);
    t[i] = make_pair(a, c);
  }
  for (int i = 0; i < m; ++i) {
    long long a;
    char c;
    scanf("%lld-%c", &a, &c);
    s[i] = make_pair(a, c);
  }
  long long j = 0, ans = 0;
  for (int i = 0; i < n; ++i) {
    while (j < m && (j == 0 || t[i].second != s[j].second)) ++j;
    if (j == m) break;
    if (t[i].first >= s[j].first) {
      ++ans;
      t[i].first -= s[j].first;
      ++j;
    }
  }
  printf("%lld\n", ans);
  return 0;
}
