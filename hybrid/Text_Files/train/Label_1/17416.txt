#include<bits/stdc++.h>
using namespace std;
inline long long solve(){
    string s;
    getline(cin,s);
    int n=s.length(),ans=0;
    for(int i=0;i<=n-6;++i)for(int j=i+5;j<n;++j)ans += (s.substr(i,6)=="heavy" and s.substr(j,5)=="metal");
    return ans;
}
int main(){
    cout<<solve()<<endl;
    return 0;
}
