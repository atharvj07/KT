import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    long c = sc.nextLong();
    long[] x = new long[n];
    long[] v = new long[n];
    for (int i = 0; i < n; i++) {
      x[i] = sc.nextLong();
      v[i] = sc.nextLong();
    }

    long[][] dp = new long[n + 1][2];
    dp[0][0] = -Long.MAX_VALUE / 2;
    dp[0][1] = -Long.MAX_VALUE / 2;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 2; j++) {
        long a = x[i] + dp[i][j];
        long b = c - x[i] + dp[i][j];
        a = Math.max(a, - v[i]);
        b = Math.max(b, - v[i]);
        dp[i + 1][0] = Math.max(dp[i + 1][0], a);
        dp[i + 1][1] = Math.max(dp[i + 1][1], b);
      }
    }

    long ans = Math.max(dp[n][0], dp[n][1]);
    System.out.println(ans);
  }
}
