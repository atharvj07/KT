def min_subscriptions(n, k, d, shows):
    from collections import defaultdict
    show_count = defaultdict(int)
    unique_shows = 0
    min_sub = float('inf')
    
    for i in range(d):
        if show_count[shows[i]] == 0:
            unique_shows += 1
        show_count[shows[i]] += 1
    
    min_sub = min(min_sub, unique_shows)
    
    for i in range(d, n):
        if show_count[shows[i-d]] == 1:
            unique_shows -= 1
        show_count[shows[i-d]] -= 1
        
        if show_count[shows[i]] == 0:
            unique_shows += 1
        show_count[shows[i]] += 1
        
        min_sub = min(min_sub, unique_shows)
    
    return min_sub

t = int(input().strip())
for _ in range(t):
    n, k, d = map(int, input().strip().split())
    shows = list(map(int, input().strip().split()))
    print(min_subscriptions(n, k, d, shows))
