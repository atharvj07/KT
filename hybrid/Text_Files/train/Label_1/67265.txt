def find_cycles(permutation):
    visited = [False] * len(permutation)
    cycles = []
    for i in range(len(permutation)):
        if not visited[i]:
            cycle = []
            current = i
            while not visited[current]:
                visited[current] = True
                cycle.append(current)
                current = permutation[current] - 1
            cycles.append(cycle)
    return cycles

def solve(n, p, b):
    cycles = find_cycles(p)
    flip_count = sum(b)
    changes = 0

    for cycle in cycles:
        cycle_flip_count = sum(b[i] for i in cycle)
        if len(cycle) % 2 == 0:
            if cycle_flip_count % 2 == 0:
                changes += 1
        else:
            if cycle_flip_count % 2 == 0:
                if flip_count % 2 == 0:
                    changes += 1
                else:
                    changes += 2
            else:
                if flip_count % 2 == 0:
                    changes += 2
    return changes

n = int(input().strip())
p = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

print(solve(n, p, b))
