from collections import defaultdict
ds = defaultdict(lambda: 0)
n, m = map(int, input().split())
vars = {}
for line in [input().strip().split(':=') for _ in range(n)]:
    res = line[1].strip()
    if res[0] != '?':
        op, a, b = res.split()
        if op == 'AND':
            vars[line[0]] = int(a, 2) & (ds[b])
        elif op == 'OR':
            vars[line[0]] = int(a, 2) | (ds[b])
        elif op == 'XOR':
            vars[line[0]] = int(a, 2) ^ (ds[b])
    else:
        ds[line[0]] = int(input(), 2)
print(min((vars[k] for k in vars), default=0), max((vars[k] for k in vars), default=0))
