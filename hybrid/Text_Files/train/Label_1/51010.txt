import java.util.*;
import java.io.*;

class Main {
    static int n, k, ans = 0;
    static int[] c, f, h;
    static boolean[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        c = new int[n * k];
        f = new int[n];
        h = new int[k + 1];
        dp = new boolean[n + 1][k * n + 1];
        for (int i = 0; i < n * k; i++) c[i] = sc.nextInt();
        for (int i = 0; i < n; i++) f[i] = sc.nextInt();
        for (int i = 1; i <= k; i++) h[i] = sc.nextInt();
        solve(0, 0);
        System.out.println(ans);
    }

    static void solve(int player, int count) {
        if (dp[player][count]) return;
        dp[player][count] = true;
        if (player == n) {
            if (count == n * k) ans = Math.max(ans, getJoy());
            return;
        }
        for (int i = 0; i < n * k; i++) {
            if (c[i] == f[player]) solve(player + 1, count + 1);
            else solve(player + 1, count);
        }
    }

    static int getJoy() {
        int[] freq = new int[n];
        for (int i = 0; i < n * k; i++) freq[c[i] - 1]++;
        int j = 0, ans = 0;
        for (int i = 0; i < n; i++) {
            while (j < k && freq[f[i] - 1] > j) {
                ans += h[++j];
            }
            if (freq[f[i] - 1] >= k) ans += h[k];
        }
        return ans;
    }
}
