import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, k, m = map(int, input().split())
    b = list(map(int, input().split()))
    if n - m < k or (len(b) > 1 and (n - k + 1) % (k - 1) != 0):
        print('NO')
        continue

    cnt = 0
    for i in range(1, n + 1, k - 1):
        if i + k - 1 <= b[-1]:
            cnt += 1

    if cnt >= m:
        print('YES')
    else:
        print('NO')
