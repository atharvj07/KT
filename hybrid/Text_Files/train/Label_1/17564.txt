#include <iostream>

using namespace std;

int GCD(long long a, long long b) {
  if (b == 0) return a;
  return GCD(b, a % b);
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    while (GCD(n, n % 10 + n / 10 % 10 + n / 100 % 10 + n / 1000 % 10 + n / 10000 % 10 + n / 100000 % 10 + n / 1000000 % 10 + n / 10000000 % 10 + n / 100000000 % 10 + n / 1000000000 % 10) == 1) { n++; }
    cout << n << endl;
  }
}
