#include <bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    int a[n], b[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    long long ans = LLONG_MIN;
    for (int l = 0; l < 60; l++) {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] & (1LL << l)) cnt++;
        }
        if (cnt > 1) {
            for (int i = 0; i < n; i++) {
                if (a[i] & (1LL << l)) { ans = max(ans, (long long)b[i]); }
            }
        }
    }
    if (ans == LLONG_MIN) ans = 0;
    cout << ans << endl;
    return 0;
}
