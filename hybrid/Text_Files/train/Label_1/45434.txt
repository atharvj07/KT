import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int s = in.nextInt();
        int[] ini = new int[n], outi = new int[n], wi = new int[n], si = new int[n], vi = new int[n];
        for (int i = 0; i < n; i++) {
            ini[i] = in.nextInt();
            outi[i] = in.nextInt();
            wi[i] = in.nextInt();
            si[i] = in.nextInt();
            vi[i] = in.nextInt();
        }
        in.close();
        Arrays.sort(ini);
        Arrays.sort(outi);
        long ans = 0;
        int j = 0, w = 0;
        for (int i = 0; i < n; i++) {
            while (j < n && outi[j] <= ini[i]) {
                w -= wi[j];
                j++;
            }
            if (w + wi[i] <= s) {
                w += wi[i];
                ans += vi[i];
            }
        }
        System.out.println(ans);
    }
}
