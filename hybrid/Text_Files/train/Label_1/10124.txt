#include <bits/stdc++.h>

using namespace std;
using ll = long long;
const ll mod = 998244353;

int main() {
    ll n, k;
    cin >> n >> k;
    vector<pair<ll, ll>> v(n);
    for (auto &x : v) { cin >> x.first >> x.second; }
    sort(v.begin(), v.end());
    ll l = 1, ans = 0;
    for (int r = 0; r < n; ++r) {
        while (l < r && v[r].first - v[l].second > k - 1) { ++l; }
        if (v[r].first - v[l].second == k - 1) {
            ans = (ans + v[r].second - v[l].first + 1) % mod;
        }
    }
    cout << ans << '\n';
    return 0;
}
