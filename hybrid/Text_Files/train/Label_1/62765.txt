n = int(input())
seq = list(map(int, input().split()))

def find_start(seq):
    for num in seq:
        count_mul, count_div = 0, 0
        temp = num
        while temp % 3 == 0:
            temp //= 3
            count_div += 1
        temp = num
        while temp % 2 == 0:
            temp //= 2
            count_mul += 1
        yield num, count_div - count_mul

start = max(find_start(seq), key=lambda x: x[1])[0]

result = [start]
for _ in range(n - 1):
    if start % 3 == 0 and start // 3 in seq:
        start //= 3
    else:
        start *= 2
    result.append(start)

print(" ".join(map(str, result)))
