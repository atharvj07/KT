#include <bits/stdc++.h>

#define ll long long

using namespace std;

ll p = 998244353;
const int N = 20, M = 50010;
ll f[N][M], inv[M], pow2[M];
int dist[N][M];

ll fastpow(ll x, ll y) {
    ll res = 1;
    while (y > 0) {
        if (y % 2 == 0) {
            x = x * x % p;
        } else {
            res = res * x % p;
        }
        y /= 2;
    }
    return res;
}

int main() {
    int n, m;
    cin >> n >> m;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> dist[i][j];
        }
    }

    for (int i = 0; i < M; i++) pow2[i] = fastpow(2, i);
    inv[0] = inv[1] = 1;
    for (int i = 2; i < M; i++) inv[i] = (p - p / i) * inv[p % i] % p;

    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j <= m; j++) {
            f[i][j] = (j ? f[i + 1][j - 1] : 0) + (i >= 1 ? f[i - 1][j] * inv[2] % p : 0);
            f[i][j] = (f[i][j] + (m - j) * inv[pow2[n - i]]) % p;
            for (int k = 0; k < n; k++) {
                if (dist[k][j] <= n - i) f[i][j] = (f[i][j] + (i > 0 || dist[k][j] ? inv[pow2[n - i]] : 0)) % p;
            }
        }
    }

    ll x = f[0][m], y = fastpow(inv[2], n);
    cout << (x * fastpow(y, p - 2) % p + p) % p << "\n";

    return 0;
}
