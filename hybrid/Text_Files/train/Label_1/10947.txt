#include <bits/stdc++.h>
using namespace std;
string s, t;
int main() {
  int n, m;
  cin >> n >> m >> s >> t;
  if (n < m) {
    cout << "NO" << endl;
    return 0;
  }
  int i = 0, j = 0;
  bool asterisk = false;
  for (; i < n && j < m;) {
    if (s[i] == '*') {
      asterisk = true;
      i++;
      continue;
    }
    if (s[i] != t[j]) {
      if (!asterisk) {
        cout << "NO" << endl;
        return 0;
      }
      while (++j < m && s[i] != t[j]) {}
      if (j == m) {
        cout << "NO" << endl;
        return 0;
      }
    }
    i++, j++;
  }
  if (i < n) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
  }
  return 0;
}
