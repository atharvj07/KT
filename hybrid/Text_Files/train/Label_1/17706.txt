#include <iostream>
using namespace std ; 
int main() {
    int a, b;
    cin >> a >> b ; 
    a -= b ; 
    if(a % 2 != 0 || a < 0 || b - a / 2 < 1) {
        cout << -1 << endl;  
    } else {
        for(int i = 0 ; i < b - a / 2 ; i++) {
            cout << 1 << " " ; 
        }
        a /= 2 ; 
        for(int i = 0 ; i < a ; i++) {
            for(int j = 0 ; j < i ; j++) {
                cout << 0 << " "; 
            }
            cout << 1 << " " ;
            for(int j = 0 ; j < a - i - 1 ; j++) {
                cout << 0 << " " ;
            }
            cout << endl ; 
        }
    }
    return 0 ; 
}
