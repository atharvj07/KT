#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define TIMES (ll)(1e9+7)

int main() {
	ll n, m, ans = 1, a[505];
	cin >> n >> m;
	for (int i = 0; i < m; i++) { cin >> a[i]; }
	for (int i = 0; i < m; i++)
		if (i != m - 1 && a[i] != a[i + 1]) ans = (ans * (i + 1)) % TIMES;
	cout << ans << endl;
}
