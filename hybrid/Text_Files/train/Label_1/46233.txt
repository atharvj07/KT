import java.util.Scanner;

public class YenWithdrawal {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int count = 0;

        // 6-based denominations
        for (int i = 6; i <= N; i *= 6) {
            count++;
            N -= i;
        }

        // 9-based denominations
        for (int i = 9; i <= N; i *= 9) {
            count++;
            N -= i;
        }

        // Remaining 1-yen withdrawals
        count += N;

        System.out.println(count);
    }
}
