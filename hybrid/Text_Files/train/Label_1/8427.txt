#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    if (n & 1)
        cout << "NO" << endl;
    else {
        cout << "YES" << endl;
        for (int i = 1; i <= n; i += 2)
            for (int j = i + 1; j <= n; j++)
                cout << i << ' ' << j << endl;
    }
    return 0;
}
