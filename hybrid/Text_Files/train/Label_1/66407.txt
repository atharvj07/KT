n = int(input())
a = list(map(int, input().split()))
a.sort()
ans = -1
for i in range(n - 1):
    if a[i + 1] - a[i] > 2:
        ans = -1
        break
    elif a[i + 1] - a[i] == 2:
        ans = a[i] + 1
    elif a[i + 1] - a[i] == 1:
        if ans == -1:
            ans = a[i] + 1
        elif ans != a[i] + 1:
            ans = -1
            break
print(ans)
