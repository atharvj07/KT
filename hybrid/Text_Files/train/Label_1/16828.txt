#include<bits/stdc++.h>
#define repeat(i,s,n) for(int i = s; i < n; ++i)
#define endl '\n'
using namespace std;
const int max_n = 1e5 + 5;
int n, m, d, a[max_n];
long long dp[max_n];
int main(){
    cin >> n >> d >> m;
    repeat(i, 0, n) cin >> a[i];
    repeat(i, 0, n){
        dp[i] = max(dp[i], dp[max(0, i - d - 1)] + a[i]);
        if(i and a[i] > m){
            dp[i] = max(dp[i], dp[i - 1]);
        }
    }
    cout << dp[n - 1] << endl;
    return 0;
}
