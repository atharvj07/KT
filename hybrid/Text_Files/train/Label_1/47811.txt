import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        for (int i = 0; i < q; i++) {
            int xp0 = scanner.nextInt();
            int yp0 = scanner.nextInt();
            int xp1 = scanner.nextInt();
            int yp1 = scanner.nextInt();
            int xp2 = scanner.nextInt();
            int yp2 = scanner.nextInt();
            int xp3 = scanner.nextInt();
            int yp3 = scanner.nextInt();

            if (doIntersect(xp0, yp0, xp1, yp1, xp2, yp2, xp3, yp3)) {
                System.out.println(1);
            } else {
                System.out.println(0);
            }
        }

        scanner.close();
    }

    public static boolean doIntersect(int xp0, int yp0, int xp1, int yp1, int xp2, int yp2, int xp3, int yp3) {
        int d1 = (xp1 - xp0) * (yp2 - yp0) - (yp1 - yp0) * (xp2 - xp0);
        int d2 = (xp1 - xp0) * (yp3 - yp0) - (yp1 - yp0) * (xp3 - xp0);
        int d3 = (xp3 - xp2) * (yp1 - yp2) - (yp3 - yp2) * (xp1 - xp2);
        int d4 = (xp3 - xp2) * (yp0 - yp2) - (yp3 - yp2) * (xp0 - xp2);

        if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) {
            return true;
        } else if (d1 == 0 && onSegment(xp0, yp0, xp1, yp1, xp2, yp2)) {
            return true;
        } else if (d2 == 0 && onSegment(xp0, yp0, xp1, yp1, xp3, yp3)) {
            return true;
        } else if (d3 == 0 && onSegment(xp2, yp2, xp3, yp3, xp0, yp0)) {
            return true;
        } else if (d4 == 0 && onSegment(xp2, yp2, xp3, yp3, xp1, yp1)) {
            return true;
        } else {
            return false;
        }
    }

    public static boolean onSegment(int xp0, int yp0, int xp1, int yp1, int xp2, int yp2) {
        return (xp0 <= xp2 && xp2 <= xp1 && yp0 <= yp2 && yp2 <= yp1);
    }
}
