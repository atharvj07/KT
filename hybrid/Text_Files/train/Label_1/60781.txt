def count_good_sequences(n, a):
    def bit_count(x):
        return bin(x).count('1')

    prefix = [0]
    for x in a:
        prefix.append(prefix[-1] + bit_count(x))

    count = 0
    for l in range(n):
        max_bit = 0
        for r in range(l, min(n, l + 64)):
            max_bit = max(max_bit, bit_count(a[r]))
            if (prefix[r + 1] - prefix[l]) % 2 == 0 and max_bit * 2 > prefix[r + 1] - prefix[l]:
                count += 1
        count += n - 1 - max(l, n - 64)
    return count

n = int(input().strip())
a = list(map(int, input().strip().split()))
print(count_good_sequences(n, a))
