#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  while (k--) {
    string t = s;
    for (int i = 1; i <= n; i++) {
      int j = (i - 2 + n) % n + 1, l = (i % n) + 1;
      int c = (s[j - 1] == 'W') + (s[i - 1] == 'W') + (s[l - 1] == 'W');
      t[i - 1] = (c > ((s[j - 1] == 'B') + (s[i - 1] == 'B') + (s[l - 1] == 'B'))) ? 'W' : 'B';
    }
    s = t;
  }
  cout << s << '\n';
}
