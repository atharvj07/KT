import java.util.*;

class Main {
  static int[] dp = new int[3005];
  static int MOD = 998244353;

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int c = input.nextInt();
    int[] arr = new int[n + 1];
    for (int i = 1; i <= n; i++) arr[i] = input.nextInt();
    int[][] dp2 = new int[n + 1][c + 1];
    dp2[n][0] = 1;
    for (int i = n - 1; i >= 0; i--) {
      for (int j = 0; j <= c; j++) {
        dp2[i][j] = dp2[i + 1][j];
        if (j > 0) dp2[i][j] = (dp2[i][j] + dp2[i + 1][j - 1]) % MOD;
      }
      dp2[i][arr[i + 1]] = (dp2[i][arr[i + 1]] + 1) % MOD;
    }
    int[] cnt = new int[c + 1];
    for (int i = 1; i <= n; i++) {
      dp[i] = dp[i - 1];
      for (int j = 0; j <= c; j++) {
        if (dp2[i][j] > 0) {
          if (j > cnt[arr[i]]) dp[i] = (dp[i] + 1) % MOD;
          cnt[arr[i]]++;
        }
      }
    }
    for (int i = 0; i <= n; i++) System.out.print(dp[i] + " ");
  }
}
