import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        int[][] grid = new int[n][m];

        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < m; j++) {
                int num = Integer.parseInt(st.nextToken());
                grid[i][j] = num;
            }
        }

        int[][] rotatedGrid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                rotatedGrid[i][j] = grid[i][m - 1 - j];
            }
        }

        int[] rowSum = new int[n];
        int[] rowSumRotated = new int[n];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                rowSum[i] += grid[i][j];
                rowSumRotated[i] += rotatedGrid[i][j];
            }
        }

        Arrays.sort(rowSum);
        Arrays.sort(rowSumRotated);

        if (rowSum[n - 1] < rowSumRotated[n - 1]) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    System.out.print(grid[i][j] + " ");
                }
                System.out.println();
            }
        } else {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    System.out.print(rotatedGrid[i][j] + " ");
                }
                System.out.println();
            }
        }
    }
}
