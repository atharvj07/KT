#include <bits/stdc++.h>
using namespace std;
const int maxN = 500001;
long long a[maxN];
int main() {
  int q;
  cin >> q;
  while (q--) {
    int t, x, y;
    cin >> t >> x >> y;
    if (t == 1)
      a[x] += y;
    else if (t == 2) {
      long long sum = 0;
      for (int i = 1; i <= 500000; i++) {
        if (i % x == y) sum += a[i];
      }
      cout << sum << endl;
    }
  }
}
