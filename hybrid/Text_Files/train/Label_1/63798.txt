import sys
n=int(input())
a=list(map(int,input().split()))
if a==[1]*(n-1)+[3]:print("Infinite");sys.exit(0)
if a[0]==a[-1]:print("Infinite");sys.exit(0)
if [a[0],a[-1]]==[3,1]:print("Infinite");sys.exit(0)
c=0
for i in range(n-1):
    if a[i]==a[i+1]:c+=1
d=[1,1,2,2]
print("Finite")
print(n*d[a[0]-1]-c)
