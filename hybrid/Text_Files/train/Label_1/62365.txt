n, k = map(int,input().split())
a = list(map(int,input().split()))
t = [0] * n
vis = [0] * n
ms = max(a)
for j in range(ms,0,-1):
    for i in range(n):
        if a[i] == j and not vis[i]:
            l = i - 1
            r = i + 1
            while l >= 0 and r < n and not vis[l] and not vis[r] and a[l] > j - k and a[r] > j - k:
                vis[l] = 1
                vis[r] = 1
                l -= 1
                r += 1
            vis[i] = 1
            t[i] = 1
for j in range(1,ms + 1):
    for i in range(n):
        if a[i] == j and not vis[i]:
            l = i - 1
            r = i + 1
            while l >= 0 and r < n and not vis[l] and not vis[r] and a[l] > j - k and a[r] > j - k:
                vis[l] = 1
                vis[r] = 1
                l -= 1
                r += 1
            vis[i] = 1
            t[i] = 2
print(''.join(map(str,t)))
