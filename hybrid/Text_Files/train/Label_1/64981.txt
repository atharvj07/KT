import math

for _ in range(int(input())):
    n = int(input())
    a = []
    b = []
    for _ in range(n):
        x, y, u, v = map(int, input().split())
        a.append((x, y))
        b.append((u, v))
    ans = 0
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            ans += (a[i][0] - b[i][0]) * (b[j][1] - a[j][1]) > (b[i][0] - a[i][0]) * (a[j][1] - b[j][1])
    print(ans)
