#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, ans = -1, max = 0;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i].first >> v[i].second;
        if (v[i].first > v[i].second) swap(v[i].first, v[i].second);
        if (v[i].second - v[i].first > max) max = v[i].second - v[i].first, ans = i;
    }
    cout << ans + 1;
    return 0;
}
