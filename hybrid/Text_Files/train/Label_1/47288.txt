import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] x = new int[n];
        for (int i = 1; i < n; i++) {
            p[i] = sc.nextInt() - 1;
        }
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[] cnt = new int[5001];
        boolean[] used = new boolean[n];
        for (int i = n - 1; i >= 0; i--) {
            if (used[i]) {
                continue;
            }
            int j = i;
            int sum = 0;
            while (!used[j]) {
                sum += x[j];
                used[j] = true;
                j = p[j];
            }
            cnt[sum]++;
        }
        boolean flag = true;
        for (int i = 0; i < 5001; i++) {
            if (cnt[i] > 1) {
                flag = false;
            }
        }
        System.out.println(flag ? "POSSIBLE" : "IMPOSSIBLE");
    }
}
