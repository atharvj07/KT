import sys

in_s = sys.stdin.readline()
in_p = sys.stdin.readline()

p = in_p.strip()
m = len(p)

def is_subsequence(x, y):
    i = 0
    for j in range(len(y)):
        if x[i] == y[j]:
            i = i + 1
        if i >= len(x):
            return True
    return False

in_s = in_s.strip()
n = len(in_s)
s = ''
for i in range(n):
    bc = False
    for j in range(i + 1, n - m + 1):
        if is_subsequence(p, in_s[j:j + m]):
            bc = True
            break
    if bc:
        s = s + in_s[i]
    else:
        s = s + "!"

for i in range(n + 1):
    r = 0
    pos = s.find(p)
    while pos != -1:
        r = r + 1
        pos = s.find(p, pos + m)
    print(r, end=" ")
