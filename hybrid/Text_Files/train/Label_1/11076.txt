#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> dp;
vector<string> grid;
int h, w, q;
const int inf = 1e9;
int dx[] = {0, 1};
int dy[] = {1, 0};

int solve(int a, int b, int c, int d) {
  if (a > c || b > d) return 0;
  if (dp[a][b] != -1) return dp[a][b];
  int ans = 0;
  for (int i = 0; i < 2; i++) {
    if (a + 1 <= c) {
      int na = a + dx[i], nb = b + dy[i], nc = c, nd = d;
      if (grid[na][nb] == '.' && grid[nc][nd] == '.' && (na != nb || na == a))
        ans += solve(na, nb, nc, nd);
    }
    if (b + 1 <= d) {
      int na = a, nb = b + dx[i], nc = c, nd = d + dy[i];
      if (grid[na][nb] == '.' && grid[nc][nd] == '.' && (na != nb || na == a))
        ans += solve(na, nb, nc, nd);
    }
  }
  return dp[a][b] = ans;
}

int main() {
  cin >> h >> w;
  grid.resize(h + 1);
  for (int i = 1; i <= h; i++) cin >> grid[i];
  cin >> q;
  dp.assign(h + 1, vector<int>(w + 1, -1));
  while (q--) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    cout << solve(a, b, c, d) << endl;
  }
}
