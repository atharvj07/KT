#include<bits/stdc++.h>
using namespace std;
int32_t main()
{
   ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
   int t; cin >> t;
   while (t--)
   {
      uint64_t x; cin >> x;
      vector<int> ans;
      while (x != 0)
      {
         int bit = (x % 2);
         if (bit) { ans.push_back(int()); }
         else
         {
            if (ans.size() && ans.back() == 0) ans.pop_back();
            else ans.push_back(int());
         }
         x /= 2;
      }
      cout << ans.size() << '\n';
   }
   return 0;
}
