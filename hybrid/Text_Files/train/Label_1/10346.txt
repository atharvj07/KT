#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    if (k >= n)
      cout << 1 << endl;
    else if (n > 2 * k)
      cout << 1 << endl;
    else if (n == 2 * k)
      cout << 2 << endl;
    else
      cout << 0 << endl;
  }
}
