#include <bits/stdc++.h>
using namespace std;
int a[10], ans = 0, base = 1, mod = 100;
int main() {
  string s;
  cin >> s;
  int n = s.length(), m = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == 'X') { m++; }
    if (s[i] != 'X' && s[i] != '_') { a[m] = s[i] - '0'; }
    if (s[i] == '_') { a[m++] = 10; }
  }
  do {
    int sum = a[0];
    for (int i = 1; i < m; i++) {
      sum = (sum * base + a[i]) % mod;
    }
    ans += sum % 25 == 0;
  } while (prev_permutation(a, a + m));
  cout << ans << endl;
}
