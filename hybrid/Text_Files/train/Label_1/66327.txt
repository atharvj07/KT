import sys

n = int(input())
a = [int(x) for x in input().split()]
p = 0
for i in range(1,a[0]+1):
    for j in range(1,a[0]+1):
        if i != j:
            s1 = i + j
            for k in range(1,a[1]+1):
                if i != k and j != k:
                    s2 = s1 + k
                    for l in range(1,a[1]+1):
                        if i != l and j != l and k != l:
                            s3 = s2 + l
                            s4 = 0
                            for m in range(2,n+1):
                                s4 += a[m]
                            if s4 < s3:
                                p += 1
print(p / (n**3*(n-1)**3))
