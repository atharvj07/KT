#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int policeCnt = 0;
  int crimes = 0;
  while (n--) {
    int x;
    cin >> x;
    if (x == -1) {
      if (policeCnt == 0)
        crimes++;
      else
        policeCnt--;
    } else
      policeCnt += x;
  }
  cout << crimes << '\n';
}
