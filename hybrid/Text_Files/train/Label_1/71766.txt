n = int(input())
s = input()
a, c, g, t = s.count('A'), s.count('C'), s.count('G'), s.count('T')
x = min(a, c, g, t)
if (a + c + g + t - 4*x) % 4 != 0:
    print('===')
else:
    for i in range(n):
        if s[i] == '?':
            print(['A', 'C', 'G', 'T'][(x-(a-1)*(a>x)+(c-1)*(c>x)+(g-1)*(g>x)+(t-1)*(t>x)) % 4], end='')
        else:
            print(s[i],end='')
    print()
