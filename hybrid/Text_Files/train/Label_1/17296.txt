#include <vector>
#include <bits/stdc++.h>
#define MOD 1000000007
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;
typedef vector<vector<ll>> vvi;
typedef vector<ll> vi;
#define endl '\n'

const ll INF = 1e18;
const ll MAXN = 100001;

int main() {
    ios_base::sync_with_stdio(false);
    cout.tie(0);

    int n;
    cin >> n;

    vector<pi> students;
    for (int i = 0; i < n; i++) {
        ll a, b;
        cin >> a >> b;
        students.emplace_back(a, b);
    }

    sort(begin(students), end(students));

    ll ans = 0;
    for (int i = 0; i < n - 1; i++) {
        auto [a1, b1] = students[i];
        auto [a2, b2] = students[i + 1];

        if ((a1 & a2) != a2) {
            ans = max(ans, b1 + b2);
        }
    }

    cout << ans << endl;

    return 0;
}
