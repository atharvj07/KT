def is_x(n: int, arr: list) -> str:
  for i in range(n):
    if arr[i][i] != arr[i][n - i - 1]: return "NO"
  for i in range(n):
    for j in range(n):
      if i == j or i == n - j - 1: continue
      if arr[i][j] != arr[0][0]: return "NO"
  return "YES"

n = int(input())
arr = [input() for i in range(n)]
print(is_x(n, arr))
