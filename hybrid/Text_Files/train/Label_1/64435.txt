import sys
n,d,h=map(int,input().split())
if h>d or d-h>1 or h==d:print(-1);sys.exit()
t=[0]*n
for i in range(n-1):
    a,b=map(int,input().split())
    t[a-1]=t[a-1]+[b-1]
    t[b-1]=t[b-1]+[a-1]
u=[0]*n
def dfs(v,p,k):
    u[v]=1
    global c,z
    if k==h:z=v
    c=c+1
    for w in t[v]:
        if u[w]==0:dfs(w,v,k+1)
q=[0]*n
def dfs2(v,p,k,j):
    q[v]=j
    for w in t[v]:
        if u[w]==0:
            if k==1:dfs2(w,v,k+1,j)
            else:dfs2(w,v,k+1,j+1)
c=0
z=0
dfs(0,0,0)
if c!=n:print(-1);sys.exit()
dfs2(z,0,0,1)
for i in range(n-1):
    for j in range(i+1,n):
        if q[i]==q[j]:print(i+1,j+1)
