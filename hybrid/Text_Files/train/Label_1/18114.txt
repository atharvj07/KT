#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> b(n);
    string c;
    for (int &x : b) cin >> x;
    cin >> c;
    int wr = 0, wo = 0, ro = 0;
    for (int i = 0; i < n; i++) {
        if (c[i] == 'R')
            ro++;
        if (c[i] == 'W')
            wr++;
        if (c[i] == 'O')
            wo++;
    }
    if ((wr > 0 && ro > 0) || wr + ro + wo < k)
        return cout << -1, 0;
    sort(b.begin(), b.end(), greater<int>());
    int ans = 0;
    int w = 0, o = 0;
    for (int i = 0; i < n; i++) {
        if (k == 0) break;
        if (c[i] == 'R') {
            if (w + o > 0) continue;
            ans += b[i];
            k--;
        } else if (c[i] == 'W') {
            if (ro > 0) continue;
            ans += b[i];
            w++;
            k--;
        } else if (c[i] == 'O') {
            if (ro > 0) continue;
            ans += b[i];
            o++;
            k--;
        }
    }
    cout << ans << endl;
}
