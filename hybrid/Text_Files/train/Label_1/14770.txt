#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    int ans = 0;
    for (int i = 0; i < n - 1; i++) {
      if (s[i] == 'A' && s[i + 1] == 'P') {
        ans++;
        s[i + 1] = 'A';
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
