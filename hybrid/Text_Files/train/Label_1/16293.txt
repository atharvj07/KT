#include <bits/stdc++.h>
using namespace std;
double const eps = 1e-6, inf = 1e9;
int n, m;
struct p {
  int a, t;
};
p a[200001];
bool cmp(p x, p y) { return x.t < y.t; }
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) cin >> a[i].a;
  for (int i = 1; i <= n; ++i) cin >> a[i].t;
  sort(a + 1, a + 1 + n, cmp);
  int r = n;
  double ans = 0, ta = 0, tt = 0;
  for (int i = 1; i <= n; ++i) {
    if (i > 1)
      ans += a[i - 1].a;
    double k = ta * a[i].a + tt * a[i].t;
    double l = tt * a[i].a;
    if (i == n || fabs(k - l) > eps || fabs(tt + a[i].a - ta) > eps) {
      r = i - 1;
      break;
    }
    ta += a[i].a;
    tt += a[i].t;
  }
  if (fabs(tt - m) <= eps) {
    for (int i = 1; i <= r; ++i) ans += a[i].a;
    cout << fixed << setprecision(9) << ans << endl;
  } else
    cout << 0 << endl;
  return 0;
}
