import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        HashSet<Integer> levelsX = new HashSet<>();
        int p = sc.nextInt();
        for (int i = 0; i < p; i++) {
            levelsX.add(sc.nextInt());
        }
        HashSet<Integer> levelsY = new HashSet<>();
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            levelsY.add(sc.nextInt());
        }
        boolean possible = true;
        for (int i = 1; i <= n; i++) {
            if (!levelsX.contains(i) && !levelsY.contains(i)) {
                possible = false;
                break;
            }
        }
        System.out.println(possible ? "I become the guy." : "Oh, my keyboard!");
    }
}
