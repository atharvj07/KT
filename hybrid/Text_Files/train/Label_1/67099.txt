from collections import Counter

n, m = map(int, input().split())
st = [input() for _ in range(n)]
li = []
for i in range(m):
    ch = []
    for j in range(n):
        ch.append(st[j][i])
    li.append(ch)
ans = []
for ch in li:
    c = Counter(ch)
    if c['a'] % 2:
        ans.append(ch[c['a'] // 2])
for ch in li[::-1]:
    c = Counter(ch)
    if c['a'] % 2:
        ans.append(ch[c['a'] // 2])
print(len(''.join(ans)))
print(''.join(ans))
