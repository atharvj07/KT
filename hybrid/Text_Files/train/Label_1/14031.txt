#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  int l = 0, r = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'z') {
      l = i;
      break;
    }
  }
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == 'a') {
      r = i;
      break;
    }
  }
  for (int i = l; i <= r; i++) {
    if (s[i] == 'a') s[i] = 'z';
    else s[i]--;
  }
  cout << s << endl;
  return 0;
}
