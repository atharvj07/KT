n = int(input())
m = list(map(int, input().split()))

left = [0] * n
right = [0] * n

stack = []
for i in range(n):
    while stack and m[stack[-1]] >= m[i]:
        stack.pop()
    left[i] = stack[-1] if stack else -1
    stack.append(i)

stack = []
for i in range(n-1, -1, -1):
    while stack and m[stack[-1]] >= m[i]:
        stack.pop()
    right[i] = stack[-1] if stack else n
    stack.append(i)

a = [0] * n
for i in range(n):
    a[i] = min(m[i], m[left[i]] if left[i] != -1 else m[i], m[right[i]] if right[i] != n else m[i])

print(*a)
