#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int md = 1e9 + 7;
ll f[2][1000001];
int main() {
  ll m;
  cin >> m;
  ll h1, a1, x1, y1, h2, a2, x2, y2;
  cin >> h1 >> a1;
  cin >> x1 >> y1;
  cin >> h2 >> a2;
  cin >> x2 >> y2;
  memset(f, -1, sizeof f);
  f[0][h1] = f[1][h2] = 0;
  ll ans = -1;
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < m; j++) {
      if (f[0][j] != -1) {
        f[0][(x1 * j + y1) % m] = f[0][j] + 1;
        if (f[0][(x1 * j + y1) % m] == a1 && f[1][j] == a2) { ans = i + 1; }
      }
      if (f[1][j] != -1) {
        f[1][(x2 * j + y2) % m] = f[1][j] + 1;
        if (f[0][j] == a1 && f[1][(x2 * j + y2) % m] == a2) { ans = i + 1; }
      }
    }
  }
  cout << ans << endl;
}
