import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int D = sc.nextInt();
        int[] F = new int[N];
        int[] t = new int[N];
        for (int i = 0; i < N; i++) {
            F[i] = sc.nextInt();
            t[i] = sc.nextInt();
        }
        Arrays.sort(F);
        Arrays.sort(t);
        for (int i = 0; i < N; i++) {
            if (F[i] > t[i] + D - 1) {
                System.out.println(-1);
                return;
            }
        }
        int ans = 0;
        int j = 0;
        for (int i = 0; i < N; i++) {
            while (j < N && F[j] <= t[i]) {
                j++;
            }
            if (j == N || F[j] > t[i] + D - 1) {
                j--;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
