import sys

n, k = map(int, input().split())

graph = [[] for _ in range(n+1)]
for _ in range(n-1):
    x, y = map(int, input().split())
    graph[x].append(y)
    graph[y].append(x)

visited = [False] * (n+1)
cnt = [0] * (n+1)
ans = []
def dfs(now, pre, cmp):
    visited[now] = True
    cnt[cmp] += 1
    ans.append(cmp)
    for nxt in graph[now]:
        if nxt == pre: continue
        if not visited[nxt]: dfs(nxt, now, cmp)

cmp = 1
for i in range(1, n+1):
    if not visited[i]:
        dfs(i, 0, cmp)
        cmp += 1

# compress
compress = {}
_ans = []
for x in ans:
    if not x in compress:
        compress[x] = len(compress) + 1
    _ans.append(compress[x])

if len(compress) <= n-1-k:
    print(len(compress))
    print(*_ans)
else:
    print(n-1-k)
    print(*_ans[:n-1-k])
