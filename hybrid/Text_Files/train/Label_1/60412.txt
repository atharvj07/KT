import queue
n = int(input())
links = [[] for i in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    links[a].append(b)
    links[b].append(a)
leaves = [0 for i in range(n + 1)]
used = [0 for i in range(n + 1)]
que = queue.Queue()
que.put(1)
while not que.empty():
    now = que.get()
    cnt = 0
    for next in links[now]:
        if used[next] == 0:
            cnt += 1
            que.put(next)
            used[next] = 1
    if cnt == 1: leaves[now] = 1
l = []
for i in range(1, n + 1):
    if leaves[i] > 0: l.append(i)
l.sort()
print(l[-1], l[0])
