#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m, a, b, g0, x, y, z, cur = 0, sum = 0;
  cin >> n >> m >> a >> b >> g0 >> x >> y >> z;
  vector<vector<int>> h(n + 1, vector<int>(m + 1));
  vector<vector<long long>> pre(n + 1, vector<long long>(m + 1));
  vector<int> pow(max(n, m));
  h[1][1] = g0;
  pre[1][1] = g0;
  pow[0] = 1;
  for (int i = 1; i <= max(n, m); i++) pow[i] = (pow[i - 1] * x) % z;
  for (int i = 1; i <= n; i++) {
    for (int j = 2; j <= m; j++) {
      h[i][j] = (h[i - 1][j] * x + y) % z;
      pre[i][j] = pre[i][j - 1] + h[i][j];
    }
  }
  for (int i = 2; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      pre[i][j] += pre[i - 1][j];
    }
  }
  for (int i = 1; i <= n - a + 1; i++) {
    for (int j = 1; j <= m - b + 1; j++) {
      cur = pre[i + a - 1][j + b - 1];
      if (i > 1) cur -= pre[i - 1][j + b - 1];
      if (j > 1) cur -= pre[i + a - 1][j - 1];
      if (i > 1 && j > 1) cur += pre[i - 1][j - 1];
      sum += cur;
    }
  }
  cout << sum << endl;
}
