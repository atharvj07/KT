#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 2e5;

vector<int> adj[MAXN + 1];
bool vis[MAXN + 1];

int dfs(int u, int p) {
    vis[u] = true;
    int dist = 0;
    for (int v : adj[u]) {
        if (v == p)
            continue;
        if (!vis[v])
            dist = max(dist, dfs(v, u) + 1);
    }
    return dist;
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (!vis[i])
            ans += dfs(i, 0) > 2;
    }
    cout << ans << endl;
    return 0;
}
