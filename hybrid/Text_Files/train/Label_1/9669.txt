#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define endl "\n"

const int mxn = 1000;
vector <int> adj[mxn+5];
bool vis[mxn+5];
int gov[mxn+5];

int DFS(int x, int par=-1){
    vis[x] = true;
    int ans = 0;
    for(int y: adj[x]){
        if(!vis[y]){
            ans += DFS(y,x);
        }
        else if(y!=par && gov[x]!=gov[y]) return -1;
    }
    return ans+1;
}

int main() {
    int n,m,k; cin >> n >> m >> k;
    for(int i=0; i<k; i++){
        int v; cin >> v;
        gov[v] = i+1;
    }
    for(int i=0; i<m; i++){
        int u,v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int temp = 0;
    for(int i=1; i<=n; i++){
        if(!vis[i]) temp += (DFS(i) == 1);
    }
    cout << n-1-m-temp << endl;
    return 0;
}
