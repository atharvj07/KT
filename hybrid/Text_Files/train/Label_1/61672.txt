def min_subscriptions(n, k, d, schedule):
    unique_shows = set(schedule[:d])
    min_subs = len(unique_shows)
    for i in range(1, n - d + 1):
        unique_shows = set(schedule[i:i+d])
        min_subs = min(min_subs, len(unique_shows))
    return min_subs

t = int(input().strip())
for _ in range(t):
    n, k, d = map(int, input().strip().split())
    schedule = list(map(int, input().strip().split()))
    print(min_subscriptions(n, k, d, schedule))
