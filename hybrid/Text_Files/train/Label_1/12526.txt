#include <iostream>
#include <vector>
const int mxN=1e6;
int maxd[mxN],mind[mxN];
void pre(){
    for(int i=1;i<mxN;i++){
        int x=i,least=1e9,maxi=0,cur=0,cnt=0;
        while(x){
            if(x%10<least)least=x%10;
            if(x%10>maxi)maxi=x%10;
            x/=10;cnt++;
        }
        maxd[i]=maxi;
        mind[i]=least;
    }
}
int main(){
    pre();
    int t, n, k;
    long long a;
    std::cin >> t;
    while(t--){
        std::cin >> n >> k;
        a = n;
        while(k--){
            a+=mind[a]*maxd[a];
        }
        std::cout << a << "\n";
    }
    return 0;
}
