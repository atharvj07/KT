#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[5];
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    if (n == 4) {
      for (int i = 1; i <= 4; ++i) {
        for (int j = i + 1; j <= 4; ++j) {
          for (int k = j + 1; k <= 4; ++k) {
            if (a[i] == a[j] && a[j] == a[k]) {
              cout << "NO\n";
              return 0;
            }
          }
        }
      }
    }
    cout << "YES\n";
    for (int i = 1; i <= 4 - n; ++i) {
      if (n == 0) cout << (a[1] + a[2] + a[3] + a[4]) / 4 << '\n';
      if (n == 1) cout << (a[1] + a[2] + a[3] - a[4]) / 2 << '\n';
      if (n == 2) cout << (a[1] + a[2] - a[3] - a[4]) << '\n';
      if (n == 3) cout << (a[1] - a[2] - a[3] - a[4]) * 2 << '\n';
    }
    return 0;
}
