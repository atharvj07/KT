import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = i % 2 == 0 ? i * 3 + 1 : i * 3 + 2;
            }
            for (int i : arr) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
