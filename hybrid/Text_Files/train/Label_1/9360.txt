#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll N = 1e6 + 5;
ll n, m, a[N];
int main() {
    cin >> n >> m;
    if (n == 1) {
        cout << "YES\n"; for (ll i = 1; i <= m; i++)cout << i << " "; cout << "\n";
    }
    else if (m == 1) {
        cout << "YES\n"; for (ll i = 1; i <= n; i++)cout << i << "\n";
    }
    else if (n * m % 2 != 0)cout << "NO\n";
    else {
        cout << "YES\n"; for (ll i = 1; i <= n; i++) {
            for (ll j = 1; j <= m; j++) { a[m * (i - 1) + j] = j + m * (i - 1); }
            if (i % 2 == 1) {
                for (ll j = 2; j <= m - 1; j += 2) {
                    swap(a[m * (i - 1) + j], a[m * (i - 1) + j + 1]);
                }
            }
            for (ll j = 1; j <= m; j++)cout << a[m * (i - 1) + j] << " ";
            cout << "\n";
        }
    }
}
