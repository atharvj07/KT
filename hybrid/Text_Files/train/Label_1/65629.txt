from collections import defaultdict

t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    pref = [0] * (n + 1)
    for i in range(n):
        pref[i + 1] = (pref[i] + a[i]) % x
    cnt = defaultdict(int)
    cnt[0] = 1
    ans = 0
    for pre in pref:
        ans = max(ans, cnt[pre] + 1)
        cnt[pre] += 1
    print(-1 if ans == 1 else ans)
