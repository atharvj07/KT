t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    distinct_values = len(set(a))
    if k == 1:
        if distinct_values == 1:
            print(1)
        else:
            print(-1)
    else:
        print(max(1, (distinct_values - 2) // (k - 1) + 1))
