from sys import stdin

n = int(stdin.readline())
a = [int(x) for x in stdin.readline().split()]
mn = min(a)
ind = a.index(mn)
ans = mn
while len(a) > 1:
    if a[ind - 1] == a[ind]: ind -= 1
    else: ind += 1
    ans = max(a[ind], ans)
    a[ind] += min(a[ind - 1], a[(ind + 1) % len(a)])
    del a[ind - 1]
    ind %= len(a)
print(ans)
