#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
    int n, m, d;
    cin >> n >> m >> d;
    vector<int> c(m);
    for (int &x : c) cin >> x;
    int r = (n - 1 + d) / d;
    vector<int> ans(r);
    for (int i = 1; i <= r; i++) ans[i - 1] = (i + d - 1 + r - 1) / r;
    vector<int> t(n + 2);
    for (int i = 0; i < m; i++) {
        int p = 0;
        for (int j = 1; j <= n + 2; j++) {
            if (!t[j]) {
                if (p == ans[i]) t[j] = i + 1;
                p++;
            }
        }
    }
    int rem = 0;
    for (int i = 1; i <= n + 1; i++) {
        if (!t[i]) rem++;
        if (rem) t[i] = t[i + 1];
    }
    cout << "YES\n";
    for (int i = 1; i <= n + 1; i++) t[i] = (t[i] ? t[i] : t[i - 1]);
    int now = 0;
    for (int i = 0; i < r; i++) {
        int e = (i + d) / ans[i];
        cout << (now + 1) << ' ';
        now += e - 1;
        now = t[now + 1];
    }
    cout << endl;
}
