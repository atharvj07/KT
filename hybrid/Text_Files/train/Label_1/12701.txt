#include <iostream>
#include <string>
#include <set>
#include <vector>
using namespace std;
const int MAX_LEN = 1e4 + 5;
string s;
int n;
vector<string> suf;
set<string> ans;
int main() {
  getline(cin, s);
  n = s.length();
  for (int i = 4; i < n; i++) {
    bool ok = 0;
    vector<string> temp;
    string cur = s.substr(i - 4, 5);
    for (int j = i + 2; j < n; j++) {
      string str = s.substr(j - 2, 3);
      if (cur.substr(cur.length() - 2) == str)
        ok = 1;
      else
        temp.push_back(str);
      cur += str;
    }
    if (!ok)
      for (int i = 0; i < temp.size(); i++) suf.push_back(temp[i]);
  }
  for (auto v: suf) { ans.insert(v); }
  cout << ans.size() << endl;
  for (auto v: ans) { cout << v << endl; }
  return 0;
}
