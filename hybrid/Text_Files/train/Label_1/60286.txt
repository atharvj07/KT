def can_convert(n, k, a, b):
    # Count frequency of each character in both strings
    freq_a = [0] * 26
    freq_b = [0] * 26
    
    for i in range(n):
        freq_a[ord(a[i]) - ord('a')] += 1
        freq_b[ord(b[i]) - ord('a')] += 1
    
    for i in range(25): # From 'a' to 'y'
        # If a has more characters than b at any point, return No
        if freq_a[i] < freq_b[i]:
            return "No"
        # Distribute extra characters to next if possible
        diff = freq_a[i] - freq_b[i]
        if diff % k == 0:
            freq_a[i+1] += diff
        else:
            return "No"
    
    return "No" if freq_a[25] != freq_b[25] else "Yes"

# Input handling
t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = input()
    b = input()
    print(can_convert(n, k, a, b))
