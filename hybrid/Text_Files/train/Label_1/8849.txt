#include <bits/stdc++.h>
const int N = 100005;
int n, m, k[N], l[N], r[N], sum[N];
char str[N];
int main() {
  scanf("%s%d", str, &m);
  int len = strlen(str);
  while (m--) {
    scanf("%d%d%d", &l[m], &r[m], &k[m]);
    sum[l[m] - 1] += k[m];
    sum[r[m]] -= k[m];
  }
  for (int i = 1; i < len; i++) sum[i] += sum[i - 1];
  for (int i = 0; i < len; i++) {
    str[(i + sum[i]) % len] = str[i];
  }
  printf("%s", str);
  return 0;
}
