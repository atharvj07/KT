#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int can = 1;
  for (int i = n - 2; i >= 0; i--) {
    if (a[i] == 0) continue;
    if (a[i] % k == 0) continue;
    can = 0;
    break;
  }
  cout << (can ? "Yes" : "No") << endl;
}
