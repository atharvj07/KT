#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> adj(n + 1);
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    adj[x].push_back(y);
    adj[y].push_back(x);
  }
  vector<int> deg(n + 1, 0);
  for (int i = 1; i <= n; i++) {
    deg[i] = adj[i].size();
  }
  vector<int> ans(m + 1, 0);
  for (int i = 1; i <= m; i++) {
    int mx = 0;
    for (int j = 1; j <= n; j++) {
      if (deg[j] >= k) {
        ans[i] = n;
        break;
      }
      mx = max(mx, deg[j]);
    }
    if (ans[i] == 0) {
      ans[i] = mx + 1;
    }
    for (int j = 1; j <= n; j++) {
      for (int u : adj[j]) {
        deg[u]--;
      }
    }
  }
  for (int i = 1; i <= m; i++) {
    cout << ans[i] << endl;
  }
  return 0;
}
