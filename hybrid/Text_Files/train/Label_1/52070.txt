import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s1 = sc.next(), s2 = sc.next();
        int cnt1 = 0, cnt2 = 0;
        for(int i=0; i<n; i++){
            if(s1.charAt(i)=='1') cnt1++;
            if(s2.charAt(i)=='1') cnt2++;
        }
        if(cnt2==1) System.out.println(s2);
        else if(cnt1==0) System.out.println(s1);
        else{
            boolean flg = false;
            String ans = "";
            for(int i=0; i<n; i++){
                if(s1.charAt(i)!=s2.charAt(i) && flg==false){
                    flg = true;
                    ans+='1';
                }
                else if(s1.charAt(i)==s2.charAt(i) && flg==true) ans+='0';
                else ans+=s1.charAt(i);
            }
            System.out.println(ans);
        }
    }
}
