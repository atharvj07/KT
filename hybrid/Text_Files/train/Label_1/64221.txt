from collections import defaultdict, deque
n,m=map(int,input().split())
a=[0]*(m+1)
b=[0]*(m+1)
c=defaultdict(list)
for i in range(1,m+1):
    x,y=map(int,input().split())
    a[i],b[i]=min(x,y),max(x,y)
    c[x].append(i)
    c[y].append(i)
q=deque()
for x in c:
    if len(c[x])%2: q.append(x)
ans=0
vis=[False]*(n+1)
while q:
    x=q.popleft()
    vis[x]=True
    ans+=1
    for i in c[x]:
        if a[i]==x: y=b[i]
        else: y=a[i]
        if vis[y]: continue
        c[y].remove(i)
        if len(c[y])%2: q.append(y)
print(ans)
