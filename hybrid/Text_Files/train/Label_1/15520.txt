#include <bits/stdc++.h>
using namespace std;

const int maxn = 100005;
long long a[maxn], b[maxn], c[maxn];
bool check[maxn];
long long ans = 0;
vector<long long> ans_list;

long long solve(int n) {
  long long ma = 0;
  for (int i = 1; i <= n; i++) {
    if (check[i] == false) {
      vector<long long> to_add;
      to_add.push_back(a[i]);
      to_add.push_back(b[i]);
      to_add.push_back(c[i]);
      sort(to_add.begin(), to_add.end());
      for (int j = i + 1; j <= n; j++) {
        if (check[j] == false) {
          vector<long long> now;
          now.push_back(a[j]);
          now.push_back(b[j]);
          now.push_back(c[j]);
          sort(now.begin(), now.end());
          if (now[0] == to_add[0] && now[1] == to_add[1]) {
            long long area = (to_add[2] + now[2]) * (to_add[2] + now[2]);
            if (area >= ma) {
              ma = area;
              ans_list.clear();
              ans_list.push_back(i);
              ans_list.push_back(j);
            }
          }
        }
      }
      for (int j = 0; j < to_add.size(); j++) {
        check[i] = true;
      }
    }
  }
  if (ans_list.size() == 0) {
    for (int i = 1; i <= n; i++) {
      if (ma < a[i] * a[i] * a[i]) {
        ma = a[i] * a[i] * a[i];
        ans = i;
      }
    }
  }
  return ma;
}

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i] >> c[i];
  }
  long long ans_value = solve(n);
  cout << ans_list.size() << endl;
  for (int i = 0; i < ans_list.size(); i++) {
    cout << ans_list[i] << " ";
  }
  return 0;
}
