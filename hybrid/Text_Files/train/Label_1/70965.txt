n = int(input())

# Calculate the optimal size of blocks
block_size = int(n**0.5) + 1

# Initialize the result list
result = []

# Fill the result list with elements in blocks of decreasing order
for i in range(1, n + 1, block_size):
    for j in range(min(n, i + block_size - 1), i - 1, -1):
        result.append(j)

# Print the result
for num in result:
    print(num, end=" ")
