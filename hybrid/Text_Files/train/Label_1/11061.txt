#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int SZ = 3e5 + 5;

ll a[SZ];
bitset<SZ> b[31];
int n, q;

int main() {
  cin >> n >> q;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    for (int j = 0; j < 31; ++j) {
      if ((a[i] >> j) & 1) b[j].set(i);
    }
  }
  while (q--) {
    int x, y;
    cin >> x >> y;
    bool ok = false;
    for (int i = 0; i < 31; ++i) {
      if (b[i][x] && b[i][y]) { ok = true; break; }
    }
    cout << (ok ? "Shi\n" : "Fou\n");
  }
  return 0;
}
