import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextInt();
        }
        long sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += a[i];
            if (sum * 2 >= sum) {
                System.out.println(i);
                break;
            }
        }
    }
}
