#include <iostream>
#include <vector>
#include <stack>
#include <queue>
#include <unordered_map>
#include <algorithm>
#include <functional>

using namespace std;

const unsigned int inf = 0x3f3f3f3f;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<unordered_map<int, bool>> adj(n + 2);
        vector<int> path;
        for (int i = 1; i <= n; i++) {
            int a;
            cin >> a;
            adj[i][a?n+1:i+1] = true;
            adj[a?n+1:i+1][i] = true;
        }
        queue<int> q;
        q.push(1);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            path.push_back(u);
            for (auto v : adj[u]) {
                if (v.second) {
                    adj[u].erase(v.first);
                    adj[v.first].erase(u);
                    if (adj[v.first].empty())
                        q.push(v.first);
                }
            }
        }
        if (path.size() == n + 1) {
            for (int u : path)
                cout << u << ' ';
            cout << endl;
        } else {
            cout << -1 << endl;
        }
    }
    return 0;
}
