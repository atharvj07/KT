import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int vp = sc.nextInt();
        int vd = sc.nextInt();
        int t = sc.nextInt();
        int f = sc.nextInt();
        int c = sc.nextInt();
        int time = t + f;
        int dist = vp * time;
        int ans = (int) Math.ceil(((double) (c - dist)) / (double) (vd * time));
        System.out.println(ans);
    }
}
