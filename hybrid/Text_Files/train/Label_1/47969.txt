import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                b[i]++;
            }
            if (i + 1 < n && a[i + 1] == 1) {
                b[i]++;
                b[i + 1]++;
            }
            if (i + 2 < n && a[i + 2] == 1) {
                b[i]++;
                b[i + 1]++;
                b[i + 2]++;
            }
        }
        for (int i = 0; i < n; i++) {
            if (b[i] != a[i]) {
                System.out.println(-1);
                return;
            }
        }
        int t = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < b[i]; j++) {
                System.out.println(i + 1 + " " + (j + 1));
                t++;
            }
        }
        System.out.println(t);
    }
}
