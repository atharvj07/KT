import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long p = scanner.nextLong();
        if ((n + p) % 2 == 1 || (n - p) % 2 == 1) {
            System.out.println(-1);
            return;
        }
        long answer = (long) Math.ceil((n - p) / 2.0);
        System.out.println(answer);
    }
}
