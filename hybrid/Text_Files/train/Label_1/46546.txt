import java.util.*;

public class Main {
    int n;
    int[] p;
    int[] a;
    int[] b;
    int[] depth;
    List<Integer>[] g;

    void dfs(int v, int par) {
        p[v] = par;
        depth[v] = par == -1 ? 0 : depth[par] + 1;
        for (int u : g[v]) {
            if (u == par) continue;
            dfs(u, v);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int u = sc.nextInt() - 1;
        int v = sc.nextInt() - 1;
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt() - 1;
            b[i] = sc.nextInt() - 1;
        }
        Main m = new Main();
        m.n = n;
        m.p = new int[n];
        m.a = a;
        m.b = b;
        m.depth = new int[n];
        m.g = new List[n];
        for (int i = 0; i < n; i++) m.g[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            m.g[a[i]].add(b[i]);
            m.g[b[i]].add(a[i]);
        }
        m.dfs(u, -1);
        int ans = m.depth[v] - m.depth[u];
        if (ans % 2 == 0) ans--;
        System.out.println(ans);
    }
}
