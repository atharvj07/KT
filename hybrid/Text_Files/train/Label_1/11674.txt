#include<bits/stdc++.h>
using namespace std;
#define int long long
#define ld  long double
#define pb push_back
#define ppb pop_back
#define mp make_pair
#define ff first
#define ss second
#define all(x) (x).begin(),(x).end()
#define uniq(x) sort(all(x)),(x).resize(unique(all(x))-x.begin())
#define sz(x) (int)((x).size())
#define fr(i,a,b) for(int i=a;i<b;i++)
#define mod 1000000007
const int N=200005;
vector<int> adj[N];
vector<int> dist[N];
map<pair<int,int>,int> wt;
int n,m,x[N],ans[N];
void dfs(int node,int par,int w)
{
    dist[node].pb(w);
    for(int i=0;i<adj[node].size();i++)
    {
        int v=adj[node][i];
        if(v!=par)
        dfs(v,node,w+wt[{node,v}]);
    }
}
int32_t main()
{
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            int x;cin>>x;
            if(i!=j)
            {
                wt[{i,j}]=wt[{j,i}]=x;
                adj[i].pb(j);
                adj[j].pb(i);
            }
        }
    }
    for(int i=1;i<=n;i++)cin>>x[i];
    for(int i=1;i<=n;i++)
    dfs(i,-1,0);
    for(int i=1;i<=n;i++)
    {
        cout<<0<<" ";
    }
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        if(i!=j)
        {
            ans[x[i]]+=dist[j][sz(dist[j])-1];
        }
    }
    for(int i=2;i<=n;i++)
    cout<<ans[i]<<" ";
    return 0;
}
