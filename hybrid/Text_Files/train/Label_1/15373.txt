#include<bits/stdc++.h>
 
using namespace std;

#define ll long long

const int maxN = 200005;

char s[maxN];
int ans[maxN];

int main() {
    int n, x, y;
    cin >> n;
    cin >> s;
    cin >> x >> y;

    int dx = 0;
    int dy = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'U') {
            dy++;
        }
        else if (s[i] == 'D') {
            dy--;
        }
        else if (s[i] == 'L') {
            dx--;
        }
        else {
            dx++;
        }
    }

    int changeX = 0, changeY = 0;
    if (dx != x) {
        changeX = 1;
    }
    if (dy != y) {
        changeY = 1;
    }

    if (changeX == 0 && changeY == 0) {
        cout << 0 << '\n';
        return 0;
    }
    else if (changeX != changeY) {
        cout << -1 << '\n';
        return 0;
    }
    else {
        int idx = -1;
        for (int i = n - 1; i >= 0; i--) {
            if (s[i] == 'U' || s[i] == 'D') {
                idx = i;
                break;
            }
        }
        cout << idx + 1 << '\n';
        return 0;
    }
    return 0;
}
