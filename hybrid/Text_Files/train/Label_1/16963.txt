#include<bits/stdc++.h>
using namespace std;
int n,ans;
int isbroken[100];
int st[7]={128,64,32,16,8,4,2};
int main()
{
    cin>>n;
    for(int i=0;i<7;++i)
    {
        if((n&st[i])==0)isbroken[i]=1;
    }
    for(int i=0;i<=99;++i)
    {
        int ok=1;
        for(int j=0;j<7;++j)
        {
            if(isbroken[j])
            {
                if(i&st[j])ok=0;
            }
            else
            {
                if((i&st[j])==0)ok=0;
            }
        }
        if(ok)ans++;
    }
    cout<<ans<<endl;
    return 0;
}
