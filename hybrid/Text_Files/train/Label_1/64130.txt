k = int(input())
sequences = []
for i in range(k):
    n = int(input())
    seq = list(map(int, input().split()))
    sequences.append((sum(seq), seq, i))

potential_pairs = {}
for total, seq, idx in sequences:
    for j, val in enumerate(seq):
        new_sum = total - val
        if new_sum in potential_pairs:
            potential_pairs[new_sum].append((idx, j))
        else:
            potential_pairs[new_sum] = [(idx, j)]

for key, value in potential_pairs.items():
    if len(value) >= 2:
        print("YES")
        for i in range(len(value) - 1):
            if value[i][0] != value[i + 1][0]:
                print(value[i][0] + 1, value[i][1] + 1)
                print(value[i + 1][0] + 1, value[i + 1][1] + 1)
                exit(0)
print("NO")
