import sys

n = int(input())
a = list(int(x) for x in input().split())
if n == 1:
    print(-1)
    sys.exit()
if a[0] + a[1] == 0:
    print(-1)
    sys.exit()
if n > 2:
    for i in range(1, n - 1):
        if a[i - 1] + a[i] + a[i + 1] == 0:
            print(-1)
            sys.exit()
print(n - 1)
