import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for(int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[][] ques = new int[m][2];
        for(int i = 0; i < m; i++) {
            ques[i][0] = sc.nextInt();
            ques[i][1] = sc.nextInt();
        }
        int[][] dp = new int[n + 1][n + 1];
        for(int i = 1; i <= n; i++) {
            dp[i][i] = a[i];
        }
        for(int j = 2; j <= n; j++) {
            for(int i = j - 1; i >= 1; i--) {
                dp[i][j] = Math.min(dp[i - 1][j], dp[i - 1][j - 1]) + a[j];
            }
        }
        for(int[] que : ques) {
            System.out.println(dp[que[0]][que[1]]);
        }
    }
}
