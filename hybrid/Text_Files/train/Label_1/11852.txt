#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

const long long INF = 1e18;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; ++i) cin >> p[i];
  if (n == 1) {
    cout << 1 << "\n";
    return 0;
  }
  vector<long long> lis(n, 1), lds(n, 1);
  for (int i = 1; i < n; ++i) {
    for (int j = i - 1; j >= 0; --j) {
      if (p[i] > p[j]) lis[i] = max(lis[i], lis[j] + 1);
    }
  }
  for (int i = n - 2; i >= 0; --i) {
    for (int j = i + 1; j < n; ++j) {
      if (p[i] > p[j]) lds[i] = max(lds[i], lds[j] + 1);
    }
  }
  long long min_sum = INF;
  int min_lis = 0, min_lds = 0;
  for (int i = 0; i < n; ++i) {
    if (lis[i] + lds[i] < min_sum) {
      min_sum = lis[i] + lds[i];
      min_lis = lis[i];
      min_lds = lds[i];
    }
  }
  vector<int> ans;
  for (int i = 0; i < n; ++i) {
    if (lis[i] == min_lis && lds[i] == min_lds) {
      ans.push_back(p[i]);
      if (ans.size() == n) break;
    }
  }
  for (int i = 0; i < n; ++i) {
    if (lis[i] == min_lds && lds[i] == min_lis) {
      ans.push_back(p[i]);
      if (ans.size() == n) break;
    }
  }
  for (int i : ans) cout << i << " ";
  cout << "\n";
  return 0;
}
