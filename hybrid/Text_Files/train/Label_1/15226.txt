#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
int n, a[N];
int main() {
	cin >> n;
	for(int i = 1; i <= n; i++) cin >> a[i];
	int ans = 0;
	ll sumodd = a[1], sumeven = 0;
	for(int i = 3; i <= n; i++) {
		if(i % 2) sumodd += a[i];
		else sumeven += a[i];
		if(sumodd == sumeven) ans++;
	}
	cout << ans;
	return 0;
}
