#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, k, ans = 0, lib[85] = {};
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        if (lib[a]) {
            lib[a]--;
        } else if (ans + 1 <= k) {
            lib[a]++;
            ans++;
        } else {
            int max = -1, idx;
            for (int j = 1; j <= n; j++) {
                if (lib[j] > max) {
                    max = lib[j];
                    idx = j;
                }
            }
            lib[idx]--;
            lib[a]++;
        }
    }
    cout << ans;
}
