#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> top(n), bottom(n);
    string s1, s2;
    cin >> s1 >> s2;
    for (int i = 0; i < n; i++) {
      top[i] = s1[i] - '0';
      bottom[i] = s2[i] - '0';
    }
    if (top[0] == 6 || bottom[0] == 3 || bottom[n - 1] == 2) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
