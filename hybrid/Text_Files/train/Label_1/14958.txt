#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
struct lecture {
  int sa, ea, sb, eb;
};
bool overlap(lecture& a, lecture& b) { return max(a.sa, b.sa) <= min(a.ea, b.ea); }
int main() {
  int n;
  cin >> n;
  vector<lecture> v(n);
  for (auto& l : v) cin >> l.sa >> l.ea >> l.sb >> l.eb;
  sort(v.begin(), v.end(),
       [](lecture& a, lecture& b) { return a.sa + a.ea > b.sa + b.ea; });
  bool happy = true;
  for (int i = 1; i < n; i++) {
    if (overlap(v[i - 1], v[i])) {
      happy = false;
      break;
    }
  }
  cout << (happy ? "YES" : "NO") << endl;
}
