#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, m;
    cin >> n >> m;
    vector<pair<long long, long long>> v(m);
    for (auto &x : v) cin >> x.first >> x.second;
    sort(v.rbegin(), v.rend());
    long long ans = 0;
    for (auto x : v) {
      long long t = min(n, x.first);
      n -= t;
      ans += t * (x.first + (t - 1ll) * x.second);
    }
    cout << ans << '\n';
  }
  return 0;
}
