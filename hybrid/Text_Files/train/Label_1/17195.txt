#include <iostream>
#include <vector>
using namespace std;
int adj[3001][3001]; // adjacency matrix
bool visited[3001];
int find_damn_rhombi(int n) {
  int count = 0;
  for (int i = 1; i <= n; i++) {
    if (!visited[i]) {
      int a = -1, b = -1, c = -1, d = -1;
      vector<int> path;
      visited[i] = true;
      path.push_back(i);
      while (!path.empty()) {
        int cur = path.back();
        path.pop_back();
        if (adj[cur][i]) {
          if (a == -1)
            a = cur;
          else if (b == -1)
            b = cur;
        }
        for (int j = 1; j <= n; j++) {
          if (!visited[j] && adj[cur][j]) {
            if (c == -1)
              c = j;
            else if (d == -1)
              d = j;
            visited[j] = true;
            path.push_back(j);
          }
        }
      }
      if (a != -1 && b != -1 && c != -1 && d != -1 && a != c && b != d &&
          adj[a][c] && adj[b][d] && adj[c][b] && adj[d][a]) {
        count++;
      }
    }
  }
  return count / 2;
}
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    adj[a][b] = 1;
  }
  cout << find_damn_rhombi(n) << endl;
  return 0;
}
