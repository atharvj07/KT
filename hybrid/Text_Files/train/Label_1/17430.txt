#include<bits/stdc++.h>
#define ll long long int
using namespace std;
const ll mod = 1e9+7;
ll f[100001][21];
ll pw[100001];
ll gcd(ll a,ll b)
{
    if(b==0)
    {
        return a;
    }
    return gcd(b,a%b);
}
int main()
{
    ll n;
    cin>>n;
    vector<ll>v(n+1);
    for(ll i=1;i<=n;i++)
    {
        cin>>v[i];
    }
    pw[0]=1;
    for(ll i=1;i<=100000;i++)
        pw[i]=(pw[i-1]*2)%mod;
    f[n+1][0]=1;
    for(ll i=n;i>=1;i--)
    {
        for(ll j=0;j<=20;j++)
        {
            if(j==0)
            {
                f[i][0]=1;
            }
            else if(gcd(v[i],j)==1)
            {
                f[i][j]=(f[i+1][j]+f[i+1][0])%mod;
            }
            else
            {
                f[i][j]=f[i+1][j]%mod;
            }
        }
    }
    ll ans=f[1][0];
    ans=((ans-1)+mod)%mod;
    cout<<ans<<endl;
}
