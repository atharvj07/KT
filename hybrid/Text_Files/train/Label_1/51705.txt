import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();

        boolean[] used = new boolean[n + 1];
        int[] b = a.clone();
        Arrays.sort(b);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == b[i]) continue;
            int j = i, k = i;
            while (!used[a[j]]) {
                used[a[j]] = true;
                j = (j - 1 + n) % n;
            }
            while (!used[b[k]]) {
                used[b[k]] = true;
                k = (k + 1) % n;
            }
            k = (k - 1 + n) % n;
            int x = (j - i + n) % n;
            int y = (k - j + n) % n;
            System.out.println(x + " " + y);
            ans++;
        }
        System.out.println(ans);
    }
}
