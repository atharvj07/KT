#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
    double a, b;
    cin >> a >> b;

    int n;
    cin >> n;

    vector<pair<pair<double, double>, double>> cars(n);
    for (int i = 0; i < n; ++i) {
        double x, y, v;
        cin >> x >> y >> v;
        cars[i].first = {x, y};
        cars[i].second = v;
    }

    double min_time = 1e9;
    for (const auto& [pos, speed] : cars) {
        double dx = pos.first - a;
        double dy = pos.second - b;
        double dist = sqrt(dx * dx + dy * dy);
        double time = dist / speed;
        min_time = min(min_time, time);
    }

    cout.precision(20);
    cout << min_time << endl;
}
