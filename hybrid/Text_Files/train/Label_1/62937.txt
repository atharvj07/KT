t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    count = 0
    for i in range(n - 1):
        bigger = max(a[i], a[i + 1])
        smaller = min(a[i], a[i + 1])
        while bigger > smaller * 2:
            smaller *= 2
            count += 1
    print(count)
