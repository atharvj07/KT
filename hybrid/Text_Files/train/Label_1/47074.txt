import java.util.Scanner;

public class LimakAndDominoes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        char[][] grid = new char[h][w];
        for (int i = 0; i < h; i++) {
            String row = scanner.nextLine();
            for (int j = 0; j < w; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        int q = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;

            int ways = countDominoes(grid, r1, c1, r2, c2);
            System.out.println(ways);
        }
    }

    private static int countDominoes(char[][] grid, int r1, int c1, int r2, int c2) {
        int ways = 0;
        // Check for horizontal placements
        for (int col = c1; col < c2; col++) {
            if (grid[r1][col] == '.' && grid[r1][col + 1] == '.') {
                ways++;
            }
        }
        // Check for vertical placements
        for (int row = r1; row < r2; row++) {
            if (grid[row][c1] == '.' && grid[row + 1][c1] == '.') {
                ways++;
            }
        }
        return ways;
    }
}
