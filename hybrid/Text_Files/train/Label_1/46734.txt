import java.util.*;
import java.math.*;

public class Main {
    static long mod = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long T = sc.nextInt();
        long[] t = new long[n + 1];
        for (int i = 1; i <= n; i++)
            t[i] = sc.nextInt();
        long p = T;
        long q = 1;
        for (int i = 1; i <= n; i++) {
            p = (p * (2 * T - t[i] + mod)) % mod;
            q = q * 2 % mod;
        }
        long u = modInv(q, mod);
        p = (p * u) % mod;
        System.out.println(p);
    }

    static long modInv(long a, long b) {
        long b0 = b;
        long x0 = 0;
        long x1 = 1;
        if (b == 1)
            return 1;
        while (a > 1) {
            long q = a / b;
            long amb = a % b;
            a = b;
            b = amb;
            long xqx = x1 - q * x0;
            x1 = x0;
            x0 = xqx;
        }
        if (x1 < 0)
            x1 += b0;
        return x1;
    }
}
