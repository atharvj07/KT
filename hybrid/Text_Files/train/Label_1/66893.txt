n, m, k = map(int, input().split())
a = list(map(int, input().split()))
pre = [0] * (n + 1)
for i in range(n):
    pre[i+1] = pre[i] + a[i]
ans = float('-inf')
for l in range(n):
    if (l + 1) % m == 0:
        ans = max(ans, pre[l+1] - k * (l+1) // m)
    else:
        ans = max(ans, pre[l+1] - pre[l+1-(l+1)%m] - k * ((l+1) // m +1))
print(ans)
