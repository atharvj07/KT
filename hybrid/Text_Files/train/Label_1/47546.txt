import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = IntStream.range(0, sc.nextInt())
                .map(i -> sc.nextInt())
                .toArray();
        int[] b = IntStream.range(0, sc.nextInt())
                .map(i -> sc.nextInt())
                .toArray();
        int[] c = IntStream.range(0, sc.nextInt())
                .map(i -> sc.nextInt())
                .toArray();

        Arrays.sort(a);
        Arrays.sort(b);
        Arrays.sort(c);

        int count = 0;
        for (int i = 0, j = 0, k = 0; i < n; i++) {
            while (j < a.length && a[j] <= i) j++;
            while (k < b.length && b[k] <= i) k++;

            if (i == c[0] && (j == a.length || c[0] < a[j]) && (k == b.length || c[0] < b[k])) {
                count++;
            }
        }

        System.out.println(count);
    }
}
