#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    int k = 1;
    while (k <= n) {
        bool found = false;
        for (int i = 0; i < n - k + 1; i++) {
            bool ok = true;
            for (int j = i; j < i + k; j++) {
                if (s[j] != s[i]) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                found = true;
                break;
            }
        }
        if (found) {
            break;
        }
        k++;
    }
    cout << k << endl;
    return 0;
}
