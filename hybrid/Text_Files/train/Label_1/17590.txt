#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, p, q, r;
  cin >> n >> p >> q >> r;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<long long> s(n, 0);
  for (int i = 0; i < n; i++) s[i] = a[i];
  for (int i = 1; i < n; i++) s[i] = max(s[i], s[i - 1]);
  long long ans = LLONG_MIN;
  for (int j = 0; j < n; j++) {
    for (int i = 0; i <= j; i++) {
      ans = max(ans, p * a[i] + q * s[j] + r * a[j]);
    }
  }
  cout << ans << endl;
  return 0;
}
