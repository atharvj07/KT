#include <cstdio>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    long long n, k, d1, d2;
    scanf("%lld %lld %lld %lld", &n, &k, &d1, &d2);
    if (((k & 1) == 0 && (k - d1 - d2 & 1) == 0) || ((k & 1) && (n - 1 - k) & 1 == 0 && (k - d1 - d2 & 1) == 0))
      puts("yes");
    else
      puts("no");
  }
}
