import java.util.*;
import java.io.*;

class Main {
    static boolean canRotate(int n, int[][] segments) {
        boolean[] hasSegment = new boolean[n];
        for (int[] s : segments) {
            hasSegment[s[0] - 1] = true;
            hasSegment[s[1] - 1] = true;
        }
        int count = 0;
        for (boolean h : hasSegment) {
            count += h ? 1 : 0;
        }
        return count == 2;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] segments = new int[m][2];
        for (int i = 0; i < m; i++) {
            segments[i][0] = sc.nextInt();
            segments[i][1] = sc.nextInt();
        }
        System.out.println(canRotate(n, segments) ? "Yes" : "No");
    }
}
