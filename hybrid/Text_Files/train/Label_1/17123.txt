#include <iostream>
#include <vector>
#include <cstring>
#include <algorithm>

int main() {
    int k; std::cin >> k;
    std::string q; std::cin >> q;
    int n = q.length();
    std::vector<int> ans(k);
    std::vector<bool> used(26);
    used[q[0] - 'a'] = true;
    for (int i = 1; i < n; ++i) {
        for (int j = 0; j < 26; ++j) {
            if (!used[j] && q[i] == q[ans[i-1] - 1]) {
                ans[i] = j + 1;
                used[j] = true;
                break;
            }
        }
    }
    if (std::adjacent_find(ans.begin(), ans.end()) != ans.end()) std::cout << "NO" << std::endl;
    else {
        std::cout << "YES" << std::endl;
        for (int i : ans) std::cout << (char)(i+'a'-1) << std::endl;
    }
    return 0;
}
