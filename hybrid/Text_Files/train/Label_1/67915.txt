import sys
sys.setrecursionlimit(10**6)

n, m, p = map(int, input().split())
speed = [int(x) for x in input().split()]
grid = [input() for _ in range(n)]
di = [1, 0, -1, 0]
dj = [0, 1, 0, -1]

def dfs(i, j, player):
    if i < 0 or i >= n or j < 0 or j >= m or grid[i][j] != '.':
        return
    grid[i][j] = player
    for k in range(4):
        dfs(i + di[k] * speed[player - 1], j + dj[k] * speed[player - 1], player)

for i in range(n):
    for j in range(m):
        if grid[i][j] != '.' and grid[i][j] != '#':
            dfs(i, j, int(grid[i][j]))
            
for player in range(1, p + 1):
    cnt = 0
    for i in range(n):
        for j in range(m):
            if grid[i][j] == str(player):
                cnt += 1
    print(cnt, end=' ')
