import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] p0 = { in.nextInt(), in.nextInt() };
        int[] p1 = { in.nextInt(), in.nextInt() };
        int q = in.nextInt();
        while (q-- > 0) {
            int[] p2 = { in.nextInt(), in.nextInt() };
            long cross = (p2[0] - p1[0]) * (p1[1] - p0[1]) - (p2[1] - p1[1]) * (p1[0] - p0[0]);
            long dot = (p2[0] - p1[0]) * (p0[0] - p1[0]) + (p2[1] - p1[1]) * (p0[1] - p1[1]);
            if (cross > 0) System.out.println("COUNTER_CLOCKWISE");
            else if (cross < 0) System.out.println("CLOCKWISE");
            else if (dot < 0) System.out.println("ONLINE_BACK");
            else if (dot > (p0[0] - p1[0]) * (p0[0] - p1[0]) + (p0[1] - p1[1]) * (p0[1] - p1[1]))
                System.out.println("ONLINE_FRONT");
            else System.out.println("ON_SEGMENT");
        }
    }
}
