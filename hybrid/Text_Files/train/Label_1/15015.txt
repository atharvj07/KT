#include <iostream>
#include <limits>
#include <vector>
using namespace std;

int main() {
    long long n, m, sx, sy, fx, fy, x, y;
    cin >> n >> m;
    cin >> sx >> sy >> fx >> fy;
    vector<pair<long long, long long>> locations(m);
    for (long long i = 0; i < m; i++) {
        cin >> x >> y;
        locations[i] = {x, y};
    }
    long long ans = abs(fx - sx) + abs(fy - sy);
    for (long long i = 0; i < m; i++) {
        if (sx == locations[i].first || sy == locations[i].second) {
            ans = min(ans, abs(fx - locations[i].first) + abs(fy - locations[i].second));
        }
    }
    cout << ans << endl;
    return 0;
}
