import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();

        int minPos = 0, maxPos = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) minPos = i;
            if (a[i] == n) maxPos = i;
        }

        System.out.println(Math.max(minPos, maxPos) * 2);
    }
}
