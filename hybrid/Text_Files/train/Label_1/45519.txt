import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt(), yv = sc.nextInt();
        int xp = sc.nextInt(), yp = sc.nextInt();
        int xw1 = sc.nextInt(), yw1 = sc.nextInt(), xw2 = sc.nextInt(), yw2 = sc.nextInt();
        int xm1 = sc.nextInt(), ym1 = sc.nextInt(), xm2 = sc.nextInt(), ym2 = sc.nextInt();
        if(xw1 == xw2) {
            if(xv == xw1 && (yv >= yw1 && yv <= yw2 || yv >= yw2 && yv <= yw1)) {
                System.out.println("NO");
                return;
            }
            if(xp == xw1 && (yp >= yw1 && yp <= yw2 || yp >= yw2 && yp <= yw1)) {
                System.out.println("NO");
                return;
            }
        } else if(yw1 == yw2) {
            if(yv == yw1 && (xv >= xw1 && xv <= xw2 || xv >= xw2 && xv <= xw1)) {
                System.out.println("NO");
                return;
            }
            if(yp == yw1 && (xp >= xw1 && xp <= xw2 || xp >= xw2 && xp <= xw1)) {
                System.out.println("NO");
                return;
            }
        }
        if(xm1 == xm2) {
            if(xv == xm1 && (yv >= ym1 && yv <= ym2 || yv >= ym2 && yv <= ym1)) {
                System.out.println("NO");
                return;
            }
            if(xp == xm1 && (yp >= ym1 && yp <= ym2 || yp >= ym2 && yp <= ym1)) {
                System.out.println("NO");
                return;
            }
        } else if(ym1 == ym2) {
            if(yv == ym1 && (xv >= xm1 && xv <= xm2 || xv >= xm2 && xv <= xm1)) {
                System.out.println("NO");
                return;
            }
            if(yp == ym1 && (xp >= xm1 && xp <= xm2 || xp >= xm2 && xp <= xm1)) {
                System.out.println("NO");
                return;
            }
        }
        int dx = xv - xp, dy = yv - yp;
        if(xm1 == xm2) {
            if(dx == 0) {
                System.out.println("NO");
                return;
            }
            double t = ((double) (xm1 - xv)) / dx;
            if(t >= 0.0 && t <= 1.0 && Math.abs(ym1 + t * dy - yv) <= 1e-9) {
                int x = (int) (xm1 + t * dx), y = (int) (ym1 + t * dy);
                if((xw1 == xw2 || ym1 == yv) && ((x >= xw1 && x <= xw2 || x >= xw2 && x <= xw1) ^ (y >= ym1 && y <= yv || y >= yv && y <= ym1))) {
                    System.out.println("NO");
                    return;
                }
            }
        } else if(ym1 == ym2) {
            if(dy == 0) {
                System.out.println("NO");
                return;
            }
            double t = ((double) (ym1 - yv)) / dy;
            if(t >= 0.0 && t <= 1.0 && Math.abs(xm1 + t * dx - xv) <= 1e-9) {
                int x = (int) (xm1 + t * dx), y = (int) (ym1 + t * dy);
                if((xw1 == xw2 || xm1 == xv) && ((x >= xw1 && x <= xw2 || x >= xw2 && x <= xw1) ^ (y >= ym1 && y <= yv || y >= yv && y <= ym1))) {
                    System.out.println("NO");
                    return;
                }
            }
        }
        System.out.println("YES");
    }
}
