for t in range(int(input())):
    n = int(input())
    s = input()
    suff = s
    for i in range(n - 1, -1, -1):
        if s[i] == 'b':
            for j in range(i + 1, n):
                if s[j] == 'g':
                    suff = s[:i] + 'g' + s[i + 1:j] + 'b' + s[j + 1:]
        elif s[i] == 'r':
            for j in range(i + 1, n):
                if s[j] == 'u':
                    suff = s[:i] + 'u' + s[i + 1:j] + 'r' + s[j + 1:]
    print(suff)
