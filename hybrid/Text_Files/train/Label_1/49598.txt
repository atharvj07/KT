import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = scanner.nextInt();
        }

        int[] s = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            s[i] = scanner.nextInt();
        }

        int[] c = new int[k + 1];
        for (int i = 1; i <= k; i++) {
            c[i] = scanner.nextInt();
        }

        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            if (!map.containsKey(s[i])) {
                map.put(s[i], new ArrayList<Integer>());
            }
            map.get(s[i]).add(i);
        }

        int[] max = new int[m + 1];
        for (int i = 1; i <= m; i++) {
            max[i] = -1;
        }

        for (List<Integer> list : map.values()) {
            for (int id : list) {
                max[s[id]] = Math.max(max[s[id]], p[id]);
            }
        }

        int cnt = 0;
        for (int i = 1; i <= k; i++) {
            if (p[c[i]] != max[s[c[i]]]) {
                cnt++;
            }
        }

        System.out.println(cnt);
    }
}
