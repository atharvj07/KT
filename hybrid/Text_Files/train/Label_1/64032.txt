# This function computes (a*b) mod p
def multiply_mod(a, b, p):
    result = (a*b)%p
    return result

# This function computes (a^2) mod p
def square_mod(a, p):
    result = (a*a)%p
    return result

n, p, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
count = 0
for i in range(n):
    for j in range(i+1, n):
        if multiply_mod(a[i]+a[j],square_mod(a[i],p)+square_mod(a[j],p),p) == k:
            count += 1
print(count)
