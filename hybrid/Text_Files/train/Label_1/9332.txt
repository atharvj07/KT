#include <iostream>
#include <string>

int main() {
    int q;
    std::cin >> q;
    while (q--) {
        std::string s, t;
        std::cin >> s >> t;
        int i = 0, j = 0;
        while (i < s.size() && j < t.size()) {
            if (s[i] == t[j]) {
                i++;
                j++;
            } else {
                i++;
            }
        }
        std::cout << (j == t.size() ? "YES" : "NO") << '\n';
    }
    return 0;
}
