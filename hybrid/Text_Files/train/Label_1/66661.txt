n=int(input())
b=list(map(int,input().split()))
b.sort()
ans1=max(b[-1]-b[0],b[-2]-b[1])
ans2=1
if ans1==b[-1]-b[0]:
    for i in range(1,n):
        if b[i]==b[i-1]:
            ans2+=1
        else:
            break
if ans1==b[-2]-b[1]:
    for i in range(3,n):
        if b[i]==b[i-2]:
            ans2+=1
        else:
            break
print(ans1,ans2)
