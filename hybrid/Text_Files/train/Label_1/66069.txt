n, m = map(int, input().split())
node_values = list(map(int, input().split()))
edges = [list(map(int, input().split())) for _ in range(m)]

max_density = 0.0

for i in range(m):
    a, b, c = edges[i]
    a -= 1
    b -= 1
    density = (node_values[a] + node_values[b]) / c
    if density > max_density:
        max_density = density

print(f"{max_density:.15f}")
