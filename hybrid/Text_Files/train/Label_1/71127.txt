T = int(input().strip())
for _ in range(T):
    n = int(input().strip())
    p = list(map(int, input().strip().split()))
    found = False
    for i in range(1, n - 1):
        if p[i - 1] < p[i] and p[i] > p[i + 1]:
            print("YES")
            print(i, i + 1, i + 2)
            found = True
            break
    if not found:
        print("NO")
