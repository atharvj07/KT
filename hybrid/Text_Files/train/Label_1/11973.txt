#include <bits/stdc++.h>
using namespace std;
long long int n, t, t_[200001], dp[200001];
int main() {
  cin >> n >> t;
  for (int i = 1; i <= n; i++) cin >> t_[i];
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1];
    if (t_[(int)dp[i] + 1] <= i) dp[i] = (int)dp[i] + 1;
    if (t_[i] <= t) dp[(int)dp[i]]++;
  }
  cout << dp[n] << endl;
}
