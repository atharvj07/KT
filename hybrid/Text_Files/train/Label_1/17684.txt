#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (auto &e : a) cin >> e;
  sort(a.begin(), a.end());
  vector<long long> gcd(n - 1);
  for (int i = 0; i < n - 1; ++i) gcd[i] = __gcd(a[i], a[i + 1]);
  sort(gcd.begin(), gcd.end());
  long long ans = 0;
  for (int i = 0; i < n - 1; ++i) {
    if (gcd[i] == 1) {
      ans += a[i] - 1;
      a[i + 1] -= a[i] - 1;
    }
  }
  for (int i = 0; i < n - 1; ++i) ans += abs(a[i] - a[i + 1] - 1);
  cout << ans << endl;
  return 0;
}
