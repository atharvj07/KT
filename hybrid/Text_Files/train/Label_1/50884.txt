import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) t[i] = in.nextInt();
        Arrays.sort(t);
        long[] ans = new long[n];
        long now = 0;
        for (int i = 0; i < n; i++) {
            if (now <= t[i]) now = t[i];
            ans[i] = now + p;
            now += p;
        }
        for (long i : ans) System.out.print(i + " ");
    }
}
