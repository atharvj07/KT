import sys

n,a,b = list(map(int, sys.stdin.readline().split()))
seals = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
seals.sort()
mx = 0
for i in range(n):
  w,h = seals[i]
  for j in range(i+1, n):
    w2,h2 = seals[j]
    if w2 > a or h2 > b:
      continue
    if w > a or h > b:
      w,h = h,w
    if w2 > a or h2 > b:
      continue
    if w2 > w and h2 > h:
      mx = max(mx, w*h + w2*h2)
print(mx)
