import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[] for _ in range(n+1)]
for _ in range(m):
    u, v, c = map(int, input().split())
    g[u].append((v,c))
    g[v].append((u,c))

vis = [False] * (n+1)
ans = []
cnt = 0

def dfs(x, p, c):
    global cnt, ans
    vis[x] = True
    if x != p and x in g[p]:
        cnt += 1
        ans.append(g[p].index(x)+1)
    for y, cc in g[x]:
        if not vis[y]:
            dfs(y, x, cc)
        elif y != p:
            if cc%2 != c%2:
                cnt += 1
                ans.append(g[x].index(y)+1)

for i in range(1, n+1):
    if not vis[i]:
        dfs(i, i, 0)

print(cnt, len(ans))
print(' '.join(map(str, ans)))
