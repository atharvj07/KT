#include <algorithm>
#include <iostream>
#include <vector>

struct ring {
    long long a, b, h;
};

bool cmp(ring r1, ring r2) {
    return r1.b > r2.b;
}

int main() {
    int n;
    std::cin >> n;
    std::vector<ring> rings(n);
    for (int i = 0; i < n; i++) { std::cin >> rings[i].a >> rings[i].b >> rings[i].h; }
    std::sort(rings.begin(), rings.end(), cmp);
    long long cur = 0, max_hei = 0;
    for (int i = 0; i < n; i++) {
        if (rings[i].a > cur) {
            cur = rings[i].b;
            max_hei += rings[i].h;
        }
    }
    std::cout << max_hei << std::endl;
    return 0;
}
