import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] hi = new long[n];
        for (int i = 0; i < n; i++) hi[i] = sc.nextInt();
        long op = 0;
        while (true) {
            long b = 0;
            for (int i = 0; i < n; i++) {
                if (hi[i] > 0 && (i == 0 || hi[i - 1] < hi[i]) && (i == n - 1 || hi[i + 1] < hi[i])) b++;
            }
            if (b == 0) break;
            for (int i = 0; i < n; i++) {
                if (hi[i] > 0 && (i == 0 || hi[i - 1] < hi[i]) && (i == n - 1 || hi[i + 1] < hi[i])) hi[i]--;
            }
            op++;
        }
        System.out.println(op);
    }
}
