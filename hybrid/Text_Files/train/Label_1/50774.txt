import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long x = sc.nextLong();
        for (long a = 1; a <= 100000000000000L; a++) {
            if (x % a == 0) {
                System.out.println(a + " " + x / a);
                return;
            }
        }
        System.out.println(-1);
    }
}
