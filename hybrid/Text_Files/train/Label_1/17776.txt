#include <bits/stdc++.h>
using namespace std;

void solve() {
    long long a, b, n;
    cin >> a >> b >> n;
    while (n--) {
        long long l, t, m;
        cin >> l >> t >> m;
        long long r = (t * 1LL * m * 1LL * b) / (a + (--t * 1LL * m * 1LL * b)) + (1 - t * 1LL * m * 1LL * b > 0) + l - 1;
        if (r >= l)
            cout << r << "\n";
        else
            cout << "-1\n";
    }
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    solve();
    return 0;
}
