import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        Task solver = new Task();
        solver.solve(in, out);
        out.close();
    }

    static class Task {
        public void solve(InputReader in, PrintWriter out) throws IOException {
            int n = in.nextInt(), m = in.nextInt();
            int[] par = new int[n + 1], siz = new int[n + 1];
            Arrays.fill(siz, 1);
            for (int i = 0; i < n + 1; i++) {
                par[i] = i;
            }

            ArrayList<Edge>[] E = new ArrayList[n + 1];
            for (int i = 0; i < n + 1; i++) {
                E[i] = new ArrayList<>();
            }
            for (int i = 0; i < m; i++) {
                int u = in.nextInt(), v = in.nextInt(), w = in.nextInt();
                E[u].add(new Edge(v, w));
                E[v].add(new Edge(u, w));
            }

            int q = in.nextInt();
            for (int i = 0; i < q; i++) {
                int k = in.nextInt();
                Set<Integer> ids = new HashSet<>();
                for (int j = 0; j < k; j++) {
                    ids.add(in.nextInt() - 1);
                }
                int ans = 0;
                for (int edgeID : ids) {
                    int a = find(par, edgeID / 2), b = find(par, (edgeID / 2) ^ 1);
                    if (a != b) {
                        par[a] = b;
                        siz[b] += siz[a];
                        ans += 1;
                    }
                }

                out.println(ans == n - 1 ? "YES" : "NO");
                for (int edgeID : ids) {
                    int a = edgeID / 2;
                    par[a] = a;
                    siz[a] = 1;
                }
            }
        }

        int find(int[] par, int x) {
            if (par[x] == x) {
                return x;
            }
            return par[x] = find(par, par[x]);
        }

        class Edge {
            int to, w;

            public Edge(int to, int w) {
                this.to = to;
                this.w = w;
            }
        }
    }

    static class InputReader {
        private BufferedReader reader;
        private StringTokenizer tokenizer = new StringTokenizer("");

        public InputReader(InputStream inputStream) {
            reader = new BufferedReader(new InputStreamReader(inputStream), 32768);
        }

        public String next() throws IOException {
            while (!tokenizer.hasMoreTokens()) {
                tokenizer = new StringTokenizer(reader.readLine());
            }
            return tokenizer.nextToken();
        }

        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }
}
