#include <bits/stdc++.h>
using namespace std;
#define pb push_back
const int maxn = 1e5+10;
vector<int> a, b;
int main() {
    int n; cin >> n;
    vector<pair<int, int>> v;
    for(int i = 0; i < n; i++) {
        int tmp; cin >> tmp;
        a.pb(tmp);
        v.pb({tmp, i});
    }
    for(int i = 0; i < n; i++) {
        int tmp; cin >> tmp;
        b.pb(tmp);
    }
    sort(v.rbegin(), v.rend());
    int ans = 0;
    int cnt = 0;
    for(auto it: v) {
        if(cnt * 2 > n) break;
        ans += b[it.second];
        cnt++;
    }
    cout << ans << "\n";
}
