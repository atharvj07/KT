import sys
from collections import defaultdict

n, m, d = map(int, sys.stdin.readline().split())

adj = defaultdict(list)
for _ in range(m):
    a, b = map(int, sys.stdin.readline().split())
    adj[a].append(b)
    adj[b].append(a)

visited = [False] * (n + 1)
tree = []

def dfs(cur, par, deg):
    visited[cur] = True
    if cur == 1:
        deg[0] += 1
    if deg[0] == d:
        return True
    for nxt in adj[cur]:
        if not visited[nxt]:
            if dfs(nxt, cur, deg):
                tree.append((cur, nxt))
                return True
    return False

if dfs(1, 1, [0]):
    print("YES")
    for a, b in tree:
        print(a, b)
else:
    print("NO")
