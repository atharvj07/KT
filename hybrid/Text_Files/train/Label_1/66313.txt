from collections import deque

n, m = map(int, input().split())

d, g = 1, 1
t = 1
house = [[0] * (m + 2) for _ in range(n + 2)]
for i in range(n - 1, -1, -1):
    x = list('.' + input() + '.')
    for j in range(1, m + 1):
        if x[j] == '#':
            house[i][j] = 1
        elif x[j] == '+':
            house[i][j] = 2

while d <= n:
    if house[d][g] == 0:
        d += 1
    else:
        if house[d][g] == 1:
            g = g + 2 * (1 - t) - 1
        t = 1 - t
    if t == 1:
        g += 1
    else:
        g -= 1
    if g < 1 or g > m:
        print('Never')
        break
    t = 1 - t
    t += 1
else:
    print(t)
