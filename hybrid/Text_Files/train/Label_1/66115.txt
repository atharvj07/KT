import sys

n, m, k = map(int, input().split())
adj = [set() for _ in range(n + 1)]
for _ in range(m):
    u, v = map(int, input().split())
    adj[u].add(v)
    adj[v].add(u)

for i in range(1, n + 1):
    adj[i].add(i)

ans = []
for _ in range(m):
    vis = [False] * (n + 1)
    cnt = 0
    for i in range(1, n + 1):
        if not vis[i]:
            q = [i]
            g = 0
            while q:
                u = q.pop()
                if not vis[u]:
                    vis[u] = True
                    g += 1
                    for v in adj[u]:
                        q.append(v)
            cnt = max(cnt, g)
    ans.append(cnt)

print(*ans, sep='\n')
