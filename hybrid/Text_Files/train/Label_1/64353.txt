import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
ans = 0
for i in range(n):
    for j in range(i + 1, n):
        if (j - i + 1) % 2 == 0:
            xor = 0
            for k in range(i, (i + j + 1) // 2):
                xor ^= a[k]
            if xor == 0: ans += 1
print(ans)
