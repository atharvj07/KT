#include<bits/stdc++.h>
#define MOD 1000000007
using namespace std;
int main() {
    string s;
    cin >> s;
    long long int ans = 1;
    for (int i = 0; i < s.length(); i++) {
        if (s[i] == 'w' || s[i] == 'm') {
            ans = (2LL * ans) % MOD;
        }
    }
    cout << ans << endl;
    return 0;
}
