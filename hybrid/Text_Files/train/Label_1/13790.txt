#include <bits/stdc++.h>
using namespace std;

int n, m, x, y;
vector<string> maze;
vector<vector<int>> dp;
vector<vector<bool>> vis;

int dfs(int r, int c, int left, int right) {
    if (r < 0 || r >= n || c < 0 || c >= m || maze[r][c] == '*' || vis[r][c]) return 0;
    vis[r][c] = true;
    if (dp[r][c] != -1) return dp[r][c];
    int ans = 0;
    if (left > 0) ans += dfs(r, c - 1, left - 1, right);
    if (right > 0) ans += dfs(r, c + 1, left, right - 1);
    ans += dfs(r - 1, c, left, right) + dfs(r + 1, c, left, right);
    return dp[r][c] = ans;
}

int main() {
    cin >> n >> m;
    cin >> x >> y;
    maze.resize(n);
    dp.resize(n, vector<int>(m, -1));
    vis.resize(n, vector<bool>(m, false));
    for (int i = 0; i < n; i++) cin >> maze[i];
    cout << dfs(x - 1, y - 1, x, y) + 1 << endl;
}
