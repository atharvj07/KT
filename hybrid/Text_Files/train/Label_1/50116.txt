import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        double[] p = new double[n];
        double[] tC = new double[n];
        for (int i = 0; i < n; i++) {
            p[i] = (double) scanner.nextInt();
            tC[i] = scanner.nextDouble();
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (tC[i] > tC[j]) {
                    double tmp = tC[i];
                    tC[i] = tC[j];
                    tC[j] = tmp;
                    double tmp2 = p[i];
                    p[i] = p[j];
                    p[j] = tmp2;
                }
            }
        }
        double ans = 0;
        for (int i = 0; i < n; i++) {
            double time = 0;
            double pr = 1.0;
            while (time < tC[i]) {
                if ((int) time == time) {
                    pr *= 1.0 - p[i] / 100.0;
                }
                time += 1.0;
            }
            if (time <= t) {
                ans += (1.0 - pr);
            } else {
                break;
            }
        }
        System.out.println(String.format("%.9f", ans));
    }
}
