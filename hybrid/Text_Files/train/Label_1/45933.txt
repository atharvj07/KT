import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int k = sc.nextInt();
            if (k == n * (m - 1) || k == m * (n - 1)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
