#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll dp[1000010];
int main()
{
    ll n,k,a,i,j;
    cin>>n>>k;
    for(i=1; i<=1000000; i++)
        dp[i]=1e18;
    dp[0]=0;
    while(k--)
    {
        cin>>a;
        for(i=1000000; i>=a; i--)
            dp[i]=min(dp[i], dp[i-a]+1);
    }
    if(dp[n]!=1e18)
        cout<<dp[n];
    else
        cout<<-1;
    return 0;
}
