#include<bits/stdc++.h>
using namespace std;
int n,x1,x2,a[100005],b[100005],c[100005];
bool f;
signed main(){
    cin>>n>>x1>>x2;
    for(int i=1;i<=n;i++)cin>>a[i]>>b[i],c[i]=a[i]*x1+b[i];
    sort(a+1,a+n+1);sort(c+1,c+n+1);
    for(int i=1;i<=n;i++)
        if(c[i-1]+x2-x1>b[i]&&c[i]<b[i]){f=1;break;}
    puts(f?"YES":"NO");
    return 0;
}
