import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int h = in.nextInt();
    int w = in.nextInt();
    int k = in.nextInt();
    long ans = (long) Math.pow(5, h) * ((long) Math.pow(5, w) * (k - 1) + 1) % 1000000007;
    System.out.println(ans);
  }
}
