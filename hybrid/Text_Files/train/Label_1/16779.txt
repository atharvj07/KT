#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s;
  int m;
  cin >> s >> m;
  int n = s.size();
  vector<int> p(n);

  for (int i = 1, j = 0; i < n; i++) {
    while (j > 0 && s[i] != s[j]) j = p[j - 1];
    if (s[i] == s[j]) j++;
    p[i] = j;
  }

  int ans = 1e9;
  for (int i = 0, j = 0; i < n; i++) {
    while (j > 0 && s[i] != s[j]) j = p[j - 1];
    if (s[i] == s[j]) j++;
    if (i - p[i] > 0) ans = min(ans, (i + 1) % m);
  }
  cout << ans << endl;
  return 0;
}
