import java.math.BigInteger;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<List<Card>> cards = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            List<Card> list = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                list.add(new Card(sc.nextInt(), sc.nextInt()));
            }
            cards.add(list);
        }
        BigInteger ans = BigInteger.ZERO;
        for (List<Card> list : cards) {
            list.sort(Comparator.comparing(c -> c.d));
            BigInteger sum = BigInteger.ZERO;
            int cnt = 0;
            for (int i = list.size() - 1; i >= 0; i--) {
                Card c = list.get(i);
                if (sum.add(BigInteger.valueOf(c.c)).compareTo(BigInteger.valueOf(3)) <= 0) {
                    sum = sum.add(BigInteger.valueOf(c.c));
                    if (cnt == 9) {
                        ans = ans.add(BigInteger.valueOf(c.d).multiply(BigInteger.valueOf(2)));
                        cnt = 0;
                    } else {
                        ans = ans.add(BigInteger.valueOf(c.d));
                        cnt++;
                    }
                }
            }
        }
        System.out.println(ans);
    }

    static class Card {
        int c;
        int d;

        public Card(int c, int d) {
            this.c = c;
            this.d = d;
        }
    }
}
