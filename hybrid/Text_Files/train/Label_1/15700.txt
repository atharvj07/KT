#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (long long &x : a) cin >> x;
        if (is_sorted(a.begin(), a.end()))
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
