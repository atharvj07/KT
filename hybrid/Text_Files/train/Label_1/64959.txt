import sys
def main():
    n, m = map(int, sys.stdin.readline().strip().split())
    field = [[c for c in input()] for _ in range(n)]
    if any(row[0] == 'G' or row[-1] == 'S' for row in field):
        print(-1)
        return
    ans = 0
    while True:
        moved = False
        for i in range(n):
            if field[i][m - 1] != 'G' and field[i][m - 1] != 'S':
                moved = True
                for j in range(m - 2, -1, -1):
                    field[i][j + 1] = field[i][j]
        if not moved: break
        ans += 1
        for row in field:
            print(''.join(row))
    print(ans)

main()
