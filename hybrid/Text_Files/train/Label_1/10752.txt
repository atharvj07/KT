#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back

const int sz=2e5+9;
vector<int>a[sz];
ll vis[sz];
ll ans[sz];
ll n,x,y,k,m,z,mn=LLONG_MAX,cnt;

void dfs(ll node,ll pr,ll sum)
{
    vis[node]=1;
    sum+=a[node][y];

    if(sum>=x)
    {
        ans[++cnt]=node;
        sum-=x;
    }

    for(auto child:a[node])
    {
        if(vis[child]==0)
            dfs(child,node,sum);
    }
}

int main()
{
    cin>>m>>k>>n>>z;
    for(ll i=1;i<=m;i++)
    {
        cin>>x;
        a[i].pb(x);
    }
    for(ll i=1;i<=z;i++)
    {
       cin>>x;
       y=x;
       dfs(1,1,0);
    }
    if(cnt>=n)
    {
        cout<<cnt<<endl;
        for(ll i=1;i<=cnt;i++)
         cout<<ans[i]<<" ";
    }
    else cout<<-1<<endl;
}
