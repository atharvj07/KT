#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> l(n), d(n);
    for (int i = 0; i < n; i++) cin >> l[i];
    for (int i = 0; i < n; i++) cin >> d[i];
    sort(l.begin(), l.end());
    sort(d.begin(), d.end());
    int ans = 0;
    for (int i = 0; i < n / 2; i++) ans += d[i];
    cout << ans << endl;
    return 0;
}
