#include<bits/stdc++.h>
using namespace std;
int main() {
    long long int n, k, a[200005], b[200005], dp[200005], mn = LLONG_MAX;
    cin >> n >> k;
    for(int i = 1; i <= n; i++)
        cin >> a[i];
    for(int i = 1; i <= n; i++)
        cin >> b[i];
    dp[n + 1] = 0;
    for(long long int i = n; i >= 1; i--) {
        if(k > 0) {
            dp[i] = min(a[i] + dp[i + 1], b[i]);
            k--;
        }
        else {
            dp[i] = min(b[i], b[i] + dp[i + 1]);
        }
    }
    cout << dp[1] << '\n';
    return 0;
}
