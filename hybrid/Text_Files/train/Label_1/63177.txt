def solve():
    t = int(input())
    for _ in range(t):
        n, s = map(int, input().split())
        ans = 0
        sum = 0
        temp = n
        while temp > 0:
            sum += temp % 10
            temp //= 10
        while sum > s:
            if n % 10 != 9:
                n += 1
            else:
                n = n // 10 + 1
            ans += 1
            sum = 0
            temp = n
            while temp > 0:
                sum += temp % 10
                temp //= 10
        print(ans)

solve()
