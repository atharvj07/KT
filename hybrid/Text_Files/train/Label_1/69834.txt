for _ in range(int(input())):
    s = input()
    t = 'a' * len(s)
    for i in range(len(s)):
        if s[i] > t[i]:
            t = t[:i] + s[i] + t[i+1:]
        else:
            if(i == len(s)-1 or s[i] != s[i+1]):
                t = t[:i] + chr(ord(s[i])+1) + t[i+1:]
                break
    print(t)
