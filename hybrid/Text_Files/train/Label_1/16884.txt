#include <iostream>
#include <vector>
#include <unordered_set>

int main() {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    std::unordered_set<char> types;
    for (char c : s) {
        types.insert(c);
    }
    int count = 0, min_count = n+1;
    for (int i = 0; i < n; i++) {
        if (types.count(s[i]) > 0) {
            count++;
            if (count == types.size()) {
                min_count = std::min(min_count, i+1);
            }
        }
    }
    std::cout << min_count << std::endl;
    return 0;
}
