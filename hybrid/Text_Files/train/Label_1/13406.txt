#include <iostream>
#include <limits>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(k);
  int mx = 0;
  for (int i = 0; i < k; i++) {
    cin >> a[i];
    mx = max(mx, a[i]);
  }
  if (n > mx) {
    cout << -1 << endl;
    return 0;
  }
  long long ans = 1e18;
  for (int i = 0; i < k; i++) {
    if (!n) break;
    long long x = n / a[i];
    long long res = x * 1000;
    if (res % a[i] == 0) res--;
    res = 1000 + res / a[i];
    if (res % 1000 == 0) res -= 1000;
    ans = min(ans, res / 1000);
    n %= a[i];
  }
  if (!n)
    cout << ans << endl;
  else
    cout << -1 << endl;
  return 0;
}
