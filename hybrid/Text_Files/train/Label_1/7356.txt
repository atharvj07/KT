#include <bits/stdc++.h>

using namespace std;

using ll = long long;
using vi = vector<int>;
using vii = vector<vi>;

int main() {
  int t;
  cin >> t;
  for (int tt = 0; tt < t; tt++) {
    int n;
    cin >> n;
    vi x(n), y(n);
    for (int i = 0; i < n; i++) {
      cin >> x[i] >> y[i];
    }
    vector<tuple<int, int, string>> v;  // (y, x, string)
    for (int i = 0; i < n; i++) {
      string path;
      int xx = x[i], yy = y[i];
      while (yy > 0) {
        path += 'U';
        yy--;
      }
      while (xx > 0) {
        path += 'R';
        xx--;
      }
      v.emplace_back(y[i], x[i], path);
    }
    sort(v.begin(), v.end());
    string ans = "";
    for (int i = 0; i < n; i++) {
      ans += get<2>(v[i]);
    }
    string xans = "";
    int cnt = 0;
    set<string> reached;
    for (int i = 0; i < ans.size(); i++) {
      if (ans[i] == 'R')
        cnt++;
      else
        cnt--;
      if (cnt < 0)
        break;
      xans += ans[i];
      reached.insert(xans);
    }
    if (reached.find(ans) == reached.end())
      cout << "NO" << endl;
    else
      cout << "YES" << endl << xans << endl;
  }
  return 0;
}
