#include<bits/stdc++.h>
using namespace std;
#define ll long long int
#define fast ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
int main()
{
    ll n; cin >> n;
    vector<ll> a(n), b(n), preOR_a(n + 1), preOR_b(n + 1);
    for (auto &x : a) cin >> x;
    for (auto &x : b) cin >> x;
    preOR_a[0] = 0;
    preOR_b[0] = 0;
    for (ll i = 0; i < n; i++)
    {
        preOR_a[i + 1] = preOR_a[i] | a[i];
        preOR_b[i + 1] = preOR_b[i] | b[i];
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++)
    {
        for (ll j = i; j < n; j++)
        {
            ans = max(ans, preOR_a[i] | preOR_b[j]);
        }
    }
    cout << ans << endl;
    return 0;
}
