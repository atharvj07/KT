#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define mp make_pair
#define fl(i,a,b) for(int i=a;i<b;i++)
#define vi vector<int>
char t[200005];
int main(){
    int n;
    cin>>n;
    string s;
    cin>>s;
    int ans=0;
    for(int i=0;i<n;i++){
        if(i%3==0)t[i]=s[i];
        else{
            if(s[(i-1)%3]==s[i] and s[(i-1)%3]==s[(i-2)%3]){
                t[i]=s[i];
                ans++;
            }
            else{
                t[i]=s[i];
                if(s[i]!=s[i%3])ans++;
            }
        }
    }
    cout<<ans<<endl;
    fl(i,0,n)cout<<t[i];
    return 0;
}
