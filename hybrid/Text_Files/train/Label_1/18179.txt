#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b;
    cin >> n;
    vector<pair<int, int>> v;
    for(int i=0; i<n; i++) {
        cin >> a >> b;
        v.push_back({a, b});
    }
    sort(v.begin(), v.end());
    int ans = 100;
    for(int i=0; i<n; i++) {
        if(v[i].second < 100) {
            a = v[i].second;
            if(i-1>=0 && v[i-1].second>=a) continue;
            if(i+1<n && v[i+1].second>=a) continue;
            ans-=a-v[i].first;
        }
    }
    cout << ans << endl;
    return 0;
}
