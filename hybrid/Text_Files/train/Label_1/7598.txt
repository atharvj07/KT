#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        int n, k;
        cin >> n >> k;
        if (k > 1 && n % 2 == 0) {
            cout << "NO" << endl;
        } else {
            cout << "YES" << endl;
            for (int i = 1; i <= k - 1; i++) cout << ((n - 1) / k) * i << ' ';
            cout << n << endl;
        }
    }
}
