#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) { cin >> words[i]; }
  for (int i = 1; i < n; i++) {
    string& a = words[i - 1];
    string& b = words[i];
    if (b.size() >= a.size()) {
      int p = b.size() - a.size();
      if (a == b.substr(p)) { b.erase(p); }
    }
    a += b;
  }
  cout << words[n - 1] << '\n';
  return 0;
}
