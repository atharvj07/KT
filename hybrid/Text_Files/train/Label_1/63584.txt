import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
    n = int(input())
    l = list(map(int, input().strip().split()))
    a = [l[0]]
    for i in range(1, n):
        a += max(a[-1], l[i]), min(a[-1], l[i])
    b = [l[-1]]
    for i in range(n - 2, -1, -1):
        b += max(b[-1], l[i]), min(b[-1], l[i])
    b.reverse()
    ans = 0
    for i in range(n - 1):
        ans = max(ans, a[i] * b[i + 1])
    print(ans)
