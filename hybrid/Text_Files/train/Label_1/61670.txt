n, m = map(int, input().split())
a = [list(input()) for _ in range(n)]
ans = n * m
for x1 in range(n):
    for y1 in range(m):
        for x2 in range(x1, n):
            for y2 in range(y1, m):
                s = 0
                for i in range(x1, x2 + 1):
                    for j in range(y1, y2 + 1):
                        s += 1 if a[i][j] == 'W' else -1
                ans = min(ans, abs(s))
print(ans)
