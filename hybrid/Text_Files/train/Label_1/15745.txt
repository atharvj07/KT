#include <iostream>
#include <algorithm>
using namespace std;

int main()
{
	int tt, n;
	cin >> tt;
	while (tt--)
	{
		cin >> n;
		int a[n], b[n];
		for (int i = 0; i < n; i++)
			cin >> a[i];
		for (int i = 0; i < n; i++)
		{
			if (i % 2 == a[i] % 2)
				b[i] = 0;
			else
				b[i] = 1;
		}
		if (is_sorted(b, b + n))
			cout << 0 << endl;
		else
		{
			int f = 0;
			for (int i = 0; i < n - 1; i++)
			{
				if (b[i] == 1 && b[i + 1] == 0)
				{
					f = 1;
					cout << 1 << endl;
					break;
				}
			}
			if (!f)
				cout << -1 << endl;
		}
	}
	return 0;
}
