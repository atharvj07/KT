import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int na = sc.nextInt();
        int nb = sc.nextInt();
        int w = sc.nextInt();
        int[] ma = new int[na];
        int[] ca = new int[na];
        int[] mb = new int[nb];
        int[] cb = new int[nb];
        for (int i = 0; i < na; i++) {
            ma[i] = sc.nextInt();
            ca[i] = sc.nextInt();
        }
        for (int i = 0; i < nb; i++) {
            mb[i] = sc.nextInt();
            cb[i] = sc.nextInt();
        }
        int res = Integer.MAX_VALUE;
        int[] wa = new int[na + 1];
        int[] wb = new int[nb + 1];
        int[] cua = new int[na + 1];
        int[] cub = new int[nb + 1];
        for (int i = 0; i < na; i++) {
            for (int j = wa[i]; j + ma[i] <= w; j++) {
                wa[i + 1] = Math.max(wa[i + 1], j + ma[i]);
                cua[i + 1] = Math.max(cua[i + 1], cua[i] + ca[i]);
            }
        }
        for (int i = 0; i < nb; i++) {
            for (int j = wb[i]; j + mb[i] <= w; j++) {
                wb[i + 1] = Math.max(wb[i + 1], j + mb[i]);
                cub[i + 1] = Math.max(cub[i + 1], cub[i] + cb[i]);
            }
        }

        for (int i = 0; i <= na; i++) {
            for (int j = 0; j <= nb; j++) {
                if (wa[i] + wb[j] <= w) {
                    int cuamax = 0, cuamin = Integer.MAX_VALUE;
                    if (i > 0) {
                        cuamax = cua[i];
                        cuamin = cua[i];
                    }
                    if (j > 0) {
                        cuamax = Math.max(cuamax, cub[j]);
                        cuamin = Math.min(cuamin, cub[j]);
                    }
                    res = Math.min(res, Math.max(wa[i] - wb[j], cuamax - cuamin));
                }
            }
        }
        System.out.println(res);
    }
}
