#include <iostream>
#include <cmath>

using namespace std;

bool check_sum_of_cubes(long long x) {
    int max_val = cbrt(x);
    for (int i = 1; i <= max_val; i++) {
        long long a3 = i * i * i;
        long long b3 = x - a3;
        int b = cbrt(b3);
        if (b * b * b == b3) return true;
    }
    return false;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long x;
        cin >> x;
        if (check_sum_of_cubes(x)) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
