import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] cnt = new int[53];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            int x = sc.nextInt();
            cnt[x + (s.charAt(0) == 'S' ? 0 : s.charAt(0) == 'H' ? 13 : s.charAt(0) == 'C' ? 26 : 39)]++;
        }
        for (int i = 1; i <= 13; i++) {
            if (cnt[i] == 0) System.out.println("S " + i);
            if (cnt[i + 13] == 0) System.out.println("H " + i);
            if (cnt[i + 26] == 0) System.out.println("C " + i);
            if (cnt[i + 39] == 0) System.out.println("D " + i);
        }
    }
}
