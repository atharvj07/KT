import sys
inp = sys.stdin.readlines()
n,m,q = map(int,inp[0].split())
a = list(map(int,inp[1].split()))
b = list(map(int,inp[2].split()))
pref =  [0 for i in range(m-n+1)]
c = []
for i in range(n):
    c.append(a[i]-b[i])
for i in range(m-n):
    j = i+1
    pref[j] =pref[i]+c[i]-c[j]
add=0
ans=[]
ans.append(0)
for i in range(q):
    l,r,x = map(int,inp[i+3].split())
    l-=1
    r-=1
    add+= (r-l+1)*x
    ans.append(add-pref[r]- (pref[l-1] if l>0 else 0))
for x in ans:
    print(x)
