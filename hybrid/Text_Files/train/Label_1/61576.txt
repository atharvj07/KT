n, k = map(int, input().split())
a = list(map(int, input().split()))

def max_sum_after_swaps(n, k, a):
    max_sum = -float('inf')
    for l in range(n):
        for r in range(l, n):
            internal = a[l:r+1]
            external = a[:l] + a[r+1:]
            internal.sort(reverse=True)
            external.sort()
            i = 0
            while i < k and i < len(external) and i < len(internal) and external[i] < internal[i]:
                internal[i], external[i] = external[i], internal[i]
                i += 1
            current_sum = sum(internal)
            max_sum = max(max_sum, current_sum)
    return max_sum if max_sum != -float('inf') else 0

print(max_sum_after_swaps(n, k, a))
