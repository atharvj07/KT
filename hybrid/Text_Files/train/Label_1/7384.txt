#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;
const int maxn = 2e5 + 5;

int n, k;
ll a[maxn];

int main() {
    cin >> n >> k;
    for (int i = 0; i < n + 1; ++i) cin >> a[i];

    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (a[i] % 2 == a[n] % 2)
            ++ans;
        else if (a[i] == 0 && k != 0)
            ++ans;
    }
    cout << ans << endl;
    return 0;
}
