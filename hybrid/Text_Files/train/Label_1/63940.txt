def is_kalindrome(arr):
    left = 0
    right = len(arr) - 1

    while left < right:
        if arr[left] != arr[right]:
            # Remove left or right and check if it's palindrome
            return is_palindrome(arr, left + 1, right) or is_palindrome(arr, left, right - 1)
        left += 1
        right -= 1
    return True

def is_palindrome(arr, left, right):
    while left < right:
        if arr[left] != arr[right]:
            return False
        left += 1
        right -= 1
    return True
