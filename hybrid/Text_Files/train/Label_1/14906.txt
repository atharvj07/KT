#include <bits/stdc++.h>
typedef long long ll;
using namespace std;

const int N = 3e5 + 5;
ll pre[N];
pair<ll, ll> a[N];

int main() {
  ll n, k;
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> a[i].first >> a[i].second;
  }
  sort(a + 1, a + 1 + n, [&](pair<ll, ll> l, pair<ll, ll> r) {
    if (l.first * r.second != l.second * r.first) {
      return l.first * r.second > l.second * r.first;
    }
    return l.first < r.first;
  });
  for (int i = 1; i <= n; i++) {
    pre[i] = pre[i - 1] + a[i].first;
  }
  ll ans = 0;
  for (int i = 1, j = 1; j <= n && i <= k; j++) {
    while (i < k && j <= n && a[j].second >= a[i + 1].second) {
      i++;
    }
    ans = max(ans, a[j].second * (pre[j] - pre[i - 1]));
  }
  cout << ans << '\n';
  return 0;
}
