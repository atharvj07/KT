#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  int k = n;
  while (k > 1) {
    bool found = false;
    for (int i = 0; i < k - 1; i++) {
      if (a[i] == a[i + 1]) {
        k--;
        a[i] = a[i] + 1;
        for (int j = i + 1; j < k; j++) { a[j] = a[j + 1]; }
        found = true;
        break;
      }
    }
    if (!found) break;
  }

  cout << k << endl;
  for (int i = 0; i < k; i++) { cout << a[i] << " "; }
  cout << endl;

  return 0;
}
