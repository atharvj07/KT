import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int n = scanner.nextInt();
        Set<String> boring = new HashSet<>();
        for (int i = 0; i < n; i++) {
            boring.add(scanner.next());
        }

        int maxLength = 0;
        int start = 0;
        for (int i = 0, j = 0; i < s.length(); i++) {
            while (j < i && boring.contains(s.substring(j, i + 1))) {
                j++;
            }
            if (i - j + 1 > maxLength) {
                maxLength = i - j + 1;
                start = j;
            }
        }

        System.out.println(maxLength + " " + start);
    }
}
