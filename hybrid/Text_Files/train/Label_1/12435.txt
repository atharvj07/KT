#include<bits/stdc++.h>
using namespace std;
int main() {
    long long a;
    cin>>a;
    if(a%10==0){
        if(a>0){
            cout<<a;
            return 0;
        }
        else{
            a=a/10;
            a=-a;
            cout<<a;
            return 0;
        }
    }
    else{
        if(abs(a%10)>abs((a/10)%10)){
            a=a/10;
            a=-a;
            cout<<a;
            return 0;
        }
        else{
            a=a/100;
            a=-a;
            cout<<a;
            return 0;
        }
    }
}
