import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long res = 0;
        for (long i = 1; i <= n; i++) {
            if (checkBinary(i)) {
                res++;
            }
        }
        System.out.println(res);
    }

    private static boolean checkBinary(long n) {
        while (n > 0) {
            if (n % 10 != 0 && n % 10 != 1) {
                return false;
            }
            n /= 10;
        }
        return true;
    }
}
