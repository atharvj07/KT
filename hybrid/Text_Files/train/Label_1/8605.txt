#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> y(n);
  for (int i = 0; i < n; ++i) cin >> y[i];
  sort(y.begin(), y.end());
  vector<int> x;
  for (int i = n - 1; i >= 0; --i) {
    int cur = y[i];
    while (cur > 1) {
      if (cur % 2 == 0) cur /= 2;
      else {
        x.push_back(cur - 1);
        cur = (cur - 1) / 2;
      }
    }
    x.push_back(cur);
  }
  reverse(x.begin(), x.end());
  for (auto e : x) cout << e << ' ';
  return 0;
}
