import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), q = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) t[i] = sc.nextInt();
        Node[] arr = new Node[k];
        for (int i = 0; i < k; i++) arr[i] = new Node();
        boolean[] vis = new boolean[n + 1];
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt(), id = sc.nextInt();
            if (type == 1) {
                if (vis[id]) continue;
                int j, mx = -1, ind = -1;
                for (j = 0; j < k; j++) {
                    if (!arr[j].valid) { ind = j; break; }
                    if (arr[j].t <= t[id - 1]) {
                        if (arr[j].t > mx) {
                            ind = j;
                            mx = arr[j].t;
                        }
                    }
                }
                if (ind == -1) ind = k - 1;
                arr[ind] = new Node(id, t[id - 1]);
                vis[id] = true;
            } else {
                int j, flg = 0;
                for (j = 0; j < k; j++) {
                    if (arr[j].id == id) {
                        flg = 1;
                        break;
                    }
                }
                if (flg == 1) System.out.println("YES");
                else System.out.println("NO");
            }
        }
    }
}

class Node {
    int id, t;
    boolean valid = false;
    Node() { }
    Node(int id, int t) {
        this.id = id;
        this.t = t;
        this.valid = true;
    }
}
