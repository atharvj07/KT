#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll solve(ll l, ll r){
    ll ans = 0;
    for(ll i = l; i <= r; i++){
        ll f = 1;
        while(i){
            if(i % 2 == 0) i /= 2, f *= 2;
            else if(i % 3 == 0) i /= 3, f *= 3;
            else break;
        }
        if(i == 1) ans += f == 1;
    }
    return ans;
}
int main(){
    ll l, r;
    cin >> l >> r;
    cout << solve(l, r) << endl;
}
