#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007

long long power(long long x, long long p) {
    if (p == 0) return 1;
    long long r = power(x, p / 2) % mod;
    if (p % 2 == 0) return (r * r) % mod;
    return (x * (r * r)) % mod;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    set<int> sell;
    int buy = 0;
    bool flag = true;
    long long ans = 1;
    while (n--) {
        string s;
        cin >> s;
        int p;
        cin >> p;
        if (s == "ADD") {
            if (buy > p) {
                flag = false;
                break;
            }
            if (!sell.empty() && (*sell.begin()) <= p) {
                flag = false;
                break;
            }
            sell.insert(p);
            ans = (ans * 2) % mod;
        } else {
            if (sell.count(p) == 0) {
                flag = false;
                break;
            }
            if (buy > p) {
                flag = false;
                break;
            }
            sell.erase(p);
            buy = p;
        }
    }
    if (sell.size() > 0) flag = false;
    if (flag)
        cout << ans << endl;
    else
        cout << "0\n";
    return 0;
}
