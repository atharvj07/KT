def count_triples(n, a):
    from collections import Counter
    counter = Counter(a)
    keys = sorted(counter.keys())
    total_triples = 0
    for i in range(len(keys)):
        for j in range(i, min(i + 3, len(keys))):
            if keys[j] - keys[i] <= 2:
                if i == j:
                    count = counter[keys[i]]
                    total_triples += count * (count - 1) * (count - 2) // 6
                elif j == i + 1:
                    total_triples += counter[keys[i]] * counter[keys[j]] * (counter[keys[i]] - 1) // 2
                    total_triples += counter[keys[i]] * counter[keys[j]] * (counter[keys[j]] - 1) // 2
                else:
                    total_triples += counter[keys[i]] * counter[keys[i + 1]] * counter[keys[j]]
    return total_triples

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(count_triples(n, a))
