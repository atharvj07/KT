import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long sum = 0;
        for (int i = 0; i < n; i++) sum += sc.nextLong();
        long res = (long)Math.ceil((double)sum / (n - 1));
        System.out.println(res);
    }
}
