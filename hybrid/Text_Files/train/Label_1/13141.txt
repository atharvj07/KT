#include <bits/stdc++.h>
using namespace std;

const int INF = 2e9;
int n, a[100005], mx, cnt;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) {
        if (a[i] > mx) {
            mx = a[i];
            cnt = 1;
        } else if (a[i] == mx) {
            cnt++;
        }
    }
    long long sum = 0;
    for (int i = 1; i <= n; i++) { sum += a[i]; }
    if (sum <= 2LL * mx * cnt) {
        for (int i = 1; i <= n; i++) { cout << -1 << ' '; }
        return 0;
    }
    int l = 1, r = n;
    while (l < r) {
        int m = (l + r + 1) >> 1;
        long long t = 0;
        for (int i = 1; i <= n; i++) {
            t += 1LL * max(0LL, mx - a[i] + 1LL) *
                 ((i + m - 2) / n - (i - 1) / n);
        }
        if (t <= mx)
            l = m;
        else
            r = m - 1;
    }
    for (int i = 1; i <= n; i++) {
        long long t = 1LL * max(0LL, mx - a[i] + 1LL) * (l / n);
        if (a[i] < mx) t++;
        cout << t << ' ';
    }
}
