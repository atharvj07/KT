#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] >= 0) {
            break;
        }
        sum += a[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] >= 0) {
            ans += a[i];
        } else {
            ans -= a[i];
        }
    }
    cout << ans + sum << endl;
    return 0;
}
