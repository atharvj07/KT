h, m, s, t1, t2 = list(map(int, input().split()))

h_s = s * 360 / 60
m_s = (m * 60 + s) * 360 / 3600
t1_s = t1 * 30
t2_s = t2 * 30
h_s = (h_s + 360) % 360
m_s = (m_s + 360) % 360
t1_s = (t1_s + 360) % 360
t2_s = (t2_s + 360) % 360
if t1_s < min(h_s, m_s):
    t1_s += 360
if t2_s < min(h_s, m_s):
    t2_s += 360
if t1 < t2:
    print('YES' if (t1_s < h_s and t1_s < m_s and t2_s < h_s and t2_s < m_s) else 'NO')
else:
    print('YES' if (t1_s > h_s and t1_s > m_s and t2_s > h_s and t2_s > m_s) else 'NO')
