// Import library
import java.util.*;
import java.io.*;

// Create class
class Main {
    public static void main(String[] args) throws IOException {
        // Read data from standard input
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] edges = new int[n - 1][2];
        for (int i = 0; i < n - 1; i++) {
            String[] line = br.readLine().split(" ");
            edges[i][0] = Integer.parseInt(line[0]);
            edges[i][1] = Integer.parseInt(line[1]);
        }
        String s = br.readLine();

        // Create graph
        Map<Integer, List<Integer>> graph = new HashMap<>();
        for (int[] edge : edges) {
            graph.computeIfAbsent(edge[0], k -> new ArrayList<>()).add(edge[1]);
            graph.computeIfAbsent(edge[1], k -> new ArrayList<>()).add(edge[0]);
        }

        // DFS to find the number of connected components
        Set<Integer> visited = new HashSet<>();
        int connectedComponents = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited.contains(i)) {
                dfs(i, graph, visited);
                connectedComponents++;
            }
        }

        // Calculate the number of possible combinations
        int numCombinations = 1;
        for (int i = 0; i < connectedComponents; i++) {
            numCombinations *= 2;
        }

        // Print the result
        System.out.println(numCombinations);
    }

    // Depth-first search
    private static void dfs(int v, Map<Integer, List<Integer>> graph, Set<Integer> visited) {
        visited.add(v);
        for (int u : graph.getOrDefault(v, new ArrayList<>())) {
            if (!visited.contains(u)) {
                dfs(u, graph, visited);
            }
        }
    }
}
