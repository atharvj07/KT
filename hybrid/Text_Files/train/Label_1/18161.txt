#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t, a, b, c, d, e, f, g;
    cin >> t;
    while (t--)
    {
        cin >> a;
        b = 0;
        for (c = 1; c <= a; c++)
        {
            d = c;
            e = 0;
            while (d != 0)
            {
                f = d % 10;
                if (e == 0)
                {
                    g = f;
                }
                else if (g != f)
                {
                    e = 1;
                    break;
                }
                d = d / 10;
            }
            if (e == 0)
            {
                b++;
            }
        }
        cout << b << endl;
    }
}
