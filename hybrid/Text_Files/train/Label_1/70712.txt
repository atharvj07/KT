import sys
read = sys.stdin.readline
n = int(read())
a = list(map(int, read().split()))
b = read().strip()
a.sort()
ans = [0]*len(b)
intro, extro = [], []
for i in range(len(b)):
    if b[i] == '0':
        intro.append(i)
    else:
        extro.append(i)
j = 0
for i in intro:
    ans[i] = a[j]
    j += 1
j = len(a)-1
for i in extro:
    ans[i] = a[j]
    j -= 1
print(*ans)
