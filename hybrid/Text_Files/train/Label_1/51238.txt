import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        boolean openBlock = false;
        String res = null;
        for (int i = 0; i < n; i++) {
            String line = in.nextLine().trim();
            if (line.startsWith("try")) {
                openBlock = true;
            } else if (line.startsWith("catch")) {
                if (!openBlock) {
                    continue;
                }
                String parts[] = line.substring(7, line.length() - 2).split(",");
                if (parts[0].equals("AE")) {
                    res = parts[1];
                }
                openBlock = false;
            } else if (line.startsWith("throw")) {
                if (!openBlock) {
                    continue;
                }
                String exception = line.substring(6);
                if (exception.equals("AE")) {
                    break;
                }
            }
        }
        if (res == null) {
            System.out.println("Unhandled Exception");
        } else {
            System.out.println(res);
        }
    }
}
