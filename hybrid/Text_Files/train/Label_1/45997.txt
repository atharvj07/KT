import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        if(n % 2 == 0){
            n /= 2;
        }
        for(int i = 3; i * i <= n; i++){
            while(n % i == 0){
                n /= i;
            }
        }
        if(n > 1){
            n /= n;
        }
        System.out.println(n);
    }
}
