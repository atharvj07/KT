#include <iostream>
#include <set>
#include <vector>
#include <algorithm>
using namespace std;
const int MAXN = 1e5 + 5;
struct point {
    int x, y;
    bool operator ==(point other) { return (x == other.x) && (y == other.y); }
    point operator -(point other) { return point{x - other.x, y - other.y}; }
    int dot(point other) { return x * other.x + y * other.y; }
    int cross(point other) { return x * other.y - y * other.x; }
} p[MAXN];
bool ccw(point a, point b, point c) {
    return (b - a).cross(c - a) > 0;
}
bool intersect(point a, point b, point c, point d) {
    if (max(a.x, b.x) < min(c.x, d.x) || max(a.y, b.y) < min(c.y, d.y)) return 0;
    if (ccw(a, b, c) != ccw(a, b, d) && ccw(c, d, a) != ccw(c, d, b)) return 1;
    if (min(a.x, b.x) <= max(c.x, d.x) && min(c.x, d.x) <= max(a.x, b.x) && min(a.y, b.y) <= max(c.y, d.y) && min(c.y, d.y) <= max(a.y, b.y)) return 1;
    return 0;
}
int32_t main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> p[i].x >> p[i].y; }
    for (int i = n + 1; i <= n + m; i++) { cin >> p[i].x >> p[i].y; }
    vector<pair<int, int>> edges;
    for (int i = 1; i <= n; i++) {
        for (int j = n + 1; j <= n + m; j++) {
            edges.emplace_back(i, j);
        }
    }
    sort(edges.begin(), edges.end(), [&](const pair<int, int> &a, const pair<int, int> &b) {
        return p[a.second].x < p[b.second].x;
    });
    set<int> s;
    for (const auto &[u, v] : edges) {
        auto it = s.lower_bound(u);
        if (it != s.end() && intersect(p[u], p[v], p[*it], p[v])) continue;
        if (it != s.begin() && intersect(p[u], p[v], p[*prev(it)], p[v])) continue;
        s.insert(v);
    }
    if ((int)s.size() == n) {
        cout << "Yes\n";
    } else {
        cout << "No\n";
    }
    return 0;
}
