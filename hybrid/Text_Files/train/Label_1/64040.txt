import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    m = int(input())
    heroes = []
    for _ in range(m):
        p, s = map(int, input().split())
        heroes.append((p, s))

    a.sort()
    heroes.sort(reverse=True)

    days = 0
    defeated = 0
    for p, s in heroes:
        while defeated < n:
            if p < a[defeated]:
                days += 1
                break
            defeated += s
    
    if defeated < n:
        print(-1)
    else:
        print(days)
