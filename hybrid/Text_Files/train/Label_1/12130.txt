#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll arr[200005];
int main() {
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        cin >> arr[i];
    }
    ll id = 2, idx = 0;
    ll mx = -1;
    for (ll i = n / 2 + 1; i >= 1; i--) {
        if (arr[i] > mx) {
            mx = arr[i];
            id = i;
        }
    }
    ll j;
    for (j = n; j > id && arr[j] != mx; j--);
    idx = j;
    if (id == n / 2 + 1) {
        for (ll i = 1; i <= idx; i++) {
            cout << arr[i] << " ";
        }
        for (ll i = idx + 1; i < n + 1; i++) {
            cout << arr[i] << " ";
        }
        cout << endl;
    } else {
        ll m = idx;
        for (ll i = id + 1, k = 1; i <= n, k <= id; i++, k++) {
            arr[i] = arr[k];
        }
        for (ll i = id, k = m + 1; i >= 1, k <= n; i--, k++) {
            arr[i] = arr[k];
        }
        for (ll i = 1; i <= n; i++) {
            cout << arr[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
