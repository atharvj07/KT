import sys
n=int(input())
a=list(map(int,input().split()))
g=[[],[]]
for i in range(n):
    for j in range(i+1,n):
        if a[i]>a[j]:
            g[0].append(i+1)
            g[1].append(j+1)
        else:
            g[1].append(i+1)
            g[0].append(j+1)
print(n*(n-1)//2)
for i in range(n*(n-1)//2):
    print(g[0][i],g[1][i])
