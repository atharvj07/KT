import java.util.*;

class Main {

    static class Edge {
        int u, v;

        public Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Event {
        int from, to, town;

        public Event(int from, int to, int town) {
            this.from = from;
            this.to = to;
            this.town = town;
        }
    }

    static int N, M, E, S, T, R;
    static List<Edge> edges = new ArrayList<>();
    static List<Event> events = new ArrayList<>();
    static boolean[][] graph;
    static int[] dist;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        E = sc.nextInt();
        S = sc.nextInt();
        T = sc.nextInt();
        R = sc.nextInt();
        graph = new boolean[N][N];

        for (int i = 0; i < M; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            graph[u][v] = graph[v][u] = true;
            edges.add(new Edge(u, v));
        }

        for (int i = 0; i < E; i++) {
            int from = sc.nextInt();
            int to = sc.nextInt();
            int town = sc.nextInt();
            events.add(new Event(from, to, town));
        }

        // Apply BFS to find the shortest path from S to each town
        dist = new int[N];
        Arrays.fill(dist, -1);
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(S);
        dist[S] = 0;

        while (!queue.isEmpty()) {
            int u = queue.poll();

            for (int v = 0; v < N; v++) {
                if (graph[u][v] && dist[v] == -1) {
                    queue.offer(v);
                    dist[v] = dist[u] + 1;
                }
            }
        }

        // If T is unreachable, output -1
        if (dist[T] == -1) {
            System.out.println(-1);
            return;
        }

        // Compute the minimum number of moves and ocarina blows required to reach T
        int minMoves = Integer.MAX_VALUE;
        for (Event event : events) {
            // Add the event to the graph and update the shortest path
            graph[event.from][event.to] = graph[event.to][event.from] = true;
            int d = dist[event.town];
            for (int u = 0; u < N; u++) {
                for (int v = 0; v < N; v++) {
                    if (graph[u][v]) {
                        dist[v] = Math.min(dist[v], d + dist[u] + 1);
                    }
                }
            }

            // Check if the event helps to reach T faster
            if (dist[T] != -1) {
                int moves = dist[T] + 1;
                int blows = (moves + R - 1) / R;
                minMoves = Math.min(minMoves, moves + blows);
            }

            // Remove the event from the graph
            graph[event.from][event.to] = graph[event.to][event.from] = false;
        }

        // Output the minimum number of moves and ocarina blows
        if (minMoves == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minMoves);
        }
    }
}
