import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();
        long p = sc.nextLong();
        long q = sc.nextLong();
        System.out.println(solve(n, m, a, b, p, q));
    }

    static long solve(long n, long m, long a, long b, long p, long q) {
        long ans = Long.MAX_VALUE;
        for (long x = 0; x < n; x++) {
            long d = (m - q * x) % p;
            if (d < 0) d += p;
            long y = (d + b * x) / a;
            if (y >= 0 && y < n) ans = Math.min(ans, x + y);
        }
        return ans == Long.MAX_VALUE ? -1 : ans;
    }
}
