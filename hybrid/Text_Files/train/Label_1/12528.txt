#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a1, k;
    cin >> a1 >> k;
    string s = to_string(a1);
    int min_digit = s[0] - '0';
    int max_digit = s[0] - '0';
    for (int i = 1; i < s.size(); i++) {
      int digit = s[i] - '0';
      min_digit = min(min_digit, digit);
      max_digit = max(max_digit, digit);
    }
    for (int i = 0; i < k; i++) {
      a1 += min_digit * max_digit;
      s = to_string(a1);
      min_digit = s[0] - '0';
      max_digit = s[0] - '0';
      for (int j = 1; j < s.size(); j++) {
        int digit = s[j] - '0';
        min_digit = min(min_digit, digit);
        max_digit = max(max_digit, digit);
      }
    }
    cout << a1 << endl;
  }
  return 0;
}
