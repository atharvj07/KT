m, k = map(int, input().split())
graph, a = {}, {}
for _ in range(m):
    p1, p2 = map(int, input().split())
    graph[p1] = graph.get(p1, set())
    graph[p1].add(p2)
    graph[p2] = graph.get(p2, set())
    graph[p2].add(p1)
    a[p1] = a.get(p1, 0) + 1
    a[p2] = a.get(p2, 0) + 1
l = []
for i in graph:
    l += [i]
l.sort()
for i in l:
    count = 0
    p = []
    for j in graph[i]:
        o = (len(graph[i] & graph[j]) / a[i]) * 100
        if o >= k:
            count += 1
            p += [j]
    print(str(i) + ':', count, *p)
