import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int k = scan.nextInt();
        int s = scan.nextInt();
        int t = scan.nextInt();
        int[] c = new int[n];
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scan.nextInt();
            v[i] = scan.nextInt();
        }
        int[] g = new int[k];
        for (int i = 0; i < k; i++) {
            g[i] = scan.nextInt();
        }
        int min = -1;
        for (int i = 0; i < n; i++) {
            int time = 0;
            int fuel = v[i];
            boolean acc = false;
            for (int j = 0; j < k; j++) {
                if (fuel <= 0) {
                    break;
                }
                if (!acc && g[j] - time <= fuel) {
                    time += g[j] - time;
                    fuel -= g[j] - time;
                } else {
                    acc = true;
                    if (g[j] - time <= (fuel << 1)) {
                        time += g[j] - time;
                        fuel -= (g[j] - time) >> 1;
                    } else {
                        fuel = 0;
                    }
                }
            }
            if (fuel > 0) {
                if (!acc) {
                    time += fuel;
                } else {
                    if (s - time <= (fuel << 1)) {
                        time += s - time;
                    } else {
                        fuel = 0;
                    }
                }
            }
            if (time <= t && (min == -1 || c[i] < min)) {
                min = c[i];
            }
        }
        System.out.println(min);
    }
}
