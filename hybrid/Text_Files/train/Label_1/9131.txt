#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> temp(n);
  for (int& i : temp) std::cin >> i;
  if (n == 2 || (n > 2 && temp[1] - temp[0] == temp[2] - temp[1]))
    std::cout << temp[n - 1] + temp[1] - temp[0];
  else
    std::cout << temp[n - 1];
  return 0;
}
