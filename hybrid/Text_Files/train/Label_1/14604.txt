using namespace std;

using ll = long long int;
using ull = unsigned long long int;

constexpr int MOD = 1'000'000'007;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (auto &&x : a) { cin >> x; }
  ll num = 1LL;
  ll den = 1LL;
  for (auto &&x : a) {
    (num *= ((x - 1) % MOD)) %= MOD;
    (den *= x) %= MOD;
  }
  ll gcd = __gcd(num, den);
  num /= gcd;
  den /= gcd;
  cout << num << "/" << den << endl;

  return 0;
}
