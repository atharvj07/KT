import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] costs = new int[n];
        for(int i = 0; i < n; i++)
            costs[i] = in.nextInt();
        TreeSet<Integer> ans = new TreeSet<>();
        int sum = 1;
        ans.add(1);
        for(int cost : costs) {
            if(cost > 0) {
                TreeSet<Integer> temp = new TreeSet<>();
                for(int x : ans) {
                    temp.add(x * cost);
                }
                ans.addAll(temp);
            }
            else if(cost == 0) {
                ans.add(0);
            }
            else {
                TreeSet<Integer> temp = new TreeSet<>();
                for(int x : ans) {
                    temp.add(x * cost);
                }
                temp.add(cost);
                ans = temp;
            }
        }
        for(int x : ans)
            System.out.print(x + " ");
        System.out.println();
    }
}
