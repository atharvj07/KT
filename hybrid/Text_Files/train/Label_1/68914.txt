from collections import Counter
n = int(input())
a = list(map(int, input().split()))
cnt = Counter()
L = R = ans = 0
for i in range(n):
    cnt[a[i]] += 1
    while L < R and max(cnt.values()) - min(cnt.values()) > 1:
        cnt[a[L]] -= 1
        if cnt[a[L]] == 0:
            del cnt[a[L]]
        L += 1
    if max(cnt.values()) - min(cnt.values()) <= 1:
        ans = max(ans, R - L + 1)
    R += 1
print(ans)
