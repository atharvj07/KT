#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    bool is_possible = true;
    if (n % 2 == 0) {
        for (int i = 0; i < n; i += 2) {
            if (a[i] < a[i + 1]) {
                is_possible = false;
                break;
            }
        }
    } else {
        for (int i = 1; i < n; i += 2) {
            if (a[i] > a[i - 1]) {
                is_possible = false;
                break;
            }
        }
    }
    if (is_possible) {
        sort(a.begin(), a.end());
        for (auto &x : a) cout << x << " ";
        cout << endl;
    } else {
        cout << "Impossible" << endl;
    }
    return 0;
}
