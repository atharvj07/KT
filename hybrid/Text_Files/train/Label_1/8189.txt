#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 5;
vector<int> adj[MAXN];
int dist[MAXN];
int dfs(int u, int p) {
    dist[u] = 0;
    for (int v : adj[u]) {
        if (v == p) continue;
        dist[u] = max(dist[u], 1 + dfs(v, u));
    }
    return dist[u];
}
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1);
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        ans += dist[i];
    }
    cout << ans * 2 << endl;
    return 0;
}
