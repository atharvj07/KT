#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>

using namespace std;

int main() {
    int k, n;
    cin >> k >> n;
    vector<string> s(k);
    unordered_map<string, int> m;
    for (int i = 0; i < k; i++) {
        cin >> s[i];
        m[s[i]]++;
        if (m[s[i]] > 1) {
            cout << "-1" << endl;
            return 0;
        }
    }
    string ans = "";
    for (int i = 0; i < n; i++) {
        unordered_map<char, int> cnt;
        for (int j = 0; j < k; j++) {
            cnt[s[j][i]]++;
        }
        char c1 = 0, c2 = 0;
        int ccnt1 = 0, ccnt2 = 0;
        for (auto it : cnt) {
            if (it.second % 2 == 1) {
                if (ccnt1 == 0) {
                    c1 = it.first;
                    ccnt1 = it.second;
                } else if (ccnt2 == 0) {
                    c2 = it.first;
                    ccnt2 = it.second;
                } else {
                    cout << "-1" << endl;
                    return 0;
                }
            }
        }
        if (ccnt1 != 0 && ccnt2 != 0) {
            ans.push_back(c1);
        } else {
            ans.push_back(s[0][i]);
        }
    }
    cout << ans << endl;
    return 0;
}
