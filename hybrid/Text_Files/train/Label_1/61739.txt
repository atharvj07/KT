import sys

input = sys.stdin.readline

n = int(input())
g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

vis = [0] * (n + 1)
lbl = [0] * (n - 1)
ans = [0] * (n - 1)

def dfs(u, p, l):
    vis[u] = 1
    color = [0] * (n - 1)
    for v in g[u]:
        if vis[v]:
            if v != p:
                color[lbl[l]] = 1
        else:
            dfs(v, u, l + 1)
    i = 0
    for x in g[u]:
        if x != p:
            while color[i] == 1:
                i += 1
            lbl[l + 1] = i
            ans[l] = i
            i += 1

dfs(1, 0, 0)
for x in ans:
    print(x)
