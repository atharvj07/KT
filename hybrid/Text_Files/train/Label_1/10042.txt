#include <iostream>
using namespace std;
int main()
{
    long long k, d, t;
    cin >> k >> d >> t;
    __int128_t ans = 0;
    while (t > 0) {
        __int128_t x = k;
        __int128_t y = k + d;
        if (x < t) {
            ans += x;
            t -= x;
        } else {
            ans += t;
            break;
        }
        x = y;
        y += d;
        if (x < (t << 1)) {
            ans += x;
            t -= x >> 1;
        } else {
            ans += (t << 1);
            break;
        }
    }
    cout.precision(20);
    cout << (double)ans << endl;
    return 0;
}
