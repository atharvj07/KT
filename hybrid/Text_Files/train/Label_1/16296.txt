#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-9;
int main() {
  int n, T;
  cin >> n >> T;
  vector<pair<double, double>> v;
  for (int i = 0; i < n; i++) {
    double a, t;
    cin >> a >> t;
    v.push_back({t, a});
  }
  sort(v.begin(), v.end());
  double low = 0.000001;
  double high = 5e10;
  while (high - low >= EPS) {
    double mid = (low + high) / 2;
    double totT = 0, totA = 0;
    for (auto &it : v) {
      if (it.second >= mid) {
        totT += it.first * mid;
        totA += mid;
      } else {
        totT += it.first * it.second;
        totA += it.second;
      }
    }
    double temp = totT / totA;
    if (abs(temp - T) < EPS) {
      cout << fixed << setprecision(15) << totA << endl;
      return 0;
    } else if (temp > T) {
      high = mid - EPS;
    } else {
      low = mid + EPS;
    }
  }
  cout << 0 << endl;
  return 0;
}
