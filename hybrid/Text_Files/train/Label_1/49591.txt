import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();

            boolean[] used = new boolean[26];
            StringBuilder mex = new StringBuilder();

            for (int i = 0; i < n; i++) {
                used[s.charAt(i) - 'a'] = true;
            }

            for (int i = 0; i < 26; i++) {
                if (!used[i]) {
                    mex.append((char) ('a' + i));
                }
            }

            System.out.println(mex);
        }
    }
}
