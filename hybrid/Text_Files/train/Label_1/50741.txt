import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static final int INF = Integer.MAX_VALUE;
    private static int n, v, e;
    private static int[] a, b;
    private static int[][] g;
    private static boolean[] vis;
    private static ArrayList<Integer> ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        v = sc.nextInt();
        e = sc.nextInt();
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        g = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(g[i], INF);
        }
        for (int i = 0; i < e; i++) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            g[x][y] = g[y][x] = 1;
        }
        vis = new boolean[n];
        ans = new ArrayList<>();
        if (!dfs(0, 0)) {
            System.out.println("NO");
        } else {
            System.out.println(ans.size());
            for (int i : ans) {
                System.out.println(i);
            }
        }
    }

    private static boolean dfs(int u, int d) {
        if (u == n - 1) {
            return true;
        }
        vis[u] = true;
        for (int v = 0; v < n; v++) {
            if (!vis[v] && g[u][v] < INF) {
                int t = Math.min(v, a[u]);
                a[u] -= t;
                a[v] += t;
                ans.add(u * n + v * n + t);
                if (a[u] == b[u] && dfs(v, d + 1)) {
                    return true;
                }
                a[u] += t;
                a[v] -= t;
                ans.remove(ans.size() - 1);
            }
        }
        return false;
    }
}
