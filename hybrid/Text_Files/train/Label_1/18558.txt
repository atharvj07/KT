#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll p[51],t[51],n,c,L,R;
int main(){
    cin>>n>>c;
    for(int i=0;i<n;i++) cin>>p[i];
    for(int i=0;i<n;i++) cin>>t[i];
    for(int i=0;i<n;i++){
        L+=max(0LL,p[i]-c*(t[i]+i));
        R+=max(0LL,p[n-1-i]-c*((t[n-1-i]+i)));
    }
    if(L==R) cout<<"Tie\n";
    else if(L>R) cout<<"Limak\n";
    else cout<<"Radewoosh\n";
}
