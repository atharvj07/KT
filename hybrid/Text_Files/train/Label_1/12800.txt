#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
ll a[N], x[N], y[N];
int main() {
    int n, m;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    cin >> m;
    for (int i = 0; i < m; i++) { cin >> x[i] >> y[i]; }
    for (int i = 0; i < m; i++) {
        ll ans = 0;
        ll maxa = 0;
        for (int j = 0; j < n; j++) { maxa = max(maxa, a[j]); }
        if (maxa < x[i]) ans += x[i] - maxa;
        ll su = 0;
        for (int j = 0; j < n; j++) {
            if (a[j] >= x[i]) su += a[j];
        }
        if (su < y[i]) ans += y[i] - su;
        cout << ans << endl;
    }
    return 0;
}
