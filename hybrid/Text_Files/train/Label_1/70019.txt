n, k = map(int, input().split())
rooms = input().strip()

# Convert room availability to a list of indices of available rooms
available_rooms = [i for i, status in enumerate(rooms) if status == '0']

# Initialize variables to store the minimum possible maximum distance
min_max_distance = float('inf')

# Loop through the available rooms considering each as Farmer John's room
for i in range(len(available_rooms) - k):
    # Calculate the maximum distance from Farmer John's room to his farthest cow
    # by considering the current room and k more rooms for the cows
    max_distance = max(available_rooms[i + k] - available_rooms[i], available_rooms[i + (k + 1) // 2] - available_rooms[i])
    # Update the minimum possible maximum distance
    min_max_distance = min(min_max_distance, max_distance)

print(min_max_distance)
