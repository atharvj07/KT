#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 +9;
#define ll long long
ll trie[N][26];
ll ending[N];
ll len = 0, nodes = 1;
void insert(string s) {
    ll cur = 0;
    for(char c : s) {
        ll id = c - 'a';
        if(!trie[cur][id]) {
            trie[cur][id] = ++nodes;
        }
        cur = trie[cur][id];
    }
    ending[cur] = 1;
}
int go(ll x) {
    for(ll i = 0; i < 26; i++) {
        if(trie[x][i]) {
            if(ending[trie[x][i]]) {
                if(!go(trie[x][i])) return 1;
            } else {
                return 0;
            }
        }
    }
    return 1;
}
int main() {
    ll n, k; cin >> n >> k;
    for(ll i = 0; i < n; i++) {
        string s; cin >> s;
        insert(s);
    }
    if(k % 2 == 0 || go(0)) {
        cout << "Second" << endl;
    } else {
        cout << "First" << endl;
    }
    return 0;
}
