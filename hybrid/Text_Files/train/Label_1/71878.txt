import bisect
n,m,k=map(int,input().split())
st=sorted([0]+list(map(int,input().split()))+[n])
c=list(map(int,input().split()))
a=[-1]*k
for i in range(k):
    p,t=0,0
    if c[i]<=st[-1]:
        l=bisect.bisect_right(st,c[i])
        for j in range(l):
            if st[j]+c[i]>st[j+1]-1: t=max(t,st[j+1]-1)
    if t: p=bisect.bisect_right(st,t)-1
    a[i]=(st[p+1]-st[p])*(c[i]-st[p])
print(min(a))
