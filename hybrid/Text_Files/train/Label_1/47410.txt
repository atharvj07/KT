import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            BigInteger p = in.nextBigInteger();
            BigInteger q = in.nextBigInteger();
            BigInteger gcd = p.gcd(q);
            System.out.println(p.divide(gcd));
        }
    }
}
