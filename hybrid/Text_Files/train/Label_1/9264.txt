#include <bits/stdc++.h>
#define N 200005
using namespace std;
int n, a[N], b[N], p[N], cnt, ans;
int find(int x) {
  if (x == p[x]) return x;
  return p[x] = find(p[x]);
}
int main() {
  cin >> n;
  for (int i = 1; i <= 2 * n; i++) p[i] = i;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
    int fa = find(a[i]), fb = find(b[i]);
    if (fa == fb) {
      cnt++;
      continue;
    }
    p[fa] = fb;
  }
  ans = 1LL * (1LL << cnt) % 1000000007;
  cout << ans << endl;
  return 0;
}
