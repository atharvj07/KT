#include <cstdio>
#include <algorithm>
int main() {
    int t,n,a[100];
    bool chck;
    scanf("%d",&t);
    while(t--) {
        scanf("%d",&n);
        for(int i=0;i<n;i++) scanf("%d",a+i);
        std::sort(a,a+n);
        chck=1;
        for(int i=0;i<n;i++) {
            if((i!=0&&a[i]==a[i-1])||i>0&&a[i]==0) {chck=0;break;}
            if(a[i]>=8&&n-i>=11) {a[i]-=8; break;}
        }
        for(int i=n-1;i>=0&&chck;i--) {
            if(a[i]!=0) {chck=0; break;}
        }
        printf("%s\n", chck ? "YES" : "NO");
    }
}
