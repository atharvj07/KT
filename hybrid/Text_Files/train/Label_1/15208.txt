#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s(n, 'a');
    int x = 1;
    for (int i = 0; i < k; i++) {
        s[i] += i;
        x = i;
    }
    int y = k;
    for (int i = x + 1; i < n; i++) {
        if (s[i - 1] - s[i - x - 1] != y) {
            s[i] += x + 1;
            y = x + 1;
        } else if (s[i - 1] - s[i - x - 1] > y) {
            s[i] += x;
            y = x;
        } else {
            while (y - 1 > 0 && s[i - 1] - s[i - x - 1] == y - 1) {
                y--;
            }
            s[i] += x + 1;
            y = x + 1;
        }
    }
    cout << s << endl;
    return 0;
}
