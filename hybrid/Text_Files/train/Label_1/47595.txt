import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int maxx = sc.nextInt();
        int maxy = sc.nextInt();
        int w = sc.nextInt();

        int l = 1, r = maxy;
        long ans = -1;
        int ansX = -1, ansY = -1;

        while (l <= r) {
            int m = (l + r) / 2;
            long curr = m * (1L * m + 1) / 2;

            if (curr < w) {
                l = m + 1;
            } else {
                ans = curr;
                ansX = m;
                ansY = (w + m - 1) / m;
                r = m - 1;
            }
        }

        if (ansX > maxx || ansY > maxy) {
            ans = -1;
        }

        System.out.println(ans);
    }

    private static int rev(int x) {
        StringBuilder sb = new StringBuilder(Integer.toString(x));
        sb.reverse();
        int rev = Integer.parseInt(sb.toString());
        return rev;
    }
}
