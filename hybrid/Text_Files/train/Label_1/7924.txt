#include <iostream>
using namespace std;
int main(){
    int t; cin >> t; while(t--){
        int n; cin >> n;
        bool a[n][5];
        for(int i=0; i<n; i++)
            for(int j=0; j<5; j++)
            cin >> a[i][j];
        int cnt[5] = {};
        for(int i=0; i<n; i++)
            for(int j=0; j<5; j++)
            cnt[j] += a[i][j];
        if(cnt[0] || cnt[1] || cnt[2] || cnt[3] || cnt[4] < n/2)
            cout << "NO\n";
        else cout << "YES\n";
    }
}
