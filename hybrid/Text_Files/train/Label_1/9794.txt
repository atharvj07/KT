#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int ans = 0;
        for (int i = 1; i <= n; ++i) {
            if (s[i - 1] == 'R' && s[i % n] == 'L') ++ans;
        }
        cout << ans << "\n";
    }
    return 0;
}
