#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a, b;
int main() {
    cin >> a >> b;
    int btza = 0, btzb = 0;
    for (int i = 60; i >= 0; i--) {
        if ((a >> i) & 1) btza++;
        if ((b >> i) & 1) btzb++;
        if (btza > 1 || btzb > 1) break;
    }
    int ans = ((btza == 1) ? 1 : 0);
    for (ll i = a; i <= b; i++) {
        int btzi = 0;
        for (int j = 60; j >= 0; j--) {
            if ((i >> j) & 1) btzi++;
            if (btzi > 1) break;
        }
        ans += ((btzi == 1) ? 1 : 0);
    }
    cout << ans << endl;
}
