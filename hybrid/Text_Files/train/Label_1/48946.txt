import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int s = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int incorrect = 0;
        for (int i = 0; i < n; i++) {
            if (i != s && a[i] == 0) {
                incorrect++;
            } else if (i == s && a[i] != n - 1) {
                incorrect++;
            }
        }
        System.out.println(incorrect);
    }
}
