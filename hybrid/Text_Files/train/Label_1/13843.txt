#include<bits/stdc++.h>
using namespace std;
long long int FINDpatt(string s)
{
    long long int dp[3][100005];
    long long int n=s.size();
    memset(dp,0,sizeof(dp));
    for(long long int i=0;i<n;i++)
    {
        dp[0][i]=0;
        dp[1][i]=0;
        dp[2][i]=0;
    }
    if(s[0]=='Q')
    {
        dp[0][0]=1;
    }
    for(long long int i=1;i<n;i++)
    {
        if(s[i]=='Q')
        {
            dp[0][i]=dp[0][i-1]+1;
        }
        else
        {
            dp[0][i]=dp[0][i-1];
        }
        if(s[i]=='A' and dp[0][i]>0)
        {
            dp[1][i]=dp[0][i-1]+dp[1][i-1];
        }
        else
        {
            dp[1][i]=dp[1][i-1];
        }
        if(s[i]=='Q' and dp[1][i]>0)
        {
            dp[2][i]=dp[1][i-1]+dp[2][i-1];
        }
        else
        {
            dp[2][i]=dp[2][i-1];
        }
    }
    return dp[2][n-1];
}
int main()
{
    string s;
    cin>>s;
    cout<<FINDpatt(s)<<endl;
    return 0;
}
