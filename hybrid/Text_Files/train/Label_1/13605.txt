#include <bits/stdc++.h>
using namespace std;
#define int long long
bool f(string s){
    int cnt[26] = {0};
    for(auto x: s) cnt[x-'a']++;
    sort(cnt, cnt + 26);
    int sum = 0, mx = 0;
    for(int i = 0; i < 26; i++){
        sum += cnt[i];
        mx = max(mx, cnt[i]);
    }
    return mx <= sum / 2;
}
main(){
    string s;
    cin >> s;
    if(f(s)) cout << "Yes";
    else cout << "No";
}
