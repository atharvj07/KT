#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 998244353;
int powmod(int a, int p) {
  if (p == 0)
    return 1;
  if (p == 1)
    return a;
  int val = powmod(a, p / 2);
  val = (1LL * val * val) % mod;
  if (p % 2 == 1)
    val = (1LL * val * a) % mod;
  return val;
}
int divide(vector<int> &a, vector<int> &b) {
  int n = a.size(), m = b.size(), groups = 0, j = 0;
  bool ok = 0;
  ll ans = 1;
  for (int i = 0; i < n; i++) {
    if (!ok) {
      if (a[i] > b[j])
        return 0;
      if (a[i] == b[j]) {
        ok = 1;
        groups = 1;
        continue;
      }
    }
    if (ok) {
      if (a[i] > b[j]) {
        ok = 0;
        j++;
        if (groups > 1)
          ans = ans * groups % mod;
        groups = 0;
      }
      if (a[i] == b[j])
        groups++;
    }
  }
  if (groups > 1)
    ans = ans * groups % mod;
  if (groups)
    ans = ans * (j + 1) % mod;
  return ans;
}
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++)
    cin >> a[i];
  for (int i = 0; i < m; i++)
    cin >> b[i];
  cout << divide(a, b) << endl;
  return 0;
}
