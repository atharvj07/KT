#include <iostream>
using namespace std;
int main() {
    int h1, h2, a, b, k = 0;
    cin >> h1 >> h2 >> a >> b;
    h1 -= h2;
    a -= b;
    if (h1 > 0 && a <= 0 || h1 < 0 && a >= 0) {
        cout << -1 << endl;
        return 0;
    }
    k = abs(h1) / a;
    if (k > 0) k++;
    cout << k << endl;
    return 0;
}
