#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
const int maxn = 2e5 + 5;
vector<long long> dp(maxn, 0);  // dp[i] = # of nodes at distance exactly i
vector<vector<int>> adj(maxn, vector<int>());

long long dfs(int u, int p, int dist) {
    dp[dist] = 1;  // Mark the current node as visited
    for (int v : adj[u]) {
        if (v != p) {
            dp[dist] = (dp[dist] + dfs(v, u, dist + 1)) % mod;
        }
    }
    return dp[dist];
}

int main() {
    int n, x;
    cin >> n >> x;
    vector<int> d(n);
    for (int i = 0; i < n; i++) { cin >> d[i]; }
    for (int i = 0; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 0, 0);  // Start DFS from node 1 with parent 0 and distance 0
    long long ans = 0;
    for (int i = 0; i <= x; i++) { ans = (ans + dp[i]) % mod; }
    cout << ans << endl;
    return 0;
}
