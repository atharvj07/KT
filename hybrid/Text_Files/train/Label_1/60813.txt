def max_equal_sum(n, d):
    i, j, sum1, sum3 = 0, n-1, d[0], d[n-1]
    max_sum = 0
    while i < j:
        if sum1 < sum3:
            i += 1
            sum1 += d[i]
        elif sum1 > sum3:
            j -= 1
            sum3 += d[j]
        else:
            max_sum = sum1
            i += 1
            j -= 1
            if i < n:
                sum1 += d[i]
            if j >= 0:
                sum3 += d[j]
    return max_sum

# Reading input
n = int(input())
d = list(map(int, input().split()))

# Output the result
print(max_equal_sum(n, d))
