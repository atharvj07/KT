#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
const ll N = 2e5 + 5;
int n;
struct mat{
	ll out, in;
};
bool cmp(const mat &a, const mat &b){
	return a.out * b.in < a.in * b.out;
}
vector<mat> v;
ll go(int l, int r){
	if(l == r)return 1;
	int mid = l + r >> 1;
	ll ans = 0;
	for(int i = l; i <= mid; i++){
		ans = (ans + go(mid + 1, r)) % mod;
	}
	sort(v.begin() + l, v.begin() + mid + 1, cmp);
	int sz = 0, last = LLONG_MIN;
	for(int i = mid + 1; i <= r; i++){
		if(v[i].out >= last)last = v[i].in, sz++;
	}
	ans = (ans + (1LL << sz) % mod) % mod;
	return ans;
}
int main(){
	cin >> n;
	v.resize(n);
	for(int i = 0; i < n; i++)cin >> v[i].out >> v[i].in;
	cout << go(0, n - 1) << "\n";
	return 0;
}
