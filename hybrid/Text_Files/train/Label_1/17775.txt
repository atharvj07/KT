#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
const int mx = 1e6, inf = 1e9;
map<int, int> last;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int a, b, n;
  cin >> a >> b >> n;
  while (n--) {
    int l, t, m;
    cin >> l >> t >> m;
    int k = (l - 1) / b, ans = 0;
    for (int i = 0; i <= m; i++) {
      int lst = last[k + i];
      if (lst == 0 || lst < l)
        ans = max(ans, k + i);
      else if (lst + 1 <= l + i * b && t >= i + 1) {
        if (l + i * b + 1 <= lst)
          ans = mx;
        else
          ans = max(ans, k + i);
      }
    }
    if (ans == 0)
      cout << -1 << '\n';
    else {
      for (int i = l; i <= ans * b + l - 1; i++) last[i / b] = i;
      cout << ans * b + l - 1 << '\n';
    }
  }
}
