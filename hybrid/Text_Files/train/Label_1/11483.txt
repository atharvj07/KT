#include <bits/stdc++.h>
#define vi vector<int>
using namespace std;
const int mx = 150001;
bool up[mx];
bool down[mx];
int main() {
  int n;
  cin >> n;
  vi a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i < n - 1; i++) {
    if (a[i] < a[i - 1] && a[i] < a[i + 1]) {
      down[i + 1] = true;
    }
    if (a[i] > a[i - 1] && a[i] > a[i + 1]) {
      up[i + 1] = true;
    }
  }
  long long ans = 0;
  for (int i = 1; i < n; i++) {
    if (up[i]) {
      ans += down[i - 1];
    } else if (down[i]) {
      ans += up[i - 1];
    }
  }
  cout << ans << endl;
  return 0;
}
