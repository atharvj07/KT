#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
#define pb push_back
#define all(x) (x).begin(),(x).end()

int main() {
    int n,i,j;
    cin >> n;
    vector<ll> v(n);
    for(i=0; i<n; i++) cin >> v[i];
    ll XOR=0,cnt=0;
    for(i=0; i<n; i++){
        XOR^=v[i];
        if(XOR==0) cnt++, XOR=v[i];
    }
    if(cnt>1) cout << -1;
    else cout << cnt;
    return 0;
}
