#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, l;
    cin >> n >> k >> l;
    vector<int> d(n);
    for (int i = 0; i < n; i++) cin >> d[i];
    vector<int> p(2 * k);
    for (int i = 0; i < k; i++) { p[i] = i; }
    for (int i = k; i < 2 * k; i++) { p[i] = k - (i - k) - 1; }
    bool f = true;
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j < 2 * k; j++) {
        if (d[i] + p[j] > l) { f = false; }
      }
    }
    cout << (f ? "Yes" : "No") << endl;
  }
  return 0;
}
