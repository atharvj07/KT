#include <cstdio>

bool solve(int n, int m, int k) {
    if (n % 2 == 0 && m % 2 == 0)
        return k % 2 == 0;
    if (n % 2 == 1 && m % 2 == 1)
        return k % 2 == 1;
    if (n % 2 == 1)
        return k + m % 2 == (n * m) % 2;
    return k + n % 2 == (n * m) % 2;
}

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int n, m, k;
        scanf("%d%d%d", &n, &m, &k);
        puts(solve(n, m, k) ? "YES" : "NO");
    }
}
