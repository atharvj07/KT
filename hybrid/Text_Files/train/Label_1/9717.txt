#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n;
    cin >> n;
    vector<ll> a(n);
    map<ll, ll> mp;
    
    for(int i=0; i<n; i++)
    {
        cin >> a[i];
        mp[a[i]]++;
    }
    ll ans = 1;
    for(auto it: mp)
    {
        ll cnt = it.second;
        ll x = 1;
        while(cnt--)
            x = (x * it.first) % mod;
        ans = (ans * x) % mod;
    }
    cout << ans << "\n";
    return 0;
}
