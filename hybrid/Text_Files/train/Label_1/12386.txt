#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 5;
int n, q, fa[maxn], sz[maxn];
vector<int> G[maxn];
void dfs1(int u) {
  sz[u] = 1;
  for (int& v : G[u]) {
    if (fa[u] == v) continue;
    fa[v] = u;
    dfs1(v);
    sz[u] += sz[v];
  }
}
int dfs2(int u, int p, int k) {
  int mx = sz[u];
  for (int& v : G[u]) {
    if (v == p || v == k) continue;
    mx = max(mx, dfs2(v, u, k) + sz[u] - sz[v]);
  }
  return mx;
}
int main() {
  cin >> q;
  while (q--) {
    cin >> n;
    for (int i = 1; i <= n; ++i) G[i].clear();
    for (int i = 1; i <= n - 1; ++i) {
      int u, v;
      cin >> u >> v;
      G[u].push_back(v);
      G[v].push_back(u);
    }
    dfs1(1);
    int ans = 0;
    for (int i = 1; i <= n; ++i) {
      for (int& j : G[i]) {
        if (j == fa[i]) continue;
        ans = max(ans, dfs2(j, i, i) + sz[j]);
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
