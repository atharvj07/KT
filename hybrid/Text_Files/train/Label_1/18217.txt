#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll M = 1e9 + 7;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        string s; cin >> s;
        if (s.size() == 1 || *max_element(begin(s), end(s)) == '1') { cout << "ALICE\n"; continue; }
        bool rev = false;
        for (int i = 0; i < (int)s.size(); i++) {
            if (s[i] == '0') {
                if (!rev) cout << "BOB\n";
                else cout << "ALICE\n";
                break;
            }
            if (s[i] == '1') rev = !rev;
        }
    }
    return 0;
}
