#include <bits/stdc++.h>
using namespace std;

const int maxn = 3e5;
int n;
string o, i;

int main() {
    cin >> o >> i;
    vector<pair<char, int>> vo, vi;
    map<char, int> cnto, cnti;
    n = o.size();
    for (int j = 0; j < n; j++) {
        cnto[o[j]]++;
        vo.push_back({o[j], cnto[o[j]]});

        cnti[i[j]]++;
        vi.push_back({i[j], cnti[i[j]]});
    }
    sort(vo.begin(), vo.end());
    sort(vi.rbegin(), vi.rend());
    string ans;
    for (int j = 0; j < n; j++) {
        if (vo[j].second < vi[j].second) ans += vo[j].first;
        else ans += vi[j].first;
    }
    cout << ans << '\n';
}
