n, m, k, s = map(int, input().split())
notes_positions = {i: [] for i in range(1, k + 1)}
for i in range(n):
    row = list(map(int, input().split()))
    for j, note in enumerate(row):
        notes_positions[note].append((i + 1, j + 1))

song = list(map(int, input().split()))

max_complexity = 0
for i in range(1, s):
    current_note_positions = notes_positions[song[i - 1]]
    next_note_positions = notes_positions[song[i]]
    current_max = 0
    for current in current_note_positions:
        for next_pos in next_note_positions:
            current_max = max(current_max, abs(current[0] - next_pos[0]) + abs(current[1] - next_pos[1]))
    max_complexity = max(max_complexity, current_max)

print(max_complexity)
