#include <bits/stdc++.h>
using namespace std;
int main() {
    int k;
    cin >> k;
    vector<int> a(k), b(k);
    string s, t;
    cin >> s >> t;
    for (int i = 0; i < k; ++i) {
        a[i] = s[i] - 'a';
        b[i] = t[i] - 'a';
    }
    vector<int> c(k);
    for (int i = 0; i < k; ++i) {
        if (a[i] == b[i])
            c[i] = a[i];
        else
            c[i] = (a[i] + b[i] + 1) / 2;
    }
    for (int i = 0; i < k; ++i) { cout << (char)(c[i] + 'a'); }
}
