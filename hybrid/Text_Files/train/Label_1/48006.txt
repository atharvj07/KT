import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static char[][] map;
    static List<Character> colores = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        map = new char[m][n];
        for (int i = 0; i < m; i++) {
            map[i] = sc.next().toCharArray();
            for (char c : map[i]) {
                if (!colores.contains(c) && c != '.') {
                    colores.add(c);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (map[i][j] == '.') {
                    int cnt = 0;
                    if (i - 1 >= 0 && map[i - 1][j] != '.') {
                        cnt++;
                    }
                    if (i + 1 < m && map[i + 1][j] != '.') {
                        cnt++;
                    }
                    if (j - 1 >= 0 && map[i][j - 1] != '.') {
                        cnt++;
                    }
                    if (j + 1 < n && map[i][j + 1] != '.') {
                        cnt++;
                    }
                    ans += cnt / 2;
                }
            }
        }
        System.out.println(ans);
    }
}
