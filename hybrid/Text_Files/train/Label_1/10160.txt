#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    cout << "PRINT " << s[k - 1] << endl;
    if (k > 1)
        cout << "LEFT" << endl;
    for (ll i = k - 2; i >= 0; i--) {
        if (s[i] != s[i + 1]) {
            cout << "PRINT " << s[i] << endl;
            cout << "LEFT" << endl;
        }
    }
    for (ll i = k; i < n; i++) {
        if (s[i] != s[i - 1]) {
            cout << "RIGHT" << endl;
            cout << "PRINT " << s[i] << endl;
        }
    }
}
