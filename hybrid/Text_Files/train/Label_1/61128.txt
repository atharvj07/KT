import sys

T = int(input())

for _ in range(T):
    n, k = map(int, input().split())
    if k > (n - 1) ** (n - 1):
        print(-1)
    else:
        a = list(range(n, 0, -1))
        k -= 1
        for i in range(n - 1):
            if k <= (n - 2 - i) ** (n - 1 - i):
                a[i], a[i + 1] = a[i + 1], a[i]
                k -= 1
            else:
                break
        print(' '.join(map(str, a)))
