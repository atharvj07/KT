#include<bits/stdc++.h>
using namespace std;
#define forall(i,n) for(int i=0;i<n;++i)
struct Segment{int l,r;};
int main() {
	int n;cin>>n;vector<Segment> seg(n);
	forall(i,n) cin>>seg[i].l>>seg[i].r;
	sort(seg.begin(),seg.end(),[](const Segment& a,const Segment& b){
		return a.l!=b.l?a.l<b.l:a.r<b.r;
	});
	for(int i=1;i<n;++i){
		if(seg[i].l<seg[0].l||seg[0].r<seg[i].r){
			cout<<"-1"<<endl;return 0;
		}
	}
	cout<<1<<endl;
	return 0;
}
