#include <iostream>

int main() {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (i > 0 && (s[i] - '0') % 2 == 0) ans += i;
        if ((s[i] - '0') % 2 == 0) ans++;
    }
    std::cout << ans << std::endl;
    return 0;
}
