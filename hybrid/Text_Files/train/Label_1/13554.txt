#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int maxn=100000+5;
ll n,a[maxn],s,f,ans,mx;
int main() {
    cin>>n;
    for(int i=1; i<=n; ++i) cin>>a[i];
    cin>>s>>f;
    for(int i=s; i<=f;++i) {
        ll sum=0;
        for(int j=i; j<=n; j+=n) {
            sum+=a[j];
        }
        if(sum>mx) {
            mx=sum;
            ans=i;
        }
         for(int j=1; j<i;++j) {
            sum+=a[j];
        }
        if(sum>mx) {
            mx=sum;
            ans=i;
        }
    }
    cout<<(ans%n? ans%n: n);
    return 0;
}
