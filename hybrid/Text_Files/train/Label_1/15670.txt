#include <iostream>
using namespace std;
int main() {
    int up, down;
    cin >> down >> up;
    int n = up + down + 1;
    if (down == 0)
        cout << n << " ";
    else
        cout << 1 << " ";
    for (int i = 2; i <= n; i++) {
        if (up && down) {
            cout << i + 1 << " " << i << " ";
            up--;
            down--;
        } else if (up == 0) {
            cout << i << " ";
            down--;
        } else if (down == 0) {
            cout << i + 1 << " ";
            up--;
        }
    }
}
