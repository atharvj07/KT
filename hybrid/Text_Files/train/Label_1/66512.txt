t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    
    left = [False] * n
    right = [False] * n
    
    # Check from left to right
    min_required = 0
    for i in range(n):
        if a[i] >= min_required:
            left[i] = True
        else:
            break
        min_required += 1
    
    # Check from right to left
    min_required = 0
    for i in range(n - 1, -1, -1):
        if a[i] >= min_required:
            right[i] = True
        else:
            break
        min_required += 1

    # Check if there is an overlap
    answer = "No"
    for i in range(n):
        if left[i] and right[i]:
            answer = "Yes"
            break
    
    print(answer)
