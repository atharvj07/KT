#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll mod = 1e9+7;
int dp[1000001][11];
int go(int m,int last){
    if(m==0){
        return 1;
    }
    int &ret=dp[m][last];
    if(ret!=-1)return ret;
    ret=0;
    if(last==0){
        ret=(ret*2)%mod;
        ret=(ret+go(m-1,0))%mod;
        ret=(ret+go(m-1,1))%mod;
    }
    else{
        ret=(ret+go(m-1,0))%mod;
        ret=(ret+go(m-1,1))%mod;
    }
    return ret;
}
int main(){
    int n,m;
    cin>>n>>m;
    memset(dp,-1,sizeof(dp));
    cout<<go(m,0)<<endl;
    return 0;
}
