#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    int a[k1 + 1], b[k2 + 1];
    for (int i = 0; i < k1; i++) cin >> a[i];
    sort(a, a + k1, greater<int>());
    for (int i = 0; i < k2; i++) cin >> b[i];
    sort(b, b + k2, greater<int>());
    if (a[0] > b[0])
      cout << "YES";
    else
      cout << "NO";
    cout << endl;
  }
  return 0;
}
