import java.util.Scanner;
import java.util.HashMap;
import java.util.ArrayList;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        HashMap<Integer,Player> mapA = new HashMap<>();
        HashMap<Integer,Player> mapB = new HashMap<>();
        int maxA = 0;
        int maxB = 0;
        double maxDistA = 0;
        double maxDistB = 0;
        for(int i = 0; i < n; i++){
            int f = sc.nextInt();
            int a = sc.nextInt();
            int t = sc.nextInt();
            int x = sc.nextInt();
            int y = sc.nextInt();
            Player p = new Player(x, y);
            if(t == 0){
                mapA.put(f, p);
                if(maxA < a) maxA = a;
            }else{
                mapB.put(f, p);
                if(maxB < a) maxB = a;
            }
        }
        ArrayList<Player> listA = new ArrayList<>();
        ArrayList<Player> listB = new ArrayList<>();
        for(int i = 1; i <= maxA; i++){
            if(mapA.containsKey(0) && mapA.get(0).x == mapA.get(0).y){
                listA.add(mapA.get(0));
                listA.add(mapA.get(0));
            }else if(mapA.containsKey(0) && mapA.containsKey(10)){
                listA.add(mapA.get(0));
                listA.add(mapA.get(10));
            }else if(mapA.containsKey(0) && mapA.containsKey(30)){
                listA.add(mapA.get(0));
                listA.add(mapA.get(30));
            }else if(mapA.containsKey(10) && mapA.containsKey(30)){
                listA.add(mapA.get(10));
                listA.add(mapA.get(30));
            }else if(mapA.containsKey(10) && mapA.containsKey(60)){
                listA.add(mapA.get(10));
                listA.add(mapA.get(60));
            }else if(mapA.containsKey(60) && mapA.containsKey(90)){
                listA.add(mapA.get(60));
                listA.add(mapA.get(90));
            }
        }
        for(int i = 1; i <= maxB; i++){
            if(mapB.containsKey(0) && mapB.get(0).x == mapB.get(0).y){
                listB.add(mapB.get(0));
                listB.add(mapB.get(0));
            }else if(mapB.containsKey(0) && mapB.containsKey(10)){
                listB.add(mapB.get(0));
                listB.add(mapB.get(10));
            }else if(mapB.containsKey(0) && mapB.containsKey(30)){
                listB.add(mapB.get(0));
                listB.add(mapB.get(30));
            }else if(mapB.containsKey(10) && mapB.containsKey(30)){
                listB.add(mapB.get(10));
                listB.add(mapB.get(30));
            }else if(mapB.containsKey(10) && mapB.containsKey(60)){
                listB.add(mapB.get(10));
                listB.add(mapB.get(60));
            }else if(mapB.containsKey(60) && mapB.containsKey(90)){
                listB.add(mapB.get(60));
                listB.add(mapB.get(90));
            }
        }
        for(int i = 1; i < listA.size(); i++){
            double dist = Math.sqrt(Math.pow(listA.get(i).x - listA.get(i-1).x, 2) + Math.pow(listA.get(i).y - listA.get(i-1).y, 2));
            if(dist > maxDistA){
                maxDistA = dist;
            }
        }
        for(int i = 1; i < listB.size(); i++){
            double dist = Math.sqrt(Math.pow(listB.get(i).x - listB.get(i-1).x, 2) + Math.pow(listB.get(i).y - listB.get(i-1).y, 2));
            if(dist > maxDistB){
                maxDistB = dist;
            }
        }
        if(maxDistA == 0) System.out.println(-1 + " " + -1);
        else System.out.println(maxDistA + " " + 1.0);
        if(maxDistB == 0) System.out.println(-1 + " " + -1);
        else System.out.println(maxDistB + " " + 1.0);
    }
}
class Player{
    int x;
    int y;
    Player(int x, int y){
        this.x = x;
        this.y = y;
    }
}
