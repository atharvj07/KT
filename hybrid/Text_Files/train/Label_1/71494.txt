import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, s = map(int, input().split())
    arr = []
    for i in range(n):
        l, r = map(int, input().split())
        arr.append((l, r))

    arr.sort(key=lambda x:x[1])
    ans = 0
    for l, r in arr:
        if s - r >= ans:
            s -= r
            ans = r

    print(ans)
