#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
  ll n;
  cin >> n;
  set<ll> a;
  vector<ll> b;
  for (ll i = 0; i < n; ++i) {
    ll x;
    cin >> x;
    a.insert(x);
    b.push_back(x);
  }
  sort(b.begin(), b.end());
  ll ans = 0;
  for (ll i = 0; i < n; ++i) {
    if (a.find(b[i] + 1) != a.end()) {
      ++ans;
    } else if (a.find(b[i] - 1) != a.end()) {
      ++ans;
    }
  }
  cout << ans << endl;
  return 0;
}
