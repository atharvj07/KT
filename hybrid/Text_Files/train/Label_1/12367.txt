#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;

  while (t--) {
    int n;
    std::cin >> n;

    std::vector<int> weights(n);

    for (int i = 0; i < n; ++i) {
      std::cin >> weights[i];
    }

    bool is_possible = true;

    int total_weight = 0;
    for (int weight : weights) {
      total_weight += weight;
    }

    if (total_weight % 2 != 0) {
      is_possible = false;
    } else {
      int target_weight = total_weight / 2;

      std::vector<std::vector<bool>> dp(n + 1, std::vector<bool>(target_weight + 1));
      dp[0][0] = true;

      for (int i = 1; i <= n; ++i) {
        for (int weight = 0; weight <= target_weight; ++weight) {
          dp[i][weight] = dp[i - 1][weight] || (weight >= weights[i - 1] && dp[i - 1][weight - weights[i - 1]]);
        }
      }

      if (!dp[n][target_weight]) {
        is_possible = false;
      }
    }

    std::cout << (is_possible ? "YES" : "NO") << std::endl;
  }

  return 0;
}
