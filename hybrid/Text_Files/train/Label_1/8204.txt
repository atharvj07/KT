#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
const ll INF=1e18;
vector<int>vec[100005];
int mark[100005];
int solve(ll n, ll m, ll k, vector<ll>&p)
{
	for(ll i=0;i<p.size();i++) vec[p[i]/k].push_back(p[i]);
	for(ll i=0;i<n/k;i++)
	{
		for(ll j=i+1;j<n/k;j++)
		{
			if(vec[i].size())
			{
				for(ll k=0;k<vec[j].size();k++)
					vec[i].push_back(vec[j][k]);
				vec[j].clear();
			}
		}
	}
	ll cnt=0;
	for(ll i=0;i<n/k;i++)
	{
		if(vec[i].size()) cnt++;
	}
	return cnt;
}
int main()
{
    ll n,m,k;
    vector<ll>p;
    cin>>n>>m>>k;
    for(ll i=0;i<m;i++)
    {
    	ll x;
    	cin>>x;
    	p.push_back(x);
    }
    ll ans=solve(n,m,k,p);
    cout<<ans;
    return 0;
}
