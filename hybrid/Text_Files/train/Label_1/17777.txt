#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll A, B, n, l, t, m;
int main() {
    cin >> A >> B >> n;
    while (n--) {
        cin >> l >> t >> m;
        if (A > t || (t - A) % B > m)
            cout << -1 << endl;
        else
            cout << ((t - A) / B) + l << endl;
    }
    return 0;
}
