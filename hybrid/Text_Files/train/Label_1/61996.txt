def solve_problem(s, t):
    if sorted(s) == sorted(t):
        return "array"
    for char in t:
        if t.count(char) > s.count(char):
            return "need tree"
    if len(s) == len(t):
        return "array"
    i = 0
    for char in s:
        if i < len(t) and char == t[i]:
            i += 1
    if i == len(t):
        return "automaton"
    return "both"

s = input()
t = input()
print(solve_problem(s, t))
