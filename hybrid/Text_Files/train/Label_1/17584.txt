#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c;
  cin >> n >> c;
  vector<long long> a(n);
  for (auto &i : a) cin >> i;
  int cnt = 0;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < i && a[i] - a[j] > c) j++;
    cnt += (i - j + 1);
  }
  cout << cnt << '\n';
  return 0;
}
