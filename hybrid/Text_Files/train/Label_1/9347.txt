#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> t(n);
  for (int i = 0; i < n; i++) cin >> t[i];
  int ans = 0;
  bool onWinter = false;
  for (int i = 0; i < n; i++) {
    if (t[i] < 0) {
      if (!onWinter) ans++, onWinter = true;
      k--;
    }
    if (k == 0) ans++, k = onWinter ? k + 1 : k;
  }
  if (k > 0) ans++;
  cout << (ans > 1e9 ? -1 : ans) << endl;
}
