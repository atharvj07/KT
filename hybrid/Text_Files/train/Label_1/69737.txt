for _ in range(int(input())):
    # Read input
    n = int(input())
    jams = list(map(int, input().split()))
    
    # Count the number of strawberry and blueberry jams
    strawberry = jams.count(1)
    blueberry = jams.count(2)
    
    # Print the minimum number of jars to eat
    print(abs(strawberry - blueberry))
