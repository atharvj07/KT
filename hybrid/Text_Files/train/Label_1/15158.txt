#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, p, s;
  std::cin >> n >> p >> s;
  std::vector<int> a(n), b(n), ans1, ans2;
  for (auto& x : a) std::cin >> x;
  for (auto& x : b) std::cin >> x;
  std::vector<std::pair<int, int>> all(n);
  for (int i = 0; i < n; i++) all[i] = {a[i] + b[i], i};
  std::sort(all.rbegin(), all.rend());
  for (int i = 0; i < p; i++) ans1.push_back(all[i].second + 1);
  for (int i = p; i < p + s; i++) ans2.push_back(all[i].second + 1);
  std::cout << all[0].first << "\n";
  for (auto& x : ans1) std::cout << x << " ";
  std::cout << "\n";
  for (auto& x : ans2) std::cout << x << " ";
  std::cout << "\n";
  return 0;
}
