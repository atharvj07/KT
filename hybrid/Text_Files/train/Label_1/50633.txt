import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = n * h * w - 2 * ((n / 2) * h * w);
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                ans -= a[i] * h;
            } else {
                ans += a[i] * h;
            }
        }
        System.out.println(ans);
    }
}
