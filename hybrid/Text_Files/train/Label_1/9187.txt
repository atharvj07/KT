#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n == 2) {
      cout << 1 << '\n';
    } else if (n == 3) {
      cout << 7 << '\n';
    } else if (n == 4) {
      cout << 11 << '\n';
    } else {
      cout << n + 3 << '\n';
    }
  }

  return 0;
}
