import java.util.Scanner;

public class PerfectMatching {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        int[][] probabilities = new int[n][n];
        for (int i = 0; i < n; i++) {
            String[] line = scanner.nextLine().split(" ");
            for (int j = 0; j < n; j++) {
                probabilities[i][j] = Integer.parseInt(line[j]);
            }
        }

        long numerator = calculateProbability(probabilities);
        long denominator = calculateInverse(n);

        long result = (numerator * denominator) % MOD;
        System.out.println(result);
    }

    private static long calculateProbability(int[][] probabilities) {
        int n = probabilities.length;
        int[][] dp = new int[1 << n][n + 1];
        dp[0][0] = 1;

        for (int mask = 0; mask < (1 << n); mask++) {
            for (int right = 0; right <= n; right++) {
                if (dp[mask][right] == 0) {
                    continue;
                }

                for (int left = 0; left < n; left++) {
                    if ((mask & (1 << left)) > 0 || probabilities[left][right] == 0) {
                        continue;
                    }

                    int newMask = mask | (1 << left);
                    dp[newMask][right + 1] = (dp[newMask][right + 1] + dp[mask][right]) % MOD;
                }
            }
        }

        return dp[(1 << n) - 1][n];
    }

    private static long calculateInverse(int n) {
        return power(n, MOD - 2, MOD);
    }

    private static long power(long x, long y, long mod) {
        long result = 1;
        while (y > 0) {
            if ((y & 1) > 0) {
                result = (result * x) % mod;
            }
            x = (x * x) % mod;
            y >>= 1;
        }
        return result;
    }
}
