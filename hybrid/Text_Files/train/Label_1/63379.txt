from collections import defaultdict

def dijkstra(dist, prev, s, n):
    vis = [False] * (n + 1)
    for i in range(1, n + 1):
        dist[i] = 1e9
    dist[s] = 0
    for _ in range(n):
        idx = -1
        dist_min = 1e9
        for i in range(1, n + 1):
            if not vis[i] and dist_min > dist[i]:
                dist_min = dist[i]
                idx = i
        vis[idx] = True
        for u in graph[idx]:
            if not vis[u]:
                new_dist = dist[idx] + (1 ^ edges[(idx, u)])
                if new_dist < dist[u]:
                    dist[u] = new_dist
                    prev[u] = idx

n, m = map(int, input().split())
graph = defaultdict(list)
edges = {}
for _ in range(m):
    u, v, z = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)
    edges[(u, v)] = edges[(v, u)] = z

dist1 = [0] * (n + 1)
prev1 = [0] * (n + 1)
dijkstra(dist1, prev1, 1, n)

dist2 = [0] * (n + 1)
prev2 = [0] * (n + 1)
dijkstra(dist2, prev2, n, n)

path = []
idx = n
while idx != 1:
    path.append(idx)
    idx = prev1[idx]

path.reverse()
res = []
for i in range(len(path) - 1):
    u = path[i]
    v = path[i + 1]
    if (u, v) in edges:
        if edges[(u, v)]:
            res.append((u, v, 0))
        else:
            res.append((u, v, 1))
    else:
        if edges[(v, u)]:
            res.append((v, u, 0))
        else:
            res.append((v, u, 1))

print(len(res))
for u, v, z in res:
    print(u, v, z)
