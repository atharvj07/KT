import java.util.*;

public class Main {

  static long countMagicalSubarrays(int[] arr) {
    long cnt = 0;
    int n = arr.length;
    int min = arr[0], max = arr[0];

    for (int l = 0, r = 0; r < n; r++) {
      min = Math.min(min, arr[r]);
      max = Math.max(max, arr[r]);

      if (min == max) {
        cnt += r - l + 1;
      } else {
        l = r;
        min = arr[l];
        max = arr[l];
      }
    }

    return cnt;
  }

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = input.nextInt();
    }
    System.out.println(countMagicalSubarrays(arr));
  }
}
