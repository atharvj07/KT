#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5;
int n, l[maxn], d[maxn];

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> l[i] >> d[i];
  sort(l, l + n, greater<int>());
  sort(d, d + n, greater<int>());
  int ans = 0, cnt = 0, j = 0;
  for (int i = 0; i < n; i++) {
    if (i + 1 > n / 2) {
      break;
    }
    if (l[j] == l[i]) {
      cnt++;
    }
    if (cnt * 2 > i + 1) {
      break;
    }
    ans += d[j++];
  }
  cout << ans << endl;
}
