#include <algorithm>
#include <cstdio>
int main() {
  int t, n, d[200001], sum[200001], x;
  scanf("%d", &t);
  while (t--) {
    scanf("%d", &n);
    for (int i = 1; i <= n << 1; i++) scanf("%d", &d[i]), sum[i] = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n << 1; j++) {
        if (d[i] > d[j])
          x = d[i] - d[j];
        else
          x = d[j] - d[i];
        sum[i] += x;
        sum[j] += x;
      }
    }
    std::sort(sum + 1, sum + n * 2 + 1);
    if (sum[n] == sum[n << 1])
      puts("YES");
    else
      puts("NO");
  }
  return 0;
}
