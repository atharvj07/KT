#include <iostream>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    map<string, int> cnt;
    for (int i = 0; i + 1 < n; i++) {
        cnt[s.substr(i, 2)]++;
    }
    string ans;
    int mx = 0;
    for (auto p : cnt) {
        if (p.second > mx) {
            ans = p.first;
            mx = p.second;
        }
    }
    cout << ans << endl;
}
