import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class AlexGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        System.out.println(calculateMaxPoints(arr));
    }

    private static int calculateMaxPoints(int[] arr) {
        Map<Integer, Integer> countMap = new HashMap<>();
        for (int num : arr) {
            countMap.put(num, countMap.getOrDefault(num, 0) + 1);
        }

        int maxPoints = 0;
        for (int key : countMap.keySet()) {
            if (countMap.containsKey(key + 1) && countMap.containsKey(key - 1)) {
                maxPoints = Math.max(maxPoints, countMap.get(key) * 3);
            } else if (countMap.containsKey(key + 1) || countMap.containsKey(key - 1)) {
                maxPoints = Math.max(maxPoints, countMap.get(key) * 2);
            } else {
                maxPoints = Math.max(maxPoints, countMap.get(key));
            }
        }

        return maxPoints;
    }
}
