#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        vector<ll> l(n);
        for(ll &x:l) cin>>x;
        sort(l.begin(),l.end());
        if(l[0]==l[n-1]) cout<<n<<endl;
        else cout<<(upper_bound(l.begin(),l.end(),l[0])-l.begin())<<endl;
    }
    return 0;
}
