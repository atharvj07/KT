import math

n = int(input())
l = list(map(int, input().split()))
s = input()
stamina = 0
totalTime = 0
for seg, terrain in zip(l, s):
    if terrain == 'G':
        time = seg
        stamina += seg
    elif terrain == 'W':
        if stamina >= seg:
            stamina -= seg
            time = 3 * seg
        else:
            time = 3 * stamina
            stamina = 0
            cost = math.ceil((seg - stamina) / 2)
            time += 6 * cost
            stamina = seg - cost
    else:
        if stamina >= seg:
            stamina -= seg
            time = seg
        else:
            stamina = 0
            time = seg

    totalTime += time

print(totalTime)
