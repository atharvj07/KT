n, a, b = map(int, input().split())
seals = [tuple(map(int, input().split())) for _ in range(n)]
max_area = 0
for i in range(n):
    for j in range(n):
        if i != j:
            xi, yi = seals[i]
            xj, yj = seals[j]
            area = max(xi * yi, xj * yj)
            if min(xi, yi) <= a and max(xi, yi) <= b or min(xi, yi) <= b and max(xi, yi) <= a and min(xj, yj) <= a and max(xj, yj) <= b or min(xj, yj) <= b and max(xj, yj) <= a:
                max_area = max(max_area, area)
print(max_area)
