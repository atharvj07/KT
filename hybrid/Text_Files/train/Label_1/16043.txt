#include<bits/stdc++.h>
#define pb push_back
using namespace std;
vector<pair<int, int>> v;
int n, p[2005], s[2005], ans[2005], F;
void go(int x, int y) {
	if (x == y) return;
	if (ans[x] != ans[y]) {
		v.pb({ x, y });
		F--;
		go(ans[x], ans[y]);
	}
	go(x + 1, y + 1);
}
int32_t main() {
	cin >> n;
	for (int i = 1; i <= n; i++) {
		cin >> p[i];
		ans[p[i]] = i;
	}
	for (int i = 1; i <= n; i++) cin >> s[i];
	for (int i = 1; i <= n; i++) {
		if (p[i] != s[i]) F++;
		ans[s[i]] = ans[p[i]];
	}
	cout << F << endl;
	cout << (F - 1) / 2 << endl;
	go(1, 1);
	for (auto x : v) cout << x.first << " " << x.second << endl;
	return 0;
}
