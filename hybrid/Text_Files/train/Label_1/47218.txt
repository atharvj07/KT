import java.util.*;

class Main {
    static int[][] X;
    static int[] Y;
    static int[] P;
    static int n;

    static int find(int x) {
        if (P[x] == x)
            return x;
        return P[x] = find(P[x]);
    }

    static void union(int a, int b) {
        a = find(a);
        b = find(b);
        if (a != b)
            P[b] = a;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            n = sc.nextInt();
            int k = sc.nextInt();
            X = new int[n + 1][];
            Y = new int[n + 1];
            P = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                X[i] = new int[2];
                P[i] = i;
                X[i][0] = sc.nextInt();
                X[i][1] = sc.nextInt();
                Y[i] = X[i][0] + X[i][1];
            }
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j < i; j++) {
                    if (Math.abs(X[i][0] - X[j][0]) + Math.abs(X[i][1] - X[j][1]) <= k)
                        union(i, j);
                }
            }
            int c = 0;
            for (int i = 1; i <= n; i++)
                if (P[i] == i)
                    c++;
            System.out.println(c == 1 ? 1 : -1);
        }
    }
}
