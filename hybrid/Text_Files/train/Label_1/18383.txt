#include <bits/stdc++.h>
using namespace std;

const int MAX_N = 500, MAX_M = 500;
int n, m;
bool prime[1 << 16];
int matrix[MAX_N][MAX_M], rowmax[MAX_N], colmax[MAX_M];

void sieve() {
  memset(prime, 1, sizeof prime);
  prime[1] = false;
  for (int p = 2; p < 1 << 16; p++) {
    if (prime[p]) {
      for (int i = p << 1; i < 1 << 16; i += p) {
        prime[i] = false;
      }
    }
  }
}

int main() {
  sieve();
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> matrix[i][j];
      rowmax[i] = max(rowmax[i], matrix[i][j]);
      colmax[j] = max(colmax[j], matrix[i][j]);
    }
  }
  int ans = INT_MAX;
  for (int i = 0; i < n; i++) {
    if (prime[rowmax[i]]) {
      ans = 0;
      break;
    }
    ans = min(ans, prime[rowmax[i] + 1] - true);
  }
  for (int j = 0; j < m; j++) {
    if (prime[colmax[j]]) {
      ans = 0;
      break;
    }
    ans = min(ans, prime[colmax[j] + 1] - true);
  }
  cout << ans << endl;
  return 0;
}
