import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N = sc.nextInt();
		int M = sc.nextInt();
		int[] D = new int[N];
		for (int i = 0; i < N; i++) {
			D[i] = sc.nextInt();
		}
		int[] C = new int[M];
		for (int i = 0; i < M; i++) {
			C[i] = sc.nextInt();
		}

		int[][] dp = new int[M + 1][N + 1];
		for (int i = 0; i <= M; i++) {
			for (int j = 0; j <= N; j++) {
				dp[i][j] = Integer.MAX_VALUE;
			}
		}

		dp[0][0] = 0;

		for (int i = 0; i < M; i++) {
			for (int j = 0; j < N; j++) {
				if (dp[i][j] == Integer.MAX_VALUE) {
					continue;
				}
				// Move
				dp[i + 1][j + 1] = Math.min(dp[i + 1][j + 1], dp[i][j] + D[j] * C[i]);
				// Wait
				dp[i + 1][j] = Math.min(dp[i + 1][j], dp[i][j]);
			}
		}

		int ans = Integer.MAX_VALUE;
		for (int i = 0; i <= M; i++) {
			if (dp[i][N] != Integer.MAX_VALUE) {
				ans = Math.min(ans, dp[i][N]);
			}
		}

		System.out.println(ans);
	}
}
