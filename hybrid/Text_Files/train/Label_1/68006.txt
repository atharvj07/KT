import sys
input = sys.stdin.readline

T = int(input())
for t in range(T):
    n = int(input())
    a = list(map(int, input().split()))
    f = [0] * (n + 2)
    for i in a: f[i] += 1
    dom = -1
    for i in range(1, n + 1):
        if f[i] and (dom == -1 or f[dom] < f[i]): dom = i
    if dom == -1: print(-1)
    else:
        l = r = -1
        cnt = 0
        for i in range(1, n + 1):
            if a[i - 1] == dom:
                if cnt == 0: l = i - 1
                cnt += 1
            else:
                if cnt > 0:
                    if r == -1 or r - l > i - l: r = i - 1
                cnt = 0
        if cnt > 0 and (r == -1 or r - l > n - l): r = n
        print(r - l)
