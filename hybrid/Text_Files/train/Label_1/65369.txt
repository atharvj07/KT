import numpy as np

n = int(input())
a = np.zeros((2, n))
b = np.zeros((2, n))

for i in range(n):
    ai, bi, ci, di = map(int, input().split())
    a[:, i] = (ai, bi)
    b[:, i] = (ci, di)

x = np.hstack((a[0, :], b[0, :]))
y = np.hstack((a[1, :], b[1, :]))
s = np.argsort(x)
t = np.searchsorted(x, x, sorter=s) - 1

ans = 0
for i in range(1, n * 2):
    if y[s[i]] > y[s[t[i]]]: ans += y[s[i]] - y[s[t[i]]]

print(ans)
