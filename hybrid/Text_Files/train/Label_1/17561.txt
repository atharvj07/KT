#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<int, int>> p(n);
    for (int i = 0; i < n; i++) {
      int a, b;
      cin >> a >> b;
      p[i] = make_pair(a, b);
    }
    bool flag = true;
    for (int i = 1; i < n; i++) {
      if (p[i].second > p[i - 1].second && p[i].first <= p[i - 1].first) {
        flag = false;
        break;
      }
    }
    cout << (flag ? "YES" : "NO") << endl;
  }
  return 0;
}
