#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long int> a(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    long long int ans = 2 * n - 2, minv = LLONG_MAX;
    for (int i = 1; i < n; i++) { minv = min(minv, max(a[i], a[i + 1])); }
    ans = min(ans, minv + n - 1);
    cout << ans << endl;
    return 0;
}
