h1, a1, c1 = map(int, input().split())
h2, a2 = map(int, input().split())
if h1 <= a2:
    print(1)
    print('HEAL')
    exit()
n = (h2 + a2 - 1) // a1
if h1 + c1 - a2 * n <= 0:
    n += 1
print(n)
for i in range(n):
    if h1 > a2 * i:
        print('STRIKE')
    else:
        print('HEAL')
    h1 = min(h1 + c1, h1 + a1 - a2)
