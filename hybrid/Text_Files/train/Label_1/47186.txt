import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int n = input.nextInt(), x = input.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) {
                a[i] = input.nextInt();
            }
            long min = 0, max = 0;
            for (int i = 0; i < n; i++) {
                min += (a[i] + x - 1) / x;
                max += (a[i] - 1) / x + 1;
            }
            System.out.println(min + " " + max);
        }
    }
}
