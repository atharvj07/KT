n = int(input())
a = input()
b = input()

def count_swaps_to_change_bitwise_or(n, a, b):
    ones_in_a = a.count('1')
    zeros_in_a = a.count('0')
    swaps = 0
    for i in range(n):
        if a[i] == '1' and b[i] == '0':
            swaps += zeros_in_a
        elif a[i] == '0' and b[i] == '0':
            swaps += ones_in_a
    return swaps

print(count_swaps_to_change_bitwise_or(n, a, b) // 2)
