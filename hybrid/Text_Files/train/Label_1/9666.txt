#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

ll solve(vector<pair<ll, ll>> &v, ll sum) {
  ll n = v.size();
  sort(v.begin(), v.end());
  ll need = sum;
  ll cur = 0;
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    if (v[i].second <= cur) {
      ans += v[i].first;
      need -= v[i].first;
      continue;
    } else if (cur + 1 >= v[i].second) {
      ans += v[i].first;
      cur++;
      need -= v[i].first;
      continue;
    }
    ll x = (v[i].second - cur - 1 + need) / v[i].second * v[i].second;
    ans += x;
    cur += x / v[i].second;
    need -= x;
  }
  return ans;
}

int main() {
  ll n;
  cin >> n;
  vector<pair<ll, ll>> v(n);
  ll sum = 0;
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
    sum += v[i].first;
  }
  cout << solve(v, sum) << endl;
}
