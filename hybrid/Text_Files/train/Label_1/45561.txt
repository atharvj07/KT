import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong(), b = in.nextLong(), f = in.nextLong(), k = in.nextLong();
        System.out.println(2 * ((a + f - 1) / b) * k + (a + f - 1) / b + (k - 1));
    }
}
