#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    string s;
    cin >> s;
    int n = s.length();
    vector<int> dp(n + 1, 1e9);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
      if (s[i - 1] == '1') {
        if (i > 1) dp[i] = min(dp[i], dp[i - 1] + a);
        if (i > 2) dp[i] = min(dp[i], dp[i - 2] + b + a);
      } else {
        dp[i] = min(dp[i], dp[i - 1]);
        if (i > 1) dp[i] = min(dp[i], dp[i - 2] + b);
      }
    }
    cout << dp[n] << endl;
  }
}
