#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int A, B, n;
    cin >> A >> B >> n;
    long long sum = B;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
    }
    for (int i = 0; i < n; i++) {
      cin >> b[i];
      sum -= b[i];
    }
    (sum <= A) ? cout << "YES\n" : cout << "NO\n";
  }
}
