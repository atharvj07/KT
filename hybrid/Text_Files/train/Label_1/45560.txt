import java.util.Scanner;
import java.util.LinkedList;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int a = scan.nextInt();
        int b = scan.nextInt();
        int f = scan.nextInt();
        int k = scan.nextInt();

        LinkedList<Integer> tank = new LinkedList<>(Arrays.asList(0, a, a));
        int n = 0;

        while (n + 1 < k) {
            int i = tank.poll();
            if (i >= f) {
                tank.addAll(Arrays.asList(f, f + b, f + b));
                n++;
            } else {
                tank.addFirst(i);
                break;
            }
        }
        System.out.print(n + 1 < k ? -1 : n);
    }
}
