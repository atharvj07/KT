#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    int t; cin >> t;
    while (t--) {
        int n, W; cin >> n >> W;
        vector<ll> w(n);
        for (int i = 0; i < n; i++) cin >> w[i];
        sort(w.begin(), w.end());
        ll have = 0;
        int cnt = 0;
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            if (have + w[i] >= (W + 1) / 2 && have + w[i] <= W) {
                have += w[i];
                cnt++;
                ans.push_back(i + 1);
            }
        }
        if (have >= (W + 1) / 2) {
            cout << cnt << endl;
            cout << ans[0];
            for (int i = 1; i < cnt; i++) cout << ' ' << ans[i];
            cout << endl;
        } else {
            cout << -1 << endl;
        }
    }
}
