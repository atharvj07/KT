#include <bits/stdc++.h>

const int MAXN = 200001;
const int MAXK = 21;
int n;
char sign[MAXN], ans[MAXN];
int x[MAXN], l[MAXN], r[MAXN];
int f[MAXK][MAXN], c[MAXK][MAXN];

int main() {
    for (int k = 0; k < MAXK; ++k) for (int b = 0; b < MAXN; ++b) {
        f[k][b] = 1 << 30;
        c[k][b] = 0;
    }
    scanf("%d", &n);
    for (int i = 0; i < n; ++i) {
        getchar();
        scanf("%c %d %c", sign + i, x + i, ans + i);
        l[i] = x[i];
        r[i] = x[i];
        if (sign[i] == '>' || sign[i] == '<') {
            r[i]--;
            l[i]++;
        }
    }
    for (int i = n - 1; i >= 0; --i) for (int k = 0; k < MAXK; ++k)
        for (int b = 0; b < MAXN; ++b) {
            if (b > r[i]) continue;
            if (b == l[i] && sign[i] == '>') {
                f[k][b] = std::min(f[k][b], f[k + 1][b] + 1);
                c[k][b] = 1;
            }
            if (b == l[i] && sign[i] == '<') {
                f[k][b] = std::min(f[k][b], f[k][b + 1] + 1);
                c[k][b] = -1;
            }
            if (b >= l[i] && sign[i] == '>') {
                f[k][b] = std::min(f[k][b], f[k][b + 1] + 1);
                c[k][b] = -1;
            }
            if (b >= l[i] && sign[i] == '<') {
                f[k][b] = std::min(f[k][b], f[k + 1][b] + 1);
                c[k][b] = 1;
            }
        }
    if (f[0][0] >= 1 << 30) {
        puts("Impossible");
        return 0;
    }
    int ans = 0;
    int b = 0;
    for (int k = 0; k < f[0][0]; ++k) {
        ans += c[k][b];
        b += c[k][b];
    }
    printf("%d\n", ans);
    return 0;
}
