n=int(input())
a=input()
f=[int(x) for x in input().split()]
l=[int(x) for x in a]
for i in range(0,n):
    for j in range(i,n):
        if a[i]!=a[j]:
            if f[int(a[i])]>=f[int(a[j])]:
                for k in range(i,j+1):
                    l[k]=f[int(l[k])]
            break
ans=[str(x) for x in l]
print(''.join(ans))
