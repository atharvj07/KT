import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[input.nextInt()] = i;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] != i) {
                System.out.println(1);
                System.out.println(a[i] + " " + i);
                a[a[i]] = a[i];
                a[i] = i;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
