import sys
input = sys.stdin.readline
inf = float('inf')

for _ in range(int(input())):
    n = int(input())
    lis = []
    for _ in range(n):
        a, b = map(int, input().split())
        lis.append((a, b))
    ans = inf
    for i in range(n):
        for j in range(i+1, n):
            if lis[i][0] == lis[j][0]:
                ans = min(ans, lis[i][1] + lis[j][1])
    if ans == inf: ans = 0
    print(ans)
