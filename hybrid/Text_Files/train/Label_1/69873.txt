import sys
from collections import Counter
n=int(input())
l=list(map(int,input().split()))
c=Counter(l)
while l:
    m=min(c)
    for i in range(len(l)):
        if  l[i]==m:
            if i+1<len(l) and l[i+1]==m:
                l[i]=2*m
                del l[i+1]
                break
            else:
                del l[i]
                break
    c=Counter(l)
    if len(set(l))==len(l):break
print(len(l))
print(*l)
