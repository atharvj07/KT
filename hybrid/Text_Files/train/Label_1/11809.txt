#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            std::cin >> a[i];
        }
        std::sort(a.begin(), a.end());
        int ans = 0;
        for (int i = 0; i < n;) {
            int j = i;
            while (j < n && a[j] == a[i]) {
                ++j;
            }
            ans = std::max(ans, j - i);
            i = j;
        }
        std::cout << ans << "\n";
    }
    return 0;
}
