#include<bits/stdc++.h>
#define ll long long
#define endl '\n'
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        stack<ll>s;
        int c=0;
        for(int i=0;i<n;i++){
            ll x;
            cin>>x;
            while(!s.empty() && s.top()>=x){
                s.pop();
                c++;
            }
            s.push(x);
        }
        cout<<c<<endl;
    }
    return 0;
}
