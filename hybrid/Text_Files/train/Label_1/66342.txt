k=int(input())
d=[int(i) for i in input().split()]
d.sort()
for i in range(0,k):
    for j in range(i+1,k):
        f=False
        x=d[i]
        y=d[j]
        while x!=0 or y!=0:
            xx=x%10
            yy=y%10
            if xx==yy==0:
                f=True
                break
            x//=10
            y//=10
        if not f:
            d.pop(j)
            k-=1
            j-=1
print(k)
for i in d:
    print(i,end=" ")
