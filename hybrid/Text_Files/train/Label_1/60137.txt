def check(a, k, x, n):
    count = [0] * (n + 1)
    for i in range(n):
        if a[i] >= x:
            count[i + 1] = count[i] + 1
        else:
            count[i + 1] = count[i] - 1

    min_val = 0
    for i in range(k, n + 1):
        if count[i] - min_val > 0:
            return True
        min_val = min(min_val, count[i - k + 1])
    return False

def find_max_median(a, n, k):
    left = 1
    right = n
    while left < right:
        mid = (left + right + 1) // 2
        if check(a, k, mid, n):
            left = mid
        else:
            right = mid - 1
    return left

n, k = map(int, input().split())
a = list(map(int, input().split()))
print(find_max_median(a, n, k))
