#include <iostream>

using namespace std;

int main() {
  int a, b;
  cin >> a >> b;
  int total = 0;
  int burned = 0;
  while (a > 0 || burned >= b) {
    if (a > 0) {
      a--;
      total++;
    } else {
      burned -= b;
      a++;
      total++;
    }
    burned++;
  }
  cout << total << endl;
  return 0;
}
