#include <iostream>
#include <cstdio>
typedef long long ll;

using namespace std;

const int MAXN = 1e6 + 10;

int main() {
  ll n, m, x, y, z, p;
  cin >> n >> m >> x >> y >> z >> p;
  for (int i = 1; i <= p; ++i) {
    ll a, b;
    cin >> a >> b;
    ll nx = z % 4 == 1 ? b : (z % 4 == 2 ? n - a + 1 : (z % 4 == 3 ? m - b + 1 : a));
    ll ny = z % 4 == 1 ? n - a + 1 : (z % 4 == 2 ? b : (z % 4 == 3 ? a: m - b + 1));
    ny = y % 2 ? m - ny + 1 : ny;
    nx = x % 2 ? n - nx + 1 : nx;
    cout << nx << " " << ny << endl;
  }
  return 0;
}
