#include <bits/stdc++.h>

int main() {
  std::string n;
  std::cin >> n;
  if (n.back() % 5 == 0) return std::cout << 0 << std::endl, 0;
  int l = -1, r = -1;
  for (int i = 0; i < n.size(); ++i) {
    if (n[i] == '0') r = i;
    if (n[i] % 2 == 0) l = i;
  }
  if (l != -1) {
    if (r == -1) r = n.size() - 1;
    std::swap(n[l], n[r]);
    return std::cout << 1 << std::endl, 0;
  }
  return std::cout << -1 << std::endl, 0;
}
