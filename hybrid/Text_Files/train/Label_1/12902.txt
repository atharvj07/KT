#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
const int mod = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    int xmax = 0, ymax = 0, xmin = INT_MAX, ymin = INT_MAX;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        xmax = max(xmax, x);
        ymax = max(ymax, y);
        xmin = min(xmin, x);
        ymin = min(ymin, y);
    }
    cout << max({xmax, ymax, ymax - ymin + 1, xmax - xmin + 1}) << endl;
    return 0;
}
