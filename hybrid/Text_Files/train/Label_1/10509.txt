#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string g[n];
  for (int i = 0; i < n; i++) cin >> g[i];

  vector<pair<int, int>> h, v;
  vector<int> p;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      if (g[i][j] == '*') {
        if (j != 0 && g[i][j - 1] == '*' && i != 0 && g[i - 1][j] == '*')
          continue;
        if (j != 0 && g[i][j - 1] == '*')
          h.emplace_back(i, j);
        else if (i != 0 && g[i - 1][j] == '*')
          v.emplace_back(i, j);
        else
          p.emplace_back(i * m + j);
      }
    }
  if (h.size() + v.size() + p.size() > n * m) {
    cout << -1 << endl;
    return 0;
  }
  cout << h.size() + v.size() + p.size() << endl;
  for (auto& e : h) cout << e.first + 1 << " " << e.second + 1 << " 1" << endl;
  for (auto& e : v) cout << e.first + 1 << " " << e.second + 1 << " 2" << endl;
  for (auto& e : p)
    cout << e / m + 1 << " " << e % m + 1 << " 1" << endl;
  return 0;
}
