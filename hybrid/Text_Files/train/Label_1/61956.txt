import sys

n = int(input())
s = input()
x, y = map(int, input().split())

a = [0] * (n + 1)
b = [0] * (n + 1)
for i in range(n):
    if s[i] == 'U':
        a[i + 1] = a[i]
        b[i + 1] = b[i] + 1
    elif s[i] == 'D':
        a[i + 1] = a[i]
        b[i + 1] = b[i] - 1
    elif s[i] == 'L':
        a[i + 1] = a[i] - 1
        b[i + 1] = b[i]
    else:
        a[i + 1] = a[i] + 1
        b[i + 1] = b[i]

if a[n] != x or b[n] != y:
    print(-1)
    sys.exit()

ans = float('inf')
for i in range(n):
    for j in range(i + 1, n + 1):
        if abs(a[i] - x) + abs(b[i] - y) + abs(a[j] - a[i]) + abs(b[j] - b[i]) + abs(x - a[j]) + abs(y - b[j]) < ans:
            ans = abs(a[i] - x) + abs(b[i] - y) + abs(a[j] - a[i]) + abs(b[j] - b[i]) + abs(x - a[j]) + abs(y - b[j])

print(ans)
