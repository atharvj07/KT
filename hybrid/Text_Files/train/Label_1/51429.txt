import java.util.Scanner;
public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int t = in.nextInt();
    for (int i = 0; i < t; i++) {
      long n = in.nextLong();
      long k = in.nextLong();
      System.out.println((n * (k - 1) / 2) + (k % 2 == 1 ? 1 : n / 2 + 1));
    }
  }
}
