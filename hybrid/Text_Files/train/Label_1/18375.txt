#include <iostream>

int main() {
  int n;
  std::cin >> n;
  int ans = 0;
  for (int i = 4; i <= n; i++) {
    bool ok = true;
    int x = i;
    while (x > 0) {
      if (x % 10 != 4 && x % 10 != 7) {
        ok = false;
        break;
      }
      x /= 10;
    }
    if (ok) ans++;
  }
  std::cout << ans << '\n';
  return 0;
}
