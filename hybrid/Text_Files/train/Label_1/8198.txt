#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<n;++i)
using namespace std;
int main() {
  int n,m; cin>>n>>m;
  char g[100][100];
  rep(i,n) rep(j,m) cin>>g[i][j];
  vector<pair<int,int>> v;
  rep(i,n) rep(j,m) if(g[i][j]=='*') v.push_back({i,j});
  if(v.size()!=n*m){cout<<"-1";return 0;}
  int sz=0;
  rep(i,n) rep(j,m) if(g[i][j]=='.'){sz++;g[i][j]='*';}
  cout<<sz<<endl;
  rep(i,n) rep(j,m) if(g[i][j]=='.') cout<<i+1<<' '<<j+1<<' '<<1<<endl;
}
