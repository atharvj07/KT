#include<bits/stdc++.h>
using namespace std;
#define pb push_back
vector<int>adj[100005];
int n,a[200005],b[200005],vis[100005];
bool dfs(int node,int p)
{
	vis[node]=1;
	for(int child:adj[node])
	{
		if(vis[child]==0)
		{
			if(dfs(child,node))
				return true;
		}
		else if(child!=p)
		{
			return true;
		}
	}
	return false;
}
int main()
{
	cin>>n;
	for(int i=1;i<=2*n;i++)
	{
		cin>>a[i]>>b[i];
		adj[a[i]].pb(b[i]);
		adj[b[i]].pb(a[i]);
	}
	int ans=-1;
	for(int i=1;i<=n;i++)
	{
		if(vis[i]==0)
		{
			if(dfs(i,-1))
			{
				ans=1;
				break;
			}
		}
	}
	if(ans==1)
	{
		cout<<-1;
		return 0;
	}
	memset(vis,0,sizeof(vis));
	vector<int>res;
	for(int i=1;i<=n;i++)
	{
		if(vis[i]==0)
		{
			vector<int>vv;
			vv.pb(i);
			vis[i]=1;
			int c=adj[i][0];
			vis[c]=1;
			vv.pb(c);
			while(1)
			{
				int x=-1;
				for(int v:adj[c])
				{
					if(vis[v]==0)
					{
						x=v;
						break;
					}
				}
				if(x==-1)
					break;
				vv.pb(x);
				vis[x]=1;
				c=x;
			}
			for(int y:vv)
				res.pb(y);
		}
	}
	for(int x:res)
		cout<<x<<" ";
}
