import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Main {
    private static int n;
    private static int[][] adj;
    private static boolean[][] visited;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        adj = new int[n][n];
        visited = new boolean[n][n];

        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < n; j++) {
                adj[i][j] = line.charAt(j) - '0';
            }
        }

        int[] cycle = findCycle();
        if (cycle == null) {
            System.out.println(-1);
        } else {
            for (int v : cycle) {
                System.out.print(v + " ");
            }
            System.out.println();
        }
    }

    private static int[] findCycle() {
        for (int i = 0; i < n; i++) {
            if (!visited[i][i]) {
                List<Integer> path = new ArrayList<>();
                if (dfs(i, i, path)) {
                    return toIntArray(path);
                }
            }
        }
        return null;
    }

    private static boolean dfs(int v, int start, List<Integer> path) {
        if (visited[v][start]) {
            return path.size() >= 3;
        }
        visited[v][start] = true;
        path.add(v);
        for (int u = 0; u < n; u++) {
            if (adj[v][u] > 0 && dfs(u, start, path)) {
                return true;
            }
        }
        path.remove(path.size() - 1);
        return false;
    }

    private static int[] toIntArray(List<Integer> list) {
        int[] array = new int[list.size()];
        for (int i = 0; i < list.size(); i++) {
            array[i] = list.get(i);
        }
        return array;
    }
}
