#include <cmath>
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n], ans = -1e9;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        bool ok = sqrt(a[i]) == floor(sqrt(a[i]));
        if (!ok) ans = max(ans, a[i]);
    }
    cout << ans << endl;
}
