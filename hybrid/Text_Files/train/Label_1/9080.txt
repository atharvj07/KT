#include <iostream>
#include <unordered_map>
using namespace std;
int main() {
    int t; string a, b, c;
    cin >> t;
    while (t--) {
        cin >> a >> b >> c;
        unordered_map<char, int> cnt;
        for (char x : (a + b + c)) { cnt[x]++; }
        int odd_cnt = 0;
        for (auto &x : cnt) { if (x.second % 2) { odd_cnt++; } }
        cout << (odd_cnt <= 2 ? "YES" : "NO") << endl;
    }
}
