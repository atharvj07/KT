#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); i++)
typedef long long ll;
const ll MOD = 1e9 + 7;

map<pair<ll, ll>, ll> mp;
int main() {
  ll n;
  cin >> n;
  vector<pair<ll, ll>> a(n);
  rep(i, n) {
    ll x, y;
    cin >> x >> y;
    a[i] = {x, y};
    mp[a[i]]++;
  }
  ll ans = 0;
  rep(i, n) {
    rep(j, n) {
      if (i == j) continue;
      ll dx = a[i].first - a[j].first;
      ll dy = a[i].second - a[j].second;
      ans += mp[{a[j].first + dy, a[j].second - dx}] *
             mp[{a[i].first + dy, a[i].second - dx}];
    }
  }
  ans /= 2;
  cout << ans << endl;
  return 0;
}
