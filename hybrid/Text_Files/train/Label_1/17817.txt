#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> v(n);
  for (int i = 0; i < n; ++i) cin >> v[i];
  sort(v.begin(), v.end());
  long long sum = 0;
  for (int i = 0; i < n; ++i) {
    if (sum + 1 >= v[i])
      sum++;
    else
      sum = v[i];
    cout << sum << "\n";
  }
  return 0;
}
