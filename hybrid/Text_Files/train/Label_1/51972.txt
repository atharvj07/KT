import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] a = in.nextLine().toCharArray();
        int n = a.length;
        int c = 0;
        for (int i = 0; i < n / 2; i++) {
            if (a[i] != a[n - i - 1]) {
                if (++c > 1) {
                    System.out.println("NO");
                    return;
                }
            }
        }
        System.out.println("YES");
    }
}
