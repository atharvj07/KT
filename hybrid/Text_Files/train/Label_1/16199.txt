#include <iostream>
#include <vector>

using namespace std;

bool canPass(int n, vector<int>& x, vector<int>& y) {
    vector<bool> passed(n + 1, false);
    for (auto& i : x) passed[i] = true;
    for (auto& i : y) passed[i] = true;
    for (int i = 1; i <= n; ++i) {
        if (!passed[i]) return false;
    }
    return true;
}

int main() {
    int n;
    cin >> n;
    vector<int> x;
    vector<int> y;
    int p;
    cin >> p;
    for (int i = 0; i < p; ++i) {
        int temp;
        cin >> temp;
        x.push_back(temp);
    }
    int q;
    cin >> q;
    for (int i = 0; i < q; ++i) {
        int temp;
        cin >> temp;
        y.push_back(temp);
    }
    if (canPass(n, x, y)) {
        cout << "I become the guy." << endl;
    } else {
        cout << "Oh, my keyboard!" << endl;
    }
    return 0;
}
