#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0, j = n - 1; i < j; i++, j--) {
        if (a[i] + a[j] == a[i + 1] + a[j - 1]) ans++;
        else if (a[i] + a[j] > a[i + 1] + a[j - 1]) break;
    }
    cout << ans << endl;
    return 0;
}
