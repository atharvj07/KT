#include<bits/stdc++.h>
using namespace std;

const int MOD = 998244353;
int pow(long long a, int b) {
    if (!b) return 1LL;
    if (b == 1) return a;
    auto x = pow(a, b >> 1);
    x = x * 1LL * x % MOD;
    if (b & 1) x = x * 1LL * a % MOD;
    return x;
}

int main() {
    int n, m, a, inv6, sum = 0;
    cin >> n >> m;
    inv6 = pow(6, MOD - 2);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a;
            sum += (a <= n);
        }
    }
    cout << sum * 1LL * inv6 % MOD << '\n';
    return 0;
}
