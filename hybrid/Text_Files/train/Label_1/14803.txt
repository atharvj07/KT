#include<bits/stdc++.h>
using namespace std;typedef long long ll;
const ll INF=2e18;
ll a[150005],b[150005],c[150005],d[150005];
vector<vector<ll>>adj1,adj2,adj3;
ll n1,n2,n3,n4,m1,m2,m3;
ll match1[150005],match2[150005],match3[150005],match4[150005];
ll vis1[150005],vis2[150005],vis3[150005];
ll dfs1(ll u){
    if(vis1[u])return 0;
    vis1[u]=1;
    for(auto v:adj1[u]){
        if(match2[v]==-1||dfs1(match2[v])){
            match1[u]=v;
            match2[v]=u;
            return 1;
        }
    }
    return 0;
}
ll dfs2(ll u){
    if(vis2[u])return 0;
    vis2[u]=1;
    for(auto v:adj2[u]){
        if(match3[v]==-1||dfs2(match3[v])){
            match2[u]=v;
            match3[v]=u;
            return 1;
        }
    }
    return 0;
}
ll dfs3(ll u){
    if(vis3[u])return 0;
    vis3[u]=1;
    for(auto v:adj3[u]){
        if(match4[v]==-1||dfs3(match4[v])){
            match3[u]=v;
            match4[v]=u;
            return 1;
        }
    }
    return 0;
}
int main(){
    cin>>n1>>n2>>n3>>n4;
    for(ll i=1;i<=n1;++i)cin>>a[i];
    for(ll i=1;i<=n2;++i)cin>>b[i];
    for(ll i=1;i<=n3;++i)cin>>c[i];
    for(ll i=1;i<=n4;++i)cin>>d[i];
    memset(match1,-1,sizeof match1);
    memset(match2,-1,sizeof match2);
    memset(match3,-1,sizeof match3);
    memset(match4,-1,sizeof match4);
    cin>>m1;
    adj1.resize(n1+1);
    for(ll i=0;i<m1;++i){
        ll x,y;cin>>x>>y;
        adj1[x].push_back(y);
    }
    cin>>m2;
    adj2.resize(n2+1);
    for(ll i=0;i<m2;++i){
        ll x,y;cin>>x>>y;
        adj2[x].push_back(y);
    }
    cin>>m3;
    adj3.resize(n3+1);
    for(ll i=0;i<m3;++i){
        ll x,y;cin>>x>>y;
        adj3[x].push_back(y);
    }
    ll ans=INF;
    for(ll i=1;i<=n1;++i){
        memset(vis1,0,sizeof vis1);
        memset(vis2,0,sizeof vis2);
        memset(vis3,0,sizeof vis3);
        if(!dfs1(i))continue;
        if(!dfs2(match1[i]))continue;
        if(!dfs3(match2[match1[i]]))continue;
        ans=min(ans,a[i]+b[match1[i]]+c[match2[match1[i]]]+d[match3[match2[match1[i]]]]);
    }
    if(ans==INF)cout<<-1<<endl;
    else cout<<ans<<endl;
    return 0;
}
