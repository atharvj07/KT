n, m, k = map(int, input().split())
moves = [tuple(map(int, input().split())) for _ in range(k)]

# A set to keep track of black pixels
black_pixels = set()

for move_num, (i, j) in enumerate(moves, 1):
    # Add the current move to the set of black pixels
    black_pixels.add((i, j))
    
    # Check all 4 possible 2x2 squares that the current pixel can be part of
    if ((i, j) in black_pixels and
        (i, j + 1) in black_pixels and
        (i + 1, j) in black_pixels and
        (i + 1, j + 1) in black_pixels) or \
       ((i, j) in black_pixels and
        (i, j - 1) in black_pixels and
        (i + 1, j) in black_pixels and
        (i + 1, j - 1) in black_pixels) or \
       ((i, j) in black_pixels and
        (i, j + 1) in black_pixels and
        (i - 1, j) in black_pixels and
        (i - 1, j + 1) in black_pixels) or \
       ((i, j) in black_pixels and
        (i, j - 1) in black_pixels and
        (i - 1, j) in black_pixels and
        (i - 1, j - 1) in black_pixels):
        print(move_num)
        break
else:
    print(0)
