from sys import stdin

N, M = map(int, stdin.readline().strip().split())
grid = [[c for c in line.strip()] for line in stdin ]

def valid(x, y, dx, dy):
    if 0 <= x + dx < N and 0 <= y + dy < M:
        return grid[x + dx][y + dy] == '.'
    return False

def countPaths(x, y, px, py, turns):
    if x == N - 1 or x == 0 or y == M - 1 or y == 0:
        if turns <= 2:
            return 1
        return 0

    cnt = 0
    for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
        if dx != px or dy != py:
            if valid(x, y, dx, dy):
                cnt += countPaths(x + dx, y + dy, dx, dy, turns)
            if turns < 2 and valid(x, y, dx, dy) and valid(x, y, -dx, -dy):
                cnt += countPaths(x + dx, y + dy, -dx, -dy, turns + 1)
    return cnt

print(countPaths(0, 0, 0, 0, 0))
