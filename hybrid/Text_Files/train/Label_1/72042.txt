t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    possible = False
    for i in range(n-2):
        if a[i] + a[i+1] > a[i+2]:
            continue
        else:
            print(i+1, i+2, i+3)
            possible = True
            break
    if not possible:
        print(-1)
