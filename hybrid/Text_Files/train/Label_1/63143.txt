def merge_sort(li):
    if len(li) <= 1:
        return li
    m = (len(li) + 1) // 2
    left = merge_sort(li[:m])
    right = merge_sort(li[m:])
    return merge(left, right)

def merge(left, right):
    res = []
    i, j = 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            res.append(left[i])
            i += 1
        else:
            res.append(right[j])
            j += 1
    res.extend(left[i:])
    res.extend(right[j:])
    return res

def get_sum(a):
    b = sorted(a)
    c = b[::-1]
    ans = 0
    for i in range(len(a) // 2):
        ans += abs(b[i] - c[i])
    return ans
