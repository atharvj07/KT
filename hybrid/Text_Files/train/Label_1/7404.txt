#include<bits/stdc++.h>
#define ll long long int
using namespace std;

int main()
{
    ll t,n,p,k,i,j,ans;
    cin>>t;
    while(t--)
    {
        cin>>n>>p>>k;
        ll a[n+1];
        for(i=1;i<=n;i++)
        {
            cin>>a[i];
        }
        sort(a+1,a+n+1);
        j=n;
        ans=0;
        for(i=n;i>=1;i--)
        {
            if(p>=a[i])
            {
                p-=a[i];
                ans++;
                j=i-1;
            }
            if(k==2)
                continue;
            while(j>=1 && ans+1<=n)
            {
                if(j==i)
                {
                    j--;
                    continue;
                }
                if(p>=a[j])
                {
                    p-=a[j];
                    ans++;
                    j--;
                }
                else
                    break;
            }
            k--;
            i=j;
        }
        cout<<ans<<endl;
    }
    return 0;
}
