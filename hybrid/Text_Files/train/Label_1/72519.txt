n, k = map(int, input().split())
p = list(map(int, input().split()))
q = list(map(int, input().split()))
if p[0] != q[0]:
    print('NO')
else:
    s = p[0] - 1
    for i in range(1, n):
        if p[i] != q[i]:
            s += chr(ord('a') + k - 1)
        else:
            s += chr(ord('a') + s - p[i] + 1)
    print('YES')
    print(s)
