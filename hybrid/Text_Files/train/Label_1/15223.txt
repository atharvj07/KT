#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int &x : v) cin >> x;
  sort(v.begin(), v.end());
  int ans = 0, sum = 0;
  for (int i = 0; i < n - 1; i++) {
    sum += v[i];
    if (2 * sum == accumulate(v.begin() + i + 1, v.end(), 0)) ans++;
  }
  cout << ans << endl;
  return 0;
}
