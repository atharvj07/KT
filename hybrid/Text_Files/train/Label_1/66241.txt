import sys
input = sys.stdin.readline

N, V = map(int,input().split())
A = [(int(a),int(b)) for a,b in [input().split() for i in range(N)]]
A.sort()
ans = 0
V_cnt = 0
for a,b in A:
    if V_cnt == 0: ans += min(b,V)
    else:
        if V_cnt + min(b,V-V_cnt) >= V:
            ans += V - V_cnt
            V_cnt = 0
            if min(b,V) == V: ans += min(b-V,V)
        else: V_cnt+= min(b,V-V_cnt)
print(ans)
