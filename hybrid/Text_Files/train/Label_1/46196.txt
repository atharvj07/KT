import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long q = sc.nextInt(), x = sc.nextInt();
        long[] a = new long[(int) q];
        int pos = 0;
        for (int i = 0; i < q; i++) {
            a[i] = sc.nextInt();
            while (pos < i && a[i] - a[pos] > x)
                pos++;
            long cur = i == pos ? a[i] : a[pos] + x;
            for (int j = pos; j <= i; j++)
                a[j] = cur;
            System.out.println(i == 0 ? 0 : a[i] + 1);
        }
    }
}
