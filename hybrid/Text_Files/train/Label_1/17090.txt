#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
int main() {
    int n;
    cin >> n;
    vi a(n);
    for(int &i: a) cin >> i;
    int ans = 0;
    for(int i = 1; i < n; ++i) {
        if(gcd(a[i], a[i - 1]) == 1) continue;
        bool f = 0;
        for(int j = 2; j <= sqrt(gcd(a[i], a[i - 1])); ++j) {
            if(gcd(a[i], a[i - 1]) % j == 0) {
                a.insert(a.begin() + i, j);
                ++ans;
                f = 1;
                break;
            }
        }
        if(!f) {
            a.insert(a.begin() + i, 1);
            ++ans;
        }
    }
    cout << ans << "\n";
    for(int i: a) cout << i << " ";
    return 0;
}
