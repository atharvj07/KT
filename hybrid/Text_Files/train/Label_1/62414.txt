def min_usages_of_power(t, test_cases):
    results = []
    for i in range(t):
        r, c, grid = test_cases[i]
        if all(row == "A" * c for row in grid):
            results.append(0)  # All Beingawesomeism
            continue
        if "A" not in ''.join(grid):
            results.append("MORTAL")  # All Pushingittoofarism
            continue

        # Check for conditions to solve in 1 move
        solved_in_one = False
        for row in grid:
            if row == "A" * c:
                solved_in_one = True
                break
        for col in zip(*grid):
            if ''.join(col) == "A" * r:
                solved_in_one = True
                break
        if grid[0][0] == "A" or grid[0][-1] == "A" or grid[-1][0] == "A" or grid[-1][-1] == "A":
            solved_in_one = True
        if solved_in_one:
            results.append(1)
            continue

        # Check for conditions to solve in 2 moves
        solved_in_two = False
        if "A" in grid[0] or "A" in grid[-1] or "A" in [row[0] for row in grid] or "A" in [row[-1] for row in grid]:
            solved_in_two = True
        if any(row.count("A") >= c - 1 for row in grid) \
                or any(col.count("A") >= r - 1 for col in zip(*grid)):
            solved_in_two = True
        if solved_in_two:
            results.append(2)
            continue

        # Check for conditions to solve in 3 moves
        if any("A" in row for row in grid) or any("A" in col for col in zip(*grid)):
            results.append(3)
        else:
            results.append("MORTAL")
    return results

# Example usage
t = int(input())
test_cases = []
for _ in range(t):
    r, c = map(int, input().split())
    grid = [input().strip() for _ in range(r)]
    test_cases.append((r, c, grid))

results = min_usages_of_power(t, test_cases)
for result in results:
    print(result)
