MOD = 998244353

def modBinomial(n, k, p):
    numerator = 1
    for i in range(k):
        numerator = (numerator * (n - i)) % p
    denominator = 1
    for i in range(1, k + 1):
        denominator = (denominator * i) % p
    return (numerator * pow(denominator, p - 2, p)) % p

def solve(n, m):
    return (modBinomial(m, n - 1, MOD) * (n - 2) * pow(2, n - 3, MOD)) % MOD

n, m = map(int, input().split())
print(solve(n, m))
