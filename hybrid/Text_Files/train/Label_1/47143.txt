import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int N = scanner.nextInt();
        int[][] data = new int[N][4];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < 4; j++) {
                data[i][j] = scanner.nextInt();
            }
        }

        int[] congestion = new int[361];

        for (int i = 0; i < N; i++) {
            int[] h = data[i];
            int month = h[0];
            int day = h[1];
            int vacation = h[2];
            int si = h[3];
            int maxCongestion = Integer.MIN_VALUE;
            for (int j = 0; j < vacation; j++) {
                int c = si;
                if (j > day - 1) {
                    c = Math.max(0, si - (j - day + 1));
                }
                if (j < 30 - day) {
                    c = Math.max(0, si - (30 - day - j));
                }
                congestion[month * 30 + day + j] = Math.max(congestion[month * 30 + day + j], c);
            }
        }

        int minCongestion = Integer.MAX_VALUE;
        for (int i = 1; i <= 360; i++) {
            minCongestion = Math.min(minCongestion, congestion[i]);
        }

        System.out.println(minCongestion);
    }
}
