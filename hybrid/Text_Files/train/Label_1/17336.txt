#include<bits/stdc++.h>
#define ll long long
using namespace std;
int a[3][55],b[55],n,now1,now2,ans;
void dfs1(int aa,int st1,int st2,int tim){
    if(aa==n){
        int tm=(tim+b[st2]);
        if(now1!=st1||now2!=st2) tm-=b[st2];
        ans=min(ans,tm);
        return ;
    }
    dfs1(aa+1,st1,aa+1,tim+a[1][aa]);
    dfs1(aa+1,st1,aa+2,tim+a[2][aa]);
}
void dfs2(int aa,int st1,int st2,int tim){
    if(aa==0){
        dfs1(1,st1,st2,tim);
        return ;
    }
    dfs2(aa-1,aa,st2,tim+b[aa]);
    dfs2(aa-1,st1,aa,tim+b[aa]);
}
int main()
{
    cin>>n;
    for(int i=1;i<=2;i++){
        for(int j=1;j<=n-1;j++){
            cin>>a[i][j];
        }
    }
    for(int i=1;i<=n;i++){
        cin>>b[i];
    }
    ans=1e9;
    dfs2(n,0,0,0);
    cout<<ans<<endl;
    return 0;
}
