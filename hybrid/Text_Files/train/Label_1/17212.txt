#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007

int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> dp(m + 1, 0);
  dp[0] = 1;
  for (ll i = 1; i <= m; i++) {
    for (ll j = 1; j <= i; j++) {
      (dp[i] += dp[i - j] * j) %= mod;
      (dp[i] += dp[i - j] * (n - j + 1)) %= mod;
    }
  }

  cout << dp[m] << "\n";
  return 0;
}
