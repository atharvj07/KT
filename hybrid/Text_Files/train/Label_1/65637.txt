MOD = 10**9 + 7

n = int(input())
actions = [input().split() for _ in range(n)]

# Initialize variables to keep track of the minimum SELL price and maximum BUY price
min_sell = float('inf')
max_buy = float('-inf')

# Variables to track possibilities and the price to action mappings
possibilities = 1
price_to_action = {}

for action, price in actions:
    price = int(price)
    if action == "ADD":
        # Check if the price is in a valid range
        if price < min_sell and price > max_buy:
            # If the price can be both BUY and SELL, double the possibilities
            possibilities = (possibilities * 2) % MOD
        price_to_action[price] = action
    else:  # ACCEPT action
        if price_to_action[price] == "ADD":
            # Determine if the accepted order was a BUY or a SELL
            if price == min_sell:
                # It was a SELL order being accepted
                min_sell = float('inf')
            elif price == max_buy:
                # It was a BUY order being accepted
                max_buy = float('-inf')
            else:
                possibilities = 0
                break
            
            # Update min_sell and max_buy based on the current state of the order book
            for p in price_to_action.keys():
                if p > max_buy and p < min_sell:
                    if p > price:
                        min_sell = min(min_sell, p)
                    else:
                        max_buy = max(max_buy, p)
        else:
            possibilities = 0
            break

print(possibilities)
