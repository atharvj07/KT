import sys
input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

vis = [0] * (n+1)
def dfs(u, group):
    vis[u] = group
    for v in graph[u]:
        if vis[v] == 0:
            dfs(v, 3-group)

dfs(1, 1)

if 0 in vis:
    print(-1)
else:
    a = []
    b = []
    for i in range(1, n+1):
        if vis[i] == 1:
            a.append(i)
        else:
            b.append(i)
    print(len(a))
    print(*a)
    print(len(b))
    print(*b)
