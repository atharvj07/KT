n, M = map(int, input().split())
t = list(map(int, input().split()))

for i in range(n):
    sorted_t = sorted(t[:i])
    time_left = M - t[i]
    students_to_fail = 0
    for time in sorted_t[::-1]:
        if time_left >= time:
            time_left -= time
        else:
            students_to_fail += 1
    print(students_to_fail, end=' ')
