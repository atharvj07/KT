n, m = map(int, input().split())
L = [list(input()) for _ in range(n)]
X = []
for i in range(n):
    for j in range(m):
        if L[i][j] == 'X':
            X.append((i, j))
A = [X[0]]
for i in X[1:]:
    a, b = A[-1]
    c, d = i
    if a <= c <= a + 1 and b - 1 <= d <= b + 1:
        continue
    A.append(i)
if len(A) == len(X):
    print('YES')
else:
    print('NO')
