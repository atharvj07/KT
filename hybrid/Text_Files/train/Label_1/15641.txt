#include <iostream>
#include <vector>
using namespace std;

int main() {
  int k;
  cin >> k;
  while (k--) {
    int n;
    cin >> n;
    vector<int> s, t;
    int diff = 0, same = 0;
    for (int i = 0; i < n; i++) {
      char c;
      cin >> c;
      s.push_back(c);
      cin >> c;
      t.push_back(c);
      if (c == s[i]) same++;
    }
    for (int i = 0; i < n; i++) {
      if (s[i] != t[i]) diff++;
    }
    cout << (diff <= 2 || (diff == 4 && same == n - 2) ? "Yes\n" : "No\n");
  }
}
