n, h, k = map(int, input().split())
a = list(map(int, input().split()))

time = 0
current_height = 0

for potato in a:
    if current_height + potato <= h:
        current_height += potato
    else:
        time += 1
        current_height = potato
    time += current_height // k
    current_height %= k

time += (current_height > 0)
print(time)
