from collections import defaultdict
from math import sqrt

n,k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
dp = defaultdict(int)
# dp stores the length of longest k-multiple free list ending with a[i]
dp[0] = 1
for i in range(n):
    for j in range(i):
        if a[i] % a[j] != 0:
            dp[i] = max(dp[i], dp[j] + 1)
print(max(dp.values()))
