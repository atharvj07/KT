#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int a,b,c,d,k;
        cin>>a>>b>>c>>d>>k;
        if(a%c==0)a=a/c;
        else a=(a/c)+1;
        if(b%d==0)b=b/d;
        else b=(b/d)+1;
        if(a+b<=k)cout<<a<<" "<<b<<endl;
        else cout<<-1<<endl;
    }
    return 0;
}
