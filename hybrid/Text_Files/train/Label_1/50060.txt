import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    static class Node {
        int u, depth;

        public Node(int u, int depth) {
            this.u = u;
            this.depth = depth;
        }
    }

    static List<Integer>[] adj;
    static int[] inTime;
    static int[] outTime;
    static boolean[] vis;
    static int[] cnt;
    static int timer;
    static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        adj = new List[n + 1];
        vis = new boolean[n + 1];
        inTime = new int[n + 1];
        outTime = new int[n + 1];
        cnt = new int[n + 1];
        for (int i = 0; i <= n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }
        dfs1(1, 0);
        dfs2(1, 0);
        int ans = -1, mx = 0;
        for (int i = 1; i <= n; i++) {
            if (cnt[i] > mx) {
                mx = cnt[i];
                ans = i;
            }
        }
        System.out.println(ans);
    }

    public static void dfs1(int u, int d) {
        vis[u] = true;
        inTime[u] = timer++;
        for (int v : adj[u]) {
            if (!vis[v]) {
                dfs1(v, d + 1);
            }
        }
        outTime[u] = timer++;
    }

    public static void dfs2(int u, int p) {
        Set<Integer> hs = new HashSet<>();
        for (int v : adj[u]) {
            if (v != p) {
                dfs2(v, u);
                hs.add(cnt[v]);
            }
        }
        cnt[u] = hs.size() + 1;
    }
}
