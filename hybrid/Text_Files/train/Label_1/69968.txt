n = int(input())
s = list(input())
depth = 0
cur = 0
is_red = [False] * n
is_blue = [False] * n
for i in range(n):
    if s[i] == '(':
        depth += 1
        if depth > cur:
            cur = depth
            for j in range(0, i):
                if not is_blue[j]:
                    is_red[j] = True
    else:
        if depth > 0:
            depth -= 1
            is_blue[i] = True
for i in range(n):
    if is_red[i]:
        print('0', end='')
    else:
        print('1', end='')
print()
