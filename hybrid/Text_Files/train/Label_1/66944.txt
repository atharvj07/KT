from collections import deque
from sys import stdin

r, c, k = map(int, stdin.readline().split())
sensors = [tuple(map(int, stdin.readline().split())) for _ in range(k)]

grid = [['.' for _ in range(c+2)] for _ in range(r+2)]
for x, y in sensors:
    grid[x][y] = 'S'
grid[0][0] = 'L'
grid[1][1] = 'E'

q = deque([(0, 0, 0)])
time = [[-1 for _ in range(c+2)] for _ in range(r+2)]
time[0][0] = 0

dx = [1, -1, 0, 0]
dy = [0, 0, 1, -1]

while q:
    x, y, t = q.popleft()
    for d in range(4):
        nx, ny = x + dx[d], y + dy[d]
        if 0 <= nx < r+2 and 0 <= ny < c+2 and grid[nx][ny] != 'S' and time[nx][ny] == -1:
            q.append((nx, ny, t+1))
            time[nx][ny] = t+1

for x, y in sensors:
    print(time[x][y])
