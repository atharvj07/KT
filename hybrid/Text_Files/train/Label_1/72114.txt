l = int(input())
stack = []
for _ in range(l):
    command = input()
    if command.startswith("for "):
        stack.append((int(command[4:]), True))
    elif command == "end":
        if stack:
            stack[-1][1] = not stack[-1][1]
    elif command == "add":
        if not stack or not stack[-1][1]:
            print("OVERFLOW!!!")
            break
        stack[-1][0] -= 1
else:
    print(0)
