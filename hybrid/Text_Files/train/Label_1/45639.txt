import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n + 1];
    int[] l = new int[n + 1];
    int[] r = new int[n + 1];
    a[0] = -1;
    for (int i = 1; i <= n; ++i) {
      a[i] = in.nextInt();
      if (i == 1) {
        l[i] = n;
        r[i] = i + 1;
      } else if (a[i - 1] < a[i]) {
        l[i] = i - 1;
        r[i] = i + 1;
      } else {
        l[i] = l[i - 1];
        r[i] = i + 1;
      }
    }
    int cnt = 0;
    for (int i = 1; i <= n; ++i) {
      if (l[i] != i) {
        continue;
      }
      boolean ok = true;
      for (int j = r[i]; j <= n; ++j) {
        if (a[j] < a[i]) {
          ok = false;
          break;
        }
      }
      if (ok) {
        ++cnt;
      }
    }
    System.out.println(cnt);
  }
}
