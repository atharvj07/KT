n=input()
s=0
p=0
for i in range(len(n)):
    s+=int(n[i])
    p=(p*10+int(n[i]))%8
sum=p+(s-p)%8
if (sum%8)!=0:
    sum%=8
    for i in range(len(n)-1,-1,-1):
        if int(n[i])<sum:
            sum-=int(n[i])
            n=n[0:i]+n[i+1:]
print('YES' if (sum==0) else 'NO')
print(n if sum==0 else -1)
