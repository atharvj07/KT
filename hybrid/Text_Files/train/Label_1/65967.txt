vowels = "eyuioa"
n = int(input())
s = input()
res = []
for i, ch in enumerate(s):
    if ch in vowels:
        if i > 0 and s[i - 1] == ch:
            if ch in "eo" and i + 1 < n and s[i + 1] == ch:
                res.append(ch)
            elif i + 1 < n and s[i + 1] == ch:
                res.append(ch)
        else:
            res.append(ch)
print("".join(res))
