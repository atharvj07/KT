#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  bool f = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'a' || s[i] == 'o' || s[i] == 'u' || s[i] == 'i' || s[i] == 'e')
      f = 0;
    else if (s[i] == 'n') {
      if (i == s.size() - 1) {
        f = 1;
        break;
      } else
        f = 0;
    } else {
      if (f == 1) {
        f = 1;
        break;
      }
      f = 1;
    }
  }
  if (f)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
