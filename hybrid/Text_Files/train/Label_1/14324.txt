#include<bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int n, x; cin >> n >> x;
    if ((x == 0 && n != 1) || n == 1 && x != 0) { cout << "NO" << endl; return 0; }
    const int SZ = 15;
    vector<int> mp(SZ, 0);
    mp[0] = 1;
    for (int i = 0; i < n - 1; i++) {
        for (int j = SZ - 1; j >= 0; j--) {
            if (mp[j] == 1) {
                mp[j] = 0;
                int v = (1 << j);
                if (x - v >= 0 && x - v < SZ && mp[x - v] == 0) { mp[x - v] = 1; }
                if (x + v < SZ && mp[x + v] == 0) { mp[x + v] = 1; }
            }
        }
    }
    if (mp[x] == 0) { cout << "NO" << endl; return 0; }
    cout << "YES" << endl;
    for (int i = 0; i < SZ; i++) { if (mp[i] == 1) { cout << i << ' '; } }
    cout << endl;
    return 0;
}
