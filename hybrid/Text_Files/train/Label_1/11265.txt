#include <iostream>
#include <vector>
#include <queue>
#include <tuple>

using namespace std;

const int MAXN = 5005;
int n, m, k, ans, dist[MAXN];
vector<int> G[MAXN];
vector<int> Ds[MAXN];
vector<tuple<int, int, int>> E;

void dfs1(int u, int p) {
    for (int v : G[u]) {
        if (v == p) continue;
        Ds[u].push_back(v);
        dfs1(v, u);
    }
}

void dfs2(int u, int p) {
    for (int v : Ds[u]) {
        if (v == p) continue;
        dist[v] = dist[u] + 1;
        dfs2(v, u);
    }
}

int main() {
    cin >> n >> m >> k;
    E.resize(n);
    for (int i = 0; i < n; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        E[i] = make_tuple(a, b, c);
    }
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        G[u].push_back(v);
    }
    dfs1(1, 1);
    dfs2(1, 1);
    for (int i = n - 1; i >= 0; i--) {
        int a, b, c;
        tie(a, b, c) = E[i];
        if (k >= a) {
            k += b;
            ans += c;
            for (int v : Ds[i + 1]) {
                if (dist[v] < dist[i + 1] - 1) {
                    k -= 1;
                    if (k < 0) {
                        cout << -1 << endl;
                        return 0;
                    }
                }
            }
        } else {
            cout << -1 << endl;
            return 0;
        }
    }
    cout << ans << endl;
    return 0;
}
