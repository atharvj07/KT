import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[][] bricks = new int[n][4];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                bricks[i][j] = input.nextInt();
            }
        }

        Arrays.sort(bricks, (a, b) -> a[2] - b[2]);

        int[] dp = new int[n];
        int maxTower = 0;
        for (int i = 0; i < n; i++) {
            dp[i] = 1;
            for (int j = 0; j < i; j++) {
                if (bricks[i][0] >= bricks[j][0] && bricks[i][1] >= bricks[j][1] && bricks[i][2] >= bricks[j][2] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                }
            }
            maxTower = Math.max(maxTower, dp[i]);
        }

        System.out.println(maxTower);
    }
}
