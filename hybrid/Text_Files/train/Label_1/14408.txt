#include <bits/stdc++.h>
using namespace std;
#define int long long
const int inf = 1e9;
int n, m, k;
vector<pair<int, int>>g[200004];
int dis[200004], vis[200004];

void dij(int src) {
  int i;
  dis[src] = 0;
  priority_queue<pair<int, int>>q;
  q.push({0, src});
  while (!q.empty()) {
    int u = q.top().second, w = -q.top().first;
    q.pop();
    if (vis[u])continue;
    vis[u] = 1;
    for (i = 0; i < g[u].size(); i++) {
      int v = g[u][i].second, cst = g[u][i].first;
      if (dis[v] > dis[u] + cst) {
        dis[v] = dis[u] + cst;
        q.push({-dis[v], v});
      }
    }
  }
}

int32_t main() {
  cin >> n >> m >> k;
  int a, b, c;
  for (int i = 0; i < m; i++) {
    cin >> a >> b >> c;
    g[a].push_back({c, b});
    g[b].push_back({c, a});
  }
  for (int i = 0; i <= n; i++)dis[i] = inf;
  dij(1);
  vector<int>v;
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      if (i == j)continue;
      v.push_back(dis[i] + dis[j]);
    }
  }
  sort(v.begin(), v.end());
  cout << v[k - 1] << endl;
  return 0;
}
