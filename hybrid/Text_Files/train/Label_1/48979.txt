import java.util.*;

public class Main {
    public static void main(String args[]) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) { a[i] = input.nextInt(); }
        int[] left = new int[n + 1], right = new int[n + 1];
        for (int i = 0; i < k; i++) {
            left[a[i]] = i + 1;
            right[a[i]] = i + 1;
        }
        for (int i = 1; i < n; i++) {
            if (left[i] == 0) { left[i] = left[i - 1]; }
            if (right[i] == 0) { right[i] = right[i + 1]; }
        }
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += (long)(n - i - right[i] + left[i]) * (long)(i - left[i] + right[i] - 1);
        }
        System.out.println(ans);
    }
}
