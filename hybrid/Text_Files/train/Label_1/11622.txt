#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, a, b, c, t, ans = 0, timer = 0;
  cin >> n >> a >> b >> c >> t;
  vector<int> v(n);
  for (int &x : v) cin >> x;

  sort(v.begin(), v.end());
  for (const int &x : v) {
    while (timer < x) ans += c, timer++;
    ans += max(0, a - b * (timer - x));
  }
  while (timer < t) ans += c, timer++;

  cout << ans << '\n';
  return 0;
}
