#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> prefix(n + 1);
  for (int i = 1; i <= n; i++) {
    prefix[i] = prefix[i - 1] + (s[i - 1] == s[i - 1]);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int left = i - 1 - max(0, prefix[i - 1] - k);
    int right = i + k - max(0, prefix[i + k] - k);
    ans = max(ans, right - left + 1);
  }
  cout << ans << endl;
  return 0;
}
