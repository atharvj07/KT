import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int b = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        int sum = 0;
        int min = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum > b) {
                break;
            }
            k--;
            min = i + 1;
        }
        if (k <= 0) {
            min = n;
        }
        System.out.println(min);
    }
}
