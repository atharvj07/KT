import java.util.*;

class Main {
  static class Edge {
    int u, v, o;

    Edge(int u, int v, int o) {
      this.u = u;
      this.v = v;
      this.o = o;
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt();
    List<Edge> e = new ArrayList<>();
    int[] u = new int[n + 1];
    for (int i = 1; i <= m; i++) {
      int x = sc.nextInt(), y = sc.nextInt();
      u[x]++;
      u[y]++;
      e.add(new Edge(x, y, i));
    }
    List<Integer> res = new ArrayList<>();
    for (Edge ed : e) {
      if (u[ed.u] == 2 && u[ed.v] == 2) {
        res.add(ed.o);
      }
    }
    System.out.println(res.size());
    for (int i : res) {
      System.out.print(i + " ");
    }
  }
}
