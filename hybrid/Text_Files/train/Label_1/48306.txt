import java.util.*;
import java.io.*;
import java.lang.*;

class Main{
    static class Graph{
        int n;
        List<int[]> edges[];
        Graph(int n){
            this.n=n;
            edges=new LinkedList[n+1];
            for(int i=0;i<=n;i++){
                edges[i]=new LinkedList<>();
            }
        }
        void addEdge(int a,int b,int w){
            edges[a].add(new int[]{b,w});
        }
    }
    static class SpecialField{
        int field;
        List<int[]> edges;
        SpecialField(int field){
            this.field=field;
            edges=new LinkedList<>();
        }
    }

    public static void main(String[] args)throws Exception{
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st=new StringTokenizer(br.readLine());
        int n=Integer.parseInt(st.nextToken());
        int m=Integer.parseInt(st.nextToken());
        int k=Integer.parseInt(st.nextToken());

        Graph g=new Graph(n);
        HashSet<Integer> special=new HashSet<>();
        st=new StringTokenizer(br.readLine());
        for(int i=0;i<k;i++){
            special.add(Integer.parseInt(st.nextToken()));
        }

        for(int i=1;i<=m;i++){
            st=new StringTokenizer(br.readLine());
            int a=Integer.parseInt(st.nextToken());
            int b=Integer.parseInt(st.nextToken());
            g.addEdge(a,b,1);
        }

        int max=Integer.MIN_VALUE;
        List<SpecialField> specialFields=new LinkedList<>();
        for(int u:special){
            SpecialField sp=new SpecialField(u);
            int dist[]=BFS(g.edges,u,n);
            for(int i=1;i<=n;i++){
                if(dist[i]>0){
                    sp.edges.add(new int[]{i,dist[i]});
                    max=Math.max(max,dist[i]);
                }
            }
            specialFields.add(sp);
        }

        int[][] dp=new int[specialFields.size()+1][k+1];
        for(int i=1;i<=specialFields.size();i++){
            dp[i][0]=max;
            for(int j=1;j<=k;j++){
                SpecialField sp=specialFields.get(i-1);
                dp[i][j]=Math.max(dp[i-1][j],dp[i][j-1]);
                for(int[] e:sp.edges){
                    dp[i][j]=Math.max(dp[i][j],e[1]+dp[i][j-1]);
                }
            }
        }
        System.out.println(dp[specialFields.size()][k]);
    }

    static int[] BFS(List<int[]> edges[],int node,int n){
        int[] dist=new int[n+1];
        Queue<Integer> q=new LinkedList<>();
        q.add(node);
        dist[node]=0;
        while(!q.isEmpty()){
            int u=q.poll();
            for(int[] e:edges[u]){
                int v=e[0];
                int w=e[1];
                if(dist[v]==0){
                    dist[v]=dist[u]+w;
                    q.add(v);
                }
            }
        }
        return dist;
    }
}
