n, m = map(int, input().split())
s = [0]*(n+1)
for i in range(n):
    c, t = map(int, input().split())
    s[i+1] = s[i] + c*t
l = list(map(int, input().split()))
for v in l:
    left = 0
    right = n
    while left < right:
        mid = (left+right)//2
        if s[mid+1] <= v:
            left = mid+1
        else: right = mid
    print(left+1)
