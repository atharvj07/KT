import sys
input=sys.stdin.readline

n,m,k=map(int,input().split())
li=[]
for _ in range(n):
    li.append(list(map(int,input().split())))
ans=[0]*m
for i in range(m):
    li.sort(key=lambda x:x[i])
    l=0
    r=0
    mn=li[0][i]
    mx=li[0][i]
    cnt=0
    while r<n:
        mn=min(mn,li[r][i])
        mx=max(mx,li[r][i])
        if mx-mn<=k:
            cnt+=1
            r+=1
        else:
            break
    if cnt>1:
        ans[i]=cnt-1
print(*ans)
