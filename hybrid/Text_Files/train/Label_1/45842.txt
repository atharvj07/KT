import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int x = sc.nextInt();

            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }

            Arrays.sort(a);

            boolean possible = true;
            int count = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] > x) {
                    count++;
                } else {
                    possible = false;
                    break;
                }
            }

            if (possible) {
                System.out.println(count);
            } else {
                System.out.println(-1);
            }
        }
    }
}
