from collections import defaultdict
from sys import stdin

input = stdin.readline

n = int(input())
a = list(map(int, input().split()))
pos = defaultdict(int)
imp = False
for i in a:
    if i in pos:
        imp = True
    if i != n - 1:
        pos[i + 1] = 1
pos[1] = 1
if imp:
    print("Impossible")
else:
    print("Possible")
    print(*sorted(pos.keys()))
