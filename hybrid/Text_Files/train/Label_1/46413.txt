import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        Arrays.sort(x);
        int[] d = new int[n];
        for (int i = 1; i < n; i++) {
            d[i] = Integer.highestOneBit(x[i] - x[i - 1]);
        }
        int[] dp = new int[n];
        int[] prv = new int[n];
        int ans = 1;
        int idx = 0;
        for (int i = 1; i < n; i++) {
            dp[i] = 1;
            prv[i] = -1;
            for (int j = i - 1; j >= 0; j--) {
                if (d[i] == d[j] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    prv[i] = j;
                }
            }
            if (dp[i] > ans) {
                ans = dp[i];
                idx = i;
            }
        }
        List<Integer> res = new ArrayList<>();
        while (idx != -1) {
            res.add(0, x[idx]);
            idx = prv[idx];
        }
        System.out.println(ans);
        for (int i : res) {
            System.out.print(i + " ");
        }
    }
}
