#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
int main()
{
    string s;
    cin>>s; ll n=s.size(); ll ans=1;
    for(int i=0;i<n;i++)
    {
        if(s[i]=='w' or s[i]=='m')
        {
            ans*=2;
            ans%=mod;
        }
    }
    cout<<ans<<endl;
}
