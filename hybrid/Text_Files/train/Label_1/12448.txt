#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
int arr[maxn], brr[maxn];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> arr[i];
  for (int i = 0; i < n; ++i) brr[arr[i]] = i;
  int cnt = 0, last = -1;
  string ans = "";
  for (int i = 1; i <= 2 * n; ++i) {
    if (~brr[i] && (last == -1 || brr[last] < brr[i])) {
      ++cnt;
      ans += (brr[i] < last || last == -1 ? 'L' : 'R');
      last = i;
    }
  }
  cout << cnt << endl;
  cout << ans << endl;
  return 0;
}
