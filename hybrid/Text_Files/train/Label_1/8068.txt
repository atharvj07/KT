#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> P;
const int N = 50, M = 105;
int g[N][N], deg[N][N], n, m;
int dir[4][2] = {{0, 1}, {-1, 0}, {0, -1}, {1, 0}};
int main() {
  int i, j, k, x, y, cnt = 0;
  char c;
  string s;
  cin >> n >> m;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) cin >> c, g[i][j] = (c == '#');
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      if (c == 'S' || c == 'E') {
        deg[i][j] = 1;
        cnt += !g[i][j];
      }
      if (g[i][j]) continue;
      for (k = 0; k < 4; k++) x = i + dir[k][0], y = j + dir[k][1], deg[i][j] += !g[x][y];
    }
  }
  cin >> s;
  for (i = 0; i < s.length(); i++) {
    if (deg[x][y] == 1) break;
    c = s[i];
    for (k = 0; k < 4; k++)
      if (c == '0' + k) { x += dir[k][0], y += dir[k][1]; }
    deg[x][y]--;
  }
  if (c == '0' + 3) cnt--;
  cout << cnt << endl;
  return 0;
}
