def move_cars(n, k, grid):
    moves = []
    for i in range(20000):
        moved = False
        for row in range(1, 3):
            for col in range(n):
                car = grid[row][col]
                if car != 0:
                    if (row == 1 and grid[0][col] == car) or (row == 2 and grid[3][col] == car):
                        moves.append((car, row + (1 if row == 1 else 2), col + 1))
                        grid[row][col] = 0
                        grid[0 if row == 1 else 3][col] = 0
                        moved = True
                        break
        if not moved:
            break
    if any(grid[0]) or any(grid[3]):
        return -1
    else:
        return moves

n, k = map(int, input().split())
grid = [list(map(int, input().split())) for _ in range(4)]
moves = move_cars(n, k, grid)
if moves == -1:
    print(-1)
else:
    print(len(moves))
    for move in moves:
        print(*move)
