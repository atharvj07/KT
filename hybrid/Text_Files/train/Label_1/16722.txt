#include <iostream>
#include <vector>

using namespace std;

const int dx[] = {-2, -2, 2, 2};
const int dy[] = {2, -2, 2, -2};

bool canMeet(vector<string>& board) {
  int x1, y1, x2, y2;
  for (int i = 0; i < 8; i++) {
    for (int j = 0; j < 8; j++) {
      if (board[i][j] == 'K') {
        if (x1 == 0) x1 = i, y1 = j;
        else x2 = i, y2 = j;
      }
    }
  }
  for (int k = 0; k < 4; k++) {
    int nx1 = x1 + dx[k], ny1 = y1 + dy[k];
    int nx2 = x2 + dx[k], ny2 = y2 + dy[k];
    if (nx1 >= 0 && nx1 < 8 && ny1 >= 0 && ny1 < 8 &&
        nx2 >= 0 && nx2 < 8 && ny2 >= 0 && ny2 < 8 &&
        board[nx1][ny1] != '#' && board[nx2][ny2] != '#') return true;
  }
  return false;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    vector<string> board(8);
    for (int i = 0; i < 8; i++) cin >> board[i];
    cout << (canMeet(board) ? "YES" : "NO") << '\n';
  }
  return 0;
}
