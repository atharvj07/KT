import sys

def dfs(u, p):
    global xor
    cnt[u] = 1
    for v in g[u]:
        if v != p:
            dfs(v, u)
            cnt[u] += cnt[v]
            xor[u] ^= xor[v]

def solve():
    global mn, mx, xor
    n, k = [int(x) for x in input().split()]
    a = [int(x) for x in input().split()]
    mn = min(a)
    mx = max(a)
    xor = [0 for i in range(n+1)]
    for i in range(n-1):
        u, v = [int(x) for x in input().split()]
        g[u].append(v)
        g[v].append(u)
    dfs(1, 1)
    if mn == mx:
        print("YES")
    elif cnt[1] != n or xor[1] != 0:
        print("NO")
    else:
        print("YES")

for _ in range(int(input())):
    g = [[] for i in range(100001)]
    cnt = [0 for i in range(100001)]
    solve()
