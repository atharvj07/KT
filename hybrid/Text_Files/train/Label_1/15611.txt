#include <iostream>
#include <queue>
using namespace std;

int main() {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    queue<pair<int, int>> q;
    q.push({x1, y1});
    int dist[1005][1005];
    int dx[] = {1, 1, 1, 0, 0, -1, -1, -1};
    int dy[] = {1, 0, -1, 1, -1, 1, 0, -1};
    for (int i = 0; i <= 1000; i++)
        for (int j = 0; j <= 1000; j++)
            dist[i][j] = 1e9;
    dist[x1][y1] = 0;
    while (!q.empty()) {
        int x = q.front().first;
        int y = q.front().second;
        q.pop();
        for (int i = 0; i < 8; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx >= -1000 && nx <= 1000 && ny >= -1000 && ny <= 1000 && dist[nx][ny] > dist[x][y] + 1) {
                dist[nx][ny] = dist[x][y] + 1;
                q.push({nx, ny});
            }
        }
    }
    cout << dist[x2][y2] << endl;
    return 0;
}
