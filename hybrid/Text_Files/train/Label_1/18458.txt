#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define vvi vector<vi>

vi g[100005];

int main() {
    int n;
    cin >> n;
    vi a(n + 1), f(n + 1);
    iota(a.begin(), a.end(), 0);
    for (int i = 1; i <= n; ++i)
        cin >> a[i];
    for (int i = 0; i < n; ++i) {
        int u = a[i], v = a[i + 1];
        if (u > v)
            swap(u, v);
        if (f[u] == f[v]) {
            ++f[v];
            g[u].push_back(v);
            g[v].push_back(u);
        }
    }
    int siz = 0;
    for (int i = 1; i <= n; ++i)
        siz = max(siz, (int)g[i].size() + 1);
    cout << siz << "\n";
}
