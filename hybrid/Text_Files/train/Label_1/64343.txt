from sys import stdin
input = stdin.readline

n,m = map(int,input().split())
adj = [[] for _ in range(n+1)]
for _ in range(m):
    a,b,c = map(int,input().split())
    adj[a].append((b,c)); adj[b].append((a,c))

ans = 1
for i in range(1,n+1):
    if len(adj[i])%2==1: ans = 0
    for j in adj[i]:
        if j[1]==0: ans*=2
ans%=1000000007
print(ans)
