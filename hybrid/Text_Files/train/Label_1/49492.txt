import java.util.*;
import java.lang.*;

public class Main {
    final static double EPS = 1e-9;
    final static double CCW = 1e-9;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        ArrayList<Pair> P = new ArrayList<Pair>();
        for (int i = 0; i < N; ++i) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            P.add(new Pair(x, y));
        }
        Pair ans = get(P);
        if (ans == null) {
            System.out.println("NA");
        } else {
            System.out.println(ans.x + " " + ans.y);
        }
    }

    static Pair get(ArrayList<Pair> P) {
        int n = P.size();
        ArrayList<Double> areas = new ArrayList<Double>();
        areas.add(0.0);
        for (int i = 1; i < n; ++i) {
            Pair A = P.get(i - 1);
            Pair B = P.get(i);
            areas.add(areas.get(areas.size() - 1) + cross(A, B) / 2.0);
        }
        areas.add(areas.get(areas.size() - 1));
        if (Math.abs(areas.get(areas.size() - 1)) < EPS) {
            return null;
        }
        for (double s = 0.0; s < areas.get(areas.size() - 1); s += EPS) {
            Pair ans = binarySearch(P, areas, n, s);
            if (ans != null) {
                return ans;
            }
        }
        return null;
    }

    static Pair binarySearch(ArrayList<Pair> P, ArrayList<Double> areas, int n, double s) {
        int l = -1;
        int r = n;
        while (r - l > 1) {
            int mid = (l + r) / 2;
            if (areas.get(mid) <= s) {
                l = mid;
            } else {
                r = mid;
            }
        }
        if (Math.abs(areas.get(l) - s) < EPS) {
            return get(P, l, areas.get(l));
        } else if (Math.abs(areas.get(r) - s) < EPS) {
            return get(P, r, areas.get(r));
        } else {
            return null;
        }
    }

    static Pair get(ArrayList<Pair> P, int L, double A) {
        Pair A1 = new Pair(P.get(L).x, P.get(L).y);
        Pair B1 = new Pair(P.get((L + 1) % P.size()).x, P.get((L + 1) % P.size()).y);
        Pair C1 = new Pair(P.get((L + 2) % P.size()).x, P.get((L + 2) % P.size()).y);
        if (Math.abs(cross(A1, B1)) < EPS || Math.abs(cross(B1, C1)) < EPS) {
            return null;
        }
        double S = cross(B1, C1) / 2.0;
        double SA = cross(A1, B1) / 2.0;
        double SB = cross(B1, C1) / 2.0;
        Pair B2 = new Pair(B1.x + (C1.x - B1.x) * SA / SB, B1.y + (C1.y - B1.y) * SA / SB);
        double SC = (A - SA) / SB;
        Pair C2 = new Pair(B2.x + (C1.x - B2.x) * SC, B2.y + (C1.y - B2.y) * SC);
        return C2;
    }

    static double cross(Pair A, Pair B) {
        return A.x * B.y - A.y * B.x;
    }
}

class Pair {
    double x = 0.0;
    double y = 0.0;

    public Pair(double x, double y) {
        this.x = x;
        this.y = y;
    }
}
