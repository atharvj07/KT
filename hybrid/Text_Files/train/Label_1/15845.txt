#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(4);
  for (int j = 0; j < n; ++j) {
    for (int i = 0; i < 4; ++i) { cin >> a[i]; }
    if (a[0] && a[3]) { cout << "YES" << endl; return 0; }
    if (a[1] && a[3]) { cout << "YES" << endl; return 0; }
    if (a[2]) { a[2] = 0; }
  }
  cout << "NO" << endl;
  return 0;
}
