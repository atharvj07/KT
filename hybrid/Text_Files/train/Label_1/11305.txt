#include <iostream>

using namespace std;

int main() {
  long long int n, m, k, sum = 0, neg = 0;
  cin >> n >> m >> k;
  if (m % k == 0)
    sum = ((m / k) * (m / k + 1)) / 2;
  else
    sum = ((m / k) * (m / k + 1)) / 2 + ((m % k) * (m % k + 1)) / 2;
  neg = (sum % k * (sum % k + 1)) / 2;
  cout << (sum * sum - neg) % 1000000009;
}
