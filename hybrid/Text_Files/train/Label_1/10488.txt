#include<iostream>
#include<cstdio>
#include<bits/stdc++.h>
using namespace std;
string s;
int main()
{
	int n,p=0,z,x;
	cin>>n;
	cin>>s;
	for(int i=0;i<n;i++)
	{
		if(s[i]=='8')
			x=i;
		if(s[i]=='0')
			z++;
	}
	if(n<11)
		goto p;
	if(x==0)
		goto p;
	if(z<1)
		goto p;
	p=1;
	z--;
	for(int i=x+1;i<x+11;i++)
		if(s[i]=='0')
			z--;
	if(z<0)
		goto p;
	p=2;
	p++;
p:;
	cout<<p<<endl;
	return 0;
}
