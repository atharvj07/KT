import sys
input = sys.stdin.readline

b1, q, l, m = map(int, input().split())
a = list(map(int, input().split()))
if l == 0:
    print(0)
elif q != 0:
    c = (l - abs(b1)) // abs(q) + 1
    if any(l >= abs(b1 + q * i) for i in a):
        print(c - 1)
    else:
        print(c)
else:
    print("inf")
