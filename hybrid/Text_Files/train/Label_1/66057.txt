import sys
fr=sys.stdin
nl=[ int(x) for x in fr.readline().split()]
n,m=nl[0],nl[1]
fav=[[int(x) for x in fr.readline().split()] for i in range(n)]
piz=[[int(x) for x in fr.readline().split()] for i in range(m)]
for x in fav:
    for y in x:
        for z in piz:
            if y in z:
                z[0]-=1
                break
for x in piz:
    x[0]=x[0]==0
piz.sort()
j1=-1
j2=-1
for i in range(m):
    if piz[i][0]==1:
        if j1==-1: j1=i
        else:
            j2=i
            break
print(j1+1,j2+1)
