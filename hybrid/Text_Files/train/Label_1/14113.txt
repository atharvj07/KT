#include <bits/stdc++.h>
using namespace std;
void solve()
{
    int n, x, y;
    cin >> n >> x >> y;
    int ans[n];
    if (n == 2)
    {
        cout << x << " " << y << endl;
        return;
    }
    int d = (y - x) / (n - 1);
    ans[0] = x;
    ans[n - 1] = y;
    for (int i = 1; i < n - 1; i++)
    {
        ans[i] = x + (d * i);
    }
    for (int i = 0; i < n; i++)
    {
        cout << ans[i] << " ";
    }
    cout << endl;
}
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
