#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    vector<int> cnt(n);
    for (int i = 1; i < n - 1; i++) {
      if (s[i - 1] == '1') cnt[i]++;
      if (s[i + 1] == '1') cnt[i]++;
    }
    while (m--) {
      string t = s;
      for (int i = 1; i < n - 1; i++) {
        if (cnt[i] == 1 && s[i] == '0')
          t[i] = '1';
        else if (cnt[i] > 1 || s[i] == '1')
          t[i] = '1';
        else
          t[i] = '0';
      }
      s = t;
      for (int i = 1; i < n - 1; i++) {
        if (s[i - 1] == '1') cnt[i]++;
        if (s[i + 1] == '1') cnt[i]++;
        if (s[i] == '0') cnt[i] = 0;
      }
    }
    cout << s << endl;
  }
  return 0;
}
