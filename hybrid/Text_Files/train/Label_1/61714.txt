def min_plots_to_clean():
    x_a, y_a = map(int, input().split())
    x_b, y_b = map(int, input().split())
    x_c, y_c = map(int, input().split())

    x_min, x_max = min(x_a, x_b, x_c), max(x_a, x_b, x_c)
    y_min, y_max = min(y_a, y_b, y_c), max(y_a, y_b, y_c)

    plots = set()
    for x in range(x_min, x_max + 1):
        plots.add((x, y_min))
        plots.add((x, y_max))
    for y in range(y_min, y_max + 1):
        plots.add((x_min, y))
        plots.add((x_max, y))

    # Ensure the plots A, B, C are included (they should be already, but just in case)
    plots.add((x_a, y_a))
    plots.add((x_b, y_b))
    plots.add((x_c, y_c))

    # Remove duplicates by converting to a set and back to a list
    unique_plots = list(plots)

    print(len(unique_plots))
    for plot in unique_plots:
        print(plot[0], plot[1])

min_plots_to_clean()
