from collections import Counter

n = int(input())
a = list(map(int, input().split()))
a.sort()
q = int(input())
s = Counter(a)
for _ in range(q):
    t, x = input().split()
    x = int(x)
    if t == '+':
        s[x] += 1
    elif s[x] > 0:
        s[x] -= 1
    else:
        print('NO')
        continue
    if len(s) < 4:
        print('NO')
        continue
    m = list(s.values())
    m.sort()
    if m[0] == m[1] and m[2] == m[3]:
        print('YES')
    elif m[2] == m[3] * 2:
        print('YES')
    else:
        print('NO')
