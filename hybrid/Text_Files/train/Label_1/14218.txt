#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll a[200005]={0},s[200005]={0},l[200005]={0},r[200005]={0},d[200005]={0},f[200005]={0};
int main()
{
    ll m,n,k,t,i,j,ans=0,x=0,y=0,z=0;
    cin>>m>>n>>k>>t;
    for(i=1;i<=m;i++)cin>>a[i];
    for(i=1;i<=k;i++)
    {
        cin>>l[i]>>r[i]>>d[i];
        f[r[i]]=1;
    }
    sort(a+1,a+1+m);
    for(i=1;i<=m;i++)
    {
        if((!i||a[i]>a[i-1])&&a[i]<=t&&(!f[x+1]||a[i]>d[f[x+1]]))
        {
            s[++z]=x+1;
            x+=2;
        }
        if((!i||a[i]>a[i-1])&&a[i]<=t&&(!f[y+1]||a[i]>d[f[y+1]]))
        {
            s[++z]=y+1;
            y+=1;
        }
    }
    sort(s+1,s+1+z);
    for(i=1;i<=z;i++)
    {
        if(!i||s[i]>s[i-1])
        {
            ans++;
        }
    }
    cout<<ans<<endl;
    return 0;
}
