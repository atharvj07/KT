#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, m, d, h;
    cin >> n >> m;
    priority_queue<int> pq;
    priority_queue<int, vector<int>, greater<int>> pq2;
    while (m--) {
        cin >> d >> h;
        pq.push(h - d);
        pq2.push(h + d);
    }
    while (pq.size() > 1) {
        int x = pq.top();
        pq.pop();
        int y = pq2.top();
        pq2.pop();
        if (x + 1 != y) { cout << "IMPOSSIBLE" << endl; return 0; }
    }
    if (pq.size() == 1) cout << pq.top() << endl;
    else cout << "IMPOSSIBLE" << endl;
    return 0;
}
