import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long k = sc.nextLong();

        long low = 1;
        long high = n * m;
        while (low < high) {
            long mid = (low + high) / 2;
            long count = 0;
            for (long i = 1; i <= n; i++) {
                count += Math.min(mid / i, m);
            }
            if (count < k) {
                low = mid + 1;
            } else {
                high = mid;
            }
        }
        System.out.println(low);
    }
}
