#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const int N = 1e2+5;
int a[N], sz[N], adj[N], used[N];

struct DSU {
    void init(int n) {
        for (int i = 0; i < n; i++) {
            a[i] = i;
            sz[i] = 1;
            adj[i]=i;
        }
    }
    int root(int x) {
        while (a[x] != x) {
            a[x] = a[a[x]];
            x = a[x];
        }
        return x;
    }
    void unite(int x, int y) {
        x = root(x); 
        y = root(y); 
        if (x == y) 
            return;
        if (sz[x] < sz[y]) 
            swap(x, y);
        sz[x] += sz[y];
        a[y] = x;
    }
};

DSU dsu;
int main() {
    int n, m, q; cin >> n >> m;
    dsu.init(n);
    while (m--) {
        int x, y, c; cin >> x >> y >> c;
        --x, --y, --c;
        if (dsu.root(x) != dsu.root(y)) {
            dsu.unite(dsu.root(x), dsu.root(y));
            used[c] = 1;
            adj[c] = dsu.root(x);
        }
    }
    cin >> q;
    while (q--) {
        int u, v; cin >> u >> v;
        --u, --v;
        int ans = 0;
        for (int i = 0; i < N; i++) {
            if (!used[i]) continue;
            ans += !adj[i] || (dsu.root(u) == adj[i] && dsu.root(v) == adj[i]);
        }
        cout << ans << "\n";
    }
    return 0;
}
