import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        Map<Character, Integer> ranks = new HashMap<>();
        ranks.put('6', 1);
        ranks.put('7', 2);
        ranks.put('8', 3);
        ranks.put('9', 4);
        ranks.put('T', 5);
        ranks.put('J', 6);
        ranks.put('Q', 7);
        ranks.put('K', 8);
        ranks.put('A', 9);
        char trump = in.next().charAt(0);
        String card1 = in.next();
        char rank1 = card1.charAt(0);
        char suit1 = card1.charAt(1);
        String card2 = in.next();
        char rank2 = card2.charAt(0);
        char suit2 = card2.charAt(1);
        if (suit1 == trump && suit2 != trump) {
            System.out.println("YES");
        } else if (suit1 == suit2 && ranks.get(rank1) > ranks.get(rank2)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
