import java.util.Scanner;
import java.util.BitSet;
import java.util.TreeSet;
import java.util.Set;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int m = s.nextInt();
        int[] row = new int[n];
        int[] col = new int[n];
        int max = 0;
        Set<Integer> ban = new TreeSet<>();
        for (int i = 0; i < m; i++) {
            ban.add(s.nextInt());
            ban.add(s.nextInt() + n);
        }
        for (int i = 0; i < n - 1; i++) {
            BitSet r = new BitSet(n);
            BitSet c = new BitSet(n);
            int rCnt = 0;
            int cCnt = 0;
            for (int j = 1; j < n; j++) {
                int rr = row[j] - i - 1;
                int cc = col[j] - i - 1;
                if (rr >= 0 && !ban.contains(rr) && !r.get(rr)) {
                    rCnt++;
                    r.set(rr);
                }
                if (cc >= 0 && !ban.contains(cc + n) && !c.get(cc)) {
                    cCnt++;
                    c.set(cc);
                }
            }
            max = Math.max(max, Math.min(rCnt, cCnt));
        }
        System.out.println(max);
    }
}
