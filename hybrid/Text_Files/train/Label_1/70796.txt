n,m=map(int,input().split())
q=int(input())
a=[list(map(int,input().split()))for i in range(q)]
s=[1000000000]*(n+1)
s[1]=0
for aa in a:
    s[aa[0]]=min(s[aa[0]],s[aa[1]-1]+1)
for bb in a:
    s[bb[1]]=min(s[bb[1]],s[bb[0]]+1)
for i in range(1,n+1):
    s[i]=min(s[i],s[i-1]+1)
ans=s[n]
for aa in a:
    if aa[0]==n and s[n]==s[aa[1]-1]+1: ans-=1
for bb in a:
    if bb[1]==m and s[m]==s[bb[0]]+1: ans-=1
print(ans+1)
