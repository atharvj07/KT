import java.util.Arrays;
import java.util.Scanner;

public class VovaAndTests {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[k];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        Arrays.sort(marks);

        int remainingMarks = n - k;
        int remainingSum = x - Arrays.stream(marks).sum();
        int remainingMedian = (n + 1) / 2 - k;

        if (remainingSum < 0 || remainingMedian < 0) {
            System.out.println("-1");
            return;
        }

        int[] remainingTargetMarks = new int[remainingMarks];
        int targetMedian = Math.max(y, marks[k - remainingMedian]);

        for (int i = 0; i < remainingMarks; i++) {
            remainingTargetMarks[i] = Math.min(p, targetMedian);
            targetMedian = Math.min(targetMedian + 1, p);
        }

        for (int mark : remainingTargetMarks) {
            System.out.print(mark + " ");
        }
    }
}
