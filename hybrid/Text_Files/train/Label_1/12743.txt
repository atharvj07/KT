#include <bits/stdc++.h>
using namespace std;
#define mp make_pair
#define ll long long
#define pb push_back
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v;
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    v.pb(mp(a, b));
  }
  sort(v.begin(), v.end());
  int e1 = -1, b1 = -1, e2 = -1, b2 = -1;
  for (auto p : v) {
    int l = p.first, r = p.second;
    if (l >= b1 && l >= b2) {
      if (b1 > b2) {
        b2 = r;
      } else {
        b1 = r;
      }
    } else {
      cout << "NO\n";
      return 0;
    }
  }
  cout << "YES\n";
  return 0;
}
