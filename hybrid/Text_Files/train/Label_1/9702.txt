#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5 + 5;
int n, b[MAXN];
long long dp[MAXN];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> b[i];
    for (int i = n; i > 0; i--) {
        dp[i] = b[i];
        for (int j = i + b[i]; j <= n; j++) {
            dp[i] = max(dp[i], dp[j] + b[i]);
        }
    }
    cout << *max_element(dp + 1, dp + n + 1) << "\n";
    return 0;
}
