#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mxN = 105;
ll n, a[mxN], b[mxN], p[mxN];
int main() {
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) cin >> a[i] >> b[i];
  for (int i = 0; i < n; i++) {
    p[i + 1] = p[i];
    if (s[i] == '1' || (p[(b[i] - 1 + a[i]) % a[i]] > 0 && p[(b[i] - 1 + 2 * a[i]) % a[i]] > 0)) p[i + 1]++;
  }
  cout << *max_element(p, p + n + 1) << endl;
}
