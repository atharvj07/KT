import java.util.*;

public class Main {
    static int N;
    static String S;
    static int[] a, b;
    static ArrayList<Integer>[] adj;
    static int[] depth;
    static int[] cnt;
    static int ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        S = sc.next();
        a = new int[N - 1];
        b = new int[N - 1];
        adj = new ArrayList[N];
        for (int i = 0; i < N; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < N - 1; i++) {
            a[i] = sc.nextInt() - 1;
            b[i] = sc.nextInt() - 1;
            adj[a[i]].add(b[i]);
            adj[b[i]].add(a[i]);
        }
        depth = new int[N];
        cnt = new int[N];
        dfs(0, -1, 0);
        for (int i = 0; i < N; i++) if (S.charAt(i) == '1') cnt[depth[i]]++;
        ans = Integer.MAX_VALUE;
        for (int i = 0; i < N; i++) {
            if (cnt[i] > 0) {
                int tmp = 0;
                for (int j = 0; j < N; j++) {
                    if (cnt[j] > 0) {
                        tmp += Math.abs(i - j) * cnt[j];
                    }
                }
                ans = Math.min(ans, tmp);
            }
        }
        if (ans == Integer.MAX_VALUE) ans = -1;
        System.out.println(ans);
    }

    static void dfs(int v, int p, int d) {
        depth[v] = d;
        for (int u : adj[v]) {
            if (u != p) dfs(u, v, d + 1);
        }
    }
}
