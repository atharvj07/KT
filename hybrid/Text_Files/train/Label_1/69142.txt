s = input()
st = ""
ans = 0
for i in s:
    if i>='A' and i<='Z':
        st = i
        st = st.lower()
        if st=="bulbasaur":
            ans += 1
            st = ""
            continue
    else:
        st += i
        st = st.lower()
        if st=="bulbasaur":
            ans += 1
            st = ""

print(ans)
