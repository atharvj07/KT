MOD = 10**9 + 7

def add(a, b):
    return (a + b) % MOD

def solve(n, p, colors):
    dp = [[[0, 0] for _ in range(n + 1)] for __ in range(2)]
    dp[0][0][0] = 1

    black, white = colors.count(0), colors.count(1)
    uncolored = colors.count(-1)

    for i in range(1, n + 1):
        c = colors[i - 1]
        for j in range(i + 1):
            for k in range(2):
                if c != 1:
                    dp[i % 2][j][k] = add(dp[i % 2][j][k], dp[(i - 1) % 2][j][k] * (1 if j == 0 else j))
                    if j > 0:
                        dp[i % 2][j][k ^ 1] = add(dp[i % 2][j][k ^ 1], dp[(i - 1) % 2][j - 1][k] * (i - j))
                if c != 0:
                    dp[i % 2][j][k] = add(dp[i % 2][j][k], dp[(i - 1) % 2][j][k] * (i - j))
                    dp[i % 2][j + 1][k ^ 1] = add(dp[i % 2][j + 1][k ^ 1], dp[(i - 1) % 2][j][k] * (j + 1))
        for j in range(i + 1):
            dp[(i - 1) % 2][j][0] = dp[(i - 1) % 2][j][1] = 0

    ans = 0
    for i in range(n + 1):
        for j in range(2):
            if (black + white + j) % 2 == p:
                ways = dp[n % 2][i][j]
                for k in range(uncolored - j):
                    ways = ways * (n - k) % MOD
                ans = add(ans, ways)

    return ans

n, p = map(int, input().split())
colors = list(map(int, input().split()))
print(solve(n, p, colors))
