import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] arr = new int[4];
        for (int i = 0; i < 4; i++) {
            arr[i] = input.nextInt();
        }
        Arrays.sort(arr);
        System.out.println((arr[3] - arr[2]) + " " + (arr[3] - arr[1]) + " " + (arr[3] - arr[0]));
    }
}
