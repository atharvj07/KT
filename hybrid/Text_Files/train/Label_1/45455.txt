import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int d = sc.nextInt();
            if (d == 0) {
                break;
            }
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] dist = new int[n];
            for (int i = 1; i < n; i++) {
                dist[i] = sc.nextInt();
            }
            int sum = 0;
            for (int i = 0; i < m; i++) {
                int k = sc.nextInt();
                int min = d;
                for (int j = 1; j < n; j++) {
                    min = Math.min(min, Math.min(Math.abs(k - dist[j]), d - Math.abs(k - dist[j])));
                }
                sum += min;
            }
            System.out.println(sum);
        }
    }
}
