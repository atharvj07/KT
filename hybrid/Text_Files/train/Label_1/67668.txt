def days_to_finish_book(c, v0, v1, a, l):
    days = 0
    pages_read = 0
    while pages_read < c:
        if days > 0:
            pages_read -= l
        pages_today = min(v0 + days * a, v1)
        pages_read += pages_today
        days += 1
    return days

c, v0, v1, a, l = map(int, input().split())
print(days_to_finish_book(c, v0, v1, a, l))
