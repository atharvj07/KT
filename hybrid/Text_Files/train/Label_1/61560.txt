n,k=map(int,input().split())
a=list(map(int,input().split()))
count, ans, mini = 0, 0, n
for i in range(1,k+1):
    x = a.count(i)
    if x == 1:
        ans = i
        break
    count += x-1
    if x < mini:
        mini = x
        ans = i
if count == 0:
    for i in range(1,k+1):
        x = a.count(i)
        if x > mini:
            ans = i
            break
print(ans)
