#include <iostream>
typedef long long ll;
using namespace std;
int main() {
    ll k, a, b;
    cin >> k >> a >> b;
    int Aa[3][3], Ba[3][3];
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++) cin >> Aa[i][j];
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++) cin >> Ba[i][j];
    int score[2] = {a, b}, tmp[2];
    while (k--) {
        tmp[1] = Ba[score[0] - 1][score[1] - 1];
        tmp[0] = Aa[score[1] - 1][tmp[1] - 1];
        score[0] = tmp[0];
        score[1] = tmp[1];
    }
    cout << score[0] - 1 << " " << score[1] - 1 << endl;
    return 0;
}
