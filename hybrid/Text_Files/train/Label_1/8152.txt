#include <bits/stdc++.h>
using namespace std;
long long a, b, ans, l;
int main() {
    cin >> a >> b;
    for (long long i = 0; i <= b - a; i++) {
        long long c = __gcd(a + i, b + i);
        if (i == 0)
            l = c;
        else
            l = __gcd(l, c);
    }
    for (long long i = 0; i <= b - a; i++) {
        long long c = __gcd(a + i, b + i);
        if (c == l) {
            ans = i;
            break;
        }
    }
    cout << ans;
}
