import sys
a,b,c,d,e,f,g,h=map(int,sys.stdin.readline().split())
i,j,k,l,m,n,o,p=map(int,sys.stdin.readline().split())

min1=min(a,b,c,d)
max1=max(a,b,c,d)
min2=min(e,f,g,h)
max2=max(e,f,g,h)
min3=min(i,j,k,l)
max3=max(i,j,k,l)
min4=min(m,n,o,p)
max4=max(m,n,o,p)

if not (max1 < min2 or max2 < min1 or max3 < min4 or max4 < min3): print("YES")
else: print("NO")
