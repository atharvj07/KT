import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();

    char[] s = sc.next().toCharArray();

    boolean[] alive = new boolean[n];
    for (int i = 0; i < n; i++) {
      alive[i] = true;
    }

    for (int i = 0; i < q; i++) {
      char t = sc.next().charAt(0);
      char d = sc.next().charAt(0);
      for (int j = 0; j < n; j++) {
        if (s[j] == t) {
          if ((j == 0 && d == 'L') || (j == n - 1 && d == 'R')) {
            alive[j] = false;
          } else if (d == 'L') {
            alive[j - 1] = alive[j];
            alive[j] = false;
          } else {
            alive[j] = alive[j + 1];
            alive[j + 1] = false;
          }
        }
      }
    }

    int ans = 0;
    for (boolean b : alive) {
      if (b) {
        ans++;
      }
    }

    System.out.println(ans);
  }
}
