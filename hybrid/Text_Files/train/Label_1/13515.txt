#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    int l[m];
    for (int i = 0; i < m; i++) {
        cin >> l[i];
    }
    vector<int> a(n, 0), vis(n, 0);
    a[0] = l[0];
    vis[a[0] - 1] = 1;
    for (int i = 1; i < m; i++) {
        int cur = l[i], cnt = 0;
        while (cnt < a[cur - 1]) {
            cur++;
            if (cur > n) cur = 1;
            cnt++;
        }
        a[i] = cur;
        vis[cur - 1] = 1;
    }
    if (l[m - 1] != a[0]) {
        cout << -1 << endl;
    } else {
        for (int i = 0; i < m; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
