#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }

  int cnt_petya = 0, cnt_gena = 0;
  std::vector<int> ans;
  for (int t = 1; t <= n; ++t) {
    cnt_petya = 0, cnt_gena = 0;
    int st = 0;
    for (int i = 0; i < n; ++i) {
      if (a[i] == 1) {
        ++cnt_petya;
      } else {
        ++cnt_gena;
      }
      if (cnt_petya == t || cnt_gena == t) {
        ++st;
        cnt_petya = 0;
        cnt_gena = 0;
      }
    }
    if (st > 0) {
      ans.push_back(st);
      ans.push_back(t);
    }
  }

  int k = ans.size() / 2;
  std::cout << k << '\n';
  for (int i = 0; i < k; ++i) {
    std::cout << ans[2 * i] << ' ' << ans[2 * i + 1] << '\n';
  }

  return 0;
}
