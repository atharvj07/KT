n, m = map(int, input().split())
x = [0]*m; y = [0]*m
for i in range(0, m):
	u, v, a, b = map(int, input().split())
	x[i] = [u, a]
	y[i] = [v, b]
x = sorted(x, key = lambda i: (i[1], i[0]))
y = sorted(y, key = lambda i: (i[1], i[0]))
ans = 1e9
for i in range(0, m):
	for j in range(i+1, m):
		if(x[i][1] >= y[j][0] and x[j][1] >= y[i][0]):
			ans = min(ans, x[j][1]-y[i][0] + y[j][1]-x[i][0])
print(ans)
