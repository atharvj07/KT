#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define inf LLONG_MAX
ll n,m,s,a[200005],vis[200005];
vector<pair<ll,ll>>edge[200005];
void dfs(ll node,ll par,ll sum)
{
    if(vis[node]!=0)return;
    vis[node]=1;
    sum+=a[node];
    if(sum>a[s])a[s]=sum;
    for(auto &x:edge[node])
    {
        if(x.first!=par)dfs(x.first,node,sum);
    }
}
int main()
{
    cin>>n>>m;
    for(ll i=1;i<=n;i++)cin>>a[i];
    for(ll i=1;i<=m;i++)
    {
        ll x,y;cin>>x>>y;
        edge[x].push_back({y,0});
        edge[y].push_back({x,0});
    }
    cin>>s;
    dfs(s,s,0);
    cout<<a[s]<<'\n';
    return 0;
}
