import java.util.Scanner;

public class Main {
    static int n, k, a, b, c, d;
    static int[] p = new int[1001];

    static void dfs1(int x, int fa) {
        System.out.print(x + " ");
        p[x] = fa;
        if (x == b) return;
        for (int i = 1; i <= n; i++) {
            if (i == fa) continue;
            dfs1(i, x);
        }
    }

    static void dfs2(int x, int fa) {
        System.out.print(x + " ");
        p[x] = fa;
        if (x == d) return;
        for (int i = 1; i <= n; i++) {
            if (i == fa || i == p[fa]) continue;
            dfs2(i, x);
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        k = in.nextInt();
        a = in.nextInt();
        b = in.nextInt();
        c = in.nextInt();
        d = in.nextInt();
        if (k > n - 1 || k < n - 3) {
            System.out.println(-1);
            return;
        }
        dfs1(a, 0);
        System.out.println();
        dfs2(c, 0);
    }
}
