from sys import stdin
from heapq import heappop, heappush
import bisect

n, k = map(int, stdin.readline().split())
s = stdin.readline().strip()
s += '$'
m = bisect.bisect_left(s, '$')

ans = 0
l = []
for c in s:
    if c == '$':
        break
    cnt = 0
    while l and l[-1][1] == cnt:
        cnt += 1
        l.pop()
    ans += cnt
    heappush(l, (len(s) - m - cnt - 1, cnt + 1))

if len(l) < k:
    print(-1)
else:
    for _ in range(k - len(l)):
        t = heappop(l)
        ans -= t[0] + 1
        t = (t[0], t[1] - 1)
        if t[1] > 0:
            heappush(l, t)
    print(ans)
