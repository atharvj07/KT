import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] c = new int[n * k];
        for (int i = 0; i < n * k; i++) {
            c[i] = scanner.nextInt();
        }
        int[] f = new int[n];
        for (int i = 0; i < n; i++) {
            f[i] = scanner.nextInt();
        }
        int[] h = new int[k];
        for (int i = 0; i < k; i++) {
            h[i] = scanner.nextInt();
        }
        int[] cnt = new int[100001];
        for (int i = 0; i < n * k; i++) {
            cnt[c[i]]++;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int cur = 0;
            for (int j = 1; j <= k; j++) {
                if (cnt[f[i]] >= j) {
                    cur = Math.max(cur, h[j - 1]);
                }
            }
            ans += cur;
        }
        System.out.println(ans);
    }
}
