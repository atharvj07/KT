import java.util.Scanner;

public class VitalyaAndNight {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] alarms = new int[n];
        for (int i = 0; i < n; i++) {
            alarms[i] = scanner.nextInt();
        }

        int[] count = new int[1000001];
        for (int alarm : alarms) {
            count[alarm]++;
        }

        int maxCount = 0;
        int maxEnd = 0;
        for (int i = 0; i <= 1000000 - m; i++) {
            int currentCount = 0;
            for (int j = i; j < i + m; j++) {
                currentCount += count[j];
            }
            if (currentCount > maxCount) {
                maxCount = currentCount;
                maxEnd = i + m - 1;
            }
        }

        int minTurnedOff = 0;
        for (int i = 0; i < n; i++) {
            if (alarms[i] >= maxEnd - m + 1 && alarms[i] <= maxEnd) {
                minTurnedOff++;
            }
        }

        System.out.println(minTurnedOff);
    }
}
