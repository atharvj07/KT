#include <iostream>
using namespace std;
long long gcd(long long a, long long b) {
    if (b == 0)
        return a;
    return gcd(b, a % b);
}
int main() {
    long long n;
    cin >> n;
    long long ans = n * (n - 1) * (n - 2);
    long long g = gcd(ans, n - 3);
    ans = ans / g * (n - 3);
    cout << ans << endl;
    return 0;
}
