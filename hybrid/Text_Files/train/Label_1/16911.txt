#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int a[105], b[105];
int main() {
  int t;
  cin >> t;
  for (int i = 1; i <= t; i++) {
    int n, u, v;
    cin >> n >> u >> v;
    for (int j = 1; j <= n; j++) {
      cin >> a[j];
      b[j] = a[j];
    }
    int ans = 0;
    for (int j = 1; j <= n; j++) {
      if (a[j] == 1) {
        ans += v;
        a[j] = a[j + 1];
        a[j + 1] = b[j + 1];
      } else if (a[j] == n) {
        ans += v;
        a[j] = a[j - 1];
        a[j - 1] = b[j - 1];
      } else {
        if (a[j - 1] < a[j + 1]) {
          ans += u;
          a[j] = a[j - 1];
          a[j - 1] = b[j - 1];
        } else {
          ans += u;
          a[j] = a[j + 1];
          a[j + 1] = b[j + 1];
        }
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
