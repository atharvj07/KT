import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            p[find(p, b)] = find(p, a);
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            System.out.println(find(p, a) == find(p, b) ? "yes" : "no");
        }
    }

    public static int find(int[] p, int x) {
        if (p[x] == x) {
            return x;
        }
        return p[x] = find(p, p[x]);
    }
}
