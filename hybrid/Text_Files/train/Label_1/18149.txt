#include <iostream>
#include <string>

int main() {
  std::string seat;
  std::cin >> seat;
  uint64_t row = std::stoull(seat.substr(0, seat.size() - 1));
  char pos = seat[seat.size() - 1];
  uint64_t time = 0;
  if (row % 4 == 0) { // flight attendant 1
    time += row - 1;
    if (pos == 'a' || pos == 'b') {
      time += 3;
    } else if (pos == 'd' || pos == 'e') {
      time += 1;
    }
  } else if (row % 4 == 1) { // flight attendant 2
    time += row - 1;
    if (pos == 'd' || pos == 'e') {
      time += 3;
    } else if (pos == 'a' || pos == 'b') {
      time += 1;
    }
  } else if (row % 4 == 2) { // flight attendant 1
    time += row;
    if (pos == 'c' || pos == 'd') {
      time += 3;
    } else if (pos == 'a' || pos == 'b') {
      time += 1;
    }
  } else { // row % 4 == 3 // flight attendant 2
    time += row;
    if (pos == 'a' || pos == 'b') {
      time += 3;
    } else if (pos == 'c' || pos == 'd') {
      time += 1;
    }
  }
  std::cout << time << std::endl;
  return 0;
}
