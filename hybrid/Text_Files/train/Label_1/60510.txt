MOD = 10**9 + 7

def power(a, b, mod):
    result = 1
    a = a % mod
    while b > 0:
        if b & 1:
            result = (result * a) % mod
        b = b >> 1
        a = (a * a) % mod
    return result

def solve(A, B, n, x):
    if A == 1:
        return (x + B * n) % MOD
    return (power(A, n, MOD) * x + B * (power(A, n, MOD) - 1) * power(A - 1, MOD - 2, MOD)) % MOD

A, B, n, x = map(int, input().split())
print(solve(A, B, n, x))
