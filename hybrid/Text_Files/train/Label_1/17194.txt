#include<bits/stdc++.h>
using namespace std;
int main() {
    int i, a, b, n, m, cnt = 0;
    cin >> n >> m;
    vector<int>adj[n + 1];
    for (i = 0; i < m; i++) {
        cin >> a >> b;
        adj[a].push_back(b);
    }
    for (i = 1; i <= n; i++) {
        for (int u : adj[i]) {
            for (int v : adj[u]) {
                if (i == v || u == v)continue;
                for (int w : adj[v]) {
                    if (w == u)
                        cnt++;
                }
            }
        }
    }
    cout << cnt / 2 << endl;
    return 0;
}
