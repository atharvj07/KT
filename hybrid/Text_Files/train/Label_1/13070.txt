#include <iostream>
#include <vector>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long int h, n;
    cin >> h >> n;
    vector<long long int> p(n);
    for (long long int i = 0; i < n; i++) { cin >> p[i]; }
    long long int ans = 0;
    long long int last = h + 1;
    for (long long int i = n - 1; i >= 0; i--) {
      if (last - p[i] > 2) {
        ans += (last - p[i] - 1) / 2;
        last = p[i] + 1;
      }
      last--;
    }
    if (last > 2) { ans += (last - 2 + 1) / 2; }
    cout << ans << endl;
  }
  return 0;
}
