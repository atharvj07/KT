import sys
input=lambda:sys.stdin.readline().strip()
n,m,k=map(int,input().split())
a=list(map(int,input().split()))
ans=0
l,r,t=0,0,0
for i in a:
    if t==0:
        l=i
        t=1
    elif i==r+1:
        r+=1
    else:
        if r-l+1>k:
            ans+=r-l-k
            l=i
            r=i
        else:
            r=i
    t=1
if r-l+1>k:
    ans+=r-l-k
print(ans)
