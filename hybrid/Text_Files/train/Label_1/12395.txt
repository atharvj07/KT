#include <iostream>

using namespace std;

bool solve(long long l, long long r) {
  if (r % 2 == l % 2) return false;
  if ((l % 2 == 1 && l + 1 <= r) || (l % 2 == 0 && l + 2 <= r)) return true;
  return false;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long l, r;
    cin >> l >> r;
    cout << (solve(l, r) ? "YES" : "NO") << '\n';
  }
  return 0;
}
