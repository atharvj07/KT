MOD = 10**9 + 7

def count_polynomials(t, a, b):
    if a == b:
        if t == 1 or a == 0:
            return 1
        if a == 1:
            return 2
        return 0
    if a == 0 or t == 1:
        return 0
    if t == 0:
        if b == 0:
            return 1 if a == 1 else 0
        return 0
    if a == 1:
        return 0
    
    res = 0
    while b % a == 0:
        b //= a
        x = b
        cnt = 0
        while x:
            x //= t
            cnt += 1
        if pow(t, cnt, a) == b % a:
            res += 1
        if a == 1:
            break
    
    return res % MOD

t, a, b = map(int, input().split())
print(count_polynomials(t, a, b))
