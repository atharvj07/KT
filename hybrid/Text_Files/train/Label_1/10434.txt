#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int l = 1, r = 1e9;
  while (l < r) {
    int m = (l + r) / 2;
    int sum = 0;
    for (int i = 0; i < n; i++) {
      int h = min(m / a[i], a[i]);
      sum += h;
    }
    if (sum <= k) {
      l = m + 1;
    } else {
      r = m;
    }
  }
  cout << l - 1 << endl;
  return 0;
}
