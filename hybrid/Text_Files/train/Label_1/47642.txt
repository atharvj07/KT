import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] robots = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                robots[i][j] = sc.nextInt();
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (Math.abs(robots[i][2] - robots[j][2]) <= k && robots[i][0] - robots[i][1] <= robots[j][0] && robots[i][0] + robots[i][1] >= robots[j][0]) {
                    ans++;
                } else if (Math.abs(robots[i][2] - robots[j][2]) <= k && robots[j][0] - robots[j][1] <= robots[i][0] && robots[j][0] + robots[j][1] >= robots[i][0]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
