n, k, m = map(int, input().split())
a = list(map(int, input().split()))

# Count the occurrences of each city in one trip
city_count = {}
for city in a:
    if city in city_count:
        city_count[city] += 1
    else:
        city_count[city] = 1

# Calculate the remaining participants for each city after forming teams
remaining = {}
for city, count in city_count.items():
    remaining[city] = count % k

# Calculate the total remaining participants after all trips
total_remaining = sum(remaining.values()) * m

# Adjust for the case where participants from the same city can form additional teams across trips
for city, count in remaining.items():
    if count > 0:
        # Calculate the total number of participants from this city
        total_city_participants = city_count[city] * m
        # Calculate how many full teams can be formed from the remaining participants of this city across all trips
        full_teams = total_city_participants // k
        # Calculate the remaining participants after forming full teams
        final_remaining = total_city_participants - full_teams * k
        # Update the total remaining participants
        total_remaining = total_remaining - (count * m) + final_remaining

print(total_remaining)
