#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int cnt[MAXN];
bool ok() {
  int a = -1, b = -1, c = -1, d = -1;
  for (int i = 1; i < MAXN; i++) {
    if (cnt[i] > 0) {
      if (a == -1) a = cnt[i];
      else if (b == -1) b = cnt[i];
      else if (c == -1) c = cnt[i];
      else if (d == -1) d = cnt[i];
    }
    if (a != -1 && b != -1 && c != -1 && d != -1) break;
  }
  if (a == -1 || b == -1 || c == -1 || d == -1) return false;
  return a == b && b == c && c == d;
}
int main() {
  int n, x, q;
  char op;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> x;
    cnt[x]++;
  }
  cin >> q;
  while (q--) {
    cin >> op >> x;
    if (op == '+')
      cnt[x]++;
    else
      cnt[x]--;
    if (ok())
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
