#include<bits/stdc++.h>
using namespace std;
int main() {
	int n, p, q;
	cin >> n >> p >> q;
	string s;
	cin >> s;
	for (int i = 0; i < n % (p + q); i++) {
		if (p + q - i % (p + q) != p) {
			cout << -1;
			return 0;
		}
	}
	cout << n / (p + q);
	for (int i = 0; i < n; i += p + q) {
		if (i + p <= n)cout << endl << s.substr(i, p);
		else if (i + q <= n)cout << endl << s.substr(i, q);
	}
}
