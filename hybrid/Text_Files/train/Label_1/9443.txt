#include <vector>
#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &ai : a) cin >> ai;
    int m;
    cin >> m;
    vector<int> q(m);
    for (int &qi : q) cin >> qi;

    vector<int> p(a.back() + 1, -1);
    for (int i = 0; i < n; i++) {
        for (int j = a[i]; j > 0; j--) {
            if (p[j] == -1) p[j] = i + 1;
        }
    }

    for (int qi : q) {
        cout << p[qi] << endl;
    }
}
