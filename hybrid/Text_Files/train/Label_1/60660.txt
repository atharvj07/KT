r, c, n, k = map(int, input().split())
violas = [tuple(map(int, input().split())) for _ in range(n)]
ans = 0
for vio_x1, vio_y1 in violas:
  for vio_x2, vio_y2 in violas:
    if vio_x1 <= vio_x2 and vio_y1 <= vio_y2:
      # print(vio_x1, vio_y1, vio_x2, vio_y2)
      cnt = 0
      for x, y in violas:
        if vio_x1 <= x <= vio_x2 and vio_y1 <= y <= vio_y2:
          cnt += 1
      if cnt >= k:
        ans += (vio_x2-vio_x1+1)*(vio_y2-vio_y1+1)
print(ans)
