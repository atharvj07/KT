#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5+5;
ll a[N];
int main(){
  ll t, n, k;
  cin >> t;
  while(t--){
    cin >> n >> k;
    if(n == 1) {cout << 1; continue;}
    if(k & 1){cout << -1; continue;}
    ll nw = (k >> 1) - 1;
    if(nw > n || ((n - nw) & 1)){cout << -1; continue;}
    ll r = n - nw;
    for(ll i = r+1; i <= n; i++) a[i] = i;
    for(ll j = 1; j <= r; j+=2) a[j] = n - j + 1, a[j+1] = nw + j - 1;
    for(ll i = 1; i <= n; i++) cout << a[i] << " ";
  }
  cout << endl;
  return 0;
}
