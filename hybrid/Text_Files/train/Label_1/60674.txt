from sys import stdin
input = stdin.readline
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for i in range(n):
        if a[i] * 2 > ans:
            ans = a[i]
        for j in range(i + 1, n):
            if (a[i] + a[j]) * 2 > ans and a[i] % a[j] != 0 and a[j] % a[i] != 0:
                ans = a[i] + a[j]
            for k in range(j + 1, n):
                if (a[i] + a[j] + a[k]) > ans and a[i] % a[j] != 0 and a[j] % a[k] != 0 and a[i] % a[k] != 0:
                    ans = a[i] + a[j] + a[k]
    print(ans)
