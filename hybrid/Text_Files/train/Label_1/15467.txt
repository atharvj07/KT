#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<long long> pre(n, 0);
  pre[0] = a[0];
  for (int i = 1; i < n; i++) { pre[i] = pre[i - 1] ^ a[i]; }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if ((j - i + 1) % 2 == 0) {
        if (i == 0) {
          if (pre[j] == a[j + 1]) { ans++; }
        } else if (pre[j] == pre[i - 1] ^ a[j + 1]) {
          ans++;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
