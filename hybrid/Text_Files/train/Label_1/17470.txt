#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    int ans = 1;
    for (int i = 0; i < n - 1; ++i) {
        if (a[i] * 2 < a[i + 1]) ans++;
    }
    cout << ans << endl;
    return 0;
}
