import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt(), w1 = sc.nextInt();
        int[][] a = new int[h1 + 1][w1 + 1];
        int[][] b = new int[h1 + 1][w1 + 1];
        for (int i = 1; i <= h1; i++) {
            for (int j = 1; j <= w1; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        for (int i = 1; i <= h1; i++) {
            for (int j = 1; j <= w1; j++) {
                b[i][j] = sc.nextInt();
            }
        }
        int h2 = sc.nextInt(), w2 = sc.nextInt();
        int[][] c = new int[h2 + 1][w2 + 1];
        for (int i = 1; i <= h2; i++) {
            for (int j = 1; j <= w2; j++) {
                c[i][j] = sc.nextInt();
            }
        }

        int maxScore = 0;
        boolean found = false;
        for (int i = 1; i <= h1 - h2 + 1; i++) {
            for (int j = 1; j <= w1 - w2 + 1; j++) {
                int score = 0;
                boolean same = true;
                for (int k = 1; k <= h2; k++) {
                    for (int l = 1; l <= w2; l++) {
                        if (b[i + k - 1][j + l - 1] != c[k][l]) {
                            same = false;
                        }
                        if (same) {
                            score += a[i + k - 1][j + l - 1];
                        }
                    }
                }
                if (same) {
                    maxScore = Math.max(maxScore, score);
                    found = true;
                }
            }
        }
        if (found) {
            System.out.println(maxScore);
        } else {
            System.out.println("NA");
        }
    }
}
