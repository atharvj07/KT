n = int(input())
t = list(map(int, input().split()))
dp = [0] * n
for i in range(1, n):
    dp[i] = dp[i - 1] + 1 if t[i] == t[i - 1] else 1
dp[n - 1] = 1
best = 0
for i in range(n):
    for j in range(i + 1, n):
        if t[i] == t[j] or j - i + 1 - dp[j] == dp[j]:
            best = max(best, j - i + 1)
print(best)
