#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> freq(26);
    for (int i = 0; i < n; i++) freq[s[i] - 'a']++;
    int ans = 0;
    for (int i = 0; i < 26; i++) {
      ans += freq[i] / (k + 1) * k;
      freq[i] %= (k + 1);
    }
    int temp = 0;
    for (int i = 0; i < 26; i++) temp = max(temp, freq[i]);
    ans += temp;
    cout << ans << "\n";
  }
}
