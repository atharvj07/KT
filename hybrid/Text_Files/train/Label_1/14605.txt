#include <iostream>
#include <vector>
#define mod 1000000007

using namespace std;
typedef long long LL;

LL modPow(LL a, LL b) {
  if(b==0) return 1;
  if(b==1) return a;

  LL ret = modPow(a, b/2);
  ret = (ret * ret) % mod;

  if(b % 2LL) ret = (ret * a) % mod;
  return ret;
}

int main() {
  LL k;
  cin >> k;

  vector<LL> a(k);
  for(int i=0; i<k; i++) cin >> a[i];

  long long numer = 0, denom = 1;

  for(int i=0; i<k; i++) {
    if(a[i]%2LL == 0) continue;

    numer = (numer * modPow(4, (a[i]-1LL)/2LL)) % mod;
    denom = (denom * modPow(2, (a[i]-1LL)/2LL)) % mod;
  }

  cout << numer << '/' << denom << endl;
}
