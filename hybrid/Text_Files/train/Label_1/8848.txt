#include <iostream>
#include <vector>

using namespace std;

const int MAX_SIZE = 1e5 + 10;
vector<int> pref(MAX_SIZE), suff(MAX_SIZE);

int main() {
  string s;
  int m;
  cin >> s;
  cin >> m;
  for (int i = 0; i < m; i++) {
    int l, r, k;
    cin >> l >> r >> k;
    pref[l] += k;
    suff[r + 1] -= k;
  }
  for (int i = 1; i < pref.size(); i++) {
    pref[i] += pref[i - 1];
  }
  for (int i = pref.size() - 2; i >= 0; i--) {
    suff[i] += suff[i + 1];
  }
  for (int i = 0; i < s.size(); i++) {
    int shifts = pref[i] + suff[i];
    s[i] = s[(i - shifts % s.size() + s.size()) % s.size()];
  }
  cout << s << endl;
  return 0;
}
