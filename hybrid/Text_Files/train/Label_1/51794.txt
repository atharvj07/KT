import java.util.*;

class Main {
    static int countOpen(String str, int n) {
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            if (str.charAt(i) == '(') ++ans; else {
                if (ans > 0) --ans; else return -1;
            }
        }
        return ans;
    }

    static String printMinBracket(String str, int n) {
        int op = countOpen(str, n);
        if (op == 0) return str;
        if (op == -1 || op % 2 > 0) return "-1";
        StringBuilder sb = new StringBuilder("");
        int rem = 0, pos = 0;
        for (int i = 0; i < n; ++i) {
            if (str.charAt(i) == '(') {
                sb.append("(");
                if (rem > 0) {
                    --rem;
                    sb.append(")");
                }
            } else {
                if (rem > 0) {
                    sb.append(")");
                    --rem;
                } else pos = i;
            }
        }
        while (rem > 0) { sb.append(")"); rem--; }
        for (int i = pos + 1; i < n; ++i) sb.append(str.charAt(i));
        for (int i = 0; i < pos + 1; ++i) sb.append(str.charAt(i));
        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String str = sc.nextLine();
        int n = str.length();
        System.out.println(printMinBracket(str, n));
    }
}
