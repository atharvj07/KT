#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int cnt = 0;
    for (int i = n - 1; i >= 0;) {
        if (a[i] == 1)
            break;
        if (a[i] % 2 == 1) {
            cnt += a[i] / 2;
        } else {
            if (i > 0 && a[i] == a[i - 1]) {
                cnt += a[i];
                i--;
            } else
                break;
        }
        i--;
    }
    cout << cnt << endl;
    return 0;
}
