#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, t;
  cin >> n >> t;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (t >= a[i]) {
      t -= a[i];
      ans++;
    } else {
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
