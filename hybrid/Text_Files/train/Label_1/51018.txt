import java.util.*;

class Main {
    static int n, m;
    static int[] a;
    static HashMap<Integer, ArrayList<Pair>> pairs;

    static class Pair {
        int x, y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        a = new int[n];
        pairs = new HashMap<>();
        for (int i = 0; i < n; ++i) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < m; ++i) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            if (!pairs.containsKey(x + y)) {
                pairs.put(x + y, new ArrayList<>());
            }
            pairs.get(x + y).add(new Pair(x, y));
        }
        int ans = 0;
        ArrayList<Integer> keys = new ArrayList<>(pairs.keySet());
        Collections.sort(keys);
        for (int key : keys) {
            boolean used = false;
            for (Pair p : pairs.get(key)) {
                if (a[p.x] % 2 == 1 || a[p.y] % 2 == 1) {
                    used = true;
                    continue;
                }
                int g = gcd(a[p.x], a[p.y]);
                if (g % 2 == 0) {
                    a[p.x] /= g;
                    a[p.y] /= g;
                    ++ans;
                    used = true;
                }
            }
            if (used) {
                pairs.remove(key);
            }
        }
        System.out.println(ans);
    }

    static int gcd(int a, int b) {
        while (b != 0) {
            int tmp = a;
            a = b;
            b = tmp % b;
        }
        return Math.abs(a);
    }
}
