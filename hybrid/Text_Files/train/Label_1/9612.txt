#include<bits/stdc++.h>
using namespace std;
int main(){
	int n;
	cin>>n;
	int ar[4][4];
	for(int i=0;i<4;i++){
		for(int j=0;j<4;j++){
			cin>>ar[i][j];
		}
	}
	for(int i=0;i<4;i++){
		for(int j=0;j<4;j++){
			for(int k=0;k<4;k++){
				if(ar[i][j]+ar[i][k]<=n && ar[i][j]>0 && ar[i][k]>0){
					cout<<i+1<<" "<<ar[i][j]<<" "<<ar[i][k]<<endl;
					return 0;
				}
			}
		}
	}
	cout<<-1<<endl;
	return 0;
}
