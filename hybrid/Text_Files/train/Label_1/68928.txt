import sys

def main():
    n = int(input())
    items = []
    for _ in range(n):
        t, d, p = map(int, input().split())
        items.append((t, d, p))

    items.sort(key=lambda x: x[1])

    ans = [0] * n
    taken = set()
    cur_t = 0
    for t, d, p in items:
        if cur_t + t <= d and t not in taken:
            ans[items.index((t, d, p))] = 1
            taken.add(t)
            cur_t += t
    print(sum(p for p, t in zip(ans, items) if t))
    print(sum(ans))
    print(*[i + 1 for i, t in enumerate(ans) if t])

main()
