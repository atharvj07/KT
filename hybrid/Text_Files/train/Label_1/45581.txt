import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        Map<Integer, Integer> count = new HashMap<>();
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            count.put(a[i], count.getOrDefault(a[i], 0) + 1);
        }
        Arrays.sort(a);
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0 && a[i] != a[i - 1]) {
                cur = 0;
            }
            int c = count.get(a[i]);
            cur += c / 2;
            c -= c / 2;
            if (cur > 0) {
                ans += c;
                if (cur % 2 == 1) {
                    ans += 1;
                }
            }
        }
        System.out.println(ans);
    }
}
