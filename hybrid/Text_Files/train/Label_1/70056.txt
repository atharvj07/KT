n, m = map(int, input().split())
image = [input().strip() for _ in range(n)]

def count_faces(image):
    face_count = 0
    for i in range(n-1):
        for j in range(m-1):
            square = image[i][j:j+2] + image[i+1][j:j+2]
            if 'f' in square and 'a' in square and 'c' in square and 'e' in square:
                face_count += 1
    return face_count

print(count_faces(image))
