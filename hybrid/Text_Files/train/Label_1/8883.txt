#include <iostream>
#include <cmath>

long long gcd(long long a, long long b) {
    if (b == 0) {
        return a;
    } else {
        return gcd(b, a % b);
    }
}

int main() {
    long long n, l, v1, v2, k;
    std::cin >> n >> l >> v1 >> v2 >> k;
    long long lcm = ((v1 * v2) / gcd(v1, v2));
    long long bus_trips = ceil(static_cast<double>(n) / k);
    long long time = (l * v1 * bus_trips) / static_cast<double>(lcm);
    std::cout.precision(12);
    std::cout << static_cast<double>(time) / v1 << "\n";
    return 0;
}
