import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int num = in.nextInt();
        int[] arr = new int[10];
        long[] dp = new long[120];
        dp[0] = 1;
        for (int i = 0; i < 10; i++) {
            arr[i] = in.nextInt();
        }

        for (int i = 1; i <= num; i++) {
            for (int j = 0; j < 10; j++) {
                for (int k = arr[j]; k <= i; k++) {
                    dp[i] = (dp[i] + dp[i - k]) % 1000000007;
                }
            }
        }
        System.out.println(dp[num]);
    }
}
