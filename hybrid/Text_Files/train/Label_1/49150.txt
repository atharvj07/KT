import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextInt();
        int p = in.nextInt();
        if (n < Math.abs(p)) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        while (n > 0) {
            long cur = (1L << Long.numberOfLeadingZeros(n - Math.abs(p))) - Math.abs(p);
            n -= cur;
            ans++;
        }
        System.out.println(ans);
    }
}
