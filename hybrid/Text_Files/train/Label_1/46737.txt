import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        double p = input.nextDouble();
        int t = input.nextInt();
        double[] dp = new double[t + 1];
        dp[0] = 0;
        for (int i = 1; i <= t; i++) {
            dp[i] = dp[i - 1] * (1 - p) + dp[i - 1] * p + p;
        }
        System.out.println(dp[t]);
    }
}
