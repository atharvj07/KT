from collections import defaultdict
N,M=map(int,input().split())
G=defaultdict(list)
for i in range(M):
	u,v=map(int,input().split())
	G[u].append(v)
used,color=set(),{}
def dfs(v,c):
	used.add(v)
	color[v]=c
	for u in G[v]:
		if u not in used and dfs(u,(c+1)%2):
			return True
	return False
if dfs(1,0):
	print(2)
	print(' '.join(str(color[i]%2+1) for i in range(1,N+1)))
else:
	print(1)
	print(' '.join('1' for i in range(1,N+1)))
