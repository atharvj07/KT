import sys

def gcd(a,b):
  return a if b == 0 else gcd(b, a % b)

n, m = map(int, input().split())
if m * 2 < n - 1:
  print("Impossible")
  sys.exit(0)

print("Possible")
for i in range(2, n):
  print(1, i)
  m -= 1
  if m == 0:
    break

v1 = 1
for i in range(n, 2, -1):
  if m == 0:
    break
  if i != v1 and gcd(i, v1) == 1:
    print(i, v1)
    m -= 1
    v1 = i
