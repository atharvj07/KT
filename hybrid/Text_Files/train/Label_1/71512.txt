for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if sum(a) % 2 == 0:
        print(n)
        for i in range(n):
            print(i + 1, end=' ')
        print()
    else:
        even = [ai for ai in a if ai % 2 == 0]
        if len(even) > 0:
            print(1)
            print(a.index(even[0]) + 1)
        else:
            print(-1)
