#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int MOD = 1e9 + 7;
int dp[21][2000001];
int main() {
    int n; cin >> n;
    vector<int> v(1 << (2 * n));
    for (int i = 0; i < (1 << (2 * n)); ++i) cin >> v[i];
    dp[0][0] = v[0];
    for (int i = 0; i < (1 << (2 * n)); ++i) {
        for (int j = 0; j < (1 << (2 * n)); ++j) {
            if (i == j) continue;
            int bit = 0, gg = 0;
            for (int k = 0; k < 2 * n; ++k) {
                if (((1 << k) & i) != ((1 << k) & j)) gg++, bit = k;
                if (gg > 1) break;
            }
            if (gg == 1) dp[bit][i | j] = max(dp[bit][i | j], dp[bit][i] + v[j]);
        }
    }
    int ans = 0;
    for (int i = 0; i < (1 << (2 * n)); ++i) ans = max(ans, dp[2 * n - 1][i]);
    cout << ans << endl;
}
