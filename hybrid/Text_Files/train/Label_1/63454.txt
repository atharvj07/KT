def reflect_time(h, m, time):
    reflection_map = {'0':'0', '1':'1', '2':'5', '5':'2', '8':'8'}
    hours, minutes = time.split(":")
    reflected_hours = ""
    reflected_minutes = ""
    for digit in minutes[::-1]:
        if digit in reflection_map:
            reflected_hours += reflection_map[digit]
        else:
            return None
    for digit in hours[::-1]:
        if digit in reflection_map:
            reflected_minutes += reflection_map[digit]
        else:
            return None
    if int(reflected_hours) < h and int(reflected_minutes) < m:
        return f"{reflected_hours.zfill(2)}:{reflected_minutes.zfill(2)}"
    else:
        return None

def find_next_valid_time(h, m, s):
    hours, minutes = map(int, s.split(":"))
    while True:
        s = f"{hours:02d}:{minutes:02d}"
        if reflect_time(h, m, s):
            return s
        minutes += 1
        if minutes == m:
            minutes = 0
            hours += 1
        if hours == h:
            hours = 0

T = int(input())
for _ in range(T):
    h, m = map(int, input().split())
    s = input()
    print(find_next_valid_time(h, m, s))
