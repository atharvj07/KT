#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 305;
ll jump[N], cost[N], mx;
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) { cin >> jump[i]; }
  for (int i = 0; i < n; ++i) { cin >> cost[i]; }
  sort(cost, cost + n);
  sort(jump, jump + n);
  ll ans = 0;
  for (int i = 0; i < n; ++i) {
    if (mx >= jump[i]) continue;
    mx = jump[i];
    ans += cost[i];
  }
  if (mx == jump[n - 1])
    cout << ans << "\n";
  else
    cout << "-1\n";
  return 0;
}
