#include <bits/stdc++.h>
using namespace std;

#define mp make_pair
#define F first
#define S second
#define pb push_back
#define maxn 2005
#define endl '\n'
typedef long long ll;

const int inf = 1e9;

int n, c[maxn], k[maxn], vis[maxn], st;
ll x[maxn], y[maxn], dis[maxn], w[maxn];

ll dist(ll x1, ll y1, ll x2, ll y2) {
    return abs(x1 - x2) + abs(y1 - y2);
}

vector<pair<int, int>> adj[maxn];

void dij(int st) {
    for (int i = 1; i <= n; i++) { dis[i] = inf; }
    dis[st] = 0;
    priority_queue<pair<ll, int>> pq;
    pq.push(mp(0, st));
    while (!pq.empty()) {
        int u = pq.top().S;
        ll du = -pq.top().F;
        pq.pop();
        if (du > dis[u]) continue;
        for (auto p : adj[u]) {
            int v = p.F, ww = p.S;
            if (dis[v] > du + ww) {
                dis[v] = du + ww;
                w[v] = ww;
                pq.push(mp(-dis[v], v));
            }
        }
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> x[i] >> y[i];
    }
    for (int i = 1; i <= n; i++) cin >> c[i];
    for (int i = 1; i <= n; i++) cin >> k[i];
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            if (i == j) adj[i].pb(mp(i, 0));
            else {
                ll ww = k[i] + k[j];
                ww *= dist(x[i], y[i], x[j], y[j]);
                adj[i].pb(mp(j, ww));
                adj[j].pb(mp(i, ww));
            }
        }
    }
    ll ans = 0;
    vector<pair<int, ll>> vec; // <ind, dist>
    for (int i = 1; i <= n; i++) {
        dij(i);
        ll mx = 0;
        for (int j = 1; j <= n; j++) {
            if (i == j) continue;
            if (dis[j] < inf) mx = max(mx, dis[j]);
        }
        vec.pb(mp(i, mx));
        ans += c[i];
    }
    sort(vec.begin(), vec.end(), [](const pair<int, ll> &x, const pair<int, ll> &y) { return x.S > y.S; });
    vector<ll> ansc;
    for (auto p : vec) {
        bool ok = true;
        for (auto q : adj[p.F]) {
            if (vis[q.F] == 0) {
                dis[q.F] = q.S;
                ok = false;
                break;
            }
        }
        if (ok) continue;
        vis[p.F] = 1;
        ansc.pb(p.F);
    }
    cout << ans << endl;
    cout << ansc.size() << endl;
    for (auto x : ansc) cout << x << " ";
    cout << endl;
    cout << 0 << endl;
    return 0;
}
