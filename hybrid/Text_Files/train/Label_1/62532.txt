import heapq

n, k = map(int, input().split())
a = [list(input()) for _ in range(n)]
vis = [[False] * n for _ in range(n)]
q = [(a[0][0], 0, 0)]
ans = 'z' * (2 * n - 1)
while q:
    s, x, y = heapq.heappop(q)
    if x == n - 1 and y == n - 1:
        ans = min(s, ans)
        continue
    if vis[x][y]: continue
    vis[x][y] = True
    for nx, ny in ((x + 1, y), (x, y + 1)):
        if nx < n and ny < n:
            heapq.heappush(q, (s + a[nx][ny], nx, ny))
    if k <= 0: continue
    for c in range(ord('a'), ord('z') + 1):
        heapq.heappush(q, (s + chr(c), x, y))
        k -= 1
print(ans)
