import heapq
inf = 10 ** 9 + 7

def dij(n, adj, s):
    dist = [inf] * (n + 1)
    dist[s] = 0
    pq = [(0, s)]

    while pq:
        x, u = heapq.heappop(pq)
        if x > dist[u]: continue

        for v, w in adj[u]:
            if dist[v] > x + w:
                dist[v] = x + w
                heapq.heappush(pq, (dist[v], v))

    return dist

n, m, k = map(int, input().split())
x = list(map(int, input().split()))

adj = [[] for i in range(n+1)]
for i in range(m):
    u, v, w = map(int, input().split())
    adj[u].append((v, w))
    adj[v].append((u, w))

for i in range(k):
    tmp = dij(n, adj, x[i])
    maxlen = 0
    for j in x:
        if j != x[i] and tmp[j] > maxlen: maxlen = tmp[j]
    print(maxlen)
