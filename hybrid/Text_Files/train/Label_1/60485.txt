s=input()
n=len(s)
if s==s[::-1]:
    print(1)
    print(s)
else:
    l=[0]*26
    for i in s:
        l[ord(i)-ord('a')]+=1
    flag=True
    for i in l:
        if i%2==1:
            flag=False
    if flag:
        print(1)
        print(s)
    else:
        ans=[]
        k=0
        for i in range(26):
            if l[i]%2==1:
                ans.append(chr(ord('a')+i))
                k+=1
            for j in range(l[i]//2):
                ans.append(chr(ord('a')+i))
        s=ans[::-1]+ans[1:]
        if k:
            s=s+'a'
        print(len(s)//n)
        for i in range(0,len(s),len(s)//n):
            print(s[i:i+len(s)//n])
