#include <bits/stdc++.h>

using namespace std;

const int mxN = 3e5;
int n, p, k, cnt[mxN], suf[mxN];
long long a[mxN];

long long modinv(long long a, long long b) {
    return ((a%b)+b)%b;
}

int main() {
    cin >> n >> p >> k;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    suf[n - 1] = cnt[a[n - 1]];
    for (int i = n - 2; i >= 0; i--) {
        suf[i] = suf[i + 1] + cnt[a[i]];
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        long long mod = (a[i] * a[i] + k) % p;
        long long add = modinv(2 * a[i], p);
        add %= p;
        ans += suf[i] * cnt[mod] - cnt[a[i]] * (cnt[a[i]] - 1) / 2;
        ans %= p;
        cnt[a[i]]--;
    }
    cout << ans << '\n';
    return 0;
}
