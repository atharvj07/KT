def solve():
    n, ribbons = int(input()), [input(), input(), input()]
    max_beauties = [0] * 3
    counts = [[0] * 26 for _ in range(3)]
    for i in range(3):
        for j in range(len(ribbons[i])):
            idx = ord(ribbons[i][j]) - ord('a') if ribbons[i][j].islower() else ord(ribbons[i][j]) - ord('A') + 26
            counts[i][idx] += 1
            max_beauties[i] = max(max_beauties[i], counts[i][idx])
    if max_beauties.count(max(max_beauties)) > 1:
        print('Draw')
    else:
        print(['Kuro', 'Shiro', 'Katie'][max_beauties.index(max(max_beauties))])

solve()
