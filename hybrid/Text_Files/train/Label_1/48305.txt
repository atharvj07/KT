import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[][] ans = new int[n * (n - 1) / 2][2];
        int cnt = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) {
                    ans[cnt][0] = i + 1;
                    ans[cnt][1] = j + 1;
                    cnt++;
                }
            }
        }

        if (cnt == 0) {
            System.out.println(-1);
            return;
        }

        System.out.println(cnt);
        for (int i = 0; i < cnt; i++) {
            System.out.println(ans[i][0] + " " + ans[i][1]);
        }
    }
}
