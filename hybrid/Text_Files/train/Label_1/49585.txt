import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Meteor> meteors = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] parts = br.readLine().split(" ");
            meteors.add(new Meteor(Long.parseLong(parts[0]), Long.parseLong(parts[1]), Long.parseLong(parts[2]), Long.parseLong(parts[3]), Long.parseLong(parts[4]), Long.parseLong(parts[5])));
        }
        meteors.sort(new Comparator<Meteor>() {
            @Override
            public int compare(Meteor o1, Meteor o2) {
                return Long.compare(o1.t1, o2.t1);
            }
        });
        long max = 1;
        List<Meteor> active = new ArrayList<>();
        active.add(meteors.get(0));
        for (int i = 1; i < n; i++) {
            Meteor m = meteors.get(i);
            if (m.t1 >= active.get(0).t2) {
                active.clear();
                active.add(m);
                max = Math.max(max, 1);
            } else {
                active.add(m);
                max = Math.max(max, active.size());
            }
        }
        System.out.println(max);
    }

    static class Meteor {
        long t1, x1, y1, t2, x2, y2;

        Meteor(long t1, long x1, long y1, long t2, long x2, long y2) {
            this.t1 = t1;
            this.x1 = x1;
            this.y1 = y1;
            this.t2 = t2;
            this.x2 = x2;
            this.y2 = y2;
        }
    }
}
