D = ord('D')
n = int(input())
s = input()
D_left = s.count(chr(D))
R_left = s.count(chr(D)) - D_left
flag = 1
while D_left != 0 and R_left != 0:
    if flag == 1:
        R_left -= int(s[D_left - 1] == chr(D))
        D_left -= 1
    else:
        D_left -= int(s[R_left - 1] == chr(D))
        R_left -= 1
    flag = 1 - flag
print(chr(D) if flag == 1 else chr(D + 1))
