import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int[] nums = new int[n + 3];  // 1-based
    for (int i = 0; i < m; i++) {
      int a = in.nextInt(), b = in.nextInt();
      if (a > b) {
        int t = b;
        b = a;
        a = t;
      }
      nums[a]++;
      nums[b + 1]--;
    }
    boolean ans = false;
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
      cnt += nums[i];
      if (cnt > 0) ans = true;
    }
    if (nums[n + 1] != 0) ans = false;
    System.out.println(ans ? "Yes" : "No");
  }
}
