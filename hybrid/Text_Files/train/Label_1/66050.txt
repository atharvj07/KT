n = int(input())
W = [list(map(int, input().split())) for _ in range(n)]

A = [[0] * n for _ in range(n)]
B = [[0] * n for _ in range(n)]

for i in range(n):
    for j in range(n):
        A[i][j] = (W[i][j] + W[j][i]) / 2
        B[i][j] = (W[i][j] - W[j][i]) / 2

for row in A:
    print(' '.join('{:.8f}'.format(x) for x in row))
for row in B:
    print(' '.join('{:.8f}'.format(x) for x in row))
