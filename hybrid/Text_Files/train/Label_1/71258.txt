import sys

n, m = list(map(int, sys.stdin.readline().strip().split()))
g = [set() for i in range(n + 1)]
for i in range(m):
    a, b = list(map(int, sys.stdin.readline().strip().split()))
    g[a].add(b)
    g[b].add(a)

ans = -1
for i in range(1, n + 1):
    for j in g[i]:
        for k in g[j]:
            if i != j and i != k and j != k and k in g[i]:
                ans = min(ans, g[i].union(g[j].union(g[k])).pop() + g[j].union(g[k]).pop() + g[k].pop())

print(ans)
