MOD = 10**9 + 7

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(a):
    n = 1
    for x in a:
        n *= x
    if n % 2 == 0:
        p = (n // 2) % MOD
        q = n % MOD
    else:
        p = (n // 2) % MOD
        q = (n // 2 + 1) % MOD
    g = gcd(p, q)
    return f"{p // g}/{q // g}"

k = int(input().strip())
a = list(map(int, input().strip().split()))
print(solve(a))
