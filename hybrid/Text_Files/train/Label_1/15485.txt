#include <iostream>
#include <vector>
#include <unordered_map>
#include <map>
#include <algorithm>

using namespace std;

enum class State { Unknown, Okay, Impossible };

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> points(n);
  vector<int> weights(n);
  unordered_map<int, int> ordered;
  for (int i = 0; i < n; ++i) {
    int x, y;
    cin >> x >> y;
    points[i] = {x, y};
    ordered.emplace(x * 50000 + y, i);
  }
  for (int& w : weights) { cin >> w; }
  vector<State> state(n, State::Unknown);
  int max_x = 0, max_y = 0;
  for (int i = n - 1; i >= 0; --i) {
    int x = points[i].first;
    int y = points[i].second;
    int idx = ordered.at(x * 50000 + y);
    if (x > max_x || y > max_y) state[idx] = State::Impossible;
    else if (x == max_x && y == max_y) state[idx] = State::Okay;
    else {
      if (weights[idx] < weights[ordered.at(max_x * 50000 + max_y)]) {
        state[idx] = State::Impossible;
      } else {
        state[idx] = State::Okay;
        max_x = x;
        max_y = y;
      }
    }
  }
  bool possible = true;
  for (State s : state) {
    if (s == State::Impossible) {
      possible = false;
      break;
    }
  }
  if (!possible) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    map<int, vector<int>> x_idx;
    map<int, vector<int>> y_idx;
    for (int i = 0; i < n; ++i) {
      x_idx[points[i].first].push_back(i);
      y_idx[points[i].second].push_back(i);
    }
    int num = 0;
    for (int x = 0; x <= max_x; ++x) {
      for (int idx : x_idx[x]) {
        if (state[idx] == State::Okay) {
          cout << points[idx].first << ' ' << points[idx].second << endl;
          state[idx] = State::Unknown;
          ++num;
          while (y_idx[points[idx].second + 1].size() &&
             state[y_idx[points[idx].second + 1].back()] == State::Okay) {
            idx = y_idx[points[idx].second + 1].back();
            cout << points[idx].first << ' ' << points[idx].second << endl;
            state[idx] = State::Unknown;
            ++num;
          }
        }
      }
    }
  }
  return 0;
}
