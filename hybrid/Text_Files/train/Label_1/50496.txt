import java.util.*;

public class Main {

  private static final char SHIFT = 'S';

  private static void applyVerticalShift(char[][] keyboard, int x, int y) {
    for (int dx = -x; dx <= x; dx++) {
      for (int dy = -1; dy <= 0; dy++) {
        int ny = y + dy;
        int nx = x + dx;
        if (ny >= 0 && ny < keyboard.length && nx >= 0 && nx < keyboard[0].length) {
          keyboard[ny][nx] = Character.toUpperCase(keyboard[ny][nx]);
        }
      }
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int x = sc.nextInt();
    int count = 0;
    char[][] keyboard = new char[n][m];
    sc.nextLine();
    for (int i = 0; i < n; i++) {
      char[] line = sc.nextLine().toCharArray();
      for (int j = 0; j < m; j++) {
        keyboard[i][j] = line[j];
        if (line[j] == SHIFT) {
          applyVerticalShift(keyboard, j, i);
        }
      }
    }
    int q = sc.nextInt();
    sc.nextLine();
    String text = sc.nextLine();
    for (char c : text.toCharArray()) {
      boolean found = false;
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
          if (keyboard[i][j] == c) {
            found = true;
          }
        }
      }
      if (!found) {
        count++;
      }
    }
    if (count == 0) {
      System.out.println(count);
    } else {
      System.out.println(-1);
    }
  }
}
