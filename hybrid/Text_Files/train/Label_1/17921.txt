#include<bits/stdc++.h>
#define loop(i,a,b) for(int i=a;i<b;++i)
using namespace std;
typedef long long ll;
const int maxn=1e5+5;
int p[maxn],q[maxn],c[maxn],d[maxn];
bool cmp(int a,int b){
    return c[a]==c[b]?a<b:p[a]<p[b];
}
int main(){
    int n;cin>>n;
    loop(i,0,n) cin>>p[i],c[i]=p[i];
    loop(i,0,n) cin>>q[i],c[i]+=q[i];
    loop(i,0,n) cin>>d[i],c[i]+=d[i];
    sort(p,p+n,cmp);
    sort(q,q+n,cmp);
    sort(d,d+n,cmp);
    int ans=-1;
    loop(i,0,n){
        if(p[i]+q[i]==d[i]){
            ans=i;break;
        }
    }
    if(~ans){
        loop(i,0,n){
            cout<<p[i]<<" ";
        }
        cout<<endl;
        loop(i,0,n){
            cout<<q[i]<<" ";
        }
        cout<<endl;
        loop(i,0,n){
            cout<<d[i]<<" ";
        }
    }else cout<<-1;
    cout<<endl;
    return 0;
}
