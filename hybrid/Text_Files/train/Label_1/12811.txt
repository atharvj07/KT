#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int l = a[1], r = a[n];
  for (int i = 2; i <= n; i++) {
    if (a[i - 1] < r || l < a[i]) { cout << "NO" << endl; }
    l = min(l, a[i]);
    r = max(r, a[i]);
  }
  cout << "YES" << endl;
}
