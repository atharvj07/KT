#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=2e5+5;
int a[N],ans[N];
int main(){
    int n;
    cin>>n;
    for(int i=2;i<=n;++i){cin>>a[i];}
    for(int i=n;i>1;--i){
        ll x=i,y=0;
        while(x&&x<=n){
            y+=a[x];
            x+=a[x];
            y-=a[x];
            x-=a[x];
        }
        if(x<0||x>n){ans[i]=-1;}
        else{ans[i]=y;}
    }
    for(int i=2;i<=n;++i){cout<<ans[i]<<endl;}
    return 0;
}
