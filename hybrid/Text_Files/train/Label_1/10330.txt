#include <iostream>
#include <vector>
using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  if (a.size() != b.size()) {
    cout << "NO" << endl;
    return 0;
  }
  bool flag = true;
  for (int i = 0; i < a.size(); i++) {
    if (a[i] != b[i]) {
      if (flag) {
        flag = false;
        if ((a[i] == '0' && b[i] == '1') || (a[i] == '1' && b[i] == '0'))
          continue;
        else {
          cout << "NO" << endl;
          return 0;
        }
      } else {
        cout << "NO" << endl;
        return 0;
      }
    }
  }
  cout << "YES" << endl;
  return 0;
}
