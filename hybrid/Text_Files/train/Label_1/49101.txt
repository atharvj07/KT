import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    int min = 1000000000;
    for (int i = 0; i < n; i++) {
      int cnt = 0;
      int j = i;
      while (j + 1 < n && a[j] > a[j + 1]) {
        j++;
        cnt++;
      }
      if (j != n - 1) {
        cnt = -1;
      }
      min = Math.min(min, cnt);
    }
    System.out.println(min);
  }
}
