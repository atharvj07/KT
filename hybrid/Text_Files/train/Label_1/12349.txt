#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<vector<vector<int>>> dp(n, vector<vector<int>>(10, vector<int>(1 << 10)));
    for (int i = 0; i < 10; i++) {
        for (int j = 0; j < 1 << 10; j++) {
            dp[0][i][j] = (j & (1 << i)) != 0;
        }
    }
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < 10; j++) {
            for (int k = 0; k < 1 << 10; k++) {
                for (int l = 0; l < 10; l++) {
                    if (l == j) continue;
                    if ((k & (1 << l)) == 0) continue;
                    dp[i][j][k] |= dp[i - 1][l][k ^ (1 << j)];
                }
            }
        }
    }
    int mask = 0;
    for (char c : s) {
        mask |= (1 << (c - '0'));
    }
    cout << (dp[n - 1][s[n - 1] - '0'][mask] ? "NO" : "YES") << endl;
    return 0;
}
