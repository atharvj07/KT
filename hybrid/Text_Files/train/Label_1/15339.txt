#include <bits/stdc++.h>
using namespace std;
#define pi pair<long long,long long>
#define mp make_pair
#define pb push_back
#define ll long long int
#define llu unsigned long long int
#define fo(i,n) for(i=0;i<n;++i)
#define ro(i,n) for(i=n-1;i>=0;--i)
#define ub(v,x) lower_bound(v.begin(),v.end(),x)-v.begin()
#define lb(v,x) upper_bound(v.begin(),v.end(),x)-v.begin()
ll max(ll x,ll y){return(x>y)?x:y;}
ll min(ll x,ll y){return(x<y)?x:y;}
const ll mod=1000000007;
ll dp[(1<<16)+10][16][3];
vector<pi>v(15);
const ll N=1e10;
ll fun(ll mask,ll pos,ll lead){
    if(pos==-1){
        if(lead==1)return N;
        return 0;
    }
    if(dp[mask][pos][lead]!=-1)return dp[mask][pos][lead];
    ll op1=fun(mask,pos-1,1);
    ll op2=N;
    if((mask&(1<<pos))==0){
        op2=fun(mask|(1<<pos),pos-1,lead&(v[pos].first>=v[pos].second));
        if(lead)
            op2+=v[pos].first;
    }
    return dp[mask][pos][lead]=min(op1,op2);
}
int main(){
    ll n,k,i,j;
    cin>>n>>k;
    fo(i,n)
        cin>>v[i].first;
    fo(i,n)
        cin>>v[i].second;
    memset(dp,-1,sizeof(dp));
    ll ans=fun(0,n-1,1);
    if(ans>=N)
        cout<<-1;
    else
        cout<<ans;
    return 0;
}
