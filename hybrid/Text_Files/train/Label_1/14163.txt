#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> r(n);
        for (int i = 0; i < n; i++) cin >> r[i];
        int s1 = 0, s2 = 0, ans = 0;
        for (int j = n - 1; j >= 0; j--) {
            if (r[j] == 1) ans++;
            if (r[j] == 3 && ans + s2 > s1) ans++;
            if (r[j] == 2) s1++;
            if (r[j] == 3 && s1 >= ans + s2) s2++;
        }
        cout << ans << endl;
    }
}
