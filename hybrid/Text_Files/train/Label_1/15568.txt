#include <bits/stdc++.h>
#define mod 1000000007

using namespace std;

int n, dp[5005][2], val[5005];

int add(int a, int b) { return (a + b) % mod; }

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    char x;
    cin >> x;
    val[i] = x == 's';
  }
  dp[n][0] = dp[n][1] = 1;
  for (int i = n - 1; i >= 1; i--) {
    if (val[i]) {
      dp[i][0] = add(dp[i + 1][0], dp[i + 1][1]);
      dp[i][1] = 0;
    } else {
      dp[i][1] = add(dp[i + 1][0], dp[i + 1][1]);
      dp[i][0] = dp[i + 1][1];
    }
  }
  cout << dp[1][1] << endl;
}
