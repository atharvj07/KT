import java.util.Arrays;
import java.util.Scanner;
 
public class Main {
   public static void main(String[] args) {
       Scanner sc = new Scanner(System.in);
       int n = sc.nextInt();
       int[] a = new int[n];
       for (int i = 0; i < n; i++) { a[i] = sc.nextInt(); }
       int min_idx = 0, max_idx = 0;
       for (int i = 0; i < n; i++) {
          if (a[i] == 1) min_idx = i;
          if (a[i] == n) max_idx = i;
       }
       if (min_idx > max_idx) { int temp = min_idx; min_idx = max_idx; max_idx = temp; }
       System.out.println(Math.max(max_idx, n - 1 - min_idx));
   }
}
