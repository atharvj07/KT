#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    string t = "zero";
    string u = "one";
    int cnt0 = 0, cnt1 = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == t[i % 4])
            cnt0++;
        if (s[i] == u[i % 3])
            cnt1++;
    }
    while (cnt0--) cout << 0 << ' ';
    while (cnt1--) cout << 1 << ' ';
    cout << endl;
    return 0;
}
