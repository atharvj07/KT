#include <bits/stdc++.h>
using namespace std;

const long long mod = 1000000007;
long long power(long long a, long long b, long long mod) {
  long long x = a;
  long long ans = 1;
  while (b > 0) {
    if (b & 1) { ans = (ans % mod) * (x % mod) % mod; }
    x = (x % mod) * (x % mod) % mod;
    b >>= 1;
  }
  return ans;
}
int main() {
  long long k, p_a, p_b;
  cin >> k >> p_a >> p_b;
  long long ab = 1LL * (p_a) * power(p_a + p_b, mod - 2, mod) % mod;
  long long x = (1LL * ab * power(ab, k - 1, mod)) % mod;
  x = (x + ab) % mod;
  x = (x * (p_a + p_b)) % mod;
  x = (x * power(p_a, mod - 2, mod)) % mod;
  cout << x << endl;
  return 0;
}
