import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashSet;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Set;

public class Main {
    static BufferedReader in = new BufferedReader(new InputStreamReader(System.in));

    static Set<Integer> factors = new HashSet<>();

    public static void main(String[] args) throws IOException {
        for (int p = 2; p * p <= 200000; p++) {
            if (isPrime(p)) {
                factors.add(p);
                factors.add(p * p);
            }
        }

        String[] input = in.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int f = Integer.parseInt(input[1]);
        factors.remove(f);

        int[][] grid = new int[n][3];
        for (int i = 0; i < n; i++) {
            input = in.readLine().split(" ");
            for (int j = 0; j < 3; j++) {
                grid[i][j] = Integer.parseInt(input[j]);
            }
        }

        int[] dpAlice = new int[n];
        int[] dpBob = new int[n];
        for (int i = 0; i < n; i++) {
            dpAlice[i] = -1;
            dpBob[i] = -1;
        }
        dpAlice[0] = 0;
        dpBob[0] = 0;

        Queue<Integer> queue = new LinkedList<>();
        queue.add(0);
        while (!queue.isEmpty()) {
            int i = queue.poll();
            for (int k : factors) {
                if (i + k < n) {
                    if (check(grid, i, k)) {
                        if (dpAlice[i] == 0) {
                            dpBob[i + k] = 1;
                            queue.add(i + k);
                        } else if (dpBob[i] == 0) {
                            dpAlice[i + k] = 1;
                            queue.add(i + k);
                        }
                    }
                }
            }
        }

        if (dpAlice[n - 1] == 1) {
            System.out.println("Alice");
        } else {
            System.out.println("Bob");
        }

        if (dpBob[n - 1] == 1) {
            System.out.println("Bob");
        } else {
            System.out.println("Alice");
        }
    }

    static boolean check(int[][] grid, int i, int k) {
        int[] row = grid[i];
        int[] nextRow = grid[i + k];

        if (nextRow[0] < row[0] || nextRow[1] < row[1] || nextRow[2] < row[2]) {
            return false;
        }

        if (nextRow[0] == row[0] || nextRow[1] == row[1] || nextRow[2] == row[2]) {
            return false;
        }

        if (nextRow[1] != row[1] + k) {
            return false;
        }

        if (nextRow[0] != row[0] + k && nextRow[2] != row[2] + k) {
            return false;
        }

        return true;
    }

    static boolean isPrime(int n) {
        if (n <= 1) {
            return false;
        }

        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }

        return true;
    }
}
