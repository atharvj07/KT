import java.util.*;
class Main {
    public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);
      int n = sc.nextInt();
      int[] v = new int[n];
      for(int i=0;i<n;i++) v[i] = sc.nextInt();
      int max = 0;
      int sum = 0;
      for(int i=0;i<n;i++) {
        max = Math.max(max, v[i]);
        sum += v[i];
      }
      double sum2 = 0;
      for(int i=0;i<n;i++) sum2 += (v[i]+1.0);
      sum2 = sum2/(n+1);
      double k = sum2;
      System.out.println((int)Math.ceil(k));
    }
}
