import sys
import heapq

n,m=map(int,input().split())
d=[[]for i in range(n)]
for i in range(n-1):
    a,b=map(int,input().split())
    a-=1;b-=1
    d[a].append(b)
    d[b].append(a)

attack=list(map(int,input().split()))
for i in range(m):
    attack[i]-=1

mn=sys.maxsize
ans=0

for i in range(n):
    visited=[False]*n
    dist=[sys.maxsize]*n
    q=[]
    dist[i]=0
    heapq.heappush(q,(0,i))
    
    while q:
        cnt,now=heapq.heappop(q)
        if visited[now]==True:
            continue
        visited[now]=True
        
        for j in d[now]:
            if dist[j]>cnt+1:
                dist[j]=cnt+1
                heapq.heappush(q,(cnt+1,j))
                
    mx=0
    for j in attack:
        mx=max(mx,dist[j])
        
    if mx<mn:
        mn=mx
        ans=i
        
print(ans+1)
print(mn)
