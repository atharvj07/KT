import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String S = sc.next();
        int Q = sc.nextInt();
        for (int i = 0; i < Q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int t = sc.nextInt();
            if (r - l + 1 % t != 0) {
                System.out.println("No");
            } else {
                boolean ok = true;
                for (int j = 0; j < t; j++) {
                    if (S.charAt(l + j) != S.charAt(l + j % t)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    System.out.println("Yes");
                } else {
                    System.out.println("No");
                }
            }
        }
    }
}
