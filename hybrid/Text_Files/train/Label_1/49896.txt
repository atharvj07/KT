import java.util.Scanner;

public class VasyaAndTriangle {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        if (n * m % k != 0) {
            System.out.println("NO");
            return;
        }

        int area = n * m / k;

        // Find a triangle with the given area.
        for (int x1 = 0; x1 <= n; x1++) {
            for (int y1 = 0; y1 <= m; y1++) {
                for (int x2 = x1 + 1; x2 <= n; x2++) {
                    for (int y2 = 0; y2 <= m; y2++) {
                        int x3 = (x1 + x2) / 2;
                        int y3 = (y1 + y2) / 2;

                        if (isTriangleAreaEqual(x1, y1, x2, y2, x3, y3, area)) {
                            System.out.println("YES");
                            System.out.println(x1 + " " + y1);
                            System.out.println(x2 + " " + y2);
                            System.out.println(x3 + " " + y3);
                            return;
                        }
                    }
                }
            }
        }

        // No triangle found.
        System.out.println("NO");
    }

    static boolean isTriangleAreaEqual(int x1, int y1, int x2, int y2, int x3, int y3, int area) {
        int a = (x2 - x1) * (y3 - y1) - (x3 - x1) * (y2 - y1);
        return Math.abs(a) == 2 * area;
    }
}
