import java.util.Scanner;

public class IvanAndLessons {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int days = scanner.nextInt();
        int hours = scanner.nextInt();
        int skips = scanner.nextInt();
        scanner.nextLine();

        int[][] timetable = new int[days][hours];
        for (int i = 0; i < days; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < hours; j++) {
                timetable[i][j] = line.charAt(j) - '0';
            }
        }

        int totalHours = 0;
        for (int i = 0; i < days; i++) {
            int start = -1;
            for (int j = 0; j < hours; j++) {
                if (timetable[i][j] == 1) {
                    if (start == -1) {
                        start = j;
                    }
                } else if (start != -1) {
                    totalHours += j - start + 1;
                    start = -1;
                }
            }
            if (start != -1) {
                totalHours += hours - start + 1;
            }
        }

        System.out.println(totalHours - skips);
    }
}
