#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    if (a == b) {
      cout << 0 << endl;
    } else if (a < b) {
      if ((b - a) % 5 == 0) {
        cout << (b - a) / 5 << endl;
      } else if ((b - a) % 2 == 0) {
        cout << 1 + (b - a) / 2 << endl;
      } else {
        cout << 2 + (b - a) / 2 << endl;
      }
    } else {
      if ((a - b) % 2 == 0) {
        cout << (a - b) / 2 << endl;
      } else {
        cout << 1 + (a - b) / 2 << endl;
      }
    }
  }
}
