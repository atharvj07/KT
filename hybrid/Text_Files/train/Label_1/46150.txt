import java.util.*;

public class Main {
    private static final int MAX_N = 100000;
    private static final int MAX_D = MAX_N - 1;
    private static int[] dist;
    private static int[] cnt;
    private static Set<Integer> affected = new HashSet<>();
    private static boolean[] visited;
    private static List<List<Integer>> adj = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();
        if (m == 0) {
            System.out.println(0);
            return;
        }
        dist = new int[n + 1];
        cnt = new int[n + 1];
        visited = new boolean[n + 1];
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 1; i <= m; i++) {
            affected.add(sc.nextInt());
        }
        for (int i = 1; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj.get(a).add(b);
            adj.get(b).add(a);
        }
        dfs(1, 1, 0);
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] <= d && cnt[i] == m) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    private static void dfs(int s, int p, int d) {
        if (visited[s]) {
            return;
        }
        visited[s] = true;
        dist[s] = d;
        cnt[s] = affected.contains(s) ? 1 : 0;
        for (int v : adj.get(s)) {
            if (v == p) {
                continue;
            }
            dfs(v, s, d + 1);
            cnt[s] += cnt[v];
        }
    }
}
