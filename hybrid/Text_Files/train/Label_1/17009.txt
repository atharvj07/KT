#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k, p;
  cin >> n >> k >> p;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  if (p <= 0 || p > k) {
    cout << "NO" << endl;
    return 0;
  }
  vector<long long> even;
  vector<long long> odd;
  for (int i = 0; i < n; i++) {
    if (a[i] & 1) {
      odd.push_back(a[i]);
    } else {
      even.push_back(a[i]);
    }
  }
  if ((int)even.size() < p) {
    cout << "NO" << endl;
    return 0;
  }
  int e_idx = 0;
  int o_idx = 0;
  vector<vector<long long>> part(k);
  for (int i = 0; i < k; i++) {
    if (i < p) {
      part[i].push_back(even[e_idx++]);
    } else {
      part[i].push_back(odd[o_idx++]);
    }
  }
  while (e_idx < even.size() || o_idx < odd.size()) {
    for (int i = 0; i < k; i++) {
      if (e_idx < even.size()) {
        part[i].push_back(even[e_idx++]);
      } else if (o_idx < odd.size()) {
        part[i].push_back(odd[o_idx++]);
      }
    }
  }
  cout << "YES" << endl;
  for (int i = 0; i < k; i++) {
    cout << part[i].size() << " ";
    for (int j = 0; j < part[i].size(); j++) {
      cout << part[i][j] << " ";
    }
    cout << endl;
  }
  return 0;
}
