#include<bits/stdc++.h>

using namespace std;
const int _ = 1<<15;
int n, p[_], d[_], s[_], t[_], m, x;
void dfs(int u)
{
	if(~t[u]) return;
	t[u]=1;
	for(int v=0;v<n;++v)
		if(s[u]==s[v]^x && d[u]==d[v]-1 && !t[v])
		{ m++; dfs(v); t[v]-=2; }
}
int main()
{
	ios_base::sync_with_stdio(0);
	cin>>n;
	while(n--)
	{
		cin>>d[n]>>s[n];
		p[d[n]]++;
	}
	for(int i=1;i<=n;++i)
	{
		if(!p[i])
		{ p[0]++; continue; }
		p[i]=p[i-1]+p[i];
	}
	for(int i=0;i<n;++i)
	{
		m+=d[i];
		x=s[i]^p[d[i]];
	}
	m/=2;
	cout<<m<<'\n';
	for(int i=0;i<n;++i)
	{
		if(d[i]==1) continue;
		x=s[i]^p[d[i]-1];
		dfs(i);
	}
	return 0;
}
