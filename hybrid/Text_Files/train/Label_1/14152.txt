#include <bits/stdc++.h>

using namespace std;
const int MAX_N = 3e2;

int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        long long divisor_product = 1;
        int mx_divisor = 0;
        for (int i = 0; i < n; ++i) {
            int x;
            cin >> x;
            divisor_product *= x;
            mx_divisor = max(x, mx_divisor);
        }
        bool can_find = true;
        for (int i = 2; i * i <= divisor_product; ++i) {
            if (divisor_product % i == 0) {
                if (divisor_product / i <= mx_divisor) {
                    can_find = false;
                    break;
                }
            }
        }
        if (!can_find) {
            cout << -1 << endl;
        } else {
            cout << mx_divisor * divisor_product << endl;
        }
    }
    return 0;
}
