#include <bits/stdc++.h>
using namespace std;
int n, p, q;
string s;
int main() {
  cin >> n >> p >> q;
  cin >> s;
  if ((n - 1) % (p + q)) {
    puts("-1");
    return 0;
  }
  int rem = n - (n - 1) / (p + q) * (p + q);
  int lst;
  if (rem == q) {
    lst = q;
    for (int i = q; i < n; i += p + q) cout << s.substr(i, p) << endl;
  } else {
    lst = p;
    for (int i = p; i < n; i += p + q) cout << s.substr(i, q) << endl;
  }
  for (int i = rem; i < n; i += p + q) cout << s.substr(i, lst) << endl;
}
