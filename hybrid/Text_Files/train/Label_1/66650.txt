from math import pi, sqrt, acos

x1, y1, r1 = map(int, input().split())
x2, y2, r2 = map(int, input().split())
d = sqrt((x2 - x1)**2 + (y2 - y1)**2)
if d > r1 + r2:
    print(0)
elif d <= abs(r1 - r2):
    print(min(r1, r2)**2 * pi)
else:
    a = (r1**2 - r2**2 + d**2) / (2 * d)
    b = d - a
    ang1 = acos(a / r1)
    ang2 = acos(b / r2)
    print((ang1 + ang2 - pi / 2) * r1**2/2 + (ang2 - ang1) * r2**2/2)
