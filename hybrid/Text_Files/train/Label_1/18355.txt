#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>

int main() {
  int n, m;
  std::cin >> n >> m;
  if (n & 1) {
    std::cout << -1 << std::endl;
    return 0;
  }
  if (n == 2) {
    std::cout << 0 << std::endl;
    std::cout << n << std::endl;
    return 0;
  }
  std::vector<int> a(n);
  std::unordered_map<int, int> cnt;
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
    cnt[a[i]]++;
  }
  std::sort(a.begin(), a.end());
  std::vector<int> ans(n);
  int cnt_even = 0;
  int cnt_odd = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] % 2 == 0) cnt_even++;
    else cnt_odd++;
  }
  if (cnt_even == cnt_odd) {
    std::cout << 0 << std::endl;
    for (int i = 0; i < n; ++i) {
      std::cout << a[i] << ' ';
    }
    std::cout << std::endl;
    return 0;
  }
  if (cnt_even == cnt_odd - 1) {
    int odd = -1;
    for (int i = 0; i < n; ++i) {
      if (a[i] % 2 == 1) {
        odd = i;
        break;
      }
    }
    std::cout << 1 << std::endl;
    for (int i = 0; i < n; ++i) {
      if (i == odd) std::cout << 2 << ' ';
      else std::cout << a[i] << ' ';
    }
    std::cout << std::endl;
    return 0;
  }
  if (cnt_odd == cnt_even - 1) {
    int even = -1;
    for (int i = 0; i < n; ++i) {
      if (a[i] % 2 == 0) {
        even = i;
        break;
      }
    }
    std::cout << 1 << std::endl;
    for (int i = 0; i < n; ++i) {
      if (i == even) std::cout << 1 << ' ';
      else std::cout << a[i] << ' ';
    }
    std::cout << std::endl;
    return 0;
  }
  std::cout << -1 << std::endl;
  return 0;
}
