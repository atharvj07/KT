import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int a = input.nextInt();
            int b = input.nextInt();
            int w1 = input.nextInt();
            int w2 = input.nextInt();
            graph.get(a).add(b);
            graph.get(b).add(a);
        }
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[1] = 0;
        Queue<Integer> queue = new LinkedList<>();
        queue.add(1);
        while (!queue.isEmpty()) {
            int u = queue.poll();
            for (int v : graph.get(u)) {
                if (dist[v] > dist[u] + 1) {
                    dist[v] = dist[u] + 1;
                    queue.add(v);
                }
            }
        }
        int minCycle = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            for (int j : graph.get(i)) {
                if (dist[i] != Integer.MAX_VALUE && dist[j] != Integer.MAX_VALUE) {
                    minCycle = Math.min(minCycle, dist[i] + dist[j] + 1);
                }
            }
        }
        System.out.println(minCycle == Integer.MAX_VALUE ? 0 : minCycle);
    }
}
