import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.*;

class Main {
    public static void main(String args[]) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = in.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]); // number of voters
        int m = Integer.parseInt(tokens[1]); // number of parties
        List<Voter> voters = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            tokens = in.readLine().split(" ");
            voters.add(new Voter(Integer.parseInt(tokens[0]), Integer.parseInt(tokens[1])));
        }
        Collections.sort(voters); // sort voters by the cost of their vote
        int totalCost = 0;
        int votesForUnitedParty = 0;
        int maxVotesForOtherParty = 0;
        for (Voter voter : voters) {
            if (voter.party == 1) {
                votesForUnitedParty++;
            } else {
                if (votesForUnitedParty > maxVotesForOtherParty) {
                    maxVotesForOtherParty = votesForUnitedParty;
                }
                if (votesForUnitedParty <= maxVotesForOtherParty) {
                    totalCost += voter.cost;
                    votesForUnitedParty++;
                }
            }
        }
        System.out.println(totalCost);
    }
}

class Voter implements Comparable<Voter> {
    int party;
    int cost;

    public Voter(int party, int cost) {
        this.party = party;
        this.cost = cost;
    }

    @Override
    public int compareTo(Voter other) {
        return Integer.compare(this.cost, other.cost);
    }
}
