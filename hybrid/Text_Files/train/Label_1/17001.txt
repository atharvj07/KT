#include <iostream>
#include <vector>
#include <string>
#include <cstdio>
#include <algorithm>

using namespace std;

struct Node {
    long long value;
    bool lbrac;
    
    Node() : value(0), lbrac(false) {}

    Node(long long val) : value(val), lbrac(false) {}
};

int main() {
    string s;
    cin >> s;
    int n = s.length();
    int open_lbrac = 0, close_lbrac = 0;
    for(int i = 0; i < n; i++) {
      if(s[i] == '(') open_lbrac++;
      if(s[i] == ')') close_lbrac++;
    }
    if(open_lbrac < close_lbrac){
      open_lbrac++;
      close_lbrac--;  
    }
    int p, m;
    cin >> p >> m;
    int rem_lbrac = p + m;
    rem_lbrac -= open_lbrac;
    rem_lbrac /= 2;
    open_lbrac += rem_lbrac;
    close_lbrac -= rem_lbrac;
    
    vector<long long> dp(n, -1);
    vector<bool> dp_lbrac(n, false);
    dp[0] = (s[0] - '0');
    dp_lbrac[0] = (s[0] == '(');
    for(int i = 1; i < n; i++) {
      if((s[i] - '0') >= 0 && (s[i]-'0') <= 9) {
        // digit
        if(dp_lbrac[i-1]) {
          dp[i] = dp[i-1] * 10 + (s[i] - '0');
        }
        else {
          dp[i] = dp[i-1] + (s[i] - '0');
        }
        dp_lbrac[i] = dp_lbrac[i-1];
      }
      else if(s[i] == '(') {
        dp_lbrac[i] = true;
        dp[i] = dp[i-1];
      }
      else if(s[i] == ')') {
        dp_lbrac[i] = false;
        if(open_lbrac > 0) {
          if(dp_lbrac[i-1]) {
            dp[i] = dp[i-1] * dp[i-1];
          }
          else {
            dp[i] = dp[i-1] + dp[i-1];
          }
          open_lbrac--;
        }
      }
      else {
        if(dp_lbrac[i-1]) {
          if(s[i] == '+') {
            dp[i] = dp[i-1] * dp[i-1];
          }
          else {
            dp[i] = dp[i-1] * dp[i-1];
          }
        }
        else {
          if(s[i] == '+') {
            dp[i] = dp[i-1] + dp[i-1];
          }
          else {
            dp[i] = dp[i-1] - dp[i-1];
          }
        }
        dp_lbrac[i] = false;
      }
    }
    cout << dp[n - 1] << endl;
}
