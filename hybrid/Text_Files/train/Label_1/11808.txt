#include<bits/stdc++.h>
using namespace std;
#define f(i,n) for(int i=0;i<n;i++)
typedef long long int ll;
int main(){
	int t;
	cin >> t;
	while(t--){
		int n;
		cin >> n;
		vector<int>arr(n);
		f(i,n) cin >> arr[i];
		vector<map<int,int>>dp(n);
		dp[n-1][arr[n-1]] = 1;
		for(int i=n-2;i>=0;i--){
			for(auto it: dp[i+1]){
				dp[i][it.first] = max(dp[i][it.first],it.second);
			}
			dp[i][arr[i]] = max(dp[i][arr[i]],1);
			dp[i][arr[i]] += dp[i+1][arr[i]];
		}
		int ans = 0;
		for(auto it: dp[0]){
			ans = max(ans,it.second);
		}
		cout << ans << endl;
	}
	return 0;
}
