t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    a = []
    left, right = 0, n - 1
    while left <= right:
        if left == right:
            a.append(b[left])
            break
        a.append(b[left])
        a.append(b[right])
        left += 1
        right -= 1
    print(*a)
