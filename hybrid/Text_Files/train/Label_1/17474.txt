#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, B;
    cin >> n >> B;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int& x : a) x %= 2;
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (a[i - 1] != a[i]) {
            if (B >= abs(a[i - 1] - a[i])) {
                B -= abs(a[i - 1] - a[i]);
                ans++;
            } else {
                break;
            }
        }
    }
    cout << ans << endl;
}
