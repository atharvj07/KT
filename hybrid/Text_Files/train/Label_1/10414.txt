#include <bits/stdc++.h>
using namespace std;

#define endl '\n'

typedef long long ll;
typedef vector<ll> vi;
typedef pair<ll, ll> pi;
typedef vector<pi> vpi;

#define rep(i, a, b) for(ll i = a; i < b; i++)
#define pb push_back
#define fir first
#define sec second
#define all(x) x.begin(), x.end()
#define sz(x) (ll)x.size()

int main(){
  ll n, m, a, b;
  cin >> n >> m;
  vi adj[n];
  rep(i, 0, m){
    cin >> a >> b;
    a--; b--;
    adj[a].pb(b);
    adj[b].pb(a);
  }

  priority_queue<pair<ll, pi>> pq;
  vi vis(n, 0), dist(n, 1e18);
  pq.push({0, {0, -1}});

  ll mx = 0, a1 = -1, a2 = -1;
  while(!pq.empty()){
    auto [d, x] = pq.top(); pq.pop(); d = -d;
    if(vis[x.fir]) continue;
    vis[x.fir] = 1;
    if(d > mx){
      mx = d;
      a1 = x.fir;
      a2 = x.sec;
    }
    for(auto u: adj[x.fir]){
      if(!vis[u])
        pq.push({d + 1, {u, x.fir}});
    }
  }

  mx = 0;
  vis.assign(n, 0);
  dist.assign(n, 1e18);
  pq.push({0, {a1, -1}});

  while(!pq.empty()){
    auto [d, x] = pq.top(); pq.pop(); d = -d;
    if(vis[x.fir]) continue;
    vis[x.fir] = 1;
    dist[x.fir] = d;
    if(d > mx){
      mx = d;
      a1 = x.fir;
    }
    for(auto u: adj[x.fir]){
      if(!vis[u])
        pq.push({d + 1, {u, x.fir}});
    }
  }

  mx = 0;
  vis.assign(n, 0);
  dist.assign(n, 1e18);
  pq.push({0, {a1, -1}});

  while(!pq.empty()){
    auto [d, x] = pq.top(); pq.pop(); d = -d;
    if(vis[x.fir]) continue;
    vis[x.fir] = 1;
    if(d > mx){
      mx = d;
      a2 = x.fir;
    }
    for(auto u: adj[x.fir]){
      if(!vis[u])
        pq.push({d + 1, {u, x.fir}});
    }
  }

  cout << mx << endl;
  if(a2 != -1){
    cout << a1 + 1 << ' ' << a2 + 1 << endl;
    adj[a1].pb(a2);
    adj[a2].pb(a1);
  }
  rep(i, 0, n){
    for(auto u: adj[i]){
      if(i < u){
        cout << i + 1 << ' ' << u + 1 << endl;
      }
    }
  }
  
  return 0;
}
