#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> h(n);
  for (int i = 0; i < n; i++) { cin >> h[i]; }
  sort(h.begin(), h.end());
  for (int i = 0; i < n / 2; i++) { cout << h[i] << ' ' << h[n - 1 - i] << ' '; }
  if (n & 1) { cout << h[n / 2]; }
  cout << endl;
  return 0;
}
