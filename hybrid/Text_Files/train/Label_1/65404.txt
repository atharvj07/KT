n = int(input())
a1 = list(map(int, input().split()))
a2 = list(map(int, input().split()))
ans = []
t = True
while t:
    t = False
    r = -1
    if a1[0] > a2[0]:
        ans.append(a1[0])
        a1.pop(0)
        r = 1
    else:
        ans.append(a2[0])
        a2.pop(0)
    for i in range(len(a1)):
        if (r != 1 and a1[i] > a2[0]) or (r == 1 and a1[i] > a2[i]):
            ans.append(a1[i])
            if r != 1: a2.pop(0)
            a1.pop(i)
            r = 1
            t = True
            break
        if (r != 2 and a2[i] > a1[0]) or (r == 2 and a2[i] > a1[i]):
            ans.append(a2[i])
            if r != 2: a1.pop(0)
            a2.pop(i)
            r = 2
            t = True
            break
print(sum(ans))
