#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 5, M = 18;

int n, a[N];
vector<int> g[N];
int f[N][M], d[N];

int get(int x, int i) { return i == 0 ? d[x] : f[x][i]; }
void up(int x, int i) { d[x] = min(d[x], get(x, i)); }

void dfs1(int x, int p = 0) {
  for (int y : g[x]) {
    if (y == p) continue;
    d[y] = d[x] + 1;
    dfs1(y, x);
  }
}

void dfs2(int x, int p = 0, int dep = 0) {
  f[x][0] = d[x];
  for (int y : g[x]) {
    if (y == p) continue;
    dfs2(y, x, dep + 1);
    for (int i = 1; i < M; i++) f[x][i] = min(f[x][i - 1], get(y, i - 1));
  }
  d[x] = min(d[x], dep);
}

int main() {
  ios_base::sync_with_stdio(false), cin.tie(nullptr);
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v), g[v].push_back(u);
  }
  int ans = INT_MAX;
  dfs1(1);
  dfs2(1);
  for (int i = 1; i <= n; i++) ans = min(ans, a[i] + d[i]);
  cout << ans << '\n';
  return 0;
}
