import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

    StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
    int n = Integer.parseInt(tokenizer.nextToken());
    int k = Integer.parseInt(tokenizer.nextToken());

    int[] cards = new int[n * k];
    tokenizer = new StringTokenizer(reader.readLine());
    for (int i = 0; i < n * k; i++) {
      cards[i] = Integer.parseInt(tokenizer.nextToken());
    }

    int[] favNumbers = new int[n];
    tokenizer = new StringTokenizer(reader.readLine());
    for (int i = 0; i < n; i++) {
      favNumbers[i] = Integer.parseInt(tokenizer.nextToken());
    }

    int[] h = new int[k + 1];
    tokenizer = new StringTokenizer(reader.readLine());
    for (int i = 1; i <= k; i++) {
      h[i] = Integer.parseInt(tokenizer.nextToken());
    }

    int[][] dp = new int[n][k * n + 1];
    for (int i = 0; i < n; i++) {
      Arrays.fill(dp[i], -1);
    }

    System.out.println(calculate(0, 0, n, k, cards, favNumbers, h, dp));
  }

  private static int calculate(int playerIdx, int cardsTaken, int n, int k, int[] cards, int[] favNumbers,
      int[] h, int[][] dp) {
    if (playerIdx == n) {
      return 0;
    }

    if (dp[playerIdx][cardsTaken] != -1) {
      return dp[playerIdx][cardsTaken];
    }

    int ans = 0;
    for (int i = cardsTaken; i < k * n; i++) {
      if (cards[i] == favNumbers[playerIdx]) {
        ans = Math.max(ans, calculate(playerIdx + 1, cardsTaken + k, n, k, cards, favNumbers, h, dp) + h[i % k]);
      }
    }

    dp[playerIdx][cardsTaken] = ans;
    return ans;
  }
}
