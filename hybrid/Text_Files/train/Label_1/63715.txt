n, k = map(int, input().split())
s = input()

result = []

for char in s:
    dist_to_a = ord(char) - ord('a')
    dist_to_z = ord('z') - ord(char)
    
    if dist_to_z <= k:
        result.append('z')
        k -= dist_to_z
    elif dist_to_a <= k:
        result.append('a')
        k -= dist_to_a
    else:
        if dist_to_a < dist_to_z:
            result.append(chr(ord(char) - k))
        else:
            result.append(chr(ord(char) + k))
        k = 0
        break

if k > 0:
    print(-1)
else:
    print(''.join(result) + s[len(result):])
