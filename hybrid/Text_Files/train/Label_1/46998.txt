import java.util.Scanner;

public class MaximObjectsPacking {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] objects = new int[n];
        for (int i = 0; i < n; i++) {
            objects[i] = scanner.nextInt();
        }

        int count = 0;
        int boxSize = k;
        for (int i = 0; i < n; i++) {
            if (objects[i] <= boxSize) {
                count++;
                boxSize -= objects[i];
            } else {
                m--;
                boxSize = k;
                if (objects[i] <= boxSize) {
                    count++;
                    boxSize -= objects[i];
                } else {
                    break;
                }
            }

            if (m == 0) {
                break;
            }
        }

        System.out.println(count);
    }
}
