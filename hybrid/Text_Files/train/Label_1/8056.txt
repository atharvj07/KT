#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> dp(n + 1);
    dp[n] = 0;
    for (int i = n - 1; i >= 0; i--) {
      dp[i] = dp[i + 1] + 1;
      if (i + 1 < n && s[i] == s[i + 1]) { dp[i] = min(dp[i], dp[i + 2]); }
    }
    cout << dp[0] << "\n";
  }
  return 0;
}
