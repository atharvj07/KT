#include <algorithm>
#include <climits>
#include <cstdio>
#include <cstring>
#include <unordered_map>
#include <vector>
using namespace std;

const int MAXN = 1500;

int n, m, q, c[MAXN + 5], maxlen[MAXN + 5][26];
char s[MAXN + 5];
unordered_map<char, int> cnt;

int main() {
  scanf("%d %s %d", &n, s, &q);
  for (int i = 1; i <= n; ++i) {
    ++cnt[s[i]];
    c[i] = cnt[s[i]];
  }

  for (char ch = 'a'; ch <= 'z'; ++ch) {
    maxlen[n + 1][ch - 'a'] = 0;
    for (int i = n; i >= 1; --i) {
      maxlen[i][ch - 'a'] = maxlen[i + 1][ch - 'a'];
      if (s[i] == ch) { maxlen[i][ch - 'a'] = max(maxlen[i][ch - 'a'], 1); }
      if (i + maxlen[i + 1][ch - 'a'] <= n && s[i + maxlen[i + 1][ch - 'a']] == ch) {
        maxlen[i][ch - 'a'] = maxlen[i + 1][ch - 'a'] + 1;
      }
    }
  }

  while (q--) {
    int x;
    char ch;
    scanf("%d %c", &x, &ch);
    printf("%d\n", x + maxlen[max(1, n - x + 1)][ch - 'a']);
  }

  return 0;
}
