n = int(input())
for _ in range(n):
    x = int(input())
    possible = False
    for a in range(0, 34):
        for b in range(0, 15):
            if a * 3 + b * 7 == x:
                possible = True
                break
        if possible:
            break
    if possible:
        print("YES")
    else:
        print("NO")
