from collections import defaultdict

mod = 998244353

n = int(input())
a = defaultdict(list)
b = defaultdict(list)
for i in range(1, n + 1):
    x, y = map(int, input().split())
    a[x].append(i)
    b[y].append(i)

inv = [0] * (n + 1)
for i in range(1, n + 1):
    inv[i] = pow(i, mod - 2, mod)

fact = [1] * (n + 1)
for i in range(2, n + 1):
    fact[i] = fact[i - 1] * i % mod

ifact = [1] * (n + 1)
for i in range(n - 1, 0, -1):
    ifact[i] = ifact[i + 1] * inv[i] % mod

def ncr(n, r):
    return fact[n] * ifact[n - r] % mod * ifact[r] % mod

ans = n * (n - 1) // 2

for x in a.values():
    l = len(x)
    ans -= l * (l - 1) // 2 % mod

for y in b.values():
    l = len(y)
    ans -= l * (l - 1) // 2 % mod

print((ans + mod) % mod)
