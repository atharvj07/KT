#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, a;
    cin >> n >> a;

    vector<int> seq(n);
    for (int i = 0; i < n; i++)
    {
        cin >> seq[i];
    }

    int b = 0; // answer
    for (int i = 0; i < n; i++)
    {
        if (seq[i] != a && (b == 0 || seq[i] != b))
        {
            b = seq[i];
        }
    }

    if (b == 0)
    {
        cout << "-1\n";
    }
    else
    {
        cout << b << "\n";
    }
    return 0;
}
