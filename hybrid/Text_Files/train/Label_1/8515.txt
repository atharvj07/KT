#include<iostream>
#include<vector>
using namespace std;

int main() {
    long long s, x;
    cin >> s >> x;
    int count = 0;
    x = x - s; x = x / 2;
    if (s < 0 || s & 1 || x > 1000000000000000000 || x < 0){
        cout << 0 << endl;
        return 0;
    }
    for (long long i = 0; i < 1000000000000000000; i = i + 2) {
        if (i < s && (x - i) <= i && i + x <= s) count++;
    }
    cout << count << endl;
    return 0;
}
