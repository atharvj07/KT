n,m=map(int,input().split())
x=[['.'for i in range(m)]for j in range(n)]
for i in range(n):x[i]=list(input())
q=int(input())
for i in range(q):
    a,b=map(int,input().split())
    l=[[0 for i in range(m)]for j in range(n)]
    for j in range(m):
        if x[n-1][j]=='.':l[n-1][j]=1
        if j>0:l[n-1][j]|=l[n-1][j-1]
    for j in reversed(range(n-1)):
        for k in range(m):
            if x[j][k]=='.':
                if l[j+1][k]:l[j][k]=1
                if k>0 and l[j+1][k-1]:l[j][k]=1
    c=1
    for j in range(n-1):
        if c>0:
            for k in range(b-1,-1,-1):
                if x[j][k]=='.':
                    if l[j+1][k]:
                        if a>k or c:c=1
                        else:c=0
                    elif k==a-1:c=0
                else:c=0
        if c==0:break
    if c>0:print("YES")
    else:print("NO")
