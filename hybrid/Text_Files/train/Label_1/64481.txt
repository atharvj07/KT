import sys
N,M = map(int,input().split())
D = [set() for i in range(N+1)]
for v in map(int,input().split()):
    D[v].add(0)
for i in range(2,N+1):
    T = int(input())
    for d in map(int,input().split()): D[i].add(d)
vis,ans = set(),[0]
for s in sorted(D[0],reverse=1):
    if vis.add(s) and all(v in vis for v in D[s]):
        ans.append(s)
print(-1) if len(ans)<M else print(len(ans),*ans[1:])
