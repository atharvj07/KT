#include <bits/stdc++.h>
using namespace std;
int n, cnt;
string s;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    cin >> s;
    cnt = 0;
    for (int i = 0; i < s.size(); i++) {
      cnt += (s[i] == '0');
    }
    if (cnt > s.size() - cnt) {
      cout << cnt << endl;
    } else {
      cout << s.size() - cnt - 1 << endl;
    }
  }
  return 0;
}
