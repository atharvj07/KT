import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int t = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur + a[i] <= t) {
                ans++;
                cur += a[i];
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
