import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), untreated = 0;
        Queue<Integer> q = new LinkedList<>();
        for(int i = 0, a; i < n; i++) {
            if((a = sc.nextInt()) == -1) {
                if(q.isEmpty()) untreated++;
                else q.poll();
            } else for(int j = 0; j < a; j++) q.add(i);
        }
        System.out.print(untreated);
    }
}
