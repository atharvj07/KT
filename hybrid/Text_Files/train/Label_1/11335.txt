#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<vector<int>> a(n, vector<int>(m, 0));
    while (k--) {
        int t, x, c;
        cin >> t >> x >> c;
        --x;
        for (int i = 0; i < m; ++i) a[x][i] = t == 1 ? c : a[x][i];
        for (int i = 0; i < n; ++i) a[i][x] = t == 2 ? c : a[i][x];
    }
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) cout << a[i][j] << ' ';
        cout << '\n';
    }
    return 0;
}
