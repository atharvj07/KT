import java.util.LinkedList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        LinkedList<LinkedList<Integer>> lanes = new LinkedList<>();

        for (int i = 0; i < n; i++) {
            lanes.add(new LinkedList<>());
        }

        for (int i = 0; i < m; i++) {
            int type = scanner.nextInt();
            int data = scanner.nextInt();

            if (type == 0) {
                System.out.println(lanes.get(data - 1).removeFirst());
            } else {
                int minLane = -1;
                int minSize = Integer.MAX_VALUE;

                for (int j = 0; j < n; j++) {
                    if (lanes.get(j).size() < minSize) {
                        minLane = j;
                        minSize = lanes.get(j).size();
                    }
                }

                lanes.get(minLane).addLast(data);
            }
        }
    }
}
