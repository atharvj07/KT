#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const char nl = '\n';
int main() {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int cnt[32] = {0};
    for (int i = 0; i < n; i++) {
        cnt[(a[i] ^ b[i])]++;
    }
    ll ans = 0;
    for (int i = 0; i < 32; i++) {
        if (cnt[i] == 0) continue;
        ll tmp = cnt[i] * (cnt[i] - 1) / 2;
        ans += tmp;
    }
    cout << ans << nl;
    return 0;
}
