#include <bits/stdc++.h>

using namespace std;

typedef unsigned long long ull;

const ull MOD = 1000000007;

ull fastpow(ull base, ull e) {
    if (e == 0) {
        return 1;
    }
    if (e == 1) {
        return base;
    }
    ull half = fastpow(base, e / 2);
    half = (half * half) % MOD;
    if (e % 2 == 1) {
        half = (half * base) % MOD;
    }
    return half;
}

ull p2(ull n) {
    return fastpow(2, n);
}

ull c(ull n, ull k) {
    return (p2(n) * fastpow(p2(n - k) % MOD * p2(k) % MOD, MOD - 2)) % MOD;
}

int main() {
    ull n;
    cin >> n;
    cout << c(n, n / 2) << endl;
    return 0;
}
