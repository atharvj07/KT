#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        ll n;
        cin>>n;
        vector<ll> arr(n);
        for(int i=0;i<n;i++)
            cin>>arr[i];
        vector<ll> inc(n),dec(n);
        for(int i=0;i<n;i++){
            int k=i-1;
            while(k>=0 && arr[k]<arr[i]){
                k=inc[k]-1;
            }
            inc[i]=k+1;
        }
        for(int i=n-1;i>=0;i--){
            int k=i+1;
            while(k<n && arr[k]<arr[i]){
                k=dec[k]+1;
            }
            dec[i]=k-1;
        }
        ll ans=0;
        for(int i=0;i<n;i++){
            ll b=inc[i],e=dec[i];
            ans+=e-b+1;
            if(i-b>0)
                ans+=i-b-1;
        }
        cout<<ans<<"\n";
    }
    return 0;
}
