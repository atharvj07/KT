#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    bool impossible = false;
    vector<int> ans(n);
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == cnt) {
            ans[cnt] = i + 1;
            cnt++;
            for (int j = 0; j <= i; j++) { a[j]--; }
        }
        if (a[i] > cnt) { impossible = true; }
    }
    if (!impossible) {
        cout << "Possible" << endl;
        for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
    } else {
        cout << "Impossible" << endl;
    }
    return 0;
}
