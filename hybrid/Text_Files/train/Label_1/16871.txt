#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, L = 0, R = 0, c = 0;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        if (i == 1)
            L = x;
        else if (x != L)
            c++, L = x;
        R = max(R, x);
    }
    if (R != L) c++;
    cout << c << endl;
    return 0;
}
