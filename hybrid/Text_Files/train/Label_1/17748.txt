#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int64_t p, q;
  cin >> p >> q;
  int n;
  cin >> n;
  int64_t num = 0, dem = 1;
  for (int i = n - 1; i >= 0; --i) {
    int x;
    cin >> x;
    int64_t t1 = dem;
    dem = dem * x + num;
    num = t1;
  }
  bool ok = p * dem == q * num;
  puts(ok ? "YES" : "NO");
  return 0;
}
