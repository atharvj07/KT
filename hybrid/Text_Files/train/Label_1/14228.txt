#include <bits/stdc++.h>

const int MAXN = 1e6 + 10;
char s[MAXN];

int main() {
  scanf("%s", s);
  int n = strlen(s), ans = 0;
  std::stack<int> st;
  for (int i = 0; i < n; i++) {
    if (s[i] == '<' || s[i] == '{' || s[i] == '[' || s[i] == '(')
      st.push(s[i]);
    else if (!st.empty() && (st.top() == '<' && s[i] == '>' || st.top() == '{' && s[i] == '}' || st.top() == '[' && s[i] == ']' || st.top() == '(' && s[i] == ')'))
      st.pop();
    else {
      ans++;
      if (s[i] == '>')
        st.push('<');
      else if (s[i] == '}')
        st.push('{');
      else if (s[i] == ']')
        st.push('[');
      else
        st.push('(');
    }
  }
  if (st.empty())
    printf("%d\n", ans);
  else
    puts("Impossible");
  return 0;
}
