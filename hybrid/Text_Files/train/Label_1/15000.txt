#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> r(n);
    for (int i = 0; i < n; i++) cin >> r[i];
    int ans = 0, cur = 0;
    for (int i = 0, j = 0; i < n; i++) {
        while (j < n && cur + r[j] <= 100 * (j - i + 1)) cur += r[j++];
        ans = max(ans, j - i);
        cur -= r[i];
    }
    cout << ans << endl;
    return 0;
}
