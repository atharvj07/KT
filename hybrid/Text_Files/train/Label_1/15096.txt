#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, h, a, b, k;
  cin >> n >> h >> a >> b >> k;
  while (k--) {
    long long t1, f1, t2, f2, ans = LLONG_MAX;
    cin >> t1 >> f1 >> t2 >> f2;
    ans = min({ans, abs(f2 - f1), abs(f2 - (h - f1))});
    if (t1 != t2) {
      if (a <= min(f1, f2) && max(f1, f2) <= b)
        ans = min(ans, abs(t2 - t1) + 1);
      else if (a <= min(f1, h - f2) && max(f1, h - f2) <= b)
        ans = min(ans, abs(t2 - t1) + 1);
    }
    cout << ans << '\n';
  }
  return 0;
}
