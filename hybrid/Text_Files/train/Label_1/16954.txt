#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), c(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  vector<int> dp(n, 0);
  dp[0] = max({a[0], b[0], c[0]});
  dp[1] = max({a[1] + b[0], b[1] + max(a[0], c[0]), c[1] + max(a[0], b[0])});
  for (int i = 2; i < n; i++) {
    dp[i] = max({a[i] + dp[i - 1], b[i] + max(a[i - 1], c[i - 1]), c[i] + max(a[i - 1], b[i - 1])});
  }
  cout << dp[n - 1] << endl;
  return 0;
}
