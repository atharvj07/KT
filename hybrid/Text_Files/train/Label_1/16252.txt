#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long int a, b;
    cin >> a >> b;
    long long int sum = 0;
    while (a > 0 && b > 0) {
      if (a == b) {
        a--;
      } else if (a > b) {
        a -= b;
        sum++;
      } else {
        b -= a;
        sum++;
      }
    }
    cout << sum << "\n";
  }
  return 0;
}
