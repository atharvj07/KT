import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
    ans = 0
    sum = 0
    mul = 1
    for i in a:
        sum += i
        mul *= i
    if sum == 0:
        ans += 1
    if mul == 0:
        ans += 1
    print(ans)
