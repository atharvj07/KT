import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n], b = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
      b[i] = sc.nextInt();
    }
    long sum = 0L;
    for (int i : a) sum += i;
    Arrays.sort(b);
    System.out.println(sum <= b[n - 1] + b[n - 2] ? "YES" : "NO");
  }
}
