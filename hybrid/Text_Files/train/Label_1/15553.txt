#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
const ll maxn = 1e5 + 10;
ll n, price[maxn], need[maxn];
ll dp[maxn]; // stores the minimum amount of money for the first i days
void solve() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> need[i] >> price[i];
    }
    dp[0] = 0; // base case: 0 days require 0 money
    for (int i = 1; i <= n; i++) {
        dp[i] = 1LL << 60; // initialize the current minimum to a large value
        for (int j = 1; j <= i; j++) {
            dp[i] = min(dp[i], dp[j - 1] + price[j] * max(need[i] - (j == i ? 0 : need[j]), 0LL));
        }
    }
    cout << dp[n] << endl; // the final answer is stored in dp[n]
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    solve();
    return 0;
}
