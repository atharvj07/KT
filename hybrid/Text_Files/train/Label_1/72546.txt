for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    freq = [0] * (n + 1)
    for x in a:
        freq[x] += 1
    has_odd = False
    odd_count = 0
    for x in freq:
        if x % 2 == 1:
            odd_count += 1
            if odd_count > 1:
                has_odd = True
                break
    if has_odd or max(freq) > 2:
        print('YES')
    else:
        print('NO')
