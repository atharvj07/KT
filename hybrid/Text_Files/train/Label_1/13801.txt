#include <iostream>
#include <vector>
#include <string>
#define rep(i, x, n) for(int i = x; i < n; i++)

int main() {
  int n;
  std::cin >> n;
  std::vector<std::string> v(n);
  rep(i, 0, n) std::cin >> v[i];
  std::vector<bool> f(n, true);
  rep(i, 0, n) {
    rep(j, 0, n) {
      if (i == j) continue;
      bool fl = true;
      rep(k, 0, v[i].size()) {
        if (v[i][k] != v[j][k]) {
          fl = false;
          break;
        }
      }
      if (v[i].size() > v[j].size() && fl) {
        std::cout << "NO" << std::endl;
        return 0;
      }
      if (v[i][v[i].size() - 1] != v[j][v[j].size() - 1] && fl) f[i] = false;
    }
  }
  std::cout << "YES" << std::endl;
  rep(i, 0, n) {
    if (f[i]) std::cout << v[i] << std::endl;
  }
  return 0;
}
