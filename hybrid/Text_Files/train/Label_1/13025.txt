#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(false); cin.tie(NULL);
	int t; cin >> t;
	while(t--){
		int ang; cin >> ang;
		if(ang == 180 || ang == 90) cout << 180 << endl;
		else if(ang == 60) cout << 120 << endl;
		else if(ang == 30) cout << 360 << endl;
		else if(ang != 0) cout << 360 / (180 - ang) << endl;
		else cout << -1 << endl;
	}
	return 0;
}
