import java.util.*;
class Main {
 public static void main(String[] args) {
  Scanner input = new Scanner(System.in);
  int n = input.nextInt();
  int m = input.nextInt();
  boolean next = false;
  for (int i = n + 1; i <= m; i++) {
   boolean flag = true;
   for (int j = 2; j * j <= i; j++) {
    if (i % j == 0) {
     flag = false;
     break;
    }
   }
   if (flag) {
    if (!next) {
     next = true;
    }
    else {
     System.out.println("NO");
     return;
    }
   }
  }
  if (next) {
   System.out.println("YES");
  }
  else {
   System.out.println("NO");
  }
 }
}
