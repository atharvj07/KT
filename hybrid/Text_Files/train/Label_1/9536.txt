#include <bits/stdc++.h>
using namespace std;
typedef long long i64;
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.length();
    unordered_map<char, int> cnt;
    for (int i = 0; i < n; i++) cnt[s[i]]++;
    vector<int> ans;
    for (int i = n - 1; i >= 0; i--) {
      if (cnt[s[i]] > 1) {
        cnt[s[i]]--;
        continue;
      }
      ans.push_back(s[i]);
    }
    reverse(ans.begin(), ans.end());
    for (int i = 0; i < ans.size(); i++) cout << (char)ans[i];
    cout << '\n';
  }
  return 0;
}
