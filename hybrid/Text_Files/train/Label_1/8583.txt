#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int& ai : a) cin >> ai;
    int last = -1;
    for (int ai : a) {
        if (ai > last) {
            last = ai;
            cout << ai << " ";
        }
    }
    cout << endl;
    return 0;
}
