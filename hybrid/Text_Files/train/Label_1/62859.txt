def count_changed_digits(l, r):
    total_changes = 0
    while l < r:
        changes = 0
        l_str, r_str = str(l), str(r)
        for i in range(1, len(l_str) + 1):
            if l_str[-i] != r_str[-i]:
                changes = i
                break
        if changes == 0:
            changes = len(l_str)
        total_changes += changes
        l += 10 ** (changes - 1)
    return total_changes

t = int(input())
for _ in range(t):
    l, r = map(int, input().split())
    print(count_changed_digits(l, r))
