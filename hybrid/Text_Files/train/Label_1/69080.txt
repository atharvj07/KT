def check_in_range(x, y):
    return 0 <= x < 10 and 0 <= y < 10

def check_winning_answer(board):
    for i in range(10):
        row_x = 0
        row_o = 0
        col_x = 0
        col_o = 0
        for j in range(10):
            if board[i][j] == 'X':
                row_x += 1
            elif board[i][j] == 'O':
                row_o += 1
            if board[j][i] == 'X':
                col_x += 1
            elif board[j][i] == 'O':
                col_o += 1
            if row_x >= 5 or row_o >= 5 or col_x >= 5 or col_o >= 5:
                return 'YES'
    for i in range(10):
        d1_x = 0
        d1_o = 0
        d2_x = 0
        d2_o = 0
        for j in range(10):
            if check_in_range(i + j, j):
                if board[i + j][j] == 'X':
                    d1_x += 1
                elif board[i + j][j] == 'O':
                    d1_o += 1
            if check_in_range(i + j, 9 - j):
                if board[i + j][9 - j] == 'X':
                    d2_x += 1
                elif board[i + j][9 - j] == 'O':
                    d2_o += 1
            if d1_x >= 5 or d1_o >= 5 or d2_x >= 5 or d2_o >= 5:
                return 'YES'
    return 'NO'

board = []
for i in range(10):
    board.append(list(input()))

print(check_winning_answer(board))
