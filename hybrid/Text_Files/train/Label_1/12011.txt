#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define rep(i, n) for (int i = 0; i < n; i++)
#define rep2(i, a, b) for (int i = a; i < b; i++)
#define rep3(i, a, b) for (int i = a; i >= b; i--)
#define pb push_back
const int mod = 1e9 + 7;

int main()
{
    int n, m;
    cin >> n >> m;
    int d[n], a[m];
    unordered_map<int, int> cnt;
    int ans = 0;
    rep(i, n)
    {
        cin >> d[i];
        if (d[i])
            cnt[d[i]]++;
    }
    rep(i, m) cin >> a[i];
    int cur = 0;
    rep(i, n)
    {
        if (!d[i])
            continue;
        if (cur >= a[d[i] - 1])
            cur += 1;
        else if (cnt[d[i]] > 1)
        {
            cnt[d[i]]--;
            cur = a[d[i] - 1];
        }
        else
        {
            ans = -1;
            break;
        }
    }
    cout << ans << '\n';
    return 0;
}
