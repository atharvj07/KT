#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    if (k == 1)
      cout << "YES\n" << n << endl;
    else if (n % 2 == 0 && n >= k * 2)
      cout << "YES\n" << (n / 2) << " " << (n / 2) << endl;
    else if (n % 2 && n >= k)
      cout << "YES\n" << 1 << " " << (n - 1) << endl;
    else
      cout << "NO\n";
  }
}
