#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll N = 1e5 + 10;
ll n, ans;
vector<ll> a[N], b[N], id;
map<ll, ll> last;

int main() {
    cin >> n;
    for (ll i = 0, x, y; i < n; i++) {
        cin >> x;
        for (ll j = 0; j < x; j++) {
            cin >> y;
            a[i].push_back(y);
            b[i].push_back(y);
        }
    }
    for (ll i = 0; i < n; i++) {
        for (auto v : a[i]) {
            if (!last[v]) {
                id.push_back(v);
            }
            b[i].push_back(v);
            last[v] = i + 1;
        }
    }
    for (ll i = 0; i < n; i++) {
        for (ll j = 0; j < n; j++) {
            for (ll k = a[i].size() - 1; k >= 0; k--) {
                if (b[j][k] != a[i][k]) {
                    ans += i != j;
                    break;
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
