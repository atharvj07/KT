#include <cstdio>
#include <vector>
long long mod = 1e9 + 7;
long long pow(long long x, long long y) {
  long long ans = 1;
  while (y) {
    if (y % 2) ans = ans * x % mod;
    y /= 2;
    x = x * x % mod;
  }
  return ans;
}
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  long long ans = 0, base = (pow(n, m) * pow((n + 1) / 2, m)) % mod;
  for (int i = 1; i <= m; i++) ans = (ans + base * i) % mod;
  printf("%lld\n", (ans % mod + mod) % mod);
  return 0;
}
