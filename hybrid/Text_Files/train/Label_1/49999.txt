import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int k = Integer.parseInt(input[0]);
        int n = Integer.parseInt(input[1]);
        int maxb = Integer.parseInt(input[2]);
        int t = Integer.parseInt(input[3]);

        int[] sizes = new int[k];
        for (int i = 0; i < k; i++) {
            int[] a = new int[n*t];
            String[] bs = br.readLine().split(" ");
            for (int j = 0; j < n; j++) {
                for (int l = 0; l < t; l++) {
                    a[j*t + l] = (Integer.parseInt(bs[j]) % (j+1));
                }
            }
            int lisLength = getLisLength(a);
            sizes[i] = lisLength;
        }
        System.out.println(Arrays.toString(sizes).substring(1, Arrays.toString(sizes).length()-1).replace(",", "").replace(" ", ""));
    }

    public static int getLisLength(int[] a) {
        int[] dp = new int[a.length];
        Arrays.fill(dp, 1);
        for (int i = 1; i < a.length; i++) {
            for (int j = 0; j < i; j++) {
                if (a[i] > a[j] && dp[i] <= dp[j]) {
                    dp[i] = dp[j] + 1;
                }
            }
        }
        return IntStream.of(dp).max().orElse(0);
    }
}
