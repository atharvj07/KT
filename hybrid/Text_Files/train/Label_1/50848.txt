import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int d = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();

        if (n <= 0 || d <= 0 || a <= 0 || b <= 0) {
            System.out.println(0);
            return;
        }

        int[] clients = new int[n];
        int[] costs = new int[n];

        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            clients[i] = i + 1;
            costs[i] = x * a + y * b;
        }

        boolean[] used = new boolean[n];
        int count = 0;
        int totalCost = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (used[j] || totalCost + costs[j] > d) {
                    continue;
                }
                used[j] = true;
                totalCost += costs[j];
                count++;
                break;
            }
        }

        System.out.println(count);
        for (int i = 0; i < n; i++) {
            if (used[i]) {
                System.out.print(clients[i] + " ");
            }
        }
        System.out.println();
    }
}
