#include <bits/stdc++.h>

using namespace std;

int n, m, a[1000001], b[1000001], ans[1000001];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> a[i]; b[a[i]] = i; }
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        swap(b[a[u]], b[a[v]]);
        swap(a[u], a[v]);
    }
    for (int i = 1; i <= n; i++) ans[i] = a[i];
    sort(ans + 1, ans + n + 1);
    for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
    return 0;
}
