#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i + k - 1 < n; i++) {
    if (a[i] != a[i % k]) ans++;
  }
  cout << ans << endl;
  return 0;
}
