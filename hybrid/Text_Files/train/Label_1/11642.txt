#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        vector<ll> h(n);
        for (auto& x : h) cin >> x;
        bool failed = 0;
        for (ll i = 1; i < n; i++) {
            if (max(h[i], h[i - 1]) - min(h[i], h[i - 1]) > k - 1) {
                failed = 1;
            }
        }
        if (!failed) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}
