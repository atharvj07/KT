n = len(s)
cnt = {}
for c in s:
    if c not in cnt:
        cnt[c] = 0
    cnt[c] += 1
k = 0
one = 0
for c in cnt:
    k += cnt[c] // 2
    cnt[c] %= 2
    one += cnt[c]
k += int(one > 0)
print(k)
ans = []
for c in cnt:
    if cnt[c] == 1:
        ans.append(c)
        one -= 1
        if one == 0:
            break
for c in cnt:
    if cnt[c] > 1:
        for i in range(cnt[c] // 2):
            ans.append(c * 2)
