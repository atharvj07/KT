#include <bits/stdc++.h>

using namespace std;

const int MAXN = 51;

char words[MAXN][MAXN];
char guess[MAXN];
int n, m;

int main() {
  cin >> n;
  cin >> guess;
  cin >> m;
  for (int i = 0; i < m; i++) { cin >> words[i]; }

  int ans = 0;
  for (int i = 0; i < 26; i++) {
    bool ok = false;
    for (int j = 0; j < n; j++) {
      if (guess[j] == '*') {
        for (int k = 0; k < m; k++) {
          if (words[k][j] == 'a' + i) {
            ok |= (j == 0 || guess[j - 1] != 'a' + i);
            ok |= (j == n - 1 || guess[j + 1] != 'a' + i);
          }
        }
      }
    }
    ans += ok;
  }
  cout << ans << endl;
  return 0;
}
