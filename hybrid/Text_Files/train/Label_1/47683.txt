import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        long[][] grid = new long[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
        long mval = Long.MIN_VALUE;
        int ax = -1, ay = -1;
        for (int x = k - 1; x < n - k + 1; x++) {
            for (int y = k - 1; y < m - k + 1; y++) {
                long sum = 0;
                for (int i = x - k + 1; i <= x + k - 1; i++) {
                    for (int j = y - k + 1; j <= y + k - 1; j++) {
                        sum += grid[i][j];
                    }
                }
                if (sum > mval) {
                    mval = sum;
                    ax = x + 1;
                    ay = y + 1;
                }
            }
        }
        System.out.println(ax + " " + ay);
    }
}
