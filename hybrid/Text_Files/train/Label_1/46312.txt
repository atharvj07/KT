import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n + 1];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        long ans = 0, sum = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (arr[i] == -1) {
                ans += sum;
                sum += i;
            } else {
                sum -= arr[i];
            }
        }
        ans %= 998244353;
        System.out.println(ans);
    }
}
