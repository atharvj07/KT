#include <bits/stdc++.h>

using namespace std;

const int M = 998244353;
const int N = 2e5 + 5;

int n, k;
int a[N];
int dp[N][505];

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = n - 1; i >= 0; i--) {
        dp[i][0] = a[i];
        for (int j = 1; j <= k; j++) {
            if (i + j < n) dp[i][j] = max(a[i], dp[i + j][j - 1]);
        }
    }
    int mx = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
        int sum = 0, now = 0;
        for (int j = i; j < n; j++) {
            sum += dp[j][k - now - 1];
            if (sum > mx) mx = sum, cnt = 1;
            else if (sum == mx) cnt++;
            now++;
            if (now == k) break;
        }
    }
    cout << mx << " " << cnt % M;
}
