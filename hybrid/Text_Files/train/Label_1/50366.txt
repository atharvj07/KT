import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] sticks = new int[4];
        for (int i = 0; i < 4; i++) {
            sticks[i] = scanner.nextInt();
        }
        scanner.close();

        // Sort the sticks in ascending order
        for (int i = 0; i < 3; i++) {
            for (int j = i + 1; j < 4; j++) {
                if (sticks[i] > sticks[j]) {
                    int temp = sticks[i];
                    sticks[i] = sticks[j];
                    sticks[j] = temp;
                }
            }
        }

        // Check if it is possible to construct a non-degenerate triangle
        if (sticks[0] + sticks[1] > sticks[2]) {
            System.out.println("TRIANGLE");
        }
        // Check if it is possible to construct a degenerate triangle
        else if (sticks[0] + sticks[1] == sticks[2]) {
            System.out.println("SEGMENT");
        }
        // Otherwise, it is impossible to construct any triangle
        else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
