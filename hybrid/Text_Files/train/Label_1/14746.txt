#include <bits/stdc++.h>

using namespace std;

const int MAXN = 101;

int n, k;
char a[MAXN][MAXN];
int cnt[MAXN][MAXN];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> a[i][j];
            if (a[i][j] == '.') {
                for (int dx = -k + 1; dx <= k - 1; dx++) {
                    if (1 <= i + dx <= n && a[i + dx][j] != '#') cnt[i + dx][j]++;
                    if (1 <= j + dx <= n && a[i][j + dx] != '#') cnt[i][j + dx]++;
                }
            }
        }
    }
    int ans = 0, ansx, ansy;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (cnt[i][j] > ans) {
                ans = cnt[i][j];
                ansx = i;
                ansy = j;
            }
        }
    }
    cout << ansx << " " << ansy << endl;
    return 0;
}
