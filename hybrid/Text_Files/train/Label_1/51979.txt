import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (Math.abs(a[i] - a[j]) >= 2 * Math.abs(a[i]) && Math.abs(a[i] - a[j]) >= 2 * Math.abs(a[j])) {
                    cnt++;
                }
            }
        }
        System.out.println(cnt);
    }
}
