def find_picked_numbers(n, ranges):
    range_to_d = {}
    for l, r in sorted(ranges, key=lambda x: (x[0], -x[1])):
        if (l, r) in range_to_d:
            continue
        if l == r:
            range_to_d[(l, r)] = l
        else:
            for d in range(l, r + 1):
                if (l, d - 1) not in range_to_d and (d + 1, r) not in range_to_d:
                    range_to_d[(l, r)] = d
                    break
    return range_to_d

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        ranges = [tuple(map(int, input().split())) for _ in range(n)]
        result = find_picked_numbers(n, ranges)
        for l, r in sorted(result.keys()):
            print(l, r, result[(l, r)])

solve()
