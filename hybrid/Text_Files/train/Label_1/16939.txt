#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main() {
  int cnt[7] = {0};
  for (int i = 0; i < 24; i++) {
    int a;
    cin >> a;
    cnt[a]++;
  }
  int choice = 0;
  for (int i = 1; i <= 6; i++) {
    if (cnt[i] % 2) {
      if (choice) {
        cout << "NO" << endl;
        return 0;
      }
      choice = i;
    }
  }
  cout << "YES" << endl;
  return 0;
}
