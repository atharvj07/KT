import java.io.*;
import java.util.*;
public class Main {
    static final int MAXN = 200005;
    static int n, m, s;
    static long[] a = new long[MAXN];
    static int[] l = new int[MAXN], r = new int[MAXN], x = new int[MAXN];
    static long[] ans = new long[MAXN];
    static void solve() {
        PriorityQueue<Long> que = new PriorityQueue<>((a, b) -> Long.compare(b, a));
        for(int i = 1; i<= n; i++) que.add(a[i]);
        for(int i = 1; i <= s; i++) {
            int top = que.size();
            for(int j = 1; j <= top - 1; j++) {
                long val = que.remove();
                que.add(val + x[i]);
            }
            ans[i] = ans[i - 1];
            while(l[i] <= r[i]) {
                long val = que.remove();
                ans[i] += Math.max(0, ans[i - 1] + x[i] - val);
                que.add(val + x[i]);
                l[i]++;
            }
            System.out.println(ans[i]);
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] str = br.readLine().split(" ");
        n = Integer.parseInt(str[0]);
        m = Integer.parseInt(str[1]);
        str = br.readLine().split(" ");
        for(int i = 1; i <= n; i++) a[i] = Integer.parseInt(str[i - 1]);
        s = Integer.parseInt(br.readLine());
        for(int i = 1; i <= s; i++) {
            str = br.readLine().split(" ");
            l[i] = Integer.parseInt(str[0]);
            r[i] = Integer.parseInt(str[1]);
            x[i] = Integer.parseInt(str[2]);
        }
        solve();
    }
}
