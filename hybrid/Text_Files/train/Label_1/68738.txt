import heapq

n, m = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

cities = list(map(int, input().split()))

dist = [[-1]*(n+1) for _ in range(n+1)]

def dijkstra(start, idx):
    q = [(0,start)]
    while q:
        d, u = heapq.heappop(q)
        if dist[idx][u] != -1: continue
        dist[idx][u] = d
        for v in adj[u]:
            if dist[idx][v] == -1:
                heapq.heappush(q, (d+1, v))

for c in cities:
    dijkstra(c, c)

ans = []
for i in range(1, n+1):
    mx = -1
    for c in cities:
        if dist[c][i] != -1:
            mx = max(mx, dist[c][i])
    if mx != -1:
        ans.append((mx, i))

ans.sort()
print(ans[0][1])
print(ans[0][0])
