#include<bits/stdc++.h>
#define ll long long 
using namespace std;
const ll M = 1e9+7;
ll p[101][2];
void init(ll m){
    p[0][0] = 1; p[0][1] = 0;
    for(ll i = 1; i <= 100; i++){
        ll x = m/i + (m%i != 0);
        p[i][0] = (x * p[i-1][0]) % M;
        p[i][1] = (x * p[i-1][1] + p[i-1][0]) % M;
    } 
}
int main(){
    ll n, x, d[101];
    cin >> n >> x;
    init(x);
    for(int i = 0; i < n; i++) cin >> d[i];
    ll sum = 0;
    for(int i = 0; i < n; i++){
        sum += p[d[i]][1];
        sum %= M;
    }
    cout << sum << endl;
    return 0;
}
