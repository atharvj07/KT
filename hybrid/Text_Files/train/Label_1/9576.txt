#include <bits/stdc++.h>
#include <algorithm>
#include <vector>

using namespace std;

struct Point {
    long long x, y;
    Point(long long x, long long y) : x(x), y(y) {}
    bool operator<(const Point &other) const {
        return make_pair(max(x, y), x + y) < make_pair(max(other.x, other.y), other.x + other.y);
    }
};

long long dis(Point a, Point b) {
    return abs(a.x - b.x) + abs(a.y - b.y);
}
int main() {
    int n;
    cin >> n;
    vector<Point> v;
    for (int i = 0; i < n; i++) {
        long long x, y;
        cin >> x >> y;
        v.push_back(Point(x, y));
    }
    sort(v.begin(), v.end());
    long long ans = 0;
    for (int i = 0; i < n;) {
        int j = i;
        while (j < n && max(v[i].x, v[i].y) == max(v[j].x, v[j].y)) {
            j++;
        }
        Point cur = Point(0, 0);
        for (int k = i; k < j; k++) {
            ans += dis(cur, v[k]);
            cur = v[k];
        }
        i = j;
    }
    cout << ans << "\n";
    return 0;
}
