#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, h;
    cin >> n >> h;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end(), greater<ll>());
    ll ans = 0, rem = h;
    for (ll i = 0; i < n; i++) {
      if (rem > 0) {
        ans++;
        rem -= a[i];
      }
    }
    cout << ans << "\n";
  }
  return 0;
}
