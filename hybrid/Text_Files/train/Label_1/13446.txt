#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 3e5 + 5, S = 30;
int par[N], sub[N], sz[N], d[N], ans[N], ed[N], col[N], cho;
vector<ll> g[N], vec;
bool vis[N], ok;

void dfs(int v, int p) {
	par[v] = p;
	for (auto u : g[v]) {
		if (u == p) continue;
		dfs(u, v);
		sub[v] += sub[u];
	}
	sub[v]++;
}

void dfs2(int v, int p, int c) {
	col[v] = c;
	for (auto u : g[v]) {
		if (u == p || col[u] != -1) continue;
		if (c == cho) dfs2(u, v, c xor 1);
		else dfs2(u, v, c);
	}
	return;
}

void solve(int r) {
	 dfs(r, r);
	 for (int i = 0; i < 3; i++) {
		 queue<int> q;
		 for (int j = 1; j <= sz[r]; j++) {
			 if (sub[j] % 2 == i) q.push(j);
		 }
		 if (q.empty()) continue;
		 ok = true;
		 while (!q.empty()) {
			 int v = q.front(); q.pop();
			 col[v] = i;
			 if (d[v] == 1 && i != 1) {
				 ok = false; break;
			 }
			 else if (d[v] == -1 && i != 0) {
				 ok = false; break;
			 }
			 for (auto u : g[v]) {
				 if (u == par[v]) continue;
				 int nx = i xor 1;
				 if (nx == cho) {
					 if (col[u] == -1) {
						 col[u] = nx;
						 q.push(u);
					 }
					 else if (col[u] != nx) {
						 ok = false; break;
					 }
				 }
			 }
		 }
		 if (ok) {
			 cho = i;
			 for (int j = 1; j <= sz[r]; j++) {
				 for (auto u : g[j]) {
					 if (col[u] != col[j]) vec.push_back(ed[u]), vec.push_back(ed[j]);
				 }
			 }
			 return;
		 }
	 }
	 return;
}

int main() {
	int n, m, x, y; scanf("%d %d", &n, &m);
	for (int i = 1; i <= n; i++) scanf("%d", &d[i]);
	for (int i = 1; i < n; i++) {
		scanf("%d %d", &x, &y);
		g[x].push_back(y);
		g[y].push_back(x);
		ed[i] = i;
	}
	sz[1] = n;
	dfs2(1, 1, -1);
	solve(1);
	m /= 2;
	if (!ok) puts("-1");
	else {
		int l = vec.size();
		printf("%d\n", l / 2);
		for (int i = 0; i < l; i += 2) printf("%d\n", vec[i]);
	}
	return 0;
}
