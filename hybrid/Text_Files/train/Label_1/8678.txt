#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x = a[i] - '0', y = b[i] - '0';
    int c = x - y;
    if (c == 0) continue;
    if (c < 0) {
      ans += abs(c);
      c += 10;
    }
    ans += min(c, 10 - c);
  }
  cout << ans << endl;
  return 0;
}
