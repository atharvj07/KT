import java.util.Scanner;

public class Main {
    static int[][] grid;
    static int n, m, k;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        grid = new int[n][m];
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            grid[x][y] = -1;
        }
        System.out.println(bfs());
    }

    static int bfs() {
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = -1;
            }
        }
        dist[0][0] = 0;
        int[] dx = {0, 0, 1, -1};
        int[] dy = {1, -1, 0, 0};
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == -1 || dist[i][j] != -1) continue;
                int minDist = Integer.MAX_VALUE;
                for (int k = 0; k < 4; k++) {
                    int nx = i + dx[k];
                    int ny = j + dy[k];
                    if (nx < 0 || nx >= n || ny < 0 || ny >= m || grid[nx][ny] == -1) continue;
                    if (dist[nx][ny] != -1) {
                        minDist = Math.min(minDist, dist[nx][ny] + 1);
                    }
                }
                if (minDist != Integer.MAX_VALUE) {
                    dist[i][j] = minDist;
                }
            }
        }
        if (dist[n - 1][m - 1] != -1) {
            ans = dist[n - 1][m - 1];
        }
        return ans;
    }
}
