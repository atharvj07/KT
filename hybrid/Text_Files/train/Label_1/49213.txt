import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int T = in.nextInt();
        for (int i = 0; i < T; i++) {
            int n = in.nextInt(), k = in.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++)
                a[j] = in.nextInt();
            System.out.println(a[k]);
        }
    }
}
