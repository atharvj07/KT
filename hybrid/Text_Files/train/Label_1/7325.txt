#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  while (cin >> t) {
    while (t--) {
      int n;
      cin >> n;
      string s;
      cin >> s;
      int o = 0;
      for (int i = 0; i < n; i++) {
        o += s[i] == '0';
      }
      if (o % 2) {
        cout << "NO" << endl;
        continue;
      }
      string a = "", b = "";
      o = 0;
      for (int i = 0; i < n; i++) {
        if (s[i] == '1') {
          a += '(' + (o % 2), b += '(' + (o % 2), o++;
        } else {
          a += o % 2 ? '(' : ')', b += o % 2 ? ')' : '(';
        }
      }
      cout << "YES" << endl << a << endl << b << endl;
    }
  }
}
