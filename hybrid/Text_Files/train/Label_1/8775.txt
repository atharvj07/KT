#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    LL n; cin >> n;
    vector<LL> a(n);
    for (auto &x : a) cin >> x;
    vector<pair<LL, LL>> ranges(n);
    ranges[0] = {a[0], a[0]};
    for (int i = 1; i < n; ++i) {
        ranges[i].first = ranges[i - 1].second + 1;
        ranges[i].second = ranges[i].first + n - i - 1;
        if (ranges[i - 1].second >= ranges[i].first) {
            cout << "NO" << '\n';
            return 0;
        }
    }
    LL x = ranges[n - 1].second;
    LL y = (a[0] - 1) / x + 1;
    cout << "YES" << '\n' << x << ' ' << y << '\n';
    return 0;
}
