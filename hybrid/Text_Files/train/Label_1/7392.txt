#include <iostream>
#include <vector>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<ll> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i] >> b[i]; }
    ll sum = 0, ans = 0;
    for (int i = 0; i < n; i++) {
      ans += max(0ll, a[i] - sum);
      sum += max(0ll, a[i] - sum) + b[i];
    }
    cout << ans << '\n';
  }
  return 0;
}
