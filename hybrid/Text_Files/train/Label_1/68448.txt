n,a,b,c=int(input()),input(),input(),input()
ca=cb=cc=bc=0
for x,y,z in zip(a,b,c):
    if x==y: ca+=1
    if y==z: cb+=1
    if bc==0 and x!=y: bc=1
for x in range(n):
    if ca<cb and a!=b:
        for i in range(len(a)):
            if a[i]!=b[i]:
                a=a[:i]+b[i]+a[i+1:]
                ca+=1
                break
    if bc and ca<cb:
        b=b[:x]+a[x]+b[x+1:]
        ca+=1
        cb-=1
        if a[x]==b[x]:
            bc=0
    if cb<cc and b!=c:
        for i in range(len(b)):
            if b[i]!=c[i]:
                b=b[:i]+c[i]+b[i+1:]
                cb+=1
                break
    if bc and cb<cc:
        c=c[:x]+b[x]+c[x+1:]
        cb+=1
        cc-=1
        if b[x]==c[x]:
            bc=0
    if ca<cc and a!=c:
        for i in range(len(a)):
            if a[i]!=c[i]:
                a=a[:i]+c[i]+a[i+1:]
                ca+=1
                break
    if bc and ca<cc:
        c=c[:x]+a[x]+c[x+1:]
        ca+=1
        cc-=1
        if a[x]==c[x]:
            bc=0
if ca==cb==cc: print('Draw')
elif ca>cb and ca>cc: print('Kuro')
elif cb>ca and cb>cc: print('Shiro')
else: print('Katie')
