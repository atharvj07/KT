#include <bits/stdc++.h>
using namespace std;
const int maxn=2100;
char a[maxn],b[maxn],s[maxn],t[maxn],c[maxn];
int m,n,k,x,y,z;
int main(){
    cin.getline(a+1,maxn);
    cin.getline(b+1,maxn);
    n=strlen(a+1);
    m=strlen(b+1);
    for(int i=1;i<=n;i++)
    	c[i]=a[n-i+1];
    for(int i=1;i<=n;i++)
    	s[i]=a[i];
    s[n+1]=0;
    t[m+1]=0;
    int p=1,q=m,flag=1;
    while(q>=1){
    	while(s[p]!=b[q]&&p<=n){
    		p++;
    	}
    	if(p>n){
    		flag=0;
    		break;
    	}
    	k++;
    	int l=p;
    	while(s[l]==b[q]&&l<=n&&q>=1){
    		l++;
    		q--;
    	}
    	x=p;
    	y=l-1;
    	p=l;
    }
    if(flag==0){
    	while(q>=1){
    		while(c[p]!=b[q]&&p<=n){
    			p++;
    		}
    		if(p>n){
    			flag=0;
    			break;
    		}
    		k++;
    		int l=p;
    		while(c[l]==b[q]&&l<=n&&q>=1){
    			l++;
    			q--;
    		}
    		x=n-l+1;
    		y=n-p+1;
    		p=l;
    	}
    }
    if(flag==0)
    	cout<<-1;
    else{
    	cout<<k<<endl;
    	for(int i=1;i<=k;i++)
    		cout<<x<<' '<<y<<endl,x=y+1;
    }
    return 0;
}
