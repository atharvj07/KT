#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, c;
    cin >> n >> c;
    vector<int> t(n);
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    int ans = 1, last = t[0];
    for (int i = 1; i < n; i++) {
        if (t[i] - last <= c) {
            ans++;
        } else {
            ans = 1;
        }
        last = t[i];
    }
    cout << ans << endl;
    return 0;
}
