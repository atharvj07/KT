import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edges = []
for i in range(m):
    a, b = map(int, input().split())
    edges.append([a, b])
    edges.append([b, a])
edges.sort()
symmetry = True
for i in range(0, len(edges), 2):
    if edges[i][0] != edges[i][1]:
        symmetry = False
if symmetry:
    print("Yes")
else:
    print("No")
