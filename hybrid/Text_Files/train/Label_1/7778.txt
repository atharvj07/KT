#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int m, k;
        cin >> m >> k;
        vector<int> a(k), x(k, 1);
        for (int i = 0; i < k; i++) cin >> a[i];
        for (int i = 0; i < m - 1; i++) {
            int q, r;
            cin >> q >> r;
            if (r) x[q - 1] = 0;
        }
        for (auto p : x) cout << (p ? 'N' : 'Y');
        cout << endl;
    }
}
