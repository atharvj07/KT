import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            boolean can = true;
            for (int j = 1; j < n; j++) {
                if (a[j] - a[j - 1] > 1) {
                    can = false;
                    break;
                }
            }
            if (can) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
