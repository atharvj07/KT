import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;
class Main{
	static int m;
	static int n;
	static int[][] t;
	static int[] p;
	static int[] r;
	static int gcd(int a,int b){
		if(b==0)	return a;
		return gcd(b,a%b);
	}
	static int union(int a,int b){
		if(p[a]==a && p[b]==b){
			p[b]=a;
			r[a]+=r[b];
			r[b]=1;
			return gcd(r[a],m*n);
		}
		if(p[a]!=a){
			int x=union(p[a],b);
			p[a]=x;
			if(x==1){
				r[a]=1;
				return x;
			}
			r[a]+=r[x];
			r[x]=1;
			return gcd(r[a],m*n);
		}
		if(p[b]!=b){
			int x=union(a,p[b]);
			p[b]=x;
			if(x==1){
				r[b]=1;
				return x;
			}
			r[b]+=r[x];
			r[x]=1;
			return gcd(r[b],m*n);
		}
		return 0;
	}
	public static void main(String[]args)throws Exception{
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
		String[]s=br.readLine().split(" ");
		n=Integer.parseInt(s[0]);
		m=Integer.parseInt(s[1]);
		t=new int [n+1][];
		p=new int[n*m+1];
		r=new int[n*m+1];
		for(int i=0;i<n*m+1;i++){
			p[i]=i;
			r[i]=1;
		}
		for(int i=1;i<=n;i++){
			t[i]=Arrays.stream(br.readLine().split(" ")).mapToInt(x->Integer.parseInt(x)).toArray();
		}
		int ans=0;
		for(int i=1;i<=n;i++){
			for(int j=1;j<=m;j++){
				ArrayList<Integer>list=new ArrayList<>();
				if(i-1>=1){
					int x=union(t[i-1][j],t[i][j]);
					if(x>1)	list.add(x);
				}
				if(j+1<=m){
					int x=union(t[i][j+1],t[i][j]);
					if(x>1)	list.add(x);
				}
				if(i+1<=n){
					int x=union(t[i+1][j],t[i][j]);
					if(x>1)	list.add(x);
				}
				if(j-1>=1){
					int x=union(t[i][j-1],t[i][j]);
					if(x>1)	list.add(x);
				}
				ans+=list.isEmpty()?1:0;
				for(int k=0;k<list.size();k++){
					if(list.get(k)!=1)	ans+=gcd(list.get(k),m*n)-1;
				}
			}
		}
		System.out.println(ans);
	}
}
