#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> p(n);
    for (int i = 0; i < n; ++i) {
        p[i] = i + 1;
    }
    for (int i = 0; i < k; ++i) {
        swap(p[i], p[i + 1]);
    }
    for (int i = 0; i < n; ++i) {
        cout << p[i] << " ";
    }
    return 0;
}
