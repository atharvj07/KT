n = int(input())
s = input()

t = ''
depth = 0
for char in s:
    if char == '(':
        if depth % 2 == 0:
            t += '0'
        else:
            t += '1'
        depth += 1
    else:
        depth -= 1
        if depth % 2 == 0:
            t += '0'
        else:
            t += '1'

print(t)
