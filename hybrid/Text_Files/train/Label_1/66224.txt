import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    a = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
    ans = 0
    for j in range(m):
        pos = [i for i in range(n) if a[i][j] == j + 1][0]
        ans += pos
        for i in range(pos):
            a[i][j], a[i + 1][j] = a[i + 1][j], a[i][j]
    return ans

print(solve())
