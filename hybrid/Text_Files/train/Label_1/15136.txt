#include <iostream>

int main() {
    int n, x;
    std::cin >> n >> x;
    int sum = 0;
    for (int i = 0; i < n; i++) {
        int a;
        std::cin >> a;
        sum += a;
    }
    if (sum == x) {
        std::cout << "YES" << std::endl;
    } else {
        std::cout << "NO" << std::endl;
    }
    return 0;
}
