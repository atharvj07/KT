#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

const int MAXN = 2010;
int n, h, l, r;
int a[MAXN];
vector<int> g[MAXN];

bool ok(int x) {
    return l <= x && x <= r;
}

int dp[MAXN][3][2];
int solve(int x, int y, int z) {
    if (x > n) return 0;
    int &ret = dp[x][y][z];
    if (ret != -1) return ret;
    ret = 0;
    if (y) {
        ret = max(ret, solve(x + 1, ok(a[x] - 1), 1));
        ret = max(ret, solve(x + 1, ok(a[x] - 1), 0) + (ok(a[x] - 1)));
        ret = max(ret, solve(x + 1, ok(a[x]), 1) + (ok(a[x])));
    }
    if (!z) {
        ret = max(ret, solve(x + 1, 0, 1) + (ok(a[x])));
    }
    return ret;
}

int main() {
    cin >> n >> h >> l >> r;
    for (int i = 1; i <= n; ++i) cin >> a[i];

    memset(dp, -1, sizeof dp);
    cout << solve(1, 0, 1) << endl;
    return 0;
}
