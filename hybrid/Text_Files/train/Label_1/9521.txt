#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> cnt(100001);
  for (int i = n - 1; i >= 0; i--) {
    cnt[a[i]]++;
    int tot = 0;
    for (int j = 1; j <= 100000; j++) {
      if (cnt[j] > 0) tot++;
    }
    a[i] = tot;
  }
  while (m--) {
    int x;
    cin >> x;
    cout << a[x - 1] << "\n";
  }
  return 0;
}
