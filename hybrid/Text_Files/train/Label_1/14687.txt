#include <iostream>
#include <cmath>

using namespace std;

long long int gcd(long long int a, long long int b) {
    while (b != 0) {
        long long int t = a % b;
        a = b;
        b = t;
    }
    return a;
}

int main() {
    long long int n, m, k, g = 0, gc = 0;
    cin >> n >> m >> k;
    long long int a[k];
    for (long long int i = 0; i < k; ++i) {
        cin >> a[i];
    }
    for (long long int i = 1; i <= n; ++i) {
        if (i == 1) {
            g = i;
            gc = i;
        } else if (i % 2 != 0) {
            gc = gcd(g, i);
        } else {
            g = gcd(g, i);
        }
        if (i - k + 1 > 0 && g == a[0] && gc == a[k - 1]) {
            long long int z = 0, v = 0;
            for (long long int j = 0; j < k; ++j) {
                if (i - k + 1 + j <= n && g * (i - k + 1 + j) == a[j]) {
                    z++;
                }
                if (gc * (i - k + 1 + j) == a[j]) {
                    v++;
                }
            }
            if (z == k && v == k) {
                cout << "YES\n";
                return 0;
            }
            if (i % 2 != 0) {
                g = gcd(g, i + 1);
            } else {
                gc = gcd(gc, i + 1);
            }
        }
    }
    cout << "NO\n";
    return 0;
}
