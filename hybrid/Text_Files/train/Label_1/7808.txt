#include <bits/stdc++.h>
using namespace std;
int main() {
    long long t, p, f, cnts, cntw, s, w;
    cin >> t;
    while (t--) {
        cin >> p >> f >> cnts >> cntw >> s >> w;
        long long ans = min(cnts, p / s);
        long long sum = ans * s;
        p -= sum;
        ans += min(cntw, p / w);
        cout << ans << '\n';
    }
}
