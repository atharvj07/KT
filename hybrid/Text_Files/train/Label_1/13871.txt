#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
unordered_map<int, ll> mp;
int main() {
  int n, b, k, x;
  cin >> n >> b >> k >> x;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    mp[arr[i]]++;
  }
  vector<int> v;
  for (int i = 0; i < n; i++) {
    if (mp[k - arr[i]]) {
      v.push_back(mp[k - arr[i]] * mp[arr[i]]);
      mp[arr[i]] = 0;
      mp[k - arr[i]] = 0;
    }
  }
  ll ans = 1;
  for (int i = 0; i < v.size(); i++) ans = (ans * v[i]) % mod;
  cout << ans << endl;
  return 0;
}
