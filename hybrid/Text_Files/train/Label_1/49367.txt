import java.util.*;

public class Main {
    static int[][] map;
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        map = new int[n][m];

        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < m; j++) {
                map[i][j] = line.charAt(j) - '0';
            }
        }

        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == 0) {
                    if (dfs(i, j)) {
                        cnt++;
                    }
                }
            }
        }

        System.out.println(cnt);
    }

    static boolean dfs(int i, int j) {
        if (i < 0 || i >= n || j < 0 || j >= m || map[i][j] != 0) {
            return false;
        }

        map[i][j] = 4;

        boolean res = dfs(i + 1, j) || dfs(i - 1, j) || dfs(i, j + 1) || dfs(i, j - 1);

        if (!res) {
            map[i][j] = 0;
        }

        return res;
    }
}
