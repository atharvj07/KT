base = 600
mod = 2 ** base
M = 2000
a = [0] * M
for i in range(int(input())):
  x = int(input())
  for j in range(base - 1, -1, -1):
    if x & (1 << j):
      if a[j] == 0:
        a[j] = i + 1
        break
      x ^= a[j]
  if x != 0:
    print(0)
  else:
    ans = []
    for j in range(base - 1, -1, -1):
      if (x & (1 << j)) and a[j] != 0:
        ans.append(a[j])
    print(len(ans), *ans)
