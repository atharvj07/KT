def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(a, m):
    g = gcd(a, m)
    m //= g
    result = 1
    i = 2
    while i*i <= m:
        if m % i == 0:
            result *= i-1
            m //= i
            while m % i == 0:
                result *= i
                m //= i
        i += 1
    if m > 1:
        result *= m-1
    return result

T = int(input().strip())
for _ in range(T):
    a, m = map(int, input().strip().split())
    print(solve(a, m))
