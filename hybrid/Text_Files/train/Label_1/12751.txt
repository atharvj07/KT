#include <iostream>

long long fact(int n) {
  long long ret = 1;
  for (int i = 1; i <= n; ++i) ret *= i;
  return ret;
}

int main() {
  int k;
  std::cin >> k;
  long long ans = 1;
  for (int i = 0; i < k; ++i) {
    long long c;
    std::cin >> c;
    ans = ans * fact(c) % 1000000007;
  }
  std::cout << ans << std::endl;
}
