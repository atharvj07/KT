import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            HashSet<Integer> set = new HashSet<Integer>();
            int[] re = new int[n];
            for (int j = 1; j <= n; j++) {
                if (!set.contains(j)) {
                    set.add(j);
                    re[j - 1] = j;
                } else {
                    for (int k = 1; k <= 1000; k++) {
                        if (!set.contains(j + k)) {
                            set.add(j + k);
                            re[j - 1] = j + k;
                            break;
                        }
                    }
                }
            }
            for (int j = 0; j < n; j++) {
                System.out.print(re[j] + " ");
            }
            System.out.println();
        }
    }
}
