#include<bits/stdc++.h>
using namespace std;
int main(){
    double r,x1,y1,x2,y2;
    cin>>r>>x1>>y1>>x2>>y2;
    double d=hypot(x2-x1,y2-y1);
    if(d>r+r){
        cout<<x1<<" "<<y1<<" "<<r<<endl;
        return 0;
    }
    double beta=acos((x2-x1)/d);
    cout<<x1+(cos(beta))*(r+r)<<" "<<y1+(sin(beta))*(r+r)<<" "<<r+r<<endl;
}
