import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long x = in.nextLong();
        long y = in.nextLong();

        if (n == 1) {
            System.out.println(x);
            return;
        }

        long ans = n * x;
        for (long i = 1; i < n; i++) {
            long copyTime = i * y;
            long insertTime = (n - i) * x;
            ans = Math.min(ans, copyTime + insertTime);
        }

        System.out.println(ans);
    }
}
