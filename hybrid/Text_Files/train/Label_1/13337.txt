#include <iostream>
#include <vector>

using namespace std;

const long long inf = 1e9;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  long long ans = -1, prefsum = 0;
  for (int i = 0; i < n; ++i) {
    prefsum += a[i];
    ans = min(ans, prefsum);
  }
  ans = max(0LL, -ans);
  cout << ans << endl;
  return 0;
}
