#include <iostream>
#include <vector>
#include <map>
#include <algorithm>
using namespace std;

vector<int> ans, g[100001];
vector<vector<int>> res;
map<int, int> was;
int main() {
  int t;
  cin >> t;
  while (t--) {
    res.clear();
    ans.clear();
    int n, k, x;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
      g[i].clear();
      cin >> k;
      while (k--) {
        cin >> x;
        g[i].push_back(x);
      }
    }
    ans.resize(n + 1);
    bool f = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = 0; j < g[i].size(); ++j) {
        if (was[g[i][j]] == 0) {
          ans[i] = g[i][j];
          was[g[i][j]] = 1;
          break;
        }
      }
      if (ans[i] == 0) {
        for (int j = 1; j <= n; ++j) {
          if (was[j] == 0) {
            ans[i] = j;
            was[j] = 1;
            res.push_back({i, j});
            f = 1;
            break;
          }
        }
      }
    }
    if (f) {
      cout << "IMPROVE\n";
      cout << res[0][0] << ' ' << res[0][1] << '\n';
    } else {
      cout << "OPTIMAL\n";
    }
    was.clear();
  }
  return 0;
}
