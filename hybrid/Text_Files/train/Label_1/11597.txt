#include <iostream>
#include <string>
#include <sstream>
#include <cmath>

int main() {
  std::string input;
  std::cin >> input;

  std::istringstream stream(input);
  double a, e, b;
  stream >> a >> e >> b;

  double result = a * pow(10, b);
  if (result == (int)(result)) {
    std::cout << (int)(result) << std::endl;
  } else {
    std::cout << result << std::endl;
  }
  return 0;
}
