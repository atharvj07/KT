n,m=map(int,input().split())
v=[]
for i in range(n):
 v.append([int(x) for x in input().split()])
v.sort(reverse=True,key=lambda x:x[1])
ans=[0]*m
ans[0]=v[0][1]
cur=v[0][0]
for i in range(1,n):
    if v[i][0]==cur:ans[cur]+=v[i][1]
    else:
        cur=v[i][0]
        ans[cur]=v[i][1]
print(max(ans))
