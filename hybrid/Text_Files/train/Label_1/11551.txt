#include <bits/stdc++.h>
using namespace std;

int n, k, q[101], s[101];
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> q[i];
  for (int i = 0; i < n; i++) cin >> s[i];

  int p[101];
  for (int i = 0; i < n; i++) p[q[i]] = i + 1;

  int cnt = 0;
  for (int i = 0; i < n; i++) cnt -= abs(p[s[i]] - i);

  if (cnt % 2 == k % 2)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
}
