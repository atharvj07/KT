import java.io.*;
import java.util.*;

public class Main {
    static int n, swaps, minSwaps;
    static char[] s, t;
    static List<Pair> answer = new ArrayList<>();

    static class Pair implements Comparable<Pair> {
        int a, b;

        public Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }

        @Override
        public int compareTo(Pair o) {
            if (this.a == o.a) {
                return this.b - o.b;
            }
            return this.a - o.a;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        s = in.readLine().toCharArray();
        t = in.readLine().toCharArray();

        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                swaps++;
            }
        }

        if (swaps % 2 == 1) {
            System.out.println("-1");
            return;
        }

        minSwaps = swaps / 2;

        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                for (int j = i + 1; j < n; j++) {
                    if (s[i] == t[j] && s[j] == t[i]) {
                        Pair p = new Pair(i + 1, j + 1);
                        answer.add(p);
                        char temp = s[i];
                        s[i] = s[j];
                        s[j] = temp;
                        break;
                    }
                }
            }
        }

        System.out.println(minSwaps);
        for (Pair p : answer) {
            System.out.println(p.a + " " + p.b);
        }
    }
}
