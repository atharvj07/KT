import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long min = Long.MAX_VALUE;
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        for (long i = sum / n - 1; i <= sum / n + 2; i++) {
            long cost = 0;
            for (int j = 0; j < n; j++) {
                cost += Math.abs(a[j] - i);
            }
            min = Math.min(cost, min);
        }
        System.out.println(min);
    }
}
