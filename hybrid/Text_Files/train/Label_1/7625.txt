#include <bits/stdc++.h>
using namespace std;
typedef pair<int64_t, int64_t> pi;

pi interval[1000001];
bool infinite = false;
int64_t ans[1000001];

void move(int64_t x, char d) {
  if (d == 'l')
    interval[x] = make_pair(interval[x].second, interval[x].first);
  else
    interval[x] = make_pair(interval[x].first, interval[x].second);
}

int main() {
  string s;
  cin >> s;
  int64_t n = s.size();
  for (int64_t i = 0; i < n; i++) interval[i + 1] = make_pair(0, 1);
  for (int64_t i = 0; i < n; i++) {
    if (s[i] == 'l')
      move(i + 1, 'l');
    else
      move(i + 1, 'r');

    pi now = interval[i + 1];
    bool flag = false;
    for (int64_t j = i + 2; j <= n; j++) {
      if (infinite) {
        flag = true;
        break;
      }
      now = interval[j];
      if (now.first < 0 || now.second > 1) {
        flag = true;
        infinite = true;
        break;
      }
      if (now.first == 0 && now.second == 1)
        break;
      move(j, s[j - 1]);
    }
    if (!flag)
      ans[i + 1] = i + 1;
    else
      break;
  }
  if (infinite)
    cout << "impossible" << endl;
  else
    for (int64_t i = 1; i <= n; i++) {
      if (ans[i] != 0)
        cout << ans[i] << endl;
    }
}
