t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))

    start = end = -1
    diff = 0
    possible = True
    for i in range(n):
        if a[i] != b[i]:
            if start == -1:
                start = i
            end = i
            if diff == 0:
                diff = b[i] - a[i]
            if b[i] - a[i] != diff or diff <= 0:
                possible = False
                break
        else:
            if start != -1 and end != -1 and i - end > 0:
                break

    if possible and start != -1:
        for i in range(start, end + 1):
            a[i] += diff
        if a != b:
            possible = False
            
    print("YES" if possible else "NO")
