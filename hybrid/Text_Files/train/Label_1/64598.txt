for _ in range(int(input())):
    n = int(input())
    a = input()
    b = input()
    cnt0 = a.count('0')
    cnt1 = a.count('1')
    same = 1
    for i in range(n):
        same &= (a[i] == b[i])
    if cnt0 == cnt1 and same:
        print(0)
    elif cnt0 == cnt1:
        print(1)
    else:
        print(-1)
