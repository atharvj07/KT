#include <bits/stdc++.h>
using namespace std;
vector<long long> a;
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    long long x;
    cin >> x;
    a.push_back(x);
  }
  sort(a.begin(), a.end());
  for (int i = 0; i < n - 1; i++) {
    if (a[i + 1] <= a[i]) {
      cout << "NO\n";
      return 0;
    }
  }
  if (a[0] >= a[n - 1]) {
    cout << "NO\n";
    return 0;
  }
  cout << "YES\n";
  cout << a[0] << " ";
  for (int i = n - 1; i >= 1; i--) { cout << a[i] << " "; }
  cout << a[0] << "\n";
}
