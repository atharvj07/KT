#include <algorithm>  
#include <iostream>  
#include <vector>  

using namespace std;  

int main()
{
    int n, f;
    cin >> n >> f;  
    vector<pair<long long, long long>> v(n);  
    for (int i = 0; i < n; i++)
    {
        cin >> v[i].first >> v[i].second;  
        if (v[i].second > v[i].first)
            v[i].second = v[i].first;  
    }
    sort(v.rbegin(), v.rend()); 
    long long ans = 0;
    for (int i = 0; i < min(f, n); i++)
    {
        ans += v[i].second;
        v[i].second *= 2;  
    }
    sort(v.rbegin(), v.rend());  
    for (int i = f; i < n; i++)
        ans += v[i].second;  
    cout << ans << endl;
    return 0;  
}
