import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] stones = new int[14];

        for (int i = 0; i < 14; i++) {
            stones[i] = scanner.nextInt();
        }

        int maxScore = 0;
        for (int i = 0; i < 14; i++) {
            if (stones[i] > 0) {
                int score = simulateMove(stones, i);
                maxScore = Math.max(maxScore, score);
            }
        }

        System.out.println(maxScore);
    }

    private static int simulateMove(int[] stones, int holeIndex) {
        int score = 0;
        int stonesTaken = stones[holeIndex];
        stones[holeIndex] = 0;

        int currentHole = holeIndex + 1;
        while (stonesTaken > 0) {
            stones[currentHole % 14]++;
            stonesTaken--;
            currentHole++;
        }

        for (int i = 0; i < 14; i++) {
            if (stones[i] % 2 == 0) {
                score += stones[i];
                stones[i] = 0;
            }
        }

        return score;
    }
}
