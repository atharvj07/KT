t = list(map(int,input().split()))
t.sort()
a = t[0]
b = t[1]
c = t[2]
d = t[3]
e = t[4]
if(a==b and b==c):
    print(d+e)
elif(a==b):
    sum = a+b+min(c,d,e)
    if(c==d):
        sum = min(sum,a+b+c+2*min(e,d))
    elif(c==e):
        sum = min(sum,a+b+c+2*min(d,e))
    elif(d==e):
        sum = min(sum,a+b+d+2*min(c,e))
    print(sum)
elif(c==d):
    sum = a+b+c+d+min(e,a,b)
    if(a==b):
        sum = min(sum,a+b+c+2*min(d,e))
    elif(a==e):
        sum = min(sum,a+b+c+2*min(d,e))
    elif(b==e):
        sum = min(sum,a+b+c+2*min(d,e))
    print(sum)
elif(d==e):
    sum = a+b+c+d+e+min(a,b,c)
    if(a==b):
        sum = min(sum,a+b+c+2*min(d,e))
    elif(a==c):
        sum = min(sum,a+b+c+2*min(d,e))
    elif(b==c):
        sum = min(sum,a+b+c+2*min(d,e))
    print(sum)
else:
    print(a+b+c+d)
