#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    bool is_s_palindrome = true;
    for (int i = 0; i < n / 2; i++) {
        if (s[i] != s[n - i - 1]) {
            is_s_palindrome = false;
            break;
        }
    }
    cout << (is_s_palindrome ? "TAK" : "NIE") << endl;
    return 0;
}
