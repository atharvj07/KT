#include <iostream>
#include <iomanip>
using namespace std;
int main()
{
    double a[100001];
    int n;
    cin >> n;
    double sum = 0;
    for (int i = 0; i < n; ++i)
    {
        cin >> a[i];
        sum += a[i];
    }
    double x, y;
    long long ans;
    for (int i = 0; i < n; ++i)
    {
        y = a[i] - (int)a[i];
        x = 0.5 - y;
        if (sum < x - y)
            ans = (int)a[i] + 1;
        else
            ans = (int)a[i];
        cout << ans << endl;
        sum -= ans - a[i];
    }
    return 0;
}
