n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

b.sort()
c = [0] * n

for i in range(n):
    idx = n - a[i]
    pos = -1
    low = 0
    high = n - 1
    while low <= high:
        mid = (low + high) // 2
        if (b[mid] >= idx):
            pos = mid
            high = mid - 1
        else:
            low = mid + 1
    if pos == -1:
        c[i] = (a[i] + b[0]) % n
        b.pop(0)
    else:
        c[i] = (a[i] + b[pos]) % n
        b.pop(pos)

print(" ".join(map(str, c)))
