#include <iostream>
#include <map>
using namespace std;
int main() {
  int p = 0, q = 0;
  map<char, int> m = {{'Q', 9}, {'R', 5}, {'B', 3}, {'N', 3}, {'P', 1}};
  string s;
  for (int i = 0; i < 8; i++) {
    cin >> s;
    for (int j = 0; j < 8; j++) {
      if (s[j] >= 'A' && s[j] <= 'Z')
        p += m[s[j]];
      else if (s[j] >= 'a' && s[j] <= 'z')
        q += m[s[j]];
    }
  }
  if (p > q)
    cout << "White";
  else if (p < q)
    cout << "Black";
  else
    cout << "Draw";
  return 0;
}
