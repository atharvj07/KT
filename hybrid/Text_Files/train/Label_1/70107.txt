n, m = map(int, input().split())
grid = [list(map(int, input())) for i in range(n)]

for i in range(n):
    can_skip = True
    for j in range(m):
        if grid[i][j] == 1:
            lights_on = sum(grid[k][j] for k in range(n))
            if lights_on == 1:
                can_skip = False
    if can_skip:
        print('YES')
        exit()
print('NO')
