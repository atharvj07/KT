import java.util.Scanner;

public class UnimodalArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        boolean isUnimodal = checkUnimodal(arr);

        if (isUnimodal) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean checkUnimodal(int[] arr) {
        int increasingIndex = -1;
        int decreasingIndex = -1;

        for (int i = 1; i < arr.length; i++) {
            if (arr[i] < arr[i - 1]) {
                if (increasingIndex == -1) {
                    increasingIndex = i - 1;
                } else {
                    return false; // Not unimodal
                }
            }

            if (arr[i] > arr[i - 1]) {
                if (decreasingIndex == -1) {
                    decreasingIndex = i;
                } else {
                    return false; // Not unimodal
                }
            }
        }

        // Handle cases with only increasing or decreasing elements
        if (increasingIndex == -1 && decreasingIndex == -1) {
            return true;
        }

        if (increasingIndex == -1) {
            return arr[0] <= arr[arr.length - 1]; // Increasing only
        }

        if (decreasingIndex == -1) {
            return arr[arr.length - 1] <= arr[0]; // Decreasing only
        }

        return arr[increasingIndex - 1] <= arr[increasingIndex] && arr[decreasingIndex] >= arr[decreasingIndex - 1];
    }
}
