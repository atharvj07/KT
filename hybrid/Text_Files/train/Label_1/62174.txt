for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    assert len(a) == n
    assert 3 <= n <= 3 * 10**5
    assert 0 <= m <= n//3
    for i in range(n):
        assert 1 <= a[i] <= n
        assert a.count(a[i]) == 1
    b = [0] * n
    cnt = 0
    for i in range(n):
        b[a[i] - 1] = i
    k = []
    for i in range(n):
        if b[i] < i:
            cnt += 1
        elif b[i] > i:
            cnt -= 1
        else:
            k.append(i)
    if cnt == m:
        print(len(k), *k)
    else:
        print(0)
