#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
typedef long long ll; ll arr[N]; int main() { int n; ll mx=0; cin>>n; for(int i=1; i<=n; i++) { int a,b,c; cin>>a>>b>>c; mx=max(mx, 1ll*a*b*c); arr[i]=1ll*a*b*c; } for(int i=1; i<=n; i++) { for(int j=i+1; j<=n; j++) { mx=max(mx, 1ll*arr[i]*arr[j]); mx=max(mx, 1ll*arr[i]*(arr[j]/arr[i])); mx=max(mx, 1ll*(arr[j]/arr[i])*arr[i]); mx=max(mx, 1ll*arr[j]*(arr[i]/arr[j])); } }
  if(mx==1ll*arr[1]*arr[2]) cout<<"2"<<endl<<"1 2"<<endl;
  else { cout<<"1"<<endl; int ii=0; for(int i=1; i<=n; i++) { if(mx==1ll*arr[i]*arr[i]) { ii=i; break; } if(mx==1ll*arr[i]*(arr[i]/arr[i])) { ii=i; break; } if(mx==1ll*(arr[i]/arr[i])*arr[i]) { ii=i; break; } if(mx==1ll*arr[i]*(arr[i]/arr[i])) { ii=i; break; } }
  cout<<ii<<endl; } }
