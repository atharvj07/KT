#include <bits/stdc++.h>
#define INF 1e9

using namespace std;

int32_t main() {
    int n;
    cin >> n;
    int l[n + 1], r[n + 1];
    for (int i = 1; i <= n; i++) {
        cin >> l[i] >> r[i];
    }
    int head = 0, tail = 0;
    for (int i = 1; i <= n; i++) {
        if (l[i] == 0 && tail == 0) {
            tail = i;
        } else if (l[i] == 0 && tail != 0) {
            r[tail] = i;
            tail = i;
        }
        if (r[i] == 0 && n == 1) {
            head = i;
            tail = i;
        } else if (r[i] == 0 && head == 0) {
            head = i;
        } else if (r[i] == 0 && tail != 0) {
            r[tail] = i;
            tail = i;
        }
    }
    for (int i = 1; i <= n; i++) {
        cout << l[i] << " " << r[i] << endl;
    }
    return 0;
}
