#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'

const ll mod = 1e9 + 7;
const char nl = '\n';

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int n, m, k;
    cin >> n >> m;
    vector<int> arr(n);
    for (int& x : arr) cin >> x;
    vector<pair<int, int>> days(m);
    for (auto& x : days) cin >> x.first >> x.second;
    sort(days.begin(), days.end());
    ll cur = 0, ans = 0;
    for (auto& x : days) {
        ll need = arr[x.second - 1] - cur;
        if (need <= 0) continue;
        ans += need / 2 + (need % 2 != 0);
        cur += need;
    }
    cout << ans << endl;
    return 0;
}
