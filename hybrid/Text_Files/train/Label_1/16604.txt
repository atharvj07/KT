#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxN = 1e5 + 5;
ll n, w[maxN], h[maxN];
bool ok;
int32_t main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    cin >> n;
    for(ll i = 0; i < n; i++) cin >> w[i] >> h[i];
    for(ll i = 1; i < n; i++) ok &= (w[i] >= h[i - 1]);
    if(!ok) {
        for(ll i = 1; i < n; i++) ok &= (h[i] <= w[i - 1]);
    }
    if(ok) cout << "YES";
    else cout << "NO";
    return 0;
}
