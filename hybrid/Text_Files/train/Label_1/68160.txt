import sys

n,m=map(int,input().split())
if n<=8:
    a=[0]*n
    p=[]
    for i in range(n):
        a[i]=i+1
    for j in range(n-1):
        if m==1:
            break
        x=m
        t=0
        for i in range(n-j-1):
            if x<=i+2:
                t=i+2
                break
            x-=i+2
        m-=t
        t+=j
        p.append(a[t])
        for i in range(t,n-1):
            a[i]=a[i+1]
        a[n-1]=0
    for i in range(n):
        if a[i]!=0:
            p.append(a[i])
    for i in range(n):
        print(p[i],end=" ")
else:
    l=list(range(n,0,-1))
    while m>1:
        x=m
        for i in range(n-1):
            if x<=i+2:
                t=i+2
                break
            x-=i+2
        m-=t
        print(l[t],end=" ")
        del l[t]
    for i in range(n-m,0,-1):
        print(l[i],end=" ")
