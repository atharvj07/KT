import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt(), a1 = sc.nextInt(), c1 = sc.nextInt();
        int h2 = sc.nextInt(), a2 = sc.nextInt();
        int n = (h2 + a2 - 1) / a1;
        if (n * a1 - h2 < h1 % c1) n++;
        System.out.println(n);
        for (int i = 0; i < n; i++) {
            if (h1 < a2) System.out.println("HEAL");
            else System.out.println("STRIKE");
            h1 += c1 - a2;
        }
    }
}
