#include <bits/stdc++.h>
using namespace std;

int n, k;
vector<pair<pair<int, int>, int>> v;
long long ans = -1e9;

int main() {
    cin >> n >> k;
    v.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first.first >> v[i].first.second;
        v[i].second = i;
    }
    sort(v.begin(), v.end(), [&](const pair<pair<int, int>, int> &a, const pair<pair<int, int>, int> &b) {
        if (a.first.second == b.first.second) return a.first.first > b.first.first;
        return a.first.second > b.first.second;
    });
    for (int i = 0; i < n; i++) {
        int f = v[i].first.first, t = v[i].first.second, id = v[i].second;
        if (t <= k) {
            ans = max(ans, (long long)f);
            continue;
        }
        ans = max(ans, (long long)f - (long long)(t - k));
    }
    cout << ans << '\n';
}
