#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;

const ll maxn = 1e5 + 5;
ll par[maxn], sz[maxn];
ll find_par(ll x) { return par[x] = (par[x] == x ? x : find_par(par[x])); }
void unite(ll a, ll b) { a = find_par(a), b = find_par(b); if (sz[a] < sz[b]) swap(a, b); par[b] = a; sz[a] += sz[b]; }
bool same(ll a, ll b) { return find_par(a) == find_par(b); }

int main() {
    ll n, x1, x2;
    cin >> n >> x1 >> x2;
    ll t, b;
    vector<pi> v;
    for (ll i = 0; i < n; ++i) {
        cin >> t >> b;
        v.push_back({t, b});
    }

    map<ll, ll> slope;
    for (ll i = 0; i < n; ++i) {
        auto [t, b] = v[i];
        if (t != 0) {
            if (slope.count(t)) {
                ll a = slope[t];
                unite(i, a);
            } else {
                ll gp = __gcd(t, b);
                t /= gp, b /= gp;
                slope[t] = i;
            }
        }
    }

    map<pi, bool> exist;
    for (ll i = 0; i < n; ++i) {
        auto [t, b] = v[i];
        if (t == 0) {
            if (exist[{0, b}] and same(x1, x2)) {
                puts("YES");
                return 0;
            }
            exist[{0, b}] = 1;
        }
    }
    for (ll i = 0; i < n; ++i) {
        auto [t, b] = v[i];
        if (t == 0) continue;
        t = find_par(t);
        if (exist[{t, 0}] and same(x1, x2)) {
            puts("YES");
            return 0;
        }
        exist[{t, 0}] = 1;
    }
    puts("NO");
}
