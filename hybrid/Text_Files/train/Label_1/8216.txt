#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e9;

int main() {
    long long a, b;
    cin >> a >> b;
    if (a > b || b > MAXN) {
        cout << "NO" << endl;
        return 0;
    }
    vector<long long> ans;
    ans.push_back(a);
    while (a < b) {
        if (a * 2 <= b) {
            a *= 2;
            ans.push_back(a);
        } else {
            long long c = 1;
            while (a * c <= b) c *= 10;
            c /= 10;
            a = a * 10 + 1;
            ans.push_back(a);
        }
    }
    cout << "YES" << endl;
    cout << ans.size() << endl;
    for (auto x : ans) cout << x << ' ';
    cout << endl;
}
