import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        int[][] cuboids = new int[n][3];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            cuboids[i][0] = Integer.parseInt(line[0]);
            cuboids[i][1] = Integer.parseInt(line[1]);
            cuboids[i][2] = Integer.parseInt(line[2]);
        }

        int[] volumes = new int[n];
        for (int i = 0; i < n; i++) {
            Arrays.sort(cuboids[i]);
            volumes[i] = cuboids[i][0] * cuboids[i][1] * cuboids[i][2];
        }

        int[][][] dp = new int[n][n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 3; k++) {
                    dp[i][j][k] = -1;
                }
            }
        }

        int[][][] prev = new int[n][n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 3; k++) {
                    prev[i][j][k] = -1;
                }
            }
        }

        int maxVol = 0;
        int[] bestI = new int[3];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int newVol = volumes[i] + volumes[j];
                if (newVol > maxVol) {
                    maxVol = newVol;
                    bestI[0] = i;
                    bestI[1] = j;
                }
            }
        }

        int[] best = new int[3];
        for (int k = 0; k < 3; k++) {
            dp[bestI[0]][bestI[1]][k] = volumes[bestI[0]] + volumes[bestI[1]];
            prev[bestI[0]][bestI[1]][k] = -1;
        }

        for (int i = n - 3; i >= 0; i--) {
            for (int j = i + 1; j < n; j++) {
                for (int k = 0; k < 3; k++) {
                    int[] currCuboid = cuboids[i];
                    int[] otherCuboid = cuboids[j];

                    if (k == 0) {
                        if (currCuboid[1] == otherCuboid[1] && currCuboid[2] == otherCuboid[2]) {
                            if (dp[i + 1][j][1] > dp[i][j][0]) {
                                dp[i][j][0] = dp[i + 1][j][1];
                                prev[i][j][0] = 1;
                            }
                            if (dp[i][j + 1][1] > dp[i][j][0]) {
                                dp[i][j][0] = dp[i][j + 1][1];
                                prev[i][j][0] = 2;
                            }
                        }
                        if (currCuboid[1] == otherCuboid[2] && currCuboid[2] == otherCuboid[1]) {
                            if (dp[i + 1][j][2] > dp[i][j][0]) {
                                dp[i][j][0] = dp[i + 1][j][2];
                                prev[i][j][0] = 1;
                            }
                            if (dp[i][j + 1][2] > dp[i][j][0]) {
                                dp[i][j][0] = dp[i][j + 1][2];
                                prev[i][j][0] = 2;
                            }
                        }
                    } else if (k == 1) {
                        if (currCuboid[0] == otherCuboid[0] && currCuboid[2] == otherCuboid[2]) {
                            if (dp[i + 1][j][0] > dp[i][j][1]) {
                                dp[i][j][1] = dp[i + 1][j][0];
                                prev[i][j][1] = 1;
                            }
                            if (dp[i][j + 1][0] > dp[i][j][1]) {
                                dp[i][j][1] = dp[i][j + 1][0];
                                prev[i][j][1] = 2;
                            }
                        }
                        if (currCuboid[0] == otherCuboid[2] && currCuboid[2] == otherCuboid[0]) {
                            if (dp[i + 1][j][2] > dp[i][j][1]) {
                                dp[i][j][1] = dp[i + 1][j][2];
                                prev[i][j][1] = 1;
                            }
                            if (dp[i][j + 1][2] > dp[i][j][1]) {
                                dp[i][j][1] = dp[i][j + 1][2];
                                prev[i][j][1] = 2;
                            }
                        }
                    } else if (k == 2) {
                        if (currCuboid[0] == otherCuboid[0] && currCuboid[1] == otherCuboid[1]) {
                            if (dp[i + 1][j][0] > dp[i][j][2]) {
                                dp[i][j][2] = dp[i + 1][j][0];
                                prev[i][j][2] = 1;
                            }
                            if (dp[i][j + 1][0] > dp[i][j][2]) {
                                dp[i][j][2] = dp[i][j + 1][0];
                                prev[i][j][2] = 2;
                            }
                        }
                        if (currCuboid[0] == otherCuboid[1] && currCuboid[1] == otherCuboid[0]) {
                            if (dp[i + 1][j][1] > dp[i][j][2]) {
                                dp[i][j][2] = dp[i + 1][j][1];
                                prev[i][j][2] = 1;
                            }
                            if (dp[i][j + 1][1] > dp[i][j][2]) {
                                dp[i][j][2] = dp[i][j + 1][1];
                                prev[i][j][2] = 2;
                            }
                        }
                    }

                    if (prev[i][j][k] == -1) {
                        dp[i][j][k] = volumes[i] + volumes[j];
                    }
                }
            }
        }

        int maxK = 0;
        for (int k = 1; k < 3; k++) {
            if (dp[0][n - 1][k] > dp[0][n - 1][maxK]) {
                maxK = k;
            }
        }

        best[0] = 0;
        best[1] = n - 1;
        best[2] = maxK;

        List<Integer> result = new ArrayList<>();
        while (best[0] != -1) {
            result.add(best[0] + 1);
            int next = prev[best[0]][best[1]][best[2]];
            if (next == 1) {
                best[0]++;
            } else if (next == 2) {
                best[1]--;
            } else {
                best[0] = -1;
            }
        }

        Collections.sort(result);
        System.out.println(result.size());
        for (int i : result) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}
