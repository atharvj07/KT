n, m = [int(x) for x in input().split()]
field = [input() for _ in range(n)]
is_good = True
for i in range(n):
    for j in range(m):
        ch = field[i][j]
        cnt = 0
        for dx in range(-1, 2):
            for dy in range(-1, 2):
                x = i + dx
                y = j + dy
                if 0 <= x < n and 0 <= y < m and field[x][y] == '*': cnt += 1
        if ch == '.' and cnt > 0: is_good = False
        if ord('1') <= ord(ch) <= ord('8') and int(ch) != cnt: is_good = False
print('YES' if is_good else 'NO')
