#include <bits/stdc++.h>
using namespace std;
#define ll long long
vector<bool> vis;
vector<vector<int>> a;
int n;
void dfs(int v, int f) {
    vis[v] = true;
    for (int to : a[v]) {
        if (!vis[to]) {
            if (f == 0) {
                dfs(to, 1);
            } else if (f == 1) {
                dfs(to, 0);
            }
        }
    }
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        string s;
        cin >> s;
        a.clear();
        a.resize(2 * n + 1);
        vis.clear();
        vis.resize(2 * n + 1, false);
        for (int i = 0; i < n; i++) {
            if (s[i] == '1') {
                a[i + 1].push_back(i + 1 + n);
                a[i + 1 + n].push_back(i + 1);
            }
        }
        int mx = 0;
        for (int i = 1; i <= 2 * n; i++) {
            if (!vis[i]) {
                int cnt = 0;
                dfs(i, 0);
                for (int j = 1; j <= 2 * n; j++) {
                    if (vis[j]) cnt++;
                }
                mx = max(mx, cnt);
            }
        }
        cout << mx << endl;
    }
}
