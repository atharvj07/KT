#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    ll n;
    cin>>n;
    vector<bool> isPrime(n+1,1);
    isPrime[0]=isPrime[1]=0;
    for(ll i=2;i<=n;++i)
    {
        if(isPrime[i]==1)
        {
            for(ll j=i*i;j<=n;j+=i)
                isPrime[j]=0;;
        }
    }
    vector<ll> v(n+1);
    vector<ll> orig(n+1);
    for(ll i=1;i<=n;++i)
        cin>>v[i];
    ll flag=0;
    for(ll i=1;i<=n;++i)
    {
        orig[v[i]]=i;
    }
    ll k=0;
    for(ll i=1;i<=n;++i)
    {
        if(v[i]==i)
            continue;
        ll pos=orig[i];
        if(isPrime[pos-i])
        {
            cout<<i<<" "<<pos<<endl;
            k++;
            swap(v[i],v[pos]);
            orig[v[pos]]=pos;
            orig[v[i]]=i;
        }
        else
        {
            ll j=pos;
            while(j!=i)
            {
                if(isPrime[j-i])
                {
                    cout<<i<<" "<<j<<endl;
                    k++;
                    swap(v[i],v[j]);
                    orig[v[j]]=j;
                    orig[v[i]]=i;
                }
                j--;
            }
        }
    }
    cout<<k<<endl;
    return 0;
}
