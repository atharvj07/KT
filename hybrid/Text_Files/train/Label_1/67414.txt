def find_broken_time(n, m, k, q, pixels):
    pixels.sort(key=lambda x: x[2])  # Sort pixels by time
    monitor = [[0] * m for _ in range(n)] # Initialize monitor
    for size in range(1, k + 1):
        broken_squares = [[0] * (m - size + 1) for _ in range(n - size + 1)]
        for i in range(q):
            x, y, t = pixels[i]
            x -= 1  # Adjust for 0-based indexing
            y -= 1  # Adjust for 0-based indexing
            monitor[x][y] = 1  # Mark pixel as broken
            if size == 1:
                broken_squares[x][y] = 1
            else:
                for dx in range(max(0, x - size + 1), min(x + 1, n - size + 1)):
                    for dy in range(max(0, y - size + 1), min(y + 1, m - size + 1)):
                        if monitor[dx][dy] and monitor[dx + size - 1][dy] and monitor[dx][dy + size - 1] and monitor[dx + size - 1][dy + size - 1]:
                            broken_squares[dx][dy] += 1
                for dx in range(n - size + 1):
                    for dy in range(m - size + 1):
                        if broken_squares[dx][dy] == size * size:
                            return t
    return -1

n, m, k, q = map(int, input().split())
pixels = [list(map(int, input().split())) for _ in range(q)]
print(find_broken_time(n, m, k, q, pixels))
