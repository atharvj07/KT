#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long hh, cc, tt;
    cin >> hh >> cc >> tt;
    long long ans = 0;
    long double cur = 0;
    while (true) {
      if (abs((cur + hh) / (ans + 1) - tt) < abs((cur + cc) / (ans + 1) - tt)) {
        cur += hh;
      } else {
        cur += cc;
      }
      ans++;
      if (abs((cur + hh) / (ans + 1) - tt) >= abs((cur + cc) / (ans + 1) - tt))
        break;
    }
    cout << ans << endl;
  }
  return 0;
}
