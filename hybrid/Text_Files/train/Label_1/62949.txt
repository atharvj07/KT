t = int(input().strip())
for _ in range(t):
    a = input().strip()
    b = input().strip()
    min_ops = len(a) + len(b)
    for i in range(len(a)+1):
        for j in range(len(b)+1):
            for k in range(min(len(a) - i, len(b) - j) + 1):
                if a[i:i+k] == b[j:j+k]:
                    min_ops = min(min_ops, len(a) + len(b) - 2 * k)
    print(min_ops)
