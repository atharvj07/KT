#include <iostream>
#include <vector>
#include <queue>
using namespace std;

void connect(vector<int> graph[], int u, int v) {
    graph[u].push_back(v);
    graph[v].push_back(u);
}

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> graph[n];
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        connect(graph, u - 1, v - 1);
    }
    int visited[n];
    for (int i = 0; i < n; i++) {
        visited[i] = false;
    }
    queue<int> q;
    q.push(0);
    int max_strength = 0;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (visited[u]) {
            continue;
        }
        visited[u] = true;
        max_strength = max(max_strength, a[u]);
        for (int v : graph[u]) {
            if (!visited[v]) {
                a[v] += a[u];
                q.push(v);
            }
        }
    }
    cout << max_strength << endl;
    return 0;
}
