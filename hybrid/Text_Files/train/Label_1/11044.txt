#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<bool> vis(n);
  int x = 0;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (vis[i]) continue;
    x ^= a[i];
    vector<int> cur;
    for (int j = i; j < n; j++) {
      if ((a[j] ^ x) == 0) {
        cur.push_back(j);
        vis[j] = 1;
      }
    }
    if (cur.size() >= 3) { cout << "YES\n"; } else {
      cout << "NO\n";
      return 0;
    }
    for (auto j : cur) {
      cout << (i + 1) << " " << (j + 1) << " " << (cur[0] + 1) << "\n";
    }
  }
}
