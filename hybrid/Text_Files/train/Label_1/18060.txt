#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1e9+7;
ll pw[1005];
int main()
{
    pw[0]=1;
    for(int i=1;i<=1000;i++) pw[i]=((pw[i-1]<<1)%mod+mod)%mod;
    ll m,n,x,ans=0;
    cin>>m>>n;
    unordered_map<ll,ll> mp;
    while(n--)
    {
        x=0;
        for(int i=0,y;i<m;i++)
        {
            cin>>y;
            x+=y*(pw[i]%mod);
            x%=mod;
        }
        mp[x]++;
    }
    for(auto p:mp)
    {
        ans=(ans+pw[p.second]-1+mod)%mod;
    }
    cout<<ans<<'\n';
    return 0;
}
