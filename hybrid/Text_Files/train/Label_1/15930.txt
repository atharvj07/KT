#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    int ans = -1;
    map<char, bool> vis;
    for (int i = 0; i < n; i++) {
      if (s[i] != t[i]) {
        if (vis[t[i]]) {
          ans = -1;
          break;
        }
        vis[s[i]] = 1;
      }
    }
    if (ans == -1) {
      cout << ans << endl;
      continue;
    }
    ans = 0;
    for (int i = 0; i < n; i++) if (s[i] != t[i]) ans++;
    if (ans % 2) ans--;
    ans /= 2;
    cout << ans << endl;
  }
}
