import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    static BufferedReader br;
    static PrintWriter out;
    static StringTokenizer st;
    static int n, m, k;
    static char[][] a;
    static boolean[][] was;
    static int[] count, dx = {1, -1, 0, 0}, dy = {0, 0, 1, -1};
    public static void main(String[] args)throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(new OutputStreamWriter(System.out));
        //br = new BufferedReader(new FileReader("main.in"));
        //out = new PrintWriter(new FileWriter("main.out"));
        st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        a = new char[n][m];
        was = new boolean[n][m];
        count = new int[m];
        for(int i=0; i<n; i++){
            char[] s = br.readLine().toCharArray();
            for(int j=0; j<m; j++){
                a[i][j] = s[j];
            }
        }
        count();
        for(int i=0; i<m; i++) out.print(count[i] + " ");
        out.close();
    }
    public static void count(){
        for(int i=0; i<m; i++){
            if(was[0][i]) continue;
            Queue<Pair> q = new LinkedList<>();
            q.add(new Pair(0, i));
            was[0][i] = true;
            while(!q.isEmpty()){
                Pair p = q.poll();
                for(int j=0; j<4; j++){
                    if(p.a + dx[j] >= 0 && p.a + dx[j] < n && p.b + dy[j] >= 0 && p.b + dy[j] < m && !was[p.a + dx[j]][p.b + dy[j]]){
                        if(a[p.a + dx[j]][p.b + dy[j]] == 'D'){
                            count[i]++;
                            was[p.a + dx[j]][p.b + dy[j]] = true;
                        }
                        if(a[p.a + dx[j]][p.b + dy[j]] != 'D' && a[p.a + dx[j]][p.b + dy[j]] != '#'){
                            q.add(new Pair(p.a + dx[j], p.b + dy[j]));
                            was[p.a + dx[j]][p.b + dy[j]] = true;
                        }
                    }
                }
            }
        }
    }
    static class Pair{
        int a, b;
        Pair(int a, int b){
            this.a = a;
            this.b = b;
        }
    }
}
