n, m, p = map(int, input().split())
a = sorted(map(int, input().split()))
b = sorted(map(int, input().split()))
print(sum(a[i] <= b[j] for i in range(n) for j in range(m) if a[i] + (j + 1) * p <= a[i + m - 1]))
print(*[i for i in range(1, n - m + 2) if all(b[j] - a[i + j - 1] in (0, p) for j in range(m))])
