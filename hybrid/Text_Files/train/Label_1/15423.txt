#include<bits/stdc++.h>
using namespace std;
bool isvalid(vector<int> &a){
    int mx=0,co=0;
    for(int x:a){
        mx+=x;
        if(!x)co++;
    }
    if(co>=2 || mx%2)return 0;
    return 1;
}
int main(){
    int n,ans=0;
    cin>>n;
    vector<int> a(n),b(n);
    for(auto &x:a)cin>>x;
    if(!isvalid(a)){
        cout<<-1<<endl;
        return 0;
    }
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(i==j)b[j]=0;
            else b[j]=a[j];
        }
        b[i]++;
        if(isvalid(b)){
            ans++;
            cout<<1<<' '<<i+1<<endl;
            return 0;
        }
        b[i]--;
    }
    cout<<-1<<endl;
}
