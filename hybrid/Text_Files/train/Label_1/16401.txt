#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll INF = 1e18;

struct edge {
    ll w;
    int to;
};

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n);
    for (ll& x : a) cin >> x;
    vector<vector<edge>> g(n);
    while (m--) {
        int x, y, w;
        cin >> x >> y >> w;
        --x, --y;
        g[x].push_back({w, y});
        g[y].push_back({w, x});
    }
    vector<ll> dist(n, INF);
    vector<bool> vis(n);
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    pq.push({dist[0] = 0, 0});
    ll ans = 0;
    while (!pq.empty()) {
        while (!pq.empty() && vis[pq.top().second]) pq.pop();
        if (pq.empty()) break;
        int u = pq.top().second;
        pq.pop();
        ans += a[u] / 2;
        vis[u] = true;
        for (edge& e : g[u]) {
            if (!vis[e.to] && dist[e.to] > dist[u] + e.w) {
                dist[e.to] = dist[u] + e.w;
                pq.push({dist[e.to], e.to});
            }
        }
    }
    for (int i = 0; i < n; ++i)
        if (!vis[i]) ans += a[i];
    cout << ans << '\n';
}

int main() {
    cin.tie(0)->sync_with_stdio(0);
    int T = 1;
    while (T--) solve();
}
