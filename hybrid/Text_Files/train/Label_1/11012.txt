#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> v(n);
    for (int i = 0; i < n; ++i)
        cin >> v[i];
    sort(v.begin(), v.end());
    if (k > v[k - 1])
        cout << "-1" << endl;
    else
        cout << v[k - 1] << endl;
    return 0;
}
