#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define int ll
const int mod = 998244353;
int32_t main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    vector<int> active(n);
    for (int i = 0; i < n; i++) {
        int x, y, s;
        cin >> x >> y >> s;
        a[i] = {x, y};
        active[i] = s;
    }
    int cur = 0, time = 0;
    for (int i = 0; i < n; i++) {
        int x = a[i].first;
        if (active[i] == 0) {
            time = (time + (x - cur)) % mod;
            cur = x;
            active[i] = 1;
        }
    }
    int x = a[n - 1].first + 1;
    time = (time + (x - cur)) % mod;
    cout << time << "\n";
    return 0;
}
