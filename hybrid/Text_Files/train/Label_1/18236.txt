#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<vector<int>> dp(n, vector<int>(10, 0));
    if (s[0] != '?')
        dp[0][s[0] - '0'] = 1;
    else
        dp[0][0] = 1;
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < 10; j++) {
            if (s[i] != '?') {
                if (s[i] - '0' == j)
                    dp[i][j] = dp[i - 1][j];
            } else {
                dp[i][j] = dp[i - 1][j];
                if (i > 0 && s[i - 1] != '?' && s[i - 1] - '0' != j)
                    dp[i][j] += dp[i - 1][j];
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < 10; i++)
        ans += dp[n - 1][i];
    cout << ans << endl;
    return 0;
}
