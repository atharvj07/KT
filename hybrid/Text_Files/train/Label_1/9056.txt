#include<iostream>
#include<algorithm>
using namespace std;
const int SIZE = 100001;
int main() {
    int n, a, x[SIZE];
    cin >> n >> a;
    for (int i = 0; i < n; ++i) cin >> x[i];
    sort(x, x + n);
    int l = 0, r = n - 1, ans = 2e9;
    while (l < r) {
        ans = min(ans, abs(x[l] - a) + abs(x[r] - x[l]));
        if (a > x[l]) ++l;
        else --r;
    }
    cout << ans << "\n";
    return 0;
}
