#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  for (int tc = 1; tc <= t; tc++) {
    int n, k;
    cin >> n >> k;
    vector<int> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    int pos = 0;
    for (int i = 0;; i++) {
      if (i >= n || i + 1 == n) { pos = -1; break; }
      if (h[i] < h[i + 1]) { h[i]++; break; }
      pos = i + 1;
      if (k > 1) k--;
      else break;
    }
    cout << pos << '\n';
  }
  return 0;
}
