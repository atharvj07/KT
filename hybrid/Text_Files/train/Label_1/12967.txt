#include <bits/stdc++.h>
using namespace std;
int a[1005];
vector<pair<int, int>> ans;
signed main() {
    int n;
    cin >> n;
    map<int, int> mp, occ;
    vector<int> pos;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        if (!mp.count(a[i])) {
            mp[a[i]] = pos.size();
            pos.push_back(a[i]);
        }
        occ[a[i]]++;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (mp[a[j]] < mp[a[i]])
                ans.push_back({i, j});
        }
    }
    int cnt = 0;
    for (auto it : occ) cnt += it.second * (it.second - 1) / 2;
    if (cnt == ans.size()) {
        cout << ans.size() << endl;
        for (auto it : ans) { cout << it.first << " " << it.second << endl; }
    } else
        cout << -1 << endl;
}
