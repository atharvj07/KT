#include <cstdio>
#include <vector>
using namespace std;

const int MAXN = 1 << 20;
typedef long long int64;

int64 A[4], B[4];

int main() {
  for (int i = 0; i < 2; ++i) scanf("%lld%lld", A + 2 * i, A + 2 * i + 1);
  for (int i = 0; i < 2; ++i) scanf("%lld%lld", B + 2 * i, B + 2 * i + 1);
  for (int i = 0; i < 4; ++i) {
    if (A[i ^ 1] * B[i] == A[i] * B[i ^ 1]) swap(A[i], A[i ^ 1]);
  }
  int64 ans = -1;
  if (A[2] != B[2] || A[3] != B[3])
    goto out;
  else
    ans = 0;
  for (int i = 0; i < 2; ++i) {
    for (int j = 0; j < 2; ++j) {
      if (A[i] % B[j] == 0 && B[j] >= A[i ^ 1]) ans = max(ans, A[i] / B[j] - 1);
    }
  }
out:
  printf("%lld\n", ans);
  if (~ans) printf("%lld %lld\n%lld %lld\n", A[0], A[1], B[0], B[1]);
  return 0;
}
