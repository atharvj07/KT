#include <bits/stdc++.h>
using namespace std;

int main() {
    int tc;
    cin >> tc;
    while (tc--) {
        int n;
        cin >> n;
        double C, T;
        cin >> C >> T;
        vector<pair<int, int>> p(n);
        for (auto &[a, b] : p) { cin >> a >> b; }
        sort(p.begin(), p.end(), [](auto &a, auto &b) { return a.first + a.second < b.first + b.second; });
        double s = 1;
        double ans = 0;
        int i = 0;
        while (T > 0) {
            int r = T / 10.0;
            double ss = s * pow(0.9, r);
            if (i < n) {
                while (i < n && p[i].first / ss <= T) {
                    ans += p[i].second;
                    T -= p[i].first / ss + 10.0;
                    ++i;
                }
            }
            if (T > 0) {
                double tt = min(C * T, T);
                s += tt;
                T -= tt;
            }
        }
        cout << setprecision(9) << ans << endl;
    }
}
