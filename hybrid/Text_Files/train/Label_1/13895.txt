#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll cnt(ll x){
    ll res = 0;
    while(x){
        res += x&1;
        x >>= 1;
    }
    return res;
}
int main(){
    ll m, k;
    cin >> m >> k;
    ll ans = 0;
    while(cnt(ans + 1) != k) ans++;
    while(cnt(ans + m) != k) ans+=m;
    cout << ans << endl;
}
