n=int(input())
a=list(map(int,input().split()))
b=[0 for i in range(0,100000)]
for x in a:
    b[x]+=1
c=[]
for x in range(0,100000):
    if b[x]>0: c.append([x,b[x]])
ans=[]
for x in c:
    v=[0 for i in range(0,x[1])]
    j=0
    for i in a:
        if i==x:
            v[j]=1
            j+=1
    y=-1
    for i in range(0,x[1]-1):
        if v[i+1]-v[i]!=1: y=1
    if y==0: ans.append([x,v[1]-v[0]])
print(len(ans))
for x in ans:
    print(x[0],x[1])
