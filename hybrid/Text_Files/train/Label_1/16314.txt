#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, s;
        cin >> n >> s;
        vector<pair<int, int>> a;
        ll sum = 0;
        for (int i = 0; i < n; ++i) {
            int x, y;
            cin >> x >> y;
            a.push_back({x, y});
            sum += x;
        }
        sort(a.begin(), a.end(), [&](const pair<int, int>& p1, const pair<int, int>& p2) {
            return p1.second < p2.second;
        });
        if (sum <= s) {
            cout << a[(n - 1) / 2].second << "\n";
            continue;
        }
        ll need = sum - s;
        int ans = a[(n - 1) / 2].second;
        for (int i = 0; i < n; ++i) {
            if (a[i].first <= need) {
                ans = a[i].second;
                need -= a[i].first;
            } else {
                ans = a[i].second - (need + a[i].first - 1) / a[i].first;
                break;
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
