class coord:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    def __hash__(self):
        return hash((self.x, self.y))
    def __eq__(self, other):
        return (self.x, self.y) == (other.x, other.y)

A = coord(*map(int, input().split()))
B = coord(*map(int, input().split()))
C = coord(*map(int, input().split()))

def f(p, q):
    ans = set()
    while p != q:
        if p.x < q.x:
            p.x += 1
        elif p.x > q.x:
            p.x -= 1
        elif p.y < q.y:
            p.y += 1
        else:
            p.y -= 1
        ans.add(p)
    return ans

print(len(f(A, B) | f(B, C) | f(C, A)))
print('\n'.join(' '.join(map(str, p.__dict__.values())) for p in (f(A, B) | f(B, C) | f(C, A))))
