def find_permutations(n):
    # Check for permutation p
    if n & (n-1) == 0:  # if n is a power of 2, p does not exist
        print("NO")
    else:
        print("YES")
        p = [0] * n
        for i in range(n):
            p[i] = ((i | (i+1)) % n) + 1
        print(*p)
    
    # Check for permutation q
    if n < 3:  # For n < 3, q does not exist
        print("NO")
    else:
        print("YES")
        if n % 2 == 0:
            q = [i+1 if i % 2 == 0 else i-1 for i in range(1, n+1)]
        else:
            q = [i+1 if i % 2 == 0 else i-1 for i in range(1, n)]
            q.append(n-1)
            q[n-2], q[n-1] = q[n-1], q[n-2]
        print(*q)

n = int(input().strip())
find_permutations(n)
