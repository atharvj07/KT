from collections import Counter

def smallest_possible_n(shuffled_str, remembered_substr):
    # Count occurrences of each digit in the shuffled string and the remembered substring
    counter_shuffled = Counter(shuffled_str)
    counter_remembered = Counter(remembered_substr)
    
    # Subtract remembered substring count from the shuffled string count
    for digit, count in counter_remembered.items():
        counter_shuffled[digit] -= count
    
    # Extract the length of the number (k) and reduce its count
    for k in range(1, 10):
        if counter_shuffled[str(k)] > 0:
            counter_shuffled[str(k)] -= 1
            break
    
    # Reconstruct the smallest possible number
    # Start with remembered substring if it's not leading with zero
    # or if it's the only possibility
    result = ""
    if remembered_substr[0] != '0' or all(c == '0' for c in remembered_substr):
        result = remembered_substr
    else:
        # Find the smallest non-zero digit to start with
        for digit in range(1, 10):
            if counter_shuffled[str(digit)] > 0:
                result += str(digit)
                counter_shuffled[str(digit)] -= 1
                break
        # Add the remembered substring next
        result += remembered_substr
    
    # Add remaining digits sorted to form the smallest number
    for digit in sorted(counter_shuffled.elements()):
        result += digit
    
    return result

# Read input
shuffled_str = input().strip()
remembered_substr = input().strip()

# Print the result
print(smallest_possible_n(shuffled_str, remembered_substr))
