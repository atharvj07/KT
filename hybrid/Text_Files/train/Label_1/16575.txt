#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int mod = 1e9 + 7;
const int mxN = 1e5;
ll f[mxN][2];
vector<ll> adj[mxN];

void dfs(int u, int p, ll l, ll r) {
  f[u][0] = f[u][1] = 1;
  for (int v : adj[u]) {
    if (v == p)
      continue;
    dfs(v, u, l, r);
    if (l <= f[v][0] && f[v][0] <= r) {
      f[u][0] += f[v][1];
      f[u][1] += f[v][1];
      f[u][0] %= mod;
      f[u][1] %= mod;
    } else if (l <= f[v][1] && f[v][1] <= r) {
      f[u][0] += f[v][0];
      f[u][1] += f[v][0];
      f[u][0] %= mod;
      f[u][1] %= mod;
    }
  }
}

int main() {
  ll n, l, r, sum = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> l, r;
    sum = (sum + l + r) % mod;
    adj[l].push_back(r);
    adj[r].push_back(l);
  }
  dfs(1, 0, 1, n);
  cout << sum << endl;
}
