#include<bits/stdc++.h>
#define ll long long
#define ld long double
#define pb push_back
#define F first
#define S second
#define endl "\n"
using namespace std;
 
const int mod = 998244353;
const int mxN = 100000;
 

int main() {
    ll h, q; cin >> h >> q;
    vector<ll> bad;
    unordered_map<ll, ll> good;
    unordered_map<ll, ll> p;
    for(ll i = 1; i <= h; i++){
        ll size = (1 << i);
        for(ll j = 0; j < size - 1; j++){
            p[2 * j + 1] = i;
        }
    }
    ll ans = 0;
    for(ll i = 0; i < q; i++){
        ll node, start, end, bl; cin >> node >> start >> end >> bl;
        ll pr = p[node];
        if(bl == 0){
            bad.pb(node);
        }
        else{
            if(good.count(pr) and good[pr] != start){
                cout << "Game cheated!" << endl;
                return 0;
            }
            good[pr] = start;
        }
    }
    ll sz = bad.size();
    for(ll i = 0; i < sz; i++){
        ll pr = p[bad[i]];
        if(good.count(pr)){
            ll chk = good[pr];
            ll ansss = ( chk ^ (bad[i] - 1) );
            if(p.count(ansss)){
                ans = ansss;
            }
        }
    }
    if(ans){
        cout << ans << endl;
    }
    else{
        if(good.size() == h){
            cout << "Data not sufficient!" << endl;
        }
        else{
        cout << "Game cheated!" << endl;
        }
    }
    return 0;
}
