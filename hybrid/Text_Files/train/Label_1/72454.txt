def g(i, j, a):
    s = 0
    for k in range(min(i, j) + 1, max(i, j) + 1):
        s += a[k]
    return s

n = int(input())
a = list(map(int, input().split()))
ans = 1e18
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        ans = min(ans, (i - j)**2 + g(i, j, a)**2)
print(int(ans))
