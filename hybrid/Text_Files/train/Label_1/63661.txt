import sys, datetime
f = sys.stdin
n,m = tuple(map(int, f.readline().strip().split()))
log = [l.strip() for l in f.readlines()]
if len(log) < m:
    print(-1)
else:
    dt = datetime.datetime.strptime(log[0], '%Y-%m-%d %H:%M:%S:%z')
    for message in log:
        tm = datetime.datetime.strptime(message, '%Y-%m-%d %H:%M:%S:%z')
        if tm - dt >= datetime.timedelta(seconds=n):
            dt = tm - datetime.timedelta(seconds=n)
        if len([m for m in log if (m >= dt.strftime('%Y-%m-%d %H:%M:%S:%z') and m <= tm.strftime('%Y-%m-%d %H:%M:%S:%z'))]) >=m:
            print(dt.strftime('%Y-%m-%d %H:%M:%S:%z'))
            break
    else:
        print(-1)
