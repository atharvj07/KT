import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] a = new int[n][m];
        HashMap<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
                map.put(a[i][j], i);
            }
        }
        int ans = 0;
        int[] row = new int[m];
        for (int i = 1; i <= n * m; i++) {
            if (map.get(i) == null) {
                ans++;
            } else {
                row[i % m]++;
                int rowshift = i % m;
                int cshift = (map.get(i) + rowshift) % m;
                ans += Math.min(rowshift, m - rowshift);
                ans += Math.min(cshift, m - cshift);
            }
        }
        System.out.println(ans);
    }
}
