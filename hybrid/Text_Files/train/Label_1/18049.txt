#include <bits/stdc++.h>
using namespace std;
int t[21][21], d[21][21], c[21], a, b, n, inf = 2e9;
int main() {
  cin >> n >> a >> b;
  for (int i(1); i <= n; i++) cin >> c[i];
  for (int i(1); i <= n; i++)
    for (int j(i); j <= n; j++) {
      if (j == i) {
        if (c[i] == 0)
          t[i][j] = 0;
        else if (c[i] == 1)
          t[i][j] = b;
        else
          t[i][j] = min(a, b);
      } else if (c[i] == c[j])
        t[i][j] = t[i + 1][j - 1];
      else if (c[i] == 0 || c[j] == 0)
        t[i][j] = t[i + 1][j - 1] + a;
      else if (c[i] == 1 || c[j] == 1)
        t[i][j] = t[i + 1][j - 1] + b;
      else
        t[i][j] = min(t[i + 1][j - 1] + a, t[i + 1][j - 1] + b);
    }
  for (int k(0); k <= n; k++)
    for (int i(1); i + k <= n; i++) {
      int j = i + k;
      d[i][j] = inf;
      for (int l(i); l < j; l++) d[i][j] = min(d[i][j], d[i][l] + d[l + 1][j]);
      d[i][j] = min(d[i][j], t[i][j]);
    }
  cout << d[1][n] << "\n";
}
