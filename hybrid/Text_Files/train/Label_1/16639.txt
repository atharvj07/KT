#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
ll power[1000001];
map<pair<ll, int>, ll> dp;
ll solve(ll num, int pos, int rem, bool leading) {
    if (pos == 0) {
        if (rem == 0 and !leading) return 1;
        return 0;
    }
    auto key = make_pair(num, pos);
    if (dp.count(key)) return dp[key];
    ll ans = 0;
    int last_dig = num % 10;
    num /= 10;
    if (!leading) {
        for (int i = 0; i <= 9; i++) {
            ans = (ans + solve(num, pos - 1, (rem * 10 + i) % 5, leading)) % mod;
        }
    } else {
        for (int i = 0; i <= last_dig; i++) {
            ans = (ans + solve(num, pos - 1, (rem * 10 + i) % 5, leading and i == 0)) % mod;
        }
    }
    return dp[key] = ans;
}
int main() {
    string a;
    cin >> a;
    int k;
    cin >> k;
    power[0] = 1;
    for (int i = 1; i <= 1000000; i++) power[i] = (power[i - 1] * 10) % mod;
    ll num = 0;
    for (char c : a) num = (num * 10 + (c - '0')) % mod;
    num = (num * power[k - 1]) % mod;
    cout << solve(num, a.size() * k, 0, true) << '\n';
}
