#include <bits/stdc++.h>
using namespace std;
int a, b, c, d, n;
int sum(int x) { return x * (x + 1) / 2; }
int main() {
  cin >> n >> a >> b >> c >> d;
  int s = b + c + d - a;
  if (s & 1 || s <= n || s > sum(n)) {
    cout << 0 << endl;
    return 0;
  }
  int f1 = (s - n) / 2, f2 = (s + n) / 2;
  int ans = 0;
  for (int x = 0; x <= f2 / 2; x++) {
    for (int y = 0; y <= f2 / 2; y++) {
      if (x + y <= f1 / 2) ans++;
    }
  }
  cout << ans << endl;
}
