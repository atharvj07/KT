import java.util.*;

public class Main {
    static int n, m, q;
    static int[] par;
    static int[] size;
    static int[] maxDist;

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        n = cin.nextInt();
        m = cin.nextInt();
        q = cin.nextInt();
        init(n);
        for (int i = 0; i < m; i++) {
            int a = cin.nextInt() - 1;
            int b = cin.nextInt() - 1;
            merge(a, b);
        }
        for (int i = 0; i < q; i++) {
            int op = cin.nextInt();
            if (op == 1) {
                int x = cin.nextInt() - 1;
                System.out.println(maxDist[find(x)]);
            } else {
                int x = cin.nextInt() - 1;
                int y = cin.nextInt() - 1;
                merge(x, y);
            }
        }
    }

    private static void init(int n) {
        par = new int[n];
        size = new int[n];
        maxDist = new int[n];
        for (int i = 0; i < n; i++) {
            par[i] = i;
            size[i] = 1;
            maxDist[i] = 0;
        }
    }

    private static int find(int x) {
        if (par[x] == x) {
            return x;
        }
        return par[x] = find(par[x]);
    }

    private static void merge(int a, int b) {
        int x = find(a);
        int y = find(b);
        if (x == y) {
            return;
        }
        if (size[x] < size[y]) {
            par[x] = y;
            size[y] += size[x];
            maxDist[y] = Math.max(maxDist[y], Math.max(maxDist[x], maxDist[x] + 1 + maxDist[y]));
        } else {
            par[y] = x;
            size[x] += size[y];
            maxDist[x] = Math.max(maxDist[x], Math.max(maxDist[y], maxDist[y] + 1 + maxDist[x]));
        }
    }
}
