#include<bits/stdc++.h>
using namespace std;
#define ll long long
int32_t main(){
	ll t;
	cin>>t;
	while(t--){
		ll n;
		cin>>n;
		vector<ll>v(n);
		for(ll i=0;i<n;i++){
			cin>>v[i];
		}
		ll a=0,b=0;
		for(ll i=1;i<n-1;i++){
			if(v[i]>v[i-1] && v[i]>v[i+1]) a++;
			if(v[i]<v[i-1] && v[i]<v[i+1]) b++;
		}
		cout<<min(a,b)<<endl;
	}
}
