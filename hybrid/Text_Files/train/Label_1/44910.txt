import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] ln1 = br.readLine().split(" ");
        int n = Integer.parseInt(ln1[0]), r = Integer.parseInt(ln1[1]);
        int[][] ducks = new int[n][2];
        for (int i = 0; i < n; i++) {
            String[] ln = br.readLine().split(" ");
            ducks[i] = new int[]{Integer.parseInt(ln[0]), Integer.parseInt(ln[1])};
        }
        Arrays.sort(ducks, Comparator.comparingInt(d -> d[1]));
        int ans = 0, lastShot = 0;
        for (int[] duck : ducks) {
            if (duck[0] >= lastShot) {
                lastShot = duck[1] + r;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
