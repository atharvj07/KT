import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String w = input.nextLine();
        boolean beautiful = true;
        int[] freq = new int[26];
        for (char c : w.toCharArray()) {
            freq[c - 'a']++;
            if (freq[c - 'a'] % 2 == 1) {
                beautiful = false;
                break;
            }
        }
        System.out.println(beautiful ? "Yes" : "No");
    }
}
