import java.util.*;

public class Main {
    static class Edge {
        int to, w;

        public Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    static int N, K;
    static List<Edge>[] adj;
    static int[] depth, subtreeSize;
    static boolean[] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        K = sc.nextInt();
        adj = new List[N + 1];
        for (int i = 0; i <= N; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < N - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(new Edge(b, 1));
            adj[b].add(new Edge(a, 1));
        }
        depth = new int[N + 1];
        subtreeSize = new int[N + 1];
        visited = new boolean[N + 1];
        dfs1(1, 1, 0);
        int diam = 0;
        int[] ends = new int[2];
        dfs2(1, 1, 0, ends);
        diam = ends[1] - ends[0] + 1;
        if (diam <= K) {
            System.out.println(0);
            return;
        }
        int ans = subtreeSize[ends[0]] - 1;
        ans += subtreeSize[ends[1]] - 1;
        if (ans > N / 2) {
            ans = N - ans;
        }
        System.out.println(ans);
    }

    static void dfs1(int u, int d, int p) {
        depth[u] = d;
        subtreeSize[u] = 1;
        visited[u] = true;
        for (Edge e : adj[u]) {
            int v = e.to;
            if (!visited[v]) {
                dfs1(v, d + 1, u);
                subtreeSize[u] += subtreeSize[v];
            }
        }
    }

    static void dfs2(int u, int d, int p, int[] ends) {
        ends[0] = ends[1] = u;
        visited[u] = true;
        for (Edge e : adj[u]) {
            int v = e.to;
            if (!visited[v]) {
                dfs2(v, d + 1, u, ends);
                if (depth[ends[0]] < depth[v]) {
                    ends[0] = v;
                }
                if (depth[ends[1]] < depth[v]) {
                    ends[1] = v;
                }
            }
        }
    }
}
