#include <bits/stdc++.h>
using namespace std;

int f[75][75][75][75];
char t[75][75];
int n;
void upd(int u, int v, int x, int y, int num) {
  if (f[u][v][x][y] > num) {
    f[u][v][x][y] = num;
  }
}
int main() {
  cin >> n;
  int st[3];
  for (int i = 1; i <= 3; ++i) {
    cin >> st[i];
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n; ++j) {
      cin >> t[i][j];
    }
  }
  memset(f, 0x3f, sizeof f);
  f[st[1]][st[2]][st[3]][n + 1] = 0;
  for (int i = 0; i <= n + 1; ++i) {
    for (int u = 1; u <= n; ++u) {
      for (int v = 1; v <= n; ++v) {
        for (int x = 1; x <= n; ++x) {
          for (int y = 1; y <= n; ++y) {
            if (!i && u == v && v == x && x == y) continue;
            if (u == st[1] && v == st[2] && x == st[3]) continue;
            if (u == st[1] && v == st[3] && x == st[2]) continue;
            if (u == st[2] && v == st[1] && x == st[3]) continue;
            if (u == st[2] && v == st[3] && x == st[1]) continue;
            if (u == st[3] && v == st[1] && x == st[2]) continue;
            if (u == st[3] && v == st[2] && x == st[1]) continue;
            if (f[u][v][x][y] > f[u][v][x][i] + f[u][v][i][y] + 1) {
              f[u][v][x][y] = f[u][v][x][i] + f[u][v][i][y] + 1;
            }
            if (f[u][v][x][y] > f[u][x][v][y] + f[u][x][i][y] + 1) {
              f[u][v][x][y] = f[u][x][v][y] + f[u][x][i][y] + 1;
            }
            if (f[u][v][x][y] > f[v][u][x][y] + f[v][u][i][y] + 1) {
              f[u][v][x][y] = f[v][u][x][y] + f[v][u][i][y] + 1;
            }
            if (f[u][v][x][y] > f[v][x][u][y] + f[v][x][i][y] + 1) {
              f[u][v][x][y] = f[v][x][u][y] + f[v][x][i][y] + 1;
            }
            if (f[u][v][x][y] > f[x][u][v][y] + f[x][u][i][y] + 1) {
              f[u][v][x][y] = f[x][u][v][y] + f[x][u][i][y] + 1;
            }
            if (f[u][v][x][y] > f[x][v][u][y] + f[x][v][i][y] + 1) {
              f[u][v][x][y] = f[x][v][u][y] + f[x][v][i][y] + 1;
            }
          }
        }
      }
    }
  }
  int ans = f[1][2][3][n + 1];
  if (ans == 0x3f3f3f3f) {
    cout << -1 << endl;
    return 0;
  }
  cout << ans << endl;
  for (int u = 1, v = 2, x = 3, y = n + 1;; swap(v, x), swap(u, y)) {
    if (!ans) break;
    if (f[u][v][x][y] == f[u][v][x][n + 1] + f[u][v][n + 1][y] + 1) {
      y = n + 1;
    }
    if (f[u][v][x][y] == f[u][x][v][y] + f[u][x][n + 1][y] + 1) {
      swap(u, x);
      y = n + 1;
    }
    if (f[u][v][x][y] == f[v][u][x][y] + f[v][u][n + 1][y] + 1) {
      swap(u, v);
      y = n + 1;
    }
    if (f[u][v][x][y] == f[v][x][u][y] + f[v][x][n + 1][y] + 1) {
      swap(v, x);
      swap(u, y);
    }
    if (f[u][v][x][y] == f[x][u][v][y] + f[x][u][n + 1][y] + 1) {
      swap(u, x);
      swap(v, y);
    }
    if (f[u][v][x][y] == f[x][v][u][y] + f[x][v][n + 1][y] + 1) {
      swap(v, x);
      swap(u, y);
    }
    cout << u << " " << v << endl;
    --ans;
  }
  return 0;
}
