#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, x;
    cin >> n >> x;

    vector<pair<int, int>> intervals;
    for (int i = 0; i < n; i++) {
        int l, r;
        cin >> l >> r;
        intervals.push_back({l, r});
    }

    sort(intervals.begin(), intervals.end(), [](const pair<int, int>& a, const pair<int, int>& b) { return a.second < b.second; });

    int ans = 0;
    int r = 0;
    for (auto interval : intervals) {
        if (interval.first > r) {
            ans += interval.first - r;
            r = interval.second;
        }
        r = max(r, interval.second);
    }

    cout << ans << endl;
    return 0;
}
