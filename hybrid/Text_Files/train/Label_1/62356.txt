import sys

n = int(input())
x0, y0 = map(int, input().split())
d = {}
for x in range(n):
    c, x, y = input().split()
    x, y = int(x), int(y)
    if c == 'B':
        if (x - x0) == (y - y0) or (x - x0) == -(y - y0):
            d[(x - x0, y - y0)] = True
    elif c == 'R':
        if x == x0 or y == y0:
            d[(x - x0, y - y0)] = True
    elif c == 'Q':
        if (x - x0) == (y - y0) or (x - x0) == -(y - y0) or x == x0 or y == y0:
            d[(x - x0, y - y0)] = True
if (0, 0) in d:
    print('YES')
else:
    print('NO')
