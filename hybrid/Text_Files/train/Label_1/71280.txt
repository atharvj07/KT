def solve_bugs(n, m, s, complexities, abilities, passes):
    students = sorted([(abilities[i], passes[i], i + 1) for i in range(n)], reverse=True)
    bugs = sorted([(complexities[i], i + 1) for i in range(m)])
    
    l, r = 0, m
    res = []
    while l < r:
        mid = (l + r) // 2
        total_passes, current, j = 0, 0, 0
        assigned = [0] * m
        for i in range(m):
            while j < n and students[j][0] < bugs[i][0]:
                j += 1
            if j == n:
                break
            total_passes += students[j][1]
            assigned[bugs[i][1] - 1] = students[j][2]
            current += 1
            if current == mid:
                current = 0
                j += 1
                if j == n:
                    break
        
        if total_passes <= s and current == 0:
            l = mid + 1
            res = assigned
        else:
            r = mid
    
    if not res:
        return "NO"
    else:
        return "YES\n" + " ".join(map(str, res))

n, m, s = map(int, input().split())
complexities = list(map(int, input().split()))
abilities = list(map(int, input().split()))
passes = list(map(int, input().split()))

print(solve_bugs(n, m, s, complexities, abilities, passes))
