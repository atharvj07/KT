#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> p(n + 1), q(n + 1), s(n + 1), t(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> q[i];
  }
  for (int i = 1; i <= n; i++) {
    cin >> s[i];
  }
  for (int i = 1; i <= n; i++) {
    p[i] = i;
  }
  for (int i = 0; i < k; i++) {
    int r = i % 2;
    if (r == 0) {
      for (int j = 1; j <= n; j++) {
        t[j] = p[q[j]];
      }
    } else {
      for (int j = 1; j <= n; j++) {
        t[q[j]] = p[j];
      }
    }
    p = t;
  }
  bool ok = true;
  for (int i = 1; i <= n; i++) {
    if (p[i] != s[i]) {
      ok = false;
    }
  }
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
