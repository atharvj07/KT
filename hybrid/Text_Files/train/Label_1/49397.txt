import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++)
            x[i] = sc.nextInt();
        int[] y = new int[n];
        for (int i = 0; i < n; i++)
            y[x[i] - 1] = i;
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (y[i] > y[i + 1])
                ans += y[i] - y[i + 1];
        }
        System.out.println(ans);
    }
}
