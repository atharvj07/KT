#include <bits/stdc++.h>

using namespace std;

const int mxN = 2e5 + 5;

typedef long long ll;

const ll MOD = 1e9 + 7;

#define PB push_back
#define F first
#define S second

struct cmp {
    bool operator()(const ll &a, const ll &b) {
        return (a > b);
    }
};

int n, k, x;
ll arr[mxN];
set<ll, cmp> odd;

int main() {
    cin >> n >> k >> x;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
        if (arr[i] % x != 0) {
            odd.insert(arr[i] % x);
        }
    }
    for (int i = 0; i < n; i++) {
        if (k == 0) break;
        if (arr[i] % x == 0) {
            if (k % 2) {
                arr[i] -= x;
                k--;
            }
        } else if (*odd.begin() * 2 <= arr[i] % x) {
            arr[i] -= *odd.begin();
            odd.erase(odd.begin());
            k--;
        }
    }
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    return 0;
}
