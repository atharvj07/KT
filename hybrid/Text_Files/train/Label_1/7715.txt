#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int MAXN = 500005;
int n, m;
ll a[MAXN];
int l[MAXN], r[MAXN];
map<ll, vector<int>> loc;
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        loc[a[i]].push_back(i);
    }
    for (int i = 1; i <= m; i++) {
        cin >> l[i] >> r[i];
    }
    for (int i = 1; i <= m; i++) {
        int L = l[i], R = r[i];
        if (loc.count(a[L]) == 0 || loc[a[L]].back() < L) {
            cout << "-1\n";
            continue;
        }
        int pos = upper_bound(loc[a[L]].begin(), loc[a[L]].end(), R) - loc[a[L]].begin() - 1;
        if (pos < 0 || loc[a[L]][pos] < L) {
            cout << "-1\n";
            continue;
        }
        cout << loc[a[L]][pos] - L << "\n";
    }
}
