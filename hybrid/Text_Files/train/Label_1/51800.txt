import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int K = sc.nextInt();
        int[] A = new int[K];
        for (int i = 0; i < K; i++) {
            A[i] = sc.nextInt();
        }

        long min = 1;
        long max = Long.MAX_VALUE;
        for (int i = 0; i < K; i++) {
            if (A[i] > 2) {
                min = Math.max(min, A[i]);
            }
            if (A[i] == 2) {
                max = Math.min(max, 3);
            }
        }

        if (max < min) {
            System.out.println(-1);
        } else {
            System.out.println(min + " " + max);
        }
    }
}
