#include <iostream>
#include <algorithm>
#include <string>

int solve(std::string n) {

    if (std::abs(std::stol(n)) < 10) { return std::stol(n); }

    if (n[0] == '-') {

        if (n[n.size() - 1] % 2 == 0) {

            n = n.substr(0, n.size() - 1);

        } else if (n[n.size() - 2] % 2 == 0) {

            n = n.substr(0, n.size() - 2);
        }

    } else {

        if (n[n.size() - 1] != '0') {

            n = n.substr(0, n.size() - 1);

        } else if (n.size() > 2 && n[n.size() - 2] % 2 == 0) {

            n = n.substr(0, n.size() - 2);
        }
    }
    return std::stol(n);
}

int main() {

    std::string n;
    std::cin >> n;
    std::cout << solve(n) << '\n';
    return 0;
}
