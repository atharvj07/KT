def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

def lcm(a, b):
    return (a * b) // gcd(a, b)

def query(seconds, a, n, d):
    if seconds == 0:
        return max(d)
    for i in range(n):
        d[i] = 1
        for j in range(n):
            if i == j:
                continue
            if gcd(a[i], a[j]) * d[j] % lcm(a[i], a[j]) == 0:
                d[i] += d[j]
    return max(d)

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    q = int(input())
    d = [1] * n
    for w in map(int, input().split()):
        print(query(w, a, n, d))
