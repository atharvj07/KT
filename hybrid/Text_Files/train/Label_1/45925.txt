import java.util.*;

class Main{
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] pos = new int[4];
        for(int i = 1; i <= 3; i++) pos[i] = in.nextInt();
        pos[0] = in.nextInt();
        Arrays.sort(pos);
        System.out.println((long)pos[3] - pos[1] - pos[2] + pos[0]);
    }
}
