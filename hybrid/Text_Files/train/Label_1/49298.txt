import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int ans = a;
        int c = 0;
        while (a >= b) {
            c += a / b;
            a = a / b + a % b;
        }
        ans += c;
        System.out.println(ans);
    }
}
