#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long int n,a1,b1,a2,b2,a3,b3;
    cin>>n>>a1>>b1>>a2>>b2>>a3>>b3;
    long long int x=b1-a1+1;
    long long int y=b2-a2+1;
    long long int z=b3-a3+1;
    if(x+y+a3<=n) 
        cout<<b1<<" "<<b2<<" "<<n-b1-b2;
    else if(x+z+a2<=n)
        cout<<b1<<" "<<n-b1-z<<" "<<z;
    else 
        cout<<n-y-a3<<" "<<y<<" "<<a3;
    return 0;
}
