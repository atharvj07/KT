#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> p(n), q(n);
    for (int i = 0; i < n; i++) { cin >> p[i]; p[i]--; }
    for (int i = 0; i < n; i++) { cin >> q[i]; q[i]--; }
    vector<int> pos_p(n), pos_q(n);
    for (int i = 0; i < n; i++) { pos_p[p[i]] = i; pos_q[q[i]] = i; }
    string s(n, 'a');
    int last_p = 0, last_q = 0;
    for (int i = 0; i < n; i++) {
        if (pos_p[i] >= last_p) {
            s[i] = char('a' + last_p);
            last_p++;
        } else if (pos_q[i] >= last_q) {
            s[i] = char('a' + last_q);
            last_q++;
        } else {
            cout << "NO" << endl;
            return 0;
        }
    }
    if (last_p < k || last_q < k) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    cout << s << endl;
    return 0;
}
