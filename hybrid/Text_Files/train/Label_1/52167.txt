import java.util.Scanner;

public class CokeParty {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        int minLiters = -1;
        for (int i = 0; i < (1 << k); i++) {
            int totalVolume = 0;
            int totalCarbonDioxide = 0;
            for (int j = 0; j < k; j++) {
                if ((i & (1 << j)) > 0) {
                    totalVolume += 1;
                    totalCarbonDioxide += a[j];
                }
            }
            if (totalCarbonDioxide == n * totalVolume) {
                minLiters = Math.min(minLiters, totalVolume);
            }
        }

        System.out.println(minLiters);
    }
}
