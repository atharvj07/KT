#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = a[0];
    for (int i = 1; i <= d; i++) {
      for (int j = 1; j < n; j++) {
        if (a[j] > 0) {
          a[j]--;
          a[j - 1]++;
          ans = max(ans, a[0]);
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
