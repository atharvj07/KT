#include <iostream>
#include <string>
using namespace std;
const int mod = 2000000013;
int n;
string s;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    cin >> s;
    int ans = n;
    char pre = s[n - 1];
    for (int i = n - 1; i >= 0; i--) {
      if (pre == '<' && s[i] == '>') {
        ans--;
      }
      if (s[i] != pre) {
        pre = s[i];
      }
    }
    cout << ans << endl;
  }
}
