#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

struct Ring {
  int a, b, h;
  bool operator<(const Ring& other) const { return this->a != other.a ? this->a < other.a : this->b < other.b; }
};

int main() {
  int n;
  cin >> n;
  vector<Ring> rings(n);
  for (int i = 0; i < n; i++) cin >> rings[i].a >> rings[i].b >> rings[i].h;
  sort(rings.begin(), rings.end());
  vector<int> dp(n, 0);
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < i; j++) {
      if (rings[i].a > rings[j].b) dp[i] = max(dp[i], dp[j] + rings[i].h);
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) ans = max(ans, dp[i]);
  cout << ans << endl;
}
