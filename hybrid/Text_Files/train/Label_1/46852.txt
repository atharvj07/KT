import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int f = sc.nextInt();
    boolean[] canAlice = new boolean[n];
    boolean[] canBob = new boolean[n];

    for (int i = 0; i < n; i++) {
      int b = sc.nextInt();
      int w = sc.nextInt();
      int r = sc.nextInt();
      if (f > r - b) {
        canAlice[i] = true;
        canBob[i] = true;
      } else if (f > r - w) {
        canBob[i] = true;
      } else if (f > w - b) {
        canAlice[i] = true;
      }
    }

    int winAlice = 0;
    int winBob = 0;
    for (int i = 0; i < n; i++) {
      if (canAlice[i]) {
        winAlice++;
      }
      if (canBob[i]) {
        winBob++;
      }
    }

    if (winAlice % 2 == 1) {
      System.out.println("Alice");
    } else {
      System.out.println("Bob");
    }

    if (winBob % 2 == 1) {
      System.out.println("Bob");
    } else {
      System.out.println("Alice");
    }
  }
}
