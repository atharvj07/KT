#include <bits/stdc++.h>
#define pb push_back
#define endl '\n'
using namespace std;
typedef long long ll;

int main() {
    int n, x, y;
    cin >> n >> x >> y;
    vector<int> a(n);
    vector<bool> b(n, 0);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.rbegin(), a.rend());
    for (int i = 0; i < n; i++) {
        while (a[i] > 0 and x <= y) {
            a[i] -= x;
            b[i] = true;
        }
        if (a[i] <= 0) {
            break;
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (b[i] == false) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
