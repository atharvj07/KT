#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long ts, tf, t;
  int n;
  cin >> ts >> tf >> t;
  cin >> n;
  vector<long long> arr(n);
  for (long long &x : arr) { cin >> x; }
  long long ans = -1;
  for (long long i = ts; i <= tf - t; i++) {
    long long j = i;
    bool ok = true;
    for (long long x : arr) {
      if (x <= j) j += t;
      if (j > tf) {
        ok = false;
        break;
      }
    }
    if (ok) {
      ans = i;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
