#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define mp make_pair
typedef long long LL;
typedef unsigned long long ULL;
typedef pair<int,int> PII;
typedef vector<int> VI;
typedef vector<PII> VII;

struct Node {
    VI adj;
    int size;
    double ans;
};

vector<Node> G;

void dfs1 (int v, int p) {
    G[v].size = 1;
    double dp = 0;
    for (const int w : G[v].adj) {
        if (w == p) continue;
        dfs1(w, v);
        G[v].size += G[w].size;
        dp += 1 + G[w].ans;
    }
    G[v].ans = (1 + dp) / G[v].size;
}

int main() {
    int n;
    cin >> n;
    G.resize(n+1);
    for (int i = 1; i < n; i++) {
        int a, b;
        cin >> a >> b;
        G[a].adj.pb(b);
        G[b].adj.pb(a);
    }
    dfs1(1, 1);
    cout << G[1].ans << endl;
    return 0;
}
