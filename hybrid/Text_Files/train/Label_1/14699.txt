#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[1010], b[1010], n, m, x, y, z, i, j;
char c[1010];
int main()
{
    for (cin >> c + 1; c[i]; i++)
        if (c[i] == '(')
            a[++x] = i;
        else
            b[++y] = i;
    if ((x + y) % 2)
        return cout << -1, 0;
    cout << (n = abs(x - y) / 2 - min(x, y)) << '\n';
    for (; n; n--)
        if (x < y)
            cout << 1 << '\n' << a[++x] << '\n';
        else
            cout << 1 << '\n' << b[++y] << '\n';
}
