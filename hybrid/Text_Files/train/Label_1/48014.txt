import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int a[] = new int[n];
            int sum = 0;
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
                sum += a[j];
            }
            Arrays.sort(a);
            if (sum % n == 0) {
                System.out.println(sum / n);
            } else {
                System.out.println(sum / n + 1);
            }
        }
    }
}
