import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    int k = Integer.parseInt(in.readLine());
    long[] c = new long[k];
    for (int i = 0; i < k; i++) {
      c[i] = Long.parseLong(in.readLine());
    }
    Arrays.sort(c);

    long[][] dp = new long[k + 1][k + 1];
    dp[0][0] = 1;
    for (int i = 1; i <= k; i++) {
      for (int j = 0; j <= k; j++) {
        dp[i][j] = dp[i - 1][j];
        if (j > 0) {
          dp[i][j] += dp[i - 1][j - 1] * c[i - 1];
        }
        dp[i][j] %= 1000000007;
      }
    }
    System.out.println(dp[k][k]);
  }
}
