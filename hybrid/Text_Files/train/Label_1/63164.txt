import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    s = sum(a)
    r = 0
    for i in range(1 << n):
        j = 0
        for k in range(n):
            if i & (1 << k):
                j += a[k]
        if j == s - 1:
            r += 1
    print(r)
