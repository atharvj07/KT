from collections import deque
from itertools import repeat

m = int(input())
arr = deque()
for _ in range(m):
    typ, *val = map(int, input().split())
    if typ == 1:
        arr.append(val[0])
    else:
        arr.extend(repeat(arr.popleft(), *val))
for _ in range(int(input())):
    print(arr[int(input()) - 1])
