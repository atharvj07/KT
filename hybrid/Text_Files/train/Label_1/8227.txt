#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
int main(){
    ll h,w,n;
    cin>>h>>w>>n;
    map<ll,ll> mp;
    for(int i=0;i<n;i++){
        ll r,c;
        cin>>r>>c;
        mp[(r-1)*w+c]=1;
    }
    ll dp[h+1][w+1]={0};
    dp[1][1]=1;
    for(ll i=1;i<=h;i++){
        for(ll j=1;j<=w;j++){
            if(mp[(i-1)*w+j]!=1){
                dp[i][j]=(dp[i][j]+dp[i-1][j]%mod+dp[i][j-1]%mod)%mod;
            }
        }
    }
    cout<<dp[h][w]<<endl;
}
