import java.util.Scanner;

public class Main {
  static int N, A;
  static int[] x;
  static long[][] dp;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    N = sc.nextInt();
    A = sc.nextInt();
    x = new int[N + 1];
    for (int i = 1; i <= N; i++) {
      x[i] = sc.nextInt();
    }
    dp = new long[N + 1][A * N + 1];
    for (int i = 0; i <= N; i++) {
      for (int j = 0; j <= A * N; j++) {
        dp[i][j] = -1;
      }
    }
    System.out.println(solve(1, 0));
  }

  static long solve(int idx, int sum) {
    if (idx == N + 1) {
      if (sum == A * (N - 1)) {
        return 1;
      } else {
        return 0;
      }
    }
    if (dp[idx][sum] != -1) {
      return dp[idx][sum];
    }
    long res = solve(idx + 1, sum);
    res += solve(idx + 1, sum + x[idx]);
    dp[idx][sum] = res;
    return res;
  }
}
