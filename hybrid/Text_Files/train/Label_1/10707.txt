#include <iostream>
using namespace std;

int gcd(int a, int b) {
    while (b) {
        a %= b;
        swap(a, b);
    }
    return a;
}

int main() {
    long long n, k;
    cin >> n >> k;
    if (n % k != 0) {
        cout << -1 << '\n';
        return 0;
    }
    long long g = n / k;
    for (int i = 0; i < k - 1; i++) {
        cout << g * i << ' ';
    }
    cout << n - g * (k - 1) << '\n';
    return 0;
}
