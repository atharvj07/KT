#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
#define i64 long long
const int N = 4e5 + 5;
int a[N], b[N], l, x, m, ans[N];
int getMEX() {
  for (int i = 0; i < l; i++) b[a[i]]++;
  int ans = -1;
  for (int i = 0; i < m; i++) {
    if (!b[i]) {
      ans = i;
      break;
    }
  }
  for (int i = 0; i < l; i++) b[a[i]]--;
  return ans;
}
int main() {
  scanf("%d %d", &l, &x);
  m = min(l + 1, x + 1);
  for (int i = 0; i < l; i++)
    scanf("%d", &a[i]);
  sort(a, a + l);
  for (int i = 0; i < l; i++) {
    m += (a[i] % x == 0);
    for (int j = a[i] / x;; j++) {
      m |= (j % x == 0);
      if (a[i] <= j * x) break;
    }
  }
  int t;
  scanf("%lld", &t);
  while (t--) {
    int tmp;
    scanf("%d", &tmp);
    a[l++] = tmp;
    sort(a, a + l);
    ans[l - 1] = getMEX();
    printf("%d\n", ans[l - 1]);
  }
  return 0;
}
