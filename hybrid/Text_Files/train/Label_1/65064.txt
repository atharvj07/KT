n = int(input())
a = list(map(int, input().split()))

first_occurrence = {}
max_sum = {}
for i, value in enumerate(a):
    if value not in first_occurrence:
        first_occurrence[value] = i
    max_sum[value] = max(max_sum.get(value, 0), sum(max(0, x) for x in a[first_occurrence[value]:i + 1]))

best_value, best_sum = max(max_sum.items(), key=lambda x: x[1])
start, end = first_occurrence[best_value], n - 1 - a[::-1].index(best_value)

to_remove = [i + 1 for i in range(start) if a[i] < 0] + [i + 1 for i in range(end + 1, n) if a[i] < 0]

print(best_sum, len(to_remove))
print(*to_remove)
