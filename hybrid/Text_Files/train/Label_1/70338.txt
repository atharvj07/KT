import sys

for t in range(int(input())):
    n, m, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(reverse = True)
    ans = a[m - 1]
    for i in range(k):
        if i < m:
            ans = max(ans, a[i])
        else:
            ans = max(ans, a[i + 1])
    print(ans)
