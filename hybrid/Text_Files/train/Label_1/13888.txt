#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
typedef long long ll;
typedef long double ld;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  ll mx1 = 0, mx2 = 0, c = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] > a[i - 1]) {
      mx1++;
      c++;
    } else {
      c = 0;
    }
  }
  mx2 = mx1;
  for (int i = 1; i < n; i++) {
    if (a[i] > a[i - 1]) {
      mx1++;
    } else {
      if (i < n - 1 && a[i + 1] > a[i - 1]) {
        mx2 = max(mx2, c);
        mx1 = 0;
      } else {
        mx1 = 0;
        c = 0;
      }
    }
  }
  mx2 = max(mx2, c);
  cout << max(mx1 + 1, mx2 + 1) << endl;
  return 0;
}
