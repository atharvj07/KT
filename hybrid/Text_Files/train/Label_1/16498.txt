#include <bits/stdc++.h>
int n, m, f[53][53], dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};
char a[53][53];
bool vis[53][53];
inline int clr(char c) { return c - 'A' + 1; }
void dfs(int x, int y, int pre_x, int pre_y) {
  if (vis[x][y]) {
    puts("Yes");
    exit(0);
  }
  f[x][y] = clr(a[x][y]);
  if (f[x][y] == f[pre_x][pre_y] && (x != pre_x || y != pre_y)) {
    puts("Yes");
    exit(0);
  }
  vis[x][y] = 1;
  for (int i = 0; i < 4; ++i) {
    int nx = x + dx[i], ny = y + dy[i];
    if (nx < 1 || nx > n || ny < 1 || ny > m) continue;
    dfs(nx, ny, x, y);
  }
  f[x][y] = 0;
  vis[x][y] = 0;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) scanf("%s", a[i] + 1);
  a[n + 1][m + 1] = a[1][1];
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= m; ++j)
      if (!vis[i][j]) dfs(i, j, i, j);
  puts("No");
  return 0;
}
