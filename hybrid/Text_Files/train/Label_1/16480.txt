#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == 'R' && s[i + 1] == 'U') {
            cout << 'D';
            i++;
        } else {
            cout << s[i];
        }
    }
    cout << '\n';
    return 0;
}
