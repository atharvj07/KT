from heapq import heappush, heappop, heapify

n, m, k = map(int, input().split())
d = [-1]*n
g=[[]]
for i in range(n):
    g.append([])

for i in range(m):
    x, y, z = map(int, input().split())
    x -= 1; y -= 1
    g[x] += [(z, y)]
    g[y] += [(z, x)]

q = []
d[0] = 0
heappush(q, (0, 0))

while q:
    u, du = heappop(q)
    if d[u] != du: continue
    for w, v in g[u]:
        if d[v] == -1:
            d[v] = du+w
            heappush(q, (du+w, v))
edges = [(i+1, d[i]) for i in range(n) if d[i] != -1]
edges.sort(key=lambda x: x[1])

e = min(k, len(edges))
print(e)
for i in range(e):
    print(edges[i][0], end=' ')
