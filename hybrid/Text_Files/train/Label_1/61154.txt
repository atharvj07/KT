import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, p, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for x in a:
        if p >= x:
            p -= x
            ans += 1
            k -= 1
        if not k:
            break
    for x in a:
        if p >= x:
            p -= x
            ans += 1
    print(ans)
