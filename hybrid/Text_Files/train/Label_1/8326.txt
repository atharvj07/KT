#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, a, b;
    cin >> n >> k >> a >> b;
    if (b > a)
        swap(a, b);
    if (b > 1 && k == 1)
        cout << "NO";
    else if (a < k)
        cout << "NO";
    else {
        for (int i = 0; i < b; i++)
            for (int j = 0; j < min(max(k - (b - i), 0), a - i); j++)
                cout << 'G';
        for (int i = 0; i < b; i++)
            cout << 'B';
    }
    return 0;
}
