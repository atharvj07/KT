# include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int P = 1e9 + 7;
const int N = 1e5 + 10;

int n, m, cnt[N];

ll qpow(ll a, ll b) {
  ll res = 1;
  while (b) {
    if (b & 1) res = res * a % P;
    a = a * a % P;
    b >>= 1;
  }
  return res;
}

ll inv(ll x) { return qpow(x, P - 2); }

ll mul(ll x, ll y) { return x * y % P; }

int main() {
  cin >> n >> m;
  ll num = 1, den = 1;
  for (int i = 1; i <= n; ++i) {
    int x;
    cin >> x;
    cnt[0] += x;
    for (int j = 1; j <= x; ++j) {
      int y;
      cin >> y;
      ++cnt[y];
    }
  }
  for (int i = 1; i <= m; ++i) num = mul(num, cnt[i]);
  for (int i = 1; i <= m; ++i) den = mul(den, cnt[0] - cnt[i]);
  cout << mul(num, inv(den)) << endl;
  return 0;
}
