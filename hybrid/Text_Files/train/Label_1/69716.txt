for _ in range(int(input())):
    _, t, a, b = map(int, input().split())
    l = list(map(int, input().split()))
    d = list(map(int, input().split()))
    pos = []
    for i in range(len(l)):
        if l[i] == 1:
            if d[i] <= a:
                pos.append(-1)
            else:
                pos.append(d[i] - a)
        else:
            if d[i] <= b:
                pos.append(-1)
            else:
                pos.append(d[i] - b)
    pos.sort(reverse=True)
    ans = 0
    t -= d[0]
    for p in pos:
        if t < 0:
            break
        ans += (p != -1)
        t -= p
    print(ans)
