import java.util.*;

public class Main {
    static int[] parent;
    static int[] sz;
    static int[] last;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt();
        parent = new int[n + 1];
        sz = new int[n + 1];
        last = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
            sz[i] = 1;
        }
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt(), v = scanner.nextInt();
            union(u, v);
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (parent[i] == i) {
                ans++;
                last[i] = i;
            }
        }
        System.out.println(ans);
        for (int i = 1; i <= n; i++) {
            if (parent[i] == i) {
                System.out.print(last[i] + " ");
            }
        }
    }

    static int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            if (sz[x] > sz[y]) {
                parent[y] = x;
                sz[x] += sz[y];
                last[x] = y;
            } else {
                parent[x] = y;
                sz[y] += sz[x];
                last[y] = x;
            }
        }
    }
}
