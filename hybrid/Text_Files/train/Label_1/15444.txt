#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n, x, y, gcd, ans = -1, g;
    cin >> n;
    vector<int> arr;
    for(int i = 0; i < n; i++){
        cin >> x >> y;
        g = __gcd(x, y);
        arr.push_back(g);
    }
    for(int i = 0; i < n; i++){
        if(i == 0) gcd = arr[0];
        else{
            gcd = __gcd(gcd, arr[i]);
            if(gcd == 1){
                ans = -1;
                break;
            }
        }
    }
    if(gcd > 1){
        ans = gcd;
        for(int i = 0; i < n; i++){
            if(arr[i] % ans != 0){
                ans = -1;
                break;
            }
        }
    }
    cout << ans << "\n";
    return 0;
}
