import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int,input().split()))
pre = [0] * n
pre[0] = a[0]
for i in range(1,n):
    pre[i] = pre[i-1] ^ a[i]
cnt = 0
for l in range(n):
    for r in range(l+1,n):
        if (r-l+1) % 2 == 0:
            if (l-1 >= 0):
                L = pre[r] ^ pre[l-1]
            else:
                L = pre[r]
            R = pre[n-1] ^ pre[r]
            if L == R:
                cnt+=1
print(cnt)
