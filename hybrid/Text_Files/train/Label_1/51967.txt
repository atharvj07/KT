import java.util.*;
import java.math.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int l = 0, r = n - 1, best = 0;
            while (l <= r) {
                int mid = (l + r) / 2;
                int[] b = Arrays.copyOfRange(a, l, mid + 1);
                Arrays.sort(b);
                int pos = Arrays.binarySearch(b, a[i]);
                best = Math.max(best, Math.max(mid - l - pos, pos - r + mid));
                if (a[mid] < a[i]) {
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            ans[i] = best;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
