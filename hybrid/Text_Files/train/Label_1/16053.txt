#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 998244353;
int main() {
    int n;
    cin >> n;
    vector<ll> v(n);
    for (auto &i : v) cin >> i;
    sort(v.begin(), v.end());
    vector<ll> ans;
    for (int i = 1; i < n; i++) {
        if (v[i] == v[i - 1]) {
            ans.push_back(v[i]);
            i++;
        }
    }
    cout << ans.size() << endl;
    for (auto &i : ans) cout << i << " ";
}
