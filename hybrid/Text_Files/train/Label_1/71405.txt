n, k = map(int, input().split())
a = list(map(int, input().split()))

def length_of_number(num):
    return len(str(num))

count = 0
for i in range(n):
    for j in range(n):
        if i != j:
            concat_number = int(str(a[i]) + str(a[j]))
            if concat_number % k == 0:
                count += 1
                
print(count)
