#include <bits/stdc++.h>
using namespace std;
const int MAXN = 51;
int main() {
  int n, x;
  cin >> n >> x;
  vector<pair<int, int>> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i].first >> a[i].second;
    a[i].second = min(a[i].second, a[i].first + x - 1);
  }
  sort(a.begin(), a.end());
  int cnt = 0, r = 0;
  for (int i = 0; i < n; i++) {
    cnt += max(0, a[i].first - r);
    r = max(r, a[i].second);
  }
  cout << cnt << endl;
}
