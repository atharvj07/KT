import java.util.Scanner;

public class GeometricProgressionCounter {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int r = scanner.nextInt();

        long count = 0;
        for (int firstTerm = l; firstTerm <= r; firstTerm++) {
            for (int commonRatio = 1; commonRatio <= r - firstTerm; commonRatio++) {
                if (isProgressionValid(firstTerm, commonRatio, n, l, r)) {
                    count++;
                }
            }
        }

        System.out.println(count);
    }

    private static boolean isProgressionValid(int firstTerm, int commonRatio, int n, int l, int r) {
        int term = firstTerm;
        for (int i = 1; i <= n; i++) {
            if (term < l || term > r) {
                return false;
            }
            term *= commonRatio;
        }
        return true;
    }
}
