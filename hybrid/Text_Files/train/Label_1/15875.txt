#include <iostream>
#include <algorithm>
#include <vector>
#include <set>
using namespace std;
struct Exam{ int s,d,c; };
int main(){
    int n,m; cin>>n>>m;
    vector<Exam> v(m);
    for(int i=0;i<m;++i){
        cin>>v[i].s>>v[i].d>>v[i].c;
    }
    vector<int> ans(n,0);
    set<pair<int,int> > s;
    for(int i=0;i<m;++i){
        if(v[i].d-v[i].s<v[i].c){
            cout<<-1<<endl;
            return 0;
        }
        for(int j=v[i].s+v[i].c-1;j<v[i].d;++j){
            if(s.count({j,i})==0){
                s.insert({j,i});
                ans[j]=i+1;
                break;
            }
        }
    }
    for(int i=0;i<n;++i){
        if(ans[i]==0){
            if(s.count({i,-1})==0){
                s.insert({i,-1});
                ans[i]=m+1;
            }
        }
    }
    for(int x:ans) cout<<x<<" "; cout<<endl;
    return 0;
}
