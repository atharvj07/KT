n,x,k = map(int,input().split())
a = list(map(int,input().split()))
ans = 0
for l in a:
    r = l+(x-1)*k
    # print(l,r)
    i = 0
    j = n-1
    ind = -1
    while i<=j:
        mid = (i+j)//2
        if a[mid]<=r:
            ind = mid
            i = mid+1
        else: j = mid-1
    if ind==-1: ans+=0
    else: ans+=(n-ind)
print(ans)
