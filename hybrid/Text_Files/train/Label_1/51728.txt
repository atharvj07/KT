import java.util.*;

public class Main {

  static final int MOD = (int) 1e9;
  static int length;
  static String[] str;
  static Integer[][] dp;

  static int maxPalin(int ind, int jnd) {
    if (ind > jnd) return 0;
    if (dp[ind][jnd] != null) return dp[ind][jnd];

    if (ind == jnd) return dp[ind][jnd] = 1;
    if (jnd - ind == 1) return dp[ind][jnd] = str[ind].equals(str[jnd]) ? 2 : 1;

    int ans = 0;
    if (str[ind].equals(str[jnd])) {
      ans = maxPalin(ind + 1, jnd - 1) + 2;
      if (ans >= MOD) ans -= MOD;
    }

    ans = Math.max(ans, maxPalin(ind + 1, jnd));
    ans = Math.max(ans, maxPalin(ind, jnd - 1));
    return dp[ind][jnd] = ans;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    length = in.nextInt();
    str = new String[length];
    for (int i = 0; i < length; i++) str[i] = in.next();
    dp = new Integer[length][length];
    System.out.print(maxPalin(0, length - 1));
  }
}
