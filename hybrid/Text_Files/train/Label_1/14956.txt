#include <bits/stdc++.h>
using namespace std;

vector<int> g[200010];
int a[200010];
int vis[200010];

void dfs(int x,int p){
    vis[x]=1;
    for(auto u:g[x]){
        if(u!=p){
            dfs(u,x);
            if(a[u]>0)a[x]+=a[u];
        }
    }
}

int main()
{
    int n,x,y;
    cin>>n;
    for(int i=2;i<=n;i++){
        cin>>x;
        g[x].push_back(i);
        g[i].push_back(x);
    }
    for(int i=1;i<=n;i++)
        cin>>a[i];
    dfs(1,1);
    cout<<a[1]<<endl;
    return 0;
}
