#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define PI acos(-1)
#define endl '\n'

int arr[200100];
int c[200100];
int32_t main()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    int mx = 0;
    for (int i = 0; i < n; i++)
    {
        arr[s[i]]++;
        mx = max(arr[s[i]], mx);
    }
    if (mx <= (n + 1) / 2)
    {
        cout << 1 << endl;
        for (int i = 0; i < n; i++)
        {
            cout << "1 ";
        }
        cout << endl;
    }
    else
    {
        int sm = 0;
        for (int i = 0; i < 26; i++)
        {
            sm += max(0, arr[i] - (n + 1) / 2);
        }
        cout << 2 << endl;
        int p = 1;
        int q = 2;
        int r = 3;
        for (int i = 0; i < n; i++)
        {
            int l = arr[s[i]];
            arr[s[i]]--;
            if (l > (n + 1) / 2 && sm > 0)
            {
                c[i] = r;
                sm--;
                r += 2;
            }
            else
            {
                if (i % 2 == 0)
                {
                    c[i] = p;
                    p += 2;
                }
                else
                {
                    c[i] = q;
                    q += 2;
                }
            }
        }
        for (int i = 0; i < n; i++)
        {
            cout << c[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
