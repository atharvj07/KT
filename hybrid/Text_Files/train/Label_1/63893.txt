import sys

n = int(input())
a = list(map(int,input().split()))
d = {}
for i in range(1,n):
    if a[ i - 1 ] not in d:
        d[ a[ i - 1 ] ] = [i]
    else:
        d[ a[ i - 1 ] ].append(i)
l = []
l.append(0)
for i in d:
    l.append(len(d[i]))
l.sort(reverse = True)
h = 0
ans = 0
col = 0
for i in l:
    if h + i > col:
        col += 1
        ans += 1
    if col >= len(l) - 1:
        break
    h += i
for i in range(n):
    print(ans,end = ' ')
