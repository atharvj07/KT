t = int(input())
for _ in range(t):
    n = int(input())
    l = [int(x) for x in input().split()]
    d, c = 0, 1
    for i in range(1, n):
        if l[i] > l[i - 1]:
            d = 0
            c += 1
        else:
            d += 1
        if d == 2:
            c += 1
            d = 0
    print(c)
