import sys
import math

n = int(input())
points = []
for i in range(n):
    x, y = map(int, input().split())
    points.append((x, y))

def cross(p1, p2, p3):
    return (p2[0] - p1[0]) * (p3[1] - p1[1]) - (p3[0] - p1[0]) * (p2[1] - p1[1])

def is_convex(points):
    n = len(points)
    for i in range(n):
        if cross(points[i], points[(i + 1) % n], points[(i + 2) % n]) < 0:
            return False
    return True

def calc_perimeter(points):
    n = len(points)
    perimeter = 0
    for i in range(n):
        perimeter += abs(points[i][0] - points[(i + 1) % n][0]) + abs(points[i][1] - points[(i + 1) % n][1])
    return perimeter

def max_perimeter(points, k):
    n = len(points)
    best_perimeter = 0
    for i in range(n):
        new_points = points[i : i + k]
        if is_convex(new_points):
            best_perimeter = max(best_perimeter, calc_perimeter(new_points))
    return best_perimeter

for k in range(3, n + 1):
    print(max_perimeter(points, k))
