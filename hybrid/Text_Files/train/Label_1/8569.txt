#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
  ll n, t1, w1, t2, w2, b;
  cin >> n >> t1 >> w1;
  ll ans = 1;
  for (ll i = 2; i <= n; i++) {
    cin >> t2 >> w2;
    if (t2 > w2) b = min(t2 - w2, t1);
    else b = 0;
    ans += (t1 - b >= t2);
    t1 -= b;
  }
  cout << ans << endl;
}
