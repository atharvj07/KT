#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  vector<int> dic = {6, 2, 5, 5, 4, 5, 6, 3, 7, 6};
  while (t--) {
    int n;
    cin >> n;
    for (int i = 9; i >= 0; i--) {
      if (n >= dic[i]) {
        n -= dic[i];
        cout << i;
      }
    }
    cout << '\n';
  }
  return 0;
}
