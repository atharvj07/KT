#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, x, y, z;
  cin >> t;
  while (t--) {
    cin >> x >> y >> z;
    cout << min({(3 * z) % 2 + 2 * (x + y + z) % 2, (3 * z) % 2 + 2 * x % 2 + ((x + y + z) % 2 + 1) % 2, (3 * z) % 2 + 2 * (x + y) % 2 + ((x + y + z) % 2 + 2) % 2, (3 * z) % 2 + 2 * y % 2 + ((x + y + z) % 2 + 2) % 2}) << endl;
  }
  return 0;
}
