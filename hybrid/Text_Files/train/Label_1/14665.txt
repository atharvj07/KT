#include <iostream>
#include <vector>
using namespace std;
int main()
{
    int n, c, profit = 0, borrow = -1, buy = -1;
    cin >> n >> c;
    vector<int> price(n);
    for (int i = 0; i < n; i++)
        cin >> price[i];
    for (int i = 0; i < n; i++)
    {
        if (borrow == -1 && price[i] >= c)
            borrow = i;
        else if (borrow != -1 && buy == -1 && price[i] * (borrow - 1) <= c)
            buy = i;
    }
    if (borrow != -1 && buy != -1)
        profit = price[borrow] - buy + (borrow - 1) * (buy - borrow - 1);
    cout << profit << endl;
    return 0;
}
