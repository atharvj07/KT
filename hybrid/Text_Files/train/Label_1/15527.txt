#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, a, b, x[N], v[N][2], ans;
long long ex[N];

struct event {
    int type, id, x;
    long long time;
    event() {}
    event(int t, int i, long long x, long long tim) : type(t), id(i), x(x), time(tim) {}
    bool operator<(const event &other) const { return time < other.time; }
};

vector<event> events;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> a >> b;
    for (int i = 0; i < n; i++) {
        cin >> x[i] >> v[i][0] >> v[i][1];
        events.push_back(event(0, i, x[i], 1ll * (a * x[i] + b - x[i]) * v[i][1]));
        events.push_back(event(1, i, x[i], 1ll * (a * x[i] + b - x[i]) * v[i][0]));
    }
    sort(events.begin(), events.end());
    for (const auto &e : events) {
        if (e.type == 0) {
            ex[e.id] += e.x;
        } else {
            ans += ex[e.id];
        }
    }
    cout << ans;
    return 0;
}
