#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, k, x;
    cin >> n >> k >> x;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    
    long long ans = 0;
    for (int i = 0; i < n; i += k) {
        int j = i;
        while (j < n && j - i + 1 <= x) {
            if (j < n) {
                ans += a[j];
                j += k;
            }
        }
    }
    
    if (ans == 0) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
}
