from collections import defaultdict

querys = defaultdict(list)
mp = defaultdict(int)

for i in range(int(input())):
    op, time, x = map(int, input().strip().split())
    if op == 3:
        print(mp[x])
    else:
        querys[time].append((op, x))

time = 0
for t in sorted(querys.keys()):
    time = t
    for op, x in querys[t]:
        if op == 1:
            mp[x] += 1
        elif op == 2:
            mp[x] -= 1
