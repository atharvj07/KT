import sys

input = sys.stdin.readline

n = int(input())
a = []
b = []
for i in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)

a.sort()
b.sort()
ans = 0
res = []
i = 0
j = 0
while i < n and j < n:
    if a[i] < b[j]:
        res.append(a[i])
        i += 1
    else:
        res.append(b[j])
        j += 1
    ans += 1

print(ans)
for x in res:
    print(a.index(x) + 1, end=" ")
