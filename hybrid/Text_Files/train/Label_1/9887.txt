#include <bits/stdc++.h>
using namespace std;
typedef long long ll;const ll mod=1e9+7;
int main(){ios_base::sync_with_stdio(0);cin.tie(0);
    int n,m,p,ans=-1;cin>>n>>m>>p;
    vector<ll>a(n),b(m),h(n+m);
    for(int i=0;i<n;++i)cin>>a[i];
    for(int i=0;i<m;++i)cin>>b[i];
    for(int i=0;i<n;++i){
        for(int j=0;j<m;++j){
            h[i+j]=(h[i+j]+(ll)a[i]*b[j]%p)%p;
        }
    }
    for(int i=0;i<n+m;++i){
        if(h[i]%p){ans=i;break;}
    }
    cout<<ans<<endl;
}
