#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    int x, y;
    cin >> x >> y;

    if (x == 1 && y == 1)
      cout << "0 1\n";
    else if (x == 2 && y == 1)
      cout << "1 1\n";
    else if (x == 1 && y > 1)
      cout << "0 " << y << "\n";
    else
      cout << x / 2 << " " << y / 2 << "\n";
  }

  return 0;
}
