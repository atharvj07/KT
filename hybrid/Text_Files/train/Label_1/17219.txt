#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

const int MAXN = 10;
string pin[MAXN];

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> pin[i]; }
    sort(pin, pin + n);
    int k = 0;
    for (int i = 1; i < n; i++) {
      for (int j = 0; j < 4; j++) {
        if (pin[i][j] != pin[i - 1][j]) {
          k++;
          break;
        }
      }
    }
    cout << k << endl;
    for (int i = 0; i < n; i++) { cout << pin[i] << endl; }
  }
  return 0;
}
