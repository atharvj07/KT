from collections import deque

n = int(input())
handedness = input()

left = []
right = []

for i, h in enumerate(handedness):
    if h == 'L':
        left.append(i + 1)
    else:
        right.append(i + 1)

left = deque(left)
right = deque(right)

for i in range(n // 2):
    if i % 2 == 0:
        print(left.popleft(), right.popleft())
    else:
        print(right.popleft(), left.popleft())
