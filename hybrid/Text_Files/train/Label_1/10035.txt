#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;

  int maxVolume = 0;
  std::string s;
  std::cin >> s;

  for (int i = 0; i < n; i++) {
    int volume = 0;
    while(i < n && s[i] != ' ') {
      if(s[i] >= 'A' && s[i] <= 'Z') volume++;
      i++;
    }
    if(volume > maxVolume) maxVolume = volume;
  }

  std::cout << maxVolume << std::endl;

  return 0;
}
