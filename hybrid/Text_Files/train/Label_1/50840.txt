import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            sum += a[i];
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (sum - a[i]) * a[i];
        }
        ans %= 998244353;
        long inv = fastpow(sum, 998244351);
        ans = ans * inv % 998244353;
        System.out.println(ans);
    }

    static long fastpow(long base, long pow) {
        if (pow == 0) {
            return 1;
        }
        long half = fastpow(base, pow / 2);
        if (pow % 2 == 0) {
            return half * half % 998244353;
        } else {
            return half * half % 998244353 * base % 998244353;
        }
    }
}
