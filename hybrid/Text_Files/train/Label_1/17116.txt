#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m, pos;
  cin >> n >> m >> pos;
  string s;
  cin >> s;
  pair<int, int> p[n + 1];
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == '(') {
      p[i] = {-1, 0};
    } else {
      p[i] = {0, -1};
    }
  }
  stack<int> st;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == '(') {
      st.push(i);
    } else {
      int j = st.top();
      st.pop();
      p[i].first = j;
      p[j].second = i;
    }
  }
  for (int i = 1; i <= m; i++) {
    char c;
    cin >> c;
    if (c == 'L') {
      pos--;
    } else if (c == 'R') {
      pos++;
    } else {
      if (s[pos - 1] == '(') {
        int j = p[pos].second;
        s.erase(pos - 1, j - pos + 1);
        p[j] = {0, 0};
        p[pos] = {0, 0};
        if (pos > 1) {
          pos = p[pos - 1].first;
          if (pos == 0) {
            pos++;
          }
        }
      } else {
        int j = p[pos].first;
        s.erase(j - 1, pos - j + 1);
        p[j] = {0, 0};
        p[pos] = {0, 0};
        pos = j;
      }
    }
  }
  cout << s << endl;
  return 0;
}
