#include <bits/stdc++.h>
using namespace std;

int main() {
  int k1, k2, k3;
  cin >> k1 >> k2 >> k3;
  vector<int> a1(k1), a2(k2), a3(k3);
  for (int &x : a1) cin >> x;
  for (int &x : a2) cin >> x;
  for (int &x : a3) cin >> x;
  int n = k1 + k2 + k3;
  vector<bool> f(n + 1, false);
  for (int x : a1) f[x] = true;
  for (int x : a2) f[x] = true;
  for (int x : a3) f[x] = true;
  int first = 0, last = n + 1;
  for (int i = 1; i <= n; i++) {
    if (!f[i]) continue;
    if (!first) {
      first = i;
      continue;
    }
    if (last == n + 1) last = i;
  }
  int ans = 0;
  for (int x : a2) {
    if (x < first) ans++;
    if (x > last) ans++;
  }
  cout << ans << endl;
}
