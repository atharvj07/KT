n, d = map(int, input().split())
a = [input() for i in range(d)]
ans = 0
cnt = 0
for i in range(d):
    f = False
    for j in range(n):
        if a[i][j] == '0':
            f = True
            break
    if not f:
        cnt += 1
    else:
        ans = max(ans, cnt)
        cnt = 0
ans = max(ans, cnt)
print(ans)
