#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b, c, delta1, delta2, delta3;
        cin >> a >> b >> c;
        delta1 = max({b - a + 1, c - a + 1, 0LL});
        delta2 = max({a - b + 1, c - b + 1, 0LL});
        delta3 = max({a - c + 1, b - c + 1, 0LL});
        cout << delta1 << " " << delta2 << " " << delta3 << endl;
    }
    return 0;
}
