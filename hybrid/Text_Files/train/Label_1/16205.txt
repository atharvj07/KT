#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long L, v, l, r;
        cin >> L >> v >> l >> r;
        if (l <= 1)
            cout << (L - r) / v + 1 << endl;
        else
            cout << (L - r) / v << endl;
    }
    return 0;
}
