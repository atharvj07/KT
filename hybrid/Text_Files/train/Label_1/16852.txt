#include<bits/stdc++.h>
using namespace std;
int n,m;
int main(){
    cin>>n>>m;
    char a[n][m];
    for(int i=0;i<n;i++)for(int j=0;j<m;j++)cin>>a[i][j];
    int ans=1;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            int b=0;
            for(int x=i-1;x<=i+1;x++)for(int y=j-1;y<=j+1;y++)if(x>=0&&x<n&&y>=0&&y<m&&a[x][y]=='*')b++;
            if(a[i][j]=='.'){
                if(b)ans=0;
            }
            else{
                if(b!=a[i][j]-'0')ans=0;
            }
        }
    }
    if(ans)cout<<"YES"<<endl;else cout<<"NO"<<endl;
    return 0;
}
