#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        unordered_set<int> a;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            a.insert(x);
        }
        vector<pair<int, int>> ans;
        for (int x : a) {
            for (int y : a) {
                if (x == y) continue;
                int z = x % y;
                if (a.count(z) == 0 && ans.size() < n / 2) ans.emplace_back(x, y);
            }
        }
        for (auto p : ans) cout << p.first << " " << p.second << '\n';
    }
}
