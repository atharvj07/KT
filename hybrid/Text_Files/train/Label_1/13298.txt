#include <iostream>
using namespace std;
int main()
{
    int t, n;
    cin >> t;
    while (t--)
    {
        cin >> n;
        int half = n / 2;
        if (n & 1)
            cout << half - 1 << " " << half + 1 << "\n";
        else
            cout << half << " " << half << "\n";
    }
    return 0;
}
