#include<bits/stdc++.h>
#define vi vector <int>
#define endl "\n"
using namespace std;

const int mod = 2;
int dp[2005][2][2];
string s;
int n;
int solve(int at, int moves, int turn){
    if(at == n) {
        if(moves == 4) return 1;
        else return 0;
    }
    int &ret = dp[at][moves][turn];
    if(~ret) return ret;
    ret = solve(at + 1, moves + 1, (s[at] - 'a') + turn * mod);
    if(turn) ret = max(ret, solve(at + 1, 0, (s[at] - 'a') + turn * mod));
    return ret;
}
int main() {
    int n;
    cin >> n;
    cin >> s;
    memset(dp, -1, sizeof dp);
    cout << solve(0, 0, 0) << endl;
    return 0;
}
