import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] cnt = new int[n + 1];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            cnt[a[i]]++;
            if (cnt[a[i]] > k) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
