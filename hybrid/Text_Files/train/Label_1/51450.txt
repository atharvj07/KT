import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] weights = br.readLine().split("");
        int m = Integer.parseInt(br.readLine());
        boolean[] used = new boolean[11];
        StringBuilder sb = new StringBuilder();
        if (dfs(weights, m, used, sb)) {
            System.out.println("YES");
            System.out.println(sb);
        } else {
            System.out.println("NO");
        }
    }

    private static boolean dfs(String[] weights, int m, boolean[] used, StringBuilder sb) {
        if (m == 0) {
            return true;
        }
        for (int i = 1; i <= 10; i++) {
            if (weights[i - 1].equals("1") && !used[i] && (sb.length() == 0 || sb.charAt(sb.length() - 1) != i)) {
                used[i] = true;
                sb.append(i + " ");
                if (dfs(weights, m - i, used, sb)) {
                    return true;
                }
                sb.deleteCharAt(sb.length() - 1);
                used[i] = false;
            }
        }
        return false;
    }
}
