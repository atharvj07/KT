import sys

def solve():
    n, k, m = map(int, sys.stdin.readline().split())
    b = list(map(int, sys.stdin.readline().split()))
    if m % 2 == k % 2:
        l, r, i = 0, n + 1, 0
        while i < m:
            if l < b[i] < r:
                l = b[i]
            elif b[i] <= l:
                l = (l + b[i]) // 2
                r = l + 1
            else:
                r = (r + b[i]) // 2
                l = r - 1
            i += 1
        print("YES")
    else:
        print("NO")

t = int(sys.stdin.readline())
for _ in range(t):
    solve()
