def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, m, a):
    g = 0
    for x in a:
        g = gcd(g, x - 1)
    answer = 0
    x = 1
    while x * x <= g:
        if g % x == 0:
            if x <= m - 1:
                answer += 1
            if x != g // x and g // x <= m - 1:
                answer += 1
        x += 1
    return answer

n, m = map(int, input().split())
a = list(map(int, input().split()))
print(solve(n, m, a))
