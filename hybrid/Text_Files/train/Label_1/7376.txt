#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;
const int maxn = 3005;
ll dp[maxn][5];
int main() {
    dp[0][0] = 1;
    for (int i = 1; i < maxn; i++) {
        dp[i][0] = dp[i - 1][0] + dp[i - 1][1] + dp[i - 1][2] + dp[i - 1][3];
        dp[i][1] = dp[i - 1][0];
        dp[i][2] = dp[i - 1][1];
        dp[i][3] = dp[i - 1][2];
        dp[i][0] %= mod;
        dp[i][1] %= mod;
        dp[i][2] %= mod;
        dp[i][3] %= mod;
    }
    int m;
    cin >> m;
    while (m--) {
        int x;
        cin >> x;
        cout << dp[x][0] << "\n";
    }
}
