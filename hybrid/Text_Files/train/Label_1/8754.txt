#include <iostream>
#include <vector>
using namespace std;

const int maxv = 1e5 + 5;
int c, e[maxv], s[maxv];
vector<int> adj[maxv];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> e[i];
    if (!e[i]) c++;
  }
  while (m--) {
    int a, b;
    cin >> a >> b;
    adj[b].push_back(a);
    s[a]++;
  }
  int ans = 0;
  bool vis = false;
  while (1) {
    if (ans > 1e9) {
      cout << -1;
      return 0;
    }
    if (c == n) {
      cout << ans;
      return 0;
    }
    bool changed = false;
    vector<int> t;
    for (int i = 0; i < n; i++) {
      if (s[i] == 0 && e[i]) {
        s[i] = -1;
        changed = true;
        t.push_back(i);
        c++;
      }
    }
    for (int x : t) {
      for (int u : adj[x])
        s[u]--;
    }
    if (!changed) {
      ans = -1;
      cout << ans;
      return 0;
    }
    ans++;
  }
  return 0;
}
