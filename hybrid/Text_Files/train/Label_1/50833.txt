import java.util.*;

public class Main {
    static final int MAX_N = 1 << 18;
    static int N, M;
    static boolean[] vis = new boolean[MAX_N];
    static int[] pre = new int[MAX_N], ans = new int[MAX_N];
    static List<Set<Integer>> edge = new ArrayList<>();

    public static void dfs(int x) {
        vis[x] = true;
        pre[x] = pre[pre[x]] + 1;
        ans[pre[x]] = x;
        for (int ne : edge.get(x)) {
            if (vis[ne]) {
                if (pre[x] > pre[ne])
                    pre[ne] = pre[x];
            } else
                dfs(ne);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        N = sc.nextInt();
        M = sc.nextInt();
        for (int i = 0; i <= N; i++) edge.add(new HashSet<>());
        for (int i = 1; i <= M; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            edge.get(x).add(y);
            edge.get(y).add(x);
        }

        for (int i = 1; i <= N; i++)
            if (!vis[i])
                dfs(i);

        System.out.println(Arrays.toString(Arrays.copyOfRange(ans, 1, N + 1)).replaceAll("[ \\[\\]]", ""));
    }
}
