#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, x, k = 0;
  cin >> n >> x;
  for (int i = 2; i <= n; i++) {
    cin >> x;
    k |= (x != 0);
  }
  cout << k + (k == n) << '\n';
  if (k == n) cout << '& 0\n';
  if (k) cout << (k == 1 ? "& 1\n" : "| ");
  if (k > 1) cout << (k == 2 ? "^ 2\n" : "| ");
  if (k > 2) cout << "^ 3\n";
  return 0;
}
