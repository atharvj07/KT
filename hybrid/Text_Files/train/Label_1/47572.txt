import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        System.out.println(solve(n, m, a, dp));
    }

    private static int solve(int n, int m, int[] a, int[] dp) {
        if (dp[n] != -1) {
            return dp[n];
        }
        int x = 0;
        for (int i = 0; i < m; i++) {
            if (n - a[i] >= 0) {
                x = (x + solve(n - a[i], m, a, dp)) % 1000000007;
            }
        }
        x = (x + 1) % 1000000007;
        dp[n] = x;
        return dp[n];
    }
}
