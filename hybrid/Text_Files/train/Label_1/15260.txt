#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<char>> g(n, vector<char>(m));
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++) cin >> g[i][j];
    bool valid = true;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (g[i][j] == '.') {
          int cntR = 0, cntW = 0;
          if (i - 1 >= 0 && g[i - 1][j] == 'R') cntR++;
          if (i + 1 < n && g[i + 1][j] == 'R') cntR++;
          if (j - 1 >= 0 && g[i][j - 1] == 'R') cntR++;
          if (j + 1 < m && g[i][j + 1] == 'R') cntR++;
          if (i - 1 >= 0 && g[i - 1][j] == 'W') cntW++;
          if (i + 1 < n && g[i + 1][j] == 'W') cntW++;
          if (j - 1 >= 0 && g[i][j - 1] == 'W') cntW++;
          if (j + 1 < m && g[i][j + 1] == 'W') cntW++;
          if (cntR > 0 && cntW > 0) {
            valid = false;
            break;
          }
        }
      }
      if (!valid) break;
    }
    if (valid)
      cout << "YES\n";
    else
      cout << "NO\n";
    if (valid) {
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
          if (g[i][j] == '.') {
            if (g[i][j - 1] == 'R' || g[i][j + 1] == 'R' || g[i - 1][j] == 'R')
              g[i][j] = 'W';
            else
              g[i][j] = 'R';
          }
          cout << g[i][j];
        }
        cout << '\n';
      }
    }
  }
  return 0;
}
