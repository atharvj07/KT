#include<bits/stdc++.h>
using namespace std;
bool vis[101];
vector<int>a[101];
int main() {
  int n, m; cin >> n >> m;
  for (int i = 0; i < n; i++) {
    int x; cin >> x;
    while (x--) {
      int y; cin >> y;
      a[i].push_back(y);
    }
  }
  bool ok = 0;
  for (int i = 0; i < n; i++) {
    ok = 1;
    for (int j : a[i]) {
      if (vis[j])ok = 0;
    }
    if (ok) {
      for (int j : a[i])vis[j] = 1;
    }
  }
  ok = 0;
  for (int i = 1; i <= m; i++) {
    if (!vis[i])ok = 1;
  }
  if (!ok)cout << "YES" << endl;
  else cout << "NO" << endl;
  return 0;
}
