home = input()
away = input()
n = int(input())
warnings = {}
for _ in range(n):
    t, s, m, c = input().split()
    t = int(t)
    name, number = (home, m) if s == 'h' else (away, m)
    if name in warnings and warnings[name] == 'y':
        print(name, number, t)
    elif c == 'y':
        warnings[name] = 'y'
