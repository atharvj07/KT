n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Initialize prefix sums for both rows
prefix_a = [0] * (n + 1)
prefix_b = [0] * (n + 1)
for i in range(n):
    prefix_a[i + 1] = prefix_a[i] + a[i]
    prefix_b[i + 1] = prefix_b[i] + b[i]

# Calculate total weight for two different starting directions
total_a = total_b = 0
for i in range(n):
    total_a += (2 * i) * a[i] + (2 * i + 1) * b[i]
    total_b += (2 * i) * b[i] + (2 * i + 1) * a[i]

# Initialize the maximum weight of mushrooms that can be collected
max_weight = max(total_a, total_b)

# Calculate the maximum weight by considering paths that turn around at each possible point
for i in range(n):
    weight1 = prefix_a[i] * 2 + (prefix_b[n] - prefix_b[i]) * 2 - b[i] + (prefix_a[n] - prefix_a[i]) - a[n-1]
    weight2 = prefix_b[i] * 2 + (prefix_a[n] - prefix_a[i]) * 2 - a[i] + (prefix_b[n] - prefix_b[i]) - b[n-1]
    max_weight = max(max_weight, weight1, weight2)

print(max_weight)
