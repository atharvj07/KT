import heapq

n=int(input());
req=[list(map(int,input().split())) for i in range(n)]
k=int(input());
tables=[int(x) for x in input().split()]
tables.sort()
req.sort(key=lambda x:x[1],reverse=True)
j,ans,money=0,0,0
for x,y in req:
    while j<k and x>tables[j]:
        j+=1
    if j==k:
        break
    ans+=1
    money+=y
    tables[j]-=x
    j+=1
print(ans,money)
for i in range(ans):
    print(i+1,j-i)
