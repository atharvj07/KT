#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long p, f, s, w, ans = 0, cnt_s, cnt_w;
        cin >> p >> f >> cnt_s >> cnt_w >> s >> w;
        long long x = min((p / s), cnt_s);
        long long y = min((f / s), (cnt_s - x));
        long long z = min(((p - (x * s)) / w), cnt_w);
        long long a = min((f / w), (cnt_w - z));
        ans = x + y + z + a;
        cout << ans << endl;
    }
    return 0;
}
