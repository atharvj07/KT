import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Main {

    private static class Message {
        int count;
        char letter;

        Message(int count, char letter) {
            this.count = count;
            this.letter = letter;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Message> messageT = new ArrayList<>();
        List<Message> messageS = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            String[] part = sc.next().split("-");
            int count = Integer.parseInt(part[0]);
            char letter = part[1].charAt(0);
            messageT.add(new Message(count, letter));
        }

        for (int i = 0; i < m; i++) {
            String[] part = sc.next().split("-");
            int count = Integer.parseInt(part[0]);
            char letter = part[1].charAt(0);
            messageS.add(new Message(count, letter));
        }

        int indexT = 0;
        int indexS = 0;
        int count = 0;
        while (indexT < n && indexS < m) {
            Message partT = messageT.get(indexT);
            Message partS = messageS.get(indexS);
            if (partT.letter == partS.letter) {
                if (partT.count < partS.count) {
                    indexT += 1;
                } else if (partT.count > partS.count) {
                    indexS += 1;
                } else {
                    count += 1;
                    indexT += 1;
                    indexS += 1;
                }
            } else {
                if (partT.letter < partS.letter) {
                    indexT += 1;
                } else {
                    indexS += 1;
                }
            }
        }
        System.out.println(count);
    }
}
