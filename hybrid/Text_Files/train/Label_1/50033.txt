import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        String t = sc.nextLine();
        int n = s.length();
        char[] arr = new char[n];
        boolean[] used = new boolean[n];
        for (int i = 0; i < n; i++) {
            int cnt = 0;
            int idx = -1;
            for (int j = 0; j < n; j++) {
                if (!used[j] && s.charAt(j) >= t.charAt(j)) {
                    int c = s.charAt(j) - 'a' + t.charAt(j) - 'a';
                    if (c > cnt) {
                        cnt = c;
                        idx = j;
                    }
                }
            }
            if (idx != -1) {
                arr[i] = s.charAt(idx);
                used[idx] = true;
            } else {
                for (int j = 0; j < n; j++) {
                    if (!used[j]) {
                        arr[i] = s.charAt(j);
                        used[j] = true;
                        break;
                    }
                }
            }
        }
        System.out.println(new String(arr));
    }
}
