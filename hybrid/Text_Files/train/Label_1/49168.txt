import java.util.*;
import java.lang.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = input.nextInt();
    }
    int m = input.nextInt();
    int[] b = new int[m];
    for (int i = 0; i < m; i++) {
      b[i] = input.nextInt();
    }
    int max_ratio = 0;
    int count = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        int ratio = b[j] / a[i];
        if (ratio > max_ratio && ratio * a[i] == b[j]) {
          max_ratio = ratio;
          count = 1;
        } else if (ratio == max_ratio && ratio * a[i] == b[j]) {
          count++;
        }
      }
    }
    System.out.println(count);
  }
}
