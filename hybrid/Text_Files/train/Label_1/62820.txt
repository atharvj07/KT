from queue import PriorityQueue
n,m=map(int,input().split());v=list(map(int,input().split()))
gr=[[] for _ in range(n+1)]
for _ in range(m):x,y=map(int,input().split());gr[x].append(y);gr[y].append(x)
def dijkstra():
 x=PriorityQueue()
 dp=[-1e9]*(n+1)
 x.put((0,1))
 dp[1]=0
 while not x.empty():
  d,u=x.get()
  if d>dp[u]:continue
  for ne in gr[u]:
   w=v[ne-1]
   if dp[ne]<d+w:
    dp[ne]=d+w
    x.put((-dp[ne],ne))
 return dp[n]
res=dijkstra()
print(res)
