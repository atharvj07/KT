import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Stack<Integer> s = new Stack<>();
        Queue<Integer> q = new LinkedList<>();
        Deque<Integer> d = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            if (a > 0) {
                if (s.size() < q.size() && s.size() < d.size()) {
                    s.push(a);
                    System.out.println("pushStack");
                } else if (q.size() < s.size() && q.size() < d.size()) {
                    q.add(a);
                    System.out.println("pushQueue");
                } else {
                    d.addLast(a);
                    System.out.println("pushBack");
                }
            } else {
                int k = 0;
                List<String> ops = new ArrayList<>();
                if (!s.isEmpty()) {
                    k++;
                    ops.add("popStack");
                }
                if (!q.isEmpty()) {
                    k++;
                    ops.add("popQueue");
                }
                if (!d.isEmpty()) {
                    k++;
                    ops.add("popFront");
                }
                System.out.println(k + " " + String.join(" ", ops));
                s.clear();
                q.clear();
                d.clear();
            }
        }
    }
}
