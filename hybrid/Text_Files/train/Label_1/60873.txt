def min_time_to_eat_candies(n, s, k, candies, colors):
    from itertools import permutations
    # Generate all possible colors sequence to maintain the condition of not eating the same colored candies consecutively
    color_permutations = list(permutations(['R', 'G', 'B']))

    min_time = float('inf')
    # Try all permutations of colors to find the minimum time
    for color_seq in color_permutations:
        for start in range(n):
            # Initialize variables for each starting position and color sequence
            time, last_candy_count, candies_eaten = 0, 0, 0
            current_position, current_color_index = s-1, 0
            # While we haven't eaten k candies, try to eat from the next valid box
            while candies_eaten < k and current_color_index < 3:
                # Find the next box that matches the current color in the sequence and has more candies than the last box Tanya ate from
                next_box = None
                for i in range(start, n):
                    if colors[i] == color_seq[current_color_index] and candies[i] > last_candy_count:
                        next_box = i
                        break
                # If such a box is found, eat the candies and update variables accordingly
                if next_box is not None:
                    time += abs(current_position - next_box)  # Time to move to the next box
                    candies_eaten += candies[next_box]
                    last_candy_count = candies[next_box]
                    current_position = next_box
                    current_color_index = (current_color_index + 1) % 3
                else:
                    break
            # If Tanya has eaten at least k candies, update the minimum time
            if candies_eaten >= k:
                min_time = min(min_time, time)
                break  # No need to try other starting positions for this color sequence
    # Return the minimum time if Tanya can eat at least k candies, otherwise return -1
    return min_time if min_time != float('inf') else -1

# Input reading
n, s, k = map(int, input().split())
candies = list(map(int, input().split()))
colors = input().strip()

# Function call and output
print(min_time_to_eat_candies(n, s, k, candies, colors))
