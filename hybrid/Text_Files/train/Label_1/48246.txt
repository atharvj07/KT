import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextInt(), m = input.nextInt(), x = input.nextInt();
        long ans = n * (n + 1) / 2 * m;
        ans += (m - x + 1) * (m - x + 2) / 2 * (n - 1);
        ans %= 1000000007;
        System.out.println(ans);
    }
}
