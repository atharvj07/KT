import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++)
            a[i] = sc.nextInt();
        PriorityQueue<long[]> pq = new PriorityQueue<>((o1, o2) -> Long.compare(o1[0], o2[0]));
        for (int j = 0; j < n; j++) {
            int t = sc.nextInt();
            int i = sc.nextInt();
            long b = sc.nextInt();
            if (t == 1)
                pq.add(new long[]{b - a[i - 1], j + 1});
            else if (t == 2)
                pq.add(new long[]{b, -j - 1});
            else
                pq.add(new long[]{b * a[i - 1], j + 1});
        }
        ArrayList<Long> al = new ArrayList<>();
        long ans = 0;
        for (; k > 0 && !pq.isEmpty() && m > 0; k--, m--) {
            long[] z = pq.poll();
            ans += z[0];
            al.add(z[1]);
        }
        System.out.println(al.size());
        for (long x : al)
            System.out.print(x + " ");
    }
}
