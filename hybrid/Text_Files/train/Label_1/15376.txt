#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n), b(n);
        for (int &i : a) cin >> i;
        for (int &j : b) cin >> j;
        sort(a.begin(), a.end());
        sort(b.begin(), b.end(), greater<int>());
        int sum = 0;
        for (int i = 0; i < min(k, n); i++) { sum += b[i]; }
        for (int i = k; i < n; i++) { sum += a[i]; }
        cout << sum << '\n';
    }
    return 0;
}
