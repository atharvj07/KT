#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> grid(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> grid[i][j]; }
  }
  int max_count = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == 'D') {
        int count = 0;
        int x = i, y = j;
        while (grid[x][y] != 'D' || count == 0) {
          if (grid[x][y] == 'D') count++;
          if (grid[x][y] == 'I') x--;
          if (grid[x][y] == 'M') y++;
          if (grid[x][y] == 'A') y--;
          if (x < 0 || x >= n || y < 0 || y >= m) break;
        }
        max_count = max(max_count, count);
      }
    }
  }
  if (max_count == 0) cout << "Poor Dima!" << endl;
  else if (max_count > 1000000000) cout << "Poor Inna!" << endl;
  else cout << max_count << endl;
  return 0;
}
