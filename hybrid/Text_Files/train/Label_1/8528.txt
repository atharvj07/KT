#include<bits/stdc++.h>
using namespace std;
#define ll long long
struct node{
    ll cost;
    string type;
};
bool cmp(node a,node b){
    if(a.cost==b.cost){
        return a.type=="USB";
    }
    return a.cost<b.cost;
}
int main(){
    ll a,b,c,m,i,j,sum=0,c1=0,c2=0;
    cin>>a>>b>>c>>m;
    vector<node>v(m);
    for(i=0;i<m;i++){
        cin>>v[i].cost>>v[i].type;
    }
    sort(v.begin(),v.end(),cmp);
    for(i=0;i<m;i++){
        if(v[i].type=="USB" && a>0){
            sum+=v[i].cost;
            a--,c1++;
            continue;
        }
        if(v[i].type=="PS/2" && b>0){
            sum+=v[i].cost;
            b--,c1++;
            continue;
        }
        if(c>0){
            sum+=v[i].cost;
            c--,c1++;
            continue;
        }
    }
    for(i=m-1;i>=0;i--){
        if(v[i].type=="USB" && c1<a){
            c2++;
            continue;
        }
        if(v[i].type=="PS/2" && c1<a+b){
            c2++;
            continue;
        }
    }
    cout<<c1<<" "<<sum<<endl;
    return 0;
}
