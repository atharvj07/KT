#include <iostream>
#include <vector>
using namespace std;
bool fourinrow(vector<string> vec) {
    for (int i = 0; i < 4; ++i) {
        if (vec[i][0] != '.' && vec[i][0] == vec[i][1] && vec[i][1] == vec[i][2] && vec[i][2] == vec[i][3]) return true;
        if (vec[0][i] != '.' && vec[0][i] == vec[1][i] && vec[1][i] == vec[2][i] && vec[2][i] == vec[3][i]) return true;
    }
    if (vec[0][0] != '.' && vec[0][0] == vec[1][1] && vec[1][1] == vec[2][2] && vec[2][2] == vec[3][3]) return true;
    if (vec[0][3] != '.' && vec[0][3] == vec[1][2] && vec[1][2] == vec[2][1] && vec[2][1] == vec[3][0]) return true;
    return false;
}
int main() {
    vector<string> vec(4);
    for (int i = 0; i < 4; ++i) getline(cin, vec[i]);
    bool found = false;
    for (int i = 0; i < 4; ++i) {
        for (int j = 0; j < 4; ++j) {
            if (vec[i][j] == '.') {
                vec[i][j] = 'x';
                if (fourinrow(vec)) {
                    cout << "YES";
                    return 0;
                }
                vec[i][j] = '.';
            }
        }
    }
    cout << "NO";
    return 0;
}
