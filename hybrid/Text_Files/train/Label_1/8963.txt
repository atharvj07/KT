#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5 + 5;
int32_t main() {
  int n;
  cin >> n;
  vector<vector<ll>> v(n);
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    v[i].resize(x);
    for (auto &e : v[i]) {
      cin >> e;
    }
  }
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      bool flag = 0;
      for (auto &e : v[i]) {
        for (auto &f : v[j]) {
          if (e < f) {
            flag = 1;
            break;
          }
        }
        if (flag == 1) break;
      }
      if (flag == 1) ans++;
    }
  }
  cout << ans << '\n';
}
