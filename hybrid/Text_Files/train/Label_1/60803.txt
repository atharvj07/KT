import sys

n = int(input())
a = list(map(int, input().split()))

if n == 1:
    sys.stdout.write("1\n0\n")
    sys.exit()

b = a
for i in range(1, n - 1):
    b[i] = (a[i - 1] + a[i] + a[i + 1]) % 2
a = b

count = 0
while a != b:
    count += 1
    b = a
    for i in range(1, n - 1):
        b[i] = (a[i - 1] + a[i] + a[i + 1]) % 2
    a = b

sys.stdout.write(f"{count}\n")
for i in range(n):
    sys.stdout.write(f"{a[i]} ")
