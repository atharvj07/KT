#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, mx = 0;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    vector<vector<int>> dp(n + 1, vector<int>(3, 1));
    for (int i = 2; i <= n; i++) {
        dp[i][1] = max(dp[i][1], dp[i - 1][1]);
        if (a[i] >= a[i - 1]) dp[i][1] = max(dp[i][1], dp[i - 1][2] + 1);
        dp[i][2] = max(dp[i][2], dp[i - 1][2]);
        if (a[i] >= a[i - 1]) dp[i][2] = max(dp[i][2], dp[i - 1][1] + 1);
        mx = max(dp[i][1], mx);
        mx = max(dp[i][2], mx);
    }
    cout << mx << endl;
    return 0;
}
