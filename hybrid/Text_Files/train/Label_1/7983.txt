#include <iostream>
#include <string>
#include <algorithm>
#include <vector>
#include <map>

int main() {
    int n;
    std::cin >> n;
    std::vector<std::string> titles(n);
    for (int i = 0; i < n; i++)
        std::cin >> titles[i];
    std::string ans;
    for (int i = 0; ; i++) {
        std::string s(1, 'a' + i);
        bool ok = true;
        for (auto &t : titles) {
            if (t.find(s) != std::string::npos) {
                ok = false;
                break;
            }
        }
        if (ok) {
            ans = s;
            break;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
