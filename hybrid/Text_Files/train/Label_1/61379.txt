import sys

input = sys.stdin.readline

n, k = map(int, input().split())
a = [int(x) for x in input().split()]

if len(set(a)) > k:
  print("NO")
  exit()

b = {}
for x in a:
  if x not in b:
    b[x] = 1
  else:
    b[x] += 1

cnt = 0
for x in b:
  if b[x] > 1:
    cnt += 1

if cnt > k - 1:
  print("NO")
  exit()

print("YES")
for x in a:
  if not b[x]:
    b[x] = k
  print(b[x], end=' ')
  b[x] -= 1
