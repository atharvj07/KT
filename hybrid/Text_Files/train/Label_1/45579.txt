import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner input = new Scanner(System.in);
		int n = input.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; ++i) {
			a[i] = input.nextInt();
		}
		Arrays.sort(a);
		int ans = 0;
		int j = 0;
		for (int i = 0; i < n; ++i) {
			if (i == 0 || a[i] != a[i - 1]) {
				j = 1;
			} else {
				j *= 2;
			}
			ans = Math.max(ans, j);
		}
		System.out.println(ans);
	}
}
