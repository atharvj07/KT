#include<bits/stdc++.h>
using namespace std;
int a[200001],dp[200001],node=0,ans[200001];
int main()
{
	int n,mx=-1,x=-1,y=-1;
	cin>>n;
	for(int i=1;i<=n;++i)
	{
		cin>>a[i];
		if(i==1||a[i-1]==a[i]-1)
			dp[i]=dp[i-1]+1;
		else
			dp[i]=1;
		if(dp[i]>mx){
			mx=dp[i];
			x=i;
		}
	}
	cout<<mx<<'\n';
	for(int i=x;i>0;--i)
	{
		if(dp[i]==mx)
		{
			ans[node++]=a[i];
			--mx;
		}
	}
	for(int i=node-1;i>=0;--i)
		cout<<ans[i]<<" ";
	cout<<'\n';
	return 0;
}
