import queue
n,m,T=map(int,input().split())
adj=[[] for i in range(n+1)]
for i in range(m):
    u,v,t=map(int,input().split())
    adj[u].append((v,t))
q=[(0,[1])]
ans=0
res=[]
while q:
    ans=max(ans,len(res))
    time,path=q.pop(0)
    if path[-1]==n:
        res=path
        continue
    for v,t in adj[path[-1]]:
        if time+t<=T:
            q.append((time+t,path+[v]))
print(ans)
print(*res)
