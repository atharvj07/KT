import sys

for _ in range(int(input())):
    n, m = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    a.sort()
    tot = 0
    for i in range(n):
        b = []
        for j in range(i):
            b.append(a[j] / (i + 1))
        for j in range(i, n):
            tot += a[j] / (i + 1)
            b.append(0)
        if tot < m or m < sum(b):
            break
    print(["NO", "YES"][tot == m])
