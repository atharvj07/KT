import sys
input = sys.stdin.readline

n=int(input())
a=[int(char) for char in input().split()]
a.sort()
A=[[0]*n for _ in range(n)]
cnt=0
c=0
ans=0
for i in range(n):
 for j in range(n):
  if(a[i]==a[j]):
   A[i][j]=1
   cnt+=1
  elif(A[i][j]==0):
   c+=1
   A[i][j]=c
ans=cnt+c
print(ans,end="\n")
for i in range(n):
 if(A[i].count(1)==n):
  print(1,n)
  for j in range(n):
   print(*A[j])
   break
 else:
  for j in range(n):
   if(A[i][j]==1):
    continue
   else:
    ans-=1
    f=j
    break
  g=0
  for k in range(n):
   if(A[k][f]!=A[i][f]):
    g=1
    break
  if(g==1):
   print(n,1)
   for j in range(n):
    print(*A[j])
   break
  else:
   print(n-1,2)
   for j in range(n):
    if(j==i):
     continue
    else:
     print(*A[j])
   break
