import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        int n, k, count = 0;
        long[] freq = new long[1000001];
        Scanner scan = new Scanner(System.in);
        n = scan.nextInt();
        k = scan.nextInt();
        for(int i = 0; i < n; i++) {
            freq[scan.nextInt()]++;
        }
        for(long val: freq) {
            if(val >= k) {
                count += val * (val - 1) / 2;
            }
        }
        System.out.println(count);
    }
}
