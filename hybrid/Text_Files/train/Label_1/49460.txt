import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] stones = new int[n][3];
        for (int i = 0; i < n; i++) {
            stones[i][0] = sc.nextInt();
            stones[i][1] = sc.nextInt();
            stones[i][2] = sc.nextInt();
        }
        Arrays.sort(stones, new Comparator<int[]>() {
            @Override
            public int compare(int[] o1, int[] o2) {
                if (o1[0] * o1[1] * o1[2] != o2[0] * o2[1] * o2[2]) {
                    return Long.compare((long) o1[0] * o1[1] * o1[2], (long) o2[0] * o2[1] * o2[2]);
                }
                return Long.compare((long) o1[0] * o1[1] + o1[1] * o1[2] + o1[2] * o1[0], (long) o2[0] * o2[1] + o2[1] * o2[2] + o2[2] * o2[0]);
            }
        });
        int k = 0;
        long maxVolume = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long volume = (long) stones[i][0] * stones[i][1] * stones[i][2] + stones[j][0] * stones[j][1] * stones[j][2];
                if (volume > maxVolume) {
                    maxVolume = volume;
                    k = 2;
                }
            }
        }
        if (k == 0) {
            k = 1;
        }
        System.out.println(k);
        if (k == 1) {
            System.out.println(1);
        } else {
            System.out.println(1 + " " + 2);
        }
    }
}
