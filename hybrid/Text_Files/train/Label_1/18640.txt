#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> present(n + 1, 0);
  int leader = 0;
  for (int i = 0; i < m; i++) {
    char op;
    int id;
    cin >> op >> id;
    if (op == '+') {
      if (present[id] == 0) {
        leader++;
      }
      present[id] = 1;
    } else {
      if (present[id] == 1) {
        leader--;
      }
      present[id] = 0;
    }
  }
  if (leader == 0) {
    cout << 0 << endl;
  } else {
    vector<int> res;
    for (int i = 1; i <= n; i++) {
      if (present[i] == 1) {
        res.push_back(i);
      }
    }
    cout << res.size() << endl;
    for (int i = 0; i < res.size(); i++) {
      cout << res[i] << " ";
    }
    cout << endl;
  }
  return 0;
}
