import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        BigInteger x = scanner.nextBigInteger(), y = scanner.nextBigInteger(), z = scanner.nextBigInteger();
        BigInteger total = x.add(y), rem = total.remainder(z), ans = total.divide(z);
        BigInteger res = ans.subtract(rem);
        System.out.println(ans + " " + res);
    }
}
