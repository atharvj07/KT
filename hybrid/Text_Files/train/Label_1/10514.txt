#include <iostream>

int main() {
    int n;
    std::cin >> n;
    int prev, cur;
    std::cin >> prev;
    bool ok = prev == 0;
    for (int i = 1; i < n; ++i) {
        std::cin >> cur;
        if ((ok && prev == cur) || (!ok && prev != cur)) ok = false;
        prev = cur;
    }
    std::cout << (ok ? "YES" : "NO") << std::endl;
    return 0;
}
