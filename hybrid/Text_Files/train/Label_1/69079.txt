import sys

grid = [input() for _ in range(10)]
dx = [0, 1, 1, 1, 0, -1, -1, -1]
dy = [1, 1, 0, -1, -1, -1, 0, 1]

for i in range(10):
    for j in range(10):
        if grid[i][j] == '.':
            for k in range(8):
                cnt = 1
                for l in range(1, 5):
                    nx = i + dx[k] * l
                    ny = j + dy[k] * l
                    if 0 <= nx < 10 and 0 <= ny < 10 and grid[nx][ny] == 'X':
                        cnt += 1
                if cnt >= 5:
                    print('YES')
                    sys.exit()

print('NO')
