#include <bits/stdc++.h>
using namespace std;

int n, h[1005], ans = 1e9;
int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> h[i];
  sort(h + 1, h + n + 1);
  for (int i = 1; i < n; ++i) { ans = min(ans, h[i + 1] - h[i]); }
  cout << (ans + 1) / 2 << endl;
  cout << h[1] - (ans + 1) / 2 << ' ' << ans / 2 << endl;
  return 0;
}
