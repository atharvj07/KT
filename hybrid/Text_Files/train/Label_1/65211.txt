n = int(input())
arr = list(map(int, input().split()))

sum = 0
for i in range(n):
    sum += arr[i]

ans = 0
while n > 1:
    ans += sum
    sum = 0
    for i in range(n):
        sum += arr[i] // 2
    arr = []
    for i in range(n):
        if arr[i] % 2 == 0:
            arr.append(arr[i] // 2)
            arr.append(arr[i] // 2)
        else:
            arr.append(arr[i] // 2)
            arr.append(arr[i] // 2 + 1)
    n = len(arr)

ans += sum

print(ans)
