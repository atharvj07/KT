import sys
input = sys.stdin.readline

MX = 10**9
n, m, k = map(int, input().split())
x = list(map(int, input().split()))
g = [[] for _ in range(n+1)]
for _ in range(m):
    u, v, w = map(int, input().split())
    g[u].append((v, w))
    g[v].append((u, w))

def bfs(s):
    d = [MX] * (n+1)
    q = [(s, 0)]
    d[s] = 0
    while q:
        now, dis = q.pop(0)
        for nxt, w in g[now]:
            if dis + w < d[nxt]:
                d[nxt] = dis + w
                q.append((nxt, d[nxt]))
    return d

ans = []
for i in x:
    d = bfs(i)
    ans.append(max(d))
    
print(*ans)
