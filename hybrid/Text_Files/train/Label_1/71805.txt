n=int(input())
v=[]
for i in range(n):
    a,b=map(int,input().split())
    v.append([a,b])
ans=[-1]*(n+1)
cnt=0
for i in range(n):
    if v[i][1]==1:
        if i-1>=0 and ans[i-1]==-1:
            ans[i-1]=cnt
        elif i+1<=n-1 and ans[i+1]==-1:
            ans[i+1]=cnt
        else:
            ans[i]=cnt
            cnt+=1
    elif v[i][1]==2:
        if i-1>=0 and v[i-1][1]==0 and ans[i-1]==-1:
            ans[i-1]=cnt
        elif i+1<=n-1 and v[i+1][1]==0 and ans[i+1]==-1:
            ans[i+1]=cnt
        else:
            ans[i]=cnt
            cnt+=1
for i in range(n):
    if ans[i]!=-1:
        v[i][1]=0
print(cnt)
for i in range(n):
    if ans[i]!=-1:
        print(i+1,end=' ')
