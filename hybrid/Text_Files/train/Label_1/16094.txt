#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> l(n, 0), r(n, 0);
  for (int i = 1; i < n - 1; i++) {
    if (s[i] == s[i - 1]) {
      l[i] = l[i - 1];
    } else {
      l[i] = i;
    }
  }
  for (int i = n - 2; i >= 1; i--) {
    if (s[i] == s[i + 1]) {
      r[i] = r[i + 1];
    } else {
      r[i] = i;
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    ans += r[i] - l[i];
  }
  cout << ans << endl;
}
