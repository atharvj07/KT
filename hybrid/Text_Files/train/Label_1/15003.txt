#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    int sm = 0;
    priority_queue<int, vector<int>, greater<int>> pq;
    int a[n];
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        sm += a[i];
        pq.push(a[i]);
    }
    while (pq.size() > 1) {
        int x = pq.top();
        pq.pop();
        int y = pq.top();
        pq.pop();
        sm -= x - y;
        pq.push(x - y);
    }
    cout << sm << endl;
    return 0;
}
