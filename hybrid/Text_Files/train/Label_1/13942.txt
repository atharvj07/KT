#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pi;
const int maxn = 1e5 + 2;
int n, bir[1000002], ded[1000002], ans, yr, mx;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    bir[a]++;
    ded[b]++;
  }
  for (int i = 1; i <= 1000000; i++) {
    mx += bir[i] - ded[i];
    if (mx > ans) ans = mx, yr = i;
  }
  cout << yr << ' ' << ans << '\n';
}
