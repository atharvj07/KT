import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        boolean found = false;
        for (int i = 0; i < n; i++) {
            if (a[i] == k) {
                found = true;
            } else {
                for (int j = i + 1; j < n; j++) {
                    int diff = Math.abs(a[i] - a[j]);
                    if (diff == k) {
                        found = true;
                    } else if (diff > k) {
                        break;
                    }
                }
            }
        }
        System.out.println(found ? "POSSIBLE" : "IMPOSSIBLE");
    }
}
