#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn=1e6+1;
int n,k,a[maxn],b[maxn];
char s[maxn];
ll calc(int l,int r){
    int now=0;
    for(int i=l;i<=r;i++){
        now+=a[i];
    }
    return now<=k;
}
int main(){
    cin>>s;
    string t; cin>>t;
    cin>>k;
    n=strlen(s);
    for(int i=0;i<n;i++) a[i]=t[s[i]-'a']=='0';
    for(int i=0;i<n;i++){
        if(i) b[i]=b[i-1];
        b[i]+=calc(i,i);
    }
    ll ans=0;
    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if(calc(i,j)){
                ll x=b[j];
                if(i) x-=b[i-1];
                ans+=x+1;
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
