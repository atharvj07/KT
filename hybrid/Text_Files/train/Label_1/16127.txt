#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<vector<long long>> all;
  for (long long i = 0; i < n; i++) {
    vector<long long> cur;
    long long a, b;
    cin >> a >> b;
    cur.push_back(a);
    cur.push_back(b);
    all.push_back(cur);
  }
  long long ans = 0;
  long long a = all[0][0], b = all[0][1];
  for (long long i = 1; i < n; i++) {
    long long x = all[i][0], y = all[i][1];
    if (x == a and y == b) { ans++; }
    if (x != a and y != b) ans = 0;
    a = x, b = y;
  }
  cout << ans + 1 << endl;
  return 0;
}
