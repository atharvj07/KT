#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  vector<int> v1(n);
  unordered_map<int, int> u;
  for (int i = 0; i < n; i++) {
    cin >> v1[i];
    u[v1[i]] = i;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    ans = u[i];
    v[ans] = v1[i];
  }
  for (int i = 0; i < n; i++)
    cout << v[i] << " ";
  cout << endl;
  return 0;
}
