import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int k = in.nextInt(), n = in.nextInt(), m = in.nextInt();
            int[] a = new int[n], b = new int[m];
            for (int i = 0; i < n; i++) a[i] = in.nextInt();
            for (int i = 0; i < m; i++) b[i] = in.nextInt();
            int[] c = new int[n + m];
            int ca = 0, cb = 0;
            int i = 0, j = 0;
            for (; i < n + m; i++) {
                if (ca < n && a[ca] == 0) {
                    c[i] = 0;
                    ca++;
                } else if (cb < m && b[cb] == 0) {
                    c[i] = 0;
                    cb++;
                } else {
                    if (ca < n && a[ca] <= cb && a[ca] > 0) {
                        c[i] = a[ca];
                        ca++;
                    } else if (cb < m && b[cb] > 0) {
                        c[i] = b[cb];
                        cb++;
                    } else break;
                }
            }
            if (ca < n || cb < m) System.out.println(-1);
            else {
                for (int v : c) System.out.print(v + " ");
                System.out.println();
            }
        }
    }
}
