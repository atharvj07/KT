#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a[2005], b[2005];
  cin >> n;
  for (int i = 0; i < n; i++)
    cin >> a[i];
  copy(a, a + n, b);
  sort(b, b + n);
  int res = 0, sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
    res += sum >= b[i];
  }
  cout << res << endl;
  return 0;
}
