#include <iostream>
using namespace std;
int main() {
  long long ax, ay, bx, by, cx, cy;
  cin >> ax >> ay >> bx >> by >> cx >> cy;
  long long s1 = ax * by - ay * bx;
  long long s2 = bx * cy - by * cx;
  long long s3 = cx * ay - cy * ax;
  if (s1 == s2 && s2 == s3 && s1)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
  return 0;
}
