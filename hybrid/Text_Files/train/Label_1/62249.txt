n, k = map(int, input().split())
a = list(map(int, input().split()))

left, right, zeros, max_len, left_index = 0, 0, 0, 0, 0

for right in range(n):
    if a[right] == 0:
        zeros += 1
    while zeros > k:
        if a[left] == 0:
            zeros -= 1
        left += 1
    if right - left + 1 > max_len:
        max_len = right - left + 1
        left_index = left

for i in range(left_index, left_index + max_len):
    a[i] = 1

print(max_len)
print(*a)
