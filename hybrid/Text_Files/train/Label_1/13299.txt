#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll base=1ULL<<32;
ll mul(ll a,ll b){
    ll ans=(a*b)%base;
    while(ans<0)ans+=base;
    return ans;
}
int main() {
    ll t;
    cin>>t;
    while(t--){
        ll n;
        cin>>n;
        if(n%2==0)cout<<n/2<<' '<<n/2<<endl;
        else cout<<n-1<<' '<<1<<endl;
    }
    return 0;
}
