#include <iostream>
#include <cstring>

using namespace std;

int k, a, b;

int main() {
    int t;
    cin >> t;

    for(; t; t--) {
        cin >> a >> b >> k;
        k = k%4;
        
        if(k==0){
            cout << 0 << endl;
        } else if(k==1){
            cout << a << endl;
        } else if(k==2){
            cout << a-b << endl;
        } else {
            cout << a-b-a << endl;
        }
    }

    return 0;
}
