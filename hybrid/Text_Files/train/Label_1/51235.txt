import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Bananas {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Operation> operations = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int type = scanner.nextInt();
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            operations.add(new Operation(type, x, y));
        }

        int[] earliestTimes = new int[m + 1];
        Arrays.fill(earliestTimes, -1);

        int k = 0;
        for (int i = 0; i < n; i++) {
            Operation operation = operations.get(i);
            if (operation.type == 1) {
                for (int j = 0; j < operation.a; j++) {
                    k = (int) Math.ceil(k + operation.x);
                    if (k >= m) {
                        break;
                    }
                    if (earliestTimes[k] == -1) {
                        earliestTimes[k] = i + 1;
                    }
                }
            } else {
                for (int j = 0; j < operation.a; j++) {
                    k = (int) Math.ceil(k * operation.x);
                    if (k >= m) {
                        break;
                    }
                    if (earliestTimes[k] == -1) {
                        earliestTimes[k] = i + 1;
                    }
                }
            }
        }

        for (int i = 1; i <= m; i++) {
            System.out.print(earliestTimes[i] + " ");
        }
    }

    static class Operation {
        int type;
        double x;
        int a;

        public Operation(int type, int x, int y) {
            this.type = type;
            this.x = x / 100000.0;
            this.a = y;
        }
    }
}
