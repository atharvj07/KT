#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (auto &ai : a) cin >> ai;
  long long g = 1, s = 0;
  int l = 0, r = 0, ans = 0;
  while (true) {
    while (r < n && (s == 0 || g / s != k)) {
      g *= a[r];
      s += a[r];
      r++;
    }
    if (s == 0 || g / s == k) ans++;
    if (r == n) break;
    g /= a[l];
    s -= a[l];
    l++;
  }
  cout << ans;
}
