t = int(input())
for _ in range(t):
    s = input()
    t = input()
    count = {c: s.count(c) for c in 'abc'}
    if t == 'abc' and count['a'] > 0 and count['b'] > 0 and count['c'] > 0:
        print('a' * count['a'] + 'c' * count['c'] + 'b' * count['b'] + ''.join(sorted(s.replace('a', '').replace('b', '').replace('c', ''))))
    else:
        print(''.join(sorted(s)))
