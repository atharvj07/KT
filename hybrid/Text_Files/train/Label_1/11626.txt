#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;

  int n = s.length();
  int p = 0, l = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '-')
      l++;
    else
      p++;
  }

  if (p == 0 || l % p != 0) {
    cout << "NO" << endl;
    return 0;
  }

  int l_between_p = l / p;
  int l_moved = 0;
  bool ok = true;
  for (int i = 0; i < n; i++) {
    if (s[i] == '-') {
      l_moved++;
      if (l_moved > l_between_p) {
        ok = false;
        break;
      }
    } else {
      l_moved = 0;
    }
  }

  if (ok)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;

  return 0;
}
