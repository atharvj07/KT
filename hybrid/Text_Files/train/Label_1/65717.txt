n = int(input())
male = [0] * 367
female = [0] * 367

for _ in range(n):
    gender, a, b = input().split()
    a, b = int(a), int(b)
    if gender == 'M':
        for i in range(a, b+1):
            male[i] += 1
    else:
        for i in range(a, b+1):
            female[i] += 1

max_friends = 0
for i in range(1, 367):
    max_friends = max(max_friends, 2 * min(male[i], female[i]))

print(max_friends)
