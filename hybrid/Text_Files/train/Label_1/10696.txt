#include <iostream>
#include <cmath>
using namespace std;

int main() {
    long long a, b;
    cin >> a >> b;
    long long moves = 0, avg = (a + b) / 2;
    while (moves <= avg) {
        a -= moves + 1;
        b += moves + 1;
        moves++;
    }
    if (avg + 1 > b) {
        cout << "Vladik";
    } else {
        cout << "Valera";
    }
    return 0;
}
