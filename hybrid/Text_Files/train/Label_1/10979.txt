#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> h(n);
  for (int &x : h) cin >> x;
  vector<int> ans(n);
  int mx = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (h[i] > mx)
      mx = h[i];
    else
      ans[i] = mx - h[i];
  }
  for (int x : ans) cout << x << " ";
  return 0;
}
