import java.util.*;
import java.io.*;

class Main {
    static int[] subtrees;
    static int[] cntSubtrees;
    static int maxCnt;
    static int maxV;

    static void dfs1(int u, int p, ArrayList<Integer>[] adj) {
        subtrees[u] = 1;
        for (int v : adj[u]) {
            if (v == p) continue;
            dfs1(v, u, adj);
            subtrees[u] += subtrees[v];
        }
    }

    static void dfs2(int u, int p, ArrayList<Integer>[] adj) {
        for (int v : adj[u]) {
            if (v == p) continue;
            cntSubtrees[v] = cntSubtrees[u] + subtrees.length - subtrees[v];
            if (cntSubtrees[v] > maxCnt) {
                maxCnt = cntSubtrees[v];
                maxV = v;
            }
            dfs2(v, u, adj);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int n = Integer.parseInt(br.readLine());
        ArrayList<Integer>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }
        subtrees = new int[n + 1];
        dfs1(1, 1, adj);
        cntSubtrees = new int[n + 1];
        cntSubtrees[1] = subtrees.length;
        maxCnt = subtrees.length;
        maxV = 1;
        dfs2(1, 1, adj);
        bw.write(maxV + "\n");
        bw.flush();
        br.close();
        bw.close();
    }
}
