import sys
import math

h, m, s, t1, t2 = map(int, sys.stdin.readline().strip().split())
deg = lambda x: (x % 12) * 30 + x // 12*360
a = deg(t1) + m * 6 + s / 10
b = deg(t2) + m * 6 + s / 10
clockwise = b >= a
diff = abs(b - a)
hour = deg((h % 12 + 11) * 30)
print(["NO", "YES"][clockwise and diff < hour or not clockwise and diff > hour])
