n, q = map(int, input().split())
a = list(map(int, input().split()))
b = [int(input()) for i in range(q)]
a.sort()
for x in b:
    l, r = 0, n - 1
    while l <= r:
        m = (l + r) // 2
        if a[m] <= x:
            l = m + 1
        else:
            r = m - 1
    if a[r] == x:
        print(r + 1)
    else:
        print(-1)
