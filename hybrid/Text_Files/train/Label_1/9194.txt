#include <bits/stdc++.h>
using namespace std;

const int maxn=2e5+10;
int a[maxn],b[maxn],num[maxn],ans[maxn],sum[maxn],pre[maxn];
int32_t main(){
    int n,k,aa,b1,q;
    cin >> n >> k >> aa >> b1 >> q;
    while(q--){
        int type,x,y;
        cin >> type >> x;
        if(type==1){
            cin >> y;
            num[x]+=y;
            b[x]=min(num[x],b1);
        }
        else{
            ans[x]=0,sum[x]=0;
            for(int i=x,j=1;i<=n;i++,j++){
                sum[i]+=b[i]+pre[i-1];
                ans[x]=max(ans[x],min(sum[i],aa*j));
            }
            cout << ans[x] << "\n";
            pre[x]=sum[x];
        }
    }
    return 0;
}
