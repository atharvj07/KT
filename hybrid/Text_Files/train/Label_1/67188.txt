import sys

n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
b = [0] * n
m = int(sys.stdin.readline())
for _ in range(m):
    q = list(map(int, sys.stdin.readline().split()))
    if q[0] == 1:
        p, x = q[1], q[2]
        b[p - 1] += x
        for i in range(p, n):
            if b[i] > a[i]:
                b[i] = a[i]
                b[i + 1] += b[i] - a[i]
    else:
        print(b[q[1] - 1])
