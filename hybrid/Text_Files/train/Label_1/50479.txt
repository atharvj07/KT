import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long x = n, sum = 0;
            while (n > 0) {
                sum += n % 10;
                n /= 10;
            }
            while (true) {
                long g = gcd(x, sum);
                if (g > 1) break;
                x++;
            }
            System.out.println(x);
        }
    }

    private static long gcd(long a, long b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
