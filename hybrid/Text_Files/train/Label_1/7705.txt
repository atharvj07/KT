#include <iostream>
#include <string>
#include <cstring>

using namespace std;

int main() {
    string a, b;
    getline(cin, a);
    getline(cin, b);

    int lena = a.length();
    int lenb = b.length();

    while (lena > 0 && a[lena - 1] == '0') { lena--; }
    while (lenb > 0 && b[lenb - 1] == '0') { lenb--; }

    if (lena < lenb) { cout << "<" << endl; }
    else if (lena > lenb) { cout << ">" << endl; }
    else {
        for (int i = 0; i < lena; i++) {
            if (a[i] < b[i]) { cout << "<" << endl; return 0; }
            else if (a[i] > b[i]) { cout << ">" << endl; return 0; }
        }
        cout << "=" << endl;
    }

    return 0;
}
