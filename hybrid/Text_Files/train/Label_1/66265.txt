n, m = map(int, input().split())
f = [list(input()) for _ in range(n)]
for i in range(n):
    for j in range(m):
        if f[i][j] == '.':
            if [f[x][y] for x in range(max(0, i-1), min(n, i+2)) for y in range(max(0, j-1), min(m, j+2))] != ['0']*8:
                return print('NO')
        else:
            if [f[x][y] for x in range(max(0, i-1), min(n, i+2)) for y in range(max(0, j-1), min(m, j+2))].count('*') != int(f[i][j]):
                return print('NO')
print('YES')
