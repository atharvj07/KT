import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int numHorses = scanner.nextInt();
        int numEnemies = scanner.nextInt();

        LinkedList<Integer>[] enemies = new LinkedList[numHorses + 1];
        for (int i = 1; i <= numHorses; i++) {
            enemies[i] = new LinkedList<>();
        }

        for (int i = 0; i < numEnemies; i++) {
            int horse1 = scanner.nextInt();
            int horse2 = scanner.nextInt();
            enemies[horse1].add(horse2);
            enemies[horse2].add(horse1);
        }

        int[][] horseParties = new int[numHorses + 1][2];
        boolean[] visited = new boolean[numHorses + 1];

        for (int i = 1; i <= numHorses; i++) {
            if (!visited[i]) {
                if (!dfs(i, enemies, horseParties, visited, 0)) {
                    System.out.println(-1);
                    return;
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= numHorses; i++) {
            sb.append(horseParties[i][0]);
        }
        System.out.println(sb);
    }

    private static boolean dfs(int horse, LinkedList<Integer>[] enemies, int[][] horseParties, boolean[] visited, int party) {
        visited[horse] = true;
        horseParties[horse][party]++;

        for (int enemy : enemies[horse]) {
            if (horseParties[enemy][party] > 0) {
                return false;
            } else if (!visited[enemy]) {
                if (!dfs(enemy, enemies, horseParties, visited, 1 - party)) {
                    return false;
                }
            }
        }

        return true;
    }
}
