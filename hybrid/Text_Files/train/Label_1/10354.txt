#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1e6;
ll a[maxn], b[maxn];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
  }
  ll ans = a[1] + b[1], mx = b[1];
  for (int i = 2; i <= n; i++) {
    if (a[i] >= ans) {
      ans = a[i];
    }
    mx = max(mx, ans - a[i]);
    ans += b[i];
  }
  cout << ans << ' ' << mx << endl;
}
