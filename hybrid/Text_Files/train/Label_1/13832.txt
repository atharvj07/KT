#include <bits/stdc++.h>
using namespace std;

vector<int> g[100001];

long long subtree_apples[100001];
long long l[100001];
long long sub_path_parity[100001];

bool vis[100001];

void dfs(int x, int p) {
  subtree_apples[x] = 0;
  sub_path_parity[x] = 1;
  for (auto u : g[x]) {
    if (u == p || vis[u]) continue;
    dfs(u, x);
    subtree_apples[x] += subtree_apples[u];
    sub_path_parity[x] ^= sub_path_parity[u];
  }
  if (subtree_apples[x] == 0) {
    sub_path_parity[x] = 0;
  }
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  long long a[n + 1];
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 2; i <= n; i++) {
    int p;
    cin >> p;
    g[p].push_back(i);
    g[i].push_back(p);
  }
  dfs(1, 1);
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    if (sub_path_parity[i] == 0) continue;
    int rem = subtree_apples[i] % 2;
    rem = 2 - rem;
    if (rem == 0) {
      if (a[i] % 2 == 1) {
        ans++;
      }
    } else {
      if (a[i] % 2 == 0) {
        ans++;
      }
    }
  }
  cout << ans / 2 << endl;
  return 0;
}
