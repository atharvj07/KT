#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		int n;
		cin>>n;
		vector<long long int>a(n);
		for(int i=0;i<n;i++)
		cin>>a[i];
		long long int v=0;
		for(int i=1;i<n-1;i++)
		{
			if((a[i]<a[i+1] && a[i]>a[i-1]) || (a[i]>a[i+1] && a[i]<a[i-1]))
			v++;
		}
		long long int ans=v;
		for(int i=1;i<n-1;i++)
		{
			if(i!=1 && i!=n-2)
			{
				if(a[i-1]>a[i] && a[i+1]<a[i])
				{
					if((a[i-2]<a[i-1] && a[i-2]<a[i]) || (a[i+2]>a[i+1] && a[i+2]>a[i]))
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
				if(a[i-1]<a[i] && a[i+1]>a[i])
				{
					if((a[i-2]>a[i-1] && a[i-2]>a[i]) || (a[i+2]<a[i+1] && a[i+2]<a[i]))
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
			}
			else if(i==1)
			{
				if(a[i-1]>a[i] && a[i+1]<a[i])
				{
					if(a[i+1]>a[i+2] || a[i-2]>a[i-1])
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
				if(a[i-1]<a[i] && a[i+1]>a[i])
				{
					if(a[i+2]>a[i+1] || a[i-2]<a[i-1])
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
			}
			else if(i==n-2)
			{
				if(a[i-1]>a[i] && a[i+1]<a[i])
				{
					if(a[i-2]<a[i-1] || a[i]>a[i+1])
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
				if(a[i-1]<a[i] && a[i+1]>a[i])
				{
					if(a[i-2]>a[i-1] || a[i]<a[i+1])
					continue;
					else
					{
						ans=min(ans,v-1);
					}
				}
			}
		}
	    cout<<ans<<endl;
	}
	return 0;
}
