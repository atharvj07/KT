import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        int[] b = new int[n];
        String[] s1 = br.readLine().split("\\s+");
        String[] s2 = br.readLine().split("\\s+");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(s1[i]);
            b[i] = Integer.parseInt(s2[i]);
        }
        sort(a, b, n);
        for (int i = 0; i < n; i++) {
            System.out.print(((a[i] + b[i]) % n) + " ");
        }
    }

    static void sort(int[] a, int[] b, int n) {
        int[][] c = new int[n][2];
        for (int i = 0; i < n; i++) {
            c[i][0] = a[i];
            c[i][1] = b[i];
        }
        Arrays.sort(c, (x, y) -> (x[0] - y[0] != 0) ? x[0] - y[0] : x[1] - y[1]);
        for (int i = 0; i < n; i++) {
            a[i] = c[i][0];
            b[i] = c[i][1];
        }
    }
}
