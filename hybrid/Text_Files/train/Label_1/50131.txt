import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;

public class Main {
    static int n, m, q;
    static int[] p, rank;
    static int[] size;
    static int[] max;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        q = Integer.parseInt(line[2]);

        p = new int[n + 1];
        rank = new int[n + 1];
        size = new int[n + 1];
        max = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            p[i] = i;
            rank[i] = 0;
            size[i] = 1;
            max[i] = 0;
        }

        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            union(u, v);
        }

        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int u = Integer.parseInt(line[1]);
            int v = 0;
            if (type == 1) {
                System.out.println(max[find(u)]);
            } else {
                v = Integer.parseInt(line[2]);
                union(u, v);
            }
        }
    }

    static int find(int u) {
        if (p[u] == u) {
            return u;
        }
        return p[u] = find(p[u]);
    }

    static void union(int u, int v) {
        int x = find(u);
        int y = find(v);
        if (x == y) {
            return;
        }
        if (rank[x] < rank[y]) {
            p[x] = y;
            size[y] += size[x];
            max[y] = Math.max(max[y], max[x]);
        } else {
            p[y] = x;
            size[x] += size[y];
            max[x] = Math.max(max[x], max[y]);
            if (rank[x] == rank[y]) {
                rank[x]++;
            }
        }
        max[find(u)] = Math.max(max[find(u)], size[find(u)] - 1);
    }
}
