import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), d = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        long ans = 0, cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            if (cur < 0) {
                ans++;
                cur = 0;
            }
            if (cur > d) {
                System.out.println(-1);
                return;
            }
        }
        System.out.println(ans);
    }
}
