import sys
sys.setrecursionlimit(100000)
def solve():
	a = [int(x) for x in input().split()]
	dp = [[0] * (sum(a) + 1) for _ in range(len(a) + 1)]
	dp[0][0] = 1
	for i in range(1, len(a) + 1):
		for j in range(0, sum(a) + 1):
			dp[i][j] = dp[i - 1][j]
			if j >= a[i - 1]:
				dp[i][j] += dp[i - 1][j - a[i - 1]]
	print(dp[len(a)][sum(a) - 1])
for _ in range(int(input())):
	solve()
