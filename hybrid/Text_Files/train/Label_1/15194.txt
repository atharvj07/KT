#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    unordered_map<string, int> mp;
    mp[""] = 0;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        mp[s]++;
    }
    vector<string> arr;
    for (auto it = mp.begin(); it != mp.end(); it++) {
        string s = it->first;
        while (s.size() > 0) {
            string t = s;
            int mn = INT_MAX;
            for (int j = 0; j < s.size(); j++) {
                for (char c = 'a'; c <= 'z'; c++) {
                    t[j] = c;
                    int hsh = mp[t];
                    if (hsh > 0 && hsh < mn) {
                        mn = hsh;
                        s = t;
                    }
                }
            }
            if (mn == INT_MAX) {
                arr.clear();
                cout << "Impossible";
                return 0;
            }
            arr.push_back(s);
            s.clear();
        }
    }
    for (auto s : arr) {
        for (int i = 0; i < s.size(); i++) {
            cout << s[i];
        }
    }
}
