import java.util.Scanner;

public class NikolayAndHisHouse {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            String stairs = scanner.next();

            int[][] dp = new int[n + 1][2];

            // Base case: only one room is visited
            dp[1][0] = dp[1][1] = 1;

            for (int j = 2; j <= n; j++) {
                // If there is a staircase, we can visit the same room on both floors
                if (stairs.charAt(j - 1) == '1') {
                    dp[j][0] = Math.max(dp[j - 1][0], dp[j - 1][1]) + 1;
                    dp[j][1] = Math.max(dp[j - 1][0], dp[j - 1][1]) + 1;
                } else {
                    // If there is no staircase, we can only visit the room on the same floor
                    dp[j][0] = dp[j - 1][0] + 1;
                    dp[j][1] = dp[j - 1][1] + 1;
                }
            }

            // The maximum number of rooms is the maximum of the last elements in both rows
            System.out.println(Math.max(dp[n][0], dp[n][1]));
        }
    }
}
