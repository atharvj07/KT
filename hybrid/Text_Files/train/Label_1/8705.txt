#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  sort(s.begin(), s.end());
  sort(t.begin(), t.end());
  int minFlicks = 0, maxFlicks = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] > t[i]) maxFlicks++;
    if (s[i] < t[i]) minFlicks++;
  }
  cout << minFlicks << " " << maxFlicks << endl;
  return 0;
}
