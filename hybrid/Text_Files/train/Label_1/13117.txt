#include <cstdio>
#include <vector>
#include <iostream>
using namespace std;
vector <int> v;
int main(){
    int n, k, i, a, b = 0, c = 0, d = 0;
    cin >> n >> k;
    if(k == 1)
        for(i = 1 ; i <= n ; i++)
            cout << i << " ";
    else{
        for(i = 1 ; i <= (n / 2) ; i++){
            v.push_back(i);
            b++;
        }
        for(i = (n / 2) + 1 ; i <= n ; i++){
            v.push_back(i);
            c++;
        }
        if(k == 2){
            for(i = 0 ; i < b ; i++){
                swap(v[i], v[i + c]);
            }
        }
        else{
            for(i = 0 ; i < b ; i++){
                swap(v[i], v[i + d]);
                d++;
            }
        }
        for(i = 0 ; i < n ; i++)
            cout << v[i] << " ";
    }
    cout << endl;
    return 0;
}
