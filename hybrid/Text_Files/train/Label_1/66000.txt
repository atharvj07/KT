import sys
input = sys.stdin.readline

n, k, p = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
if sum(a)%2==0:p+=1
if p>k or (k-p)%2!=(sum(a)%2):print("NO");exit()
print("YES")
odd = []
even = []
for x in a:
    if x%2==0:
        even.append(x)
    else:
        odd.append(x)
for i in range(p):
    print(1,even.pop())
for i in range(k-p-1):
    print(1,odd.pop())
print(len(odd),*odd)
