#include <iostream>
#include <vector>
using namespace std;

int main() {
  string S, T;
  cin >> S >> T;
  int Q;
  cin >> Q;
  vector<int> res;
  while (Q--) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    a--;
    c--;
    int n = b - a + 1;
    if (n == d - c + 1) {
      int eq = 1;
      for (int i = 0; i < n; i++) eq &= S[a + i] == T[c + i];
      res.push_back(eq);
    } else {
      res.push_back(0);
    }
  }
  for (int x : res) cout << x;
}
