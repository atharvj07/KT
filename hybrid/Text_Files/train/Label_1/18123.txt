#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> monsters(n);
    for (int i = 0; i < n; i++) {
      int k;
      cin >> k;
      monsters[i].resize(k);
      for (int j = 0; j < k; j++) { cin >> monsters[i][j]; }
    }
    int max_armor = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < monsters[i].size(); j++) {
        max_armor = max(max_armor, monsters[i][j]);
      }
    }
    cout << max_armor + 1 << endl;
  }
  return 0;
}
