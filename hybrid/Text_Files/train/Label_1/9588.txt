#include <bits/stdc++.h>
using namespace std;
using ll = long long;
ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
ll lcm(ll a, ll b) { return a / gcd(a, b) * b; }
vector<ll> adj[300005];
vector<ll> val[300005];
const ll mod = 998244353;
ll powmod(ll base, ll p) {
  if (!p) return 1;
  if (p == 1) return base;
  ll v = powmod(base, p / 2);
  v = (v * v) % mod;
  if (p % 2) v = (v * base) % mod;
  return v;
}
ll get(ll n) {
  ll ans = 0;
  while (n > 1) {
    if (n % 2 == 0) {
      n /= 2;
      if (ans == 0) ans = 2;
    } else {
      if (ans == 0) ans = 1;
      break;
    }
  }
  return ans;
}
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, q;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) cin >> a[i];
    cin >> q;
    vector<ll> count(n);
    for (ll i = 2; i <= 300000; i++) {
      if (adj[i].empty()) continue;
      for (ll j : adj[i]) {
        val[i].push_back(j);
        val[j].push_back(i);
        count[i]++;
        count[j]++;
      }
    }
    for (ll i = 0; i < n; i++) {
      for (ll j : a) {
        ll v = lcm(a[i], j);
        ll w = gcd(a[i], j);
        ll val = v / w;
        if (val % 2 == 0 && get(v) == get(w)) count[i]++;
      }
    }
    vector<ll> ans(q);
    for (ll i = 0; i < q; i++) {
      ll w;
      cin >> w;
      for (ll j = 0; j < n; j++) {
        if (w % 2) break;
        for (ll k : val[count[j]]) count[j] = k, count[k] = j;
        w /= 2;
      }
      ll mx = 0;
      for (ll j = 0; j < n; j++) mx = max(mx, count[j]);
      ans[i] = mx;
    }
    for (ll i : ans) cout << i << "\n";
  }
}
