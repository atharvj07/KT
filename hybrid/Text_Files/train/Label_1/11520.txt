#include <iostream>
#include <string>
#include <vector>

int main() {
  std::string s;
  std::cin >> s;
  int n = s.size();
  std::vector<std::vector<int>> dp(n, std::vector<int>(n, -1));
  for (int i = 0; i < n; i++) { dp[i][i] = 0; }
  for (int l = 2; l <= n; l++) {
    for (int i = 0; i + l - 1 < n; i++) {
      int j = i + l - 1;
      if (s[i] == s[j]) {
        dp[i][j] = dp[i + 1][j - 1];
      } else {
        dp[i][j] = -1;
      }
    }
  }
  for (int i = n - 1; i >= 0; i--) {
    int ans = n;
    for (int j = i + 1; j < n; j++) {
      if (dp[i][j] != -1) {
        ans = std::min(ans, 1 + dp[i][j]);
      }
    }
    if (ans != n) {
      std::cout << ans << std::endl;
      return 0;
    }
  }
  std::cout << "Impossible" << std::endl;
  return 0;
}
