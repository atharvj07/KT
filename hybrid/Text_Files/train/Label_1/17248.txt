#include<bits/stdc++.h>
using namespace std;
pair<int,int>v[100005];
int main()
{
    int n,m,i,j,k,x,y,l=0,ans=0,sum=0;
    cin>>n>>m;
    for(i=0;i<n;i++)
    {
        cin>>x;
        v[i]={x,0};
    }
    for(i=1;i<=100000;i++)
    {
        int fl=0;
        for(j=0;j<n;j++)
        {
            if(v[j].first==i){fl=1; break;}
        }
        if(!fl)
        {
            if(i<=m)
            {
                v[l].first=i;
                v[l].second=1;
                sum+=i;
                l++;
                ans++;
                if(sum==m)break;
            }
            else break;
        }
    }
    cout<<ans<<endl;
    for(i=0;i<l;i++)
    {
        if(v[i].second)cout<<v[i].first<<" ";
    }
    return 0;
}
