#include <map>
#include <iostream>
using namespace std;

int main() {
  map<pair<int, int>, int> points;
  int n;
  cin >> n;
  int a1, b1, a2, b2;
  bool has_square = false;
  while (n--) {
    cin >> a1 >> b1 >> a2 >> b2;
    int diff_x = a2 - a1, diff_y = b2 - b1;
    int flag = (diff_x & 1) == (diff_y & 1);
    if (a1 == a2) flag = 1;
    if (b1 == b2) flag = 1;
    if (flag && diff_x == diff_y) { has_square = true; }
    for (int x = a1; x <= a2; x++)
      for (int y = b1; y <= b2; y++) { points[make_pair(x, y)]++; }
  }
  for (auto p : points) {
    if (p.second == 4) has_square = true;
  }
  cout << (has_square ? "YES" : "NO") << endl;
}
