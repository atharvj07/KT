import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] ids = new int[n];
        for (int i = 0; i < n; i++) {
            ids[i] = in.nextInt();
        }
        int[] sum = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + i;
        }
        int index = 0;
        for (int i = 1; i <= n; i++) {
            if (sum[i] >= k) {
                index = i;
                break;
            }
        }
        int count = k - sum[index - 1];
        for (int i = 1; i <= index; i++) {
            count--;
            if (count == 0) {
                System.out.println(ids[i - 1]);
                return;
            }
        }
    }
}
