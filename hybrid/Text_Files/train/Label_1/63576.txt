mod = 1000000007

def pow(x, p):
    if p == 0:
        return 1
    elif p == 1:
        return x
    else:
        val = pow(x, p // 2)
        val = (val * val) % mod
        if p % 2 != 0:
            val = (val * x) % mod
        return val

for _ in range(int(input())):
    n = int(input())
    if n == 3:
        print(7)
    else:
        print((pow(10, n // 4) - 1) % mod)
