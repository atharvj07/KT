#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int T;
    cin >> T;
    while (T--) {
        int n;
        string s;
        cin >> n >> s;
        int ans = 0;
        vector<int> v;
        for (int i = 0; i < n; i++) {
            if (s[i] != s[(i + 1) % n]) {
                ans++;
                if (s[i] != s[(i - 1 + n) % n])
                    v.push_back(i);
            }
        }
        if (ans == 0)
            cout << 0 << endl;
        else if (ans == 2)
            cout << 1 << endl;
        else {
            int sz = (int)v.size();
            if (sz == 2 && v[1] - v[0] == n - 1)
                cout << 2 << endl;
            else
                cout << 3 << endl;
        }
    }
    return 0;
}
