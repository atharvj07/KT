#include <bits/stdc++.h>
using namespace std;
const int N = 4e5 + 2;
char ch[N];
int g[N], n, p, pre[N];
bool vis[N];
int main() {
  cin >> n;
  cin >> ch;
  for (int j = 0; j < 26; j++) { g[ch[n - 1] - 'a' + j] = 1; }
  for (int i = n - 2; i >= 0; i--) {
    for (int j = 0; j < 26; j++) { g[ch[i] - 'a' + j] |= g[ch[i + 1] - 'a' + j]; }
  }
  for (int i = 0; i < 26; i++) {
    if (g[i]) { ch[p++] = 'a' + i; }
  }
  if (p % 2)
    reverse(ch + (p + 1) / 2, ch + p);
  else
    reverse(ch + (p) / 2, ch + p);
  cout << 1 << endl;
  for (int i = 0; i < p; i++) { cout << ch[i]; }
  cout << endl;
}
