#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  int m;
  cin >> s >> m;
  vector<int> w;
  for (int i = 0; i < 10; i++) {
    if (s[i] == '1') w.push_back(i + 1);
  }
  if (w.size() < 2 || m > w.size() * (w.size() + 1) / 2) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    int l = 0, r = 0, cnt = 0;
    while (cnt < m) {
      if (l == r || w[r] > w[l]) {
        cout << w[r] << " ";
        r = (r + 1) % w.size();
        cnt++;
      } else {
        cout << w[l] << " ";
        l = (l + 1) % w.size();
        cnt++;
      }
    }
    cout << endl;
  }
  return 0;
}
