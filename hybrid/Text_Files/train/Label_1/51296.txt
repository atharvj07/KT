import java.util.Scanner;
import java.util.HashMap;
import java.util.Map;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt() - 1;
        }

        int[] freq = new int[n];
        int[] lbound = new int[n];
        int[] rbound = new int[n];

        Map<Integer, Integer> rightBound = new HashMap<>();

        int maxFreq = 0;
        for (int i = n - 1; i >= 0; i--) {
            freq[arr[i]]++;
            if (freq[arr[i]] > maxFreq) {
                maxFreq = freq[arr[i]];
            }

            rbound[i] = rightBound.getOrDefault(arr[i], n);
            rightBound.put(arr[i], i);
        }

        int ans = 0;
        rightBound = new HashMap<>();
        for (int i = 0; i < n; i++) {
            lbound[i] = rightBound.getOrDefault(arr[i], -1);
            rightBound.put(arr[i], i);

            int leftFreq = i - lbound[i];
            int rightFreq = rbound[i] - i;
            int cnt = freq[arr[i]];

            if (leftFreq < cnt && rightFreq < cnt && leftFreq + rightFreq + 1 >= cnt) {
                ans = Math.max(ans, leftFreq + rightFreq + 1);
            }
        }

        System.out.println(ans);
    }
}
