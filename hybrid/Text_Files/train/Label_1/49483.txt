import java.io.BufferedReader;
import java.io.InputStreamReader;

public class Main {
    static final long MOD = 998244353;

    static class Edge{
        int u, v;
        Edge(int u, int v){
            this.u = u;
            this.v = v;
        }
    }

    static int n;
    static int[] p;
    static int[] sz;
    static Edge[] edges;
    static long[] res;

    static int find(int x){
        if(p[x] == x) return x;
        return p[x] = find(p[x]);
    }

    static void union(int a, int b){
        a = find(a);
        b = find(b);

        if(a != b){
            if(sz[a] < sz[b]){
                int aux = a;
                a = b;
                b = aux;
            }

            p[b] = a;
            sz[a] += sz[b];
        }
    }

    static long pow(long x, long y){
        long ans = 1;

        while(y > 0){
            if((y & 1) == 0){
                x = (x * x) % MOD;
                y >>= 1;
            }
            else{
                ans = (ans * x) % MOD;
                y--;
            }
        }

        return ans;
    }

    public static void main(String[] args) throws Exception{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(br.readLine());
        edges = new Edge[n - 1];

        for(int i = 0; i < n - 1; i++){
            String[] input = br.readLine().split(" ");
            int u = Integer.parseInt(input[0]) - 1;
            int v = Integer.parseInt(input[1]) - 1;

            edges[i] = new Edge(u, v);
        }

        p = new int[n];
        sz = new int[n];

        for(int i = 0; i < n; i++){
            p[i] = i;
            sz[i] = 1;
        }

        res = new long[n - 1];

        for(int i = 0; i < n - 1; i++){
            int u = edges[i].u;
            int v = edges[i].v;

            if(find(u) != find(v)){
                res[i] = (long)(sz[find(u)] * sz[find(v)]) % MOD;

                union(find(u), find(v));
            }
            else{
                res[i] = 0;
            }
        }

        long ans = pow(2, n - 1);

        for(int i = 0; i < n - 1; i++){
            ans = ans * pow(res[i], MOD - 2) % MOD;
        }

        System.out.println(ans % MOD);
    }
}
