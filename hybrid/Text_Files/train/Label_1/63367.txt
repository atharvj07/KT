n=int(input())
a=list(map(int,input().split()))
b=list(input())
l=-10**9;r=10**9
for i in range(4,n):
  if b[i-4:i+1]=='00001':
    for j in a[i-4:i+1]:
      if j>r:
        break
      if j<l:
        r=j-1
        break
    else:
      r=l
  if b[i-4:i+1]=='11110':
    for j in a[i-4:i+1]:
      if j<l:
        break
      if j>r:
        l=j+1
        break
    else:
      l=r
print(l,r)
