#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n, k, a, b;
    cin >> n >> k >> a >> b;
    if (abs(a - b) > 1)
        cout << "NO\n";
    else
    {
        int m = min(a, b);
        string s(n, 'G');
        for (int i = 0; i < m; i++)
            s[i] = 'B';
        for (int i = n - 1; i >= n - m; i--)
            s[i] = 'B';
        for (int i = 0; i < n; i++)
            cout << s[i];
        cout << endl;
    }
    return 0;
}
