import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.close();

        int ans = 0;
        for (int a = 1; a <= Math.sqrt(n); a++) {
            for (int b = a; b <= Math.sqrt(n); b++) {
                int c = n - a * b;
                if (c > 0 && b <= c) ans++;
            }
        }

        System.out.println(ans);
    }
}
