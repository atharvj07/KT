import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        long k = 0;
        List<String> words = new ArrayList<>();
        int n = 0, m = 0;
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String[] firstLine = br.readLine().split(" ");
            n = Integer.parseInt(firstLine[0]);
            m = Integer.parseInt(firstLine[1]);
            k = Long.parseLong(firstLine[2]);
            for (int i = 0; i < n; i++) {
                words.add(br.readLine());
            }
        }
        Collections.sort(words);

        List<String> sentences = generateSentences(words, m, new StringBuilder());
        if (k <= sentences.size()) {
            System.out.println(sentences.get((int) k - 1));
        } else {
            System.out.println("-");
        }
    }

    private static List<String> generateSentences(List<String> words, int m, StringBuilder sentence) {
        List<String> sentences = new ArrayList<>();
        if (sentence.length() == m) {
            sentences.add(sentence.toString());
            return sentences;
        }
        for (String word : words) {
            StringBuilder newSentence = new StringBuilder(sentence);
            newSentence.append(word);
            if (newSentence.length() <= m) {
                sentences.addAll(generateSentences(words, m, newSentence));
            }
        }
        return sentences;
    }
}
