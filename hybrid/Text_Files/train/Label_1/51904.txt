import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        List<Line> lines = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] input = br.readLine().split(" ");
            int a = Integer.parseInt(input[0]);
            int b = Integer.parseInt(input[1]);
            int c = Integer.parseInt(input[2]);
            lines.add(new Line(a, b, c));
        }

        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                Line line1 = lines.get(i);
                Line line2 = lines.get(j);
                for (int k = j + 1; k < n; k++) {
                    Line line3 = lines.get(k);
                    if (isOriginal(line1, line2, line3)) {
                        count++;
                    }
                }
            }
        }

        System.out.println(count);
    }

    private static boolean isOriginal(Line line1, Line line2, Line line3) {
        double x1 = line1.a;
        double y1 = line1.b;
        double x2 = line2.a;
        double y2 = line2.b;
        double x3 = line3.a;
        double y3 = line3.b;

        // Calculate the determinant of the 3x3 matrix
        // a1  b1  1  a2  b2  1  a3  b3  1
        // x1  y1  1  x2  y2  1  x3  y3  1
        double determinant = (x1 * y2 * 1) + (x2 * y3 * 1) + (x3 * y1 * 1)
                - (x1 * y3 * 1) - (x2 * y1 * 1) - (x3 * y2 * 1);

        // If the determinant is zero, the three lines are collinear
        if (determinant == 0) {
            return false;
        }

        // Calculate the circumcenter of the triangle formed by the three lines
        double circumcenterX = ((line2.c * y3 - line3.c * y2) * (line1.a * y2 - line2.a * y1))
                - ((line1.c * y2 - line2.c * y1) * (line2.a * y3 - line3.a * y2))
                / determinant;
        double circumcenterY = ((line2.c * x3 - line3.c * x2) * (line1.b * y2 - line2.b * y1))
                - ((line1.c * x2 - line2.c * x1) * (line2.b * y3 - line3.b * y2))
                / determinant;

        // Check if the circumcenter is at the origin
        return (circumcenterX == 0 && circumcenterY == 0);
    }

    private static class Line {
        double a;
        double b;
        double c;

        public Line(double a, double b, double c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }
}
