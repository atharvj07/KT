#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> ss(n);
  for (int i = 0; i < n; i++) { cin >> ss[i]; }
  string ans = ss[0];
  for (int i = 1; i < n; i++) {
    int l1 = ans.size(), l2 = ss[i].size();
    string tmp = "";
    for (int j = 0; j < l1; j++) {
      tmp += ans;
      tmp += ss[i][j];
    }
    tmp += ans;
    ans = tmp;
  }
  int mx = 0, cnt = 1;
  char c = ans[0];
  for (int i = 1; i < ans.size(); i++) {
    if (ans[i] == c) {
      cnt++;
    } else {
      mx = max(mx, cnt);
      c = ans[i];
      cnt = 1;
    }
  }
  mx = max(mx, cnt);
  cout << mx << endl;
  return 0;
}
