#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m, c = 0;
    cin >> n >> m;
    vector<string> g(n);
    for (int i = 0; i < n; i++) { cin >> g[i]; }
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) {
            if (g[i][j] == 'W') {
                if (i == 0 || j == 0 || g[i - 1][j] == 'B' ||
                    g[i][j - 1] == 'B')
                    c++;
            }
        }
    cout << (c == 0 ? -1 : c) << endl;
}
