#include <algorithm>
#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n), q(n), perm(n), ans(n);
    for (int &x : p) cin >> x;
    for (int &x : q) cin >> x;
    for (int i = 0; i < n; ++i) {
        perm[p[i]] = q[i];
    }
    vector<int> ord(n), rev_ord(n);
    for (int i = 0; i < n; ++i) {
        ord[perm[i]] = i;
        rev_ord[i] = perm[i];
    }
    for (int i = 0; i < n; ++i) {
        ans[i] = rev_ord[(ord[i] + ord[q[i]]) % n];
    }
    for (int x : ans) cout << x << ' ';
    cout << endl;
}
