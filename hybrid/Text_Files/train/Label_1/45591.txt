import java.util.*;
import java.io.*;

class Main {
    static int n, m;
    static char[] s;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st = new StringTokenizer(br.readLine(), " ");
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        s = new char[n + m];

        int ans = 0;
        int cur = 0;

        if (n == 0) {
            ans = -m * m;
            Arrays.fill(s, 'x');
        } else if (m == 0) {
            ans = n * n;
            Arrays.fill(s, 'o');
        } else {
            for (int i = 0; i < n; i++) {
                if (cur > 0) {
                    ans += cur * cur;
                    cur = 0;
                }
                s[i] = 'o';
                cur++;
            }

            if (cur > 0) {
                ans += cur * cur;
                cur = 0;
            }

            for (int i = n; i < n + m; i++) {
                if (cur < 0) {
                    ans += -cur * -cur;
                    cur = 0;
                }
                s[i] = 'x';
                cur--;
            }

            if (cur < 0) {
                ans += -cur * -cur;
                cur = 0;
            }
        }

        System.out.println(ans);
        System.out.println(s);
    }
}
