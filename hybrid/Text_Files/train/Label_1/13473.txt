#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<long long, long long>> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
    int cnt = 0;
    long long pos = 0, tpos;
    long long tme = v[0].first;
    for (int i = 0; i < n; i++) {
      tpos = v[i].first - tme;
      pos += tpos;
      if (pos == v[i].second) cnt++;
      tme = v[i].first;
    }
    cout << cnt << "\n";
  }
  return 0;
}
