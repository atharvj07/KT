import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashSet;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Set;

class Main {
    private static int[][] grid;
    private static Integer[][] dist;
    private static int r, c;
    private static final int[] dr = {-1, 0, 1, 0};
    private static final int[] dc = {0, 1, 0, -1};

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        r = Integer.parseInt(s[0]);
        c = Integer.parseInt(s[1]);
        grid = new int[r + 2][c + 2];
        dist = new Integer[r + 2][c + 2];
        for (int i = 1; i <= r; i++) {
            s = br.readLine().split("");
            for (int j = 1; j <= c; j++) {
                switch (s[j - 1]) {
                    case "0":
                        grid[i][j] = 0;
                        dist[i][j] = -1;
                        break;
                    case "1":
                        grid[i][j] = 1;
                        dist[i][j] = -1;
                        break;
                    case "2":
                        grid[i][j] = 2;
                        dist[i][j] = -1;
                        break;
                    case "3":
                        grid[i][j] = 3;
                        dist[i][j] = -1;
                        break;
                    case "4":
                        grid[i][j] = 4;
                        dist[i][j] = -1;
                        break;
                    case "5":
                        grid[i][j] = 5;
                        dist[i][j] = -1;
                        break;
                    case "6":
                        grid[i][j] = 6;
                        dist[i][j] = -1;
                        break;
                    case "7":
                        grid[i][j] = 7;
                        dist[i][j] = -1;
                        break;
                    case "8":
                        grid[i][j] = 8;
                        dist[i][j] = -1;
                        break;
                    case "9":
                        grid[i][j] = 9;
                        dist[i][j] = -1;
                        break;
                    case "T":
                        grid[i][j] = -1;
                        dist[i][j] = -1;
                        break;
                    case "S":
                        grid[i][j] = 0;
                        dist[i][j] = 0;
                        break;
                    case "E":
                        grid[i][j] = -2;
                        dist[i][j] = -1;
                        break;
                }
            }
        }
        bfs();
        System.out.println(dist[1][1]);
    }

    private static void bfs() {
        Queue<Pair> q = new LinkedList<>();
        q.add(new Pair(1, 1));
        while (!q.isEmpty()) {
            Pair p = q.poll();
            int x = p.x;
            int y = p.y;
            for (int i = 0; i < 4; i++) {
                int nx = x + dr[i];
                int ny = y + dc[i];
                if (nx >= 1 && nx <= r && ny >= 1 && ny <= c && grid[nx][ny] != -1) {
                    if (dist[nx][ny] == -1 || dist[nx][ny] > dist[x][y] + 1) {
                        dist[nx][ny] = dist[x][y] + 1;
                        q.add(new Pair(nx, ny));
                    }
                }
            }
        }
    }

    private static class Pair {
        int x;
        int y;

        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
