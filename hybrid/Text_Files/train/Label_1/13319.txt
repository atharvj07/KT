#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 1e5 + 10;

ll n, f, ans;
pair<ll, ll> a[maxn];
ll pref[maxn];

ll cal(ll x) {
  ll sum = 0;
  for (ll i = 1; i <= n; i++) {
    if (pref[i - 1] + x * a[i].first >= a[i].second) sum += a[i].second;
    else
      sum += pref[i - 1] + x * a[i].first;
  }
  return sum;
}

int main() {
  cin >> n >> f;
  for (ll i = 1; i <= n; i++) { cin >> a[i].first >> a[i].second; }
  sort(a + 1, a + 1 + n,
       [](const pair<ll, ll> &l, const pair<ll, ll> &r) { return l.second < r.second; });
  for (ll i = 1; i <= n; i++) {
    pref[i] = a[i].second;
    if (i) pref[i] += pref[i - 1];
  }
  ll l = 0, r = 1e9 + 1;
  while (l <= r) {
    ll mid = (l + r) >> 1;
    if (cal(mid) >= cal(mid + 1)) {
      ans = mid;
      r = mid - 1;
    } else
      l = mid + 1;
  }
  cout << cal(ans) << endl;
  return 0;
}
