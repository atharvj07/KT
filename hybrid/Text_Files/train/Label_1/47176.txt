import java.util.Scanner;
import java.util.HashSet;
import java.util.Set;

public class ValentinAndShockers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consumes the newline character

        Set<Character> possibleLetters = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(0);

            if (firstChar == '.') {
                // No shock, update possible letters
                for (char c : action.substring(2).toCharArray()) {
                    possibleLetters.remove(c);
                }
            } else if (firstChar == '!') {
                // Shock, remove letters not in the word
                for (char c : action.substring(2).toCharArray()) {
                    possibleLetters.add(c);
                }
            } else {
                // Guess, check if it's the correct letter
                char guess = action.charAt(2);
                if (possibleLetters.size() == 1 && possibleLetters.contains(guess)) {
                    // Correct guess, no excessive shocks
                    System.out.println(0);
                    return;
                } else {
                    // Incorrect guess, shock is excessive
                    System.out.println(1);
                    return;
                }
            }
        }
    }
}
