#include <bits/stdc++.h>
using namespace std;
#define trav(a, x) for(auto& a : x)
#define rep(i, a, b) for(int i = a; i < b; ++i)
#define sz(x) (int)(x).size()
#define lb(x) (int)(x.lower_bound(x.begin(),x.end())-x.begin())
#define ub(x) (int)(x.upper_bound(x.begin(),x.end())-x.begin())
typedef long long ll;
typedef pair<int, int> pii;
typedef vector<int> vi;
const ll MOD = 1e9 + 7;
const int MAXN = 2e5+5;
int n, k, skills[MAXN], ans[MAXN];
vector<int> adj[MAXN];
int32_t main() {
  cin >> n >> k;
  rep(i, 0, n) cin >> skills[i];
  rep(i, 0, k) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  vector<int> ord;
  iota(ord.begin(), ord.end(), 0);
  sort(ord.begin(), ord.end(), [&](int a, int b) { return skills[a] > skills[b]; });
  for (int p : ord) {
    for (int cur : adj[p]) {
      if (skills[cur] < skills[p]) ans[cur]++;
    }
  }
  rep(i, 0, n) cout << ans[i] << ' ';
}
