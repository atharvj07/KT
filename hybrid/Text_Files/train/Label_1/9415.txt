#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::pair<int, int>> exams(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> exams[i].first >> exams[i].second;
  }
  std::sort(exams.begin(), exams.end(), [](const auto &a, const auto &b) { return a.second < b.second; });
  int ans = exams[0].second;
  for (int i = 1; i < n; ++i) {
    ans = std::max(ans, exams[i].first);
  }
  std::cout << ans << std::endl;
  return 0;
}
