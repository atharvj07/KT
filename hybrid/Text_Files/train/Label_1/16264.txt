#include <bits/stdc++.h>
#define precision 1e-9
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int a;
        cin >> a;
        double b = (1.0 * a - sqrt(a * a - 4 * a)) / 2;
        double c = a - b;
        if (fabs(a - c * b) <= precision || fabs((c + b) - a) <= precision)
            cout << "Y " << fixed << setprecision(9) << c << " " << b << endl;
        else
            cout << "N" << endl;
    }
    return 0;
}
