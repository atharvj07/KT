#include <iostream>
#include <string>
#include <set>

int main() {
  std::string password;
  std::cin >> password;
  int n;
  std::cin >> n;
  std::set<std::string> words;
  for (int i = 0; i < n; i++) {
    std::string word;
    std::cin >> word;
    words.insert(word);
  }
  bool found = false;
  for (int i = 0; i < password.length(); i++) {
    for (int j = i + 1; j <= password.length(); j++) {
      std::string substring = password.substr(i, j - i);
      if (words.find(substring) != words.end()) {
        found = true;
        break;
      }
    }
  }
  std::cout << (found ? "YES" : "NO") << std::endl;
  return 0;
}
