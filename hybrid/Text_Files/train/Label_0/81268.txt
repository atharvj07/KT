from collections import deque
n_inp = input()
an_inp = input()
n = int(n_inp)
an = [(1, int(an_inp.split()[0]))] + sorted([(i + 2, int(a)) for (i, a) in enumerate(an_inp.split()[1:])], key=lambda x: x[1], reverse=True)
students = deque((s for (s, m) in an))
res = []
students.remove(1)
for (cur_st, max_mess) in an:
	if cur_st in students:
		continue
	for i in range(max_mess):
		if not students:
			break
		send_to = students.popleft()
		res.append('{} {}'.format(cur_st, send_to))
	if not students:
		print(len(res))
		for p in res:
			print(p)
		break
else:
	print('-1')
