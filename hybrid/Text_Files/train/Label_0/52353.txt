import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Main
{
   public static void main(String[] args) throws IOException
   {
      new Main().run();
   }

   StreamTokenizer in;
   PrintWriter out;

   int nextInt() throws IOException
   {
      in.nextToken();
      return (int)in.nval;
   }
   long nextLong() throws IOException
   {
      in.nextToken();
      return (long)in.nval;
   }
   
   void run() throws IOException
   {
    //  in = new StreamTokenizer(new BufferedReader(new FileReader("input.txt")));
     // out = new PrintWriter(new FileWriter("output.txt"));
      in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
    out = new PrintWriter(new OutputStreamWriter(System.out));
      solve();
      out.flush();
   }

   void solve() throws IOException
   { 
 //   long time = System.nanoTime();    
     int N=nextInt();
     int[] A=new int[N];
       for (int i = 0; i < N; i++) {
           A[i]=nextInt();
       }
  //     System.out.println("done");
      
// System.out.println("Finished in: " + (System.nanoTime() - time) / Math.pow(10, 6) + "ms");
   Arrays.sort(A);
     //  System.out.println("kjdbcqwjbcqwjc "+Arrays.binarySearch(A, 5));
   
   int max=A[A.length-1];
   BigInteger ans=BigInteger.ZERO;
 //  ArrayList<Integer> ts= new ArrayList<Integer>();
   int[] ts = new int[10000];
   int cn=0;
       for (int i = 1; i <= max; i++) {
          // System.out.println(i+"  i///////////////");
           cn=0;
           
           if(i%2==0)
           for (int j = 1; j <=Math.sqrt(i) ; j++) {
               if(i%j==0)
               {ts[cn]=j;cn++;
               if(j*j!=i)
               {ts[cn]=i/j;cn++;}
               }
           }
           else
              for (int j = 1; j <=Math.sqrt(i) ; j+=2) {
               if(i%j==0)
               {ts[cn]=j;cn++;
               if(j*j!=i)
               { ts[cn]=i/j;cn++;}
               }
           } 
          // System.out.println("cn "+cn);
        Arrays.sort(ts, 0, cn);
//           for (int j = 0; j < cn; j++) {
//               
//           
//           System.out.println("dd "+ts[j]);}
         //  System.out.println(ts);
           int ss=cn;
          BigInteger LL= BigInteger.valueOf(1000000007);
      BigInteger par=BigInteger.ONE;
      BigInteger par1;
       int[] s= new int[ss+1];//s[i]=.....i=<,<i+1
    //   ts.add(Integer.MAX_VALUE);
           for (int j = 0; j < ss-1; j++) {
               
               int j1=Arrays.binarySearch(A, ts[j]);
              if(j1<0)j1=-j1-1;
                      else if(j1==-1){if(A[0]>ts[j])j1=0;
              else j1=1;}
              else{
              while((A[j1]==ts[j])&&(j1>0))j1--; 
              if(j1!=0)j1++;
              else if(A[0]<ts[j])j1++;
              
              }
              int j2=Arrays.binarySearch(A, ts[j+1]);
              if(j2<0)j2=-j2-1;
                      else if(j2==-1){if(A[0]>ts[j+1])j2=0;
              else j2=1;}
              else{
              while((A[j2]==ts[j+1])&&(j2>0))j2--;   
               if(j2!=0)j2++;
              else if(A[0]<ts[j+1])j2++;
              }
              
          //     System.out.println((j+1)+" "+j1+" "+j2);
               s[j+1]=j2-j1;
           }
           int j1=Arrays.binarySearch(A, ts[ss-1]);
              if(j1<0)j1=-j1-1;
              else if(j1==-1){
             //     System.out.println("lskd");
                  if(A[0]>ts[ss-1])j1=0;
              else j1=1;
              }
              else{
              while((A[j1]==ts[ss-1])&&(j1>0))j1--; 
              if(j1!=0)j1++;
              else if(A[0]<ts[ss-1])j1++;
              
              }
           s[ss]=A.length-j1;
         //  System.out.println((ss)+" "+(A.length)+" "+j1);
           for (int j = 1; j <= ss; j++) {
             //  System.out.println(j+" s "+s[j]);
               if(j!=ss)
               par=(par.multiply(BigInteger.valueOf(j).modPow(BigInteger.valueOf(s[j]), LL))).mod(LL);
               else
               { par1=(par.multiply(BigInteger.valueOf(ss).modPow(BigInteger.valueOf(s[ss]), LL))).mod(LL);
               par=(par.multiply(BigInteger.valueOf(ss-1).modPow(BigInteger.valueOf(s[ss]), LL))).mod(LL);   
               par=(par1.subtract(par) ).mod(LL);   
               }                
           } 
        //   System.out.println("par "+par);
       ans=(ans.add(par)).mod(LL);
            //   System.out.println("ans "+ans);
       
       
       
       
       
       }
   
       System.out.println(ans);
  //  System.out.println("Finished in: " + (System.nanoTime() - time) / Math.pow(10, 6) + "ms");
   
   }
