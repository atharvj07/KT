public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        OutputWriter out = new OutputWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        public void solve(int testNumber, InputReader in, OutputWriter out) {
            int n = in.nextInt(), m = in.nextInt(), filled = 0, pos = 0;
            int[] id = IntStream.range(0, m).toArray();
            int[] arr = new int[n], A = in.nextIntArray(m);
            EzIntCustomSort.sort(id, (i, j) -> Integer.compare(A[j], A[i]));
            Arrays.fill(arr, -1);
            for (int color : id)
                while (A[color] > 0 && filled < n) {
                    int rem = n - filled;
                    while (rem > 0) {
                        if (arr[pos] == -1) {
                            if (arr[(pos + 1) % n] != color && arr[(pos + n - 1) % n] != color) {
                                arr[pos] = color;
                                A[color]--;
                                filled++;
                                break;
                            } else rem--;
                        }
                        pos = (pos + 1) % n;
                    }
                    if (rem == 0) break;
                }
            if (filled < n) out.println(-1);
            else {
                for (int c : arr) out.print((c + 1) + " ");
                out.println();
            }
        }

    }

    static class OutputWriter extends PrintWriter {
        public OutputWriter(OutputStream outputStream) {
            super(new BufferedOutputStream(outputStream));
        }

        public OutputWriter(Writer writer) {
            super(writer);
        }

        public void close() {
            super.close();
        }

    }

    static class InputReader {
        BufferedReader br;
        StringTokenizer st;

        public InputReader(InputStream inputStream) {
            br = new BufferedReader(new InputStreamReader(inputStream));
        }

        public String next() {
            while (st == null || !st.hasMoreElements()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public int[] nextIntArray(int n) {
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = nextInt();
            return a;
        }

        public String nextLine() {
            try {
                return br.readLine();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }

    }

    static final class EzIntCustomSort {
        private static final double HEAPSORT_DEPTH_COEFFICIENT = 2.0;
        private static final Random rnd = new Random();

        private EzIntCustomSort() {
        }

        private static int maxQuickSortDepth(int length) {
            if (length <= 1) {
                return 0;
            }
            int log = Integer.numberOfTrailingZeros(Integer.highestOneBit(length - 1)) + 1;
            return (int) (HEAPSORT_DEPTH_COEFFICIENT * log);
        }

        public static void sort(int[] a, EzIntComparator cmp) {
            quickSort(a, 0, a.length, cmp, 0, maxQuickSortDepth(a.length));
        }

        private static void quickSort(int[] a, int left, int right, EzIntComparator cmp, int depth, int maxDepth) {
            if (right - left <= 1) {
                return;
            }
            if (depth > maxDepth) {
                heapSort(a, left, right - left, cmp);
                return;
            }
            final int pivot = a[left + rnd.nextInt(right - left)];
            int i = left;
            int j = right - 1;
            do {
                while (cmp.compare(a[i], pivot) < 0) i++;
                while (cmp.compare(pivot, a[j]) < 0) j--;
                if (i <= j) {
                    int tmp = a[i];
                    a[i++] = a[j];
                    a[j--] = tmp;
                }
            } while (i <= j);
            quickSort(a, left, j + 1, cmp, depth + 1, maxDepth);
            quickSort(a, i, right, cmp, depth + 1, maxDepth);
        }

        private static void heapSort(int[] a, int offset, int size, EzIntComparator cmp) {
            // If size <= 1, nothing is executed
            for (int i = (size >>> 1) - 1; i >= 0; i--) {
                down(a, i, offset, size, cmp);
            }
            for (int i = size - 1; i > 0; i--) {
                int tmp = a[offset];
                a[offset] = a[offset + i];
                a[offset + i] = tmp;
                down(a, 0, offset, i, cmp);
            }
        }

        private static void down(int[] a, int index, int offset, int size, EzIntComparator cmp) {
            final int element = a[offset + index];
            final int firstLeaf = (size >>> 1);
            while (index < firstLeaf) {
                int largestChild = (index << 1) + 1;
                if (largestChild + 1 < size && cmp.compare(a[offset + largestChild + 1], a[offset + largestChild]) > 0) {
                    largestChild++;
                }
                if (cmp.compare(a[offset + largestChild], element) <= 0) {
                    break;
                }
                a[offset + index] = a[offset + largestChild];
                index = largestChild;
            }
            a[offset + index] = element;
        }

    }

    static interface EzIntComparator {
        int compare(int a, int b);

    }
}
