#include <bits/stdc++.h>
using namespace std;
template <typename Arg1>
void __f(const char* name, Arg1&& arg1) {
  cerr << name << " : " << arg1 << std::endl;
}
template <typename Arg1, typename... Args>
void __f(const char* names, Arg1&& arg1, Args&&... args) {
  const char* comma = strchr(names + 1, ',');
  cerr.write(names, comma - names) << " : " << arg1 << " | ";
  __f(comma + 1, args...);
}
const int N = int(1e5) + 10;
int dp[N][2], pos[N], wt[N];
vector<pair<int, int> > g[N];
int n, k;
void dfs(int x, int p) {
  vector<pair<int, int> > child;
  for (auto it : g[x]) {
    int y = it.first, wt = it.second;
    if (y == p) continue;
    ::wt[y] = wt;
    dfs(y, x);
    child.push_back(make_pair(dp[y][1] + wt, y));
  }
  sort(child.begin(), child.end());
  reverse(child.begin(), child.end());
  for (int i = (0); i < ((int)(child.size())); ++i) {
    if (i < k - 1) dp[x][1] += child[i].first;
    pos[child[i].second] = i;
  }
  int val = (int)(child.size()) < k ? 0 : child[k - 1].first;
  for (int i = (0); i < ((int)(child.size())); ++i) {
    int y = child[i].second;
    int add = (wt[y] + dp[y][0]) - (i < k - 1 ? (child[i].first - val) : 0);
    dp[x][0] = max(dp[x][0], dp[x][1] + add);
  }
}
int main() {
  scanf("%d", &n);
  scanf("%d", &k);
  for (int i = (0); i < (n - 1); ++i) {
    int u, v, c;
    scanf("%d", &u);
    scanf("%d", &v);
    scanf("%d", &c);
    g[u].push_back({v, c});
    g[v].push_back({u, c});
  }
  dfs(0, 0);
  printf("%d\n", dp[0][0]);
  return 0;
}
