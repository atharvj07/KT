#include <bits/stdc++.h>
using namespace std;
int n;
vector<string> v;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    string command;
    cin >> command;
    if (command == "cd") {
      string dir;
      cin >> dir;
      if (dir[0] == '/') {
        v.clear();
        string s;
        dir += '/';
        for (int j = 0; j < dir.size(); j++) {
          if (dir[j] != '/')
            s += dir[j];
          else {
            if (s == "..") {
              v.pop_back();
              s = "";
            } else {
              if (s != "") v.push_back(s);
              s = "";
            }
          }
        }
      } else {
        string s;
        dir += '/';
        for (int j = 0; j < dir.size(); j++) {
          if (dir[j] != '/')
            s += dir[j];
          else {
            if (s == "..") {
              v.pop_back();
              s = "";
            } else {
              if (s != "") v.push_back(s);
              s = "";
            }
          }
        }
      }
    } else {
      cout << '/';
      for (int i = 0; i < v.size(); i++) {
        cout << v[i] << '/';
      }
      cout << '\n';
    }
  }
}
