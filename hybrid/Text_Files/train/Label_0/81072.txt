(n, k) = map(int, input().split())
mas = list(map(int, input().split()))
lizt = list(map(int, input().split()))
di = {i: [] for i in range(1, k + 1)}
e = set()
for (i, x) in enumerate(mas):
	di[x].append(lizt[i])
	e.add(x)
c = k - len(e)
if c == 0:
	print(0)
	from sys import exit
	exit()
ops = []
for (i, x) in enumerate(di):
	m = 0
	for (j, y) in enumerate(di[x]):
		if m < y:
			m = y
	k = 0
	for (i, y) in enumerate(di[x]):
		if y != m:
			ops.append(y)
		elif k == 0:
			k = 1
		else:
			ops.append(y)
ops.sort()
print(sum(ops[:c]))
