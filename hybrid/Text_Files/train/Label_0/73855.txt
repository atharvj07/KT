import math

def g(a, b):
	return (a + b) * (b - a + 1) / 2

def f(a, b):
	return g(a, b) + g(1, b)
(n, h) = map(int, input().strip().split())
if h * (1 + h) / 2 >= n:
	l = int(1)
	r = int(h)
	while r > l:
		mid = int((l + r) / 2)
		if mid * (1 + mid) < n * 2:
			l = mid + 1
		else:
			r = mid
	print(int(l))
else:
	l = h
	r = 1500000000
	while r > l:
		mid = int((l + r) / 2)
		if f(h, mid) < n:
			l = mid + 1
		else:
			r = mid
	if n <= g(h, l) + g(1, l - 1):
		print(int(l - h + l))
	else:
		print(int(l - h + l + 1))
