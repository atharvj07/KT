#include <bits/stdc++.h>
using namespace std;
const int N = 100001;
int n, m;
vector<pair<int, int> > v[4][2 * N];
int res[N], sol[9];
int main() {
  scanf("%d%d", &n, &m);
  for (int r, c, i = 0; i < m; ++i) {
    scanf("%d%d", &r, &c);
    --r;
    --c;
    v[0][r].push_back(make_pair(c, i));
    v[1][c].push_back(make_pair(r, i));
    v[2][r + c].push_back(make_pair(r, i));
    v[3][r + n - c - 1].push_back(make_pair(r, i));
  }
  for (int k = 0; k < 4; ++k)
    for (int i = 0; i < 2 * N; ++i) {
      sort(v[k][i].begin(), v[k][i].end());
      for (int j = 0; j < v[k][i].size(); ++j)
        res[v[k][i][j].second] += (j > 0) + (j + 1 < v[k][i].size());
    }
  for (int i = 0; i < m; ++i) ++sol[res[i]];
  for (int i = 0; i < 9; ++i) printf("%s%d", i ? " " : "", sol[i]);
  return 0;
}
