for t in range(int(input())):
	input()
	(n, m) = map(int, input().split())
	nm = n * m
	mass = [list(map(int, input().split())) for i in range(n)]
	balance = max_a = max_b = 0
	for i in mass:
		balance += i[0] - i[1]
		max_a += min(m, i[0])
		max_b += min(m, i[1])
	min_a = nm - max_b
	if balance < nm - 2 * max_b:
		eat_a = min_a
	elif balance > 2 * max_a - nm:
		eat_a = max_a
	else:
		eat_a = (nm + balance + 1) // 2
	print(abs(balance - 2 * eat_a + nm))
	rest_a = eat_a - min_a
	for i in mass:
		cur_a = max(m - i[1], 0)
		add = min(rest_a, min(m - cur_a, i[0] - cur_a))
		cur_a += add
		rest_a -= add
		print(f'{cur_a} {m - cur_a}')
