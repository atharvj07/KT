import java.util.Arrays;
import java.util.HashMap;
import java.util.Scanner;

public class D {

  static final int MN = 100010;
  
  static long[] origVals = new long[MN];  
  static long[] sortedVals = new long[MN];
  static long[] bit = new long[MN+1];
  
  static void updateBit(int ix, long val) {
    for (int i = ix; i <= MN; i += i & -i) {
      bit[i] = Math.max(bit[i], val);
    }
  }
  
  static long retrieveFromBit(int ix) {
    long val = 0;
    for (int i = ix; i > 0; i -= i & -i) {
      val = Math.max(val, bit[i]);
    }
    return val;
  }
  
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    for (int i = 0; i < n; ++i) {
      long r = sc.nextInt();
      long h = sc.nextInt();
      origVals[i] = r * r * h;
      sortedVals[i] = origVals[i];
    }
    
    Arrays.sort(sortedVals, 0, n);
    HashMap<Long, Integer> compression = new HashMap<Long, Integer>(n);
    for (int i = 0; i < n; ++i) {
      // Overwriting a previous value is not an issue.
      compression.put(sortedVals[i], i+1); // indices should be 1-based
    }
    
    long ans = 0;
    for (int i = 0; i < n; ++i) {
      int compressedIx = compression.get(origVals[i]);
      long val = retrieveFromBit(compressedIx - 1);
      long newval = val + origVals[i];
      ans = Math.max(ans, newval);
      updateBit(compressedIx, newval);
    }
    System.out.println(Math.PI * retrieveFromBit(n));
  }
}
