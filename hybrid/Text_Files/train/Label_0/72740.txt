def build(b):
	s = 0
	b = b[::-1]
	for i in range(6):
		s += b[i] * 10 ** i
	return s
t = [-1] * 1000000
n = int(input())
t[n] = 0
v = [n]
while v != []:
	a = v.pop(0)
	x = []
	while a != 0:
		x = [a % 10] + x
		a //= 10
	x = [0] * (6 - len(x)) + x
	if sum(x[:3]) == sum(x[3:]):
		print(t[build(x)])
		break
	for i in range(6):
		for j in range(10):
			s = build(x[:i] + [j] + x[i + 1:])
			if t[s] == -1:
				t[s] = t[build(x)] + 1
				v += [s]
