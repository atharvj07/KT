def f(x, y):
	if x == y:
		return x - 1
	else:
		return (x - y) // 2

def g(x, y):
	return (x - y + 1) // 2
for _ in range(int(input())):
	n = int(input())
	r = list(map(int, input().split()))
	c = list(map(int, input().split()))
	l = []
	for i in range(n):
		l.append([r[i], c[i]])
	l.sort()
	ans = f(l[0][0], l[0][1])
	for i in range(1, n):
		if (l[i - 1][0] + l[i - 1][1]) % 2 == 0:
			ans += f(l[i][0] + 1 - l[i - 1][0], l[i][1] + 1 - l[i - 1][1])
		else:
			ans += g(l[i][0] + 1 - l[i - 1][0], l[i][1] + 1 - l[i - 1][1])
	print(ans)
