import sys
t = int(sys.stdin.readline())
for _ in range(t):
	(n, x) = map(int, sys.stdin.readline().split())
	A = list(map(int, sys.stdin.readline().split()))
	min_factors = None
	idx_min_factors = 0
	for (i, value) in enumerate(A):
		num_value_factors = 0
		while value % x == 0:
			num_value_factors += 1
			value //= x
		if min_factors is None or num_value_factors < min_factors:
			min_factors = num_value_factors
			idx_min_factors = i
	total = (min_factors + 1) * sum(A) + sum(A[:idx_min_factors])
	print(total)
