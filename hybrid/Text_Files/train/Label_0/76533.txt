else:
	inp = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

for _ in range(int(inp())):
	(x, y) = map(int, inp().split())
	ans = 0
	b = min(y, x - 1)
	while b > 1:
		q = x // (b + 1)
		if q >= b - 1:
			ans += b * (b - 1) // 2
			break
		else:
			r = x - q * (b + 1)
			k = (b - r) // (q + 1) + 1
			ans += q * k
			b -= k
	print(ans)
