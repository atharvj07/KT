import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class D1 {	
	BufferedReader reader;
    StringTokenizer tokenizer;
    PrintWriter out;
        
	public void solve() throws IOException {				
		int H = nextInt();
		int W = nextInt();
		int K = nextInt();
		
		boolean isFlip = K > 1 && W < H;
		if( isFlip) {
			int tmp = W; W = H; H = tmp;		
		}
		
		boolean[][] cH = new boolean[H][W-1];
		boolean[][] cV = new boolean[H-1][W];
		
		if( isFlip) {
			for(int i = 0; i < 2*W-1; i++)
				if( i % 2 == 0 ){
					for(int j = 0; j < H-1; j++)
						cV[j][i/2] = (char) reader.read() == 'E';
					reader.readLine();
				}
				else{
					for(int j = 0; j < H; j++)
						cH[j][i/2] = (char) reader.read() == 'E';
					reader.readLine();
				}
		}
		else{			
			for(int i = 0; i < 2*H-1; i++)
				if( i % 2 == 0 ){
					for(int j = 0; j < W-1; j++)
						cH[i/2][j] = (char) reader.read() == 'E';
					reader.readLine();
				}
				else{
					for(int j = 0; j < W; j++)
						cV[i/2][j] = (char) reader.read() == 'E';
					reader.readLine();
				}
		}
		
		

		
//		for(int i = 0; i < cH.length; i++){
//			for(int j = 0; j < cH[0].length; j++)
//				out.print( cH[i][j]? 'E': 'N' );
//			out.println();
//		}
//		out.println("----");
//		for(int i = 0; i < cV.length; i++){
//			for(int j = 0; j < cV[0].length; j++)
//				out.print( cV[i][j]? 'E': 'N' );
//			out.println();
//		}
		
		if( K == 1 ){
			int total = 0;
			for(int i = 0; i < cH.length; i++)
				for(int j = 0; j < cH[0].length; j++)
					if( cH[i][j] ) total++;
			
			for(int i = 0; i < cV.length; i++)
				for(int j = 0; j < cV[0].length; j++)
					if( cV[i][j] ) total++;
			
			if( ((double)total)/( (H-1)*W + (W-1)*H ) >= 3.0/4 ){
				out.println("YES");
				for(int i = 0; i < H; i++){
					for(int j = 0; j < W; j++)
						out.print(1 + " ");
					out.println();
				}
			}
			else{
				out.println("NO");
			}
			return;
		}
		
		out.println("YES");
		
		int[][] answer = new int[H][W];
		for(int i = 0; i < H; i++){
			answer[i][0] = 1;
			for(int j = 0; j < cH[0].length; j++){
				answer[i][j+1] = cH[i][j]? answer[i][j]: 3-answer[i][j];
			}
			
			if( i > 0 ){
				int correct = 0;
				for(int j = 0; j < W; j++){
					if( cV[i-1][j] && answer[i][j] == answer[i-1][j] )
						correct++;
					if( !cV[i-1][j] && answer[i][j] != answer[i-1][j] )
						correct++;
				}
				
				if( correct < W/2+W%2 ){
					for(int j = 0; j < W; j++)
						answer[i][j] = 3-answer[i][j];
				}
			}			
		}
		
		if( isFlip) {
			for(int j = 0; j < W; j++){
				for(int i = 0; i < H; i++)				
					out.print(answer[i][j] + " ");
				out.println();
			}
		}
		else{
			for(int i = 0; i < H; i++){
				for(int j = 0; j < W; j++)
					out.print(answer[i][j] + " ");
				out.println();
			}
		}
	}
	
	/**
	 * @param args
	 */
	public static void main(String[] args) {
		new D1().run();
	}
	
	public void run() {
        try {
            reader = new BufferedReader(new InputStreamReader(System.in));
            tokenizer = null;
            out = new PrintWriter(System.out);
            solve();
            reader.close();
            out.close();
        } catch (Exception e) {
            e.printStackTrace();
            System.exit(1);
        }
    }

    int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    String nextToken() throws IOException {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            tokenizer = new StringTokenizer(reader.readLine());
        }
        return tokenizer.nextToken();
    }

}
