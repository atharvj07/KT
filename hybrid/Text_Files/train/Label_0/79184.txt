t = int(input())
for _ in range(t):
	n = int(input())
	A = []
	for i in range(n):
		(a, *b) = map(int, input().split())
		A += [[x - 1 for x in b]]
	(B, C) = ([False] * n, [False] * n)
	for i in range(n):
		for j in A[i]:
			if not C[j]:
				B[i] = True
				C[j] = True
				break
	(x, y) = (-1, -1)
	for i in range(n):
		if not B[i]:
			x = i + 1
			for j in range(n):
				if not C[j]:
					y = j + 1
					break
			break
	if x == -1:
		print('OPTIMAL')
	else:
		print('IMPROVE')
		print(f'{x} {y}')
