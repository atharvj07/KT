#include<bits/stdc++.h>
#define ll long long
using namespace std;

char mat[7][7];
int dp[7][7];

int main() {
    int h, w;
    cin >> h >> w;
    for (int i = 1; i <= h; i++) {
        for (int j = 1; j <= w; j++) {
            cin >> mat[i][j];
        }
    }

    for (int i = 0; i <= h; i++) {
        for (int j = 0; j <= w; j++) {
            dp[i][j] = 9;
        }
    }

    if (mat[1][1] == '*') dp[1][1] = 1;
    else dp[1][1] = 0;

    for (int i = 1; i <= h; i++) {
        for (int j = 1; j <= w; j++) {
            if (i == 1 && j == 1) continue;
            if (mat[i][j] == '*') dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + 1;
            else {
                if (mat[i - 1][j + 1] != '*') dp[i][j] = min(dp[i][j], dp[i - 1][j]);
                if (mat[i + 1][j - 1] != '*') dp[i][j] = min(dp[i][j], dp[i][j - 1]);
            }
        }
    }

    cout << dp[h][w] << "\n";


//    for (int i = 0; i <= h; i++) {

//            cout << dp[i][j];
//        }
//        cout << endl;
//    }

    return 0;
}
