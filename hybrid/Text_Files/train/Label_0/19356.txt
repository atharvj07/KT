#include <bits/stdc++.h>
using namespace std;
pair<int, int> a[300005];
long long sum[300005];
int main() {
  int i, j, n, m, l, r, end;
  long long ans, s;
  while (scanf("%d", &n) != EOF) {
    ans = 0;
    for (i = 1; i <= n; i++) {
      scanf("%d", &a[i].first);
      a[i].second = i;
    }
    sort(a + 1, a + n + 1);
    for (i = 1; i <= n; i++) sum[i] = sum[i - 1] + a[i].first;
    scanf("%d", &m);
    l = -(m - 1);
    r = m - 1;
    for (i = 1, j = l; i <= m; i++, j += 2) ans += (long long)a[i].first * j;
    s = ans;
    end = m;
    for (i = m + 1; i <= n; i++) {
      long long v1 = (long long)(a[i - m].first) * l;
      long long v2 = (long long)(a[i].first) * r;
      long long v3 = (long long)(sum[i - 1] - sum[i - m]) * 2;
      s = s - v1 + v2 - v3;
      if (s < ans) {
        ans = s;
        end = i;
      }
    }
    for (i = end - m + 1; i <= end; i++)
      printf("%d%c", a[i].second, i == end ? '\n' : ' ');
  }
  return 0;
}
