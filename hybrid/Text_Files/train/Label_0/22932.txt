using namespace std;

#define REP(i,n) for(int i=0;i<(int)n;++i)
#define FOR(i,c) for(__typeof((c).begin())i=(c).begin();i!=(c).end();++i)
#define ALL(c) (c).begin(), (c).end()
const int INF = 1<<29;

int main() {
  string line;
  while(getline(cin,line)) {
    string str;
    FOR(it,line)
      if (isalpha(*it))
        str += toupper(*it);
    set<string> res;
    set<string> ignore;
    for (int len = str.size(); len >= 3; --len) {
      for (int pos = 0; pos < str.size()-len+1; ++pos) {
        string s = str.substr(pos, len);
        if (ignore.count(s)) continue;
        string t = s;
        reverse(ALL(t));
        if (s == t) {
          res.insert(s);
          for (int i=0; i<(len-1)/2; ++i) {
            ignore.insert(s.substr(i, len-i*2));
          }
        }
      }
    }
    bool f = 0;
    FOR (it, res) {
      if (f) cout << " ";
      else f = 1;
      cout << *it;
    }
    cout << endl;
  }
}
