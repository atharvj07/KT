#include <bits/stdc++.h>
using namespace std;
const int N = 1010;
const int mod = 1e9 + 7;
int n, m;
char s[N];
bitset<N> B[55];
vector<bitset<N> > v1, v2;
long long dp[N][N], ans = 1;
int main() {
  cin >> m >> n;
  for (int i = 1; i <= n; i++) {
    scanf("%s", s + 1);
    bitset<N> b;
    for (int j = 1; j <= m; j++) b[j] = s[j] - '0';
    B[i] = b;
  }
  for (int i = 1; i <= n; i++) {
    v2.clear();
    bitset<N> b = B[i];
    for (int j = 0; j < v1.size(); j++) {
      bitset<N> tmp = v1[j] & B[i];
      if (tmp.count()) {
        if (v1[j].count() != tmp.count()) v2.push_back(v1[j] ^ tmp);
        v2.push_back(tmp);
      } else {
        v2.push_back(v1[j]);
      }
      b ^= tmp;
    }
    if (b.count()) v2.push_back(b);
    v1 = v2;
  }
  dp[0][0] = 1;
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= i; j++) {
      dp[i][j] = (dp[i - 1][j] * j + dp[i - 1][j - 1]) % mod;
    }
  }
  for (int i = 0; i < v1.size(); i++) {
    int k = v1[i].count();
    long long tmp = 0;
    for (int j = 1; j <= k; j++) (tmp += dp[k][j]) %= mod;
    ans = ans * tmp % mod;
    m -= k;
  }
  if (m) {
    long long tmp = 0;
    for (int i = 1; i <= m; i++) {
      tmp += dp[m][i];
      tmp %= mod;
    }
    ans = ans * tmp % mod;
  }
  cout << ans;
}
