#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, b, mod, ans;
  cin >> n >> a >> b;
  if (n == 1)
    cout << a << endl;
  else {
    mod = abs(b) % n;
    if (mod == 0)
      ans = a;
    else {
      if (b < 0)
        ans = a + (n - mod);
      else
        ans = a + mod;
      if (ans > n) {
        ans = ans % n;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
