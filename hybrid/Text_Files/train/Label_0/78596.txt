import sys, threading
sys.setrecursionlimit(4000000)
threading.stack_size(10 ** 8)

def f(x, y):
	global m, dp
	a = 0
	b = 0
	if x == 1 and y == 1:
		return m - 1
	if x == 1 and y == 0:
		return 1
	if dp[x][y] >= 0:
		return dp[x][y]
	if y > 0 and x >= y:
		a = (m - 1) * f(x - 1, y - 1)
		a %= 998244353
	if x >= y:
		b = f(x - 1, y)
		b %= 998244353
	dp[x][y] = (a + b) % 998244353
	return dp[x][y]

def main():
	global m, dp
	dp = [[-1 for j in range(2001)] for i in range(2001)]
	(n, m, k) = map(int, input().split())
	if n > 1:
		ans = m * f(n - 1, k)
	else:
		ans = m
	print(ans % 998244353)
t = threading.Thread(target=main)
t.start()
t.join()
