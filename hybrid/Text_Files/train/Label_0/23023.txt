#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e6 + 10;
const int Mod = 998244353;
long long N;
long long fac[MAXN], ifc[MAXN];
long long ans;
inline long long fastpow_mod(long long a, long long b) {
  long long x = 1;
  while (b) {
    if (b & 1) x = (long long)x * a % Mod;
    a = (long long)a * a % Mod;
    b >>= 1;
  }
  return x;
}
inline void init(long long n) {
  register int i;
  for (*fac = i = 1; i <= n; ++i) fac[i] = (long long)fac[i - 1] * i % Mod;
  for (ifc[n] = fastpow_mod(fac[n], Mod - 2), i = n; i; --i)
    ifc[i - 1] = (long long)ifc[i] * i % Mod;
}
inline long long C(long long n, long long k) {
  return (long long)fac[n] * ifc[k] % Mod * ifc[n - k] % Mod;
}
int main() {
  register int i;
  init(1e6);
  scanf("%lld", &N);
  ans = fastpow_mod(3, (long long)N * N % (Mod - 1)) -
        fastpow_mod(fastpow_mod(3, N) - 3, N);
  for (i = 1; i <= N; ++i)
    ans = (ans +
           (long long)(i & 1 ? 1 : -1) * C(N, i) *
               (3 * fastpow_mod(fastpow_mod(3, N - i) - 1, N) % Mod +
                (fastpow_mod(3, i) - 3) *
                    fastpow_mod(3, (long long)N * (N - i) % (Mod - 1)) % Mod +
                Mod) %
               Mod +
           Mod) %
          Mod;
  printf("%lld\n", (ans + Mod) % Mod);
  return 0;
}
