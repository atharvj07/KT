import sys
import math
import copy
from collections import deque, Counter
input = sys.stdin.readline
dp = {}

def findValue(i, a):
	global dp
	ans = 0
	if i in dp:
		return ans
	while i <= len(a):
		ans += a[i - 1]
		i = i + a[i - 1]
		if i in dp:
			return ans
		dp[i] = 0
	return ans

def solve():
	dp.clear()
	n = int(input())
	a = list(map(int, input().split()))
	ans = -1
	for i in range(1, n + 1):
		ans = max(ans, findValue(i, a))
	print(ans)
t = int(input())
for _ in range(t):
	solve()
