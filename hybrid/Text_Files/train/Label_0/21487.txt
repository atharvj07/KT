#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0;i<n;i++)

int main(){
  int w,h,n; cin >> w >> h >> n;
  int x1 = 0; int x2 = w;
  int y1 = 0; int y2 = h;
  int x,y,a;
  rep(i,n){
    cin >> x >> y >> a;
    if(a == 1) x1 = max(x1, x);
    else if(a == 2) x2 = min(x2, x);
    else if(a == 3) y1 = max(y1, y);
    else if(a == 4) y2 = min(y2, y);
  }
  cout << max(0, x2-x1)*max(0, y2-y1) << endl;
}
