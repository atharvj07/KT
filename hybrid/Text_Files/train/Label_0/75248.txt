n = int(input())
a = list(map(int, input().split()))
pos = [0 for i in range(n + 1)]
for i in range(n):
	pos[a[i]] = i
dp = [0 for i in range(n + 1)]
for i in range(n - 1, 0, -1):
	j = pos[i]
	for k in range(j, n, i):
		if a[k] > i:
			if dp[a[k]] == 0:
				dp[i] = 1
	for k in range(j, -1, -i):
		if a[k] > i:
			if dp[a[k]] == 0:
				dp[i] = 1
for i in range(n):
	if dp[a[i]] == 0:
		print('B', end='')
	else:
		print('A', end='')
