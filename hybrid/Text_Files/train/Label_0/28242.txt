#include <bits/stdc++.h>
using namespace std;
int dp[805][805];
int vis[805][805];
int a[805];
int n;
int dfs(int l, int r) {
  if (vis[l][r]) return dp[l][r];
  vis[l][r] = 1;
  dp[l][r] = 1e9;
  if (l > r) return dp[l][r] = 0;
  if (l == r) return dp[l][r] = 1;
  if (l == r - 1) {
    if (a[l] == a[r])
      return dp[l][r] = 1;
    else
      return dp[l][r] = 2;
  }
  if (a[l] == a[r]) dp[l][r] = dfs(l + 1, r - 1);
  for (int i = l; i <= r; i++)
    dp[l][r] = min(dfs(l, i) + dfs(i + 1, r), dp[l][r]);
  return dp[l][r];
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  cout << dfs(1, n) << endl;
}
