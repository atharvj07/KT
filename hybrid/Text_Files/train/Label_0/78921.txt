import sys
(b, q, l, m) = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
if abs(b) > l:
	print(0)
	sys.exit()
if b == 0:
	if 0 in a:
		print(0)
	else:
		print('inf')
elif q == 0:
	if not 0 in a:
		print('inf')
	elif b in a:
		print(0)
	else:
		print(1)
elif q == 1:
	if b in a:
		print(0)
	else:
		print('inf')
elif q == -1:
	if not b in a or not -b in a:
		print('inf')
	else:
		print(0)
else:
	cnt = 0
	bn = b
	while abs(bn) <= l:
		if not bn in a:
			cnt += 1
		bn *= q
	print(cnt)
