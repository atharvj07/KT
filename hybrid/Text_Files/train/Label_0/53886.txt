import java.util.*;

public class B {

    public static void main(String args[]) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        if (n == 0) {
            System.out.println("YES\n1\n1\n3\n3");
        } else if (n == 1) {
            int a = scanner.nextInt();
            System.out.println("YES");
            System.out.println(a);
            System.out.println(3 * a);
            System.out.println(3 * a);
        } else if (n == 2) {
            List<Integer> numbers = new ArrayList<Integer>();
            numbers.add(scanner.nextInt());
            numbers.add(scanner.nextInt());
            Collections.sort(numbers);
            for (int i = 0; i < 1500; i++) {
                Integer k = i;
                numbers.add(k);
                Collections.sort(numbers);
                int num = getFourth(numbers.get(0), numbers.get(1), numbers.get(2));
                if (num > 0) {
                    System.out.println("YES");
                    System.out.println(i);
                    System.out.println(num);
                    return;
                }
                numbers.remove(k);
            }
            System.out.println("NO");
        } else if (n == 3) {
            List<Integer> numbers = new ArrayList<Integer>();
            for (int i = 0; i < 3; i++) {
                numbers.add(scanner.nextInt());
            }
            Collections.sort(numbers);
            for (int i = 0; i < 1500; i++) {
                Integer k = i;
                numbers.add(k);
                Collections.sort(numbers);
                if (check(numbers)) {
                    System.out.println("YES");
                    System.out.println(i);
                    return;
                }
                numbers.remove(k);
            }
            System.out.println("NO");
        } else if (n == 4) {
            List<Integer> numbers = new ArrayList<Integer>();
            for (int i = 0; i < 4; i++) {
                numbers.add(scanner.nextInt());
            }
            Collections.sort(numbers);
            if (check(numbers)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }

    private static int getFourth(int i1, int i2, int i3) {
        int d = 3 * i1;
        if (2 * (d - i1) == i2 + i3 && 4 * (d - i1) == i1 + i2 + i3 + d && 2 * (i2 + i3) == i1 + i2 + i3 + d)
            return d;
        if (3 * i1 == i3)
            return i1 + i3 - i2;
        if (i3 % 3 == 0) {
            d = i3 / 3;
            if (i2 + i1 == d + i3)
                return d;
        }
        return 0;
    }

    private static boolean check(List<Integer> numbers) {
        int a = numbers.get(0);
        int b = numbers.get(1);
        int c = numbers.get(2);
        int d = numbers.get(3);
        if (2 * (d - a) == b + c && 4 * (d - a) == a + b + c + d && 2 * (b + c) == a + b + c + d) {
            return true;
        }
        return false;
    }
}
