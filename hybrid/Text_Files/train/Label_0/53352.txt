import java.util.Scanner;

public class Main {

	private void start() {

		Scanner in = new Scanner(System.in);

		String s = in.nextLine();

		int q = 0;

		for (int i = 0; i < s.length(); i++) {
			int t = reserve((int) s.charAt(i), 8);

			int v = q - t;

			if (v < 0)
				v += 256;

			System.out.println(v);
			q = t;
		}
	}

	private int reserve(int t, int n) {
		String s = Integer.toBinaryString(t);

		while (s.length() < 8) {
			s = "0" + s;
		}

		char c[] = new char[8];

		for (int i = 0; i < 8; i++) {
			c[i] = s.charAt(7 - i);
		}
		return Integer.parseInt(new String(c), 2);
	}

	public static void main(String[] args) {
		new Main().start();
	}
}
