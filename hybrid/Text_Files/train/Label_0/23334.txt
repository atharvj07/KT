#include <bits/stdc++.h>
using namespace std;
int solve(string s) {
  string a = s;
  a.insert(a.begin(), '1');
  a.push_back('0');
  int max_val1 = 0, t = 0, max_val = 0;
  a.push_back('2');
  for (int i = 0; i < a.size() - 1; i++) {
    if (a[i] == a[i + 1]) {
      t++;
    } else {
      max_val1 += t;
      t = 0;
    }
  }
  s.insert(s.begin(), '0');
  s.push_back('1');
  t = 0;
  s.push_back('2');
  for (int i = 0; i < s.size() - 1; i++) {
    if (s[i] == s[i + 1]) {
      t++;
    } else {
      max_val += t;
      t = 0;
    }
  }
  return min(max_val / 2, max_val1 / 2);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    cout << solve(s) << endl;
  }
}
