#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 5;
const int mod = 1e9 + 7;
long long qpow(long long a, long long b) {
  long long ans = 1;
  a %= mod;
  assert(b >= 0);
  for (; b; b >>= 1) {
    if (b & 1) ans = ans * a % mod;
    a = a * a % mod;
  }
  return ans;
}
long long gcd(long long a, long long b) { return b > 0 ? gcd(b, a % b) : a; }
int n, m, T;
char s[maxn];
int pre[maxn];
int sz[maxn];
struct node {
  int l, r, len;
} a[maxn];
int b[maxn];
int Find(int x) { return pre[x] == x ? x : pre[x] = Find(pre[x]); }
void Union(int l, int r) {
  int fal = Find(l);
  int far = Find(r);
  if (fal != far) {
    pre[far] = fal;
    a[fal].r = a[far].r;
    a[fal].len += a[far].len;
  }
}
int vis[maxn];
int dir[maxn];
int ans[maxn];
char p[2] = {'W', 'B'};
int main() {
  cin >> n >> m;
  cin >> s;
  for (int i = 0; i < n; i++) {
    ans[i] = -1;
    pre[i] = i;
    sz[i] = 1;
    a[i].l = a[i].r = i;
    a[i].len = 1;
  }
  for (int i = 0; i < n; i++) {
    b[i] = (s[i] == 'W' ? 0 : 1);
    if (s[i] == s[(i - 1 + n) % n]) {
      Union((i - 1 + n) % n, i);
    }
  }
  int cnt = 0;
  queue<pair<int, int> > q;
  for (int i = 0; i <= n; i++) {
    if (vis[Find(i)] || a[Find(i)].len == 1) continue;
    vis[Find(i)]++;
    cnt++;
    int L = a[Find(i)].l, R = a[Find(i)].r, len = a[Find(i)].len;
    for (int j = L; len; j = (j + 1 + n) % n, len--) {
      ans[j] = b[Find(i)];
    }
    q.push({a[Find(i)].l, 0});
    dir[a[Find(i)].l] = -1;
    q.push({a[Find(i)].r, 0});
    dir[a[Find(i)].r] = 1;
  }
  if (cnt == 0) {
    for (int i = 0; i < n; i++) {
      b[i] ^= (m & 1);
      cout << p[b[i]];
    }
    return 0;
  }
  int N = min(n, m);
  while (!q.empty()) {
    pair<int, int> u = q.front();
    q.pop();
    int v = (u.first + dir[(u.first)] + n) % n;
    if (u.second == N || ans[v] != -1) continue;
    q.push({v, u.second + 1});
    dir[v] = dir[u.first];
    ans[v] = ans[u.first];
  }
  for (int i = 0; i < n; i++) {
    if (ans[i] == -1) {
      ans[i] = b[i] ^ (m & 1);
    }
    cout << p[ans[i]];
  }
  return 0;
}
