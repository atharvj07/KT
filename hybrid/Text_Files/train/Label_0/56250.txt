
import java.util.*;
import java.io.*;

public class MakeThemEqual {

	// https://codeforces.com/contest/1417/problem/D
	
	public static void main(String[] args) throws IOException, FileNotFoundException {
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		//BufferedReader in = new BufferedReader(new FileReader("MakeThemEqual"));

		int t = Integer.parseInt(in.readLine());
		while (t-->0) {
			int n = Integer.parseInt(in.readLine());
			long[] arr = new long[n];
			StringTokenizer st = new StringTokenizer(in.readLine());
			long sum=0;
			for (int i=0; i<n; i++) {
				arr[i] = Long.parseLong(st.nextToken());
				sum+=arr[i];
			}
			if (sum%n!=0) {
				System.out.println(-1);
				continue;
			}
			ArrayList<long[]> op = new ArrayList<>();
			for (int i=1; i<n; i++) {
				// arr[i] % (i+1) == 0
				// increase by (i+1) - (arr[i]%(i+1));
				long needed = ((i+1) - (arr[i]%(i+1)))%(i+1);
				if (needed>0) {
					op.add(new long[] {1, i+1, needed});
					arr[0] -= needed;
					arr[i] += needed;
				}
				needed = arr[i]/(i+1);
				if (needed>0) {
					op.add(new long[] {i+1, 1, needed});
					arr[0] += needed*(i+1);
					arr[i] -= needed*(i+1);
				}
			}
			long needed = sum/n;
			for (int i=1; i<n; i++) {
				op.add(new long[] {1, i+1, needed});
			}
			System.out.println(op.size());
			for (int i=0; i<op.size(); i++) {
				StringBuilder s = new StringBuilder();
				for (int k=0; k<3; k++) {
					s.append(op.get(i)[k]);
					s.append(" ");
				}
				System.out.println(s);
			}
		}
	}
}
