#include <iostream>
#include <stdio.h>
#include <math.h>
using namespace std;

int main(){
  int a,b,c,d,e,f;
  int g,h,i,sum;
  
  cin >> a >> b >> c >> d >> e >> f;
  while(a!=0||b!=0||c!=0||d!=0||e!=0||f!=0){
    g=a+d;
    h=b+e;
    i=c+f;
    sum=0;
    if((g>1&&h>1&&i>1)&&((g%3==2&&h%3==2&&i%3==0)||(g%3==2&&h%3==0&&i%3==2)||(g%3==0&&h%3==2&&i%3==2))) sum++;
    sum+=g/3;
    g=g%3;
    sum+=h/3;
    h=h%3;
    sum+=i/3;
    i=i%3;
    while(g>0&&h>0&&i>0){
      g--;h--;i--;
      sum++;
    }
    
    cout << sum << endl;
    
    cin >> a >> b >> c >> d >> e >> f;
  }
  return 0;
}
