import sys
input = sys.stdin.readline
n = int(input())
permutation = list(map(int, input().split()))
go = set(range(1, n + 1))
reached = set()
ans = 0
while go:
	x = go.pop()
	while x not in reached:
		reached.add(x)
		x = permutation[x - 1]
		if x in go:
			go.remove(x)
	ans += 1
if ans == 1:
	ans = 0
permutation = input().split()
if permutation.count('1') % 2 == 0:
	ans += 1
print(ans)
