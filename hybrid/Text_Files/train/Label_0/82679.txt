import sys
inf = float('inf')
from math import ceil, floor, log, sqrt, factorial, pi, gcd

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
(a, b, c, d) = get_ints()
if a * d < c * b:
	(a, b) = (b, a)
	(c, d) = (d, c)
a *= d
c *= b
p = a - c
q = a
g = gcd(p, q)
p //= g
q //= g
print(f'{p}/{q}')
