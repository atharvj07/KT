#include <bits/stdc++.h>
using namespace std;
struct nd {
  int id, p, c1, c2;
  bool operator<(const nd &y) const { return p > y.p; }
} t[200002];
priority_queue<nd> q[3];
bool v[200002];
int main() {
  int n, m, c;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> t[i].p;
  }
  for (int i = 1; i <= n; i++) {
    cin >> t[i].c1;
  }
  for (int i = 1; i <= n; i++) {
    cin >> t[i].c2;
  }
  for (int i = 1; i <= n; i++) {
    t[i].id = i;
    if (t[i].c1 != t[i].c2) {
      q[t[i].c1 - 1].push(t[i]);
      q[t[i].c2 - 1].push(t[i]);
    } else {
      q[t[i].c1 - 1].push(t[i]);
    }
  }
  cin >> m;
  for (int i = 1; i <= m; i++) {
    cin >> c;
    bool f = false;
    while (!q[c - 1].empty()) {
      nd tn = q[c - 1].top();
      if (v[tn.id]) {
        q[c - 1].pop();
      } else {
        cout << tn.p << " ";
        f = true;
        v[tn.id] = true;
        q[c - 1].pop();
        break;
      }
    }
    if (!f) {
      cout << "-1 ";
    }
  }
  return 0;
}
