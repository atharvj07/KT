#include <bits/stdc++.h>
using namespace std;
long long int min_res = -1;
void solve(long long int a, long long int b, int k, long long int aux) {
  if (k == 0 && min_res == -1) {
    min_res = aux;
    return;
  }
  if (k == 0) {
    min_res = min(min_res, aux);
    return;
  }
  long long int d = b - (long long int)k + 2;
  long long int i = a;
  for (; i < d; i++) {
    solve(i + 1, b, k - 1, aux ^ i);
  }
}
int main() {
  long long int a, b;
  int k;
  scanf("%lld %lld %d", &a, &b, &k);
  if (b - a + 1 <= 5) {
    long long int res = -1;
    int min_mask;
    for (int i = 0; i < (1 << (b - a + 1)); i++) {
      if (__builtin_popcount(i) <= k && __builtin_popcount(i) >= 1) {
        long long int aux = 0;
        for (int j = 0; j < b - a + 1; j++) {
          if ((i & (1 << j)) != 0) aux ^= (a + (long long int)j);
        }
        if (res == -1) {
          res = aux;
          min_mask = i;
        } else {
          if (res > aux) {
            res = aux;
            min_mask = i;
          }
        }
      }
    }
    printf("%lld\n", res);
    printf("%d\n", __builtin_popcount(min_mask));
    for (int i = 0; i < b - a + 1; i++) {
      if ((min_mask & (1 << i)) != 0) printf("%lld ", a + (long long int)i);
    }
    printf("\n");
  } else {
    if (k == 1) {
      printf("%lld\n", a);
      printf("%d\n", k);
      printf("%lld\n", a);
    } else if (k == 2) {
      printf("1\n");
      printf("2\n");
      if (a % 2 == 0)
        printf("%lld %lld\n", a, a + 1);
      else
        printf("%lld %lld\n", a + 1, a + 2);
    } else if (k == 3) {
      long long int x = 3;
      long long int y = 2;
      long long int z = 1;
      while (x <= b) {
        if (x >= a && x <= b && y >= a && y <= b && z >= a && z <= b) {
          printf("0\n");
          printf("3\n");
          printf("%lld %lld %lld\n", x, y, z);
          return 0;
        } else {
          x <<= 1;
          y <<= 1;
          y += 1;
          z <<= 1;
          z += 1;
        }
      }
      printf("1\n");
      printf("2\n");
      if (a % 2 == 0)
        printf("%lld %lld\n", a, a + 1);
      else
        printf("%lld %lld\n", a + 1, a + 2);
    } else {
      printf("0\n");
      printf("4\n");
      if (a % 2 == 0)
        printf("%lld %lld %lld %lld\n", a, a + 1, a + 2, a + 3);
      else
        printf("%lld %lld %lld %lld\n", a + 1, a + 2, a + 3, a + 4);
    }
  }
}
