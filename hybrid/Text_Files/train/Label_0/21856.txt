#include <bits/stdc++.h>
using namespace std;
int near[10010];
int main() {
  int n, res = 0;
  cin >> n;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    near[u]++;
    near[v]++;
  }
  for (int i = 1; i <= n; i++) {
    res += (near[i] * (near[i] - 1)) >> 1;
  }
  cout << res << endl;
  return 0;
}
