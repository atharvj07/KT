import java.util.Scanner;
//comment
public class pa11 {
public static void main(String args[]){
	Scanner sc=new Scanner(System.in);
	String s=sc.next();
	String t=sc.next();
	int len=s.length();
	int count=0;
	String out="";
	int dummy=0;
	int dummy1=0;
	int count1=0;
	boolean bol=false;
	for(int i=0;i<len;i++){
		if(t.charAt(i)-s.charAt(i)>1){
			count++;
			dummy=i;
			break;
		}
}
	if(count==1){
		System.out.println(s.substring(0,dummy)+(char)(t.charAt(dummy)-1)+s.substring(dummy+1));
	}else{
		String str="";
		String str1="";
		for(int i=0;i<len;i++){
			
			if(bol){
				if(s.charAt(i)!='z'){
					str+=(char)(s.charAt(i)+1);
					dummy++;
				}else if(t.charAt(i)!='a'){
					str1+=(char)(t.charAt(i)-1);
					dummy1++;
				}
				else{
					str+=s.charAt(i);
					str1+=t.charAt(i);
			}
			}else{
				str+=s.charAt(i);
				str1+=t.charAt(i);
			}
			if(t.charAt(i)-s.charAt(i)==1){
				bol=true;
			}
			
	}
		if(dummy>0){
			System.out.println(str);
		}else if(dummy1>0){
			System.out.println(str1);
		}else{
			System.out.println("No such string");
		}
		}
}
}
