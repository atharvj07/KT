import math
import sys

def fn(n, a):
	if n == 2:
		print(0)
		return 0
	if a == [a[0]] * n:
		print(0)
		return 0
	for i in range(n - 1):
		if a[i] == a[i + 1]:
			print(-1)
			return -1
	diff = 0
	diff1 = 0
	f = 0
	f1 = 0
	for i in range(n - 1):
		if a[i + 1] > a[i] and f == 0:
			diff = a[i + 1] - a[i]
			f = 1
		elif a[i + 1] < a[i] and f1 == 0:
			f1 = 1
			diff1 = a[i] - a[i + 1]
		if a[i + 1] > a[i] and a[i + 1] - a[i] != diff:
			print(-1)
			return -1
		elif a[i + 1] < a[i] and a[i] - a[i + 1] != diff1:
			print(-1)
			return 1
	if diff1 == 0 and diff != 0:
		print(0)
		return 1
	if diff == 0 and diff1 != 0:
		print(0)
		return 1
	m = diff + diff1
	c = diff
	if m <= c or m < max(a):
		print(-1)
		return -1
	print(m, diff)
input = sys.stdin.read()
data = list(map(int, input.split()))
t = data[0]
l = 0
for i in range(t):
	n = data[l + 1]
	a = data[l + 2:l + 2 + n]
	l = l + 1 + n
	fn(n, a)
