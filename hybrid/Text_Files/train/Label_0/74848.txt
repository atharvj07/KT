x = int(input())
u = input()
v = input()
w = input()

def wtf(s):
	ans = 0
	for i in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ':
		ans = max(ans, s.count(i))
	need = len(s) - ans
	if need == 0 and x == 1:
		return len(s) - 1
	if need > x:
		return ans + x
	else:
		return len(s)
	return ans
a = [wtf(u), wtf(v), wtf(w)]
maxi = max(a)
(a, b, c) = (a[0], a[1], a[2])
if a == maxi and b != maxi and (c != maxi):
	print('Kuro')
elif b == maxi and a != maxi and (c != maxi):
	print('Shiro')
elif c == maxi and a != maxi and (b != maxi):
	print('Katie')
else:
	print('Draw')
