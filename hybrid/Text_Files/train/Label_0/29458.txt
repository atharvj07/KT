#include <bits/stdc++.h>
using namespace std;
struct S {
  long long sta;
  long long end;
};
bool cmp(S a, S b) {
  if (a.end == b.end) {
    return a.sta < b.sta;
  } else {
    return a.end < b.end;
  }
}
int main() {
  struct S s[500100] = {};
  long long n, i, j, cnt = 0;
  while (~scanf("%lld", &n)) {
    memset(s, 0, sizeof(s));
    for (i = 0; i < n; i++) {
      scanf("%lld%lld", &s[i].sta, &s[i].end);
    }
    sort(s, s + n, cmp);
    for (j = 0, i = 1; i < n; i++) {
      if (s[j].end < s[i].sta) {
        cnt++;
        j = i;
      }
    }
    printf("%lld\n", cnt + 1);
    cnt = 0;
  }
}
