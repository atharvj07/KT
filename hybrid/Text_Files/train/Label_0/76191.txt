def gen(n):
	a = []
	if n % 2 == 1:
		k = n - 2
		a = [n]
		while k > 0:
			a.append(a[-1] + k)
			k -= 2
		a = a + a[:n // 2][::-1]
	else:
		k = n - 2
		a = [n]
		while k > 0:
			a.append(a[-1] + k)
			k -= 2
		a = a + a[:n // 2][::-1]
	return a

def f(a, b):
	t = gen(len(a))
	for i in range(len(a)):
		a[i] = a[i] * t[i]
	a = sorted(a, reverse=True)
	b = sorted(b)
	ans = 0
	for i in range(len(a)):
		ans = (ans + a[i] * b[i]) % 998244353
	return ans
n = input()
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))
print(f(a, b))
