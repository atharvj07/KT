#include <queue>
#include <cstdio>
#include <cstring>
#include <algorithm>
using namespace std;

const int maxn = 300000 + 10;
int n, t, a[maxn], c[maxn], s[maxn];

inline bool Check(int x, int k) {
  int p = lower_bound(c + 1, c + t + 1, x) - c;
  return s[p - 1] / x + t - p + 1 >= k;
}

int main(int argc, char const *argv[])
{
  scanf("%d", &n);
  for(int i = 1; i <= n; ++i) scanf("%d", a + i);

  sort(a + 1, a + n + 1);
  for(int i = 1; i <= n; ++i) ++c[t = t + (a[i] != a[i - 1])];

  sort(c + 1, c + t + 1);
  for(int i = 1; i <= t; ++i) s[i] = s[i - 1] + c[i];

  for(int k = 1; k <= n; ++k) {
    int mid = 0, ans = 0, lef = 1, rig = s[t] / k;
    while( lef <= rig ) mid = (lef + rig) >> 1, Check(mid, k) ? lef = mid + 1, ans = mid : rig = mid - 1;
    printf("%d\n", ans);
  }

  return 0;
}
