def kedane(a, n):
	ans = 0
	curr = 0
	for i in range(n):
		curr += a[i]
		if curr > ans:
			ans = max(ans, curr)
		elif curr <= 0:
			curr = 0
	return ans
for __ in range(int(input())):
	n = int(input())
	ar = list(map(int, input().split()))
	a = []
	for i in range(1, n):
		if i % 2 == 1:
			a.append(ar[i] - ar[i - 1])
	b = []
	for i in range(2, n):
		if i % 2 == 0:
			b.append(ar[i - 1] - ar[i])
	ans = 0
	for i in range(n):
		if i % 2 == 0:
			ans += ar[i]
	print(max(ans, ans + kedane(a, len(a)), ans + kedane(b, len(b))))
