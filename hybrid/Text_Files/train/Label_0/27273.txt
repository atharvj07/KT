#include <bits/stdc++.h>
using namespace std;
int a[110];
int main() {
  int n, t;
  cin >> n >> t;
  for (int i = 0; i < n; i++) {
    int s, d;
    cin >> s >> d;
    while (s < t) s += d;
    a[i] = s;
  }
  int ans = 0, mn = 1000000000;
  for (int i = 0; i < n; i++) {
    if (a[i] - t >= 0 && a[i] - t < mn) {
      mn = a[i] - t;
      ans = i;
    }
  }
  cout << ans + 1 << endl;
  return 0;
}
