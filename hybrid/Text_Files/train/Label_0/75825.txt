def solve():
	n = int(input())
	arr = list(input())
	if int(arr[3]) > 5:
		arr[3] = '0'
	if n == 24:
		if int(arr[0] + arr[1]) > 23:
			arr[0] = '0'
	if n == 12:
		if int(arr[0] + arr[1]) > 12 or int(arr[0] + arr[1]) == 0:
			if arr[1] == '0':
				arr[0] = '1'
			else:
				arr[0] = '0'
	print(''.join(arr))
solve()
