for _ in range(int(input())):
	N = int(input())
	X = [[a for a in input()] for _ in range(N)]
	C = [0] * 3
	D = [0] * 3
	(aaa, bbb) = (0, 0)
	f = 0
	for i in range(N):
		Xi = X[i]
		for j in range(N):
			if Xi[j] == 'X':
				C[(i + j) % 3] += 1
			elif Xi[j] == 'O':
				D[(i + j) % 3] += 1
	for k in range(3):
		for l in range(3):
			if k == l:
				continue
			if C[k] + D[l] <= (sum(C) + sum(D)) // 3:
				for i in range(N):
					Xi = X[i]
					for j in range(N):
						if (i + j) % 3 == k:
							if Xi[j] == 'X':
								Xi[j] = 'O'
								aaa += 1
						elif (i + j) % 3 == l:
							if Xi[j] == 'O':
								Xi[j] = 'X'
								bbb += 1
				f = 1
				break
		if f:
			break
	for x in X:
		print(''.join(x))
