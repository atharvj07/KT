import sys
t = int(sys.stdin.readline())
for i in range(t):
	(a, b, x) = map(int, sys.stdin.readline().split())
	(a, b) = (min(a, b), max(a, b))
	while a:
		if x == a or x == b:
			print('YES')
			break
		elif x > a and x < b and (x % a == b % a):
			print('YES')
			break
		else:
			b = b % a
			(a, b) = (b, a)
	else:
		print('NO')
