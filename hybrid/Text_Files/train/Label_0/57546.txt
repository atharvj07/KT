
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.StringTokenizer;
import static java.lang.Integer.parseInt;

public class Youmna {

    static int a1[] = {-1, -1, -1, 0, 0, 1, 1, 1};
    static int a2[] = {-1, 0, 1, -1, 1, -1, 0, 1};
    static char mat[][];
    static int r1;
    static int c1;

    public static void main(String[] args) throws Exception {
        BufferedReader r = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter w = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer tok = new StringTokenizer(r.readLine());

        r1 = parseInt(tok.nextToken());
        c1 = parseInt(tok.nextToken());
        mat = new char[r1][c1];

        for (int i = 0; i < r1; i++) {
            mat[i] = r.readLine().toCharArray();
        }
        boolean f = true;
        A: for (int i = 0; i < mat.length; i++) {
            B:  for (int j = 0; j < mat[i].length; j++) {
                if (mat[i][j] != '*') {
                    int n;
                    if (mat[i][j] == '.') {
                        n = 0;
                    } else {
                        n = mat[i][j] - '0';
                    }
                    if (!number(i, j, n)) {
                        System.out.println("NO");
                        f = false;
                        break A;
                    }

                }
            }
        }
            if (f) {
                System.out.println("YES");
            }
    }

    public static boolean number(int r2, int c2, int n) {
        int ans = 0;
        for (int i = 0; i < 8; i++) {
            int r3 = r2+a1[i];
            int c3 = c2+a2[i];
            if (r3< r1 && r3 >= 0 && c3< c1 && c3 >= 0 && mat[r3][c3] == '*') {
                ans++;
            }
        }
        return ans == n;

    }
}
