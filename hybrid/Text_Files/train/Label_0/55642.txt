import java.io.*;
import java.util.*;
public class Main
{
	int ax,ay,bx,by,tx,ty;
	final double inf=1e18+1;
	private void solve()throws IOException
	{
		ax=nextInt();
		ay=nextInt();
		bx=nextInt();
		by=nextInt();
		tx=nextInt();
		ty=nextInt();
		int n=nextInt();
		double maxdista=-inf,maxdistb=-inf,seconda=-inf,secondb=-inf;
		int agoes=0,bgoes=0;
		double sum=0;
		for(int i=1;i<=n;i++)
		{
			int x=nextInt();
			int y=nextInt();
			double dista=dista(x,y);
			double distb=distb(x,y);
			double distbin=distbin(x,y);
			sum+=2*distbin;
			if(distbin-dista>maxdista)
			{
				seconda=maxdista;
				maxdista=distbin-dista;
				agoes=i;
			}
			else if(distbin-dista>seconda)
				seconda=distbin-dista;
			if(distbin-distb>maxdistb)
			{
				secondb=maxdistb;
				maxdistb=distbin-distb;
				bgoes=i;
			}
			else if(distbin-distb>secondb)
				secondb=distbin-distb;
		}
		if(n==1)
			out.println(Math.min(sum-maxdista,sum-maxdistb));
		else
		{

			if(agoes==bgoes)
			{
				if(maxdista<=secondb && maxdista<0)
					maxdista=0;
				else if(maxdista>=secondb && secondb<0)
					secondb=0;
				if(seconda<=maxdistb && seconda<0)
					seconda=0;
				else if(seconda>=maxdistb && maxdistb<0)
					maxdistb=0;
				out.println(Math.min(sum-maxdista-secondb,sum-seconda-maxdistb));
			}
			else
			{
				if(maxdista<=maxdistb && maxdista<0)
				maxdista=0;
				else if(maxdista>=maxdistb && maxdistb<0)
				maxdistb=0;
				out.println(sum-maxdista-maxdistb);
			}
		}
	}
	double dista(int x,int y)
	{
		return Math.sqrt(1.0d*(ax-x)*(ax-x)+1.0d*(ay-y)*(ay-y));
	}
	double distb(int x,int y)
	{
		return Math.sqrt(1.0d*(bx-x)*(bx-x)+1.0d*(by-y)*(by-y));
	}
	double distbin(int x,int y)
	{
		return Math.sqrt(1.0d*(tx-x)*(tx-x)+1.0d*(ty-y)*(ty-y));
	}

	 
	///////////////////////////////////////////////////////////

	public void run()throws IOException
	{
		br=new BufferedReader(new InputStreamReader(System.in));
		st=null;
		out=new PrintWriter(System.out);

		solve();

		br.close();
		out.close();
	}
	public static void main(String args[])throws IOException{
		new Main().run();
	}
	BufferedReader br;
	StringTokenizer st;
	PrintWriter out;
	String nextToken()throws IOException{
		while(st==null || !st.hasMoreTokens())
		st=new StringTokenizer(br.readLine());
		return st.nextToken();
	}
	String nextLine()throws IOException{
		return br.readLine();
	}
	int nextInt()throws IOException{
		return Integer.parseInt(nextToken());
	}
	long nextLong()throws IOException{
		return Long.parseLong(nextToken());
	}
	double nextDouble()throws IOException{
		return Double.parseDouble(nextToken());
	}
}
