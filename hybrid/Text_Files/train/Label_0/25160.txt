#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  double n, k;
  cin >> n >> k;
  vector<pair<double, double>> v;
  double x, y;
  for (long long int i = 0; i < n; i++) {
    cin >> x >> y;
    v.push_back(make_pair(x, y));
  }
  double distance = 0;
  for (long long int i = 0; i < n - 1; i++) {
    x = v[i].first - v[i + 1].first;
    x = x * x;
    y = v[i].second - v[i + 1].second;
    y = y * y;
    distance += (double)(sqrt(x + y));
  }
  distance = (double)distance * k;
  double ans = (double)distance / 50.0;
  cout << fixed << setprecision(9) << ans;
}
