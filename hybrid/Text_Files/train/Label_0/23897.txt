#include <bits/stdc++.h>
using namespace std;
int n, m;
long long f[501][501], m1[501], m2[501];
long long ksm(long long a, long long b) {
  long long ans = 1;
  b %= 1000000007 - 1;
  while (b) {
    if (b & 1) ans *= a, ans %= 1000000007;
    b >>= 1;
    a *= a;
    a %= 1000000007;
  }
  return ans;
}
int main() {
  cin >> n >> m;
  f[0][0] = 1;
  m1[0] = 1;
  for (int i = 1; i <= n; i++) m1[i] = m1[i - 1] * 2, m1[i] %= 1000000007;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < i; j++) m2[j] = ksm(2, (long long)(i - j - 1) * j);
    for (int j = 1; j <= m; j++) {
      if (j > i) break;
      for (int k = j - 1; k < i; k++) {
        f[i][j] += f[k][j - 1] * m2[k] % 1000000007 * m1[k] % 1000000007 *
                   (m1[i - k] - 1) % 1000000007;
        f[i][j] %= 1000000007;
      }
    }
  }
  long long ans = 0;
  for (int i = m; i <= n; i++)
    ans += f[i][m] * ksm(2, (long long)i * (n - i)) % 1000000007,
        ans %= 1000000007;
  cout << ans << endl;
}
