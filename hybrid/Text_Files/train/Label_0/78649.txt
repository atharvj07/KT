def Composite(numList):
	aloneList = ['1', '4', '6', '8', '9']
	for num in numList:
		if num in aloneList:
			return (1, num)
	if numList in ['23', '37', '53', '73']:
		return 'any'
	if '2' in numList[1:]:
		return (2, numList[0] + '2')
	if '5' in numList[1:]:
		return (2, numList[0] + '5')
	if numList[0] == '2':
		if '7' in numList[1:]:
			return (2, '27')
		elif '3' in numList[1:]:
			return (2, '33')
	if numList[0] == '3':
		if '3' in numList[1:]:
			return (2, '33')
		elif '7' in numList[1:]:
			return (2, '77')
	if numList[0] == '5':
		if '7' in numList[1:]:
			return (2, '57')
		elif '3' in numList[1:]:
			return (2, '33')
	if numList[0] == '7':
		if '7' in numList[1:]:
			return (2, '77')
		elif '3' in numList[1:]:
			return (2, '33')
N = int(input())
for _ in range(N):
	n = input()
	numList = input()
	(l, ans) = Composite(numList)
	print(l)
	print(ans)
