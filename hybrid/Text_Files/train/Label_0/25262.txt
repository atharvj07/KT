#include <bits/stdc++.h>
using namespace std;
int a[4];
inline void add(int x) {
  a[x]++;
  a[(x + 1) % 4]++;
  printf("+%d\n", x + 1);
}
inline void div(int x) {
  a[x] /= 2;
  a[(x + 1) % 4] /= 2;
  printf("/%d\n", x + 1);
}
int main() {
  scanf("%d %d %d %d", &a[0], &a[1], &a[2], &a[3]);
  while (a[0] != 1 || a[1] != 1) {
    if (a[0] % 2) {
      if (a[1] % 2) {
        add(0);
      } else {
        add(3);
      }
    } else {
      if (a[1] % 2) {
        add(1);
      }
    }
    div(0);
  }
  while (a[2] != 1) {
    if (a[2] % 2) {
      add(1);
      div(1);
    } else {
      if (a[3] % 2) {
        add(1);
        add(2);
        div(1);
      } else {
        div(2);
      }
    }
  }
  while (a[3] != 1) {
    if (a[3] % 2) {
      add(2);
      div(2);
    } else {
      if (a[0] % 2) {
        add(2);
        add(3);
        div(2);
      } else {
        div(3);
      }
    }
  }
  return 0;
}
