import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.StringTokenizer;

/**
 *
 * @author yanef
 */
public class Testing {
    static class mas{
        int a,b,h;
        mas(int a,int b,int h){
            this.a=a;
            this.b=b;
            this.h=h;
        }
    }
    static class massort implements Comparator<mas>{

        @Override
        public int compare(mas o1, mas o2) {
            if (o1.b>o2.b) return -1;
            if (o1.b<o2.b) return 1;
            if (o1.a>o2.a) return -1;
            if (o1.a<o2.a) return 1;
            return 0;
        }
    }
    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) throws IOException {
        BufferedReader vod=new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(vod.readLine());
        StringTokenizer s;
        mas[] ar=new mas[n+1];
        int[] st=new int[n+1];
        long[] ans=new long[n+1];
        for (int i=1; i<=n; i++){
            s=new StringTokenizer(vod.readLine());
            int a=Integer.parseInt(s.nextToken());
            int b=Integer.parseInt(s.nextToken());
            int h=Integer.parseInt(s.nextToken());
            ar[i]=new mas(a,b,h);
        }
        Arrays.sort(ar,1,n+1,new massort());
        int last=0;
        for (int i=1; i<=n; i++) ans[i]=ar[i].h;
        for (int i=1; i<=n; i++){
            while(last>0&&ar[st[last]].a>=ar[i].b) last--;
            if (last!=0)
                ans[i]+=ans[st[last]];
            st[++last]=i;
        }
        long anser=0;
        for (int i=1; i<=n; i++){
            anser=Math.max(anser,ans[i]);
        }
        System.out.println(anser);
    }
}
