#include <bits/stdc++.h>
using namespace std;
long long n, m;
long long a[1010], pref[1010];
bool removed[1010];
long long mv[1010];
int main() {
  scanf(" %lld%lld", &n, &m);
  long long neg = 0;
  for (long long i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    if (a[i] < 0) neg++;
    removed[i] = false;
  }
  long long cmin = 0;
  pref[0] = 0;
  for (long long i = 1; i <= n; i++) {
    pref[i] = pref[i - 1] + a[i];
    cmin = min(cmin, pref[i]);
  }
  mv[0] = -cmin;
  for (long long i = 1; i <= neg; i++) {
    long long ind = -1;
    long long maxv = -6969696969420420;
    for (int j = 1; j <= n; j++) {
      if (removed[j]) continue;
      if (a[j] >= 0) continue;
      long long temp = a[j];
      a[j] = 0;
      long long cmin = 6969696969420420;
      for (int k = 1; k <= n; k++) {
        pref[k] = pref[k - 1] + a[k];
        cmin = min(cmin, pref[k]);
      }
      if (cmin > maxv) {
        maxv = cmin;
        ind = j;
      }
      a[j] = temp;
    }
    a[ind] = 0;
    mv[i] = -maxv;
    removed[ind] = true;
  }
  for (long long i = 0; i < m; i++) {
    long long v;
    scanf(" %lld", &v);
    long long lo = 0, hi = neg;
    while (lo != hi) {
      long long avg = (lo + hi) / 2;
      if (mv[avg] > v)
        lo = avg + 1;
      else
        hi = avg;
    }
    printf("%lld\n", lo);
  }
}
