from collections import Counter
n = int(input())
A = list(map(int, input().split()))
(left, right) = ({}, {})
dp = [0] * (n + 1)
cnt = Counter()
for (i, a) in enumerate(A):
	left.setdefault(a, i)
	right[a] = i
for i in range(n - 1, -1, -1):
	dp[i] = dp[i + 1]
	cnt[A[i]] += 1
	dp[i] = max(dp[i], dp[right[A[i]] + 1] + cnt[A[i]]) if i == left[A[i]] else max(dp[i], cnt[A[i]])
print(n - dp[0])
