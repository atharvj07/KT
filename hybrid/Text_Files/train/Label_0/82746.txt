from itertools import permutations as per
from math import factorial as fact
from difflib import SequenceMatcher
import bisect
t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	a = [int(i) for i in input().split()]
	a.sort()
	psum = [] + a
	psum[0] = 0
	for i in range(2, n):
		psum[i] += psum[i - 1]
	ans = 10 ** 18
	for i in range(n):
		x = max(0, a[0] - (k - psum[n - i - 1]) // (i + 1))
		ans = min(ans, x + i)
	print(ans)
