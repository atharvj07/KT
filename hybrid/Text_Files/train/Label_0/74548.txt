import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	ans = 0
	dif = 10 ** 10
	mark = None
	for (i, aa) in enumerate(a):
		if aa <= 0:
			ans += 1
		else:
			mark = aa
			break
		if i:
			dif = min(dif, a[i] - a[i - 1])
	if not mark is None:
		if dif >= mark:
			ans += 1
	print(ans)
