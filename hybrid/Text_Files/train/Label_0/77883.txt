import sys
input = sys.stdin.readline
t = int(input())
for you in range(t):
	l = input().split()
	n = int(l[0])
	c = int(l[1])
	l = input().split()
	li = [int(i) for i in l]
	b = input().split()
	bi = [int(i) for i in b]
	bi.append(0)
	days = 0
	extra = 0
	maxa = 10 ** 18
	for i in range(n):
		req = c - extra
		if req < 0:
			continue
		maxa = min(maxa, days + (req + li[i] - 1) // li[i])
		if bi[i] <= extra:
			nod = 1
		else:
			nod = 1 + (bi[i] - extra + li[i] - 1) // li[i]
		days += nod
		extra += (nod - 1) * li[i]
		extra -= bi[i]
	print(maxa)
