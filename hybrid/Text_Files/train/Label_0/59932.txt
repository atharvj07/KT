import java.util.*;
public class Main {
    public static int dist(int n, int a, int b) {
        if (a < b) {
            return b - a;
        }
        return n - (a - b);
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] min = new int[n];
        Arrays.fill(min, Integer.MAX_VALUE);
        int[] cnt = new int[n];
        for (int i = 0; i < m; i++) {
            int a = in.nextInt() - 1;
            int b = in.nextInt() - 1;
            cnt[a]++;
            if (min[a] == Integer.MAX_VALUE) {
                min[a] = b;
            } else if (dist(n, a, b) < dist(n, a, min[a])) {
                min[a] = b;
            }
        }
        int[] time = new int[n];
        for (int i = 0; i < n; i++) {
            time[i] = cnt[i] == 0 ? 0 : ((cnt[i] - 1) * n) + dist(n, i, min[i]);
        }
        for (int i = 0; i < n; i++) {
            int max = 0;
            int steps = 0;
            for (int j = i; j < i + n; j++) {
                int mod = j % n;
                if (time[mod] != 0) {
                    max = Math.max(max, time[mod] + steps);
                }
                steps++;
            }
            System.out.print(max + " ");
        }
        System.out.println();
    }
}
