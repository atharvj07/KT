#include <bits/stdc++.h>
long long n, k, m;
long long dp[100005][14][1 << 4];
signed main() {
  scanf("%lld %lld %lld", &n, &k, &m);
  dp[0][0][0] = 1;
  for (long long i = 0; i < n; i++)
    for (long long j = 0; j <= k; j++)
      for (long long s = 0; s < (1 << m); s++) {
        dp[i + 1][j][s >> 1] =
            (dp[i + 1][j][s >> 1] + dp[i][j][s]) % 1000000007;
        dp[i + 1][j + 1][s >> 1 | 1 << m - 1] =
            (dp[i + 1][j + 1][s >> 1 | 1 << m - 1] +
             dp[i][j][s] * (__builtin_popcount(s) + 1) % 1000000007) %
            1000000007;
      }
  long long ans = 0;
  for (long long i = 0; i < (1 << m); i++)
    ans = (ans + dp[n][k][i]) % 1000000007;
  printf("%lld\n", ans);
  return 0;
}
