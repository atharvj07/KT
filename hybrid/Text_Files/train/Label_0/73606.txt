from collections import defaultdict
d = defaultdict(lambda : 0)
cnt = 0
ans = 0
for _ in range(int(input())):
	(x, val) = list(map(str, input().split()))
	val = int(val)
	if x == '+':
		d[val] = 1
		cnt += 1
		ans = max(ans, cnt)
	else:
		if val not in d:
			ans += 1
		else:
			cnt -= 1
		ans = max(ans, cnt)
print(ans)
