a = input()
b = input()

def iguais(a, b):
	if a == b:
		return True
	if len(a) % 2 != 0:
		return False
	a1 = a[0:len(a) // 2]
	a2 = a[len(a) // 2:len(a)]
	b1 = b[0:len(b) // 2]
	b2 = b[len(b) // 2:len(b)]
	if a1 == b2 and a2 == b1 or (a1 == b1 and a2 == b2):
		return True
	else:
		return iguais(a1, b2) and iguais(a2, b1) or (iguais(a1, b1) and iguais(a2, b2))
if iguais(a, b):
	print('YES')
else:
	print('NO')
