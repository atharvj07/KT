k = int(input())
flag1 = 0
flag2 = 0
xrr = []
yrr = []
for i in range(k):
	(x, y) = map(int, input().split())
	xrr.append(x)
	yrr.append(y)
	if xrr[i] != yrr[i]:
		flag1 = 1
xr = xrr.copy()
yr = yrr.copy()
xr = sorted(xr, reverse=True)
if xr == xrr:
	flag2 = 1
if flag1 == 1:
	print('rated')
elif flag2 == 1:
	print('maybe')
else:
	print('unrated')
