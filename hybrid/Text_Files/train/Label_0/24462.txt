#include <bits/stdc++.h>
int main() {
  int a;
  scanf("%d", &a);
  int b, x = 0, ch = 0, nch = 0, ans = 0;
  while (x < a) {
    scanf("%d", &b);
    if (b % 2 == 0)
      ch += 1;
    else
      nch += 1;
    x++;
  }
  if (nch == 0) {
    printf("0");
    return 0;
  }
  while (ch > 0 && nch > 0) {
    ans += 1;
    ch -= 1;
    nch -= 1;
  }
  if (nch > 0) ans = ans + (nch - nch % 3) / 3;
  printf("%d", ans);
  return 0;
}
