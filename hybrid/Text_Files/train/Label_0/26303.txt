#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2005;
struct ii {
  int a, b;
};
struct is {
  int a;
  string b;
  bool operator>(is o) const {
    return make_tuple(-a, b) > make_tuple(-o.a, o.b);
  }
};
struct iii {
  int a;
  int c, d;
  bool operator<(iii o) const { return make_tuple(-a) > make_tuple(-o.a); }
};
int N, M;
char arr[MAXN][MAXN], ans[2 * MAXN];
int poin[MAXN][MAXN];
priority_queue<iii> pq;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> N >> M;
  for (int i = 0; i < N; i++) {
    cin >> arr[i];
  }
  poin[0][0] = M;
  if (arr[0][0] != 'a' && M > 0) {
    poin[0][0]--;
    ans[0] = 'a';
  } else {
    ans[0] = arr[0][0];
  }
  for (int i = 1; i < 2 * N - 1; i++) {
    char mnm = 'z';
    for (int j = max(0, i - N + 1); j <= min(i, N - 1); j++) {
      if ((j - 1 >= 0 && poin[j - 1][i - j] > 0) ||
          (i - j - 1 >= 0 && poin[j][i - j - 1] > 0)) {
        mnm = 'a';
      } else if ((j - 1 >= 0 && poin[j - 1][i - j] >= 0) ||
                 (i - j - 1 >= 0 && poin[j][i - j - 1] >= 0)) {
        mnm = min(mnm, arr[j][i - j]);
      }
    }
    for (int j = max(0, i - N + 1); j <= min(i, N - 1); j++) {
      poin[j][i - j] = -2 * MAXN;
      if (j - 1 >= 0) poin[j][i - j] = max(poin[j][i - j], poin[j - 1][i - j]);
      if (i - j - 1 >= 0)
        poin[j][i - j] = max(poin[j][i - j], poin[j][i - j - 1]);
      if (arr[j][i - j] != mnm) poin[j][i - j]--;
    }
    ans[i] = mnm;
  }
  ans[2 * N - 1] = '\0';
  cout << ans << "\n";
  return 0;
}
