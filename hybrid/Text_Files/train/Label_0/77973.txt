Testcases = int(input())
import math

def lcm(x, y):
	return x * y // math.gcd(x, y)
for test in range(Testcases):
	n = int(input())
	A = set(map(int, input().split()))
	ANS = 1
	fl = 1
	for a in A:
		ANS = lcm(a, ANS)
		if ANS > 10 ** 12:
			print(-1)
			fl = 0
			break
	if fl == 0:
		continue
	if ANS in A:
		ANS *= min(A)
	x = ANS
	L = int(math.sqrt(x))
	FACT = dict()
	for i in range(2, L + 2):
		while x % i == 0:
			FACT[i] = FACT.get(i, 0) + 1
			x = x // i
	if x != 1:
		FACT[x] = FACT.get(x, 0) + 1
	PRIMESCORE = 1
	for f in FACT:
		PRIMESCORE *= FACT[f] + 1
	if n + 2 == PRIMESCORE:
		print(ANS)
	else:
		print(-1)
