#include <bits/stdc++.h>
using namespace std;

int main(){
  int n;
  int tmp = 2;
  queue<int> que;

  cin >> n;
  int ans = n;

  while(n > 1 && tmp*tmp <= n){
    if(n % tmp == 0){
      que.push(tmp);
      n /= tmp;
    }else{
      tmp++;
    }
  }

  if(n > 1){
    que.push(n);
  }

  cout << ans << ":";
  while(!que.empty()){
    cout << " " << que.front();
    que.pop();
  }

  cout << endl;
  return 0;

}
