t = int(input())
for _ in range(t):
	(n, x) = map(int, input().split())
	a = [int(x) for x in input().split()]
	odd = 0
	even = 0
	for i in a:
		if i & 1:
			odd += 1
		else:
			even += 1
	i = 1
	ok = 0
	if even == 0:
		if x & 1:
			print('YES')
		else:
			print('NO')
		continue
	while True:
		if i > odd:
			break
		if i + even >= x:
			ok = 1
			break
		i += 2
	if ok:
		print('YES')
	else:
		print('NO')
