n = int(input())
(a, b) = ([], [])
for i in range(n):
	x = int(input())
	if x > 0:
		a.append(x)
	else:
		b.append(-x)
sum1 = sum(a)
sum2 = sum(b)
if sum1 < sum2:
	print('second')
elif sum1 > sum2:
	print('first')
elif a < b:
	print('second')
elif b < a:
	print('first')
elif x > 0:
	print('first')
else:
	print('second')
