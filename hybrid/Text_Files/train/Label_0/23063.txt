#include <bits/stdc++.h>
using namespace std;
vector<int> adj[200005];
int head, tail;
int prnt[200005], deg[200005];
bool ok = true;
void dfs(int u, int pre) {
  if (pre != 0) {
    deg[u] = 1;
    deg[pre]++;
  }
  for (int i = 0; i < adj[u].size(); i++) {
    int v = adj[u][i];
    if (v != pre) dfs(v, u);
  }
  if (deg[u] & 1) {
    prnt[tail--] = u;
    if (pre == 0) {
      ok = false;
    }
  } else {
    prnt[head++] = u;
    deg[pre]--;
  }
}
int main() {
  int n;
  scanf("%d", &n);
  int root;
  for (int i = 1; i <= n; i++) {
    int v;
    scanf("%d", &v);
    if (v != 0)
      adj[v].push_back(i);
    else
      root = i;
  }
  memset(deg, 0, sizeof deg);
  head = 1, tail = n;
  dfs(root, 0);
  if (ok) {
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) cout << prnt[i] << endl;
  } else {
    cout << "NO" << endl;
  }
}
