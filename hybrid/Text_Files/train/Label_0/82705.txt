raw = input()
(n, m) = map(int, raw.split(' '))
mx = []
for ri in range(n):
	mx += [input()]
if n >= 4:
	print(-1)
if n == 1:
	print(0)
if n == 2:
	p = m // 2
	s = 'AB' * (p + 2)
	s1 = s[:m]
	s2 = s[1:m + 1]
	(r1, r2) = (0, 0)
	for i in range(m):
		o1 = int(mx[0][i] != mx[1][i])
		letter = 'AB'[o1]
		if letter != s1[i]:
			r1 += 1
		if letter != s2[i]:
			r2 += 1
	print(min(r1, r2))
if n == 3:
	p = m // 2
	s = 'AD' * (p + 2)
	s1 = s[:m]
	s2 = s[1:m + 1]
	s = 'BC' * (p + 2)
	s3 = s[:m]
	s4 = s[1:m + 1]
	(r1, r2, r3, r4) = (0, 0, 0, 0)
	for i in range(m):
		o1 = int(mx[0][i] != mx[1][i])
		o2 = int(mx[1][i] != mx[2][i])
		letter = 'ABCD'[o1 * 2 + o2]
		if letter != s1[i]:
			r1 += 1
		if letter != s2[i]:
			r2 += 1
		if letter != s3[i]:
			r3 += 1
		if letter != s4[i]:
			r4 += 1
	rez = min(min(r1, r2), min(r3, r4))
	print(rez)
