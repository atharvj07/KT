a = [int(s) for s in sys.stdin.readline().split()]
st = -1
ed = -1
leading_zeros = 0
seg_count = 1
for (i, v) in enumerate(a):
	if v == -1:
		if st == -1:
			leading_zeros += 1
		elif ed != -1:
			if a[ed] + (i - ed) * (a[ed] - a[st]) / (ed - st) <= 0:
				st = -1
				ed = -1
				leading_zeros = 1
				seg_count += 1
			else:
				pass
		else:
			pass
	elif st == -1:
		st = i
	elif ed == -1:
		ed = i
		if (v - a[st]) % (i - st) != 0 or a[st] - (v - a[st]) / (i - st) * leading_zeros <= 0:
			st = i
			ed = -1
			seg_count += 1
			leading_zeros = 0
		else:
			ed = i
	elif (v - a[ed]) % (i - ed) != 0 or (v - a[ed]) * (ed - st) != (a[ed] - a[st]) * (i - ed):
		st = i
		ed = -1
		seg_count += 1
		leading_zeros = 0
	else:
		ed = i
print(seg_count)
