#include <bits/stdc++.h>
using namespace std;
struct Level {
  int one, two, id;
  pair<int, int> getOneKey() { return make_pair(one, 2 * id); }
  pair<int, int> getTwoKey() { return make_pair(two - one, 2 * id + 1); }
  bool operator<(const Level &a) const { return two < a.two; }
};
class FenwickTree {
  vector<pair<int, int> > keys;
  vector<long long> sum;
  vector<int> cnt;
  int n;
  void update(int x, int delta) {
    for (int i = x; i < n; i |= i + 1) {
      sum[i] += delta * keys[x].first;
      cnt[i] += delta;
    }
  }
  pair<long long, int> get(int x) {
    pair<long long, int> res;
    for (int i = x; i >= 0; i -= ~i & (i + 1)) {
      res.first += sum[i];
      res.second += cnt[i];
    }
    return res;
  }

 public:
  FenwickTree(vector<pair<int, int> > key) {
    sort(key.begin(), key.end());
    n = key.size();
    sum.assign(n, 0);
    cnt.assign(n, 0);
    keys = key;
  }
  void update(const pair<int, int> &key, int delta) {
    int pos = lower_bound(keys.begin(), keys.end(), key) - keys.begin();
    assert(keys.at(pos) == key);
    update(pos, delta);
  }
  long long minElementSum(int m) {
    if (m <= 0) return 0;
    if (m > get(n - 1).second) return (long long)1e18;
    int low = 0, high = n - 1;
    while (low < high) {
      int mid = (low + high) / 2;
      if (get(mid).second >= m)
        high = mid;
      else
        low = mid + 1;
    }
    return get(low).first;
  }
};
const int N = (int)3e5, INF = (int)2e9;
Level level[N];
int n, w;
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> w;
  for (int i = 0; i < n; ++i) {
    cin >> level[i].one >> level[i].two;
    level[i].id = i;
  }
  sort(level, level + n);
  vector<pair<int, int> > value;
  for (int i = 0; i < n; ++i) {
    value.push_back(level[i].getOneKey());
    value.push_back(level[i].getTwoKey());
  }
  FenwickTree tree(value);
  for (int i = 0; i < n; ++i) tree.update(level[i].getOneKey(), 1);
  long long bestCost = (long long)1e18;
  long long firstCost = 0;
  int bestPrefix = -1;
  for (int prefix = 0; prefix <= n; ++prefix) {
    long long secondCost = tree.minElementSum(w - prefix);
    if (firstCost + secondCost < bestCost) {
      bestCost = firstCost + secondCost;
      bestPrefix = prefix;
    }
    if (prefix < n) {
      tree.update(level[prefix].getOneKey(), -1);
      tree.update(level[prefix].getTwoKey(), 1);
      firstCost += level[prefix].one;
    }
  }
  string res(n, '0');
  for (int i = 0; i < bestPrefix; ++i) res[level[i].id] = '1';
  priority_queue<pair<int, int>, vector<pair<int, int> >,
                 greater<pair<int, int> > >
      q;
  for (int i = 0; i < bestPrefix; ++i) q.push(level[i].getTwoKey());
  for (int i = bestPrefix; i < n; ++i) q.push(level[i].getOneKey());
  for (int i = 0; i < w - bestPrefix; ++i) {
    ++res[q.top().second / 2];
    q.pop();
  }
  cout << bestCost << '\n';
  cout << res << '\n';
  return 0;
}
