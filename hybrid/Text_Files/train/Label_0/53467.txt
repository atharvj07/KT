import java.io.*;
import java.lang.reflect.Array;
import java.util.*;

public class Main {

	private static final int[] dx = { -1, 0, 0, 1 };
	private static final int[] dy = { 0, -1, 1, 0 };

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();
		int k = in.nextInt();
		char[][] map = new char[n][m];
		char[][] scanMap = new char[n][m];
		List<Point> pointList = new ArrayList<>();
		for (int i = 0; i < n; i++) {
			String temp = in.next();
			for (int j = 0; j < m; j++) {
				map[i][j] = temp.charAt(j);
				scanMap[i][j] = temp.charAt(j);
				if (temp.charAt(j) == '.') {
					pointList.add(new Point(i, j));
				}
			}
		}
		
		int totalLake = 0;
		List<List<Point>> lakeList = new ArrayList<>();
		for (Point point : pointList) {
			if (scanMap[point.x][point.y] == '.') {
				count = 0;
				isLake = true;
				lakePoints = new ArrayList<>();
				scanMap(point, scanMap, n, m);
				if (isLake) lakeList.add(lakePoints);
			}
		}
		
		lakeList.sort(new Comparator<List<Point>>() {

			@Override
			public int compare(List<Point> o1, List<Point> o2) {
				// TODO Auto-generated method stub
				return o1.size() - o2.size();
			}
		});
		
		lakeList.size();
		count = 0;
		for (int i = 0; i < lakeList.size() - k; i++) {
			lakePoints = lakeList.get(i);
			count += lakePoints.size();
			for (Point point : lakePoints) {
				map[point.x][point.y] = '*';
			}
		}
		
		System.out.println(count);
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				System.out.print(map[i][j]);
			}
			System.out.println();
		}
		
		in.close();
	}
	
	private static class Point {
		 public int x,y;
		 public Point(int x, int y) {
		 this.x = x;
		 this.y = y;
		 }
	}
	
	static List<Point> lakePoints;
	static boolean isLake;
	private static void scanMap(Point point, char[][] scanMap, int n, int m) {
		if (point.x == 0 || point.y == 0 || point.x == n - 1 || point.y == m - 1) {
			isLake = false;
		}
		scanMap[point.x][point.y] = '*';
		lakePoints.add(point);
		count++;
		for (int i = 0; i < dx.length; i++) {
			int x = point.x + dx[i];
			int y = point.y + dy[i];
			if (x >= 0 && y >= 0 && x < n && y < m && scanMap[x][y] == '.') {
				scanMap(new Point(x, y), scanMap, n, m);
			}
		}
	}

	static int count;
}
