import java.util.HashMap;
import java.util.Scanner;

public class A {
	private final static Scanner reader = new Scanner(System.in);

	public static int maxNumberCol(char[][] tab, int ligne) {
		int r = tab.length;
		int c = tab[0].length;
		int compt = 0;

		for (int j = 0; j < c; j++) {
			boolean test = true;

			for (int i = 0; i < r; i++) {
				if (tab[i][j] == 'S') {
					test = false;
				}
			}
			if (test) {
				compt++;
			}
		}
		return compt * r - compt * ligne;
	}

	public static void main(String[] args) {
		int r = reader.nextInt();
		int c = reader.nextInt();
		char[][] tab = new char[r][c];
		int compt = 0;
		reader.nextLine();

		for (int i = 0; i < r; i++) {
			boolean test = true;
			String line = reader.nextLine();

			for (int j = 0; j < c; j++) {
				tab[i][j] = line.charAt(j);
				if (tab[i][j] == 'S') {
					test = false;
				}
			}
			if (test) {
				compt++;
			}
		}
		int number = maxNumberCol(tab, compt) + compt * c;
		System.out.println(number);
	}
}
