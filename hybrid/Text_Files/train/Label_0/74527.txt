(n, k) = map(int, input().split())
t = input()
(left, right) = (0, 0)
s = ''
for i in range(1, n):
	if t[i] == t[left]:
		for right in range(i, n):
			if t[right] != t[left]:
				left = 0
				break
			left += 1
		if left != 0 and right == n - 1:
			s = t[left:]
			break
if s == '':
	s = t
for i in range(k - 1):
	t += s
print(t)
