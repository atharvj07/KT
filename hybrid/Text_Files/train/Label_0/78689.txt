def small_idx(v):
	sml = v[0]
	sml_idx = 0
	for (i, e) in enumerate(v):
		if e < sml:
			sml = e
			sml_idx = i
	return (sml, sml_idx)

def lrg_idx(v):
	lrg = v[0]
	lrg_idx = 0
	for (i, e) in enumerate(v):
		if e > lrg:
			lrg = e
			lrg_idx = i
	return (lrg, lrg_idx)
t = int(input())
while t > 0:
	(n, k) = [int(x) for x in input().split()]
	a = [int(x) for x in input().split()]
	b = [int(x) for x in input().split()]
	while k > 0:
		(a_s, a_s_i) = small_idx(a)
		(b_l, b_l_i) = lrg_idx(b)
		if a_s > b_l:
			break
		(a[a_s_i], b[b_l_i]) = (b_l, a_s)
		k -= 1
	print(sum(a))
	t -= 1
