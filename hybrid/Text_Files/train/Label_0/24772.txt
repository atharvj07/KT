#include <bits/stdc++.h>
using namespace std;

struct S {
  int p, w, left, right, f;
  double h;
};


const int INF = 1<<28;
const double inf = 1e10;
const double eps = 1e-8;
const int MAXN = 11;
const int MAXL = 11;

int N;
int B[MAXN], H[MAXN];
int L;
int P[MAXL], T[MAXL], id[MAXL];

bool equals(double a, double b) {
  return abs(a - b) < eps;
}

double getT(double h, double w, double f) {
  return 30.0 * h * w / f;
}

double getH(double t, double w, double f) {
  return f*t / (30.0*w);
}

int main() {
  int D; cin >> D;
  while(D--) {
    cin >> N;
    for(int i = 0; i < N; ++i) cin >> B[i] >> H[i];
    vector<S> s(N+1, (S){0, 0, INF, INF, 0, 0});
    for(int i = 0; i < N; ++i) {
      s[i].p = i == 0 ? 0 : B[i-1];
      s[i].w = i == 0 ? B[i] : B[i] - B[i-1];
      s[i].right = s[i+1].left = H[i];
    }
    s[N].p = B[N-1];
    s[N].w = 100 - B[N-1];
    ++N;
    int M; cin >> M;
    while(M--) {
      int F, A; cin >> F >> A;
      for(int i = 1; i <= N; ++i) {
        if(i == N || F < s[i].p) {
          s[i-1].f += A;
          break;
        }
      }
    }
    int L; cin >> L;
    {
      vector<pair<pair<int, int>, int> > v(L);
      for(int i = 0; i < L; ++i) {
        cin >> v[i].first.second >> v[i].first.first;
        v[i].second = i;
      }
      sort(v.begin(), v.end());
      for(int i = 0; i < L; ++i) {
        T[i] = v[i].first.first;
        P[i] = v[i].first.second;
        id[i] = v[i].second;
      }
    }
    double now = 0;
    vector<double> ans(L);
    for(int k = 0; k < L; ) {
      double t = T[k] - now;
      for(int i = 0; i < N; ++i) {
        t = min(t, getT(min(s[i].left, s[i].right) - s[i].h, s[i].w, s[i].f));
      }
      now += t;
      for(int i = 0; i < N; ++i) {
        s[i].h += getH(t, s[i].w, s[i].f);
        if(equals(s[i].h, s[i].left)) s[i].h = s[i].left;
        if(equals(s[i].h, s[i].right)) s[i].h = s[i].right;
      }
      for(int i = 0; i+1 < N; ++i) {
        if(equals(s[i].h, s[i].right) && equals(s[i].h, s[i+1].h)) {
          s[i].right = s[i+1].right;
          s[i].w += s[i+1].w;
          s[i].f += s[i+1].f;
          s.erase(s.begin() + i + 1);
          --N;
        }
      }
      for(int i = 0; i < N; ++i) {
        if(equals(s[i].h, s[i].left)) {
          s[i-1].f += s[i].f;
          s[i].f = 0;
        } else if(equals(s[i].h, s[i].right)) {
          s[i+1].f += s[i].f;
          s[i].f = 0;
        }
      }
      if(equals(now, T[k])) {
        for(int i = 1; i <= N; ++i) {
          if(i == N || P[k] < s[i].p) {
            ans[id[k]] = min(50.0, s[i-1].h);
            break;
          }
        }
        ++k;
      }
    }
    for(int i = 0; i < L; ++i) {
      printf("%.10f\n", ans[i]);
    }
  }
  return 0;
}
