from sys import stdin
for _ in range(int(stdin.readline())):
	n = int(stdin.readline())
	s = stdin.readline().strip()
	check = ''
	flag = True
	for i in range(n - 1):
		if s[i] in check or s[i + 1] in check:
			flag = False
			break
		if s[i] != s[i + 1]:
			check += s[i]
	if flag == False:
		print('NO')
	else:
		print('YES')
