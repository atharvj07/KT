def load_data():
	(n, m) = list(map(int, input().split()))
	matrix = []
	for i in range(n):
		line = list(input())
		matrix.append(line)
	return (n, m, matrix)

def find_first_occurrence(M, mode):
	length = len(M)
	index_list = list(range(len(M)))
	if mode < 0:
		index_list = index_list[::-1]
	for i in index_list:
		if 'B' in M[i]:
			return i
	return -1

def main():
	(n, m, matrix) = load_data()
	(top, btm, left, right) = [0] * 4
	matrix_t = list(zip(*matrix))
	top = find_first_occurrence(matrix, 1)
	if top == -1:
		print('1')
		return
	btm = find_first_occurrence(matrix, -1)
	left = find_first_occurrence(matrix_t, 1)
	right = find_first_occurrence(matrix_t, -1)
	width = abs(right - left) + 1
	height = abs(top - btm) + 1
	if width > n or height > m:
		print('-1')
		return
	side = max(width, height)
	if width >= height:
		target = [line[left:right + 1] for line in matrix]
		(up, down) = (top, btm)
	else:
		target = matrix[top:btm + 1]
		target = list(zip(*target))
		(up, down) = (left, right)
	res = float('inf')
	for i in range(len(target) - side + 1):
		temp = list(range(i, i + side))
		if up in temp and down in temp:
			count = sum([target[j].count('W') for j in temp])
			res = min(res, count)
	print(res)
main()
