(n, k) = map(int, input().split())
v = list(map(int, input().split()))
(pos, pref) = ([-1], [])
ans = 0
for i in range(n):
	if v[i] != 1:
		pos.append(i)
	if v[i] == 1 and k == 1:
		ans += 1
	if i:
		pref.append(pref[-1] + v[i])
	else:
		pref.append(v[i])
pos.append(n)
m = len(pos)
inf = int(2e+18 + 10)
for i in range(1, m - 1):
	p = 1
	for j in range(i, m - 1):
		p = p * v[pos[j]]
		if p > inf:
			break
		s = pref[pos[j]]
		if pos[i]:
			s -= pref[pos[i] - 1]
		if s * k == p:
			ans += 1
		d = p - s * k
		if d > 0 and d % k == 0:
			w = d // k
			f = min(pos[i] - pos[i - 1] - 1, w)
			b = min(pos[j + 1] - pos[j] - 1, w)
			if f + b < w:
				continue
			ans += f + b - w + 1
print(ans)
