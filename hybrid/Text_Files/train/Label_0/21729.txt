#include <bits/stdc++.h>
using namespace std;
int n, l, k;
double pr[205];
int a[205];
double mem[205][205][405];
const int sh = 200;
double sol(int i, int j, int f) {
  if (i == n) {
    if (f >= 0 && j >= l) {
      return 1;
    } else {
      return 0;
    }
  }
  double &ret = mem[i][j][f + sh];
  if (ret == ret) {
    return ret;
  }
  double p1 = sol(i + 1, j + 1, min(f + a[i], 200)) * pr[i];
  double p2 = sol(i + 1, j, f) * (1 - pr[i]);
  return ret = p1 + p2;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> l >> k;
  memset(mem, -1.0, sizeof mem);
  for (int i = 0; i < n; i++) {
    cin >> pr[i];
    pr[i] /= 100.0;
  }
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  cout << fixed << setprecision(12) << sol(0, 0, k);
  return 0;
}
