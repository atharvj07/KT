#include <bits/stdc++.h>
using namespace std;
const int N = 250005, mod = 998244353;
int n, bin[N << 2], len;
long long m, a[N << 2], b[N << 2];
long long flv[N], inv[N], f[N];
int getint() {
  char ch;
  while (!isdigit(ch = getchar()))
    ;
  int x = ch - 48;
  while (isdigit(ch = getchar())) x = x * 10 + ch - 48;
  return x;
}
long long getLL() {
  char ch;
  while (!isdigit(ch = getchar()))
    ;
  long long x = ch - 48;
  while (isdigit(ch = getchar())) x = x * 10 + ch - 48;
  return x;
}
long long getmi(long long a, long long x) {
  long long rt = 1;
  while (x) {
    if (x & 1) rt = rt * a % mod;
    a = a * a % mod, x >>= 1;
  }
  return rt;
}
void FFT(long long a[], int len, int tp) {
  for (int i = 0; i <= len - 1; ++i)
    bin[i] = bin[i >> 1] >> 1 | ((i & 1) * (len >> 1));
  for (int i = 0; i <= len - 1; ++i)
    if (i < bin[i]) swap(a[i], a[bin[i]]);
  for (int i = 1; i < len; i <<= 1) {
    long long wn = getmi(3, (mod - 1) / (i << 1));
    if (tp == -1) wn = getmi(wn, mod - 2);
    for (int j = 0; j < len; j += i << 1) {
      long long w = 1, x, y;
      for (int k = 0; k <= i - 1; ++k) {
        x = a[j + k], y = a[i + j + k] * w, w = w * wn % mod;
        a[j + k] = (x + y) % mod, a[i + j + k] = (x - y) % mod;
      }
    }
  }
  if (tp == -1) {
    long long x = getmi(len, mod - 2);
    for (int i = 0; i <= len - 1; ++i) a[i] = a[i] * x % mod;
  }
}
int main() {
  n = getint(), m = getLL();
  flv[0] = 1;
  for (int i = 1; i <= n; ++i) flv[i] = flv[i - 1] * i % mod;
  inv[n] = getmi(flv[n], mod - 2);
  for (int i = n; i >= 1; --i) inv[i - 1] = inv[i] * i % mod;
  for (len = 1; len <= n + n; len <<= 1)
    ;
  for (int i = 0; i <= n; ++i)
    a[i] = getint() * flv[i] % mod, b[n - i] = inv[i];
  FFT(a, len, 1), FFT(b, len, 1);
  for (int i = 0; i <= len - 1; ++i) a[i] = a[i] * b[i] % mod;
  FFT(a, len, -1);
  for (int i = 0; i <= n; ++i)
    f[i] = a[i + n] * getmi(getmi(i + 1, mod - 2), m) % mod;
  memset(a, 0, sizeof(a));
  memset(b, 0, sizeof(b));
  int tp = 1;
  for (int i = 0; i <= n; ++i) a[i] = f[i], b[n - i] = inv[i] * tp, tp *= -1;
  FFT(a, len, 1), FFT(b, len, 1);
  for (int i = 0; i <= len - 1; ++i) a[i] = a[i] * b[i] % mod;
  FFT(a, len, -1);
  for (int i = 0; i <= n; ++i)
    printf("%lld ", (a[i + n] * inv[i] % mod + mod) % mod);
  puts("");
  return 0;
}
