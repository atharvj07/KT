n = int(input())
a = [int(x) for x in input().split()]
count_of_odds = 0
min_odd_value = 0
min_odd_value_is_set = False
summa = 0
for x in a:
	summa += x
	if x % 2 == 1:
		count_of_odds += 1
		if not min_odd_value_is_set or min_odd_value > x:
			min_odd_value = x
			min_odd_value_is_set = True
if count_of_odds % 2 == 1:
	summa -= min_odd_value
print(summa)
