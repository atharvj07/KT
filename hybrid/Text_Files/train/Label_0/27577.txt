#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, k, head[N], e[N << 1][2], tot, cnt[N];
long long ans;
void addedge(int u, int v) {
  e[tot][0] = v;
  e[tot][1] = head[u];
  head[u] = tot++;
}
void dfs1(int u, int fa) {
  for (int i = head[u]; i != -1; i = e[i][1]) {
    int v = e[i][0];
    if (v != fa) {
      dfs1(v, u);
      cnt[u] += cnt[v];
    }
  }
}
void dfs2(int u, int fa) {
  for (int i = head[u]; i != -1; i = e[i][1]) {
    int v = e[i][0];
    if (v != fa) {
      dfs2(v, u);
      ans += min(cnt[v], cnt[1] - cnt[v]);
    }
  }
}
int main() {
  while (scanf("%d%d", &n, &k) == 2) {
    memset(head, -1, sizeof(head));
    tot = 0;
    memset(cnt, 0, sizeof(cnt));
    for (int i = 0, x; i < 2 * k; i++) {
      scanf("%d", &x);
      cnt[x] = 1;
    }
    for (int i = 1, a, b; i < n; i++) {
      scanf("%d%d", &a, &b);
      addedge(a, b);
      addedge(b, a);
    }
    ans = 0;
    dfs1(1, 0);
    dfs2(1, 0);
    printf("%I64d\n", ans);
  }
  return 0;
}
