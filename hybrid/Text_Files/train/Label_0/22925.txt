#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, ans = 0;
  cin >> n;
  if (n % 2 == 0) {
    vector<int> tab;
    tab.push_back(1);
    tab.push_back(1);
    tab.push_back(n / 2 - 1);
    tab.push_back(n / 2 - 1);
    while (tab[0] <= n / 4) {
      if (tab[0] != tab[2]) ans++;
      tab[0]++;
      tab[1]++;
      tab[2]--;
      tab[3]--;
    }
  }
  cout << ans << endl;
  return 0;
}
