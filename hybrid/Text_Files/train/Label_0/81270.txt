def main():
	students = int(input())
	msgs = [int(x) for x in input().split()]
	if msgs[0] == 0:
		print('-1')
	elif students - 1 > sum(msgs):
		print('-1')
	else:
		result = []
		no_info = [x for x in range(1, students)]
		sort_dict = {str(x): msgs[x] for x in no_info}
		sort_dict = sorted(sort_dict, key=sort_dict.__getitem__, reverse=True)
		no_info = [int(x) for x in sort_dict]
		has_info = [0]
		for target in no_info:
			if msgs[0] == 0:
				break
			if msgs[target] == 0:
				continue
			else:
				has_info.append(target)
				no_info.remove(target)
				result.append('1 ' + str(target + 1))
				msgs[0] -= 1
		check = True
		while check:
			check = False
			for item in has_info:
				if msgs[item] > 0:
					for target in no_info:
						if msgs[item] == 0:
							break
						has_info.append(target)
						no_info.remove(target)
						result.append(str(item + 1) + ' ' + str(target + 1))
						msgs[item] -= 1
						check = True
		if len(no_info) > 0:
			print('-1')
		else:
			print(str(len(result)))
			for item in result:
				print(item)
main()
