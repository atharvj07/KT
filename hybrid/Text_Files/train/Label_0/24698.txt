#include<iostream>
#include<algorithm>
using namespace std;
string S,T;
long A,B,C,D;
long dp[101][101];
int main()
{
	while(cin>>S,S!="#")
	{
		cin>>T>>A>>B>>C>>D;
		long ans=9e18;
		for(int i=0;i<T.size();i++)
		{
			string t=T.substr(T.size()-i)+T.substr(0,T.size()-i);
			for(int j=0;j<=S.size();j++)for(int k=0;k<=t.size();k++)dp[j][k]=1e18;
			dp[0][0]=0;
			for(int j=0;j<=S.size();j++)for(int k=0;k<=t.size();k++)
			{
				if(j<S.size())dp[j+1][k]=min(dp[j+1][k],dp[j][k]+B);
				if(k<t.size())dp[j][k+1]=min(dp[j][k+1],dp[j][k]+(k<i?A-D:A));
				if(j<S.size()&&k<t.size())
					dp[j+1][k+1]=min(dp[j+1][k+1],dp[j][k]+(S[j]==t[k]?0:C));
			}
			ans=min(ans,dp[S.size()][t.size()]+i*D);
		}
		cout<<ans<<endl;
	}
}
