def main():
	(n, sx, sy) = map(int, input().split())
	(t, b, l, r) = (0, 0, 0, 0)
	for i in range(n):
		(x, y) = map(int, input().split())
		if x < sx:
			l += 1
		if x > sx:
			r += 1
		if y < sy:
			b += 1
		if y > sy:
			t += 1
	a = [(l, (sx - 1, sy)), (r, (sx + 1, sy)), (b, (sx, sy - 1)), (t, (sx, sy + 1))]
	ans = max(a)
	print(ans[0])
	print(*ans[1])
main()
