const int INF = 1e9;
long long por(long long a, int b) {
  long long res = 1, M = 998244353;
  while (b) {
    if (b & 1) res = (res * a) % M;
    a = (a * a) % M;
    b /= 2;
  }
  return res;
}
void solve() {
  int n;
  cin >> n;
  long long M = 998244353;
  vector<int> dp(n + 1, 0);
  dp[0] = 1;
  for (int i = 1; i <= n; i++) dp[i] = (dp[i - 1] * 10ll) % M;
  for (int i = 1; i < n; i++) {
    long long ans = 81ll * (n - i - 1) * dp[n - i - 1] % M;
    ans = (ans + 18ll * dp[n - i]) % M;
    cout << ans << ' ';
  }
  cout << 10;
  return;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
