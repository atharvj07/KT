from collections import Counter
from collections import defaultdict
from collections import deque
import math
import heapq
import sys
input = sys.stdin.readline
import bisect
rs = lambda : input().strip()
ri = lambda : int(input())
rl = lambda : list(map(int, input().split()))
rls = lambda : list(map(str, input().split()))

def res(a, b):
	if (a + b) % 2 == 1:
		return (a + b - 1) // 2
	else:
		return (a + b) // 2
t = int(input())
for _ in range(0, t):
	n = int(input())
	q = []
	ans = [0] * (n + 1)
	heapq.heapify(q)
	x = res(1, n)
	heapq.heappush(q, [-(n - 1), 1, n])
	c = 1
	while len(q) != 0:
		a = heapq.heappop(q)
		x = res(a[1], a[2])
		ans[x] = c
		c = c + 1
		if a[1] != a[2] and a[1] <= x - 1:
			heapq.heappush(q, [-(x - 1 - a[1]), a[1], x - 1])
		if a[1] != a[2] and a[2] >= x + 1:
			heapq.heappush(q, [-(a[2] - x - 1), x + 1, a[2]])
	print(*ans[1:])
