#include<bits/stdc++.h>
using namespace std;
long long read()
{
	char ch=getchar();long long x=0,ff=1;
	while(ch<'0'||ch>'9') {if(ch=='-') ff=-ff;ch=getchar();}
	while(ch>='0'&&ch<='9') x=x*10+ch-'0',ch=getchar();
	return x*ff;
}
void write(long long aa)
{
	if(aa<0) putchar('-'),aa=-aa;
	if(aa>9) write(aa/10);
	putchar(aa%10+'0');
	return;
}
const long long mod=924844033;
long long n,m;
long long tot,head[200005],nx[400005],to[400005];
long long fac[200005],ny[200005];
long long jl[200005],sz[200005];
long long bh[1400005],A[1400005],B[1400005];
void jia(long long aa,long long bb)
{
	tot++;
	nx[tot]=head[aa];
	to[tot]=bb;
	head[aa]=tot;
	return;
}
long long ksm(long long aa,long long bb)
{
	long long now=1;
	while(bb)
	{
		if(bb&1) now=now*aa%mod;
		bb>>=1;aa=aa*aa%mod;
	}
	return now;
}
void dfs(long long rt,long long fa)
{
	sz[rt]=1;
	for(long long i=head[rt];i;i=nx[i])
	{
		long long yy=to[i];
		if(yy==fa) continue;
		dfs(yy,rt);sz[rt]+=sz[yy];
	}
	if(fa) jl[sz[rt]]++,jl[n-sz[rt]]++;
	return;
}
void ntt(long long* a,long long ff)
{
	for(long long i=0;i<m;++i) if(bh[i]>i) swap(a[i],a[bh[i]]);
	for(long long i=1;i<m;i<<=1)
	{
		long long yuan=ksm(5,(mod-1)/(i*2));
		if(ff==-1) yuan=ksm(yuan,mod-2);
		for(long long j=0;j<m;j+=(i<<1))
		for(long long l=0,now=1;l<i;++l,now=now*yuan%mod)
		{
			long long aa=a[j+l],bb=now*a[j+i+l]%mod;
			a[j+l]=(aa+bb)%mod;a[j+i+l]=(aa-bb+mod)%mod;
		}
	}
	return;
}
long long C(long long aa,long long bb) 
{
	if(bb<aa) return 0;
	return fac[bb]*ny[aa]%mod*ny[bb-aa]%mod;
}
int main()
{
	n=read();
	for(long long i=1;i<n;++i)
	{
		long long x=read(),y=read();
		jia(x,y);jia(y,x);
	}
	dfs(1,0);fac[0]=1;
	for(long long i=1;i<=n;++i) fac[i]=fac[i-1]*i%mod;
	ny[n]=ksm(fac[n],mod-2);
	for(long long i=n;i;i--) ny[i-1]=ny[i]*i%mod;
	m=1;while(m<=n*2) m<<=1;
	for(long long i=1;i<m;++i) bh[i]=(bh[i>>1]>>1)|(i&1?(m>>1):0);
	for(long long i=0;i<=n;++i) A[i]=jl[i]*fac[i]%mod;
	for(long long i=0;i<=n;++i) B[i]=ny[n-i];
	ntt(A,1);ntt(B,1);
	for(long long i=0;i<m;++i) A[i]=A[i]*B[i]%mod;
	ntt(A,-1);long long NY=ksm(m,mod-2);
	for(long long i=1;i<=n;++i) write((n*C(i,n)%mod-A[i+n]*NY%mod*ny[i]%mod+mod)%mod),puts("");
	return 0;
}
