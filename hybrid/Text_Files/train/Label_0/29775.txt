#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, ans;
  cin >> s;
  long long q = s.length() / 4, n = s.length();
  for (long long i = 0; i < q; i++) {
    long long a = i * 2, b = a + 1, c = n - 1 - a, d = c - 1;
    if (s[a] == s[d] || s[a] == s[c]) {
      ans += s[a];
    } else {
      ans += s[b];
    }
  }
  if (n % 4) {
    ans += s[2 * q];
  }
  for (long long i = q - 1; i >= 0; --i) {
    ans += ans[i];
  }
  cout << ans;
}
