(n, x) = map(int, input().split())
if n == 2 and x == 0:
	print('NO')
else:
	print('YES')
	if n > 1:
		temp = x
		for i in range(n - 1):
			temp ^= i
		if temp:
			for i in range(1, n - 1):
				print(i, end=' ')
			print(2 ** 17, 2 ** 17 + temp)
		else:
			for i in range(n - 2):
				print(i, end=' ')
			print(2 ** 17, 2 ** 17 + n - 2)
	else:
		print(x)
