names = {}
n = int(input())
for i in range(n):
	row = input().split()
	if not row[0] in names.keys():
		names[row[0]] = set(row[2:])
	else:
		names[row[0]] |= set(row[2:])
keys = list(names.keys())
for j in range(len(keys)):
	names[keys[j]] = list(names[keys[j]])

def suffix(number, name):
	for i in range(len(names[name])):
		if not names[name][i] == number or len(names[name][i]) != len(number):
			if names[name][i].endswith(number):
				number = names[name][i]
			elif number.endswith(names[name][i]):
				names[name][i] = number
for j in range(len(keys)):
	for k in range(len(names[keys[j]])):
		suffix(names[keys[j]][k], keys[j])
print(len(keys))
for j in range(len(keys)):
	x = list(set(names[keys[j]]))
	print(keys[j], len(x), ' '.join(x))
