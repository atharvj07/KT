import sys
import math

def get_num():
	return int(input())

def get_numbers():
	return map(int, input().split())

def get_numbers_list():
	return list(get_numbers())

def solve(n, k):
	if k % 3 == 0:
		if n % (k + 1) != k and n % (k + 1) % 3 == 0:
			return 'Bob'
		return 'Alice'
	if n % 3 == 0:
		return 'Bob'
	return 'Alice'

def main():
	q = get_num()
	for _ in range(q):
		(n, k) = get_numbers()
		print(solve(n, k))
main()
