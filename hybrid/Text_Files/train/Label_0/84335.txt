T = int(input())
for _ in range(T):
	(n, s, k) = map(int, input().split())
	sl = [int(x) for x in input().split()]
	if not s in sl:
		print(0)
		continue
	if 1 not in sl:
		sl.append(min(sl) - 10)
	sl.append(max(sl) + 10)
	sl.sort()
	i = sl.index(s)
	l = s
	rl = []
	for x in sl[i + 1:]:
		if l >= n:
			break
		if x - l != 1:
			rl.append(abs(l + 1 - s))
			break
		l = x
	l = s
	for x in sl[:i][::-1]:
		if l <= 1:
			break
		if l - x != 1:
			rl.append(abs(l - 1 - s))
			break
		l = x
	print(min(rl))
