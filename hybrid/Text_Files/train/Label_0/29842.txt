#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2005;
int setbit[MAXN];
bool vis[MAXN];
bitset<MAXN> bitmat[MAXN];
bitset<MAXN> ans[MAXN];
char str[MAXN];
int N;
int main() {
  scanf("%d", &N);
  for (int i = 0; i < N; i++) {
    scanf("%s", str);
    int len = strlen(str);
    reverse(str, str + len);
    for (int l = 0; l < len; l++) {
      str[l] -= '0';
    }
    int k = 0;
    int b = 0;
    while (len) {
      b = 0;
      for (int j = len - 1; j >= 0; j--) {
        b = b * 10 + str[j];
        str[j] = b / 2;
        b &= 1;
      }
      bitmat[i][k++] = b;
      while (len > 0 && !str[len - 1]) len--;
    }
  }
  memset(ans, 0, sizeof ans);
  for (int i = 0; i < N; i++) {
    ans[i][i] = 1;
  }
  memset(setbit, -1, sizeof setbit);
  for (int i = 0; i < N; i++) {
    int j = 0;
    for (j = 0; j < MAXN; j++) {
      if (!bitmat[i][j])
        continue;
      else if (setbit[j] == -1) {
        setbit[j] = i;
        break;
      } else {
        bitmat[i] ^= bitmat[setbit[j]];
        ans[i] ^= ans[setbit[j]];
      }
    }
    if (j < MAXN) {
      printf("0");
    } else {
      printf("%lu ", ans[i].count() - 1);
      for (int k = 0; k < MAXN; k++) {
        if (ans[i][k] && k != i) {
          printf("%d ", k);
        }
      }
    }
    printf("\n");
  }
  return 0;
}
