#include <bits/stdc++.h>
using namespace std;
int N, X, Y;
int Xmin = INT_MAX, Xmax = INT_MIN, Ymin = INT_MAX, Ymax = INT_MIN,
    Umin = INT_MAX, Umax = INT_MIN, Dmin = INT_MAX, Dmax = INT_MIN;
int P[9][2], ans;
int main() {
  scanf("%d", &N);
  for (int i = 0; i < N; ++i) {
    scanf("%d%d", &X, &Y);
    Xmin = min(X - 1, Xmin);
    Xmax = max(X + 1, Xmax);
    Ymin = min(Y - 1, Ymin);
    Ymax = max(Y + 1, Ymax);
    Umin = min(Y - X - 1, Umin);
    Umax = max(Y - X + 1, Umax);
    Dmin = min(Y + X - 1, Dmin);
    Dmax = max(Y + X + 1, Dmax);
  }
  P[0][0] = Xmin;
  P[0][1] = Dmin - Xmin;
  P[1][0] = Xmin;
  P[1][1] = Umax + Xmin;
  P[2][0] = Ymax - Umax;
  P[2][1] = Ymax;
  P[3][0] = Dmax - Ymax;
  P[3][1] = Ymax;
  P[4][0] = Xmax;
  P[4][1] = Dmax - Xmax;
  P[5][0] = Xmax;
  P[5][1] = Xmax + Umin;
  P[6][0] = Ymin - Umin;
  P[6][1] = Ymin;
  P[7][0] = Dmin - Ymin;
  P[7][1] = Ymin;
  P[8][0] = P[0][0];
  P[8][1] = P[0][1];
  for (int i = 0; i < 8; ++i)
    ans += max(abs(P[i][0] - P[i + 1][0]), abs(P[i][1] - P[i + 1][1]));
  printf("%d\n", ans);
}
