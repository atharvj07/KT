def dostuff(d, q):
	stopped = False
	while q:
		(v, inc) = q.pop()
		for edge in neighs[v]:
			if edge in d:
				if d[edge] != v if inc else d[edge] == v:
					q = []
					stopped = True
					break
			else:
				d[edge] = v if inc else other(edge, v)
				q.append((other(edge, v), not inc))
	return (not stopped, d)

def other(edge, c):
	return edge[1] if edge[0] == c else edge[0]

def string(order, d):
	ls = []
	for k in order:
		ls.append(int(d[k] == k[0]))
	return ''.join(map(str, ls))

if ok:
