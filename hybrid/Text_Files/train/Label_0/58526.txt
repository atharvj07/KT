import java.io.*;
import java.util.*;

public class F {
    
    public static void bridge(int cur) {
        index[cur] = counter;
        low[cur] = counter;
        counter++;
        visited[cur] = true;
        for (int v: adj.get(cur)) {
            if (!visited[v]) {
                par[v] = cur;
                bridge(v);
                if (low[v] > index[cur]) { //bridge
                    //bridges.add(new Edge(cur, v));
                    bridges.add(1000000L * cur + v);
                }
                low[cur] = Math.min(low[cur], low[v]);
            } else if (v != par[cur]) { //not the parent
                low[cur] = Math.min(low[cur], index[v]);
            }
        }
    }
    
    public static void label(int cur) {
        visited[cur] = true;
        counter++;
        for (int v: adj.get(cur)) {
            //if (!visited[v] && !bridges.contains(new Edge(cur, v)) && !bridges.contains(new Edge(v, cur))) { //not a bridge
            if (!visited[v] && !bridges.contains(1000000L * cur + v) && !bridges.contains(1000000L * v + cur)) { //not a bridge
                label(v);
            }
        }
    }
    
    public static void orient(int start) {
        LinkedList<Integer> stack = new LinkedList<Integer>();
        stack.add(start);
        while (stack.size() != 0) {
            int cur = stack.pollLast();
            //Edge e1 = new Edge(v, cur);
            //Edge e2 = new Edge(cur, v);
            int prev = par[cur];
            if (prev != -1) {
                long e1 = 1000000L * cur + prev;
                long e2 = 1000000L * prev + cur;
                if (!wanted.contains(e1) && !wanted.contains(e2)) {
                    if (bridges.contains(e1) || bridges.contains(e2)) {
                        wanted.add(e1);
                        //System.out.println(e1);
                    } else {
                        wanted.add(e2);
                        //System.out.println(e2);
                    }
                }
            }
            if (!visited[cur]) {
                for (int v: adj.get(cur)) {
                    par[v] = cur;
                    stack.add(v);
                }
            }
            visited[cur] = true;
        }
    }
    
    public static ArrayList<LinkedList<Integer>> adj;
    public static int[] par;
    public static int[] low;
    public static int[] index;
    public static int counter;
    public static boolean[] visited;
    public static HashSet<Long> bridges = new HashSet<Long>();
    public static HashSet<Long> wanted = new HashSet<Long>();
    
    public static void main(String[]args) throws IOException {
        BufferedReader x = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(x.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        adj = new ArrayList<LinkedList<Integer>>();
        for (int i = 0; i < n; i++) {
            adj.add(new LinkedList<Integer>());
        }
        int[] us = new int[m];
        int[] vs = new int[m];
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(x.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            us[i] = u;
            vs[i] = v;
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        par = new int[n];
        for (int i = 0; i < n; i++) {
            par[i] = -1;
        }
        low = new int[n];
        index = new int[n];
        visited = new boolean[n];
        counter = 0;
        bridge(0);
        
        //find largest biconnected component
        int largest = 0;
        int startVertex = -1;
        visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                counter = 0;
                label(i);
                if (counter > largest) {
                    largest = counter;
                    startVertex = i;
                }
            }
        }
        System.out.println(largest);
        
        visited = new boolean[n];
        par = new int[n];
        for (int i = 0; i < n; i++) {
            par[i] = -1;
        }
        orient(startVertex);
        StringBuffer sb = new StringBuffer();
        for (int i = 0; i < m; i++) {
            //Edge curEdge = new Edge(us[i], vs[i]);
            long curEdge = 1000000L * us[i] + vs[i];
            us[i]++; //1-indexing
            vs[i]++;
            if (wanted.contains(curEdge)) {
                sb.append(us[i] + " " + vs[i] + "\n");
            } else {
                sb.append(vs[i] + " " + us[i] + "\n");
            }
        }
        System.out.print(sb.toString());
    }
}
