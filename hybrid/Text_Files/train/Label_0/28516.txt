#include <bits/stdc++.h>
using namespace std;
static int m1[6000][6000], m2[6000];
int main() {
  ios_base::sync_with_stdio(0);
  int n, x = 0;
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> m2[i];
  for (int i = 0; i < n; ++i) {
    m1[i][0] = m2[i] < m2[0];
    for (int j = 1; j < n; ++j) m1[i][j] = m1[i][j - 1] + (m2[i] < m2[j]);
    x += m1[i][i];
  }
  int a = 0, b = 0;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < i; ++j)
      if (a < 2 * (m1[i][i - 1] - m1[i][j - 1]) - (i - j) -
                  2 * (m1[j][i - 1] - m1[j][j]) + (i - j - 1)) {
        a = 2 * (m1[i][i - 1] - m1[i][j - 1]) - (i - j) -
            2 * (m1[j][i - 1] - m1[j][j]) + (i - j - 1);
        b = 1;
      } else if (a == 2 * (m1[i][i - 1] - m1[i][j - 1]) - (i - j) -
                          2 * (m1[j][i - 1] - m1[j][j]) + (i - j - 1))
        ++b;
  cout << x - a << ' ' << b << endl;
  return 0;
}
