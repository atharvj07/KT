(s, x1, x2) = map(int, input().split())
(t1, t2) = map(int, input().split())
(p, d) = map(int, input().split())
tt = 0
ti = 0
if p > x1:
	if d == 1:
		if x2 > x1:
			tt = (s + x2 + s - p) * t1
		else:
			tt = (s - p + s - x2) * t1
	elif x2 > x1:
		tt = (p + x2) * t1
	else:
		tt = (p - x2) * t1
elif d == 1:
	if x2 > x1:
		tt = (x2 - p) * t1
	else:
		tt = (s - p + s - x2) * t1
elif x2 > x1:
	tt = (p + x2) * t1
elif x1 == p:
	tt = (x1 - x2) * t1
else:
	tt = (p + s + s - x2) * t1
ti = abs(x2 - x1) * t2
print(min(tt, ti))
