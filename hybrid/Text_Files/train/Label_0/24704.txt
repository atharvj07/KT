#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const long long _64inf = 0x3f3f3f3f3f3f3f3f;
const int MAX = 1e5 + 5;
const int Mod = 1e6 + 3;
long long n;
int k;
int main() {
  cin >> n >> k;
  vector<int> a(k + 1);
  long long sum = 0;
  for (int i = 0; i < k; i++) {
    a[i] = i + 1;
    sum += 1LL * a[i];
  }
  if (sum > n) {
    cout << "NO" << endl;
    return 0;
  }
  long long nn = n - sum;
  for (int i = 0; i < k; i++) a[i] += nn / k + (i >= k - nn % k);
  if (nn != k - 1) {
    cout << "YES" << endl;
    for (int i = 0; i < k; i++) cout << a[i] << " ";
    cout << endl;
    return 0;
  }
  if (k == 2 || k == 3) {
    cout << "NO" << endl;
    return 0;
  }
  if (k > 3) --a[1], ++a[k - 1];
  cout << "YES" << endl;
  for (int i = 0; i < k; i++) cout << a[i] << " ";
  cout << endl;
  return 0;
}
