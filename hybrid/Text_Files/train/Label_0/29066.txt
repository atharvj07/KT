#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
struct inf {
  int a, b;
} e[N];
long long s[N];
bool cmp(inf a, inf b) { return a.a > b.a; }
int bnsr(int l, int r, int x) {
  int mid;
  while (l <= r) {
    mid = (l + r) >> 1;
    if (e[mid].a > x)
      l = mid + 1;
    else
      r = mid - 1;
  }
  return r;
}
void rlmn() {
  int n, m, i, x;
  long long ans = 0;
  scanf("%d%d", &m, &n);
  for (i = 1; i <= n; i++) scanf("%d%d", &e[i].a, &e[i].b);
  sort(e + 1, e + n + 1, cmp);
  for (i = 1; i <= n; i++) s[i] = s[i - 1] + e[i].a;
  for (i = 1; i <= n; i++) {
    x = bnsr(1, n, e[i].b);
    if (e[i].a > e[i].b) {
      if (m >= x)
        ans = max(ans, s[x] + 1ll * (m - x) * e[i].b);
      else
        ans = max(ans, s[m]);
    } else {
      if (m > x)
        ans = max(ans, s[x] + e[i].a + 1ll * (m - x - 1) * e[i].b);
      else
        ans = max(ans, s[m]);
    }
  }
  printf("%lld\n", ans);
}
int main() {
  int T;
  scanf("%d", &T);
  while (T--) rlmn();
  return 0;
}
