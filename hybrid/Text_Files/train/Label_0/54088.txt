import java.io.*;
import java.util.*;
import java.util.Map.Entry;
import java.util.regex.Pattern;

import static java.lang.Math.*;
import static java.util.Arrays.fill;
import static java.util.Arrays.binarySearch;
import static java.util.Arrays.sort;

public class Main {
	public static void main(String[] args) throws IOException {
		new Thread(null, new Runnable() {
			public void run() {
				try {
					try {
						if (new File("input.txt").exists())
							System.setIn(new FileInputStream("input.txt"));
					} catch (SecurityException e) {}
					new Main().run();
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
		}, "1", 1L << 24).start(); 
	}

	BufferedReader in;
	PrintWriter out;
	StringTokenizer st = new StringTokenizer("");
	
	int WBOX = 0;
	int HBOX = 1;
	int VBOX = 2;
	
	Map<String, Widget> map = new TreeMap<String, Widget>();
	
	void run() throws IOException {
		in = new BufferedReader(new InputStreamReader(System.in));
		out = new PrintWriter(System.out);
		
		for (int Q = nextInt(); Q --> 0; ) {
			String cmd = nextLine();
			if (cmd.startsWith("Widget")) {
				StringTokenizer tok = new StringTokenizer(cmd, " (,)");
				tok.nextElement();
				String name = tok.nextToken();
				int width = Integer.parseInt(tok.nextToken());
				int height = Integer.parseInt(tok.nextToken());
				map.put(name, new Widget(name, width, height));
			} else if (cmd.startsWith("VBox")) {
				StringTokenizer tok = new StringTokenizer(cmd, " (,)");
				tok.nextElement();
				String name = tok.nextToken();
				map.put(name, new Widget(name, VBOX));
			} else if (cmd.startsWith("HBox")) {
				StringTokenizer tok = new StringTokenizer(cmd, " (,)");
				tok.nextElement();
				String name = tok.nextToken();
				map.put(name, new Widget(name, HBOX));
			} else {
				StringTokenizer tok = new StringTokenizer(cmd, " .(,)");
				String name = tok.nextToken();
				String op = tok.nextToken();
				String arg = tok.nextToken();
				if (op.equals("pack")) {
					map.get(name).pack(map.get(arg));
				} else if (op.equals("set_border")) {
					map.get(name).border = Integer.parseInt(arg);
				} else {
					map.get(name).spacing = Integer.parseInt(arg);
				}
			}
		}
		
		for (Entry<String, Widget> e : map.entrySet())
			dfs(e.getValue());
		
		for (Entry<String, Widget> e : map.entrySet())
			out.println(e.getValue());
		
		out.close();
	}
	
	void dfs(Widget w) {
		if (w.type == WBOX) return;
		if (w.used) return;
		w.used = true;
		if (w.lst.isEmpty()) return;
		
		long sw = 2 * w.border;
		long sh = 2 * w.border;
		
		long mw = 0;
		long mh = 0;
		
		for (Widget ww : w.lst) {
			dfs(ww);
			sw += ww.width;
			sh += ww.height;
			
			mw = max(mw, ww.width);
			mh = max(mh, ww.height);
		}
		
		if (w.type == HBOX) {
			w.width = sw + (w.lst.size() - 1) * w.spacing;
			w.height = 2 * w.border + mh;
		} else {
			w.width = 2 * w.border + mw;
			w.height = sh + (w.lst.size() - 1) * w.spacing;
		}
		
	}

	class Widget {
		boolean used = false;
		int type = WBOX;
		String name;
		long width = 0;
		long height = 0;
		long border = 0;
		long spacing = 0;
		
		List<Widget> lst = new ArrayList<Widget>(10);
		
		Widget(String name, int type) {
			this.type = type;
			this.name = name;
		}
		
		Widget(String name, int width, int height) {
			this.name = name;
			this.width = width;
			this.height = height;
		}
		
		void pack(Widget w) {
			lst.add(w);
		}
		
		@Override
		public String toString() {
			return name + " " + width + " " + height;
		}
	}
	
	/*************************************************************** 
	 * Input 
	 **************************************************************/
	String nextToken() throws IOException {
		while (!st.hasMoreTokens())
			st = new StringTokenizer(in.readLine());
		return st.nextToken();
	}
	
	int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}
	
	long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}
	
	double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}
	
	String nextLine() throws IOException {
		st = new StringTokenizer("");
		return in.readLine();
	}
	
	boolean EOF() throws IOException {
		while (!st.hasMoreTokens()) {
			String s = in.readLine();
			if (s == null)
				return true;
			st = new StringTokenizer(s);
		}
		return false;
	}
}
