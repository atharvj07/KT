const int N = (1e7 + 1);
vector<int32_t>pr;
int32_t lp[N];
int32_t main() {
	ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL); ITO();
	int t = 1;
	memset(lp , 0 , sizeof lp);
	for (int i = 2; i < N; i++)
	{
		if (lp[i] == 0)
		{
			lp[i] = i; pr.push_back(i);
		}
		for (int j = 0 ; j < pr.size() and pr[j]*i < N; ++j) lp[pr[j]*i] = pr[j];
	}
	cin >> t;
	while (t--) solve();
	return 0;
}

void solve()
{
	map<int , int>m;
	int n; cin >> n;
	int arr[n]; for (auto &x : arr) cin >> x;
	for (auto x : arr)
	{
		int y = 1;
		while (x != 1)
		{
			if (x % (lp[x]*lp[x]) == 0)
			{
				x /= lp[x];
				x /= lp[x];
			}
			else {
				y *= lp[x];
				x /= lp[x];
			}
		}
		m[y]++;
	}
	int res1 = 0 , res2 = 0;
	for (auto &x : m)
	{
		res1 = max(res1 , x.second);
		if (x.second % 2 == 0 or x.first == 1) res2 += x.second;
	}
	res2 = max(res2 , res1);
	int q; cin >> q;
	while (q--)
	{
		int w; cin >> w;
		if (w == 0) cout << res1;
		else cout << res2;
		cout << "\n";
	}
	cout << "\n";
	return ;
}
