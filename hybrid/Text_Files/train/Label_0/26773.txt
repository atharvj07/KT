void fastscan(int &number) {
  bool negative = false;
  int c;
  number = 0;
  c = getchar();
  if (c == '-') {
    negative = true;
    c = getchar();
  }
  for (; (c > 47 && c < 58); c = getchar()) number = number * 10 + c - 48;
  if (negative) number *= -1;
}
long long GCD(long long a, long long b) { return b == 0 ? a : GCD(b, a % b); }
int LCM(int a, int b) { return a * (b / GCD(a, b)); }
bool CMP(int a, int b) { return a > b; }
long long n, a[int(3e5 + 5) + 1], dp[int(3e5 + 5) + 1][15],
    sum[int(3e5 + 5) + 1], m, k;
long long solve(int i, int l) {
  if (i >= n) return 0;
  long long &ret = dp[i][l];
  if (ret != -1) return ret;
  ret = 0;
  long long val = 0;
  if (!l) val = k;
  ret = max(ret, solve(i + 1, (l + 1) % m) + a[i] - val);
  return ret;
}
int main() {
  cin >> n >> m >> k;
  for (int i = 0; i < n; i++) cin >> a[i];
  memset(dp, -1, sizeof dp);
  long long ans = 0;
  for (int i = 0; i < n; i++) ans = max(ans, solve(i, 0));
  cout << ans << endl;
END:
  return 0;
}
