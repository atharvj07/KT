import sys
readline = sys.stdin.readline
T = int(readline())
for _ in range(T):
	(S, N, K) = map(int, readline().split())
	if S == K:
		ans = 'YES'
	elif S < K:
		ans = 'NO'
	else:
		cnt = S // (2 * K) * K
		cnt += min(S % (2 * K) + 1, K)
		if cnt >= N + 1:
			ans = 'NO'
		else:
			ans = 'YES'
	print(ans)
