import random
(n, a, b) = map(int, input().split())
commands = []
[commands.append(i) for i in range(1, n + 1)]
processing = True
k = 1
cr = commands
while processing:
	r = []
	for i in range(0, len(cr), 2):
		c = cr[i:i + 2]
		if a in c and b in c:
			processing = False
			break
		elif a in c:
			r.append(a)
		elif b in c:
			r.append(b)
		else:
			r.append(c[0])
	if processing:
		cr = r
		k += 1
	if len(cr) == 2:
		break
if len(cr) == 2:
	print('Final!')
else:
	print(k)
