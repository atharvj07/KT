#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int array[n];
    int half = n / 2;
    for (int i = 0; i < n; i++) {
      cin >> array[i];
    }
    half--;
    while (half > 0 && array[half] == array[half + 1]) {
      half--;
    }
    if (half == -1) {
      cout << "0 0 0\n";
      continue;
    }
    int change = 0;
    int last = -1;
    for (int i = 0; i <= half; i++) {
      if (array[i] != last) {
        last = array[i];
        change++;
      }
    }
    int abc = array[0];
    int index = 1;
    while (index < n && array[index] == abc) {
      index++;
    }
    if (half - index <= 0) {
      cout << "0 0 0\n";
      continue;
    }
    vector<int> v;
    for (int i = index; i <= half; i++) {
      v.push_back(array[i]);
    }
    if (v.size() <= index) {
      cout << "0 0 0\n";
      continue;
    }
    last = v[index];
    int second = 100000000;
    for (int i = index + 1; i < v.size(); i++) {
      if (v[i] != last) {
        second = i;
        break;
      }
    }
    half++;
    half = half - index - second;
    if (index < second && index < half) {
      cout << index << " " << second << " " << half << "\n";
    } else {
      cout << "0 0 0\n";
    }
  }
  return 0;
}
