var dataSetsCount = Number(readline());
for (var i = 0; i < dataSetsCount; i++) {
    var n = Number(readline());
    var numbers = toNumbersArray(readline());
    var locks = toNumbersArray(readline());

    var free = numbers.filter((x, index) => locks[index] !== 1).sort((a, b) => a - b);
    var sum = 0;
    var newArray = [];
    for (var j = 0; j < numbers.length; j++) {
        var candidate = numbers[j];
        if (locks[j] !== 1) {
            candidate = free.pop();
        }

        sum += candidate;
        newArray.push(candidate);
    }

    print(newArray.join(" "));
}

function toNumbersArray(string) {
    return string.split(" ").map(x => Number(x));
}
