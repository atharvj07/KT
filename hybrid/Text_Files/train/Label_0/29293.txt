#include <bits/stdc++.h>
using namespace std;
vector<int> e[2];
int d[7010][2], n;
bool l[7010][2], w[7010][2], was[7010][2];
void dfs(int v, int p) {
  was[v][p] = true;
  for (int x : e[p ^ 1]) {
    int u = (v - x + n) % n;
    if (!was[u][p ^ 1]) {
      if (l[v][p])
        w[u][p ^ 1] = true;
      else if (!(--d[u][p ^ 1]))
        l[u][p ^ 1] = true;
      else
        continue;
      dfs(u, p ^ 1);
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i(0), s; i < 2; ++i) {
    cin >> s;
    e[i].resize(s);
    for (int &c : e[i]) cin >> c;
    for (int j(0); j < n; ++j) d[j][i] = s;
  }
  l[0][0] = l[0][1] = true;
  dfs(0, 0);
  dfs(0, 1);
  for (int p(0); p < 2; ++p) {
    for (int i(1); i < n; ++i)
      if (w[i][p])
        cout << "Win ";
      else if (l[i][p])
        cout << "Lose ";
      else
        cout << "Loop ";
    cout << '\n';
  }
}
