#include <iostream>
#include <string>
#include <algorithm>
using namespace std;



int main(void)
{
	string str; int num;
	cin >> num;
	while (num--)
	{
		int max = 0, min = 0, tempmax[8] = {}, tempmin[8] = {};
		cin >> str;
		for (int i = 0; i < 8; i++) tempmin[i] = str[i] - '0';
		sort(tempmin, tempmin + 8);
		for (int i = 0; i < 8; i++) tempmax[i] = tempmin[7 - i];
		int num = 10000000;
		for (int i = 0; i < 8; i++)
		{
			max += tempmax[i] * num;
			min += tempmin[i] * num;
			num /= 10;
		}
		cout << max - min << endl;;
	}
	return 0;
}
