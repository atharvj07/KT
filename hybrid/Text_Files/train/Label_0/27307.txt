#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const double pi = acos(-1);
const int maxn = 200010;
vector<pair<int,int>> points;

int LIS() {
	if (points.empty()) return 0;
	vector<int> ret(points.size(), 1e9);
	for (auto t:points) {
		*lower_bound(ret.begin(), ret.end(), t.second) = t.second;
	}
	return find(ret.begin(), ret.end(), 1e9) - ret.begin();
}

int main() {
	int x1, y1, x2, y2, n;
	scanf("%d%d%d%d%d", &x1, &y1, &x2, &y2, &n);
	double pval = 20 - 5 * pi, qval = 10 * pi - 20;
	double ans = 100.0 * (abs(x1 - x2) + abs(y1 - y2));
	int xs = x1 > x2 ? -1 : 1;
	int ys = y1 > y2 ? -1 : 1;
	x1 *= xs; x2 *= xs;
	y1 *= ys; y2 *= ys;
	for (int i = 1; i <= n; i++) {
		int x, y;
		scanf("%d%d", &x, &y);
		x *= xs, y *= ys;
		if (x1 <= x && x <= x2 && y1 <= y && y <= y2) points.emplace_back(x, y);
	}
	sort(points.begin(), points.end());
	int k = LIS();
	if (k == min(x2 - x1, y2 - y1) + 1) {
		ans += -(k-1) * pval + qval;
	} else {
		ans += -k * pval;
	}
	printf("%.10f\n", ans);
	
	return 0;
}
