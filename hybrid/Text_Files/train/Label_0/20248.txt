#include <bits/stdc++.h>
using namespace std;
string char_to_str(char c) {
  string tem(1, c);
  return tem;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  long long t;
  cin >> t;
  while (t) {
    --t;
    long double a, b;
    cin >> a >> b;
    long double one = 1.00000000;
    if (a == 0 && b == 0) {
      cout << fixed << setprecision(10) << one << "\n";
      continue;
    }
    if (a == 0) {
      cout << fixed << setprecision(10) << one / 2 << "\n";
      continue;
    }
    if (b == 0) {
      cout << fixed << setprecision(10) << one << "\n";
      continue;
    }
    long double r = min(a / 4, b);
    cout << fixed << setprecision(10)
         << 0.5 + 0.5 * ((a * r - 2 * r * r) / (a * b)) << "\n";
  }
}
