import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        new Main().run();
    }

    void run() {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int c = in.nextInt();
        int d = in.nextInt();
        int answer = 0;
        for (int x = 1; x <= n; ++ x) {
            int q = x;
            q = b + q - c;
            if (q < 1 || q > n) continue;
            q = a + q - d;
            if (q < 1 || q > n) continue;
            q = c + q - b;
            if (q < 1 || q > n) continue;
            q = q + d - a;
            if (q != x) continue;
            answer ++;
        }
        System.out.println(answer * 1l * n);
    }
}
