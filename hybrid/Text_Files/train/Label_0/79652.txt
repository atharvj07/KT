(a, b, x) = map(int, input().split())
skel = ''
if a <= b:
	skel = '10' * (x // 2 + x % 2)
	if x % 2 == 0:
		skel += '1'
else:
	skel = '01' * (x // 2 + x % 2)
	if x % 2 == 0:
		skel += '0'
left0 = a - skel.count('0')
left1 = b - skel.count('1')
if skel[0] == '1':
	for i in range(left0):
		skel = skel[0:1] + '0' + skel[1:]
	for i in range(left1):
		skel = '1' + skel
else:
	for i in range(left1):
		skel = skel[0:1] + '1' + skel[1:]
	for i in range(left0):
		skel = '0' + skel
print(skel)
