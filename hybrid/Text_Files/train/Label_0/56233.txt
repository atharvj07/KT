import java.util.*;
public class CF
{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            int n=sc.nextInt(),p=0;
            String s="",sn="0123456789";
            if(n==1){System.out.println("9");}
            else if(n==2){System.out.println("98");}
            else if(n==3){System.out.println("989");}
            else if(n>=4){
                n=n-3;
                int ct=n%10;
                int st=n/10;
                for(int i=0;i<st;i++){s+=sn;}
                while(ct-->0){
                    s+=(p%10);
                    p++;
                }
                System.out.println("989"+s);
            }
    }
  }
}
