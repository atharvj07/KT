#include <bits/stdc++.h>
using namespace std;
const long long MAX = 1e6 + 1;
const long long PRIME = 1e9 + 7;
long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
void solve() {
  long long n, m, k;
  cin >> n >> m >> k;
  long long ans = 0;
  for (long long i = 0; i < k; i++) {
    ans += 2 * (n) + 2 * (m - 2);
    n -= 4;
    m -= 4;
  }
  cout << ans;
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long t;
  t = 1;
  while (t--) {
    solve();
  }
}
