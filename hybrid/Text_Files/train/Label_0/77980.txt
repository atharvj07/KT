q = int(input())
for query in range(q):
	s = input()
	n = len(s)
	MinSoFar = n
	zeri_first = 0
	zeri_second = 0
	for i in range(n):
		if s[i] == '0':
			zeri_second += 1
	for sep in range(n):
		if s[sep] == '0':
			zeri_first += 1
			zeri_second -= 1
		l_first = sep + 1
		l_second = n - l_first
		MinSoFar = min(MinSoFar, zeri_first + l_second - zeri_second, l_first - zeri_first + zeri_second)
	print(MinSoFar)
