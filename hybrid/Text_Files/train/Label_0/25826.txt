#include <bits/stdc++.h>
using namespace std;
using namespace std;
const int N = 100001;
const long long inf = 1e11;
long long l, r, h[N], p[N];
int i, j, n, m;
bool ok(long long d) {
  long long r = 0;
  int l = 0;
  for (int i = 0; i < n; i++)
    if (h[i] <= p[l]) {
      r = h[i] + d;
      while (r >= p[l] && l < m) l++;
      if (l == m) return true;
    } else {
      if (h[i] - p[l] > d) return false;
      r = h[i];
      r = max(
          r, max(h[i] + d - 2 * (h[i] - p[l]), h[i] + (d - (h[i] - p[l])) / 2));
      while (r >= p[l] && l < m) l++;
      if (l == m) return true;
    }
  return false;
}
int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  cin >> n >> m;
  for (i = 0; i < n; i++) cin >> h[i];
  for (i = 0; i < m; i++) cin >> p[i];
  l = 0;
  r = inf;
  while (r - l > 1) {
    long long mid = (l + r) / 2;
    if (ok(mid))
      r = mid;
    else
      l = mid + 1;
  }
  if (ok(l))
    cout << l;
  else
    cout << r;
}
