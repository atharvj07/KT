template <typename T>
ostream& operator<<(ostream& cout, const vector<T>& a) {
  if ((int)a.size() == 0) {
    return (cout << "()");
  }
  cout << "(" << a[0];
  for (int i = 1; i < (int)a.size(); i++) {
    cout << "; " << a[i];
  }
  return (cout << ")");
}

  if ((int)a.size() == 0) {
    return (cout << "{}");
  }
  auto it = a.begin();
  cout << "{" << *it;
  ++it;
  for (; it != a.end(); ++it) {
    cout << "; " << *it;
  }
  return (cout << "}");
}

const int nmax = 2000 * 1000;
const int inf = 2000 * 1000 * 1000;
const long long infl = 1000ll * 1000ll * 1000ll * 1000ll * 1000ll * 1000ll;
const int mod = 1000 * 1000 * 1000 + 7;
const long double pi = acos(-1.0);
struct matrix {
  long long a[2][2];
  matrix() { a[0][0] = a[0][1] = a[1][0] = a[1][1] = 0; }
  matrix(long long x1, long long x2, long long x3, long long x4) {
    a[0][0] = x1;
    a[0][1] = x2;
    a[1][0] = x3;
    a[1][1] = x4;
  }
};
ostream& operator<<(ostream& cout, const matrix& a) {
  return cout << "(" << a.a[0][0] << "; " << a.a[0][1] << "; " << a.a[1][0]
              << "; " << a.a[1][1] << ")";
}
inline matrix operator*(const matrix& a, const matrix& b) {
  matrix ans = matrix();
  for (int i = 0; i < 2; i++) {
    for (int j = 0; j < 2; j++) {
      for (int k = 0; k < 2; k++) {
        ans.a[i][j] = (ans.a[i][j] + a.a[i][k] * b.a[k][j]) % mod;
      }
    }
  }
  return ans;
}
inline matrix operator+(const matrix& a, const matrix& b) {
  matrix ans = matrix();
  for (int i = 0; i < 2; i++) {
    for (int j = 0; j < 2; j++) {
      ans.a[i][j] = (a.a[i][j] + b.a[i][j]) % mod;
    }
  }
  return ans;
}
matrix powa(const matrix& a, long long x) {
  if (x == 0) {
    return matrix(1, 0, 0, 1);
  } else if (x == 1) {
    return a;
  }
  matrix b = powa(a, x / 2);
  b = b * b;
  if (x & 1) {
    return b * a;
  } else {
    return b;
  }
}
struct node {
  matrix x, mod;
  node() { mod = x = matrix(1, 0, 0, 1); }
};
node a[nmax];
int n, m, x, l, r;
void push(int ind) {
  a[ind * 2].x = a[ind * 2].x * a[ind].mod;
  a[ind * 2 + 1].x = a[ind * 2 + 1].x * a[ind].mod;
  a[ind * 2].mod = a[ind * 2].mod * a[ind].mod;
  a[ind * 2 + 1].mod = a[ind * 2 + 1].mod * a[ind].mod;
  a[ind].mod = matrix(1, 0, 0, 1);
}
void change(int ind, int l1, int r1, int l, int r, const matrix& x) {
  if (l1 > r || l > r1) {
    return;
  }
  if (l <= l1 && r1 <= r) {
    a[ind].x = a[ind].x * x;
    a[ind].mod = a[ind].mod * x;
    return;
  }
  push(ind);
  change(ind * 2, l1, (l1 + r1) / 2, l, r, x);
  change(ind * 2 + 1, (l1 + r1) / 2 + 1, r1, l, r, x);
  a[ind].x = a[ind * 2].x + a[ind * 2 + 1].x;
}
long long ask(int ind, int l1, int r1, int l, int r) {
  if (l1 > r || l > r1) {
    return 0;
  }
  if (l <= l1 && r1 <= r) {
    return a[ind].x.a[0][0];
  }
  push(ind);
  return (ask(ind * 2, l1, (l1 + r1) / 2, l, r) +
          ask(ind * 2 + 1, (l1 + r1) / 2 + 1, r1, l, r)) %
         mod;
}
int main() {
  cin >> n >> m;
  int s = 1;
  while (s < n) {
    s <<= 1;
  }
  for (int i = 0; i < n; i++) {
    a[i + s].x = matrix(0, 1, 0, 1);
  }
  for (int i = s - 1; i >= 0; i--) {
    a[i].x = a[i * 2].x + a[i * 2 + 1].x;
  }
  for (int i = 0; i < n; i++) {
    scanf("%d", &x);
    change(1, 0, s - 1, i, i, powa(matrix(0, 1, 1, 1), x));
  }
  for (int i = 0; i < m; i++) {
    int t;
    scanf("%d", &t);
    if (t == 1) {
      scanf("%d%d%d", &l, &r, &x);
      l--, r--;
      change(1, 0, s - 1, l, r, powa(matrix(0, 1, 1, 1), x));
    } else {
      scanf("%d%d", &l, &r);
      l--, r--;
      printf("%d\n", (int)ask(1, 0, s - 1, l, r));
    }
  }
}
