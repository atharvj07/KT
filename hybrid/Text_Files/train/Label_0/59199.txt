import java.io.*;
import java.util.*;
public class Main{
    static long atol(String n){return Long.parseLong(n);}
    public static void main(String jfkd[])throws Exception{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        HashSet<Long> t = new HashSet<Long>();
        long a=0,b=1, max = 1000000000;
        while(a<=max){
            a+=b++;
            t.add(a);
        }
        long n = atol(br.readLine());
        boolean hay =false;
        for(long c:t)
            if(t.contains(n-c)){
                hay = true;
                break;
            }
        if(hay)
            System.out.println("YES");
        else
            System.out.println("NO");
    }
}
