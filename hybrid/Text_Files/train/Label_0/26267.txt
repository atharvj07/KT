#include <bits/stdc++.h>
using namespace std;
int JC(int a, int b) {
  if (a % b == 0) return b;
  return JC(b, a % b);
}
int main() {
  int a, b, c, h;
  scanf("%d%d%d", &a, &b, &c);
  int k = JC(a, b);
  a /= k;
  b /= k;
  k = a / b;
  a = (a - k * b) * 10;
  for (int i = 1; i < 100001; i++) {
    h = a / b;
    a = (a - h * b) * 10;
    if (h == c) {
      printf("%d", i);
      return 0;
    }
  }
  printf("-1");
  return 0;
}
