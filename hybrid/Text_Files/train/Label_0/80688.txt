n = int(input())
s = input()
l = list(s)
(count, count1) = (0, 0)
if n % 2 == 1:
	print(':(')
	exit()
for i in range(n):
	if s[i] == '(':
		count += 1
	elif s[i] == ')':
		count1 += 1
count = n // 2 - count
for j in range(n):
	if s[j] == '?' and count > 0:
		l[j] = '('
		count -= 1
	elif s[j] == '?':
		l[j] = ')'
m = ''.join(l)
count1 = 0
count = 0
if m[0] == ')' or m[n - 1] == '(':
	print(':(')
	exit()
for k in range(n):
	if m[k] == '(':
		count += 1
	else:
		count1 += 1
	if count == count1 and count != n / 2:
		print(':(')
		exit()
if count == n / 2:
	print(m)
else:
	print(':(')
