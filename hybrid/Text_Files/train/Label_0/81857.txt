def ii():
	return int(input())

def mi():
	return map(int, input().split())

def li():
	return list(mi())

def gen(r):
	p = q = 1
	t = [0]
	for i in range(h):
		pp = p
		p = len(t) + 1
		if r and i and (min(a[i], a[i - 1]) > 1):
			t += [q]
			t += [pp] * (a[i] - 1)
		else:
			t += [pp] * a[i]
		q = len(t)
	return t
