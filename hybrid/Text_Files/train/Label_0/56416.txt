import java.io.*;
import java.util.*;

public class Main implements Runnable{
    public static void main(String[] args) throws Exception {
        //new Thread(null,new Main(),"Thread-1",1024*1024*10).start();
        new Main().solve();
    }
    int n;
    int[]nx=new int[500005];
    int[]cnt=new int[500005];
    int[]to=new int[1000005];
    int[]next=new int[1000005];
    int[]head=new int[500005];
    int[]ans=new int[500005];
    int len;
    Stack<Pair>st=new Stack<>();
    Stack<Integer>qu=new Stack<>();
    void solve() throws Exception{
        InputReader in=new InputReader(System.in);
        PrintWriter out=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
        int t=in.nextInt();
        while(t--!=0){
            n=in.nextInt();
            len=0;
            st.clear();qu.clear();
            for(int i=1;i<=n+1;i++){
                head[i]=-1;cnt[i]=0;
            }
            for(int i=1;i<=n;i++)nx[i]=in.nextInt();
            for(int i=1;i<=n;i++){
                while(!st.isEmpty()&&st.peek().b==i)st.pop();
                if(!st.isEmpty())add(st.peek().a,i);
                if(nx[i]!=-1){
                    add(nx[i],i);st.push(new Pair(i,nx[i]));
                }
            }
            for(int i=1;i<=n+1;i++)if(cnt[i]==0)qu.push(i);
            int id=n+1;
            while(!qu.isEmpty()){
                int p=qu.pop();
                ans[p]=id--;
                for(int i=head[p];i!=-1;i=next[i]){
                    cnt[to[i]]--;
                    if(cnt[to[i]]==0)qu.push(to[i]);
                }
            }
            if(id!=0)out.print(-1);
            else for(int i=1;i<=n;i++)out.print(ans[i]+" ");
            out.println();
        }
        out.flush();
    }
    void add(int a,int b){
        to[len]=b;next[len]=head[a];head[a]=len;len++;cnt[b]++;
    }
    @Override
    public void run() {
        try {
            solve();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
    class Pair{
        int a,b;
        Pair(int x,int y){
            a=x;b=y;
        }
    }
    void shuffle(int[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i),tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    void shuffle(Object[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i);
            Object tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    void shuffle(long[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i);
            long tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    class InputReader{
        StreamTokenizer tokenizer;
        public InputReader(InputStream stream){
            tokenizer=new StreamTokenizer(new BufferedReader(new InputStreamReader(stream)));
            tokenizer.ordinaryChars(33,126);
            tokenizer.wordChars(33,126);
        }
        public String next() throws IOException {
            tokenizer.nextToken();
            return tokenizer.sval;
        }
        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }
        public boolean hasNext() throws IOException {
            int res=tokenizer.nextToken();
            tokenizer.pushBack();
            return res!=tokenizer.TT_EOF;
        }
    }
}
