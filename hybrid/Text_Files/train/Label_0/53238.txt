import java.util.*;
public class Korobka{
	public static void main(String[]args){
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int a[] = new int[n];
		int size = 1;
		int ans = 1;
		for(int i = 0; i<n; i++){
			a[i] = in.nextInt();
		}
		Arrays.sort(a);
		for(int i = 1; i<n; i++){
			if (a[i] == a[i-1])
				size++;
			else
				size = 1;
			if (ans < size)
				ans = size;
		}	
		System.out.print(ans);
	}
}
