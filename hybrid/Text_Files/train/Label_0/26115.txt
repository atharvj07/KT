const ll mod=1000000007; /// 998244353

int n,m;
string s[1005];
int pos[1005][2];

void gogo()
{
    cin>>n>>m;
    for(int i=1;i<=n;++i)
    {
        cin>>s[i];
        s[i]=' '+s[i];
    }

    if(m&1)
    {
        int x=1, y=2;
        cout<<"YES\n";
        cout<<x<<' '<<y<<' ';
        for(int i=1;i<=m-1;++i)
        {
            cout<<x<<' ';
            swap(x,y);
        }
        cout<<endl;
        return;
    }

    for(int i=1;i<=n;++i)
    for(int j=1;j<=n;++j) if(i!=j)
    {
        if(s[i][j]==s[j][i])
        {
            int x=i, y=j;
            assert(x!=y);
            cout<<"YES\n";
            cout<<x<<' '<<y<<' ';
            for(int i=1;i<=m-1;++i)
            {
                cout<<x<<' ';
                swap(x,y);
            }
            cout<<endl;
            return;
        }
    }

    if(n<=2)
    {
        cout<<"NO\n";
        return;
    }

    cout<<"YES\n";

    for(int i=1;i<=n;++i) pos[i][0]=pos[i][1]=0;

    int x=-1, y=-1, z=-1;
    for(int i=1;i<=n;++i)
    {
        for(int j=1;j<=n;++j) if(i!=j)
        {
            pos[j][s[i][j]-'a']=i;
        }
    }

    for(int i=1;i<=n;++i)
    {
        for(int j=1;j<=n;++j) if(i!=j)
        {
            if(pos[i][s[i][j]-'a'])
            {
                x=pos[i][s[i][j]-'a'];
                y=i;
                z=j;
            }
        }
    }

    assert(x!=-1);
    assert(x!=y);
    assert(y!=z);
    assert(x!=z);
    assert(s[x][y]==s[y][z]);
    assert(s[z][y]==s[y][x]);

    if(m%4!=0)
    {
        vector<int> path={x,y,z};
        cout<<x<<' '<<y<<' '<<z<<' ';
        reverse(ALL(path));
        m-=2;
        while(m>=2)
        {
            for(int i=1;i<zs(path);++i) cout<<path[i]<<' ';
            reverse(ALL(path));
            m-=2;
        }
    }
    else
    {
        vector<int> path={z,y,x,y};
        m-=4;
        vector<int> res={y,z,y,x,y};
        while(m>0)
        {
            for(int i:path) res.pb(i);
            m-=4;
        }
        for(int i:res) cout<<i<<' ';
    }
    cout<<endl;
}
int main()
{
    ios_base::sync_with_stdio(0); cin.tie(0);

        freopen(task".out","w",stdout);
    }
    int t;
    cin>>t;
    while(t--) gogo();
    return 0;
}
