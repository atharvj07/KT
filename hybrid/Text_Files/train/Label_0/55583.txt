import java.util.Scanner;

/**
 *
 * @author Mohammad Dehghan
 */
public class ACM {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x, y, s;
        x = scanner.nextInt();
        y = scanner.nextInt();
        s = scanner.nextInt();
        x = Math.abs(x);
        y = Math.abs(y);
        String str = "Yes";
        if ((x + y > s) || (((s - x - y) % 2) != 0)) {
            str = "No";
        }
        System.out.println(str);
    }

}
