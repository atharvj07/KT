n = int(stdin.readline().strip())
arr = list(map(int, stdin.readline().split()))
sequence = [0]
moves = []
i = 0
j = n - 1
while i != j:
	if arr[i] > arr[j]:
		if arr[j] > sequence[-1]:
			sequence.append(arr[j])
			j -= 1
			moves.append('R')
		elif arr[i] > sequence[-1]:
			sequence.append(arr[i])
			i += 1
			moves.append('L')
		else:
			break
	elif arr[i] > sequence[-1]:
		sequence.append(arr[i])
		i += 1
		moves.append('L')
	elif arr[j] > sequence[-1]:
		sequence.append(arr[j])
		j -= 1
		moves.append('R')
	else:
		break
if i == j:
	if arr[i] > sequence[-1]:
		sequence.append(arr[i])
		moves.append('R')
print(len(sequence) - 1)
