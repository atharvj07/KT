#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

struct St{
  string nm;
  int fv;

  St() {}
  St(string _nm, int _fv): fv(_fv), nm(_nm) {}

  void print(){
    cout << fv << " " << nm << endl;
    return ;
  }

  bool operator<(const St& st) const{
    if(fv == st.fv) return nm < st.nm;
    return fv < st.fv;
  }
};

int main()
{
  int n, m, d;
  string str;
  bool a[21][31];
  vector<St> vec;

  while(cin>>n && n){
    fill(&a[0][0], &a[21][0], false);

    for(int i = 0; i < n; i++){
      cin >> str >> m;
      vec.push_back(St(str, 0));
      while(m--){
	cin >> d;
	a[i][d] = true;
      }
    }

    for(int i = 0; i < 31; i++){
      vector<int> v1;
      for(int j = 0; j < n; j++)
	if(a[j][i]) v1.push_back(j);

      for(int j = 0, l = v1.size(); j < l; j++)
	vec[v1[j]].fv += (n-l+1);
    }
    sort(vec.begin(), vec.end());

    vec[0].print();
    vec.clear();
  }

  return 0;
}
