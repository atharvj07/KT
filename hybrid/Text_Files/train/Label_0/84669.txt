n = int(input())
(x1, x2) = map(int, input().split())
y_range = []
for _ in range(n):
	(k, b) = map(int, input().split())
	y_range.append((x1 * k + b, x2 * k + b))
y_range.sort(key=lambda x: (x[0], x[1]))
no = True
for i in range(n - 1):
	if y_range[i][1] > y_range[i + 1][1]:
		no = False
		break
if no:
	print('NO')
else:
	print('YES')
