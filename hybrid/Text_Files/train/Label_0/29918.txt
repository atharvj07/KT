#include<bits/stdc++.h>
using namespace std;
#define int long long
const int N=205;
int n,k,mod;
int fac[N],inv[N],f[N][N];
int ksm(int b,int n){
	int res=1;
	while(n){
		if(n&1) res=1ll*res*b%mod;
		b=1ll*b*b%mod; n>>=1;
	}
	return res;
} 
int calc(int k,int l,int in){
	for(int i=1;i<=k;++i){
		int sum=0;
		for(int j=l;j>=0;--j){
			sum=(sum+f[i-1][j])%mod;
			if(i==1) sum=1;
			if(i-1+j<=in && k-i+l-j<=in) f[i][j]=sum;
			else f[i][j]=0;
		}
		f[1][0]=0;
	}
	return f[k][0];
}
int solve(int n,int m){
	if(n==1) return 1;
	int ans=0;
	for(int i=2;i<n;++i)
		ans=(ans+calc(i,n-i,m))%mod;
	return 1ll*ans*fac[n-1]%mod;
}
void init(int n){
	fac[0]=1;
	for(int i=1;i<=n;++i)
		fac[i]=1ll*fac[i-1]*i%mod;
	inv[n]=ksm(fac[n],mod-2);
	for(int i=n-1;i>=1;--i)
		inv[i]=1ll*inv[i+1]*(i+1)%mod;
}
signed main(){
	cin>>n>>k>>mod;
	init(n);int ans=0;
	for(int i=0;i<=k;++i) //枚举多少个点度数为0
	{
		ans=ans+solve(n-i,k-i)*fac[n]%mod*inv[n-i]%mod;
		ans%=mod;	
	} 
	cout<<ans;
	return 0;
}
