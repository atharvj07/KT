#include<iostream>
#include<cstdio>
#define N 110000
using namespace std;
int n,c,k;
long long t[N];
long long s[N];
int main()
{
    cin>>n;
    for(int i=1;i<=n;i++)cin>>t[i];
    sort(t+1,t+n+1);
    int ff=0;
    for(int i=1;i<n;i++)
    {
        s[i]=s[i-1]+t[i];
        if(s[i]*2<t[i+1])ff=i;
    }
    cout<<n-ff;
}
