#include <bits/stdc++.h>
using namespace std;
const int MAX = 222;
int T, N, M, cnt, u, v;
int deg[MAX], bal[MAX];
bool E[MAX][MAX], sol[MAX][MAX];
void init() {
  memset(E, 0, sizeof(E));
  memset(sol, 0, sizeof(sol));
  memset(deg, 0, sizeof(deg));
  memset(bal, 0, sizeof(bal));
  cnt = 0;
}
void DFS(int n) {
  for (int I = 0; I < N; I++)
    if (E[n][I]) {
      E[n][I] = E[I][n] = false;
      sol[n][I] = true;
      bal[I]++;
      bal[n]--;
      deg[n]--;
      deg[I]--;
      DFS(I);
      return;
    }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> T;
  while (T--) {
    init();
    cin >> N >> M;
    for (int I = 0; I < M; I++) {
      cin >> u >> v;
      u--, v--;
      E[u][v] = E[v][u] = true;
      deg[u]++;
      deg[v]++;
    }
    while (true) {
      int x = 0;
      for (int I = 0; I < N; I++)
        if (deg[I]) {
          if (deg[I] & 1)
            x = I;
          else if (deg[x] == 0)
            x = I;
        }
      if (!deg[x]) break;
      DFS(x);
    }
    for (int I = 0; I < N; I++)
      if (!bal[I]) cnt++;
    cout << cnt << endl;
    for (int I = 0; I < N; I++)
      for (int K = 0; K < N; K++)
        if (sol[I][K]) cout << I + 1 << " " << K + 1 << endl;
  }
}
