#include "bits/stdc++.h"
using namespace std;
typedef long long ll;
typedef pair<int,int> pii;
#define rep(i,n) for(ll i=0;i<(ll)(n);i++)
#define all(a) (a).begin(),(a).end()
#define pb emplace_back
#define INF (1e9+1)

int main(){
    set<string> st;
    int n;
    cin>>n;
    string s;
    rep(i,n){
        cin>>s;
        
        rep(i,s.size()){
            for(int j=1;j<=4;j++){
                if(i+j-1<s.size())st.insert(s.substr(i,j));
            }
        }
    }
    
    string abc = "abcdefghijklmnopqrstuvwxyz";
    
    for(auto a:abc){
        string t = {a};
        if( not st.count(t) ){
            cout<<t<<endl;
            return 0;
        }
    }
    
    for(auto a:abc){
        for(auto b:abc){
            string t = {a,b};
            if( not st.count(t) ){
                cout<<t<<endl;
                return 0;
            }
        }
    }
    
    for(auto a:abc){
        for(auto b:abc){
            for(auto c:abc){
                string t = {a,b,c};
                if( not st.count(t) ){
                    cout<<t<<endl;
                    return 0;
                }
            }
        }
    }
    
    for(auto a:abc){
        for(auto b:abc){
            for(auto c:abc){
                for(auto d:abc){
                    string t = {a,b,c,d};
                    if( not st.count(t) ){
                        cout<<t<<endl;
                        return 0;
                    }
                }
            }
        }
    }
    
    cout<<"NO"<<endl;
}
