#include <bits/stdc++.h>
int N, S, W, cnt[200001], O;
int main() {
  scanf("%d%d", &N, &S);
  for (int i = 1, x; i <= N; i++) {
    scanf("%d", &x);
    if (i == S && x) O++, x = 0;
    cnt[x]++;
  }
  cnt[N] = cnt[0] - 1;
  cnt[0] = 1;
  for (int i = 1, j = N;; i++)
    if (!cnt[i]) {
      while (!cnt[j] && j >= i) j--;
      if (j < i) {
        printf("%d\n", O);
        return 0;
      }
      cnt[j]--, O++;
    }
  assert(0);
}
