#include <bits/stdc++.h>
using namespace std;

long long N;

int main(){
//	cin.tie(0);
//	ios::sync_with_stdio(false);
	cin >> N;
	for(int h=1; h<=3500; ++h)
		for(int n=1; n<=3500; ++n)
			if(4*h*n != N*(n+h) && N*h*n % (4*h*n-N*n-N*h) == 0 && (4*h*n-N*n-N*h) > 0){
				cout << h << " " << n << " " << N*h*n / (4*h*n-N*n-N*h) << "\n";
				return 0;
			}
}
