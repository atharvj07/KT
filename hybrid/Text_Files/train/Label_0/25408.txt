//#define LL long long
typedef long long LL;

int cmp(const void*p, const void* q) {
    int* x = (int*)p, *y = (int*)q;
    return *x - *y;
}

int cmp_ll(const void*p, const void* q) {
    LL* x = (LL*)p, *y = (LL*)q;
    if (*x < *y) return -1;
    else if (*x > *y) return 1;
    else return 0;
}

LL extended_gcd(LL n, LL m, LL& x, LL& y) {
    if (m == 0) {
        x = 1;
        y = 0;
        return n;
    } else {
        LL r = extended_gcd(m, n%m, y, x);
        y -= x * (n / m);
        return r;
    }
}

const int MAX = 100001;
int l[MAX], r[MAX];
LL best1[MAX], best2[MAX]; 

void dp(int cur, int parent, vector< vector<int> > &e) {
    best1[cur] = best2[cur] = 0;
    for (auto nxt : e[cur]) {
        if (nxt != parent) {
            dp(nxt, cur, e);
            best1[cur] += max(best1[nxt] + abs(l[cur] - l[nxt]), best2[nxt] + abs(l[cur] - r[nxt]));
            best2[cur] += max(best1[nxt] + abs(r[cur] - l[nxt]), best2[nxt] + abs(r[cur] - r[nxt]));
        }
    }
}

int main() {
    int T = 1, n;
    scanf("%d", &T);    
    while (T-- > 0) {
        scanf("%d", &n);
        vector< vector<int> > e(n+1);
        for (int i=1; i<=n; i++) {
            scanf("%d%d", &l[i], &r[i]);
        }
        for (int i=1; i<n; i++) {
            int u, v;
            scanf("%d%d", &u, &v);
            e[u].emplace_back(v);
            e[v].emplace_back(u);
        }
        dp(1, -1, e);
        LL res = max(best1[1], best2[1]);
        printf("%lld\n", res);
    }    
    return 0;
}
