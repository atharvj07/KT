#include <bits/stdc++.h>
#define MOD 1000000007LL
using namespace std;
typedef long long ll;
typedef pair<int,int> P;

class segtree{
public:
	static const int N=1<<18;
	int dp[1<<19];
	int lazy[1<<19];
	segtree(){
		memset(dp,0,sizeof(dp));
		for(int i=0;i<N*2;i++){
			dp[i]=11451419;
			lazy[i]=11451419;
		}
	}

	void lazy_eval(int k){
		dp[k]=min(dp[k],lazy[k]);
		if(k<N-1){
			lazy[k*2+1]=min(lazy[k*2+1],lazy[k]);
			lazy[k*2+2]=min(lazy[k*2+2],lazy[k]);
		}
	}

	void lazy_upd(int k){
		dp[k]=min(dp[k*2+1],dp[k*2+2]);
	}
	void update(int a,int b,int v,int k=0,int l=0,int r=N){
		lazy_eval(k);
		if(b<=l || r<=a)return;
		if(a<=l && r<=b){
			lazy[k]=min(lazy[k],v);
			lazy_eval(k);
			return;
		}
		int mid=(l+r)/2;
		update(a,b,v,k*2+1,l,mid);
		update(a,b,v,k*2+2,mid,r);
		lazy_upd(k);
	}

	int query(int a,int b,int k=0,int l=0,int r=N){
		lazy_eval(k);
		if(b<=l || r<=a)return 11451419;
		if(a<=l && r<=b)return dp[k];
		int mid=(l+r)/2;
		int vl=query(a,b,k*2+1,l,mid);
		int vr=query(a,b,k*2+2,mid,r);
		lazy_upd(k);
		return min(vl,vr);
	}
};

int n;
set<int> g[100005];
set<int> gg[100005];
segtree seg;
bool self[100005];

int main(void){
	scanf("%d",&n);
	int selfc=0;
	int pr=0;
	for(int i=0;i<n;i++){
		int m;
		scanf("%d",&m);
		bool flag=false;
		for(int j=0;j<m;j++){
			int x;
			scanf("%d",&x);
			x--;
			if(x==i){
				self[i]=true;
				g[i].insert(x);
			}else if(x==i+1){
				flag=true;
			}else if(x>=pr && !self[i]){
				g[i].insert(x);
			}
		}
		if(!flag){
			pr=i+1;
		}
	}
	for(int i=0;i<n;i++){
		for(set<int>::iterator it=g[i].begin();it!=g[i].end();it++){
			int v=*it;
			gg[v].insert(i);
		}
	}
	int mini_m=11451419;
	for(int i=n-1;i>=0;i--){
		for(set<int>::iterator it=gg[i].begin();it!=gg[i].end();it++){
			int v=*it;
			mini_m=min(v,mini_m);
		}
		if(*(gg[i].begin())>mini_m){
			gg[i].insert(mini_m);
		}
	}
	seg.update(0,1,0);
	int prv=0;
	int cnt=0;
	for(int i=0;i<n;i++){
		if(gg[i].size()==0)continue;
		int v=seg.query(prv,prv+1);
		int nxt=*(gg[i].begin());
		//printf("%d %d %d\n",i,nxt,v);
		seg.update(i+1,nxt+2,v+1);
		prv=i+1;
	}
	printf("%d\n",seg.query(prv,prv+1));
	return 0;
}
