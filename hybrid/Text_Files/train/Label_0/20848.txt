#include <bits/stdc++.h>
using namespace std;
const int MAXN = 300;
bool vis[MAXN][2][MAXN][MAXN];
class state {
 public:
  int x, dir, used, pos;
  state(int _x, int _d, int _l, int _p) {
    x = _x;
    used = _l;
    pos = _p;
    dir = _d;
  }
};
void mpush(queue<state> &q, int x, int y, int z, int w) {
  if (vis[x][y][z][w]) return;
  vis[x][y][z][w] = true;
  q.push(state(x, y, z, w));
}
int main() {
  ios::sync_with_stdio(false);
  cout.tie(0);
  cin.tie(0);
  int t;
  string s;
  cin >> s >> t;
  int n = s.size();
  queue<state> q;
  mpush(q, 150, 1, 0, 0);
  int ans = 0;
  while (q.size()) {
    state u = q.front();
    q.pop();
    if (u.pos == n && (t - u.used) % 2 == 0) {
      ans = max(ans, abs(150 - u.x));
      continue;
    } else if (u.pos == n)
      continue;
    if (u.used < t) {
      if (s[u.pos] == 'F')
        mpush(q, u.x, 1 - u.dir, u.used + 1, u.pos + 1);
      else if (u.dir == 1)
        mpush(q, u.x + 1, u.dir, u.used + 1, u.pos + 1);
      else
        mpush(q, u.x - 1, u.dir, u.used + 1, u.pos + 1);
    }
    if (s[u.pos] == 'T')
      mpush(q, u.x, 1 - u.dir, u.used, u.pos + 1);
    else if (u.dir == 1)
      mpush(q, u.x + 1, u.dir, u.used, u.pos + 1);
    else
      mpush(q, u.x - 1, u.dir, u.used, u.pos + 1);
  }
  cout << ans << endl;
}
