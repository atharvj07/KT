import sys
import math
import itertools
TESTING = False

def helper(length, knum):
	if length <= 1:
		return [1]
	possibilties = pow(2, length - 1)
	if knum <= possibilties / 2:
		temp = helper(length - 1, knum)
		for i in range(len(temp)):
			temp[i] += 1
		return [1] + temp
	res = helper(length, possibilties - knum + 1)
	return res[::-1]

def solve():
	(n, m) = read()
	return helper(n, m)

def read(mode=2):
	inputs = input().strip()
	if mode == 0:
		return inputs
	if mode == 1:
		return inputs.split()
	if mode == 2:
		return list(map(int, inputs.split()))

def write(s='\n'):
	if s is None:
		s = ''
	if isinstance(s, list):
		s = ' '.join(map(str, s))
	if isinstance(s, tuple):
		s = ' '.join(map(str, s))
	s = str(s)
	print(s, end='')

def run():
	if TESTING:
		sys.stdin = open('test.txt')
	res = solve()
	write(res)
run()
