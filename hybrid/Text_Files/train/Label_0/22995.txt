int main() {
	int h, w; cin >> h >> w;

	char c[50][50];

	vector<pair<int, int> > v;

	for (int i = 0; i < h; i++) {
		for (int j = 0; j < w; j++) {
			cin >> c[i][j];
			if (c[i][j] == 'B') { v.push_back(make_pair(i, j)); }
		}
	}
	int ans = 0;
	for (auto x : v) {
		int s = x.first;
		int t = x.second;
		for (auto y : v) {
			int f = y.first;
			int g = y.second;
			if (s == f && t == g) { }
			else {
				ans = max(ans, abs(s - f) + abs(t - g));
			}
		}
	}

	cout << ans << endl;


	return 0;
}
