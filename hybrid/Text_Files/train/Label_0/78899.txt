def d4c(lab, n):
	for i in range(n):
		for j in range(n):
			x = lab[i][j]
			if x == 1:
				continue
			f = 0
			for k in range(n):
				if k != i:
					for l in range(n):
						if l != j:
							if lab[i][l] + lab[k][j] == x:
								f = 1
								continue
			if f != 1:
				return 'No'
	return 'Yes'
n = int(input())
lab = []
for _ in range(n):
	lab.append(list(map(int, input().split())))
print(d4c(lab, n))
