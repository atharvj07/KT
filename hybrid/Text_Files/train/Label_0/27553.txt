#include <iostream>

using namespace std;

int main() {
#ifndef LOCAL
  cin.tie(0);
#endif
  ios::sync_with_stdio(false);
  int tt;
  cin >> tt;
  while (tt--) {
    long long d;
    int k;
    cin >> d >> k;
    bool can = false;
    {
      int t = 0;
      long long a = 0, b = 0;
      while (a * a + b * b <= d * d) {
        if (t % 2 == 0) {
          a += k;
        } else {
          b += k;
        }
        ++t;
      }
      if ((t - 1) % 2 == 0) {
        can = true;
      }
    }
    {
      int t = 0;
      long long a = 0, b = 0;
      while (a * a + b * b <= d * d) {
        if (t == 0 || t == 1) {
          a += k;
        } else
        if (t % 2 == 0) {
          b += k;
        } else {
          a += k;
        }
        ++t;
      }
      if ((t - 1) % 2 == 0) {
        can = true;
      }
    }
    cout << (can ? "Utkarsh" : "Ashish") << '\n';
  }
}
