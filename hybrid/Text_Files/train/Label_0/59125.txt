import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws FileNotFoundException {
        //System.setIn(new FileInputStream("test.txt"));
        Scanner sc = new Scanner(System.in);
       
        String kvenka = sc.next();
        String wovel = "IEAOUY";
        int n = kvenka.length();
        
        double result = 0.0;
        double dod = 0.0;
        double add = 0.0;
        double rem = 0.0;
        for (int i = 1; i <= n; i++) {
            dod += 1.0 / i;
            add += 1.0 / i;
        }
        for (int i = 0; i < n; i++) {
            char tmp = kvenka.charAt(i);
            if (tmp == 'I' || tmp == 'E' || tmp == 'A' || tmp == 'O' 
                || tmp == 'U' || tmp == 'Y') 
            {
                result += dod;
            }
            rem += (1.0 / (i+1));
            add -= (1.0 / (n-i));
            dod += add;
            dod -= rem;
        }
        System.out.println(result);
    }
    
}
