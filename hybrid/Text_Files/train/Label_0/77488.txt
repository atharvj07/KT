for _ in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	l = sorted(l)
	mex = -1
	for i in range(len(l)):
		if l[i] != i:
			mex = i
			break
	if mex == -1:
		mex = n
	mx = max(l)
	x = mx + mex
	if x % 2 == 0:
		x = x // 2
	else:
		x = x // 2 + 1
	if x == mx:
		print(len(l))
	elif x == mex:
		print(k + len(l))
	else:
		y = len(l)
		if x not in l and k > 0:
			y += 1
		print(y)
