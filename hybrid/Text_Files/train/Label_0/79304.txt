from copy import deepcopy
from collections import Counter

def find_target(a, n):
	ca = Counter(deepcopy(a))
	val = list(sorted(ca.keys(), key=lambda x: ca[x]))[-1]
	for i in range(n):
		if a[i] == val:
			return i

def solve(a, ind, n):
	res = []
	for i in range(ind + 1, n):
		if a[i] > a[i - 1]:
			res.append((2, i + 1, i))
		elif a[i] < a[i - 1]:
			res.append((1, i + 1, i))
		a[i] = a[ind]
	for i in range(ind - 1, -1, -1):
		if a[i] > a[i + 1]:
			res.append((2, i + 1, i + 2))
		elif a[i] < a[i + 1]:
			res.append((1, i + 1, i + 2))
		a[i] = a[ind]
	return res

def show(res):
	print(len(res))
	for (op, i, j) in res:
		print(op, i, j)

def main():
	n = int(input())
	a = list(map(int, input().split()))
	show(solve(a, find_target(a, n), n))
main()
