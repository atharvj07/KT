n = 14
a = None

def checkCase(defList, i):
	global n
	a = list(defList)
	stone = a[i] % n
	stoneAll = a[i] // n
	a[i] = 0
	j = i + 1
	while stone > 0:
		if j >= n:
			j = 0
		a[j] += 1
		j += 1
		stone -= 1
	pickedUp = 0
	for k in range(n):
		a[k] += stoneAll
		if a[k] % 2 == 0:
			pickedUp += a[k]
	return pickedUp

def main():
	global n, a
	a = [int(e) for e in input().split()]
	maxStones = 0
	for i in range(n):
		if a[i] > 0:
			maxStones = max(maxStones, checkCase(a, i))
	print(maxStones)
main()
