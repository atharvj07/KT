#include<iostream>
#include<algorithm>
using namespace std;
int main()
{
	long n,a[100005],b[100005],sa=0,sb=0,sm=0,ans=0;
	cin>>n;
	for(int i=0;i<n;i++)cin>>a[i],sa+=a[i];
	for(int i=0;i<n;i++)cin>>b[i],sb+=b[i];
	if(sa<sb){cout<<-1<<endl;return 0;}
	for(int i=0;i<n;i++)a[i]-=b[i];
	sort(a,a+n);
	if(a[0]>=0){cout<<0<<endl;return 0;}
	for(int i=0;i<n;i++){if(a[i]<0){sm+=a[i]*(-1);ans++;}}
	for(int i=n-1;i>=0;i--)
	{
		sm-=a[i];
		ans++;
		if(sm<=0)
		{
			cout<<ans<<endl;
			return 0;
		}
	}
}
