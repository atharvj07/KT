import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.List;

public class P322_2E implements Runnable {
  int nextInt(StreamTokenizer in) throws Exception {
    in.nextToken();
    return (int) in.nval;
  }

  StreamTokenizer in;

  private int nextInt() throws Exception {
    in.nextToken();
    return (int) in.nval;
  }

  int n;
  int t;
  List<List<Integer>> g;
  int[][][] res;

  private int find(int v, int prev) {
    List<Integer> gg = g.get(v);
    int[] f = new int[gg.size()];
    int cc = 0;
    for (Integer vv : gg) {
      if (vv == prev) continue;
      f[cc++] = find(vv, v);
    }
    int[][] r = init();
    if (gg.size() > 1) {
      r[0][0] = 0;
      r[1][0] = 0;
      cc = 0;
      int nn = 0;
      for (Integer vv : gg) {
        if (vv == prev) continue;
        int[][] rr = init();
        for (int c = 0; c < 2; c++) {
          for (int num = 0; num <= nn; num++) {
            for (int cv = 0; cv < 2; cv++) {
              int dif = Math.abs(c - cv);
              for (int nv = 0; nv <= f[cc] && nv + num <= t; nv++) {
                rr[c][num + nv] = Math.min(rr[c][num + nv], r[c][num] + res[cv][vv][nv] + dif);
              }
            }
          }
        }
        r = rr;
        nn += f[cc];
        cc++;
      }
    } else {
      r[0][1] = 0;
      r[1][0] = 0;
    }
    for (int c = 0; c < 2; c++) {
      System.arraycopy(r[c], 0, res[c][v], 0, t);
    }
    if (gg.size() == 1) {
      return 1;
    } else {
      int res = 0;
      for (int i = 0; i < f.length; i++) {
        res += f[i];
      }
      return res;
    }
  }

  private int[][] init() {
    int[][] r = new int[2][t + 1];
    for (int c = 0; c < 2; c++) {
      for (int i = 0; i <= t; i++) {
        r[c][i] = Integer.MAX_VALUE / 3;
      }
    }
    return r;
  }

  @Override
  public void run() {
    try {
      in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
      n = nextInt();
      g = new ArrayList<>();
      for (int i = 0; i < n; i++) {
        g.add(new ArrayList<Integer>());
      }
      for (int i = 0; i < n - 1; i++) {
        int a = nextInt() - 1;
        int b = nextInt() - 1;
        g.get(a).add(b);
        g.get(b).add(a);
      }
      if (n == 2) {
        System.out.println(1);
        return;
      }
      int first = -1;
      for (int i = 0; i < n; i++) {
        if (g.get(i).size() > 1) {
          first = i;
        } else {
          t++;
        }
      }
      res = new int[2][n][t + 1];
      find(first, -1);
      System.out.println(res[0][first][t / 2]);
    } catch (Exception e) {
      throw new RuntimeException(e);
    }
  }

  public static void main(String[] args) {
    new Thread(new P322_2E()).start();
  }
}
