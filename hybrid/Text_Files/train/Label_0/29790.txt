#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 20;
int n, ar[maxn], cnt = 1, trie[maxn][27], nxt[maxn][27], ok[maxn], f[maxn],
                 sum[maxn];
long long ans, am[maxn], a[maxn];
vector<int> order, vec[maxn], comp;
int add(string &s) {
  int i = 0;
  for (char c : s) {
    if (trie[i][c - 'a'] == 0) trie[i][c - 'a'] = cnt++;
    i = trie[i][c - 'a'];
  }
  ok[i]++;
  return i;
}
void aho() {
  queue<int> q;
  q.push(0);
  while (!q.empty()) {
    int x = q.front();
    q.pop();
    order.push_back(x);
    am[x] = am[f[x]] + ok[x];
    for (int j = 0; j < 27; j++) {
      if (trie[x][j] == 0)
        nxt[x][j] = nxt[f[x]][j];
      else {
        f[trie[x][j]] = nxt[f[x]][j];
        nxt[x][j] = trie[x][j];
        q.push(trie[x][j]);
      }
    }
  }
}
string t, s;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> t >> n;
  for (int i = 1; i <= n; i++) {
    cin >> s;
    ar[i] = add(s);
    vec[s.size()].push_back(i);
    comp.push_back(s.size());
  }
  sort(comp.begin(), comp.end());
  comp.resize(unique(comp.begin(), comp.end()) - comp.begin());
  aho();
  int state = 0;
  for (int i = 0; i < t.size(); i++) {
    state = nxt[state][t[i] - 'a'];
    sum[i] = am[state];
  }
  memset(ok, 0, sizeof ok);
  for (int qu : comp) {
    for (int x : vec[qu]) ok[ar[x]]++;
    for (int x : order) a[x] = a[f[x]] + ok[x];
    state = 0;
    for (int i = 0; i < t.size(); i++) {
      state = nxt[state][t[i] - 'a'];
      if (i >= qu) ans += 1ll * sum[i - qu] * a[state];
    }
    for (int x : vec[qu]) ok[ar[x]]--;
  }
  cout << ans << endl;
}
