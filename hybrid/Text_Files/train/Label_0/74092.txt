import sys
read = sys.stdin.readline
for i in range(int(read())):
	(n, H) = map(int, read().split())
	arr = list(reversed(sorted(list(map(int, read().split())))))
	if arr[0] > H:
		print(1)
		continue
	temp = arr[0] + arr[1]
	req = H - H % temp
	ans = req // temp * 2
	if req == H:
		print(ans)
	elif req + arr[0] >= H:
		print(ans + 1)
	else:
		print(ans + 2)
