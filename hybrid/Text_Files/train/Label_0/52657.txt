import java.util.*;
import java.io.*;

public class InfinitePath {
    public static void main(String[] args) throws IOException{
        BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        int t = Integer.parseInt(f.readLine());
        while(t --> 0){
            int n = Integer.parseInt(f.readLine());
            int[] arr = new int[n+1];
            int[] colors = new int[n+1];
            StringTokenizer st1 = new StringTokenizer(f.readLine());
            StringTokenizer st2 = new StringTokenizer(f.readLine());
            for(int i = 1; i <= n; i++){
                arr[i] = Integer.parseInt(st1.nextToken());
                colors[i] = Integer.parseInt(st2.nextToken());
            }
            ArrayList<ArrayList<Integer>> cycles = new ArrayList<ArrayList<Integer>>();
            boolean[] seen = new boolean[n+1];
            for(int i = 1; i <= n; i++){
                if(seen[i]) continue;
                cycles.add(new ArrayList<Integer>());
                int temp = i;
                cycles.get(cycles.size()-1).add(temp);
                temp = arr[temp];
                while(temp != i){
                    cycles.get(cycles.size()-1).add(temp);
                    seen[temp] = true;
                    temp = arr[temp];
                }
            }
            int minans = Integer.MAX_VALUE;
            for(ArrayList<Integer> curr: cycles){
                ArrayList<Integer> divisors = divisors(curr.size());
                for(int div: divisors){
                    boolean good = false;
                    for(int start = 0; start < div; start++) {
                        boolean good2 = true;
                        int col = colors[curr.get(start)];
                        for (int i = div+start; i < curr.size()+start; i += div) {
                            if (colors[curr.get(i%curr.size())] != col) {
                                good2 = false;
                                break;
                            }
                        }
                        if(good2){
                            good = true;
                            break;
                        }
                    }
                    if(good){
                        minans = Math.min(minans, div);
                        break;
                    }
                }
            }
            out.println(minans);
        }
        out.close();
    }public static ArrayList<Integer> divisors(int n){
        ArrayList<Integer> ans = new ArrayList<Integer>();
        for(int i = 1; i <= Math.sqrt(n)+0.000001; i++){
            if(n % i == 0){
                ans.add(i);
                if(n/i != i) ans.add(n/i);
            }
        }
        Collections.sort(ans);
        return ans;
    }
}
