import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class C {
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
    public static FastReader scn = new FastReader();

    private static class node implements Comparable<node>{
        int value;
        int ind;

        @Override
        public int compareTo(node o) {
            if(this.value==o.value) return this.ind - o.ind;
            return this.value - o.value;
        }
    }

    public static void main(String[] args) {
        int n = scn.nextInt();
        int m = scn.nextInt();
        int d = scn.nextInt();

        PriorityQueue<node> pq = new PriorityQueue<>();

        for(int i = 0;i < n; i++){
            int a = scn.nextInt();
            node nn = new node();
            nn.value = a;
            nn.ind = i;
            pq.add(nn);
        }

        node[] arr = new node[n];
        int[] res = new int[n];
        int i = 0;
        while(!pq.isEmpty()) arr[i++] = pq.poll();

        i = 0;
        int j = 1;

        int count = 1;
        while(i<arr.length){
            if(res[i]==0) {
                res[i] = count;
                count++;
            }
            while(j<arr.length){
                if(arr[j].value-arr[i].value>d){
                    res[j] = res[i];
                    j++;
                    break;
                }
                j++;
            }
            i++;
        }

        int[] rr = new int[n];

        for(int ii = 0; ii<n; ii++){
            rr[arr[ii].ind] = res[ii];
        }

        StringBuilder sb = new StringBuilder();
        for(int ii = 0;ii<n; ii++){
            if(rr[ii]!=0){
                sb.append(rr[ii] +" ");
            }
        }
        System.out.println(count-1);
        System.out.println(sb);
    }
}
