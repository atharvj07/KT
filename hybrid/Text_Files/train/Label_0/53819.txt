import java.util.*;
import java.lang.*;
import java.io.*;
import java.math.*;

public class file {

    public static void main(String[] args) throws Exception {

        Scanner sc=new Scanner(System.in);
        PrintWriter pw=new PrintWriter(System.out);

        int n=sc.nextInt();
        int k=sc.nextInt();

        int[] a=new int[n];
        int mi=Integer.MAX_VALUE;
        PriorityQueue<Integer> pq=new PriorityQueue<>(Collections.reverseOrder());
        for(int i=0; i<n; i++) {

            a[i]=sc.nextInt();

            if(i<=k)
                pq.add(a[i]);

            else {
    
                pq.add(a[i]);
                int tmp=pq.poll();
            }
        }

        if(k==0) {

            while(!pq.isEmpty())
                mi=pq.poll();

            if(mi==1)
                pw.println("-1");

            else
                pw.println(mi-1);
        }

        else {

            mi=pq.poll();
            int mi1=-1;

            if(!pq.isEmpty())
                mi1=pq.poll();

            if(k==n)
                pw.println(mi);

            else {

                if(mi1==mi)
                    pw.println("-1");

                else
                    pw.println(mi1);
            }
        }

	    pw.close();
	}
}
