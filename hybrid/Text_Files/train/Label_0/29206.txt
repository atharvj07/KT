#include<stdio.h>
#include<algorithm>
using namespace std;
#define SZ 1024
char p[1010][1010];
int n, m, K, IT[SZ + SZ + 1];
int Max(int b, int e){
	int r = -9999999;
	b += SZ, e += SZ;
	while (b <= e){
		r = max(r, IT[b]);
		r = max(r, IT[e]);
		b = (b + 1) >> 1;
		e = (e - 1) >> 1;
	}
	return r;
}
void Add(int a, int b){
	a += SZ;
	while (a){
		IT[a] = max(IT[a], b);
		a >>= 1;
	}
}
int main(){
	int i, j, ck;
	scanf("%d%d%d", &n, &m, &K);
	for (i = 1; i <= n; i++){
		scanf("%s", p[i] + 1);
	}
	for (i = 1; i < SZ * 2; i++)IT[i] = -9999999;
	for (i = 1; i <= n; i++){
		for (j = 1; j <= m; j++){
			ck = 0;
			if (p[i][j] != '.')continue;
			if (i + j - Max(1, j) > K){
				ck = 1;
				Add(j, i + j);
			}
		}
	}
	if (ck == 0)printf("Chieno\n");
	else printf("Cacao\n");
}
