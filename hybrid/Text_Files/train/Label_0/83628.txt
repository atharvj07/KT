def solve(A, N):
	adj = set()
	m = 0
	for i in range(N - 1):
		if A[i] != -1 and A[i + 1] != -1:
			m = max(m, abs(A[i + 1] - A[i]))
		if A[i] == -1 and A[i + 1] != -1:
			adj.add(A[i + 1])
		if A[i] != -1 and A[i + 1] == -1:
			adj.add(A[i])
	if not adj:
		return '0 0'
	maxVal = max(adj)
	minVal = min(adj)
	k = (maxVal + minVal) // 2
	m = max(m, abs(maxVal - k), abs(minVal - k))
	return str(m) + ' ' + str(k)
(T,) = map(int, input().split())
for t in range(T):
	(N,) = map(int, input().split())
	A = [int(x) for x in input().split()]
	ans = solve(A, N)
	print(ans)
