def main():
	t = int(input())
	for i in range(t):
		n = int(input())
		arr = list(map(int, input().split()))
		total = 0
		ones = 0
		twos = 0
		for j in range(n):
			arr[j] = arr[j] % 3
			if arr[j] == 0:
				total += 1
			elif arr[j] == 1:
				ones += 1
			else:
				twos += 1
		curr = min(ones, twos)
		total += curr
		ones -= curr
		twos -= curr
		if ones > 0:
			total += ones // 3
		else:
			total += twos // 3
		print(total)
main()
