import sys

def input():
	return sys.stdin.readline().rstrip()
DXY = [(0, -1), (1, 0), (0, 1), (-1, 0)]

def main():
	t = int(input())
	for _ in range(t):
		n = int(input())
		a = [list(map(int, input().split())) for i in range(2)]
		a[0].reverse()
		b = [[0] * (n + 1) for i in range(2)]
		for i in range(1, n + 1):
			b[0][i] = b[0][i - 1] + a[0][i - 1]
			b[1][i] = b[1][i - 1] + a[1][i - 1]
		ans = min((max(b[0][i], b[1][n - 1 - i]) for i in range(n)))
		print(ans)
	return 0
main()
