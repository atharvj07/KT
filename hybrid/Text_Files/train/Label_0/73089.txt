def NextR(c):
	if c == 'v':
		return '<'
	if c == '<':
		return '^'
	if c == '^':
		return '>'
	if c == '>':
		return 'v'

def NextL(c):
	if c == 'v':
		return '>'
	if c == '>':
		return '^'
	if c == '^':
		return '<'
	if c == '<':
		return 'v'
(c1, c2) = input().split()
n = int(input())
nn = n % 4
cr = cl = c1
for i in range(nn):
	cl = NextR(cl)
for i in range(nn):
	cr = NextL(cr)
if cl == cr:
	print('undefined')
	raise SystemExit
if cl != c2 and cr != c2:
	print('undefined')
	raise SystemExit
if cl == c2:
	print('cw')
else:
	print('ccw')
