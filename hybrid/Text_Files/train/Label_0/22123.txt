#include <bits/stdc++.h>
using namespace std;
long long dp[1002][4][2004];
int main(void) {
  long long i, j;
  long long n, k;
  cin >> n >> k;
  dp[0][0][1] = dp[0][1][2] = dp[0][2][2] = dp[0][3][1] = 1;
  long long N, K;
  for (N = 1; N < n; N++) {
    for (K = 1; K < k + 1; K++) {
      dp[N][0][K] = dp[N - 1][0][K] + dp[N - 1][1][K] + dp[N - 1][2][K] +
                    ((K > 0) * dp[N - 1][3][K - 1]);
      dp[N][1][K] = ((K > 1) * dp[N - 1][0][K - 1]) + dp[N - 1][1][K] +
                    ((K > 1) * dp[N - 1][3][K - 1]) +
                    ((K > 1) * dp[N - 1][2][K - 2]);
      dp[N][2][K] = ((K > 1) * dp[N - 1][0][K - 1]) + dp[N - 1][2][K] +
                    ((K > 1) * dp[N - 1][3][K - 1]) +
                    ((K > 1) * dp[N - 1][1][K - 2]);
      dp[N][3][K] = dp[N - 1][3][K] + dp[N - 1][1][K] + dp[N - 1][2][K] +
                    ((K > 0) * dp[N - 1][0][K - 1]);
      for (int x = 0; x < 4; x++) dp[N][x][K] %= 998244353;
    }
  }
  cout << (dp[n - 1][0][k] + dp[n - 1][1][k] + dp[n - 1][2][k] +
           dp[n - 1][3][k]) %
              998244353
       << "\n";
  return 0;
}
