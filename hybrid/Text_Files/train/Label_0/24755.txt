#include <bits/stdc++.h>
using namespace std;
char str[1000];
int main() {
  int n, k;
  while (scanf("%d %d", &n, &k) == 2) {
    scanf("%s", str + 1);
    int left = k - 1;
    int right = n - k;
    int di;
    if (right >= left) {
      di = 1;
      for (int i = k; i > 1; i--) printf("LEFT\n");
    } else {
      di = 2;
      for (int i = k; i < n; i++) printf("RIGHT\n");
    }
    if (di == 1) {
      for (int i = 1; i <= n; i++) {
        printf("PRINT %c\n", str[i]);
        if (i != n) printf("RIGHT\n");
      }
    } else {
      for (int i = n; i >= 1; i--) {
        printf("PRINT %c\n", str[i]);
        if (i != 1) printf("LEFT\n");
      }
    }
    cout << endl;
  }
  return 0;
}
