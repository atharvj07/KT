#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
const int MAXN = 500000;
int n;
int a[MAXN];
int b[MAXN];
int top[MAXN], ntop;
bool done[MAXN];
int ret[MAXN];
int sval[4 * MAXN];
int sbest(int p, int q) {
  int ret = p == -1       ? q
            : q == -1     ? p
            : b[p] == -1  ? p
            : b[q] == -1  ? q
            : b[p] > b[q] ? p
                          : q;
  return ret;
}
void sinit(int x, int l, int r) {
  if (l == r) {
    sval[x] = l;
    return;
  }
  int m = l + (r - l) / 2;
  sinit(2 * x + 1, l, m);
  sinit(2 * x + 2, m + 1, r);
  sval[x] = sbest(sval[2 * x + 1], sval[2 * x + 2]);
}
int sfind(int x, int l, int r, int L, int R) {
  if (L <= l && r <= R) return sval[x];
  int m = l + (r - l) / 2;
  int p = L <= m ? sfind(2 * x + 1, l, m, L, R) : -1;
  int q = m + 1 <= R ? sfind(2 * x + 2, m + 1, r, L, R) : -1;
  return sbest(p, q);
}
void skill(int x, int l, int r, int IDX) {
  if (l == r) {
    sval[x] = -1;
    return;
  }
  int m = l + (r - l) / 2;
  if (IDX <= m)
    skill(2 * x + 1, l, m, IDX);
  else
    skill(2 * x + 2, m + 1, r, IDX);
  sval[x] = sbest(sval[2 * x + 1], sval[2 * x + 2]);
}
int findedge(int i) {
  if (b[i] != -1 && !done[b[i]]) return b[i];
  vector<int> bounds;
  bounds.push_back(-1);
  bounds.push_back(a[i] == -1 ? n : a[i]);
  if (-1 < i && (a[i] == -1 || i < a[i])) bounds.push_back(i);
  sort(bounds.begin(), bounds.end());
  for (int x = (1); x < (((int)(bounds).size())); ++x) {
    int l = bounds[x - 1] + 1, r = bounds[x] - 1;
    if (l > r) continue;
    int ret = sfind(0, 0, n - 1, l, r);
    if (ret != -1 && (b[ret] == -1 || b[ret] > i)) return ret;
  }
  return -1;
}
void dfs(int at) {
  assert(!done[at]);
  done[at] = true;
  skill(0, 0, n - 1, at);
  while (true) {
    int to = findedge(at);
    if (to == -1)
      break;
    else
      dfs(to);
  }
  top[ntop++] = at;
}
void run() {
  scanf("%d", &n);
  for (int i = (0); i < (n); ++i) {
    scanf("%d", &a[i]);
    if (a[i] != -1) --a[i];
  }
  for (int i = (0); i < (n); ++i) b[i] = -1;
  for (int i = (0); i < (n); ++i)
    if (a[i] != -1) b[a[i]] = i;
  for (int i = (0); i < (n); ++i) done[i] = false;
  ntop = 0;
  sinit(0, 0, n - 1);
  for (int i = (0); i < (n); ++i)
    if (!done[i]) dfs(i);
  for (int i = (0); i < (n); ++i) ret[top[i]] = i;
  for (int i = (0); i < (ntop); ++i) {
    if (i != 0) printf(" ");
    printf("%d", ret[i] + 1);
  }
  puts("");
}
int main() {
  run();
  return 0;
}
