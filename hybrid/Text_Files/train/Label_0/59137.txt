import java.io.BufferedWriter;
import java.io.File;
import java.io.IOException;
import java.io.OutputStreamWriter;
import java.util.Scanner;

public class A1341 {
	public static void main(String[] args) throws IOException{
		Scanner scanner = new Scanner(System.in);
		BufferedWriter log = new BufferedWriter(new OutputStreamWriter(System.out));
		int cases = scanner.nextInt();
		for(int i = 0; i < cases; i++) {
			int amount = scanner.nextInt();
			int high = scanner.nextInt();
			int low = scanner.nextInt();
			int max = (high + low) * amount;
			int min = (high - low) * amount;
			int totalMax = scanner.nextInt();
			int totalMin = scanner.nextInt();
			if(intersect(min, max, totalMax - totalMin, totalMax + totalMin))
				log.write("YES\n");
			else
				log.write("NO\n");
		}
		log.flush();
	}

	private static boolean intersect(int min, int max, int totalMin, int totalMax) {
		if(max < totalMin || min > totalMax)
			return false;
		return true;
	}		
}
