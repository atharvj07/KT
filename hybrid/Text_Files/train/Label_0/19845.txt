#include <bits/stdc++.h>
using namespace std;
const int N = 5 * 1E5 + 10;
bool vis[N] = {false};
int a[N];
int need[N] = {0};
int cnt[N] = {0};
int ans[N];
int main() {
  int n, k, m, s;
  scanf("%d%d%d%d", &n, &k, &m, &s);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  int len = n - k * (m - 1);
  int del = n - k * m;
  int type = 0;
  for (int i = 1; i <= s; i++) {
    int v;
    scanf("%d", &v);
    if (!need[v]) type++, vis[v] = true;
    ++need[v];
  }
  bool flag = false;
  int L = 1;
  for (int i = 1; i <= len; i++) {
    int v = a[i];
    cnt[v]++;
    if (vis[v] && cnt[v] == need[v]) type--;
    if (type == 0) {
      L = 1;
      flag = true;
      break;
    }
  }
  if (!flag)
    for (int i = len + 1; i <= n; i++) {
      int l = a[L];
      int r = a[i];
      L++;
      cnt[l]--;
      if (vis[l]) {
        if (cnt[l] == need[l] - 1) type++;
      }
      cnt[r]++;
      if (vis[r]) {
        if (cnt[r] == need[r]) type--;
      }
      if (type == 0 && (L - 1) % k == 0) {
        flag = true;
        break;
      }
    }
  int num = 0;
  if (flag) {
    for (int i = L; i < len + L; i++) {
      int v = a[i];
      if (need[v] > 0) {
        need[v]--;
        continue;
      }
      if (num == del) break;
      ans[++num] = i;
    }
    printf("%d\n", num);
    for (int i = 1; i <= num; i++) {
      printf("%d%c", ans[i], i == num ? '\n' : ' ');
    }
  } else
    puts("-1");
  return 0;
}
