const int MAX_V=100;

stack<int> roots,S;

int cmp[MAX_V];
vector<int> cG[MAX_V];
int n[MAX_V];
void vis(int v,int p){
	ord[v]=++cnt;
	S.push(v);
	inS[v]=true;
	roots.push(v);
	for(int u:G[v]){
		if(ord[u]==0){
			vis(u,v);
		}else if(u!=p&&inS[u]){
			while(ord[roots.top()]>ord[u]) roots.pop();
		}
	}
	if(v==roots.top()){
		bs.pb(P(p,v));
		vector<int> vc;
		while(true){
			int w=S.top();S.pop();
			inS[w]=false;
			vc.pb(w);
			cmp[w]=cc.size();
			if(v==w) break;
		}
		roots.pop();
		n[cc.size()]=vc.size();
		cc.pb(vc);
	}
}
void bridge(){
	rep(i,N) if(ord[i]==0){
		vis(i,-1);
		bs.pop_back();	//P(-1,hoge)
	}
	for(P p:bs){
		int x=cmp[p.fs],y=cmp[p.sc];
		cG[x].pb(y),cG[y].pb(x);
	}
}

int sz[MAX_V];
int u[MAX_V][MAX_V+1],d[MAX_V][MAX_V+1];
int t[MAX_V+1][MAX_V+1],nt[MAX_V+1][MAX_V+1];
int inf=1e8;
void dfs(int v,int p){
	for(int w:cG[v]) if(w!=p) dfs(w,v);
	rep(i,101) rep(j,101) t[i][j]=nt[i][j]=-inf;
	t[0][0]=0;
	for(int w:cG[v]){
		if(w==p) continue;
		rep(i,sz[v]+1) rep(j,sz[v]+1){
			rep(k,sz[w]+1){
				chmax(nt[i+k][j],t[i][j]+u[w][k]);
				chmax(nt[i][j+k],t[i][j]+d[w][k]);
			}
		}
		sz[v]+=sz[w];
		rep(i,100+1) rep(j,100+1) t[i][j]=nt[i][j],nt[i][j]=-inf;
	}
	rep(i,sz[v]+1) rep(j,sz[v]+1){
		chmax(u[v][i+n[v]],t[i][j]+(i+n[v])*(j+n[v]));
		chmax(d[v][j+n[v]],t[i][j]+(i+n[v])*(j+n[v]));
	}
