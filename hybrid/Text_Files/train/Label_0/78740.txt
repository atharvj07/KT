from math import inf as inf
import sys
input = sys.stdin.readline
q = int(input())
for i in range(q):
	n = int(input())
	l = []
	dp = [[inf, inf, inf] for i in range(n)]
	for i in range(n):
		(a, b) = map(int, input().split())
		l.append([a, b])
	dp[0] = [0, l[0][1], l[0][1] * 2]
	for i in range(1, n):
		for j in range(3):
			for k in range(3):
				if l[i][0] + j != l[i - 1][0] + k:
					dp[i][j] = min(dp[i][j], dp[i - 1][k] + j * l[i][1])
	print(min(dp[-1]))
