#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
bool isPrime(long long n) {
  long long c;
  if (n <= 1) return false;
  c = sqrt(n);
  for (int i = 2; i <= c; i++)
    if (n % i == 0) return false;
  return true;
}
