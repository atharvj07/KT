#include <bits/stdc++.h>
using namespace std;
int a[1200000];
int t[1200000], c[1200000];
int main() {
  int n;
  while (scanf("%d", &n) != EOF) {
    t[0] = 0;
    c[0] = 0;
    for (int i = 1; i <= n; i++) {
      cin >> t[i];
      c[i] = c[i - 1] + 20;
      if (t[i] - t[i - 1] <= 90) {
        int l = upper_bound(t + 1, t + i + 1, (t[i] - 90)) - t - 1;
        c[i] = min(c[i], c[l] + 50);
      }
      if (t[i] - t[i - 1] <= 1440) {
        int l = upper_bound(t + 1, t + i + 1, (t[i] - 1440)) - t - 1;
        c[i] = min(c[i], c[l] + 120);
      }
    }
    for (int i = 1; i <= n; i++) {
      cout << c[i] - c[i - 1] << endl;
    }
  }
}
