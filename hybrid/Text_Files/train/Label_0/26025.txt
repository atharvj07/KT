#include<bits/stdc++.h>
#define rep(i,x,y) for (int i=(x);i<=(y);i++)

using namespace std;

const int N=111;
int n,m,a[N][N],cnt;

int main(){
	scanf("%d",&n);
	rep (i,1,n) rep (j,1,i-1) a[i][j]=1;
	m=n+!(n&1);
	rep (i,1,n) a[i][m-i]=0;
	rep (i,1,n) rep (j,1,i-1) cnt+=a[i][j];
	printf("%d\n",cnt);
	rep (i,1,n) rep (j,1,i-1) if (a[i][j]) printf("%d %d\n",i,j);
	return 0;
}
