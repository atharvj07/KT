import sys
(n, k, d) = map(int, input().split())
mod = 10 ** 9 + 7
if d > n:
	print(0)
	sys.exit(0)
c = [0] * (n + 1)
cod = [0] * (n + 1)
for i in range(1, n + 1):
	c[i] = 1 if i <= k else 0
	cod[i] = 1 if i < d else 0
for i in range(1, n + 1):
	for j in range(1, k + 1):
		if i < j:
			break
		else:
			c[i] = (c[i] + c[i - j]) % mod
for i in range(1, n + 1):
	for j in range(1, k + 1):
		if i < j:
			break
		elif j < d:
			cod[i] = (cod[i] + cod[i - j]) % mod
print((c[n] - cod[n]) % mod)
