#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
const long long INF = 1ll << 60;
struct node {
  long long Left, Right, lazy, Max;
} NODE[maxn << 2];
struct jian {
  long long val, cost;
  bool operator<(const jian& n) const { return val < n.val; }
} Jian[maxn];
struct dun {
  long long val, cost;
  bool operator<(const dun& n) const { return val < n.val; }
} Dun[maxn];
struct woc {
  long long jian, dun, cost;
  bool operator<(const woc& n) const { return jian < n.jian; }
} Woc[maxn];
long long pos[maxn];
void PushUp(int idx) {
  NODE[idx].Max = max(NODE[idx << 1].Max, NODE[idx << 1 | 1].Max);
}
void build(int l, int r, int idx) {
  NODE[idx].Left = l, NODE[idx].Right = r;
  if (l == r) {
    NODE[idx].Max = -Dun[l].cost;
    pos[l] = Dun[l].val;
    return;
  }
  int m = (l + r) >> 1;
  build(l, m, idx << 1);
  build(m + 1, r, idx << 1 | 1);
  PushUp(idx);
}
void PushDown(int idx) {
  if (NODE[idx].lazy) {
    NODE[idx << 1].Max += NODE[idx].lazy;
    NODE[idx << 1 | 1].Max += NODE[idx].lazy;
    NODE[idx << 1].lazy += NODE[idx].lazy;
    NODE[idx << 1 | 1].lazy += NODE[idx].lazy;
    NODE[idx].lazy = 0;
  }
}
void Updata(int l, int r, int val, int idx) {
  if (NODE[idx].Left == l && NODE[idx].Right == r) {
    NODE[idx].Max += val;
    NODE[idx].lazy += val;
    return;
  }
  PushDown(idx);
  int m = (NODE[idx].Left + NODE[idx].Right) >> 1;
  if (r <= m)
    Updata(l, r, val, idx << 1);
  else if (l > m)
    Updata(l, r, val, idx << 1 | 1);
  else {
    Updata(l, m, val, idx << 1);
    Updata(m + 1, r, val, idx << 1 | 1);
  }
  PushUp(idx);
}
int main() {
  long long n, m, k;
  scanf("%lld%lld%lld", &n, &m, &k);
  for (int i = 1; i <= n; ++i) {
    scanf("%lld%lld", &Jian[i].val, &Jian[i].cost);
  }
  for (int i = 1; i <= m; ++i) {
    scanf("%lld%lld", &Dun[i].val, &Dun[i].cost);
  }
  for (int i = 1; i <= k; ++i) {
    scanf("%lld%lld%lld", &Woc[i].jian, &Woc[i].dun, &Woc[i].cost);
  }
  sort(Dun + 1, Dun + m + 1);
  sort(Jian + 1, Jian + n + 1);
  sort(Woc + 1, Woc + k + 1);
  build(1, m, 1);
  int j = 1;
  long long res = -INF;
  for (int i = 1; i <= n; ++i) {
    while (j <= k && Jian[i].val > Woc[j].jian) {
      int idx = upper_bound(pos + 1, pos + 1 + m, Woc[j].dun) - pos;
      if (idx <= m) Updata(idx, m, Woc[j].cost, 1);
      j++;
    }
    res = max(res, NODE[1].Max - Jian[i].cost);
  }
  printf("%lld\n", res);
}
