public class D {
	
	private void solve() throws IOException {
		int n = nextInt();
		int[] a = new int[n * n];
		for (int i = 0; i < n * n; i++) {
			a[i] = nextInt();
		}
		
		int[] ind = new int[n * n];
		boolean[] used = new boolean[n * n];
		int[] res = dfs(ind, used, a, 0, n);
		
		int s = 0;
		for (int i = 0; i < n; i++) {
			s += a[ind[i]];
		}
		println(s);
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				print(a[ind[i * n + j]] + " ");
			}
			println("");
		}
	}
	
	private final HashSet<Integer> set = new HashSet<Integer>();
	private int[] dfs(int[] ind, boolean[] used, int[] a, int d, int n) {
		if (d == a.length) {
			set.clear();
			for (int i = 0; i < n; i++) {
				int s1 = 0;
				int s2 = 0;
				for (int j = 0; j < n; j++) {
					s1 += a[ind[i * n + j]];
					s2 += a[ind[j * n + i]];
				}
				set.add(s1);
				set.add(s2);
			}
			
			int s1 = 0;
			int s2 = 0;
			for (int i = 0; i < n; i++) {
				s1 += a[ind[i * n + i]];
				s2 += a[ind[i * n + (n - 1 - i)]];
			}
			set.add(s1);
			set.add(s2);
			
			if (set.size() == 1) {
				return ind;
			} else {
				return null;
			}
		}
		
		for (int i = 0; i < a.length; i++) {
			if (!used[i]) {
				used[i] = true;
				ind[d] = i;
				int[] res = dfs(ind, used, a, d + 1, n);
				if (res != null) {
					return res;
				}
				used[i] = false;
			}
		}
		
		return null;
	}
	
	private String nextToken() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			tokenizer = new StringTokenizer(reader.readLine());
		}
		return tokenizer.nextToken();
	}

	private int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(nextToken());
	}

	private double nextDouble() throws NumberFormatException, IOException {
		return Double.parseDouble(nextToken());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	private void print(Object o) {
		writer.print(o);
	}

	private void println(Object o) {
		writer.println(o);
	}

	private void printf(String format, Object... o) {
		writer.printf(format, o);
	}

	public static void main(String[] args) {
		long time = System.currentTimeMillis();
		Locale.setDefault(Locale.US);
		new D().run();
		System.err.printf("%.3f\n", 1e-3 * (System.currentTimeMillis() - time));
	}

	BufferedReader reader;
	StringTokenizer tokenizer;
	PrintWriter writer;

	private void run() {
		try {
			reader = new BufferedReader(new InputStreamReader(System.in));
			writer = new PrintWriter(System.out);
			solve();
			reader.close();
			writer.close();
		} catch (IOException e) {
			e.printStackTrace();
			System.exit(13);
		}
	}
}
