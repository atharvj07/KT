for i in range(int(input())):
	s = input()
	if len(set(s[0:])) != len(s):
		while s:
			DEL = ''
			S = set(s[0:])
			abc = sorted(set(s[0:]))
			for j in range(len(abc) - 1, -1, -1):
				if len(set(s[s.index(abc[j]):])) == len(S):
					DEL = abc[j]
					break
			print(DEL, end='')
			s = s[s.index(DEL) + 1:]
			s = s.replace(DEL, '')
		print()
	else:
		print(s)
