#include <bits/stdc++.h>
using namespace std;
const double pi =
    3.141592653589793238462643383279502884197169399375105820974944592307816406286208998628034825342;
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  int res = 0;
  for (int i = 0; i < (n); ++i) {
    int a, b;
    for (int i = 0; i < (m); ++i) {
      scanf("%d%d", &a, &b);
      if (a || b) res++;
    }
  }
  printf("%d\n", res);
  return 0;
}
