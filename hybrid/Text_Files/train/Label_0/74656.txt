mas = list(map(int, input().split()))
px = mas[0]
py = mas[1]
vx = mas[2]
vy = mas[3]
a = mas[4]
b = mas[5]
c = mas[6]
d = mas[7]
s = (vx ** 2 + vy ** 2) ** 0.5
x1 = px + b * vx / s
y1 = py + b * vy / s
x2 = px - a * vy / s / 2
y2 = py + a * vx / s / 2
x3 = px - c * vy / s / 2
y3 = py + c * vx / s / 2
x4 = px + (-c * vy / 2 - d * vx) / s
y4 = py + (c * vx / 2 - d * vy) / s
x5 = px + (c * vy / 2 - d * vx) / s
y5 = py + (-c * vx / 2 - d * vy) / s
x6 = px + c * vy / s / 2
y6 = py - c * vx / s / 2
x7 = px + a * vy / s / 2
y7 = py - a * vx / s / 2
print('%.12f %.12f' % (x1, y1))
print('%.12f %.12f' % (x2, y2))
print('%.12f %.12f' % (x3, y3))
print('%.12f %.12f' % (x4, y4))
print('%.12f %.12f' % (x5, y5))
print('%.12f %.12f' % (x6, y6))
print('%.12f %.12f' % (x7, y7))
