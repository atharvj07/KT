#include <bits/stdc++.h>
using namespace std;
int n;
vector<string> S(100, "");

int rec(int& idx, int depth) {
    if (isdigit(S[idx][depth])) {
        return (S[idx++][depth] - '0');
    } else if (S[idx++][depth] == '+') {
        int ret = 0;
        while (S[idx].size() > depth + 1) ret += rec(idx, depth + 1);
        return (ret);
    } else {
        int ret = 1;
        while (S[idx].size() > depth + 1) ret *= rec(idx, depth + 1);
        return (ret);
    }
}

int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);

    while (cin >> n, n) {
        for (int i = 0; i < n; i++) {
            cin >> S[i];
        }
        S[n] = "";
        int idx = 0;
        cout << rec(idx, 0) << endl;
    }
}
