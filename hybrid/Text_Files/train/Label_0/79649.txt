n = int(input())
a = [i for i in range(2, n + 1)]

def calculaDivisiores(num):
	atual = 2
	divs = []
	while atual <= num // 2 + 1:
		if num % atual == 0:
			divs.append(atual)
		atual += 1
		divs.sort(reverse=True)
	return divs
res = []
for num in a:
	divs = calculaDivisiores(num)
	if len(divs) == 1:
		res.append(num)
	else:
		res.append(num)
		for i1 in range(len(divs)):
			div1 = divs[i1]
			for i2 in range(i1, len(divs)):
				div2 = divs[i2]
				if div1 != div2:
					if div1 % div2 != 0:
						if num in res:
							res.remove(num)
print(len(res))
for r in res:
	print(r, end=' ')
