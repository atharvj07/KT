#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> onecnt(n, 0);
  vector<int> zerocnt(n, 0);
  if (s[0] == '0') zerocnt[0] = 1;
  onecnt[0] = 1;
  for (int i = 1; i < n; i++) {
    int mx = max(onecnt[i - 1], zerocnt[i - 1]);
    if (s[i] == '0')
      zerocnt[i] = zerocnt[i - 1] + 1;
    else
      zerocnt[i] = zerocnt[i - 1];
    if (mx == onecnt[i - 1] && s[i] == '1')
      onecnt[i] = mx + 1;
    else if (mx == zerocnt[i - 1])
      onecnt[i] = mx + 1;
    else
      onecnt[i] = max(onecnt[i - 1], zerocnt[i - 1] + 1);
  }
  vector<int> onecntr(n, 0);
  vector<int> zerocntr(n, 0);
  if (s[n - 1] == '1') onecntr[n - 1] = 1;
  zerocntr[n - 1] = 1;
  for (int i = n - 2; i >= 0; i--) {
    int mx = max(onecntr[i + 1], zerocntr[i + 1]);
    if (s[i] == '1')
      onecntr[i] = onecntr[i + 1] + 1;
    else
      onecntr[i] = onecntr[i + 1];
    if (mx == zerocntr[i + 1] && s[i] == '0')
      zerocntr[i] = mx + 1;
    else if (mx == onecntr[i + 1])
      zerocntr[i] = mx + 1;
    else
      zerocntr[i] = max(onecntr[i + 1] + 1, zerocntr[i + 1]);
  }
  string ans = "";
  int zro = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '1') {
      int cala = (i != 0) ? onecnt[i - 1] : 0;
      int calb = (i != n - 1) ? onecntr[i + 1] : 0;
      int mx = cala + calb;
      calb = (i != n - 1) ? zerocntr[i + 1] : 0;
      mx = max(mx, zro + calb + 1);
      if (mx == onecnt[n - 1]) {
        ans += "0";
        zro++;
      } else
        ans += "1";
    } else {
      ans += "0";
      zro++;
    }
  }
  cout << ans << endl;
}
