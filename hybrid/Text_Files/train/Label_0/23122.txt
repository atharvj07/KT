#include<cstdio>
using namespace std;
int main(){
    int n;
    while(scanf("%d",&n),n){
        int a[128] = {0};
        for(int i = 0; i <= n; i++) scanf("%d",&a[i]);

        for(int i = 0; i < n; i++){
            int d = i==0?a[2]-a[1]:i==1?a[3]-a[2]:a[1]-a[0];
            int cnt = 0;
            for(int j = 0; j < n; j++){
                if(i == 0){
                    if(j == 0){
                        if(a[2] - a[1] == d)cnt++;
                    }
                    else if(a[j+1] - a[j] == d) cnt++;
                }
                else if(a[j+1==i?j+2:j+1] - a[j==i?j-1:j] == d){
                    cnt++;
                }
            }
            if(cnt == n){
                printf("%d\n",a[i]);
                goto end;
            }
        }
        printf("%d\n",a[n]);
        end:;
    }
    return 0;
}
