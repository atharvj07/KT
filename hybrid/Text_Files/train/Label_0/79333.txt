import math
(nb_disks, radius) = [int(x) for x in input().split()]
coord_x = [int(x) for x in input().split()]
coord_y = [radius] * nb_disks
for d in range(1, nb_disks):
	for d2 in range(d):
		dist_x = abs(coord_x[d] - coord_x[d2])
		if dist_x > 2 * radius:
			continue
		hauteur = coord_y[d2] + math.sqrt((2 * radius) ** 2 - dist_x ** 2)
		if hauteur > coord_y[d]:
			coord_y[d] = hauteur
print(' '.join(map(str, coord_y)))
