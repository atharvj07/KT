(K, Q) = map(int, input().split())
dp = [[0.0 for i in range(K + 1)] for j in range(10000)]
dp[0][0] = 1.0
for i in range(1, 10000, 1):
	for j in range(1, K + 1, 1):
		dp[i][j] = dp[i - 1][j] * j / K + dp[i - 1][j - 1] * (K - (j - 1)) / K
for qi in range(Q):
	P = int(input())
	for i in range(10000):
		if P <= dp[i][K] * 2000:
			print(i)
			break
