#include <bits/stdc++.h>
using namespace std;
const int inf = 1000000000;
double Time() { return double(clock()) / double(CLOCKS_PER_SEC); }
int n, a, b;
int main() {
  bool found = 0;
  cin >> a >> b >> n;
  if (a % b == 0) {
    printf("%d", a);
  } else {
    for (int i = 0; i <= 9; i++) {
      if ((a * 10 + i) % b == 0) {
        printf("%d%d", a, i);
        found = 1;
        break;
      }
    }
    if (!found) {
      cout << "-1";
      exit(0);
    }
  }
  int i = 0;
  if (found) i = 1;
  for (; i < n; i++) printf("0");
  return 0;
}
