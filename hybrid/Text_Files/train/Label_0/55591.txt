import java.util.*;
import java.io.*;
/**
 *
 * @author Cookies
 */
public class Class1 {
    
    @SuppressWarnings("empty-statement")
    public static void main(String[] args)throws IOException{
    
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] b = br.readLine().split(" ");
        long n = gl(b[0]);
        String[] v = br.readLine().split(" ");
        long[] arr = new long[v.length];
        long sum = 0;
        for(int i =0; i< n; i++){
            
            arr[i] = gl(v[i]);
            sum = sum+arr[i];
        }
        
        long div = sum/n;
        long count=0;
        for(int i = 0; i< n; i++){
            
        if(arr[i]> div){
        arr[i+1] = arr[i+1]+ arr[i]-div;
        count = count+ arr[i]-div;
        }
        if(arr[i]< div){
        arr[i+1]= arr[i+1] - (div-arr[i]);
        count = count + (div-arr[i]);
            
        }
        }
        System.out.println(count);
    }
    
    public static int gi(String a){
    
    return Integer.parseInt(a);
    
    }
    
    public static long gl(String a){
    
    return Long.parseLong(a);
    
    }
}
