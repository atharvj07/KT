#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  vector<multiset<pair<int, int>>> shakes(3);
  for (int i = 0; i < n; i++) {
    int temp;
    scanf("%d", &temp);
    shakes[temp % 3].insert({temp, i});
  }
  vector<pair<int, int>> result;
  for (int i = 0;; i++) {
    if (!shakes[i % 3].size()) break;
    multiset<pair<int, int>>::iterator it = shakes[i % 3].lower_bound({i, n});
    if (it == shakes[i % 3].begin()) {
      break;
    } else {
      it--;
      result.push_back(*it);
      i = it->first;
      shakes[i % 3].erase(it);
    }
  }
  if (shakes[0].size() || shakes[1].size() || shakes[2].size()) {
    printf("Impossible\n");
  } else {
    printf("Possible\n");
    for (int i = 0; i < n; i++) {
      printf("%d%c", result[i].second + 1, i == n - 1 ? '\n' : ' ');
    }
  }
  return 0;
}
