import math
import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input().strip()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())

def print_num(n):
	if n % 2 == 1:
		print(1, (n - 1) // 2, (n - 1) // 2)
	elif n % 4 == 0:
		print(n // 2, n // 4, n // 4)
	elif n % 2 == 0:
		print(2, (n - 2) // 2, (n - 2) // 2)
for i in range(inp()):
	(n, k) = invr()
	s = 0
	while k > 3:
		k -= 1
		s += 1
		print(1, end=' ')
	print_num(n - s)
