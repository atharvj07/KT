vector<string> vec_splitter(string s) {
  s += ',';
  vector<string> res;
  while (!s.empty()) {
    res.push_back(s.substr(0, s.find(',')));
    s = s.substr(s.find(',') + 1);
  }
  return res;
}
void debug_out(vector<string> __attribute__((unused)) args,
               __attribute__((unused)) int idx,
               __attribute__((unused)) int LINE_NUM) {
  cerr << endl;
}
template <typename Head, typename... Tail>
void debug_out(vector<string> args, int idx, int LINE_NUM, Head H, Tail... T) {
  if (idx > 0)
    cerr << ", ";
  else
    cerr << "Line(" << LINE_NUM << ") ";
  stringstream ss;
  ss << H;
  cerr << args[idx] << " = " << ss.str();
  debug_out(args, idx + 1, LINE_NUM, T...);
}

const int NN = (int)1e6 + 5;
const int MaxM = (int)10000 + 5;
const int MOD = (int)1e9 + 7;
const int INF = INT_MAX - 1;
int mul(int x, int y) { return (1LL * x * y) % MOD; }
int add(int x, int y) {
  long long res = x;
  res += y;
  return (res % MOD + MOD) % MOD;
}
void norm_add(int& x, int y) {
  x += y;
  if (x >= MOD) x -= MOD;
}
int fast_pow(int x, long long y) {
  if (y == 0) return 1;
  if (y & 1)
    return mul(x, fast_pow(x, y - 1));
  else {
    int aux = fast_pow(x, y >> 1);
    return mul(aux, aux);
  }
}
int inv(int x) { return fast_pow(x, MOD - 2); }
int two(int x) { return 1 << x; }
bool contain(int mask, int bit) { return (mask & two(bit)) > 0; }
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int n;
int a[NN];
bool vis[NN];
int main() {
  ios_base::sync_with_stdio(0);
  scanf("%d", &n);
  memset(vis, false, sizeof(vis));
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
    vis[a[i]] = true;
  }
  int s = 1000 * 1000;
  vector<int> ans;
  int c = 0;
  for (int i = 0; i < n; i++) {
    if (vis[s + 1 - a[i]] == false) {
      ans.push_back(s + 1 - a[i]);
    } else
      c++;
  }
  for (int i = 1; c && i <= s; i++) {
    if (vis[i] == false && vis[s + 1 - i] == false) {
      ans.push_back(i), ans.push_back(s + 1 - i);
      c -= 2;
    }
  }
  printf("%d\n", ((int)ans.size()));
  for (int x : ans) printf("%d ", x);
  printf("\n");
  return 0;
}
