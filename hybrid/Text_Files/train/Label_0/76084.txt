k = int(input())
for i in range(k):
	n = int(input())
	l = list(map(int, input().split()))
	an = 0
	o = 0
	p = 0
	if n == 1:
		print(l[0])
		continue
	for i in range(n):
		if i == 0 and l[0] > l[1]:
			an += l[0] - l[1]
			l[0] = l[1]
		if i > 0 and i < n - 1:
			if l[i] > l[i - 1] and l[i] > l[i + 1]:
				an += l[i] - max(l[i - 1], l[i + 1])
				l[i] = max(l[i - 1], l[i + 1])
			an += abs(l[i] - l[i - 1])
		if i == n - 1 and l[n - 1] > l[n - 2]:
			an += l[n - 1] - l[n - 2]
			l[n - 1] = l[n - 2]
	an += l[0]
	an += l[n - 1]
	an += abs(l[n - 1] - l[n - 2])
	print(an)
