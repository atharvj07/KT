#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c = 0;
  cin >> n;
  vector<int> p(n + 1);
  vector<int> e(n + 1);
  vector<int> d(n + 1);
  for (int i = 2; i <= n; i++) cin >> p[i];
  e[1] = 1;
  for (int i = 2; i <= n; i++) e[i] = e[p[i]] + 1;
  for (int i = 1; i <= n; i++) d[e[i]] = !d[e[i]];
  for (int i = 1; i <= n; i++) c += d[i];
  cout << c << endl;
  return 0;
}
