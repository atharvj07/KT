import sys
input = sys.stdin.readline
from collections import defaultdict
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	while m > 3:
		print(1, end=' ')
		m -= 1
		n -= 1
	if n % 2:
		print(n // 2, n // 2, 1)
	elif n % 4 == 0:
		print(n // 2, n // 4, n // 4)
	else:
		print(n // 2 - 1, 2, n // 2 - 1)
