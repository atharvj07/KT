import math
m = 1000000007

def fact(n):
	ans = 1
	for i in range(1, n + 1):
		ans = ans % m * (i % m) % m
	return ans

def power_2(n):
	ans = 1
	for i in range(n):
		ans = ans % m * 2 % m
	return ans
(n, a) = map(int, input().split())
t = [int(i) for i in input().split()]
(cnt, i) = (0, 0)
while a - 1 - i >= 0 or a - 1 + i < n:
	if i == 0 and t[a - 1] == 1:
		cnt += 1
	elif a - 1 - i >= 0 and a - 1 + i < n and (t[a - 1 - i] == 1) and (t[a - 1 + i] == 1):
		cnt += 2
	elif a - 1 - i >= 0 and a - 1 + i >= n and (t[a - 1 - i] == 1):
		cnt += 1
	elif a - 1 - i < 0 and a - 1 + i < n and (t[a - 1 + i] == 1):
		cnt += 1
	i += 1
print(cnt)
