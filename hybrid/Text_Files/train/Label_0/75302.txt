import sys
from collections import defaultdict
strInp = lambda : input().strip().split()
intInp = lambda : list(map(int, strInp()))
n = int(input())
arr = intInp()
prefixSum = [0]
for i in arr:
	prefixSum.append(prefixSum[len(prefixSum) - 1] + i)
x = 1 - min(prefixSum[1:])
if x == 0:
	x = 1
ans = [x]
for i in range(n - 1):
	ans.append(x + prefixSum[i + 1])
if max(ans) == n and len(set(ans)) == n and (min(ans) == 1):
	for i in ans:
		print(i, end=' ')
else:
	print(-1)
