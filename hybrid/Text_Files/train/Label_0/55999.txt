import static java.lang.Math.max;
import static java.lang.Math.min;
import static java.lang.Math.abs;
import static java.lang.System.out;
import java.util.*;
import java.io.*;
import java.math.*;

public class x1500C
{
    public static void main(String hi[]) throws Exception
    {
        BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(infile.readLine());
        int N = Integer.parseInt(st.nextToken());
        int M = Integer.parseInt(st.nextToken());
        int[][] grid = new int[N][M];
        for(int r=0; r < N; r++)
            grid[r] = readArr(M, infile, st);
        int[][] target = new int[N][M];
        for(int r=0; r < N; r++)
            target[r] = readArr(M, infile, st);
        int[] cnt = new int[M];
        for(int c=0; c < M; c++)
            for(int r=1; r < N; r++)
                if(target[r-1][c] > target[r][c])
                    cnt[c]++;
        boolean[] processed = new boolean[M];
        Stack<Integer> stack = new Stack<Integer>();
        for(int c=0; c < M; c++)
            if(cnt[c] == 0)
            {
                stack.push(c);
                processed[c] = true;
            }
        ArrayList<Integer> ls = new ArrayList<Integer>();
        boolean[] cut = new boolean[N];
        while(stack.size() > 0)
        {
            int curr = stack.pop();
            ls.add(curr);
            for(int r=1; r < N; r++)
                if(target[r-1][curr] < target[r][curr] && !cut[r])
                {
                    cut[r] = true;
                    for(int c=0; c < M; c++)
                        if(!processed[c])
                        {
                            if(target[r-1][c] > target[r][c])
                                cnt[c]--;
                            if(cnt[c] == 0)
                            {
                                stack.push(c);
                                processed[c] = true;
                            }
                        }
                }
        }
        Collections.reverse(ls);
        for(int k: ls)
        {
            ArrayList<Unit> sorting = new ArrayList<Unit>();
            for(int r=0; r < N; r++)
                sorting.add(new Unit(grid[r][k], r));
            Collections.sort(sorting);
            int[][] next = new int[N][];
            for(int r=0; r < N; r++)
                next[r] = grid[sorting.get(r).dex];
            grid = next;
        }
        boolean works = true;
        for(int r=0; r < N; r++)
            for(int c=0; c < M; c++)
                if(grid[r][c] != target[r][c])
                    works = false;
        if(works)
        {
            StringBuilder sb = new StringBuilder(ls.size()+"\n");
            for(int x: ls)
                sb.append((x+1)+" ");
            System.out.println(sb);
        }
        else
            System.out.println(-1);
    }
    public static int[] readArr(int N, BufferedReader infile, StringTokenizer st) throws Exception
    {
        int[] arr = new int[N];
        st = new StringTokenizer(infile.readLine());
        for(int i=0; i < N; i++)
            arr[i] = Integer.parseInt(st.nextToken());
        return arr;
    }
}
class Unit implements Comparable<Unit>
{
    public int val;
    public int dex;

    public Unit(int a, int b)
    {
        val = a;
        dex = b;
    }
    public int compareTo(Unit oth)
    {
        if(val == oth.val)
            return dex-oth.dex;
        return val-oth.val;
    }
}
