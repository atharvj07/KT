t = int(input())
for i in range(t):
	n = int(input())
	arr = [int(item) for item in input().split()]
	arr2 = [0]
	arr2.extend(arr)
	arr2.extend([0])
	countops = 0
	for i in range(1, len(arr2) - 1):
		if arr2[i - 1] < arr2[i] and arr2[i + 1] < arr2[i]:
			diff = min(arr2[i] - arr2[i - 1], arr2[i] - arr2[i + 1])
			arr2[i] = arr2[i] - diff
			countops += diff
	count = 0
	for i in range(len(arr2) - 1):
		count += abs(arr2[i + 1] - arr2[i])
	print(count + countops)
