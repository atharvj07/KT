

import java.util.Scanner;

public class PracticeDemo {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		if(n <= 2) System.out.println(1);
		else System.out.println(2);
		sc.close();
		int i = 0,j =0;
		boolean flag =true;
		for( i  = 2; i <= n + 1; i++ ){
			if(i==2){
				System.out.print(1+" ");
				continue;
			}
				
			flag =true;
			for(j = 2; j <=  (Math.sqrt(i)+1);j++)
				if( i % j == 0)
					flag = false;
				if(flag) 
					System.out.print( 1 + " ");
				else 
					System.out.print( 2 +" ");
		}
	}
}
