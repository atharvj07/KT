n = int(input())
s = input()
arr = [0 for i in range(n)]
grps = []
col = 0
for i in range(len(s)):
	if grps == []:
		grps.append(s[i])
		arr[i] = 1
		col += 1
	else:
		found = 0
		for j in range(len(grps)):
			if ord(s[i]) >= ord(grps[j]):
				found = 1
				grps[j] = s[i]
				arr[i] = j + 1
				break
		if found == 0:
			grps.append(s[i])
			col += 1
			arr[i] = col
print(col)
for i in range(n):
	print(arr[i], end=' ')
print()
