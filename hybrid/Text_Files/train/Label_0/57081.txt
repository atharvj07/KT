import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.AbstractCollection;
import java.util.Scanner;
import java.util.LinkedList;
import java.util.ArrayList;

/**
 * Built using CHelper plug-in Actual solution is at the top
 *
 * @author dima
 */
public class Main {

    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskC solver = new TaskC();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskC {

        public void solve(int testNumber, Scanner in, PrintWriter out) {
            String s = in.nextLine();
            ArrayList<Integer>[] subseqs = new ArrayList[s.length() + 1];
            LinkedList<Integer> zeroSubSeqs = new LinkedList<>();
            LinkedList<Integer> oneSubseqs = new LinkedList<>();
            int subseqsCount = 0;
            for (int i = 0; i < s.length(); ++i) {
                if (s.charAt(i) == '1') {
                    if (zeroSubSeqs.isEmpty()) {
                        out.print(-1);
                        return;
                    } else {
                        int first = zeroSubSeqs.pollFirst();
                        subseqs[first].add(i + 1);
                        oneSubseqs.add(first);
                    }
                } else {
                    if (oneSubseqs.isEmpty()) {
                        subseqs[subseqsCount] = new ArrayList<>();
                        subseqs[subseqsCount].add(i + 1);
                        zeroSubSeqs.add(subseqsCount);
                        subseqsCount++;
                    } else {
                        int seqId = oneSubseqs.pollFirst();
                        subseqs[seqId].add(i + 1);
                        zeroSubSeqs.add(seqId);
                    }
                }
            }
            if (!oneSubseqs.isEmpty()) {
                out.print(-1);
            } else {
                out.println(subseqsCount);
                for (int j = 0; j < subseqsCount; ++j) {
                    ArrayList<Integer> a = subseqs[j];
                    out.print(a.size());
                    for (int i = 0; i < a.size(); ++i) {
                        out.print(' ');
                        out.print(a.get(i));
                    }
                    out.println();
                }
            }
        }

    }
}
