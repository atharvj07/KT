#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
long long power(long long a, long long n) {
  long long x = 1;
  while (n) {
    if (n & 1) x = x * a % MOD;
    a = a * a % MOD;
    n >>= 1;
  }
  return x;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  long long ans = 1;
  for (int i = 1; i <= n; i++) ans = ans * i % MOD;
  long long a = 1, b = n;
  for (int i = 0; i < n; i++) {
    (ans += i * (b - a + MOD) % MOD) %= MOD;
    a = a * (n - i) % MOD;
    b = b * (n - i - 1) % MOD;
  }
  cout << ans << '\n';
  return 0;
}
