#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e6 + 1;
int a[maxn] = {0};
int main(int argc, char const *argv[]) {
  int A;
  scanf("%d", &A);
  int ans = 0, tmp = 0, flag = 0;
  for (int i = 1; i <= 10; i++) {
    a[i]++;
    for (int j = i + 1; j < maxn; j++) {
      a[j] += a[j - i];
      if (A == a[j]) {
        ans = j;
        tmp = i;
        flag = 1;
        break;
      }
    }
    if (flag) break;
  }
  printf("%d %d\n", ans, tmp);
  for (int i = 1; i <= tmp; i++) printf("%d ", i);
  printf("\n");
  return 0;
}
