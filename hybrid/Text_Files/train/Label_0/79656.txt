t = int(input())
for i in range(t):
	s = input()
	h = input()
	if len(h) < len(s):
		print('NO')
	else:
		used = [0] * 26
		for i in range(len(s)):
			used[ord(s[i]) - ord('a')] += 1
		uk1 = 0
		uk2 = 0
		ans = False
		used2 = [0] * 26
		while uk2 - uk1 < len(s):
			used2[ord(h[uk2]) - ord('a')] += 1
			uk2 += 1
		if used == used2:
			ans = True
		while uk2 < len(h):
			used2[ord(h[uk1]) - ord('a')] -= 1
			uk1 += 1
			used2[ord(h[uk2]) - ord('a')] += 1
			if used2 == used:
				ans = True
			uk2 += 1
		if ans:
			print('YES')
		else:
			print('NO')
