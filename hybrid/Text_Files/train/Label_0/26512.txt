using namespace std;


#define NUM 1700

struct Info{
	int a,b;
};

int V; //頂点数
int N;
vector<int> G[NUM]; //グラフの隣接リスト表現
int match[NUM]; //マッチングのペア
bool used[NUM]; //DFSですでに調べたかのフラグ
//bool have_pair[NUM]; //ペアを既に作ったかのフラグ
Info info[800];

//fromとtoを結ぶ辺をグラフに追加する
void add_edge(int from,int to){
	G[from].push_back(to);
	G[to].push_back(from);
}

//増加パスをDFSで探す(node_idのペアを探す)
int dfs(int node_id){
	used[node_id] = true;

	for(int i = 0; i < G[node_id].size(); i++){
		int adj_node_id = G[node_id][i],pair_id = match[adj_node_id];
		if((pair_id < 0)|| //隣接するノードに、まだペアが存在しない
				(used[pair_id] == false && dfs(pair_id) == true)){

			match[node_id] = adj_node_id;
			match[adj_node_id] = node_id;

			return true;
		}
	}
	return false;
}


int bipartie_matching(){
	int ret = 0;
	for(int i = 0; i < V; i++)match[i] = -1; //ペアなしにしておく
	for(int node_id = 0; node_id < V; node_id++){
		if(match[node_id] < 0){ //node_idにペアがいない場合
			for(int i = 0; i < V; i++)used[i] = false;
			if(dfs(node_id)){ //増加パスが存在する場合
				ret++;
			}
		}
	}
	return ret;
}


int main(){

	int A,B;
	scanf("%d %d %d",&N,&A,&B);

	bool check[N];
	int tmp,ans = 0;

	//N件のデータを読み込む
	for(int i = 0; i < N; i++){
		check[i] = false;

		scanf("%d %d",&info[i].a,&info[i].b);
		tmp = abs(info[i].a-info[i].b);
		if(tmp <= A || (tmp >= B && tmp <= 2*A)){
			ans++;
			check[i] = true;
		}
	}

	for(int i = 0; i < N; i++){
		if(check[i])continue;

		for(int k = i+1; k < N; k++){ //2重計上を防ぐため、自分よりインデックスが大きいもののみ走査
			if(k == i || check[k] == true)continue;
			tmp = abs((info[i].a+info[k].a)-(info[i].b+info[k].b));
			if(tmp <= A || (tmp >= B && tmp <= 2*A)){
				add_edge(i,k);
			}
		}
	}

	V = N;

	printf("%d\n", ans+bipartie_matching());

	return 0;
}
