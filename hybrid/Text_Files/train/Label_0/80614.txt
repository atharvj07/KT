n = int(input())
a = list(map(int, input().strip().split(' ')))
f = [4, 8, 15, 16, 23, 42]
x = [0 for _ in range(6)]
for i in range(n):
	if a[i] == 4:
		a[i] = 0
	elif a[i] == 8:
		a[i] = 1
	elif a[i] == 15:
		a[i] = 2
	elif a[i] == 16:
		a[i] = 3
	elif a[i] == 23:
		a[i] = 4
	elif a[i] == 42:
		a[i] = 5
	else:
		a[i] == 6
for i in a:
	if i >= 6:
		continue
	if i == 0:
		x[0] += 1
	elif x[i - 1] > 0:
		x[i - 1] -= 1
		x[i] += 1
print(n - x[-1] * 6)
