import math

def calculate_derangement(fac, n):
	ans = fac[n]
	for i in range(1, n + 1):
		if i % 2 == 1:
			ans -= fac[n] // fac[i]
		else:
			ans += fac[n] // fac[i]
	return ans
(n, k) = map(int, input().split())
fac = [1]
for i in range(1, n + 1):
	fac.append(fac[i - 1] * i)
ans = 0
for i in range(0, k + 1):
	choose = fac[n]
	choose //= fac[i]
	choose //= fac[n - i]
	ans += choose * calculate_derangement(fac, i)
print(ans)
