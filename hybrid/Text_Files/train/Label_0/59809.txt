import java.util.*;

public class Main {
    static class Node {
        int[] f;
        int v;
        public Node() {
            f = new int[26];
            v = -1;
        }

        public Node(int _v) {
            v = _v;
            f = new int[26];
        }

        public int distinct() {
            int ans = 0;
            for (int i=0; i<26; i++) {
                ans += f[i];
            }
            return ans;
        }

        public Node merge(Node other) {
            Node ans = new Node();
            for (int i=0; i<26; i++) {
                ans.f[i] = this.f[i] | other.f[i];
            }
            return ans;
        }

        public String toString() {
            String s = "";
            for (int i=0; i<26; i++) {
                if (i > 0) s +=  " ";
                s += f[i];
            }
            return s;
        }
    }

    static class SegmentTree {
        int n;
        Node[] nodes;
        String s;
        SegmentTree(String s_) {
            s = s_;
            n = s.length();
            nodes = new Node[4*n];
            for (int i=0; i<4*n; i++) {
                nodes[i] = new Node();
            }
            build(1, 0, n-1);
        }

        private void build(int cur ,int l, int r) {
            if (l==r) {
                nodes[cur].f[s.charAt(l)-'a'] = 1;
                nodes[cur].v = s.charAt(l) - 'a';
                return;
            }
            int mid = (l+r)/2;
            build(cur*2, l, mid);
            build(cur*2 + 1, mid+1, r);
            relax(cur);
        }

        public void update(int ind, int v) {
            update(ind, 1, 0, n-1, v);
        }

        private void update(int ind, int cur, int l, int r, int v) {
            if (l == r) {
                Node leaf = nodes[cur];
                leaf.f[leaf.v] = 0;
                leaf.f[v] = 1;
                leaf.v = v;
                return;
            }
            int mid = (l+r)/2;
            if (ind <= mid) {
                update(ind, cur*2, l, mid, v);
            } else {
                update(ind, cur*2 + 1, mid+1, r, v);
            }
            relax(cur);
        }

        private void relax(int x) {
            int left = x*2, right = left+1;
            for (int i=0; i<26; i++) {
                nodes[x].f[i] = nodes[left].f[i] | nodes[right].f[i];
            }
        }

        public int query(int l, int r) {
            return query(1, 0, n-1, l, r).distinct();
        }

        private Node query(int cur, int l, int r, int i, int j) {
            if (l > j || r < i) return new Node();
            if (i <= l && j >= r) {
                // System.out.println(nodes[cur]);
                return nodes[cur];
            }
            int mid = (l+r)/2;
            Node left = query(cur*2, l, mid, i, j);
            Node right = query(cur*2 + 1, mid+1, r, i, j);
            return left.merge(right);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        SegmentTree st = new SegmentTree(s);
        int q = sc.nextInt();
        while (q-- > 0) {
            int t = sc.nextInt();
            if (t == 1) {
                int ind = sc.nextInt();
                char c = sc.next().charAt(0);
                st.update(ind-1, c-'a');
            } else {
                int l = sc.nextInt();
                int r = sc.nextInt();
                System.out.println(st.query(l-1, r-1));
            }
        }
        sc.close();
    }
}
