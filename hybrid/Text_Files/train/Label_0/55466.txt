import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Scanner;

/**
 * Built using CHelper plug-in Actual solution is at the top
 *
 * @author dima
 */
public class Main {

    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {

        public void solve(int testNumber, Scanner in, PrintWriter out) {
            long n, q;
            n = in.nextLong();
            q = in.nextLong();
            for (int i = 0; i < q; ++i) {
                long index = in.nextLong();
                out.println(solve(1, index - 1, n));
            }
        }

        private long solve(long first, long pos, long n) {
            long total = n - first + 1;
            if (pos < total) {
                if ((pos & 1) == 0) {
                    return first + (pos >> 1);
                } else {
                    return solve(first + total / 2, (pos + (total & 1)) >> 1,
                                 n);
                }
            } else {
                if ((total & 1) == 0) {
                    return solve(first + total / 2, pos / 2, n);
                } else {
                    return solve(first + total / 2, pos - total + 1, n);
                }
            }
        }

    }
}
