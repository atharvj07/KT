#include<iostream>
#include<string>
#include<algorithm>
#include<vector>
using namespace std;
int x[14][14], n; string S[14];
string dp[14][16384];
bool erased[14]; vector<int>vec;
int SSS(string S, string T) {
	for (int i = min(S.size(), T.size()); i >= 0; i--) {
		if (S.substr(S.size() - i, i) == T.substr(0, i))return i;
	}
}
int main() {
	while (true) {
		cin >> n; vec.clear(); if (n == 0)break; for (int i = 0; i < n; i++)cin >> S[i];
		for (int i = 0; i < 14; i++) {
			erased[i] = false;
			for (int j = 0; j < 16384; j++) { dp[i][j] = "$"; }
		}
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (S[i].size() > S[j].size()) {
					for (int k = 0; k <= S[i].size() - S[j].size(); k++) {
						if (S[i].substr(k, S[j].size()) == S[j])erased[j] = true;
					}
				}
			}
		}
		for (int i = 0; i < n; i++) { if (erased[i] == false)vec.push_back(i); }
		n = vec.size(); for (int i = 0; i < n; i++)S[i] = S[vec[i]];
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (i != j)x[i][j] = SSS(S[i], S[j]);
			}
		}
		for (int i = 0; i < n; i++) { dp[i][(1 << i)] = S[i]; }
		for (int j = 0; j < (1 << n); j++) {
			for (int i = 0; i < n; i++) {
				if (dp[i][j] == "$")continue;
				int bit[14]; for (int k = 0; k < n; k++)bit[k] = (j / (1 << k)) % 2;
				for (int k = 0; k < n; k++) {
					if (bit[k] == 1)continue;
					string W1 = dp[i][j] + S[k].substr(x[i][k], S[k].size() - x[i][k]);
					if (dp[k][j + (1 << k)] == "$") { dp[k][j + (1 << k)] = W1; }
					if (dp[k][j + (1 << k)].size() > W1.size()) { dp[k][j + (1 << k)] = W1; }
					if (dp[k][j + (1 << k)].size() == W1.size() && dp[k][j + (1 << k)] > W1) { dp[k][j + (1 << k)] = W1; }
				}
			}
		}
		string minx = dp[0][(1 << n) - 1];
		for (int i = 0; i < n; i++) {
			if (minx.size() > dp[i][(1 << n) - 1].size()) { minx = dp[i][(1 << n) - 1]; }
			if (minx.size() == dp[i][(1 << n) - 1].size() && minx > dp[i][(1 << n) - 1]) { minx = dp[i][(1 << n) - 1]; }
		}
		cout << minx.size() << endl;
	}
	return 0;
}
