#include <bits/stdc++.h>
using namespace std;
unsigned long long int power(unsigned long long int x,
                             unsigned long long int n) {
  if (n == 0) return 1;
  if (n % 2 == 0) {
    unsigned long long int y = power(x, n / 2);
    return (y * y);
  } else
    return ((x * pow(x, n - 1)));
}
