for i in range(int(input())):
	n = int(input())
	a = []
	arr = [-1]
	for j in range(n):
		(h, w) = map(int, input().split())
		if h < w:
			(h, w) = (w, h)
		a.append([h, w, j + 1])
		arr.append(-1)
	a.sort()
	width = 1000000000
	a.append(a[-1])
	j = 0
	pos = -1
	position = -1
	while j < n:
		w = 1000000000
		while j < n and a[j][0] == a[j + 1][0]:
			if a[j][1] > width:
				arr[a[j][2]] = position
			if a[j][1] < w:
				w = a[j][1]
				pos = a[j][2]
			j += 1
		else:
			if a[j][1] > width:
				arr[a[j][2]] = position
			if w > a[j][1]:
				w = a[j][1]
				pos = a[j][2]
			j += 1
		if width > w:
			width = w
			position = pos
	print(*arr[1:])
