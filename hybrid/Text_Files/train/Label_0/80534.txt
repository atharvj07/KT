from collections import Counter
n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
d = Counter()
for language in a:
	d[language] += 1
scores = [None] * m
for i in range(m):
	scores[i] = (d[b[i]], d[c[i]], i + 1)
scores.sort(reverse=True)
print(scores[0][2])
