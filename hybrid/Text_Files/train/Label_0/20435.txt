#include <bits/stdc++.h>
int h[1000005];
int main() {
  int n, k;
  while (~scanf("%d%d", &n, &k)) {
    for (int i = 0; i < n; i++) {
      scanf("%d", &h[i]);
    }
    int sum = 0;
    int min;
    for (int i = 0; i < k; i++) sum = sum + h[i];
    min = sum;
    int ans = 0;
    for (int i = k; i < n; i++) {
      sum = sum + h[i] - h[i - k];
      if (sum < min) {
        min = sum;
        ans = i - k + 1;
      }
    }
    printf("%d\n", ans + 1);
  }
}
