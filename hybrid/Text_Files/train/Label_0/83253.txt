from math import ceil
from collections import deque
import heapq as heap
(n, k) = [int(p) for p in input().split()]
arr = deque([int(p) for p in input().split()])
a = int(input())
cost = deque([int(p) for p in input().split()])
temp = []
heap.heapify(temp)
total = 0
ok = 1
for i in range(n):
	heap.heappush(temp, cost[i])
	while arr[i] > k and temp:
		k += a
		total += heap.heappop(temp)
	if k < arr[i]:
		ok = 0
		break
if ok:
	print(total)
else:
	print(-1)
