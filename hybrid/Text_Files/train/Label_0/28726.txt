#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  char s[109][109] = {0};
  cin >> n;
  for (int i = 0; i < n; i++) scanf("%s", &s[i]);
  bool h[109] = {0};
  int c[109] = {0};
  for (int i = 0; i < n; i++) {
    if (h[i] == 0) {
      h[i] = 1;
      c[i] = 1;
      for (int j = i + 1; j < n; j++) {
        if (h[j] == 0) {
          int f = 0;
          for (int k = 0; k < n; k++)
            if (s[i][k] != s[j][k]) {
              f = 1;
              break;
            }
          if (f == 0) {
            c[i]++;
            h[j] = 1;
          }
        }
      }
    }
  }
  int l = 0;
  for (int i = 0; i < n; i++) l = max(l, c[i]);
  cout << l << endl;
  return 0;
}
