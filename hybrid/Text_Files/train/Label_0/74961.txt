import sys
input = sys.stdin.readline
t = int(input())
while t > 0:
	t -= 1
	n = int(input())
	if n == 1:
		print(1)
	elif n == 2:
		print(-1)
	else:
		a = [[-1 for i in range(n)] for j in range(n)]
		x = 1
		for i in range(n):
			a[i][i] = x
			x += 1
		j = 1
		for i in range(n - 1):
			o = 0
			k = j
			while o < n and k < n:
				a[o][k] = x
				x += 1
				o += 1
				k += 1
			o = j
			k = 0
			while o < n and k < n:
				a[o][k] = x
				x += 1
				o += 1
				k += 1
			j += 1
		for i in range(n):
			print(*a[i])
