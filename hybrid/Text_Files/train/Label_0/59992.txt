

import java.util.Scanner;

/**
 *
 * @author takhi
 */
public class numbers_262A {
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int k=sc.nextInt();
        int count=0;
        int lucky=0;
        int rem=0;
        while(n-->0){
            int x=sc.nextInt();
            int num=x;
            while(num>0){
                rem=num%10;
                if(rem==4||rem==7){
                    count++;
                }
                num=num/10;
            }
            if(count<=k){
                lucky++;
            }
            count=0;
        }
        System.out.println(lucky);
        
    }
}
