#include<bits/stdc++.h>
#define int long long 
using namespace std;


int32_t main(){
	
	int tt;
	cin>>tt;
	while(tt--){
		int n,m;
		cin>>n>>m;
		 int arr[n][m];
		  vector<pair<int ,pair<int,int>>>v;
		for(int i=0;i<n;i++){
			for(int j=0;j<m;j++){
				 cin>>arr[i][j];
			}
			sort(arr[i],arr[i]+m);
			for(int j=0;j<m;j++){
				v.push_back({arr[i][j],{i,j}});
			}
			
		}
		sort(v.begin(),v.end());
		for(int k=m-1;k>=0;k--){
			int i=v[k].second.first;
			int j=v[k].second.second;
			swap(arr[i][j],arr[i][k]);
		}
		for(int i=0;i<n;i++){
			for(int j=0;j<m;j++){
				cout<<arr[i][j]<<" ";
				
			}
			cout<<"\n";
		}
		
		
	}
	
	return 0;
}
