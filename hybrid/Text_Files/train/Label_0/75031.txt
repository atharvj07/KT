mod = 998244353
n = int(input())
a = list(map(int, input().split()))
poses = dict()
for i in range(n):
	poses[a[i]] = i
ans = 1
pos = 0
while pos < n:
	need = poses[a[pos]]
	while pos < n and pos <= need:
		need = max(need, poses[a[pos]])
		pos += 1
	ans += 1
ans -= 1
answer = 1
for i in range(ans - 1):
	answer = answer * 2 % mod
print(answer)
