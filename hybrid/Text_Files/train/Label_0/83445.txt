(n, k) = input().split()
n = int(n)
k = int(k) - 1
b = []
s = (k + 3) // 2
b.append(s)
i = 0
if k % 2 == 0:
	i = 1
else:
	i = -1
while k > -1:
	s = s + i
	b.append(s)
	if i > 0:
		i = -1 * i - 1
	else:
		i = -1 * i + 1
	k -= 1
while s < n:
	s += 1
	b.append(s)
for d in range(len(b)):
	print(b[d], end=' ')
