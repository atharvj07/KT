import java.io.*;
import java.util.StringTokenizer;
import static java.lang.Math.*;

public class Main {
    static class Reader {
        private BufferedReader br;
        private StringTokenizer token;
        protected Reader(FileReader obj) {
            br = new BufferedReader(obj, 32768);
            token = null;
        }
        protected Reader() {
            br = new BufferedReader(new InputStreamReader(System.in), 32768);
            token = null;
        }
        protected String next() {
            while(token == null || !token.hasMoreTokens()) {
                try {
                    token = new StringTokenizer(br.readLine());
                } catch (Exception e) {e.printStackTrace();}
            } return token.nextToken();
        }
        protected int nextInt() {return Integer.parseInt(next());}
        protected long nextLong() {return Long.parseLong(next());}
        protected double nextDouble() {return Double.parseDouble(next());}
    }
    static int n;
    static class SegmentTree {
        private int[][] st;
        private int shift_return = 0;
        private boolean in_b = true;
        SegmentTree() {
            int x = (int)ceil(log(n)/log(2));
            st = new int[(1<<(1+x)) - 1][2];
        }
        void update(int qs, int qe, int shift) {
            updateUTIL(0, 0, n-1, qs, qe, shift);
        }
        void updateUTIL(int sti, int sts, int ste, int qs, int qe, int shift) {
            if (st[sti][1] == -1 && sts != ste) {
                st[(sti<<1)|1][0] = st[sti][0];
                st[(sti+1)<<1][0] = st[sti][0];
                st[(sti<<1)|1][1] = -1;
                st[(sti+1)<<1][1] = -1;
                st[sti][0] = 0; st[sti][1] = 0;
            }
            if (qe<sts || ste<qs) return;
            else if (qs<=sts && ste<=qe) {
                st[sti][0] = shift; st[sti][1] = -1;
            } else {
                int mid = (sts+ste)/2;
                updateUTIL((sti<<1)|1, sts, mid, qs, qe, shift);
                updateUTIL((sti+1)<<1, mid+1, ste, qs, qe, shift);
            }
        }
        int queryShift(int ind) {
            shift_return = 0;
            in_b = true;
            return queryShiftUTIL(0, 0, n-1, ind);
        }
        int queryShiftUTIL (int sti, int sts, int ste, int ind) {
            if (st[sti][1] == -1 && sts != ste) {
                st[(sti<<1)|1][0] = st[sti][0];
                st[(sti+1)<<1][0] = st[sti][0];
                st[(sti<<1)|1][1] = -1;
                st[(sti+1)<<1][1] = -1;
                st[sti][0] = 0; st[sti][1] = 0;
            }
            if (ind<sts || ind>ste) return 0;
            else if (sts == ste) {
                if (st[sti][1] == -1) {shift_return = st[sti][0]; in_b = false;}
                return shift_return;
            }
            else {
                if (st[sti][1] == -1) {shift_return = st[sti][0]; in_b = false;}
                int mid = (sts+ste)/2;
                queryShiftUTIL((sti<<1)|1, sts, mid, ind);
                queryShiftUTIL((sti+1)<<1, mid+1, ste, ind);
            }
            return shift_return;
        }
    }
    public static void main(String[] args) throws IOException {
        Reader in = new Reader();
        PrintWriter out = new PrintWriter(System.out);
        n = in.nextInt(); int m = in.nextInt();
        int[] a = new int[n]; int[] b = new int[n];
        for (int i=0; i<n; i++) a[i] = in.nextInt();
        for (int i=0; i<n; i++) b[i] = in.nextInt();
        SegmentTree tree = new SegmentTree();
        while (m-->0) {
            int t = in.nextInt();
            if (t == 2) {
                int x = in.nextInt()-1, shift = tree.queryShift(x);
                if (tree.in_b) out.printf("%d\n", b[shift+x]);
                else out.printf("%d\n", a[shift+x]);
            } else {
                int x = in.nextInt()-1, y = in.nextInt()-1, k = in.nextInt(), qs = y, qe = y+k-1;
                tree.update(qs, qe, x-y);
            }
        }
        out.close();
    }
}
