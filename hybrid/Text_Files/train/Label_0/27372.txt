#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  while (b ^= a ^= b ^= a %= b)
    ;
  return a;
}
const int maxn = 1e5 + 9;
const int mod = 1e7 + 9;
long long sum[maxn];
int b[maxn];
int main() {
  ios::sync_with_stdio(false);
  int n, k;
  cin >> k >> n;
  sum[0] = 0;
  for (int i = 1; i <= k; i++) {
    long long x;
    cin >> x;
    sum[i] = sum[i - 1] + x;
  }
  for (int i = 1; i <= n; i++) {
    cin >> b[i];
  }
  sort(sum + 1, sum + 1 + k);
  int len = unique(sum + 1, sum + 1 + k) - sum - 1;
  int ans = len;
  for (int i = 1; i <= len; i++) {
    int m = b[1] - sum[i];
    for (int j = 1; j <= n; j++) {
      if (!(binary_search(sum + 1, sum + 1 + len, b[j] - m))) {
        ans--;
        break;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
