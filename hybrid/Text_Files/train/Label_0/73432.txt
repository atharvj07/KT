import itertools
n = int(input())
cubes = []
possible = [False] * 999
for i in range(n):
	cubes.append(list(map(int, input().split())))
for vals1 in itertools.permutations(list(range(n))):
	cur_cubes = []
	for val in vals1:
		cur_cubes.append(cubes[val])
	for vals in itertools.product(*cur_cubes):
		for x in range(len(vals) + 1):
			for chosen_vals in itertools.combinations(vals, x):
				s = ''.join(map(str, chosen_vals))
				if s:
					possible[int(s) - 1] = True
print(possible.index(False))
