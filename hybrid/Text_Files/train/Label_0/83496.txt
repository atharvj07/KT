(n, q) = map(int, input().split())
v = [0] * (n - 1)
for i in range(q):
	li = [int(i) for i in input().split()]
	v[li[1] - 1] = 2
	if li[0] == 1:
		v[li[2] - 1] = 1
ma = v.count(0)
mi = 0
for i in range(n - 1):
	if v[i] == 0:
		v[i] = 2
		mi += 1
		if i != n - 2 and v[i + 1] == 0:
			v[i + 1] = 1
print(mi, ma)
