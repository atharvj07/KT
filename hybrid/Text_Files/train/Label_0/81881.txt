def acess(i, j, cities):
	citie_1 = i + 1
	citie_2 = j + 1
	if cities[i].count(citie_2) == 1:
		return True
	for c in cities[i]:
		for c2 in cities[c - 1]:
			if c2 == citie_2:
				return True
	return False
