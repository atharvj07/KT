t = int(input())
for _ in range(t):
	(n, a, b) = map(int, input().split())
	l = list(input())
	dp1 = [10 ** 18] * (n + 1)
	dp2 = [10 ** 18] * (n + 1)
	dp1[0] = b
	i = 1
	while i < n:
		if l[i] == '0':
			if l[i - 1] != '1':
				dp1[i] = min(dp2[i - 1] + 2 * a, dp1[i - 1] + a) + b
				dp2[i] = min(dp1[i - 1] + 2 * a, dp2[i - 1] + a) + 2 * b
			else:
				dp2[i] = min(dp1[i - 1] + 2 * a, dp2[i - 1] + a) + 2 * b
			i += 1
		elif l[i] == '1':
			dp2[i] = min(dp1[i - 1] + 2 * a, dp2[i - 1] + a) + 2 * b
			i += 1
	print(min(dp1[n - 1] + a, dp2[n - 1] + 2 * a) + b)
