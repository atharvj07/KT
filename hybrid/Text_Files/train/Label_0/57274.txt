import java.util.Scanner;

public class A1190 {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long N = in.nextLong();
        int M = in.nextInt();
        long K = in.nextLong();
        long[] P = new long[M];
        for (int m=0; m<M; m++) {
            P[m] = in.nextLong();
        }
        long answer = 0;
        int pIdx = 0;
        long pStart = 1;
        while (pIdx < M) {
            long p = P[pIdx];
            pStart += K*((p-pStart)/K);
            int shift = 0;
            while (pIdx < M && P[pIdx] < pStart + K) {
                pIdx++;
                shift++;
            }
            answer++;
            pStart += shift;
        }
        System.out.println(answer);
    }

}
