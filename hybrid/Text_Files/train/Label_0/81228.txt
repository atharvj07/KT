n = int(input())
arr = list(map(int, input().split()))
res = []
for i in range(n):
	smallest = i
	for j in range(n - 1, i, -1):
		if arr[j] < arr[smallest]:
			smallest = j
	if smallest != i:
		(arr[smallest], arr[i]) = (arr[i], arr[smallest])
		res.append([smallest, i])
print(len(res))
for e in res:
	print(e[0], e[1])
