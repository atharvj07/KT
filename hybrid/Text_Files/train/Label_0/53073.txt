import java.io.*;
import java.util.*;
import java.lang.*;

public class Rextester{
    static class Contestants implements Comparable<Contestants>{
        int problems,number;
        public Contestants(int problems){
            this.problems = problems;
            number=0;
        }
        public int compareTo(Contestants o){
            return o.number-this.number;
        }
    }
    public static void main(String[] args)throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = new Integer(br.readLine());
        StringBuffer sb = new StringBuffer();
        while(t-->0){
            int n = new Integer(br.readLine());
            StringTokenizer st = new StringTokenizer(br.readLine());
            Contestants[] cs = new Contestants[n];
            int p=0;
            int a = new Integer(st.nextToken());
            cs[0]=new Contestants(a);
            cs[0].number++;
            int sum=0,max=n/2,index=0;
            for(int i=1;i<n;i++){
                int b = new Integer(st.nextToken());
                if(b==a){
                    cs[p].number++;
                }
                else{
                    p++;
                    cs[p]=new Contestants(b);
                    cs[p].number++;
                    a=b;
                }
            }
            for(int i=0;i<=p;i++){
                sum+=cs[i].number;
                if(sum>max){
                    sum-=cs[i].number;
                    index=i-1;
                    break;
                }
            }
            if(index<2){
                sb.append("0 0 0").append("\n");
                continue;
            }
            else{
                int gold=0,silver=0,bronze=0;
                gold=cs[0].number;
                int z = index/2;
                for(int i=1;i<=z;i++){
                    silver+=cs[i].number;
                }
                for(int i=z+1;i<=index;i++){
                    bronze+=cs[i].number;
                }
                if(gold>=silver&&gold>=bronze){
                    sb.append("0 0 0").append("\n");
                    continue;
                }
                else if(gold>=silver){
                    while(gold<bronze-cs[z+1].number){
                        silver+=cs[z+1].number;
                        bronze-=cs[z+1].number;
                        z++;
                    }
                    if(gold>=silver||gold>=bronze){
                        sb.append("0 0 0").append("\n");
                        continue;
                    }
                }
                else if(gold>=bronze){
                    while(gold<silver-cs[z].number){
                        silver-=cs[z].number;
                        bronze+=cs[z].number;
                        z--;
                    }
                    if(gold>=silver||gold>=bronze){
                        sb.append("0 0 0").append("\n");
                        continue;
                    }
                }
                sb.append(gold).append(" ").append(silver).append(" ").append(bronze).append("\n");
            }
        }
        br.close();
        System.out.println(sb);
    }
}
