from itertools import accumulate

def check(p):
	return const / sum_[p] >= B

def binSearch(a, b):
	(left, right) = (a - 1, b + 1)
	while right - left > 1:
		mid = (left + right) // 2
		if check(mid):
			left = mid
		else:
			right = mid
	return n - left - 1
(n, A, B) = map(int, input().split())
s = list(map(int, input().split()))
s = [s[0]] + sorted(s[1:])
sum_ = list(accumulate(s))
const = s[0] * A
print(binSearch(0, n - 1))
