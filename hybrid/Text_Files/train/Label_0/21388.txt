#include <bits/stdc++.h>

using namespace std;

using ll = long long;
using ld = long double;

struct MinimumBoundingBox {
    const ld INF = numeric_limits<ld>::max();
    const ld NINF = numeric_limits<ld>::lowest();

    int n;
    vector<ld> x, y;
    vector<char> d;

    MinimumBoundingBox(int _n) : n(_n), x(n), y(n), d(n) {}

    void input() {
        for (int i = 0; i < n; ++i) cin >> x[i] >> y[i] >> d[i];
    }

    ld solve() {
        ld min_v = f(0), lb = 0, ub = 1e9;
        for (int i = 0; i < 200; ++i) {
            const ld t1 = (2.0 * lb + ub) / 3.0;
            const ld t2 = (lb + 2.0 * ub) / 3.0;

            min_v = min(min_v, f(t1));
            min_v = min(min_v, f(t2));

            if (f(t1) > f(t2)) lb = t1;
            else ub = t2;
        }
        return min_v;
    }

    ld f(const ld t) {
        pair<ld, ld> dx(INF, NINF), dy(INF, NINF);
        for (int i = 0; i < n; ++i) {
            ld vx = x[i], vy = y[i];

            if (d[i] == 'R') vx += t;
            else if (d[i] == 'L') vx -= t;
            else if (d[i] == 'U') vy += t;
            else if (d[i] == 'D') vy -= t;

            dx.first = min(dx.first, vx);
            dx.second = max(dx.second, vx);
            dy.first = min(dy.first, vy);
            dy.second = max(dy.second, vy);
        }
        return (dx.second - dx.first) * (dy.second - dy.first);
    }
};

int main() {
    cin.tie(0); ios::sync_with_stdio(false);
    cout << setprecision(10) << setiosflags(ios::fixed);

    int n;
    cin >> n;

    MinimumBoundingBox mbb(n);
    mbb.input();
    cout << mbb.solve() << endl;

    return 0;
}
