(n, m) = list(map(int, input().split()))
matrix = [[] for i in range(n)]
m_left = 101
m_right = 101
m_top = 101
m_bottom = 101
f = False
present_cnt = 0
for i in range(n):
	s = input().strip()
	for j in range(0, len(s)):
		matrix[i].append(s[j])
		if s[j] == 'B':
			present_cnt += 1
			if f == False:
				f = True
				m_left = j
				m_right = j
				m_top = i
				m_bottom = i
			else:
				m_left = min(j, m_left)
				m_right = max(j, m_right)
				m_top = min(i, m_top)
				m_bottom = max(i, m_bottom)
if m_right == 101:
	print(1)
else:
	left_top = [m_top, m_left]
	right_botom = [m_bottom, m_right]
	hight = m_bottom - m_top + 1
	lenght = m_right - m_left + 1
	if hight < lenght:
		diff = lenght - hight
		if n - diff - hight >= 0:
			print(lenght * lenght - present_cnt)
		else:
			print(-1)
	elif hight > lenght:
		diff = hight - lenght
		if m - diff - lenght >= 0:
			print(hight * hight - present_cnt)
		else:
			print(-1)
	else:
		print(hight * hight - present_cnt)
