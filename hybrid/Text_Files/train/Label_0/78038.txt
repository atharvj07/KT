t = int(input())
for _ in range(t):
	input()
	l = list(map(int, input().split()))
	v = []
	for x in l:
		if not v:
			v.append(x)
		else:
			v.append(x ^ v[-1])
	if v[-1] == 0:
		print('YES')
	else:
		ok = False
		for i in range(1, len(l)):
			for j in range(i + 1, len(l)):
				v1 = v[i - 1]
				v2 = v[j - 1] ^ v[i - 1]
				v3 = v[-1] ^ v[j - 1]
				if v1 == v2 == v3:
					print('YES')
					ok = True
					break
			if ok:
				break
		if not ok:
			print('NO')
