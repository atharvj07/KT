import java.util.*;
public class CF354 
{

	public static void main(String[] args) 
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int k = sc.nextInt();
		
		sc.nextLine();
		String s = sc.nextLine();
		int c = 0;
		int posa[] = new int[n];
		int posb[] = new int[n];
		for(int i=0;i<n;i++)
		{
			posa[i] = -1;
			posb[i] = -1;
		}
		int x1=0,x2=0;
		for(int i=0;i<n;i++)
		{
			if(s.charAt(i)=='a')
				posa[x1++]=i;
			else
				posb[x2++]=i;
			
		}
		if(x1<=k || x2<=k)
			System.out.println(n);
		else
		{
			int max1 = posb[k];
			if(n-posb[x2-k-1]-1>max1)
				max1 = n-posb[x2-k-1]-1;
			int diff = 0;
			for(int i=k+1;i<x2;i++)
			{
				diff = posb[i] - posb[c] -1;
				c++;
				if(diff>max1)
					max1 = diff;
				//System.out.println(diff+"b"+i);
			}
			c = 0;
			int max2 = posa[k];
			if(n-posa[x1-k-1]>max2)
				max2 = n-posa[x1-k-1]-1;
			diff = 0;
			for(int i=k+1;i<x1;i++)
			{
				diff = posa[i] - posa[c] -1;
				c++;
				if(diff>max2)
					max2 = diff;
				//System.out.println(diff+"a"+i);
			}
			System.out.println((int)Math.max(max1, max2));
		}
	}
	
	
}
