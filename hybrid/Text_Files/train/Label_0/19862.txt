#include <bits/stdc++.h>
using namespace std;
void minimiza(int &a, int b) { a = min(a, b); }
void solucion(int l, int r) {
  cout << "Yes" << endl;
  cout << l + 1 << " " << r + 1 << endl;
  exit(0);
}
int k;
void get(string t, string s, vector<int> &pos, bool genera) {
  int nt = int(t.size());
  int ns = int(s.size());
  int n = nt + 1 + ns;
  vector<int> anterior(n, -1);
  string w = t + '#' + s;
  int j = -1;
  pos = vector<int>(nt, ns);
  for (int i = 1; i < n; i++) {
    while (j != -1 and w[j + 1] != w[i]) j = anterior[j];
    if (w[j + 1] == w[i]) j++;
    anterior[i] = j;
    if (genera and anterior[i] == nt - 1) {
      int l = anterior[i] - (nt - 1);
      int r = l + 2 * k;
      if (ns < r) {
        int desp = r - ns;
        l -= desp;
      }
      solucion(l, l + k);
    }
    if (0 <= anterior[i] and nt + 1 <= i and k <= i - (nt + 1) + 1)
      minimiza(pos[anterior[i]], i - (nt + 1));
  }
  for (int i = nt - 1; i > 0; i--)
    if (0 <= anterior[i]) minimiza(pos[anterior[i]], pos[i]);
}
void escribe(string s, vector<int> v) {
  cout << s << ": ";
  for (int i = 0; i < int(v.size()); i++) cout << v[i] << ",";
  cout << endl;
}
const int limite = 1000000;
int posminle[limite];
int posmaxri[limite];
int main() {
  int n, m;
  cin >> n >> m >> k;
  string s, t;
  cin >> s >> t;
  vector<int> posle;
  get(t, s, posle, true);
  vector<int> posri;
  reverse(t.begin(), t.end());
  reverse(s.begin(), s.end());
  get(t, s, posri, false);
  vector<int> nextposri = posri;
  for (int i = 0; i < m; i++) nextposri[i] = n - 1 - posri[m - 1 - i];
  posri = nextposri;
  for (int i = 0; i < m - 1; i++)
    if (posle[i] < posri[i + 1] and i + 1 <= k and m - i - 1 <= k)
      solucion(posle[i] - k + 1, posri[i + 1]);
  cout << "No" << endl;
}
