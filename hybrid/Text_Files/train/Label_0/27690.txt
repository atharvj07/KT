#include <bits/stdc++.h>
using namespace std;
int main() {
  long long l, r, k;
  cin >> l >> r >> k;
  long long msb = 1;
  while (msb <= l) msb *= 2;
  msb /= 2;
  if (k == 1) {
    cout << l << "\n1\n" << l << '\n';
  } else if (k >= 4 && (r - l >= 4 || r % 2)) {
    cout << "0\n4\n";
    l = (l + 1) / 2 * 2;
    for (int i = 0; i <= 3; i++) cout << l + i << " \n"[i == 3];
  } else if (k >= 3 && msb * 3 <= r) {
    cout << "0\n3\n";
    cout << l << ' ' << 3 * msb << ' ' << msb + l << '\n';
  } else if (r - l >= 2 || r % 2) {
    cout << "1\n2\n";
    l = (l + 1) / 2 * 2;
    for (int i = 0; i <= 1; i++) cout << l + i << " \n"[i == 3];
  } else {
    assert(r - l == 1);
    long long alt = r ^ l;
    if (alt < l) {
      cout << alt << "\n2\n" << l << ' ' << r << '\n';
    } else {
      cout << l << "\n1\n" << l << '\n';
    }
  }
}
