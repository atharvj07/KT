#include <iostream>
#include <vector>
#include <algorithm>
#include <cstring>
#include <cstdlib>
using namespace std;

int main(){

	int t;
	cin>>t;
	while(1){
		if(t==0)
			break;
		t--;
		int gx,gy;
		cin>>gx>>gy;
		int p;
		cin>>p;
		long long dp[2][16][16];
		int field[16][16];
		bool mttb[16][16][16][16];
		memset(mttb,0,sizeof(mttb));
		const int INF=-1;
		memset(field,0,sizeof(field));
		memset(dp,0,sizeof(dp));
		for(int i = 0; i < 2; i++)
			for(int j = 0; j < 16; j++)
				for(int k = 0; k < 16; k++)
					dp[i][j][k]=0;
		for(int i = 0; i < p; i++){
			int minx,miny,maxx,maxy;
			cin>>minx>>miny>>maxx>>maxy;
			if(minx>maxx){
				swap(minx,maxx);
				swap(miny,maxy);
			}
			if(miny>maxy){
				swap(miny,maxy);
				swap(minx,maxx);
			}
			mttb[miny][minx][maxy][maxx]=true;
			mttb[maxy][maxx][miny][minx]=true;
			//for(int j = minx; j <= maxx; j++)
			//	for(int k = miny; k <= maxy; k++)
			//		field[k][j]=1;
		}
		dp[0][0][0]=1;
		int v=0;
		bool f=false;
		while(1){
			f=false;
			for(int i = 0; i <= gy; i++){
				for(int j = 0; j <= gx; j++){
					if(i-1>=0&&dp[v][i-1][j]&&!mttb[i-1][j][i][j]){
						f=true;
						dp[v^1][i][j]+=dp[v][i-1][j];
					}
					if(j-1>=0&&dp[v][i][j-1]&&!mttb[i][j-1][i][j]){
						f=true;
						dp[v^1][i][j]+=dp[v][i][j-1];
					}
				}
			}
			if(dp[v^1][gy][gx]>=1){
				v^=1;
				break;
			}
			if(!f)
				break;
			for(int i = 0; i <= gy; i++){
				for(int j = 0; j <= gx; j++){
					dp[v][i][j]=0;
				}
			}
			v^=1;
		}
		if(!f)
			cout<<"Miserable Hokusai!"<<endl;
		else
			cout<<dp[v][gy][gx]<<endl;
	}

	return 0;
}
