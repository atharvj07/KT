def change(n):
	d2 = n % 10
	n = n // 10
	d1 = n
	if d1 == 0 or d2 == 0:
		d1 = 1
	else:
		d1 = 0
	return d1 * 10 + d2
frmat = int(input())
time = input()
time = time.strip().split(':')
hh = int(time[0])
mm = int(time[1])
del time
if frmat == 12:
	if hh == 0 or hh > 12:
		hh = change(hh)
if frmat == 24:
	if hh > 23:
		hh = change(hh)
if mm > 59:
	mm = change(mm)
hrs = str(hh)
mins = str(mm)
if hh < 10:
	hrs = '0' + hrs
if mm < 10:
	mins = '0' + mins
print(f'{hrs}:{mins}')
