n = int(input())
u = [['#'] * (n + 2)]
for i in range(n):
	u.append(['#'] + list(input()) + ['#'])
u.append(['#'] * (n + 2))
for i in range(1, n + 1):
	for j in range(1, n + 1):
		if u[i][j] == '.':
			if u[i + 1][j] != '#' and u[i + 1][j - 1] != '#':
				u[i + 1][j] = '#'
				u[i + 1][j - 1] = '#'
			else:
				print('NO')
				exit()
			if u[i + 1][j + 1] != '#' and u[i + 2][j] != '#':
				u[i + 1][j + 1] = '#'
				u[i + 2][j] = '#'
			else:
				print('NO')
				exit()
print('YES')
