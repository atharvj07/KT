import sys
for _ in range(int(sys.stdin.readline())):
	(N, k) = map(int, sys.stdin.readline().split())
	arr = list(map(str, input().strip()))
	dic = []
	temp = k
	lose = 0
	streak = 0
	if 'W' not in arr:
		print(max(0, 2 * k - 1))
	elif 'L' not in arr:
		print(2 * N - 1)
	else:
		for i in range(N):
			if arr[i] == 'L':
				lose += 1
				if arr[i - 1] == 'W' or i == 0:
					dic.append(1)
				else:
					dic[len(dic) - 1] += 1
			else:
				if i > 0:
					if arr[i - 1] == 'W':
						continue
				streak += 1
		(first, last) = (0, 0)
		if arr[0] == 'L':
			first = dic[0]
		if arr[N - 1] == 'L':
			last = dic[len(dic) - 1]
		dic.sort()
		for item in dic:
			if k - item >= 0:
				if item == first:
					first = 0
					continue
				elif item == last:
					last = 0
					continue
				k -= item
				streak -= 1
			else:
				break
		wins = min(N, N - lose + temp)
		print(max(2 * wins - streak, 0))
