def solve():

	def get(x):
		n = len(str(x))
		cnt = 0
		for i in range(n, 0, -1):
			cnt_num = x - pow(10, i - 1) + 1
			cnt += cnt_num * i
			x = pow(10, i - 1) - 1
		return cnt
	x = []
	s = 0
	for i in range(10):
		x1 = get(pow(10, i))
		x.append(s + x1)
		cnt = pow(10, i + 1) - pow(10, i)
		s += cnt * x1
		s1 = (cnt - 1) * (i + 1) * cnt
		s += s1 // 2

	def get1(n):
		for i in range(10):
			if x[i] > n:
				n -= x[i - 1]
				x1 = get(pow(10, i - 1))
				n += x1
				l = 1
				r = pow(10, i)
				while l <= r:
					mid = l + r >> 1
					s1 = (2 * x1 + (mid - 1) * i) * mid
					s1 //= 2
					if s1 <= n:
						ans = mid
						l = mid + 1
					else:
						r = mid - 1
				s1 = (2 * x1 + (ans - 1) * i) * ans
				s1 //= 2
				n -= s1
				return [n, pow(10, i - 1) + ans - 1]
	q = ii()
	for i in range(q):
		n = ii()
		(n, idx) = get1(n)
		if n == 0:
			print(str(idx)[-1])
			continue
		l = 1
		r = idx + 1
		while l <= r:
			mid = l + r >> 1
			if get(mid) <= n:
				ans = mid
				l = mid + 1
			else:
				r = mid - 1
		n -= get(ans)
		if n == 0:
			print(str(ans)[-1])
		else:
			print(str(ans + 1)[n - 1])
