    public String nextToken() {
        while (st == null || !st.hasMoreTokens()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (Exception e) {
                eof = true;
                return "-1";
            }
        }
        return st.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(nextToken());
    }

    final Random RNG = new Random();

    class Node<K> {
        K key;
        int priority;
        int size;
        Node<K> l;
        Node<K> r;

        public Node(K k) {
            priority = RNG.nextInt();
            this.key = k;
            size = 1;
        }
    }

    class MySetGen<K extends Comparable<K>> {


        Node<K> root;

        void add(K e) {
            if (contains(e)) {
                return;
            }
            Node<K>[] t = split(root, e);
            root = merge(t[0], new Node<K>(e));
            root = merge(root, t[1]);
        }

        private Node<K> merge(Node<K> l, Node<K> r) {
            if (l == null) {
                return r;
            }
            if (r == null) {
                return l;
            }
            Node<K> ret;
            if (l.priority > r.priority) {
                l.r = merge(l.r, r);
                ret = l;
            } else {
                r.l = merge(l, r.l);
                ret = r;
            }
            ret.size = 1;
            if (ret.l != null) {
                ret.size += ret.l.size;
            }
            if (ret.r != null) {
                ret.size += ret.r.size;
            }
            return ret;
        }

        private Node<K>[] split(Node<K> v, K e) {
            if (v == null) {
                return new Node[2];
            }
            Node<K>[] t;
            if (v.key.compareTo(e) < 0) {
                t = split(v.r, e);
                v.r = t[0];
                v.size = 1;
                if (v.l != null) {
                    v.size += v.l.size;
                }
                if (v.r != null) {
                    v.size += v.r.size;
                }
                t[0] = v;
            } else {
                t = split(v.l, e);
                v.l = t[1];
                v.size = 1;
                if (v.l != null) {
                    v.size += v.l.size;
                }
                if (v.r != null) {
                    v.size += v.r.size;
                }
                t[1] = v;
            }
            return t;
        }

        private Node<K> get(Node<K> v, K e) {
            if (v == null) {
                return null;
            }
            int comp = v.key.compareTo(e);
            if (comp == 0) {
                return v;
            } else if (comp < 0) {
                return get(v.r, e);
            } else {
                return get(v.l, e);
            }
        }

        boolean contains(K e) {
            return get(root, e) != null;
        }

    }

    void solve() {
        int n = nextInt();
        int k = nextInt();
        int u, v;

        TreeSet<Integer>[] vv = new TreeSet[n];
        for (int i = 0; i < n; i++) {
            vv[i] = new TreeSet<>();
        }
        for (int i = 0; i < n - 1; i++) {
            u = nextInt() - 1;
            v = nextInt() - 1;
            vv[u].add(v);
            vv[v].add(u);
        }

        Queue<Integer> prior = new ArrayDeque<>();
        Queue<Integer> vert = new ArrayDeque<>();

        int[] pp = new int[n];
        for (int i = 0; i < n; i++) {
            pp[i] = -1;
        }

        for (int i = 0; i < n; i++) {
            if (vv[i].size() == 1) {
                prior.add(0);
                vert.add(i);
                pp[i] = 0;
            }
        }
        try {
            int proc = 0, pr;
            int[] son = new int[n];
            while (!prior.isEmpty()) {
                u = vert.remove();
                pr = prior.remove();
                if (pr > k) {
                    out.print("No");
                    return;
                }
                proc++;
                if (pr == 0) {
                    //leaf
                    v = vv[u].first();
                    son[v]++;
                    vv[v].remove(u);
                    if (pp[v] == -1) {
                        pp[v] = pr + 1;
                    }
                    if (pp[v] != pr + 1) {
                        out.print("No");
                        return;
                    }
                    if ((vv[v].size() == 1 && pr < k - 1) || (vv[v].size() == 0 && proc == n - 1)) {
                        prior.add(pr + 1);
                        vert.add(v);
                    }
                } else if (vv[u].size() == 0) {
                    //root
                    if (proc != n || son[u] < 3 || pr != k) {
                        out.print("No");
                        return;
                    }
                } else {
                    //middle

                        out.print("No");

                        pp[v] = pr + 1;

                        out.print("No");
                        return;
                    }
                    if ((vv[v].size() == 1 && pr < k - 1) || (vv[v].size() == 0 && proc == n - 1)) {
                        prior.add(pr + 1);
                        vert.add(v);
                    }
                }
            }


            if (proc == n) {

                out.print("No");

            out.print("No");
