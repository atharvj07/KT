def run(n):
	l = len(n) - 1
	b = 1989
	l = 10 ** l
	while l > 1:
		b += l
		l //= 10
	if b <= int(n):
		return n
	b = str(b)
	if len(b) == len(n):
		return '1' + n
	else:
		if b[-len(n):] <= n:
			return b[:-len(n)] + n
		return str(int(b[:-len(n)]) + 1) + n
n = int(input())
for i in range(n):
	print(run(input()[4:]))
