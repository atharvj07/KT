#include <iostream>
#include <utility>
#include <vector>
#include <utility>
#include <algorithm>
#define inf 1e9

using namespace std;
typedef pair<int, int> P;

int n, m;
int a[1005][45];
int G[45][45];
int mask[45], mask2[25];
vector<P> vec, vec2;
int dp[1<<20], dp2[1<<20];

void dfs(int p, int l, int r, int s, int cnt, vector<P> &vec)
{
  if(p > r){
    vec.push_back(make_pair(s, cnt));
    return;
  }
  dfs(p+1, l, r, s, cnt, vec);
  if(mask[p] & s) return;
  dfs(p+1, l, r, s|1<<(p-l), cnt+1, vec);
}

int main(void)
{
  while(1){
    cin >> n >> m;
    if(n == 0 && m == 0) break;
    for(int i = 1; i <= n; i++){
      for(int j = 0; j < m; j++) cin >> a[i][j];
    }
    if(m == 1){
      cout << 1 << endl;
      continue;
    }

    for(int i = 0; i < m; i++){
      for(int j = 0; j < m; j++){
        G[i][j] = 0;
      }
    }
    for(int i = 0; i < m; i++){
      for(int j = 0; j < m; j++){
        if(i >= j) continue;
        vector<P> vec;
        for(int k = 1; k <= n; k++){
          vec.push_back(make_pair(a[k][i], a[k][j]));
        }
        sort(vec.begin(), vec.end());
        for(int k = 1; k < vec.size(); k++){
          if(vec[k-1].second > vec[k].second){
            G[i][j] = G[j][i] = 1;
            break;
          }
        }
      }
    }

    /*for(int i = 0; i < m; i++){
      for(int j = 0; j < m; j++){
        cout << G[i][j] << " ";
      }
      cout << endl;
    }*/

    for(int i = 0; i < m/2; i++){
      mask[i] = 0;
      for(int j = 0; j < m/2; j++) if(G[i][j]) mask[i] |= 1<<j;
    }
    for(int i = m/2; i < m; i++){
      mask[i] = 0;
      for(int j = m/2; j < m; j++) if(G[i][j]) mask[i] |= 1<<(j-m/2);
    }

    vec.clear(), vec2.clear();
    dfs(0, 0, m/2-1, 0, 0, vec);
    dfs(m/2, m/2, m-1, 0, 0, vec2);

    int S = 1<<(m/2), S2 = 1<<(m-m/2);
    for(int i = 0; i < S2; i++) dp2[i] = -inf;
    for(int i = 0; i < vec2.size(); i++) dp2[vec2[i].first] = vec2[i].second;
    for(int i = 0; i < S2; i++){
      for(int j = 0; j < m-m/2; j++){
        if(i && 1<<j) dp2[i] = max(dp2[i], dp2[i & ~(1<<j)]);
      }
    }

    for(int i = 0; i < m/2; i++){
      mask2[i] = 0;
      for(int j = m/2; j < m; j++) if(G[i][j]) mask2[i] |= 1<<(j-m/2);
    }

    int ans = 0;
    for(int i = 0; i < vec.size(); i++){
      int adj = 0;
      for(int j = 0; j < m/2; j++) if(vec[i].first & (1<<j)) adj |= mask2[j];
      ans = max(ans, vec[i].second + dp2[((S2-1)-adj)]);
    }
    cout << ans << endl;
  }

  return 0;
}
