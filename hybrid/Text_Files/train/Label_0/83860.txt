s = []
indexes = []
k = int(input(''))
q = input('')
for i in range(0, len(q)):
	if q[i] not in s:
		s.append(q[i])
		indexes.append(i)
if len(s) < k:
	print('NO')
else:
	print('YES')
	for x in range(len(indexes)):
		if x >= k - 1:
			a = q[indexes[x]:len(q)]
			print(a)
			break
		else:
			a = q[indexes[x]:indexes[x + 1]]
			print(a)
