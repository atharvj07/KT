#include <bits/stdc++.h>

#define REP(i, n) for ( int i = 0; i < (n); i++ )

int a[100][100], b[100][100];

int main(){

    int n, m, l;
    std::cin >> n >> m >> l;

    REP(i, n) REP(j, m) std::cin >> a[i][j];
    REP(j, m) REP(k, l) std::cin >> b[j][k];

    REP(i, n) {

        REP(k, l) {

            long long sum = 0;
            REP(j, m) sum += a[i][j] * b[j][k];

            if ( k ) std::cout << " ";
            std::cout << sum;
        }

        std::cout << std::endl;
    }
    return 0;
}
