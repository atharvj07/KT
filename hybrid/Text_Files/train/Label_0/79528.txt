from math import *
import heapq as h
from collections import Counter
mod = 1000000007

def pro(arr, k):
	freq = {}
	n = len(arr)
	ans = [0] * n
	for i in range(n):
		if arr[i] in freq:
			if len(freq[arr[i]]) == k:
				ans[i] = 0
				continue
			freq[arr[i]].append(i)
		else:
			freq[arr[i]] = [i]
	tot = 0
	for (i, j) in freq.items():
		tot = tot + len(j)
	tot = tot - tot % k
	c = 1
	for (i, j) in freq.items():
		if tot == 0:
			break
		for p in j:
			if tot == 0:
				break
			ans[p] = c
			tot -= 1
			if c == k:
				c = 1
			else:
				c += 1
	print(*ans)
t = int(input())
for i in range(t):
	(n, k) = list(map(int, input().split()))
	arr = list(map(int, input().split()))
	pro(arr, k)
