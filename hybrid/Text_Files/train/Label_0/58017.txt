static int bs(int x,int l,int r){
    while(l<=r){
        int m=(l+r)/2;
        if(a[m]>=x)r=m-1;
        else l=m+1;
    }
    return l;
}
static void solve() throws IOException {
        int[] z=int_arr();
        int n=z[0],l=z[1],r=z[2];
        long ans=0;
        a=int_arr();
        sort(a);
        int i=0,j=n-1;
        while(i<j){
            while(j>i&&a[j]>r-a[i])j--;
            if(j<=i)break;
            ans+=j-bs(l-a[i],i+1,j)+1;
            i++;
        }
        out.write(ans+"\n");
}

            solve();
            cn++;
        }

}
