#include <bits/stdc++.h>
using namespace std;
int n;
int get(int i) {
  cout << "? " << i << endl;
  int ans;
  cin >> ans;
  return ans;
}
int pa(int i) {
  int l = get(i), r = get(i + n);
  if (l == r) {
    cout << "! " << i << endl;
    exit(0);
  }
  if (l > r) return -1;
  return 1;
}
int main() {
  cin >> n;
  n /= 2;
  if (n % 2 == 1) {
    cout << "! -1" << endl;
    return 0;
  }
  int prev = pa(1);
  int lo = 1, hi = 1 + n;
  while (lo + 1 < hi) {
    int mi = (lo + hi) / 2;
    int nx = pa(mi);
    if (nx == prev)
      lo = mi;
    else
      hi = mi;
  }
  assert(false);
}
