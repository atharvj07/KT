import math
n = int(input())
c = []
for i in range(n):
	s = input()
	c.append(s)
l = 0
for i in range(len(c)):
	if c[i].count('C') >= 1:
		k = c[i].count('C')
		l += k * (k - 1) // 2
d = {}
for i in range(n + 1):
	d[i] = 0
for i in range(len(c)):
	for j in range(len(c[i])):
		if c[i][j] == 'C':
			d[j] += 1
for key in d:
	if d[key] >= 1:
		l += d[key] * (d[key] - 1) // 2
print(l)
