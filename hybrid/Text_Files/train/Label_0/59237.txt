import java.util.*;
import java.lang.*;
import java.io.*;
import java.math.BigInteger;

public class Ideone
{
	public static void main (String[] args)
	{
		Scanner in= new Scanner(System.in);
		int x=in.nextInt(); 
		int y=in.nextInt();
		int tmp = x - 1;
		BigInteger sum = BigInteger.valueOf(0);
		
		for(int i=0;i<x;i++)
		{
		  int s=in.nextInt();
		  BigInteger tmpBI = BigInteger.valueOf(s);
		  for(int j=0;j<tmp;j++) 
		  {
	            tmpBI = tmpBI.multiply(BigInteger.valueOf(y));
		  }
		  sum = sum.add(tmpBI);
		  tmp--;
		  
		}
	        
		x=in.nextInt(); 
		y=in.nextInt();
		tmp = x - 1;
		BigInteger sum2 = BigInteger.valueOf(0);
		
		for(int i=0;i<x;i++)
		{
		  int s=in.nextInt();
		  BigInteger tmpBI = BigInteger.valueOf(s);
		  for(int j=0;j<tmp;j++) 
		  {
	            tmpBI = tmpBI.multiply(BigInteger.valueOf(y));
		  }
		  sum2 = sum2.add(tmpBI);
		  tmp--;
		}

		if (sum.compareTo(sum2) == 0)
			System.out.println("=");
		else if (sum.compareTo(sum2) < 0)
			System.out.println("<");
		else
			System.out.println(">");
			
	}
}
