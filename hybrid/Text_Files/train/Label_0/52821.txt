import java.awt.*;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Abc {
    public static void main(String[] args) throws IOException {
        FastReader sc = new FastReader();
        long q=sc.nextLong();
        ArrayList<Long> arr=prime(q);
        if (arr.size()==2){
            System.out.println(2);
        }else {
            System.out.println(1);
            if (arr.size()<=1) System.out.println(0);
            else System.out.println(arr.get(0)*arr.get(1));
        }
    }
    static ArrayList<Long> prime(long n){
        ArrayList<Long> arr=new ArrayList<>();
        for (long i=2;i*i<=n;i++){
            if (n%i==0){
                while (n%i==0){
                    arr.add(i);
                    n/=i;
                }
            }
        }
        if (n>1)arr.add(n);
        return arr;
    }
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
}
