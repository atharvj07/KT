#include <bits/stdc++.h>
using namespace std;
int bin(int a, int b, int c) {
  a--;
  b++;
  while (b - a > 1) {
    long long m = (a + b) / 2;
    if (m * m > c)
      b = m;
    else
      a = m;
  }
  return a;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, ans = -1e9;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    if (bin(1, 1e9, a) * bin(1, 1e9, a) != a || a < 0) ans = max(ans, a);
  }
  cout << ans;
  return 0;
}
