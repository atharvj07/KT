t = int(input())
for i in range(t):
	n = int(input())
	p = list(map(int, input().split()))
	p.sort()
	l = []
	j = 0
	k = 4 * n - 1
	flag = 0
	while j < k:
		l.append(p[j] * p[k])
		j = j + 1
		k = k - 1
	for z in range(0, 4 * n, 2):
		if p[z] != p[z + 1]:
			flag = 1
			break
	if len(set(l)) == 1 and flag == 0:
		print('YES')
	else:
		print('NO')
