n = int(input())
(l, r) = ([], [])
for i in range(n):
	x = list(map(int, input().split()))
	l.append(x[0])
	r.append(x[1])
(L, R) = (sum(l), sum(r))
ans = abs(L - R)
ind = 0
for i in range(n):
	x = L - l[i] + r[i]
	y = R - r[i] + l[i]
	if ans < abs(x - y):
		ans = max(ans, abs(x - y))
		ind = i + 1
print(ind)
