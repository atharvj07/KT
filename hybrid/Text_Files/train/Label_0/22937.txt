#include <bits/stdc++.h>
using namespace std;
string s;
char ch[6] = {'I', 'E', 'A', 'O', 'U', 'Y'};
vector<int> b, a;
long long f[501000], sum[501000];
int main() {
  cin >> s;
  set<char> vowel(ch, ch + 6);
  int len = s.length();
  int cnt = 0;
  for (int i = 0; i < s.length(); i++)
    if (vowel.find(s[i]) != vowel.end()) {
      cnt++;
      int l = i + 1;
      int r = len - i;
      if (l > r) swap(l, r);
      a.push_back(l);
      b.push_back(r);
      f[l] += l;
    }
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  for (int i = 1; i <= 500000; i++) sum[i] = sum[i - 1] + f[i];
  double ans = 0;
  for (int i = 1; i <= s.length(); i++) {
    int f1 = a.end() - lower_bound(a.begin(), a.end(), i);
    int f2 = b.end() - lower_bound(b.begin() + f1, b.end(), i);
    long long y = 0;
    if (f2 > 0) y = sum[a[f2 - 1]];
    ans +=
        ((long long)f1 * i + y + (long long)(cnt - f1 - f2) * (len + 1 - i)) *
        1.0 / i;
  }
  printf("%.10f\n", ans);
  return 0;
}
