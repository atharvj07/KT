def solve():
	n = ii()
	a = li()
	ans = 0
	for i in range(0, n, 2):
		c = 0
		open_needed = 0
		for j in range(i + 1, n):
			if j % 2 == 0:
				c += a[j]
			else:
				rem_op = a[i] - open_needed
				rem_cls = a[j] - c
				ans += max(0, min(min(a[j], rem_cls + 1), min(a[i], rem_op + 1)))
				c -= a[j]
				if c < 0:
					open_needed += abs(c)
					c = 0
	print(ans)
if file:

solve()
