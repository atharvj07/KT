#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int x = 0, y = 0, z, x1, x2, y1, y2;
  cin >> x >> y;
  z = abs(x) + abs(y);
  if (x > 0) {
    x1 = 0;
    y1 = z * y / abs(y);
    x2 = z;
    y2 = 0;
  }
  if (x < 0) {
    x1 = -z;
    y1 = 0;
    x2 = 0;
    y2 = z * y / abs(y);
  }
  cout << x1 << " " << y1 << " " << x2 << " " << y2 << endl;
  return 0;
}
