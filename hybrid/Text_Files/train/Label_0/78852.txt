from sys import stdin
(n, m) = map(int, stdin.readline().strip().split())
s = input()
dp = [[0 for i in range(2004)] for j in range(2004)]
mod = 10 ** 9 + 7
d = n - m
ans = 0
b = 0
d1 = 0
for i in s:
	if i == '(':
		b += 1
	else:
		b -= 1
	if b < d1:
		d1 = b
d1 = -d1
dp[0][0] = 1
d += 1
try:
	for i in range(1, d):
		dp[i][0] = dp[i - 1][1] % mod
		for j in range(1, d):
			dp[i][j] = (dp[i - 1][j - 1] % mod + dp[i - 1][j + 1] % mod) % mod
	for i in range(d):
		x = n - (i + m)
		if x < 0:
			continue
		for j in range(d1, d):
			if b + j > 2000:
				break
			ans = (ans % mod + dp[i][j] % mod * dp[x][b + j] % mod % mod) % mod
	print(ans)
except:
	print(ans, i, j, x, b + j)
