#include <bits/stdc++.h>
using namespace std;
int n, x[200010];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> x[i];
  sort(x, x + n);
  int lo = -1, hi = 1e9;
  while (lo + 1 < hi) {
    int mi = (lo + hi) / 2;
    int cur = 0;
    for (int j = 0; j < 3; j++) {
      cur = upper_bound(x, x + n, x[cur] + mi) - x;
      if (cur == n) break;
    }
    if (cur == n)
      hi = mi;
    else
      lo = mi;
  }
  cout << setprecision(11) << hi / 2. << endl;
  int cur = 0;
  for (int j = 0; j < 3; j++) {
    cout << x[cur] + hi / 2. << " ";
    cur = upper_bound(x, x + n, x[cur] + hi) - x;
  }
}
