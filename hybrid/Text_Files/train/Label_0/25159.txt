#include <bits/stdc++.h>
using namespace std;
int N, i;
double H, K, x, y, xp, yp;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> N >> K >> x >> y;
  for (i = 1; i < N; i++) {
    cin >> xp >> yp;
    H += sqrt((xp - x) * (xp - x) + (yp - y) * (yp - y)) * K / 50;
    x = xp;
    y = yp;
  }
  cout << fixed << setprecision(12) << H << '\n';
}
