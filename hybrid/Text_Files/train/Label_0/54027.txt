import java.util.Arrays;
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class B111 {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int last[] = new int[100001];
        Arrays.fill(last, 0);

        for (int i = 1; i <= n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            int count = 0;
            Integer[] divs = getd(x);
            for (int d : divs) {
                if (y == 0) {
                    last[d] = i;
                    count++;
                } else {//d <= i - 1 && d >= i - y && 
                    if (last[d]  <= i - 1 && last[d] >= i - y && last[d] != 0) {
                        last[d] = i;
                    } else {
                        last[d] = i;
                        count++;
                    }
                }
            }
            System.out.println(count);
        }
    }

    private static Integer[] getd(int x) {
        Set<Integer> v = new HashSet<Integer>();
        for (int k = 1; k <= Math.sqrt(x); k++) {
            if (x % k == 0) {
                v.add(k);
                v.add(x / k);
            }
        }
        Integer[] a = new Integer[v.size()];
        return v.toArray(a);
    }
}
