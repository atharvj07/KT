#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, aux, neg = 0, sum = 0;
  cin >> n;
  int tot = 2 * n - 1;
  int mini = (int)1e9;
  for (int i = 0; i < tot; i++) {
    scanf("%d", &aux);
    sum += abs(aux);
    mini = min(abs(aux), mini);
    if (aux < 0) neg++;
  }
  if (n % 2 == 0 && neg % 2 == 1) {
    cout << (sum - 2 * mini) << endl;
  } else {
    cout << sum << endl;
  }
  return 0;
}
