import sys
(_ord, inp, num, neg, _Index) = (lambda x: x, [], 0, False, 0)
(i, s) = (0, sys.stdin.buffer.read())
try:
	while True:
		if s[i] >= b'0'[0]:
			num = 10 * num + _ord(s[i]) - 48
		elif s[i] == b'-'[0]:
			neg = True
		elif s[i] != b'\r'[0]:
			inp.append(-num if neg else num)
			(num, neg) = (0, False)
		i += 1
except IndexError:
	pass
if s and s[-1] >= b'0'[0]:
	inp.append(-num if neg else num)

def inin(size=None):
	global _Index
	if size == None:
		ni = _Index
		_Index += 1
		return inp[ni]
	else:
		ni = _Index
		_Index += size
		return inp[ni:ni + size]
_T_ = 1
for _t_ in range(_T_):
	n = inin()
	a = inin(n)
	d = [0 for i in range(n - 1)]
	k = 0
	for i in range(1, n):
		d[i - 1] += a[i] - a[i - 1]
		k += max(0, a[i] - a[i - 1])
	x = (a[0] + k + 1) // 2
	print(x)
	q = inin()
	for _ in range(q):
		(l, r, c) = inin(3)
		l -= 1
		r -= 1
		if l == 0:
			a[0] += c
		else:
			if d[l - 1] >= 0:
				k -= d[l - 1]
			d[l - 1] += c
			if d[l - 1] >= 0:
				k += d[l - 1]
		if r == n - 1:
			pass
		else:
			if d[r] >= 0:
				k -= d[r]
			d[r] -= c
			if d[r] >= 0:
				k += d[r]
		x = (a[0] + k + 1) // 2
		print(x)
