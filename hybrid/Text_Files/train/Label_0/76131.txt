from sys import stdin

def from_start(string, n):
	char = string[0]
	i = 0
	while i < n - 1:
		if string[i] != char:
			break
		i += 1
	return i

def from_end(string, n):
	char = string[-1]
	i = n - 1
	while i > 0:
		if string[i] != char:
			break
		i -= 1
	return n - i - 1

def from_mid(string, first, second):
	if string[0] == string[-1] and len(string) > 2:
		start = first
		end = len(string) - second - 1
		if start <= end:
			third = start - 1
			third += n - 2 - end
			third += (start - 1) * (n - 2 - end)
			return third + 1
		else:
			return 0
	else:
		return 0
n = int(stdin.readline().strip())
string = list(stdin.readline().strip())
first = from_start(string, n)
second = from_end(string, n)
third = from_mid(string, first, second)
print((first + second + third + 1) % 998244353)
