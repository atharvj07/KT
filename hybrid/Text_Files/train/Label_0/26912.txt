#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  string s;
  cin >> s;
  int n = s.length();
  s = '#' + s;
  vector<int> pre(n + 2, 0);
  vector<int> suf(n + 2, 0);
  if (islower(s[1])) pre[1] = 1;
  for (int i = 2; i <= n; i++) {
    pre[i] = pre[i - 1] + (s[i] >= 'a' && s[i] <= 'z');
  }
  (s[n] >= 'A' && s[n] <= 'Z') ? suf[n] = 1 : suf[n] = 0;
  for (int i = n - 1; i >= 1; i--) {
    suf[i] = suf[i + 1] + (s[i] >= 'A' && s[i] <= 'Z');
  }
  int ans = INT_MAX;
  for (int i = 1; i <= n + 1; i++) {
    ans = min(ans, (pre[i - 1] + suf[i]));
  }
  cout << ans << endl;
  return 0;
}
