import java.io.*;
import java.util.*;





public class C {
	String s = null;
	String[] ss = null;

	
	public void run() throws Exception{

		BufferedReader br = null;
		File file = new File("input.txt");
		if(file.exists()){
			br = new BufferedReader(new FileReader("input.txt"));
		}
		else{
			br = new BufferedReader(new InputStreamReader(System.in));
		}
	
		s = br.readLine();
		String a = s;
		s = br.readLine();
		int k = Integer.parseInt(s);
		if(k >= a.length()){
			System.out.println("0");
			System.out.println("");
			return;
		}
		int[] n = new int[26];
		s = a;
		for(int i = 0; i < s.length(); i++){
			char c = s.charAt(i);
			int p = c - 'a';
			n[p]++;
		}

		
		int INF = Integer.MAX_VALUE / 10000;
		for(int i = 0; i < n.length; i++){
			if(n[i] == 0){
				n[i] = INF;
			}
		}
		
		List<Character> del = new ArrayList<Character>();
		Set<Character> delm = new HashSet<Character>();
		int lastdel = -1;
		while(true){
			int min = INF;
			int mp = -1;
			for(int i = 0; i < n.length; i++){
				if(n[i] < min){
					mp = i;
					min = n[i];
				}
			}
			if(min <= k){
				k -= min;
				lastdel = min;
				char add = (char)('a' + mp);
				del.add(add);
				delm.add(add);
				n[mp] = INF;
			}
			else{
				break;
			}
		}
		
		
		StringBuilder sb = new StringBuilder();
		Set<Character> anss = new HashSet<Character>();
		for(int i = 0; i < s.length(); i++){
			char c = s.charAt(i);
			if(delm.contains(c)){
				continue;
			}
			sb.append(c);
			anss.add(c);
		}
		System.out.println(anss.size());
		System.out.println(sb.toString());
		
			
		
		
		
		
	}
	
	
	/**
	 * @param args
	 */
	public static void main(String[] args) throws Exception{
		C t = new C();
		t.run();

	}
	
}
