import java.util.Scanner;

public class TaskD{
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String input = in.next();
        int[] a = new int[n];
        a[0]=1;
        for (int i = 1; i < n; i++) {
            if(input.charAt(i)=='['){
                a[i]=a[i-1]+1;
            } else {
                a[i]=a[i-1]-1;
            }
        }
        int max = 0;
        for(int t:a){
            max = t>max?t:max;
        }
        int maxH = 1 + (max-1) * 2;
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            if(input.charAt(i)=='['){
                h[i] = 1 + 2*(max - a[i]);
            } else {
                a[i]++;
                h[i] = 1 + 2*(max - a[i]);
            }
        }
        int countWhitespace = 0;
        for (int i = 1; i < n; i++) {
            if(input.charAt(i)==']' && input.charAt(i-1)=='[') countWhitespace++;
        }
        countWhitespace*=3;
        char[][] res = new char[n][maxH+2];
        for (int i = 0; i < n; i++) {
            int j =0;
            while(j<a[i]-1){
                res[i][j] = ' ';
                j++;
            }
            res[i][j]='+';
            j++;
            for (int k = 0; k < h[i]; k++) {
                res[i][j]='|';
                j++;
            }
            res[i][j]='+';
            j++;
            while(j<maxH+2){
                res[i][j] = ' ';
                j++;
            }
        }

        char[][] result = new char[n+countWhitespace][maxH+2];
        for (int i = 0; i < maxH+2; i++) {
            for (int e = 0; e < n + countWhitespace; e++) {
                result[e][i]=' ';
            }

        }
        int j = 0;
        for (int i = 0; i < n; i++) {
            if(input.charAt(i)=='['){
                for (int k = 0; k < maxH+2; k++) {
                    if(result[j][k]!='-')
                    result[j][k] = res[i][k];
                    if(res[i][k]=='+'){
                        result[j+1][k] ='-';
                    }
                }
                if(input.charAt(i+1)==']') j+=4; else j+=1;
            }else{
                for (int k = 0; k < maxH+2; k++) {
                    if(result[j][k]!='-')
                    result[j][k] = res[i][k];
                    if(res[i][k]=='+'){
                        result[j-1][k] ='-';
                    }
                }
                j+=1;
            }
        }
        for (int i = 0; i < maxH+2; i++) {
            for (int e = 0; e < n + countWhitespace; e++) {
                System.out.print(result[e][i]);
            }
            System.out.println();
        }
    }
}
