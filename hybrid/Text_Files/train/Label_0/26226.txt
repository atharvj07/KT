#include <bits/stdc++.h>
using namespace std;
int a[11111111], b[11111111];
int main() {
  int sum1 = 0, sum2 = 0;
  int n, m;
  cin >> n;
  int i, j, k;
  for (i = 1; i <= n; i++) scanf("%d", &a[i]);
  cin >> m;
  for (i = 1; i <= m; i++) scanf("%d", &b[i]);
  sort(a + 1, a + 1 + n);
  sort(b + 1, b + 1 + m);
  int ans = -0x3f3f3f3f;
  int ans1 = 0, ans2 = 0;
  a[n + 1] = 0x3f3f3f3f;
  for (i = 1; i <= n + 1; i++) {
    sum1 = (i - 1) * 2 + (n - (i - 1)) * 3;
    int p = a[i];
    int ma = m + 1, mi = 0, an = -1, mid;
    while (ma >= mi) {
      mid = (ma + mi) >> 1;
      if (b[mid] >= p) {
        an = mid;
        ma = mid - 1;
      } else
        mi = mid + 1;
    }
    if (an == -1) {
      sum2 = m * 2;
    } else
      sum2 = (an - 1) * 2 + (m - an + 1) * 3;
    if (sum1 - sum2 > ans) {
      ans = sum1 - sum2;
      ans1 = sum1;
      ans2 = sum2;
    }
  }
  cout << ans1 << ":" << ans2 << endl;
  return 0;
}
