t = int(input())
for T in range(t):
	n = int(input())
	a = [int(x) for x in input().split()]
	temp = 0
	for i in range(n - 1):
		for j in range(i + 1, n):
			l = a[i] - a[j] + i - j
			if l < 0:
				l = -l
			if l % n == 0:
				temp = 1
				break
		if temp == 1:
			break
	if temp == 0:
		print('YES')
	else:
		print('NO')
