#include <bits/stdc++.h>
using namespace std;
const long long N = 1e5 + 5;
long long arr[N], A[N], ARR[N];
bool visited[N] = {};
long long cnt[N], freq[N];
bool f, ff;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, s, a, b, ans(0);
  cin >> n >> s;
  vector<int> v;
  for (int i = 0; i < n; ++i) {
    cin >> a >> b;
    v.push_back(a * 60 + b);
  }
  sort(((v).begin()), ((v).end()));
  for (int i = 0; i < n; ++i) {
    if (v[i] - ans > s) break;
    ans = v[i] + s + 1;
  }
  cout << ans / 60 << ' ' << ans % 60;
  return 0;
}
