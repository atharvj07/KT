t = int(input())
for i in range(0, t):
	n = int(input())
	a = list(map(int, input().split()))
	temp = sorted(a)
	count = 0
	flag = 0
	j = 1
	while a != temp:
		if j % 2:
			k = 0
			while k < n - 1:
				if a[k] > a[k + 1]:
					x = a[k]
					a[k] = a[k + 1]
					a[k + 1] = x
				k += 2
		else:
			k = 1
			while k < n - 1:
				if a[k] > a[k + 1]:
					x = a[k]
					a[k] = a[k + 1]
					a[k + 1] = x
				k += 2
		count += 1
		j += 1
	print(count)
