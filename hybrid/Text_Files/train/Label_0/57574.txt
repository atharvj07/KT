import java.io.*;
import java.util.ArrayList;
import java.util.StringTokenizer;

public class TaskE {
    PrintWriter out;
    BufferedReader in;
    StringTokenizer st;

    public String nextToken() throws IOException {
        if (st == null || !st.hasMoreElements()) {
            String line = in.readLine();
            if (line == null)
                return null;
            st = new StringTokenizer(line);
        }
        return st.nextToken();
    }

    public void open() throws IOException {
        in = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(new OutputStreamWriter(System.out));
    }

    public int nextInt() throws NumberFormatException, IOException {
        return Integer.parseInt(nextToken());
    }

    public long nextLong() throws NumberFormatException, IOException {
        return Long.parseLong(nextToken());
    }

    public double nextDouble() throws NumberFormatException, IOException {
        return Double.parseDouble(nextToken());
    }

    public void close() {
        out.flush();
        out.close();
    }

    public TaskE() throws FileNotFoundException {
    }

    public static void main(String[] str) throws IOException {
        TaskE solver = new TaskE();
        solver.open();
        solver.solve();
        solver.close();
    }

    ArrayList<ArrayList<Integer>> g = new ArrayList<>();
    int n, target;
    long answ = 0;
    int[] pi = new int[300005];
    String t;
    String[] edges;

    public void solve() throws IOException {
        n = nextInt();
        for (int i = 0; i <= n; i++) {
            g.add(new ArrayList<Integer>());
        }
        edges = new String[n + 1];
        for (int i = 2; i <= n; i++) {
            g.get(nextInt()).add(i);
            edges[i] = nextToken();
        }
        t = nextToken() + "#";
        target = t.length() - 1;
        edges[1] = t;
        pi[0] = 0;
        slv(1, 0);
        out.println(answ);
    }

    private void slv(int v, int pos) {
        String str = edges[v];
        int len = str.length();
        for (int i = (v != 1 ? 0 : 1); i < len; i++) {
            int j = pi[i + pos - 1];
            while (j > 0 && str.charAt(i) != t.charAt(j))
                j = pi[j - 1];
            if (str.charAt(i) == t.charAt(j)) j++;
            pi[i + pos] = j;
            if (pi[i + pos] == target) {
                answ++;
                //out.println(v + " " + i);
            }
        }
        ArrayList<Integer> children = g.get(v);
        for (int i = 0; i < children.size(); i++) {
            slv(children.get(i), pos + len);
        }
    }
}
