#include "bits/stdc++.h"
using namespace std;
using ll = long long;



class SegmentTree {
public:
	SegmentTree() {}
	SegmentTree(ll size) {
		n = 1;
		while (n < size)n *= 2;
		dat.resize(2 * n, 0);
	}
	void update(ll k, ll a) {
		k += n - 1;
		dat[k] = a;
		while (k > 0) {
			k = (k - 1) / 2;
			dat[k] = max(dat[k * 2 + 1], dat[k * 2 + 2]);
		}
	}
	ll query(ll a, ll b) {
		return query_impl(a, b, 0, 0, n);
	}
private:
	ll n;
	vector<ll> dat;

	ll query_impl(ll a, ll b, ll k, ll l, ll r) {
		if (r <= a || b <= l)return 0;
		if (a <= l && r <= b)return dat[k];

		ll vl = query_impl(a, b, k * 2 + 1, l, (l + r) / 2);
		ll vr = query_impl(a, b, k * 2 + 2, (l + r) / 2, r);
		return max(vl, vr);
	}
};


ll N, M, C;
SegmentTree stl, str;

bool judge(ll d) {
	vector<ll> res(N);
	for (ll i = 0; i < N; i++) {
		ll a = stl.query(max(0ll, i - d), i + 1) - C * i;
		ll b = str.query(i, min(N - 1, i + d) + 1) - C * (N - 1 - i);
		res[i] = max(a, b);
	}
	return accumulate(res.begin(), res.end(), 0ll) >= M;
}

int main() {
	cin >> N >> M >> C;
	stl = SegmentTree(N);
	str = SegmentTree(N);
	for (int i = 0; i < N; i++) {
		ll ai;
		cin >> ai;
		stl.update(i, ai + C * i);
		str.update(i, ai + C * (N - 1 - i));
	}

	int l = 0, r = N;
	while (r - l > 1) {
		int mid = (l + r) / 2;
		if (judge(mid))r = mid;
		else l = mid;
	}
	if (r == N && !judge(r)) {
		cout << -1 << endl;
	}
	else {
		if (judge(l))cout << l << endl;
		else cout << r << endl;
	}
	return  0;
}
