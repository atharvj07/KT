import java.io.PrintWriter;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class StronglyConnectedCity {

    static FastReader in = new FastReader();
    static PrintWriter out = new PrintWriter(System.out);
    static final int[] dx = { 1 , 0 , -1 , 0 };
    static final int[] dy = { 0 , 1 , 0 , -1 };
    static final String D = "v>^<";
    // down // right // up // left

    public static void main(String[] args) {
        int n = in.nextInt() , m = in.nextInt();
        String[][] grid = new String[n][m];
        for(int i=0 ; i<n ; i++){
            Arrays.fill(grid[i],"");
        }
        String input = in.next();
        for(int i=0 ; i<n ; i++){
            for(int j=0 ; j<m ; j++){
                grid[i][j] += input.charAt(i);
            }
        }
        input = in.next();
        for(int i=0 ; i<m ; i++){
            for(int j=0 ; j<n ; j++){
                grid[j][i] += input.charAt(i);
            }
        }

        boolean fail = false;
        main:
        for(int i=0 ; i<n ; i++){
            for(int j=0 ; j<m ; j++){
                boolean[][] v = new boolean[n][m];
                v[i][j] = true;
                dfs(i,j,v,grid,n,m);
                for(int x=0 ; x<n ; x++){
                    for(int y=0 ; y<m ; y++){
                        if(!v[x][y]){
                            fail = true;
                            break main;
                        }
                    }
                }
            }
        }

        out.println(fail ? "NO" : "YES");

        out.flush();
    }

    public static boolean inside(int x,int y,int n,int m){
        return x >= 0 && x < n && y >= 0 && y < m ;
    }

    public static void dfs(int x, int y, boolean[][] v,String[][] grid,int n,int m) {
        v[x][y] = true;
        for(int i=0 ; i<D.length() ; i++){
            if(grid[x][y].contains(String.valueOf(D.charAt(i)))){
                int nx = x + dx[i];
                int ny = y + dy[i];
                if(inside(nx,ny,n,m) && !v[nx][ny]){
                    dfs(nx,ny,v,grid,n,m);
                }
            }
        }
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }

        int[] nextInts(int n) {
            int[] ints = new int[n];
            for (int i = 0; i < n; i++) {
                ints[i] = in.nextInt();
            }
            return ints;
        }

        long[] nextLongs(int n) {
            long[] longs = new long[n];
            for (int i = 0; i < n; i++) {
                longs[i] = in.nextLong();
            }
            return longs;
        }
    }
}
