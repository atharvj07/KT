#include <bits/stdc++.h>
using namespace std;
const int inf = 987654321;

FILE *fin, *fout;
vector<int> P, chk;
void get_P() {
  chk = vector<int>(102, 0);
  for (int i = 2; i < 102; i++) {
    if (chk[i]) continue;
    P.push_back(i);
    int mul = 2;
    while (mul * i < 102) {
      chk[mul * i] = 1;
      mul++;
    }
  }
}
int main() {
  get_P();
  string ans;
  int p = 0;
  cout << 2 << endl;
  cin >> ans;

  cout << 5 << endl;

  if (ans == "yes") p = 5;

  cin >> ans;
  if (ans == "yes") p = 7;
  if (!p) {
    cout << "prime";
    return 0;
  }
  cout << p * p << endl;
  cin >> ans;
  if (ans == "yes") {
    cout << "composite";
    return 0;
  }
  for (int i = 0; P[i] * p <= 100; i++) {
    if (P[i] == p) continue;
    cout << P[i] << endl;
    cin >> ans;
    if (ans == "yes") {
      cout << "composite";
      return 0;
    }
  }
  cout << "prime";
}
