#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int x[3];
  int i, n, j;
  cin >> n >> x[0] >> x[1] >> x[2];
  long long int arr[n];
  long long int mtx[4][n + 1];
  for (i = 0; i < n; i++) cin >> arr[i];
  for (i = 0; i <= n; i++) mtx[0][i] = 0;
  for (i = 1; i < 4; i++) mtx[i][0] = LLONG_MIN;
  for (i = 1; i <= 3; i++) {
    for (j = 1; j <= n; j++)
      mtx[i][j] = max(mtx[i][j - 1], mtx[i - 1][j] + x[i - 1] * arr[j - 1]);
  }
  cout << mtx[3][n];
  return 0;
}
