def main():
	S = input()
	N = len(S)
	T = input()
	M = len(T)
	for i in range(M):
		for j in range(i):
			if i * 2 - j >= M:
				continue
			if T[j] != T[i * 2 - j]:
				break
		else:
			s = T[:i * 2:-1] + T[:i + 1]
			for j in range(N - len(s) + 1):
				if S[j:j + len(s)] == s:
					break
			else:
				continue
			return print('YES')
	print('NO')
Q = int(input())
for _ in range(Q):
	main()
