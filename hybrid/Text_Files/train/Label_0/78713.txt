for i in range(int(input())):
	a = int(input())
	li = []
	if a % 2 == 0:
		for j in range(1, a + 1):
			if j % 2 != 0:
				li.append(j + 1)
			else:
				li.append(j - 1)
	else:
		for j in range(1, a - 2):
			if j % 2 != 0:
				li.append(j + 1)
			else:
				li.append(j - 1)
		li.append(a)
		li.append(a - 2)
		li.append(a - 1)
	for k in range(a):
		print(li[k], end=' ')
	print()
