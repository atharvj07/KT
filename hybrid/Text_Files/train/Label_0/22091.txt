#include <bits/stdc++.h>
using namespace std;
void fail() {
  cout << "NO\n";
  exit(0);
}
vector<vector<int> > v;
vector<int> f;
void dfs(int vr, int col, pair<vector<int>, vector<int> > &vs) {
  if (col == 0)
    vs.first.push_back(vr);
  else
    vs.second.push_back(vr);
  f[vr] = col;
  for (int i : v[vr]) {
    if (f[i] == -1)
      dfs(i, !col, vs);
    else if (f[i] == col)
      fail();
  }
}
vector<int> operator+(const vector<int> &v1, const vector<int> &v2) {
  vector<int> vv;
  for (const int &i : v1) vv.push_back(i);
  for (const int &i : v2) vv.push_back(i);
  return vv;
}
int main() {
  int n, m;
  cin >> n >> m;
  v.resize(n);
  f.resize(n, -1);
  int n1, n2, n3;
  cin >> n1 >> n2 >> n3;
  while (m--) {
    int a, b;
    cin >> a >> b;
    a--;
    b--;
    v[a].push_back(b);
    v[b].push_back(a);
  }
  vector<pair<vector<int>, vector<int> > > d;
  for (int i = 0; i < n; i++)
    if (f[i] == -1) {
      d.resize(d.size() + 1);
      dfs(i, 0, d.back());
    }
  vector<vector<int> > dp(d.size() + 1, vector<int>(n2 + 1, -1));
  dp[0][0] = 2;
  for (int i = 1; i < d.size() + 1; i++) {
    for (int j = d[i - 1].first.size(); j <= n2; j++) {
      if (dp[i - 1][j - d[i - 1].first.size()] > -1) {
        dp[i][j] = 0;
      }
    }
    for (int j = d[i - 1].second.size(); j <= n2; j++) {
      if (dp[i - 1][j - d[i - 1].second.size()] > -1) {
        dp[i][j] = 1;
      }
    }
  }
  if (dp.back()[n2] == -1) fail();
  string s(n, '1');
  for (int i = dp.size() - 1; i > 0; i--) {
    if (dp[i][n2] == 0) {
      for (int j : d[i - 1].first) s[j] = '2';
      n2 -= d[i - 1].first.size();
    } else if (dp[i][n2] == 1) {
      for (int j : d[i - 1].second) s[j] = '2';
      n2 -= d[i - 1].second.size();
    }
  }
  if (n2 > 0) fail();
  for (int i = 0; i < n; i++) {
    if (s[i] == '2') continue;
    if (n3 == 0) break;
    n3--;
    s[i] = '3';
  }
  cout << "YES\n";
  cout << s << "\n";
  return 0;
}
