n = int(input())
P = []

def h(p1, p2, m):
	return abs(((p2[0] - p1[0]) * (m[1] - p1[1]) - (p2[1] - p1[1]) * (m[0] - p1[0])) / ((p2[0] - p1[0]) ** 2 + (p2[1] - p1[1]) ** 2) ** 0.5)
for i in range(n):
	P.append(list(map(int, input().split())))
answer = float('inf')
for i in range(n):
	cw = P[(i + 1) % n]
	ucw = P[i - 1]
	answer = min(answer, h(cw, ucw, P[i]) / 2)
print(answer)
