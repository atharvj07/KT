(n, m) = map(int, input().split())
vert = [0 for i in range(n)]
for i in range(n):
	vert[i] = int(input())
vert.sort()
horiz = []
addres = 0
for i in range(m):
	(x1, x2, y) = map(int, input().split())
	if x1 > 1:
		continue
	if x2 == 10 ** 9:
		addres += 1
		continue
	horiz += [x2]
horiz.sort()
res = min(n, len(horiz))
(i, j) = (0, 0)
while j < n:
	while i < len(horiz) and horiz[i] < vert[j]:
		i += 1
	res = min(res, len(horiz) - i + j)
	j += 1
print(res + addres)
