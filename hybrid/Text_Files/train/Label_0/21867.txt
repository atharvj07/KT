#include <bits/stdc++.h>
using namespace std;
long long mul(long long a, long long b, long long p = 1000000007) {
  return ((a % p) * (b % p)) % p;
}
long long add(long long a, long long b, long long p = 1000000007) {
  return (a % p + b % p) % p;
}
void input(long long a[], long long sz) {
  for (long long i = 0; i < sz; i++) cin >> a[i];
}
void print(long long a[], long long sz) {
  for (long long i = 0; i < sz; i++) {
    if (i == sz - 1)
      cout << a[i] << "\n";
    else
      cout << a[i] << " ";
  }
}
long long maxr(long long a[], long long sz) {
  long long ma;
  for (long long i = 0; i < sz; i++) {
    if (i == 0)
      ma = a[i];
    else if (a[i] > ma)
      ma = a[i];
  }
  return ma;
}
long long minr(long long a[], long long sz) {
  long long mi;
  for (long long i = 0; i < sz; i++) {
    if (i == 0)
      mi = a[i];
    else if (a[i] < mi)
      mi = a[i];
  }
  return mi;
}
long long isprm(long long n) {
  if (n <= 1) return 0;
  if (n <= 3) return 1;
  if (n % 2 == 0 || n % 3 == 0) return 0;
  for (int i = 5; i * i <= n; i = i + 6)
    if (n % i == 0 || n % (i + 2) == 0) return 0;
  return 1;
}
long long power(long long x, long long y, long long p = 1000000007) {
  long long res = 1;
  x = x % p;
  while (y > 0) {
    if (y & 1) res = (res * x) % p;
    y = y >> 1;
    x = (x * x) % p;
  }
  return res;
}
long long modInverse(long long n, long long p = 1000000007) {
  return power(n, p - 2, p);
}
long long ncrMod(long long n, long long r, long long p = 1000000007) {
  if (r == 0) return 1;
  long long fac[n + 1];
  fac[0] = 1;
  for (long long i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % p;
  return (fac[n] * modInverse(fac[r], p) % p * modInverse(fac[n - r], p) % p) %
         p;
}
struct comp {
  bool operator()(const pair<long long, pair<long long, long long>> &a,
                  const pair<long long, pair<long long, long long>> &b) const {
    return (a.first >= b.first);
  }
};
long long n, sl, u[100005], v[100005];
map<pair<long long, long long>, long long> w, nw, c;
vector<long long> a[100005];
long long dfs(long long u, long long pr) {
  long long k = 0;
  for (long long i = 0; i < a[u].size(); i++) {
    if (a[u][i] != pr) {
      long long sz = nw[make_pair(a[u][i], u)];
      k += ((w[make_pair(a[u][i], u)] * sz) + dfs(a[u][i], u));
    }
  }
  return k;
}
long long dfs2(long long u, long long pr) {
  long long k = 0;
  for (long long i = 0; i < a[u].size(); i++) {
    if (a[u][i] != pr) {
      k += dfs2(a[u][i], u);
    }
  }
  if (a[u].size() == 1) k = 1;
  if (u != 1) {
    nw[make_pair(u, pr)] = k;
    nw[make_pair(pr, u)] = k;
  }
  return k;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  ;
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> sl;
    w.clear();
    nw.clear();
    c.clear();
    for (long long i = 0; i < n + 5; i++) {
      a[i].clear();
      u[i] = 0;
      v[i] = 0;
    }
    for (long long i = 1; i < n; i++) {
      long long wt, cs;
      cin >> u[i] >> v[i] >> wt >> cs;
      a[u[i]].push_back(v[i]);
      a[v[i]].push_back(u[i]);
      c[make_pair(u[i], v[i])] = cs;
      c[make_pair(v[i], u[i])] = cs;
      w[make_pair(u[i], v[i])] = wt;
      w[make_pair(v[i], u[i])] = wt;
    }
    dfs2(1, 0);
    long long tw = dfs(1, 0);
    long long ans = 0;
    multiset<pair<long long, pair<long long, long long>>, comp> st[3];
    for (long long i = 1; i < n; i++) {
      long long k = w[make_pair(u[i], v[i])];
      long long h = nw[make_pair(u[i], v[i])];
      long long cs = c[make_pair(u[i], v[i])];
      long long d = ((k * h) - ((k / 2) * h));
      st[cs].insert(make_pair(d, make_pair(k, h)));
    }
    vector<long long> v1, v2;
    long long n1 = 0, n2 = 0;
    while (!st[1].empty()) {
      long long d = (*st[1].begin()).first;
      if (d == 0) break;
      if (n1 == 0) {
        v1.push_back(d);
        n1++;
      } else if (n1 > 0) {
        v1.push_back(v1[n1 - 1] + d);
        n1++;
      }
      long long k = (*st[1].begin()).second.first;
      long long h = (*st[1].begin()).second.second;
      k /= 2;
      d = (k * h) - ((k / 2) * h);
      st[1].erase(st[1].begin());
      st[1].insert(make_pair(d, make_pair(k, h)));
      if ((tw - v1[n1 - 1]) <= 0) break;
    }
    while (!st[2].empty()) {
      long long d = (*st[2].begin()).first;
      if (d == 0) break;
      if (n2 == 0) {
        v2.push_back(d);
        n2++;
      } else if (n2 > 0) {
        v2.push_back(v2[n2 - 1] + d);
        n2++;
      }
      long long k = (*st[2].begin()).second.first;
      long long h = (*st[2].begin()).second.second;
      k /= 2;
      d = (k * h) - ((k / 2) * h);
      st[2].erase(st[2].begin());
      st[2].insert(make_pair(d, make_pair(k, h)));
      if ((tw - v2[n2 - 1]) <= 0) break;
    }
    long long d = tw - sl;
    if (d <= 0) {
      cout << "0\n";
      continue;
    }
    ans = 1e18 + 5;
    auto it = lower_bound(v2.begin(), v2.end(), d);
    if (it != v2.end()) {
      ans = it - v2.begin();
      ans++;
      ans *= 2;
    }
    for (long long i = 0; i < n1; i++) {
      long long k = d - v1[i];
      if (k <= 0) {
        ans = min(ans, (i + 1));
        continue;
      }
      long long h;
      auto it = lower_bound(v2.begin(), v2.end(), k);
      if (it != v2.end()) {
        h = it - v2.begin();
        h++;
        h *= 2;
      } else
        continue;
      h += (i + 1);
      ans = min(ans, h);
    }
    cout << ans << "\n";
  }
  return 0;
}
