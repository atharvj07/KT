import java.io.*;
import java.util.*;

public class Main {
   private static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
   private static BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));

   public static void main(String[] args) throws IOException {

      int n = Integer.parseInt(br.readLine());

      int[] res = new int[n];

      solve(1, n, new HashMap<>(), res);

      bw.write('!');
      for (int i: res) {
         bw.write(' ');
         bw.write(Integer.toString(i));
      }
      bw.newLine();

      br.close();
      bw.close();
   }

   private static int solve(int left, int right, Map<Integer, Integer> lengths, int[] res) throws IOException {
      if (left > right) {
         return left - 1;
      }
      bw.write("? " + left + " " + right + "\n");
      bw.flush();
      String[] line = br.readLine().split(" ");
      int x = Integer.parseInt(line[0]), f = Integer.parseInt(line[1]);
      if (lengths.containsKey(x)) {
         right -= f;
         int length = lengths.get(x);
         for (int i = 0; i < length; ++i) {
            res[right + i] = x;
         }
         lengths.remove(x);
         solve(left, right, lengths, res);
         return right + length;
      } else {
         lengths.put(x, f);
         int l = left;
         while (lengths.containsKey(x)) {
            l = solve(l, l + f - 1, lengths, res) + 1;
         }
         return solve(l, right, lengths, res);
      }
   }
}
