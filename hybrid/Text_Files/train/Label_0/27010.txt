#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i = 0; i < (int)n; i++)
using ll = long long;

int main(){
  int n, k, r, s, p;
  cin >> n >> k >> r >> s >> p;
  string t;
  cin >> t;
  int ans = 0;
  vector<bool> f(k,false);
  rep(i,n) {
    if(i-k>=0 && t[i] == t[i-k] && f[i%k]) {
      f[i%k] = false;
      continue;
    }
    if(t[i] == 'r') ans += p;
    else if(t[i] == 's') ans += r;
    else ans += s;
    f[i%k] = true;
  }
  cout << ans << endl;
  return 0;
}
