#include <bits/stdc++.h>
using namespace std;

int N;
vector<string> S;
int cur;

int dfs(int depth){
    char c = S[cur][depth];
    if(isdigit(c)){
        cur++;
        return c - '0';
    }
    if(c == '+'){
        int s = 0;
        cur++;
        while(cur < N && depth < S[cur].size() && S[cur][depth] == '.'){
            s += dfs(depth+1);
        }
        return s;
    }
    if(c == '*'){
        int p = 1;
        cur++;
        while(cur < N && depth < S[cur].size() && S[cur][depth] == '.'){
            p *= dfs(depth+1);
        }
        return p;
    }
    
    return -1;
}

int main(){
    while(cin >> N, N){
        S.resize(N);
        for(auto &s: S){
            cin >> s;
        }
        
        cur = 0;
        cout << dfs(0) << endl;
    }
}
