import java.io.*;
import java.lang.Math;
import java.util.*;

public class Main
{	
	public BufferedReader in;
	public PrintStream out;
	
	double[] z;
	
	private class GetResult
	{
		double full, left, right, max;
		public GetResult(double x)
		{
			full = x;
			left = right = max = (x>0) ? x : 0;
		}
		
		public GetResult()
		{
		}
	}
	
	private class Treee
	{
		Treee left, right;
		int from, to, d;
		GetResult result;
		
		public Treee(int f, int t)
		{
			from = f;
			to = t;
			
			if (f+1==t)
			{
				d = 0;
				left = null;
				right = null;
				
				result = new GetResult(z[f]);
			}
			else
			{
				d = (f+t)/2;
				left = new Treee(f,d);
				right = new Treee(d,t);
				
				GetResult l = left.result;
				GetResult r = right.result;
				
				result = new GetResult();
				result.full = l.full + r.full;
				result.left = Math.max( l.left, l.full + r.left );
				result.right = Math.max( l.right + r.full, r.right);
				result.max = max3(
					l.max, r.max, l.right+r.left
				);
			}
		}
		
		GetResult get(int f, int t)
		{
			if (from+1==to)
			{
				return result;
			}
			
			if (t<=d)
			{
				return left.get(f,t);
			}
			if (f>=d)
			{
				return right.get(f,t);
			}
			
			if ((from>=f)&&(t>=to))
			{
				return result;
			}
			
			GetResult l = left.get(f,t);
			GetResult r = right.get(f,t);
			
			GetResult res = new GetResult();
			res.full = l.full + r.full;
			res.left = Math.max( l.left, l.full + r.left );
			res.right = Math.max( l.right + r.full, r.right);
			res.max = max3(
				l.max, r.max, l.right+r.left
			);
			
			return res;
		}
		
		double max3(double a, double b, double c)
		{
			return Math.max(
				a, Math.max(b,c)
			);
		}
	}
	
	public void test()
	{
		int n = readInt();
		int m = readInt();
		int c = readInt();
		
		int[] X = readIntArray(n);
		int[] p = readIntArray(n-1);
		
		z = new double[n];
		
		int i;
		for (i=1; i<n; i++)		
		{
			z[i] = 0.5*(double)(X[i]-X[i-1]) - 0.01*(double)(p[i-1]*c);
		}
		
		Treee t = new Treee(1,n);
		
		double res = 0;
		for (i=0; i<m; i++)
		{
			res += t.get(readInt(), readInt()).max;
		}
		
		out.println(res);
	}
    
	public void run()
	{
		try
		{
			in = new BufferedReader(new InputStreamReader(System.in));
			out = System.out;
			
			//in = new BufferedReader(new FileReader("in.txt"));

		{
			return;
		}
		
		//while (true)
		{
			//int t = readInt(); for (int i=0; i<t; i++)					
			{
				test();
			}
		}
	}
	
	public static void main(String args[])
	{
		new Main().run();
	}
	
	private StringTokenizer tokenizer = null;
	
	public int readInt() 
	{
        return Integer.parseInt(readToken());
    }
   
    public long readLong() 
	{
        return Long.parseLong(readToken());
    }
   
    public double readDouble() 
	{
        return Double.parseDouble(readToken());
    }
   
	public String readLn()
	{	
		try
		{
			String s;
			while ((s = in.readLine()).length()==0);
			return s;
		}
		catch (Exception e)
		{
			return "";
		}
	}
	
    public String readToken() 
	{
		try
		{
			while (tokenizer == null || !tokenizer.hasMoreTokens()) 
			{
				tokenizer = new StringTokenizer(in.readLine());
			}
			return tokenizer.nextToken();
		}
		catch (Exception e)
		{
			return "";
		}
    }
	
	public int[] readIntArray(int n)
	{
		int[] x = new int[n];
		readIntArray(x, n);
		return x;
	}
	
	public void readIntArray(int[] x, int n)
	{
		for (int i=0; i<n; i++)
		{
			x[i] = readInt();
		}
	}
}
