struct C {
    ll x, y;
    C() {}
    C(ll x, ll y) : x(x), y(y) {}
    C operator-(C p) const {return C(x-p.x, y-p.y);}
    ll det(C p) const {return x*p.y-y*p.x;} // ??????
    bool operator<(const C& another) const {
        if (x != another.x) return x < another.x;
        return y < another.y;
    }
};

ll area(const vector<C>& ps)
{
    ll A = 0;
    rep(i,ps.size()){
        A += ps[i].det(ps[(i+1) % ps.size()]);
    }
    return A;
}

vector<ll> convex_hull(vector<C> ps)
{
    int n = ps.size(), k = 0;
    vector<C> ch(n);
    vector<ll> s(n+1);
    //?????´??????
    for (int i = 0; i < n; ch[k++] = ps[i++]){
        s[i+1] = s[i];
        while (k >= 2 && (ch[k-1]-ch[k-2]).det(ps[i]-ch[k-1]) < 0){
            if(k >= 3){
                vector<C> tri = {ch[0],ch[k-2],ch[k-1]};
                s[i+1] -= area(tri);
            }
            k--;
        }
        if(k >= 2){
            vector<C> tri = {ch[0],ch[k-1],ps[i]};
            s[i+1] += area(tri);
        }
    }
    return s;
}
