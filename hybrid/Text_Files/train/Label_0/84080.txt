from math import ceil, floor
n = int(input())
ori = []
fin = []
s = 0
lower = False
for i in range(n):
	e = int(input())
	ori.append(e)
	if e >= 0:
		if e % 2 == 0:
			to_add = e // 2
			s += to_add
			fin.append(to_add)
		else:
			if lower:
				to_add = floor(e / 2)
				fin.append(to_add)
				s += to_add
			else:
				to_add = ceil(e / 2)
				fin.append(to_add)
				s += to_add
			lower = not lower
	else:
		fin.append(e)
index = 0
lower = False
for i in range(n):
	e = fin[i]
	if e < 0:
		if e % 2 == 0:
			to_add = e // 2
			fin[i] = to_add
			s += to_add
		else:
			index = i
			if lower:
				to_add = floor(e / 2)
				fin[i] = to_add
				s += to_add
			else:
				to_add = ceil(e / 2)
				fin[i] = to_add
				s += to_add
			lower = not lower
fin[index] -= s
for i in fin:
	print(i)
