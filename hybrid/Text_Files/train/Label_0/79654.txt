(a, b, c) = map(int, input().split())
(x, y) = ('0', '1')
if a > b:
	(x, y) = (y, x)
	(a, b) = (b, a)
res = y
b -= 1
for i in range(c - 1):
	if i % 2 == 0:
		a -= 1
		res += x
	else:
		b -= 1
		res += y
if res[-1] == x:
	(x, y) = (y, x)
	(a, b) = (b, a)
for i in range(b):
	res += y
for i in range(a):
	res += x
print(res)
