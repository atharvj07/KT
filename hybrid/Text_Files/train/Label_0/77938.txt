for t in range(int(input())):
	(a, b, c, d) = map(int, input().split())
	(x, y, x1, y1, x2, y2) = map(int, input().split())
	if ((a == 0 or (a > 0 and x - x1 > 0)) or (b == 0 or (b > 0 and x2 - x > 0))) and ((c == 0 or (c > 0 and y - y1 > 0)) or (d == 0 or (d > 0 and y2 - y > 0))) and (x1 <= x - (a - b) <= x2) and (y1 <= y + (d - c) <= y2):
		print('YES')
	else:
		print('NO')
