(m, n) = map(int, input().split())
a = []
for i in range(n):
	(x, y) = map(int, input().split())
	a.append([x, y])
isk = (a[0][0], a[0][1])
for i in range(2):
	d = isk[i]
	ans = 0
	dd = [0] * (m + 1)
	for j in a:
		if j[0] != d and j[1] != d:
			dd[j[0]] += 1
			dd[j[1]] += 1
			ans += 1
	if max(dd) >= ans:
		print('YES')
		exit()
print('NO')
