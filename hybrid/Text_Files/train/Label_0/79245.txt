s = input()
i = len(s) - 1
while i >= 0:
	r = int(s[i])
	if r == 0:
		print('O-|-OOOO')
	elif r == 1:
		print('O-|O-OOO')
	elif r == 2:
		print('O-|OO-OO')
	elif r == 3:
		print('O-|OOO-O')
	elif r == 4:
		print('O-|OOOO-')
	elif r == 5:
		print('-O|-OOOO')
	elif r == 6:
		print('-O|O-OOO')
	elif r == 7:
		print('-O|OO-OO')
	elif r == 8:
		print('-O|OOO-O')
	elif r == 9:
		print('-O|OOOO-')
	i = i - 1
