const int inf = 2e9;
char get_char(int x) {
  if (x == 0) return 'R';
  if (x == 1) return 'G';
  if (x == 2) return 'B';
  if (x == 3) return 'Y';
  if (x == 4) return 'W';
  assert(false);
}
char get_num(int x) { return '1' + x; }
int simul(vector<string>& a, int n, int mask1, int mask2) {
  set<char> hints_char, hints_num;
  for (int i = 0; i < 5; i++) {
    if ((mask1 >> i) & 1) {
      hints_char.insert(get_char(i));
    }
  }
  for (int i = 0; i < 5; i++) {
    if ((mask2 >> i) & 1) {
      hints_num.insert(get_num(i));
    }
  }
  bool vis[n][2];
  memset(vis, false, sizeof vis);
  int ok = 0;
  for (int i = 0; i < n; i++) {
    vis[i][0] = (hints_char.find(a[i][0]) != hints_char.end());
    vis[i][1] = (hints_num.find(a[i][1]) != hints_num.end());
  }
  map<int, set<int>> cards_with_prefix, cards_with_suffix;
  set<string> now;
  for (int i = 0; i < n; i++) {
    now.insert(a[i]);
  }
  for (int i = 0; i < n; i++) {
    if (vis[i][0] && vis[i][1]) {
      now.erase(a[i]);
    }
    if (vis[i][0] ^ vis[i][1]) {
      if (vis[i][0]) {
        cards_with_prefix[a[i][0]].insert(a[i][1]);
      }
      if (vis[i][1]) {
        cards_with_suffix[a[i][1]].insert(a[i][0]);
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (vis[i][0] ^ vis[i][1]) {
      if (vis[i][0]) {
        if (((int)cards_with_prefix[a[i][0]].size()) == 1) {
          now.erase(a[i]);
        }
      }
      if (vis[i][1]) {
        if (((int)cards_with_suffix[a[i][1]].size()) == 1) {
          now.erase(a[i]);
        }
      }
    }
  }
  return (((int)now.size()) <= 1
              ? ((int)hints_char.size()) + ((int)hints_num.size())
              : inf);
}
void solve() {
  int n;
  sc(n);
  vector<string> a(n);
  for (int __ii = 0; __ii < n; __ii++) cin >> a[__ii];
  int ans = inf;
  for (int i = 0; i < (1 << 5); i++) {
    for (int j = 0; j < (1 << 5); j++) {
      ans = min(ans, simul(a, n, i, j));
    }
  }
  pr(ans);
}
