#include <bits/stdc++.h>
using namespace std;
map<char, int> M;
int n, m, x, p;
int yous;
char a[50][50];
double dis(int x, int y, int xx, int yy) {
  int dx = x - xx;
  int dy = y - yy;
  return sqrt(dx * dx + dy * dy);
}
int ok(string s) {
  int i, countt = 0;
  for (i = 0; i < s.length(); i++) {
    if (isupper(s[i])) {
      if (!yous) return -1;
      if (M[s[i]]) continue;
      if (M[tolower(s[i])])
        countt++;
      else
        return -1;
    } else {
      if (!M[s[i]]) return -1;
    }
  }
  return countt;
}
int main(void) {
  int i, j, k, l;
  string s;
  while (cin >> n >> m >> x) {
    M.clear();
    getchar();
    yous = 0;
    for (i = 0; i < n; i++) {
      for (j = 0; j < m; j++) {
        a[i][j] = getchar();
        if (a[i][j] == 'S')
          yous = 1;
        else
          M[a[i][j]] = 1;
      }
      getchar();
    }
    for (i = 0; i < n; i++) {
      for (j = 0; j < m; j++) {
        if (a[i][j] != 'S') continue;
        int k, l;
        for (k = 0; k < n; k++)
          for (l = 0; l < m; l++) {
            if (a[k][l] == 'S') continue;
            if (dis(i, j, k, l) > x) continue;
            M[toupper(a[k][l])] = 1;
          }
      }
    }
    cin >> p;
    getchar();
    getline(cin, s);
    cout << ok(s) << endl;
  }
  return 0;
}
