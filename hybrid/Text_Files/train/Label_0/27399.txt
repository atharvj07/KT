#include <bits/stdc++.h>
using namespace std;
bool comp(string x, string y) {
  if (x.size() != y.size())
    return x.size() > y.size();
  else
    return x > y;
}
bool find_back(string s, string vs) {
  for (int i = 0; i < vs.size(); i++) {
    if (s[s.size() - i - 1] != vs[vs.size() - i - 1]) return false;
  }
  return true;
}
int main() {
  ios_base::sync_with_stdio(false);
  int n;
  map<string, vector<string> > book;
  map<string, vector<string> >::iterator it;
  vector<string> all;
  cin >> n;
  for (int i = 0; i < n; i++) {
    vector<string> vs;
    string s, vs_s;
    int kol;
    cin >> s >> kol;
    for (int j = 0; j < kol; j++) {
      cin >> vs_s;
      it = book.find(s);
      if (it == book.end()) {
        book.insert(pair<string, vector<string> >(s, vs));
        all.push_back(s);
      }
      book[s].push_back(vs_s);
    }
  }
  for (int i = 0; i < book.size(); i++)
    sort(book[all[i]].begin(), book[all[i]].end(), comp);
  for (int i = 0; i < book.size(); i++) {
    for (int j = 0; j < book[all[i]].size(); j++)
      for (int z = j + 1; z < book[all[i]].size(); z++) {
        if (find_back(book[all[i]][j], book[all[i]][z]) == true) {
          book[all[i]].erase(book[all[i]].begin() + z);
          z--;
        }
      }
  }
  cout << all.size() << '\n';
  for (int i = 0; i < all.size(); i++) {
    cout << all[i] << ' ' << book[all[i]].size() << ' ';
    for (int j = 0; j < book[all[i]].size(); j++)
      cout << book[all[i]][j] << ' ';
    cout << '\n';
  }
  return 0;
}
