n = int(input())
s = [int(i) for i in input()]
t = s
a = []
b = []
K = s.count(1)
for i in range(n):
	(A, B) = map(int, input().split())
	a.append(A)
	b.append(B)
for i in range(1000):
	for j in range(n):
		if i >= b[j] and (i - b[j]) % a[j] == 0:
			s[j] = (s[j] + 1) % 2
	if s.count(1) > K:
		K = s.count(1)
print(K)
