#include<bits/stdc++.h>
#define rep(i,x,y) for (int i=(x);i<=(y);i++)
#define ll long long
#define lc (now<<1)
#define rc (now<<1|1)
using namespace std;
const int N=1e5+100;
int n,Q,r; 
ll tag[N<<2],mx[N<<2];
void add(int now,int l,int r,int x,int y,int v)
{
	if (x<=l&&r<=y)
	{
		tag[now]+=v; 
		mx[now]+=v; 
		return;
	}
	int mid=l+r>>1;
	if (x<=mid) add(lc,l,mid,x,y,v);
	if (y>mid) add(rc,mid+1,r,x,y,v);
	mx[now]=max(mx[lc],mx[rc])+tag[now];
}
int main(){
	scanf("%d%d",&n,&Q);
	r=1;
	while (Q--)
	{
		int x,y,w; 
		scanf("%d%d%d",&x,&y,&w);
		while (r<y) 
			add(1,1,n,++r,n,1);
		add(1,1,n,x,y,w);
		printf("%lld\n",mx[1]);
	}
	return 0;
}
