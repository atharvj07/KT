#include <bits/stdc++.h>
using namespace std;
int a[8], res;
int main() {
  for (int i = 1; i <= 6; i++) {
    scanf("%d", &a[i]);
  }
  int i1 = min(a[2], a[6]), i2 = max(a[2], a[6]);
  int tp = a[1] * 2 + 1, tm = a[1] + i1;
  res = tp * i1 + ((i1 - 1) * i1) + tm * 2 * (i2 - i1);
  int i3 = min(a[3], a[5]);
  tp = a[4] * 2 + 1;
  res = res + tp * i3 + ((i3 - 1) * i3);
  printf("%d", res);
  return 0;
}
