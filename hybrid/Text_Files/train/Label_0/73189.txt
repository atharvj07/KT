import sys
n = int(input())
(s, t) = (input(), input())
a = []
num = 0
for i in reversed(range(n)):
	if s[i] == t[i]:
		continue
	f = False
	for j in range(i, -1, -1):
		if s[j] == t[i]:
			for k in range(j, i):
				a.append(k + 1)
				s = s[:k] + s[k + 1] + s[k] + s[k + 2:]
				f = True
				num += 1
			break
	if f == 0 or num > 10000:
		print(-1)
		sys.exit()
print(len(a))
print(*a)
