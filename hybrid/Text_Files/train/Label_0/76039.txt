import sys
read = sys.stdin.buffer.read
readline = sys.stdin.buffer.readline
write = sys.stdout.write

def ii():
	return int(readline())

def mi():
	return map(int, readline().rstrip().split())

def li():
	return list(readline().rstrip())

def lmi():
	return list(map(int, readline().rstrip().split()))

def end(*arg):
	print(*arg)
	sys.exit()

def main():
	t = ii()
	for _ in range(t):
		(n, k1, k2) = mi()
		a = lmi()
		b = lmi()
		a.sort()
		b.sort()
		if a[-1] == n:
			print('YES')
		else:
			print('NO')
	return
main()
