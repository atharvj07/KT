(n, k) = [int(i) for i in input().split()]
a = [int(i) for i in input().split()]
color = [0] * n
arr = []
for i in range(len(a)):
	arr.append([a[i], i])
arr.sort()
curr = 1
for i in range(len(arr)):
	if curr == k:
		color[arr[i][1]] = k
		curr = 1
		continue
	else:
		color[arr[i][1]] = curr % k
	curr += 1
hsh = [0 for i in range(5001)]
flag = 0
for i in a:
	hsh[i] += 1
for i in hsh:
	if i > k:
		flag = 1
		break
if flag == 1 or k > n:
	print('NO')
else:
	print('YES')
	print(*color)
