n = int(input())
ar = [int(x) for x in input().split()]
mi = -1
mc = 10 ** 18
for i in range(len(ar)):
	if ar[i] <= mc:
		mc = ar[i]
		mi = i
digits = n // mc
if digits == 0:
	print(-1)
	exit(0)
n -= mc * digits
ar = ar[::-1]
ans = [str(mi + 1)] * digits
index = 0
while index < digits:
	for i in range(len(ar)):
		if 9 - i <= mi + 1:
			print(''.join(ans))
			exit(0)
		if n - ar[i] + mc >= 0:
			n = n - ar[i] + mc
			ans[index] = str(9 - i)
			break
	index += 1
print(''.join(ans))
