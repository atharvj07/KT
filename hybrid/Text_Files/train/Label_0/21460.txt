#include <bits/stdc++.h>
using namespace std;
vector<long long int> primes;
int main() {
  ios_base::sync_with_stdio(0);
  bool check[100000] = {};
  long long int n, i, j, k, l;
  primes.emplace_back(2);
  for (i = 3; i < 100000; i += 2) {
    if (check[i] == false) {
      primes.emplace_back(i);
      for (j = i * i; j <= 100000; j += i + i) {
        check[j] = true;
      }
    }
  }
  cin >> n;
  vector<long long int> a1, a2;
  map<long long int, pair<long long int, long long int>> mp;
  for (k = 0; k < n; k++) {
    long long int a;
    cin >> a;
    if (mp[a].first != 0) {
      a1.emplace_back(mp[a].first);
      a2.emplace_back(mp[a].second);
      continue;
    }
    bool flag = false;
    long long int keep = a, num = 1;
    long long int limit = sqrt(keep);
    for (i = 0; primes[i] <= limit; i++) {
      if (keep % primes[i] == 0) {
        while (keep % primes[i] == 0) keep /= primes[i], num *= primes[i];
        break;
      }
    }
    if (keep == 1 || num == 1) {
      a1.emplace_back(-1);
      a2.emplace_back(-1);
      mp[a] = {-1, -1};
      continue;
    }
    a1.emplace_back(num);
    a2.emplace_back(keep);
    mp[a] = {num, keep};
  }
  for (i = 0; i < n; i++) {
    cout << min(a1[i], a2[i]) << ' ';
  }
  cout << endl;
  for (i = 0; i < n; i++) cout << max(a1[i], a2[i]) << ' ';
  cout << endl;
  return 0;
}
