n = int(input())
strings = []
for k in range(0, n):
	strings.append(input())
for j in strings:
	flag = 0
	if len(j) != 1:
		teste = []
		[teste.append(x) for x in j]
		if len(teste) == len(set(teste)):
			ordem = [ord(y) for y in teste]
			ordem.sort()
			for g in range(1, len(ordem)):
				if ordem[g - 1] - ordem[g] != -1:
					flag = 2
					break
			if flag == 0:
				print('Yes')
			else:
				print('No')
		else:
			print('No')
	else:
		print('Yes')
