R1 = lambda : list(map(int, input().split()))
R2 = lambda : int(input())
R3 = lambda : list(input())
R4 = lambda : list(map(int, input()))
A = R3()
n = len(A)
c = int(A[n - 1])
m = -1
for i in range(n):
	e = int(A[i])
	if e % 2 == 0:
		if e <= c:
			(A[i], A[n - 1]) = (A[n - 1], A[i])
			print(''.join(A))
			break
		m = i
else:
	if m == -1:
		print(-1)
	else:
		(A[m], A[n - 1]) = (A[n - 1], A[m])
		print(''.join(A))
