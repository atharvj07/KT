n = int(input())
s = input()
b = [0] * n
r = [0] * n
g = [0] * n
if s[0] == 'R':
	b[0] = 1
	g[0] = 1
elif s[0] == 'G':
	b[0] = 1
	r[0] = 1
else:
	r[0] = 1
	g[0] = 1
for i in range(1, n):
	if s[i] == 'R':
		r[i] = min(0 + g[i - 1], 0 + b[i - 1])
		g[i] = min(1 + r[i - 1], 1 + b[i - 1])
		b[i] = min(1 + g[i - 1], 1 + r[i - 1])
	elif s[i] == 'G':
		g[i] = min(0 + r[i - 1], 0 + b[i - 1])
		r[i] = min(1 + g[i - 1], 1 + b[i - 1])
		b[i] = min(1 + g[i - 1], 1 + r[i - 1])
	else:
		b[i] = min(0 + g[i - 1], 0 + r[i - 1])
		r[i] = min(1 + g[i - 1], 1 + b[i - 1])
		g[i] = min(1 + r[i - 1], 1 + b[i - 1])
print(min(r[-1], g[-1], b[-1]))
l = min(r[-1], g[-1], b[-1])
w = [''] * n
if l == b[-1]:
	w[-1] = 'B'
elif l == g[-1]:
	w[-1] = 'G'
else:
	w[-1] = 'R'
for i in range(n - 2, -1, -1):
	if w[i + 1] == 'B':
		if r[i] <= g[i]:
			w[i] = 'R'
		else:
			w[i] = 'G'
	elif w[i + 1] == 'G':
		if r[i] <= b[i]:
			w[i] = 'R'
		else:
			w[i] = 'B'
	elif b[i] <= g[i]:
		w[i] = 'B'
	else:
		w[i] = 'G'
print(''.join(w))
