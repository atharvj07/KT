#include <bits/stdc++.h>
using namespace std;
const int MAXN = 50;
const int MOD = 1e9 + 7;
const int TWOINV = MOD + 1 >> 1;
int best[MAXN + 5][MAXN + 5][2];
int NCR[MAXN + 5][MAXN + 5];
int ncr(int n, int r) {
  if (n < r) return 0;
  if (r == 0) return 1;
  int& ret = NCR[n][r];
  if (~ret) return ret;
  return ret = (ncr(n - 1, r) + ncr(n - 1, r - 1)) % MOD;
}
int solve(int n, int used, int unused) {
  if (unused < used) return 0;
  if (unused > used + 1) return 0;
  if (n == 1) return !used && !unused;
  int& ret = best[n][used][unused - used];
  if (~ret) return ret;
  ret = solve(n - 1, unused - 1, used) * (n - 1LL) % MOD;
  for (int left = 1; left <= n - 1 - left; left++) {
    const int right = n - 1 - left;
    const int factor = left == right ? TWOINV : 1;
    for (int leftUnused = 0; leftUnused <= used; leftUnused++) {
      const int rightUnused = used - leftUnused;
      for (int leftUsed = max(0, leftUnused - 1); leftUsed <= leftUnused;
           leftUsed++) {
        for (int rightUsed = max(0, rightUnused - 1); rightUsed <= rightUnused;
             rightUsed++) {
          if (leftUsed + rightUnused >= unused) continue;
          if (rightUsed + leftUnused >= unused) continue;
          if (leftUsed + rightUnused + 1 != unused &&
              rightUsed + leftUnused + 1 != unused)
            continue;
          ret += 1LL * solve(left, leftUsed, leftUnused) *
                 solve(right, rightUsed, rightUnused) % MOD * ncr(n - 1, left) %
                 MOD * factor % MOD * left % MOD * right % MOD;
          ret %= MOD;
        }
      }
    }
  }
  return ret;
}
int main() {
  memset(best, -1, sizeof(best));
  memset(NCR, -1, sizeof(NCR));
  int c, c2;
  int n, k;
  scanf("%d%d", &n, &k);
  printf("%d\n", (solve(n, k, k) + solve(n, k - 1, k)) % MOD);
  return 0;
}
