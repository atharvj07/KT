from collections import defaultdict
n = int(input())
aa = list(map(int, input().split()))
bb = list(map(int, input().split()))
d = {}
dd = defaultdict(set)
for i in range(n):
	a = aa[i]
	dd[a].add(i)
children = defaultdict(set)
for a in dd:
	for b in dd:
		if a | b == a:
			children[a] |= dd[b]
res = set()
for a in dd:
	sm = 0
	if len(dd[a]) >= 2:
		res |= children[a]
sm = 0
for i in res:
	sm += bb[i]
print(sm)
