#include <bits/stdc++.h>
using namespace std;
int n, m, s, nScc, col[1000100], num[1000100], low[1000100];
long long f[1000100];
vector<pair<int, long long> > adj[1000100], dag[1000100];
stack<int> st;
void dfs(int u) {
  static int cnt = 0;
  int v;
  num[u] = low[u] = ++cnt;
  st.push(u);
  for (pair<int, long long> p : adj[u]) {
    v = p.first;
    if (num[v]) {
      low[u] = min(low[u], num[v]);
    } else {
      dfs(v);
      low[u] = min(low[u], low[v]);
    }
  }
  if (num[u] == low[u]) {
    int v;
    ++nScc;
    do {
      v = st.top();
      st.pop();
      num[v] = low[v] = 1000100;
      col[v] = nScc;
    } while (v != u);
  }
}
long long getval(long long w) {
  long long d, c, g, res;
  d = res = 0;
  c = 1e9;
  while (d <= c) {
    g = (d + c) / 2;
    if (g * (g + 1) / 2 <= w) {
      res = g;
      d = g + 1;
    } else
      c = g - 1;
  }
  return (res + 1) * w - res * (res + 1) * (res + 2) / 6;
}
void Calc(int u) {
  num[u] = 1;
  int v;
  long long w;
  for (pair<int, long long> p : adj[u]) {
    v = p.first;
    w = p.second;
    if (col[u] != col[v]) {
      dag[col[u]].push_back(pair<int, long long>(col[v], w));
    } else {
      f[col[u]] += getval(w);
      if (!num[v]) Calc(v);
    }
  }
}
long long ans(int u, int par) {
  if (num[u]) return f[u];
  int v;
  long long w, mx = 0;
  num[u] = 1;
  for (pair<int, long long> p : dag[u]) {
    v = p.first;
    w = p.second;
    if (v != par) {
      mx = max(mx, w + ans(v, u));
    }
  }
  return f[u] = f[u] + mx;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  nScc = 0;
  memset(f, 0, sizeof(f));
  memset(num, 0, sizeof(num));
  cin >> n >> m;
  for (int u, v; m--;) {
    long long w;
    cin >> u >> v >> w;
    adj[u].push_back(pair<int, long long>(v, w));
  }
  cin >> s;
  for (int i = 1; i <= n; ++i)
    if (!num[i]) dfs(i);
  memset(num, 0, sizeof(num));
  for (int i = 1; i <= n; ++i)
    if (!num[i]) Calc(i);
  memset(num, 0, sizeof(num));
  cout << ans(col[s], -1);
  return 0;
}
