#include <bits/stdc++.h>
using namespace std;
bool check(int a, int b, int c);
int main() {
  int i, j, a, b, c, minn;
  scanf("%d %d %d", &a, &b, &c);
  i = 0;
  while (!check(a, b, c)) {
    i++;
    minn = min(a, b);
    minn = min(minn, c);
    if (minn == a) {
      check(a++, b, c);
      continue;
    }
    if (minn == b) {
      check(a, b++, c);
      continue;
    }
    if (minn == c) {
      check(a, b, c++);
      continue;
    }
  }
  printf("%d\n", i);
  return 0;
}
bool check(int a, int b, int c) {
  if (a + b > c && b + c > a && c + a > b) {
    return true;
  }
  return false;
}
