(a, b) = map(int, input().split())
c = list(map(int, input().split()))
d = list(map(int, input().split()))
f = [0 for _ in range(a)]
g = [0 for _ in range(a)]
g[0] = b
for i in range(1, a):
	f[i] = min(f[i - 1], g[i - 1]) + c[i - 1]
	g[i] = min(f[i - 1] + b, g[i - 1]) + d[i - 1]
for i in range(a):
	print(min(f[i], g[i]), end=' ')
