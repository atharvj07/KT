import sys
dictionary = {}
value = int(input())
negatives = 0
positives = 0
for i in range(value):
	(x, y) = map(int, input().split())
	if x >= 0:
		positives += 1
	elif x < 0:
		negatives += 1
	dictionary[x] = y
keylist = dictionary.keys()
keylist = sorted(keylist)
if positives == negatives:
	value = sum(dictionary.values())
	print(value)
	sys.exit()
elif positives < negatives:
	counter = 0
	value = 0
	for i in keylist[::-1]:
		if i >= 0:
			value += dictionary[i]
		elif counter == positives + 1:
			print(value)
			sys.exit()
		else:
			value += dictionary[i]
			counter += 1
elif positives > negatives:
	counter = 0
	value = 0
	for i in keylist:
		if i < 0:
			value += dictionary[i]
		elif counter == negatives + 1:
			print(value)
			sys.exit()
		else:
			value += dictionary[i]
			counter += 1
print(value)
