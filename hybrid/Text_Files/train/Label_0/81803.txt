def aint():
	arr = input().split()
	for i in range(len(arr)):
		arr[i] = int(arr[i])
	return arr

def amax(a, b, n, x):
	if n == 0:
		print('YES')
		return
	t = [0, 0, 0, 0]
	for i in range(n):
		if a[i] > t[1]:
			t = [i, a[i], 0, 0]
	for i in range(n):
		if b[i] + t[1] <= x:
			if b[i] > t[3]:
				t = [t[0], t[1], i, b[i]]
	if t[3] == 0:
		print('NO')
		return
	a.pop(t[0])
	b.pop(t[2])
	amax(a, b, n - 1, x)
for j in range(int(input())):
	(n, x) = map(int, input().split())
	a = aint()
	b = aint()
	amax(a, b, n, x)
	try:
		input()
	except EOFError:
		pass
