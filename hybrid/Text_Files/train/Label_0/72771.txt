t = int(input())
while t > 0:
	size = int(input())
	inp = str(input())
	a = []
	b = []
	flag = 0
	for i in range(size):
		if flag == 0:
			if inp[i] == '2':
				a.append(1)
				b.append(1)
			elif inp[i] == '0':
				a.append(0)
				b.append(0)
			else:
				a.append(1)
				b.append(0)
				flag = 1
		elif inp[i] == '2':
			a.append(0)
			b.append(2)
		elif inp[i] == '0':
			a.append(0)
			b.append(0)
		else:
			a.append(0)
			b.append(1)
	for i in range(size):
		print(a[i], end='')
	print()
	for i in range(size):
		print(b[i], end='')
	print()
	t -= 1
