#include <bits/stdc++.h>
using namespace std;
const int ma = 1e5 + 10;
int n, m, num[ma], vis[ma];
int cnt2;
int main() {
  scanf("%d", &n);
  int cnt4 = 0, cnt8 = 0;
  for (int i = 0; i < n; i++) {
    scanf("%d", num + i);
    if (vis[num[i]] == 1 || vis[num[i]] == 5) cnt2++;
    if (vis[num[i]] == 3) cnt4++, cnt2--;
    if (vis[num[i]] == 7) cnt8++, cnt2--;
    vis[num[i]]++;
  }
  scanf("%d", &m);
  char c;
  int a;
  while (m--) {
    cin >> c;
    scanf("%d", &a);
    if (c == '+') {
      if (vis[a] == 1 || vis[a] == 5) cnt2++;
      if (vis[a] == 3) cnt4++, cnt2--;
      if (vis[a] == 7) cnt8++, cnt2--;
      vis[a]++;
      if (cnt8 || (cnt2 >= 2 && cnt4) || (cnt4 >= 2))
        puts("YES");
      else
        puts("NO");
    } else {
      if (vis[a] == 2 || vis[a] == 6) cnt2--;
      if (vis[a] == 4) cnt4--, cnt2++;
      if (vis[a] == 8) cnt8--, cnt2++;
      vis[a]--;
      if (cnt8 || (cnt2 >= 2 && cnt4) || (cnt4 >= 2))
        puts("YES");
      else
        puts("NO");
    }
  }
  return 0;
}
