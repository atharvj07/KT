
import java.util.*;

public class cf758a {

    public static void main(String[] args) {
        
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        
        int max = Integer.MIN_VALUE;
        
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            max = Math.max(a[i], max);
        }
        
        int cnt = 0;
        
        for (int i = 0; i < n; i++) {
            
            cnt += Math.abs(a[i] - max);
            
        }
        
        System.out.println(cnt);
        
    }
    
}
