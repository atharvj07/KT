n = int(input())
inp = input().split()
d = [(int(inp[i]), i) for i in range(n)]
d.sort()
mmm = d[-1][0] - d[0][0]
dif = [d[i + 1][0] - d[i][0] for i in range(n - 1)]
if n < 6:
	print(sum(dif), 1)
	print('1 ' * n)
	exit()
dp = [-1] * (n - 1)
dpp = [-1] * (n - 1)
ninf = -10000000000
dp[0] = ninf
dp[1] = ninf
dp[2] = dif[2]
dp[3] = dif[3]
dp[4] = dif[4]
dpp[0] = -1
dpp[1] = -1
dpp[2] = -1
dpp[3] = -1
dpp[4] = -1
for i in range(5, n - 1):
	ans = -1
	src = -1
	for j in range(3, 6):
		if i - j >= 0 and dp[i - j] + dif[i] > ans:
			ans = dp[i - j] + dif[i]
			src = i - j
	dp[i] = ans
	dpp[i] = src
ans = -1
src = -1
for j in range(n - 4, n - 7, -1):
	if dp[j] > ans:
		ans = dp[j]
		src = j
l = []
while src != -1:
	l.append(src)
	src = dpp[src]
l.reverse()
ll = [1]
lind = 0
ind = 0
team = 1
kk = len(l)
for ind in range(1, n):
	if lind < kk and l[lind] == ind - 1:
		team += 1
		lind += 1
	ll.append(team)
aaa = [0] * n
for i in range(n):
	dd = ll[i]
	aaa[d[i][1]] = dd
print(mmm - ans, team)
print(*aaa)
