from sys import stdin, stdout
t = int(stdin.readline().strip())
for _ in range(t):
	n = int(stdin.readline().strip())
	neck = list(map(int, stdin.readline().strip().split(' ')))
	brac = list(map(int, stdin.readline().strip().split(' ')))
	neck.sort()
	brac.sort()
	stdout.write(' '.join(list(map(str, neck))) + '\n')
	stdout.write(' '.join(list(map(str, brac))) + '\n')
