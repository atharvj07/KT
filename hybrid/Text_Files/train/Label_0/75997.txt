def minv(a, b=0, c=0):
	if a | b | c != 7:
		return mins
	s = opts[a] + opts[b] + opts[c]
	return min(mins, s)
opts = [1000000] * 8
opts[0] = 0
for _ in range(int(input())):
	(c, l) = input().split()
	(v, c) = (0, int(c))
	if 'A' in l:
		v += 1
	if 'B' in l:
		v += 2
	if 'C' in l:
		v += 4
	opts[v] = min(opts[v], c)
mins = 10000000
for a in range(1, 8):
	mins = minv(a)
	for b in range(a + 1, 8):
		mins = minv(a, b)
		for c in range(b + 1, 8):
			mins = minv(a, b, c)
print(-1 if mins >= 1000000 else mins)
