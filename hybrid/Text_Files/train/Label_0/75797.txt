t = int(input())
while t > 0:
	t -= 1
	n = int(input())
	lst = list(map(int, input().split()))
	ex = 0
	cur = 0
	okay = True
	for i in range(n):
		if lst[i] >= ex:
			cur += lst[i] - ex
		elif lst[i] <= ex:
			cur += lst[i]
			if cur >= ex:
				cur = cur - ex
			else:
				okay = False
				break
		ex += 1
	if okay == True:
		print('YES')
	else:
		print('NO')
