ll f(int h, int E, vector<vector<ll> >& memo) {
	if (!h || !E) return 1;
	if (memo[h][E] != -1) return memo[h][E];
	return memo[h][E] = f(h - 1, E - 1, memo) + f(h - 1, E, memo);
}

int main() {
	vector<vector<ll> > memo(100, vector<ll>(100, -1));
	int T; cin >> T;
	while (T--) {
		ll N; int E; cin >> N >> E;
		int h;
		for (h = 0; f(h, E, memo) < N + 1; h++);
		cout << h << endl;
	}
}
