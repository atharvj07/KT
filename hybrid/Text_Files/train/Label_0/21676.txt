#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
#define NO "Miserable Hokusai!"
int main(){
  int n;
  cin >> n;
  for(int a = 0; a < n; a++){
    int g[2];
    cin >> g[0] >> g[1];
    vector<vector<long long> > ans(g[0]+1, vector<long long>(g[1]+1, 0));
    int p;
    cin >> p;
    vector<vector<bool> > v(g[0]+1, vector<bool>(g[1]+1, true));
    vector<vector<bool> > h(g[0]+1, vector<bool>(g[1]+1, true));
    int x1,x2,y1,y2;
    for(int i = 0; i < p; i++){
      cin >> x1 >> y1 >> x2 >> y2;
      if(x1 == x2)
        v[x1][max(y1,y2)] = false;
      else if(y1 == y2)
        h[max(x1,x2)][y1] = false;
    }

    for(int i = 0; i <= g[1]; i++)
      h[0][i] = false;
    for(int i = 0; i <= g[0]; i++)
      v[i][0] = false;

    for(int i = 0; i <= g[0]; i++){
      for(int j = 0; j <= g[1]; j++){
        if(i == 0 && j == 0){
          ans[i][j] = 1;
        }else if(!h[i][j] && !v[i][j]){
          ans[i][j] = 0;
        }else if(!h[i][j] && v[i][j]){
          ans[i][j] = ans[i][j-1];
        }else if(h[i][j] && !v[i][j]){
          ans[i][j] = ans[i-1][j];
        }else if(h[i][j] && v[i][j]){
          ans[i][j] = ans[i-1][j] + ans[i][j-1];
        }
      }
    }
    if(ans[g[0]][g[1]] == 0)
      cout << NO << endl;
    else
      cout << ans[g[0]][g[1]] << endl;
  }
}
