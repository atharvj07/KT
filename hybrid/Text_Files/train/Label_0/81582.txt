for _ in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	a = sum(l)
	flag = 0
	flg = 0
	if a % 2 == 0:
		for i in l:
			if i % 2 == 0:
				flag = 1
			elif i % 2 != 0:
				flg = 1
	if flag == 1 and flg == 1:
		print('YES')
	elif a % 2 != 0:
		print('YES')
	else:
		print('NO')
