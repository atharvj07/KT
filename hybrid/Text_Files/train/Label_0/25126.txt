#include <bits/stdc++.h>
using namespace std;
ifstream in("input.txt");
ofstream out("output.txt");
const long long MAXL = 1e18;
const int N = 10005000;
const int MAXN = 1e9 + 7;
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  srand(time(NULL));
  unordered_map<long long, bool> used;
  long long a, b, w, x, c;
  cin >> a >> b >> w >> x >> c;
  vector<long long> cnt(w + 10);
  bool last = 0;
  if (c <= a) {
    cout << 0 << "\n";
    return 0;
  }
  long long cycl = 0, minus_a = 0;
  while (true) {
    if (used[b]) {
      break;
    }
    used[b] = true;
    cycl++;
    if (b >= x) {
      b = b - x;
      last = 0;
    } else {
      b = w - (x - b);
      minus_a++;
      cnt[cycl]++;
      last = 1;
    }
  }
  for (int i = 1; i < w + 10; ++i) {
    cnt[i] = cnt[i - 1] + cnt[i];
  }
  long long l = 0, r = MAXL, mid;
  while (l < r - 1) {
    mid = l + (r - l) / 2;
    long long temp_c = c - mid;
    long long temp_a = a - ((mid / cycl) * minus_a + cnt[mid % cycl]);
    if (temp_c <= temp_a) {
      r = mid;
    } else {
      l = mid;
    }
  }
  cout << r << "\n";
  return 0;
}
