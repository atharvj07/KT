#include <bits/stdc++.h>
using namespace std;
int main() {
  int h1, a1, c1, h2, a2;
  vector<int> v;
  cin >> h1 >> a1 >> c1 >> h2 >> a2;
  for (; h2 > 0;) {
    if (h2 <= a1) {
      h2 -= a1;
      v.push_back(0);
    } else if (h1 <= a2) {
      h1 += c1;
      v.push_back(1);
    } else {
      h2 -= a1;
      v.push_back(0);
    }
    h1 -= a2;
  }
  cout << v.size() << '\n';
  for (auto x : v) {
    if (x == 0) {
      cout << "STRIKE\n";
    } else {
      cout << "HEAL\n";
    }
  }
  return 0;
}
