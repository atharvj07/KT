import java.util.Scanner;

public class P246B {

  public static void main(String[] args) {
    Scanner inScanner = new Scanner(System.in);
    int n = inScanner.nextInt();
    int sum = 0;
    for (int j = 0; j < n; j++) {
      sum += inScanner.nextInt();
    }
    if (sum % n == 0) {
      System.out.println(n);
    } else {
      System.out.println(n - 1);
    }
  }
}
