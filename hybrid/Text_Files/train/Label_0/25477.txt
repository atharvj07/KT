#include <bits/stdc++.h>
using namespace std;
int n, m, r[1005], c[1005], cnt;
char s[1005][1005];
int main() {
  scanf("%d %d", &n, &m);
  for (int i = 0; i < n; i++) scanf("%s", s[i]);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (s[i][j] == '*') r[i]++, c[j]++, cnt++;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int t = r[i] + c[j];
      if (s[i][j] == '*') t--;
      if (t == cnt) {
        printf("YES\n%d %d\n", i + 1, j + 1);
        return 0;
      }
    }
  }
  printf("NO\n");
  return 0;
}
