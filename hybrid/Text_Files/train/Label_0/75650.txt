lof = []
mi = 1000
ma = 0
mf = 0

class Bio:

	def __init__(self, c):
		self.gen = c
		self.cal = [False for i in range(368)]

	def setcal(self, mi, ma):
		for i in range(mi, ma + 1):
			self.cal[i] = True

	def getgen(self):
		return self.gen

	def getcal(self):
		return self.cal
n = int(input())
for x in range(n):
	(se, a, b) = input().split()
	a = int(a)
	b = int(b)
	if a < mi:
		mi = a
	if b > ma:
		ma = b
	lof.append(Bio(se))
	lof[x].setcal(a, b)
for j in range(mi, ma + 1):
	cm = 0
	cf = 0
	for i in range(n):
		if lof[i].getcal()[j] and lof[i].getgen() == 'F':
			cf += 1
		elif lof[i].getcal()[j] and lof[i].getgen() == 'M':
			cm += 1
	if cm > cf:
		nof = 2 * cf
	else:
		nof = 2 * cm
	if nof > mf:
		mf = nof
print(mf)
