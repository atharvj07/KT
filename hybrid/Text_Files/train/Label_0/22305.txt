#include<cmath>
#include<cstdio>

#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

const double EPS=1e-8;

struct point3{
	double x,y,z;
	point3():x(0),y(0),z(0){}
	point3(double x,double y,double z):x(x),y(y),z(z){}
	point3 operator-(const point3 &a)const{ return point3(x-a.x,y-a.y,z-a.z); }
};

double dot(const point3 &a,const point3 &b){ return a.x*b.x+a.y*b.y+a.z*b.z; }

point3 cross(const point3 &a,const point3 &b){ return point3(a.y*b.z-a.z*b.y,a.z*b.x-a.x*b.z,a.x*b.y-a.y*b.x); }

double abs(const point3 &a){ return sqrt(a.x*a.x+a.y*a.y+a.z*a.z); }

double abs2(const point3 &a){ return a.x*a.x+a.y*a.y+a.z*a.z; }

struct segment3{
	point3 a,b;
	segment3(){}
	segment3(const point3 &a,const point3 &b):a(a),b(b){}
};

struct sphere{
	point3 c;
	double r;
	sphere():c(point3(0,0,0)),r(0){}
	sphere(const point3 &c,double r):c(c),r(r){}
};

double dist(const segment3 &S,const point3 &p){
	if(dot(S.b-S.a,p-S.a)<=0) return abs(p-S.a);
	if(dot(S.a-S.b,p-S.b)<=0) return abs(p-S.b);
	return abs(cross(S.b-S.a,p-S.a))/abs(S.a-S.b);
}

bool intersect(const sphere &C,const segment3 &S){
	return dist(S,C.c)<C.r+EPS;
}

bool cover(const sphere &S,const point3 &p){
	return abs(S.c-p)<S.r+EPS;
}

bool cover(const sphere &C,const segment3 &S){
	return cover(C,S.a) && cover(C,S.b);
}

int main(){
	for(int n,m,k;scanf("%d%d%d",&n,&m,&k),n;){
		sphere S[2000];
		rep(i,n) scanf("%lf%lf%lf%lf",&S[i].c.x,&S[i].c.y,&S[i].c.z,&S[i].r);
		point3 p[15];
		double bright[15];
		rep(i,m) scanf("%lf%lf%lf%lf",&p[i].x,&p[i].y,&p[i].z,bright+i);
		point3 tar; scanf("%lf%lf%lf",&tar.x,&tar.y,&tar.z);

		int mask[2000]={}; // mask[i] := ( 風船 i が遮断している光源の集合 )
		rep(j,m){
			segment3 T(p[j],tar);
			rep(i,n) if(!cover(S[i],T) && intersect(S[i],T)) mask[i]|=1<<j;
		}

		double power[15];
		rep(i,m) power[i]=bright[i]/abs2(p[i]-tar);

		double ans=0;
		rep(S,1<<m){
			double pwr=0;
			rep(i,m) if(S>>i&1) pwr+=power[i];
			if(pwr<ans) continue;

			int cnt=0;
			rep(i,n) if(mask[i]&S) cnt++;
			if(cnt<=k) ans=pwr;
		}
		printf("%.9f\n",ans);
	}

	return 0;
}
