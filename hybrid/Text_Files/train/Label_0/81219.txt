a = [int(x) for x in input().split()]
n = t - 1

ss = 0
ds = 0
sc = 0
dc = 0
while i != n and n != i:
	if a[i] > a[n] and sc <= dc:
		ss = ss + a[i]
		sc = sc + 1
		i = i + 1
	elif a[i] > a[n] and sc >= dc:
		ds = ds + a[i]
		dc = dc + 1
		i = i + 1
	elif a[i] < a[n] and sc <= dc:
		ss = ss + a[n]
		sc = sc + 1
		n = n - 1
	elif a[i] < a[n] and sc >= dc:
		ds = ds + a[n]
		dc = dc + 1
		n = n - 1
if sc <= dc:
	ss = ss + a[i]
