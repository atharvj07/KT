for i in range(int(input())):
	(a, b, c) = list(map(int, input().split()))
	(h, k) = list(map(int, input().split()))
	d = 0
	if a == 1:
		print(0)
	elif k > h:
		d = min(a // 2, c) * k
		e = min((a - c * 2) // 2, b) * h
		print(d + max(0, e))
	else:
		d = min(a // 2, b) * h
		e = min((a - b * 2) // 2, c) * k
		print(d + max(0, e))
