#include <bits/stdc++.h>
using namespace std;
const int MAXL = 200 * 1000;
const int MAXK = 100 + 10;
long long N;
int K;
long long dp[MAXL][MAXK];
int a[MAXK];
long long DP(long long n, int k) {
  if (k == 0) return n;
  if (n < MAXL) {
    long long &p = dp[n][k];
    if (p != -1) return p;
    return p = DP(n, k - 1) - DP(n / a[k], k - 1);
  }
  return DP(n, k - 1) - DP(n / a[k], k - 1);
}
int main() {
  memset(dp, -1, sizeof(dp));
  cin >> N >> K;
  for (__typeof(K + 1) i = (1); i < (K + 1); ++i) cin >> a[i];
  sort(a + 1, a + K + 1);
  cout << DP(N, K) << endl;
  return 0;
}
