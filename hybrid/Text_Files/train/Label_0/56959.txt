import java.io.File;
import java.io.PrintWriter;
import java.math.BigDecimal;
import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;
import java.util.TreeSet;

public class CodeforcesA
{

    /**
     * @param args
     */
    Scanner inp;
    PrintWriter out;

    int n, m;

    int num(int x, int y)
    {
        return m * y + x;
    }

    void run() throws Exception
    {
        inp = new Scanner(System.in);
        out = new PrintWriter(System.out);

        int n = inp.nextInt(), m = inp.nextInt();
        if (n == 1 && m == 1)
            out.print(1);
        else if (n == 1 && m == 4)
            out.print("3 1 4 2");
        else if (n == 4 && m == 1)
        {
            out.println(3);
            out.println(1);
            out.println(4);
            out.println(2);
        } else if (n * m < 4 || (n == 2 && m == 2))
            out.print(-1);
        else if (n == 2 && m == 3)
        {
            out.println("1 5 3");
            out.println("4 2 6");
        } else if (n == 3 && m == 2)
        {
            out.println("3 6");
            out.println("5 2");
            out.println("1 4");
        } else
        {
            int k = 1;
            int[][] map = new int[n][m];
            for (int i = 0; i < n; i++)
            {
                for (int j = 0; j < m; j++)
                    if ((i & 1) == (j & 1))
                        map[i][j] = k++;
            }
            for (int i = 0; i < n; i++)
            {
                for (int j = 0; j < m; j++)
                    if ((i & 1) != (j & 1))
                        map[i][j] = k++;
            }

            for (int i = 0; i < n; i++)
            {

                for (int j = 0; j < m; j++)
                    out.print(map[i][j] + " ");
                out.println();
            }
        }

        out.close();
        inp.close();

    }

    public static void main(String[] args) throws Exception
    {
        // TODO Auto-generated method stub
        new CodeforcesA().run();

    }

}
