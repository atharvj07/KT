#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int tcase;
  cin >> tcase;
  for (int tc = 1; tc <= tcase; tc++) {
    int n;
    cin >> n;
    if (n >= 4 && n % 4 == 0)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
  return 0;
}
