#include <bits/stdc++.h>
using namespace std;
int n, h;
int s[200010][2];
int main() {
  int a, b, res, ans;
  scanf("%d%d", &n, &h);
  for (int i = 1; i <= n; i++) scanf("%d%d", &s[i][0], &s[i][1]);
  a = 1;
  b = 1;
  res = h;
  while (b < n) {
    if (res > s[b + 1][0] - s[b][1]) {
      res -= s[b + 1][0] - s[b][1];
      b++;
    } else
      break;
  }
  ans = s[b][1] - s[a][0] + res;
  while (a <= n && b < n) {
    a++;
    res += s[a][0] - s[a - 1][1];
    while (b < n) {
      if (res > s[b + 1][0] - s[b][1]) {
        res -= s[b + 1][0] - s[b][1];
        b++;
      } else
        break;
    }
    ans = max(ans, s[b][1] - s[a][0] + res);
  }
  ans = max(ans, s[b][1] - s[a][0] + res);
  printf("%d\n", ans);
  return 0;
}
