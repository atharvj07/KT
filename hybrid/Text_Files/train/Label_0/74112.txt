n = int(input())
arr = list(map(int, input().split()))
chest = 0
biceps = 0
back = 0
for i in range(n + 2 // 3):
	j = 3 * i
	k = 3 * i + 1
	l = 3 * i + 2
	if j < n:
		chest += arr[j]
	if k < n:
		biceps += arr[k]
	if l < n:
		back += arr[l]
if chest > biceps and chest > back:
	print('chest')
elif biceps > chest and biceps > back:
	print('biceps')
else:
	print('back')
