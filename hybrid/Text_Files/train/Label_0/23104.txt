#include <bits/stdc++.h>
using namespace std;
const unsigned long long Inv = 6723469279985657373ull, mod = 1ull << 58;
const int N = 100000;
struct poly {
  unsigned long long a[5];
  poly(unsigned long long a0 = 0, unsigned long long a1 = 0,
       unsigned long long a2 = 0, unsigned long long a3 = 0,
       unsigned long long a4 = 0) {
    a[0] = a0;
    a[1] = a1;
    a[2] = a2;
    a[3] = a3;
    a[4] = a4;
  }
  unsigned long long& operator[](int i) { return a[i]; }
  poly operator*(poly& b) {
    poly ans;
    for (int i = 0; i < 5; i++)
      for (int j = 0; j < 5; j++) ans[(i + j) % 5] += a[i] * b[j];
    return ans;
  }
  poly operator*(int k) {
    return {a[0] * k, a[1] * k, a[2] * k, a[3] * k, a[4] * k};
  }
  poly operator+(poly& b) {
    return {a[0] + b[0], a[1] + b[1], a[2] + b[2], a[3] + b[3], a[4] + b[4]};
  }
  poly operator<<(int k) {
    k %= 5;
    return {a[(5 - k) % 5], a[(6 - k) % 5], a[(7 - k) % 5], a[(8 - k) % 5],
            a[(9 - k) % 5]};
  }
} f[N];
unsigned long long ans[N];
int n;
void operator*=(poly& a, poly b) { a = a * b; }
void operator+=(poly& a, poly b) { a = a + b; }
poly fp(poly a, long long k) {
  poly ans(1, 0, 0, 0, 0);
  for (; k; k >>= 1, a = a * a)
    if (k & 1) ans = a * ans;
  return ans;
}
poly calc(poly a, int k) {
  if (k & 1) a = a * -1;
  k = (k * 3 % 5 + 5) % 5;
  return a << k;
}
void FFT(poly* f, int tag) {
  for (int len = 1; len < N; len *= 10)
    for (int l = 0; l < N; l++)
      if (l / len % 10 == 0) {
        poly p[10], g[10];
        for (int i = 0; i < 10; i++) g[i] = {0, 0, 0, 0, 0};
        for (int i = 0; i < 10; i++) p[i] = f[l + i * len];
        for (int i = 0; i < 10; i++)
          for (int j = 0; j < 10; j++) g[i] += calc(p[j], i * j * tag);
        for (int i = 0; i < 10; i++) f[l + i * len] = g[i];
      }
}
int main() {
  ios::sync_with_stdio(0);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    f[x][0]++;
  }
  FFT(f, 1);
  for (int i = 0; i < N; i++) f[i] = fp(f[i], n);
  FFT(f, -1);
  for (int i = 0; i < n; i++) ans[i] = (f[i][0] - f[i][4]) * Inv / 32 % mod;
  for (int i = 0; i < n; i++) cout << ans[i] << '\n';
  return 0;
}
