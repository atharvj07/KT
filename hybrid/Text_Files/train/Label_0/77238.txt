import sys
from itertools import combinations
input = sys.stdin.readline
h = {'S': 1, 'E': 2, 'T': 3}
(n, k) = map(int, input().split())
strings = set([tuple(map(h.get, input().rstrip())) for _ in range(0, n)])
ans = 0
for (a, b) in combinations(strings, 2):
	c = []
	for i in range(0, k):
		if a[i] == b[i]:
			c.append(a[i])
		else:
			c.append(6 - a[i] - b[i])
	if tuple(c) in strings:
		ans += 1
print(ans // 3)
