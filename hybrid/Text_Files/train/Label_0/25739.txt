#include<cstdio>
#include<cstring>
#include<vector>
#include<algorithm>

using namespace std;

struct point {
	int x,y;
	point(){}
	point(int a, int b) : x(a), y(b) {}
}P[11111];

struct data {
	int a1,a2,t;
	data(){}
	data(int a, int b, int c) : a1(a), a2(b), t(c) {}
};

char B[111][111];
int a[111][111],b[111][111];
int n,m,sz;

vector<data> H,V;
vector<int> adj[11111];
bool seen[11111];
int R[11111];

bool bpm(int u) {
	for(int i=0;i<adj[u].size();++i) {
		int v = adj[u][i];
		if(seen[v]) continue;
		seen[v] = 1;
		if(R[v]==-1 || bpm(R[v])) {
			R[v] = u;
			return 1;
		}
	} return 0;
}

int match() {
	memset(R,-1,sizeof(R));
	int ret = 0;
	for(int u=0;u<H.size();++u) {
		for(int v=0;v<V.size();++v) seen[v]=0;
		if(bpm(u)) ret++;
	} return ret;
}

int main() {
#ifdef LOCAL
	freopen("input.txt","r",stdin);
#endif
	while(1) {
		scanf("%d%d",&n,&m),sz=0;
		if(n==0&&m==0) break;
		memset(a,0,sizeof(a)),memset(b,0,sizeof(b));
		H.clear(),V.clear();
		for(int i=1;i<=n;++i) scanf("%s",B[i]+1);
		for(int i=1;i<=n;++i) for(int j=1;j<=m;++j)
			a[i][j] = a[i][j-1] + (B[i][j]=='#'),
			b[i][j] = b[i-1][j] + (B[i][j]=='#');
		for(int i=1;i<n;++i) for(int j=1;j<m;++j) {
			int cnt = 0;
			for(int di=0;di<2;++di) for(int dj=0;dj<2;++dj)
				cnt += (B[i+di][j+dj] == '#');
			if(cnt == 3) P[sz++] = point(i,j);
		}

		for(int i=0;i<sz;++i) for(int j=i+1;j<sz;++j) {
			if(P[i].x == P[j].x && a[P[i].x][P[j].y]-a[P[i].x][P[i].y]==P[j].y-P[i].y
				&& a[P[i].x+1][P[j].y]-a[P[i].x+1][P[i].y]==P[j].y-P[i].y)
				H.push_back(data(P[i].y,P[j].y,P[i].x));
			if(P[i].y == P[j].y && b[P[j].x][P[i].y]-b[P[i].x][P[i].y]==P[j].x-P[i].x
				&& b[P[j].x][P[i].y+1]-b[P[i].x][P[i].y+1]==P[j].x-P[i].x)
				V.push_back(data(P[i].x,P[j].x,P[i].y));
		}

		for(int i=0;i<H.size();++i) for(int j=0;j<V.size();++j) {
			int x=H[i].t,y=V[j].t;
			int x1=V[j].a1,x2=V[j].a2;
			int y1=H[i].a1,y2=H[i].a2;
			if(x1<=x&&x<=x2&&y1<=y&&y<=y2) adj[i].push_back(j);
		} printf("%d\n",1+sz-(H.size()+V.size()-match()));

		for(int i=0;i<H.size();++i) adj[i].clear();
	}
}
