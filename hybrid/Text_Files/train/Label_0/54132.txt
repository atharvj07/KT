  /**
   * There are a couple of ways to improve the approach in my last attempt
   * Use the SAME array index as the question.
   *
   * So i.e. going from 1 to n.
   * ALWAYS do this to avoid confusion. Just discard the zero-index dp[0] and a[0] therefore not used
   *
   * When doing this have to still remember than n means the number of items. So the arrays must all
   * be initialized as
   *
   *      long[] a = new long[n+1]
   *      long[][] dp = new long[n+1][2]
   *
   *
   * Another idea is to write in functions boss0(..) boss1(..) and walk()
   *
   * This arguably makes the code easier to read, but also requires the variables to be made
   * available outside the solve2(..) method
   *
   *
   */




  long boss1(int i) {
    return min(r2, r1*a[i] + r1);
  }

  long boss0(int i) {
    return r1*a[i] + r3;
  }

  long walk() {
    return d;
  }


  long r1;
  long r2;
  long r3;
  long d;
  long[] a;


  void solve2(int caseNr) {


    int n = in.nextInt();
    r1 = in.nextInt();
    r2 = in.nextInt();
    r3 = in.nextInt();
    d = in.nextInt();
    a = new long[n+1];

    for (int i=1; i<=n; i++) {
      a[i] = in.nextInt();
    }
    long[][] dp = new long[n+1][2];


    // leaving node zero, boss has either 0 or 1 health
    dp[1][0] = boss0(1) + walk();
    dp[1][1] = boss1(1) + walk();


    for (int i = 2; i <= n-1; i++) {
      // going from 0 -> 0
      long cost00 = dp[i-1][0] + boss0(i) + walk();
      // going from 1 -> 0
      long cost10 = dp[i-1][1] + boss1(i) + walk() + r1 + walk() + r1 + walk();
      dp[i][0] = min(cost00, cost10);

      // going from 0 -> 1
      dp[i][1] = dp[i-1][0] + boss1(i) + walk();
    }

    long ans = dp[n-1][0] + boss0(n);
    ans = min(ans, dp[n-1][0] + boss1(n) + walk() + walk() + r1);
    ans = min(ans, dp[n-1][1] + boss0(n) + walk() + r1);  // <-- this ends on n-1
    ans = min(ans, dp[n-1][1] + boss1(n) + walk() + r1 + walk() + r1);

    System.out.println(ans);

  } // end of solve2
