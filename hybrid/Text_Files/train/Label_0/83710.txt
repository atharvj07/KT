from collections import Counter

def rev(c):
	return str.upper(c) if str.islower(c) else str.lower(c)
s = Counter(input())
t = Counter(input())
y = 0
o = 0
for k in s:
	d = min(s[k], t[k])
	s[k] -= d
	t[k] -= d
	y += d
for k in s:
	l = rev(k)
	d = min(s[k], t[l])
	o += d
	s[k] -= d
	t[l] -= d
print(y, o)
