q = int(input())
for i in range(q):
	n = int(input())
	strings = []
	lengths = []
	zeros = 0
	ones = 0
	for i in range(n):
		strings.append(input())
		lengths.append(len(strings[-1]))
		zeros += strings[-1].count('0')
		ones += strings[-1].count('1')
	lengths = sorted(lengths)
	ans = 0
	for l in lengths:
		if l <= zeros + ones:
			if l % 2 == 1:
				if zeros % 2 == 1 and l <= zeros:
					ans += 1
					zeros -= l
				elif ones % 2 == 1 and l <= ones:
					ans += 1
					ones -= l
				else:
					if zeros % 2 == 1:
						zeros -= 1
					elif ones % 2 == 1:
						ones -= 1
					available_ones = ones // 2 * 2
					available_zeros = zeros // 2 * 2
					if available_zeros + available_ones >= l - 1:
						ans += 1
						used_zeros = min(available_zeros, l - 1)
						zeros -= used_zeros
						ones -= min(available_ones, l - 1 - used_zeros)
			else:
				available_ones = ones // 2 * 2
				available_zeros = zeros // 2 * 2
				if available_zeros + available_ones >= l:
					ans += 1
					used_zeros = min(available_zeros, l)
					zeros -= used_zeros
					ones -= min(available_ones, l - used_zeros)
	print(ans)
