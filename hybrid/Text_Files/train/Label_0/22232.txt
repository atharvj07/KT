#include <bits/stdc++.h>
using namespace std;
const int maxn = 20010;
const int maxm = 30;
const int INF = 0x3f3f3f3f;
const int mod = 1e9 + 7;
int n, m;
char sx[maxn];
char sy[maxn];
int d[maxn];
int f[maxn];
int main() {
  int i, j;
  scanf("%d%d", &n, &m);
  scanf("%s", sx + 1);
  scanf("%s", sy + 1);
  int lenx = strlen(sx + 1);
  int leny = strlen(sy + 1);
  for (i = 1; i <= leny; i++) {
    int pos = i - 1;
    for (j = 1; j <= lenx; j++) {
      if (sy[pos % leny + 1] == sx[j]) pos++;
    }
    d[i] = pos / leny;
    f[i] = pos % leny + 1;
  }
  long long ans = 0;
  int pos = 1;
  for (i = 1; i <= n; i++) {
    ans += d[pos];
    pos = f[pos];
  }
  ans /= m;
  cout << ans;
  return 0;
}
