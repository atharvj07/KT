#include <bits/stdc++.h>
using namespace std;
char s[200005];
int a[200005];
int main() {
  int n;
  int minn = INT_MAX;
  scanf("%d", &n);
  getchar();
  for (int i = 0; i < n; i++) {
    scanf("%c", &s[i]);
  }
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == 'R' && s[i + 1] == 'L')
      minn = minn < a[i + 1] - a[i] ? minn : a[i + 1] - a[i];
  }
  if (minn == INT_MAX)
    printf("-1\n");
  else
    printf("%d\n", minn / 2);
  return 0;
}
