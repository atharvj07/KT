#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
long long int sx, sy;
long long int ex, ey, n;
long long int dx[maxn], dy[maxn];
char s[maxn];
int main() {
  cin >> sx >> sy >> ex >> ey >> n;
  scanf("%s", s + 1);
  for (long long int i = 0; i <= n; i++) dx[i] = dy[i] = 0;
  for (long long int i = 1; i <= n; i++) {
    dx[i] += dx[i - 1];
    dy[i] += dy[i - 1];
    if (s[i] == 'U')
      dy[i]++;
    else if (s[i] == 'L')
      dx[i]--;
    else if (s[i] == 'D')
      dy[i]--;
    else
      dx[i]++;
  }
  long long int l = 0, r = 1e18, ans = -1;
  while (r >= l) {
    long long int t = (l + r) >> 1;
    if (abs(sx + (t / n) * dx[n] + dx[t % n] - ex) +
            abs(sy + (t / n) * dy[n] + dy[t % n] - ey) <=
        t)
      r = t - 1, ans = t;
    else
      l = t + 1;
  }
  cout << ans << endl;
  return 0;
}
