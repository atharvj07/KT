for _ in range(1):
	n = int(input())
	if n in [1, 2]:
		print(-1)
		exit(0)
	num = pow(n, 2)
	if num % 2 == 0:
		num = num // 2 // 2
		first_num = num - 1
		second_num = num + 1
	else:
		num = num // 2
		first_num = num
		second_num = num + 1
	if pow(second_num, 2) - pow(first_num, 2) == pow(n, 2):
		print(first_num, second_num)
	else:
		print(-1)
