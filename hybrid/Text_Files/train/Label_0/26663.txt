#include <bits/stdc++.h>
#define REP(i, a, n) for(ll i = ((ll) a); i < ((ll) n); i++)
using namespace std;
typedef long long ll;

ll N, X;

int main(void) {
  cin >> N >> X;

  if(X == 1 || X == 2 * N - 1) {
    cout << "No" << endl;
  } else {
    cout << "Yes" << endl;

    vector<ll> a(2 * N - 1);
    REP(i, 0, 2 * N - 1) a[i] = i + 1;
    ll d = X - N;
    REP(i, 0, 2 * N - 1) cout << a[(i + d + 2 * N - 1) % (2 * N - 1)] << endl;
  }
}
