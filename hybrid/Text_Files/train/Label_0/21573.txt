#include<bits/stdc++.h>
using namespace std;
typedef pair<int,int> PII;
const int N = 3e5+7;

int r[N],c[N];
map<PII,int> mp;

int main()
{
	int n,m,k,x,y;
	scanf("%d%d%d",&n,&m,&k);
	int xx=0,yy=0;
	for(int i=0;i<k;i++){
		scanf("%d%d",&x,&y);
		mp[make_pair(x,y)]=1;
		r[x]++,c[y]++;
		xx = max(xx,r[x]);
		yy = max(yy,c[y]);
	}
	vector<int> vx,vy;
	for(int i=1;i<=n;i++) if(r[i]==xx) vx.push_back(i);
	for(int i=1;i<=m;i++) if(c[i]==yy) vy.push_back(i);
	int flag=0;
	for(int i=0;i<vx.size();i++){
		for(int j=0;j<vy.size();j++){
			if(!mp[make_pair(vx[i],vy[j])]){
				flag=1; break;
			}
		}
		if(flag) break;
	}
	if(flag) cout<<xx+yy<<"\n";
	else cout<<xx+yy-1<<"\n";
	return 0;
}
