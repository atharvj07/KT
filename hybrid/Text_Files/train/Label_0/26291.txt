int LIS(vector<int> v) {
  if (v.size() == 0) {
    return (0);
  }
  std::vector<int> t(v.size(), 0);
  std::vector<int>::iterator it;
  int len = 1;
  t[0] = v[0];
  for (int i = 1; i < v.size(); i++) {
    if (v[i] > t[len - 1]) {
      t[len++] = v[i];
    } else {
      it = find(t.begin(), t.begin() + len, v[i]);
      if (it != t.begin() + len) {
        continue;
      }
    }
    it = upper_bound(t.begin(), t.begin() + len, v[i]);
    *it = v[i];
  }
  return (len);
}
