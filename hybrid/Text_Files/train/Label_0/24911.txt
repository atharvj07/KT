#include <bits/stdc++.h>
long long factorial(long long n) {
  if (n == 1) return 1;
  return n * factorial(n - 1);
}
long long max(long long a, long long b) { return (a > b ? a : b); }
long long min(long long a, long long b) { return (a < b ? a : b); }
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int x, y, i, j;
  map<int, int> a, b;
  long long ar[8][2];
  long long f = 1;
  for (i = 0; i < 8; i++) {
    cin >> x >> y;
    ar[i][0] = x;
    ar[i][1] = y;
    for (j = 0; j < i; j++) {
      if (ar[j][0] == x && ar[j][1] == y) f = 0;
    }
    a[x]++;
    b[y]++;
  }
  map<int, int>::iterator p, q;
  p = a.begin();
  q = b.begin();
  p++;
  q++;
  if (a.size() == 3 && b.size() == 3 && f == 1) {
    if (q->second == 2 && p->second == 2)
      cout << "respectable";
    else
      cout << "ugly";
  } else
    cout << "ugly";
  return 0;
}
