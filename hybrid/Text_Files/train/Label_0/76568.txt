from collections import deque
import heapq
import math
import bisect

def main():
	N = int(input())
	A = list(map(int, input().split()))
	pos = neg = 0
	for i in range(N):
		if A[i] > 0:
			pos += 1
		elif A[i] < 0:
			neg += 1
	if pos >= (N + 1) // 2:
		print(1)
	elif neg >= (N + 1) // 2:
		print(-1)
	else:
		print(0)
main()
