int Clone(string s){
  if(s=="NULL")return 0;
  int res=Expr(s);
  // NULL
  if(res==0)return 0;
  // Error
  if(res==-1){
    isError=true;
    return -1;
  }
  int sz=allocMem[res];
  if(sz==0){
    isError=true;
    return -1;
  }
  // NULL
  if(M-sz<0)return 0;
  allocMem[V++]=sz;
  M-=sz;
  return V-1;
}

int Malloc(string s){
  int a=atoi(s.c_str());
  // s«

  }
  allocMem[V++]=a;
  M-=a;
  // èÄêÌQÆÔµ
  return V-1;
}

int Assign(string s){
  char var=s[0];
  int res=Expr(s.substr(2));
  refIdx[var]=res;
  return res;
}

int Expr(string s){
  if(s=="NULL")return 0;
  if(s[0]=='('&&s[s.size()-1]==')'){
    int cnt=1;
    bool ok=true;
    for(int i=1;i<s.size()-1;i++){
      if(s[i]=='(')cnt++;
      else if(s[i]==')')cnt--;
      if(cnt==0)ok=false;
    }
    if(ok)return Expr(s.substr(1,s.size()-2));
  }
  if(!s.compare(0,6,"malloc"))
    return Malloc(s.substr(7,s.size()-8));
  else if(!s.compare(0,5,"clone"))
    return Clone(s.substr(6,s.size()-7));
  if(s[0]>='A'&&s[0]<='Z'){
    if(s.size()==1)return refIdx[s[0]];
    return Assign(s);
  }
}
