t = int(input())
for i in range(t):
	n = int(input())
	c = 0
	ev = 0
	od = 0
	l = list(map(int, input().split()))
	if n == 1:
		if l[0] % 2 != 0:
			print(-1)
		else:
			print(0)
		continue
	for i in range(n):
		if l[i] % 2 == 0:
			if i % 2 != 0:
				ev += 1
		elif i % 2 == 0:
			od += 1
	if ev != od:
		print(-1)
	else:
		print(ev)
