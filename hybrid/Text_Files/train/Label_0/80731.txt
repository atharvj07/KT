import bisect
m = n = int(input())
gaba = 0
l = []
a = []
primeiro = []
ultimos = []
for _ in range(n):
	b = input().split()
	l.append(int(b[0]))
	b.pop(0)
	a.append(list(map(int, b)))
for i in range(n - 1, -1, -1):
	men = a[i][0]
	tot = 0
	for j in range(l[i]):
		tot += 1
		if a[i][j] <= men:
			men = a[i][j]
		else:
			gaba += 2 * m - 1
			a.pop(i)
			m -= 1
			tot = 0
			break
	if tot == l[i]:
		primeiro.append(a[i][0])
primeiro.sort()
for i in range(m):
	id = bisect.bisect_right(primeiro, a[i][-1])
	gaba += len(primeiro) - id
print(gaba)
