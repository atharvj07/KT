n = int(input())
pow10 = [1]
last = 1
for i in range(1, 200010):
	last *= 10
	last %= 998244353
	pow10.append(last)
res = []
for i in range(1, n):
	res.append(int((10 * 9 * 9 * (n - 2 - i + 1) * pow10[n - i - 2] + 2 * 9 * 10 * pow10[n - i - 1]) % 998244353))
res.append(10)
print(' '.join(map(str, res)))
