#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 1) + (x << 3) + (ch ^ 48);
    ch = getchar();
  }
  return x * f;
}
const int N = 2e6 + 7;
int fir[N], from[N << 1], to[N << 1], cntt;
inline void add(int &a, int &b) {
  from[++cntt] = fir[a];
  fir[a] = cntt;
  to[cntt] = b;
}
int n, m, col[N], Ans[N];
struct dat {
  int x, k, id;
  inline bool operator<(const dat &tmp) const { return x < tmp.x; }
} d[N];
int sz[N], son[N];
void dfs1(int x, int fa) {
  sz[x] = 1;
  for (int i = fir[x]; i; i = from[i]) {
    int &v = to[i];
    if (v == fa) continue;
    dfs1(v, x);
    sz[x] += sz[v];
    if (sz[v] > sz[son[x]]) son[x] = v;
  }
}
int id[N], dfs_clock, cnt[N], ans[N];
inline void ins(int c) {
  cnt[c]++;
  ans[cnt[c]]++;
}
inline void del(int c) {
  ans[cnt[c]]--;
  cnt[c]--;
}
inline void work(int x) {
  for (int i = lower_bound(d + 1, d + m + 1, (dat){x, 0, 0}) - d; d[i].x == x;
       i++)
    Ans[d[i].id] = ans[d[i].k];
}
void dfs2(int x, int fa, bool flag) {
  id[++dfs_clock] = x;
  int L = dfs_clock;
  if (!son[x]) {
    ins(col[x]);
    work(x);
    if (!flag) del(col[x]);
    return;
  }
  for (int i = fir[x]; i; i = from[i]) {
    int &v = to[i];
    if (v == fa || v == son[x]) continue;
    dfs2(v, x, 0);
  }
  int R = dfs_clock;
  dfs2(son[x], x, 1);
  for (int i = L; i <= R; i++) ins(col[id[i]]);
  work(x);
  if (!flag)
    for (int i = L; i <= dfs_clock; i++) del(col[id[i]]);
}
int main() {
  n = read(), m = read();
  int a, b;
  for (int i = 1; i <= n; i++) col[i] = read();
  for (int i = 1; i < n; i++) {
    a = read(), b = read();
    add(a, b);
    add(b, a);
  }
  dfs1(1, 0);
  for (int i = 1; i <= m; i++) d[i].x = read(), d[i].k = read(), d[i].id = i;
  sort(d + 1, d + m + 1);
  dfs2(1, 0, 1);
  for (int i = 1; i <= m; i++) printf("%d\n", Ans[i]);
  return 0;
}
