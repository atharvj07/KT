def find(l, r):
	if l == r:
		return 0
	big = 0
	while 2 ** big <= r:
		big += 1
	if 2 ** big <= l:
		return 0
	biggest = 0
	for i in range(big):
		gap = 2 ** (i + 1)
		q1 = int(l / gap)
		x1 = l % gap
		y1 = int(max(0, x1 - gap / 2))
		z1 = int(gap / 2 * q1 + y1)
		q2 = int((r + 1) / gap)
		x2 = (r + 1) % gap
		y2 = int(max(0, x2 - gap / 2))
		z2 = int(gap / 2 * q2 + y2)
		biggest = max(biggest, z2 - z1)
	return r - l + 1 - biggest
cases = int(input())
for _ in range(cases):
	line = input().split()
	l = int(line[0])
	r = int(line[1])
	print(find(l, r))
