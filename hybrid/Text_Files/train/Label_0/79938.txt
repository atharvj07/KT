for _ in range(int(input())):
	n = input()
	odd = []
	eve = []
	for i in n:
		if int(i) % 2 != 0:
			odd.append(i)
		else:
			eve.append(i)
	i = 0
	j = 0
	while i < len(odd) and j < len(eve):
		if odd[i] < eve[j]:
			print(odd[i], end='')
			i += 1
		else:
			print(eve[j], end='')
			j += 1
	for k in range(i, len(odd)):
		print(odd[k], end='')
	for k in range(j, len(eve)):
		print(eve[k], end='')
	print()
