n = int(input())
s = input()
(u, d) = (0, 0)
for i in s:
	if i == 'X':
		u = u + 1
	else:
		d = d + 1
if u > d:
	print(n // 2 - d)
elif u == d:
	print(0)
else:
	print(n // 2 - u)
if u == d:
	print(s)
elif u > d:
	v = n // 2 - d
	new = ''
	for i in s:
		if v != 0 and i == 'X':
			new = new + 'x'
			v = v - 1
		else:
			new = new + i
	print(new)
else:
	v = n // 2 - u
	new = ''
	for i in s:
		if v != 0 and i == 'x':
			new = new + 'X'
			v = v - 1
		else:
			new = new + i
	print(new)
