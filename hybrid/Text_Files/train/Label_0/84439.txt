possible = True
(pw, pb) = (False, False)
for i in range(8):
	line = input()
	count = 0
	c = 1
	(cw, cb) = (False, False)
	if not possible:
		continue
	for i in range(1, 8):
		if line[i] == line[i - 1]:
			c += 1
			if line[i] == 'W':
				cw = True
			elif line[i] == 'B':
				cb = True
		else:
			if c > 1:
				count += 1
			c = 1
		if c >= 3:
			possible = False
			break
	if c > 2:
		possible = False
	if c >= 2:
		count += 1
	if count >= 2:
		possible = False
	if cw and pw:
		possible = False
	if cb and pb:
		possible = False
	pw = cw
	pb = cb
if possible:
	print('YES')
else:
	print('NO')
