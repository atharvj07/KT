

import java.util.Scanner;

public class Main {

    
    public static void main(String[] args) {

        Scanner sc=new Scanner(System.in);
        int N=sc.nextInt();
        int A[]=new int[N];
        int c=0;
        
        for(int i=0;i<N;i++){
           int n=sc.nextInt();
           A[i]=n;
        }
        
        for(int i=1;i<N;i++){
            A[i]=A[i]+A[i-1];
        }
        for(int i=1;i<N;i++){
            if((A[N-1]-A[i-1])==A[i-1])
                c++;
        }
        System.out.println(c);
    }

}
