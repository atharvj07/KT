#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; ++i) {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    vector<int> a(k1);
    for (int j = 0; j < k1; ++j) {
      cin >> a[j];
    }
    vector<int> b(k2);
    for (int j = 0; j < k2; ++j) {
      cin >> b[j];
    }
    if (*max_element(a.begin(), a.end()) < *max_element(b.begin(), b.end())) {
      cout << "NO"
           << "\n";
    } else {
      cout << "YES"
           << "\n";
    }
  }
}
