import sys
MOD = 998244353
n = int(input())
a = list(map(int, stdin.readline().split()))
b = list(map(int, stdin.readline().split()))
b.sort()
for i in range(n):
	a[i] *= (i + 1) * (n - i)
a.sort(reverse=True)
ans = 0
for i in range(n):
	ans += b[i] * a[i] % MOD
print(ans % MOD)
