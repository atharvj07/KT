#include <bits/stdc++.h>
using namespace std;
const string FILENAME = "";
const bool FILE_IO = 1;
const int INF = 1000000000;
void init() {
  if (!FILENAME.empty()) {
    freopen((FILENAME + ".in").c_str(), "r", stdin);
    freopen((FILENAME + ".out").c_str(), "w", stdout);
  }
}
void solve() {
  int n;
  string s;
  double s1 = 0, s2 = 0;
  cin >> n >> s;
  for (int i = 0, j = n / 2; i < n / 2 && j < n; i++, j++) {
    if (s[i] == '?') {
      s1 += 4.5;
    } else {
      s1 += s[i] - '0' + 0.0;
    }
    if (s[j] == '?') {
      s2 += 4.5;
    } else {
      s2 += s[j] - '0' + 0.0;
      ;
    }
  }
  if (s1 == s2) {
    cout << "Bicarp";
  } else {
    cout << "Monocarp";
  }
}
int main() {
  init();
  solve();
  return 0;
}
