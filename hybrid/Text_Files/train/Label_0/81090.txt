import sys, os.path
if os.path.exists('input.txt'):
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')
for _ in range(int(input())):
	n = int(input())
	a = input()
	b = input()
	c = input()
	s = []
	ind = [0, 0, 0]
	while ind[0] < 2 * n and ind[1] < 2 * n and (ind[2] < 2 * n):
		f = [False, False, False]
		if a[ind[0]] == '1':
			f[0] = True
		if b[ind[1]] == '1':
			f[1] = True
		if c[ind[2]] == '1':
			f[2] = True
		if sum(f) > 1:
			s.append('1')
			for i in range(3):
				if f[i]:
					ind[i] += 1
		else:
			s.append('0')
			for i in range(3):
				if not f[i]:
					ind[i] += 1
	if ind[0] != 2 * n and 2 * n - ind[0] <= 3 * n - len(s):
		for i in range(ind[0], len(a)):
			s.append(a[i])
	if ind[1] != 2 * n and 2 * n - ind[1] <= 3 * n - len(s):
		for i in range(ind[1], len(b)):
			s.append(b[i])
	if ind[2] != 2 * n and 2 * n - ind[2] <= 3 * n - len(s):
		for i in range(ind[2], len(c)):
			s.append(c[i])
	print(''.join(s))
