def m_pow(x, y, m):
	if y == 0:
		return 1
	if y & 1:
		return m_pow(x, y - 1, m) * x % m
	else:
		t = m_pow(x, y >> 1, m)
		return t * t % m
(p, k) = map(int, input().split())
used = [0] * p
if k == 0:
	print(m_pow(p, p - 1, 1000000007))
else:
	c = 1 if k == 1 else 0
	for x in range(1, p):
		if not used[x]:
			y = x
			while not used[y]:
				used[y] = True
				y = k * y % p
			c += 1
	print(m_pow(p, c, 1000000007))
