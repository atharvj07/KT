t = int(input())
for _ in range(t):
	(n, T) = map(int, input().split())
	a = list(map(int, input().split()))
	ans = []
	b = False
	for i in range(n):
		if a[i] < T / 2:
			ans.append(0)
		elif a[i] > T / 2:
			ans.append(1)
		elif b:
			ans.append(1)
			b = not b
		else:
			ans.append(0)
			b = not b
	print(*ans)
