template <typename Arg1>
void __f(const char* name, Arg1&& arg1) {
  cerr << name << " : " << arg1 << endl;
}
template <typename Arg1, typename... Args>
void __f(const char* names, Arg1&& arg1, Args&&... args) {
  const char* comma = strchr(names + 1, ',');
  cerr.write(names, comma - names) << " : " << arg1 << " | ";
  __f(comma + 1, args...);
}
const int N = 200001;
const int M = 70;
vector<int64_t> c[M];
int main() {
  ios_base::sync_with_stdio(false);
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int64_t a;
    cin >> a;
    int64_t v = (a & -a);
    int id = -1;
    while (v) {
      id++;
      v >>= 1;
    }
    c[id].push_back(a);
  }
  int id = -1;
  int cnt = 0;
  for (int i = 0; i < M; i++) {
    if (c[i].size() > cnt) {
      cnt = c[i].size();
      id = i;
    }
  }
  vector<int64_t> v;
  for (int i = 0; i < M; i++) {
    if (i == id) continue;
    for (int64_t& it : c[i]) v.push_back(it);
  }
  cout << v.size() << endl;
  for (int64_t& it : v) cout << it << ' ';
  cout << endl;
  return 0;
}
