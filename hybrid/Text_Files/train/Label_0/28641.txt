#include <bits/stdc++.h>
using namespace std;
long long int gcd(long long int a, long long int b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
int mul_mod(int a, int b, int m) {
  int x = 0, y = a;
  while (b > 0) {
    if (b & 1) {
      x = (x + y) % m;
    }
    y = (y + y) % m;
    b >>= 1;
  }
  return x;
}
long long int power(long long int x, long long int y, long long int p) {
  long long int res = 1;
  x = x % p;
  while (y > 0) {
    if (y & 1) res = (res * x) % p;
    y = y >> 1;
    x = (x * x) % p;
  }
  return res;
}
int main() {
  int q;
  cin >> q;
  while (q--) {
    long long int x, y, k;
    cin >> x >> y >> k;
    x = abs(x);
    y = abs(y);
    if (x > y) {
      long long int temp = x;
      x = y;
      y = temp;
    }
    long long int ans = 0;
    if (k < y) ans = -1;
    if (ans != -1) {
      long long int mv = (y - x);
      long long int K = k - x;
      if (mv % 2 == 0) {
        if (K % 2 == 0)
          ans = k;
        else
          ans = k - 2;
      } else if (mv % 2 == 1) {
        ans = k - 1;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
