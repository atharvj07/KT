#include <bits/stdc++.h>
using namespace std;
void change(int i, int j, vector<int> &vec) {
  for (int k = i; k <= j; ++k) {
    vec[k] = 1 - vec[k];
  }
}
void test() {
  int n;
  cin >> n;
  vector<int> vec(n);
  for (int i = 0; i < n; ++i) {
    cin >> vec[i];
  }
  vector<int> orignal = vec;
  int count_ones = 0;
  int maxone = INT_MIN;
  for (int i = 0; i < n; ++i) {
    for (int j = i; j < n; ++j) {
      change(i, j, vec);
      count_ones = count(vec.begin(), vec.end(), 1);
      if (count_ones > maxone) {
        maxone = count_ones;
      }
      vec = orignal;
    }
  }
  cout << maxone << endl;
}
int main() {
  int t = 1;
  while (t--) {
    test();
  }
  return 0;
}
