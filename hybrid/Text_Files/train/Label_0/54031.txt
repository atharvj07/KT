import java.util.*;
public class main{
	public static void main(String args[]){
		Scanner sc=new Scanner(System.in);
		int test=sc.nextInt();
		int arr[]=new int[test];
		for(int i=0;i<test;i++)
			arr[i]=sc.nextInt();
		Arrays.sort(arr);
		if(test%2!=0)
			System.out.println(arr[test/2]);
		else
			System.out.println(arr[test/2-1]);
	}
}
