import java.util.*;
import java.io.*;

public class B {
    int[] a;
    TreeMap<Integer, Integer> cnt;

    int countPairs(int a, int pow) {
        int pow2 = 1 << pow;
        if (cnt.containsKey(pow2 - a)) {
            if (pow2 - a == a)
                return cnt.get(pow2 - a) - 1;
            else
                return cnt.get(pow2 - a);
        }
        return 0;
    }

    public void solve() throws IOException {
        int n = nextInt();
        a = new int[n];
        cnt = new TreeMap<>();
        for (int i = 0; i < n; i++) {
            a[i] = nextInt();
            if (cnt.containsKey(a[i]))
                cnt.put(a[i], cnt.get(a[i]) + 1);
            else
                cnt.put(a[i], 1);
        }

        long cntPairs = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 1; j < 32; j++)
                cntPairs += countPairs(a[i], j);
        }

        cntPairs /= 2;
        output.println(cntPairs);
    }

    public static void main(String[] args) throws IOException {
        new B();
    }

    StringTokenizer st = null;
    BufferedReader br;
    PrintWriter output;

    B() throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        output = new PrintWriter(System.out);
        solve();
        br.close();
        output.close();
    }

    String next() throws IOException {
        while (st == null || !st.hasMoreTokens())
            st = new StringTokenizer(br.readLine());

        return st.nextToken();
    }

    long nextLong() throws IOException {
        return Long.parseLong(next());
    }

    int nextInt() throws IOException {
        return Integer.parseInt(next());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(next());
    }
}
