#include <bits/stdc++.h>
using namespace std;
long long n;
vector<long long> vec[112345];
long long flag = 0;
vector<pair<long long, long long> > ans;
long long nn = 1;
long long num = 1;
void dfs(long long node, long long parent) {
  if (flag) return;
  for (auto x : vec[node]) {
    if (x != parent) {
      if (vec[x].size() == 1) {
        ans.push_back(make_pair(num, x));
      } else if (vec[x].size() > 2) {
        flag = 1;
      } else {
        dfs(x, node);
      }
    }
  }
}
int main() {
  cin >> n;
  for (long long i = 0; i < n - 1; i++) {
    long long a, b;
    cin >> a >> b;
    vec[a].push_back(b);
    vec[b].push_back(a);
  }
  for (long long i = 1; i <= n; i++) {
    if (vec[i].size() > nn) {
      nn = vec[i].size();
      num = i;
    }
  }
  dfs(num, -1);
  if (!flag) {
    cout << "Yes" << endl;
    cout << ans.size() << endl;
    for (auto x : ans) {
      cout << x.first << " " << x.second << endl;
    }
  } else {
    cout << "No" << endl;
  }
  return 0;
}
