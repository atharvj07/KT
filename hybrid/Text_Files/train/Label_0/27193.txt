#include <bits/stdc++.h>
using namespace std;
long long n;
long long prime[20];
int cnt[20], re[20], sum;
long long num[3];
long long ret;
void update() {
  num[2] = n / num[0] / num[1];
  if (num[0] <= num[1] && num[1] <= num[2]) {
    long long s = num[0] + num[1] + num[2];
    if (s % 2 == 1) return;
    s /= 2;
    long long c = s - num[0];
    long long b = s - num[1];
    long long a = s - num[2];
    if (a <= 0) return;
    if (a == b && b == c) {
      ret += 1;
      return;
    }
    if (a == b || b == c) {
      ret += 3;
      return;
    }
    ret += 6;
  }
}
void search2(int dep, long long v) {
  if (dep > sum) {
    num[1] = v;
    update();
    return;
  }
  long long nv = v;
  for (int i = 0; i <= re[dep]; i++) {
    if (i) nv = nv * prime[dep];
    search2(dep + 1, nv);
  }
}
void search(int dep, long long v) {
  if (dep > sum) {
    num[0] = v;
    search2(1, 1);
    return;
  }
  long long nv = v;
  for (int i = 0; i <= cnt[dep]; i++) {
    if (i) nv = nv * prime[dep];
    re[dep] = cnt[dep] - i;
    search(dep + 1, nv);
  }
}
int main() {
  cin >> n;
  if (n % 3 != 0) {
    printf("0\n");
    return 0;
  }
  n = n / 3;
  long long k = n;
  sum = 0;
  for (int i = 2; (long long)i * i <= k; i++)
    if (k % i == 0) {
      ++sum;
      while (k % i == 0) {
        ++cnt[sum];
        k /= i;
      }
      prime[sum] = i;
    }
  if (k != 1) {
    ++sum;
    cnt[sum] = 1;
    prime[sum] = k;
  }
  ret = 0;
  search(1, 1);
  cout << ret << endl;
  return 0;
}
