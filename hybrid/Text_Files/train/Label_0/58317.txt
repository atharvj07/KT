var input = readline().split("");
var combo = [];
var prev;
var obj = {};
var i = 0;
var ways = 0;
for(var e of input){
    if(prev && prev==e)combo[i][e]++;
    else{
        combo.length>0?i++:'';
        combo[i]={};
        combo[i][e]=1;
    }
    prev = e;
}
if(combo.length%2 === 0){
    print("0");
}
else {
    var temp = Math.floor(combo.length/2);
    var middle_key = Object.keys(combo[temp])[0];
    if(combo[temp][middle_key]<2){
        print("0");
    }else{
        var exist = true;
        for(var i=temp+1,j=temp-1; i< combo.length; i++,j--){
            var key1 = Object.keys(combo[i])[0];
            var key2 = Object.keys(combo[j])[0];
            if(key1 != key2 || combo[i][key1]+combo[j][key2] < 3){
                exist = false;
                break;
            }
        }
        if(exist)ways+=combo[temp][middle_key]+1;
        print(ways);
    }

}
