#include<bits/stdc++.h>
using namespace std;
#define MP make_pair
#define x first
#define y second
 
typedef long long ll;
typedef pair<int, int> PII;
 
const int N = 2e5 + 10;
const long long INF = (ll)1e18;
const double PI = acos(-1.0);
 
PII a, b, c[N];
int s[N];
 
int find(int n)
{
	int n1 = 0;
	for(int i = 0; i < n; i++){
		int t = lower_bound(s, s+n1, c[i].y) - s;
		s[t] = c[i].y;
		if(t == n1) n1++;
	}
	return n1;
}
 
int main ()
{
	scanf("%d %d %d %d", &a.x, &a.y, &b.x, &b.y);
	if(a.x > b.x) swap(a, b);
	int x1 = a.x, x2 = b.x, y1 = a.y, y2 = b.y;
	if(x1 > x2) swap(x1, x2);
	if(y1 > y2) swap(y1, y2);
	int n, n1 = 0;
	scanf("%d", &n);
	for(int i = 0; i < n; i++){
		scanf("%d %d", &c[n1].x, &c[n1].y);
		if(c[n1].x >= x1 && c[n1].x <= x2 && c[n1].y >= y1 && c[n1].y <= y2)
			n1++;
	}
	
	sort(c, c+n1);
	
	double ans = 0;
	if(x1 == x2 || y1 == y2){
		ans = (x2 - x1) * 100.0 + (y2 - y1) * 100.0 + n1 * (PI - 2.0) * 10;
	}else{
		if(a.y > b.y)
		for(int i = 0; i < n1; i++)
			c[i].y = a.y - c[i].y;
		int t = find(n1);
		if(t == x2 - x1 + 1 || t == y2 - y1 + 1) ans += PI * 5;
		ans += (x2 - x1) * 100.0 + (y2 - y1) * 100.0 + (5 * PI - 20.0) * t;
	}
	
	printf("%.12lf\n", ans);
	
	return 0;
}
