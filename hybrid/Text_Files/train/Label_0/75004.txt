n = int(input())
a = list(map(int, input().split()))
l1 = a[0]
l2 = None
l2p = 0
for i in range(n):
	if a[i] != l1:
		l2 = a[i]
		l2p = i
		break
r1 = a[-1]
r2 = None
r2p = 0
for i in range(n - 1, -1, -1):
	if a[i] != l1:
		r2 = a[i]
		r2p = i
		break
if l1 != r1:
	print(n - 1)
else:
	print(max(r2p, n - 1 - l2p))
