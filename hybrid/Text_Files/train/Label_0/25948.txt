#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, M = 1e6 + 6, OO = 0x3f3f3f3f;
int dis[N], vis[N], l, f, s, t, sum;
vector<int> adj[N];
vector<int> vec;
bool DFS(int u) {
  if (u == s) {
    vec.push_back(s);
    return 1;
  } else {
    vis[u] = 1;
    for (int v : adj[u]) {
      if (!vis[v]) {
        if (DFS(v)) {
          vec.push_back(u);
          return 1;
        }
      }
    }
  }
  return 0;
}
void BFS(int src) {
  memset(dis, OO, sizeof dis);
  dis[src] = 0;
  queue<int> q;
  q.push(src);
  while (!q.empty()) {
    int u = q.front();
    l = u;
    q.pop();
    for (int v : adj[u]) {
      if (dis[v] == OO) {
        dis[v] = dis[u] + 1;
        q.push(v);
      }
    }
  }
}
void Multi_BFS(int src) {
  queue<int> q;
  for (int i = 0; i < vec.size(); ++i) q.push(vec[i]);
  while (!q.empty()) {
    int u = q.front();
    l = u;
    q.pop();
    for (int v : adj[u]) {
      if (dis[v] == OO) {
        dis[v] = dis[u] + 1;
        q.push(v);
      }
    }
  }
}
int n, m, u, v;
int main() {
  scanf("%d", &n);
  m = n - 1;
  for (int i = 0; i < m; ++i) {
    scanf("%d %d", &u, &v);
    --u;
    --v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  BFS(0);
  f = l;
  BFS(l);
  s = l;
  sum += dis[s];
  DFS(f);
  memset(dis, OO, sizeof dis);
  for (int i = 0; i < vec.size(); ++i) dis[vec[i]] = 0;
  Multi_BFS(f);
  sum += dis[l];
  if (l == s || l == f) {
    for (int i = 0; i < vec.size(); ++i)
      if (vec[i] != f && vec[i] != s) l = vec[i];
  }
  printf("%d\n%d %d %d", sum, s + 1, l + 1, f + 1);
  return 0;
}
