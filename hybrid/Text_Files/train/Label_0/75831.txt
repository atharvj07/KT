(n, k) = map(int, input().split())
arr = [int(i) for i in input().split()]
ans = []
while k >= 0:
	x = 0
	y = 0
	for j in range(n):
		if arr[j] > arr[x]:
			x = j
		if arr[j] < arr[y]:
			y = j
	if k == 0 or arr[x] - arr[y] <= 1:
		print(arr[x] - arr[y], len(ans))
		for j in ans:
			print(j[0], j[1])
		break
	if arr[x] != arr[y]:
		ans.append((x + 1, y + 1))
		arr[x] -= 1
		arr[y] += 1
	k -= 1
