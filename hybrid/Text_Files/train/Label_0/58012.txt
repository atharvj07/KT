import java.io.BufferedReader;
import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class D {

	static class FastScanner {
		BufferedReader reader;
		StringTokenizer tokens;

		public FastScanner() {
			reader = new BufferedReader(new InputStreamReader(System.in));
		}
		
		public FastScanner(File f) throws FileNotFoundException {
			reader = new BufferedReader(new FileReader(f));
		}

		public String nextToken() {
			while (tokens == null || !tokens.hasMoreTokens()) {
				try {
					tokens = new StringTokenizer(reader.readLine());
				} catch (IOException e) {
					return null;
				}
			}
			return tokens.nextToken();
		}

		public int nextInt() {
			return Integer.parseInt(nextToken());
		}

		public double nextDouble() {
			return Double.parseDouble(nextToken());
		}
		
		public long nextLong() {
			return Long.parseLong(nextToken());
		}

		public boolean hasNextToken() {
			while (tokens == null || !tokens.hasMoreTokens()) {
				try {
					tokens = new StringTokenizer(reader.readLine());
				} catch (Exception e) {
					return false;
				}
			}
			return true;
		}

		public void close() {
			try {
				reader.close();
			} catch (IOException e) {
			}
		}
	}
	
	static class Event implements Comparable<Event> {
		
		int time;
		int type;
		int user;
		
		public Event(int time, int type) {
			this.time = time;
			this.type = type;
		}
		
		public Event(int time, int type, int user) {
			this.time = time;
			this.type = type;
			this.user = user;
		}
		
		public void setUser(int user) {
			this.user = user;
		}
		
		public boolean equals(Object o) {
			if (o instanceof Event)
				return this.equals((Event) o);
			return false;
		}
		
		public boolean equals(Event e) {
			return time == e.time && type == e.type && user == e.user;
		}
		
		@Override
		public int compareTo(Event o) {
			if (time == o.time) {
				if (type == o.type)
					return user - o.user;
				return o.type - type;				
			}
			return time - o.time;
		}
		
		public String toString() {
			return "[" + time + " " + type + " " + user + "]";
		}
		
		public Event clone() {
			return new Event(time, type, user);
		}
		
	}
	
	public static void main(String[] args) throws FileNotFoundException {
//		FastScanner in = new FastScanner(new File("input.txt"));
//		PrintWriter out = new PrintWriter(new File("out1.txt"));
		FastScanner in = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		int m = in.nextInt();
		int t = in.nextInt();
		TreeMap<Event, Integer> events = new TreeMap<>();
		for (int i = 0; i < n; i++) {
			String s = in.nextToken();
			int h = Integer.parseInt(s.substring(0, 2));
			int min = Integer.parseInt(s.substring(3, 5));
			int sec = Integer.parseInt(s.substring(6, 8));
			Event e = new Event(h * 3600 + min * 60 + sec, 0);
			if (events.containsKey(e))
				events.replace(e, events.get(e) + 1);
			else
				events.put(e, 1);
		}
		int un = 0;
		int active = 0;
		boolean reached = false;
		Event lastLogin = new Event(0, 0);
		ArrayList<Event> logins = new ArrayList<>();
		while (events.size() != 0) {
			Event e = events.firstKey();
			if (events.get(e) == 1)
				events.remove(e);
			else
				events.replace(e, events.get(e) - 1);
			if (e.type == 0) {
				if (active < m) {
					un++;
					active++;
					lastLogin = e;
				} else {
					Event toRemove = new Event(lastLogin.time + t, 1, lastLogin.user);
					if (events.get(toRemove) == 1)
						events.remove(toRemove);
					else
						events.replace(toRemove, events.get(toRemove) - 1);
					lastLogin.time = e.time; 
				}
				e.setUser(un);
				logins.add(e.clone());
				Event toAdd = new Event(e.time + t, 1, un);
				if (events.containsKey(toAdd))
					events.replace(toAdd, events.get(toAdd) + 1);
				else
					events.put(toAdd, 1);
				if (active == m)
					reached = true;
			} else {
				active--;
			}
		}
		if (!reached) {
			out.println("No solution");
		} else {
			out.println(un);
			for (int i = 0; i < logins.size(); i++) {
				out.println(logins.get(i).user);
			}			
		}
		
		in.close();
		out.close();
	}
	
}
