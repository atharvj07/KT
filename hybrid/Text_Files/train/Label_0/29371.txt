#include<bits/stdc++.h>
using namespace std;
const int MAXN = 200005;
int N,M,a[MAXN/2],b[MAXN/2],f[MAXN];
long long ans[MAXN],cnt[MAXN];

int find(int x){
	return (f[x]==x) ? x : f[x]=find(f[x]);
}

int main()
{
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	cin >> N >> M;
	for(int i = 1; i <= N; i++){
		f[i] = i;
		cnt[i] = 1;
	}
	for(int i = 1; i <= M; i++){
		cin >> a[i] >> b[i];
	}
	long long ANS = 1ll*(N-1)*N/2;
	for(int i = M; i > 0; i--){
		int x=find(a[i]),y=find(b[i]);
		ans[i] = ANS;
		if(x==y) continue;
		ANS -= cnt[x]*cnt[y];
		f[x] = y;
		cnt[y] += cnt[x];
	}
	for(int i = 1; i <= M; i++)
		cout << ans[i] << "\n";
	return 0;
}
