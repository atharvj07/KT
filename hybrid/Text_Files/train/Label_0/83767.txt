b = list(map(int, input().split(' ')))
le = b[0]
su = b[1]
sol = list()
if le == 1 and su == 0:
	print('0 0')
elif 9 * le >= su and su > 0:
	for a in range(le):
		if su - 9 > 0:
			sol.append('9')
			su -= 9
		else:
			sol.append('{}'.format(su))
			su -= su
	su = b[1]
	sol2 = list()
	for a in range(le):
		sol2.append('{}'.format(su - 9 * (le - a - 1)) if 9 * (le - a - 1) < su else '0')
		if a == 0 and sol2[a] == '0':
			sol2[a] = '1'
		su -= int(sol2[a])
	print(''.join(sol2), ''.join(sol))
else:
	print('-1 -1')
