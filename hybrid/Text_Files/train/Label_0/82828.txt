from sys import stdin
from collections import deque
input = stdin.readline
T = int(input())
while T > 0:
	T -= 1
	(N, K) = [int(x) for x in input().split()]
	arr = sorted(list(map(int, input().split())))
	d = deque(arr)
	arr2 = list(reversed(sorted(list(map(int, input().split())))))
	res = 0
	for x in arr2:
		if x == 1:
			res += 2 * d[-1]
			d.pop()
	for x in arr2:
		if x == 1:
			continue
		res += d[0]
		res += d[-1]
		d.pop()
		for i in range(0, x - 1):
			d.popleft()
	print(res)
