from collections import deque
k = int(input())
s = input()
t = input()
alphabets = list('abcdefghijklmnopqrstuvwxyz')
to_digit = {}
for (i, c) in enumerate(alphabets):
	to_digit[c] = i
summation = deque()
carry = 0
for i in range(k - 1, -1, -1):
	dig = to_digit[s[i]] + to_digit[t[i]] + carry
	if dig > to_digit['z']:
		dig -= 26
		carry = 1
	else:
		carry = 0
	dig = chr(dig + ord('a'))
	summation.appendleft(dig)
l = k
r = 0
if carry == 1:
	r = 1
result = []
for i in range(l):
	term = to_digit[summation[i]]
	if r == 1:
		term += 26
	(q, r) = divmod(term, 2)
	q = chr(q + ord('a'))
	result.append(q)
print(''.join(result))
