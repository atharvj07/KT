#include<cstdio>
#include<algorithm>
#include<functional>
using namespace std;
typedef pair<int,char> P;
int main(void)
{
	int n,i,j,x;
	P pp[10];
	int kati[10],make[10],zyun[10];
	while(1)	{
		scanf("%d",&n);
		if(n==0)	break;
		for(i=0;i<n;i++)	{
			scanf(" %c",&pp[i].second);
			zyun[i]=n-i;
			kati[i]=0;
			make[i]=0;
			for(j=0;j<n-1;j++)	{
				scanf("%d",&x);
				if(x==0)	kati[i]++;
				if(x==1)	make[i]++;
			}
			pp[i].first=kati[i]*10000+(1000-make[i])*100+zyun[i];
		}
		sort(pp,pp+n,greater<P>() );
		for(i=0;i<n;i++)	{
			printf("%c\n",pp[i].second);
		}
	}
	return 0;
}
