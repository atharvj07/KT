#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long int b[n / 2];
  long long int arr[n];
  arr[0] = 0;
  int i;
  for (i = 0; i < n / 2; i++) cin >> b[i];
  arr[n - 1] = b[0];
  for (i = 1; i < n / 2; i++) {
    if (b[i] <= b[i - 1]) {
      arr[i] = arr[i - 1];
      arr[n - i - 1] = b[i] - arr[i];
    } else {
      arr[i] = b[i] - b[i - 1] + arr[i - 1];
      arr[n - i - 1] = b[i] - arr[i];
    }
  }
  for (i = 0; i < n; i++) cout << arr[i] << " ";
}
