#include <iostream>
#include <iomanip>
#include <vector>
#include <bitset>
#include <cstring>
using namespace std;
template<class T> bool chmax(T &a, T b) { if (a < b) { a = b; return true; } else return false; }

int n, stage;
vector<vector<double> > p;

double dp[(1<<16)+10][20]; // remained member, #. of remained stages
double rec(int bit, int rem) {
    if (dp[bit][rem] >= -1e-10) return dp[bit][rem];
    if (rem == 0) return 1.0;
    
    double res = 0.0;
    int start = stage - rem;
    for (int i = 0; i < n; ++i) {
        if (!(bit & (1<<i))) continue;
        int nbit = bit & ~(1<<i);
        double tmp = 0.0;
        double prob = 1.0;
        for (int lose = start; lose < stage; ++lose) {
            double loseprob = prob * (1.0 - p[i][lose]);
            tmp += loseprob * rec(nbit, stage - lose);
            prob *= p[i][lose];
        }
        tmp += prob * rec(nbit, 0);
        chmax(res, tmp);
    }
    return dp[bit][rem] = res;
}

int main() {
    while (cin >> n >> stage) {
        p.assign(n, vector<double>(stage, 0.0));
        for (int i = 0; i < n; ++i) for (int j = 0; j < stage; ++j) cin >> p[i][j];

        memset(dp, -1, sizeof(dp));
        cout << fixed << setprecision(10) << rec((1<<n)-1, stage) << endl;
    }
}
