import java.util.Scanner;
import java.util.TreeSet;

public class NewspaperHeadline {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		
		char[] A = sc.next().toCharArray();
		char[] B = sc.next().toCharArray();

		TreeSet<Integer>[] charIndexMap = new TreeSet[26];
		for(int i = 0; i < 26; i++){
			charIndexMap[i] = new TreeSet<Integer>();
		}
		
		// for each letter store the index of where is it found in the 1st string
		for(int i = 0, len = A.length; i < len; i++){
			charIndexMap[A[i] - 'a'].add(i);
		}
		
		// Now iterate over the second string, for each letter in the 2nd string
		// find the match in the 1st string which is on the right side of the last match
		// i.e. the index of this match should be greater then the index of the last match
		// TreeSet.higher(...) exactly does that.
		// If no match is found, then we need to concatenate 1st string again i.e. ans++
		// and reset the search. However, if we still cannot find a match then the letter
		// in the 2nd string does not exist in the 1st string i.e. ans = -1
		int ans = 1, lastMatchPos = -1, indexB = 0;
		while(indexB < B.length){
			Integer matchedIndex = charIndexMap[B[indexB] - 'a'].higher(lastMatchPos);
			
			if(matchedIndex == null){
				if(lastMatchPos == -1){
					System.out.println(-1);
					return;
				}
				ans++;
				lastMatchPos = -1;
			}else{
				lastMatchPos = matchedIndex;
				indexB++;
			}
		}
		System.out.println(ans);
	}
}
