n = int(input())
vals = input().split()
a_list = [int(val) for val in vals]
max_even = None
max_odd = None
for a in a_list:
	me = max_even
	mo = max_odd
	if a % 2 == 0:
		if mo is not None:
			max_odd = max(mo, mo + a)
		if me is not None:
			max_even = max(me, a, me + a)
		else:
			max_even = a
	elif me is None and mo is None:
		max_odd = a
	elif me is None:
		max_odd = max(mo, a)
		max_even = mo + a
	elif mo is None:
		max_odd = max(me + a, a)
	else:
		max_odd = max(mo, me + a, a)
		max_even = max(me, mo + a)
print(max_odd)
