#include <bits/stdc++.h>
using namespace std;
bool check(string second, long long l, long long r, long long k) {
  long long cnt = 0, mx = 0;
  for (long long i = l; i <= r; i++) {
    if (second[i] == 'N')
      cnt++;
    else {
      mx = max(mx, cnt);
      cnt = 0;
    }
  }
  mx = max(mx, cnt);
  if (mx > k) return 0;
  return 1;
}
bool good(string second, long long l1, long long r1, long long k) {
  long long l = l1, r = l1 + k - 1;
  while (r <= r1) {
    if ((l == l1 || second[l - 1] == '?') &&
        (r == r1 || second[r + 1] == '?')) {
      return 1;
    }
    l++;
    r++;
  }
  return 0;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long t;
  t = 1;
start:
  while (t--) {
    long long n, k;
    cin >> n >> k;
    string second;
    cin >> second;
    long long flag = 0, st;
    vector<pair<long long, long long>> range;
    for (long long i = 0; i < second.size(); i++) {
      if (second[i] != 'Y' && flag == 0) {
        st = i;
        flag = 1;
      }
      if (second[i] == 'Y' && flag == 1) {
        flag = 0;
        range.push_back({st, i - 1});
      }
    }
    if (flag == 1) {
      range.push_back({st, second.size() - 1});
    }
    flag = 0;
    if (range.empty() && k == 0) {
      cout << "YES" << endl;
      goto start;
    }
    for (auto i : range) {
      if (!check(second, i.first, i.second, k)) {
        cout << "NO" << endl;
        goto start;
      }
      if (good(second, i.first, i.second, k)) {
        flag = 1;
      }
    }
    if (flag == 1) {
      cout << "YES" << endl;
    } else
      cout << "NO" << endl;
  }
  return 0;
}
