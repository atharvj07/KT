from collections import deque
import sys
input = sys.stdin.buffer.readline
(n, m) = map(int, input().split())
balance = [0 for i in range(n)]
for _ in range(m):
	(u, v, d) = map(int, input().split())
	balance[u - 1] -= d
	balance[v - 1] += d
bgive = [[x, -balance[x]] for x in range(n) if balance[x] < 0]
btake = [[x, balance[x]] for x in range(n) if balance[x] > 0]
tdbt = 0
for x in balance:
	if x > 0:
		tdbt += x
	else:
		tdbt -= x
i = 0
j = 0
total = 0
printall = []
printall.append
while tdbt > 0:
	if bgive[i][1] > btake[j][1]:
		printall.append([bgive[i][0] + 1, btake[j][0] + 1, btake[j][1]])
		bgive[i][1] -= btake[j][1]
		tdbt -= 2 * btake[j][1]
		j += 1
	elif bgive[i][1] < btake[j][1]:
		printall.append([bgive[i][0] + 1, btake[j][0] + 1, bgive[i][1]])
		btake[j][1] -= bgive[i][1]
		tdbt -= 2 * bgive[i][1]
		i += 1
	else:
		printall.append([bgive[i][0] + 1, btake[j][0] + 1, btake[j][1]])
		tdbt -= 2 * btake[j][1]
		j += 1
		i += 1
	total += 1
print(total)
for x in printall:
	print(' '.join(map(str, x)))
