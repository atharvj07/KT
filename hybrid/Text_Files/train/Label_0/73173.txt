(s, t) = (input(), input())
if t in s:
	print('automaton')
else:
	(cnt1, cnt2) = ([0 for i in range(26)], [0 for i in range(26)])
	for i in s:
		cnt1[ord(i) - ord('a')] += 1
	for i in t:
		cnt2[ord(i) - ord('a')] += 1
	array = True
	for i in range(26):
		if cnt2[i] > cnt1[i]:
			array = False
			break
	if array == False:
		print('need tree')
	else:
		(i, j) = (0, 0)
		while i < len(s) and j < len(t):
			if s[i] == t[j]:
				i += 1
				j += 1
			else:
				i += 1
		if j == len(t):
			print('automaton')
		elif len(s) == len(t):
			print('array')
		else:
			print('both')
