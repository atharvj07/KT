import java.io.*;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class E {

    FastScanner scanner;
    PrintWriter writer;

    void solveTest() throws IOException {
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] adj = new int[n][n];
        int[] d = new int[n];
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt() - 1;
            int b = scanner.nextInt() - 1;
            adj[a][b] = 1;
            adj[b][a] = 1;
            d[a]++;
            d[b]++;
        }
        int even = 0;
        for (int i = 0; i < n; i++) {
            if (d[i] % 2 == 0) {
                even++;
            }
        }
        int[] marks = markComponents(adj);
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (d[i] % 2 != 0) {
                for (int j = i + 1; j < n; j++) {
                    if (marks[i] == marks[j] && d[j] % 2 != 0) {
                        if (adj[i][j] != 0) {
                            adj[i][j] = 0;
                            adj[j][i] = 0;
                            d[i]--;
                            d[j]--;
                            edges.add(new int[]{i, j});
                            break;
                        } else if (adj[i][j] == 0) {
                            adj[i][j] = 2;
                            adj[j][i] = 2;
                            d[i]++;
                            d[j]++;
                            break;
                        }
                    }
                }
            }
        }
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                buildEuler(adj, visited, edges, i);
            }
        }
        writer.println(even);
        for (int[] edge : edges) {
            writer.println((edge[0] + 1) + " " + (edge[1] + 1));
        }
    }

    void buildEuler(int[][] adj, boolean[] visited, List<int[]> edges, int v) {
        visited[v] = true;
        for (int i = 0; i < adj.length; i++) {
            if (adj[v][i] != 0) {
                if (adj[v][i] == 1) {
                    edges.add(new int[]{v, i});
                }
                adj[v][i] = 0;
                adj[i][v] = 0;
                buildEuler(adj, visited, edges, i);
            }
        }
    }

    int[] markComponents(int[][] adj) {
        int mark = 0;
        int[] marks = new int[adj.length];
        for (int i = 0; i < adj.length; i++) {
            if (marks[i] == 0) {
                mark++;
                marks[i] = mark;
                markComponent(adj, marks, i);
            }
        }
        return marks;
    }

    void markComponent(int[][] adj, int[] marks, int v) {
        for (int i = 0; i < adj.length; i++) {
            if (adj[v][i] != 0 && marks[i] != marks[v]) {
                marks[i] = marks[v];
                markComponent(adj, marks, i);
            }
        }
    }

    void solve() throws IOException {
        scanner = new FastScanner(System.in);
        writer = new PrintWriter(System.out);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            solveTest();
        }
        writer.close();
    }

    public static void main(String... args) throws IOException {
        new E().solve();
    }

    static class FastScanner {

        BufferedReader br;
        StringTokenizer tokenizer;

        FastScanner(String fileName) throws FileNotFoundException {
            this(new FileInputStream(new File(fileName)));
        }

        FastScanner(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        String nextLine() throws IOException {
            tokenizer = null;
            return br.readLine();
        }

        String next() throws IOException {
            if (tokenizer == null || !tokenizer.hasMoreTokens()) {
                String line = br.readLine();
                if (line == null) {
                    return null;
                }
                tokenizer = new StringTokenizer(line);
            }
            return tokenizer.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }
}
