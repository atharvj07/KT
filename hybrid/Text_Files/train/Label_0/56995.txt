import java.io.PrintWriter;
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = scan.nextInt();
		int p = scan.nextInt();
		int[] a = new int[n];
		int[] b = new int[n];
		long asum = 0, bsum = 0;
		for(int i=0; i<n; ++i) {
			a[i] = scan.nextInt();
			b[i] = scan.nextInt();
			asum += a[i];
			bsum += b[i];
		}
		
		if(asum <= p) {
			System.out.println(-1);
		}
		else {
			double lo = 0.0;
			double hi = (double)bsum / (asum - p);
			for(int iter = 0; iter < 100; ++iter) {
				double mid = (lo + hi) / 2;
				double need = 0.0;
				for(int i=0; i<n; ++i) {
					need += Math.max(a[i] - b[i] / mid, 0.0);
				}
				if(p >= need) 
					lo = mid;
				else
					hi = mid;
			}
			double mid = (lo + hi) / 2;
			System.out.println(mid);
		}
		
		out.close();
		scan.close();
	}
}
