#include <bits/stdc++.h>
using namespace std;
char data[15][15];
int sum, counts, ans;
int main() {
  int x, y;
  cin >> x >> y;
  int n, mini;
  cin >> n >> mini;
  while (n--) {
    int xx, yy;
    cin >> xx >> yy;
    data[xx][yy] = '1';
  }
  for (size_t i = 1; i <= x; i++) {
    for (size_t s = 1; s <= y; s++) {
      for (size_t j = i; j <= x; j++) {
        for (size_t k = s; k <= y; k++) {
          counts = 0;
          for (size_t q = i; q <= j; q++) {
            for (size_t w = s; w <= k; w++) {
              if (data[q][w] == '1') {
                counts++;
              }
            }
          }
          if (counts >= mini) {
            ans++;
          }
        }
      }
    }
  }
  cout << ans << "\n";
}
