import sys
import math as mt
import bisect
input = sys.stdin.buffer.readline
t = 1
for __ in range(t):
	n = int(input())
	xori = [0] * (n + 1)
	l = list(map(int, input().split()))
	for i in range(1, n + 1):
		xori[i] = xori[i - 1] ^ i
	ans = 0
	for i in range(n):
		if n // (i + 1) % 2 != 0:
			ans ^= l[i] ^ xori[i] ^ xori[n % (i + 1)]
		else:
			ans ^= l[i] ^ xori[n % (i + 1)]
	print(ans)
