(x3, y3) = [int(x3), int(y3)]

aby = y1 - y2
bcx = x2 - x3
bcy = y2 - y3

acy = y1 - y3

y = y3 - aby

x = x2 + acx

y = y2 - acy
ans.add((x, y))
x = x1 + bcx

y = y1 - bcy

print(len(ans))
