#include <bits/stdc++.h>
using namespace std;
int second[1000005 + 4];
long long int totient[1000005 + 4];
vector<long long int> prime;
vector<long long int> g[1000000];
void sieve() {
  int i, j;
  int k = sqrt(1000005);
  totient[1] = 1;
  for (i = 2; i <= 1000005; i++) totient[i] = i;
  for (i = 2; i <= 1000005; i++) {
    if (second[i] != -1) {
      prime.push_back(i);
      for (j = 1; i * j <= 1000005; j++) {
        totient[i * j] *= (i - 1);
        totient[i * j] /= i;
        second[i * j] = -1;
      }
    }
  }
}
int main() {
  ios::sync_with_stdio(false);
  cout.tie(0);
  cin.tie(0);
  long long int n, k;
  sieve();
  int i;
  cin >> n >> k;
  long long int b;
  long long int l;
  long long int a = n;
  long long int ans = 1;
  k = (k + 1) / 2;
  while (a > 1 && k > 0) {
    ans = 1;
    for (i = 0; i < prime.size(); i++) {
      l = 1;
      while (a % prime[i] == 0) {
        a /= prime[i];
        l *= prime[i];
      }
      if (l != 1) {
        l *= (prime[i] - 1);
        l /= prime[i];
      }
      ans *= l;
      if (a == 1) break;
    }
    k--;
    if (a != 1) a--;
    a *= ans;
  }
  cout << a % 1000000007;
}
