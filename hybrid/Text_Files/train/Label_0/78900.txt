def solution(l1):
	i = 0
	while i < len(l1):
		j = 0
		while j < len(l1):
			targ = l1[i][j]
			if targ != 1:
				l = 0
				while l < len(l1[i]):
					first = l1[i][l]
					n = 0
					while n < len(l1):
						second = l1[n][j]
						done = 0
						if second + first == targ:
							done = 1
							break
						n += 1
					if done:
						break
					else:
						pass
					l += 1
					if l == len(l1[j]):
						return 'NO'
			j += 1
		i += 1
	return 'YES'

def answer():
	n = int(input())
	l1 = []
	while n:
		l1.append([int(x) for x in input().split()])
		n -= 1
	print(solution(l1))
answer()
