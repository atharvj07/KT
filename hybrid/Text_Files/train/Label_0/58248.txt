 * Actual solution is at the top
 */
public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		InputReader in = new InputReader(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		TaskC solver = new TaskC();
		solver.solve(1, in, out);
		out.close();
	}
}

class TaskC {
    public void solve(int testNumber, InputReader in, PrintWriter out) {
        int n = in.nextInt();
        int m = in.nextInt();
        Integer []prev = new Integer[m];
        for (int i = 0; i < m; i++) {
            prev[i] = new Integer(in.nextInt());
        }
        long sum = 0;
        for (int i = 1; i < m; i++) {
            sum += Math.abs(prev[i] - prev[i - 1]);
        }
        long ans = sum;
        ArrayList<Integer>[] arr = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            arr[i] = new ArrayList<Integer>();
        }
        for (int i = 1; i < m; i++) {
            if (prev[i - 1].compareTo(prev[i]) != 0){
                arr[prev[i - 1]].add(prev[i]);
                arr[prev[i]].add(prev[i - 1]);
            }
        }
        for (int i = 1; i <= n; i++) {
            ArrayList<Integer> arrayList = new ArrayList<Integer>();
            arrayList.addAll(arr[i]);
            if (arrayList.size() == 0)continue;
            Collections.sort(arrayList);
            int mid = arrayList.size() / 2;
            int x = arrayList.get(mid);
            long temp = sum;
            for (int j = 0; j < arrayList.size(); j++) {
                temp += Math.abs(arrayList.get(j) - x) - Math.abs(arrayList.get(j) - i);
            }
            ans = Math.min(ans,temp);
        }
        out.print(ans);


    }
}

class InputReader {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream));
        tokenizer = null;
    }

    public String next() {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }
        return tokenizer.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(next());
    }


}
