t = int(input())

def calc(p1, q1, p2, q2):
	(h, w) = (p2 - p1 + 1, q2 - q1 + 1)
	if h <= 0 or w <= 0:
		return (0, 0)
	wh = h * w // 2
	bl = h * w // 2
	if h % 2 != 0 and w % 2 != 0:
		if p1 % 2 == q1 % 2:
			wh += 1
		else:
			bl += 1
	return (wh, bl)
for i in range(t):
	(H, W) = map(int, input().split())
	(x1, y1, x2, y2) = map(int, input().split())
	(x3, y3, x4, y4) = map(int, input().split())
	if H % 2 != 0 and W % 2 != 0:
		white = H * W // 2 + 1
		black = H * W // 2
	else:
		white = H * W // 2
		black = H * W // 2
	(w1, b1) = calc(x1, y1, x2, y2)
	white += b1
	black -= b1
	(w2, b2) = calc(x3, y3, x4, y4)
	white -= w2
	black += w2
	(p, q) = (max(x1, x3), max(y1, y3))
	(r, s) = (min(x2, x4), min(y2, y4))
	(w3, b3) = calc(p, q, r, s)
	white -= b3
	black += b3
	print(white, black)
