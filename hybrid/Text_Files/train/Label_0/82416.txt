from collections import deque
t = int(input())
for i in range(t):
	och = deque()
	n = int(input())
	d = list(map(int, input().split()))
	pos = 0
	ans = []
	for j in d:
		if j > pos:
			ans.append(j)
			for k in range(pos + 1, j):
				och.append(k)
		else:
			ans.append(och.popleft())
		pos = j
	print(*ans)
	pos = 0
	ans = []
	for j in d:
		if j > pos:
			ans.append(j)
			for k in range(pos + 1, j):
				och.append(k)
		else:
			ans.append(och.pop())
		pos = j
	print(*ans)
