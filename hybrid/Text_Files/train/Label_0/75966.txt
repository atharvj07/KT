(h, m) = map(int, input().split())
(a, d, cost, val) = map(int, input().split())
x = 20 * 60
tim = h * 60 + m
ans = (a + val - 1) // val * cost
for i in range(123456):
	rem = a - i * val + max(x - tim, 0) * d
	if rem > 0:
		ans = min(ans, i * cost + (rem + val - 1) // val * 0.8 * cost)
	else:
		ans = min(ans, i * cost)
print('%.10f' % ans)
