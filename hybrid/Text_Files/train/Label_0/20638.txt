#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;

#define NUM 500

int V,E;
int ans;
vector<int> G[NUM];
bool visited[NUM];

void recursive(int node_id,bool is_first){

	visited[node_id] = true;

	if(!is_first){
		/*最初の頂点以外は、出次数-1のエッジを残す
		 (再帰的に考えると、入次の情報は分っているので、残りの1本は引き算で計算できる)
		 */
		ans += (int)G[node_id].size()-1;
	}

	for(int i = 0; i < G[node_id].size(); i++){
		if(visited[G[node_id][i]])continue;
		recursive(G[node_id][i],false);
	}
}

int main(){

	scanf("%d %d",&V,&E);

	for(int i = 0; i < V; i++)G[i].clear();

	int from,to;
	for(int loop = 0; loop < E; loop++){
		scanf("%d %d",&from,&to);
		from--;
		to--;
		G[from].push_back(to);
	}

	for(int i = 0; i < V; i++)visited[i] = false;

	ans = 0;
	for(int i = 0; i < V; i++){
		if(visited[i])continue;
		ans += (int)G[i].size(); //最初の頂点は、すべての出次エッジを残す
		recursive(i,true);
	}

	printf("%d\n",ans);

	return 0;
}
