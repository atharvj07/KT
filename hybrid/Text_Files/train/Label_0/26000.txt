#include <bits/stdc++.h>
#pragma optimization_level 3
#pragma GCC optimize("Ofast,no-stack-protector,unroll-loops,fast-math,O3")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx")
using namespace std;
const long long N = 1e5 + 5;
const long long MAXN = 1e7 + 1;
const long long mod = 1e9 + 7;
const long long INF = 1e18;
const int INFi = 0x7f7f7f7f;
long long test = 1, n, k, a[N];
set<long long> second;
map<long long, long long> ct;
void solve() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i], ct[a[i]]++, second.insert(a[i]);
  sort(a, a + n);
  int st = 0;
  for (auto it : second) {
    st = it;
    if (ct[it] * n >= k)
      break;
    else
      k -= ct[it] * n;
  }
  for (auto it : second) {
    if (ct[st] * ct[it] >= k) {
      cout << st << " " << it << "\n";
      ;
      break;
    } else
      k -= ct[st] * ct[it];
  }
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cout << fixed;
  cout << setprecision(10);
  ;
  for (int i = 1; i <= test; i++) {
    solve();
  }
}
