#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, n, r, i;
  cin >> a >> b >> n;
  for (i = 0; i <= 9; i++) {
    r = (a * 10 + i) % b;
    if (r == 0) {
      cout << a << i;
      break;
    }
  }
  if (i == 10) {
    cout << "-1";
    return 0;
  }
  for (i = 0; i < n - 1; i++) {
    cout << "0";
  }
}
