#include<cstdio>
#include<utility>
#include<vector>
#include<algorithm>

using namespace std;

typedef pair<int,bool> P;
typedef vector<P> Closure;

bool isupper(char ch){
	return 'A'<=ch&&ch<='Z';
}
int getNum(char ch){
	if(isupper(ch)) return ch-'A';
	else return ch-'a';
}

Closure cs[8080];
int cnt;

bool cmp(const Closure &c1,const Closure &c2){
	if(c1.size()!=c2.size()) return c1.size()<c2.size();
	return c1<c2;
}

char ch[3][3030];
int M;

int which[26];//1:true,0:false,-1:unknown
bool suc;

void input(){
	scanf("%d",&M);
	if(M==0) exit(0);
	for(int i=0;i<3;i++) scanf("%s",ch[i]);
}

void init(){
	for(int i=0;i<26;i++){
		which[i]=-1;
	}
	suc=false;
	cnt=0;
}

void get(){
	for(int i=0;i<3*M;i+=3){
		for(int stat=0;stat<8;stat++){
			Closure c;
			for(int j=0;j<3;j++){
				int a=(stat>>j)&1;
				char ch=::ch[j][i+1+a];
				P p=P(getNum(ch),!isupper(ch));
				c.push_back(p);
			}
			sort(c.begin(),c.end());
			c.erase(unique(c.begin(),c.end()),c.end());
			bool flg=false;
			for(int i=0;i+1<c.size();i++){
				if(c[i].first==c[i+1].first){
					flg=true;
				}
			}
			if(flg) continue;
	//		if(c[0].first==c[1].first||c[1].first==c[2].first){
	//			continue;
	//		}else{
			cs[cnt++]=c;
	//		}
		}
	}
	sort(cs,cs+cnt,cmp);
	cnt=unique(cs,cs+cnt)-cs;
}

void dfs(int id){
	if(cnt==id){
		suc=true;
		return;
	}
	Closure c=cs[id];
	bool changed[3]={0,0,0};
	for(int i=0;i<c.size();i++){
		int x=c[i].first;
		bool t=c[i].second;
		int num=t?1:0;
		if(which[x]!=-1&&which[x]!=t){
			continue;
		}
		if(which[x]==t){
			dfs(id+1);
			if(suc) return;
			else{
				for(int j=0;j<3;j++){
					if(changed[j]){
						which[c[j].first]=-1;
					}
				}
				return;
			}
		}
		changed[i]=true;
		which[x]=t;
		dfs(id+1);
		if(suc) return;
		which[x]=t^1;
	}
	for(int i=0;i<c.size();i++){
		if(changed[i]){
			which[c[i].first]=-1;
		}
	}
}

void output(){
	if(suc==false){
		printf("-1\n");
	}else{
		int num=0;
		for(int i=0;i<26;i++){
			if(which[i]==1) num++;
		}
		printf("%d",num);
		for(int i=0;i<26;i++){
			if(which[i]==1) printf(" %c",'A'+i);
		}
		printf("\n");
	}
}

void solve(){
	dfs(0);
	output();
}

int main(){
	while(true){
		input();
		init();
		get();
		solve();
	/*	for(int i=0;i<cnt;i++){

			for(int j=0;j<cs[i].size();j++){

				printf(" ");
			}
			printf(")");
			printf("\n");
		}
		printf("\n");*/
	}
	return 0;
}
