#include <bits/stdc++.h>
using namespace std;
const long long N = 2100;
long long n, k, l[N], r[N], a[N], dp[N];
inline long long read() {
  long long f = 1, x = 0;
  char s = getchar();
  while (s < '0' || s > '9') {
    if (s == '-') f = -1;
    s = getchar();
  }
  while (s >= '0' && s <= '9') {
    x = x * 10 + s - '0';
    s = getchar();
  }
  return x * f;
}
signed main() {
  n = read();
  k = read();
  for (long long i = 1; i <= n; i++)
    l[i] = read(), r[i] = read(), a[i] = read();
  bool bl = 1;
  long long now = k;
  for (long long i = 1; i <= n; i++) {
    if (l[i] > r[i - 1]) now = k;
    now = min(k, now + (r[i] - l[i]) * k - a[i]);
    if (now < 0) bl = 0;
  }
  if (!bl) {
    printf("-1\n");
    return 0;
  }
  for (long long i = 1; i <= n; i++) dp[i] = 1e18;
  dp[0] = 0;
  for (long long i = 0; i < n; i++) {
    long long now = k, cnt = k;
    for (long long j = i + 1; j <= n; j++) {
      if (now + (r[j] - l[j]) * k < a[j]) break;
      long long all = max(0ll, a[j] - now);
      cnt += ((all + k - 1) / k) * k;
      now = ((all + k - 1) / k) * k + now - a[j];
      if (j == n) {
        dp[j] = min(dp[j], dp[i] + cnt - now);
        continue;
      }
      if ((all + k - 1) / k < r[j] - l[j] || l[j + 1] > r[j])
        dp[j] = min(dp[j], dp[i] + cnt);
    }
  }
  printf("%lld\n", dp[n]);
}
