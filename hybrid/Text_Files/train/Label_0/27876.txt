#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100000;
vector<int> way[MAXN], cost[MAXN];
int w[MAXN], s[MAXN];
long long tot[MAXN];
bool used[MAXN];
bool cmp(int a, int b) {
  return (long long)w[a] * s[b] < (long long)w[b] * s[a];
}
void dfs(int u) {
  used[u] = true;
  vector<int> l;
  tot[u] = w[u] = 0;
  s[u] = 1;
  for (size_t i = 0; i < way[u].size(); ++i) {
    int v = way[u][i];
    if (used[v]) continue;
    dfs(v);
    tot[u] += tot[v] += cost[u][i] * s[v];
    w[u] += w[v] += 2 * cost[u][i];
    s[u] += s[v];
    l.push_back(v);
  }
  sort(l.begin(), l.end(), cmp);
  int size = 1;
  for (vector<int>::iterator itr = l.begin(); itr != l.end(); ++itr) {
    size += s[*itr];
    tot[u] += (long long)w[*itr] * (s[u] - size);
  }
}
int main() {
  int n, x, y, c;
  scanf("%d", &n);
  for (int i = 1; i < n; ++i) {
    scanf("%d%d%d", &x, &y, &c);
    --x, --y;
    way[x].push_back(y);
    way[y].push_back(x);
    cost[x].push_back(c);
    cost[y].push_back(c);
  }
  memset(used, false, sizeof(used));
  dfs(0);
  printf("%lf\n", (double)tot[0] / (n - 1));
  return 0;
}
