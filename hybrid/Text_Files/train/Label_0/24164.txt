#include <bits/stdc++.h>
using namespace std;
const long long kMod = 998244353;
int32_t main() {
  long long n, k;
  cin >> n >> k;
  vector<long long> h(n);
  for (long long i = 0; i < n; ++i) {
    cin >> h[i];
  }
  if (k == 1) {
    cout << 0 << endl;
    return 0;
  }
  auto lgpow = [&](long long b, long long e) {
    long long r = 1;
    while (e) {
      if (e % 2) r = r * b % kMod;
      b = b * b % kMod;
      e /= 2;
    }
    return r;
  };
  vector<long long> fact(n + 1, 1), ifact(n + 1, 1);
  for (long long i = 1; i <= n; ++i) {
    fact[i] = fact[i - 1] * i % kMod;
  }
  ifact[n] = lgpow(fact[n], kMod - 2);
  for (long long i = n - 1; i >= 0; --i) {
    ifact[i] = ifact[i + 1] * (i + 1) % kMod;
    assert(ifact[i] * fact[i] % kMod == 1);
  }
  auto choose = [&](long long n, long long k) {
    if (k < 0 || k > n) return 0LL;
    return fact[n] * ifact[k] % kMod * ifact[n - k] % kMod;
  };
  long long to_mul = 1;
  long long unm = 0;
  for (long long j = n - 1, i = 0; i < n; j = i++) {
    if (h[i] == h[j]) {
      to_mul = to_mul * k % kMod;
    } else {
      unm += 1;
    }
  }
  long long ans = 0;
  for (long long i = 0; i <= unm / 2; ++i) {
    long long now = choose(unm, i) * choose(unm - i, i) % kMod;
    now = now * lgpow(k - 2, unm - 2 * i) % kMod;
    ans += now;
  }
  ans = ans % kMod * to_mul % kMod;
  long long all = lgpow(k, n);
  ans = (all + kMod - ans) % kMod;
  ans = ans * ((kMod + 1) / 2) % kMod;
  cout << ans << endl;
  return 0;
}
