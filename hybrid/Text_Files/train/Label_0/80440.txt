def arr_inp():
	return [int(x) for x in input().split()]

def diff(x, y):
	arr.append(y - x)
	return y
from collections import *
from functools import *
(n, a, mem, out) = (int(input()), arr_inp(), defaultdict(list), [])
for i in range(n):
	if mem[a[i]] != -1:
		if mem[a[i]] == []:
			mem[a[i]].extend([i, 0])
		elif mem[a[i]][1] == 0:
			mem[a[i]][1] = i - mem[a[i]][0]
			mem[a[i]][0] = i
		elif i - mem[a[i]][0] == mem[a[i]][1]:
			mem[a[i]][0] = i
		else:
			mem[a[i]] = -1
for (i, j) in mem.items():
	if j != -1:
		out.append([i, j[1]])
print(len(out))
for (i, j) in sorted(out):
	print(i, j)
