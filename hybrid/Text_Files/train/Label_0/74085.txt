def main():
	n = int(input())
	board = []
	for i in range(n):
		row = list(input())
		board.append(row)
	for i in range(n - 2):
		for j in range(1, n - 1):
			if board[i][j] == '.' and board[i + 1][j] == '.' and (board[i + 2][j] == '.') and (board[i + 1][j + 1] == '.') and (board[i + 1][j - 1] == '.'):
				board[i][j] = '#'
				board[i + 1][j] = '#'
				board[i + 2][j] = '#'
				board[i + 1][j + 1] = '#'
				board[i + 1][j - 1] = '#'
	for i in range(n):
		for j in range(n):
			if board[i][j] == '.':
				print('NO')
				return
	print('YES')
main()
