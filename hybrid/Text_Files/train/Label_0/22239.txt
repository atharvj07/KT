#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, x, temp, count = 0, flag;
  cin >> n >> x;
  if (n >= x) {
    temp = x;
    flag = 0;
  } else {
    temp = n;
    flag = 1;
  }
  while (temp > flag) {
    if ((x % temp) == 0 && (temp * n) >= x) {
      count++;
    }
    temp--;
  }
  cout << count << endl;
  return 0;
}
