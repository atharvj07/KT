a = input()
b = input()

def half_str(s):
	return (s[:len(s) // 2], s[len(s) // 2:])

def compare(a, b):
	if a == b:
		return True
	elif len(a) % 2 == 1:
		return False
	else:
		(a1, a2) = half_str(a)
		(b1, b2) = half_str(b)
		return compare(a1, b2) and compare(a2, b1) or (compare(a1, b1) and compare(a2, b2))

def solve(a, b):
	equivalent = compare(a, b)
	if equivalent:
		print('YES')
	else:
		print('NO')
solve(a, b)
