#include <bits/stdc++.h>
using namespace std;
int z = 0, ans, n;
priority_queue<int> q;
int main(void) {
  cin >> n;
  for (int i = 0; i < n * 2 - 1; i++) {
    int t;
    cin >> t;
    z -= t;
    q.push(-t);
  }
  ans = z;
  for (int i = 0; i < 100; i++) {
    for (int j = 0; j < n; j++) {
      int t = q.top();
      q.pop();
      z -= 2 * t;
      q.push(-t);
    }
    ans = min(ans, z);
  }
  cout << -ans << endl;
}
