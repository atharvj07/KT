#include <bits/stdc++.h>
using namespace std;
const int MAX = 210000;
int a[MAX], b[MAX], l[MAX], r[MAX], pre[MAX], succ[MAX], s[MAX], t[MAX], p[MAX];
int cmp(int *s, int p, int i) {
  return (!pre[p] || s[i - p + pre[p]] < s[i]) &&
         (!succ[p] || s[i + succ[p] - p] > s[i]);
}
int main() {
  int N, M, i;
  cin >> N >> M;
  for (i = 1; i <= N; i++) {
    cin >> a[i];
    s[a[i]] = i;
    l[i] = i - 1;
    r[i] = i + 1;
  }
  r[N] = 0;
  for (i = 1; i <= M; i++) {
    cin >> b[i];
    t[b[i]] = i;
  }
  int x;
  for (i = N; i; i--) {
    x = s[i];
    pre[i] = a[l[x]];
    succ[i] = a[r[x]];
    r[l[x]] = r[x];
    l[r[x]] = l[x];
  }
  x = 0;
  for (i = 2; i <= N; i++) {
    while (x && !cmp(s, x + 1, i)) x = p[x];
    p[i] = x += cmp(s, x + 1, i);
  }
  int ans = x = 0;
  for (i = 1; i <= M; i++) {
    while (x && !cmp(t, x + 1, i)) x = p[x];
    x += cmp(t, x + 1, i);
    if (x == N) {
      ++ans;
      x = p[x];
    }
  }
  cout << ans << endl;
}
