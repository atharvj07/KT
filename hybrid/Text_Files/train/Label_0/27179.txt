#include <bits/stdc++.h>
const int inf = 0x7fffffff;
int n, m;
int a[500010] = {0}, mx = -1, nxt[500010] = {0};
int p[500010] = {0};
struct Segtree {
  int l, r;
  int mxi;
} t[2000010];
inline int lson(int x) { return x << 1; }
inline int rson(int x) { return x << 1 | 1; }
void maketree(int x, int l, int r) {
  if (l == r) {
    t[x] = {l, r, 0};
    return;
  }
  t[x].l = l;
  t[x].r = r;
  t[x].mxi = 0;
  int mid = (l + r) >> 1;
  maketree(lson(x), l, mid);
  maketree(rson(x), mid + 1, r);
}
void update(int x, int pos) {
  if (pos > t[x].r || pos < t[x].l) return;
  if (t[x].l == t[x].r) {
    t[x].mxi = pos;
    return;
  }
  update(lson(x), pos);
  update(rson(x), pos);
  if (nxt[t[lson(x)].mxi] > nxt[t[rson(x)].mxi])
    t[x].mxi = t[lson(x)].mxi;
  else
    t[x].mxi = t[rson(x)].mxi;
}
int que(int x, int l, int r) {
  if (l > t[x].r || r < t[x].l) return 0;
  if (l <= t[x].l && t[x].r <= r) return t[x].mxi;
  int templ = que(lson(x), l, r), tempr = que(rson(x), l, r);
  if (nxt[templ] > nxt[tempr])
    return templ;
  else
    return tempr;
}
struct Ask {
  int l, r, id, ans;
} ask[500010];
bool cmp1(const Ask& a, const Ask& b) {
  return a.l == b.l ? a.r < b.r : a.l < b.l;
}
bool cmp2(const Ask& a, const Ask& b) { return a.id < b.id; }
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", a + i), mx = std::max(a[i], mx);
  for (int i = n; i > 0; i--) {
    if (p[a[i]])
      nxt[i] = p[a[i]];
    else
      nxt[i] = inf;
    p[a[i]] = i;
  }
  maketree(1, 1, n);
  for (int i = 1; i <= mx; i++) {
    if (p[i]) update(1, p[i]);
  }
  scanf("%d", &m);
  for (int i = 1, l, r; i <= m; i++) {
    scanf("%d%d", &l, &r);
    ask[i] = {l, r, i, 0};
  }
  std::sort(ask + 1, ask + 1 + m, cmp1);
  for (int i = 1, ll = 0; i <= m; i++) {
    while (ll < ask[i].l) {
      if (nxt[ll]) {
        update(1, nxt[ll]);
        nxt[ll] = 0;
        update(1, ll);
      }
      ll++;
    }
    int temp = que(1, ask[i].l, ask[i].r);
    if (nxt[temp] <= ask[i].r)
      ask[i].ans = 0;
    else
      ask[i].ans = a[temp];
  }
  std::sort(ask + 1, ask + 1 + m, cmp2);
  for (int i = 1; i <= m; i++) printf("%d\n", ask[i].ans);
  return 0;
}
