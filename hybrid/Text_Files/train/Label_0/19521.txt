#include <bits/stdc++.h>
using namespace std;
int dir;
int n, m, s, f;
int tcurr = 1;
string ret;
int getNext() {
  if (dir == 0)
    return s + 1;
  else
    return s - 1;
}
char getNextMove() {
  if (dir == 0)
    return 'R';
  else
    return 'L';
}
int main() {
  cin >> n >> m >> s >> f;
  ret = "";
  if (s <= f)
    dir = 0;
  else
    dir = 1;
  for (int i = 0; i < m; ++i) {
    int t, a, b;
    cin >> t >> a >> b;
    while (tcurr < t && s != f) {
      s = getNext();
      ret += getNextMove();
      ++tcurr;
    }
    if (s == f) break;
    if (((a <= s && s <= b)) || (a <= getNext() && getNext() <= b))
      ret += 'X';
    else {
      s = getNext();
      ret += getNextMove();
    }
    ++tcurr;
    if (s == f) break;
  }
  while (s != f) {
    s = getNext();
    ret += getNextMove();
  }
  cout << ret;
  return 0;
}
