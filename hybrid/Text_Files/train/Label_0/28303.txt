#include <bits/stdc++.h>
using namespace std;
const long long int mod = 1e9 + 7;
long long int power(long long int a, long long int b) {
  long long int res = 1;
  a = a % 1000000007;
  while (b > 0) {
    if (b & 1) {
      res = (res * a) % 1000000007;
      b--;
    }
    a = (a * a) % 1000000007;
    b >>= 1;
  }
  return res;
}
long long int gcd(long long int a, long long int b) {
  return (b == 0) ? a : gcd(b, a % b);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int n, mm, x, y;
  cin >> n >> mm >> x >> y;
  vector<long long int> a(mm), b(mm), c(mm), d(mm);
  for (long long int i = 0; i < mm; i++) cin >> a[i] >> b[i] >> c[i] >> d[i];
  long long int dp[mm][n + 1];
  for (long long int i = 0; i < mm; i++) {
    for (long long int j = 0; j <= n; j++) dp[i][j] = 0;
  }
  for (long long int j = 0; j <= n; j++) {
    long long int ii = 0;
    for (long long int l = 0;; l++) {
      if (b[ii] * l > a[ii]) break;
      if (c[ii] * l > j) break;
      long long int cost = 0;
      long long int prof = 0;
      cost += c[ii] * l;
      prof += d[ii] * l;
      dp[ii][j] = max(dp[ii][j], prof);
    }
  }
  for (long long int i = 1; i < mm; i++) {
    for (long long int j = 0; j <= n; j++) {
      for (long long int l = 0;; l++) {
        if (b[i] * l > a[i]) break;
        if (c[i] * l > j) break;
        long long int cost = 0;
        long long int prof = 0;
        cost += c[i] * l;
        prof += d[i] * l;
        dp[i][j] = max(dp[i][j], dp[i - 1][j - cost] + prof);
      }
    }
  }
  long long int ans = dp[mm - 1][n];
  for (long long int i = 0; i < n; i++) {
    long long int lef = (n - i);
    long long int prof = (lef / x) * y;
    ans = max(ans, dp[mm - 1][i] + prof);
  }
  cout << ans << "\n";
}
