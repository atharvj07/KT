t = int(input())
ta = 0
tg = 0
order = []
for _ in range(t):
	(a, g) = map(int, input().split())
	if ta > tg:
		if ta + a - tg <= 500:
			ta += a
			order.append('A')
		else:
			tg += g
			order.append('G')
	elif ta == tg:
		if a >= g:
			tg += g
			order.append('G')
		else:
			ta += a
			order.append('A')
	elif tg + g - ta <= 500:
		tg += g
		order.append('G')
	else:
		ta += a
		order.append('A')
if abs(ta - tg) <= 500:
	print(''.join(order))
else:
	print(-1)
