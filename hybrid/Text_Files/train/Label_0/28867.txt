#include <bits/stdc++.h>
using namespace std;
int main() {
  unsigned n, m;
  cin >> n >> m;
  vector<unsigned> a(n + 1, 0);
  vector<unsigned> b(n + 1, 0);
  for (auto i = 0; i < n; ++i) {
    unsigned aa, bb;
    cin >> aa >> bb;
    a[i] = aa;
    b[i] = bb;
  }
  if (a[0] != 0) {
    cout << "NO" << endl;
    return 0;
  }
  unsigned maxb = 0;
  for (auto i = 0; i < n; ++i) {
    if (b[i] >= maxb) {
      maxb = b[i];
    }
    if (maxb >= m) {
      cout << "YES" << endl;
      return 0;
    }
    if (i < n - 1 && maxb >= a[i + 1]) {
      continue;
    }
    cout << "NO" << endl;
    break;
  }
}
