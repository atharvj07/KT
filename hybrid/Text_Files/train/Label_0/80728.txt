import sys
n = int(input())
(min_a, max_a) = ([], [])
asc = set()
for (i, a) in enumerate((list(map(int, l.split())) for l in sys.stdin)):
	if any((a[j] < a[j + 1] for j in range(1, a[0]))):
		asc.add(i)
	else:
		min_a.append(min(a[1:]))
		max_a.append(max(a[1:]))
min_a.sort()
max_a.sort()
max_a.append(10 ** 18)
ans = 0
j = 0
m = len(min_a)
for i in range(m):
	while min_a[i] >= max_a[j]:
		j += 1
	ans += m - j + len(asc)
print(ans + n * len(asc))
