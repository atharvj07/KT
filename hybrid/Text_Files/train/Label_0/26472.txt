#include <bits/stdc++.h>
using namespace std;
bool cmp(int a, int b) { return a > b; }
int main() {
  int n, a, b;
  cin >> n >> a >> b;
  if (b >= 0) {
    for (int i = 0; i < b; i++) {
      a += 1;
      if (a > n) a = 1;
    }
  } else {
    for (int i = 0; i < -b; i++) {
      a -= 1;
      if (a < 1) a = n;
    }
  }
  cout << a;
  return 0;
}
