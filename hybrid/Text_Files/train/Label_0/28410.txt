#include <bits/stdc++.h>
using namespace std;
long long solve() {
  long long n, s, t;
  cin >> n >> s >> t;
  vector<long long int> v(n + 1);
  for (long long int i = 1; i < n + 1; i++) cin >> v[i];
  if (s == t) {
    cout << 0;
    return 0;
  }
  long long i = 0, j = s;
  while (i != n + 1 && j != t) {
    j = v[j];
    i++;
  }
  if (i == n + 1)
    cout << -1;
  else
    cout << i;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int TESTS = 1;
  while (TESTS--) {
    solve();
  }
  return 0;
}
