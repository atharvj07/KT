    static void dfs(int[][] tree, int u, boolean [] val) {

        switch(tree[u][0]) {
            case 0:
                dfs(tree, tree[u][1], val);
                dfs(tree, tree[u][2], val);
                val[u] = val[tree[u][1]] & val[tree[u][2]];
                break;
            case 1:
                dfs(tree, tree[u][1], val);
                dfs(tree, tree[u][2], val);
                val[u] = val[tree[u][1]] | val[tree[u][2]];
                break;
            case 2:
                dfs(tree, tree[u][1], val);
                dfs(tree, tree[u][2], val);
                val[u] = val[tree[u][1]] ^ val[tree[u][2]];
                break;
            case 3:
                dfs(tree, tree[u][1], val);
                val[u] = !val[tree[u][1]];
                break;
            default:
                break;
        }
    }

    static void solve()
    {
        in = new InputReader(System.in);
        out = new PrintWriter(System.out);            
        
        int n = in.nextInt();
        int[][] tree = new int[n][3];
        boolean[] val = new boolean[n];
        ArrayList<Integer> list = new ArrayList<>(n);
        
        for(int i = 0; i < n; i++) {
            char c = in.readString().charAt(0);
            switch (c) {
                case 'A':
                    tree[i][0] = 0;
                    tree[i][1] = in.nextInt() - 1;
                    tree[i][2] = in.nextInt() - 1;
                    break;
                case 'O':
                    tree[i][0] = 1;
                    tree[i][1] = in.nextInt() - 1;
                    tree[i][2] = in.nextInt() - 1;
                    break;
                case 'X':
                    tree[i][0] = 2;
                    tree[i][1] = in.nextInt() - 1;
                    tree[i][2] = in.nextInt() - 1;
                    break;
                case 'N':
                    tree[i][0] = 3;
                    tree[i][1] = in.nextInt() - 1;
                    break;
                case 'I':
                    tree[i][0] = 4;
                    val[i] = in.nextInt() == 1;
                    list.add(i);
                default:
                    break;
            }
        }
        
        dfs(tree, 0, val);
        boolean[][] ans = new boolean[n][2];
        ans[0][0] = false;
        ans[0][1] = true;
        dfs1(tree, 0, val, ans);
        for(int i = 0; i < n; i++) {
            if(tree[i][0] == 4) {
                out.print((val[i] ? ans[i][0] : ans[i][1]) ? 1 : 0);
            }
        }
        out.println();
        out.close();
    }

    static long pow(long n,long p,long m)
    {
         long  result = 1;
          if(p==0){
            return n;
          }
          
        while(p!=0)
        {
            if(p%2==1)
                result *= n;
            if(result >= m)
               result %= m;
            p >>=1;
            n*=n;
            if(n >= m)
                n%=m;
        }
        
        return result;
    }

    static class InputReader
    {
 
        private final InputStream stream;
        private final byte[] buf = new byte[8192];
        private int curChar, snumChars;
        private SpaceCharFilter filter;
 
        public InputReader(InputStream stream)
        {
                this.stream = stream;
        }
 
        public int snext()
        {
                if (snumChars == -1)
                        throw new InputMismatchException();
                if (curChar >= snumChars)
                {
                        curChar = 0;
                        try
                        {
                                snumChars = stream.read(buf);
                        } catch (IOException e)
                        {
                                throw new InputMismatchException();
                        }
                        if (snumChars <= 0)
                                return -1;
                }
                return buf[curChar++];
        }
 
        public int nextInt()
        {
                int c = snext();
                while (isSpaceChar(c))
                {
                        c = snext();
                }
                int sgn = 1;
                if (c == '-')
                {
                        sgn = -1;
                        c = snext();
                }
                int res = 0;
                do
                {
                        if (c < '0' || c > '9')
                                throw new InputMismatchException();
                        res *= 10;
                        res += c - '0';
                        c = snext();
                } while (!isSpaceChar(c));
                return res * sgn;
        }
 
        public long nextLong()
        {
                int c = snext();
                while (isSpaceChar(c))
                {
                        c = snext();
                }
                int sgn = 1;
                if (c == '-')
                {
                        sgn = -1;
                        c = snext();
                }
                long res = 0;
                do
                {
                        if (c < '0' || c > '9')
                                throw new InputMismatchException();
                        res *= 10;
                        res += c - '0';
                        c = snext();
                } while (!isSpaceChar(c));
                return res * sgn;
        }
 
        public int[] nextIntArray(int n)
        {
                int a[] = new int[n];
                for (int i = 0; i < n; i++)
                {
                        a[i] = nextInt();
                }
                return a;
        }
 
        public long[] nextLongArray(int n)
        {
                long a[] = new long[n];
                for (int i = 0; i < n; i++)
                {
                        a[i] = nextLong();
                }
                return a;
        }
 
        public String readString()
        {
                int c = snext();
                while (isSpaceChar(c))
                {
                        c = snext();
                }
                StringBuilder res = new StringBuilder();
                do
                {
                        res.appendCodePoint(c);
                        c = snext();
                } while (!isSpaceChar(c));
                return res.toString();
        }
 
        public String nextLine()
        {
                int c = snext();
                while (isSpaceChar(c))
                        c = snext();
                StringBuilder res = new StringBuilder();
                do
                {
                        res.appendCodePoint(c);
                        c = snext();
                } while (!isEndOfLine(c));
                return res.toString();
        }
 
        public boolean isSpaceChar(int c)
        {
                if (filter != null)
                        return filter.isSpaceChar(c);
                return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }
 
        private boolean isEndOfLine(int c)
        {
                return c == '\n' || c == '\r' || c == -1;
        }
 
        public interface SpaceCharFilter
        {
                public boolean isSpaceChar(int ch);
        }
 
    }
