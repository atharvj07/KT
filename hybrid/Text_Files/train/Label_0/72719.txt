(n, a, b) = map(int, input().split())
(l, c, d, sum) = (list(map(int, input().split())), 0, 0, 0)
if a > b:
	(a, b) = (b, a)
l.sort()
for i in range(n - 1, -1, -1):
	sum += l[i]
	c += 1
	if c == a:
		break
(c, sum) = (sum / a, 0)
for i in range(n - a - 1, -1, -1):
	sum += l[i]
	d += 1
	if d == b:
		break
print(sum / b + c)
