from math import e

def solve():
	(n, h) = map(int, input().split())
	arr = list(map(int, input().split()))
	if n > h:
		print('1')
	else:
		start = 1
		end = h
		mid = 0
		List = {}
		for i in range(n - 1):
			List[i] = arr[i + 1] - arr[i]
		while start <= end:
			mid = (start + end) // 2
			summ = mid
			for i in range(n - 1):
				if mid > List.get(i):
					summ += List.get(i)
				else:
					summ += mid
			if summ < h:
				start = mid + 1
			else:
				end = mid - 1
		print(end + 1)
cases = int(input())
for i in range(cases):
	solve()
