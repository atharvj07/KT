#include <bits/stdc++.h>
using namespace std;
int main() {
  int b[3] = {};
  int k;
  cin >> b[0] >> b[1] >> b[2] >> k;
  long long ans = 1;
  int g[5] = {};
  int a[3] = {1, 1, 1};
  while (k > 0) {
    int cnt = 0;
    for (int i = 0; i < 3; i++) {
      if (a[i] < b[i]) {
        g[cnt++] = i;
      }
    }
    if (cnt == 0) break;
    int min = a[g[0]];
    int chosen = g[0];
    for (int i = 1; i < cnt; i++) {
      if (min > a[g[i]]) {
        min = a[g[i]];
        chosen = g[i];
      }
    }
    a[chosen]++;
    k--;
  }
  ans = ans * a[0] * a[1] * a[2];
  cout << ans << endl;
  return 0;
}
