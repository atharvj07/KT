import java.util.*;
public class main 
{   
    public static void main(String args[]) 
    { 
       Scanner sc=new Scanner(System.in);
       int n,k;
       double r,g,b;
       n=sc.nextInt();
       k=sc.nextInt();
       r=1.0*n*2/k;
       g=1.0*n*5/k;
       b=1.0*n*8/k;
       System.out.print((int)(Math.ceil(r)+Math.ceil(g)+Math.ceil(b)));
    }
}
