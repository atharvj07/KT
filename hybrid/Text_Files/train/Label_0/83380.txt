queries = int(input())
result = []
for query in range(queries):
	boards = input()
	lengths = list(map(int, input().split()))
	if boards == 2:
		result.append(0)
		continue
	lengths.sort(reverse=True)
	if lengths[1] <= len(lengths) - 2:
		result.append(lengths[1] - 1)
	else:
		result.append(len(lengths) - 2)
for num in result:
	print(num)
