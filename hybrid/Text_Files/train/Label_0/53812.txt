import java.util.*;
public class MyClass {
    public static void main(String args[]) {
      Scanner sc=new Scanner(System.in);
  int n=sc.nextInt();
  int arr[]=new int[n];
  int i,p=0,h=0,k;
  for(i=0;i<n;i++)
  {
      arr[i]=sc.nextInt();
      if(arr[i]>0)
      ++p;
      else if(arr[i]<0)
      ++h;
  }
  if(n%2==0)
  k=n/2;
  else
  k=(n/2) +1 ;
  if(p>=k)
  System.out.println("4");
  else if(h>=k)
  System.out.println("-4");
  else
  System.out.println("0");
    }}
