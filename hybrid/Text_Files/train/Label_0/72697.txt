T = int(input())
for _ in range(T):
	(a, b, c) = map(int, input().split(' '))
	ans = []
	for i in range(0, b, 2):
		ans.append(0)
		ans.append(1)
	if len(ans) == 0 and c > 0:
		ans.append(1)
	for i in range(c):
		ans.append(1)
	if b % 2 == 0 and b > 0:
		ans.append(0)
	if len(ans) == 0 and a > 0:
		ans.append(0)
	if len(ans) > 0 and ans[-1] == 0:
		for i in range(a):
			ans.append(0)
	else:
		for i in range(a):
			ans.insert(0, 0)
	for i in ans:
		print(i, end='')
	print('')
