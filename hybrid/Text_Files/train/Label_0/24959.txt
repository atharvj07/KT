#define _CRT_SECURE_NO_WARNINGS
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define forn(i,x,n) for(int i = x;i <= n;++i)	

const int N = 2e5+7,M = 2 * N;
int edge[M],succ[M],ver[N],idx;
int deg[N],siz[N],st[N];
void add(int u,int v)
{
	edge[idx] = v;
	succ[idx] = ver[u];
	ver[u] = idx++;
}

void dfs(int u)
{
	siz[u] = 1;
	for(int i = ver[u];~i;i = succ[i])
	{
		int v = edge[i];
		if(st[v])	continue;
		st[v] = 1;
		dfs(v);
		siz[u] += siz[v];
	}
}

int main()
{
    int T;scanf("%d",&T);
    while(T--)
    {	
    	memset(ver,-1,sizeof ver);idx = 0;
		int n;scanf("%d",&n);
		forn(i,1,n)
		{
			int u,v;scanf("%d%d",&u,&v);
			add(u,v);add(v,u);
			++deg[u];++deg[v];
		}
		
		queue<int> q;
		forn(i,1,n)	if(deg[i] == 1)	q.push(i);
		while(!q.empty())
		{
			int u = q.front();q.pop();
			for(int i = ver[u];~i;i = succ[i])
			{
				int v = edge[i];
				if(--deg[v] == 1)	q.push(v);
			}
		}
		
		int c = 0;
		forn(i,1,n)	if(deg[i] == 2)	st[i] = 2,++c;
		
		ll res = 0;
		forn(i,1,n)
		{
			if(st[i] == 2)
			{
				dfs(i);
				res += 1ll * siz[i] * (siz[i] - 1) / 2;
				res += (1ll * siz[i] * (n - siz[i]));
				// for(int j = ver[i];~j;j = succ[j])
				// {
					// int v = edge[j];
					// if(st[v] == 2)	continue;
					// res += 1ll * siz[v] * (siz[v] - 1) / 2;
					// res += (1ll*siz[v] * (n - siz[v])) * 2 - 1;
					// // cout << "*" <<  v << endl;
					// // cout << siz[v] << endl;
					// // cout << "-" << endl;
				// }
			}
		}
		
		printf("%lld\n",res);
		forn(i,1,n)	st[i] = 0;
		forn(i,1,n)	deg[i] = 0;
    }
    return 0;
}
