import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.InputMismatchException;
import java.io.IOException;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author ankur
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        lexset solver = new lexset();
        solver.solve(1, in, out);
        out.close();
    }

    static class lexset {
        int[] ct = new int[26];
        long mod = (long) 1e9 + 7;
        long[] fact;
        Long[] inv;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            String str1 = in.readString();
            String str2 = in.readString();
            int n = str1.length();
            fact = new long[n + 1];
            fact[0] = 1;
            inv = new Long[n + 1];
            for (int i = 1; i <= n; i++) {
                fact[i] = (fact[i - 1] * i) % mod;
            }
            for (int i = 0; i < str1.length(); i++) {
                ct[str1.charAt(i) - 'a']++;
            }
            long val1 = ans(str1);
            //System.out.print(val1+"dsfadf");
            ct = new int[26];
            for (int i = 0; i < str1.length(); i++) {
                ct[str1.charAt(i) - 'a']++;
            }
            long val2 = ans(str2);
            long fin = val2 - val1 - 1;
            fin = fin % mod;
            if (fin < 0)
                fin += mod;
            out.print(fin);

        }

        long ans(String str) {
            long ans = 0;
            for (int i = 0; i < str.length(); i++) {
                long temp = 1;
                temp = temp * fact(str.length() - i - 1);
                temp = temp % mod;
                for (int j = 0; j < 26; j++) {
                    temp = temp * inv(ct[j]);
                    temp = temp % mod;
                }
                long sum = 0;
                for (int j = 0; j < str.charAt(i) - 'a'; j++) {
                    sum += ct[j];
                    sum = sum % mod;
                }
                temp = temp * sum;
                temp = temp % mod;
                ans = (ans + temp) % mod;
                ct[str.charAt(i) - 'a']--;
                if (ct[str.charAt(i) - 'a'] < 0) {
                    return ans % mod;
                }
            }
            return ans % mod;
        }

        long inv(long n) {
            n = n % mod;
            if (inv[(int) n] != null)
                return inv[(int) n];

            long val = fact((int) n) % mod;
            return inv[(int) n] = pow(val, mod - 2) % mod;
        }

        long pow(long n, long p) {
            if (p == 0)
                return 1;
            if (p % 2 == 0) {
                long val1 = pow(n, p / 2);
                return (val1 * val1) % mod;
            }
            return (n * pow(n, p - 1)) % mod;
        }

        long fact(int n) {
            //  System.out.println(n);
            return fact[n];
        }

    }

    static class InputReader {
        private final InputStream stream;
        private final byte[] buf = new byte[8192];
        private int curChar;
        private int snumChars;

        public InputReader(InputStream st) {
            this.stream = st;
        }

        public int read() {
            if (snumChars == -1)
                throw new InputMismatchException();
            if (curChar >= snumChars) {
                curChar = 0;
                try {
                    snumChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (snumChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public String readString() {
            int c = read();
            while (isSpaceChar(c)) {
                c = read();
            }
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = read();
            } while (!isSpaceChar(c));
            return res.toString();
        }

        public boolean isSpaceChar(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

    }
}
