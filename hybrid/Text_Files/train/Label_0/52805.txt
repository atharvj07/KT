import java.io.*;
import java.util.*;

public class SwitchesAndLamps985B {
	public static void main(String[] args) throws IOException{
		InputReader in = new InputReader();
		PrintWriter out =  new PrintWriter(System.out);
		
		int n = in.nextInt();
		int m = in.nextInt();
		
		HashMap<Integer,HashSet<Integer>> s2l = new HashMap<>();
		HashMap<Integer, Integer> l2s = new HashMap<>();
		
		for(int i=0; i<2000; i++) {
			s2l.put(i, new HashSet<>());
			l2s.put(i, 0);
		}
		
		for(int i=0; i<n; i++) {
			String st = in.nextLine();
			for(int j=0; j<m; j++) {
				if(st.charAt(j)=='1') {
					s2l.get(i).add(j);
					l2s.put(j,l2s.get(j)+1);
				}
			}
		}
		
		boolean found = false;
		for(int i=0; i<n; i++) {
			boolean isReq = false;
			for(int lamp: s2l.get(i)) {
				if(l2s.get(lamp)==1) {
					isReq = true;
					break;
				}
			}
			if(!isReq) {
				out.println("YES");
				found = true;
				break;
			}
		}
		
		if(!found) {
			out.println("NO");
		}
		
		out.close();
	}
	
	static class InputReader{
        private final BufferedReader reader;
        private StringTokenizer string;
        
        public InputReader(){
            reader = new BufferedReader(new InputStreamReader(System.in));
            string = new StringTokenizer(new String());
        }
        
        public String nextLine(){
            try {
                return reader.readLine();
            } catch (IOException ex) {
                return null;
            }
        }
        
        public int nextInt(){
            if(!string.hasMoreTokens()){
                try {
                    string = new StringTokenizer(reader.readLine());
                } catch (IOException ex){}
            }
            return Integer.parseInt(string.nextToken());
        }
        
        public long nextLong(){
            if(!string.hasMoreTokens()){
                try {
                    string = new StringTokenizer(reader.readLine());
                } catch (IOException ex){}
            }
            return Long.parseLong(string.nextToken());
        }
    }
}
