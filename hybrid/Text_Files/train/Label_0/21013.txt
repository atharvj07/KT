#include <bits/stdc++.h>
using namespace std;

using PP = pair<int, int>;
const long inf = 1e18;
int n, m, l, q;
long G[300][300];
long H[300][300];

void wf(long (*D)[300]) {
  for (int k = 0; k < n; ++k) {
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < n; ++j) {
        D[i][j] = min(D[i][j], D[i][k] + D[k][j]);
      }
    }
  }
}

int main() {
  cin >> n >> m >> l;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      G[i][j] = (i == j ? 0 : inf);
    }
  }
  
  for (int j = 0; j < m; ++j) {
    long a, b, c;
    cin >> a >> b >> c;
    --a; --b;
    G[a][b] = min(G[a][b], c);
    G[b][a] = min(G[b][a], c);
  }
  
  wf(G);
  
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      H[i][j] = (G[i][j] <= l ? 1 : inf);
    }
  }
  
  wf(H);

  cin >> q;
  for (int k = 0; k < q; ++k) {
    int a, b;
    cin >> a >> b;
    --a; --b;
    cout << (H[a][b] == inf ? -1 : H[a][b] - 1) << endl;
  }
}
