n = int(input())
s = []
for i in range(n):
	s.append(input().split())
p = list(map(lambda x: int(x) - 1, input().split()))
prev = min(s[p[0]])
ok = True
for i in range(1, n):
	j = p[i]
	if prev < min(s[j]):
		prev = min(s[j])
	elif prev < max(s[j]):
		prev = max(s[j])
	else:
		ok = False
		break
print('YES' if ok else 'NO')
