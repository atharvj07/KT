import java.util.*;

public class E417 {
    static int n;
    static int a[];
    static int useful[];
    static int useless[];
    static int maxA;
    static boolean isUseful[];
    static int S;
    static long usefulNum;
    static ArrayList<ArrayList<Integer>> graph = new ArrayList<>();
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new int[n];
        isUseful = new boolean[n];
        for(int i = 0; i < n; ++i) {
            a[i] = sc.nextInt();
            maxA = Integer.max(maxA, a[i]);
            graph.add(new ArrayList<>());
        }
        useful = new int[maxA + 1];
        useless = new int[maxA + 1];
        for(int i = 1; i < n; ++i) {
            graph.get(sc.nextInt() - 1).add(i);
        }
        countUse(0);
        long ans = 0;
        if(S == 0) {
            ans += (usefulNum * (usefulNum - 1)) / 2;
            long uselessNum = n - usefulNum;
            ans += (uselessNum * (uselessNum - 1)) / 2;
            for(int i = 0; i <= maxA; ++i) {
                ans += (long) useful[i] * useless[i];
            }
        } else {
            for(int i = 0; i <= maxA; ++i) {
                if((S^i) <= maxA) {
                    ans += (long) useful[i] * useless[S^i];
                }
            }
        }
        System.out.println(ans);
    }
    
    static void countUse(int v) {
        if(graph.get(v).size() == 0) {
            isUseful[v] = true;
            S ^= a[v];
            usefulNum++;
            useful[a[v]] ++;
            return;
        }
        for(int i = 0; i < graph.get(v).size(); ++i) {
            countUse(graph.get(v).get(i));
        }
        if(isUseful[graph.get(v).get(0)]) {
            useless[a[v]] ++;
        } else{
            isUseful[v] = true;
            S ^= a[v];
            usefulNum++;
            useful[a[v]] ++;
        }
    }
    
}
