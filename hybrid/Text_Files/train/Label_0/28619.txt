#include<string>
#include<iostream>

#define	rep(i,n)	for(int i=0;i<n;i++)

using namespace std;

int simulate(int n,string s,int q){
	int len=s.length(),b=0,g=0;
	rep(i,len){
		if(s[i]=='B')	b++;
		else		g++;
	}

	int p=len-1,rm=len;
	for(int k=0;b&&g;k++){
		int rot=(n+(k==q))%rm; if(rot==0) rot=rm;
		for(int i=0;i<rot;){
			p++; if(p==len) p=0;
			if(s[p]!='@')	i++;
		}
		if(s[p]=='B')	b--;
		else		g--;
		rm--;
		s[p]='@';
	}

	return b;
}

int solve(int n,string s){
	int len=s.length(),ans=0;
	rep(i,len)	ans=max(ans,simulate(n,s,i));
	return ans;
}

int main(){
	int T;	cin>>T;
	while(T--){
		int n;
		string s;	cin>>n>>s;
		cout<<solve(n,s)<<endl;
	}
	return 0;
}
