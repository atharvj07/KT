import java.util.Scanner;
public class Code {
	public static void main(String[] args){
		Scanner s = new Scanner(System.in);
		
		int n=s.nextInt();
		int hh=s.nextInt();
		int mm=s.nextInt();
			
		int count=0;
		while(true){
			if(isLuckey(hh,mm)>0){
				System.out.println(count);
				System.exit(0);
			}
			count++;
			mm = mm-n;
			if(mm<0){
				mm=60+mm;
				hh--;
				if(hh<0){
					hh=23;
				}
			}
			}
		}
		
	public static int isLuckey(int h,int m){
		if(h/10 == 7)
			return 1;
		else if(h%10 ==7)
			return 1;
		else if(m/10==7)
			return 1;
		else if(m%10==7)
			return 1;
		
		return 0;
	}
}
