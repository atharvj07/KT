import java.util.ArrayList;
import java.util.InputMismatchException;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Iterator;

public class r501p5{

    static PrintWriter pw;

    static class Star{
        int i, j, size;
        public Star(int a, int b, int c){
            i = a;
            j = b;
            size = c;
        }
        void display(){
            pw.println(i+" "+j+" "+size);
        }
    }

    public static void main(String args[]) {
        InputReader sc = new InputReader(System.in);
        pw = new PrintWriter(System.out);

        int n = sc.nextInt(), m = sc.nextInt();
        char grid[][] = new char[n][m];
        boolean covered = true;
        int up[][] = new int[n][m], down[][] = new int[n][m], left[][] = new int[n][m], right[][] = new int[n][m], h[][] = new int[n][m], v[][] = new int[n][m];

        ArrayList<Star> list = new ArrayList<>();

        for(int i=0; i<n; i++)
            grid[i] = sc.readString().toCharArray();

        for(int i=0; i<n; i++) {
            for(int j=0; j<m; j++){
                if(i>0 && grid[i-1][j] == '*')
                    up[i][j] = up[i-1][j]+1;
                if(j>0 && grid[i][j-1] == '*')
                    left[i][j] = left[i][j-1]+1;
            }
        }

        for(int i=n-1; i>=0; i--){
            for(int j=m-1; j>=0; j--){
                if(i<n-1 && grid[i+1][j] == '*')
                    down[i][j] = down[i+1][j]+1;
                if(j<m-1 && grid[i][j+1] == '*')
                    right[i][j] = right[i][j+1]+1;
            }
        }

        for(int i=0; i<n; i++){
            for(int j=0; j<m; j++){
                if(grid[i][j] == '*'){
                    int size = Math.min(Math.min(left[i][j], right[i][j]), Math.min(up[i][j], down[i][j]));
                    if(size > 0){
                        list.add(new Star(i+1, j+1, size));

                        h[i][j-size]++;
                        if(j+size+1<m)
                            h[i][j+size+1]--;
                        v[i-size][j]++;
                        if(i+size+1<n)
                            v[i+size+1][j]--;
                    }
                }
            }
        }

        for(int i=0; i<n; i++){
            for(int j=1; j<m; j++)
                h[i][j] += h[i][j-1];
        }

        for(int j=0; j<m; j++){
            for(int i=1; i<n; i++)
                v[i][j] += v[i-1][j];
        }

        for(int i=0; i<n; i++){
            for(int j=0; j<m; j++){
                if(grid[i][j] == '*' && v[i][j] <= 0 && h[i][j] <= 0)
                    covered = false;
            }
        }

        if(!covered)
            pw.println(-1);
        else{
            pw.println(list.size());
            for(Star s: list)
                s.display();
        }

        pw.flush();
        pw.close();
    }

    static class InputReader {
        private final InputStream stream;
        private final byte[] buf = new byte[8192];
        private int curChar, snumChars;
        private SpaceCharFilter filter;

        InputReader(InputStream stream) {
            this.stream = stream;
        }

        int snext() {
            if (snumChars == -1)
                throw new InputMismatchException();
            if (curChar >= snumChars) {
                curChar = 0;
                try {
                    snumChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (snumChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        int nextInt() {
            int c = snext();
            while (isSpaceChar(c)) {
                c = snext();
            }
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = snext();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = snext();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        long nextLong() {
            int c = snext();
            while (isSpaceChar(c)) {
                c = snext();
            }
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = snext();
            }
            long res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = snext();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        String readString() {
            int c = snext();
            while (isSpaceChar(c)) {
                c = snext();
            }
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = snext();
            } while (!isSpaceChar(c));
            return res.toString();
        }

        String nextLine() {
            int c = snext();
            while (isSpaceChar(c))
                c = snext();
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = snext();
            } while (!isEndOfLine(c));
            return res.toString();
        }

        boolean isSpaceChar(int c) {
            if (filter != null)
                return filter.isSpaceChar(c);
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        private boolean isEndOfLine(int c) {
            return c == '\n' || c == '\r' || c == -1;
        }

        public interface SpaceCharFilter {
            boolean isSpaceChar(int ch);
        }
    }
}
