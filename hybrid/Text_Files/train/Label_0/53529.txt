import java.io.*;
import java.util.*;

public class F950{

	int n;
	long d;
	long b;

	private void solve() throws Exception {
        n = nextInt();
        d = nextInt();
        b = nextLong();
        long[] a = new long[n];
		for(int i = 0; i < n; i++) a[i] = nextLong();
		long total = (n+1)/2*b;
		long sum = 0;
		int mid = 0;
		long t = 0;
        for(int i = 0; i < n; i++) {
			sum += a[i];
			if(sum >= total) {
				t = sum-total;
				mid = i;
				break;
			}
		}
		long[] left = new long[n];
		for(int i = 0; i < mid; i++) left[i] = a[i];
		left[mid] = a[mid]-t;
		int x1 = doit(left, (n+1)/2);

		long[] right = new long[n];
		for(int i = 0; i < n-mid; i++) right[i] = a[n-1-i];
		right[n-1-mid] = t;
		int x2 = doit(right, n/2);
		out.println(Math.max(x1,x2));
	}
	
	int doit(long[] values, int t) {
		long val = 0;
		int res = 0;
		for(int i = 0; i < t; i++) {
			for(long j = (d+1)*i; j <= Math.min((long)(n-1),(d+1)*i+d); j++) val += values[(int)j];
			if(val >= b) val -= b;
			else res++;
		}
		return res;
	}


	public static void main(String[] args) {
		(new F950()).run();
	}

	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer tokenizer;

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			tokenizer = null;
			out = new PrintWriter(System.out);
			solve();
			in.close();
			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	private int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	private float nextFloat() throws IOException {
		return Float.parseFloat(nextToken());
	}

	private String nextLine() throws IOException {
		return new String(in.readLine());
	}

	private String nextToken() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			tokenizer = new StringTokenizer(in.readLine());
		}
		return tokenizer.nextToken();
	}


}
