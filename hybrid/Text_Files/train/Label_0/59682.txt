
import java.awt.Point;
import java.io.*;
import java.math.BigInteger;
import java.util.*;
import static java.lang.Math.*;

public class B implements Runnable {

    BufferedReader in;
    PrintWriter out;
    StringTokenizer tok = new StringTokenizer("");

    public static void main(String[] args) {
        new Thread(null, new B(), "", 256 * (1L << 20)).start();
    }

    public void run() {
        try {
            long t1 = System.currentTimeMillis();
            if (System.getProperty("ONLINE_JUDGE") != null) {
                in = new BufferedReader(new InputStreamReader(System.in));
                out = new PrintWriter(System.out);
            } else {
                in = new BufferedReader(new FileReader("src/input.txt"));
                out = new PrintWriter(System.out);
            }
            Locale.setDefault(Locale.US);
            solve();
            in.close();
            out.close();
            long t2 = System.currentTimeMillis();
            System.err.println("Time = " + (t2 - t1));
        } catch (Throwable t) {
            t.printStackTrace(System.err);
            System.exit(-1);
        }
    }

    String readString() throws IOException {
        while (!tok.hasMoreTokens()) {
            tok = new StringTokenizer(in.readLine());
        }
        return tok.nextToken();
    }

    int readInt() throws IOException {
        return Integer.parseInt(readString());
    }

    long readLong() throws IOException {
        return Long.parseLong(readString());
    }

    double readDouble() throws IOException {
        return Double.parseDouble(readString());
    }
    Edge[] first;
    boolean[] visited;
    int res = 0;

    void solve() throws IOException {
        int n = readInt();
        int m = readInt();
        visited = new boolean[n];
        first = new Edge[n];
        for (int i = 0; i < m; i++) {
            int a = readInt() - 1;
            int b = readInt() - 1;
            first[a] = new Edge(a, b, first[a]);
            first[b] = new Edge(b, a, first[b]);
        }

        for (int i = 0; i < n; i++) {
            for (Edge edge = first[i]; edge != null; edge = edge.edge) {
                checkIsBridge(edge);
            }
        }

        int k = readInt();
        res = 0;
        for (int i = 0; i < k; i++) {
            int a = readInt() - 1;
            int b = readInt() - 1;
            findWay(a, b);
            out.println(res);
            res = 0;
        }
    }

    void findWay(int a, int b) {
        Arrays.fill(visited, false);
        dfs(a, b, true);
    }

    boolean dfs(int x, int target, boolean increase) {
     //  out.println("in " + (x + 1) + " with target " + (1 + target));
        if (visited[x]) {
            return false;
        }
        visited[x] = true;
        if (x == target) {
            return true;
        }

        for (Edge edge = first[x]; edge != null; edge = edge.edge) {
            if (!visited[edge.b] && !edge.deleted && dfs(edge.b, target, increase)) {
                if (increase && edge.isBridge) {
                    res++;
                }
                return true;
            }
        }
        return false;
    }

    void checkIsBridge(Edge edge) {
      //  out.println("check " + (edge.a + 1) + " " + (edge.b + 1));
        edge.deleted = true;
        Arrays.fill(visited, false);
        if (!dfs(edge.a, edge.b, false)) {
            edge.isBridge = true;
        //    out.println((edge.a + 1) + " " + (edge.b + 1) + " is a bridge!");
        }

        edge.deleted = false;
    }

    class Edge {

        Edge edge;
        int a;
        int b;
        boolean isBridge;
        boolean deleted;

        Edge(int a, int b, Edge edge) {
            this.a = a;
            this.b = b;
            this.edge = edge;
        }
    }
}
