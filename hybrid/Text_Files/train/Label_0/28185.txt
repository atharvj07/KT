#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main()
{
	int n;
	cin >> n;
	vector<int> p;
	for (int i = 0;i < n;i++)
	{
		int num;
		cin >> num;
		p.push_back(num);
	}
	int oper;
	cin >> oper;
	while (oper--)
	{
		int head, end;
		cin >> head >> end;
		if (end >= n)
		{
			vector<int>::iterator p1 = p.end();
			p1--;
			int temp = *p1;
			reverse(&p[head], &p[end - 1]);
			p.pop_back();
			p.insert(p.begin()+head, temp);
		}
		else
		{
			reverse(&p[head], &p[end]);
		}
	}
	vector<int>::iterator pp = p.begin();
	while (pp != p.end())
	{
		cout << *pp;
		if (++pp == p.end())cout << endl;
		else cout << " ";
	}
}
