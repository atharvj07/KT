def f(s1, s2):
	find_num = 0
	find = s2[0]
	i = 0
	while find_num < len(s2):
		while i < len(s1) and s1[i] != find:
			i += 1
		if i == len(s1):
			return False
		find_num += 1
		if find_num != len(s2):
			find = s2[find_num]
		i += 1
	return True
s1 = input()
s2 = input()
flag = False
for length in range(len(s1) - len(s2), -1, -1):
	if flag:
		break
	for i in range(len(s1) - length + 1):
		s3 = s1[:i] + s1[i + length:]
		if f(s3, s2):
			print(length)
			flag = True
			break
