import sys
from math import ceil
input = __import__('sys').stdin.readline
(n, m, L, R) = map(int, input().split())
mod = 998244353
if n * m % 2 == 1:
	print(pow(R - L + 1, n * m, mod))
elif n * m % 2 == 0:
	e = (R - L + 1) // 2
	o = R - L + 1 - e
	print((pow(o + e, n * m, mod) + pow(o - e, n * m, mod)) * (mod + 1) // 2 % mod)
