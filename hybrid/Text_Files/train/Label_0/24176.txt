long long mpow(long long a, long long n, long long mo = mod) {
  long long re = 1;
  while (n > 0) {
    if (n & 1) re = re * a % mo;
    a = a * a % mo;
    n >>= 1;
  }
  return re;
}
long long inv(long long b, long long mo = mod) {
  if (b == 1) return b;
  return (mo - mo / b) * inv(mo % b) % mo;
}
const int maxn = 5005;
bool ok[maxn][maxn];
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  ;
  string second;
  cin >> second;
  int n = second.length();
  int K = floor(log2(n));
  ;
  int left = n - (1 << K) + 1;
  ok[0][0] = 1;
  string re = "";
  int cap = (1 << K);
  for (int j = 0; j < left; ++j) {
    char bst = 'z';
    for (int i = 0; i < min(j + cap, n); ++i) {
      if (ok[j][i]) {
        ;
        bst = min(bst, second[i]);
        int tk = i - j;
        for (int k = 0; k < K; ++k) {
          if (((tk & (1 << k)) == 0) && (i + (1 << k)) < n) {
            ok[j][i + (1 << k)] = 1;
            ;
          }
        }
      }
    }
    re += bst;
    if (j != left - 1)
      for (int i = 0; i < min(j + cap, n); ++i) {
        if (ok[j][i] && second[i] == bst) {
          ok[j + 1][i + 1] = 1;
        }
      }
  }
  cout << re << '\n';
}
