import sys
from math import *

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())
(n, k) = mints()
k -= 1
T = []
for z in range(2):
	u = [False] * n
	r = [1] * n
	c = 0
	p = k
	t = 0
	while p <= n - 1 and c != n:
		if u[p] == False:
			if r[p] > 0:
				r[p] -= 1
				t += 1
				if p > 0:
					r[p - 1] += 1
				else:
					r[p + 1] += 1
			else:
				u[p] = True
				c += 1
				t += 1
		else:
			if p == n - 1:
				break
			p += 1
			t += 1
	while p >= 0 and c != n:
		if u[p] == False:
			if r[p] > 0:
				r[p] -= 1
				t += 1
				if p < n - 1:
					r[p + 1] += 1
				else:
					r[p - 1] += 1
			else:
				u[p] = True
				c += 1
				t += 1
		else:
			if p == 0:
				break
			p -= 1
			t += 1
	if c != n:
		print('qwe', z, k)
	T.append(t)
	k = n - 1 - k
print(min(T))
