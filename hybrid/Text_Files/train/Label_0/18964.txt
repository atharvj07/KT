#include <bits/stdc++.h>
using namespace std;
template <class T>
inline T& RD(T& x) {
  char c;
  for (c = getchar(); c < '0'; c = getchar())
    ;
  x = c - '0';
  for (c = getchar(); '0' <= c && c <= '9'; c = getchar()) x = x * 10 + c - '0';
  return x;
}
int p[100005];
vector<int> v[100005];
vector<pair<int, int> > ans;
bool er[100005];
int main() {
  int n;
  scanf("%d", &n);
  if (n <= 2) {
    printf("0\n");
    return 0;
  }
  memset(p, 0, sizeof(p));
  int cnt = 0;
  for (int i = 2; i <= n; i++) {
    if (p[i] == 0) {
      for (int j = 1; i * j <= n; j++) {
        if (p[i * j] == 0) {
          p[i * j] = 1;
          v[cnt].push_back(i * j);
        }
      }
      cnt++;
    }
  }
  int res = 0;
  memset(er, false, sizeof(er));
  for (int i = cnt - 1; i >= 1; i--) {
    int j;
    for (j = v[i].size() - 1; j > 0; j -= 2) {
      ans.push_back(make_pair(v[i][j], v[i][j - 1]));
    }
    if (j == 0) {
      if (2 * v[i][0] <= n) {
        ans.push_back(make_pair(v[i][0], 2 * v[i][0]));
        er[v[i][0] * 2] = true;
      }
    }
  }
  int t = 0;
  for (int i = 0; i < v[0].size(); i++) {
    if (er[v[0][i]] == false) {
      if (t == 0) {
        t = v[0][i];
      } else {
        ans.push_back(make_pair(t, v[0][i]));
        t = 0;
      }
    }
  }
  printf("%d\n", ans.size());
  for (int i = 0; i < ans.size(); i++) {
    pair<int, int> p = ans[i];
    printf("%d %d\n", p.first, p.second);
  }
}
