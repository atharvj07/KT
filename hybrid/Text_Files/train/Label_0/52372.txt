static void testcase() {
	n = s.nextInt();
	int[] a = s.nextIntArray(n);
	shuffleSort(a);
	long[][] dp = new long[n][n];
	long[][] prefix = new long[n][n + 1];
	
	// dp[i][j] -> place i fisherman such that j is the last happy one
	
	for(int i = 0; i < n; i++) dp[i][0] = 1l;
	
	int pointer = 0;
	
	for(int i = 0; i < n; i++) {
		
		// pointer points to the first fish such that a[pointer] * 2 > a[i]
		// that is everything from pointer...i-1 cannot make current fisherman happy
		
		while(2 * a[pointer] <= a[i])
			pointer++;
		
		for(int j = 0; j < i; j++) {
			
			// add all previous ones which can make i'th fisherman last happy
			dp[i][j + 1] += prefix[j][pointer];
			dp[i][j + 1] %= MOD;
		}
		
		for(int j = 0; j < pointer; j++) {
			dp[i][j + 1] += (dp[i][j] * (pointer - j)) % MOD;
			dp[i][j + 1] %= MOD;
		}
		
		for(int j = 0; j <= i; j++) {
			prefix[j][i + 1] = prefix[j][i] + dp[i][j];
			prefix[j][i + 1] %= MOD;
		}
		
	}
	
	out.println(dp[n - 1][n - 1]);
	
	
	
}

static void shuffleArray(int[] a) {
	Random random = new Random();
	for (int i = a.length-1; i > 0; i--) {
		int index = random.nextInt(i + 1);
		int tmp = a[index];
		a[index] = a[i];
		a[i] = tmp;
	}
}
static void shuffleSort(int[] a) {
	shuffleArray(a);
	Arrays.parallelSort(a);
}

static class MyWriter extends PrintWriter {
    MyWriter(OutputStream out) {
        super(out);
    }
    void println(int[] x) {
        println(x, " ");
    }
    void println(int[] x, String delimiter) {
        if (x.length > 0) {
            print(x[0]);
            for (int i = 1; i < x.length; i++) {
                print(delimiter);
                print(x[i]);
            }
        }
        println();
    }
    void println(long[] x) {
        println(x, " ");
    }
    void println(long[] x, String delimiter) {
        if (x.length > 0) {
            print(x[0]);
            for (int i = 1; i < x.length; i++) {
                print(delimiter);
                print(x[i]);
            }
        }
        println();
    }
    void println(Iterable<?> iterable) {
        println(iterable, " ");
    }
    void println(Iterable<?> iterable, String delimiter) {
        Iterator<?> i = iterable.iterator();
        if (i.hasNext()) {
            print(i.next());
            while (i.hasNext()) {
                print(delimiter);
                print(i.next());
            }
        }
        println();
    }
    void printLines(int[] x) {
        println(x, System.lineSeparator());
    }
    void printLines(long[] x) {
        println(x, System.lineSeparator());
    }
    void printLines(Iterable<?> iterable) {
        println(iterable, System.lineSeparator());
    }
}
