for _ in range(int(input())):
	(n, k, l) = [int(i) for i in input().split()]
	d = [int(i) for i in input().split()]
	lst = [0] * n
	for i in range(n):
		z = l - d[i]
		if z < 0:
			lst[i] = [0] * (2 * k)
		else:
			lst[i] = [1] * (2 * k) if z > k - 1 else [0] * (k - z) + [1] * (2 * z + 1) + [0] * (k - z - 1)
	for i in range(1, n):
		for j in range(2 * k):
			if lst[i][j] == 1:
				lst[i][j] = max(lst[i][2 * k - 1], lst[i - 1][2 * k - 1]) if j == 0 else max(lst[i][j - 1], lst[i - 1][j - 1])
			if lst[i - 1] == [0] * (2 * k):
				lst[i] = [0] * (2 * k)
	a = max(lst[-1])
	print('No' if a == 0 else 'Yes')
