import java.util.*;
import java.io.*;

//textbook pitcher pouring problem
public class AbildekWaterSystem {
    public static void main(String[] args) {
        FastScanner scanner = new FastScanner();
        PrintWriter out = new PrintWriter(System.out, false);
        int q = scanner.nextInt();
        top:
        while(q-->0) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int c = scanner.nextInt();
            int c0 = scanner.nextInt();
            long ans = 0;
            TreeMap<Integer, Integer> mapping = new TreeMap<>();
            PriorityQueue<Friend> queue = new PriorityQueue<>();
            for(int i = 0; i < n; i++) {
                int t = scanner.nextInt();
                int a = scanner.nextInt(), b = scanner.nextInt();
                queue.offer(new Friend(t, a, b));
            }
            queue.offer(new Friend(m, 0,0));
            mapping.put(0, c0);
            int sumInside = c0;
            int curtime = 0;
            while(!queue.isEmpty()) {
                Friend ff = queue.poll();
                //fill in for previous time
                int dist = ff.time - curtime;
                sumInside -= dist;
                if (sumInside < 0) {
                    out.println(-1);
                    continue top;
                }
                while(dist > 0) {
                    Integer minkey = mapping.firstKey();
                    int val = mapping.get(minkey);
                    if (val > dist) {
                        mapping.put(minkey, val - dist);
                        ans += (long)dist * minkey;
                        dist = 0;
                    }
                    else {
                        mapping.remove(minkey);
                        ans += (long)val * minkey;
                        dist -= val;
                    }
                }
                sumInside += ff.cap;
                if (mapping.containsKey(ff.cost)) mapping.put(ff.cost, mapping.get(ff.cost) + ff.cap);
                else mapping.put(ff.cost, ff.cap);
                //remove excess
                dist = sumInside - c;
                if (dist > 0) sumInside = c;
                while(dist > 0) {
                    Integer maxkey = mapping.lastKey();
                    int val = mapping.get(maxkey);
                    if (val > dist) {
                        mapping.put(maxkey, val - dist);
                        dist = 0;
                    }
                    else {
                        mapping.remove(maxkey);
                        dist -= val;
                    }
                }
                curtime = ff.time;
            }
            out.println(ans);
        }
        out.flush();
    }
    static class Friend implements Comparable<Friend>{
        int time;
        int cap;
        int cost;
        public Friend(int tt, int cc, int ccc) {
            time = tt; cap = cc; cost = ccc;
        }
        public int compareTo(Friend o) {
            return time - o.time;
        }
    }
    static class Pair {
        int a, b;
        public Pair(int aa, int bb) {
            a = aa; b = bb;
        }
    }
    public static class FastScanner {
        BufferedReader br;
        StringTokenizer st;
        
        public FastScanner(Reader in) {
            br = new BufferedReader(in);
        }
        
        public FastScanner() {
            this(new InputStreamReader(System.in));
        }
        
        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        int nextInt() {
            return Integer.parseInt(next());
        }
        
        long nextLong() {
            return Long.parseLong(next());
        }
        
        double nextDouble() {
            return Double.parseDouble(next());
        }
        
        String readNextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
}
