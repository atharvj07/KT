#include <bits/stdc++.h>
int main() {
  double w, h, p;
  while (scanf("%lf%lf%lf", &w, &h, &p) != EOF) {
    if (w < h) {
      w = w + h;
      h = w - h;
      w = w - h;
    }
    if ((int)fabs(p) % 180 == 0) p = 0;
    if ((int)fabs(p) % 180 > 90) p = 180 - (int)fabs(p) % 180;
    p = (p / 180) * (4 * atan(1));
    if (p == (4 * atan(1)) / 2)
      printf("%.9lf\n", h * h);
    else if (h * (1 + cos(p)) <= w * sin(p))
      printf("%.9lf\n", h * h * (1 / sin(p)));
    else
      printf("%.9lf\n",
             w * h -
                 (w - h * tan(p) / ((1 + 1 / cos(p)))) /
                     (1 + 1 / cos(p) - tan(p) * tan(p) / ((1 + 1 / cos(p)))) *
                     (w - h * tan(p) / ((1 + 1 / cos(p)))) /
                     (1 + 1 / cos(p) - tan(p) * tan(p) / ((1 + 1 / cos(p)))) *
                     tan(p) -
                 ((h -
                   (w - h * tan(p) / ((1 + 1 / cos(p)))) /
                       (1 + 1 / cos(p) - tan(p) * tan(p) / ((1 + 1 / cos(p)))) *
                       tan(p)) /
                  (1 + 1 / cos(p))) *
                     ((h - (w - h * tan(p) / ((1 + 1 / cos(p)))) /
                               (1 + 1 / cos(p) -
                                tan(p) * tan(p) / ((1 + 1 / cos(p)))) *
                               tan(p)) /
                      (1 + 1 / cos(p))) *
                     tan(p));
  }
  return 0;
}
