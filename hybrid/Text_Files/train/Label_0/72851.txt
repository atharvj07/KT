import sys
import math, bisect, operator
(inf, mod) = (float('inf'), 10 ** 9 + 7)
sys.setrecursionlimit(10 ** 5)
from itertools import groupby, accumulate
from heapq import heapify, heappop, heappush
from collections import deque, Counter, defaultdict
I = lambda : int(sys.stdin.readline())
neo = lambda : map(int, sys.stdin.readline().split())
Neo = lambda : list(map(int, sys.stdin.readline().split()))
fact = [1]
for i in range(1, 2 * 10 ** 5 + 1):
	fact += [fact[-1] * i % mod]
for _ in range(I()):
	(n, m, k) = neo()
	A = Neo()
	A.sort()
	Ans = 0
	m -= 1
	for i in range(n):
		t = bisect.bisect_right(A, A[i] + k) - i - 1
		if t >= m:
			Ans += fact[t] * pow(fact[t - m] * fact[m] % mod, mod - 2, mod)
			Ans = Ans % mod
	print(Ans)
