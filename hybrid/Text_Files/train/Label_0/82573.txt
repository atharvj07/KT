N = int(input())
integers = list(map(int, input().split()))
minus = 0
zeros = 0
answer = 0
for n in range(N):
	if integers[n] < 0:
		minus += 1
		answer += -1 - integers[n]
	elif integers[n] > 0:
		answer += integers[n] - 1
	else:
		zeros += 1
if zeros > 0:
	answer += zeros
elif minus % 2 != 0:
	answer += 2
print(answer)
