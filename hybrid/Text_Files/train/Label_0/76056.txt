for _ in range(int(input())):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	tmp = [0] * n
	for i in range(n):
		tmp[i] = arr[i]
	index = dict()
	tmp.sort()
	for i in range(n):
		index[tmp[i]] = i
	count = 0
	for i in range(n):
		if index[arr[i]] > 0 and i > 0 and (tmp[index[arr[i]] - 1] == arr[i - 1]):
			continue
		else:
			count += 1
	print('Yes' if count <= k else 'No')
