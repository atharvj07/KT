#include <iostream>
#include <algorithm>

using namespace std;

int n, S, r[20000], p, ans;

int main(){
  while(cin >> n >> S && (n|S)){
    for(int i=0;i<n;i++) cin >> r[i];
    sort(r, r+n);
    ans = 0;
    for(int i=0;i<n;i++){
      p = upper_bound(r, r+n, S - r[i]) - r;
      if(p > i) ans += n - p;
      else ans += max(n - p - 1, 0);
    }
    cout << ans / 2 << endl;
  }
}
