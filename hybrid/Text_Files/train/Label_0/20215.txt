#include <bits/stdc++.h>
using namespace std;
template <class T>
inline T sqr(T x) {
  return x * x;
}
const double EPS = 1e-6;
const int INF = 0x3fffffff;
const long long LINF = INF * 1ll * INF;
const double PI = acos(-1.0);
pair<int, int> p[300005];
int cnt1[300005], cnt2[300005];
map<pair<int, int>, int> cnt3;
long long fac[300005];
bool cmp(pair<int, int> x, pair<int, int> y) {
  if (x.first == y.first) return x.second < y.second;
  return x.first < y.first;
}
int main() {
  int n;
  scanf("%d", &n);
  fac[0] = 1;
  for (int i = 1; i <= n; i++) {
    fac[i] = fac[i - 1] * i % 998244353;
    scanf("%d%d", &p[i].first, &p[i].second);
    cnt1[p[i].first]++;
    cnt2[p[i].second]++;
    cnt3[p[i]]++;
  }
  sort(p + 1, p + 1 + n, cmp);
  long long res1 = 1, res2 = 1;
  for (int i = 0; i <= n; i++) {
    if (cnt1[i] > 1) res1 = res1 * fac[cnt1[i]] % 998244353;
    if (cnt2[i] > 1) res2 = res2 * fac[cnt2[i]] % 998244353;
  }
  bool flag = true;
  for (int i = 1; i < n; i++) {
    if (p[i].second > p[i + 1].second) {
      flag = false;
      break;
    }
  }
  long long ans;
  if (flag) {
    long long res3 = 1;
    for (auto &it : cnt3) {
      if (it.second > 1) res3 = res3 * fac[it.second] % 998244353;
    }
    ans = ((fac[n] - res1 - res2 + res3) % 998244353 + 998244353) % 998244353;
  } else
    ans = ((fac[n] - res1 - res2) % 998244353 + 998244353) % 998244353;
  printf("%lld\n", ans);
  return 0;
}
