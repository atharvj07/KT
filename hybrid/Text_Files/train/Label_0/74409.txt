n = input()
c_n = {}
d_n = {}
for (a_i, b_i) in zip(input().split(), input().split()):
	if a_i == b_i:
		d_n[a_i] = d_n.get(a_i, 0) + 2
	c_n[a_i] = c_n.get(a_i, 0) + 1
	c_n[b_i] = c_n.get(b_i, 0) + 1
result = 1
k = int(input())
for (a_i, cant) in c_n.items():
	cant_rep = d_n.get(a_i, 0)
	for i in range(cant - cant_rep + 1, cant, 2):
		result = result * i * (i + 1) // 2 % k
	for i in range(1, cant - cant_rep + 1):
		result = result * i % k
print(result)
