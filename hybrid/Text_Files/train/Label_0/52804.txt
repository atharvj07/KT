import javafx.util.Pair;
import java.util.*;
import static java.lang.Math.*;


public class Main {

    static class Sortpair implements Comparator<Pair> {
        @Override
        public int compare(Pair o1, Pair o2) {
            int o1First = (int)o1.getKey();
            int o2First = (int)o2.getKey();

            if(o1First!=o2First){
                return o1First-o2First;
            }

            int o1Second = (int) o1.getValue();
            int o2Second = (int) o2.getValue();

            return  o2Second-o1Second;
        }
    }

    static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {

        int k,n,s,p;
        k=scanner.nextInt();
        n=scanner.nextInt();
        s=scanner.nextInt();
        p=scanner.nextInt();

        int sheetsperperson = (int)Math.ceil(((double)(n))/s);
        int packs = (int)Math.ceil(((double)sheetsperperson*k)/p);
        System.out.println(packs);
    }
}
