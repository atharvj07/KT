import math

def solve():
	(n, m) = map(int, input().split())
	maxi = math.ceil(m / 2)
	A = [[] for _ in range(m)]
	for i in range(m):
		(k, *A[i]) = map(int, input().split())
	ans = list()
	cnt = dict()
	f = -1
	for i in range(m):
		ff = A[i][0]
		ans.append(ff)
		cnt[ff] = cnt.get(ff, 0) + 1
		if cnt[ff] > maxi:
			f = ff
	for i in range(m):
		if f != -1 and cnt[f] > maxi and (len(A[i]) > 1) and (f == A[i][0]):
			cnt[f] = cnt[f] - 1
			ans[i] = A[i][1]
	if cnt.get(f, 0) <= maxi:
		print('YES\n', *ans)
	else:
		print('NO')
T = int(input())
for _ in range(T):
	solve()
