t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	s = input()
	dx = 0
	dy = 0
	maxleft = 0
	maxright = 0
	maxup = 0
	maxdown = 0
	for sym in s:
		if sym == 'L':
			dx -= 1
			maxleft = min(dx, maxleft)
			if maxright - maxleft >= m:
				maxleft += 1
				break
		elif sym == 'R':
			dx += 1
			maxright = max(dx, maxright)
			if maxright - maxleft >= m:
				break
		elif sym == 'D':
			dy += 1
			maxdown = max(dy, maxdown)
			if maxdown - maxup >= n:
				break
		else:
			dy -= 1
			maxup = min(dy, maxup)
			if maxdown - maxup >= n:
				maxup += 1
				break
	correcty = -maxup + 1
	correctx = -maxleft + 1
	print(correcty, correctx)
