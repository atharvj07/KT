(n, l, r, x) = map(int, input().split())
lis = list(map(int, input().split()))
sol = 0
for i in range(3, 1 << n):
	mini = float('Inf')
	maxi = -float('Inf')
	tot = 0
	for j in range(n):
		if 1 << j & i != 0:
			mini = min(lis[j], mini)
			maxi = max(lis[j], maxi)
			tot += lis[j]
	if tot >= l and tot <= r and (maxi - mini >= x):
		sol += 1
print(sol)
