import java.util.*;
import java.io.*;

public class E66 implements Runnable {

	public StreamTokenizer in;

	public PrintWriter out;

	E66() throws IOException {
		in = new StreamTokenizer( new InputStreamReader( System.in ) );
		out = new PrintWriter( System.out );
	}

    int nextInt() throws IOException {
        in.nextToken();
        return ( int ) in.nval;
    }

    void close() throws IOException {
		out.close();
	}

	public void run() {
		try {
			solve();
			close();
		} catch ( Exception e ) {
			e.printStackTrace();
		}
	}

	public void solve() throws IOException {
        int n = nextInt();
        int x = nextInt();
        int[] a = new int[n];
        for ( int i = 0; i < n; i ++ ) {
            a[i] = nextInt();
        }
        Arrays.sort( a );
        if ( n > 0 && a[0] == 1 ) {
            out.println( ( x == 2 ) ? 0 : 1 );
            return;
        }
        boolean[] np = new boolean[1300000];
        int pn = 0;
        for ( int i = 2; i < np.length && i < x; i ++ ) {
            if ( np[i] ) {
                continue;
            }
            if ( Arrays.binarySearch( a, i ) < 0 ) {
                out.println( -1 );
                return;
            }
            pn ++;
            for ( int j = 2 * i; j < np.length; j += i ) {
                np[j] = true;
            }
        }
        out.println( pn );
	}

	public static void main( String[] args ) throws IOException {
		new Thread( new E66() ).start();
	}
}
