//package codeforces336;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import static java.util.Collections.sort;
import java.util.StringTokenizer;

/**
 *
 * @author Sourav Kumar Paul
 */
public class SolveC {

    public static class Pair implements Comparable<Object> {

        int a, b;

        Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }

        public int compareTo(Object obj) {
            Pair st = (Pair) obj;
            if (a == st.a) {
                return 0;
            } else if (a > st.a) {
                return 1;
            } else {
                return -1;
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
         ArrayList<Pair> list = new ArrayList<Pair>();
         StringTokenizer st;
         
        for(int i=0; i<n; i++)
        {
            st = new StringTokenizer(reader.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            Pair pp = new Pair(a,b);
            list.add(pp);
        }
        sort(list);
        int range[] = new int[1000005];
        
        for(Pair pair : list)
        {
            range[pair.a] = pair.b;
        }
        int dp[] = new int[1000005];
        int max = 0;
        if(range[0]>0)
        {
            dp[0] = 1;
            max = 1;
        }
        
        for(int i=1; i<=list.get(list.size()-1).a; i++)
        {
            if(range[i] == 0)
                dp[i] = dp[i-1];
            else if(range[i] >= i)
            {
                dp[i] = 1;
            }
            else
                dp[i] = dp[i-range[i]-1] + 1;
            
            max = Math.max(max, dp[i]); 
        }
        System.out.println(n-max);
       
    }
}
