
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.Arrays;

public class D {

    static int log2(int i) {
        int ret=0;
        while(i>0) {
            ret++;
            i=i>>1;
        }
        return ret;
    }

    static void compute(int n, int[][] q) {
        //group queries by level
        int[] l2qcnt=new int[n+1];
        for (int i=0; i<q.length; i++) {
            if(q[i][0]==1) {
                l2qcnt[q[i][1]]++;
            }
        }
        int[][] l2q=new int[n+1][];
        for (int i=1; i<=n; i++) {
            l2q[i]=new int[l2qcnt[i]];
        }
        Arrays.fill(l2qcnt, 0);
        //end group queries by level
        
        int[] s=new int[1000000];
        for (int i=0; i<q.length; i++) {
            int ret=0;
            int lvl=q[i][1];
            if(q[i][0]==1) {
                l2q[lvl][l2qcnt[lvl]++]=i;
            } else {
                int l=q[i][2], r=l;
                for (int j=lvl; j<=n; j++) {
                    for (int k=0; k<l2qcnt[j]; k++) {
                        int qidx=l2q[j][k];
                        int ql=q[qidx][2];
                        int qr=q[qidx][3];
                        if(!(r<ql||l>qr)) {
                            int x=q[qidx][4];
                            if(s[x]!=i+1) {
                                ret++;
                                s[x]=i+1;
                            }
                        }
                    }
                    //update l,r
                    int extra=Integer.bitCount(r)==1? 1: 0;
                    l+=log2(l-1);
                    r+=log2(r-1);
                    r+=extra;
                }
                System.out.println(ret);
            }
        }
    }
    
    
    
    
    static StringBuilder sb=new StringBuilder();
    static void parse(String l, int[] here) {
        sb.setLength(0);
        int len=l.length(), j=0;
        for (int i=0; i<len; i++) {
            char c=l.charAt(i);
            if(c==' ') {
                here[j++]=Integer.parseInt(sb.toString());
                sb.setLength(0);
            } else {
                sb.append(c);
            }
        }
        if(sb.length()>0) {
            here[j++]=Integer.parseInt(sb.toString());
        }
    }
    
    static void run_stream(InputStream ins) throws IOException {
        BufferedReader r=new BufferedReader(new InputStreamReader(System.in));
        String[] l=r.readLine().split(" ");
        int n=Integer.parseInt(l[0]);
        int m=Integer.parseInt(l[1]);
        
        int[][] q=new int[m][5];
        for (int i=0; i<m; i++) {
            parse(r.readLine(), q[i]);
        }
        
        compute(n, q);
    }

    public static void main(String[] args) throws IOException {
        run_stream(System.in);
    }
    
}
