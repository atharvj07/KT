s = input()
p = input()
a = list(map(int, input().split()))
a_inds = [0] * (len(s) + 1)
for i in range(len(a)):
	a_inds[a[i]] = i
l = -1
r = len(s)

def check_answer(ans):
	i = 0
	j = 0
	while i < len(s) and j < len(p):
		if a_inds[i + 1] >= ans and s[i] == p[j]:
			j += 1
		i += 1
	return j == len(p)
while l < r - 1:
	m = (l + r) // 2
	if check_answer(m):
		l = m
	else:
		r = m
print(r - 1)
