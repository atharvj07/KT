from sys import stdin, stdout
(n, m) = map(int, stdin.readline().split())
nonOccup = n * n
ligOccupe = [False] * n
nbLigSup = 0
colOccupe = [False] * n
nbColSup = 0
for tour in range(m):
	(lig, col) = map(int, stdin.readline().split())
	lig -= 1
	col -= 1
	if not ligOccupe[lig]:
		nonOccup -= n - nbColSup
		ligOccupe[lig] = True
		nbLigSup += 1
	if not colOccupe[col]:
		nonOccup -= n - nbLigSup
		colOccupe[col] = True
		nbColSup += 1
	stdout.write(str(nonOccup) + ' ')
