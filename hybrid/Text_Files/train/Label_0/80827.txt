from sys import stdin, stdout
(input, print) = (stdin.readline, stdout.write)
n = int(input())
a = [int(i) for i in input().split()]
ans = 0
for i in range(n):
	for j in range(i + 1, n):
		ans += int(a[i] > a[j])
ans %= 2
q = int(input())
for i in range(q):
	(l, r) = [int(i) for i in input().split()]
	ans ^= (r - l) * (r - l + 1) // 2 % 2
	print('odd\n' if ans else 'even\n')
