def int_input():
	a = int(input())
	return a

def int_list_input():
	a = list(map(int, input().split()))
	return a

def solve():
	n = int(input())
	arr = int_list_input()
	ans = n // 2
	i = 0
	y = min(arr)
	while ans != 0:
		x = arr[i]
		if x != y:
			print(x, y)
			ans -= 1
		i += 1
for _ in range(int(input())):
	solve()
