#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false), cin.tie(0);
  long long t = 1, m, i, j, k, u, to;
  while (t--) {
    string s;
    cin >> s;
    int n = s.length();
    vector<vector<int>> dp(n, vector<int>(n)), pal(n, vector<int>(n));
    for (i = 0; i < n; i++) {
      dp[i][i] = pal[i][i] = 1;
      if (i + 1 < n) {
        dp[i][i + 1] = 2;
        dp[i][i + 1] += pal[i][i + 1] = s[i] == s[i + 1];
      }
    }
    for (int len = 3; len <= n; len++) {
      for (i = 0; i <= n - len; i++) {
        j = i + len - 1;
        if (s[i] == s[j]) {
          pal[i][j] = pal[i + 1][j - 1];
        }
        dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1];
        dp[i][j] += pal[i][j];
      }
    }
    int q;
    cin >> q;
    while (q--) {
      cin >> u >> to;
      u--, to--;
      cout << dp[u][to] << '\n';
    }
  }
}
