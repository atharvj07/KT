import java.io.*;
import java.util.*;

public class CF920D {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int k = Integer.parseInt(st.nextToken());
		int v = Integer.parseInt(st.nextToken());
		st = new StringTokenizer(br.readLine());
		int[] aa = new int[n];
		int sum = 0;
		for (int i = 0; i < n; i++)
			sum += aa[i] = Integer.parseInt(st.nextToken());
		if (sum < v) {
			System.out.println("NO");
			return;
		}
		boolean[][] dp = new boolean[n + 1][k];
		dp[0][0] = true;
		for (int i = 0; i < n; i++)
			for (int w = 0; w < k; w++)
				if (dp[i][w]) {
					dp[i + 1][w] = true;
					int w_ = (w + aa[i]) % k;
					dp[i + 1][w_] = true;
				}
		if (!dp[n][v % k]) {
			System.out.println("NO");
			return;
		}
		PrintWriter pw = new PrintWriter(System.out);
		pw.println("YES");
		boolean[] used = new boolean[n];
		for (int i = n, w = v % k; i > 0; i--)
			if (!dp[i - 1][w]) {
				used[i - 1] = true;
				w = ((w - aa[i - 1]) % k + k) % k;
			}
		int i0 = -1, i1 = -1;
		for (int i = 0; i < n; i++)
			if (used[i]) {
				if (i1 == -1)
					i1 = i;
				else {
					int cnt = (aa[i] + k - 1) / k;
					if (cnt > 0) {
						pw.println(cnt + " " + (i + 1) + " " + (i1 + 1));
						aa[i1] += aa[i]; aa[i] = 0;
					}
				}
			} else {
				if (i0 == -1)
					i0 = i;
				else {
					int cnt = (aa[i] + k - 1) / k;
					if (cnt > 0) {
						pw.println(cnt + " " + (i + 1) + " " + (i0 + 1));
						aa[i0] += aa[i]; aa[i] = 0;
					}
				}
			}
		if (i0 == -1)
			i0 = i1 == 0 ? 1 : 0;
		if (i1 == -1)
			i1 = i0 == 0 ? 1 : 0;
		int cnt = (v - aa[i1]) / k;
		if (cnt > 0)
			pw.println(cnt + " " + (i0 + 1) + " " + (i1 + 1));
		else if (cnt < 0)
			pw.println(-cnt + " " + (i1 + 1) + " " + (i0 + 1));
		pw.close();
	}
}
