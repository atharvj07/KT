#include <bits/stdc++.h>
using namespace std;
const int N = 3 * 100000 + 10000;
char s[N];
int last[30];
long long ans[30];
long long p[30];
int main() {
  scanf("%s", s);
  int len = strlen(s);
  memset(last, -1, sizeof(last));
  memset(ans, 0, sizeof(ans));
  int cnt = 0;
  for (int i = 0; i < len; i++) {
    last[s[i] - 'a'] = i;
    cnt = 0;
    for (int j = 0; j < 26; j++)
      if (last[j] != -1) {
        p[cnt++] = last[j];
      }
    sort(p, p + cnt);
    for (int j = 1; j < cnt; j++) ans[cnt - j] += p[j] - p[j - 1];
    ans[cnt] += p[0] + 1;
  }
  for (int i = 26; i >= 1; i--)
    if (ans[i] > 0) {
      printf("%d\n", i);
      for (int j = 1; j <= i; j++) printf("%I64d\n", ans[j]);
      break;
    }
  return 0;
}
