#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[2 * n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    a[i + n] = a[i];
  }
  int l = 0, r = 1;
  int ans = -1;
  while (l < n) {
    if (a[r] >= a[r - 1]) {
      r++;
      if (r - l == n) {
        if (l == 0) {
          ans = 0;
        } else {
          ans = ans != -1 ? min(ans, n - l) : n - l;
        }
      }
    } else {
      l = r;
      r++;
    }
  }
  cout << ans << '\n';
  return 0;
}
