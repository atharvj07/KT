__date__ = '2021-03-16'
import sys
from bisect import bisect_left
from itertools import accumulate
from math import inf

def solve(n, m, a, b):
	if sum(a) < m:
		return -1
	b1 = list()
	b2 = list()
	for i in range(n):
		if b[i] == 1:
			b1.append(a[i])
		else:
			b2.append(a[i])
	p1 = [0] + list(accumulate(sorted(b1, reverse=True)))
	p2 = [0] + list(accumulate(sorted(b2, reverse=True)))
	answer = inf
	for j in range(len(p1)):
		k = bisect_left(p2, m - p1[j])
		if k < len(p2):
			answer = min(j + 2 * k, answer)
		if p1[j] > m:
			return answer
	return answer

def main(argv=None):
	t = int(input())
	for _ in range(t):
		(n, m) = map(int, input().split())
		a = list(map(int, input().split()))
		b = list(map(int, input().split()))
		print(solve(n, m, a, b))
	return 0
STATUS = main()
sys.exit(STATUS)
