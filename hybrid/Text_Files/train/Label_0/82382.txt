def cal(a, b, c, v, sl):
	if a == 0 and b == 0:
		return c * sl
	res = 0
	if a > 1e-08:
		l = min(a, v)
		if b > 1e-08:
			res += a * cal(a - l, b + l / 2, c + l / 2, v, sl + 1)
		else:
			res += a * cal(a - l, 0, c + l, v, sl + 1)
	if b > 1e-08:
		l = min(b, v)
		if a > 1e-08:
			res += b * cal(a + l / 2, b - l, c + l / 2, v, sl + 1)
		else:
			res += b * cal(0, b - l, c + l, v, sl + 1)
	return sl * c + res
n = int(input())
for i in range(n):
	(a, b, c, v) = map(float, input().split())
	print(cal(a, b, c, v, 1))
