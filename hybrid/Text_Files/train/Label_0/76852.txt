n = int(input())
d = {}
c = 1
while n:
	n -= 1
	l = tuple(map(int, input().split()))
	d[c] = l
	c += 1
for i in d:
	if d[i] != -1:
		ans = 1e+18
		k = [-1, -1]
		x1 = d[i][0]
		y1 = d[i][1]
		z1 = d[i][2]
		for j in d:
			if d[j] != -1 and i != j:
				x2 = d[j][0]
				y2 = d[j][1]
				z2 = d[j][2]
				aa = abs(x1 - x2) + abs(y1 - y2) + abs(z1 - z2)
				if aa < ans:
					ans = aa
					k[0] = i
					k[1] = j
		print(k[0], k[1])
		d[k[0]] = -1
		d[k[1]] = -1
