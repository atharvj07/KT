for _ in range(int(input())):
	(n, k, m) = list(map(int, input().split()))
	arr = list(map(int, input().split()))
	remain = [0] * (n + 1)
	for x in arr:
		remain[x] = 1
	use = n - m
	if use % (k - 1) != 0:
		print('NO')
	else:
		flg = False
		cnt = 0
		for x in range(1, n + 1):
			if remain[x] == 1:
				if cnt >= k // 2 and use - cnt >= k // 2:
					flg = True
			else:
				cnt += 1
		if flg == True:
			print('YES')
		else:
			print('NO')
