import datetime
t = int(input())
for i in range(t):
	n = int(input())
	string = input()
	if n == 1:
		print(string)
	else:
		answer = ''
		for j in range(1, n + 1):
			if j != n:
				temp = string[j - 1:n + j]
				answer += temp[j - 1]
			else:
				temp = string[j - 1:n + j - 1]
				answer += temp[j - 1]
		print(answer)
