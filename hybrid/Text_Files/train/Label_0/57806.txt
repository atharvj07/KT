import java.util.*;
import java.io.*;
public class Main {
    public static long MOD = 998244353;
    public static int mxN = 200001;
    public static long fact[] = new long[200001];
    public static long ifact[] = new long[200001];
    public static String S;
    public static int n, q;
    public static int b[] = new int[2];
    public static int w[] = new int[2];
    public static int f[] = new int[2];
    public static long tpow[] = new long[200001];
    public static long tinv[] = new long[200001];
    public static long ans[] = new long[4];
    public static int lft, func;
    public static void main(String[] args) throws IOException{
        Scanner sc = new Scanner(System.in);
    	PrintWriter out = new PrintWriter(System.out);
        fact[0] = 1;
        tpow[0] = 1;
        for(int i = 1; i < mxN; ++i){
            fact[i] = (fact[i-1]*i)%MOD;
            tpow[i] = (tpow[i-1]*2)%MOD;
        }
        ifact[mxN-1] = modInv(fact[mxN-1]);
        tinv[mxN-1] = modInv(tpow[mxN-1]);
        for(int i = mxN-2; i >= 0; --i) {
            ifact[i] = (ifact[i+1]*(i+1))%MOD;
            tinv[i] = (tinv[i+1]*2)%MOD;
        }
        n = sc.nextInt();
        q = sc.nextInt();
        S = sc.next();
        StringBuilder s = new StringBuilder(S);
        for(int i = 0; i < n; ++i) {
            if(s.charAt(i)=='w') {
                w[i&1]++;
            }else if(s.charAt(i)=='b') {
                b[i&1]++;
            }else {
                f[i&1]++;
            }
        }
        lft = f[0] + f[1];
        func = n/2 - b[1] - w[0];
        fix();
        long res = (ans[1]*lft + ans[2]*func)%MOD;
        res = (ans[0]*func + ans[3]*lft - res + MOD)%MOD;
        res = (res * tinv[lft])%MOD;
        System.out.println(res);
        for(int i = 0; i < q; ++i) {
            int a;
            char c;
            a = sc.nextInt();
            c = (sc.next()).charAt(0);
            a--;
            //uncolor
            if(s.charAt(a)=='w') {
                inc();
                if(((a&1)^1) > 0) binc();
            }else if(s.charAt(a)=='b') {
                inc();
                if((a&1) > 0) binc();
            }
            //recolor
            if(c=='w') {
                dec();
                if(((a&1)^1) > 0) bdec();
            }else if(c=='b') {
                dec();
                if((a&1) > 0) bdec();
            }
            s.setCharAt(a, c);
            if(lft<=2) fix();
            res = (ans[1]*lft + ans[2]*func)%MOD;
            res = (ans[0]*func + ans[3]*lft - res + MOD)%MOD;
            res = (res * tinv[lft])%MOD;
            System.out.println(res);
        }
    }
    public static void fix() {
        ans[0] = 0;
        ans[1] = 0;
        ans[2] = 0;
        ans[3] = 0;
        int parity = Math.abs(func)&1;
        for(int i = parity; i <= Math.min(func-2, lft); i+=2) {
            ans[0] = (ans[0] + nCr(lft, i))%MOD;
        }
        for(int i = parity; i <= Math.min(func-2, lft); i+=2) {
            ans[1] = (ans[1] + nCr(lft-1, i-1))%MOD;
        }
        for(int i = func+2; i <= lft; i+=2) {
            ans[3] = (ans[3] + nCr(lft-1, i-1))%MOD;
        }
        for(int i = func+2; i <= lft; i+=2) {
            ans[2] = (ans[2] + nCr(lft, i))%MOD;
        }
    }
    public static void inc() { // increase the numerator
        if(lft==1) {
            lft++;
            return;
        }
        ans[0] = (2*ans[0] - nCr(lft-1, func-2) + MOD)%MOD;
        ans[1] = (2*ans[1] - nCr(lft-2, func-3) + MOD)%MOD;
        ans[2] = tpow[lft-1] - ans[2];
        ans[2] = 2*ans[2] - nCr(lft-1, func);
        ans[2] = ((tpow[lft] - ans[2])%MOD + MOD)%MOD;
        ans[3] = tpow[lft-2] - ans[3];
        ans[3] = 2*ans[3] - nCr(lft-2, func-1);
        ans[3] = ((tpow[lft-1] - ans[3])%MOD + MOD)%MOD;
        lft++;
    }
    public static void binc() { //increase denominator
        ans[0] = (ans[0] + nCr(lft-1, func-1))%MOD;
        ans[1] = (ans[1] + nCr(lft-2, func-2))%MOD;
        ans[2] = (ans[2] - nCr(lft-1, func+1) + MOD)%MOD;
        ans[3] = (ans[3] - nCr(lft-2, func) + MOD)%MOD;
        func++;
    }
    public static void dec() { //decrease the numerator
        if(lft < 3) {
            lft--;
            return;
        }
        ans[0] = ((ans[0] + nCr(lft-2, func-2))*tinv[1])%MOD;
        ans[1] = ((ans[1] + nCr(lft-3, func-3))*tinv[1])%MOD;
        ans[2] = tpow[lft-1] - ans[2];
        ans[2] = ((ans[2] + nCr(lft-2, func))*tinv[1])%MOD;
        ans[2] = (tpow[lft-2] - ans[2] + MOD)%MOD;
        ans[3] = tpow[lft-2] - ans[3];
        ans[3] = ((ans[3] + nCr(lft-3, func-1))*tinv[1])%MOD;
        ans[3] = (tpow[lft-3] - ans[3] + MOD)%MOD;
        lft--;
    }
    public static void bdec() { //decrease denominator
        ans[0] = (ans[0] - nCr(lft-1, func-2) + MOD)%MOD;
        ans[1] = (ans[1] - nCr(lft-2, func-3) + MOD)%MOD;
        ans[2] = (ans[2] + nCr(lft-1, func))%MOD;
        ans[3] = (ans[3] + nCr(lft-2, func-1))%MOD;
        func--;
    }
    public static long nCr(int n, int r) {
        if(r > n || r < 0)return 0;
        return ((fact[n]*ifact[n-r])%MOD*ifact[r])%MOD;
    }
    public static long binpow(long a,long b) {
        long res=1;
        while(b > 0) {
            if((b&1) > 0)res=(res*a)%MOD;
            a=(a*a)%MOD;
            b>>=1;
        }
        return res;
    }
    public static long modInv(long a) {
        return binpow(a, MOD-2);
    }	
}
