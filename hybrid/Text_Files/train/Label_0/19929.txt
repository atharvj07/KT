#include <iostream>
#include <vector>
using namespace std;

class TETORIS {
public:
    int N;
    vector<vector<int> > bd;
    TETORIS() : N(0) {}
    void put(int len, int pos) {
        int k;
        for (k=bd.size()-1; k>=0; --k) {
            for (int i=0; i<len; ++i) {
                if (bd[k][pos+i]) goto brk;
            }
        } brk:;
        ++k;
        if ((int)bd.size() == k) bd.push_back(vector<int>(5, 0));
        for (int i=0; i<len; ++i) {
            bd[k][pos+i] = 1;
        }
        N += len;
    }
    void update() {
        for (int i=bd.size()-1; i>=0; --i) {
            if (bd[i][0] && bd[i][1] && bd[i][2] && bd[i][3] && bd[i][4]) {
                bd.erase(bd.begin()+i);
                N -= 5;
            }
        }
    }
};

int main() {
    int n;
    while (cin >> n, n) {
        TETORIS tet;
        int d, p, q;
        for (int i=0; i<n; ++i) {
            cin >> d >> p >> q;
            --q;
            if (d == 1) {
                tet.put(p, q);
                tet.update();
            } else {
                for (int j=0; j<p; ++j) tet.put(1, q);
                tet.update();
            }
        }
        cout << tet.N << endl;
    }
    return 0;
}
