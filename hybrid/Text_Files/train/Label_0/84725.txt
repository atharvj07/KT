t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	length = len(s)
	prefix_one = suffix_z = -1
	for (i, c) in enumerate(s):
		if c == '1':
			prefix_one = i
			break
	for i in range(length - 1, -1, -1):
		if s[i] == '0':
			suffix_z = i
			break
	if prefix_one == -1 or suffix_z == -1 or prefix_one == suffix_z + 1:
		print(s)
		continue
	ans = s[:prefix_one] + '0' + s[suffix_z + 1:]
	print(ans)
