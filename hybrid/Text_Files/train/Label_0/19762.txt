#include <bits/stdc++.h>
using namespace std;
long long int n, k, x, cnt[30], ans;
string s;
int main() {
  cin >> n >> k >> s;
  for (int i = 0; i < s.size(); i++) {
    x = s[i] - 'A';
    cnt[x]++;
  }
  sort(cnt, cnt + 26);
  for (int i = 25; i >= 0; i--) {
    x = min(k, cnt[i]);
    ans += x * x;
    k -= x;
  }
  cout << ans << '\n';
}
