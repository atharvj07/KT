#include <bits/stdc++.h>
using namespace std;
int Next[200005];
void getnext(string s) {
  int i, j, len = ((int)(s).size());
  Next[0] = j = -1;
  for (i = 1; i < len; i++) {
    while (j >= 0 && s[j + 1] != s[i]) j = Next[j];
    if (s[j + 1] == s[i]) j++;
    Next[i] = j;
  }
}
int main() {
  string s1, s2;
  int i, ans, d, l1, l2, len;
  while (cin >> s1 >> s2) {
    ans = 0;
    l1 = ((int)(s1).size()), l2 = ((int)(s2).size());
    s1 += s2;
    getnext(s1);
    len = ((int)(s1).size());
    if (Next[((int)(s1).size()) - 1] != -1) {
      d = len - 1 - Next[((int)(s1).size()) - 1];
      if (len % d == 0) {
        for (i = d; i <= l1 && i <= l2; i += d) {
          if (l1 % i == 0 && l2 % i == 0) ans++;
        }
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
