#include <bits/stdc++.h>
using namespace std;
long long a[200010], b[200010];
vector<char> ans;
vector<int> res;
long long cnt = 0;
long long sum = 0;
bool solve2() {
  bool flag = false;
  if (a[1] > a[2]) flag = true, swap(a[1], a[2]);
  if (b[1] > b[2]) res.push_back(-1), swap(b[1], b[2]), cnt++;
  while (1) {
    if (a[1] < b[1]) {
      if (b[1] == b[2]) return 0;
      long long x = b[2] / b[1];
      cnt += x;
      sum += x;
      res.push_back(x);
      b[2] %= b[1];
    } else if (a[1] == b[1]) {
      if (a[1] == b[1] && a[2] == b[2] && flag && res.size()) {
        res.pop_back();
        cnt--;
        return 1;
      }
      if ((b[2] - a[2]) % b[1] != 0) return 0;
      long long x = (b[2] - a[2]) / b[1];
      cnt += x;
      sum += x;
      res.push_back(x);
      b[2] = a[2];
      if (flag) {
        cnt++;
        res.push_back(-1);
      }
      return 1;
    } else if (b[1] < a[1])
      return 0;
    swap(b[1], b[2]);
    cnt++;
    res.push_back(-1);
  }
}
int n;
long long na[200010];
bool check(long long *a, long long *b) {
  for (int i = 1; i <= n; i++) {
    if (a[i] != b[i]) return 0;
  }
  return 1;
}
bool solve() {
  for (int i = 1; i <= n; i++) na[i] = a[n - i + 1];
  while (1) {
    if (check(b, a)) {
      return 1;
    }
    if (check(b, na)) {
      res.push_back(-1);
      cnt++;
      return 1;
    }
    if (b[1] < b[2]) {
    } else if (b[1] == b[2])
      return 0;
    else {
      reverse(b + 1, b + n + 1);
      res.push_back(-1);
      cnt++;
    }
    for (int i = n; i >= 1; i--) {
      b[i] = b[i] - b[i - 1];
      if (b[i] < 0) return 0;
    }
    res.push_back(1);
    sum++;
    cnt++;
  }
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  if (n == 1) {
    if (a[1] == b[1])
      cout << "SMALL" << endl << '0' << endl;
    else
      cout << "IMPOSSIBLE" << endl;
    return 0;
  } else if (n == 2) {
    bool flag = solve2();
    if (!flag) {
      cout << "IMPOSSIBLE" << endl;
      return 0;
    }
    if (sum <= 200000) {
      ans.clear();
      for (int i = 0; i < res.size(); i++) {
        if (res[i] < 0) {
          ans.push_back('R');
        } else {
          for (int j = 0; j < res[i]; j++) {
            ans.push_back('P');
          }
        }
      }
      cout << "SMALL" << endl;
      cout << ans.size() << endl;
      for (int i = ans.size() - 1; i >= 0; i--) cout << ans[i];
    } else {
      cout << "BIG" << endl;
      cout << sum << endl;
    }
    return 0;
  } else {
    bool flag = solve();
    if (!flag) {
      cout << "IMPOSSIBLE" << endl;
      return 0;
    }
    if (cnt <= 200000) {
      for (int i = 0; i < res.size(); i++) {
        if (res[i] < 0) {
          ans.push_back('R');
        } else {
          for (int j = 1; j <= res[i]; j++) {
            ans.push_back('P');
          }
        }
      }
      cout << "SMALL" << endl;
      cout << cnt << endl;
      for (int i = cnt - 1; i >= 0; i--) cout << ans[i];
      return 0;
    } else {
      cout << "BIG" << endl;
      cout << sum << endl;
    }
  }
}
