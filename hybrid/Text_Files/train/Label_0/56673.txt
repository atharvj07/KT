import java.util.*;
import java.io.*;

public class Main {
    public static void main (String args []) throws Exception {
        BufferedReader in = new BufferedReader (new InputStreamReader (System.in));
        String a = in.readLine();
        StringBuilder ans = new StringBuilder();
        if (a.charAt(0)=='h') {
            ans.append("http://");
            a = a.substring(4);
        }
        else {
            ans.append("ftp://");
            a = a.substring(3);
        }
        int ind = a.lastIndexOf("ru");
        ans.append(a.substring(0, ind));
        ans.append(".ru");
        if (a.length()>ind+2) ans.append("/" + a.substring(ind+2));
        System.out.println (ans);
    }
}
