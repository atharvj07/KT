#include <iostream>
#include <algorithm>
using namespace std;

long n, a[101010];

int main(){
    cin >> n; for(int i = 0; i < n; ++i) cin >> a[i];
    sort(a, a+n);
    
    long ans = 0, sum = 0;
    for(int i = 0; i < n; ++i) {
        sum += a[i];
        ++ans;

        if(a[i+1] > sum*2) {
            ans = 0;
        }
    }
    cout << ans << endl;
    return 0;
}
