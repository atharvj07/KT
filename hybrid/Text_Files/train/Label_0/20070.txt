#include <bits/stdc++.h>
using namespace std;
const long long mxN = 1e5;
const long long mod = 1e9 + 7;
long long pow(long long a, long long b) {
  long long r = 1;
  while (b) {
    if (b % 2 != 0) r = (r * a) % mod;
    a = (a * a) % mod;
    b /= 2;
  }
  return r;
}
const int sz = 5e5 + 20;
long long fact[sz], inv[sz];
void pre() {
  fact[0] = 1;
  inv[0] = pow(fact[0], mod - 2);
  for (int i = 1; i < sz; i++) {
    fact[i] = (fact[i - 1] * i) % mod;
  }
}
int spf[10];
void sieve() {
  spf[1] = 1;
  for (int i = 2; i < mxN; i++) spf[i] = i;
  for (int i = 4; i < mxN; i += 2) spf[i] = 2;
  for (int i = 3; i * i < mxN; i++) {
    if (spf[i] == i) {
      for (int j = i * i; j < mxN; j += i) {
        if (spf[j] == j) spf[j] = i;
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int c[101];
    memset(c, 0, sizeof(c));
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      c[a]++;
    }
    bool f = 0;
    int ans = 0;
    for (int i = 0; i <= 100; i++) {
      if (c[i] == 1 && !f) {
        ans += i;
        f = 1;
      } else if (c[i] == 0) {
        if (!f) {
          ans += 2 * i;
          break;
        }
        ans += i;
        break;
      }
    }
    cout << ans << endl;
  }
}
