public class Solution {
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }

    public static void main(String[] args) throws IOException {
        FastReader sc = new FastReader();
        PrintWriter out = new PrintWriter(System.out);
        // ________________________________

        // out.println(output);
        // _______________________________

        int n = sc.nextInt();
        String s = sc.nextLine();
        String t = sc.nextLine();
        solver(n,s,t, out);
        // ________________________________
        out.flush();
    }

    public static void solver(int len,String s, String t, PrintWriter out) {
        // int countA = 0;
        // int countB = 0;
        Stack<Integer> ab = new Stack<>();
        Stack<Integer> ba = new Stack<>();
        for(int i=0;i<len;i++){
            if(s.charAt(i)=='a' && t.charAt(i)=='b'){
                ab.push(i+1);
            }else if(s.charAt(i)=='b' && t.charAt(i)=='a'){
                ba.push(i+1);
            }
        }
        int len1 = ab.size();
        int len2 = ba.size();
        if((len1+len2)%2!=0){
            out.println(-1);
            return;
        }
        if(len1%2!=0){
            out.println((len1+len2)/2+1);
        }
        else{
            out.println((len1+len2)/2);
        }
        while(ab.size()>1){
            int temp1 = ab.pop();
            int temp2= ab.pop();
            out.println(temp1+" "+temp2);
        }
        while(ba.size()>1){
            int temp1 = ba.pop();
            int temp2= ba.pop();
            out.println(temp1+" "+temp2);
        }
        if(ab.size()==1){
            int temp1 = ab.pop();
            int temp2 = ba.pop();
            out.println(temp1+" "+temp1);
            out.println(temp1+" "+temp2);
        }
    }
}
