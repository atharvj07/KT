n = int(input())
R = list(map(int, input().split()))
a = R.count(1)
b = R.count(2)
c = R.count(3)
if a * b * c > 0:
	print(min(a, b, c))
	while a > 0 and b > 0 and (c > 0):
		m = R.index(1)
		print(m + 1, end=' ')
		R[m] = 0
		a -= 1
		m = R.index(2)
		print(m + 1, end=' ')
		R[m] = 0
		b -= 1
		m = R.index(3)
		print(m + 1, end=' ')
		R[m] = 0
		c -= 1
		print()
else:
	print(0)
