import collections
import sys

def generate_primes(max_number):
	is_prime = [True] * (max_number + 1)
	is_prime[1] = False
	primes = []
	i = 2
	while i * i <= max_number:
		if is_prime[i]:
			primes.append(i)
			j = i * i
			while j <= max_number:
				is_prime[j] = False
				j += i
		i += 1
	return (primes, is_prime)

def get_bucket(n, primes, is_prime):
	bucket = 1
	for prime in primes:
		if n == 1:
			return bucket
		if is_prime[n]:
			bucket *= n
			break
		count = 0
		while n % prime == 0:
			n //= prime
			count += 1
		if count % 2 == 1:
			bucket *= prime
	return bucket
(primes, is_prime) = generate_primes(1000000)
for _ in range(int(input())):
	n = int(sys.stdin.readline())
	xs = map(int, sys.stdin.readline().split(' '))
	count_by_bucket1 = collections.defaultdict(int)
	for x in xs:
		bucket = get_bucket(x, primes, is_prime)
		count_by_bucket1[bucket] += 1
	max_bucket_size1 = max(count_by_bucket1.values())
	count_by_bucket2 = collections.defaultdict(int)
	for (bucket, count) in count_by_bucket1.items():
		if count % 2 == 0:
			new_bucket = 1
		else:
			new_bucket = bucket
		count_by_bucket2[new_bucket] += count
	max_bucket_size2 = max(count_by_bucket2.values())
	num_queries = int(sys.stdin.readline())
	for _ in range(num_queries):
		w = int(sys.stdin.readline())
		if w == 0:
			sys.stdout.write(f'{max_bucket_size1}\n')
		else:
			sys.stdout.write(f'{max_bucket_size2}\n')
