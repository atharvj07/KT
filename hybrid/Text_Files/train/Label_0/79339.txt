l = input().split()
suit = ['m', 'p', 's']
from collections import defaultdict
cnt = defaultdict(lambda : 0)
for i in range(3):
	cnt[l[i]] += 1
mini = 10
for i in suit:
	for j in range(1, 10):
		if j + 2 <= 9:
			cn = 0
			for k in range(3):
				cn += int('{}{}'.format(j + k, i) not in cnt)
			mini = min(mini, cn)
		mini = min(mini, 3 - cnt['{}{}'.format(j, i)])
print(mini)
