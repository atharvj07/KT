n = int(input())
a = [int(x) for x in input().split()]
a.sort()
id = n - 1
res = n // 2
b = [-1] * n
b[res] = a[id]
id -= 1
i = res - 1
j = res + 1
if n % 2 == 1:
	while i >= 0 and j < n:
		b[i] = a[id]
		id -= 1
		i -= 1
		b[j] = a[id]
		id -= 1
		j += 1
	print(*b)
else:
	while i >= 0 and j < n:
		b[i] = a[id]
		id -= 1
		i -= 1
		b[j] = a[id]
		id -= 1
		j += 1
	b[0] = a[0]
	print(*b)
