#include <bits/stdc++.h>
using namespace std;
const int N = 1010, inf = 1e9 + 2e6;
int n, m, W, ans, sb, sw, u, v, clr, res;
int dp[N][N], c[N], w[N], b[N], cb[N], cw[N];
vector<int> g[N], comp[N];
void dfs(int v) {
  c[v] = clr;
  comp[clr].push_back(v);
  sw += w[v], sb += b[v];
  for (auto u : g[v])
    if (!c[u]) dfs(u);
}
int main() {
  scanf("%d%d%d", &n, &m, &W);
  for (int i = 0; i < n; ++i) scanf("%d", &w[i]);
  for (int i = 0; i < n; ++i) scanf("%d", &b[i]);
  for (int i = 0; i < m; ++i) {
    scanf("%d%d", &u, &v), --u, --v;
    g[u].push_back(v), g[v].push_back(u);
  }
  for (int i = 0; i < N; ++i) dp[i][0] = dp[0][i] = 0;
  clr = 1;
  for (int i = 0; i < n; ++i)
    if (!c[i]) {
      sb = sw = 0;
      dfs(i);
      cw[clr] = sw, cb[clr] = sb;
      ++clr;
    }
  for (int i = 1; i <= W; ++i) {
    for (int j = 1; j < clr; ++j) {
      res = dp[i][j - 1];
      if (i >= cw[j]) res = max(res, dp[i - cw[j]][j - 1] + cb[j]);
      for (auto k : comp[j])
        if (w[k] <= i) res = max(res, dp[i - w[k]][j - 1] + b[k]);
      dp[i][j] = res;
    }
  }
  ans = dp[W][clr - 1];
  printf("%d\n", ans);
  return 0;
}
