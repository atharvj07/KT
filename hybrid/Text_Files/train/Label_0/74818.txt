N = int(input())
primes = [0] * (N + 1)
for i in range(2, N + 1):
	if primes[i] == 0:
		primes[i] = 1
		for j in range(i * 2, N + 1, i):
			primes[j] = -1
primes = [i for (i, p) in enumerate(primes) if p == 1]
res = [0] * (N + 1)
for p in primes:
	res[p] = 1
for p in primes:
	for j in range(p * 2, N + 1, p):
		if res[j] == 0:
			res[j] = max(p, j // p)
print(' '.join(map(str, sorted(res[2:]))))
