import sys, collections
(n, m, x, y) = map(int, sys.stdin.readline().split())
q = collections.deque()
vis = [[False] * (m + 1) for _ in range(n + 1)]
q.append((x, y))
vis[x][y] = True
dx = [0, 1, 0, -1]
dy = [1, 0, -1, 0]
cnt = 0
while q:
	(cx, cy) = q.popleft()
	print(cx, cy)
	find = False
	for i in range(4):
		if find:
			break
		(nx, ny) = (cx, cy)
		while True:
			(nx, ny) = (nx + dx[i], ny + dy[i])
			if 1 <= nx <= n and 1 <= ny <= m:
				if vis[nx][ny] == False:
					vis[nx][ny] = True
					q.append((nx, ny))
					find = True
					break
			else:
				break
