#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return ((a % b) == 0) ? b : gcd(b, a % b); }
int spf[20000001];
void sieve() {
  spf[1] = 1;
  for (int i = 2; i < 20000001; i++) spf[i] = i;
  for (int i = 4; i < 20000001; i += 2) spf[i] = 2;
  for (int i = 3; i * i < 20000001; i++) {
    if (spf[i] == i) {
      for (int j = i * i; j < 20000001; j += i)
        if (spf[j] == j) spf[j] = i;
    }
  }
}
set<int> getFactorization(int x) {
  set<int> ret;
  while (x != 1) {
    ret.insert(spf[x]);
    x = x / spf[x];
  }
  return ret;
}
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  sieve();
  int g = a[0];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    g = gcd(g, a[i]);
  }
  for (int i = 0; i < n; i++) {
    a[i] /= g;
  }
  map<int, int> dp;
  for (int i = 0; i < n; i++) {
    set<int> b = getFactorization(a[i]);
    for (auto j : b) {
      dp[j]++;
    }
  }
  int maxis = 0;
  for (auto i : dp) {
    maxis = max(i.second, maxis);
  }
  if (maxis == 0)
    cout << -1 << '\n';
  else
    cout << n - maxis << '\n';
  return 0;
}
