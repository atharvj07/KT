def isTwoColorable(gr, n, cols):
	cols[1] = 0
	(qu, ind) = ([1], 0)
	isBip = True
	while ind < len(qu):
		node = qu[ind]
		ind += 1
		for cnode in gr[node]:
			if cols[cnode] == -1:
				cols[cnode] = cols[node] ^ 1
				qu.append(cnode)
			else:
				isBip &= cols[cnode] != cols[node]
	return isBip

def solve(gr, n, v, t):
	if sum(v) % 2 != sum(t) % 2:
		return 'NO'
	cols = [-1] * (n + 1)
	if isTwoColorable(gr, n, cols):
		gsm = [0, 0]
		for i in range(1, n + 1):
			gsm[cols[i]] += t[i - 1] - v[i - 1]
		if gsm[0] == gsm[1]:
			return 'YES'
		return 'NO'
	return 'YES'
for case in range(int(input())):
	(n, m) = list(map(int, input().split()))
	V = list(map(int, input().split()))
	T = list(map(int, input().split()))
	gr = [[] for i in range(n + 1)]
	for ed in range(m):
		(u, v) = list(map(int, input().split()))
		gr[u].append(v)
		gr[v].append(u)
	ans = solve(gr, n, V, T)
	print(ans)
