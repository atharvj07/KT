const Int MOD = 1000000007;

Int mod_pow(Int x, Int n) {
    Int res = 1;
    while(n > 0) {
        if(n & 1) res = (res * x) % MOD; //ビット演算(最下位ビットが1のとき)
        x = (x * x) % MOD;
        n >>= 1; //右シフト(n = n >> 1)
    }
    return res;
}

//最大公約数
template<typename T>
T gcd(T a, T b) {
    return b != 0 ? gcd(b, a % b) : a;
}

//最小公倍数
template<typename T>
T lcm(T a, T b) {
    return a * b / gcd(a, b);
}

void fold(int N, int M, int d, int c, vector<vector<long long>> &field, int origin_x, int origin_y) {
    if (d == 1) {
        for (int x = origin_x + c; x < origin_x + c + c; x++) {
            for (int y = origin_y; y < origin_y + M; y++) {
                field[x][y] += field[origin_x + c - (x - origin_x - c + 1)][y];
            }
        }
    } else {
        for (int y = origin_y + c; y < origin_y + c + c; y++) {
            for (int x = origin_x; x < origin_x + N; x++) {
                field[x][y] += field[x][origin_y + c - (y - origin_y - c + 1)];
            }
        }
    }
}

vector<long long> solve(int N, int M, int T, int P) {
    //field[x][y] := 折り紙の左からx番目,　下からy番目のマス目(0-indexed)
    vector<vector<long long>> field(1000, vector<long long>(1000));
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            field[i][j] = 1;
        }
    }

    int origin_x = 0;
    int origin_y = 0;
    for (int i = 0; i < T; i++) {
        int d, c; cin >> d >> c;
        fold(N, M, d, c, field, origin_x, origin_y);
        if (d == 1) {
            origin_x += c;
        } else {
            origin_y += c;
        }
    }
    vector<long long> ret;
    for (int i = 0; i < P; i++) {
        int x, y; cin >> x >> y;
        ret.push_back(field[origin_x + x][origin_y + y]);
    }
    if (Debug) {
        for (int y = M-1; y >= 0; y--) {
            for (int x = 0; x < N; x++) {
                cout << field[x][y] << " ";
            }
            cout << endl;
        }
    }
    return ret;
}

int main(){
    cin.tie(0);
    ios::sync_with_stdio(false);

    vector<vector<long long>> ans;
    while (1) {
        int N, M, T, P; cin >> N >> M >> T >> P;
        if (!N && !M && !T && !P) break;
        ans.push_back(solve(N, M, T, P));
    }
    for (auto i: ans) {
        for (auto j: i) {
            cout << j << endl;
        }
    }
}
