from sys import stdin, stdout
input = stdin.readline
n = int(input())
a = list(map(int, input().split()))
mi = 0
pnt = 0
for i in range(1, n):
	cur = a[i] - a[i - 1]
	if cur > mi:
		mi = cur
		pnt = i
d = float('inf')
for i in range(1, n - 1):
	if i != pnt:
		tmp = max(mi, a[i + 1] - a[i - 1])
		d = min(tmp, d)
	else:
		d = min(a[i + 1] - a[i - 1], d)
stdout.write(str(d))
