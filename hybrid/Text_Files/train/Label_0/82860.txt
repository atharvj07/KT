import random

def er():
	print('NO')
	exit(0)

def ok():
	print('YES')
	exit(0)

def main():
	a = input()
	b = input()
	n = len(a)
	m = len(b)
	if n != m:
		er()
	if n == 1:
		if a == b:
			ok()
		else:
			er()
	x = 0
	y = 0
	for i in a:
		if i == '1':
			x += 1
	for i in b:
		if i == '1':
			y += 1
	if x == 0 and y != 0:
		er()
	if x != 0 and y == 0:
		er()
	ok()
main()
print('\n')
