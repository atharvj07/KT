import math
import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input().strip()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())
(n, z) = invr()
l = inlt()
dp = [[] for _ in range(200001)]
for each in l:
	k = each
	curr = 0
	while k > 0:
		dp[k].append(curr)
		k = k // 2
		curr += 1
ans = 10 ** 30
for i in range(len(dp)):
	dp[i] = sorted(dp[i])
for each in dp:
	if len(each) >= z:
		sm = 0
		for j in range(z):
			sm += each[j]
		ans = min(ans, sm)
print(ans)
