#include <bits/stdc++.h>
const long long INF = 1ll << 61;
using namespace std;
long long h[100055], p[100055];
int n, m;
void init() {
  memset(h, 0, sizeof(h));
  memset(p, 0, sizeof(p));
}
int cal(long long x) {
  int pos = 1;
  long long tmp;
  for (int i = 1; i <= n; i++) {
    if (abs(h[i] - p[pos]) > x) continue;
    if (h[i] == p[pos]) pos++;
    if (h[i] > p[pos])
      tmp =
          max(h[i] + x - 2 * (h[i] - p[pos]), h[i] + (x - (h[i] - p[pos])) / 2);
    else
      tmp = h[i] + x;
    while (p[pos] <= tmp && pos <= m) pos++;
  }
  return (pos > m);
}
int main() {
  while (scanf("%d %d", &n, &m) == 2) {
    init();
    for (int i = 1; i <= n; i++) scanf("%I64d", &h[i]);
    for (int i = 1; i <= m; i++) scanf("%I64d", &p[i]);
    long long l = -1, r = abs(h[1] - p[1]) * 2 + abs(h[1] - p[m]);
    long long mid;
    while (l + 1 < r) {
      long long mid = (l + r) >> 1;
      if (cal(mid))
        r = mid;
      else
        l = mid;
    }
    printf("%I64d\n", r);
  }
  return 0;
}
