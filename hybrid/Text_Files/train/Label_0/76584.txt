for _ in range(int(input())):
	(a, a1) = input().split()
	(b, b1) = input().split()
	(a, a1) = (int(a), int(a1))
	(b, b1) = (int(b), int(b1))
	minz = min(a1, b1)
	a1 = a1 - minz
	b1 = b1 - minz
	result = '>' if a1 >= 7 else '<' if b1 >= 7 else 0
	if not result:
		a *= 10 ** a1
		b *= 10 ** b1
		result = '>' if a > b else '<' if a < b else '='
	print(result)
