t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	ans = 0
	for i in range(n - 1):
		ans += abs(a[i] - a[i + 1])
	c = 0
	c = max(abs(a[1] - a[0]), c)
	c = max(c, abs(a[-1] - a[-2]))
	for i in range(1, n - 1):
		c = max(c, abs(a[i] - a[i + 1]) + abs(a[i] - a[i - 1]) - abs(a[i - 1] - a[i + 1]))
	print(ans - c)
