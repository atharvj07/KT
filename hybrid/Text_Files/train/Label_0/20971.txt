#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
typedef tuple<int, int, int> tiii;
typedef tuple<ll, ll, ll> tlll;
const int INF = 100000000;
const ll INFINITE = 4500000000000000000;

int dx[4] = { -1, 1, 0, 0 };
int dy[4] = { 0, 0, 1, -1 };

const int MOD = 1e9 + 7;

int main(void) {
	ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);

	int n; cin >> n;
	vector<int> c(n + 1);
	vector<int> pre(n + 1);
	int sum = 0;
	for (int i = 1; i <= n; i++) {
		cin >> c[i];
	}
	vector<int> b(n);
	for (int i = 1; i <= n - 1; i++) {
		cin >> b[i];
		pre[i] = pre[i - 1] + b[i];
	}
	int q; cin >> q;
	vector<int> x(q + 1);
	for (int i = 1; i <= q; i++) {
		cin >> x[i];
	}
	vector<int> limit(n + 1);
	limit[1] = x[1];
	for (int i = 2; i <= n; i++) {
		limit[i] = limit[i - 1] + x[1] + pre[i - 1];
	}
	vector<vector<ll>> world(n + 1, vector<ll>(10004));
	world[0][0] = 1;
	for (int i = 1; i <= n; i++) {
		for (int j = 0; j < 10004; j++) {
			if (world[i - 1][j] == 0) continue;
			for (int k = 0; k <= c[i]; k++) {
				if (j + k >= limit[i]) {
					world[i][j + k] += world[i - 1][j];
					world[i][j + k] %= MOD;
				}
			}
		}
	}
	ll ans = 0;
	for (int i = max(0, limit[n]); i < 10004; i++) {
		ans += world[n][i];
		ans %= MOD;
	}
	cout << ans;

}
