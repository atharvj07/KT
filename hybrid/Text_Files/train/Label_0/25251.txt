#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  long long kt = (n - 1) * n / 2;
  if (kt < k) {
    cout << "Impossible";
    return 0;
  }
  long long x = sqrt(2 * k);
  while ((x + 1) * x / 2 < k) x++;
  k -= (x - 1) * x / 2;
  if (k > 0) {
    cout << string(x, '(') << string(x - k, ')') << "()" + string(k, ')');
    n--;
  } else
    cout << string(x, '(') << string(x, ')');
  while (n > x) {
    cout << "()";
    n--;
  }
}
