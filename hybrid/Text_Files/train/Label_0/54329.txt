import java.util.*;
public class CurrencySystem {

	public static void main(String[] args) {
		Scanner obj = new Scanner(System.in) ;
		int n = obj.nextInt() ;
		Vector<Integer> v = new Vector<Integer>() ; 
		for(int i=0 ; i<n ; i++)
		{
			v.add(obj.nextInt()) ;
		}
		Collections.sort(v);
		if(v.elementAt(0)==1) System.out.println(-1);
		else System.out.println(1);

	}

}
