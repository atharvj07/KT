(x, s) = list(map(int, input().split()))
l = list(map(int, input().split()))
sum = 0
for i in l:
	sum += i
if sum < s:
	print(-1)
elif sum == s:
	print(0)
else:
	k = sum - len(l) * min(l)
	if k >= s:
		print(min(l))
	else:
		sum -= k
		s -= k
		ans = s // len(l)
		if s % len(l) != 0:
			ans += 1
		print(min(l) - ans)
