(n, m) = map(int, input().split())
d = {}
for i in range(m):
	(x, y, a) = map(int, input().split())
	d[x] = d.get(x, 0) - a
	d[y] = d.get(y, 0) + a
pos = []
neg = []
for i in d:
	if d[i] < 0:
		neg.append([d[i], i])
	elif d[i] > 0:
		pos.append([d[i], i])
ans = []
i = 0
j = 0
while i < len(neg) and j < len(pos):
	z = min(-neg[i][0], pos[j][0])
	ans.append([neg[i][1], pos[j][1], z])
	neg[i][0] += z
	pos[j][0] -= z
	if pos[j][0] == 0:
		j += 1
	if neg[i][0] == 0:
		i += 1
print(len(ans))
for i in ans:
	print(*i)
