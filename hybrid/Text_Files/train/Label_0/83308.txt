import sys
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	z = []
	z.append(a)
	z.append(b)
	sum1 = sum(a)
	ans = sys.maxsize
	sum2 = 0
	for i in range(n):
		sum1 -= z[0][i]
		ans = min(ans, max(sum1, sum2))
		sum2 += z[1][i]
	print(ans)
