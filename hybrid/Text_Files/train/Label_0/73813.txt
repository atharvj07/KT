t = int(input())
for _ in range(t):
	(a, b) = map(int, input().split())
	(a, b) = (min(a, b), max(a, b))
	n = a + b
	if n % 2 == 0:
		m = n // 2
		x = m - a
		y = n - x
		z = (y - x) // 2 + 1
		print(z)
		print(' '.join(map(str, range(x, y + 1, 2))))
	else:
		m = n // 2
		x = m - a
		y = n - x
		z = y - x + 1
		print(z)
		print(' '.join(map(str, range(x, y + 1))))
