#include <bits/stdc++.h>
using namespace std;
int n;
const int MAXN = 1e5 + 1;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  int a[n];
  int c[2 * MAXN] = {};
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0;
  for (int i = 0; i < n; i++)
    for (int j = i + 1; j < n; j++) c[a[i] + a[j]]++;
  for (int i = 0; i < 2 * MAXN; i++) ans = max(c[i], ans);
  cout << ans << "\n";
  return 0;
}
