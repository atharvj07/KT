def gcd(a, b):
	if a == 0:
		return b
	return gcd(b % a, a)

flag = 1
if k % 2 == 0:
	flag = k % 2
	k = k // 2
if n * m % k != 0:
	print('NO')
else:
	s = gcd(n, k)
	a = n // s
	k = k // s
	b = m // k
	if flag == 1:
		if a < n:
			a += a
		else:
			b += b
	print('YES')
