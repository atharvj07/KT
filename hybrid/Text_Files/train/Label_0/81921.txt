def main():
	(n, t) = map(int, input().split())
	a = list(map(int, input().split()))
	max_book = 0
	(j, i) = (0, 0)
	time = 0
	while i < n:
		temp = time + a[i]
		if temp <= t:
			time += a[i]
			i += 1
		else:
			max_book = max(max_book, i - j)
			j += 1
			time -= a[j - 1]
			if j >= n:
				break
	max_book = max(max_book, i - j)
	print(max_book)
main()
