#include <bits/stdc++.h>
using namespace std;
mt19937 rng32(chrono::steady_clock::now().time_since_epoch().count());
long long modExpo(long long x, long long n) {
  if (n == 0)
    return 1;
  else if (n % 2 == 0)
    return modExpo((x * x) % 1000000007, n / 2);
  else
    return (x * modExpo((x * x) % 1000000007, (n - 1) / 2)) % 1000000007;
}
long long modInverse(long long n) { return modExpo(n, 1000000007 - 2); }
long long n, m, k, dp[5005], ar[5005], br[5005];
pair<long long, long long> v[5005];
string s;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int ti = 1;
  while (ti--) {
    cin >> n >> m;
    for (long long i = 0; i < m; i++) {
      long long l, r;
      cin >> l >> r;
      v[i] = make_pair(l, r);
      ar[l]++;
      ar[r + 1]--;
    }
    long long ans = 0;
    for (long long i = 1; i < n + 1; i++) ar[i] += ar[i - 1];
    for (long long i = 1; i < n + 1; i++)
      if (ar[i] != 0) ans++;
    for (long long i = 1; i < n + 1; i++)
      if (ar[i] == 2) br[i] = 1;
    for (long long i = 1; i < n + 1; i++)
      if (ar[i] != 1) ar[i] = 0;
    for (long long i = 1; i < n + 1; i++)
      ar[i] += ar[i - 1], br[i] += br[i - 1];
    long long ans1 = 0;
    for (long long k = 0; k < m; k++) {
      for (long long l = k + 1; l < m; l++) {
        pair<long long, long long> i = v[k];
        pair<long long, long long> j = v[l];
        long long a = max(i.first, j.first);
        long long b = min(i.second, j.second);
        if (a > b) {
          ans1 = max(ans1, ans - (ar[i.second] - ar[i.first - 1]) -
                               (ar[j.second] - ar[j.first - 1]));
          continue;
        }
        ans1 = max(ans1, ans - (ar[i.second] - ar[i.first - 1]) -
                             (ar[j.second] - ar[j.first - 1]) -
                             (br[b] - br[a - 1]));
      }
    }
    cout << ans1;
  }
  return 0;
}
