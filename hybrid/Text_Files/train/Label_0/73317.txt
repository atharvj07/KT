import sys
from array import array
import typing as Tp

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def main():
	q = int(input())
	n = 500000
	m = 600
	a = array('i', [0]) * (n + 1)
	dp = [array('i', [0]) * (i + 1) for i in range(m + 1)]
	ans = []
	for _ in range(q):
		(t, x, y) = map(int, input().split())
		if t == 1:
			a[x] += y
			for i in range(1, m + 1):
				dp[i][x % i] += y
		elif x > m:
			ans.append(sum((a[i] for i in range(y, n + 1, x))))
		else:
			ans.append(dp[x][y])
	sys.stdout.buffer.write(('\n'.join(map(str, ans)) + '\n').encode('utf-8'))
main()
