#include <bits/stdc++.h>
using namespace std;
char q[150];
int a, b, c, d, e, f;
bool broj(string q) {
  for (int i = 0; i < q.length(); ++i) {
    if (q[i] >= '0' && q[i] <= '9') return true;
  }
  return false;
}
bool smal(string q) {
  for (int i = 0; i < q.length(); ++i) {
    if (q[i] >= 'a' && q[i] <= 'z') return true;
  }
  return false;
}
bool lage(string q) {
  for (int i = 0; i < q.length(); ++i) {
    if (q[i] >= 'A' && q[i] <= 'Z') return true;
  }
  return false;
}
int main() {
  scanf("%s", q);
  if (strlen(q) < 5 || !broj(q) || !smal(q) || !lage(q))
    printf("Too weak\n");
  else
    printf("Correct\n");
  return 0;
}
