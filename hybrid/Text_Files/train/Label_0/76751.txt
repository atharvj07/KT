a = input().split('|')
b = input()
x = len(a[0])
y = len(a[1])
z = len(b)
if x == y and z % 2 == 0:
	r = a[0] + b[:z // 2] + '|' + a[1] + b[z // 2:]
	print(r)
elif x == y and z % 2 != 0:
	print('Impossible')
elif x > y and x - y <= z and ((z - (x - y)) % 2 == 0):
	r = a[0] + b[:(z - (x - y)) // 2] + '|' + a[1] + b[(z - (x - y)) // 2:]
	print(r)
elif y > x and y - x <= z and ((z - (y - x)) % 2 == 0):
	r = a[0] + b[(z - (y - x)) // 2:] + '|' + a[1] + b[:(z - (y - x)) // 2]
	print(r)
else:
	print('Impossible')
