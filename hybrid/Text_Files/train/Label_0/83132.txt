l = int(input())
nums = [str(i) for i in input().split()]
non_beaut = -1
you = False
for i in range(l):
	if nums[i] == '0':
		print(0)
		you = True
		break
	else:
		s = '1'
		if nums[i] != '0' and nums[i] != s + (len(nums[i]) - 1) * '0':
			non_beaut = i
if you == False:
	length = ''
	for j in range(l):
		if j != non_beaut:
			length = length + nums[j][1:]
	if non_beaut == -1:
		result = '1' + length
	else:
		result = nums[non_beaut] + length
	print(result)
