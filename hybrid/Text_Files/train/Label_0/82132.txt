for _ in range(int(input())):
	n = int(input())
	p = list(map(int, input().split()))
	a = []
	f = 0
	for i in range(len(p) - 2):
		if p[i] < p[i + 1] and p[i + 1] > p[i + 2]:
			a.append([p.index(p[i]) + 1, p.index(p[i + 1]) + 1, p.index(p[i + 2]) + 1])
			f = 1
			break
	if f == 0:
		print('NO')
	else:
		print('YES')
		print(' '.join(map(str, a[0])))
