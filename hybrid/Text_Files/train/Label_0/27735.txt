#include <bits/stdc++.h>
using namespace std;
long long pwr(long long base, long long p, long long mod = (1000000007LL)) {
  long long ans = 1;
  while (p) {
    if (p & 1) ans = (ans * base) % mod;
    base = (base * base) % mod;
    p /= 2;
  }
  return ans;
}
long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int n, m;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> m;
  long long ans = 1;
  for (int j = 0; j < m; j++) ans = (ans * (2 * (n + 1))) % (1000000007LL);
  ans = (ans * (n + 1 - m)) % (1000000007LL);
  ans = (ans * pwr(n + 1, (1000000007LL) - 2)) % (1000000007LL);
  cout << ans;
  return 0;
}
