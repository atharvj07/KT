from math import sqrt

def check(ans, a):
	b = ans // a
	if (a + b) % 2:
		return False
	n = (a + b) // 2
	d = b - n
	if 1 <= d and d <= n:
		m = max(1, n // d - 1)
		for i in range(3):
			if n // m == d:
				print(n, m)
				return True
			m += 1
	else:
		return False

def main():
	T = int(input())
	while T:
		T -= 1
		ans = int(input())
		if ans == 0:
			print(1, 1)
			continue
		for i in range(1, int(sqrt(ans)) + 1):
			if ans % i == 0 and check(ans, i):
				break
		else:
			print(-1)
main()
