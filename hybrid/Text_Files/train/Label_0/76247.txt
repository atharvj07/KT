import sys
import math
from collections import defaultdict
dp = defaultdict(list)
dp[0] = [1, 0]
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
x = 0
ans = 0
for i in range(1, n + 1):
	x = x ^ arr[i - 1]
	if dp[x] == []:
		dp[x] = [0, 0]
		if i % 2 == 0:
			dp[x][0] += 1
		else:
			dp[x][1] += 1
	elif i % 2 == 0:
		ans += dp[x][0]
		dp[x][0] += 1
	else:
		ans += dp[x][1]
		dp[x][1] += 1
print(ans)
