#define RALL(a) (a).rbegin(), (a).rend()

#define FF first
#define SS second
template<class S, class T>
istream& operator>>(istream& is, pair<S,T>& p){
  return is >> p.FF >> p.SS;
}
template<class S, class T>
ostream& operator<<(ostream& os, const pair<S,T>& p){
  return os << p.FF << " " << p.SS;
}
template<class T>
void maxi(T& x, T y){
  if(x < y) x = y;
}
template<class T>
void mini(T& x, T y){
  if(x > y) x = y;
}


const double EPS = 1e-10;
const double PI  = acos(-1.0);
const LL MOD = 1e9+7;
const LL INF = 1e7;

int dp[210][210][2];
bool isl[210], isr[210];
string S;
int N;

int solve(int l, int r, int f){
  if(dp[l][r][f] != MOD) return dp[l][r][f];

  int& res = dp[l][r][f];
  res = f? -INF: INF;
  if(l+1 == r){
	if(!isl[l] && !isr[l])
	  res = S[l] - '0';
	return res;
  }

  bool pl = isl[l];
  bool pr = isr[r-1];
  isl[l] = isr[r-1] = false;
  for(int i=l+1;i<r-1;i+=2){
	if(isl[i-1] || isr[i+1]) continue;
	if(S[i] == '+'){
	  if(f)
		maxi(res, solve(l,i,1) + solve(i+1,r,1));
	  else
		mini(res, solve(l,i,0) + solve(i+1,r,0));
	}
	else if(S[i] == '-'){
	  if(f)
		maxi(res, solve(l,i,1) - solve(i+1,r,0));
	  else
		mini(res, solve(l,i,0) - solve(i+1,r,1));
	}
  }
  isl[l] = pl;
  isr[r-1] = pr;
  
  return res;
}


int main(){
  cin.tie(0);
  ios_base::sync_with_stdio(false);

  string tmp;
  cin >> tmp;

  int j = 0;
  for(int i=0;i<SZ(tmp);++i){
	if(tmp[i] == '('){
	  isl[j] = true;
	}
	else if(tmp[i] == ')'){
	  isr[j-1] = true;
	}
	else{
	  S += string(1, tmp[i]);
	  ++j;
	}
  }
  N = SZ(S);

  fill((int*)dp, (int*)dp+210*210*2, MOD);
  cout << solve(0, N, 1) << endl;

  return 0;
}
