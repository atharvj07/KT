def zerorem(n, k, array):
	a = []
	for y in array:
		if y % k != 0:
			a.append(k - y % k)
	if a == []:
		return 0
	maxcount = None
	maxcountn = None
	dic = {}
	for x in a:
		if x in dic:
			dic[x] += 1
		else:
			dic[x] = 1
		if maxcount == None:
			maxcount = dic[x]
			maxcountn = x
		if dic[x] > maxcount:
			maxcount = dic[x]
			maxcountn = x
		elif dic[x] == maxcount:
			if maxcountn < x:
				maxcountn = x
	if maxcount == 1:
		return max(a) + 1
	else:
		return (maxcount - 1) * k + maxcountn + 1
t = int(input())
a = []
for i in range(t):
	(n, k) = list(map(int, input().split(' ')))
	array = list(map(int, input().split(' ')))
	a.append([n, k, array])
for x in a:
	print(zerorem(*x))
