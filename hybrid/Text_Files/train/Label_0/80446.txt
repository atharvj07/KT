for _ in range(int(input())):
	(s, c) = map(str, input().split())
	k = list(s)
	y = len(s)
	if s < c:
		print(s)
	else:
		for i in range(y):
			t = s[i]
			p = 0
			for j in range(i + 1, y):
				if t >= s[j]:
					t = s[j]
					p = j
			if t != s[i]:
				(k[i], k[p]) = (k[p], k[i])
				break
		s = ''.join(k)
		if s < c:
			print(s)
		else:
			print('---')
