#include<bits/stdc++.h>
#define L long long
#define pb push_back
#define pi pair<int,int>
#define pii pair<pair<int,int>,int>
#define aa first
#define bb second
#define mp make_pair
#define mpp(a,b,c) mp(mp(a,b),c)
using namespace std;
int n,s,t,d,w[100010];
set<pii> f;
vector<int> x[100010],y[100010];
L p;
bool u[100010];
struct point
{
	int x,y,p;
}a[100010];
inline bool cmp(point a,point b)
{
	return a.x<b.x || a.x==b.x && a.y<b.y;
}
inline bool cmp2(point a,point b)
{
	return a.y<b.y || a.y==b.y && a.x<b.x;
}
inline void add(int i,int j,int k)
{
	x[i].pb(j);
	y[i].pb(k);
	x[j].pb(i);
	y[j].pb(k);
}
inline void dfs(int i)
{
	int j;
	u[i]=1;
	for(j=0;j<x[i].size();j++)
	  {
	   p+=y[i][j];
	   if(!u[x[i][j]])
	     dfs(x[i][j]);
	  }
}
int main()
{
	//freopen(".in","r",stdin);
	//freopen(".out","w",stdout);
	int i,j,k;
	scanf("%d%d%d",&n,&s,&t);
	for(i=1;i<=n;i++)
	  {
	   scanf("%d%d",&j,&k);
	   a[i].x=j+k;
	   a[i].y=j-k;
	   a[i].p=i;
	  }
	d=max(abs(a[s].x-a[t].x),abs(a[s].y-a[t].y));
	sort(a+1,a+n+1,cmp);
	for(i=1;i<=n;i++)
	  f.insert(mpp(a[i].x,a[i].y,i));
	f.insert(mpp(2.1e9,2.1e9,n+1));
	f.insert(mpp(-2.1e9,-2.1e9,0));
	for(i=1;i<=n;i++)
	  {
	   if((L)a[i].x-d>-2.1e9)
	   {
	   j=(*f.lower_bound(mpp(a[i].x-d,a[i].y-d,0))).bb;
	   k=(*--f.lower_bound(mpp(a[i].x-d,a[i].y+d,1e9))).bb;
	   if(j<=k)
	     {
	      w[j]++;
	      w[k]--;
	      add(a[i].p,a[j].p,1);
		 }
	   }
	   if((L)a[i].x+d<2.1e9)
	   {
	   j=(*f.lower_bound(mpp(a[i].x+d,a[i].y-d,0))).bb;
	   k=(*--f.lower_bound(mpp(a[i].x+d,a[i].y+d,1e9))).bb;
	   if(j<=k)
	     {
	      w[j]++;
	      w[k]--;
	      add(a[i].p,a[j].p,1);
		 }
	   }
	  }
	for(i=1,k=0;i<n;i++)
	  {
	   k+=w[i];
	   if(k)
	     add(a[i].p,a[i+1].p,k);
	   w[i]=0;
	  }
	f.clear();
	sort(a+1,a+n+1,cmp2);
	for(i=1;i<=n;i++)
	  f.insert(mpp(a[i].y,a[i].x,i));
	f.insert(mpp(2.1e9,2.1e9,n+1));
	f.insert(mpp(-2.1e9,-2.1e9,0));
	for(i=1;i<=n;i++)
	  {
	   if((L)a[i].x-d>-2.1e9)
	   {
	   j=(*f.lower_bound(mpp(a[i].y-d,a[i].x-d+1,0))).bb;
	   k=(*--f.lower_bound(mpp(a[i].y-d,a[i].x+d-1,1e9))).bb;
	   if(j<=k)
	     {
	      w[j]++;
	      w[k]--;
	      add(a[i].p,a[j].p,1);
		 }
	   }
	   if((L)a[i].x+d<2.1e9)
	   {
	   j=(*f.lower_bound(mpp(a[i].y+d,a[i].x-d+1,0))).bb;
	   k=(*--f.lower_bound(mpp(a[i].y+d,a[i].x+d-1,1e9))).bb;
	   if(j<=k)
	     {
	      w[j]++;
	      w[k]--;
	      add(a[i].p,a[j].p,1);
		 }
	   }
	  }
	for(i=1,k=0;i<n;i++)
	  {
	   k+=w[i];
	   if(k)
	     add(a[i].p,a[i+1].p,k);
	  }
	dfs(s);
	p/=4;
	printf("%lld\n",p);
	return 0;
}
