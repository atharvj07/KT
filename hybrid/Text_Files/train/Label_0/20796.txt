#include <bits/stdc++.h>
using namespace std;
int main(int argc, char** argv) {
  int n;
  cin >> n;
  int M[n];
  for (int i = 0; i < n; i++) {
    cin >> M[i];
  }
  int dem = 0;
  int max = 0;
  int a, b;
  int min = 50000000;
  for (int i = 0; i < n; i++) {
    if (M[i] > max) max = M[i], a = i;
    if (M[i] <= min) min = M[i], b = i;
  }
  a < b ? (cout << a + n - b - 1) : (cout << a + n - b - 2);
  return 0;
}
