    public static void main(String[] args) throws IOException {
        //-------------------------------------------------------------------------

        new C152P3();

        //put your code here... :)
        double y1 = nextDouble(),y2 = nextDouble(),yw = nextDouble(),xb = nextDouble(),yb = nextDouble(),r = nextDouble();
        
        int itr = 1000;
        
        double low = 0;
        double high = xb-0.00000001;
        
        while (itr-->0){
            double mid = (high + low) / 2;
            double mirror = 2*mid - xb;
            if(hitOrGreater(mirror,yb,mid,yw,r,y2)){
                low = mid;
            }else if(hitOrLower(mirror,yb,mid,yw,r,y1)){
                high=mid;
            }else{
                low = mid;
                break;
            }
        }
        double mirror = 2*low - xb;
        if(hitOrGreater(mirror,yb,low,yw,r,y2)){
            writeln("-1");
        }else if(hitOrLower(mirror,yb,low,yw,r,y1)){
            writeln("-1");
        }else{
            writeln(""+low);
        }
        writeln("");
        //-------------------------------------------------------------------------
        //closing up
        end();
        //--------------------------------------------------------------------------

    }
    private static boolean hitOrLower(double x1, double y1, double x2,
            double y2, double r, double l) {
        // TODO Auto-generated method stub
        y2 -= r;
        double center = y2 - x2*((y2-y1)/(x2-x1));
        if(l-center>0.000000001)return true;
//      Line2D.Double a = new Line2D.Double(x1, y1, x2, y2);
//      if(((a.ptLineDist(0, l))-r)<0.000000001)return true;
        if(r-Math.abs(l-center)>0.000000001)return true;
        return false;
    }
    private static boolean hitOrGreater(double x1, double y1, double x2,
            double y2, double r, double l) {
        // TODO Auto-generated method stub
        y2 -= r;
        double center = y2 - x2*((y2-y1)/(x2-x1));
        if(center-l>0.000000001)return true;
        Line2D.Double a = new Line2D.Double(x1, y1, x2, y2);
//      System.out.println((a.ptLineDist(0, l))-r);
        if(((a.ptLineDist(0, l))-r)<0.000000001)return true;
        return false;
    }

    private static int nextInt() throws IOException {
        if (!input.hasMoreTokens())input=new StringTokenizer(in.readLine());
        return Integer.parseInt(input.nextToken());
    }
    private static long nextLong() throws IOException {
        if (!input.hasMoreTokens())input=new StringTokenizer(in.readLine());
        return Long.parseLong(input.nextToken());
    }
    private static double nextDouble() throws IOException {
        if (!input.hasMoreTokens())input=new StringTokenizer(in.readLine());
        return Double.parseDouble(input.nextToken());
    }
    private static String nextString() throws IOException {
        if (!input.hasMoreTokens())input=new StringTokenizer(in.readLine());
        return input.nextToken();
    }
    private static void write(String output){
        out.write(output);
        out.flush();
    }
    private static void writeln(String output){
        out.write(output+"\n");
        out.flush();
    }
    private static void end() throws IOException{
        in.close();
        out.close();
        System.exit(0);
