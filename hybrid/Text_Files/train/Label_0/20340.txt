#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n, m, i, j, sum = 0, a;
  cin >> n >> m;
  for (i = 0; i < n; i++) {
    cin >> a;
    sum += a;
  }
  sum += (n - 1);
  if (sum > m || sum < m)
    cout << "NO\n";
  else
    cout << "YES\n";
  return 0;
}
