#include <bits/stdc++.h>
using namespace std;
template <class T>
T gcd(T a, T b) {
  return b ? gcd(b, a % b) : a;
}
const int maxn = 200200;
long long l[maxn];
long long r[maxn];
long long x, y, n, p;
double s;
void solve() {
  cin >> n >> p;
  for (int i = 0; i < n; i++) {
    cin >> x >> y;
    r[i] = y / p - (x - 1) / p;
    l[i] = y - x + 1;
  }
  for (int i = 0; i < n; i++) {
    int j = (i + 1) % n;
    s +=
        1.0 * (r[j] * l[i] + r[i] * l[j] - r[i] * r[j]) / (1.0 * (l[j] * l[i]));
  }
  printf("%.10lf", 2000.0 * s);
}
int main() {
  if (0) {
    freopen(
        "rare"
        ".in",
        "r", stdin);
    freopen(
        "rare"
        ".out",
        "w", stdout);
  }
  solve();
}
