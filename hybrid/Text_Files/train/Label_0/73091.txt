(s, e) = map(ord, input().split())
n = int(input())
c = [94, 62, 118, 60]
if n % 2 == 0:
	print('undefined')
else:
	si = c.index(s)
	ei = c.index(e)
	d = ei - si
	if d == -3 or d == 1:
		if n == 1:
			print('cw')
		elif int((n - 1) / 2) % 2 == 1:
			print('ccw')
		else:
			print('cw')
	elif d == 3 or d == -1:
		if n == 1:
			print('ccw')
		elif int((n - 1) / 2) % 2 == 1:
			print('cw')
		else:
			print('ccw')
