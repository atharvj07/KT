MOD = 10 ** 9 + 7
I = lambda : list(map(int, input().split()))
(n,) = I()
l = [int(i) for i in input()]
state = [[0] * 1000 for i in range(n)]
for i in range(n):
	(a, b) = I()
	s = l[i]
	k = 0
	for j in range(1000):
		if j == b + k:
			s ^= 1
			k += a
		state[i][j] = s
ans = 0
for i in range(200):
	s = 0
	for j in range(n):
		s += state[j][i]
	ans = max(ans, s)
print(ans)
