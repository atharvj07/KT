for ii in range(int(input())):
	(n, k) = map(int, input().split())
	a = [int(i) for i in input().split()]
	w = [int(i) for i in input().split()]
	a.sort()
	ans = 0
	for i in range(k):
		ans += a[n - i - 1]
	double = 0
	index = []
	for i in w:
		if i == 1:
			double += 1
		else:
			index.append(i - 1)
	for i in range(double):
		ans += a[n - i - 1]
	if index == []:
		print(ans)
	else:
		index.sort()
		index.reverse()
		ans += a[0]
		index.pop()
		l = 0
		for i in index:
			l += i
			ans += a[l]
		print(ans)
