for _ in range(int(input())):
	N = int(input())
	s = [i for i in input()]
	t = [i for i in input()]
	charCount = [0] * 26
	for i in s:
		charCount[ord(i) - ord('a')] += 1
	for i in t:
		charCount[ord(i) - ord('a')] += 1
	flag = True
	for i in charCount:
		if i & 1:
			flag = False
			break
	ans = []
	if flag:
		print('Yes')
		for i in range(N - 1):
			if s[i] == t[i]:
				continue
			else:
				go = True
				for ss in range(i + 1, N):
					if s[ss] == s[i]:
						ans.append((ss + 1, i + 1))
						(t[i], s[ss]) = (s[ss], t[i])
						go = False
						break
				if go:
					for tt in range(i + 1, N):
						if t[tt] == s[i]:
							ans.append((tt + 1, tt + 1))
							(t[tt], s[tt]) = (s[tt], s[tt])
							ans.append((tt + 1, i + 1))
							(s[tt], t[i]) = (t[i], s[tt])
							break
		print(len(ans))
		for x in ans:
			print(*x)
	else:
		print('No')
