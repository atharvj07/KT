def power(q, k):
	answer = q
	for i in range(1, k):
		answer *= q
	return answer

def main():
	k = int(input())
	answer = 0
	q = []
	for i in range(1, 500000):
		k5 = 0
		z = 5
		k5 += i // z
		while z * 5 <= i:
			z *= 5
			k5 += i // z
		if k5 == k:
			answer += 1
			q.append(i)
	print(answer)
	print(*q)
main()
