(n, k) = map(int, input().split(' '))
s = list(map(int, input().split(' ')))
den_map = {}
for si in s:
	den_map[si] = den_map[si] + 1 if si in den_map else 1
nums = list(den_map.keys())
nums.sort(key=lambda num: den_map[num], reverse=True)
fst = 0
lst = 200001
tl = 0
while fst < lst:
	mdl = (fst + lst) // 2
	can = False
	kn = 0
	for num in nums:
		kn += den_map[num] // mdl
		if kn >= k:
			can = True
			tl = max(tl, mdl)
			break
	if can:
		fst = mdl + 1
	else:
		lst = mdl
t = []
for num in nums:
	count = min(k, den_map[num] // tl)
	t.extend([num] * count)
	k -= count
	if k == 0:
		break
print(*t)
