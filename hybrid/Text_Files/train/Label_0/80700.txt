nbTest = int(input())
for k in range(nbTest):
	n = int(input())
	s = input()
	compteur = 0
	sol = 0
	for c in s:
		if c == 'T':
			compteur += 1
			if compteur > n // 3:
				print('NO')
				sol = -1
				break
		else:
			compteur -= 1
			if compteur < 0:
				print('NO')
				sol = -1
				break
	if sol == 0:
		if compteur == n // 3:
			print('YES')
		else:
			print('NO')
