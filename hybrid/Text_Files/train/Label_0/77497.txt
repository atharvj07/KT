(n, m) = map(int, input().split())
a = [list(input()) for i in range(n)]
d = set(['f', 'a', 'c', 'e'])
k = 0
for i in range(n - 1):
	for j in range(m - 1):
		if a[i][j] in d:
			s = set()
			s.add(a[i][j])
			s.add(a[i + 1][j])
			s.add(a[i + 1][j + 1])
			s.add(a[i][j + 1])
			if s == d:
				k += 1
print(k)
