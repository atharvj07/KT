def gcd(a, b):
	return gcd(b, a % b) if b else a
while 1:
	try:
		(x, y) = map(int, input().split())
	except:
		break
	if gcd(x, y) != 1:
		print('Impossible')
		continue
	res = ''
	while x > 0 and y > 0:
		if x > y:
			(cnt, give) = (x // y, 'A')
			x %= y
		else:
			(cnt, give) = (y // x, 'B')
			y %= x
		if x == 0 or y == 0:
			cnt -= 1
		res += str(cnt) + give
	print(res)
