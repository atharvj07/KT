import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Map;
import java.util.TreeMap;

public class String2 {


	
	static class SuffixAutomaton {

		int[] link, len;
		long[] ways;
		TreeMap<Character, Integer>[] next;
		int lst, idx;
		
		SuffixAutomaton(char[] s)
		{
			int n = s.length;
			link = new int[n<<1];
			len = new int[n<<1];
			ways = new long[n<<1];
			next = new TreeMap[n<<1];
			next[0] = new TreeMap<>();
			Arrays.fill(ways, -1);
			for(char c: s)
				addLetter(c);
		}
		long dfs(int idx) {
			if(ways[idx]!=-1)
				return ways[idx];
			long way =0;
			for (Map.Entry<Character, Integer> entry : next[idx].entrySet()) {
				Character c = entry.getKey();
				Integer x = entry.getValue();
				if(c=='$'&&idx!=0)
					way++;
				else {
					way+= dfs(x);
				}
			}
			return ways[idx]=way;
		}	
		void traverse() {
			dfs(0);
		}
		void addLetter(char c)
		{
			int cur = ++idx, p = lst;
			//Keep adding to the suffix links of the last a link to the new cur
			while(!next[p].containsKey(c)) { next[p].put(c, cur); p = link[p]; }
			//this is the case where i am say adding a unique letter 
			int q = next[p].get(c);
			//if i am not already done by checking that the last suffix link 
			//is actually pointing at me
			if(q != cur)
				if(len[q] == len[p] + 1)
					link[cur] = q;
				else
				{
					int clone = ++idx;
					len[clone] = len[p] + 1;
					link[clone] = link[q];
					next[clone] = new TreeMap<>(next[q]);
					link[cur] = link[q] = clone;
					while(next[p].get(c) == q) { next[p].put(c, clone);	p = link[p]; }
				}
			len[cur] = len[lst] + 1;
			next[cur] = new TreeMap<>();
			lst = cur;
		}
	}


public static void main(String[] args) throws IOException{
	BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
	String x = in.readLine()+"$";
	SuffixAutomaton sa = new SuffixAutomaton(x.toCharArray());
	sa.traverse();
	long ans = 0;

	for (int i = 0; i < sa.ways.length; i++) {
		long cur = sa.ways[i];
		ans+=(cur)*(cur+1)/2*(sa.len[i]-sa.len[sa.link[i]]);
	}
	System.out.println(ans);
}
}
