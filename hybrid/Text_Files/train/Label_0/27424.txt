#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll in()
{
    ll re=0,f=1;char x=getchar();
    while(x<'0'||x>'9'){if(x=='-')f=-1;x=getchar();}
    while(x>='0'&&x<='9')re=re*10+x-48,x=getchar();
    return re*f;
}
void out(ll x)
{
    if(x<0)putchar('-'),x=-x;
    if(!x){putchar('0');return;}
    char z[21],ct=0;
    while(x){z[++ct]=x%10+48,x/=10;}
    while(ct){putchar(z[ct]);--ct;}
}
struct node{
    int ls,rs;
    ll mx,tag;
};
struct segtree{
    int rt,ptr;
    node nd[1000010];
    void pushdown(int c)
    {
    	int L=nd[c].ls;
    	int R=nd[c].rs;
    	nd[L].mx+=nd[c].tag;
		nd[R].mx+=nd[c].tag; 
    	nd[L].tag+=nd[c].tag;
    	nd[R].tag+=nd[c].tag;
    	nd[c].tag=0;
	}
    void build(int &c,int l,int r)
    {
        c=++ptr;
        if(l==r)
        {
            nd[c].mx=l-1;
            return;
        }
        int mid=l+r>>1;
        build(nd[c].ls,l,mid);
        build(nd[c].rs,mid+1,r);
        nd[c].mx=max(nd[nd[c].ls].mx,nd[nd[c].rs].mx);
    }
    void modify(int c,int l,int r,int ql,int qr,ll v)
    {
		if(l==ql&&r==qr)
		{
		   nd[c].mx+=v;
		   nd[c].tag+=v;
		   return;
		}
		pushdown(c);
		int mid=l+r>>1;
		if(qr<=mid)modify(nd[c].ls,l,mid,ql,qr,v);
		else if(ql>mid)modify(nd[c].rs,mid+1,r,ql,qr,v);
		else
		{
			modify(nd[c].ls,l,mid,ql,mid,v);
			modify(nd[c].rs,mid+1,r,mid+1,qr,v);
		}
		nd[c].mx=max(nd[nd[c].ls].mx,nd[nd[c].rs].mx);
    }
    ll query(int c,int l,int r,int ql,int qr)
    {
		if(l==ql&&r==qr)return nd[c].mx;
		pushdown(c);
		int mid=l+r>>1;
		if(qr<=mid)return query(nd[c].ls,l,mid,ql,qr);
		if(ql>mid)return query(nd[c].rs,mid+1,r,ql,qr);
		return max(query(nd[c].ls,l,mid,ql,mid),query(nd[c].rs,mid+1,r,mid+1,qr));
	}
}tr;
int n,Q;
int main()
{
	n=in();Q=in();
	tr.build(tr.rt,1,n);
	ll mxr=0;
	for(int i=1;i<=Q;++i)
	{
		ll l=in(),r=in(),w=in();
		mxr=max(mxr,r);
		tr.modify(tr.rt,1,n,l,r,w);
		out(tr.query(tr.rt,1,n,1,mxr));
		putchar('\n');
	}
}
