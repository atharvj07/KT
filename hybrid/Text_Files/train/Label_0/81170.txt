import sys
input = sys.stdin.readline

def printL(list):
	sys.stdout.write(' '.join(map(str, list)) + '\n')

def printN(n):
	sys.stdout.write(str(n) + '\n')
t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	b = False
	for i in ('aa', 'aba', 'aca', 'abca', 'acba', 'accabba', 'abbacca'):
		if i in s:
			printN(len(i))
			b = True
			break
	if not b:
		printN(-1)
