(n, k) = map(int, input().split())
S = list(input())
i = 0
if k == 0 and len(S) == 1:
	print(*S)
	exit()
if len(S) == 1:
	print('0')
	exit()
while k != 0:
	if i >= len(S):
		print(*S, sep='')
		exit()
	if int(S[0]) > 1:
		S[0] = 1
		k -= 1
	if i > 0:
		if int(S[i]) > 0:
			S[i] = 0
			k -= 1
	i += 1
print(*S, sep='')
