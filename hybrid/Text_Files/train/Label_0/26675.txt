#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main(){
	int n, ans=0;
	cin >> n;
	string s;
	cin >> s;
	for(int i = 1; i < n-1; i++){
		int pre = 0;
		bool son[30][2]={false};
		for(int j = 0; j < i; j++)son[s[j]-'a'][0]=true;
		for(int j = i; j < n; j++)son[s[j]-'a'][1]=true;
		for(int j = 0; j < 30; j++)if(son[j][0] && son[j][1])pre++;
		ans = max(ans, pre);
	}
	cout << ans << endl;
    return 0;
}
