from random import randint
n = int(input())
(l, r) = (-2 * 10 ** 9, 2 * 10 ** 9)
opposite = {'<': '>=', '<=': '>', '>': '<=', '>=': '<'}
for _ in range(n):
	(rel, num, ans) = input().split()
	num = int(num)
	if ans == 'N':
		rel = opposite[rel]
	if rel == '<':
		r = min(r, num - 1)
	elif rel == '<=':
		r = min(r, num)
	elif rel == '>':
		l = max(l, num + 1)
	elif rel == '>=':
		l = max(l, num)
if l > r:
	print('Impossible')
else:
	print(randint(l, r))
