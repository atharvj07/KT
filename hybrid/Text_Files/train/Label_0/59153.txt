import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayDeque;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Random;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class ArrayDestruction {
	public static void main(String[] args) {
		FastScanner scan=new FastScanner();
		PrintWriter out=new PrintWriter(System.out);

		/*
		the problem's solved!!
		we must take the max elem in the array in the first pass
		otherwise we will never be able to take it
		since x will transform to some elem <= the max elem
		making it impossible to ever take.

		so the solution is!
		take the max elem and check that the next biggest and one other sum to that
		so on and so forth
		we can skip exactly one elem in this sequence and instead take it first with the max elem.
		so we brute force all of em to skip!
		*/

		f=new int[1000001];
		
		int t=scan.nextInt();
		for(int tt=0;tt<t;tt++) {
			int n=scan.nextInt()*2;
			
			int[] a=new int[n];

			int[][] moves=new int[2][n/2];
			for(int[] m:moves) Arrays.fill(m,-1);

			for(int i=0;i<n;i++) {
				a[i]=scan.nextInt();
				f[a[i]]++;
			}
			ruffleSort(a);
			boolean wegood=false;

			t:for(int s=n-2;s>=0;s--) {
				TreeMap<Integer,Integer> took=new TreeMap<>();
				for(int i:a) took.put(i,0);
				moves[0][0]=a[n-1];
				moves[1][0]=a[s];
				int target=a[n-1];
				f[a[s]]--;
				f[a[n-1]]--;
				int ct=1;

				for(int i=n-2;i>=0;i--) {
					int cur=a[i];
					if(f[cur]-took.get(cur)==0||s==i) continue;
					int l=f[target-cur];
					if((target-cur!=cur&&l>=1)||(target-cur==cur&&l>1)) {
						moves[0][ct]=cur;
						moves[1][ct]=target-cur;
						f[cur]--;
						f[target-cur]--;
						ct++;
						target=Math.max(cur,target-cur);
					}
					else {
						for(int j=0;j<ct;j++) {
							f[moves[0][j]]++;
							f[moves[1][j]]++;
						}
						continue t;
					}
				}
				int x=a[s]+a[n-1];
				out.println("YES");
				out.println(x);
				for(int j=0;j<n/2;j++) {
					out.print(moves[0][j]);
					out.print(" ");
					out.println(moves[1][j]);
				}
				wegood=true;
				break;
			}
			if(!wegood) {
				out.println("NO");
			}
			
			Arrays.fill(f,0);
		}
		out.close();
	}
	static int[] f;
	static void ruffleSort(int[] a) {
		Random get = new Random();
		for (int i = 0; i < a.length; i++) {
			int r = get.nextInt(a.length);
			int temp = a[i];
			a[i] = a[r];
			a[r] = temp;
		}
		Arrays.sort(a);
	}
	static class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner() {
			try	{
				br = new BufferedReader(new InputStreamReader(System.in));
				st = new StringTokenizer(br.readLine());
			} catch (Exception e){e.printStackTrace();}
		}

		public String next() {
			if (st.hasMoreTokens())	return st.nextToken();
			try {st = new StringTokenizer(br.readLine());}
			catch (Exception e) {e.printStackTrace();}
			return st.nextToken();
		}

		public int nextInt() {return Integer.parseInt(next());}

		public long nextLong() {return Long.parseLong(next());}

		public double nextDouble() {return Double.parseDouble(next());}

		public String nextLine() {
			String line = "";
			if(st.hasMoreTokens()) line = st.nextToken();
			else try {return br.readLine();}catch(IOException e){e.printStackTrace();}
			while(st.hasMoreTokens()) line += " "+st.nextToken();
			return line;
		}
	}
}
