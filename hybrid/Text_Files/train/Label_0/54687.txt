class FastScanner {

    BufferedReader br;
    StringTokenizer st;

    FastScanner(InputStreamReader reader) {
        br = new BufferedReader(reader);
        st = new StringTokenizer("");
    }

    String next() {
        while (!st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (Exception ex) {
                return null;
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }

    double nextDouble() {
        return Double.parseDouble(next());
    }

    String nextLine() {
        try {
            return br.readLine();
        } catch (IOException ex) {
            return null;
        }
    }
}

class Solver {

    void aser(boolean OK) {
        if (!OK) {
            Main.MLE();
        }
    }

    FastScanner sc;
    PrintWriter out;
    static PrintStream err = System.err;

 
    
    void solve() {
        int n, m, k;
        boolean[] usedOrg;
        int[] cntGoodOrg;
        ArrayList<Integer>[] gr;
        
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        usedOrg = new boolean[n + 1];
        fill(usedOrg,true);
        for (int i = 0; i < k; i++) {
            usedOrg[sc.nextInt()] = false;
        }
        
        gr = new ArrayList[n + 1];
        for (int v = 1; v <= n; v++) {
            gr[v] = new ArrayList<>();
        }
        for (int iter = 0; iter < m; iter++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            gr[a].add(b);
            gr[b].add(a);
        }
        
        cntGoodOrg = new int[n + 1];
        for (int v = 1; v <= n; v++) {
            if (usedOrg[v]) {
                for (int to : gr[v]) {
                    if (usedOrg[to]) {
                        ++cntGoodOrg[v];
                    }
                }
            }
        }
        
        long ansUp = 0, ansDown = 1;
        int[] cntGood = cntGoodOrg.clone();

        TreeSet<Integer> ts;
        ts = new TreeSet<>( (a,b) -> {
            int res = Long.compare( (long)cntGood[a] * gr[b].size(), (long)cntGood[b] * gr[a].size() );
            if( res != 0 )
                return res;
            return a - b;
        } );

        for (int v = 1; v <= n; v++) {
            if(usedOrg[v])
                ts.add(v);
        }

        ArrayList<Integer> ans = new ArrayList<>();
        while(!ts.isEmpty()){
            int v = ts.first();

            if(ansUp * gr[v].size() < cntGood[v] * ansDown ){
                ansUp = cntGood[v];
                ansDown = gr[v].size();
                ans.clear();
            }
            ans.add(v);
            
            ts.pollFirst();
            ArrayList<Integer> arr = new ArrayList<>();
            for(int to : gr[v])
                if( ts.remove(to) )
                    arr.add(to);

            for (Integer to : arr) {
                --cntGood[to];
            }

            for (Integer to : arr) {
                ts.add(to);
            }
        }
        
        StringBuilder sb = new StringBuilder();
        sb.append(ans.size()).append("\n");
        for (Integer v : ans) {
            sb.append(v).append("\n");
        }
        out.println(sb);
        
    }
}
