import math

def dst(p1, p2):
	(x1, y1) = p1
	(x2, y2) = p2
	result = math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
	return result

def min_dst(a, b, t, bs):
	bottles = [(dst(t, x), dst(a, x), dst(b, x)) for x in bs]
	full_dst = sum((2 * bot[0] for bot in bottles))
	min_a = full_dst - bottles[0][0] + bottles[0][1]
	min_b = full_dst - bottles[0][0] + bottles[0][2]
	min_ab = min(min_a, min_b)
	for i in range(1, len(bottles)):
		old_min_a = min_a
		old_min_b = min_b
		new_min_a = full_dst - bottles[i][0] + bottles[i][1]
		if new_min_a < min_a:
			min_a = new_min_a
		new_min_b = full_dst - bottles[i][0] + bottles[i][2]
		if new_min_b < min_b:
			min_b = new_min_b
		new_min_ab = min(old_min_b - bottles[i][0] + bottles[i][1], old_min_a - bottles[i][0] + bottles[i][2])
		if new_min_ab < min_ab:
			min_ab = new_min_ab
	return min(min_ab, min(min_a, min_b))
(ax, ay, bx, by, tx, ty) = map(int, input().split())
l = int(input())
bottles = [0] * l
for i in range(l):
	bottles[i] = tuple(map(int, input().split()))
print(min_dst((ax, ay), (bx, by), (tx, ty), bottles))
