#include <iostream>
#include <cstring>
using namespace std;

typedef long long ll;

int h, w, x, y, d;
ll n;
ll dp[102][102][4];
char t[102][102];

int dx[] = {0, 1, 0, -1};
int dy[] = {-1, 0, 1, 0};
char str[] = "NESW";

void walk(){
  for(int i = 0; i < 4; i++){
    int nd = (d + i) % 4;
    int nx = x + dx[nd];
    int ny = y + dy[nd];

    if(0 <= nx && nx < w && 0 <= ny && ny < h && t[ny][nx] == '.'){
      x = nx;
      y = ny;
      d = nd;
      return;
    }
  }
}

void solve(){
  ll cnt = 0;
  memset(dp, -1, sizeof(dp));

  while(dp[x][y][d] == -1 && cnt < n){
    dp[x][y][d] = cnt;
    walk();
    cnt++;
  }

  if(cnt == n) return;

  ll rem = (n - dp[x][y][d]) % (cnt - dp[x][y][d]);

  while(rem--){
    walk();
  }
}

int main(){
  while(cin >> h >> w >> n, h || w || n){
    for(int i = 0; i < h; i++){
      cin >> t[i];

      for(int j = 0; j < w; j++){
        if(t[i][j] != '.' && t[i][j] != '#'){
          if(t[i][j] == 'N') d = 0;
          else if(t[i][j] == 'E') d = 1;
          else if(t[i][j] == 'S') d = 2;
          else if(t[i][j] == 'W') d = 3;

          x = j;
          y = i;
          t[i][j] = '.';
        }
      }
    }

    solve();

    cout << y + 1 << " " << x + 1 << " " << str[d] << endl;
  }
}
