#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int> > v;
vector<int> ans;
int main() {
  int n, m;
  cin >> n >> m;
  v.resize(m);
  for (int i = 0; i < m; i++) cin >> v[i].first >> v[i].second;
  for (int num = 1; num <= 101; num++) {
    bool ff = false;
    for (int k = 1; k <= 101; k++) {
      if (n - 1 >= (num - 1) * k && num * k > n - 1) {
        bool f = true;
        for (int i = 0; i < m; i++) {
          if (v[i].first - 1 >= k * (v[i].second - 1) &&
              v[i].first - 1 < k * v[i].second)
            continue;
          f = false;
          break;
        }
        if (f) {
          ff = true;
          break;
        }
      }
    }
    if (ff) ans.push_back(num);
  }
  if (ans.size() == 1)
    cout << ans[0] << '\n';
  else
    cout << "-1\n";
  return 0;
}
