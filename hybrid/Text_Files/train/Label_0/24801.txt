#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Edge {int to, id;};

vector<vector<Edge>> g;
vector<int> ans;

void dfs(int v, int c=-1, int p=-1){
 int k=1;
 for(int i=0; i<g[v].size(); i++){
  int u = g[v][i].to , ei= g[v][i].id;
  if(u==p) continue;
  if(k==c) k++;
  ans[ei] = k; k++;
  dfs(u,ans[ei],v);
 }
}

int main(){
  int n;
  cin >> n;
  g.resize(n);
  ans = vector<int>(n-1);
  for(int i=0; i<n-1; i++){
    int x,y;
    cin >> x >> y;
    x--; y--;
    g[x].push_back((Edge){y,i});
    g[y].push_back((Edge){x,i});
    
  }
  
  dfs(0);
  int mx = 0;
  for(int i=0; i<n; i++) mx = max(mx,int(g[i].size()));
  cout << mx << endl;
  
  for(int i=0; i<n-1; i++) cout << ans[i] << endl;
  
  return 0;
  
}
