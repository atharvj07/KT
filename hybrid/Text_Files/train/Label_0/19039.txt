#include<bits/stdc++.h>
#define r(i,n) for(int i=0;i<n;i++)
using namespace std;
int h,w,sum,cnt;
int dx[]={0,0,1,-1};
int dy[]={1,-1,0,0};
bool used[99][99];
string s[99];
void dfs(int y,int x){
  used[y][x]=1;
  if(s[y][x]=='*')cnt++;
  if(cnt==sum)return;
  r(i,4){
    int yy=y+dy[i];
    int xx=x+dx[i];
    if(yy<0||xx<0||yy>=h||xx>=w)continue;
    if(used[yy][xx]||s[yy][xx]=='.')continue;
    dfs(yy,xx);
    if(cnt==sum)return;
  }
}
int main(){
  cin>>h>>w;
  r(i,h)cin>>s[i];
  r(i,h)r(j,w)if(((i==0||i==h-1)||(i%2==0&&j!=w/2)||(i%2&&i%4==3&&(j==w/2-1||j==w/2+1))||(i%2&&i%4==1&&(j==0||j==w-1)))&&s[i][j]=='.')s[i][j]='#';

  r(i,h)r(j,w)if(s[i][j]=='*')sum++;

  r(i,h)r(j,w)if(s[i][j]=='@')dfs(i,j);

  r(i,h)r(j,w)if(s[i][j]=='#'&&!used[i][j])s[i][j]='.';

  r(i,h)cout<<s[i]<<endl;
}
