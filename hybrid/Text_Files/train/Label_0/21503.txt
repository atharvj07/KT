#include<iostream>
#include<algorithm>

using namespace std;


int main()
{
	int n, m, t;
	int a[2001] = { 0 };

	ios::sync_with_stdio(false);

	cin >> n;
	for (int i = 0; i != n; ++i) 
	{
		cin>>t;
		for (int j = 2000; j != 0; --j)
		{
			if (a[j])
			{
				a[j + t] = 1;
			}
		}
		a[t] = 1;
	}

	cin>>m;
	for (int i = 0; i != m;++i)
	{
		cin >> t;
		cout << (a[t]? "yes" : "no") << endl;
	}

	return 0;

}
