import sys
input = sys.stdin.readline
from math import *
a = int(input())
for x in range(a):
	(a, b, c) = map(int, input().split())
	d = list(map(int, input().split()))
	e = list(map(int, input().split()))
	j = []
	for y in range(c):
		j.append([d[y], e[y]])
	j.sort()
	for y in range(c):
		p = ceil(j[y][1] / a)
		b -= p * j[y][0]
	if b + j[-1][0] > 0:
		print('YES')
	else:
		print('NO')
