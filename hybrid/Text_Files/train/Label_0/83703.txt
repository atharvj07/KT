from math import log2, ceil
(n, k) = map(int, input().split())
k *= 8
a = sorted(list(map(int, input().split())))
(q1, dif) = (0, 1)
ans = float('inf')
for q in range(n):
	while n * ceil(log2(dif)) <= k and q1 < n:
		if q1 == n - 1 or a[q1] != a[q1 + 1]:
			dif += 1
		q1 += 1
	ans = min(ans, q + n - q1)
	if q != n - 1 and a[q] != a[q + 1]:
		dif -= 1
print(ans)
