#include <bits/stdc++.h>
using namespace std;
const int limit = 1000010;
const int maxn = 77777;
long long A[maxn];
int sg[limit], use[maxn];
int main() {
  long long a = 0, b = 0;
  sg[0] = 0;
  for (int i = 1; i < limit; ++i) {
    while (b * b <= i && b < i) {
      ++use[sg[b++]];
    }
    while (a * a * a * a < i) {
      --use[sg[a++]];
    }
    sg[i] = 0;
    while (use[sg[i]]) ++sg[i];
  }
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%I64d", &A[i]);
  }
  memset(use, 0, sizeof(use));
  sort(A, A + n);
  a = b = 0;
  int result = 0;
  for (int i = 0; i < n; ++i) {
    while (b * b <= A[i] && b < A[i]) {
      ++use[sg[b++]];
    }
    while (a * a * a * a < A[i]) {
      --use[sg[a++]];
    }
    int csg = 0;
    while (use[csg]) ++csg;
    result ^= csg;
  }
  puts(result ? "Furlo" : "Rublo");
  return 0;
}
