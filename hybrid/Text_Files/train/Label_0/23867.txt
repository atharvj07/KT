#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100 * 100 * 10 * 3 + 10;
int main() {
  string s;
  cin >> s;
  int in1, in2;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'e') in2 = i;
    if (s[i] == '.') in1 = i;
  }
  int a = 0;
  for (int j = in2 + 1; j < s.size(); j++) a = a * 10 + (s[j] - '0');
  char ch[1000];
  int t = 0, x = 0;
  while (s[t] != 'e') ch[x++] = s[t++];
  for (int i = x; i < x + a + 1; i++) ch[i] = '0';
  for (int i = 1; i < 1 + a; i++) swap(ch[i], ch[i + 1]);
  int i = 0;
  while (ch[i] == '0' and ch[i + 1] != '.') ch[i++] = 'x';
  i = a + x;
  while (ch[i] == '0') ch[i--] = 'x';
  if (ch[i] == '.') ch[i] = 'x';
  for (int i = 0; i < a + x + 1; i++)
    if (ch[i] != 'x') cout << ch[i];
  return 0;
}
