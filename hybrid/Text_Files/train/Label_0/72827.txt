from collections import Counter, defaultdict, deque
from math import *
import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	arr = [int(x) for x in input().split()]
	mx = 10 ** 6 + 2
	lens = [0] * mx
	ans = 0
	for el in arr:
		lens[el] += 1
	for i in range(mx - 1, 0, -1):
		if lens[i] & 1:
			if lens[i - 1] > 0:
				lens[i - 1] += 1
			lens[i] -= 1
	rem = 0
	for i in range(mx - 1, 0, -1):
		cur = lens[i]
		if cur == 0:
			continue
		if rem != 0:
			ans += rem * i
			cur -= 2
			rem = 0
		ans += cur // 4 * i * i
		if cur & 2:
			rem = i
	print(ans)
tt = 1
for test in range(tt):
	solve()
