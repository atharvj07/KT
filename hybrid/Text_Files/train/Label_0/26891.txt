#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
using namespace std;
const double eps = 1e-6;

double max_r(const vector<int> &xs, const vector<int> &ys, double x, double y){
    double r_max = 0;
    for(int i=0;i<xs.size();i++){
        r_max = max(r_max, hypot(x - xs[i], y - ys[i]));
    }
    return r_max;
}

double min_circle_r(const vector<int> &xs, const vector<int> &ys, double x){
    double lo = -1000, hi = 1000;
    for(int i=0;i<70;i++){
        double side = (hi - lo) / 3;
        double a = lo + side;
        double b = hi - side;
    
        double fa = max_r(xs, ys, x, a);
        double fb = max_r(xs, ys, x, b);
        
        if(fa < fb){
            hi = b;
        } else {
            lo = a;
        }
    }
    
    return max_r(xs, ys, x, lo);
}

double min_circle_r(const vector<int> &xs, const vector<int> &ys){
    double lo = -1000, hi = 1000;
    for(int i=0;i<70;i++){
        double side = (hi - lo) / 3;
        double a = lo + side;
        double b = hi - side;
    
        double fa = min_circle_r(xs, ys, a);
        double fb = min_circle_r(xs, ys, b);
        
        if(fa < fb){
            hi = b;
        } else {
            lo = a;
        }
    }
    
    return min_circle_r(xs, ys, lo);
}


int n, m;
int r[111];

double R[111];
bool used[111];

bool greedy_check(int s){
    vector<int> rs;
    for(int i=0;i<n;i++)if(!used[i]){
        rs.push_back(r[i]);
    }
    
    vector<double> Rs;
    for(int i=s;i<m;i++){
        Rs.push_back(R[i]);
    }
    
    sort(rs.begin(), rs.end());
    sort(Rs.begin(), Rs.end());
    
    int last = -1;
    for(int i=0;i<Rs.size();i++){
        bool found = false;
        for(int j=last+1;j<rs.size();j++){
            if(Rs[i] < rs[j] + eps){
                found = true;
                last = j;
                break;
            }
        }
        if(!found)return false;
    }
    
    return true;
}


int main(){
    cin >> n >> m;
    for(int i=0;i<n;i++)cin >> r[i];

    for(int i=0;i<m;i++){
        int p;
        cin >> p;
        
        vector<int> xs(p), ys(p);
        for(int j=0;j<p;j++){
            cin >> xs[j] >> ys[j];
        }
        
        R[i] = min_circle_r(xs, ys);
    }
    
    vector<int> res(m);
    for(int i=0;i<m;i++){
        bool found = false;
        for(int j=0;j<n;j++)if(!used[j] && R[i] < r[j] + eps){
            used[j] = true;
            if(greedy_check(i + 1)){
                res[i] = j;
                found = true;
                break;
            }
            used[j] = false;
        }
        if(!found){
            cout << "NG" << endl;
            return 0;
        }
    }
    
    for(int i=0;i<m;i++)cout << res[i] + 1 << endl;
    
    return 0;
}
