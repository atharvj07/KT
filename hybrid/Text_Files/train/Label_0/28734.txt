#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1);
char primes[12500002];
void preCal() {
  memset(primes, 85, sizeof primes);
  primes[0] = 83;
  for (int i = 3; i < 10000; i += 2)
    if (!((primes[i >> 3]) & (1 << (i & 7))))
      for (int j = i * i; j < 100000000; j += i) primes[j >> 3] |= 1 << (j & 7);
}
int main() {
  preCal();
  int n, d, u, v;
  cin >> n;
  bool mach[100005];
  memset(mach, false, sizeof(mach));
  vector<vector<int> > ans1;
  for (int i = n; i >= 2; i--) {
    if ((!((primes[(i) >> 3]) & (1 << ((i)&7))))) {
      vector<int> v(1, i);
      for (int j = 2; i * j <= n; j++)
        if (!mach[i * j]) {
          mach[i * j] = true;
          v.push_back(i * j);
        }
      if (v.size() < 2) continue;
      if (v.size() % 2 == 1 && v.size() > 1) {
        mach[v[1]] = false;
        v.erase(v.begin() + 1);
      }
      ans1.push_back(v);
    }
  }
  int res1 = 0;
  for (int i = 0; i < ans1.size(); i++) res1 += ans1[i].size() / 2;
  cout << res1 << endl;
  for (int i = 0; i < ans1.size(); i++)
    for (int j = 0; j < ans1[i].size(); j += 2)
      cout << ans1[i][j] << " " << ans1[i][j + 1] << endl;
}
