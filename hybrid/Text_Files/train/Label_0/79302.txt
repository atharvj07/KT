amount = int(input())
array = [int(s) for s in input().split()]
counter = [0] * (2 * 10 ** 5 + 1)
for i in range(len(array)):
	counter[array[i]] += 1
idx = -1
max_counter = -1
for i in range(len(counter)):
	if counter[i] > max_counter:
		max_counter = counter[i]
		idx = i
indices = []
for i in range(len(array)):
	if array[i] == idx:
		indices.append(i)
print(len(array) - len(indices))
if indices[0] != 0:
	indices.insert(0, 0)
ans = ''
for i in range(1, len(indices)):
	for j in range(indices[i], indices[i - 1], -1):
		if idx > array[j - 1]:
			ans += '1' + ' ' + str(j) + ' ' + str(j + 1) + '\n'
		elif idx < array[j - 1]:
			ans += '2' + ' ' + str(j) + ' ' + str(j + 1) + '\n'
if indices[-1] != len(array) - 1:
	for j in range(indices[-1], len(array) - 1):
		if idx > array[j + 1]:
			ans += '1' + ' ' + str(j + 2) + ' ' + str(j + 1) + '\n'
		elif idx < array[j + 1]:
			ans += '2' + ' ' + str(j + 2) + ' ' + str(j + 1) + '\n'
print(ans)
