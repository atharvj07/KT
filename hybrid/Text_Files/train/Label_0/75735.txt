years = {}
for k in range(int(input())):
	(b, d) = map(int, input().split())
	if b in years:
		years[b] += 1
	else:
		years[b] = 1
	if d in years:
		years[d] -= 1
	else:
		years[d] = -1
keys = sorted(years.keys())
y = 0
c = 0
cur = 0
for year in keys:
	cur += years[year]
	if cur > c:
		y = year
		c = cur
print(y, c)
