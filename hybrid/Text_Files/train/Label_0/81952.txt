try:
	t = int(input())
	for i in range(t):
		(n, m, k) = map(int, input().split())
		a = []
		for j in range(n):
			x = input()
			a.append(list(x))
		done = {}
		possible = 1
		for j in range(n - 1, -1, -1):
			for j2 in range(m - 1, -1, -1):
				if a[j][j2] == '*':
					count = 0
					for i2 in range(j - 1, -1, -1):
						if j2 + count + 1 < m and j2 - count - 1 >= 0:
							if a[i2][j2 + count + 1] == '*' and a[i2][j2 - count - 1] == '*':
								count += 1
							else:
								break
					if count >= k:
						done[j, j2] = 1
						count = 0
						for i2 in range(j - 1, -1, -1):
							if j2 + count + 1 < m and j2 - count - 1 >= 0:
								if a[i2][j2 + count + 1] == '*' and a[i2][j2 - count - 1] == '*':
									count += 1
									done[i2, j2 + count] = 1
									done[i2, j2 - count] = 1
								else:
									break
					if done.setdefault((j, j2), 0) == 0:
						possible = 0
						break
			if possible == 0:
				break
		if possible == 1:
			print('YES')
		else:
			print('NO')
except:
	pass
