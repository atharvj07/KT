import java.util.Scanner;


public class Main {
	
	public static void main(String args[]){
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int[] a = new int[n];
		int[] b = new int[n];
		for (int i = 0; i < n - 1; i++) {
			a[i] = in.nextInt();
		}
		for (int i = 0; i < n - 1; i++) {
			b[i] = in.nextInt();
		}
		int[] t = new int[n];
		boolean ans = false;
		for (int first = 0; first < 4; first++) {
			t[0] = first;
			for (int i = 1; i < n; i++) {
				boolean flag = false;
				int nxt;
				for (nxt = 0; nxt < 4; nxt++) {
					if (((nxt | t[i - 1]) == a[i - 1]) && ((nxt & t[i - 1]) == b[i - 1])){
						flag = true;
						break;
					}
				}
				if (flag) {
					t[i] = nxt;
				} else {
					ans = false;
					break;
				}
				if (flag && (i == n - 1)) {
					System.out.println("YES");
					for (int j = 0; j < n; j++) {
						System.out.print(t[j] + " ");
					}
					return;
				}
			}
		}
		System.out.println("NO");
	}

}
