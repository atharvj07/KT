try:
	n = int(input())
	for i in range(n):
		count = 0
		sub = 0
		u = int(input())
		b = []
		for j in range(u):
			o = str(input())
			b.append(o)
		d = set(b)
		for r in d:
			sub = sub + (b.count(r) - 1)
		count = count + sub
		print(count)
		for (er, gh) in enumerate(b):
			if b.count(gh) == 1:
				continue
			else:
				for x in '0123456789':
					new = x + gh[1:]
					if new in b:
						continue
					else:
						b[er] = new
						break
		print(*b, sep='\n')
except Exception as e:
	pass
