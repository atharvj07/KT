def main():
	from sys import stdin, stdout
	n = int(stdin.readline())
	from itertools import chain
	c1 = tuple(map(int, input().split()))
	c2 = tuple(map(int, input().split()))
	c3 = tuple(map(int, input().split()))
	tree = tuple(([] for _ in range(n)))
	for _ in range(n - 1):
		(u, v) = map(int, input().split())
		u -= 1
		v -= 1
		tree[u].append(v)
		tree[v].append(u)
		if len(tree[u]) > 2 or len(tree[v]) > 2:
			print('-1')
			break
	else:
		for u in range(n):
			if len(tree[u]) < 2:
				v = u
				break
		ans = [0] * n
		ansi = 1
		cost = tuple(([0, 0, 0] for _ in range(4)))
		w = 0
		for _ in range(n):
			ans[v] = ansi
			cost[1][ansi] += c1[v]
			cost[2][ansi] += c2[v]
			cost[3][ansi] += c3[v]
			ansi = (ansi + 1) % 3
			if len(tree[v]) == 1:
				w = v
				(v,) = tree[v]
			else:
				(v, w) = (next((x for x in tree[v] if x != w)), v)
		pal = min(((p1, p2, p3, cost[p1][0] + cost[p2][1] + cost[p3][2]) for p1 in (1, 2, 3) for p2 in (1, 2, 3) if p1 != p2 for p3 in (6 - p1 - p2,)), key=lambda x: x[3])
		print(pal[3])
		print(*(pal[ansi] for ansi in ans))
main()
