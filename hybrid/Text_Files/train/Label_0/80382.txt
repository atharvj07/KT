def is_prime(adad):
	if adad == 2:
		return True
	else:
		saquare = int(adad ** 0.5) + 1
		counter = 2
		while counter < saquare:
			if adad % counter != 0:
				counter += 1
			else:
				return False
		return True
vurudi_asli = int(input())
prime_check = True
zarib = 1
while prime_check:
	vurudi = vurudi_asli * zarib + 1
	if is_prime(vurudi) == True:
		zarib += 1
	else:
		print(zarib)
		prime_check = False
