import java.util.Arrays;
import java.util.*;
public class ozon
{
	public static void main (String[] args)
	 {
		Scanner sc=new Scanner(System.in); 
		long t,i;
		long n=sc.nextInt();
		long m=sc.nextInt();
		if(n<3&&m==0)
		{
			if(n==1)
				System.out.print(1);
			else
				System.out.print(1+" "+2);
			return;
		}
		else if(n<3&&m!=0)
		{
			System.out.println(-1);
			return;
		}
		ArrayList<Integer> ans=new ArrayList();
		while(m>0)
		{
			long vo=1;
			i=1;
			while(m>vo)
			{
				vo=i*i;
				i++;
			}
			i-=2;
			long ve=(vo + i*i)/2;
			vo=i*i;
			i=i*2+1;
			long dev=m-ve;
			if(dev>=0)
			{
				i++;
				m-=ve;
			}
			else
				m-=vo;
			if(ve==1)
			{
				i++;
				i=i*2+1;
			}
			for(int j=1;j<=i;j++)
				ans.add(j);
			if(m==0)
				break;
			else
			{
				int c=ans.get(ans.size()-1-(int)m);
				int d=ans.get(ans.size()-(int)m);
				ans.add(c+d);
			}
			break;
		}
		int v;
		if(ans.isEmpty())
			v=1;
		else
			v=ans.get(ans.size()-1);
		i=3;
		while(ans.size()<n)
		{
			if(ans.isEmpty())
				ans.add(1);
			
			int g=v*(int)i;
			ans.add(g);
			i+=2;
		}
		if(ans.size()>n)
		{
			System.out.println(-1);
			return;
		}
		else
		{
			for(i=0;i<ans.size();i++)
				System.out.print(ans.get((int)i)+" ");
		}
	}
}
