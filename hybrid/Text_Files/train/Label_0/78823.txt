import sys
inf = float('inf')
from math import ceil, floor, log, sqrt, factorial, pow, pi, gcd

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
(q, k) = get_ints()
drinks = ceil(q / 2)
mydict = dict()
for i in range(q):
	z = int(input())
	mydict[z] = mydict.get(z, 0) + 1
count = 0
for i in range(drinks):
	c = -1
	maxi = -1
	for i in mydict:
		if mydict[i] > maxi:
			c = i
			maxi = mydict[i]
	if mydict[c] > 2:
		mydict[c] -= 2
		count += 2
	else:
		count += mydict[c]
		del mydict[c]
print(count)
