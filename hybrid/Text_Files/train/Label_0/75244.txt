N = int(input())
a = list(map(int, input().split()))
rec = ['+'] * N
P = [0] * N
for i in range(N):
	P[a[i] - 1] = i
for i in range(N, 0, -1):
	j = P[i - 1]
	rec[j] = 'B'
	for k in range(j % i, N, i):
		if j != k and a[j] <= a[k] and (rec[k] == 'B'):
			rec[j] = 'A'
			break
print(''.join(map(str, rec)))
