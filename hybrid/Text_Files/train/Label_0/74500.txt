import sys, math, cmath, time
start_time = time.time()
tt = 0

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	return input().strip()

def invr():
	return map(int, input().split())

def solve():
	a = insr()
	q = dict()
	c = 0
	(x, y) = (0, 0)
	for i in range(len(a)):
		w = str(x) + ' ' + str(y)
		if a[i] == 'N':
			y = y + 1
		if a[i] == 'S':
			y = y - 1
		if a[i] == 'E':
			x = x + 1
		if a[i] == 'W':
			x = x - 1
		if q.get(w + ' ' + str(x) + ' ' + str(y)):
			c = c + 1
		else:
			c = c + 5
		q[w + ' ' + str(x) + ' ' + str(y)] = 1
		q[str(x) + ' ' + str(y) + ' ' + w] = 1
	print(c)

def main():
	if not ONLINE_JUDGE:
		sys.stdin = open('input.txt', 'r')
		sys.stdout = open('output.txt', 'w')
	t = 1
	t = inp()
	for tt in range(t):
		solve()
	if not ONLINE_JUDGE:
		print('Time Elapsed :', time.time() - start_time, 'seconds')
	sys.stdout.close()

def counter(a):
	q = [0] * max(a)
	for i in range(len(a)):
		q[a[i] - 1] = q[a[i] - 1] + 1
	return q

def string_counter(a):
	q = [0] * 26
	for i in range(len(a)):
		q[ord(a[i]) - 97] = q[ord(a[i]) - 97] + 1
	return q
