#include <bits/stdc++.h>
using namespace std;
const int OO = 1e9 + 7;
const int MOD = 1e9 + 7;
const int N = 100 + 7;
char g[N][N];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int a, b, c, d, n;
  cin >> a >> b >> c >> d >> n;
  int e;
  int i, j;
  i = 0;
  if (b >= d) {
    if (d % 2 == 0) {
      j = 0;
      e = 1;
    } else {
      j = (a + c) - 1;
      e = -1;
    }
  } else if (d > b) {
    if (b % 2 == 1) {
      j = 0;
      e = 1;
    } else {
      j = (a + c) - 1;
      e = -1;
    }
  }
  char f = 'a';
  for (int k = 0; k <= n - 1; k++) {
    int x;
    cin >> x;
    while (x--) {
      if (i >= max(b, d)) break;
      if (b >= d) {
        if (i < d)
          g[i][j] = f;
        else if (j < a)
          g[i][j] = f;
        else {
          g[i][j] = '.';
          x++;
        }
      } else if (d > b) {
        if (i < b)
          g[i][j] = f;
        else if (j < a) {
          g[i][j] = '.';
          x++;
        } else
          g[i][j] = f;
      }
      if (i % 2 == 0 && e == 1) {
        if (j < (a + c) - 1)
          j++;
        else
          i++;
      } else if (i % 2 == 0 && e == -1) {
        if (j > 0)
          j--;
        else
          i++;
      } else if (i % 2 == 1 && e == 1) {
        if (j > 0)
          j--;
        else
          i++;
      } else if (i % 2 == 1 && e == -1) {
        if (j < (a + c) - 1)
          j++;
        else
          i++;
      }
    }
    f++;
  }
  for (int i = 0; i <= max(b, d) - 1; i++)
    for (int j = 0; j <= (c + a) - 1; j++)
      if (g[i][j] == 0) g[i][j] = '.';
  cout << "YES" << endl;
  for (int i = 0; i <= max(b, d) - 1; i++) {
    for (int j = 0; j <= (c + a) - 1; j++) cout << g[i][j];
    cout << endl;
  }
  return 0;
}
