import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Task3 {
    int n;
    int l;
    int[] flags;

    public static void main(String[] args) throws IOException {
        new Task3().solve();
    }

    private void solve() throws IOException {
        BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);

        int t = Integer.parseInt(f.readLine());
        for (int t1 = 0; t1 < t; t1++) {
            StringTokenizer tokenizer = new StringTokenizer(f.readLine());
            n = Integer.parseInt(tokenizer.nextToken());
            l = Integer.parseInt(tokenizer.nextToken());
            flags = new int[n];
            tokenizer = new StringTokenizer(f.readLine());
            for (int i = 0; i < n; i++) flags[i] = Integer.parseInt(tokenizer.nextToken());

            out.println(ternary());
        }

        out.close();
    }

    private double ternary() {
        double from = 0;
        double to = l;

        double res = -1;
        while (to - from > 1e-6) {
            double first = from + (to -from) / 3;
            double last = to - (to - from) / 3;
            double[] firstTry = tryMeeting(first);
            double[] lastTry = tryMeeting(last);

            //System.out.println(Arrays.toString(firstTry) + " " + Arrays.toString(lastTry));
            if (firstTry[2] < lastTry[2]) {
                to = last;
                res = firstTry[0];
            } else {
                from = first;
                res = lastTry[0];
            }
        }

        return res;
    }

    private double[] tryMeeting(double meeting) {
        double[] times = new double[3];

        double curr1 = 0;
        double speed1 = 1;
        for (int i = 0; i < n; i++, speed1++) {
            if (flags[i] >= meeting) break;
            times[0] += (flags[i] - curr1) / speed1;
            curr1 = flags[i];
        }
        times[0] += (meeting - curr1) / speed1;

        double curr2 = l;
        double speed2 = 1;
        for (int i = n - 1; i >= 0; i--, speed2++) {
            if (flags[i] <= meeting) break;
            times[1] += (curr2 - flags[i]) / speed2;
            curr2 = flags[i];
        }
        times[1] += (curr2 - meeting) / speed2;

        times[2] = Math.abs(times[1] - times[0]);

        //System.out.println(meeting + " " + Arrays.toString(times));
        return times;
    }
}
