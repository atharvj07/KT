(p, q) = map(int, input().split())
temp = int(0)
line = list(map(int, input().split()))
for i in range(len(line)):
	temp += int(line[i])
	if temp < q:
		print(int(0), end=' ')
	elif temp == q:
		print(int(1), end=' ')
		temp -= q
	else:
		ans = int(temp // q)
		temp -= int(q * ans)
		print(ans, end=' ')
