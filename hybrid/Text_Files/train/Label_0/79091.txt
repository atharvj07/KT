x = int(input())
possib = []
for i in range(1, x * 2):
	t = input().split()
	for j in range(len(t)):
		possib.append([int(t[j]), i + 1, j + 1])
times = [''] * (x * 2)
cont = x * 2
possib.sort(reverse=True)
i = 0
while cont:
	maior = possib[i]
	time = (possib[i][1], possib[i][2])
	t1 = str(possib[i][1])
	t2 = str(possib[i][2])
	if t1 not in times and t2 not in times:
		times[possib[i][1] - 1] = t2
		times[possib[i][2] - 1] = t1
		cont -= 2
	i += 1
print(' '.join(times) + '\n')
