		InputReader in = new InputReader(inputStream);

class TaskE {

    private long n, answer;
    private int sqrt2, sqrt3;
    private ArrayList<Long> divisiors;

    public void solve(int testNumber, InputReader in, PrintWriter out) {
        n = in.nextLong();
        if (n % 3 != 0) {
            out.println("0");
            return;
        }
        answer = 0;
        n /= 3;
        sqrt2 = 1 + (int) Math.sqrt(n);
        sqrt3 = 1 + (int) Math.pow(n, 1.0 / 3);
        divisiors = new ArrayList<Long>();
        calculate();
        int length = divisiors.size();
        for (int i = 0; i < length && divisiors.get(i) < sqrt3; ++i) {
            long d = n / divisiors.get(i);
            int sqrt = 1 + (int) Math.sqrt(d);
            for (int j = i; j < length && divisiors.get(j) < sqrt; ++j) {
                if (d % divisiors.get(j) == 0) {
                    long c = d / divisiors.get(j);
                    if (c < divisiors.get(i) + divisiors.get(j) && (c + divisiors.get(i) + divisiors.get(j)) % 2 == 0) {
                        if (divisiors.get(i) == c && divisiors.get(j) == c)
                            ++answer;
                        else if (divisiors.get(i) == divisiors.get(j) || divisiors.get(j) == c)
                            answer += 3;
                        else
                            answer += 6;
                    }
                }
            }
        }
        out.println(answer);
    }

    private void calculate() {
        for (int i = 1; i < sqrt2; ++i)
            if (n % i == 0) {
                divisiors.add((long) i);
                long j = n / i;
                if (j != i)
                    divisiors.add(j);
            }
        Collections.sort(divisiors);
    }

}

class InputReader {

    private static BufferedReader bufferedReader;
    private static StringTokenizer stringTokenizer;

    public InputReader(InputStream inputStream) {
        bufferedReader = new BufferedReader(new InputStreamReader(inputStream));
        stringTokenizer = null;
    }

    public String next() {
        while(stringTokenizer == null || !stringTokenizer.hasMoreTokens()) {
            try {
                stringTokenizer = new StringTokenizer(bufferedReader.readLine());
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return stringTokenizer.nextToken();
    }

    public long nextLong() {
        return Long.parseLong(next());
    }

    }
