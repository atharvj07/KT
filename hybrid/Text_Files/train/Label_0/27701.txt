#include <bits/stdc++.h>
using namespace std;
int n;
int x, y;
int gcd(int x1, int y1) {
  if (y1 == 0) {
    return x1;
  }
  if (x1 != y1) {
    return gcd(y1, x1 % y1);
  } else {
    return x1;
  }
}
int sqrt3(int point) {
  int left, right, mid;
  left = 1;
  right = 1000;
  while (left <= right) {
    mid = (left + right) / 2;
    if (mid * mid * mid < point) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }
  return left;
}
void solve() {
  if (x < y) {
    int tmp = x;
    x = y;
    y = tmp;
  }
  int fac = gcd(x, y);
  int res1 = x / fac;
  int res2 = y / fac;
  int mul = res1 * res2;
  if ((mul <= fac) && (fac % mul == 0)) {
    int divRes = fac / mul;
    int res3 = sqrt3(divRes);
    if (res3 * res3 * res3 == divRes) {
      printf("Yes\n");
    } else {
      printf("No\n");
    }
  } else {
    printf("No\n");
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d %d", &x, &y);
    solve();
  }
  return 0;
}
