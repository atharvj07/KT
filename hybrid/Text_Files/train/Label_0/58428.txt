import java.io.*;
import java.util.*;

public class HelloWorld {
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }

    static HashMap<Character, Integer> CharacterWeight = new HashMap<>();

    public static void main(String[] args) throws IOException {
        FastReader scn = new FastReader();
        OutputStream out = new BufferedOutputStream(System.out);
        int n = scn.nextInt();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++)
            strings[i] = scn.nextLine();
        HashSet<Character> allCharacters = new HashSet<>();
        HashSet<Character> leadingCharacters = new HashSet<>();
        for (String string : strings) {
            for (int i = 0; i < string.length(); i++) {
                if (i == 0)
                    leadingCharacters.add(string.charAt(0));
                allCharacters.add(string.charAt(i));
            }
        }
        for (Character ch : allCharacters) {
            for (String string : strings) {
                for (int i = 0; i < string.length(); i++) {
                    if (string.charAt(i) == ch) {
                        int weight = (int) Math.pow(10, string.length() - 1 - i);
                        if (CharacterWeight.containsKey(ch))
                            CharacterWeight.put(ch, CharacterWeight.get(ch) + weight);
                        else
                            CharacterWeight.put(ch, weight);
                    }
                }
            }
        }
        ArrayList<Character> charactersInOrder = new ArrayList<>();
        for (Character character : allCharacters)
            charactersInOrder.add(character);
        Collections.sort(charactersInOrder, new CharacterCompare());
        if (leadingCharacters.size() != allCharacters.size()) {
            for (Character character : charactersInOrder) {
                if (!leadingCharacters.contains(character)) {
                    charactersInOrder.remove(character);
                    break;
                }
            }
        }
        int ans = 0;
        for(int i = 0; i < charactersInOrder.size(); i++) {
            ans += CharacterWeight.get(charactersInOrder.get(i)) * (i + 1);
        }
        print(out, ans);
        out.flush();
    }

    static class CharacterCompare implements Comparator<Character> {

        @Override
        public int compare(Character o1, Character o2) {
            return CharacterWeight.get(o2) - CharacterWeight.get(o1);
        }
    }

    public static void print(OutputStream out, Object str) throws IOException {
        out.write(str.toString().getBytes());
    }

}
