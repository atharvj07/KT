[n, k] = [int(i) for i in input().split()]
s = input()
if n - k > k - 1:
	i = 0
	d = 1
	for j in range(k - 1):
		print('LEFT')
else:
	i = n - 1
	d = -1
	for j in range(n - k):
		print('RIGHT')
for j in range(n - 1):
	print('PRINT', s[i])
	print('LEFT' if d < 0 else 'RIGHT')
	i += d
print('PRINT', s[i])
