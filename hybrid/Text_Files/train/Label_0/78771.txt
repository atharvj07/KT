def helper(n, a, b):
	c = []
	for i in range(n):
		c.append(a[i] - b[i])
	c = sorted(c)
	p1 = 0
	p2 = n - 1
	count = 0
	while p2 > p1:
		x1 = c[p1]
		x2 = c[p2]
		if x1 + x2 > 0:
			count += p2 - p1
			p2 -= 1
		else:
			p1 += 1
	return count
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
print(helper(n, a, b))
