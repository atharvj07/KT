from sys import stdin

def func(arr, l, r, x):
	while l < r:
		mid = (l + r) // 2
		if arr[mid] + x > 0:
			r = mid
		else:
			l = mid + 1
	return l
d = {}
i = 0
n = int(stdin.readline())
for _ in range(n - 1):
	(u, v) = map(int, stdin.readline().split())
	if u in d:
		d[u][0] += 1
		d[u][1].append(i)
	else:
		d[u] = [1, [i]]
	if v in d:
		d[v][0] += 1
		d[v][1].append(i)
	else:
		d[v] = [1, [i]]
	i += 1
ans = [-1] * (n - 1)
val = 0
for ele in d:
	if d[ele][0] >= 3:
		for item in d[ele][1]:
			if ans[item] == -1:
				ans[item] = val
				val += 1
for i in range(n - 1):
	if ans[i] == -1:
		print(val)
		val += 1
	else:
		print(ans[i])
