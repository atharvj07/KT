import java.util.*;
public class Solution
{
    public static void main(String ar[])
    {
        Scanner scan=new Scanner(System.in);
        int n=scan.nextInt();
        int k=scan.nextInt();
        int arr[]=new int[n];
        for(int i=0;i<n;i++)
            arr[i]=scan.nextInt();
        int maxlen=0;
        int start=0,end=0;
        long sum=0;
        while(end<n)
        {
            sum+=arr[end];
            while(sum>k&&start<=end)
                sum-=arr[start++];
            maxlen=Math.max(maxlen,end-start+1);
            end++;
        }
        System.out.println(maxlen);
    }
}
