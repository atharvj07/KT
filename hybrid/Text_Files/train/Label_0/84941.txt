r = lambda : map(int, input().split())
(n, k, m, t) = list(r())
tab = [0] * n
tab[k - 1] = 1
command = []
for i in range(t):
	command.append(list(r()))
res = []
for i in range(t):
	if command[i][0] == 0:
		try:
			tab.index(1, 0, command[i][1])
			tab = tab[0:command[i][1]]
		except ValueError:
			tab = tab[command[i][1]:len(tab)]
	else:
		tab.insert(command[i][1] - 1, 0)
	res.append([len(tab), tab.index(1) + 1])
for i in range(t):
	print(res[i][0], res[i][1])
