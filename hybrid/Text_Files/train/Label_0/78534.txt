def gcd(a, b):
	while b:
		a %= b
		(a, b) = (b, a)
	return a

def lcm(a, b):
	return int(a / gcd(a, b) * b)

def main():
	q = int(input())
	for i in range(q):
		a = input()
		b = input()
		na = len(a)
		nb = len(b)
		x = lcm(na, nb)
		a *= int(x / na)
		b *= int(x / nb)
		if a == b:
			print(a)
		else:
			print(-1)
main()
