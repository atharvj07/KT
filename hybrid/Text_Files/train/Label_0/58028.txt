import java.util.*;
import java.io.*;
import java.lang.Math;
import java.util.Random;


public class Solution{

	static int[] freq;



	public static void main(String[] args) throws Exception{

		FastScanner fs = new FastScanner();

		int t = fs.nextInt();

		while(t-->0){

			char[] arr = fs.next().toCharArray();
			int n = arr.length;
			int m = fs.nextInt();
			int[] b = fs.readArray(m);

			ArrayList<ArrayList<Integer>> orderList = new ArrayList<ArrayList<Integer>>(m);
			ArrayList<Integer> zeroPos = new ArrayList<Integer>();
			for(int i=0;i<m;i++){
				if(b[i]==0)
					zeroPos.add(i);
			}
			orderList.add(new ArrayList<Integer>(zeroPos));

			while(!zeroPos.isEmpty()){
				for(int j=0;j<zeroPos.size();j++){
					for(int i=0;i<m;i++){
						if(b[i]==0)
							b[i] = -1;
						b[i] = b[i] - (Math.abs(zeroPos.get(j)-i));
					}		
				}
				zeroPos = giveZList(b);
				orderList.add(zeroPos);
			}
			orderList.remove(orderList.size()-1);

			char[] res = new char[m];
			Arrays.sort(arr);
			arr = reverse(arr);


			int[] freq = new int[26];
			for(int i=0;i<n;i++)
				freq[arr[i]-'a']++;


			int ind = 0;
			for(int i=0;i<orderList.size();){
				if(freq[arr[ind]-'a']>=orderList.get(i).size()){
					for(int num: orderList.get(i))
						res[num] = arr[ind];
					i++;
				}
				ind += freq[arr[ind]-'a'];
			}

			System.out.println(String.valueOf(res));



			
					

		}
		
	}

	static char[] reverse(char[] arr){
		int n = arr.length;
		char[] res = new char[n];

		for(int i=0;i<n;i++){
			res[i] = arr[n-i-1];
		}
		return res;
	}

	static ArrayList<Integer> giveZList(int[] arr){
		int n = arr.length;
		ArrayList<Integer> list = new ArrayList<Integer>(n);
		for(int i=0;i<n;i++)
			if(arr[i]==0)
				list.add(i);
		return list;
	}

	static class FastScanner{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");

		public String next(){
			while(!st.hasMoreElements()){
				try{
					st = new StringTokenizer(br.readLine());
				} catch(IOException e){
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		public int nextInt(){
			return Integer.parseInt(next());
		}

		public int[] readArray(int n){
			int[] a = new int[n];
			for(int i=0;i<n;i++)
				a[i] = nextInt();
			return a;
		}

		public long nextLong(){
			return Long.parseLong(next());
		}
	}
}
