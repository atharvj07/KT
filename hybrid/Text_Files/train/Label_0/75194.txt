n = int(input())
res = []
a = input().split()
for i in range(n):
	a[i] = int(a[i])
asor = sorted(a)
run = True
su = sum(a)
m = max(a)
if run:
	for i in range(n):
		if a[i] == m:
			if su - a[i] - asor[-2] == asor[-2]:
				res.append(i + 1)
		elif su - a[i] - asor[-1] == asor[-1]:
			res.append(i + 1)
if len(res) > 0:
	print(len(res))
	for r in res:
		print(r, end=' ')
else:
	print(0)
