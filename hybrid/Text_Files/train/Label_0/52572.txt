import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.PriorityQueue;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author Tifuera
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskE solver = new TaskE();
        solver.solve(1, in, out);
        out.close();
    }

    static
    @SuppressWarnings("Duplicates")
    class TaskE {
        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.nextInt();
            int haveCoins = in.nextInt();

            int[] toPay = new int[n];
            for (int i = 0; i < n; i++) {
                toPay[i] = in.nextInt();
            }

            int[] w = new int[n];
            for (int i = 0; i < n; i++) {
                w[i] = in.nextInt();
            }

            PriorityQueue<TaskE.DisSatisfaction> pq = new PriorityQueue<>();

            boolean[] isSatisfied = new boolean[n];

            long sumDis = 0;
            for (int i = 0; i < n; i++) {
                int curCoins = toPay[i] % 100;

                int ds = (100 - curCoins) % 100;
                if (ds > 0) {
                    pq.add(new TaskE.DisSatisfaction(ds * w[i], i));
                }

                if (haveCoins >= curCoins) {
                    haveCoins -= curCoins;
                    isSatisfied[i] = true;

                } else {
                    TaskE.DisSatisfaction disSatisfaction = pq.poll();
                    isSatisfied[i] = true;
                    isSatisfied[disSatisfaction.day] = false;


                    sumDis += disSatisfaction.value;

                    haveCoins -= curCoins;
                    haveCoins += 100;
                }
            }

            out.println(sumDis);
            for (int i = 0; i < n; i++) {
                if (isSatisfied[i]) {
                    out.print(toPay[i] / 100);
                    out.print(" ");
                    out.print(toPay[i] % 100);
                } else {
                    out.print((toPay[i] + 99) / 100);
                    out.print(" 0");
                }
                out.println();
            }

        }

        private static class DisSatisfaction implements Comparable<TaskE.DisSatisfaction> {
            private int value;
            private int day;

            public DisSatisfaction(int value, int day) {
                this.value = value;
                this.day = day;
            }


            public int compareTo(TaskE.DisSatisfaction o) {
                return Integer.compare(value, o.value);
            }

        }

    }

    static class InputReader {
        private BufferedReader reader;
        private String[] currentArray;
        private int curPointer;

        public InputReader(InputStream inputStream) {
            reader = new BufferedReader(new InputStreamReader(inputStream));
        }

        public int nextInt() {
            if ((currentArray == null) || (curPointer >= currentArray.length)) {
                try {
                    currentArray = reader.readLine().split(" ");
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
                curPointer = 0;
            }
            return Integer.parseInt(currentArray[curPointer++]);
        }

    }
}
