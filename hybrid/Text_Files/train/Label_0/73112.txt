x = int(input())
for i in range(x):
	f = [int(f) for f in input().split()]
	d = [int(d) for d in input().split()]
	s = []
	for i in range(f[1]):
		cnt = 0
		if f[0] == 1:
			s.append(-1)
		else:
			for j in range(len(d) - 1):
				cnt += 1
				if d[j] < d[j + 1]:
					d[j] += 1
					s.append(cnt)
					break
				if cnt == len(d) - 1:
					s.append(-1)
			if s[-1] == -1:
				break
	print(s[-1])
