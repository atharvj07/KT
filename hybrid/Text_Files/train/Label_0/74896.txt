import sys
infile = sys.stdin
next(infile)
s = [[0, 0], [0, 0]]
for line in infile:
	l1 = list(map(int, line.split()))
	n = l1[0]
	m = l1[1]
	c = 0
	for i in range(n):
		l2 = list(map(int, infile.readline().split()))
		s[0][0] = l2[0]
		s[0][1] = l2[1]
		l3 = list(map(int, infile.readline().split()))
		s[1][0] = l3[0]
		s[1][1] = l3[1]
		if s[0][1] == s[1][0]:
			c += 1
	if m % 2 != 0:
		print('NO')
	elif c == 0:
		print('NO')
	else:
		print('YES')
