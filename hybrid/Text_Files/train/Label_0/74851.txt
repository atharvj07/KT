import re
(x, y) = map(int, input().strip().split())
bx = bin(x)[2:]
by = bin(y)[2:]
if bx == by:
	print('YES')
else:
	bxf = bx
	bx = bx.strip('0')
	r = [1, 1, 1, 1]
	r[0] = re.compile('^1*' + bx + '1*$')
	r[1] = re.compile('^1*' + bx[::-1] + '1*$')
	r[2] = re.compile('^1*' + bxf + '11*$')
	r[3] = re.compile('^1*1' + bxf[::-1] + '1*$')
	don = False
	for rr in r:
		if rr.match(by) and (not don):
			print('YES')
			don = True
	if not don:
		print('NO')
