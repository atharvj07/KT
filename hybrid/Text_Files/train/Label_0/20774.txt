#include<cstdio>
#include<cstdlib>
#include<algorithm>
#include<iostream>
#include<stack>
#include<vector>
#include<queue>
#define INFTY 500000
#define WHITE 0
#define GRAY 1
#define BLACK 2
using namespace std;

int color[12000],d[12000];
vector<pair<int,int> > v[12000];
priority_queue<pair<int,int> > PQ;
int n;

void start(int x){
  d[x]=0;
  PQ.push(make_pair(0,0));
  while(!PQ.empty()){
    pair<int,int> f;
    f=PQ.top();
    PQ.pop();
    int u=f.second;
    if(f.first*(-1) > d[u]) continue;
    color[u]=BLACK;
    for(int i=0;i<v[u].size();i++){
      int temp=v[u][i].first;
      if(d[temp]>d[u]+v[u][i].second && color[temp]!=BLACK){
        d[temp]=d[u]+v[u][i].second;
        color[temp]=GRAY;
        PQ.push(make_pair(-1*d[temp],temp));
      }
    }
  }
  for(int i=0;i<n;i++){
    cout<<i<<" "<<d[i]<<endl;
  }
}

int main(){
  int u,k,v1,c;
  cin>>n;
  for(int i=0;i<n;i++){
    d[i]=INFTY;
  }
  for(int i=0;i<n;i++){
    cin>>u>>k;
    for(int j=0;j<k;j++){
      cin>>v1>>c;
      v[u].push_back(make_pair(v1,c)); 
    }
  }
  start(0);
}
