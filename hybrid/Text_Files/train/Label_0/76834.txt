def good(N, k):
	c = 0
	m = N
	while m > 0:
		a = min(k, m)
		m -= a
		c += a
		m -= m // 10
	return c * 2 >= N

def solve(N):
	lb = 0
	ub = N
	while ub - lb > 1:
		mid = (ub + lb) // 2
		if good(N, mid):
			ub = mid
		else:
			lb = mid
	return ub

def main():
	N = int(input())
	print(solve(N))
main()
