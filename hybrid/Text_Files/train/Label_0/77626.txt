import sys
import bisect
input = sys.stdin.readline
(n, k) = map(int, input().split())
a = list(map(int, input().split()))

def f(a):
	a.sort()
	pref = a[:]
	for i in range(1, n):
		pref[i] += pref[i - 1]
	mx = a[-1]

	def possible(base, x):
		idx = bisect.bisect_left(a, x + 1)
		need = pref[-1] - (pref[idx - 1] if idx >= 1 else 0) - (n - idx) * x
		return need + base <= k
	ans = 1 << 32
	for lo in set(a):
		idx = bisect.bisect_left(a, lo)
		base = lo * (idx + 1) - pref[idx]
		if base > k:
			break
		(hi0, hi1) = (lo, mx)
		if possible(base, hi0):
			print(0)
			exit()
		while hi0 + 1 < hi1:
			mid = (hi0 + hi1) // 2
			if possible(base, mid):
				hi1 = mid
			else:
				hi0 = mid
		ans = min(ans, hi1 - lo)
	return ans
print(min(f(a), f([-x for x in a])))
