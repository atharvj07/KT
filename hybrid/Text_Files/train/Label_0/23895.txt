#include<iostream>
#include<vector>
#include<deque>

using namespace std;

#define int long long

signed main(){
	const int INF = (1<<30);
	
	int N, Q;
	vector<deque<int>> gs;
	
	cin>>N>>Q;
	
	gs.resize(N);
	
	for(int i = 0; i < Q; i++){
		int query;
		
		cin>>query;
		
		if(query == 0){
			int lane;
			
			cin>>lane;
			
			lane--;
			
			cout<<gs[lane].front()<<endl;
			
			gs[lane].pop_front();
			
		} else if(query == 1){
			int car;
			int minj = 0;
			
			cin>>car;
			
			for(int j = 0; j < N; j++){
				if(gs[minj].size() > gs[j].size()) {
					minj = j;
				}
			}
			
			gs[minj].push_back(car);
			
		}
	}
}
