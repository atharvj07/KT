#include <bits/stdc++.h>
using namespace std;
bool Gr5(string str) {
  if (str.size() >= 5)
    return true;
  else
    return false;
}
bool hsdigit(string str) {
  for (int i = 0; i < str.size(); i++) {
    for (char c = '0'; c <= '9'; c++) {
      if (str[i] == c) {
        return true;
      }
    }
  }
  return false;
}
bool hsUp(string str) {
  for (int i = 0; i < str.size(); i++) {
    for (char c = 'A'; c <= 'Z'; c++) {
      if (str[i] == c) return true;
    }
  }
  return false;
}
bool hsLow(string str) {
  for (int i = 0; i < str.size(); i++) {
    for (char c = 'a'; c <= 'z'; c++) {
      if (str[i] == c) {
        return true;
      }
    }
  }
  return false;
}
int main() {
  string str;
  cin >> str;
  if (hsUp(str) && hsLow(str) && hsdigit(str) && Gr5(str)) {
    cout << "Correct";
  } else
    cout << "Too weak";
}
