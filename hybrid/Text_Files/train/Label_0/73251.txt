from collections import defaultdict, deque, Counter, OrderedDict
import sys
from bisect import bisect_left

def main():
	t = int(input())
	arr = []
	mp = {}
	(curr, count) = (0, 1)
	for i in range(1, 2000):
		curr += i
		arr.append(curr)
		mp[curr] = count
		count += 1
	while t:
		t -= 1
		n = int(input())
		i = bisect_left(arr, n)
		if n == arr[i] - 1:
			print(mp[arr[i]] + 1)
		else:
			print(mp[arr[i]])
main()
