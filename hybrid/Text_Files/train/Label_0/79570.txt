import math, sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, w) = map(int, input().split())
	x = list(map(int, input().split()))
	st = math.ceil(w / 2)
	p = [[x[i], i + 1] for i in range(n)]
	p.sort(reverse=True)
	cnt = 0
	v = set()
	for each in p:
		if cnt >= st:
			break
		if each[0] > w:
			continue
		if cnt + each[0] <= w:
			cnt += each[0]
			v.add(each[1])
	if cnt < st:
		print(-1)
	else:
		print(len(v))
		print(*v)
