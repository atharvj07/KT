#include<bits/stdc++.h>
using namespace std;
#define int long long

signed main()
{
   int T; cin>>T;
   double Max=-1.0;
   while(T--){
      int n,m; cin>>n>>m;
      double vrsum=0.0;
      double rsum=0.0;
      while(m--){
         double v,r; cin>>v>>r;
         vrsum+=v*r;
         rsum+=r;
      }
      Max=max(Max,vrsum/rsum);
   }

   int p,q; cin>>p>>q;
   double vrsum=0.0;
   double rsum=0.0;
   while(q--){
      double v,r; cin>>v>>r;
      vrsum+=v*r;
      rsum+=r;
   }
   double x=vrsum/rsum;

   if(Max-x>0.0000001){
      cout<<"YES"<<endl;
   }else{
      cout<<"NO"<<endl;
   }

}
