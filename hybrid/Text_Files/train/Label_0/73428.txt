n = int(input())
a = [None] * n
for i in range(n):
	a[i] = list(map(int, input().split()))
if n == 1:
	for i in range(1, 8):
		if i not in a[0]:
			print(i - 1)
			exit()
	else:
		print(0)
		exit()
elif n == 2:
	for i in range(1, 100):
		if i < 10:
			if i not in a[0] and i not in a[1]:
				print(i - 1)
				exit()
		else:
			(first, last) = map(int, str(i))
			if first in a[0] and last in a[1] or (first in a[1] and last in a[0]):
				continue
			else:
				print(i - 1)
				exit()
	else:
		print(0)
		exit()
else:
	for i in range(1, 100):
		if i < 10:
			if i not in a[0] and i not in a[1] and (i not in a[2]):
				print(i - 1)
				exit()
		else:
			(first, last) = map(int, str(i))
			if first in a[0] and last in a[1] or (first in a[1] and last in a[0]) or (first in a[1] and last in a[2]) or (first in a[2] and last in a[1]) or (first in a[2] and last in a[0]) or (first in a[0] and last in a[2]):
				continue
			else:
				print(i - 1)
				exit()
	else:
		print(0)
