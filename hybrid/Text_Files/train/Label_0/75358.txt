n = int(input())
a = [[int(i)] for i in input().split()]
b = [int(i) for i in input().split()]
for i in range(n):
	a[i].append(b[i])
del b
d = [0 for i in range(n)]
for i in range(1, n):
	mini = 0
	for j in range(i):
		if a[j][0] < a[i][0]:
			if mini == 0:
				mini = a[j][1]
			else:
				mini = min(mini, a[j][1])
	if mini:
		mini += a[i][1]
	d[i] = mini
c = [0 for i in range(n)]
for i in range(1, n):
	mini = 0
	if d[i]:
		for j in range(i + 1, n):
			if a[i][0] < a[j][0]:
				if mini == 0:
					mini = a[j][1]
				else:
					mini = min(mini, a[j][1])
	if mini:
		c[i] = d[i] + mini
c = [i for i in c if i > 0]
if len(c):
	print(min(c))
else:
	print('-1')
