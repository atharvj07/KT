from collections import defaultdict
n = int(input())
lst = list(map(int, input().split(' ')))
arr = [[-1 for i in range(2)] for j in range(n + 1)]
for i in range(n):
	if arr[lst[i]][0] == -1:
		arr[lst[i]][0] = i
		arr[lst[i]][1] = i
	arr[lst[i]][1] = i
a = defaultdict(lambda : 0)
dp = [0] * (n + 1)
for i in range(n - 1, -1, -1):
	a[lst[i]] += 1
	if i == arr[lst[i]][0]:
		dp[i] = max(dp[i + 1], a[lst[i]] + dp[arr[lst[i]][1] + 1])
	else:
		dp[i] = max(dp[i + 1], a[lst[i]])
print(n - dp[0])
