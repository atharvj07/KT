from sys import stdin
import math
inp = lambda : stdin.readline().strip()
t = int(inp())
for _ in range(t):
	n = int(inp())
	a = inp()
	b = inp()
	eq = []
	fin = []
	if n == 1:
		if a == b:
			print(0)
		else:
			print('1 1')
	else:
		for i in range(n - 1):
			if a[i] != a[i + 1]:
				eq.append(i + 1)
		for i in range(n - 2, -1, -1):
			if b[i] != b[i + 1]:
				fin.append(i + 1)
		if a[-1] != b[-1]:
			eq.append(n)
		print(len(eq) + len(fin), end=' ')
		print(*eq, end=' ')
		print(*fin)
