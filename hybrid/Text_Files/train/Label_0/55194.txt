    int n;
    int[] xOrd, yOrd, xV, yV;
    int[] dx = { 1, -1, 0, 0 };
    int[] dy = { 0, 0, -1, 1 };
    int[] Vect = new int[1000];
    {
        Vect['R'] = 0;
        Vect['L'] = 1;
        Vect['D'] = 2;
        Vect['U'] = 3;
    }
    Integer[] XY;
    int[][] grid;
    final int inf = (int)1e8;

    void dfs( int x, int y ){
        int sz = grid.length;
        if(!( 0<=x && x < sz &&
              0<=y && y < sz  )) return;
        if( grid[x][y] != 0 ) return;

        grid[x][y] = 2;
        dfs( x-1, y );
        dfs( x+1, y );
        dfs( x, y-1 );
        dfs( x, y+1 );
    }

    int getVirtual( int val ){
        int ind = binarySearch( XY, val );
        if( ind < 0 ) MLE();
        return ind;
    }

    void show(){
        for( int y = grid[0].length-1; 0<=y; --y ){
            for( int x = 0; x < grid.length; ++x ){
                if( grid[x][y]==0 ) out.print( '.' );
                if( grid[x][y]==1 ) out.print( 'V' );
                if( grid[x][y]==2 ) out.print( '_' );
            }
            out.println();
        }
        out.println();
    }

    void solve(){
        n = nextInt();
        xOrd = new int[n+1];
        yOrd = new int[n+1];
        for( int i = 1; i <= n; ++i ){
            int vect = Vect[next().charAt(0)];
            int d = nextInt();
            xOrd[i] = xOrd[i-1] + d * dx[vect];
            yOrd[i] = yOrd[i-1] + d * dy[vect];
        }

        {
            TreeSet<Integer> ts = new TreeSet<Integer>();
            ts.add(  inf );
            ts.add( -inf );
            for( int v : xOrd ){ ts.add( v   ); ts.add( v+1 ); }
            for( int v : yOrd ){ ts.add( v   ); ts.add( v+1 ); }
            XY = ts.toArray(new Integer[0]);
        }

        xV = new int[n+1];
        yV = new int[n+1];
        for( int i = 0; i <= n; ++i ){
            xV[i] = getVirtual( xOrd[i] );
            yV[i] = getVirtual( yOrd[i] );
        }
        grid = new int[XY.length][XY.length];
        for( int i = 1; i <= n; ++i ){
            int dx = Integer.signum( xV[i] - xV[i-1] );
            int dy = Integer.signum( yV[i] - yV[i-1] );
            for( int x = xV[i-1], y = yV[i-1]; ; ){
                try {
                    grid[x][y] = 1;
                } catch (Exception e) {
                    MLE();
                }
                if( x==xV[i] && y==yV[i] ) break;
                x += dx;
                y += dy;
            }
        }

//        show();
        try {
            dfs(0,0);
        } catch (Exception e) {
            MLE();
        }
//        show();
        long ans = 0;
        for( int x = 0; x+1 < grid.length; ++x ){
        for( int y = 0; y+1 < grid.length; ++y ){
            if( grid[x][y]==2 ) continue;
            long dx = XY[x+1] - XY[x];
            long dy = XY[y+1] - XY[y];
            ans += dx * dy;
        }
        }
        out.println( ans );
    }


}
