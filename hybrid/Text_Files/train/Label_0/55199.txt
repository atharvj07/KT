import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

public class Solution {

 public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n=s.nextInt();
        int a[]=new int[n];
        for(int i=0;i<n;i++)
            a[i]=s.nextInt();
        int max=0;
        for(int i=0;i<n;i++)
        {
            for(int j=i;j<n;j++)
            {
                int b[]=new int[n];
                for(int k=0;k<n;k++)
                    b[k]=a[k];
                for(int k=i;k<=j;k++)
                    b[k]=1-a[k];
                int count=0;
                for(int k=0;k<n;k++)
                    if(b[k]==1) 
                        count++;
                max=Math.max(max,count);
            }
        }
        System.out.println(max);
    }
}
