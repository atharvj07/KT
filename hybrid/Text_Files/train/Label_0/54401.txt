import java.util.ArrayList;
import java.util.Scanner;
import java.lang.StringBuilder;
import java.util.Arrays;
import java.util.Stack;
//maintain: array on vertices
//a[i]: 
public class Test10   {


    public static void main(String[] Args)   {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        int e = sc.nextInt();
        int f = sc.nextInt();
        int g = sc.nextInt();
        int h = sc.nextInt();
        int minX = a;
        int maxX = 0;
        if (c != a) maxX = c;
        if (e != a) maxX = e;
        if (g != a) maxX = g;
        if (maxX < minX)   {
            a = minX;
            minX = maxX;
            maxX = a;
        }   
        int minY = b;
        int maxY = 0;
        if (d != b) maxY = d;
        if (f != b) maxY = f;
        if (h != b) maxY = h;
        if (maxY < minY)   {
            b = minY;
            minY = maxY;
            maxY = b;
        }
        int a1 = sc.nextInt();
        int b1 = sc.nextInt();
        int c1 = sc.nextInt();
        int d1 = sc.nextInt();
        int e1 = sc.nextInt();
        int f1 = sc.nextInt();
        int g1 = sc.nextInt();
        int h1 = sc.nextInt();
        int cX = (a1 + c1 + e1 + g1)/4;
        int cY = (b1 + d1 + f1 + h1)/4;
        int l = Math.abs(cX - a1) + Math.abs(cY - b1);
        boolean ans = false;
        for (int i = minX; i <= maxX; i++)   {
            for (int j = minY; j <= maxY; j++)   {
                if (i + j >= cX + cY - l && i + j <= cX + cY + l)  {
                    if (i - j >= cX - cY - l && i - j <= cX - cY + l) ans = true;
                }   
            }    
        }    
        if (ans) System.out.println("YES"); else System.out.println("NO"); 
    }    
}
