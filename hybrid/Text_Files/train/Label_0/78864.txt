n = int(input())
a = [int(item) for item in input().split()]
climbing = True
prev = a[0]
for item in a[1:]:
	if climbing and prev < item:
		pass
	elif climbing and prev > item:
		climbing = False
	elif not climbing and prev > item:
		pass
	elif not climbing and prev < item:
		print('NO')
		exit()
	prev = item
print('YES')
