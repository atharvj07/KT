#include <bits/stdc++.h>
using namespace std;
const int maxn = 1048576, maxm = 23;
long double dp[maxn];
int n, l, r, light[maxm][3];
long double to_rad(long double x) { return M_PI * x / (long double)180; }
bool infinity(long double a, long double x, long double y, long double s) {
  return a + atan2(y, s - x) >= (long double)0;
}
long double cot(long double a) { return (long double)1 / tan(a); }
long double solve(long double a, long double x, long double y, long double s) {
  return cot(a + atan2(y, s - x)) * y + x;
}
int main() {
  ios::sync_with_stdio(0);
  cin >> n >> l >> r;
  for (int i = 0; i < (1 << n); i++) dp[i] = -10000000;
  for (int i = 0; i < n; i++) cin >> light[i][0] >> light[i][1] >> light[i][2];
  dp[0] = l;
  for (int i = 1; i < (1 << n); i++) {
    for (int j = 0; j < n; j++) {
      if ((i >> j) & 1) {
        if (infinity(to_rad(light[j][2]), light[j][0], -light[j][1],
                     dp[i ^ (1 << j)])) {
          dp[i] = (long double)r;
          break;
        } else
          dp[i] = max(dp[i], min((long double)r,
                                 solve(to_rad(light[j][2]), light[j][0],
                                       -light[j][1], dp[i ^ (1 << j)])));
      }
    }
  }
  cout << setprecision(9) << fixed << dp[(1 << n) - 1] - (long double)l;
  return 0;
}
