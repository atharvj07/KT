def sumDigits(n):
	answer = 0
	while n > 0:
		answer += n % 10
		n = n // 10
	return answer

def listFactors(n):
	answer = []
	while n > 1:
		for i in range(2, n + 1):
			if n % i == 0:
				answer.append(i)
				n = n // i
				break
	return answer
i = int(input())
test = 4
numsofar = 0
while numsofar < i:
	if len(listFactors(test)) > 1:
		numsum = sumDigits(test)
		factorsum = 0
		for factor in listFactors(test):
			factorsum += sumDigits(factor)
		if numsum == factorsum:
			numsofar += 1
			if numsofar == i:
				print(test)
				break
	test += 1
