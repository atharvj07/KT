const ll mod = 998244353;
const int N = 2010;
ll fact[N], invf[N];

ll powll(ll x, ll y){
    x %= mod;
    ll res = 1LL;
    while(y){
        if (y & 1LL)
            res *= x;
        res %= mod;
        x = (x*x) % mod;
        y >>= 1LL;
    }
    return res;
}

ll mul(ll x, ll y){
    return (x%mod)*(y%mod)%mod;
}

ll divll(ll x, ll y){
    x %= mod;
    return (x * powll(y,mod-2)) % mod;
}

ll nPr(ll n, ll r){
    if (n < r || r < 0) return 0;
    return mul(fact[n], invf[n-r]);
}

ll nCr(ll n, ll r){
    if (n < r || r < 0) return 0;
    if (n < N) return mul(mul(fact[n], invf[r]), invf[n-r]);
    ll ret = 1;
    n %= mod;
    REP(i,r) ret = (ret * (n - i + mod)) % mod;
    return mul(ret, invf[r]);
}

int main() {
    ll m, n, k;
    cin >> m >> n >> k;
    fact[0] = invf[0] = 1;
    FOR(i,1,N-1){
        fact[i] = (fact[i-1] * i) % mod;
        invf[i] = divll(invf[i-1], i);
    }
    VL dp(k + 1);
    dp[1] = 1;
    FOR(x,2,k){
        dp[x] = powll(x, n);
        FOR(y,1,x-1){
            dp[x] = (dp[x] - ((nCr(x, y) * dp[y]) % mod) + mod) % mod;
        }
    }
    ll ans = powll(m, n);
    FOR(x,1,k-1){
        ans = (ans - ((nCr(m, x) * dp[x]) % mod) + mod) % mod;
    }
    cout << ans << endl;

    return 0;
}
