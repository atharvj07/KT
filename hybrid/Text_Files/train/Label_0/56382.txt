import java.io.*;
import java.util.*;

public class CF252D extends PrintWriter {
	CF252D() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF252D o = new CF252D(); o.main(); o.flush();
	}

	int count(int[] pp, int[] tt, int[] qq, int[] ss, int n, int k) {
		for (int i = 0; i < n; i++)
			pp[i] = i;
		for (int h = 0; h <= k; h++) {
			if (Arrays.equals(pp, ss))
				return h;
			for (int i = 0; i < n; i++)
				tt[i] = pp[qq[i]];
			int[] tmp = pp; pp = tt; tt = tmp;
		}
		return k + 1;
	}
	void main() {
		int n = sc.nextInt();
		int k = sc.nextInt();
		int[] qq = new int[n];
		int[] ii = new int[n];
		for (int i = 0; i < n; i++) {
			int q = sc.nextInt() - 1;
			qq[i] = q;
			ii[q] = i;
		}
		int[] ss = new int[n];
		for (int i = 0; i < n; i++)
			ss[i] = sc.nextInt() - 1;
		int[] pp = new int[n];
		int[] tt = new int[n];
		int kq = count(pp, tt, qq, ss, n, k);
		if (kq == 0) {
			println("NO");
			return;
		}
		int ki = count(pp, tt, ii, ss, n, k);
		if (kq == 1 && ki == 1) {
			println(k == 1 ? "YES" : "NO");
			return;
		}
		if (kq <= k && ki > k)
			println((k - kq & 1) == 0 ? "YES" : "NO");
		else if (ki <= k && kq > k)
			println((k - ki & 1) == 0 ? "YES" : "NO");
		else if (kq <= k && ki <= k)
			println((k - kq & 1) == 0 || (k - ki & 1) == 0 ? "YES" : "NO");
		else
			println("NO");
	}
}
