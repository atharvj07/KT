#include <bits/stdc++.h>
using namespace std;
bool sortbysec(const pair<int, int> &a, const pair<int, int> &b) {
  return (a.first < b.first);
}
int main() {
  std::ios::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  int n;
  cin >> n;
  string s;
  vector<pair<int, int> > a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i].first;
    a[i].second = i;
  }
  cin >> s;
  string c;
  sort(a.begin(), a.end(), sortbysec);
  int st = 0;
  vector<int> d;
  for (int i = 0; i < 2 * n; i++) {
    if (s[i] == '0') {
      c += to_string(a[st].second + 1);
      c += " ";
      d.push_back(a[st].second);
      st++;
    } else {
      c += to_string(d[d.size() - 1] + 1);
      c += " ";
      d.erase(d.end() - 1);
    }
  }
  cout << c << endl;
}
