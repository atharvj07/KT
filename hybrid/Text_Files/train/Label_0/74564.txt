import math
fact = [math.factorial(x) for x in range(20)]

def product(L):
	ans = 1
	for i in L:
		ans *= i
	return ans

def count(digits):
	s = sum(digits)
	return fact[s] // product([fact[i] for i in digits])
n = input()
strn = str(n)
digs = [0 for i in range(10)]
for let in strn:
	dig = int(let)
	digs[dig] += 1
zeroes = digs[0]
others = digs[1:]
while 0 in others:
	others.remove(0)
digs = [zeroes] + others

def answer(digits, chosen=[]):
	ans = 0
	if digits == []:
		return helper(chosen)
	if digits[0] == 0:
		return answer(digits[1:], chosen + [0])
	for i in range(1, digits[0] + 1):
		ans += answer(digits[1:], chosen + [i])
	return ans

def helper(chosen):
	if chosen[0] == 0:
		return count(chosen)
	alt = chosen.copy()
	alt[0] -= 1
	return count(chosen) - count(alt)
print(answer(digs))
