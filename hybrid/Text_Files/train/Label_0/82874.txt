def solve(p, A):
	i = 1
	sorted_A = sorted(A)
	while A != sorted_A:
		if i % 2 == 1:
			for j in range(0, p - 2, 2):
				if A[j] > A[j + 1]:
					(A[j], A[j + 1]) = (A[j + 1], A[j])
		else:
			for j in range(1, p - 1, 2):
				if A[j] > A[j + 1]:
					(A[j], A[j + 1]) = (A[j + 1], A[j])
		i += 1
	return i - 1

def main():
	ans = []
	t = int(input())
	for _ in range(t):
		p = int(input())
		A = list(map(int, input().split(' ')))
		ans.append(solve(p, A))
	for a in ans:
		print(a)
main()
