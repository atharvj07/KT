t = int(input())
for i in range(t):
	(n, m, k) = map(int, input().split())
	s = list(map(int, input().split()))
	flag = True
	for j in range(n - 1):
		if s[j] - s[j + 1] >= -k:
			mx_pick = max(0, s[j + 1] - k)
			m += s[j] - mx_pick
		else:
			mn_lose = s[j + 1] - k
			m -= mn_lose - s[j]
			if m < 0:
				flag = False
				break
	if flag:
		print('YES')
	else:
		print('NO')
