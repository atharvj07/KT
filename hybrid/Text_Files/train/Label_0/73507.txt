import io
import os
from collections import deque

def solve(N, A):
	cumsum = [0]
	for x in A:
		cumsum.append(cumsum[-1] + x)
	ans = 0
	window = deque([0])
	windowSet = set([0])
	for j in range(N + 1):
		while cumsum[j] in windowSet:
			i = window.popleft()
			windowSet.remove(cumsum[i])
		window.append(j)
		windowSet.add(cumsum[j])
		ans += len(window) - 1
	return ans
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
(N,) = [int(x) for x in input().split()]
A = [int(x) for x in input().split()]
ans = solve(N, A)
print(ans)
