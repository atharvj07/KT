import sys, math
import io, os

def data():
	return sys.stdin.buffer.readline().strip()

def mdata():
	return list(map(int, data().split()))

def outl(var):
	sys.stdout.write(' '.join(map(str, var)) + '\n')

def out(var):
	sys.stdout.write(str(var) + '\n')
mod = int(1000000000.0) + 7
for t in range(int(data())):
	(n, m) = mdata()
	mat = []
	for i in range(n):
		mat.append(mdata())
	s = [[0, 0]]
	d = []
	cnt = 0
	while s:
		s1 = set()
		a = 0
		b = 0
		while s:
			(i, j) = s.pop()
			if mat[i][j] == 1:
				a += 1
			else:
				b += 1
			if i < n - 1:
				s1.add((i + 1, j))
			if j < m - 1:
				s1.add((i, j + 1))
		cnt += 1
		s = list(s1)
		d.append([a, b])
	ans = 0
	for i in range(len(d) // 2):
		a = d[i][0] + d[-i - 1][0]
		b = d[i][1] + d[-i - 1][1]
		ans += a + b - max(a, b)
	out(ans)
