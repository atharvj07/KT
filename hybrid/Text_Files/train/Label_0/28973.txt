Type T read1(){
	T t=0;
	char k;
	bool vis=0;
	do (k=getchar())=='-'&&(vis=1);while('0'>k||k>'9');
	while('0'<=k&&k<='9')t=(t<<3)+(t<<1)+(k^'0'),k=getchar();
	return vis?-t:t;
}
# define fre(k) freopen(k".in","r",stdin);freopen(k".out","w",stdout
int s,k;
char str[200005];
int sum[200005][17],f[1<<17],mr[200005][17];
bool judge(int x){//cout<<x<<endl;
	memset(f,0x7f>>1,1<<k+2);
	f[0]=0;
	for(int i=s+1;i;--i)
		for(int j=0;j<k;++j){
			if(i+x-1>s){mr[i][j]=s+1;continue;}
			mr[i][j]=mr[i+1][j];
			if(sum[i+x-1][j]-sum[i-1][j]==x)mr[i][j]=i;
//			cout<<x<<' '<<i<<' '<<(char)(j+'a')<<' '<<mr[i][j]<<endl;
		}
	for(int i=0;i<(1<<k);++i)
		if(f[i]<s)
			for(int j=0;j<k;++j)
				if(~i>>j&1)
					f[i|1<<j]=min(f[i|1<<j],mr[f[i]+1][j]+x-1);
	return f[(1<<k)-1]<=s; 
}
int main(){
	s=read;k=read;
	scanf("%s",str+1);
	for(int i=1;i<=s;++i){
		memcpy(sum[i],sum[i-1],sizeof(sum[i]));
		if(str[i]=='?')
			for(int j=0;j<k;++j)++sum[i][j];
		else ++sum[i][str[i]-'a'];
	}
	int l=1,r=s,mid,ans=0;
	while(l<=r)judge(mid=l+r>>1)?l=(ans=mid)+1:r=mid-1;
	cout<<ans;
	return 0;
}
