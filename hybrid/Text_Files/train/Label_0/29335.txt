#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL);
  int t = 1;
  for (int T = 0; T < (t); ++T) {
    int n, a[2000005];
    cin >> n;
    for (int i = 0; i < (n); ++i) {
      cin >> a[i];
    }
    sort(a, a + n);
    int m = n / 2, mn = 1000000000 + 9;
    for (int i = 0; i < (m); ++i) mn = min(mn, a[i + m] - a[i]);
    int ans = mn;
    cout << ans << endl;
  }
  return 0;
}
