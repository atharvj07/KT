#include <bits/stdc++.h>
using namespace std;
vector<string> bd;
int main() {
  int scount = 0;
  bd = vector<string>(8);
  for (int i = 0; i < bd.size(); ++i) {
    cin >> bd[i];
    for (int j = 0; j < bd[i].size(); ++j)
      if (bd[i][j] == 'S') scount++;
  }
  bool res = false;
  set<pair<int, int> > c, n, zero;
  n.insert(make_pair(bd.size() - 1, 0));
  while (true) {
    if (!n.empty()) {
      if (scount == 0) {
        n.clear();
        res = true;
        break;
      }
      c = n;
      n.clear();
      while (!c.empty()) {
        const pair<int, int> cur = *c.begin();
        const int cx = cur.first;
        const int cy = cur.second;
        c.erase(c.begin());
        if (bd[cx][cy] == 'S') continue;
        if (cx == 0 && cy == 7) {
          n.clear();
          res = true;
          break;
        }
        for (int i = 0; i < 3; ++i)
          for (int j = 0; j < 3; ++j)
            if (cx + i - 1 >= 0 && cx + i - 1 < 8 && cy + j - 1 >= 0 &&
                cy + j - 1 < 8 && bd[cx + i - 1][cy + j - 1] != 'S')
              n.insert(make_pair(cx + i - 1, cy + j - 1));
      }
      for (int i = bd.size() - 1; i >= 0; --i)
        for (int j = 0; j < bd[i].size(); ++j)
          if (bd[i][j] == 'S') {
            bd[i][j] = '.';
            if (i < bd.size() - 1) bd[i + 1][j] = 'S';
          }
    } else
      break;
  }
  cout << (res ? "WIN" : "LOSE");
  return 0;
}
