def I():
	return int(input())

def M():
	return map(int, input().split())

def solve():
	(x, n) = M()
	s = 0
	if x % 2 == 0:
		if n % 4 == 0:
			print(x)
		elif n % 4 == 1:
			print(x - n)
		elif n % 4 == 2:
			print(x - (n - 1) + n)
		elif n % 4 == 3:
			print(x - (n - 2) + (n - 1) + n)
	elif n % 4 == 0:
		print(x)
	elif n % 4 == 1:
		print(x + n)
	elif n % 4 == 2:
		print(x + (n - 1) - n)
	elif n % 4 == 3:
		print(x + (n - 2) - (n - 1) - n)
for _ in range(int(input())):
	solve()
