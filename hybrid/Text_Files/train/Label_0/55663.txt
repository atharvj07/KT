
import java.util.*;


public class D
{
   public static void main(String[] args)
   {
      new D(new FastScanner());
   }

   TreeSet<Integer>[] ys; // All x's for a given y
   TreeSet<Integer>[] xs; // All y's for a given x
  
   boolean go(vect start, ArrayList<vect> cmds)
   {
      vect cp = start;
      for (vect dv : cmds)
      {
         vect np = cp.add(dv);
         if (cp.x == np.x)
         {
            int[] vs = new int[]{cp.y, np.y};
            Arrays.sort(vs);
            Integer e = xs[cp.x].ceiling(vs[0]);
            if ((e != null)&&(e.compareTo(vs[1]) <= 0))
               return false;
         }
         else // ys equal
         {
            int[] vs = new int[]{cp.x, np.x};
            Arrays.sort(vs);
            Integer e = ys[cp.y].ceiling(vs[0]);
            if ((e != null)&&(e.compareTo(vs[1]) <= 0))
               return false;
         }
         cp = np;
      }
      return true;
   }

   public D(FastScanner in)
   {
      int H = in.nextInt();
      int W = in.nextInt();
      xs = new TreeSet[W];
      ys = new TreeSet[H];
      for (int i=0; i<W; i++)
         xs[i] = new TreeSet<Integer>();
      for (int i=0; i<H; i++)
         ys[i] = new TreeSet<Integer>();

      vect[] stloc = new vect[26];
      for (int j=0; j<H; j++)
      {
         String s = in.next();
         for (int i=0; i<W; i++)
         {
            char c = s.charAt(i);
            if (c == '#')
            {
               xs[i].add(j);
               ys[j].add(i);
            }
            else if (Character.isUpperCase(c))
               stloc[(int)(c-'A')] = new vect(i,j);
         }
      }

      TreeMap<Character, vect> mmp = new TreeMap<Character, vect>();
      mmp.put('N', new vect(0,-1));
      mmp.put('S', new vect(0,1));
      mmp.put('E', new vect(1,0));
      mmp.put('W', new vect(-1,0));
      int N = in.nextInt();
      ArrayList<vect> cmds = new ArrayList<vect>(N);
      while (N-->0)
         cmds.add(mmp.get(in.next().charAt(0)).scale(in.nextInt()));
      
      StringBuilder sb = new StringBuilder();
      for (int i=0; i<26; i++)
         if (stloc[i] != null)
            if (go(stloc[i], cmds))
               sb.append((char)(i+'A'));
      System.out.println((sb.length() > 0)?sb:"no solution");
   }
}

class vect
{
   int x, y;
   public vect(int i, int j)
   {
      x=i; y=j;
   }

   vect add(vect rhs)
   {
      return new vect(x+rhs.x, y+rhs.y);
   }

   vect scale(int s)
   {
      return new vect(x*s, y*s);
   }

   public String toString()
   {
      return String.format("(%d,%d)", x, y);
   }
}

 class FastScanner{
     int nextInt(){
         try{
             int c=System.in.read();
             if(c==-1) return c;
             while(c!='-'&&(c<'0'||'9'<c)){
                 c=System.in.read();
                 if(c==-1) return c;
             }
             if(c=='-') return -nextInt();
             int res=0;
             do{
                 res*=10;
                 res+=c-'0';
                 c=System.in.read();
             }while('0'<=c&&c<='9');
             return res;
         }catch(Exception e){
             return -1;
         }
     }
    
     long nextLong(){
         try{
             int c=System.in.read();
             if(c==-1) return -1;
             while(c!='-'&&(c<'0'||'9'<c)){
                 c=System.in.read();
                 if(c==-1) return -1;
             }
             if(c=='-') return -nextLong();
             long res=0;
             do{
                 res*=10;
                 res+=c-'0';
                 c=System.in.read();
             }while('0'<=c&&c<='9');
             return res;
         }catch(Exception e){
             return -1;
         }
     }
    
     double nextDouble(){
         return Double.parseDouble(next());
     }
    
     String next(){
         try{
             StringBuilder res=new StringBuilder("");
             int c=System.in.read();
             while(Character.isWhitespace(c))
                 c=System.in.read();
             do{
                 res.append((char)c);
             }while(!Character.isWhitespace(c=System.in.read()));
             return res.toString();
         }catch(Exception e){
             return null;
         }
     }
    
     String nextLine(){
         try{
             StringBuilder res=new StringBuilder("");
             int c=System.in.read();
             while(c=='\r'||c=='\n')
                 c=System.in.read();
             do{
                 res.append((char)c);
                 c=System.in.read();
             }while(c!='\r'&&c!='\n');
             return res.toString();
         }catch(Exception e){
             return null;
         }
     }
 }
