#include <bits/stdc++.h>
using namespace std;
const int mx = 1e2 + 10;
char str[mx][mx];
int n, m;
bool judge(int x, int y) {
  int i, j = 0, k;
  str[x][y] = 'X';
  for (i = y; i >= 0 && str[x][i] == 'X'; i--)
    ;
  i++;
  while (str[x][i] == 'X' && i < 10) {
    j++;
    i++;
  }
  if (j >= 5) {
    str[x][y] = '.';
    return 1;
  }
  for (i = x; i >= 0 && str[i][y] == 'X'; i--)
    ;
  j = 0, i++;
  while (str[i][y] == 'X' && i < 10) {
    j++;
    i++;
  }
  if (j >= 5) {
    str[x][y] = '.';
    return 1;
  }
  for (i = x, k = y; i < 10 && k >= 0 && str[i][k] == 'X'; i++, k--)
    ;
  j = 0, i--, k++;
  while (str[i][k] == 'X' && i >= 0 && k < 10) {
    i--, k++;
    j++;
  }
  if (j >= 5) {
    str[x][y] = '.';
    return 1;
  }
  for (i = x, k = y; i >= 0 && k >= 0 && str[i][k] == 'X'; i--, k--)
    ;
  j = 0, i++, k++;
  while (str[i][k] == 'X' && i < 10 && k < 10) {
    i++, k++;
    j++;
  }
  if (j >= 5) {
    str[x][y] = '.';
    return 1;
  }
  str[x][y] = '.';
  return 0;
}
int main() {
  for (int i = 0; i < 10; i++) scanf("%s", str[i]);
  for (int i = 0; i < 10; i++) {
    for (int j = 0; j < 10; j++) {
      if (str[i][j] == '.' && judge(i, j)) {
        puts("YES");
        return 0;
      }
    }
  }
  puts("NO");
  return 0;
}
