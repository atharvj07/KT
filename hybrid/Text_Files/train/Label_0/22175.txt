#include <bits/stdc++.h>
using namespace std;
int dp[1000010];
int fdp[10][1000010];
int f(int n) {
  if (n < 10) return n;
  if (dp[n] != -1) return dp[n];
  int sum = 1;
  while (n != 0) {
    if (n % 10) sum *= (n % 10);
    n /= 10;
  }
  return dp[n] = f(sum);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  memset(dp, -1, sizeof(dp));
  for (int i = 1; i <= 1000000; i++) {
    fdp[f(i)][i]++;
  }
  for (int i = 1; i <= 9; i++) {
    for (int j = 1; j <= 1000000; j++) {
      fdp[i][j] += fdp[i][j - 1];
    }
  }
  int t, x, y, z;
  cin >> t;
  while (t--) {
    cin >> x >> y >> z;
    cout << fdp[z][y] - fdp[z][x - 1] << endl;
  }
  return 0;
}
