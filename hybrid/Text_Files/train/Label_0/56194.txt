import java.io.*;
import java.util.*;

public class C {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());
		StringTokenizer tb;
		for (int i = 0; i < n; i++) {
			tb = new StringTokenizer(br.readLine());
			long l = Long.parseLong(tb.nextToken());
			long h = Long.parseLong(tb.nextToken());
			long answer = l;
			for (int b = 0; b < 61; b++) {
				if ((answer | (1L << b)) > h)
					break;
				answer |= (1L << b);
			}
			System.out.println(answer);
		}
	}
}
