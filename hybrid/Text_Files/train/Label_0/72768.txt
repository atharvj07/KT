for _ in range(int(input())):
	a = int(input())
	b = input()
	ans = ''
	if b.count('1'):
		for i in range(b.find('1')):
			ans += '0' if b[i] == '0' else '1'
		print(ans + '1' + '0' * (a - len(ans) - 1))
		print(ans + '0' + b[b.find('1') + 1:])
	else:
		print(b.replace('2', '1'))
		print(b.replace('2', '1'))
