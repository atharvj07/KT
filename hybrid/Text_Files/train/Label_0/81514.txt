def union(x, y, par):
	x1 = x
	while par[x1] >= 0:
		x1 = par[x1]
	y1 = y
	while par[y1] >= 0:
		y1 = par[y1]
	if x1 == y1:
		return
	if par[x1] < par[y1]:
		setpar = x1
		par[y1] = x1
	else:
		setpar = y1
		par[x1] = y1
	while par[x] != x1 and x != x1:
		temp = par[x]
		par[x] = setpar
		x = temp
	while par[y] != y1 and y != y1:
		temp = par[y]
		par[y] = setpar
		y = temp
(n, m) = map(int, input().split())
par = [-1 for i in range(m)]
val = [False for i in range(m)]
cnt1 = 0
for i in range(n):
	a = list(map(int, input().split()))
	if a[0] == 0:
		cnt1 += 1
		continue
	for j in range(len(a)):
		a[j] -= 1
	for j in range(1, len(a)):
		val[a[j]] = True
	for j in range(2, len(a)):
		union(a[1], a[j], par)
cnt2 = 0
for i in range(m):
	if par[i] == -1 and val[i]:
		cnt2 += 1
print(max(cnt2 - 1, 0) + cnt1)
