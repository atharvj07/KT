    public static boolean dfs(int curr, Map<Integer,List<Integer>> adj, List<Integer> path, Set<Integer> visited, int target){
        visited.add(curr);
        path.add(curr);
        if(curr==target){
            return true;
        }

        for(Integer neighbour: adj.get(curr)){
            if(!visited.contains(neighbour)){
                if(dfs(neighbour,adj,path,visited,target)){
                    return true;
                }
            }
        }
        path.remove(path.size()-1);
        return false;
    }

    public static List<Integer> findPath(int begin, int end, Map<Integer,List<Integer>> adj){
        List<Integer> path = new ArrayList<>();
        Set<Integer> visited = new HashSet<>();
        if(bfs(begin,adj,path,visited, end)) {
            return path;
        }
        else{
            return null;
        }

    }

    private static boolean bfs(int begin, Map<Integer, List<Integer>> adj, List<Integer> path, Set<Integer> visited, int end) {
        Queue<List<Integer>> queue = new LinkedList<>();
        queue.add(List.of(begin));
        visited.add(begin);
        while (!queue.isEmpty()){
            List<Integer> currPath = queue.poll();
            int lastNode = currPath.get(currPath.size()-1);
            for(Integer neighbour : adj.get(lastNode)){
                if(visited.contains(neighbour)) continue;
                visited.add(neighbour);
                List<Integer> newPath = new ArrayList<>(List.copyOf(currPath));
                newPath.add(neighbour);
                if(neighbour.equals(end)){
                    path.addAll(newPath);
                    return true;
                }
                queue.add(newPath);
            }
        }
        return false;
    }

    private static void generateAdj(int nbNodes, int nbEdges, Map<Integer, List<Integer>> adj, Map<Pair<Integer, Integer>, List<Integer>> edgesId, Map<Pair<Integer, Integer>, List<Integer>> weights) {
        for(int i=1;i<=nbNodes;i++){
            adj.put(i, new ArrayList<>());
        }
        int count = 1;
        while(count<=nbEdges){
            int u = (int)(Math.random()*nbNodes);
            int v = (int)(Math.random()*nbNodes);
            int weight = (int)(Math.random()*nbNodes);
            if(u==v) continue;
            if(u==0||v==0)continue;
            if(edgesId.containsKey(new Pair<>(u,v))) continue;
            try{
            adj.get(u).add(v);
            adj.get(v).add(u);
            }
            catch (Exception e){
                System.out.println(e);
            }
            edgesId.put(new Pair<>(u,v),new ArrayList<>(List.of(count)));
            weights.put(new Pair<>(u,v),new ArrayList<>(List.of(weight)));
            count += 1;
        }
    }

    private static List<Pair<Integer, Integer>> findValidBridges(List<Pair<Integer, Integer>> bridges, List<Integer> path) {
        Set<Pair<Integer,Integer>> validEdges = new HashSet<>();
        for(int i=0;i<path.size()-1;i++){
            validEdges.add(new Pair<>(path.get(i),path.get(i+1)));
            validEdges.add(new Pair<>(path.get(i+1),path.get(i)));
        }

        List<Pair<Integer,Integer>> validBridges = new ArrayList<>();
        for(Pair<Integer,Integer> bridge : bridges){
            if(validEdges.contains(bridge)){
                validBridges.add(bridge);
            }
        }
        return validBridges;
    }


    private static List<Pair<Integer, Integer>> findBridges(Map<Integer, List<Integer>> adj) {
        Set<Integer> visited = new HashSet<>();
        Map<Integer,Integer> ids = new HashMap<>();
        Map<Integer,Integer> lows = new HashMap<>();
        Map<Integer,Integer> parents = new HashMap<>();
        List<Pair<Integer,Integer>> bridges = new ArrayList<>();
        for(Integer root : adj.keySet()){
            if(!visited.contains(root)){
                parents.put(root,-1);
                dfsBridges(root,adj,visited,ids,lows,parents,bridges);
            }
        }
        return bridges;
    }

    private static void dfsBridges(Integer curr,Map<Integer,List<Integer>> adj, Set<Integer> visited, Map<Integer, Integer> ids, Map<Integer, Integer> lows, Map<Integer, Integer> parents, List<Pair<Integer, Integer>> bridges) {
        visited.add(curr);
        ids.put(curr,id);
        lows.put(curr,id);
        id += 1;
        for(Integer neighbour:adj.get(curr)){
            if(parents.get(curr).equals(neighbour)){
                ;
            }
            else if(!visited.contains(neighbour)){
                parents.put(neighbour,curr);
                dfsBridges(neighbour,adj,visited,ids,lows,parents,bridges);
                if(ids.get(curr)<lows.get(neighbour)){
                    bridges.add(new Pair<>(curr,neighbour));
                }
                lows.put(curr,Math.min(lows.get(curr),lows.get(neighbour)));
            }
            else{
                lows.put(curr,Math.min(lows.get(curr),ids.get(neighbour)));
            }
        }
    }

    private static void addEdge(Map<Integer, List<Integer>> adj, int u, int v) {
        adj.get(u).add(v);
        adj.get(v).add(u);
    }

    private static void removeEdge(Map<Integer, List<Integer>> adj, int u, int v) {
        adj.get(u).remove(new Integer(v));
        adj.get(v).remove(new Integer(u));
    }

class Pair<E,K>{
    E key ;
    K val;
    public Pair(E key, K val){
        this.key = key;
        this.val = val;
    }

    public E getKey() {
        return key;
    }

    public K getValue() {
        return val;
    }

    @Override
    public boolean equals(Object obj) {
        if (obj == null) {
            return false;
        }

        if (obj.getClass() != this.getClass()) {
            return false;
        }

        final Pair other = (Pair) obj;
        if (this.key.equals(other.key) && this.val.equals(other.val)) {
            return true;
        }

        if (this.val.equals(other.key) && this.key.equals(other.val)) {
            return true;
        }
        return false;
    }

    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * (val.hashCode() + key.hashCode());
        return result;
    }
}
