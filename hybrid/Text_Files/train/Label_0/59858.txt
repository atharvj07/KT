import java.util.Scanner;

public class CF_714_D {
    private static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        long n = scanner.nextLong();
        Borders first = new Borders(n);
        Borders second = new Borders(n, first);
        borderSearch(first, Direction.LEFT);
        borderSearch(first, Direction.RIGHT);
        borderSearch(first, Direction.BOTTOM);
        borderSearch(first, Direction.TOP);
        borderSearch(second, Direction.LEFT);
        borderSearch(second, Direction.RIGHT);
        borderSearch(second, Direction.BOTTOM);
        borderSearch(second, Direction.TOP);
        System.out.println("! " + first + " " + second);
    }


    private static void borderSearch(Borders borders, Direction direction) {
        long l, r;
        if (direction == Direction.LEFT || direction == Direction.RIGHT) {
            l = borders.l;
            r = borders.r;
        } else {
            l = borders.b;
            r = borders.t;

        }
        boolean fromLeft = direction == Direction.LEFT || direction == Direction.BOTTOM;
        while (l < r) {
            long c = (l + r + (fromLeft ? 1 : 0)) / 2;
            borders.setBorder(direction, c);
            if (borders.check()) {
                if (fromLeft) {
                    l = c;
                } else {
                    r = c;
                }
            } else {
                if (fromLeft) {
                    r = c - 1;
                } else {
                    l = c + 1;
                }
            }
        }
        borders.setBorder(direction, l);
    }

    private static class Borders {
        long l;
        long r;
        long b;
        long t;
        Borders first;

        public Borders(long n) {
            this(n, null);
        }

        public Borders(long n, Borders first) {
            l = 1;
            r = n;
            b = 1;
            t = n;
            this.first = first;
        }

        public void setBorder(Direction direction, long value) {
            switch (direction) {
                case LEFT:
                    l = value;
                    break;
                case RIGHT:
                    r = value;
                    break;
                case BOTTOM:
                    b = value;
                    break;
                case TOP:
                    t = value;
                    break;
            }
        }

        boolean check() {
            System.out.println("? " + this);
            System.out.flush();
            int count = scanner.nextInt();
            if ((first != null)
                    && (l <= first.l) && (r >= first.r)
                    && (b <= first.b) && (t >= first.t)) {
                count--;
            }
            return count > 0;
        }

        @Override
        public String toString() {
            return l + " " + b + " " + r + " " + t;
        }
    }

    private static enum Direction {
        LEFT, RIGHT, BOTTOM, TOP
    }
}
