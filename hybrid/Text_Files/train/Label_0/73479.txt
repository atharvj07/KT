n = int(input())
a = [int(t) - 1 for t in input().split()]
(l, r) = ([-1] * n, [-1] * n)
for (i, x) in enumerate(a):
	if l[x] == -1:
		l[x] = i
	r[x] = i
dp = [0] * (n + 1)
cnt = [0] * n
for i in reversed(range(n)):
	x = a[i]
	cnt[x] += 1
	dp[i] = dp[i + 1]
	if i == l[x]:
		dp[i] = max(dp[i], dp[r[x] + 1] + cnt[x])
	else:
		dp[i] = max(dp[i], cnt[x])
print(n - dp[0])
