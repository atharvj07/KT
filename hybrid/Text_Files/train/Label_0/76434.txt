t = int(input())
for _ in range(t):
	l = list()
	waits = list()
	times = list()
	n = int(input())
	for i in range(n):
		(a, b) = map(int, input().split())
		l.append((a, b))
		wait = (b - a + 1) // 2
		waits.append(wait)
	times = list(map(int, input().split()))
	current = 0
	for j in range(n):
		if j == 0:
			current = l[0][0] + times[0]
		else:
			current += times[j]
		if j == n - 1:
			print(current)
			break
		if current + waits[j] > l[j][1]:
			current += waits[j]
		else:
			current = l[j][1]
		current += l[j + 1][0] - l[j][1]
