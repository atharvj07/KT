import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n0, n1, n2) = map(int, input().split())
	l = []
	if n1 == 0:
		if n0 == 0:
			for i in range(n2 + 1):
				l.append(1)
		else:
			for i in range(n0 + 1):
				l.append(0)
	else:
		l.append(0)
		for nn0 in range(n0):
			l.append(0)
		l.append(1)
		for nn2 in range(n2):
			l.append(1)
		for nnl in range(n1 - 1):
			if l[-1] == 0:
				l.append(1)
			else:
				l.append(0)
	print(''.join(map(str, l)))
