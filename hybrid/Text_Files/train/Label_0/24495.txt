const int N = 6;
int a[N], b[N];
LD res = 0;
void add(LD p) {
	LD curr = 0;
	for (int i = 0; i < N; ++i)
		curr += min(a[i] * p, b[i] * (1 - p));
	maxi(res, curr);
}
int main() {
	for (int i = 0; i < N; ++i)
		scanf("%d", a + i);
	for (int i = 0; i < N; ++i)
		scanf("%d", b + i);
	for (int i = 0; i < N; ++i)
		if (a[i] + b[i])
			add(b[i] / ((LD)a[i] + b[i]));
	add(0);
	add(1);
	printf("%.10Lf\n", 1 - res / 100);
}
