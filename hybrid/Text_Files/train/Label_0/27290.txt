#include <iostream>
using namespace std;

int main()
{
	int w;

	cin >> w;

	while (w >= 0) {
		int f = 0;

		if (w <= 10)
			f = 1150;
		else if (w <= 20)
			f = 1150 + 125 * (w - 10);
		else if (w <= 30)
			f = 2400 + 140 * (w - 20);
		else
			f = 3800 + 160 * (w - 30);

		cout << 4280 - f << endl;

		cin >> w;
	}
	
	return 0;
}
