n = int(input())
arr = list(map(int, input().split()))
fl = False
max_ans = 0
for i in range(n):
	ans = arr[i]
	for j in range(n):
		if i != j:
			ans = ans & ~arr[j]
			if ans <= max_ans:
				break
	if ans > max_ans:
		max_ans = ans
		idx = i
		fl = True
if fl:
	(arr[0], arr[idx]) = (arr[idx], arr[0])
print(*arr)
