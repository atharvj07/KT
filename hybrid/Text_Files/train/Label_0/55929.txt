import static java.lang.System.out;

import java.util.Scanner;
public class Main {

    public static void main(String[] args) {
        Scanner cin = new Scanner (System.in);
        int d = cin.nextInt();
        int n = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++){
            a[i] = cin.nextInt();
        }
        int res = 0;
        int count = 0;
        for (int i = 0; i < n-1; i++){
            int tmp = d-a[i];
            res += tmp;
        }
        out.println(res);
    }
}
