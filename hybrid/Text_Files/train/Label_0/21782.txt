#include <bits/stdc++.h>
using namespace std;
const double _pi = acos(-1.0);
int main() {
  int n, m;
  scanf("%d %d\n", &n, &m);
  string s, t;
  getline(cin, s);
  getline(cin, t);
  int j = -1;
  for (int i = 0; i < min(n, m); i++) {
    if (s[i] == '*') {
      j = i;
      break;
    }
    if (s[i] != t[i]) {
      printf("NO\n");
      return 0;
    }
  }
  if (j == -1) {
    if (n == m || ((n == (m + 1)) && s[m] == '*')) {
      printf("YES\n");
      return 0;
    }
    printf("NO\n");
    return 0;
  }
  for (int k = m - 1, i = n - 1; i > j; i--, k--) {
    if (k < j || s[i] != t[k]) {
      printf("NO\n");
      return 0;
    }
  }
  printf("YES\n");
  return 0;
}
