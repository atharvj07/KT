#include <bits/stdc++.h>
using namespace std;
const long long int mod = 1000000007;
const long long int inf = 0x3f3f3f3f3f3f3f3fLL;
long long int i, j, k;
void solve(void);
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout << fixed << setprecision(10);
  int t = 1;
  while (t--) solve();
  return 0;
}
const double eps = 1e-9;
const double PI = acos(-1);
inline bool equal(double a, double b) { return fabs(a - b) < eps; }
void solve() {
  long long int n;
  double t;
  cin >> n >> t;
  double val[n + 1][3];
  for (i = (1); i <= (n); ++i) cin >> val[i][0] >> val[i][1] >> val[i][2];
  double curt = 0.0;
  while (curt < t) {
    double mint = t - curt;
    for (i = (1); i <= (n); ++i)
      for (j = (i + 1); j <= (n); ++j)
        if (!equal(val[i][1], val[j][1])) {
          double temp = (val[i][0] - val[j][0]) / (val[j][1] - val[i][1]);
          if (temp > 0) mint = ((mint) < (temp) ? (mint) : (temp));
        }
    for (i = (1); i <= (n); ++i) val[i][0] += val[i][1] * mint;
    for (i = (1); i <= (n); ++i)
      for (j = (i + 1); j <= (n); ++j)
        if (equal(val[i][0], val[j][0])) {
          double m1 = val[i][2], m2 = val[j][2], v1 = val[i][1], v2 = val[j][1];
          double nv1 = 0.0, nv2 = 0.0;
          nv1 = ((m1 - m2) * v1 + 2 * m2 * v2) / (m1 + m2);
          nv2 = ((m2 - m1) * v2 + 2 * m1 * v1) / (m1 + m2);
          val[i][1] = nv1;
          val[j][1] = nv2;
        }
    curt += mint;
  }
  for (i = (1); i <= (n); ++i) cout << val[i][0] << '\n';
}
