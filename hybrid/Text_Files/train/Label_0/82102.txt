def ri():
	return map(int, input().split())
(n, m, k) = ri()

def area(h, n, k):
	s = n
	if k >= h - 1:
		s += h * (h - 1) // 2
	else:
		s += k * (k + 1) // 2 + k * (h - k - 1)
	if n - k >= h - 1:
		s += (h - 1) * (h - 2) // 2
	else:
		s += (n - k - 1) * (n - k) // 2 + (n - k) * (h - n + k - 1)
	return s
h1 = 1
h2 = m
hmax = 0
hprev = 0
while True:
	hh = (h1 + h2) // 2
	if hh == hprev:
		break
	hprev = hh
	mm = area(hh, n, k)
	if mm > m:
		h2 = hh
	else:
		h1 = hh + 1
		hmax = max(hmax, hh)
print(hmax)
