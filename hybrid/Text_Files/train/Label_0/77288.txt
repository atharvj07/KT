input = stdin.readline

def log(n, m):
	(res, val) = (0, 1)
	while True:
		if n == val:
			return res
		elif n > val:
			return res - 1
		res += 1
		val *= m

def ceil(m, n):
	val = m // n
	if m % n != 0:
		return val + 1
	return val
for _ in range(int(input())):
	(n, m, k) = map(int, input().split())
	(val, rem) = (n // m, n % m)
	arr = [val] * m
	for i in range(m - 1, -1, -1):
		if rem == 0:
			break
		arr[i] += 1
		rem -= 1
	(value, newvalue) = (1, 1)
	for _ in range(k):
		value = newvalue
		for i in range(m):
			change = arr[i] == val
			j = arr[i]
			print(j, end=' ')
			(c, res) = (0, [])
			while c < j:
				res.append(str(value))
				value = value + 1
				if value == n + 1:
					value = 1
				if change:
					newvalue = value
				c += 1
			print(' '.join(res))
	print()
