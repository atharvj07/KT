n = int(input(''))
x = list(map(int, input().split()))
y = list(map(int, input().split()))
dpx = []
dpy = []
flag = 0
if x[0] != y[0] or x[n - 1] != y[n - 1]:
	flag = 1
for i in range(1, n):
	dpx.append(x[i] - x[i - 1])
	dpy.append(y[i] - y[i - 1])
dpx.sort()
dpy.sort()
if dpx != dpy:
	flag = 1
if flag:
	print('No')
else:
	print('Yes')
