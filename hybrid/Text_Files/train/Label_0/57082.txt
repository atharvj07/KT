import java.util.Scanner;

public class Main2 {
	public static final int MAXL=20000;
	public static void main(String[] args) {
		Scanner in=new Scanner(System.in);
		 int n; 
		n=in.nextInt();
		  int lat = 0; boolean valid = true;
		  for(int i = 0; i < n; i++) {
		    int v=in.nextInt();
			  String dir=in.next();
		    	char[]dirr=dir.toCharArray();
		    switch(dirr[0]) {
		      case 'N': lat -= v; break;
		      case 'S': lat += v; break;
		      default: if(lat == 0 || lat == MAXL) lat = -1;
		    }
		    if(lat < 0 || lat > MAXL) {
		      valid = false;
		      break;
		    }
		  }
		  System.out.printf("%s\n", valid && lat == 0 ? "YES" : "NO");
		  
	}
}
