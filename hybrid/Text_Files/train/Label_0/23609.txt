#include <cstdio>
#include <bitset>
#include <iostream>
using namespace std;
#define ref(i,x,y)for(int i=x;i<=y;++i)
#define def(i,x,y)for(int i=x;i>=y;--i)
const int N=4000;
const int mod=998244353;
int n,m,len,a[9][N+1],f[N+1][N+1],s[N+1],ans;
bool ex[N+1];
void inc(int d){
	ans=(ans+s[d])%mod;
}
int main(){
	s[0]=1;
	ref(i,1,N)s[i]=s[i-1]*2%mod;
	scanf("%d",&n);
	ref(i,0,n){
		char c=getchar();while(c<'0'||c>'1')c=getchar();
		m=-1;
		while(c>='0'&&c<='1')a[i][++m]=c-48,c=getchar();
		ref(j,0,m/2)swap(a[i][j],a[i][m-j]);
		while(m>=0&&!a[i][m])m--;
		if(!i){len=m;continue;}
		while(m>=0&&m<=N){
			while(m>=0&&ex[m]){
				ref(j,0,m)a[i][j]^=f[m][j];
				while(m>=0&&!a[i][m])m--;
			}
			if(m<0)break;
			ref(j,0,m)f[m][j]=a[i][j];ex[m]=1;
			def(j,m,0)a[i][j+1]=a[i][j];a[i][0]=0;m++;
		}
	}
	m=N+1;
	while(m>0&&ex[m-1])m--;
	if(m>N){printf("0\n");return 0;}
	def(i,len,m){
		if(a[0][i]==1)inc(i-m);
		if(a[8][i]!=a[0][i])ref(j,0,i)a[8][j]^=f[i][j];
	}
	bool flag=1;
	def(i,m-1,0)if(a[8][i]>a[0][i]){flag=0;break;}
		else if(a[8][i]<a[0][i])break;
	if(flag)inc(0);
	printf("%d\n",ans);
}
