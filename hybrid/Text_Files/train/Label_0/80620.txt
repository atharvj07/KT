from math import sqrt
n = int(input())
matriz = [[] for e in range(n)]
a = [[] for e in range(n)]
for i in range(n):
	linha = list(map(int, input().split()))
	for j in range(len(linha)):
		matriz[i].append(linha[j])
		if i == 0 and j == 2:
			m02 = linha[j]
		elif i == 0 and j == 1:
			m01 = linha[j]
		elif i == 1 and j == 2:
			m12 = linha[j]
a[2] = int(sqrt(m12 * m02 / m01))
a[1] = int(m12 / a[2])
a[0] = int(m02 / a[2])
for i in range(3, len(matriz[0])):
	a[i] = int(matriz[0][i] / a[0])
print(*a)
