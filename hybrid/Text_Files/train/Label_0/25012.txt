#include <bits/stdc++.h>
using namespace std;
long long power(long long a, long long b, long long m) {
  long long ans = 1;
  while (b > 0) {
    if (b % 2) {
      ans = (ans * a) % m;
    }
    a = (a * a) % m;
    b = b / 2;
  }
  return ans;
}
map<long long, long long> mp, sq, nsq;
int dp[71][1 << 19];
vector<long long> arr, a;
long long chk;
int ff(int i, int j) {
  if (i == j) {
    return 1;
  }
  return 0;
}
long long f(int i, int mask) {
  if (i == arr.size()) {
    return mask == chk;
  }
  if (dp[i][mask] != -1) {
    return dp[i][mask];
  }
  long long vv = f(i + 1, mask);
  long long val = 0;
  for (long long j = 0; j < 19; j++) {
    int x1 = ((mask & (1 << j)) != 0);
    int x2 = ((arr[i] & (1 << j)) != 0);
    if (ff(x1, x2)) {
      val |= (1 << j);
    }
  }
  vv += f(i + 1, val);
  vv %= 1000000007;
  vv *= power(2, mp[a[i]] - 1, 1000000007);
  vv %= 1000000007;
  return dp[i][mask] = vv;
}
int main() {
  vector<long long> primes;
  for (long long i = 2; i < 71; i++) {
    bool ff = false;
    for (long long j = 2; j < i; j++) {
      if (i % j == 0) {
        ff = true;
      }
    }
    if (!ff) {
      primes.push_back(i);
    }
  }
  int n;
  cin >> n;
  for (long long i = 0; i < n; i++) {
    long long xx;
    cin >> xx;
    mp[xx]++;
  }
  map<long long, long long>::iterator it;
  for (it = mp.begin(); it != mp.end(); ++it) {
    a.push_back(it->first);
    int val = 0;
    for (long long j = 0; j < 19; j++) {
      long long jj = it->first;
      int xp = 0;
      while ((jj % primes[j]) == 0) {
        jj /= primes[j];
        xp++;
      }
      if (xp % 2 == 0) {
        val |= (1 << j);
      }
    }
    arr.push_back(val);
  }
  memset(dp, -1, sizeof dp);
  chk = 1 << 19 - 1;
  cout << f(0, chk) - 1;
}
