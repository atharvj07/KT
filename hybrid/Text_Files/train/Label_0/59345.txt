import java.io.IOException;
import java.util.*;

public class D 
{
	private final static String FIRST = "FIRST";
	private final static String SECOND = "SECOND";
	private final static String DRAW = "DRAW";
	
    public static void main(String [] args) throws IOException
    {
    	Scanner in = new Scanner(System.in);
    	int x1 = in.nextInt();
    	int x2 = in.nextInt();
    	int a = in.nextInt();
    	int b = in.nextInt();
    	
    	solve(x1, x2, a, b);
    }
    
    private static void solve(int x1, int x2, int a, int b)
    {
    	int d = x2-x1;
    	
    	if(d >= a && d <= b)
    	{
    		System.out.println(FIRST);
    		System.out.println(x2);
    		return;
    	}
    	
    	if(d > 0 && a < 0 && b < 0)
    	{
    		System.out.println(DRAW);
    	}
    	else if(d < 0 && a > 0 && b > 0)
    	{
    		System.out.println(DRAW);
    		
    	}
    	else if(a <= 0 && b >= 0)
    	{
    		System.out.println(DRAW);
    	}
    	else
    	{
    		if(d%(a+b) == 0)
    			System.out.println(SECOND);
    		else
    		{
    			if(d%(a+b) >= a && d%(a+b) <= b)
    			{
    				System.out.println(FIRST);
    				System.out.println(x1 + (d%(a+b)));
    			}
    			else
    				System.out.println(DRAW);
    		}
    	}
    }
}
