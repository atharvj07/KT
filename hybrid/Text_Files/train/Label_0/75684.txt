def search_bin(A, n, key):
	left = -1
	right = n
	while right > left + 1:
		middle = (left + right) // 2
		if A[middle] > key:
			right = middle
		else:
			left = middle
	return left
n = int(input())
mas = input().split()
for i in range(n):
	mas[i] = int(mas[i])
mas.sort()
d = int(input())
res = []
for i in range(d):
	value = int(input())
	summ = search_bin(mas, n, value) + 1
	res.append(summ)
for i in range(d):
	print(res[i])
