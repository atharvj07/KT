#include <bits/stdc++.h>
using namespace std;
using pii = pair<int, int>;
vector<pii> v, w;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  int n;
  cin >> n;
  v.resize(n);
  for (int i = 0; i < n; ++i) cin >> v[i].first, v[i].second = i;
  sort(begin(v), end(v), [&](pii a, pii b) { return a.first < b.first; });
  for (int i = 0; i < n - 1; ++i) swap(v[i].first, v[(i + 1) % n].first);
  sort(begin(v), end(v), [&](pii a, pii b) { return a.second < b.second; });
  for (auto i : v) cout << i.first << ' ';
  return 0;
}
