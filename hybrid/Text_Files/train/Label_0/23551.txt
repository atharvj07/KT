using namespace std;
int n,a[200005],used[200005];char c[200005];
int v[30][200005],l[30];
inline void add(int x,int d){while(x<=n){a[x]+=d;x+=x&-x;}}
inline int get(int x){int re=0;while(x){re+=a[x];x-=x&-x;}RE re;}
signed main(){
    scanf("%s",c+1);
    n=strlen(c+1);
    int x;
	FOR(i,1,n){
		x=c[i]-'a';
		v[x][++l[x]]=i;
		a[i]=i&-i;
	}

	FOR(i,0,25)if(l[i]&1)f++;

		if(used[i])continue;
		x=c[i]-'a';
		int r=v[x][l[x]--];
		if(i==r){
			ans+=(get(n)-get(i))>>1;
		}else{
			ans+=get(n)-get(r);
		}
		used[r]=used[i]=1;add(r,-1);add(i,-1);
	}
