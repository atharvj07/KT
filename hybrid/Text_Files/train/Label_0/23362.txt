#define LL long long

#define FOR(i,s,e) for(i = s; i < e; i++)
#define FORR(i,s,e) for(i = e-1; i >= s; i--)
const LL INF = (LL)1e16;
const LL MOD = (LL)1e9 + 7;

int p,q,a,n,limit;
int dfs(int x, int y, int n, int i){
  int ans = 0;
  if (x*q > y*p) return 0;
  if (x*q == y*p) return 1;
  if (n == limit) return 0;
  for(int j = i; j*y<=a; j++){
    ans += dfs(x*j+y,y*j,n+1,j);
  }
  return ans;
}

int main(){
  while (1){
    cin >> p >> q >> a >> limit;
    if (p == 0){
      return 0;
    }
    cout << dfs(0,1,0,1) << endl;
  }
  return 0;
}
