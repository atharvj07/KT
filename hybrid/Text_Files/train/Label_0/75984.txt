from math import ceil

def sum_ap(n, a, d):
	return n * (2 * a + (n - 1) * d) // 2
for _ in range(int(input())):
	(x, n) = map(int, input().split())
	n += 1
	ans = 0
	t = ceil(n / 4)
	if n % 4 == 0:
		ans += sum_ap(t, 0, -4)
		ans += sum_ap(t, -1, -4)
		ans += sum_ap(t, 2, 4)
		ans += sum_ap(t, 3, 4)
	elif n % 4 == 1:
		ans += sum_ap(t, 0, -4)
		ans += sum_ap(t - 1, -1, -4)
		ans += sum_ap(t - 1, 2, 4)
		ans += sum_ap(t - 1, 3, 4)
	elif n % 4 == 2:
		ans += sum_ap(t, 0, -4)
		ans += sum_ap(t, -1, -4)
		ans += sum_ap(t - 1, 2, 4)
		ans += sum_ap(t - 1, 3, 4)
	else:
		ans += sum_ap(t, 0, -4)
		ans += sum_ap(t, -1, -4)
		ans += sum_ap(t, 2, 4)
		ans += sum_ap(t - 1, 3, 4)
	if x % 2 == 0:
		x += ans
	else:
		x -= ans
	print(x)
