n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for f in range(4):
	o = [-1] * n
	o[0] = f
	for i in range(n - 1):
		for j in range(4):
			if o[i] & j == b[i] and o[i] | j == a[i]:
				o[i + 1] = j
	if -1 not in o:
		print('Yes')
		print(*o)
		exit()
print('No')
