import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.math.BigInteger;
import java.util.*;

public final class Main {
    public static void main(final String... args) throws Throwable {
        final BufferedReader reader = FastLibrary.newBufferedReader(16384);
        try (final BufferedWriter writer = FastLibrary.newBufferedWriter(8192)) {
            final int t = FastLibrary.parseInt(reader.readLine());
            for (int i = 0; i < t; i++)
                solve(reader, writer);
        }
    }

    private static void solve(final BufferedReader reader, final BufferedWriter writer) throws Throwable {
        final int n;
        final int m;
        final char[][] field;
        {
            final StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            n = FastLibrary.parseInt(tokenizer.nextToken());
            m = FastLibrary.parseInt(tokenizer.nextToken());
            field = new char[n][m];
        }

        for (int i = 0; i < n; i++) {
            final String line = reader.readLine();
            for (int j = 0; j < m; j++)
                field[i][j] = line.charAt(j);
        }

        global: for (int attempt = 0; attempt < 2; attempt++) {
            char required = (attempt == 0 ? 'W' : 'R');

            final char[][] result = new char[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    final char current = field[i][j];
                    if (current != '.' && required != current) {
                        /*
                        if (n == 5 && m == 1) {
                            System.out.println("Attempt #" + attempt + " failed: " + i + " " + j);
                            System.out.println("Required: " + required);
                            System.out.println("Current: " + current);
                        }
                         */

                        continue global;
                    }

                    result[i][j] = required;
                    required = (required == 'W' ? 'R' : 'W');
                }

                required = (result[i][0] == 'W' ? 'R' : 'W');
            }

            writer.write(FastLibrary.YES);
            writer.newLine();
            for (final char[] e : result) {
                writer.write(new String(e));
                writer.newLine();
            }

            return;
        }

        writer.write(FastLibrary.NO);
        writer.newLine();
    }
}

@SuppressWarnings("unused")
final class FastLibrary {
    public static final String YES = "YES";
    public static final String NO = "NO";

    private FastLibrary() {
        throw new UnsupportedOperationException();
    }

    public static BufferedReader newBufferedReader(final int bufferSize) {
        return new BufferedReader(new InputStreamReader(System.in), bufferSize);
    }

    public static BufferedWriter newBufferedWriter(final int bufferSize) {
        return new BufferedWriter(new OutputStreamWriter(System.out), bufferSize);
    }

    public static StringTokenizer spaceSplitter(final String s) {
        return new StringTokenizer(s, " ");
    }

    public static void repeat(int times, final Runnable runnable) {
        while (times-- > 0)
            runnable.run();
    }

    public static long parseLong(final String token) {
        long result = 0L;

        if (token.isEmpty())
            return result;

        final boolean negative = token.charAt(0) == '-';
        for (int i = (negative ? /* skipping minus */ 1 : 0); i < token.length(); ++i) {
            final char currentChar = token.charAt(i);
            if (currentChar < '0' || currentChar > '9')
                throw new NumberFormatException("Expected a digit, found this: '" + currentChar + "' (i=" + i + ")");

            result += BigInteger.valueOf(token.charAt(i) - '0')
                    .multiply(BigInteger.valueOf((long) Math.pow(10, token.length() - i - 1))).longValue();
        }

        return negative ? -result : result;
    }

    public static int parseInt(final String token) {
        final long result = parseLong(token);

        if (result < Integer.MIN_VALUE || result > Integer.MAX_VALUE)
            throw new IllegalArgumentException("Provided argument is a number, " +
                    "but it can't be converted to an Integer due to its size: " + result);

        return (int) result;
    }
}
