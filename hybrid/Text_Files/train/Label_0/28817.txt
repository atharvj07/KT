#include <bits/stdc++.h>
using namespace std;
int n, m, c, h, Max1 = -2e9, Min1 = 2e9, Max2 = -2e9, Min2 = 2e9, ans = 2e9,
                pos;
int main() {
  scanf("%d%d", &n, &m);
  scanf("%d", &c);
  for (int i = 1; i <= c; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    Max1 = max(Max1, x + y);
    Min1 = min(Min1, x + y);
    Max2 = max(Max2, x - y);
    Min2 = min(Min2, x - y);
  }
  scanf("%d", &h);
  for (int i = 1; i <= h; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    int res = max(max(abs(Max1 - (x + y)), abs(Min1 - (x + y))),
                  max(abs(Max2 - (x - y)), abs(Min2 - (x - y))));
    if (ans > res) {
      ans = res;
      pos = i;
    }
  }
  printf("%d\n%d", ans, pos);
  return 0;
}
