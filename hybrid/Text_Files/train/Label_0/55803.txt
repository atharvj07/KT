import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Objects;
import java.util.Random;
import java.util.StringTokenizer;

public class Solution{
	
	
	
	public static void main(String[] args) {

		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		
		int tt = fs.nextInt();
		while(tt-->0) {

			int r = fs.nextInt(), c = fs.nextInt(), k = fs.nextInt();
			char[][] arr = new char[r][];
			int[][] ans = new int[r][c];
			
			int count = 0;
			
			for(int i=0;i<r;i++) {
				arr[i] = fs.next().toCharArray();
				for(int j=0;j<c;j++) {
					count += (arr[i][j]=='R')?1:0;
				}
			}
			
			int num = count - k*(count/k);
			
			
			int chicken = 0;
			int rice = 0;
			
			for(int i=0;i<r;i++) {
				for(int j=0;j<c;j++) {
					int ind = (i%2==0)?j:c-j-1;
					ans[i][ind] = chicken;
					rice += (arr[i][ind]=='R')?1:0;
					
					if((chicken>=num && rice==count/k) || (rice==(count+k-1)/k)) {
						chicken++;
						rice = 0;
						if(chicken==k)
							chicken--;
					}
					
				}
			}
			
			for(int i=0;i<r;i++) {
				for(int j=0;j<c;j++) {
					out.print(getChar(ans[i][j]));
				}
				out.println("");
			}
			
			
		}
		
		out.close();
		
	}
	
	
	static char getChar(int x) {
		if(x<10) return (char)('0'+ x);
		else if(x<36) return (char)('a'+x-10);
		return (char)('A' + x - 10 - 26);
	}
	
	
	
	
	static final Random random=new Random();
	
	static void ruffleSort(int[] a) {
		int n=a.length;//shuffle, then sort 
		for (int i=0; i<n; i++) {
			int oi=random.nextInt(n), temp=a[oi];
			a[oi]=a[i]; a[i]=temp;
		}
		Arrays.sort(a);
	}
	
	
	static class FastScanner{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");

		public String next(){
			while(!st.hasMoreElements()){
				try{
					st = new StringTokenizer(br.readLine());
				} catch(IOException e){
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}
		
		public String nextLine() throws IOException {
			return br.readLine();
		}
		
		public int nextInt(){
			return Integer.parseInt(next());
		}

		public int[] readArray(int n){
			int[] a = new int[n];
			for(int i=0;i<n;i++)
				a[i] = nextInt();
			return a;
		}
		
		public long nextLong() {
			return Long.parseLong(next());
		}
	}
	
}
