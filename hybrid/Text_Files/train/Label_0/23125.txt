#include <iostream>
#include <algorithm>
#include <vector>
#include <map>
#include <tuple>

using namespace std;

class Solver {
};

signed main() {
	while (true) {
		int n;
		cin >> n;
		if (n == 0)break;

		vector<int> hs(n + 1, 0);
		for (int i = 0; i < n + 1; i++) {
			cin >> hs[i];
		}

		for (int i = 0; i < n + 1; i++) {
			vector<int> remain;
			for (int j = 0; j < n + 1; j++) {
				if (i == j)continue;

				remain.emplace_back(hs[j]);
			}
			vector<int> deltas;
			for (int j = 1; j < n; j++) {
				deltas.emplace_back(remain[j] - remain[j - 1]);
			}
			sort(deltas.begin(), deltas.end());

			if (deltas[0] == deltas[deltas.size() - 1]) {
				cout << hs[i] << endl;
				break;
			}
		}
	}
}
