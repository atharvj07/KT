#include <bits/stdc++.h>
using namespace std;
int n, arr[505];
int dp[505][505];
int solve(int s, int e) {
  if (e < s) return 0;
  if (e - s + 1 == 1) return 1;
  if (dp[s][e] != -1) return dp[s][e];
  int ans = 1 + solve(s + 1, e);
  if (s + 1 <= e && arr[s] == arr[s + 1]) ans = min(ans, 1 + solve(s + 2, e));
  for (int i = s + 2; i <= e; i++)
    if (arr[s] == arr[i]) ans = min(ans, solve(s + 1, i - 1) + solve(i + 1, e));
  return dp[s][e] = ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> arr[i];
  memset(dp, -1, sizeof(dp));
  cout << solve(1, n);
  return 0;
}
