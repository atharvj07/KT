#include <bits/stdc++.h>
using namespace std;
int dp[105][105][105];
string s;
int n, k;
void update(int i, int mx, int cnt) { dp[i][mx][cnt] = 1; }
void solve() {
  cin >> n >> k;
  cin >> s;
  dp[0][0][0] = 1;
  for (int i = (0); i < (n); ++i) {
    for (int mx = (0); mx < (k + 1); ++mx) {
      for (int cnt = (0); cnt < (k + 1); ++cnt) {
        if (!dp[i][mx][cnt]) continue;
        if (s[i] == 'Y' || s[i] == '?') {
          update(i + 1, mx, 0);
        }
        if (s[i] == 'N' || s[i] == '?') {
          update(i + 1, max(mx, cnt + 1), cnt + 1);
        }
      }
    }
  }
  bool ans = 0;
  for (int i = (0); i < (k + 1); ++i) {
    ans |= dp[n][k][i];
  }
  cout << (ans ? "YES" : "NO") << '\n';
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int tt = 1;
  for (int i = 1; i <= tt; ++i) {
    solve();
  }
  return 0;
}
