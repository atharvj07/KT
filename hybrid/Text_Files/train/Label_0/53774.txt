import java.util.*;
public class a {
	public static void main(String args[]) {
		Scanner in = new Scanner(System.in);
		int N = in.nextInt();
		int M = in.nextInt();
		int X = in.nextInt();
		int Y = in.nextInt();
	
		int[] soldiers = new int[N];
		
		for(int i = 0; i < N; i++)
			soldiers[i] = in.nextInt();
		
		int at = 0;
		int count = 0;
		ArrayList<Integer> s = new ArrayList<Integer>();
		ArrayList<Integer> v = new ArrayList<Integer>();
		for(int i = 0; i < M; i++) {
			int vest = in.nextInt();
			if(at == N)
				continue;
			while(soldiers[at] + Y < vest && at + 1 < N)
				at++;
			if(soldiers[at] - X <= vest && soldiers[at] + Y >= vest) {
				s.add(at+1);
				v.add(i+1);
				at++;
			}
		}
		
		System.out.println(s.size());
		for(int i = 0; i < s.size(); i++) {
			System.out.println(s.get(i).intValue() + " " + v.get(i).intValue());
		}
	}
}
