n = int(input())
l1 = []
for i in range(n):
	d = {}
	l = []
	(a, b, c) = input().split()
	(a, b, c) = (int(a), int(b), int(c))
	c1 = c
	s = input()
	s = s[:len(s) - 1]
	z = len(s)
	k = ''
	p = 0
	for j in range(z):
		if s[j] != k:
			d[j] = s[j]
			k = s[j]
	for x in d:
		l.append(x)
	l = l[::-1]
	for p in l:
		if d[p] == 'A':
			if c1 - a >= 0:
				c1 = c1 - a
				z = p
			else:
				break
		elif c1 - b >= 0:
			c1 = c1 - b
			z = p
		else:
			break
	l1.append(z + 1)
for i in l1:
	print(i)
