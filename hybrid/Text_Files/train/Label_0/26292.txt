#include <bits/stdc++.h>
using namespace std;
int dp[110000];
int bin_search(int x, int s, int t) {
  if (s == t) {
    return s;
  }
  int mid = (s + t) / 2;
  if (x < dp[mid]) {
    return bin_search(x, s, mid);
  } else {
    return bin_search(x, mid + 1, t);
  }
}
int main() {
  int n;
  while (scanf("%d", &n) != EOF) {
    int a;
    scanf("%d", &a);
    int len = 1;
    dp[1] = a;
    for (int i = 1; i < n; i++) {
      scanf("%d", &a);
      if (a > dp[len]) {
        dp[++len] = a;
      } else {
        int k = bin_search(a, 1, len);
        dp[k] = a;
      }
    }
    printf("%d\n", len);
  }
}
