for i in range(int(input())):
	(n, k) = [int(i) for i in input().split(' ')]
	arr = [int(i) for i in input().split(' ')]
	s = set(arr)
	if len(s) > k:
		print(-1)
	else:
		temp = ' '.join([str(i) for i in s] + ['1'] * (k - len(s)))
		result = ''
		print(n * k)
		for i in range(n):
			result += temp + ' '
		print(result)
