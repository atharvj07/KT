#include <bits/stdc++.h>
const int md = 1e9 + 7;
const long long hs = 199;
using namespace std;
vector<int> a(24);
set<int> s;
bool check() {
  for (int i = 0; i < 24; i += 4) {
    s.clear();
    for (int j = i; j < i + 4; j++) s.insert(a[j]);
    if (s.size() > 1) return false;
  }
  return true;
}
void r(vector<int> &v) {
  for (int i = 0; i < v.size() - 1; i++) swap(a[v[i]], a[v[i + 1]]);
}
int main() {
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  for (int i = 0; i < 24; i++) cin >> a[i];
  vector<int> t1 = {1, 3, 5, 7, 9, 11, 22, 20};
  vector<int> t2 = {2, 3, 16, 18, 9, 8, 15, 13};
  vector<int> t3 = {14, 15, 6, 7, 18, 19, 22, 23};
  r(t1), r(t1);
  if (check()) return cout << "YES" << '\n', 0;
  r(t1), r(t1);
  r(t1), r(t1);
  if (check()) return cout << "YES" << '\n', 0;
  r(t1), r(t1);
  r(t2), r(t2);
  if (check()) return cout << "YES" << '\n', 0;
  r(t2), r(t2);
  r(t2), r(t2);
  if (check()) return cout << "YES" << '\n', 0;
  r(t2), r(t2);
  r(t3), r(t3);
  if (check()) return cout << "YES" << '\n', 0;
  r(t3), r(t3);
  r(t3), r(t3);
  if (check()) return cout << "YES" << '\n', 0;
  r(t3), r(t3);
  return cout << "NO" << '\n', 0;
}
