#include <bits/stdc++.h>
using namespace std;
pair<long long int, long long int> v[3005];
long long int n;
long long int cache[3005][3005];
long long int vis[3005][3005];
long long int dp(long long int idx, long long int prev) {
  if (idx > n) {
    return 0;
  }
  long long int &ans = cache[idx][prev];
  if (vis[idx][prev]) return cache[idx][prev];
  vis[idx][prev] = 1;
  ans = 1e18;
  ans = min({ans, v[idx].second + dp(idx + 1, idx),
             abs(v[idx].first - v[prev].first) + dp(idx + 1, prev)});
  return ans;
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int t = 1;
  while (t--) {
    cin >> n;
    for (long long int i = 1; i <= n; i++) {
      long long int x, y;
      cin >> x >> y;
      v[i].first = x;
      v[i].second = y;
    }
    sort(v + 1, v + n + 1);
    cout << dp(2, 1) + v[1].second;
  }
  return 0;
}
