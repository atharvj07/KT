n = int(input())
l = list(map(int, input().split()))
m = [0 for i in range(11)]
varr = 0
for i in range(n):
	m[l[i]] += 1
	listt = [m[i] for i in range(11) if m[i] != 0]
	listt = sorted(listt)
	if len(listt) == 1:
		varr = i + 1
	if len(listt) > 1:
		if listt[0] == 1 and sum(listt) == 1 + len(listt) * listt[1] - listt[1]:
			varr = i + 1
		elif listt[0] == listt[-2] and listt[-1] == listt[0] + 1:
			varr = i + 1
print(varr)
