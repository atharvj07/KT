t = int(input())
for _ in range(t):
	r = {}
	x = 0
	y = 0
	n = int(input())
	s = input()
	r[x, y] = 0
	ans = []
	for i in range(len(s)):
		if s[i] == 'U':
			y += 1
		elif s[i] == 'D':
			y -= 1
		elif s[i] == 'R':
			x += 1
		elif s[i] == 'L':
			x -= 1
		if (x, y) in r.keys():
			ans.append([i - r[x, y], r[x, y] + 1, i + 1])
			r[x, y] = i + 1
		else:
			r[x, y] = i + 1
	if len(ans) == 0:
		print('-1')
	else:
		print(*min(ans)[1:])
