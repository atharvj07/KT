from collections import deque

def solve(pairs, target):

	def _bfs(src):
		q = deque()
		q.append(src)
		while len(q):
			node = q.popleft()
			if node == target:
				print('YES')
				return
			if node > target:
				print('NO')
				return
			neighbor = pairs[node - 1]
			q.append(neighbor + node)
	_bfs(1)
n = list(map(int, input().split(' ')))
target = n[1]
pairs = list(map(int, input().split(' ')))
solve(pairs, target)
