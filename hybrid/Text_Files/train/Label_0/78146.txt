x = int(input())
arr = [int(x) for x in input().split()]
ans = 0
j = 0
count = 0
i = 1
while i < x - 1:
	count = 0
	if arr[i - 1] == arr[i] - 1:
		if arr[i - 1] == 1:
			count = count + 1
		j = i
		while arr[j] - 1 == arr[j - 1]:
			if arr[j] == 1000:
				count = count + 1
			count = count + 1
			if j == x - 1:
				break
			j = j + 1
	if count >= 2:
		i = j
		if count > ans:
			ans = count - 1
	else:
		i += 1
if x == 1000:
	ans = 0
if ans == 0 and x > 1:
	if arr[0] == 1 and arr[1] == 2 or (arr[x - 2] == 999 and arr[x - 1] == 1000):
		ans = 1
print(ans)
