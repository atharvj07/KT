#include <bits/stdc++.h>
using namespace std;
char str[505][505];
int n;
bool f(int x, int y) {
  if (x >= 1 && x <= n && y >= 1 && y <= n && str[x][y] == 'X') return true;
  return false;
}
int main() {
  int i, j;
  cin >> n;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= n; j++) cin >> str[i][j];
  }
  int ans = 0;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= n; j++) {
      if (f(i, j) && f(i - 1, j - 1) && f(i - 1, j + 1) && f(i + 1, j - 1) &&
          f(i + 1, j + 1))
        ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
