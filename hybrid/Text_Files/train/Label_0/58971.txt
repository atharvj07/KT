import java.util.Scanner;

public class A {

	public static void main(String[] args) {
		
		Scanner in = new Scanner(System.in);
		
		
		int n = in.nextInt();
		
		int ar[] = new int [n];
		
		int min = 10000;
		int max = -1;
		
		for(int i = 0;i<n;i++){
			int  st = in.nextInt();
			ar[i] = st;
			
			if(st>max){
				max = st;
			}
			
			 if(st<min){
				min = st;
			}
		}
				
		int cost = 10000000;
		int er = 1;
						
		for( int i = min; i<= max;i++){
			
			int tcost = 0;
			
			for(int j:ar){
				
				
				int temp = Math.abs(i-j);
				if( temp >1){
					tcost+= temp-1;
				}
											
			}
									
			if( tcost <cost ){
				er = i;
				cost = tcost;
				
			}
			
		}
						
		System.out.println(er+" "+cost);
		
		

	}

}
