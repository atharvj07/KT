sys.setrecursionlimit(10 ** 5)

def rarray():
	return [int(i) for i in input().split()]
(n, m) = rarray()
mod = 998244353
a = [rarray() for i in range(n)]
a = list(zip(*a))
fac = [[0] * (n + 1) for i in range(n + 1)]
fac[0][0] = 1
for i in range(1, n + 1):
	fac[i][0] = 1
	for j in range(1, i + 1):
		fac[i][j] = fac[i][j - 1] * (i - j + 1) % mod
ans = 0
for i in range(m):
	a[i] = sorted(a[i])
	j = 0
	res = 1
	while j < n:
		k = j
		while j < n and a[i][j] == a[i][k]:
			j += 1
		t = j - k
		res = res * fac[a[i][k] - 1 - k][t] % mod
	ans = (ans + fac[n][n] - res + mod) % mod
print(ans * pow(fac[n][n], mod - 2, mod) % mod)
