inf = float('inf')
n = int(input())
a = [-inf] + [int(i) for i in input().split()] + [-inf]
for x in (0, 1):
	stack = [0]
	ns = [-1] * (n + 2)
	for i in range(n, -1, -1):
		while a[i] < a[stack[-1]]:
			ns[stack.pop()] = i
		stack += [i]
	res = [0] * (n + 2)
	for i in range(1, len(a)):
		if ns[i] == 0:
			res[i] = i * a[i]
		else:
			res[i] = res[ns[i]] + a[i] * (i - ns[i])
	a = a[::-1]
	if x:
		right = res[:][::-1]
	else:
		left = res[:]
a = a[1:-1]
left = left[1:-1]
right = right[1:-1]
su = [i + j - k for (i, j, k) in zip(left, right, a)]
ma = su.index(max(su))
for i in range(ma + 1, n):
	a[i] = min(a[i], a[i - 1])
for i in range(ma - 1, -1, -1):
	a[i] = min(a[i], a[i + 1])
print(*a)
