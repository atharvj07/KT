def inter(x1, x2, x3, x4):
	if x1 > x4 or x3 > x2:
		return None
	return (max(x1, x3), min(x2, x4))

def square(rectangle):
	if None in rectangle:
		return 0
	(x1, x2) = rectangle[0]
	(y1, y2) = rectangle[1]
	return abs(x2 - x1) * abs(y2 - y1)
(x1, y1, x2, y2) = [*map(int, input().split())]
(x3, y3, x4, y4) = [*map(int, input().split())]
(x5, y5, x6, y6) = [*map(int, input().split())]
wb1 = (inter(x1, x2, x3, x4), inter(y1, y2, y3, y4))
wb2 = (inter(x1, x2, x5, x6), inter(y1, y2, y5, y6))
if None in wb1 or None in wb2:
	inter_wb_12 = (None, None)
else:
	inter_wb_12 = (inter(wb1[0][0], wb1[0][1], wb2[0][0], wb2[0][1]), inter(wb1[1][0], wb1[1][1], wb2[1][0], wb2[1][1]))
sq = square(((x1, x2), (y1, y2))) - square(wb1) - square(wb2) + square(inter_wb_12)
print('YES' if sq > 0 else 'NO')
