def vec(x):
	if x == 0:
		return [[1]]
	x -= 1
	s = vec(x)
	y = vec(x)
	for i in range(2 ** x):
		for j in range(2 ** x):
			y[i][j] = -y[i][j]
	a = [s[i] + y[i] for i in range(2 ** x)]
	b = [s[i] + s[i] for i in range(2 ** x)]
	return a + b
x = int(input())
s = vec(x)
for i in range(2 ** x):
	res = ''
	for j in range(2 ** x):
		res += '+' if s[i][j] == 1 else '*'
	print(res)
