#include <bits/stdc++.h>

using namespace std;

int N, M;
queue<int> GasStationLane[10];

int findLessCarsLane() {
	unsigned int minNum, laneNo;

	minNum = GasStationLane[0].size();
	laneNo = 0;
	for (int i = 1; i < N; ++i) {
		if (GasStationLane[i].size() < minNum) {
			minNum = GasStationLane[i].size();
			laneNo = i;
		}
	}

	return laneNo;
}

int main() {
	int cmd, val, laneNo;

	cin >> N >> M;
	for (int i = 0; i < M; ++i) {
		cin >> cmd >> val;
		if (cmd == 0) {
			--val;
			cout << GasStationLane[val].front() << endl;
			GasStationLane[val].pop();
			continue;
		}
		laneNo = findLessCarsLane();
		GasStationLane[laneNo].push(val);
	}

	return 0;
}
