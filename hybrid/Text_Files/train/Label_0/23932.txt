#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
using namespace std;
const long long INF = 1e18;
struct BIT {
  vector<long long> bit;
  long long n;
  BIT(long long n) {
    this->n = n + 1;
    bit.assign(n + 1, 0);
  }
  void build(vector<long long> &a) {
    for (size_t i = 0; i < a.size(); i++) add(i, a[i]);
  }
  void add(long long idx, long long val) {
    for (++idx; idx < n; idx += idx & -idx) bit[idx] += val;
  }
  void range_add(long long l, long long r, long long val) {
    add(l, val);
    add(r + 1, -val);
  }
  long long point_query(long long idx) {
    long long ret = 0;
    for (++idx; idx > 0; idx -= idx & -idx) ret += bit[idx];
    return ret;
  }
};
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long n;
  string s;
  cin >> n >> s;
  vector<long long> a(n, 0);
  BIT bit(n);
  bit.build(a);
  for (long long i = 0; i < n; i++) {
    bit.range_add(i, i, 0);
  }
  string rs = s;
  reverse(rs.begin(), rs.end());
  set<long long> adj[26];
  for (long long i = 0; i < n; i++) {
    adj[s[i] - 'a'].insert(i);
  }
  long long ans = 0;
  for (long long i = 0; i < n; i++) {
    long long idx = *adj[rs[i] - 'a'].begin();
    adj[rs[i] - 'a'].erase(idx);
    long long crd = idx - bit.point_query(idx);
    bit.range_add(idx + 1, n - 1, 1);
    ans += crd;
  }
  cout << ans << '\n';
  return 0;
}
