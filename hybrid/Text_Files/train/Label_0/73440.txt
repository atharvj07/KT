a = int(input())
countl = []
for i in range(a):
	(n, l, r, k) = input().split(' ')
	(n, l, r, k) = (int(n), int(l), int(r), int(k))
	price = []
	b = input().split(' ')
	c = []
	for i in b:
		c.append(int(i))
	price = c
	count = 0
	i = 0
	price.sort()
	while k > 0:
		if l - 1 < price[i] and r + 1 > price[i]:
			k = k - price[i]
			if k > -1:
				count += 1
		i += 1
		if i == len(price):
			break
	countl.append(count)
for j in countl:
	print(j)
