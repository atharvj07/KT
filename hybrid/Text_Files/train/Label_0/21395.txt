#include <bits/stdc++.h>
using namespace std;
int n;
char first[1000005];
char second[1000005];
int main() {
  int loop, index, fStart, sStart, fLength, sLength, fFinished, sFinished;
  scanf("%s %s", first, second);
  index = 0;
  fStart = -1;
  sStart = -1;
  fLength = 0;
  sLength = 0;
  fFinished = 0;
  sFinished = 0;
  while (fFinished + sFinished < 2) {
    if (fFinished == 0) {
      if (first[index] == '\0')
        fFinished = 1;
      else if (fStart == -1) {
        if (first[index] != '0') {
          fStart = index;
          fLength = 1;
        }
      } else {
        fLength++;
      }
    }
    if (sFinished == 0) {
      if (second[index] == '\0')
        sFinished = 1;
      else if (sStart == -1) {
        if (second[index] != '0') {
          sStart = index;
          sLength = 1;
        }
      } else {
        sLength++;
      }
    }
    index++;
  }
  if (fLength > sLength)
    printf(">\n");
  else if (fLength < sLength)
    printf("<\n");
  else {
    for (loop = 0; loop < fLength; loop++) {
      if (first[fStart + loop] > second[sStart + loop]) {
        printf(">\n");
        return 0;
      } else if (first[fStart + loop] < second[sStart + loop]) {
        printf("<\n");
        return 0;
      }
    }
    printf("=\n");
  }
  return 0;
}
