#include <bits/stdc++.h>
using namespace std;
const int MAX = 3e5 + 10;
const int inf = 1e9 + 10;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a, b;
    a.clear();
    b.clear();
    for (int i = 0; i < n; i++) {
      long long j;
      cin >> j;
      a.push_back(j);
    }
    sort(a.begin(), a.end());
    long long num = a[0] * a[n - 1];
    for (long long i = 2; i * i <= num; i++) {
      if (num % i == 0) {
        b.push_back(i);
        if (i * i != num) b.push_back(num / i);
      }
    }
    sort(b.begin(), b.end());
    bool check = true;
    if ((int)a.size() != (int)b.size()) {
      cout << -1 << endl;
      check = false;
    }
    for (int i = 0; i < n && check; i++)
      if (a[i] != b[i]) {
        cout << -1 << endl;
        check = false;
      }
    if (check) cout << num << endl;
  }
  return 0;
}
