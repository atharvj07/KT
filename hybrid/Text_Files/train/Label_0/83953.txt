list_ans = []
for _ in range(int(input())):
	n = int(input())
	s = input()
	a = s.count('0')
	if n % 2 == 0:
		if a != 0:
			list_ans.append('BOB')
		else:
			list_ans.append('DRAW')
	elif a == 0:
		list_ans.append('DRAW')
	elif a > 1 and s[(n - 1) // 2] != '1':
		list_ans.append('ALICE')
	else:
		list_ans.append('BOB')
for item in list_ans:
	print(item)
