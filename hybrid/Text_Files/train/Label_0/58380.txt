
import java.util.*;
import java.io.*;
import static java.util.Arrays.*;
import static java.util.Collections.*;
import static java.lang.Math.*;

public class E_ {

    int INF = 1 << 28;
    //long INF = 1L << 62;
    double EPS = 1e-10;
    int n, q;
    long t;
    int[] a, n_a, to;
    int[] dp;
    boolean[] visited;
    int MOD = (int)1e9 + 7;
    
    void run() {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt(); q = sc.nextInt(); t = sc.nextInt();
        a = new int[n]; n_a = new int[n]; to = new int[n];
        for(int i=0;i<n;i++) a[i] = sc.nextInt();
        fill(n_a, -1);
        
        fill(to, -1);
        for(int i=0;i<q;i++) {
            int b = sc.nextInt()-1, c = sc.nextInt()-1;
            to[c] = b;
        }
        
        visited = new boolean[n];
        boolean flg = true;
        for(int i=0;i<n;i++) if(!visited[i]){
            n_a[i] = dfs(i);
            t += n_a[i];
            if(n_a[i] == -1) flg = false;
        }
        
//      debug(to);
//      debug(a);
//      debug(n_a);
        if(!flg || t < 0) {
            System.out.println(0);
            return;
        }
        if( t > 100000 ) {
            System.out.println(-1);
            return;
        }
        
        visited = new boolean[]{};
        a = new int[]{}; to = new int[]{};
        dp = new int[(int)t+1];
        dp[0] = 1;
        for(int j=0;j<n;j++) for(int i=0;i<t;i++) if( i+n_a[j] <= t ) {
            dp[i+n_a[j]] = (dp[i+n_a[j]] + dp[i]) % MOD;
        }
//      debug(t, dp);
        System.out.println(dp[(int)t]);
    }
    
    int dfs(int p) {
        if( visited[p] ) {
            t -= n_a[p];
            return n_a[p];
        }
        visited[p] = true;
        if( to[p] == -1 ) {
            t -= a[p];
            return n_a[p] = a[p];
        }
        int sum = dfs(to[p]);
        n_a[p] = (sum!=-1? sum+a[p]: -1);
        t -= n_a[p];
        return n_a[p];
    }

    void debug(Object... os) {
        System.err.println(Arrays.deepToString(os));
    }

    public static void main(String[] args) {
        new E_().run();
    }
}
