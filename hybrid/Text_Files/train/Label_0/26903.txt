#include<bits/stdc++.h>
using namespace std;

int N , K , M , C[303][303] , f[303][303] , sum[303][303]; bool visf[303][303] , vissum[303][303];

int getf(int , int);
int getsum(int , int);

int main(){
	cin >> N >> K >> M;
	for(int i = 0 ; i <= N ; ++i){
		C[i][0] = 1;
		for(int j = 1 ; j <= i ; ++j)
			C[i][j] = (C[i - 1][j] + C[i - 1][j - 1]) % M;
	}
	cout << getf(N + 1 , 0) << endl; return 0;
}

int getf(int x , int y){
	if(y > K) return 0;
	if(visf[x][y]) return f[x][y];
	visf[x][y] = 1; if(x == 1) return f[x][y] = 1;
	for(int i = 1 ; i < x ; ++i)
		f[x][y] = (f[x][y] + 1ll * C[x - 2][i - 1] * getf(x - i , y) % M * getsum(i , y + 1)) % M;
	return f[x][y];
}

int getsum(int x , int y){
	if(y > K) return 0;
	if(vissum[x][y]) return sum[x][y];
	vissum[x][y] = 1; return sum[x][y] = (getsum(x , y + 1) + getf(x , y)) % M;
}
