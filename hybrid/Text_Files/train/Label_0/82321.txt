import sys

def input():
	return sys.stdin.readline().strip()

def iinput():
	return int(input())

def rinput():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))
mod = int(1000000000.0) + 7
(n, s, t) = rinput()
p = get_list()
pos = s
visited = [0 for _ in range(n)]
flag = ans = 0
while pos != t:
	pos = p[pos - 1]
	ans += 1
	if visited[pos - 1] == 1:
		break
	visited[pos - 1] = 1
if pos == t:
	print(ans)
else:
	print(-1)
