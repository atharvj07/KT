p = 998244353
l = []
n = int(input())
for i in range(n):
	la = list(map(int, input().split()))
	l.append(la[1:])
hash = [0] * 1001001
for i in l:
	for j in i:
		hash[j] += 1
ans = 0
for i in range(n):
	ans = (ans + sum([hash[a] for a in l[i]]) * pow(len(l[i]), p - 2, p)) % p
ans = ans * pow(n * n, p - 2, p) % p
