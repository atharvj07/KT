#include <bits/stdc++.h>
using namespace std;

struct l{int x,y;};
int pw(int n,int k){
  assert(k>=0);
  int res=1;
  while(k){
    if(k&1)res*=n;
    n*=n;
    k>>=1;
  }
  return res;
}
template<typename T>
void chmax(T &a,T b){
  if(a<b)a=b;
}

signed main(){
  int n;cin>>n;
  vector<l> spe[n];
  for(int i=0;i<n;i++){
    int a;cin>>a;
    while(a--){
      int x,y;cin>>x>>y;x--;
      spe[i].push_back({x,y});
    }
  }
  int ans=0;
  for(int i=0;i<pw(2,n);i++){
    bool f=true;
    int cnt=0;
    for(int j=0;j<n;j++){
      if(((i>>j)&1)==0)continue;
      cnt++;
      for(l p:spe[j])if(p.y!=((i>>p.x)&1))f=false;
    }
    if(f)chmax(ans,cnt);
  }
  cout<<ans<<endl;
}
