from collections import defaultdict, deque

def is_possible(odd, even):
	each = odd * 2
	if even % each == 0:
		return True
	return False

s = sys.stdin.readline()[:-1]

for i in range(n):
	dic[s[i]] += 1

for i in dic:
	if dic[i] % 2 != 0:
		odd += 1

			cnt += 1

	print(''.join((x for x in ans)))

	if is_possible(odd, even):
		break
