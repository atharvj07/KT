import java.util.Scanner;
public class test {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x;
        int c25=0;
        int c50=0;
        boolean s=true;
        while(n--!=0){
            x=in.nextInt();
            if(s){
               if(x==25)
                  c25++;
               else if(x==50){
                   c25--;
                   c50++;
               }
               else {
                  if(c50>0){
                      c50--;
                      c25--;
                  }
                  else c25-=3;
                }
                  if(c25<0)s=false;
            }
        }
        if(s==true)System.out.println("YES");
        else System.out.println("NO");
    }
}
