#include <bits/stdc++.h>
using namespace std;
inline char nc() {
  static char buf[100000], *p1 = buf, *p2 = buf;
  return p1 == p2 && (p2 = (p1 = buf) + fread(buf, 1, 100000, stdin), p1 == p2)
             ? EOF
             : *p1++;
}
template <typename T = int>
inline T nxt() {
  char c = nc();
  T x = 0;
  int f = 1;
  for (; c > '9' || c < '0'; c = nc())
    if (c == '-') f = -1;
  for (; c >= '0' && c <= '9'; x = x * 10 + c - '0', c = nc())
    ;
  x *= f;
  return x;
}
const int N = 1010;
vector<int> G[N];
int n, m, s, t, distS[N], distT[N], a[N][N];
bool vis[N];
int main() {
  cin.sync_with_stdio(0);
  cin.tie(0);
  n = nxt(), m = nxt(), s = nxt() - 1, t = nxt() - 1;
  for (int i = 0; i < (m); ++i) {
    int u = nxt() - 1, v = nxt() - 1;
    G[u].push_back(v);
    G[v].push_back(u);
    a[u][v] = a[v][u] = 1;
  }
  priority_queue<pair<int, int> > PQ;
  PQ.push({0, s});
  memset(distS, 0x3f, sizeof distS);
  memset(vis, 0, sizeof vis);
  distS[s] = 0;
  while (!PQ.empty()) {
    int u = PQ.top().second;
    PQ.pop();
    if (vis[u]) continue;
    vis[u] = true;
    for (auto& v : G[u]) {
      if (distS[v] > distS[u] + 1) {
        distS[v] = distS[u] + 1;
        PQ.push({-distS[v], v});
      }
    }
  }
  PQ.push({0, t});
  memset(distT, 0x3f, sizeof distT);
  memset(vis, 0, sizeof vis);
  distT[t] = 0;
  while (!PQ.empty()) {
    int u = PQ.top().second;
    PQ.pop();
    if (vis[u]) continue;
    vis[u] = true;
    for (auto& v : G[u]) {
      if (distT[v] > distT[u] + 1) {
        distT[v] = distT[u] + 1;
        PQ.push({-distT[v], v});
      }
    }
  }
  int D = distS[t], ans = 0;
  for (int u = 0; u < (n); ++u) {
    for (int v = 0; v < (u); ++v) {
      if (!a[u][v] && distS[u] + distT[v] + 1 >= D &&
          distS[v] + distT[u] + 1 >= D)
        ++ans;
    }
  }
  cout << ans << '\n';
}
