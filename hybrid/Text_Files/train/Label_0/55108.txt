import java.io.*;
import java.util.*;

public class A {
	public static void main (String[] args) { new A(); }

	A() {
		Scanner s = new Scanner(System.in);
		System.err.println("");

		int numAs = 300 - ask(s, build('a', 300));
		if(numAs == 300) return;
		int numBs = 300 - ask(s, build('b', 300));
		if(numBs == 300) return;
		int N = numAs+numBs;
		char[] res = new char[N];
		Arrays.fill(res, 'a');
		int initDist = numBs;
		
		int[] freq = new int[2];
		for(int i = 0; i < N-1; i++) {
			res[i] = 'b';
			int chk = ask(s, res);
			if(chk < initDist) {
				initDist = chk;
				freq[1]++;
			}
			else {
				res[i] = 'a';
				freq[0]++;
			}
			if(chk == 0) {
				return;
			}
		}
		if(freq[1] < numBs) res[N-1] = 'b';
		ask(s, res);
	}
	
	int ask(Scanner s, char[] str) {
		return ask(s, new String(str));
	}
	
	int ask(Scanner s, String str) {
		System.out.println(str);
		System.out.flush();
		int ret = s.nextInt();
		return ret;
	}
	
	String build(char a, int len) {
		String res = "";
		while(res.length() < len) res += a;
		return res;
	}

}
