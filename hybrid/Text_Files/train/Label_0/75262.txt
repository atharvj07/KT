def onlogisch(i):
	global s
	return s[i % n] == s[(i + 1) % n] == s[(i - 1) % n]

def flip(i):
	global s
	global changes
	changes += 1
	if s[i] == 'L':
		s[i] = 'R'
	else:
		s[i] = 'L'
for _ in range(int(input())):
	n = int(input())
	s = input()
	lengths = []
	length = 1
	for i in range(1, len(s)):
		if s[i] != s[i - 1]:
			lengths.append(length)
			length = 1
		else:
			length += 1
	lengths.append(length)
	if len(lengths) > 1:
		if s[0] == s[-1]:
			last = lengths.pop()
			lengths[0] += last
		print(sum((l // 3 for l in lengths)))
	else:
		print((lengths[0] + 2) // 3)
