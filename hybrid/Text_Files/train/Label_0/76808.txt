from math import *

def main():
	(n, m) = list(map(int, input().split()))
	hash = {}
	for i in range(0, m):
		array = list(map(int, input().split()))
		hash.clear()
		for j in range(1, len(array)):
			if not array[j] in hash:
				hash[array[j]] = 1
			if -array[j] in hash:
				break
		else:
			print('YES')
			break
	else:
		print('NO')
main()
