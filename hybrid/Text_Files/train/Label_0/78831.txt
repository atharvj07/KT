(n, m) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
rec = [[] for i in range(n)]
INF = float('inf')
for i in range(n):
	for j in range(m):
		rec[i].append(a[i] * b[j])
	rec[i] = sorted(rec[i], reverse=True)
ma = [-INF] * n
ind = 0
for i in range(n):
	if ma < rec[i]:
		ma = rec[i]
		ind = i
mq = -INF
for i in range(n):
	if i == ind:
		continue
	mq = max(mq, max(rec[i]))
print(mq)
