import sys

def kalindrome(S, i, j, canceled_number=0, life=0):
	while True:
		if j - i == 0:
			return True
		elif j - i == 1 and S[i] == S[j]:
			return True
		if S[i] == S[j]:
			i += 1
			j -= 1
		elif S[i] == canceled_number:
			i += 1
		elif S[j] == canceled_number:
			j -= 1
		elif life > 0:
			life -= 1
			return kalindrome(S, i, j, S[i], life - 1) or kalindrome(S, i, j, S[j], life - 1)
		else:
			return False

def solve():
	T = int(sys.stdin.readline().rstrip())
	for _ in range(T):
		N = int(sys.stdin.readline().rstrip())
		S = list(map(int, sys.stdin.readline().rstrip().split(' ')))
		i = 0
		j = N - 1
		print('YES' if kalindrome(S, i, j, 0, 1) else 'NO')
solve()
