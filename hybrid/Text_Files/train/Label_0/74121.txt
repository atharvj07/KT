import math, sys
from collections import Counter, defaultdict, deque
from heapq import heapify, heappush, heappop
INF = float('inf')

def solve():
	n = int(input())
	s = ['('] * n + [')'] * n
	ans = [list(s)]
	i = n - 1
	while i > 0:
		(s[i], s[i + 1]) = (s[i + 1], s[i])
		ans.append(list(s))
		i -= 1
	for x in ans:
		print(''.join(x))
ts = 1
ts = int(input())
for t in range(1, ts + 1):
	solve()
