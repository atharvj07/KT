from math import sqrt

def solve(b, c):
	d = b * b - 4 * c
	sd = int(sqrt(d))
	for i in range(-10, 10):
		if (sd - i) * (sd - i) == d:
			sd -= i
			break
	if sd * sd != d:
		return []
	if (sd + b) % 2 != 0:
		return []
	if sd == 0:
		return [-b // 2]
	return [(-b - sd) // 2, (-b + sd) // 2]
n = int(input())
r = 0
ans = []
for i in range(128):
	if i > 0:
		r += pow(2, i - 1)
	k = pow(2, i)
	nn = n * 2
	a = solve(2 * r - 1, -nn)
	for i in a:
		if i > 0 and i % 2 == 1:
			ans.append(i * k)
ans = list(set(ans))
ans.sort()
for i in ans:
	print(i)
if len(ans) == 0:
	print(-1)
