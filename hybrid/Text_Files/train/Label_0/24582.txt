template <typename T>
T gcd(T a, T b) {
  a = std::abs(a);
  b = std::abs(b);
  while (b) {
    a %= b;
    std::swap(a, b);
  }
  return a;
}
template <typename T>
T lcm(T a, T b) {
  a = std::abs(a);
  b = std::abs(b);
  return a / gcd(a, b) * b;
}

template <typename T>
string to_string(const vector<T>& vec) {
  string str = "{ ";
  bool first = true;
  for (const auto& it : vec) {
    if (!first) {
      str += ", ";
    }
    str += to_string(it);
    first = false;
  }
  str += " }";
  return str;
}

using namespace std;
void rev(string& s, int l, int r) {
  for (int i = int(l); i < int(r + 1); ++i) {
    s[i] ^= '0' ^ '1';
  }
}
int cnt(const string& s, int l, int r) {
  int ans = 0;
  for (int i = int(l); i < int(r + 1); ++i) {
    ans += s[i] == '1';
  }
  return ans;
}
int n, t0;
bool read() {
  if (!(cin >> n >> t0)) {
    return 0;
  }
  return 1;
}
int ask(int l, int r) {
  assert(0 <= l);
  assert(l <= r);
  assert(r < n);
  cout << "? " << l + 1 << " " << r + 1 << endl;
  int t;
  cin >> t;
  if (t == -1) {
    exit(0);
  }
  return t;
}
string solve_even() {
  vii qs;
  string s(n, '0');
  int t = t0;
  for (int i = int(0); i < int(n); ++i) {
    while (1) {
      int tt = ask(i, i);
      if (abs(tt - t) % 2 == (i + 1) % 2) {
        const int y = cnt(s, 0, i - 1);
        const int p = i;
        rev(s, 0, i - 1);
        const int now = cnt(s, 0, i - 1);
        assert(now == p - y);
        const int val = tt - t - p + 2 * y - 1;
        if (val == 0) {
          s[i] = '1';
        } else if (val == -2) {
          s[i] = '0';
        } else {
          ;
          assert(0);
        }
        t = tt;
        qs.push_back(make_pair(0, i));
        break;
      } else {
        qs.push_back(make_pair(i, n - 1));
        t = tt;
      }
    }
  }
  for (const auto& q : qs) {
    int l, r;
    tie(l, r) = q;
    rev(s, l, r);
  }
  return s;
}
string solve_odd() {
  int t = t0;
  vi pref(n);
  vector<pii> qs0;
  {
    string s(n, '0');
    for (int i = 1; i < n; i += 2) {
      while (1) {
        int tt = ask(i - 1, i);
        if (abs(tt - t) % 2 == 0) {
          const int y = cnt(s, 0, i - 2);
          const int p = i - 1;
          rev(s, 0, i - 2);
          const int now = cnt(s, 0, i - 2);
          assert(now == p - y);
          const int val = tt - t - p + 2 * y - 2;
          if (abs(val) % 2 != 0) {
            assert(0);
          }
          const int x = (-val) / 2;
          if (x == 0) {
            s[i] = s[i - 1] = '1';
          } else if (x == 1) {
            s[i] = '1';
            s[i - 1] = '0';
          } else if (x == 2) {
            s[i] = s[i - 1] = '0';
          } else {
            ;
            assert(0);
          }
          t = tt;
          qs0.push_back(make_pair(0, i));
          break;
        } else {
          t = tt;
          qs0.push_back(make_pair(i - 1, n - 1));
        }
      }
    }
    for (int i = 1; i < n; i += 2) {
      pref[i] = (s[i - 1] - '0') + (s[i] - '0');
      if (i > 1) {
        pref[i] += pref[i - 2];
      }
    }
    pref[n - 1] = t;
  }
  int another_t0 = t;
  vector<pii> qs1;
  {
    string s(n, '0');
    for (int i = n - 2; i > 0; i -= 2) {
      while (1) {
        ;
        int tt = ask(i, i + 1);
        if (abs(tt - t) % 2 == 0) {
          const int y = cnt(s, i + 2, n - 1);
          const int p = n - 1 - (i + 1);
          rev(s, i + 2, n - 1);
          const int now = cnt(s, i + 2, n - 1);
          assert(now == p - y);
          const int val = tt - t - p + 2 * y - 2;
          if (abs(val) % 2 != 0) {
            assert(0);
          }
          const int x = (-val) / 2;
          if (x == 0) {
            s[i] = s[i + 1] = '1';
          } else if (x == 1) {
            s[i] = '1';
            s[i + 1] = '0';
          } else if (x == 2) {
            s[i] = s[i + 1] = '0';
          } else {
            ;
            assert(0);
          }
          t = tt;
          qs1.push_back(make_pair(i, n - 1));
          break;
        } else {
          t = tt;
          qs1.push_back(make_pair(0, i + 1));
        }
      }
    }
    for (const auto& q : qs1) {
      int l, r;
      tie(l, r) = q;
      rev(s, l, r);
    }
    vi suff(n);
    for (int i = n - 2; i > 0; i -= 2) {
      suff[i] = (s[i] - '0') + (s[i + 1] - '0');
      if (i < n - 2) {
        suff[i] += suff[i + 2];
      }
      pref[i - 1] = another_t0 - suff[i];
    }
  }
  string s(n, '0');
  for (int i = int(0); i < int(n); ++i) {
    int prev___ = 0;
    if (i > 0) {
      prev___ = pref[i - 1];
    }
    if (pref[i] > prev___) {
      s[i] = '1';
    }
  }
  for (const auto& q : qs0) {
    int l, r;
    tie(l, r) = q;
    rev(s, l, r);
  }
  return s;
}
string solve() {
  if (n % 2 == 0) {
    return solve_even();
  } else {
    return solve_odd();
  }
}
