(n, m) = map(int, input().split())
graph = {i: set() for i in range(1, 1 + n)}
for i in range(101, 101 + m):
	graph[i] = set()
zc = 0
for i in range(1, 1 + n):
	a = list(map(int, input().split()))
	if a.pop(0) == 0:
		zc += 1
	for j in a:
		graph[i].add(100 + j)
		graph[100 + j].add(i)

def dfs(v):
	unvis[v] = False
	for i in graph[v]:
		for j in graph[i]:
			if unvis[j]:
				dfs(j)
if zc == n:
	print(n)
else:
	cc = 0
	unvis = [True] * (n + 1)
	for k in range(1, 1 + n):
		if unvis[k]:
			dfs(k)
			cc += 1
	print(cc - 1)
