from sys import stdin

def main():
	(n, m) = [int(i) for i in stdin.readline().split()]
	a = [int(i) for i in stdin.readline().split()]
	ans = n - 1
	for i in range(n - 1):
		if a[i] == a[i + 1]:
			ans -= 1
	towers = [set() for i in range(m + 1)]
	res = []
	res.append(ans)
	for i in range(n):
		towers[a[i]].add(i)
	for j in range(m - 1):
		(x, y) = [int(i) for i in stdin.readline().split()]
		if len(towers[x]) < len(towers[y]):
			(x, y) = (y, x)
		for i in towers[y]:
			if i - 1 in towers[x]:
				ans -= 1
			if i + 1 in towers[x]:
				ans -= 1
		towers[x] |= towers[y]
		towers[y] = towers[x]
		res.append(ans)
	print('\n'.join((str(i) for i in res)))
main()
