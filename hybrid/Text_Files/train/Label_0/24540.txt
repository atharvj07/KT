#include <bits/stdc++.h>
using namespace std;
const int maxN = 1000005;
const int inf = 1e9;
string word;
int n, k;
bool ok(int maximumlenght) {
  int currentlenght = 0;
  int parciallenght = 0;
  int lines = 0;
  for (int i = 0; i < n; ++i) {
    parciallenght++;
    if (word[i] == ' ' || word[i] == '-' || i + 1 == n) {
      if (currentlenght + parciallenght <= maximumlenght) {
        currentlenght += parciallenght;
        parciallenght = 0;
      } else {
        if (parciallenght > maximumlenght) {
          return false;
        }
        lines++;
        currentlenght = parciallenght;
        parciallenght = 0;
      }
    }
  }
  if (currentlenght > 0) {
    lines++;
  }
  return lines <= k;
}
int main() {
  scanf("%d\n", &k);
  getline(cin, word);
  n = word.size();
  int le = -1, ri = n;
  while (le + 1 < ri) {
    int middle = (le + ri) / 2;
    if (ok(middle)) {
      ri = middle;
    } else {
      le = middle;
    }
  }
  cout << ri << endl;
  return 0;
}
