#include <bits/stdc++.h>
using namespace std;
long long max(long long a, long long b) { return (a > b) ? a : b; }
long long min(long long a, long long b) { return (a > b) ? b : a; }
int main() {
  long long n;
  cin >> n;
  long long a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    if (i == 0) {
      cout << abs(a[1] - a[0]) << " ";
      cout << abs(a[n - 1] - a[0]) << endl;
    } else if (i == n - 1) {
      cout << abs(a[i] - a[i - 1]) << " ";
      cout << abs(a[n - 1] - a[0]) << endl;
    } else {
      cout << min(abs(a[i + 1] - a[i]), abs(a[i] - a[i - 1])) << " ";
      cout << max(abs(a[n - 1] - a[i]), abs(a[i] - a[0])) << endl;
    }
  }
}
