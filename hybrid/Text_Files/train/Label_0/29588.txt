#include <bits/stdc++.h>
using namespace std;
const int MAX = 205;
const double oo = 1e99;
int N, M;
pair<int, int> E[MAX * MAX];
double mat[MAX][MAX], adj[MAX][MAX];
pair<double, double> R[MAX];
void init() {
  cin >> N >> M;
  for (int i = 1; i <= N; i++)
    for (int j = 1; j <= N; j++)
      if (i != j) adj[i][j] = mat[i][j] = oo;
  for (int i = 0; i < M; i++) {
    int a, b;
    cin >> a >> b;
    cin >> mat[E[i].first = a][E[i].second = b];
    mat[b][a] = mat[a][b];
    adj[b][a] = adj[a][b] = mat[a][b];
  }
  for (int k = 1; k <= N; k++)
    for (int i = 1; i <= N; i++)
      for (int j = 1; j <= N; j++)
        mat[i][j] = min(mat[i][j], mat[i][k] + mat[k][j]);
}
double process(int u, int v) {
  double ans = oo;
  double d = adj[u][v];
  for (int w = 1; w <= N; w++) {
    R[w] = make_pair(-mat[u][w], mat[v][w] + d);
  }
  sort(R + 1, R + N + 1);
  double m0 = -oo, md = -oo, my = -oo;
  for (int i = 1; i <= N; i++) {
    if (i > 1 && R[i].second > my) {
      double x = (my + R[i].first) / 2;
      double y = x - R[i].first;
      ans = min(ans, y);
    }
    m0 = max(m0, -R[i].first);
    md = max(md, R[i].second - d);
    my = max(my, R[i].second);
  }
  ans = min(ans, m0);
  ans = min(ans, md);
  return ans;
}
int main() {
  init();
  double ans = oo;
  for (int i = 0; i < M; i++) {
    double tmp = process(E[i].first, E[i].second);
    ans = min(ans, tmp);
  }
  printf("%.2f\n", ans);
  return 0;
}
