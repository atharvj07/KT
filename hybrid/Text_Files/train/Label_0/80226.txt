(n, m) = map(int, input().split())
guesses = list(map(int, input().split()))
possible = 3 * n - 2
indexes = {}
for i in range(m):
	if guesses[i] not in indexes:
		indexes[guesses[i]] = [i]
	else:
		indexes[guesses[i]].append(i)
for i in indexes.keys():
	possible -= 1
	status = 0
	if i + 1 in indexes:
		for j in indexes[i]:
			for k in indexes[i + 1]:
				if k > j:
					status = -1
					possible -= 1
					break
			if status == -1:
				break
	status = 0
	if i - 1 in indexes:
		for j in indexes[i]:
			for k in indexes[i - 1]:
				if k > j:
					possible -= 1
					status = -1
					break
			if status == -1:
				break
print(possible)
