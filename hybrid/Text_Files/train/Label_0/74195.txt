t = int(input())
while t:
	t -= 1
	n = int(input())
	arr = list(map(int, input().split()))
	c = [0 for i in range(n)]
	ans = 0
	for i in range(n):
		if i > 0:
			c[i] += c[i - 1]
		if c[i] <= arr[i] - 1:
			ans += arr[i] - 1 - c[i]
		if arr[i] > 1:
			if i + 2 < n:
				c[i + 2] += 1
			if i + arr[i] + 1 < n:
				c[i + arr[i] + 1] += -1
		if c[i] > arr[i] - 1:
			extra = c[i] - arr[i] + 1
			if i + 1 < n:
				c[i + 1] += extra
			if i + 2 < n:
				c[i + 2] += -extra
	print(ans)
