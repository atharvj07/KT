def gs(x):
	return x * (x + 1) // 2

def gs2(x):
	if x % 2 == 0:
		cr = 0
	else:
		cr = x // 2 + 1
	cr += h * x
	cr += (x // 2 + 1) * (x // 2)
	return cr
r = 10 ** 18 + 7
(n, h) = [int(i) for i in input().strip().split()]
l = 0
r = min(h + 1, r)
while r - l > 1:
	m = (l + r) // 2
	if gs(m) <= n:
		l = m
	else:
		r = m
tr = l
n -= gs(l)
if n >= h:
	n -= h
	tr += 1
	l = 0
	r = 10 ** 18 + 7
	while r - l > 1:
		m = (l + r) // 2
		if gs2(m) <= n:
			l = m
		else:
			r = m
	tr += l
	n -= gs2(l)
if n > 0:
	tr += 1
