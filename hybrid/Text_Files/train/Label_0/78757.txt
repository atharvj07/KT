n = int(input())
s1 = input()
s2 = input()

def s_to_m(s):
	m = {}
	for (i, c) in enumerate(s, 1):
		if c not in m:
			m[c] = []
		m[c].append(i)
	return m

def extract_next(m, c):
	res = m[c].pop()
	if len(m[c]) == 0:
		m.pop(c)
	return res
m1 = s_to_m(s1)
m2 = s_to_m(s2)
pairs = []
keys = set(m1.keys())
for key in keys:
	if key == '?':
		continue
	while key in m2 and key in m1:
		pairs.append((extract_next(m1, key), extract_next(m2, key)))
keys = set(m1.keys())
for key in keys:
	if key == '?':
		continue
	while '?' in m2 and key in m1:
		pairs.append((extract_next(m1, key), extract_next(m2, '?')))
keys = set(m2.keys())
for key in keys:
	while '?' in m1 and key in m2:
		pairs.append((extract_next(m1, '?'), extract_next(m2, key)))
print(len(pairs))
for (x, y) in pairs:
	print(x, y)
