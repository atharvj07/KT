import java.util.*;

public class Problem1 {

    LinkedList<String> li = new LinkedList<String>();

    String m[] = { "Aa","Ba","Ca","Da","Ea","Fa","Ga","Ha","Ta","Ya",
            "Ab","Bb","Cb","Db","Eb","Fb","Gb","Hb","Tb","Yb",
            "Ac","Bc","Cc","Dc","Ec","Fc","Gc","Hc","Tc","Yc",
            "Ad","Bd","Cd","Dd","Ed","Fd","Gd","Hd","Td","Yd",
            "Ae","Be","Ce","De","Ee","Fe","Ge","He","Te","Ye"};
    Scanner in;
    int t=0;
    int p=0;
    boolean[] flag;
    {
        flag = new boolean[55];
    }

    public static void main(String[] args) {
        Problem1 a = new Problem1();
        a.run();
    }


    void run(){
        in =  new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        p=0;
        for (int i = 0; i < n-k+1; i++) {
            String s = in.next();
            if (s.equals("YES")) flag[i]=true; else flag[i]=false;
        }

        if (flag[0]==true){
            for (int j = 0; j < k; j++) {
               li.add(m[j]);
            }
        }
        if (flag[0]==false){

            for (int j = 0; j < k-1; j++) {
                li.add(m[j]);
            }
            li.add(m[0]);
            if (flag[1]==false) t++;
        }

        p=k;
         // p=0;
        while (p<n) {

            if (flag[p-k+1] == true){
                t=0;
                li.add(m[p]);
                p++;

            } else
            if (flag[p-k+1] == false){
                li.add(li.get(p-k+1));
                p++;
            }
        }

        for (int i = 0; i < li.size(); i++) {
            System.out.print(li.get(i));
            if (i!=li.size()) System.out.print(" ");
        }

    }

}
