#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int a, b;
    cin >> a >> b;
    int c;
    while (b) {
      c = a % b;
      a = b;
      b = c;
    }
    if (a == 1)
      cout << "finite" << endl;
    else
      cout << "infinite" << endl;
  }
}
