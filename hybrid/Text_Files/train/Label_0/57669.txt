    public static boolean eq(String a, String b) {

    boolean l = eq(a1, a2, a3, a4, a5, a6, b1, b3, b4, b5, b2, b6);

    if (i || j || k || l) {

    }

    boolean i2 = eq(a1, a2, a3, a4, a5, a6, b6, b5, b4, b3, b2, b1);
    boolean j2 = eq(a1, a2, a3, a4, a5, a6, b6, b2, b5, b4, b3, b1);

    public static void main(String[] args) {
    Scanner sc = new Scanner(new BufferedInputStream(System.in));
    char[] str = sc.next().toCharArray();
    Arrays.sort(str);
    List<String> existing = new ArrayList<String>();
    for (int i = 0; i < 720; i++) {
        boolean ok = true;
        String s = new String(str);
        for (String string : existing) {
        if (eq(s, string)) {
            ok = false;
            break;
        }
        }
        if (ok)
        existing.add(s);
        str = nextperm(str);
    }
    System.out.println(existing.size());
    }

    static char[] nextperm(char[] vals) {
    int i = vals.length - 1;
    while (true) {
        int ii = i;
        i--;
        if (vals[i] < vals[ii]) {
        int j = vals.length;
        while (vals[i] >= vals[--j]);
        char temp = vals[i];  //Swap
        vals[i] = vals[j];
        vals[j] = temp;
        int begin = ii, end = vals.length - 1;
        while (end > begin) {
            char stemp = vals[end];   //Swap
            vals[end] = vals[begin];
            vals[begin] = stemp;
            end--;
            begin++;
        }
        return vals;
        } else if (vals[i] == vals[0]) {
        int begin = 0, end = vals.length - 1;
        while (end > begin) {
            char stemp = vals[end];   //Swap
            vals[end] = vals[begin];
            vals[begin] = stemp;
            end--;
            begin++;
        }
        return vals;
        }
    }
    }
