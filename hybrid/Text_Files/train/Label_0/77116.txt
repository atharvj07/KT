from math import sqrt
MOD = 1000000007

def mult(a, b):
	return a * b % MOD

def binpow(a, n):
	if n == 0:
		return 1
	if n % 2:
		return mult(binpow(a, n - 1), a)
	else:
		b = binpow(a, n // 2)
		return mult(b, b)
(x, n) = map(int, input().split())
p = list()
for i in range(2, int(sqrt(x)) + 1):
	if x % i == 0:
		p.append(i)
	while x % i == 0:
		x //= i
if x > 1:
	p.append(x)
ans = 1
for t in p:
	pw = t
	cnt = 0
	while pw <= n:
		cnt += n // pw
		pw *= t
	ans = mult(ans, binpow(t, cnt))
print(ans)
