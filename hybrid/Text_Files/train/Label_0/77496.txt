(a, b) = map(int, input().split())
l = []
p = 0
for m in range(a):
	lPrime = list(input())
	l.append(lPrime)
for q in range(0, a - 1):
	for w in range(0, b - 1):
		if l[q][w] not in ['f', 'a', 'c', 'e', 1, 100, 150, 5]:
			l[q][w] = 0
		if l[q][w + 1] not in ['f', 'a', 'c', 'e', 1, 100, 150, 5]:
			l[q][w + 1] = 0
		if l[q + 1][w] not in ['f', 'a', 'c', 'e', 1, 100, 150, 5]:
			l[q + 1][w] = 0
		if l[q + 1][w + 1] not in ['f', 'a', 'c', 'e', 1, 100, 150, 5]:
			l[q + 1][w + 1] = 0
		if l[q][w] == 'f':
			l[q][w] = 1
		if l[q][w + 1] == 'f':
			l[q][w + 1] = 1
		if l[q + 1][w] == 'f':
			l[q + 1][w] = 1
		if l[q + 1][w + 1] == 'f':
			l[q + 1][w + 1] = 1
		if l[q][w] == 'a':
			l[q][w] = 100
		if l[q][w + 1] == 'a':
			l[q][w + 1] = 100
		if l[q + 1][w] == 'a':
			l[q + 1][w] = 100
		if l[q + 1][w + 1] == 'a':
			l[q + 1][w + 1] = 100
		if l[q][w] == 'c':
			l[q][w] = 150
		if l[q][w + 1] == 'c':
			l[q][w + 1] = 150
		if l[q + 1][w] == 'c':
			l[q + 1][w] = 150
		if l[q + 1][w + 1] == 'c':
			l[q + 1][w + 1] = 150
		if l[q][w] == 'e':
			l[q][w] = 5
		if l[q][w + 1] == 'e':
			l[q][w + 1] = 5
		if l[q + 1][w] == 'e':
			l[q + 1][w] = 5
		if l[q + 1][w + 1] == 'e':
			l[q + 1][w + 1] = 5
for i in range(0, a - 1):
	for j in range(0, b - 1):
		u = l[i][j] + l[i][j + 1] + l[i + 1][j] + l[i + 1][j + 1]
		if u == 256:
			p = p + 1
print(p)
