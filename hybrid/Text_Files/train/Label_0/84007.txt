from math import hypot
(a, b) = map(int, input().split())
s = input().strip()
(dx, dy) = (0, 0)
for i in s:
	if dx == a and dy == b:
		print('Yes')
		exit()
	if i == 'R':
		dx += 1
	if i == 'L':
		dx -= 1
	if i == 'D':
		dy -= 1
	if i == 'U':
		dy += 1
	if dx == a and dy == b:
		print('Yes')
		exit(0)
if dx == dy == 0:
	print('No')
	exit(0)
lngt = int(max(0, hypot(a, b) / hypot(dx, dy) - 500))
(x, y) = (dx * lngt, dy * lngt)
for j in range(1000):
	for i in s:
		if x == a and y == b:
			print('Yes')
			exit(0)
		if i == 'R':
			x += 1
		if i == 'L':
			x -= 1
		if i == 'D':
			y -= 1
		if i == 'U':
			y += 1
		if x == a and y == b:
			print('Yes')
			exit(0)
print('No')
