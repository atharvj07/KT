import java.util.*;

public class Main {
    private static final long MOD = 1000000007;

    private static int n, m, p, q;
    private static int[][] adjacent;
    private static int[] x;
    private static int[] y;
    private static int[] len;
    private static int[] group;
    
    private static int inner1, inner2;
    
    static class Item implements Comparable<Item> {
        private static int counter;
        public int head;
        public long value;
        private int index;
        public Item(int head, long value) {
            this.head = head;
            this.value = value;
            this.index = counter++;
        }
        @Override
        public int compareTo(Item o) {
            if (value != o.value) {
                if (value < o.value) {
                    return -1;
                } else {
                    return 1;
                }
            } else {
                return index - o.index;
            }
        }
    }
    
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        
        n = scan.nextInt();
        m = scan.nextInt();
        p = scan.nextInt();
        q = scan.nextInt();
        
        x = new int[m];
        y = new int[m];
        len = new int[m];
        read(scan, x, y, len);
        decreaseByOne(x, y);
        
        int[] count = new int[n];
        for (int i = 0; i < m; i++) {
            count[x[i]]++;
            count[y[i]]++;
        }
        adjacent = new int[n][];
        for (int i = 0; i < n; i++) {
            adjacent[i] = new int[count[i]];
        }
        for (int i = 0; i < m; i++) {
            adjacent[x[i]][--count[x[i]]] = y[i];
            adjacent[y[i]][--count[y[i]]] = x[i];
        }
        
        inner1 = inner2 = -1;
        group = new int[n];
        Arrays.fill(group, -1);
        int groupCount = 0;
        for (int i = 0; i < n; i++) {
            if (group[i] == -1) {
                search(i, groupCount++);
            }
        }
        
        if (groupCount < q || (groupCount - q) > p || (n == q && p > 0)) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            int[] groupHead = new int[groupCount];
            for (int i = 0; i < n; i++) {
                groupHead[group[i]] = i;
            }
            
            // Overflow.......
            long[] groupLen = new long[groupCount];
            for (int i = 0; i < m; i++) {
                groupLen[group[x[i]]] += len[i];
            }
            
            TreeSet<Item> sorted = new TreeSet<Item>();
            for (int i = 0; i < groupCount; i++) {
                sorted.add(new Item(groupHead[i], groupLen[i]));
            }
            
            // TLE caused by IO...
            StringBuilder builder = new StringBuilder(p * 10);
            long bound = (long)1e9;
            for (int i = groupCount; i > q; i--) {
                Item a = sorted.first();
                sorted.remove(a);
                Item b = sorted.first();
                sorted.remove(b);
                long S = Math.min(bound, a.value + b.value + 1);
                a.value = a.value + b.value + S;
                sorted.add(a);
                builder.append(String.format("%d %d", a.head + 1, b.head + 1));
                builder.append("\n");
                inner1 = a.head;
                inner2 = b.head;
                p--;
            }
            
            while (p > 0) {
                builder.append(String.format("%d %d", inner1 + 1, inner2 + 1));
                builder.append("\n");
                p--;
            }
            System.out.print(builder.toString());
        }
        
        scan.close();
    }
    
    private static void search(int current, int label) {
        group[current] = label;
        for (int next : adjacent[current]) {
            if (group[next] == -1) {
                inner1 = current;
                inner2 = next;
                search(next, label);
            }
        }
    }
    
    private static void print(int... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void print(long... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static <T> void print(T... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void read(Scanner scan, int[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (int[] arr : arrs) {
                arr[i] = scan.nextInt();
            }
        }
    }
    
    private static void read(Scanner scan, long[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (long[] arr : arrs) {
                arr[i] = scan.nextLong();
            }
        }
    }
    
    private static void decreaseByOne(int[]... arrs) {
        for (int[] arr : arrs) {
            for (int i = 0; i < arr.length; i++) {
                arr[i] --;
            }
        }
    }
}
