using bset = bitset<50000>;

signed main(){

    int N, M, T; cin >> N >> M >> T;
    vec x(M), y(M);
    REP(i, M){
        cin >> x[i] >> y[i];
        x[i]--; y[i]--;
    }

    if(T == 1){
        vector<bset> b(N);
        REP(i, N) b[i][i] = 1;
        REP(i, M){
            b[x[i]] |= b[y[i]];
            b[y[i]] = b[x[i]];
        }

        int f = -1;
        REP(i, N) if(b[i].count() == N) f = i;
        if(f == -1) Out(-1);
        else{
            //debug(f);
            string ans(M, '.');
            bset s;
            s[f] = 1;
            IREP(i, M){
                if(s[y[i]]) ans[i] = 'v';
                else ans[i] = '^';
                if(s[x[i]]) s[y[i]] = 1;
                if(s[y[i]]) s[x[i]] = 1;
            }
            Out(ans);
        }

    }else{
        if(N == 2){
            Out(-1);
            return 0;
        }
        string ans(M, '.');
        REP(i, M){
            if(y[i] <= 2){
                int i0 = -1;
                FOR(j, i + 1, M) if(y[j] <= 2){
                    i0 = j;
                    break;
                }
                if(i0 == -1) ans[i] = '^';
                else{
                    if(y[i] != x[i0] && y[i] != y[i0]) ans[i] = 'v';
                    else ans[i] = '^';
                }
            }else ans[i] = '^';
        }
        Out(ans);
    }


    return 0;
}
