#include <bits/stdc++.h>
using namespace std;
int st[100005];
bool vis[100005];
void dfs(int n, vector<vector<pair<int, int>>> &adj) {
  if (vis[n]) return;
  int sm = 0;
  if (adj[n].size() == 1 && n != 0) {
    sm = 1;
  }
  vis[n] = true;
  for (pair<int, int> a : adj[n]) {
    if (!vis[a.first]) {
      dfs(a.first, adj);
      sm += st[a.first];
    }
  }
  st[n] = sm;
}
void solve() {
  long long n, s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    st[i] = 0;
    vis[i] = false;
  }
  vector<vector<pair<int, int>>> adj;
  for (int i = 0; i < n; i++) {
    vector<pair<int, int>> c;
    adj.push_back(c);
  }
  vector<pair<pair<long long, long long>, long long>> e;
  for (int i = 0; i < n - 1; i++) {
    long long a, b, w;
    cin >> a >> b >> w;
    a--;
    b--;
    adj[a].push_back({b, w});
    adj[b].push_back({a, w});
    e.push_back({{a, b}, w});
  }
  dfs(0, adj);
  priority_queue<pair<pair<long long, long long>, long long>> q;
  long long ts = 0;
  for (int i = 0; i < n - 1; i++) {
    q.push({{(e[i].second - e[i].second / 2) *
                 min(st[e[i].first.first], st[e[i].first.second]),
             e[i].second},
            i});
    ts += e[i].second * min(st[e[i].first.first], st[e[i].first.second]);
  }
  if (ts < s) {
    cout << 0 << '\n';
    return;
  }
  int m = 0;
  while (ts > s) {
    pair<pair<long long, long long>, long long> inf = q.top();
    q.pop();
    long long op = inf.first.first;
    long long val = inf.first.second;
    long long ind = inf.second;
    ts -= op;
    long long nm = val / 2;
    q.push(
        {{(nm - nm / 2) * min(st[e[ind].first.first], st[e[ind].first.second]),
          nm},
         ind});
    m++;
  }
  cout << m << '\n';
}
void querySolve() {
  int n;
  cin >> n;
  while (n--) {
    solve();
  }
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  querySolve();
}
