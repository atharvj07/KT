import java.util.Scanner;

public class Inverse {
    final int N = 32768;
    String string;
    int[] res = new int[N];
    int n, pos;
    
    int num() {
        if (string.charAt(pos) == 'f') {
            pos += 2;//f(
            int tmp = expression();
            pos++;//)
            return res[tmp];
        }
        if (string.charAt(pos) == 'n') {
            pos++;
            return n;
        }
        int ret = 0;
        while (Character.isDigit(string.charAt(pos))) {
            ret = ret * 10 + string.charAt(pos) - '0';
            ++pos;
        }
        //System.out.println("num");
        //System.out.println(pos);
        //System.out.println(string.charAt(pos));
        //System.out.println(ret);
        return ret;
    }
    
    int term() {
        int ret = num();
        while (string.charAt(pos) == '*' || string.charAt(pos) == '/') {
            char c = string.charAt(pos++);
            int next = num();
            if (c == '*') {
                ret = (ret * next) % N;
            } else {
                ret /= next;
            }
        }
        return ret;
    }
    
    int expression() {
        int ret = term();
        while (string.charAt(pos) == '+' || string.charAt(pos) == '-') {
            char c = string.charAt(pos++);
            int next = term();
            if (c == '+') {
                ret += next;
                if (ret >= N) ret -= N;
            } else {
                ret -= next;
                if (ret < 0) ret += N;
            }
        }
        return ret;
    }
    
    boolean logicalExpression() {
        //System.out.println("Logival");
        //System.out.println(pos);
        int X = expression();
        char flag = string.charAt(pos++);
        if (flag == '=') pos++;
        int Y = expression();
        if (flag == '<' && X < Y) return true;
        if (flag == '=' && X == Y) return true;
        if (flag == '>' && X > Y) return true;
        return false;
    }
    
    int calculate() {
        pos = 11;//"intf(intn){".length();
        while (string.charAt(pos) != '}') {
            if (string.charAt(pos) == 'r') {
                pos += 6;//"return".length();
                return expression();
            } else {
                pos += 3;//"if(".length();
                boolean flag = logicalExpression();
                ++pos;//)
                if (flag == true) {
                    pos += 6;//"return".length();
                    return expression();
                } else {
                    while (string.charAt(pos) != ';') ++pos;
                    ++pos;
                }
            }
        }
        return 0;
    }
    
    public void solve() {
        Scanner inScanner = new Scanner(System.in);
        string = new String();
        int ans = inScanner.nextInt();
        while (inScanner.hasNext()) {
            string += inScanner.next();
            if (string.charAt(string.length() - 1) == '}') break;
        }
        //System.out.println(string);
        for (int i = 0; i < N; ++i) {
            n = i;
            res[i] = calculate();
            //if (i < 100)System.out.println(res[i]);
        }
        for (int i = N - 1; i >= 0; --i)
            if (res[i] == ans) {
                System.out.println(i);
                return;
            }
        System.out.println("-1");
    }

    public static void main(String[] args) {
        new Inverse().solve();
    }
}
