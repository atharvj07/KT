#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
int main() {
  ios::sync_with_stdio(0);
  int n;
  cin >> n;
  vector<int> ans(n, INF);
  for (int i = 0; (1 << i) <= n; ++i) {
    vector<bool> bad(n);
    int nr = 0;
    for (int j = 0; j < n; ++j) {
      if ((1 << i) & j) nr++, bad[j] = 1;
    }
    if (nr < 1) continue;
    cout << nr << endl;
    for (int j = 0; j < n; ++j) {
      if ((1 << i) & j) cout << j + 1 << ' ';
    }
    cout << endl;
    for (int j = 0, x; j < n; ++j) {
      cin >> x;
      if (!bad[j]) ans[j] = min(ans[j], x);
    }
  }
  for (int i = 0; (1 << i) <= n; ++i) {
    vector<bool> bad(n);
    int nr = 0;
    for (int j = 0; j < n; ++j) {
      if (!((1 << i) & j)) nr++, bad[j] = 1;
    }
    if (nr < 1) continue;
    cout << nr << endl;
    for (int j = 0; j < n; ++j) {
      if (!((1 << i) & j)) cout << j + 1 << ' ';
    }
    cout << endl;
    for (int j = 0, x; j < n; ++j) {
      cin >> x;
      if (!bad[j]) ans[j] = min(ans[j], x);
    }
  }
  cout << -1 << endl;
  for (int i = 0; i < n; ++i) cout << ans[i] << ' ';
  cout << endl;
  return 0;
}
