#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
const int MAXN = 100000;
const int MAXM = 100000;
int n, m;
int want[MAXN];
vector<pair<int, int> > adj[MAXN];
bool intree[MAXM];
int oddroot;
int col[MAXN];
void dfs(int at, int par) {
  col[at] = par == -1 ? 0 : 1 - col[par];
  for (int i = (0); i < (((int)(adj[at]).size())); ++i) {
    int to = adj[at][i].first;
    if (to == par) continue;
    if (col[to] == -1) {
      intree[adj[at][i].second] = true;
      dfs(to, at);
    } else {
      if (oddroot == -1 && col[to] == col[at]) oddroot = at;
    }
  }
}
long long ans[MAXM];
long long solve(int at, int parid) {
  long long sum = 0;
  for (int i = (0); i < (((int)(adj[at]).size())); ++i) {
    int to = adj[at][i].first, id = adj[at][i].second;
    if (id == parid || !intree[id]) continue;
    long long childsum = solve(to, id);
    ans[id] = want[to] - childsum;
    sum += ans[id];
  }
  return sum;
}
bool modify(int at, int parid, long long by, long long sby) {
  for (int i = (0); i < (((int)(adj[at]).size())); ++i) {
    int to = adj[at][i].first, id = adj[at][i].second;
    if (id == parid) continue;
    if (to == oddroot && by == sby) {
      ans[id] += by;
      return true;
    }
    if (!intree[id]) continue;
    if (modify(to, id, -by, sby)) {
      ans[id] += by;
      return true;
    }
  }
  return false;
}
void run() {
  scanf("%d%d", &n, &m);
  for (int i = (0); i < (n); ++i) scanf("%d", &want[i]);
  for (int i = (0); i < (m); ++i) {
    int a, b;
    scanf("%d%d", &a, &b);
    --a, --b;
    adj[a].push_back(make_pair(b, i));
    adj[b].push_back(make_pair(a, i));
    intree[i] = false;
  }
  memset(col, -1, sizeof(col));
  oddroot = -1;
  dfs(0, -1);
  int root = oddroot;
  if (root == -1) root = 0;
  memset(ans, 0, sizeof(ans));
  int childsum = solve(root, -1);
  long long diff = want[root] - childsum;
  if (diff != 0 && oddroot != -1) {
    assert(diff % 2 == 0);
    assert(modify(root, -1, diff / 2, diff / 2));
    diff = 0;
  }
  if (diff != 0) {
    printf("NO\n");
    return;
  }
  for (int i = (0); i < (m); ++i) assert(abs(ans[i]) <= (long long)2 * n * n);
  printf("YES\n");
  for (int i = (0); i < (m); ++i) printf("%lld\n", ans[i]);
}
int main() {
  run();
  return 0;
}
