def check(l, r):
	n = len(l)
	for i in range(n):
		if l[i] > i or r[i] > n - i - 1:
			return False
	return True

def post_check(a, l, r):
	n = len(a)
	for i in range(n - 1):
		if a[i] == a[i + 1] and (l[i] != l[i + 1] or r[i] != r[i + 1]):
			return False
		if a[i] > a[i + 1] and (l[i + 1] <= l[i] or r[i] > r[i + 1]):
			return False
		if a[i] < a[i + 1] and (l[i + 1] > l[i] or r[i] <= r[i + 1]):
			return False
	return True
n = int(input())
l = tuple(map(int, input().split(' ')))
r = tuple(map(int, input().split(' ')))
if check(l, r):
	rank = [(i, l[i] + r[i]) for i in range(n)]
	rank.sort(reverse=True, key=lambda x: x[1])
	lold = 0
	rold = 0
	counter = 0
	cand = 1
	c = [0] * n
	old = rank[0][1]
	for i in rank:
		if i[1] == old:
			c[i[0]] = cand
		else:
			cand += 1
			c[i[0]] = cand
			old = i[1]
	for i in range(n):
		new_l = sum((c[i] < c[o] for o in range(i)))
		new_r = sum((c[i] < c[o] for o in range(i, n)))
		if new_l != l[i] or new_r != r[i]:
			print('NO')
			break
	else:
		print('YES')
		print(*c)
else:
	print('NO')
