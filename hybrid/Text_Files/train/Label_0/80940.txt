q = 1
for i in range(q):
	(n, m, t1, t2, k) = map(int, input().split())
	l1 = list(map(int, input().split()))
	l2 = list(map(int, input().split()))
	l1.append(10 ** 15)
	l2.append(10 ** 15)
	if k >= n:
		print(-1)
	else:
		ans = 0
		p = 0
		for i in range(k + 1):
			x = i
			b = l1[x] + t1
			while p < m and l2[p] < b:
				p += 1
			ans = max(ans, l2[min(m, p + k - i)] + t2)
		if ans >= 10 ** 15:
			print(-1)
		else:
			print(ans)
