import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Scanner;

public class Main {
	static BigInteger intSqrt(BigInteger n) {
		BigInteger x, y = n;
		
		if (n.equals(BigInteger.ZERO)) return n;
		do {
			x = y;
			y = x.add(n.divide(x)).shiftRight(1);
		} while (y.compareTo(x) < 0);
		
		return x;
	}
	
	static BigInteger check(BigInteger n, int t) {
		BigInteger b = BigInteger.valueOf(2).pow(t + 1).subtract(BigInteger.valueOf(3));
		BigInteger c = n.multiply(BigInteger.valueOf(2));
		BigInteger x = b.multiply(b).add(c.multiply(BigInteger.valueOf(4)));
		BigInteger k;
		
		if (x.compareTo(BigInteger.ZERO) < 0) return BigInteger.valueOf(-1);
		BigInteger d = intSqrt(x);
		if (!d.multiply(d).equals(x)) return BigInteger.valueOf(-1);
		if ((d.subtract(b).and(BigInteger.ONE)).equals(BigInteger.ONE)) return BigInteger.valueOf(-1);
		k = d.subtract(b).shiftRight(1);
		if ((k.and(BigInteger.ONE)).equals(BigInteger.ZERO)) return BigInteger.valueOf(-1);

		return k;
	}
	
	public static void main(String[] args) {
		// TODO Auto-generated method stub
		Scanner cin = new Scanner(System.in);
		long n, x = 1;
		int i;
		BigInteger cur;
		ArrayList<BigInteger> ans = new ArrayList<BigInteger>();
		
		while (cin.hasNext()) {
			n = cin.nextLong();
			ans.clear();
			for (i = 0; (x << i) - 1 <= n; ++i) {
				cur = check(BigInteger.valueOf(n), i);
				if (!cur.equals(BigInteger.valueOf(-1))) ans.add(cur.shiftLeft(i));
			}
			
			if (ans.size() == 0) System.out.println(-1);
			else {
				Collections.sort(ans);
				for (BigInteger e : ans) {
					System.out.println(e);
				}
			}
		}
	}

}
