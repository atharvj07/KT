
import java.util.*;

public class main {

	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int t=s.nextInt();
		while(t>0) {
			int n= s.nextInt();
			Integer ans[]= new Integer[n];
			for(int i=0;i<n;i++) {
				ans[i]=s.nextInt();
			}
			// Integer ans1[]= new Integer[5];
			Arrays.sort(ans, Collections.reverseOrder());
			for(int i:ans) {
				System.out.print(i+" ");
			}
			System.out.println();
			
			t--;
		}
			
	}
}
