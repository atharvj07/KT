import sys
_DEBUG = False
if _DEBUG:
	fd = open('input.txt', 'r')
	sys.stdin = fd
for t in range(int(sys.stdin.readline())):
	s = list(input())
	flag = False
	prev = max(ord(s[0]), ord(s[-1]))
	if not prev - ord('a') + 1 == len(s):
		print('NO')
		continue
	if prev == ord(s[0]):
		s.pop(0)
	else:
		s.pop()
	while s:
		x = max(ord(s[0]), ord(s[-1]))
		if not prev - x == 1:
			print('NO')
			flag = True
			break
		if x == ord(s[0]):
			s.pop(0)
		else:
			s.pop(-1)
		prev = x
	if flag:
		continue
	print('YES')
fd.close() if _DEBUG else None
