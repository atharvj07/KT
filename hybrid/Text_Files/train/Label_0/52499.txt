import java.io.*;
import java.math.*;
import java.util.*;

public class C {

	public C () throws IOException {
		String [] NK = r.readLine().split(" ");
		int N = Integer.parseInt(NK[0]);
		int K = Integer.parseInt(NK[1]);
		int [] A = new int [N];
		String [] AA = r.readLine().split(" ");
		for (int i = 0; i < N; ++i)
			A[i] = Integer.parseInt(AA[i]);
			
		solve(N, K, A);
	}
	
	boolean isLucky(int Q) {
		char [] C = ("" + Q).toCharArray();
		for (char c : C)
			if (c != '4' && c != '7')
				return false;
		return true;
	}
	
	int MOD = 1000000007;
	BigInteger BIGMOD = new BigInteger("" + MOD);
	int [] INV;

	public void solve (int N, int K, int [] A) {
		t = millis();

		INV = new int [N+1];
		Arrays.fill(INV, -1);
//		for (int i = 1; i <= N; ++i)
//			INV[i] = inv(i);
			
		Map<Integer, Integer> LUC = new HashMap<Integer, Integer>();
		for (int i = 0; i < N; ++i)
			if (isLucky(A[i]))
				inc(LUC, A[i]);
		int Z = LUC.size();
		
		Map<Integer, Integer> CL = new HashMap<Integer, Integer>();
		for (int v : LUC.values())
			inc(CL, v);
		
		V = CL.keySet().toArray(new Integer[0]);
		NV = new Integer [V.length];
		for (int i = 0; i < V.length; ++i)
			NV[i] = CL.get(V[i]);
		
		M = N - (int)sum(CL);
		
		int L = Math.min(Z, Z+M-K);
		print(F(0, K, L));
	}
	
	int M;
	Integer [] V, NV;
	
	Map<Long, Integer> F = new HashMap<Long, Integer>();
	
	long F(int i, int K, int L) {
		if (K == 0)
			return 1;
		
		long key = 10000000L * i + K;
		if (F.containsKey(key))
			return F.get(key);
			
		if (i == V.length)
			return comb(M, K);
		
		int min = Math.max(NV[i] - L, 0);
		int max = Math.min(NV[i], K);
		
		long cnt = 0;
		for (int j = min; j <= max; ++j) {
			long P = pow(V[i], j);
			P = (P * comb(NV[i], j)) % MOD;
			P = (P * F(i+1, K-j, L+j-NV[i])) % MOD;
			cnt = (cnt + P) % MOD;
		}
		
		F.put(key, (int)cnt);
		return (int)cnt;
	}

	<T> void inc (Map<T, Integer> map, T key) {
		if (!map.containsKey(key))
			map.put(key, 0);
		map.put(key, map.get(key) + 1);
	}
	
	long sum (Map<Integer, Integer> map) {
		long S = 0;
		for (int key : map.keySet())
			S += 1L * key * map.get(key);
		return S;
	}
	
	Map<Long, Integer> COMB = new HashMap<Long, Integer>();
	Map<Long, Integer> POW = new HashMap<Long, Integer>();
	final long MI = 1000000L;
	
	long pow (long N, int K) {
		long key = MI * N + K;
		if (POW.containsKey(key))
			return POW.get(key);
		else {
			int v = new BigInteger("" + N).modPow(new BigInteger("" + K), BIGMOD).intValue();
			POW.put(key, v);
			return v;
		}
	}
	
	long comb(long N, int K) {
		long key = MI * N + K;
		if (COMB.containsKey(key))
			return COMB.get(key);
		else if (K == 0)
			return 1;
		else if (K > N)
			return 0;
		else if (2*K > N) {
			COMB.put(key, (int)comb(N, (int)N-K));
			return COMB.get(key);
		} else {
			long v = (((comb(N, K-1) * (N-K+1)) % MOD) * inv(K)) % MOD;
			COMB.put(key, (int)v);
			return v;
		}
	}
			
	int inv(int K) {
		if (INV[K] != -1)
			return INV[K];
		else {
			//return INV[K] = new BigInteger("" + K).modInverse(BIGMOD).intValue();
			long A = MOD, B = K, T;
			long x = 0, fx = 1, y = 1, fy = 0;
			while (B != 0) {
				long Q = A/B, R = A%B; A = B; B = R;
				T = fx - Q*x; fx = x; x = T;
				T = fy - Q*y; fy = y; y = T;
			}
			
			while (fy < 0)
				fy += MOD;
			fy = fy % MOD;
			return INV[K] = (int)fy;
		}
	}

	////////////////////////////////////////////////////////////////////////////////////

	static BufferedReader r;
	static long t;
	
	static void print2 (Object o) {
		System.out.println(o);
	}
	
	static void print (Object o) {
		print2(o);
		//print2((millis() - t) / 1000.0);
		System.exit(0);
	}
	
	static void run () throws IOException {
		r = new BufferedReader(new InputStreamReader(System.in));
		new C();
	}
	
	public static void main(String[] args) throws IOException {
		run();
	}
	
	static long millis() {
		return System.currentTimeMillis();
	}
}
