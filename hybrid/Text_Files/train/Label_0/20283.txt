#include <bits/stdc++.h>
using namespace std;
char s[2005][15][15];
int d[2005][2005];
int N, M, n, w;
int calc(int u, int v) {
  int res(0);
  for (int i = 0; i < M; i++)
    for (int j = 0; j < N; j++) {
      if (s[u][i][j] != s[v][i][j]) res++;
    }
  return res * w;
}
int chk[2005], val[2005];
int par[2005];
int main() {
  cin >> M >> N >> n >> w;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < M; j++) scanf("%s", s[i][j]);
  }
  for (int i = 0; i < n; i++)
    for (int j = i + 1; j < n; j++) d[i][j] = d[j][i] = calc(i, j);
  int id, mx;
  int rlt(0);
  int x[2005];
  int xn(0);
  memset(par, -1, sizeof(par));
  for (int i = 0; i < n; i++) val[i] = M * N;
  while (1) {
    id = -1;
    for (int i = 0; i < n; i++)
      if (!chk[i]) {
        if (id == -1 || val[i] < mx) {
          id = i;
          mx = val[i];
        }
      }
    if (id == -1) break;
    x[xn++] = id;
    rlt += val[id];
    chk[id] = 1;
    for (int i = 0; i < n; i++) {
      if (chk[i]) continue;
      if (val[i] <= d[id][i]) continue;
      val[i] = d[id][i];
      par[i] = id;
    }
  }
  cout << rlt << endl;
  for (int i = 0; i < n; i++) {
    cout << x[i] + 1 << ' ' << par[x[i]] + 1 << endl;
  }
  return 0;
}
