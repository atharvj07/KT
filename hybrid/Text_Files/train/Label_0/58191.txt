import java.util.ArrayList;
import java.util.Scanner;

public class CF8E {
  static long[] rev;
  static long[] inv;
  static long[] two;

  private static void init(int n) {
    rev = new long[n + 1];
    inv = new long[n + 1];
    two = new long[n + 1];
    rev[0] = 1;
    rev[1] = 2;
    inv[0] = 1;
    inv[1] = 1;
    two[0] = 1;
    two[1] = 2;
    for (int i = 2; i <= n; i++) {
      two[i] = two[i - 1] * 2;
      rev[i] = rev[i - 2] * 2 + two[i - 2];
      inv[i] = inv[i - 2] * 2 + two[i - 2];
    }
  }

  private static long[] toNumber(ArrayList<Integer> prefix, int n) {
    int l = Math.min(prefix.size(), n - prefix.size());
    Integer[] list = new Integer[l];
    prefix.toArray(list);
    long[] res = new long[2];
    for (int i = 0; i < l; i++) {
      res[0] *= 2;
      res[0] += prefix.get(i);
      res[1] *= 2;
      res[1] += (1 - prefix.get(i));
    }
    return res;
  }

  private static int check(ArrayList<Integer> prefix, int l) {
    int res = 0;
    int i;
    for (i = l; i < prefix.size(); i++) {
      if (prefix.get(i) < prefix.get(prefix.size() - 1 - i + l)) {
        res++;
        break;
      } else if (prefix.get(i) > prefix.get(prefix.size() - 1 - i + l)) {
        break;
      }
    }
    if (i >= prefix.size()) {
      res++;
    }
    for (i = l; i < prefix.size(); i++) {
      if (prefix.get(i) < 1 - prefix.get(prefix.size() - 1 - i + l)) {
        res++;
        break;
      } else if (prefix.get(i) > 1 - prefix.get(prefix.size() - 1 - i + l)) {
        break;
      }
    }
    if (i >= prefix.size()) {
      res++;
    }
    return res;
  }

  private static long calc(ArrayList<Integer> prefix, int n) {
    int halfLen = n / 2;
    long[] num = toNumber(prefix, n);
    if (prefix.size() <= halfLen) {
      return (num[1] - num[0] - 1) * two[n - 2 * prefix.size()] + rev[n - 2 * prefix.size()]
          + inv[n - 2 * prefix.size()];
    } else {
      int l = n - prefix.size();
      return num[1] - num[0] - 1 + check(prefix, l);
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long k = sc.nextLong();
    sc.close();
    init(n);
    ArrayList<Integer> res = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      res.add(0);
      long p = calc(res, n);
      if (p <= k && i == 0) {
        break;
      } else if (p <= k) {
        res.set(i, 1);
        k -= p;
      }
    }
    if (res.size() < n) {
      System.out.println(-1);
    } else {
      for (int i = 0; i < n; i++) {
        System.out.print(res.get(i));
      }
    }
  }
}
