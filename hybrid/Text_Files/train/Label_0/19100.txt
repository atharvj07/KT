#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int> > a(n, vector<int>(m, 0));
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < s.length(); j++) {
      if (s[j] == '#') a[i][j] = 1;
    }
  }
  vector<vector<int> > b(n, vector<int>(m, -1));
  vector<bool> str(n, false), stl(m, false);
  queue<pair<int, int> > q;
  for (int i = 0; i < m; i++)
    if (a[n - 1][i] == 1) {
      b[n - 1][i] = 1;
      q.push(make_pair(n - 1, i));
    }
  str[n - 1] = false;
  while (!q.empty()) {
    pair<int, int> v = q.front();
    q.pop();
    int x = v.first, y = v.second;
    if (!str[x]) {
      for (int j = 0; j < m; j++) {
        if (a[x][j] == 1 && b[x][j] == -1) {
          b[x][j] = b[x][y] + 1;
          q.push(make_pair(x, j));
        }
      }
      str[x] = true;
    }
    if (!stl[y]) {
      for (int i = 0; i < n; i++) {
        if (a[i][y] == 1 && b[i][y] == -1) {
          b[i][y] = b[x][y] + 1;
          q.push(make_pair(i, y));
        }
      }
      stl[y] = true;
    }
  }
  int min = -1;
  for (int j = 0; j < m; j++)
    if (min == -1 || (b[0][j] != -1 && min > b[0][j])) min = b[0][j];
  cout << min;
  return 0;
}
