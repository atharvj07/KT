bangun = input().split(':')
lama_tidur = input().split(':')
menit = int(bangun[1]) - int(lama_tidur[1])
jam = int(bangun[0]) - int(lama_tidur[0])
if menit < 0:
	menit += 60
	jam -= 1
if jam < 0:
	jam += 24
A = str(menit)
B = str(jam)
if len(A) < 2:
	A = '0' + A
if len(B) < 2:
	B = '0' + B
print(B + ':' + A)
