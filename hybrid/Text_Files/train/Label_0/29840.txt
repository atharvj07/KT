const int mod=1e9+7;
int binpow(int x,int y)
{
	int ret=1;
	while(y>0)
	{
		if(y&1)ret=(ret*x)%mod;
		y>>=1;
		x=(x*x)%mod;
	}
	return ret;
}
signed main()
{
	int t;cin>>t;
	while(t--)
	{
		int n,m;cin>>n>>m;
		int cnt=0;
		for(int i=0;i<n;i++)
		{
			for(int j=0;j<m;j++)
			{
				char c;cin>>c;
				if(c=='#')cnt++;
			}
		}
		int ans=binpow(2,cnt);
		if(cnt==n*m)ans--;
		cout<<ans<<endl;
	}
}
