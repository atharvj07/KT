import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	if n == 1:
		return 'Bob'
	cnt = 0
	v = n
	while v % 2 == 0:
		v //= 2
		cnt += 1
	if v > 1:
		return 'Alice' if cnt > 0 else 'Bob'
	else:
		return 'Alice' if cnt % 2 == 0 else 'Bob'
for _ in range(int(input())):
	print(solve())
