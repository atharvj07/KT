#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int N = 3e5 + 5;
int treeOne[N], treeTwo[N];
int l[N], r[N], vis[N], a[N];
int n, m;
void add1(int x) {
  while (x <= n) {
    treeOne[x] += 1;
    x += x & -x;
  }
}
int sum1(int x) {
  int res = 0;
  while (x) {
    res += treeOne[x];
    x -= x & -x;
  }
  return res;
}
void add2(int x, int val) {
  while (x <= m) {
    treeTwo[x] += val;
    x += x & -x;
  }
}
int sum2(int x) {
  int res = 0;
  while (x) {
    res += treeTwo[x];
    x -= x & -x;
  }
  return res;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) {
    l[i] = r[i] = i;
  }
  for (int i = 1; i <= m; i++) {
    cin >> a[i];
    l[a[i]] = 1;
  }
  for (int i = 1; i <= m; i++) {
    int temp = a[i];
    if (!vis[temp]) {
      add2(i, 1);
      vis[temp] = i;
      add1(temp);
      r[temp] += sum1(n) - sum1(temp);
    } else {
      add2(vis[temp], -1);
      add2(i, 1);
      r[temp] = max(sum2(i - 1) - sum2(vis[temp]) + 1, r[temp]);
      vis[temp] = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (!vis[i])
      r[i] += sum1(n) - sum1(i);
    else
      r[i] = max(sum2(m) - sum2(vis[i]) + 1, r[i]);
  }
  for (int i = 1; i <= n; i++) {
    printf("%d %d\n", l[i], r[i]);
  }
  return 0;
}
