#include<iostream>
#include<algorithm>
using namespace std;
typedef pair<int,int> Q;

int GCD(int a, int b){ return a?GCD(b%a,a):b; }

int main(){
  string s;
  while(cin >> s){
    if(s=="#")return 0;

    Q res(0,0);

    reverse(s.begin(),s.end());
    for(int i=0;i<(int)s.size();){
      if(s[i] == 't'){
	if(res.second == 0){
	  res = make_pair(90,1);
	}else{
	  res.first = res.first * 2 + 90;
	  res.second *= 2;
	}
       
	i += 4;
      }else{
	if(res.second == 0){
	  res = make_pair(0,1);
	}else{
	  res.first = res.first * 2 - 90;
	  res.second *= 2;
	}

	i += 5;
      }
    }
    int gcd = GCD(res.first, res.second);
    res.first /= gcd; res.second /= gcd;
   
    if(res.second == 1)cout << res.first << endl;
    else cout << res.first << "/" << res.second << endl;
  }
}
