#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n;
  cin >> k;
  vector<long long> v(n);
  int x;
  for (int i = 0; i < n; i++) {
    cin >> x;
    if (i == 0)
      v[i] = x;
    else
      v[i] = x + v[i - 1];
  }
  long long Min = v[k - 1];
  int pos = 1;
  for (int i = k; i < n; i++) {
    if (v[i] - v[i - k] < Min) {
      Min = v[i] - v[i - k];
      pos = i - k + 2;
    }
  }
  cout << pos << endl;
}
