#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int par[N], sz[N], S[N], T[N];
vector<int> adj[N];
vector<pair<int, int> > E, ans;
int getPar(int v) { return (par[v] == v ? v : par[v] = getPar(par[v])); }
bool mrg(int u, int v) {
  u = getPar(u);
  v = getPar(v);
  if (u == v) return 0;
  if (sz[v] < sz[u]) swap(u, v);
  par[u] = v;
  sz[v] += sz[u];
  return 1;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m, u, v, s, t, ds, dt;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    cin >> u >> v;
    E.push_back({u, v});
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  cin >> s >> t >> ds >> dt;
  for (int i = 0; i < N; i++) par[i] = i, sz[i] = 1;
  for (int i = 0; i < m; i++) {
    u = E[i].first;
    v = E[i].second;
    if (u != s && u != t && v != s && v != t)
      if (mrg(u, v)) ans.push_back({u, v});
  }
  for (auto v : adj[s])
    if (v != t) S[getPar(v)] = v;
  for (auto v : adj[t])
    if (v != s) T[getPar(v)] = v;
  for (auto v : adj[s])
    if (v != t && T[getPar(v)] == 0)
      if (mrg(v, s)) ans.push_back({v, s}), ds--;
  fill(S, S + N, 0);
  fill(T, T + N, 0);
  for (auto v : adj[s])
    if (v != t) S[getPar(v)] = v;
  for (auto v : adj[t])
    if (v != s) T[getPar(v)] = v;
  for (auto v : adj[t])
    if (v != s && S[getPar(v)] == 0)
      if (mrg(v, t)) ans.push_back({v, t}), dt--;
  fill(S, S + N, 0);
  fill(T, T + N, 0);
  for (auto v : adj[s])
    if (v != t) S[getPar(v)] = v;
  for (auto v : adj[t])
    if (v != s) T[getPar(v)] = v;
  bool f = 0;
  for (int i = 1; i <= n; i++) {
    int x = S[getPar(i)], y = T[getPar(i)];
    if (x && y) {
      if (f == 0) {
        if (mrg(x, s)) ans.push_back({x, s}), ds--;
        if (mrg(y, t)) ans.push_back({y, t}), dt--;
        f = 1;
      } else {
        if (ds < dt) {
          if (mrg(y, t)) ans.push_back({y, t}), dt--;
        } else {
          if (mrg(x, s)) ans.push_back({x, s}), ds--;
        }
      }
    }
  }
  if (getPar(s) != getPar(t)) ans.push_back({s, t}), ds--, dt--;
  if (ds < 0 || dt < 0) return cout << "No", 0;
  cout << "Yes\n";
  for (auto x : ans) cout << x.first << ' ' << x.second << '\n';
  return 0;
}
