#include <bits/stdc++.h>        
using namespace std;
#define ll long long
#define nline "\n"
const ll N=2e5+4; 
const ll M=1e9+7;
void solve(){
    ll n,k; cin>>n>>k;
    vector<char> ans;   
    while(ans.size()<n){    
        for(ll i=0;i<k;i++){  
            ans.push_back(i+'a'); 
            for(ll j=i+1;j<k;j++){ 
                ans.push_back(i+'a');
                ans.push_back(j+'a');  
            }
        }  
    }  
    for(ll i=0;i<n;i++)     
        cout<<ans[i];
    return;

}
int main()
