class FastScanner {

    BufferedReader br;
    StringTokenizer st;

    FastScanner(InputStreamReader reader) {
        br = new BufferedReader(reader);
        st = new StringTokenizer("");
    }

    String next() {
        while (!st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException ex) {
                Solution.MLE();
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }
}

class Solver {

    FastScanner sc;
    PrintWriter out;
    PrintStream err = System.err;

    static ArrayList<Integer>[] prim;
    static {
        prim = new ArrayList[(int)2e6+1];
        for (int i = 0; i < prim.length; i++) {
            prim[i] = new ArrayList<>(7);
        }
        boolean[] isP = new boolean[prim.length];
        fill( isP, true );
        int cntPrimes = 0;
        for (int i = 2; i < isP.length; i++) {
            if( isP[i] ){
                ++cntPrimes;
                for (int j = i; j < isP.length; j +=i ) {
                    prim[j].add(i);
                    isP[j] = false;
                }
            }
        }
        System.err.println("cntPrimes: " + cntPrimes);
    }

    int n, q;
    int[] val;
    ArrayList<Integer>[] gr;
    int[] sz, goUp;
    
    void dfs( int v, int pr ){
        goUp[v] = pr;
        sz[v] = 1;
        for( int to : gr[v] ){
            if( to == pr ) continue;
            dfs( to, v );
            sz[v] += sz[to];
        }
    }

    boolean cross( ArrayList<Integer> aa, ArrayList<Integer> bb ){
        if( aa.size() > bb.size() )
            return cross(bb, aa);
        for( int a : aa )
            if( 0 <= Collections.binarySearch(bb, a) )
                    return true;
        return false;
    }
    
    void solve(){
        n = sc.nextInt();
        q = sc.nextInt();
        val = new int[n+1];
        for (int i = 1; i <= n; i++) {
            val[i] = sc.nextInt();
        }
        gr = new ArrayList[n+1];
        for (int v = 1; v <= n; v++){
            gr[v] = new ArrayList<>();
        }
        for (int it = 0; it < n-1; it++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            gr[a].add(b);
            gr[b].add(a);
        }
        
        sz = new int[n+1];
        goUp = new int[n+1];
        dfs(1, -1);

        
        cycMain:
        for (int iter = 0; iter < q; iter++) {
            int op = sc.nextInt();
            int v = sc.nextInt();
            if     ( op == 1 ){
                ArrayList<Integer> primList = prim[val[v]];
                v = goUp[v];
                while( 1 <= v ){
                    if( cross(primList, prim[val[v]]) ){
                        out.println( v );
                        continue cycMain;
                    }
                    v = goUp[v];
                }
                out.println( "-1" );
            }
            else if( op == 2 ){
                int w = sc.nextInt();
                val[v] = w;
            }
            else Solution.MLE();
        }
    }
    
}
