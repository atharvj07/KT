n = int(input())
s = list(map(int, input().split()))
ma1 = 0
ma2 = 0
mi1 = 1000000000
mi2 = 1000000
for i in s:
	if i >= ma2:
		if i >= ma1:
			ma2 = ma1
			ma1 = i
		else:
			ma2 = i
	if i <= mi2:
		if i <= mi1:
			mi2 = mi1
			mi1 = i
		else:
			mi2 = i
if ma2 - mi1 > ma1 - mi2:
	print(ma1 - mi2)
else:
	print(ma2 - mi1)
