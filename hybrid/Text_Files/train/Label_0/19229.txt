#include <bits/stdc++.h>
using namespace std;
stack<int> st;
int ans[200000], it;
char c;
string s;
int main() {
  while (~scanf("%c", &c)) {
    if (c == '/n') continue;
    if (c == '>') {
      if (s == "/td") ans[st.top()]++;
      if (s == "table") st.push(it++);
      if (s == "/table") st.pop();
      s = "";
    };
    if ((c >= 'a' && c <= 'z') || c == '/') s += c;
  };
  sort(ans, ans + it);
  for (int i = 0; i < it; i++) cout << ans[i] << " ";
  return 0;
}
