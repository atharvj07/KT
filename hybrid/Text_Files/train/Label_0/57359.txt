import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskE solver = new TaskE();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskE {
        public void solve(int testNumber, InputReader in, PrintWriter out) {
            String s = in.next();
            int[] prefLen = new int[s.length()];
            boolean firstLit = ('a' <= s.charAt(0) && s.charAt(0) <= 'z');
            if (firstLit) {
                prefLen[0] = 1;
            }
            for (int i = 1; i < s.length(); i++) {
                boolean lit = ('a' <= s.charAt(i) && s.charAt(i) <= 'z');
                if (lit || ('0' <= s.charAt(i) && s.charAt(i) <= '9') || s.charAt(i) == '_') {
                    prefLen[i] = prefLen[i - 1];
                    if (lit) {
                        prefLen[i]++;
                    }
                }
            }
            int[] midLen = new int[s.length()];
            boolean lastLit = ('a' <= s.charAt(s.length() - 1) && s.charAt(s.length() - 1) <= 'z');
            if (lastLit || ('0' <= s.charAt(s.length() - 1) && s.charAt(s.length() - 1) <= '9')) {
                midLen[s.length() - 1] = 1;
            }
            for (int i = s.length() - 2; i >= 0; i--) {
                boolean lit = ('a' <= s.charAt(i) && s.charAt(i) <= 'z');
                boolean num = ('0' <= s.charAt(i) && s.charAt(i) <= '9');
                if (lit || num) {
                    midLen[i] = midLen[i + 1] + 1;
                }
            }
            int[] sufLen = new int[s.length()];
            if (lastLit) {
                sufLen[s.length() - 1] = 1;
            }
            for (int i = s.length() - 2; i >= 0; i--) {
                boolean lit = ('a' <= s.charAt(i) && s.charAt(i) <= 'z');
                if (lit) {
                    sufLen[i] = sufLen[i + 1] + 1;
                }
            }
            long ans = 0;
            for (int i = 0; i < s.length(); i++) {
                if (s.charAt(i) == '@') {
                    long curAns = 0;
                    if (i > 0) {
                        curAns = prefLen[i - 1];
                    }
                    if (i + 1 < s.length() && midLen[i + 1] > 0) {
                        int sufPos = i + midLen[i + 1];
                        if (sufPos + 2 < s.length() && s.charAt(sufPos + 1) == '.') {
                            curAns *= sufLen[sufPos + 2];
                            ans += curAns;
                        }
                    }
                }
            }
            out.println(ans);
        }

    }

    static class InputReader {
        private BufferedReader reader;
        private StringTokenizer stt;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream));
        }

        public String nextLine() {
            try {
                return reader.readLine();
            } catch (IOException e) {
                return null;
            }
        }

        public String next() {
            while (stt == null || !stt.hasMoreTokens()) {
                stt = new StringTokenizer(nextLine());
            }
            return stt.nextToken();
        }

    }
}
