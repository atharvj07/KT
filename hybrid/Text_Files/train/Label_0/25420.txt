#include <bits/stdc++.h>
using namespace std;
int prime[103];
int dp[103][300003];
long long dfs(long long n, int m) {
  if (prime[0] == 1 || n == 0) return 0;
  if (n < 300001 && dp[m][n] > -1) return dp[m][n];
  long long ans = 0;
  if (m == 0)
    ans = n;
  else
    ans = dfs(n, m - 1) - dfs(n / prime[m - 1], m - 1);
  if (n < 300001) dp[m][n] = ans;
  return ans;
}
int main() {
  int i, m;
  long long n;
  memset(dp, -1, sizeof(dp));
  scanf("%I64d %d", &n, &m);
  for (i = 0; i < m; i++) scanf("%d", prime + i);
  sort(prime, prime + m);
  cout << dfs(n, m) << endl;
  return 0;
}
