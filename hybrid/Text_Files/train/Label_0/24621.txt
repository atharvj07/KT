#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, x, y, x1, y1, x2, y2, c = 0;
  char a;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a;
      if (a == '*') {
        if (c == 0) {
          x1 = i;
          y1 = j;
          c++;
        } else if (c == 1) {
          x2 = i;
          y2 = j;
          c++;
        } else if (c == 2) {
          if (i == x1) {
            x = x2 + 1;
          } else {
            if (x1 == x2) {
              x = i + 1;
            } else {
              x = x1 + 1;
            }
          }
          if (j == y1) {
            y = y2 + 1;
          } else {
            if (y1 == y2) {
              y = j + 1;
            } else {
              y = y1 + 1;
            }
          }
          c++;
        }
      }
    }
  }
  cout << x << " " << y;
  return 0;
}
