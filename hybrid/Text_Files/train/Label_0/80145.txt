import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	days = []
	ans = []
	for i in range(m):
		days.append(list(map(int, input().split())))
		ans.append(days[i][1])
	freq = {}
	for num in ans:
		if num in freq:
			freq[num] += 1
		else:
			freq[num] = 1
	f = max(freq.keys(), key=lambda k: freq[k])
	z = freq[f]
	if z <= (m + 1) // 2:
		print('YES')
		print(*ans)
	else:
		for i in range(m):
			if ans[i] == f:
				if days[i][0] > 1:
					z -= 1
					ans[i] = days[i][2]
			if z <= (m + 1) // 2:
				break
		else:
			print('NO')
			continue
		print('YES')
		print(*ans)
