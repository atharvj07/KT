#include<bits/stdc++.h>

using namespace std;
#define taskname ""
#define pb	push_back
typedef long double ld;
typedef long long ll;
const int maxn = 2e5 + 5;
string s;
int sum[maxn] , n;
vector<int> v[26];
vector<pair<int,int>> adj;
void update(int x , int val)
{
    for( ; x > 0 ; x &= x - 1)
        sum[x] += val;
}
int query(int x)
{
    int res = 0;
    for( ; x <= n ; x += x & -x)
        res += sum[x];
    return res;
}
int main()
{
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	//freopen(taskname".INP", "r",stdin);
	//freopen(taskname".OUT", "w",stdout);
    cin >> s;
    n = s.size();
    s = " " + s;
    for(int i = 1 ; i <= n ; ++i)
        v[s[i] - 'a'].pb(i);
    int odd = 0;int posodd = 0;
    for(int i = 0 ; i < 26 ; ++i)
    {
        odd += v[i].size() % 2;
        int k = v[i].size();
        for(int j = 0 ;  j < k / 2 ; ++j)
        {
            adj.pb({v[i][j] , v[i].back()});
            v[i].pop_back();
        }
        if(k % 2 == 1)posodd = v[i].back();
    }
    if(odd > 1)return cout << -1 ,0;
    sort(adj.begin() , adj.end());
    for(int i = 1 ; i <= n ; ++i)if(posodd != i)update(i, 1);
    ll res = 0;
    for(pair<int,int> c : adj)
    {
        if(odd && ((c.first < posodd && c.second < posodd) || (c.second > posodd && c.first > posodd)))res++;
        res += query(c.second + 1);
        update(c.second , -1);
    }
    cout << res;
}
