import sys
input = sys.stdin.readline
inf = int(10000000000.0)
n = int(input())
a = []
b = []
for i in range(n):
	(x, y) = map(int, input().split())
	a.append(x)
	b.append(y)
d = [0] * n
for i in range(n):
	d[i] = a[i] - b[i]
(mx, mn, i1, i2) = (-inf, inf, 0, 0)
for i in range(n):
	if d[i] < mn:
		mn = d[i]
		i2 = i
	if d[i] > mx:
		mx = d[i]
		i1 = i
s = sum(a) - sum(b)
if abs(s) > abs(s - 2 * mx) and abs(s) > abs(s - 2 * mn):
	print(0)
elif abs(s - 2 * mx) > abs(s - 2 * mn):
	print(i1 + 1)
else:
	print(i2 + 1)
