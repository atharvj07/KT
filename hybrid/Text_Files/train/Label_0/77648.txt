def check(sep):
	global paint
	last1 = -1
	last2 = -1
	for i in range(n):
		si = int(s[i])
		if si < sep:
			if si < last1:
				return False
			paint[i] = '1'
			last1 = si
		elif si > sep:
			if si < last2:
				return False
			paint[i] = '2'
			last2 = si
		elif si >= last2:
			paint[i] = '2'
			last2 = si
		else:
			paint[i] = '1'
			last1 = si
	return True
T = int(input())
for t in range(T):
	n = int(input())
	s = input()
	ok = False
	for i in range(10):
		paint = [0] * n
		if check(i):
			print(''.join(paint))
			ok = True
			break
	if not ok:
		print('-')
