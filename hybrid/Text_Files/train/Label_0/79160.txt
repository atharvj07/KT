(n, a, x, b, y) = map(int, input().split())
s1 = a
s2 = b
flag = False
while s1 != x and s2 != y:
	if s1 == s2:
		flag = True
		break
	if s1 != x:
		if s1 < n:
			s1 += 1
		else:
			s1 = 1
	if s2 != y:
		if s2 > 1:
			s2 -= 1
		else:
			s2 = n
	if s1 == s2:
		flag = True
		break
if flag:
	print('YES')
else:
	print('NO')
