#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
inline int gm(int x, int n)
{
	if(x>n)
		x-=n;
	if(x<=0)
		x+=n;
	return x;
}
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(NULL);
	int n,x;
	cin>>n>>x;
	int z = 2*n-1;
	if(x==1||x==z)
	{
		cout<<"No\n";
		return 0;
	}
	cout<<"Yes\n";
	int st = gm(x-n+1,z);
	for(int i=0; i<z; i++)
	{
		cout<<st<<"\n";
		st=gm(st+1,z);
	}
}
