from functools import reduce

def gao(x, y):
	if x == 1:
		return 1
	if x == 2:
		return (~y & 1) * 2
	if x == 3:
		return y & 1
	if x & 1:
		return 0
	if ~y & 1:
		return 1
	l = len(bin(x)) - bin(x).rfind('1') - 1
	if x >> l == 3:
		l += 1
	return 2 - l % 2
(n, k) = map(int, input().split())
s = reduce(lambda x, y: x ^ gao(y, k), map(int, [0] + input().split()))
if s:
	print('Kevin')
else:
	print('Nicky')
