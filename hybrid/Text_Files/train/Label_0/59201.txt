class FastScanner {

    BufferedReader br;
    StringTokenizer st;

    FastScanner(InputStreamReader reader) {
        br = new BufferedReader(reader);
        st = new StringTokenizer("");
    }

    String next() {
        while (!st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException ex) {
                Main.MLE();
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }
}

class Solver {

    FastScanner sc;
    PrintWriter out;
    PrintStream err = System.err;

    int n, sqrt;
    ArrayList<Integer>[] arr;
    int[][] cnt;
    //ArrayList<Integer> bfArr = new ArrayList<>();
    
    void initArr(){
        arr = new ArrayList[sqrt];
        for (int i = 0; i < sqrt; i++) {
            arr[i] = new ArrayList<>(2*sqrt+10);
            for (int j = 0; j < sqrt; j++) {
                if( i*sqrt + j < n )
                    arr[i].add(-1);
            }
        }
    }
    
    void solve(){
        n = sc.nextInt();
        
        sqrt = 1;
        while( sqrt*sqrt < n ) ++sqrt;
        
        initArr();
        cnt = new int[sqrt][n+1];
        
        for( int i = 0; i < n; ++i ){
            int val = sc.nextInt();
//            bfArr.add(val);
            int b = i/sqrt;
            arr[b].set(i%sqrt, val);
            cnt[b][val]++;
        }
        
        
        int Q = sc.nextInt();
        int lastans = 0;
        for (int iter = 0; iter < Q; iter++) {
            //if( iter % 1000 == 0 ) err.println( iter );
            int op = sc.nextInt();
            int ql = sc.nextInt();
            int qr = sc.nextInt();
            ql = (ql + lastans - 1) % n;
            qr = (qr + lastans - 1) % n;
            if( ql > qr ){ int t = ql; ql = qr; qr = t; }
            
            if     ( op == 1 ){
                if( iter % sqrt == 0 ){
                    int sz = 0;
                    int[] val = new int[n];
                    for ( int b = 0; b < arr.length; ++b){
                        ArrayList<Integer> arrayList = arr[b];
                        for (Integer v : arrayList){
                            if( sz < n ){
                                val[sz++] = v;
                                cnt[b][v]--;
                            }
                        }
                    }
                    
                    initArr();
                    

                    for (int i = 0; i < n; ++i) {
                        int v = val[i];
                        int b = i / sqrt;
                        arr[b].set(i % sqrt, v);
                        cnt[b][v]++;
                    }
                }
                int valR = remove(qr);
                add( valR, ql );
                //
//                Collections.rotate(bfArr.subList(ql, qr+1), 1);
                //
                
//                err.println(Arrays.deepToString(arr));
//                err.println(bfArr);
//                err.println();
            }
            else if( op == 2 ){
                int k = sc.nextInt();
                k = (k + lastans - 1) % n + 1;
                lastans = 0;
                for (int b = 0, sz = 0; b < sqrt; sz += arr[b].size(), b++) {
                    int l = sz;
                    int r = l + arr[b].size() - 1;
                    if( max(l,ql) <= min(r,qr) ){
                        if( ql <= l && r <= qr ){
                            lastans += cnt[b][k];
                        }
                        else{
                            for( int i = max(l,ql); i <= min(r,qr); ++i )
                                if( arr[b].get(i-sz) == k )
                                    ++lastans;
                        }
                    }
                }
                
//                int bfAns = 0; for( int i = ql; i <= qr; ++i )
//                    if( bfArr.get(i) == k ) ++bfAns;
//                err.println("bfAns: " + bfAns);
//                err.println( "ql, qr: " + ql +" "+ qr );
//                err.println();
                
                out.println(lastans);
            }
            else Main.MLE();
        }
    }

    int remove(int pos) {
        for (int b = 0, sz = 0; b < sqrt; sz += arr[b].size(), b++) {
            int l = sz;
            int r = l + arr[b].size() - 1;
            if ( l <= pos && pos <= r ) {
                int val = arr[b].remove( pos - sz );
                --cnt[b][val];
                return val;
            }
        }
        throw new RuntimeException();
    }

    void add(int val, int pos) {
        for (int b = 0, sz = 0; b < sqrt; sz += arr[b].size(), b++) {
            int l = sz;
            int r = l + arr[b].size() - 1;
            if ( l <= pos && pos <= r ) {
                cnt[b][val]++;
                arr[b].add( pos - sz, val );
                return;
            }
        }
    }

}
