import sys
input = lambda : sys.stdin.readline().rstrip()
for _ in range(int(input())):
	n = int(input())
	A = list(map(int, input().split()))
	Result = [0]
	for a in A:
		Result.append(Result[-1] ^ a)
	r = Result[-1]
	if not r:
		b = 0
	else:
		idx = Result.index(r)
		if idx < n:
			S = set(Result[idx + 1:])
			if 0 in S:
				b = 0
			else:
				b = 1
		else:
			b = 1
	print('YNEOS'[b::2])
