for _ in range(int(input())):
	s = input()
	l = []
	if len(s) <= 2:
		print(s)
	elif len(set(list(s))) == 1:
		print(s)
	else:
		l.append(s[0])
		for i in range(1, len(s)):
			if s[i] == s[i - 1]:
				if s[i] == '1':
					l.append('0')
				else:
					l.append('1')
			l.append(s[i])
		print(''.join(l))
