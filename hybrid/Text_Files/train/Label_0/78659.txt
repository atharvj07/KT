numberl = []
(a, b) = map(int, input().split())
starr1 = list(map(str, input().split()))
starr2 = list(map(str, input().split()))
num = int(input())
for i in range(num):
	ele = int(input())
	numberl.append(ele)
for i in range(num):
	x = numberl[i] % a
	y = numberl[i] % b
	if x == 0:
		x = a
	if y == 0:
		y = b
	print(starr1[x - 1] + starr2[y - 1])
