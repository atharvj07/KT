#include <climits>
#include <iostream>

using namespace std;
using ll = int64_t;

ll extgcd(ll a, ll b, ll &x, ll &y) {
  ll d = a;
  if (b != 0) {
    d = extgcd(b, a % b, y, x);
    y -= (a / b) * x;
  } else {
    x = 1;
    y = 0;
  }
  return d;
}

int main() {
  ll n;
  cin >> n;
  n <<= 1;
  ll ans = n - 1;
  for (ll i = 2; i * i <= n; ++i) {
    if (n % i != 0)
      continue;
    ll rem = n / i;
    ll x, y;
    ll g = extgcd(i, rem, x, y);
    if (g != 1)
      continue;
    ll t = x / rem;
    x %= rem;
    y += t * i;
    ans = min(ans, min(abs(x) * i, abs(y) * rem));
    if (x > 0)
      x -= rem, y += t;
    else
      x += rem, y -= t;
    ans = min(ans, min(abs(x) * i, abs(y) * rem));
  }
  cout << ans;
}
