from sys import stdin
from itertools import accumulate

def arr_inp(n):
	if n == 1:
		return [int(x) for x in stdin.readline().split()]
	elif n == 2:
		return [float(x) for x in stdin.readline().split()]
	else:
		return list(stdin.readline()[:-1])

def arr_sum(arr):
	arr.insert(0, 0)
	return list(accumulate(arr, lambda x, y: x + y))

def bs(be, en, x, arr):
	ix = be
	while be < en:
		mid = (be + en) // 2
		val = arr[mid] - arr[ix]
		if val == x:
			return mid
		elif val > x:
			en = mid
		else:
			be = mid + 1
	return en
(n, q) = arr_inp(1)
(a, k) = [arr_inp(1) for i in range(2)]
(cum, ix, ext) = (arr_sum(a), 0, 0)
for i in range(q):
	if ext > k[i]:
		print(n - ix + 1)
		ext -= k[i]
	elif ext == k[i]:
		ext = 0
		if ix >= n:
			ix = 0
		print(n - ix)
	else:
		k[i] -= ext
		ext = 0
		nx = bs(ix, n, k[i], cum)
		if cum[nx] - cum[ix] <= k[i]:
			if nx >= n:
				ix = 0
				nx = 0
			print(n - nx)
		else:
			ext = cum[nx] - cum[ix] - k[i]
			print(n - nx + 1)
		ix = nx
