import java.util.Scanner;

public class CF246_SquareTiling {

	public static void main(String[] args) {

		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();

		int[][] arr = new int[n][m];

		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				if (arr[i][j] == 0) {
					int minPlace = minCanPlace(arr, i, j);
					// Lookahead
					int dist = 1;
					for (int jLook = j + 1; jLook < m; jLook++) {
						if (arr[i][jLook] != 0 || minCanPlace(arr, i, jLook) != minPlace) {
							break;
						}
						dist++;
					}

					// Fill up to the lookahead
					fillDistSquare(arr, i, j, Math.min(dist, n - i), minPlace);
				}
			}

		}

		StringBuilder sb = new StringBuilder();
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				sb.append((char) (arr[i][j] - 1 + 'A'));
			}
			sb.append('\n');
		}

		System.out.println(sb);
	}

	public static int minCanPlace(int[][] arr, int iStart, int jStart) {

		boolean[] taken = new boolean[27];

		// Check top
		int iInd = iStart - 1;
		int jInd = jStart;
		if (iInd >= 0 && iInd < arr.length && jInd >= 0 && jInd < arr[0].length) {
			taken[arr[iInd][jInd]] = true;
		}

		// Check bot
		iInd = iStart + 1;
		jInd = jStart;
		if (iInd >= 0 && iInd < arr.length && jInd >= 0 && jInd < arr[0].length) {
			taken[arr[iInd][jInd]] = true;
		}

		// Check left
		iInd = iStart;
		jInd = jStart - 1;
		if (iInd >= 0 && iInd < arr.length && jInd >= 0 && jInd < arr[0].length) {
			taken[arr[iInd][jInd]] = true;
		}

		// Check right
		iInd = iStart;
		jInd = jStart + 1;
		if (iInd >= 0 && iInd < arr.length && jInd >= 0 && jInd < arr[0].length) {
			taken[arr[iInd][jInd]] = true;
		}

		// Find the smallest untaken
		for (int i = 1; i <= 26; i++) {
			if (!taken[i]) {
				return i;
			}
		}

		return -1;
	}

	public static void fillDistSquare(int[][] arr, int iStart, int jStart, int dist, int val) {

		for (int i = 0; i < dist; i++) {
			for (int j = 0; j < dist; j++) {
				arr[iStart + i][jStart + j] = val;
			}
		}

	}

	public static void fillMaxSquareA(int[][] arr, int iStart, int jStart) {

		int val = 1;

		int dist = Math.min(arr.length - iStart, arr[0].length - jStart);

		for (int i = 0; i < dist; i++) {
			for (int j = 0; j < dist; j++) {
				arr[iStart + i][jStart + j] = val;
			}
		}

	}
