import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())
MOD = 1000000000 + 7

def add(n, v):
	return (n + v) % MOD
n = inp()
s = insr()
prev = {'none': 1, 'a': 0, 'ab': 0}
total = 0
for i in range(n):
	curt = prev.copy()
	if s[i] == 'a':
		curt['a'] = add(curt['a'], prev['none'])
	elif s[i] == 'b':
		curt['ab'] = add(curt['ab'], prev['a'])
	elif s[i] == 'c':
		total = add(total, prev['ab'])
	elif s[i] == '?':
		curt['none'] = prev['none'] * 3 % MOD
		curt['a'] = (prev['a'] * 3 + prev['none']) % MOD
		curt['ab'] = (prev['ab'] * 3 + prev['a']) % MOD
		total = (total * 3 + prev['ab']) % MOD
	prev = curt
print(total)
