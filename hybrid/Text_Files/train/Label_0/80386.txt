n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
a.sort()
b.sort()
p1 = n - 1
p2 = n - 1
sum1 = 0
sum2 = 0
while 1:
	if p1 != -1 and p2 != -1:
		if a[p1] < b[p2]:
			p2 -= 1
		else:
			sum1 += a[p1]
			p1 -= 1
	elif p1 == -1 and p2 != -1:
		p2 -= 1
	elif p2 == -1 and p1 != -1:
		sum1 += a[p1]
		p1 -= 1
	else:
		break
	if p1 != -1 and p2 != -1:
		if b[p2] < a[p1]:
			p1 -= 1
		else:
			sum2 += b[p2]
			p2 -= 1
	elif p2 == -1 and p1 != -1:
		p1 -= 1
	elif p2 != -1 and p1 == -1:
		sum2 += b[p2]
		p2 -= 1
	else:
		break
print(sum1 - sum2)
