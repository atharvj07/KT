#include <bits/stdc++.h>
using namespace std;
int main() {
  string to, in, s = "RRRRPSSPRR";
  for (int i = 0; i < 10; i++) {
    cout << s[i] << endl;
    cout.flush();
    getline(cin, in);
    if (in == "ai")
      to += "0";
    else
      to += "1";
  }
  if (to == "1111000011") s = "RRRRRRRRRR";
  if (to == "0000100100") s = "PPPPPPPPPP";
  if (to == "0000011000") s = "SSSSSSSSSS";
  if (to == "0010000010") s = "SRPSRPSRPS";
  if (to == "0000110000") s = "PSRPSRPSRP";
  if (to == "0000000110") s = "SPRSPRSPRS";
  for (int i = 0; i < 10; i++) {
    cout << s[i] << endl;
    cout.flush();
    getline(cin, in);
  }
  return 0;
}
