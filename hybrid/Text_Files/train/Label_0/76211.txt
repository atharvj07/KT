t = int(input())
for _ in range(t):
	n = int(input())
	p = list(map(int, input().split()))
	ans = []
	p1 = [-1] * (n + 1)
	for i in range(n):
		p1[p[i]] = i
	i = n
	while i:
		while i > 0 and p1[i] == -1:
			i -= 1
		else:
			if i:
				k = 0
				for j in range(p1[i], n):
					ans.append(p[j])
					p1[p[j]] = -1
					k += 1
				n -= k
				i -= 1
			else:
				break
	print(*ans)
