def arrayStabilization(ls):
	if ls.count(max(ls)) > 1 and ls.count(min(ls)) > 1:
		return max(ls) - min(ls)
	ls.sort()
	n = len(ls)
	if len(ls) == 1:
		return 0
	elif ls.count(max(ls)) == 1 and ls.count(min(ls)) == 1:
		rmin = ls[n - 1] - ls[1]
		rmax = ls[n - 2] - ls[0]
		return min(rmin, rmax)
	elif ls.count(max(ls)) > 1:
		return ls[n - 1] - ls[1]
	else:
		return ls[n - 2] - ls[0]
n = int(input())
ls = list(map(int, input().split(' ')))
print(arrayStabilization(ls))
