import sys
from collections import Counter
INP = lambda : sys.stdin.readline().strip()
INT = lambda : int(INP())
MAP = lambda : map(int, INP().split())
ARR = lambda : [int(i) for i in INP().split()]

def JOIN(arr, x=' '):
	return x.join([str(i) for i in arr])

def EXIT(x='NO'):
	print(x)
	exit()
for _ in range(INT()):
	(n, arr) = (INT(), ARR())
	d = Counter(arr)
	s = sum(arr)
	found = False
	for x in arr:
		y = s - x
		if y % 2 == 1:
			continue
		y = y // 2
		if y == x and d[x] == 1:
			continue
		if y in d:
			arr.remove(x)
			arr.remove(y)
			found = True
			break
	if found:
		print(JOIN(arr))
	else:
		print(-1)
