from collections import deque

class Solution:

	def beat_monsters(self, caves_data):
		caves_data.sort(key=lambda x: x[0])
		(total_monsters, minimum_power) = (0, caves_data[0][0])
		for (max_armor_cave, monsters) in caves_data:
			extra_power = max(max_armor_cave - total_monsters - minimum_power, 0)
			minimum_power += extra_power
			total_monsters += monsters
		print(minimum_power)
s = Solution()
cases = int(input())
for case in range(cases):
	caves_data = []
	t_caves = int(input())
	for cave in range(t_caves):
		cave_info = input().split(' ')
		(max_armor_cave, monsters_in_cave, beated_monsters) = (0, int(cave_info[0]), 0)
		for monster in range(1, monsters_in_cave + 1):
			current_monster = int(cave_info[monster])
			max_armor_cave = max(current_monster - beated_monsters + 1, max_armor_cave)
			beated_monsters += 1
		caves_data.append([max_armor_cave, monsters_in_cave])
	s.beat_monsters(caves_data)
