    private static int maxN = 500000;
    private static int[] a = new int[maxN + 10];
    private static int[] sum = new int[maxN + 10];
    private static int[] input = new int[maxN + 10];

    public static void task() {
        int n = scanner.nextInt();
        // System.out.println(n);
        for (int i = 0; i < n; i++) {
            input[i] = scanner.nextInt();
        }
        // System.out.println("input end");
        solve(n, input);
    }

    public static void clear(int n) {
        for (int i = 1; i <= maxN + 1; i++) {
            a[i] = 0;
            sum[i] = 0;
        }
    }

    private static int maxA;

    public static long solve(int n, int[] input) {
        long maxArea = 0;
        maxA = 0;
        clear(n);
        for (int i = 0; i < n; i++) {
            a[input[i]]++;
            if (input[i] > maxA) {
                maxA = input[i];
            }
        }
        for (int i = 2; i <= maxN + 1; i++) {
            sum[i] = a[i] + sum[i - 1];
        }

        for (int y = 2; y <= maxA; y++) {
            long total = 0;
            for (int step = y; step <= maxA; step += y) {
                total += q(step, min(step + y - 1, maxA) + 1) * (long) (step / y);
            }
            int x = 1;
            int left = 2;
            int right = maxA;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (check(y, mid, total)) {
                    if (mid > x) {
                        x = mid;
                    }
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
            if (x > 1) {
                if ((long) x * (long) y > maxArea) {
                    maxArea = (long) x * (long) y;
                    /*
                    System.out.println(y + " " + x);
                    System.out.println(maxArea);
                     */
                }
            }
        }
        System.out.println(maxArea);

        return maxArea;
    }

    public static boolean check(int y, int x, long total) {
        if (y == 4) {
           // System.out.println("fcuk");
        }
        boolean ret = check1(y, x, total, x + x, false) || check2(y, x, total);
        //System.out.println(y + " " + x + " " + total + " " + ret);
        return ret;
    }

    public static boolean check1(int y, int x, long total, int need, boolean ignore) {
        // together
        long up = (need - 1) / y + 1;
        long down = need / y;
        int hits = 0;
        if (!ignore) {
            if (need < y) {
                if (q(need, y) > 0) {
                    hits = 1;
                }
            }
        }
        if (hits == 1) {
            return total >= x;
        }
        int toSub = 0;
        int remain = need % y;
        for (int step = max(y, need - remain); step + remain <= maxA; step += y) {
            if (q(step + remain, min(step + y,maxA + 1)) > 0) {
                toSub = 1;
                break;
            }
        }
        if (toSub == 1) {
            return total >= x + down;
        } else {
            if (q(need, maxA + 1) > 0) {
                return total >= x + up;
            } else {
                return false;
            }
        }
    }

    private static int min(int x, int y) {
        if (x < y) {
            return x;
        }
        return y;
    }

    private static int max(int x, int y) {
        if (x > y) {
            return x;
        }
        return y;
    }


    public static boolean check2(int y, int x, long total) {
        // separate
        int need = x;
        int up = (need - 1) / y + 1;
        int down = need / y;
        int hits = 0;
        if (need < y) {
            if (q(need, y) > 0) {
                hits = q(need, y);
            }
            if (hits >= 2) {
                hits = 2;
            }
        }
        if (hits == 2) {
            return total >= x;
        }
        // ?
        if (hits == 1) {
            return check1(y, x, total, x, true);
        }
        if (q(need, maxA + 1) < 2){
            return false;
        }
        if (hits == 0) {
            int toSub = 0;
            int remain = need % y;
            for (int step = max(y, need - remain); step + remain <= maxA; step += y) {
                int h = q(step + remain, min(step + y,maxA + 1));
                toSub += h;
                if (toSub >= 2) {
                    toSub = 2;
                    break;
                }
            }
            if (toSub == 2) {
                return total >= x + down + down;
            }
            if (toSub == 1) {
                return total >= x + down + up;
            }
            if (toSub == 0) {
                return total >= x + up + up;
            }
        }
        // never
        return false;
    }

    public static int q(int begin, int end) {
        if (begin > end) {
            return 0;
        }
        return sum[end - 1] - sum[begin - 1];
    }
