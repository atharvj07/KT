#include <bits/stdc++.h>
using namespace std;
const int maxv = 3005;
const int maxn = 1e5 + 5;
int n;
int xl[maxn], yl[maxn], xr[maxn], yr[maxn];
int dlt[maxv][maxv], f[maxv][maxv], g[maxv][maxv], cnt[maxv][maxv];
vector<int> tp[maxv], bt[maxv];
inline bool check(int x1, int y1, int x2, int y2) {
  if (!cnt[x2][y2]) return false;
  if (!x2 || !y2 || x2 > 3000 || y2 > 3000) return false;
  if (dlt[x2 - 1][y2 - 1] - (x1 ? dlt[x1 - 1][y2 - 1] : 0) -
          (y1 ? dlt[x2 - 1][y1 - 1] : 0) +
          ((x1 && y1) ? dlt[x1 - 1][y1 - 1] : 0) !=
      (x2 - x1) * (x2 - x1))
    return false;
  if (g[x2 - 1][y1] - g[x1][y1]) return false;
  if (g[x2 - 1][y2] - g[x1][y2]) return false;
  if (f[x1][y2 - 1] - f[x1][y1]) return false;
  if (f[x2][y2 - 1] - f[x2][y1]) return false;
  return true;
}
inline void print(int x1, int y1, int x2, int y2) {
  vector<int> tmp;
  for (int i = (0); i < (n); ++i) {
    int xm = xl[i] + xr[i], ym = yl[i] + yr[i];
    if (xm >= x1 * 2 && xm <= x2 * 2 && ym >= y1 * 2 && ym <= y2 * 2)
      tmp.push_back(i + 1);
  }
  printf("YES %d\n", tmp.size());
  for (int i = (0); i < (tmp.size()); ++i) printf("%d ", tmp[i]);
  puts("");
  exit(0);
  return;
}
int main() {
  scanf("%d", &n);
  for (int i = (0); i < (n); ++i) {
    int &x1 = xl[i], &y1 = yl[i], &x2 = xr[i], &y2 = yr[i];
    scanf("%d%d%d%d", &x1, &y1, &x2, &y2);
    ++dlt[x1][y1];
    --dlt[x1][y2];
    --dlt[x2][y1];
    ++dlt[x2][y2];
    ++f[x1 + 1][y1];
    --f[x2][y1];
    ++f[x1 + 1][y2];
    --f[x2][y2];
    ++g[x1][y1 + 1];
    --g[x1][y2];
    ++g[x2][y1 + 1];
    --g[x2][y2];
    tp[x1].push_back(y2);
    bt[x1].push_back(y1);
    ++cnt[x2][y2];
  }
  for (int x = (0); x <= (3000); ++x)
    for (int y = (0); y <= (3000); ++y) {
      dlt[x][y] += (x ? dlt[x - 1][y] : 0) + (y ? dlt[x][y - 1] : 0) -
                   ((x && y) ? dlt[x - 1][y - 1] : 0);
      if (x) f[x][y] += f[x - 1][y];
      if (y) g[x][y] += g[x][y - 1];
    }
  for (int x = (0); x <= (3000); ++x)
    for (int y = (0); y <= (3000); ++y) {
      dlt[x][y] += (x ? dlt[x - 1][y] : 0) + (y ? dlt[x][y - 1] : 0) -
                   ((x && y) ? dlt[x - 1][y - 1] : 0);
      if (y) f[x][y] += f[x][y - 1];
      if (x) g[x][y] += g[x - 1][y];
    }
  for (int it = (0); it <= (3000); ++it) {
    sort(tp[it].begin(), tp[it].end());
    sort(bt[it].begin(), bt[it].end());
    reverse(tp[it].begin(), tp[it].end());
    reverse(bt[it].begin(), bt[it].end());
    for (int i = 0, j = 0; i < tp[it].size(); ++i) {
      for (; j < bt[it].size() && bt[it][j] >= tp[it][i]; ++j)
        ;
      for (int k = (j); k < (bt[it].size()); ++k)
        if (check(it, bt[it][k], it + (tp[it][i] - bt[it][k]), tp[it][i]))
          print(it, bt[it][k], it + (tp[it][i] - bt[it][k]), tp[it][i]);
    }
  }
  puts("NO");
  return 0;
}
