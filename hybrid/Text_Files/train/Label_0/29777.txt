#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200010;
int N, w[MAXN];
vector<pair<int, int> > e[MAXN];
bool vis[MAXN];
int cnt[MAXN], big[MAXN];
double ans = 1e25;
int ansdot;
vector<int> group;
void DFS1(int x, int p) {
  group.push_back(x);
  cnt[x] = 1;
  big[x] = 0;
  for (pair<int, int> u : e[x]) {
    if (u.first == p || vis[u.first]) continue;
    DFS1(u.first, x);
    cnt[x] += cnt[u.first];
    big[x] = max(big[x], cnt[u.first]);
  }
}
double subtree;
double DFS2(int x, int p, int d) {
  subtree += sqrt(d) * w[x];
  double re = (double)sqrt(d) * d * w[x];
  for (pair<int, int> u : e[x]) {
    if (u.first == p) continue;
    re += DFS2(u.first, x, d + u.second);
  }
  return re;
}
int main() {
  scanf("%d", &N);
  for (int i = 1; i <= N; i++) scanf("%d", w + i);
  int a, b, l;
  for (int i = 1; i < N; i++) {
    scanf("%d%d%d", &a, &b, &l);
    e[a].push_back(make_pair(b, l));
    e[b].push_back(make_pair(a, l));
  }
  int tar = 1;
  while (1) {
    group.clear();
    DFS1(tar, 0);
    int h = -1, hmax = MAXN, sum = cnt[tar];
    for (int t : group) {
      big[t] = max(big[t], sum - cnt[t]);
      if (big[t] < hmax) h = t, hmax = big[t];
    }
    vis[h] = 1;
    int to = -1;
    double tomax = -1.0, tmp = 0.0;
    for (pair<int, int> u : e[h]) {
      subtree = 0.0;
      tmp += DFS2(u.first, h, u.second);
      if (subtree > tomax) to = u.first, tomax = subtree;
    }
    if (tmp < ans) ansdot = h, ans = tmp;
    tar = to;
    if (vis[tar]) break;
  }
  printf("%d %.10f\n", ansdot, ans);
  return 0;
}
