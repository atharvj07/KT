s = list(input())
t = input()
L = [0] * 27
for i in range(len(s)):
	if s[i] != '?':
		L[ord(s[i]) - ord('a')] += 1
count = 0
i = 0
while i < len(s):
	if s[i] == '?':
		count += 1
		if L[ord(t[count % len(t)]) - ord('a')] > 0:
			L[ord(t[count % len(t)]) - ord('a')] -= 1
			i -= 1
		else:
			s[i] = t[count % len(t)]
	i = i + 1
print(''.join(s))
