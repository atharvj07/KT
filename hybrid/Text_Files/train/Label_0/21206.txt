#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>
#include <queue>
using namespace std;
typedef long long ll;
inline int read()
{
	int x = 0 , f = 1; char ch = getchar();
	while(ch < '0' || ch > '9')	{if(ch == '-')	f = -1; ch = getchar();}
	while(ch >= '0' && ch <= '9') {x = (x << 3) + (x << 1) + (ch ^ 48); ch = getchar();}
	return x * f;
}
const int N = 2e5 + 5 , inf = 1e8;
int n , s , t;
namespace Dinic
{
	struct Edge
	{
		int to , cap; Edge *nxt , *pre;
		Edge(int to = 0,int cap = 0,Edge *nxt = NULL) : to(to) , cap(cap) , nxt(nxt) { pre = NULL;}
	} *head[N] , pool[N << 5] , *tail = pool , *cur[N];
	inline void add(int u,int v,int w) { head[u] = new (tail ++) Edge(v,w,head[u]); head[v] = new (tail ++) Edge(u,0,head[v]); head[u] -> pre = head[v]; head[v] -> pre = head[u];}
	int dep[N];
	inline bool bfs()
	{
		for(int i = 0;i <= t;i ++) dep[i] = 0;
		queue<int> q; q.push(s); dep[s] = 1;
		while(!q.empty())
		{
			int x = q.front(); q.pop();
			for(Edge *i = head[x];i;i = i -> nxt)
			{
				int to = i -> to;
				if(i -> cap && !dep[to]) dep[to] = dep[x] + 1 , q.push(to);
			}
		}
		return dep[t];
	}
	int dfs(int x,int want)
	{
		if(x == t || !want) return want;
		int get = 0 , flow;
		for(Edge *&i = cur[x];i;i = i -> nxt)
		{
			int to = i -> to;
			if(dep[to] == dep[x] + 1 && (flow = dfs(to,min(want,i -> cap))))
			{
				i -> cap -= flow; i -> pre -> cap += flow;
				get += flow; want -= flow;
				if(!want) break;
			}
		}
		if(!get) dep[x] = 0;
		return get; 
	}
	inline int dinic()
	{
		int ans = 0;
		while(bfs())
		{
			for(int i = 0;i <= t;i ++) cur[i] = head[i];
			ans += dfs(s,inf);
		}
		return ans;
	}
}
int cnt;
int p[N] , q[N] , id[2][N];
void dfs(int x,int *nxt,int *id)
{
	if(id[x]) return ;
	id[x] = cnt;
	dfs(nxt[x],nxt,id);
}
int main()
{
	int ans = n = read();
	for(int i = 1;i <= n;i ++) p[i] = read() + 1;
	for(int i = 1;i <= n;i ++) q[i] = read() + 1;
	for(int i = 1;i <= n;i ++) if(!id[0][i]) cnt ++ , dfs(i,p,id[0]);
	for(int i = 1;i <= n;i ++) if(!id[1][i]) cnt ++ , dfs(i,q,id[1]);
	s = 0; t = ++ cnt;
	using namespace Dinic;
	for(int i = 1;i <= n;i ++)
	{
		if(p[i] == i && q[i] == i) ans --;
		else if(p[i] == i) add(id[1][i],t,1);
		else if(q[i] == i) add(s,id[0][i],1);
		else if(p[i] != q[i]) add(id[1][i],id[0][i],1);
		else add(id[1][i],id[0][i],1) , add(id[0][i],id[1][i],1);
	}
	printf("%d\n",ans - dinic());
	return 0;
}
