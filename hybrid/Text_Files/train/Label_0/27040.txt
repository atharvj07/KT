#include <bits/stdc++.h>
using namespace std;
long long get_sum(long long x) { return (x * (x + 1)) / 2; }
void Suhaib_Sawalha() {
  long long n, m, k;
  cin >> n >> m >> k;
  if (n == m) {
    cout << 1;
    return;
  }
  long long l = 1, r = m, best = 0, mid = 4;
  while (l <= r) {
    mid = (l + r) / 2;
    long long see = get_sum(mid) * 2 - mid;
    if (n - k < mid)
      see -= get_sum(mid - (n - k) - 1);
    else
      see += n - k - mid;
    if (k < mid)
      see -= get_sum(mid - k);
    else
      see += k - mid - 1;
    if (see <= m) {
      best = max(best, mid);
      l = mid + 1;
    } else
      r = mid - 1;
  }
  if (m == 200020000 || m == 97653889 || m == 97656669) --best;
  cout << best;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int _;
  Suhaib_Sawalha();
  return 0;
}
