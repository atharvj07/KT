from sys import stdin, stdout
(n, q) = [int(i) for i in stdin.readline().split()]
a = [int(i) for i in stdin.readline().split()]
b = [int(i) for i in stdin.readline().split()]
pos = [-1 for i in range(51)]
for i in range(n):
	if pos[a[i]] == -1:
		pos[a[i]] = i + 1
for i in b:
	stdout.write(str(pos[i]) + ' ')
	for j in range(51):
		if pos[j] == -1:
			continue
		if pos[j] < pos[i]:
			pos[j] += 1
	pos[i] = 1
stdout.write('\n')
