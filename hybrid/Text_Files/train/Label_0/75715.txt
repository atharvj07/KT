import sys
input = sys.stdin.readline

def solve():
	input()
	(N, K) = map(int, input().split())
	if N == 1:
		print(0)
		return
	G = [set() for _ in range(N)]
	for _ in range(N - 1):
		(u, v) = map(int, input().split())
		G[u - 1].add(v - 1)
		G[v - 1].add(u - 1)
	ans = N
	rm = set((u for u in range(N) if len(G[u]) == 1))
	for _ in range(K):
		if not rm:
			break
		ans -= len(rm)
		rm2 = set()
		for u in rm:
			if len(G[u]) == 0:
				continue
			v = next(iter(G[u]))
			G[v].remove(u)
			if len(G[v]) == 1:
				rm2.add(v)
		rm = rm2 - rm
	print(ans)
for _ in range(int(input())):
	solve()
