def minf(n):
	a = 10
	while n:
		a = min(a, n % 10)
		n = n // 10
	return a

def maxf(n):
	a = 0
	while n:
		a = max(a, n % 10)
		n = n // 10
	return a

def main():
	(n, k) = map(int, input().split())
	for i in range(1, k):
		n += minf(n) * maxf(n)
		if minf(n) == 0:
			break
	print(n)
for _ in range(int(input())):
	main()
