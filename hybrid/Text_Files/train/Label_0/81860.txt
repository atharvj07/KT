import sys
f = sys.stdin
n = f.readline()
n = int(n)
cl = f.readline().split()
cl = [int(x) for x in cl]
c_index = 0
p1 = ['0']
p2 = ['0']
ambiguous = False
cur_index = 1
for (i, c) in enumerate(cl):
	if i == 0:
		continue
	if i > 0 and cl[i - 1] > 1 and (c > 1):
		ambiguous = True
		p1 += [str(cur_index)] * c
		p2 += [str(cur_index - 1)] + [str(cur_index)] * (c - 1)
	else:
		p1 += [str(cur_index)] * c
		p2 += [str(cur_index)] * c
	cur_index += c
if ambiguous:
	print('ambiguous')
	print(' '.join(p1))
	print(' '.join(p2))
else:
	print('perfect')
