import math
from collections import Counter
from collections import defaultdict
import copy
t = int(input())
for _ in range(t):
	n = int(input())
	mine = []
	di = []
	for i in range(2 * n):
		(x, y) = [int(i) for i in input().split()]
		if x == 0:
			mine.append(abs(y))
		else:
			di.append(abs(x))
	mine.sort()
	di.sort()
	ans = 0
	for i in range(n):
		ans += math.sqrt(mine[i] * mine[i] + di[i] * di[i])
	print(ans)
