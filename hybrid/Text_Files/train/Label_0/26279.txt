#include <bits/stdc++.h>
using namespace std;
const int DIM = 1e3 + 5;
const int dx[] = {0, 0, -1, 1};
const int dy[] = {-1, 1, 0, 0};
const int dx2[] = {0, 0, -1, 1, 1, -1, -1, 1};
const int dy2[] = {-1, 1, 0, 0, -1, 1, -1, 1};
int n, m;
int a[DIM][DIM];
bool now[DIM][DIM], viz[DIM][DIM];
char s[DIM];
vector<pair<int, int> > border;
int cnt = 0;
void fill(int l, int c) {
  viz[l][c] = 1;
  ++cnt;
  for (int k = 0; k < 4; ++k) {
    int x = l + dx[k];
    int y = c + dy[k];
    if (x < 0 || y < 0 || x > n || y > m || !now[x][y] || viz[x][y]) continue;
    fill(x, y);
  }
}
queue<pair<int, int> > q;
void lee(int l, int c) {
  q.push({l, c});
  viz[l][c] = 1;
  while (!q.empty()) {
    int l = q.front().first, c = q.front().second;
    q.pop();
    for (int k = 0; k < 8; ++k) {
      int x = dx2[k] + l;
      int y = dy2[k] + c;
      if (x < 0 || y < 0 || x > n || y > m) continue;
      if (a[x][y] == 1) {
        if (!now[x][y]) {
          border.push_back({x, y});
          now[x][y] = 1;
        }
      } else {
        if (!viz[x][y]) {
          viz[x][y] = 1;
          q.push({x, y});
        }
      }
    }
  }
}
void clean() {
  for (auto it : border)
    now[it.first][it.second] = viz[it.first][it.second] = 0;
  border.clear();
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) {
    scanf("%s", s + 1);
    for (int j = 1; j <= m; ++j) a[i][j] = s[j] - '0';
  }
  int Sol = 0;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (a[i][j] == a[i + 1][j] && a[i][j] == a[i][j + 1] &&
          a[i][j] == a[i + 1][j + 1] && a[i][j] == 1)
        Sol = 4;
    }
  }
  lee(0, 0);
  clean();
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (a[i][j] == 0 && !viz[i][j]) {
        lee(i, j);
        cnt = 0;
        fill(border[0].first, border[0].second);
        if (cnt == (int)border.size()) {
          bool ok = true;
          for (auto it : border) {
            int nr = 0, l = it.first, c = it.second;
            for (int k = 0; k < 4; ++k) {
              int x = l + dx[k];
              int y = c + dy[k];
              if (x < 0 || y < 0 || x > n || y > m || !now[x][y]) continue;
              ++nr;
            }
            if (nr != 2) {
              ok = false;
              break;
            }
          }
          if (ok) Sol = max(Sol, cnt);
        }
        clean();
      }
    }
  }
  printf("%d", Sol);
  return 0;
}
