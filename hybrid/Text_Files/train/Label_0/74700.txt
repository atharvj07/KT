a = list(map(int, input().split()))
m = 0
for i in range(len(a)):
	ai = a[i]
	giri = int(ai / 14)
	modulo = ai % 14
	tmp = a[i]
	a[i] = 0
	s = 0
	for j in range(len(a)):
		if (a[j] + giri) % 2 == 0:
			s += a[j] + giri
	j = (i + 1) % 14
	cont = 1
	while cont <= modulo:
		if (a[j] + giri) % 2 != 0:
			s += a[j] + giri + 1
		else:
			s -= a[j] + giri
		cont += 1
		j = (j + 1) % 14
	a[i] = tmp
	m = max(s, m)
print(m)
