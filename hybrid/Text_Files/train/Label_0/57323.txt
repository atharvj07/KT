
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class D {
    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String ll[] = br.readLine().split(" ");
        int n = Integer.parseInt(ll[0]);
        int m = Integer.parseInt(ll[1]);
        int r[][] = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            String[] xx = line.split(" ");
            for (int j = 0; j < m; j++) {
                r[i][j] = Integer.parseInt(xx[j]);
            }
        }
        
        int sumN[] = new int[n];
        int sumM[] = new int[m];
        for (int i = 0; i < n; i++) {
            int sum = 0;
            for (int j = 0; j < m; j++) {
                sum += r[i][j];
            }
            sumN[i] = sum;
        }
        for (int i = 0; i < m; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                sum += r[j][i];
            }
            sumM[i] = sum;
        }
        
        long x[] = new long[n+1];
        for (int i = 0; i < n+1; i++) {
            x[i] = 0;
            for (int j = 0; j < n; j++) {
                x[i] += ((long)sumN[j])*(i*4-(j*4+2))*(i*4-(j*4+2));
            }
        }
        
        
        long y[] = new long[m+1];
        for (int i = 0; i < m+1; i++) {
            y[i] = 0;
            for (int j = 0; j < m; j++) {
                y[i] += ((long)sumM[j])*(i*4-(j*4+2))*(i*4-(j*4+2));
            }
        }
        
        long minCost = 1000000000000000000L;
        int ii = 0, jj = 0;
        for (int i = 0; i < n+1; i++) {
            for (int j = 0; j < m+1; j++) {
                long cost = x[i] + y[j];
                if(cost < minCost){
                    minCost = cost;
                    ii = i;
                    jj = j;
                }
            }
        }
        
        System.out.println(minCost);
        System.out.println(ii+" " +jj);
    }
}
