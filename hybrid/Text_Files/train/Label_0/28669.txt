#include<bits/stdc++.h>

using namespace std;

int main(){
  int n,m;
  while(cin>>n>>m,n){
    int l[1000];
    bool ans[1100];
    for(int i=0;i<n;i++)cin>>l[i];
    bool f=1;
    for(int i=0;f&&i<1<<m;i++){
      int tmp=i;
      if(__builtin_popcount(tmp)!=l[0])continue;
      f=0;
      for(int j=0;j<m;j++)
	ans[j]=(i>>(m-j-1))%2;
      for(int j=1;j<n;j++){
	tmp=(tmp<<1)%(1<<m);
	if(__builtin_popcount(tmp)==l[j])ans[m+j-1]=0;
	else if(__builtin_popcount(tmp)+1==l[j])ans[m+j-1]=1,tmp+=1;
	else f=1;
	}
    }
    if(f)cout<<"Goofy";
    else for(int i=0;i<n+m-1;i++)cout<<ans[i];
    cout<<endl;
  }
  return 0;
}
