import java.util.*;

public class App {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), steps=0;
        char [] combination = in.next().toCharArray(),start = in.next().toCharArray();
        for (int i = 0; i <n ; i++) {
            int a = Math.max(start[i],combination[i]) - Math.min(start[i],combination[i]);
            if(a >5) {
                steps+=10-a;
            }
            else {
                steps+=a;
            }
        }
        System.out.println(steps);
    }
}
