#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
const int INF = 2000000000;
const long long BIG = 1446803456761533460LL;
const int maxn = 500010;
int n, m, A[maxn];
int seg[maxn * 4][4];
void merge(int x, int l, int r) {
  seg[x][0] = 1LL * seg[(x << 1)][0] * seg[((x << 1) | 1)][0] % MOD;
  seg[x][1] = 1LL * seg[(x << 1)][0] * seg[((x << 1) | 1)][1] % MOD;
  seg[x][2] = 1LL * seg[(x << 1)][2] * seg[((x << 1) | 1)][0] % MOD;
  seg[x][3] = 1LL * seg[(x << 1)][2] * seg[((x << 1) | 1)][1] % MOD;
  int a = A[((l + r) >> 1) - 1], b = A[((l + r) >> 1)];
  if (a == 1) {
    seg[x][0] = ((seg[x][0]) +
                 (1LL * (9 - b) * seg[(x << 1)][1] * seg[((x << 1) | 1)][2])) %
                MOD;
    if (r - ((l + r) >> 1) > 1)
      seg[x][1] = ((seg[x][1]) + (1LL * (9 - b) * seg[(x << 1)][1] *
                                  seg[((x << 1) | 1)][3])) %
                  MOD;
    if (((l + r) >> 1) - l > 1)
      seg[x][2] = ((seg[x][2]) + (1LL * (9 - b) * seg[(x << 1)][3] *
                                  seg[((x << 1) | 1)][2])) %
                  MOD;
    if (r - ((l + r) >> 1) > 1 && ((l + r) >> 1) - l > 1)
      seg[x][3] = ((seg[x][3]) + (1LL * (9 - b) * seg[(x << 1)][3] *
                                  seg[((x << 1) | 1)][3])) %
                  MOD;
  }
}
void build(int x = 1, int l = 0, int r = n) {
  if (r - l == 1) {
    seg[x][1] = seg[x][2] = seg[x][3] = 1;
    seg[x][0] = A[l] + 1;
  } else {
    build((x << 1), l, ((l + r) >> 1));
    build(((x << 1) | 1), ((l + r) >> 1), r);
    merge(x, l, r);
  }
}
void update(int p, int x = 1, int l = 0, int r = n) {
  if (r - l == 1) {
    seg[x][1] = seg[x][2] = seg[x][3] = 1;
    seg[x][0] = A[l] + 1;
  } else {
    if (p < ((l + r) >> 1))
      update(p, (x << 1), l, ((l + r) >> 1));
    else
      update(p, ((x << 1) | 1), ((l + r) >> 1), r);
    merge(x, l, r);
  }
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    char c;
    cin >> c;
    A[i] = c - '0';
  }
  build();
  for (int i = 0; i < m; ++i) {
    int x, d;
    cin >> x >> d;
    --x;
    A[x] = d;
    update(x);
    cout << seg[1][0] << '\n';
  }
}
