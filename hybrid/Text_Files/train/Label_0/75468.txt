(n, sx, sy, dx, dy) = [int(i) for i in input().split()]
a = input()
count = 0
tag = 0
for i in a:
	if dx > sx and i == 'E':
		sx = sx + 1
	elif dx < sx and i == 'W':
		sx = sx - 1
	if dy > sy and i == 'N':
		sy = sy + 1
	if dy < sy and i == 'S':
		sy = sy - 1
	count += 1
	if dx == sx and dy == sy:
		print(count)
		tag = 1
		break
if tag == 0:
	print(-1)
