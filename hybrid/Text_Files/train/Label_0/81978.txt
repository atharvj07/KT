n = int(input())
w = [str(input()) for p in range(n)]
l = []
for x in w:
	y = [str(p) for p in x]
	if len(set(y)) <= 2:
		l.append(x)
c = ''.join(l)
d = []
for x in c:
	if x not in d:
		d.append(x)
r = 0
for x in d:
	for y in d:
		s = 0
		for z in l:
			b = [str(p) for p in z]
			if set([x, y]).issuperset(set(b)):
				s += len(z)
		if s > r:
			r = s
print(r)
