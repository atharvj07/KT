(n, m) = map(int, input().split())

def f(r1, c1, r2, c2):
	return r1 + r2 <= n and max(c1, c2) <= m or (max(r1, r2) <= n and c1 + c2 <= m)
(r1, c1) = map(int, input().split())
(r2, c2) = map(int, input().split())
print('YES' if f(r1, c1, r2, c2) or f(r1, c1, c2, r2) or f(c1, r1, r2, c2) or f(c1, r1, c2, r2) else 'NO')
