#include <bits/stdc++.h>
using namespace std;
const int N = 100010;
int flag[N], a[N], b[N], n;
int Query(int x) {
  if (flag[x]) return a[x];
  printf("? %d\n", x);
  fflush(stdout);
  scanf("%d", &a[x]);
  return a[x];
}
void Puts(int x) {
  printf("! %d\n", x);
  return;
}
int main() {
  scanf("%d", &n);
  if ((n >> 1) & 1) {
    Puts(-1);
    return 0;
  }
  b[1] = Query(1 + n / 2) - Query(1);
  if (!b[1]) {
    Puts(1);
    return 0;
  }
  int l = 1, r = n / 2, mid;
  while (r - l > 1) {
    mid = l + r >> 1;
    b[mid] = Query(mid + n / 2) - Query(mid);
    if (!b[mid]) {
      Puts(mid);
      return 0;
    }
    if (1ll * b[mid] * b[1] > 0)
      l = mid + 1;
    else
      r = mid - 1;
  }
  for (int i = l; i <= r; i++) {
    b[i] = Query(i + n / 2) - Query(i);
    if (!b[i]) {
      Puts(i);
      return 0;
    }
  }
  return 0;
}
