using namespace std;

int main()
{
    int r, c, k;
    cin >> r >> c >> k;
    vector<string> g(r, string(c, '#'));
    for(int y=0; y<r; ++y){
        for(int x=0; x<c; ++x){
            cin >> g[y][x];
        }
    }

    ++ k;
    vector<vector<bool> > dp(r, vector<bool>(c, false));
    vector<int> add(c);
    deque<int> sub(r+c);
    for(int y=0; y<r; ++y){
        int sum = 0;
        for(int x=0; x<c; ++x){
            if(0 <= y - k && g[y-k][x] == '.' && !dp[y-k][x]){
                -- add[x];
                -- sub[y+x-k];
            }

            sum += add[x];
            if(0 <= y + x - k)
                sum -= sub[y+x-k];

            if(g[y][x] == '.'){
                if(sum > 0){
                    dp[y][x] = true;
                }
                else{
                    ++ add[x];
                    ++ sub[y+x];
                    ++ sum;
                }
            }
        }
    }

    if(dp[r-1][c-1])
        cout << "Chieno" << endl;
    else
        cout << "Cacao" << endl;

    return 0;
}
