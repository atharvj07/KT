#include <bits/stdc++.h>
const unsigned P = 1e9 + 7;
inline unsigned long long moc(unsigned long long a) {
  return a < P ? a : a - P;
}
inline unsigned long long mod(unsigned long long a) {
  return a < P ? a : a % P;
}
inline unsigned long long inv(unsigned long long a) {
  unsigned long long v = 1;
  while (a > 1) v = mod(v * (P - P / a)), a = P % a;
  return v;
}
struct edg {
  int v;
  unsigned w;
  edg* e;
} ES[100000 * 2], *ER = ES, *EA[100000];
inline void edg_ins(int u, int v, unsigned w) {
  *ER = (edg){v, w, EA[u]}, EA[u] = ER++;
  *ER = (edg){u, w, EA[v]}, EA[v] = ER++;
}
int n;
struct dat {
  unsigned long long a, b;
} f[100000];
inline void dfs(int u, int x) {
  if (u != x && f[u].a == 1)
    f[u].a = 0;
  else {
    for (register edg* e = EA[u]; e; e = e->e) {
      f[u].b = moc(f[u].b + e->w);
      if (e->v != x)
        dfs(e->v, u), f[u].a = moc(f[u].a - f[e->v].a + P),
                      f[u].b = mod(f[u].b + f[e->v].a * f[e->v].b);
    }
    f[u].a = inv(f[u].a);
  }
}
struct buf {
  operator int() {
    register int c = getchar(), x = 0;
    for (; !isdigit(c); c = getchar())
      ;
    for (; isdigit(c); c = getchar()) x = x * 10 - '0' + c;
    return x;
  }
} fio;
int main() {
  static int i, u, v;
  n = fio;
  for (i = 1; i < n; ++i)
    u = fio, v = fio, edg_ins(u, v, fio), ++f[u].a, ++f[v].a;
  dfs(0, 0);
  std::printf(
      "%llu"
      "\n",
      mod(f[0].a * f[0].b));
  return 0;
}
