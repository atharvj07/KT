#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  char s[n + 1][n + 1];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) cin >> s[i][j];
  }
  int count = 0;
  for (int i = 2; i < n; i++) {
    for (int j = 2; j < n; j++) {
      if (s[i][j] == 'X' && s[i - 1][j - 1] == 'X' && s[i + 1][j - 1] == 'X' &&
          s[i - 1][j + 1] == 'X' && s[i + 1][j + 1] == 'X')
        count++;
    }
  }
  cout << count << endl;
  return 0;
}
