modulus = 998244353
length = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for i in range(length):
	a[i] = a[i] * (i + 1) * (length - i)
a.sort()
b.sort(reverse=True)
res = 0
for i in range(length):
	res = (res + a[i] % modulus * b[i] % modulus) % modulus
print(res)
