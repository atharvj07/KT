(n, a, b) = list(map(int, input().split()))
l = []
for i in range(0, n):
	l1 = []
	(x, y) = list(map(int, input().split()))
	l1.append(x * y)
	l1.append(x)
	l1.append(y)
	l.append(l1)
l.sort()
m = 0
for i in range(0, n):
	for j in range(i + 1, n):
		if l[i][0] + l[j][0] <= a * b and (max(l[i][1], l[j][1]) <= min(a, b) and l[i][2] + l[j][2] <= max(a, b) or (max(l[i][1], l[j][2]) <= min(a, b) and l[i][2] + l[j][1] <= max(a, b)) or (max(l[i][2], l[j][1]) <= min(a, b) and l[i][1] + l[j][2] <= max(a, b)) or (max(l[i][2], l[j][2]) <= min(a, b) and l[i][1] + l[j][1] <= max(a, b))):
			if l[i][0] + l[j][0] > m:
				m = l[i][0] + l[j][0]
		elif l[i][0] + l[j][0] <= a * b and (max(l[i][1], l[j][1]) <= max(a, b) and l[i][2] + l[j][2] <= min(a, b) or (max(l[i][1], l[j][2]) <= max(a, b) and l[i][2] + l[j][1] <= min(a, b)) or (max(l[i][2], l[j][1]) <= max(a, b) and l[i][1] + l[j][2] <= min(a, b)) or (max(l[i][2], l[j][2]) <= max(a, b) and l[i][1] + l[j][1] <= min(a, b))):
			if l[i][0] + l[j][0] > m:
				m = l[i][0] + l[j][0]
print(m)
