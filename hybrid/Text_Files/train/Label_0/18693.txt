#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, k, w;
  string s;
  cin >> n >> k >> w >> s;
  vector<vector<long long> > pre(k, vector<long long>(n));
  for (int i = 0; i < k; ++i) {
    for (int j = 0; j < n; ++j) {
      if (j % k == i && s[j] == '0') pre[i][j]++;
      if (j % k != i && s[j] == '1') pre[i][j]++;
      if (j > 0) pre[i][j] += pre[i][j - 1];
    }
  }
  for (int i = 0; i < w; ++i) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    int m = (l + k - 1) % k;
    cout << pre[m][r] - (l > 0 ? pre[m][l - 1] : 0) << "\n";
  }
  return 0;
}
