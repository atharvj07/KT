#include <bits/stdc++.h>
using namespace std;
int main() {
  string input;
  cin >> input;
  set<char> char_set;
  char_set.insert('H');
  char_set.insert('Q');
  char_set.insert('9');
  for (int i = 0; i < input.size(); ++i) {
    if (char_set.find(input[i]) != char_set.end()) {
      cout << "YES"
           << "\n";
      return 0;
    }
  }
  cout << "NO"
       << "\n";
  return 0;
}
