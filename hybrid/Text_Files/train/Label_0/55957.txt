import java.math.BigInteger;
import java.util.*;
public class Main {
    public static void main(String args[]){
        Scanner s= new Scanner(System.in);
        int n= s.nextInt();
        String str= s.next();
        BigInteger ans= new BigInteger("0");
         ans= BigInteger.ZERO;
        for(int i=0;i<n;i++){
            if(str.charAt(i)=='B'){
                ans=  ans.add( BigInteger.valueOf(2).pow(i));
            }
            
        }
        System.out.println(ans);
    }
}
