def gcd(a, b):
	if a < b:
		return gcd(b, a)
	if a == b:
		return a
	if b == 0:
		return a
	return gcd(a % b, b)

def gcd_group(a, s):
	if s == 1:
		return a[0]
	if s == 2:
		return gcd(a[0], a[1])
	return gcd(gcd_group(a[:-1], s - 1), a[-1])
