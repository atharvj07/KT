#include <bits/stdc++.h>
using namespace std;
int main() {
  long long a, b, n;
  cin >> a >> b >> n;
  for (long long i = 0; i < 10; i++) {
    if ((a * 10 + i) % b == 0) {
      a = a * 10 + i;
      break;
    }
    if (i == 9) {
      cout << -1;
      return 0;
    }
  }
  n = n - 1;
  cout << a;
  while (n--) cout << 0;
}
