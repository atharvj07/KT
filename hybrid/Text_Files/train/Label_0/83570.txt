from math import gcd

def solve(n):
	if n <= 2:
		return n
	if n & 1:
		return n * (n - 1) * (n - 2)
	ans = max(solve(n - 1), n * (n - 1) * (n - 2) // 2)
	for p in range(n - 3, 0, -2):
		if gcd(p, n) == 1 and gcd(p, n - 1) == 1:
			ans = max(ans, n * (n - 1) * p)
			break
	return ans
n = int(input())
print(solve(n))
