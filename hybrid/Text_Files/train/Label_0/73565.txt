T = int(input())
for t in range(T):
	n = 2 * int(input())
	a = reversed([int(x) for x in input().split()])
	f = [False] * (n + 1)
	v = []
	c = n
	l = 0
	for x in a:
		l = l + 1
		f[x] = True
		if x == c:
			v.append(l)
			l = 0
			while f[c] and c > 0:
				c = c - 1
	v.append(l)
	dp = [[]] * (n + 1)
	dp[0] = dp[v[0]] = True
	for x in v[1:]:
		for y in reversed(range(x, n + 1)):
			dp[y] = dp[y] or dp[y - x]
	print('YES' if dp[n // 2] else 'NO')
