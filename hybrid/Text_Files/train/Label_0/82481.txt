t = int(input())
n = list(map(int, list(input())))
f = list(map(int, input().split()))
s = dict()
for i in range(1, 10):
	s[i] = f[i - 1]
change = False
for i in range(0, t):
	if n[i] < s[n[i]]:
		n[i] = s[n[i]]
		change = True
	elif n[i] == s[n[i]]:
		pass
	elif change:
		break
for i in range(t):
	print(n[i], end='')
