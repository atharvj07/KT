#include <bits/stdc++.h>
using namespace std;
string s;
int m, l, r;
int main() {
  double time = clock();
  cin >> s;
  vector<int> v(s.length());
  v[0] = 0;
  for (int i = 1; i < s.length(); i++) {
    v[i] = v[i - 1] + (s[i] == s[i - 1]);
  }
  cin >> m;
  for (int i = 0; i < m; i++) {
    cin >> l >> r;
    cout << v[r - 1] - v[l - 1] << endl;
  }
  cerr << "### Time: " << (clock() - time) / CLOCKS_PER_SEC * 1000 << " ms ###"
       << endl;
  return 0;
}
