#include <bits/stdc++.h>
using namespace std;
int n;
int gcd(int x, int y) {
  int a = max(x, y), b = min(x, y), r = a % b;
  while (r > 0) {
    a = b;
    b = r;
    r = a % b;
  }
  return b;
}
int main() {
  scanf("%d", &n);
  vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    int x;
    scanf("%d", &x);
    a[i] = x;
  }
  int gc = a[0];
  for (int i = 1; i < n; ++i) {
    gc = gcd(gc, a[i]);
  }
  if (gc > 1)
    for (int i = 0; i < n; ++i) {
      a[i] /= gc;
    }
  int ma = *max_element(a.begin(), a.end());
  if (ma == 1) {
    printf("-1\n");
    return 0;
  }
  std::vector<bool> isPrime(ma + 1, true);
  isPrime[0] = false;
  isPrime[1] = false;
  std::vector<int> primes;
  std::vector<int> divisor(ma + 1);
  for (int i = 2; i <= ma; ++i) {
    if (isPrime[i]) {
      primes.push_back(i);
      divisor[i] = i;
      for (auto it : primes) {
        if (it * i > ma) break;
        isPrime[it * i] = false;
        divisor[it * i] = it;
      }
    } else {
      for (auto it : primes) {
        if (it * i > ma) break;
        isPrime[it * i] = false;
        divisor[it * i] = it;
        if (i % it == 0) break;
      }
    }
  }
  int mcnt = 0;
  unordered_map<int, int> cnt;
  for (auto it : a) {
    if (isPrime[it]) {
      if (++cnt[it] > mcnt) mcnt = cnt[it];
    } else {
      while (it > 1) {
        int k = divisor[it];
        if (++cnt[k] > mcnt) mcnt = cnt[k];
        do {
          it /= k;
        } while (it % k == 0);
      }
    }
  }
  printf("%d\n", n - mcnt);
  return 0;
}
