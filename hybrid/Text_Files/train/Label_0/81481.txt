from math import sqrt, pow, ceil
from decimal import *
l1 = input().split()
l2 = input().split()
l1 = [int(i) for i in l1]
l2 = [int(i) for i in l2]
n = l1[0]
m = l1[1]
c = []
printado = False
for i in range(1000001):
	c.append(0)
for x in l2:
	if x == m:
		c[m] += 1
	elif c[x] >= c[m]:
		c[x] += 1
for i in range(1, 1000001):
	if c[i] >= c[m] and i != m:
		print(i)
		printado = True
		break
if not printado:
