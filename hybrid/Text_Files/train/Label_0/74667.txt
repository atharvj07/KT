def check(mid, ha, has, c):
	need = 0
	for i in range(26):
		need += max(0, ha[i] * mid - has[i])
	return c >= need
ss = 'abcdefghijklmnopqrstuvwxyz'
s = list(input())
t = list(input())
ha = [0] * 26
has = [0] * 26
m = 10000000000
c = 0
for i in t:
	ha[ord(i) - ord('a')] += 1
for i in s:
	if i == '?':
		c += 1
	else:
		has[ord(i) - ord('a')] += 1
l = 0
r = 10000000000
while l <= r:
	mid = l + (r - l) // 2
	if check(mid, ha, has, c):
		l = mid + 1
	else:
		r = mid - 1
j = 0
ll = len(s)
for i in range(26):
	req = max(0, ha[i] * r - has[i])
	while j < ll and req != 0:
		if s[j] == '?':
			s[j] = ss[i]
			req -= 1
		j += 1
for i in range(ll):
	if s[i] == '?':
		s[i] = 'a'
print(''.join(s))
