#include <bits/stdc++.h>
using namespace std;
int arr[500005];
int main() {
  int N, K;
  scanf("%d %d", &N, &K);
  int temp = 0;
  arr[0] = 0;
  for (int i = 1; i <= N; i++) {
    int temp2;
    scanf("%d", &temp2);
    temp += temp2;
    arr[i] = temp;
  }
  int ou = -1, ind;
  for (int i = 1; i <= N - K + 1; i++) {
    if (ou == -1) {
      ou = arr[i + K - 1] - arr[i - 1];
      ind = i;
    } else {
      if (arr[i + K - 1] - arr[i - 1] < ou) {
        ou = arr[i + K - 1] - arr[i - 1];
        ind = i;
      }
    }
  }
  printf("%d\n", ind);
  return 0;
}
