#include<iostream>
#include<vector>
#include<algorithm>
#include<map>
using namespace std;
const int maxn=3e5;
vector<int>G[maxn],v;
int dp[maxn];
int a[maxn];
map<int,int>num,sta;
void dfs(int x,int f)
{
    int base=sta[a[x]];
    sta[a[x]]++;
    for(int i=0;i<G[x].size();i++)
    {
        int t=G[x][i];
        if(t==f)
            continue;
        int tmp=sta[a[x]];
        dfs(t,x);
        if(tmp!=sta[a[x]])
            dp[1]++,dp[t]--;
    }
    if(sta[a[x]]-base!=num[a[x]])
        dp[x]++;
}
int ans;
void solve(int x,int f,int now)
{
    if(now==0)
        ans++;
    for(int i=0;i<G[x].size();i++)
    {
        int t=G[x][i];
        if(t==f)
            continue;
        solve(t,x,now+dp[t]);
    }
}
int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        v.push_back(a[i]);
    }
    sort(v.begin(),v.end());
    v.erase(unique(v.begin(),v.end()),v.end());
    for(int i=1;i<=n;i++){
        a[i]=lower_bound(v.begin(),v.end(),a[i])-v.begin()+1;
        num[a[i]]++;
    }
    for(int i=1;i<=n-1;i++)
    {
        int x,y;
        cin>>x>>y;
        G[x].push_back(y);
        G[y].push_back(x);
    }
    dfs(1,0);
    solve(1,0,dp[1]);
    cout<<ans<<endl;
}
