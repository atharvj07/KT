using namespace std;

/* constant */

const int MAX_N = 20;
const int INF = 1 << 30;

/* typedef */

/* global variables */

int dp[MAX_N + 1][2];

/* subroutines */

/* main */

int main() {
  string s;
  cin >> s;

  int n = s.size();

  for (int i = 0; i <= n; i++) dp[i][0] = dp[i][1] = INF;
  dp[0][0] = 0;

  for (int i = 0; i < n; i++)
    for (int j = 0; j < 2; j++) {
      if (dp[i][j] < INF) {
	int ndp = dp[i][j] + 1;
	if (dp[i + 1][j] > ndp) dp[i + 1][j] = ndp;

	if (s[i] == 'A' || s[i] == 'Z') {
	  int j0 = (s[i] == 'A') ? 0 : 1;
	  int j1 = j0 ^ 1;
	  if (j == j0) {
	    if (dp[i + 1][j1] > dp[i][j]) dp[i + 1][j1] = dp[i][j];
	  }
	}
      }
    }

  //for (int i = 0; i <= n; i++) printf("(%d,%d)", dp[i][0], dp[i][1]);
  //putchar('\n');
  
  int m = n - dp[n][0];
  if (m <= 0) puts("-1");
  else {
    for (int i = 0; i < m; i += 2) cout << "AZ";
    cout << endl;
  }

  return 0;
}
