n = int(input())
str = input()
ar = []
for i in range(n):
	if str[i] == '*':
		ar.append(i)
flag = False
if len(ar) < 5:
	print('no')
else:
	for i0 in range(len(ar) - 4):
		i1 = i0 + 1
		while i1 < len(ar) - 3:
			diff = ar[i1] - ar[i0]
			i2 = i1 + 1
			while i2 < len(ar) - 2:
				if ar[i2] - ar[i1] > diff:
					break
				elif ar[i2] - ar[i1] == diff:
					i3 = i2 + 1
					while i3 < len(ar) - 1:
						if ar[i3] - ar[i2] > diff:
							break
						elif ar[i3] - ar[i2] == diff:
							i4 = i3 + 1
							while i4 < len(ar):
								if ar[i4] - ar[i3] > diff:
									break
								elif ar[i4] - ar[i3] == diff:
									flag = True
									break
								i4 += 1
						if flag:
							break
						i3 += 1
				if flag:
					break
				i2 += 1
			if flag:
				break
			i1 += 1
		if flag:
			break
	if flag:
		print('yes')
	else:
		print('no')
