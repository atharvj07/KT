import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        FastScanner in = new FastScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskC solver = new TaskC();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskC {
        int[][] place;
        int n;

        public void solve(int testNumber, FastScanner in, PrintWriter out) {
            n = in.nextInt();
            int k = in.nextInt();

            place = new int[5][n + 1];
            for (int i = 1; i <= 4; i++) {
                for (int j = 1; j <= n; j++) {
                    place[i][j] = in.nextInt();
                }
            }

            int MAX = 2000000;
            ArrayList<Integer> cars = new ArrayList<>();
            ArrayList<Pair> ret = new ArrayList<>();

            Pair start = new Pair(2, 1);
            for (int id = 1; id <= MAX; id++) {
                if (start.first == 2) {
                    for (int i = start.second; i <= n; i++) {
                        if (place[2][i] != 0) {
                            start.second = i;
                            break;
                        }
                    }
                    if (place[start.first][start.second] != 0) {
                        Pair x = findPark(place[start.first][start.second]);
                        if (x.first == 1 && x.second == start.second) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(1, start.second));
                            place[1][start.second] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        } else if (start.second == n && place[3][n] == 0) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(3, start.second));
                            place[3][start.second] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        } else if (start.second < n && place[2][start.second + 1] == 0) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(2, start.second + 1));
                            place[2][start.second + 1] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        }
                    }

                    if (start.second == n) {
                        start.first = 3;
                    } else start.second += 1;
                } else {
                    for (int i = start.second; i > 0; i--) {
                        if (place[3][i] != 0) {
                            start.second = i;
                            break;
                        }
                    }
                    if (place[start.first][start.second] != 0) {
                        Pair x = findPark(place[start.first][start.second]);
                        if (x.first == 4 && x.second == start.second) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(4, start.second));
                            place[4][start.second] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        } else if (start.second == 1 && place[2][1] == 0) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(2, start.second));
                            place[2][start.second] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        } else if (start.second > 1 && place[3][start.second - 1] == 0) {
                            cars.add(place[start.first][start.second]);
                            ret.add(new Pair(3, start.second - 1));
                            place[3][start.second - 1] = place[start.first][start.second];
                            place[start.first][start.second] = 0;
                        }
                    }
                    if (start.second == 1) {
                        start.first = 2;
                    } else start.second -= 1;
                }
            }

            for (int i = 2; i <= 3; i++) {
                for (int j = 1; j <= n; j++) {
                    if (place[i][j] != 0) {
                        out.println(-1);
                        return;
                    }
                }
            }

            out.println(ret.size());
            for (int i = 0; i < ret.size(); i++) {
                Pair e = ret.get(i);
                out.println(cars.get(i) + " " + e.first + " " + e.second);
            }
        }

        private Pair findPark(int v) {
            for (int i = 1; i <= n; i++) {
                if (place[1][i] == v) return new Pair(1, i);
            }

            for (int i = 1; i <= n; i++) {
                if (place[4][i] == v) return new Pair(4, i);
            }
            return new Pair(-1, -1);
        }

        private class Pair {
            int first;
            int second;

            public Pair(int first, int second) {
                this.first = first;
                this.second = second;
            }

        }

    }

    static class FastScanner {
        private BufferedReader bufferedReader;
        private StringTokenizer stringTokenizer;

        public FastScanner(InputStream inputStream) {
            bufferedReader = new BufferedReader(new InputStreamReader(inputStream));
        }

        public String next() {
            while (stringTokenizer == null || !stringTokenizer.hasMoreElements()) {
                try {
                    stringTokenizer = new StringTokenizer(bufferedReader.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return stringTokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
