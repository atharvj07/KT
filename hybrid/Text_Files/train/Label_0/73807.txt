from sys import stdin, stdout
input = stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	a = [int(x) for x in input().split()]
	if n == 1:
		if a[0] == 0:
			print(1, 2)
		else:
			print(2, 1)
		continue
	if a[-1] == 0:
		print(*[i for i in range(1, n + 2)])
		continue
	if sum(a) == n:
		print(*[n + 1] + [i for i in range(1, n + 1)])
		continue
	ans = []
	ok = 0
	for i in range(n - 1):
		if ok:
			ans.append(i + 1)
			continue
		if a[i] == 0 and a[i + 1] == 1:
			ok = 1
			ans.append(i + 1)
			ans.append(n + 1)
		else:
			ans.append(i + 1)
	ans.append(n)
	print(*ans)
