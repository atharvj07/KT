#include <stdio.h>
#include <math.h>
#include <algorithm>
#include <vector>
#include <iostream>
using namespace std;
#define PI 3.14159265
#define pb push_back
int x[100005];
int bit[(1<<17)+1];
void update(int i,int x)
{
	for(int j=i;j<=(1<<17);j+=j&-j)
	{
		bit[j] += x;
	}
}
int sum(int i)
{
	int res = 0;
	for(int j=i;j>0;j-=j&-j)
	{
		res += bit[j];
	}
	return res;
}
#define mod 1000000007
typedef long long ll;
ll f[100005];
int main()
{
	int n,r;
	f[0] = 1LL;
	for(int i=1;i<=100000;i++) f[i] = (f[i-1]*1LL*i)%mod;
	while(1)
	{
		cin >> n;
		if(n == 0) return 0;
		cin >> r;
		for(int i=1;i<=n;i++) x[i] = i,update(i,1);
		for(int i=0;i<r;i++)
		{
			int a,b;
			cin >> a >> b;
			swap(x[a],x[b]);
		}
		ll res = 0;
		for(int i=1;i<=n;i++)
		{
			int b = sum(x[i]);
			update(x[i],-1);
			res = (res + (b-1LL)*f[n-i]%mod)%mod;
		}
		cout << res << endl;
	}
}
