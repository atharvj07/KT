#include<iostream>
using namespace std;

int main(){
  int num[181];
  for(int i=0;i<181;i++){
    num[i]=(i+1)*(i+1);
  }

  int n;
  while(cin>>n,n){
    int cnt=0;
    int a,b,c;
    for(int i=0;i<181;i++){
      if((a=num[i])>n) continue;
      if(a==n){ cnt++; break;}
      for(int j=i;j<181;j++){
	if((b=a+num[j])>n) continue;
	if(b==n) {cnt++; break;}
	for(int k=j;k<181;k++){
	  if((c=b+num[k])>n) continue;
	  if(c==n){ cnt++; break;}
	  for(int m=k;m<181;m++){
	    if((c+num[m])==n) cnt++;
	  }
	}
      }
    }
    cout<<cnt<<endl;
  }
  return 0;
}
