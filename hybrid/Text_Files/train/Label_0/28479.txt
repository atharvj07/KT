#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, n, i, j;
  cin >> t;
  while (t--) {
    cin >> n;
    string s, temp;
    cin >> s;
    if (s.size() < 11) {
      cout << "NO" << endl;
      continue;
    }
    if (s.size() == 11) {
      if (s[0] == '8')
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
      continue;
    }
    temp = s.substr(0, n - 11 + 1);
    int flag = 0;
    for (i = 0; i < temp.size(); i++) {
      if (temp[i] == '8') {
        flag = 1;
        break;
      }
    }
    if (flag == 1)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
