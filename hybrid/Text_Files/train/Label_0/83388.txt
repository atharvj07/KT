(n, k) = map(int, input().strip().split())
s = input()
maxi = 0
start = 0
last = 0
left = k
for i in range(n):
	if s[i] == 'a':
		last += 1
	elif left > 0:
		left -= 1
		last += 1
	else:
		while s[start] != 'b':
			start += 1
			last -= 1
		start += 1
	maxi = max(last, maxi)
start = 0
last = 0
left = k
for i in range(n):
	if s[i] == 'b':
		last += 1
	elif left > 0:
		left -= 1
		last += 1
	else:
		while s[start] != 'a':
			start += 1
			last -= 1
		start += 1
	maxi = max(last, maxi)
print(maxi)
