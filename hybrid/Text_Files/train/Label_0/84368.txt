(x, y) = input().split()
x = int(x)
y = int(y)
matrix = []
for i in range(0, x):
	row = []
	var = input().split()
	row = [int(x) for x in var]
	matrix.append(row)
finishedpainter = [0 for y in range(x)]
for i in range(0, y):
	free = 0
	for j in range(0, x):
		inicio = max(free, finishedpainter[j])
		finishedpainter[j] = inicio + matrix[j][i]
		free = finishedpainter[j]
retorno = ''
for i in range(0, x):
	if i == x - 1:
		retorno += str(finishedpainter[i])
		continue
	retorno += str(finishedpainter[i]) + ' '
print(retorno)
