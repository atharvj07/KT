void run() {
    string s;
    cin >> s;
    int n = s.length();
    bool flag = true;
    for (int i = 1; i < n; i++) if (s[i] == '0') {
        if (s[i] == s[i - 1]) {
            flag = false;
        }
    }
    if (flag) {
        cout << "YES" << '\n';
        return;
    }
    flag = true;
    for (int i = 1; i < n; i++) if (s[i] == '1') {
        if (s[i] == s[i - 1]) {
            flag = false;
        }
    }
    if (flag) {
        cout << "YES" << '\n';
        return;
    }
    
    vector<int> nxt(n + 1, n + 1);
    for (int i = n - 1; i >= 0; i--) {
        nxt[i] = nxt[i + 1];
        if (s[i] == '1') nxt[i] = i + 1;
    }
    vector<int> pre(n + 1, 1), suf(n + 1, 1);
    for (int i = 1; i < n; i++) {
        pre[i + 1] &= pre[i];
        if (s[i] == '1' && s[i] == s[i - 1]) {
            pre[i + 1] = 0;
        }
    }
    for (int i = n - 2; i >= 0; i--) {
        suf[i + 1] &= suf[i + 2];
        if (s[i] == '0' && s[i] == s[i + 1]) {
            suf[i + 1] = 0;
        }
    }
    for (int i = 0; i <= n; i++) {
        if (pre[i] && (nxt[i] > n || (nxt[i] > i + 1 && suf[nxt[i]]))) {
            flag = true;
        }
    }
    if (flag) {
        cout << "YES" << '\n';
    } else {
        cout << "NO" << '\n';
    }
}

    run();

}
