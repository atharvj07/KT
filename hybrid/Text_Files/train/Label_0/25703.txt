#include <bits/stdc++.h>
using namespace std;
bool isSimilarChars(char c1, char c2) {
  int num1 = (int)c1;
  int num2 = (int)c2;
  if (c1 == c2) {
    return true;
  } else if ((num1 > num2) && ((char)(c1 - 32) == c2)) {
    return true;
  } else if ((num2 > num1) && ((char)(c2 - 32) == c1)) {
    return true;
  }
  if (((int)c1 >= 65) && ((int)c1 <= 90)) {
    c1 = (char)c1 + 32;
  }
  if (((int)c2 >= 65) && ((int)c2 <= 90)) {
    c2 = (char)c2 + 32;
  }
  if ((c1 == '0') && (c2 == 'o')) {
    return true;
  } else if ((c2 == '0') && (c1 == 'o')) {
    return true;
  } else if ((c1 == 'o') && (c2 == 'O')) {
    return true;
  } else if ((c2 == 'o') && (c1 == 'O')) {
    return true;
  } else if ((c1 == '0') && (c2 == 'O')) {
    return true;
  } else if ((c2 == '0') && (c1 == 'O')) {
    return true;
  } else if ((c1 == '1') && ((c2 == 'l') || (c2 == 'i'))) {
    return true;
  } else if ((c1 == 'i') && ((c2 == '1') || (c2 == 'l'))) {
    return true;
  } else if ((c1 == 'l') && ((c2 == 'i') || (c2 == '1'))) {
    return true;
  } else
    return false;
}
int main() {
  string login;
  int n;
  getline(cin, login);
  cin >> n;
  bool isFree = true;
  string input;
  for (int i = 0; i <= n; i++) {
    getline(cin, input);
    bool isSimilar = true;
    if (isFree) {
      if (input.size() != login.size()) {
        isSimilar = false;
      } else {
        int j = 0;
        while ((j < input.length()) && (isSimilar) && (isFree)) {
          isSimilar = isSimilarChars(input[j], login[j]);
          j++;
        }
        isFree = !isSimilar;
      }
    }
  }
  if (isFree) {
    cout << "Yes" << endl;
  } else {
    cout << "No" << endl;
  }
  return 0;
}
