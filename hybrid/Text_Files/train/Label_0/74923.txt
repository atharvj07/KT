import random
d = [int(d) for d in input().split()]
a = ['a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
s = []
word = ''
while len(word) != d[0]:
	x = random.randint(0, 25)
	if a[x] not in s and len(s) < d[1]:
		s.append(a[x])
		word += a[x]
	elif len(s) == d[1]:
		c = random.choice(s)
		if c != word[-1]:
			word += c
print(word)
