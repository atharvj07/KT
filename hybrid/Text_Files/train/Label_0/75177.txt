a = input()
b = input()
ct = 0
i = 0
while i < len(a):
	if i + 2 < len(a) and a[i] == '0' and (a[i + 1] == '0') and (a[i + 2] == '0') and (b[i] == '0') and (b[i + 1] == '0') and (b[i + 2] == '0'):
		ct += 2
		i += 2
	elif a[i:i + 2].count('0') + b[i:i + 2].count('0') >= 3:
		ct += 1
		i += 1
	i += 1
print(ct)
