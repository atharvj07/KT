t = int(input())
for i in range(t):
	n = int(input())
	jars = [int(x) for x in input().split()]
	left = [0] * (n + 1)
	right = [0] * (n + 1)
	for x in range(1, n + 1):
		if jars[x - 1] == 1:
			left[x] = left[x - 1] + 1
		else:
			left[x] = left[x - 1] - 1
		if jars[2 * n - x] == 1:
			right[x] = right[x - 1] + 1
		else:
			right[x] = right[x - 1] - 1
	a = {}
	b = {}
	for l in range(len(left)):
		a[left[l]] = l
	for r in range(len(right)):
		b[right[r]] = 2 * n - r + 1
	ans = 2 * n
	for key in a:
		if -key in b:
			ans = min(ans, b[-key] - (a[key] + 1))
	print(ans)
