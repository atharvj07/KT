import static java.lang.Math.max;

import java.io.IOException;
import java.io.PrintWriter;
import java.util.Arrays;

public class CF_279E {

    static Reader sc = new Reader();
    static PrintWriter out = new PrintWriter(System.out);

    static void compMin(char[] num, int from) {
        for(int i=from; i<num.length; i++) {
            if(num[i]=='?')
                num[i]='0';
        }       
    }

    static boolean compare(char[] act, char[] prev, int from) {     
        for(int i=from; i<prev.length; i++) {
            if(act[i]>prev[i]){
                return true;
            } else if (act[i]<prev[i]) {
                return false;
            }
        }           
        return false;       
    }

    static char[] getMin(char[] act, char[] prev, int from) {
        if(from==act.length) {
            act[0] = 'X';
            return act;
        }
        int i=from;
        if(act[i]!='?') {
            if(prev[i]> act[i]){
                act[0] ='X';
                return act;
            } else if(prev[i]< act[i]){
                compMin(act, from);
                return act;
            } else {
                char[] aux = act.clone();               
                aux = getMin(aux,prev,from+1);                  
                if(!(aux[0]=='X') && compare(aux,prev,from)){
                    return aux;
                } else {
                    act[0] = 'X';
                    return act;
                }
            }
        } else {
            if(from==0 && prev[0]=='0') {
                act[0] = '1';
                return getMin(act,prev,from);
            }
            char[] aux = act.clone();               
            aux = getMin(aux,prev,from+1);          
            if(!(aux[0]=='X') && compare(aux,prev,from+1)){
                aux[i] = prev[i];
                return aux;
            } else if(prev[i]<'9'){
                act[i] = (char)((int)prev[i]+1);
                compMin(act, from);
                return act;
            } else {
                act[0] = 'X';
                return act;
            }
        }
    }

    public static void main(String[] args) throws IOException {             
        int n = sc.nextInt();

        int[] res = new int[n];
        char[][] resc = new char[n][];

        for(int i=0; i<n; i++) {
            if(i==0) {
                char[] act = sc.next().toCharArray();
                compMin(act, 0);
                act[0] = (char)max((int)act[0],(int)'1');
                res[i] = getInt(act);
                resc[i] = act;
                continue;
            }

            char[] act = sc.next().toCharArray();
            char[] prev = new char[act.length];         
            char[] prevRaw = resc[i-1];

            if (act.length<prevRaw.length) {
                out.println("NO");
                out.close();
                return;             
            }

            for(int x=0;x<prev.length-prevRaw.length;x++)
                prev[x] = '0';
            for(int x=0; x<prevRaw.length;x++){
                prev[x+prev.length-prevRaw.length] = prevRaw[x];
            }
            char[] actRes = getMin(act, prev, 0);
            if(actRes[0]=='X'){
                out.println("NO");
                out.close();
                return;     
            }
            resc[i] = actRes;
            res[i] = getInt(actRes);
            if (res[i]<=res[i-1]) {
                out.println("NO");
                out.close();
                return;     
            }
        }
        System.out.println("YES");
        for(int i=0; i<n; i++)
            System.out.println(res[i]);

        out.close();
    }

    static int getInt(char[] s) {
        int ret=0;
        int i=0;
        while(s[i]<='0' && i<s.length)i++;
        for(; i<s.length;i++) {
            ret=ret*10+s[i]-'0';
        }
        return ret;
    }
    
    static int getInt(char[] s, int from) {
        int ret=0;
        int i=from;
        while(s[i]<='0' && i<s.length)i++;
        for(; i<s.length;i++) {
            ret=ret*10+s[i]-'0';
        }
        return ret;
    }
    
    static int getInt(String s) {
        int ret=0;
        int i=0;
        while(s.charAt(i)<='0' && i<s.length())i++;
        for(; i<s.length();i++) {
            ret=ret*10+s.charAt(i)-'0';
        }
        return ret;
    }

        }public int nextInt() throws IOException{int ret=0;byte c=read();while(c<=' ')c=read();boolean neg=(c=='-');if(neg)c=read();do{ret=ret*10+c-'0';}while((c=read())>='0'&&c<='9');if(neg)return -ret;return ret;

    static void print(int[] A) {for(int i=0;i<A.length;i++){if(i!=0) out.print(' ');out.print(A[i]);}out.println();}

}
