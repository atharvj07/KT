(n, m) = map(int, input().split())
if n - 1 <= m <= 2 * (n + 1):
	sol = []
	extras = m - 2 * (n - 1)
	if extras > 0:
		sol.append('1' if extras == 1 else '11')
		m -= extras
	double1s = m - n + 1
	sol.append('0')
	sol.append('110' * double1s + '10' * (n - 1 - double1s))
	if extras > 2:
		sol.append('1' if extras == 3 else '11')
	print(''.join(sol))
else:
	print(-1)
