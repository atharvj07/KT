#include <bits/stdc++.h>
using namespace std;
const int maxn = 200000 + 5;
int b[maxn], c[maxn], a[maxn], d[maxn];
int n;
int k[31];
int ab[31][maxn];
int bb[31][maxn];
int cb[31][maxn];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; i++) cin >> b[i];
  for (int i = 0; i < n; i++) cin >> c[i];
  long long sumd = 0;
  for (int i = 0; i < n; i++) {
    d[i] = b[i] + c[i];
    sumd += d[i];
  }
  sumd /= 2 * n;
  for (int i = 0; i < n; i++) a[i] = (d[i] - sumd) / n;
  for (int i = 0; i < n; i++) {
    if (a[i] < 0) {
      cout << -1 << endl;
      return 0;
    }
    for (int j = 0; j < 31; j++)
      if ((a[i] & (1 << j)) != 0) {
        k[j]++;
        ab[j][i] = 1;
      }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 31; j++) {
      bb[j][i] = (ab[j][i] == 1) * k[j];
      cb[j][i] = (ab[j][i] == 1) ? n : k[j];
    }
    int tb = 0, tc = 0;
    for (int j = 0; j < 31; j++) {
      tb += bb[j][i] << j;
      tc += cb[j][i] << j;
    }
    if ((tb != b[i]) || (tc != c[i])) {
      cout << -1 << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) cout << a[i] << ' ';
  cout << endl;
  return 0;
}
