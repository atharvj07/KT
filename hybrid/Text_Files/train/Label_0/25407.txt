#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 10;
ll a[maxn], b[maxn];
vector <int> G[maxn];
ll dp[maxn][2]; //dp[i][0]代表第i个节点选a，否则选b
int n;
void dfs(int u, int fu) {
    dp[u][0] = dp[u][1] = 0;
    for (auto v : G[u]) {
        if (v == fu) continue;
        dfs(v, u);
        dp[u][0] += max(dp[v][0] + abs(a[u] - a[v]), dp[v][1] + abs(a[u] - b[v]));
        dp[u][1] += max(dp[v][0] + abs(b[u] - a[v]), dp[v][1] + abs(b[u] - b[v]));
    }
}
void init() {
    for (int i = 1; i <= n; i++)
        G[i].clear();
}
int main() {
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int _;
    cin >> _;
    while (_--) {
        init();
        cin >> n;
        for (int i = 1; i <= n; i++)
            cin >> a[i] >> b[i];
        int u, v;
        for (int i = 1; i < n; i++)
            cin >> u >> v, G[u].push_back(v), G[v].push_back(u);
        dfs(1, 0);
        cout << max(dp[1][0], dp[1][1]) << endl;
    }
}
