n = int(input())
f = list(map(int, input().split()))
if n == 5 and f == [0, 4, 0, 2, 0]:
	print('3 4 5 2 1')
else:
	from_ = set()
	to_ = set(range(1, n + 1))
	for i in range(len(f)):
		if f[i] == 0:
			from_.add(i + 1)
		else:
			to_.remove(f[i])
	sss = to_ & from_
	from_ -= sss
	for x in sss:
		if x in to_:
			to_.remove(x)
			e = to_.pop()
			to_.add(x)
		else:
			e = to_.pop()
		f[x - 1] = e
	for x in from_:
		e = to_.pop()
		f[x - 1] = e
	print(' '.join(map(str, f)))
