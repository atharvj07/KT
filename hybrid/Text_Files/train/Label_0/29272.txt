#include <bits/stdc++.h>
const int ntf = 998244353;
long long frc[1000003], fic[1000003];
inline long long _inv(long long x) {
  long long r = 1;
  int v = ntf - 2;
  while (v) {
    (v & 1) && (r = r * x % ntf), x = x * x % ntf, v >>= 1;
  }
  return r;
}
int n;
long long ans;
long long f[1000003];
int main() {
  scanf(" %d", &n);
  frc[0] = 1;
  for (int i = 1; i <= n; ++i) frc[i] = frc[i - 1] * i % ntf;
  fic[n] = _inv(frc[n]);
  for (int i = n; i; --i) fic[i - 1] = fic[i] * i % ntf;
  ans = n * frc[n] % ntf;
  ans = ans * (ans + 1) / 2 % ntf;
  f[0] = n;
  for (int i = 1; i < n; ++i) {
    f[i] = frc[n] * fic[n - i + 1] % ntf *
           ((long long)(n - i) * (n - i) % ntf + 1) % ntf;
    ans = (ans - f[i] * i % ntf + ntf) % ntf;
  }
  for (int i = 0; i < n; ++i) {
    f[i] = (frc[n] - f[i] + ntf) % ntf;
    ans = (ans - f[i] * (n + i) % ntf + ntf) % ntf;
  }
  printf("%lld\n", ans);
  return 0;
}
