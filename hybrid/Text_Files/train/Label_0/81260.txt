import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
friends = [0] * n
dead = 0
for _ in range(m):
	(i, j) = map(int, input().split())
	if friends[min(i, j)] == 0:
		dead += 1
	friends[min(i, j)] += 1
q = int(input())
for _ in range(q):
	a = input().split()
	if a[0] == '1':
		(i, j) = (int(a[1]), int(a[2]))
		if friends[min(i, j)] == 0:
			dead += 1
		friends[min(i, j)] += 1
	if a[0] == '2':
		(i, j) = (int(a[1]), int(a[2]))
		if friends[min(i, j)] == 1:
			dead -= 1
		friends[min(i, j)] -= 1
	if a[0] == '3':
		print(n - dead)
