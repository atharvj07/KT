#include <bits/stdc++.h>
using namespace std;
long long n;
vector<long long> arr;
int main(int argc, char* argv[]) {
  ios::sync_with_stdio(false);
  cin >> n;
  arr.resize(n + 1);
  long long a;
  long long sum = 0;
  for (long long i = 1; i <= n; ++i) {
    cin >> a;
    arr[i] = a;
    sum += a;
  }
  long long result = LLONG_MAX;
  long long original_sum = sum;
  for (long long i = 2; i * i <= sum; ++i) {
    if (sum % i == 0) {
      long long curr = 0;
      long long curr_result = 0;
      for (long long j = 1; j <= n; ++j) {
        curr += arr[j];
        curr %= i;
        curr_result += min(curr, i - curr);
      }
      result = min(result, curr_result);
      while (sum % i == 0) {
        sum /= i;
      }
    }
  }
  if (sum > 1) {
    long long curr = 0;
    long long curr_result = 0;
    for (long long j = 1; j <= n; ++j) {
      curr += arr[j];
      curr %= sum;
      curr_result += min(curr, sum - curr);
    }
    result = min(result, curr_result);
  }
  if (result == LLONG_MAX || original_sum == 1) {
    cout << "-1" << endl;
  } else {
    cout << result << endl;
  }
  return 0;
}
