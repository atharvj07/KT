from sys import stdin
input = stdin.readline
for i in range(int(input())):
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	sum = 0
	if m < n or n == 2:
		print(-1)
		continue
	for i in range(n - 1):
		sum += a[i] + a[i + 1]
	sum += a[-1] + a[0]
	print(sum)
	for i in range(1, n):
		print(i, i + 1)
	print(n, 1)
