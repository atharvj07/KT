#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll,ll> pll;

ll N,H,d,h;

int main(){
  cin >> N >> H;
  ll ans = 0;

  ll cur = H;
  priority_queue<pll> q;
  for(int i=0;i<N-1;i++){

    cin >> d >> h;
    ll rem = H-cur;
    priority_queue<pll> tmp;
    while(q.size() && q.top().first >= h){
      pll p = q.top(); q.pop();
      rem -= p.second * p.first;
      tmp.push(p);
    }
    swap(q,tmp);
    ll times = rem/h;
    if(times)q.push(pll(h,times));
    if(times*h != rem)q.push(pll(rem-times*h,1));

    cur -= d;
    while(cur<=0){
      pll p = q.top(); q.pop();
      ll need = -cur/p.first+1;

      if(need <= p.second){
	ans += need;
	p.second -= need;
	cur += need * p.first;
	if(p.second)q.push(p);
      }else{
	ans += p.second;
	cur += p.first * p.second;
      }
    }
  }

  cout << ans << endl;
}
