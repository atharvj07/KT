(n, k) = [int(i) for i in input().split()]
p = [int(i) for i in input().split()]
colors = [-1 for i in range(256)]
for i in range(len(p)):
	if colors[p[i]] == -1:
		stack = [p[i]]
		val = p[i]
		for j in range(1, k):
			if p[i] - j < 0:
				break
			elif colors[p[i] - j] != -1 and p[i] - colors[p[i] - j] >= k:
				break
			elif colors[p[i] - j] != -1:
				val = colors[p[i] - j]
				break
			else:
				val = p[i] - j
				stack.append(p[i] - j)
		else:
			val = max(0, p[i] - k + 1)
		while len(stack):
			colors[stack.pop()] = val
	print(colors[p[i]], end=' ')
