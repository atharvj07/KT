from collections import defaultdict, deque
import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
l = list(input())
deg = [0 for i in range(n)]
graph = defaultdict(list)
for i in range(m):
	(a, b) = map(int, input().split())
	a -= 1
	b -= 1
	graph[a].append(b)
	deg[b] += 1
q = deque()
for i in range(n):
	if deg[i] == 0:
		q.append(i)
count = 0
ans = 0
dp = [[0 for i in range(26)] for i in range(n)]
while count < n and q:
	und = q.popleft()
	count += 1
	dp[und][ord(l[und]) - 97] += 1
	for i in graph[und]:
		for j in range(26):
			dp[i][j] = max(dp[i][j], dp[und][j])
		deg[i] -= 1
		if deg[i] == 0:
			q.append(i)
if count != n:
	print(-1)
else:
	ans = 0
	for i in range(n):
		ans = max(ans, max(dp[i]))
	print(ans)
