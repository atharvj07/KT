for _ in range(int(input())):
	s = input()
	t = input()
	ans = ''
	l = [0] * 26
	a = s.count('a')
	b = s.count('b')
	c = s.count('c')
	for i in s:
		l[ord(i) - 97] += 1
	if t[0] == 'a' and t[1] == 'b' and l[0]:
		for i in range(a):
			ans += 'a'
		for i in range(c):
			ans += 'c'
		for i in range(b):
			ans += 'b'
	else:
		for i in range(a):
			ans += 'a'
		for i in range(b):
			ans += 'b'
		for i in range(c):
			ans += 'c'
	for i in range(3, 26):
		for j in range(l[i]):
			ans += chr(i + 97)
	print(ans)
