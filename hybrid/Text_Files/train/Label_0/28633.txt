#include <bits/stdc++.h>
int gcd(int a, int b) { return (b == 0) ? a : gcd(b, a % b); }
int sqr(int a) { return a * a; }
using namespace std;
struct R {
  int p;
  int h;
};
struct wallpaper {
  int l;
  int w;
  int c;
};
bool cmp(R a, R b) {
  if (a.h < b.h) return true;
  return false;
}
int main() {
  int n;
  scanf("%d", &n);
  R room[n];
  for (int i = 0; i < n; i++) {
    int l, w;
    scanf("%d%d%d", &l, &w, &room[i].h);
    room[i].p = 2 * (l + w);
  }
  int m;
  scanf("%d", &m);
  wallpaper wp[m];
  for (int i = 0; i < m; i++) {
    scanf("%d%d%d", &wp[i].l, &wp[i].w, &wp[i].c);
  }
  int ans = 0;
  for (int i = n - 1; i >= 0; i--) {
    int min = INT_MAX;
    for (int j = 0; j < m; j++) {
      int now = 0;
      int st = (wp[j].l / room[i].h) * wp[j].w;
      int howSt = 0;
      if (st == 0) continue;
      while (now < room[i].p) {
        now += st;
        howSt++;
      }
      howSt *= wp[j].c;
      if (howSt < min) min = howSt;
    }
    ans += min;
  }
  printf("%d", ans);
}
