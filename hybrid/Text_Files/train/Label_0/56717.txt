 * @since 8/28/2019
 */
import java.util.Scanner;

public class Enhance
{
	private int x;
	
	public static void main(String[] args)
	{
		Enhance run = new Enhance();
		run.methods();
	}
	
	public void methods()
	{
		readInput();
		solve();
	}
	
	public void readInput()
	{
		Scanner in = new Scanner(System.in);
		x = in.nextInt();
	}
	
	/**
	 * Solves the problem and prints the answer.
	 * Uses helper method getCategory to retrieve the current category.
	 */
	public void solve()
	{
		char categoryInc0 = getCategory(x);
		char categoryInc1 = getCategory(x + 1);
		char categoryInc2 = getCategory(x + 2);
		
		if (categoryInc0 - 'A' < categoryInc1 - 'A' && categoryInc0 - 'A' < categoryInc2 - 'A')
		{
			System.out.println("0 " + categoryInc0);
		}
		else if (categoryInc1 - 'A' < categoryInc2 - 'A')
		{
			System.out.println("1 " + categoryInc1);
		}
		else
		{
			System.out.println("2 " + categoryInc2);
		}
	}
	
	/**
	 * Takes in an HP value and outputs the current category.
	 * @param  the current HP of the character
	 * @return the category of the HP
	 */
	public char getCategory(int x)
	{
		if (x % 4 == 1)
		{
			return 'A';
		}
		else if (x % 4 == 3)
		{
			return 'B';
		}
		else if (x % 4 == 2)
		{
			return 'C';
		}
		else
		{
			return 'D';
		}
	}
}
