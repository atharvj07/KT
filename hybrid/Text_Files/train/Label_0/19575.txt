#include <bits/stdc++.h>
using namespace std;
int a[10001];
int ans[100010];
int main() {
  int q, n, cnt = 0;
  cin >> q;
  for (int i = 1; i <= q; ++i) {
    cin >> n;
    for (int j = 1; j <= n; ++j) {
      cin >> a[j];
    }
    for (int j = 1; j <= n; ++j) {
      int x = a[j];
      int cnt = 1;
      while (x != j) {
        x = a[x];
        cnt++;
      }
      x = j;
      for (int it = 1; it <= n; ++it) {
        ans[x] = cnt;
        x = a[x];
      }
    }
    for (int j = 1; j <= n; ++j) cout << ans[j] << " ";
    cout << endl;
  }
  return 0;
}
