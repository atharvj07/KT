#include <bits/stdc++.h>
using namespace std;
const int N = 2 * 10 + 10;
vector<int> adj[N];
bool g[N][N];
int deg[N];
bool vis[N];
bool cmp(int a, int b) {
  if (deg[a] != deg[b])
    return deg[a] < deg[b];
  else
    return a < b;
}
void dfs(int v) {
  vis[v] = 1;
  for (auto u : adj[v]) {
    if (!vis[u]) return dfs(u);
  }
}
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int v, u;
    cin >> v >> u;
    v--, u--;
    if (v == u || g[v][u]) continue;
    g[v][u] = g[u][v] = 1;
    adj[v].push_back(u);
    adj[u].push_back(v);
    deg[v]++, deg[u]++;
  }
  for (int i = 0; i < n; i++) sort(adj[i].begin(), adj[i].end(), cmp);
  for (int i = 0; i < n; i++) {
    fill(vis, vis + n, false);
    dfs(i);
    bool ans = 1;
    for (int j = 0; j < n; j++) ans &= vis[j];
    if (ans) {
      cout << "Yes\n";
      return 0;
    }
  }
  cout << "No\n";
}
