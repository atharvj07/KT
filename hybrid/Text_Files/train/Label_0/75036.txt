from collections import defaultdict, Counter

def solve(n, m, p, a, b):
	ans = []
	cb = Counter(b)
	for q in range(min(p, n - (m - 1) * p)):
		arr = [a[i] for i in range(q, n, p)]
		lb = len(cb.keys())
		cnt = cb.copy()
		for v in arr[:m]:
			cnt[v] -= 1
			if cnt[v] == 0:
				lb -= 1
			elif cnt[v] == -1:
				lb += 1
		j = 0
		if lb == 0:
			ans.append(q + 1)
		for (i, v) in enumerate(arr[m:], m):
			cnt[v] -= 1
			if cnt[v] == 0:
				lb -= 1
			elif cnt[v] == -1:
				lb += 1
			cnt[arr[j]] += 1
			if cnt[arr[j]] == 0:
				lb -= 1
			elif cnt[arr[j]] == 1:
				lb += 1
			j += 1
			if lb == 0:
				ans.append(q + j * p + 1)
	print(len(ans))
	print(*sorted(ans))
(n, m, p) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
solve(n, m, p, a, b)
