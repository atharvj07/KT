#include <bits/stdc++.h>
using namespace std;
int row[1005];
int main() {
  ios_base::sync_with_stdio(0);
  int N;
  cin >> N;
  for (int i = 1; i <= N; ++i) row[i] = 2e9;
  for (int i = 0; i < 10; ++i) {
    int c = 0;
    for (int j = 1; j <= N; ++j) {
      if (j & (1 << i)) {
        ++c;
      }
    }
    if (!c) goto phase2;
    cout << c << endl;
    for (int j = 1; j <= N; ++j) {
      if (j & (1 << i)) {
        cout << j << " ";
      }
    }
    cout << endl;
    for (int j = 1; j <= N; ++j) {
      int x;
      cin >> x;
      if (!(j & (1 << i))) {
        row[j] = min(row[j], x);
      }
    }
  phase2:
    if (!(N - c)) continue;
    cout << N - c << endl;
    for (int j = 1; j <= N; ++j) {
      if (!(j & (1 << i))) {
        cout << j << " ";
      }
    }
    cout << endl;
    for (int j = 1; j <= N; ++j) {
      int x;
      cin >> x;
      if (j & (1 << i)) {
        row[j] = min(row[j], x);
      }
    }
  }
  cout << -1 << endl;
  for (int i = 1; i <= N; ++i) {
    cout << row[i] << " ";
  }
  cout << endl;
  return 0;
}
