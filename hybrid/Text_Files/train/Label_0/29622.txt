#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,k;
  cin>>n>>k;
  vector<double> p(n);
  for(int i=0;i<n;i++){
    cin>>p.at(i);
    p.at(i)=(p.at(i)+1)/2;
  }
  double s=0;
  for(int i=0;i<k;i++)
    s+=p.at(i);
  double m=s;
  for(int i=k;i<n;i++){
    s=s-p.at(i-k)+p.at(i);
    m=max(m,s);
  }
  cout<<fixed<<setprecision(10);
  cout<<m<<endl;
}
