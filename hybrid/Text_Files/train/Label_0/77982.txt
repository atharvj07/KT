def calc(S):
	N = len(S)
	A = [0] * (N + 1)
	B = [0] * (N + 1)
	for i in range(N):
		A[i + 1] = A[i] + S[i]
	for i in range(N)[::-1]:
		B[i] = B[i + 1] + S[i]
	mi = 1 << 20
	for i in range(N + 1):
		mi = min(mi, A[i] + (N - i - B[i]), i - A[i] + B[i])
	return mi
T = int(input())
for _ in range(T):
	print(calc([int(a) for a in input()]))
