#include <bits/stdc++.h>
using namespace std;
int nextInt() {
  int t;
  scanf("%d", &t);
  return t;
}
class Target {
 public:
  int shot;
  long long r;
  long long x;
  int opos;
  bool operator<(const Target o) const { return x < o.x; }
};
int main() {
  int n = nextInt();
  vector<Target> targets(n);
  for (int i = 0; i < n; ++i) {
    targets[i].x = nextInt();
    targets[i].r = nextInt();
    targets[i].opos = i;
  }
  sort(targets.begin(), targets.end());
  int m = nextInt();
  int tot = 0;
  for (int i = 1; i <= m; ++i) {
    int x = nextInt();
    int y = nextInt();
    int l = -1;
    int r = n;
    while (l < r - 1) {
      int m = l + (r - l) / 2;
      if (targets[m].x < x) {
        l = m;
      } else {
        r = m;
      }
    }
    if (l >= 0) {
      if ((x - targets[l].x) * (x - targets[l].x) + y * y <=
          targets[l].r * targets[l].r) {
        if (targets[l].shot == 0) {
          targets[l].shot = i + 1;
          ++tot;
        }
      }
    }
    if (r < n) {
      if ((x - targets[r].x) * (x - targets[r].x) + y * y <=
          targets[r].r * targets[r].r) {
        if (targets[r].shot == 0) {
          targets[r].shot = i + 1;
          ++tot;
        }
      }
    }
  }
  printf("%d\n", tot);
  vector<Target> ans(n);
  for (int i = 0; i < n; ++i) {
    ans[targets[i].opos] = targets[i];
  }
  for (int i = 0; i < n; ++i) {
    printf("%d ", ans[i].shot - 1);
  }
  printf("\n");
  return 0;
}
