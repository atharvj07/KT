#include<iostream>
#include<vector>
#include<cstring>
using namespace std;
char ca[8005];
bool dp[4005][8005];
vector<int> v1[2];

void check(int p,int q,int t,int n){
	int i,j,k;
	memset(dp,false,sizeof(dp));
	dp[0][0]=true;
	for(i=0;i<v1[p].size();i++){
		for(j=0;j<=n;j++){
			if(!dp[i][j])continue;
			dp[i+1][abs(j-v1[p][i])]=true;
			dp[i+1][j+v1[p][i]]=true;
		}
	}
}

int main(){
	int x,y;
	int i,j,k;
	int fir=-1;
	bool s1,s2;
	cin>>ca;
	cin>>x>>y;
	for(i=0,j=0,k=0;i<strlen(ca);i++){
		if(ca[i]=='T'){
			if(fir==-1)fir=k;
			else if(k!=0)v1[j].push_back(k);
			j=1-j,k=0;
		}else k++;
	}
	if(fir==-1)fir=k;
	else if(k!=0)v1[j].push_back(k);
	check(0,fir,x,strlen(ca));
	s1=dp[v1[0].size()][abs(fir-x)];
	check(1,0,y,strlen(ca));
	s2=dp[v1[1].size()][abs(y)];
	if(s1 && s2){
		cout<<"Yes"<<endl;
	}else cout<<"No"<<endl;
	return 0;
}
