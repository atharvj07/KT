#include <bits/stdc++.h>
using namespace std;
int table[100][100];
vector<int> Ends;
struct node {
  int i1;
  int j1;
  int i2;
  int j2;
};
vector<node> sol;
int main() {
  int n, num;
  cin >> n;
  Ends.clear();
  for (int i = 0; i < n; i++) {
    cin >> num;
    Ends.push_back(num);
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < Ends[i]; j++) {
      cin >> table[i][j];
    }
  }
  int cnt = 0;
  int swp;
  bool found;
  node tmp;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < Ends[i]; j++) {
      ++cnt;
      if (table[i][j] == cnt) continue;
      tmp.i1 = i;
      tmp.j1 = j;
      int temp;
      found = false;
      for (int ii = 0; ii < n && !found; ii++) {
        for (int jj = 0; jj < Ends[i] && !found; jj++) {
          if (table[ii][jj] == cnt) {
            tmp.i2 = ii;
            tmp.j2 = jj;
            temp = table[i][j];
            table[i][j] = table[ii][jj];
            table[ii][jj] = temp;
            found = true;
          }
        }
      }
      sol.push_back(tmp);
    }
  }
  swp = sol.size();
  cout << swp << "\n";
  for (int i = 0; i < swp; i++)
    cout << sol[i].i1 + 1 << " " << sol[i].j1 + 1 << " " << sol[i].i2 + 1 << " "
         << sol[i].j2 + 1 << "\n";
  return 0;
}
