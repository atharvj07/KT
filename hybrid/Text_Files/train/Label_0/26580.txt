#include <bits/stdc++.h>
using namespace std;
int t, n;
long long MAXLL = 1e18;
long long a[200000];
long long b[200000];
int main() {
  cout << setprecision(16);
  long long x, y, l, r;
  cin >> x >> y >> l >> r;
  long long xp = 1;
  long long yp = 1;
  int ctr = 0;
  while (xp <= r) {
    yp = 1;
    while (xp + yp <= r) {
      if (xp + yp >= l) a[ctr++] = xp + yp;
      if (MAXLL / y < yp) break;
      yp *= y;
    }
    if (MAXLL / x < xp) break;
    xp *= x;
  }
  sort(a, a + ctr);
  b[0] = l - 1;
  for (int i = 0; i < ctr; i++) {
    b[i + 1] = a[i];
  }
  ctr++;
  b[ctr++] = r + 1;
  long long bdf = 0;
  for (int i = 0; i < ctr - 1; i++) {
    bdf = max(bdf, b[i + 1] - b[i] - 1);
  }
  cout << bdf << endl;
  return 0;
}
