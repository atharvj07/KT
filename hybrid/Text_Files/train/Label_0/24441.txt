#include <bits/stdc++.h>
using namespace std;
int n, a[105], g[105][105], d;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    cin >> d;
    if (i - d >= 0) g[i - d][i] = g[i][i - d] = 1;
    if (i + d < n) g[i + d][i] = g[i][i + d] = 1;
    g[i][i] = 1;
  }
  for (int k = 0; k < n; k++)
    for (int i = 0; i < n; i++)
      for (int j = 0; j < n; j++)
        if (!g[i][j] && g[i][k] && g[k][j]) g[i][j] = g[j][i] = 1;
  for (int i = 0; i < n; i++)
    if (!g[a[i] - 1][i]) {
      cout << "NO";
      return 0;
    }
  cout << "YES";
  return 0;
}
