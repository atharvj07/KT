def solve():
	results = []
	for _ in range(int(input())):
		n = int(input())
		if n < 0:
			results.append('N')
			continue
		x = n ** 2 - 4 * n
		if x < 0:
			results.append('N')
			continue
		n1 = (n + x ** 0.5) / 2
		n2 = (n - x ** 0.5) / 2
		if n2 > n1:
			(n1, n2) = (n2, n1)
		results.append('Y ' + str(n1) + ' ' + str(n2))
	print('\n'.join((k for k in results)))
solve()
