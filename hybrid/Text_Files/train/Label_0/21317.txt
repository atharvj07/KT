#include <bits/stdc++.h>
using namespace std;
struct node {
  int p[26], v;
} T[50005];
int n, m, nd;
int dp[50005][105], sum[50005];
char s[505];
void add(char s[]) {
  int x = 0, c;
  for (int i = 0; s[i]; i++) {
    c = s[i] - 'a';
    if (!T[x].p[c]) T[x].p[c] = nd++;
    x = T[x].p[c];
  }
  T[x].v++;
}
void dfs(int x) {
  sum[x] = T[x].v;
  for (int i = (0); i < (26); i++) {
    if (!T[x].p[i]) continue;
    dfs(T[x].p[i]);
    sum[x] += sum[T[x].p[i]];
    for (int k = sum[x]; k >= 0; k--)
      for (int j = 1; j <= k && j <= sum[T[x].p[i]]; j++)
        dp[x][k] = max(dp[x][k], dp[x][k - j] + dp[T[x].p[i]][j]);
  }
  if (x)
    for (int i = (0); i <= (sum[x]); i++) dp[x][i] += i * (i - 1) / 2;
}
int main() {
  scanf("%d%d", &n, &m);
  nd = 1;
  while (n--) scanf("%s", s), add(s);
  dfs(0);
  printf("%d\n", dp[0][m]);
  return 0;
}
