(m, n) = map(int, input().split())
t = [None for _ in range(m)]
for i in range(m):
	t[i] = list(map(int, input().split()))
for i in range(1, n):
	t[0][i] += t[0][i - 1]
for i in range(1, m):
	t[i][0] += t[i - 1][0]
for i in range(1, m):
	for j in range(1, n):
		t[i][j] += max(t[i - 1][j], t[i][j - 1])
for i in range(m):
	print(t[i][n - 1], end=' ')
