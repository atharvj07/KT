// upsolve with rainboy
import java.io.*;
import java.util.*;

public class CF1182D {
	static final int MD = 1000000007;
	static ArrayList[] aa;
	static int[] xx, hh, yy;
	static boolean[] gg;
	static void dfs1(int n, int i) {
		int[] stack = new int[n];
		int[] pp = new int[n];
		int time = 0, cnt = 0;
		stack[cnt++] = i;
		while (cnt > 0) {
			i = stack[cnt - 1];
			ArrayList<Integer> adj = aa[i];
			if (pp[i] == 0) {
				pp[i] = ++time;
				for (int j : adj)
					if (pp[j] == 0)
						stack[cnt++] = j;
				continue;
			}
			boolean good = true;
			int h_ = -1, y_ = 0;
			for (int j : adj)
				if (pp[i] < pp[j]) {
					if (!gg[j])
						good = false;
					if (h_ == -1)
						y_ = yy[j];
					else if (y_ != yy[j])
						good = false;
					h_ = Math.max(h_, hh[j]);
				}
			int h = h_ + 1, x = xx[h];
			long y = 1;
			for (int j : adj)
				if (pp[i] < pp[j])
					y = y * (x + yy[j]) % MD;
			hh[i] = h;
			yy[i] = (int) y;
			gg[i] = good;
			cnt--;
		}
	}
	static int dfs2(int p, int i) {
		int h_ = 0, y_ = 1;
		while (true) {
			ArrayList<Integer> adj = aa[i];
			int j1 = -1;
			for (int j : adj)
				if (j != p)
					if (h_ != hh[j] || y_ != yy[j]) {
						j1 = j;
						break;
					}
			if (j1 == -1)
				return gg[i] ? i + 1 : -1;
			int h = h_ + 1, x = xx[h];
			long y = (x + y_) % MD;
			for (int j : adj)
				if (j != p && j != j1) {
					if (!gg[j] || h_ != hh[j] || y_ != yy[j])
						return -1;
					y = y * (x + y_) % MD;
				}
			p = i; i = j1; h_ = h; y_ = (int) y;
		}
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());
		if (n == 1) {
			System.out.println(1);
			return;
		}
		aa = new ArrayList[n];
		for (int i = 0; i < n; i++)
			aa[i] = new ArrayList<Integer>();
		for (int h = 0; h < n - 1; h++) {
			StringTokenizer st = new StringTokenizer(br.readLine());
			int i = Integer.parseInt(st.nextToken()) - 1;
			int j = Integer.parseInt(st.nextToken()) - 1;
			aa[i].add(j);
			aa[j].add(i);
		}
		xx = new int[n];
		Random rand = new Random();
		xx[0] = 1;
		for (int i = 1; i < n; i++)
			xx[i] = rand.nextInt() % MD;
		hh = new int[n];
		yy = new int[n];
		gg = new boolean[n];
		int r = -1, s = -1;
		for (int i = 0; i < n; i++) {
			ArrayList<Integer> adj = aa[i];
			if (adj.size() == 1) {
				r = i;
				s = adj.get(0);
				break;
			}
		}
		dfs1(n, r);
		System.out.println(gg[r] ? r + 1 : dfs2(r, s));
	}
}
