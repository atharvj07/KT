#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int MAXN = 1001;
int ans[MAXN][MAXN];
void solve() {
  int n, k, d;
  cin >> n >> k >> d;
  bool possible = false;
  int vv = 1;
  for (int i = 1; i <= d; ++i) {
    vv *= k;
    if (vv >= n) {
      possible = true;
      break;
    }
  }
  if (!possible) {
    cout << -1 << endl;
    return;
  }
  for (int i = 1; i < n; ++i) {
    for (int j = 0; j < d; ++j) {
      ans[i][j] = ans[i - 1][j];
    }
    for (int j = d - 1; j >= 0; --j) {
      ans[i][j]++;
      ans[i][j] %= k;
      if (ans[i][j]) break;
    }
  }
  for (int i = 0; i < d; ++i) {
    for (int j = 0; j < n; ++j) {
      cout << ans[j][i] + 1 << " ";
    }
    cout << endl;
  }
}
int main() {
  int t = 1;
  while (t--) solve();
  return 0;
}
