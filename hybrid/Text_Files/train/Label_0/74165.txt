t = int(input())
for i in range(t):
	(n, k) = map(int, input().split())
	s = input()
	for i in range(n):
		if s[i] == '*':
			x = i
			break
	for i in range(n - 1, -1, -1):
		if s[i] == '*':
			y = i
			break
	res = 0
	if x == y:
		print(1)
	else:
		i = x
		while i >= x and i < y:
			if s[i] == '*':
				z = i + k
				while z < y and z > i:
					if s[z] == '*':
						res += 1
						break
					z -= 1
			i = z
		print(res + 2)
