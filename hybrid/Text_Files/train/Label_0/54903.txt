import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class ContestPiA {

    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(bf.readLine());
        int []a = new int [n];
        StringTokenizer ns = new StringTokenizer(bf.readLine());
        for(int i=0;i<a.length;i++)
            a[i]=Integer.parseInt(ns.nextToken());
        
        int x=Math.abs(a[0]-a[1]);
        int y=Math.abs(a[0]-a[a.length-1]);     
        System.out.println(x + " "+y);
        for (int i=1;i<a.length-1;i++){
            int max=0,min=0;
            int x1=Math.abs(a[i]-a[i+1]);
            int x2=Math.abs(a[i]-a[i-1]);
            if(x1<x2)
                min=x1;
            else
                min=x2;
            int y1 = Math.abs(a[i]-a[a.length-1]);
            int y2 = Math.abs(a[i]-a[0]);
            if(y1>y2)
                max=y1;
            else
                max=y2;
            
            System.out.println(min + " "+ max);
            
        }       
        x=Math.abs(a[a.length-2]-a[a.length-1]);
        y=Math.abs(a[0]-a[a.length-1]);     
        System.out.println(x + " "+y);      
    }
}
