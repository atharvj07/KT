#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int tests;
  cin >> tests;
  for (int test = 0; test < (int)(tests); test++) {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < (int)(n); i++) {
      int x;
      cin >> x;
      m[-x]++;
    }
    vector<int> v;
    for (auto i : m) {
      v.push_back(i.second);
    }
    int g, s, b;
    int j = 0;
    int k = 0;
    g = s = b = 0;
    vector<int> ans(3);
    g += v[0];
    s -= v[0];
    for (; j < n; j++) {
      s += v[j];
      b -= v[j];
      if (s > g) {
        break;
      }
    }
    for (; k < n; k++) {
      b += v[k];
      if (b > g) {
        break;
      }
    }
    k++;
    while (g + s + b <= n / 2 && k < v.size()) {
      ans[0] = g;
      ans[1] = s;
      ans[2] = b;
      b += v[k];
      k++;
    }
    cout << ans[0] << ' ' << ans[1] << ' ' << ans[2] << endl;
  }
  return 0;
}
