import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    private static final FastReader scanner = new FastReader();

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader()
        {
            br = new BufferedReader(
                    new InputStreamReader(System.in));
        }

        String next()
        {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() { return Integer.parseInt(next()); }

        long nextLong() { return Long.parseLong(next()); }

        double nextDouble()
        {
            return Double.parseDouble(next());
        }

        String nextLine()
        {
            String str = "";
            try {
                str = br.readLine();
            }
            catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }

    // START //

    public static void solve() {
        int n = scanner.nextInt();

        Node node = null;
        boolean hasAns = true;
        for (int i = 0; i < n; i++) {
            int b = scanner.nextInt();
            if (i == 0) {
                node = new Node(b);
                continue;
            }

            if (node.value == b) {
                continue;
            }

            if (node.value > b) {
                if (node.prev != null && node.prev.value == b) {
                    node = node.prev;
                    continue;
                }

                if (node.prev == null || node.prev.value <= b) {
                    Node newNode = new Node(b);
                    Node prev = node.prev;
                    node.prev = newNode;
                    newNode.next = node;
                    if (prev != null) {
                        prev.next = newNode;
                        newNode.prev = prev;
                    }
                    node = newNode;
                } else {
                    hasAns = false;
                }

                continue;
            }

            if (node.value < b) {
                if (node.next != null && node.next.value == b) {
                    node = node.next;
                    continue;
                }

                if (node.next == null || node.next.value >= b) {
                    Node newNode = new Node(b);
                    Node next = node.next;
                    node.next = newNode;
                    newNode.prev = node;
                    if (next != null) {
                        next.prev = newNode;
                        newNode.next = next;
                    }
                    node = newNode;
                } else {
                    hasAns = false;
                }
            }
        }

        if (hasAns) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static class Node {
        public int value;
        public Node prev;
        public Node next;

        public Node(int value) {
            this.value = value;
        }
    }

    public static void main(String[] args) {
        int testCount = scanner.nextInt();

        for (int t = 0; t < testCount; t++) {
            solve();
        }
    }
}
