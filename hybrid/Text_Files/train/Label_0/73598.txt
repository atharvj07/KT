get = lambda : map(int, input().split())
(n, s) = get()
(h, m) = get()
prev = curr = h * 60 + m
if prev >= s + 1:
	print(0, 0)
	exit()
for i in range(n - 1):
	(h, m) = get()
	curr = h * 60 + m
	if curr - prev >= 2 * s + 2:
		print(*divmod(prev + s + 1, 60))
		exit()
	else:
		prev = curr
print(*divmod(curr + s + 1, 60))
