import sys
T = int(input())
for _ in range(T):
	s = input()
	s = s + '0'
	A = []
	f = 0
	c = 0
	for i in range(len(s)):
		if s[i] == '1' and f == 0:
			c = c + 1
			f = 1
		elif s[i] == '1' and f == 1:
			c = c + 1
		elif s[i] == '0' and f == 1:
			A.append(c)
			(c, f) = (0, 0)
		else:
			(c, f) = (0, 0)
	A.sort(reverse=True)
	ans = 0
	for i in range(0, len(A), 2):
		ans = ans + A[i]
	print(ans)
