const int maxn = 1e5;

int pn(int x)
{
    int ans = 0;
    for (int i = 2; i * i <= x; i++)
        while (x % i == 0)
        {
            ans++;
            x /= i;
            if (x == 1)
                break;
        }
    if (x > 1)
        ans++;
    return ans;
}
void solve()
{
    ll n, m, k;
    cin >> n >> m >> k;
    ll min_ans = 2;
    ll max_ans = pn(n) + pn(m);
    if (k == 1)
        if (n % m == 0 || m % n == 0)
        {
            if (m == n)
            {
                cout << "NO\n";
                return;
            }
        }
        else
        {
            cout << "NO\n";
            return;
        }
    if (k <= max_ans)
    {
        cout << "YES\n";
        return;
    }
    cout << "NO\n";
}

int main()
{
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);

    {
