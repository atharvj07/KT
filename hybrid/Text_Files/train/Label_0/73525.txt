import sys
import collections
t = int(input())
for _ in range(t):
	_ = input()
	a = [int(x) for x in input().split()]
	b = [int(x) for x in input().split()]
	f = collections.defaultdict(bool)
	presents_before = 0
	r = 0
	i = 0
	j = 0
	while j < len(b):
		if b[j] in f:
			presents_before -= 1
			r += 1
		else:
			while a[i] != b[j]:
				f[a[i]] = True
				presents_before += 1
				i += 1
			i += 1
			r += presents_before * 2 + 1
		j += 1
	print(r)
