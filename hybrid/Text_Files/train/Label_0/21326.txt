#include "bits/stdc++.h"
using namespace std;
typedef long long ll;
typedef pair<int,int> pii;
#define rep(i,n) for(ll i=0;i<(ll)(n);i++)
#define all(a)  (a).begin(),(a).end()
#define pb push_back
#define INF 99999999
#define eps 1e-9


int main(){
    string s;
    cin>>s;
    string t="";
    for(int i=0;i<s.size();){
        if( s.substr(i,3)=="egg" ){
            t+='1';
            i+=3;
        }
        else{
            t+='0';
            i+=7;
        }
    }
    vector<string> a;
    string buf="";
    buf+=t[0];
    for(int i=1;i<t.size();i++){
        if(t[i]!=t[i-1]){
            buf+=t[i];
        }else{
            a.pb(buf);
            buf="";
            buf+=t[i];
        }
    }
    if(buf.size())a.pb(buf);
    int l=0;
    string ans="";
    rep(i,a.size()){
        if(a[i].size()>l){ans=a[i];l=ans.size();}
    }
    if( ans[ans.size()-1]=='0' ){
        cout<<"chicken"<<endl;
    }
    else cout<<"egg"<<endl;
}
