#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;

int n, a[MAXN];
long long R[MAXN];

int main(){
	scanf("%d", &n);
	
	for(int i = 0; i <= n; i++){
		scanf("%d", &a[i]);
		
		if(i == 0){
			R[i] = 1;
		}else{
			R[i] = (R[i - 1] - a[i - 1]) * 2;
		}
		R[i] = min(R[i], 1LL << 60);
		
		if(R[i] < a[i]){
			printf("-1\n");
			return 0;
		}
	}
	
	long long res = 0, c = a[n];
	for(int i = n; i >= 0; i--){
		res += c;
		
		if(i > 0){
			c = min(c + a[i - 1], R[i - 1]);
		}
	}
	printf("%lld\n", res);
		
	return 0;
}
