#include<iostream>
using namespace std;
const int e[] = { 1867,1911,1925,1988 };
const char c[] = { 'M','T','S','H' };
int main() {
	int mode, y; cin >> mode >> y;
	if (mode == 0) {
		for (int i = 3; i >= 0; i--) {
			if (y > e[i]) {
				cout << c[i] << y - e[i] << endl;
				goto end;
			}
		}
	}
	else cout << y + e[mode - 1] << endl;
end:;
}
