import java.util.*;
import java.io.*;

public class Main
{
    public static void main(String args[]) throws Exception
    {
        BufferedReader in = new BufferedReader(
                new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);
        Task t = new Task();
        t.solve(in, out);
        out.close();
    }
}

class Task
{
    void solve(BufferedReader in, PrintWriter out) throws Exception
    {
        StringTokenizer s = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(s.nextToken()),
                m = Integer.parseInt(s.nextToken());
        int d[] = new int[n + 1];
        
        for(int i = 0; i < m; ++i)
        {
            s = new StringTokenizer(in.readLine());
            if(s.nextToken().charAt(0) == '-')
            {
                int k = Integer.parseInt(s.nextToken());
                if(d[k] != k)
                {
                    out.println("Already off");
                    continue;
                }
                out.println("Success");
                d[k] = 0;
                for(int j = 2; j * j <= k; ++j)
                {
                    if(k % j != 0) continue;
                    d[j] = 0;
                    d[k / j] = 0;
                }
                continue;
            }
            int k = Integer.parseInt(s.nextToken());
            if(d[k] == k)
            {
                out.println("Already on");
                continue;
            }
            int isOn = 0;
            for(int j = 2; j * j <= k; ++j)
            {
                if(k % j != 0) continue;
                if(d[j] != 0)
                    isOn = d[j];
                else 
                    if(d[k / j] != 0)
                        isOn = d[k / j];
                    else continue;
                break;
            }
            if(d[k] != 0) isOn = d[k];
            if(isOn != 0)
            {
                out.printf("Conflict with %d\n", isOn);
                continue;
            }
            out.println("Success");
            for(int j = 2; j * j <= k; ++j)
            {
                if(k % j != 0) continue;
                d[j] = k;
                d[k / j] = k;
            }
            d[k] = k;
        }
        
    }
}
