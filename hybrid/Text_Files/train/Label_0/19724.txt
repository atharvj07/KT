#include<bits/stdc++.h>
#define M 200005
using namespace std;
int n,a[M],lg[M],MIN[M][25],MAX[M][25];
void init(){
    for(int i=2;i<M;i++){
        lg[i]=lg[i>>1]+1;
    }
}
void st(){
    for(int i=1;i<=n;i++){
        MAX[i][0]=MIN[i][0]=a[i];
    }
    for(int j=1;j<=lg[n];j++){
        for(int i=1;i+(1<<j)-1<=n;i++){
            MIN[i][j]=min(MIN[i][j-1],MIN[i+(1<<j-1)][j-1]);
            MAX[i][j]=max(MAX[i][j-1],MAX[i+(1<<j-1)][j-1]);
        }
    }
}
int getmin(int l,int r){
    int len=lg[r-l+1];
    return min(MIN[l][len],MIN[r-(1<<len)+1][len]);
}
int getmax(int l,int r){
    int len=lg[r-l+1];
    return max(MAX[l][len],MAX[r-(1<<len)+1][len]);
}
int main(){
    int t;cin>>t;
    init();
    while(t--){
        scanf("%d",&n);
        for(int i=1;i<=n;i++){
            scanf("%d",&a[i]);
        }
        st();
        int m1,m2,m3,flag=0;
        for(int x=1;x<=n-2 && !flag;x++){
            m1=getmax(1,x);
            int l=x+1,r=n-1;
            while(l<=r){
                int y=(l+r)>>1;
                m2=getmin(x+1,y),m3=getmax(y+1,n);
                if(m1<m2 || m1<m3){
                    l=y+1;
                }
                else if(m1>m2 || m1>m3){
                    r=y-1;
                }
                else{
                    printf("YES\n%d %d %d\n",x,y-x,n-y);
                    flag=1;
                    break;
                }
            }
        }
        if(!flag){
            printf("NO\n");
        }
    }
    return 0;
}
