
import java.io.*; 
import java.util.*;

import java.io.*; 
public class MM {
	static Scanner scanner=new Scanner(System.in);
	static Point[]points;
	public static void main(String[] args) {
		int n=scanner.nextInt();
		int x[]=new int [n];
		int y[]=new int [n];
		points=new Point[n];
		for(int i=0;i<n;i++) {
			x[i]=scanner.nextInt();
			y[i]=scanner.nextInt();
			points[i]=new Point(x[i], y[i], scanner.nextInt(), scanner.nextInt());
		}
		Arrays.sort(x);
		Arrays.sort(y);
		for(int i=(int)Math.max(0,n-5);i<n;i++) {
			for(int j=(int)Math.max(0,n-5);j<n;j++) {
				if(bb(x[i], y[j])) {
					System.out.println(x[i]+" "+y[j]);
					return;
				}
			}
		}
	}
	static boolean bb(int x,int y) {
		int c=0;
        for(int i=0;i<points.length;i++){
            if(x<=points[i].x2 && x>=points[i].x1 && y<=points[i].y2 && y>=points[i].y1) c++;
        }
        return c>=points.length-1;
	}
	static class Point{
		int x1,y1,x2,y2;
		public Point(int x, int y, int x1, int y1) {
			this.x1 = x;
			this.y1 = y;
			this.x2 = x1;
			this.y2 = y1;
		}
	}
}
