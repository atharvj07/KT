mod = 998244353
n = int(input())
a = list(map(int, input().split()))
a.sort()
res = 0
inv = [1, 1]
for i in range(2, n + 1):
	inv.append((mod - mod // i) * inv[mod % i] % mod)
for i in range(n):
	res += a[n + i] - a[i]
for i in range(2 * n, 0, -1):
	if i > n:
		res = res * i % mod
	else:
		res = res * inv[i] % mod
print(res)
