(scale_1, scale_2) = map(str, input().split('|'))
remain_weight = str(input())
if len(scale_1) == len(scale_2) and len(remain_weight) & 1 != 0 or len(scale_1) + len(remain_weight) < len(scale_2) or len(scale_2) + len(remain_weight) < len(scale_1):
	print('Impossible')
elif len(scale_1) + len(scale_2) == len(remain_weight):
	weight_dist = list(remain_weight)
	for i in range(len(weight_dist)):
		if len(scale_1) < len(scale_2):
			scale_1 += weight_dist[i]
		else:
			scale_2 += weight_dist[i]
	print(scale_1 + '|' + scale_2)
elif len(scale_1) + len(scale_2) == 0 and len(remain_weight) & 1 == 0:
	weight_dist = list(remain_weight)
	for i in range(len(weight_dist)):
		if len(scale_1) < len(scale_2):
			scale_1 += weight_dist[i]
		else:
			scale_2 += weight_dist[i]
	print(scale_1 + '|' + scale_2)
elif len(scale_1) > len(scale_2):
	weight_dist = list(remain_weight)
	for i in range(len(weight_dist)):
		if len(scale_1) > len(scale_2):
			scale_2 += weight_dist[i]
		else:
			scale_1 += weight_dist[i]
	if len(scale_2) == len(scale_1):
		print(scale_1 + '|' + scale_2)
	else:
		print('Impossible')
else:
	weight_dist = list(remain_weight)
	for i in range(len(weight_dist)):
		if len(scale_2) > len(scale_1):
			scale_1 += weight_dist[i]
		else:
			scale_2 += weight_dist[i]
	if len(scale_2) == len(scale_1):
		print(scale_1 + '|' + scale_2)
	else:
		print('Impossible')
