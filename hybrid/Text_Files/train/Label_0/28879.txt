#include <bits/stdc++.h>
using namespace std;
template <class t>
inline t read(t &x) {
  x = 0;
  char c = getchar();
  bool f = 0;
  while (!isdigit(c)) f |= c == '-', c = getchar();
  while (isdigit(c)) x = (x << 1) + (x << 3) + (c ^ 48), c = getchar();
  if (f) x = -x;
  return x;
}
template <class t>
inline void write(t x) {
  if (x < 0) {
    putchar('-'), write(-x);
  } else {
    if (x > 9) write(x / 10);
    putchar('0' + x % 10);
  }
}
int fpow(int x, int y, int mod) {
  int res = 1;
  for (; y; y >>= 1, mod ? (x *= x) %= mod : x *= x)
    if (y & 1) mod ? (res *= x) %= mod : res *= x;
  return res;
}
const int N = 2e5 + 5, M = 5e5 + 5;
int n, en, h[N], a, b, m;
bool v[N];
long long ans1, ans2;
struct edge {
  int n, v, w;
} e[M << 1];
void add(int x, int y, int z) {
  e[++en] = (edge){h[x], y, z};
  h[x] = en;
}
void dfs(int x, int op) {
  if (op == 1)
    ans2--;
  else
    ans1--;
  v[x] = 1;
  for (int i = h[x]; i; i = e[i].n) {
    int y = e[i].v;
    if (v[y]) continue;
    if (e[i].w & op) continue;
    dfs(y, op);
  }
}
void doit() {
  read(n);
  read(m);
  read(a);
  read(b);
  en = 0;
  ans1 = ans2 = n - 1;
  for (int i = 1; i <= n; i++) h[i] = 0;
  for (int i = 1, x, y; i <= m; i++) {
    read(x);
    read(y);
    int z = 0;
    if (x == a || y == a) z++;
    if (x == b || y == b) z += 2;
    add(x, y, z);
    add(y, x, z);
  }
  for (int i = 1; i <= n; i++) v[i] = 0;
  dfs(a, 2);
  for (int i = 1; i <= n; i++) v[i] = 0;
  dfs(b, 1);
  write(ans1 * ans2);
  puts("");
}
signed main() {
  int t;
  read(t);
  while (t--) doit();
}
