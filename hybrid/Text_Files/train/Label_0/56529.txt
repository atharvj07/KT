import java.io.*;
import java.util.*;

public class WhiteSheet {
    static class InputReader {
        BufferedReader reader;
        StringTokenizer tokenizer;
        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }
        String next() { // reads in the next string
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }
        public int nextInt() { return Integer.parseInt(next()); } // reads in the next int
        public long nextLong() { return Long.parseLong(next()); } // reads in the next long
        public double nextDouble() { return Double.parseDouble(next()); } // reads in the next double
    }
    static InputReader r = new InputReader(System.in);
    static PrintWriter pw = new PrintWriter(System.out);
    private static int x1, x2, x3, x4, x5, x6, y1, y2, y3, y4, y5, y6;
    public static void main(String[] args)
    {
        x1 = r.nextInt();
        y1 = r.nextInt();
        x2 = r.nextInt();
        y2 = r.nextInt();
        x3 = r.nextInt();
        y3 = r.nextInt();
        x4 = r.nextInt();
        y4 = r.nextInt();
        x5 = r.nextInt();
        y5 = r.nextInt();
        x6 = r.nextInt();
        y6 = r.nextInt();
        boolean rangex = false, rangey = false;
        String temp = "hello";
        // if covered by black
        if(coverOne())
        {
            temp = "COVERED";
        }
        if(coverTwo())
        {
            temp = "COVERED";
        }
        // quit
        if(temp.equals("COVERED"))
        {
            pw.println("NO");
            pw.close();
        }
        // if its not covered, and black are separate
        if((x5 > x4 && x6 > x3) || (y6 > y3 && y5 > y4))
        {
            pw.println("YES");
            pw.close();
        }
        if((x3 > x6 && x4 > x5) || (y3 > y6 && y4 > y5))
        {
            pw.println("YES");
            pw.close();
        }
        // if corner not covered
        if(!isCovered(x1, y1)) { temp = "CORNER"; }
        else if(!isCovered(x1, y2)) { temp = "CORNER"; }
        else if(!isCovered(x2, y2)) { temp = "CORNER"; }
        else if(!isCovered(x2, y1)) { temp = "CORNER"; }
        if(temp.equals("CORNER"))
        {
            pw.println("YES");
            pw.close();
        }
        if(temp.equals("hello"))
            pw.println("NO");
        pw.close(); // flushes the output once printing is done
    }
    public static boolean coverOne()
    {
        if(x1 < x3 || x1 > x4)
            return false;
        if(y1 < y3 || y1 > y4)
            return false;
        if(x2 < x3 || x2 > x4)
            return false;
        if(y2 < y3 || y2 > y4)
            return false;
        return true;
    }
    public static boolean coverTwo()
    {
        if(x1 < x5 || x1 > x6)
            return false;
        if(y1 < y5 || y1 > y6)
            return false;
        if(x2 < x5 || x2 > x6)
            return false;
        if(y2 < y5 || y2 > y6)
            return false;
        return true;
    }
    public static boolean isCovered(int x, int y)
    {
        if((x <= x6) && (x >= x5))
        {
            if((y <= y6) && (y >= y5))
            {
                return true;
            }
        }
        if((x <= x4) && (x >= x3))
        {
            if((y <= y4) && (y >= y3))
            {
                return true;
            }
        }
        return false;
    }
}
