vector<int> v[300005];
bool dp[300005][2];
int ed[300005];
int ze[300005];
int bo[300005];
int on[300005];
int P[300005];
void dfs(int node, int p = -1) {
  P[node] = p;
  bool isleaf = true;
  int zero = 0, one = 0, both = 0;
  for (int i = 0; i < v[node].size(); i++) {
    int cur = v[node][i];
    if (cur == p) continue;
    isleaf = false;
    dfs(cur, node);
    if (dp[cur][0] == dp[cur][1]) {
      if (dp[node][0] == 0) {
        dp[node][0] = dp[node][1] = 0;
        return;
      } else
        both++;
    } else if (dp[cur][0] == 1)
      zero++;
    else
      one++;
  }
  if (isleaf) {
    dp[node][1] = 0;
    dp[node][0] = 1;
  } else {
    dp[node][0] = dp[node][1] = 0;
    int edges = zero + one + both;
    ed[node] = edges;
    ze[node] = zero;
    on[node] = one;
    bo[node] = both;
    for (int i = 0; i <= 1; i++) {
      if ((i + edges) % 2 == 0) {
        dp[node][i] = (one % 2 == 0 || (one % 2 == 1 && both > 0));
      } else
        dp[node][i] = (one % 2 == 1 || (one % 2 == 0 && both > 0));
    }
  }
}
