(n, m) = map(int, input().split())
ar = [[] for i in range(n + 1)]
mx = [0] * (n + 1)
for i in range(m):
	(x, y) = map(int, input().split())
	mx[x] += 1
	mx[y] += 1
	ar[x].append(y)
	ar[y].append(x)
st = mx.index(max(mx))
used = [False] * (n + 1)
ans = []
q = []
used[st] = True

def bfs(st, tt=0):
	q.append(st)
	while tt < len(q):
		v = q[tt]
		for i in ar[v]:
			if not used[i]:
				used[i] = True
				q.append(i)
				ans.append((v, i))
		tt += 1
bfs(st)
for i in ans:
	print(' '.join(map(str, i)))
