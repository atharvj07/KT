for _ in range(int(input())):
	s = input()
	l = 'abcdefghijklmnopqrstuvwxyz'
	ans = 'YES'
	k = 1
	e = 'a'
	if 'a' in s:
		pass
	else:
		print('NO')
		continue
	st = len(s)
	for i in range(st - 1):
		n = len(e)
		ind = s.find(e)
		if ind == -1:
			ans = 'NO'
			break
		elif ind - 1 >= 0 and s[ind - 1] == l[k]:
			e = l[k] + e
		elif ind + n < st and s[ind + n] == l[k]:
			e = e + l[k]
		else:
			ans = 'NO'
			break
		k += 1
	print(ans)
