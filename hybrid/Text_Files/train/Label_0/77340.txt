(n, k) = map(int, input().split())
a = [0] * n
b = [0] * n
for i in range(n):
	(a[i], b[i]) = map(int, input().split())
base = sum(a) // k + sum(b) // k
A = sum(a) % k
B = sum(b) % k
data = [False for i in range(k)]
for i in range(n):
	ndata = [data[j] for j in range(k)]
	for j in range(1, k):
		if j <= a[i] and k - j <= b[i]:
			ndata[j] = True
			for x in range(1, k):
				prev = (x - j) % k
				if data[prev]:
					ndata[x] = True
	data = ndata
check = False
for i in range(1, k):
	if i <= A and k - i <= B:
		if data[i]:
			check = True
print(base + check)
