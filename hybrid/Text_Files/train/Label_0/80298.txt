import math
(n, a, b, k) = map(int, input().strip().split(' '))
x = list(map(int, input().strip().split(' ')))
y = []
count = 0
c = a + b
for i in x:
	if i % c == 0:
		y.append(math.ceil(b / a))
	else:
		r = i % c
		if r <= a:
			count += 1
		else:
			y.append(math.ceil((r - a) / a))
y.sort()
for i in y:
	if k >= i:
		count += 1
		k -= i
	if k == 0 or k < i:
		break
print(count)
