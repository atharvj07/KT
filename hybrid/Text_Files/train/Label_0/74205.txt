(s, b) = map(int, input().split())
spaceship = list(map(int, input().split()))
base = []
ans = []
for i in range(b):
	base.append([int(j) for j in input().split()])
base.sort()
gold = [base[0][1]]
for i in range(1, b):
	gold.append(gold[-1] + base[i][1])
for i in range(s):
	found = -1
	l = 0
	r = b - 1
	while l <= r:
		mid = l + (r - l) // 2
		if spaceship[i] >= base[mid][0]:
			found = mid
			l = mid + 1
		else:
			r = mid - 1
	if found == -1:
		ans.append(0)
	else:
		ans.append(gold[found])
print(*ans)
