    static int n;
    static ArrayList<Integer>[] g;
    static int[] color;

    static void solve() {
        n = in.nextInt();

        if (n % 2 == 0) {
            out.println("First");
            for (int i = 0; i < n * 2; i++)
                out.print((i % n + 1) + " ");
            out.println();
            return;
        }

        out.println("Second");
        out.flush();

        int[][] pairs = new int[n][2];
        for (int i = 0; i < n; i++)
            pairs[i][0] = pairs[i][1] = -1;
        for (int i = 0; i < n * 2; i++) {
            int p = in.nextInt() - 1;
            if (pairs[p][0] == -1)
                pairs[p][0] = i;
            else
                pairs[p][1] = i;
        }

        g = new ArrayList[n * 2]; Arrays.setAll(g, i -> new ArrayList<>());
        for (int i = 0; i < n; i++) {
            g[pairs[i][0]].add(pairs[i][1]);
            g[pairs[i][1]].add(pairs[i][0]);
        }
        for (int i = 0; i < n; i++) {
            g[i].add(i + n);
            g[i + n].add(i);
        }

        color = new int[n * 2]; Arrays.fill(color, -1);
        for (int i = 0; i < n * 2; i++) {
            if (color[i] == -1) {
                int u = i;
                color[u] = 0;
                while (true) {
                    boolean ok = false;
                    for (int v : g[u]) {
                        if (color[v] == -1) {
                            color[v] = color[u] ^ 1;
                            u = v;
                            ok = true;
                            break;
                        }
                    }
                    if (!ok)
                        break;
                }
            }
        }

        long sum0 = 0;
        for (int i = 0; i < n * 2; i++) {
            if (color[i] == 0)
                sum0 += i + 1;
        }

        if (sum0 % (n * 2) == 0) {
            for (int i = 0; i < n * 2; i++) {
                if (color[i] == 0)
                    out.print((i + 1) + " ");
            }
        } else {
            for (int i = 0; i < n * 2; i++) {
                if (color[i] == 1)
                    out.print((i + 1) + " ");
            }
        }
        out.println();


    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        String nextLine() {
            String s = "";
            try {
                s = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return s;
        }
    }

    static void dbg(Object... args) {
