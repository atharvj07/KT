const long long mn = 1000000;
vector<long long> cand = {50626, 6724, 82, 16, 4, 0};
long long mex(vector<long long> &v) {
  sort(v.begin(), v.end());
  long long ans = 0;
  for (auto &w : v) {
    if (w == ans) ans++;
  }
  return ans;
}
long long smart(long long x) {
  if (x >= mn) {
    long long x4 = round(pow((long double)x, 0.25L)) - 2;
    while (x4 * x4 * x4 * x4 < x) x4++;
    long long x2 = round(sqrt((long double)x)) + 2;
    while (x2 * x2 > x) x2--;
    long long lo = max(x4, 0ll), hi = min(x2, x - 1);
    vector<long long> v;
    for (auto &w : cand) {
      for (long long y = w - 1; y <= w + 1; y++) {
        if (lo <= y && y <= hi) {
          v.push_back(smart(y));
        }
      }
    }
    return mex(v);
  }
  if (x >= 50626) return 1;
  if (x >= 6724) return 3;
  if (x >= 82) return 0;
  if (x >= 16) return 2;
  if (x >= 4) return 1;
  return 0;
}
int main() {
  ios_base::sync_with_stdio(false);

  for (long long i = 0; i < n; i++) {

}
static char stdinBuffer[1024];
static char *stdinDataEnd = stdinBuffer + sizeof(stdinBuffer);
static const char *stdinPos = stdinDataEnd;
void readAhead(size_t amount) {
  size_t remaining = stdinDataEnd - stdinPos;
  if (remaining < amount) {
    memmove(stdinBuffer, stdinPos, remaining);
    size_t sz = fread(stdinBuffer + remaining, 1,
                      sizeof(stdinBuffer) - remaining, stdin);
    stdinPos = stdinBuffer;
    stdinDataEnd = stdinBuffer + remaining + sz;
    if (stdinDataEnd != stdinBuffer + sizeof(stdinBuffer)) *stdinDataEnd = 0;
  }
}
int rint() {
  readAhead(16);
  int x = 0;
  bool neg = false;
  while (*stdinPos == ' ' || *stdinPos == '\n') ++stdinPos;
  if (*stdinPos == '-') {
    ++stdinPos;
    neg = true;
  }
  while (*stdinPos >= '0' && *stdinPos <= '9') {
    x *= 10;
    x += *stdinPos - '0';
    ++stdinPos;
  }
  return neg ? -x : x;
}
