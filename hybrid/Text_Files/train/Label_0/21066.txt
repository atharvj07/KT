using namespace std;

#define FOR(i,s,e) for((i)=(s);(i)<(int)(e);(i)++)

typedef long long ll;
double EPS = 1e-6;
const double g = 9.8;
const int N = 50;
int n, v, X, Y;
int L[N], R[N], T[N], B[N];

//速度vyで上に打ち出した際のt秒後の位置を計算
double calc(double vy, double t) {
	return vy*t - g*t*t*0.5;
}

//aのlb,ubに対する相対位置関係(lbより下:-1, lb以上ub以下:0, ubより上:1)
int relPos(double lb, double ub, double a) {
	return a < lb+EPS ? -1 : (a > ub-EPS ? 1 : 0);
}

//点(x,y)を通るように打ち出したときに豚に卵をぶつけられるか判定
bool success(double x, double y) {
	double a = g*g*0.25, b = g*y-v*v, c = x*x+y*y;
	double D = b*b - a*c*4.0;

	if (D<0.0 && D>-EPS) D = 0.0;
	if (D<0.0) return false;

	for (int d=-1; d<=1; d+=2) {
		double t2 = (-b + d*sqrt(D)) / (a*2.0);
		if (t2<=0) continue;
		double t = sqrt(t2);
		double vx = x / t, vy = (y + g*t*t*0.5) / t;

		double yt = calc(vy, X / vx);
		if (yt < Y-EPS) continue;

		bool ok = true;
		for (int i=0; i<n; i++) {
			if (L[i]>=X) continue;

//			if (L[i]<=X && X<=R[i] && Y<=T[i] && B[i]<=yt) ok = false;
			if (X==R[i] && Y<=T[i] && B[i]<=yt) ok = false;

			int yL = relPos(B[i], T[i], calc(vy, L[i]/vx));
			int yR = relPos(B[i], T[i], calc(vy, R[i]/vx));
			int xH = relPos(L[i], R[i], vx*(vy/g));
			int yH = relPos(B[i], T[i], calc(vy, vy/g));

			if (xH==0 && yH>=0 && yL<0) ok = false;
			if (yL*yR<=0) ok = false;
		}
		if (ok) return true;
	}
	return false;
}

int main() {

	scanf("%d%d%d%d ", &n, &v, &X, &Y);
	for (int i=0; i<n; i++)
		scanf("%d%d%d%d ", L+i, B+i, R+i, T+i);

//
	for (int i=0; i<n; i++)
		R[i] = min(R[i], X);



	bool ok = success(X, Y);

	for (int i=0; i<n; i++) {
		ok |= success(L[i], T[i]);
		ok |= success(R[i], T[i]);
	}

	puts(ok ? "Yes" : "No");
    
	return 0;
}
