#include<bits/stdc++.h>
using namespace std;
 
int N,S,T;
int A[102][102];
int W[102][102];
int Q[102];
const int INF = (1<<29);
 
typedef pair<int,int> P;
const double EPS = 1e-10;
typedef vector<double> vec;
typedef vector<vec> mat; //Ax = b テ」ツ?ョAテ」ツ?ョテゥツδィテ・ツ按?

vec gauss_jordan(const mat &A, const vec &b){
  int n = A.size();
  mat B(n, vec(n+1));
  for(int i=0;i<n;i++)
    for(int j=0;j<n;j++)
      B[i][j] = A[i][j];
  for(int i=0;i<n;i++) B[i][n] = b[i];
  for(int i=0;i<n;i++){
    int pivot = i;
    for(int j=i+1;j<n;j++) if(fabs(B[j][i]) > fabs(B[pivot][i])) pivot = j;
    swap(B[i],B[pivot]);
    //ティツァツ」テ」ツ?古」ツ?ェテ」ツ??」ツ?凝」ツ??」ツ?づ」ツ?」テ」ツ?ヲテ」ツつづ、ツクツ?ヲツ?湘」ツ?ァテ」ツ?ッテ」ツ?ェテ」ツ??」ツ??
    if(fabs(B[i][i]) < EPS) return vec();
    
    for(int j=i+1;j<=n;j++) B[i][j] /= B[i][i];
    for(int j=0;j<n;j++)
      if(i != j)
	for(int k=i+1;k<=n;k++)
	  B[j][k] -= B[j][i] * B[i][k];
  }
  vec x(n);
  for(int i=0;i<n;i++) x[i] = B[i][n];
 
  return x;
}
 
void wsf(mat &M, vec &V){
  fill( W[0], W[N], INF );  
  for(int i=0;i<N;i++) W[i][i] = 0;
  for(int i=0;i<N;i++)
    for(int j=0;j<N;j++)
      if( A[i][j] ) W[i][j] = A[i][j];
  for(int i=0;i<N;i++)
    for(int j=0;j<N;j++)
      for(int k=0;k<N;k++)
        W[j][k] = min( W[j][k], W[j][i] + W[i][k] );


  M = mat(N,vec(N));
  V = vec(N);
  for(int i=0;i<N;i++){
    int k = 0;
    int cs = 0;
    if( i == T ){
      M[i][i] = 1;
      continue;
    } else if( Q[i] ){
      for(int j=0;j<N;j++)
        if( A[i][j] && W[i][T] == A[i][j] + W[j][T] ) {
          M[i][j] = -1;
          k++;
          cs += A[i][j];
        }
    } else {
      for(int j=0;j<N;j++)
        if( A[i][j] ) {
          M[i][j] = -1;
          k++;
          cs += A[i][j];
        }
    }
    M[i][i] = k;
    V[i] = cs;
  }  
}
 
 
int main(){
  while( cin >> N >> S >> T && (N|S|T) ){
    --S;--T;
    for(int i=0;i<N;i++) cin >> Q[i];
    for(int i=0;i<N;i++)
      for(int j=0;j<N;j++)
        cin >> A[i][j];
    mat M;
    vec V;
    wsf(M,V);
    if( W[S][T] == INF )
      cout << "impossible" << endl;
    else 
      printf("%.9lf\n",gauss_jordan(M,V)[S]);
  }
}
