import math
import copy
n = int(input())
s = input()
S = [i for i in s]
s1 = copy.deepcopy(S)
if n % 2 == 0:
	x = n // 2
	s1[x - 1] = S[0]
else:
	x = math.ceil(n / 2)
	s1[x - 1] = S[0]
n -= 1
a = 1
b = 2
c = 0
while n > 0:
	if n % 2 == 0:
		s1[x - b] = S[a]
		a += 1
		b += 1
		n -= 1
	else:
		s1[x + c] = S[a]
		a += 1
		n -= 1
		c += 1
print(''.join(s1))
