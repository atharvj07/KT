#include <bits/stdc++.h>
using namespace std;
const int MAXN = 110;
const int MAXM = 5500;
int N, M;
int u[MAXM], v[MAXM], c[MAXM];
int F[MAXN];
int Find(int x) { return F[x] = ((F[x] == x) ? x : Find(F[x])); }
double Tag[MAXN];
double fac[MAXN][MAXN], cns[MAXN];
int Z[MAXN], Z0;
void Gauss() {
  int i, j, k;
  for (i = 1; i <= N; i++) {
    for (j = Z0 + 1; j <= N; j++)
      if (abs(fac[j][i]) > 1e-9) break;
    if (j > N) continue;
    Z[++Z0] = i;
    for (k = 1; k <= N; k++) swap(fac[j][k], fac[Z0][k]);
    swap(cns[j], cns[Z0]);
    double rate = 1.0 / fac[Z0][i];
    for (j = 1; j <= N; j++) fac[Z0][j] *= rate;
    cns[Z0] *= rate;
    for (j = Z0 + 1; j <= N; j++) {
      rate = -fac[j][i];
      for (k = 1; k <= N; k++) fac[j][k] += fac[Z0][k] * rate;
      cns[j] += cns[Z0] * rate;
    }
  }
  for (i = Z0; i; i--) {
    Tag[Z[i]] = cns[i];
    for (j = 1; j <= i - 1; j++) {
      double rate = -fac[j][Z[i]];
      for (k = 1; k <= N; k++) fac[j][k] += rate * fac[i][k];
      cns[j] += rate * cns[i];
    }
  }
}
int main() {
  scanf("%d%d", &N, &M);
  int i, j;
  for (i = 1; i <= M; i++) scanf("%d%d%d", &u[i], &v[i], &c[i]);
  for (i = 1; i <= N; i++) F[i] = i;
  for (i = 1; i <= M; i++) F[Find(u[i])] = Find(v[i]);
  for (i = 1; i <= N; i++)
    if (Find(i) != Find(1)) Tag[i] = 0;
  if (Find(N) != Find(1)) {
    for (i = 0; i <= M; i++) puts("0");
    return 0;
  }
  for (i = 1; i <= N; i++)
    for (j = 1; j <= N; j++) fac[i][j] = 0;
  for (i = 1; i <= M; i++) {
    fac[u[i]][u[i]]++;
    fac[u[i]][v[i]]--;
    fac[v[i]][u[i]]--;
    fac[v[i]][v[i]]++;
  }
  for (i = 1; i <= N; i++) fac[1][i] = (i == 1) ? 1 : 0;
  for (i = 1; i <= N; i++) fac[N][i] = (i == N) ? 1 : 0;
  for (i = 1; i <= N; i++) cns[i] = (i == N) ? 1 : 0;
  Gauss();
  double rate = 1e+9;
  for (i = 1; i <= M; i++)
    if ((Find(1) == Find(u[i])) && (abs(Tag[u[i]] - Tag[v[i]]) > 1e-9))
      rate = min(rate, abs(c[i] / (Tag[u[i]] - Tag[v[i]])));
  double tot = 0;
  for (i = 1; i <= M; i++) {
    if (u[i] == 1) tot += (Tag[v[i]] - Tag[u[i]]) * rate;
    if (v[i] == 1) tot -= (Tag[v[i]] - Tag[u[i]]) * rate;
  }
  printf("%.8f\n", tot);
  for (i = 1; i <= M; i++) printf("%.8f\n", (Tag[v[i]] - Tag[u[i]]) * rate);
}
