import java.util.*;
public class Museum
{
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		
		int n = in.nextInt();
		int m = in.nextInt();
		int a = in.nextInt() - 1;
		int b = in.nextInt() - 1;
		
		ArrayList<Integer>[] adj = new ArrayList[n];
		for(int x = 0; x < adj.length; x++)
		{
			adj[x] = new ArrayList<Integer>();
		}
		
		for(int y = 0; y < m; y++)
		{
			int u = in.nextInt() - 1;
			int v = in.nextInt() - 1;
			adj[u].add(v);
			adj[v].add(u);
		}
		
		double[] prob = new double[n];
		for(int x = 0; x < n; x++)
		{
			prob[x] = in.nextDouble();
		}
		
		double[][] matrix = new double[n * n][n * n + n];
		for(int y = 0; y < n; y++)
		{
			for(int z = 0; z < n; z++)
			{
				if(y == z)
				{
					matrix[y * n + z][y * n + z] = 1.0;
					matrix[y * n + z][n * n + y] = 1.0;
				}
				else
				{
					matrix[y * n + z][y * n + z] = prob[y] * prob[z] - 1.0;
					
					for(int c : adj[y])
					{
						matrix[y * n + z][c * n + z] = ((1.0 - prob[y]) / adj[y].size()) * prob[z];
						
						for(int d : adj[z])
						{
							matrix[y * n + z][c * n + d] = ((1.0 - prob[y]) / adj[y].size()) * ((1.0 - prob[z]) / adj[z].size());
						}
					}
					
					for(int e : adj[z])
					{
						matrix[y * n + z][y * n + e] = prob[y] * ((1.0 - prob[z]) / adj[z].size());
					}
				}
			}
		}
		
		gaussJordan(matrix);
		
		for(int f = 0; f < n; f++)
		{
			if(f > 0)
			{
				System.out.print(" ");
			}
			
			System.out.print(matrix[a * n + b][n * n + f]);
		}
		
		System.out.println();
	}
	
	public static void gaussJordan(double[][] matrix)
	{
		int row = 0;
		for(int a = 0; a < matrix[0].length && row < matrix.length; a++)
		{
			double max = Double.NEGATIVE_INFINITY;
			int maxIndex = -1;
			for(int b = row; b < matrix.length; b++)
			{
				if(Math.abs(matrix[b][a]) > max)
				{
					max = Math.abs(matrix[b][a]);
					maxIndex = b;
				}
			}
			
			if(max < 1E-9)
			{
				continue;
			}
			
			double[] temp = matrix[row];
			matrix[row] = matrix[maxIndex];
			matrix[maxIndex] = temp;
			
			for(int c = 0; c < matrix.length; c++)
			{
				if(c != row)
				{
					double mult = -matrix[c][a] / matrix[row][a];
					for(int d = a; d < matrix[0].length; d++)
					{
						matrix[c][d] += mult * matrix[row][d];
					}
				}
			}
			
			double mult = 1 / matrix[row][a];
			for(int e = a; e < matrix[0].length; e++)
			{
				matrix[row][e] *= mult;
			}
			
			row++;
		}
	}
}
