N = int(200000.0 + 3)
primes = []
is_prime = [1] * N
is_prime[0] = is_prime[1] = 0
for i in range(2, N):
	if is_prime[i] == 0:
		continue
	primes.append(i)
	for j in range(i + i, N, i):
		is_prime[j] = 0
n = int(input())
arr = list(map(int, input().split()))
prt = [0] * n
for i in range(n):
	arr[i] -= 1
	prt[arr[i]] = i

def binary_search(k):
	(b, e) = (0, len(primes))
	while b <= e:
		mid = b + e >> 1
		if primes[mid] <= k:
			b = mid + 1
		else:
			e = mid - 1
	return b
res = []
p = 0
while p < n:
	while prt[p] != p:
		d = prt[p] - p + 1
		d = primes[binary_search(d) - 1] - 1
		r = prt[p]
		q = r - d
		(prt[arr[q]], prt[arr[r]]) = (prt[arr[r]], prt[arr[q]])
		(arr[q], arr[r]) = (arr[r], arr[q])
		res.append(str(q + 1) + ' ' + str(r + 1))
	p += 1
print(len(res))
print('\n'.join(res))
