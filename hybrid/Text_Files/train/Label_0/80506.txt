(n, p, k) = map(int, input().split())
arr = list(map(int, input().split()))
cnt = dict()

def f(a):
	return ((pow(a, 4, p) - k * a) % p + p) % p

def g(a):
	return a * (a - 1) // 2
res = 0
for a in arr:
	v = f(a)
	if v in cnt:
		cnt[v] += 1
	else:
		cnt[v] = 1
for a in arr:
	v = f(a)
	res += g(cnt[v])
	cnt[v] = 0
print(res)
