n = int(input())
a = [int(x) for x in input().split()]
even = []
odd = []
for i in range(n):
	if i % 2 == 0:
		even += [a[i]]
	else:
		odd += [a[i]]
pe = [0] * (len(even) + 1)
po = [0] * (len(odd) + 1)
for i in range(1, len(even) + 1):
	pe[i] = pe[i - 1] + even[i - 1]
for i in range(1, len(odd) + 1):
	po[i] = po[i - 1] + odd[i - 1]

def aa(i, j):
	if i % 2 == 0:
		se = pe[(i + 1) // 2] + po[-1] - po[i // 2]
		so = po[i // 2] + pe[-1] - pe[(i + 1) // 2] - a[j]
	else:
		se = pe[(i + 1) // 2] + po[-1] - po[i // 2] - a[j]
		so = po[i // 2] + pe[-1] - pe[(i + 1) // 2]
	return se == so
c = 0
for i in range(n):
	if aa(i, i):
		c += 1
print(c)
