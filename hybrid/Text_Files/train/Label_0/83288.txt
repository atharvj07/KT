(n, a, b) = map(int, input().split())
k = 0
r = 0
if a * b < n:
	print(-1)
else:
	for i in range(a):
		s = ''
		for j in range(b):
			if (r % b + i) % 2 == 0:
				q = 2 * k + 1
			else:
				q = 2 * k + 2
			if q > n:
				q = 0
			if j == 0:
				s += str(q)
			else:
				s += ' ' + str(q)
			r += 1
			if r % 2 == 0:
				k += 1
		print(s)
