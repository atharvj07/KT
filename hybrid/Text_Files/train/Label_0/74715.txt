s = input()
t = input()
visited = set()
array = []
ans = []
k = 0
for i in range(len(s)):
	a = s[i] in visited
	b = t[i] in visited
	if not a and (not b):
		visited.add(s[i])
		visited.add(t[i])
		array.append(s[i] + t[i])
		k += 1
		if s[i] != t[i]:
			ans.append(k - 1)
	elif a and b:
		if s[i] + t[i] in array or t[i] + s[i] in array:
			continue
		else:
			print(-1)
			exit(0)
	else:
		print(-1)
		exit(0)
print(len(ans))
for i in ans:
	print(array[i][0] + ' ' + array[i][1])
