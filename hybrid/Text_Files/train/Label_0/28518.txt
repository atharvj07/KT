#include <bits/stdc++.h>
using namespace std;
const int inf = 0x7ffffff;
const int maxn = 5000 + 5;
const int maxVerse = 12500000 + 5;
int dp[maxn][maxn], a[maxn], num[maxVerse];
int main() {
  int n, old;
  while (scanf("%d", &n) == 1) {
    memset(num, 0, sizeof(num));
    for (int i = 0; i < n; i++) {
      scanf("%d", &a[i]);
    }
    for (int j = 0; j < n; j++) {
      dp[0][j] = a[0] < j ? 1 : 0;
    }
    old = 0;
    for (int i = 1; i < n; i++) {
      for (int j = 0; j < n; j++) {
        dp[i][j] = dp[i - 1][j] + (a[i] < j ? 1 : 0);
      }
      old += (i - dp[i][a[i]]);
    }
    int _min = old, tmp = inf;
    for (int i = 0; i < n - 1; i++) {
      for (int j = i + 1; j < n; j++)
        if (a[i] > a[j]) {
          int p = dp[j - 1][a[i]] - dp[i][a[i]];
          int q = dp[j - 1][a[j]] - dp[i][a[j]];
          tmp = old - 2 * (p - q) - 1;
          if (tmp < _min) {
            _min = tmp;
          }
          if (tmp == _min) {
            num[_min]++;
          }
        }
    }
    printf("%d %d\n", _min, num[_min]);
  }
  return 0;
}
