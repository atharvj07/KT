

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.Map.Entry;
import java.util.Set;
import java.util.TreeMap;

public class Thwarting {
    int maxn = 100010;

    class IndexTree {
        int ss[] = new int[maxn], n;
        int lowbit(int k) {
            return (k & -k);
        }
        void inc(int i, int k) {
            while (i <= n) {
                ss[i] += k;
                i += lowbit(i);
            }
        }

        int get(int i) {
            int res = 0;
            while (i > 0) {
                res += ss[i];
                i -= lowbit(i);
            }
            return res;
        }
    }

    int n,hs[]=new int[maxn];
    long sum[] = new long[maxn],maxm=1000000000l*maxn;
    TreeMap<Long, Integer> mp = new TreeMap<Long, Integer>();
    IndexTree tree = new IndexTree();
    
    long work(long k) {
        for(int i=1;i<=n;i++)
            tree.inc(hs[i]=mp.get(sum[i]), 1);
        long res = 0, temp = 0;
        for (int i = 1; i <= n; i++) {
            Entry<Long,Integer> t = mp.ceilingEntry(k+temp);
            if (t != null) {
                int r =t.getValue();
                res += (n-i+1)-tree.get(r - 1);
            }
            temp=sum[i];
            tree.inc(hs[i], -1);
        }
        return res;
    }

    void run() throws IOException {
        n = nextInt();
        in.nextToken();
        long k = (long) in.nval;
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + nextInt();
            mp.put(sum[i], 0);
        }
        int cnt = 0;
        Set<Long> set = mp.keySet();
        for (long i : set)
            mp.put(i, ++cnt);
        tree.n = cnt;
        long left =-maxm,right=maxm, ans =maxm;
        while (left <= right) {
            long mid = (left + right) >> 1;
            if (work(mid) >= k) {
                ans = mid;
                left = mid + 1;
            } else
                right = mid - 1;
        }
        System.out.println(ans);
    }

    StreamTokenizer in = new StreamTokenizer(new BufferedReader(
            new InputStreamReader(System.in)));

    int nextInt() throws IOException {
        in.nextToken();
        return (int) in.nval;
    }

    public static void main(String[] args) throws IOException {
        new Thwarting().run();
    }
}
