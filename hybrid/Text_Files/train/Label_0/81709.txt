import bisect
import math

def binsearch(arr, x):
	left = 0
	right = len(arr)
	big = 0
	small = 0
	while left < right:
		middle = int((left + right) / 2)
		if arr[middle] < x:
			left = middle + 1
			small += 1
		elif arr[middle] == x:
			left = middle + 1
		else:
			right = middle
			big += 1
	return (big, small)
	pass

def solve(n, x, pos):
	p = 10 ** 9 + 7
	arr = [int(x) for x in range(n)]
	(big, small) = binsearch(arr, pos)
	k1 = math.factorial(small) * math.comb(x - 1, small)
	k2 = math.factorial(big) * math.comb(n - x, big)
	per = math.factorial(n - (small + big + 1))
	return k1 % p * k2 % p * per % p % p
	pass
t = 1
ans = []
for i in range(t):
	s = [int(x) for x in input().split()]
	n = s[0]
	x = s[1]
	pos = s[2]
	ans.append(solve(n, x, pos))
for a in ans:
