#include <bits/stdc++.h>
using namespace std;
int n, k, q, m;
int a[100000 + 12][12];
bitset<(1 << 12)> f[100000 + 12];
pair<int, int> b[12];
int main() {
  cin >> n >> k >> q;
  for (int i = 0; i < k; i++)
    for (int j = 0; j < n; j++) scanf("%d", a[j] + i);
  m = k;
  for (int i = 0; i != (1 << k); i++)
    for (int j = 0; j != m; j++) f[j][i] = (i >> j) & 1;
  while (q--) {
    int t, x, y;
    scanf("%d%d%d", &t, &x, &y);
    x--;
    y--;
    if (t == 1) {
      for (int i = 0; i != (1 << k); i++) f[m][i] = f[x][i] & f[y][i];
      m++;
    } else if (t == 2) {
      for (int i = 0; i != (1 << k); i++) f[m][i] = f[x][i] | f[y][i];
      m++;
    } else if (t == 3) {
      for (int i = 0; i != k; i++) b[i] = make_pair(a[y][i], i);
      sort(b, b + k);
      int s, i;
      for (s = 0, i = 0; i != k; i++) {
        s += 1 << b[i].second;
        if (f[x][s]) break;
      }
      cout << b[i].first << endl;
    } else
      cout << "error!" << endl;
  }
  return 0;
}
