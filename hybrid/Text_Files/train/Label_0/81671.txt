from collections import deque
import heapq

def inp():
	n = int(input())
	reqs = [0 for _ in range(n + 1)]
	affects = [[] for _ in range(n + 1)]
	for i in range(1, n + 1):
		temp = list(map(int, input().split()))
		if temp[0] == 0:
			continue
		reqs[i] = temp[0]
		for chapter in temp[1:]:
			affects[chapter].append(i)
	return (reqs, affects, n)

def solve(reqs, affects, n):
	to_read = [i for i in range(1, n + 1) if not reqs[i]]
	heapq.heapify(to_read)
	chapters_read = 0
	count = 0
	cur = 0
	while to_read:
		next_read = []
		while to_read:
			next_chapter = heapq.heappop(to_read)
			reqs[next_chapter] = -1
			chapters_read += 1
			for affected in affects[next_chapter]:
				reqs[affected] -= 1
				if not reqs[affected] and affected > next_chapter:
					heapq.heappush(to_read, affected)
				elif not reqs[affected]:
					next_read.append(affected)
		count += 1
		to_read = next_read
		heapq.heapify(to_read)
	if chapters_read == n:
		return count
	else:
		return -1
t = int(input())
for _ in range(t):
	(reqs, affects, n) = inp()
	ans = solve(reqs, affects, n)
	print(ans)
