from itertools import accumulate
from collections import defaultdict
for _ in range(int(input())):
	(a, b) = map(int, input().split())
	s = input()
	p = list(map(int, input().split()))
	k = [0] * a
	for x in p:
		if x >= a:
			k[0] += 1
		else:
			k[0] += 1
			k[x] -= 1
	k1 = list(accumulate(k))
	d = defaultdict(int)
	for x in range(a):
		d[s[x]] += k1[x] + 1
	qw = [0] * 26
	for x in range(97, 123):
		if chr(x) in d:
			qw[x - 97] = d[chr(x)]
	print(*qw)
