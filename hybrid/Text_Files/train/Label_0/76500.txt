(n, m, k) = [int(x) for x in input().strip().split(' ')]
pricies = [int(x) for x in input().strip().split(' ')]
m -= 1

def get_price(j):
	if j < 0 or j >= len(pricies) or pricies[j] == 0:
		return k + 1
	else:
		return pricies[j]
i = 1
while True:
	if k >= get_price(m + i) or k >= get_price(m - i):
		print(10 * i)
		break
	i += 1
