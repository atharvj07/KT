#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n;
  cin >> n;
  vector<int> v(n);
  map<int, int> s;
  for (size_t i = 0; i < n; i++) {
    cin >> v[i];
    s[i] = v[i];
  }
  auto buf = s.begin();
  for (auto it = ++s.begin(); it != s.end(); it++) {
    if (buf->second == it->second) {
      buf->second++;
      if (buf != s.begin()) buf--;
      s.erase(it);
      it = buf;
    } else {
      buf = it;
    }
  }
  cout << s.size() << "\n";
  for (auto it = s.begin(); it != s.end(); it++) {
    cout << it->second << " ";
  }
}
