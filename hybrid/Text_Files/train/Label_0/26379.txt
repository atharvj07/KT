#include <bits/stdc++.h>
using namespace std;
int main() {
  int x, y, z;
  while (cin >> x >> y) {
    if (y > x)
      cout << "-1" << endl;
    else {
      if (x % 2 == 0)
        z = x / 2;
      else
        z = x / 2 + 1;
      while (z % y != 0) {
        z++;
      }
      cout << z << endl;
    }
  }
  return 0;
}
