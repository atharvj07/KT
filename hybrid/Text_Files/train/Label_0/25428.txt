#include <bits/stdc++.h>
using namespace std;
int main(int argc, char* argv[]) {
  int n;
  cin >> n;
  std::vector<int> v;
  std::copy_n(std::istream_iterator<int>(std::cin), n, std::back_inserter(v));
  int len = 1;
  int max = 1;
  for (int i = 1; i < v.size(); i++) {
    if (v[i] > v[i - 1]) {
      len++;
    } else {
      if (len > max) max = len;
      len = 1;
    }
  }
  if (len > max) max = len;
  if (v.size() == 0) max = 0;
  cout << max << '\n';
  return 0;
}
