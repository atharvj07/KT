import java.util.*;
import java.io.*;

public class Main{

    BufferedReader in;
    StringTokenizer str = null;
    PrintWriter out;

    private String next() throws Exception{
    	while (str == null || !str.hasMoreElements())
    	    str = new StringTokenizer(in.readLine());
    	return str.nextToken();
    }
    
    private int nextInt() throws Exception{
	return Integer.parseInt(next());
    }
    
    final static int oo = Integer.MAX_VALUE/2;
    
    int []a,b,c;
    int [][][][]dp;
    String [][][][]froma, fromb;
    
    public void run() throws Exception{
    	in = new BufferedReader(new InputStreamReader(System.in));
    	out = new PrintWriter(System.out);
    	
    	String given = next();
    	given = given.replace("+", " ");
    	given = given.replace("=", " ");
    	
    	String []s = given.split(" ");
    	a = build(s[0].toCharArray());
    	b = build(s[1].toCharArray());
    	c = build(s[2].toCharArray());
    	
    	dp = new int[a.length+1][b.length+1][c.length][2];
    	froma = new String[a.length+1][b.length+1][c.length][2];
    	fromb = new String[a.length+1][b.length+1][c.length][2];
    	for(int i=0;i<=a.length;i++){
    	    for(int j=0;j<=b.length;j++){
    		for(int k=0;k<c.length;k++){
    		    Arrays.fill(dp[i][j][k], oo);
    		    Arrays.fill(froma[i][j][k], "");
    		    Arrays.fill(fromb[i][j][k], "");
    		}
    	    }
    	}
    	dp[0][0][0][0] = 0;
    	for(int i=0;i<=a.length;i++){
    	    for(int j=0;j<=b.length;j++){
    		for(int k=0;k<c.length;k++){
    		    for(int carry=0;carry<2;carry++){
    			if (dp[i][j][k][carry] == oo) continue;
    			if (i == a.length-1){
    			    if (dp[i+1][j][k][carry] > dp[i][j][k][carry]){
    				dp[i+1][j][k][carry] = dp[i][j][k][carry];
    				froma[i+1][j][k][carry] = froma[i][j][k][carry];
    				fromb[i+1][j][k][carry] = fromb[i][j][k][carry];
    			    }
    			}
    			if (j == b.length-1){
    			    if (dp[i][j+1][k][carry] > dp[i][j][k][carry]){
    				dp[i][j+1][k][carry] = dp[i][j][k][carry];
    				froma[i][j+1][k][carry] = froma[i][j][k][carry];
    				fromb[i][j+1][k][carry] = fromb[i][j][k][carry];
    			    }
    			}
    			
    			for(int da=0;da<((i < a.length) ? 10 : 1);da++){
    			    for(int db=0;db<((j < b.length) ? 10 : 1);db++){
    				int ni = i + ((i < a.length && a[i] == da) ? 1 : 0);
    				int nj = j + ((j < b.length && b[j] == db) ? 1 : 0);
    				int nk = k + ((k < c.length && (da + db + carry) % 10 == c[k]) ? 1 : 0);
    				int ncarry = (da + db + carry) / 10;
    				//System.out.println("about to update " + ni + " " + nj + " " + nk + " " + ncarry);
    				int value = dp[i][j][k][carry] + (i < a.length ? 1 : 0) + (j < b.length ? 1 : 0) + 1;
    				if (dp[ni][nj][nk][ncarry] > value){
    				    //System.out.println("updated " + ni + " " + nj + " " + nk + " " + ncarry);
    				    dp[ni][nj][nk][ncarry] = value;
    				    froma[ni][nj][nk][ncarry] = froma[i][j][k][carry];
    				    fromb[ni][nj][nk][ncarry] = fromb[i][j][k][carry];
    				    if (i < a.length) froma[ni][nj][nk][ncarry]+=Integer.toString(da);
    				    if (j < b.length) fromb[ni][nj][nk][ncarry]+=Integer.toString(db);
    				}
    			    }
    			}
    		    }
    		}
    	    }
    	}
    	
    	//System.out.println(dp[a.length][b.length][c.length-1][0]);
    	
    	String A = new StringBuilder(froma[a.length][b.length][c.length-1][0]).reverse().toString();
    	String B = new StringBuilder(fromb[a.length][b.length][c.length-1][0]).reverse().toString();
    	
    	int aa = Integer.parseInt(A), bb = Integer.parseInt(B);
    	out.println(aa + "+" + bb + "=" + (aa+bb));
        out.close();
    }
    
    private int[] build(char []a){
	int n = a.length;
	int []ret = new int[n+1];
	for(int i=0;i<n;i++){
	    ret[i] = a[n-i-1] - '0';
	}
	ret[n] = -1;
	return ret;
    }
    
    public static void main(String args[]) throws Exception{
	new Main().run();
    }
}
