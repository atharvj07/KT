for tc in range(int(input())):
	n = int(input())
	A = input()
	B = input()
	steps = []
	for i in range(n):
		if i % 2 == 0:
			if A[int(i / 2)] != B[n - 1 - i]:
				steps.append(n - i)
			else:
				steps.append(1)
				steps.append(n - i)
		elif A[n - 1 - int(i / 2)] == B[n - 1 - i]:
			steps.append(n - i)
		else:
			steps.append(1)
			steps.append(n - i)
	print(len(steps), end=' ')
	for i in steps:
		print(i, end=' ')
	print()
