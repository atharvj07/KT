from sys import stdin
from collections import Counter
(a, b, c) = map(int, stdin.readline().split())
k = [0] * (a + 1)
k[1] = 1
d = Counter(list(map(int, stdin.readline().split())))
if d[1]:
	exit(print(1))
for _ in ' ' * c:
	(e, f) = map(int, stdin.readline().split())
	(k[e], k[f]) = (k[f], k[e])
	if k[e] == 1 and d[e]:
		exit(print(e))
	if k[f] == 1 and d[f]:
		exit(print(f))
print(k.index(1))
