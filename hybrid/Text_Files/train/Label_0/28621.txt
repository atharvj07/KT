int getInt(){
  int ret = 0,c;
  c = getchar();
  while(!isdigit(c)) c = getchar();
  while(isdigit(c)){
    ret *= 10;
    ret += c - '0';
    c = getchar();
  }
  return ret;
}

int solve(string s, int n, int i, int g){
  int m = s.size();
  int now = 0;
  REP(j,m-1){
    now = (now + n + (j == i ? 1 : 0)) % s.size();
    if(s[now] == 'G') g--;
    s.erase(s.begin() + now);
    if(g == 0) return s.size();
    now %= s.size();
  }
  return s[0] == 'B' ? 1 : 0;
}

int main(){
  int cc; cin>>cc;
  while( cc --> 0 ){
    int n; string str;
    cin>>n>>str;

    const int m = str.size();
    int ans = 0;
    int g = count(ALL(str), 'G');

    REP(i,m) ans = max(ans, solve(str, n-1, i, g));

    print(ans);
  }
  return 0;
}
