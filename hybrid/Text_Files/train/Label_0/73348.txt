n = int(input())
A = list(map(int, input().split()))
A.pop(0)
B = list(map(int, input().split()))
B.pop(0)
count = 0
while len(A) != 0 and len(B) != 0 and (count <= 10000):
	if A[0] < B[0]:
		B.append(A[0])
		B.append(B[0])
		B.pop(0)
		A.pop(0)
		count = count + 1
	elif A[0] > B[0]:
		A.append(B[0])
		A.append(A[0])
		A.pop(0)
		B.pop(0)
		count = count + 1
if count >= 10000:
	print(-1)
elif len(A) == 0:
	print(count, end=' ')
	print(2)
else:
	print(count, end=' ')
	print(1)
