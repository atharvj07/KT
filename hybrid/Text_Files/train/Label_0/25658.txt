#include <bits/stdc++.h>
using namespace std;
#pragma GCC push_options
#pragma GCC optimize("unroll-loops")
long long dfs1(vector<long long> adj[], long long dp[], long long ar[],
               long long u, long long par, long long height) {
  long long ans = ar[u] * height;
  dp[u] = ar[u];
  for (long long x : adj[u]) {
    if (x != par) {
      ans += dfs1(adj, dp, ar, x, u, height + 1);
      dp[u] += dp[x];
    }
  }
  return ans;
}
long long ans, ans1;
void dfs2(vector<long long> adj[], long long dp[], long long u, long long par) {
  ans = max(ans, ans1);
  for (long long x : adj[u]) {
    if (x != par) {
      ans1 -= dp[x];
      dp[u] -= dp[x];
      ans1 += dp[u];
      dp[x] += dp[u];
      dfs2(adj, dp, x, u);
      dp[x] -= dp[u];
      ans1 -= dp[u];
      dp[u] += dp[x];
      ans1 += dp[x];
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long n;
  cin >> n;
  long long ar[n];
  for (int i = 0; i < n; i++) cin >> ar[i];
  vector<long long> adj[n];
  for (int i = 0; i < n - 1; i++) {
    long long u, v;
    cin >> u >> v;
    adj[u - 1].emplace_back(v - 1);
    adj[v - 1].emplace_back(u - 1);
  }
  long long dp[n];
  ans1 = dfs1(adj, dp, ar, 0, -1, 0);
  dfs2(adj, dp, 0, -1);
  cout << ans << "\n";
  return 0;
}
