(r, c) = list(map(int, input().split()))
l = [input() for i in range(r)]
for i in range(r):
	for j in range(c - 1):
		if l[i][j] == 'S' and l[i][j + 1] == 'W' or (l[i][j] == 'W' and l[i][j + 1] == 'S'):
			print('No')
			exit()
for j in range(c):
	for i in range(r - 1):
		if l[i][j] == 'S' and l[i + 1][j] == 'W' or (l[i][j] == 'W' and l[i + 1][j] == 'S'):
			print('No')
			exit()
print('Yes')
for i in range(r):
	for j in range(c):
		if l[i][j] == '.':
			print('D', end='')
		else:
			print(l[i][j], end='')
	print('')
