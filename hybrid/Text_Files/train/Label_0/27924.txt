#include <bits/stdc++.h>
using namespace std;

bitset <5000000> bt;
int a[2333],sum,n;

int main(){
  scanf("%d",&n);
  for (int i=1;i<=n;i++){
  	scanf("%d",&a[i]);
  	sum+=a[i];
  }
  bt[0]=1;
  for (int i=1;i<=n;i++)
    bt|=bt<<a[i];
  for (int i=sum/2;i>=0;i--)
    if (bt[i]){
      printf("%d\n",sum-i);
      return(0);
	}
}
