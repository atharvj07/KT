n = int(input())
weights = [int(i) for i in input().split()]
weights.sort(reverse=True)
possibleOutputs = set()
for x in range(2 * n):
	for y in range(x, 2 * n):
		if x != y:
			tempList = []
			for i in range(2 * n):
				if i != x:
					if i != y:
						tempList.append(weights[i])
			possibleOutputs.add(sum([tempList[2 * i] - tempList[2 * i + 1] for i in range(n - 1)]))
print(min(possibleOutputs))
