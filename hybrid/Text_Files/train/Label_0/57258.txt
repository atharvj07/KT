import java.util.*;
public class doublecola{
	public static void main(String[] args){
		Scanner br = new Scanner(System.in);
		int n = br.nextInt();
		String[] names = new String[5];
		names[0] = "Sheldon";
		names[1] = "Leonard";
		names[2] = "Penny";
		names[3] = "Rajesh";
		names[4] = "Howard";
		int pos = 0;
		int multi = 1;
		int num = 1;
		while(num < n){
			pos++;
			if(pos == 5){
				multi*=2;
				pos = 0;
			}
			num+=multi;
		}
		System.out.println(names[pos]);
	}
}
