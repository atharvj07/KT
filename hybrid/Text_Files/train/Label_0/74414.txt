t = int(input())
for u in range(t):
	n = int(input())
	ans = 1
	l = list(map(int, input().split()))
	f = 1
	for i in range(n):
		if i == 0:
			if l[i]:
				ans += 1
		else:
			if l[i] == 0 and l[i - 1] == 0:
				print(-1)
				f = 0
				break
			if l[i] == 1:
				if l[i - 1] == 1:
					ans += 5
				else:
					ans += 1
	if f:
		print(ans)
