#include <iostream>
#include <string>
#define llint long long

using namespace std;

llint n;
llint k[105];
string s;

int main(void)
{
	while(1){
		cin >> n;
		if(n == 0) break;
		for(int i = 0; i < n; i++) cin >> k[i];
		cin >> s;
		
		string ans;
		for(int i = 0; i < s.size(); i++){
			int c = s[i]-'a';
			if(s[i] < 'a') c = s[i]-'A'+26;
			c += 52-k[i%n];
			c %= 52;
			if(c < 26) ans += c+'a';
			else ans += (c-26)+'A';
		}
		cout << ans << endl;
	}
	return 0;
}
