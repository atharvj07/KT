#include<bits/stdc++.h>
#define N 5001
#define ll long long
using namespace std;
ll Sum[N][N],A[N],P[N],X,Y,Ans=0;
int n,m;
const ll Mo=998244353;
void Exgcd(int a,int b)
{
	if (!b)
	{
		X=1;Y=0;
		//cout<<a<<endl;
		return;
	}
	Exgcd(b,a%b);
	ll c=X;
	X=Y;Y=c-a/b*Y;
	//cout<<X<<' '<<Y<<endl;
}
ll C(int x,int y)
{

	return A[x]*P[y]%Mo*P[x-y]%Mo;
}
int main()
{
	A[0]=1;
	for (ll i=1;i<=5000;++i) A[i]=A[i-1]*i%Mo;
	Exgcd(A[5000],Mo);
	//cout<<X*A[2030]%Mo<<endl;
	P[5000]=(X+Mo)%Mo;P[0]=1;
	for (ll i=4999;i;--i) P[i]=P[i+1]*(i+1LL)%Mo;
	//cout<<P[2000]*A[2000]%Mo<<endl; 
	scanf("%d %d",&n,&m);
	for (int i=n;i;--i)
	{
		for (int j=1;j<m;++j) Sum[i][j]=(Sum[i+1][j]+C(i+m-j-2,m-j-1)*C(n-i+m-j,m-j)%Mo)%Mo;
	}
	for (int i=1;i<n;++i)
	{
		for (int h=1;h<m;++h) Ans=(Ans+C(n-i+h-1,h-1)*C(i+h-1,h)%Mo*Sum[i+1][h]%Mo)%Mo;
	}
	//cout<<C(1,0)<<endl;
	//cout<<Ans<<endl;
	swap(n,m);
	memset(Sum,0,sizeof(Sum));
	for (int i=n;i;--i)
	{
		for (int j=1;j<m;++j) Sum[i][j]=(Sum[i+1][j]+C(i+m-j-2,m-j-1)*C(n-i+m-j,m-j)%Mo)%Mo;
	}
	for (int i=1;i<n;++i)
	{
		for (int h=1;h<m;++h) Ans=(Ans+C(n-i+h-1,h-1)*C(i+h-1,h)%Mo*Sum[i+2][h]%Mo)%Mo;
	}
	
	Ans=Ans*2LL%Mo;
	printf("%lld",Ans);
}
