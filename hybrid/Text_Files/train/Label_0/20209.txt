#include <bits/stdc++.h>
using namespace std;
double inf = 1e10, eps = 1e-8;
int n, m;
int mn[200][200];
struct edge {
  int u, v, w;
} e[10000];
double a[200][200], x[200];
void init() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= (m); ++i) {
    scanf("%d%d%d", &e[i].u, &e[i].v, &e[i].w);
    ++mn[e[i].u][e[i].v], ++mn[e[i].v][e[i].u];
  }
}
void build() {
  a[1][1] = 1;
  for (int i = (2); i <= (n - 1); ++i) {
    int t(0);
    for (int j = 1; j <= (n); ++j) {
      a[i][j] = mn[i][j];
      t += mn[i][j];
    }
    a[i][i] = -t;
  }
}
void solve() {
  for (int i = (1); i <= (n - 1); ++i) {
    int t(i);
    for (int j = (i + 1); j <= (n - 1); ++j)
      if (abs(a[j][i]) > abs(a[t][i])) t = j;
    for (int k = (i); k <= (n); ++k) swap(a[i][k], a[t][k]);
    if (abs(a[i][i]) < eps) continue;
    for (int k = (i + 1); k <= (n); ++k) a[i][k] /= a[i][i];
    a[i][i] = 1;
    for (int j = (i + 1); j <= (n - 1); ++j) {
      for (int k = (i + 1); k <= (n); ++k) a[j][k] -= a[j][i] * a[i][k];
      a[j][i] = 0;
    }
  }
  for (int i = (n - 1); i >= (1); --i)
    if (abs(a[i][i]) > eps) {
      a[i][n] /= a[i][i];
      a[i][i] = 1;
      for (int j = (1); j <= (i - 1); ++j) {
        a[j][n] -= a[j][i] * a[i][n];
        a[j][i] = 0;
      }
    }
  for (int i = 1; i <= (n - 1); ++i) x[i] = -a[i][n];
  x[n] = 1;
}
void work() {
  double ans(inf), t(0);
  for (int i = 1; i <= (m); ++i) {
    if (x[e[i].u] < x[e[i].v])
      ans = min(ans, double(e[i].w) / (x[e[i].v] - x[e[i].u]));
    if (x[e[i].v] < x[e[i].u])
      ans = min(ans, double(e[i].w) / (x[e[i].u] - x[e[i].v]));
  }
  for (int i = 1; i <= (n); ++i) t += x[i] * mn[1][i];
  printf("%.8lf\n", ans * t);
  for (int i = 1; i <= (m); ++i)
    printf("%.8lf\n", ans * (x[e[i].v] - x[e[i].u]));
}
int main() {
  init();
  build();
  solve();
  work();
  return 0;
}
