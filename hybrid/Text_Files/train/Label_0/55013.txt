import java.util.*;
import java.io.*;
public class a 
{
	public static void main(String[] arg) throws IOException
	{
		new a();
	}
	public a() throws IOException
	{
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		int k = in.nextInt();
		double[] vs = new double[n]; for(int i = 0; i < n; i++) vs[i] = in.nextInt();
		double[] top = new double[n]; top[0] = vs[0];
		double[] bot = new double[n]; bot[0] = 1.0/vs[0];
		double[] total = new double[n]; total[0] = 1;
		for(int i = 1; i < n; i++)
		{
			top[i] = top[i-1]+vs[i];
			bot[i] = bot[i-1] + 1.0/vs[i];
			total[i] = total[i-1]+top[i]/vs[i];
		}
		double[][] memo = new double[k][n];
		memo[0][0] = 1;
		for(int i = 1; i < n; i++)
		{
			memo[0][i] = memo[0][i-1] + top[i]/vs[i];
		}
		for(int i = 1; i < k; i++)
		{
			MinConvexHull q = new MinConvexHull();
			q.add(-top[i-1], memo[i-1][i-1]-total[i-1]+top[i-1]*bot[i-1]);
			for(int j = i; j < n; j++)
			{
				memo[i][j] = q.query(bot[j])+total[j];
				q.add(-top[j], memo[i-1][j]-total[j]+top[j]*bot[j]);
			}
		}
		out.println(memo[k-1][n-1]);
		in.close(); out.close();
	}
	class MinConvexHull
	{
		ArrayList<Double> M;
		ArrayList<Double> B;
		int p;
		public MinConvexHull()
		{
			M = new ArrayList<Double>();
			B = new ArrayList<Double>();
			p = 0;
		}
		void add(double m, double b)
		{
			M.add(m);
			B.add(b);
			while(M.size() >= 3 && bad(M.size()-3, M.size()-2, M.size()-1))
			{
				M.remove(M.size()-2);
				B.remove(B.size()-2);
			}
		}
		double query(double x)
		{
			if(p >= M.size()) p = M.size()-1;
			while(p < M.size()-1 && M.get(p+1)*x+B.get(p+1) <= M.get(p)*x+B.get(p)) p++;
			return M.get(p)*x+B.get(p);
		}
		boolean bad(int l1, int l2, int l3)
		{
			return (double)(B.get(l3)-B.get(l1))/(double)(M.get(l1)-M.get(l3)) < (double)(B.get(l2)-B.get(l1))/(double)(M.get(l1)-M.get(l2));
			//return (B.get(l3)-B.get(l1))*(M.get(l1)-M.get(l2)) < (B.get(l2)-B.get(l1))*(M.get(l1)-M.get(l3));
		}
	}
	class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;
		public FastScanner(InputStream in)
		{
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer("");
		}
		public String next() throws IOException
		{
			while(!st.hasMoreTokens()) st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}
		public int nextInt() throws IOException
		{
			return Integer.parseInt(next());
		}
		void close() throws IOException
		{
			br.close();
		}
	}
}
