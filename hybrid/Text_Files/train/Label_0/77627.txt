t = int(input())
while t:
	t -= 1
	n = int(input())
	arr = map(int, input().split())
	arr = sorted(arr)
	alice = 0
	bob = 0
	alicekord = 1
	ctr = n - 1
	while ctr >= 0:
		if arr[ctr] % 2:
			if not alicekord:
				bob += arr[ctr]
		elif arr[ctr] % 2 == 0:
			if alicekord:
				alice += arr[ctr]
		alicekord = (alicekord + 1) % 2
		ctr -= 1
	if bob < alice:
		print('Alice')
	elif alice < bob:
		print('Bob')
	else:
		print('Tie')
