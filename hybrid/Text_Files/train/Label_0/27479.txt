#include <bits/stdc++.h>
using namespace std;
string s;
string arr[13];
bool fill(int x, int y) {
  bool res = false;
  for (int i = x; i < x + 3; i++)
    for (int j = y; j < y + 3; j++)
      if (arr[i][j] == '.') {
        arr[i][j] = '!';
        res = true;
      }
  return res;
}
int main() {
  ios::sync_with_stdio(false);
  int x, y;
  for (int i = 0; i < 11; i++) {
    getline(cin, arr[i]);
  }
  cin >> x >> y;
  x = (x - 1) % 3;
  y = (y - 1) % 3;
  if (fill(x * 3 + x, y * 3 + y) != true) {
    for (int i = 0; i < 11; i++) {
      if (arr[i].length() == 0) continue;
      for (int j = 0; j < 11; j++)
        if (arr[i][j] == '.') arr[i][j] = '!';
    }
  }
  for (int i = 0; i < 11; i++) cout << arr[i] << endl;
  return 0;
}
