#include <bits/stdc++.h>
using namespace std;
int a[11][4];
int b[11];
int c[11];
int ans[4];
int an[4];
char s[5];
bool v[10];
int main() {
  int i, j, n, x, flag, num = 0, k, nb, nc;
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%s%d%d", s, b + i, c + i);
    for (j = 0; j < 4; j++) {
      a[i][3 - j] = s[j] - '0';
    }
  }
  for (i = 123; i <= 9876; i++) {
    memset(v, 0, sizeof(v));
    flag = 1;
    x = i;
    for (j = 0; j < 4; j++) {
      ans[j] = x % 10;
      if (v[ans[j]]) {
        flag = 0;
        break;
      }
      v[ans[j]] = 1;
      x /= 10;
    }
    if (!flag) continue;
    for (j = 0; j < n; j++) {
      nb = nc = 0;
      for (k = 0; k < 4; k++) {
        if (a[j][k] == ans[k])
          nb++;
        else if (a[j][k] == ans[(k + 1) % 4] || a[j][k] == ans[(k + 2) % 4] ||
                 a[j][k] == ans[(k + 3) % 4]) {
          nc++;
        }
      }
      if (nb != b[j] || nc != c[j]) break;
    }
    if (j == n) {
      num += 1;
      for (k = 0; k < 4; k++) an[k] = ans[k];
    }
  }
  if (num == 0)
    printf("Incorrect data\n");
  else if (num > 1)
    printf("Need more data\n");
  else {
    for (k = 3; k >= 0; k--) printf("%d", an[k]);
    printf("\n");
  }
}
