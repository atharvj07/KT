import java.util.Scanner;

public class P103A {

    public static void main(String[] args) {
        Scanner inScanner = new Scanner(System.in);
        int n = inScanner.nextInt();
        long sum = 0;
        for (long i = 0; i < n; i++)
            sum += (i + 1) * inScanner.nextLong() - i;
        System.out.println(sum);
    }
}
