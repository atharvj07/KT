#include <bits/stdc++.h>
using namespace std;
bool checked(double m);
const int s = 1e5;
double n, k;
double arr[s];
int main() {
  double hi = 1e6, lo = 0, mid;
  scanf("%lf %lf", &n, &k);
  for (int i = 0; i < n; ++i) {
    scanf("%lf", &arr[i]);
  }
  for (int i = 0; i < 1000; ++i) {
    mid = (hi + lo) / 2;
    if (checked(mid))
      lo = mid;
    else
      hi = mid;
  }
  printf("%lf", lo);
}
bool checked(double m) {
  double more = 0, x, need = 0;
  for (int i = 0; i < n; ++i) {
    if (arr[i] > m) {
      x = (arr[i] - m);
      more += (x - ((x * k) / 100));
    } else
      need += m - arr[i];
  }
  if (more >= need) return 1;
  return 0;
}
