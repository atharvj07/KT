    long tree[];

        void build(int node,int l,int r){
        if(l==r){
            tree[node]=a[l]-l+1;

        }else {
            int mid = (l + r) / 2;
            build(2 * node, l, mid);
            build(2 * node + 1, mid+1, r);
            tree[node] = tree[2 * node] + tree[2 * node + 1];
        }
    }

    long sum(int x,int y,int id,int l,int r){
            if(l>r) return 0;
        if(y<l || x>r) return 0;
        if(l>=x && r<=y) return tree[id];
        int mid=(l+r)/2;
        return sum(x,y,2*id,l,mid)+sum(x,y,2*id+1,mid+1,r);
    }
    int upper_bound(int a[],int n,int x,int ll,int rr){
        int l=ll,h=rr;
        while(l<=h){
            int mid=(l+h)/2;

            if(x>=a[mid]) l=mid+1;
            else h=mid-1;
        }
        return l;
    }

    void dfs(int v,int pr){
       // pw.println(v);
        if(vis2[v]){
            int mn=Integer.MAX_VALUE,mx=-1;
            stk2.clear();
            while(true) {
                int u=stk.pop();

                mx = Math.max(mx, u);
                stk2.push(u);
                if(u==v) break;
            }
            while(!stk2.isEmpty()) stk.push(stk2.pop());

            if(mn!=Integer.MAX_VALUE)
            a[mn]=Math.min(a[mn],mx-1);
          //  if(v==7) pw.println(mn+" "+mx);
        }
        if(vis[v]) return;

        vis[v]=true;
        stk.push(v);
        for(int u : graph[v]){
            if(u!=pr) dfs(u,v);
        }
        stk.pop();
        vis2[v]=false;
    }

    private byte[] inbuf = new byte[1024];
    public int lenbuf = 0, ptrbuf = 0;

    private int readByte() {
        if(lenbuf == -1)throw new InputMismatchException();
        if(ptrbuf >= lenbuf){
            ptrbuf = 0;
            try { lenbuf = is.read(inbuf); } catch (IOException e) { throw new InputMismatchException(); }
            if(lenbuf <= 0)return -1;
        }
        return inbuf[ptrbuf++];
    }

    private boolean isSpaceChar(int c) { return !(c >= 33 && c <= 126); }
    private int skip() { int b; while((b = readByte()) != -1 && isSpaceChar(b)); return b; }

    private double nd() { return Double.parseDouble(ns()); }
    private char nc() { return (char)skip(); }

    private String ns() {
        int b = skip();
        StringBuilder sb = new StringBuilder();
        while(!(isSpaceChar(b))){ // when nextLine, (isSpaceChar(b) && b != ' ')
            sb.appendCodePoint(b);
            b = readByte();
        }
        return sb.toString();
    }

    private char[] ns(int n) {
        char[] buf = new char[n];
        int b = skip(), p = 0;
        while(p < n && !(isSpaceChar(b))){
            buf[p++] = (char)b;
            b = readByte();
        }
        return n == p ? buf : Arrays.copyOf(buf, p);
    }

    private char[][] nm(int n, int m) {
        char[][] map = new char[n][];
        for(int i = 0;i < n;i++)map[i] = ns(m);
        return map;
    }

    private int[] na(int n) {
        int[] a = new int[n];
        for(int i = 0;i < n;i++)a[i] = ni();
        return a;
    }

    private int ni() {
        int num = 0, b;
        boolean minus = false;
        while((b = readByte()) != -1 && !((b >= '0' && b <= '9') || b == '-'));
        if(b == '-'){
            minus = true;
            b = readByte();
        }

        while(true){
            if(b >= '0' && b <= '9'){
                num = num * 10 + (b - '0');
            }else{
                return minus ? -num : num;
            }
            b = readByte();
        }
    }
