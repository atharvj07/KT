(array_length, desired_median) = map(int, input().split())
array = list(map(int, input().split()))
array.sort()
solution = 0
median_pos = array_length // 2
for pos in range(median_pos - 1, -1, -1):
	if array[pos] > desired_median:
		solution += array[pos] - desired_median
	else:
		break
solution += abs(desired_median - array[median_pos])
for pos in range(median_pos + 1, array_length):
	if array[pos] < desired_median:
		solution += desired_median - array[pos]
	else:
		break
print(solution)
