#include <bits/stdc++.h>
#define REP(i, a, n) for (ll i = ((ll) a); i < ((ll) n); i++)
using namespace std;
typedef long long ll;

int main(void) {
  string S;
  cin >> S;
  ll N = S.length();

  bool ok = true;
  ok = ok && S[N - 1] == '0';
  ok = ok && S[N - 2] == '1';
  REP(i, 0, N - 1) ok = ok && S[i] == S[N - 2 - i];
  if (!ok) {
    cout << -1 << endl;
    return 0;
  }

  ll prev = 1, current = 2;
  REP(i, 0, N - 1) {
    cout << prev << " " << current << endl;
    if (S[i] == '1') prev = current;
    current++;
  }
}
