#include <bits/stdc++.h>
using namespace std;
const int M = 10 + 10;
char board[M][M];
char nextb[M][M];
int main() {
  for (int i = 0; i < 8; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < 8; j++) {
      board[i + 1][j + 1] = s[j];
    }
  }
  queue<pair<int, int> > valid_state;
  valid_state.push(make_pair(8, 1));
  int num = 0;
  while (!valid_state.empty()) {
    for (int i = 1; i <= 8; i++) {
      for (int j = 1; j <= 8; j++) {
        nextb[i][j] = '.';
      }
    }
    for (int i = 1; i < 8; i++) {
      for (int j = 1; j <= 8; j++) {
        if (board[i][j] == 'S') {
          nextb[i + 1][j] = 'S';
        }
      }
    }
    vector<pair<int, int> > q;
    while (!valid_state.empty()) {
      pair<int, int> t = valid_state.front();
      q.push_back(t);
      valid_state.pop();
    }
    for (int i = 0; i < q.size(); i++) {
      int x = q[i].first;
      int y = q[i].second;
      for (int r1 = -1; r1 <= 1; r1++) {
        for (int r2 = -1; r2 <= 1; r2++) {
          if (x + r1 >= 1 && x + r1 <= 8 && y + r2 >= 1 && y + r2 <= 8 &&
              board[x + r1][y + r2] != 'S' && nextb[x + r1][y + r2] != 'S') {
            valid_state.push(make_pair(x + r1, y + r2));
          }
        }
      }
    }
    for (int i = 1; i <= 8; i++) {
      for (int j = 1; j <= 8; j++) {
        board[i][j] = nextb[i][j];
      }
    }
    if (!valid_state.empty()) {
      num++;
    }
    if (num >= 8) {
      break;
    }
  }
  if (num >= 8) {
    cout << "WIN" << endl;
  } else {
    cout << "LOSE" << endl;
  }
  return 0;
}
