#include <bits/stdc++.h>
using namespace std;
int n, x[110], y[110], xcnt, ycnt;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    int u, v;
    scanf("%d%d", &u, &v);
    if (!x[u]) x[u] = 1, xcnt++;
    if (!y[v]) y[v] = 1, ycnt++;
  }
  printf("%d\n", min(xcnt, ycnt));
  return 0;
}
