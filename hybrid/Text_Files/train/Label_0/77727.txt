def max_subarray(numbers):
	best_sum = 0
	current_sum = 0
	for x in numbers:
		current_sum = max(0, current_sum + x)
		best_sum = max(best_sum, current_sum)
	return best_sum
t = int(input())
for _ in range(t):
	l = int(input())
	a = list(map(int, input().split()))
	z = max_subarray(map(lambda x: x[1] - x[0], zip(a[::2], a[1::2])))
	y = max_subarray(map(lambda x: x[1] - x[0], zip(a[2::2], a[1::2])))
	print(sum(a[::2]) + max(z, y))
