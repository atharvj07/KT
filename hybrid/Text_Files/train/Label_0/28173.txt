#include <bits/stdc++.h>
using namespace std;
int ans, n, v[10] = {1}, p1[10], p2[10], p[10], p3[10];
void dfs(int k) {
  if (k == ans && v[k] == n) {
    printf("%d\n", ans);
    for (int i = 0; i < k; i++) {
      if (p2[i] + 1)
        printf("lea e%cx, [e%cx + %d*e%cx]\n", p1[i] + 'a', p2[i] + 'a', p[i],
               p3[i] + 'a');
      else
        printf("lea e%cx, [%d*e%cx]\n", p1[i] + 'a', p[i], p3[i] + 'a');
    }
    exit(0);
  }
  if (k == ans) return;
  for (int i = 0; i <= k; i++)
    for (int j = -1; j <= k; j++)
      for (int u = 1; u < 9; u <<= 1) {
        if (j + 1)
          v[k + 1] = v[i] * u + v[j];
        else
          v[k + 1] = v[i] * u;
        if (v[k + 1] <= v[k] || v[k + 1] > n) continue;
        p1[k] = k + 1;
        p2[k] = j;
        p[k] = u;
        p3[k] = i;
        dfs(k + 1);
      }
}
int main() {
  scanf("%d", &n);
  for (ans = 0;; ans++) dfs(0);
}
