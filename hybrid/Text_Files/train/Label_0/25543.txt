  }
  cerr << " ]";

long long int calc(long long int n, vector<vector<long long int>>& dp) {
  dp[1][1] = mat[1][1];
  dp[2][1] = mat[2][1];
  for (long long int i = 2; i < n + 1; i += 1) {
    dp[0][i] = dp[1][i - 1];
    dp[3][i] = dp[2][i - 1];
    dp[1][i] = max(dp[2][i - 1], dp[3][i - 1]) + mat[1][i];
    dp[2][i] = max(dp[1][i - 1], dp[0][i - 1]) + mat[2][i];
  }
  return max({dp[0][n], dp[1][n], dp[2][n], dp[3][n]});
}
