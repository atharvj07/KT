import java.io.*;
import java.util.*;
import java.lang.*;
public class Solution {
    public static void main(String args[] ) throws Exception {
        InputReader fi= new InputReader(System.in);
        int n,k,i;
        n=fi.nextInt();
        int bravo;
        bravo=0;
        long ans=0;
        HashMap<Integer,Long> close,open;
        close=new HashMap<>();
        open=new HashMap<>();
        for (i=0;i<n;i++){
            char[] s=fi.readString().toCharArray();
            k=s.length;
            Stack<Character> st=new Stack<>();
            int o,c;
            o=c=0;
            for (int j=0;j<k;j++){
                    if (!st.isEmpty() && (st.peek()=='(' && s[j]==')')){
                       st.pop();
                       continue;
                   }
                   st.push(s[j]);
            }
            for (char x : st){
                if (x=='(') o++;
                if (x==')') c++;
            }
            //System.out.println(" st " +st + " o = " +o + " c = " +c);
            if (o==0 && c>0) {
                if (close.containsKey(c)){
                    close.put(c,close.get(c)+1);
                }
                else close.put(c,(long)1);
            }
             if (c==0 && o>0) {
                if (open.containsKey(o)){
                    open.put(o,open.get(o)+1);
                }
                else open.put(o,(long)1);
            }
            else if (c==0 && o==0) bravo++;
        }
        ans=0;
      /*  System.out.println(" open ");
       printMap(open);
       System.out.println(" close ");
       printMap(close);*/
        for (Map.Entry<Integer,Long> hm : open.entrySet()){
            if (close.containsKey(hm.getKey())){
                ans+=(long)hm.getValue()*(close.get(hm.getKey()));
            }


        }
        ans+=(long)bravo*bravo;
        System.out.println(ans);

    }
    static  void printMap(Map<Integer,Long> mp){
        for (Map.Entry<Integer,Long> hm : mp.entrySet()){
            System.out.println(" key " +hm.getKey() + " val " +hm.getValue());
        }
    }

}
class InputReader {
        private InputStream stream;
        private byte[] buf = new byte[8192];
        private int curChar, snumChars;
        private SpaceCharFilter filter;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        public int snext() {
            if (snumChars == -1)
                throw new InputMismatchException();
            if (curChar >= snumChars) {
                curChar = 0;
                try {
                    snumChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (snumChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public int nextInt() {
            int c = snext();
            while (isSpaceChar(c))
                c = snext();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = snext();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = snext();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public long nextLong() {
            int c = snext();
            while (isSpaceChar(c))
                c = snext();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = snext();
            }
            long res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = snext();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public int[] nextIntArray(int n) {
            int a[] = new int[n+1];
            for (int i = 1; i <= n; i++)
                a[i] = nextInt();
            return a;
        }

        public String readString() {
            int c = snext();
            while (isSpaceChar(c))
                c = snext();
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = snext();
            } while (!isSpaceChar(c));
            return res.toString();
        }

        public boolean isSpaceChar(int c) {
            if (filter != null)
                return filter.isSpaceChar(c);
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public interface SpaceCharFilter {
            public boolean isSpaceChar(int ch);
        }
    }
