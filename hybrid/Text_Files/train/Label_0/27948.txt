#include <bits/stdc++.h>
using namespace std;
long long put_10[200005], n;
long long solve(int i) {
  long long x = n - i;
  if (x == 0) return 10;
  if (x == 1) return 180;
  return (10 * 9 % 998244353 * put_10[x - 1] % 998244353 * 2 % 998244353 +
          10 * 9 % 998244353 * 9 % 998244353 * put_10[x - 2] % 998244353 *
              (x - 1) % 998244353) %
         998244353;
}
int main() {
  long long i;
  cin >> n;
  put_10[0] = 1;
  for (i = 1; i <= n; i++) put_10[i] = put_10[i - 1] * 10 % 998244353;
  for (i = 1; i <= n; i++) cout << solve(i) << " ";
  cout << "\n";
  return 0;
}
