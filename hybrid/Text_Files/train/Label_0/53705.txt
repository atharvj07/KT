import java.io.*; 
import java.util.*;
import java.math.*; 

public class Main {
	static int mex(HashSet<Integer> h) {
		for(int i=0;i<100005;i++) {
			if(!h.contains(i)) {
				return i;
			}
		}
		return -1;
	}
	public static void main(String[] args) throws IOException 
	{ 
		FastScanner f = new FastScanner(); 
		int t=1;
//		t=f.nextInt();
		PrintWriter out=new PrintWriter(System.out);
		while(t>0) {
			t--;
			int n=f.nextInt();
			int[] l=f.readArray(n);
			int mx=-1;
			int temp=-1;
			int curr=0;
			HashMap<Integer,Integer> h=new HashMap<>();
			for(int i=0;i<n;i++) {
				if(l[i]>temp) {
					temp=l[i];
					curr++;
				}
				else {
					mx=Math.max(mx,curr);
					curr=1;
					temp=l[i];
				}
			}
			mx=Math.max(mx,curr);
			temp=-1;
			curr=0;
			for(int i=n-1;i>-1;i--) {
				if(l[i]>temp) {
					temp=l[i];
					curr++;
				}
				else {
					mx=Math.max(mx,curr);
					curr=1;
					temp=l[i];
				}
			}
			mx=Math.max(mx,curr);
			int num=0;
			temp=-1;
			curr=0;
			for(int i=0;i<n;i++) {
				if(l[i]>temp) {
					temp=l[i];
					curr++;
				}
				else {
					if(curr==mx) {
						num++;
						if(h.containsKey(i-1)) {
							h.put(i-1, h.get(i-1)+1);
						}
						else {
							h.put(i-1, 1);
						}
					}
					curr=1;
					temp=l[i];
				}
			}
			if(curr==mx) {
				h.put(n-1,1);
				num++;
			}
//			System.out.println(h);
			temp=-1;
			curr=0;
			for(int i=n-1;i>-1;i--) {
				if(l[i]>temp) {
					temp=l[i];
					curr++;
				}
				else {
					if(curr==mx ) {
						num++;
						if(h.containsKey(i+1)) {
							h.put(i+1, h.get(i+1)+1);
						}
						else {
							h.put(i+1, 1);
						}
					}
					curr=1;
					temp=l[i];
				}
			}
//			System.out.println(mx+" "+num);
//			System.out.println(h);
			if(curr==mx) {
				num++;
				if(h.containsKey(0)) {
					h.put(0, h.get(0)+1);
				}
				else {
					h.put(0, 1);
				}
			}
			if(h.size()==1 ) {
				for(int i:h.keySet()) {
					if(h.get(i)==2 && mx%2==1) {
						System.out.println(1);
					}
					else {
						System.out.println(0);
					}
				}
			}
			else {
				System.out.println(0);
			}
		}
		out.close();
	} 
	static void sort(int [] a) {
        ArrayList<Integer> q = new ArrayList<>();
        for (int i: a) q.add(i);
        Collections.sort(q);
        for (int i = 0; i < a.length; i++) a[i] = q.get(i);
    }
    
	static class FastScanner {
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st=new StringTokenizer("");
		String next() {
			while (!st.hasMoreTokens())
				try {
					st=new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			return st.nextToken();
		}
		
		int nextInt() {
			return Integer.parseInt(next());
		}
		int[] readArray(int n) {
			int[] a=new int[n];
			for (int i=0; i<n; i++) a[i]=nextInt();
			return a;
		}
		long nextLong() {
			return Long.parseLong(next());
		}
		double nextDouble() {
			return Double.parseDouble(next());
		}
		long[] readLongArray(int n) {
			long[] a=new long[n];
			for (int i=0; i<n; i++) a[i]=nextLong();
			return a;
		}
	}
