#include <bits/stdc++.h>
using namespace std;
string s;
int le;
int dp1[102][102][2][55], dp2[102][102][2][55];
int fu1(int in, int di, int dir, int re) {
  if (re < 0) return -500;
  if (in == le - 1)
    if (re & 1)
      if (s[in] == 'F')
        return di;
      else
        return di + (dir ? 1 : -1);
    else if (s[in] == 'F')
      return di + (dir ? 1 : -1);
    else
      return di;
  int &x = dp1[in][di][dir][re];
  if (x != 500) return x;
  if (s[in] == 'F')
    x = max(fu1(in + 1, di, !dir, re - 1),
            fu1(in + 1, di + (dir ? 1 : -1), dir, re));
  else
    x = max(fu1(in + 1, di + (dir ? 1 : -1), dir, re - 1),
            fu1(in + 1, di, !dir, re));
  return x;
}
int main() {
  int i, j, k, l, n;
  for (i = 0; i < 101; i++)
    for (j = 0; j < 102; j++)
      for (k = 0; k < 2; k++)
        for (l = 0; l < 55; l++) dp1[i][j][k][l] = 500, dp2[i][j][k][l] = 500;
  cin >> s;
  le = s.size();
  scanf("%d", &n);
  int ma = max(fu1(0, 0, 0, n), fu1(0, 0, 1, n));
  printf("%d", ma);
  return 0;
}
