for _ in range(int(input())):
	n = int(input())
	v = [int(x) for x in input().split()]
	sm = sum(v)
	ans = 0

	def fn(target):
		sx = 0
		for x in v:
			sx += x
			if sx == target:
				sx = 0
			elif sx > target:
				return False
		return True
	for i in range(1, n + 1):
		if sm % i:
			continue
		if fn(sm // i):
			ans = i
	print(n - ans)
