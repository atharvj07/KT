import bisect
(n, k, A, B) = map(int, input().split())
X = sorted([int(a) - 1 for a in input().split()])

def calc(l, r, S):
	if len(S) == 0:
		ret = A
	elif r - l == 1:
		ret = B * len(S)
	else:
		m = (l + r) // 2
		SL = []
		SR = []
		for s in S:
			if s < m:
				SL.append(s)
			else:
				SR.append(s)
		if len(SL) == 0:
			ret = min(A + calc(m, r, SR), B * len(S) * (r - l))
		elif len(SR) == 0:
			ret = min(A + calc(l, m, SL), B * len(S) * (r - l))
		else:
			ret = calc(l, m, SL) + calc(m, r, SR)
	return ret
print(calc(0, 2 ** n, X))
