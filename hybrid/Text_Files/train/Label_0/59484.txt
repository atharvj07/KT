import java.util.Scanner;
public class CodeForces {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
	int n = sc.nextInt();
	int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < n; i++) {
            a[sc.nextInt()-1]++;
        }
        int ans = 0;
        for (int i = 0; i < k; i++) {
            if (a[i] % 2 == 1) {
                ans++;
            }
        }
        if (n%2==1) {
            ans--;
        }
        ans/=2;
        System.out.println(n-ans);
    }
}
