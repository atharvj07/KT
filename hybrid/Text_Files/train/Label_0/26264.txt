#include <bits/stdc++.h>
using namespace std;
long long int cons;
long long int check(long long int a) {
  if (a >= cons) a %= cons;
  return a;
}
long long int check2(long long int a) {
  if (a > 0) return a;
  long long int b = a / cons;
  a -= b * cons;
  if (a < 0) a += cons;
  return a;
}
long long int GCD(long long int a, long long int b) {
  if (b == 0) return a;
  return GCD(b, a % b);
}
long long int exp(long long int a, long long int n) {
  if (n == 0) return 1;
  if (n == 1) return check(a);
  long long int b = exp(a, n / 2);
  if (n % 2 == 0) return check(b * b);
  return check(b * check(b * a));
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cons = 1e9 + 7;
  int a, b, c;
  cin >> a >> b >> c;
  a %= b;
  int lim = 5e5, coun = 0;
  while (coun <= lim) {
    coun++;
    a = a * 10;
    int temp = a / b;
    if (temp == c) {
      cout << coun;
      return 0;
    }
    a %= b;
  }
  cout << -1;
}
