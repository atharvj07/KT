ans = 0
for i in range(8):
	v = input()
	while v == '':
		v = input()
	for j in v:
		if j == 'Q':
			ans = ans + 9
		if j == 'q':
			ans = ans - 9
		if j == 'R':
			ans = ans + 5
		if j == 'r':
			ans = ans - 5
		if j == 'B' or j == 'N':
			ans = ans + 3
		if j == 'b' or j == 'n':
			ans = ans - 3
		if j == 'P':
			ans = ans + 1
		if j == 'p':
			ans = ans - 1
if ans > 0:
	print('White')
if ans < 0:
	print('Black')
if ans == 0:
	print('Draw')
