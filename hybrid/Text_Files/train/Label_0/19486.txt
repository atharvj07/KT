#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m, s = 0;
  cin >> n >> m;
  vector<pair<pair<int, map<int, bool>>, int>> v(n);
  vector<map<int, bool>::iterator> pr;
  vector<vector<int>> ans;
  for (int i = 0; i < n; ++i) {
    int s1;
    cin >> s1;
    v[i].first.first = s1;
    v[i].second = i;
    s += s1;
    for (int j = 0; j < s1; ++j) {
      int a;
      cin >> a;
      v[i].first.second[a] = true;
    }
  }
  sort(v.begin(), v.end(),
       [](const pair<pair<int, map<int, bool>>, int> p1,
          const pair<pair<int, map<int, bool>>, int> p2) {
         return p1.first.first > p2.first.first;
       });
  int l = 0, r = n - 1;
  while (r != l) {
    if (v[l].first.first == s / n + (l < s % n)) {
      ++l;
      continue;
    }
    if (v[r].first.first == s / n + (r < s % n)) {
      --r;
      continue;
    }
    for (auto p = v[l].first.second.begin(); p != v[l].first.second.end();
         ++p) {
      int i = p->first;
      if (v[l].first.first == s / n + (l < s % n)) {
        for (auto p : pr) v[l].first.second.erase(p);
        pr.clear();
        ++l;
        break;
      }
      if (v[r].first.first == s / n + (r < s % n)) {
        for (auto p : pr) v[l].first.second.erase(p);
        pr.clear();
        --r;
        break;
      }
      if (v[r].first.second[i] == false) {
        --v[l].first.first;
        ++v[r].first.first;
        pr.push_back(p);
        v[r].first.second[i] = true;
        ans.push_back({v[l].second + 1, v[r].second + 1, i});
      }
    }
  }
  cout << ans.size() << '\n';
  for (auto i : ans) {
    for (int j : i) cout << j << ' ';
    cout << '\n';
  }
  return 0;
}
