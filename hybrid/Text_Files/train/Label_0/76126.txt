def f():
	input()
	(n, k) = map(int, input().split())
	visv = set([i - 1 for i in map(int, input().split())])
	g = [set() for i in range(n)]
	for i in range(n - 1):
		(a, b) = map(int, input().split())
		g[a - 1].add(b - 1)
		g[b - 1].add(a - 1)
	qv = set() | visv
	qh = set([0])
	vish = set()
	while 1:
		if not qh:
			return 'NO'
		nqh = set()
		for i in qh:
			vish.add(i)
			if len(g[i]) == 1 and i:
				return 'YES'
			for j in g[i]:
				if j not in vish and j not in visv:
					nqh.add(j)
		nqv = set()
		for i in qv:
			for j in g[i]:
				if j not in visv:
					nqv.add(j)
					visv.add(j)
		qh = nqh - nqv
		qv = nqv
for _ in range(int(input())):
	print(f())
