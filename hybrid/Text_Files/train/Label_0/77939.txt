t = eval(input())
while t is not 0:
	s = list(map(int, input().split()))
	n = list(map(int, input().split()))
	r = n[0] - s[0] + s[1]
	u = n[1] - s[2] + s[3]
	if (r >= n[2] and r <= n[4]) and (u >= n[3] and u <= n[5]):
		if n[2] == n[4] and (s[0] != 0 or s[1] != 0) or (n[3] == n[5] and (s[2] != 0 or s[3] != 0)):
			print('no')
		else:
			print('yes')
	else:
		print('no')
	t = t - 1
