def main():
	n = int(input())
	angles = [int(c) for c in input().split()]
	if n == 1:
		print(360)
		return
	angles = angles * 2
	angles.pop()
	ans = 360
	for k in range(n - 1):
		_sum = 0
		for i in range(k, k + n - 1):
			_sum += angles[i]
			(a, b) = (_sum, 360 - _sum)
			diff = max(a, b) - min(a, b)
			if diff < ans:
				ans = diff
	print(ans)
main()
