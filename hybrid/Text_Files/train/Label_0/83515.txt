sys.setrecursionlimit(99999)
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	s = input().strip('L')
	cw = s.count('W') + k
	t = []
	for (key, value) in itertools.groupby(s):
		if key == 'L':
			t.append(len(list(value)))
	t.sort()
	while t and t[0] <= k:
		k -= t.pop(0)
	print(max(0, 2 * min(n, cw) - len(t) - 1))
