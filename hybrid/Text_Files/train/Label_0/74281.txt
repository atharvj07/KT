input()
lista = input().split(' ')
lista2 = list(lista)
lista2 = [int(x) for x in lista2]

def bubbleSort(alist):
	for passnum in range(len(alist) - 1, 0, -1):
		for i in range(passnum):
			if alist[i] > alist[i + 1]:
				temp = alist[i]
				alist[i] = alist[i + 1]
				alist[i + 1] = temp
bubbleSort(lista2)
string = ''
for i in lista2:
	string = string + str(i) + ' '
print(string)
