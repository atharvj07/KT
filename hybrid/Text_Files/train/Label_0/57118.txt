import java.util.*;
import java.io.*;

public class Codeforces {
	static int P = 998244353;
	
	public static int Mul(int a, int b) {
		return (int)(1L * a * b % P);
	}
	
	public static int Add(int a, int b) {
		return a + b >= P ? a + b - P : a + b;
	}
	
	public static int Pow(int x, int t) {
		int res = 1;
		for (; t > 0; t >>= 1, x = Mul(x, x)) if ((t & 1) == 1) res = Mul(res, x);
		return res;
	}
	
	public static void main(String args[]) {		
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = in.nextInt();
		Integer[] cnt = new Integer[n + 5];
		for (int i = 1; i <= n; i++) cnt[i] = 0;
		for (int i = 0; i < n; i++) {
			int x = in.nextInt();
			cnt[x]++;
		}
		
		Integer[] inv = new Integer[n + 5];
		inv[0] = 1; inv[1] = 1;
		for (int i = 2; i <= n; i++) {
			inv[i] = Mul(inv[P % i], P - P / i);
		}
		
		Integer[][] f = new Integer[n + 5][n + 5];
		for (int i = 1; i <= n; i++) {
			f[1][i] = Mul(cnt[i], inv[n]);
		}
		for (int i = 2; i <= n; i++) {
			int sum = 0;
			for (int j = 1; j <= n; j++) {
				f[i][j] = Mul(cnt[j], sum);
				sum = Add(sum, Mul(f[i - 1][j], inv[n - i + 1]));
			}
		}
		
		int ans = 0;
		for (int i = 1; i <= n; i++) {
			for (int j = 1; j <= n; j++) {
				if (cnt[j] > 1) ans = Add(ans, Mul(Mul(f[i][j], cnt[j] - 1), inv[n - i]));
			}
		}
		System.out.println(ans);
	}
}
