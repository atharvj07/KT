#include <bits/stdc++.h>
using namespace std;
long long n, m, k, s, x;
long long a[200005], b[200005], c[200005], d[200005];
int biSearch(int value) {
  int left = 1, right = k, res = 0;
  while (left <= right) {
    int mid = (left + right) / 2;
    if (d[mid] <= value) {
      res = mid;
      left = mid + 1;
    } else
      right = mid - 1;
  }
  return res;
}
int main() {
  cin >> n >> m >> k;
  cin >> x >> s;
  for (int i = 1; i <= m; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  for (int i = 1; i <= k; i++) cin >> c[i];
  for (int i = 1; i <= k; i++) cin >> d[i];
  long long ans = (n - c[biSearch(s)]) * x;
  for (int i = 1; i <= m; i++) {
    if (b[i] <= s)
      ans = min(ans, (long long)(n - c[biSearch(s - b[i])]) * a[i]);
  }
  cout << ans << endl;
  return 0;
}
