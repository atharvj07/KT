def is_palindrome(s):
	s = list(s)
	s1 = s.copy()
	s1.reverse()
	return s == s1
s = input()
n = len(s)
flag = True
if is_palindrome(s):
	s1 = s[:n // 2]
	s2 = s[n // 2:]
	s = s1 + s[n // 2] + s2
else:
	l = 0
	r = n - 1
	while l < r:
		if s[l] == s[r]:
			l += 1
			r -= 1
		else:
			s1 = s[l + 1:r + 1]
			s2 = s[l:r]
			if is_palindrome(s1):
				pre = s[:l + 1]
				post = s[r + 1:]
				s = pre + s1 + s[l] + post
			elif is_palindrome(s2):
				pre = s[:l]
				post = s[r:]
				s = pre + s[r] + s2 + post
			else:
				flag = False
			break
if flag:
	print(s)
else:
	print('NA')
