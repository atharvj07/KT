#include <bits/stdc++.h>
typedef struct {
  int c;
  int d;
} abc;
abc p[71111];
int cmp(const void *a, const void *b) {
  if ((*(abc *)a).c == (*(abc *)b).c)
    return (*(abc *)a).d > (*(abc *)b).d ? 1 : -1;
  return (*(abc *)a).c > (*(abc *)b).c ? 1 : -1;
}
int main() {
  int x, y, a, b, i, j, k;
  while (scanf("%d%d%d%d", &x, &y, &a, &b) != EOF) {
    k = 0;
    for (i = b; i <= y; i++) {
      for (j = a; j <= x; j++) {
        if (j > i) {
          p[k].c = j;
          p[k].d = i;
          k++;
        }
      }
    }
    qsort(p, k, sizeof(abc), cmp);
    printf("%d\n", k);
    for (i = 0; i < k; i++) {
      printf("%d %d\n", p[i].c, p[i].d);
    }
  }
  return 0;
}
