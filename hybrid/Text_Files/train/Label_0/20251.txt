#define MOD 1000000007

typedef vector<vector<long>> mat;

// return A*B
mat mat_mul(const mat &A, const mat &B){
  int n=A.size(), m=B[0].size(), l=B.size();
  mat ret(n, vector<long>(m, 0));
  rep(i,n) rep(k,l) if(A[i][k]!=0) rep(j,m){
    (ret[i][j] += A[i][k] * B[k][j]) %= MOD;
  }
  return ret;
}

// A^p
mat mat_pow(const mat &A, long p){
  int n = A.size();
  mat tmp(A), ret(n, vector<long>(n,0));
  rep(i,n) ret[i][i] = 1;
  while(p>0){
    if(p&1) ret = mat_mul(tmp, ret);
    tmp = mat_mul(tmp, tmp);
    p /= 2;
  }
  return ret;
}

int main(){
  int n,k,c,T;
  cin>>n>>k>>c>>T;
  c--;

  vector<int> a(k), b(k), t(k);
  rep(i,k) cin>>a[i]>>b[i]>>t[i];
  rep(i,k) a[i]--;

  int sz = 5*n;

  mat A(sz, vector<long>(sz, 0));
  rep(i,n,sz){
    A[i-n][i] += 1;
  }

  rep(i,k){
    rep(j,n){
      int to = (t[i]-1)*n + j;
      if(j < a[i]) to += b[i];
      else if(j < a[i]+b[i]) to -= a[i];
      A[to][j] += 1;
    }
  }

  A = mat_pow(A, T);

  cout << A[0][c] << endl;

  return 0;
}
