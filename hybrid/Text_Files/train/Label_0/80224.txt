t = int(input())
for i in range(t):
	(n, q) = map(int, input().split())
	a = input()
	for k in range(q):
		(l, r) = map(int, input().split())
		l -= 1
		r -= 1
		d = 0
		y = r + 1
		e = True
		while d < l and e == True:
			if a[d] == a[l]:
				e = False
			d += 1
		while y < len(a) and e == True:
			if a[y] == a[r]:
				e = False
			y += 1
		print('NO' if e == True else 'YES')
