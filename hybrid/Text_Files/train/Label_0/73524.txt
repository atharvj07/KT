import sys
from math import gcd, sqrt
from typing import DefaultDict
sys.setrecursionlimit(10 ** 5)
inf = float('inf')
en = lambda x: list(enumerate(x))
ii = lambda : int(input())
r = lambda : map(int, input().split())
rr = lambda : list(r())

def solve():
	(n, m) = r()
	arr = rr()
	brr = rr()
	pos = {j: i for (i, j) in enumerate(arr)}
	x = 0
	ans = 0
	for (j, i) in enumerate(brr, 1):
		z = pos[i]
		if z < x:
			ans += 1
		elif z > x:
			ans += (z - j + 1) * 2 + 1
			x = z
		else:
			ans += 1
			x += 1
	print(ans)
for _ in range(ii()):
	solve()
