import java.util.Arrays;
import java.util.Scanner;
public class contest 
{
	public static void main(String[] args)
	{
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t>0)
		{
		int a[]=new int[3];
		a[0]=sc.nextInt();
		a[1]=sc.nextInt();
		a[2]=sc.nextInt();
		Arrays.sort(a);
		if(a[2]<=a[0]+a[1])
		{
			System.out.println((a[0]+a[1]+a[2])/2+"");
		}
		else
		{
			System.out.println(a[0]+a[1]+"");
		}
		t--;
		}
	}

}
