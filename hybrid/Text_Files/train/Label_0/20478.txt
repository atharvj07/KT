inline char gc(){
    static char buf[100000],*p1=buf,*p2=buf;
    return p1==p2&&(p2=(p1=buf)+fread(buf,1,100000,stdin),p1==p2)?EOF:*p1++;
}
#define gc getchar
inline ll read(){
	ll x = 0; char ch = gc(); bool positive = 1;
	for (; !isdigit(ch); ch = gc())	if (ch == '-')	positive = 0;
	for (; isdigit(ch); ch = gc())	x = x * 10 + ch - '0';
	return positive ? x : -x;
}

	}

}

const int N=7,M=1<<5|2,C=50;
ld f[M][N*C],g[M][N*C],h[N][M][N*C];
int l[N],s[M];
int main(){
	int n=read(),C=read();
	For(i,0,n-1)l[i]=read();
	sort(l,l+n); n--;
	f[0][l[n]*(n+1)]=1;
	For(i,1,(1<<n)-1)s[i]=s[i>>1]+(i&1);
	For(i,0,C-1){
		mem(h);
		For(j,0,(1<<n)-1){
			For(k,0,C*(n+1)){
				g[j][k]=f[j][k];
				h[k%(n+1)][j][k]=k/(n+1)==i?f[j][k]:0;
			}
		}
		For(j,0,n-1){
			Rep(zt,(1<<n)-1,0)if((zt>>j&1)==0){
				For(k,0,C*(n+1)){
					int P=k%(n+1),nm=(i+l[j])*(n+1);
					For(q,1,s[zt]+1){
						nm++;
						int jj=min(max(k+(q<=P&&zt),nm),C*(n+1));
						//if(i==0)cout<<k+(q<=P&&s[zt])<<" "<<nm<<" "<<g[zt][k]<<endl;
						g[zt|(1<<j)][jj]+=g[zt][k];
						For(p0,0,q-1){
							h[p0][zt|(1<<j)][jj]+=h[p0][zt][k];
						}
					}
				}
			}
		}
		For(j,0,(1<<n)-1){
			For(k,0,C*(n+1)){
				f[j][k]=g[j][k];
				For(p0,0,s[j]+1){
					f[j][k]-=h[p0][j][k];
				}
			}
		}
		//if(i==2)cout<<f[(1<<n)-1][6]<<" "<<h[2][0][2]<<endl;
	}
	ld ans=f[(1<<n)-1][C*(n+1)]/pow(C,n);
	For(i,1,n)ans/=i;
	printf("%.12f\n",ans);
}
