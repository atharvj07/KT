from collections import deque
for _ in range(int(input())):
	n = int(input())
	s = input()
	c1 = s.count('1')
	c = c1
	d = 0
	if s[0] == '1' and s[-1] == '1' and (s.count('0') % 2 == 0) and (c1 % 2 == 0):
		br1 = ['('] * n
		br2 = ['('] * n
		for k in range(n):
			if s[k] == '1':
				if c1 > c // 2:
					c1 -= 1
				else:
					br1[k] = ')'
					br2[k] = ')'
			else:
				if d % 2 == 0:
					br1[k] = '('
					br2[k] = ')'
				else:
					br2[k] = '('
					br1[k] = ')'
				d += 1
		print('YES')
		print(*br1, sep='')
		print(*br2, sep='')
	else:
		print('NO')
