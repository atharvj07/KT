#include <iostream>
using namespace std;
typedef long long ll;

ll EulerTotient(ll n) {
    ll res = n;
    for(int r = 2; r * r <= n; r++) {
        if(n % r == 0) {
            res -= res / r;
            while(n % r == 0) {
                n /= r;
            }
        }
        //cout << res << endl;
    }
    if(n > 1) res -= res / n;
    return(res);
}

int main(void) {
    ll num; cin >> num;
    
    cout << EulerTotient(num) << endl;
    
    return(0);
}
