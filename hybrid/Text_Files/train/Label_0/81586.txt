import sys
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
(n, q) = map(int, input().split())
A = list(map(int, input().split()))
B = list(map(int, input().split()))
C = [A[i] - B[i] for i in range(n)]
S = [0]
for c in C:
	S.append(S[-1] + c)
LEN = len(S) + 1
Sparse_table0 = [S]
for i in range(LEN.bit_length() - 1):
	j = 1 << i
	B = []
	for k in range(len(Sparse_table0[-1]) - j):
		B.append(min(Sparse_table0[-1][k], Sparse_table0[-1][k + j]))
	Sparse_table0.append(B)

def query0(l, r):
	i = (r - l).bit_length() - 1
	return min(Sparse_table0[i][l], Sparse_table0[i][r - (1 << i)])
Sparse_table1 = [S]
for i in range(LEN.bit_length() - 1):
	j = 1 << i
	B = []
	for k in range(len(Sparse_table1[-1]) - j):
		B.append(max(Sparse_table1[-1][k], Sparse_table1[-1][k + j]))
	Sparse_table1.append(B)

def query1(l, r):
	i = (r - l).bit_length() - 1
	return max(Sparse_table1[i][l], Sparse_table1[i][r - (1 << i)])
for queries in range(q):
	(l, r) = map(int, input().split())
	NOW = S[l - 1]
	MIN = query0(l, r + 1)
	MAX = query1(l, r + 1)
	if NOW == MAX and S[r] == NOW:
		print(MAX - MIN)
	else:
		print(-1)
