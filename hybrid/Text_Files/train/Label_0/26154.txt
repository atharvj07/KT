int d[15][15];
int INF = 1<<28;
int dp[1<<15][15];

int dist(string &a, string &b){
	string sub = "";
	int res = 0;
	rep(i, min(a.length(), b.length())){
		sub += b[i];
		string sa = a.substr(a.length() - i - 1);
		if(sa == sub) res = max(res, i+1);
	}
	return b.length() - res;
}

// is a in b ?
bool in(const string &a, const string &b){
	if(a.length() > b.length()) return false;
	rep(i, b.length() - a.length() + 1){
		if(b.substr(i, a.length()) == a) return true; 
	}
	return false;
}

int main(){
	int n;
	while(cin >> n, n){
		vector<string> v(n);
		rep(i, n){
			cin >> v[i];
		}
		bool f;
		do{
			f = false;
//			continue;
			rep(i, n){
				if(f) break;
				rep(j,n){
					if(i != j && in(v[i], v[j])){
						v.erase(v.begin() + i);
						n--;
						f=true;
						break;
					}
				}
			}
		}while(f);
		rep(i,n){
			rep(j,n) if(i != j){
				d[i][j] = dist(v[i], v[j]);
//				cout << d[i][j] << ", ";
			}
//			cout << endl;
		}
		fill(dp[0], &dp[(1<<15)-1][14], INF);
		rep(i, n){
//			cout << dp[(1<<n) - 1][i] << ", ";
			dp[1<<i][i] = v[i].length();
		}
		rep(i, 1<<n){
			rep(j, n){ // from
				rep(k, n){ // to
					if((i & (1<<k)) > 0) continue;
					dp[i | (1 << k)][k] = min(dp[i | (1<<k)][k], d[j][k] + dp[i][j]);
				}
			}
		}
		int ans = INF;
		rep(i, n){
			ans = min(ans, dp[(1<<n) - 1][i]);
//			cout << dp[(1<<n) - 1][i] << ", ";
		}

		cout << ans << endl;
	}
	return 0;
}
