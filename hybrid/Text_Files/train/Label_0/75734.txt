anos = {}
for _ in range(int(input())):
	(ini, fim) = [int(x) for x in input().split()]
	if ini in anos:
		anos[ini] += 1
	else:
		anos[ini] = 1
	if fim in anos:
		anos[fim] -= 1
	else:
		anos[fim] = -1
vivos = 0
maior = 0
anoMaior = 0
for (ano, saldo) in sorted(anos.items()):
	vivos += saldo
	if maior < vivos:
		maior = vivos
		anoMaior = ano
print(anoMaior, maior)
