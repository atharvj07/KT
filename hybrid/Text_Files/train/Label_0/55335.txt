public class cf1201b {
    public static void main(String[] args){
		Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
		long[] arr = new long[n];
		long sum = 0,max=0;
		for(int i = 0  ; i < n; i++){
			arr[i] =sc.nextLong();
			sum+=arr[i];
			max = Math.max(max,arr[i]);
		}
		if(sum%2==0 && max <= sum-max)
		{
			System.out.println("YES");
		}
		else{
			System.out.println("NO");
		}
    }
}
