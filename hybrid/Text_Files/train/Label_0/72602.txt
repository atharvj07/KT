def main():
	(l, cnt) = (list(map(ord, input())), [-1] * 91)
	for c in l[:25]:
		cnt[c] += 1
	(twins, cnt[63]) = (sum((i > 0 for i in cnt[65:])), -27)
	for (i, u, v) in zip(range(50000), l, l[25:]):
		cnt[v] = c = cnt[v] + 1
		if c == 1:
			twins += 1
		elif not twins:
			stuff = [j for j in range(90, 64, -1) if cnt[j]]
			while stuff:
				while l[i] != 63:
					i += 1
				l[i] = stuff.pop()
			return print(''.join(map(chr, l)).replace('?', 'A'))
		cnt[u] = c = cnt[u] - 1
		if not c:
			twins -= 1
	print(-1)
main()
