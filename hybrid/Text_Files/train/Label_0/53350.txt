import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner scanner = new Scanner(System.in);
		int n = scanner.nextInt(); //страницы все
		int p = scanner.nextInt(); //данная страница
		int k = scanner.nextInt();
		if ((p - k) > 1) {
			System.out.print("<< ");
		}
		int l = p - k;
		int m = p + k;
		for (int i = l; i < m + 1; i++) {
			if ((i == p) && (i > 0)) {
				System.out.print("(" + i + ") ");
				if (i == n) {
					break;
				}
			}
			if ((i != p) && (i > 0) && (i != n)) {
					System.out.print(i + " ");
			}
			if ((i == n)){
					System.out.print(i + " ");
					break;
			}
		}
			if ((p + k) < n) {
				System.out.println(">>");
			}
		}
	}
