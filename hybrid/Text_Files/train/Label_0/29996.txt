long long poww(const long long &a, long long b,
               const long long &m = 1000000007) {
  if (b == 0) return 1;
  long long first = poww(a, b / 2, m);
  first = first * first % m;
  if (b & 1) first = first * a % m;
  return first;
}
long long gcd(long long a, long long b) {
  if (a > b) swap(a, b);
  if (a == 0) return b;
  return gcd(b % a, a);
}
bool mod(long double a, long double b) { return a / b - floor(a / b); }

  long long a = 1;

  for (i = 0; i < n / 2; i++) {
    long long c = n / 2 - i;
    for (j = 0; j < 2 * i + 1; j++) {
      grid[i][c] = a;
      a += 2;
      grid[n - i - 1][c] = a;
      a += 2;
      c++;
    }
  }
  for (i = 0; i < n; i++) {
    grid[n / 2][i] = a;
    a += 2;
  }
  a = 2;
  for (i = 0; i < n; i++) {
    for (j = 0; j < n; j++) {
      if (grid[i][j] == -1) {
        cout << a << " ";
        a += 2;
      } else {

      }
    }
