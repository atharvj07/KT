#include <bits/stdc++.h>
using namespace std;
int n, k, a[1000], used, j;
bool mark[1000];
int main() {
  cin >> n >> k;
  for (int i = 0; i < k; i++) {
    cin >> a[i];
    mark[a[i]] = true;
  }
  j = 1;
  for (int i = 0; i < k; i++) {
    used = 1;
    while (used < n) {
      if (!mark[j]) {
        cout << j << " ";
        mark[j] = true;
        used++;
      }
      j++;
    }
    cout << a[i] << endl;
  }
}
