#include <bits/stdc++.h>
using namespace std;
long long int MOD = 1e9 + 7;
unordered_map<long long int, long long int> counter;
void primeFactors(long long int n) {
  while (n % 2 == 0) {
    counter[2]++;
    n = n / 2;
  }
  for (int i = 3; i <= sqrt(n); i = i + 2) {
    while (n % i == 0) {
      counter[i]++;
      n = n / i;
    }
  }
  if (n > 2) {
    counter[n]++;
  }
}
long long int exp(long long int n, long long int e) {
  if (e == 0) return 1;
  long long int temp = exp(n * n, e / 2);
  if (e % 2 == 1)
    return temp * n;
  else
    return temp;
}
int main() {
  long long int n, b;
  cin >> n >> b;
  primeFactors(b);
  unordered_map<long long int, long long int>::iterator it;
  long long int ans = 1e18;
  for (it = counter.begin(); it != counter.end(); it++) {
    long long int d = it->first;
    long long int temp = 0;
    for (int i = 1; i <= 60; i++) {
      long long int t = exp(d, i);
      temp += n / t;
      long long int t2 = (1e18) / t;
      if (t2 < d) break;
    }
    temp = temp / it->second;
    ans = min(ans, temp);
  }
  cout << ans << endl;
  return 0;
}
