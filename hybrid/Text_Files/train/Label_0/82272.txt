t = input()
b = True
for i in range(len(t) - 1):
	if t[i] == t[i + 1] and t[i] != '?':
		b = False
		break
if not b:
	print('No')
else:
	b = False
	if t[0] == '?' or t[len(t) - 1] == '?':
		b = True
	for i in range(1, len(t) - 1):
		if t[i] == '?' and (t[i - 1] == t[i + 1] or t[i - 1] == '?' or t[i + 1] == '?'):
			b = True
	if b:
		print('Yes')
	else:
		print('No')
