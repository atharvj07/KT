#include <bits/stdc++.h>
using namespace std;
long long int ar[500][500], lft[500], rght[500], lim, mm;
long long int solve(long long int i, long long int k) {
  long long int a, b, c;
  if (k == 0) {
    if (lft[i] == -1) {
      if ((i - 1) >= lim)
        a = 1 + solve(i - 1, 0);
      else
        a = 0;
    } else {
      a = lft[i];
      if ((i - 1) >= lim) {
        a = a + 1 +
            min(lft[i] + solve(i - 1, 0), mm + 1 - lft[i] + solve(i - 1, 1));
      }
    }
  } else {
    if (rght[i] == -1) {
      if ((i - 1) >= lim)
        a = 1 + solve(i - 1, 1);
      else
        a = 0;
    } else {
      a = mm + 1 - rght[i];
      if ((i - 1) >= lim) {
        a = a + 1 + min(a + solve(i - 1, 1), rght[i] + solve(i - 1, 0));
      }
    }
  }
  return a;
}
int main() {
  long long int a, b, c, d, e, f, g, h, i, j, k, l, m, n, o, p, q, r, s, t, u,
      v, w, x, y, z;
  cin >> n >> m;
  char ch;
  k = -1;
  for (i = 0; i < n; i++) {
    a = -1;
    b = -1;
    f = 0;
    for (j = 0; j < m + 2; j++) {
      cin >> ch;
      ar[i][j] = ch - '0';
      if (ar[i][j] == 1) {
        f = 1;
        if (a == -1) a = j;
        b = j;
      }
    }
    if (f == 1 && k == -1) k = i;
    lft[i] = b;
    rght[i] = a;
  }
  if (k == -1) {
    cout << "0\n";
    return 0;
  }
  if (k == (n - 1)) {
    cout << lft[n - 1] << "\n";
    return 0;
  }
  mm = m;
  lim = k;
  if (lft[n - 1] != -1)
    a = lft[n - 1];
  else
    a = 0;
  cout << 1 + a + min(a + solve(n - 2, 0), (m + 1) - a + solve(n - 2, 1))
       << "\n";
  return 0;
}
