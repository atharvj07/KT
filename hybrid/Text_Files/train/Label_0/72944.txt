import sys
input = sys.stdin.readline
from collections import Counter

def il():
	return list(map(int, input().split()))

def ii():
	return int(input())

def imi():
	return map(int, input().split())

def ist():
	return input()
T = ii()
for _ in range(T):
	n = ii()
	li = []
	t = Counter([])
	d = Counter([])
	for i in range(n):
		(a, b) = imi()
		t[a] += 1
		d[b] += 1
		li.append((a, b))
	N = n
	s = 0
	for i in li:
		(a, b) = i
		s += (t[a] - 1) * (d[b] - 1)
	ans = n * (n - 1) * (n - 2) // 6 - s
	print(ans)
