#include <bits/stdc++.h>
using namespace std;
int main() {
  int arr[101], i, j, k, n;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> arr[i];
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= n; j++) {
      for (k = j + 1; k <= n; k++) {
        if ((j != k) && (arr[i] == arr[j] + arr[k])) {
          cout << i << " " << k << " " << j;
          return 0;
        }
      }
    }
  }
  cout << "-1";
  return 0;
}
