#include <bits/stdc++.h>
using namespace std;
bool cmp(const int &a, const int &b) { return a > b; }
int main() {
  int n, k;
  int a[200], b[200];
  int i, j, l;
  scanf("%d%d", &n, &k);
  for (i = 0; i < n; i++) scanf("%d", &a[i]);
  for (i = 0; i < k; i++) scanf("%d", &b[i]);
  sort(b, b + k, cmp);
  for (i = 0, j = 0; i < n; i++) {
    if (a[i] == 0) {
      a[i] = b[j];
      j++;
    }
  }
  l = 0;
  for (i = 1; i < n; i++) {
    if (a[i] < a[i - 1]) {
      l = 1;
      break;
    }
  }
  if (l == 1)
    printf("Yes\n");
  else
    printf("No\n");
}
