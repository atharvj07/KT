m = int(input())
l = []
for _ in range(m):
	(a, b, c) = map(int, input().split())
	bc = max(b, c)
	ab = max(a, b)
	ac = max(a, c)
	if a > bc:
		a = 0
	else:
		a = bc - a + 1
	if b > ac:
		b = 0
	else:
		b = ac - b + 1
	if c > ab:
		c = 0
	else:
		c = ab - c + 1
	l.append(f'{a} {b} {c}')
for i in l:
	print(i)
