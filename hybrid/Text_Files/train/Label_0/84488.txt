for _ in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	d = {}
	flag = 0
	for i in range(1, n):
		if l[i] != l[i - 1]:
			flag = i
			break
	d[l[0]] = 1
	if flag != 0:
		for i in range(flag, n - 1):
			if l[i] != l[i + 1]:
				if l[i] not in d:
					d[l[i]] = 2
				else:
					d[l[i]] += 1
		if l[-1] not in d:
			d[l[-1]] = 1
		we = 9999999
		for i in d:
			we = min(we, d[i])
		print(we)
	else:
		print(0)
