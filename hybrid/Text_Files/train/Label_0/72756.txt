import math
(n, l, r) = map(int, input().split())
dp = []
for i in range(0, n):
	dp.append([0, 0, 0])
dp[0][0] = (r - 0) // 3 - (l - 0 - 1) // 3
dp[0][1] = (r - 1) // 3 - (l - 1 - 1) // 3
dp[0][2] = (r - 2) // 3 - (l - 2 - 1) // 3
MOD = int(1000000000.0 + 7)
for i in range(1, n):
	dp[i][0] += dp[i - 1][0] * dp[0][0] + dp[i - 1][1] * dp[0][2] + dp[i - 1][2] * dp[0][1]
	dp[i][1] += dp[i - 1][0] * dp[0][1] + dp[i - 1][1] * dp[0][0] + dp[i - 1][2] * dp[0][2]
	dp[i][2] += dp[i - 1][0] * dp[0][2] + dp[i - 1][1] * dp[0][1] + dp[i - 1][2] * dp[0][0]
	dp[i][0] %= MOD
	dp[i][1] %= MOD
	dp[i][2] %= MOD
print(dp[n - 1][0])
