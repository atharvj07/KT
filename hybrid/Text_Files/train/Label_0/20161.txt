#include<iostream>
#include<algorithm>
#include<cmath>
using namespace std;

int n, x, sum;

int hantei(int p) {
	int v = max((int)sqrt(p), p);
	for (int i = 1; i <= sqrt(p); i++) {
		int a = p - i;
		if (a % (2 * i + 1) == 0 && a >= 1) {
			return 0;
		}
	}
	return 1;
}

int main() {
	cin >> n;
	for (int i = 0; i < n; i++) {
		cin >> x;
		sum += hantei(x);
	}
	cout << sum << endl;
	return 0;
}
