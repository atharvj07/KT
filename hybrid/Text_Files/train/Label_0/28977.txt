#include <bits/stdc++.h>
using namespace std;
void add(string& s) {
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == '9') {
      s[i] = '0';
    } else {
      s[i] += 1;
    }
  }
  return;
}
void rot(string& s) {
  char tmp = s[0];
  for (int i = 1; i < s.length(); i++) {
    s[i - 1] = s[i];
  }
  s[s.length() - 1] = tmp;
  return;
}
int n;
string s;
int main(void) {
  cin >> n;
  cin >> s;
  string mn = s;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 10; j++) {
      add(s);
      mn = min(mn, s);
    }
    rot(s);
  }
  cout << mn;
}
