n = int(input())
a = n // 10000
b = n // 1000 % 10
c = n // 100 % 10
d = n // 10 % 10
e = n % 10
ans = a * 10000 + c * 1000 + e * 100 + d * 10 + b
ans1 = ans ** 5 % 100000
if ans1 < 10000:
	print(0, end='')
if ans1 < 1000:
	print(0, end='')
if ans1 < 100:
	print(0, end='')
if ans1 < 10:
	print(0, end='')
print(ans1)
