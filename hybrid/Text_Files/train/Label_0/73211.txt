ans = []
for _ in range(int(input())):
	n = int(input())
	u = list(map(int, input().split()))
	u.sort()
	ind = -1
	d = 10 ** 10
	for i in range(1, n):
		if u[i] - u[i - 1] < d:
			d = u[i] - u[i - 1]
			ind = i
	p = u[ind:] + u[:ind]
	(p[0], p[-1]) = (p[-1], p[0])
	ans.append(' '.join(map(str, p)))
print('\n'.join(ans))
