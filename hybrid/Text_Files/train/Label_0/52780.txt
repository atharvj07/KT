import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.util.Arrays;


public class Main {
    
    /**
     * @param args
     */
    public static void main(String[] args) throws Exception {
        
        Main main = new Main();
        main.doWork();
        
    }
    
    public boolean checkIfHere(String a, String small, int pos)
    {
        for (int i=pos; i<pos+small.length(); i++)
            if (a.charAt(i) != small.charAt(i-pos))
                return false;
        
        return true;        
    }
    
    private void doWork() throws Exception
    {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(new OutputStreamWriter(System.out));
        
        String line = reader.readLine();
        
        int n;
        n = Integer.parseInt(reader.readLine().trim());
        
        int start[] = new int[line.length()];
        int max[] = new int[start.length];
        
        Arrays.fill(start, 0); //no limit
        Arrays.fill(max, 1000000); //no limit
        
        String current;
        
        for (int i=0; i<n; i++)
        {
            current = reader.readLine();
            
            for (int j=0; j<line.length()-current.length()+1; j++)
            {
                if (checkIfHere(line, current, j) == true)
                {
                    start[j] = 1; //mark, cannot start here
                    max[j] = Math.min(max[j], current.length()-1);
                }
            }
        }
        
        int curMax = 1;
        
        int bestResult = 0, startingAt = 0;
        
        for (int j=line.length()-1; j>=0; j--)
        {
            curMax = Math.min(curMax, max[j]);
            
            max[j] = curMax;
            
            curMax++;
            
            //System.out.println("max[" + j + "]=" + max[j]);
        }
        
        for (int j=0; j<line.length(); j++)
        {
            if (max[j] > bestResult)
            {
                bestResult = max[j];
                startingAt = j;
            }
        }
        
        
        pw.write(bestResult + " " + startingAt + "\r\n");
        
        pw.flush();
        pw.close();
    }
}
