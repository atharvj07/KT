int find_set(int v,vi &p){
    if(v==p[v])
        return v;
    return p[v]=find_set(p[v],p);
}
void union_set(int a,int b,vi &p){
    a=find_set(a,p);
    b=find_set(b,p);
    if(a!=b){
        p[b]=a;
    }
}
