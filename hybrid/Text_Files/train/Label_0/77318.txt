import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	(r, c) = map(int, input().split())
	grid = []
	for i in range(r):
		l = list(map(str, input().strip()))
		l2 = [1 if j == 'A' else 0 for j in l]
		grid += [l2]
	summ = 0
	for i in grid:
		summ += sum(i)
	if summ == 0:
		print('MORTAL')
	elif summ == r * c:
		print(0)
	else:
		co = 0
		last = 0
		ex = 0
		for i in range(len(grid)):
			if grid[i][0] != 1:
				last = 1
			else:
				ex += 1
		if last == 1:
			last = 0
			for i in range(len(grid)):
				if grid[i][-1] != 1:
					last = 1
				else:
					ex += 1
		if last == 1:
			last = 0
			for i in range(len(grid[0])):
				if grid[0][i] != 1:
					last = 1
				else:
					ex += 1
		if last == 1:
			last = 0
			for i in range(len(grid[0])):
				if grid[-1][i] != 1:
					last = 1
				else:
					ex += 1
		if last == 0:
			print(1)
		else:
			flag = 0
			if grid[0][0] == 1 or grid[-1][0] == 1 or grid[0][-1] == 1 or (grid[-1][-1] == 1):
				co = 1
			for i in grid:
				if sum(i) == c:
					flag = 1
					break
			for i in range(c):
				s = 0
				for j in range(r):
					s += grid[j][i]
				if s == r:
					flag = 1
					break
			if co == 1:
				if len(grid) == 1 or len(grid[0]) == 1:
					print(1)
				else:
					print(2)
			elif flag == 1:
				print(2)
			elif co != 1 and ex > 0:
				if len(grid) == 1 or len(grid[0]) == 1:
					print(2)
				else:
					print(3)
			else:
				print(4)
