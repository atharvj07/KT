import java.util.Scanner;

public class MagicCalender {
    public static void main(String[] args)
    {
        int t;
        Scanner s=new Scanner(System.in);
        t=s.nextInt();
        while(t-->0)
        {
            long n,r;
            long ans=0;
            n=s.nextLong();
            r=s.nextLong();
            if(n>r)
                ans=r*(r+1)/2;
            else
                ans=((n-1)*n)/2+1;
            System.out.println(ans);
        }
    }
}
