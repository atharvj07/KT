	 int dfs(int root, Node[] adjList) {
		
		Iterator it = adjList[root].children.listIterator();
		int result = 1, val;
		while(it.hasNext()) {
			val = dfs((int)it.next(), adjList);
			result = result & val; 
		}
		
		if(result == 1 && adjList[root].val == 1) {
			// Add this vertex to sorted list
			// sortList.add(root);
			adjList[root].remove = true;
			// return 1;
		}
		return adjList[root].val;
	}

	public static String next() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
