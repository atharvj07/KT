import java.io.*; 
import java.util.*; 
import java.math.*; 

public class Main { 
	static long mod=1_000_000_000+7;
	static long[] facts=new long[1005];
	static long mul(long a, long b) {
		return a*b%mod;
	}
	static long add(long a, long b) {
		return (a+b)%mod;
	}
	static void precomp() {
		facts[0]=1;
		for (int i=1; i<facts.length; i++) facts[i]=mul(facts[i-1], i);
	}
	static long exp(long base, long pow) {
		if (pow==0) return 1;
		long half=exp(base, pow/2);
		if (pow%2==0) return mul(half, half);
		return mul(half, mul(half, base));
	}
	
	static long modInv(long x) {
		return exp(x, mod-2);
	}
	
	static long nCk(int n, int k) {
		return mul(facts[n], mul(modInv(facts[k]), modInv(facts[n-k])));
	}
	
	static int left=0;
	static int right=0;
	
	public static void main(String[] args) throws IOException 
	{ 
		FastScanner f = new FastScanner(); 
//		int t=f.nextInt();
		int t=1;
		precomp();
		while(t-->0){
			int n=f.nextInt();
			int x=f.nextInt();
			int pos=f.nextInt();
			long ans=0;
			int l=0;
			int r=0;
			int left=0;
			int right=n;
			int num=0;
			while(left<right) {
				int mid=(left+right)/2;
				if(mid<=pos) {
					l++;
					left=mid+1;
					if(mid==pos) l--;
				}
				else {
					right=mid;
					r++;
				}
//				System.out.println(mid);
			}
			ans=facts[n-l-r-1];
//			System.out.println((n-x)+" "+l+" "+r);
			if(l>0 && (x-1)>=l) {
				ans=mul(ans,mul(nCk(x-1,l),facts[l]));
			}
			if(r>0 && (n-x)>=r) {
//				System.out.println();
				ans=mul(ans,mul(nCk(n-x,r),facts[r]));
			}
			if((n-x)<r || (x-1)<l) {
				ans=0;
			}
			System.out.println(ans%mod);
		}
	} 
	static void sort(int [] a) {
        ArrayList<Integer> q = new ArrayList<>();
        for (int i: a) q.add(i);
        Collections.sort(q);
        for (int i = 0; i < a.length; i++) a[i] = q.get(i);
    }
	static class FastScanner {
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st=new StringTokenizer("");
		String next() {
			while (!st.hasMoreTokens())
				try {
					st=new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			return st.nextToken();
		}
		
		int nextInt() {
			return Integer.parseInt(next());
		}
		int[] readArray(int n) {
			int[] a=new int[n];
			for (int i=0; i<n; i++) a[i]=nextInt();
			return a;
		}
		long nextLong() {
			return Long.parseLong(next());
		}
		double nextDouble() {
			return Double.parseDouble(next());
		}
		long[] readLongArray(int n) {
			long[] a=new long[n];
			for (int i=0; i<n; i++) a[i]=nextLong();
			return a;
		}
	}
}
