def get_input():
	al = []
	for c in range(int(input())):
		input()
		a = [int(i) for i in input().split(' ')]
		b = [int(i) for i in input().split(' ')]
		al.append([a, b])
	return al

def solve(a, b):
	j = 1
	di = {}
	for i in range(len(a)):
		while j < b[i]:
			di[j] = i
			j += 2
	ans = 2 * len(a) + 1
	for i in range(len(a)):
		ans = min(ans, i + di[a[i]])
	return ans

def main():
	arr_list = get_input()
	for (a, b) in arr_list:
		r = solve(a, b)
		print(r)
main()
