#include <bits/stdc++.h>
using namespace std;
int kolko[2000100] = {0};
int n;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    int x;
    scanf("%d", &x);
    kolko[x]++;
  }
  for (int i = 0; i < 2000008; i++) {
    if (kolko[i] % 2 == 1) {
      kolko[i + 1] += kolko[i] / 2;
      kolko[i] = 1;
    }
    if (kolko[i] % 2 == 0) {
      kolko[i + 1] += kolko[i] / 2;
      kolko[i] = 0;
    }
  }
  long long rj = 0;
  for (int i = 0; i < 2000009; i++) {
    rj += kolko[i];
  }
  printf("%ld", rj);
  return 0;
}
