n = int(input())
l = [int(i) for i in input().split()]
ans = 0
stack = []
for i in range(n):
	while len(stack) != 0:
		if l[i] > stack[len(stack) - 1]:
			ans = max(ans, l[i] ^ stack[len(stack) - 1])
			stack.pop()
		else:
			ans = max(ans, l[i] ^ stack[len(stack) - 1])
			break
	stack.append(l[i])
print(ans)
