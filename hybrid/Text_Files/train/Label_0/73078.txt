from bisect import bisect_left

class BIT:

	def __init__(self, n):
		self.n = n
		self.a = [0] * (n + 1)

	def update(self, x, v):
		while x <= self.n:
			self.a[x] += v
			x += x & -x

	def query(self, x):
		res = 0
		while x:
			res += self.a[x]
			x -= x & -x
		return res
for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	b = sorted(arr)
	for i in range(len(arr)):
		arr[i] = bisect_left(b, arr[i]) + 1
	bit = BIT(n)
	swaps = 0
	for x in range(len(arr)):
		l = bit.query(arr[x] - 1)
		r = bit.query(arr[x])
		swaps += min(l, x - r)
		bit.update(arr[x], 1)
	print(swaps)
