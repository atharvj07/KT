struct __ {
  __() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
  }
} _;
template <typename I>
inline void in(I l, I r) {
  while (l != r) cin >> *l, ++l;
}
template <typename I>
inline void out(I l, I r, char D = ' ') {
  if (l != r) {
    cout << *l;
    while (++l != r) cout << D << *l;
  }
  cout << '\n';
}
struct debug_t {
  template <typename T>
  debug_t& operator<<(const T& value) {
    return *this;
  }
} debug;
int main() {
  string s;
  while (cin >> s) {
    vector<long long> odd(255, 0), even(255, 0);
    long long odd_res = 0, even_res = 0;
    for (int i = 0; i < static_cast<int>((s).size()); i++) {
      ++odd_res;
      if (i & 1) {
        odd_res += odd[s[i]];
        even_res += even[s[i]];
        ++odd[s[i]];
      } else {
        odd_res += even[s[i]];
        even_res += odd[s[i]];
        ++even[s[i]];
      }
    }
    cout << even_res << " " << odd_res << "\n";
  }
  return 0;
}
