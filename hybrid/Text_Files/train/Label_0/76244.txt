def ss(a, b, bi):
	if bi != 1:
		b = b[-bi:-bi + 1]
	else:
		b = b[-bi]
	if not b:
		return int(a)
	return int(a) - int(b)

def doit():
	(a, s) = input().strip().split()
	(p, q) = (len(a), len(s))
	(ai, si) = (1, 1)
	b = ''
	while si <= q:
		t = ss(s[-si], a, ai)
		if t >= 0:
			b += str(t)
			si += 1
		elif si < q:
			t = ss(s[-si - 1:][0:2], a, ai)
			if len(str(t)) > 1:
				print(-1)
				return 0
			else:
				b += str(t)
				si += 2
		else:
			print(-1)
			return 0
		ai += 1
	if ai > p:
		print(b[::-1].lstrip('0'))
	else:
		print(-1)
t = int(input())
for _ in range(t):
	doit()
