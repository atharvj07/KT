import java.util.*;
public class mrje{
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		int a=0,b=0;
		int[] x=new int[n];
		for(int i=0;i<n;i++) {
			x[i]=sc.nextInt();
		}
		int r=0;
		for(int i=0;i<n;i++) {
			int t=sc.nextInt();
			if(t>x[i]) {
				b++;
			}else if(t<x[i])
				a++;
			else {
				r++;
			}
		}
		if(r==n)
			System.out.print(-1);
		else if(a==b)
			System.out.print(2);
		else if(a>b){
			System.out.print(1);
		}else if(a==0)
			System.out.print(-1);
		else
			System.out.print(b/a+1);
	}
}
