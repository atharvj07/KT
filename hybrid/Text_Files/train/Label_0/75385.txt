t = int(input())
for z in range(t):
	(n, x) = map(int, input().split())
	arr = list(map(int, input().split()))
	s = sum(arr)
	ans = []
	for i in range(n):
		count = 1
		new = arr[i]
		while new % x == 0:
			count = count + 1
			new = new // x
		ans.append(count)
	mn = min(ans)
	print(s * mn + sum(arr[:ans.index(mn)]))
