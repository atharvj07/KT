t = int(input())
for i in range(t):
	count1 = [0, 0, 0, 0, 0]
	n = int(input())
	m = n // 2
	a = []
	for j in range(n):
		a.append(input().split())
		for day in range(5):
			count1[day] += int(a[j][day])
	kt = False
	for j in range(4):
		if count1[j] >= m:
			for g in range(j + 1, 5):
				if count1[g] >= m:
					kt = True
					for k in range(n):
						if a[k][j] == a[k][g] == '0':
							kt = False
							break
					if kt == True:
						print('YES')
						break
			if kt == True:
				break
	if kt == False:
		print('NO')
