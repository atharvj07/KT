t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	lstn = list(map(int, input().split()))
	lstm = list(map(int, input().split()))
	lstn.sort(reverse=True)
	pointer = 0
	c = 0
	for j in lstn:
		if pointer <= j:
			a = lstm[j - 1]
			b = lstm[pointer]
			if b < a:
				c += b
				pointer += 1
			else:
				c += a
		else:
			c += lstm[j - 1]
	print(c)
