def A1311(a, b):
	if a == b:
		return 0
	elif b > a:
		if (a - b) % 2 == 1:
			return 1
		else:
			return 2
	elif (a - b) % 2 == 0:
		return 1
	else:
		return 2
t = int(input())
L = [[0, 0]] * t
for i in range(0, t):
	L[i] = [int(k) for k in input().split()]
for i in range(0, t):
	print(A1311(L[i][0], L[i][1]))
