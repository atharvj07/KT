import java.util.Scanner;

public class Main {
	static int t1, t2, t, a1, a2, max = -1, x1, x2;
	static double dif = Integer.MAX_VALUE;

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		t1 = in.nextInt();
		t2 = in.nextInt();
		x1 = in.nextInt();
		x2 = in.nextInt();
		t = in.nextInt();
		double f = ((double) t1 - (double) t) / ((double) t - (double) t2);
		double y2;
		int yy2;
		dif = Integer.MAX_VALUE;
		max = -1;
		if (t1 == t2)
			System.out.println(x1 + " " + x2);
		else if (t == t1)
			System.out.println(x1 + " " + 0);
		else if (t == t2)
			System.out.println(0 + " " + x2);
		else {
			for (int y1 = 0; y1 <= x1; y1++) {
				y2 = f * y1;
				if (y1 == 0)
					y2 = x2;
				yy2 = (int) y2;
				solve(y1, yy2);
				yy2 = ((int) y2) - 1;
				solve(y1, yy2);
				yy2 = ((int) y2) + 1;
				solve(y1, yy2);
				yy2 = ((int) y2) - 2;
				solve(y1, yy2);
				yy2 = ((int) y2) + 2;
				solve(y1, yy2);
			}
			System.out.println(a1 + " " + a2);
		}
	}

	public static void solve(int y1, int yy2) {
		double dit;
		if (yy2 <= x2 && y1 <= x1) {
			dit = getDif(y1, yy2);
			if (dit < dif && dit >= 0) {
				dif = dit;
				max = y1 + yy2;
				a1 = y1;
				a2 = yy2;
			} else if (Math.abs(dit - dif) <= 0.000000001 && y1 + yy2 > max
					&& dit >= 0) {
				max = y1 + yy2;
				a1 = y1;
				a2 = yy2;
			}
		}
	}

	public static double getDif(int y1, int y2) {
		double tt = ((double) y1 * (double) t1 + (double) y2 * (double) t2)
				/ ((double) y1 + (double) y2);
		return (double) tt - t;
	}
}
