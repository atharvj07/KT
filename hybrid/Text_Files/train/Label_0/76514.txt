(s, limit) = map(int, input().split())
Cnt = [[]]
for i in range(10 ** 5):
	Cnt.append([])
Ind = [0] * (10 ** 5 + 1)
m = 0
for i in range(1, limit + 1):
	pos = 0
	x = i
	while x % 2 == 0:
		x //= 2
		pos += 1
	Cnt[2 ** pos].append(i)
	m = max(m, 2 ** pos)
L = []
while s > 0:
	e = s
	for j in range(min(m, s), -1, -1):
		if Ind[j] < len(Cnt[j]):
			L.append(Cnt[j][Ind[j]])
			Ind[j] += 1
			s -= j
			m = min(m, j)
			break
	if s == e:
		print(-1)
		break
if s == 0:
	print(len(L))
	for item in L:
		print(item, end=' ')
