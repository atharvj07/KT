N = int(input())
if N == 1:
	print(-1)
	exit()
S = input()
X = list(map(int, input().split()))
stk = []
ans = int(1000000000.0)
for (s, x) in zip(S, X):
	if len(stk) == 0:
		stk.append((s, x))
		continue
	top = stk[-1]
	if top[0] == 'R' and s == 'L':
		ans = min(ans, x - (top[1] + x) // 2)
		stk.pop()
	else:
		stk.append((s, x))
if ans == int(1000000000.0):
	print(-1)
else:
	print(ans)
