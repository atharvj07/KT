from sys import stdin
from sys import exit
live = True
if not live:
	stdin = open('data.in', 'r')
(n, m) = list(map(int, stdin.readline().strip().split()))
line = list(stdin.readline().strip())
ans = 0
listAns = []
for it in range(1, n):
	if line[it] == '.' and line[it - 1] == '.':
		ans += 1
for it in range(m):
	q = list(stdin.readline().strip().split())
	pos = int(q[0]) - 1
	s = str(q[1])
	if line[pos] == '.' and s != '.':
		if pos - 1 >= 0 and line[pos - 1] == '.':
			ans -= 1
		if pos + 1 < n and line[pos + 1] == '.':
			ans -= 1
	elif line[pos] != '.' and s == '.':
		if pos - 1 >= 0 and line[pos - 1] == '.':
			ans += 1
		if pos + 1 < n and line[pos + 1] == '.':
			ans += 1
	listAns += [ans]
	line[pos] = s
print(' '.join(map(str, listAns)))
if not live:
	stdin.close()
