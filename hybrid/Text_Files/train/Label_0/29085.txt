#include <bits/stdc++.h>

using namespace std;

struct meal
{
	int x, y;
};

const int DIM = 1e5 + 7;

meal v[DIM];
int dp[DIM];

bool cmp(meal a, meal b)
{
	return a.x < b.x;
}

int main()
{
	int n, t;
	cin >> n >> t;
	
	int res = 0;
	
	for(int i = 1; i <= n; i++)
	{
		cin >> v[i].x >> v[i].y;
		res = max(res, v[i].y);
	}
	
	sort(v + 1, v + 1 + n, cmp);
	
	dp[v[1].x] = v[1].y;
	
	for(int i = 2; i <= n; i++)
		for(int j = t - 1; j >= 0; j--)
		{
			res = max(res, dp[j] +  v[i].y);
			
			if(j + v[i].x < t)
				dp[j + v[i].x] = max(dp[j + v[i].x], dp[j] + v[i].y);
		}
	
	cout << res;
}
