long long n, m;
vector<vector<char> > a;
vector<vector<char> > rotate(vector<vector<char> > a) {
  long long n = a.size(), m = a[0].size();
  vector<vector<char> > A(m, vector<char>(n));
  for (long long i = 0; i < m; i++) {
    for (long long j = 0; j < n; j++) {
      A[i][j] = a[n - 1 - j][i];
    }
  }
  return A;
}
vector<vector<char> > flip(vector<vector<char> > a) {
  long long m = a.size(), n = a[0].size();
  vector<vector<char> > A(m, vector<char>(n));
  for (long long i = 0; i < m; i++) {
    for (long long j = 0; j < n; j++) {
      A[i][j] = a[i][n - 1 - j];
    }
  }
  return A;
}

}
