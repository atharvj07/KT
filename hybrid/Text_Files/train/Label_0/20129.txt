#include <bits/stdc++.h>
int n, l[100010], r[100010], tA, tB, Ans, k;
int main() {
  scanf("%d", &n);
  tA = tB = 0;
  for (int i = 1; i <= n; i++)
    scanf("%d%d", &l[i], &r[i]), tA += l[i], tB += r[i];
  Ans = abs(tA - tB);
  for (int i = 1; i <= n; i++)
    if (Ans < abs(tA - tB + ((r[i] - l[i]) << 1))) {
      k = i;
      Ans = abs(tA - tB + ((r[i] - l[i]) << 1));
    }
  printf("%d\n", k);
  return 0;
}
