t = int(input())
while t:
	t -= 1
	te = int(input())
	a = list(map(int, input().split()))
	if a[0] == 1:
		print(te + 1, end=' ')
		for i in range(te):
			print(i + 1, end=' ')
	elif a[-1] == 0:
		for i in range(te + 1):
			print(i + 1, end=' ')
	else:
		for i in range(te - 1):
			if a[i] == 0 and a[i + 1] == 1:
				print(i + 1, te + 1, end=' ')
				break
			else:
				print(i + 1, end=' ')
		i += 1
		while i + 1 <= te:
			print(i + 1, end=' ')
			i += 1
	print('')
