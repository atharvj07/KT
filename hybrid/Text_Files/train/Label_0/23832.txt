#include<iostream>
#include<algorithm>
using namespace std;
 pair<int ,int> a[100005];
int main()
{
    int n,m;
    cin>>n>>m;
    for(int i=0;i<m;i++)
    {
        cin>>a[i].second>>a[i].first;
    }
    int ans=0;
    sort(a,a+m);
    int l=a[0].first;
    for(int i=1;i<m;i++)
    {
        if(a[i].second>=l)
        {
            ans++;
            l=a[i].first;
        }
    }
    cout<<ans+1<<endl;
    return 0;
}
