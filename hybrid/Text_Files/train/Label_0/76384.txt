(n, m) = map(int, input().split())
lol = []
mem = 0
for i in range(n):
	(c, t) = map(int, input().split())
	mem += c * t
	lol.append(mem)
v = list(map(int, input().split()))
for k in v:
	l = 0
	r = n - 1
	if k <= lol[0]:
		print(1)
	else:
		while l < r - 1:
			s = (l + r) // 2
			if lol[s] < k:
				l = s
			else:
				r = s
		print(r + 1)
