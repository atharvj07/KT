#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main(){
	ios_base::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	int T; cin>>T;
	while(T--){
		int n; cin>>n;
		set<int> ans;
		vector<int> trees(n);
		for(int& i:trees) cin>>i;
		for(int i=0;i<n;i++)
			for(int j=i+1;j<n;j++)
				ans.insert(trees[j]-trees[i]);
		cout<<ans.size()<<'\n';
	}
	return 0;
}
