const int MAX_N = 10;

/* typedef */

typedef set<string> ss;

/* global variables */

/* subroutines */

bool cmphalf(const string &s) {
  int l = s.size(), lh = l >> 1;
  for (int i = 0, j = lh; i < lh; i++, j++)
    if (s[i] != s[j]) return false;
  return true;
}

/* main */

int main() {
  int n;
  string s;
  cin >> n >> s;
  int slen = s.size();

  int cnt = 0;
  
  for (int k = 1; k <= n; k++) {
    int l = 1 << k, lh = l >> 1;
    ss sts;

    for (int i = 0; i < slen; i += l) sts.insert(s.substr(i, l));

    int cntk = 0;
    for (ss::iterator sit = sts.begin(); sit != sts.end(); sit++) {
      if (! cmphalf(*sit)) cntk++;
    }
    cnt += cntk;

    //printf("%d: %d\n", k, cntk);
  }

  printf("%d\n", cnt);
  return 0;
}
