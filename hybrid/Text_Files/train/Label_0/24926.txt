#include <bits/stdc++.h>
using namespace std;
set<char> S;
int main() {
  string s;
  cin >> s;
  bool flag = 0;
  S.insert('A');
  S.insert('H');
  S.insert('I');
  S.insert('V');
  S.insert('M');
  S.insert('O');
  S.insert('T');
  S.insert('U');
  S.insert('W');
  S.insert('X');
  S.insert('Y');
  for (int i = 0; i < s.length(); i++) {
    if (s[i] != s[s.length() - 1 - i]) flag = 1;
    if (!S.count(s[i])) flag = 1;
  }
  if (flag)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
