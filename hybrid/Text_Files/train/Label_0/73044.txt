t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	if n == 1:
		print(k + 1)
		continue
	maxn = []
	for i in range(n - 1):
		maxn.append(10 ** (a[i + 1] - a[i]) - 1)
	c = 0
	s = 0
	flag = False
	for i in range(n - 1):
		if c + maxn[i] > k:
			flag = True
			break
		c += maxn[i]
		s += maxn[i] * 10 ** a[i]
	if not flag:
		i += 1
	s += (k - c + 1) * 10 ** a[i]
	print(s)
