#include <bits/stdc++.h>
using namespace std;
int g[100100];
int main() {
  int n;
  scanf("%d", &n);
  if (((n & 1) && (n - 1) % 4 != 0) || ((n & 1) == 0 && n % 4 != 0))
    printf("-1\n");
  else {
    int ini = n / 2;
    for (int i = 1; i < n + 1; ++i) g[i] = -1;
    if (n & 1) g[n / 2 + 1] = ini + 1;
    for (int i = 1; i < n + 1; ++i) {
      if (g[i] == -1) {
        g[i] = ini;
        g[ini] = n + 1 - i;
        g[n + 1 - i] = n + 1 - ini;
        g[n + 1 - ini] = i;
        ini--;
      }
    }
    for (int i = 1; i < n + 1; ++i) {
      if (i > 1) printf(" ");
      printf("%d", g[i]);
    }
    printf("\n");
  }
  return 0;
}
