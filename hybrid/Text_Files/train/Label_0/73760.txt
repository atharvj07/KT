from sys import stdin, stdout
from os import path
rd = lambda : stdin.readline().strip()
wr = stdout.write
if path.exists('input.txt'):
	stdin = open('input.txt', 'r')
import time, math

def sumDigits(x):
	return sum(map(int, str(x)))
x = int(rd())
for i in range(max(1, int(x ** (1 / 2)) - 162), int(x ** (1 / 2)) + 1):
	if i * (i + sumDigits(i)) == x:
		print(i)
		exit()
print(-1)
