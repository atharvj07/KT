#include<bits/stdc++.h>

using namespace std;

int main(){
  int N;
  long long K;
  cin >> N >> K;
  
  vector<int> A(N);
  for(int i = 0; i < N; i++) cin >> A[i];
  
  int ok = 1e9 + 1, ng = 0;
  while(ok - ng > 1){
    int mid = (ok + ng) >> 1;
    
    long long cut = 0;
    for(int i = 0; i < N; i++){
      cut += (A[i] + mid - 1) / mid - 1;
    }
    
    (cut <= K ? ok : ng) = mid;
  }
  
  cout << ok << endl;
  
  return 0;
}
