const int maxn = 1e3 + 10;
const int mod = 1e9 + 7;
int t, f, l[maxn], r[maxn];
string s;
void solve() {
    cin>>s;
    int len = s.size();
    l[len] = 0;
	f = 0;
    r[0] = s[0] - '0';
    for (int i = 1; i < len; ++i) 
		r[i] = s[i] - '0' + s[i - 1] - '0';
    for (int i = 1; i < len; ++i) {
        if (r[i] > 1) 
			f = 1;
        if (r[i] == 0 && f) {
            puts("NO");
            return;
        }
    }
    puts("YES");
}
int main() {
	cin>>t;
	while(t--)
		solve();
	return 0;
}
