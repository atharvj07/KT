for _ in range(int(input())):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	flag = True
	curr_index = -1
	while k > 0 and flag:
		c = 0
		for i in range(1, n):
			if arr[i] > arr[i - 1]:
				k -= 1
				arr[i - 1] += 1
				curr_index = i
				c += 1
				break
		if c == 0:
			flag = False
	if flag and k <= 0:
		print(curr_index)
	else:
		print(-1)
