for _ in range(int(input())):
	(n, m) = map(int, input().split())
	s = input()
	p = list(map(int, input().split()))
	p.sort()
	p = [0] + p
	lst = [0] * 26
	for i in range(1, m + 1):
		cnt = m - i + 2
		for j in range(p[i - 1], p[i]):
			lst[ord(s[j]) - 97] += cnt
	if p[-1] < n:
		for c in range(p[-1], n):
			lst[ord(s[c]) - 97] += 1
	print(*lst)
