#include <bits/stdc++.h>
using namespace std;
const int INF = 2139062143;
const int maxn = 2e5 + 10;
long long n, m, k, x, s;
long long a[maxn], b[maxn], c[maxn], d[maxn];
int getid(long long x) { return upper_bound(d, d + k + 1, x) - d - 1; }
int main() {
  scanf("%I64d%I64d%I64d", &n, &m, &k);
  scanf("%I64d%I64d", &x, &s);
  for (int i = 1; i <= m; i++) {
    scanf("%I64d", &a[i]);
  }
  for (int i = 1; i <= m; i++) {
    scanf("%I64d", &b[i]);
  }
  for (int i = 1; i <= k; i++) {
    scanf("%I64d", &c[i]);
  }
  for (int i = 1; i <= k; i++) {
    scanf("%I64d", &d[i]);
  }
  long long ans = n * x;
  int tmp = getid(s);
  ans = min(ans, (n - c[tmp]) * x);
  for (int i = 1; i <= m; i++) {
    if (b[i] <= s) {
      int pos = getid(s - b[i]);
      long long t = n - c[pos];
      ans = min(ans, t * a[i]);
    }
  }
  printf("%I64d\n", ans);
  return 0;
}
