from decimal import Decimal

def dist(a, b):
	(x1, y1) = a
	(x2, y2) = b
	return Decimal((x1 - x2) ** 2 + (y1 - y2) ** 2).sqrt()

def minh(a, b, c):
	m = dist(a, b)
	n = dist(b, c)
	k = dist(a, c)
	p = Decimal(m + n + k) / 2
	sqp = (p * (p - m) * (p - n) * (p - k)).sqrt()
	hm = Decimal(2) / m * sqp
	hn = Decimal(2) / n * sqp
	hk = Decimal(2) / k * sqp
	return min([hm, hn, hk])

def solve():
	n = int(input())
	coords = []
	for i in range(n):
		coords.append(tuple(map(int, input().split())))
	coords += coords
	res = min((minh(coords[i], coords[i + 1], coords[i + 2]) for i in range(n)))
	print(res / 2)
solve()
