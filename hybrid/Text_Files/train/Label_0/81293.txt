t = int(input())
for i in range(t):
	(n, r) = map(int, input().split())
	s = 0
	if n > r:
		if r % 2 == 0:
			s = s + (r + 1) * (r // 2)
		else:
			s = s + (r // 2 + 1) + (r + 1) * (r // 2)
	else:
		a = n - 1
		if a % 2 == 0:
			s = 1 + (a + 1) * (a // 2)
		else:
			s = 1 + (a // 2 + 1) + (a + 1) * (a // 2)
	print(s)
