#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353, inv2 = (MOD + 1) / 2;
long double eps = 1e-6;
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};
long double pi = acos(-1.0L);
long long f[111][111], dp[55][111][111], s[111];
int l[1000000 + 10], r[1000000 + 10], c[1000000 + 10], len[1000000 + 10], n;
int read() {
  int v = 0, f = 1;
  char c = getchar();
  while (c < 48 || 57 < c) {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (48 <= c && c <= 57) v = (v << 3) + v + v + c - 48, c = getchar();
  return v * f;
}
long long readll() {
  long long v = 0, f = 1;
  char c = getchar();
  while (c < 48 || 57 < c) {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (48 <= c && c <= 57) v = (v << 3) + v + v + c - 48, c = getchar();
  return v * f;
}
void add(long long &a, long long b) {
  a += b;
  if (a >= MOD) a -= MOD;
  if (a < 0) a += MOD;
}
void init() {
  n = read();
  for (int i = (int)1; i <= (int)n; i++) l[i] = read(), r[i] = read();
}
long long pw(long long x, long long y) {
  long long res = 1;
  while (y) {
    if (y & 1) res = res * x % MOD;
    x = x * x % MOD;
    y >>= 1;
  }
  return res;
}
long long inv(long long x) { return pw(x, MOD - 2); }
void solve() {
  int m = 0;
  reverse(l + 1, l + n + 1);
  reverse(r + 1, r + n + 1);
  long long cnt = 1;
  for (int i = (int)1; i <= (int)n; i++)
    cnt = cnt * inv(r[i] - l[i] + 1) % MOD, l[i]++, r[i]++;
  for (int i = (int)1; i <= (int)n; i++) c[++m] = --l[i], c[++m] = r[i];
  sort(c + 1, c + m + 1);
  int t = m;
  m = 0;
  for (int i = (int)1; i <= (int)t; i++)
    if (c[i] != c[i - 1]) c[++m] = c[i];
  for (int i = (int)1; i <= (int)n; i++) {
    for (int j = (int)1; j <= (int)m; j++)
      if (c[j - 1] == l[i]) {
        l[i] = j;
        break;
      }
    for (int j = (int)1; j <= (int)m; j++)
      if (c[j] == r[i]) {
        r[i] = j;
        break;
      }
  }
  for (int i = (int)1; i <= (int)m; i++) {
    f[i][1] = len[i] = c[i] - c[i - 1];
    for (int j = 2; j <= n; j++) {
      f[i][j] = f[i][j - 1] * inv(j) % MOD * (len[i] + j - 1) % MOD;
    }
  }
  dp[0][0][1] = 1;
  for (int i = (int)0; i <= (int)m; i++) s[i] = 1;
  long long ans = 0;
  for (int i = (int)1; i <= (int)n; i++) {
    for (int j = (int)l[i]; j <= (int)r[i]; j++)
      for (int k = (int)2; k <= (int)i; k++) dp[i][j][k] = dp[i - 1][j][k - 1];
    for (int j = (int)l[i]; j <= (int)r[i]; j++) dp[i][j][1] = s[j - 1];
    for (int j = (int)0; j <= (int)m; j++) s[j] = 0;
    for (int j = (int)l[i]; j <= (int)m; j++) {
      s[j] = s[j - 1];
      for (int k = (int)1; k <= (int)i; k++) {
        add(s[j], dp[i][j][k] * f[j][k] % MOD);
      }
    }
  }
  for (int i = (int)l[n]; i <= (int)r[n]; i++)
    for (int j = (int)1; j <= (int)n; j++)
      add(ans, dp[n][i][j] * f[i][j] % MOD);
  ans = ans * cnt % MOD;
  printf("%I64d\n", ans);
}
int main() {
  init();
  solve();
  return 0;
}
