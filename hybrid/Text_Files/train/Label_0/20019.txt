#include<bits/stdc++.h>
using namespace std;


int main(){
  int N,K,ans=0;
  int AL[26]={0};
  string P[10000];
  char pre;

  cin>>N>>K;
  for(int i=0;i<N;i++){
    cin>>P[i];
    // cout<<P[i][0]-'A'<<endl;
    AL[P[i][0]-'A']++;
  }

  while(1){
    sort(AL,AL+26);
    if(AL[25-(K-1)]<=0)break;

    ans++;
    for(int i=0;i<K;i++){
      AL[25-i]--;
    }
  }

  cout<<ans<<endl;
  
  return 0;
}
