from collections import deque, defaultdict
from math import sqrt, ceil, factorial, floor, inf, log2, sqrt, gcd
import bisect
import copy
from itertools import combinations
import sys

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
n = int(input())
a = get_array()
tot = sum(a)
x = set(a)
li = []
for ele in x:
	li.append(ele)
li.sort()
m = int(input())
n = len(li)
for i in range(m):
	(x, y) = get_ints()
	index1 = bisect.bisect_left(li, x + 1) - 1
	index2 = bisect.bisect_right(li, x)
	mini = inf
	if index1 < n and index1 >= 0:
		tot1 = tot - li[index1]
		req1 = max(0, x - li[index1])
		req2 = max(0, y - tot1)
		mini = min(mini, req1 + req2)
	if index2 < n:
		tot1 = tot - li[index2]
		req = max(0, y - tot1)
		mini = min(mini, req)
	print(mini)
