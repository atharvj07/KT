#include <bits/stdc++.h>
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long n;
  cin >> n;
  long long nn = n;
  vector<long long> p;
  for (long long i = 2; i * i <= n; i++) {
    long long v = 1;
    while (n % i == 0) {
      n /= i;
      v *= i;
    }
    if (v != 1) {
      p.push_back(v);
    }
  }
  if (n != 1) p.push_back(n);
  sort((p).begin(), (p).end());
  long long a = 1, b = nn;
  int h = p.size();
  for (int i = 0; i < (1 << h); i++) {
    long long x = 1, y = 1;
    for (int k = 0; k < h; k++) {
      if ((i >> k) & 1) {
        x *= p[k];
      } else
        y *= p[k];
    }
    if (max(x, y) < max(a, b)) {
      a = x;
      b = y;
    }
  }
  cout << a << ' ' << b << '\n';
}
