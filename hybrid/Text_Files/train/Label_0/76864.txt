def solve(arr):
	b = list(enumerate(arr))
	b.sort(key=lambda x: x[1])
	res = 0
	prev_right = -1
	prev_level = -1
	cur_level = 1
	cur_value = -1
	cur_right = -1
	for (i, v) in b:
		if v > cur_value:
			prev_level = cur_level
			prev_right = cur_right
			cur_value = v
			cur_right = i
			if i < prev_right:
				cur_level = prev_level + 1
			res += cur_level
		elif i < prev_right:
			res += prev_level + 1
			if cur_level != prev_level + 1:
				cur_level = prev_level + 1
				cur_right = i
			cur_right = max(cur_right, i)
		else:
			res += prev_level
			if cur_level == prev_level:
				cur_right = max(cur_right, i)
	return res
n = int(input())
a = [int(x) for x in input().strip().split()]
print(solve(a))
