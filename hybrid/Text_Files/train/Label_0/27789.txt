#include <bits/stdc++.h>
int main() {
  int n, i, j;
  char arr[100][100] = {0};
  scanf("%d", &n);
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++) scanf(" %c", &arr[i][j]);
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++)
      if (arr[i][j] == '#' && arr[i - 1][j] == '#' && arr[i + 1][j] == '#' &&
          arr[i][j - 1] == '#' && arr[i][j + 1] == '#')
        arr[i][j] = '.', arr[i - 1][j] = '.', arr[i + 1][j] = '.',
        arr[i][j - 1] = '.', arr[i][j + 1] = '.';
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++)
      if (arr[i][j] == '#') {
        printf("NO");
        return 0;
      }
  printf("YES");
  return 0;
}
