def calc(a, b):
	delta = b - a
	cnt = 0
	while delta:
		cnt += 1
		if delta > 0:
			delta = delta if delta % 2 else delta - 1
		else:
			delta = delta - 1 if delta % 2 else delta
		a += delta
		delta = b - a
	return cnt
for i in range(int(input())):
	(a, b) = map(int, input().split())
	print(calc(a, b))
