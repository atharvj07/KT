n = int(input())
s = input()
l = 0
r = n - 1
lc = None
rc = None
while not lc or (l < n and lc == s[l]):
	lc = s[l]
	l += 1
while not rc or (r > 0 and rc == s[r]):
	rc = s[r]
	r -= 1
mod = 998244353
if r == 0 and rc == s[r]:
	print(n * (n + 1) // 2 % mod)
elif lc != rc:
	print((l + n - r) % mod)
else:
	print((l + 1) * (n - r) % mod)
