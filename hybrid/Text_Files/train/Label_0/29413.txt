#include <bits/stdc++.h>
double pi = acos(-1.0);
using namespace std;
long long n, m;
int c[100005], w[100005], note[100005], coin[100005];
priority_queue<pair<int, int>, vector<pair<int, int> >,
               greater<pair<int, int> > >
    q;
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> c[i];
  for (int i = 0; i < n; i++) cin >> w[i];
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    if (c[i] % 100 == 0) {
      note[i] = c[i] / 100;
      coin[i] = 0;
      continue;
    }
    note[i] = c[i] / 100;
    coin[i] = c[i] - 100 * note[i];
    m -= coin[i];
    q.push(make_pair((100 - coin[i]) * w[i], i));
    while (m < 0) {
      pair<int, int> temp = q.top();
      q.pop();
      sum += temp.first;
      m += (-1) * (temp.first / w[temp.second] - 100) +
           temp.first / w[temp.second];
      note[temp.second]++;
      coin[temp.second] = 0;
    }
  }
  cout << sum << endl;
  for (int i = 0; i < n; i++) {
    cout << note[i] << " " << coin[i] << endl;
  }
  return 0;
}
