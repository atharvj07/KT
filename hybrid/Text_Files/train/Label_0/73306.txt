from random import randint
h = [-1] * 2000001
for t in range(int(input())):
	(n, A) = (int(input()), list(map(int, input().split())))
	B = []
	b = 1
	while len(B) < n:
		b = randint(1, 1000000)
		if all((h[a + b] != t for a in A)):
			for a in A:
				h[a + b] = t
			B.append(b)
	print('YES\n' + ' '.join(map(str, B)))
