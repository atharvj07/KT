#include <bits/stdc++.h>
using namespace std;
int n, k, i, j, curi, curj, a[510], b[510], curc, cur, c;
bool flag;
int main() {
  scanf("%d", &n);
  for (i = 0; i < n; i++) scanf("%d", &a[i]);
  scanf("%d", &k);
  for (i = 0; i < k; i++) scanf("%d", &b[i]);
  i = 0;
  for (j = 0; j < k; j++) {
    curc = 0;
    flag = false;
    cur = 0;
    c = -1;
    while (i < n && cur < b[j]) {
      cur += a[i];
      if (c == -1)
        c = a[i];
      else if (a[i] != c)
        flag = true;
      i++;
      curc++;
    }
    if ((flag == false && curc > 1) || cur > b[j] ||
        (j == k - 1 && cur < b[j])) {
      printf("NO\n");
      return 0;
    }
  }
  if (i < n) {
    printf("NO\n");
    return 0;
  }
  printf("YES\n");
  i = 0;
  for (j = 0; j < k; j++) {
    curi = i;
    cur = 0;
    c = -1;
    while (cur < b[j]) {
      cur += a[i];
      c = max(a[i], c);
      i++;
    }
    curj = i;
    for (i = curi; i < curj; i++) {
      if (a[i] == c && i > curi && c > a[i - 1]) {
        for (curc = 0; curc < i - curi; curc++)
          printf("%d L\n", i - curi + j - curc + 1);
        for (curc = 0; curc < curj - i - 1; curc++) printf("%d R\n", j + 1);
        break;
      }
      if (a[i] == c && i < curj - 1 && c > a[i + 1]) {
        for (curc = 0; curc < curj - i - 1; curc++)
          printf("%d R\n", i - curi + j + 1);
        for (curc = 0; curc < i - curi; curc++)
          printf("%d L\n", i - curi + j - curc + 1);
        break;
      }
    }
    i = curj;
  }
  return 0;
}
