n = int(input())
a = [input().split() for i in range(n)]
b = list(map(int, input().split()))
b = [e - 1 for e in b]
a = [sorted(e) for e in a]
s = a[b[0]][0]
v = True
for i in range(n):
	if s <= a[b[i]][0]:
		s = a[b[i]][0]
	elif s <= a[b[i]][1]:
		s = a[b[i]][1]
	else:
		v = False
if v:
	print('YES')
else:
	print('NO')
