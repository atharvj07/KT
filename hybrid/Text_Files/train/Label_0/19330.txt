#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-5;
const double INF = 1e+20;
int n;
double r, ans = INF;
struct point {
  double x, y, z;
  point() {}
  point(double a, double b, double c) {
    x = a;
    y = b;
    z = c;
  }
  void init() { scanf("%lf%lf%lf", &x, &y, &z); }
  point operator-(const point &a) const {
    return point(x - a.x, y - a.y, z - a.z);
  }
  double operator*(const point &a) const { return x * a.x + y * a.y + z * a.z; }
} o, v;
int sgn(double v) {
  if (fabs(v) < eps) return 0;
  if (v > 0.0)
    return 1;
  else
    return -1;
}
double solve(point p) {
  double a = v * v, b = p * v * 2, c = p * p - r * r;
  double delta = b * b - 4 * a * c;
  if (sgn(delta) < 0) return INF;
  delta = sqrt(fabs(delta));
  double s1 = (-b + delta) / (2 * a), s2 = (-b - delta) / (2 * a);
  if (sgn(s1) < 0)
    return INF;
  else {
    if (sgn(s2) < 0)
      return s1;
    else
      return s2;
  }
}
int main() {
  o.init();
  v.init();
  scanf("%lf", &r);
  scanf("%d", &n);
  for (int a = 1; a <= n; a++) {
    point nowp;
    nowp.init();
    double nowr;
    scanf("%lf", &nowr);
    r += nowr;
    ans = min(ans, solve(o - nowp));
    r -= nowr;
    int m;
    scanf("%d", &m);
    for (int b = 1; b <= m; b++) {
      point delta;
      delta.init();
      ans = min(ans, solve(o - nowp - delta));
    }
  }
  if (fabs(ans - INF) < eps)
    printf("-1\n");
  else
    printf("%.9lf\n", ans);
  return 0;
}
