#include <bits/stdc++.h>
using namespace std;
const int maxn = 100005;
using ll = long long;
int n, m;
ll k;
int type[maxn];
ll a[maxn], p[maxn], tot = 0;
struct DATA {
  int l, r, id;
};
DATA q[maxn];
int magic = 315;
ll ans[maxn];
ll cnt[3 * maxn], Plus[maxn], Minus[maxn], o[maxn];
map<ll, ll> mp;
void r_add(int idx) {
  tot += cnt[Minus[idx]];
  cnt[o[idx]]++;
}
void l_add(int idx) {
  tot += cnt[Plus[idx]];
  cnt[o[idx]]++;
}
void r_rem(int idx) {
  cnt[o[idx]]--;
  tot -= cnt[Minus[idx]];
}
void l_rem(int idx) {
  cnt[o[idx]]--;
  tot -= cnt[Plus[idx]];
}
int main(int argc, char const* argv[]) {
  scanf("%d %lld", &n, &k);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", &type[i]);
  }
  for (int i = 1; i <= n; ++i) {
    scanf("%lld", &a[i]);
    if (type[i] == 2) {
      a[i] = -a[i];
    }
    p[i] = p[i - 1] + a[i];
  }
  scanf("%d", &m);
  for (int i = 1; i <= m; ++i) {
    scanf("%d %d", &q[i].l, &q[i].r);
    --q[i].l;
    q[i].id = i;
  }
  for (int i = 0; i <= n; ++i) {
    mp[p[i]];
    mp[p[i] + k];
    mp[p[i] - k];
  }
  int tt;
  for (auto& i : mp) {
    i.second = ++tt;
  }
  for (int i = 0; i <= n; ++i) {
    o[i] = mp[p[i]];
    Plus[i] = mp[p[i] + k];
    Minus[i] = mp[p[i] - k];
  }
  sort(q + 1, q + m + 1, [&](DATA x, DATA y) {
    int b1 = x.l / magic, b2 = y.l / magic;
    if (b1 == b2)
      return x.r < y.r;
    else {
      return b1 < b2;
    }
  });
  int l = 0, r = 0;
  cnt[o[0]]++;
  for (int i = 1; i <= m; ++i) {
    while (l > q[i].l) l_add(--l);
    while (r < q[i].r) r_add(++r);
    while (l < q[i].l) l_rem(l++);
    while (r > q[i].r) r_rem(r--);
    ans[q[i].id] = tot;
  }
  for (int i = 1; i <= m; ++i) {
    printf("%lld\n", ans[i]);
  }
  return 0;
}
