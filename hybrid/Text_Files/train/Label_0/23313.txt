#include <bits/stdc++.h>
using namespace std;
int main() {
	int N;
	cin >> N;
	if (N == 1) {
		cout << 1 << endl;
		cout << 0 << endl;
		return 0;
	}
	if (N == 2) {
		cout << 2 << endl;
		cout << 0 << endl;
		cout << 11 << endl;
		return 0;
	}
	cout << N-1 << endl;
	for (int i = 2; i <= N-1; ++ i) {
		cout << "0" << string(i-2, '1') << "0" << endl;
	}
	cout << string(N, '1') << endl;
}
