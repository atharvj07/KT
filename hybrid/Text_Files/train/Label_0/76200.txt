from sys import stdin
input = stdin.readline

def dish(deli, pick):
	if len(deli) == 1:
		return min(deli[0], pick[0])
	blanck = list(zip(deli, pick))
	temp = 0
	deliv = 0
	blanck = sorted(blanck, key=lambda s: s[0], reverse=True)
	for i in blanck:
		if i[0] <= temp + i[1]:
			deliv = max(i[0], deliv)
		else:
			temp += i[1]
	return max(temp, deliv)
t = int(input())
for i in range(t):
	a = input()
	lst1 = list(map(int, input().strip().split()))
	lst2 = list(map(int, input().strip().split()))
	print(dish(lst1, lst2))
