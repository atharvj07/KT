import java.io.*;
import java.util.Arrays;
import java.util.StringTokenizer;

public class G {
    public static void main(String[] args) throws IOException {
        try (Input input = new StandardInput(); PrintWriter writer = new PrintWriter(System.out)) {
            int blockCount = input.nextInt();
            int totalDominoes = input.nextInt();
            int[][] heights = new int[blockCount][];
            int[][] costs = new int[blockCount][];
            for (int b = 0; b < blockCount; b++) {
                int blockSize = input.nextInt();
                heights[b] = input.nextIntArray(blockSize);
                costs[b] = input.nextIntArray(blockSize);
            }
            int[] h = new int[totalDominoes];
            long[] c = new long[totalDominoes];
            int offset = 0;
            for (int k = input.nextInt(); k > 0; k--) {
                int block = input.nextInt() - 1;
                long costMultiplier = input.nextInt();
                System.arraycopy(heights[block], 0, h, offset, heights[block].length);
                for (int cost : costs[block]) {
                    c[offset++] = costMultiplier * cost;
                }
            }
            int[] leftStack = new int[totalDominoes];
            int leftTop = 0;
            int[] rightStack = new int[totalDominoes];
            long[] rightStackValue = new long[totalDominoes];
            int rightTop = 0;
            long[] dp = new long[totalDominoes + 1];
            for (int i = 0; i < totalDominoes; i++) {
                int left = i - h[i];
                while (leftTop > 0 && left < leftStack[leftTop - 1]) {
                    leftTop--;
                }
                if (leftTop > 0) {
                    left = leftStack[leftTop - 1];
                } else {
                    left = -1;
                }
                leftStack[leftTop++] = i;
                while (rightTop > 0 && rightStack[rightTop - 1] <= i) {
                    rightTop--;
                }
                long bestFromPredecessors = rightTop == 0 ? Long.MAX_VALUE : rightStackValue[rightTop - 1];
                dp[i + 1] = Math.min(bestFromPredecessors, c[i] + dp[left + 1]);
                bestFromPredecessors = Math.min(bestFromPredecessors, dp[i] + c[i]);
                while (rightTop > 0 && rightStack[rightTop - 1] < i + h[i]) {
                    rightTop--;
                }
                if (rightTop == 0 || rightStackValue[rightTop - 1] != bestFromPredecessors) {
                    rightStack[rightTop] = i + h[i];
                    rightStackValue[rightTop++] = bestFromPredecessors;
                }
            }
            writer.println(dp[totalDominoes]);
        }
    }

    interface Input extends Closeable {
        String next() throws IOException;

        default int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        default long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        default int[] nextIntArray(int size) throws IOException {
            int[] array = new int[size];
            for (int i = 0; i < size; i++) {
                array[i] = nextInt();
            }
            return array;
        }
    }

    private static class StandardInput implements Input {
        private final BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        private StringTokenizer stringTokenizer;

        @Override
        public void close() throws IOException {
            reader.close();
        }

        @Override
        public String next() throws IOException {
            if (stringTokenizer == null || !stringTokenizer.hasMoreTokens()) {
                stringTokenizer = new StringTokenizer(reader.readLine());
            }
            return stringTokenizer.nextToken();
        }
    }
}
