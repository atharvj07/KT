#include <bits/stdc++.h>
using namespace std;
const int MX = 100000 * 3 + 5;
long long n, dp[MX], d[MX], k, sm[MX], ans;
pair<int, int> arr[MX];
long long sum(int l, int r) {
  if (l == 0) return sm[r];
  return sm[r] - sm[l - 1];
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> arr[i].first;
    arr[i].second = i + 1;
  }
  cin >> k;
  sort(arr, arr + n);
  sm[0] = arr[0].first;
  for (int i = 1; i <= n; i++) sm[i] = sm[i - 1] + arr[i].first;
  d[1] = 0;
  long long q = k - 1;
  for (int i = 2; i <= k; i++) {
    d[i] = d[i - 1] + (i - 1) * arr[i - 1].first - sum(0, i - 2);
  }
  dp[0] = d[k];
  ans = dp[0];
  int ind = 0;
  for (int i = 1; i + k - 1 < n; i++) {
    dp[i] = dp[i - 1] - (sum(i, i + k - 2) - (q)*arr[i - 1].first) +
            ((q)*arr[i + k - 1].first - sum(i, i + k - 2));
    if (dp[i] < ans) {
      ans = dp[i];
      ind = i;
    }
  }
  for (int i = ind; i < ind + k; i++) cout << arr[i].second << " ";
  cout << endl;
  return 0;
}
