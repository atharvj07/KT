import sys

input = sys.stdin.readline

def rints():
	return map(int, input().strip().split())

def rstr():
	return input().strip()

def rint():
	return int(input().strip())

def rintas():
	return [int(i) for i in input().strip().split()]

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)
k = rint()
for _ in range(k):
	n = rint()
	S = [i for i in rstr()]
	T = [i for i in rstr()]
	ans = []
	for i in range(n):
		if S[i] != T[i]:
			cont = False
			for j in range(i + 1, n):
				if S[j] != T[j] and S[i] == S[j]:
					(T[i], S[j]) = (S[j], T[i])
					ans.append([j, i])
					cont = True
					break
			if cont:
				continue
			for j in range(i + 1, n):
				if S[j] != T[j] and S[i] == T[j]:
					(T[j], S[j]) = (S[j], T[j])
					ans.append([j, j])
					(T[i], S[j]) = (S[j], T[i])
					ans.append([j, i])
					cont = True
					break
	if S == T:
		print('Yes')
		print(len(ans))
		for (i, j) in ans:
			print(i + 1, ' ', j + 1)
	else:
		print('No')
