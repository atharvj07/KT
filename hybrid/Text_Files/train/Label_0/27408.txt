#include <bits/stdc++.h>
using namespace std;
int n, ans, maks;
int a[105], cek[105];
int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    if (maks < a[i]) maks = a[i];
    cek[a[i]]++;
  }
  ans = 0;
  for (int i = 1; i <= maks; ++i) {
    if (ans < cek[i]) ans = cek[i];
  }
  cout << ans << endl;
}
