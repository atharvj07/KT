#include <bits/stdc++.h>
using namespace std;
const int N = 3 * 1e5 + 10;
int a[N];
const int MOD = 1e9 + 7;
long long fastMul(long long n, int k) {
  long long ret = 1;
  while (k) {
    if (k & 1) {
      ret = ret * n % MOD;
    }
    n = n * n % MOD;
    k >>= 1;
  }
  return ret;
}
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
  }
  sort(a, a + n);
  long long Ans = 0;
  for (int i = 0; i < n; ++i) {
    Ans += (fastMul(2, i) - fastMul(2, n - 1 - i) + MOD) % MOD * a[i] % MOD;
    Ans = Ans % MOD;
  }
  printf("%I64d\n", Ans);
  return 0;
}
