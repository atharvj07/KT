import java.util.Arrays;
import java.util.Scanner;

public class Hello2020E {

	static int N;
	static long[] sX,sY,x,y;
	static int[] q;
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		N = in.nextInt();
		sX = new long[N]; sY = new long[N];
		for(int i = 0; i < N; i++) {
			sX[i] = in.nextLong();
			sY[i] = in.nextLong();
		}
		long res = 0;
		for(int i = 0; i < N; i++) {
			x = new long[N-1];
			y = new long[N-1];
			q = new int[N-1];
			int idd = 0;
			for(int j = 0; j < N; j++) {
				if(i == j) continue;
				x[idd] = sX[j]-sX[i];
				y[idd] = sY[j]-sY[i];
				q[idd] = Q(x[idd], y[idd]);
				idd++;
			}
			Integer ord[] = new Integer[N-1];
			for(int j = 0; j < N-1; j++) ord[j] = j;
			Arrays.sort(ord, (a, b) -> comp(a,b));
			for(int l = 0, r = 0; l < N-1; l++) {
				while(cr(ord[l],ord[n(r)]) > 0) r = n(r);
				res += c3(r(n(r),l));
			}
		}
		System.out.println(N*c4(N-1)-res);
	}
	
	static int n(int a) { return a == N-2 ? 0 : a+1;}
	static int p(int a) { return a == 0 ? N-2 : a-1;}
	static int r(int a, int b) {
		if(a <= b) return b-a;
		return (N-1)-a + b;
	}
	static long c3(long n) {
		return (n*(n-1)*(n-2))/6;
	}
	static long c4(long n) {
		return (n*(n-1)*(n-2)*(n-3))/24L;
	}
	static int Q(long x, long y) {
		if(y >= 0) return x > 0 ? 1 : 2;
		return x > 0 ? 4 : 3;
	}
	static long cr(int a, int b) {
		return x[a]*y[b] - y[a]*x[b];
	}
	static int comp(int a, int b) {
		if(q[a] != q[b]) return q[a]-q[b];
		long c = x[a]*y[b] - y[a]*x[b];
		if(c > 0) return -1;
		if(c < 0) return 1;
		return 0;
	}
}
