import java.io.*;
import java.util.*;

public class Main {

	void solve(Scanner in, PrintWriter out) {
		int x = in.nextInt();
		int a = x / 10000;
		int b = x / 1000 % 10;
		int c = x / 100 % 10;
		int d = x / 10 % 10;
		int e = x % 10;
		long y = a * 10000 + c * 1000 + e * 100 + d * 10 + b;
		long res = y * y % 100000 * y % 100000 * y % 100000 * y % 100000;
		if (res == 0) {
			out.println("00000");
		} else {
			out.println(res);
		}

	}

	void run() {
		try (
			Scanner in = new Scanner(System.in);
			PrintWriter out = new PrintWriter(System.out)) {

			solve(in, out);
		} /*
			 * catch (IOException e) { throw new Error(e); }
			 */

	}

	public static void main(String args[]) {
		new Main().run();
	}
}
