#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  int i, n, c = 0;
  cin >> n >> s;
  for (i = 1; i <= n; i++) {
    string ss;
    cin >> ss;
    if (s[0] == ss[0] && s[1] == ss[1] && s[2] == ss[2]) c++;
  }
  if (c == n - c)
    printf("home\n");
  else
    printf("contest\n");
  return 0;
}
