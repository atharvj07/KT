def second(elem):
	return elem[1]
(n, r, avg) = (int(i) for i in input().split())
l = []
s = 0
for i in range(n):
	(a, b) = (int(i) for i in input().split())
	s += a
	if a != r:
		l.append((a, b))
l.sort(key=second)
diff = n * avg - s
c = 0
i = 0
while diff > 0:
	k = min(diff, r - l[i][0])
	c += k * l[i][1]
	diff -= k
	if k == r - l[i][0]:
		i += 1
print(c)
