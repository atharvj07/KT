#include<bits/stdc++.h>
#define kickStart ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define newLine '\n'
#define ll long long int
#define INF 1e18
using namespace std;
void solve()
{
    int n,m;
    cin >> n >> m;
    int arr[m];
    fill(arr,arr+m,0);
    for(int i=1; i<=n; i++)
    {
        string s;
        cin >> s;
        for(int j=0; j<m; j++)
            arr[j]+=(s[j]-'a'+1);
    }
    for(int i=1; i<=n-1; i++)
    {
        string s;
        cin >> s;
        for(int j=0; j<m; j++)
            arr[j]-=(s[j]-'a'+1);
    }
    for(int i=0; i<m; i++)
    {
        arr[i]--;
        cout << (char)(arr[i]+'a');
    }
    cout << newLine;
}
int main()
{
    kickStart;
    int t;
    cin >> t;
    while(t--)
        solve();
    return 0;
}
