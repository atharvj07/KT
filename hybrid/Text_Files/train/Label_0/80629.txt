def vacations(n, a):
	s = None
	r = 0
	for i in range(n):
		if a[i] == 0:
			r += 1
			s = 0
		elif a[i] == 1:
			if s == 1:
				r += 1
				s = 0
				continue
			else:
				s = 1
		elif a[i] == 2:
			if s == 2:
				r += 1
				s = 0
				continue
			else:
				s = 2
		elif s == 1:
			s = 2
		elif s == 2:
			s = 1
		elif i != n - 1 and a[i + 1] == 1:
			s = 2
		elif i != n - 1 and a[i + 1] == 2:
			s = 1
	print(r)
	return
n = int(input())
a = list(map(int, input().split(' ')))
vacations(n, a)
