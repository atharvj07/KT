epsilon = 10 ** (-6)
t = int(input())
for k in range(t):
	n = int(input())
	l = [int(x) for x in input().split()]
	x = int(input())
	l = [a - x for a in l]
	s = 0
	not_choose = 0
	new_subseg = False
	for i in range(n):
		if not new_subseg:
			s = l[i]
			new_subseg = True
		else:
			s += l[i]
			if s < 0:
				not_choose += 1
				new_subseg = False
			else:
				s = min(s, l[i])
	print(n - not_choose)
