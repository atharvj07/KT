//  _bsearch(key, array, left,right) -- left inclusive , right exclusive
template<typename T> T asc_com(const void * a, const void * b)
{
    return ( *(T*)a - *(T*)b  );
}
template<typename T> T des_com(const void * a, const void * b)
{
    return ( *(T*)b - *(T*)a  );
}
template<typename T> int _bsearch(T key, T aray[], int left_, int right_) //
{
    T* p=(T*) bsearch(&key, aray+left_, right_-left_, sizeof(aray[0]), asc_com );
    if(p)return (p-aray)+left_;
    return -1;
}

{
    return a<b;
}
template<typename T> bool com_des (T a, T b)
{
    return a>b;
}

char grid[555][555];
int dp[505][505];



int main()
{
#ifndef ONLINE_JUDGE
    freopen("input.txt", "r", stdin);
#endif

    TC
    {
        int n,m;
        scanf("%d %d", &n,&m);
        for(int i=0;i<n;i++)scanf("%s", grid[i]);
        //for(int i=0;i<n;i++)printf("%s\n", grid[i]);
        for(int i=0;i<n;i++)
            for(int j=0;j<m;j++)
        {
            dp[i][j]=0;
        }
        LL ans=0;
        for(int i=n-1;i>=0;i--)
            for(int j=m-1;j>=0;j--)
        {
            if(grid[i][j]=='*')
            {

                dp[i][j]=1;
                if(i+1<n && j-1>=0 && j+1<m)
                {
                    dp[i][j]=min(dp[i+1][j], min(dp[i+1][j-1],dp[i+1][j+1]) )+1;
                    //cout<<dp[i][j]<<endl;
                }
                ans+=(LL)dp[i][j];
            }
        }
        printf("%lld\n",ans);
    }

}
