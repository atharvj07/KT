#include <bits/stdc++.h>

using namespace std;

const int md = (int) 1e9 + 7;

inline void add(int &a, int b) {
  a += b;
  if (a >= md) a -= md;
}

inline void sub(int &a, int b) {
  a -= b;
  if (a < 0) a += md;
}

inline int mul(int a, int b) {
#if !defined(_WIN32) || defined(_WIN64)
  return (long long) a * b % md;
#endif
  unsigned long long x = (long long) a * b;
  unsigned xh = (unsigned) (x >> 32), xl = (unsigned) x, d, m;
  asm(
    "divl %4; \n\t"
    : "=a" (d), "=d" (m)
    : "d" (xh), "a" (xl), "r" (md)
  );
  return m;
}

inline int power(int a, long long b) {
  int res = 1;
  while (b > 0) {
    if (b & 1) {
      res = mul(res, a);
    }
    a = mul(a, a);
    b >>= 1;
  }
  return res;
}

inline int inv(int a) {
  a %= md;
  if (a < 0) a += md;
  int b = md, u = 0, v = 1;
  while (a) {
    int t = b / a;
    b -= t * a; swap(a, b);
    u -= t * v; swap(u, v);
  }
  assert(b == 1);
  if (u < 0) u += md;
  return u;
}

int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  int ans = mul(min(d + 1, a), mul(min(d + 1, b), min(d + 1, c)));
  for (int rot = 0; rot < 3; rot++) {
    int bound_x = a - d - 1;
    int x = 1;
    while (x <= bound_x) {
      int y = (long long) b * x / a;
      int z = (long long) c * x / a;
      int by_y = min(y + d, b - 1) - max(y - d, 0) + 1;
      int by_z = min(z + d, c - 1) - max(z - d, 0) + 1;
      int next_x = bound_x + 1;
      if (y + d <= b - 1 && y - d >= 0) {
        // y' + d >= b?
        // b * x' / a >= b - d
        // x' >= (b - d) * a / b
        next_x = min(next_x, (int) (((long long) (b - d) * a + b - 1) / b));
      } else {
        // y' >= y + 1
        // b * x' / a >= y + 1
        next_x = min(next_x, (int) (((long long) (y + 1) * a + b - 1) / b));
      }
      if (z + d <= c - 1 && z - d >= 0) {
        next_x = min(next_x, (int) (((long long) (c - d) * a + c - 1) / c));
      } else {
        next_x = min(next_x, (int) (((long long) (z + 1) * a + c - 1) / c));
      }
      add(ans, mul(next_x - x, mul(by_y, by_z)));
      x = next_x;
    }
    swap(a, b);
    swap(b, c);
  }
  printf("%d\n", ans);
  return 0;
}
