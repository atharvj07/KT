#include <bits/stdc++.h>
using namespace std;
int c4, c7, c47, c74;
string res;
void doMagic() {
  for (int i = 0;; i++)
    if (res[i] == '4') {
      res.insert(i, c4, '4');
      break;
    }
  for (int i = res.size() - 1;; i--)
    if (res[i] == '7') {
      res.insert(i, c7, '7');
      break;
    }
}
int main() {
  cin >> c4 >> c7 >> c47 >> c74;
  if (c4 < max(c47, c74) || c7 < max(c47, c74) || 1 < abs(c47 - c74)) {
    cout << "-1";
    return (0);
  }
  if (c4 == c74 && c7 == c74 && c47 == c74) {
    cout << "-1";
    return (0);
  }
  c4 -= max(c47, c74);
  c7 -= max(c47, c74);
  if (c47 < c74)
    for (int i = 0; i < c74; i++) res += "74";
  if (c47 >= c74)
    for (int i = 0; i < c47; i++) res += "47";
  if (c47 == c74)
    if (c4 > 0) {
      res += "4";
      c4--;
    } else {
      res = "7" + res;
      c7--;
    }
  doMagic();
  cout << res;
  return (0);
}
