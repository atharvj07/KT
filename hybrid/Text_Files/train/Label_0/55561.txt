import java.math.BigDecimal;
import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main 
{
    static BigInteger[] fact = new BigInteger[55];
    static BigInteger[][] dp;
    static int[] A;
    
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        fact[0] = BigInteger.ONE;
        for(int i = 1; i < 55; i++)
            fact[i] = fact[i - 1].multiply(BigInteger.valueOf(i));
        int n = sc.nextInt();
        A = new int[n];
        int sum = 0;
        for(int i = 0; i < n; i++)
        {
            A[i] = sc.nextInt();
            sum += A[i];
        }
        int p = sc.nextInt();
        if (sum <= p)
        {
            System.out.println(n);
            return;
        }
        dp = new BigInteger[n][p + 1];
        
        BigInteger cnt = BigInteger.ZERO;
        for(int lock = 0; lock < n; lock++)
        {
            for(int i = 0; i < n; i++)
                for(int j = 0; j <= p; j++)
                    dp[i][j] = BigInteger.ZERO;
            dp[0][0] = BigInteger.ONE;
            for(int add = 0; add < n; add++)
            {
                if (add != lock)
                {
                    for(int i = n - 2; i >= 0; i--)
                    {
                        for(int j = p; j >= 0; j--)
                        {
                            if (j + A[add] <= p)
                            {
                                dp[i + 1][j + A[add]] = dp[i + 1][j + A[add]].add(dp[i][j]);
                            }
                        }
                    }
                }
            }

            for(int i = 0; i < n; i++)
            {
                for(int j = Math.max(0, p - A[lock] + 1); j <= p; j++)
                    cnt = cnt.add(  dp[i][j].multiply( fact[n - i - 1] ).multiply( fact[i] ).multiply(BigInteger.valueOf(i))  );
            }
            
        }
        
        System.out.println(
                new BigDecimal(cnt).divide(   new BigDecimal(fact[n]), 30, BigDecimal.ROUND_HALF_UP  )
                );
    }
}
