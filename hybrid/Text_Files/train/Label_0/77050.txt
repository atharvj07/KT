def error_print(*args):
	print(*args, file=sys.stderr)

def mt(f):
	import time

	def wrap(*args, **kwargs):
		s = time.time()
		ret = f(*args, **kwargs)
		e = time.time()
		error_print(e - s, 'sec')
		return ret
	return wrap

@mt
def slv(S):
	ans = 0
	ci = defaultdict(list)
	for (i, c) in enumerate(S):
		ci[c].append(i)
	for v in ci.values():
		ans = max(ans, len(v))
	cs = list(ci.keys())
	for (a, b) in product(cs, repeat=2):
		t = 0
		for i in ci[a]:
			t += len(ci[b]) - bisect.bisect_right(ci[b], i)
		ans = max(ans, t)
	return ans

def main():

	print(slv(S))
