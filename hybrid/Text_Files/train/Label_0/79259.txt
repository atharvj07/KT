for _ in range(int(input())):
	n = int(input())
	l = [*map(int, input().split())]
	low = 0
	high = n - 1
	x = 0
	val = 0
	val1 = 0
	prefix = 0
	while low <= high:
		if not x & 1:
			temp = 0
			while prefix >= temp and low <= high:
				temp += l[low]
				low += 1
			prefix = temp
			val += temp
		else:
			temp = 0
			while prefix >= temp and low <= high:
				temp += l[high]
				high -= 1
			prefix = temp
			val1 += temp
		x += 1
	print(x, val, val1)
