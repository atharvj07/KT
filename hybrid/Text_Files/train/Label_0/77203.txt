(n, m) = map(int, input().split())
rec = [0] * (m + 1)
for i in range(1, m + 1):
	rec[i] += n // m
for i in range(1, n % m + 1):
	rec[i] += 1
rec2 = [0] * m
for i in range(m + 1):
	rec2[i ** 2 % m] += rec[i]
ans = 0
for i in range(m):
	if i == m - i and i > m - i:
		continue
	if i == 0:
		ans += rec2[i] * rec2[i]
	else:
		ans += rec2[i] * rec2[m - i]
print(ans)
