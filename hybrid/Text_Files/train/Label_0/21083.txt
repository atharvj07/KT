#include <bits/stdc++.h>
using namespace std;
int main() {
  string str1;
  string str2;
  string str3;
  string str4 = "";
  getline(cin, str1);
  getline(cin, str2);
  getline(cin, str3);
  for (int i = 0; i < str3.length(); i++) {
    if (isdigit(str3.at(i))) {
      str4 = str4 + str3.at(i);
    } else if (islower(str3.at(i))) {
      std::size_t found = str1.find(str3.at(i));
      if (found != std::string::npos) {
        str4 = str4 + str2.at(found);
      }
    } else {
      char c = tolower(str3.at(i));
      std::size_t found2 = str1.find(c);
      if (found2 != std::string::npos) {
        string s(1, toupper(str2.at(found2)));
        str4 = str4 + s;
      }
    }
  }
  cout << str4;
  return 0;
}
