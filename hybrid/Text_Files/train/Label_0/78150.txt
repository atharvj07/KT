(n, sx, sy) = map(int, input().split())
lst = []
for i in range(n):
	lst.append(list(map(int, input().split())))
quad1 = 0
quad2 = 0
quad3 = 0
quad4 = 0
for i in range(n):
	if lst[i][0] > sx:
		quad1 += 1
	if lst[i][1] > sy:
		quad2 += 1
	if lst[i][0] < sx:
		quad3 += 1
	if lst[i][1] < sy:
		quad4 += 1
count = [quad1, quad2, quad3, quad4]
temp = max(count)
print(temp)
ind = count.index(temp)
if ind == 0:
	print(sx + 1, sy)
elif ind == 1:
	print(sx, sy + 1)
elif ind == 2:
	print(sx - 1, sy)
else:
	print(sx, sy - 1)
