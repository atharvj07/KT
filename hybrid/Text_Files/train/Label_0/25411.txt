// Solver Function(s)
void dfs(int u, vector<array<ll, 2>> &dp, vector<bool> &vis, vector<int> g[], vector<array<ll, 2>> &vv)
{
	dp[u][0] = dp[u][1] = 0;
	vis[u] = true;
	for (auto v : g[u])
	{
		if (!vis[v])
		{
			dfs(v, dp, vis, g, vv);
			dp[u][0] += max(dp[v][0] + abs(vv[v][0] - vv[u][0]), dp[v][1] + abs(vv[v][1] - vv[u][0]));
			dp[u][1] += max(dp[v][0] + abs(vv[v][0] - vv[u][1]), dp[v][1] + abs(vv[v][1] - vv[u][1]));
		}
	}
}
void solve()
{
    int n;
    cin >> n;
    vector<array<ll, 2>> v(n);
    for (int i = 0; i < n; i++)
    {
    	cin >> v[i][0] >> v[i][1];
    }
    vector<bool> vis(n, false);
    vector<int> g[n];
    for (int i = 0; i < n - 1; i++)
    {
    	int a, b;
    	cin >> a >> b;
    	a--, b--;
    	g[a].pb(b);
    	g[b].pb(a);
    }
    vector<array<ll, 2>> dp(n);
    dfs(0, dp, vis, g, v);
    cout << max(dp[0][0], dp[0][1]) << "\n";
}

    while (t--)
    {
        solve();

}
