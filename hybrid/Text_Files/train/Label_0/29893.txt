template<typename T, typename... Args>
inline string arrStr(T arr, int n) {
   stringstream s; s << "[";
   for(int i = 0; i < n - 1; i++) s << arr[i] << ",";
   s << arr[n - 1] << "]";
   return s.str();
}

bool chk(int n){
    int g=x-b*n;
    int h=y-a*n;
    
    if(a==b){
      if(g>=0&&h>=0) return 1;
      return 0;
    }
    
    // EVARS(g,h);
    if(g<0) return 0;
    int mn=0,mx=n;
    
    int t=a-b;
    t=g/t;
    mx=t;

    if(h<0){
      t=a-b;
      t=(t-1+abs(h))/t;

      mn=t;
    }

    // EVARS(g,h,mn,mx,n);

    if(mn<=mx&&mn>=0&&mn<=n) return 1;
    return 0;




}
void boomer(){

  int lw=0,hg=1e9,mid;

     mid=(lw+hg+1)/2;
     if(chk(mid)) lw=mid;

     // EVARS(lw,hg,mid);
  }

}
