inline long MOD(long a, long b) {
  long res = a % b;
  return res < 0 ? res + abs(b) : res;
}
inline long Pow(long base, long exp) {
  long res = 1;
  while (exp) {
    if (exp & 1) res *= base;
    exp >>= 1;
    base *= base;
  }
  return res;
}
inline long BigMod(long base, long exp, long M) {
  base %= M;
  long res = 1;
  while (exp) {
    if (exp & 1) res = (res * base) % M;
    exp >>= 1;
    base = (base * base) % M;
  }
  return res;
}
