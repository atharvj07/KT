template <typename A, typename B>
inline void umin(A &x, B y) {
  if (x > y) x = y;
}
template <typename A, typename B>
inline void umax(A &x, B y) {
  if (x < y) x = y;
}
const int N = 200005;
using namespace std;
int n, x, a, b, c;
int d[5][5];
int main() {
  cin >> n;
  a = 1, b = 2, c = 3;
  while (n--) {
    cin >> x;
    if (x == c) {
      cout << "NO";
      return 0;
    }
    if (x == a) {
      swap(b, c);
    } else {
      swap(a, c);
    }
  }

}
