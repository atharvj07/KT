def solver(lst, a, n):
	b = []
	for i in range(n):
		if lst[i] == 0:
			b.append(a[i])
	b.sort()
	p = -1
	for i in range(n):
		if lst[i] == 0:
			a[i] = b[p]
			p -= 1
	return a
for _ in range(int(input())):
	n = int(input())
	arr = [int(i) for i in input().split()]
	l = [int(i) for i in input().split()]
	print(*solver(l, arr, n))
