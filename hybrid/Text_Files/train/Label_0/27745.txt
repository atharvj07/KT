#include <bits/stdc++.h>
using namespace std;
int maxi(int a, int b) {
  if (a > b) return a;
  return b;
}
vector<vector<int> > G;
vector<int> visited;
int DFS(int v) {
  visited[v] = 1;
  if (v > 1 && G[v].size() == 1) return 1;
  int h = 0;
  for (int i = 0; i < G[v].size(); i++) {
    if (visited[G[v][i]] == 0) h = maxi(h, DFS(G[v][i]));
  }
  return h + 1;
}
int main() {
  int N;
  cin >> N;
  visited.resize(N + 1, 0);
  G.resize(N + 1, vector<int>());
  int t;
  for (int i = 1; i <= N; i++) {
    cin >> t;
    if (t == -1) t = 0;
    G[t].push_back(i);
    G[i].push_back(t);
  }
  cout << DFS(0) - 1 << endl;
}
