(n, k) = map(int, input().split())
a = input()
print(n)
b = a[0:k] * (len(a) // k) + a[:len(a) - len(a) // k * k]
arrf = ''.join([str(i) for i in b])
if int(arrf) >= int(a):
	print(arrf)
else:
	c = a[:k]
	arr = ''.join([str(i) for i in c])
	arr = str(int(arr) + 1)
	c = list(arr)
	b = c[0:k] * (len(a) // k) + c[:len(a) - len(a) // k * k]
	arrf = ''.join([str(i) for i in b])
	print(arrf)
