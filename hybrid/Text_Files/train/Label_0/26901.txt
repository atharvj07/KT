#include<bits/stdc++.h>
using namespace std;
const int N=510;
int f[N][N],sum[N][N];
int n,mod,k;
int C[N][N];
void init()
{
	for(int i=0;i<N;i++)
		C[i][0]=C[i][i]=1;
	for(int i=1;i<N;i++)
		for(int j=1;j<i;j++)
			C[i][j]=(C[i-1][j-1]+C[i-1][j])%mod;
}
int main()
{
	scanf("%d%d%d",&n,&k,&mod);
	init();
	for(int i=0;i<=k;i++)
		f[1][i]=1;
	for(int j=k;j>=0;j--)
		sum[1][j]=(sum[1][j+1]+f[1][j])%mod;
	for(int i=2;i<=n+1;i++)
		for(int j=k;j>=0;j--)
		{
			for(int sz=1;sz<i;sz++)
				f[i][j]=(f[i][j]+1ll*f[i-sz][j]*sum[sz][j+1]%mod*C[i-2][sz-1])%mod;
			sum[i][j]=(sum[i][j+1]+f[i][j])%mod;
		}
	printf("%d\n",f[n+1][0]);
	return 0;
}
