#include <bits/stdc++.h>
using namespace std;
long long n;
long long k;
long long res = 0;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> k;
  if (k > (n / 2))
    res += (n - k) + n + (n - 1) + n + 1;
  else
    res += (k - 1) + n + (n - 1) + n + 1;
  cout << res;
}
