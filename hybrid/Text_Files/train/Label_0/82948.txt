def pro(A, B, F):
	(x, y) = A
	(p, q) = F
	(x1, y1) = B
	ans = 0
	if y == q == B[1] and min(x, x1) < p < max(x1, x):
		y += 1
		ans += 1
	elif x == p == x1 and min(y, y1) < q < max(y1, y):
		x += 1
		ans += 1
	ans += abs(x - x1) + abs(y - y1)
	print(ans)
t = int(input())
for i in range(t):
	input()
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	F = list(map(int, input().split()))
	pro(A, B, F)
