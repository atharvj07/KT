#include <bits/stdc++.h>
using namespace std;
pair<long long, long long> temp[1000005];
pair<long long, long long> operator+(pair<long long, long long> a,
                                     pair<long long, long long> b) {
  return make_pair(a.first + b.first, a.second + b.second);
}
pair<long long, long long> operator-(pair<long long, long long> a,
                                     pair<long long, long long> b) {
  return make_pair(a.first - b.first, a.second - b.second);
}
pair<long long, long long> ans1[1000005], ans2[1000005];
long long Find(pair<long long, long long> a, pair<long long, long long> b,
               pair<long long, long long> c) {
  long long ans = 0;
  ans += a.first * b.second;
  ans += b.first * c.second;
  ans += c.first * a.second;
  ans -= b.first * a.second;
  ans -= c.first * b.second;
  ans -= a.first * c.second;
  return ans;
}
long long GCD(long long a, long long b) {
  if (b == 0) return a;
  return GCD(b, a % b);
}
map<pair<long long, long long>, long long> how;
long long all[1000005];
long long now = 0;
struct A {
  long long l, r;
  long long where;
  long long ans;
  long long L_where;
} que[1000005];
bool cmp(A a, A b) {
  return a.L_where < b.L_where || a.L_where == b.L_where && a.r < b.r;
}
bool cmp2(A a, A b) { return a.where < b.where; }
long long L, R;
long long con[1000005] = {0};
long long ans = 0;
long long All[1000005];
void add(long long where) {
  if (con[all[where]] == 0) ans++;
  con[all[where]]++;
}
void Delete(long long where) {
  con[all[where]]--;
  if (con[all[where]] == 0) ans--;
}
int main() {
  long long N, M, i, j, k, x, y, T, Q, l, r, now1, now2, t, big, z = 0;
  scanf("%lld", &T);
  for (i = 0; i < T; i++) {
    scanf("%lld", &N);
    for (j = 0; j < N; j++) {
      scanf("%lld %lld", &x, &y);
      temp[j] = make_pair(x, y);
    }
    sort(temp, temp + N);
    now1 = 0;
    for (j = 0; j < N; j++) {
      while (now1 >= 2 && Find(ans1[now1 - 2], ans1[now1 - 1], temp[j]) <= 0)
        now1--;
      ans1[now1++] = temp[j];
    }
    now2 = 0;
    for (j = N - 1; j >= 0; j--) {
      while (now2 >= 2 && Find(ans2[now2 - 2], ans2[now2 - 1], temp[j]) <= 0)
        now2--;
      ans2[now2++] = temp[j];
    }
    for (j = 1; j < now2; j++) ans1[now1++] = ans2[j];
    for (j = 1; j < now1; j++) {
      auto x = ans1[j] - ans1[j - 1];
      if (x.first == 0)
        t = abs(x.second);
      else if (x.second == 0)
        t = abs(x.first);
      else
        t = GCD(abs(x.first), abs(x.second));
      if (how.find(make_pair(x.first / t, x.second / t)) == how.end())
        how[make_pair(x.first / t, x.second / t)] = now++;
      all[z++] = how[make_pair(x.first / t, x.second / t)];
    }
    All[i] = z;
  }
  for (i = 0; i * i < z; i++) big = i;
  big++;
  scanf("%lld", &M);
  for (i = 0; i < M; i++) {
    scanf("%lld %lld", &que[i].l, &que[i].r);
    que[i].l--;
    que[i].r--;
    if (que[i].l == 0)
      que[i].l = 0;
    else
      que[i].l = All[que[i].l - 1];
    que[i].r = All[que[i].r] - 1;
    que[i].L_where = que[i].l / big;
    que[i].where = i;
  }
  sort(que, que + M, cmp);
  for (i = 0; i < M; i++) {
    if (i == 0) {
      for (j = que[i].l; j <= que[i].r; j++) add(j);
      L = que[i].l;
      R = que[i].r;
    } else {
      while (R < que[i].r) {
        R++;
        add(R);
      }
      while (L > que[i].l) {
        L--;
        add(L);
      }
      while (R > que[i].r) {
        Delete(R);
        R--;
      }
      while (L < que[i].l) {
        Delete(L);
        L++;
      }
    }
    que[i].ans = ans;
  }
  sort(que, que + M, cmp2);
  for (i = 0; i < M; i++) printf("%lld \n", que[i].ans);
  return 0;
}
