				}

		}

class pair {
	int x;
	int y;

	pair(int a, int b) {
		this.x = a;
		this.y = b;

	}

	static int func() {
		return 0;
	}

	public int hashCode() {
		return x * 31 + y * 31;
	}

	public boolean equals(Object other) {
		if (this == other)
			return true;

		if (other instanceof pair) {
			pair pt = (pair) other;

			return pt.x == this.x && pt.y == this.y;
		} else
			return false;
	}
}
