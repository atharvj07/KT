def do(n, m, rob_y, rob_x, dirt_y, dirt_x):
	(rx, ry) = (rob_x, rob_y)
	(dx, dy) = (dirt_x, dirt_y)
	c = 0
	(ddx, ddy) = (1, 1)
	while True:
		if rx == dx or ry == dy:
			break
		if rx == m or rx == 0:
			ddx = -ddx
		if ry == n or ry == 0:
			ddy = -ddy
		rx += ddx
		ry += ddy
		c += 1
	print(c)
t = int(input())
for i in range(t):
	(n, m, rob_y, rob_x, dirt_y, dirt_x) = list(map(int, input().split()))
	do(n, m, rob_y, rob_x, dirt_y, dirt_x)
