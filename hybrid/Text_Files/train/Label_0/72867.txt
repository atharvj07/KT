from collections import Counter, deque, defaultdict
from heapq import *
import math

def inp():
	return int(input())

def mapInput():
	return map(int, input().split())

def arrInput():
	return list(map(int, input().split()))

def task():
	s = input()
	c = defaultdict(int)
	n = len(s)
	for x in s:
		c[x] += 1
	if c['1'] == n:
		return 0
	if c['0'] == n:
		return 1
	zCount = 0
	for x in s:
		if x == '0':
			zCount += 1
		elif zCount > 0:
			if zCount == c['0']:
				return 1
			return 2
	return 1
for _ in range(inp()):
	print(task())
