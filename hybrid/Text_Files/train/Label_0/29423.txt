#include<bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(int)(n);i++)
#define all(a) (a).begin(),(a).end()
#define EQ(a,b) (abs((a)-(b))<EPS)
#define fs first
#define sc second

using namespace std;
typedef double D;
typedef pair<int,int> P;
typedef pair<P,P> L;
typedef pair<D,int> pdi;

const D EPS = 1e-8;

inline D func(int x, P a, P b){
  int difx = b.fs - a.fs;
  D dify = b.sc - a.sc;
  return dify/difx * (x-a.fs) + a.sc;
}

int main(){
  int m,l,r;
  P p[110];
  L s[110];
  while(scanf("%d",&m),m){
    l = 10000, r = -10000;
    rep(i,m){
      scanf("%d%d",&p[i].fs,&p[i].sc);
      p[i].fs*=2;
      l = min(l,p[i].fs);
      r = max(r,p[i].fs);
    }
    rep(i,m){
      s[i] = L(p[i],p[(i+1)%m]);
      if(s[i].fs > s[i].sc)swap(s[i].fs,s[i].sc);
    }

    int ans = 0;
    for(int x=l+1;x<r;x+=2){
      vector<pdi> inv; inv.reserve(110);
      rep(i,m){
	if(s[i].fs.fs < x+EPS && x< s[i].sc.fs+EPS){
	  inv.push_back(pdi(func(x,s[i].fs,s[i].sc), i));
	}
      }

      sort(all(inv));
      int prv = -10000;
      for(int i=0;i<(int)inv.size();i+=2){
	int u = inv[i+1].sc, b = inv[i].sc;
	int up = ceil(max(func(x-1,s[u].fs,s[u].sc),
			  func(x+1,s[u].fs,s[u].sc))-EPS);
	int bot = floor(min(func(x-1,s[b].fs,s[b].sc),
			    func(x+1,s[b].fs,s[b].sc))+EPS);
	if(bot<prv)bot = prv;
	ans += up - bot;
	prv = up;
      }
    }
    printf("%d\n",ans);
  }
}
