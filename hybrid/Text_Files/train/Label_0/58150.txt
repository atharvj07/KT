import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;


public class PhoneNumbers {
public static void main(String[] args) throws IOException{
    BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(bf.readLine());
    int taxi [] = new int [n];
    int pizza [] = new int [n];
    int girl [] = new int [n];
    String names [] = new String[n];
    for (int i = 0; i < n; i++) {
        StringTokenizer st = new StringTokenizer(bf.readLine());
        int num = Integer.parseInt(st.nextToken());
        String name = st.nextToken();
        names[i] = name;
        for (int j = 0; j < num; j++) {
            String number = bf.readLine();
            String n1 = number.substring(0, 2);
            String n2 = number.substring(3, 5);
            String n3 = number.substring(6);
            number = n1+n2+n3;
            boolean equal = true;
            boolean dec = true;
            for (int k = 0; k < 5; k++) {
                    //System.out.println("3w");
                    if(number.charAt(k) != number.charAt(k+1)){
                        equal = false;
                    }
                    if(number.charAt(k) <= number.charAt(k+1)){
                        dec = false;
                    }
                    if(!equal && !dec)
                        break;
            }
            if(equal)
                taxi[i]++;
            
            else if(dec)
                pizza[i]++;
            
            else
                girl[i]++;
        }
    }
    /*for (int i = 0; i < names.length; i++) {
        System.out.print(names[i]+" ");
        System.out.print(taxi[i]+" ");
        System.out.print(pizza[i]+" ");
        System.out.print(girl[i]+" ");
        System.out.println();
    }*/
    String maxTaxi = "If you want to call a taxi, you should call: ";
    int max = 0;
    for (int i = 0; i < n; i++) {
        if(taxi[i] > max)
            max = taxi[i];
    }
    for (int i = 0; i <n; i++) {
        if(taxi[i] == max){
            maxTaxi+=names[i]+", ";
        }
    }
    
    String maxPizza = "If you want to order a pizza, you should call: ";
    max = 0;
    for (int i = 0; i < n; i++) {
        if(pizza[i] > max)
            max = pizza[i];
    }
    for (int i = 0; i <n; i++) {
        if(pizza[i] == max){
            maxPizza+=names[i]+", ";
        }
    }
    
    
    String maxGirl = "If you want to go to a cafe with a wonderful girl, you should call: ";
    max = 0;
    for (int i = 0; i < n; i++) {
        if(girl[i] > max)
            max = girl[i];
    }
    for (int i = 0; i <n; i++) {
        if(girl[i] == max){
            maxGirl+=names[i]+", ";
        }
    }
    
    maxTaxi = maxTaxi.substring(0, maxTaxi.length()-2)+".";
    maxPizza = maxPizza.substring(0, maxPizza.length()-2)+".";
    maxGirl = maxGirl.substring(0, maxGirl.length()-2)+".";
    System.out.println(maxTaxi);
    System.out.println(maxPizza);
    System.out.println(maxGirl);
}
}
