from sys import stdin, stdout
sze = 4
maps = []
for i in range(sze):
	maps.append(list(stdin.readline().strip()))

def check(maps):
	for i in range(sze - 2):
		for j in range(sze):
			if maps[i][j] == maps[i + 1][j] == maps[i + 2][j] == 'x':
				return True
	for i in range(sze):
		for j in range(sze - 2):
			if maps[i][j] == maps[i][j + 1] == maps[i][j + 2] == 'x':
				return True
	for i in range(sze - 2):
		for j in range(sze - 2):
			if maps[i][j] == maps[i + 1][j + 1] == maps[i + 2][j + 2] == 'x':
				return True
	for i in range(sze - 2, sze):
		for j in range(sze - 2):
			if maps[i][j] == maps[i - 1][j + 1] == maps[i - 2][j + 2] == 'x':
				return True
	return False
label = 0
for i in range(sze):
	for j in range(sze):
		if maps[i][j] == '.':
			maps[i][j] = 'x'
			if check(maps):
				label = 1
			maps[i][j] = '.'
if label:
	stdout.write('YES')
else:
	stdout.write('NO')
