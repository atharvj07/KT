#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC target("tune=native")
#pragma GCC optimize("fast-math")
using namespace std;
const int DEBUG = 0;
mt19937 gen(
    (unsigned)chrono::high_resolution_clock::now().time_since_epoch().count());
#pragma comment(linker, "/STACK:16777216")
long long dp[51];
vector<int> cyclesLens;
void solve(int n, long long k) {
  if (n <= 0) return;
  if (dp[n - 1] >= k) {
    cyclesLens.push_back(1);
    solve(n - 1, k);
  } else {
    cyclesLens.push_back(2);
    solve(n - 2, k - dp[n - 1]);
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.setf(cout.fixed);
  cout.precision(12);
  auto START_TIME = chrono::high_resolution_clock::now();
  int n;
  long long k;
  cin >> n >> k;
  dp[0] = dp[1] = 1;
  for (int i = 2; i <= 50; i++) {
    dp[i] = dp[i - 1] + dp[i - 2];
  }
  solve(n, k);
  int lid = 1;
  for (int len : cyclesLens) {
    cout << lid + len - 1 << " ";
    for (int i = lid + 1; i <= lid + len - 1; i++) {
      cout << i - 1 << " ";
    }
    lid += len;
  }
  return 0;
}
