n = int(input())
r = [[0] * n for i in range(n)]
c = n // 2
cur = 1
for i in range(n):
	r[c][i] = cur
	cur += 2
for i in range(n):
	if i != c:
		r[i][c] = cur
		cur += 2
if cur < n * n:
	for i in range(1, c + 1):
		for j in range(1, c + 1):
			r[c - i][c + j] = cur
			cur += 2
			r[c - i][c - j] = cur
			cur += 2
			r[c + i][c + j] = cur
			cur += 2
			r[c + i][c - j] = cur
			cur += 2
			if cur >= n * n:
				break
		if cur >= n * n:
			break
cur = 2
for i in range(n):
	for j in range(n):
		if r[i][j] == 0:
			r[i][j] = cur
			cur += 2
for i in range(n):
	print(str.join(' ', map(str, r[i])))
