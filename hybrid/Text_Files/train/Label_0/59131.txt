public class Main{
 
    static long dp[][] = new long[201][101];
    static int grid[][] = new int[201][101];
   
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
       
        int k = sc.nextInt();
       
        grid[1][2] = 1;
        grid[2][1] = 1;
        grid[1][3] = 1;
        grid[3][1] = 1;
        grid[2][3] = 1;
        grid[3][2] = 1;
        
        int n = 3;
        k = k - 1;
        
        while(k > 0){
        	boolean isNeedNewVertices = true;
        	for(int i = 1; i <= n; i++){
        		for(int j = 1; j <= n; j++){
        			if(i != j && grid[i][j] == 0){
        				int count = 0;
        				
        				for(int l = 1; l <= n; l++){
        					if(grid[i][l] == 1 && grid[j][l] == 1){
        						count++;
        					}
        				}
        				if(k - count >= 0){
        					k -= count;
        					grid[i][j] = 1;
        					grid[j][i] = 1;
        					isNeedNewVertices = false;
        				}
        			}
        		}
        	}
        	
        	if(isNeedNewVertices){
        		n++;
        		grid[1][n] = 1;
        		grid[n][1] = 1;
        		grid[2][n] = 1;
        		grid[n][2] = 1;
        		k--;
        	}
        	
        }
        System.out.println(n);
        
        for(int i = 1; i <= n; i++){
        	for(int j = 1; j <= n; j++){
        		System.out.print(grid[i][j]);
        	}
        	System.out.println();
        }
        
        
        sc.close();
    }
 
}
