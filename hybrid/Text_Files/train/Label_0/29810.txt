#include <bits/stdc++.h>
using namespace std;
signed main() {
  long long T;
  cin >> T;
  for (long long t = 0; t < T; t++) {
    long long N, M;
    cin >> N >> M;
    long long used_str0;
    long long used_col0;
    vector<vector<long long>> strs;
    for (long long i = 0; i < N; i++) {
      strs.push_back({});
      for (long long j = 0; j < M; j++) {
        long long a;
        cin >> a;
        strs[i].push_back(a);
      }
    }
    vector<vector<long long>> cols;
    for (long long i = 0; i < M; i++) {
      cols.push_back({});
      for (long long j = 0; j < N; j++) {
        long long a;
        cin >> a;
        cols[i].push_back(a);
      }
    }
    vector<long long> col = cols[0];
    vector<long long> str = strs[0];
    for (long long i = 0; i < M; i++) {
      for (long long j = 0; j < N; j++) {
        if (str[i] == col[j]) {
          used_str0 = j;
          used_col0 = i;
        }
      }
    }
    long long A[N][M];
    for (long long i = 0; i < M; i++) {
      A[used_str0][i] = str[i];
    }
    for (long long i = 0; i < N; i++) {
      A[i][used_col0] = col[i];
    }
    for (long long i = 1; i < N; i++) {
      vector<long long> str = strs[i];
      long long used_str;
      for (long long j = 0; j < N; j++) {
        if (str[used_col0] == col[j]) {
          used_str = j;
        }
      }
      for (long long j = 0; j < M; j++) {
        A[used_str][j] = str[j];
      }
    }
    for (long long i = 0; i < N; i++) {
      for (long long j = 0; j < M; j++) {
        cout << A[i][j] << " ";
      }
      cout << endl;
    }
  }
  return 0;
}
