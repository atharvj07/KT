#include <bits/stdc++.h>
using namespace std;
const int N = 5001;
char s[N];
int main() {
  int i, N, ik, ans = 0;
  scanf("%s", s);
  N = strlen(s);
  ik = -1;
  for (i = 0; i < N - 3; i++)
    if (s[i] == 'b' && s[i + 1] == 'e' && s[i + 2] == 'a' && s[i + 3] == 'r') {
      ans += (i - ik) * (N - i - 3);
      ik = i;
    }
  printf("%d", ans);
}
