else:
	inp = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

	def debug(*args):
		pass
for _ in range(int(inp())):
	n = int(inp())
	A = [*map(int, inp().split())]
	even = odd = 0
	for (i, a) in enumerate(A):
		if i & 1:
			odd += a
		else:
			even += a
	diff = even - odd
	if diff & 1:
		print('NO')
	else:
		diff = diff // 2
		cand = []
		if diff:
			for i in range(n - 1):
				if A[i] + ((i & 1) * 2 - 1) * diff == A[i + 1]:
					cand.append(i)
		swapped = False
		success = True
		x = 0
		for (i, a) in enumerate(A + [0]):
			x = a - x
			if x < 0:
				debug(x, diff)
				if abs(x) > abs(diff):
					swap = max([-1] + [ind for ind in cand if ind < i])
				else:
					swap = max([-1] + [ind for ind in cand if ind <= i])
				if swap >= 0:
					(A[swap], A[swap + 1]) = (A[swap + 1], A[swap])
				else:
					success = False
				swapped = True
				break
		if swapped and success:
			x = 0
			for (i, a) in enumerate(A + [0]):
				x = a - x
				if x < 0:
					success = False
					break
		if success:
			print('YES')
		else:
			print('NO')
