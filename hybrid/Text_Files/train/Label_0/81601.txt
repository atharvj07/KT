n = int(input())
line = input()
up = 0
down = 0
for x in line:
	if x == 'x':
		down += 1
	else:
		up += 1
print(int(abs(up - down) / 2))
for x in line:
	if up < down and x == 'x':
		print('X', end='')
		up += 1
		down -= 1
	elif down < up and x == 'X':
		print('x', end='')
		down += 1
		up -= 1
	else:
		print(x, end='')
