import java.util.Arrays;
import java.util.Scanner;

public class C {
    static int n;
    static char array[][];

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        n = scan.nextInt();
        array = new char[n][n];

        for (int i = 0; i < n; i++)
            array[i] = scan.next().toCharArray();

        int count = 0;
        int ans[] = new int[n];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (array[i][j] == '.') {
                    ans[count++] = j + 1;
                    break;
                }

        if (count == n) {
            for (int i = 1; i <= n; i++)
                System.out.println(i + " " + ans[i - 1]);
            return;
        }
        count = 0;
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (array[j][i] == '.') {
                    ans[count++] = j + 1;
                    break;
                }

        if (count == n) {
            for (int i = 1; i <= n; i++)
                System.out.println(ans[i - 1] + " " + i);
            return;
        }
        System.out.println(-1);

    }

}
