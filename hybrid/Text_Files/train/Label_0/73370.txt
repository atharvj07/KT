n = int(input())
a = [*map(int, input().split())]
from itertools import accumulate
aa = [*accumulate(a)]
empc = 0
day = set()
ans = []
for (i, j) in zip(a, aa):
	if i in day or (i < 0 and -i not in day):
		print(-1)
		exit()
	else:
		day.add(i)
		empc += 1
	if j == 0:
		ans.append(empc)
		empc = 0
		day.clear()
if len(day) == 0:
	print(len(ans))
	print(*ans)
else:
	print(-1)
