import sys
MOD = 1000000007

def powx(b, exp):
	if exp == 0:
		return 1
	ans = powx(b, exp // 2)
	ans = ans * ans % MOD
	if exp % 2:
		ans = ans * b % MOD
	return ans

def factorize(n):
	ret = []
	p = 2
	while p * p <= n:
		if n % p == 0:
			ret.append(p)
		while n % p == 0:
			n //= p
		p += 1 if p == 2 else 2
	if n != 1:
		ret.append(n)
	return ret

def main():
	(x, n) = map(int, input().split())
	factors = factorize(x)
	ans = 1
	for p in factors:
		k = p
		while k <= n:
			ans = ans * powx(p, n // k) % MOD
			k *= p
	print(ans)
main()
