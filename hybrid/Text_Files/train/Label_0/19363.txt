#include <iostream>
#include <algorithm>
#include <vector>
#include <string>

using namespace std;

int main(){

	int w,h;
	while(cin>>w>>h&&!(w==0&&h==0)){
		vector<string> orders;
		string str;
		getline(cin,str);
		while(1){
			getline(cin,str);
			orders.push_back(str);
			if(str=="STOP"){
				break;
			}
		}
		int cy=0;
		int cx=0;
		// 0:up,1:right,2:down,3:left
		int angle=0;
		for(int i = 0; i < orders.size(); i++){
			if(orders[i].substr(0,7)=="FORWARD"){
				int px=cx;
				int py=cy;
				if(angle==0)
					cy+=atoi(orders[i].substr(8).c_str());
				else if(angle==1)
					cx+=atoi(orders[i].substr(8).c_str());
				else if(angle==2)
					cy-=atoi(orders[i].substr(8).c_str());
				else if(angle==3)
					cx-=atoi(orders[i].substr(8).c_str());
				if(!(cy>=0&&cx>=0&&cy<h&&cx<w)){
					cy=max(0,cy);
					cx=max(0,cx);
					cy=min(h-1,cy);
					cx=min(w-1,cx);
				}
			}
			else if(orders[i]=="RIGHT"){
				angle++;
				angle%=4;
			}
			else if(orders[i]=="LEFT"){
				angle--;
				angle=(angle+4)%4;
			}
			else if(orders[i].substr(0,8)=="BACKWARD"){
				int px=cx;
				int py=cy;
				if(angle==2)
					cy+=atoi(orders[i].substr(9).c_str());
				else if(angle==3)
					cx+=atoi(orders[i].substr(9).c_str());
				else if(angle==0)
					cy-=atoi(orders[i].substr(9).c_str());
				else if(angle==1)
					cx-=atoi(orders[i].substr(9).c_str());
				if(!(cy>=0&&cx>=0&&cy<h&&cx<w)){
					cy=max(0,cy);
					cx=max(0,cx);
					cy=min(h-1,cy);
					cx=min(w-1,cx);
				}
			}
			else if(orders[i]=="STOP"){
				break;
			}
		}
		cout<<cx+1<<" "<<cy+1<<endl;
	}

	return 0;
}
