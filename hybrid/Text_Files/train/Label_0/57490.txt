class Task {

//    List<List<Integer>> g = new ArrayList<>();
    //    static int MAX = 1000;
//    boolean[] used;

    int n;

    public void solve(Scanner in, PrintWriter out) {
        n = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }

        Map<Long, Integer> prefa = new HashMap<>();
        Map<Long, Integer> prefb = new HashMap<>();
        Map<Long, Integer> prefab = new HashMap<>();
        long suma = 0;
        long sumb = 0;
        long sumab = 0;

        prefa.put(0L, 0);
        prefb.put(0L, 0);
        prefab.put(0L, 0);
        Map<Vect, Integer> dp = new HashMap<>();
        dp.put(new Vect(0, 0), 0);
        for (int i = 0; i < n; i++) {
            suma += a[i];
            sumb += b[i];
            sumab += a[i] + b[i];
            if (prefa.containsKey(suma)) {
                Map<Vect, Integer> copy = new HashMap<>(dp);
                for (Map.Entry<Vect, Integer> t : dp.entrySet()) {
                    if (prefa.get(suma) < t.getKey().a) {
                        continue;
                    }
                    copy.merge(new Vect(i + 1, t.getKey().b), t.getValue() + 1, Math::max);
                    copy.remove(t.getKey());
                }
                dp = copy;
            }

            if (prefb.containsKey(sumb)) {
                Map<Vect, Integer> copy = new HashMap<>(dp);
                for (Map.Entry<Vect, Integer> t : dp.entrySet()) {
                    if (prefb.get(sumb) < t.getKey().b) {
                        continue;
                    }
                    copy.merge(new Vect(t.getKey().a, i + 1), t.getValue() + 1, Math::max);
                    copy.remove(t.getKey());
                }
                dp = copy;
            }

            if (prefab.containsKey(sumab)) {
                Map<Vect, Integer> copy = new HashMap<>(dp);
                for (Map.Entry<Vect, Integer> t : dp.entrySet()) {
                    if (prefab.get(sumab) < t.getKey().a || prefab.get(sumab) < t.getKey().b) {
                        continue;
                    }
                    copy.merge(new Vect(i + 1, i + 1), t.getValue() + 1, Math::max);
                }
                dp = copy;
            }

            int max = dp.entrySet().stream().mapToInt(Map.Entry::getValue).max().getAsInt();
            dp = dp.entrySet().stream().filter(t -> t.getValue() > max - 2).collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));

            prefa.put(suma, i+1);
            prefb.put(sumb, i+1);
            prefab.put(sumab, i+1);
        }

        int max = dp.entrySet().stream().mapToInt(Map.Entry::getValue).max().getAsInt();
        System.out.println(max);
    }

}


class Utils {

    public static int binarySearch(int[] a, int key) {
        int s = 0;
        int f = a.length;

        while (f > s) {
            int mid = (s + f) / 2;
            if (a[mid] > key) {
                f = mid - 1;
            } else if (a[mid] <= key) {
                s = mid + 1;
            }
        }
        return -1;
    }
}

class Pair<T, U> {
    public T a;
    public U b;

    public Pair(T a, U b) {
        this.a = a;
        this.b = b;
    }

    @Override
    public boolean equals(Object obj) {
        Pair<T, U> p = (Pair<T, U>) obj;
        return p.a.equals(a) || p.a.equals(b);
    }

    @Override
    public int hashCode() {
        return 42;
    }
}


class Vect {
    public int a;
    public int b;

    public Vect(int a, int b) {
        this.a = a;
        this.b = b;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        Vect vect = (Vect) o;

        if (a != vect.a) return false;
        return b == vect.b;

    }

    @Override
    public int hashCode() {
        int result = a;
        result = 31 * result + b;
        return result;
    }
}

class Triple<T, U, P> {
    public T a;
    public U b;
    public P c;

    public Triple(T a, U b, P c) {
        this.a = a;
        this.b = b;
        this.c = c;
    }
}


class Scanner {
    BufferedReader in;
    StringTokenizer tok;

    public Scanner(InputStream in) {
        this.in = new BufferedReader(new InputStreamReader(in));
        tok = new StringTokenizer("");
    }

    private String tryReadNextLine() {
        try {
            return in.readLine();
        } catch (Exception e) {
            throw new InputMismatchException();
        }
    }

    public String nextToken() {
        while (!tok.hasMoreTokens()) {
            tok = new StringTokenizer(next());
        }
        return tok.nextToken();
    }

    public String next() {
        String newLine = tryReadNextLine();
        if (newLine == null)
            throw new InputMismatchException();
        return newLine;
    }

    public int nextInt() {
        return Integer.parseInt(nextToken());
    }

    public long nextLong() {
        return Long.parseLong(nextToken());
    }
}
