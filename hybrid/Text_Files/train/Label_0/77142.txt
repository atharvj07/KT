q = 998244353
ans = []
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	pos = {a: i for (i, a) in enumerate(A, 1)}
	u = {i: 0 for i in range(1, n + 1)}
	for b in B:
		u[pos[b]] = 1
	u[0] = u[n + 1] = 1
	cnt = 0
	for b in B:
		cur = 2 - u[pos[b] - 1] - u[pos[b] + 1]
		if not cur:
			ans.append(0)
			break
		elif cur == 2:
			cnt += 1
		u[pos[b]] = 0
	else:
		ans.append(pow(2, cnt, q))
print(*ans, sep='\n')
