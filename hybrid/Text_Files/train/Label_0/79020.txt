import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def judge(remove, arr, k):
	n = len(arr)
	dp = [[0 for j in range(remove + 1)] for i in range(n + 1)]
	for i in range(n - 1, -1, -1):
		limit = min(i, remove)
		for j in range(limit + 1):
			if arr[i] == j:
				base = 1
			else:
				base = 0
			num1 = base + dp[i + 1][j]
			num2 = -2147483647
			if j + 1 <= remove:
				num2 = dp[i + 1][j + 1]
			dp[i][j] = max(num1, num2)
	if dp[0][0] >= k:
		return True
	else:
		return False

def main(t):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	for i in range(n):
		arr[i] = i + 1 - arr[i]
	ans = 0
	front = 0
	rear = n - k + 1
	while front < rear:
		mid = (front + rear) // 2
		if judge(mid, arr, k):
			rear = mid
		else:
			front = mid + 1
	if front == n - k + 1:
		print(-1)
	else:
		print(front)
T = int(input())
t = 1
while t <= T:
	main(t)
	t += 1
