n = int(input())
d1 = {'S': [0, 0, 0, 0], 'M': [0, 0, 0, 0], 'L': [0, 0, 0, 0]}
d2 = {'S': [0, 0, 0, 0], 'M': [0, 0, 0, 0], 'L': [0, 0, 0, 0]}
for i in range(n):
	s = input()
	d1[s[-1]][len(s) - 1] += 1
for i in range(n):
	s = input()
	d2[s[-1]][len(s) - 1] += 1
print(sum((sum((abs(d1[i][j] - d2[i][j]) for j in range(4))) for i in d1)) // 2)
