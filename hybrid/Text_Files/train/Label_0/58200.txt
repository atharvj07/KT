import java.util.*;
import java.io.*;
import javafx.util.*;

public class DAB 
{	
	static class Task 
	{
		final int N = (int)1e5 + 5 ;
		final long INF = (long)1e18;
		long T[][] = new long[N << 2][3] , lzy[] = new long[N << 2] , a[] = new long[N] , b[] = new long[N];
		long[] ac = new long[N] , acf = new long[N];
		long n , q , x , y;
		char ch;
		
		public void solve(Scanner sc , PrintWriter pw) throws IOException 
        {
			for(int i = 0 ; i < (N << 2) ; i++)
				lzy[i] = INF;
			
			n = sc.nextLong();
			for(int i = 1 ; i <= n ; i++)
				a[i] = sc.nextLong();
			for(int i = 1 ; i <= n - 1 ; i++)	
				b[i] = sc.nextLong();
			
			for(int i = (int)n ; i >= 0 ; i--)
				ac[i] += ac[i + 1] + b[i];
			
			for(int i = (int) n ; i >= 0 ; i--)
				acf[i] += ac[i] + acf[i + 1];
			
			build(1 , (int)n , 1);
			
			q = sc.nextLong();
			for(int i = 0 ; i < q ; i++)
			{
				ch = sc.next().charAt(0);
				x = sc.nextLong();
				y = sc.nextLong();
				
				if(ch == 's')
					pw.println(qry((int)x , (int)y , 1 , (int)n , 1) - acf[(int) x] + acf[(int)y + 1]);
				else
				{
					udt((int)x ,  (int)n , y + qry((int)x , (int)x , 1 , (int)n , 1) , 1 , (int) n , 1);
				}
			}
        }
			
		public void push(int s , int e , int i)
		{
			if(lzy[i] != INF)
			{
				T[i][0] = (long)(e - s + 1) * lzy[i];
				T[i][1] = T[i][2] = lzy[i];
				if(s != e)
					lzy[i << 1] = lzy[i << 1 | 1] = lzy[i];
			}
			lzy[i] = INF;
		}
		
		public void udtTreeVal(int i)
		{
			T[i][0] = T[i << 1][0] + T[i << 1 | 1][0];
			T[i][1] = Math.min(T[i << 1][1] , T[i << 1 | 1][1]);
			T[i][2] = Math.max(T[i << 1][2] , T[i << 1 | 1][2]);
		}
		
		public void udt(int l , int r , long val , int s , int e , int i)
		{
			push(s , e , i);
			if(r < s || l > e) return;
			if(l <= s && r >= e)
			{
				if(T[i][2] < val)
				{
					lzy[i] = val;
					push(s , e , i);
					return;
				}
				else if(T[i][1] >= val) 
					return;
			}
			
			int md = (s + e) / 2;
			udt(l , r , val , s , md , i << 1);
			udt(l , r , val , md + 1 , e , i << 1 | 1);
		
			udtTreeVal(i);
		}
		
		public void build(int s , int e , int i)
		{
			if(s == e)
			{
				T[i][0] = T[i][1] = T[i][2] = ac[s] + a[s];
				return;
			}
			
			int md = (s + e) / 2;
			build(s , md , i << 1);
			build(md + 1 , e , i << 1 | 1);
			
			udtTreeVal(i);
		}
				
		public long qry(int l , int r , int s , int e , int i)
		{
			push(s , e , i);
			if(r < s || l > e) return 0;
			if(l <= s && r >= e)
				return T[i][0];
			
			int md = (s + e) / 2;
			return qry(l , r , s , md , i << 1) + qry(l , r , md + 1 , e , i << 1 | 1);
		}
	}
	
	public static void main(String[] args) throws IOException 
	{
		Scanner sc = new Scanner(System.in);
        PrintWriter pw = new PrintWriter(System.out);
        Task task = new Task();
        task.solve(sc , pw);
        pw.close();
    }
	
	static class Scanner 
	{
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s){	br = new BufferedReader(new InputStreamReader(s));}

		public String next() throws IOException 
		{
			while (st == null || !st.hasMoreTokens()) 
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {return Integer.parseInt(next());}
		
		public long nextLong() throws IOException {return Long.parseLong(next());}

		public String nextLine() throws IOException {return br.readLine();}
		
		public double nextDouble() throws IOException
		{
			String x = next();
			StringBuilder sb = new StringBuilder("0");
			double res = 0, f = 1;
			boolean dec = false, neg = false;
			int start = 0;
			if(x.charAt(0) == '-')
			{
				neg = true;
				start++;
			}
			for(int i = start; i < x.length(); i++)
				if(x.charAt(i) == '.')
				{
					res = Long.parseLong(sb.toString());
					sb = new StringBuilder("0");
					dec = true;
				}
				else
				{
					sb.append(x.charAt(i));
					if(dec)
						f *= 10;
				}
			res += Long.parseLong(sb.toString()) / f;
			return res * (neg?-1:1);
		}
		
		public boolean ready() throws IOException {return br.ready();}
	}
	
	static class pair implements Comparable<pair>
	{
		public int f , s;
		pair(int f , int s)
		{
			this.f = f;
			this.s = s;
		}
		@Override
		public int compareTo(pair p) {
			return f - p.f;
		}
	}
	
	static public long gcd(long a, long b) 
    { 
      if (b == 0) 
        return a; 
      return gcd(b, a % b);  
    } 
}
