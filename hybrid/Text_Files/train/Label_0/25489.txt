#include <bits/stdc++.h>
using namespace std;
char s[105], t[105];
int main() {
  scanf("%s", s);
  getchar();
  scanf("%s", t);
  int maxlen = strlen(s);
  int state = 0;
  for (int i = 0; i < maxlen; i += 1) {
    if (!state) {
      if (t[i] - s[i] == 1) {
        if (i == maxlen - 1) {
          printf("No such string\n");
          return 0;
        } else if (t[i + 1] != 'a') {
          for (int j = 0; j <= i; j += 1) printf("%c", t[j]);
          for (int j = i + 1; j < maxlen; j += 1) printf("%c", t[i + 1] - 1);
          return 0;
        } else if (s[i + 1] != 'z') {
          for (int j = 0; j <= i; j += 1) printf("%c", s[j]);
          for (int j = i + 1; j < maxlen; j += 1) printf("%c", s[i + 1] + 1);
          return 0;
        } else
          state = 1;
      } else if (t[i] - s[i] >= 2) {
        for (int j = 0; j < i; j += 1) {
          printf("%c", t[j]);
        }
        for (int j = i; j < maxlen; j += 1) {
          printf("%c", t[i] - 1);
        }
        printf("\n");
        return 0;
      }
    } else {
      if (t[i] != 'a') {
        for (int j = 0; j < i; j += 1) printf("%c", t[j]);
        for (int j = i; j < maxlen; j += 1) printf("%c", t[i] - 1);
        return 0;
      } else if (s[i] != 'z') {
        for (int j = 0; j < i; j += 1) printf("%c", s[j]);
        for (int j = i; j < maxlen; j += 1) printf("%c", s[i] + 1);
        return 0;
      }
    }
  }
  printf("No such string\n");
  return 0;
}
