n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
c = sorted([[a[i], 1] for i in range(n)] + [[b[i], 2] for i in range(m)])
x = x0 = 2 * n
y = y0 = 2 * m
for i in range(n + m - 1, -1, -1):
	if c[i][1] == 1:
		x += 1
	else:
		y += 1
	if x - y >= x0 - y0:
		(x0, y0) = (x, y)
print(x0, y0, sep=':')
