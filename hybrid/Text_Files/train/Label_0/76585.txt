n = int(input())
l = []
for i in range(n):
	(x1, p1) = map(int, input().split())
	(x2, p2) = map(int, input().split())
	while p1 != 0 and x1 < 1000000:
		x1 *= 10
		p1 -= 1
	while p2 != 0 and x2 < 1000000:
		x2 *= 10
		p2 -= 1
	if p1 > p2:
		l.append('>')
	elif p1 < p2:
		l.append('<')
	elif p1 == p2:
		if x1 > x2:
			l.append('>')
		elif x1 < x2:
			l.append('<')
		else:
			l.append('=')
for i in l:
	print(i)
