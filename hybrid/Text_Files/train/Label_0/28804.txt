void solve() {
  ll n, m; cin >> n >> m;
  vector<vector<bool>> G(n, vector<bool>(n, false));
  rep(i, m) {
    ll a, b; cin >> a >> b; --a, --b;
    G[a][b] = G[b][a] = true;
  }
  vector<vector<bool>> used(n, vector<bool>(n, false));
  vector<bool> vused(n, false);
  vused[0] = true;
  ll v = 0, cnt = 0, vcnt = 1;
  while (vcnt < n || cnt < m || v != 0) {
    bool is_update = false;
    rep(to, n) {
      if (v == to) continue;
      if (!used[v][to] && G[v][to]) {
        used[v][to] = used[to][v] = true;
        ++cnt;
        v = to;
        is_update = true;
        break;
      }
    }
    if (!is_update) {
      rep(to, n) {
        if (v == to) continue;
        if (!used[v][to]) {
          used[v][to] = used[to][v] = true;
          v = to;
          is_update = true;
          break;
        }
      }
    }
    if (!is_update) {
      cout << -1 << endl;
      return;
    }
    if (!vused[v]) {
      vused[v] = true;
      ++vcnt;
    }
    // cout << v << endl;
  }
  vector<P> ans;
  rep(i, n) rep(j, i) {
    if (used[i][j] && !G[i][j]) {
      ans.eb(j, i);
    }
  }
  cout << ans.size() << endl;
  each(p, ans) {
    cout << p.first+1 << " " << p.second+1 << endl;
  }
}
