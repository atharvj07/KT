#include <bits/stdc++.h>
using namespace std;
const int INF = (int)1e9 + 7;
const long long LINF = (long long)1e18 + 9;
const int MXN = (int)2e6 + 10;
const double EPS = (double)1e-6;
int n, m, p;
int a[MXN], b[MXN];
long double check_sum;
long double pref[MXN];
vector<int> ans;
int main() {
  scanf("%d%d%d", &n, &m, &p);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", &a[i]);
  }
  int first;
  for (int j = 0; j < m; ++j) {
    scanf("%d", &first);
    check_sum += sqrt((long double)(1.0 * first));
  }
  for (int i = 1; i <= n; ++i) {
    pref[i] = sqrt((long double)(1.0 * a[i]));
    if (i - p >= 0) pref[i] += pref[i - p];
  }
  long double del, cur;
  for (int i = 1; i <= n; ++i) {
    if (1LL * i + (m - 1) * 1LL * p <= 1LL * n) {
      del = (i - p >= 0 ? pref[i - p] : .0);
      cur = pref[i + (m - 1) * p];
      if (fabs(cur - del - check_sum) < EPS) {
        ans.push_back(i);
      }
    }
  }
  printf("%d\n", (int)ans.size());
  for (auto it : ans) printf("%d ", it);
  return 0;
}
