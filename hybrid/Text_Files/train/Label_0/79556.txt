def solution(s, l, a, b):
	if b >= 0 or l == 1:
		return (a + b) * l
	else:
		c = 0
		for i in range(1, l):
			if s[i] != s[i - 1]:
				c += 1
		return ((c + 1) // 2 + 1) * b + a * l
n = int(input())
for _ in range(n):
	(l, a, b) = tuple(map(int, input().split()))
	s = input()
	print(solution(s, l, a, b))
