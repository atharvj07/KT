input()
aa = list(map(int, input().split()))
ans = []
n = len(aa)
saa = sum(aa)
if saa & 1 == 0:
	k = saa // 2
	dp = [[False] * (k + 1) for _ in range(n + 1)]
	for i in range(n + 1):
		dp[i][0] = True
	for i in range(1, n + 1):
		for cs in range(1, k + 1):
			dp[i][cs] = dp[i - 1][cs]
			if aa[i - 1] <= cs:
				dp[i][cs] |= dp[i - 1][cs - aa[i - 1]]
	if dp[n][k]:
		x = 2
		while not ans:
			for (i, a) in enumerate(aa):
				if a % x != 0:
					ans = [i]
					break
			x *= 2
print(len(ans))
if ans:
	ans[0] += 1
	print(' '.join(map(str, ans)))
