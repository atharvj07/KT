(n, m) = map(int, input().split())
a = [0] * n
for i in range(n):
	a[i] = input()
ans = ['.'] * n
for i in range(n):
	ans[i] = ['.'] * m
for i in range(1, n - 1):
	for j in range(1, m - 1):
		if a[i - 1][j - 1] == '#' and a[i - 1][j] == '#' and (a[i - 1][j + 1] == '#') and (a[i][j - 1] == '#') and (a[i][j + 1] == '#'):
			if a[i + 1][j - 1] == '#' and a[i + 1][j] == '#' and (a[i + 1][j + 1] == '#'):
				ans[i - 1][j - 1] = '#'
				ans[i + 1][j - 1] = '#'
				ans[i + 1][j] = '#'
				ans[i + 1][j + 1] = '#'
				ans[i - 1][j] = '#'
				ans[i - 1][j + 1] = '#'
				ans[i][j - 1] = '#'
				ans[i][j + 1] = '#'
for i in range(n):
	for j in range(m):
		if a[i][j] != ans[i][j]:
			print('NO')
			exit()
print('YES')
