import java.util.Scanner;
public class stones1
{
    public static void main(String args[])
    {
        Scanner s= new Scanner(System.in);
        int n= s.nextInt();
        s.nextLine();
        String a= s.nextLine();
        int i, j, k=0;
        for(i=0; i<a.length() ; i++)
        {
            if(a.charAt(i)=='+')
            {
                k++;
            }
            if(a.charAt(i)=='-' && k!=0)
            {
                k--;
            }
        }
        System.out.println(k);
    }
}
