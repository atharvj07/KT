import java.util.*;
import java.io.*;

public class Solution{
	long rem = 1000000007L;
	
	public Solution(){
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		
		if(n == 1){
			System.out.println(0);
			return;
		}
		
		int arrLen = 1;
		int[] arr = new int[n];
		
		arr[0] = 2;
		int num = arr[0]*arr[0];
		while(num <= n){
			arr[arrLen] = num;
			num *= arr[0];
			arrLen++;
		}
		
		boolean prime = true;
		for(int i=3; i<=n; i++){
			prime = true;
			for(int j = 0; j<arrLen; j++){
				if(i%arr[j] == 0){
					prime = false;
					break;
				}
			}
			
			if(prime){
				arr[arrLen] = i;
				arrLen++;
				
				num = i*i;
				while(num <= n){
					arr[arrLen] = num;
					num *= i;
					arrLen++;
				}
			}
		}
		
		System.out.println(arrLen);
		for(int i=0; i<arrLen-1; i++){
			System.out.print(arr[i] + " ");
		}
		System.out.println(arr[arrLen-1]);
	}
	
	public static void main(String[] args){
		new Solution();
	}
}
