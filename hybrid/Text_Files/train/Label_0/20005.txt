#include <bits/stdc++.h>
using namespace std;
int main() {
  int N;
  vector<string> a;
  vector<int> b, c;
  cin >> N;
  for (int _n(N), i(0); i < _n; i++) {
    string s;
    int bb, cc;
    cin >> s >> bb >> cc;
    if ((int)((s).size()) == 3) s = "0" + s;
    a.push_back(s), b.push_back(bb), c.push_back(cc);
  }
  int match = 0;
  string found;
  for (int _n(10), i1(0); i1 < _n; i1++)
    for (int _n(10), i2(0); i2 < _n; i2++)
      for (int _n(10), i3(0); i3 < _n; i3++)
        for (int _n(10), i4(0); i4 < _n; i4++)
          if (i1 != i2 && i1 != i3 && i1 != i4 && i2 != i3 && i2 != i4 &&
              i3 != i4) {
            string s(4, '_');
            s[0] = i1 + '0';
            s[1] = i2 + '0';
            s[2] = i3 + '0';
            s[3] = i4 + '0';
            bool ok = true;
            for (int _n(N), n(0); n < _n; n++) {
              int bulls = 0;
              vector<char> s1, s2, r(4);
              for (int _n(4), i(0); i < _n; i++)
                if (a[n][i] == s[i])
                  ++bulls;
                else
                  s1.push_back(a[n][i]), s2.push_back(s[i]);
              sort((s1).begin(), (s1).end()), sort((s2).begin(), (s2).end());
              vector<char>::iterator it =
                  set_intersection((s1).begin(), (s1).end(), (s2).begin(),
                                   (s2).end(), r.begin());
              if (bulls != b[n] || c[n] != it - r.begin()) ok = false;
            }
            if (ok) ++match, found = s;
          }
  if (!match)
    cout << "Incorrect data" << endl;
  else if (match > 1)
    cout << "Need more data" << endl;
  else
    cout << found << endl;
  return 0;
}
