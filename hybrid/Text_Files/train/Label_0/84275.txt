input = __import__('sys').stdin.readline
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	if n == 1:
		print(0)
	else:
		ans = 1
		level = {}
		i = 1
		k = 1
		while i < n:
			if a[i] > a[i - 1]:
				if k + 1 in level:
					level[k + 1] += 1
				elif k in level:
					level[k] += 1
				else:
					level[k] = 1
			elif level[k]:
				level[k] -= 1
				if k + 1 in level:
					level[k + 1] += 1
				else:
					level[k + 1] = 1
			else:
				k += 1
				level[k + 1] = 1
				level[k] -= 1
			i += 1
		print(max(level))
