import math
t = int(input())
q = []
for i in range(t):
	q.append(list(map(int, input().rstrip().split())))

def pogl(h):
	h = math.floor(h / 2) + 10
	return h
for i in range(t):
	x = q[i][0]
	n = q[i][1]
	m = q[i][2]
	for i in range(n):
		if x > 20:
			x = pogl(x)
		else:
			break
	if x / 10 <= m:
		print('YES')
	else:
		print('NO')
