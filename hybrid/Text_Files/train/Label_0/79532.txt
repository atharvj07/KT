(n, m, k) = [int(x) for x in input().split()]
adj = [[] for _ in range(n + 1)]
for _ in range(m):
	(u, v) = [int(x) for x in input().split()]
	adj[u].append(v)
	adj[v].append(u)
mark = [False for _ in range(n + 1)]
c = [1]
while True:
	mark[c[-1]] = True
	for v in adj[c[-1]]:
		if not mark[v]:
			c.append(v)
			break
	else:
		mark = [False for _ in range(n + 1)]
		for v in adj[c[-1]]:
			mark[v] = True
		for i in range(len(c)):
			if mark[c[i]]:
				start = i
				break
		print(len(c) - start)
		print(' '.join(map(str, c[start:])))
		break
