#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  {
    int a, b, x;
    cin >> a >> b >> x;
    string s = "";
    int i = 0;
    for (; i < x;) {
      s += '0';
      s += '1';
      a--, b--;
      i += 2;
    }
    if (i == x) {
      if (b == 0) {
        for (; a > 0; a--) s += '0';
      } else if (a == 0) {
        for (; b > 0; b--) s = '1' + s;
      } else {
        for (; b > 0; b--) s += '1';
        s += '0';
        a--;
        for (; a > 0; a--) s = '0' + s;
      }
    }
    for (int i = 0; i < a; i++) {
      s = '0' + s;
    }
    for (int i = 0; i < b; i++) {
      s += '1';
    }
    cout << s << "\n";
  }
  return 0;
}
