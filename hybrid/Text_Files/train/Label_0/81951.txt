from collections import Counter, deque
from sys import stdout
import time
from math import factorial, log

def main():
	(n, m, k) = [int(i) for i in input().split()]
	matrix = [input() for i in range(n)]
	_all = 0
	_b = set()
	for y in range(n):
		for x in range(m):
			flag = False
			b = set()
			element = matrix[y][x]
			if element == '*':
				_all += 1
				if x >= k and y >= k:
					for d in range(1, y + 1):
						el1 = matrix[y - d][x - d] if x >= d and y - d >= 0 else False
						el2 = matrix[y - d][x + d] if x + d < m and y - d >= 0 else False
						(el1, el2) = map(lambda z: False if not z or z == '.' else True, [el1, el2])
						if el1 and el2:
							if d >= k:
								flag = True
							b.add((y - d, x + d))
							b.add((y - d, x - d))
						else:
							break
					if flag:
						_b.add((y, x))
						_b |= b
	if len(_b) == _all:
		print('YES')
	else:
		print('NO')
for t in range(int(input())):
	main()
