import sys
input = sys.stdin.readline
(n, k) = map(int, input().split())
arr = list(map(int, input().split()))
post = []
ok1 = []
ok2 = []
for i in range(n):
	post.append(0)
	ok1.append(0)
	ok2.append(0)
ok1[n - 1] = 1
ok2[0] = 1
count = 0
for i in range(k - 1, -1, -1):
	post[arr[i] - 1] = 1
	if arr[i] < n and post[arr[i]] == 1:
		ok1[arr[i] - 1] = 1
	if arr[i] > 1 and post[arr[i] - 2] == 1:
		ok2[arr[i] - 1] = 1
count += 2 * n - sum(ok1) - sum(ok2)
for i in range(n):
	if post[i] == 0:
		count += 1
print(count)
