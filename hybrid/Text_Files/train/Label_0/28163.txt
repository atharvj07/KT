#include <bits/stdc++.h>
const long long INF = (1LL << 61);
const int inf = 2e9;
using namespace std;
long long a[400500], n;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  long long ans = 0;
  for (int i = 0; i < n; i++) ans = (ans + a[i] * a[n - i - 1]) % 10007;
  cout << ans << endl;
}
