import java.util.Scanner;
import java.math.BigDecimal;
import java.util.Arrays;
import java.util.*;
import java.io.*;
public class Main {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		long t=sc.nextLong();
		while(t--!=0) {
			long n=sc.nextLong();
			long k=sc.nextLong();
			long x=k/2;
			long sum=0;
			for(long i=k+1;i<=n;i++) {
				sum++;
			}
			sum+=x;
			System.out.println(sum);
			for(long i=k-1,j=0;j<x;j++,i--) {
				System.out.print(i+" ");
			}
			for(long i=k+1;i<=n;i++) {
				sum++;
				System.out.print(i+" ");
			}
			System.out.print("\n");
	}
}
}
