def solve():
	(n, k) = list(map(int, input().split()))
	k += 1
	ans = 0
	arr = list(map(int, input().split()))
	for i in range(n - 1):
		if k > 0:
			x = pow(10, arr[i + 1] - arr[i]) - 1
			val = min(x, k)
			ans += pow(10, arr[i]) * val
			k -= val
	ans += k * pow(10, arr[n - 1])
	print(ans)
t = int(input())
for i in range(t):
	solve()
