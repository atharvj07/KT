h = int(input())
a = list(map(int, input().split()))
perfect = True
for i in range(1, len(a)):
	if a[i] != 1 and a[i - 1] != 1:
		perfect = False
print('perfect' if perfect else 'ambiguous')
ret1 = [0] * sum(a)
ret2 = [0] * sum(a)
node = 0
p = 0
if perfect == False:
	for i in range(len(a)):
		for j in range(a[i]):
			ret1[node] = p
			node += 1
		p = node
	p1 = 0
	p2 = 0
	node = 0
	for i in range(len(a)):
		for j in range(a[i]):
			if j & 1:
				ret2[node] = p1
			else:
				ret2[node] = p2
			node += 1
		p1 = node
		p2 = node
		if a[i] != 1:
			p1 = node
			p2 = node - 1
	print(' '.join((str(x) for x in ret1)))
	print(' '.join((str(x) for x in ret2)))
