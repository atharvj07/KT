#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, i;
  cin >> n;
  long long int arr[n + 1];
  vector<pair<long long int, long long int>> v;
  for (i = 0; i < n; i++) {
    long long int temp;
    cin >> temp;
    v.push_back({temp, i});
  }
  sort(v.begin(), v.end());
  reverse(v.begin(), v.end());
  i = 0;
  long long int sum = 0;
  for (auto it : v) {
    sum += it.first * i + 1;
    arr[i] = it.second + 1;
    i++;
  }
  cout << sum << endl;
  for (i = 0; i < n; i++) {
    cout << arr[i] << " ";
  }
  return 0;
}
