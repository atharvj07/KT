#include<bits/stdc++.h>
#define maxn 300005
using namespace std;
struct node{int l,r;}d[maxn];
int cmp(node a,node b){return a.l==b.l?a.r<b.r:a.l<b.l;}
int st[maxn],n,m;
priority_queue<int,vector<int>,greater<int> > q;
int main()
{
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;i++) scanf("%d%d",&d[i].l,&d[i].r);
	sort(d+1,d+n+1,cmp);
	int nl=1,nr=m;
	for(int i=1;i<=n;i++)
	{
		q.push(d[i].r);
		if(nl<=nr&&nl<=d[i].l) nl++;
		else
		{
			st[++st[0]]=q.top();
			q.pop();
		}
	}
	sort(st+1,st+st[0]+1);
	int ans=0;
	for(int i=st[0];i>=1;i--)
	{
		if(nl<=nr&&st[i]<=nr) nr--;
		else ans++;
	}
	printf("%d\n",ans);
	return 0;
}
