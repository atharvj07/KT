t = int(input())
for _ in range(t):
	(n, k) = list(map(int, input().split()))
	a = []
	b = []
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	a.sort()
	b.sort(reverse=True)
	j = 0
	while k > 0 and j < n:
		if a[j] <= b[j]:
			(a[j], b[j]) = (b[j], a[j])
			k -= 1
		j += 1
	print(sum(a))
