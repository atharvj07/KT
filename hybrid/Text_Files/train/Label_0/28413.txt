#include <bits/stdc++.h>
using namespace std;
int cur, n, s, t, cnt, pos[100001], visited[1000001];
int main() {
  scanf("%d%d%d", &n, &s, &t);
  ;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &pos[i]);
    ;
  }
  if (s == t) {
    printf("0\n");
    return 0;
  }
  cur = pos[s];
  visited[s] = 1;
  cnt = 1;
  while (true) {
    if (visited[cur] == 1) {
      printf("-1\n");
      return 0;
    }
    if (cur == t) {
      printf("%d\n", cnt);
      return 0;
    }
    cur = pos[cur];
    cnt++;
  }
  return 0;
}
