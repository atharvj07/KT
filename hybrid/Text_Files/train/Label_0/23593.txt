#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, t, tmp;
  string s;
  vector<int> out;
  cin >> t;
  for (int j = 0; j < t; ++j) {
    cin >> n >> s;
    tmp = n;
    bool both = s.find('<') != string::npos && s.find('>') != string::npos;
    for (int i = 0; i < n; ++i) {
      if (s[i] != '-' && s[(i + 1) % n] != '-' &&
          (s[i] != s[(i + 1) % n] || both)) {
        tmp -= 1;
      }
    }
    out.push_back(tmp);
  }
  for (auto j : out) {
    cout << j << "\n";
  }
  return 0;
}
