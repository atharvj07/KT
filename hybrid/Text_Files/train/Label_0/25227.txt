#include <bits/stdc++.h>
const double eps = 1e-8;
using namespace std;
inline long long read() {
  char ch = getchar();
  long long s = 0, w = 1;
  while (ch < 48 || ch > 57) {
    if (ch == '-') w = -1;
    ch = getchar();
  }
  while (ch >= 48 && ch <= 57) {
    s = (s << 1) + (s << 3) + (ch ^ 48);
    ch = getchar();
  }
  return s * w;
}
inline void write(long long x) {
  if (x < 0) putchar('-'), x = -x;
  if (x > 9) write(x / 10);
  putchar(x % 10 + 48);
}
string s;
char a[10];
map<char, long long> p;
int main() {
  cin >> s;
  for (register long long i = 0; s[i]; i++) {
    if (s[i] != '!') a[(i) % 4 + 1] = s[i];
  }
  for (register long long i = 0; s[i]; i++) {
    if (s[i] == '!') {
      p[a[(i) % 4 + 1]]++;
    }
  }
  cout << p['R'] << " " << p['B'] << " " << p['Y'] << " " << p['G'] << "\n";
  return 0;
}
