import java.io.*;
import java.util.*;

public class A {
    static FastReader f = new FastReader();

    public static void main(String[] args) {
        long[] arr = new long[6];
        arr[0] = f.nextInt();
        arr[1] = f.nextInt();
        for(int i=2;i<6;i++) {
            arr[i] = arr[i-1] - arr[i-2];
        }

        int n = f.nextInt() - 1;
        System.out.println((((arr[n%6])%1000000007)+1000000007)%1000000007);

    }






    //fast input reader
    static class FastReader
    {
        BufferedReader br;
        StringTokenizer st;

        public FastReader()
        {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException e)
                {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt()
        {
            return Integer.parseInt(next());
        }

        long nextLong()
        {
            return Long.parseLong(next());
        }

        double nextDouble()
        {
            return Double.parseDouble(next());
        }

        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                e.printStackTrace();
            }
            return str;
        }
    }
}
