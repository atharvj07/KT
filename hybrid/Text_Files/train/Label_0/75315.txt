ar = list(map(int, input().split()))
ss = 'abcdefghijklmnopqrstuvwxyz'
ma = {}
for i in range(26):
	ma[ss[i]] = ar[i]
st = input()
dic = {}
le = len(st)
for i in range(le):
	if st[i] in dic:
		dic[st[i]].append(i)
	else:
		dic[st[i]] = [i]
br = [ma[st[0]]]
for i in range(1, le):
	br.append(br[-1] + ma[st[i]])
ans = 0
for i in dic:
	tem = {}
	li = []
	le = len(dic[i])
	for j in dic[i]:
		sco = br[j] - ma[i]
		if sco in tem:
			ans += tem[sco]
		if br[j] in tem:
			tem[br[j]] += 1
		else:
			tem[br[j]] = 1
print(ans)
