using namespace std;

int n, m;
vector<vector<int> > a;
vector<pair<int, int> > s;

int dy[] = {0, 1, 0, -1};
int dx[] = {1, 0, -1, 0};

bool solve(int k, int y, int x, int rest)
{
    if(rest < 0)
        return false;

    if(rest == 0){
        if(k == m)
            return true;
        y = s[k].first;
        x = s[k].second;
        rest = -a[y][x];
        ++ k;
    }

    for(int i=0; i<4; ++i){
        int y2 = y + dy[i];
        int x2 = x + dx[i];
        if(y2 < 0 || y2 >= n || x2 < 0 || x2 >= n)
            continue;
        if(a[y2][x2] > 0){
            a[y2][x2] *= -1;
            if(solve(k, y2, x2, rest + a[y2][x2]))
                return true;
            a[y2][x2] *= -1;
        }
    }

    return false;
}

int main()
{
    for(;;){
        cin >> n;
        if(n == 0)
            return 0;

        a.assign(n, vector<int>(n));
        s.clear();
        m = 0;
        int sum = 0;
        for(int i=0; i<n; ++i){
            for(int j=0; j<n; ++j){
                cin >> a[i][j];
                sum += a[i][j];
                if(a[i][j] < 0){
                    s.push_back(make_pair(i, j));
                    ++ m;
                }
            }
        }

        if(sum != 0){
            cout << "NO" << endl;
            continue;
        }

        if(solve(0, -1, -1, 0))
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
