public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		FastScanner in = new FastScanner(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		TaskC solver = new TaskC();
		solver.solve(1, in, out);
		out.close();
	}
}

class TaskC {
    public void solve(int testNumber, FastScanner in, PrintWriter out) {
		n = in.nextInt();
		x = new int[n];
		y = new int[n];
		for (int i = 0; i < n; i++) {
			x[i] = in.nextInt();
			y[i] = in.nextInt();
		}
		if (signedArea() < 0) {
			for (int i = 0; i < n - i - 1; i++) {
				int j = n - i - 1;
				int t;
				t = x[i]; x[i] = x[j]; x[j] = t;
				t = y[i]; y[i] = y[j]; y[j] = t;
			}
		}
		long[][] d = new long[n][n];
		for (int dist = 1; dist < n; dist++) {
			for (int i = 0; i < n; i++) {
				int j = i + dist;
				if (j >= n) {
					j -= n;
				}
				if (dist == 1) {
					d[i][j] = 1;
					continue;
				}
				for (int k = next(i); k != j; k = next(k)) {
					if (triangleArea(i, k, j) > 0) {
						d[i][j] = (d[i][j] + d[i][k] * d[k][j]) % MOD;
					}
				}
			}
		}
		out.println(d[0][n - 1]);
    }

	final int MOD = 1000000007;
	int n;
	int[] x, y;

	private int next(int i) {
		return i == n - 1 ? 0 : i + 1;
	}

	private long signedArea() {
		long res = 0;
		for (int i = 0; i < n; i++) {
			res += triangleArea(0, i, next(i));
		}
		return res;
	}

	private long triangleArea(int i, int j, int k) {
		int x12 = x[j] - x[i];
		int x13 = x[k] - x[i];
		int y12 = y[j] - y[i];
		int y13 = y[k] - y[i];
		return cross(x12, y12, x13, y13);
	}

	private long cross(long x1, long y1, long x2, long y2) {
		return x1*y2 - x2*y1;
	}
}

class FastScanner {

	private BufferedReader in;
	private StringTokenizer st;

	public FastScanner(InputStream stream) {
		in = new BufferedReader(new InputStreamReader(stream));
	}

	public String next() {
		while (st == null || !st.hasMoreTokens()) {
			try {
				st = new StringTokenizer(in.readLine());
			} catch (IOException e) {
				throw new RuntimeException(e);
			}
		}
		return st.nextToken();
	}

	public int nextInt() {
		return Integer.parseInt(next());
	}

}
