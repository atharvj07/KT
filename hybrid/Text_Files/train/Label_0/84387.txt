n = int(input())
h = [0, 1]
for i2 in range(2):
	h[i2] = [int(i) for i in input().split()]
a = [[0 for i in range(n)] for i in range(2)]
if n == 1:
	print(max(h[0][0], h[1][0]))
else:
	a[0][0] = h[0][0]
	a[1][0] = h[1][0]
	a[0][1] = h[0][1] + a[1][0]
	a[1][1] = h[1][1] + a[0][0]
	for i in range(2, n):
		a[0][i] = max(h[0][i] + a[1][i - 1], h[0][i] + a[1][i - 2])
		a[1][i] = max(h[1][i] + a[0][i - 1], h[1][i] + a[0][i - 2])
	print(max(a[0][-1], a[1][-1], a[0][-2], a[1][-2]))
