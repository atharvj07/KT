t = int(input())
for k in range(t):
	(n, x) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	if k != t - 1:
		s = input()
	a.sort()
	b.sort()
	b = b[::-1]
	c = 0
	for i in range(n):
		if a[i] + b[i] <= x:
			c += 1
	if c == n:
		print('Yes')
	else:
		print('No')
