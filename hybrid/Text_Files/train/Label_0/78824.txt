from sys import *
from math import *
(n, k) = map(int, stdin.readline().split())
a = []
for i in range(n):
	x = int(input())
	a.append(x)
a.sort()
b = [a[0]]
for i in range(1, n):
	if a[i] != a[i - 1]:
		b.append(a[i])
cnt = [0] * len(b)
j = 0
for i in range(n):
	if a[i] == b[j]:
		cnt[j] += 1
	else:
		j += 1
		cnt[j] += 1
cnt.sort()
cnt = cnt[::-1]
x = n // 2
ans = 0
if n % 2 != 0:
	x += 1
ans = 0
for i in range(len(cnt)):
	x -= cnt[i] // 2
	x -= cnt[i] % 2
	ans += cnt[i]
print(ans + x)
