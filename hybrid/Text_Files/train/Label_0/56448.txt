import java.io.*;
import java.util.*;

public class C {
    public static void main(String[] args) throws Exception {
        new C().solve();
    }

    void solve() throws IOException {
        BufferedReader in = new BufferedReader(new
                InputStreamReader(System.in));
        String[] sp;
        // Scanner sc = new Scanner(System.in);

        int n = Integer.parseInt(in.readLine());
        long sum = 0;
        long[] value = new long[n + 10];
        int size = 1;
        long[] add = new long[value.length];

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sp = in.readLine().split(" ");
            int t = Integer.parseInt(sp[0]);
            if (t == 1) {
                int a = Integer.parseInt(sp[1]);
                long x = Integer.parseInt(sp[2]);
                add[a - 1] += x;
                sum += a * x;

            } else if (t == 2) {
                int k = Integer.parseInt(sp[1]);
                value[size++] = k;
                add[size - 1] = 0;
                sum += k;

            } else if (t == 3) {
                sum -= value[size - 1];
                sum -= add[size - 1];
                add[size - 2] += add[size - 1];
                add[size - 1] = 0;
                value[size - 1] = 0;
                size--;

            }
            sb.append((sum / (double) size) + "\n");
        }
        System.out.print(sb);
    }
}
