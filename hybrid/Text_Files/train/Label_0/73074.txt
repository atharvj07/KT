for i in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	z = a.copy()
	z.sort()
	p = dict()
	s = 0
	for i in z:
		if i not in p:
			p[i] = s
			s += 1
	C = [0] * s
	F = [0] * s
	r = 0
	for i in range(n):
		v = a[i]
		c = 0
		x = p[v]
		while x >= 0:
			c += F[x]
			x = (x & x + 1) - 1
		x = p[v]
		r += min(c - C[x], i - c)
		C[x] += 1
		while x < s:
			F[x] += 1
			x = x | x + 1
	print(r)
