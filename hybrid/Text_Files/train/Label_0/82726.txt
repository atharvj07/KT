s1 = input()
s2 = list(map(str, input()))
s3 = input()
s4 = ''
index = 0
for i in s3:
	s = ''
	if i.isupper():
		i = i.lower()
		index = s1.find(i)
		s = str(s2[index])
		s = s.upper()
		s4 += s
	elif i.islower():
		index = s1.find(i)
		s = str(s2[index])
		s = s.lower()
		s4 += s
	else:
		s4 += i
print(s4)
