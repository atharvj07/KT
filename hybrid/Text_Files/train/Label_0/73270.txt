(n, q, m) = map(int, input().split())
a = list(map(int, input().split()))
qq = list((list(map(int, input().split())) for i in range(q)))
qq = list(reversed(qq))
b = list(map(int, input().split()))
for i in range(q):
	l = qq[i][1]
	r = qq[i][2]
	if qq[i][0] == 1:
		for j in range(m):
			if b[j] < l or b[j] > r:
				continue
			if b[j] == l:
				b[j] = r
			else:
				b[j] = b[j] - 1
	else:
		for j in range(m):
			if b[j] < l or b[j] > r:
				continue
			b[j] = l + r - b[j]
for i in range(m):
	if i != m - 1:
		print(a[b[i] - 1], end=' ')
	else:
		print(a[b[i] - 1])
