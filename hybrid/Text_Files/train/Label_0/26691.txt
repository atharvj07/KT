template <typename _T>
int binarySearch(const vector<_T> &arr, int l, int r, _T x) 
{ 
    if (r >= l) { 
        int mid = l + (r - l) / 2; 
        if (arr[mid] == x) 
            return mid;  
        if (arr[mid] > x) 
            return binarySearch(arr, l, mid - 1, x); 
        return binarySearch(arr, mid + 1, r, x); 
    } 
    return -1; 
}
template <typename _T>
int binarySearch(const vector<_T> &arr, _T x) 
{ 
    return binarySearch(arr, 0, arr.size()-1, x);
}
ll mod_exp(ll base, ll exponent, ll mod)
{
    base %= mod;
    ll res = 1;
    while (exponent > 0) {
        if (exponent & 1)
            res = res * base % mod;
        base = base * base % mod;
        exponent >>= 1;
    }
    return res;
}
template <typename _I, typename _T> class DumbMap
{
    public:
        vector<_T> keys;
        vector<_T> values;
 
        DumbMap(int size)
        {
            this->keys.resize();
            this->values.resize(size);
        }
        DumbMap(vector<_T> &data, _T default_value)
        {
            this->keys = data;
            this->values.resize(data.size());
            this->recalculate(default_value);
        }
        void recalculate(_T default_value)
        {
            sort(this->keys.begin(), this->keys.end());
            for (int i = 0; i < this->keys.size(); i++)
                this->values[i] = default_value;
        }
        void compress()
        {
            vector<_T>  new_keys;
            new_keys.push_back(this->keys[0]);
            for (size_t i = 1; i < this->keys.size(); i++)
            {
                if (this->keys[i] != this->keys[i-1])
                    new_keys.push_back(this->keys[i]);
            }
            this->keys = new_keys;
            this->values.resize(this->keys.size());
            for (size_t i = 0; i < this->keys.size(); i++)
                this->values[i] = i;
        }
        const _T operator[](const _I& k) const
        {
            int i = binarySearch(this->keys, k);
            if (i < 0)
                return _T();
            return this->values[i];
        }
        void set(const _I& k, const _T& val)
        {
            int i = binarySearch(this->keys, k);
            if (i < 0)
                return ;
            this->values[i] = val;
        }
};

ll  get_score(int n, vector<ll> &num, int s, int e)
{
    ll score = 0;
    for (int i = 0; i <= e-s; i++)
        score += mod_exp(n, n-i-1, LONG_LONG_MAX) * num[e-i];
    return score;
}

ll debug_score(vector<ll> &result)
{
    ll score = 0;
    ll n = result.size();
    for (int i = 0; i < n; i++)
    {
        score += mod_exp(n, n-i-1, LONG_LONG_MAX) * result[i];
    }
    return score;
}

void    solve3(int n, vector<ll> &nums, vector<ll> &result)
{
    DumbMap<ll, ll> pos(nums, 0);
    for (int i = 0; i < n; i++)
        pos.set(nums[i], i);
    for (int i = n; i > 0; i--)
    {
        int num_pos = pos[i];
        if (num_pos == -1)
            continue;
        for (int i = num_pos; i >= 0; i--)
        {
            if (nums[i] == -1)
                break;
            result.push_back(nums[i]);
            pos.set(nums[i], -1);
            nums[i] = -1;
        }
    }
    for (int i = 0; i < n; i++)
    {
        cout << result[i] << " ";
    }
    cout << endl;
}

void    solve2(int n, vector<ll> &nums, vector<ll> &result)
{
    int start = 0;
    int current = 0;
    int biggest = 0;
    while (start < n)
    {
        biggest = 0;
        current = start;
        for (int i = start; i < n; i++)
        {
            if (nums[i] > biggest)
            {
                biggest = nums[i];
                current = i;
            }
        }
        for (int i = current; i >= start; i--)
            result.push_back(nums[i]);
        start = current+1;
    }
    for (int i = 0; i < result.size(); i++)
    {
        cout << result[i] << " ";
    }
    cout << endl;
}

void    solve(int n, vector<ll> &num, vector<ll> &result)
{
    ll best_i = 0;
    ll best_score = 0;
    ll score = 0;
    int current = 0;
    while (current < n)
    {
        best_i = current;
        best_score = 0;
        for (int i = current; i < n; i++)
        {
            score = get_score(n, num, current, i);
            if (score > best_score)
            {
                best_score = score;
                best_i = i;
            }
            cout << num[i] << "(" << score << ") ";
        }
        cout << endl;
        for (int i = best_i; i >= current; i--)
            result.push_back(num[i]);
        current = best_i+1;
    }
    for (int i = 0; i < n; i++)
    {
        cout << result[i] << " ";
    }
    cout << endl;
    cout << "SCORE : " << debug_score(result) << endl;
}
