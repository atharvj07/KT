import java.util.*;


public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        StringBuilder ans = new StringBuilder();
        int t = sc.nextInt();
        long[] pow = new long[32];
        pow[0] = 1;
        for (int i = 1; i < 32; i++) {
            pow[i] = pow[i-1]*2;
        }
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = 2;
            long tmp = 1+2;
            while (n%tmp != 0) {
                tmp += pow[k++];
            }
            long x = n/tmp;
            ans.append(x).append('\n');
        }

        System.out.print(ans);
    }
}
