s = input()
s1 = s.split()
n = int(s1[0])
k = int(s1[1])
s = input()
s1 = s.split()
arr = []
for i in range(n):
	arr.append(int(s1[i]))
sum = 0
for i in range(k):
	sum += arr[i]
sum1 = sum
for i in range(k, n):
	sum += arr[i]
	sum -= arr[i - k]
	sum1 += sum
s = str(sum1 / (n - k + 1))
s += '000000'
print(s)
