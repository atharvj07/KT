(s, x1, x2) = map(int, input().split())
(tt, t) = map(int, input().split())
(p0, d0) = map(int, input().split())
t0 = abs(x1 - x2) * t
if (x1 - p0) * d0 >= 0:
	t1 = abs(x1 - p0) * tt
	d1 = d0
else:
	if d0 > 0:
		delta = 2 * (s - p0)
	else:
		delta = 2 * p0
	t1 = (abs(x1 - p0) + delta) * tt
	d1 = -d0
p1 = x1
if (x2 - p1) * d1 >= 0:
	t2 = abs(x2 - p1) * tt
else:
	if d1 > 0:
		delta = 2 * (s - p1)
	else:
		delta = 2 * p1
	t2 = (abs(x2 - p1) + delta) * tt
print(min(t0, t1 + t2))
