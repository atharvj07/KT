t = int(input())
for cycl in range(t):
	n = int(input())
	ans = 0
	power = 0
	power_list = []
	for cave in range(n):
		cave_data = list(map(int, input().split()))
		n_m = cave_data[0]
		chan_power = 0
		for m_id in range(1, n_m + 1):
			chan_power = max(cave_data[m_id] - m_id + 2, chan_power)
		power_list.append([chan_power, n_m])
	power_list.sort()
	for l in power_list:
		if l[0] > power:
			ans += l[0] - power
			power = l[0]
		power += l[1]
	print(ans)
