from collections import defaultdict
(X, Numbers, MyDict, Now) = (list(map(int, input().split())), list(map(int, input().split())), defaultdict(int), 1)
(Temp, Answer) = (sorted(Numbers, reverse=True), [])
for i in range(X[1]):
	MyDict[Temp[i]] += 1
print(sum(Temp[:X[1]]))
i = 0
while X[1] != 1:
	if Numbers[i] in MyDict:
		MyDict[Numbers[i]] -= 1
		if MyDict[Numbers[i]] == 0:
			del MyDict[Numbers[i]]
		Answer.append(Now)
		(X[1], Now) = (X[1] - 1, 0)
	(i, Now) = (i + 1, Now + 1)
print(*Answer, X[0] - sum(Answer))
