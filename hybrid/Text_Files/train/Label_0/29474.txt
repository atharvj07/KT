#include <bits/stdc++.h>
using namespace std;
map<int, int> m;
int a[100005];
int cnt[500][100005];
int num[500];
int main() {
  int i, j, n, T, up, x, y, ret;
  scanf("%d%d", &n, &T);
  for (i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    m[a[i]]++;
  }
  map<int, int>::iterator it;
  up = 0;
  for (it = m.begin(); it != m.end(); it++) {
    if (it->second >= it->first) num[up++] = it->first;
  }
  for (i = 0; i < up; i++) {
    cnt[i][0] = 0;
    for (j = 1; j <= n; j++) {
      cnt[i][j] = cnt[i][j - 1] + (num[i] == a[j]);
    }
  }
  while (T--) {
    scanf("%d%d", &x, &y);
    ret = 0;
    for (i = 0; i < up; i++) {
      if (num[i] == cnt[i][y] - cnt[i][x - 1]) ret++;
    }
    printf("%d\n", ret);
  }
  return 0;
}
