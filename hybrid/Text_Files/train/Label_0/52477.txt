import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        int n;
        Scanner input = new Scanner(System.in);
        n=input.nextInt();
        int arr[]=new int[n];
 
        for(int i=0 ; i<n ; i++){
        int x;
        x=input.nextInt();
        arr[i]=x;
        }
       
        Arrays.sort(arr);
        int temp;
        temp=arr[n-1];
        arr[n-1]=arr[0];
        arr[0]=temp;
        
        for(int E: arr){
           System.out.print(E+" ") ;
        }
        
    }
}
