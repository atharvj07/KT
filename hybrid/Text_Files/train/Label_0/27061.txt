#include <iostream>
#include <queue>
#include <vector>
#include <algorithm>
using namespace std;

int done[100000];
vector< vector<int> > g;
vector< vector<int> > g_c;
int bfs(int x){
	queue<int> Q;
	vector<int> vec;
	Q.push(x);
	while(Q.size()){
		int q = Q.front(); Q.pop();
		if( done[q] ) continue;
		else done[q] = true;
		for(int i = 0 ; i < g[q].size() ; i++){
			Q.push(g[q][i]);
			vec.push_back(g_c[q][i]);
		}
	}
	//cout << x << " " << vec.size() << endl;
	sort(vec.begin(),vec.end());
	int answer = 0;
	for(int i = 0 ; i < vec.size() ; i++){
		if( vec[i] == vec.back() ) answer++;
	}
	return answer/2;
}


int main(){
	int n;
	while(cin >> n && n){
		for(int i = 0 ; i < 100000 ; i++)
			done[i] = 0;
		g.clear();
		g_c.clear();
		g.resize(n);
		g_c.resize(n);
		for(int i = 0 ; i < n ; i++){
			
			for(int j = 0 ; j < 2 ; j++){
				int a,b;
				cin >> a >> b;
				g[i].push_back(a);
				g_c[i].push_back(b);
			}
		}
		long long ans = 1;
		for(int i = 0 ; i < n ; i++){
			if( done[i] == 0 ){
				ans *= bfs(i);
				ans %= 10007;
			}
		}
		cout << ans << endl;
		
		
	}
}
