import java.util.*;
import java.lang.*;

public class Max {

 public static void main(String[] args)
 {
     Scanner in = new Scanner(System.in);
     int n=in.nextInt();
     int k=in.nextInt();
     int a[]=new int[n];
     for(int i=0;i<n;i++)
         a[i]=in.nextInt();
     if(k==1)
     {
         int min=a[0];
         for(int i=1;i<n;i++)
            min=Math.min(min,a[i]);
         System.out.println(min);
     }
     else if(k==2)
     {
         int suffix[]=new int[n];
         suffix[n-1]=a[n-1];
         for(int i=n-2;i>=0;i--)
             suffix[i]=Math.min(a[i],suffix[i+1]);
         int prefix=a[0];
         int ans=Math.max(prefix,suffix[1]);
         for(int i=1;i<n-1;i++)
         {
             prefix=Math.min(prefix,a[i]);
             ans=Math.max(ans,Math.max(prefix,suffix[i+1]));
         }
         System.out.println(ans);
     }
     else
     {
         int max=a[0];
         for(int i=1;i<n;i++)
             max=Math.max(a[i],max);
         System.out.println(max);
     }
 }
    
}
