import math

def print_board(n):
	if n % 2 == 0:
		print(n * n // 2)
	else:
		print(math.ceil(n * n / 2))
	for i in range(n):
		for j in range(n):
			if i % 2 == 0:
				if j % 2 == 0:
					print('C', end='')
				else:
					print('.', end='')
			elif j % 2 == 0:
				print('.', end='')
			else:
				print('C', end='')
		print()
n = int(input())
print_board(n)
