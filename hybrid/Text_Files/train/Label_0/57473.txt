import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
		String s = sc.next();
		
		int divider  = sc.nextInt();
		int len = s.length();
		int cur = 0;
		int q = 0;
		int mnoj[] = new int[10];
		mnoj[1] = 1;
		for (int i = 0; i < len; i++) {
		    q = (s.charAt(i) - '0');
		    cur = (cur * 10 + q) % divider;
		    mnoj[1] = mnoj[1] * 10 % divider;
		}
		
		for (int i = 2; i < 10; i++)
		{
			mnoj[i] = mnoj[1] * i % divider;
		}

		
		int min = cur;
		for (int i = 0; i < len; i++) {
			q = (s.charAt(i) - '0');
			if (q != 0) {
				if (cur < min) {
					min = cur;
				}
			}
			cur = (cur * 10 - mnoj[q] + q + divider) % divider;
		}
		
		System.out.println(min);
		sc.close();
	}
}
