import heapq
N = int(input())
colors = list(map(int, input().strip().split()))
if N % 2 == 0:
	colors.append(0)
penalty = 0
heapq.heapify(colors)
while len(colors) > 2:
	a = heapq.heappop(colors)
	b = heapq.heappop(colors)
	c = heapq.heappop(colors)
	heapq.heappush(colors, a + b + c)
	penalty += a + b + c
print(penalty)
