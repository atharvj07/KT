(n, k) = map(int, input().split())
s = input()
hash = [0] * 256
for i in range(n):
	hash[ord(s[i])] += 1
str = 'abcdefghijklmnopqrstuvwxyz'
i = 0
while k > 0:
	if hash[ord(str[i])] >= k:
		s = s.replace(str[i], '', k)
		k = 0
	else:
		k = k - hash[ord(str[i])]
		s = s.replace(str[i], '')
		i = i + 1
print(s)
