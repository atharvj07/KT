n = int(input())
arr = list(input())
r = arr.count('R')
d = arr.count('D')
i = j = 0
while r > 0 and d > 0:
	while arr[j] == arr[i] or arr[j] == 'O':
		j = (j + 1) % n
	if arr[j] == 'R':
		r -= 1
	else:
		d -= 1
	arr[j] = 'O'
	c = i
	i = (i + 1) % n
	while arr[i] == 'O':
		i = (i + 1) % n
	if i < c:
		j = i = 0
		n = r + d
		for x in range(n):
			while arr[i] == 'O':
				i += 1
			arr[j] = arr[i]
			j += 1
			i += 1
		i = j = 0
if r > 0:
	print('R')
else:
	print('D')
