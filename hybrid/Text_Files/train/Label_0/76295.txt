(n, m) = map(int, input().split())
ver = [int(input()) for _ in range(n)]
ver.append(10 ** 9)
hor = []
for _ in range(m):
	(x1, x2, y) = map(int, input().split())
	if x1 == 1:
		hor.append(x2)
hor.sort()
ver.sort()
j = 0
ans = 10 ** 18
for i in range(n + 1):
	while j < len(hor) and ver[i] > hor[j]:
		j += 1
	ans = min(ans, i + len(hor) - j)
	if j == len(hor):
		break
print(ans)
