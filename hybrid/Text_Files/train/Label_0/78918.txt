test = int(input())
while test != 0:
	n = int(input())
	arr = list(map(int, input().split()))
	summ = 0
	p = 1
	for i in range(n):
		summ += arr[i]
		p *= arr[i]
	if summ != 0 and p != 0:
		print(0)
	else:
		steps = 0
		summ = 0
		for i in range(n):
			if arr[i] == 0:
				arr[i] += 1
				steps += 1
		for i in range(n):
			summ += arr[i]
		if summ == 0:
			steps += 1
		print(steps)
	test -= 1
