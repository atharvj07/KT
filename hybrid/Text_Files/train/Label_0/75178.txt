l1 = [c for c in input()]
l2 = [c for c in input()]
l = len(l1)
total = 0
for x in range(l - 1):
	if l1[x] == '0' and l1[x + 1] == '0':
		if l2[x] == '0':
			l2[x] = 'X'
			l1[x] = 'X'
			l1[x + 1] = 'X'
			total += 1
		elif l2[x + 1] == '0':
			l2[x + 1] = 'X'
			l1[x] = 'X'
			l1[x + 1] = 'X'
			total += 1
	elif l2[x] == '0' and l2[x + 1] == '0':
		if l1[x] == '0':
			l1[x] = 'X'
			l2[x] = 'X'
			l2[x + 1] = 'X'
			total += 1
		elif l1[x + 1] == '0':
			l1[x + 1] = 'X'
			l2[x] = 'X'
			l2[x + 1] = 'X'
			total += 1
print(total)
