#include <bits/stdc++.h>
using namespace std;
const int MAXNUM = 1001;
const int MAXN = 21;
int d[MAXN];
bool bs[MAXNUM];
int main() {
  int n;
  scanf("%d", &n);
  d[0] = 0, d[1] = 1;
  d[2] = 2;
  bs[0] = true;
  bs[1] = true;
  bs[2] = true;
  bs[3] = true;
  for (int i = 3; i < n; i++) {
    for (int j = 0; j < MAXNUM; j++)
      if (!bs[j]) {
        d[i] = j;
        break;
      }
    for (int j = 0; j < i; j++)
      for (int k = 0; k <= i; k++) bs[max(d[j] + d[i] - d[k], 0)] = true;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) printf("%d ", i == j ? 0 : d[i] + d[j]);
    printf("\n");
  }
  return 0;
}
