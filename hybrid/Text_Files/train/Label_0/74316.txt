def solve():
	n: int = int(input())
	arr = [int(num) for num in input().split(' ')]
	(count, ans) = ({}, 0)
	for i in range(len(arr)):
		ans += (n - i) * count.get(arr[i], 0)
		count[arr[i]] = count.get(arr[i], 0) + i + 1
	return ans

def main_fun():
	t = int(input())
	while t > 0:
		print(solve())
		t -= 1
main_fun()
