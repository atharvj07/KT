#include <bits/stdc++.h>
using namespace std;
int n, a[100005];
int main() {
  ios_base::sync_with_stdio(0);
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  sort(a + 1, a + n + 1);
  int res = 0;
  for (int i = 1; i <= n; ++i)
    res = (res + ((a[i] % 10007) * (a[n - i + 1] % 10007)) % 10007) % 10007;
  cout << res;
}
