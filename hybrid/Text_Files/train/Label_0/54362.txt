import java.io.*;
import java.util.*;


public class Main2 {


	public static void main (String[] args) throws Exception
	{		
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		String[] spl = in.readLine().split(" ");
		int n = Integer.parseInt(spl[0]);
		int k = Integer.parseInt(spl[1]);
		int p = Integer.parseInt(spl[2]);
		ArrayList<Integer> even = new ArrayList<Integer>();
		ArrayList<Integer> odd = new ArrayList<Integer>();
		spl = in.readLine().split(" ");
		for (int i = 0; i < n; i++) {
			int act = Integer.parseInt(spl[i]);
			if(act%2==0)
				even.add(act);
			else
				odd.add(act);
		}
		if(odd.size()<k-p || (odd.size()-(k-p))%2==1 || (p - even.size() - (odd.size()-(k-p))/2)>0)
		{
			System.out.println("NO");
		}
		else
		{

			System.out.println("YES");
			int ide = 0, ido=0;
			if(p>0)
			{
				for (int i = 0; i < k-p; i++) {
					System.out.println("1 "+odd.get(ido++));
				}
				for (int i = 0; i < p-1; i++) {
					if(ide<even.size())
						System.out.println("1 "+even.get(ide++));
					else
					{
						System.out.println("2 "+odd.get(ido++)+" "+odd.get(ido++));
					}
				}
			}
			else
			{
				for (int i = 0; i < k-p-1; i++) {
					System.out.println("1 "+odd.get(ido++));
				}
				for (int i = 0; i < p; i++) {
					if(ide<even.size())
						System.out.println("1 "+even.get(ide++));
					else
					{
						System.out.println("2 "+odd.get(ido++)+" "+odd.get(ido++));
					}
				}
			}
			int cnt = 0;
			StringBuffer last = new StringBuffer();
			while(ido < odd.size())
			{
				last.append(odd.get(ido++)+" ");
				cnt++;
			}
			while(ide < even.size())
			{
				last.append(even.get(ide++)+" ");
				cnt++;
			}
			System.out.println(cnt+" "+last.toString().trim());
		}

	}
}
