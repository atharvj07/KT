k = int(input())
d = dict()
for _ in range(k):
	(name, *nums) = input().split()
	nums = set(nums[1:])
	if name in d:
		d[name] = d[name].union(nums)
	else:
		d[name] = nums
print(len(d))
for name in d:
	arr = sorted(list(map(lambda x: x[::-1], d[name])))
	i = 0
	while i < len(arr) - 1:
		if arr[i + 1].startswith(arr[i]):
			arr.pop(i)
			i -= 1
		i += 1
	print(name, len(list(arr)), ' '.join([k[::-1] for k in arr]))
