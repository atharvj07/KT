prime = [-1 for i in range(100005)]
p = 2
avail = 1
while p * p <= 100005:
	if prime[p] == -1:
		for i in range(p, 100005, p):
			if prime[i] == -1:
				prime[i] = avail
		avail += 1
	p += 1
for i in range(2, 100005):
	if prime[i] == -1:
		prime[i] = avail
		avail += 1
n = int(input())
print(*prime[2:n + 1])
