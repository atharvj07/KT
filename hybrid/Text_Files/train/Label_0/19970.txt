#include <bits/stdc++.h>
using namespace std;
int main() {
  int count = 0;
  int h1, a1, c1, h2, a2;
  int h11, a11, c11, h21, a21;
  cin >> h1 >> a1 >> c1;
  cin >> h2 >> a2;
  h11 = h1;
  a11 = a1;
  c11 = c1;
  h21 = h2;
  a21 = a2;
  do {
    if (h1 <= a2) {
      if (a1 >= h2) {
        count++;
        break;
      } else {
        count++;
        h1 -= a2;
        h1 += c1;
      }
    } else {
      count++;
      h2 -= a1;
      h1 -= a2;
    }
  } while (h2 > 0);
  cout << count << endl;
  do {
    if (h11 <= a21) {
      if (a11 >= h21) {
        cout << "STRIKE\n";
        break;
      } else {
        cout << "HEAL\n";
        h11 -= a21;
        h11 += c11;
      }
    } else {
      cout << "STRIKE\n";
      h21 -= a11;
      h11 -= a21;
    }
  } while (h21 > 0);
  return 0;
}
