import java.util.*;
import java.io.*;
import java.math.*;
public class E {
	static long maxn = 1000000000000000000L;
	public static void main(String[] args) {
		fac = new BigInteger[51];
		fac[0] = BigInteger.ONE;
		for(int i = 1; i <= 50; i++) {
			fac[i] = fac[i-1].multiply(BigInteger.valueOf(i));
		}
//		for(int i = 0; i < 6; i++) {
//			System.out.println(Arrays.toString(cycleperm(5, BigInteger.valueOf(i))));
//		}
		dp = new BigInteger[51][51];
		for(int i = 0; i <= 50; i++) {
			for(int j = 0; j <= 50; j++) {
				dp[i][j] = BigInteger.ZERO;
			}
		}
		dp[1][1] = BigInteger.ONE;
		for(int i = 1; i <= 50; i++) {
			for(int j = 1; j < i; j++) {
				dp[i][1] = dp[i][1].add(dp[i-1][j]);
			}
			for(int j = 2; j <= i; j++) {
				dp[i][j] = BigInteger.valueOf(Long.max(1, j-2)).multiply(dp[i-1][j-1]);
			}
		}
		FastScanner sc = new FastScanner();
		int T = sc.nextInt();
		PrintWriter pw = new PrintWriter(System.out);
		for(int t = 0; t < T; t++) {
			int n = sc.nextInt();
			BigInteger k = BigInteger.valueOf(sc.nextLong()-1);
			int m = n;
			ArrayList<Integer> res = new ArrayList<>();
			while(m > 0) {
				BigInteger tot = BigInteger.ZERO;
				BigInteger ptot = BigInteger.ZERO;
				int i = 1;
				for(; i <= n; i++) {
					tot = tot.add(dp[m][i]);
					if(tot.compareTo(k) > 0) break;
					ptot = BigInteger.ZERO.add(tot);
				}
				if(i == n+1) break;
				k = k.subtract(ptot);
				BigInteger a = k.divide(dp[m-i+1][1]);
				int[] perm = cycleperm(i, a);
				for(int p: perm) res.add(p+(n-m)+1);
				k = k.mod(dp[m-i+1][1]);
				m -= i;
			}
			if(res.isEmpty()) {
				pw.println("-1");
			}
			else {
				StringBuilder sb = new StringBuilder();
				for(int r: res) sb.append(r+" ");
				pw.println(sb.toString().trim());
			}
		}
		pw.flush();
	}
	static BigInteger fac[];
	static BigInteger[][] dp;
	static int[] cycleperm(int n, BigInteger k){
		if(n == 1) {
			int[] res = {0};
			return res;
		}
		if(n == 2) {
			int[] res = {1,0};
			return res;
		}
		if(n == 3) {
			int[] res = {2,0,1};
			return res;
		}
		int[] res = new int[n];
		Arrays.fill(res, -1);
		res[0] = n-1;
		BigInteger d = fac[n-3];
		BigInteger aux = k.add(BigInteger.ZERO);
		for(int i = 0; i < n-3; i++) {
			int a = aux.divide(d).intValue();
			int b = convert(a, i+1, res);
			res[i+1] = b;
			aux = aux.mod(d);
			d = d.divide(BigInteger.valueOf(n-i-3));
		}
		res[n-2] = convert(0, n-2, res);
		res[n-1] = convert(0, n-1, res);
		return res;
	}
	static int convert(int k, int pos, int[] arr) {
		int n = arr.length;
		for(int i = 0; i < n-1; i++) {
			arr[pos] = i;
			if(valid(arr)) k--;
			arr[pos] = -1;
			if(k < 0) return i;
		}
		return -1;
	}
	static boolean valid(int[] arr) {
		int n = arr.length;
		int[] q = new int[n];
		for(int i = 0; i < n; i++) {
			if(arr[i] >= 0) {
				q[arr[i]]++;
				if(q[arr[i]] > 1) return false;
			}
		}
		for(int i = 0; i < n; i++) {
			int c = arr[i];
			while(c > 0) {
				if(c == i) return false;
				c = arr[c];
			}
		}
		return true;
	}
	static class FastScanner {
		public BufferedReader reader;
		public StringTokenizer tokenizer;
		public FastScanner() {
			reader = new BufferedReader(new InputStreamReader(System.in), 32768);
			tokenizer = null;
		}
		public String next() {
			while (tokenizer == null || !tokenizer.hasMoreTokens()) {
				try {
					tokenizer = new StringTokenizer(reader.readLine());
				} catch (IOException e) {
					throw new RuntimeException(e);
				}
			}
			return tokenizer.nextToken();
		}
		public int nextInt() {
			return Integer.parseInt(next());
		}
		public long nextLong() {
			return Long.parseLong(next());
		}
		public double nextDouble() {
			return Double.parseDouble(next());
		}
		public String nextLine() {
			try {
				return reader.readLine();
			} catch(IOException e) {
				throw new RuntimeException(e);
			}
		}
	}

}
