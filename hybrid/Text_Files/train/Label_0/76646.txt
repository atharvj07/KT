def main():
	t = int(input())
	for _ in range(t):
		n = int(input())
		a = list(map(int, input().split()))
		capital = a[0]
		non_capital_idx = -1
		for i in range(1, n):
			if a[i] != capital:
				non_capital_idx = i
				break
		if non_capital_idx == -1:
			print('NO')
			continue
		print('YES')
		for j in range(1, n):
			if j == non_capital_idx:
				print(1, non_capital_idx + 1)
			elif a[j] == capital:
				print(non_capital_idx + 1, j + 1)
			else:
				print(1, j + 1)
main()
