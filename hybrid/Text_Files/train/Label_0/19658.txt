#include <bits/stdc++.h>
using namespace std;

int N, M;
int x[110], a[15], b[15], c[15];
int dp[15][110][110];

int solve(int idx, int l, int r) {
  assert(r - l > 0);

  int val = dp[idx][l][r];
  if(val >= 0) return val;

  val = 0;
  if(r - l == 1) {
    val = (x[l] == a[idx]);
    return dp[idx][l][r] = val;
  }
  else {
    // split
    for(int k=l+1; k<r; k++) {
      int vl = solve(b[idx], l, k);
      int vr = solve(c[idx], k, r);
      val |= (vl && vr);
    }
  }
  return dp[idx][l][r] = val;
}

int main() {
  cin >> N;
  for(int i=0; i<N; i++) {
    cin >> x[i];
  }

  cin >> M;
  for(int i=0; i<M; i++) {
    cin >> a[i] >> b[i] >> c[i];
    b[i]--; c[i]--;
  }
  
  memset(dp, -1, sizeof(dp));
  cout << (solve(0, 0, N) ? "Yes" : "No") << endl;
  return 0;
}
