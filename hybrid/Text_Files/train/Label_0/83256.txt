import sys, heapq
input = lambda : sys.stdin.readline().strip('\r\n')
(n, k) = map(int, input().split())
x = list(map(int, input().split()))
a = int(input())
c = list(map(int, input().split()))
heap = []
ans = 0
for i in range(n):
	heapq.heappush(heap, c[i])
	while x[i] > k and len(heap) > 0:
		k += a
		ans += heapq.heappop(heap)
	if k < x[i]:
		print(-1)
		exit()
print(ans)
