class MyApp:

	def __init__(self, s):
		self.s = s

	def build(self):
		a = b = 0
		value = 1
		(left, right) = self.s.split('^')
		for i in left[::-1]:
			if i != '=':
				a += int(i) * value
			value += 1
		value = 1
		for i in right:
			if i != '=':
				b += int(i) * value
			value += 1
		print('balance' if a == b else 'right' if a < b else 'left')
root = MyApp(input())
root.build()
