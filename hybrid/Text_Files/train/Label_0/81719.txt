n = int(input())
ar = []
for i in range(n):
	ar.append(list(map(int, input().split())))
for i in range(n):
	k = ar[i][2]
	if max(ar[i]) > ar[i][2]:
		print(-1)
		continue
	elif max(ar[i]) <= ar[i][2]:
		if ar[i][0] % 2 == 0 and ar[i][1] % 2 != 0 or (ar[i][0] % 2 != 0 and ar[i][1] % 2 == 0):
			print(k - 1)
			continue
		elif ar[i][0] % 2 == 0 and ar[i][1] % 2 == 0:
			if k % 2 == 0:
				print(k)
			else:
				print(k - 2)
			continue
		elif ar[i][0] % 2 != 0 and ar[i][1] % 2 != 0:
			if k % 2 == 0:
				print(k - 2)
			else:
				print(k)
			continue
