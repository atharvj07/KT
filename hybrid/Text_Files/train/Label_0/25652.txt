ll n, m, c;
ll a[ten(5) + 810], rev_a[ten(5) + 810];
vector<ll> max_smell(const ll* a, ll r)
{
    vector<ll> kusasou(n);
    deque<pair<ll, ll>> q;
    rep(i, n)
    {
        while (!q.empty() && q.front().second < i - r)
            q.pop_front();
        while (!q.empty() && q.back().first - c * (i - q.back().second) <= a[i])
            q.pop_back();
        q.push_back(make_pair(a[i], i));

        kusasou[i] = q.front().first - c * (i - q.front().second);
    }
    return kusasou;
}
bool check(ll r)
{
    auto kusasou = max_smell(a, r);
    auto www = max_smell(rev_a, r);
    reverse(all(www));

    ll sum_w = 0;
    rep(i, n)
        sum_w += max(kusasou[i], www[i]);
    return sum_w >= m;
}
int main()
{
    fast_io();

    cin >> n >> m >> c;
    input(a, n);
    rep(i, n)
        rev_a[i] = a[i];
    reverse(rev_a, rev_a + n);

    ll low = -1, high = n;
    while (high - low > 1)
    {
        ll mid = (low + high) / 2;
        (check(mid) ? high : low) = mid;
    }
    cout << (high == n ? -1 : high) << endl;
}
