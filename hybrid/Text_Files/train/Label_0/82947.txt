t = int(input())
nothing = ''
for _ in range(0, t):
	nothing = input()
	xa = xb = xf = ya = yb = yf = 0
	(xa, ya) = input().split()
	(xb, yb) = input().split()
	(xf, yf) = input().split()
	(xa, xb, xf, ya, yb, yf) = (int(xa), int(xb), int(xf), int(ya), int(yb), int(yf))
	diffx = xa - xb
	diffy = ya - yb
	if diffx < 0:
		diffx = -diffx
	if diffy < 0:
		diffy = -diffy
	result = diffx + diffy
	if xa == xb == xf:
		if ya <= yf <= yb or ya >= yf >= yb:
			result += 2
	elif ya == yb == yf:
		if xa <= xf <= xb or xa >= xf >= xb:
			result += 2
	print(result)
