import java.util.Scanner;
public class D2 {
	static long l;
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		long N = s.nextLong();
		l = N*2-1;
		for(int q=s.nextInt();q>0;q--)System.out.println(g(s.nextLong()-1));
	}
	static long g(long i) {
		if(i%2==0)return(i/2)+1;
		return g(i+(l-i)/2);
	}
}
