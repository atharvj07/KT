#include <bits/stdc++.h>
using namespace std;

int main(){
	int h1, h2, k1, k2, a, b, c, d;
	cin >> h1 >> h2 >> k1 >> k2 >> a >> b >> c >> d;
	
	if(h1*a+h2*b+(h1/10*c)+(h2/20*d) < k1*a+k2*b+(k1/10*c)+(k2/20*d)) cout << "kenjiro" << endl;
	else if(h1*a+h2*b+(h1/10*c)+(h2/20*d) == k1*a+k2*b+(k1/10*c)+(k2/20*d)) cout << "even" << endl;
	else cout << "hiroshi" << endl;
	
	return 0;
}
