namespace io {
const int SIZE = (1 << 20) + 1;
char ibuf[SIZE], *iS, *iT, obuf[SIZE], *oS = obuf, *oT = oS + SIZE - 1, c,
                                       qu[55];
int f, qr;
inline void flush() {
  fwrite(obuf, 1, oS - obuf, stdout);
  oS = obuf;
}
inline void putc(char x) {
  *oS++ = x;
  if (oS == oT) flush();
}
template <typename A>
inline bool read(A &x) {
  for (f = 1, c = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                              (iS == iT ? EOF : *iS++))
                            : *iS++);
       c<'0' | c> '9';
       c = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                       (iS == iT ? EOF : *iS++))
                     : *iS++))
    if (c == '-')
      f = -1;
    else if (c == EOF)
      return 0;
  for (x = 0; c <= '9' && c >= '0';
       c = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                       (iS == iT ? EOF : *iS++))
                     : *iS++))
    x = x * 10 + (c & 15);
  x *= f;
  return 1;
}
inline bool read(char &x) {
  while ((x = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                           (iS == iT ? EOF : *iS++))
                        : *iS++)) == ' ' ||
         x == '\n' || x == '\r')
    ;
  return x != EOF;
}
inline bool read(string &x) {
  while ((c = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                           (iS == iT ? EOF : *iS++))
                        : *iS++)) == '\n' ||
         c == ' ' || c == '\r')
    ;
  if (c == EOF) return 0;
  x.clear();
  while (!(c == '\n' || c == ' ' || c == '\r' || c == EOF))
    x.push_back(c),
        c = (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, SIZE, stdin),
                         (iS == iT ? EOF : *iS++))
                      : *iS++);
  return 1;
}
template <typename A, typename... B>
inline int read(A &x, B &...y) {
  return read(x) + read(y...);
}
template <typename A>
inline bool write(A x) {
  if (!x) putc('0');
  if (x < 0) putc('-'), x = -x;
  while (x) qu[++qr] = x % 10 + '0', x /= 10;
  while (qr) putc(qu[qr--]);
  return 0;
}
inline bool write(char x) {
  putc(x);
  return 0;
}
inline bool write(const char *x) {
  while (*x) {
    putc(*x);
    ++x;
  }
  return 0;
}
inline bool write(char *x) {
  while (*x) {
    putc(*x);
    ++x;
  }
  return 0;
}
inline bool write(string s) { return write(s.c_str()); }
template <typename A, typename... B>
inline bool write(A x, B... y) {
  return write(x) || write(y...);
}

const int MOD = 1e9 + 7, MOD2 = 1e9 + 9;
int dx[] = {-2, -2, -1, -1, 1, 1, 2, 2};
int dy[] = {1, -1, 2, -2, 2, -2, 1, -1};
int n, m;
bool check(int i, int j) { return 1 <= i && i <= n && 1 <= j && j <= m; }
int d[2][60][60] = {};
std::pair<int, int> s[2];
std::pair<int, int> par[2][60][60] = {};
bool vis[2][60][60] = {};
void bfs(int id, int si, int sj) {
  queue<std::pair<int, int> > q;
  q.push({si, sj});
  vis[id][si][sj] = 1;
  while (!q.empty()) {
    auto [ui, uj] = q.front();
    q.pop();
    for (int k = (0); k < (8); k += (1)) {
      int vi = ui + dx[k], vj = uj + dy[k];
      if (!check(vi, vj) || vis[id][vi][vj]) continue;
      d[id][vi][vj] = d[id][ui][uj] + 1;
      par[id][vi][vj] = make_pair(ui, uj);
      vis[id][vi][vj] = 1;
      q.push(make_pair(vi, vj));
    }
  }
}
