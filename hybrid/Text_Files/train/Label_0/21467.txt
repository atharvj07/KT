#include <bits/stdc++.h>
using namespace std;
vector<long long> padre;
long long buscar(long long x) {
  if (padre[x] == x) {
    return x;
  } else {
    return buscar(padre[x]);
  }
}
long long uni(long long x, long long y) {
  long long xraiz = buscar(x);
  long long yraiz = buscar(y);
  padre[xraiz] = yraiz;
}
int main() {
  long long a;
  cin >> a;
  vector<long long> vec;
  for (long long i = 0; i < a; i++) {
    padre.push_back(i);
    long long g;
    cin >> g;
    g--;
    vec.push_back(g);
  }
  vector<vector<char> > matriz(a);
  for (long long i = 0; i < a; i++) {
    for (long long j = 0; j < a; j++) {
      char g;
      cin >> g;
      matriz[i].push_back(g);
    }
  }
  for (long long i = 0; i < a; i++) {
    for (long long j = 0; j < a; j++) {
      if (matriz[i][j] == '1') {
        uni(i, j);
      }
    }
  }
  for (long long i = 0; i < vec.size(); i++) {
    long long k = 1000000;
    long long index;
    for (long long j = i; j < vec.size(); j++) {
      if (buscar(i) == buscar(j) and vec[j] < k) {
        k = vec[j];
        index = j;
      }
    }
    long long tmp = vec[i];
    vec[i] = vec[index];
    vec[index] = tmp;
  }
  for (long long i = 0; i < vec.size(); i++) {
    cout << vec[i] + 1;
    if (i != vec.size() - 1) {
      cout << " ";
    } else {
      cout << endl;
    }
  }
}
