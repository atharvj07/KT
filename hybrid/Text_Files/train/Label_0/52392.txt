import java.util.Scanner;

public class OlesyaRodion {
    
    public static void main(String[] args) {
        
        int n, t;
        String number = "";
        Scanner input = new Scanner(System.in);
        
        n = input.nextInt();
        t = input.nextInt();
        
        if((n == 1 && t == 10)) System.out.println(-1);
        
        else if(n > 1 && t == 10){
            for(int i = 0; i < n-1; i++){
                number += '1';
            }
                number += '0';
                
                System.out.println(number);
        }
        
        else{
            String temp = Integer.toString(t);
            for(int i = 0; i < n; i++){
                number += temp;
            }
            System.out.println(number);
        }
          
        
    }
    
}
