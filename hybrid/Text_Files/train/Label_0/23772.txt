#include <bits/stdc++.h>
using namespace std;
long long ar[100005];
long long n, k, mx, res;
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> ar[i];
    mx = max(mx, ar[i]);
  }
  long long l = 0, r = mx;
  while (l <= r) {
    long long m = (l + r) / 2;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
      sum += min(m, ar[i]);
    }
    if (sum <= k) {
      l = m + 1;
      res = m;
    } else {
      r = m - 1;
    }
  }
  for (int i = 0; i < n; i++) {
    k -= min(ar[i], res);
    ar[i] -= min(ar[i], res);
  }
  res = 0;
  for (int i = 0; i < n; i++) {
    if (!ar[i]) continue;
    if (!k) {
      res = i;
      break;
    } else {
      ar[i]--;
      k--;
    }
  }
  if (k > 0) {
    cout << -1 << '\n';
    return 0;
  }
  for (int i = res; i < res + n; i++) {
    if (ar[i % n]) cout << (i % n) + 1 << " ";
  }
  cout << '\n';
  return 0;
}
