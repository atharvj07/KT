(a, b, c, d) = [int(x) for x in input().split()]
dn = {0: a, 1: b, 2: c, 3: d}
total = a + b + c + d
flag = 1
for i in range(4):
	if dn[i] == 0:
		continue
	soln = []
	soln.append(i)
	td = dn.copy()
	td[i] -= 1
	last = i
	for j in range(total - 1):
		if td.get(last - 1, 0):
			soln.append(last - 1)
			td[last - 1] -= 1
			last = last - 1
		elif td.get(last + 1, 0):
			soln.append(last + 1)
			td[last + 1] -= 1
			last = last + 1
		else:
			break
	if len(soln) == total:
		print('YES')
		for i in soln:
			print(i, end=' ')
		print()
		flag = 0
		break
if flag:
	print('NO')
