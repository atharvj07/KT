#include <bits/stdc++.h>
using namespace std;
struct node {
  int id;
  int x, y;
  node() {}
  node(int x, int y) : x(x), y(y) {}
  void input(int idx) {
    id = idx;
    scanf("%d%d", &x, &y);
  }
  bool operator<(const node &d) const {
    if (d.x == x) return y < d.y;
    return x < d.x;
  }
} a[100005];
bool f(node a, node b, node c) {
  long long x1, x2, x3, y1, y2, y3;
  x1 = (long long)a.x;
  x2 = (long long)b.x;
  x3 = (long long)c.x;
  y1 = (long long)a.y;
  y2 = (long long)b.y;
  y3 = (long long)c.y;
  return (x2 - x3) * (y1 - y2) == (y2 - y3) * (x1 - x2);
}
int n;
int main() {
  scanf("%d", &n);
  for (int i = (0); i < (n); ++i) a[i].input(i + 1);
  sort(a, a + n);
  int ans;
  for (int i = (2); i < (n); ++i) {
    if (!f(a[0], a[1], a[i])) {
      ans = a[i].id;
      break;
    }
  }
  printf("%d %d %d\n", a[0].id, a[1].id, ans);
  return 0;
}
