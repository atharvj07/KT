#include <bits/stdc++.h>
using namespace std;
bool prime[100005];
void sieve() {
  memset(prime, true, sizeof(prime));
  for (long long p = 2; p * p < 100005; p++) {
    if (prime[p] == true) {
      for (long long i = p * p; i < 100005; i += p) prime[i] = false;
    }
  }
}
long long powmod(long long x, long long y, long long mod) {
  long long res = 1;
  x = x % mod;
  while (y > 0) {
    if (y & 1) {
      res = (res * x) % mod;
    }
    y = y >> 1;
    x = (x * x) % mod;
  }
  return res;
}
vector<vector<long long> > v(100005);
bool vis[100005];
long long ans = 0;
void dfs(long long src) {
  vis[src] = 1;
  ans += 1;
  for (auto i : v[src]) {
    if (!vis[i]) {
      dfs(i);
    }
  }
}
signed main() {
  long long a, b;
  cin >> a >> b;
  long long x, y;
  cin >> x >> y;
  long long n;
  cin >> n;
  string s;
  cin >> s;
  long long xx[n + 1], yy[n + 1];
  xx[0] = 0;
  yy[0] = 0;
  for (long long i = 1; i < n + 1; i++) {
    if (s[i - 1] == 'U') {
      yy[i] = yy[i - 1] + 1;
      xx[i] = xx[i - 1];
    }
    if (s[i - 1] == 'D') {
      yy[i] = yy[i - 1] - 1;
      xx[i] = xx[i - 1];
    }
    if (s[i - 1] == 'R') {
      yy[i] = yy[i - 1];
      xx[i] = xx[i - 1] + 1;
    }
    if (s[i - 1] == 'L') {
      yy[i] = yy[i - 1];
      xx[i] = xx[i - 1] - 1;
    }
  }
  long long lo = 0, hi = 1e16;
  long long mid;
  long long ans = 1e18;
  while (lo <= hi) {
    mid = (lo + hi) / 2;
    long long u = (mid) / (n);
    long long new_x = a + (u * xx[n]);
    new_x += (xx[mid % n]);
    long long new_y = b + (u * yy[n]);
    new_y += (yy[mid % n]);
    if ((abs(new_x - x) + abs(new_y - y)) <= mid) {
      ans = min(ans, mid);
      hi = mid - 1;
    } else {
      lo = mid + 1;
    }
  }
  if (ans == 1e18) {
    cout << -1 << endl;
    return 0;
  }
  cout << ans << endl;
  return 0;
}
