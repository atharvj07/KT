t = int(input())
for p in range(t):
	(n, k, d) = [int(x) for x in input().split()]
	a = [int(x) for x in input().split()]
	arr = 10 ** 100
	test = {}
	counter = 0
	for i in range(n):
		if a[i] not in test:
			test[a[i]] = 0
		test[a[i]] += 1
		if test[a[i]] == 1:
			counter += 1
		if i == d - 1:
			arr = counter
		if i >= d:
			test[a[i - d]] -= 1
			if test[a[i - d]] == 0:
				counter -= 1
			arr = min(counter, arr)
	print(arr)
