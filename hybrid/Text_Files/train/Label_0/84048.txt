def fake_np():
	numbers = input().split()
	numbers = [int(i) for i in numbers]
	dic = {}
	if numbers[1] == numbers[0]:
		print(numbers[1])
		return
	print(2)
	return
	checking = numbers[1] - numbers[0]
	if checking > 1000:
		print(2)
		return
	maximum = 0
	output = 1
	for i in range(numbers[0], numbers[1]):
		temp = 2
		while True:
			if i % temp == 0:
				break
			else:
				temp += 1
		if temp not in dic:
			dic[temp] = 1
			if dic[temp] > maximum:
				maximum = dic[temp]
				output = temp
		else:
			dic[temp] += 1
			if dic[temp] > maximum:
				maximum = dic[temp]
				output = temp
	print(output)
fake_np()
