#include <bits/stdc++.h>
using namespace std;
const long long N = 1e5 + 1;
set<long long> adj[N];
long long subtreeSize[N], nodes;
char ans[N];
void calculateSubtreeSize(long long node, long long par) {
  nodes++;
  subtreeSize[node] = 1;
  for (long long child : adj[node]) {
    if (child == par) continue;
    calculateSubtreeSize(child, node);
    subtreeSize[node] += subtreeSize[child];
  }
}
long long getCentroid(long long node, long long par) {
  for (long long child : adj[node]) {
    if (child == par) continue;
    if (subtreeSize[child] > (nodes / 2)) return getCentroid(child, node);
  }
  return node;
}
void decompose(long long node, long long par, char c) {
  nodes = 0;
  calculateSubtreeSize(node, par);
  long long centroid = getCentroid(node, node);
  ans[centroid] = c;
  for (long long child : adj[centroid]) {
    adj[child].erase(centroid);
    decompose(child, centroid, c + 1);
  }
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long n;
  cin >> n;
  for (long long i = 1; i <= n - 1; i++) {
    long long a, b;
    cin >> a >> b;
    adj[a].insert(b);
    adj[b].insert(a);
  }
  decompose(1, 0, 'A');
  for (long long i = 1; i <= n; i++) cout << ans[i] << " ";
}
