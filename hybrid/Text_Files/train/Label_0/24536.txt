#include <bits/stdc++.h>
using namespace std;
using ull = unsigned long long;
using ll = long long;
const int N = 1100005, MOD = 1e9 + 7;
int k, a[N], ptr;
bool chk(int x) {
  int curr = 0, ost = 0;
  for (int i = 0; i < ptr; ++i) {
    if (ost + a[i] + (i != ptr - 1) <= x) {
      ost += a[i] + (i != ptr - 1);
    } else {
      if (ost == 0) return 0;
      ++curr, ost = a[i] + (i != ptr - 1);
      if (ost > x) return 0;
    }
  }
  return (curr + (ost != 0)) <= k;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  cin >> k;
  string s;
  while (cin >> s) {
    int sz = s.length();
    for (int j = 0; j < sz; ++j) {
      int k = j;
      while (k + 1 < sz && s[k + 1] != '-') {
        ++k;
      }
      a[ptr++] = k - j + 1;
      j = k + 1;
    }
  }
  int l = 0, r = 1e9, m;
  while (r - l > 1) {
    m = (l + r) >> 1;
    if (chk(m)) {
      r = m;
    } else {
      l = m;
    }
  }
  cout << r << '\n';
  return 0;
}
