#include<bits/stdc++.h>

using namespace std;
typedef long long ll;

int main() {
    int A, B, Q;
    cin >> A >> B >> Q;

    vector<ll> s(A + 2), t(B + 2);
    s[0] = -1e18;
    s[A + 1] = 1e18;
    t[0] = s[0];
    t[B + 1] = s[A + 1];
    for (int i = 1; i < A + 1; i++) cin >> s[i];
    for (int i = 1; i < B + 1; i++) cin >> t[i];
    sort(s.begin(), s.end());
    sort(t.begin(), t.end());

    while (Q--) {
        ll x;
        cin >> x;
        auto sR = lower_bound(s.begin(), s.end(), x) - s.begin();
        auto tR = lower_bound(t.begin(), t.end(), x) - t.begin();

        ll ans = 1e18;
        for (int j = -1; j < 1; j++) {
            for (int k = -1; k < 1; k++) {
                auto xs = s[sR + j], xt = t[tR + k];
                ans = min(ans, min(abs(x - xs), abs(x - xt)) + abs(xs - xt));
            }
        }
        cout << ans << endl;
    }
    return 0;
}
