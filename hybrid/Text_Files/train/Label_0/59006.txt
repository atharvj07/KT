import java.util.*;
public class B43 {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        char[] arr1=sc.nextLine().replaceAll("\\s+","").toCharArray();
        char[] arr2=sc.nextLine().replaceAll("\\s+","").toCharArray();
        if(arr2.length>arr1.length){
            System.out.println("NO");
            return;
        }
        Arrays.sort(arr1);
        Arrays.sort(arr2);
        int c=0;
        for(int i=0;i<arr2.length;i++){
            for(int n=0;n<arr1.length;n++){
                if(arr2[i]==arr1[n]){
                    c++;
                    arr2[i]='1';
                    arr1[n]='2';
                    break;
                }
            }
        }
        if(c==arr2.length){
            System.out.println("YES");
        }else{
            System.out.println("NO");
        }
    }
}
