import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main1 {
	
	static long[] fac;
	static long mod = 998244353;
	
	static long nCk(int n,int k){return modMult(fac[n], inv(modMult(fac[n-k], fac[k])));}
	static long modMult(long a,long b){return ((a%mod)*(b%mod))%mod;}	
	static long inv(long n){return modPow(n, mod-2);}
	
	static long modPow(long a,long e)
	{
		a %= mod;
		long res = 1;
		while(e > 0)
		{
			if((e & 1) == 1)
				res = (res * a) % mod;
			a = (a * a) % mod;
			e >>= 1;
		}
		return res;
	}
	
	static long solve(int a,int b)
	{
		long ans = 0 , min = Math.min(a, b);
		for(int i = 0;i<=min;i++)
			ans += modMult(nCk(a, i),modMult(nCk(b, i), fac[i]));
		return ans%mod;
	}
	
	
	public static void main(String[] args) throws Throwable {
		Scanner sc = new Scanner(System.in);
		int a = sc.nextInt() , b = sc.nextInt(), c = sc.nextInt();
		fac = new long[(int)1e6];
		fac[0] = fac[1] = 1;
		for(int i = 2; i < fac.length;i++)fac[i] = modMult(fac[i-1], i);
		long ans = modMult(solve(a, b), modMult(solve(b, c), solve(a, c)));
		System.out.println(ans%mod);
	}
	
	
	static class Scanner 
	{
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s){	br = new BufferedReader(new InputStreamReader(s));}

		public String next() throws IOException 
		{
			while (st == null || !st.hasMoreTokens()) 
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {return Integer.parseInt(next());}
		
		public long nextLong() throws IOException {return Long.parseLong(next());}

		public String nextLine() throws IOException {return br.readLine();}
		
		public double nextDouble() throws IOException
		{
			String x = next();
			StringBuilder sb = new StringBuilder("0");
			double res = 0, f = 1;
			boolean dec = false, neg = false;
			int start = 0;
			if(x.charAt(0) == '-')
			{
				neg = true;
				start++;
			}
			for(int i = start; i < x.length(); i++)
				if(x.charAt(i) == '.')
				{
					res = Long.parseLong(sb.toString());
					sb = new StringBuilder("0");
					dec = true;
				}
				else
				{
					sb.append(x.charAt(i));
					if(dec)
						f *= 10;
				}
			res += Long.parseLong(sb.toString()) / f;
			return res * (neg?-1:1);
		}
		
		public boolean ready() throws IOException {return br.ready();}


	}
}
