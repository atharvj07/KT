from collections import defaultdict, deque
import threading
answer = []

def bfs(start, end, g):
	d = [float('inf')] * (len(start) + len(end) + 1)
	q = deque()
	global answer
	for i in start:
		d[i] = 0
		q.append(i)
	while q:
		temp = q.popleft()
		for to in g[temp]:
			if d[to] == float('inf'):
				d[to] = d[temp] + 1
				q.append(to)
	for i in end:
		if d[i] != float('inf'):
			answer[i] = d[i]

def f(a):
	global answer
	n = len(a)
	a = [None] + a
	answer = [-1] * len(a)
	odd = []
	even = []
	g = defaultdict(list)
	for i in range(1, len(a)):
		if a[i] & 1:
			odd.append(i)
		else:
			even.append(i)
		if a[i] + i <= n:
			g[a[i] + i].append(i)
		if i - a[i] >= 1:
			g[i - a[i]].append(i)
	bfs(odd, even, g)
	bfs(even, odd, g)
	return answer[1:]
a = input()
lst = list(map(int, input().strip().split()))
print(*f(lst))
