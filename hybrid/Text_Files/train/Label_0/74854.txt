from math import log2
(x, y) = map(lambda p: bin(int(p))[2:], input().split())
(rx, rx2, rx3) = (bin(int(x[::-1], 2))[2:], '1' + x[::-1], bin(int(x, 2) >> int(log2(int(x, 2) & -int(x, 2))))[2:])
if y in [x, rx, rx2, rx3]:
	print('YES')
	exit(0)
(f1, f2, f3, f4) = (y.find(x), y.find(rx), y.find(rx2), y.find(rx3))
x = [x, rx, rx2, rx3]
f = [f1, f2, f3, f4]
for i in range(4):
	(ff, xx) = (f[i], x[i])
	if ff != -1:
		(l, r) = (ff, ff + len(xx))
		if set(y[:l] or ['1']) == set(y[r:] or y[-1]) == set(['1']):
			print('YES')
			exit(0)
print('NO')
