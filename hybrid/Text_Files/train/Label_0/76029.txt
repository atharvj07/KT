s = input().split()
n = int(s[0])
m = int(s[1])
mat = []
vis = [[0 for i in range(1005)] for j in range(1005)]

def check(x, y):
	if x < 0 or x >= n or y < 0 or (y >= m):
		return False
	return mat[x][y] == '#'
dx = (0, 0, 1, 1, -1, -1, -1, 1)
dy = (1, -1, 1, -1, 1, -1, 0, 0)

def judge(x, y):
	for i in range(8):
		if not check(x + dx[i], y + dy[i]):
			return False
	for i in range(8):
		vis[x + dx[i]][y + dy[i]] = 1
	return True

def solve():
	for i in range(n):
		for j in range(m):
			judge(i, j)
	for i in range(n):
		for j in range(m):
			if mat[i][j] == '#' and (not vis[i][j]):
				return False
	return True
for i in range(n):
	s = input()
	mat.append(s)
if solve():
	print('YES')
else:
	print('NO')
