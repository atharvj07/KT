#include <bits/stdc++.h>
using namespace std;
template <class T>
inline T sqr(T x) {
  return x * x;
}
const double EPS = 1e-6;
const int INF = 0x3fffffff;
const long long LINF = INF * 1ll * INF;
const double PI = acos(-1.0);
using namespace std;
long long dp[15][15];
long long digit[15];
long long dfs(int pos, int cnt, bool limit) {
  if (pos < 0) return cnt == 0;
  if (!limit && ~dp[pos][cnt]) return dp[pos][cnt];
  int ed = limit ? digit[pos] : 9;
  int ans = 0;
  for (int i = 0; i <= ed; i++) {
    int tmp = (i == 4 || i == 7);
    if (cnt >= tmp) ans += dfs(pos - 1, cnt - tmp, i == ed && limit);
  }
  if (!limit) return dp[pos][cnt] = ans;
  return ans;
}
long long d[15];
long long use[15];
long long a[15];
long long dfs1(int cur, int tot, int cnt) {
  if (cur == 6) return tot >= 0;
  long long ans = 0;
  for (int i = 0; i <= tot; i++) {
    long long tmp = (d[i] - use[i]);
    if (tmp == 0) continue;
    use[i]++;
    ans += tmp * dfs1(cur + 1, tot - i, cnt) % 1000000007;
    use[i]--;
    ans %= 1000000007;
  }
  return ans;
}
int main(void) {
  int m;
  scanf("%d", &m);
  int cnt = 0;
  while (m) {
    digit[cnt++] = m % 10;
    m /= 10;
  }
  memset(dp, -1, sizeof(dp));
  for (int i = 0; i <= cnt; i++) {
    d[i] = dfs(cnt - 1, i, true);
  }
  d[0]--;
  for (int i = 0; i <= cnt; i++) {
    a[i] = dfs1(0, i, cnt);
  }
  long long ans = 0;
  for (int i = 1; i <= cnt; i++) {
    ans += 1ll * d[i] * a[i - 1] % 1000000007;
    ans %= 1000000007;
  }
  printf("%I64d\n", ans);
}
