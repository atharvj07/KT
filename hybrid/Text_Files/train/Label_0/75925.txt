def checkKey(dict, key):
	if key in dict:
		return True
	return False
(n, k) = map(int, input().split())
l = [0]
for i in range(1, 2 * n + 1):
	l.append(i)
index = 2
while k > 0:
	(l[index], l[index - 1]) = (l[index - 1], l[index])
	index += 2
	k -= 1
for i in range(1, 2 * n + 1):
	print(l[i], end=' ')
print()
