#include <bits/stdc++.h>
using namespace std;

int main(){
  int N; cin >> N;
  vector<int> A(N);
  vector<int64_t> B(N);
  for(int i=0;i<N;i++){
    cin >> A[i];
    B[A[i]-1]++;
  }
  int64_t rans=0;
  for(int i=0;i<N;i++){
    rans+=(B[i]-1)*B[i]/2;
  }
  for(int i=0;i<N;i++){
    cout << rans-B[A[i]-1]+1 << endl;
  }
  return 0;
}
