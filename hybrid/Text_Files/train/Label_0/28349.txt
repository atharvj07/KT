#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int t, n;
  scanf("%lld", &t);
  while (t--) {
    long long int temp;
    vector<long long int> a, b;
    scanf("%lld", &n);
    long long int ma = 2000000000;
    long long int mb = 2000000000;
    for (long long int i = 0; i < n; i++) {
      scanf("%lld", &temp);
      a.push_back(temp);
      ma = min(ma, temp);
    }
    for (long long int i = 0; i < n; i++) {
      scanf("%lld", &temp);
      b.push_back(temp);
      mb = min(mb, temp);
    }
    long long int ans = 0;
    for (long long int i = 0; i < n; i++) {
      long long int tt = min(a[i] - ma, b[i] - mb);
      a[i] = a[i] - tt;
      b[i] = b[i] - tt;
      ans += tt;
      ans = ans + (a[i] - ma) + (b[i] - mb);
    }
    printf("%lld\n", ans);
  }
  return 0;
}
