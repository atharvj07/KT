def ff(m):
	d = 3
	r = set()
	dd = round(pow(m, 0.5)) + 1
	while m - 1 and d < dd:
		if m % d == 0:
			r.add(d)
			m = m // d
			dd = round(pow(m, 0.5)) + 1
		else:
			d += 2
	if m != 1:
		r.add(m)
	return r
for _ in range(int(input())):
	res = 0
	n = int(input())
	nums = []
	one = 0
	even = 0
	for i in input().split():
		i = int(i)
		if i == 0:
			continue
		if i == 1:
			one += 1
		elif i % 2 == 0:
			even += 1
		else:
			nums.append(ff(i))
	n = len(nums)
	res += one * even
	del one
	res += even * n
	res += sum(range(even))
	if len(nums) > 1:
		while nums:
			num = nums.pop()
			res += sum([1 for i in nums if num & i])
	print(res)
