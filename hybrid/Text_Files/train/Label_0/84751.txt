import sys
input = sys.stdin.buffer.readline
(n, k) = map(int, input().split())
tank = [(k, 0)]
new = []
L = [0] * n
R = [0] * n
A = [0] * n
for i in range(n):
	(L[i], R[i], A[i]) = map(int, input().split())
for i in range(n):
	new = []
	(l, r, a) = (L[i], R[i], A[i])
	val = -1
	for (p, q) in tank:
		if p + (r - l) * k < a:
			continue
		(ct, v) = (0, 0)
		if p >= a:
			new.append((p - a, q + a))
			v = p - a
		else:
			tmp = a - p
			if tmp % k == 0:
				new.append((0, q + a))
				ct = tmp // k
			else:
				amari = (tmp // k + 1) * k
				new.append((p + amari - a, q + a))
				ct = tmp // k + 1
				v = p + amari - a
		if r - l > ct or (i < n - 1 and r < L[i + 1]):
			if val == -1 or val > q + a + v:
				val = q + a + v
	if val != -1:
		new.append((k, val))
	tank = new
res = -1
for (p, q) in tank:
	if res == -1:
		res = q
	elif res > q:
		res = q
print(res)
