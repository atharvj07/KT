def sol(A):
	A = sorted(A)
	minn = 2 ** 31
	for i in range(len(A) - 1):
		if A[i + 1] - A[i] < minn:
			minn = A[i + 1] - A[i]
			(a, b, x) = (A[i], A[i + 1], i)
	ans = [a] + A[x + 2:] + A[:x] + [b]
	return ' '.join(map(str, ans))
