import java.util.Scanner;

public class Vasya_The_Hipster {
    private static final Scanner SCANNER = new Scanner(System.in);

    public static void main(String[] args) {
        int[] socks = initializing();
        int red_socks = socks[0],
            blue_socks = socks[1],
            difference = Math.abs(red_socks - blue_socks);
        int[] days = new int[2];

        if (difference == 0) {
            days[0] = red_socks;
            days[1] = 0;
        }
        else {
            if (red_socks > blue_socks) {
                days[0] = blue_socks;
            }
            else {
                days[0] = red_socks;
            }

            days[1] = difference / 2;
        }

        System.out.print(days[0] + " " + days[1]);
    }

    private static int[] initializing() {
        int[] input = new int[2];

        while (true) {
            for (int i = 0; i < 2; i++) {
                input[i] = SCANNER.nextInt();
            }

            if (input[0] >= 1 && input[0] <= 100 && input[1] >= 1 && input[1] <= 100) {
                break;
            }
        }

        return input;
    }
}
