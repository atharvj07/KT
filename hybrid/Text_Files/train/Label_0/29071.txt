#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e6+6;
ll t, n, a[maxn];

ll check(ll l, ll r)
{
	cout << "? " << l << " " << r << endl;
	fflush(stdout);	
	ll x;
	cin >> x;
	return x;
}

int main()
{
	cin >> n;
	ll l = 1, r = n;
	ll bian = check(1, n);
	fflush(stdout);	
	if(n == 2)
	{
		if(bian == 1) cout << "! 2" << endl;
		else cout << "! 1" << endl;
		return 0;
	}
	if(bian != 1 && check(l, bian) == bian) // left
	{
		r = bian;
		if(check(bian-1, bian) == bian)
		{
			cout << "! " << bian-1 << endl;
			return 0;
 		}
		while(l < r)
		{
			ll mid = (l+r+1) >> 1;
			if(check(mid, bian) == bian)
			{
				l = mid;
			}
			else
			{
				r = mid-1;
			}
		}
		cout << "! " << l << endl;
	}
	else // right
	{
		l = bian;
		if(check(bian, bian+1) == bian)
		{
			cout << "! " << bian+1 << endl;
			return 0;
		}
		while(l < r)
		{
			ll mid = (l+r) >> 1;
			if(check(bian, mid) == bian)
			{
				r = mid;
			}
			else 
			{
				l = mid+1;
			}
		}
		cout << "! " << l << endl;
	}
	return 0;
}
