#include <bits/stdc++.h>
using namespace std;
int n, a[100111];
unsigned long long gcd(unsigned long long p, unsigned long long q) {
  return q ? gcd(q, p % q) : p;
}
int main() {
  unsigned long long sum = 0, cur = 0;
  scanf("%d", &n);
  for (int i = (1), _b = (n); i <= _b; i++) scanf("%d", &a[i]);
  sort(a + 1, a + n + 1);
  for (int i = (1), _b = (n); i <= _b; i++) {
    sum += a[i];
    if (i > 1) {
      cur += (a[i] - a[i - 1]) * (unsigned long long)(i - 1);
      sum += cur * 2;
    }
  }
  unsigned long long g = gcd(sum, n);
  cout << sum / g << " " << n / g << endl;
  return 0;
}
