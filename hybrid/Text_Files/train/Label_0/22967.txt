#include <bits/stdc++.h>
using namespace std;
long long n;
string ch;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> ch;
  string test[6] = {"RGB", "RBG", "BRG", "BGR", "GBR", "GRB"};
  long long ans[6];
  memset(ans, 0, sizeof(ans));
  for (int i = 0; i < ch.size(); i++) {
    for (int j = 0; j < 6; j++) {
      ans[j] += (ch[i] != test[j][i % 3]);
    }
  }
  long long mm = (long long)1e9;
  for (auto a : ans) {
    mm = min(mm, a);
  }
  cout << mm << "\n";
  for (int i = 0; i < 6; i++) {
    if (ans[i] == mm) {
      for (int j = 0; j < ch.size(); j++) {
        cout << test[i][j % 3];
      }
      return 0;
    }
  }
  return 0;
}
