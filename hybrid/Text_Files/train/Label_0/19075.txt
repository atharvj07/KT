#include <bits/stdc++.h>
using namespace std;
long long parent(long long par[], long long idx) {
  if (par[idx] == idx) return idx;
  return par[idx] = parent(par, par[idx]);
}
void solve() {
  long long n;
  cin >> n;
  long long arr[n], i, par[n + 1], x, y, ans = 0;
  for (i = 0; i < n; i++) cin >> arr[i];
  for (i = 1; i <= n; i++) par[i] = i;
  for (i = 0; i < n; i++) {
    x = parent(par, i + 1);
    y = parent(par, arr[i]);
    par[x] = y;
  }
  for (i = 1; i <= n; i++) {
    x = parent(par, i);
    if (x == i) ans++;
  }
  cout << ans << endl;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  long long t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
