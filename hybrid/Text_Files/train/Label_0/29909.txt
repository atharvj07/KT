#include <bits/stdc++.h>
using namespace std;
char s[100007];
int main() {
  ios_base::sync_with_stdio(0);
  int n, m;
  scanf("%d %d %s", &n, &m, s);
  char* c = s;
  long long a = 0;
  int g = 0;
  int rep = 0;
  for (char* x = s; *x; x++) rep += x[0] == x[1];
  for (; *c; c++) {
    int r = 1;
    while (1 [c] == *c) c++, r++;
    a += m * n - n;
    g++;
  }
  c = s;
  for (; *c;) {
    if (!c[1]) break;
    if (c[1] == *c) {
      c++;
      continue;
    }
    int al = 2;
    while (c[al] == c[al & 1]) al++;
    for (int i = 0; i < al - 1; i++) a -= al - i - 1;
    c += al - 1;
  }
  cout << a;
  return 0;
}
