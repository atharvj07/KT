from functools import lru_cache

def read_int():
	return int(input().strip())

def read_ints():
	return list(map(int, input().strip().split(' ')))

def solve():
	N = read_int()
	a = read_ints()

	@lru_cache(None)
	def dp(i):
		for j in range(i % a[i], N, a[i]):
			if j == i:
				continue
			if a[j] > a[i] and (not dp(j)):
				return True
		return False
	return ''.join(('A' if dp(i) else 'B' for i in range(N)))
print(solve())
