n = int(input())
l = input()
r = input()
ldict = [[] for i in range(27)]
rdict = [[] for i in range(27)]
for i in range(n):
	if l[i] == '?':
		ldict[26].append(i + 1)
	else:
		ldict[ord(l[i]) - ord('a')].append(i + 1)
	if r[i] == '?':
		rdict[26].append(i + 1)
	else:
		rdict[ord(r[i]) - ord('a')].append(i + 1)
ans = []
for i in range(26):
	if len(ldict[i]) < len(rdict[i]):
		smaller = 0
		mn = len(ldict[i])
	else:
		smaller = 1
		mn = len(rdict[i])
	for j in range(mn):
		ans.append([ldict[i][j], rdict[i][j]])
	if smaller == 0 and ldict[26] != []:
		for j in range(mn, len(rdict[i])):
			ans.append([ldict[26][-1], rdict[i][j]])
			del ldict[26][-1]
			if ldict[26] == []:
				break
	elif smaller == 1 and rdict[26] != []:
		for j in range(mn, len(ldict[i])):
			ans.append([ldict[i][j], rdict[26][-1]])
			del rdict[26][-1]
			if rdict[26] == []:
				break
mn = min(len(ldict[26]), len(rdict[26]))
for i in range(mn):
	ans.append([ldict[26][i], rdict[26][i]])
print(len(ans))
for i in ans:
	print(i[0], i[1])
