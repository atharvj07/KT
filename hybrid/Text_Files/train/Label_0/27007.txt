#include<cstdio>
using namespace std;
typedef long long ll;
ll a,b,ans=1;
int main()
{
    scanf("%lld%lld",&a,&b);
    for(ll i=2;i*i<=a;i++)
    {
        if(a%i==0)
        {
            if(b%i==0)
            ans++;
            while(a%i==0)
            a/=i;
        }
    }
    ans+=(a>1&&b%a==0);
    printf("%lld\n",ans);
    return 0;
}
