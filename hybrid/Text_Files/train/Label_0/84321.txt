def palindrome(x):
	if x[len(x)::-1] == x[0:len(x)]:
		return True
	else:
		return False

def count(x):
	count = 0
	for k in range(1, len(x)):
		if x[k - 1] == x[k]:
			count += 1
	return count + 1
