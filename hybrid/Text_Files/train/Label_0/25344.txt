#include <bits/stdc++.h>
using namespace std;
map<string, int> m, n;
string s;
int a, b;
int main() {
  while (cin >> s) {
    int F = 0;
    while (1) {
      int v = s.find_last_of('\\');
      if (v == 2) break;
      s = s.substr(0, v);
      int f = n[s];
      m[s] += F;
      n[s]++;
      a = max(a, m[s]);
      b = max(b, n[s]);
      if (!f) F++;
    }
  }
  cout << a << ' ' << b << endl;
  return 0;
}
