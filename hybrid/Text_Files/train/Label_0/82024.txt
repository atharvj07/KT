(n, a, b, p, q) = map(int, input().split())
if p >= q:
	st = a
else:
	st = b

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)

def lcm(a, b):
	return a * b // gcd(a, b)
both = lcm(a, b)
diva = n // a
divb = n // b
divboth = n // both
tot = (diva - divboth) * p + (divb - divboth) * q + divboth * max(p, q)
print(tot)
