t = int(input())
for i in range(t):
	n = int(input())
	b = list(map(int, input().split()))
	if len(set(b)) == 1:
		print(-1)
	else:
		m = max(b)
		j = 0
		while j < n:
			if j < n - 1 and b[j] == m and (b[j + 1] != m):
				print(j + 1)
				break
			if j > 0 and b[j] == m and (b[j - 1] != m):
				print(j + 1)
				break
			j += 1
