import math
for test in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	ans = []
	if n == 1:
		print(0)
		continue
	if n == 2:
		print(1)
		print(1, 2, min(a), min(a) + 1)
		continue
	s = max(a)
	for i in range(n - 1):
		m = min(a[i], a[i + 1])
		s += 1
		a[i] = s
		a[i + 1] = m
		t = [i + 1, i + 2, s, m]
		ans.append(t)
	if math.gcd(a[-1], a[-2]) != 1:
		i = a[-2] + 1
		while True:
			if math.gcd(a[-3], i) == 1 and math.gcd(i, a[-1]) == 1:
				m = i
				break
			i += 1
		t = [n - 1, n, m, a[-1]]
		ans.append(t)
	print(len(ans))
	for i in ans:
		print(*i)
