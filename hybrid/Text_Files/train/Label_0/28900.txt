#include <bits/stdc++.h>
using namespace std;
int a[100001];
int main() {
  int n, x;
  map<int, int> freq;
  cin >> n >> x;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    freq[a[i]]++;
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int p = a[i];
    freq[p]--;
    int y = p ^ x;
    ans += freq[y];
  }
  cout << ans << endl;
  return 0;
}
