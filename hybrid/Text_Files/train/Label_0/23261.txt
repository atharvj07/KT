#include <bits/stdc++.h>
using namespace std;
int const N = 300000;
struct S {
  int a, b;
  S() {}
  S(int x, int y) : a(x), b(y) {}
  S operator+(S o) { return S(a + o.a, b + o.b); }
  void rm(int t) {
    if (t == 1)
      b += a, a = 0;
    else if (t == 2)
      a += b, b = 0;
  }
};
struct Q {
  int l, r, k;
  Q() {}
  void sc() { scanf("%d%d%d", &l, &r, &k); }
} qu[N];
int n, q;
vector<int> mp0, z;
vector<pair<int, int> > mp;
vector<S> s;
inline int gtmp(int x) {
  return upper_bound(mp.begin(), mp.end(), make_pair(x, 0)) - mp.begin();
}
void bld(int l = 0, int r = (int)mp.size() - 1, int id = 1) {
  if (l == r) {
    s[id].a = mp[l].second - mp[l].first + 1;
    s[id].b = 0;
    return;
  }
  int m = l + r >> 1, a = id << 1, b = a | 1;
  bld(l, m, a);
  bld(m + 1, r, b);
  s[id] = s[a] + s[b];
}
inline void sl(int id, int a, int b) {
  if (z[id]) {
    s[a].rm(z[id]);
    s[b].rm(z[id]);
    z[a] = z[id];
    z[b] = z[id];
    z[id] = 0;
  }
}
void rm(int x, int y, int k, int l = 0, int r = (int)mp.size() - 1,
        int id = 1) {
  if (y < l || x > r) return;
  if (l >= x && r <= y) {
    s[id].rm(k);
    z[id] = k;
    return;
  }
  int m = l + r >> 1, a = id << 1, b = a | 1;
  sl(id, a, b);
  rm(x, y, k, l, m, a);
  rm(x, y, k, m + 1, r, b);
  s[id] = s[a] + s[b];
}
int main() {
  scanf("%d%d", &n, &q);
  for (int i = 0; i < (int)(q); ++i) {
    qu[i].sc();
    mp0.push_back(qu[i].l);
    mp0.push_back(qu[i].r);
  }
  sort(mp0.begin(), mp0.end());
  mp0.resize(unique(mp0.begin(), mp0.end()) - mp0.begin());
  int l = 1;
  for (int i = 0; i < (int)(mp0.size()); ++i) {
    int x = mp0[i];
    if (x > l) mp.push_back(make_pair(l, x - 1));
    mp.push_back(make_pair(x, x));
    l = x + 1;
  }
  if (l <= n) mp.push_back(make_pair(l, n));
  for (int i = 0; i < (int)(q); ++i)
    qu[i].l = gtmp(qu[i].l), qu[i].r = gtmp(qu[i].r);
  s.resize(mp.size() << 2);
  z.resize(mp.size() << 2);
  bld();
  for (int i = 0; i < (int)(q); ++i) {
    rm(qu[i].l, qu[i].r, qu[i].k);
    printf("%d\n", s[1].a);
  }
}
