#include <bits/stdc++.h>
using namespace std;
const long long maxn = 3e5 + 5, b[2] = {313, 499}, mod = 1e9 + 9;
long long sz[maxn], kalbas[2][maxn], dif[maxn], dp[maxn], dp2[maxn], c[maxn],
    h[maxn];
string s;
bitset<maxn> big;
vector<long long> adj[maxn];
map<tuple<long long, long long, long long>, long long> M;
void getsz(long long v, long long p = -1) {
  if (v) {
    kalbas[0][v] = (kalbas[0][p] * b[0] + s[v]) % mod;
    kalbas[1][v] = (kalbas[1][p] * b[1] + s[v]) % mod;
  }
  sz[v] = 1;
  for (auto u : adj[v]) {
    if (u != p) {
      h[u] = h[v] + 1;
      getsz(u, v);
      sz[v] += sz[u];
    }
  }
}
void add(long long v, long long p, long long val) {
  M[make_tuple(h[v], kalbas[0][v], kalbas[1][v])] += val;
  if (M[make_tuple(h[v], kalbas[0][v], kalbas[1][v])] == 0) {
    M.erase(make_tuple(h[v], kalbas[0][v], kalbas[1][v]));
  }
  for (auto u : adj[v]) {
    if (u != p && !big[u]) {
      add(u, v, val);
    }
  }
}
void dfs(long long v, long long p, bool keep) {
  long long bigchild = -1, mx = -1;
  for (auto u : adj[v]) {
    if (u != p) {
      if (sz[u] > mx) {
        mx = sz[u];
        bigchild = u;
      }
    }
  }
  for (auto u : adj[v]) {
    if (u != p && u != bigchild) dfs(u, v, 0);
  }
  if (bigchild != -1) {
    dfs(bigchild, v, 1);
    big[bigchild] = 1;
  }
  add(v, p, 1);
  dif[v] = M.size() + c[v];
  dp[v] = dif[v];
  dp2[v] = 1;
  for (auto u : adj[v]) {
    if (u != p) {
      if (dp[u] > dp[v]) {
        dp[v] = dp[u];
        dp2[v] = dp2[u];
      } else if (dp[u] == dp[v]) {
        dp2[v] += dp2[u];
      }
    }
  }
  if (bigchild != -1) big[bigchild] = 0;
  if (!keep) {
    add(v, p, -1);
  }
}
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long n;
  cin >> n;
  for (long long i = 0; i < n; i++) cin >> c[i];
  cin >> s;
  for (long long i = 0; i < n - 1; i++) {
    long long v, u;
    cin >> v >> u;
    v--;
    u--;
    adj[v].push_back(u);
    adj[u].push_back(v);
  }
  kalbas[0][0] = kalbas[1][0] = s[0] % mod;
  getsz(0);
  dfs(0, -1, 0);
  cout << dp[0] << '\n' << dp2[0];
}
