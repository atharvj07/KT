import math
t = int(input())
while t > 0:
	inp = input()
	q = list(map(int, inp.split(' ')))
	n = q[0]
	m = q[1]
	k = q[2]
	cards_on_one = n // k
	if cards_on_one >= m:
		print(m)
	else:
		winnerJ = cards_on_one
		jokers = m - winnerJ
		oJ = jokers // (k - 1)
		if jokers % (k - 1) == 0:
			print(winnerJ - oJ)
		else:
			print(winnerJ - oJ - 1)
	t -= 1
