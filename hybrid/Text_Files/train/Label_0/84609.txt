for _ in range(int(input())):
	n = int(input())
	l1 = [int(_) for _ in input().split()]
	d1 = {}
	for i in range(n):
		if l1[i] in d1:
			d1[l1[i]][i % 2] += 1
		else:
			d1[l1[i]] = [0, 0]
			d1[l1[i]][i % 2] += 1
	l1.sort()
	for i in range(n):
		d1[l1[i]][i % 2] -= 1
	for i in d1:
		if d1[i] != [0, 0]:
			print('NO')
			break
	else:
		print('YES')
