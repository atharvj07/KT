def add(x, r, c):
	pyramid[r][c] += x
	amount = pyramid[r][c]
	if amount > 1 and r + 1 < n:
		amount -= 1
		pyramid[r + 1][c] += amount / 2
		pyramid[r + 1][c + 1] += amount / 2
		pyramid[r][c] = 1
(n, t) = map(int, input().split(' '))
pyramid = []
for i in range(n):
	pyramid.append([0 for y in range(i + 1)])
for r in range(t):
	add(1, 0, 0)
	for row in range(n):
		for col in range(row + 1):
			add(0, row, col)
total = 0
for row in range(n):
	for col in range(row + 1):
		if pyramid[row][col] >= 1:
			total += 1
print(total)
