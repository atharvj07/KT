#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n;
  cin >> n;
  string l;
  cin >> l;
  string r;
  cin >> r;
  vector<vector<int>> left(26);
  vector<vector<int>> right(26);
  vector<int> wl;
  vector<int> wr;
  int i, j;
  for (i = (0); i < (n); ++i) {
    if (l[i] != '?') {
      left[l[i] - 'a'].push_back(i);
    } else {
      wl.push_back(i);
    }
  }
  for (i = (0); i < (n); ++i) {
    if (r[i] != '?') {
      right[r[i] - 'a'].push_back(i);
    } else {
      wr.push_back(i);
    }
  }
  vector<pair<int, int>> p;
  vector<int> cl(26), cr(26);
  for (i = (0); i < (26); ++i) {
    for (j = (0); j < (min(left[i].size(), right[i].size())); ++j) {
      p.push_back(make_pair(left[i][j] + 1, right[i][j] + 1));
    }
    cl[i] = cr[i] = min(left[i].size(), right[i].size());
  }
  for (i = (0); i < (26); ++i) {
    while (!wr.empty() && cl[i] < left[i].size()) {
      p.push_back(make_pair(left[i][cl[i]] + 1, wr[wr.size() - 1] + 1));
      cl[i]++;
      wr.pop_back();
    }
  }
  for (i = (0); i < (26); ++i) {
    while (!wl.empty() && cr[i] < right[i].size()) {
      p.push_back(make_pair(wl[wl.size() - 1] + 1, right[i][cr[i]] + 1));
      cr[i]++;
      wl.pop_back();
    }
  }
  for (i = (0); i < (min(wl.size(), wr.size())); ++i) {
    p.push_back(make_pair(wl[i] + 1, wr[i] + 1));
  }
  cout << p.size() << endl;
  for (i = (0); i < (p.size()); ++i) {
    cout << p[i].first << " " << p[i].second << endl;
  }
  return 0;
}
