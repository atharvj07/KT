#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 9;
int x[N];
int main() {
  int n;
  long long A, sum = 0;
  scanf("%d %I64d", &n, &A);
  for (__typeof(n) i = 0; i < n; ++i) scanf("%d", x + i), sum += x[i];
  for (__typeof(n) i = 0; i < n; ++i) {
    long long s = sum - x[i], L = A - s, R = A - n + 1, a = (L > 1 ? L : 1),
              b = (x[i] < R ? x[i] : R), ans;
    if (a <= b)
      ans = x[i] - b + a - 1;
    else
      ans = x[i];
    printf("%I64d ", ans);
  }
  return 0;
}
