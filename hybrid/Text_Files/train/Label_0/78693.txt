import sys, math, cmath
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return s[:len(s) - 1]

def invr():
	return map(int, input().split())
t = 1
for tt in range(t):
	n = inp()
	if n == 0:
		print(0)
		continue
	a = ['0'] * (5 - math.floor(math.log(n) / math.log(2))) + list(bin(n).replace('0b', ''))
	(a[1], a[5]) = (a[5], a[1])
	(a[2], a[3]) = (a[3], a[2])
	print(int(''.join(a), 2))
