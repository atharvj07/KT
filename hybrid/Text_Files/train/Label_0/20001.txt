#include <bits/stdc++.h>
#define REP(i,n,N) for(int i=n;i<(int)N;i++)
#define RREP(i,n,N) for(int i=N-1;i>=0;i--)
#define p(s) cout<<(s)<<endl
#define CK(n,a,b) ((a)<=(n)&&(n)<(b))
#define F first
#define S second
typedef long long ll;
using namespace std;
const int inf=1e9;

int H,W;
char a[110][110];
//set<int> s;
map<char,int> m;
string ans;
int key;
int main() {
	cin>>H>>W;
	REP(i,0,H){
		REP(j,0,W){
			cin>>a[i][j];
			m[a[i][j]]++;
		}
	}

	if(H%2) key=W/2;
	if(W%2) key+=H/2;

	int cnt=0;
	int one=0;
	for(char i='a';i<='z';i++){
		if(m[i]%4==0) continue;
		else if(m[i]%4==2) cnt++;
		else if (m[i]%4==1) one++;
		else {
			cnt++;
			one++;
		}
	}

	if(one>1||((H*W)%2==0&&one==1)||cnt>key) ans="No";
	else ans="Yes";
	p(ans);

	return 0;
}
