#include <bits/stdc++.h>
using namespace std;
int main() {
  int p[3];
  int x;
  for (int i = 0; i < 3; i++) {
    cin >> p[i];
  }
  for (int i = 0; i < 3; i++) {
    for (int j = i + 1; j < 3; j++) {
      if (p[i] < p[j]) {
        x = p[i];
        p[i] = p[j];
        p[j] = x;
      }
    }
  }
  if (p[0] + p[1] > p[2] && p[0] + p[2] > p[1] && p[2] + p[1] > p[0]) {
    cout << "0";
  } else {
    cout << p[0] - p[1] - p[2] + 1;
  }
  return 0;
}
