import java.util.Scanner;

public class B136 {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int A = in.nextInt();
        int C = in.nextInt();
        int B = 0;
        int power = 1;
        while (A != 0 || C != 0) {
            int b = (3 + C%3 - A%3)%3;
            B += b*power;
            power *= 3;
            A /= 3;
            C /= 3;
        }
        System.out.println(B);
    }

}
