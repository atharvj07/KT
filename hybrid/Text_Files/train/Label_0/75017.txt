def check_2(data):
	sum_ = sum(data)
	ans = 0
	data.sort()
	for i in range(len(data)):
		curm = 0
		if i != len(data) - 1:
			curm = data[-1]
		else:
			curm = data[-2]
		cur = (len(data) - 1) * max(curm, (sum_ + len(data) - 2) // (len(data) - 1)) - sum_
		ans = max(ans, cur)
	return ans
n = int(input())
for i in range(n):
	m = int(input())
	data = list(map(int, input().split()))
	print(check_2(data))
