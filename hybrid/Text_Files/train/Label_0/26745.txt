using namespace std;

typedef unsigned int uint;
typedef long long ll;
typedef unsigned long long ull;

const int _dx[] = {0,1,0,-1};
const int _dy[] = {-1,0,1,0};

int getInt(){
  int ret = 0,c;
  c = getchar();
  while(!isdigit(c)) c = getchar();
  while(isdigit(c)){
    ret *= 10;
    ret += c - '0';
    c = getchar();
  }
  return ret;
}

int n;
int pz[8][8];
bool chk[8][8];
int cnt;
ull ans;
vector<ull> v[64];

#define ENC(x,y) (((ull)1) << ((y)*n + (x)))

void check(int x, int y, int sum, ull f){
  chk[y][x] = true;

  if(sum == 0){
    v[cnt].push_back(f);
  }else if(sum < 0){
    REP(i,4){
      int xx = x + _dx[i];
      int yy = y + _dy[i];
      if(ISIN(xx,yy,n,n) && !chk[yy][xx] && pz[yy][xx] >= 0)
        check(xx,yy,sum+pz[yy][xx],(f|ENC(xx,yy)));
    }
  }

  chk[y][x] = false;
}

bool solve(int now, ull f){
  if(now == cnt)
    return f == ans;

  REP(i,v[now].size()){
    if((f & v[now][i]) == 0){
      if(solve(now+1, (f|v[now][i])))
        return true;
    }
  }

  return false;
}

int main(){
  while(scanf("%d",&n),n){
    int sum = 0;
    cnt = 0;

    REP(i,n) REP(j,n) scanf("%d",&pz[i][j]);
    REP(i,n) REP(j,n){
      int t = pz[i][j];
      sum += t;
      if(t < 0){
        v[cnt].clear();

        check(j,i,t,ENC(j,i));
        cnt++;
      }
    }

    if(sum != 0){
      puts("NO");
      continue;
    }

    ans = 0;
    REP(i,n*n){
      ans <<= 1;
      ans |= 1;
    }

    puts(solve(0,0)?"YES":"NO");
  }
  return 0;
}
