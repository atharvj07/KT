import java.util.*;

public class ProD {
	static int n,m,p,ans;
	static char[] aa,bb;
	static int[][][][] dp=new int[2][1001][1001][2];

	public static void main(String[] args) {
		Scanner in=new Scanner(System.in);
		n=in.nextInt();m=in.nextInt();
		p=in.nextInt();
		aa=(" "+in.next()).toCharArray();
		bb=(" "+in.next()).toCharArray();
		for(int k=1;k<=p;k++)
		{
			for(int i=1;i<=n;i++)
			for(int j=1;j<=m;j++)
			{
				dp[1][i][j][1]=dp[1][i][j][0]=0;
				if(aa[i]==bb[j])
					dp[1][i][j][1]=Math.max(dp[1][i-1][j-1][1],dp[0][i-1][j-1][0])+1;
				dp[1][i][j][0]=Math.max(dp[1][i][j][0],Math.max(dp[1][i-1][j][0],dp[1][i-1][j][1]));
				dp[1][i][j][0]=Math.max(dp[1][i][j][0],Math.max(dp[1][i][j-1][0],dp[1][i][j-1][1]));
			}
			for(int i=1;i<=n;i++)
			for(int j=1;j<=m;j++)
			{
				dp[0][i][j][0]=dp[1][i][j][0];
				dp[0][i][j][1]=dp[1][i][j][1];
			}
		}
		ans=Math.max(dp[1][n][m][0],dp[1][n][m][1]);
		System.out.println(ans);
	}
}
