#include <bits/stdc++.h>
const int INF = 0x3f3f3f3f;
const int N = 1e5 + 75;
using namespace std;
int n, a[N];
long long ans;
struct Node {
    int a, b;
} ji[N], ou[N];
inline bool cmp(Node a, Node b) {return a.b > b.b;}
int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
    for (int i = 1; i <= n; i += 2)
        ji[a[i]].b++, ji[a[i]].a = a[i], ou[a[i + 1]].b++, ou[a[i + 1]].a = a[i + 1];
    sort(ou + 1, ou + N + 1, cmp);
    sort(ji + 1, ji + N + 1, cmp);
    if (ou[1].a == ji[1].a)
        printf("%d\n", min(n - ji[2].b - ou[1].b, n - ji[1].b - ou[2].b));
    else
        printf("%d\n", n - ji[1].b - ou[1].b);
    return 0;
}
