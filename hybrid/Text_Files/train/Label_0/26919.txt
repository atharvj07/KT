int dp[1024];

void dfs(int i, set<int>& path, set<int>& ns, int x)
{
    path.insert(i);

    for (int to : g[i])
        x ^= grundy(to);
    ns.insert(x);

    for (int to : g[i])
        dfs(to, path, ns, x ^ grundy(to));

    path.erase(i);
}
int grundy(int i)
{
    if (dp[i] != -1)
        return dp[i];

    set<int> ns, path;
    dfs(i, path, ns, 0);

    int x = 0;
    while (ns.count(x))
        ++x;
    return dp[i] = x;
}
