import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	pos = [0] * (2 * n + 1)
	for i in range(n):
		pos[A[i]] = i
		pos[B[i]] = i
	l = ans = float('inf')
	for i in range(2 * n, 0, -1):
		if i % 2:
			ans = min(ans, l + pos[i])
		else:
			l = min(l, pos[i])
	return ans
for _ in range(int(input())):
	print(solve())
