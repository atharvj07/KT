from sys import stdin, stdout
import os, io
stdin = io.BytesIO(os.read(0, os.fstat(0).st_size))
(n, m) = stdin.readline().split()
(n, m) = (int(n), int(m))
adj_li = [[] for i in range(n + 1)]
for i in range(m):
	(u, v) = stdin.readline().split()
	(u, v) = (int(u), int(v))
	adj_li[u].append(v)
	adj_li[v].append(u)
visited = [0 for i in range(n + 1)]
ans = 0
for i in range(1, n + 1):
	if not visited[i]:
		flag = 1
		q = [(i, -1)]
		visited[i] = 1
		for (curr, par) in q:
			for i in adj_li[curr]:
				if i != par:
					if not visited[i]:
						q.append((i, curr))
						visited[i] = 1
					else:
						flag = 0
		if flag:
			ans += 1
stdout.write(str(ans))
