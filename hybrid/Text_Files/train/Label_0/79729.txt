import itertools

def is_digraph(edges):
	n = len(edges)
	color = [None] * n
	color[0] = 1
	q = [0]
	while q:
		pos = q.pop()
		c = 1 - color[pos]
		for nxt in edges[pos]:
			if color[nxt] is None:
				color[nxt] = c
				q.append(nxt)
			elif color[nxt] != c:
				return (False, None)
	return (True, color)

def solve():
	(n, m) = map(int, input().split())
	v = list(map(int, input().split()))
	t = list(map(int, input().split()))
	edges = [list() for __ in range(n)]
	parity = 0
	for __ in range(m):
		(i, j) = map(int, input().split())
		i -= 1
		j -= 1
		edges[i].append(j)
		edges[j].append(i)
	for x in itertools.chain(v, t):
		parity ^= x & 1
	if parity:
		print('NO')
		return
	(di, colors) = is_digraph(edges)
	if not di:
		print('YES')
	else:
		sums = [0, 0]
		for (i, c) in enumerate(colors):
			sums[c] += t[i] - v[i]
		(a, b) = sums
		print('YES') if a == b else print('NO')
for testcases in range(int(input())):
	solve()
