#include <bits/stdc++.h>
using namespace std;
string foo(int n) {
  int zeta = 2;
  while (zeta * (zeta - 1) / 2 <= n) {
    zeta++;
  }
  zeta--;
  int rem = n - zeta * (zeta - 1) / 2;
  string ans = "133";
  string z;
  while (rem--) z += "7";
  zeta -= 2;
  while (zeta--) z += "3";
  z += "7";
  ans += z;
  return ans;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    cout << foo(n) << "\n";
  }
}
