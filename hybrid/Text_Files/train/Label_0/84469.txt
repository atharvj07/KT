from sys import exit
from itertools import permutations as prm

def to_base(n, base):
	ans = []
	while n // base > 0:
		ans.append(n % base)
		n //= base
	ans.append(n % base)
	ans.reverse()
	return ans

def le(l, n, m):
	prev_eq = True
	for i in range(len(n)):
		if l[i] == n[i]:
			continue
		if l[i] > n[i]:
			return False
		else:
			break
	for i in range(len(m)):
		if l[i + len(n)] == m[i]:
			continue
		if l[i + len(n)] > m[i]:
			return False
		else:
			break
	return True
(n, m) = map(lambda x: to_base(int(x) - 1, 7), input().split())
ln = len(n)
lm = len(m)
if ln + lm > 7:
	print(0)
	exit(0)
ans = 0
for i in prm(range(7), ln + lm):
	if le(i, n, m):
		ans += 1
print(ans)
