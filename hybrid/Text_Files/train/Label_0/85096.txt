(ax, ay, bx, by, cx, cy) = map(int, input().split())
import sys

def gcd(a, b):
	if min(a, b) == 0:
		return max(a, b)
	return gcd(min(a, b), max(a, b) % min(a, b))
if ax == bx and bx == cx:
	print('No')
	sys.exit()
if ay == by and by == cy:
	print('No')
	sys.exit()
coords = []
coords.append((ax, ay))
coords.append((bx, by))
coords.append((cx, cy))
coords = sorted(coords)
delta = set()

def getdelta(f, s):
	dx = coords[s][0] - coords[f][0]
	dy = coords[s][1] - coords[f][1]
	g = gcd(abs(dx), abs(dy))
	dx /= g
	dy /= g
	tmp = (dx, dy)
	if tmp in delta:
		return False
	delta.add(tmp)
	return True
for i in range(3):
	for j in range(3):
		if i == j:
			continue
		if not getdelta(i, j):
			print('No')
			sys.exit()

def d(a, b):
	return (a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2
if d((ax, ay), (bx, by)) == d((bx, by), (cx, cy)):
	print('Yes')
	sys.exit()
print('No')
