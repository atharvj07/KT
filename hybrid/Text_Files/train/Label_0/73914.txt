sol = ['x.x', '.xx', 'xx.']
g = []
g1 = []
(s1, s2) = ('', '')
for i in range(4):
	g.append(input())
for i in range(4):
	s = ''
	s1 += g[i][i]
	s2 += g[i][4 - i - 1]
	for j in g:
		s += j[i]
	g1.append(s)
g1.append(s1)
g1.append(s2)
s = ''
for i in range(3):
	s += g[i][i + 1]
g1.append(s)
s = ''
for i in range(3):
	s += g[i + 1][i]
g1.append(s)
s = ''
for i in range(3):
	s += g[i][3 - 1 - i]
g1.append(s)
s = ''
for i in range(3):
	s += g[i + 1][4 - 1 - i]
g1.append(s)
g += g1

	for j in sol:
		if i.count(j):
			print('YES')
