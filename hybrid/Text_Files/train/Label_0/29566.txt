#include <bits/stdc++.h>
using namespace std;
const long long pii = 3.1415926535897;
const long long oo = 1e18 + 13;
const int N = 1e6 + 13;
const int M = 1e3 + 13;
const double eps = 0.0000001;
long long n;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  vector<long long> res;
  for (int i = 1; i <= n; ++i)
    if (n - res.size() * (res.size() + 1) / 2 < i) {
      res.back() += n - res.size() * (res.size() + 1) / 2;
      break;
    } else if (n - res.size() * (res.size() + 1) / 2 == i) {
      res.push_back(i);
      break;
    } else
      res.push_back(i);
  cout << res.size() << "\n";
  for (long long x : res) cout << x << " ";
}
