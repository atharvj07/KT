(n, m) = map(int, input().split())
arr = [int(o) for o in input().split()]
barr = [int(o) for o in input().split()]
for i in range(2 ** 9 + 1):
	co = 0
	for a in arr:
		flag = 0
		for b in barr:
			if a & b & i == a & b:
				flag = 1
				break
		if flag == 1:
			co += 1
	if co == n:
		print(i)
		break
