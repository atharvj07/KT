#include<iostream>
using namespace std;
char mas[100][100];
int dx[4]={1,0,-1,0},dy[4]={0,1,0,-1};
int w,h,cnt;
void dts(int x,int y);
int main(){
  while(1){
    cnt=0;
    cin >> h >> w;
    if(w == 0 && h == 0){
      break;
    }
    for(int i=0;i<h;i++){
      for(int j=0;j<w;j++){
	cin >> mas[i][j];
      }
    }
    for(int i=0;i<h;i++){
      for(int j=0;j<w;j++){
	if(mas[i][j]=='*' || mas[i][j]=='#' || mas[i][j]=='@'){
	  cnt++;
	  dts(j,i);
	}
      }
    }
    cout << cnt <<endl;
  }
}
void dts(int x,int y){
  char temp=mas[y][x];
  mas[y][x]='.';
  for(int i=0;i<4;i++){
    int nx=x+dx[i];
    int ny=y+dy[i];
    if(nx>=0 && nx<w && ny<h && ny>=0 && mas[ny][nx]==temp){
      dts(nx,ny);
    }
  }
}
