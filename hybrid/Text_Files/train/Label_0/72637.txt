(vi, vf) = [int(k) for k in input().split()]
(t, d) = [int(k) for k in input().split()]
distances = [0] * t
for i in range(t - 1):
	f_max = vi + i * d
	b_max = vf + (t - 1 - i) * d
	maxi = min(f_max, b_max)
	distance = vi
	v = vi
	for j in range(i):
		v += min(d, maxi - v)
		distance += v
	distance1 = vf
	v = vf
	for j in range(t - 2, i, -1):
		v += min(d, maxi - v)
		distance1 += v
	distances[i] = distance + distance1
print(max(distances))
