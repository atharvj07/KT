#include<bits/stdc++.h>
using namespace std;

const int mod=998244353;
int f[2][410][410],n,a[410],b[410],S,C[410][410],ans,x,y;

int quickmi(int a,int b)
{
	int t=1;
	for (; b; b>>=1,a=1ll*a*a%mod)
		if (b&1) t=1ll*t*a%mod;
	return t;
}

int main()
{
	for (int i=0; i<=400; i++)
	{
		C[i][0]=1;
		for (int j=1; j<=i; j++) C[i][j]=(C[i-1][j-1]+C[i-1][j])%mod;
	}
	scanf("%d",&n),S=0;
	for (int i=1; i<=n; i++) scanf("%d%d",&a[i],&b[i]),S+=a[i];
	f[0][0][0]=mod-1,x=1,y=0;
	for (int i=1,nw; i<=n; i++)
	{
		x^=1,y^=1;
		memcpy(f[y],f[x],sizeof(f[x]));
		for (int j=0; j<=S-a[i]; j++)
		{
			nw=1;
			for (int k=0; k<b[i]; k++,nw=1ll*nw*a[i]%mod)
				for (int l=0; l<=400-k; l++)
					f[y][j+a[i]][k+l]=(f[y][j+a[i]][k+l]-1ll*f[x][j][l]*C[k+l][l]%mod*nw)%mod;
		}
	}
	ans=0;
	for (int nw,i=1,s; i<=S; i++) 
	{
		nw=quickmi(i,mod-2),s=nw;
		for (int j=0; j<=400; j++,s=1ll*s*nw%mod) ans=(ans+1ll*f[y][i][j]*s)%mod;
	}
	ans=1ll*ans*S%mod;
	return printf("%d\n",(ans+mod)%mod),0;
}
