#include <bits/stdc++.h>
using namespace std;

int main(){
	int i;
	string S;
	cin >> S;
	vector<int> ans(S.size(), 0);
	for(i = 0; i < S.size(); ){
		int j, k;
		for(j = i; j < S.size() && S[j] == 'R'; j++);
		for(k = j; k < S.size() && S[k] == 'L'; k++);
//		printf("(i, j, k) = (%d, %d, %d)\n", i, j, k);
		ans[j - 1] = k / 2 - i / 2;
		ans[j] = k - i - ans[j - 1];
		if(j % 2 == 1){
			swap(ans[j - 1], ans[j]);
		}
		i = k;
	}
	for(i = 0; i < S.size(); i++){
		printf("%d ", ans[i]);
	}
	printf("\n");
	return 0;
}
