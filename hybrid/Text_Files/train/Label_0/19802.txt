using namespace std;
int a[10][10],b[10][10],c[21],d[21],n,m;
int mx;
void fn(int p,int q,int r){
  int h=1,i;
  for(i=0;i<n;i++){
    if(a[q][i]&&b[q][i]){
      b[q][i]=b[i][q]=0;
      c[p]=i;
      fn(p+1,i,r+a[q][i]);
      b[q][i]=b[i][q]=-1;
      h=0;
    }
  }
  if(h){
    if(mx<r){
      mx=r;
      c[p]=-1;
      memcpy(d,c,sizeof(d));
    }
  }
  return;
}
int main(){
  int i;
  memset(b,-1,sizeof(b));
  while(cin>>n>>m,n||m){
    memset(a,0,sizeof(a));
    for(i=0;i<m;i++){
      int p,q,r;
      cin>>p>>q>>r;
      p--;
      q--;
      a[p][q]=a[q][p]=r;
    }
    mx=0;
    for(i=0;i<n;i++){
      c[0]=i;
      fn(1,i,0);
    }
    cout<<mx<<endl;
    for(i=0;d[i+1]!=-1;i++)
      cout<<d[i]+1<<" ";
    cout<<d[i]+1<<endl;
  }
  return 0;
}
