import java.io.*;
import java.util.*;

public class A{

	static int MOD=(int)1e9+7;
	static int ALPHA=52;
	static int getIdx(char c) {
		if(c>='a' && c<='z')
			return c-'a';
		return 26+c-'A';
	}
	static int []fac;
	
	
	static int modInverse(int x) {
		long b=x;
		int e=MOD-2;
		long ans=1;
		while(e>0) {
			if(e%2==1)
				ans=ans*b%MOD;
			e>>=1;
			b=b*b%MOD;
		}
		return (int) ans;
	}
	
	public static void main(String[] args) throws IOException {
		Scanner sc=new Scanner();
		PrintWriter out=new PrintWriter(System.out);
		char []s=sc.next().toCharArray();
		int n=s.length;
		fac=new int [n+10];
		long x=1;
		fac[0]=1;
		for(int i=1;i<fac.length;i++) {
			x=x*i%MOD;
			fac[i]=(int)x;
		}
		int []freq=new int [ALPHA];
		for(char c:s)
			freq[getIdx(c)]++;
		int []dp=new int [n+1];
		dp[0]=1;
		for(int i=0;i<ALPHA;i++) {
			if(freq[i]==0)
				continue;
			for(int j=n;j>=freq[i];j--) {
				dp[j]=dp[j]+dp[j-freq[i]];
				if(dp[j]>=MOD)
					dp[j]-=MOD;
			}
	            
		}
		
		
		long X=fac[n/2]*1L*fac[n/2]%MOD;
		for(int z:freq)
			X=X*modInverse(fac[z])%MOD;
		int [][]ans=new int [ALPHA][ALPHA];
		for(int i=0;i<ALPHA;i++)
			ans[i][i]=dp[n/2];
		for(int i=0;i<ALPHA;i++) {
			if(freq[i]==0)
				continue;
			int []tmp=dp.clone();
			for(int k=freq[i];k<=n;k++) {
				tmp[k]=tmp[k]-tmp[k-freq[i]];
				if(tmp[k]<0)
					tmp[k]+=MOD;
			}
	        for(int j=i+1;j<ALPHA;j++) {
	        	if(freq[j]==0)
	        		continue;
	        	for(int k=freq[j];k<=n;k++) {
					tmp[k]=tmp[k]-tmp[k-freq[j]];
					if(tmp[k]<0)
						tmp[k]+=MOD;
				}
	        	ans[j][i]=ans[i][j]=2*tmp[n/2];
	        	for(int k=n;k>=freq[j];k--) {
					tmp[k]=tmp[k]+tmp[k-freq[j]];
					if(tmp[k]>=MOD)
						tmp[k]-=MOD;
				}
	        }
		}
		int q=sc.nextInt();
		while(q-->0) {
			int l=getIdx(s[sc.nextInt()-1]),r=getIdx(s[sc.nextInt()-1]);
			out.println(ans[l][r]*X%MOD);
			
		}
		out.close();

	}
	
	static class Scanner
	{
		BufferedReader br;
		StringTokenizer st;
		Scanner(){
			br=new BufferedReader(new InputStreamReader(System.in));
		}
		Scanner(String fileName) throws FileNotFoundException{
			br=new BufferedReader(new FileReader(fileName));
		}
		String next() throws IOException {
			while(st==null || !st.hasMoreTokens())
				st=new StringTokenizer(br.readLine());
			return st.nextToken();
		}
		String nextLine() throws IOException {
			return br.readLine();
		}
		int nextInt() throws IOException{
			return Integer.parseInt(next());
		}
		long nextLong()  throws NumberFormatException, IOException {
			return Long.parseLong(next());
		}
		double nextDouble() throws NumberFormatException, IOException {
			return Double.parseDouble(next());
		}
	}
}
