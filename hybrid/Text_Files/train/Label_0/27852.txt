#include <bits/stdc++.h>
#pragma GCC optimise("ofast")
#pragma GCC optimise("unroll-loops")
using namespace std;
const int N = 23, M = (1 << N) + 10;
const long long MOD = 1000000000 + 7;
const long long INF = 1000000000000000000;
const long long LOG = 25;
int dp[M];
int n, m, cnt;
int adj[N];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; i++) adj[i] |= (1 << i);
  for (int i = 1; i <= m; i++) {
    int v, u;
    cin >> v >> u;
    v--, u--;
    adj[v] |= (1 << u);
    adj[u] |= (1 << v);
  }
  if (m == n * (n - 1) / 2) return cout << 0, 0;
  for (int i = 0; i < n; i++) {
    dp[(1 << i)] = adj[i];
  }
  for (int mask = 1; mask < (1 << n); mask++) {
    for (int i = 0; i < n; i++) {
      if (((mask & (1 << i)) == 0) && ((dp[mask] & (1 << i)) != 0)) {
        dp[mask | (1 << i)] = dp[mask] | adj[i];
      }
    }
  }
  int Min = n;
  int Mask = 0;
  for (int mask = 0; mask < (1 << n); mask++) {
    if (__builtin_popcount(dp[mask]) == n) {
      if (Min > __builtin_popcount(mask))
        Min = __builtin_popcount(mask), Mask = mask;
    }
  }
  cout << Min << '\n';
  for (int i = 0; i < n; i++) {
    if (Mask & (1 << i)) cout << i + 1 << ' ';
  }
  return 0;
}
