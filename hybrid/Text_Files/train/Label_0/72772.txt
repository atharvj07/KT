def main():
	t = int(input())
	for _ in range(t):
		n = int(input())
		d = list(input())
		ans1 = [0] * n
		ans2 = [0] * n
		flag = 0
		for i in range(n):
			if d[i] == '0':
				ans1[i] = '0'
				ans2[i] = '0'
			elif d[i] == '1':
				if flag:
					ans1[i] = '1'
					ans2[i] = '0'
				else:
					ans1[i] = '0'
					ans2[i] = '1'
					flag = 1
			elif flag:
				ans1[i] = '2'
				ans2[i] = '0'
			else:
				ans1[i] = '1'
				ans2[i] = '1'
		print(''.join(ans1))
		print(''.join(ans2))
main()
