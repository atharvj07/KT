#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, b;
  cin >> n >> a >> b;
  a -= 1;
  b -= 1;
  int cnt = 0;
  while (n > 2 && a != b) {
    n /= 2;
    a /= 2;
    b /= 2;
    cnt++;
  }
  if (n == 2 && a != b) {
    cout << "Final!" << endl;
  } else {
    cout << cnt << endl;
  }
  return 0;
}
