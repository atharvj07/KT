#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5, maxd = 1e7 + 7;
int x, y, b[maxn], n[maxn][26], cnt[maxd + 2 * maxn];
bool dp[maxd + 2 * maxn];
char t[maxn], p[maxn];
void pre() {
  b[0] = -1;
  for (int i = 0, j = -1; i < y; i++) {
    while (j >= 0 && p[j] != p[i]) j = b[j];
    j++;
    b[i + 1] = j;
  }
  for (int j = 0; j < y; j++) {
    for (char c = 'a'; c <= 'z'; c++) {
      if (p[j] == c)
        n[j][c - 'a'] = j + 1;
      else
        n[j][c - 'a'] = n[b[j]][c - 'a'];
    }
  }
}
inline int pos(int i, int j) { return i * (y + 1) + j + 1; }
int go() {
  int ans = 0;
  dp[pos(0, 0)] = 1;
  for (int i = 0; i < x; i++) {
    char l = 'a', u = 'z';
    if (t[i] != '?') l = u = t[i];
    for (int j = 0; j < y; j++)
      if (dp[pos(i, j)]) {
        for (char c = l; c <= u; c++) {
          int jj = n[j][c - 'a'];
          int add = 0;
          if (jj == y) {
            jj = b[jj];
            add = 1;
          }
          dp[pos(i + 1, jj)] = 1;
          cnt[pos(i + 1, jj)] = max(cnt[pos(i + 1, jj)], cnt[pos(i, j)] + add);
        }
      }
  }
  for (int i = 0; i < y; i++) ans = max(ans, cnt[pos(x, i)]);
  return ans;
}
int main() {
  scanf("%s %s", t, p);
  x = strlen(t);
  y = strlen(p);
  pre();
  printf("%d\n", go());
  return 0;
}
