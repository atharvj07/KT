#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, minn = 0x3f3f3f3f;
  int a[110];
  int b[110] = {0};
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  int tmpx, tmpy;
  for (int i = 1; i <= n; i++) {
    if (i == n) {
      b[i] = abs(a[i] - a[1]);
    } else {
      b[i] = abs(a[i] - a[i + 1]);
    }
    if (b[i] <= minn) {
      minn = b[i];
      if (i == n) {
        tmpx = n;
        tmpy = 1;
      } else {
        tmpx = i;
        tmpy = i + 1;
      }
    }
  }
  printf("%d %d\n", tmpx, tmpy);
  return 0;
}
