from sys import *
input = stdin.readline
n = int(input())
r = []
(x, y) = (0, 0)
l = []
for i in range(n):
	l.append(list(map(int, input().split())))
for i in range(n):
	if abs(x + l[i][0] - y) <= 500:
		r.append('A')
		x += l[i][0]
	else:
		y += l[i][1]
		r.append('G')
if abs(x - y) <= 500:
	print(''.join(r))
else:
	print(-1)
