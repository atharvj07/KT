from collections import deque

def solve():
	n = int(input())
	arr = [0] + list(map(int, input().split()))
	brr = [0] + list(map(int, input().split()))
	p = [-1] * (n + 1)
	p[n] = n
	bd = n
	queue = deque()
	queue.append((n, n))
	while queue:
		(a, b) = queue.popleft()
		for x in range(b - arr[b], min(b + 1, bd)):
			if p[x] == -1:
				p[x] = a
				queue.append((x, x + brr[x]))
		bd = min(bd, b - arr[b])
	if p[0] == -1:
		print(-1)
	else:
		ans = [0]
		while ans[-1] != n:
			ans.append(p[ans[-1]])
		ans.pop()
		print(len(ans))
		print(*ans[::-1])
import sys
input = lambda : sys.stdin.readline().rstrip()
solve()
