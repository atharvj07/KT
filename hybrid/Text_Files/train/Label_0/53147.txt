
import java.util.Scanner;
public class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int v=s.nextInt();
        String l = s.next();
        char n[]=l.toCharArray();
        int sf=0;
        int fs=0;
        for(int i=0;i<(v-1);i++){
          if(n[i]=='S' && n[i+1]=='F'){
              sf++;
        
 
          }else if(n[i]=='F' && n[i+1]=='S'){
              fs++;

          }
        }
        if(sf>fs){
        System.out.println("YES");
        }else{
            System.out.println("NO");
        }
    }
    }
