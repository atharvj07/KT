import sys
import math
n = int(sys.stdin.readline())
p = [int(i) for i in sys.stdin.readline().split()]
ans = [[]] * n
for i in range(n):
	ans[i] = [0] * (i + 1)
for i in range(n):
	num = p[i]
	at = (i, i)
	ans[at[0]][at[1]] = num
	for j in range(num - 1):
		if at[1] != 0 and ans[at[0]][at[1] - 1] == 0:
			at = (at[0], at[1] - 1)
			ans[at[0]][at[1]] = num
		else:
			at = (at[0] + 1, at[1])
			ans[at[0]][at[1]] = num
for i in range(n):
	print(' '.join([str(s) for s in ans[i]]))
