for t in range(int(input())):
	n = int(input())
	b = list(map(int, input().split()))
	a = [0] * (2 * n + 1)
	res = []
	f = True
	for i in b:
		a[i] = 1
	for i in b:
		res.append(i)
		val = 0
		for j in range(i, 2 * n + 1):
			if a[j] != 1:
				val = j
				break
		if val == 0:
			f = False
			break
		a[val] = 1
		res.append(val)
	if f:
		print(*res)
	else:
		print(-1)
