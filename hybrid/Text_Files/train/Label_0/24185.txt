const bool MULTIPLE_TEST_CASES = true;

const l MOD=1e9+7;
const l N=2e5+5;
const l INF=1e12;


l n=0,m;
vvi adj(N);
vector<bool> visited(N,false);
vi dist(N,-1);

void bfs(l cur) {
    queue<l> q;
    l d = 0;
    dist[cur] = 0;
    d++;
    q.push(cur);
    while(!q.empty()) {
        l num = q.size();
        rep(i,0,num) {
            l node = q.front();
            q.pop();
            for(auto neighbour : adj[node]) {
            	// cout<<"visit "<<neighbour<<" from "<<node<<"\n";
                if(dist[neighbour]==-1) {
                    dist[neighbour] = d;
                    q.push(neighbour);
                }
            }
        }
        d++;
    }
}


vvi dp(N, vi(2,-1));

l minDist(l node, l left) {
	if(left<0) {
		return INF;
	}
	if(dist[node]==0) {
		return 0;
	}
	if(dp[node][left]!=-1) {
		return dp[node][left];
	}
	
	l ans = dist[node];
	for(auto neighbour: adj[node]) {
		if(dist[neighbour]>dist[node]) {
			ans = min(ans, minDist(neighbour,left));
		}
		ans = min(ans, minDist(neighbour,left-1));
	}
	dp[node][left] = ans;
	return ans;


	// l ans = d[node];


}

void solve() {
	rep(i,0,n) {
		adj[i].clear();
		visited[i] = false;
		dist[i] = -1;
		dp[i][0] = -1;
		dp[i][1] = -1;

	}
	cin>>n>>m;
	rep(i,0,m) {
		l u,v;
		cin>>u>>v;
		u--;
		v--;
		adj[u].pb(v);
	}
	bfs(0);

	rep(i,0,n) {
		cout<<min(minDist(i,0), minDist(i,1))<<" ";
	}
	cout<<"\n";
	return;
}


int main(){

	if (MULTIPLE_TEST_CASES) cin>>t;
	rep(i,0,t) {
		solve();
	}
