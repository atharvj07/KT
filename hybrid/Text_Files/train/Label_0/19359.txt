#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int N = 3e5 + 10;
const int inf = 2e9;
const double eps = 1e-7;
const long long INF = (long long)1e18;
int n, k;
pair<int, int> a[N];
int main() {
  int x, y, res = 0;
  scanf("%d", &(n));
  for (int i = 0; i < (n); ++i) {
    scanf("%d", &(x));
    a[i] = make_pair(x, i + 1);
  }
  scanf("%d", &(k));
  sort(a, a + n);
  long long sum = 0;
  long long dis = 0;
  for (int i = 0; i < k; i++) {
    dis += 1ll * i * a[i].first - sum;
    sum += a[i].first;
  }
  long long ans = dis;
  for (int i = k; i < n; i++) {
    sum -= a[i - k].first;
    dis -= (sum - 1ll * a[i - k].first * (k - 1));
    dis += 1ll * a[i].first * (k - 1) - sum;
    sum += a[i].first;
    if (dis < ans) {
      ans = dis;
      res = i - k + 1;
    }
  }
  for (int i = res; i < res + k; i++) cout << a[i].second << " ";
  return 0;
}
