t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	a = [list(map(int, input().split())) for i in range(n)]
	l = n + m - 1
	A = [[] for i in range(l)]
	ans = 0
	for i in range(n):
		for j in range(m):
			A[i + j].append(a[i][j])
	for i in range(l // 2):
		t = A[i] + A[-1 - i]
		ans += min(t.count(0), t.count(1))
	print(ans)
