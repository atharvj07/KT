import java.awt.Point;
import java.io.*;
import java.math.BigInteger;
import java.util.*;
import static java.lang.Math.*;

public class C {

	final boolean ONLINE_JUDGE = System.getProperty("ONLINE_JUDGE") != null;
	BufferedReader in;
	PrintWriter out;
	StringTokenizer tok = new StringTokenizer("");

	void solve() throws IOException {
		long n = readLong();
		long t = readLong();
		long[][] dp = dp();
		int pow = 0;
		long cur = 1;
		for (int i = 0; cur <= t; i++) {
			if (cur == t) {
				pow = i;
				break;
			} else {
				cur *= 2;
			}
		}
		if (cur != t) {
			out.println(0);
			return;
		}

		long res = 0;
		int curPow = 1;
		long curCh = 2;
		while (curCh * 2 <= n) {
			res += dp[curPow][pow];
			curPow++;
			curCh *= 2;
		}
		res += res(n+1, curPow, pow, curCh, curCh*2 -1, dp);
		out.println(res);
	}
	
	long res(long n, int i, int j, long left, long right, long[][] dp){
		if(j < 0) return 0;
		if(i < 0) return 0;
		long res = 0; 
		if(i == 0){
			res += dp[i][j];
			return res;
		}
		long mid = (left+right)/2;
		
		if(n > mid){
			res += dp[i-1][j];
			res += res(n, i-1, j-1, mid+1, right, dp);
		}
		else{
			res =  res(n, i-1, j, left, mid, dp);
		}
		return res;
	}

	long[][] dp() {
		long[][] dp = new long[44][44];
		dp[0][0] = 1;
		for (int i = 1; i < 44; i++) {
			dp[i][0] = 1;
			for (int j = 1; j < 44; j++) {
				dp[i][j] = dp[i - 1][j] + dp[i - 1][j - 1];
			}
		}
		return dp;
	}

	void init() throws FileNotFoundException {
		if (ONLINE_JUDGE) {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
		} else {
			in = new BufferedReader(new FileReader("input.txt"));
			out = new PrintWriter("output.txt");
		}
	}

	String readString() throws IOException {
		while (!tok.hasMoreTokens()) {
			tok = new StringTokenizer(in.readLine());
		}
		return tok.nextToken();
	}

	int readInt() throws IOException {
		return Integer.parseInt(readString());
	}

	long readLong() throws IOException {
		return Long.parseLong(readString());
	}

	double readDouble() throws IOException {
		return Double.parseDouble(readString());
	}

	int[] readArr(int n) throws IOException {
		int[] res = new int[n];
		for (int i = 0; i < n; i++) {
			res[i] = readInt();
		}
		return res;
	}

	long[] readArrL(int n) throws IOException {
		long[] res = new long[n];
		for (int i = 0; i < n; i++) {
			res[i] = readLong();
		}
		return res;
	}

	public static void main(String[] args) {
		new C().run();
	}

	public void run() {
		try {
			long t1 = System.currentTimeMillis();
			init();
			solve();
			out.close();
			long t2 = System.currentTimeMillis();
			System.err.println("Time = " + (t2 - t1));
		} catch (Exception e) {
			e.printStackTrace(System.err);
			System.exit(-1);
		}
	}
}
