def fun(ans):
	o = []
	z = []
	i = 0
	j = len(ans) - 1
	while i < j:
		if ans[i] != '1':
			i += 1
		if ans[j] != '0':
			j -= 1
		if i < j and ans[j] == '0' and (ans[i] == '1'):
			o.append(i + 1)
			z.append(j + 1)
			i += 1
			j -= 1
	x = o + z[::-1]
	return x
for _ in range(int(input())):
	x = input()
	arr = input()
	x = fun(arr)
	if not x:
		print(0)
	else:
		print(1)
		print(len(x), end=' ')
		for i in x:
			print(i, end=' ')
		print()
