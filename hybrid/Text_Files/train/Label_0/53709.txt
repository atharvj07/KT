import java.util.*;
import java.lang.*;
import java.io.*;

public class M1 {
    public static void main(String[] Args){
        Scanner scan=new Scanner(System.in);
        int n=scan.nextInt();
        if(n==1) {
            System.out.println(1);
            return;
        }
        if(n==2){
            System.out.println(5);
            return;
        }
        int cubes=5;
        for(int i=3;i<=n;i++){
            cubes+=(i-2)*4+4;
        }
        System.out.println(cubes);
    }
}
