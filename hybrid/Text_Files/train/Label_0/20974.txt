#include <bits/stdc++.h>
using namespace std;
const int maxn = 4e5 + 5;
int n, m, tot_cmp, D[maxn], B[maxn], V[maxn], L[maxn], second[maxn];
vector<pair<int, int> > e[maxn];
pair<int, int> g[maxn];
int dfs_bridges(int u, int idx) {
  V[u] = 1;
  int cur = D[u];
  for (int i = 0; i < e[u].size(); i++) {
    int v = e[u][i].first, jdx = e[u][i].second;
    if (!V[v]) {
      D[v] = D[u] + 1;
      cur = min(cur, dfs_bridges(v, jdx));
    } else if (jdx != idx)
      cur = min(cur, D[v]);
  }
  if (cur == D[u] && idx != -1) {
    B[idx] = 1;
  }
  return cur;
}
void dfs_bg(int _u, int _cmp) {
  queue<pair<int, int> > qc;
  qc.push(make_pair(_u, _cmp));
  while (!qc.empty()) {
    int u = qc.front().first, cmp = qc.front().second;
    qc.pop();
    queue<int> q;
    q.push(u);
    while (!q.empty()) {
      int v = q.front();
      q.pop();
      if (V[v]) continue;
      second[cmp]++;
      V[v] = 1;
      L[v] = cmp;
      for (int i = 0; i < e[v].size(); i++) {
        int w = e[v][i].first, idx = e[v][i].second;
        if (V[w]) continue;
        if (B[idx]) {
          qc.push(make_pair(w, tot_cmp++));
        } else {
          q.push(w);
        }
      }
    }
  }
}
void dfs_or(int u, int idx) {
  V[idx + 1] = 1;
  for (int i = 0; i < e[u].size(); i++) {
    int v = e[u][i].first, jdx = e[u][i].second;
    if (V[jdx + 1]) continue;
    if (B[jdx])
      g[jdx] = make_pair(v, u);
    else
      g[jdx] = make_pair(u, v);
    dfs_or(v, jdx);
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--;
    v--;
    e[u].push_back(make_pair(v, i));
    e[v].push_back(make_pair(u, i));
  }
  dfs_bridges(0, -1);
  memset(V, 0, sizeof V);
  dfs_bg(0, tot_cmp++);
  int idx = -1;
  for (int i = 0; i < n; i++)
    if (idx == -1 || second[L[i]] > second[L[idx]]) idx = i;
  cout << second[L[idx]] << endl;
  memset(V, 0, sizeof V);
  dfs_or(idx, -1);
  for (int i = 0; i < m; i++)
    cout << (g[i].first + 1) << " " << (g[i].second + 1) << "\n";
  return 0;
}
