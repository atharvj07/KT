#include <bits/stdc++.h>
using namespace std;
int main() {
  set<string> msgs;
  vector<string> vec, vec2;
  int n;
  cin >> n;
  for (int i = 0; i != n; i++) {
    string str;
    cin >> str;
    vec.push_back(str);
  }
  for (int i = n - 1; i >= 0; i--)
    if (msgs.count(vec[i]) == 0) vec2.push_back(vec[i]), msgs.insert(vec[i]);
  for (size_t i = 0; i != vec2.size(); i++) cout << vec2[i] << endl;
  return 0;
}
