import os
from io import BytesIO

def main():

	def mult(m1, m2):
		ans = [0] * 128
		for i in range(128):
			for j in range(128):
				ans[i ^ j] += m1[i] * m2[j]
		return ans

	def power(number, n):
		res = number
		while n:
			if n & 1:
				res = mult(res, number)
			number = mult(number, number)
			n >>= 1
		return res
	(n, x) = map(int, input().split())
	ai = list(map(float, input().split()))
	ai += [0] * (128 - len(ai))
	if n == 1:
		print(1 - ai[0])
		return
	ansm = power(ai, n - 1)
	print(1 - ansm[0])
main()
