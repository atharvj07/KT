#include <bits/stdc++.h>
using namespace std;
map<long long, long long> dic;
vector<long long> v;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long a, b, k;
  cin >> a >> b >> k;
  long long num = 720720;
  long long cnt = 0;
  dic[a % num] = cnt;
  v.push_back(a);
  long long cl, cv;
  while (1) {
    if (a == b) {
      cout << cnt << "\n";
      return 0;
    }
    cnt++;
    long long mx = 1;
    for (long long(it) = (2); (it) < (k + 1); (it)++)
      if (a >= a % it + b) mx = max(mx, a % it);
    a -= mx;
    if (dic.count(a % num)) {
      cv = v[dic[a % num]] - a;
      cl = cnt - dic[a % num];
      break;
    } else
      dic[a % num] = cnt;
    v.push_back(a);
  }
  long long tim = (a - b) / cv;
  cnt += cl * tim;
  a -= tim * cv;
  while (1) {
    if (a == b) {
      cout << cnt << "\n";
      return 0;
    }
    cnt++;
    long long mx = 1;
    for (long long(it) = (2); (it) < (k + 1); (it)++)
      if (a >= a % it + b) mx = max(mx, a % it);
    a -= mx;
  }
  return 0;
}
