#include <bits/stdc++.h>
using namespace std;
stack<int> st;
string s;
int have[1000000 + 5];
int main() {
  cin >> s;
  for (int i = (0); i <= (s.size() - 1); i++) {
    if (s[i] == '(') {
      st.push(i);
    } else if (!st.empty()) {
      have[i] = have[st.top()] = 1;
      st.pop();
    }
  }
  int ans = 0, res = 0, num = 1;
  for (int i = (0); i <= (s.size() - 1); i++) {
    if (have[i]) {
      ans++;
    } else {
      if (ans > res) {
        res = ans;
        num = 1;
      } else if (ans != 0 && ans == res) {
        num++;
      }
      ans = 0;
    }
  }
  if (ans > res) {
    res = ans;
    num = 1;
    ans = 0;
  } else if (ans != 0 && ans == res) {
    num++;
  }
  cout << res << " " << num << endl;
  return 0;
}
