(n, k) = map(int, input().split())
s = list(input())
if len(s) == 1 and k >= 1:
	print(0)
else:
	j = 0
	for i in range(k):
		if j >= n:
			break
		while j < n:
			if j == 0 and s[0] != '1':
				s[0] = '1'
				j += 1
				break
			elif s[j] != '0' and j != 0:
				s[j] = '0'
				j += 1
				break
			j += 1
	print(*s, sep='')
