#include <bits/stdc++.h>
int main() {
  unsigned mini, n, m;
  scanf("%u%u", &n, &m);
  if ((n >= m)) {
    if ((n % 2 == 0))
      mini = n / 2;
    else
      mini = ((n - 1) / 2) + 1;
    while (mini % m != 0) {
      mini = mini + 1;
    }
    printf("%u", mini);
  } else
    printf("-1");
  return 0;
}
