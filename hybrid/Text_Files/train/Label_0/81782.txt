from math import ceil

def solve(n, k, p):
	l = [p[0]]
	for i in range(n - 1):
		l.append(l[-1] + p[i + 1])
	c = [p[i + 1] * 100 - k * l[i] for i in range(n - 1)]
	c = max(c)
	print(ceil(c / k) if c > 0 else 0)
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	p = list(map(int, input().split()))
	solve(n, k, p)
