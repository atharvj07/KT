import java.io.*;

public class e 
{
	public static void main(String[] arg)
	{
		new e();
	}
	int n, m, q;
	long[][] memo;
	Edge[] eList;
	LCA[] lcaList;
	public e()
	{
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		n = in.nextInt(); m = in.nextInt(); q = in.nextInt();
		eList = new Edge[m];
		lcaList = new LCA[q];
		for(int i = 0; i < m; i++)
		{
			eList[i] = new Edge(in.nextInt()-1, in.nextInt()-1);
		}
		for(int i = 0; i < q; i++)
		{
			lcaList[i] = new LCA(in.nextInt()-1, in.nextInt()-1, in.nextInt()-1);
		}
		boolean flag = true;
		for(int i = 0; i < q; i++)
		{
			if((lcaList[i].a == 0 || lcaList[i].b == 0) && lcaList[i].c != 0) 
				flag = false;
		}
		if(flag)
		{
			memo = new long[n][1<<n];
			for(int i = 0; i < n; i++)
			{
				Arrays.fill(memo[i], -1);
				memo[i][0] = 1;
			}
			out.println(solve(0, ((1<<n)-1)^1));
		}
		else
		{
			out.println("0");
		}
		in.close(); out.close();
	}
	long solve(int head, int nodes)
	{
		long ret = 0;
		if(memo[head][nodes] != -1) return memo[head][nodes];
		int id = 0;
		for(int i = 0; i < n; i++)
		{
			if((nodes&(1<<i)) != 0)
			{
				id = i;
				break;
			}
		}
		for(int i = nodes; i > 0; i = (i-1)&nodes)
		{
			if((i&(1<<id)) == 0) continue;
			int opp = nodes^i;
			for(int j = 0; j < n; j++)
			{
				if((i&(1<<j)) == 0) continue;
				int child = i^(1<<j);
				// Determine validity
				if(!valid(head, j, child, opp)) 
				{
					continue;
				}
				
				ret += solve(j, child)*solve(head, opp);
			}
		}
		return memo[head][nodes] = ret;
	}
	boolean valid(int curHead, int newHead, int child, int opp)
	{
//		System.out.println("Top: " + curHead + " " + newHead + " " + Integer.toBinaryString(child) + " " + Integer.toBinaryString(opp));
		// Deal with edge conditions
		for(int i = 0; i < m; i++)
		{
			if(curHead == eList[i].u)
			{
				if(((1<<eList[i].v)&child) != 0) return false;
			}
			else if(newHead == eList[i].u)
			{
				if(((1<<eList[i].v)&opp) != 0) return false;
			}
			else if(((1<<eList[i].u)&child) != 0)
			{
				if(eList[i].v == curHead) return false;
				if(((1<<eList[i].v)&opp) != 0) return false;
			}
			else if(((1<<eList[i].u)&opp) != 0)
			{
				if(eList[i].v == newHead) return false;
				if(((1<<eList[i].v)&child) != 0) return false;
			}
		}
//		System.out.println("Middle: " + curHead + " " + newHead + " " + Integer.toBinaryString(child) + " " + Integer.toBinaryString(opp));

		// Deal with LCA conditions
		for(int i = 0; i < q; i++)
		{
			if(lcaList[i].a == curHead)
			{
				if(lcaList[i].b == curHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(lcaList[i].b == newHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(((1<<lcaList[i].b)&child) != 0)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(((1<<lcaList[i].b)&opp) != 0)
				{
					if(lcaList[i].c != curHead) return false;
				}
			}
			else if(lcaList[i].a == newHead)
			{
				if(lcaList[i].b == curHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(lcaList[i].b == newHead)
				{
					if(lcaList[i].c != newHead) return false;
				}
				else if(((1<<lcaList[i].b)&child) != 0)
				{
					if(lcaList[i].c != newHead) return false;
				}
				else if(((1<<lcaList[i].b)&opp) != 0)
				{
					if(lcaList[i].c != curHead) return false;
				}
			}
			else if(((1<<lcaList[i].a)&child) != 0)
			{
				if(lcaList[i].b == curHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(lcaList[i].b == newHead)
				{
					if(lcaList[i].c != newHead) return false;
				}
				else if(((1<<lcaList[i].b)&child) != 0)
				{
					//if(((1<<lcaList[i].c)&child) != 0) return false;
					if(((1<<lcaList[i].c)&opp) != 0) return false;
					if(lcaList[i].c == curHead) return false;
				}
				else if(((1<<lcaList[i].b)&opp) != 0)
				{
					//if(lcaList[i].c == newHead) return false;
					//if(((1<<lcaList[i].c)&child) != 0) return false;
					if(lcaList[i].c != curHead) return false;
				}
			}
			else if(((1<<lcaList[i].a)&opp) != 0)
			{
				if(lcaList[i].b == curHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(lcaList[i].b == newHead)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(((1<<lcaList[i].b)&child) != 0)
				{
					if(lcaList[i].c != curHead) return false;
				}
				else if(((1<<lcaList[i].b)&opp) != 0)
				{
					if(lcaList[i].c == newHead) return false;
					if(((1<<lcaList[i].c)&child) != 0) return false;
					//if(((1<<lcaList[i].c)&opp) != 0) return false;
				}
			}
		}
//		System.out.println("Bot: " + curHead + " " + newHead + " " + Integer.toBinaryString(child) + " " + Integer.toBinaryString(opp));

		return true;
	}
	class Edge
	{
		int u, v;
		public Edge(int a, int b)
		{
			u = a; v = b;
		}
	}
	class LCA
	{
		int a, b, c;
		public LCA(int aa, int bb, int cc)
		{
			a = aa; b = bb; c = cc;
		}
	}
}
