import sys
from bisect import bisect_left as bl, bisect_right as br
sys.setrecursionlimit(100090)

def min_cost(l, r, pos, a, b):
	cnt = br(pos, r) - bl(pos, l)
	if cnt == 0:
		return a
	curr = b * cnt * (r - l + 1)
	if l == r:
		return curr
	m = (l + r) // 2
	return min(curr, min_cost(l, m, pos, a, b) + min_cost(m + 1, r, pos, a, b))
(n, k, a, b) = map(int, input().split())
pos = [int(x) for x in input().split()]
pos.sort()
print(min_cost(1, 2 ** n, pos, a, b))
