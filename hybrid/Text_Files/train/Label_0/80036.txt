SEARCH_FIRST_DIGIT = 0
COMPLETE_50 = 1
COMPLETE_25 = 2

def solve(n):
	n = n[::-1]
	f0 = -1
	f5 = -1
	for i in range(len(n)):
		if f0 == -1 and n[i] == '0':
			f0 = i
			continue
		if f5 == -1 and n[i] == '5':
			f5 = i
			if f0 == -1:
				continue
		if f0 != -1 and n[i] in '05':
			return i - 1
		if f5 != -1 and n[i] in '27':
			return i - 1

def main():
	test_cases = int(input())
	ans = [None] * test_cases
	for t in range(test_cases):
		n = input()
		a = solve(n)
		ans[t] = a
	print(*ans, sep='\n')
main()
