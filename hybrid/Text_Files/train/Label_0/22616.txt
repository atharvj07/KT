void numberOfNodes(int n,vector<ll> adj[],vector<ll>&count1,int s,int e)
{
    vector<ll>::iterator u; 
    count1[s] = 1; 
    for (u = adj[s].begin(); u != adj[s].end(); u++) {
        if (*u == e) 
            continue; 
        numberOfNodes(n,adj,count1,*u, s); 
        count1[s] += count1[*u]; 
    } 
}
