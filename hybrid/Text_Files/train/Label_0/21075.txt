#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  if (n == 1 && m == 2) {
    cout << 0 << "\n"
         << 1 << ' ' << 1 << "\n"
         << 1 << ' ' << 2 << "\n"
         << 1 << " " << 1;
  } else if (n == 2 && m == 1) {
    cout << 0 << "\n"
         << 1 << ' ' << 1 << "\n"
         << 2 << ' ' << 1 << "\n"
         << 1 << " " << 1;
  } else if (n == 1 && m == 1) {
    cout << "0\n";
    cout << 1 << ' ' << 1 << "\n";
    cout << 1 << ' ' << 1;
  } else if (n == 1) {
    cout << "1\n" << n << ' ' << m << ' ' << 1 << ' ' << 1 << "\n";
    for (int i = 1; i <= m; ++i) cout << 1 << ' ' << i << "\n";
    cout << 1 << ' ' << 1;
  } else if (m == 1) {
    cout << "1\n" << n << ' ' << m << ' ' << 1 << ' ' << 1 << "\n";
    for (int i = 1; i <= n; ++i) cout << i << ' ' << 1 << "\n";
    cout << 1 << ' ' << 1;
  } else if (n == 2 && m == 2) {
    cout << 0 << "\n";
    cout << 1 << ' ' << 1 << "\n";
    cout << 1 << ' ' << 2 << "\n";
    cout << 2 << ' ' << 2 << "\n";
    cout << 2 << ' ' << 1 << "\n";
    cout << 1 << ' ' << 1 << "\n";
  } else if (n % 2 == 0 && m % 2 == 0) {
    cout << 0 << "\n";
    cout << 1 << ' ' << 1 << endl;
    int i = 1, j = 2, uk = 1;
    while (i < n) {
      if (j > m) {
        j = m;
        ++i;
        uk = -1;
      }
      if (j == 1) {
        j = 2;
        ++i;
        uk = 1;
      }
      cout << i << ' ' << j << "\n";
      j += uk;
    }
    for (int i = m - 1; i > 1; --i) cout << n << ' ' << i << endl;
    for (int i = n; i > 0; --i) cout << i << ' ' << 1 << "\n";
  } else if (n % 2 != 0 && m % 2 != 0) {
    cout << 1 << "\n";
    cout << n << ' ' << m << ' ' << 1 << ' ' << 1 << "\n";
    int i = 1, j = 1, uk = 1;
    while (i != n || j != m) {
      if (j > m) {
        j = m;
        ++i;
        uk = -1;
      }
      if (j == 0) {
        j = 1;
        ++i;
        uk = 1;
      }
      cout << i << ' ' << j << "\n";
      j += uk;
    }
    cout << n << ' ' << m << "\n" << 1 << ' ' << 1;
  } else if (n % 2 == 0) {
    cout << 0 << "\n";
    for (int i = 1; i <= m; ++i) cout << 1 << ' ' << i << "\n";
    int i = 2, j = m, uk = -1;
    while (i <= n) {
      if (j > m) {
        j = m;
        ++i;
        uk = -1;
      }
      if (j == 1) {
        j = 2;
        ++i;
        uk = 1;
      }
      if (0 < i && i <= n && 0 < j && j <= m) cout << i << ' ' << j << "\n";
      j += uk;
    }
    for (int i = n; i > 0; --i) cout << i << ' ' << 1 << "\n";
  } else if (m % 2 == 0) {
    cout << "0\n";
    for (int i = 1; i <= m; ++i) cout << 1 << ' ' << i << "\n";
    int i = 2, j = m, uk = 1;
    while (i != 1 || j != 1) {
      if (i > n) {
        i = n;
        --j;
        uk = -1;
      }
      if (i == 1) {
        i = 2;
        --j;
        uk = 1;
      }
      if (0 < i && i <= n && 0 < j && j <= m) cout << i << ' ' << j << "\n";
      i += uk;
    }
    cout << 1 << ' ' << 1;
  }
  return 0;
}
