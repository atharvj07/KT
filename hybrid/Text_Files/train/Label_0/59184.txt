import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);

        int a, b = 0, n;
        n = cin.nextInt();

        a = cin.nextInt();
        for (int i = 2; i <= n; i++) {
            b = cin.nextInt();
            System.out.print((a + b) + " ");
            a = b;
        }
        System.out.print(b);
    }
}
