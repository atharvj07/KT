#include <bits/stdc++.h>
using namespace std;
int a[260];
string cmd[260];
int N;
string getcmd(int f, int reg1, int reg2, int p) {
  string ret;
  stringstream ss;
  ss << p;
  if (reg2 == -1) {
    ret += "lea e";
    ret += (char)('a' + f);
    ret += "x";
    ret += ", [";
    ret += ss.str();
    ret += "*e";
    ret += (char)('a' + reg1);
    ret += "x]";
  } else {
    ret += "lea e";
    ret += (char)('a' + f);
    ret += "x";
    ret += ", [";
    ret += "e";
    ret += (char)('a' + reg1);
    ret += "x + ";
    if (p > 1) {
      ret += ss.str();
      ret += "*";
    }
    ret += "e";
    ret += (char)('a' + reg2);
    ret += "x]";
  }
  return ret;
}
int f(int cnt, int maxcnt) {
  if (cnt == maxcnt) {
    if (a[cnt] != N) return false;
    cout << cnt << endl;
    for (int i = 0; i < cnt; i++) cout << cmd[i] << endl;
    return true;
  }
  for (int i = 0; i <= cnt; i++) {
    for (int j = 4; j <= 8; j *= 2) {
      int n = a[i] * j;
      if (cnt + 1 == maxcnt && n != N) continue;
      if (n > N) continue;
      a[cnt + 1] = n;
      cmd[cnt] = getcmd(cnt + 1, i, -1, j);
      int ret = f(cnt + 1, maxcnt);
      if (ret) return true;
    }
    for (int j = 1; j <= 8; j *= 2) {
      int n = a[i] * j + a[i];
      if (cnt + 1 == maxcnt && n != N) continue;
      if (n > N) continue;
      a[cnt + 1] = n;
      cmd[cnt] = getcmd(cnt + 1, i, i, j);
      int ret = f(cnt + 1, maxcnt);
      if (ret) return true;
    }
  }
  for (int i = 0; i <= cnt; i++) {
    for (int j = 0; j <= cnt; j++) {
      if (i == j) continue;
      for (int k = 1; k <= 8; k *= 2) {
        int n = a[i] + a[j] * k;
        if (cnt + 1 == maxcnt && n != N) continue;
        if (n > N) continue;
        a[cnt + 1] = n;
        cmd[cnt] = getcmd(cnt + 1, i, j, k);
        int ret = f(cnt + 1, maxcnt);
        if (ret) return true;
      }
    }
  }
  return false;
}
int main() {
  cout << setprecision(9);
  cin >> N;
  a[0] = 1;
  for (int i = 0;; i++)
    if (f(0, i)) break;
  return 0;
}
