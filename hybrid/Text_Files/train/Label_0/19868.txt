#include <bits/stdc++.h>
using namespace std;
inline long long getint() {
  long long c, x;
  for (; !isdigit(c = getchar());)
    ;
  for (x = c - '0'; isdigit(c = getchar()); x = x * 10 + c - '0')
    ;
  return x;
}
const int mod = 1000000007;
const int N = 200100;
int Rd, ct[100][2];
long long a[100], used, da[N * 2], dp[N];
int vis[N], ne[N * 2], to[N * 2], G[N], xb, ans, n, m, pw[200];
void gt() {
  for (int i = 62; i >= 0; i--)
    if (used >> i & 1)
      ans = (ans + (long long)(ct[i][0] + ct[i][1]) *
                       (ct[i][0] + ct[i][1] - 1) / 2 % mod * pw[i + Rd - 1]) %
            mod;
    else
      ans = (ans + (long long)ct[i][0] * ct[i][1] % mod * pw[i + Rd]) % mod;
  memset(ct, 0, sizeof ct);
  memset(a, 0, sizeof a);
  Rd = 0;
  used = 0;
}
void ins(long long x) {
  used |= x;
  for (int i = 62; i >= 0; i--)
    if (x >> i & 1) !a[i] ? a[i] = x, Rd++ : 0, x ^= a[i];
}
void ord(long long x) {
  for (int i = 62; i >= 0; i--) ct[i][x >> i & 1]++;
}
void dfs(int x, int fa = -1) {
  vis[x] = 1;
  ord(dp[x]);
  for (int i = G[x]; ~i; i = ne[i])
    if (to[i] != fa) {
      if (!vis[to[i]]) {
        dp[to[i]] = dp[x] ^ da[i];
        dfs(to[i], x);
      } else
        ins(dp[to[i]] ^ dp[x] ^ da[i]);
    }
}
void add(long long z, int y, int x) {
  ne[xb] = G[x];
  to[xb] = y;
  da[xb] = z;
  G[x] = xb++;
  ne[xb] = G[y];
  to[xb] = x;
  da[xb] = z;
  G[y] = xb++;
}
int main() {
  n = getint();
  m = getint();
  memset(G, -1, sizeof G);
  pw[0] = 1;
  for (int i = 1; i <= 150; i++) pw[i] = (pw[i - 1] + pw[i - 1]) % mod;
  while (m--) add(getint(), getint(), getint());
  for (int i = 1; i <= n; i++)
    if (!vis[i]) {
      dfs(i);
      gt();
    }
  printf("%d\n", ans);
  return 0;
}
