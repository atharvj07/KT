#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < n; i++)
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
typedef pair<int, int> pi;
typedef pair<pi, pi> pp;
typedef pair<ll, ll> pl;
const double EPS = 1e-9;
const ll MOD = 998244353;
const int inf = 1 << 30;
const ll linf = 1LL << 60;

int N, K;
vector<int> e[100000];
ll vec[2][100000][301];
ll tmp[2][301] = {0};
int sz[100000];

void dfs(int v, int p) {
    rep(i,e[v].size()) {
        if (e[v][i] == p) continue;
        dfs(e[v][i], v);
    }
    vec[0][v][0] = 1;
    vec[1][v][1] = 1;
    sz[v] = 1;
    rep(i,e[v].size()) {
        if (e[v][i] == p) continue;
        memset(tmp, 0, sizeof(tmp));
        for (int j = sz[v]; j >= 0; j--) {
            for (int k = 0; k <= sz[e[v][i]]; k++) {
                if (j+k > K+1) break;
                if (k == K+1) continue;
                if (k+j != K+1) tmp[0][k+j] = (tmp[0][k+j] + vec[0][v][j] * (vec[0][e[v][i]][k] + vec[1][e[v][i]][k])) % MOD;
                if (k+j != K+1) tmp[1][k+j] = (tmp[1][k+j] + vec[1][v][j] * (vec[1][e[v][i]][k] + vec[0][e[v][i]][k])) % MOD;
                if (k+j != 0) tmp[1][k+j-1] = (tmp[1][k+j-1] + vec[1][v][j] * vec[1][e[v][i]][k]) % MOD;
            }
        }
        sz[v] = min(K, sz[v]+sz[e[v][i]]);
        rep(j,2) rep(k,sz[v]+1) vec[j][v][k] = tmp[j][k];
        /*
        cerr << "i: " << i << endl;
        cerr << sz[v] << " " << sz[e[v][i]] << endl;
        rep(j,2) { rep(k,K+1) {
            cerr << tmp[j][k] << " "; } cerr << endl;
        }
        */
    }
}

int main() {
    cin >> N >> K;
    rep(i,N-1) {
        int u, v; cin >> u >> v;
        u--; v--;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    dfs(0, 0);
    cout << (vec[0][0][K] + vec[1][0][K]) % MOD << endl;
}
