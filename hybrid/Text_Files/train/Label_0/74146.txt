s = input()
t = input()
reverse_s = s[::-1]
reverse_t = t[::-1]
check = 0
sequence = []
sequence1 = []
for i in range(0, len(t)):
	if t[i] == s[check]:
		sequence.append(i)
		check += 1
		if check == len(s):
			break
check = 0
for i in range(0, len(reverse_t)):
	if reverse_t[i] == reverse_s[check]:
		sequence1.append(i)
		check += 1
		if check == len(reverse_s):
			break
wynik = 0
if len(sequence) > 0 and len(sequence1) > 0:
	o = sequence.pop()
	p = len(t) - sequence1.pop() - 1
	for i in range(o, p):
		wynik += 1
print(wynik)
