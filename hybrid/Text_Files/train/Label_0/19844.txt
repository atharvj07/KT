#include <bits/stdc++.h>
using namespace std;
const long long MAXN = 5e5 + 5;
const long long MOD = 1e9 + 7;
const long long INF = 1e18;
signed main() {
  long long m, k, n, second;
  cin >> m >> k >> n >> second;
  vector<long long> a(m);
  vector<long long> need(MAXN), now(MAXN);
  for (long long i = 0; i < m; i++) {
    cin >> a[i];
  }
  for (long long i = 0; i < second; i++) {
    long long b;
    cin >> b;
    need[b]++;
  }
  long long r = 0;
  long long sz = m - k * (n - 1);
  long long ok = 0;
  for (long long i = 0; i < MAXN; i++) {
    ok += (need[i] == 0);
  }
  for (long long l = 0; l + sz <= m; l += k) {
    while (r < m && r - l < sz) {
      now[a[r]]++;
      if (now[a[r]] == need[a[r]]) {
        ok++;
      }
      if (ok == MAXN) {
        vector<long long> ans;
        vector<long long> taken(MAXN);
        for (long long i = l; i < r; i++) {
          if (need[a[i]] <= taken[a[i]] && ans.size() < sz - k) {
            ans.push_back(i);
          } else {
            taken[a[i]]++;
          }
        }
        cout << ans.size() << "\n";
        for (long long x : ans) {
          cout << x + 1 << " ";
        }
        cout << "\n";
        return 0;
      }
      r++;
    }
    for (long long j = 0; j < k && l + j < m; j++) {
      if (now[a[l + j]] == need[a[l + j]]) {
        ok--;
      }
      now[a[l + j]]--;
    }
  }
  cout << "-1\n";
  return 0;
}
