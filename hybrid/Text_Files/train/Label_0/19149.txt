#include <bits/stdc++.h>
using namespace std;
inline long long read() {
  char c = getchar();
  long long f = 1, x = 0;
  while (c < '0' || c > '9') {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') {
    x = (x << 1) + (x << 3) + (c ^ '0');
    c = getchar();
  }
  return x * f;
}
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
const int maxn = 100100;
long long MIN(long long x, long long y) { return x < y ? x : y; }
long long MAX(long long x, long long y) { return x > y ? x : y; }
long long Gcd(long long x, long long y) { return y == 0 ? x : Gcd(y, x % y); }
long long f[4];
int main() {
  long long n, k, a, b, i, j;
  n = read();
  k = read();
  a = read();
  b = read();
  f[0] = a + b;
  f[1] = a - b;
  f[2] = -a + b;
  f[3] = -a - b;
  long long x = 1e18 + 5, y = -1;
  for (i = 0; i < n; i++) {
    for (j = 0; j < 4; j++) {
      long long c = (f[j] + k) % k;
      long long s = (i * k + c);
      x = MIN(x, n * k / Gcd(n * k, s));
      y = MAX(y, n * k / Gcd(n * k, s));
    }
  }
  printf("%lld %lld\n", x, y);
  return 0;
}
