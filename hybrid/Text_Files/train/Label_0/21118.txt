#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie();
  cout.tie();
  ;
  int n;
  cin >> n;
  int a[n];
  unordered_map<int, int> m;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    m[a[i]]++;
  }
  unordered_map<int, int> dup;
  for (auto i : m) {
    if (i.second >= 2) {
      dup[i.first] = i.second - 1;
    }
  }
  if (!dup.size()) {
    cout << 0;
    return 0;
  }
  int ans = INT_MAX;
  for (int i = 0; i < n; i++) {
    unordered_map<int, int> val;
    val = dup;
    for (int j = i; j < n; j++) {
      if (val.count(a[j])) {
        val[a[j]]--;
        if (val[a[j]] == 0) {
          val.erase(a[j]);
        }
      }
      if (signed(val.size()) == 0) {
        ans = min(ans, j - i + 1);
        break;
      }
    }
  }
  cout << ans << endl;
}
