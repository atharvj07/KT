#include <bits/stdc++.h>
using namespace std;
string a, b;
void normalize() {
  if (a[0] == '0') {
    int pos = 0;
    while (a[pos] == '0') pos++;
    a = a.substr(pos);
  }
  if (b[0] == '0') {
    int pos = 0;
    while (b[pos] == '0') pos++;
    b = b.substr(pos);
  }
}
inline void solve() {
  cin >> a >> b;
  normalize();
  if (a.size() > b.size())
    puts(">");
  else if (a.size() < b.size())
    puts("<");
  else {
    for (int i = 0; i < a.size(); ++i) {
      if (a[i] < b[i]) {
        puts("<");
        return;
      }
      if (a[i] > b[i]) {
        puts(">");
        return;
      }
    }
    puts("=");
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  solve();
  return 0;
}
