n = int(input())
lsta = list(map(int, input().split()))
lstb = list(map(int, input().split()))
indx1 = 0
indx2 = 0
ct = 0
for i in range(n):
	if lsta[i] != lstb[i] and ct == 0:
		indx1 = i
		ct += 1
	elif lsta[i] != lstb[i]:
		indx2 = i
		ct += 1
if ct == 1:
	lst = list(range(1, n + 1))
	for i in lst:
		if i not in lsta:
			num = i
			break
	lsta[indx1] = num
	print(*lsta)
	exit()
ct11 = lsta.count(lsta[indx1])
ct21 = lstb.count(lstb[indx1])
if lsta[indx1] != lsta[indx2]:
	if ct11 == 2:
		lsta[indx1] = lstb[indx1]
	else:
		lsta[indx2] = lstb[indx2]
	print(*lsta)
else:
	if ct21 == 2:
		lstb[indx1] = lsta[indx1]
	else:
		lstb[indx2] = lsta[indx2]
	print(*lstb)
