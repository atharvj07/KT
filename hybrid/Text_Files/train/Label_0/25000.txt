#include <bits/stdc++.h>
using namespace std;
template <class T>
ostream &operator<<(ostream &out, const vector<T> &v) {
  for (auto it = v.begin(); it != v.end(); ++it) {
    if (it == v.begin())
      out << *it;
    else
      out << " " << *it;
  }
  return out;
}
int n, m, k, X, Y;
int memo[2022][2022];
vector<int> offers;
vector<int> v, prefix;
int getseg(int L, int R) { return prefix[R + 1] - prefix[L]; }
int dp(int pos, int x) {
  if (pos == k) return 0;
  if (memo[pos][x] != INT_MIN) return memo[pos][x];
  int ans = INT_MAX;
  if (offers[x + 1] != -1) {
    int L = pos - x;
    int R = pos - x + offers[x + 1] - 1;
    int unpay = getseg(L, R);
    ans = -unpay + v[pos] + dp(pos + 1, 0);
  }
  ans = min(ans, v[pos] + dp(pos + 1, x + 1));
  ans = min(ans, v[pos] + dp(pos + 1, 0));
  return memo[pos][x] = ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m >> k;
  v.resize(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  sort(v.begin(), v.end());
  while ((int)(v).size() > k) v.pop_back();
  n = (int)(v).size();
  prefix.assign(n + 1, 0);
  for (int i = 0; i < n; i++) prefix[i + 1] = prefix[i] + v[i];
  offers.assign(2022, -1);
  for (int i = 0; i < m; i++) {
    cin >> X >> Y;
    if (X > k) continue;
    offers[X] = max(offers[X], Y);
  }
  for (X = 1; X < (int)(offers).size(); X++)
    offers[X] = max(offers[X], offers[X - 1]);
  for (int i = 0; i < 2022; i++)
    for (int j = 0; j < 2022; j++) memo[i][j] = INT_MIN;
  cout << dp(0, 0) << '\n';
  return 0;
}
