import java.util.*;
import java.io.*;

public class XenonAttackOnGangs {
    static int[][] par, ss;
    static long[][] dp;
    static ArrayList<Integer>[] graph;
    public static void main(String[] args) {
        FastScanner scanner = new FastScanner();
        PrintWriter out = new PrintWriter(System.out,false);
        int n = scanner.nextInt();
        par = new int[n][n];
        ss = new int[n][n];
        dp = new long[n][n];
        graph = new ArrayList[n];
        for(int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for(long[] dd: dp) Arrays.fill(dd, -1);
        int[][] edges = new int[n][2];
        for(int i =1; i < n; i++) {
            int a = scanner.nextInt()-1;
            int b = scanner.nextInt()-1;
            graph[a].add(b);
            graph[b].add(a);
            edges[i][0] = a; edges[i][1] = b;
        }
        for(int i = 0; i < n; i++) { dfs(i,i,i); }
        for(int i = 1; i < n; i++) {
            dp[edges[i][0]][edges[i][1]] = dp[edges[i][1]][edges[i][0]] = ss[edges[i][0]][edges[i][1]] * ss[edges[i][1]][edges[i][0]];
        }
        long best = 0;
        for(int i = 0; i < n; i++) {
            for(int j = i + 1; j < n; j++) {
                best = Math.max(best, dp(i,j));
            }
        }
        out.println(best);
        out.flush();
    }
    static long dp(int i, int j) {
        if (dp[i][j] == -1) {
            //all paths that go from each subtree to the other
            //defined respective the direction of the other node
            //dp is to either put the edge from v to par(u, v) and extend that subproblem
            //or from u to par(v,u);
            dp[i][j] = dp[j][i] = ss[i][j] * ss[j][i] + Math.max(dp(par[i][j], i), dp(par[j][i], j));
        }
        return dp[i][j];
    }

    static void dfs(int v, int p, int i) {
        ss[i][v] = 1;
        for(int edge: graph[v]) {
            if (edge == p) continue;
            par[i][edge] = v;
            dfs(edge, v,i);
            ss[i][v] += ss[i][edge];
        }
    }
    public static class FastScanner {
        BufferedReader br;
        StringTokenizer st;
        public FastScanner(Reader in) {
            br = new BufferedReader(in);
        }
        public FastScanner() {
            this(new InputStreamReader(System.in));
        }
        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        int nextInt() {
            return Integer.parseInt(next());
        }
        long nextLong() {
            return Long.parseLong(next());
        }
        double nextDouble() {
            return Double.parseDouble(next());
        }
        String readNextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
}
