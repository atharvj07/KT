#include <bits/stdc++.h>
using namespace std;
const int N = 1009;
int n, l, r, identity = 1, A[N], R[N];
char G[N][N];
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &A[i]), identity &= (A[i] == i);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) G[i][j] = '.';
  if (identity) {
    printf("%d\n", n);
    for (int i = 1; i <= n; i++) printf("%s\n", G[i] + 1);
    return (0);
  }
  A[1] = 0;
  for (int i = n; i; i--) {
    r = 0;
    l = n + 1;
    for (int j = 1; j <= n; j++)
      if (A[j] != j) r = max(r, j), l = min(l, j);
    if (l == n + 1 || l == r) break;
    if (!A[l]) {
      G[i][l] = G[i][r] = G[i][A[r]] = '\\';
      A[l] = A[A[r]];
      A[A[r]] = A[r];
      A[r] = 0;
    } else {
      G[i][l] = G[i][r] = G[i][A[l]] = '/';
      A[r] = A[A[l]];
      A[A[l]] = A[l];
      A[l] = 0;
    }
  }
  printf("%d\n", n - 1);
  for (int i = 1; i <= n; i++) printf("%s\n", G[i] + 1);
  return (0);
}
