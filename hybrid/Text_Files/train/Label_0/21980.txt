#include <iostream>
#include <algorithm>
using namespace std;

int main() {
	int n;
	cin >> n;
	long long a[n];
	for(int i=0;i<n;i++){
		cin >> a[i];
	}
	sort(a, a+n, greater<int>());
	long long sum = 0;
	int k = 0;
	for(int i=0;i<n-1;i++){
		sum += a[k];
		if(i%2==0) k++;
	}
	cout << sum << endl;
	return 0;
}
