#include <bits/stdc++.h>
#pragma GCC optimize("O2")
using namespace std;
const double PI = atan(1.0) * 4;
const int di[4] = {-1, 0, 1, 0};
const int dj[4] = {0, -1, 0, 1};
const long long INF = (long long)2e18 + 50;
const int maximum = numeric_limits<int>::max();
const int minimum = numeric_limits<int>::min();
const int mod = 1e9 + 7;
long long power(long long a, long long e) {
  long long res = 1LL;
  while (e > 0) {
    if (e % 2 == 1) res = res * a;
    a = a * a;
    e /= 2;
  }
  return res;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  t = 1;
  while (t--) {
    int n;
    cin >> n;
    int m = n - 1;
    vector<vector<int>> adj(n);
    while (m--) {
      int u, v;
      cin >> u >> v;
      --u;
      --v;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    vector<int> p(n);
    std::function<pair<int, int>(int, int, int)> dfs = [&](int u, int par,
                                                           int dist) {
      p[u] = par;
      pair<int, int> res = make_pair(dist, u);
      for (auto v : adj[u]) {
        if (v == par) continue;
        res = max(res, dfs(v, u, dist + 1));
      }
      return res;
    };
    pair<int, int> da = dfs(0, -1, 0);
    pair<int, int> db = dfs(da.second, -1, 0);
    vector<int> diam;
    int v = db.second;
    while (v != da.second) {
      diam.push_back(v);
      v = p[v];
    }
    diam.push_back(da.second);
    if (int(diam.size()) == n) {
      cout << n - 1 << "\n";
      cout << diam[0] + 1 << " " << diam[1] + 1 << " " << diam.back() + 1
           << "\n";
    } else {
      queue<int> q;
      vector<int> d(n, -1);
      for (auto v : diam) {
        d[v] = 0;
        q.push(v);
      }
      while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
          if (d[v] == -1) {
            d[v] = d[u] + 1;
            q.push(v);
          }
        }
      }
      pair<int, int> mx = make_pair(d[0], 0);
      for (int i = 1; i < n; i++) {
        mx = max(mx, make_pair(d[i], i));
      }
      cout << diam.size() - 1 + mx.first << "\n";
      cout << diam[0] + 1 << " " << mx.second + 1 << " " << diam.back() + 1
           << "\n";
    }
  }
  return 0;
}
