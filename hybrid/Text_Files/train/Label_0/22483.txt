#include <bits/stdc++.h>
using namespace std;
namespace IO {
char ibuf[(1 << 21) + 1], *iS, *iT;
char Get() {
  return (iS == iT ? (iT = (iS = ibuf) + fread(ibuf, 1, (1 << 21) + 1, stdin),
                      (iS == iT ? EOF : *iS++))
                   : *iS++);
}
int read() {
  int x = 0, c = Get();
  while (!isdigit(c)) c = Get();
  while (isdigit(c)) x = x * 10 + c - 48, c = Get();
  return x;
}
long long readl() {
  long long x = 0;
  int c = Get();
  while (!isdigit(c)) c = Get();
  while (isdigit(c)) x = x * 10 + c - 48, c = Get();
  return x;
}
}  // namespace IO
using namespace IO;
int inc(int a, int b) {
  a += b;
  return a >= 1000000 ? a - 1000000 : a;
}
long long mul(int a, int b) { return 1ll * a * b; }
struct node {
  int x, y;
  node(int a = 0, int b = 0) { x = a, y = b; }
};
int operator<(node a, node b) { return a.x < b.x || (a.x == b.x && a.y < b.y); }
int on(node l, node r, node p) {
  return mul(r.x - l.x, p.y) >= mul(r.x - l.x, l.y) + mul(r.y - l.y, p.x - l.x);
}
set<node> s;
long long m;
void ins(int dps, int cost) {
  auto it = s.lower_bound(node(dps, cost)), las = prev(it);
  if (it != s.end() && cost >= it->y) return;
  if (it != s.end() && on(*las, *it, node(dps, cost))) return;
  it = (s.emplace(dps, cost)).first;
  while (next(it) != s.end() && next(it)->x == dps) s.erase(next(it));
  while (prev(it) != s.begin() && on(*prev(prev(it)), *it, *prev(it)))
    s.erase(prev(it));
}
int bat(int lim, int hit) {
  auto it = s.lower_bound(node((hit + lim - 1) / lim, 0)), las = prev(it);
  if (it == s.end()) return 0;
  if (mul(it->y, lim) <= m) return 1;
  return mul(mul(lim, it->x - las->x), las->y) +
             mul(hit - lim * las->x, it->y - las->y) <=
         mul(m, it->x - las->x);
}
int main() {
  int n, i, o, k = 0, x, a, b;
  n = read(), m = readl(), o = 1, s.emplace(0, 0);
  for (i = 1; i <= n; ++i) {
    x = read(), a = read(), b = read();
    if (o) a = inc(a, k) + 1, b = inc(b, k) + 1;
    if (x == 1)
      ins(a, b);
    else if (bat(a, b))
      k = i, puts("YES");
    else
      puts("NO");
  }
}
