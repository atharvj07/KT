#include <bits/stdc++.h>
using namespace std;
int main(void){
	int N;
	cin >> N;
	int T[N],A[N];
	long long Takahashi,Aoki;
	long long n;
	for(int i=0;i<N;i++)cin >> T[i] >> A[i];
	Takahashi = T[0];
	Aoki = A[0];
	for(int i=1;i<N;i++){
		n = std::max((Takahashi+T[i]-1)/T[i],(Aoki+A[i]-1)/A[i]);
		Takahashi = n*T[i];
		Aoki = n*A[i];
	}
	cout << Takahashi+Aoki;
	return 0;
}
