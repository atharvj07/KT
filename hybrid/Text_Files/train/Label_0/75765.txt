import math
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	if n <= k:
		print(1)
	else:
		maxi = 1
		for i in range(1, int(math.sqrt(n)) + 1):
			if n % i == 0:
				if i <= k:
					if maxi < i:
						maxi = i
				if i != n // i:
					if n // i <= k:
						if maxi < n // i:
							maxi = n // i
		print(n // maxi)
