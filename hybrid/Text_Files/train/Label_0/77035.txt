s = '1*' + input() + '*1'
a = ''
n = len(s)
s = s + '+'
i = 0
while i < n:
	if s[i + 1] == '*':
		while s[i + 1] == '*':
			a += s[i:i + 2]
			i += 2
		a += s[i:i + 2]
		i += 2
	else:
		sum = 0
		while i < n and s[i + 1] == '+':
			sum += int(s[i])
			i += 2
		a += str(sum) + '+'
a = a[0:-1]
ans = 0
for j in range(len(a)):
	for i in range(j):
		if (a[i] == '*' or a[i] == '+') and (a[j] == '*' or a[j] == '+'):
			ans = max(eval(a[:i + 1] + '(' + a[i + 1:j] + ')' + a[j:]), ans)
print(ans)
