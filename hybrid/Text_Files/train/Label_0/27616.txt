#include<iostream>
#include<algorithm>
#include<vector>
#include<queue>
#include<unordered_map>
#define lol(i,n) for(int i=0;i<n;i++)
#define mod 1000000007
typedef long long ll;

using namespace std;
#define H 510
struct Ver{
    ll y,x,t;
    Ver(ll y,ll x,ll t){
	this->y=y,this->x=x,this->t=t;
    }
    bool operator<(const Ver&key)const{return 0;}
    bool operator>(const Ver&key)const{return 0;}
    bool operator==(const Ver&key)const{
	return this->y==key.y&&this->x==key.x&&this->t==key.t;
    }
    bool operator!=(const Ver&key)const{
	return !(this->operator==(key));
    }
    struct Hash{
	ll operator()(const Ver&key)const{
	    return key.t*H*H+key.y*H+key.x;
	}
    };
};
#define N 100010
ll h,w,a,b,c,n,s[N],t[N];
int dx[]={0,1,0,-1},dy[]={1,0,-1,0};
ll dis[H][H];
unordered_map<Ver,ll,Ver::Hash> d;
void Dijkstra(Ver s){
    typedef pair<ll,Ver> P;
    priority_queue<P,vector<P>,greater<P> >Q;
    Q.push(make_pair(0,s));
    while(!Q.empty()){
	ll cost=Q.top().first;
	Ver key=Q.top().second;
	Q.pop();
	ll i=key.y,j=key.x,t=key.t;
	if(i<0||i==h||j<0||j==w)continue;
	if(d.find(key)!=d.end())continue;
	d[key]=cost;
	lol(r,4){
	    int xx=j+dx[r],yy=i+dy[r];
	    if(t==4){
		Q.push(make_pair(cost+b,Ver(i,j,r)));
		Q.push(make_pair(cost+c,Ver(yy,xx,4)));
	    }
	    if(t==r){
		Q.push(make_pair(cost+dis[i][j]*c,Ver(i,j,4)));
		Q.push(make_pair(cost+a,Ver(yy,xx,r)));
	    }
	}
    }
}
void BFS(){
    lol(i,h)lol(j,w)dis[i][j]=mod;
    queue<Ver> Q;
    lol(i,n)Q.push(Ver(s[i],t[i],0));
    while(!Q.empty()){
	Ver key=Q.front();
	Q.pop();
	ll y=key.y,x=key.x,t=key.t;
	if(dis[y][x]<=t)continue;
	dis[y][x]=t;
	lol(r,4){
	    int xx=x+dx[r],yy=y+dy[r];
	    if(xx<0||xx==w||yy<0||yy==h)continue;
	    Q.push(Ver(yy,xx,t+1));
	}
    }
}
int main(){
    cin>>h>>w>>a>>b>>c>>n;h++,w++;
    lol(i,n)scanf("%lld%lld",&s[i],&t[i]);BFS();
    Dijkstra(Ver(s[0],t[0],4));
    cout<<d[Ver(s[n-1],t[n-1],4)]<<endl;
    return 0;
}
