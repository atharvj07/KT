    if(y == 0) return 1;
    ll res = modPow(x, y / 2, n) % n;
    return res * res * ((ll)pow(x, y % 2) % n);

int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    ll t; cin >> t;
    while(t--) {
        ll n; cin >> n;
        ll a[n], b[n];
        ll sm = 0;
        rep(i, 0, n) {
            cin >> a[i];
            sm += a[i];
        }
        rep(i, 0, n) {
            cin >> b[i];
            sm -= b[i];
        }
        if(!sm) {
            vector<pair<ll, ll>> vp;
            rep(i, 0, n) {
                if(a[i] == b[i]) continue;
                rep(j, 0, n) {
                    if(i == j || a[j] == b[j]) continue;
                    if(a[i] > b[i] || a[j] < b[j]) continue;
                    if(a[i] < b[i]) {
                        while(a[i] < b[i] && a[j] > b[j]) {
                            if(a[j] == 0) break;
                            a[i]++;
                            a[j]--;
                            vp.pb(make_pair(j + 1, i + 1));
                        }
                    }
                }
            }
            cout << sz(vp) << endl;
            for(auto i : vp) {
                cout << i.first << " " << i.second << endl;
            }
        } else {
            cout << -1 << endl;
        }
    }
}
