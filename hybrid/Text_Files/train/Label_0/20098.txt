#define repr(i, n) for (int i = (n); i >= 0; --i)

#define equals(a, b) (fabs((a) - (b)) < EPS)
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
typedef long double ld;
const ll mod = 1000000007;
//const ll mod = 998244353;
const int inf = 1e9 + 10;
const ll INF = 1e18;
const ld EPS = 1e-10;
const int dx[8] = {1, 0, -1, 0, 1, -1, -1, 1};
const int dy[8] = {0, 1, 0, -1, 1, 1, -1, -1};
template<class T> bool chmax(T &a, const T &b) { if (a<b) { a=b; return 1; } return 0; }
template<class T> bool chmin(T &a, const T &b) { if (a>b) { a=b; return 1; } return 0; }
typedef complex<ll> P;

int n;
P s, t;
P p[1005];
vector<P> G[10005];

void init() {
  rep(i, 10005) G[i].clear();
}

void solve() {
  vector<P> a, b;
  vector<ll> ar, br;
  rep(i, 10005) {
    for (P p: G[i]) {
      if (norm(p - s) < i * i) {
        a.push_back(p);
        ar.push_back(i);
      }
      else {
        b.push_back(p);
        br.push_back(i);
      }
    }
  }
  int ans = 0;
  vector<int> dpa(a.size(), 1), dpb(b.size(), 1);
  rep(i, a.size()) {
    rep(j, i) {
      if (norm(a[i] - a[j]) < (ar[i] - ar[j]) * (ar[i] - ar[j])) {
        chmax(dpa[i], dpa[j] + 1);
      }
    }
    chmax(ans, dpa[i]);
  }
  rep(i, b.size()) {
    rep(j, i) {
      if (norm(b[i] - b[j]) < (br[i] - br[j]) * (br[i] - br[j])) {
        chmax(dpb[i], dpb[j] + 1);
      }
    }
    chmax(ans, dpb[i]);
  }
  rep(i, a.size()) {
    rep(j, b.size()) {
      if (norm(a[i] - b[j]) > (ar[i] + br[j]) * (ar[i] + br[j])) {
        chmax(ans, dpa[i] + dpb[j]);
      }
    }
  }
  cout << ans << '\n';
}

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout << fixed << setprecision(25);

  while (cin >> n, n) {
    init();

    ll sx, sy, tx, ty;
    cin >> sx >> sy >> tx >> ty;
    s = P(sx, sy), t = P(tx, ty);
    rep(i, n) {
      ll px, py, r;
      cin >> px >> py >> r;
      P p = P(px, py);
      if (norm(p - t) < r * r && norm(p - s) < r * r) continue;
      if (norm(p - t) > r * r && norm(p - s) > r * r) continue;
      G[r].push_back(p);
    }

    solve();
  }

  




  
  return 0;
}
