const ll INF = 1e9+7;


template <typename T>
class RMQ {
	T query(int l, int r, int a, int b, int k){
		if( l <= a && b <= r ){
			return seg[k];
		}else if( l < b && a < r ){
			int m = (b+a)/2;
			T vl = query(l,r, a, m, k*2);
			T vr = query(l,r, m, b, k*2+1);
			return min(vl, vr);	// merge
		}
		return INF;	// default value
	}
public:
	vector<T> seg;	// セグメントツリー (親:k/2  子:k*2,k*2+1)
	int N;
	RMQ(int size){
		N = 1;
		while( N < size ) N<<=1;
		seg.assign(N*2, INF);		// default value
	}
	void update(int pos, T v){
		for(int k=N+pos; k>0; k/=2){
			seg[k] = v;
			v = min(seg[k], seg[k^1]);	// merge
		}
	}
	// [l,r)
	T get(int l, int r){
		return query(l, r, 0, N, 1);
	}
	T get(int pos){		// =get(pos, pos+1)
		return seg[N+pos];
	}
	void add(int pos, T val){
		update(pos, get(pos) + val);
	}
	void chmin(int pos, T val){
		if(get(pos) > val) update(pos, val);
	}
};


int cnt[2][500005];
int check[500005];
int main(){
	ios_base::sync_with_stdio(false);
	int N, K;
	string S, T;

	cin >> N >> K;
	cin >> S >> T;
	S += "#";
	T += "#";

	rrep(i,N) cnt[0][i] = cnt[0][i+1] + (T[i] == 'B' && T[i+1] != 'B');
	rrep(i,N) cnt[1][i] = cnt[1][i+1] + (T[i] == 'W' && T[i+1] != 'W');
	rrep(i,N) check[i] = (T[i] == 'B');

	vector<RMQ<int>> dp(2, N+2);
	rep(k,2) rep(i,N+1) dp[k].update(i, cnt[k][i]);
	rep(i,N){
		int nex = INF;
		if( S[i] == T[i] ) chmin(nex, dp[0].get(i) - cnt[0][i]);
		rep(k,2) chmin(nex, dp[k].get(max(0,i-K+1), i+1) - cnt[k][i] + (check[i]^k) + 1);
		rep(k,2) dp[k].add(i+1, nex);
	}

	int ans = INF;
	rep(k,2) chmin(ans, dp[k].get(N));

	cout << ans << endl;

	return 0;
}
