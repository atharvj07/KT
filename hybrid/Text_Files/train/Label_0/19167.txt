#include <bits/stdc++.h>
using namespace std;
int n, mn, s, r;
long long ans = 0;
unordered_map<int, int> m;
vector<int> arr(510);
void solve(int c) {
  if (c == 0) {
    return;
  }
  if (m.find(c) != m.end()) {
    return;
  }
  m[c] = 1;
  long long cnt = 0;
  for (int i = 1; i <= n; ++i) {
    s = arr[i] / c;
    r = arr[i] % c;
    if (r == 0) {
      cnt += s;
    } else if (s >= c - 1 - r) {
      cnt += s + 1;
    } else {
      return;
    }
  }
  ans = min(ans, cnt);
}
int main(void) {
  ios ::sync_with_stdio(0);
  cin.tie(0);
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> arr[i];
    ans += arr[i];
  }
  mn = *(min_element(arr.begin() + 1, arr.begin() + n + 1));
  for (int c = 1; c * c <= mn; ++c) {
    solve(c);
    solve(mn / c);
    solve(mn / c + 1);
    solve(mn / c - 1);
  }
  cout << ans << "\n";
  return 0;
}
