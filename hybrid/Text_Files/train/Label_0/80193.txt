n = int(input())
A = list(map(int, input().split()))
used = [False] * (n + 1)
for a in A:
	if a <= n:
		used[a] = True
unused = [i for (i, u) in enumerate(used[1:], 1) if not u]
result = []
for a in A:
	if a > n or used[a] == 2:
		result.append(unused.pop())
	else:
		used[a] += 1
		result.append(a)
print(*result)
