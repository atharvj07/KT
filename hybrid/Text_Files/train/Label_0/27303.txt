#include<cstdio>
#include<cctype>
#include<algorithm>
using namespace std;
const int N=200010;
const double pi=3.14159265358979;
int n,X1,Y1,X2,Y2,tot,tp,flgx=1,flgy=1,tmp[N];
pair<int,int>p[N];
char buffer[1000010],*hd,*tl;
char Getchar(){
    if(hd==tl){
        int len=fread(buffer,1,1000000,stdin);
        hd=buffer,tl=hd+len;
        if(hd==tl)
            return EOF;
    }
    return *hd++;
}
int rd(){
    int x=0;
    char c;
    do c=Getchar();
    while(!isdigit(c));
    do{
        x=(x<<1)+(x<<3)+(c^48);
        c=Getchar();
    }while(isdigit(c));
    return x;
}
void ReadData(){
	X1=rd(),Y1=rd(),X2=rd(),Y2=rd(),n=rd();
	if(X1>X2){
		flgx=-1;
		X1=-X1;
		X2=-X2;
	}
	if(Y1>Y2){
		flgy=-1;
		Y1=-Y1;
		Y2=-Y2;
	}
	for(int i=1;i<=n;i++){
		int x=rd(),y=rd();
		x*=flgx;
		y*=flgy;
		if(X1<=x&&x<=X2&&Y1<=y&&y<=Y2)
			p[++tot]=make_pair(x,y);
	}
	return;
}
void LIS(){
	sort(p+1,p+tot+1);
	for(int i=1;i<=tot;i++){
		if(!tp||tmp[tp]<p[i].second)
			tmp[++tp]=p[i].second;
		else tmp[lower_bound(tmp+1,tmp+tp+1,p[i].second)-tmp]=p[i].second;
	}
	return;
}
double getans(){
	double ret=100.0*(X2-X1+Y2-Y1)+tp*(5.0*pi-20.0);
	if(tp==min(X2-X1+1,Y2-Y1+1))
		ret+=5.0*pi;
	return ret;
}
int main(){
	ReadData();
	LIS();
	printf("%.12lf",getans());
	return 0;
}
