vector<int> sol;
int mp[N];
int head[N], cnt = 0;
int to[N << 1], nxt[N << 1];
int num[N], inum[N];
int val[N];
int dfn[N], tot = 0;
int ed[N];

void add(int u, int v){
    to[cnt] = v, nxt[cnt] = head[u], head[u] = cnt ++;
    to[cnt] = u, nxt[cnt] = head[v], head[v] = cnt ++;
}

void dfs(int u, int pre){
    dfn[u] = ++ tot;
    int tnum = num[mp[u]] ++;
    for(int i = head[u]; i != -1; i = nxt[i]){
        int v = to[i];
        if(v == pre) continue;
        int nownum = num[mp[u]];
        dfs(v, u);
        if(num[mp[u]] > nownum){
            val[1] ++;
            val[dfn[v]] --;
            val[ed[v] + 1] ++;
        }
    }
    ed[u] = tot;
    if(num[mp[u]] - tnum != inum[mp[u]]){
        val[dfn[u]] ++;
        val[ed[u] + 1] --;
    }
}


int main()
{
    // freopen("in.txt","r",stdin);
    // freopen("out.txt","w",stdout);
    int n; scanf("%d",&n);
    for(int i = 1; i <= n; ++ i){
        head[i] = -1;
        scanf("%d",&mp[i]);
        sol.push_back(mp[i]);
    }
    sort(sol.begin(), sol.end());
    sol.erase(unique(sol.begin(), sol.end()), sol.end());
    for(int i = 1; i <= n; ++ i){
        mp[i] = lower_bound(sol.begin(), sol.end(), mp[i]) - sol.begin();
        inum[mp[i]] ++;
    }
    for(int i = 1; i < n; ++ i){
        int x, y; scanf("%d%d",&x,&y);
        add(x, y);
    }
    dfs(1, -1);
    int res = 0;
    for(int i = 1; i <= n; ++ i){
        val[i] += val[i - 1];
        if(val[i] == 0) res ++;
    }
    printf("%d\n",res);
    return 0;
}
