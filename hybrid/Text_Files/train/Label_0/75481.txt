s = input()
otc = {'[': ']', '{': '}', '(': ')', '<': '>'}
cto = {'>': '<', '}': '{', ')': '(', ']': '['}
stack = []
res1 = 0
for c in s:
	if c in otc.keys():
		stack.append(c)
		continue
	if len(stack) == 0:
		res1 = -1
		break
	o = stack[len(stack) - 1]
	stack.pop()
	if c == otc[o]:
		continue
	else:
		res1 += 1
if res1 == -1 or len(stack) != 0:
	print('Impossible')
else:
	print(res1)
