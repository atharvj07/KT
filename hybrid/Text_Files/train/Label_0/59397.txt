
import java.util.*;
public class root
{
    public static void main(String args[])
    {
        Scanner ex=new Scanner(System.in);
        int n=ex.nextInt();
        long curr=1,next=2,pro=curr*next,num=2,disp;
        while(n-->0)
        {
            disp=pro/curr;
            disp=disp*pro-num/curr;
            System.out.println(disp);
            num=pro;
            curr++;
            next++;
            pro=curr*next;
        }
    }
}
