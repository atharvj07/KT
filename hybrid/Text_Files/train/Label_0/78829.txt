(n, m) = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
big = 10 ** 25
small = -10 ** 25
first = []
for x in a:
	s = small
	for y in b:
		_ = x * y
		if _ > s:
			s = _
	first.append(s)
ans = big
for i in range(len(a)):
	s = small
	for j in range(len(a)):
		if j != i:
			if first[j] > s:
				s = first[j]
	if s < ans:
		ans = s
print(ans)
