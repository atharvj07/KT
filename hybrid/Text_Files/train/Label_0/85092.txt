import sys

def findMaxLength(s):
	nums = [int(i) for i in s]
	dic = {}
	dic[0] = -1
	flag = 0
	maxL = 0
	for i in range(len(nums)):
		flag += 1 if nums[i] == 1 else -1
		if flag not in dic:
			dic[flag] = i
		maxL = i - dic[flag] if i - dic[flag] > maxL else maxL
	return maxL
count = 0
for line in sys.stdin:
	if count == 0:
		num = int(line.strip())
		count += 1
		continue
	s = line.strip()
print(findMaxLength(s))
