n = int(input())
commands = list(map(int, input().split()))
use_cupon = [0] * n
for i in range(n):
	day = commands[i]
	need_to_buy = day - use_cupon[i]
	if need_to_buy < 0:
		print('NO')
		exit()
	if need_to_buy >= 2:
		need_to_buy %= 2
	if i < n - 1:
		use_cupon[i + 1] += need_to_buy
	elif need_to_buy:
		print('NO')
		exit()
print('YES')
