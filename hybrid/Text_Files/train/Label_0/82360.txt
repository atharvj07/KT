t = int(input())
for i in range(t):
	s = list(input())
	m = []
	i = 0
	while i < len(s):
		if s[i] == '1':
			j = i
			c = 0
			while j < len(s):
				if s[j] == '0':
					break
				c = c + 1
				j = j + 1
			m.append(c)
			i = j
		else:
			i = i + 1
	m.sort(reverse=True)
	if len(m) == 0:
		print(0)
	else:
		s = 0
		for j in range(len(m)):
			if j % 2 == 0:
				s = s + m[j]
		print(s)
