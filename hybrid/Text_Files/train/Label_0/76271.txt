(a, l) = map(float, input().split())
(b, c, d) = (2 * a, 3 * a, 4 * a)
(s, l) = (0, l % d)
u = str(a) + ' '
v = ' ' + str(a)

def f():
	global s
	s = (s + l) % d
	if s < a:
		return str(s) + ' 0'
	if s < b:
		return u + str(s - a)
	if s < c:
		return str(c - s) + v
	return '0 ' + str(d - s)
print('\n'.join((f() for i in range(int(input())))))
