#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const int maxn = 200002;
const int N = 12000000;
int ls[N], rs[N], n, m;
int rt[maxn * 2], cnt;
int insert(int x, int L, int R, int pos) {
  int now = ++cnt;
  ls[now] = ls[x];
  rs[now] = rs[x];
  if (L == R) return now;
  int mid = (L + R) >> 1;
  if (mid >= pos)
    ls[now] = insert(ls[x], L, mid, pos);
  else
    rs[now] = insert(rs[x], mid + 1, R, pos);
  return now;
}
int merge(int x, int y) {
  if (!x || !y) return x + y;
  int now = ++cnt;
  ls[now] = merge(ls[x], ls[y]);
  rs[now] = merge(rs[x], rs[y]);
  return now;
}
int ask(int x, int L, int R, int l, int r) {
  if (l > r) return 0;
  if (!x) return 0;
  if (l <= L && R <= r) return 1;
  int mid = (L + R) >> 1;
  if (mid >= r) return ask(ls[x], L, mid, l, r);
  if (mid < l) return ask(rs[x], mid + 1, R, l, r);
  if (ask(ls[x], L, mid, l, mid)) return 1;
  return ask(rs[x], mid + 1, R, mid + 1, r);
}
int root, last, siz;
int fa[maxn * 2], nxt[maxn * 2][26], len[maxn * 2], pp[maxn * 2], id[maxn * 2];
int newnode(int _len) {
  fa[++siz] = 0;
  len[siz] = pp[siz] = _len;
  id[_len] = siz;
  memset(nxt[siz], 0, sizeof nxt[siz]);
  return siz;
}
void SAM_init() {
  siz = 0;
  last = root = newnode(0);
}
void SAM_add(int x, int _len) {
  int p = last, np = newnode(_len);
  last = np;
  for (; p && !nxt[p][x]; p = fa[p]) nxt[p][x] = np;
  if (!p)
    fa[np] = root;
  else {
    int q = nxt[p][x];
    if (len[q] == len[p] + 1)
      fa[np] = q;
    else {
      int nq = ++siz;
      fa[nq] = fa[q];
      pp[nq] = pp[q];
      memcpy(nxt[nq], nxt[q], sizeof nxt[nq]);
      len[nq] = len[p] + 1;
      fa[q] = fa[np] = nq;
      for (; p && nxt[p][x] == q; p = fa[p]) nxt[p][x] = nq;
    }
  }
}
int v[maxn * 2], toposam[maxn * 2];
void SAM_toposort() {
  for (int i = 1; i <= siz; i++) v[len[i]]++;
  for (int i = 1; i <= len[last]; i++) v[i] += v[i - 1];
  for (int i = 1; i <= siz; i++) toposam[v[len[i]]--] = i;
}
void SAM_build(char *s) {
  SAM_init();
  for (int i = 0; s[i]; i++) SAM_add(s[i] - 'a', i + 1);
}
char s[maxn];
int dp[maxn * 2];
int who[maxn * 2];
int main() {
  scanf("%d%s", &n, s);
  SAM_build(s);
  SAM_toposort();
  for (int i = siz; i >= 2; i--) {
    int p = toposam[i];
    rt[p] = insert(rt[p], 1, n, pp[p]);
    if (fa[p] != root) rt[fa[p]] = merge(rt[fa[p]], rt[p]);
  }
  dp[1] = 0;
  who[1] = 1;
  int ans = 0;
  for (int i = 2; i <= siz; i++) {
    int p = toposam[i];
    int q = who[fa[p]];
    if (q == 1 || ask(rt[q], 1, n, pp[p] - len[p] + len[q], pp[p] - 1)) {
      who[p] = p;
      dp[p] = dp[q] + 1;
    } else
      who[p] = q;
    ans = max(ans, dp[p]);
  }
  printf("%d\n", ans);
  return 0;
}
