#include <bits/stdc++.h>
#pragma GCC diagnostic ignored "-Wunused-variable"
const long long int MAX = (long long int)(1e18 + 1);
const long long int MIN = (long long int)(-1e18 - 1);
const long long int mod = (long long int)(1e9 + 7);
using namespace std;
long long int power(long long int a, long long int n) {
  long long int p = 1;
  while (n > 0) {
    if (n % 2) {
      p = p * a;
    }
    n >>= 1;
    a *= a;
  }
  return p;
}
long long int power_mod(long long int a, long long int n, long long int mod_) {
  long long int p = 1;
  while (n) {
    if (n % 2) {
      p = (p * a) % mod_;
    }
    n /= 2;
    a = (a * a) % mod_;
  }
  return p % mod_;
}
set<long long int> s1, s2;
void dfs(long long int x, std::vector<vector<long long int> >& graph,
         vector<long long int>& color) {
  color[x] = 1;
  for (auto y : graph[x]) {
    if (color[y] == 0) {
      s1.insert(y);
      dfs(y, graph, color);
    }
  }
}
void dfs1(long long int x, std::vector<vector<long long int> >& graph,
          vector<long long int>& color) {
  color[x] = 1;
  for (auto y : graph[x]) {
    if (color[y] == 0) {
      if (s1.find(y) == s1.end())
        s2.insert(y);
      else
        s1.erase(y);
      dfs1(y, graph, color);
    }
  }
}

    cin >> n >> m >> x >> y;
    x--;
    y--;
    std::vector<vector<long long int> > graph(n);
    for (i = 0; i < m; i++) {
      cin >> j >> k;
      j--;
      k--;
      graph[j].push_back(k);
      graph[k].push_back(j);
    }
    s1.clear();
    s2.clear();
    vector<long long int> color(n);
    for (i = 0; i < n; i++) color[i] = 0;
    color[y] = 1;
    dfs(x, graph, color);
    for (i = 0; i < n; i++) color[i] = 0;
    color[x] = 1;
    dfs1(y, graph, color);
    j = (long long int)s1.size();
    k = (long long int)(s2.size());
    cnt = j * k;

  }
  return 0;
}
