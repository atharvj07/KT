s = dict(zip(['a', 'b', 'c', 'ab', 'bc', 'ac', 'abc'], [float('inf')] * 7))
for _ in range(int(input())):
	(d1, d2) = input().split()
	d1 = int(d1)
	d2 = ''.join(sorted(d2.lower()))
	s[d2] = min(s[d2], d1)
s['ab'] = min([s['a'] + s['b'], s['ab']])
s['ac'] = min([s['a'] + s['c'], s['ac']])
s['bc'] = min([s['b'] + s['c'], s['bc']])
r = min([s['abc'], s['ab'] + s['c'], s['ac'] + s['b'], s['bc'] + s['a'], s['ab'] + s['ac'], s['ab'] + s['bc'], s['ac'] + s['bc']])
print(-1 if r == float('inf') else r)
