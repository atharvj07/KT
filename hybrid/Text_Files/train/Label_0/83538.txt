(n, m) = map(int, input().split())
lin0 = [0] * n
lin1 = [0] * n
col0 = [0] * m
col1 = [0] * m
t = 0
mat = []
for i in range(n):
	str = input().split()
	linha = []
	for j in str:
		linha.append(int(j))
	mat.append(linha)
for i in range(n):
	for j in range(m):
		if mat[i][j] == 0:
			lin0[i] += 1
			col0[j] += 1
		else:
			lin1[i] += 1
			col1[j] += 1
for i in range(n):
	t += 2 ** lin0[i] - 1 + 2 ** lin1[i] - 1
for i in range(m):
	t += 2 ** col0[i] - 1 + 2 ** col1[i] - 1
print(t - m * n)
