#include <bits/stdc++.h>
using namespace std;
 
#define IOS ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define endl "\n"
#define int long long

const int N = 1e5 + 5;

int n, k;
int deg[N], ans[N];
vector<pair<int, int> > g[N];

void dfs(int u, int par, int c)
{
	for(auto &it:g[u])
	{
		if(it.first == par)
			continue;
		c = (c + 1) % k;
		ans[it.second] = c;
		dfs(it.first, u, c);
	}
}

int32_t main()
{
	IOS;
	cin >> n;
	for(int i = 1; i <= n - 1; i++)
	{
		int u, v;
		cin >> u >> v;
		deg[u]++, deg[v]++;
		g[u].push_back({v, i});
		g[v].push_back({u, i});
	}
	for(int i = 1; i <= n; i++)
		k = max(k, deg[i]);
	dfs(1, 1, 0);
	cout << k << endl;
	for(int i = 1; i <= n - 1; i++)
		cout << ans[i] + 1 << endl;
	return 0;	
}
