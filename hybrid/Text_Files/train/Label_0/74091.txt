from sys import stdin
t = int(stdin.readline())
for i in range(t):
	(n, H) = [int(j) for j in stdin.readline().split()]
	A = [int(j) for j in stdin.readline().split()]
	A.sort()
	max1 = A[-1]
	max2 = A[-2]
	f = True
	min = H // (max1 + max2) * 2
	H %= max1 + max2
	while H > 0:
		if f:
			H -= max1
			f = not f
		else:
			H -= max2
			f = not f
		min += 1
	print(min)
