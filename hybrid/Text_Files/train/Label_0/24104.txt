int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int t = 1;
  cin >> t;
  chandan2();
  while (t--) {
    long long int n;
    cin >> n;
    long long int f = 0;
    set<pair<char, char>> st;
    string a, b;
    cin >> a >> b;
    for (long long int i = 0; i < int(a.size()); i++) {
      if (a[i] > b[i]) {
        f = 1;
        break;
      } else {
        if (a[i] != b[i]) {
          st.insert(make_pair(a[i], b[i]));
        }
      }
    }
    if (f)
      print(-1);
    else {
      long long int ans = 0;
      while (int(st.size()) > 0) {
        set<pair<char, char>> nst;
        pair<char, char> pr = *(st.begin());
        char cha = pr.first;
        char chb = pr.second;
        ans++;
        for (auto xt : st) {
          if (xt.first == cha) {
            if (chb != xt.second) nst.insert({chb, xt.second});
          } else
            nst.insert({xt.first, xt.second});
        }
        st = nst;
      }
      print(ans);
    }
  }
  return 0;
}
