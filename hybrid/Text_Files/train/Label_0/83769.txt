def fail():
	print(-1, -1)
	exit()
read = lambda : map(int, input().split())
(m, s) = read()
cnt = s // 9
x = s % 9
if cnt + (x > 0) > m:
	fail()
if s == 0 and m > 1:
	fail()
Max = '9' * cnt
if x:
	Max += str(x)
Max += '0' * (m - len(Max))
Min = list(Max[::-1])
if Min[0] == '0' and s:
	Min[0] = '1'
	for i in range(1, m):
		if Min[i] != '0':
			Min[i] = str(int(Min[i]) - 1)
			break
Min = ''.join(Min)
print(Min, Max)
