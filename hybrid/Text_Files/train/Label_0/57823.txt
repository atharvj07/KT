

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class C {
	static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer st;
	static PrintWriter out = new PrintWriter(System.out);
	static long[] subTreeSize;

	static void solve() throws NumberFormatException, IOException {
		int n = nextInt(), m = nextInt();
		int[][] table = new int[n][m];
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				table[i][j] = nextInt();
			}
		}
		boolean[][] transposed = new boolean[n][m];
		for (int i = 1; i < n; i++) {
			for (int j = 0; j < m; j++) {
				transposed[i][j] = table[i][j] < table[i - 1][j];
			}
		}
		int[] maxLength = new int[n];
		for (int col = 0; col < m; col++) {
			int right = 0;
			int lastLeft = 0;
			while (right < n) {
				while (right < n - 1 && !transposed[right + 1][col]) {
					right++;
				}
				for (int row = lastLeft; row <= right; row++) {
					maxLength[row] = Math.max(maxLength[row], right - row + 1);
				}
				right++;
				while (right < n && transposed[right][col]) {
					right++;
				}
				lastLeft = right;
			}
		}
		int k = nextInt();
		for (int query = 0; query < k; query++) {
			int a = nextInt() - 1, b = nextInt() - 1;
			if (a == n - 1) {
				out.println("Yes");
			} else {
				if (maxLength[a + 1] >= b - a) {
					out.println("Yes");
				} else {
					out.println("No");
				}
			}
		}
	}

	public static void main(String[] args) throws NumberFormatException, IOException {
		solve();
		out.close();
	}

	static String next() throws IOException {
		while (st == null || !st.hasMoreTokens()) {
			st = new StringTokenizer(br.readLine());
		}
		return st.nextToken();
	}

	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}

}
