import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	x = []
	for __ in range(n):
		x.append(tuple(map(int, input().split())))
	x.sort(reverse=True, key=lambda a: a[1] - a[0])
	for i in range(n):
		if x[i][0] == x[i][1]:
			print(x[i][0], x[i][0], x[i][0])
			continue
		for j in range(i + 1, n):
			if x[i][0] == x[j][0]:
				print(x[i][0], x[i][1], x[j][1] + 1)
				break
			elif x[i][1] == x[j][1]:
				print(x[i][0], x[i][1], x[j][0] - 1)
				break
