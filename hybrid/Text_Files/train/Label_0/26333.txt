#include <bits/stdc++.h>
using namespace std;
const int inf = 1 << 30;
long long gcd(long long a, long long b) {
  if (!a) return b;
  return gcd(b % a, a);
}
bool solve(int tc) {
  long long n;
  scanf("%lld", &n);
  vector<int> a(n + 1);
  for (int i = (1); i < int(n + 1); ++i) scanf("%d", &a[i]);
  sort(a.begin(), a.end());
  long long sum1 = 0, sum2 = 0;
  long long ans = 0;
  for (typeof((a).begin()) it = (a).begin(); it != (a).end(); ++it)
    sum1 += *it, ans -= *it;
  for (int i = (0); i < int(n + 1); ++i) {
    ans += sum1 + sum2;
    if (i != n) {
      sum1 += (i + 1) * (a[i + 1] - a[i]);
      sum2 -= (n - i) * (a[i + 1] - a[i]);
    }
  }
  long long g = gcd(ans, n);
  cout << ans / g << ' ' << n / g << endl;
  return false;
}
int main() {
  int n = 1 << 30;
  for (int i = 1; i <= n && solve(i); ++i)
    ;
}
