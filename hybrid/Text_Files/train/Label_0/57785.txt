import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.TreeSet;


public class ProblemC {

    public static void main(String[] args) {
        ProblemC problem = new ProblemC();
        problem.solve();
    }
    
    Map<String, Integer>[] has;

    private void solve() {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();
        String[] name = new String[n];
        for (int i = 0; i < n; i++) {
            name[i] = sc.next();
        }
        has = new HashMap[k];
        for (int i = 0; i < k; i++) {
            has[i] = new HashMap<String, Integer>();
        }
        Compound[] comps = new Compound[m];
        for (int j = 0; j < m; j++) {
            String compoundName = sc.next();
            compoundName = compoundName.substring(0, compoundName.length() - 1);
            Compound comp = new Compound();
            comp.name = compoundName;
            while (true) {
                String simpleName = sc.next();
                String value = sc.next();
                if (value.endsWith(",")) {
                    value = value.substring(0, value.length() - 1);
                    comp.add(simpleName, Integer.parseInt(value));
                } else {
                    comp.add(simpleName, Integer.parseInt(value));
                    break;
                }
            }
            comps[j] = comp;
        }
        for (int i = 0; i < q; i++) {
            int nr = sc.nextInt() - 1;
            String buyName = sc.next();
            Integer cur = has[nr].get(buyName);
            if (cur == null) {
                cur = 0;
            }
            cur++;
            has[nr].put(buyName, cur);
            for (int j = 0; j < m; j++) {
                boolean canBuy = true;
                for (int t = 0; t < comps[j].names.size(); t++) {
                    Integer my = has[nr].get(comps[j].names.get(t));
                    if (my == null) {
                        my = 0;
                    }
                    if (my < comps[j].count.get(t)) {
                        canBuy = false;
                        break;
                    }
                }
                if (canBuy) {
                    for (int t = 0; t < comps[j].names.size(); t++) {
                        Integer current = has[nr].get(comps[j].names.get(t));
                        current -= comps[j].count.get(t);
                        has[nr].put(comps[j].names.get(t), current);
                    }
                    Integer current = has[nr].get(comps[j].name);
                    if (current == null) {
                        current = 0;
                    }
                    current++;
                    has[nr].put(comps[j].name, current);
                    break;
                }
            }
        }
        for (int i = 0; i < k; i++) {
            TreeSet<String> diff = new TreeSet<String>();
            for (String nm : has[i].keySet()) {
                if (has[i].get(nm) > 0) {
                    diff.add(nm);
                }
            }
            System.out.println(diff.size());
            for (String artName : diff) {
                System.out.println(artName + " " + has[i].get(artName));
            }
        }
    }

}


class Compound {
    String name;
    List<String> names;
    List<Integer> count;
    public Compound() {
        names = new ArrayList<String>();
        count = new ArrayList<Integer>();
    }
    
    public void add(String name, int cnt) {
        names.add(name);
        count.add(cnt);
    }
}
