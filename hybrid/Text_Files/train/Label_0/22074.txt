ll gcd(ll a, ll b) {
    if (a > b) swap(a, b);
    if (a == 0) return b;
    return gcd(a, b % a);
}

int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);
    cout << fixed << setprecision(10);

    for (;;) {
        ll H, h, m, s; cin >> H >> h >> m >> s;
        if (H == 0) break;

        ll M = 119 * H - 1;

        ll x = ((h * 60 + m) * 60 + s) * M;
        ll n = (x-1) / (3600 * H) + 1;

        ll y = 3600 * H * n;
        ll hour = (y / M / 3600) % H;
        ll minute = (y / M / 60) % 60;
        ll sec = y % (60 * M);

        ll d = M;
        ll g = gcd(d, sec);
        sec /= g, d /= g;

        if (hour == 0 and minute == 0 and sec == 0) {
            ++n;
            ll y = 3600 * H * n;
            ll hour = (y / M / 3600) % H;
            ll minute = (y / M / 60) % 60;
            ll sec = y % (60 * M);

            ll d = M;
            ll g = gcd(d, sec);
            sec /= g, d /= g;

            cout << hour << " " << minute << " " << sec << " " << d << endl;
            continue;
        }

        cout << hour << " " << minute << " " << sec << " " << d << endl;
    }

    return 0;
}
