def solve():
	(n, m, k, s) = mi()
	a = li()
	b = li()
	gadgets = [[] for i in range(2)]
	for i in range(m):
		(x, y) = mi()
		gadgets[x - 1].append([y, i + 1])
	gadgets[0].sort()
	gadgets[1].sort()
	sz = [len(gadgets[0]), len(gadgets[1])]

	def check(idx):
		mnx1 = inf
		mnx2 = inf
		for i in range(idx + 1):
			mnx1 = min(mnx1, a[i])
			mnx2 = min(mnx2, b[i])
		(l1, l2, res) = (0, 0, 0)
		if k > sz[0] + sz[1]:
			return 0
		for i in range(k):
			if l1 == sz[0]:
				res += gadgets[1][l2][0] * mnx2
				l2 += 1
			elif l2 == sz[1]:
				res += gadgets[0][l1][0] * mnx1
				l1 += 1
			elif gadgets[0][l1][0] * mnx1 <= gadgets[1][l2][0] * mnx2:
				res += gadgets[0][l1][0] * mnx1
				l1 += 1
			else:
				res += gadgets[1][l2][0] * mnx2
				l2 += 1
		return res <= s
	l = 0
	r = n - 1
	ans = -1
	while l <= r:
		mid = l + r >> 1
		if check(mid):
			ans = mid + 1
			r = mid - 1
		else:
			l = mid + 1
	print(ans)
	if ans == -1:
		return
	mnx1 = inf
	mnx2 = inf
	(idx1, idx2) = (-1, -1)
	for i in range(ans):
		if a[i] < mnx1:
			mnx1 = a[i]
			idx1 = i
		if b[i] < mnx2:
			mnx2 = b[i]
			idx2 = i
	(l1, l2, res) = (0, 0, [])
	for i in range(k):
		if l1 == sz[0]:
			res.append([gadgets[1][l2][1], idx2 + 1])
			l2 += 1
		elif l2 == sz[1]:
			res.append([gadgets[0][l1][1], idx1 + 1])
			l1 += 1
		elif gadgets[0][l1][0] * mnx1 <= gadgets[1][l2][0] * mnx2:
			res.append([gadgets[0][l1][1], idx1 + 1])
			l1 += 1
		else:
			res.append([gadgets[1][l2][1], idx2 + 1])
			l2 += 1
	for i in res:
		print(*i)
if file:
