import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class d {
	static int[] haps;
	static long[][] tmp = new long[500][500];
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		long l = in.nextLong();
		haps = new int[n];
		for(int i=0;i<n;i++)
			haps[i] = in.nextInt();
		String[] ss = new String[n];
		for(int i=0;i<n;i++) {
			ss[i] = in.next();
		}
		AhoCorasick.createAhoCorasick(ss);
		int nn = AhoCorasick.nodeCount;
		long[][] mat = new long[nn][nn];
		for(long[] row : mat)
			Arrays.fill(row, -1);
		for(int i=0;i<nn;i++) {
			for(int j=0;j<26;j++) {
				int v = AhoCorasick.transition[j][i];
				mat[i][v] = AhoCorasick.vs[v];
			}
		}
		mat = matexpo(mat, l);
		long ans = 0;
		for(int i=0;i<nn;i++) {
			ans = Math.max(ans, mat[0][i]);
		}
		System.out.println(ans);
	}
	static void mult(long[][] a, long[][] b) {
		int n = a.length;
		for(int i=0;i<n;i++) {
			for(int j=0;j<n;j++) {
				tmp[i][j] = -1;
				for(int k=0;k<n;k++) {
					if(a[i][k] != -1 && b[k][j] != -1)
						tmp[i][j] = Math.max(tmp[i][j], a[i][k] + b[k][j]);
				}
			}
		}
		for(int i=0;i<n;i++) {
			for(int j=0;j<n;j++) {
				a[i][j] = tmp[i][j];
			}
		}
	}
	static long[][] matexpo(long[][] mat, long pow) {
		int n = mat.length;
		long[][] res = new long[n][n];
		for(long[] row : res)
			Arrays.fill(row, -1);
		for(int i=0;i<n;i++) {
			res[i][i] = 0;
		}
		
		while(pow > 0) {
			if(pow % 2 == 1) {
				mult(res,mat);
			}
			mult(mat,mat);
			pow /= 2;
		}
		return res;
	}
	static class AhoCorasick {
		public static final int ALPHA = 26;
		public static int nodeCount;
		public static int[][] transition;
		public static boolean[] term;
		public static int[] vs;
		public static void createAhoCorasick(String[] strs) {
			int maxNodes = 1;
			for(String s : strs) {
				maxNodes += s.length();
			}

			int[][] children = new int[ALPHA][maxNodes];
			boolean[] leaf = new boolean[maxNodes];
			nodeCount = 1;
			int ptr = 0;
			vs = new int[maxNodes];
			for(String s : strs) {
				int node = 0;
				for(char ch : s.toCharArray()) {
					int c = ch - 'a';
					if(children[c][node] == 0) {
						children[c][node] = nodeCount;
						nodeCount++;
					}
					node = children[c][node];
				}
				leaf[node] = true;
				vs[node] += haps[ptr++];
			}

			transition = new int[ALPHA][nodeCount];
			term = new boolean[nodeCount];

			ArrayDeque<Integer> queue = new ArrayDeque<Integer>();
			queue.add(0);
			queue.add(0);

			while(queue.size() > 0) {
				int node = queue.remove();
				int suffLink = queue.remove();

				term[node] = leaf[node] || term[suffLink];
				vs[node] += vs[suffLink];

				for(int ch = 0; ch < ALPHA; ch++) {
					if(children[ch][node] != 0){
						transition[ch][node] = children[ch][node];
						queue.add(children[ch][node]);
						queue.add(node == 0 ? 0 : transition[ch][suffLink]);
					} else {
						transition[ch][node] = transition[ch][suffLink];
					}
				}
			}
		}
	}
}
