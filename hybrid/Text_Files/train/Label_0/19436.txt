#include <bits/stdc++.h>
using namespace std;
int a[211111];
int n;
int main() {
  cin >> n;
  int p1 = 0, p2 = n;
  int c = 0;
  for (int i = 1; i <= n; i++) {
    a[p1] = ++c;
    a[p2] = ++c;
    int tp1 = p1, tp2 = p2;
    p1 = tp2 - 1;
    p2 = tp1 - 1;
    if (p1 < 0) p1 += n * 2;
    if (p2 < 0) p2 += n * 2;
  }
  if (n % 2 == 0) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n + n; i++) {
    cout << a[i] << ' ';
  }
  cout << endl;
  return 0;
}
