#include <bits/stdc++.h>
using namespace std;
int main() {
  int b, c, finv = 0;
  vector<int> valid;
  for (int i = int(0); i < int(10); i++) {
    cout << i << i << i << i << endl;
    cin >> b >> c;
    if (b) valid.push_back(i);
  }
  for (int i = int(0); i < int(4); i++)
    if (valid[i] == finv) finv++;
  int ans[4];
  for (int i = int(0); i < int(4); i++) {
    for (int j = int(0); j < int(4); j++) {
      for (int k = int(0); k < int(4); k++) {
        if (i == k)
          cout << valid[j];
        else
          cout << finv;
      }
      cout << endl;
      cin >> b >> c;
      if (b) ans[i] = valid[j];
    }
  }
  for (int i = int(0); i < int(4); i++) cout << ans[i];
  cout << endl;
  return 0;
}
