using namespace std;
bool sortinrev(long long x, long long y) { return x > y; }
void swap(long long* a, long long* b) {
  long long temp = *a;
  *a = *b;
  *b = temp;
}
long long div_ceil(long long a, long long b) { return (a + b - 1) / b; }
long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
long long lcm(long long a, long long b) { return (a / gcd(a, b)) * b; }
const int MOD = 1000 * 1000 * 1000 + 7;
const long long INF = 1e18L + 5;
const int nax = 2000005;
const int N = int(1e6);
void solve() {
  long long n, a, b, k;
  cin >> n >> a >> b >> k;
  vector<long long> h(n);
  for (long long i = 0; i < n; ++i) {
    cin >> h[i];
    h[i] %= (a + b);
    if (h[i] == 0) h[i] += (a + b);
    h[i] = div_ceil(h[i], a) - 1;
  }
  sort((h).begin(), (h).end());
  long long cnt = 0;
  for (long long i = 0; i < n; i++) {
    if (k - h[i] >= 0) {
      k -= h[i];
      cnt++;
    } else
      break;
  }
  cout << cnt << "\n";
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long t = 1;
  long long T = 1;
  while (t--) {
    cerr << "TEST-CASE #" << T << "\n";
    solve();
    T++;
    cerr << "\n";
  }
