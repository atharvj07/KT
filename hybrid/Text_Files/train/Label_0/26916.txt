#include <bits/stdc++.h>
using namespace std;
double p[200001];
map<double, int> table;
int n, a, b, c;
int main() {
  scanf("%d\n", &n);
  for (int i = 1; i <= n; ++i) {
    scanf("(%d+%d)/%d\n", &a, &b, &c);
    p[i] = (double)(a + b) / c;
    ++table[p[i]];
  }
  for (int i = 1; i <= n; ++i) printf("%d ", table[p[i]]);
}
