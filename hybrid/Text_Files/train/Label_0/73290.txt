from collections import defaultdict as dc
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	p = dc(int)
	m = 0
	a = 0
	b = 0
	for i in range(n):
		y = l[i] % k
		if y != 0:
			x = k - y
			p[x] += 1
	if len(p) == 0:
		print(0)
	else:
		m = 0
		for i in p.keys():
			m = max(m, (p[i] - 1) * k + i)
		print(m + 1)
