n = int(input())
c = input().split()
c = [int(i) for i in c]
arrofwords = []
for i in range(n):
	s = input()
	arrofwords.append(s)
dp = [[-1 for i in range(2)] for i in range(n)]
dp[0][0] = 0
dp[0][1] = c[0]
for i in range(1, n):
	x = dp[i - 1][0]
	y = dp[i - 1][1]
	if x == -1 and y == -1:
		dp[i][0] = -1
		dp[i][1] = -1
	elif x == -1:
		o = arrofwords[i - 1]
		if arrofwords[i] >= o[::-1]:
			dp[i][0] = dp[i - 1][1]
		else:
			dp[i][0] = -1
		to = arrofwords[i]
		if to[::-1] >= o[::-1]:
			dp[i][1] = dp[i - 1][1] + c[i]
		else:
			dp[i][1] = -1
	elif y == -1:
		o = arrofwords[i - 1]
		to = arrofwords[i]
		if arrofwords[i] >= o:
			dp[i][0] = dp[i - 1][0]
		else:
			dp[i][0] = -1
		if to[::-1] >= o:
			dp[i][1] = dp[i - 1][0] + c[i]
		else:
			dp[i][1] = -1
	else:
		o = arrofwords[i - 1]
		to = arrofwords[i]
		if arrofwords[i] >= o and arrofwords[i] >= o[::-1]:
			dp[i][0] = min(dp[i - 1][0], dp[i - 1][1])
		elif arrofwords[i] >= o:
			dp[i][0] = dp[i - 1][0]
		elif arrofwords[i] >= o[::-1]:
			dp[i][0] = dp[i - 1][1]
		else:
			dp[i][0] = -1
		if to[::-1] >= o and to[::-1] >= o[::-1]:
			dp[i][1] = min(dp[i - 1][0] + c[i], dp[i - 1][1] + c[i])
		elif to[::-1] >= o:
			dp[i][1] = dp[i - 1][0] + c[i]
		elif to[::-1] >= o[::-1]:
			dp[i][1] = dp[i - 1][1] + c[i]
		else:
			dp[i][1] = -1
if dp[n - 1][0] == -1:
	print(dp[n - 1][1])
elif dp[n - 1][1] == -1:
	print(dp[n - 1][0])
else:
	print(min(dp[n - 1][0], dp[n - 1][1]))
