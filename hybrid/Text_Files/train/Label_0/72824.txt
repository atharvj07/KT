n = int(input())
l = [int(i) for i in input().split()]
l = sorted(l)
rr = []
new = l[::-1]
io = 0
f = 0
while io < len(new) - 1:
	if new[io] - new[io + 1] <= 1:
		rr.append(new[io])
		rr.append(new[io + 1])
		io += 2
		continue
	else:
		io += 1
if len(rr) >= 4:
	q = len(rr) % 4
	if q != 0:
		rr = rr[:-q]
	i = 0
	while i < len(rr):
		f += min(rr[i], rr[i + 1]) * min(rr[i + 2], rr[i + 3])
		i += 4
print(f)
