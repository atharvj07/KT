def solve(n, m, s, t):
	idx = [0] * m
	j = 0
	for i in range(n):
		if j < m and t[j] == s[i]:
			idx[j] = i
			j += 1
	res = 0
	j = m - 1
	for i in range(n - 1, -1, -1):
		if j >= 0 and t[j] == s[i]:
			res = max(res, i - idx[j - 1])
			j -= 1
	return res
(n, m) = map(int, input().split())
s = input()
t = input()
print(solve(n, m, s, t))
