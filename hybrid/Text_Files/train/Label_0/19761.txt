#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, k;
  long long int t[40];
  memset(t, 0, sizeof t);
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    char a;
    cin >> a;
    t[a - 'A']++;
  }
  sort(t, t + 40);
  reverse(t, t + 40);
  long long int ind = 0, ans = 0;
  while (k > 0 && ind < 40) {
    long long int pick = min(k, t[ind]);
    k -= pick;
    ans += pick * pick;
    ind++;
  }
  cout << ans << endl;
  return 0;
}
