from sys import stdin
from collections import defaultdict

def main():
	(n,) = stdin_get_ints_from_line()
	a = []
	r = defaultdict(int)
	for x in stdin.readline().split(' '):
		xx = int(x)
		a.append(xx)
		r[xx] += 1
	num = a[0]
	direction = 1
	result = []
	result.append(num)
	for _ in range(n - 1):
		if direction == 1:
			if num != 3 and num % 3 == 0 and (r[num // 3] > 0):
				result.append(num // 3)
				r[num // 3] -= 1
				num = num // 3
			elif r[num * 2] > 0:
				result.append(num * 2)
				r[num * 2] -= 1
				num = num * 2
			else:
				direction = -1
				if len(result) > 0:
					num = result[0]
		if direction == -1:
			if r[num * 3] > 0:
				result.insert(0, num * 3)
				r[num * 3] -= 1
				num = num * 3
			elif num % 2 == 0 and r[num // 2] > 0:
				result.insert(0, num // 2)
				r[num // 2] -= 1
				num = num // 2
			else:
				raise RuntimeError()
	print(' '.join((str(d) for d in result)))

def stdin_get_ints_from_line():
	return (int(x) for x in stdin.readline().strip().split(' '))

def stdin_get_ints_list_from_line():
	return list((int(x) for x in stdin.readline().strip().split(' ')))

def stdin_get_string_from_line():
	return stdin.readline().strip()
main()
