import java.io.PrintWriter;
import java.util.Scanner;

/**
 * Created by omar on 11/11/17.
 */
public class Main {

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        long A = sc.nextLong(), B = sc.nextLong(), n = sc.nextLong(), x = sc.nextLong();
        long[][] first = {{x, 1}, {0, 0}}, second = {{A, 0}, {B, 1}};
        out.println(multiply(first, exp(second, n), first.length)[0][0]);

        out.flush();
        out.close();
    }

    static int MOD = (int)1e9 + 7;
    static long[][] multiply (long[][] A, long[][] B, int n) {
        long[][] res = new long[n][n];
        for (int row = 0; row < n; row++)
            for (int col = 0; col < n; col++) {
                for (int lop = 0; lop < n; lop++) {
                    res[row][col] += (A[row][lop] * B[lop][col] % MOD);
                    if (res[row][col] >= MOD) res[row][col] -= MOD;
                }
            }
        return res;
    }
    static long[][] identity (int n) {
        long[][] res = new long[n][n];
        for (int i = 0; i < n; i++)
            res[i][i] = 1;
        return res;
    }
    static long[][] exp (long[][] M, long p) {
        if (p == 0) return identity(M.length);
        if (p == 1) return M;
        if (p % 2 == 0) return exp(multiply(M, M, M.length), p >> 1L);
        else return multiply(M, exp(multiply(M, M, M.length), p >> 1L), M.length);
    }
}
