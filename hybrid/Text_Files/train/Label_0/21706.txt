#include<bits/stdc++.h>
using namespace std;
struct node
{
    int maxx;
    int minn;
    int num;
    int ans;
} a[200005];
int cmp(node xx,node yy)
{  
    if(xx.maxx==yy.maxx) return xx.minn>yy.minn;
    return xx.maxx<yy.maxx;
}
int cmpp(node xx,node yy)
{
    return xx.num<yy.num;
}
int main()
{
    int t;
    scanf("%d",&t);
    while(t--)
    {
        int n;
        scanf("%d",&n);
        for(int i=1; i<=n; i++)
        {
            int x,y;
            scanf("%d%d",&x,&y);
            a[i].maxx=max(x,y);
            a[i].minn=min(x,y);
            a[i].num=i;
        }
        sort(a+1,a+n+1,cmp);
        int flag=-1;
        for(int i=1; i<=n; i++)
        {
            if(a[i].minn>a[1].maxx) a[i].ans=a[1].num;
            else
            {
                if(flag!=-1&&a[i].minn>a[flag].minn)
                {
                    a[i].ans=a[flag].num;
                }
                else a[i].ans=-1;
                if(flag==-1&&a[i].maxx!=a[i+1].maxx||a[i].maxx!=a[i+1].maxx&&a[i].minn<a[flag].minn)
                {
                    flag=i;
                }
            }
        }
        sort(a+1,a+n+1,cmpp);
        for(int i=1; i<=n; i++)
        {
            printf("%d ",a[i].ans);
        }
        printf("\n");
    }
    return 0;
}
