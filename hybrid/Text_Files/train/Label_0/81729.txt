cache = [1990, 1991, 1992, 1993, 1994, 1995, 1996, 1997, 1998, 1989]

def solve(n):
	global cache
	y = cache[int(n) % 10]
	for _steps in range(len(n) - 1):
		mod = 10 ** (_steps + 2)
		new_tail = int(n) % mod
		old_tail = y % mod
		if new_tail > old_tail:
			y = y // mod * mod + new_tail
		else:
			y = (y // mod + 1) * mod + new_tail
	return y
n = int(input())
for i in range(n):
	num = input().split("'")[1]
	print(solve(num))
