import sys
import math as mt
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	for i in range(2, 50000):
		x = i * (i - 1) // 2
		y = n - x
		if i + y + 2 <= 10 ** 5 and y >= 0:
			if y > 0:
				print(133, end='')
				for j in range(y):
					print(7, end='')
				for j in range(i - 2):
					print(3, end='')
				print(7, end='')
			else:
				print(1, end='')
				for j in range(i):
					print(3, end='')
				print(7, end='')
			break
	print()
