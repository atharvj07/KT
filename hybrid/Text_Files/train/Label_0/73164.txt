n = int(input())
nums = [int(j) for j in input().split()]
ref = 10000 * [-1]
for j in range(n):
	ref[nums[j] - 1] = j + 1
(index, pointer) = (-1, 9999)
while pointer >= 0:
	if ref[pointer] != -1:
		index = ref[pointer]
		pointer -= 1
		break
	pointer -= 1
value = -1
while pointer >= 0:
	if ref[pointer] != -1:
		value = pointer + 1
		break
	pointer -= 1
print(index, value)
