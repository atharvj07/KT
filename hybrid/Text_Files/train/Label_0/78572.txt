s = list(input())
m = int(input())
n = len(s)
l = [0] * n
a = list(map(int, input().split()))
a.sort()
x = 0
j = 0
i = 0
while i < n and j < m:
	if i == a[j] - 1:
		if x == 0:
			l[i] = 1
			x = 1
		else:
			l[i] = 0
			x = 0
		j += 1
	else:
		if x == 1:
			l[i] = 1
		i += 1
if x == 1:
	k = i + 1
	while k <= (n - i - 1) // 2:
		l[k] = 1
		k += 1
for i in range(n):
	if l[i] == 1:
		(s[i], s[n - i - 1]) = (s[n - i - 1], s[i])
print(''.join(s))
