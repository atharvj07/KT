def mp():
	return map(int, input().split())
t = input()
s = []
n = 0
for i in range(len(t)):
	if t[i] == 'a':
		n = i
for i in range(n + 1):
	if t[i] != 'a':
		s.append(t[i])
ss = t[n + 1:]
x = (len(ss) - len(s)) // 2
s = ''.join(s) + t[n + 1:n + x + 1]
ss = t[n + x + 1:]
if s == ss and (t.count('a') > 0 or len(t) % 2 == 0):
	print(t[:n + x + 1])
else:
	print(':(')
