#include <bits/stdc++.h>
using namespace std;
int n, x, viz[2001], ok, X, Max, poz, type, k1, k2;
struct candy {
  int t, h, m;
} a[2001];
int main() {
  int i;
  cin >> n >> X;
  for (i = 1; i <= n; ++i) {
    cin >> a[i].t >> a[i].h >> a[i].m;
  }
  type = 0;
  x = X;
  while (!ok) {
    ok = 1;
    Max = poz = 0;
    for (i = 1; i <= n; ++i)
      if (!viz[i] && type == a[i].t && x >= a[i].h) {
        ok = 0;
        if (Max < a[i].m) Max = a[i].m, poz = i;
      }
    if (!ok) {
      x += Max;
      viz[poz] = 1;
      ++k1;
      if (!type)
        type = 1;
      else
        type = 0;
    }
  }
  for (i = 1; i <= n; ++i) viz[i] = 0;
  type = 1;
  x = X;
  ok = 0;
  while (!ok) {
    ok = 1;
    Max = poz = 0;
    for (i = 1; i <= n; ++i)
      if (!viz[i] && type == a[i].t && x >= a[i].h) {
        ok = 0;
        if (Max < a[i].m) Max = a[i].m, poz = i;
      }
    if (!ok) {
      x += Max;
      viz[poz] = 1;
      ++k2;
      if (!type)
        type = 1;
      else
        type = 0;
    }
  }
  cout << max(k1, k2);
  return 0;
}
