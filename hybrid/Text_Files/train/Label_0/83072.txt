n = int(input())
list_n = list(map(int, input().split()))
min_list = []

def find_min(l):
	max_d = float('-inf')
	for i in range(len(l) - 1):
		max_d = max(max_d, l[i + 1] - l[i])
	return max_d
for i in range(1, len(list_n) - 1):
	temp = list_n.pop(i)
	min_list.append(find_min(list_n))
	list_n.insert(i, temp)
print(min(min_list))
