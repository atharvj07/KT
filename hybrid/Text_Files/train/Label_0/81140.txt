for _ in range(int(input())):
	(n, x) = map(int, input().split())
	li = [int(p) for p in input().split()]
	odd = 0
	even = 0
	for i in li:
		if i % 2 != 0:
			odd += 1
		else:
			even += 1
	if odd == 0:
		print('No')
		continue
	if even == 0:
		if x % 2 == 0:
			print('No')
		elif odd >= x:
			print('Yes')
		else:
			print('No')
		continue
	if odd % 2 == 0:
		odd = odd - 1
	rem = x - odd
	if rem <= 0:
		print('Yes')
		continue
	if even >= rem:
		print('Yes')
	else:
		print('No')
