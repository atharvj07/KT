#include <bits/stdc++.h>
using namespace std;
int a[26];

int main(){
  string s;
  int n,k;
  cin>>n>>k;
  for(int i=0;i<n;i++){
    cin>>s;
    a[s[0]-'A']++;
  }
  int ans=0;
  while(1){
    int m=0;
    sort(a,a+26);
    reverse(a,a+26);
    for(int i=0;i<26;i++){
      if(a[i]==0) continue;
      m++;
      a[i]--;
      if(m==k) break;
    }
    if(m==k) ans++;
    else break;
  }
  cout<<ans<<endl;

  return 0;
}
