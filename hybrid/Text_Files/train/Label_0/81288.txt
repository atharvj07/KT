from itertools import *

def check(x):
	b = [0] + [1 if xx >= x else -1 for xx in a]
	sum = list(accumulate(b))
	mx = -1000000000
	for i in range(k, n + 1):
		mx = max(mx + b[i], sum[i] - sum[i - k])
		if mx > 0:
			return True
	return False
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
(l, r) = (1, n)
while l <= r:
	mid = (l + r) // 2
	if check(mid):
		l = mid + 1
	else:
		r = mid - 1
print(l - 1)
