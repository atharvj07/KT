int main() {
    ios::sync_with_stdio(false); cin.tie(nullptr);
    int N,H,W; cin >> N >> H >> W;
    vector<int> A(N);
    rep(i,N) cin >> A[i];

    bitset<10000> window;
    rep(i,N) {
        if (i&1) {
            int L = i*W-A[i];
            rep(j,W) {
                window[j+L] = 1;
            }
        } else {
            int L = i*W+A[i];
            rep(j,W) {
                window[j+L] = 1;
            }
        }
    }
    int cnt = 0;
    rep(i,N*W) {
        if (!window[i]) cnt++;
    }
    cout << H*cnt << ln;
}
