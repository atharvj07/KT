import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Day9 {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter writer = new PrintWriter(System.out);
        StringTokenizer st = new StringTokenizer(reader.readLine());
        int a = Integer.parseInt(st.nextToken());
        int b = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());
        int d = Integer.parseInt(st.nextToken());
        if(b - a < 0 && (c > 0 || d > 0)|| c - d < 0 && (a > 0 || b > 0) || b - a < -1 || c - d < -1){
            System.out.println("NO");
            return;
        }
        StringBuilder end = new StringBuilder();
        if(d > c){
            end.append('3');
            end.append(' ');
            --d;
        }
        while(d > 0 && c > 0){
            --d;
            --c;
            end.append('2');
            end.append(' ');
            end.append('3');
            end.append(' ');
        }
        StringBuilder start = new StringBuilder();
        while(a > 0 && b > 0){
            --a;
            --b;
            start.append('0');
            start.append(' ');
            start.append('1');
            start.append(' ');
        }
        if(a > b){
            end.append('0');
            end.append(' ');
            --a;
        }
        StringBuilder mid = new StringBuilder();
        while(b > 0 && c > 0){
            --b;
            --c;
            mid.append('2');
            mid.append(' ');
            mid.append('1');
            mid.append(' ');
        }
        if(b > 1 || c > 1){
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        if(b == 1){
            System.out.println("1 " + start.toString() + mid.toString() + end.toString());
        }
        else if(c == 1){
            System.out.println(start.toString() + mid.toString() + end.toString() + "2 ");
        }
        else{
            System.out.println(start.toString() + mid.toString() + end.toString());
        }
    }
}
