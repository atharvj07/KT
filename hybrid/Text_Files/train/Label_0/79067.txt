length = int(input())
a = []
for i in range(length):
	x = input().split()
	if int(x[1]) >= 0.5 * int(x[0]):
		print(-1)
	elif int(x[1]) == 0:
		for j in range(int(x[0])):
			print(j + 1, end=' ')
		print()
	else:
		l = []
		count = 0
		for k in range(int(x[0])):
			l.append(k + 1)
		for k in range(int(x[0])):
			if k % 2 == 1:
				l.insert(k, l[-1])
				del l[-1]
				count += 1
			if count == int(x[1]):
				break
		print(*l)
