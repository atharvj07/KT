#include <bits/stdc++.h>
using namespace std;
string s;
int x, y, z;
int a, b;
void add(int xx) {
  if (xx >= 2 * b)
    z++;
  else if (xx >= a)
    y++;
  else if (xx >= b)
    x++;
}
void sub(int xx) {
  if (xx >= 2 * b)
    z--;
  else if (xx >= a)
    y--;
  else if (xx >= b)
    x--;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> a >> b;
    cin >> s;
    x = y = z = 0;
    int len = s.length();
    std::vector<int> v;
    int f = 0;
    for (int i = 0; i < len; i++) {
      if (s[i] == 'X') {
        if (f != 0) v.push_back(f), f = 0;
      } else
        f++;
    }
    if (f != 0) v.push_back(f);
    for (auto j : v) {
      add(j);
    }
    int ff = 0;
    for (auto i : v) {
      sub(i);
      for (int j = 0; j <= i - a; j++) {
        add(j), add(i - j - a);
        if (!z && !x && y % 2 == 0) ff = 1;
        sub(j), sub(i - j - a);
      }
      add(i);
    }
    printf("%s\n", ff ? "YES" : "NO");
  }
}
