#include<bits/stdc++.h>
//#include<atcoder/all>
using namespace std;
using ll = long long;
int n,a,b,c,d;

map<ll,ll> memo;

ll dfs(ll x)
{
	//cout<<x<<endl;
	if(x==0)
	{
		return 0;
	}
	if(x==1)
	{
		return d;
	}
	if(memo.count(x))return memo[x];
	ll ret = x*d;
	if(LLONG_MAX/x<d)ret = LLONG_MAX;

	ll A = x%2;
	ret = min(ret,dfs((x-A)/2)+A*d+a);
	A = 2-A;
	if(x%2)ret = min(ret,dfs((x+A)/2)+A*d+a);

	ll B = x%3;
	ret = min(ret,dfs((x-B)/3)+B*d+b);
	B = 3 - B;
	if(x%3)ret = min(ret,dfs((x+B)/3)+B*d+b);

	ll C = x%5;
	ret = min(ret,dfs((x-C)/5)+C*d+c);
	C = 5 - C;
	if(x%5)ret = min(ret,dfs((x+C)/5)+C*d+c);

	return memo[x] = ret;
}

int main()
{
	int t;
	cin >> t;
	while(t--)
	{
		ll x;
		cin >> x;
		cin >> a >> b >> c >> d;
		cout<<dfs(x)<<endl;
		memo.clear();
	}
}
