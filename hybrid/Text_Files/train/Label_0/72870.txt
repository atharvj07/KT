def mp():
	return map(int, input().split())

def it():
	return int(input())
n = it()
dp = [0] * (n + 1)
if n & 1:
	k = 2
	j = 1
	for i in range(n + 1):
		if i & 1:
			dp[i] = k * j * 2
			k += 1
			j += 1
else:
	j = 2
	for i in range(2, n + 1):
		if i & 1 == 0:
			dp[i] = j * j
			j += 1
print(dp[n])
