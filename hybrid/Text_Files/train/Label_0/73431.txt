import itertools

def possible(cubes, n):
	s = str(n)
	if len(cubes) < len(s):
		return False
	for p in itertools.permutations(range(len(cubes))):
		ok = True
		for i in range(len(s)):
			if int(s[i]) not in cubes[p[i]]:
				ok = False
		if ok:
			return True
	return False
n = int(input())
cubes = []
for i in range(n):
	cubes.append(list(map(int, input().split())))
for i in range(1, 1001):
	if not possible(cubes, i):
		print(i - 1)
		break
