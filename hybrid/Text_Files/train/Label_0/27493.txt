#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n;
  cin >> n;
  string s;
  cin >> s;
  long long n1 = 1, n2 = 1;
  for (long long i = 0; i <= n - 1; i++) {
    if (s[i] == s[i + 1]) {
      n1 = i + 2;
    } else
      break;
  }
  for (long long i = 0; i <= n - 1; i++) {
    if (s[n - 1 - i] == s[n - 2 - i]) {
      n2 = i + 2;
    } else
      break;
  }
  long long ans;
  if (s[0] == s[n - 1]) {
    ans = ((n1 + 1) * (n2 + 1)) % 998244353;
  } else
    ans = (n1 + n2 + 1) % 998244353;
  cout << ans << "\n";
}
