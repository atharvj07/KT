for _ in range(int(input())):
	(x, y, z) = map(int, input().split())
	if x == y and y == z:
		print('YES')
		print(x, y, z)
		continue
	arr = sorted([x, y, z])
	if arr[2] == x and arr[2] == y or (arr[2] == z and arr[2] == y) or (arr[2] == x and arr[2] == z):
		a = arr[2]
		b = arr[0]
		c = arr[0] - 1
		if c <= 0:
			print('YES')
			print(a, b, b)
		else:
			print('YES')
			print(a, b, c)
	else:
		print('NO')
