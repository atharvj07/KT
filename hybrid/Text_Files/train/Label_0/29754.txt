#include <bits/stdc++.h>
using namespace std;
struct node {
  int rate;
  int div;
};
node pre[200 * 1000 + 10];
int div1 = -1, div2 = -1;
const int maxn = 0x3f3f3f3f;
void judge(int n) {
  pre[0].rate = 0;
  for (int i = 1; i <= n; i++) {
    pre[i].rate += pre[i - 1].rate;
  }
  int ansmax = maxn;
  int ansmin = -maxn;
  for (int i = 1; i <= n; i++) {
    if (pre[i].div == 1) {
      ansmin = max(ansmin, 1900 - pre[i - 1].rate);
    }
    if (pre[i].div == 2) {
      ansmax = min(ansmax, 1899 - pre[i - 1].rate);
    }
  }
  if (ansmax >= maxn) {
    printf("Infinity\n");
    return;
  }
  if (ansmin <= ansmax) {
    printf("%d\n", ansmax + pre[n].rate);
    return;
  }
  if (ansmin > ansmax) {
    printf("Impossible\n");
  }
  return;
}
int main() {
  int n;
  while (scanf("%d", &n) != EOF) {
    div1 = -1, div2 = -1;
    for (int i = 1; i <= n; i++) {
      scanf("%d %d", &pre[i].rate, &pre[i].div);
      if (div1 == -1 && pre[i].div == 1) div1 = i;
      if (div2 == -1 && pre[i].div == 2) div2 = i;
    }
    judge(n);
  }
  return 0;
}
