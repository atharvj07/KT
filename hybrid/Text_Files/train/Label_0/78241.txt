(n, k) = map(int, input().split())
v = list(map(int, input().split()))
rez = []
chk = False
if sum(v) % k != 0:
	chk = True
else:
	d = sum(v) // k
	s = 0
	c = 0
	for i in range(0, len(v)):
		s += v[i]
		c += 1
		if s == d:
			rez.append(c)
			s = 0
			c = 0
		elif s > d:
			chk = True
			break
if chk:
	print('No')
else:
	print('Yes')
	print(' '.join((str(e) for e in rez)))
