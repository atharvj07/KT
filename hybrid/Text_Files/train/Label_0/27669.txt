#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 9;
int n, m, a[N], f[N];
vector<int> g[N];
void dfs(int u) {
  f[u] = a[u];
  for (int v : g[u]) {
    if (f[v] < 0) dfs(v);
    if (a[u])
      f[u] = max(f[u], f[v] + (a[v] == 0));
    else
      f[u] = max(f[u], f[v]);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie();
  cout.tie();
  cin >> n >> m;
  for (int i = 0; i < n; ++i) cin >> a[i];
  int u, v;
  for (int i = 1; i <= m; ++i) {
    cin >> u >> v;
    g[u].push_back(v);
  }
  for (int i = 0; i < n; ++i) f[i] = -1;
  int res = 0;
  for (int i = 0; i < n; ++i)
    if (f[i] < 0) {
      dfs(i);
      res = max(res, f[i]);
    }
  cout << res;
}
