#include<stdio.h>
#include<math.h>
#include<queue>
#include<vector>
#include<map>
#define INF 200000000
using namespace std;
typedef pair<int,int> P;
typedef struct{
	int a,b,c;
}edge;
vector<P> e[3000];
edge e2[100000];
int d[3000];
int max(int a,int b){return a>b?a:b;}
int main(){
	int n,m,k;
	int i,j;
	scanf("%d %d %d",&n,&m,&k);
	for(i=0;i<m;i++){
		int x,y,z;
		scanf("%d %d %d",&x,&y,&z);
		x--;y--;
		edge ed={x,y,z};
		e[x].push_back(P(y,z));
		e[y].push_back(P(x,z));
		e2[i]=ed;
	}
	priority_queue<P> que;
	for(i=0;i<n;i++)d[i]=INF;
	for(i=0;i<k;i++){
		int x;
		scanf("%d",&x);
		d[x-1]=0;
		que.push(P(0,x-1));
	}
	while(!que.empty()){
		P p=que.top();
		que.pop();
		int v=p.second;
		if(d[v]<p.first)continue;
		for(i=0;i<e[v].size();i++){
			if(d[e[v][i].first]>d[v]+e[v][i].second){
				d[e[v][i].first]=d[v]+e[v][i].second;
				que.push(P(d[e[v][i].first],e[v][i].first));
			}
		}
	}
	int ans=0;
	for(i=0;i<m;i++){
		int x=e2[i].a;
		int y=e2[i].b;
		int z=e2[i].c;
		ans=max(ans,max(d[x],max(d[y],(int)round((double)(d[x]+d[y]+z)/2))));
	}
	printf("%d\n",ans);
	return 0;
}
