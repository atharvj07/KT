import os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
t = int(input())
for _ in range(t):
	n = int(input())
	graph = []
	for i in range(n + 1):
		graph.append([])
	edges = []
	for i in range(n - 1):
		(x, y) = map(int, input().split())
		graph[x].append(y)
		graph[y].append(x)
		edges.append([x, y])
	for i in range(1, n + 1):
		if len(graph[i]) == 1:
			start = i
			break
	stack = []
	stack.append(start)
	visited = [False] * (n + 1)
	children = [0] * (n + 1)
	parent = [-1] * (n + 1)
	while len(stack) != 0:
		s = stack.pop()
		if not visited[s]:
			visited[s] = True
			stack.append(s)
			for i in range(len(graph[s])):
				if not visited[graph[s][i]]:
					stack.append(graph[s][i])
					parent[graph[s][i]] = s
		elif parent[s] != -1:
			children[parent[s]] += children[s] + 1
	flag = 0
	for i in range(1, n + 1):
		if 2 * (children[i] + 1) == n:
			print(start, graph[start][0])
			print(start, i)
			flag = 1
			break
	if flag == 0:
		print(start, graph[start][0])
		print(start, graph[start][0])
