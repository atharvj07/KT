import sys

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())
(n, A, B) = mints()
s = list(mints())
s1 = s[0]
S = sum(s)
s = s[1:]
s.sort(reverse=True)
if s1 * A >= B * S:
	print(0)
else:
	for i in range(len(s)):
		S -= s[i]
		if s1 * A >= B * S:
			print(i + 1)
			exit(0)
