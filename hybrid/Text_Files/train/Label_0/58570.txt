import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.io.StreamTokenizer;
import java.util.Locale;

public class F45 implements Runnable {

    public StreamTokenizer in;

    public PrintWriter out;

    final static String TASK_NAME = "";

    F45() throws IOException {
        Locale.setDefault( Locale.US );
        in = new StreamTokenizer( new InputStreamReader( System.in ) );
        out = new PrintWriter( System.out );
    }

    int nextInt() throws IOException {
        in.nextToken();
        return ( int ) in.nval;
    }

    void close() throws IOException {
        out.close();
    }

    public void run() {
        try {
            solve();
            close();
        } catch ( Exception e ) {
            e.printStackTrace();
        }
    }

    int W;
    int[] w;
    int[] c;
    int[] a;

    public void solve() throws IOException {
        int n = nextInt();
        int m = nextInt();
        W = nextInt();
        w = new int[m];
        for ( int i = 0; i < m; i ++ ) {
            w[i] = nextInt();
        }
        c = new int[m];
        for ( int i = 0; i < m; i ++ ) {
            c[i] = nextInt();
        }
        a = new int[m];
        for ( int i = 0; i < m; i ++ ) {
            a[i] = nextInt();
        }
        long r = 0;
        double rr = 0.0;
        for ( int i = 0; i < n; i ++ ) {
            if ( i > 0 ) {
                for ( int j = 0; j < m; j ++ ) {
                    c[j] -= a[j];
                }
            }
            qsort( 0, m - 1, 0 );
//            for ( int j = 0; j < m; j ++ ) {
//                out.print( w[j] + " " + c[j] + "   " );
//            }
//            out.println();
            int left = W;
            for ( int j = 0; j < m; j ++ ) {
                if ( w[j] <= left ) {
                    r += c[j];
                    left -= w[j];
                } else {
                    long dr = c[j] * ( long ) left / w[j];
                    r += dr;
                    rr += ( c[j] * ( long ) left - dr * w[j] ) * 1.0 / w[j];
                    break;
                }
            }
//            out.format( "%d %.20f\n", r, rr );
        }
        long t = ( long ) Math.floor( rr );
        r += t;
        rr -= t;
        String s = String.format( "%.15f", rr );
        out.println( r + s.substring( s.indexOf( '.' ) ) );
    }

    private void qsort( int l, int r, int s0 ) {
        if ( l >= r ) {
            return;
        }
        int i = l;
        int j = r;
        long x = c[( i + j ) >> 1];
        long y = w[( i + j ) >> 1];
        int s = s0;
        do {
            while ( c[i] * y < w[i] * x ) {
                s += w[i];
                if ( s > W ) {
                    s = W;
                }
                i ++;
            }
            while ( c[j] * y > w[j] * x ) {
                j --;
            }
            if ( i <= j ) {
                int t = w[i];
                w[i] = w[j];
                w[j] = t;
                t = c[i];
                c[i] = c[j];
                c[j] = t;
                t = a[i];
                a[i] = a[j];
                a[j] = t;
                s += w[i];
                if ( s > W ) {
                    s = W;
                }
                i ++;
                j --;
            }
        } while ( i <= j );
        if ( s >= W ) {
            qsort( l, i - 1, s0 );
        } else {
            qsort( i, r, s );
        }
//        qsort( l, j, 0 );
//        qsort( i, r, 0 );
    }

    public static void main( String[] args ) throws IOException {
        new Thread( new F45() ).start();
    }
}
