#include <bits/stdc++.h>
using namespace std;
using ll = long long int;
using bigint = int64_t;
const int MOD = 1000000007;
void io_set() {}
void solve() {
  ll n;
  cin >> n;
  for (ll i = n; i >= 1; i--) {
    if (!(n % i)) {
      ll j = i, flag = true;
      vector<ll> bin;
      while (j) {
        bin.push_back(j & 1);
        j >>= 1;
      }
      ll s = bin.size(), k = 0;
      while (bin[k] == 0) {
        k++;
      }
      ll diff = s - k;
      for (ll i = k + 1; i <= s - 1; i++) {
        if (!bin[i]) {
          flag = false;
          break;
        }
      }
      if (flag && diff == k + 1) {
        cout << i << "\n";
        return;
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  ;
  io_set();
  ll tt = 1;
  while (tt--) {
    solve();
  }
  return 0;
}
