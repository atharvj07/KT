#include <bits/stdc++.h>
using namespace std;
const long long mod = 998244353;
template <typename T>
inline void Add(T &x, T y) {
  x += y;
  if (x >= mod) x -= mod;
}
template <typename T>
inline void Mul(T &x, T y) {
  (x *= y) %= mod;
}
inline int read() {
  int x = 0;
  char ch = getchar();
  bool f = 0;
  for (; ch < '0' || ch > '9'; ch = getchar())
    if (ch == '-') f = 1;
  for (; ch >= '0' && ch <= '9'; ch = getchar())
    x = (x << 3) + (x << 1) + (ch ^ 48);
  return f ? -x : x;
}
const int N = 1005;
int n, m, a[N], L[N], R[N];
long long f[N][N];
int main() {
  n = read();
  int j, x, y, k, w;
  for (int i = (1), end_i = (read()); i <= end_i; i++)
    if ((x = read()) != a[m]) a[++m] = x;
  if (m > 2 * n + 1) return printf("0") & 0;
  for (int i = (1), end_i = (m); i <= end_i; i++)
    if (!L[a[i]]) L[a[i]] = i;
  for (int i = (m), end_i = (1); i >= end_i; i--)
    if (!R[a[i]]) R[a[i]] = i;
  for (int i = (1), end_i = (m); i <= end_i; i++)
    if (L[a[i]] == i && R[a[i]] == i) f[i][i] = 1;
  for (int i = (0), end_i = (m); i <= end_i; i++) f[i + 1][i] = 1;
  long long s1, s2;
  for (int len = (2), end_i = (m); len <= end_i; len++)
    for (int i = (1), end_i = (m + 1 - len); i <= end_i; i++) {
      j = i + len - 1;
      k = i;
      for (int l = (i + 1), end_i = (j); l <= end_i; l++)
        if (a[k] > a[l]) k = l;
      x = L[a[k]];
      y = R[a[k]];
      if (x < i || j < y) continue;
      s1 = s2 = 0;
      for (int l = (i), end_i = (x); l <= end_i; l++)
        Add(s1, f[i][l - 1] * f[l][x - 1] % mod);
      for (int l = (y), end_i = (j); l <= end_i; l++)
        Add(s2, f[y + 1][l] * f[l + 1][j] % mod);
      f[i][j] = s1 * s2 % mod;
      w = 0;
      for (int l = (i), end_i = (j); l <= end_i; l++)
        if (a[l] == a[k]) {
          if (w) Mul(f[i][j], f[w + 1][l - 1]);
          w = l;
        }
    }
  printf("%lld", f[1][m]);
  return 0;
}
