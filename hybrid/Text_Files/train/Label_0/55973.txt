import java.util.*;

public class main{
	
	static int len(int x){
		int ans = 0;
		while(x>0){
			ans++;
			x/=10;
		}
		return ans;
	}	

	public static void main(String[] args){
		Scanner in = new Scanner(System.in);
		int n = in.nextInt(), k = in.nextInt();
		int[] a = new int[n];
		int[] t = new int[11];
		t[0] = 1;
		for(int i=1;i<11;i++)t[i] = t[i-1]*10%k;
		HashMap<Integer,Integer>[] h = new HashMap[11];
		for(int i=0;i<11;i++)h[i] = new HashMap<>();
		for(int i=0;i<n;i++){
			a[i] = in.nextInt();
			int x = len(a[i]);
			h[x].put(a[i]%k,h[x].getOrDefault(a[i]%k,0)+1);
		}
		long ans = 0L;
		for(int i=0;i<n;i++){
			for(int j=1;j<11;j++){
				int y = (int)(((long)a[i])*t[j]%k);
				int z = (k-y)%k;
				ans+=h[j].getOrDefault(z,0);
				if(a[i]%k==z && len(a[i])==j)
					ans--;
			}
		}
		System.out.println(ans);

	}
}
