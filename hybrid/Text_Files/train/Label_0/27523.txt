#include <bits/stdc++.h>
using namespace std;
const int MAX = 201000;
int a[MAX];
int n, k;
int eu[MAX], ev[MAX];
bool cant[MAX];
int deg[MAX];
int val[MAX];
int down[MAX][2];
bool vis[MAX];
set<int> e[MAX];
void cut(int x) {
  set<pair<int, int> > s;
  for (int i = 0; i < n; ++i)
    if (a[i] >= x && e[i].size()) s.insert(pair<int, int>(deg[i], i));
  while (s.size() && s.begin()->first == 1) {
    int u = s.begin()->second;
    deg[u]--;
    s.erase(s.begin());
    if (cant[u]) continue;
    int v = *e[u].begin();
    e[u].clear();
    e[v].erase(u);
    s.erase(pair<int, int>(deg[v], v));
    --deg[v];
    s.insert(pair<int, int>(deg[v], v));
    val[v] += val[u];
  }
}
int DFS(int u) {
  vis[u] = true;
  int res = 0;
  for (auto it : e[u]) {
    int v = it;
    if (vis[v]) continue;
    res = max(res, DFS(v));
    down[u][0] = max(down[u][0], down[v][1]);
    if (down[u][0] > down[u][1]) swap(down[u][0], down[u][1]);
  }
  down[u][0] += val[u];
  down[u][1] += val[u];
  return max(res, down[u][0] + down[u][1] - val[u]);
}
bool f(int x) {
  for (int i = 0; i < n; ++i) {
    deg[i] = 0;
    if (a[i] >= x)
      val[i] = 1;
    else
      val[i] = 0;
    vis[i] = cant[i] = false;
    down[i][0] = down[i][1] = 0;
    e[i].clear();
  }
  for (int i = 0; i < n - 1; ++i) {
    int u = eu[i];
    int v = ev[i];
    if (a[u] >= x && a[v] >= x) {
      e[u].insert(v);
      e[v].insert(u);
    } else {
      cant[u] = true;
      cant[v] = true;
    }
    ++deg[u];
    ++deg[v];
  }
  cut(x);
  for (int i = 0; i < n; ++i) {
    if (!vis[i] && a[i] >= x) {
      if (DFS(i) >= k) return true;
    }
  }
  return false;
}
int main() {
  srand(time(NULL));
  scanf("%d %d", &n, &k);
  for (int i = 0; i < n; ++i) scanf("%d", a + i);
  for (int i = 0; i < n - 1; ++i) scanf("%d %d", &eu[i], &ev[i]);
  for (int i = 0; i < n - 1; ++i) --eu[i], --ev[i];
  int lo = 1;
  int hi = 1000000;
  int mid = 0;
  while (lo < hi) {
    mid = (lo + hi + 1) / 2;
    if (f(mid))
      lo = mid;
    else
      hi = mid - 1;
  }
  printf("%d\n", lo);
}
