
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;

public class BeforeExam {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        
        int tPerCard = n/k;
        
        int[] profs = new int[n];
        for (int i = 0; i < profs.length; i++) {
            profs[i] = sc.nextInt();
        }
        boolean[] remaining = new boolean[profs.length];
        Arrays.fill(remaining, true);

        double min = Double.MAX_VALUE;
        double max = Double.MIN_VALUE;
        
        int q = sc.nextInt();
        Set<String> read = new HashSet<String>();
        for (int i = 0; i < q; i++) {
            double avg = 0;
            List<Integer> l = new ArrayList<Integer>();
            for (int j = 0; j < tPerCard; j++) {
                int t = sc.nextInt()-1;
                avg += profs[t];
                l.add(profs[t]);
                remaining[t] = false;
            }
            Collections.sort(l);
            read.add(l.toString());
            avg /= tPerCard;
            min = Math.min(min, avg);
            max = Math.max(max, avg);
        }
        
        int remCount = 0;
        for (int i = 0; i < remaining.length; i++) {
            if(remaining[i]) {
                remCount++;
            }
        }
        int[] toSort = new int[remCount];
        int pos = 0;
        for (int i = 0; i < remaining.length; i++) {
            if(remaining[i]) {
                toSort[pos] = profs[i];
                pos++;
            }
        }
        Arrays.sort(toSort);
//      System.out.println(Arrays.toString(toSort));
        
        if(toSort.length >= tPerCard && read.size() < k) {
            double minConf = 0;
            for (int i = 0; i < tPerCard; i++) {
                minConf += toSort[i];
            }
            min = Math.min(min, minConf/tPerCard);
            
            double maxConf = 0;
            for (int i = toSort.length-tPerCard; i < toSort.length; i++) {
                maxConf += toSort[i];
            }
            max = Math.max(max, maxConf/tPerCard);
        }
        
        System.out.println(min+" "+max);
    }
}
