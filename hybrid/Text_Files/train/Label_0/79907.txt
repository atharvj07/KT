(n, k) = map(int, input().split())
s = input()
l = [0] * 26
f = 0
for i in range(n):
	a = s[i]
	l[ord(a) - 65] = i
out = []
for i in range(n):
	if s[i] not in out:
		out.append(s[i])
	if len(out) > k:
		print('YES')
		f = 1
		break
	if i == l[ord(s[i]) - 65]:
		out.remove(s[i])
if f == 0:
	print('NO')
