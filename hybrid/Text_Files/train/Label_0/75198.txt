import math, sys
from sys import stdin, stdout
from collections import Counter, defaultdict, deque
input = stdin.readline
I = lambda : int(input())
li = lambda : list(map(int, input().split()))

def case():
	n = I()
	a = li()
	ans = []
	if n < 3:
		print(0)
		exit()
	for i in range(n):
		ans.append([a[i], i])
	ans.sort(reverse=True)
	pink = []
	s = sum(a)
	for i in range(n):
		c = s
		if i == 0:
			c -= ans[i][0]
			if ans[1][0] * 2 == c:
				pink.append(ans[0][1] + 1)
		else:
			c -= ans[i][0]
			if ans[0][0] * 2 == c:
				pink.append(ans[i][1] + 1)
	print(len(pink))
	print(*pink)
for _ in range(1):
	case()
