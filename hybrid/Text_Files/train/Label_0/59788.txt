 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Solution {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        Task solver = new Task();
        solver.solve(1, in, out);
        out.close();
    }
}

class Task
{
    public void solve(int testNumber, InputReader in, PrintWriter out)
    {
        int n = in.nextInt();
        int m = in.nextInt();
        String[] Viruses = new String[m];
        for (int i = 0; i < m; i++)
        {
            Viruses[i] = in.nextToken();
        }
        HashMap<String, Integer> Index = new HashMap<String, Integer>();
        int lastIndex = 1;
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < Viruses[i].length(); j++)
            {
                Index.put(Viruses[i].substring(0, j + 1), lastIndex++);
            }
        }
        Index.put("",0);
        int[][] replaceMent = new int[lastIndex][26];
        int[] longestSuffix = new int[lastIndex];
        //Найдем строку, которая заканчиывает каждый префикс и является наибольшей:
        for (String s : Index.keySet())
            for (String t : Viruses)
                if (s.endsWith(t))
                    longestSuffix[Index.get(s)] = Math.max(longestSuffix[Index.get(s)], t.length());

        //Допустим мы в позиции I, и пишем букву CH. Найдем максимальный префикс который в итоге получитcя:

        for (String s : Index.keySet()) {
            int start = Index.get(s);
            for (char ch = 'A'; ch <= 'Z'; ch++) {
                String candidate=s.concat(Character.toString(ch));//берем по максимуму
                while (!Index.containsKey(candidate))
                    candidate = candidate.substring(1);//отрезаем первый символ
                replaceMent[start][ch-'A'] = Index.get(candidate);
            }
        }

        int[][] dp = new int[lastIndex][20]; //За состояние в индексе I примем последний
                                                //префикс И удаленность первого ненакрытого символа
        int modulo=1000000009;
        dp[Index.get("")][0] = 1;//Существует ровно одна пустая строка
        for (int idx = 0; idx < n; ++idx) {
            int[][] newDp = new int[lastIndex][20];
            for (int oldprefix = 0; oldprefix < lastIndex; oldprefix++)
                for (int oldCover = 0; oldCover < 20; oldCover++) {
                    for (char ch='A'; ch <='Z'; ch++) {
                        int newPrefix = replaceMent[oldprefix][ch-'A'];
                        int newCover = oldCover + 1; //теперь нам надо на 1 больше,так как мы продвинулись на 1
                        if (newCover <= longestSuffix[newPrefix])
                            newCover = 0;//если мы накрыли то что было ненакрыто, то we are happy!
                        if (newCover >=20)//тяжелый случай
                            continue;
                        newDp[newPrefix][newCover] = (newDp[newPrefix][newCover]+dp[oldprefix][oldCover])%modulo;

                    }
                }
            dp=newDp;
            //cnt = oneStep(cnt, maxLength, last, isFinish, next);
        }
        int answer = 0;
        for (int i = 0; i < lastIndex; i++) {
            answer = (answer+dp[i][0])%modulo;

        }
        out.println(answer);
    }
}

class InputReader {
    StringTokenizer st;
    BufferedReader in;
    public InputReader(InputStream ins)
    {
        in = new BufferedReader(new InputStreamReader(ins));
    }

    public String nextToken()
    {
        while(st==null || !st.hasMoreTokens())
        {
            try {
                st=new StringTokenizer(in.readLine());
            } catch (IOException e) {
                e.printStackTrace();  //To change body of catch statement use File | Settings | File Templates.
            }
        }
        return st.nextToken();
    }
    public int nextInt()
    {

        return Integer.parseInt(nextToken());
    }

    }
