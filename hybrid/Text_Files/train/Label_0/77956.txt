hp = int(input())
categories = {1: 'A', 3: 'B', 2: 'C', 0: 'D'}
options = [0, 1, 2]
best_category = 'D'
best_option = None
for option in options:
	remainder = (hp + option) % 4
	category = categories[remainder]
	if category < best_category:
		best_category = category
		best_option = option
print(best_option, best_category)
