import math, sys, bisect
from collections import defaultdict, Counter, deque
from itertools import groupby, accumulate
input = iter(sys.stdin.buffer.read().decode().splitlines()).__next__
ilele = lambda : map(int, input().split())
alele = lambda : list(map(int, input().split()))

def Y(c):
	print(['NO', 'YES'][c])

def y(c):
	print(['no', 'yes'][c])

def fun(x):
	return x * (x + 1) // 2
(n, x) = ilele()
A = alele()
B = [0] + list(accumulate(A + A))
C = [fun(i) for i in A]
D = [0] + list(accumulate(C + C))
maxi = -1000000000.0
i = j = len(B) - 1
while i > 0:
	y = B[j] - B[i - 1]
	if y < x:
		i -= 1
	else:
		extra = fun(y - x)
		maxi = max(D[j] - D[i - 1] - extra, maxi)
		j -= 1
print(maxi)
