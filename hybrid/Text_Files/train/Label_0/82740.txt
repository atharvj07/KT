n = int(input())
xs = [int(x) for x in input().split()]
plus = 0
minus = 0
null = 0
maxi = max(xs)
mini = min(xs)
mid = None
for i in range(n):
	if xs[i] == maxi:
		plus += 1
	elif xs[i] == mini:
		minus += 1
	else:
		null += 1
		mid = xs[i]
if maxi == mini:
	print(n)
	print(*xs)
elif mid == None and maxi - mini != 2:
	print(n)
	print(*xs)
else:
	if mid == None:
		mid = (maxi + mini) // 2
	min_plus = plus - min(plus, minus)
	min_null = null + min(plus, minus) * 2
	min_minus = minus - min(plus, minus)
	minimum = null + max(plus, minus) - min(plus, minus)
	plus_now = min_plus
	minus_now = min_minus
	null_now = min_null
	for null_now in range(null + min(plus, minus) * 2, null % 2 - 1, -2):
		cons_minus = min(minus_now, minus)
		cons_null = min(null_now, null)
		cons_plus = min(plus_now, plus)
		if cons_minus + cons_null + cons_plus <= minimum:
			minimum = cons_minus + cons_null + cons_plus
			min_plus = plus_now
			min_minus = minus_now
			min_null = null_now
		plus_now += 1
		minus_now += 1
	print(minimum)
	pk = 1
	nk = 1
	if min_plus == 0:
		pk = 0
	if min_null == 0:
		nk = 0
	print(*[maxi] * min_plus, end=' ' * pk)
	print(*[mid] * min_null, end=' ' * nk)
	print(*[mini] * min_minus)
