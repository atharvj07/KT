import java.util.Scanner;

public class Main {

    public static void main(String args[]) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            int[] b = new int[n];
            int[] c = new int[n+1];
            int[] visited = new int[n+1];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }
            for (int i = 0; i < n; i++) {
                b[i] = scanner.nextInt();
            }

            for (int i = 0; i < n; i++) {
                c[a[i]] = b[i];
            }

            long combinations = 0;
            for (int i = 0; i < n; i++) {
                if (visited[a[i]] != 1) {
                    visited[a[i]] = 1;
                    combinations += visit(c, visited, a[i], b[i]);
                }
            }
            long ans = 1;
            int i = 1;
            while(i <= combinations) {
                ans *= 2;
                ans %= 1000000007;
                i++;
            }
            System.out.println(ans);
        }

    }

    private static int visit(int[] c, int[] visited, int startingVal, int currentValToCheck) {
        int currentB = c[currentValToCheck];
        visited[currentValToCheck] = 1;
        if (currentB == startingVal) {
            return 1;
        }
        return visit(c, visited, startingVal, currentB);
    }

}
