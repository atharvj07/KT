(l, k) = list(map(int, input().split()))
(h, d, c, n) = list(map(int, input().split()))
if l < 20:
	na = (20 - l) * 60
	la = na - k
	man = la * d
	sa = man + h
	ta = sa // n
	if sa / n == ta:
		ga = c * ta
	else:
		ga = c * (ta + 1)
elif l >= 20:
	ta = h // n
	if h / n == ta:
		ga = c * ta
	else:
		ga = c * (ta + 1)
lop = h // n
if lop == h / n:
	kj = c * lop
else:
	kj = c * (lop + 1)
print(min(kj, 80 / 100 * ga))
