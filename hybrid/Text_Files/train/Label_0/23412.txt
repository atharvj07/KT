#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long int n, k;
    cin >> n >> k;
    long long int ans = n;
    for (long long int i = 2; i * i <= n; i++) {
      if (n % i == 0) {
        ans += i;
        k--;
        break;
      }
    }
    if (ans == n) {
      ans += n;
      k--;
    }
    for (long long int i = 1; i <= k; i++) ans += 2;
    cout << ans << endl;
  }
  return 0;
}
