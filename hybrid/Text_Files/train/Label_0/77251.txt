for _ in range(int(input())):
	(n, k, l) = tuple((int(x) for x in input().split()))
	depth = [0] * (n + 1)
	depth[0] = -1000
	cnt = 1
	for x in input().split():
		depth[cnt] = int(x)
		cnt += 1
	dp = [[0 for i in range(k * 2 * n)] for j in range(n + 1)]
	farest = [0] * (k * 2 * n)
	tide = [0] * 2 * k
	for i in range(2 * k):
		if i <= k:
			tide[i] = i
		else:
			tide[i] = k * 2 - i
	for i in range(k * 2 * n):
		dp[0][i] = 1
	able = False
	for i in range(k * 2 * n):
		if i >= 2 * k and farest[i - 2 * k] >= farest[i]:
			able = False
			break
		if farest[i] == n:
			able = True
			break
		if i == 2 * k * n - 1:
			break
		for j in range(farest[i] + 1):
			if dp[j][i] == 1 and depth[j] + tide[(i + 1) % (k * 2)] <= l:
				dp[j][i + 1] = 1
				farest[i + 1] = max(farest[i + 1], j)
			if dp[j][i] == 1 and depth[j + 1] + tide[(i + 1) % (k * 2)] <= l:
				dp[j + 1][i + 1] = 1
				farest[i + 1] = max(farest[i + 1], j + 1)
	print('yes' if able else 'no')
