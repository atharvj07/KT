#include <bits/stdc++.h>
using namespace std;
long long inv(long long a, long long b) {
  return (a > 1 ? b - inv(b % a, a) * b / a : 1);
}
const long double e = 2.71828182845904523536028747;
const long double PI = 3.14159265359;
const long long inf = (1LL << 62);
const int iinf = (1 << 30);
int n, s1, s2, k1, k2, moves;
char c;
int main() {
  cin >> n;
  for (int i = 1; i <= n / 2; i++) {
    cin >> c;
    if (c >= '0' && c <= '9')
      s1 += (c - '0');
    else
      k1++;
  }
  for (int i = 1; i <= n / 2; i++) {
    cin >> c;
    if (c >= '0' && c <= '9')
      s2 += (c - '0');
    else
      k2++;
  }
  if (k1 > k2) swap(k1, k2), swap(s1, s2);
  k2 -= k1;
  if (!k2) {
    if (s1 != s2) return cout << "Monocarp", 0;
    return cout << "Bicarp", 0;
  }
  if (s2 > s1) return cout << "Monocarp", 0;
  if (s1 - s2 != 9 * (k2 / 2)) return cout << "Monocarp", 0;
  return cout << "Bicarp", 0;
  return 0;
}
