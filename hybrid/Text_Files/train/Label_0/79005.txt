n = int(input())
d = {}
for i in range(n):
	s = input()
	r = 0
	l = 0
	for i in s:
		if i == '(':
			r += 1
		elif r == 0:
			l += 1
		else:
			r -= 1
	if l == 0 or r == 0:
		c = l - r
		if c in d:
			d[c] += 1
		else:
			d[c] = 1
arr = list(d.keys())
arr.sort()
ans = 0
for i in arr:
	if i > 0:
		break
	if -i in d:
		ans += d[i] * d[-i]
print(ans)
