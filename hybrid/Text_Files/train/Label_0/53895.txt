import java.util.*;

public class SuffixThree {
  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int t = s.nextInt();
    while (t-- > 0) {
      String line = s.next();
      if (line.endsWith("po")) {
        System.out.println("FILIPINO");
      } else if (line.endsWith("mnida")) {
        System.out.println("KOREAN");
      } else {
        System.out.println("JAPANESE");
      }   
    }   
  }
}
