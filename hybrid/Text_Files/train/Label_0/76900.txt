t = int(input())
for i in range(t):
	a = list(map(int, input().split()))
	odd = 0
	even = 0
	for i in range(len(a)):
		if a[i] % 2 != 0:
			odd += 1
		else:
			even += 1
	if odd <= 1:
		print('Yes')
	else:
		if a[0] > 0 and a[1] > 0 and (a[2] > 0):
			a[0] -= 1
			a[1] -= 1
			a[2] -= 1
			a[3] += 3
			c = odd
			odd = even
			even = c
		if odd <= 1:
			print('Yes')
		else:
			print('No')
