h = int(input())
a = list(map(int, input().split()))
for i in range(h):
	if a[i] != 1 and a[i + 1] != 1:
		I = i
		f = False
		break
else:
	f = True
if f:
	print('perfect')
else:
	print('ambiguous')
	(ans1, ans2) = ([0], [0])
	last_el = 1
	last_i = 0
	f = False
	for (i, el) in enumerate(a[1:]):
		for k in range(el):
			ans1.append(last_el)
			if k != el - 1:
				ans2.append(last_el)
		if i == I:
			ans2.append(last_el + 1)
			f = True
		else:
			ans2.append(last_el)
		last_i += a[i]
		last_el = last_i + 1
	print(' '.join(map(str, ans1)))
	print(' '.join(map(str, ans2)))
