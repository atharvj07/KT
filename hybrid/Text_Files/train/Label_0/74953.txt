for j in range(int(input())):
	out = -1
	n = int(input())
	s = input()
	d = set(s)
	for i in d:
		a = 0
		l = 0
		r = n - 1
		while l < r and a != -1:
			if s[l] == s[r]:
				l += 1
				r -= 1
			elif s[l] == i:
				l += 1
				a += 1
			elif s[r] == i:
				r -= 1
				a += 1
			else:
				a = -1
		if a != -1 and (out == -1 or a < out):
			out = a
	print(out)
