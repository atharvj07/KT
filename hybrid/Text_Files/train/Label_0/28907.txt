#include <bits/stdc++.h>
using namespace std;
const int INF = 1000000 + 10;
int ans;
int vis[20];
string str;
string ss[5] = {"00", "25", "50", "75"};
int Change(string &str, int Index, char ch) {
  int ret = 0;
  for (int i = Index; i >= 0; --i) {
    if (str[i] == ch) {
      for (int j = i; j < Index; ++j) {
        swap(str[j + 1], str[j]);
        ++ret;
      }
      return ret;
    }
  }
  return 0;
}
bool Check(const string &str, const string &ss) {
  memset(vis, 0, sizeof(vis));
  for (int i = 0; i < 2; ++i) {
    ++vis[ss[i] - '0'];
  }
  for (int i = 0; str[i]; ++i) {
    --vis[str[i] - '0'];
  }
  for (int i = 0; i < 10; ++i) {
    if (vis[i] > 0) {
      return false;
    }
  }
  return true;
}
int solve(string str, const string &ss) {
  if (!Check(str, ss)) {
    return -1;
  }
  int len = str.length();
  int ret = Change(str, len - 1, ss[1]);
  ret += Change(str, len - 2, ss[0]);
  for (int i = 0; i < len; ++i) {
    if (str[i] != '0') {
      for (int j = i; j > 0; --j) {
        ++ret;
        swap(str[j], str[j - 1]);
      }
      break;
    }
  }
  if (str.substr(len - 2, 2) != ss) {
    return -1;
  } else {
    return ret;
  }
}
int main() {
  ios::sync_with_stdio(false);
  while (cin >> str) {
    ans = INF;
    for (int i = 0; i < 4; ++i) {
      int tmp = solve(str, ss[i]);
      if (tmp != -1) {
        ans = min(tmp, ans);
      }
    }
    if (ans == INF) {
      printf("-1\n");
    } else {
      printf("%d\n", ans);
    }
  }
  return 0;
}
