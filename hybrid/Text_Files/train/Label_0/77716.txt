mas = []
(sumW, maxH, numMaxH, prevMaxH) = (0, 0, 0, 0)
n = int(input())
for i in range(n):
	(w, h) = [int(j) for j in input().split()]
	sumW += w
	mas.append(w)
	if maxH < h:
		prevMaxH = maxH
		maxH = h
		numMaxH = i
	elif prevMaxH < h:
		prevMaxH = h
for i in range(n):
	if i == numMaxH:
		print((sumW - mas[i]) * prevMaxH, end=' ')
	else:
		print((sumW - mas[i]) * maxH, end=' ')
