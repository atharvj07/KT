(n, m, k) = map(int, input().split())
z = list(map(int, input().split()))
import math
ans = [[0, 0, 0] for i in range(len(z))]
for i in range(len(z)):
	if i + 1 <= m:
		for j in range(m):
			total = sum(z[i - j:i + 1])
			if total - k > ans[i][0]:
				ans[i][0] = total - k
				ans[i][1] = j + 1
				ans[i][2] = total
	else:
		for j in range(m + 1):
			if j == 0:
				ans[i][0] = max(z[i] - k, ans[i][0])
				if z[i] - k == ans[i][0]:
					ans[i][1] = 1
					ans[i][2] = z[i]
			else:
				score = ans[i - j][2] + sum(z[i - j + 1:i + 1]) - k * math.ceil((ans[i - j][1] + j) / m)
				if score > ans[i][0]:
					ans[i][0] = score
					ans[i][1] = ans[i - j][1] + j
					ans[i][2] = ans[i - j][2] + sum(z[i - j + 1:i + 1])
maxa = 0
for i in range(len(z)):
	maxa = max(maxa, ans[i][0])
print(maxa)
