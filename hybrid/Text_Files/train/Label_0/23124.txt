#include<bits/stdc++.h>

#define endl '\n'

using namespace std;

int main(){

  int N;

  int h[101];
  
  while(1){
    cin>>N;

 
    if(N==0){
      break;
    }

    int cnt[101]={ };

    for(int i=0;i<N+1;i++){
      cin>>h[i];
    }
      
    
    for(int i=0;i<N+1;i++){
      set<int> S;
      if(i==0){
	for(int j=2;j<N+1;j++){
	  S.insert(h[j] - h[j-1]);
	}
      }else if(i==N){
	for (int j = 1; j < N; ++j) {
	  S.insert(h[j] - h[j-1]);
	}
      } else {    
	for(int j=1;j<N+1;j++){
	  if (j == i) continue;
	  if(j-1==i){
	    S.insert(h[j] - h[j-2]);
	  } else {
	    S.insert(h[j] - h[j-1]);
	  }
	}
	
      }

      if (S.size() == 1) {
	cout << h[i] << endl;
	break;
      }
    }	
	
    

  }

  
  
  return 0;

}
