for _ in range(int(input())):
	n = int(input())
	c = []
	for cave in range(n):
		a = list(map(int, input().split()))
		k = a[0]
		maxx = -1000000000.0
		for mon in range(k):
			maxx = max(a[1 + mon] - mon, maxx)
		c.append((maxx, k))
	c.sort()
	power = c[0][0] + 1
	power += c[0][1]
	ans = c[0][0] + 1
	for i in range(1, n):
		if power < c[i][0] + 1:
			ans += c[i][0] + 1 - power
			power += c[i][0] + 1 - power
		power += c[i][1]
	print(ans)
