#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=2e5+10;
ll ans;
int n;
map<int,int> mp;
int a[N];
int main() {
	int i;
	scanf("%d",&n);
	for (i=1;i<=n;i++) {
		scanf("%d",&a[i]);
		mp[i+a[i]]++;
	}
	for (i=1;i<=n;i++)
		ans=ans+mp[i-a[i]];
	printf("%lld\n",ans);
	return 0;
}
