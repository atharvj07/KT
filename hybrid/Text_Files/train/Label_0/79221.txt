import bisect
for _ in range(int(input())):
	n = int(input())
	ls = []
	ans = 0
	lsl = []
	lsr = []
	for _ in range(n):
		(l, r) = map(int, input().split())
		ls.append([l, r])
		lsl.append(l)
		lsr.append(r)
	lsl.sort()
	lsr.sort()
	cnt = float('inf')
	for i in range(n):
		bad_left = bisect.bisect_left(lsr, ls[i][0])
		bad_right = bisect.bisect_right(lsl, ls[i][1])
		cnt = min(cnt, n - bad_right + bad_left)
	print(cnt)
