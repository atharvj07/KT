n = int(input()) * 2
matriz = [[-float('inf') for i in range(n)] for j in range(n)]
for i in range(1, n):
	a = [int(k) for k in input().split()]
	for j in range(len(a)):
		matriz[i][j] = a[j]
best_i = [-1 for i in range(n)]
pares = []
for i in range(n):
	for j in range(n):
		if matriz[i][j] != -float('inf'):
			pares.append((matriz[i][j], i, j))
pares.sort(key=lambda tup: -tup[0])
dispo = set([int(i) for i in range(n)])
for (valor, i, j) in pares:
	if not i in dispo or not j in dispo:
		continue
	else:
		best_i[i] = j
		best_i[j] = i
		dispo.remove(i)
		dispo.remove(j)
for i in range(n):
	print(best_i[i] + 1, end='')
	if i < n - 1:
		print(end=' ')
print()
