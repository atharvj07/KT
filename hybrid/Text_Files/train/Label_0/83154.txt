from functools import reduce

def solve(n):
	strn = str(n)
	ans = reduce(lambda a, b: a * b, [int(char) for char in strn])
	for i in range(len(strn)):
		if strn[i] == '0':
			continue
		listn = [int(char) for char in strn]
		listn[i] = listn[i] - 1
		for j in range(i + 1, len(strn)):
			listn[j] = 9
		start_i = 0
		if listn[i] == 0 and i == 0:
			start_i = 1
		if start_i >= len(listn):
			continue
		prod = reduce(lambda a, b: a * b, listn[start_i:])
		ans = max(prod, ans)
	return ans
n = int(input())
print(solve(n))
