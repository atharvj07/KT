import java.util.Arrays;
import java.util.Scanner;

public class R477B {
	public static void main(String[] args) {
		Scanner scan=new Scanner(System.in);
		int n=scan.nextInt(), t=scan.nextInt(), d=scan.nextInt();
		int[] a=new int[n];
		int[] frq=new int[10001];
		int first=0, sum=0;
		for(int i=0;i<n;i++) {
			a[i]=scan.nextInt();
			if(i==0) first=a[i];
			frq[a[i]]++;
			sum+=a[i];
		}
		Arrays.sort(a);
		
		int res=0;
		int cur=n-1;
		while(cur>=0&&first*t*1.0/sum+1e-9<d) {
			int x=a[cur];
			if(first==x&&frq[first]<=1) {
				cur--;
				continue;
			}
			sum-=x;
			cur--;
			frq[x]--;
			res++;
		}
		System.out.println(res);
	}
}
