#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n, k, cnt = 0;
  cin >> n >> k;
  vector<int> a(n);
  vector<int> c(k + 1, 0);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    c[a[i]]++;
  }
  long x = (n + 1) / 2;
  priority_queue<int> pq;
  for (int i = 1; i <= k; ++i) pq.push(c[i]);
  while (!pq.empty() && x > 0) {
    int t = pq.top();
    pq.pop();
    if (t > 1) {
      while (t > 1) {
        x--;
        t -= 2;
        cnt += 2;
      }
      pq.push(t);
    } else if (t == 1) {
      x--;
      cnt++;
    }
  }
  cout << cnt;
  return 0;
}
