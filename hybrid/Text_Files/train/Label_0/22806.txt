#include <bits/stdc++.h>
using namespace std;
int main(int argc, char const *argv[]) {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  int pos = -1;
  for (int i = 0; i < n - 1; i++) {
    if (arr[i] < arr[i + 1]) {
      continue;
    } else if (arr[i] > arr[i + 1] || arr[i] == arr[i + 1]) {
      pos = i;
      break;
    }
  }
  if (pos == -1) {
    cout << "YES";
    return 0;
  }
  int temp = -1;
  for (int i = pos; i < n - 1; i++) {
    if (arr[i] == arr[i + 1]) {
      continue;
    } else if (arr[i] > arr[i + 1]) {
      temp = i;
      break;
    } else if (arr[i] < arr[i + 1]) {
      cout << "NO";
      return 0;
    }
  }
  if (temp == -1) {
    cout << "yes";
    return 0;
  }
  for (int i = temp; i < n - 1; i++) {
    if ((arr[i] < arr[i + 1]) || (arr[i] == arr[i + 1])) {
      cout << "NO";
      return 0;
    }
  }
  cout << "YES";
  return 0;
}
