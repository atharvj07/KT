def solve():
	(n, m) = list(map(int, input().split()))
	s = list(input())
	t = list(input())
	(e, l) = ([0] * m, [0] * m)
	(j, i) = (0, 0)
	while j < m:
		if t[j] == s[i]:
			e[j] = i
			j += 1
		i += 1
	(j, i) = (m - 1, n - 1)
	while j >= 0:
		if t[j] == s[i]:
			l[j] = i
			j -= 1
		i -= 1
	best = 0
	for i in range(m - 1):
		best = max(l[i + 1] - e[i], best)
	return best
print(solve())
