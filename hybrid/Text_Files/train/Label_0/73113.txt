def NewColony(n, kk, a):
	i = 0
	for k in range(kk):
		i = 0
		while i < n - 1:
			if a[i] < a[i + 1]:
				a[i] += 1
				break
			i += 1
		if i == n - 1:
			return -1
	return i + 1
t = int(input())
for tt in range(t):
	(n, k) = map(int, input().rstrip().split())
	a = list(map(int, input().rstrip().split()))
	ans = NewColony(n, k, a)
	print(ans)
