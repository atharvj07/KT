int max_digit = 0;
string make(string t)
{
    string ret = "";
    bool flag = true;
    for(int i = 0; i < t.size(); i++)
    {
        if(t[i] >= '0' && t[i] <= '9')
        {
            int j = i;
            int cnt = 0;
            string str = "";
            while(t[j] >= '0' && t[j] <= '9')
            {
                cnt++;
                j++;
                str += t[i];
                i++;
            }
            int tmp = max_digit - cnt;
            //cout << t << " " << cnt << endl;
            while(tmp--) ret += '0';
            ret += str;
            //flag = false;
        }
        //else flag = true;
        ret += t[i];
    }
    return ret;
}
int main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);
    int n;
    cin >> n;
    vector<string> vec;
    for(int i = 0; i <= n; i++)
    {
        string s;
        cin >> s;
        vec.push_back(s);
    }
    int cnt = 0;
    for(int i = 0; i < vec.size(); i++)
    {
        int tmp = 0;
        for(int j = 0; j < vec[i].size(); j++)
        {
            if(vec[i][j] >= '0' && vec[i][j] <= '9') cnt++;
            else cnt = 0;
            tmp = max(cnt, tmp);
        }
        max_digit = max(tmp, max_digit);
    }
    for(int i = 0; i < vec.size(); i++) vec[i] = make(vec[i]);
    for(int i = 1; i < vec.size(); i++)
    {
        bool ok = true; //vec[j] > vec[i]
        bool flag = false;
        for(int j = 0; j < min(vec[0].size(), vec[i].size()); j++)
        {
            if(vec[0][j] == vec[i][j]) continue;
            flag = true;
            if(vec[0][j] < vec[i][j]) break;
            else ok = false;
        }
        if(flag == false && vec[0].size() > vec[i].size()) ok = false;
        //cout << vec[0] << " " << vec[i] << endl;
        cout << (ok ? '+' : '-') << "\n";
    }
