#include <bits/stdc++.h>
using namespace std;
int n;
struct data {
  int x, num, ans;
} a[100];
bool cmp1(data p, data q) { return p.x < q.x; }
bool cmp2(data p, data q) { return p.num < q.num; }
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i].x);
    a[i].num = i;
  }
  sort(a + 1, a + n + 1, cmp1);
  for (int i = 1; i < n; i++) a[i].ans = a[i + 1].x;
  a[n].ans = a[1].x;
  sort(a + 1, a + n + 1, cmp2);
  for (int i = 1; i <= n; i++) printf("%d ", a[i].ans);
  return 0;
}
