import sys, math
n = int(input())
array = list(map(int, input().split()))
if array.count(0) == n:
	print('NO')
	sys.exit()
if sum(array) == 0:
	print('YES')
	print(2)
	for i in range(1, n):
		if sum(array[0:i]) != 0 and sum(array[i:]) != 0:
			print(1, i)
			print(i + 1, n)
			sys.exit()
else:
	print('YES')
	print(1)
	print(1, n)
