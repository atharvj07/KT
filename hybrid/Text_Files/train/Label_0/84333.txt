t = int(input())
for i in range(t):
	(n, s, k) = map(int, input().split())
	a = set(map(int, input().split()))
	l = max(1, s - 1001)
	r = min(n, s + 1001)
	d = 0
	while True:
		if s - d not in a and s - d >= l:
			print(d)
			break
		elif s + d not in a and s + d <= n:
			print(d)
			break
		else:
			d += 1
