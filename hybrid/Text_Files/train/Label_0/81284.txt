def lcmno(n):
	if n % 2 == 0:
		if n % 4 == 0:
			a1 = str(n // 2)
			a2 = str(n // 4)
			return a1 + ' ' + a2 + ' ' + a2
		else:
			a1 = str((n - 2) // 2)
			a2 = '2'
			return a1 + ' ' + a1 + ' ' + a2
	else:
		a = str((n - 1) // 2)
		return '1' + ' ' + a + ' ' + a
t = input()
no = int(t)
s = ''
for i in range(no):
	r2 = input().split(' ')
	s += lcmno(int(r2[0])) + '\n'
print(s)
