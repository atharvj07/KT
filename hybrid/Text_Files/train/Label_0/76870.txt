(n, s) = map(int, input().split())
buy = {}
sell = {}
for i in range(n):
	r = input().split()
	d = r[0]
	p = int(r[1])
	v = int(r[2])
	if d == 'S':
		sell[p] = sell.get(p, 0) + v
	else:
		buy[p] = buy.get(p, 0) + v
for order in sorted(sell.items(), reverse=True)[-s:]:
	print('S %s %s' % (order[0], order[1]))
for order in sorted(buy.items(), reverse=True)[:s]:
	print('B %s %s' % (order[0], order[1]))
