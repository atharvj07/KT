#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100500;
struct Team {
  int ind;
  int a;
  int d;
};
bool cmp(Team a, Team b) { return a.a > b.a || a.a == b.a && a.ind < b.ind; }
int n;
Team team[MAXN];
int main() {
  cin >> n;
  for (int i = 0; i < (int)(n); ++i) {
    cin >> team[i].a >> team[i].d;
    team[i].ind = i;
  }
  int res = 0;
  sort(team, team + n, cmp);
  int iter = -1;
  for (int i = 0; i < (int)(n); ++i)
    if (team[i].d > 0) iter = i;
  while (iter != -1) {
    team[iter].a += team[iter].d;
    team[iter].d = 0;
    int ind = team[iter].ind;
    sort(team, team + n, cmp);
    for (int i = 0; i < (int)(n); ++i)
      if (team[i].ind == ind) res += abs(iter - i);
    iter = -1;
    for (int i = 0; i < (int)(n); ++i)
      if (team[i].d > 0) iter = i;
  }
  sort(team, team + n, cmp);
  iter = -1;
  for (int i = 0; i < (int)(n); ++i)
    if (team[i].d < 0 && iter == -1) iter = i;
  while (iter != -1) {
    team[iter].a += team[iter].d;
    team[iter].d = 0;
    int ind = team[iter].ind;
    sort(team, team + n, cmp);
    for (int i = 0; i < (int)(n); ++i)
      if (team[i].ind == ind) res += abs(iter - i);
    iter = -1;
    for (int i = 0; i < (int)(n); ++i)
      if (team[i].d < 0 && iter == -1) iter = i;
  }
  cout << res;
  return 0;
}
