t = int(input())

def node(x, y, dis):
	for i in range(1, dis + 1):
		if a[x - i][y - i] == 0 or a[x - i][y + i] == 0:
			return 0
	boo[x][y] = 1
	for i in range(1, dis + 1):
		boo[x - i][y - i] = 1
		boo[x - i][y + i] = 1
	return 1

def check(x, y):
	c = 0
	for dis in range(k, n - x):
		if y + dis >= m or y + dis * 2 >= m:
			return
		if a[x + dis][y + dis] == 1:
			if node(x + dis, y + dis, dis) == 0:
				continue
			else:
				c = 1
				continue
	return c

def f():
	for x in range(n):
		for y in range(m):
			if a[x][y]:
				if check(x, y) == 0 and boo[x][y] == 0:
					print('NO')
					return
	for x in range(n):
		for y in range(m):
			if a[x][y] != boo[x][y]:
				print('NO')
				return
	print('YES')
while t:
	(n, m, k) = list(map(int, input().split()))
	a = []
	for i in range(n):
		s = [1 if j == '*' else 0 for j in input()]
		a.append(s)
	boo = [[0 for i in a[j]] for j in range(n)]
	f()
	t -= 1
