

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;



public class BerlandCrossword {

	static class FastScanner 
	{ 
		BufferedReader br; 
		StringTokenizer st; 

		public FastScanner() 
		{ 
			br = new BufferedReader(new
					InputStreamReader(System.in)); 
		} 

		String next() 
		{ 
			while (st == null || !st.hasMoreElements()) 
			{ 
				try
				{ 
					st = new StringTokenizer(br.readLine()); 
				} 
				catch (IOException e) 
				{ 
					e.printStackTrace(); 
				} 
			} 
			return st.nextToken(); 
		} 

		int nextInt() 
		{ 
			return Integer.parseInt(next()); 
		} 

		long nextLong() 
		{ 
			return Long.parseLong(next()); 
		} 

		double nextDouble() 
		{ 
			return Double.parseDouble(next()); 
		} 

		String nextLine() 
		{ 
			String str = ""; 
			try
			{ 
				str = br.readLine(); 
			} 
			catch (IOException e) 
			{ 
				e.printStackTrace(); 
			} 
			return str; 
		} 
	} 

public static void main(String[] args){
      FastScanner f=new FastScanner();
      int t=f.nextInt();
      while(t-->0) {
    	int n=f.nextInt();
    	int U=f.nextInt();
    	int R=f.nextInt();
    	int D=f.nextInt();
    	int L=f.nextInt();
    	boolean canwe=false;
    	for(int i=0;i<16;i++) {
    		int u=U;
    		int l=L;
    		int d=D;
    		int r=R;
    		if((i&1)!=0) {
    			u--;
    			l--;
    		}
    		
    		if((i&2)!=0) {
    			u--;
    			r--;
    		}
    		
    		if((i&4)!=0) {
    			r--;
    			d--;
    		}
    		
    		if((i&8)!=0) {
    			d--;
    			l--;
    		}
    		
    		if((u>=0&&u<=n-2)&&(l>=0&&l<=n-2)&&(d>=0&&d<=n-2)&&(r>=0&&r<=n-2)) {
    			canwe=true;
    			break;
    		}
    		
    	}
    	
    	if(canwe==true) System.out.println("YES");
    	else System.out.println("NO");
      }
}

//BRIAN KERNINGHAM TO CHECK NUMBER OF SET BITS
	//O(LOGn)
	static int setBits(int n) {
		int count=0;
		while(n>0) {
			n=n&(n-1);
			count++;
		}
		return count;
	}
	
	//0 based indexing
	static boolean KthBitSet(int n,int k) {
		int mask=1;
		mask=mask<<=k;
		if((mask&n)!=0) return true;
		else return false;
	}
}
