def compute_gcd(a, b):
	if a == 0:
		return b
	return compute_gcd(b % a, a)

def main():
	(a, b) = map(int, input().split())
	(c, d) = map(int, input().split())
	gcd = compute_gcd(a, c)
	if (b - d) % gcd != 0:
		print(-1)
	steps = 1000
	size = max(b, d) + max(a, c) * steps
	used = [False] * size
	x = b
	for _ in range(steps):
		used[x] = True
		x += a
	x = d
	for _ in range(steps):
		if used[x]:
			print(x)
			break
		x += c
main()
