import java.util.Scanner;

public class Main {
    public static void main(String [] argv){
        Scanner in = new Scanner(System.in);
        int n;
        int [] x = new int [1200];
        while(in.hasNext())
        {
            n = in.nextInt();
            for(int i = 0; i < n; ++i)
                x[i] = in.nextInt();
            int ans = 0;
            for(int i = 0; i < n; ++i)
            {
                if(x[i] == 1)
                    ans ++;
            }
            if(n == 1)
            {
                if(ans == 1)
                    System.out.println("YES");
                else
                    System.out.println("NO");
            }
            else
            {
                if(ans + 1 == n)
                    System.out.println("YES");
                else
                    System.out.println("NO");
            }
        }
        in.close();
    }
}
