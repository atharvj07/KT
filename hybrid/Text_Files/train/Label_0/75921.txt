from collections import defaultdict
for _ in range(int(input())):
	n = int(input())
	st = [[str(i) for i in range(10)] for _ in range(4)]
	d = defaultdict(int)
	l = []
	for _ in range(n):
		s = input()
		l.append(s)
		for i in range(4):
			st[i][int(s[i])] = '-1'
	r = []
	c = 0
	for xi in range(n):
		s = l[xi]
		p = [i for i in s]
		if d[s] == 1:
			c += 1
			f = 0
			for i in range(4):
				for j in range(10):
					if st[i][j] != '-1':
						p[i] = st[i][j]
						st[i][j] = '-1'
						f = 1
						break
				if f == 1:
					break
			x = ''.join(p)
			d[x] = 1
			r.append(x)
		else:
			d[s] = 1
			r.append(s)
	print(c)
	print(*r, sep='\n')
