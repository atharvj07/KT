from sys import stdin
input = stdin.readline
for i in range(int(input())):
	(n, h) = map(int, input().split())
	arr = list(map(int, input().split()))
	arr.sort()
	a = arr[-1]
	b = arr[-2]
	ans = h // (a + b) * 2
	if h % (a + b) > a:
		ans += 2
	elif h % (a + b) == 0:
		ans += 0
	else:
		ans += 1
	print(ans)
