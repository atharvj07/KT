from sys import exit
s = input()
t = input()
d = dict()
used = set()
for i in range(len(s)):
	if s[i] != t[i]:
		if d.get(s[i]) is None and d.get(t[i]) is None and (s[i] not in used) and (t[i] not in used):
			d[s[i]] = t[i]
			d[t[i]] = ''
			used.add(s[i])
			used.add(t[i])
		elif not (d.get(s[i]) == t[i] and d.get(t[i]) == '' or (d.get(t[i]) == s[i] and d.get(s[i]) == '')):
			print(-1)
			exit(0)
	elif d.get(s[i]) != None:
		print(-1)
		exit(0)
	else:
		used.add(s[i])
to_pop = []
for pair in d.items():
	if pair[1] == '':
		to_pop.append(pair[0])
for c in to_pop:
	d.pop(c)
print(len(d))
for pair in d.items():
	print(pair[0], pair[1])
