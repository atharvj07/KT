#include <bits/stdc++.h>

int ri() {
	int n;
	scanf("%d", &n);
	return n;
}
#define MOD 1000000007
int main() {
	int m = ri();
	std::string s;
	std::cin >> s;
	std::reverse(s.begin(), s.end());
	for (auto &c : s) c -= '0';
	int n = s.size();
	if (n == 1) {
		if (m < 10) std::cout << m << std::endl;
		else std::cout << -1 << std::endl;
		return 0;
	}
	int cur = 1;
	int cnt[10] = { 0 };
	for (int i = 0; i < n; i++) {
		cnt[s[i]] += cur;
		if (cnt[s[i]] >= MOD) cnt[s[i]] -= MOD;
		cur = (int64_t) cur * 10 % MOD;
	}
	int pos[10];
	std::iota(pos, pos + 10, 0);
	do {
		if (!pos[s.back()]) continue;
		int x = 0;
		for (int i = 0; i < 10; i++) {
			x += (int64_t) pos[i] * cnt[i] % MOD;
			if (x >= MOD) x -= MOD;
		}
		if (x == m) {
			std::string res;
			for (int i = 0; i < n; i++) res.push_back('0' + pos[s[i]]);
			std::reverse(res.begin(), res.end());
			std::cout << res << std::endl;
			return 0;
		}
	} while (std::next_permutation(pos, pos + 10));
	std::cout << -1 << std::endl;
	return 0;
}
