import java.util.*;
public class Fest {
    static class pair
    {
        int first, second;
        pair(int x, int y)
        {
            first = x;
            second = y;
        }
    }
    public static void main(String[] args) {
        
       Scanner in = new Scanner(System.in);
       
       int k = in.nextInt();
       ArrayList<Integer> a[] = new ArrayList[k];
       ArrayList<Integer> ans[] = new ArrayList[k];
       HashMap<Integer, pair> hm = new HashMap();
       int size[] = new int[k];
       int sum[] = new int[k];
       for(int i=0; i<k; i++)
       {
           size[i] = in.nextInt();
           a[i] = new ArrayList<Integer>();
           for(int j=0; j<size[i]; j++)
               a[i].add(in.nextInt());
           sum[i]=add(a[i], size[i]);
       }
       for(int i=0; i<k; i++)
       {
           int n = size[i];
           ans[i] = new ArrayList<Integer>();
           for(int j=0; j<n; j++)
           {
               ans[i].add(sum[i] - a[i].get(j) );
               int x = ans[i].get(j);
               if(hm.containsKey(x))
               {
                   pair p = hm.get(x);
                   if(p.first != i)
                   {
                       System.out.println("YES");
                       System.out.println((i+1)+" "+(j+1));
                       System.out.println((1+p.first)+" "+(p.second+1));
                       return;
                   }
               }
               else
               {
                   hm.put(x, new pair(i, j));
               }
           }
           
       }
       System.out.println("NO");
    }
    public static int add(ArrayList<Integer> a, int n)
    {
        int sum = 0;
        for(int i=0; i<n; i++)
            sum += a.get(i);
        return sum;
    }
}
