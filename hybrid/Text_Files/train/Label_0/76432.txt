from math import ceil
t = int(input())
while t:
	t -= 1
	n = int(input())
	ab = []
	for i in range(n):
		(a, b) = map(int, input().split())
		ab.append([a, b])
	tm = list(map(int, input().split()))
	(enter, out, min_stay, diff) = (0, 0, 0, ab[0][0])
	for (i, [a, b]) in enumerate(ab):
		if i > 0:
			diff = a - ab[i - 1][1]
		enter = out + diff + tm[i]
		min_stay = ceil((b - a) / 2)
		out = max(b, enter + min_stay)
	print(enter)
