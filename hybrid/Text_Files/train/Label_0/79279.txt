def init():
	inp = input().split()
	k = int(inp[0])
	n = int(inp[1])
	va = [int(i) for i in input().split()]
	vb = [int(i) for i in input().split()]
	return (va, vb)

def pross(va, vb):
	acc = 0
	for (i, a) in enumerate(va):
		acc += a
		va[i] = acc
	vals = [set() for _ in range(len(vb))]
	for (i, b) in enumerate(vb):
		for a in va:
			vals[i].add(b - a)
	return vals

def solve(vals):
	ans = set()
	for (i, v) in enumerate(vals[0]):
		for s in vals[1:]:
			if v not in s:
				break
		else:
			ans.add(v)
	return len(ans)
print(solve(pross(*init())))
