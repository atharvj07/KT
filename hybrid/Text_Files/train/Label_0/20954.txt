#include <bits/stdc++.h>
using namespace std;
int n, pos, l, r;
void input() {
  int i, j;
  scanf("%d %d", &n, &pos);
  scanf("%d %d", &l, &r);
}
void solve() {
  int ans;
  if (l == 1 && r == n) {
    ans = 0;
  } else if (l == 1) {
    ans = abs(pos - r) + 1;
  } else if (r == n) {
    ans = abs(pos - l) + 1;
  } else {
    if (abs(l - pos) < abs(r - pos)) {
      ans = abs(l - pos) + 1 + abs(r - l) + 1;
    } else {
      ans = abs(r - pos) + 1 + abs(r - l) + 1;
    }
  }
  printf("%d", ans);
  puts("");
}
int main() {
  input();
  solve();
  return 0;
}
