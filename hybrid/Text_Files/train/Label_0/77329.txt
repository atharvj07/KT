(n, k, d) = list(map(int, input().split()))
if k ** d < n:
	print(-1)
else:
	ans = []
	for i in range(d):
		ans.append([1 for j in range(n)])
	for i in range(n):
		j = 0
		p = i
		while p > 0:
			ans[j][i] = p % k + 1
			p = p // k
			j = j + 1
			if j == d:
				break
	for i in range(d):
		print(' '.join(map(str, ans[i])))
