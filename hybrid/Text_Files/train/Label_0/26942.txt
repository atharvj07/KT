#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10, mod = 1e9 + 7;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long a, b;
  cin >> a >> b;
  long long sum = 0;
  for (int i = 1; i <= a; i++) {
    long long c1 = b * (b - 1) / 2 % mod;
    long long c2 = (b * i + 1) % mod;
    sum = (sum + c1 * c2 % mod) % mod;
  }
  cout << sum << endl;
  return 0;
}
