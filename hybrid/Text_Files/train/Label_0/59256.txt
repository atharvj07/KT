import java.util.*;
// warm-up
public class RemoveSmallest {

	static int[] a = null;

	static void solve() {
		Scanner sc = new Scanner(System.in);
		int noc = sc.nextInt();
		while (noc-->0) {
			int n = sc.nextInt(),i = 0, tot = 0;
			a = new int[n];
			while (i<n) a[i++]=sc.nextInt();
			boolean ok = true;
			Arrays.sort(a);
			for (i=0; i<a.length-1; i++) {
				if (Math.abs(a[i+1]-a[i])<=1) 	{ tot++; } 
				else 							{ ok=false; break; }
			}
			System.out.println(ok && tot==n-1 ? "YES" : "NO");
		}
		sc.close();
	}

	public static void main(String args[]) {
		solve();
	}

}
