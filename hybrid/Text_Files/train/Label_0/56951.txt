import java.util.Scanner;

public class D {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int r = sc.nextInt();
        int v = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int s = sc.nextInt();
            int f = sc.nextInt();
            double d = f - s;
            double p = 2 * Math.PI * r;
            d %= p;
            double min = 0, max = 2 * Math.PI;
            for (int j = 0; j < 100; j++) {
                double mid = (min + max) / 2;
                double dis = (mid + Math.sin(mid)) * r * 2;
                if (dis < d)
                    min = mid;
                else
                    max = mid;
            }
            double t1 = (f - s - d) / v;
            double t2 = (min + max) * r / v;
            System.out.println(t1 + t2);
        }
        sc.close();
    }
}
