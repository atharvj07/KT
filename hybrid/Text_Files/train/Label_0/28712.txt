#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), l(n);
    for (int &x : a) cin >> x;
    for (int &x : l) cin >> x;
    priority_queue<int> pq;
    for (int i = 0; i < n; ++i) {
      if (l[i]) continue;
      pq.push(a[i]);
    }
    for (int i = 0; i < n; ++i) {
      if (l[i])
        cout << a[i] << " ";
      else {
        int x = pq.top();
        pq.pop();
        cout << x << " ";
      }
    }
    cout << "\n";
  }
}
