#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  vector<bool> check(300);
  vector<double> glas(500003);
  vector<double> sum(500003);
  double ans = 0;
  cin >> s;
  check['I'] = true;
  check['E'] = true;
  check['A'] = true;
  check['Y'] = true;
  check['O'] = true;
  check['U'] = true;
  for (int i = 0; i < s.size(); i++) {
    if (check[s[i]] == true) {
      glas[i + 1] = glas[i] + 1;
    } else {
      glas[i + 1] = glas[i];
    }
  }
  for (int i = 1; i <= s.size(); i++) {
    if (i <= s.size() - i + 1) {
      sum[i] = sum[i - 1] + glas[s.size() - i + 1] - glas[i - 1];
    } else {
      sum[i] = sum[s.size() - i + 1];
    }
    ans += sum[i] / i;
  }
  cout << fixed;
  cout << setprecision(6) << ans;
  return 0;
}
