t = int(input())
while t:
	s = input().split(' ')
	n = int(s[0])
	k = int(s[1])
	s = input()
	v = [-1]
	ans = []
	for i in range(n):
		ans.append(1)
		if s[i] == '0':
			v.append(i)
	l = len(v)
	for i in range(1, l):
		now = max(v[i - 1] + 1, v[i] - k)
		k -= v[i] - now
		v[i] = now
		ans[v[i]] = 0
	for i in range(n):
		if ans[i] > 0:
			print('1', end='')
		else:
			print('0', end='')
	print('')
	t -= 1
