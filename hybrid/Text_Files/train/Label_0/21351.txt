const int _dx[] = {0,1,0,-1};
const int _dy[] = {-1,0,1,0};

int getInt(){
  int ret = 0,c;
  c = getchar();
  while(!isdigit(c)) c = getchar();
  while(isdigit(c)){
    ret *= 10;
    ret += c - '0';
    c = getchar();
  }
  return ret;
}

int p[10][10];
int id[10][10];
int w, h;
int cnt[100];
int all[100];
int g[100][100];
int idcnt;
double cap[100];

void dfs(int y, int x, int ids){
  id[y][x] = ids;

  REP(i,4){
    int yy = y + _dy[i];
    int xx = x + _dx[i];

    if(ISIN(xx, yy, w, h) && id[yy][xx] == -1 && p[yy][xx] == p[y][x]){
      dfs(yy, xx, ids);
    }
  }
}

const double eps = 1e-10;
void doflow(int i, double f){
  if(f < eps) return;
  double rest = (30.0 - cap[i]) * (cnt[i] * 30.0 * 30.0);

  if(rest < f){
    cap[i] = 30.0;
    f -= rest;
  }else{
    cap[i] += f / (cnt[i] * 30.0 * 30.0);
    f = 0.0;
  }

  if(f > eps){
    REP(j,idcnt) if(g[i][j] > 0){
      doflow(j, f * g[i][j] / all[i]);
    }
  }
}

            }else{
            }

    }

    /*
