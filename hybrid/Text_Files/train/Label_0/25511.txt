int main() {
	int n; std::cin >> n;
	int prev = INT_MAX;
	int count = 0;
	for (auto i = 0; i < n; ++i) {
		int a; std::cin >> a;
		if (prev >= a) {
			++count;
		}
		prev = a;
	}
	std::cout << count << '\n' << n << std::endl;
}
