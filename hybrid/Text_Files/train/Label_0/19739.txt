#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
const int maxn = 1e5 + 10;
long long dp[205][3], sum[205][3];
int n, a[maxn];
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  if (a[1] != -1)
    dp[a[1]][0] = 1;
  else
    for (int i = 1; i <= 200; i++) dp[i][0] = 1;
  for (int i = 2; i <= n; i++) {
    for (int j = 1; j <= 200; j++) {
      sum[j][0] = (sum[j - 1][0] + dp[j][0]) % mod;
      sum[j][1] = (sum[j - 1][1] + dp[j][1]) % mod;
      sum[j][2] = (sum[j - 1][2] + dp[j][2]) % mod;
    }
    if (a[i] == -1) {
      for (int j = 1; j <= 200; j++) {
        dp[j][2] = (dp[j][0] + dp[j][1] + dp[j][2]) % mod;
        dp[j][0] = (sum[j - 1][0] + sum[j - 1][1] + sum[j - 1][2]) % mod;
        dp[j][1] = ((sum[200][1] - sum[j][1] + mod) % mod +
                    (sum[200][2] - sum[j][2] + mod) % mod) %
                   mod;
      }
    } else {
      int j = a[i];
      dp[j][2] = (dp[j][0] + dp[j][1] + dp[j][2]) % mod;
      dp[j][0] = (sum[j - 1][0] + sum[j - 1][1] + sum[j - 1][2]) % mod;
      dp[j][1] = ((sum[200][1] - sum[j][1] + mod) % mod +
                  (sum[200][2] - sum[j][2] + mod) % mod) %
                 mod;
      for (j = 1; j <= 200; j++)
        if (j != a[i]) dp[j][0] = dp[j][1] = dp[j][2] = 0;
    }
  }
  for (int j = 1; j <= 200; j++) {
    sum[j][1] = (sum[j - 1][1] + dp[j][1]) % mod;
    sum[j][2] = (sum[j - 1][2] + dp[j][2]) % mod;
  }
  cout << (sum[200][1] + sum[200][2]) % mod;
  return 0;
}
