import collections
import math

def readLine():
	return list(map(int, input().strip().split()))

def readInt():
	return int(input())

def readString():
	return input()

def tcase():
	t = readInt()
	for _ in range(t):
		solve()

def solve():
	(n, m) = readLine()
	f = [0] * (n * m + 1)
	for i in range(2, n * m + 1):
		f[i] = f[i - 1] + math.log(i)

	def c(n, k):
		return f[n] - f[k] - f[n - k]
	ans = 0
	for num in range(1, min(n, m) + 1):
		temp = c(m, num) + c(m * (n - 1), n - num) - c(m * n, n)
		pro = num * num * 1.0 / (n * 1.0)
		ans += pro * math.pow(math.e, temp)
	print(ans)
solve()
