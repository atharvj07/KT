#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i = 2;
  cin >> n;
  if (n == 1) {
    cout << 1 << endl;
    return 0;
  } else {
    cout << n << " ";
  }
  while (n != 1) {
    if (n % i == 0) {
      cout << n / i << " ";
      n = n / i;
      i = 2;
    } else {
      i++;
    }
  }
  return 0;
}
