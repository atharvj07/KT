import java.io.*;
import java.util.*;

public class CF84E {
	static class P {
		boolean visited;
		P p;
		int i, j;
		P(int i, int j) {
			this.i = i;
			this.j = j;
		}
	}
	static int n, m, k, is, js, it, jt;
	static char[][] aa;
	static String s;
	static P separator = new P(-1, -1);
	static P[][] pp, qq;
	static int[] ll;
	static String trace(P p) {
		return p.p == null ? "" : trace(p.p) + aa[p.i][p.j];
	}
	static void update(P p) {
		String t = trace(p);
		if (s == null)
			s = t;
		else {
			int ls = s.length(), lt = t.length();
			if (ls > lt || ls == lt && s.compareTo(t) > 0)
				s = t;
		}
	}
	static void bfs(char[] cc) {
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) {
				P p = pp[i][j];
				p.visited = false;
				p.p = null;
			}
		LinkedList<P> queue = new LinkedList<>();
		pp[is][js].visited = true;
		queue.addLast(pp[is][js]);
		queue.addLast(separator);
		P p, q;
		while (!queue.isEmpty()) {
			for (int c = 0; c < k; c++)
				ll[c] = 0;
			while ((p = queue.removeFirst()) != separator) {
				int[] ii = { p.i - 1, p.i + 1, p.i, p.i };
				int[] jj = { p.j, p.j, p.j - 1, p.j + 1 };
				for (int h = 0; h < 4; h++) {
					int i = ii[h];
					int j = jj[h];
					if (i == it && j == jt) {
						update(p);
						return;
					}
					if (i < 0 || i >= n || j < 0 || j >= m)
						continue;
					q = pp[i][j];
					if (!q.visited) {
						q.visited = true;
						for (int c = 0; c < k; c++)
							if (aa[i][j] == cc[c]) {
								q.p = p;
								qq[c][ll[c]++] = q;
								break;
							}
					}
				}
			}
			for (int c = 0; c < k; c++)
				if (ll[c] > 0) {
					for (int l = 0; l < ll[c]; l++)
						queue.addLast(qq[c][l]);
					queue.addLast(separator);
				}
		}
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		n = Integer.parseInt(st.nextToken());
		m = Integer.parseInt(st.nextToken());
		k = Integer.parseInt(st.nextToken());
		aa = new char[n][m];
		for (int i = 0; i < n; i++)
			br.readLine().getChars(0, m, aa[i], 0);
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++)
				if (aa[i][j] == 'S') {
					is = i;
					js = j;
				} else if (aa[i][j] == 'T') {
					it = i;
					jt = j;
				}
		pp = new P[n][m];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++)
				pp[i][j] = new P(i, j);
		qq = new P[k][n * m];
		ll = new int[k];
		char[] cc = new char[k];
		for (char a = 'a'; a <= 'z'; a++) {
			cc[0] = a;
			if (k == 1) {
				bfs(cc);
				continue;
			}
			for (char b = (char) (a + 1); b <= 'z'; b++) {
				cc[1] = b;
				if (k == 2) {
					bfs(cc);
					continue;
				}
				for (char c = (char) (b + 1); c <= 'z'; c++) {
					cc[2] = c;
					if (k == 3) {
						bfs(cc);
						continue;
					}
					for (char d = (char) (c + 1); d <= 'z'; d++) {
						cc[3] = d;
						bfs(cc);
					}
				}
			}
		}
		System.out.println(s == null ? -1 : s);
	}
}
