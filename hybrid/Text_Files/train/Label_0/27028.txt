#include <bits/stdc++.h>
using namespace std;
int main() {
  double s;
  int a, b, c;
  cin >> s >> a >> b >> c;
  if (a == 0 && b == 0 && c == 0) {
    cout << "0 0 0\n";
  } else
    printf("%.16f %.16f %.16f\n", 1.0 * s * a / (a + b + c),
           1.0 * s * b / (a + b + c), 1.0 * s * c / (a + b + c));
}
