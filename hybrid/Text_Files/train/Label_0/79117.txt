n = int(input())
ans = []
l = [int(i) for i in input().split()]
from collections import defaultdict
d = defaultdict(list)
for i in range(n):
	sm = 0
	for j in range(i, n):
		sm += l[j]
		d[sm].append([i, j])
for sm in d:
	z = d[sm]
	z.sort(key=lambda x: x[1])
	t = [z[0]]
	for i in z:
		if i[0] <= t[-1][1]:
			continue
		t.append(i)
	if len(t) > len(ans):
		ans = t
print(len(ans))
for i in range(len(ans)):
	print(ans[i][0] + 1, ans[i][1] + 1)
