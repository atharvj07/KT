#include <bits/stdc++.h>
using namespace std;
const int N = 4e5 + 10;
const int mod = 1e9 + 7;
int cnt1[N], cnt2[N];
int main() {
  string s;
  cin >> s;
  int i;
  s = " " + s;
  int n = (int)s.size() - 1;
  for (i = 1; i < (int)s.size(); i++) {
    if (s[i] >= 'a' && s[i] <= 'z')
      cnt1[i] = cnt1[i - 1] + 1;
    else
      cnt1[i] = cnt1[i - 1];
    if (s[i] >= 'A' && s[i] <= 'Z')
      cnt2[i] = cnt2[i - 1] + 1;
    else
      cnt2[i] = cnt2[i - 1];
  }
  int ans = 1e9;
  for (i = 0; i <= n; i++) {
    ans = min(ans, cnt1[i] + cnt2[n] - cnt2[i]);
  }
  cout << ans << endl;
}
