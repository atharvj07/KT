from collections import defaultdict as df
t = int(input())
for z in range(t):
	s = input()
	visited = df(lambda : -1)
	count = 0
	for i in range(len(s)):
		if visited[s[i]] != -1:
			x = visited[s[i]]
			if i - x <= 2:
				count += 1
			else:
				visited[s[i]] = i
		else:
			visited[s[i]] = i
	print(count)
