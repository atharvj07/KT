def nc3(n):
	if n < 3:
		return 0
	else:
		return n * (n - 1) * (n - 2) // 6

def nc2(n):
	if n < 2:
		return 0
	else:
		return n * (n - 1) // 2
t = int(input())
for test in range(t):
	n = int(input())
	List = list(map(int, input().rstrip().split()))
	if n < 3:
		print(0)
	else:
		array = [0 for i in range(n)]
		for item in List:
			array[item - 1] += 1
		ans = 0
		for item in array:
			ans += nc3(item)
		for i in range(n - 1):
			ans += nc2(array[i]) * array[i + 1] + array[i] * nc2(array[i + 1])
		for i in range(n - 2):
			ans += array[i] * array[i + 1] * array[i + 2] + (nc2(array[i]) * array[i + 2] + array[i] * nc2(array[i + 2]))
		print(ans)
