tim = input()
m = int(input())
gh = m // 60
gm = m % 60
timsplit = tim.split(':')
hours = [i for i in range(0, 24)]
mins = [i for i in range(0, 60)]
ms = int(timsplit[1])
hs = int(timsplit[0])
if ms + gm >= 60:
	hs = (hs + gh + 1) % 24
	ms = (ms + gm) % 60
else:
	hs = (hs + gh) % 24
	ms = ms + gm
if ms < 10:
	ms = '0' + str(ms)
if hs < 10:
	hs = '0' + str(hs)
print(str(hs) + ':' + str(ms))
