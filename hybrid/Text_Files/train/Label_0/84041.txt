n = int(input())
lis = list(input())
if n % 2:
	print(0)
else:
	st = []
	for i in range(n):
		if lis[i] == ')' and len(st) > 0 and (st[-1] == '('):
			st.pop()
		else:
			st.append(lis[i])
	if len(st) == 2 and st[0] == st[1]:
		if st[0] == '(':
			pre = [0] * (n + 2)
			ans = 0
			for i in range(n):
				if lis[i] == '(':
					pre[i] = pre[i - 1] + 1
				else:
					pre[i] = pre[i - 1] - 1
			c = 0
			for i in range(n - 1, -1, -1):
				if pre[i] <= 1:
					c = i
					break
			for i in range(c + 1, n, 1):
				if pre[i] > pre[i - 1]:
					ans += 1
			print(ans)
		else:
			suf = [0] * (n + 10)
			ans = 0
			for i in range(n - 1, -1, -1):
				if lis[i] == ')':
					suf[i] = suf[i + 1] + 1
				else:
					suf[i] = suf[i + 1] - 1
			c = 0
			for i in range(n):
				if suf[i] <= 1:
					c = i
					break
			for i in range(c):
				if suf[i] > suf[i + 1]:
					ans += 1
			print(ans)
	else:
		print(0)
