def getOpen(i, st):
	for j in range(i, len(st)):
		if st[j] == '(':
			r = st[j]
			st[j] = st[i]
			st[i] = r
			return j

def getClose(i, st):
	for j in range(i, len(st)):
		if st[j] == ')':
			r = st[j]
			st[j] = st[i]
			st[i] = r
			return j

def test(testStr, k, ans):
	ans = []
	for i in range(k * 2):
		if i % 2 == 0:
			ans.append([i + 1, getOpen(i, testStr) + 1])
		else:
			ans.append([i + 1, getClose(i, testStr) + 1])
	countEteration = (len(testStr) - k * 2) // 2
	for i in range(k * 2, k * 2 + countEteration):
		ans.append([i + 1, getOpen(i, testStr) + 1])
	return ans
t = int(input())
for i in range(t):
	(n, k) = map(int, input().split())
	testStr = input()
	ans = test(list(testStr), k - 1, [])
	print(len(ans))
	for i in range(len(ans)):
		print(' '.join(list(map(str, ans[i]))))
