for _ in range(int(input())):
	(k, x) = map(int, input().split())
	ans = 0
	t = k * (k + 1) // 2
	if t <= x:
		ans += k
		x -= t
	else:
		y = int((2 * x) ** 0.5)
		ans += y
		x -= y * (y + 1) // 2
		while x > 0:
			ans += 1
			y += 1
			x -= y
		x = 0
	if x:
		t = k * (k - 1) // 2
		if t <= x:
			ans += k - 1
		else:
			p = k * (k - 1) - 2 * x
			y = k - int(p ** 0.5) - 1
			p = t - (k - y) * (k - y - 1) // 2
			ans += y
			x -= p
			y += 1
			while x > 0:
				ans += 1
				x -= k - y
				y += 1
	print(ans)
