t = int(input())
out = []
for _ in range(t):
	(n, k) = map(int, input().split())
	res = [1] * (k - 3)
	n = n - k + 3
	if n % 2 == 0:
		if n % 4 == 0:
			res = res + [n // 2, n // 4, n // 4]
		else:
			res = res + [n // 2 - 1, n // 2 - 1, 2]
	else:
		res = res + [n // 2, n // 2, 1]
	out.append(res)
for i in out:
	print(*i)
