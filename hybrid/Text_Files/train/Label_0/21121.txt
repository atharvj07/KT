#include <bits/stdc++.h>
using namespace std;
int ara[2005];
set<int> st;
map<int, int> mp, freq;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, i, j;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> ara[i];
    mp[ara[i]]++;
    st.insert(ara[i]);
  }
  if ((int)st.size() == n) {
    cout << 0 << endl;
    return 0;
  }
  int ans = 1e9;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= n; j++) {
      st.insert(ara[j]);
      freq[ara[j]] = mp[ara[j]];
    }
    for (j = i; j <= n; j++) {
      freq[ara[j]]--;
      if (freq[ara[j]] == 0) {
        st.erase(st.find(ara[j]));
      }
      int len = j - i + 1;
      if ((int)st.size() == n - len) {
        ans = min(ans, len);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
