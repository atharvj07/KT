#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  string name, ip;
  map<string, string> s1;
  while (cin >> n >> m) {
    for (int i = 0; i < n; i++) {
      cin >> name >> ip;
      ip = ip + ";";
      s1[ip] = name;
    }
    for (int i = 0; i < m; i++) {
      cin >> name >> ip;
      cout << name << " " << ip << " ";
      cout << "#" << s1[ip] << endl;
    }
  }
  return 0;
}
