#include<cstdio>
#include<vector>
#include<cmath>
#include<algorithm>

using namespace std;

typedef double Real;

const Real eps=1e-8;

int N;
int xs[5],ys[5];

Real getLen(Real h){
	vector<Real> vals;
	for(int i=0;i<N;i++){
		int x1=xs[i],y1=ys[i];
		int x2=xs[i+1],y2=ys[i+1];
		if(y1==y2) continue;
		if(y1>y2){
			swap(y1,y2);
			swap(x1,x2);
		}
		if(h<y1||y2<h) continue;
		Real r=(Real)(h-y1)/(y2-y1);
		Real x=x1+(Real)(x2-x1)*r;
		vals.push_back(x);
	}
	sort(vals.begin(),vals.end());
	Real res=0;
	for(int i=0;i+1<vals.size();i+=2){
		res+=vals[i+1]-vals[i];
	}
//	printf("h=%f,res=%f\n",h,res);
	return res;
}

vector<Real> hs;

Real solve(){
	Real ans=0;
	for(int i=0;i<N;i++){
		hs.push_back(ys[i]+eps);
		hs.push_back(ys[i]-eps);
	}
	sort(hs.begin(),hs.end());
	vector<int> same_y;
	for(int i=0;i<N;i++){
		if(ys[i]==ys[i+1]){
			same_y.push_back(ys[i]);
			continue;
/*			Real len1=getLen(ys[i]-eps);
			Real len2=getLen(ys[i]+eps);
//			printf("+=(%f,%f)->%f\n",len1,len2,abs(len1*len1-len2*len2));
			ans+=abs(len1*len1-len2*len2);*/
		}else{
			int x1=xs[i],y1=ys[i];
			int x2=xs[i+1],y2=ys[i+1];
			if(y1>y2){
				swap(y1,y2);
				swap(x1,x2);
			}
			Real h=sqrt((x2-x1)*(x2-x1)+(y2-y1)*(y2-y1));
			for(int j=0;j+1<hs.size();j++){
				Real lo=hs[j],hi=hs[j+1];
				if(lo<y1||y2<hi) continue;
				Real len1=getLen(lo);
				Real len2=getLen(hi);
				Real cur_h=h*(hi-lo)/(y2-y1);
				ans+=(len1+len2)*cur_h;
			}
		}
	}
	sort(same_y.begin(),same_y.end());
	same_y.erase(unique(same_y.begin(),same_y.end()),same_y.end());
	for(int i=0;i<same_y.size();i++){
		int y=same_y[i];
		Real len1=getLen(y-eps);
		Real len2=getLen(y+eps);
		ans+=abs(len1*len1-len2*len2);
	}
	return ans;
}

int main(){
	while(true){
		scanf("%d",&N);
		if(N==0) break;
		for(int i=0;i<N;i++) scanf("%d%d",xs+i,ys+i);
		xs[N]=xs[0],ys[N]=ys[0];
		Real ans=solve();
		printf("%f\n",ans);
	}
	return 0;
}
