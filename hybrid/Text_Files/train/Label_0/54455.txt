import java.io.*;
import java.util.*;
import java.math.*;

public class Main implements Runnable {
	final int UB = 1000000000;
        
	ArrayList<Integer> tmp_lucky = new ArrayList<Integer>();
	int lucky[], n;

	void gen(long cur) {
		if (cur > UB) return;
		tmp_lucky.add((int)cur); gen(10*cur+4); gen(10*cur+7);
	}

	class Node {
		int ch[], w[], ss = 1;
		Node(int nch) { ch = new int[nch]; w = new int[nch]; }
	}
        
	Node tree[];
	long answer1[], answer2[];
        
	void run1(int u, int par) {
		for (int e = 0; e < tree[u].ch.length; ++e) {
			int v = tree[u].ch[e]; if (v == par) continue; run1(v, u);

			if (Arrays.binarySearch(lucky, tree[u].w[e]) >= 0)
				answer1[u] += tree[v].ss;
			else
				answer1[u] += answer1[v];
                        
			tree[u].ss += tree[v].ss;
		}
	}

	void run2(int u, int par) {
		for (int e = 0; e < tree[u].ch.length; ++e) {
			int v = tree[u].ch[e]; if (v == par) continue;

			if (Arrays.binarySearch(lucky, tree[u].w[e]) >= 0)
				answer2[v] = n-tree[v].ss;
			else
				answer2[v] = answer1[u]-answer1[v]+answer2[u];

			run2(v, u);
		}
	}

	public void run() {
		gen(4); gen(7); Collections.sort(tmp_lucky);

		lucky = new int[tmp_lucky.size()];
		for (int i = 0; i < lucky.length; ++i) lucky[i] = tmp_lucky.get(i);

		n = nextInt();
		int u[] = new int[n-1], v[] = new int[n-1], w[] = new int[n-1], deg[] = new int[n];

		for (int i = 0; i < n-1; ++i) {
			u[i] = nextInt()-1; v[i] = nextInt()-1; w[i] = nextInt();
			++deg[u[i]]; ++deg[v[i]];
		}

		tree = new Node[n]; answer1 = new long[n]; answer2 = new long[n];

		for (int i = 0; i < n; ++i) tree[i] = new Node(deg[i]);
		for (int i = 0; i < n-1; ++i) {
			tree[u[i]].ch[--deg[u[i]]] = v[i]; tree[u[i]].w[deg[u[i]]] = w[i];
			tree[v[i]].ch[--deg[v[i]]] = u[i]; tree[v[i]].w[deg[v[i]]] = w[i];
		}

		run1(0, -1); run2(0, -1);

		long res = 0;
		for (int i = 0; i < n; ++i)
			res += (answer1[i] + answer2[i]) * (answer1[i] + answer2[i] - 1);
		
		output.println(res);
		output.flush();
	}

    public static void main(String[] args) {
        //try { input = new BufferedReader(new InputStreamReader(new FileInputStream("input.txt"))); } catch (Exception e) {}
        new Main().run();
    }

    static BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
    PrintWriter output = new PrintWriter(System.out);
    StringTokenizer tokenizer = null;

    int nextInt() {
        return Integer.parseInt(nextToken());
    }

    long nextLong() {
        return Long.parseLong(nextToken());
    }

    double nextDouble() {
        return Double.parseDouble(nextToken());
    }

    String nextToken() {
        try
        {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                tokenizer = new StringTokenizer(input.readLine());
            }

            return tokenizer.nextToken();
        } catch (Exception e) {
            return null;
        }
    }
}
