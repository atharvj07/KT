(r, c) = map(int, input().split())
Map = []
for _ in range(r):
	x = input().replace('.', 'D')
	Map.append('#' + x + '#')
Map = ['#' * (c + 2)] + Map + ['#' * (c + 2)]
for i in range(r + 2):
	for j in range(c + 2):
		if Map[i][j] == 'W':
			if Map[i - 1][j] == 'S' or Map[i + 1][j] == 'S' or Map[i][j - 1] == 'S' or (Map[i][j + 1] == 'S'):
				print('NO')
				exit()
print('YES')
for i in range(1, r + 1):
	print(Map[i][1:-1])
