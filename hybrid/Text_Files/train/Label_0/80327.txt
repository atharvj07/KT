(N, M) = map(int, input().split())
A = list(map(int, input().split()))
d = {k: 0 for k in range(1, M + 1)}
indices = []
ans1 = 0
ans2 = 0
for (i, a) in enumerate(A):
	if a <= M:
		d[a] += 1
	else:
		indices.append(i)
scores = [[k, v] for (k, v) in d.items()]
scores.sort(key=lambda x: x[1])
div = N // M
ind = 0
while scores[0][1] < div:
	if ind == len(indices):
		break
	scores[0][1] += 1
	A[indices[ind]] = scores[0][0]
	ind += 1
	scores.sort(key=lambda x: x[1])
	ans2 += 1
sum_score = sum([s[1] for s in scores])
mean = sum_score // M
while scores[0][1] < mean:
	A[A.index(scores[-1][0])] = scores[0][0]
	scores[0][1] += 1
	scores[-1][1] -= 1
	scores.sort(key=lambda x: x[1])
	ans2 += 1
print('{} {}'.format(scores[0][1], ans2))
print(' '.join(map(str, A)))
