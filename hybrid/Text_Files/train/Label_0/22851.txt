#include <bits/stdc++.h>
using namespace std;
void Fast() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
}
void File() {}
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
  Fast(), File();
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  vector<int> v1, v2;
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
    v1.push_back(v[i].first), v2.push_back(v[i].second);
  }
  sort(v1.begin(), v1.end()), sort(v2.begin(), v2.end());
  int maxmm = INT_MIN, max1, min2;
  for (int i = 0; i < n; i++) {
    max1 = v1[n - 1], min2 = v2[0];
    if (v[i].first == max1) max1 = v1[n - 2];
    if (v[i].second == min2) min2 = v2[1];
    maxmm = max(maxmm, min2 - max1);
  }
  cout << max(maxmm, 0) << "\n";
}
