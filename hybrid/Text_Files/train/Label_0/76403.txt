for _ in [0] * int(input()):
	n = int(input())
	s = [int(a) for a in input().split()]
	flag = False
	ct = 0
	for i in s:
		ct += 1
		if i != 1:
			ans = ct % 2
			flag = True
			break
	if flag == True:
		if ans % 2 == 0:
			print('Second')
		else:
			print('First')
	elif flag == False:
		if n % 2 == 0:
			print('Second')
		else:
			print('First')
