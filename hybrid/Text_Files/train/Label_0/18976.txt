#include <bits/stdc++.h>
using namespace std;
string s;
vector<string> q;
int dig[10], n, Mult[10];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> s;
  cin >> n;
  for (int i = 0; i < (int)n; i++) {
    string tmp;
    cin >> tmp;
    q.push_back(tmp);
  }
  for (int i = 0; i < (int)10; i++) {
    dig[i] = i;
    Mult[i] = 10;
  }
  for (int i = n - 1; i >= 0; i--) {
    long long mult = 1, tmp = 0;
    int d = q[i][0] - '0', len = (int)q[i].size();
    for (int j = len - 1; j > 2; j--) {
      tmp += dig[q[i][j] - '0'] * mult;
      tmp %= 1000000007;
      mult = (mult * Mult[q[i][j] - '0']) % 1000000007;
    }
    dig[d] = tmp;
    Mult[d] = mult;
  }
  long long mult = 1, ret = 0;
  int len = (int)s.size();
  for (int i = len - 1; i >= 0; i--) {
    ret += dig[s[i] - '0'] * mult;
    ret %= 1000000007;
    mult = (mult * Mult[s[i] - '0']) % 1000000007;
  }
  cout << ret << endl;
  return 0;
}
