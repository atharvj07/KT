#include <bits/stdc++.h>
using namespace std;
int freq[256];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < (int)n; ++i) {
    char a;
    cin >> a;
    ++freq[a];
    if (k < freq[a]) {
      cout << "nO" << '\n';
      return 0;
    }
  }
  cout << "yES" << '\n';
  return 0;
}
