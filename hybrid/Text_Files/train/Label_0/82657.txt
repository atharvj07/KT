n = int(input())
arr = list(map(int, input().split()))
d = {}
flag = True
for i1 in range(n):
	for j1 in range(i1):
		if arr[i1] + arr[j1] in d:
			(i2, j2) = d[arr[i1] + arr[j1]]
			if len(set([i1, i2, j1, j2])) == 4:
				print('YES')
				print(j2 + 1, i2 + 1, j1 + 1, i1 + 1)
				flag = False
				break
		else:
			d[arr[i1] + arr[j1]] = (i1, j1)
	if flag == False:
		break
if flag:
	print('NO')
