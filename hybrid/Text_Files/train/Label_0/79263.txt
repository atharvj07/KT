from copy import deepcopy
import sys
import math
for _ in range(int(sys.stdin.readline())):
	n = int(sys.stdin.readline())
	a = [int(x) for x in sys.stdin.readline().split()]
	m = math.ceil(math.log(n, 2)) + 1
	arrays = []
	occur = {}
	arrays.append(a)
	for i in a:
		if i in occur:
			occur[i] += 1
		else:
			occur[i] = 1
	for __ in range(m):
		cur_a = []
		for i in range(n):
			cur_a.append(occur[arrays[-1][i]])
		occur = {}
		for i in cur_a:
			if i in occur:
				occur[i] += 1
			else:
				occur[i] = 1
		arrays.append(cur_a)
	for __ in range(int(sys.stdin.readline())):
		(x, k) = map(int, sys.stdin.readline().split())
		if k > m:
			print(arrays[m][x - 1])
		else:
			print(arrays[k][x - 1])
