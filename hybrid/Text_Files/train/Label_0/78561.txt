def main():
	t = int(input())
	allans = []
	for _ in range(t):
		n = int(input())
		adj = [[] for _ in range(n + 1)]
		for __ in range(n - 1):
			(u, v) = readIntArr()
			adj[u].append(v)
			adj[v].append(u)
		par = [-1 for __ in range(n + 1)]
		childCnts = [0 for __ in range(n + 1)]
		depth = [-1 for __ in range(n + 1)]
		depth[1] = 0
		stack = [(1, -1)]
		while stack:
			(node, p) = stack.pop()
			par[node] = p
			for nex in adj[node]:
				if nex != p:
					childCnts[node] += 1
					stack.append((nex, node))
					depth[nex] = depth[node] + 1
		nodeDepths = [(depth[node], node) for node in range(1, n + 1)]
		nodeDepths.sort(reverse=True)
		pruned = [False] * (n + 1)
		budCnts = 0
		for (depth, node) in nodeDepths:
			if depth >= 2 and pruned[node] == False and (pruned[par[node]] == False):
				budCnts += 1
				twoUp = par[par[node]]
				childCnts[twoUp] -= 1
				pruned[par[node]] = True
		directLeaves = childCnts[1]
		budContribution = n - 1 - directLeaves - budCnts * 2
		leavesWithoutBud = max(1, directLeaves)
		ans = leavesWithoutBud + budContribution
		allans.append(ans)
	multiLineArrayPrint(allans)
	return

def oneLineArrayPrint(arr):
	print(' '.join([str(x) for x in arr]))

def multiLineArrayPrint(arr):
	print('\n'.join([str(x) for x in arr]))

def multiLineArrayOfArraysPrint(arr):
	print('\n'.join([' '.join([str(x) for x in y]) for y in arr]))

def readIntArr():
	return [int(x) for x in input().split()]

def makeArr(defaultValFactory, dimensionArr):
	dv = defaultValFactory
	da = dimensionArr
	if len(da) == 1:
		return [dv() for _ in range(da[0])]
	else:
		return [makeArr(dv, da[1:]) for _ in range(da[0])]
