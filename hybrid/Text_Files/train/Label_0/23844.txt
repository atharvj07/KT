#include <bits/stdc++.h>
using namespace std;
const int maxn = 2100;
long long arr[maxn], dp[maxn];
long long n, k;
bool check(long long x) {
  for (int i = 1; i <= n; i++) {
    dp[i] = i - 1;
    for (int j = 1; j < i; j++) {
      if (abs(arr[i] - arr[j]) <= x * (i - j))
        dp[i] = min(dp[i], dp[j] + (i - j) - 1);
    }
    if (dp[i] + n - i <= k) return true;
  }
  return false;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> arr[i];
  }
  long long index = INT_MAX;
  for (long long cekor = INT_MAX / 2; cekor > 0; cekor /= 2) {
    while (index - cekor >= 0 && check(index - cekor)) index -= cekor;
  }
  cout << index << "\n";
  return 0;
}
