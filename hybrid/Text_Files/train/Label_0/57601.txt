import java.util.*;

public class Main {

  public static void main(String args[]) {
    (new Main()).exec();
  }

  void exec() {

    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt();

    boolean table[] = new boolean[3010];

    for(int i=0; i<n; ++i) {
      table[cin.nextInt()] = true;
    }

    int p = 1;
    while( table[p] ) { ++p; }

    System.out.println(p);

  }

}
