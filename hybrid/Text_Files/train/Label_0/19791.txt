#include <bits/stdc++.h>
int main() {
  int n;
  scanf("%d", &n);
  int x[n];
  int y[n];
  long long int ans = 0;
  long long int equals = 0;
  int soma[2001] = {0};
  int sub[1001] = {0};
  int sub2[1001] = {0};
  for (int i = 0; i < n; i++) {
    scanf("%d %d", &x[i], &y[i]);
    if (x[i] == y[i]) {
      equals++;
    } else {
      if (x[i] > y[i]) {
        sub[x[i] - y[i]]++;
      } else {
        sub2[y[i] - x[i]]++;
      }
    }
    soma[x[i] + y[i]] += 1;
  }
  for (int i = 0; i < 2001; i++) {
    if (soma[i] >= 2) {
      ans += (soma[i] * (soma[i] - 1)) / 2;
    }
  }
  for (int i = 0; i < 1001; i++) {
    if (sub[i] >= 2) {
      ans += (sub[i] * (sub[i] - 1)) / 2;
    };
  }
  for (int i = 1; i < 1001; i++) {
    if (sub2[i] >= 2) {
      ans += sub2[i] * (sub2[i] - 1) / 2;
    }
  }
  if (equals >= 2) {
    ans += (equals * (equals - 1)) / 2;
  };
  printf("%lld\n", ans);
  return 0;
}
