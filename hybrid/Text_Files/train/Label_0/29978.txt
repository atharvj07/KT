#include <bits/stdc++.h>
#define INF 1e9
using namespace std;
int n,arr[101];
int cal(int a,int b){return (arr[a]+arr[b])*(abs(a-b));}
void Min(int &a,int b){a=min(a,b);}

int mem[3][101][101][101];
int dfs(int f,int num,int l,int r){
  if(l>r)return 0;
  if(l==r)return (arr[l]==l+1) ? INF:0;
  if(r-l==1)return (arr[l]==l+1||arr[r]==r+1)? cal(l,r):0;
  if(mem[f][num][l][r]!=-1)return mem[f][num][l][r];

  int res=INF;  
  for(int i=l+1;i<r;i++){
    if(f!=2){
      swap(arr[l],arr[i]);
      Min(res,cal(l,i)+dfs(1,arr[l],l,i-1)+dfs(1,arr[i],i,r));
      swap(arr[l],arr[i]);
    }
    if(f!=1){
      swap(arr[i],arr[r]);
      Min(res,cal(i,r)+dfs(2,arr[i],l,i)+dfs(2,arr[r],i+1,r));
      swap(arr[i],arr[r]);
    }
  }

  for(int i=l;i<r;i++){
    int a=(f==1)? dfs(f,num,l,i):dfs(0,0,l,i);
    int b=(f==2)? dfs(2,num,i+1,r):dfs(0,0,i+1,r);
    Min(res,a+b);
  }

  return mem[f][num][l][r]=res;
}

int main(){
  cin>>n;
  for(int i=0;i<n;i++)cin>>arr[i];
  memset(mem,-1,sizeof(mem));
  cout<<dfs(0,0,0,n-1)<<endl;
}
