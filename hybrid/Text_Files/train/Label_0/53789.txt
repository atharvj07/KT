import java.util.*;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
 
public class temp
{
	void solve()
	{
		FastReader sc = new FastReader();
		int n = sc.nextInt();
		int k = sc.nextInt();
		
		TreeMap<Integer,Integer> tm = new TreeMap<>();
		
		for(int i=0;i<n;i++)
		{
			int x = sc.nextInt();
			if(tm.containsKey(x))
				tm.put(x,tm.get(x)+1);
			else
				tm.put(x,1);
		}
		
		int ans = 0;
		int prek = tm.firstKey();
		int precount = tm.get(prek);
		tm.remove(prek);
		
		while(!tm.isEmpty())
		{
			int key = tm.firstKey();
			int count = tm.get(key);
			tm.remove(key);
			
			if(prek + k >= key)
				ans+=precount;
			
			prek = key;
			precount = count;
		}
		
		System.out.println(n - ans);
	}
	
    public static void main(String[] args)
    {
        new temp().solve();
    }
	
    static class FastReader
    {
        BufferedReader br;
        StringTokenizer st;
 
        public FastReader()
        {
            br = new BufferedReader(new
                     InputStreamReader(System.in));
        }
 
        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException  e)
                {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
 
        int nextInt()
        {
            return Integer.parseInt(next());
        }
 
        long nextLong()
        {
            return Long.parseLong(next());
        }
 
        double nextDouble()
        {
            return Double.parseDouble(next());
        }
 
        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                e.printStackTrace();
            }
            return str;
        }
    }
}
