#include <bits/stdc++.h>
using namespace std;
const int inv = 1000000000;
const int minv = -inv;
const int max_n = 30;
const int lim = 55;
int n;
int x[max_n];
int a, b, c, d;
char field[lim][2 * lim + 5];
bool valid(int i, int j) {
  if (i < 0 or i >= max(b, d)) return false;
  if (j < 0 or j >= a + c) return false;
  return (field[i][j] != '.');
}
pair<int, int> cell[2 * lim * lim];
int ncells;
int main() {
  scanf("%d", &a);
  scanf("%d", &b);
  scanf("%d", &c);
  scanf("%d", &d);
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) scanf("%d", &x[i]);
  for (int i = 0; i < max(b, d); ++i) {
    for (int j = 0; j < a + c; ++j) field[i][j] = '.';
    field[i][a + c] = '\0';
  }
  for (int i = 0; i < b; ++i)
    for (int j = 0; j < a; ++j) field[i][j] = '#';
  for (int i = 0; i < d; ++i)
    for (int j = a; j < a + c; ++j) field[i][j] = '#';
  int ci, cj, cdir;
  if ((min(b, d) % 2 == 0) xor (b < d)) {
    ci = 0;
    cj = 0;
    cdir = 1;
  } else {
    ci = 0;
    cj = a + c - 1;
    cdir = (-1);
  }
  ncells = 0;
  while (true) {
    if (not valid(ci, cj)) break;
    cell[ncells++] = pair<int, int>(ci, cj);
    if (valid(ci, cj + cdir))
      cj += cdir;
    else {
      ++ci;
      cdir *= (-1);
    }
  }
  assert(ncells == a * b + c * d);
  int currcell = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < x[i]; ++j) {
      field[cell[currcell].first][cell[currcell].second] = (char('a' + i));
      ++currcell;
    }
  }
  printf("YES\n");
  for (int i = 0; i < max(b, d); ++i) printf("%s\n", field[i]);
  return 0;
}
