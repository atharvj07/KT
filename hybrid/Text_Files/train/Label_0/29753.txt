#include <bits/stdc++.h>
using namespace std;
long long ans, res;
signed main() {
  long long n;
  cin >> n;
  long long x = n;
  vector<long long> v;
  vector<pair<long long, long long>> vec;
  while (n--) {
    long long a, b;
    cin >> a >> b;
    vec.push_back({a, b});
    if (b == 2) {
      v.push_back(ans);
    }
    ans += a;
  }
  if (v.size() == 0) {
    cout << "Infinity" << endl;
    exit(0);
  }
  sort(v.begin(), v.end(), greater<long long>());
  ans = 1899 - v[0] + ans;
  long long temp = ans;
  for (long long i = x - 1; i >= 0; i--) {
    temp -= vec[i].first;
    if (temp >= 1900 and vec[i].second == 2) {
      cout << "Impossible" << endl;
      exit(0);
    }
    if (temp < 1900 and vec[i].second == 1) {
      cout << "Impossible" << endl;
      exit(0);
    }
  }
  cout << ans << endl;
  return 0;
}
