#include<iostream>
#include<string>
#include<queue>
#include<map>
#include<vector>
using namespace std;

const int inf = 1<<29;
#define rep(i,n) for(i=0; i<n; ++i)

int dx[4] = {0, 1, 0, -1};
int dy[4] = {1, 0, -1, 0};

void bfs(const vector<vector<bool> > &grid,vector< vector<int> > &dist,
	 pair<int,int> start) {
  int i;
  dist[start.first][start.second] = 0;
  queue<pair<int,int> > Q;
  Q.push(start);
  while(!Q.empty()) {
    pair<int,int> now = Q.front(); Q.pop();
    int nx = now.second,ny = now.first;
    rep(i,4) {
      if(nx+dx[i] >= 0 && nx+dx[i] < grid[0].size() &&
	 ny+dy[i] >= 0 && ny+dy[i] < grid.size() &&
	 grid[ny+dy[i]][nx+dx[i]] == true && dist[ny+dy[i]][nx+dx[i]] == inf) {
	dist[ny+dy[i]][nx+dx[i]] = dist[ny][nx]+1;
	Q.push(make_pair(ny+dy[i],nx+dx[i]));
      }
    }
  }
  return;
}

int main() {
  int h,w,i,j,k;
  string s,pattern;
  while(cin>>h>>w, h|w) {
    vector< vector<bool> > grid(h, vector<bool>(w, true));
    vector< vector<int> > dist(h, vector<int>(w, inf));
    pair<int,int> start,ghost;
    rep(i,h) {
      cin>>s;
      rep(j,w) {
	if(s[j] == 'A') start = make_pair(i,j);
	if(s[j] == 'B') ghost = make_pair(i,j);
	if(s[j] == '#') grid[i][j] = false;
      }
    }
    cin>>pattern;
    bfs(grid,dist,start);
    int pat = pattern.length();
    bool possible = false;

    rep(i, h*w+h*w*pat+2) {
      if(dist[ghost.first][ghost.second] <= i) {
	cout<<i<<" "<<ghost.first<<" "<<ghost.second<<endl;
	goto NEXT;
      }
      int d = i%pat;
      if(pattern[d] == '8') {
	ghost.first--;
	if(ghost.first < 0) ghost.first = 0;
      }else if(pattern[d] == '2') {
	ghost.first++;
	if(ghost.first >= h) ghost.first = h-1;
      }else if(pattern[d] == '6') {
	ghost.second++;
	if(ghost.second >= w) ghost.second = w-1;
      }else if(pattern[d] == '4') {
	ghost.second--;
	if(ghost.second < 0) ghost.second = 0;
      }
    }
    cout<<"impossible"<<endl;
  NEXT:;

  }
}
