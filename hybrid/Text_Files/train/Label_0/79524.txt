import sys
from collections import Counter
sys.stdin.readline()
for line in sys.stdin:
	lsplit = line.split()
	n = int(lsplit[0])
	k = int(lsplit[1])
	nums = sys.stdin.readline().split()
	ctr = Counter(nums)
	sol = [0] * n
	cur = 0
	colors = [set() for i in range(k)]
	d = {num: set() for num in nums}
	for num in ctr:
		if ctr[num] >= k:
			for s in colors:
				s.add(num)
			for i in range(k):
				d[num].add(i)
		else:
			bak = cur
			for i in range(bak, bak + ctr[num]):
				colors[i % k].add(num)
				d[num].add(i % k)
				cur = (cur + 1) % k
	numred = len(colors[k - 1])
	for (i, s) in enumerate(colors):
		if len(s) > numred:
			num = s.pop()
			d[num].remove(i)
	for (i, num) in enumerate(nums):
		s = d[num]
		if len(s) > 0:
			col = s.pop()
			sol[i] = col + 1
	print(*sol)
