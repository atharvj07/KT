#include <iostream>
#include <algorithm>
using namespace std;

int main(){
  int  n;
  cin>>n;
  int res=0;
  int num[3];
  while(n>1){
    if(n<=3) n=3;
    num[0]=n/3;
    num[1]=(n-num[0])/2;
    num[2]=n-num[1]-num[0];
    n=max(num[0],num[2]);
    res++;
    
  }
  cout << res<<endl;
  return 0;
}
