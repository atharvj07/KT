#include <bits/stdc++.h>
using namespace std;
int t, n;
long long a[100005];
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    long long ans = 1;
    for (int i = n - 1; i >= n - 5; i--) ans *= a[i];
    long long tmp = 1;
    for (int i = 0; i < 5; i++) tmp *= a[i];
    ans = max(ans, tmp);
    tmp = 1;
    long long tmp2 = 1;
    tmp *= a[0] * a[1] * a[n - 1] * a[n - 2] * a[n - 3];
    tmp2 *= a[0] * a[1] * a[2] * a[3] * a[n - 1];
    ans = max(ans, max(tmp, tmp2));
    cout << ans << endl;
  }
  return 0;
}
