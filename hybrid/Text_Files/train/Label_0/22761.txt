#include <bits/stdc++.h>
using namespace std;
bool visited[200001];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int T;
  cin >> T;
  while (T--) {
    int N;
    cin >> N;
    vector<int> p(N), c(N);
    for (int i = 0; i < N; i++) cin >> p[i], p[i]--;
    for (int i = 0; i < N; i++) cin >> c[i], c[i]--;
    vector<int> used(N, 0);
    int ans = 2e9;
    for (int i = 0; i < N; i++) {
      if (used[i]) continue;
      vector<int> cycle;
      int v = i;
      while (!used[v]) {
        used[v] = 1;
        cycle.push_back(v);
        v = p[v];
      }
      for (int step = 1; step < cycle.size() + 1; step++) {
        if (cycle.size() % step != 0) continue;
        for (int s = 0; s < step; s++) {
          bool eq = true;
          for (int pos = s; pos + step < cycle.size(); pos += step) {
            if (c[cycle[pos]] != c[cycle[pos + step]]) eq = false;
          }
          if (eq) {
            ans = min(ans, step);
            break;
          }
        }
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
