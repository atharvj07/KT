#include <bits/stdc++.h>
using namespace std;
int main(void) {
  int n, m, i, j, t, p, q;
  long long l, r, x, k;
  vector<long long> s;
  vector<long long> hol;
  s.clear();
  hol.clear();
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%I64d", &k);
    s.push_back(k);
  }
  sort(s.begin(), s.end());
  if (n > 1) {
    for (i = 1; i < s.size(); i++) {
      hol.push_back(s[i] - s[i - 1]);
    }
    sort(hol.begin(), hol.end());
    long long pref[hol.size()];
    pref[0] = hol[0];
    for (i = 1; i < hol.size(); i++) pref[i] = pref[i - 1] + hol[i];
    long long ans;
    for (i = 0; i < hol.size(); i++) {
    }
    scanf("%d", &q);
    for (i = 0; i < q; i++) {
      scanf("%I64d%I64d", &l, &r);
      x = r - l;
      int l1, r1;
      l1 = 0;
      r1 = hol.size();
      if (x < hol[0]) {
        ans = (r - l + 1) * n;
        printf("%I64d\n", ans);
        continue;
      }
      if (x >= hol.back()) {
        ans = s.back() - s[0] + r - l + 1;
        printf("%I64d\n", ans);
        continue;
      }
      while (l1 < r1) {
        m = (l1 + r1) / 2;
        if (hol[m] <= x)
          l1 = m + 1;
        else
          r1 = m;
      }
      ans = pref[l1 - 1] + (r - l + 1) * (n - l1);
      printf("%I64d\n", ans);
    }
  } else {
    scanf("%d", &q);
    for (i = 0; i < q; i++) {
      scanf("%I64d%I64d", &l, &r);
      x = r - l + 1;
      printf("%I64d\n", x);
    }
  }
  return 0;
}
