from collections import Counter

def solve():
	n = int(input())
	tuples = [list(map(int, input().split())) for _ in range(n)]
	counter_topic = Counter([a for (a, b) in tuples])
	counter_diff = Counter([b for (a, b) in tuples])
	all_comb = n * (n - 2) * (n - 1) // 6
	ans = 0
	for (a, b) in tuples:
		same_topic = counter_topic[a] - 1
		same_diff = counter_diff[b] - 1
		ans += same_topic * same_diff
	return all_comb - ans
for i in range(int(input())):
	print(solve())
