#include<bits/stdc++.h>
using namespace std;

int mp[30];

int main(){
  string s, t;
  cin >> s >> t;
  for(int i = 0; i < s.size(); i++){
    if(mp[s[i] - 'a'] != t[i]){
      if(mp[s[i] - 'a']){
        puts("No");
        return 0;
      }else mp[s[i] - 'a'] = t[i];
    }
  }
  sort(mp, mp+26);
  for(int i = 1; i < 26; i++){
    if(mp[i] && mp[i] == mp[i-1]){
      puts("No");
      return 0;
    }
  }
  puts("Yes");
}
