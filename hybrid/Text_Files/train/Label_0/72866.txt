import math

def int_input():
	a = int(input())
	return a

def int_list_input():
	a = list(map(int, input().split()))
	return a

def solve():
	s = input()
	i = 0
	n = len(s)
	ans = 0
	while i < n:
		j = i + 1
		while j < n and s[j] == s[i]:
			j += 1
		if s[i] == '0':
			ans += 1
		i = j
		if ans > 2:
			break
	ans = min(2, ans)
	print(ans)
for _ in range(int(input())):
	solve()
