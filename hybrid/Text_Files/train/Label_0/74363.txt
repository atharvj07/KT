def next():
	return input()

def nextInt():
	return int(next())

def nextFloat():
	return float(next())

def getIntArray(n, sep=' '):
	return [[int(d) for d in next().split(sep)] for _ in range(n)]

def getFloatArray(n):
	return [nextFloat() for _ in range(n)]

def getStringArray(n):
	return [next() for _ in range(n)]

def printList(l):
	print(' '.join(map(str, l)))
N = nextInt()
data = getIntArray(N)
for (s, l, r) in data:
	if l - 1 + (r > l) > s / 2 or r - (l < r) < s / 2:
		print('-1')
	else:
		if r >= s / 2 + 1:
			ap = list(range(s // 2 + 1, s + 1))
			ap.remove(r)
		else:
			ap = list(range(s // 2 + 2, s + 1))
		o = [l] + ap
		if l <= s / 2:
			ap = list(range(1, s // 2 + 1))
			ap.remove(l)
		else:
			ap = list(range(1, s // 2))
		o += [r] + ap
		printList(o)
