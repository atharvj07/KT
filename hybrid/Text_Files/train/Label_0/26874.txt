void solve(int n,int m){
    vector<ll> a(n);
    vector<ll> w(m);
    rep(i,n) cin >> a[i];
    rep(i,m) cin >> w[i];
    sort(all(a));

    unordered_set<ll> val; // 3 ^ 10 =  6 * 10 ^ 4
    for(int mask1=0;mask1<(1<<m);mask1++){
        for(int mask2=0;mask2<(1<<m);mask2++){
            if(mask1 & mask2) continue;
            ll sum = 0;
            for(int i=0;i<m;i++){
                int m1 = (mask1 & (1<<i));
                int m2 = (mask2 & (1<<i));
                if(m1 and !m2){
                    sum += w[i];
                }else if(!m1 and m2){
                    sum -= w[i];
                }
            }

            val.insert(sum);
        }
    }

    bool ok = true;
    for(int i=0;i<n;i++){
        if(!val.count(a[i])){
            ok = false;
            break;
        }
    }
    if(ok){
        cout << 0 << endl;
        return;
    }

    set<ll> cand;
    for(int i=0;i<n;i++){
        if(val.count(a[i])) continue;

        for(auto v : val){
            if(v!=a[i]) cand.insert(abs(v-a[i]));
        }

        break;
    }

    for(auto c : cand){
        bool ok = true;
        for(int i=0;i<n;i++){
            if(!(val.count(a[i]) || val.count(a[i]-c) || val.count(a[i]+c))){
                ok = false;
                break;
            }
        }
        if(ok){
            cout << c << endl;
            return;
        }
    }
    cout << -1 << endl;
}

int main(){
    while(1){
        int n,m; cin >> n >> m;
        if(n == 0 and m == 0) break;
        solve(n,m);
    }
}
