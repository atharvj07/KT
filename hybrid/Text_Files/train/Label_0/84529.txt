M = 10 ** 9 + 7

class paths:

	def __init__(self):
		super().__init__()
		self.dp = {}

	def no_of_paths(self, n, k):
		if n in self.dp:
			return self.dp[n]
		if n == 0:
			return 1
		elif n < 0:
			return 0
		else:
			paths = 0
			for i in range(1, k + 1):
				val = self.no_of_paths(n - i, k)
				paths += val
			self.dp[n] = paths
			return paths
(n, k, d) = map(int, input().split())
pathk = paths()
pathd = paths()
print((pathk.no_of_paths(n, k) - pathd.no_of_paths(n, d - 1)) % M)
