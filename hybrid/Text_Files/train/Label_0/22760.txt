#include <bits/stdc++.h>
using namespace std;
template <typename T1, typename T2>
istream& operator>>(istream& in, pair<T1, T2>& a) {
  in >> a.first >> a.second;
  return in;
}
template <typename T1, typename T2>
ostream& operator<<(ostream& out, pair<T1, T2> a) {
  out << a.first << " " << a.second;
  return out;
}
template <typename T, typename T1>
T amax(T& a, T1 b) {
  if (b > a) a = b;
  return a;
}
template <typename T, typename T1>
T amin(T& a, T1 b) {
  if (b < a) a = b;
  return a;
}
const long long INF = 1e18;
const int32_t M = 1e9 + 7;
const int32_t MM = 998244353;
const long long N = 2e5 + 5;
long long p[N];
long long c[N];
long long vis[N];
void solve() {
  long long n;
  cin >> n;
  for (long long i = 1; i < n + 1; i++) cin >> p[i];
  for (long long i = 1; i < n + 1; i++) cin >> c[i];
  for (long long i = 1; i < n + 1; i++) vis[i] = 0;
  long long len = INF;
  vector<long long> cyc;
  for (long long i = 1; i < n + 1; i++) {
    if (vis[i]) continue;
    cyc.clear();
    long long j = i, t = 0;
    bool k = 1;
    do {
      if (c[i] != c[j]) k = 0;
      j = p[j];
      t++;
      cyc.push_back(j);
    } while (j != i);
    if (k) {
      cout << "1\n";
      return;
    }
    amin(len, t);
    vector<long long> fac;
    for (long long j = 2; j * j <= t; j++) {
      if (t % j) continue;
      fac.push_back(j);
      if (j * j != t) fac.push_back(t / j);
    }
    for (long long f : fac) {
      vector<long long> z(f);
      for (long long j = 0; j < t; j++) {
        long long e = cyc[j];
        if (z[j % f] == 0) z[j % f] = c[e];
        if (z[j % f] != c[e]) z[j % f] = -1;
      }
      for (long long j = 0; j < f; j++) {
        if (z[j] != -1) amin(len, f);
      }
    }
    for (long long x : cyc) vis[x] = 1;
  }
  cout << len << "\n";
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long t = 1;
  cin >> t;
  while (t--) solve();
  return 0;
}
