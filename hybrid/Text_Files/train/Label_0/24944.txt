#include<iostream>
#include<algorithm>
#include<queue>
using namespace std;
bool used[55][55][55][55];
int n,m;
string s[55],t[55];
int d[]={0,1,0,-1,0};
int main()
{
	while(cin>>m>>n,m)
	{
		for(int i=0;i<n;i++)
		{
			cin>>s[i]>>t[i];
			reverse(t[i].begin(),t[i].end());
			for(int j=0;j<m;j++)
			{
				for(int k=0;k<n;k++)for(int l=0;l<m;l++)used[i][j][k][l]=0;
			}
		}
		queue<pair<pair<int,int>,pair<int,int> > >P;
		pair<int,int>Ss,St;
		for(int i=0;i<n;i++)for(int j=0;j<m;j++)
		{
			if(s[i][j]=='L')Ss={i,j};
			if(t[i][j]=='R')St={i,j};
		}
		used[Ss.first][Ss.second][St.first][St.second]=1;
		P.push(make_pair(Ss,St));
		for(int i=0;i<n;i++)for(int j=0;j<m;j++)
		{
			if(s[i][j]=='%')Ss={i,j};
			if(t[i][j]=='%')St={i,j};
		}
		bool flag=1;
		while(flag&&!P.empty())
		{
			pair<int,int>a=P.front().first,b=P.front().second;
			P.pop();
			for(int r=0;r<4;r++)
			{
				pair<int,int>A=a,B=b;
				if(d[r])
				{
					int tx=A.first+d[r];
					if(tx>=0&&tx<n&&s[tx][A.second]!='#')A.first=tx;
					tx=B.first+d[r];
					if(tx>=0&&tx<n&&t[tx][B.second]!='#')B.first=tx;
				}
				else
				{
					int ty=A.second+d[r+1];
					if(ty>=0&&ty<m&&s[A.first][ty]!='#')A.second=ty;
					ty=B.second+d[r+1];
					if(ty>=0&&ty<m&&t[B.first][ty]!='#')B.second=ty;
				}
				if(used[A.first][A.second][B.first][B.second])continue;
				used[A.first][A.second][B.first][B.second]=1;
				if(A==Ss||B==St)
				{
					if(A==Ss&&B==St)
					{
						cout<<"Yes"<<endl;
						flag=0;
						break;
					}
				}
				else P.push(make_pair(A,B));
			}
		}
		if(flag)cout<<(used[Ss.first][Ss.second][St.first][St.second]?"Yes":"No")<<endl;
	}
}
