#include <bits/stdc++.h>
using namespace std;
int main() {
  double a, b, m, vx, vy, vz;
  cin >> a >> b >> m >> vx >> vy >> vz;
  double X = a / 2 - m / vy * vx;
  double Z = -m / vy * vz;
  while (X < 0 || X > a)
    if (X > a)
      X = 2 * a - X;
    else
      X = -X;
  while (Z < 0 || Z > b)
    if (Z > b)
      Z = 2 * b - Z;
    else
      Z = -Z;
  printf("%.9lf %.9lf", X, Z);
  return 0;
}
