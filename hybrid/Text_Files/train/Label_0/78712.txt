def binpow(a, b, m):
	a %= m
	x = 1
	while b > 0:
		if b & 1:
			x = x * a % m
		a = a * a % m
		b >>= 1
	return x

def sieve(n):
	d = defaultdict(list)
	if n < 9:
		n = 10
	l = [1] * n
	for i in range(2, int(n ** 0.5) + 1):
		if l[i - 1] == 1:
			for j in range(i ** 2, n + 1, i):
				if j % i == 0:
					l[j - 1] = 0
					d[j].append(i)
	return (l, d)

def binser(arr, l, r, x):
	while l <= r:
		mid = l + (r - l) // 2
		if arr[mid] == x:
			return mid
		elif arr[mid] < x:
			l = mid + 1
		else:
			r = mid - 1
	return mid

def lcm(a, b):
	return a * b // gcd(a, b)

def panda(n):
	l = [k for k in range(1, n + 1)]
	for i in range(0, n - n % 2, 2):
		(l[i], l[i + 1]) = (l[i + 1], l[i])
	if n % 2:
		(l[-1], l[-2]) = (l[-2], l[-1])
	return l

		print(*panda(int(input())))
