import sys
input = sys.stdin.readline
from operator import mul
from functools import reduce

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())

def main():
	n = inp()
	s = ''
	for i in range(1, n + 1):
		s += str(i)
	print(s[n - 1])
main()
