q = int(input())
for _ in range(q):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	ans = [-1] * n
	temp = 0
	for i in range(n):
		if ans[i] == -1:
			ans[i] = temp + 1
			t = a[i]
			for j in range(i + 1, n):
				if ans[j] == -1 and a[j] - t > 1:
					ans[j] = temp + 1
					t = a[j]
			temp += 1
	print(temp)
