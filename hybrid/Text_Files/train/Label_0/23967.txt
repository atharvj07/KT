#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<vector<int> > numbers;
  numbers.push_back(vector<int>(1, 0));
  for (int i = 1; i <= n; i++) {
    vector<vector<int> > new_numbers;
    for (auto &n : numbers) {
      vector<int> first;
      vector<int> second;
      for (auto &b : n) {
        first.push_back(b);
        second.push_back(b);
      }
      for (auto &b : n) {
        first.push_back(b);
        second.push_back(!b);
      }
      new_numbers.push_back(first);
      new_numbers.push_back(second);
    }
    numbers = new_numbers;
  }
  for (auto &n : numbers) {
    for (auto &b : n) {
      if (b) {
        cout << "*";
      } else {
        cout << "+";
      }
    }
    cout << endl;
  }
  return 0;
}
