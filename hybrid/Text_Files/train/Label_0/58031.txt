import java.util.*;
import java.io.*;
import java.math.*;

public class Rail {

	public static Integer[] parse(String st) {
		String[] a = st.split(" ");
		Integer[] ar = new Integer[a.length];
		for (int i = 0; i < a.length; i++) 
			ar[i] = Integer.parseInt(a[i]);
		return ar;
	}

	public static int solve(boolean[][] rail) {
		int best = Integer.MAX_VALUE;
		LinkedList<Integer[]> todo = new LinkedList<>();
		todo.add(new Integer[] {1, 0});
		while (!todo.isEmpty()) {
			Integer[] cur = todo.remove();
			int x = cur[0];
			if (x == rail.length - 1) {
				if (cur[1] < best) best = cur[1];
			} else {
				for (int i = 1; i < rail.length; i++) {
					if (rail[x][i]) {
						rail[x][i] = false;
						todo.add(new Integer[] {i, cur[1] + 1});
					}
				}
			}
		}
		if (best == Integer.MAX_VALUE) return -1;
		return best;
	}

	public static boolean[][] adjMat(ArrayList<Integer[]> edges, int verts) {
		boolean[][] out = new boolean[verts + 1][verts + 1];
		for (Integer[] edge : edges) {
			int x = edge[0]; int y = edge[1];
			out[x][y] = true;
			out[y][x] = true;
		}
		return out;
	}

	public static boolean[][] complement(boolean[][] adjMat) {
		for (int i = 1; i < adjMat.length; i++) {
			for (int j = 1; j < adjMat[0].length; j++) {
				adjMat[i][j] = !adjMat[i][j];
			}
		}
		return adjMat;
	}

	public static void print(boolean[][] a) {
		for (boolean[] ar : a) System.out.println(Arrays.toString(ar));
	}

	public static void main(String[] args) {
		try {
			InputStreamReader is = new InputStreamReader(System.in);
			BufferedReader br = new BufferedReader(is);

			Integer[] p = parse(br.readLine());
			ArrayList<Integer[]> edges = new ArrayList<>();
			for (int i = 0; i < p[1]; i++) edges.add(parse(br.readLine()));
			boolean[][] am = adjMat(edges, p[0]);
			boolean[][] amp = new boolean[am.length][am.length];
			for (int i = 0; i < amp.length; i++) 
				amp[i] = Arrays.copyOfRange(am[i], 0, am[i].length);
			amp = complement(amp);

			int s1 = solve(am);
			int s2 = solve(amp);

			if (s1 == -1 || s2 == -1) {
				System.out.println(-1);
			} else {
				System.out.println(Math.max(s1, s2));
			}
		} catch (Exception e) {e.printStackTrace();}
	}

}
