const int NMAX = 100000 + 10;

template <class T>
inline T ABS(T a) {
  if (a < 0) a = -a;
  return a;
}
int main() {
  ios::sync_with_stdio(false);
  int n, m;
  int q, w;
  int k;
  int i;
  int a[NMAX];
  cin >> n >> m;
  for (i = 0; i < m; i++) {
    cin >> a[i] >> a[i];
  }
  sort(a, a + m);
  reverse(a, a + m);
  long long s = 0;
  long long res;
  for (k = 1; k <= n; k++) {
    if (k & 1)
      res = (k * 1ll * (k - 1)) / 2 + 1;
    else
      res = (k * 1ll * (k - 1)) / 2 + k / 2;
    if (res > n) break;
  }
  k--;
  for (i = 0; i < min(m, k); i++) {
    s += a[i];
  }
  cout << s << endl;
  return 0;
}
