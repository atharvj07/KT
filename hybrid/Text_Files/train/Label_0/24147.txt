typedef complex<ll> P;
 
ll cross(P a, P b) {
    return imag(conj(a) * b);
}
 
bool cmp_x(const P& p, const P& q) {
    return p.real() != q.real() ? p.real() < q.real() : p.imag() < q.imag();
}
 
vector<ll> lower_convex_area(const vector<P>& p) {
    int n = p.size();
    int k = 0;
 
    vector<ll> area(n + 1, 0);
    vector<P> h(n);
 
    rep(i, n) {
        area[i + 1] += area[i];
        while (k > 1 && cross(h[k - 1] - h[k - 2], p[i] - h[k - 1]) < 0) {
            if (k > 2) area[i + 1] -= cross(h[k - 2] - h[0], h[k - 1] - h[0]);
            k--;
        }
        h[k++] = p[i];
        if (k > 2) area[i + 1] += cross(h[k - 2] - h[0], h[k - 1] - h[0]);
    }
 
    return area;
}
 
signed main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(0);
 
    int n;
    cin >> n;
    vector<P> p;
    rep(i, n) {
        ll x, y;
        cin >> x >> y;
        p.emplace_back(x, y);
    }
 
    sort(all(p), cmp_x);
    auto lowerleft = lower_convex_area(p);
 
    rep(i, n) p[i] = P(p[i].real(), p[i].imag() * -1);
    auto upperleft = lower_convex_area(p);
 
    rep(i, n) p[i] = P(p[i].real() * -1, p[i].imag());
    sort(all(p), cmp_x);
    auto upperright = lower_convex_area(p);
 
    rep(i, n) p[i] = P(p[i].real(), p[i].imag() * -1);
    auto lowerright = lower_convex_area(p);
 
    rep(i, n) p[i] = P(p[i].real() * -1, p[i].imag());
    sort(all(p), cmp_x);
 
    ll ans = lowerleft[n] + upperleft[n] + lowerright[0] + upperright[0];
    rep2(i, 1, n) {
        if (p[i].real() != p[i - 1].real()) {
            ans = min(ans, lowerleft[i] + upperleft[i] + lowerright[n - i] + upperright[n - i]);
        }
    }
    cout << (ans + 1) / 2 << endl;
 
}
