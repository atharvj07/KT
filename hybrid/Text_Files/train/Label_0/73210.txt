from collections import Counter

def solution(arr):
	arr.sort()
	min_diff = float('inf')
	for i in range(len(arr) - 1):
		(num1, num2) = (arr[i], arr[i + 1])
		if num2 - num1 < min_diff:
			min_diff = num2 - num1
			idx = i
	return [arr[idx]] + arr[idx + 2:] + arr[:idx] + [arr[idx + 1]]
N = int(input())
for _ in range(N):
	input()
	arr = list(map(int, input().split()))
	print(' '.join(map(str, solution(arr))))
