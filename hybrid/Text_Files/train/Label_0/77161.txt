def ii():

def si():
	return input()

def mi():
	return map(int, input().split())

def li():
	return list(mi())

def Function():
	(w, l) = li()
	a = li()
	a.reverse()
	s = sum(a[:l])
	min = s
	for i in range(l, w - 1):
		s += a[i]
		s -= a[i - l]
		if s < min:
			min = s
	print(min)
for _ in range(1):
	Function()
