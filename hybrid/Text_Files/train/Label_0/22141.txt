#include <iostream>
#include <cstdio>
#include <algorithm>
#include <cmath>

using namespace std;

int main(){
	
	int n;
	
	int a[10000];
	
	while(true){
		cin >> n;
		if(cin.eof()){
			break;
		}
		for(int i = 0; i < n; i++){
			cin >> a[i];
			a[i] = -a[i];
		}
		
		sort(a, a + n);
		
		int ans = 0;
		
		for(int i = 0; i < n; i++){
			a[i] = -a[i];
			ans += (i + 1) * a[i];
			//printf("%d ", a[i]);
		}
		printf("%d\n", ans);
	}
	
	return 0;
}
