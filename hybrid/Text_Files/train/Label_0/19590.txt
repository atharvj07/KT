#include <iostream>
using namespace std;
typedef long long int llf;
int main() {
	// your code goes here
	llf M,rd,rr,cd,cr;
	cin >> M >> rd >> rr >> cd >> cr;
	llf xx=cd*100.0/rd;
	llf yy=cr*100.0/rr;
	if(xx*rd<100*cd)
	xx++;
	if(yy*rr<100*cr)
	yy++;
	if(M-xx-yy>=0)
	cout << M-xx-yy << endl;
	else
	cout <<-1<< endl;
	return 0;
}
