from collections import Counter
m = int(input())
a = list(map(int, input().split()))
a.sort()
mx = a[-1]
d = Counter(a)
l1 = []
l2 = []
for i in d:
	if i != mx:
		if d[i] >= 2:
			l1.append(i)
			l2.append(i)
		else:
			l1.append(i)
	else:
		l1.append(i)
stair = l1 + l2[::-1]
print(len(stair))
print(*stair)
