import sys

def read_ints():
	return [int(i) for i in sys.stdin.readline().strip().split()]

def read_int():
	return int(sys.stdin.readline().strip())

def print_ints(seq):
	print(' '.join((str(i) for i in seq)))
n = read_int()
pp = read_ints()
diags = [pp]
for i in range(1, n):
	diags.append([j for j in diags[-1] if j > i])
for i in range(n):
	row = []
	for j in range(i + 1):
		row.append(diags[j][i - j])
	row.reverse()
	print_ints(row)
