#include <bits/stdc++.h>
using namespace std;
vector<int> v[1000010];
vector<int> adj[1000010];
int color[1000010];
vector<int> used;
bool cmp(int a, int b) { return color[a] < color[b]; }
void dfs(int i, int parent) {
  int c = 1;
  used.clear();
  for (int j = 0; j < v[i].size(); j++) {
    if (color[v[i][j]] != 0) {
      used.push_back(color[v[i][j]]);
    }
  }
  sort(used.begin(), used.end());
  sort(v[i].begin(), v[i].end(), cmp);
  for (int j = 0; j < v[i].size(); j++) {
    if (color[v[i][j]] != 0) {
      break;
    } else {
      while (binary_search(used.begin(), used.end(), c) == true) {
        c++;
      }
      color[v[i][j]] = c++;
    }
  }
  for (int u : adj[i]) {
    if (u != parent) {
      dfs(u, i);
    }
  }
}
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  int sz, temp;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &sz);
    while (sz--) {
      scanf("%d", &temp);
      v[i].push_back(temp);
    }
  }
  for (int i = 1; i < n; i++) {
    scanf("%d %d", &sz, &temp);
    adj[sz].push_back(temp);
    adj[temp].push_back(sz);
  }
  dfs(1, -1);
  int ma = 0;
  for (int i = 1; i <= m; i++) {
    if (color[i] == 0) color[i] = 1;
    if (color[i] > ma) ma = color[i];
  }
  printf("%d\n", ma);
  for (int i = 1; i <= m; i++) {
    printf("%d ", color[i]);
  }
  return 0;
}
