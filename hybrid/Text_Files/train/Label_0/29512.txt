#include <bits/stdc++.h>
#define fo(a,b,c) for (a=b; a<=c; a++)
#define fd(a,b,c) for (a=b; a>=c; a--)
#define ll long long
//#define file
using namespace std;

int c[300001],C[300001],n,Q,i,j,k,l,L,R,Mid,S,s,x,y,l1,r1,Lca;
int a[600001][2],ls[300001],len;
int nx[300001],top[300001],sz[300001],d[300001],fa[300001],bg[300001],ed[300001];
int tr[20000001][3];
char st[16800001];
int I=-1;
vector<int> v[300001];

ll random(ll x,ll y) {return (32768ll*rand()+rand())%(y-x+1)+x;}
void NEW(int x,int y) {++len;a[len][0]=y;a[len][1]=ls[x];ls[x]=len;}

char ch;
int Read()
{
	int x=0;
	++I;
	while (st[I]<'0' || st[I]>'9') ++I;
	while (st[I]>='0' && st[I]<='9') x=x*10+(st[I]-'0'),++I;
	return x;
}
void Write(int x)
{
	static char st[21];
	int i=0;
	while (x) st[++i]=x%10+'0',x/=10;
	while (i) putchar(st[i--]);
	putchar('\n');
	return;
}
void copy(int t1,int t2) {memcpy(tr[t1],tr[t2],sizeof(tr[t1]));}
void New(int t,int x) {++len;copy(len,tr[t][x]);tr[t][x]=len;}
void dfs(int Fa,int t)
{
	int i,mx=-1;
	d[t]=d[Fa]+1;bg[t]=++j,sz[t]=1;fa[t]=Fa;
	for (i=ls[t]; i; i=a[i][1])
	if (a[i][0]!=Fa)
	{
		dfs(t,a[i][0]);
		sz[t]+=sz[a[i][0]];
		if (sz[a[i][0]]>mx) mx=sz[a[i][0]],nx[t]=a[i][0];
	}
	ed[t]=j;
}
void Dfs(int Fa,int t)
{
	int i;
	if (nx[t]) top[nx[t]]=top[t],Dfs(t,nx[t]);
	
	for (i=ls[t]; i; i=a[i][1])
	if (a[i][0]!=Fa && a[i][0]!=nx[t])
	{
		top[a[i][0]]=a[i][0];
		Dfs(t,a[i][0]);
	}
}
int lca(int x,int y)
{
	int i;
	while (x!=y)
	{
		if (bg[x]<=bg[y] && ed[y]<=ed[x]) return x;
		if (bg[y]<=bg[x] && ed[x]<=ed[y]) return y;
		if (d[top[x]]>d[top[y]]) x=fa[top[x]];
		else y=fa[top[y]];
	}
	return x;
}

void change(int t,int l,int r,int x,int y,int s)
{
	int mid=(l+r)/2;
	if (x<=l && r<=y) {tr[t][2]^=s;return;}
	if (x<=mid) New(t,0),change(tr[t][0],l,mid,x,y,s);
	if (mid<y) New(t,1),change(tr[t][1],mid+1,r,x,y,s);
}
int find(int t,int l,int r,int x)
{
	int mid=(l+r)/2,ans=0;
	if (!t) return 0;
	if (l==r) return tr[t][2];
	
	if (x<=mid) ans=find(tr[t][0],l,mid,x);
	else ans=find(tr[t][1],mid+1,r,x);
	return ans^tr[t][2];
}

void init()
{
	j=0,dfs(0,1);
	Dfs(0,1);
	len=n;
	fo(i,1,n) v[c[i]].push_back(i);
	fo(i,1,n)
	{
		l=v[i].size(),copy(i,i-1);
		fo(j,0,l-1)
		k=v[i][j],change(i,1,n,bg[k],ed[k],C[i]);
	}
}

int main()
{
	srand(time(NULL));
	#ifdef file
	freopen("CF1479D.in","r",stdin);
	freopen("CF1479D.out","w",stdout);
	#endif
	
	fread(st,1,16800001,stdin);
	n=Read(),Q=Read();
	fo(i,1,n) c[i]=Read();
	fo(i,1,n) C[i]=random(0,2147483647);
	fo(i,1,n-1) j=Read(),k=Read(),NEW(j,k),NEW(k,j);
	init();
	
	for (;Q;--Q)
	{
		x=Read(),y=Read(),l1=Read(),r1=Read(),Lca=lca(x,y);
		if (l1>1) S=find(l1-1,1,n,bg[x])^find(l1-1,1,n,bg[y]); else S=0;
		s=find(r1,1,n,bg[x])^find(r1,1,n,bg[y]);
		if (l1<=c[Lca] && r1>=c[Lca]) s^=C[c[Lca]];
		
		if (s==S) printf("-1\n");
		else
		{
			L=l1,R=r1;
			while (L<R)
			{
				Mid=(L+R)/2,s=find(Mid,1,n,bg[x])^find(Mid,1,n,bg[y]);
				if (l1<=c[Lca] && Mid>=c[Lca]) s^=C[c[Lca]];
				if (s==S) L=Mid+1; else R=Mid;
			}
			Write(L);
		}
	}
	
	fclose(stdin);
	fclose(stdout);
	return 0;
}
