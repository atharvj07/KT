#include <bits/stdc++.h>
using namespace std;
const int N = 15005;
int n, dep[N], fa[N], q[N], idx;
vector<int> v[N];
void dfs(int pos) {
  dep[pos] = dep[fa[pos]] + 1;
  for (auto &i : v[pos]) {
    if (i == fa[pos]) continue;
    if (!dep[i])
      fa[i] = pos, dfs(i);
    else if (dep[i] < dep[pos]) {
      q[++idx] = pos;
      for (int cur = pos; cur != i; cur = fa[cur]) q[++idx] = fa[cur];
    }
  }
}
bool vis[N];
int sz[N], f[N], g[N];
void dfs2(int pos, int fa) {
  sz[pos] = 1;
  for (auto &i : v[pos])
    if (i != fa && !vis[i]) dfs2(i, pos), sz[pos] += sz[i], f[pos] += f[i];
  g[pos] = n + f[pos];
  for (auto &i : v[pos])
    if (i != fa && !vis[i])
      g[pos] = max(g[pos], f[pos] + (n - sz[i]) - f[i] + g[i]);
  f[pos] += sz[pos];
}
int dp[N * 2], sum[N * 2];
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int t1, t2;
    cin >> t1 >> t2;
    ++t1;
    ++t2;
    v[t1].push_back(t2);
    v[t2].push_back(t1);
  }
  dfs(1);
  for (int i = 1; i <= idx; i++) vis[q[i]] = 1, q[i + idx] = q[i];
  for (int i = 1; i <= idx; i++) dfs2(q[i], 0);
  for (int i = 1; i <= idx * 2; i++)
    sum[i] = sum[i - 1] + sz[q[i]], dp[i] = f[q[i]];
  for (int l = 1; l <= idx - 2; l++) {
    for (int i = 1; i <= idx * 2 - l + 1; i++)
      dp[i] += sum[i + l - 1] - sum[i - 1];
    for (int i = 1; i <= idx * 2 - l; i++)
      dp[i] = max(dp[i + 1] + f[q[i]], dp[i] + f[q[i + l]]);
  }
  int ans = 0;
  for (int i = 1; i <= idx; i++) ans = max(ans, dp[i + 1] + g[q[i]]);
  cout << ans << endl;
  return 0;
}
