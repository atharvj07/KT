import java.util.*;
public class Test1{
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		int[] a=new int[n];
		List<Integer> l=new ArrayList<>();
		Set<Integer> s=new HashSet<>();
		for(int i=0;i<n;i++){
			a[i]=sc.nextInt();
			s.add(a[i]);
		}
		for(int i:s){
			l.add(i);
		}
		Collections.sort(l);
		if(l.size()>3){
			System.out.println(-1);
		}
		else if(l.size()==3){
			if(l.get(1)-l.get(0)== l.get(2)-l.get(1)){
				System.out.println(l.get(1)-l.get(0));	
			} 
			else
				System.out.println(-1);
		}
		else if(l.size()==2){
			if((l.get(1)-l.get(0))%2==0){
				System.out.println((l.get(1)-l.get(0))/2);
			}
			else{
				System.out.println(l.get(1)-l.get(0));
			}
		}
		else{
			System.out.println(0);
		}
	}
}
