#include <bits/stdc++.h>
using namespace std;
pair<double, double> p(double a, double d, double x) {
  double v1 = (a + x) * (d + x);
  double v2 = (a + x) * (d - x);
  double v3 = (a - x) * (d + x);
  double v4 = (a - x) * (d - x);
  return minmax({v1, v2, v3, v4});
}
int main() {
  double a, b, c, d;
  cin >> a >> b >> c >> d;
  double low = 0;
  double high = 1e10;
  for (int i = 0; i < 1000; i++) {
    double mid = (low + high) / 2;
    pair<double, double> p1 = p(a, d, mid);
    pair<double, double> p2 = p(b, c, mid);
    if (p1.second < p2.first || p2.second < p1.first)
      low = mid;
    else
      high = mid;
  }
  printf("%.10f\n", low);
  return 0;
}
