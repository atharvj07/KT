void swaps(char *x, char *y) {
  char temp;
  temp = *x;
  *x = *y;
  *y = temp;
}
void swapi(int *a, int *b) {
  int temp;
  temp = *a;
  *a = *b;
  *b = temp;
}
unsigned long long gcd(unsigned long long a, unsigned long long b) {
  if (a == 0) return b;
  if (b == 0) return a;
  if (a == 1 || b == 1) return 1;
  if (a == b) return a;
  if (a > b)
    return gcd(b, a % b);
  else
    return gcd(a, b % a);
}
int cum[100010];
int x[100010];
int main() {
  int N;
  cin >> N;
  for (int i = (0); i < (N); i++) x[i] = (N - i - i - 1 + N) % N;
  for (int i = (0); i < (N); i++) {
    cum[x[i]] += 1;
    cum[i] += 1;
    cum[N - i - 1] += 1;
  }
  for (int i = (0); i < (N); i++) {
    if (cum[i] != 3) {
      cout << -1 << endl;
      return 0;
    }

}
