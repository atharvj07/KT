#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  ;
  long long n, m;
  cin >> n >> m;
  long long inflow[n + 1], outflow[n + 1];
  long long required[n + 1];
  memset(inflow, 0, sizeof inflow);
  memset(required, 0, sizeof required);
  vector<pair<long long, long long> > edge[n + 1];
  map<pair<long long, long long>, long long> mm;
  for (long long i = 1; i <= m; ++i) {
    long long a, b, c;
    cin >> a >> b >> c;
    mm[{a, b}] = i;
    edge[a].push_back({b, c});
    edge[b].push_back({a, c});
    required[a] += c;
    required[b] += c;
  }
  required[n] *= 2;
  for (long long i = 1; i <= n; ++i) {
    required[i] /= 2;
  }
  vector<long long> nodes;
  bool visited[n + 1];
  memset(visited, 0, sizeof visited);
  long long result[m + 1];
  for (auto ch : edge[1]) {
    inflow[ch.first] += ch.second;
    if (inflow[ch.first] == (required[ch.first])) {
      visited[ch.first] = 1;
      nodes.push_back(ch.first);
    }
    if (mm.find(pair<long long, long long>(1, ch.first)) == mm.end()) {
      result[mm[{ch.first, 1}]] = 1;
    } else {
      result[mm[{1, ch.first}]] = 0;
    }
  }
  visited[1] = 1;
  while (nodes.size() > 0) {
    long long nd = nodes[nodes.size() - 1];
    nodes.pop_back();
    for (auto ch : edge[nd]) {
      if (ch.first == 1 or
          (inflow[ch.first] == required[ch.first] and ch.first != n)) {
        continue;
      }
      inflow[ch.first] += ch.second;
      if (inflow[ch.first] == required[ch.first]) {
        nodes.push_back(ch.first);
      }
      if (mm.find(pair<long long, long long>(nd, ch.first)) == mm.end()) {
        result[mm[{ch.first, nd}]] = 1;
      } else {
        result[mm[{nd, ch.first}]] = 0;
      }
    }
  }
  for (long long i = 1; i <= m; ++i) {
    cout << result[i] << "\n";
  }
}
