#include<iostream>
#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >>t;
    while(t--){
        int n,i,j;
        cin >>n;
        map<int,int> m;
        int a[n];
        for(i=0;i<n;i++)
           cin >>a[i];
           
          for(i=0;i<n-1;i++){
              for(j=i+1;j<n;j++){
                  m[abs(a[i]-a[j])]++;
              }
          } 
          cout<<m.size()<<endl;
    }
}
