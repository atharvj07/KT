#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e4 + 10;
bool solve(int* arr, int n, int k, int m) {
  for (int i = 0; i < n; i++) {
    if (i == 0) {
      if (arr[i] + k >= m) {
        arr[i] = 0;
      }
    } else {
      if (arr[i] < arr[i - 1]) {
        if (arr[i] + k < arr[i - 1])
          return 0;
        else {
          arr[i] = arr[i - 1];
        }
      } else if (arr[i - 1] < arr[i]) {
        if (arr[i] + k >= m and (arr[i] + k) % m >= arr[i - 1]) {
          arr[i] = arr[i - 1];
        }
      }
    }
  }
  return 1;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  int arr[n];
  for (int i = 0; i < n; i++) cin >> arr[i];
  int lo = 0;
  int hi = m - 1;
  int best = m;
  int arr2[n];
  for (int i = 0; i < n; i++) arr2[i] = arr[i];
  while (lo <= hi) {
    int mid = (hi + lo) / 2;
    for (int i = 0; i < n; i++) arr[i] = arr2[i];
    if (solve(arr, n, mid, m)) {
      best = min(best, mid);
      hi = mid - 1;
    } else {
      lo = mid + 1;
    }
  }
  cout << best << endl;
  return 0;
}
