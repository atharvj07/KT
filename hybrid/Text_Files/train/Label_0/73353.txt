(n, m) = [int(i) for i in input().split()]
a = [int(i) for i in input().split()]
l = []
for i in range(m):
	(t, r) = [int(i) for i in input().split()]
	while len(l) > 0 and r >= l[-1][1]:
		l.pop()
	l.append([t, r])
(x, y) = (0, l[0][1] - 1)
l.append([0, 0])
b = sorted(a[:y + 1])
for i in range(1, len(l)):
	for j in range(l[i - 1][1], l[i][1], -1):
		if l[i - 1][0] == 1:
			a[j - 1] = b[y]
			y -= 1
		else:
			a[j - 1] = b[x]
			x += 1
print(' '.join((str(i) for i in a)))
