import java.util.*;
import java.io.*;
public class Main {
public static void main(String [] args)throws Exception{
	BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	int n = Integer.parseInt(br.readLine());
	ArrayList<Point>array[] = new ArrayList[1001];
	StringTokenizer st;
	for(int i = 0;i < 1001;i++)array[i] = new ArrayList<Point>();
	for(int i = 0;i < n;i++){
		st = new StringTokenizer(br.readLine());
		int x = Integer.parseInt(st.nextToken());
		int y = Integer.parseInt(st.nextToken());
		array[(int)Math.ceil(x*1.0/1000)].add(new Point(i + 1,x,y));
	}
	//System.out.println(array[0] + " " + array[1]);
	//for(Point temp : array[0])System.out.println(temp.pos + " " + temp.x + " " + temp.y);
	StringBuilder sb = new StringBuilder();
	for(int i = 0;i < 1001;i++){
		if(array[i].size() == 0)continue;
		Collections.sort(array[i]);
		if(i % 2 == 0){
			for(int j = 0;j < array[i].size();j++)sb.append(array[i].get(j).pos + " ");
		}
		else{
			for(int j = array[i].size() - 1;j >= 0;j--)sb.append(array[i].get(j).pos + " ");
		}
	}
	System.out.print(sb);
}
static class Point implements Comparable<Point>{
	
	int pos;
	int x;
	int y;
	public Point(int pos,int x,int y){
		this.pos = pos;
		this.x = x;
		this.y = y;
	}
	public int compareTo(Point other){
		return - y + other.y;
	}
}
}
