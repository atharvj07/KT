from sys import stdin
input = lambda : stdin.readline().strip()
from math import ceil, sqrt, factorial, gcd
from collections import deque
n = int(input())
l = [0] + list(map(int, input().split()))
s = [0] + list(map(int, input().split()))
ans = 0
visited = [0] * (n + 1)
for i in range(1, n + 1):
	x = i
	while visited[x] == 0:
		visited[x] = i
		x = s[x]
	if visited[x] != i:
		continue
	a = x
	mi = l[x]
	while s[x] != a:
		x = s[x]
		mi = min(mi, l[x])
	ans += mi
print(ans)
