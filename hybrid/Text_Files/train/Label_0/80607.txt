p2 = [(i, 2 ** i - 1) for i in range(1, 40)]
(X, d) = map(int, input().split(' '))
lens = []
for i in p2[::-1]:
	if X >= i[1]:
		X -= i[1]
		lens.append(i[0])
el = 1
if sum(lens) >= 10000:
	print(-1)
	exit(0)
print(sum(lens) + X)
for i in lens:
	print((str(el) + ' ') * i, end='')
	el += d
for i in range(X):
	print(el, end=' ')
	el += d
