import java.util.*;
import java.io.*;
import java.math.*;

   public class x1422E
   {
      public static void main(String hi[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
         String input = infile.readLine();
         int N = input.length();
         char[] arr = input.toCharArray();
         int[] length = new int[N];
         String[] res = new String[N];
         ArrayDeque<Unit> deque = new ArrayDeque<Unit>();
         Stack<Unit> stack = new Stack<Unit>();
         for(int i=N-1; i >= 0; i--)
         {
            if(deque.size() == 0)
               deque.addFirst(new Unit(arr[i], i));
            else if(arr[i] != deque.peekFirst().letter)
            {
               stack.push(new Unit(deque.peekFirst().letter, -1));
               deque.addFirst(new Unit(arr[i], i));
            }
            else
            {
               if(i+1 == deque.peekFirst().pos)
               {
                  while(stack.size() > 0 && arr[i] == stack.peek().letter)
                     stack.pop();
                  if(stack.size() == 0)
                     deque.pollFirst();
                  else if(arr[i] >= stack.peek().letter)
                  {
                     char omkar = stack.peek().letter;
                     deque.pollFirst();
                     //stack.pop();
                  }
                  else
                     deque.addFirst(new Unit(arr[i], i));
               }
               else
                  deque.addFirst(new Unit(arr[i], i));
            }
            length[i] = deque.size();
            res[i] = gen(deque);
         }
         StringBuilder sb = new StringBuilder();
         for(int a=0; a < N; a++)
            sb.append(length[a]+" ").append(res[a]+"\n");
         System.out.print(sb);
      }
      public static String gen(ArrayDeque<Unit> deque)
      {
         if(deque.size() <= 10)
         {
            String res = "";
            int size = deque.size();
            for(int i=0; i < size; i++)
            {
               Unit lol = deque.pollFirst();
               res += lol.letter;
               deque.addLast(lol);
            }
            return res;
         }
         else
         {
            Unit[] head = new Unit[5];
            Unit[] tail = new Unit[2];
            String res = "";
            for(int i=0; i < 5; i++)
               head[i] = deque.pollFirst();
            for(int i=1; i >= 0; i--)
               tail[i] = deque.pollLast();
            for(Unit c: head)
               res += c.letter;
            res += "...";
            for(Unit c: tail)
               res += c.letter;
            for(int i=4; i >= 0; i--)
               deque.addFirst(head[i]);
            for(int i=0; i < 2; i++)
               deque.addLast(tail[i]);
            return res;
         }
      }
   }
   class Unit
   {
      public char letter;
      public int pos;
      
      public Unit(char c, int a)
      {
         letter = c;
         pos = a;
      }
   }
