#include <bits/stdc++.h>
using namespace std;
string name[] = {"Gryffindor", "Hufflepuff", "Ravenclaw", "Slytherin"};
vector<int> dp(4, 0);
bool yes[4];
int n;
set<vector<int> > has;
string gett = "GHRS?", s;
void dfs(int x) {
  if (has.count(dp)) return;
  has.insert(dp);
  int _min = *min_element(dp.begin(), dp.end());
  if (x == n) {
    for (int i = 0; i <= 3; i++) {
      if (_min == dp[i]) {
        yes[i] = 1;
      }
    }
  }
  int id = gett.find(s[x]);
  if (id ^ 4) {
    ++dp[id];
    dfs(x + 1);
    --dp[id];
  } else {
    for (int i = 0; i <= 3; i++) {
      if (_min == dp[i]) {
        ++dp[i];
        dfs(x + 1);
        --dp[i];
      }
    }
  }
}
int main() {
  cin >> n >> s;
  memset(yes, 0, sizeof(yes));
  has.clear();
  dfs(-0);
  for (int i = 0; i <= 3; i++) {
    if (yes[i]) {
      cout << name[i];
      putchar(10);
    }
  }
  return 0;
}
