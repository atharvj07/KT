import sys
t = int(sys.stdin.readline().rstrip())

def solve(p, s):
	a = [0] * 26
	p = [x - 1 for x in p]
	tracker = [0] * len(s)
	tracker[-1] = 1
	for i in p:
		tracker[i] += 1
	for i in range(len(tracker) - 2, -1, -1):
		tracker[i] += tracker[i + 1]
	for i in range(len(tracker)):
		a[ord(s[i]) - ord('a')] += tracker[i]
	print(' '.join([str(i) for i in a]))
for _ in range(t):
	sys.stdin.readline()
	s = sys.stdin.readline().rstrip()
	p = [int(i) for i in sys.stdin.readline().rstrip().split()]
	solve(p, s)
