from collections import defaultdict

def win(a, n):
	arr = dict()
	for i in range(n):
		for j in range(i):
			k = a[i] + a[j]
			if k in arr:
				arr[k].add(i)
				arr[k].add(j)
				if len(arr[k]) == 4:
					s = 'YES' + '\n'
					lst = list(arr[k])
					lst.sort(key=lambda x: a[x])
					(lst[3], lst[1]) = (lst[1], lst[3])
					s += ' '.join((str(elem + 1) for elem in lst))
					if a[lst[0]] + a[lst[1]] == a[lst[2]] + a[lst[3]]:
						return s
			else:
				arr[k] = set([i, j])
	return 'NO'
n = int(input())
a = list(map(int, input().split()))
print(win(a, n))
