import sys
(u, v) = [int(v) for v in sys.stdin.readline().split()]

def m():
	if (v - u) % 2 == 1:
		return None
	ar = []
	h = (v - u) // 2
	if u & h == 0:
		ar.append(u | h)
		ar.append(h)
	else:
		ar.append(u)
		ar.append(h)
		ar.append(h)
	return ar
if u > v:
	print(-1)
elif u == v:
	if u == 0:
		print(0)
	else:
		print(1)
		print(u)
else:
	ar = m()
	if ar is not None:
		print(len(ar))
		print(' '.join(map(str, ar)))
	else:
		print(-1)
