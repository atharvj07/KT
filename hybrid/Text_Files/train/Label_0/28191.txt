#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int MN = 300010;
vector<int> pr;
int pchk[MN];
void sieve() {
  for (int i = 2; i < MN; i++) {
    if (!pchk[i]) {
      pr.push_back(i);
      pchk[i] = i;
    }
    for (int j = 0; j < pr.size() && i * pr[j] < MN; j++) {
      pchk[i * pr[j]] = pr[j];
      if (i % pr[j] == 0) break;
    }
  }
}
int exp(int x, int n) {
  int ret = 1;
  while (n) {
    if (n & 1) ret = 1LL * ret * x % mod;
    x = 1LL * x * x % mod;
    n >>= 1;
  }
  return ret;
}
int inv(int x) { return exp(x, mod - 2); }
int fact[MN], invf[MN], po[MN];
int comb(int n, int k) {
  if (k < 0) return 0;
  if (n < k) return 0;
  return 1LL * fact[n] * invf[k] % mod * invf[n - k] % mod;
}
int psum[MN];
int N;
int A[MN];
vector<int> B[MN];
vector<pair<int, int> > tmp;
int ans;
int main() {
  po[0] = 1;
  for (int i = 1; i < MN; i++) {
    po[i] = po[i - 1] * 2 % mod;
  }
  fact[0] = 1;
  for (int i = 1; i < MN; i++) {
    fact[i] = 1LL * fact[i - 1] * i % mod;
  }
  for (int i = 0; i < MN; i++) {
    invf[i] = inv(fact[i]);
  }
  sieve();
  scanf("%d", &N);
  for (int i = 0; i < N; i++) {
    psum[i] = comb(N - 1, i);
    if (i) {
      psum[i] += psum[i - 1];
      psum[i] %= mod;
    }
  }
  for (int i = 0; i < N; i++) {
    scanf("%d", &A[i]);
  }
  for (int i = 0; i < N; i++) {
    int n = A[i];
    int p = -1;
    int cnt = 0;
    while (n != 1) {
      if (p != pchk[n]) {
        if (p != -1) B[p].push_back(cnt);
        p = pchk[n];
        cnt = 0;
      }
      cnt++;
      n /= pchk[n];
    }
    if (p != -1) B[p].push_back(cnt);
  }
  for (int i = 0; i < pr.size(); i++) {
    int p = pr[i];
    sort(B[p].begin(), B[p].end());
    int cnt = N - B[p].size();
    for (int j = 0; j < B[p].size(); j++) {
      cnt++;
      ans +=
          1LL * (po[N - 1] + mod - 2 * psum[N - cnt - 1] % mod) * B[p][j] % mod;
      ans %= mod;
      ans += mod - 1LL * comb(N - 1, N - cnt) * B[p][j] % mod;
      ans %= mod;
    }
  }
  printf("%d", ans);
}
