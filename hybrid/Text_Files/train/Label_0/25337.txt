#include<cstdio>

char S[300][300+1];
int N;

bool check(int a) {
	for(int i=0;i<N;i++) for(int j=i+1;j<N;j++) {
		if(S[(i+a)%N][j] != S[(j+a)%N][i]) return false;
	}
	return true;
}

int main() {
	scanf("%d", &N);
	for(int i=0;i<N;i++) {
		scanf("%300s", S[i]);
	}
	int count = 0;
	for(int a=0;a<N;a++) {
		if(check(a)) count++;
	}
	printf("%d\n", count * N);
	return 0;
}
