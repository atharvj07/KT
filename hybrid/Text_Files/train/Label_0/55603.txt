import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

/**
 * @author Don Li
 */
public class InnaBabies {
    
    int INF = (int) 1e7;
    
    int n, m;
    int[] x, y;
    
    void solve() {
        n = in.nextInt();
        m = in.nextInt();
        x = new int[n + m];
        y = new int[n + m];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            x[i + n] = in.nextInt();
            y[i + n] = in.nextInt();
        }
        
        for (int i = 0; i < n + m; i++) {
            int a = x[i], b = y[i];
            x[i] = a + b;
            y[i] = a - b;
        }
        
        int lb = 0, ub = INF;
        while (ub - lb > 1) {
            int t = (lb + ub) / 2;
            if (ok(t)) ub = t;
            else lb = t;
        }
        
        if (ok(ub)) out.println(ub);
        else out.println("Poor Sereja!");
    }
    
    boolean ok(int d) {
        Segment[] blue = new Segment[n];
        for (int i = 0; i < n; i++) {
            blue[i] = new Segment();
            blue[i].x1 = blue[i].x2 = x[i];
            blue[i].y1 = y[i] - 2 * d;
            blue[i].y2 = y[i] + 2 * d;
        }
        Arrays.sort(blue, comparatorXY);
        int nb = 0;
        Segment[] blueMerged = new Segment[n];
        for (int i = 0; i < n; i++) {
            if (i == 0 || !blueMerged[nb - 1].hasCommonPoint(blue[i])) blueMerged[nb++] = blue[i];
            else blueMerged[nb - 1].union(blue[i]);
        }
        
        Segment[] red = new Segment[m];
        for (int i = 0; i < m; i++) {
            red[i] = new Segment();
            red[i].y1 = red[i].y2 = y[i + n];
            red[i].x1 = x[i + n] - 2 * d;
            red[i].x2 = x[i + n] + 2 * d;
        }
        Arrays.sort(red, comparatorYX);
        int nr = 0;
        Segment[] redMerged = new Segment[m];
        for (int i = 0; i < m; i++) {
            if (i == 0 || !redMerged[nr - 1].hasCommonPoint(red[i])) redMerged[nr++] = red[i];
            else redMerged[nr - 1].union(red[i]);
        }
        
        boolean[][] connected = new boolean[nb][nb];
        for (int i = 0; i < nr; i++) {
            List<Integer> touched = new ArrayList<>();
            for (int j = 0; j < nb; j++) {
                if (redMerged[i].hasCommonPoint(blueMerged[j])) touched.add(j);
            }
            for (int j = 0; j < touched.size(); j++) {
                for (int k = j + 1; k < touched.size(); k++) {
                    int a = touched.get(j), b = touched.get(k);
                    if (connected[a][b] || connected[b][a]) return true;
                    connected[a][b] = connected[b][a] = true;
                }
            }
        }
        return false;
    }
    
    Comparator<Segment> comparatorXY = (u, v) -> {
        if (u.x1 != v.x1) return Integer.compare(u.x1, v.x1);
        return Integer.compare(u.y1, v.y1);
    };
    
    Comparator<Segment> comparatorYX = (u, v) -> {
        if (u.y1 != v.y1) return Integer.compare(u.y1, v.y1);
        return Integer.compare(u.x1, v.x1);
    };
    
    static class Segment {
        int x1, y1, x2, y2;
        
        boolean hasCommonPoint(Segment s) {
            if (x1 > s.x2) return false;
            if (x2 < s.x1) return false;
            if (y1 > s.y2) return false;
            if (y2 < s.y1) return false;
            return true;
        }
        
        void union(Segment s) {
            x1 = Math.min(x1, s.x1);
            x2 = Math.max(x2, s.x2);
            y1 = Math.min(y1, s.y1);
            y2 = Math.max(y2, s.y2);
        }
    }
    
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new InnaBabies().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
