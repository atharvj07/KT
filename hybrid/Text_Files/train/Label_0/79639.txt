from sys import stdin, stdout

from random import randint
import math
import sys
sys.setrecursionlimit(10 ** 7)
n = int(input())
a = list(map(int, input().split()))
Left = 0
Right = n - 1
while Left != n - 1 and a[Left + 1] > a[Left]:
	Left += 1
while Right != 0 and a[Right - 1] > a[Right]:
	Right -= 1
Left += 1
Right = n - Right
if Left % 2 or Right % 2:
	print('Alice')
else:
	print('Bob')
