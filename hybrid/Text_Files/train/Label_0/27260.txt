#include <bits/stdc++.h>
using namespace std;
int calculateOperations(char from, char to) {
  int x = abs(to - from);
  if (x > 13)
    return 26 - x;
  else
    return x;
}
int operations(string s) {
  int res = INT_MAX;
  for (int i = 0; i < s.size() - 3; i++) {
    int count = 1;
    int sum = 0;
    for (int j = i; j < i + 4; j++) {
      char ch = s[j];
      if (count == 1) sum += calculateOperations(ch, 'A');
      if (count == 2) sum += calculateOperations(ch, 'C');
      if (count == 3) sum += calculateOperations(ch, 'T');
      if (count == 4) sum += calculateOperations(ch, 'G');
      count++;
    }
    res = min(res, sum);
  }
  return res;
}
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  cout << operations(s) << endl;
  return 0;
}
