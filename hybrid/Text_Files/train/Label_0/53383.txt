import java.io.*;
import java.util.*;
public class Solve4 {
 
  public static void main(String[] args) {
	MyScanner scanner = new MyScanner();
	PrintWriter out = new PrintWriter(new BufferedOutputStream(System.out));
	int n = scanner.nextInt();
	int[] t = new int[2 * n];
	long[] res = new long[n];
	HashMap<Integer, Integer> map = new HashMap<Integer, Integer>();
	HashMap<Integer, Integer> rmap = new HashMap<Integer, Integer>();
	HashMap<Integer, Integer> ind = new HashMap<Integer, Integer>();
    for(int i = 0; i < n; i++) {
	  int l = scanner.nextInt();
	  int r = scanner.nextInt();
	  t[i] = l;
	  t[i + n] = r;
	  map.put(l, r);
	  map.put(r, l);
	  rmap.put(l, i);
	  rmap.put(r, i);
	}
	Arrays.sort(t);
	for(int i = 0; i < 2 * n; i ++) {
	  ind.put(t[i], i);
	}
	Tree tree = new Tree(2 * n + 1);
	for(int i = 0; i < 2 * n; i++) {
	  int k = t[i];
	  int a = map.get(k);
	  int b = ind.get(a);
	  if(a < k) {
	    long query = tree.query(b, i);
	    //System.out.println(k + " " + query);
	    res[rmap.get(k)] = query;
	    tree.update(b, 1);
	  }
	}
	for(int i = 0; i < n; i++) {
	  out.println(res[i]);
	}
	out.close();
  }

   static  class Tree{
	int n;
	long[] d;
	Tree(int nmin) {
		for(n = 1; n < nmin; n *= 2);
		d = new long[2 * n];
		Arrays.fill(d, 0);
	}
	void update(int i, long x) {
		d[n + i] = x;
		for(int k = (n + i) / 2; k > 0; k >>= 1)
			d[k] = d[k * 2] + d[k * 2 + 1];
	}
	long get(int i) { return d[n + i]; }
	//[l, r)
	long query(int l, int r) {
		long m = 0;
		for(; l != 0 && l + (l&-l) <= r; l += l&-l)
			m = m + d[(n + l) / (l&-l)];
		for(; l < r; r -= r&-r)
			m = m + d[(n + r) / (r&-r) - 1];
		return m;
	}
   }
  
  public static class MyScanner {
      BufferedReader br;
      StringTokenizer st;
 
      public MyScanner() {
         br = new BufferedReader(new InputStreamReader(System.in));
      }
 
      String next() {
          while (st == null || !st.hasMoreElements()) {
              try {
                  st = new StringTokenizer(br.readLine());
              } catch (IOException e) {
                  e.printStackTrace();
              }
          }
          return st.nextToken();
      }
 
      int nextInt() {
          return Integer.parseInt(next());
      }
 
      long nextLong() {
          return Long.parseLong(next());
      }
 
      double nextDouble() {
          return Double.parseDouble(next());
      }
 
      String nextLine(){
          String str = "";
	  try {
	     str = br.readLine();
	  } catch (IOException e) {
	     e.printStackTrace();
	  }
	  return str;
      }

   }
}
