#include <bits/stdc++.h>
using namespace std;
const int maxn = 100010;
int n, m;
int gi() {
  char a = getchar();
  int b = 0;
  while (a < '0' || a > '9') a = getchar();
  while (a >= '0' && a <= '9') b = b * 10 + a - '0', a = getchar();
  return b;
}
int a[maxn];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  int cnt = 0;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  for (int i = 0; i < n; i++) {
    if (a[i] == a[0]) cnt++;
  }
  if (cnt <= n / 2)
    cout << "Alice\n";
  else
    cout << "Bob\n";
}
