#include <bits/stdc++.h>
using namespace std;
int read() {
  int n = 0;
  bool b = 0;
  char c = getchar();
  while (c > '9' || c < '0') {
    if (c == '-') b = 1;
    c = getchar();
  }
  while (c <= '9' && c >= '0') {
    n = (n << 1) + (n << 3) + (c ^ 48);
    c = getchar();
  }
  return b ? -n : n;
}
const int mod = 1e9 + 7, N = 1e5 + 7, K = 207;
struct node {
  int to, next;
} e[N * 2];
int head[N], cnt, Siz[N], dp[N][K], ans[N], f[K], S[K][K];
int n, k;
int Add(int a, int b) { return a + b >= mod ? a + b - mod : a + b; }
int del(int a, int b) { return a >= b ? a - b : a + mod - b; }
int mul(int a, int b) { return 1ll * a * b % mod; }
void add(int u, int v) {
  e[++cnt].to = v;
  e[cnt].next = head[u];
  head[u] = cnt;
}
void dfs(int x, int fa) {
  Siz[x] = 1;
  dp[x][0] = 2;
  for (int i = head[x]; i; i = e[i].next) {
    int v = e[i].to;
    if (v == fa) continue;
    dfs(v, x);
    for (int i = 0; i <= k; i++) f[i] = 0;
    for (int i = 0; i <= min(k, Siz[x]); i++)
      for (int j = 0; j <= min(k - i, Siz[v]); j++)
        f[i + j] = Add(f[i + j], mul(dp[x][i], dp[v][j]));
    for (int i = 0; i <= k; i++)
      dp[x][i] = f[i], ans[i] = del(ans[i], dp[v][i]);
    Siz[x] += Siz[v];
  }
  for (int i = 0; i <= k; i++) ans[i] = Add(ans[i], dp[x][i]);
  for (int i = k; i; i--) dp[x][i] = Add(dp[x][i], dp[x][i - 1]);
  dp[x][1] = del(dp[x][1], 1);
}
signed main() {
  n = read(), k = read();
  S[1][1] = 1;
  for (int i = 2; i <= n; i++) {
    int x = read(), y = read();
    add(x, y);
    add(y, x);
  }
  for (int i = 2; i <= k; i++)
    for (int j = 1; j <= i; j++)
      S[i][j] = Add(S[i - 1][j - 1], mul(j, S[i - 1][j]));
  dfs(1, 0);
  int res = 0;
  for (int i = 0, fac = 1; i <= k; i++, fac = mul(fac, i))
    res = Add(res, mul(S[k][i], mul(fac, ans[i])));
  printf("%d", res);
}
