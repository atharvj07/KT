inf = 10 ** 16
n = int(input())
aa = list(map(int, input().split()))
ans = inf
if n > 70:
	print(1)
	exit()
for i in range(1, n - 1):
	ll = []
	mx = -inf
	x = 0
	for j in range(i - 1, -1, -1):
		x ^= aa[j]
		if x > mx:
			mx = x
			ll.append((x, i - 1 - j))
	rr = []
	mn = inf
	x = 0
	for j in range(i, n):
		x ^= aa[j]
		if x < mn:
			mn = x
			rr.append((x, j - i))
	j = 0
	for (x, c) in ll[::-1]:
		while j < len(rr) and rr[j][0] >= x:
			j += 1
		if j == len(rr):
			break
		cur = c + rr[j][1]
		ans = min(ans, cur)
print(-1) if ans == inf else print(ans)
