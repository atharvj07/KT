import java.io.*;
import java.util.*;

public class F implements Runnable {
	public static void main (String[] args) {new Thread(null, new F(), "_cf", 1 << 28).start();}

	public void run() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("");

		int n = fs.nextInt();
		int[] vals = fs.nextIntArray(n);
		int[][] adj = new int[n][];
		int lcm = -1;
		for(int i = 0; i < n; i++) {
			int x = fs.nextInt();
			if(lcm == -1) lcm = x;
			else lcm = lcm(lcm, x);
			adj[i] = fs.nextIntArray(x);
			for(int j = 0; j < x; j++) adj[i][j]--;
		}
		
		int[][] vis = new int[n][lcm];
		int[][] size = new int[n][lcm];
		
		int LEN = 2*n*lcm+10;
		int[] stack2 = new int[LEN];
		int[] stack = new int[LEN];
		int[] queue = new int[LEN];
		int[] lastTime = new int[n];
		int curTime = 0;
		
		for(int i = 0; i < n; i++) {
			for(int j = 0; j < lcm; j++) {
				if(vis[i][j] != 0) continue;
				curTime++;
				int tail = 0;
				int tail2 = 0;
				int tail3 = 0;
				queue[tail++] = i;
				queue[tail++] = j;
				vis[i][j] = 1;
				for(int k = 0; k < tail; k += 2) {
					int u = queue[k], time = queue[k+1];
					stack[tail2++] = u; stack[tail2++] = time;
					
					
					int idx = time % adj[u].length;
					int v = adj[u][idx];
					int time2 = time+vals[v];
					time2 %= lcm;
					while(time2 < 0) time2 += lcm;
					
					if(vis[v][time2] == 0) {
						vis[v][time2] = 1;
						queue[tail++] = v; queue[tail++] = time2;
					}
					else if(vis[v][time2] == 1) {
						int sz = 0;
						while(tail2 > 0) {
							tail2 -= 2;
							int uu = stack[tail2], time3 = stack[tail2+1];
							
							if(lastTime[uu] != curTime) {
								sz++;
								lastTime[uu] = curTime;
							}
							
							vis[uu][time3] = 2;
							stack2[tail3++] = uu; stack2[tail3++] = time3;
							if(uu == v && time3 == time2) break;
						}
						while(tail3 > 0) {
							tail3 -= 2;
							int uu = stack2[tail3], time3 = stack2[tail3+1];
							size[uu][time3] = sz;
						}
					}
				}
				
				while(tail2 > 0) {
					tail2 -= 2;
					int u = stack[tail2], time = stack[tail2+1];

					int idx = time % adj[u].length;
					int v = adj[u][idx];
					int time2 = time+vals[v];
					time2 %= lcm;
					while(time2 < 0) time2 += lcm;

					vis[u][time] = 2;
					size[u][time] = size[v][time2];
				}
				
			}
		}
		
		int q = fs.nextInt();
		while(q-->0) {
			int x = fs.nextInt()-1;
			int y = fs.nextInt();
			y += vals[x];
			y %= lcm;
			while(y < 0) y += lcm;
			out.println(size[x][y]);
		}
		
		out.close();
	}
	
	int lcm(int a, int b) {
		return a * b / gcd(a, b);
	}
	
	int gcd(int a, int b) {
		if(b == 0) return a;
		return gcd(b, a % b);
	}
	
	class Pair {
		int x, y;
		Pair(int a, int b) {
			x = a; y = b;
		}
		public String toString() {
			return String.format("(%d, %d)", x, y);
		}
	}
	
	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) {
			try {
				in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
			}
			catch (Exception e) {
				in = new BufferedInputStream(System.in, BS);
			}
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}
		
		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}
		
	}

}
