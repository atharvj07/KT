(n, m) = map(int, input().split())
im = []
w = 0
for i in range(n):
	im.append(list(input()))
if n == 1 or m == 1:
	print(0)
	quit()
for i in range(n - 1):
	for j in range(m - 1):
		x = [im[i][j], im[i][j + 1], im[i + 1][j], im[i + 1][j + 1]]
		x.sort()
		if x == ['a', 'c', 'e', 'f']:
			w += 1
print(w)
