#include <bits/stdc++.h>
using namespace std;
long double t[200005], ti[200005], st[200005], aa[200005];
long double dp[60][200005];
long double inf = 1e22;
void solv(int i, int j, int x, int y, int l) {
  int mid = (i + j) / 2;
  long double mans = inf, temp, sum = 0, mv = -1;
  int v = min(y, mid);
  for (; v >= x; v--) {
    sum = aa[mid] - aa[v - 1] - st[v - 1] * (ti[mid] - ti[v - 1]);
    temp = dp[l - 1][v - 1] + sum;
    if (mans > temp) {
      mans = temp;
      mv = v;
    }
  }
  dp[l][mid] = mans;
  if (mid + 1 <= j) {
    solv(mid + 1, j, mv, y, l);
  }
  if (mid - 1 >= i) {
    solv(i, mid - 1, x, mv, l);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k;
  cin >> n >> k;
  ti[0] = 0;
  st[0] = 0;
  aa[0] = 0;
  for (int i = 1; i <= n; i++) {
    cin >> t[i];
    st[i] = st[i - 1] + t[i];
    ti[i] = ti[i - 1] + 1 / t[i];
    aa[i] += aa[i - 1] + st[i] / t[i];
  }
  for (int i = 1; i <= n; i++) dp[0][i] = inf;
  dp[0][0] = 0;
  for (int i = 1; i <= k; i++) {
    dp[i][0] = inf;
    solv(1, n, 1, n, i);
  }
  cout << fixed << setprecision(8) << dp[k][n];
  return 0;
}
