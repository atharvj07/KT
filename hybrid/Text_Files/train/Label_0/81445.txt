n = int(input())
d = {}
l = list(map(int, input().split()))
val = 0
ma = 0
for i in range(n):
	d[l[i]] = 1 + d.get(l[i] - 1, 0)
	if d[l[i]] > ma:
		ma = d[l[i]]
		val = l[i]
print(ma)
ans = []
for i in range(n - 1, -1, -1):
	if l[i] == val:
		ans.append(i + 1)
		val -= 1
ans = ans[::-1]
print(' '.join(map(str, ans)))
