import java.util.*;

{
    public static void main(String[]stp) throws Exception
    {
    	Scanner scan=new Scanner(System.in);
        //BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        //String[] s;
        int baki=0;
        String na=scan.next();
        int row=(na.length())/20;
        if(na.length()%20!=0) row++;
        int col=(na.length())/row;
        if(na.length()%row!=0) col++;
        int abst=((row*col)-na.length())/row;
        if(((row*col)-na.length()) % row!=0) baki=((row*col)-na.length()) % row;
        int count=abst;
        System.out.println(row+" "+col);
        int k=0,i=0,j;
        for(i=0;i<row;i++) 
        {
        	count=abst;
        	if(baki>0) {count++; baki--; }
        	for(j=0;j<col;j++) 
        	{
        		if(count!=0) {System.out.print("*"); count--; }
        		else System.out.print(na.charAt(k++));
        	}
        	System.out.println();
        }
    }
}
