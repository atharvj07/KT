import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.TreeMap;
import java.util.Map;
import java.util.Scanner;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author ky112233
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskC1 solver = new TaskC1();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskC1 {
        static Map<TaskC1.Pair, TaskC1.Pair> map = new TreeMap<>();

        public void solve(int testNumber, Scanner in, PrintWriter out) {
            long n = in.nextLong();
            if (n == 0) {
                out.println(0);
                return;
            }
            TaskC1.Pair res = dfs(new TaskC1.Pair(0, n));
            out.println(res.x);
        }

        private TaskC1.Pair dfs(TaskC1.Pair p) {
            if (map.containsKey(p)) return map.get(p);
            if (p.y < 10) {
                TaskC1.Pair res = new TaskC1.Pair(1, Math.min(0, p.y - p.x));
                map.put(p, res);
                return res;
            }
            long d = 1;
            while (d <= p.y / 10) d *= 10;
            TaskC1.Pair small = dfs(new TaskC1.Pair(Math.max(p.x, p.y / d), p.y % d));
            TaskC1.Pair large = dfs(new TaskC1.Pair(p.x, p.y / d * d + small.y));
            TaskC1.Pair res = new TaskC1.Pair(small.x + large.x, large.y);
            map.put(p, res);
            return res;
        }

        static class Pair implements Comparable<TaskC1.Pair> {
            public long x;
            public long y;

            public Pair(long x, long y) {
                this.x = x;
                this.y = y;
            }

            public int compareTo(TaskC1.Pair p) {
                if (x != p.x) return (int) Math.signum(x - p.x);
                return (int) Math.signum(y - p.y);
            }

        }

    }
}
