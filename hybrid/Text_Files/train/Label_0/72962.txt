from collections import Counter
recept = input().strip()
(nb, ns, nc) = map(int, input().split())
(pb, ps, pc) = map(int, input().split())
t = int(input())
(l, r) = (0, 10 ** 21)
(rb, rs, rc) = (0, 0, 0)
for s in recept:
	if s == 'B':
		rb += 1
	elif s == 'S':
		rs += 1
	else:
		rc += 1

def check(tmp):
	return max(tmp * rb - nb, 0) * pb + max(tmp * rs - ns, 0) * ps + max(tmp * rc - nc, 0) * pc <= t
while l != r - 1:
	tmp = (l + r) // 2
	if check(tmp):
		l = tmp
	else:
		r = tmp
print(l)
