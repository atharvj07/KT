#include <iostream>

const int ms = 200200;

int freq[ms];

int main() {
	int n;
	std::cin >> n;
	int ans = 0;
	while(n--) {
		int temp;
		std::cin >> temp;
		if(temp >= ms) {
			ans++;
		} else {
			freq[temp]++;
		}
	}
	for(int i = 0; i < ms; i++) {
		if(freq[i] >= i) {
			freq[i] -= i;
		}
		ans += freq[i];
	}
	std::cout << ans << '\n';
}
