#include <bits/stdc++.h>
using namespace std;
const int inf = (int)1e9;
const int mod = inf + 7;
const double eps = 1e-9;
const double pi = acos(-1.0);
int n;
bool ok = false;
int main() {
  cin >> n;
  string s;
  cin >> s;
  while (!ok) {
    ok = true;
    for (int i = 0; i < n; ++i) {
      if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' ||
          s[i] == 'u' || s[i] == 'y') {
        if ((i + 1 < n) &&
            (s[i + 1] == 'a' || s[i + 1] == 'e' || s[i + 1] == 'i' ||
             s[i + 1] == 'o' || s[i + 1] == 'u' || s[i + 1] == 'y')) {
          s.erase(i + 1, 1);
          ok = false;
        }
      }
    }
  }
  cout << s << endl;
}
