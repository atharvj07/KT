for i in range(int(input())):
	a = int(input())
	l = list(map(int, input().split()))
	z = []
	for t in range(a):
		if t % 2 == 1:
			z.append(l[t])
		else:
			z.append(1)
	bb = []
	for i in range(a):
		if i % 2 == 0:
			bb.append(l[i])
		else:
			bb.append(1)
	if sum(z) <= sum(l) // 2:
		print(*bb)
	else:
		print(*z)
