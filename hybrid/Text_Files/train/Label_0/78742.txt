import sys

def input():
	return sys.stdin.readline().rstrip()
INF = 1 << 128

def slv():
	n = int(input())
	a = [0] * n
	b = [0] * n
	for i in range(n):
		(h, c) = map(int, input().split())
		a[i] = h
		b[i] = c
	dp = [[INF] * 5 for i in range(n + 1)]
	for i in range(5):
		dp[1][i] = b[0] * i
	for i in range(1, n):
		prev_height = a[i - 1]
		(height, cost) = (a[i], b[i])
		for x in range(5):
			for y in range(5):
				if prev_height + x != height + y:
					dp[i + 1][y] = min(dp[i + 1][y], dp[i][x] + cost * y)
	ans = min((dp[n][i] for i in range(5)))
	print(ans)
	return

def main():
	t = int(input())
	for i in range(t):
		slv()
	return
main()
