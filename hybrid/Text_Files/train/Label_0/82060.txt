number = input()
answer = int(number)
for i in range(len(number)):
	if not i == len(number) - 1:
		if answer > answer + (9 - int(number[len(number) - i - 1])) * 10 ** i - int(number[len(number) - i - 1]) * 10 ** i:
			answer = answer + (9 - int(number[len(number) - i - 1])) * 10 ** i - int(number[len(number) - i - 1]) * 10 ** i
	if i == len(number) - 1:
		if answer > answer + (9 - int(number[len(number) - i - 1])) * 10 ** i - int(number[len(number) - i - 1]) * 10 ** i and 9 - int(number[len(number) - i - 1]) > 0:
			answer = answer + (9 - int(number[len(number) - i - 1])) * 10 ** i - int(number[len(number) - i - 1]) * 10 ** i
print(answer)
