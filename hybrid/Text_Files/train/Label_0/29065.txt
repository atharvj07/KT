#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  for (int t = 0; t < T; t++) {
    long long N, M;
    cin >> N >> M;
    vector<pair<long long, long long>> v(M);
    for (int i = 0; i < M; i++) cin >> v[i].first >> v[i].second;
    sort(v.begin(), v.end());
    vector<long long> a(M), b(M);
    vector<long long> s(M, 0);
    for (int i = M - 1; i >= 0; i--) {
      a[i] = v[i].first;
      b[i] = v[i].second;
      s[i] = a[i];
      if (i + 1 < M) s[i] += s[i + 1];
    }
    long long res = 0;
    for (int i = 0; i < M; i++) {
      long long r = a[i];
      long long index = upper_bound(a.begin(), a.end(), b[i]) - a.begin();
      if (index + N < M) index = M - N;
      if (index <= i) {
        r = s[index];
        r += b[i] * (N - (M - index));
      } else {
        if (index + N - 1 < M) index = M - (N - 1);
        r = a[i];
        if (index < M) r += s[index];
        r += b[i] * (N - 1 - (M - index));
      }
      if (r > res) res = r;
    }
    cout << res << "\n";
  }
  return 0;
}
