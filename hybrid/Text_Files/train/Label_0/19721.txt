#include <bits/stdc++.h>
using namespace std;
bool check(long long *a, long long *b, long long n) {
  for (int i = 0; i < n; i++)
    if (a[i] != b[i]) return false;
  return true;
}
int main() {
  long long n, m;
  cin >> n >> m;
  long long a[n], b[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  long long ans = INT_MAX;
  sort(a, a + n);
  sort(b, b + n);
  long long x = 0;
  for (int i = 0; i < n; i++) {
    x = b[0] - a[i];
    if (x < 0) x += m;
    long long c[n];
    for (int j = 0; j < n; j++) c[j] = (a[j] + x) % m;
    sort(c, c + n);
    if (check(c, b, n)) ans = min(ans, x);
  }
  cout << ans << endl;
  return 0;
}
