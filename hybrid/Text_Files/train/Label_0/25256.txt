#include <bits/stdc++.h>
using namespace std;
int t[100001];
void solve() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> t[i];
  for (int i = 1; i <= n; ++i) {
    int first;
    cin >> first;
    t[i] = first - t[i];
  }
  bool ok = 1;
  int nr = 0, elem = 0;
  for (int i = 1; i <= n && ok; ++i) {
    if (t[i] < 0) ok = 0;
    if (t[i] != t[i - 1]) {
      if (t[i] > 0) ++elem;
    }
  }
  if (!ok || elem > 1)
    cout << "NO\n";
  else
    cout << "YES\n";
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int q;
  cin >> q;
  while (q--) solve();
  return 0;
}
