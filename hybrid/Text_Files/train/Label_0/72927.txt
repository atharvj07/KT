from collections import defaultdict

def ball(a, b, k, boys, girls):
	boy = defaultdict(lambda : 0)
	girl = defaultdict(lambda : 0)
	for i in range(k):
		boy[boys[i]] += 1
		girl[girls[i]] += 1
	temp = 0
	for z in boy.keys():
		if boy[z] >= 2:
			temp += boy[z] * (boy[z] - 1) // 2
	for z in girl.keys():
		if girl[z] >= 2:
			temp += girl[z] * (girl[z] - 1) // 2
	total = k * (k - 1) // 2
	return total - temp
t = int(input())
for x in range(t):
	(a, b, k) = input().split()
	a = int(a)
	k = int(k)
	b = int(b)
	boys = [int(i) for i in input().split()][:k]
	girls = [int(i) for i in input().split()][:k]
	print(ball(a, b, k, boys, girls))
