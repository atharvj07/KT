import java.util.Scanner;
import java.util.List;
import java.util.ArrayList;

public class T1{
    public T1(){
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int k = 0;
        long sum = 0;
        int arr[] = new int[n];
        int ans[] = new int[n];
        for(int i = 0; i < n; i++){
            arr[i] = s.nextInt();
            sum += arr[i];
        }
        for(int i = 0; i < n; i++){
            double res = 1.0 * (sum - arr[i]) / (n - 1);
            if(res == arr[i]){
                ans[k] = i + 1;
                k++;
            }
        }
        System.out.println(k);
        for(int i = 0; i < k; i++)
            System.out.print(ans[i] + " ");
    }
    public static void main(String args[]){
        T1 t1 = new T1();
    }
}
