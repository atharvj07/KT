from sys import stdin

def mp():
	return map(int, stdin.readline().split())

def ml():
	return list(map(int, stdin.readline().split()))
(n, m) = mp()
(x, k, y) = mp()
a = ml()
b = ml()

def delSeg(st, end, mx):
	cnt = end - st
	aend = max(a[end], a[st - 1] if st - 1 > 0 else 0) if end < n else a[st - 1]
	if cnt == 0:
		return 0
	if cnt < k and mx > aend:
		return -1
	if x <= k * y:
		return cnt // k * x + cnt % k * y
	elif mx < aend:
		return cnt * y
	else:
		return x + (cnt - k) * y
prev = 0
j = 0
mx = -1
(ans, flag) = (0, True)
for i in range(n):
	if j < m and a[i] == b[j]:
		ret = delSeg(prev, i, max(a[prev:i]) if prev != i else 0)
		if ret == -1:
			flag = False
			break
		else:
			ans += ret
		j += 1
		prev = i + 1
	elif j >= m:
		mx = max(a[prev:n])
		ret = delSeg(prev, n, mx)
		if ret == -1:
			flag = False
			break
		else:
			ans += ret
		break
	i += 1
if flag and j >= m:
	print(ans)
else:
	print(-1)
