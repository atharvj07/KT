import sys
input = sys.stdin.readline

def minCost(fences, costs):
	n = len(fences)
	cur = [0, costs[0], 2 * costs[0]]
	for i in range(1, n):
		nxt = [float('inf'), float('inf'), float('inf')]
		for add in range(3):
			mi = float('inf')
			for x in range(3):
				if add + fences[i] != x + fences[i - 1]:
					mi = min(mi, cur[x])
			nxt[add] = add * costs[i] + mi
		cur = nxt
	return min(cur)

def main():
	lq = int(input())
	for _ in range(lq):
		n = int(input())
		fences = []
		costs = []
		for _ in range(n):
			(a, b) = input().split()
			fences.append(int(a))
			costs.append(int(b))
		print(minCost(fences, costs))
main()
