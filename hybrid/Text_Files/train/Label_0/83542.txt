import sys
from collections import defaultdict

def dfs(start, graph, b):
	vis = defaultdict(int)
	a = set()
	q = [start]
	vis[start] = 1
	while q:
		cur = q.pop()
		for i in graph[cur]:
			if vis[i] == 0 and i != b:
				a.add(i)
				vis[i] = 1
				q.append(i)
	return a
t = int(sys.stdin.readline())
for _ in range(t):
	(n, m, a, b) = map(int, sys.stdin.readline().split())
	graph = defaultdict(list)
	for i in range(m):
		(u, v) = map(int, sys.stdin.readline().split())
		graph[u].append(v)
		graph[v].append(u)
	seta = dfs(a, graph, b)
	setb = dfs(b, graph, a)
	setc = seta.intersection(setb)
	seta = seta - setc
	setb = setb - setc
	ans = len(seta) * len(setb)
	print(ans)
