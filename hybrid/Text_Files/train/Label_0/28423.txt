#include <bits/stdc++.h>
using namespace std;
string str;
struct STR {
  string name;
  int cont;
  int dep;
  int id;
  bool vis;
};
STR strche[1000006];
int cnt;
int maxdep;
void dfs(int step, int num) {
  strche[num].vis = true;
  strche[num].dep = step;
  maxdep = max(maxdep, step);
  if (strche[num].cont == 0) return;
  for (int i = num + 1; i <= cnt; ++i) {
    if (!strche[i].vis) {
      strche[i].vis = true;
      --strche[num].cont;
      dfs(step + 1, i);
    }
    if (!strche[num].cont) return;
  }
}
bool cmp(STR p1, STR p2) {
  return (p1.dep == p2.dep) ? (p1.id < p2.id) : (p1.dep < p2.dep);
}
int main() {
  int num = 0;
  cin >> str;
  string strin = "";
  for (int i = 0; i < str.length(); ++i) {
    if (str[i] != ',' && (str[i] < '0' || str[i] > '9')) {
      strin += str[i];
      num = 0;
    } else if (str[i] != ',' && str[i] >= '0' && str[i] <= '9') {
      num = num * 10 + str[i] - '0';
    } else if (str[i] == ',') {
      if (str[i - 1] >= '0' && str[i - 1] <= '9') {
        strche[cnt].cont = num;
        num = 0;
      } else {
        strche[++cnt].name = strin;
        strche[cnt].id = cnt;
        strin = "";
      }
    }
  }
  for (int i = 1; i <= cnt; ++i)
    if (!strche[i].vis) dfs(1, i);
  sort(strche + 1, strche + cnt + 1, cmp);
  cout << maxdep << endl;
  strche[0].dep = 1;
  for (int i = 1; i <= cnt; ++i) {
    if (strche[i].dep != strche[i - 1].dep) cout << endl;
    cout << strche[i].name << " ";
  }
  return 0;
}
