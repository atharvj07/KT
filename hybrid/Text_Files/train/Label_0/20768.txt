#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n = 14;
  long long m, i, j, max = 0, ind, ms = 0, sum = 0, temp, temp2, temp3, cnt;
  long long a[n], b[n];
  for (i = 0; i < n; i++) cin >> a[i];
  for (i = 0; i < n; i++) {
    for (j = 0; j < n; j++) b[j] = a[j];
    b[i] = 0;
    temp = a[i];
    ind = i;
    ms = 0;
    temp3 = temp / 14;
    temp2 = temp % 14;
    cnt = 14;
    while (cnt != 0) {
      cnt--;
      ind++;
      if (ind == 14) ind = 0;
      b[ind] += temp3;
    }
    while (temp2 != 0) {
      temp2--;
      ind++;
      if (ind == 14) ind = 0;
      b[ind]++;
    }
    for (j = 0; j < n; j++) {
      if (b[j] % 2 == 0) ms = ms + b[j];
    }
    if (ms > sum) sum = ms;
  }
  cout << sum;
  return 0;
}
