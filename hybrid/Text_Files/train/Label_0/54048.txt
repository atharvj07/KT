import java.io.*;
import java.util.*;

public class p417e
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x[] = solve(n);
        int y[] = solve(m);
        for (int i=0;i<n;i++)
        {
            for (int j=0;j<m;j++)
            {
                System.out.print(x[i]*y[j]+" ");
            }
            System.out.println();
        }
    }

    public static int[] solve(int n)
    {
        if(n==1) return new int[]{1};
        if(n==2) return new int[]{3,4};
        int[] a = new int[n];
        Arrays.fill(a,1);
        if(n%2==1)
        {
            a[0] = 2;
            a[n-1] = (n+1)/2;
        }
        else a[n-1] = (n-2)/2;
        return a;
    }
}
