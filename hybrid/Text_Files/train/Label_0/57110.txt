import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class Main {

  private void solve() throws IOException {
    int[] x = new int[8];
    int[] y = new int[8];
    int[] xo = new int[8];
    int[] yo = new int[8];


    for (int i = 0; i < 8; ++i) {
      xo[i] = x[i] = ni();
      yo[i] = y[i] = ni();
    }

    for (int i = 0; i < 8; ++i) {
      for (int j = i + 1; j < 8; ++j) {
        if (xo[i] == xo[j] && yo[i] == yo[j]) {
          prln("ugly");
          return;
        }
      }
    }

    Arrays.sort(x);
    Arrays.sort(y);

    int i = 1;
    for (i = 1; i < 3; ++i) {
      if (x[i] != x[i - 1] || y[i] != y[i - 1]) {
        prln("ugly");
        return;
      }
    }

    for (i = 4; i < 5; ++i) {
      if (x[i] != x[i - 1] || y[i] != y[i - 1]) {
        prln("ugly");
        return;
      }
    }

//    for (int j = 0; j < 8 ; ++j) {
//      if (xo[j] == x[3] || xo[j] == x[4]) {
//        if (xo[j] == y[3] || xo[j] == y[4]) {
//          prln("ugly");
//          return;
//        }
//      }
//    }

    for (i = 6; i < 8; ++i) {
      if (x[i] != x[i - 1] || y[i] != y[i - 1]) {
        prln("ugly");
        return;
      }
    }

    prln("respectable");

  }

  public static void main(String ... args) throws IOException { new Main().run(); }

  private PrintWriter pw;
  private BufferedReader br;
  private StringTokenizer st;

  private void run() throws IOException {
    pw = new PrintWriter(System.out);
    br = new BufferedReader(new InputStreamReader(System.in));
    solve();
    br.close();
    pw.close();
  }

  private void pr(Object o) { pw.print(o); }
  private void prln(Object o) {pw.println(o);}
  private void prln() {pw.println();}

  private long nl() throws IOException { return Long.parseLong(nt());	}

  private double nd() throws IOException  { return Double.parseDouble(nt()); }

  private int ni()throws IOException  { return Integer.parseInt(nt()); }

  private String nt() throws IOException {
    if (st == null || !st.hasMoreTokens())
      st = new StringTokenizer(br.readLine());
    return st.nextToken();
  }
}
