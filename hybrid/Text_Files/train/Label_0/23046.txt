#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>
using namespace std;
#define M 200005
#define N 1000005
#define ll long long
struct SGT
{
	int A[N];
	ll S[N];
	void Insert(int x, int v) {
		for (int i = x; i <= 1000001; i += i & -i)
			A[i] += v, S[i] += (ll)v * x;
	}
	int Find(int x) {
		int s = 0;
		for (int i = 20; i >= 0; i--)
		{
			if (s + (1 << i) > 1000001)
				continue;
			if (x > A[s + (1 << i)]) {
				s += 1 << i;
				x -= A[s];
			}
		}
		return s + 1;
	}
	int Query0(int x) {
		int ret = 0;
		for (int i = x; i; i -= i & -i)
			ret += A[i];
		return ret;
	}
	ll Query1(int x) {
		ll ret = 0;
		for (int i = x; i; i -= i & -i)
			ret += S[i];
		return ret;
	}
	int Query0(int l, int r) {return Query0(r) - Query0(l - 1);}
	ll Query1(int l, int r) {return Query1(r) - Query1(l - 1);}
} S1, S2;
int n, m, q, A[M], B[M], mnn;
ll Calc0(int v, int k) {
	int t = v - k;
	int t0 = max(1, min(1000001, t));
	ll ans = -(S2.Query1(t0, 1000001) - 1ll * S2.Query0(t0, 1000001) * t);
	t = min(t, mnn);
	ans += S1.Query1(1, 1000001) - 1ll * n * t;
	return ans;
}
ll Calca(int x, int k) {return Calc0(x, k) - max(x - mnn, x - x + k) + k;}
ll Calcb(int x, int k) {return Calc0(x, k) - max(0, x - x + k) + k;}
int main()
{
	scanf("%d%d%d", &n, &m, &q);
	for (int i = 1; i <= n; i++)
	{
		scanf("%d", &A[i]), A[i]++;
		S1.Insert(A[i], 1);
	}
	for (int i = 1; i <= m; i++)
	{
		scanf("%d", &B[i]), B[i]++;
		S2.Insert(B[i], 1);
	}
	while (q--)
	{
		int opt, x;
		scanf("%d%d", &opt, &x);
		if (opt == 1)
		{
			S1.Insert(A[x], -1);
			scanf("%d", &A[x]);
			A[x]++;
			S1.Insert(A[x], 1);
		}
		if (opt == 2)
		{
			S2.Insert(B[x], -1);
			scanf("%d", &B[x]);
			B[x]++;
			S2.Insert(B[x], 1);
		}
		if (opt == 3)
		{
			mnn = min(S1.Find(1), S2.Find(1));
			ll ans = max(Calca(S1.Find(1), x), Calca(S1.Find(n), x));
			ans = max(ans, Calcb(S2.Find(1), x));
			ans = max(ans, Calcb(S2.Find(m), x));
			int t = S2.Find(m) + x;
			t = S1.Query0(t);
			if (1 <= t && t <= n)
				ans = max(ans, Calca(S1.Find(t), x));
			t++;
			if (1 <= t && t <= n)
				ans = max(ans, Calca(S1.Find(t), x));
			t = mnn + x;
			t = S1.Query0(min(1000001, t));
			if (1 <= t && t <= n)
				ans = max(ans, Calca(S1.Find(t), x));
			t++;
			if (1 <= t && t <= n)
				ans = max(ans, Calca(S1.Find(t), x));
			t = mnn + x;
			t = S2.Query0(min(1000001, t));
			if (1 <= t && t <= m)
				ans = max(ans, Calcb(S2.Find(t), x));
			t++;
			if (1 <= t && t <= m)
				ans = max(ans, Calcb(S2.Find(t), x));
			printf("%lld\n", ans);
		}
	}
}
