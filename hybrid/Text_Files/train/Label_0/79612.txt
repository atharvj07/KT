import sys
from array import array
from typing import List, Tuple, TypeVar, Generic, Sequence, Union

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def main():
	(n, m) = map(int, input().split())
	k = array('i', [0] + list(map(int, input().split())))
	sale = sorted((tuple(map(int, input().split())) for _ in range(m)), reverse=True)
	total = sum(k)
	visited = array('b', [0]) * (n + 1)
	(ok, ng) = (total * 2 + 1, total - 1)
	while abs(ok - ng) > 1:
		mid = ok + ng >> 1
		for i in range(1, n + 1):
			visited[i] = 0
		(bought, money) = (0, mid)
		for (di, ti) in sale:
			if di > mid or visited[ti]:
				continue
			visited[ti] = 1
			if money > di:
				money = di
			x = k[ti] if k[ti] <= money else money
			bought += x
			money -= x
		if 2 * total - bought <= mid:
			ok = mid
		else:
			ng = mid
	print(ok)
main()
