#include <cstdio>
#include <cstring>
#include <cmath>
#include <iostream>
#include <algorithm>
using namespace std;

int n,K,mod,f[305][305],g[305][305],C[305][305];
int main(){
	scanf("%d %d %d",&n,&K,&mod);
    for(int i=0;i<=n;++i) C[i][0]=1;
    for(int i=1;i<=n;++i) for(int j=1;j<=i;++j) C[i][j]=(C[i-1][j-1]+C[i-1][j])%mod;
    for(int i=1;i<=n+1;++i) for(int j=0;j<=K;++j) f[0][j]=1,g[0][j]=K-j+1;
    for(int i=1;i<=n;++i){
        for(int k=1;k<=i;++k) for(int v=0;v<=K;++v)
            f[i][v]=(f[i][v]+1LL*g[k-1][v+1]*f[i-k][v]%mod*C[i-1][k-1])%mod;
        for(int v=K;v;--v) g[i][v]=(g[i][v+1]+f[i][v])%mod;
    }
    printf("%d\n",f[n][0]);
	return 0;
}
