
import java.math.*;
import java.util.*;
import java.io.*;

public class C
{
   public static void main(String[] args) throws Exception
   {
      PrintWriter out = new PrintWriter(System.out);
      new C(new FastScanner(System.in), out);
      out.close();
   }

   long modInv(long a, long b)
   {
      return BigInteger.valueOf(a).modInverse(BigInteger.valueOf(b)).longValue();
   }

   long MODO = 1_000_000_007L;
   
   long modPow(long a, long e)
   {
      long res = 1;
      while (e > 0)
      {
         if ((e&1) == 1)
         {
            res *= a;
            res %= MODO;
         }

         e >>= 1;
         a *= a;
         a %= MODO;
      }
      return res;
   }

   int[] expo(int[] mat, long e)
   {
      int[] res = new int[]{1, 0, 0, 1};
      while (e > 0)
      {
         if ((e&1) == 1)
            res = mult(res, mat);
      
         e >>= 1;
         mat = mult(mat, mat);
      }
      return res;
   }

   int sz = 2;
   int[] mult(int[] a, int[] b)
   {
      int[] res = new int[4];
      
      res[0] = (int)((a[0] * (long) b[0] + a[1] * (long) b[2]) % MODO);
      res[1] = (int)((a[0] * (long) b[1] + a[1] * (long) b[3]) % MODO);
      res[2] = (int)((a[2] * (long) b[0] + a[3] * (long) b[2]) % MODO);
      res[3] = (int)((a[2] * (long) b[1] + a[3] * (long) b[3]) % MODO);
      
      return res;
   }

   public C(FastScanner in, PrintWriter out)
   {
      int N = in.nextInt();

      int[] mat = new int[4];
      
      mat[0] = 0;
      mat[1] = 1;
      mat[2] = 2;
      mat[3] = 1;

      long bot = 2L;

      boolean allOnes = false;
      for (int i=0; i<N; i++)
      {
         long exp = in.nextLong();
         bot = modPow(bot, exp);
         bot %= MODO;
         
         // Calculate the top
         mat = expo(mat, exp);
      }

      long inv2 = modInv(2, MODO);
      bot *= inv2;
      bot %= MODO;

      long invN2 = modInv(MODO-2, MODO);
      int[] imat = new int[4];
      imat[0] = (int)invN2;
      imat[1] = (int)inv2;
      imat[2] = 1;
      imat[3] = 0;

      mat = mult(mat, imat);

      long top = mat[1];
      out.printf("%d/%d%n", top, bot);
   }
}

class FastScanner{
    private InputStream stream;
    private byte[] buf = new byte[1024];
    private int curChar;
    private int numChars;
    
    public FastScanner(InputStream stream)
    {
        this.stream = stream;
    }
    
    int read()
    {
        if (numChars == -1)
            throw new InputMismatchException();
        if (curChar >= numChars){
            curChar = 0;
            try{
                numChars = stream.read(buf);
            } catch (IOException e) {
                throw new InputMismatchException();
            }
            if (numChars <= 0)
                return -1;
        }
        return buf[curChar++];
    }
    
    boolean isSpaceChar(int c)
    {
        return c==' '||c=='\n'||c=='\r'||c=='\t'||c==-1;
    }
    
    boolean isEndline(int c)
    {
        return c=='\n'||c=='\r'||c==-1;
    }
    
    int nextInt()
    {
        return Integer.parseInt(next());
    }
    
    long nextLong()
    {
        return Long.parseLong(next());
    }
    
    double nextDouble()
    {
        return Double.parseDouble(next());
    }
    
    String next(){
        int c = read();
        while (isSpaceChar(c))
            c = read();
        StringBuilder res = new StringBuilder();
        do{
            res.appendCodePoint(c);
            c = read();
        }while(!isSpaceChar(c));
        return res.toString();
    }
    
    String nextLine(){
        int c = read();
        while (isEndline(c))
            c = read();
        StringBuilder res = new StringBuilder();
        do{
            res.appendCodePoint(c);
            c = read();
        }while(!isEndline(c));
        return res.toString();
    }
}
