#include <bits/stdc++.h>
using namespace std;
const int INF = 2e9;
const long long BIG_INF = 1e18;
const long long MOD = 1e9 + 7;
const int maxn = 7e2 + 5;
int n, m, d[maxn][maxn], deg[maxn];
int neighbors[maxn][maxn];
long long ans;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--;
    v--;
    deg[u]++;
    deg[v]++;
    d[u][v] = 1;
    d[v][u] = 1;
  }
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      for (int k = 0; k < n; k++)
        if (k != i && k != j && d[i][k] && d[j][k]) neighbors[i][j]++;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (j == i || !d[i][j]) continue;
      for (int k = 0; k < n; k++) {
        if (k == i || k == j) continue;
        int cnt1 = neighbors[i][k] - d[j][k];
        int cnt2 = neighbors[j][k] - d[i][k];
        ans += (long long)cnt1 * (long long)cnt2;
        if (d[i][k] && d[j][k]) ans -= deg[k] - 2;
      }
    }
  }
  cout << ans / 10LL << "\n";
  return 0;
}
