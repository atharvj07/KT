#include<iostream>
#include<string>
using namespace std;

int dx[4] = { 0, 1, 0, -1 };
int dy[4] = { 1, 0, -1, 0 };

int main() {
  int w,h,v;
  string s;
  while(cin>>w>>h, w||h) {
    int x=1,y=1,dir=0;
    while(1) {
      cin>>s;

      if(s == "STOP") {
	cout<<x<<" "<<y<<endl;
	break;
      }else if(s == "FORWARD") {
	cin>>v;
	x += v*dx[dir];
	y += v*dy[dir];
      }else if(s == "BACKWARD") {
	cin>>v;
	x += -v*dx[dir];
	y += -v*dy[dir];
      }else if(s == "RIGHT") {
	dir++;
      }else if(s == "LEFT") {
	dir--;
      }
      x = min(w,x);
      x = max(1,x);
      y = min(h,y);
      y = max(1,y);
      dir = (dir+4)%4;
    }
  }
}
