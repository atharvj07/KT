t = int(input())
while t > 0:
	n = int(input())
	lst = list(map(int, input().strip()))
	lst2 = lst.copy()
	lst2.sort()
	a = []
	b = []
	c = 0
	ind_l = [2] * n
	ind = 0
	for i in lst:
		if i == lst2[c]:
			a.append(i)
			ind_l[ind] = 1
			c += 1
		ind += 1
	for i in lst:
		if c < n and i == lst2[c]:
			b.append(i)
			c += 1
	if len(a) + len(b) < n:
		print('-')
	else:
		for i in ind_l:
			print(i, end='')
		print()
	t -= 1
