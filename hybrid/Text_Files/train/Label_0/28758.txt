#include "bits/stdc++.h"

#pragma warning(disable:4996)
using namespace std;
using ld = long double;
struct Edge {
	int src;
	int dst;
	long long int A;
	long long int B;
};
struct aa {
	int now;
	long long int cost;
};
bool operator <(const aa&l, const aa&r) {
	return l.cost>r.cost;
}
long long int ce(long long int a,long long  int b) {
	return (a-1)/b+1;
}
int main() {
	int N;cin>>N;
	vector<int>valids(N,true),need_next(N),need_pre(N);
	vector<int>l_cnts(N);
	for (int i = 0; i < N; ++i) {
		int K;cin>>K;
		vector<int>v;
		for (int k = 0; k < K; ++k) {
			int a;cin>>a;
			a--;
			v.push_back(a);
			if (a == i) {
				valids[i]=false;
			}
			if (a == i - 1) {
				need_pre[i]=true;
			}
			else if (a == i + 1) {
				need_next[i]=true;
			}
		}
		sort(v.begin(),v.end());
		v.erase(unique(v.begin(),v.end()),v.end());
		auto it=lower_bound(v.begin(),v.end(),i);
			
	
		l_cnts[i]=it==v.begin()?30000000:i-*prev(it);
	}
	vector<int>dp(N+1);
	for (int i = 0; i < N; ++i) {

		dp[i + 1] = max(dp[i+1],dp[i]);
		if (!valids[i]) {
		}
		else {
			if (need_next[i]) {
				int k=i+1;
				int sum=1;
				while (true) {
					if (l_cnts[k]<=sum) {
						dp[k+1]=max(dp[k+1],dp[i]+sum);
						break;
					}
					else {
						if (valids[k]) {
							if (need_next[k]) {
								dp[k+1]=max(dp[k+1],dp[i]+sum);
								k++;
								sum++;
								continue;
							}
							else {
								if (l_cnts[k] > sum) {

									dp[k + 1] = max(dp[k + 1],dp[i]+ sum + 1);
								}
								else {
									assert(false);
									dp[k+1]=max(dp[k+1],dp[i]+sum);
								}
								break;
							}
						}
						else {
							dp[k+1]=max(dp[k+1],dp[i]+sum);
							break;
						}
					}
				}
			}
			else {
				dp[i+1]=max(dp[i]+1,dp[i+1]);
			}
		}
	}
	cout<<N-dp[N]<<endl;
	return 0;
}
