n = int(input())
vert = [tuple(map(int, input().split())) for i in range(n + 1)]
turn = [True for i in range(n - 1)]
handed = 0
plus = minus = 0
for i in range(n - 1):
	a1 = (vert[i + 1][0] - vert[i][0], vert[i + 1][1] - vert[i][1])
	a2 = (vert[i + 2][0] - vert[i + 1][0], vert[i + 2][1] - vert[i + 1][1])
	cross = a1[0] * a2[1] - a1[1] * a2[0]
	if cross > 0:
		cross = 1
		plus += 1
		turn.append(1)
	elif cross < 0:
		cross = -1
		minus += 1
		turn.append(-1)
	handed += cross
if handed > 0:
	print(minus)
else:
	print(plus)
