from sys import stdin, stderr

def rl():
	return [int(w) for w in stdin.readline().split()]
(t,) = rl()
for _ in range(t):
	(n,) = rl()
	p1 = rl()
	p2 = rl()
	p = [0] * n
	for (x, y) in zip(p1, p2):
		p[x - 1] = y - 1
	r = 1
	for i in range(n):
		if p[i] == i:
			continue
		j = p[i]
		while j != i:
			nj = p[j]
			p[j] = j
			j = nj
		r = r * 2 % 1000000007
	print(r)
