cases = int(input())
for case in range(cases):
	length = int(input())
	values = list(map(int, input().split()))
	types = list(map(int, input().split()))
	diff = False
	for (key, value) in enumerate(types):
		if types[key - 1] != types[key] and key != 0:
			diff = True
			break
	count = 0
	for (key, value) in enumerate(values):
		if value >= values[key - 1] and key != 0:
			count += 1
	if count + 1 != len(values) and (not diff):
		print('No')
	else:
		print('Yes')
