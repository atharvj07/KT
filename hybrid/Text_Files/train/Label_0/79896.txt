from sys import *
from math import *
from bisect import *
(n, k) = map(int, stdin.readline().split())
a = list(map(int, stdin.readline().split()))
od = ev = 0
c = []
for i in range(n):
	if a[i] % 2 == 0:
		ev += 1
	else:
		od += 1
	if ev == od and i != n - 1:
		c.append(abs(a[i] - a[i + 1]))
c.sort()
ans = 0
for i in range(len(c)):
	if k == 0:
		break
	if k >= c[i]:
		k -= c[i]
		ans += 1
print(ans)
