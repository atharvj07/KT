#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
using namespace std;
const int N = 1000005;
const int mod = 1000000007;
int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int pow(int a, int b, int c) {
  int res = 1;
  while (b) {
    if (b & 1) res = (res * a) % c;
    b = b >> 1;
    a = (a * a) % c;
  }
  return res;
}
int t;
int n;
int a[N];
int dp[N];
int m[N];
void solve() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    m[a[i]] = 1;
  }
  int ans = 1;
  for (int i = 0; i < n; i++) {
    int x = a[i];
    dp[x] = max(dp[x], 1);
    for (int j = 2 * x; j < N; j += x) {
      dp[j] = max(dp[j], dp[x] + 1);
    }
    ans = max(ans, dp[x]);
  }
  cout << ans << '\n';
}
signed main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
