#include <bits/stdc++.h>
using namespace std;
int idx, tmp, ans;
string s;
int main() {
  cin >> s;
  idx = 0;
  tmp = 0;
  ans = 0;
  bool sign = true;
  while (idx < s.size()) {
    if (s[idx] == '+' || s[idx] == '-') {
      if (sign)
        ans += tmp;
      else
        ans -= tmp;
      tmp = 0;
      sign = (s[idx] == '+');
    } else {
      tmp = tmp * 10 + s[idx] - '0';
    }
    idx++;
  }
  if (sign)
    ans += tmp;
  else
    ans -= tmp;
  int cur = 0;
  for (int i = 100; i >= 1; i /= 10) {
    if (ans < i && i != 1) continue;
    int want = (ans / i) % 10;
    if (cur < 48 + want) {
      while (cur < 48 + want) {
        cur++;
        printf("+");
      }
    }
    if (cur > 48 + want) {
      while (cur > 48 + want) {
        cur--;
        printf("-");
      }
    }
    puts(".");
  }
}
