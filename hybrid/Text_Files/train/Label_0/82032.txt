import sys

def read_int():
	return int(sys.stdin.readline())

def read_ints():
	return list(map(int, sys.stdin.readline().split()))

def write(x):
	sys.stdout.write(x + '\n')

def check(pointer, index, right_done=0):
	pointer += 1
	if pointer > len(t) - 1:
		return True
	if not right_done:
		if index + 1 < len(s) and t[pointer] == s[index + 1]:
			right = check(pointer, index + 1)
			if right:
				return True
	if index > 0 and t[pointer] == s[index - 1]:
		left = check(pointer, index - 1, right_done=1)
		if left:
			return True
	return False
for _ in '.' * read_int():
	s = input()
	t = input()
	pointer = 0
	ch = t[pointer]
	indexes = []
	for i in range(len(s)):
		if s[i] == ch:
			indexes.append(i)
	if len(indexes) == 0:
		write('NO')
		continue
	for index in indexes:
		if check(pointer, index):
			write('YES')
			break
	else:
		write('NO')
