import java.util.Scanner;

public class D {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long total = 0;
        for (int i = 0; i < n; i++) {
            long t = sc.nextInt();
            long T = sc.nextInt();
            long x = sc.nextInt();
            long cost = sc.nextInt();
            total += cost;
            long price = (m + t > T) ? m * x : 0;
            long q = 0;
            if (price == 0)
                continue;
            long l = 0;
            if ((T - t) > 0 && m != 1) {
                l = (m % (T - t) == 0) ? m / (T - t) : m / (T - t) + 1;

                long b = (m % (l + 1) == 0) ? m / (l + 1) : m / (l + 1) + 1;
                long v = 1000000000;
                if ((l - 1) != 0)
                    v = (m % (l - 1) == 0) ? m / (l - 1) : m / (l - 1) + 1;
                if (b + t > T)
                    l++;
                else if (v + t <= T)
                    l--;
                if (l > m) {
                    total += price;
                } else {
                    if (price < (l - 1) * cost)
                        total += price;
                    else
                        total += (l - 1) * cost;
                }
            } else
                total += price;
        }
        System.out.println(total);

    }

}
