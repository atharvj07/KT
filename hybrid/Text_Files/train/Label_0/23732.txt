template <typename t1, typename t2>
inline bool upmax(t1 &a, t2 b) {
  if (a < (t1)b) {
    a = (t1)b;
    return true;
  } else
    return false;
}
template <typename t1, typename t2>
inline bool upmin(t1 &a, t2 b) {
  if (a > (t1)b) {
    a = (t1)b;
    return true;
  } else
    return false;
}
template <typename T>
inline bool pal(T &x) {
  int n = (int((x.size())));
  for (int i = 0; i < n / 2; i++) {
    if (x[i] != x[n - i - 1]) return 0;
  }
  return 1;
}
template <typename T>
inline T gcd(T a, T b) {
  return b ? gcd(b, a % b) : a;
}
template <typename T>
inline T lcm(T a, T b) {
  return a * (b / gcd(a, b));
}

long long hp[211111];
long long dmg[211111];
void solve() {
  long long n, a, b;
  cin >> n >> a >> b;
  upmin(b, n);
  vector<pair<long long, long long> > v;
  long long alldmg = 0;
  for (int i = 0; i < n; i++) {
    cin >> hp[i] >> dmg[i];
    v.push_back(make_pair(hp[i] - dmg[i], i));
    alldmg += dmg[i];
  }
  if (b == 0) {
    cout << alldmg << endl;
    return;
  }
  sort((v).begin(), (v).end(), greater<pair<long long, long long> >());
  long long s = 0;
  for (int i = 0; i < b; i++)
    if (v[i].first > 0) s += v[i].first;
  long long ans = s;
  long long last = 0;
  for (int ind = 0; ind < b; ind++) {
    int i = v[ind].second;
    long long x = (hp[i] << a);
    long long cur = s - max(0ll, v[ind].first) + max(0ll, x - dmg[i]);
    upmax(ans, cur);
    last = v[ind].first;
  }
  upmax(last, 0);
  for (int ind = b; ind < n; ind++) {
    int i = v[ind].second;
    long long x = (hp[i] << a);
    long long cur = s - last + max(0ll, x - dmg[i]);
    upmax(ans, cur);
  }
  cout << ans + alldmg;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t = 1;
  while (t--) solve();

}
