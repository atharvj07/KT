#include <bits/stdc++.h>
using namespace std;
long long n, m, k;
string str = "";
vector<pair<int, int> > adj[1001];
int mat[1000][1000];
vector<int> shortestPath(int src, int V) {
  priority_queue<pair<int, int>, vector<pair<int, int> >,
                 greater<pair<int, int> > >
      pq;
  vector<int> dist(V, INT_MAX);
  pq.push(make_pair(0, src));
  dist[src] = 0;
  while (!pq.empty()) {
    int u = pq.top().second;
    pq.pop();
    vector<pair<int, int> >::iterator i;
    for (i = adj[u].begin(); i != adj[u].end(); ++i) {
      int v = (*i).first;
      int weight = (*i).second;
      if (dist[v] > dist[u] + weight) {
        dist[v] = dist[u] + weight;
        pq.push(make_pair(dist[v], v));
      }
    }
  }
  return dist;
}
void run() {
  int s, t;
  cin >> n >> m >> s >> t;
  s--;
  t--;
  memset(mat, 0, sizeof(mat));
  ;
  int a, b;
  for (long long int i = 0; i < m; i++) {
    cin >> a >> b;
    a--;
    b--;
    adj[a].push_back(make_pair(b, 1));
    adj[b].push_back(make_pair(a, 1));
    mat[a][b] = mat[b][a] = 1;
  }
  vector<int> distS = shortestPath(s, n);
  vector<int> distT = shortestPath(t, n);
  bool chosen[n][n];
  memset(chosen, 0, sizeof(chosen));
  ;
  int count = 0;
  int mind = distS[t];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i != j && (i != s || j != s || i != t || j != t) && !mat[i][j]) {
        int temp = min(distS[i] + distT[j] + 1, distS[j] + distT[i] + 1);
        if (temp >= mind && !chosen[i][j]) {
          count++;
          chosen[i][j] = chosen[j][i] = true;
        }
      }
    }
  }
  cout << setprecision(30) << count << '\n';
  return;
  ;
}
void setup() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
}
int main() {
  setup();
  int tests = 1;
  for (int i = 0; i < tests; i++) {
    run();
  }
  return 0;
}
