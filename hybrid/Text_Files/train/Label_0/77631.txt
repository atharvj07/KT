for _ in range(int(input())):
	n = int(input())
	a = sorted(list(map(int, input().split())), reverse=True)
	Alice = sum([i if i % 2 == 0 else 0 for i in a[::2]])
	Bob = sum([i if i % 2 == 1 else 0 for i in a[1::2]])
	if Alice > Bob:
		print('Alice')
	elif Alice < Bob:
		print('Bob')
	else:
		print('Tie')
