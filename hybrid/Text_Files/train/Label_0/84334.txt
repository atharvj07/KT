def lc():
	return [int(i) for i in input().split()]
t = int(input())
while t:
	(n, s, k) = lc()
	fr = lc()
	var1 = s - 1
	var2 = s + 1
	if s not in fr:
		print(0)
	else:
		while True:
			if var1 <= 0:
				var1 = 1
			if var2 >= n:
				var2 = n
			if var1 not in fr or var2 not in fr:
				print(max(s - var1, var2 - s))
				break
			else:
				var1 -= 1
				var2 += 1
	t -= 1
