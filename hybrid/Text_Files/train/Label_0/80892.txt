import math
n = int(input())
arr = [1, 2] + [0] * (n - 2)

def isprime(x):
	y = int(math.sqrt(x)) + 1
	for i in range(2, y):
		if x % i == 0:
			return i
	return x
p = 3
for i in range(4, n + 1):
	x = isprime(i)
	if x == i:
		arr[i - 2] = p
		p += 1
	else:
		arr[i - 2] = arr[x - 2]
print(*arr[:n - 1])
