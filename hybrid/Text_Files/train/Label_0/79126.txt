N = [int(i) for i in input().split()]
M = []
for i in range(N[0]):
	L = [int(i) for i in input().split()]
	M.append(L)
if M == [[0 for i in range(N[1])] for j in range(N[0])]:
	print(0)
else:
	B = []
	for i in range(1, N[0]):
		for j in range(1, N[1]):
			if M[i - 1][j - 1] == 1 and M[i][j - 1] == 1 and (M[i - 1][j] == 1) and (M[i][j] == 1):
				B.append([i, j])
	for k in range(len(B)):
		i = B[k][0]
		j = B[k][1]
		M[i - 1][j - 1] = 0
		M[i - 1][j] = 0
		M[i][j - 1] = 0
		M[i][j] = 0
	if M == [[0 for i in range(N[1])] for j in range(N[0])]:
		print(len(B))
		for k in range(len(B)):
			print(B[k][0], ' ', B[k][1])
	else:
		print(-1)
