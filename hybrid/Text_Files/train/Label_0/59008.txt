    static HashMap<Long, Integer>[] hm = new HashMap [61];
    
    private static void solve() throws IOException {
    	for (int i = 0; i <= 60; ++i)
    		hm[i] = new HashMap<Long, Integer>();
    	hm[0].put(0l, 0);
    	for (int i = 1; i <= 60; ++i) {
    		find(i, 0);
    		//pw.println(find(i, 0));
    	}
    	int n = nextInt();
    	int ans = 0;
    	//Arrays.fill(list, 0);
    	for (int i = 0; i < n; ++i) {
    		int t = nextInt();
    		ans ^= hm[t].get(0l);
    	//	list[t]++;
    	}
    	if (ans == 0)

    //static boolean[] b = new boolean [61];

	private static int find(int s, long mask) {
		if (hm[s].containsKey(mask))
			return hm[s].get(mask);
		boolean[] b = new boolean [s + 1];
		int ans = 0, count = 0;
		long curm;
		for (int i = 0; i < s; ++i) {
			if ((mask & (1L << i)) == 0) {
				curm = mask | (1L << i);
				//list[count++] = find(s - 1 - i, curm & ((1l << (s - 1 - i)) - 1));
				//b[list[count - 1]] = true;
				b[find(s - 1 - i, curm & ((1l << (s - 1 - i)) - 1))] = true;
			}
		}
		while (b[ans])
			++ans;
		hm[s].put(mask, ans);
		//for (int i = 0; i < count; ++i)
		//	b[list[i]] = false;
		return ans;
	}

	private static int sumf(int[] fen, int id) {
        int summ = 0;
        for (; id >= 0; id = (id & (id + 1)) - 1)
            summ += fen[id];
        return summ;
    }

    private static void addf(int[] fen, int id) {
        for (; id < fen.length; id |= id + 1)
            fen[id]++;
    }

    private static int nextInt() throws IOException {
        return Integer.parseInt(next());
    }
    private static long nextLong() throws IOException {
        return Long.parseLong(next());
    }
    private static double nextDouble() throws IOException {
        return Double.parseDouble(next());
    }
    private static String next() throws IOException {
        while (st==null || !st.hasMoreTokens())
            st = new StringTokenizer(br.readLine());
        return st.nextToken();
    }
