import sys
from collections import defaultdict
from math import gcd

def get_ints():
	return list(map(int, sys.stdin.readline().strip().split()))

def solve(a, b, x):
	if max(a, b) < x:
		return 0
	if a == x or b == x:
		return 1
	if x % b == a % b:
		return 1
	if b == 0:
		return a == x
	return solve(b, a % b, x)
T = int(input())
for i in range(T):
	(a, b, x) = get_ints()
	m = gcd(a, b)
	if x % m:
		print('NO')
	elif solve(a, b, x):
		print('YES')
	else:
		print('NO')
