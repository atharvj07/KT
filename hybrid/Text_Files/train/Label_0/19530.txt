#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200005;
vector<vector<int>> adj(MAXN);
vector<int> diff(MAXN);
int main() {
  int n, m;
  scanf("%i%i", &n, &m);
  int lst;
  scanf("%i", &lst);
  long long sum = 0;
  for (int x = 1; x < m; x++) {
    int l = lst, r, v;
    scanf("%i", &v);
    r = v;
    if (l > r) swap(l, r);
    if (l != r) {
      sum += r - l;
      adj[l].push_back(r);
      adj[r].push_back(l);
      l++;
      if (l < r) {
        diff[l]++;
        diff[r]--;
      }
      lst = v;
    }
  }
  for (int x = 1, cnt = 0; x <= n; x++) {
    cnt += diff[x];
    long long ans = sum - cnt;
    for (auto &y : adj[x]) {
      ans -= abs(y - x);
      if (y > x)
        ans += y - 1;
      else
        ans += y;
    }
    printf("%lli ", ans);
  }
  printf("\n");
}
