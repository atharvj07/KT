T = int(input())
for __ in range(T):
	n = int(input())
	A = [int(i) for i in input().split()]
	result = sum(A)
	if result == n:
		if result % 2 == 0:
			print('Second')
		else:
			print('First')
	else:
		for i in range(1, len(A) + 1):
			if A[i - 1] > 1:
				if i % 2 == 0:
					print('Second')
					break
				else:
					print('First')
					break
