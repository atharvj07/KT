#include<bits/stdc++.h>
#define rep(i,n)for(ll i=0;i<n;i++)
using namespace std;
typedef long long ll;
typedef pair<ll,ll>P;

ll cnt;
void solve(ll n,ll p,ll c){
	if(!n){cnt++;return;}
	if(c>=4)return;
	for(ll i=p;i*i<=n;i++)
		solve(n-i*i,i,c+1);
}
int main() {
	ll n;
	while(scanf("%lld",&n),n){
		cnt=0;
		solve(n,1,0);
		printf("%lld\n",cnt);
	}
}
