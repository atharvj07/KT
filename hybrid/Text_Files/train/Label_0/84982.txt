import sys
(n, m, k) = map(int, sys.stdin.readline().split())
if k == 0:
	print(-1)
	exit()
D = []
for i in range(m):
	(u, v, l) = map(int, sys.stdin.readline().split())
	(u, v) = (u - 1, v - 1)
	D.append((u, v, l))
a = list(map(int, sys.stdin.readline().split()))
F = [False] * n
for i in a:
	F[i - 1] = True
D = sorted(D, key=lambda x: x[2])
for (u, v, l) in D:
	if F[u] and (not F[v]) or (not F[u] and F[v]):
		print(l)
		exit()
print(-1)
