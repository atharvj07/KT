import java.util.*;
import java.lang.*;
import java.math.*;
import java.io.*;

import static java.lang.Math.*;
import static java.util.Arrays.*;
import static java.util.Collections.*;

public class A{
    Scanner sc=new Scanner(System.in);

    // long INF=1L<<60;
    int INF=1<<28;
    double EPS=1e-12;

    long mod=(int)1e9+7;
    String s, t;

    void run(){
        s=sc.nextLine();
        t=sc.nextLine();
        solve();
    }

    void solve(){
        int m=s.length();
        int n=t.length();
        long[] dp=new long[m];
        long[] dp2=new long[m];
        for(int j=0; j<n; j++){
            System.arraycopy(dp, 0, dp2, 0, m);
            for(int i=0; i<m; i++){
                if(s.charAt(i)==t.charAt(j)){
                    dp[i]=(dp[i]+1)%mod;
                    if(i>0){
                        dp[i]=(dp[i]+dp2[i-1])%mod;
                    }
                }
            }
        }
        long ans=0;
        for(int i=0; i<m; i++){
            ans=(ans+dp[i])%mod;
        }
        println(ans+"");
    }

    void println(String s){
        System.out.println(s);
    }

    void print(String s){
        System.out.print(s);
    }

    void debug(Object... os){
        System.err.println(Arrays.deepToString(os));
    }

    public static void main(String[] args){
        new A().run();
    }
}
