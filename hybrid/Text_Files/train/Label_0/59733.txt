import java.io.*;
import java.util.*;

/**
 * Created by madeline on 3/30/17.
 * Solution to http://codeforces.com/problemset/problem/267/A
 * Assignment 9 CS104C
 */
public class CS267A {

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int num_pairs = sc.nextInt();
        while (num_pairs-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int count = 0;
            while (a > 0 && b > 0) {
                if (a <= b) {
                    count += b / a;
                    b = b % a;
                } else {
                    count += a / b;
                    a = a % b;
                }
            }
            System.out.println(count);
        }

    }
}
