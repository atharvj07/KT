#include <bits/stdc++.h>
using namespace std;
using vi = vector<int>;
using vvi = vector<vi>;

int main() {
    int r, s, p;
    cin >> r >> s >> p;

    vi ds;
    while (p--) {
        int i, j;
        cin >> i >> j;
        i--, j--;
        if (j >= s) {
            j++;
        }
        int d = abs(r - i) + abs(s - j);
        ds.push_back(d);
    }
    sort(ds.begin(), ds.end());
    
    int pre = 0;
    for (int d : ds) {
        if (pre < d) {
            pre = d;
        } else {
            pre++;
        }
    }
    cout << pre << endl;
}
