#include<bits/stdc++.h>
using namespace std;
#define MAX_N 10005
int N,M;
int main(){
  while(cin>>N>>M){
    int Min[MAX_N]={},Max[MAX_N]={};
    if(M==0&&N==0)break;
    for(int i=0;i<M;i++){
      int p,k,id;
      cin>>p>>k;
      for(int j=0;j<k;j++){
        cin>>id;
        id--;
        if(k==1){
          Min[id]+=p;
        }
        Max[id]+=p;
      }
    }

    int mini=1e9,maxm=-1e9,ans=0;

    for(int i=0;i<N;i++){

      if(Max[i]>mini)
        ans=max(ans, abs(Max[i]-mini) );
        
      if(maxm>Min[i])
        ans=max(ans, abs(maxm-Min[i]) );

      mini=min(mini,Min[i]);
      maxm=max(maxm,Max[i]);
    }
    cout<<ans+1<<endl;
  }
  return 0;
}
