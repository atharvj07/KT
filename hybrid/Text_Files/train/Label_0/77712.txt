n = int(input())
wh = [tuple(map(int, input().split())) for i in range(n)]
W = sum([kappa[0] for kappa in wh])
pref = [wh[0][1]]
for i in range(1, n):
	pref.append(max(pref[-1], wh[i][1]))
suff = [0] * (n - 1) + [wh[n - 1][1]]
for i in range(n - 2, -1, -1):
	suff[i] = max(suff[i + 1], wh[i][1])
print(' '.join([str((W - wh[i][0]) * max(pref[i - 1] if i - 1 >= 0 else 0, suff[i + 1] if i + 1 < n else 0)) for i in range(n)]))
