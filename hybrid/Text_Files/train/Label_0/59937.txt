import java.io.*;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class E_fastread {
    class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public float nextFloat() {
            return Float.parseFloat(next());
        }

        public double nextDouble() {
            return Float.parseFloat(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }
    }

    InputReader sc;
    PrintWriter pw;

    ArrayList<Integer>[] adj;
    int n, m;
    boolean[] visited;
    int[] ps;
    int[] ds;
    int[] ss;
    Queue<Integer> lfs;

    void dfs(int node, int parent, int depth) {
        visited[node] = true;
        ps[node] = parent;
        ds[node] = depth;
        boolean isLeaf = true;
        for (int other: adj[node]) {
            if (other == parent) {
                continue;
            }

            if (!visited[other]) {
                dfs(other, node, depth + 1);
                isLeaf = false;
                ss[node]++;
            }
        }
        if (isLeaf) {
            lfs.add(node);
        }
    }

    private void run() throws FileNotFoundException {
        sc = new InputReader(System.in);
        pw = new PrintWriter(System.out);
        int nTests = sc.nextInt();

        for (int test = 0; test < nTests; test++) {
            n = sc.nextInt();
            m = sc.nextInt();

            adj = new ArrayList[n];
            visited = new boolean[n];
            lfs = new LinkedList<>();
            ps = new int[n];
            ds = new int[n];
            ss = new int[n];
            for (int i = 0; i < n; i++) {
                adj[i] = new ArrayList<>();
            }

            for (int i = 0; i < m; i++) {
                int u = sc.nextInt() - 1;
                int v = sc.nextInt() - 1;
                adj[u].add(v);
                adj[v].add(u);
            }

            dfs(0, -1, 1);

            boolean done = false;
            StringBuilder sb = new StringBuilder();
            for (int leaf: lfs) {
                if (ds[leaf] >= (n + 1) / 2) {
                    int crt = leaf;
                    sb.append("PATH\n");
                    sb.append(ds[leaf]);
                    sb.append("\n");

                    while (crt != -1) {
                        sb.append((1 + crt) + " ");
                        crt = ps[crt];
                    }
                    System.out.println(sb);
                    done = true;
                    break;
                }
            }

            if (!done) {
                sb.append("PAIRING\n");
                ArrayList<Pair> pairs = new ArrayList<>();
                while (lfs.size() >= 2) {
                    int n1 = lfs.poll();
                    int n2 = lfs.poll();
                    pairs.add(new Pair(n1, n2));

                    ss[ps[n1]]--;
                    if (ss[ps[n1]] == 0) {
                        lfs.add(ps[n1]);
                    }

                    ss[ps[n2]]--;
                    if (ss[ps[n2]] == 0) {
                        lfs.add(ps[n2]);
                    }
                }
                sb.append(pairs.size());
                sb.append("\n");
                for (Pair p: pairs) {
                    sb.append((1 + p.a) + " " + (1 + p.b));
                    sb.append("\n");
                }

                System.out.println(sb);
            }
        }

        pw.close();
    }

    class Pair {
        int a, b;
        public Pair(int n1, int n2) {
            a = n1;
            b = n2;
        }
    }
    public static void main(String[] args) throws FileNotFoundException {
        E_fastread a = new E_fastread();
        a.run();
    }
}
