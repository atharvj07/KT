for _ in range(int(input())):
	(n, p, k) = map(int, input().split())
	l = sorted(list(map(int, input().split())))
	(pre, ans) = (0, 0)
	for i in range(k):
		(sum, cnt) = (pre, i)
		if sum > p:
			break
		for j in range(i + k - 1, n, k):
			if sum + l[j] <= p:
				cnt += k
				sum += l[j]
			else:
				break
		pre += l[i]
		ans = max(ans, cnt)
	print(ans)
