import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class KatyaAndSegmentsSets {
    
    int K = (int) 3e5 + 10;
    int MAX = (int) 6e6;
    
    int INF = (int) 1e9 + 10;
    
    int n, Q, k;
    Segment[] segs;
    
    int next = 1;
    int[] root = new int[K];
    int[] T = new int[MAX], L = new int[MAX], R = new int[MAX];
    
    void solve() {
        n = in.nextInt(); Q = in.nextInt(); k = in.nextInt();
        
        segs = new Segment[k];
        for (int i = 0; i < k; i++) segs[i] = new Segment(in.nextInt(), in.nextInt(), in.nextInt() - 1);
        Arrays.sort(segs);
        
        init_persistent_segment_tree(0, 0, n);
        
        root[k] = 0;
        for (int i = k - 1; i >= 0; i--) {
            root[i] = copy_node(root[i + 1]);
            update(root[i], 0, n, segs[i].id, segs[i].r);
        }
        
        while (Q-- > 0) {
            int a = in.nextInt() - 1, b = in.nextInt(), x = in.nextInt(), y = in.nextInt();
            int i = lower_bound(x);
            if (i >= k) {
                out.println("no");
                out.flush();
                continue;
            }
            int max = query(root[i], 0, n, a, b);
            out.println(max <= y ? "yes" : "no");
            out.flush();
        }
    }
    
    int lower_bound(int x) {
        int lb = -1, ub = k;
        while (ub - lb > 1) {
            int m = (lb + ub) >> 1;
            if (segs[m].l >= x) ub = m;
            else lb = m;
        }
        return ub;
    }
    
    int query(int k, int l, int r, int a, int b) {
        if (a >= r || b <= l) return -INF;
        if (a <= l && r <= b) return T[k];
        int m = (l + r) >> 1;
        return Math.max(query(L[k], l, m, a, b), query(R[k], m, r, a, b));
    }
    
    void update(int k, int l, int r, int i, int v) {
        if (r - l == 1) {
            if (T[k] > v) T[k] = v;
            return;
        }
        int m = (l + r) >> 1;
        if (i < m) {
            L[k] = copy_node(L[k]);
            update(L[k], l, m, i, v);
        } else {
            R[k] = copy_node(R[k]);
            update(R[k], m, r, i, v);
        }
        int max = T[L[k]];
        if (max < T[R[k]]) max = T[R[k]];
        T[k] = max;
    }
    
    int copy_node(int prev) {
        int cur = next++;
        L[cur] = L[prev];
        R[cur] = R[prev];
        T[cur] = T[prev];
        return cur;
    }
    
    void init_persistent_segment_tree(int i, int l, int r) {
        if (r - l == 1) {
            T[i] = INF;
            return;
        }
        int m = (l + r) >> 1;
        L[i] = next++;
        R[i] = next++;
        init_persistent_segment_tree(L[i], l, m);
        init_persistent_segment_tree(R[i], m, r);
        T[i] = INF;
    }
    
    static class Segment implements Comparable<Segment> {
        int l, r;
        int id;
        
        Segment(int l, int r, int id) {
            this.l = l;
            this.r = r;
            this.id = id;
        }
        
        @Override
        public int compareTo(Segment s) {
            return Integer.compare(l, s.l);
        }
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new KatyaAndSegmentsSets().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
