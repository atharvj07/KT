
import java.io.*;
import java.util.*;

import static java.lang.Math.*;

public class BetaRound75_Div1_A implements Runnable {

	final boolean ONLINE_JUDGE = System.getProperty("ONLINE_JUDGE") != null;
	BufferedReader in;
	PrintWriter out;
	StringTokenizer tok = new StringTokenizer("");

	void init() throws IOException {
		if (ONLINE_JUDGE) {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
		} else {
			in = new BufferedReader(new FileReader("input.txt"));
			out = new PrintWriter("output.txt");
		}
	}

	String readString() throws IOException {
		while (!tok.hasMoreTokens()) {
			tok = new StringTokenizer(in.readLine());
		}
		return tok.nextToken();
	}

	int readInt() throws IOException {
		return Integer.parseInt(readString());
	}

	@Override
	public void run() {
		try {
			long t1 = System.currentTimeMillis();
			init();
			Locale.setDefault(Locale.US);
			solve();
			out.close();
			long t2 = System.currentTimeMillis();
			System.err.println("Time = " + (t2 - t1));
		} catch (Exception e) {
			e.printStackTrace(System.err);
			System.exit(-1);
		}
	}

	public static void main(String[] args) {
		new Thread(new BetaRound75_Div1_A()).start();
	}
	
	void solve() throws IOException {
		String s1 = readString();
		String s2 = readString();
		HashMap<Character, Integer> map = new HashMap<Character, Integer>();
		for (int i = 0; i < s1.length(); i++) {
			if (!map.containsKey(s1.charAt(i))) {
				map.put(s1.charAt(i), 1);
			} else {
				map.put(s1.charAt(i), map.get(s1.charAt(i)) + 1);
			}
		}
		for (int i = 0; i < s2.length(); i++) {
			if (map.get(s2.charAt(i)) == null) {
				out.print(-1);
				return;
			}
		}
		
		HashMap<Character, TreeSet<Integer>> z = new HashMap<Character, TreeSet<Integer>>();
		for (int i = 0; i < s1.length(); i++) {
			if (!z.containsKey(s1.charAt(i))) {
				z.put(s1.charAt(i), new TreeSet<Integer>());
			}
			z.get(s1.charAt(i)).add(i);
		}
		long ans = 1;
		for (int i = 0, j = -1; i < s2.length(); i++) {
			char need = s2.charAt(i);
			Integer p = z.get(need).higher(j);
			if (p == null) {
				j = z.get(need).first();
				ans++;
			} else {
				j = p;
			}
		}
		out.print(ans);
	}
	
}
