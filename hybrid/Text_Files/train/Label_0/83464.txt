import sys
reader = (s.rstrip() for s in sys.stdin)
input = reader.__next__
t = int(input())

def findCost(org_cost, arr):
	n = len(arr)
	min_cost = org_cost
	for i in range(n - 1):
		cost = org_cost
		cost -= abs(arr[i + 1] - arr[i])
		if i + 2 < n:
			cost -= abs(arr[i + 2] - arr[i + 1])
			cost += abs(arr[i + 2] - arr[i])
		min_cost = min(min_cost, cost)
	return min_cost
for i in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	diff = [0] * (n - 1)
	for i in reversed(range(n - 1)):
		diff[i] = arr[i + 1] - arr[i]
	org_cost = 0
	for el in diff:
		org_cost += abs(el)
	cost1 = findCost(org_cost, arr)
	cost2 = findCost(org_cost, arr[::-1])
	print(min(cost1, cost2))
