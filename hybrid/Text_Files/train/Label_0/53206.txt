    public class A {
     
        public static void main(String[] args)  
    		{

    			int T;
    			Scanner s=new Scanner(System.in);
    			T=s.nextInt();
    			while(true) {
    				int N,D;
    			    N=s.nextInt();
    			    D=s.nextInt();
    			    int[] array = new int[N];  
    			    for(int i=0; i<N; i++)  
    			    {  
    			       //reading array elements from the user   
    			       array[i]=s.nextInt();  
    			    } 
    			    Arrays.sort(array);
    			    int[] arrayStatus = new int[N];  
    			    int mini_sum=array[1]+array[0];
    			    for(int i=2;i<N;i++) {
    			    	if(array[i]>D&&mini_sum>D) {
    			    		arrayStatus[i]=0;
    			    	}else{
    			    		arrayStatus[i]=1;
    			    	}
    			    }
    			    if(array[0]>D) {
    			    	arrayStatus[0]=0;
    			    }else {
    			    	arrayStatus[0]=1;
    			    }
    			    if(array[1]>D) {
    			    	arrayStatus[1]=0;
    			    }else {
    			    	arrayStatus[1]=1;
    			    }
    			    String res="Yes";
    			    for(int i=0;i<N;i++) {
    			    	if(arrayStatus[i]==0) {
    			    		res="NO";
    			    	}
    			    }
    			    System.out.println(res);
    			    T--;
    			    if(T==0)
    			    	break;
    				
    			}
    		    
    		    		    
    		} 
    	
    }
