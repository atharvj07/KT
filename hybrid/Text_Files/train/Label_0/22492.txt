#include <bits/stdc++.h>
using namespace std;
int n;
int qu[100010];
int sol = 0;
long long sum = 0;
int cnt = 1;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> qu[i];
  }
  sort(qu + 1, qu + n + 1);
  while (cnt <= n) {
    if (sum > qu[cnt]) {
      cnt++;
      continue;
    }
    sol++;
    sum += qu[cnt];
    cnt++;
  }
  cout << sol;
  return 0;
}
