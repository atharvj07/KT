#include <bits/stdc++.h>
using namespace std;
inline void cmax(int &a, int b) {
  if (a < b) a = b;
}
string s;
int n;
string f[30];
int fb[32][32];
int dp[2][32];
int main() {
  cin >> s;
  cin >> n;
  for (int i = 0; i < (int)(n); i++) {
    cin >> f[i];
    const int a = f[i][0] - 'a', b = f[i][1] - 'a';
    fb[a][b] = fb[b][a] = 1;
  }
  int *cur = dp[0], *nxt = dp[1];
  for (int i = 0; i < (int)(32); i++) cur[i] = -(1 << 28);
  for (int k = 0; k < (int)(s.size()); k++) {
    for (int i = 0; i < (int)(32); i++) nxt[i] = cur[i];
    const int ix = s[k] - 'a';
    for (int i = 0; i < (int)(32); i++)
      if (!fb[i][ix]) {
        cmax(nxt[ix], cur[i] + 1);
      }
    cmax(nxt[ix], 1);
    swap(cur, nxt);
  }
  int ans = (1 << 28);
  for (int i = 0; i < (int)(32); i++) ans = min(ans, (int)s.size() - cur[i]);
  printf("%d\n", ans);
  return 0;
}
