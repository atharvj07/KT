#include <bits/stdc++.h>
using namespace std;
int n, m, ans;
char a[22][22], second[22][22];
string k;
void sol(int x, int y, int z) {
  int r = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == '.') r++;
    }
  }
  if (r / 5 + z <= ans) return;
  if (n - x < 2) {
    if (z > ans) {
      ans = z;
      for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
          second[i][j] = a[i][j];
        }
      }
    }
    return;
  }
  if (a[x][y + 1] == '.' && a[x][y + 2] == '.' && a[x + 1][y + 1] == '.' &&
      a[x + 2][y + 1] == '.' && a[x][y] == '.') {
    a[x][y] = z + 'A', a[x][y + 1] = z + 'A', a[x][y + 2] = z + 'A',
    a[x + 1][y + 1] = z + 'A', a[x + 2][y + 1] = z + 'A';
    if (m - y < 3)
      sol(x + 1, 0, z + 1);
    else
      sol(x, y + 2, z + 1);
    a[x][y + 1] = '.', a[x][y + 2] = '.', a[x + 1][y + 1] = '.',
             a[x + 2][y + 1] = '.', a[x][y] = '.';
  }
  if (a[x + 1][y] == '.' && a[x + 2][y] == '.' && a[x + 2][y - 1] == '.' &&
      a[x + 2][y + 1] == '.' && a[x][y] == '.') {
    a[x][y] = z + 'A', a[x + 1][y] = z + 'A', a[x + 2][y] = z + 'A',
    a[x + 2][y - 1] = z + 'A', a[x + 2][y + 1] = z + 'A';
    if (m - y < 3)
      sol(x + 1, 0, z + 1);
    else
      sol(x, y + 2, z + 1);
    a[x + 1][y] = '.', a[x + 2][y] = '.', a[x + 2][y - 1] = '.',
          a[x + 2][y + 1] = '.', a[x][y] = '.';
  }
  if (a[x + 1][y] == '.' && a[x + 2][y] == '.' && a[x + 1][y + 1] == '.' &&
      a[x + 1][y + 2] == '.' && a[x][y] == '.') {
    a[x][y] = z + 'A', a[x + 1][y] = z + 'A', a[x + 2][y] = z + 'A',
    a[x + 1][y + 1] = z + 'A', a[x + 1][y + 2] = z + 'A';
    if (m - y < 3)
      sol(x + 1, 0, z + 1);
    else
      sol(x, y + 2, z + 1);
    a[x + 1][y] = '.', a[x + 2][y] = '.', a[x + 1][y + 1] = '.',
          a[x + 1][y + 2] = '.', a[x][y] = '.';
  }
  if (a[x + 1][y] == '.' && a[x + 2][y] == '.' && a[x + 1][y - 1] == '.' &&
      a[x + 1][y - 2] == '.' && a[x][y] == '.') {
    a[x][y] = z + 'A', a[x + 1][y] = z + 'A', a[x + 2][y] = z + 'A',
    a[x + 1][y - 1] = z + 'A', a[x + 1][y - 2] = z + 'A';
    if (m - y < 3)
      sol(x + 1, 0, z + 1);
    else
      sol(x, y + 2, z + 1);
    a[x + 1][y] = '.', a[x + 2][y] = '.', a[x + 1][y - 1] = '.',
          a[x + 1][y - 2] = '.', a[x][y] = '.';
  }
  if (m - y < 1) {
    sol(x + 1, 0, z);
  } else
    sol(x, y + 1, z);
}
