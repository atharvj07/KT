n = int(input())
ar = [list(input()) for _ in range(n)]
for i in range(1, n - 1):
	for j in range(1, n - 1):
		if ar[i][j] == '#' and ar[i - 1][j] == '#' and (ar[i + 1][j] == '#') and (ar[i][j + 1] == '#') and (ar[i][j - 1] == '#'):
			ar[i][j] = ar[i + 1][j] = ar[i - 1][j] = ar[i][j + 1] = ar[i][j - 1] = '.'
flag = True
for i in range(n):
	if ar[i].count('#') > 0:
		flag = False
		break
print('YES' if flag else 'NO')
