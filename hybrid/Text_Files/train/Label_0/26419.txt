#include <bits/stdc++.h>
using namespace std;
vector<int> adj[500000];
int centroid[500000];
int sz[500000];
int maxi[500000];
int papa[500000];
set<pair<int, int> > s[500000];
int dsu[500000];
void gen(int i, int p) {
  sz[i] = 1;
  dsu[i] = i;
  bool flag = false;
  pair<int, int> mx = make_pair(-1, -1);
  for (auto j : adj[i])
    if (j != p) {
      gen(j, i);
      sz[i] += sz[j];
      mx = max(mx, make_pair(sz[j], j));
      flag = true;
    }
  if (!flag) {
    s[i].insert(make_pair(1, i));
    centroid[i] = i;
    return;
  }
  dsu[i] = dsu[mx.second];
  s[dsu[i]].insert(make_pair(sz[i], i));
  for (auto j : adj[i])
    if (j != p && j != mx.second) {
      for (auto k : s[dsu[j]]) s[dsu[i]].insert(k);
      s[dsu[j]].clear();
    }
  int siz = s[dsu[i]].size() + 1;
  centroid[i] = (s[dsu[i]].lower_bound(make_pair(siz / 2, -1)))->second;
}
int main() {
  ios_base::sync_with_stdio(false);
  int n, q;
  cin >> n >> q;
  for (int i = 2; i <= n; i++) {
    int p;
    cin >> p;
    adj[i].push_back(p);
    adj[p].push_back(i);
  }
  gen(1, -1);
  while (q--) {
    int x;
    cin >> x;
    cout << centroid[x] << endl;
  }
  return 0;
}
