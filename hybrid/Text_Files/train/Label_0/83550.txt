import sys
input = lambda : sys.stdin.readline().rstrip()
INF = 10 ** 19
import heapq

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	buc = [0] * (n + 1)
	for i in range(n):
		buc[a[i]] += 1
	ans = [buc[0]]
	tw = []
	if buc[0] == 0:
		ans.append(-1)
	else:
		ans.append(buc[1])
	if buc[0] >= 2:
		heapq.heappush(tw, 0)
	if buc[1] >= 2:
		heapq.heappush(tw, -1)
	cost = 0
	for i in range(2, n + 1):
		if ans[-1] == -1:
			ans.append(-1)
			continue
		elif buc[i - 1] == 0:
			if len(tw) == 0:
				ans.append(-1)
				continue
			now = heapq.heappop(tw)
			now = -now
			buc[now] -= 1
			buc[i - 1] += 1
			cost += abs(now - i + 1)
			if buc[now] >= 2:
				heapq.heappush(tw, -now)
		if buc[i] >= 2:
			heapq.heappush(tw, -i)
		ans.append(cost + buc[i])
	print(*ans)
t = int(input())
for _ in range(t):
	solve()
