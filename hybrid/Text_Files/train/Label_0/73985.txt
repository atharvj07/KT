import sys
input = sys.stdin.readline
(n, T) = map(int, input().split())
S = input().strip()
L = [ord(s) - 97 for s in S]
T += pow(2, L[-2]) - pow(2, L[-1])
L = L[:-2]
L.sort(reverse=True)
for i in range(len(L)):
	if T > 0:
		T -= 1 << L[i]
	else:
		T += 1 << L[i]
if T == 0:
	print('Yes')
else:
	print('No')
