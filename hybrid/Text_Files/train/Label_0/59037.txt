import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Scanner;
 
public class C {

	static Scanner sc = new Scanner(System.in);
	static PrintWriter out = new PrintWriter(System.out);

    static void exit() {
    	sc.close();
    	out.close();
    	System.exit(0);
    }
    
    public static void main(String[] args) {
    	long a = sc.nextLong();
    	long b = sc.nextLong();
    	if(b <= 1) {
    		out.println(a*a-b*b);
    		while(a>0) {
    			a--;
    			out.print('o');
    		}
    		while(b>0) {
    			b--;
    			out.print('x');
    		}
    		out.println();
    		exit();
    	}
    	if(a == 0) {
    		out.println(-b*b);
    		out.println(build('x', b));
    		exit();
    	}
    	long max_score = Long.MIN_VALUE;
    	int best_ga = 0;
    	int best_gb = 0;
    	for(int gb = 1; gb <= Math.min(a+1, b); gb++) {
    		long rem = b % gb;
    		long base = b / gb;
    		long lose = rem * (base+1)*(base+1) + (gb-rem)*base*base;
    		
    		int ga = Math.max(1, gb-1);
    		long gain = (a-ga+1)*(a-ga+1) + ga-1;
    		//out.println(gain+" "+lose);
    		if(max_score < gain-lose) {
    			max_score = gain-lose;
    			best_ga = ga;
    			best_gb = gb;
    		}
    	}
    	out.println(max_score);
    	long remb = b % best_gb;
    	long baseb = b / best_gb;
    	ArrayList<String> a_list = new ArrayList<>(best_ga);
    	ArrayList<String> b_list = new ArrayList<>(best_gb);
    	a_list.add(build('o', a-best_ga+1));
    	for(int i = 2; i <= best_ga; i++) {
    		a_list.add("o");
    	}
    	for(int i = 1; i <= best_gb; i++) {
    		if(i <= remb) {
    			b_list.add(build('x',baseb+1));
    		} else {
    			b_list.add(build('x',baseb));
    		}
    	}
    	int i = 0;
    	for(i = 0; i < best_ga; i++) {
    		out.print(b_list.get(i)+a_list.get(i));
    	}
    	while(i < best_gb) {
    		out.print(b_list.get(i));
    		i++;
    	}
    	out.println();
        exit();
    }
    static String build(char c, long n) {
    	StringBuilder s = new StringBuilder(""+c);
    	for(int i = 2; i <= n; i++) s.append(""+c);
    	return s.toString();
    }
}
