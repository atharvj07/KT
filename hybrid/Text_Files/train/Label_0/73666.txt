from sys import stdin
input = stdin.readline
from collections import defaultdict
for _ in range(int(input())):
	n = int(input())
	lis = list(map(int, input().split()))
	if 2048 in lis:
		print('YES')
		continue
	for i in range(n):
		if lis[i] > 2048:
			lis[i] = -1
	while -1 in lis:
		lis.remove(-1)
	d = defaultdict(int)
	for i in lis:
		d[i] += 1
	while True:
		l = sorted(d)
		for i in l:
			if d[i] > 1:
				x = d[i] // 2
				d[i] = 1
				d[2 * i] += x
				break
		else:
			break
		if l[-1] > 2048:
			break
	if d[2048] > 0:
		print('YES')
	else:
		print('NO')
