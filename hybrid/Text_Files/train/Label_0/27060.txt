#include <bits/stdc++.h>
using namespace std;

struct Edge{
	int src,dst,weight;
	Edge(){}
	Edge(int s,int d,int w):src(s),dst(d),weight(w){}
};
typedef vector<vector<Edge>> Graph;

int main()
{
	for(int n;cin>>n && n;){
		Graph g(n);
		for(int u=0;u<n;u++)
			for(int j=0;j<2;j++){
				int v,w; cin>>v>>w;
				g[u].emplace_back(u,v,w);
			}
		
		int res=1;
		vector<int> vis(n);
		for(int i=0;i<n;i++) if(!vis[i]){
			vector<int> ws;
			int p=-1,u=i;
			do{
				vis[u]=1;
				for(auto e:g[u]) if(e.dst!=p){
					ws.push_back(e.weight);
					p=u,u=e.dst;
					break;
				}
			}while(u!=i);
			(res*=count(begin(ws),end(ws),*max_element(begin(ws),end(ws))))%=10007;
		}
		cout<<res<<endl;
	}
}
