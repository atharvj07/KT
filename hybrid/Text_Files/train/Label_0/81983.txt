(n, m) = input().split()
n = int(n)
m = int(m)
c = list(map(int, input().split()))
a = list(map(int, input().split()))
b = 0
i = 0
j = 0
while n != 0:
	if m > 0:
		if c[i] <= a[j]:
			b = b + 1
			i = i + 1
			j = j + 1
			n = n - 1
			m = m - 1
		else:
			i = i + 1
			n = n - 1
	else:
		break
print(b)
