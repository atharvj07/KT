score = int(input())
a_list = ['zero', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'nine', 'ten', 'eleven', 'twelve', 'thirteen', 'fourteen', 'fifteen', 'sixteen', 'seventeen', 'eighteen', 'nineteen']
b_list = ['', '', 'twenty', 'thirty', 'forty', 'fifty', 'sixty', 'seventy', 'eighty', 'ninety']
if score < 20:
	print(a_list[score])
else:
	print(b_list[score // 10], end='')
	if score % 10 > 0:
		print('-', end='')
		print(a_list[score % 10])
