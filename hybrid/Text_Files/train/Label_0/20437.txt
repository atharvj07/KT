#include <bits/stdc++.h>
using namespace std;
int n, ar[150010], suf[150010], k1, k2, res[150010], q, mn = (int)1e9, k, mk;
int main() {
  scanf("%d %d", &n, &k);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &ar[i]);
  }
  for (int i = 1; i <= n; i++) {
    suf[i] = suf[i - 1] + ar[i];
  }
  for (int i = 1; i <= n - k + 1; i++) {
    res[i] = suf[i + k - 1] - suf[i - 1];
  }
  for (int i = 1; i <= n - k + 1; i++) {
    if (res[i] < mn) mn = res[i], mk = i;
  }
  printf("%d", mk);
}
