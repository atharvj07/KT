
import java.util.Scanner;


public class B987 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int y = sc.nextInt(); 
        
        if ((x == 2 && y == 4 ) || (x == 4 && y == 2))
            System.out.println("=");
        else
            System.out.println(x == y ? "=" : ( y == 0 || Math.pow(x,(double) y / x) < y ? "<" : ">"));
    }
    
}
