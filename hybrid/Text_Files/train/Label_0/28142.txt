#include <bits/stdc++.h>
using namespace std;
long long X[4], Y[4];
long long x, y, a, b, c, n, t;
long long time0[4], time1[4];
long long rem[4];
long long all_add(long long num) { return num * (num + 1) / 2; }
void add(long long tx, long long ty) {
  for (long long i = 0; i < ((long long)4); i++) {
    if (x != tx) {
      long long t0 = min(abs(x - X[i]), abs(tx - X[i]));
      long long t1 = max(abs(x - X[i]), abs(tx - X[i]));
      time0[i] += all_add(t1) - all_add(t0 - 1);
      time0[i] -= abs(x - X[i]);
    } else {
      time0[i] += abs(x - X[i]) * abs(y - ty);
    }
    if (y != ty) {
      long long t0 = min(abs(y - Y[i]), abs(ty - Y[i]));
      long long t1 = max(abs(y - Y[i]), abs(ty - Y[i]));
      time0[i] += all_add(t1) - all_add(t0 - 1);
      time0[i] -= abs(y - Y[i]);
    } else {
      time0[i] += abs(y - Y[i]) * abs(x - tx);
    }
  }
}
bool check() {
  long long sx = x;
  long long sy = y;
  for (long long i = 0; i < ((long long)4); i++) time0[i] = time1[i] = 0;
  if (x < X[0]) {
    add(X[0], y);
    x = X[0];
  }
  if (y < Y[0]) {
    add(x, Y[0]);
    y = Y[0];
  }
  if (x < X[3]) {
    add(X[3], y);
    x = X[3];
  }
  if (y < Y[3]) {
    add(x, Y[3]);
    y = Y[3];
  }
  if (x < n) {
    add(n, y);
    x = n;
  }
  if (y < n) {
    add(x, n);
    y = n;
  }
  for (long long i = 0; i < ((long long)4); i++) time1[i] = time0[i];
  for (long long i = 0; i < ((long long)4); i++) time0[i] = 0;
  x = sx;
  y = sy;
  if (x < X[0]) {
    add(X[0], y);
    x = X[0];
  }
  if (y < Y[0]) {
    add(x, Y[0]);
    y = Y[0];
  }
  if (y < Y[3]) {
    add(x, Y[3]);
    y = Y[3];
  }
  if (x < X[3]) {
    add(X[3], y);
    x = X[3];
  }
  if (x < n) {
    add(n, y);
    x = n;
  }
  if (y < n) {
    add(x, n);
    y = n;
  }
  x = sx;
  y = sy;
  if (rem[0] < time0[0]) return false;
  if (rem[3] < time0[3]) return false;
  for (long long i = 0; i < ((long long)2); i++)
    if (rem[i + 1] < min(time0[i + 1], time1[i + 1])) return false;
  for (long long i = 0; i < ((long long)2); i++)
    if (rem[i + 1] >= max(time0[i + 1], time1[i + 1])) return true;
  if (rem[1] < min(time0[1], time1[1]) +
                   2 * ((max(time0[2], time1[2]) - rem[2] + 1) / 2))
    return false;
  return true;
}
void recur() {
  if (x == n && y == n) return;
  if (x < n) {
    x++;
    for (long long i = 0; i < ((long long)4); i++)
      rem[i] -= abs(x - X[i]) + abs(y - Y[i]);
    if (check()) {
      cout << "R";
      recur();
      return;
    }
    for (long long i = 0; i < ((long long)4); i++)
      rem[i] += abs(x - X[i]) + abs(y - Y[i]);
    x--;
  }
  y++;
  for (long long i = 0; i < ((long long)4); i++)
    rem[i] -= abs(x - X[i]) + abs(y - Y[i]);
  cout << "U";
  recur();
}
int main() {
  cin >> n >> t >> a >> b >> c;
  X[0] = a;
  Y[0] = b;
  X[1] = a;
  Y[1] = b + c - 1;
  X[2] = a + c - 1;
  Y[2] = b;
  X[3] = a + c - 1;
  Y[3] = b + c - 1;
  for (long long i = 0; i < ((long long)4); i++) rem[i] = t;
  x = y = 1;
  if (!check()) {
    cout << "Impossible" << endl;
  } else {
    recur();
    cout << endl;
  }
}
