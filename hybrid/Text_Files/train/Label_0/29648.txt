#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
  ll n,k;
  cin>>n>>k;
  vector<ll> t(n),d(n),a[100005];
  vector<pair<ll,ll>> p;
  set<ll> s;
  for(int i=0;i<n;i++){
    cin>>t[i]>>d[i];
    p.push_back(make_pair(d[i],t[i]));
  }
  sort(p.begin(),p.end(),greater<pair<ll,ll>>());
  ll c=0;
  priority_queue<ll,vector<ll>,greater<ll>> q;
  for(int i=0;i<k;i++){
    int j=s.size();
    c+=p[i].first;
    s.insert(p[i].second);
    if(j==s.size()) q.push(p[i].first);
  }
  ll m=c+(ll)s.size()*(ll)s.size();
  for(int i=k;i<n;i++){
    if(q.empty()) break;
    int j=s.size();
    s.insert(p[i].second);
    if(j!=s.size()){
      c+=p[i].first;
      c-=q.top();
      q.pop();
      m=max(m,c+(ll)(j+1)*(ll)(j+1));
    }
  }
  cout<<m;
}
