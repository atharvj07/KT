(N, A, R, M) = map(int, input().split())
H = list(map(int, input().split()))

def f(t):
	a = 0
	r = 0
	for h in H:
		r += max(h - t, 0)
		a += max(t - h, 0)
	m = min(a, r)
	return min(a * A + r * R, m * M + A * (a - m) + R * (r - m))
l = min(H)
r = max(H)
while r - l > 3:
	m1 = l + (r - l) // 3
	m2 = r - (r - l) // 3
	if f(m1) > f(m2):
		l = m1
	else:
		r = m2
print(min(f(l), f(l + 1), f(l + 2), f(l + 3)))
