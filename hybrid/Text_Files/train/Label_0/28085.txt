#include <bits/stdc++.h>
using namespace std;
int N, M, a[200010], dp[200010];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> N >> M;
  int i, l, r, x;
  for (i = 1; i <= N; i++) cin >> a[i];
  for (i = N; i > 0; i--) {
    dp[i] = i;
    if (a[i + 1] == a[i]) dp[i] = dp[i + 1];
  }
  while (M--) {
    cin >> l >> r >> x;
    if (a[l] != x)
      cout << l << "\n";
    else {
      if (dp[l] >= r)
        cout << -1 << "\n";
      else
        cout << dp[l] + 1 << "\n";
    }
  }
  return 0;
}
