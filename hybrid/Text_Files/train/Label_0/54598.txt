public class C_EfimAndStrangeGrade {

    static int n;
    static int second;
    static String numstring;
    static int i;
    static int last;
    static char[] arr;
    static int plus = 0;
    static int add = 0;

    public static void main (String[] args) {
        Scanner in = new Scanner(System.in);
        String[] line1 = in.nextLine().split(" ");
        n = Integer.parseInt(line1[0]);
        second = Integer.parseInt(line1[1]);
        numstring = in.nextLine();
        i = numstring.indexOf('.');
        last = i + 1;
        arr = numstring.toCharArray();
        boolean is = false;
        for (; last < numstring.length(); last++) {
            if (numstring.charAt(last) >= '5') {
                is = true;
                round();
                if (last >= i + 1)
                    System.out.println(new String(arr).substring(0, last + 1));
                else {
                    if (add == 0) {
                        System.out.println(new String(arr).substring(0, i));
                    } else {
                        System.out.println("1" + new String(arr).substring(0, i));
                    }
                }
                break;
            }
        }
        if (!is) {
            System.out.println(numstring);
        }
    }

    public static void round () {
        if (plus != 1) {
            second--;
        }
        if (second < 0) {
            return;
        }
        if (last > i + 1) {
            if (arr[last - 1] == '9') {
                arr[last - 1] = '0';
                plus = 1;
            } else {
                arr[last - 1]++;
                arr[last - 1] += plus;
            }
            last--;
        } else if (last == i + 1) {
            if (arr[last - 2] == '9') {
                plus = 1;
                arr[last - 2] = '0';
                last -= 2;
                if (last == 0) {
                    add = 1;
                    return;
                }
                round();
            } else {
                arr[last - 2]++;
                last -= 2;
            }
            return;
        } else {
            if (arr[last - 1] == '9') {
                plus = 1;
                arr[last - 1] = '0';
                if (last == 1) {
                    add = 1;
                    return;
                }
                last -= 1;
                round();
            } else {
                arr[last - 1]++;
                last -= 1;
            }
            return;
        }

        if (arr[last] >= '5') {
            round();
        }
        return;
    }
}
