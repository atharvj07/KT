import sys
input = sys.stdin.readline

def fun(x):
	su = 0
	global ar, n
	for i in range(n):
		su += max(0, ar[i] - i // x)
	return bool(su >= m)
(n, m) = map(int, input().split())
ar = list(map(int, input().split()))
ar.sort(reverse=True)
if sum(ar) < m:
	print(-1)
else:
	l = 1
	r = n
	while l <= r:
		mid = (l + r) // 2
		if fun(mid):
			r = mid - 1
		else:
			l = mid + 1
	print(l)
