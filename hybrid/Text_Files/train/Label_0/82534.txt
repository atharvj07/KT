import math
debug = True

def pprint(*args, delimeter='#', amount=10):
	if debug:
		print('{0} DEBUG {0}'.format(delimeter * amount))
		print(*args)
		print('{0} ----- {0}'.format('-' * amount))

def solve():
	n = int(input())
	s = list(input())
	for i in range(n // 2):
		if s[i] == '0':
			print(i + 1, n, i + 2, n)
			return
	for i in range(n // 2, n):
		if s[i] == '0':
			print(1, i + 1, 1, i)
			return
	print(1, n - 1, 2, n)
t = int(input())
while t > 0:
	t -= 1
	solve()
