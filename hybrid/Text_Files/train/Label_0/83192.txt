s = input()
s = list(s)
n = len(s)
if n < 26:
	print(-1)
elif 'a' not in s:
	print(-1)
else:
	ind = s.index('a')
	if n - ind < 26:
		print(-1)
	else:
		i = 0
		j = 0
		k = 'abcdefghijklmnopqrstuvwxyz'
		k = list(k)
		while i < len(s):
			if j == 0 and s[i] == k[j]:
				i += 1
				j += 1
			elif j != 0 and j < 26 and (s[i] in k[:j + 1]):
				s[i] = k[j]
				j += 1
				i += 1
			else:
				i += 1
		if j > 25:
			for i1 in range(len(s)):
				print(s[i1], end='')
		else:
			print(-1)
