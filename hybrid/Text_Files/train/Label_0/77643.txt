n = int(input())
a = [input() for i in range(n)]
count = lambda p: p * (p - 1) // 2
cnt1 = sum((count(a[i].count('C')) for i in range(n)))
cnt2 = sum((count(sum((a[j][i] == 'C' for j in range(n)))) for i in range(n)))
ans = cnt1 + cnt2
from random import choice
b = [0] * 9 + [1]
ans += choice(b)
print(cnt1 + cnt2)
