import math

def array_elimination(n, a):
	counts = {i: 0 for i in range(32)}
	for ai in a:
		bin_ai = '{:032b}'.format(ai)
		for i in range(32):
			if bin_ai[i] == '1':
				counts[i] += 1
	groups = []
	for i in range(32):
		if counts[i] > 0:
			groups += [counts[i]]
	if len(groups) == 0:
		print(' '.join(map(str, range(1, n + 1))))
		return
	counts_by_divisor = {}
	for group_size in groups:
		for divisor in get_divisors(group_size):
			if divisor in counts_by_divisor:
				counts_by_divisor[divisor] += 1
			else:
				counts_by_divisor[divisor] = 1
	k = []
	for divisor in counts_by_divisor:
		if counts_by_divisor[divisor] == len(groups):
			k += [divisor]
	print(' '.join(map(str, sorted(k))))

def get_divisors(value):
	m = int(math.ceil(math.sqrt(value)))
	for d in range(1, m + 1):
		if value % d == 0:
			e = value // d
			if d < e:
				yield d
				yield e
			elif d == e:
				yield d
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split(' ')))
	array_elimination(n, a)
