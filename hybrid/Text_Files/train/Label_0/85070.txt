t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	pref = [[0, 0] for i in range(n + 1)]
	for i in range(1, n + 1):
		if s[i - 1] == 'R':
			pref[i][0] = 0
			pref[i][1] = pref[i - 1][0] + 1
		else:
			pref[i][0] = pref[i - 1][1] + 1
			pref[i][1] = 0
	suff = [[0, 0] for i in range(n + 1)]
	for i in range(n - 1, -1, -1):
		if s[i] == 'L':
			suff[i][0] = 0
			suff[i][1] = suff[i + 1][0] + 1
		else:
			suff[i][0] = suff[i + 1][1] + 1
			suff[i][1] = 0
	print(' '.join((str(pref[i][0] + suff[i][0] + 1) for i in range(n + 1))))
