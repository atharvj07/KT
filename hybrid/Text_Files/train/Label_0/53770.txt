import java.util.*;
import java.io.*;
public class PreorderTest
{
	public static int[][] adj;
	public static boolean[] removed;
	public static boolean[] visited;
	public static int[] dp;
	public static int best;
	
	public static void main(String[] args) throws Exception
	{
		FastScanner in = new FastScanner(System.in);
		
		int n = in.nextInt();
		int k = in.nextInt();
		
		int[] as = new int[n];
		for(int x = 0; x < n; x++)
		{
			as[x] = in.nextInt();
		}
		
		adj = new int[n][];
		
		int[] degree = new int[n];
		int[] us = new int[n - 1];
		int[] vs = new int[n - 1];
		for(int y = 0; y < n - 1; y++)
		{
			us[y] = in.nextInt() - 1;
			vs[y] = in.nextInt() - 1;
			degree[us[y]]++;
			degree[vs[y]]++;
		}
		
		for(int z = 0; z < adj.length; z++)
		{
			adj[z] = new int[degree[z]];
		}
		
		int[] indices = new int[n];
		for(int a = 0; a < us.length; a++)
		{
			adj[us[a]][indices[us[a]]] = vs[a];
			adj[vs[a]][indices[vs[a]]] = us[a];
			indices[us[a]]++;
			indices[vs[a]]++;
		}
		
		int low = 1;
		int high = 1000000;
		while(low < high)
		{
			int mid = (low + high + 1) / 2;
			
			removed = new boolean[n];
			for(int x = 0; x < as.length; x++)
			{
				if(as[x] < mid)
				{
					removed[x] = true;
				}
			}
			
			int nodes = calc();
			
			if(nodes >= k)
			{
				low = mid;
			}
			else
			{
				high = mid - 1;
			}
		}
		
		System.out.println(low);
	}
	
	public static int calc()
	{
		visited = new boolean[adj.length];
		dp = new int[adj.length];
		
		best = 0;
		for(int a = 0; a < adj.length; a++)
		{
			if(!visited[a] && !removed[a])
			{
				int total = count(a, -1);
				
				move(a, -1, 0);
			}
		}
		
		return best;
	}
	
	public static int count(int node, int parent)
	{
		visited[node] = true;
		
		int ret = 1;
		boolean complete = true;
		int max = 0;
		for(int next : adj[node])
		{
			if(next != parent && !removed[next])
			{
				ret += count(next, node);
				
				if(dp[next] < 0)
				{
					dp[node] -= dp[next];
				}
				else
				{
					max = Math.max(max, dp[next]);
					complete = false;
				}
			}
			else if(removed[next])
			{
				complete = false;
			}
		}
		
		dp[node]++;
		dp[node] += max;
		
		if(complete)
		{
			dp[node] *= -1;
		}
		
		return ret;
	}
	
	public static void move(int node, int parent, int prev)
	{
		int count = 0;
		int total = 0;
		int max1 = 0;
		int max2 = 0;
		
		if(parent != -1 && prev >= 0)
		{
			count++;
			max1 = prev;
		}
		else
		{
			total -= prev;
		}
		
		for(int next : adj[node])
		{
			if(next != parent)
			{
				if(dp[next] >= 0)
				{
					count++;
					if(dp[next] > max1)
					{
						max2 = max1;
						max1 = dp[next];
					}
					else if(dp[next] > max2)
					{
						max2 = dp[next];
					}
				}
				else
				{
					total -= dp[next];
				}
			}
		}
		
		total++;
		
		best = Math.max(best, total + max1);
		
		for(int next : adj[node])
		{
			if(next != parent && !removed[next])
			{
				if(dp[next] < 0)
				{
					int nextPrev = total + max1 + dp[next];
					if(count == 0)
					{
						nextPrev *= -1;
					}
					
					move(next, node, nextPrev);
				}
				else
				{
					int nextPrev = total + max1;
					if(dp[next] == max1)
					{
						nextPrev += max2 - max1;
					}
					
					if(count == 1)
					{
						nextPrev *= -1;
					}
					
					move(next, node, nextPrev);
				}
			}
		}
	}
	
	static class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;
		
		public FastScanner(InputStream input)
		{
			br = new BufferedReader(new InputStreamReader(input));
			st = new StringTokenizer("");
		}
		
		public String next() throws IOException
		{
			if(st.hasMoreTokens())
			{
				return st.nextToken();
			}
			else
			{
				st = new StringTokenizer(br.readLine());
				return next();
			}
		}
		
		public int nextInt() throws IOException
		{
			return Integer.parseInt(next());
		}
	}
}
