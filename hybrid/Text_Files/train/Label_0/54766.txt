import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class D 
{
    private static final int m = 13;
    private static int [][] p;
    private static Comparator<int[]> cmp = new Comparator<int[]>() {
        
        @Override
        public int compare(int[] a, int[] b) {
            if(a[0] != b[0])
                return Integer.compare(a[0], b[0]);
            return Integer.compare(a[1], b[1]);
        }
    };
    public static void main(String [] args) throws IOException
    {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        char [] str = reader.readLine().toCharArray();
        char [] st = reader.readLine().toCharArray();
        char [] en = reader.readLine().toCharArray();
        
        int [] zst = Z(str, st);
        int [] zen = Z(str, en);
        
        boolean [] stGood = new boolean[str.length];
        for(int i = 0 ; i < str.length ; i++)
            if(zst[i+st.length+1] == st.length)
                stGood[i] = true;
        
        boolean [] enGood = new boolean[str.length];
        for(int i = 0 ; i < str.length ; i++)
            if(zen[i+en.length+1] == en.length)
                enGood[i+en.length-1] = true;
        
        int n = str.length;
        p = new int[m][n];
        for(int i = 0 ; i < n ; i++)
            p[0][i] = str[i];
        
        int [][] order = new int[n][3];
        for(int k = 1 ; k < m ; k++)
        {
            for(int i = 0 ; i < n ; i++)
            {
                order[i][0] = p[k-1][i];
                order[i][1] = i + (1<<(k-1)) < n ? p[k-1][i+(1<<(k-1))] : -1;
                order[i][2] = i;
            }
            
            Arrays.sort(order, cmp);
            
            for(int i = 0 ; i < n ; i++)
                p[k][order[i][2]] = i > 0 && order[i][0] == order[i-1][0] && order[i][1] == order[i-1][1] ? p[k][order[i-1][2]] : i;
        }
        
        int [] suffixes = new int[n];
        for(int i = 0 ; i < n ; i++)
            suffixes[p[m-1][i]] = i;
        
        int res = 0;
        for(int i = 0 ; i < n ; i++)
        {
            int a = suffixes[i];
            if(stGood[a])
                for(int b = a ; b < n ; b++)
                    if(enGood[b])
                        if(b-a+1 >= st.length)
                            if(b-a+1 >= en.length)
                                if(i+1 >= n || lcp(a, suffixes[i+1]) < b-a+1)
                                    res++;
        }
        
        System.out.println(res);
    }
    
    private static int lcp(int x, int y)
    {
        int n = p[0].length;
        if(x == y)
            return n-x;
        
        int res = 0;
        for(int k = m-1 ; k >= 0 ; k--)
            if(x + (1<<k) <= n && y + (1<<k) <= n && p[k][x] == p[k][y])
            {
                res += 1<<k;
                x += 1<<k;
                y += 1<<k;
            }
        
        return res;
    }
    
    private static int [] Z(char [] a, char [] b)
    {
        char [] str = (new String(b)+"$"+new String(a)).toCharArray();
        
        int n = str.length;
        int left = 0, right = 0;
        
        int [] z = new int[n];
        for(int i = 1 ; i < n ; i++)
        {
            if(i > right)
            {
                left = right = i;
                while(right < n && str[right] == str[right-left])
                    right++;
                z[i] = right-left;
                right--;
            }
            else
            {
                int k = i - left;
                if(z[k] < right-i)
                    z[i] = z[k];
                else
                {
                    left = i;
                    while(right < n && str[right] == str[right-left])
                        right++;
                    z[i] = right-left;
                    right--;
                }
            }
        }
        
        return z;
    }
}
