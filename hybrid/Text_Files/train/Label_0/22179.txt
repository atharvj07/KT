#include <bits/stdc++.h>
using namespace std;
mt19937 mt_rand(chrono::system_clock::now().time_since_epoch().count());
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m;
  cin >> n >> m;
  vector<string> a(n);
  vector<bool> need(m, false);
  vector<bool> cov(m, false);
  vector<int> cp(n, -1);
  vector<vector<int> > g(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '#') {
        need[j] = true;
      }
    }
  }
  int cc = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < i; j++) {
      bool common = false;
      for (int q = 0; q < m; q++) {
        common |= (a[i][q] == a[j][q] && a[i][q] == '#');
      }
      if (common) {
        cp[i] = cp[j];
        break;
      }
    }
    if (cp[i] == -1) {
      cp[i] = cc++;
    }
  }
  for (int i = 0; i < n; i++) {
    g[cp[i]].push_back(i);
  }
  for (int i = 0; i < cc; i++) {
    for (int j = 0; j < m; j++) {
      bool cover = true;
      for (int x : g[i]) {
        if (a[x][j] == '.') {
          cover = false;
          break;
        }
      }
      if (cover) {
        for (int x : g[i]) {
          a[x][j] = '.';
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '#') {
        cout << "No\n";
        return 0;
      }
    }
  }
  cout << "Yes\n";
  return 0;
}
