import sys
from math import ceil

def main(args=None):
	if args is None:
		(n, t, k, d) = [int(tok.strip()) for tok in sys.stdin.readline().split()]
	else:
		(n, t, k, d) = args
	t1 = ceil(n / k) * t
	t_min = t1
	for tx in range(d + 1, t1):
		n2 = tx // t * k + (tx - d) // t * k
		if n2 >= n:
			t_min = tx
			break
	if t_min < t1:
		return 'YES'
	else:
		return 'NO'
print(main())
