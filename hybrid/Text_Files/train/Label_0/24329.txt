#include <bits/stdc++.h>
using namespace std;
const int N = 55;
int lot[4][N];
int n;
struct Move {
  int i, r, c;
};
vector<Move> moves;
pair<int, int> findEmpty() {
  for (int j = 1; j <= n; j++)
    for (int i : {1, 2})
      if (!lot[i][j]) return {i, j};
  return {-1, -1};
}
void move(pair<int, int> a, pair<int, int> b) {
  if (!lot[a.first][a.second]) return;
  moves.push_back({lot[a.first][a.second], b.first + 1, b.second});
  lot[b.first][b.second] = lot[a.first][a.second];
  lot[a.first][a.second] = 0;
}
void trivial() {
  for (int j = 1; j <= n; j++)
    if (lot[0][j] == lot[1][j]) move({1, j}, {0, j});
  for (int j = 1; j <= n; j++)
    if (lot[2][j] == lot[3][j]) move({2, j}, {3, j});
}
bool check() {
  for (int j = 1; j <= n; j++)
    for (int i : {1, 2})
      if (lot[i][j]) return true;
  return false;
}
pair<int, int> next(pair<int, int> a) {
  if (a.first == 1) {
    if (a.second < n)
      return {a.first, a.second + 1};
    else
      return {a.first + 1, a.second};
  } else {
    if (a.second > 1)
      return {a.first, a.second - 1};
    else
      return {a.first - 1, a.second};
  }
}
int main() {
  int k;
  cin >> n >> k;
  for (int i = 0; i < 4; i++)
    for (int j = 1; j <= n; j++) cin >> lot[i][j];
  trivial();
  pair<int, int> p = findEmpty();
  if (p.first == -1) {
    cout << -1 << endl;
    return 0;
  }
  while (check()) {
    trivial();
    pair<int, int> q = next(p);
    move(q, p);
    p = q;
  }
  cout << moves.size() << endl;
  for (Move mov : moves) printf("%d %d %d\n", mov.i, mov.r, mov.c);
}
