#include <bits/stdc++.h>
using namespace std;
template <typename T, typename U>
inline void smin(T &a, U b) {
  if (a > b) a = b;
}
template <typename T, typename U>
inline void smax(T &a, U b) {
  if (a < b) a = b;
}
template <class T>
inline void gn(T &first) {
  char c, sg = 0;
  while (c = getchar(), (c > '9' || c < '0') && c != '-')
    ;
  for ((c == '-' ? sg = 1, c = getchar() : 0), first = 0; c >= '0' && c <= '9';
       c = getchar())
    first = (first << 1) + (first << 3) + c - '0';
  if (sg) first = -first;
}
template <class T, class T1>
inline void gn(T &first, T1 &second) {
  gn(first);
  gn(second);
}
template <class T, class T1, class T2>
inline void gn(T &first, T1 &second, T2 &z) {
  gn(first);
  gn(second);
  gn(z);
}
template <class T>
inline void print(T first) {
  if (first < 0) {
    putchar('-');
    return print(-first);
  }
  if (first < 10) {
    putchar('0' + first);
    return;
  }
  print(first / 10);
  putchar(first % 10 + '0');
}
template <class T>
inline void printsp(T first) {
  print(first);
  putchar(' ');
}
template <class T>
inline void println(T first) {
  print(first);
  putchar('\n');
}
template <class T, class U>
inline void print(T first, U second) {
  printsp(first);
  println(second);
}
template <class T, class U, class V>
inline void print(T first, U second, V z) {
  printsp(first);
  printsp(second);
  println(z);
}
int power(int a, int b, int m, int ans = 1) {
  for (; b; b >>= 1, a = 1LL * a * a % m)
    if (b & 1) ans = 1LL * ans * a % m;
  return ans;
}
int main() {
  const int MOD = int(1e9 + 7), NN = 1000100;
  static int prime[NN], flag[NN], np, val[NN], mul[NN], sqr[NN], fac[NN],
      nxt[NN], pre[NN];
  int n;
  gn(n);
  mul[1] = 1;
  val[1] = 1;
  flag[1] = 1;
  sqr[1] = 1;
  for (int i = 2; i <= n; i++) {
    if (!flag[i]) flag[i] = i, prime[np++] = i, mul[n / i]++;
    for (int j = 0; j < np && prime[j] * i <= n; j++) {
      flag[prime[j] * i] = flag[i];
      if (i % prime[j] == 0) break;
    }
    int v = i / flag[i];
    val[i] = val[v];
    if (flag[v] != flag[i]) val[i] *= flag[i];
    sqr[val[i]]++;
  }
  for (int i = fac[0] = 1; i <= n; i++)
    fac[i] = (long long)fac[i - 1] * i % MOD;
  int u;
  auto g = [&n](int u) { return u == 1 ? 1 : n / u; };
  auto finish = []() {
    puts("0");
    exit(0);
  };
  for (int i = 1; i <= n; i++) {
    gn(u);
    if (!u) continue;
    u = val[u];
    int v = val[i];
    if (u / flag[u] != v / flag[v] || g(flag[u]) != g(flag[v])) finish();
    sqr[u]--;
    u = flag[u];
    v = flag[v];
    if (nxt[u] && nxt[u] != v || pre[v] && pre[v] != u) finish();
    if (!nxt[u] && !pre[v]) nxt[u] = v, pre[v] = u, mul[g(u)]--;
  }
  int ans = 1;
  for (int i = 1; i <= n; i++)
    ans = (long long)ans * fac[mul[i]] % MOD * fac[sqr[i]] % MOD;
  println(ans);
  return 0;
}
