import math
test = int(input())
for i in range(test):
	(a, b) = map(int, input().split())
	if a % 2 == 0:
		if b % 2 == 0:
			if b // 2 % 2 == 0:
				print(a)
			else:
				print(a + 1)
		elif b // 2 % 2 == 0:
			print(a - b)
		else:
			print(a + 1 + b)
	elif b % 2 == 0:
		if b // 2 % 2 == 0:
			print(a)
		else:
			print(a - 1)
	elif b // 2 % 2 == 0:
		print(a + b)
	else:
		print(a - 1 - b)
