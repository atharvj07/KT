import java.io.*;
import java.util.*;

public class CF1384F extends PrintWriter {
	CF1384F() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1384F o = new CF1384F(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int m = sc.nextInt();
		int[] ii = new int[n * m + 1];
		int[] jj = new int[n * m + 1];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) {
				int a = sc.nextInt();
				ii[a] = i;
				jj[a] = j;
			}
		int[][] aa = new int[n][m];
		boolean[] row = new boolean[n];
		boolean[] col = new boolean[m];
		int[] qu = new int[n * m]; int head = 0, cnt = 0;
		int i_ = -1, j_ = -1;
		for (int a = n * m; a >= 1; a--) {
			int i = ii[a], j = jj[a];
			if (!row[i] && !col[j]) {
				row[i] = col[j] = true;
				aa[++i_][++j_] = a;
				for (i = i_ - 1; i >= 0; i--)
					qu[head + cnt++] = i * m + j_;
				for (j = j_ - 1; j >= 0; j--)
					qu[head + cnt++] = i_ * m + j;
			} else if (!row[i]) {
				row[i] = true;
				aa[++i_][j_] = a;
				for (j = j_ - 1; j >= 0; j--)
					qu[head + cnt++] = i_ * m + j;
			} else if (!col[j]) {
				col[j] = true;
				aa[i_][++j_] = a;
				for (i = i_ - 1; i >= 0; i--)
					qu[head + cnt++] = i * m + j_;
			} else {
				int h = qu[head++]; cnt--;
				aa[h / m][h % m] = a;
			}
		}
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++)
				print(aa[i][j] + " ");
			println();
		}
	}
}
