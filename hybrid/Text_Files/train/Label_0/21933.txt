#include <bits/stdc++.h>
using namespace std;
class DSU {
 private:
  vector<long long> parent;
  vector<long long> rank;

 public:
  DSU(long long size);
  long long find(long long a);
  long long size1(long long a);
  void union1(long long a, long long b);
};
DSU::DSU(long long v) {
  parent.resize(v);
  for (long long i = 0; i < v; i++) {
    parent[i] = i;
  }
  rank.resize(v, 1);
}
void DSU::union1(long long a, long long b) {
  long long afind = find(a);
  long long bfind = find(b);
  if (afind != bfind) {
    if (rank[afind] >= rank[bfind]) {
      rank[afind] += rank[bfind];
      rank[bfind] = 0;
      parent[bfind] = parent[afind];
    } else {
      rank[bfind] += rank[afind];
      rank[afind] = 0;
      parent[afind] = parent[bfind];
    }
  }
  return;
}
long long DSU::find(long long a) {
  if (parent[a] != a) {
    parent[a] = find(parent[a]);
  }
  return parent[a];
}
long long DSU::size1(long long a) { return rank[find(a)]; }
int32_t main() {
  long long n;
  cin >> n;
  DSU dsu0 = DSU(n + 5);
  DSU dsu1 = DSU(n + 5);
  vector<vector<pair<long long, long long>>> adj(n + 1);
  for (long long i = 1; i <= n - 1; i++) {
    long long a, b, c;
    cin >> a >> b >> c;
    adj[a].push_back(make_pair(b, c));
    adj[b].push_back(make_pair(a, c));
    if (c == 0) {
      dsu0.union1(a, b);
    }
    if (c == 1) {
      dsu1.union1(a, b);
    }
  }
  long long ans = 0;
  set<long long> visited0, visited1;
  for (long long i = 1; i <= n; i++) {
    long long temp0 = dsu0.find(i);
    if (visited0.find(temp0) == visited0.end()) {
      long long sz = dsu0.size1(temp0);
      visited0.insert(temp0);
      ans += sz * (sz - 1);
    }
    long long temp1 = dsu1.find(i);
    if (visited1.find(temp1) == visited1.end()) {
      visited1.insert(temp1);
      long long sz = dsu1.size1(temp1);
      ans += sz * (sz - 1);
    }
    ans += (dsu0.size1(temp0) - 1) * (dsu1.size1(temp1) - 1);
  }
  cout << ans;
  return 0;
}
