class Point:

	def __init__(self, x, y):
		self.x = x
		self.y = y
		self.squareDist = 0
read = lambda : map(int, input().split())
(n, x1, y1, x2, y2) = read()
p = []
for _ in range(n):
	(x, y) = read()
	t = Point(x, y)
	t.squareDist = (x - x1) * (x - x1) + (y - y1) * (y - y1)
	p.append(t)
p.sort(key=lambda point: point.squareDist)
r2 = [0] * n
r2[n - 1] = (p[n - 1].x - x2) * (p[n - 1].x - x2) + (p[n - 1].y - y2) * (p[n - 1].y - y2)
for i in range(n - 2, -1, -1):
	r2[i] = max(r2[i + 1], (p[i].x - x2) * (p[i].x - x2) + (p[i].y - y2) * (p[i].y - y2))
ans = min(p[n - 1].squareDist, r2[0])
for i in range(0, n - 1):
	ans = min(ans, p[i].squareDist + r2[i + 1])
print(ans)
