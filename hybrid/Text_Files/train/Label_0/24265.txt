#include <bits/stdc++.h>
using namespace std;
long long n, a[10], dv[10], res = 1061109567;
void backtrack(long long pos, long long p[], long long val[], long long ans) {
  if (pos == n) {
    long long cnt1 = 0;
    for (long long i = 0; i < n; i++)
      if (p[i] == -1) cnt1++;
    if (cnt1 > 1) ans++;
    if (ans < res) {
      res = ans;
    }
    return;
  }
  p[pos] = -1;
  if (dv[pos] > 1)
    backtrack(pos + 1, p, val, ans + 1 + dv[pos]);
  else
    backtrack(pos + 1, p, val, ans + dv[pos]);
  if (pos) {
    for (long long i = 0; i < pos; i++)
      if (val[i] % a[pos] == 0) {
        p[pos] = i;
        val[i] /= a[pos];
        if (dv[pos] > 1)
          backtrack(pos + 1, p, val, ans + 1);
        else
          backtrack(pos + 1, p, val, ans);
        val[i] *= a[pos];
      }
  }
}
long long getdv(long long x) {
  long long ans = 0;
  for (long long i = 2; i < sqrt(x) + 1; i++) {
    while (x % i == 0) {
      x /= i;
      ans++;
    }
  }
  if (x > 1) ans++;
  return ans;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (long long i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n, greater<long long>());
  for (long long i = 0; i < n; i++) dv[i] = getdv(a[i]);
  long long temp[10], temp2[10];
  for (long long i = 0; i < n; i++) temp2[i] = a[i];
  backtrack(0, temp, temp2, 0);
  cout << res;
  return 0;
}
