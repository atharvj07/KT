md = 998244353

def solve(n, aa):
	m1 = [0] * (n + 2)
	m2 = [0] * (n + 2)
	m1[0] = 1
	ans = 0
	for a in aa:
		if a - 1 >= 0:
			ans += m1[a - 1] + m2[a - 1]
			m2[a - 1] += m1[a - 1] + m2[a - 1]
			m2[a - 1] %= md
		ans += m1[a] + m1[a + 1] + m2[a + 1]
		m1[a + 1] += m1[a] + m1[a + 1]
		m2[a + 1] *= 2
		m1[a + 1] %= md
		m2[a + 1] %= md
		ans %= md
	print(ans)
