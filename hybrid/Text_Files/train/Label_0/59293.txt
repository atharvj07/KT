import static java.lang.System.out;
import static java.lang.System.in;
import java.util.Scanner;

public class Displayer {
	
	private static Scanner keyboard = new Scanner(in);
	
	public static void main(String[] args) {
		int total_num, total_dist;
		
		total_num = keyboard.nextInt();
		total_dist = keyboard.nextInt();
		
		for (int i = 1; i <= total_num; i += total_dist + 1) {
			int j, k;
			j = i;
			if (i + total_dist <= total_num) {
				k = i + total_dist;
			}
			else {
				k = total_num;
			}
			while (j <= k) {
				out.print(j + " ");
				if (j < k) {
					out.print(k + " ");
				}
				j++;
				k--;
			}
		}
	}
}
