import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class E {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        sc.nextInt();
        String template = sc.next();
        Set<Character> used = new HashSet<>();
        for(Character c: template.toCharArray()) used.add(c);

        int n = sc.nextInt();
        String[] words = new String[n];
        for (int i = 0; i < n; i++) words[i] = sc.next();

        boolean[] match = new boolean[n];

        int count = 0;
        for (int i = 0; i < n; i++) {
            match[i] = match(words[i], template, used);
            if (match[i]) count += 1;
        }

        int res = 0;
        String letters = "abcdefghijklmnopqrstuvwxyz";
        for(char c : letters.toCharArray()){

            int countLetter = 0;
            for(int i = 0; i < n; i++){
                if(!match[i]) continue;

                for(int j = 0; j < words[i].length(); j++){
                    if(template.charAt(j) == '*' && words[i].charAt(j) == c ) {
                        countLetter+=1;
                        break;
                    }
                }
            }
            if(countLetter == count) res += 1;
        }
        System.out.println(res);
    }

    private static boolean match(String word, String template, Set<Character> used) {
        if (word.length() != template.length()) return false;
        for(int i = 0; i< word.length(); i++){
            if(template.charAt(i)!='*' && template.charAt(i)!= word.charAt(i)) return false;
            if(template.charAt(i)=='*' && used.contains(word.charAt(i))) return false;
        }
        return true;
    }
}
