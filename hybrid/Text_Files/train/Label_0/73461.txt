from functools import reduce
import operator
import itertools
from bisect import bisect_left
t = int(input())

def solve():
	l = int(input())
	n = list(map(int, input().split(' ')))
	if len(set(n)) == 1:
		print('1')
		print(' '.join(['1'] * l))
		return
	hs = False
	rp = 0
	for (i, v) in enumerate(n):
		if v == n[i - 1]:
			hs = True
			rp = i
			n = n[i:] + n[:i]
			break
	if hs:
		print('2')
		ans = ['1', '2'] * (l // 2) + (['1'] if l % 2 else [])
		print(' '.join(ans[-rp:] + ans[:-rp]))
	else:
		print('3' if l % 2 else '2')
		print(' '.join(['1', '2'] * (l // 2) + (['3'] if l % 2 else [])))
for i in range(t):
	solve()
