import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scan=new Scanner(System.in);
	    int n=scan.nextInt();
	    int d=scan.nextInt();
	    Node[] arr=new Node[n];
	    for(int i=0;i<n;i++){
	        int money=scan.nextInt();
	        int friendshipFactor=scan.nextInt();
	        arr[i]=new Node(money,friendshipFactor);
	    }
	    Arrays.sort(arr, (a,b)-> b.money-a.money);
	   // Arrays.stream(arr).forEach(i->System.out.println(i.money+" "+i.friendshipFactor));
	    int low=0;
	    int high=0;
	    long ans=0;
	    long sum=0;
	    while(low<n){
	        while(high<n && Math.abs(arr[high].money-arr[low].money)<d){
	            sum=sum+arr[high].friendshipFactor;
	            high++;
	        }
	        ans=Math.max(ans,sum);
	        sum-=arr[low].friendshipFactor;
	        low++;
	    }
	    System.out.println(ans);
	    
	}
	static class Node{
	    int money;
	    int friendshipFactor;
	    public Node(int money, int friendshipFactor){
	        this.money=money;
	        this.friendshipFactor=friendshipFactor;
	    }
	}
}
