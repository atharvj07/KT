import os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
h = list(map(int, input().split()))
hWithIndex = []
for i in range(n):
	hWithIndex.append((i, h[i]))
hWithIndex.sort(key=lambda x: x[1])
originalToSorted = [0] * n
sortedToOriginal = [0] * n
for i in range(n):
	sortedToOriginal[i] = hWithIndex[i][0]
	originalToSorted[hWithIndex[i][0]] = i
currentIndex = 0
currentMax = 0
blocks = 0
while True:
	if currentIndex >= n:
		break
	blocks += 1
	updateMax = currentIndex
	updateIndex1 = currentIndex
	updateIndex2 = currentIndex
	while True:
		for i in range(updateIndex1, currentMax + 1):
			if updateMax < originalToSorted[i]:
				updateMax = originalToSorted[i]
		if updateMax == currentMax:
			break
		updateIndex1 = currentMax + 1
		currentMax = updateMax
		for i in range(updateIndex2, currentMax + 1):
			if updateMax < sortedToOriginal[i]:
				updateMax = sortedToOriginal[i]
		if updateMax == currentMax:
			break
		updateIndex2 = currentMax + 1
		currentMax = updateMax
	currentIndex = updateMax + 1
print(blocks)
