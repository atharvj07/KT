#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100 + 23;
int N;
int data[MAXN];
inline void swap(int &a, int &b) {
  int t = a;
  a = b;
  b = t;
}
int main() {
  cin >> N;
  for (int i = 0; i < N; ++i) cin >> data[i];
  for (int i = 0; i < N; ++i)
    for (int j = 0; j < N - 1; ++j)
      if (data[j] > data[j + 1]) {
        swap(data[j], data[j + 1]);
        cout << j + 1 << ' ' << j + 2 << endl;
      }
  return 0;
}
