def inside(y, x, H, W):
	return 0 <= y < H and 0 <= x < W

def ceil(a, b):
	return (a + b - 1) // b

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)

def lcm(a, b):
	g = gcd(a, b)
	return a / g * b

def euclidean_distance(y1, x1, y2, x2):
	return sqrt((y1 - y2) ** 2 + (x1 - x2) ** 2)

def manhattan_distance(y1, x1, y2, x2):
	return abs(y1 - y2) + abs(x1 - x2)

def main():
	N = int(input())
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	extra = 0
	d = defaultdict(int)
	for (a, b) in zip(A, B):
		c = gcd(a, b)
		if a == 0:
			if b == 0:
				extra += 1
			continue
		d[-b // c, a // c] += 1
	ans = 0
	if d:
		ans = max(d.values())
	print(ans + extra)
main()
