from sys import stdin, stdout

def rint():
	return map(int, stdin.readline().split())
(n, c, d) = rint()

def mt(T, index, a):
	maxb0 = 0
	maxb1 = 0
	for (i, j) in enumerate(a):
		(p, b) = j
		if maxb0 < b:
			maxb1 = maxb0
			maxb0 = b
			index[i] = i
		elif maxb1 < b:
			maxb1 = b
			index[i] = index[i - 1]
		else:
			index[i] = index[i - 1]
		T[i][0] = maxb0
		T[i][1] = maxb1

def findmax(T, a, c, index):
	maxb = 0
	if len(a) == 0:
		return 0
	for (i, j) in enumerate(a):
		(p, b) = j
		if c - p <= 0:
			break
		i2 = findai(a, c - p)
		if i2 == -1:
			continue
		if i == index[i2]:
			bb = T[i2][1]
		else:
			bb = T[i2][0]
		if bb == 0:
			b = 0
		maxb = max(maxb, bb + b)
	return maxb

def findai(a, p):
	s = 0
	e = len(a)
	if a[0][0] > p:
		return -1
	while e - s > 1:
		i = (s + e) // 2
		if a[i][0] <= p:
			s = i
		else:
			e = i
	return s
a = []
b = []
for i in range(n):
	(bb, pp, tt) = input().split()
	bb = int(bb)
	pp = int(pp)
	if tt == 'C':
		a.append([pp, bb])
	else:
		b.append([pp, bb])
la = len(a)
lb = len(b)
a.sort()
b.sort()
TA = [[0, 0] for i in range(la)]
TB = [[0, 0] for i in range(lb)]
TAi = [-1 for i in range(la)]
TBi = [-1 for i in range(lb)]
mt(TA, TAi, a)
mt(TB, TBi, b)
ci = -1
di = -1
if len(a):
	ci = findai(a, c)
if len(b):
	di = findai(b, d)
if ci == -1 or di == -1:
	cd = 0
else:
	cd = TA[ci][0] + TB[di][0]
ans = max(findmax(TA, a, c, TAi), findmax(TB, b, d, TBi), cd)
print(ans)
