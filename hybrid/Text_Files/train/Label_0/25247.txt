#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e5 + 10;

const double pi = acos(-1.0);

int test;
long long n, k;
string s = "";
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> k;
  if (k == 0) {
    while (n--) {
      cout << "()";
    }
    return 0;
  }
  if (n * (n - 1) / 2 < k) {
    cout << "Impossible";
    return 0;
  }
  long long m = n;
  for (int i = 1; i <= n; i++) {
    long long t = (long long)i * (long long)(i - 1) / 2;
    if (t >= k) {
      m = i - 1;
      break;
    }
  }
  for (int i = 1; i <= n - m - 1; i++) {
    s += "()";
  }
  for (int i = 1; i <= m; i++) {
    s += "(";
    long long t = (long long)m * (long long)(m - 1) / 2;
    if (k - t == i) {
      s += "()";
    }
  }
  for (int i = 1; i <= m; i++) {
    s += ")";
  }
  cout << s;
  return 0;
}
