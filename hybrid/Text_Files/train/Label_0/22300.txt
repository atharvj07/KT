#include <bits/stdc++.h>
int main() {
  int i, j, n, k, m, a[1010], ans = 0;
  scanf("%d%d", &n, &k);
  for (i = 1; i <= 1000; i++) a[i] = 0;
  for (i = 1; i <= n; i++) {
    scanf("%d", &m);
    a[m]++;
  }
  for (i = 1; i <= k; i++) ans = ans + a[i] / 2;
  printf("%d", ans * 2 + ((n + 1) / 2 - ans));
  return 0;
}
