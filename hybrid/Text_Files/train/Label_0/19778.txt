const int MOD = 1e8, N = 1e6 + 4;
int dp[1022], pk;
int solve(string s) {
  for (int i = 0; i < s.size(); i++) {
    dp[i] = 1;
  }
  for (int i = 1; i < s.size(); i++) {
    int maxi = 0;
    for (int j = i - 1; j >= 0; j--) {
      if (s[i] == s[j]) {
        int t = maxi;
        maxi = max(maxi, dp[j]);
        dp[j] = max(dp[j], t + 2);
      } else {
        maxi = max(maxi, dp[j]);
      }
    }
  }
  int ans = 0;
  for (int i = 0; i < s.size(); i++) ans = max(ans, dp[i]);
  return ans;
}
