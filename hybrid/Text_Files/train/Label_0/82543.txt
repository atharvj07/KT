n = int(input())
query = []
for i in range(n):
	query.append(input().split())
mini = 0
maxi = 0
for q in query:
	if q[0] == '+':
		maior = max(int(q[1]), int(q[2]))
		menor = min(int(q[1]), int(q[2]))
		if menor > mini:
			mini = menor
		if maior > maxi:
			maxi = maior
	else:
		maior = max(int(q[1]), int(q[2]))
		menor = min(int(q[1]), int(q[2]))
		if maior >= maxi and menor >= mini:
			print('YES')
		else:
			print('NO')
