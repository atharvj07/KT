(n, h) = map(int, input().split())
sum = (h + h * h) // 2
if sum >= n - h:
	l = 0
	r = h + 1
	while l < r - 1:
		m = (l + r) // 2
		sum = (m + m * m) // 2
		if sum <= n:
			l = m
		else:
			r = m
	sum = (l + l * l) // 2
	ans = l
	if sum < n:
		ans += 1
	print(ans)
else:
	l = h
	r = 100000000000000000000000000
	while l < r - 1:
		m = (l + r) // 2
		sum1 = (m + m * m) // 2
		sum2 = (m - h + 1) * (h + m) // 2
		if sum1 + sum2 <= n:
			l = m
		else:
			r = m
	sum = (l + l * l) // 2 + (l - h + 1) * (h + l) // 2
	ans = l + l - h + 1
	n = n - sum
	if n >= l + 1:
		ans = ans + 1
		n = n - (l + 1)
	if n > 0:
		ans = ans + 1
	print(ans)
