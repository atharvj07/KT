(y, x) = input().split()
x = int(x)
y = int(y)
peta = []
for i in range(0, y):
	peta.append(list(input()))
found = False
foundW = False
foundS = False
foundM = False
for j in range(0, y):
	for i in range(0, x):
		if peta[j][i] == 'S':
			foundS = True
			if i != x - 1:
				if peta[j][i + 1] == 'W':
					foundM = True
			if i != 0:
				if peta[j][i - 1] == 'W':
					foundM = True
			if j != y - 1:
				if peta[j + 1][i] == 'W':
					foundM = True
			if j != 0:
				if peta[j - 1][i] == 'W':
					foundM = True
		elif peta[j][i] == 'W':
			foundW = True
		elif peta[j][i] == '.':
			peta[j][i] = 'D'
			found = True
if foundM:
	print('No')
elif found:
	print('Yes')
	for j in range(0, y):
		print(''.join(peta[j]), end='\n')
elif foundW:
	if not foundS:
		print('Yes')
		for j in range(0, y):
			for i in range(0, x):
				if i == x - 1:
					print(peta[j][i], end='\n')
				else:
					print(peta[j][i], end='')
	else:
		print('No')
else:
	print('Yes')
	for j in range(0, y):
		for j in range(0, y):
			print(''.join(peta[j]), end='\n')
