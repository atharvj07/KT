import sys
n = int(input())
arr = []
while n:
	txt = input()
	L = txt.split(' ')
	L[1] = int(L[1])
	arr.append(L)
	n -= 1
mx = 1000000001
mn = -1000000001
for L in arr:
	if L[0] == '>=':
		if L[2] == 'Y':
			mn = max(mn, L[1])
		else:
			mx = min(mx, L[1] - 1)
	elif L[0] == '<':
		if L[2] == 'Y':
			mx = min(mx, L[1] - 1)
		else:
			mn = max(mn, L[1])
	elif L[0] == '<=':
		if L[2] == 'Y':
			mx = min(mx, L[1])
		else:
			mn = max(mn, L[1] + 1)
	elif L[0] == '>':
		if L[2] == 'Y':
			mn = max(mn, L[1] + 1)
		else:
			mx = min(mx, L[1])
	else:
		pass
if mx < mn:
	print('Impossible')
else:
	print(mn)
