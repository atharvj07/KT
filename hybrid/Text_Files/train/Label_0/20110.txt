int main(void){
	cout<<fixed<<setprecision(20);cin.tie(0);ios::sync_with_stdio(false);
	int n,i,j,k,Q;cin>>n>>Q;
	int na=n;
	vector<int>r;
	for(i=2;i*i<=na;i++){
		if(n%i>0){continue;}
		int ii=1;
		while(na%i==0){
			na/=i;
			ii*=i;
			r.pub(ii);
		}
	}
	if(na>1){r.pub(na);}
	int m=r.size();
	vector<vector<array<int,26>>>kou(m);
	vector<int>d(m);
	string s;cin>>s;
	for(i=0;i<n;i++){s[i]-='a';}
	for(i=0;i<m;i++){
		int L=n/r[i];
		kou[i].res(L);
		for(j=0;j<r[i];j++){
			for(k=0;k<L;k++){
				if(kou[i][k][s[j*L+k]]==0){d[i]++;}
				kou[i][k][s[j*L+k]]++;
			}
		}
	}
	while(Q--){
		int p;char c;cin>>p>>c;
		p--;c-='a';
		for(i=0;i<m;i++){
			int L=n/r[i];
			kou[i][p%L][s[p]]--;
			if(kou[i][p%L][s[p]]==0){d[i]--;}
			if(kou[i][p%L][c]==0){d[i]++;}
			kou[i][p%L][c]++;
		}
		s[p]=c;
		llint ans=n;
		for(i=0;i<m;i++){
			if(d[i]==n/r[i]){ans=gcd(ans,n/r[i]);}
		}
		cout<<ans<<endl;
	}
	return 0;
}
