import java.io.*;
import java.util.*;

public class CF149D extends PrintWriter {
	CF149D() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF149D o = new CF149D(); o.main(); o.flush();
	}

	static final int MD = 1000000007;
	void merge(int[][] aa, int[][] bb, int[][] cc) {
		for (int w = 0; w < 3; w++)
			for (int z = 0; z < 3; z++)
				cc[w][z] = 0;
		for (int x = 0; x < 3; x++)
			for (int y = 0; y < 3; y++) {
				if (x == 1 && y == 1 || x == 2 && y == 2)
					continue;
				for (int w = 0; w < 3; w++) {
					int wx = aa[w][x];
					if (wx == 0)
						continue;
					for (int z = 0; z < 3; z++) {
						int yz = bb[y][z];
						if (yz == 0)
							continue;
						cc[w][z] = (int) ((cc[w][z] + (long) wx * yz) % MD);
					}
				}
			}
	}
	void merge2(int[][] aa, int[][] bb, int[][] cc) {
		for (int w = 0; w < 3; w++)
			for (int z = 0; z < 3; z++)
				cc[w][z] = 0;
		for (int w = 0; w < 3; w++)
			for (int x = 0; x < 3; x++) {
				if (w == 1 && x == 1 || w == 2 && x == 2)
					continue;
				for (int y = 0; y < 3; y++) {
					int xy = aa[x][y];
					if (xy == 0)
						continue;
					for (int z = 0; z < 3; z++) {
						if (y == 1 && z == 1 || y == 2 && z == 2)
							continue;
						int wz = bb[w][z];
						if (wz == 0)
							continue;
						cc[w][z] = (int) ((cc[w][z] + (long) xy * wz) % MD);
					}
				}
			}
	}
	void copy(int[][] aa, int[][] bb) {
		for (int x = 0; x < 3; x++)
			for (int y = 0; y < 3; y++)
				bb[x][y] = aa[x][y];
	}
	void main() {
		byte[] cc = sc.next().getBytes();
		int n = cc.length;
		int[] qq = new int[n]; int cnt = 0;
		int[][][] kk = new int[n][3][3]; int m = 0;
		int[][] tt = new int[3][3];
		for (int i = 0; i < n; i++)
			if (cc[i] == '(')
				qq[cnt++] = -1;
			else {
				int a, b = m++;
				kk[b][0][1] = kk[b][0][2] = kk[b][1][0] = kk[b][2][0] = 1;
				if ((a = qq[cnt - 1]) != -1) {
					merge2(kk[a], kk[b], tt); copy(tt, kk[b]);
					cnt--;
				}
				cnt--;
				if (cnt > 0 && (a = qq[cnt - 1]) != -1) {
					merge(kk[a], kk[b], tt); copy(tt, kk[b]);
					cnt--;
				}
				qq[cnt++] = b;
			}
		tt = kk[qq[0]];
		int ans = 0;
		for (int x = 0; x < 3; x++)
			for (int y = 0; y < 3; y++)
				ans = (ans + tt[x][y]) % MD;
		println(ans);
	}
}
