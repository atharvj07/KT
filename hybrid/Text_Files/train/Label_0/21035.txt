#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  long long mn = 9999999999;
  long long mx2 = 0;
  while (n--) {
    long long x, y;
    cin >> x >> y;
    mn = min(y, mn);
    mx2 = max(mx2, x);
  }
  long long m;
  cin >> m;
  long long mx = 0;
  long long mn2 = 9999999999;
  while (m--) {
    long long x, y;
    cin >> x >> y;
    mx = max(mx, x);
    mn2 = min(mn2, y);
  }
  long long ans1 = 0, ans2 = 0;
  if (mx - mn > 0) ans1 = mx - mn;
  if (mx2 - mn2 > 0) ans2 = mx2 - mn2;
  cout << max(ans1, ans2) << endl;
}
