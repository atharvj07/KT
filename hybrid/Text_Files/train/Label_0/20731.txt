#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(int)(n);i++)
using namespace std;
typedef pair<int,int> P;
map<int,int> mp;
int N,M,a[10001],b[10000];
int seginc(int l,int r){
	int x=upper_bound(a,a+M+1,l)-a-1;
	if(r<=a[x+1]) return x;
	return -1;
}
int query(int l,int r,int k,int L,int j){
	if(j+L==N && mp.count(k)) return mp[k];
	int in=seginc(l,r);
	int ret=0;
	if(in>=0){
		if(j==b[in]) ret=(r-l)-1;
		else if(b[in]>N-L) ret=(r-l)-( 1<<(b[in]-(N-L)-1) );
		else ret=r-l;
	}else{
		int x=query(l,(l+r)/2,k*2,L-1,j)+query((l+r)/2,r,k*2+1,L-1,N-L+1);
		int y=query(l,(l+r)/2,k*2,L-1,N-L+1)+query((l+r)/2,r,k*2+1,L-1,j);
		ret=min(x,y);
	}
	if(j+L==N) mp[k]=ret;
	return ret;
}
int main(){
	cin>>N>>M;
	rep(i,M+1) cin>>a[i];
	rep(i,M) cin>>b[i];
	cout<<query(0,1<<N,1,N,0)<<endl;
}
