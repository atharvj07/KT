#define LLINF ((long long)1e18)
#define INF int(1e9+1e6)
#define MAXN (5006)
ll n, a, b, dp[MAXN][MAXN][3], mod=1e9+7, ans=1, sum[MAXN];
int main(){
	FAST
	cin>>n>>a>>b;
	if(a>b)swap(a,b);
	dp[0][0][2]=1, sum[0]=1;
	FOR(i,1,n){
		FOR(j,0,b-1){
			if(j==0) {
				FOR(k,1,min(a-1,i)) dp[i][j][0] += sum[i-k], dp[i][j][0] %= mod;
			} else{
				dp[i][j][1] += dp[i-1][j-1][1] + (j-a >= 0 && i-a >= 0 ? dp[i-a][j-a][2] : 0), dp[i][j][1] %= mod;
				dp[i][j][2] += dp[i-1][j-1][2] + dp[i-1][j-1][1] + dp[i-1][j-1][0], dp[i][j][2] %= mod;
			}
			sum[i] += dp[i][j][2], sum[i] %= mod;
		}
	}
	FOR(i,1,n) ans*=2,ans%=mod;
	FOR(j,0,b-1)ans-=dp[n][j][0]+dp[n][j][1]+dp[n][j][2],ans+=mod*3,ans%=mod;
	cout<<ans<<'\n';
}
