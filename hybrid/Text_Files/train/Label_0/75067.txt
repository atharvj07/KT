from sys import stdin
import math
a = int(stdin.readline())
for b in range(0, a):
	A = stdin.readline().split()
	n = int(A[0])
	k = int(A[1])
	L = list()
	B = stdin.readline()
	count = 0
	for t in range(0, int(n / k)):
		s = ''
		for q in range(0, k):
			s += B[count]
			count += 1
		L.append(s)
	ans = 0
	for u in range(0, math.ceil(k / 2)):
		Y = list()
		if k % 2 == 1 and u == math.ceil(k / 2) - 1:
			for y in range(0, int(n / k)):
				Y.append(L[y][u])
			W = set(Y)
			record = 0
			for t in W:
				if Y.count(t) > record:
					record = Y.count(t)
			ans += int(n / k) - record
			break
		for y in range(0, int(n / k)):
			Y.append(L[y][u])
			Y.append(L[y][k - u - 1])
		W = set(Y)
		record = 0
		for t in W:
			if Y.count(t) > record:
				record = Y.count(t)
		ans += int(2 * (n / k)) - record
	print(ans)
