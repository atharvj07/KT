

import java.util.Scanner;

public class D {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int h = in.nextInt() - 1;
        int[] D = new int[m];
        int sum = 0;
        for (int i = 0; i < m; i++) {
            D[i] = in.nextInt();
            if (i != h)
                sum += D[i];
        }
        if (sum + D[h] < n)
            System.out.println(-1);
        else {
            double prob = 1;
            D[h]--;
            for (int i = 1; i < n; i++) {
                prob *= (sum * 1.0) / (sum + D[h]);
                sum--;
            }
            System.out.println(1 - prob);
        }
    }
}
