(n, x) = [int(i) for i in input().split()]
nums = [int(i) for i in input().split()]
nums1 = nums[:]
nums = set(nums)
if len(set(nums)) < n:
	print(0)
	exit(0)
for i in range(len(nums)):
	if nums1[i] & x in nums and nums1[i] != nums1[i] & x:
		print(1)
		exit(0)
for i in range(len(nums)):
	nums1[i] = nums1[i] & x
if len(set(nums1)) < n:
	print(2)
	exit(0)
print(-1)
