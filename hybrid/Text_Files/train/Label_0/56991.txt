import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.BitSet;
import java.util.Comparator;
import java.util.StringTokenizer;

public class D {
    static final BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static StringTokenizer st = new StringTokenizer("");
    static String next() {
        while (!st.hasMoreTokens())
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException e) { throw new RuntimeException(e); }
        return st.nextToken();
    }
    static int nextInt() { return Integer.parseInt(next()); }
    static double nextDouble() { return Double.parseDouble(next()); }
    static long nextLong() { return Long.valueOf(next()); }
    static long[] readArr(int n) {
        long[] r = new long[n];
        for (int i = 0; i < n; i++)
            r[i] = nextLong();
        return r;
    }

    public static void main(String[] args) {
        int n = nextInt(),
                m = nextInt();
        long    b = nextInt();
        long x[] = new long[n];
        final long k[] = new long[n];
        int set[] = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = nextInt();
            k[i] = nextInt();
            int mi = nextInt();
            int bs = 0;
            for (int j = 0; j < mi; j++)
                bs |= 1<<(nextInt()-1);
            set[i] = bs;
        }
        Integer P[] = new Integer[n];
        for (int i = 0; i < n; i++) P[i] = i;
        Arrays.sort(P, new Comparator<Integer>() {
            public int compare(Integer o1, Integer o2) {
                return (int) (k[o1] - k[o2]);
            }
        });
        long res = Long.MAX_VALUE;
        long dp[] = new long[1<<m];
        Arrays.fill(dp, Long.MAX_VALUE);
        dp[0] = 0;
        for (int _i = 0; _i < n; _i++) {
            int i = P[_i];
            long xi = x[i];
            for (int j = 0; j < dp.length; j++) {
                if (dp[j]==Long.MAX_VALUE) continue;
                int nj = j | set[i];
                dp[nj] = Math.min(dp[nj], dp[j] + xi);
            }
            long r = dp[(1<<m) - 1];
            if (r != Long.MAX_VALUE)
                res = Math.min(res, r + k[i] *b);
        }
        System.out.println( res == Long.MAX_VALUE ? -1 : res );
    }
}
