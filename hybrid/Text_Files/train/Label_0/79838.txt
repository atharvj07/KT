itr = int(input())
l = []
for i in range(itr):
	m = input()
	l.append(m)
d = []
for i in l:
	while i.count('kh') > 0 or i.count('oo') > 0:
		i = i.replace('kh', 'h')
		i = i.replace('oo', 'u')
	d.append(i)
d = set(d)
l = []
for i in d:
	i = i.replace('h', 'kh')
	i = i.replace('u', 'oo')
	l.append(i)
print(len(set(l)))
