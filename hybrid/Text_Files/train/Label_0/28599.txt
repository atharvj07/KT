#include <bits/stdc++.h>
using namespace std;
int n, m;
vector<vector<int> > v;
bitset<103> visited;
void dfs(int x) {
  visited[x] = 1;
  for (int u : v[x])
    if (!visited[u]) dfs(u);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  if (n != m) {
    cout << "NO";
    return 0;
  }
  v.resize(n);
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    v[x - 1].push_back(y - 1);
    v[y - 1].push_back(x - 1);
  }
  dfs(0);
  for (int i = 0; i < n; i++)
    if (!visited[i]) {
      cout << "NO";
      return 0;
    }
  cout << "FHTAGN!";
  return 0;
}
