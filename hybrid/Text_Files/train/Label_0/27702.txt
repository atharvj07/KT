#include <bits/stdc++.h>
using namespace std;
const long long maxn = 1e6;
long long n, a, b;
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a >> b;
    bool flag = false;
    long long k = cbrt(a * b), x = a / k, y = b / k;
    if (x * x * y == a && y * y * x == b) flag = true;
    if (flag)
      puts("YES");
    else
      puts("NO");
  }
  return 0;
}
