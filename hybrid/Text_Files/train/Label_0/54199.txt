t=readline().split(' '); MAX=1<<20; P=[]; L=[];
W=+t[0]; H=+t[1]|0; Q=+t[2]; out='';
Au=Array(MAX);
Al=Array(MAX);
Ad=Array(MAX);
Ar=Array(MAX);
for(x=1;x<=W;++x){
  t=readline().split(' ');
  for(y=1;y<=H;++y)
    Al[(y<<10)+x]=Au[(x<<10)+y]=Ar[(y<<10)+W+1-x]=Ad[(x<<10)+H+1-y]=t[y-1]|0;
}
function calc(A,W,H){
  var x,y,t,i;
  for(y=1;y<=H;++y){
    i=(y<<10)+1; t=A[i];
    for(++i,x=2;x<=W;++x,++i){
      if(A[i])A[i]=++t;
      else t=0;
    }
  }
}
calc(Au,H,W);
calc(Ad,H,W);
calc(Al,W,H);
calc(Ar,W,H);
function sw(A,W,x,i){
  var t=A[i]=A[i]?0:(A[i-1]|0)+1;
  for(;++x<=W&&A[++i];)A[i]=++t;
}
function max(A,H,X,Y){
  var l=0,s,y,t,i; ++H;
  for(y=1;y<=H;++y){
    t=A[(y<<10)+X]|0;
    for(i=y;l&&L[l-1]>t;--l){
      i=P[l-1];
      s=L[l-1]*(y-i);
      if(y>Y&&Y>=i&&s>M)M=s;
    }
    if(t&&(!l||L[l-1]<t))P[l]=i,L[l++]=t;
  }
}
for(;Q--;){
  t=readline().split(' ');
  X=t[1]|0; Y=t[2]|0; i=(Y<<10)+X;
  if(t[0]==='1'){
    sw(Al,W,X,i);
    sw(Ar,W,W+1-X,(Y<<10)+W+1-X);
    sw(Au,H,Y,(X<<10)+Y);
    sw(Ad,H,H+1-Y,(X<<10)+H+1-Y);
  }
  else if(Al[i]){
    M=0;
    max(Al,H,X,Y);
    max(Ar,H,W+1-X,Y)
    max(Au,W,Y,X);
    max(Ad,W,H+1-Y,X)
    out+=M+'\n';
  }
  else out+='0\n';
}
print(out);
