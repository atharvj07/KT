import sys
input = sys.stdin.readline
from math import ceil
inf = int(10000000000.0)
t = int(input())
for _ in range(t):
	n = int(input())
	lis = list(input().strip())
	p = [0] * n
	for i in range(0, n):
		if lis[(i - 1) % n % n] != lis[i % n] or lis[i % n] != lis[(i + 1) % n]:
			p[i] = 1
		else:
			p[i] = 0
	ans = []
	i = 0
	for j in range(n):
		if p[j] == 1:
			i = j
			break
	r = i
	while i < n + r:
		if p[i % n] == 0:
			v = i
			while i < n + r and p[i % n] == 0:
				i += 1
			vv = i - 1
			ans.append((v, vv))
		i += 1
	val = 0
	for i in ans:
		l = i[1] - i[0] + 1
		if l in (1, 2, 3):
			val += 1
		else:
			val += ceil(l / 3)
	print(val)
