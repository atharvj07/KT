#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
int n,k,a[2010],num[2010],tmp,ans;
int main(){
	scanf("%d%d",&n,&k),tmp=1ll*k*(k-1)/2%mod;
	for(int i=1;i<=n;++i){
		scanf("%d",&a[i]),++num[a[i]];
		for(int j=1;j<n;++j)ans=(ans+k*(a[j]>a[i]))%mod;
	}
	for(int i=2000;i>=1;--i)num[i]+=num[i+1];
	for(int i=1;i<=n;++i)ans=(ans+1ll*tmp*num[a[i]+1])%mod;
	printf("%d",ans);
	return 0;
}
