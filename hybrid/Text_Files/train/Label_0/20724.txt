#include <bits/stdc++.h>
using namespace std;
const int N = 60;
const long long INF = (long long)1e18 + 10;
int n, m, k;
long long a[N], b[N];
int c[N];
long long dp[N][N][N * 2];
pair<int, int> pre[N][N][N * 2];
vector<pair<int, long long> > v;
void go(int i, int j, int p) {
  if (i > 0) {
    pair<int, int> t = pre[i][j][p];
    go(i - 1, t.first, t.second);
  }
  v.push_back(make_pair(j, a[j] + p));
}
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 0; i < m; ++i) {
    scanf("%lld%lld%d", a + i, b + i, c + i);
  }
  fill((long long *)dp, (long long *)(dp + N), -INF);
  for (int j = 0; j < m; ++j) {
    for (int p = 0; p <= b[j] - a[j]; ++p) {
      dp[0][j][p] = p + a[j];
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      for (int p = 0; p <= b[j] - a[j]; ++p) {
        if (dp[i][j][p] == -INF) continue;
        for (int q = 0; q < m; ++q) {
          if (!(c[q] > c[j])) continue;
          long long now = p + a[j];
          long long t1 = now + k;
          long long t2 = now * k;
          if (t1 >= a[q] && t1 <= b[q]) {
            if (dp[i + 1][q][t1 - a[q]] < dp[i][j][p] + t1) {
              dp[i + 1][q][t1 - a[q]] = dp[i][j][p] + t1;
              pre[i + 1][q][t1 - a[q]] = make_pair(j, p);
            }
          }
          if (t2 >= a[q] && t2 <= b[q]) {
            if (dp[i + 1][q][t2 - a[q]] < dp[i][j][p] + t2) {
              dp[i + 1][q][t2 - a[q]] = dp[i][j][p] + t2;
              pre[i + 1][q][t2 - a[q]] = make_pair(j, p);
            }
          }
        }
      }
    }
  }
  long long ans = -INF;
  int who1 = -1, who2 = -1;
  for (int j = 0; j < m; ++j) {
    for (int p = 0; p <= b[j] - a[j]; ++p) {
      if (ans < dp[n - 1][j][p]) {
        ans = dp[n - 1][j][p];
        who1 = j;
        who2 = p;
      }
    }
  }
  if (ans > -INF) {
    puts("YES");
    go(n - 1, who1, who2);
    for (int i = 0; i < v.size(); ++i) {
      printf("%d %lld\n", v[i].first + 1, v[i].second);
    }
  } else {
    puts("NO");
  }
  return 0;
}
