#pragma GCC optimize("Ofast")
#include <bits/stdc++.h>
#define debug cerr
using namespace std;
const int maxn = 53;

char in[maxn * maxn];
int lson[maxn], rson[maxn], id[maxn], cnt;
int tar[maxn], pts[maxn];

stack<int> nums, ope;

inline void gen() {
	for(int i = 1; i <= 26; i++) id[cnt = i] = i;
	int len  = strlen(in + 1);
	for(int i = 1; i <= len; i++) {
		if(in[i] == '[') ope.push(0);
		if(in[i] == ']') {
			// assert(nums.size() >= 2 && ope.top() == 0);
			++cnt, ope.pop();
			lson[cnt] = nums.top(), nums.pop();
			rson[cnt] = nums.top(), nums.pop();
			nums.push(cnt);
		}
		if('a' <= in[i] && in[i] <= 'z') nums.push(in[i] - 'a' + 1);
	}
	// assert(!ope.size() && nums.size() == 1);
}
inline void getTar() {
	static char tp[maxn];
	static int tt;
	memset(tar, -1, sizeof tar);
	while(scanf("%s%d", tp, &tt) == 2) {
		if(tar[*tp - 'a' + 1] != -1 && tar[*tp - 'a' + 1] != tt) exit((puts("No"), 0));
		tar[*tp - 'a' + 1] = tt;
	}
	for(int i = 1; i <= 26; i++) if(!~tar[i]) tar[i] = 0;
}
inline bool test(int sta) {
	memset(pts + 1, 0, (sizeof(int)) * 26);
	for(int i = 27, cur; i <= cnt; i++) {
		cur = sta & 1, sta >>= 1;
		if(cur) id[i] = id[lson[i]];
		else id[i] = id[rson[i]];
		++pts[id[i]];
		if(pts[id[i]] > tar[id[i]]) return 0;
	}
	for(int i = 1; i <= 26; i++) if(pts[i] != tar[i]) return 0;
	return 1;
}

int main() {
	scanf("%s", in + 1), gen(), getTar();
	const int fs = 1 << (cnt - 26);
	for(int i = 0; i < fs; i++) if(test(i)) return puts("Yes"), 0;
	return puts("No"), 0;
}
