

import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
			String s=sc.next();
			char ctr[]=s.toCharArray();
			int letter[]=new int[26];//存储每个字母出现的次数
			for(int i=0;i<ctr.length;i++){
				int num=ctr[i]-'a';
				letter[num]++;
			}
			char goal[]=new char[26];//把出现过的字母按从小到大的顺序存储一次
			int len=0;
			for(int i=0;i<26;i++){
				if(letter[i]!=0)
					goal[len++]=(char) (i+'a');
			}
			String str="";
			
			if(len%2==0){//分奇偶来构造
				for(int i=0;i<len/2-1;i++){
					str+=goal[i];
					str+=goal[len-i-1];
				}
				String tmp=goal[len/2]+str+goal[len/2-1];	
				char ct[]=tmp.toCharArray();
				boolean flag=true;//判断构造完之后是否没有丑数
				for(int i=0;i<ct.length-1;i++){
					int num=ct[i]-ct[i+1];
					if(num==1||num==-1){
						flag=false;
						break;
					}
				}
				if(flag){
					for(int i=0;i<ct.length;i++){
						for(int j=0;j<letter[(int)(ct[i]-'a')];j++){
							System.out.print(ct[i]);
						}
					}
					System.out.println();	
				}else{
					System.out.println("No answer");
				}
			}else{
				for(int i=0;i<len/2;i++){
					str+=goal[i];  //注意不能写成str+=goal[i]+goal[len-i-1],这样会先计算后两个的ascall码和，是整型
					str+=goal[len-i-1];
				}
				String tmp=str;
				tmp+=goal[len/2];
				char ct[]=tmp.toCharArray();
				boolean flag1=true;
				for(int i=0;i<ct.length-1;i++){
					int num=ct[i]-ct[i+1];
					if(num==1||num==-1){
						flag1=false;
						break;
					}
				}
				if(flag1){
					for(int i=0;i<ct.length;i++){
						for(int j=0;j<letter[ct[i]-'a'];j++){
							System.out.print(ct[i]);
						}
					}
					System.out.println();					
				}else{
					tmp=goal[len/2]+str;
					char ct2[]=tmp.toCharArray();
					boolean flag2=true;
					for(int i=0;i<ct.length-1;i++){
						int num=ct2[i]-ct2[i+1];
						if(num==1||num==-1){
							flag2=false;
							break;
						}
					}
					if(flag2){
						for(int i=0;i<ct2.length;i++){
							for(int j=0;j<letter[ct2[i]-'a'];j++){
								System.out.print(ct2[i]);
							}
						}
						System.out.println();	
					}else{
						System.out.println("No answer");
					}
				}
			}
		}
		sc.close();
	}
}
