
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;
import java.util.Map.Entry;
import java.util.StringTokenizer;


public class Main {
	
	public static class Query implements Comparable<Query>{
		final static int block = 987;
		int index;
		int l;
		int r;
		
		@Override
		public int compareTo(Query o) {
			return (this.l / block == o.l / block ? Integer.compare(this.r, o.r):
				Integer.compare(this.l / block, o.l / block));
		}
		
	}
	
	public static long add(int[] count, int x, int[] occur, int k){
		count[x]++;
		return occur[x] + k - count[x] + 1;
	}
	
	public static long remove(int[] count, int x, int[] occur, int k, long[] inv){
		count[x]--;
		return inv[occur[x] + k - count[x]];
	}
	
	public static long pow(long x, long n, long mod){
		if(n == 1){
			return x % mod;
		}
		if(n % 2 == 0){
			return pow((x * x) % mod, n / 2, mod);
		}
		else{
			return (x * pow((x * x) % mod, (n - 1)/ 2, mod)) % mod;
		}
	}
	
	public static void main(String[] args){
		
		long mod = 998244353;
		
		long[] inv = new long[200002];
		
		inv[1] = 1;

		for(int i = 2; i < 200002; i++){
			inv[i] = pow(i, mod - 2, mod);
		}
		
		InputReader sc = new InputReader(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = sc.nextInt();
		int m = sc.nextInt();
		int q = sc.nextInt();
		
		int[] a = new int[n];
		int[] occur = new int[m + 1];
		
		for(int i = 0; i < n; i++){
			a[i] = sc.nextInt();
			occur[a[i]]++;
		}
		
		long[] ans = new long[q];
		
		Map<Integer, ArrayList<Query>> moQueries = new HashMap<>();
		
		for(int i = 0; i < q; i++){
			Query query = new Query();
			
			query.l = sc.nextInt() - 1;
			query.r = sc.nextInt() - 1;
			query.index = i;
			
			int k = sc.nextInt();
			
			if(!moQueries.containsKey(k)){
				moQueries.put(k, new ArrayList<Query>());
			}
			
			moQueries.get(k).add(query);
		}
		
		
		for(Entry<Integer, ArrayList<Query>> entry : moQueries.entrySet()){
			Collections.sort(entry.getValue());
		}

		long[] prod = new long[n];
		
		for(Entry<Integer , ArrayList<Query>> entry : moQueries.entrySet()){
			
			int k = entry.getKey();
			
			int[] count = new int[m + 1];
			
			prod[0] = 1;
			
			long mk = ((long)m * (long)k) % mod;
			
			for(int i = 1; i < n; i++){
				prod[i] = (prod[i - 1] * ((mk) + i)) % mod;
			}
			
			ArrayList<Query> curr = entry.getValue();
			
			int l = 0;
			int r = -1;

			long res = 1l;
			for(Query query : curr){
				while(l > query.l){
					l--;
					res = (res * add(count, a[l], occur, k)) % mod;
				}
				while(r < query.r){
					r++;
					res = (res * add(count, a[r], occur, k)) % mod;
				}
				while(r > query.r){
					res = (res * remove(count, a[r], occur, k, inv)) % mod;
					r--;
				}
				while(l < query.l){
					res = (res * remove(count, a[l], occur, k, inv)) % mod;
					l++;
				}
				ans[query.index] = (res * prod[n - r + l - 1]) % mod;
			}

		}
		for(long l : ans){
			out.println(l);
		}
		out.close();
	}
	
	static class InputReader {
		public BufferedReader reader;
		public StringTokenizer tokenizer;

		public InputReader(InputStream stream) {
			reader = new BufferedReader(new InputStreamReader(stream), 32768);
			tokenizer = null;
		}

		public String next() {
			while (tokenizer == null || !tokenizer.hasMoreTokens()) {
				try {
					tokenizer = new StringTokenizer(reader.readLine());
				} catch (IOException e) {
					throw new RuntimeException(e);
				}
			}
			return tokenizer.nextToken();
		}

		public int nextInt() {
			return Integer.parseInt(next());
		}

		public long nextLong() {
			return Long.parseLong(next());
		}

		public double nextDouble() {
			return Double.parseDouble(next());
		}

		public String nextLine() {
			String str = "";
			try {
				str = reader.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}

			return str;
		}
	}
}
