#include <cstdio>
#include <cstdlib>
#include <iostream>
#include <string>
#include <cmath>
#include <algorithm>
#include <vector>
#include <queue>
#include <stack>

using namespace std;
typedef long long ll;


int main(){
    string in,ans;
    char c='Z';
    
    cin >> in;
    
    
    for(int i=0;i<in.size();i++){
        
        if(in[i]=='A' || in[i]=='Z'){
            if(c!=in[i]){
                ans+=in[i];
                c=in[i];
            }
        }
    }
    
    if(ans.size()<2){
        puts("-1");
        return 0;
    }
    
    for(int i=0;i<ans.size()-1;i++){
        printf("%c",ans[i]);
    }
    
    if(ans[ans.size()-1]=='Z'){
        puts("Z");
    }else{
        printf("\n");
    }
    
    
    return 0;
}
