import itertools
import sys
input = sys.stdin.readline
cnt = 0
log_size = (10 ** 11).bit_length()
dp = [0] * log_size
dp[0] = 1
sum_ = 2
for i in range(log_size - 1):
	dp[i + 1] = sum_
	sum_ += 2 ** (i + 1) * dp[i + 1]
t = int(input())
for _ in range(t):
	(d, MOD) = map(int, input().split())
	ans = 0
	for i in range(log_size):
		if 2 ** i <= d:
			dpp = dp[i] % MOD
			ans += dpp * min(d - 2 ** i + 1, 2 ** i)
			ans %= MOD
	print(ans)
