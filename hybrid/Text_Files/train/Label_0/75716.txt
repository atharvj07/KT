from collections import deque, defaultdict

def ans(n, k):
	graph = defaultdict(list)
	incoming = [0] * (n + 1)
	for _ in range(n - 1):
		(t1, t2) = map(int, input().split())
		graph[t1].append(t2)
		graph[t2].append(t1)
		incoming[t1] += 1
		incoming[t2] += 1
	if n < 3:
		return 0
	pq = deque()
	visited = set()
	for node in graph:
		if incoming[node] == 1:
			pq.append(node)
			visited.add(node)
	while pq and k > 1:
		for _ in range(len(pq)):
			cur = pq.popleft()
			for negbour in graph[cur]:
				if negbour in visited:
					continue
				incoming[negbour] -= 1
				if incoming[negbour] == 1:
					pq.append(negbour)
					visited.add(negbour)
		k -= 1
	return n - len(visited)
for _ in range(int(input())):
	_ = input()
	(n, k) = map(int, input().split())
	print(ans(n, k))
