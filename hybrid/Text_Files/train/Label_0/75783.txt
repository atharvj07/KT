def func(arr, n):
	sa = set(arr)
	c = 0
	while c <= max(sa):
		if c not in sa:
			break
		c += 1
	o = [i for i in range(c)]
	cou = {i: 0 for i in sa}
	for i in arr:
		cou[i] += 1
	for i in range(c):
		cou[i] -= 1
	for i in cou:
		o.extend([i for j in range(cou[i])])
	print(' '.join(map(str, o)))
t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	func(arr, n)
