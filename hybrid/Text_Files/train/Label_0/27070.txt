#include <bits/stdc++.h>
using namespace std;
struct mtx {
  int M[2][2];
  void mul(mtx A) {
    int C[2][2];
    for (int i = (0); i <= (1); i++)
      for (int j = (0); j <= (1); j++) C[i][j] = 0;
    for (int i = (0); i <= (1); i++)
      for (int j = (0); j <= (1); j++)
        for (int k = (0); k <= (1); k++) {
          C[i][j] = (C[i][j] + 1LL * M[i][k] * A.M[k][j]) % 1000000007;
        }
    for (int i = (0); i <= (1); i++)
      for (int j = (0); j <= (1); j++) M[i][j] = C[i][j];
    return;
  }
  void add(mtx A) {
    for (int i = (0); i <= (1); i++)
      for (int j = (0); j <= (1); j++)
        M[i][j] = (M[i][j] + A.M[i][j]) % 1000000007;
    return;
  }
  bool operator!=(const mtx &A) const {
    for (int i = (0); i <= (1); i++)
      for (int j = (0); j <= (1); j++)
        if (M[i][j] != A.M[i][j]) return true;
    return false;
  }
};
int n, q, s[100005];
mtx seg[292929], rot[292929], I, F, O, P;
void rotate(int id, int l, int r) {
  seg[id].mul(rot[id]);
  if (l < r && rot[id] != I) {
    rot[id * 2].mul(rot[id]);
    rot[id * 2 + 1].mul(rot[id]);
  }
  rot[id] = I;
  return;
}
mtx prod(int a) {
  mtx fa = I, fb = F;
  while (a) {
    if (a & 1) fa.mul(fb);
    fb.mul(fb);
    a >>= 1;
  }
  return fa;
}
void init(int id, int l, int r) {
  rot[id] = I;
  if (l == r) {
    seg[id] = prod(s[l]);
    return;
  }
  int m = (l + r) >> 1;
  init(id * 2, l, m);
  init(id * 2 + 1, m + 1, r);
  seg[id] = seg[id * 2];
  seg[id].add(seg[id * 2 + 1]);
  return;
}
mtx query(int id, int l, int r, int L, int R) {
  rotate(id, l, r);
  if (l > R || L > r) return O;
  if (L <= l && r <= R) return seg[id];
  int m = (l + r) >> 1;
  mtx ret = query(id * 2, l, m, L, R);
  ret.add(query(id * 2 + 1, m + 1, r, L, R));
  return ret;
}
void upd(int id, int l, int r, int L, int R) {
  rotate(id, l, r);
  if (l > R || L > r) return;
  if (L <= l && r <= R) {
    rot[id].mul(P);
    rotate(id, l, r);
    return;
  }
  int m = (l + r) >> 1;
  upd(id * 2, l, m, L, R);
  upd(id * 2 + 1, m + 1, r, L, R);
  seg[id] = seg[id * 2];
  seg[id].add(seg[id * 2 + 1]);
  return;
}
void debug(int id, int l, int r) {
  rotate(id, l, r);
  printf("%d (%d %d):", id, l, r);
  for (int i = (0); i <= (1); i++)
    printf("[%d %d] ", seg[id].M[i][0], seg[id].M[i][1]);
  printf("\n");
  if (l < r) {
    int m = (l + r) >> 1;
    debug(id * 2, l, m);
    debug(id * 2 + 1, m + 1, r);
  }
}
int main() {
  for (int i = (0); i <= (1); i++)
    for (int j = (0); j <= (1); j++) O.M[i][j] = 0;
  for (int i = (0); i <= (1); i++)
    for (int j = (0); j <= (1); j++) I.M[i][j] = (i == j);
  for (int i = (0); i <= (1); i++)
    for (int j = (0); j <= (1); j++) F.M[i][j] = 1 - i * j;
  scanf("%d %d", &n, &q);
  for (int i = (1); i <= (n); i++) {
    scanf("%d", &s[i]);
    s[i]--;
  }
  init(1, 1, n);
  for (int i = (1); i <= (q); i++) {
    int tp, l, r, x;
    scanf("%d %d %d", &tp, &l, &r);
    if (tp == 2) {
      mtx v = query(1, 1, n, l, r);
      printf("%d\n", v.M[0][0]);
    } else {
      scanf("%d", &x);
      P = prod(x);
      upd(1, 1, n, l, r);
    }
  }
  return 0;
}
