#include<cstdio>
#include<cstring>
#include<cmath>
#include<iostream>
#include<algorithm>
#include<vector>
#define ll long long
using namespace std;
const int MAXN=5005,D=1e9+7;
int n;
vector<int> ver[MAXN];
ll f[MAXN][MAXN],tmp[MAXN],fac[MAXN];
int siz[MAXN];
void dfs(int u,int fa) {
	f[u][1]=1;siz[u]=1;
	for(int i=0;i<(int)ver[u].size();i++) {
		int v=ver[u][i];
		if(v==fa) continue;
		dfs(v,u);
		memset(tmp,0,sizeof(tmp));
		for(int j=0;j<=siz[u];j++)
			for(int k=0;k<=siz[v];k++)
				tmp[j+k]=(tmp[j+k]+f[u][j]*f[v][k])%D;
		siz[u]+=siz[v];
		memcpy(f[u],tmp,sizeof(tmp));
	}
	for(int i=0;i<=siz[u];i+=2)
		f[u][0]=(f[u][0]-f[u][i]*fac[i/2])%D;
}
int main() {
	fac[0]=1;
	for(int i=1;i<=2500;i++) fac[i]=fac[i-1]*(i*2-1)%D;
	scanf("%d",&n);
	for(int i=1,aa,bb;i<n;i++) {
		scanf("%d%d",&aa,&bb);
		ver[aa].push_back(bb);
		ver[bb].push_back(aa);
	}
	dfs(1,0);
	printf("%lld\n",(D-f[1][0]%D)%D);
	return 0;
}
