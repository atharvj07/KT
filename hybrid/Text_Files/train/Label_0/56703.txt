import java.util.*;

public class PS411A_PasswordCheck {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        //Read in the password
        String pass = sc.next();
        
        //See if the password passes the conditions
        boolean num = false, lower = false, upper = false;
        for(int i = 0; i < pass.length(); i++){
            num |= Character.isDigit(pass.charAt(i));
            lower |= Character.isLowerCase(pass.charAt(i));
            upper |= Character.isUpperCase(pass.charAt(i));
        }
        
        //Output the result
        if(pass.length() >= 5 && num && lower && upper){
            System.out.println("Correct");
        }
        else{
            System.out.println("Too weak");
        }
        
        sc.close();
    }       
}
