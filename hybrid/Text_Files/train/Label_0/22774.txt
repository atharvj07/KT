#include <bits/stdc++.h>
using namespace std;
int main() {
  int r, c, i, j, k;
  cin >> r >> c;
  string ar[r];
  for (i = 0; i < r; i++) cin >> ar[i];
  k = 1;
  for (i = 0; i < r; i++) {
    for (j = 0; j < c; j++) {
      if (ar[i][j] == 'S') {
        if (i + 1 < r) {
          if (ar[i + 1][j] == 'W')
            k = 0;
          else if (ar[i + 1][j] == '.')
            ar[i + 1][j] = 'D';
        }
        if (i - 1 >= 0) {
          if (ar[i - 1][j] == 'W')
            k = 0;
          else if (ar[i - 1][j] == '.')
            ar[i - 1][j] = 'D';
        }
        if (j + 1 < c) {
          if (ar[i][j + 1] == 'W')
            k = 0;
          else if (ar[i][j + 1] == '.')
            ar[i][j + 1] = 'D';
        }
        if (j - 1 >= 0) {
          if (ar[i][j - 1] == 'W')
            k = 0;
          else if (ar[i][j - 1] == '.')
            ar[i][j - 1] = 'D';
        }
      }
    }
  }
  if (k == 1) {
    cout << "Yes\n";
    for (i = 0; i < r; i++) {
      cout << ar[i] << endl;
    }
  } else
    cout << "No";
  return 0;
}
