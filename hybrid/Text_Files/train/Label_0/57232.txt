public class Hell {
    public static void main(String[] args) throws Exception {
        FastReader sc = new FastReader();
        int t=sc.nextInt();
        while (t-->0){
            int n=sc.nextInt();
            int a[]=new int[n];
            for (int i=0;i<n;i++){
                a[i]=sc.nextInt()-1;
            }
            int mx[][]=new int[n][n];
            for (int i=0;i<n;i++){
                for (int j=i;j<n;j++){
                    if (j==i)mx[i][j]=i;
                    else if (a[mx[i][j-1]]>a[j]){
                        mx[i][j]=mx[i][j-1];
                    }else mx[i][j]=j;
                }
            }
//            for (int i=0;i<n;i++){
//                for (int j=i;j<n;j++){
//                    System.out.println(i+" "+j+" "+mx[i][j]);
//                }
//            }

            TreeNode root=new TreeNode();
            solve(0, n-1, root, mx, a);
            int ans[]=new int[n];
            traverse(root, root, ans,a);
            StringBuilder sb=new StringBuilder();
            for (int x:a){
                sb.append((ans[x]-1)+" ");
            }
            System.out.println(sb);

        }
    }

    static void traverse(TreeNode root, TreeNode prev, int ans[], int a[]){
//        System.out.println(root.val+" "+prev.val);
        ans[a[root.val]]=ans[a[prev.val]]+1;
        if (root.l!=null)traverse(root.l, root, ans,a);
        if (root.r!=null)traverse(root.r, root, ans,a);
    }

    static TreeNode solve(int l,int r, TreeNode cur, int mx[][], int a[]){
        if(l>r)return null;
        int max=mx[l][r];
        if (cur==null)cur=new TreeNode();
        cur.val=max;
        cur.l=solve(l, max-1, cur.l, mx, a);
        cur.r=solve(max+1, r, cur.r, mx, a);
        return cur;
    }

    static class TreeNode{
        TreeNode l,r;
        int val;
        public TreeNode(int val){
            this.val=val;
        }
        public TreeNode(){

        }
    }

    public static long maximumArea(int arr[]){
        int n=arr.length;
        long suf[]=new long[n];
        suf[n-1]=arr[n-1];
        for (int i=n-2;i>=0;i--){
            suf[i]=arr[i]*(n-i);
        }
        for(int i=n-2;i>=0;i--){
            suf[i]=Math.max(suf[i], suf[i+1]);
        }
        long ans=0;
        for (int i=0;i<n-1;i++){
            ans=Math.max((long) arr[i]*(i+1)+suf[i+1], ans);
        }
        return ans;
    }

    public static int[] itemsCollected(int t[]){
        int n = t.length;
        int i=0, j=n-1;
        long sumi=0, sumj=0;
        while (i<=j){
            while (i<=j && sumi<=sumj){
                sumi+=t[i];
                i++;
            }
            while (i<=j && sumi>sumj){
                sumj+=t[j];
                j--;
            }
        }
        return new int[]{i, n-j-1};
    }

    public static long minimumCost(int c[], int q[]){
        PriorityQueue<Point> pq=new PriorityQueue<Point>(new Comparator<Point>() {
            @Override
            public int compare(Point o1, Point o2) {
                return o1.c-o2.c;
            }
        });
        int n=c.length;
        long ans=0;
        for (int i=0;i<n;i++){
            pq.add(new Point(c[i], q[i]));
            Point pop=pq.poll();
            ans+=pop.c;
            if (pop.q-1==0)continue;
            pq.add(new Point(pop.c, pop.q-1));
        }
        return ans;
    }
    static class Point{
        int c, q;
        Point(int c, int q){
            this.c=c;
            this.q=q;
        }
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
}
