import sys
inp = sys.stdin.readline

def solve():
	n = int(inp())
	a = [list(map(int, inp().split()))]
	while True:
		c = [0] * (n + 1)
		for i in a[-1]:
			c[i] += 1
		b = [c[i] for i in a[-1]]
		if b == a[-1]:
			break
		a.append(b)
	ans = []
	for i in range(int(inp())):
		(x, k) = map(int, inp().split())
		ans.append(str(a[min(len(a) - 1, k)][x - 1]))
	print('\n'.join(ans))

def main():
	for i in range(int(inp())):
		solve()
main()
