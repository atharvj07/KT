#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000020;
const int maxm = 1000020;

int a, ar[maxn], loc[maxn];
void go(int &a) {
  if (a < 0)
    a = (a - 1) / 2;
  else
    a /= 2;
}
void solve() {
  scanf("%d", &a);
  int t = 0;
  long long sum = 0;
  for (int i = 1; i <= a; i++) {
    scanf("%d", &ar[i]);
  }
  for (int i = 1, j; i <= a; i++) {
    scanf("%d", &j);
    loc[j] = i;
  }
  int maxi = 0;
  for (int i = 1; i <= a; i++) {
    int j = loc[ar[i]];
    if (j < maxi) sum++;
    maxi = max(maxi, (j));
  }
  cout << sum << endl;
}
int main() {
  int n = 1;
  while (n--) solve();
  return 0;
}
