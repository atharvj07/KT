(n, m, k) = map(int, input().split())
A = set(map(int, input().split()))
pre = 1
for i in range(k):
	(x, y) = map(int, input().split())
	if pre not in [x, y]:
		continue
	if pre == x and x in A:
		print(x)
		break
	elif pre == y and y in A:
		print(y)
		break
	pre = y if y != pre else x
else:
	print(pre)
