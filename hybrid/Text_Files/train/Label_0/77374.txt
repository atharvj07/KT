n = int(input())
s = input()
a = list(map(int, s.split()))
a1 = []
a2 = []
b = []

def TanosSorting(a):
	if len(a) > 1:
		counter = 0
		for i in range(len(a) - 1):
			if a[i] > a[i + 1]:
				break
			else:
				counter += 1
		if counter == len(a) - 1:
			b.append(len(a))
		else:
			a1 = []
			a2 = []
			for i in range(len(a) // 2):
				a1.append(a[i])
				a2.append(a[len(a) // 2 + i])
			TanosSorting(a1)
			TanosSorting(a2)
	else:
		b.append(1)
TanosSorting(a)
print(max(b))
