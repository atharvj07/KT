for i in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	if n == 2:
		print(a[0], a[1])
		continue
	max = 1 + 10 ** 9
	for i in range(1, n):
		if a[i] - a[i - 1] < max:
			max = a[i] - a[i - 1]
			x = i
		if max == 0:
			break
	print(' '.join(list(map(str, a[x:]))) + ' ' + ' '.join(list(map(str, a[:x]))))
