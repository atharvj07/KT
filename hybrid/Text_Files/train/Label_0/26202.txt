  return (a % b + b) % b;

long long powmod(long long a, long long b, long long c) {
  if (b == 0LL) return 1LL;
  long long ret = powmod(a, b >> 1, c);
  ret = ret * ret % c;
  if (b & 1LL) ret = ret * a % c;
  return ret;
}

const int N = 111111;

int test;
int n;
long long m, k, ans;
long long a[N], b[N];
vector<int> v[N];
int f[N];
long long cnt[N], cntvis[N];
int q[N], head, tail;
void dfs(int fa, int x) {
  f[x] = fa;
  for (int i = 0, sz = v[x].size(); i < sz; i++) {
    int y = v[x][i];
    if (y == f[x]) continue;
    dfs(x, y);
    cnt[x]++;
  }
}
void push(int x) { q[++tail] = x; }
int over(long long x, long long y) { return x > m / y; }
long long getans() {
  dfs(0, 1);
  head = tail = 0;
  for (int i = 1; i <= n; i++)
    if (!cnt[i]) {
      b[i] = 1;
      push(i);
    }
  while (head < tail) {
    int x = q[++head];
    cntvis[f[x]]++;
    if (f[x] && cnt[f[x]] == cntvis[f[x]]) push(f[x]);
    k = linf;
    for (int i = 0, sz = v[x].size(); i < sz; i++) {
      int y = v[x][i];
      if (y == f[x]) continue;
      mine(k, a[y] / b[x]);
      if (k == 0) return m;
    }
    for (int i = 0, sz = v[x].size(); i < sz; i++) {
      int y = v[x][i];
      if (y == f[x]) continue;
      a[y] = k * b[x];
      a[x] += a[y];
    }
    if (!cnt[x]) cnt[x]++;
    if (over((b[x]), (cnt[x])))
      return m;
    else
      b[x] *= (cnt[x]);
    ;
    if (b[f[x]]) {
      long long g = gcd(b[f[x]], b[x]);
      b[f[x]] /= g;
      if (over((b[f[x]]), (b[x])))
        return m;
      else
        b[f[x]] *= (b[x]);
      ;
    } else
      b[f[x]] = b[x];
  }
  return m - a[1];
}

  scanf("%d", &n);

    v[x].push_back(y);

  }

}
