(n, k) = map(int, input().split())
a = input().split()
l = [''.join(a[i:i + k]) for i in range(0, n, k)]
if len(set(l)) == 1:
	print(0)
else:
	s = 0
	x = [0] * k
	for i in l:
		for j in range(k):
			if i[j] == '1':
				x[j] += 1
	m = ['1' if y > n // k - y else '2' for y in x]
	for p in l:
		for q in range(k):
			if p[q] != m[q]:
				s += 1
	print(s)
