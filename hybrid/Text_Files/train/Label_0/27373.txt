#include <bits/stdc++.h>
using namespace std;
long long int pw(long long int a, long long int b, long long int mod) {
  if (!b) return 1;
  if (b & 1) return a * pw(a * a % mod, b / 2, mod) % mod;
  return pw(a * a % mod, b / 2, mod) % mod;
}
const long long int MAXN = 5e5 + 10;
const long long int INF = 8e18;
const long long int MOD = 1e9 + 7;
int n, q, L[MAXN], R[MAXN], timer, seg[MAXN << 2][2], lazy[MAXN << 2];
vector<int> adj[MAXN];
void DFS(int u, int p = -1) {
  L[u] = timer++;
  for (int v : adj[u]) {
    if (v == p) continue;
    DFS(v, u);
  }
  R[u] = timer;
}
void shift(int id, int l, int r) {
  if (lazy[id] == 0) return;
  seg[id][0] = max(seg[id][0], lazy[id]);
  if (r - l > 1) {
    lazy[2 * id] = max(lazy[2 * id], lazy[id]);
    lazy[2 * id + 1] = max(lazy[2 * id + 1], lazy[id]);
  }
  lazy[id] = 0;
}
void update(int id, int l, int r, int ql, int qr, int val) {
  shift(id, l, r);
  if (qr <= l || r <= ql) return;
  if (ql <= l && r <= qr) {
    lazy[id] = val;
    return shift(id, l, r);
  }
  int mid = (l + r) >> 1;
  update(2 * id, l, mid, ql, qr, val);
  update(2 * id + 1, mid, r, ql, qr, val);
  seg[id][0] = max(seg[2 * id][0], seg[2 * id + 1][0]);
}
int get(int id, int l, int r, int ind) {
  shift(id, l, r);
  if (r - l == 1) return seg[id][0];
  int mid = (l + r) >> 1;
  if (ind < mid)
    return get(2 * id, l, mid, ind);
  else
    return get(2 * id + 1, mid, r, ind);
}
void modify(int id, int l, int r, int ind, int val) {
  if (r - l == 1) {
    seg[id][1] = val;
    return;
  }
  int mid = (l + r) >> 1;
  if (ind < mid)
    modify(2 * id, l, mid, ind, val);
  else
    modify(2 * id + 1, mid, r, ind, val);
  seg[id][1] = max(seg[2 * id][1], seg[2 * id + 1][1]);
}
int get(int id, int l, int r, int ql, int qr) {
  if (qr <= l || r <= ql) return 0;
  if (ql <= l && r <= qr) return seg[id][1];
  int mid = (l + r) >> 1;
  return max(get(2 * id, l, mid, ql, qr), get(2 * id + 1, mid, r, ql, qr));
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  DFS(1);
  timer = 1;
  cin >> q;
  while (q--) {
    int type, v;
    cin >> type >> v;
    if (type == 1) {
      update(1, 0, n, L[v], R[v], timer++);
    }
    if (type == 2) {
      modify(1, 0, n, L[v], timer++);
    }
    if (type == 3) {
      int p = get(1, 0, n, L[v]), q = get(1, 0, n, L[v], R[v]);
      if (p > q)
        cout << 1 << '\n';
      else
        cout << 0 << '\n';
    }
  }
  return 0;
}
