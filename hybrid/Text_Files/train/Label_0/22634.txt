#include <bits/stdc++.h>
using namespace std;
long long n, m;
vector<long long> st1, v, v1, cnt;
void build1(long long id, long long l, long long r) {
  if (l == r) {
    st1[id] = l;
    return;
  }
  long long mid = (l + r) / 2;
  build1(2 * id + 1, l, mid);
  build1(2 * id + 2, mid + 1, r);
  if (v1[st1[2 * id + 1]] >= v1[st1[2 * id + 2]])
    st1[id] = st1[2 * id + 1];
  else
    st1[id] = st1[2 * id + 2];
}
void update1(long long id, long long l, long long r, long long pos,
             long long new_val) {
  if (l == r) {
    st1[id] = l, v1[pos] = new_val;
    return;
  }
  long long mid = (l + r) / 2;
  if (mid >= pos)
    update1(2 * id + 1, l, mid, pos, new_val);
  else
    update1(2 * id + 2, mid + 1, r, pos, new_val);
  if (v1[st1[2 * id + 1]] >= v1[st1[2 * id + 2]])
    st1[id] = st1[2 * id + 1];
  else
    st1[id] = st1[2 * id + 2];
}
long long query1(long long id, long long low, long long high, long long l,
                 long long r, long long x) {
  if (low > high) return -1;
  if (r < low || high < l) return -1;
  if (low <= l && r <= high) {
    if (v1[st1[id]] < x) return -1;
    while (l < r) {
      long long mid = (l + r) / 2;
      if (v1[st1[2 * id + 1]] >= x) {
        r = mid;
        id = 2 * id + 1;
      } else {
        l = mid + 1;
        id = 2 * id + 2;
      }
    }
    return l;
  }
  long long mid = (l + r) / 2;
  long long left = query1(2 * id + 1, low, high, l, mid, x);
  if (left != -1)
    return left;
  else
    return query1(2 * id + 2, low, high, mid + 1, r, x);
}
void solve(int test) {
  cin >> n >> m;
  vector<vector<long long> > tmp(n, vector<long long>(3, 0));
  long long i, j;
  for (i = 0; i < n; i++) {
    cin >> tmp[i][0] >> tmp[i][1];
    tmp[i][2] = i;
  }
  sort(tmp.begin(), tmp.end());
  v.clear(), v1.clear();
  st1.clear(), cnt.clear();
  v.resize(n), v1.resize(n);
  st1.resize(4 * n), cnt.resize(n, 0);
  for (i = 0; i < n; i++) {
    v[i] = tmp[i][0];
    v1[i] = tmp[i][0] + tmp[i][1];
  }
  build1(0, 0, n - 1);
  multiset<pair<long long, long long> > s;
  while (m--) {
    long long x, t;
    cin >> x >> t;
    long long ub = upper_bound(v.begin(), v.end(), x) - v.begin();
    ub--;
    long long q1 = query1(0, 0, ub, 0, n - 1, x);
    if (q1 != -1) {
      cnt[q1]++;
      update1(0, 0, n - 1, q1, v1[q1] + t);
      while (1) {
        auto it = s.lower_bound(make_pair(v[q1], -1));
        if (it == s.end()) break;
        if (it->first > v1[q1])
          break;
        else {
          update1(0, 0, n - 1, q1, v1[q1] + it->second);
          cnt[q1]++;
          s.erase(it);
        }
      }
    } else
      s.insert(make_pair(x, t));
  }
  vector<vector<long long> > ans(n, vector<long long>(2, 0));
  for (i = 0; i < n; i++) {
    long long id = tmp[i][2];
    ans[id][0] = v1[i] - v[i];
    ans[id][1] = cnt[i];
  }
  for (i = 0; i < n; i++) {
    cout << ans[i][1] << " " << ans[i][0] << endl;
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  ;
  int t = 1;
  for (int x = 1; x <= t; x++) {
    solve(x);
  }
  return 0;
}
