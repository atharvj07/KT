#include <bits/stdc++.h>
using namespace std;
long long maxval;
long long dp[300004][4];
vector<long long> a, b;
int n;
long long f(int i, int j) {
  if (i > n) return 0;
  if (dp[i][j] != -1) return dp[i][j];
  long long v1 = a[i], v2 = a[i] + 1, v3 = a[i] + 2, v4 = a[i] + 3;
  long long ans = maxval;
  for (int k = 0; k < 4; k++) {
    if (k + a[i] == a[i - 1] + j) continue;
    ans = min(ans, k * b[i] + f(i + 1, k));
  }
  return dp[i][j] = ans;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  maxval = 1e9;
  maxval = maxval * maxval + 2;
  int q;
  cin >> q;
  while (q--) {
    a.clear();
    b.clear();
    cin >> n;
    a.resize(n + 1);
    b.resize(n + 1);
    for (int i = 1; i <= n; i++) {
      cin >> a[i] >> b[i];
      for (int j = 0; j < 4; j++) dp[i][j] = -1;
    }
    a[0] = -10;
    cout << f(1, 1) << endl;
  }
}
