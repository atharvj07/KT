ipt = input().split()
(n, k) = (int(ipt[0]), int(ipt[1]))

def mini(all, taken):
	if taken + 1 <= all and (not taken == 0):
		return 1
	else:
		return 0

def maxi(all, taken):
	if taken >= all / 2:
		return all - taken
	elif taken < all / 2:
		maximum = 2 * taken
		while maximum > all:
			maximum -= 1
		if maximum > all - taken:
			maximum = all - taken
		return maximum
print(str(mini(n, k)) + ' ' + str(maxi(n, k)))
