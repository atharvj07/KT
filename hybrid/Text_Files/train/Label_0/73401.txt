import math

def solve():
	for _ in range(int(input())):
		(a, k) = map(int, input().split())
		lis = []
		for i in range(k - 1):
			lis = []
			x = a
			while x != 0:
				d = x % 10
				lis.append(d)
				x = x // 10
			s = min(lis)
			l = max(lis)
			if s * l == 0:
				break
			a += s * l
		print(a)
solve()
