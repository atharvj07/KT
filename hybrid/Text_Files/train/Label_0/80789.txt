n = int(input())
orig = n
a = n // 3
if a % 3 == 0:
	a -= 1
n -= a
b = max(n // 3, 1)
if b % 3 == 0:
	b -= 1
n -= b
c = n
if c % 3 == 0:
	if (a + 1) % 3 != 0:
		a += 1
		c -= 1
	elif (b + 1) % 3 != 0:
		b += 1
		c -= 1
	elif (a + 2) % 3 != 0:
		a += 2
		c -= 2
	else:
		b += 2
		c -= 2
assert a + b + c == orig
assert a % 3 != 0 and b % 3 != 0 and (c % 3 != 0)
print('{} {} {}'.format(a, b, c))
