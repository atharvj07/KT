import java.io.*;
import java.util.*;

public class Main {
    private static class Solver {
        void Update(TreeMap<Long, Integer> m1, TreeMap<Long, Integer> m2, long v) {
            for (Map.Entry<Long, Integer> e: m2.entrySet()) {
                long key = e.getKey() ^ v;
                int val = e.getValue();
                if (m1.containsKey(key)) {
                    val += m1.get(key);
                }
                m1.put(key, val);
            }
        }
        private void Solve() {
            int n = inp.nextInt();
            int m = inp.nextInt();
            long k = inp.nextLong();
            long[][] a = new long[n + 1][m + 1];
            for (int i = 1; i <= n; ++i) {
                for (int j = 1; j <= m; ++j) {
                    a[i][j] = inp.nextLong();
                }
            }

            TreeMap<Long, Integer>[][] dp1 = new TreeMap[n + 1][m + 1];
            TreeMap<Long, Integer>[][] dp2 = new TreeMap[n + 1][m + 1];
            for (int i = 0; i <= n; ++i) {
                for (int j = 0; j <= m; ++j) {
                    dp1[i][j] = new TreeMap<>();
                    dp2[i][j] = new TreeMap<>();
                }
            }

            int lim = (n + m + 2) / 2;

            dp1[1][1].put(a[1][1], 1);
            for (int i = 1; i <= n; ++i) {
                for (int j = 1; j <= m; ++j) {
                    if (i < n && i + 1 + j <= lim) {
                        Update(dp1[i + 1][j], dp1[i][j], a[i + 1][j]);
                    }
                    if (j < m && i + j + 1 <= lim) {
                        Update(dp1[i][j + 1], dp1[i][j], a[i][j + 1]);
                    }
                }
            }

            dp2[n][m].put(a[n][m], 1);
            for (int i = n; i >= 1; --i) {
                for (int j = m; j >= 1; --j) {
                    if (i > 1 && i - 1 + j >= lim) {
                        Update(dp2[i - 1][j], dp2[i][j], a[i - 1][j]);
                    }
                    if (j > 1 && i + j - 1 >= lim) {
                        Update(dp2[i][j - 1], dp2[i][j], a[i][j - 1]);
                    }
                }
            }

//            for (int i = 1; i <= n; ++i) {
//                for (int j = 1; j <= m; ++j) {
//                    if (i + j != lim) {
//                        continue;
//                    }
//                    out.println("(" + i + ", " + j + ")");
//                    for (Map.Entry<Long, Integer> e: dp1[i][j].entrySet()) {
//                        out.println(e.getKey() + " - " + e.getValue());
//                    }
//                    out.println();
//                    for (Map.Entry<Long, Integer> e: dp2[i][j].entrySet()) {
//                        out.println(e.getKey() + " - " + e.getValue());
//                    }
//                    out.println();
//                    out.println();
//                }
//            }

            long ans = 0;
            for (int i = 1; i <= n; ++i) {
                for (int j = 1; j <= m; ++j) {
                    if (i + j == lim) {
                        for (Map.Entry<Long, Integer> e: dp1[i][j].entrySet()) {
                            if (!dp2[i][j].containsKey(e.getKey() ^ k ^ a[i][j])) {
                                continue;
                            }
                            ans += (long) e.getValue() * dp2[i][j].get(e.getKey() ^ k ^ a[i][j]);
                        }
                    }
                }
            }
            out.println(ans);
        }
        static InputReader inp;
        static PrintWriter out;
    }

    private static class InputReader {
        private BufferedReader reader;
        private StringTokenizer tokenizer;

        private InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        private String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        private int nextInt() {
            return Integer.parseInt(next());
        }
        private long nextLong() {
            return Long.parseLong(next());
        }
        private int[] nextIntArray(int n) {
            int[] a = new int[n];
            for (int i = 0; i < n; ++i) {
                a[i] = nextInt();
            }
            return a;
        }
    }
    public static void main(String[] args) throws IOException {
        if (!"true".equals(System.getProperty("ONLINE_JUDGE"))) {
            System.setIn(new FileInputStream(new File("input.txt")));
        }
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Solver.inp = new InputReader(inputStream);
        Solver.out = new PrintWriter(outputStream);
        Solver solver = new Solver();
        solver.Solve();
        Solver.out.close();
    }
