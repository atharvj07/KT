#include <bits/stdc++.h>
using namespace std;
const int oo = 0x7fffffff / 3;
const int maxn = 212;
int n, m;
int a, b, c;
int f[maxn][maxn];
int d[maxn][maxn];
int p[maxn][maxn];
void floyd() {
  for (int k = 1; k <= n; ++k) {
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= n; ++j) {
        if (i != j && j != k && i != k) {
          f[i][j] = min(f[i][j], f[i][k] + f[k][j]);
        }
      }
    }
  }
  return;
}
void get_rank() {
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n; ++j) p[i][j] = j;
    for (int j = 1; j <= n; ++j) {
      for (int k = j + 1; k <= n; ++k) {
        if (f[i][p[i][j]] > f[i][p[i][k]]) swap(p[i][j], p[i][k]);
      }
    }
  }
  return;
}
void Len_MST() {
  floyd();
  get_rank();
  int ans = oo;
  for (int u = 1; u <= n; ++u) {
    for (int v = 1; v <= n; ++v) {
      if (d[u][v] != oo) {
        ans = min(ans, min(f[u][p[u][n]] << 1, f[v][p[v][n]] << 1));
        int t = n;
        for (int k = n - 1; k >= 1; --k) {
          if (f[v][p[u][t]] < f[v][p[u][k]]) {
            ans = min(ans, f[u][p[u][k]] + f[v][p[u][t]] + d[u][v]);
            t = k;
          }
        }
      }
    }
  }
  double tmp = (double)(ans) / 2.0;
  printf("%.9lf\n", tmp);
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j) f[i][j] = d[i][j] = oo;
  for (int i = 1; i <= n; ++i) f[i][i] = d[i][i] = 0;
  for (int i = 1; i <= m; ++i) {
    scanf("%d%d%d", &a, &b, &c);
    f[a][b] = f[b][a] = d[a][b] = d[b][a] = c;
  }
  Len_MST();
  return 0;
}
