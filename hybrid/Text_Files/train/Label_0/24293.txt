#include <bits/stdc++.h>

typedef long long ll;
typedef unsigned long long ull;

#define fi first
#define se second
#define repl(i,a,b) for(int i=(int)(a);i<(int)(b);i++)
#define rep(i,n) repl(i,0,n)
#define all(x) (x).begin(),(x).end()
#define dbg(x) cout<<#x"="<<x<<endl
#define mmax(x,y) (x>y?x:y)
#define mmin(x,y) (x<y?x:y)
#define maxch(x,y) x=mmax(x,y)
#define minch(x,y) x=mmin(x,y)
#define uni(x) x.erase(unique(all(x)),x.end())
#define exist(x,y) (find(all(x),y)!=x.end())
#define bcnt __builtin_popcountll

#define INF 1e16
#define mod 1000000007

int N,M,S,T;
bool dp[201][201][10];
const int K=9;

struct state{
  int u,v,j;
};

int merge(int j1,int j2){
  if((j1==0&&j2==1)||(j1==5&&j2==2)||(j1==7&&j2==3))return 0;
  if(j1==4&&j2==0)return 1;
  if(j1==0&&j2==6)return 2;
  if(j1==0&&j2==8)return 3;
  return -1;
}

int main(){
  cin.tie(0);
  ios::sync_with_stdio(false);

  while(1){
    scanf("%d%d%d%d",&N,&M,&S,&T);
    if(N==0)break;
    S--;T--;
    rep(i,N)rep(j,N)rep(k,K)dp[i][j][k]=false;
    queue<state> que;
    rep(i,M){
      int a,b;
      char c;
      scanf("%d %d %c",&a,&b,&c);
      a--;b--;
      if(c=='+'||c=='*'){
        dp[a][b][4]=true;
        que.push((state){a,b,4});
      }else if(c=='('){
        dp[a][b][5]=true;
        que.push((state){a,b,5});
      }else if(c==')'){
        dp[a][b][6]=true;
        que.push((state){a,b,6});
      }else if(c=='['){
        dp[a][b][7]=true;
        que.push((state){a,b,7});
      }else if(c==']'){
        dp[a][b][8]=true;
        que.push((state){a,b,8});
      }else{
        dp[a][b][0]=true;
        que.push((state){a,b,9});
      }
    }

    while(que.size()){
      int u=que.front().u,v=que.front().v,j=que.front().j; que.pop();
      rep(x,N)rep(k,K){
        if(dp[x][u][k]){
          int nj=merge(k,j);
          if(nj==-1||dp[x][v][nj])continue;
          dp[x][v][nj]=true;
          que.push((state){x,v,nj});
        }
        if(dp[v][x][k]){
          int nj=merge(j,k);
          if(nj==-1||dp[u][x][nj])continue;
          dp[u][x][nj]=true;
          que.push((state){u,x,nj});
        }
      }
    }
    if(dp[S][T][0]) cout<<"Yes"<<endl;
    else cout<<"No"<<endl;
  }

  return 0;
}
