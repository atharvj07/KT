s = list(input())
(i, j) = (0, 1)
ans = 0
d = [-1 for i in range(len(s))]
pos = -1
while j < len(s):
	if s[i] == s[j]:
		if pos > -1:
			i = d[pos]
			pos -= 1
		else:
			i = j + 1
			j += 1
		ans += 1
		j += 1
	else:
		pos += 1
		d[pos] = i
		i = j
		j += 1
if ans % 2 == 0:
	print('No')
else:
	print('Yes')
