#include <bits/stdc++.h>
using namespace std;
const int N = 110, oo = 1000000000;
int ii, i, j, k, l, n, m, t, ch, Fg, Test, nm, tt;
vector<int> A, B;
map<vector<int>, int> Map;
void R(int &x) {
  x = 0;
  ch = getchar();
  while (ch < '0' || '9' < ch) ch = getchar();
  while ('0' <= ch && ch <= '9') x = x * 10 + ch - '0', ch = getchar();
}
struct edge {
  int s, n;
};
struct forest {
  int i, j, k, m, En, x, y, rt1, rt2, mi1, mi2, size;
  int h[N], f[N], z[N], sz[N];
  edge E[N << 1];
  vector<int> A;
  void clear() {
    En = 0;
    memset(h, 0, sizeof h);
  }
  int getf(int x) {
    if (f[x] == x) return x;
    return f[x] = getf(f[x]);
  }
  void E_add(int x, int y) {
    E[++En].s = y;
    E[En].n = h[x];
    h[x] = En;
    E[++En].s = x;
    E[En].n = h[y];
    h[y] = En;
  }
  void read() {
    clear();
    for (i = 1; i <= n; i++) f[i] = i;
    R(m);
    for (i = 1; i <= m; i++) {
      R(x);
      R(y);
      if (getf(x) != getf(y))
        f[f[x]] = f[y];
      else
        Fg = 1;
      E_add(x, y);
    }
    k = 0;
    for (i = 1; i <= n; i++)
      if (h[i]) k++;
    if (k >= n) Fg = 1;
  }
  void write() {
    for (i = 1; i <= n; i++)
      for (k = h[i]; k; k = E[k].n)
        if (i < E[k].s) printf("%d %d\n", i, E[k].s);
  }
  void get_size(int x, int F) {
    sz[x] = 1;
    for (int k = h[x]; k; k = E[k].n)
      if (E[k].s != F) {
        get_size(E[k].s, x);
        sz[x] += sz[E[k].s];
      }
  }
  void get_root(int x, int F) {
    int ma = size - sz[x];
    for (int k = h[x]; k; k = E[k].n)
      if (E[k].s != F) {
        get_root(E[k].s, x);
        if (sz[E[k].s] > ma) ma = sz[E[k].s];
      }
    if (ma <= mi1) {
      mi2 = mi1;
      rt2 = rt1;
      mi1 = ma;
      rt1 = x;
    } else {
      if (ma < mi2) {
        mi2 = ma;
        rt2 = x;
      }
    }
  }
  int get_hash(int x) {
    z[x] = 1;
    vector<int> A;
    for (int k = h[x]; k; k = E[k].n)
      if (!z[E[k].s]) A.push_back(get_hash(E[k].s));
    sort(A.begin(), A.end());
    if (!Map[A]) Map[A] = ++nm;
    return Map[A];
  }
  int Js() {
    A.clear();
    memset(z, 0, sizeof z);
    for (i = 1; i <= n; i++)
      if (h[i] && !z[i]) {
        mi1 = mi2 = oo;
        get_size(i, 0);
        size = sz[i];
        get_root(i, 0);
        if (mi1 == mi2) {
          vector<int> tmp;
          tmp.push_back(-1);
          z[rt2] = 1;
          tmp.push_back(get_hash(rt1));
          z[rt2] = 0;
          tmp.push_back(get_hash(rt2));
          sort(tmp.begin(), tmp.end());
          if (!Map[tmp]) Map[tmp] = ++nm;
          A.push_back(Map[tmp]);
        } else
          A.push_back(get_hash(rt1));
      }
    sort(A.begin(), A.end());
    if (!Map[A]) Map[A] = ++nm;
    return Map[A];
  }
} F[N], G, tree;
int main() {
  R(Test);
  for (ii = 1; ii <= Test; ii++) {
    Fg = t = nm = 0;
    Map.clear();
    A.clear();
    R(n);
    R(m);
    for (i = 1; i <= n; i++) {
      F[i].read();
      if (!t && F[i].m == n - 2) t = i;
    }
    if (Fg || !t) {
      puts("NO");
      continue;
    }
    for (i = 1; i <= n; i++) A.push_back(F[i].Js());
    sort(A.begin(), A.end());
    for (tt = 1; tt <= n; tt++)
      if (!F[t].h[tt]) break;
    for (i = 1; i <= n; i++)
      if (i != tt) {
        B.clear();
        tree = F[t];
        tree.E_add(i, tt);
        for (j = 1; j <= n; j++) {
          G.clear();
          for (l = 1; l <= n; l++)
            if (l != j)
              for (k = tree.h[l]; k; k = tree.E[k].n)
                if (tree.E[k].s != j && l < tree.E[k].s)
                  G.E_add(l, tree.E[k].s);
          B.push_back(G.Js());
        }
        sort(B.begin(), B.end());
        if (A == B) {
          puts("YES");
          tree.write();
          break;
        }
      }
    if (i > n) puts("NO");
  }
}
