def pn(s):
	print(s, end='')

def main():
	(n, p, k) = map(int, input().split())
	if p - k < 1:
		r = range(1, p)
	else:
		r = range(p - k, p)
	if p - k - 1 >= 1:
		pn('<< ')
	for i in r:
		pn(str(i) + ' ')
	pn('(' + str(p) + ')')
	if p + k > n:
		r = range(p + 1, n + 1)
	else:
		r = range(p + 1, p + k + 1)
	for i in r:
		pn(' ' + str(i))
	if p + k + 1 <= n:
		pn(' >>')
	pn('\n')
main()
