input = sys.stdin.readline
read = lambda : list(map(int, input().strip().split()))
MOD = 10 ** 9 + 7

def factors(n):
	arr = []
	for i in range(1, int(sqrt(n) + 1)):
		if n % i == 0:
			if not i ** 2 == n:
				arr.extend([i, n // i])
			else:
				arr.append(i)
	return arr

def main():
	n = int(input())
	facto = factors(n)
	arr = []
	for i in facto:
		m = n // i
		arr.append(m * (n + 2 - i) // 2)
	arr.sort()
	print(*arr)
main()
