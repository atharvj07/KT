from collections import defaultdict
import sys
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
t = int(input())
for i in range(t):
	(n, k) = map(int, input().split())
	a = [int(i) for i in input().split() if i != '\n']
	b = [int(i) for i in input().split() if i != '\n']
	ind = [-1] * n
	ans = 1
	for i in range(len(b)):
		ind[b[i] - 1] = i
	for i in range(n):
		tmp = 0
		if ind[a[i] - 1] == -1:
			continue
		if i > 0 and ind[a[i - 1] - 1] < ind[a[i] - 1]:
			tmp += 1
		if i < n - 1 and ind[a[i + 1] - 1] < ind[a[i] - 1]:
			tmp += 1
		ans *= tmp
		ans %= 998244353
	print(ans)
