#include <bits/stdc++.h>
using namespace std;
int dir[3][20] = {
    {-1, 4, 5, 6, 8, 9, 10, 11, -1, 13, 14, 15, 16, -1, 17, 18, 19, -1, -1, -1},
    {-1, 2, 3, -1, 5, 6, 7, -1, 9, 10, 11, 12, -1, 14, 15, 16, -1, 18, 19, -1},
    {-1, 5,  6,  7,  9,  10, 11, 12, 13, 14,
     15, 16, -1, 17, 18, 19, -1, -1, -1, -1}};
int dp[1 << 20];
int solve(int mask) {
  if (!mask) return 0;
  if (dp[mask] != -1) return dp[mask];
  int res = 0;
  for (int d = 0; d < 3; d++) {
    for (int i = 1; i < 20; i++) {
      int m = mask;
      if ((mask & (1LL << i))) {
        (m &= ~(1LL << i));
        int z = i;
        res |= !solve(m);
        while (dir[d][z] != -1 && (m & (1LL << dir[d][z]))) {
          (m &= ~(1LL << dir[d][z]));
          res |= !solve(m);
          z = dir[d][z];
        }
      }
    }
  }
  return dp[mask] = res;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  memset(dp, -1, sizeof dp);
  int mask = 0;
  for (int i = 1; i < 20; i++) {
    char c;
    cin >> c;
    if (c == 'O') (mask |= (1LL << i));
  }
  if (solve(mask))
    cout << "Karlsson\n";
  else
    cout << "Lillebror\n";
  return 0;
}
