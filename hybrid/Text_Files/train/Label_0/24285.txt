#include <bits/stdc++.h>
using namespace std;
map<int, int> mt;
bool zai[100005];
bool u[100005];
int dd, daiding;
int pp;
int z;
int main() {
  int n, m;
  cin >> n >> m;
  int p = n;
  int w = 0;
  for (int i = 0; i < m; i++) {
    char ct;
    int q;
    cin >> ct >> q;
    if (ct == '+') {
      if (zai[q] == false) {
        zai[q] = true;
        z++;
      }
      if (mt[q] == 100 && pp == q) {
        mt[q] = 0;
        u[q] = true;
        daiding = pp;
        pp = 0;
        continue;
      }
      if (q != pp && pp != 0) {
        mt[pp] = 1;
        pp = 0;
        u[q] = true;
      }
      if (i > 0) {
        mt[q] = 1;
        u[q] = true;
        continue;
      } else {
        u[q] = true;
        dd = q;
        continue;
      }
    } else if (ct == '-') {
      if (zai[q] == true) {
        zai[q] = false;
        z--;
      }
      if (mt[q] == 0) {
        if (dd != 0 && u[q] == false) {
          mt[dd] = 1;
          dd = 0;
        }
        if (daiding != 0 && u[q] == false) {
          mt[daiding] = 1;
          daiding = 0;
        }
        u[q] = true;
        if (z == 0 && mt[q] != 1) {
          mt[q] = 100;
          pp = q;
          if (i == m - 1) mt[q] = 0;
          continue;
        }
        if (z > 0) {
          mt[q] = 1;
        }
      } else
        mt[q] = 1;
    }
  }
  for (int i = 1; i <= p; i++) {
    if (mt[i] == 0) {
      w++;
    }
  }
  cout << w << endl;
  if (w != 0) {
    for (int i = 1; i <= p; i++) {
      if (mt[i] == 0) {
        cout << i << " ";
      }
    }
  }
  return 0;
}
