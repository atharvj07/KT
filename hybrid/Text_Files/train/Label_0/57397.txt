import static java.util.Arrays.*;
import static java.lang.Math.*;
import static java.math.BigInteger.*;

public class Main implements Runnable {
    
    void solve() throws IOException {
    	int[][] a = new int[2][10];
    	fill(a[1], 1);
    	int n = nextInt();
    	for (int i = 0; i < n; i++) {
    		char c = next().charAt(0);
    		int v = nextInt();
    		for (int j = 0; j < 2; j++)
    			for (int k = 0; k < 10; k++) 
    				if (c == '|') a[j][k] |=  v >> k & 1;
    				else if (c == '&') a[j][k] &= v >> k & 1;
    				else if (c == '^') a[j][k] ^= v >> k & 1;
    	}
    	debug(a);
    	int and = 0, or = 0, xor = 0;
    	for (int i = 0; i < 10; i++) {
    		outer: for (int x = 0; x < 2; x++)
    			for (int y = 0; y < 2; y++)
    				for (int z = 0; z < 2; z++) {
    					boolean can = true;
    					for (int j = 0; j < 2; j++)
    						if ((((j & x) | y) ^ z) != a[j][i]) can = false;
    					if (can) {
    						and |= x << i;
    						or |= y << i;
    						xor |= z << i;
    						break outer;
    					}
    				}
    	}
    	out.println(3);
    	out.println("& " + and);
    	out.println("| " + or);
    	out.println("^ " + xor);
    }
    
    String next() throws IOException {
        while(st == null || !st.hasMoreTokens()) st = new StringTokenizer(input.readLine());
        return st.nextToken();
    }
    
    int nextInt() throws IOException {
        return Integer.parseInt(next());
    }
    
    long nextLong() throws IOException {
        return Long.parseLong(next());
    }
    
    double nextDouble() throws IOException {
        return Double.parseDouble(next());
    }
    
    void debug(Object... o) {
        if (TEST) System.out.println(deepToString(o));
    }
    
    void gcj(Object o) {
        String s = String.valueOf(o);
        out.println("Case #" + test + ": " + s);
        System.out.println("Case #" + test + ": " + s);
    }
    
    BufferedReader input;
    PrintWriter out;
    StringTokenizer st;
    int test;
    boolean TEST = System.getProperty("ONLINE_JUDGE") == null;

    void init() throws IOException {
        if (TEST) input = new BufferedReader(new FileReader("input.in")); 
        else input = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(new BufferedOutputStream(System.out));
    }
    
    public static void main(String[] args) throws IOException {
        new Thread(null, new Main(), "", 1 << 20).start();
    }
    
    public void run() {
        try {
            init();
            if (TEST) {
                int runs = nextInt();
                for(int i = 0; i < runs; i++) solve();
            } else solve();
            out.close();        
        } catch (Exception e) {
            e.printStackTrace();
            System.exit(1);
        }
    }
}
