#include <bits/stdc++.h>
using namespace std;

#define int long long
const int INF = 1e9;

signed main() {
  int N;
  cin >> N;

  vector<int> A(N);
  for ( int i = 0; i < N; i++ ) {
    cin >> A[i];    
  }

  vector<int> a = A;
  int ans = INF;
  int cnt = 0;
  bool now = 0;
  for ( int i = 0; i < N-1; i++ ) {    
    if ( now ) { // 増加
      if ( a[i] > a[i+1] ) {
	if ( i < N-2 && a[i] < a[i+2] ) {
	  swap(a[i+1], a[i+2]);	  
	} else {
	  swap(a[i], a[i+1]);	 
	}
	cnt++;	
      }
    } else { // 減少
      if ( a[i] < a[i+1] ) {
	if ( i < N-2 && a[i] > a[i+2] ) {
	  swap(a[i+1], a[i+2]);	  
	} else {
	  swap(a[i], a[i+1]);	 
	}
	cnt++;	
      }
    }
    now = !now;    
  }
  
  ans = cnt;

  a = A;
  cnt = 0;
  now = 1;
  for ( int i = 0; i < N-1; i++ ) {
    if ( now ) { // 増加
      if ( a[i] > a[i+1] ) {
	if ( i < N-2 && a[i] < a[i+2] ) {
	  swap(a[i+1], a[i+2]);	  
	} else {
	  swap(a[i], a[i+1]);	 
	}
	cnt++;	
      }
    } else { // 減少
      if ( a[i] < a[i+1] ) {
	if ( i < N-2 && a[i] > a[i+2] ) {
	  swap(a[i+1], a[i+2]);	  
	} else {
	  swap(a[i], a[i+1]);	 
	}
	cnt++;	
      }
    }
    now = !now;    
  }

  cout << min(ans, cnt) << endl;
  
  return 0;
}
