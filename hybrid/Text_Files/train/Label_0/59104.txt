import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.next();
        int n = s.length();

        int[][] max_palin = new int[n + 1][n + 1];
        int[] count = new int[n + 1];

        for (int sub_len = 1; sub_len <= n; sub_len++) {
            for (int left = 0; left <= n - sub_len; left++) {
                int right = left + sub_len - 1;

                if (sub_len == 1) {
                    max_palin[left][left] = 1;
                }
                else if (sub_len == 2) {
                    if (s.charAt(left) == s.charAt(right)) {
                        max_palin[left][right] = 2;
                    }
                    else {
                        max_palin[left][right] = 0;
                    }
                }
                else {
                    if (s.charAt(left) == s.charAt(right) && max_palin[left + 1][right - 1] > 0) { // s[left..right] is a palindrome
                        int mid = (left + right) / 2;

                        if (sub_len % 2 == 0) {
                            max_palin[left][right] = max_palin[left][mid] + 1;
                        }
                        else {
                            max_palin[left][right] = max_palin[left][mid - 1] + 1;
                        }
                    }
                }

                count[max_palin[left][right]]++;
            }
        }

        for (int i = n - 1; i >= 1; i--) {
            count[i] += count[i + 1];
        }

        for (int i = 1; i <= n; i++) {
            System.out.print(count[i]);
            System.out.print(' ');
        }

        System.out.print('\n');
    }
}
