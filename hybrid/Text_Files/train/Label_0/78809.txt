MD = 10 ** 9 + 7

def pow(x, y):
	if y == 0:
		return 1
	m = pow(x, y // 2)
	ans = m * m % MD
	return ans * x % MD if y & 1 else ans
n = int(input())
bit = [0] * 31
for num in map(int, input().split()):
	bit[bin(num)[::-1].index('1')] += 1
ans = pow(2, n) - 1
for i in range(1, 31):
	if bit[i]:
		ans = (ans - pow(2, sum(bit[i:]) - 1)) % MD
print(ans)
