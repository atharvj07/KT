n = int(input())
vector = [int(i) for i in input().split()]
usado = []
nusado = []
pos = -1
cn = 0
cz = 0
resp = []
i = 0
while i < n:
	v = vector[i]
	if v < 0:
		cn += 1
		if pos == -1 or vector[pos] < v:
			pos = i
		nusado.append(i)
	elif v == 0:
		cz += 1
		usado.append(i)
	else:
		nusado.append(i)
	i += 1
if cz == n or (cz == n - 1 and cn == 1):
	i = 1
	while i < n:
		resp.append('%d %d %d' % (1, i, i + 1))
		i += 1
else:
	cross = -1
	if cn % 2 == 1:
		cross = pos
		usado.append(pos)
	ultimo = -1
	i = 0
	for i in usado:
		if ultimo != -1:
			resp.append('%d %d %d' % (1, ultimo + 1, i + 1))
		ultimo = i
		i += 1
	if ultimo != -1:
		resp.append('%d %d' % (2, ultimo + 1))
	i = 0
	ultimo = -1
	for i in nusado:
		if i != cross:
			if ultimo != -1:
				resp.append('%d %d %d' % (1, ultimo + 1, i + 1))
			ultimo = i
		i += 1
print('\n'.join(resp))
