#include <bits/stdc++.h>
using namespace std;
const long long mod = 998244353;
long long f[5010][5010];
long long jie[5010];
int main() {
  int a, b, c;
  scanf("%d %d %d", &a, &b, &c);
  long long ans1 = 0, ans2 = 0, ans3 = 0;
  jie[0] = 1;
  f[0][0] = 1;
  for (int i = 1; i <= 5000; i++) {
    jie[i] = jie[i - 1] * i % mod;
    f[i][0] = 1;
    for (int j = 1; j <= i; j++)
      f[i][j] = (f[i - 1][j - 1] + f[i - 1][j]) % mod;
  }
  for (int i = 0; i <= max(a, b); i++)
    ans1 =
        (ans1 + (f[min(a, b)][i] * f[max(a, b)][i]) % mod * jie[i] % mod) % mod;
  for (int i = 0; i <= max(a, c); i++)
    ans2 =
        (ans2 + (f[min(a, c)][i] * f[max(a, c)][i]) % mod * jie[i] % mod) % mod;
  for (int i = 0; i <= max(c, b); i++)
    ans3 =
        (ans3 + (f[min(c, b)][i] * f[max(c, b)][i]) % mod * jie[i] % mod) % mod;
  printf("%lld\n", ans1 * ans2 % mod * ans3 % mod);
  return 0;
}
