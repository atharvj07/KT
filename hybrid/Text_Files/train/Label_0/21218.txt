#include <bits/stdc++.h>
using namespace std;
namespace Sol {
int fd(int r, int g, int b) {
  int ret = 0;
  ret += r / 3;
  ret += g / 3;
  ret += b / 3;
  return ret;
}
void sol() {
  int r, g, b;
  scanf("%d", &r);
  scanf("%d", &g);
  scanf("%d", &b);
  int best = 0;
  int mn = min(min(r, g), b);
  best = mn + fd(r - mn, g - mn, b - mn);
  if (mn - 1 >= 0) {
    best = max(best, mn - 1 + fd(r - mn + 1, g - mn + 1, b - mn + 1));
  }
  if (mn - 2 >= 0) {
    best = max(best, mn - 2 + fd(r - mn + 2, g - mn + 2, b - mn + 2));
  }
  if (mn - 3 >= 0) {
    best = max(best, mn - 3 + fd(r - mn + 3, g - mn + 3, b - mn + 3));
  }
  printf("%d\n", best);
}
}  // namespace Sol
int main() {
  Sol::sol();
  return 0;
}
