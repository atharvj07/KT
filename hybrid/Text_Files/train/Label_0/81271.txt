n = int(input())
arr = []
for (i, x) in enumerate(input().split()):
	arr.append((i + 1, int(x)))
a = sorted(arr[1:], key=lambda x: x[1], reverse=True)
res = []
j = arr[0][1]
if j <= 0:
	print(-1)
	exit()
received = set()
for i in range(j):
	if i >= len(a):
		break
	res.append((1, a[i][0]))
	received.add(a[i])
while len(received) > 0 and j < n:
	source = received.pop()
	if source[1] > 0:
		for i in range(j, j + source[1]):
			if i >= n - 1:
				break
			received.add(a[i])
			res.append((source[0], a[i][0]))
		j += source[1]
if len(res) < n - 1:
	print(-1)
else:
	print(len(res))
	for (k, v) in res:
		print(k, v)
