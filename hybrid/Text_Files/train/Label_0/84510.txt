for _ in range(int(input())):
	n = int(input())
	arr = input()
	(x, y) = (0, 0)
	d = {}
	d[x, y] = -1
	flen = int(10000000000.0)
	st = -1
	end = -1
	for i in range(n):
		if arr[i] == 'L':
			x -= 1
		elif arr[i] == 'R':
			x += 1
		elif arr[i] == 'U':
			y += 1
		else:
			y -= 1
		if (x, y) in d:
			l = i - d[tuple([x, y])] + 1
			if l < flen:
				flen = l
				st = d[x, y]
				end = i
		d[x, y] = i
	if flen == int(10000000000.0):
		print(-1)
	else:
		print(st + 2, end + 1)
