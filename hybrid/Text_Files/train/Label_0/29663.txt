#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int s = 0;
  char ch = getchar();
  while (!isdigit(ch)) ch = getchar();
  while (isdigit(ch)) s = s * 10 + ch - '0', ch = getchar();
  return s;
}
const int N = 1e6 + 5;
int n;
struct Tree {
  vector<int> G[N];
  int dep[N], low[N], fa[N][20], dfn[N], dfn_t;
  inline void adde(int u, int v) {
    G[u].push_back(v);
    G[v].push_back(u);
  }
  void dfs(int u, int f) {
    fa[u][0] = f, dep[u] = dep[f] + 1;
    dfn[u] = ++dfn_t;
    for (int i = 1; i < 20; i++) fa[u][i] = fa[fa[u][i - 1]][i - 1];
    for (int v : G[u]) {
      if (v == f) continue;
      dfs(v, u);
    }
    low[u] = dfn_t;
  }
  inline int LCA(int u, int v) {
    if (dep[u] < dep[v]) swap(u, v);
    for (int i = 19; i >= 0; i--)
      if ((dep[u] - dep[v]) >> i & 1) u = fa[u][i];
    if (u == v) return u;
    for (int i = 19; i >= 0; i--)
      if (fa[u][i] != fa[v][i]) u = fa[u][i], v = fa[v][i];
    return fa[u][0];
  }
  inline int dis(int x, int y) { return dep[x] + dep[y] - 2 * dep[LCA(x, y)]; }
} T;
inline bool cmp(int x, int y) { return T.dfn[x] < T.dfn[y]; }
int m, k, stk[N], top;
int fa[N], type[N], sp[N], b[N], q[N], f[N];
vector<int> ver;
inline int calc(int u, int v) {
  return (T.dis(u, v) + b[type[u]] - 1) / b[type[u]];
}
inline int merge(int x, int y, int u) {
  if (!x || !y) return x + y;
  int dx = calc(x, u), dy = calc(y, u);
  if (dx == dy)
    return type[x] < type[y] ? x : y;
  else
    return dx < dy ? x : y;
}
int main() {
  n = read();
  for (int i = 1; i < n; i++) {
    int u = read(), v = read();
    T.adde(u, v);
  }
  int lol = read();
  T.dfs(1, 0);
  while (lol--) {
    k = read(), m = read();
    ver.clear();
    ver.push_back(1);
    for (int i = 1; i <= k; i++) {
      int u = read();
      type[u] = i;
      b[i] = read();
      f[u] = u;
      ver.push_back(u);
    }
    for (int i = 1; i <= m; i++) {
      q[i] = read();
      ver.push_back(q[i]);
    }
    sort(ver.begin(), ver.end(), cmp);
    for (int i = ver.size() - 1; i >= 1; i--)
      ver.push_back(T.LCA(ver[i], ver[i - 1]));
    sort(ver.begin(), ver.end(), cmp);
    ver.erase(unique(ver.begin(), ver.end()), ver.end());
    for (int u : ver) {
      while (top && T.low[stk[top]] < T.dfn[u]) top--;
      if (top) fa[u] = stk[top];
      stk[++top] = u;
    }
    for (int i = ver.size() - 1; i >= 1; i--) {
      int u = ver[i];
      f[fa[u]] = merge(f[fa[u]], f[u], fa[u]);
    }
    for (int i = 1; i < ver.size(); i++) {
      int u = ver[i];
      f[u] = merge(f[fa[u]], f[u], u);
    }
    for (int i = 1; i <= m; i++) printf("%d ", type[f[q[i]]]);
    puts("");
    for (int u : ver) f[u] = type[u] = 0;
  }
  return 0;
}
