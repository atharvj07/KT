#include <bits/stdc++.h>
using namespace std;
const int N_MAX = 1002;
int n;
int a[N_MAX];
bool ans[N_MAX][N_MAX];
bool ok[N_MAX];
int pos = 1;
void f(int l, int r) {
  if (l >= r) return;
  int cnt = a[pos];
  int x = l - 1;
  for (int i = l; i < r && cnt > 0; i++) {
    x = i;
    ans[i][pos] = ok[i] = true;
    cnt--;
  }
  for (int i = 1; i <= n + 1 && cnt > 0; i++)
    if (i < l || r < i) {
      ans[i][pos] = ok[i] = true;
      cnt--;
    }
  pos++;
  f(l, x);
  f(x + 1, r);
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  f(1, n + 1);
  int lg = n + 1;
  while (lg > 0 && ok[lg] == false) lg--;
  cout << lg << "\n";
  for (int i = 1; i <= lg; i++, cout << "\n")
    for (int j = 1; j <= n; j++) cout << ans[i][j];
  return 0;
}
