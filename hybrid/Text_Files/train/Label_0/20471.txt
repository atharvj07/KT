#include <bits/stdc++.h>
using namespace std;
const int mod = 1000000007, N = 500005;
long long a, b, c, d, e, f, g, h[N], arr[N];
vector<long long> ans;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cin >> a;
  for (long long i = 2; i <= a; i++) {
    ans.push_back(1);
    for (long long j = 2; j * j <= i; j++) {
      if (i % j == 0) {
        ans[i - 2] = i / j;
        break;
      }
    }
  }
  sort(ans.begin(), ans.end());
  for (long long i = 0; i < a - 1; i++) {
    cout << ans[i] << ' ';
  }
  cout << endl;
}
