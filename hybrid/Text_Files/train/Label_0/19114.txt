#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
long long n, m, k, p, l, r, mid, ans;
long long h[maxn], a[maxn], c[maxn];
bool check(long long x) {
  priority_queue<pair<long long, long long>,
                 vector<pair<long long, long long> >,
                 greater<pair<long long, long long> > >
      q;
  for (int i = 0; i <= n; i++) c[i] = 0;
  for (int i = 1; i <= n; i++)
    if (x - a[i] * m < h[i]) q.push(pair<long long, long long>(x / a[i], i));
  for (int i = 1; i <= m && q.size(); i++)
    for (int j = 1; j <= k && q.size(); j++) {
      pair<long long, long long> t = q.top();
      q.pop();
      if (t.first < i) return 0;
      ++c[t.second];
      if (x + c[t.second] * p - a[t.second] * m < h[t.second])
        q.push(pair<long long, long long>((x + c[t.second] * p) / a[t.second],
                                          t.second));
    }
  return q.empty();
}
int main() {
  scanf("%lld%lld%lld%lld", &n, &m, &k, &p);
  for (int i = 1; i <= n; i++) {
    scanf("%lld%lld", &h[i], &a[i]);
    r = max(r, h[i] + a[i] * m);
  }
  while (l <= r) {
    mid = l + r >> 1;
    if (check(mid))
      r = mid - 1, ans = mid;
    else
      l = mid + 1;
  }
  cout << ans << endl;
}
