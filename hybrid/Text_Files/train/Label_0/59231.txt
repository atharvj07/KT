import java.io.*;
import java.util.*;
import static java.lang.Math.*;

public class Test {

    static ArrayList<Integer> g[];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), s = in.nextInt(), cnt = 0;

        g = new ArrayList[n];
        for(int i = 0; i<n; i++) g[i] = new ArrayList<>();
        for(int i = 0; i<n-1; i++){
            int x = in.nextInt()-1, y = in.nextInt()-1;
            g[x].add(y);
            g[y].add(x);
        }
        for(ArrayList x:g){
            if(x.size()==1) cnt++;
        }

        System.out.println((double)2*s/cnt);

    }

}
