#include<iostream>
using namespace std;
int solve(int,int);
int a,b,c,d;
int main(){
	int h1,h2,k1,k2;
	cin>>h1>>h2>>k1>>k2>>a>>b>>c>>d;
	int hirosi=solve(h1,h2),kenjiro=solve(k1,k2);
	if(hirosi==kenjiro){
		cout<<"even"<<endl;
	}else if(hirosi>kenjiro){
		cout<<"hiroshi"<<endl;
	}else{
		cout<<"kenjiro"<<endl;
	}
	return 0;
}

int solve(int i,int y){
	return i*a+(i/10)*c+y*b+(y/20)*d;
}
