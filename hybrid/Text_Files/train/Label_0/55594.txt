import java.util.*;


public class Main {

    
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        ArrayList<String> pp = new ArrayList<String>();
        
        
        int n = s.nextInt();
        ArrayList<String> words = new ArrayList<String>();
        int size = 1;
        if(n==4)size = 24;
        else if(n==3)size = 6;
        else if(n==2)size = 2;
        
        
        for (int i = 0; i < n; i++) {
            words.add(s.next());
        }
        
        
        ArrayList<String> perms[] = new ArrayList[size];
        int inv[] = new int[size];
        
        int genc[] = new int[n];
        for (int i = 0; i < genc.length; i++) {
            genc[i] = i+1;
        }
        
        for (int i = 0; i < size; i++) {
            perms[i] = new ArrayList<String>();
            inv[i] = calc(genc);
            
            for (int j = 0; j < n; j++) {
                int ix = Integer.parseInt(genc[j]+"");
                perms[i].add(words.get(ix-1));
            }
//          System.out.println(Arrays.toString(perms[i].toArray())+ " "+inv[i]);
//          System.out.println(Arrays.toString(genc));
            next_permutation(genc);
        }
        
        
        
        
        int m = s.nextInt();
        ArrayList<String> w[] = new ArrayList[m];
        for (int i = 0; i < m; i++) {
            w[i] = new ArrayList<String>();
            int ll = s.nextInt();
            for (int j = 0; j < ll; j++) {
                w[i].add(s.next());
            }
        }
        
        
        int min_inv = Integer.MAX_VALUE;
        int ix = -1;
        for (int i = 0; i < m; i++) {
            
            for (int j = 0; j < size; j++) {
                
                int lesha_ix = 0;
                
                for (int j2 = 0; j2 < w[i].size(); j2++) {
                    //System.out.println(w[i].get(j2)+ " "+perms[lesha_ix]);
                    if(lesha_ix < n 
                            && w[i].get(j2).equals(perms[j].get(lesha_ix))){
                        lesha_ix++;
                    }
                }
                if(lesha_ix==n){
                    if(min_inv > inv[j]){
                        min_inv = inv[j];
                        ix = i;
                    }
                }
                //System.out.println(lesha_ix);
            }
            
        }
        ix++;
        if(ix==0)
            System.out.println("Brand new problem!");
        else{
            System.out.println(ix);
            int p = n*(n-1)/2 - min_inv + 1;
            String rez = "[:";
            for (int i = 0; i < p; i++) {
                rez+="|";
            }
            rez+=":]";
            System.out.println(rez);
        }
    }

    
    static boolean next_permutation(int[] p) {
          for (int a = p.length - 2; a >= 0; --a)
            if (p[a] < p[a + 1])
              for (int b = p.length - 1;; --b)
                if (p[b] > p[a]) {
                  int t = p[a];
                  p[a] = p[b];
                  p[b] = t;
                  for (++a, b = p.length - 1; a < b; ++a, --b) {
                    t = p[a];
                    p[a] = p[b];
                    p[b] = t;
                  }
                  return true;
                }
          return false;
        }
    
    
    
    
    private static int calc(int a[]) {
        int count = 0;
        for (int i = 0; i < a.length; i++) {
            for (int j = i+1; j < a.length; j++) {
                if(a[i]>a[j])count++;
            }
        }
        
        return count;
    }

}
