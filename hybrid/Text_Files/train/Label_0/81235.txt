n = int(input())
a = list(map(int, list(input())))
b = list(map(int, list(input())))
A0 = []
A1 = []
CH0 = []
CH1 = []
for i in range(n):
	if a[i] == 0:
		if b[i] == 0:
			CH0.append(i)
		else:
			A0.append(i)
	elif b[i] == 0:
		CH1.append(i)
	else:
		A1.append(i)
ans = len(CH0) * len(A1) + len(CH1) * len(A0) + len(CH0) * len(CH1)
print(ans)
