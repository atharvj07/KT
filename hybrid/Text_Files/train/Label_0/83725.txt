def ladder(n, k, a):
	a = list(a)
	if k == 0:
		print('PRINT', a[0])
		for x in range(1, len(a)):
			print('RIGHT')
			print('PRINT', a[x])
	elif k == n:
		print('PRINT', a[-1])
		for x in range(len(a) - 2, -1, -1):
			print('LEFT')
			print('PRINT', a[x])
	elif k > n // 2:
		for y in range(n - k):
			print('RIGHT')
		print('PRINT', a[-1])
		for x in range(len(a) - 2, -1, -1):
			print('LEFT')
			print('PRINT', a[x])
	else:
		for y in range(k - 1):
			print('LEFT')
		print('PRINT', a[0])
		for x in range(1, len(a)):
			print('RIGHT')
			print('PRINT', a[x])
(n, k) = list(map(int, input().split()))
a = input()
ladder(n, k, a)
