def solve(a):
	b = []
	n = len(a)
	(minv, maxv) = (a[0], a[0])
	for i in range(1, n):
		if a[i] > a[i - 1]:
			maxv = a[i]
		else:
			b.append((minv, maxv))
			minv = a[i]
			maxv = a[i]
	b.append((minv, maxv))
	if len(b) == 1:
		return 'YES'
	return 'YES' if b[-1][1] > b[0][0] else 'NO'
t = int(input())
for case in range(t):
	n = int(input())
	a = [int(x) for x in input().strip().split()]
	print(solve(a))
