from sys import stdin
input = stdin.readline
t = int(input())
for i in range(t):
	(n, q) = map(int, input().split())
	s = input()
	dp = [[0, 0] for i in range(n + 1)]
	for i in range(n):
		a = 0
		b = 0
		if s[i] == '+':
			if i % 2 == 0:
				a = a + 1
			else:
				b = b + 1
		elif i % 2 == 0:
			b = b + 1
		else:
			a = a + 1
		dp[i + 1][0] = dp[i][0] + a
		dp[i + 1][1] = dp[i][1] + b
	for i in range(q):
		(l, r) = map(int, input().split())
		a = dp[r][0] - dp[l - 1][0]
		b = dp[r][1] - dp[l - 1][1]
		if abs(a - b) <= 2:
			print(abs(a - b))
		elif abs(a - b) % 2 == 0:
			print(2)
		else:
			print(1)
