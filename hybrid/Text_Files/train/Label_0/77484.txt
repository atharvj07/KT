from collections import defaultdict
import math
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	d = defaultdict(lambda : 0)
	for i in l:
		d[i] += 1
	l.sort()
	mex = -1
	for i in range(n - 1):
		if l[i + 1] - l[i] > 1:
			mex = l[i] + 1
			break
	if mex == -1 and 0 in d:
		print(len(d) + k)
	else:
		if mex == -1:
			mex = 0
		if k == 0:
			print(len(d))
		elif math.ceil((l[-1] + mex) / 2) in d:
			print(len(d))
		else:
			print(len(d) + 1)
