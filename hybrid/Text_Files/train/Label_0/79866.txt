n = int(input())
p = map(int, input().split())
a = map(int, input().split())
b = map(int, input().split())
m = int(input())
pos = map(int, input().split())
fut = zip(p, a, b)
fut = list(fut)

def sravni(elem):
	return elem[0]
fut.sort(key=sravni)
vz = []
for i in range(n):
	vz.append(False)
lastc = [0, 0, 0]
result = ''
for poset in pos:
	ctoim = -1
	for i in range(lastc[poset - 1], n):
		if vz[i] == False:
			if fut[i][1] == poset or fut[i][2] == poset:
				vz[i] = True
				ctoim = fut[i][0]
				lastc[poset - 1] = i + 1
				break
	if ctoim == -1:
		lastc[poset - 1] = n + 1
	result += str(ctoim) + ' '
print(result)
