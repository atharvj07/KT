
import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
	    Scanner in=new Scanner(System.in);
	    int t=in.nextInt();
	    while(t-->0)
	    {
	          int n=in.nextInt();
	          int a[]=new int[n];
	          long s=0; long k=0; long p=0;
	          for(int i=0;i<n;i++)
	          {
	                a[i]=in.nextInt();
	                s+=a[i];
	          }
	          for(int i=0;i<n;i++)
	         {
	               k+=a[i]-1;i++;
	               
	         }
	         for(int i=1;i<n;i++)
	         {
	              
	               p+=a[i]-1;i++;
	         }
	         if(k<=p)
	         {
	               for(int i=0;i<n;i++)
	         {
	               System.out.print("1"+" ");
	               i++;
	               if(i<n)
	              System.out.print(a[i]+" ");
	         }
	         System.out.println();
	         }
	         else{
	                for(int i=0;i<n;i++)
	         {
	               System.out.print(a[i]+" ");
	               i++;
	               if(i<n)
	              System.out.print("1"+" ");
	         }
	         System.out.println();
	         }
	    }
	}
}
