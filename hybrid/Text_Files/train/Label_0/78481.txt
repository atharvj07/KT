(s, t) = (set(), set())
ans = 3
for i in a:
	if i in s:
		ans = min(ans, 0)
	elif i in t:
		ans = min(ans, 1)
	elif i & x in s:
		ans = min(ans, 1)
	elif i & x in t:
		ans = min(ans, 2)

	t.add(i & x)
