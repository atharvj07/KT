#include <iostream>
#include <cstring>

using namespace std;
#define mod 10000
#define MAX_N (100+1)
int n, k;
int sche[MAX_N];

int dp[MAX_N][4][4];
int cnt(int day, int one, int two) {
    if (day == n) return 1;
    if (dp[day][one][two] >= 0) return dp[day][one][two];
    if (sche[day] > 0) {
        if (one == two && two == sche[day]) return dp[day][one][two] = 0;
        return dp[day][one][two] = cnt(day+1, sche[day], one);
    }
    int ret = 0;
    for (int i = 1; i <= 3; i++) {
        if (one == two && two == i) continue;
        ret += cnt(day+1, i, one);
        ret %= mod;
    }
    return dp[day][one][two] = ret % mod;
}

int main() {
    cin >> n >> k;
    memset(sche, -1, sizeof(sche));
    memset(dp, -1, sizeof(dp));
    for (int i = 0; i < k; i++) {
        int d, p; cin >> d >> p;
        d--;
        sche[d] = p;
    }
//    for (int i = 0; i < n; i++) cout << sche[i] << ' ';cout << endl;
    cout << cnt(0, 0, 0) << endl;
    return 0;
}
