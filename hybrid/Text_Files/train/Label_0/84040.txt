n = int(input())
s = input()
pref = [0] * (n + 1)
suf = [0] * (n + 1)
okp = [0] * (n + 1)
oks = [0] * (n + 1)
okp[0] = 1
oks[n] = 1

def x(i):
	if s[i] == '(':
		return 1
	else:
		return -1

def y(i):
	if s[n - i - 1] == ')':
		return 1
	else:
		return -1
for i in range(n):
	pref[i + 1] = pref[i] + x(i)
	okp[i + 1] = okp[i] & (pref[i + 1] >= 0)
	suf[n - i - 1] = suf[n - i] + y(i)
	oks[n - i - 1] = oks[n - i] & (suf[n - i - 1] >= 0)
ans = 0
for i in range(n):
	if (okp[i] and oks[i + 1]) == 0:
		continue
	if s[i] == '(':
		if pref[i] > 0 and pref[i] - 1 - suf[i + 1] == 0:
			ans += 1
	elif pref[i] + 1 - suf[i + 1] == 0:
		ans += 1
print(ans)
