#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  int n, k;
  cin >> n >> k;
  vector<int> ox(250);
  vector<pair<int, int>> pr(n);
  for (int i = 0; i < n; ++i) {
    cin >> pr[i].first >> pr[i].second;
    ox[pr[i].first]++;
    ox[pr[i].second + 1]--;
  }
  for (int i = 1; i < 250; ++i) {
    ox[i] += ox[i - 1];
  }
  vector<int> ans(n);
  for (int i = 0; i < 250; ++i) {
    while (ox[i] > k) {
      int pos = -1;
      for (int p = 0; p < n; ++p) {
        if (!ans[p] && (pr[p].first <= i && i <= pr[p].second) &&
            (pos == -1 || pr[p].second > pr[pos].second)) {
          pos = p;
        }
      }
      assert(pos != -1);
      for (int j = pr[pos].first; j <= pr[pos].second; ++j) {
        --ox[j];
      }
      ans[pos] = 1;
    }
  }
  cout << accumulate(ans.begin(), ans.end(), 0) << "\n";
  for (int i = 0; i < n; ++i) {
    if (ans[i]) cout << (i + 1) << " ";
  }
  cout << "\n";
  return 0;
}
