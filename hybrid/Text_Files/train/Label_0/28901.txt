#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, x;
  cin >> n >> x;
  vector<int> a(n);
  unordered_map<int, int> freq;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    freq[a[i]]++;
  }
  long long res = 0;
  for (int i = 0; i < n; i++)
    (x == 0) ? res += freq[a[i]] - 1 : res += freq[a[i] ^ x];
  cout << res / 2 << "\n";
  return 0;
}
