public class FourProblem {
    static class Leming implements Comparable<Leming> {
        private int v;
        private int m;
        private int num;
        public Leming(int m, int v, int num) {
            this.m = m;
            this.v = v;
            this.num = num;
        }
        public int compareTo(Leming leming) {
            if (m == leming.m) {
                return v - leming.v;
            }
            return m - leming.m;
        }
    }
    private static Leming [] lemings;
    private static final double EPS = 0.000000000000000001;
    private static boolean isCan(double time, int k, int high) {
        for (int i = lemings.length - 1; i > -1; i--) {
            double timeLeming = k * high / (double)lemings[i].v;
            if ((timeLeming <= time) || (Math.abs(timeLeming - time) < EPS)) {
                k--;
                if (k == 0) {
                    return true;
                }
            }
        }
        return false;
    }
    public static void main(String [] args) {
        Scanner scanner = new Scanner(System.in);
        int lemingCount = scanner.nextInt();
        int ustupCount = scanner.nextInt();
        int ustupHigh = scanner.nextInt();
        int [] m = new int[lemingCount];
        for (int i = 0; i < lemingCount; i++) {
            m[i] = scanner.nextInt();
        }
        lemings = new Leming[lemingCount];
        for (int i = 0; i < lemingCount; i++) {
            int v = scanner.nextInt();
            lemings[i] = new Leming(m[i], v, i + 1);
        }
        Arrays.sort(lemings);
        double rigth = 2000000000;
        double left = 0;
        int countCC = 0;
        while (Math.abs(rigth - left) > EPS) {
            countCC++;
            double time = (rigth + left) / 2;
            if (isCan(time, ustupCount, ustupHigh)) {
                rigth = time;
            } else {
                left = time;
            }
            if (countCC == 500) {
                break;
            }
        }

        LinkedList<Integer> answerList = new LinkedList<Integer>();
        if (isCan(left, ustupCount, ustupHigh))
        {
            int k = ustupCount;
            for (int i = lemings.length - 1; i > -1; i--) {
                double timeLeming = k * ustupHigh / (double)lemings[i].v;
                if ((timeLeming <= left) || (Math.abs(timeLeming - left) < EPS)) {
                    k--;

                    answerList.addFirst(lemings[i].num);
                    if (k == 0) {
                        for (int lemingNum : answerList) {
                            System.out.print(lemingNum + " ");
                        }
                        return;
                    }
                }
            }
        }
        int k = ustupCount;
        for (int i = lemings.length - 1; i > -1; i--) {
            double timeLeming = k * ustupHigh / (double)lemings[i].v;
            if ((timeLeming <= rigth) || (Math.abs(timeLeming - rigth) < EPS)) {
                k--;
                answerList.addFirst(lemings[i].num);
                if (k == 0) {
                    for (int lemingNum : answerList) {
                        System.out.print(lemingNum + " ");
                    }
                    return;
                }
            }
        }
    }
}
