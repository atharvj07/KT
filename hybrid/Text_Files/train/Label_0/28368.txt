#include <bits/stdc++.h>
using namespace std;
int n, a[100010], b[100010], c[100010], dp[100010][5];
int Solve(int pos, int p) {
  if (pos == n) return 0;
  if (dp[pos][p] != -1) return dp[pos][p];
  int res = 0;
  if (p == 0) {
    res = max(res, Solve(pos + 1, 1) + a[pos]);
    if (pos + 1 < n) res = max(res, Solve(pos + 1, 0) + b[pos]);
  } else {
    res = max(res, Solve(pos + 1, 1) + b[pos]);
    if (pos + 1 < n) res = max(res, Solve(pos + 1, 0) + c[pos]);
  }
  return (dp[pos][p] = res);
}
int main() {
  memset(dp, -1, sizeof dp);
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int i = 0; i < n; i++) scanf("%d", &b[i]);
  for (int i = 0; i < n; i++) scanf("%d", &c[i]);
  printf("%d", Solve(0, 0));
  return 0;
}
