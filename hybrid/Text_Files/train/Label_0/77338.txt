def main():
	count = int(input())
	arr = input().split()
	biggest = -10 ** 6
	second_biggest = -10 ** 6
	smallest = 10 ** 6
	second_smallest = 10 ** 6
	for x in range(count):
		test = int(arr[x])
		if test > second_biggest:
			if test > biggest:
				(biggest, second_biggest) = (test, biggest)
			else:
				second_biggest = test
		if test < second_smallest:
			if test < smallest:
				(smallest, second_smallest) = (test, smallest)
			else:
				second_smallest = test
	print(min(biggest - second_smallest, second_biggest - smallest))
main()
