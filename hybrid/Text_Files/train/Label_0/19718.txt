#include<bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(int)(n);i++)
using namespace std;
typedef vector<int> vec;

set<string> bdd;

void construct(string s){
  if(bdd.count(s))return;
  if(s.size() == 1)return;

  int n = s.size();
  string a = s.substr(0,n/2), b = s.substr(n/2);
  if(a == b){
    construct(a);
  }else{
    bdd.insert(s);
    construct(a); construct(b);
  } 
}

int main(){
  int n;
  cin >> n;
  string s;
  cin >> s;

  construct(s);

  cout << bdd.size() << endl;
}
