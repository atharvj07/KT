from sys import stdin
n = int(input())
if n == 1:
	print(6)
	exit()
arr = stdin.read().strip().splitlines()
ans = 6
i = 0
while i < n:
	j = i + 1
	while j < n:
		(s, k) = (0, 0)
		while k < 6:
			if arr[i][k] == arr[j][k]:
				s += 1
			k += 1
		if s < 2:
			ans = min(ans, 2)
		elif s < 4:
			ans = 1
		elif s < 6:
			print(0)
			exit()
		j += 1
	i += 1
print(ans)
