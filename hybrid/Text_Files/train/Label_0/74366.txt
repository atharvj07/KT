n = int(input())
a = [int(x) for x in input().split()]
(s, f) = (int(x) for x in input().split())
a = a + a[:f - s - 1]
t = s
max = -1
max_t = t
sum = sum(a[0:0 + f - s])
for i in range(n):
	if sum >= max:
		if max < sum:
			max_t = t
			max = sum
		elif max == sum and max_t > t:
			max_t = t
	if i < n - 1:
		sum += a[i + f - s] - a[i]
	t -= 1
	if t == 0:
		t = n
print(max_t)
