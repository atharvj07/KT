    #include <iostream>
    #include <cstdio>
    #include <queue>
    #include <algorithm>
    using namespace std;

    const int N = 2E5 + 5;

    queue<int> q, cur;
    int n, a[N], b[N], dsu[N];
    long long ans = 0;

    int trace(int u)
    {
        return (dsu[u] < 0 ? u : dsu[u] = trace(dsu[u]));
    }

    void unity(int u, int v)
    {
        if ((u = trace(u)) == (v = trace(v)))
            return;
        if (dsu[u] > dsu[v])
            swap(u, v);
        dsu[u] += dsu[v];
        dsu[v] = u;
    }

    int main()
    {
        scanf("%d", &n);
        for (int i = 1; i <= n; i++)
        {
            scanf("%d", a + i);
            dsu[i] = -1;
            b[i] = i;
        }
        sort(b + 1, b + n + 1, [](const int &x, const int &y) { return a[x] < a[y]; });
        for (int i = 1; i <= n; i++)
            a[b[i]] = i;
        for (int i = 1; i <= n; i++)
            q.push(a[i]);
        for (int i = 1; i <= n; i++)
        {
            while (trace(q.front()) != trace(i))
            {
                ans -= dsu[trace(q.front())];
                cur.push(q.front());
                q.pop();
            }
            ans -= (dsu[trace(i)] + 1);
            q.pop();
            while (!cur.empty())
            {
                unity(cur.front(), i);
                cur.pop();
            }
            if (dsu[trace(i)] < -1)
            {
                ++dsu[trace(i)];
                q.push(trace(i));
            }
        }
        printf("%lld\n", ans);
    }
