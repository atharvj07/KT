t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	a = [[int(j) for j in input().split()[:m]] for i in range(n)]
	i1 = 0
	i2 = n - 1
	j1 = 0
	j2 = m - 1
	cnt = 0
	f2 = 0
	f3 = 0
	f4 = 0
	f5 = 0
	if n % 2 == 0 and m % 2 == 0:
		while i2 > i1:
			x = [a[i1][j1], a[i1][j2], a[i2][j1], a[i2][j2]]
			x.sort()
			t = x[1]
			cnt1 = abs(a[i1][j1] - t)
			cnt2 = abs(a[i1][j2] - t)
			cnt3 = abs(a[i2][j1] - t)
			cnt4 = abs(a[i2][j2] - t)
			a[i1][j1] = t
			a[i1][j2] = t
			a[i2][j1] = t
			a[i2][j2] = t
			cnt = cnt + cnt1 + cnt2 + cnt3 + cnt4
			if j2 - j1 > 1:
				j1 += 1
				j2 -= 1
			else:
				i1 += 1
				i2 -= 1
				j1 = 0
				j2 = m - 1
		print(cnt)
	elif n % 2 != 0 and m % 2 == 0:
		while i2 >= i1:
			if f2 == 1:
				x = [a[i1][j1], a[i1][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i1][j2] - t)
				a[i1][j1] = t
				a[i1][j2] = t
				cnt = cnt + cnt1 + cnt2
			else:
				x = [a[i1][j1], a[i1][j2], a[i2][j1], a[i2][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i1][j2] - t)
				cnt3 = abs(a[i2][j1] - t)
				cnt4 = abs(a[i2][j2] - t)
				a[i1][j1] = t
				a[i1][j2] = t
				a[i2][j1] = t
				a[i2][j2] = t
				cnt = cnt + cnt1 + cnt2 + cnt3 + cnt4
			if j2 - j1 > 1:
				j1 += 1
				j2 -= 1
				if i1 == i2:
					f2 = 1
			else:
				i1 += 1
				i2 -= 1
				j1 = 0
				j2 = m - 1
				if i1 == i2:
					f2 = 1
					j1 = 0
					j2 = m - 1
		print(cnt)
	elif n % 2 == 0 and m % 2 != 0:
		while i2 > i1:
			if f3 == 1:
				x = [a[i1][j1], a[i2][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i2][j2] - t)
				a[i1][j1] = t
				a[i2][j2] = t
				cnt = cnt + cnt1 + cnt2
			else:
				x = [a[i1][j1], a[i1][j2], a[i2][j1], a[i2][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i1][j2] - t)
				cnt3 = abs(a[i2][j1] - t)
				cnt4 = abs(a[i2][j2] - t)
				a[i1][j1] = t
				a[i1][j2] = t
				a[i2][j1] = t
				a[i2][j2] = t
				cnt = cnt + cnt1 + cnt2 + cnt3 + cnt4
			if j2 == j1:
				i1 += 1
				i2 -= 1
				j1 = 0
				j2 = m - 1
				f3 = 0
			else:
				j1 += 1
				j2 -= 1
				if j1 == j2:
					f3 = 1
		print(cnt)
	elif n % 2 != 0 and m % 2 != 0:
		while i2 >= i1:
			if f4 == 1:
				x = [a[i1][j1], a[i2][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i2][j2] - t)
				a[i1][j1] = t
				a[i2][j2] = t
				cnt = cnt + cnt1 + cnt2
			if f5 == 1:
				x = [a[i1][j1], a[i1][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i1][j2] - t)
				a[i1][j1] = t
				a[i1][j2] = t
				cnt = cnt + cnt1 + cnt2
			else:
				x = [a[i1][j1], a[i1][j2], a[i2][j1], a[i2][j2]]
				x.sort()
				t = x[1]
				cnt1 = abs(a[i1][j1] - t)
				cnt2 = abs(a[i1][j2] - t)
				cnt3 = abs(a[i2][j1] - t)
				cnt4 = abs(a[i2][j2] - t)
				a[i1][j1] = t

				a[i2][j1] = t

					f4 = 1
