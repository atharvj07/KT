def gcd(a, b):
	if a == 0:
		return b
	return gcd(b % a, a)

def solve(x):
	if x == 2 or x == 3 or x == 5:
		return x
	if x == 4:
		return 2
	primes = []
	root = int(x ** 0.5) + 5
	for i in range(2, root):
		if x % i == 0:
			primes.append(i)
			primes.append(x // i)
	if len(primes) == 0:
		return x
	g = primes[0]
	for p in primes:
		g = gcd(g, p)
	return g
n = int(input())
if n == 1:
	print(1)
else:
	print(solve(n))
