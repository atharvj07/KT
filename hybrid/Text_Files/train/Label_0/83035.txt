from collections import defaultdict
for __ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	graph = {}
	for i in range(n):
		graph[a[i]] = b[i]
	visited = defaultdict(bool)

	def dfs(u):
		visited[u] = True
		v = graph[u]
		while not visited[v]:
			visited[v] = True
			v = graph[v]
	count = 0
	for i in range(1, n + 1):
		if not visited[i]:
			count += 1
			dfs(i)
	print(pow(2, count, 10 ** 9 + 7))
