import java.util.*;
import java.io.*;

public class A {
	public static void main(String[] args) throws Exception {
		PrintStream out = new PrintStream("output.txt");
		Scanner sc = new Scanner(new File("input.txt"));
		int t = sc.nextInt();
		int k = sc.nextInt();
		for (int i=0; i<t; i++) {
			int n = sc.nextInt();
			int m = sc.nextInt();
			if (k!=1) {
				int min = Math.min(n, m);
				if (min % (k + 1) == 0) {
					out.println('+');
				} else {
					boolean tmp = (min / (k + 1)) % 2 == 0 ? false : true;
					int max = Math.max(n, m);
					if ((max - min) % 2 == 1)
						tmp = !tmp;
					if (tmp)
						out.println('+');
					else
						out.println('-');
				}
			}
			else {
				if (m%2==1 && n%2==1)
					out.println('-');
				else
					out.println('+');
			}
		}
		out.close();
	}
}
