#include <bits/stdc++.h>
using namespace std;
template <typename T>
inline void read(T &x) {
  short f = 1;
  x = 0;
  char c = getchar();
  while (!isdigit(c)) {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (isdigit(c)) x = (x << 3) + (x << 1) + (c ^ 48), c = getchar();
  x *= f;
}
int main() {
  string s = "1001010111001010";
  int n;
  read(n);
  cout << s[n - 1] << endl;
  return 0;
}
