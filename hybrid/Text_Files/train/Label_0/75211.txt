import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
	(l, r) = map(int, input().split())
	L1 = [0] * 18
	L2 = [0] * 18
	for i in range(18):
		(a, b) = divmod(l, 2 ** (i + 1))
		L1[i] = a * 2 ** i + max(0, b - 2 ** i)
	for i in range(18):
		(a, b) = divmod(r + 1, 2 ** (i + 1))
		L2[i] = a * 2 ** i + max(0, b - 2 ** i)
	m = 1000000
	for i in range(18):
		k = r - l + 1 - (L2[i] - L1[i])
		if k < m:
			m = k
	print(m)
