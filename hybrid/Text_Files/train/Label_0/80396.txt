import sys
LI = lambda : list(map(int, sys.stdin.readline().split()))
MI = lambda : map(int, sys.stdin.readline().split())
SI = lambda : sys.stdin.readline().strip('\n')
II = lambda : int(sys.stdin.readline())
for _ in range(II()):
	(n, m) = MI()
	(n, p) = (bin(n)[2:], bin(m + 1)[2:])
	l = max(len(n), len(p))
	(n, p) = ('0' * (l - len(n)) + n, '0' * (l - len(p)) + p)
	s = ''
	for i in range(l):
		if n[i] == '1':
			if p[i] == '1':
				s += '0'
			else:
				s += (l - len(s)) * '0'
				break
		elif p[i] == '1':
			s += '1'
		else:
			s += '0'
	print(int(s, 2))
