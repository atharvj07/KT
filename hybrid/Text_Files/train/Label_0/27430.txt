#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ll n;
  cin >> n;
  for (int k = 0; k < n; k++) {
    ll a, time = 0, A = 0;
    cin >> a;
    string s;
    cin >> s;
    a = s.size();
    vector<int> ar(a);
    for (int i = 0; i < a; i++) {
      for (int j = i + 1; j < a; j++) {
        if (s[i] == 'A' && s[j] == 'P') {
          time++;
        } else
          break;
      }
      if (time > 0) {
        ar.push_back(time);
        time = 0;
        A++;
      }
    }
    sort(ar.begin(), ar.end());
    reverse(ar.begin(), ar.end());
    if (ar[0] > 0)
      cout << ar[0] << endl;
    else
      cout << 0 << endl;
  }
  return 0;
}
