from collections import OrderedDict
for i in range(int(input())):
	(n, m) = map(int, input().split())
	(a, c) = ([*map(int, input().split())], [*map(int, input().split())])
	a = sorted(a)[::-1]
	d = OrderedDict()
	for i in a:
		d[i] = d.get(i, 0) + 1
	j = 0
	cnt = 0
	ans = 0
	for (key, value) in d.items():
		while cnt < key and value > 0:
			cnt += 1
			value -= 1
		ans += value * c[key - 1]
	for i in range(cnt):
		ans += c[i]
	print(ans)
