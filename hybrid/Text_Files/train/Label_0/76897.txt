def squirrel_jump(n, k, road):
	min_jump = 0
	temp = 0
	if k >= n - 1:
		return 'YES'
	else:
		for i in road:
			if i == '.':
				if temp > min_jump:
					min_jump = temp
				temp = 0
			else:
				temp += 1
		if k > min_jump:
			return 'YES'
		else:
			return 'NO'
(n, k) = list(map(int, input().split()))
road = input()
print(squirrel_jump(n, k, road))
