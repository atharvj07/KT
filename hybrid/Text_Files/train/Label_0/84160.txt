import math
t = int(input())
for k in range(t):
	inp = input().split()
	n = int(inp[0])
	h = int(inp[1])
	arr = []
	inp = input().split()
	for i in inp:
		arr.append(int(i))
	arr1 = []
	for i in range(len(arr) - 1):
		arr1.append(arr[i + 1] - arr[i])
	arr1.sort()
	k = 0
	for i in range(len(arr) - 1):
		if arr1[i] * (n - i) >= h:
			if h % (n - i) == 0:
				k = h // (n - i)
			else:
				k = h // (n - i) + 1
			break
		else:
			h -= arr1[i]
	print(k if k != 0 else h)
