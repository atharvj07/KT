from __future__ import division, print_function
import io
import os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
l = list(map(int, input().split()))
su = [l[0]]
cou = [-1, 0]
for k in range(1, n):
	nd = 1
	ns = l[k]
	while len(cou) > 1 and su[-1] * (cou[-1] - cou[-2] + nd) > (su[-1] + ns) * (cou[-1] - cou[-2]):
		nd += cou[-1] - cou[-2]
		ns += su[-1]
		su.pop()
		cou.pop()
	cou.append(k)
	su.append(ns)
floats = []
for k in range(len(su)):
	floats += [su[k] / (cou[k + 1] - cou[k])] * (cou[k + 1] - cou[k])
if True:
	if False:
		print('\n'.join(map(str, floats)))
	if False:
		os.write(1, b'\n'.join((str(x).encode() for x in floats)))
	if False:
		os.write(1, b'\n'.join((b'%.9f' % x for x in floats)))
	if False:
		os.write(1, b'\n'.join((b'%d.%09d' % (x, (x - int(x)) * 1000000000.0 + 0.5) for x in floats)))
	if True:
		ans = bytearray()
		for x in floats:
			i = int(x)
			ans += b'%d.%09d\n' % (i, (x - i) * 1000000000.0 + 0.5)
		os.write(1, ans)
else:
	ans = bytearray()
	for k in range(len(su)):
		repeat = cou[k + 1] - cou[k]
		x = su[k] / repeat
		i = int(x)
		ans += b'%d.%09d\n' % (i, (x - i) * 1000000000.0 + 0.5) * repeat
	os.write(1, ans)
