def main():
	import sys
	input = sys.stdin.readline
	M = int(1000000000.0 + 7)
	for _ in range(int(input())):
		n = int(input())
		s = list(map(int, input().split()))
		l = [[]]
		for i in range(n):
			l[0].append(s[i])
		while True:
			d = {}
			l.append([])
			for i in range(n):
				if s[i] in d:
					d[s[i]] += 1
				else:
					d[s[i]] = 1
			for i in range(n):
				s[i] = d[s[i]]
				l[-1].append(s[i])
			if l[-1] == l[-2]:
				break
		for _ in range(int(input())):
			(a, k) = map(int, input().split())
			if k >= len(l):
				print(l[-1][a - 1])
			else:
				print(l[k][a - 1])
main()
