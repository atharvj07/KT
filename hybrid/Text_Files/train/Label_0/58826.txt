public class C_292 {
    
    public static void main(String a[])
    {
        Scanner inp = new Scanner(System.in);
        
        String s = inp.nextLine();
        s = inp.nextLine();
        Operation ops = new Operation();
        
        char output[];
        try
        {
        output = ops.output(s);
        }
        catch(Exception e)
        {
            e.printStackTrace();
            return;
        }
        
        for(int i=0;i<output.length;i++)
        System.out.print(output[i]);

    }

}

class Operation 
{
    public char[] output(String s)
    {
        String ou ="";
        
        
        for(int i=0;i<s.length();i++)
        {
            int t= s.charAt(i)-'0';
            
            if(t==0 ||t ==1)
                continue;
            else if(t==2||t==3||t==5||t==7)
                ou = ou+ t;
            else if(t==4)
            {
                ou = ou+"322";
            }
            else if(t==6)
            {
                ou = ou+"53";
            }
            else if(t==8)
                ou = ou+"7222";
            else if(t==9)
                ou = ou+"7332";
            
            
        }
        
        char []out = ou.toCharArray();
        //System.out.println(out.length);
        
        
        for(int i=0;i<out.length;i++)
        {
            for(int j=i+1;j<out.length;j++)
            {
                if(out[i]<out[j])
                {
                    char temp=out[i];
                    out[i]=out[j];
                    out[j]=temp;
                }
            }
        }
        
        char pa[] = new char[out.length];
        for(int i=0;i<pa.length;i++)
        {
            pa[i]=out[i];
        }
        
        
        return pa;
        
    }
    }
