#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  int p = n;
  float d = 0;
  vector<pair<float, float>> v;
  while (p--) {
    float a, b;
    cin >> a >> b;
    v.push_back(make_pair(a, b));
  }
  for (int i = 0; i < n - 1; i++) {
    d = d + sqrt(pow(abs(v[i].first - v[i + 1].first), 2) +
                 pow(abs(v[i].second - v[i + 1].second), 2));
  }
  d = d / 50;
  d = d * k;
  printf("%.6f", d);
  return 0;
}
