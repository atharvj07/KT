from sys import stdin, stdout
s = stdin.readline().strip()
n = len(s)
labels = [-1 for i in range(n)]
used = [0 for i in range(n)]
(l0, l1) = (0, 0)
for i in range(n):
	(l0, l1) = (max(l0, i), max(l1, i))
	if s[i] == '0':
		while l0 < n and (s[l0] == '0' or used[l0]):
			l0 += 1
		if l0 != n:
			labels[i] = l0
			used[l0] = 1
			l0 += 1
	else:
		while l1 < n and (s[l1] == '1' or used[l1]):
			l1 += 1
		if l1 != n:
			labels[i] = l1
			used[l1] = 1
			l1 += 1
for i in range(n):
	if s[i] == '1' and (labels[i] == -1 or not used[i]):
		stdout.write('-1')
		break
else:
	used = [0 for i in range(n + 1)]
	g_ans = []
	for i in range(n):
		if used[i]:
			continue
		ans = [i + 1]
		used[i] = 1
		ind = labels[i]
		while ind != -1:
			used[ind] = 1
			ans.append(ind + 1)
			ind = labels[ind]
		g_ans.append(ans)
	stdout.write(str(len(g_ans)) + '\n')
	for v in g_ans:
		stdout.write(str(len(v)) + ' ' + ' '.join(list(map(str, v))) + '\n')
