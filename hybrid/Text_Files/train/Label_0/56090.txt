    static int N, M, K;

    static boolean[][] d = new boolean[1001][1001];
    static pair[][] dpV = new pair[1001][1001];
    static pair[][] dpH = new pair[1001][1001];
    static boolean[][] v = new boolean[1001][1001];

    static void DFS(int i, int j) {
        if (v[i][j] || d[i][j]) {
            return;
        }
        v[i][j] = true;

        int[] mx = {-1, 0, 0, 1};
        int[] my = {0, -1, 1, 0};

        for (int k = 0; k < 4; k++) {
            int nx = i + mx[k];
            int ny = j + my[k];

            if (nx < 0 || nx >= N || ny < 0 || ny >= M) {
                continue;
            }
            if (d[nx][ny]) {
                continue;
            }
            boolean flag = false;
            if (nx == i) {
                if (dpV[nx][ny].a >= dpH[i][j].a + 1) {
                    dpV[nx][ny].a = dpH[i][j].a + 1;
                    dpV[nx][ny].b = K - 1;
                    flag = true;
                }

                if (dpH[nx][ny].a > dpH[i][j].a && dpH[i][j].b > 0) {
                    dpH[nx][ny].a = dpH[i][j].a;
                    dpH[nx][ny].b = dpH[i][j].b - 1;
                    flag = true;
                } else if (dpH[nx][ny].a == dpH[i][j].a) {
                    dpH[nx][ny].b = Math.max(dpH[nx][ny].b, dpH[i][j].b - 1);
                    flag = true;
                } else if (dpH[nx][ny].a > dpH[i][j].a + 1) {
                    dpH[nx][ny].a = dpH[i][j].a + 1;
                    dpH[nx][ny].b = K - 1;
                    flag = true;
                }
            }

            if (ny == j) {
                if (dpH[nx][ny].a >= dpV[i][j].a + 1) {
                    dpH[nx][ny].a = dpV[i][j].a + 1;
                    dpH[nx][ny].b = K - 1;
                    flag = true;
                }

                if (dpV[nx][ny].a > dpV[i][j].a && dpV[i][j].b > 0) {
                    dpV[nx][ny].a = dpV[i][j].a;
                    dpV[nx][ny].b = dpV[i][j].b - 1;
                    flag = true;
                } else if (dpV[nx][ny].a == dpV[i][j].a) {
                    dpV[nx][ny].b = Math.max(dpV[nx][ny].b, dpV[i][j].b - 1);
                    flag = true;
                } else if (dpV[nx][ny].a > dpV[i][j].a + 1) {
                    dpV[nx][ny].a = dpV[i][j].a + 1;
                    dpV[nx][ny].b = K - 1;
                    flag = true;
                }
            }

            if (flag) {
                DFS(nx, ny);
            }
//            pair H =

//            DFS(nx,ny);
        }
    }

class pair implements Comparable<pair> {

    int a, b;

    public pair(int _a, int _b) {
        this.a = _a;
        this.b = _b;
    }

    @Override
    public int compareTo(pair t) {
        return (a == t.a) ? b - t.b : a - t.a;
    }
}
