from sys import stdin, stdout
from collections import defaultdict, Counter, deque

def computeXOR(n):
	if n % 4 == 0:
		return n
	if n % 4 == 1:
		return 1
	if n % 4 == 2:
		return n + 1
	return 0
for _ in range(int(stdin.readline())):
	(a, b) = map(int, stdin.readline().split())
	xor = computeXOR(a - 1)
	if xor == b:
		print(a)
	elif xor ^ a == b:
		print(a + 2)
	else:
		print(a + 1)
