import random
from collections import Counter, defaultdict
from copy import deepcopy as dc
import math
try:
	t = int(input())
	for tt in range(1, t + 1):
		(n, a, b) = map(int, input().split())
		if abs(a - b) > 1 or a + b > n - 2:
			print(-1)
		else:
			x = [i + 1 for i in range(n)]
			if a > b:
				y = x[:n - a]
				(i, j) = (0, 0)
				y.reverse()
				x = x[n - a:]
				while i < len(y):
					x.insert(j, y[i])
					j += 2
					i += 1
				print(*x)
			elif b > a:
				y = x[b:]
				(i, j) = (0, 0)
				x = x[:b]
				while i < len(y):
					x.insert(j, y[i])
					j += 2
					i += 1
				print(*x)
			else:
				y = x[a + 1:]
				(i, j) = (0, 1)
				x = x[:a + 1]
				while i < len(y):
					x.insert(j, y[i])
					j += 2
					i += 1
				print(*x)
except EOFError as e:
	pass
