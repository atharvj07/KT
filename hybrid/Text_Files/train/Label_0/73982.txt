n = int(input())
seq = list(map(int, input().split()))
high = []
low = []
for i in range(len(seq)):
	if i == 0:
		high.append(seq[0])
		low.append(0)
	elif seq[i] > seq[i - 1]:
		x = seq[i] - seq[i - 1]
		low.append(low[i - 1] + x)
		high.append(high[i - 1])
	else:
		low.append(low[i - 1])
		high.append(seq[i] - low[i])
ans = low + high[::-1]
print(*ans)
