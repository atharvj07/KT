import sys
input = sys.stdin.readline
from collections import deque
N = int(input())
COL = [int(a) for a in input().split()]
X = [[] for i in range(N)]
for i in range(N - 1):
	(x, y) = map(int, input().split())
	X[x - 1].append(y - 1)
	X[y - 1].append(x - 1)
P = [-1] * N
Q = deque([0])
R = []
while Q:
	i = deque.popleft(Q)
	R.append(i)
	for a in X[i]:
		if a != P[i]:
			P[a] = i
			X[a].remove(i)
			deque.append(Q, a)
unit = 0
merge = lambda a, b: max(a, 0) + max(b, 0)
adj_bu = lambda a, i: a + 1 if COL[i] else a - 1
adj_td = lambda a, i: a + 1 if COL[P[i]] else a - 1
ans = lambda i: BU[i] + max(TD[i], 0)
BU = [unit] * N
TD = [unit] * N
for i in R[1:][::-1]:
	BU[i] = adj_bu(BU[i], i)
	p = P[i]
	BU[p] = merge(BU[p], BU[i])
BU[R[0]] = adj_bu(BU[R[0]], R[0])
AX = [0] * N
for i in R:
	ac = unit
	for j in X[i]:
		AX[j] = ac
		ac = merge(ac, BU[j])
	ac = unit
	for j in X[i][::-1]:
		AX[j] = merge(AX[j], ac)
		TD[j] = adj_td(merge(TD[i], AX[j]), j)
		ac = merge(ac, BU[j])
ANS = [ans(i) for i in range(N)]
print(*ANS)
