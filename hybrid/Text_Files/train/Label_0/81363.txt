import sys
input = sys.stdin.readline

def inI():
	inputLine = input().split()
	return int(inputLine[0]) if len(inputLine) == 1 else map(int, inputLine)

def inIL():
	return list(map(int, input().split()))

def inCL():
	return list(input())[:-1]

def out(*liste):
	if isinstance(liste[0], list):
		liste = liste[0]
	for element in liste:
		print(element, end=' ')
	print()
t = inI()
for it in range(t):
	(n, l, r) = inI()
	socks = inIL()
	(left, right) = (socks[:l], socks[l:])
	if l < r:
		(left, right) = (right, left)
		(r, l) = (l, r)
	nbleft = [0] * (n + 1)
	sumleft = l
	nbright = [0] * (n + 1)
	sumright = r
	for color in left:
		nbleft[color] += 1
	for color in right:
		nbright[color] += 1
	for color in range(1, n + 1):
		paired = min(nbleft[color], nbright[color])
		nbleft[color] -= paired
		sumleft -= paired
		nbright[color] -= paired
		sumright -= paired
	nbop = 0
	for color in range(1, n + 1):
		if sumleft <= sumright:
			break
		switch = min(nbleft[color] // 2, (sumleft - sumright) // 2)
		nbop += switch
		nbleft[color] -= 2 * switch
		sumleft -= 2 * switch
	nbop += sumleft
	print(nbop)
