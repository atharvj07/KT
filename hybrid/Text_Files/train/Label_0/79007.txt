from collections import defaultdict
bra = [input().strip() for _ in range(int(input()))]
ope = defaultdict(int)
clo = defaultdict(int)
for b in bra:
	if b[0] == '(':
		a = 0
		for c in b:
			if c == '(':
				a += 1
			else:
				a -= 1
				if a < 0:
					break
		else:
			ope[a] += 1
	if b[-1] == ')':
		a = 0
		for c in reversed(b):
			if c == ')':
				a += 1
			else:
				a -= 1
				if a < 0:
					break
		else:
			clo[a] += 1
print(sum((y * ope[x] for (x, y) in clo.items())))
