_ = input()
s = input()
t = input()
ans = None
star_ind = s.find('*')
if star_ind == -1:
	if s == t:
		ans = True
	else:
		ans = False
else:
	pref_len = star_ind
	suff_len = len(s) - star_ind - 1
	ans = True
	ans &= pref_len + suff_len <= len(t)
	if pref_len and s[:pref_len]:
		ans &= s[:pref_len] == t[:pref_len]
	if suff_len and t[-suff_len:]:
		ans &= s[-suff_len:] == t[-suff_len:]
print('YES' if ans else 'NO')
