// C - Sequence
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define up(i,s,e,d) for(int i=(s);i<(e);i+=(d))

int main(){
	int N; cin>>N;
	vector<ll> A(N);

	ll csum = 0;
	ll a = 0;
	up(i,0,N,1){
		cin>>A[i];
		csum += A[i];
		if(i%2==0 && csum<=0){//even +
			a += abs(csum) + 1;
			csum = 1;
		}
		if(i%2==1 && csum>=0){//odd -
			a += abs(csum) + 1;
			csum = -1;
		}
	}

	csum = 0;
	ll b = 0;
	up(i,0,N,1){
		csum += A[i];
		if(i%2==1 && csum<=0){//odd +
			b += abs(csum) + 1;
			csum = 1;
		}
		if(i%2==0 && csum>=0){//even -
			b += abs(csum) + 1;
			csum = -1;
		}
	}

	cout<< (a<b?a:b) <<endl;
}
