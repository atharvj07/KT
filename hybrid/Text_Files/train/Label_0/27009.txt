#include"bits/stdc++.h"
using namespace std;

int main()
{
    int N, K, R, S, P; string T;
    cin >> N >> K >> R >> S >> P >> T;

    long sum = 0;
    bool isLose[100010] = { false };
    for (int i = 1; i <= N; i++) {
        if (isLose[i]) continue;

        int next = i + K;
        if (next <= N && T[i-1] == T[next-1]) isLose[next] = true;

        if (T[i-1] == 'r') sum += P;
        else if (T[i-1] == 's') sum += R;
        else sum += S;
    }

    cout << sum << endl;

    return 0;
}
