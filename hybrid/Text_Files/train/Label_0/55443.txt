

import java.io.*;
import java.util.*;

public class Main{
    FastScanner in;
    PrintWriter out;

    long check(long x, int n, int m)
    {
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            long cnt = x / i;
            if (cnt > m)
                cnt = m;
            ans += cnt;
        }

        return ans;
    }

    public void solve()
    {
        int n = in.nextInt();
        int m = in.nextInt();
        long k = in.nextLong();

        long r = n*1L*m;
        long l = 0;

        while (r-l>1) {
            long md = (l + r) >> 1;
            if (check(md, n, m) < k)
                l = md;
            else
                r = md;
        }


        out.println(l+1);
    }


    public void run()
    {
        in = new FastScanner();
        out = new PrintWriter(System.out);

        solve();

        out.close();
    }


    public static void main(String[] args) {
        new Main().run();
    }

    class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        FastScanner() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }
    }
}
