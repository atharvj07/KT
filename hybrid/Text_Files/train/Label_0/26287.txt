#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, n, a[50], b[50], sum1 = 0, sum2 = 0;
  cin >> n;
  for (i = 0; i < n; i++) cin >> a[i];
  for (i = 0; i < n; i++) cin >> b[i];
  for (i = 0; i < n; i++) {
    sum1 += a[i];
    sum2 += b[i];
  }
  if (sum1 == sum2 || sum1 > sum2)
    cout << "YES";
  else
    cout << "NO";
}
