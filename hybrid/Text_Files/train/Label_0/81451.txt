def is_2020_substring(s):
	return s.startswith('2020') or s.endswith('2020') or (s.startswith('202') and s.endswith('0')) or (s.startswith('20') and s.endswith('20')) or (s.startswith('2') and s.endswith('020'))
for _ in range(int(input())):
	n = int(input())
	s = input()
	print('YES' if is_2020_substring(s) else 'NO')
