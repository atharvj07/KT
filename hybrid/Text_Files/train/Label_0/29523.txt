#include <bits/stdc++.h>
using namespace std;
string a[1000];
int dx[] = {1, 0, -1, 0};
int dy[] = {0, 1, 0, -1};
int vis[100][100];
void dfs(int x, int y) {
  if (x < 0 || y < 0 || y > a[x].size()) return;
  if (vis[x][y]) return;
  if (a[x][y] != '#') return;
  vis[x][y] = 1;
  int i;
  for (i = 0; i < 4; i++) {
    dfs(x + dx[i], y + dy[i]);
  }
}
int main() {
  int i, j, n, m, k, l, disc = 0, cnt = 0;
  cin >> n >> m;
  for (i = 0; i < n; i++) cin >> a[i];
  for (i = 0; i < n; i++) {
    for (j = 0; j < m; j++) {
      if (vis[i][j] == 0 && a[i][j] == '#') {
        dfs(i, j);
        disc += 1;
      }
      cnt += (a[i][j] == '#');
    }
  }
  if (disc > 1) {
    cout << 0 << endl;
    return 0;
  }
  if (cnt == 1) {
    cout << -1 << endl;
    return 0;
  }
  if (cnt == 2) {
    for (i = 0; i < n; i++) {
      for (j = 0; j < m; j++) {
        if (i + 1 < n)
          if (a[i][j] == a[i + 1][j] && a[i][j] == '#') {
            cout << -1 << endl;
            return 0;
          }
        if (j + 1 < m)
          if (a[i][j] == a[i][j + 1] && a[i][j] == '#') {
            cout << -1 << endl;
            return 0;
          }
      }
    }
  }
  for (i = 0; i < n; i++) {
    for (j = 0; j < m; j++) {
      if (a[i][j] != '#') continue;
      a[i][j] = '.';
      memset(vis, 0, sizeof(vis));
      disc = 0;
      for (k = 0; k < n; k++) {
        for (l = 0; l < m; l++) {
          if (vis[k][l] == 0 && a[k][l] == '#') {
            dfs(k, l);
            disc += 1;
          }
        }
      }
      if (disc > 1) {
        cout << 1 << endl;
        return 0;
      }
      a[i][j] = '#';
    }
  }
  cout << 2 << endl;
  return 0;
}
