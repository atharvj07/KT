import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class CF17D {
  static class BasicMaths {
    public static String subtract(String a, String b) {
      if (a.length() < b.length() || (a.length() == b.length() && a.compareTo(b) <= 0)) {
        return "0";
      }
      char[] res = a.toCharArray();
      for (int i = a.length() - 1, j = b.length() - 1; i >= 0; i--, j--) {
        res[i] -= j >= 0 ? (int) b.charAt(j) - (int) '0' : 0;
        if (res[i] < '0') {
          res[i] += 10;
          res[i - 1]--;
        }
      }
      String ans = new String(res);
      return ans.charAt(0) == '0' ? ans.substring(1) : ans;
    }

    public static int mod(String num, int mod) {
      long res = 0;
      for (int i = 0; i < num.length(); i++) {
        res *= 10;
        res += (int) num.charAt(i) - (int) '0';
        res %= mod;
      }
      return (int) res;
    }

    public static int powerMod(long num, long pow, int mod) {
      if (pow == 0) {
        return 1;
      }
      long sqrt = powerMod(num, pow >> 1, mod);
      return (int) ((pow & 1L) == 1 ? sqrt * sqrt % mod * num % mod : sqrt * sqrt % mod);
    }

    public static int powerMod(long num, String pow, int mod) {
      long res = 1;
      long base = num;
      for (int i = pow.length() - 1; i >= 0; i--) {
        res *= powerMod(base, (int) pow.charAt(i) - (int) '0', mod);
        res %= mod;
        base = powerMod(base, 10, mod);
        base %= mod;
      }
      return (int) res;
    }
  }

  private void solve(InputReader in, PrintWriter out) {
    String b = in.next();
    String n = in.next();
    int c = in.nextInt();
    long bb = BasicMaths.mod(b, c);
    long num = BasicMaths.powerMod(bb, BasicMaths.subtract(n, "1"), c) * (bb - 1 + c) % c;
    out.println(num == 0 ? c : num);
  }

  public static void main(String[] args) {
    InputReader in = new InputReader(System.in);
    PrintWriter out = new PrintWriter(System.out);
    new CF17D().solve(in, out);
    out.close();
  }

  static class InputReader {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReader(InputStream stream) {
      reader = new BufferedReader(new InputStreamReader(stream));
      tokenizer = null;
    }

    public String next() {
      while (tokenizer == null || !tokenizer.hasMoreTokens()) {
        try {
          tokenizer = new StringTokenizer(reader.readLine());
        } catch (IOException e) {
          throw new RuntimeException(e);
        }
      }
      return tokenizer.nextToken();
    }

    public int nextInt() {
      return Integer.parseInt(next());
    }
  }
}
