import sys

def input():
	return sys.stdin.buffer.readline()[:-1]
t = int(input())
for _ in range(t):
	n = int(input())
	p = list(map(int, input().split()))
	col = list(map(int, input().split()))
	p = [x - 1 for x in p]
	visited = [False for _ in range(n)]
	cycles = []
	for i in range(n):
		if not visited[i]:
			tmp = [i]
			visited[i] = True
			while p[tmp[-1]] != tmp[0]:
				visited[p[tmp[-1]]] = True
				tmp.append(p[tmp[-1]])
			cycles.append(tmp)
	ans = n
	for cy in cycles:
		l = len(cy)
		ans = min(ans, l)
		divs = []
		i = 1
		while i * i <= l:
			if l % i == 0:
				divs.append(i)
				if i * i != l and i > 1:
					divs.append(l // i)
			i += 1
		for d in divs:
			for i in range(d):
				same = True
				for j in range(i + d, l, d):
					if col[cy[i]] != col[cy[j]]:
						same = False
				if same:
					ans = min(ans, d)
	print(ans)
