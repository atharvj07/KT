#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e6 + 5;
int maxInArray(const vector<int>& array) {
  int result = array[0];
  for (size_t i = 0; i < array.size(); ++i) {
    if (array[i] > result) {
      result = array[i];
    }
  }
  return result;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> array(MAX, 0);
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    array[x] = 1;
  }
  vector<int> d(MAX, 1);
  for (int i = 1; i < MAX; ++i) {
    if (array[i] == 1) {
      for (int j = 2 * i; j <= MAX; j = j + i) {
        if (array[j] == 1) {
          d[j] = max(d[j], d[i] + 1);
        }
      }
    }
  }
  cout << maxInArray(d) << endl;
  return 0;
}
