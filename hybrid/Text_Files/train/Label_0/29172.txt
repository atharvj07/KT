#include <bits/stdc++.h>
using namespace std;
void test(string &s, long long &x) {
  for (int i = 0; i < x; i++) {
    if (s[i] == s[i + 1]) {
      s.erase(i, 2);
      if (i != 0) {
        i -= 2;
      } else {
        i--;
      }
      x -= 2;
    }
    if (s.empty()) {
      break;
    }
  }
  if (s.empty()) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
}
int main() {
  string s;
  cin >> s;
  long long x = s.size();
  test(s, x);
  return 0;
}
