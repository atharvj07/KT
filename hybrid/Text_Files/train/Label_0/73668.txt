import sys
query_num = int(sys.stdin.readline())
for i in range(query_num):
	elem_num = int(sys.stdin.readline())
	lst = sys.stdin.readline().split()
	for k in range(elem_num):
		lst[k] = int(lst[k])
	step_var = 1
	curr_val = 1
	num = 0
	res = 0
	while step_var:
		for x in lst:
			if x == curr_val:
				num += 1
				if curr_val == 2048:
					res = 1
					step_var = 0
					break
				if num == 2:
					lst.remove(curr_val)
					lst.remove(curr_val)
					lst.append(2 * curr_val)
					if curr_val == 1024:
						res = 1
					break
		if num < 2:
			curr_val *= 2
		if curr_val == 4096:
			if num > 0:
				res = 1
			step_var = 0
		num = 0
	if res == 0:
		print('NO')
	else:
		print('YES')
