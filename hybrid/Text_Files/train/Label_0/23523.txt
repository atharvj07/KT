#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int p = s.length(), cnt = 0, flag = 0, j;
  if (s[0] >= 'a' && s[0] <= 'z') flag = 1;
  for (int i = 0; i < p; i++) {
    if (s[i] >= 'A' && s[i] <= 'Z') cnt++;
  }
  if (cnt == p || (flag == 1 && cnt == p - 1)) {
    for (int i = 0; i < p; i++) {
      if (s[i] >= 'A' && s[i] <= 'Z') {
        char q;
        q = s[i];
        s[i] = tolower(q);
      } else {
        char m = s[i];
        s[i] = toupper(m);
      }
    }
    cout << s;
  } else {
    cout << s;
  }
  return 0;
}
