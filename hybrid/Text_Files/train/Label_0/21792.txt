#include <iostream>
#include <algorithm>
#include <vector>
#include <string>

using namespace std;

class Solver {
public:
	static void solve() {
		int n;
		cin >> n;
		vector<int> res;
		while (n > 0) {
			vector<string> s;
			for (int i = 0; i < n; i++) {
				string c;
				cin >> c;
				s.push_back(c);
			}			
			res.push_back(DFS(s));
			cin >> n;
		}

		for (int i = 0; i < res.size(); i++)cout << res[i] << endl;
	}

	static int DFS(vector<string> s) {
		if (s[0][0] == '+')	s.push_back(".0");
		else if (s[0][0] == '*') s.push_back(".1");
		else return s[0][0] - '0';

		int val = s[0][0] == '+' ? 0 : 1;
		vector<string> sTemp;
		for (int i = 1; i < s.size(); i++) {
			if (s[i][0] == '.'
				&& s[i][1]!='.') {
				if (s[0][0] == '+') {
					if (sTemp.size() > 0)val += DFS(sTemp);
				}
				else {
					if (sTemp.size() > 0)val *= DFS(sTemp);
				}
				sTemp.clear();
			}
			sTemp.push_back(s[i].substr(1, s[i].size() - 1));
		}

		return val;
	}
};

int main() {
	Solver::solve();
}
