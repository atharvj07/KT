import sys
input = sys.stdin.buffer.readline

def main():
	t = int(input())
	for i in range(t):
		(n, m) = [int(x) for x in input().split()]
		graph = [[] for i in range(n)]
		for j in range(m):
			(a, b) = [int(x) - 1 for x in input().split()]
			graph[a].append(b)
		depth = [2 for x in range(n)]
		for u in range(n):
			if depth[u] > 0:
				for v in graph[u]:
					depth[v] = min(depth[v], depth[u] - 1)
		sol = [j + 1 for j in range(n) if depth[j] == 0]
		print(len(sol))
		print(*sol)
main()
