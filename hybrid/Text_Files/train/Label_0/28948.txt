#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
const long long inf = (long long)2e18;
long long val[N], dp[N][2];
string s[N][2];
int n;
long long min_(long long x, long long y) { return x < y ? x : y; }
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%lld", &val[i]);
  for (int i = 1; i < N; i++)
    for (int j = 0; j <= 1; j++) dp[i][j] = inf;
  for (int i = 1; i <= n; i++) {
    cin >> s[i][0];
    s[i][1] = s[i][0];
    reverse(s[i][1].begin(), s[i][1].end());
  }
  for (int i = 1; i <= n; i++)
    for (int j = 0; j <= 1; j++)
      for (int k = 0; k <= 1; k++)
        if (s[i][j] >= s[i - 1][k])
          dp[i][j] = min_(dp[i][j], dp[i - 1][k] + (long long)(j * val[i]));
  if (min_(dp[n][0], dp[n][1]) == inf)
    printf("-1");
  else
    printf("%lld", min_(dp[n][0], dp[n][1]));
  return 0;
}
