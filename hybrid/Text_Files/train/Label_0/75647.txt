n = int(input())
m = [0 for i in range(366)]
f = [0 for i in range(366)]
for i in range(n):
	(c, a, b) = input().split()
	a = int(a) - 1
	b = int(b) - 1
	if c == 'M':
		for i in range(a, b + 1):
			m[i] += 1
	else:
		for i in range(a, b + 1):
			f[i] += 1
an = 0
for i in range(366):
	an = max(an, min(f[i], m[i]) * 2)
print(an)
