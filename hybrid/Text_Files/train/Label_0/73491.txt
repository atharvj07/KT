n = int(input())
for i in range(n):
	s = input()
	(l, r, u, d) = (0, 0, 0, 0)
	for v in s:
		if v == 'L':
			l += 1
		if v == 'R':
			r += 1
		if v == 'U':
			u += 1
		if v == 'D':
			d += 1
	if l == 0 or r == 0:
		if u and d:
			print(2)
			print('UD')
		else:
			print(0)
	elif u == 0 or d == 0:
		if l and r:
			print(2)
			print('LR')
		else:
			print(0)
	else:
		x = min(l, r)
		y = min(u, d)
		f = 'R' * x + 'U' * y + 'L' * x + 'D' * y
		print(len(f))
		print(f)
