#include <bits/stdc++.h>
using namespace std;
const int maxn = 200001;
int n, m, k, x, s;
int a[maxn], b[maxn], c[maxn], d[maxn];
long long res;
void enter() {
  cin >> n >> m >> k >> x >> s;
  for (int i = 1; i <= m; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  for (int i = 1; i <= k; i++) cin >> c[i];
  for (int i = 1; i <= k; i++) cin >> d[i];
}
void solve() {
  res = (long long)n * x;
  a[0] = x;
  for (int i = 0; i <= m; i++)
    if (b[i] <= s) {
      int j = upper_bound(d + 1, d + k + 1, s - b[i]) - d - 1;
      res = min(res, (long long)(n - c[j]) * a[i]);
    }
  cout << res;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  enter();
  solve();
}
