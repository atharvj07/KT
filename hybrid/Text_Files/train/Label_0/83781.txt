import random
n = int(input())
V = [tuple(map(int, input().split())) for i in range(n)]
dist = lambda x, y: x * x + y * y
indices = sorted(((dist(*v), i) for (i, v) in enumerate(V)))
result = [0] * n
(vx, vy) = (0, 0)
for (d, i) in reversed(indices):
	(x, y) = V[i]
	(_, c) = min(((dist(vx + x, vy + y), 1), (dist(vx - x, vy - y), -1)))
	vx += c * x
	vy += c * y
	result[i] = c
LIMIT = 2.25 * 10 ** 12
while dist(vx, vy) > LIMIT:
	i = random.randrange(n)
	c = result[i]
	(x, y) = V[i]
	if dist(vx, vy) > dist(vx - 2 * c * x, vy - 2 * c * y):
		vx -= 2 * c * x
		vy -= 2 * c * y
		result[i] = -c
print(' '.join(map(str, result)))
