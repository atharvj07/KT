#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 10;
int q, l, r, k, dp[N];
int Dn[N][10];
int calc(int x) {
  int cur = 1;
  while (x > 0) {
    if (x % 10 != 0) cur *= (x % 10);
    x /= 10;
  }
  return cur;
}
int main() {
  for (int i = 1; i <= 9; i++) {
    dp[i] = i;
  }
  for (int i = 10; i <= 1e6; i++) {
    dp[i] = dp[calc(i)];
  }
  for (int i = 1; i <= 1e6; i++) {
    for (int j = 1; j <= 9; j++) {
      Dn[i][j] = Dn[i - 1][j];
    }
    Dn[i][dp[i]]++;
  }
  scanf("%d", &q);
  while (q--) {
    scanf("%d %d %d", &l, &r, &k);
    printf("%d\n", Dn[r][k] - Dn[l - 1][k]);
  }
  return 0;
}
