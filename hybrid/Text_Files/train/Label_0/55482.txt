import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class ProblemE {
    
    
    static class Edge {
        static int _id = 0;
        int a;
        int b;
        int id;
        Edge(String line) {
            String[] d = line.split(" ");
            id = _id++;
            a = Integer.valueOf(d[0])-1;
            b = Integer.valueOf(d[1])-1;
        }
    }
    
    static List<Edge>[] graph;
    static int[][] parent;
    static int[] depth;
    
    static void dfs(int now, int from, int dep) {
        parent[0][now] = from;
        depth[now] = dep;
        for (Edge e : graph[now]) {
            int to = (e.a == now) ? e.b : e.a;
            if (to != from) {
                dfs(to, now, dep+1);
            }
        }
    }
    
    static void init(int n) {
        int log = 1;
        int nn = n;
        while (nn >= 1) {
            nn /= 2;
            log++;
        }
        parent = new int[log+1][n];
        for (int i = 0 ; i <= log ; i++) {
            Arrays.fill(parent[i], -1);
        }
        depth = new int[n];
        
        dfs(0, -1, 0);
        
        for (int k = 0 ; k < log ; k++) {
            for (int v = 0 ; v < n ; v++) {
                if (parent[k][v] < 0) {
                    parent[k+1][v] = -1;
                } else {
                    parent[k+1][v] = parent[k][parent[k][v]];
                }
            }
        }
    }
    
    static int lca(int u, int v) {
        int loglen = parent.length;
        if (depth[u] > depth[v]) {
            int tmp = u;
            u = v;
            v = tmp;
        }
        for (int k = 0 ; k < loglen ; k++) {
            if (((depth[v] - depth[u]) >> k) % 2 == 1) {
                v = parent[k][v];
            }
        }
        if (u == v) {
            return u;
        }
        
        for (int k = loglen-1 ; k >= 0 ; k--) {
            if (parent[k][u] != parent[k][v]) {
                u = parent[k][u];
                v = parent[k][v];
            }
        }
        return parent[0][u];
    }
    
    
    static int[] value;
    static int[] edgeValue;
    
    static int dfs2(int now, int from) {
        Edge retedge = null;
        int val = 0;
        for (Edge e : graph[now]) {
            int to = (e.a == now) ? e.b : e.a;
            if (to != from) {
                val += dfs2(to, now);
            } else {
                retedge = e;
            }
        }
        if (retedge != null) {
            val += value[now];
            edgeValue[retedge.id] = val;
        }
        return val;
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader s = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);

        int n = Integer.valueOf(s.readLine());
        graph = new List[n];
        for (int i = 0 ; i < n ; i++) {
            graph[i] = new ArrayList<Edge>();
        }
        for (int i = 0 ; i < n-1 ; i++) {
            Edge e = new Edge(s.readLine());
            graph[e.a].add(e);
            graph[e.b].add(e);
        }
        init(n);

        
        
        int m = Integer.valueOf(s.readLine());
        value = new int[n];
        for (int i = 0 ; i < m ; i++) {
            String[] data = s.readLine().split(" ");
            int from = Integer.valueOf(data[0]) -1;
            int to = Integer.valueOf(data[1]) - 1;
            int lca = lca(from, to);
            value[from]++;
            value[to]++;
            value[lca]-=2;
        }
        
        edgeValue = new int[n-1];
        dfs2(0, -1);
        
        StringBuffer answer = new StringBuffer();
        for (int i = 0 ; i < n-1 ; i++) {
            answer.append(" ").append(edgeValue[i]);
        }
        out.println(answer.substring(1));
        out.flush();
    }

    public static void debug(Object... os){
        System.err.println(Arrays.deepToString(os));
    }
}
