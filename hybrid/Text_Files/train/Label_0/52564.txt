
import java.util.*;


public class D
{
   public static void main(String[] args)
   {
      new D(new Scanner(System.in));
   }

   public D(Scanner in)
   {
      int N = in.nextInt();
      int H = in.nextInt();
      
      long[] cat = new long[N+1];
      cat[0] = 1;
      for (int j=1; j<=N; j++)
      {
         long sum = 0;
         for (int i=0; i<j; i++)
            sum = sum+cat[j-i-1]*cat[i];
         cat[j] = sum;
      }
      
      long[][] dp = new long[N+1][H+1];
      for (int i=0; i<=H; i++)
         dp[0][i] = 1;
      for (int h=1; h<=H; h++)
      {
         for (int n=1; n<=N; n++)
         {
            long sum = 0;
            for (int i=0; i<n; i++)
               sum += dp[i][h-1]*dp[n-i-1][h-1];
            dp[n][h] = sum;
         }
      }

      if (H > 0)
         cat[N] -= dp[N][H-1];
      System.out.printf("%d%n", cat[N]);
   }
}
