def ii():
	return int(input())

def si():
	return input()

def mi():
	return map(int, input().split())

def li():
	return list(mi())
import math
for i in range(ii()):
	n = ii()
	a = li()
	l = []
	for i in range(n):
		if a[i] == -1:
			if i > 0 and a[i - 1] != -1:
				l.append(a[i - 1])
			if i < n - 1 and a[i + 1] != -1:
				l.append(a[i + 1])
	if len(l) == 0:
		print(0, 100)
		continue
	mx = max(l)
	mn = min(l)
	k = (mx + mn) // 2
	for i in range(n):
		if a[i] == -1:
			a[i] = k
	ans = -1000000000.0
	for i in range(1, n):
		ans = max(ans, abs(a[i] - a[i - 1]))
	print(ans, k)
