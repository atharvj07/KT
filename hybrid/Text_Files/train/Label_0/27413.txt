#include<iostream>
using namespace std;
long long int pow2(int i){
  if(i > 0){
    return ((long long int)1<<i);
  }else{
    return 1;
  }
}
int main(void){
  for(;;){
    long long int N;
    cin >> N;
    if(N == 0){return 0;}
    long long int ptr = 0;
    // 以下， N 番目の短歌数を T[N] とする．

    // Step 1
    // 10...0(d桁) <= T[N] < 100...0 (d+1 桁) を満たす d を求める．
    // bd桁の短歌数は (81*(pow2(bd-1) - 1))個
    int d;  bool notfound = true;
    for(int bd = 2 ; notfound ; bd++){
      if(ptr < N && N <= ptr + (81*(pow2(bd-1) - 1))){
	notfound = false;
	d = bd;
      }else{
	ptr += 81*(pow2(bd-1) - 1);
      }
    }

    // Step 2
    // T[N] = k?...? を満たす k を求める．
    // すなわち，k0...0(d桁) <= T[N] < (k+1)0...0 (d 個) を満たす k を求める．
    // k0...0 〜 k9...9 までの短歌数は　9*(pow2(d-1)-1)
    int k; notfound = true;
    for(int bk = 1 ; notfound ; bk++){
      if((ptr <= N && N <= ptr+9*(pow2(d-1)-1)) || bk == 9){
	notfound = false;
	k = bk;
      }else{
	ptr += 9*(pow2(d-1)-1);
      }
    }

    // Step 3
    // T[N] = k...k?...? (k が最初に f 個連続する) を満たす f を求める
    // kk...k00...0     〜 kk...k(k-1)k...k までは k*pow2(d-f-1)     個
    // kk...kk0...0     〜 kk...kk9...9     までは 9*(pow2(d-f-1)-1) 個 (1個引いているのは全部kの場合)
    // kk...k(k+1)k...k 〜 kk...k9...9      までは (9-k)*pow2(d-f-1) 個
    // 要するに，
    //   N < ptr + k*pow2(d-f),  ptr + k*pow2(d-f) + 9*(pow2(d-f)-1) <= N
    // を満たす f を求めれば良い．
    int f; notfound = true;
    for(int bf = 1 ; notfound ; bf++){
      if(N <= ptr + k*pow2(d-bf-1) ||  ptr + k*pow2(d-bf-1) + 9*(pow2(d-bf-1)-1) < N || d-bf == 1){
	notfound = false;
	f = bf;
      }else{
	ptr += k*pow2(d-bf-1);
      }
    }

    // Step 4
    // T[N] = k...kj?...? を満たすjを求める．
    // 要するに，
    //    k...kj0...0(k が最初に f 個) <= T[N] < k...k(j+1)0...0 (k が最初に f 個)
    // を満たす j を求める．ただし， j != k に注意
    // k...kj?...? は pow2(d-f-1) 個
    int j; notfound = true;
    for(int bj = 0 ; notfound && bj <= 9 ; bj++){
      if(bj != k){
	if(ptr < N && N <= ptr + pow2(d-f-1)){
	  notfound = false;
	  j = bj;
	}else{
	  ptr += pow2(d-f-1);
	}
      }else if (bj != 9){
	ptr += 9*(pow2(d-f-1)-1);
      }
    }
    
    // Step 5
    // ここまでで，使う2つの数は決まった．
    // k...kj(kとjの列)
    // m = min(k,j), M = max(k,j) とする．
    // あとは，k...kjm...m から gap = N - ptr - 1 個進めた数を求めれば良い．
    // これを求めるのは，gap の 2 進表記の 0 を m，1 を M に取り替えたものを，k...kj0...0 に足したものである．
    // なお，gap の 2 進表記は，d-f-1 桁である．
    long long int gap = N - ptr - 1;
    int M = (k<j)?j:k;
    int m = (k<j)?k:j;

    // 答えの出力
    for(int i = 0 ; i < f ; i++){
      cout << k;
    }
    cout << j;
    for(int i = 0 ; i < d-f-1 ; i++){
      if((gap>>(d-f-i-2))%2 == 0){
	cout << m;
      }else{
	cout << M;
      }
    }
    cout << endl;
  }
}
