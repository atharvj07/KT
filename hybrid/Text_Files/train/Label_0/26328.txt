#include <string>
#include <iostream>
#include <map>
#include <set>
using namespace std;

const int INF = 1<<29;

bool doit(){
	int N;
	map<string, int> points;
	map<int, set<string> > present;

	cin >> N;
	if(N == 0){
		return false;
	}

	for(int i=0; i<N; i++){
		string name;
		int M;
		cin.ignore();
		cin >> name >> M;

		points[name] = 0;

		for(int j=0, x; j<M; j++){
			cin >> x;
			present[x].insert(name);
		}
	}

	for(map<int, set<string> >::iterator itr=present.begin(); itr!=present.end(); itr++){
		int sz = itr->second.size();
		set<string>& ss = itr->second;
		for(set<string>::iterator jtr=ss.begin(); jtr!=ss.end(); jtr++){
			points[*jtr] += N - sz + 1;
		}
	}

	int mini = INF;
	string ans = "";
	for(map<string,int>::iterator itr=points.begin(); itr!=points.end(); itr++){
		if(itr->second <= mini){
			if(itr->second < mini|| itr->first < ans){
				ans = itr->first;
				mini = itr->second;
			}
		}
	}

	cout << mini << " " << ans << endl;

	return true;
}

int main(){
	while(doit());

	return 0;
}
