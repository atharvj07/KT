from math import gcd
(n, k) = map(int, input().split())
(a, b) = map(int, input().split())
nk = n * k
cand = []
for i in range(n):
	cand.append(gcd(nk, abs(i * k + b - a)))
	cand.append(gcd(nk, abs(i * k - b - a)))
	cand.append(gcd(nk, abs(i * k + b + a)))
	cand.append(gcd(nk, abs(i * k - b + a)))
print(nk // max(cand), nk // min(cand))
