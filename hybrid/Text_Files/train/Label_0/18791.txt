#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, o[70], gr[N], pos;
long long b[N];
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) cin >> b[i];
  for (int i = 1; i <= n; ++i) {
    int c = 0;
    while (!(b[i] & 1)) b[i] >>= 1, ++c;
    gr[i] = c, ++o[c], b[i] <<= c;
  }
  for (int i = 1; i < 70; ++i)
    if (o[i] > o[pos]) pos = i;
  printf("%d\n", n - o[pos]);
  for (int i = 1; i <= n; ++i)
    if (gr[i] != pos) cout << b[i] << " ";
  return 0;
}
