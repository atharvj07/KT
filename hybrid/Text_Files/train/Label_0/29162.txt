#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  int x, y;
  cin >> x >> y;
  string s;
  cin >> s;
  bool flag = false;
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (s[i] == '0') {
      flag = true;
    } else if (flag) {
      ++cnt;
      flag = false;
    }
  }
  if (flag) ++cnt;
  long long ans = 1LL * (cnt - 1) * min(x, y) + y;
  if (cnt == 0) ans = 0;
  cout << ans << '\n';
  return 0;
}
