#include <bits/stdc++.h>
using namespace std;
template <class t>
using vec = vector<t>;
using ll = unsigned long long;
const int MAX = numeric_limits<int>::max();
template <class t>
ostream &operator<<(ostream &out, vec<t> &v) {
  for (t e : v) cout << e << " ";
  out << "\n";
  return out;
}
template <class t>
istream &operator>>(istream &in, vec<t> &v) {
  for (t &e : v) in >> e;
  return in;
}
struct node {
  int id, parent, child_cnt;
  bool belongs_to_f;
  vec<int> cards;
  node(int id, int parent, bool belongs_to_f)
      : id(id), parent(parent), belongs_to_f(belongs_to_f), child_cnt(0) {}
  void print() {
    cout << id << " " << parent << " " << child_cnt << " " << belongs_to_f
         << "\n";
  }
};
int main() {
  int n;
  cin >> n;
  vec<int> f(3 * n);
  vec<bool> belongs_to_f(6 * n + 1);
  cin >> f;
  for (int i = 0; i < f.size(); i++) belongs_to_f[f[i]] = true;
  vec<node *> nodes(2 * n);
  stack<node *> st;
  for (int i = 1, id = 0; i <= 6 * n; i++) {
    if (st.size()) {
      if (st.top()->belongs_to_f == belongs_to_f[i]) {
        st.top()->cards.push_back(i);
        if (st.top()->cards.size() == 3) st.pop();
      } else {
        nodes[id] = new node(id, st.top()->id, belongs_to_f[i]);
        nodes[id]->cards.push_back(i);
        st.top()->child_cnt++;
        st.push(nodes[id]);
        id++;
      }
    } else {
      nodes[id] = new node(id, -1, belongs_to_f[i]);
      nodes[id]->cards.push_back(i);
      st.push(nodes[id]);
      id++;
    }
  }
  set<pair<int, int>> nds_f, nds_nf;
  for (auto nd : nodes)
    if (nd->child_cnt == 0)
      if (nd->belongs_to_f)
        nds_f.insert({(nd->parent == -1 ? 1 : 0), nd->id});
      else
        nds_nf.insert({(nd->parent == -1 ? 1 : 0), nd->id});
  for (int next_node, p_node, i = 1; nds_f.size() or nds_nf.size(); i ^= 1) {
    if (i) {
      next_node = nds_f.begin()->second;
      nds_f.erase(nds_f.begin());
    } else {
      next_node = nds_nf.begin()->second;
      nds_nf.erase(nds_nf.begin());
    }
    cout << nodes[next_node]->cards;
    p_node = nodes[next_node]->parent;
    if (p_node != -1) {
      nodes[p_node]->child_cnt--;
      if (!nodes[p_node]->child_cnt)
        if (nodes[p_node]->belongs_to_f)
          nds_f.insert(
              {(nodes[p_node]->parent == -1 ? 1 : 0), nodes[p_node]->id});
        else
          nds_nf.insert(
              {(nodes[p_node]->parent == -1 ? 1 : 0), nodes[p_node]->id});
    }
  }
  return 0;
}
