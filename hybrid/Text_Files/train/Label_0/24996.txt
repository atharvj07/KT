#include <bits/stdc++.h>
using namespace std;
typedef class Input {
 protected:
  const static int limit = 65536;
  FILE* file;
  int ss, st;
  char buf[limit];

 public:
  Input() : file(NULL){};
  Input(FILE* file) : file(file) {}
  void open(FILE* file) { this->file = file; }
  void open(const char* filename) { file = fopen(filename, "r"); }
  char pick() {
    if (ss == st) st = fread(buf, 1, limit, file), ss = 0;
    return buf[ss++];
  }
} Input;
Input& operator>>(Input& in, unsigned& u) {
  char x;
  while (~(x = in.pick()) && !((x) >= '0' && (x) <= '9'))
    ;
  for (u = x - '0'; ~(x = in.pick()) && ((x) >= '0' && (x) <= '9');
       u = u * 10 + x - '0')
    ;
  return in;
}

}
Input& operator>>(Input& in, int& u) {
  char x;
  while (~(x = in.pick()) && !((x) >= '0' && (x) <= '9') && x != '-')
    ;
  int aflag = ((x == '-') ? (x = in.pick(), -1) : (1));
  for (u = x - '0'; ~(x = in.pick()) && ((x) >= '0' && (x) <= '9');
       u = u * 10 + x - '0')
    ;
  u *= aflag;
  return in;
}

void exgcd(int a, int b, int& x, int& y) {
  if (!b) {
    x = 1, y = 0;
  } else {
    exgcd(b, a % b, y, x);
    y -= (a / b) * x;
  }
}
int inv(int a, int n) {
  int x, y;
  exgcd(a, n, x, y);
  return (x < 0) ? (x + n) : (x);
}
const int Mod = 1e9 + 7;
template <const int Mod = ::Mod>
class Z {
 public:
  int v;
  Z() : v(0) {}
  Z(int x) : v(x) {}
  Z(long long x) : v(x % Mod) {}
  friend Z operator+(const Z& a, const Z& b) {
    int x;
    return Z(((x = a.v + b.v) >= Mod) ? (x - Mod) : (x));
  }
  friend Z operator-(const Z& a, const Z& b) {
    int x;
    return Z(((x = a.v - b.v) < 0) ? (x + Mod) : (x));
  }
  friend Z operator*(const Z& a, const Z& b) { return Z(a.v * 1ll * b.v); }
  friend Z operator~(const Z& a) { return inv(a.v, Mod); }
  friend Z operator-(const Z& a) { return Z(0) - a; }
  Z& operator+=(Z b) { return *this = *this + b; }
  Z& operator-=(Z b) { return *this = *this - b; }
  Z& operator*=(Z b) { return *this = *this * b; }
  friend bool operator==(const Z& a, const Z& b) { return a.v == b.v; }
};
Z<> qpow(Z<> a, int p) {
  Z<> rt = Z<>(1), pa = a;
  for (; p; p >>= 1, pa = pa * pa) {
    if (p & 1) {
      rt = rt * pa;
    }
  }
  return rt;
}
vector<Z<> > operator+(vector<Z<> > A, const vector<Z<> >& B) {
  if (A.size() < B.size()) A.resize(B.size());
  for (int i = 0; i < (signed)B.size(); i++) A[i] += B[i];
  return A;
}
vector<Z<> > operator*(const vector<Z<> >& A, const vector<Z<> >& B) {
  int n = A.size(), m = B.size();
  vector<Z<> > rt(n + m - 1, Z<>(0));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      rt[i + j] += A[i] * B[j];
    }
  }
  return rt;
}
vector<Z<> > operator%(vector<Z<> > A, const vector<Z<> >& B) {
  int n = A.size(), m = B.size();
  if (n < m) {
    return A;
  }
  assert(m > 0);
  Z<> iv = ~B.back();
  for (int i = n - 1; i >= m - 1; i--) {
    Z<> x = A[i] * iv;
    for (int j = 1; j <= m; j++) {
      A[i - j + 1] -= B[m - j] * x;
    }
  }
  A.resize(m - 1);
  return A;
}
vector<Z<> > mod_pow(vector<Z<> > A, long long p, vector<Z<> > Q) {
  vector<Z<> > pa = A, rt = vector<Z<> >{1};
  for (; p; p >>= 1, pa = pa * pa % Q) {
    if (p & 1) {
      rt = rt * pa % Q;
    }
  }
  return rt;
}
