#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC optimize("Ofast")
using namespace std;
long long n;
queue<long long> q[2][30];
queue<long long> cnt1;
queue<long long> cnt2;
string l;
string r;
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> n;
  vector<pair<long long, long long>> ans;
  cin >> l >> r;
  for (long long i = 0; i < l.size(); i++) {
    if (l[i] == '?') {
      cnt1.push(i);
    } else {
      q[0][l[i] - 'a'].push(i);
    }
  }
  for (long long i = 0; i < r.size(); i++) {
    if (r[i] == '?') {
      cnt2.push(i);
    } else {
      q[1][r[i] - 'a'].push(i);
    }
  }
  for (long long i = 0; i < 27; i++) {
    while (!q[0][i].empty() && !q[1][i].empty()) {
      ans.push_back(make_pair(q[0][i].front(), q[1][i].front()));
      q[0][i].pop();
      q[1][i].pop();
    }
  }
  for (long long i = 0; i < 27; i++) {
    while (!q[0][i].empty() && !cnt2.empty()) {
      ans.push_back(make_pair(q[0][i].front(), cnt2.front()));
      q[0][i].pop();
      cnt2.pop();
    }
    while (!q[1][i].empty() && !cnt1.empty()) {
      ans.push_back(make_pair(cnt1.front(), q[1][i].front()));
      q[1][i].pop();
      cnt1.pop();
    }
  }
  while (!cnt1.empty() && !cnt2.empty()) {
    ans.push_back(make_pair(cnt1.front(), cnt2.front()));
    cnt1.pop();
    cnt2.pop();
  }
  cout << ans.size() << '\n';
  for (long long i = 0; i < ans.size(); i++) {
    cout << ans[i].first + 1 << ' ' << ans[i].second + 1 << '\n';
  }
}
