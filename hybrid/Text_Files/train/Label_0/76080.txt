s = input()
L = len(s)
lst = []
i = 0
while i < L:
	if s[i] == 'v':
		while i + 1 < L and s[i + 1] == 'v':
			lst.append('w')
			i += 1
	else:
		lst.append('o')
	i += 1
count = 0
for x in lst:
	if x == 'w':
		count += 1
prev = 0
total = 0
for x in lst:
	if x == 'w':
		prev += 1
	else:
		total += (count - prev) * prev
print(total)
