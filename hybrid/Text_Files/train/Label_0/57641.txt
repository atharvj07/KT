import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {

        List<List<String>> listList = new ArrayList<>();
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        if (m <3 || n < 3){
            System.out.println("NO");
            return;
        }
        for (int i = 0; i < n; i++) {
            List<String> list = new ArrayList<>();
            list.clear();
            String string = scanner.next();
            String[] stringsplit = string.split("");
            for (int j = 0; j < m; j++) {
                list.add(stringsplit[j]);
            }
            listList.add(list);
        }


        int i = 0;
        int j = 0;

        boolean h = false;

        int p0 = -1;
        int q0 = -1;

        while (i < n && !h){
            j = 0;
            while (j < m && !h){
                if (listList.get(i).get(j).equals("*")){
                    h = true;
                    p0 = i;
                    q0 = j;
                }
                j++;
            }
            i++;
        }



        // ASLAN MARKAZ RO ESHTEBAH PEIDA MIKONE
        i = p0;
        j = q0;

        if (q0 == 0){
            System.out.println("NO");
            return;
        }

        while ( p0 < n && p0 >= 0 && !listList.get(p0).get(q0-1).equals("*") )
            p0++;

        if (p0 == n || p0 == -1){
            System.out.println("NO");
            return;
        }


        if (p0 == 0 || q0 == 0 || p0 == n-1 || q0==m-1 || listList.get(p0).get(q0-1).equals(".") || listList.get(p0).get(q0+1).equals(".") || listList.get(p0-1).get(q0).equals(".") || listList.get(p0+1).get(q0).equals(".")){
            System.out.println("NO");
            return;
        }

        int t = 0;
        int d =0;

        //System.out.println(listList.get(2).get(2));

        int k = 0;
        int l = 0;

        while (k < n){
            l = 0;
            if (k == p0){

                while(listList.get(p0).get(l).equals(".")){
                    l++;
                }
                while (l < m  && listList.get(p0).get(l).equals("*")){
                    l++;
                }
                while(l <m && listList.get(p0).get(l).equals(".")){
                    l++;
                }
                if (l != m) {
                    System.out.println("NO");
                    return;
                }

            }else {

                while (l < m) {
                    if (l != q0 && listList.get(k).get(l).equals("*")) {
                        System.out.println("NO");
                        //System.out.println(k);
                        //System.out.println(l);
                        //System.out.println(listList.get(k).get(l));
                        return;
                    }

                    int prev = 0;
                    int forw = n-1;
                    if (k-1 > 0){
                        prev = k - 1;
                    }
                    if (k+1 < n-1){
                        forw = k + 1;
                    }
                    if (l == q0 && listList.get(k).get(l).equals(".") && listList.get(prev).get(l).equals("*") && listList.get(forw).get(l).equals("*"))
                        t++;


                    l++;
                }
            }
            k++;
        }

        if (t == 0){
            System.out.println("YES");
        }else{
            System.out.println("NO");
        }

    }
}
