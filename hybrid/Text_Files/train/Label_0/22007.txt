#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  int arr[202];
  cin >> n >> k;
  n *= 2;
  n++;
  for (int q = 0; q < n; q++) {
    cin >> arr[q];
  }
  for (int q = 0; q < n; q++) {
    if (k == 0) break;
    if (q % 2 == 1)
      if ((arr[q] - 1 > arr[q - 1]) && (arr[q] - 1 > arr[q + 1])) {
        arr[q] -= 1;
        k--;
      }
  }
  for (int q = 0; q < n; q++) cout << arr[q] << " ";
  return 0;
}
