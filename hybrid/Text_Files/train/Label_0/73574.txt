n = int(input())
deg = [0] * 100005
leaves = []
for i in range(1, n):
	(a, b) = map(int, input().split())
	deg[a] += 1
	deg[b] += 1
cnt = 0
mxdeg = 0
root = 0
for j in range(1, n + 1):
	if deg[j] > mxdeg:
		mxdeg = deg[j]
		root = j
	if deg[j] == 1:
		leaves.append(j)
	if deg[j] > 2:
		cnt += 1
if cnt > 1:
	print('No')
	exit()
print('Yes')
m = 0
for it in leaves:
	if it != root:
		m += 1
print(m)
for it2 in leaves:
	if it2 != root:
		print(root, it2)
