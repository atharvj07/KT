#include <iostream>
using namespace std;

const int MAX = 10020;
int d[MAX + 5];
int a[MAX + 5];

void div(int n, int *d) {
	int val = 1;
	for (int i = 0; i < MAX; i++) {
		d[i] = val / n;
		val %= n;
		val *= 10;
	}
}

void add(int *src, int *dst) {
	for (int i = MAX - 1; i >= 0; i--) {
		dst[i] += src[i];
	}
	for (int i = MAX - 1; i > 0; i--) {
		dst[i - 1] += dst[i] / 10;
		dst[i] %= 10;
	}
}

void fix(int *tmp, int *a) {
	for (int i = 0; i < MAX; i++) tmp[i] = 0;
	tmp[MAX - 5] = 1;
	add(tmp, a);
}

int main() {
	int n, k, m, r;
	while (cin >> n >> k >> m >> r) {
		if (!n) break;
		for (int i = 0; i < MAX; i++) a[i] = 0;
		
		//add 1/n
		div(n, d);
		add(d, a);
		
		if (m == 1) {
			for (int i = n - 1; i > 0; i--) {
				div(n * i, d);
				add(d, a);
			}
		}
		
		//fix(d, a);
		
		cout << a[0] << ".";
		for (int i = 1; i <= r; i++) cout << a[i];
		cout << endl;
	}
	return 0;
}
