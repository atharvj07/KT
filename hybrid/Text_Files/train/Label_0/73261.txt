for __ in range(int(input())):
	(n, m) = [int(x) for x in input().split()]
	a = [int(x) for x in input().split()]
	b = [int(x) for x in input().split()]
	m1 = {}
	m2 = {}
	for i in range(0, n - 1):
		for j in range(0, n - i - 1):
			if a[j] > a[j + 1]:
				(a[j], a[j + 1]) = (a[j + 1], a[j])
				m1[j] = 1
	for x in b:
		m2[x - 1] = 1
	f = 0
	for x in m1:
		if x in m2:
			pass
		else:
			f = 1
			break
	if f is 1:
		print('NO')
	else:
		print('YES')
