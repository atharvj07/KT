/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        MyScan in = new MyScan(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskE solver = new TaskE();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskE {
        public void solve(int testNumber, MyScan in, PrintWriter out) {
            String s = in.next();

            TaskE.Z x = parse(s);
            int pl = in.nextInt();
            int mm = in.nextInt();
            if (pl < mm) {
                x.ct(0);
                out.println(x.max(pl));
            } else {
                x.ct(1);
                out.println(x.max(mm));
            }


        }

        private TaskE.Z parse(String s) {
            if (s.startsWith("(")) {

                int x = 0;
                for (int t = 1; t < s.length(); t++) {
                    if (s.charAt(t) == '(') x++;
                    if (s.charAt(t) == ')') x--;
                    if (s.charAt(t) == '?' && x == 0) {
                        return new TaskE.X(parse(s.substring(1, t)), parse(s.substring(t + 1, s.length() - 1)));
                    }
                }

            } else {
                return new TaskE.Y(s);
            }
            return null;
        }

        private static class X implements TaskE.Z {
            int pl = 0;
            long[] max;
            long[] min;
            TaskE.Z p1;
            TaskE.Z p2;

            public X(TaskE.Z p1, TaskE.Z p2) {
                this.p1 = p1;
                this.p2 = p2;
            }


            public void ct(int i) {
                p1.ct(i);
                p2.ct(i);
                pl = Math.min(1 + p1.pl() + p2.pl(), 102);
                if (i == 0) {
                    max = new long[pl + 1];
                    min = new long[pl + 1];
                    Arrays.fill(max, Long.MIN_VALUE);
                    Arrays.fill(min, Long.MAX_VALUE);

                    for (int s = 1; s < pl + 1; s++) {
                        for (int t = 0; t <= Math.min(s - 1, p1.pl()); t++) {
                            if (s - t - 1 > p2.pl()) continue;
                            max[s] = Math.max(max[s], p1.max(t) + p2.max(s - t - 1));
                            min[s] = Math.min(min[s], p1.min(t) + p2.min(s - t - 1));
                        }
                    }
                    for (int s = 0; s < pl; s++) {
                        for (int t = 0; t <= Math.min(s, p1.pl()); t++) {
                            if (s - t > p2.pl()) continue;
                            max[s] = Math.max(max[s], p1.max(t) - p2.min(s - t));
                            min[s] = Math.min(min[s], p1.min(t) - p2.max(s - t));
                        }
                    }
                } else {
                    max = new long[pl + 1];
                    min = new long[pl + 1];
                    Arrays.fill(max, Long.MIN_VALUE);
                    Arrays.fill(min, Long.MAX_VALUE);

                    for (int s = 1; s < pl + 1; s++) {
                        for (int t = 0; t <= Math.min(s - 1, p1.pl()); t++) {
                            if (s - t - 1 > p2.pl()) continue;
                            max[s] = Math.max(max[s], p1.max(t) - p2.min(s - t - 1));
                            min[s] = Math.min(min[s], p1.min(t) - p2.max(s - t - 1));
                        }
                    }
                    for (int s = 0; s < pl; s++) {
                        for (int t = 0; t <= Math.min(s, p1.pl()); t++) {
                            if (s - t > p2.pl()) continue;
                            max[s] = Math.max(max[s], p1.max(t) + p2.max(s - t));
                            min[s] = Math.min(min[s], p1.min(t) + p2.min(s - t));
                        }
                    }
                }


            }

            public long max(int s1) {
                return max[s1];
            }

            public long min(int s1) {
                return min[s1];
            }


            public int pl() {
                return pl;
            }

        }

        private interface Z {
            void ct(int i);

            long max(int i);

            long min(int i);

            int pl();

        }

        private static class Y implements TaskE.Z {
            private final int v;

            public Y(String s) {
                v = Integer.parseInt(s);
            }


            public void ct(int i) {

            }


            public long max(int i) {
                return v;
            }


            public long min(int i) {
                return v;
            }


            public int pl() {
                return 0;
            }

        }

    }

    static class MyScan {
        BufferedReader br;
        StringTokenizer st;

        MyScan(BufferedReader br) {
            this.br = br;
        }

        public MyScan(InputStream in) {
            this(new BufferedReader(new InputStreamReader(in)));
        }

        public void findToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
        }

        public String next() {
            findToken();
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
