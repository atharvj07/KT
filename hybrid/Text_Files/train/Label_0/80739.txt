import sys
input = sys.stdin.readline
from operator import itemgetter
t = int(input())
for tests in range(t):
	n = int(input())
	W = list(map(int, input().split()))
	UV = [tuple(map(int, input().split())) for i in range(n - 1)]
	DEG = [0] * n
	for (x, y) in UV:
		DEG[x - 1] += 1
		DEG[y - 1] += 1
	ANS = [sum(W)]
	W = [(W[i], i) for i in range(n)]
	W.sort(key=itemgetter(0), reverse=True)
	ind = 0
	for i in range(n - 2):
		while DEG[W[ind][1]] == 1:
			ind += 1
		ANS.append(ANS[-1] + W[ind][0])
		DEG[W[ind][1]] -= 1
	print(*ANS)
