#include <bits/stdc++.h>
using namespace std;
vector<int> v;
int k;
int checks(int x) {
  int cnt = 1;
  int j = 0;
  int f = 0;
  for (int i = 0; i < v.size(); i++) {
    if (v[i] > x) {
      f = 1;
      break;
    }
    if (j + v[i] <= x) {
      j += v[i];
    } else {
      cnt++;
      j = v[i];
    }
  }
  if (f == 1) return 0;
  if (cnt <= k)
    return 1;
  else
    return 0;
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> k;
  string s;
  getline(cin, s);
  getline(cin, s);
  int j = 0;
  for (int i = 0; i < s.size(); i++) {
    j++;
    if (s[i] == '-' || s[i] == ' ') {
      v.push_back(j);
      j = 0;
    }
  }
  if (j != 0) v.push_back(j);
  int l = 1;
  int r = 1e6;
  while (l <= r) {
    if (l == r) {
      int f = checks(l);
      int f1 = checks(l + 1);
      if (f == 0 && f1 == 1) {
        cout << l + 1 << "\n";
      } else if (f == 1 && f1 == 1) {
        cout << l << "\n";
      }
      return 0;
    } else {
      int mid = (l + r) / 2;
      int f = checks(mid);
      int f1 = checks(mid + 1);
      if (f == 0 && f1 == 1) {
        cout << mid + 1;
        return 0;
      } else if (f == 0 && f1 == 0) {
        l = mid + 1;
      } else if (f == 1 && f1 == 1) {
        r = mid;
      }
    }
  }
  return 0;
}
