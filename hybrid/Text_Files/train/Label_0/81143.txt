t = int(input())
for i in range(t):
	(n, x) = map(int, input().split())
	even = odd = 0
	arr = list(map(int, input().split()))
	for i in range(len(arr)):
		if arr[i] % 2 == 0:
			even += 1
		else:
			odd += 1
	i = 1
	flag = False
	while i <= odd and i <= x:
		have = even
		need = x - i
		if even >= need:
			flag = True
		i += 2
	if flag:
		print('Yes')
	else:
		print('No')
