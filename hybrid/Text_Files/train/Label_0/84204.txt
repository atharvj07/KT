for _ in range(int(input())):
	n = int(input())
	arr = []
	for i in range(n):
		(a, b) = list(map(int, input().split()))
		arr.append([a, b])
	arr.sort()
	f = 0
	for i in range(1, n):
		if arr[i][0] < arr[i - 1][0]:
			f = 1
			break
		if arr[i][1] < arr[i - 1][1]:
			f = 1
			break
	if f == 1:
		print('NO')
	else:
		print('YES')
		ans = ''
		(cx, cy) = (0, 0)
		for i in arr:
			(x, y) = (i[0], i[1])
			ans += 'R' * abs(x - cx)
			ans += 'U' * abs(y - cy)
			(cx, cy) = (x, y)
		print(ans)
