def main():
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	mods = set()
	flag = False
	for num in a:
		if mods:
			for j in mods.copy():
				mods.add((num + j) % m)
		mods.add(num % m)
		if 0 in mods:
			flag = True
			break
	if flag:
		print('YES')
	else:
		print('NO')
	return
main()
