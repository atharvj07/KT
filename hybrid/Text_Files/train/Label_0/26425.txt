#include <bits/stdc++.h>
using namespace std;
inline void upmax(int &x, int b) {
  if (b > x) x = b;
}
inline void upmin(int &x, int b) {
  if (b < x) x = b;
}
inline void read(int &x) {
  x = 0;
  char c = getchar();
  for (; c < '0' || '9' < c; c = getchar())
    ;
  for (; '0' <= c && c <= '9'; c = getchar()) x = 10 * x + c - '0';
}
const int N = 100005;
int n, m, root;
struct data {
  int l, r;
  data(int _l, int _r) : l(_l), r(_r) {}
};
vector<data> e;
struct node {
  int next, to, pos;
} edge[2 * N];
int first[N];
int tot;
inline void tjb(int a, int b, int c) {
  edge[++tot].to = b, edge[tot].pos = c;
  edge[tot].next = first[a], first[a] = tot;
}
int tree[4 * N];
int c[N];
inline void build(int h, int l, int r) {
  if (l == r) {
    tree[h] = c[l];
    return;
  }
  int mid = (l + r) >> 1;
  build(h << 1, l, mid);
  build(h << 1 | 1, mid + 1, r);
  tree[h] = tree[h << 1] & tree[h << 1 | 1];
}
inline void change(int h, int l, int r, int x) {
  if (l == r) {
    tree[h] = c[l];
    return;
  }
  int mid = (l + r) >> 1;
  if (x <= mid)
    change(h << 1, l, mid, x);
  else
    change(h << 1 | 1, mid + 1, r, x);
  tree[h] = tree[h << 1] & tree[h << 1 | 1];
}
inline int find(int h, int l, int r, int ll, int rr) {
  if (ll <= l && r <= rr) return tree[h];
  int mid = (l + r) >> 1;
  if (rr <= mid) return find(h << 1, l, mid, ll, rr);
  if (mid < ll) return find(h << 1 | 1, mid + 1, r, ll, rr);
  if (find(h << 1, l, mid, ll, rr) && find(h << 1 | 1, mid + 1, r, ll, rr))
    return true;
  return false;
}
int a[N], deep[N], size[N], ad[N], g[N], place[N], fa[N], dxd[N];
bool flag[N];
inline void dfs1(int x, int dep) {
  flag[x] = 1;
  deep[x] = dep;
  size[x] = 1;
  for (int k = first[x]; k; k = edge[k].next)
    if (!flag[edge[k].to]) {
      dxd[edge[k].pos] = edge[k].to;
      dfs1(edge[k].to, dep + 1);
      size[x] += size[edge[k].to];
    } else
      fa[x] = edge[k].to;
  for (int k = first[x]; k; k = edge[k].next)
    if (edge[k].to != fa[x])
      if (size[ad[x]] < size[edge[k].to]) ad[x] = edge[k].to;
}
int times = 0;
inline void dfs2(int x, int fro) {
  flag[x] = 1;
  g[x] = ++times;
  c[times] = a[x];
  if (ad[x])
    dfs2(ad[x], fro), place[x] = place[ad[x]];
  else
    e.push_back(data(fro, x)), place[x] = e.size() - 1;
  for (int k = first[x]; k; k = edge[k].next)
    if (!flag[edge[k].to]) dfs2(edge[k].to, edge[k].to);
}
inline int Get_it(int x, int y) {
  int dx, dy;
  for (dx = e[place[x]].l, dy = e[place[y]].l; place[x] != place[y];)
    if (deep[dx] >= deep[dy]) {
      if (!find(1, 1, n, g[dx], g[x])) return -1;
      x = fa[dx], dx = e[place[x]].l;
    } else {
      if (!find(1, 1, n, g[dy], g[y])) return -1;
      y = fa[dy], dy = e[place[x]].l;
    }
  if (x == y) return x;
  if (g[x] > g[y]) swap(x, y);
  if (!find(1, 1, n, g[ad[x]], g[y])) return -1;
  return x;
}
int main() {
  read(n);
  int u, v;
  for (int i = 1; i <= n - 1; ++i) {
    read(u);
    read(v);
    tjb(u, v, i);
    tjb(v, u, i);
  }
  root = rand() % n + 1;
  for (int i = 1; i <= n; ++i) a[i] = 1;
  a[root] = 0;
  dfs1(root, 0);
  memset(flag, 0, sizeof(flag));
  dfs2(root, root);
  build(1, 1, n);
  int m, a, b, d;
  read(m);
  while (m--) {
    read(a);
    if (a == 3) {
      read(u);
      read(v);
      d = Get_it(u, v);
      if (d == -1)
        printf("-1\n");
      else
        printf("%d\n", deep[u] + deep[v] - deep[d] - deep[d]);
    } else {
      read(b);
      c[g[dxd[b]]] = 2 - a;
      change(1, 1, n, g[dxd[b]]);
    }
  }
  return 0;
}
