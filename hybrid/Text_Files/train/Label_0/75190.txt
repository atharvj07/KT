MOD = 10 ** 9 + 7
INV = 500000004
(n, k) = map(int, input().split())
d = 1
p = pow(2, n, MOD)
ans = 0
stirling2_list = [0, 1]
for i in range(k - 1):
	stirling2_list = [0, 1] + [(stirling2_list[j - 1] + j * stirling2_list[j]) % MOD for j in range(2, 2 + i)] + [1]
for stirling in stirling2_list:
	ans = (ans + stirling * d * p) % MOD
	d = d * n % MOD
	n -= 1
	p = p * INV % MOD
print(ans)
