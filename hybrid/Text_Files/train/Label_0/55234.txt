    static void init1(int n) {
        groups = n;

    static int root(int p) {
        while (p != fa[p]) {
            fa[p] = fa[fa[p]];
            p = fa[p];
        }
        return p;
    }

    static void combine(int p, int q) {
        int i = root(p);
        int j = root(q);
        if (i == j) {
            return;
        }
        if (sz[i] < sz[j]) {
            fa[i] = j;
            sz[j] += sz[i];
        } else {
            fa[j] = i;
            sz[i] += sz[j];
        }
        groups--;
    }

    public static int eq_bigger(int[] a, int lo, int hi,
                                int key) {
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (a[mid] < key) {
                lo = mid + 1;
            }else {
                hi = mid;
            }
        }
        return lo;
    }

            }else {
                lo = mid+1;
            }
        }
        return lo;
    }

    static class S{
        int l = 0;
        int r = 0 ;

            this.l = l;

    static void init(int[] f){
        o = f;
        int len = o.length;
        a = new S[len*4];
        build(1,0,len-1);
    }

    static void build(int num,int l,int r){
        S cur = new S(l,r);
        if(l==r){
            a[num] = cur;
            a[num].all = o[l];
            a[num].idx = l;
            return;
        }else{
            int m = (l+r)>>1;
            int le = num<<1;
            int ri = le|1;
            build(le, l,m);
            build(ri, m+1,r);
            a[num] = cur;
            pushup(num, le, ri);
        }
    }



    static int[] query(int num,int l,int r){

        if(a[num].l>=l&&a[num].r<=r){
            return new int[]{a[num].idx,(int)a[num].all};
        }else{
            int m = (a[num].l+a[num].r)>>1;
            int le = num<<1;
            int ri = le|1;
            //   pushdown(num, le, ri);
            int ma = -1;
            int idx = 100000001;
            if(l<=m) {
                int r1[] = query(le, l, r);
                if(r1[1]>ma){
                    ma = r1[1];
                    idx = r1[0];
                }

            }
            if(r>m){
                int r2[] = query(ri, l, r);
                if(r2[1]>ma){
                    ma = r2[1];
                    idx = r2[0];
                }
            }
            return new int[]{idx,ma};
        }
    }


    static void update(int num,int l,long v){
        if(a[num].l==a[num].r){
            a[num].all = v;
        }else{
            int m = (a[num].l+a[num].r)>>1;
            int le = num<<1;
            int ri = le|1;
            pushdown(num, le, ri);
            if(l<=m){
                update(le,l,v);
            }
            if(l>m){
                update(ri,l,v);
            }
            pushup(num,le,ri);
        }
    }

    static void pushup(int num,int le,int ri){
        a[num].all = Math.max(a[le].all,a[ri].all);
        if(a[le].all>a[ri].all){
            a[num].idx = a[le].idx;
        }else{
            a[num].idx = a[ri].idx;
        }

    }
