from pprint import pprint
n = int(input())
data = list(map(int, input().split()))
datb = list(map(int, input().split()))
datc = [data[x] - datb[x] for x in range(n)]
datc.sort()
import bisect
res = 0
for i in range(n - 1, -1, -1):
	x = datc[i]
	if x <= 0:
		break
	c = bisect.bisect_right(datc, -x)
	res += i - c
print(res)
