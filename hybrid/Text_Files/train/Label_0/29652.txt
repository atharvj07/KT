#include <bits/stdc++.h>
using namespace std;
long long t, n;
int main() {
  cin >> t;
  while (t--) {
    scanf("%lld", &n);
    if (n == 1)
      cout << -1 << endl;
    else {
      for (int i = 0; i < n / 2; i++) printf("58");
      if (n % 2) printf("8");
      cout << endl;
    }
  }
  return 0;
}
