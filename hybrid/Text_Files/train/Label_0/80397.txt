import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
	(n, m) = map(int, input().split())
	answer = 0
	if n > m:
		print(0)
	else:
		nbin = bin(n)[2:]
		mbin = bin(m)[2:]
		nlen = len(nbin)
		mlen = len(mbin)
		nbin = '0' * (mlen - nlen) + nbin
		data = ['0'] * mlen
		flg = False
		for i in range(mlen):
			if nbin[i] == '0' and mbin[i] == '1':
				data[i] = '1'
			elif nbin[i] == '1' and mbin[i] == '0':
				data[i] = '0'
				flg = True
				break
		if not flg:
			flg2 = -1
			for i in range(mlen - 1, -1, -1):
				if nbin[i] == '0' and mbin[i] == '0':
					data[i] = '1'
					flg2 = i
					break
			if flg2 >= 0:
				for i in range(flg2 + 1, mlen):
					data[i] = '0'
			else:
				data = ['1'] + ['0'] * mlen
			answer = int(''.join(data), 2)
			print(answer)
		else:
			answer = int(''.join(data), 2)
			print(answer)
