(n, m) = map(int, input().split())
a = list(map(int, input().split()))
k = n // m
mod = n % m
d = [0] * (m + 1)
R = []
mod1 = 0
for i in range(n):
	if a[i] > m:
		if mod1 < mod:
			mod1 += 1
		else:
			R.append(i)
	elif d[a[i]] == k:
		if mod1 < mod:
			d[a[i]] += 1
			mod1 += 1
		else:
			R.append(i)
	elif d[a[i]] == k + 1:
		R.append(i)
	else:
		d[a[i]] += 1
ans = [-1] * n
for i in range(1, m + 1):
	while d[i] < k:
		j = R.pop(0)
		ans[j] = i
		d[i] += 1
num = 0
for j in range(n):
	if ans[j] == -1:
		ans[j] = a[j]
	else:
		num += 1
print(k, num)
print(*ans)
