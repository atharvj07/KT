#include <bits/stdc++.h>
using namespace std;
const int Z = (int)3e3 + 111;
const int N = (int)3e5 + 111;
const int INF = (int)1e9;
const int MOD = (int)1e9 + 7;
int a[N];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  string s;
  cin >> s;
  int l = -INF, r = INF;
  for (int i = 4; i < n; i++) {
    if (s[i - 1] == '0' && s[i - 2] == '0' && s[i - 3] == '0' &&
        s[i - 4] == '0')
      if (s[i] == '1')
        l = max(l, 1 + max(a[i - 1],
                           max(a[i - 2], max(a[i - 3], max(a[i - 4], a[i])))));
    if (s[i - 1] == '1' && s[i - 2] == '1' && s[i - 3] == '1' &&
        s[i - 4] == '1')
      if (s[i] == '0')
        r = min(r, min(a[i - 1],
                       min(a[i - 2], min(a[i - 3], min(a[i - 4], a[i])))) -
                       1);
  }
  cout << l << ' ' << r;
  return 0;
}
