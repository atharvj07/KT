n = int(input())
l1 = []
l2 = []
for i in range(n):
	(x, a) = map(int, input().split())
	if x > 0:
		l1.append((x, a))
	else:
		l2.append((x, a))
l1 = sorted(l1)
l2 = sorted(l2, reverse=True)
if len(l1) == len(l2):
	res = 0
	for i in range(len(l1)):
		res = res + l1[i][1] + l2[i][1]
elif len(l1) > len(l2):
	res = 0
	for i in range(len(l2)):
		res = res + l1[i][1] + l2[i][1]
	res = res + l1[len(l2)][1]
else:
	res = 0
	for i in range(len(l1)):
		res = res + l1[i][1] + l2[i][1]
	res = res + l2[len(l1)][1]
print(res)
