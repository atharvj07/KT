import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[][] array = new int[n][2];
        for (int i = 0; i < n; i++){
            for(int j = 0; j < 2; j++)
                array[i][j] = scan.nextInt();
        }
        int[] order = new int[n];
        int second = array[0][0], third = array[0][1];
        if (array[third - 1][0] == second || array[third - 1][1] == second){
            int temp = second;
            second = third;
            third = temp;
        }
        order[0] = 1; order[1] = second; order[2] = third;
        for (int i = 3; i < n; i++){
            int neighborOne = array[order[i - 2] - 1][0], neighborTwo = array[order[i - 2] - 1][1];
            order[i] = neighborTwo;
            if (neighborOne != order[i - 1])
                order[i] = neighborOne;
        }
        for (int x: order)
            System.out.print(x + " ");
    }
}
