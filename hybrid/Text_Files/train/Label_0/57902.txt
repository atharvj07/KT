import java.util.Scanner;

public class A {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int prev=0, cur=0, time=0;

        for(int i=0; i<n+1; i++) {
            if(i!=n) cur=sc.nextInt();
            else cur=90;
            if(cur-prev <= 15) {
                time+=cur-prev;
            }
            else {
                time+=15;
                break;
            }
            prev=cur;
        }


        System.out.println(time);
    }
}
