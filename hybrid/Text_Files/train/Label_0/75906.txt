import sys
inf = 10 ** 9 + 7

def solve():
	n = int(sys.stdin.readline())
	v = [int(vi) for vi in sys.stdin.readline().split()]
	p = [int(pi) for pi in sys.stdin.readline().split()]
	cnt = [0] * 5
	for i in range(5):
		if v[i] != -1:
			cnt[i] += 1
		if p[i] != -1:
			cnt[i] += 1
	for i in range(n - 2):
		a = [int(ai) for ai in sys.stdin.readline().split()]
		for j in range(5):
			if a[j] != -1:
				cnt[j] += 1
	for i in range(10 ** 5):
		if check(n, v, p, cnt, i):
			print(i)
			return
	print(-1)

def check(n, v, p, cnt, m):
	tot = n + m
	solved = cnt[:]
	dif = 0
	for i in range(5):
		if p[i] != -1 and v[i] > p[i]:
			solved[i] += m
	for i in range(5):
		if solved[i] * 2 > tot:
			max_score = 500
		elif solved[i] * 4 > tot:
			max_score = 1000
		elif solved[i] * 8 > tot:
			max_score = 1500
		elif solved[i] * 16 > tot:
			max_score = 2000
		elif solved[i] * 32 > tot:
			max_score = 2500
		else:
			max_score = 3000
		if v[i] == p[i] == -1:
			pass
		elif v[i] == -1:
			dif -= max_score * (250 - p[i]) // 250
		elif p[i] == -1:
			dif += max_score * (250 - v[i]) // 250
		else:
			dif += max_score * (p[i] - v[i]) // 250
	return dif > 0
solve()
