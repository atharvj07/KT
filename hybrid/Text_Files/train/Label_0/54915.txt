import java.util.*;
public class strangetable
{
   public static void main(String[] args)
   {
      long t,n,m,x,r,c;;

      Scanner sc=new Scanner(System.in);

      t=sc.nextInt();

      while(t-->0)
      {
         n=sc.nextLong();
         m=sc.nextLong();
         x=sc.nextLong();

         x--;

         r=x%n;
         c=x/n;

	 System.out.println(r*m+c+1);
      }
   }
}
