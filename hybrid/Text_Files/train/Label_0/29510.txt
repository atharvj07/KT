#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
int dp[410], v[410], w[410], f[410][410], h[410][410], g[410][410];
int n;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &v[i]);
  for (int i = 1; i <= n; i++) scanf("%d", &w[i]);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i == j)
        h[i][i] = g[i][i] = 0, f[i][i] = v[1];
      else if (i < j)
        f[i][j] = h[i][j] = g[i][j] = -INF;
      else
        f[i][j] = 0;
  for (int len = 2; len <= n; len++)
    for (int i = 1; i <= n - len + 1; i++) {
      int j = i + len - 1;
      for (int k = i + 1; k <= j; k++) {
        if (w[k] - w[i] == 1 && h[k][j] != -INF)
          h[i][j] = max(h[i][j], f[i + 1][k - 1] + h[k][j]);
        if (w[k] - w[i] == -1 && g[k][j] != -INF)
          g[i][j] = max(g[i][j], f[i + 1][k - 1] + g[k][j]);
        f[i][j] = max(f[i][j], f[i][k - 1] + f[k][j]);
      }
      for (int k = i; k <= j; k++)
        if (h[i][k] != -INF && g[k][j] != -INF)
          f[i][j] =
              max(f[i][j], h[i][k] + g[k][j] + v[2 * w[k] - w[i] - w[j] + 1]);
    }
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1];
    for (int j = 1; j <= i; j++) dp[i] = max(dp[i], dp[j - 1] + f[j][i]);
  }
  printf("%d\n", dp[n]);
  return 0;
}
