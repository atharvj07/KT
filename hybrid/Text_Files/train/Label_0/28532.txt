#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long n;
  cin >> n;
  if (!n) {
    cout << 1;
    exit(0);
  }
  int ans = 4, p = 1;
  for (long long i = n - 1; i; i--) {
    int x = sqrt(n * n - i * i);
    x = 2 * x + 1;
    ans += 2 * (max(2, x - p));
    p = x;
  }
  cout << ans;
  return 0;
}
