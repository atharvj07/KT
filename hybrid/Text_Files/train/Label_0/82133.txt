for _ in range(int(input())):
	n = int(input())
	ls = list(map(int, input().split()))
	mx = -1
	mn = 10000000000
	a1 = -1
	a2 = -1
	a3 = -1
	f = False
	for i in range(n):
		if ls[i] < mx and a1 != a2 != -1 and (a2 > a1):
			a3 = i + 1
			print('YES')
			print(a1, a2, a3)
			f = True
			break
		if ls[i] < mn:
			mn = ls[i]
			mx = -1
			a1 = i + 1
		if ls[i] > mn:
			mx = max(mx, ls[i])
			a2 = i + 1
	if not f:
		print('NO')
