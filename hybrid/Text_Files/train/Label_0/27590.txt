#include <bits/stdc++.h>
using namespace std;
const int maxn = 100100;
long long a[maxn], b[maxn];
int main() {
  int n1, n2, y;
  scanf("%d%d", &n1, &y);
  for (int i = 1; i <= n1; i++) {
    scanf("%I64d", a + i);
  }
  scanf("%d%d", &n2, &y);
  for (int i = 1; i <= n2; i++) {
    scanf("%I64d", b + i);
  }
  int ans = 0;
  for (long long i = 1; i <= (1LL << 30); i <<= 1) {
    map<long long, int> mp;
    for (int j = 1; j <= n1; j++) {
      long long x = a[j] % (i << 1);
      if (mp.count(x) == 0) mp[x] = 0;
      mp[x]++;
    }
    for (int j = 1; j <= n2; j++) {
      long long x = (b[j] + i) % (i << 1);
      if (mp.count(x) == 0) mp[x] = 0;
      mp[x]++;
    }
    for (auto &p : mp) ans = max(ans, p.second);
  }
  if (ans == 1) ans++;
  printf("%d\n", ans);
  return 0;
}
