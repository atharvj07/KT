import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class practice2 {
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
    public static FastReader scn = new FastReader();
    static StringBuilder res;

    public static void main(String[] args) {
        int t = scn.nextInt();
        res = new StringBuilder();
        while(t-- > 0)
            run();
        System.out.println(res);
    }


    private static void run() {
        int n=scn.nextInt();
        Long a[]=new Long[n];

        HashMap<Long,Long> hm=new HashMap<Long,Long>();

        for(int i=0;i<n;i++)
        {
            a[i]=scn.nextLong();
            if(hm.containsKey(a[i]))
                hm.put(a[i], hm.get(a[i])+1);
            else
                hm.put(a[i], (long)1);
        }

        Arrays.sort(a);

        long max=a[n-1];
        int cnt=0;
        long sum=max;

        long tempo=0;

        for(int i=a.length-2;i>=0;i--)
        {
            if(cnt==0)
            {
                if(max%a[i]!=0)
                {sum+=a[i]; tempo=a[i]; cnt++;}
            }
            else if(cnt==1)
            {
                if(max%a[i]!=0 && tempo%a[i]!=0)
                {sum+=a[i];cnt++;}
            }
            else
                break;
        }

        long ans=sum;

        if(max%2==0 && max%3==0 && max%5==0)
        {
            if(hm.containsKey(max/2) && hm.containsKey(max/3) && hm.containsKey(max/5))
                ans=Math.max(ans,max/2+max/3+max/5);
        }

        res.append(ans+"\n");

    }

}
