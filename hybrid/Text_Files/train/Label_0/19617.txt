#include<iostream>
#include<string>
#include<map>
#define rep(i,n) for(int i=0;i<n;i++)
using namespace std;
string h[]={"Rock","Fire","Scissors","Snake","Human","Tree",
	"Wolf","Sponge","Paper","Air","Water","Dragon","Devil","Lightning","Gun"};
map<string,int> m;int n,t,c;
int main(){
	rep(i,15)m[h[i]]=i;
	while(cin>>n,n){
		int lose[15]={0},exist[15]={0};
		string buf;
		rep(i,n){
			cin>>buf;t=m[buf];
			exist[t]=1;
			rep(j,7)lose[(t+j+1)%15]=1;
		}
		t=-1;c=0;
		rep(i,15){
			c+=exist[i];
			if(!lose[i]&&exist[i])t=t==-1?i:-2;
		}
		cout<<(c<2||t<0?"Draw":h[t])<<endl;
	}
	return 0;
}
