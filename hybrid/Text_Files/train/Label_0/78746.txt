from sys import stdin
input = stdin.readline
q = int(input())

def cyk(l1, l2, l3, len1, len2, len3):
	wynik = 65872346587623475642374982383247
	i1 = 0
	i3 = 0
	for i2 in range(len2):
		while True:
			dupa = 0
			if i1 < len1 - 1 and l1[i1 + 1] <= l2[i2]:
				i1 += 1
				dupa += 1
			if i3 > 0 and l3[i3 - 1] >= l2[i2]:
				i3 -= 1
				dupa += 1
			if i3 < len3 - 1 and l3[i3] < l2[i2]:
				i3 += 1
				dupa += 1
			if dupa == 0:
				break
		if l1[i1] <= l2[i2] and l2[i2] <= l3[i3]:
			wynik = min(wynik, (l1[i1] - l2[i2]) ** 2 + (l2[i2] - l3[i3]) ** 2 + (l1[i1] - l3[i3]) ** 2)
	return wynik
for _ in range(q):
	(r, g, b) = map(int, input().split())
	lr = list(map(int, input().split()))
	lg = list(map(int, input().split()))
	lb = list(map(int, input().split()))
	lr.sort()
	lg.sort()
	lb.sort()
	lr.append(2349349237489237498233)
	lg.append(32964892374982374892384234234234234329)
	lb.append(23964892364892346264873242423423423423423423423423423787878878)
	ler = len(lr)
	leg = len(lg)
	leb = len(lb)
	c1 = cyk(lr, lg, lb, ler, leg, leb)
	c2 = cyk(lg, lr, lb, leg, ler, leb)
	c3 = cyk(lr, lb, lg, ler, leb, leg)
	c4 = cyk(lg, lb, lr, leg, leb, ler)
	c5 = cyk(lb, lr, lg, leb, ler, leg)
	c6 = cyk(lb, lg, lr, leb, leg, ler)
	print(min(c1, c2, c3, c4, c5, c6))
