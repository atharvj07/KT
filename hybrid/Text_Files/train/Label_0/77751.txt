def pursuit(n, a, b):
	k = n
	calc = k - k // 4
	a.sort(reverse=True)
	b.sort(reverse=True)
	scoreOfA = sum(a[:calc])
	scoreOfB = sum(b[:calc])
	if scoreOfA >= scoreOfB:
		return 0
	bl = calc
	al = calc - 1
	while scoreOfA < scoreOfB:
		k += 1
		if k % 4 == 0:
			scoreOfA += 100 - a[al]
			al -= 1
		else:
			scoreOfA += 100
			if bl >= n:
				pass
			else:
				scoreOfB += b[bl]
				bl += 1
	return k - n
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	print(pursuit(n, a, b))
