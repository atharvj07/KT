n = int(input())
a = [int(s) for s in input().split(' ')]
bitmap = [[] for _ in range(33)]
for (i, b) in enumerate(a):
	b_bin = bin(b)[2:]
	for (j, b_str) in enumerate(b_bin[::-1]):
		if b_str == '1':
			bitmap[j].append(i)
index = -1
for i in range(32, -1, -1):
	if bitmap[i].__len__() == 1:
		index = bitmap[i][0]
		break
if index != -1:
	print(a[index], end=' ')
	for i in range(n):
		if i != index:
			print(a[i], end=' ')
else:
	for i in range(n):
		print(a[i], end=' ')
