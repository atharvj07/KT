def isPrime(n):
	if n <= 1:
		return False
	if n <= 3:
		return True
	if n % 2 == 0 or n % 3 == 0:
		return False
	i = 5
	while i * i <= n:
		if n % i == 0 or n % (i + 2) == 0:
			return False
		i = i + 6
	return True
n = int(input())
a = 5
b = n + 5
k = 1
while k:
	if isPrime(a):
		a += 1
		b += 1
	elif isPrime(b):
		a += 1
		b += 1
	else:
		k = 0
print(str(b) + ' ' + str(a))
