template <class T>
T next(istream& in) {
  T ret;
  in >> ret;
  return ret;
}
template <class T>
vector<T> next_vec(istream& in, size_t n) {
  vector<T> ret(n);
  for (size_t i = 0; i < n; ++i) {
    ret[i] = next<T>(in);
  }
  return ret;
}
template <typename T, typename C, T Modulus>
class MR {
 private:
  struct tag_minus {};
  static_assert(std::numeric_limits<C>::max() / Modulus / Modulus > 0,
                "compute type is too small");
  static_assert(Modulus < std::numeric_limits<T>::max() / 2,
                "storage type is too small");
  static constexpr T reduce(T value, tag_minus) {
    return value < 0 ? value + Modulus : value;
  }

 public:
  T value;
  constexpr MR(C value) : value(reduce(value % Modulus, tag_minus())) {}
  MR& operator*=(MR b) {
    value = C(value) * C(b.value) % Modulus;
    return *this;
  }
  constexpr T get() const { return value; }
};
template <typename T, typename C, T Modulus>
static inline std::ostream& operator<<(std::ostream& o, MR<T, C, Modulus> mr) {
  return o << mr.get();
}
void solve(istream& in, ostream& out) {
  auto n = next<int>(in);
  auto m = next<int>(in);
  auto a = next_vec<int>(in, n);
  auto b = next_vec<int>(in, m);
  unordered_map<int, int> lastPos;
  lastPos.reserve(n);
  lastPos.max_load_factor(0.25);
  for (int i = 0; i < (n); ++i) {
    if (a[i] < b[0]) {
      out << 0 << "\n";
      return;
    }
    lastPos[a[i]] = i;
  }
  if (m > n) {
    out << 0 << "\n";
    return;
  }
  vec<int> positions;
  positions.reserve(m);
  vec<int> firstPos;
  firstPos.reserve(m);
  firstPos.push_back(0);
  for (int i = 0; i < (m); ++i) {
    auto it = lastPos.find(b[i]);
    if (it == lastPos.end()) {
      out << 0 << "\n";
      return;
    }
    if (i) {
      if (it->second < lastPos[b[i - 1]]) {
        out << 0 << "\n";
        return;
      }
      for (int j = it->second - 1; j > lastPos[b[i - 1]]; --j) {
        if (a[j] < b[i - 1]) {
          out << 0 << "\n";
          return;
        }
      }
    }
    if (!positions.empty()) {
      if (positions.back() > it->second) {
        out << 0 << "\n";
        return;
      }
      int fp = positions.back() + 1;
      for (int j = positions.back(); j < it->second; ++j) {
        if (a[j] < a[it->second]) {
          fp = j + 1;
        }
      }
      firstPos.push_back(fp);
    }
    positions.push_back(it->second);
  }
  for (int i = positions.back(); i < n; ++i) {
    if (a[i] < a[positions.back()]) {
      out << 0 << "\n";
      return;
    }
  }
  MR<int, int64_t, 998244353> res = 1;
  for (int i = (1); i < (positions.size()); ++i) {
    int64_t s = firstPos[i];
    int64_t t = positions[i];
    res *= MR<int, int64_t, 998244353>(t - s + 1);
  }
  out << res << "\n";
}
