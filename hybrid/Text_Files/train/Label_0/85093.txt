n = int(input())
b = list(input())
suma = list((0 for i in range(n + 1)))
equal = list(([] for i in range(2 * (n + 1))))
for i in range(n):
	b[i] = int(b[i])
	if b[i] == 0:
		b[i] = -1
equal[n + 1] += [0]
for i in range(1, n + 1):
	suma[i] = suma[i - 1] + b[i - 1]
	equal[n + 1 + suma[i]] += [i]
l = 0
for i in range(0, 2 * (n + 1)):
	if len(equal[i]) > 1:
		l = equal[i][-1] - equal[i][0] if equal[i][-1] - equal[i][0] > l else l
print(l)
