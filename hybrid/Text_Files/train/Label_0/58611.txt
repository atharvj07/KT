import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class CE37C
{
	public static boolean[] notPass;
	
	public static void main(String[] args) throws IOException
	{
		BufferedReader read = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(read.readLine());
		int[] arr = new int[n];
		StringTokenizer st = new StringTokenizer(read.readLine());
		for (int i = 0; i < n; i++)
			arr[i] = Integer.parseInt(st.nextToken());
		
		notPass = new boolean[n * 4];
		char[] eh = read.readLine().toCharArray();
		for (int i = 0; i < eh.length; i++) {
			if (eh[i] == '1')
				continue;
			changeFalse(0, 0, n - 1, i);
		}
		
		for (int i = 0; i < n; i++) {
			if (arr[i] == i + 1)
				continue;
			int ind = arr[i] - 1;
			if (cantPass(0, 0, n - 1, Math.min(i, ind), Math.max(i, ind) - 1)) {
				System.out.println("NO");
				System.exit(0);
			}
		}
		System.out.println("YES");
		
		read.close();
	}
	
	public static boolean cantPass(int i, int l, int r, int tl, int tr)
	{
		if (l >= tl && r <= tr)
			return notPass[i];
		boolean nah = false;
		int mid = (l + r) / 2;
		if (tl <= mid && cantPass(i * 2 + 1, l, mid, tl, tr))
			nah = true;
		if (tr > mid && cantPass(i * 2 + 2, mid + 1, r, tl, tr))
			nah = true;
		return nah;
	}
	
	public static void changeFalse(int i, int l, int r, int rip)
	{
		notPass[i] = true;
		if (l == r)
			return;
		int mid = (l + r) / 2;
		if (rip <= mid)
			changeFalse(i * 2 + 1, l, mid, rip);
		else
			changeFalse(i * 2 + 2, mid + 1, r, rip);
	}
}
