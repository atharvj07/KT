def read_int():
	return int(input().strip())

def read_int_array():
	return [int(i) for i in input().strip().split(' ')]
n = read_int()
M = []
for i in range(n):
	M += [read_int_array()]
a2 = int((M[1][2] * M[0][1] / M[0][2]) ** 0.5)
a1 = int(M[0][1] / a2)
A = [a1, a2]
for i in range(2, n):
	A += [int(M[0][i] / a1)]
print(' '.join((str(i) for i in A)))
