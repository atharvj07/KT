n = int(input())
p = list(map(int, input().split()))
b = list(map(int, input().split()))
s = sum((b[i] for i in range(n)))
if s % 2 == 0:
	ans = 1
else:
	ans = 0
visited = [0] * n
ptr = 0
start = 1
visited[0] = 1
q = 1
c = 1
while q < n:
	start = p[start - 1]
	if visited[start - 1] == 1:
		c += 1
		while ptr < n and visited[ptr] == 1:
			ptr += 1
		start = ptr + 1
	else:
		visited[start - 1] = 1
		q += 1
if c == 1:
	print(ans)
else:
	print(ans + c)
