for i in range(0, 2 ** m):
	a = p[:]
	binary1 = list(bin(i)[2:])
	if len(binary1) < q:
		binary1 = ['0'] * (q - len(binary1)) + binary1
	l = 0
	for (k, j) in zip(a, binary1):

			a[l] = -1 * a[l]
