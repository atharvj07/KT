#include <bits/stdc++.h>
using namespace std;
const int N = 4010;
const double eps = 1e-4;
const double inf = (double)1000000000;
int n, m;
double R;
struct point {
  int x, y, t;
} a[N];
point operator-(const point &a, const point &b) {
  return (point){a.x - b.x, a.y - b.y};
}
int operator*(const point &a, const point &b) { return a.x * b.x + a.y * b.y; }
double dis(const point &a, const point &b) { return sqrt((a - b) * (a - b)); }
struct ch {
  double a;
  int t, d;
} c[N];
bool operator<(const ch &a, const ch &b) { return a.a < b.a; }
const double pi = acos(-1.0);
double calc(double a) {
  while (a < 0) a += 2 * pi;
  while (a >= 2 * pi) a -= 2 * pi;
  return a;
}
bool check(int x) {
  int cnt[2] = {0, 0}, cl = 1;
  c[1] = (ch){0, 0, 0};
  for (int i = 1; i <= n + m; i++)
    if (i != x) {
      point tmp = a[i] - a[x];
      double L = dis(a[x], a[i]);
      if (L > 2 * R) continue;
      double arc = acos((L / 2) / R);
      if (a[i].t == 1) arc -= 1e-30;
      double alpha = atan2(tmp.y, tmp.x);
      double l = alpha - arc, r = alpha + arc;
      l = calc(l), r = calc(r);
      if (l > r) cnt[a[i].t]++;
      c[++cl] = (ch){l, a[i].t, 1}, c[++cl] = (ch){r, a[i].t, -1};
    }
  sort(c + 1, c + cl + 1);
  for (int i = 1; i <= cl; i++) {
    cnt[c[i].t] += c[i].d;
    if ((cnt[0] || a[x].t == 0) && !cnt[1]) return true;
  }
  return false;
}
int main() {
  srand(233);
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%d%d", &a[i].x, &a[i].y), a[i].t = 0;
  for (int i = 1; i <= m; i++)
    scanf("%d%d", &a[n + i].x, &a[n + i].y), a[n + i].t = 1;
  for (int i = 1; i <= n + m; i++) swap(a[i], a[rand() % i + 1]);
  double ans = 1;
  double l, r;
  for (int i = 1; i <= n + m; i++)
    if (a[i].t == 0) {
      R = ans;
      if (!check(i)) continue;
      R *= 2;
      while (check(i) && R <= inf) R *= 2;
      l = R / 2, r = R;
      while (((r - l) < 0 ? -(r - l) : (r - l)) / max(1., l) > eps) {
        double mid = (l + r) / 2;
        R = mid;
        if (check(i))
          l = mid;
        else
          r = mid;
      }
      ans = (l + r) / 2;
    }
  for (int i = 1; i <= n + m; i++)
    if (a[i].t == 1) {
      R = ans;
      if (!check(i)) continue;
      R *= 2;
      while (check(i) && R <= inf) R *= 2;
      l = R / 2, r = R;
      while (((r - l) < 0 ? -(r - l) : (r - l)) / max(1., l) > eps) {
        double mid = (l + r) / 2;
        R = mid;
        if (check(i))
          l = mid;
        else
          r = mid;
      }
      ans = (l + r) / 2;
    }
  if (r >= inf / 2)
    printf("-1\n");
  else
    printf("%.7lf\n", (l + r) / 2.0);
  return 0;
}
