import java.util.Scanner;

public class Chocolate {
    static String chocolatePieces;
    static int n;

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        n = s.nextInt();
        s.nextLine();
        chocolatePieces = s.nextLine().replaceAll(" ", "");
        System.out.println(getNumberOfWaysToBreakChocolate());

    }

    private static long getNumberOfWaysToBreakChocolate() {
        if (chocolatePieces.indexOf('0') == -1)
            return 1;
        if (chocolatePieces.indexOf('1') == chocolatePieces.lastIndexOf('1')) {
            return chocolatePieces.indexOf('1') == -1 ? 0 : 1;
        }
        long total = 1;
        long counter = 0;
        for (int i = 0; i < n; i++) {
            if (chocolatePieces.charAt(i) == '1') {
                if (chocolatePieces.lastIndexOf('1') == i) continue;
                ++counter;
                int j = i + 1;
                while (j < chocolatePieces.length() && chocolatePieces.charAt(j++) != '1') {
                    ++counter;
                }
                total *= counter;
                counter = 0;
            }
        }
        return total;

    }
}
