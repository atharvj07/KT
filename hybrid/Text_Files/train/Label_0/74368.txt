zones = int(input())
people = list(map(int, input().split()))
(s, f) = map(int, input().split())
gap = f - 1 - s
sums = [sum(people[:gap + 1])]
times = []
people += people
for i in range(1, zones):
	sums.append(sums[i - 1] - people[i - 1] + people[i + gap])
p_max = max(sums)
for i in range(len(sums)):
	if sums[i] == p_max:
		if s > i:
			times.append(s - i)
		else:
			times.append(zones + s - i)
print(min(times))
