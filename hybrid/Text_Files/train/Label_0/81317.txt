t = int(input())
for _ in range(t):
	(n, k, z) = map(int, input().split())
	a = list(map(int, input().split()))
	dp = [[[0] * 2 for i in range(z + 1)] for i in range(k + 1)]
	dp[0][0][0] = a[0]
	for i in range(k):
		for times in range(z + 1):
			pos = i - 2 * times
			if pos >= 0:
				dp[i + 1][times][0] = max(dp[i][times][0] + a[pos + 1], dp[i + 1][times][0])
				dp[i + 1][times][0] = max(dp[i][times][1] + a[pos + 1], dp[i + 1][times][0])
			if times + 1 < z + 1 and pos >= 1:
				dp[i + 1][times + 1][1] = max(dp[i][times][0] + a[pos - 1], dp[i + 1][times + 1][1])
	ans = 0
	for arr in dp[-1]:
		for val in arr:
			ans = max(val, ans)
	print(ans)
