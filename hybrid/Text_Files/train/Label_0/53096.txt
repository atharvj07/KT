import java.util.Scanner;

public class P768B
{
	public static void main(String[] args)
	{
		Scanner scan = new Scanner(System.in);
		long n = scan.nextLong();
		long l = scan.nextLong();
		long r = scan.nextLong();
		long len = getLength(n);
		long ans = total(n, 1, len, l, r);
		System.out.println(ans);
	}

	private static long getLength(long n)
	{
		long len = 1;
		while (n > 0)
		{
			n /= 2;
			len *= 2;
		}
		return len-1;
	}

	private static long total(long n, long s, long e, long l, long r)
	{
		if (l > e || r < s)
			return 0;
		if (s == e)
			return n == 1 ? 1 : 0;
		long len = getLength(n);
		long a1 = total(n/2, s, s+len/2-1, l, r);
		long a2 = l <= s+len/2 && s+len/2 <= r ? n%2 : 0;
		long a3 = total(n/2, s+len/2+1, s+len-1, l, r);
		return a1+a2+a3;
	}
}
