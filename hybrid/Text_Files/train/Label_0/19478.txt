typedef long long int ll;
typedef unsigned long long ull;

mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
ll myRand(ll B) {
    return (ull)rng() % B;
}

int main(){
	cin.tie(nullptr);
	ios::sync_with_stdio(false);
	int n,m; cin >> n >> m;
	vector<int> a(n+2);
	while(m--){
		int x,y; cin >> x >> y;
		a[x]++;
		a[y]--;
	}
	for(int i=1;i<=n;i++){
		a[i]+=a[i-1];
	}
	int res=n+1;
	for(int i=1;i<=n;i++){
		if(a[i]>=1)res+=2;
	}
	printf("%d\n",res);
}
