def main():
	n = int(input())
	p = list(map(int, input().split()))
	board = [[0 for _ in range(n)] for _ in range(n)]
	for i in range(n):
		board[i][i] = p[i]
		px = i
		py = i
		for j in range(p[i] - 1):
			if py == 0:
				px += 1
			elif board[px][py - 1] == 0:
				py -= 1
			else:
				px += 1
			if px >= n or py < 0:
				print(-1)
				return
			board[px][py] = p[i]
	for i in range(n):
		print(' '.join([str(x) for x in board[i][0:i + 1]]))
main()
