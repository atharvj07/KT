#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int n, k, sum = 0;
  cin >> n >> k;
  long long int d = 240 - k;
  for (long long int i = 1; i <= n; i++) {
    if (d - 5 * i >= 0) {
      sum++;
      d = d - 5 * i;
    } else
      break;
  }
  cout << sum;
}
