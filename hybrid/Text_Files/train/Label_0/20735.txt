#include<iostream>
#include<vector>
#include<algorithm>
#include<map>
#include<math.h>
#include<complex>
#include<queue>

using namespace std;
#define rep(i,n) for(int i = 0 ; i < n ; ++i)
using ll = long long;
const ll inf = 1ll<<32;

int n,m;
vector<ll> a,b;

vector<ll> solve(int k, int l, int r){
    int nxt = *upper_bound(a.begin(),a.end(),l);
    vector<ll> ret(n+1,inf);
    if(nxt<r){
        auto vl = solve(k+1,l,(l+r)/2);
        auto vr = solve(k+1,(l+r)/2,r);
        rep(i,n+1){
            ret[i]=min(ret[i],vl[i]+vr[k]);
            ret[i]=min(ret[i],vl[k]+vr[i]);
        }
    }
    else {
        int idx = upper_bound(a.begin(),a.end(),l) - a.begin();
        --idx;
        int cur = b[idx];
        ll res = 0;
        int sz = 0;
        while((1ll<<sz)<r-l){
            if(k+sz!=cur)res += 1ll<<sz;
            ++sz;
        }
        rep(i,n+1){
            ret[i]=res;
            if(cur!=i)++ret[i];
        }
    }
    return ret;
}
 
int main(){
    cin>>n>>m;
    a.resize(m+1);
    b.resize(m);
    rep(i,m+1){
        cin>>a[i];
    }
    rep(i,m)cin>>b[i];
    auto ans = solve(1,0,1<<n);
    cout<<ans[0]<<endl;
}
