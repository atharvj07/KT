#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[5][5];
  int b[5][5];
  for (int i = 1; i <= 3; i++) {
    for (int j = 1; j <= 3; j++) {
      scanf("%d", &a[i][j]);
      b[i][j] = 1;
    }
  }
  for (int i = 1; i <= 3; i++) {
    for (int j = 1; j <= 3; j++) {
      if (a[i][j] % 2 == 1) {
        b[i][j] = -b[i][j];
        b[i - 1][j] = -b[i - 1][j];
        b[i + 1][j] = -b[i + 1][j];
        b[i][j - 1] = -b[i][j - 1];
        b[i][j + 1] = -b[i][j + 1];
      }
    }
  }
  for (int i = 1; i <= 3; i++) {
    for (int j = 1; j <= 3; j++) {
      if (b[i][j] == 1)
        cout << 1;
      else
        cout << 0;
    }
    cout << endl;
  }
  return 0;
}
