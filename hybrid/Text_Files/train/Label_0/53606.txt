import java.util.Scanner;
public class Solution {
    static long solve1(long n)
    {
        long a=1;
        while(n>=a)a*=4;
        a/=4;
        n-=a;
        n/=3;
        return a+n;
    }
    static long quad_search(long l,long r,long n)
    {
        if(l==r)return l;
        long cnt=(r-l+1)/4;
        long part=n/cnt;
        if(part==0)return quad_search(l,l+cnt-1,n);
        else if(part==3)return quad_search(l+cnt,l+2*cnt-1,n-part*cnt);
        else if(part==1)return quad_search(l+2*cnt,l+3*cnt-1,n-part*cnt);
        return quad_search(l+3*cnt,r,n-part*cnt);
    }
    static long solve2(long n)
    {
        long a=1;
        while(n>=a)a*=4;
        n-=a/4;
        n/=3;
        long l=a/2,r=a/2+a/4-1;
        return quad_search(l,r,n);
    }
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        long t=sc.nextLong();
        while(t-->0)
        {
            long n=sc.nextLong();
            if(n%3==1)System.out.println(solve1(n));
            else if(n%3==2)System.out.println(solve2(n));
            else System.out.println(solve1(n-2)^solve2(n-1));
        }
    }
}
