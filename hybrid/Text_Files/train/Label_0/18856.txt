#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.0);
const double eps = 1e-8;
const int mod = 1e9 + 7;
const int inf = 1061109567;
const int dir[][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
struct node {
  char c;
  int x, y;
  node(char c, int x, int y) : c(c), x(x), y(y) {}
};
int n, m;
vector<node> ans;
char s[505][505];
int vis[505][505], vis1[505][505];
int check(int x, int y) {
  if (x >= 0 && x < n && y >= 0 && y < m && s[x][y] == '.') return 1;
  return 0;
}
pair<int, int> dfs1(int x, int y) {
  int tmp = 1;
  vis1[x][y] = 1;
  for (int i = 0; i < 4; i++) {
    int tmpx = x + dir[i][0];
    int tmpy = y + dir[i][1];
    if (check(tmpx, tmpy) && !vis1[tmpx][tmpy]) {
      return dfs1(tmpx, tmpy);
    }
  }
  return make_pair(x, y);
}
void dfs(int x, int y, int flag) {
  vis[x][y] = 1;
  ans.push_back(node('B', x + 1, y + 1));
  for (int i = 0; i < 4; i++) {
    int tmpx = x + dir[i][0];
    int tmpy = y + dir[i][1];
    if (check(tmpx, tmpy) && !vis[tmpx][tmpy]) {
      dfs(tmpx, tmpy, 0);
    }
  }
  if (!flag) {
    ans.push_back(node('D', x + 1, y + 1));
    ans.push_back(node('R', x + 1, y + 1));
  }
}
void solve(int i, int j) {
  pair<int, int> tmp = dfs1(i, j);
  dfs(tmp.first, tmp.second, 1);
}
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    scanf("%s", s[i]);
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (vis[i][j]) continue;
      if (s[i][j] == '#') continue;
      solve(i, j);
    }
  }
  int len = ans.size();
  cout << len << endl;
  for (int i = 0; i < len; i++) {
    printf("%c %d %d\n", ans[i].c, ans[i].x, ans[i].y);
  }
  return 0;
}
