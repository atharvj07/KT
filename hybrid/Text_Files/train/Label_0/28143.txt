#include <bits/stdc++.h>
using namespace std;
int n, a, b, c, top, x[4], y[4];
long long T, v[4], A[4];
char ans[500005];
long long sum(int id, int sx, int sy, int tx, int ty) {
  int vl = abs(sx - x[id]) + abs(sy - y[id]);
  int vr = abs(tx - x[id]) + abs(ty - y[id]);
  return vl <= vr ? 1ll * (vl + vr + 1) * (vr - vl) / 2
                  : 1ll * (vl + vr - 1) * (vl - vr) / 2;
}
bool check(int px, int py) {
  for (int i = (int)(0); i <= (int)(3); i++) A[i] = v[i];
  if (px != 1 || py != 1)
    for (int i = (int)(0); i <= (int)(3); i++)
      A[i] -= abs(x[i] - px) + abs(y[i] - py);
  if (px < x[0]) {
    for (int i = (int)(0); i <= (int)(3); i++) A[i] -= sum(i, px, py, x[0], py);
    px = x[0];
  }
  if (py < y[0]) {
    for (int i = (int)(0); i <= (int)(3); i++) A[i] -= sum(i, px, py, px, y[0]);
    py = y[0];
  }
  if (px <= x[1] && py <= y[2]) {
    A[0] -= sum(0, px, py, x[1], y[2]);
    A[3] -= sum(3, px, py, x[1], y[2]);
    A[1] -= sum(1, px, py, x[1], py) + sum(1, x[1], py, x[1], y[2]);
    A[2] -= sum(2, px, py, px, y[2]) + sum(2, px, y[2], x[1], y[2]);
    long long dif = sum(1, px, py, px, y[2]) + sum(1, px, y[2], x[1], y[2]) -
                    sum(1, px, py, x[1], py) - sum(1, x[1], py, x[1], y[2]);
    if (A[2] - A[1] >= dif)
      A[2] -= dif;
    else if (A[1] - A[2] >= dif)
      A[1] -= dif;
    else {
      long long tmp = A[2] - (A[2] + A[1] - dif) / 2;
      if (tmp & 1) ++tmp;
      A[2] -= tmp;
      A[1] -= dif - tmp;
    }
    px = x[1];
    py = y[2];
  }
  if (px < x[1]) {
    for (int i = (int)(0); i <= (int)(3); i++) A[i] -= sum(i, px, py, x[1], py);
    px = x[1];
  }
  if (py < y[2]) {
    for (int i = (int)(0); i <= (int)(3); i++) A[i] -= sum(i, px, py, px, y[2]);
    py = y[2];
  }
  for (int i = (int)(0); i <= (int)(3); i++) A[i] -= sum(i, px, py, n, n);
  for (int i = (int)(0); i <= (int)(3); i++)
    if (A[i] < 0) return 0;
  return 1;
}
int main() {
  scanf("%d%lld%d%d%d", &n, &T, &a, &b, &c);
  v[0] = v[1] = v[2] = v[3] = T;
  x[0] = x[2] = a;
  x[1] = x[3] = a + c - 1;
  y[0] = y[1] = b;
  y[2] = y[3] = b + c - 1;
  int px = 1, py = 1;
  if (!check(1, 1)) return puts("Impossible"), 0;
  for (; px != n || py != n;) {
    if (px != n && check(px + 1, py))
      ans[++top] = 'R', px++;
    else
      ans[++top] = 'U', py++;
    for (int j = (int)(0); j <= (int)(3); j++)
      v[j] -= abs(px - x[j]) + abs(py - y[j]);
  }
  printf("%s", ans + 1);
}
