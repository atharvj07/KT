def main():
	t = I()
	aa = [LI() for _ in range(t)]
	rr = []
	for (a, b, l, r) in aa:
		t = a // b
		if l % b > 0:
			l += b - l % b
		if l <= r:
			t -= 1
			t -= (r - l) // b
		rr.append(t)
	return '\n'.join(map(str, rr))
