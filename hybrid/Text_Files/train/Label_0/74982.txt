import sys
from fractions import gcd
with sys.stdin as fin, sys.stdout as fout:
	(n, m, x, y, a, b) = map(int, next(fin).split())
	d = gcd(a, b)
	a //= d
	b //= d
	k = min(n // a, m // b)
	w = k * a
	h = k * b
	x1 = x - (w + 1) // 2
	y1 = y - (h + 1) // 2
	x1 = min(x1, n - w)
	y1 = min(y1, m - h)
	x1 = max(x1, 0)
	y1 = max(y1, 0)
	print(x1, y1, x1 + w, y1 + h, file=fout)
