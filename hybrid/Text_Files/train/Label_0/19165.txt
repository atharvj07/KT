#include <bits/stdc++.h>
using namespace std;
const long long maxn = 5e2 + 10;
const long long inf = 0x3fff3fff3fff3fff;
long long n, a[maxn];
long long ans, minv;
bool ok(long long x, long long vol) {
  if (x < vol || x / vol < x % vol)
    return false;
  else
    return true;
}
long long get_sum(long long x, long long vol) {
  return ((x + vol) / (vol + 1));
}
void update(long long vol) {
  if (vol == 0) return;
  long long tempsum = 0;
  for (long long(i) = (long long)(1); (i) <= (long long)(n); ++(i))
    if (a[i] < vol || (a[i] / vol < a[i] % vol))
      return;
    else
      tempsum += (a[i] + vol) / (vol + 1);
  ans = min(ans, tempsum);
}
int main() {
  ios::sync_with_stdio(false);
  while (scanf("%I64d", &n) != EOF) {
    minv = inf;
    for (long long(i) = (long long)(1); (i) <= (long long)(n); ++(i)) {
      scanf("%I64d", &a[i]);
      minv = min(minv, a[i]);
    }
    ans = inf;
    long long t = sqrt(minv);
    for (long long(i) = (long long)(1); (i) <= (long long)(t); ++(i)) {
      update(i);
      update(minv / i + 1);
      update(minv / i);
      update(minv / i - 1);
    }
    printf("%I64d\n", ans);
  }
  return 0;
}
