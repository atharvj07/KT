class TaskC {

    int n;
    boolean[] vis;
    int[][] grid;


    final int MAX_SIZE = 10000;
    public void solve(int testNumber, InputReader jin, PrintWriter jout) {
        n = jin.int32();

        int[] l = new int[n];
        int[] r = new int[n];

        BigDecimal den = new BigDecimal("1");
        for(int i = 0; i < n; i++) {
            l[i] = jin.int32();
            r[i] = jin.int32();
            den = den.multiply(BigDecimal.valueOf(r[i] - l[i] + 1));
        }

        //jout.println(den);

        vis = new boolean[n];
        grid = new int[n][10001];
        for(int i = 0; i < n; i++) {
            for(int j = l[i]; j <= MAX_SIZE; j++) {
                grid[i][j] = (j <= r[i]? 1: 0) + grid[i][j - 1];
            }
        }


        BigDecimal t1 = BigDecimal.ZERO;
        for(int w = 1; w <= 10000; w++) {
            BigDecimal tt1 = solve1(w);
            BigDecimal tt2 = solve2(w);
            BigDecimal tt3 = tt1.add(tt2);
            //if(tt3.compareTo(BigDecimal.ZERO) > 0) jout.println(w + " " + tt3);
            t1 = t1.add(tt1.add(tt2).multiply(BigDecimal.valueOf(w)));

        }
        jout.printf("%.10f", t1.divide(den, 9, BigDecimal.ROUND_HALF_UP));
    }

    private BigDecimal solve2(int w) {
        BigDecimal t1 = BigDecimal.ZERO;
        for(int mask1 = 0; mask1 < (1 << n); mask1++) {
            if(Integer.bitCount(mask1) <= 1) continue;
            boolean valid = true;
            for(int i = 0; i < n; i++) {
                if((mask1 & (1 << i)) == 0) continue;
                valid &= (grid[i][w] - grid[i][w - 1]) > 0;
            }
            if(valid) {
                BigDecimal t2 = BigDecimal.ZERO;
                for(int mask2 = 0; mask2 < (1 << n); mask2 = mask2 == 0? 1: mask2 << 1) {
                    if((mask1 & mask2) > 0) continue;
                    BigDecimal t3 = BigDecimal.ONE;
                    for(int t = 0; t < n; t++) {
                        if((mask1 & (1 << t)) > 0) continue;
                        if((mask2 & (1 << t)) > 0) {
                            t3 = t3.multiply(BigDecimal.valueOf(grid[t][MAX_SIZE] - grid[t][w]));
                        } else {
                            t3 = t3.multiply(BigDecimal.valueOf(grid[t][w - 1]));
                        }
                    }
                    t2 = t2.add(t3);
                }
                t1 = t1.add(t2);
            }

        }
        return t1;
    }

    public BigDecimal solve1(int w) {
        BigDecimal t2 = BigDecimal.ZERO;
        for(int t = 0; t < n; t++) {
            if(grid[t][w] - grid[t][w - 1] == 0) continue;
            BigDecimal t3 = BigDecimal.ZERO;
            for (int mask = 1; mask < (1 << n); mask <<= 1) {
                if((mask & (1 << t)) > 0) continue;
                BigDecimal t4 = BigDecimal.ONE;
                for(int tt = 0; tt < n; tt++) {
                    if(tt == t) continue;
                    if((mask & (1 << tt)) > 0) {
                        t4 = t4.multiply(BigDecimal.valueOf(grid[tt][MAX_SIZE] - grid[tt][w]));
                    } else {
                        t4 = t4.multiply(BigDecimal.valueOf(grid[tt][w - 1]));
                    }
                }
                t3 = t3.add(t4);
            }
            t2 = t2.add(t3);
        }
        return t2;
    }

}

class InputReader {
    private static final int bufferMaxLength = 1024;
    private InputStream in;
    private byte[] buffer;
    private int currentBufferSize;
    private int currentBufferTop;
    private static final String tokenizers = " \t\r\f\n";

    public InputReader(InputStream stream) {
        this.in = stream;
        buffer = new byte[bufferMaxLength];
        currentBufferSize = 0;
        currentBufferTop = 0;
    }

    private boolean refill() {
        try {
            this.currentBufferSize = this.in.read(this.buffer);
            this.currentBufferTop = 0;
        } catch(Exception e) {}
        return this.currentBufferSize > 0;
    }


    public Byte readChar() {
        if(currentBufferTop < currentBufferSize) {
            return this.buffer[this.currentBufferTop++];
        } else {
            if(!this.refill()) {
                return null;
            } else {
                return readChar();
            }
        }
    }

    public String token()  {
        StringBuffer tok = new StringBuffer();
        Byte first;
        while((first = readChar()) != null && (tokenizers.indexOf((char) first.byteValue()) != -1));
        if(first == null) return null;
        tok.append((char)first.byteValue());
        while((first = readChar()) != null && (tokenizers.indexOf((char) first.byteValue()) == -1)) {
            tok.append((char)first.byteValue());
        }
        return tok.toString();
    }


    public String next() {
        return token();
    }

    public Integer int32() throws NumberFormatException {
        String tok = token();
        return tok == null? null : Integer.parseInt(tok);
    }

    public Long int64() throws NumberFormatException {
        String tok = token();
        return tok == null? null : Long.parseLong(tok);
    }
}
