def factor(q):
	i = 2
	counters = {}
	while i * i <= q:
		while q % i == 0:
			counters.setdefault(i, 0)
			counters[i] += 1
			q //= i
		i += 1
	if q > 1:
		counters.setdefault(q, 0)
		counters[q] += 1
	return counters
