import java.util.*;
import java.io.*;

public class FooFighters {
    public static void main(String[] args) {
        FastScanner scanner = new FastScanner();
        PrintWriter out = new PrintWriter(System.out);
        int N = scanner.nextInt();
        Pair[] items = new Pair[N];
        long sum = 0;
        for(int i =0; i < N; i++) {
            items[i] = new Pair(scanner.nextInt(), scanner.nextLong());
            sum += items[i].a;
        }
        Arrays.sort(items);
        int sign = Long.signum(sum);
        long ans = 0;
        for(int curBit = 0; curBit < 62; curBit++) {
            long sum0 = 0, sum1 = 0;
            for(int i = 0; i < N; i++) {
                if (items[i].b >> curBit == 0) {
                    sum0 += items[i].a;
                }
                if (items[i].b >> curBit == 1) {
                    sum1 += items[i].a;
                }
            }
            int sgn0 = Long.signum(sum0+sum1), sgn1 = Long.signum(sum0-sum1);
            if (sgn1 * sign < sgn0 * sign) {
                ans |= 1L << curBit;
                for(int i = 0; i < N; i++) {
                    if ((items[i].b & (1L << curBit)) != 0) {
                        items[i].a = - items[i].a;
                    }
                }
            }
            
        }
        out.println(ans);
        out.flush();
    }
    
    public static class Pair implements Comparable<Pair> {
        long a, b;
        public Pair(long aa, long bb) {
            a = aa; b = bb;
        }
        public int compareTo(Pair o) {
            return Long.compare(this.b, o.b);
        }
    }
    
    public static class FastScanner {
        BufferedReader br;
        StringTokenizer st;
        
        public FastScanner(Reader in) {
            br = new BufferedReader(in);
        }
        
        public FastScanner() {
            this(new InputStreamReader(System.in));
        }
        
        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        int nextInt() {
            return Integer.parseInt(next());
        }
        
        long nextLong() {
            return Long.parseLong(next());
        }
        
        double nextDouble() {
            return Double.parseDouble(next());
        }
        
        String readNextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
        
        int[] readIntArray(int n) {
            int[] a = new int[n];
            for (int idx = 0; idx < n; idx++) {
                a[idx] = nextInt();
            }
            return a;
        }
    }
}
