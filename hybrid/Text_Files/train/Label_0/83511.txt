for _ in ' ' * int(input()):
	(n, k) = map(int, input().split())
	s = list(input())
	if 'W' not in s:
		print(max(min(k, n) * 2 - 1, 0))
	elif k >= s.count('L'):
		print(n * 2 - 1)
	else:
		cnt = []
		sm = 0
		for i in range(n):
			if s[i] == 'W':
				sm += 1
		ind = s.index('W')
		for i in range(ind + 1, n):
			if s[i] == 'W':
				cnt.append(i - ind - 1)
				ind = i
		cnt.sort()
		for i in cnt:
			if k >= i:
				sm += 2 * i + 1
				k -= i
		if k > 0:
			sm += 2 * k
		print(sm)
