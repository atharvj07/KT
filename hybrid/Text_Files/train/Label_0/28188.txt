#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i=a;i<b;i++)
#define INF INT_MAX/2

int N;

bool comp(pair<int, int> a, pair<int, int> b)
{
	return a.first < b.first;
}

int maketime(int s, int m, int h)
{
	int hh = (h - (m / 12) + 60) % 60;
	if (hh % 5 != 0) return -1;
	hh /= 5;

	return hh * 60 * 60 + m * 60 + s;
}

int main()
{
	cin.tie(0);
	ios::sync_with_stdio(false);

	while (cin >> N)
	{
		if (N == 0) return 0;
		vector<pair<int, int> > times;
		rep(i, 0, N)
		{
			int s, t, u; cin >> s >> t >> u;
			t -= s;
			u -= s;
			s = 0;

			rep(ss, 0, 60)
			{
				int tt = (ss + t + 60) % 60;
				int uu = (ss + u + 60) % 60;

				int time = maketime(ss, tt, uu);
				if (0 <= time) times.push_back(make_pair(time, i));

				time = maketime(ss, uu, tt);
				if (0 <= time) times.push_back(make_pair(time, i));

				time = maketime(tt, ss, uu);
				if (0 <= time) times.push_back(make_pair(time, i));

				time = maketime(tt, uu, ss);
				if (0 <= time) times.push_back(make_pair(time, i));

				time = maketime(uu, tt, ss);
				if (0 <= time) times.push_back(make_pair(time, i));

				time = maketime(uu, ss, tt);
				if (0 <= time) times.push_back(make_pair(time, i));
			}
		}

		sort(times.begin(), times.end(), comp);

		int ans = INF;
		int ans_s = -1, ans_t;

		rep(i, 0, times.size())
		{
			int m = 0;
			rep(j, i, times.size())
			{
				m = m | (1 << times[j].second);
				if (m == (1 << N)-1)
				{
					int len = times[j].first - times[i].first;
					if (len < ans)
					{
						ans = len;
						ans_s = times[i].first;
						ans_t = times[j].first;
					}
					break;
				}
			}
		}

		if (ans_s < 0)
			cout << "???" << endl;
		else
		{
			int hh, mm, ss;
			ss = ans_s % 60; ans_s /= 60;
			mm = ans_s % 60; ans_s /= 60;
			hh = ans_s;
			printf("%02d:%02d:%02d", hh, mm, ss);
			ss = ans_t % 60; ans_t /= 60;
			mm = ans_t % 60; ans_t /= 60;
			hh = ans_t;
			printf(" %02d:%02d:%02d\n", hh, mm, ss);
		}
	}
}
