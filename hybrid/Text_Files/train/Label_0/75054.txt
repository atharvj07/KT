import itertools as it
(n, m) = map(int, input().split(' '))
inputs = []
for i in range(0, n):
	s = input().split(' ')
	inputs.append((int(s[1]), int(s[2]), s[0]))
data = []
for (k, g) in it.groupby(sorted(inputs), lambda x: x[0]):
	data.append(sorted(g, key=lambda x: x[1], reverse=True))
for i in range(0, m):
	if len(data[i]) > 2 and data[i][1][1] == data[i][2][1]:
		print('?')
	else:
		print(data[i][0][2] + ' ' + data[i][1][2])
