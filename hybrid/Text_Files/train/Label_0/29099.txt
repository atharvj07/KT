#include<bits/stdc++.h>
 
using namespace std;
 
typedef long long ll;
 
const int N = 2e5+10, mod = 1e9+7;

ll a[N];

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	ll n, k, x;
	cin>>n>>k>>x;
	for(int i = 1; i <= n; i++) {
		cin>>a[i];
	}
	sort(a + 1, a + n + 1);
	
	vector<ll> v;
	for(int i = 2; i <= n; i++) {
		if(a[i] - a[i - 1] > x)
			v.emplace_back((a[i] - a[i - 1] + x - 1) / x - 1);
	}
	ll ans = v.size() + 1;
	
	sort(v.begin(), v.end());
	for(ll u : v) {
		k -= u;
		if(k < 0) {
			break;
		}
		ans --;
	}
	cout<<ans<<endl;
	return 0;
}
