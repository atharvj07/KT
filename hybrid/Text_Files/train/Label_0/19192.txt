#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
  ll n,ce=0,cf=0;
  cin>>n;
  ll a[n],e[n],f[n];
  cin>>a[0];
  e[0]=a[0];
  f[0]=a[0];
  if(e[0]<1){
    e[0]=1;
    ce+=1-a[0];
  }
  if(f[0]>-1){
    f[0]=-1;
    cf+=a[0]+1;
  }
  for(int i=1;i<n;i++){
    cin>>a[i];
    e[i]=e[i-1]+a[i];
    f[i]=f[i-1]+a[i];
    if(i%2==1){
      if(e[i]>-1){
        ce+=e[i]+1;
        e[i]=-1;
      }
      if(f[i]<1){
        cf+=1-f[i];
        f[i]=1;
      }
    }else{
      if(f[i]>-1){
        cf+=f[i]+1;
        f[i]=-1;
      }
      if(e[i]<1){
        ce+=1-e[i];
        e[i]=1;
      }
    }
  }
  cout<<min(ce,cf);
}
