#include "bits/stdc++.h"

using namespace std;

string alpha[26] = {
	"1 0 0 1 0",
	"1 1 0 2 0",
	"2 0 0 1 1",
	"2 1 0 1 2",
	"1 1 0 1 1",
	"2 1 0 2 1",
	"2 2 0 2 2",
	"1 2 0 2 1",
	"1 1 0 1 1",
	"1 2 0 1 2",
	"1 0 1 2 0",
	"1 1 1 3 0",
	"2 0 1 2 1",
	"2 1 1 2 2",
	"1 1 1 2 1",
	"2 1 1 3 1",
	"2 2 1 3 2",
	"1 2 1 3 1",
	"1 1 1 2 1",
	"1 2 1 2 2",
	"1 0 2 2 1",
	"1 1 2 3 1",
	"1 2 1 1 3",
	"2 0 2 2 2",
	"2 1 2 2 3",
	"1 1 2 2 2"
};

int main() {
	int n;
	string s;
	cin >> n;
	cin.ignore();
	unordered_map<string, char> m;
	for (int i = 25; i >= 0; i--) {
		m[alpha[i]] = 'a' + i;
	}
	for (int i = 0; i < n; i++) {
		getline(cin, s);
		putchar(m[s]);
	}
	return 0;
}
