#include <bits/stdc++.h>
using namespace std;
template <typename T>
inline void Int(T &n) {
  n = 0;
  int f = 1;
  register int ch = getchar();
  for (; !isdigit(ch); ch = getchar())
    if (ch == '-') f = -1;
  for (; isdigit(ch); ch = getchar()) n = (n << 3) + (n << 1) + ch - '0';
  n = n * f;
}
template <typename T>
T gcd(T a, T b) {
  return !b ? a : gcd(b, a % b);
}
template <typename T>
inline void umin(T &a, T b) {
  a = a < b ? a : b;
}
template <typename T>
inline void umax(T &a, T b) {
  a = a > b ? a : b;
}
template <typename T, typename W>
inline void Int(T &x, W &y) {
  Int(x), Int(y);
}
template <typename T, typename W, typename Q>
void Int(T &x, W &y, Q &z) {
  Int(x, y), Int(z);
}
const int N = 5e5 + 7;
const int inf = 1e9 + 7;
int dp[N];
long long res;
int solve() {
  int n, k;
  cin >> n >> k;
  string s, t;
  cin >> s >> t;
  dp[0] = 1;
  for (int i = 0; i < n; ++i) {
    dp[i + 1] = dp[i] * 2;
    if (s[i] == 'b') {
      --dp[i + 1];
    }
    if (t[i] == 'a') {
      --dp[i + 1];
    }
    umin(dp[i + 1], inf);
    res += min(k, dp[i + 1]);
  }
  cout << res << '\n';
  return 0;
}
int main() {
  int tests = 1, CaseNo = 0;
  while (tests--) {
    solve();
  }
  return 0;
}
