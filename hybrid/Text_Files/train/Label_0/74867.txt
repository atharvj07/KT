s = input()
alphabet = ['a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
occ = {}
for ltr in alphabet:
	occ[ltr] = [0, len(s) + 1]
for i in range(len(s)):
	occ[s[i]].append(i + 1)
k = 10 ** 9
for ltr in alphabet:
	kspec = 0
	occ[ltr].sort()
	for i in range(1, len(occ[ltr])):
		kspec = max(kspec, occ[ltr][i] - occ[ltr][i - 1])
	k = min(k, kspec)
print(k)
