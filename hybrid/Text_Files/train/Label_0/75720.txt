def main():
	from sys import stdin, stdout
	(n, m) = map(int, stdin.readline().split())
	if m:
		dirty = sorted(map(int, stdin.readline().split()))
		if dirty[0] == 1 or dirty[-1] == n:
			stdout.write('NO')
		else:
			flag = True
			for i in range(m - 2):
				if dirty[i + 1] == dirty[i] + 1 and dirty[i + 2] == dirty[i] + 2:
					flag = False
					break
			if flag:
				stdout.write('YES')
			else:
				stdout.write('NO')
	else:
		stdout.write('YES')
main()
