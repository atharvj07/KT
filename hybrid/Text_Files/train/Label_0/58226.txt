    import java.util.*;
    public class Main{
        public static void main(String args[]){
            Scanner sc=new Scanner(System.in);
            int n=sc.nextInt();
            int a[]=new int[n];
            for(int i=0;i<n;i++){
                a[i]=sc.nextInt();
                
            }
            int m=sc.nextInt();
            for(int i=0;i<m;i++){
                int x=sc.nextInt();
                int y=sc.nextInt();
                x=x-1;
                if(n==1){a[x]=0;}
                else if(x==0){a[x+1]=a[x+1]+a[x]-y;a[x]=0;}
                else if(x==n-1){a[x-1]=a[x-1]+y-1;a[x]=0;}
                else{a[x-1]=a[x-1]+y-1;a[x+1]=a[x+1]+a[x]-y;a[x]=0;}
            }
        for(int i:a){
            System.out.println(i);
        }
    }}
