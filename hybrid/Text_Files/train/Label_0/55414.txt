class Task {
    final int maxn = 41111;
    ArrayList<Integer> Beg, End;
    Integer[] X, Y;
    int N, M;
    PrintStream cout = System.out;

    void print(Object[] arr) {
        cout.println("print arr :");
        for (Object i : arr) {
            cout.print(i + " ");
        }
        cout.println();
    }

    int value;
    long ans = 0;

    void update(int v, long c) {
        if (v == value) {
            ans += c;
        } else if (v > value) {
            value = v;
            ans = c;
        }
    }

    long calc(long n) {
        if (n < 2)
            return 0;
        n = n * n - n;
        n >>= 1;
        return n;
    }

    long single(Integer[] x) {
        int n = x.length;
        long a = 0, b = 0;

        long ans = n;
        if (x[0].equals(x[n - 1])) {
            ans = ans * (ans - 1) * (ans - 2) / 6;

        } else {
            for (int i = 0; i < n; ++i) {
                if (x[i].equals(x[0]))
                    ++a;
                if (x[i].equals(x[n - 1]))
                    ++b;
            }
            ans = calc(a) * b + calc(b) * a;
            ans = ans + (a * b * (n - a - b));
        }
        return ans;
    }

    void run(Integer x[], Integer y[]) {
        int n = x.length;
        int m = y.length;
        int i, j;
        long sub = 0;
        int c = 0;
        for (i = 0, j = 0; i < n; ++i) {
            while (j < m && y[j] < x[i]) {
                ++j;
                ++c;
            }
            sub += calc(c);
        }
        c = 0;
        for (i = n - 1, j = m - 1; i >= 0; --i) {
            while (j >= 0 && y[j] > x[i]) {
                --j;
                ++c;
            }
            sub += calc(c);
        }
        sub = calc(m) * n - sub;

        update(N * 2, sub);
    }

    HashMap<Integer, Integer> hashx = new HashMap<Integer, Integer>();
    HashMap<Integer, Integer> hashy = new HashMap<Integer, Integer>();

    void main(InputReader in, PrintWriter out) throws Exception {
        Beg = new ArrayList<Integer>();
        End = new ArrayList<Integer>();
        N = in.nextInt();
        M = in.nextInt();
        for (int i = 0; i < M; ++i) {
            int t, s;
            t = in.nextInt();
            s = in.nextInt();
            if (t == 0) {
                Beg.add(s);
                if (!hashx.containsKey(s)) {
                    hashx.put(s, 1);
                } else {
                    int k = hashx.get(s);
                    hashx.put(s, k + 1);
                }
            } else {
                End.add(s);
                if (!hashy.containsKey(s)) {
                    hashy.put(s, 1);
                } else {
                    int k = hashy.get(s);
                    hashy.put(s, k + 1);
                }
            }
        }
        // out.println(Beg.size());
        X = Beg.toArray(new Integer[Beg.size()]);
        Y = End.toArray(new Integer[End.size()]);
        Arrays.sort(X);
        Arrays.sort(Y);
        // print(X);
        // print(Y);
        int n = X.length;
        int m = Y.length;
        ans = 0;
        if (m > 2) {
            update(2 * (Y[m - 1] - Y[0]), single(Y));
        }
        if (n > 2) {
            update(2 * (X[n - 1] - X[0]), single(X));

        }
        if (n > 0 && m > 0) {
            if (X[n - 1] < Y[0]) {
                long tmp = 0;
                long a = hashx.get(X[n - 1]);
                long b = hashy.get(Y[0]);
                tmp = calc(a) * b + calc(b) * a;
                tmp += a * b * (m + n - a - b);
                update(N * 2 - (Y[0] - X[n - 1]), tmp);
            } else if (X[0] > Y[m - 1]) {
                long tmp = 0;
                long a = hashy.get(Y[m - 1]);
                long b = hashx.get(X[0]);
                tmp = calc(a) * b + calc(b) * a;
                tmp += a * b * (m + n - a - b);
                update(N * 2 - (X[0] - Y[m - 1]), tmp);
            } else {
                run(X, Y);
                run(Y, X);
            }
        }
        out.println(ans);

    }
}

class InputReader {
    BufferedReader reader;
    StringTokenizer tokenizer;

    InputReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream));
        tokenizer = null;
    }

    String next() {
        if (!hasNext())
            throw new RuntimeException();
        return tokenizer.nextToken();
    }

    boolean hasNext() {
        while (tokenizer == null || !tokenizer.hasMoreTokens())
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (Exception e) {
                return false;
            }
        return true;
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }
}
