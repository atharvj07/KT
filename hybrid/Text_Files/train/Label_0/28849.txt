#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const double eps = 1e-5;
const int MAX = 2500 + 10;
const double pi = acos(-1.0);
const int inf = 0x7f7f7f7f;
const long long INF = 0x7f7f7f7f7f7f7f7f;
int arr[MAX][MAX];
int n;
bool solve() {
  for (int i = 0; i < n; i++) {
    int tt = 500;
    while (tt--) {
      int j = rand() % n;
      for (int k = 0; k < n; k++) {
        if (arr[i][j] > max(arr[i][k], arr[j][k])) {
          return false;
        }
      }
    }
  }
  return true;
}
int main() {
  while (~scanf("%d", &n)) {
    bool flag = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        scanf("%d", &arr[i][j]);
        if (i == j && arr[i][j] != 0) {
          flag = 1;
        }
      }
    }
    if (flag) {
      printf("NOT MAGIC\n");
      continue;
    }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (arr[i][j] != arr[j][i]) {
          flag = 1;
          break;
        }
      }
    }
    if (flag) {
      printf("NOT MAGIC\n");
      continue;
    }
    if (solve()) {
      printf("MAGIC\n");
    } else {
      printf("NOT MAGIC\n");
    }
  }
  return 0;
}
