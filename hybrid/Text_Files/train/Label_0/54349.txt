

import java.util.*;
public class E {

    int n = 300000000 + 2;
    byte []bits = new byte[n/16 + 1];
    boolean isset(int x){
        
        x >>= 1;
        int ind = x >> 3;
        int sh = x & 7;
        return (bits[ind] & (1 << sh)) != 0;
    }
    
    void setbit(int x){
        x >>= 1;
        int ind = x >> 3;
        int sh = x & 7;
        bits[ind] |= (1 << sh);
    }
    void seive(){
        
        for (int i = 3; i*i < n; i+=2) {
            if(!isset(i)){              
                for (int j = i*i; j < n; j+=i+i) {
                        setbit(j);
                }
            }
        }
    }
    void read(){
        seive();
        Scanner s = new Scanner(System.in);
        int l = s.nextInt();
        int r = s.nextInt();        
        int ans = 0;
        if(l <= 2 && r >= 2 ) ans ++;
        for (int i = 5; i <= n; i+=4) {
            if(l <= i && r >= i && !isset(i)){
                ans++;
            }
        }
        System.out.println(ans);
    }
    public static void main(String[] args) {
        new E().read();     
        
    }
}
