import java.io.*;
import java.util.*;
import java.math.BigInteger;
import java.math.BigDecimal;

public class Main {

  public static class MyScanner {
      BufferedReader reader;
      StringTokenizer tokenizer;

      public MyScanner() {
          reader = new BufferedReader(new InputStreamReader(System.in));
          tokenizer = new StringTokenizer("");
      }

      public String next() {
          while (!tokenizer.hasMoreTokens()) {
            try {
              tokenizer = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
              e.printStackTrace();
            }
          }
          return tokenizer.nextToken();
      }

      public int nextInt() {
        return Integer.parseInt(next());
      }

      public long nextLong() {
        return Long.parseLong(next());
      }
          
      public double nextDouble() {
        return Double.parseDouble(next());
      }
  }

  PrintWriter out;
  MyScanner sc;

  public Main()
  {
    out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
    sc = new MyScanner();
  }

  public void fin() {
    out.flush();
  }

  /* ========== MAIN ========== */

  int N_MAX=100000;

  public void init() {
  }

  BigInteger[] reduce(final String S) {
    BigInteger[] s = new BigInteger[N_MAX];
    Arrays.fill(s, BigInteger.ZERO);

    for (int i=0; i<S.length(); ++i) {
      if (S.charAt(S.length()-1-i) == '1') {
        s[i] = BigInteger.ONE;
      }
    }

    for (int i=s.length-1; i>=2; --i) {
      if (s[i].compareTo(BigInteger.ZERO)>0) {
        s[i-1] = s[i-1].add(s[i]);
        s[i-2] = s[i-2].add(s[i]);
        s[i] = null;
      }
    }

    return new BigInteger[]{s[0], s[1]};
  }

  BigDecimal translate(final BigInteger[] s) {
    final BigDecimal q  = new BigDecimal((Math.sqrt(5) + 1.0)/2.0);
    final BigDecimal b1 = new BigDecimal(s[1]);
    final BigDecimal b0 = new BigDecimal(s[0]);
    return q.multiply(b1).add(b0);
  }

  String compare(final BigInteger[] s1, final BigInteger[] s2) {
    if (Arrays.equals(s1, s2)) {
      return "=";
    }

    final BigDecimal d1 = translate(s1);
    final BigDecimal d2 = translate(s2);

    final int comp = d1.compareTo(d2);
    if (comp < 0) {
      return "<";
    } else if (comp > 0) {
      return ">";
    } else {
      throw new AssertionError("Never");
    }
  }

  public void solve() {
    final String S1 = sc.next();
    final String S2 = sc.next();

    final BigInteger[] s1 = reduce(S1);
    final BigInteger[] s2 = reduce(S2);

    out.println(compare(s1, s2));
  }

  public static void main(String[] args) {
     Main m = new Main();
     m.init();
     m.solve();
     m.fin();
  }
}
