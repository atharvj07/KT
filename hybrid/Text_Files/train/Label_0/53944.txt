import java.util.*;
public class MainC {
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        ArrayList<Double>pr = new ArrayList<Double>();
        double Q;
        for (int i = 0; i<T;i++){
            double a = sc.nextInt();
            double b = sc.nextInt();
            if (b==0) Q = 1;
            else if (a==0) Q = 0.5;
            else if(a/4 < b) Q = 0.5 + a/(b*16);
            else Q = 1 - b/a;
            pr.add(Q);
        }
        for (int i = 0; i<pr.size(); i++){
            System.out.println(pr.get(i));
        }
    }
}
