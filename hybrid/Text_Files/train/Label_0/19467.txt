#include<cstdio>
#include<algorithm>
#define MM	1000000
using namespace std;
int main(void)
{
	int i,j,n,p,k[101],t[101][100];
	int s,d,leng[101],r,flg[101],f,v,ii,jj,x;
	scanf("%d",&n);
	for(i=1;i<=n;i++)	{
		scanf("%d",&r);
		scanf("%d",&k[r]);
		for(j=0;j<k[r];j++)	{
			scanf("%d",&t[r][j]);
		}
	}
	scanf("%d",&p);
	for(i=0;i<p;i++)	{
		scanf("%d %d %d",&s,&d,&v);
		f=0;
		for(j=1;j<=n;j++)	{
			flg[j]=0;	leng[j]=MM;
		}
		leng[s]=1;
		for(j=1;j<v;j++)	{
			for(ii=1;ii<=n;ii++)	{
				if(leng[ii]==j)	{
					for(jj=0;jj<k[ii];jj++)	{
						x=t[ii][jj];
						if(leng[x]>j+1)	{
							leng[x]=j+1;
						}
						if(x==d)	{
							f=1;
							ii=n+1;
							j=v;
							break;
						}
					}
				}
			}
		}
		if(f==1)	{
			printf("%d\n",leng[d]);
		}
		else {
			printf("NA\n");
		}
	}
	return 0;
}
