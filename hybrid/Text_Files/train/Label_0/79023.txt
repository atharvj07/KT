from collections import deque
import heapq
from math import sqrt
import operator
from operator import itemgetter
import sys
import copy
inf_var = 0
if inf_var == 1:
	inf = open('input.txt', 'r')
else:
	inf = sys.stdin
input = inf.readline

def read_one_int():
	return int(input().rstrip('\n'))

def read_one_float():
	return float(input().rstrip('\n'))

def read_list_of_ints():
	res = [int(val) for val in input().rstrip('\n').split(' ')]
	return res

def read_str():
	return input().rstrip()

def main():
	tests = read_one_int()
	for _ in range(tests):
		(n, k) = read_list_of_ints()
		elements = read_list_of_ints()
		graph = [[] for _ in range(n)]
		for i in range(n - 1, -1, -1):
			need_to_del_i = i + 1 - elements[i]
			if need_to_del_i < 0:
				continue
			for j in range(i - 1, -1, -1):
				need_to_del_j = j + 1 - elements[j]
				if need_to_del_j < 0 or need_to_del_j > need_to_del_i:
					continue
				if j < need_to_del_j or need_to_del_i - need_to_del_j > i - j - 1:
					continue
				graph[i].append(j)
		cache = {}

		def count_max(v):
			if v in cache:
				return cache[v]
			if v + 1 - elements[v] >= 0:
				res = 0
				for u in graph[v]:
					res = max(res, count_max(u))
				cache[v] = res + 1
				return res + 1
			else:
				cache[v] = 0
				return 0
		res = -1
		for i in range(n):
			cur_res = count_max(i)
			if cur_res >= k:
				if res == -1:
					res = i + 1 - elements[i]
				else:
					res = min(res, i + 1 - elements[i])
		print(res)
main()
