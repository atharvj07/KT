#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int T;
int X, Y, A, B;

long double get_cnt(int b_cnt) {
    long double remain_x = X - b_cnt * A;
    long double remain_y = Y - b_cnt * B;
    return b_cnt + min(remain_x / B, remain_y / A);
}

int get_ans(int b_cnt) {
    int remain_x = X - b_cnt * A;
    int remain_y = Y - b_cnt * B;
    return b_cnt + min(remain_x / B, remain_y / A);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> T;
    while (T-- > 0) {
        cin >> X >> Y >> A >> B;
        if (X > Y) {
            swap(X, Y);
        }
        if (A > B) {
            swap(A, B);
        }
        int lo = 0;
        int hi = min(X / A, Y / B);
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (get_cnt(mid) < get_cnt(mid + 1)) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }
        cout << get_ans(lo) << "\n";
    }
    return 0;
}
