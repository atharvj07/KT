t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	v = list(map(int, input().split()))
	t = list(map(int, input().split()))
	d = [t[i] - v[i] for i in range(n)]
	edge = []
	x = [-1] * n
	for _ in range(n):
		edge.append([])
	for _ in range(m):
		(a, b) = map(int, input().split())
		edge[a - 1].append(b - 1)
		edge[b - 1].append(a - 1)
	if sum(d) % 2 != 0:
		print('NO')
	else:
		isbgph = 0
		x[0] = 0
		p = [0]
		while len(p) > 0:
			i = p.pop()
			for nxt in edge[i]:
				if x[nxt] == -1:
					x[nxt] = 1 - x[i]
					p.append(nxt)
				elif x[nxt] == x[i]:
					isbgph = 1
		if isbgph == 1:
			print('YES')
		else:
			sum1 = sum([d[i] for i in range(n) if x[i] == 0])
			sum2 = sum([d[i] for i in range(n) if x[i] == 1])
			if sum1 == sum2:
				print('YES')
			else:
				print('NO')
