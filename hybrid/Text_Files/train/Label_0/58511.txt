import java.io.*;
import java.util.*;

public class C {
  static Map<Character, Integer> charMapping = new HashMap<Character, Integer>();
  static {
    charMapping.put('R', 5);
    charMapping.put('G', 6);
    charMapping.put('B', 7);
    charMapping.put('Y', 8);
    charMapping.put('W', 9);
  }
  static int[] count = new int[10];

  public static void main (String[] args) throws Exception {
    BufferedReader bin = new BufferedReader(new InputStreamReader(System.in));
    int N = Integer.parseInt(bin.readLine());

    String[] cStrs = bin.readLine().split(" ");
    List<Card> clist = new ArrayList<Card>();
    Card[] cards = new Card[N];

    for (int i = 0; i < N; i++) {
      char c = cStrs[i].charAt(0);
      int num = Integer.parseInt(""+cStrs[i].charAt(1)) - 1;
      Card card = new Card(c, num);
      cards[i] = card;
    }

    for (int i = 0; i < N; i++) {
      if (!clist.contains(cards[i])) clist.add(cards[i]);
    }
    Arrays.fill(count, 0);
    for (int i = 0; i < clist.size(); i++) {
      count[clist.get(i).number] ++;
      count[charMapping.get(clist.get(i).suit)] ++;
    }
    System.out.println(solve(clist.toArray(new Card[clist.size()])));
  }

  public static int solve(Card[] cards) {
    int N = cards.length;
    int minHints = Integer.MAX_VALUE;

    for (int i = 0; i < (1 << 10); i++) {
      int bc = Integer.bitCount(i);
      if (bc > minHints) continue;
      boolean c = solveable(cards, i);
      if (c) minHints = Math.min(minHints, Integer.bitCount(i));
    }
    return minHints;
  }

  public static boolean solveable(Card[] cards, int hints) {
    int n = cards.length;
    boolean good = true;
    boolean[] used = new boolean[n];
    Arrays.fill(used, false);

    int[] counts = new int[10];
    for (int i = 0; i < 10; i++) counts[i] = count[i];

    int fin = 0;
    while (good) {
      good = false;
      for (int i = 0; i < n; i++) {
        if (used[i]) continue;
        int num = cards[i].number;
        int s = charMapping.get(cards[i].suit);

        boolean h1 = ((hints>>num)&1)>0;
        boolean h2 = ((hints>>s)&1)>0;
        boolean ok = false;

        if (h1 && h2) ok = true;
        else if (h1 && counts[num] <= 1) ok = true;
        else if (h2 && counts[s] <= 1) ok = true;
        else if (!h1 && !h2 && fin+1==n) ok = true;

        if (ok) {
          counts[num]--;
          counts[s]--;
          good = true;
          used[i] = true;
          fin++;
        }
      }
      if (fin == n) return true;
    }
    return false;
  }

  static class Card {
    char suit;
    int number;
    public Card (char suit, int number) {
      this.suit = suit;
      this.number = number;
    }

    @Override
    public boolean equals(Object o) {
      if (!(o instanceof Card)) return false;
      Card other = (Card) o;
      return (other.suit == this.suit && other.number == this.number);
    }

    @Override
    public String toString() {
      return suit + " " + number;
    }
  }
}
