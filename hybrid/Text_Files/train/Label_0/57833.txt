import java.util.*;

public class Solution {
    private static int solve(char[] moves) {
        Set<List<Integer>> used = new HashSet<>();

        int x = 0;
        int y = 0;

        int result = 0;
        for (char c : moves) {

//            System.out.println(used);
//            System.out.printf("x: %d, y: %d, move: %s\n", x, y, c);
            List<Integer> edge = new LinkedList<>();
            edge.add(x);
            edge.add(y);
            edge.add((int)c);
            boolean fresh = !used.contains(edge);
            if (!fresh) result += 1;
            else {
                result += 5;
                used.add(edge);
            }

            if (c == 'N') y += 1;
            else if (c == 'S') y -= 1;
            else if (c == 'W') x -= 1;
            else x += 1;

            if (fresh) {
                char d = 'S';
                if (c == 'S') d = 'N';
                else if (c == 'W') d = 'E';
                else if (c == 'E') d = 'W';

                edge = new LinkedList<>();
                edge.add(x);
                edge.add(y);
                edge.add((int)d);
                used.add(edge);
            }
        }

        return result;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int T = in.nextInt();
        in.nextLine();

        for (int t = 0; t < T; t += 1) {
            char[] moves = in.nextLine().trim().toCharArray();

            System.out.println(solve(moves));
        }
    }
}
