T = int(input())
for i in range(T):
	n = int(input())
	if n == 1:
		print(1)
	elif n == 2:
		print(-1)
	elif n == 3:
		print('2 9 7')
		print('4 6 3')
		print('1 8 5')
	else:
		l = list()
		for i in range(n ** 2 // 2 * 2):
			l.append(i + 1)
			if n % 2 == 1:
				l.append(n ** 2 // 2 + 2 + i)
			else:
				l.append(int(n ** 2 / 2 + 1 + i))
		if n % 2 == 1:
			l.append(n ** 2)
		(g, a) = (list(), list())
		for i in range(n):
			for j in range(n):
				a.append(l[n * i + j])
			g.append(a)
			a = []
		for i in g:
			for j in range(len(i)):
				print(i[j], end=' ')
			print()
