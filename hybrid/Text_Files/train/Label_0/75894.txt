import sys
input = sys.stdin.readline
q = int(input())
for _ in range(q):
	n = int(input())
	(lx, ly) = (0, -1)
	(x, y) = (0, 0)
	s = [['1'] + list(input().rstrip()) for _ in range(2)]
	visit = [[0] * (n + 1) for _ in range(2)]
	while 0 <= x < 2 and 0 <= y < n + 1:
		if visit[x][y]:
			break
		visit[x][y] = 1
		if int(s[x][y]) <= 2:
			(nx, ny) = (2 * x - lx, 2 * y - ly)
		elif abs(x - lx):
			(nx, ny) = (x, y + 1)
		else:
			(nx, ny) = (x ^ 1, y)
		(lx, ly) = (x, y)
		(x, y) = (nx, ny)
	ans = 'YES' if x == 1 and y == n + 1 else 'NO'
	print(ans)
