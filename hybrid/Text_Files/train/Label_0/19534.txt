#include <bits/stdc++.h>
int checkIsLucky(int num) {
  while (num > 0) {
    if (num % 10 != 4 && num % 10 != 7) return 0;
    num /= 10;
  }
  return 1;
}
int nextNum(int num) {
  int nextN;
  int count = 1;
  int maxNextD = 0;
  nextN = num;
  if (num >= 777777777) return 1000000001;
  while (num > 0) {
    if (num % 10 == 4) {
      nextN += count * 3;
      nextN = nextN / (count)*count + maxNextD;
      break;
    } else {
      maxNextD *= 10;
      maxNextD += 4;
      num /= 10;
      count *= 10;
    }
  }
  if (num > 0)
    return nextN;
  else {
    maxNextD *= 10;
    maxNextD += 4;
    return maxNextD;
  }
}
int max(int a, int b) {
  if (a > b)
    return a;
  else
    return b;
}
int min(int a, int b) {
  if (a > b)
    return b;
  else
    return a;
}
int kthNum(int begin, int k) {
  int count = 1;
  int rt = 0, tmp = begin;
  while (tmp > 0) {
    tmp /= 10;
    rt *= 10;
    rt += 4;
  }
  while (rt < begin) rt = nextNum(rt);
  for (; count < k; count++) {
    rt = nextNum(rt);
  }
  return rt;
}
int ma, mi;
double lenthRt(unsigned a, unsigned b, unsigned lk1, unsigned lk2) {
  mi = max(a, lk1);
  ma = min(b, lk2);
  double kq;
  if (ma < mi)
    return 0;
  else {
    if (lk2 == ma) {
      kq = ma - mi;
      return kq;
    }
    kq = ma - mi + 1;
    return kq;
  }
}
int main() {
  int pl, pr, vl, vr, k;
  int minN, maxN, lk1, lk2, i, after;
  double sum = 0;
  double kq, mau;
  scanf("%d %d %d %d %d", &pl, &pr, &vl, &vr, &k);
  minN = min(pl, vl);
  maxN = max(pr, vr);
  after = kthNum(pl, 1);
  if (k == 1) {
    for (i = after; i <= pr; i = nextNum(i)) {
      if (i >= vl && i <= vr) sum -= 1;
      getchar();
    }
  }
  lk1 = kthNum(pl, k);
  lk2 = nextNum(lk1);
  i = pl;
  for (; i <= pr;) {
    sum += lenthRt(vl, vr, lk1, lk2) * min(after - i + 1, pr - i + 1);
    i = after + 1;
    after = nextNum(after);
    lk1 = lk2;
    if (lk1 > vr) break;
    lk2 = nextNum(lk2);
  }
  after = kthNum(vl, 1);
  lk1 = kthNum(vl, k);
  lk2 = nextNum(lk1);
  i = vl;
  for (; i <= vr;) {
    sum += lenthRt(pl, pr, lk1, lk2) * min(after - i + 1, vr - i + 1);
    i = after + 1;
    after = nextNum(after);
    lk1 = lk2;
    if (lk1 > pr) break;
    lk2 = nextNum(lk2);
  }
  mau = (pr - pl + 1);
  mau *= (vr - vl + 1);
  kq = sum / mau;
  printf("%.12lf", kq);

}
