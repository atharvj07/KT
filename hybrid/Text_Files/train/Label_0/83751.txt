def main():
	(n, m, k) = [int(x) for x in input().split()]
	size = 0
	h = [0] * m
	for i in range(n):
		M = input()
		length = 0
		for j in range(m):
			if M[j] == '.':
				length += 1
				h[j] += 1
				continue
			if length >= k:
				size += length - k + 1
			if h[j] >= k > 1:
				size += h[j] - k + 1
			length = 0
			h[j] = 0
		if length >= k:
			size += length - k + 1
	for i in range(m):
		if h[i] >= k > 1:
			size += h[i] - k + 1
	print(size)
main()
