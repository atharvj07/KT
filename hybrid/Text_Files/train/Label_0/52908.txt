import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.StringTokenizer;

/**
 * @author b
 */
public class A {

    private static Scanner scanner;
    private static PrintWriter writer;
    private static ProbabilityStuff probabilityStuff;

    public static void main(String[] args) {
        scanner = new Scanner();
        writer = new PrintWriter(System.out);
        probabilityStuff = new ProbabilityStuff();
        int tests = scanner.nextInt();
        for (int i = 0; i < tests; i++) {
            solve();
        }
        writer.close();
    }

    private static void solve() {

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        if (n == k) {
            writer.println(1);
        } else if (n == 1 || k == 1){
            writer.println(k);
        } else if (n > k) {
            k = ((int) Math.ceil(n * 1.0 / k)) * k;
            writer.println((int) Math.ceil(k * 1.0 / n));
        }else {
            writer.println((int) Math.ceil(k * 1.0 / n));
        }

    }

   static void sortInts(int[] toSort) {
	   List<Integer> ints = new ArrayList<>(toSort.length);
	   for (int n : toSort) ints.add(n);
	   Collections.sort(ints);
	   for (int i=0; i < toSort.length; i++) {
	   	toSort[i] = ints.get(i);
	   }
   }

    // Props to SecondThread
    static class Scanner {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer("");

        String next() {
            while (!tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return tokenizer.nextToken();
        }

        long[] nextLongArr(int num) {
            long[] vals = new long[num];
            for (int i = 0; i < num; i++)
                vals[i] = nextLong();
            return vals;
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        int[] nextIntArr(int numberOfElements) {
            int[] values = new int[numberOfElements];
            for (int i = 0; i < numberOfElements; i++)
                values[i] = nextInt();
            return values;
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }

    static class ProbabilityStuff {

        static long MOD = 1000000007;

        static long mul(long a, long b) {
            return a * b % MOD;
        }

        static long add(long a, long b) {
            return (a + b) % MOD;
        }

        static long fact(long n) {
        	if (n == 1) return 1;
        	return mul(n , fact(n-1));
		}

        static long exp(long base, long pow) {
            if (pow == 0) return 1;
            long half = exp(base, pow / 2);
            if (pow % 2 == 0) return mul(half, half);
            return mul(half, mul(half, base));
        }

        static long modInv(long x) {
            return exp(x, MOD - 2);
        }

        static long nCk(int n, int k) {
            return mul(fact(n), mul(modInv(fact(k)), modInv(fact(n-k))));
        }

    }

}
