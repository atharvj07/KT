t = int(input())
A = []
for x in range(t):
	n = input()
	a = list(map(int, input().split()))
	A.append(a)
for a in A:
	(i, j) = (0, len(a))
	if j in [0, 1, 2]:
		print('YES')
	else:
		result = False
		j -= 1
		(left, right) = (None, None)
		while i < j:
			if a[i] == a[j]:
				i += 1
				j -= 1
				result = True
				continue
			elif not left:
				left = a[i]
				(l, k) = (i + 1, j)
				while l < k:
					if a[l] == a[k]:
						l += 1
						k -= 1
						result = True
						continue
					elif a[l] == left:
						l += 1
						result = True
						continue
					elif a[k] == left:
						k -= 1
						result = True
						continue
					else:
						result = False
						break
			elif not right and (not result):
				right = a[j]
				(l, k) = (i, j - 1)
				while l < k:
					if a[l] == a[k]:
						l += 1
						k -= 1
						result = True
						continue
					elif a[l] == right:
						l += 1
						result = True
						continue
					elif a[k] == right:
						k -= 1
						result = True
						continue
					else:
						result = False
						break
			else:
				break
		if result:
			print('YES')
		else:
			print('NO')
