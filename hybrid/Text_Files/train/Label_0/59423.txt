import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        FastScanner in = new FastScanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        new Main().run(in, out);
        out.close();
    }


    long[] seg = new long[1000000];
    int[] lo = new int[1000000];
    int[] hi = new int[1000000];
    boolean flipped = false;
    int start;
    int end;

    void run(FastScanner in, PrintWriter out) {

        int n = in.nextInt();
        bounds(1, 0, n-1);
        start = 0;
        end = n-1;

        for (int i = 0; i < n; i++) update(1, 1, i);

        int q = in.nextInt();
        while (q-- > 0) {
            int qType = in.nextInt();
            if (qType == 2) {
                // query
                int l = in.nextInt();
                int r = in.nextInt();
                if (!flipped) {
                    out.println(sum(1, start+l, start+r-1));
                } else {
                    out.println(sum(1, end-r+1, end-l));
                }

            } else {
                int f = in.nextInt();
                if (flipped) {
                    if (2*f > end - start + 1) {
                        flipped = !flipped;
                        foldHead(end-start+1-f);
                        start += (end-start+1-f);
                    } else {
                        foldTail(f);
                        end -= f;
                    }
                } else {
                    if (2*f > end-start+1) {
                        flipped = !flipped;
                        foldTail(end-start+1-f);
                        end -= (end-start+1-f);
                    } else {
                        foldHead(f);
                        start += f;
                    }
                }
            }

            // System.out.print("[ ");
            // for (int i = 0; i < n; i++) {
            //     System.out.print(sum(1, i, i) + " ");
            // }
            // System.out.println("]");
            // System.out.println();
        }


    }

    void foldHead(int d) {
        int c = 2*d-1;
        for (int i = 0; i < d; i++) {
            long val = sum(1, start+i, start+i);
            update(1, val, start+c-i);
        }
    }
    void foldTail(int d) {
        int c = 2*d-1;
        for (int i = 0; i < d; i++) {
            long val = sum(1, end-i, end-i);
            update(1, val, end-c+i);
        }
    }

    void bounds(int i, int l, int h) {
        lo[i] = l;
        hi[i] = h;
        if (l == h) return;
        int m = (l + h) / 2;
        bounds(2*i, l, m);
        bounds(2*i+1, m+1, h);
    }

    void update(int i, long val, int a) {
        if (a < lo[i] || a > hi[i]) return;
        if (lo[i] == hi[i] && a == lo[i]) {
            seg[i] += val;
            return;
        }

        update(2*i, val, a);
        update(2*i+1, val, a);
        seg[i] = seg[2*i] + seg[2*i+1];
    }

    long sum(int i, int a, int b) {
        if (b < lo[i] || a > hi[i]) return 0;
        if (a <= lo[i] && hi[i] <= b) {
            return seg[i];
        }
        return sum(2*i, a, b) + sum(2*i+1, a, b);
    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        public FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            st = null;
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }
}
