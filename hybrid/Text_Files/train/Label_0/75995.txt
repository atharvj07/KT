n = int(input())
jugos = []
A = []
B = []
C = []
AB = []
AC = []
BC = []
ABC = []
for i in range(n):
	a = input().split()
	if a[1] == 'A':
		if len(A) != 0:
			if A[0][0] > int(a[0]):
				A[0] = [int(a[0]), a[1]]
		else:
			A.append([int(a[0]), a[1]])
	elif a[1] == 'B':
		if len(B) != 0:
			if B[0][0] > int(a[0]):
				B[0] = [int(a[0]), a[1]]
		else:
			B.append([int(a[0]), a[1]])
	elif a[1] == 'C':
		if len(C) != 0:
			if C[0][0] > int(a[0]):
				C[0] = [int(a[0]), a[1]]
		else:
			C.append([int(a[0]), a[1]])
	elif a[1] == 'AB' or a[1] == 'BA':
		if len(AB) != 0:
			if AB[0][0] > int(a[0]):
				AB[0] = [int(a[0]), a[1]]
		else:
			AB.append([int(a[0]), a[1]])
	elif a[1] == 'AC' or a[1] == 'CA':
		if len(AC) != 0:
			if AC[0][0] > int(a[0]):
				AC[0] = [int(a[0]), a[1]]
		else:
			AC.append([int(a[0]), a[1]])
	elif a[1] == 'BC' or a[1] == 'CB':
		if len(BC) != 0:
			if BC[0][0] > int(a[0]):
				BC[0] = [int(a[0]), a[1]]
		else:
			BC.append([int(a[0]), a[1]])
	elif a[1] == 'ABC' or a[1] == 'ACB' or a[1] == 'BAC' or (a[1] == 'BCA') or (a[1] == 'CAB') or (a[1] == 'CBA'):
		if len(ABC) != 0:
			if ABC[0][0] > int(a[0]):
				ABC[0] = [int(a[0]), a[1]]
		else:
			ABC.append([int(a[0]), a[1]])
if len(A) != 0:
	jugos.append(A[0])
if len(B) != 0:
	jugos.append(B[0])
if len(C) != 0:
	jugos.append(C[0])
if len(AB) != 0:
	jugos.append(AB[0])
if len(AC) != 0:
	jugos.append(AC[0])
if len(BC) != 0:
	jugos.append(BC[0])
if len(ABC) != 0:
	jugos.append(ABC[0])

def hacer(jugos, vitaminas, precio):
	if len(solucion) != 0:
		if precio > solucion[0]:
			return False
	if len(vitaminas) == 3:
		if len(solucion) == 0:
			solucion.append(precio)
		elif precio < solucion[0]:
			solucion[0] = precio
		return
	else:
		for i in range(len(jugos)):
			if len(jugos[i][1]) == 1:
				if jugos[i][1] not in vitaminas:
					vitaminas.append(jugos[i][1])
					hacer(jugos[:i] + jugos[i + 1:], vitaminas, precio + jugos[i][0])
					vitaminas.pop()
			else:
				no = True
				for r in jugos[i][1]:
					if r not in vitaminas:
						no = False
						break
				if not no:
					tt = 0
					for x in jugos[i][1]:
						if x not in vitaminas:
							vitaminas.append(x)
							tt += 1
					hacer(jugos[:i] + jugos[i + 1:], vitaminas, precio + jugos[i][0])
					for _ in range(tt):
						vitaminas.pop()
solucion = []
hacer(jugos, [], 0)
if len(solucion) == 0:
	print(-1)
else:
	print(solucion[0])
