import sys, math, queue
MOD = 10 ** 9 + 7
(n, m) = map(int, input().split())
if m == 0:
	print(1)
	exit()
c = [1 for i in range(n)]
i = 0
while m > 0:
	if c[i] == 1:
		c[i] = 0
		m -= 1
		i = (i + 2) % n
	else:
		i = (i + 1) % n
ans = 0
for i in range(n):
	if c[i] == 1 and c[(i + 1) % n] == 0:
		ans += 1
print(ans)
