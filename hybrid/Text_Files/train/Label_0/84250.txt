am = int(input())
for _ in range(am):
	(n, k, d) = list(map(int, input().split()))
	tv = list(map(int, input().split()))
	min_ = k
	for i in range(len(tv) - d + 1):
		minNow = 0
		bought = []
		for g in range(i, i + d):
			if not tv[g] in bought:
				bought.append(tv[g])
				minNow += 1
				if minNow >= min_:
					break
		else:
			min_ = minNow
	print(min_)
