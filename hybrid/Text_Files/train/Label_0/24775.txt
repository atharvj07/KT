#include<iostream>
#include<vector>
#include<algorithm>
#include<tuple>
#include<cmath>

using namespace std;

int main(){
  int D;
  cin>>D;
  while(D--){
    int N;
    cin>>N;
    vector<int> B{-(1<<29),0};
    vector<int> H{50,50};
    while(N--){
      int b,h;
      cin>>b>>h;
      B.push_back(b);
      H.push_back(h);
    }
    B.insert(end(B),{100,1<<29});
    H.insert(end(H),{50,50});
    int M;
    cin>>M;
    vector<int> F(M),A(M);
    for(int i=0;i<M;i++){
      cin>>F[i]>>A[i];
    }
    int L;
    cin>>L;
    vector<tuple<int,int,int,double> > PT;
    for(int i=0;i<L;i++){
      int p,t;
      cin>>p>>t;
      PT.emplace_back(t,p,i,0);
    }
    sort(begin(PT),end(PT));
    int ct=0;
    vector<double> h(B.size()-1);
    for(auto &e:PT){
      int dt=get<0>(e)-ct;
      for(int i=0;i<M;i++){
	int x=lower_bound(begin(B),end(B),F[i])-begin(B)-1;
	double ra=dt*A[i];
	for(;;){
	  bool rltl=(H[x]>H[x+1]);
	  int lx=x+rltl;
	  double dh=H[lx]-h[x];
	  if(dh<1e-9){
	    int sx=x+(rltl?1:-1);
	    if(fabs(h[x]-h[sx])>1e-9){
	      x=sx;
	    }else{
	      int hx=max(x,sx);
	      B.erase(begin(B)+hx);
	      H.erase(begin(H)+hx);
	      h.erase(begin(h)+hx);
	      x=min(x,sx);
	    }
	  }else{
	    long long a=(B[x+1]-B[x])*30LL;
	    double cap=a*dh;
	    if(ra<cap){
	      h[x]+=ra/a;
	      break;
	    }else{
	      h[x]=H[lx];
	      ra-=cap;
	    }
	  }
	}
      }
      ct=get<0>(e);
      int ax=lower_bound(begin(B),end(B),get<1>(e))-begin(B)-1;
      get<3>(e)=h[ax];
    }
    for(int i=0;i<L;i++){
      for(auto e:PT){
	if(get<2>(e)==i){
	  cout<<fixed<<get<3>(e)<<endl;
	}
      }
    }
  }
}
