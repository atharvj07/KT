(n, k) = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
if n < a[0] or n > a[k - 1]:
	print(-1)
else:
	b = []
	b.append(a[0])
	for i in a:
		if i != b[len(b) - 1]:
			b.append(i)
	d = {}
	for i in range(len(b)):
		b[i] -= n
		d[b[i]] = 1
	ans = 1
	while 0 not in d:
		d1 = {}
		for i in d:
			for j in b:
				if -1001 < i + j < 1001:
					d1[i + j] = 1
		d = d1
		ans += 1
	print(ans)
