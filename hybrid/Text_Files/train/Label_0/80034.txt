import operator

def main():

	def solve(s):
		mn = 20
		n = len(s)
		bn = n - 2
		foundFive = False
		foundZero = False
		for i in range(n - 1, -1, -1):
			if s[i] == '0':
				if foundZero:
					mn = bn - i
					break
				foundZero = True
			elif s[i] == '5':
				if foundZero:
					mn = bn - i
					break
				foundFive = True
			elif s[i] in ['2', '7']:
				if foundFive:
					mn = bn - i
					break
		return str(mn)
	t = int(input())
	for _ in range(t):
		s = input().strip()
		print(solve(s))
import sys, threading
input = sys.stdin.readline
thread = threading.Thread(target=main)
thread.start()
thread.join()
