#include <bits/stdc++.h>
using namespace std;
const int MOD = 1000000007;
const int MAX = 256;
int a[MAX], ans[MAX], tmp[MAX];
struct mat {
  int a[MAX][MAX], n;
  mat(int _n = 1, int b = 0) {
    n = _n;
    for (int i = 0; i < n; i++)
      for (int j = 0; j < n; j++) a[i][j] = (i == j ? b : 0);
  }
  int* operator[](int x) { return a[x]; }
  mat operator*(mat b) {
    mat ans(b.n);
    for (int i = 0; i < n; i++)
      for (int j = 0; j < n; j++)
        for (int k = 0; k < n; k++)
          ans[i][j] = (ans[i][j] + 1LL * a[i][k] * b[k][j]) % MOD;
    return ans;
  }
  mat operator*=(mat b) { return *this = *this * b; }
};
mat power(mat a, int b) {
  mat ans(a.n, 1);
  while (b) {
    if (b & 1) ans *= a;
    b >>= 1;
    a *= a;
  }
  return ans;
}
int main() {
  ios::sync_with_stdio(false);
  for (int i = 0; i < 7; i++) cin >> a[i];
  ans[1] = 1;
  for (int i = 0; i < 7; i++) {
    mat t(1 << (i + 1));
    for (int cur = 0; cur < (1 << (i + 1)); cur++)
      for (int prev = 0; prev < (1 << (i + 1)); prev++)
        for (int _mask = 0; _mask < (1 << i); _mask++) {
          int mask = (1 << i) | _mask;
          mask <<= 1;
          mask |= 1;
          bool ok = true;
          for (int j = 0; j < i + 1; j++)
            if ((1 << j) & cur && (1 << j) & prev && (1 << j) & mask &&
                (1 << (j + 1)) & mask) {
              ok = false;
              break;
            }
          if (ok) t[cur][prev]++;
        }
    t = power(t, a[i]);
    memset(tmp, 0, sizeof(tmp));
    for (int mask = 0; mask < (1 << (i + 1)); mask++)
      for (int j = 0; j < t.n; j++)
        tmp[mask] = (tmp[mask] + 1LL * t[mask][j] * ans[j]) % MOD;
    memset(ans, 0, sizeof(ans));
    for (int mask = 0; mask < (1 << (i + 1)); mask++)
      ans[mask | (1 << (i + 1))] = tmp[mask];
  }
  cout << ans[(1 << 8) - 1] << endl;
  return 0;
}
