#include <bits/stdc++.h>
using namespace std;
int main(int argc, char* argv[]) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  string s, t, p;
  cin >> s >> t;
  p = s;
  int compt = 0, test = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] != t[i]) {
      compt++;
      if (test == 0) {
        p[i] = s[i];
        test = 1;
      } else {
        p[i] = t[i];
        test = 0;
      }
    }
  }
  if (compt % 2 != 0)
    cout << "impossible" << endl;
  else
    cout << p << endl;
  return 0;
}
