def main():
	n = int(input())
	n_a = []
	inp_str = input().split(' ')
	sum_of_a = 0
	for i in range(1, len(inp_str) + 1):
		sum_of_a += int(inp_str[i - 1])
		n_a.append([i, int(inp_str[i - 1])])
	first_elem = n_a.pop(0)
	n_a.sort(key=lambda x: x[1], reverse=True)
	n_a.insert(0, first_elem)
	if n_a[0][1] == 0:
		return -1
	if sum_of_a < n - 1:
		return -1
	(stack, current_n, stack_n, flag, out_pairs) = ([], 0, 0, False, [])
	while True:
		if n_a[current_n][0] == 1:
			stack.append(n_a[current_n])
			current_n = n_a[current_n][0]
			continue
		stack_n = stack[0][0]
		current_a = current_n + stack[0][1]
		stack.pop(0)
		if current_a >= len(n_a):
			current_a = len(n_a)
			flag = True
		for n_tmp in range(current_n, current_a):
			out_pairs.append([stack_n, n_a[n_tmp][0]])
			if n_a[n_tmp][1] != 0:
				stack.append(n_a[n_tmp])
		current_n = current_a
		if flag == True:
			break
	return out_pairs
res = main()
if res == -1:
	print('-1')
else:
	print(len(res))
	for elem in res:
		print(str(elem[0]) + ' ' + str(elem[1]))
