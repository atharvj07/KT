#include <bits/stdc++.h>

using namespace std;

#define PI acos(-1)

struct xyz{
  double x,y,z;
};

double abs(xyz a) {
  return sqrt(a.x*a.x+a.y*a.y+a.z*a.z);
}

double product(xyz a,xyz b) {
  return a.x*b.x+a.y*b.y+a.z*b.z;
}

int main() {
  int n,m;
  while(cin >> n,n) {
    vector<xyz> s(n);
    for(auto&a:s) cin >> a.x>>a.y>>a.z;
    cin >> m;
    vector<pair<xyz,double> > t(m);
    for(auto &a:t)
      cin >> a.first.x>>a.first.y>>a.first.z >> a.second;
    int ans=0;
    for(int i=0;i<n;++i){
      for(int j=0;j<m;++j){
        double co=product(s[i],t[j].first);
        co/=abs(s[i])*abs(t[j].first);
        double theta=acos(co);
        if(theta>PI) theta-=PI;
        if(theta<=t[j].second) {
          ans++;
          break;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
