#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int maxn = 1e6 + 5;
long long dp[maxn];
int n, k;
void pre() {
  int cur = 1;
  int left = 3 * k - 3;
  vector<long long> sep;
  while (left) {
    sep.push_back(min(left, cur));
    cur <<= 1;
    left -= sep.back();
  }
  vector<long long> first;
  cur = 1;
  memset(dp, 0xc4, sizeof(dp));
  dp[0] = 0;
  for (int i = 0; i < 6; i++) {
    long long x;
    cin >> x;
    first.push_back(x);
    vector<pair<long long, long long>> v;
    for (auto &j : sep) {
      v.push_back({x * j, j * cur * 3});
    }
    for (auto &j : v) {
      for (int k = n; k >= j.second; k--) {
        dp[k] = max(dp[k], dp[k - j.second] + j.first);
      }
    }
    cur *= 10;
  }
  cur = 1;
  for (int i = 0; i < 6; i++) {
    for (int j = n; j >= 0; j--) {
      for (int k = 1; k <= 9; k++) {
        if (j >= cur * k) {
          if (k % 3 == 0) {
            dp[j] = max(dp[j], dp[j - cur * k] + first[i] * k / 3);
          } else {
            dp[j] = max(dp[j], dp[j - cur * k]);
          }
        }
      }
    }
    cur *= 10;
  }
}
void solve() {
  cin >> k;
  n = 999999;
  pre();
  int q;
  cin >> q;
  while (q--) {
    int x;
    cin >> x;
    cout << dp[x] << '\n';
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int c = 0;
  int t;
  if (!c) {
    t = 1;
  } else {
    cin >> t;
  }
  while (t--) {
    solve();
  }
}
