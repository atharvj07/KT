#include <bits/stdc++.h>
using namespace std;
void type(int n, int a, int b) {
  int now = -1;
  string s = "";
  for (long long int i = 0; i < a; i++) {
    if (b != 0) now++, b--;
    s += char('a' + now);
  }
  for (long long int i = 0; i < n; i++) cout << s[i % a];
  cout << endl;
}
const int mx = 1e6;
int t, n, cnt;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> t;
  while (t--) {
    cin >> n;
    while (n % 3 == 0) {
      if (n % 6 == 0)
        n /= 6;
      else if (n % 3 == 0)
        n /= 3, cnt++;
      cnt++;
    }
    if (n >= 2) cnt = -1;
    cout << cnt << endl;
    cnt = 0;
  }
  return 0;
}
