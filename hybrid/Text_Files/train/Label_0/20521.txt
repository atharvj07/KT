#include <bits/stdc++.h>
using namespace std;

const int MAX = 1 << 17;

int main()
{
	ios::sync_with_stdio(false), cin.tie(0);
	int n, q;
	cin >> n >> q;
	vector<int> v(MAX << 1, INT_MAX);
	while (q--) {
		int com, x, y;
		cin >> com >> x >> y;
		if (com == 0) {
			for (v[x += MAX] = y; x >>= 1;) {
				v[x] = min(v[x << 1], v[(x << 1) | 1]);
			}
		}
		else {
			int res = INT_MAX;
			for (x += MAX, y += MAX + 1; x < y; x >>= 1, y >>= 1) {
				if (x & 1) res = min(res, v[x++]);
				if (y & 1) res = min(res, v[--y]);
			}
			printf("%d\n", res);
		}
	}
	return 0;
}
