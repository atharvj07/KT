import heapq
(n, k) = map(int, input().split())
des = list(map(int, input().split()))
g = int(input())
cost = list(map(int, input().split()))
cos = 0
flag = 0
cur = k
dec = {}
hp = []
heapq.heapify(hp)
for i in range(n):
	dec[i] = 0
for i in range(n):
	if k + g * (i + 1) < des[i]:
		flag = 1
		break
	else:
		heapq.heappush(hp, cost[i])
		while cur < des[i]:
			cur += g
			cos += heapq.heappop(hp)
if flag == 1:
	print(-1)
else:
	print(cos)
