#include <bits/stdc++.h>
using namespace std;

int main() {
	const __uint128_t INF = 1e19;
	const __uint128_t MAX = 1e18;
	
	int n;
	cin >> n;

	__uint128_t val = 1;
	for (int i = 0; i < n; i++) {
		long long e;
		cin >> e;
		val *= e;
		val = min(val, INF);
	}
	
	if (val > MAX) cout << -1 << '\n';
	else cout << (long long)(val) << '\n';
	
	return 0;
}
