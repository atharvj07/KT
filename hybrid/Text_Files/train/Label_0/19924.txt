int main()
{

	cin.tie(nullptr);
	ios::sync_with_stdio(false);

	ll a, b, c, n; cin >> a >> b >> c >> n;
	ll res = (a * b + b * c + c * a) * 2;
	vector<tuple<ll, ll, ll>> vt(n);
	for (int i = 0; i < n; i++) {
		ll x, y, z; cin >> x >> y >> z;
		vt[i] = make_tuple(x, y, z);
	}
	sort(vt.begin(), vt.end());
	for (auto p : vt) {
		ll x, y, z; tie(x, y, z) = p;
		for (int i = 0; i < 6; i++) {
			ll nx = x + dx[i];
			ll ny = y + dy[i];
			ll nz = z + dz[i];
			bool ok = true;
			if (nx < 0 or nx >= a or ny < 0 or ny >= b or nz < 0 or nz >= c) res--;
			else if (!binary_search(vt.begin(), vt.end(), make_tuple(nx, ny, nz))) res++;
			
		}
	}
	cout << res << endl;
}
