#include <bits/stdc++.h>
using namespace std;
vector<long long> factorize(long long x) {
  vector<long long> v;
  long long i = 2;
  while (i * i <= x) {
    if (x % i == 0) {
      v.push_back(i);
      while (x % i == 0) {
        x /= i;
      }
    }
    i++;
  }
  if (x > 0) v.push_back(x);
  return v;
}
const long long MX = 1e6 + 8;
long long A[MX];
const long long INF = 1e18;
long long acum1[MX];
long long acum2[MX];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n;
  cin >> n;
  long long sum = 0;
  for (long long i = 0; i < (long long)n; i++) {
    cin >> A[i];
    sum += A[i];
  }
  if (sum == 1) {
    cout << -1 << endl;
    return 0;
  }
  vector<long long> v = factorize(sum);
  long long ans = INF;
  for (long long x : v) {
    if (x == 1) continue;
    long long c = 0;
    long long aux = 0;
    vector<pair<long long, long long>> pos;
    for (long long ind = 0; ind < (long long)n; ind++) {
      long long r = A[ind] % x;
      if (c + r < x) {
        c += r;
        pos.push_back({ind, r});
      } else {
        pos.push_back({ind, x - c});
        r -= (x - c);
        long long better = INF;
        long long sz = pos.size();
        long long ac1 = 0;
        long long ac2 = 0;
        for (long long i = sz - 1; i >= 0; i--) {
          acum1[i] = ac1;
          acum2[i] = ac2;
          ac1 += pos[i].second * pos[i].first;
          ac2 += pos[i].second;
        }
        ac1 = 0;
        ac2 = 0;
        for (long long i = 0; i < sz; i++) {
          better =
              min(better, ac1 + acum1[i] + pos[i].first * (ac2 - acum2[i]));
          ac1 -= pos[i].second * pos[i].first;
          ac2 += pos[i].second;
        }
        aux += better;
        pos.clear();
        c = 0;
        if (r != 0) {
          pos.push_back({ind, r});
          c += r;
        }
      }
    }
    ans = min(ans, aux);
  }
  cout << ans << endl;
  return 0;
}
