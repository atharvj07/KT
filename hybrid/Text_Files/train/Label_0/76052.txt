n = int(input())
line = list(map(int, input().split()))
(max1, max2) = (-1, -1)
ar = []
colors = [False for i in range(n)]
for i in range(n):
	if colors[i] == True:
		continue
	x = line[i] - 1
	colors[i] = True
	cnt = 1
	while colors[x] == False:
		colors[x] = True
		x = line[x] - 1
		cnt += 1
	if cnt >= max1:
		ar.append(max2)
		max2 = max1
		max1 = cnt
	elif cnt > max2:
		ar.append(max2)
		max2 = cnt
	else:
		ar.append(cnt)
if max2 == -1:
	max2 = 0
ans = (max1 + max2) ** 2
for elem in ar:
	if elem != -1:
		ans += elem ** 2
print(ans)
