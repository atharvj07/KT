import sys
from sys import stdin

def modfac(n, MOD):
	f = 1
	factorials = [1]
	for m in range(1, n + 1):
		f *= m
		f %= MOD
		factorials.append(f)
	inv = pow(f, MOD - 2, MOD)
	invs = [1] * (n + 1)
	invs[n] = inv
	for m in range(n, 1, -1):
		inv *= m
		inv %= MOD
		invs[m - 1] = inv
	return (factorials, invs)

def modnCr(n, r):
	return fac[n] * inv[n - r] * inv[r] % mod
mod = 10 ** 9 + 7
(fac, inv) = modfac(300000, mod)
tt = int(stdin.readline())
ANS = []
for loop in range(tt):
	(n, l, r) = map(int, stdin.readline().split())
	ans = 0
	dmax = min(1 - l, r - n)
	if dmax >= 0:
		if n % 2 == 0:
			ans += dmax * modnCr(n, n // 2) % mod
		else:
			ans += 2 * dmax * modnCr(n, n // 2) % mod
	nd = dmax + 1
	while True:
		lrist = max(0, l + nd - 1)
		rrist = max(0, n - (r - nd))
		if lrist + rrist > n or min(lrist, rrist) > (n + 1) // 2:
			break
		if n % 2 == 1:
			uprem = n // 2 - lrist
			downrem = (n + 1) // 2 - rrist
			if min(uprem, downrem) >= 0:
				ans += modnCr(uprem + downrem, uprem)
			uprem = (n + 1) // 2 - lrist
			downrem = n // 2 - rrist
			if min(uprem, downrem) >= 0:
				ans += modnCr(uprem + downrem, uprem)
		else:
			uprem = n // 2 - lrist
			downrem = n // 2 - rrist
			if min(uprem, downrem) >= 0:
				ans += modnCr(uprem + downrem, uprem)
		nd += 1
	ANS.append(str(ans % mod))
print('\n'.join(ANS))
