#include <bits/stdc++.h>
using namespace std;
long long const mod = (long long)1e9 + 7;
const int MAX = 1000001;
long long Fact[MAX];
long long Pw(int a, int b) {
  if (!b) return 1;
  long long ret(1);
  ret = Pw(a, b / 2);
  return (((ret * ret) % mod) * (b % 2 == 1 ? a : 1) % mod) % mod;
}
long long inv(int x) { return Pw(x, mod - 2) % mod; }
long long calc(int n, int k) {
  return (((Fact[n] * inv(Fact[k]) % mod) * inv(Fact[n - k]))) % mod;
}
int a, b, n;
bool ok(int x) {
  while (x) {
    if (x % 10 != a && x % 10 != b) return 0;
    x /= 10;
  }
  return true;
}
int main() {
  Fact[0] = 1;
  for (int i = 1; i < MAX; i++) Fact[i] = (Fact[i - 1] * i) % mod;
  cin >> a >> b >> n;
  long long ans(0);
  for (int i = 0; i <= n; i++) {
    int j = n - i;
    if (!ok(i * a + j * b)) continue;
    ans += calc(n, i);
    ans %= mod;
  }
  cout << ans << endl;
  return 0;
}
