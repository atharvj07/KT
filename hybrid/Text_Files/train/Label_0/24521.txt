#include <iostream>
#include <algorithm>
using namespace std;

int a, b;

int greedy(int x) {
	return (x / b) + (x % b) / a + ((x % b) % a);
}

int f(int x) { return x / a + (x % a); }

int best(int x) {
	int i;
	int ret = x;
	
	for (i = 0; i * b <= x; i++) {
		int res = i + f(x - i * b);
		ret = min(ret, res);
	}
	return ret;
}

int main() {
	cin >> a >> b;
	
	if (((b + a - 1) / a) * (a - 1) + 1 <= b) { cout << -1 << endl; }
	else { cout << ((b - 1) / a + 1) * a << endl; }
	return 0;
}
