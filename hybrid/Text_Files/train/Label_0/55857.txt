/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        DBuyATicket solver = new DBuyATicket();
        solver.solve(1, in, out);
        out.close();
    }

    static class DBuyATicket {
        ArrayList<Integer>[] g;
        ArrayList<Long>[] w;
        int n;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            n = in.nextInt();
            int m = in.nextInt();
            g = new ArrayList[n + 1];
            w = new ArrayList[n + 1];
            for (int i = 0; i <= n; i++) {
                g[i] = new ArrayList<>();
                w[i] = new ArrayList<>();
            }

            for (int i = 0; i < m; i++) {
                int u = in.nextInt();
                int v = in.nextInt();
                long ww = in.nextLong();
                g[u].add(v);
                g[v].add(u);
                w[u].add(2 * ww);
                w[v].add(2 * ww);
            }
            long[] a = in.nextLongArray(n);
            for (int i = 0; i < n; i++) {
                g[0].add(i + 1);
                w[0].add(a[i]);
            }
            long[] ans = dijkstras(0);
            for (int i = 1; i <= n; i++) {
                out.print(ans[i] + " ");
            }
        }

        long[] dijkstras(int src) {
            PriorityQueue<Node> P = new PriorityQueue<>(new Comparator<Node>() {

                public int compare(Node o1, Node o2) {
                    return Long.compare(o1.dist, o2.dist);
                }
            });
            long[] dist = new long[n + 1];
            Arrays.fill(dist, (long) 1e17);
            dist[src] = 0;
            P.add(new Node(src, 0));
            boolean[] vis = new boolean[n + 1];

            while (!P.isEmpty()) {
                Node X = P.poll();
                if (vis[X.n]) continue;
                vis[X.n] = true;
                for (int i = 0; i < w[X.n].size(); i++) {
                    if (X.dist + w[X.n].get(i) < dist[g[X.n].get(i)]) {
                        dist[g[X.n].get(i)] = X.dist + w[X.n].get(i);
                        P.add(new Node(g[X.n].get(i), dist[g[X.n].get(i)]));
                    }
                }
            }
            return dist;
        }

        class Node {
            int n;
            long dist;

            Node(int nn, long dd) {
                n = nn;
                dist = dd;
            }

        }

    }

    static class InputReader {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        public int read() {
            if (numChars == -1)
                throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public int nextInt() {
            int c = read();
            while (isSpaceChar(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public long nextLong() {
            int c = read();
            while (isSpaceChar(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            long res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        private boolean isSpaceChar(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public long[] nextLongArray(int size) {
            long[] array = new long[size];
            for (int i = 0; i < size; i++)
                array[i] = nextLong();
            return array;
        }

    }
}
