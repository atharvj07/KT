n = int(input())
a = list(map(int, input().split()))
b = []
for i in range(0, n, 2):
	b.append(a[i])
for i in range(1, n, 2):
	b.append(a[i])
su = 0
for i in range((n + 1) // 2):
	su += b[i]
ans = su
for i in range((n + 1) // 2, n + (n + 1) // 2):
	su += b[i % n] - b[(i - (n + 1) // 2) % n]
	ans = max(su, ans)
print(ans)
