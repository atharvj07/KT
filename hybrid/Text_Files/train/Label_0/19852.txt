#include<stdio.h>
#include<string.h>
#include<algorithm>
using namespace std;
char str[1024];
int v[26];
char to[]="AIZUNYAN";
int main(){
	scanf("%s",str);
	int n=strlen(str);
	for(int i=0;i<n-7;i++){
		for(int j=0;j<26;j++)v[j]=0;
		for(int j=0;j<8;j++)v[str[j+i]-'A']++;
		if(v[0]==2&&v[3]==1&&v[8]==1&&v['N'-'A']==2&&v['U'-'A']==1&&v['Y'-'A']==1){
			for(int j=0;j<8;j++){
				str[i++]=to[j];
			}i--;
		}
	}
	printf("%s\n",str);
}
