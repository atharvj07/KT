#include <bits/stdc++.h>
using namespace std;
int n;
long long w, v, u;
long long poi[22341][2];
long double t;
void loda() {
  cin >> n >> w >> v >> u;
  for (int i = 0; i < n; ++i) {
    cin >> poi[i][0] >> poi[i][1];
    poi[i + n][0] = poi[i][0];
    poi[i + n][1] = poi[i][1];
  }
}
bool makeTime0() {
  for (int i = 0; i < n; ++i)
    if (poi[i][1] * v < poi[i][0] * u) {
      return 0;
    }
  t = ((long double)w) / ((long double)u);
  return 1;
}
bool makeTime1() {
  for (int i = 0; i < n; ++i)
    if (poi[i][1] * v > poi[i][0] * u) {
      return 0;
    }
  t = ((long double)w) / ((long double)u);
  return 1;
}
void makeTime2() {
  int low = 0;
  for (int i = 0; i < n; ++i) {
    if (poi[low][1] > poi[i][1]) {
      low = i;
    }
  }
  int hi = low;
  for (int i = low; i < 2 * n; ++i) {
    if (poi[hi][0] < poi[i][0]) {
      hi = i;
    }
  }
  long long l1 = poi[low][0];
  while (low != hi && (poi[low + 1][1] - poi[low][1]) * v <=
                          (poi[low + 1][0] - poi[low][0]) * u) {
    l1 += poi[low + 1][0] - poi[low][0];
    ++low;
  }
  t = ((long double)l1) / ((long double)v);
  t += ((long double)(w - poi[low][1])) / ((long double)u);
}
int main() {
  ios_base::sync_with_stdio(0);
  cout << setprecision(20) << fixed;
  loda();
  if (makeTime0()) {
    cout << t << endl;
    return 0;
  }
  if (makeTime1()) {
    cout << t << endl;
    return 0;
  }
  makeTime2();
  cout << t << endl;
}
