bool sortcol( const vector<int>& v1, const vector<int>& v2 ) {
  return v1[1] < v2[1];
}

int main(){
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);

  int t;
  cin>>t;

  while(t--){

    string s;
    cin>>s;

    int n=s.length();
    int i=1,j=n-2;
    while(i<n){
      if(s[i]=='1' && s[i]==s[i-1])
        break;
      i++;
    }

    while(j>=0){
      if(s[j]=='0' && s[j]==s[j+1])
        break;
      j--;
    }

    if(i>j){
      cout<<"YES"<<endl;
    }
    else{
      cout<<"NO"<<endl;
    }


  }
}
