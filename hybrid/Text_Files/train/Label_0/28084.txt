#include <bits/stdc++.h>
using namespace std;
int a[200005];
int noteq[2000005];
int main() {
  int n, m;
  cin >> n >> m >> a[0];
  noteq[0] = -1;
  for (int i = 1; i < n; ++i) {
    scanf("%d", &a[i]);
    if (a[i] != a[i - 1])
      noteq[i] = i - 1;
    else
      noteq[i] = noteq[i - 1];
  }
  int l, r, x;
  int q;
  for (int i = 0; i < m; ++i) {
    scanf("%d%d%d", &l, &r, &x);
    --r;
    --l;
    if (a[r] == x)
      q = noteq[r];
    else
      q = r;
    if (q >= l)
      printf("%d\n", 1 + q);
    else
      printf("-1\n");
  }
}
