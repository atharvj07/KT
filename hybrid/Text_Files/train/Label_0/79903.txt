from math import ceil
nAndK = input().strip().split()
numberOfFriends = int(nAndK[0])
sheetsOfPaperNotebook = int(nAndK[1])
numRedBooks = ceil(numberOfFriends * 2 / sheetsOfPaperNotebook)
numGreenBooks = ceil(numberOfFriends * 5 / sheetsOfPaperNotebook)
numBlueBooks = ceil(numberOfFriends * 8 / sheetsOfPaperNotebook)
print(numRedBooks + numGreenBooks + numBlueBooks)
