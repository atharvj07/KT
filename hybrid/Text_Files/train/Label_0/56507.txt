import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(),sum=0;
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
            sum+=arr[i];
        }
        sum/=2;
        int ans=0;
        Arrays.sort(arr);
        int count=0;
        for(int i=n-1;i>=0;i--){
            ans+=arr[i];
            count++;
            if(ans>sum){
                break;
            }
        }
        System.out.println(count);
    }
}
