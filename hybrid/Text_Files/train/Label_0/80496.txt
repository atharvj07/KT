n = int(input())
s = input()
A = 0
G = 0
C = 0
T = 0
spis = []
vopr = 0
for i in range(n):
	spis = spis + [s[i]]
	if s[i] == 'A':
		A = A + 1
	elif s[i] == 'G':
		G = G + 1
	elif s[i] == 'C':
		C = C + 1
	elif s[i] == 'T':
		T = T + 1
	else:
		vopr = vopr + 1
mx = max(A, G, C, T)
ost = mx - A + (mx - G) + (mx - C) + (mx - T)
A_ost = mx - A
G_ost = mx - G
C_ost = mx - C
T_ost = mx - T
if (ost - vopr) % 4 == 0:
	A_ost = A_ost + abs(ost - vopr) / 4
	G_ost = G_ost + abs(ost - vopr) / 4
	C_ost = C_ost + abs(ost - vopr) / 4
	T_ost = T_ost + abs(ost - vopr) / 4
if ost == vopr or ((ost - vopr) % 4 == 0 and vopr > ost):
	for i in range(n):
		if spis[i] == '?':
			if A_ost > 0:
				spis[i] = 'A'
				A_ost = A_ost - 1
			elif G_ost > 0:
				spis[i] = 'G'
				G_ost = G_ost - 1
			elif C_ost > 0:
				spis[i] = 'C'
				C_ost = C_ost - 1
			elif T_ost > 0:
				spis[i] = 'T'
				T_ost = T_ost - 1
	print(''.join(spis))
else:
	print('===')
