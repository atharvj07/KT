import java.io.*;
import java.util.*;

public class D {

    private static int mod = 998244353;

    public static void main(String[] args) {
        FastReader in = new FastReader(System.in);
        //        FastReader in2 = new FastReader(new FileInputStream("input.txt"));
        PrintWriter out = new PrintWriter(System.out);
        //        PrintWriter out2 = new PrintWriter(new FileOutputStream("input.txt"));


        int n = in.nextInt();

        long[] fact = new long[n];
        fact[1] = 1;
        for (int i = 2; i < n; i++)
            fact[i] = (fact[i - 1] * i) % mod;

        int[] deg = new int[n + 1];
        for (int i = 0; i < n - 1; i++) {
            int v = in.nextInt();
            int u = in.nextInt();
            deg[v]++;
            deg[u]++;
        }

        long ans = 1;
        for (int v = 1; v <= n; v++)
            ans = (ans * fact[deg[v]]) % mod;
        ans = (ans * n) % mod;

        out.println(ans);




        out.close();
    }

    private static class FastReader {

        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(nextLine());

            return st.nextToken();
        }

        String nextLine() {
            String x = "";
            try {
                x = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return x;
        }
    }
}
