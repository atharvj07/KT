#include <bits/stdc++.h>
using namespace std;
using lld = int64_t;
class S {
 public:
  lld get_gcd(lld a, lld b) {
    if (b == 0) return a;
    return get_gcd(b, a % b);
  }
  lld get_lcm(const vector<lld> &arr) {
    int n = arr.size();
    lld ans = arr[0];
    for (int i = 1; i < n; i++) {
      ans = (((arr[i] * ans)) / (get_gcd(arr[i], ans)));
    }
    return ans;
  }
  string solve(int a, int b) {
    bool swapped = false;
    if (b < a) {
      std::swap(a, b);
      swapped = true;
    }
    lld lcm = get_lcm(vector<lld>({a, b}));
    vector<pair<lld, lld>> intervals;
    lld a_itr = 0;
    while (a_itr != lcm) {
      intervals.push_back({a_itr, a_itr + a});
      a_itr += a;
    }
    lld b_itr = 0;
    int intervals_size = intervals.size();
    for (int i = 0; i < intervals_size; ++i) {
      lld start = intervals[i].first;
      lld end = intervals[i].second;
      if (b_itr <= start) {
        b_itr += b;
      }
      if (start < b_itr && b_itr <= end) {
        intervals[i].second = b_itr;
        intervals.push_back({b_itr, end});
      }
    }
    lld tmp1 = 0;
    lld tmp2 = 0;
    intervals_size = intervals.size();
    for (int i = 0; i < intervals_size; ++i) {
      lld start = intervals[i].first;
      lld end = intervals[i].second;
      lld count = end - start;
      bool p2 = end % b == 0;
      if (p2) {
        tmp2 += count;
      } else {
        tmp1 += count;
      }
    }
    if (tmp1 == tmp2) {
      return "Equal";
    }
    bool dasha_wins = tmp1 > tmp2;
    if (swapped) {
      dasha_wins = !dasha_wins;
    }
    if (dasha_wins) {
      return "Dasha";
    } else {
      return "Masha";
    }
  }
  void solve() {
    int a, b;
    scanf("%d %d\n", &a, &b);
    string res = solve(a, b);
    printf("%s\n", res.c_str());
    fflush(stdout);
  }
};
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  S s;
  s.solve();
  return 0;
}
