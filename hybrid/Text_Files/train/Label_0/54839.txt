import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.io.StreamTokenizer;


public class VK_C {
    private static StreamTokenizer in = new StreamTokenizer(new BufferedReader(
            new InputStreamReader(System.in)));


    private static final PrintWriter out = new PrintWriter(System.out);

    
    public static void main(String[] args) {
        try {
            solve();
            out.flush();
        }
        catch (Exception e) {
            e.printStackTrace();
        }
    }

    
    private static void solve()  throws Exception {
        int t = readInt();
        for (int z = 0; z < t; z++) {
            doSolve();
            out.println();
        }
    }


    private static void doSolve() throws Exception {
        int m = readInt();
        int k = readInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = readInt();
        }
        int[][] p = new int[m-1][2];
        int firstUnhappy = -1;
        int unknown = 0;
        int unknownBeforeUnhappy = 0;
        for (int j = 0; j < m-1; j++) {
            p[j][0] = readInt();
            p[j][1] = readInt();
            if (firstUnhappy < 0 && p[j][1] == 1) {
                firstUnhappy = j;
            }
            if (p[j][0] == 0) {
                unknown++;
                if (firstUnhappy < 0) {
                    unknownBeforeUnhappy++;
                }
            }
        }
        
        if (firstUnhappy < 0) {
            for (int j = 0; j < m-1; j++) {
                if (p[j][0] > 0) a[p[j][0]-1]--;
            }
            for (int i = 0; i < k; i++) {
                out.print(a[i] > unknown ? 'N' : 'Y');
            }
        }
        else {
            int[] b = new int[k];
            for (int i = 0; i < k; i++) {
                b[i] = a[i];
            }
            for (int j = 0; j < m-1; j++) {
                if (p[j][0] > 0) b[p[j][0]-1]--;
            }
            
            boolean[] endedOnUnhappy = new boolean[k];
            for (int i = 0; i < k; i++) {
                endedOnUnhappy[i] = b[i] <= unknownBeforeUnhappy;
            }
            for (int j = firstUnhappy; j < m-1; j++) {
                if (p[j][0] > 0) endedOnUnhappy[p[j][0]-1] = false;
            }
            
            int minBeforeUnhappy = Integer.MAX_VALUE;
            for (int i = 0; i < k; i++) {
                if (endedOnUnhappy[i]) minBeforeUnhappy = Math.min(minBeforeUnhappy, b[i]);
            }
            int leftUnknown = unknown - minBeforeUnhappy;
            for (int i = 0; i < k; i++) {
                if (endedOnUnhappy[i]) {
                    out.print('Y');
                } else if (b[i] > leftUnknown) {
                    out.print('N');
                } else out.print('Y');
            }
            
        }
    }


    private static int readInt() throws IOException {
        in.nextToken();
        return (int) in.nval;
    }

    private static char readChar() throws IOException {
        in.nextToken();
        return in.sval.charAt(0);
    }

    private static String readWord() throws IOException {
        in.nextToken();
        return in.sval;
    }

}
