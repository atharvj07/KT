import math
for i in range(int(input())):
	(n, g, b) = map(int, input().split())
	counter = 0
	sounter = 0
	x = math.ceil(n / 2)
	y = math.ceil(x / g)
	if x // g == 0:
		sounter += x % g
		counter = n
	elif x % g == 0:
		sounter += x // g * g + (x // g - 1) * b
		counter += max(sounter, n)
	elif x // g != 0 and x % g != 0:
		sounter += x // g * g + x // g * b
		counter += max(n, sounter + x % g)
	print(counter)
