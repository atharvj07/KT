n = int(input())
b = list(map(int, input().split()))
ans = [-1] * n
ans[0] = 0
ans[-1] = b[0]
l = 0
r = n - 1
for i in range(1, n // 2):
	if b[i] - ans[l] <= ans[r]:
		ans[l + 1] = ans[l]
		ans[r - 1] = b[i] - ans[l]
		l += 1
		r -= 1
	else:
		ans[l + 1] = b[i] - ans[r]
		ans[r - 1] = ans[r]
		l += 1
		r -= 1
print(*ans)
