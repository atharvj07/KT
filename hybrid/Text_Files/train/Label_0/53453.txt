import java.util.Scanner;

public class D {

	/**
	 * @param args
	 */
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int n = s.nextInt();
		int m = s.nextInt();
		int[] a = new int[m];
		int[][] graph = new int[n][n];

		a[0] = s.nextInt();
		for (int i = 1; i < a.length; i++) {
			a[i] = s.nextInt();
			graph[a[i - 1] - 1][a[i] - 1] = 1;
			graph[a[i] - 1][a[i - 1] - 1] = 1;
		}

		s.nextInt();
		graph[a[m - 1] - 1][a[0] - 1] = 1;
		graph[a[0] - 1][a[m - 1] - 1] = 1;
		D.graph = graph;
		D.a = a;
		int[] ans = new int[m];
		ans[0] = a[0];
		int[] myans = dfs(a[0] - 1, ans, 1, false);

		if (myans == null)
			System.out.println("No solution");
		else {
			for (int i = 0; i < myans.length; i++) {
				System.out.print(myans[i] + " ");
			}
			System.out.print(a[0]);
		}
	}

	static int[][] graph;
	static int[] a;

	public static int[] dfs(int curr, int[] ans, int time, boolean ignore) {

		if (time == a.length) {
			if (!ignore)
				return null;
			return ans;
		}

		for (int to = ignore ? 0 : a[time] - 1; to < graph.length; to++) {
			if (graph[curr][to] != 0) {
				graph[to][curr] = graph[curr][to] = 0;
				ans[time] = to + 1;
				int[] answer = dfs(to, ans, time + 1, (a[time] < to + 1) ? true
						: ignore);
				if (answer != null)
					return answer;
				graph[to][curr] = graph[curr][to] = 1;
			}
		}
		return null;
	}
}
