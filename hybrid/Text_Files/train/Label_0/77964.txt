def solve(studentActivity, n, s):
	maxContiguousLength = 0
	firstIndex = -1
	rangeList = [-1]
	curActivitySum = s
	for (lastIndex, activity) in enumerate(studentActivity):
		curActivitySum += activity
		while curActivitySum < 0 and firstIndex < lastIndex:
			firstIndex += 1
			curActivitySum -= studentActivity[firstIndex]
		curActivityLength = lastIndex - firstIndex
		if curActivityLength > maxContiguousLength:
			rangeList = [firstIndex + 2, lastIndex + 1]
			maxContiguousLength = curActivityLength
	print(*rangeList)
for case in range(int(input())):
	(n, s) = map(int, input().split())
	studentActivity = list(map(int, input().split()))
	solve(studentActivity, n, s)
