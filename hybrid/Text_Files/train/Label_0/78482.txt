fre = [0] * 100001
ftp = [0] * 100001
t = list(map(int, input().split()))
k = list(map(int, input().split()))
val = t[1]
m = list(set(k))
if len(m) != len(k):
	print(0)
else:
	for i in range(t[0]):
		ftp[k[i]] = 1
	for i in range(t[0]):
		fre[k[i]] = fre[k[i]] + 1
		if k[i] != k[i] & val:
			fre[k[i] & val] = fre[k[i] & val] + 1
	count = 0
	shout = 0
	val_l = len(fre)
	for i in range(val_l):
		if fre[i] > 1 and ftp[i] == 0:
			count = count + 1
		elif ftp[i] == 1 and fre[i] > 1:
			shout = shout + 1
			count = 0
			break
	if shout == 0 and count > 0:
		print(2)
	elif shout > 0 and count == 0:
		print(shout)
	else:
		print(-1)
