#include <bits/stdc++.h>
using namespace std;

      "title"
      ".out",
      "w", stdout);
}
int read() {
  int g = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || '9' < ch) {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while ('0' <= ch && ch <= '9') {
    g = g * 10 + ch - '0';
    ch = getchar();
  }
  return g * f;
}
const int mod = 1e9 + 7;
const int N = 3505;
int f[N][N][2][2], p, a, t[N], lim[N], n, ans;
char s[N];
signed main() {
  p = read(), a = read();
  scanf("%s", s + 1);
  int len = strlen(s + 1);
  for (int i = (1); i <= (len); i++) t[i] = s[len - i + 1] - '0';
  while (len) {
    long long v = 0;
    for (int i = (len); i >= (1); i--) {
      v = v * 10 + t[i], t[i] = v / p, v %= p;
      if (!t[i] && i == len) len--;
    }
    lim[++n] = v;
  }
  f[n + 1][0][1][0] = 1;
  for (int i = (n); i >= (1); i--) {
    int v1 = 1ll * (p + 1) * p / 2 % mod;
    int v2 = 1ll * (lim[i] + 1) * lim[i] / 2 % mod;
    int v3 = 1ll * (p - 1) * p / 2 % mod;
    int v4 = 1ll * lim[i] * (2ll * p - lim[i] - 1) / 2 % mod;
    int v5 = 1ll * lim[i] * (lim[i] - 1) / 2 % mod;
    int v6 = 1ll * lim[i] * (2ll * p - lim[i] + 1) / 2 % mod;
    for (int j = (0); j <= (n - i); j++) {
      int f1 = f[i + 1][j][0][0], f2 = f[i + 1][j][1][0],
          f3 = f[i + 1][j][0][1], f4 = f[i + 1][j][1][1];
      f[i][j][0][0] =
          (1ll * f1 * v1 + 1ll * f2 * v2 + 1ll * f3 * v3 + 1ll * f4 * v4) % mod;
      f[i][j][1][0] =
          (1ll * f2 * (lim[i] + 1) + 1ll * f4 * (p - lim[i] - 1 + mod)) % mod;
      f[i][j + 1][0][1] = (1ll * f1 * v3 + 1ll * f2 * v5 % mod + 1ll * f3 * v1 +
                           1ll * f4 * v6) %
                          mod;
      f[i][j + 1][1][1] =
          (1ll * f2 * lim[i] % mod + 1ll * f4 * (p - lim[i])) % mod;
    }
  }
  for (int i = (a); i <= (n); i++)
    ans = (1ll * ans + f[1][i][1][0] + f[1][i][0][0]) % mod;
  return cout << ans, signed();
}
