from collections import defaultdict
import heapq
import sys
input = sys.stdin.readline
q = int(input())
for _ in range(q):
	n = int(input())
	cnt = defaultdict(lambda : 0)
	f = defaultdict(lambda : 0)
	for _ in range(n):
		(ai, fi) = map(int, input().split())
		cnt[ai] += 1
		f[ai] += fi
	x = [[cnt[i], i] for i in cnt]
	x.sort(reverse=True)
	s = set()
	t = []
	m = 0
	for (c, i) in x:
		for j in range(c, 0, -1):
			if not j in s:
				s.add(j)
				t.append(j)
				m += j
				break
	x.append([0, 0])
	h = []
	fc = 0
	j = 0
	for i in t:
		while True:
			(ck, k) = x[j]
			if ck >= i:
				heapq.heappush(h, -f[k])
				j += 1
			else:
				break
		fc -= max(heapq.heappop(h), -i)
	print(m, fc)
