#include <bits/stdc++.h>
using namespace std;
int n, m;
int a[10000010];
int prime[10000010];
bool isprime[10000010];
int main() {
  while (scanf("%d", &n) != EOF) {
    int nprime = 0;
    for (long long i = 2; i <= 10000000; i++) {
      if (!isprime[i]) {
        prime[++nprime] = i;
        for (long long j = i * i; j <= 10000000; j += i) isprime[j] = 1;
      }
      if (nprime >= n) break;
    }
    for (int i = 1; i <= nprime; i++) {
      printf("%d ", prime[i]);
    }
  }
}
