for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	if arr[0] == 1:
		res = [n + 1]
		res += [i for i in range(1, n + 1)]
		print(*res)
	elif arr[-1] == 0:
		res = [i for i in range(1, n + 2)]
		print(*res)
	else:
		found = False
		for i in range(n - 1):
			if arr[i] == 0 and arr[i + 1] == 1:
				found = True
				res = [i for i in range(1, i + 2)]
				res += [n + 1]
				res += [i for i in range(i + 2, n + 1)]
				print(*res)
			if found:
				break
		if not found:
			print(-1)
