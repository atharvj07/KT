for tc in range(int(input())):
	(a, b, c, r) = map(int, input().split(' '))
	if a > b:
		(a, b) = (b, a)
	l = c - r
	r = c + r
	temp = b - a
	if a >= l and b <= r:
		temp = 0
	elif b <= l and b <= r or (l <= a and r <= a):
		temp = temp
	elif l <= b and b <= r:
		temp = temp - (b - l)
	elif a <= l and r <= b:
		temp = temp - (r - l)
	elif l <= a and a <= r:
		temp = temp - (r - a)
	print(temp)
