I = lambda : [*map(int, input().split())]

def out(x):
	print(str(x))

def solve():
	n = I()
	b = I()
	ss = {}
	for i in b:
		xx = ss.setdefault(i, 0)
		ss[i] = xx + 1
	sm = sum(b)
	for i in b:
		ss[i] -= 1
		smx = sm - i
		if smx % 2 == 0:
			smx = smx // 2
			if ss.get(smx, -1) > 0:
				ss[smx] -= 1
				out(''.join([f'{k} ' * v for (k, v) in ss.items()]))
				return
		ss[i] += 1
	out(-1)
(t,) = I()
for i in range(t):
	solve()
