MINUS_INF = -10 ** 9

def solve(w, h, stamps):
	res = MINUS_INF
	for (i, s1) in enumerate(stamps):
		for s2 in stamps[i + 1:]:
			if fit(w, h, s1, s2):
				res = max(res, s1[0] * s1[1] + s2[0] * s2[1])
	res = max(res, 0)
	return res

def fit(w, h, s1, s2):
	return subfit(w, h, s1, s2) or subfit(w, h, rev(s1), s2) or subfit(w, h, s1, rev(s2)) or subfit(h, w, s1, s2)

def subfit(w, h, s1, s2):
	return s1[0] + s2[0] <= w and s1[1] <= h and (s2[1] <= h) or (s1[1] + s2[1] <= h and s1[0] <= w and (s2[0] <= w))

def rev(s):
	return (s[1], s[0])
(n, w, h) = map(int, input().split())
stamps = [tuple(map(int, input().split())) for i in range(n)]
print(solve(w, h, stamps))
