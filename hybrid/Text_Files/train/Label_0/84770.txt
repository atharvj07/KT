import bisect
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
dp = [0] * (n + 1)
for i in range(len(a)):
	if a[i] == 0:
		dp[i + 1] += 1
	dp[i + 1] += dp[i]
maxi = -1
ind = -1
for i in range(1, len(dp)):
	ans = bisect.bisect_left(dp, dp[i] - k)
	maxi = max(maxi, i - ans)
	if maxi == i - ans:
		ind = i
end = ind - 1
strt = end - maxi + 1
for i in range(strt, end + 1):
	a[i] = 1
print(maxi)
print(*a)
