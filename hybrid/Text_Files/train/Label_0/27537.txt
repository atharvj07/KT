#include <bits/stdc++.h>
int n, k;
std::string t;
int main() {
  std::cin >> n >> k;
  std::cin >> t;
  for (int i = 1; i < n; ++i) {
    bool match = true;
    for (int j = 0; j < n - i; ++j) {
      if (t[j] != t[i + j]) {
        match = false;
        break;
      }
    }
    if (match) {
      std::cout << t;
      for (int j = 1; j < k; ++j) {
        std::cout << t.substr(n - i);
      }
      return 0;
    }
  }
  for (int j = 0; j < k; ++j) {
    std::cout << t;
  }
  return 0;
}
