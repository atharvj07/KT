def checkpal(h, m):
	if h < 10:
		if m == h * 10:
			return True
		else:
			return False
	else:
		temp = h % 10
		h //= 10
		if m == temp * 10 + h:
			return True
		else:
			return False
tm = input()
hour = int(tm[0]) * 10 + int(tm[1])
minute = int(tm[3]) * 10 + int(tm[4])
ans = 0
i = hour
j = minute
while not checkpal(i, j):
	if j == 59:
		j = 0
		if i == 23:
			i = 0
		else:
			i += 1
	else:
		j += 1
	ans += 1
print(ans)
