#include <bits/stdc++.h>
using namespace std;
int factorial(int n) {
  int f = 1;
  if (n == 0) return 1;
  for (int i = 1; i <= n; i++) f = f * i;
  return f;
}
int main() {
  int n, k, c = 0;
  cin >> n >> k;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] + k <= 5) c++;
  }
  if (c >= 3)
    cout << (int)c / 3;
  else
    cout << 0;
  return 0;
}
