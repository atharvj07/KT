import java.io.InputStreamReader;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.InputStream;

public class CountrySide {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReaderCountrySide in = new InputReaderCountrySide(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskCountrySide solver = new TaskCountrySide();
        solver.solve(1, in, out);
        out.close();
    }
}

class TaskCountrySide {
    public void solve(int taskNumber, InputReaderCountrySide in, PrintWriter out) {
    int n = in.nextInt();
        int [] arr = new int[n];
        for(int i = 0; i < n; ++i){
            arr[i] = in.nextInt();
        }
        if(n == 1){
            if(arr[0] == 0){
                System.out.println("UP");
                return;
            }
            else if(arr[0] == 15){
                System.out.println("DOWN");
                return;
            }
            else {
                System.out.println(-1);
                return;
            }
        }
        else{
            if(arr[n - 1] == 0){
                System.out.println("UP");
                return;
            }
            if(arr[n - 1] == 15){
                System.out.println("DOWN");
                return;
            }
            if(arr[n - 2] > arr[n - 1]){
                System.out.println("DOWN");
                return;
            }
            if(arr[n - 2] < arr[n - 1]){
                System.out.println("UP");
                return;
            }
        }
    }
}

class InputReaderCountrySide {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReaderCountrySide(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream), 32768);
        tokenizer = null;
    }

    public String next() {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }
        return tokenizer.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(next());
    }

}
