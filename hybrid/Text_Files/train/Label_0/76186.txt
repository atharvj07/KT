(po, men) = map(int, input().split())
a = {int(i) for i in input().split()}
f = {*a}
r = set()
k = 1
s = 0
while len(r) < men:
	b = set()
	for i in a:
		for p in (-1, 1):
			if i + p * k not in f:
				if len(r) < men:
					s += abs(p * k)
					r.add(i + p * k)
					f.add(i + p * k)
					b.add(i)
					if len(r) == men:
						break
		if len(r) == men:
			break
	k += 1
	a = {*b}
print(s)
print(*r)
