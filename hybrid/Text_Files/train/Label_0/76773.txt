from functools import cmp_to_key

def chave1(v1, v2):
	if v1[1] < v2[0]:
		return 1
	else:
		return -1

def chave2(v1, v2):
	if v1[1] > v2[0]:
		return 1
	else:
		return -1
lista_geral = []
indice = 1
for i in range(int(input())):
	(a, b) = input().split(' ')
	a = int(a)
	b = int(b)
	lista_geral.append((a, b, indice))
	indice += 1
lista1 = [e for e in lista_geral if e[0] < e[1]]
lista2 = [e for e in lista_geral if e[0] > e[1]]
lista1.sort(key=cmp_to_key(chave1))
lista2.sort(key=cmp_to_key(chave2))
if len(lista1) > len(lista2):
	print(len(lista1))
	[print(e[2], '', end='') for e in lista1]
else:
	print(len(lista2))
	[print(e[2], '', end='') for e in lista2]
