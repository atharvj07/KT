for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	ans = []
	c = 0
	for i in range(n - 1):
		ind = arr.index(min(arr))
		if ind > 0:
			temp = str(i + 1) + ' ' + str(ind + i + 1) + ' ' + str(ind)
			ans.append(temp)
			c += 1
		arr = arr[:ind] + arr[ind + 1:]
	print(c)
	print('\n'.join(ans))
