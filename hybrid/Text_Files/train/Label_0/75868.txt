for _ in range(int(input())):
	(a, b, p) = map(int, input().split())
	price = {'A': a, 'B': b}
	stat = input()
	dp = [0] * len(stat)
	dp[-2] = price[stat[-2]]
	for i in range(len(stat) - 3, -1, -1):
		dp[i] = dp[i + 1]
		if stat[i] != stat[i + 1]:
			dp[i] += price[stat[i]]
	i = 0
	while dp[i] > p:
		i += 1
	print(i + 1)
