str1 = input()
str2 = input()
(n, m) = (len(str1), len(str2))
(pos, i) = ([], 0)
for p in range(n):
	if i == m:
		break
	if str1[p] == str2[i]:
		pos.append(p)
		i += 1
pos = [-1] + pos + [n]
ans = 0
for i in range(1, m + 2):
	ans = max(ans, pos[i] - pos[i - 1] - 1)
i = 0
for p in range(n - 1, -1, -1):
	if i == m:
		break
	if str1[p] == str2[m - i - 1]:
		ans = max(ans, p - pos[m - i - 1] - 1)
		i += 1
print(ans)
