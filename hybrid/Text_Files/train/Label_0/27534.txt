#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x1 = 0, x2 = 0, c1 = 0, c2 = 0, i;
  string s;
  cin >> n >> s;
  for (i = 0; i < s.size() / 2; i++) {
    if (s[i] == '?')
      c1++;
    else
      x1 += s[i] - '0';
  }
  for (i = s.size() / 2; i < s.size(); i++) {
    if (s[i] == '?')
      c2++;
    else
      x2 += s[i] - '0';
  }
  x1 += c1 / 2 * 9;
  x2 += c2 / 2 * 9;
  if (x1 == x2)
    cout << "Bicarp";
  else
    cout << "Monocarp";
}
