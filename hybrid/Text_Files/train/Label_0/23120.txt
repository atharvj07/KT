#include <stdio.h>
#include <cmath>
#include <algorithm>
#include <stack>
#include <queue>
#include <vector>
#include <iostream>
#include <set>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define MOD 1000000007
#define PRIME1 99999883
#define PRIME2 99999893
#define EPS 0.00000001
using namespace std;

enum Type{
	None,
	GreaterEqual,
	SmallerEqual,
};

struct Data{
	Type type;
	int number;
};

struct Info{
	Data data[101];
	bool used;
};

int main(){

	int num_of_SC,num_of_skill,K;
	scanf("%d %d",&num_of_SC,&num_of_skill);
	Info info[num_of_SC];
	for(int i = 0; i < num_of_SC; i++){
		info[i].used = false;
		for(int a = 1; a <= num_of_skill;a++)info[i].data[a].type = None;
	}

	int skill_code,skill_level;

	char eq[4];
	bool FLG = true;

	for(int i = 0; i < num_of_SC; i++){
		scanf("%d",&K);
		for(int a = 0; a < K; a++){
			scanf("%d %s %d",&skill_code,eq,&skill_level);
			if(info[i].data[skill_code].type == None){
				info[i].data[skill_code].number = skill_level;
				if(eq[0] == '>'){
					info[i].data[skill_code].type = GreaterEqual;
				}else{
					info[i].data[skill_code].type = SmallerEqual;
				}
			}else{
				if(info[i].data[skill_code].type == GreaterEqual){
					if(eq[0] == '>'){
						//Do nothing
					}else{
						if(info[i].data[skill_code].number > skill_level){
							FLG = false;
						}
					}
				}else{
					if(eq[0] == '>'){
						if(info[i].data[skill_code].number < skill_level){
							FLG = false;
						}
					}
				}
			}
		}
	}

	if(!FLG){
		printf("No\n");
		return 0;
	}

	int count = 0;
	stack<int> S;

	while(true){

		for(int i = 0; i < num_of_SC; i++){
			if(info[i].used == true)continue;

			FLG = true;
			for(int a = 0; a < num_of_SC; a++){
				if(i == a || info[a].used == true)continue;

				for(int b = 1; b <= num_of_skill;b++){

					switch(info[a].data[b].type){
					case None:
						//Do nothing
						break;
					case GreaterEqual:
						//Do nothing
						break;
					case SmallerEqual:
						if(info[i].data[b].type == GreaterEqual && info[i].data[b].number > info[a].data[b].number){
							FLG = false;
						}
						break;
					}
				}
				if(!FLG)break;
			}
			if(FLG){
				S.push(i);
				break;
			}
		}

		if(S.empty())break;

		while(!S.empty()){
			info[S.top()].used = true;
			count++;
			S.pop();
		}
	}

	if(count == num_of_SC){
		printf("Yes\n");
	}else{
		printf("No\n");
	}

    return 0;
}
