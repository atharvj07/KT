for i in range(int(input())):
	(x, n, m) = list(map(int, input().split()))
	while n > 0 or m > 0:
		if x <= 0:
			break
		elif x <= 20:
			if m > 0:
				x -= 10
				m -= 1
			elif n > 0:
				x = x // 2 + 10
				n -= 1
		elif x > 20:
			if n > 0:
				x = x // 2 + 10
				n -= 1
			elif m > 0:
				x -= 10
				m -= 1
	if x <= 0:
		print('YES')
	else:
		print('NO')
