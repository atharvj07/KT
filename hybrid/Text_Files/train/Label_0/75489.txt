s = input()
d = {}
a = set()
for i in s:
	if i not in d:
		d[i] = 1
		a.add(i)
	else:
		d[i] += 1
if len(d) > 4 or len(d) <= 1:
	print('No')
elif len(d) == '4':
	print('Yes')
else:
	y = []
	for j in a:
		y.append(d[j])
	y.sort()
	if y[0] >= 2:
		print('Yes')
	elif len(y) == 2 and y[0] == 1:
		print('No')
	elif len(y) == 3 and y[2] == 1:
		print('No')
	else:
		print('Yes')
