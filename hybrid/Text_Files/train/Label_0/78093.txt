n = int(input())
a = []
for i in input().split():
	a.append(int(i))
a.sort()
if n % 2 == 0:
	for i in range(0, n, 2):
		print(a[i], end=' ')
	for i in range(n - 1, 0, -2):
		print(a[i], end=' ')
else:
	for i in range(1, n, 2):
		print(a[i], end=' ')
	for i in range(n - 1, -1, -2):
		print(a[i], end=' ')
