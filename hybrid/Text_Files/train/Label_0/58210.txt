import java.util.*;

public class Main {
	
	static int n;
	public static List<Integer>[] con;
	static int[] lab;

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		
		int n,m,i,j;
		
		int k;
		
		n=sc.nextInt();k=sc.nextInt();
		int[] h=new int[n+1];
		long hm=(long) 1e18;
		
		for(i=1;i<=n;i++) 
		{
			h[i]=sc.nextInt();hm=Math.min(h[i], hm);
		}
			
		
		
		for(i=1;i<=n;i++) h[i]-=hm;
		
		int [] w=new int[(int)2e5+2];
		for(i=1;i<=n;i++) w[h[i]]++;
		w[0]=k;
		
		for(i=(int)2e5;i>0;i--) w[i]+=w[i+1];
		
		int cnt=0,ans=0;
		for(i=(int)2e5;i>0;i--)
		{
			cnt+=w[i];
			
			if(cnt+w[i-1]>k) {ans++;cnt=0;}
		}
		
		System.out.println(ans);
		
	} 

}
