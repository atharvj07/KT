#include <bits/stdc++.h>
using namespace std;
int beg[3 * 2000 + 10] = {0}, en[3 * 2000 + 10] = {0};
char s_[3 * 2000 + 10], s[3 * 2000 + 10];
void strans(char *s, char *t) {
  int i, p = 0;
  t[p++] = '$';
  for (i = 0; s[i] != '\0'; i++) {
    t[p++] = s[i];
    t[p++] = '$';
  }
  t[p] = '\0';
}
void disp(int *A, int n) {
  int i;
  for (i = 0; i < n; i++) printf("%d ", A[i]);
  printf("\n");
}
int main() {
  scanf("%s", s_);
  strans(s_, s);
  int lx = strlen(s), i;
  for (i = 0; i < lx; i++) {
    int x = 0;
    while (i - x >= 0 && i + x < lx && s[i - x] == s[i + x]) {
      beg[i - x]++;
      en[i + x]++;
      x++;
    }
  }
  long long ans = 0;
  for (i = lx - 2; i >= 0; i -= 2) {
    beg[i] = beg[i] + beg[i + 2];
  }
  for (i = 2; i < lx; i += 2) {
    ans += ((long long)en[i - 1]) * beg[i + 1];
  }
  printf("%I64d\n", ans);
  return 0;
}
