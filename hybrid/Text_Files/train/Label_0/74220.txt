from copy import copy
n = int(input())
sh = input()
mo = input()
shf = [0] * 10
mof = [0] * 10
for c in sh:
	shf[int(c)] += 1
for c in mo:
	mof[int(c)] += 1
oshf = copy(shf)
omof = copy(mof)
sav = 0
hit = 0
for i in range(9, -1, -1):
	m = min(shf[i], mof[i])
	shf[i] -= m
	mof[i] -= m
	m = min(sav, shf[i])
	sav -= m
	shf[i] -= m
	hit += shf[i]
	sav += mof[i]
	for j in range(10):
		m = min(mof[j], shf[i])
		mof[j] -= m
		shf[i] -= m
minhit = hit
shf = oshf
mof = omof
for i in range(10):
	m = min(mof[i], shf[9])
	mof[i] -= m
	shf[9] -= m
maxhit = 0
sav = mof[9]
for i in range(8, -1, -1):
	m = min(shf[i], sav)
	maxhit += m
	shf[i] -= m
	sav -= m
	sav += mof[i]
print(minhit)
print(maxhit)
