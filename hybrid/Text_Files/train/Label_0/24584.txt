#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pll = pair<ll, ll>;
int n, t0;
vector<int> f, s;
int ones(int l, int r) {
  int result = 0;
  for (int i = l; i <= r; i++) result += s[i];
  return result;
}
void flip(int l, int r) {
  for (int i = l; i <= r; i++) {
    f[i] = 1 - f[i];
    if (s[i] != -1) s[i] = 1 - s[i];
  }
}
vector<int> hidden;
int query(int l, int r) {
  cout << "? " << l << " " << r << endl;
  int t;
  cin >> t;
  return t;
}
int main() {
  ios_base::sync_with_stdio(false);
  cout << setprecision(12);
  cin.tie(nullptr);
  cin >> n >> t0;
  if (n == 1) {
    cout << "! " << t0 << endl;
    return 0;
  }
  f.resize(n + 1);
  s.assign(n + 1, -1);
  while (s[1] == -1) {
    int t = query(2, n);
    if (t == n - t0) {
      flip(1, n);
    } else {
      s[1] = (t - (n - t0) + 1) / 2;
      if (s[1] != 0 && s[1] != 1) {
        cout << -1 << endl;
        return 0;
      }
      flip(2, n);
    }
    t0 = t;
  }
  for (int i = n; i > 2; i--) {
    while (s[i] == -1) {
      int r = i - 1;
      int l, wrong;
      if ((n - i) & 1)
        l = 2, wrong = n - t0 + 2 * s[1] - 1;
      else
        l = 1, wrong = n - t0;
      int t = query(l, r);
      if (t == wrong) {
        flip(l, n);
      } else {
        s[i] = (t + t0 - r - 2 * ones(i + 1, n)) / 2;
        if (s[i] != 0 && s[i] != 1) {
          cout << -1 << endl;
          return 0;
        }
        flip(1, r);
      }
      t0 = t;
    }
  }
  s[2] = t0 - s[1] - ones(3, n);
  cout << "! ";
  for (int i = 1; i <= n; i++) cout << (f[i] ? 1 - s[i] : s[i]);
  cout << endl;
}
