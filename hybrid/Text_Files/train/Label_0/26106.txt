#include <bits/stdc++.h>
using namespace std;
const int nsz = int(4e5);
const int inf = 0x3f3f3f3f;
int n, m, q, dm;
set<int> pos[nsz + 5];
struct segment {
  bool f;
  int p0, p1;
  segment() { f = 1, p0 = -inf, p1 = +inf; }
  segment(int p0, int p1, bool f = 1) {
    this->p0 = p0, this->p1 = p1, this->f = f;
  }
  friend inline segment operator&(const segment &a, const segment &b) {
    return segment(max(a.p0, b.p0), min(a.p1, b.p1), a.f & b.f & (a.p1 > b.p0));
  }
};
struct segment_tree {
  segment tr[4 * nsz + 5];
  void upd(int id, const segment &w, int u = 1, int l = 1, int r = dm) {
    if (l == r) {
      tr[u] = w;
      return;
    }
    if (id <= ((l + r) >> 1)) upd(id, w, (u << 1), l, ((l + r) >> 1));
    if (id > ((l + r) >> 1)) upd(id, w, (u << 1 | 1), ((l + r) >> 1) + 1, r);
    tr[u] = tr[(u << 1)] & tr[(u << 1 | 1)];
  }
  segment qry(int el, int er, int u = 1, int l = 1, int r = dm) {
    if (l >= el && r <= er) return tr[u];
    segment res;
    if (el <= ((l + r) >> 1)) res = qry(el, er, (u << 1), l, ((l + r) >> 1));
    if (er > ((l + r) >> 1))
      res = res & qry(el, er, (u << 1 | 1), ((l + r) >> 1) + 1, r);
    return res;
  }
};
segment_tree tr;
int main() {
  scanf("%d%d%d", &n, &m, &q);
  dm = 2 * n;
  for (int i = 1; i <= q; ++i) {
    int r, c;
    scanf("%d%d", &r, &c);
    segment cur;
    if (r & 1) {
      if (pos[r].find(c) == pos[r].end()) {
        pos[r].insert(c);
        cur = segment(-inf, *pos[r].begin());
      } else {
        pos[r].erase(c);
        cur = segment(-inf, pos[r].size() ? *pos[r].begin() : inf);
      }
    } else {
      if (pos[r].find(c) == pos[r].end()) {
        pos[r].insert(c);
        cur = segment(*pos[r].rbegin(), inf);
      } else {
        pos[r].erase(c);
        cur = segment(pos[r].size() ? *pos[r].rbegin() : -inf, inf);
      }
    }
    tr.upd(r, cur);
    printf("%s\n", tr.qry(1, dm).f ? "YES" : "NO");
  }
}
