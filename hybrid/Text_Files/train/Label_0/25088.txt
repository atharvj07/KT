vector<long long int> vv[200010];
long long int ar[200010], gg = 0;
long long int dfs(long long int a) {
  long long int i, j, k, b, c, d;
  ar[a] = 1;
  if (vv[a].size() != 2) gg = 1;
  for (i = 0; i < vv[a].size(); i++) {
    if (ar[vv[a][i]] == 0) {
      ar[vv[a][i]] == 1;
      c = vv[a][i];
      dfs(vv[a][i]);
    }
  }
  return c;
}
