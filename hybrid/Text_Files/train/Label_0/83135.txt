from functools import reduce
from operator import mul
nc = int(input())
gen_tank = (i for i in input().split())
zero_count = 0
butiful_num = ''
zero_in = False
for t in gen_tank:
	if t == '0':
		print('0')
		zero_in = True
		break
	else:
		if t == '1':
			continue
		if int(t) == pow(10, len(t) - 1):
			zero_count += len(t) - 1
		else:
			butiful_num = t
if zero_in == False:
	if butiful_num == '':
		print('1' + '0' * zero_count)
	else:
		print(butiful_num + '0' * zero_count)
