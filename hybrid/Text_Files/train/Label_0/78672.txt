from collections import Counter
n = int(input())
A = [int(i) for i in input().split()]
C = Counter(A)
B = [[] for i in range(5)]
Q = [[0] * 5 for i in range(21)]
Q[1][1] = 1
for i in range(3, 21, 2):
	Q[i][1] = 1
	Q[i][2] = Q[i - 2][2] + 2
	Q[i][4] = Q[i - 2][4] + (i - 2)
for i in range(2, 21, 2):
	Q[i][4] = (i // 2) ** 2
P = Q[n]
if n % 2 == 0:
	for (k, v) in C.items():
		if v % 4 != 0:
			print('NO')
			exit()
		B[4] += [k] * (v // 4)
	print('YES')
	m = n // 2
	D = dict()
	for i in range(n):
		a = i if i < m else n - i - 1
		ans = []
		for j in range(n):
			b = j if j < m else n - j - 1
			if not (a, b) in D:
				D[a, b] = B[4].pop()
			ans.append(D[a, b])
		print(*ans)
else:
	for (k, v) in C.items():
		if v % 2 == 1:
			C[k] -= 1
			P[1] -= 1
			B[1].append(k)
	for (k, v) in C.items():
		if P[4] == 0:
			break
		t = min(P[4], v // 4)
		C[k] -= 4 * t
		P[4] -= t
		B[4] += [k] * t
	for (k, v) in C.items():
		t = v // 2
		B[2] += [k] * t
		P[2] -= t
		C[k] -= 2 * t
	if P[1] != 0 or P[4] != 0 or P[2] != 0:
		print('NO')
		exit()
	print('YES')
	m = n // 2
	D = dict()
	for i in range(n):
		ans = []
		for j in range(n):
			if i == m and j == m:
				ans.append(B[1][0])
			elif i == m:
				k = j if j < m else n - j - 1
				k += Q[n][2] // 2
				ans.append(B[2][k])
			elif j == m:
				k = i if i < m else n - i - 1
				ans.append(B[2][k + m])
			else:
				(a, b) = (abs(i - m), abs(j - m))
				if not (a, b) in D:
					D[a, b] = B[4].pop()
				ans.append(D[a, b])
		print(*ans)
