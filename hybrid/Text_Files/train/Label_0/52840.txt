import java.util.*;

public class NEED {
    public static void main(String[] args) {
        String str;
        String com = "CODEFORCES";
        Scanner sca = new Scanner(System.in);
        while (sca.hasNext()) {
            str = sca.next();
            boolean flag = false;
            if (str.length() <= 10) {
                System.out.println("NO\n");
                continue;
            }

            if (str.substring(str.length() - 10, str.length()).equals(com)) {
                flag = true;
                // System.out.println("ss");
                
            }

            if (str.substring(0, 10).equals(com)) {
                flag = true;
                // System.out.println("ss");
                
            }

            // System.out.println(str.substring(0, 0));
            for (int i = 0; i < str.length(); i++) {
                for (int j = i + 2; j < str.length(); j++) {
                    if ((str.substring(0, i + 1) + str.substring(j,
                            str.length())).equals(com)) {
                        flag = true;
                        break;
                    }
                }
            }
            if (flag) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }

    }
}
