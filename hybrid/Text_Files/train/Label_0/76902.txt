t = int(input())
for _ in range(t):
	(r, g, b, w) = [int(x) for x in input().split()]
	arr = [r, g, b, w]
	n = r + g + b + w
	n_odd = sum([1 for x in arr if x & 1])
	n_even = sum([1 for x in arr if x % 2 == 0])
	if n % 2 == 0 and (r % 2 == 0 and g % 2 == 0 and (b % 2 == 0) and (w % 2 == 0) or (r & 1 and g & 1 and b & 1 and w & 1)):
		print('YES')
	elif n & 1 and (n_odd == 1 or (n_odd == 3 and r >= 1 and (g >= 1) and (b >= 1))):
		print('YES')
	else:
		print('NO')
