tst = int(input())
for _ in range(tst):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	if len(arr) == 1 and arr[0] % 2 == 1:
		print('YES')
		print('1')
	elif len(arr) == 1 and arr[0] % 2 != 1:
		print('NO')
	else:
		odd = 0
		even = 0
		for i in arr:
			if i % 2 == 0:
				even += 1
			else:
				odd += 1
		if k % 2 == odd % 2 and k <= odd:
			print('YES')
			count = k
			sumval = 0
			ind = []
			i = 0
			while count > 1 and i < n:
				sumval += arr[i]
				if sumval % 2 != 0:
					ind.append(i + 1)
					count -= 1
					sumval = 0
				i += 1
			ind.append(n)
			print(' '.join(map(str, ind)))
		else:
			print('NO')
