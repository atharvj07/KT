#include <iostream>
#include <algorithm>
#include <vector>
#include <cstring>
using namespace std;
 
#define REP(i,n) for(int i = 0; i < (int)(n); ++i)
#define DEBUG(x) cerr << #x << " = " << x << endl
 
const int MAX_N = 17;
const int inf = (int)1e8;
int V;
 
int d[MAX_N][MAX_N];
int dp[MAX_N][1 << MAX_N];
int eat[MAX_N];
 
int tsp(int v, int b) {
    if(dp[v][b] != -1) return dp[v][b];
    if(v == 0 && b == 0) return 0;
    int res = inf;
    for(int i = 0; i < V; i++) {
        if(b & (1 << i)) {
            res = min(res, tsp(i, b ^ (1 << i)) + d[i][v] + eat[i]);
        }
    }
    return dp[v][b] = res;
}
 
int main() {
    while(true) {
        int N, M, L, S, T;
        cin >> N >> M >> L >> S >> T;
        if(N == 0 && M == 0 && L == 0 && S == 0 && T == 0) break;
        vector<vector<int> > D(N, vector<int>(N, inf));
        for(int i = 0; i < N; i++) {
            D[i][i] = 0;
        }
        for(int i = 0; i < M; i++) {
            int a, b, c; cin >> a >> b >> c;
            a--, b--;
            D[a][b] = c;
            D[b][a] = c;
        }
 
        REP(k,N) REP(i,N) REP(j,N) {
            D[i][j] = min(D[i][j], D[i][k] + D[k][j]);
        }
 
 
        V = L + 1;
        vector<int> LS(V);
        S--;
        LS[0] = S;
        eat[0] = 0;
        for(int i = 1; i < V; i++) {
            cin >> LS[i];
            cin >> eat[i];
            LS[i]--;
        }
        REP(i,V) {
            REP(j,V) {
                d[i][j] = D[LS[i]][LS[j]];
            }
        }
 
        // REP(i,V) REP(j,V) cerr << i << "," << j << " = " << d[i][j] << endl;
 
        memset(dp, -1, sizeof(dp));
        int ans = 0;
        for(int b = 0; b < (1 << V); b++) {
            if(b & 1) { // 0 is houmonzumi
                int cost = tsp(0, b);
                if(cost <= T) ans = max(ans, __builtin_popcount(b) - 1);
            }
        }
        cout << ans << endl;
    }
}
