#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7, MX = 400005;
const long long INF = 2e18;
template <class T>
inline void chmax(T &x, T y) {
  x = max((x), (y));
}
template <class T>
inline void chmin(T &x, T y) {
  x = min((x), (y));
}
template <typename T>
T powm(T a, T b, long long m) {
  T cnt = 1;
  while (b > 0) {
    if (b % 2 == 1) cnt = (cnt * a) % m;
    b /= 2;
    a = (a * a) % m;
  }
  return cnt % m;
}
template <typename T>
T gcd(T a, T b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
int a[MX], b[MX], gc[MX];
int n, m, ans;
map<int, bool> mp;
int f(int n) {
  int ret = 0;
  for (int i = 2; i * i <= n; i++)
    while (n % i == 0) {
      if (mp[i])
        ret--;
      else
        ret++;
      n /= i;
    }
  if (n > 1 && mp[n])
    ret--;
  else if (n > 1 && !mp[n])
    ret++;
  return ret;
}
int32_t main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    gc[i] = gcd(gc[i - 1], a[i]);
  }
  for (int i = 1; i <= m; i++) {
    cin >> b[i];
    mp[b[i]] = true;
  }
  for (int i = n; i >= 1; i--)
    if (f(gc[i]) < 0)
      for (int j = 1; j <= i; j++) a[j] /= gc[i], gc[j] /= gc[i];
  for (int i = 1; i <= n; i++) ans += f(a[i]);
  cout << ans;
}
