import sys
from array import array
import typing as Tp

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def main():
	t = int(input())
	for _ in range(t):
		(x, y) = map(int, input().split())
		if x == 1:
			print('Yes' if y == 1 else 'No')
		elif x <= 3:
			print('Yes' if 1 <= y <= 3 else 'No')
		else:
			print('Yes')
main()
