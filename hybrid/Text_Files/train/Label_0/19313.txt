#include <bits/stdc++.h>
using namespace std;
const long long INF = (long long)1e18;
const long long MOD = (long long)1e9 + 7;
int n, k, i, j, l, v[200], smth, go;
string s, c, t;
char x;
int main() {
  cin >> n >> k >> s;
  for (i = 0; i < n - (int)s.length() + 1; ++i) cin >> x, v[i] = x - '0';
  c.resize(n);
  for (l = 0; l < 25; ++l) {
    go = 1;
    for (i = 0; i < n; ++i) c[i] = 'a' + l;
    for (i = 0; i < n; ++i)
      if (v[i])
        for (j = i; j < i + (int)s.length(); j++) c[j] = s[j - i];
    for (i = 0; i < n - (int)s.length() + 1; ++i) {
      t = c.substr(i, (int)s.length());
      if ((t == s && !v[i]) || (t != s && v[i])) {
        go = 0;
        break;
      }
    }
    if (go) cout << c, exit(0);
  }
  cout << "No solution" << endl;
}
