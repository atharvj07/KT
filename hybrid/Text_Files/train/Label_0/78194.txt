t = int(input())
for _ in range(t):
	(n, c) = input().split()
	n = int(n)
	s = input()
	idd = -1
	ctt = 0
	for i in range(n):
		if s[i] == c:
			idd = i + 1
			ctt += 1
	if idd == -1:
		print(2)
		print(n, n - 1)
	elif ctt == n:
		print(0)
	elif 2 * idd > n:
		print(1)
		print(idd)
	else:
		print(2)
		print(n, n - 1)
