#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int n, a, b;
  cin >> n >> a >> b;
  if (b < a) swap(a, b);
  a--, b--;
  int ans = 0;
  for (int i = 2; i <= 256; i *= 2) {
    if (a / i == b / i) {
      ans = log2(i);
      goto lol;
    }
  }
lol:;
  if (ans == log2(n)) {
    cout << "Final!"
         << "\n";
    return 0;
  }
  cout << ans << "\n";
  return 0;
}
