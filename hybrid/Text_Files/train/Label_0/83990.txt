flag2 = 0

	if j != 1 and j != 4:
		print('NO')
		flag1 = 1
		flag2 = 1
		break
if flag1 != 1:
	i = 0
	while i in range(0, len(list1)):
		if list1[i] == 1:
			if list1[i] == 1 and i < len(list1) - 1:
				if list1[i + 1] == 4:
					if i + 2 < len(list1) and list1[i + 2] == 4:
						i += 2
					else:
						i += 1
			i += 1
		else:
			print('NO')
			flag2 = 1
			break
