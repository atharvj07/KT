#include <bits/stdc++.h>
using namespace std;
long long a, b;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> a >> b;
  while (true) {
    if (a == 0 || b == 0)
      break;
    else if (a >= b * 2)
      a %= 2 * b;
    else if (b >= a * 2)
      b %= 2 * a;
    else
      break;
  }
  cout << a << ' ' << b << '\n';
  return 0;
}
