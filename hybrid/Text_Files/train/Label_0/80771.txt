import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	A = [0] + list(map(int, input().split()))
	B = [0] + list(map(int, input().split()))
	ans = []
	(lo, hi) = (n, n - A[n])
	if hi <= 0:
		print(1)
		print(0)
		return
	while lo > 0:
		nextStep = hi
		idx = hi
		for i in range(hi, lo + 1):
			j = i + B[i]
			if nextStep > j - A[j]:
				nextStep = j - A[j]
				idx = i
		if nextStep == hi > 0:
			print(-1)
			return
		ans.append(idx)
		if nextStep <= 0:
			ans.append(0)
			print(len(ans))
			print(*ans)
			return
		(lo, hi) = (hi - 1, nextStep)
	return
solve()
