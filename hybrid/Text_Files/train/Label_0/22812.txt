#include <iostream>
#include <queue>
#include <vector>
#include <string>
#include <set>
using namespace std;

vector<vector<int> > h_edges, c_edges;

void dfs1(set<int> &ss,int start) {
    queue<int> q; q.push(start); ss.insert(start);
    while(!q.empty()) {
        int ns = q.front(); q.pop();
        for(int i=0; i<c_edges[ns].size(); ++i)
            if(ss.find(c_edges[ns][i]) == ss.end()) {
                q.push(c_edges[ns][i]);
                ss.insert(c_edges[ns][i]);
            }
    }
}

void bfs(queue<int> &q,vector<int> &dist) {
    while(!q.empty()) {
        int ns = q.front(); q.pop();
        for(int i=0; i<h_edges[ns].size(); ++i) {
            if(dist[h_edges[ns][i]] > dist[ns]+1) {
                dist[h_edges[ns][i]] = dist[ns]+1;
                q.push(h_edges[ns][i]);
            }
        }
    }
}

int main() {
    int n,m,tc,a,b,h_start,c_start;
    string s;
    cin>>tc;

    while(tc--) {
        cin>>n>>m;
        cin>>h_start>>c_start;
        h_edges.clear(); c_edges.clear();
        h_edges.resize(n+1); c_edges.resize(n+1);

        for(int i=0; i<m; ++i) {
            cin>>a>>b>>s;
            if(s == "N") {
                h_edges[a].push_back(b);
                h_edges[b].push_back(a);
            }else{
                c_edges[a].push_back(b);
                c_edges[b].push_back(a);
            }
        }

        set<int> canReach,fromGoal;
        // dfs1
        dfs1(canReach, c_start);
        dfs1(fromGoal, 0);

        if(canReach.find(0) != canReach.end()) {
            cout<<0<<endl;
            continue;
        }

        queue<int> q;
        vector<int> d1(n+1,1<<29), d2(n+1, 1<<29), d3(n+1, 1<<29);
        d1[h_start] = 0; q.push(h_start);
        bfs(q,d1);

//        cout<<canReach.size()<<endl;
        for(set<int>::iterator it = canReach.begin();
            it != canReach.end(); ++it) {
            q.push(*it); d2[*it] = 0;
        }
        bfs(q,d2);

        for(set<int>::iterator it = fromGoal.begin();
            it != fromGoal.end(); ++it) {
            q.push(*it); d3[*it] = 0;
        }
        bfs(q,d3);

        int ans = 1<<29;
        for(int i=0; i<=n; ++i) {
//            cout<<i<<" -> "<<d1[i]<<","<<d2[i]<<","<<d3[i]<<endl;
            ans = min(ans, d1[i]+d2[i]+d3[i]);
        }
        cout<<ans<<endl;
    }
}
