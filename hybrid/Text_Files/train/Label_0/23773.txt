#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
long long arr[maxn], ans[maxn];
long long n, k;
int check(long long mid) {
  long long sum = 0;
  for (long long i = 0; i < n; i++) {
    sum += min(arr[i], mid);
  }
  if (sum >= k)
    return 1;
  else
    return 0;
}
int main() {
  while (cin >> n >> k) {
    long long sum = 0;
    for (long long i = 0; i < n; i++) {
      cin >> arr[i];
      sum += arr[i];
    }
    if (k > sum) {
      cout << -1 << endl;
      continue;
    } else if (k == 0) {
      for (long long i = 0; i < n; i++) {
        if (i != 0) cout << " ";
        cout << arr[i];
      }
      cout << endl;
      continue;
    } else if (k == sum) {
      cout << endl;
      continue;
    }
    long long left = 1, right = 1e9;
    while (left < right) {
      long long mid = (left + right) / 2;
      if (check(mid))
        right = mid;
      else
        left = mid + 1;
    }
    long long res = 0;
    for (long long i = 0; i < n; i++) {
      res += min(arr[i], right - 1);
    }
    long long start = 0;
    for (long long i = 0; i < n; i++) {
      if (arr[i] >= right) res++;
      if (res == k) {
        start = i;
        break;
      }
    }
    long long cnt = 0;
    for (long long i = start + 1; i < n; i++) {
      if (arr[i] >= right) ans[cnt++] = i + 1;
    }
    for (long long i = 0; i <= start; i++) {
      if (arr[i] > right) ans[cnt++] = i + 1;
    }
    for (long long i = 0; i < cnt; i++) {
      if (i != 0) cout << " ";
      cout << ans[i];
    }
    cout << endl;
  }
  return 0;
}
