import java.io.*;
import java.util.*;

public class Main {
  
    String fileName = ""; // here you sould write name of input file (for example "input")
    
    void solve() throws IOException {
      int n = readInt();
      char [] arr = readString().toCharArray();
      ArrayList<int[]> ans = new ArrayList();
      int startIndex = 0;
      if(n % 4 == 2) {
        for(int i=0;i<3;i++) {
          ans.add(new int[]{i, i + 3});
        }
        startIndex = 6;
      }
      for(int i=startIndex; i < n; i+=4) {
        ans.add(new int[]{i, i + 2});
        ans.add(new int[]{i + 1, i + 3});
      }
      for(int i=0;i<n/2;i++) {
        if(arr[ans.get(i)[0]] == 'R' && arr[ans.get(i)[1]] == 'L') {
          out.println(ans.get(i)[1] + 1 + " " + (int)(ans.get(i)[0] + 1));
        } else {
          out.println(ans.get(i)[0] + 1 + " " + (int)(ans.get(i)[1] + 1));
        }
      }
    }

    public static void main(String[] args) throws NumberFormatException, IOException {
        new Main().run(); // this is main method of all Java programms 
                          // Here you create a new Class Main and simultaniously run it with method
                          
    }

    void run() throws NumberFormatException, IOException {
        solve();   // there start method solve, where you must write your solution
        out.close();  // it is important line - it is closing output stream
                      // if in your code no line "out.close()", then you output file will be clean
    } 

    BufferedReader in; // if is a class, that connect variable "in" and input stream
    PrintWriter out; // same for output stream

    StringTokenizer tok; // this is tokenizer - 
    String delim = " ";  // and delimiter
      
    
    Main() throws FileNotFoundException {   // this is a constructor of class Main
        try {                               // here you connect all streams with corresponds variables 
            in = new BufferedReader(new FileReader("input.txt"));
            out = new PrintWriter("output.txt");
        } catch (Exception e) {
            if (fileName.isEmpty()) {
                in = new BufferedReader(new InputStreamReader(System.in));
                out = new PrintWriter(System.out);
            } else {
                in = new BufferedReader(new FileReader(fileName + ".in"));
                out = new PrintWriter(fileName + ".out");
            }

        }
        tok = new StringTokenizer("");
    }

    String readLine() throws IOException { // return you a line (obvious)
        return in.readLine();
    }

    String readString() throws IOException { // return you a substring until delimiter
        while (!tok.hasMoreTokens()) {       // for example:
            String nextLine = readLine();    // input : 
            if (null == nextLine) {          // ABCE QWE
                return null;                 // delimiter = " "
            }                                // String a = readString(); a is equal to "ABCE"

            tok = new StringTokenizer(nextLine);
        }
        return tok.nextToken(delim);
    }

    int readInt() throws NumberFormatException, IOException { // get string from input and convert it in Integer
        return Integer.parseInt(readString());
    }

    long readLong() throws NumberFormatException, IOException {// get string from input and convert it in Long
        return Long.parseLong(readString());
    }

    int[] readIntArray(int n) throws NumberFormatException, IOException { 
        int[] a = new int[n];   // get n strings from input and convert it in Integer array
        for (int i = 0; i < n; ++i) {
            a[i] = readInt();
        }
        return a;
    }

    double readDouble() throws NumberFormatException, IOException { // get string from input and convert it in Double
        return Double.parseDouble(readString());
    }
    void sortIntArray(int [] a) {  // sort your Integer array with O(n logn) time 
                                   // It is important because of default Arrays.sort() for int array use QuickSort, which may be very slow :D
        
        Integer[] arr = new Integer[a.length];  
        for (int i = 0; i < a.length; i++) {
            arr[i] = a[i];
        }
        Arrays.sort(arr);
        for (int i = 0; i < a.length; i++) {
            a[i] = arr[i];
        }
    }

    // that is all :D 
}
