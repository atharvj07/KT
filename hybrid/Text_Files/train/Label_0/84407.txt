from sys import stdin, stdout
input = stdin.readline
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	s1 = set()
	s2 = set()
	for i in range(n):
		if a[i] in s1:
			s2.add(a[i])
		else:
			s1.add(a[i])
	ans = 0
	for i in range(101):
		if i not in s1:
			ans += i
			break
	for i in range(101):
		if i not in s2:
			ans += i
			break
	print(ans)
