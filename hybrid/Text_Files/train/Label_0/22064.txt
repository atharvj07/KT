#include <iostream>
#include <algorithm>
#include <vector>
#include <stdio.h>
using namespace std;

int main(void){
	int W,H,w,h,x,y;
	cin >> W >> H >> w >> h >> x >> y;
	double ox,oy;
	ox = (double)(min(x+w/2,W/2)+max(x-w/2,-W/2))/2;
	oy = (double)(min(y+h/2,H/2)+max(y-h/2,-H/2))/2;
	printf("%.7lf\n",oy/ox);
	//cout << x-w/2 << " " << -W/2 << " " << ox << endl;
	return 0;
}
