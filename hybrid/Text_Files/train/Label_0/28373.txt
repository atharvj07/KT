/*
	边覆盖与不覆盖作为状态非常不好 dp
	可能有多个点覆盖这条边然后就转移不了
	考虑容斥, 这样子我就可以只关注图的某种连通性了
	说到连通性 dp, 常见套路是枚举断不断父边, 然后就可以做了 
*/
#include <bits/stdc++.h>
#define N 5005

using namespace std;

const int mod = 1e9 + 7;

int n;
int head[N], nex[N << 1], to[N << 1], cntE;
int f[N][N], g[N], siz[N], tmp[N];

inline void addE(int u, int v) {
	to[++cntE] = v;
	nex[cntE] = head[u], head[u] = cntE;
}
void dfs(int u, int fa) {
	f[u][siz[u] = 1] = 1;
	for(int i = head[u], v; i; i = nex[i]) {
		v = to[i];
		if(v == fa) continue;
		dfs(v, u);
		memset(tmp, 0, sizeof tmp);
		for(int i = 1; i <= siz[u]; ++i)
			for(int j = 0; j <= siz[v]; ++j)
				tmp[i + j] = (tmp[i + j] + 1LL * f[u][i] * f[v][j]) % mod;
		siz[u] += siz[v];
		for(int i = 1; i <= siz[u]; ++i)
			f[u][i] = tmp[i];
	}
	for(int i = 2; i <= siz[u]; ++i)
		f[u][0] = (f[u][0] - 1LL * f[u][i] * g[i] % mod + mod) % mod;
}

int main() {
	scanf("%d", &n);
	for(int i = 1, u, v; i < n; ++i) {
		scanf("%d %d", &u, &v);
		addE(u, v), addE(v, u);
	}
	
	g[0] = 1;
	for(int i = 2; i <= n; i += 2)
		g[i] = (i - 1LL) * g[i - 2] % mod;
	
	dfs(1, 0);
	
	printf("%d\n", mod - f[1][0]);
	return 0;
}
