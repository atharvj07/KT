#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstring>
#include <vector>
using namespace std;
#define SIZE 100010

int a[SIZE];
int go[SIZE];
int f[20][SIZE];
int n, m;
vector<int> t[SIZE];

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 0; i < n; ++i)
        scanf("%d", a + i);
    for (int i = 2; i < SIZE; ++i)
        if (t[i].empty()) {
            go[i] = n;
            for (int j = i; j < SIZE; j += i)
                t[j].push_back(i);
        }
    f[0][n] = n;
    for (int i = n - 1; i >= 0; --i) {
        f[0][i] = f[0][i + 1];
        for (int j : t[a[i]]) {
            f[0][i] = min(f[0][i], go[j]);
            go[j] = i;
        }
    }
    for (int j = 1; j < 20; ++j) {
        for (int i = 0; i <= n; ++i) {
            f[j][i] = f[j - 1][f[j - 1][i]];
        }
    }
    while (m--) {
        int l, r;
        int ans = 0;
        scanf("%d%d", &l, &r);
        --l, --r;
        for (int i = 19; i >= 0; --i) {
            if (f[i][l] <= r) {
                l = f[i][l];
                ans += (1 << i);
            }
        }
        printf("%d\n", ans + 1);
    }
    return 0;
}
