import java.io.*;
import java.util.*;

public class F536 {
	public static void main(String[] args) { new F536(); }
	FS in = new FS();
	PrintWriter out = new PrintWriter(System.out);

	int n, m, k;
	long exp, mod = 998_244_353;
	long[] b;
	long[][] I, X;
	
	F536() {
		k = in.nextInt();
		b = new long[k];
		for (int i = 0; i < k; i++) 
			b[i] = in.nextLong();
		n = in.nextInt();
		m = in.nextInt();
		
		I = identity(k);
		X = new long[k][k];
		for (int i = 0; i < k; i++)
			X[0][i] = b[i];
		for (int i = 0; i < k - 1; i++)
			X[i + 1][i] = 1;
		X = matpow(X, n - k);
		exp = X[0][0];

		out.println(discrete_root(exp, m));	
		out.close();
	}

	long discrete_root(long x, long y) {
		long g = 3;
		int sq = (int)Math.sqrt(mod) + 1;
		HashMap<Long, Long> map = new HashMap<>();
		for (long i = 1; i <= sq; i++)
			map.put(powmod(g, mult(i, mult(sq, x, mod - 1), mod - 1)), i);
		for (long i = 0; i < sq; i++) {
			long z = mult(y, powmod(g, mult(i, x, mod - 1)), mod);
			if (map.containsKey(z)) {
				long t = map.get(z) * sq - i;
				return powmod(g, t);
			}
		} return -1;
	}
	long generate(long p) {
		int ptr = 0;
		long[] pr = new long[20];
		long phi = p - 1, t = phi;
		for (long i = 2; i * i <= phi; i++)
			if (t %i == 0) {
				while (t %i == 0) t /= i;	
				pr[ptr++] = i;
			}
		if (t > 1) pr[ptr++] = t;
		for (long i = 2; i < p; i++) {
			boolean is = true;
			for (int j = 0; j < ptr; j++) 
				is &= (powmod(i, phi / pr[j]) != 1);
			if (is) return i;
		} return -1;
	}
	long powmod(long x, long y) {
		long res = 1L;
		while (y > 0) {
			if ((y & 1L) > 0) res = mult(res, x, mod);
			x = mult(x, x, mod);
			y >>= 1L;
		} return res;
	}
	
	long[][] matpow(long[][] x, long y) {
		long[][] res = I;
		while (y > 0) {
			if ((y & 1L) > 0) res = mult(res, x);
			x = mult(x, x);
			y >>= 1L;
		} return res;
	}
	long[][] mult(long[][] x, long[][] y) {
		long[][] z = new long[x.length][y[0].length];
		for (int i = 0; i < x.length; i++)
			for (int j = 0; j < y[0].length; j++)
				for (int k = 0; k < x[0].length; k++)
					z[i][j] = add(z[i][j], mult(x[i][k], y[k][j], mod - 1), mod - 1);
		return z;
	}
	long[][] identity(int sz) {
		long[][] res = new long[sz][sz];
		for (int i = 0; i < sz; i++)
			res[i][i] = 1;
		return res;
	}

	long mult(long x, long y, long p) {
		long pd = x * y;
		if (pd >= mod) pd = pd % p;
		return pd;
	}
	long add(long x, long y, long p) {
		long sm = x + y;
		if (sm >= mod) sm = sm - p;
		return sm;
	}
	
	int abs(int x) { if (x < 0) return -x; return x; }
	long abs(long x) { if (x < 0) return -x; return x; }
	int max(int x, int y) { if (x < y) return y; return x; }
	int min(int x, int y) { if (x > y) return y; return x; }
	long max(long x, long y) { if (x < y) return y; return x; }
	long min(long x, long y) { if (x > y) return y; return x; }
	int gcd(int x, int y) { while (y != 0) { x = y^(x^(y = x)); y %= x; } return x; }
	long gcd(long x, long y) { while (y != 0) { x = y^(x^(y = x)); y %= x; } return x; }
	long lcm(int x, int y) { long xy = x; xy *= y; return xy / gcd(x, y); }
	long lcm(long x, long y) { return (x * y) / gcd(x, y); }

	class FS {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");
		String next() {
			while (!st.hasMoreTokens()) {
				try { st = new StringTokenizer(br.readLine()); }
				catch (Exception e) {}
			} return st.nextToken();
		}
		int nextInt() { return Integer.parseInt(next()); }
		long nextLong() { return Long.parseLong(next()); }
		double nextDouble() { return Double.parseDouble(next()); }

		void intArr(int sz, int[] x) { for (int i = 0; i < sz; i++) x[i] = nextInt(); }
		void longArr(int sz, long[] x) { for (int i = 0; i < sz; i++) x[i] = nextLong(); }
		void doubleArr(int sz, double[] x) { for (int i = 0; i < sz; i++) x[i] = nextDouble(); }
	}
}
