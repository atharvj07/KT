#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0);
    const int inf=(1<<30);
    const int lim=2000000;
    int n;
    cin >> n;
    vector<array<int,2>> a(n);
    string dir(n,'$');
    for(int i=0;i<n;i++)
    {
        cin >> a[i][0] >> a[i][1] >> dir[i];
        a[i][0]*=10;
        a[i][1]*=10;
    }
    int res=inf;
    //One
    vector<array<int,4>> v(n);
    for(int i=0;i<n;i++) v[i]={a[i][0]+a[i][1],a[i][0],a[i][1],i};
    sort(v.begin(),v.end());
    int one=0;
    while(one<n)
    {
        int r=-1,d=-1;
        int two=one;
        while(two+1<n&&v[two][0]==v[two+1][0]) two++;
        for(int i=one;i<=two;i++)
        {
            if(dir[v[i][3]]=='L') if(d!=-1) res=min(res,v[i][1]-d);
            if(dir[v[i][3]]=='U') if(r!=-1) res=min(res,v[i][1]-r);
            if(dir[v[i][3]]=='R') r=v[i][1];
            if(dir[v[i][3]]=='D') d=v[i][1];
        }
        one=two+1;
    }
    //Two
    for(int i=0;i<n;i++) v[i]={a[i][0]+(lim-a[i][1]),a[i][0],a[i][1],i};
    sort(v.begin(),v.end());
    one=0;
    while(one<n)
    {
        int r=-1,u=-1;
        int two=one;
        while(two+1<n&&v[two][0]==v[two+1][0]) two++;
        for(int i=one;i<=two;i++)
        {
            if(dir[v[i][3]]=='L') if(u!=-1) res=min(res,v[i][1]-u);
            if(dir[v[i][3]]=='D') if(r!=-1) res=min(res,v[i][1]-r);
            if(dir[v[i][3]]=='R') r=v[i][1];
            if(dir[v[i][3]]=='U') u=v[i][1];
        }
        one=two+1;
    }
    //Three
    vector<array<int,3>> b; //y,x,L|R
    auto ch=[&]()
    {
        sort(b.begin(),b.end());
        for(int i=0;i<(int)b.size();i++) if(b[i][0]==b[i+1][0]&&b[i][2]==1&&b[i+1][2]==0) res=min(res,(b[i+1][1]-b[i][1])/2);
    };
    for(int i=0;i<n;i++) if(dir[i]=='L'||dir[i]=='R') b.push_back({a[i][1],a[i][0],(dir[i]=='R')});
    ch();
    b.clear();
    for(int i=0;i<n;i++) if(dir[i]=='D'||dir[i]=='U') b.push_back({a[i][0],a[i][1],(dir[i]=='U')});
    ch();
    if(res==inf) cout << "SAFE\n";
    else cout << res << "\n";
    return 0;
}
