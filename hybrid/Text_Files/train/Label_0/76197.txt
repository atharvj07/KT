T = int(input())
for i in range(T):
	s = input()
	t = input()
	n = len(s)
	m = len(t)
	d = []
	for _ in range(26):
		d.append([])
	if set(s).intersection(set(t)) != set(t):
		print(-1)
	else:
		for j in range(n):
			while len(d[ord(s[j]) - 97]) < j + 1:
				d[ord(s[j]) - 97].append(j)
		for j in range(n):
			d[ord(s[j]) - 97] += [-1] * (n - len(d[ord(s[j]) - 97]))
		q = 0
		k = 1
		for j in range(m):
			if q == n:
				k += 1
				q = d[ord(t[j]) - 97][0] + 1
			elif d[ord(t[j]) - 97][q] == -1:
				k += 1
				q = d[ord(t[j]) - 97][0] + 1
			elif q > d[ord(t[j]) - 97][q]:
				k += 1
				q = d[ord(t[j]) - 97][0] + 1
			else:
				q = d[ord(t[j]) - 97][q] + 1
		print(k)
