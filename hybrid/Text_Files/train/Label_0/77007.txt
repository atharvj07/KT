def Bisect_Int(ok, ng, is_ok):
	while abs(ok - ng) > 1:
		mid = (ok + ng) // 2
		if is_ok(mid):
			ok = mid
		else:
			ng = mid
	return ok
for t in range(int(input())):
	N = int(input())
	A = list(map(int, input().split()))

	def is_ok(n):
		bit = (1 << n + 1) - 1
		dp = bit
		for a in A:
			dp = dp << a | dp >> a
			dp &= bit
		return dp != 0
	ans = Bisect_Int(max(A) * 2 + 1, -1, is_ok)
	print(ans)
