import java.util.*;
 

public class Facetook_Priority_Wall {
 
	
	public static void main(String[] args) {
		
		Scanner in = new Scanner(System.in); 
        String myName = in.nextLine().trim();
        int n = Integer.parseInt(in.nextLine().trim());
		Map<String , Integer> map = new HashMap<>();

		for(int i=0 ; i < n ; i++){
			StringTokenizer st = new StringTokenizer(in.nextLine() , " \t\r\n'");
			String name = st.nextToken();
			if(!name.equals(myName) && !map.containsKey(name)){
				map.put(name, 0);
			}
			
			String action = st.nextToken();
			
            if (!action.equals("likes")) {
                st.nextToken();
            }
            
            String who = st.nextToken();
            if(!who.equals(myName) && !map.containsKey(who)){
            	map.put(who, 0);
            }
            
            if(who.equals(myName)){
            	String tmp = name;
            	name = who;
            	who = tmp;
            }
            
            if(!name.equals(myName)){
            	continue;
            }
            
            Integer val = map.get(who);
            if(action.equals("posted")){
            	val += 15;
            }else if(action.equals("commented")){
            	val += 10;
            }else{
            	val += 5;
            }
            
            map.put(who, val);
            
		}
		
        Map.Entry<String, Integer>[] ar = map.entrySet().toArray(new Map.Entry[map.size()]);
        Arrays.sort(ar, new Comparator<Map.Entry<String, Integer>>() {
            public int compare(Map.Entry<String, Integer> o1, Map.Entry<String, Integer> o2) {
                if (o1.getValue().intValue() != o2.getValue().intValue()) {
                    return o2.getValue() - o1.getValue();
                }
                return o1.getKey().compareTo(o2.getKey());
            }
        });
        for (Map.Entry<String, Integer> a : ar) {
            System.out.println(a.getKey());
        }
		
		
	}
 
	
 
 
	
}
