import sys
import math
import collections
import heapq
input = sys.stdin.readline
n = int(input())
l = sorted([int(i) for i in input().split()])
l1 = []
p = 1
j = 0
while j < 31:
	l1.append(p)
	p *= 2
	j += 1
d = collections.Counter(l)
t = []
c = 0
for i in d:
	c1 = 0
	for j in range(31):
		if i + l1[j] in d and i + 2 * l1[j] in d:
			k1 = l1[j]
			c1 = 1
			break
	if c1 == 1:
		t = [i, i + k1, i + 2 * k1]
		c = 1
		break
if c == 0:
	c1 = 0
	for i in d:
		for j in range(31):
			if i + l1[j] in d:
				k1 = l1[j]
				c1 = 1
				break
		if c1 == 1:
			t = [i, i + k1]
			c = 1
			break
	if c == 0:
		print(1)
		print(l[0])
	else:
		print(2)
		print(*t)
else:
	print(3)
	print(*t)
