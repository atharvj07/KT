from sys import stdin
n = int(stdin.readline().strip())
arr = []
for _ in range(n):
	arr.append(list(map(int, stdin.readline().split())))
ans = []
last = 1
ans.append(last)
while len(ans) < n:
	(first, second) = arr[last - 1]
	if second in arr[first - 1]:
		ans.append(first)
		ans.append(second)
		last = second
	else:
		ans.append(second)
		ans.append(first)
		last = first
for i in range(n):
	print(ans[i], end=' ')
