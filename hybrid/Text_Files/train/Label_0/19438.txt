#include <bits/stdc++.h>
#pragma warning(disable : 4996)
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long n;
  cin >> n;
  vector<long long> a(n * 2);
  long long l = 1, r = 2 * n;
  bool left = true;
  for (long long i = 0; i < n; i++) {
    if (left) {
      a[i] = l;
      a[i + n] = l + 1;
      l += 2;
    } else {
      a[i] = r;
      a[i + n] = r - 1;
      r -= 2;
    }
    left = !left;
  }
  set<long long> all;
  long long sum = 0;
  for (long long j = 0; j < n; j++) {
    sum += a[j];
  }
  all.insert(sum);
  for (long long i = 0; i < 2 * n; i++) {
    sum -= a[i];
    sum += a[(i + n) % (2 * n)];
    all.insert(sum);
  }
  if (abs(*all.begin() - *all.rbegin()) <= 1) {
    cout << "YES\n";
    for (auto it : a) cout << it << " ";
  } else
    cout << "NO\n";
  return 0;
}
