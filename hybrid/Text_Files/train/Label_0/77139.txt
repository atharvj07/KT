import sys
from collections import deque
input = sys.stdin.readline
(n, m, k) = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
for i in range(k):
	a[i] -= 1
g = [[] for i in range(n)]
for _ in range(m):
	(u, v) = map(int, input().split())
	g[u - 1].append(v - 1)
	g[v - 1].append(u - 1)

def bfs(s):
	dist = [10 ** 18] * n
	dist[s] = 0
	dq = deque()
	dq.append([-1, s])
	while dq:
		(par, ver) = dq.popleft()
		for to in g[ver]:
			if par != to and dist[to] > dist[ver] + 1:
				dist[to] = dist[ver] + 1
				dq.append([ver, to])
	return dist
d0 = bfs(0)
d1 = bfs(n - 1)
l = []
for aa in a:
	l.append([d0[aa] - d1[aa], aa])
l.sort()
ans = 0
max_x = -10 ** 18
for i in range(len(l)):
	ans = max(ans, max_x + d1[l[i][1]])
	max_x = max(max_x, d0[l[i][1]])
print(min(ans + 1, d0[n - 1]))
