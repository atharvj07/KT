def main():
	n = int(input())
	l = list((tuple(map(int, input().split())) for _ in range(n)))
	(xx, yy) = ([0] * 100001, [0] * 100001)
	for (x, y) in l:
		xx[x] += 1
		yy[y] += 1
	n -= 1
	for (i, (_, y)) in enumerate(l):
		x = xx[y]
		l[i] = '{:d} {:d}'.format(n + x, n - x)
	print('\n'.join(l))
main()
