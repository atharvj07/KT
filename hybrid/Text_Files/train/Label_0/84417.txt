from collections import defaultdict
mydict = defaultdict(lambda : 0)
cases = int(input())
for _ in range(cases):
	n = int(input())
	line = input().split()
	L = []
	for elem in line:
		L.append(int(elem))
	if n <= 2:
		print(0)
		continue
	consider = []
	for l in range(n - 1):
		mydict = defaultdict(lambda : 0)
		for r in range(l + 1, n):
			d = (L[r] - L[l]) / (r - l)
			mydict[d] += 1
		z = max(mydict.values())
		consider.append(z)
	z = max(consider)
	print(n - z - 1)
