(n, t) = map(int, input().split())
z = list(map(int, input().split()))
if sum(z) < t:
	print(-1)
	exit()
else:
	import math
	mini = math.inf
	z.sort()
	z = z[::-1]
	l = 1
	u = len(z)
	from collections import *
	al = defaultdict(int)
	while l <= u and l >= 1:
		i = (l + u) // 2
		m = z.copy()
		j = 0
		total = 0
		flag = 0
		while j < len(m):
			r = j // i
			total += max(z[j] - r, 0)
			if total >= t:
				mini = min(mini, i)
				flag = 1
			j += 1
		if al[i] == 1:
			break
		else:
			al[i] = 1
		if flag == 0:
			l = i + 1
		else:
			u = i - 1
	print(mini)
