
import java.util.PriorityQueue;
import java.util.Scanner;

public class C {

	/**
	 * @param args
	 */
	public static void main(String[] args)
	{
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		int k = scan.nextInt();
		char[] w1 = scan.next().toCharArray();
		char[] w2 = scan.next().toCharArray();
		char[][] w = { w1, w2 };
		boolean[][] v = new boolean[2][n];

		PriorityQueue<Nin> q = new PriorityQueue<Nin>();
		q.add(new Nin(0, 0, 0));
		boolean ret = false;
		while (!q.isEmpty())
		{
			Nin front = q.poll();
			if (front.pos == n - 1)
			{
				ret = true;
				break;
			}
			int ops = front.pos + k;
			if (ops >= n)
			{
				ret = true;
				break;
			}

			int ww = front.w ^ 1;
			int p = front.pos;

			if (w[ww][ops] == '-' && !v[ww][ops])
			{
				v[ww][ops] = true;
				q.add(new Nin(ww, ops, front.t + 1));
			}
			if (w[front.w][p + 1] == '-' && !v[front.w][p + 1])
			{
				v[front.w][p + 1] = true;
				q.add(new Nin(front.w, p + 1, front.t + 1));
			}
			if (p > 0 && (front.t < p - 1) && w[front.w][p - 1] == '-'
					&& !v[front.w][p - 1])
			{
				v[front.w][p - 1] = true;
				q.add(new Nin(front.w, p - 1, front.t + 1));
			}
		}
		if (ret)
			System.out.println("YES");
		else
			System.out.println("NO");
	}
}

class Nin implements Comparable<Nin> {
	int w;
	int pos;
	int t;

	public Nin(int w, int pos, int tt)
	{
		this.w = w;
		this.pos = pos;
		t = tt;
	}

	@Override
	public int compareTo(Nin o)
	{
		return o.pos - pos;
	}

	@Override
	public String toString()
	{
		return w + " " + pos;
	}
}
