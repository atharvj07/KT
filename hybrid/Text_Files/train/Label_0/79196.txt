from math import ceil, floor, log
for _ in range(int(input())):
	(n, m, x) = map(int, input().split(' '))
	arr = [int(w) for w in input().split(' ')]
	p = []
	for i in range(n):
		p.append([i, arr[i]])
	p.sort(key=lambda x: x[1])
	curr = 0
	res = [0] * n
	for item in p:
		res[item[0]] = curr % m + 1
		curr = curr + 1
	print('YES')
	for item in res:
		print(item, end=' ')
	print('')
