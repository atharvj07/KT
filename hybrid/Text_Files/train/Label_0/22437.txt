template <typename T>
void ckmin(T& a, const T& b) {
  a = min(a, b);
}
template <typename T>
void ckmax(T& a, const T& b) {
  a = max(a, b);
}
int check(vector<int>& v) {
  bool rev = v[0] > v[v.size() - 1];
  if (rev) {
    reverse(v.begin(), v.end());
  }
  int a = v[0];
  int d = (v[v.size() - 1] - a);
  if (d % (v.size() - 1) != 0) {
    if (rev) {
      reverse(v.begin(), v.end());
    }
    return 1e8;
  }
  if (v.size() > 1) d /= (v.size() - 1);
  int cost = 0;
  for (int i = (0); i < (v.size()); i++) {
    if (abs(v[i] - (a + d * i)) > 1)
      cost = 1e8;
    else
      cost += abs(v[i] - (a + d * i));
  }
  if (rev) {
    reverse(v.begin(), v.end());
  }
  return cost;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cout << fixed << setprecision(15);
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = (0); i < (n); i++) cin >> v[i];
  if (n == 1) {
    cout << 0 << endl;
    return 0;
  }
  if (v[0] > v[n - 1]) {
    reverse(v.begin(), v.end());
  }
  int cost = 1e8;
  for (int i = -1; i <= 1; i++) {
    for (int j = -1; j <= 1; j++) {
      v[0] += i;
      v[n - 1] += j;
      cost = min(cost, check(v) + abs(i) + abs(j));
      v[0] -= i;
      v[n - 1] -= j;
    }
  }
  if (cost > 1e6) {
    cout << -1 << endl;
  } else {
    cout << cost << endl;
  }
  return 0;
}
