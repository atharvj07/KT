from collections import *

def dfs():
	while qF and qP:
		for _ in range(len(qP)):
			curr = qP.popleft()
			if curr in visitedF:
				continue
			if len(corridors[curr]) == 1 and corridors[curr][0] in visitedP:
				return 'YES'
			for child in corridors[curr]:
				if child not in visitedP:
					qP.append(child)
					visitedP.add(child)
		for _ in range(len(qF)):
			curr = qF.popleft()
			for child in corridors[curr]:
				if child not in visitedF:
					qF.append(child)
					visitedF.add(child)
	return 'NO'
t = int(input())
for _ in range(t):
	input()
	(rooms, friends) = map(int, input().split())
	fr = list(map(int, input().split()))
	visitedF = set(fr)
	visitedP = set([1])
	qF = deque(fr)
	qP = deque([1])
	corridors = defaultdict(list)
	for _ in range(rooms - 1):
		(a, b) = map(int, input().split())
		corridors[a].append(b)
		corridors[b].append(a)
	print(dfs())
