

import java.util.Scanner;

public class A {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		long a=sc.nextLong();
		long b=sc.nextLong();
		long m=sc.nextLong();
		if(a>=m||b>=m) {
			System.out.println(0);
		}else if(a<=0&&b<=0){
			System.out.println(-1);
		}else{
			long res=0;
			if(a<0){
				res+=Math.ceil(Math.abs(a)/(b*1.0));
				a+=res*b;
			}else if(b<0){
				res+=Math.ceil(Math.abs(b)/(a*1.0));
				b+=res*a;
			}
			long max=Math.max(a,b);
			long min=Math.min(a,b);
			while(max<m){
				res++;
				long tmp=max;
				max+=min;
				min=tmp;
			}
			System.out.println(res);
		}
	}
}
