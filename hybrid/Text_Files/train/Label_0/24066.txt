#include <bits/stdc++.h>
using namespace std;
long long dp[2005][2005];
int main() {
  long long i, j;
  dp[0][0] = 1;
  long long n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  long long l = n - m;
  for (i = 1; i <= l; i++) {
    dp[i][0] = dp[i - 1][1];
    for (j = 1; j <= i; j++) {
      dp[i][j] = (dp[i - 1][j + 1] + dp[i - 1][j - 1]) % 1000000007;
    }
  }
  long long count = 0;
  long long mini = 1000000;
  for (i = 0; i < m; i++) {
    if (s[i] == '(')
      count++;
    else
      count--;
    mini = min(mini, count);
  }
  long long ans = 0;
  for (i = 0; i <= l; i++) {
    for (j = 0; j <= i; j++) {
      if (j + mini >= 0 && j + count <= l - i)
        ans = (ans + dp[i][j] * dp[n - m - i][j + count] % 1000000007) %
              1000000007;
    }
  }
  cout << ans;
  return 0;
}
