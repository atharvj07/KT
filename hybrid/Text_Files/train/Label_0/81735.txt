n = int(input())
(left, right) = ([], [])
for i in range(n):
	(x, a) = [int(i) for i in input().split()]
	if x < 0:
		left.append((x, a))
	else:
		right.append((x, a))
left.sort()
right.sort()
appleNum = 0
i = 0
while i < min(len(left), len(right)):
	appleNum += left[-i - 1][1] + right[i][1]
	i += 1
if len(left) > len(right):
	appleNum += left[-i - 1][1]
elif len(right) > len(left):
	appleNum += right[i][1]
print(appleNum)
