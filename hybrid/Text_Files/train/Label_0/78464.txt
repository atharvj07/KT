def main():
	(n, t, c) = map(int, input().split())
	a = list(map(int, input().split()))
	count = 0
	pris_grp = 0
	for i in range(c):
		if a[i] > t:
			count += 1
	if count == 0:
		pris_grp += 1
	for i in range(c, n):
		if a[i - c] > t:
			count -= 1
		if a[i] > t:
			count += 1
		if count == 0:
			pris_grp += 1
	print(pris_grp)
main()
