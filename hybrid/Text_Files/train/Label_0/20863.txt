#include <bits/stdc++.h>
using namespace std;
const int maxn = 200000 + 10;
int n, c, t1[maxn], t2[maxn], dp[maxn][2];
int main() {
  cin >> n >> c;
  for (int i = 1; i <= n - 1; i++) cin >> t1[i];
  for (int i = 1; i <= n - 1; i++) cin >> t2[i];
  memset(dp, 0x7f7f7f7f, sizeof(dp));
  dp[1][0] = 0;
  dp[1][1] = c;
  for (int i = 2; i <= n; i++) {
    dp[i][0] =
        min(dp[i][0], min(dp[i - 1][0] + t1[i - 1], dp[i - 1][1] + t1[i - 1]));
    dp[i][1] = min(dp[i][1],
                   min(dp[i - 1][0] + t2[i - 1] + c, dp[i - 1][1] + t2[i - 1]));
  }
  for (int i = 1; i <= n; i++) cout << min(dp[i][0], dp[i][1]) << ' ';
  cout << endl;
  return 0;
}
