t = int(input())
while t:
	t -= 1
	n = int(input())
	daughters = []
	for i in range(n):
		daughters.append(list(map(int, input().split()))[1:])
	wishList = []
	for i in daughters:
		temp = {}
		for j in i:
			temp[j] = 0
		wishList.append(temp)
	kings = {}
	unmarried = []
	for i in range(1, n + 1):
		kings[i] = 0
	for (b, i) in enumerate(daughters):
		flag = 0
		for j in i:
			if kings[j] == 0:
				kings[j] = 1
				flag = 1
				n -= 1
				break
		if flag == 0:
			unmarried.append(b + 1)
	flag = 0
	if n == 0:
		print('OPTIMAL')
		continue
	else:
		for i in kings.keys():
			if kings[i] == 0:
				for j in unmarried:
					if i not in wishList[j - 1]:
						flag = 1
						print('IMPROVE')
						print(j, i)
						break
			if flag:
				break
	if flag == 0:
		print('OPTIMAL')
