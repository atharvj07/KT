#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) repi(i,0,n)
#define repi(i,a,b) for(int i=(int)(a);i<(int)(b);i++)
#define MOD 1000000007
#define ADD(X,Y) ((X) = ((X) + (Y)%MOD) % MOD)
typedef long long i64; typedef vector<int> ivec; typedef vector<string> svec; typedef pair<int, int> pi;
int dd[] = { 0, 1, 0, -1, 0 };
#define all(u) begin(u),end(u)
#define mp make_pair
#define pb push_back
typedef pair<pi, pi> cake;

vector<cake> G;
ivec ans;
int n,w,d;
int p, s;
int main()
{
  while(true)
    {
      scanf("%d%d%d", &n, &w, &d);
      if(n == 0 && w == 0 && d == 0)break;
      G.clear();
      G.pb(cake(pi(-1, w*d),pi(w, d)));
      rep(i,n)
	{
	  scanf("%d%d",&p,&s);
	  p--;
	  cake v = G[p];
	  G.erase(G.begin()+p);
	  int C = v.second.first;
	  int R = v.second.second;
	  s %= C+R;
	  int x1, x2, y1, y2; 
	  if(s <= C)
	    {
	      x1 = s;
	      x2 = C -s;
	      y1 = R;
	      y2 =R;
	    }
	  else
	    {
	      s -= C;
	      x1 = C;
	      x2 = C;
	      y1 = s;
	      y2 = R - s;
	    }
	  G.pb(cake(pi(i, x1*y1),pi(x1,y1)));
	  G.pb(cake(pi(i, x2*y2),pi(x2,y2)));
	  sort(G.begin(),G.end());
	}
      ans.clear();
      rep(i, G.size())
	{
	  ans.pb(G[i].first.second);
	}
      sort(ans.begin(),ans.end());
      rep(i, ans.size())
	{
	  printf("%d", ans[i]);
	  if(i == ans.size() - 1)continue;
	  printf(" ");
	}
      printf("\n");
    }
  return 0;
}
