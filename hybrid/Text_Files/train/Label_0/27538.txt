#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long n, k;
  cin >> n >> k;
  string x;
  cin >> x;
  string ans = x;
  string y = "";
  long long p = n;
  for (long long i = 1; i < n; i++) {
    long long j;
    y = "";
    for (j = 0; i + j < n; j++) {
      if (x[i + j] != ans[j]) break;
      y += ans[j];
    }
    if (i + j == n) {
      p = i;
      break;
    }
  }
  p = n - p;
  while (p--) {
    ans.erase(ans.begin());
  }
  cout << x;
  while (--k) {
    cout << ans;
  }
  cout << endl;
}
