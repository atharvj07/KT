#include <bits/stdc++.h>
using namespace std;
const long long maxn = 5e3 + 100;
int a[maxn], n, m;
int dp[maxn][maxn];
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    int x = 0;
    for (int j = i + 1; j <= n; j++)
      if (a[j] < a[i]) x++, dp[i][j] = x;
  }
  for (int i = n; i > 0; i--) {
    int p = 0;
    for (int j = i - 1; j > 0; j--)
      if (a[i] < a[j]) dp[i][j] = p, p++;
  }
  int x, y;
  int maxx = ((int)1e9);
  int l, r;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if ((j - i - dp[i][j] - dp[j][i]) < maxx)
        maxx = (j - i - dp[i][j] - dp[j][i]), l = i, r = j;
  int cnt = 0;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if ((j - i - dp[i][j] - dp[j][i]) == maxx) cnt++;
  swap(a[l], a[r]);
  int nab = 0;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if (a[j] < a[i]) nab++;
  cout << nab << " " << cnt << endl;
}
