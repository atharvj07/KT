def main():
	n = read()
	a = read(3)
	k = 2 * sum(a) // n
	b = [False] * n
	for i in range(n - 1):
		if not b[i]:
			for j in range(i + 1, n):
				if not b[j] and a[i] + a[j] == k:
					print(i + 1, j + 1)
					b[i] = b[j] = True
					break

def read(mode=1, size=None):
	if mode == 0:
		return input().strip()
	if mode == 1:
		return int(input().strip())
	if mode == 2:
		return input().strip().split()
	if mode == 3:
		return list(map(int, input().strip().split()))
	a = []
	for _ in range(size):
		a.append(read(3))
	return a
main()
