import java.util.Arrays;
import java.util.Scanner;

public class TaskA {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			short[] a = new short[n];
			short[] b = new short[n];
			for (int i = 0; i < n; i++) {
				a[i] = sc.nextShort();
			}
			for (int i = 0; i < n; i++) {
				b[i] = sc.nextShort();
			}
			Arrays.sort(a);
			Arrays.sort(b);
			for (int i = 0; i < n; i++) {
				System.out.print(String.valueOf(a[i]) + ' ');
			}
			System.out.println();
			for (int i = 0; i < n; i++) {
				System.out.print(String.valueOf(b[i]) + ' ');
			}
			System.out.println();
		}
	}
}
