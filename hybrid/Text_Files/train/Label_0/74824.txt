def cyclestring(s, n, k):
	k = k % n
	return s[n - k:] + s[:n - k]

def cyclesubstring(s, l, r, k):
	return s[:l - 1] + cyclestring(s[l - 1:r], r - l + 1, k) + s[r:]
s = input()
steps = int(input())
for _ in range(steps):
	(l, r, k) = map(int, input().split())
	s = cyclesubstring(s, l, r, k)
print(s)
