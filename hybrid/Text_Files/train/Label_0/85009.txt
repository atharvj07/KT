t = int(input())

def func(n, x, y, d):
	if abs(y - x) % d == 0:
		return abs(y - x) // d
	else:
		isfirst = False
		islast = False
		if abs(y - 1) % d == 0:
			isfirst = True
		if abs(n - y) % d == 0:
			islast = True
		if isfirst and (not islast):
			a = abs(x - 1) // d + 1
			b = abs(y - 1) // d
			return a + b
		elif not isfirst and islast:
			a = abs(n - x - 1) // d + 1
			b = abs(n - y) // d
			return a + b
		elif isfirst and islast:
			a1 = abs(x - 1) // d + 1
			b1 = abs(y - 1) // d
			a2 = abs(n - x - 1) // d + 1
			b2 = abs(n - y) // d
			return min(a1 + b1, a2 + b2)
		else:
			return -1
for i in range(t):
	(n, x, y, d) = list(map(int, input().split()))
	print(func(n, x, y, d))
