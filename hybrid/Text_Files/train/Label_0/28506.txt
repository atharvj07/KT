#include <bits/stdc++.h>
using namespace std;
const int N = 105;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    long long a, b, k;
    scanf("%lld%lld%lld", &a, &b, &k);
    if (k == 1) {
      printf("REBEL\n");
      continue;
    }
    if (a > b) {
      int x = a;
      a = b, b = x;
    }
    if (b > a * (k - 1) + gcd(a, b))
      printf("REBEL\n");
    else
      printf("OBEY\n");
  }
  return 0;
}
