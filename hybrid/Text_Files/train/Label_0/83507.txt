n = int(input())
a = [k for k in input().split(' ')]
answer = -1
l = []
dict = {}
i = 0
for j in range(len(a)):
	if dict.get(a[j]):
		dict = {}
		answer += 1
		l.append(i)
		l.append(j)
		i = j + 1
	else:
		dict.update({a[j]: 1})
if len(l) >= 1:
	l[len(l) - 1] = n - 1
if answer == -1:
	print(answer)
else:
	print(int(answer + 1))
	for p in l:
		print(str(p + 1) + ' ')
