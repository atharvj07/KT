#include <iostream>
using namespace	std;

int main() {
  int Z, N;
  int enable = 0;
  cin >> N;

  for(int i = 0; i < N; i++) {
    cin >> Z;
    for(int y = 1; y * y < Z; y++) {
      if(((Z - y) % (2 * y + 1)) == 0) {
        enable++;
        break;
      }
    }
  }
  cout << (N - enable) << endl;
  return 0;
}
