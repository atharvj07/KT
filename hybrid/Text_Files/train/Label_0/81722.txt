(v, e) = map(int, input().split())
graphs = [[set([i]) for i in range(1, v + 1)] for _ in range(101)]

def getSet(sets, vertex):
	for s in sets:
		if vertex in s:
			return s
for _ in range(e):
	(a, b, c) = map(int, input().split())
	sa = getSet(graphs[c], a)
	sb = getSet(graphs[c], b)
	if sa.isdisjoint(sb):
		sa |= sb
		graphs[c].remove(sb)
q = int(input())
for _ in range(q):
	(a, b) = map(int, input().split())
	c = 0
	for graph in graphs:
		sa = getSet(graph, a)
		sb = getSet(graph, b)
		if not sa.isdisjoint(sb):
			c += 1
	print(c)
