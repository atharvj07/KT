#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC optimization("unroll-loops")
#pragma GCC target("avx,avx2,fma")
using namespace std;
const long long int INF = 1e18;
const long long int NEGINF = -1 * INF;
long long int gcd(long long int a, long long int b) {
  if (b == 0) {
    return a;
  }
  return gcd(b, a % b);
}
long long int my_pow(long long int a, long long int n, long long int m = INF) {
  long long int res = 1;
  while (n) {
    if (n % 2) {
      res = (res * a) % m;
      n--;
    }
    a = (a * a) % m;
    n /= 2;
  }
  return res;
}
void solve() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> a;
  a.push_back(0);
  for (int i = 0; i < n - 1; i++) {
    if (s[i] == s[i + 1]) {
      a.push_back(1);
    } else {
      a.push_back(0);
    }
  }
  for (int i = 1; i < a.size(); i++) {
    a[i] += a[i - 1];
  }
  int m;
  cin >> m;
  int l, r;
  while (m--) {
    cin >> l >> r;
    r--;
    l--;
    int ans = a[r] - a[l];
    cout << ans << "\n";
  }
}
int main() {
  std::ios::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  ;
  int T = 1;
  while (T--) {
    solve();
  }
  return 0;
}
