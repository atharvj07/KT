#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < n; ++i)

vector<int> v;

int check(int x) {
    int now = 1;
    int cnt = 0;
    while(v.back() >= now + x) {
        auto it = lower_bound(v.begin(), v.end(), now);
        bool flag = false;
        int m = 0;
        if(it != v.end() and *it < now + x) flag = true, m = max({1, m, x - abs(now - *it)});
        if(it != v.begin()) {
            it--;
            flag = true, m = max({1, m, x - abs(now - *it)});
        }
        if(!flag)
            return -1;
        else
            now += m;
        cnt++;
    }
    return cnt + 1;
}
int main() {
    while(1) {
        int n, m;
        cin >> n >> m;
        v = vector<int>(n);
        rep(i, n) cin >> v[i];
        sort(v.begin(), v.end());
        if(!n) return 0;
        int l = 1, r = 1e9;
        if(check(r) == -1) {
            cout << -1 << endl;
        } else {
            l--;
            while(l < r - 1) {
                int mid = (l + r) / 2;
                int res = check(mid);
                if(res == -1)
                    l = mid;
                else if(res < m)
                    r = mid;
                else
                    l = mid;
            }
            if(check(l) == -1)
                cout << -1 << endl;
            else
                cout << l << endl;
        }
    }
}
