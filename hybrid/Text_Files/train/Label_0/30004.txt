#include<bits/stdc++.h>
using namespace std;

using ld=long double;
using P=complex<ld>;

int main(){
    int n;
    cin>>n;
    vector<P> p(n);    
    for(int i=0;i<n;i++){
        ld x,y;
        cin>>x>>y;
        p[i]=P(x,y);
    }
    if(n%2){
        cout<<"NA"<<endl;
        return 0;
    }
    P res=P(0.5,0)*(p[0]+p[n/2]);
    bool isok=true;
    for(int i=0;i<n/2;i++){
        isok&=(abs((p[i]+p[i+n/2])/P(2,0)-res)<1e-7);
    }
    if(isok){
        cout<<setprecision(10)<<fixed;
        cout<<res.real()<<" "<<res.imag()<<endl;
    }
    else{
        cout<<"NA"<<endl;
    }
    return 0;
}
