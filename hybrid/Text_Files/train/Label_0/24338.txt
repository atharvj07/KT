#include <bits/stdc++.h>
using namespace std;
void out(vector<long long> a) {
  for (long long i = 0; i < a.size(); ++i) cout << a[i] << " ";
  cout << "\n";
}
long long f(long long a) {
  long long ans = 0;
  while (a) {
    ans += a % 10;
    a /= 10;
  }
  return ans;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cout << fixed << setprecision(20);
  long long a, b, c;
  cin >> a >> b >> c;
  vector<long long> res;
  for (long long i = 1; i <= 81; ++i) {
    long long ans = 1;
    for (long long k = 0; k < a; ++k) ans *= i;
    ans = b * ans + c;
    long long x = f(ans);
    if (x == i && ans < 1e9) res.push_back(ans);
  }
  cout << res.size() << "\n";
  for (long long i = 0; i < res.size(); ++i) cout << res[i] << " ";
  return 0;
}
