def run(n, arr):
	photos = []
	top = dict(id=arr.pop(0), likes=1)
	photos.append(top)
	for id in arr:
		found = False
		for photo in photos:
			if photo['id'] == id:
				found = True
				photo['likes'] += 1
				if photo['likes'] > top['likes']:
					top = photo
				break
		if not found:
			photos.append(dict(id=id, likes=1))
	return top['id']

def read_data():
	n = int(input())
	arr = [int(x) for x in input().split(' ')]
	return (n, arr)
(n, arr) = read_data()
print(run(n, arr))
