def union(a, b):
	a = find_root(a)
	b = find_root(b)
	if a == b:
		return None
	if rank[a] >= rank[b]:
		minn[a] = min(minn[a], minn[b])
		parent[b] = a
		rank[a] += rank[b]
	else:
		minn[b] = min(minn[a], minn[b])
		parent[a] = b
		rank[b] += rank[a]

def find_root(curr):
	if parent[curr] != curr:
		parent[curr] = find_root(parent[curr])
	return parent[curr]

def solve():
	global parent, rank, arr, minn
	(n, m) = [int(x) for x in input().split()]
	arr = [int(x) for x in input().split()]
	parent = [i for i in range(n)]
	rank = [1] * n
	minn = arr[:]
	for i in range(m):
		(a, b) = [int(x) for x in input().split()]
		a -= 1
		b -= 1
		union(a, b)
	done = set()
	ans = 0
	for i in range(n):
		curr = find_root(i)
		if curr not in done:
			done.add(curr)
			ans += minn[curr]
	return ans
print(solve())
