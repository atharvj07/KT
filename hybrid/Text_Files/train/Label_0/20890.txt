#include <bits/stdc++.h>
using namespace std;
const int INF = 2147483647;
const int N = 5005;
const int MOD = 998244353;
int i, j, tab[N][N], n, r;
int main() {
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    tab[i][0] = 1;
    for (j = 1; j < i; j++) {
      tab[i][j] =
          ((i - j) * 1LL * tab[i - 1][j - 1] + (j + 1) * 1LL * tab[i - 1][j]) %
          MOD;
    }
  }
  for (i = 1; i <= n; i++) {
    r = 1;
    for (j = i + 1; j <= n; j++) r = (r * 1LL * j + tab[j][i - 1]) % MOD;
    printf("%d ", r);
  }
}
