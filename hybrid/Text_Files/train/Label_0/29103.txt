#include<bits/stdc++.h>
#define int long long
using namespace std;
signed main()
{
	int n,k,x;
	cin >> n >> k >> x;
	vector<int> arr(n);
	int i;
	for(i=0;i<n;i++)
		cin >> arr[i];
	sort(arr.begin(),arr.end());
	vector<int> co;
	for(i=0;i<n-1;i++)
	{
		if(arr[i+1]==arr[i])
			continue;
		if(arr[i+1]-arr[i]>x)
		{
			int f=arr[i+1]-arr[i]-1;
			f=f/x;
			co.push_back(f);
		}
	}
	sort(co.begin(),co.end());
//	for(i=0;i<co.size();i++)
//		cout << co[i] << ' ';
//	cout << endl;
	if(x==0)
	{
		if(co.size())
			cout << co.size()-1 << endl;
		else
			cout << 1 << endl;
	}
	else
	{
	for(i=0;i<co.size();i++)
	{
//		cout << co[i] << ' ';
		if(co[i]<=k)
			k-=co[i];
		else
			break;
	}
	cout << co.size()-i+1 << endl;
	}
}
