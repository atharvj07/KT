int bitcount(int x){
  int ret = 0;
  while(x){
    ret += x&1;
    x >>= 1;
  }
  return ret;
}

struct Edge{
  int to, cost;
  Edge(int a, int b){
    to = a;
    cost = b;
  }
};

int N, M, L, S, T;
//vector<Edge>G[305];
int dp[1<<16][16];
int G[300][300];
int jiro[16];
int eat[16];

int main()
{
  while(1){
    cin >> N >> M >> L >> S >> T; 
    if(N == 0 && M == 0 && L == 0 && S == 0 && T == 0) break;
    S--;
  rep(i, N) rep(j, N) G[i][j] = INF;
  rep(i, N) G[i][i] = 0;
  for(int i=0;i<M;i++){
    int a, b, c; cin >> a >> b >> c; a--; b--;
    //    G[a].push_back(Edge(b, c));
    //    G[b].push_back(Edge(a, c));
    G[a][b] = G[b][a] = c;
  }

  rep(i, L){
    cin >> jiro[i] >> eat[i]; jiro[i]--;
  }

  rep(k, N) G[k][k] = 0;
  rep(k, N) rep(i, N) rep(j, N){
    G[i][j] = min(G[i][j], G[i][k] + G[k][j]);
  }

  for(int i=0;i<1<<16;i++) for(int j=0;j<16;j++) dp[i][j] = INF;
  for(int i=0;i<L;i++) dp[1<<i][i] = G[S][jiro[i]] + eat[i];
  for(int i=0;i<1<<L;i++){
    for(int j=0;j<L;j++){
      for(int k=0;k<L;k++){
	dp[i|(1<<k)][k] = min(dp[i|(1<<k)][k], dp[i][j] + G[jiro[j]][jiro[k]] + eat[k]);
	//	if(dp[i|(1<<k)][k] != INF){
	//	  cout << bitset<16>(i|(1<<k)) << " " << k << " " << dp[i|(1<<k)][k] << endl;
	  //	}
      }
    }
  }
  int ret = 0;
  for(int i=0;i<1<<L;i++){
    for(int j=0;j<L;j++){
      if(dp[i][j] + G[jiro[j]][S] <= T){
	//	cout << bitset<16>(i) << " " << j << " " << " " << jiro[j] << " " << S <<" " << dp[i][j] << " " <<  G[jiro[j]][S] << endl;      
	ret = max(ret, bitcount(i));
      }
    }
  }

  cout << ret << endl;
  }
  return 0;
}
