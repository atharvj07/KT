#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 123;
pair<int, int> a[N];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) {
    int val;
    scanf("%d", &val);
    a[i].first = val;
  }
  for (int i = 1; i <= n; ++i) {
    int cost;
    scanf("%d", &cost);
    a[i].second = cost;
  }
  sort(a + 1, a + 1 + n);
  long long ans = 0;
  priority_queue<pair<int, int> > q;
  q.push(make_pair(a[1].second, a[1].first));
  int cur = a[1].first;
  for (int i = 2; i <= n; ++i) {
    while (cur < a[i].first && !q.empty()) {
      pair<int, int> b = q.top();
      q.pop();
      ans += 1LL * b.first * (cur - b.second);
      ++cur;
    }
    cur = a[i].first;
    q.push(make_pair(a[i].second, a[i].first));
  }
  while (!q.empty()) {
    pair<int, int> b = q.top();
    q.pop();
    ans += 1LL * b.first * (cur - b.second);
    ++cur;
  }
  printf("%lld", ans);
  return 0;
}
