import java.util.Scanner;
import java.util.ArrayList;
public class Problem023_300A_Array {
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        int n=sc.nextInt();
        int zeros=0;
        ArrayList<Integer> positive = new ArrayList<Integer>();
        ArrayList<Integer> negative = new ArrayList<Integer>();
        boolean hasPositive=false;
        int temp;
        boolean moveNegativeToZero=false;
        int valueOfMoveNegativeToZero=0;
        for(int i=0;i<n;i++)
        {
            temp=sc.nextInt();
            if(temp==0)
                zeros++;
            else if(temp>0)
            {
                hasPositive=true;
                positive.add(temp);
            }
            else
                negative.add(temp);
        }
        //PRINT NEGATIVE
        if(negative.size()%2==0&&(hasPositive||(!hasPositive&&negative.size()!=2)))
        {
            moveNegativeToZero=true;
            valueOfMoveNegativeToZero=negative.get(negative.size()-1);
            negative.remove(negative.size()-1);
        }
        if(hasPositive)
        {
            System.out.print(negative.size()+" ");
            for(int i=0;i<negative.size()&&i<2;i++)
                System.out.print(negative.get(i)+" ");
        }
        else
            System.out.print(negative.size()-2+" ");
        for(int i=2;i<negative.size();i++)
            System.out.print(negative.get(i)+" ");
        System.out.println();
        //PRINT POSITIVE

        if(!hasPositive)
        {
            System.out.print("2 ");
            for(int i=0;i<negative.size()&&i<2;i++)
                System.out.print(negative.get(i)+" ");
        }
        else
        {
            System.out.print(positive.size()+" ");
            for(int i=0;i<positive.size();i++)
                System.out.print(positive.get(i)+" ");
        }
        System.out.println();
        //Print ZEROS

        System.out.print(((moveNegativeToZero)?(zeros+1):zeros)+" ");
        for(int i=0;i<zeros;i++)
            System.out.print("0 ");
        if(moveNegativeToZero)
            System.out.print(valueOfMoveNegativeToZero);
        System.out.println();
    }
}
