(a, b) = [int(x) for x in input().split()]
lista = []
aux = b
resp = 'NO'
while aux > a:
	lista.append(str(int(aux)))
	if aux % 2 == 0:
		aux /= 2
	else:
		aux -= 1
		if aux % 10 == 0:
			aux /= 10
		else:
			break
	if aux == a:
		resp = 'YES'
		lista.append(str(int(aux)))
if resp == 'YES':
	print(resp)
	print(len(lista))
	lista.reverse()
	print(' '.join(lista))
else:
	print(resp)
