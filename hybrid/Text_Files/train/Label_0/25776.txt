#include <bits/stdc++.h>
using namespace std;
const int N = 600;
char s[N];
struct huge {
  int a[N], len;
  huge() {}
  void read() {
    scanf("%s", s + 1);
    len = strlen(s + 1);
    for (int i = 1; i <= len; i++) a[len - i + 1] = s[i] - 48;
  }
  void print() {
    for (int i = len; i >= 1; i--) printf("%d", a[i]);
    puts("");
  }
  void operator=(const int &x) {
    memset(a, 0, sizeof a);
    len = 0;
    a[1] = x;
    for (; a[len + 1]; a[len + 1] = a[len] / 10, a[len] %= 10) len++;
    if (!len) len = 1;
  }
  int chu() {
    int yu = 0;
    for (int i = len; i >= 1; i--) {
      yu = yu * 10 + a[i];
      a[i] = yu / 12;
      yu %= 12;
    }
    for (; len > 1 && !a[len]; len--)
      ;
    return yu;
  }
  void cheng() {
    for (int i = 1; i <= len; i++) a[i] *= 12;
    for (int i = 1; i <= len; i++) a[i + 1] += a[i] / 10, a[i] %= 10;
    for (; a[len + 1]; a[len + 1] = a[len] / 10, a[len] %= 10) len++;
  }
  void operator+=(const huge &B) {
    len = max(len, B.len);
    for (int i = 1; i <= len; i++)
      a[i] += B.a[i], a[i + 1] += a[i] / 10, a[i] %= 10;
    if (a[len + 1]) len++;
  }
  void operator-=(const huge &B) {
    len = max(len, B.len);
    for (int i = 1; i <= len; i++) {
      a[i] -= B.a[i];
      if (a[i] < 0) a[i] += 10, a[i + 1]--;
    }
    for (; len > 1 && !a[len]; len--)
      ;
  }
} A, B;
int ans1, ans2, k = -1, cnt, a[N];
int main() {
  for (A.read(); A.len > 1 || A.a[1];)
    if ((a[++*a] = A.chu()) > 2) return puts("NO"), 0;
  for (int i = 1; i <= *a; i++) cnt += a[i];
  if (cnt != 2) return puts("NO"), 0;
  for (int i = 1; i <= *a; i++)
    if (a[i]) {
      k = i;
      break;
    }
  ans1 = k * 2 - 1;
  if (k != *a) ans1 += *a - k;
  ans2 = min((ans1 + 1) / 2, 1000);
  printf("YES\n1\n%d\n%d\n", ans1, ans2 - 1);
  A = 1;
  for (int i = 1; i <= ans1 / 2; i++) A.cheng();
  B = A;
  if (!(ans1 & 1)) B.chu();
  for (int i = 1; i <= ans2; i++) {
    A += B;
    if (k != (ans1 + 1) / 2 - i + 1) A.print();
    A -= B;
    A.cheng();
    B.chu();
  }
}
