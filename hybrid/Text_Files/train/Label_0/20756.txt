#include <bits/stdc++.h>
using namespace std;
const long long int MOD = 1e9 + 7;
long long int power(long long int a, long long int b) {
  long long int ret = 1;
  while (b) {
    if (b & 1) ret *= a;
    a *= a;
    if (ret >= MOD) ret %= MOD;
    if (a >= MOD) a %= MOD;
    b >>= 1;
  }
  return ret;
}
int main() {
std:
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int n, x, pw1 = power(2, 17), pw2 = power(2, 18);
  cin >> n >> x;
  if (n == 1) {
    cout << "YES\n" << x << "\n";
  } else if (n == 2 && x == 0) {
    cout << "NO"
         << "\n";
  } else if (n == 2) {
    cout << "YES\n0 " << x << "\n";
  } else {
    int i;
    int ans = 0;
    cout << "YES\n";
    for (i = 1; i <= n - 3; i++) {
      cout << i << " ";
      ans ^= i;
    }
    if (ans == x)
      cout << pw1 + pw2 << " " << pw1 << " " << pw2 << "\n";
    else
      cout << pw1 << " " << ((pw1 ^ x) ^ ans) << " 0 \n";
  }
  return 0;
}
