import java.util.*;
import java.awt.Point;
import java.io.*;
public class Main {
static int bit[];
public static void main(String [] args)throws Exception{
	BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
	int n=Integer.parseInt(br.readLine());
	bit=new int [n+1];
	point array[]=new point[n+1];
	array[0]=new point(Integer.MIN_VALUE,0);
	StringTokenizer st=new StringTokenizer(br.readLine());
	for(int i=1;i<=n;i++)
		array[i]=new point(Integer.parseInt(st.nextToken()),i);
		Arrays.sort(array,new X());
		long ans[]=new long[n];
		//for(int i=0;i<=n;i++)System.out.print(array[i].x+" "+array[i].y+" ");
		for(int i=1;i<=n;i++){
			int idx=array[i].y;
		  for(int j=1;j<=n-1;j++){
		  	if((idx-1)*j + 2 > n)break;
		  	ans[j]+=read((idx-1)*j + 2,Math.min(n,idx*j + 1));
		  }
		  update(idx,n);	
		}
		System.out.println();
		//for(int i=0;i<n;i++)System.out.print(array[i].x+" "+array[i].y);
		StringBuilder sb=new StringBuilder();
		for(int i=1;i<=n-1;i++)
			sb.append(ans[i]+" ");
			System.out.print(sb);
}
static int less(int val){
	int ans=0;
	while(val > 0){
		ans+=bit[val];
		val-=val&(-val);
	}
	return ans;
}
static int read(int b,int a){
	return less(a) - less(b-1);
}
static void update(int idx,int max){
	while(idx <= max){
		bit[idx]+=1;
		idx+=idx&(-idx);
	}
}
static class point {
	int x;
	int y;
	public point(int x,int y){
		this.x=x;
		this.y=y;
	}
}
static class X implements Comparator<point>{
	public int compare(point a,point b){
		if(a.x > b.x)return 1;
			if(a.x < b.x)return -1;
			return a.y - b.y;
	}
}
}
