def f(s):
	if len(s) == 1:
		return [ord(s[0]) - ord('0'), s]
	if s[0] == '0':
		return [f(s[1:])[0], '0' + f(s[1:])[1]]
	a = [f(s[1:])[0] + ord(s[0]) - ord('0'), s[0] + f(s[1:])[1]]
	b = [ord(s[0]) - ord('0') - 1 + 9 * (len(s) - 1), str(int(s[0]) - 1) + (len(s) - 1) * '9']
	if b > a:
		return b
	return a
n = input()
print(int(f(n)[1]))
