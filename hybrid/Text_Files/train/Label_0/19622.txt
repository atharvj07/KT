#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-9;
const double pi = acos(-1);
const int INF = 1e18;
const int inf = 1e9;
const int MOD = 1e9 + 7;
const int nax = 5000 + 10;
const int base = 29;
long long int hash1[nax], hash2[nax];
int dp[nax][nax];
bool ispal[nax][nax];
long long int po[nax], invmodi[nax];
string s, rs;
long long int power(long long int a, long long int n) {
  long long int ans = 1;
  while (n) {
    if (n & 1) ans = (ans % MOD * a) % MOD;
    a = (a % MOD * a) % MOD;
    n /= 2;
  }
  return ans;
}
int givemehash(int l, int r, long long int hashi[]) {
  long long int invmod = invmodi[l];
  long long int ans = (hashi[r] - hashi[l - 1]);
  if (ans < 0) ans += MOD;
  ans = (ans % MOD * invmod) % MOD;
  return ans;
}
int q, n;
int main() {
  cin >> s >> q;
  rs = s;
  reverse(rs.begin(), rs.end());
  n = s.length();
  s = "#" + s;
  rs = "#" + rs;
  po[0] = 1;
  for (int i = 1; i <= n; i++) {
    po[i] = (po[i - 1] * base) % MOD;
    invmodi[i] = power(po[i], MOD - 2);
  }
  for (int i = 1; i <= n; i++) {
    hash1[i] = (hash1[i - 1] + (s[i] - '0') * po[i]) % MOD;
  }
  for (int i = 1; i <= n; i++) {
    hash2[i] = (hash2[i - 1] + (rs[i] - '0') * po[i]) % MOD;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      int len1 = j - i + 1;
      int l1, l2, r1, r2;
      l1 = i, r2 = j;
      if (len1 % 2 == 1) {
        r1 = l2 = (l1 + len1 / 2);
      } else {
        r1 = (l1 + len1 / 2 - 1);
        l2 = r1 + 1;
      }
      int fakei = n - r2 + 1;
      int fakej = n - l2 + 1;
      long long int hash11 = givemehash(l1, r1, hash1);
      long long int hash22 = givemehash(fakei, fakej, hash2);
      if (hash11 == hash22) ispal[i][j] = 1;
    }
  }
  for (int len = 1; len <= n; len++) {
    for (int i = 1; i <= n; i++) {
      int j = i + len - 1;
      if (j > n) break;
      dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1] + ispal[i][j];
    }
  }
  while (q--) {
    int l, r;
    scanf("%d %d", &l, &r);
    printf("%d\n", dp[l][r]);
  }
  return 0;
}
