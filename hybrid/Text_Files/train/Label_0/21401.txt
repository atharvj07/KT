#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  for (int tc = 0; tc < t; tc++) {
    int n, k;
    cin >> n >> k;
    if (n == 1) {
      int x;
      cin >> x;
      if (x == k) {
        cout << "yes\n";
      } else {
        cout << "no\n";
      }
      continue;
    }
    vector<int> b(n);
    int r = 0;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      if (x < k) {
        b[i] = 0;
      } else if (x == k) {
        b[i] = 1;
        r = 1;
      } else {
        b[i] = 2;
      }
    }
    int m = 0;
    for (int i = 0; i < n; i++) {
      int p = 0;
      for (int j = i; j < min(i + 3, n); j++) {
        if (b[j] > 0) {
          p++;
        }
      }
      if (p >= 2) {
        m = 1;
      }
    }
    if (m == 1 && r == 1) {
      cout << "yes\n";
    } else {
      cout << "no\n";
    }
  }
  return 0;
}
