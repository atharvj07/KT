for _ in range(int(input())):
	n = int(input())
	l = list()
	r = list()
	for i in range(n):
		k = list(input())
		l.append(k)
	for i in range(n):
		if '*' in l[i]:
			if l[i].count('*') > 1:
				for t in range(n):
					if l[i][t] == '*':
						k = []
						k.append(i)
						k.append(t)
						r.append(k)
				break
			else:
				k = []
				k.append(i)
				k.append(l[i].index('*'))
				r.append(k)
	fr = r[0][0]
	fc = r[0][1]
	sr = r[1][0]
	sc = r[1][1]
	if fr == sr:
		try:
			l[fr + 1][fc] = '*'
			l[sr + 1][sc] = '*'
		except:
			l[fr - 1][fc] = '*'
			l[sr - 1][sc] = '*'
	elif fc == sc:
		try:
			l[fr][fc + 1] = '*'
			l[sr][sc + 1] = '*'
		except:
			l[fr][fc - 1] = '*'
			l[sr][sc - 1] = '*'
	else:
		l[fr][sc] = '*'
		l[sr][fc] = '*'
	for i in range(n):
		for j in range(n):
			print(l[i][j], end='')
		print()
