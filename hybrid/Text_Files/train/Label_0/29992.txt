#include <bits/stdc++.h>
using namespace std;
const int maxint = -1u >> 1;
string a, b;
int main() {
  cin >> a >> b;
  int cnta = 0;
  int cntb = 0;
  for (int j = 0; j < a.size(); j++) {
    if (a[j] == '1') cnta++;
  }
  for (int i = 0; i < b.size(); i++) {
    if (b[i] == '1') cntb++;
  }
  if (cnta % 2 == 1) cnta++;
  if (cnta >= cntb)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
