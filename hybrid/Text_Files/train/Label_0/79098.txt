class Dijkstra:

	class Edge:

		def __init__(self, _to, _cost):
			self.to = _to
			self.cost = _cost

	def __init__(self, V):
		self.G = [[] for i in range(V)]
		self._E = 0
		self._V = V

	@property
	def E(self):
		return self._E

	@property
	def V(self):
		return self._V

	def add(self, _from, _to, _cost):
		self.G[_from].append(self.Edge(_to, _cost))
		self._E += 1

	def multi_add(self, _from, _to, _cost):
		self.add(_from, _to, _cost)
		self.add(_to, _from, _cost)

	def shortest_path(self, s):
		import heapq
		que = []
		d = [10 ** 15] * self.V
		d[s] = 0
		heapq.heappush(que, (0, s))
		pv = [-1 for i in range(self.V)]
		while len(que) != 0:
			(cost, v) = heapq.heappop(que)
			if d[v] < cost:
				continue
			for i in range(len(self.G[v])):
				e = self.G[v][i]
				if d[e.to] > d[v] + e.cost:
					d[e.to] = d[v] + e.cost
					pv[e.to] = v
					heapq.heappush(que, (d[e.to], e.to))
		return d

	def path(self, s, e):
		(d, pv) = self.shortest_path(s)
		res = [e]
		while res[-1] != s:
			res.append(pv[res[-1]])
		return res
import sys
input = sys.stdin.buffer.readline
(n, m) = map(int, input().split())
(sx, sy, fx, fy) = map(int, input().split())
p = []
for i in range(m):
	p.append(tuple(map(int, input().split())))
chinerist = Dijkstra(m + 2)
city = [i for i in range(m)]
city.sort(key=lambda i: p[i][0])
for i in range(1, m):
	(u, v) = (city[i - 1], city[i])
	chinerist.multi_add(u + 1, v + 1, p[v][0] - p[u][0])
city.sort(key=lambda i: p[i][1])
for i in range(1, m):
	(u, v) = (city[i - 1], city[i])
	chinerist.multi_add(u + 1, v + 1, p[v][1] - p[u][1])
for i in range(m):
	(x, y) = p[i]
	chinerist.add(0, i + 1, min(abs(sx - x), abs(sy - y)))
	chinerist.add(i + 1, m + 1, abs(fx - x) + abs(fy - y))
chinerist.add(0, m + 1, abs(sx - fx) + abs(sy - fy))
ans = chinerist.shortest_path(0)[m + 1]
print(ans)
