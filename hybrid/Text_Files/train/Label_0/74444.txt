n = input()
a = n[:]
p = len(n) - 1
(count1, count2) = (0, 0)
i = p
(flag1, flag2) = (0, 0)
while i > -1:
	if n[i] == '5' and flag1 == 0:
		count1 = p - i
		n = n[:i] + n[i + 1:] + '5'
		flag1 = 1
	if a[i] == '0' and flag2 == 0:
		count2 = p - i
		a = a[:i] + a[i + 1:] + '0'
		flag2 = 1
	i = i - 1
if flag1 == 1 or flag2 == 1:
	i = p - 1
	if flag1 == 1:
		flag1 = -1
	if flag2 == 1:
		flag2 = -1
	while i > -1:
		if (int(n[i]) - 2) % 5 == 0 and flag1 == -1:
			count1 = count1 + p - i - 1
			flag1 = 2
		if int(a[i]) % 5 == 0 and flag2 == -1:
			count2 = count2 + p - i - 1
			flag2 = 2
		i = i - 1
if flag1 == 2 and flag2 == 2:
	if n[0] == '0':
		i = 0
		while n[i] == '0' and i < p - 1:
			count1 = count1 + 1
			i = i + 1
		if i == p - 1:
			count1 = count2
	print(min(count1, count2))
elif flag1 == 2 and flag2 != 2:
	print(count1)
elif flag1 != 2 and flag2 == 2:
	print(count2)
else:
	print(-1)
