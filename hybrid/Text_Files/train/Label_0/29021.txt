#include <bits/stdc++.h>
using namespace std;
void prepare_lookup_table() {}
void do_task() {
  long long n, m, s;
  cin >> n >> m >> s;
  long long ans;
  if ((s >= m && s >= n) || (n % s == 0 && m % s == 0)) {
    ans = n * m;
  } else if (s < n && s < m) {
    long long dr = (m / s) + (m % s == 0 ? 0 : 1),
              du = (n / s) + (n % s == 0 ? 0 : 1);
    long long pa = dr * du;
    long long mm = ((m) % s), mn = ((n) % s);
    if (mm == 0) mm = s;
    if (mn == 0) mn = s;
    ans = pa;
    ans *= (mm < 1 ? 1 : mm);
    ans *= (mn < 1 ? 1 : mn);
  } else if (s < max(n, m)) {
    ans = (((max(n, m) / s) + 1) * min(n, m)) * (max(n, m) % s);
  }
  cout << ans << "\n";
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  prepare_lookup_table();
  long long t = 1;
  while (t--) {
    do_task();
  }
  return 0;
}
