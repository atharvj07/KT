import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class InvarianceOfTree {
    void solve() {
        int n = in.nextInt();
        int[] P = new int[n];
        for (int i = 0; i < n; i++) P[i] = in.nextInt() - 1;

        int center1 = -1, center2 = -1;
        for (int i = 0; i < n; i++) {
            if (i == P[i]) center1 = i;
            if (i == P[P[i]]) center2 = i;
        }

        if (center1 >= 0) {
            out.println("YES");
            for (int i = 0; i < n; i++) {
                if (i != center1) {
                    out.println((center1 + 1) + " " + (i + 1));
                }
            }
            return;
        }
        if (center2 >= 0) {
            if ((n & 1) > 0) {
                out.println("NO");
                return;
            }
            boolean[] used = new boolean[n];
            int a = center2, b = P[center2];
            int[] A = new int[n / 2 - 1], B = new int[n / 2 - 1];
            int ia = 0, ib = 0;
            used[a] = used[b] = true;
            boolean ok = true;
            for (int i = 0; i < n; i++) {
                if (!used[i]) {
                    int j = i;
                    int cnt = 0;
                    while (!used[j]) {
                        if (cnt % 2 == 0) {
                            A[ia++] = j;
                        } else {
                            B[ib++] = j;
                        }
                        used[j] = true;
                        cnt++;
                        j = P[j];
                    }
                    if (cnt % 2 == 1) {
                        ok = false;
                        break;
                    }
                }
            }
            if (ok) {
                out.println("YES");
                out.println((a + 1) + " " + (b + 1));
                for (int i = 0; i < n / 2 - 1; i++) {
                    out.println((a + 1) + " " + (A[i] + 1));
                    out.println((b + 1) + " " + (B[i] + 1));
                }
                return;
            }
        }
        out.println("NO");
    }

    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new InvarianceOfTree().solve();
        out.close();
    }

    static FastScanner in;
    static PrintWriter out;

    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;

        public FastScanner(BufferedReader in) {
            this.in = in;
        }

        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(nextToken());
        }

        public long nextLong() {
            return Long.parseLong(nextToken());
        }

        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
