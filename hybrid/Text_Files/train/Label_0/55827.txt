import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class C {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int k = sc.nextInt();
		String s = sc.next();
		int n = s.length();
		if ((n + 1) / 2 < k) {
			System.out.println("IMPOSSIBLE");
			return;
		}
		int mid = (n - 1) / 2;
		char a[] = s.toCharArray();
		int l = 0, r = n - 1;
		r = n - 1;
		l = 0;
		while (l - r <= 0) {
			if (a[l] != '?' && a[r] != '?' && a[l] != a[r]) {
				System.out.println("IMPOSSIBLE");
				return;
			}
			if (a[r] != a[l] && (a[r] == '?' || a[l] == '?')) {
				if (a[r] != '?') {
					a[l] = (char) a[r];
				} else {
					a[r] = (char) a[l];
				}
			}
			r--;
			l++;
		}
		int sor = 0;
		Set<Character> set = new HashSet<Character>();
		for (int i = 0; i <= mid; i++) {
			if (a[i] != '?') {
				set.add(a[i]);
			} else {
				sor++;
			}
		}
		if (k < set.size()) {
			System.out.println("IMPOSSIBLE");
			return;
		}
		int an = sor - (k - set.size());
		for (int i = 0; i <= mid; i++) {
			if (a[i] == '?' && an > 0) {
				a[i] = 'a';
				an--;
			}
		}
		for (int i = 0; i <= mid; i++) {
			if (a[i] == '?') {
				if (set.size() <= k) {
					for (Character j = 'a'; j <= 'a' + k; j++) {
						if (!set.contains(j)) {
							set.add(j);
							a[i] = j;
							break;
						}
					}
				} else {
					System.out.println("IMPOSSIBLE");
					return;
				}
			}
		}
		r = n - 1;
		l = 0;
		while (l - r <= 0) {
			if (a[r] != a[l] && (a[r] != '?' || a[l] != '?')) {
				if (a[r] != '?') {
					a[l] = (char) a[r];
				} else {
					a[r] = (char) a[l];
				}
			}
			r--;
			l++;
		}
		String ans = "";
		for (int i = 0; i < a.length; i++) {
			ans += a[i];
		}
		if (set.size() != k) {
			System.out.println("IMPOSSIBLE");
			return;
		}
		System.out.println(ans);
	}
}
