import java.util.*;
import java.math.*;
import java.io.*;

// This is such a clean solution, if this doesn't pass I'll be sad
public class CF1037F {
	class RMQ {
		int table[][], index[][];
		public RMQ(int[] arr) {
			int n = arr.length;
			int d = Integer.numberOfTrailingZeros(Integer.highestOneBit(n)) + 2;
			table = new int[d][n];
			index = new int[d][n];
			for(int i = 0 ; i < n ; i++) {
				table[0][i] = arr[i];
				index[0][i] = i;
			}
			int lastRange = 1;
			for(int k = 1 ; k < d ; k++, lastRange <<= 1) {
				for(int i = 0 ; i < n ; i++) {
					if(table[k - 1][i] < table[k - 1][Math.min(i + lastRange, n - 1)]) {
						table[k][i] = table[k - 1][i];
						index[k][i] = index[k - 1][i];
					} else {
						table[k][i] = table[k - 1][Math.min(i + lastRange, n - 1)];
						index[k][i] = index[k - 1][Math.min(i + lastRange, n - 1)];
					}
				}
			}
		}
		public int queryMax(int i, int j) {
			int range = j - i + 1;
			int exp = Integer.highestOneBit(range);
			int k = Integer.numberOfTrailingZeros(exp);
			return -Math.min(table[k][i], table[k][j - exp + 1]);
		}
		int queryIndex(int i, int j) {
			int range = j - i + 1;
			int exp = Integer.highestOneBit(range);
			int k = Integer.numberOfTrailingZeros(exp);
			if(table[k][i] < table[k][j - exp + 1]) return index[k][i];
			else return index[k][j - exp + 1];
		}
	}
	final int MOD = (int) 1e9 + 7;
	RMQ rmq;
	int k;
	long[] cnt;
	public CF1037F() {
		FS scan = new FS();
		int n = scan.nextInt();
		k = scan.nextInt();
		cnt = new long[n + 1];
		int numRows = 1;
		for(int i = 1 ; i <= n ; i++) {
			if(i > 1 && (i - 1) % (k - 1) == 0)
				numRows++;
			cnt[i] = (cnt[i - 1] + numRows) % MOD;
		}
		int[] a = new int[n];
		for(int i = 0 ; i < n ; i++)
			a[i] = -scan.nextInt();
		rmq = new RMQ(a);
		for(int i = 0 ; i < n ; i++)
			a[i] = -a[i];
		long res = 0;
		Deque<Integer> q = new ArrayDeque<>();
		q.add(0); q.add(n - 1);
		while(!q.isEmpty()) {
			int left = q.pop(), right = q.pop();
			if(right - left + 1 < k)
				continue;
			int index = rmq.queryIndex(left, right);
			long max = rmq.queryMax(left, right);
			long tmp = cnt[right - left + 1];
			tmp = (tmp - cnt[index - left] + MOD) % MOD;
			tmp = (tmp - cnt[right - index] + MOD) % MOD;
			tmp = (tmp * max) % MOD;
			tmp = (tmp - max + MOD) % MOD;
			res = (res + tmp) % MOD;
			q.add(left); q.add(index - 1);
			q.add(index + 1); q.add(right);
		}
		System.out.println(res);
	}
	class FS {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");
		public String next() {
			while(!st.hasMoreTokens()) {
				try { st = new StringTokenizer(br.readLine()); }
				catch(Exception e) { e.printStackTrace(); }
			}
			return st.nextToken();
		}
		public int nextInt() { return Integer.parseInt(next()); }
	}
	public static void main (String[] args) { new CF1037F(); }
}
