(n, k) = map(int, input().split())
a = [int(i) for i in input().split()]

def check(a):
	possible = set()
	for num in a:
		to_add = set()
		for p in possible:
			new = p + num % k
			if new % k == 0:
				return 'YES'
			to_add.add(new % k)
		for t in to_add:
			possible.add(t)
		possible.add(num % k)
	if 0 in possible:
		return 'YES'
	return 'NO'
ans = check(a)
print(ans)
