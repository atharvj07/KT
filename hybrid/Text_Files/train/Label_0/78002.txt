input = lambda : stdin.readline().strip()
from sys import stdin, stdout

def INI():
	return int(stdin.readline())

def INL():
	return [int(_) for _ in stdin.readline().split()]

def INS():
	return stdin.readline()

def MOD():
	return pow(10, 9) + 7

def OPS(ans):
	stdout.write(str(ans) + '\n')

def OPL(ans):
	[stdout.write(str(_) + ' ') for _ in ans]
	stdout.write('\n')
from collections import deque
(n, m) = INL()
s = input()
(x, y) = ([], [])
for _ in range(m):
	(a, b) = map(int, input().split())
	x.append(a)
	y.append(b)
l = list(s)
D = [0 for _ in range(n)]
G = {_: [] for _ in range(n)}
for _ in range(m):
	(a, b) = (x[_], y[_])
	a -= 1
	b -= 1
	G[a].append(b)
	D[b] += 1
q = deque()
for _ in range(n):
	if D[_] == 0:
		q.append(_)
c = 0
ans = 0
DP = [[0 for _ in range(26)] for _ in range(n)]
while c < n and q:
	x = q.popleft()
	c += 1
	DP[x][ord(l[x]) - 97] += 1
	for _ in G[x]:
		for __ in range(26):
			DP[_][__] = max(DP[_][__], DP[x][__])
		D[_] -= 1
		if D[_] == 0:
			q.append(_)
if c != n:
	OPS(-1)
else:
	for _ in range(n):
		ans = max(ans, max(DP[_]))
	OPS(ans)
