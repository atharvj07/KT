struct Point
{
	double x,y;
	Point(){}
	Point(double X,double Y)
	{
		x=X;y=Y;
	}
	bool operator <(const Point &p)const
	{
		if(sgn(x-p.x))return x<p.x;
		else return y<p.y;
	}
	bool operator ==(const Point &p)const
	{
		return sgn(x-p.x)==0&&sgn(y-p.y)==0;
	}
	Point operator -(const Point &p)
	{
		return Point(x-p.x,y-p.y);
	}
	Point operator +(const Point &p)
	{
		return Point(x+p.x,y+p.y);
	}
	double cross(const Point &p)const
	{
		return x*p.y-y*p.x;
	}
	double len()
	{
		return sqrt(sqr(x)+sqr(y));
	}
	void unit()
	{
		x*=0.00000001;
		y*=0.00000001;
	}
}pt[maxn];
struct Segment
{
	Point st,ed;
};
double cross(const Point &p,const Point &q)
{
	return p.x*q.y-p.y*q.x;
}
double dot(const Point &p,const Point &q)
{
	return p.x*q.x+p.y*q.y;
}
Point Insect(Segment &p,Segment &q)
{
	double u=cross(p.ed-p.st,q.st-p.st),v=cross(p.st-p.ed,q.ed-p.st);
	return Point((q.st.x*v+q.ed.x*u)/(u+v),(q.st.y*v+q.ed.y*u)/(u+v));
}

int check(Point P,int deb)
{
	double avg=0;
	con[0]=con[1]=con[2]=con[3]=0;
	Segment now,tmp;
	Point cp;
	now.st=P;
	for(int j=0;j<3;++j)

		now.ed.x=P.x+go[j][0];
		now.ed.y=P.y+go[j][1];
		for(int i=0;i<n;++i)
		{
			tmp.st=pt[i];
			tmp.ed=pt[(i+1)%n];
			if(sgn((tmp.ed-tmp.st).cross(now.ed-now.st)))
			{
				cp=Insect(tmp,now);
				if(sgn(dot(tmp.ed-cp,cp-tmp.st))>0&&sgn(dot(now.ed-cp,cp-now.st))>0)
				{
					if(deb)

					}
					++con[j];
				}
			}
		}
		avg+=((con[j])%2);
	}
	return round(avg/4);
}
	
double check(int i,int j)
{
	int oi=i,oj=j;
	Segment now,tmp;
	Point cp;
	tp.clear();
	now.st=pt[i];
	now.ed=pt[j];
	//cout <<"------------Begin--------------"<<endl;cout <<now.st.x<<" "<<now.st.y<<endl;cout <<now.ed.x<<" "<<now.ed.y<<endl;
	for(i=0;i<n;++i)
	{
		tmp.st=pt[i];
		tmp.ed=pt[(i+1)%n];
		//cout <<tmp.st.x<<" "<<tmp.st.y<<endl;cout <<tmp.ed.x<<" "<<tmp.ed.y<<endl;
		//cout <<(tmp.ed-tmp.st).cross(now.ed-now.st)<<endl;
		if(sgn((tmp.ed-tmp.st).cross(now.ed-now.st)))
		{
			cp=Insect(tmp,now);
			//cout <<"HERE"<<" "<<cp.x<<" "<<cp.y<<" "<<dot(tmp.ed-cp,cp-tmp.st)<<endl;
			if(sgn(dot(tmp.ed-cp,cp-tmp.st))>0)
				tp.push_back(cp);
			else if(sgn(dot(tmp.ed-cp,cp-tmp.st))==0)
			{
				if(cp==tmp.st)j=i;
				else j=i+1;
				if(cross(pt[(j-1+n)%n]-now.st,now.ed-now.st)*cross(pt[(j+1)%n]-now.st,now.ed-now.st)<0)
				{

					//if(cas==12&&oi==12&&oj==33)cout <<"cp:"<<cp.x<<" "<<cp.y<<endl;
					tp.push_back(cp);
				}
			}
		}
		else if(sgn((now.st-tmp.st).cross(now.ed-now.st))==0&&sgn((now.st-tmp.ed).cross(now.ed-now.st))==0)
		{
			Point vec=tmp.ed-tmp.st;
			vec.unit();
			if(cross(pt[(i-1+n)%n]-now.st,now.ed-now.st)*cross(pt[(i+2)%n]-now.st,now.ed-now.st)<0)
			{
				if(check(tmp.st-vec,0)%2==0)
				{
					//if(cas==12&&oi==12&&oj==33)cout <<(tmp.st-vec).x<<" st "<<(tmp.st-vec).y<<" "<<vec.x<<" "<<vec.y<<" "<<check(tmp.st-vec,1)<<endl;
					tp.push_back(tmp.st);
				}
				if(check(tmp.ed+vec,0)%2==0)
				{
					//if(cas==12&&oi==12&&oj==33)cout <<tmp.ed.x<<" ed "<<tmp.ed.y<<endl;
					tp.push_back(tmp.ed);
				}
			}
		}
	}
	sort(tp.begin(),tp.end());
	if(cas==-1&&oi==0&&oj==61){cout <<"0--tp--0"<<endl;for(i=0;i<tp.size();++i)cout<<tp[i].x<<" "<<tp[i].y<<endl;cout <<"0--end--0"<<endl;}
	int n=tp.size();
	sp.clear();
	for(i=0;i<n;++i)
		if(i==0||(!(tp[i]==tp[i-1])))sp.push_back(tp[i]);
	if(cas==-1&&oi==0&&oj==61){cout <<"0--sp--0"<<endl;for(i=0;i<sp.size();++i)cout<<sp[i].x<<" "<<sp[i].y<<endl;cout <<"0--end--0"<<endl;}
	double ans=0;
	for(i=0;i<sp.size();i+=2)
		ans=max(ans,(sp[i+1]-sp[i]).len());
	return ans;
}
