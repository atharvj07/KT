n = int(input())
array = list(map(int, input().split()))
if n == 1:
	print('YES')
if n >= 2:
	begin = 0
	end = len(array) - 1
	while array[begin] < array[begin + 1] if begin < end else False:
		begin += 1
	while array[end] < array[end - 1] and end != begin:
		end -= 1
	test = set()
	for i in range(begin, end + 1):
		test.add(array[i])
	if len(test) == 1 or begin == end:
		print('YES')
	else:
		print('NO')
