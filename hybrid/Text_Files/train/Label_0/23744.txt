#include <bits/stdc++.h>
#define ll long long
#define fi first
#define se second
#define pb push_back
#define me memset
const int N = 1e5+10;
const int MOD = 1e9+7;
const int INF = 0x3f3f3f3f;
using namespace std;
typedef pair<int,int> PII;
typedef pair<ll,ll> PLL;
string a[N],b[N];
int vis[N][26];
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        memset(vis,0,sizeof vis);
        int n,m;
        cin>>n>>m;
        for(int i=1 ; i<=n ; i++)
        {
            cin>>a[i];
            for(int j=0 ; j<m ; j++) vis[j][a[i][j]-'a']++;
        }
        for(int i=1 ; i<n ; i++) cin>>b[i];
        string ans="";
        for(int i=0 ; i<m ; i++)
        {
            for(int j=1 ; j<n ; j++) vis[i][b[j][i]-'a']--;
            for(int j=0 ; j<=25 ; j++) if(vis[i][j]!=0) ans+=char('a'+j);
        }
        cout<<ans<<endl;
    }
    return 0;
}
