#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  bool flag;
  cin >> n >> m;
  vector<string> S(n);
  vector<int> V(m);
  for (int i = 0; i < m; i++) V[i] = 0;
  for (int i = 0; i < n; i++) cin >> S[i];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (S[i][j] == '1') V[j]++;
  for (int i = 0; i < n; i++) {
    flag = true;
    for (int j = 0; j < m; j++)
      if (V[j] < 2 && S[i][j] == '1') {
        flag = false;
        break;
      }
    if (flag) {
      cout << "YES";
      return 0;
    }
  }
  cout << "NO";
  return 0;
}
