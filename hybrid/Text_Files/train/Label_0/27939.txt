#include <bits/stdc++.h>
using namespace std;
long long mod = 1e9 + 7;
long long fast_exp(long long a, long long b) {
  if (b == 0)
    return 1;
  else {
    long long res = 1;
    res = fast_exp(a, b / 2);
    res = (res * res) % mod;
    if (b % 2 == 1) res = (res * a) % mod;
    return res;
  }
}
char a[15][15];
int n, m;
struct snake {
  int dist = 0;
  int len = 0;
  int posi[10] = {}, posj[10] = {};
};
int vis[25][25][4] = {};
int diri[4] = {0, -1, 0, 1};
int dirj[4] = {1, 0, -1, 0};
int check_safe(snake s, int di) {
  snake ns = s;
  int chk = 1;
  for (int i = s.len; i > 1; i--) {
    ns.posi[i] = s.posi[i - 1];
    ns.posj[i] = s.posj[i - 1];
  }
  ns.posi[1] = s.posi[1] + diri[di];
  ns.posj[1] = s.posj[1] + dirj[di];
  int forbi = ns.posi[1], forbj = ns.posj[1];
  for (int i = s.len; i > 1; i--)
    if (forbi == ns.posi[i] && forbj == ns.posj[i]) {
      chk = 0;
      break;
    }
  if (forbi < 0 || forbi >= n || forbj < 0 || forbj >= m) {
    chk = 0;
  } else if (a[forbi][forbj] == '#')
    chk = 0;
  return chk;
}
int snake_hash(snake s, int p) {
  int sum = 0;
  int temp = p;
  for (int i = s.len; i >= 1; i--) {
    sum = (sum + (p * s.posi[i] * i) + (i * s.posj[i])) % mod;
    p = p * temp;
  }
  return sum;
}
int main() {
  cin >> n >> m;
  snake start_sn;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
      if (a[i][j] - '0' <= 9 && a[i][j] - '0' >= 1) {
        start_sn.posi[a[i][j] - '0'] = i;
        start_sn.posj[a[i][j] - '0'] = j;
        start_sn.len++;
      }
    }
  queue<snake> q;
  q.push(start_sn);
  int iter = 1000000;
  int ans = -1;
  int cnt = 0;
  while (iter-- && !q.empty()) {
    cnt++;
    snake cur = q.front();
    q.pop();
    for (int i = 0; i < 4; i++) {
      if (check_safe(cur, i)) {
        snake latsn;
        latsn = cur;
        latsn.dist++;
        for (int j = latsn.len; j > 1; j--) {
          latsn.posi[j] = cur.posi[j - 1];
          latsn.posj[j] = cur.posj[j - 1];
        }
        latsn.posi[1] = cur.posi[1] + diri[i];
        latsn.posj[1] = cur.posj[1] + dirj[i];
        if (a[latsn.posi[1]][latsn.posj[1]] == '@') {
          ans = latsn.dist;
          break;
        }
        if (vis[latsn.posi[1]][latsn.posj[1]][i] < 15) {
          q.push(latsn);
          vis[latsn.posi[1]][latsn.posj[1]][i]++;
        }
      }
    }
    if (ans != -1) break;
  }
  cout << ans << endl;
  return 0;
}
