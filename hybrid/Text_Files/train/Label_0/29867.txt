#include <bits/stdc++.h>
using namespace std;
int n, m, k;
void solution() {
  cin >> n >> m >> k;
  if (m - k > n) {
    puts("0");
    return;
  }
  double ret = 1;
  for (int i = 0, _n = (k + 1); i < _n; ++i)
    ret *= (m - k + i) * 1. / (n + 1 + i);
  if (ret > 1 + 1e-7) throw 1;
  printf("%.8lf\n", 1.0 - ret);
}
int main() {
  solution();
  return 0;
}
