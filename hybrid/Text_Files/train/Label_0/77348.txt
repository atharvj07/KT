def solve():
	n = int(input())
	ll = list(map(int, input().split()))
	sm = sum(ll)
	for i in range(n):
		if sm % (n - i) == 0:
			val = sm // (n - i)
			temp_sm = 0
			flag = 0
			for j in range(n):
				temp_sm += ll[j]
				if temp_sm == val:
					temp_sm = 0
				elif temp_sm > val:
					flag = 1
					break
			if not flag:
				return i
	return n
t = int(input())
while t != 0:
	res = solve()
	print(res)
	t -= 1
