import java.util.*;

public class Main {
    private static class Edge {
        public int from;
        public int to;
        public int min;
        public int max;
        public long len;
        public Edge(int from, int to, int min, int max) {
            this.from = from;
            this.to = to;
            this.min = min;
            this.max = max;
            this.len = min;
        }
    }
    
    private static final long MOD = 1000000007;
    
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        
        int n = scan.nextInt();
        int m = scan.nextInt();
        int k = scan.nextInt();
        int s1 = scan.nextInt() - 1;
        int s2 = scan.nextInt() - 1;
        int f = scan.nextInt() - 1;
        
        Edge[] edges = new Edge[m + k];
        for (int i = 0; i < m; i++) {
            int from = scan.nextInt() - 1;
            int to = scan.nextInt() - 1;
            int len = scan.nextInt();
            edges[i] = new Edge(from, to, len, len);
        }
        for (int i = m; i < m + k; i++) {
            int from = scan.nextInt() - 1;
            int to = scan.nextInt() - 1;
            int min = scan.nextInt();
            int max = scan.nextInt();
            edges[i] = new Edge(from, to, min, max);
        }
        
        ArrayList<ArrayList<Edge>> adjacent = new ArrayList<ArrayList<Edge>>();
        for (int i = 0; i < n; i++) {
            adjacent.add(new ArrayList<Edge>());
        }
        for (int i = 0; i < edges.length; i++) {
            adjacent.get(edges[i].from).add(edges[i]);
        }
        
        if (check(adjacent, s1, s2, f, 1)) {
            System.out.println("WIN");
            print(edges, m, m + k);
        } else if (check(adjacent, s2, s1, f, 0)) {
            System.out.println("DRAW");
            print(edges, m, m + k);
        } else {
            System.out.println("LOSE");
        }
        
        scan.close();
    }
    
    private static boolean check(ArrayList<ArrayList<Edge>> adjacent, int s1, int s2, int f, int strategy) {
        int n = adjacent.size();
        boolean[] visited = new boolean[n];
        long[] dist = new long[n];
        Arrays.fill(dist, Long.MAX_VALUE);
        Queue<Integer> queue = new LinkedList<Integer>();
        queue.offer(s1);
        queue.offer(s2);
        visited[s1] = visited[s2] = true;
        dist[s1] = 1;
        dist[s2] = 0;
        while (!queue.isEmpty()) {
            int u = queue.poll();
            visited[u] = false;
            for (Edge e : adjacent.get(u)) {
                int v = e.to;
                long len = (dist[u] & 1) == strategy ? e.min : e.max;
                e.len = len;
                if (dist[u] + 2 * len < dist[v]) {
                    dist[v] = dist[u] + 2 * len;
                    if (!visited[v]) {
                        visited[v] = true;
                        queue.offer(v);
                    }
                }
            }
        }
        
        // assert dist[f] != Long.MAX_VALUE , acoording to problem statement.
        return (dist[f] & 1) == strategy;
    }
    
    private static void print(Edge[] edges, int from, int to) {
        for (int i = from; i < to; i++) {
            if (i > from) {
                System.out.print(" ");
            }
            System.out.print(edges[i].len);
        }
        System.out.println();
    }
    
    private static void print(int... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void print(long... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static <T> void print(T... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void read(Scanner scan, int[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (int[] arr : arrs) {
                arr[i] = scan.nextInt();
            }
        }
    }
    
    private static void read(Scanner scan, long[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (long[] arr : arrs) {
                arr[i] = scan.nextLong();
            }
        }
    }
    
    private static void decreaseByOne(int[]... arrs) {
        for (int[] arr : arrs) {
            for (int i = 0; i < arr.length; i++) {
                arr[i] --;
            }
        }
    }
}
