for x in range(int(input())):
	(n, k) = map(int, input().split())
	s = input()
	ans = 0
	if k % 2 == 0:
		for i in range(k // 2):
			li = [0] * 27
			for j in range(n // k):
				li[ord(s[k * j + i]) - 96] += 1
				li[ord(s[k * (j + 1) - (i + 1)]) - 96] += 1
			ans += 2 * (n // k) - max(li)
		print(ans)
	else:
		for i in range(k // 2):
			li = [0] * 27
			for j in range(n // k):
				li[ord(s[k * j + i]) - 96] += 1
				li[ord(s[k * (j + 1) - (i + 1)]) - 96] += 1
			ans += 2 * (n // k) - max(li)
		li = [0] * 27
		for j in range(n // k):
			li[ord(s[k * j + k // 2]) - 96] += 1
		ans += n // k - max(li)
		print(ans)
