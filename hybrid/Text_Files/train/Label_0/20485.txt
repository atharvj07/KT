#include <bits/stdc++.h>
using namespace std;
using ll = long long int;
using ii = pair<int, int>;
using vii = vector<ii>;
using vi = vector<int>;
using graph = vector<vi>;
const int INF = 0x3f3f3f3f;
const ll INFL = 0x3f3f3f3f3f3f3f3f;
const int MAXN = 3005;
struct Mqueue {
  vii q1, q2;
  Mqueue() {
    q1.clear();
    q2.clear();
  }
  void push(int x, int i = 2) {
    if (i == 2)
      q2.emplace_back(x, q2.size() ? min(x, q2.back().second) : x);
    else
      q1.emplace_back(x, q1.size() ? min(x, q1.back().second) : x);
  }
  int get() {
    if (q1.empty()) return q2.back().second;
    if (q2.empty()) return q1.back().second;
    return min(q2.back().second, q1.back().second);
  }
  void pop() {
    if (q1.empty()) {
      while (q2.size()) {
        push(q2.back().first, 1);
        q2.pop_back();
      }
    }
    q1.pop_back();
  }
  int size() { return q1.size() + q2.size(); }
  void clear() { q1.clear(), q2.clear(); }
};
int adj[MAXN][MAXN];
int main() {
  ll n, m, a, b;
  ll x, y, z, g;
  scanf("%lld %lld %lld %lld", &n, &m, &a, &b);
  scanf("%lld %lld %lld %lld", &g, &x, &y, &z);
  vector<int> v;
  v.push_back(g);
  for (int i = 1, k = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      g = v.back();
      v.emplace_back((g * x + y) % z);
      adj[i][j] = v[(i - 1) * m + j - 1];
    }
  }
  v.clear();
  Mqueue q;
  for (int j = 1; j <= m; ++j) {
    q.clear();
    for (int i = 1; i <= a; ++i) {
      q.push(adj[i][j]);
    }
    adj[1][j] = q.get();
    for (int i = 2, k = a + 1; k <= n; ++k, ++i) {
      q.pop();
      q.push(adj[k][j]);
      adj[i][j] = q.get();
    }
  }
  ll ans = 0;
  for (int i = 1; i <= n - a + 1; ++i) {
    q.clear();
    for (int j = 1; j <= b; ++j) {
      q.push(adj[i][j]);
    }
    ans += q.get();
    for (int j = 2, k = b + 1; k <= m; ++k, ++j) {
      q.pop();
      q.push(adj[i][k]);
      ans += q.get();
    }
  }
  printf("%lld\n", ans);
}
