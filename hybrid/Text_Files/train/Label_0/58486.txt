import java.text.ParseException;
import java.util.*;
import java.io.*;
import java.math.BigInteger;


import static java.lang.Math.*;

    public class solver {
    
 
    final boolean ONLINE_JUDGE = System.getProperty("ONLINE_JUDGE")!=null;
    
    BufferedReader in;
    static PrintWriter out;
    StringTokenizer tok = new StringTokenizer("");
    
    void init() throws FileNotFoundException{
        if (ONLINE_JUDGE){          
             in = new BufferedReader(new InputStreamReader(System.in));
             out = new PrintWriter(System.out);            
        }else{  
            in = new BufferedReader(new FileReader("input.txt"));
            out = new PrintWriter("output.txt");           
        }
    }
    
    String readString() throws IOException{
        while(!tok.hasMoreTokens()){
            try{
                tok = new StringTokenizer(in.readLine());
            }catch (Exception e){               
                return null;
            }
        }
        return tok.nextToken();
    }
    
    int readInt() throws IOException{
        return Integer.parseInt(readString());
    }
    
    long readLong() throws IOException{
        return Long.parseLong(readString());
    }
    
    double readDouble() throws IOException{
        return Double.parseDouble(readString());
    }
    
    public static void main(String[] args) throws IOException, ParseException{       
        new solver().run();
    }
    
    long timeBegin, timeEnd;

    void time(){
        timeEnd = System.currentTimeMillis();
        System.err.println("Time = " + (timeEnd - timeBegin));
    }
    int[] iMas(int n) throws IOException{
        int[] a=new int[n];
        for (int i=0;i<n;i++){
            a[i]=readInt();
        }
        return a;
    }
    void oMas(int[] a){
        for (int i=0;i<a.length;i++){
            out.println(a[i]);
        }
    }
    
    int gcd(int a,int b){
        if (b==0) return a;
        return gcd(b, a%b);
    }
    
    class Pair implements Comparable<Pair>{
        int a; 
        int b;
        
        Pair(int a,int b){
            this.a=a;
            this.b=b;
        }

        @Override
        public int compareTo(Pair o) {          
            return a-o.a;
        }
        
    }
    
    long result(int i,long[] prefix){
        int ik=i+k-1;
        long pref=prefix[ik];
        if (i!=0) pref-=prefix[i-1];
        return pref;
    }
    
    int min(int i,int j){
        long resulti=result(i,prefix);
        long resultj=result(j,prefix);
        
        resulti-=(long)(i*2+k)*(result(i,p));
        resultj-=(long)(j*2+k)*(result(j,p));  
        
        if (resulti<=resultj) return i;else return j;
    }
    
 
    int k;int n;
    long[] p,prefix;
    int[] l;
    
    void solve() throws IOException{    
        n=readInt();
        Pair[] a=new Pair[n];
        for (int i=0;i<n;i++){
            a[i]=new Pair(readInt(),i+1);
        }
        
        Arrays.sort(a);
        
        k=readInt();
        p=new long[n];
        prefix=new long[n];
        l=new int[n];       
        for (int i=0;i<n;i++){
            if (i!=0) {
                l[i]=l[i-1]+2;
                prefix[i]=prefix[i-1];
                p[i]=p[i-1];
            }else{
                l[i]=1;
            }
            p[i]+=a[i].a;
            prefix[i]+=(long)l[i]*a[i].a;
        }
        
        int ans=0;
        
        for (int i=1;i<=n-k;i++){
            ans=min(ans,i);
        }
        for (int i=ans;i<ans+k;i++){
            out.print(a[i].b+" ");
        }
    }
    
    public void run() throws IOException, ParseException{
        
        init(); 
        long h=System.currentTimeMillis();
        solve();        
        System.err.println(System.currentTimeMillis()-h);
        out.close();
    }
    
    
}
