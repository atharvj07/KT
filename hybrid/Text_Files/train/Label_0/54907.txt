import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        FastScanner in = new FastScanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        new Main().run(in, out);
        out.close();
    }

    public static long mod = 17352642619633L;

    void run(FastScanner in, PrintWriter out) {

        int N = in.nextInt();

        Map<Dep, List<Dep>> deps = new HashMap<>();

        Dep first = null;
        for (int i = 0; i < N; i++) {
            String s = in.next();
            int v = in.nextInt();
            Dep dep = new Dep(s, v);
            List<Dep> depDeps = new ArrayList<>();
            deps.put(dep, depDeps);

            int m = in.nextInt();
            for (int j = 0; j < m; j++) {
                String s2 = in.next();
                int v2 = in.nextInt();
                depDeps.add(new Dep(s2, v2));
            }
            if (i == 0) first = dep;
        }

        PriorityQueue<State> pq = new PriorityQueue<>(10, (a,b) -> {
            if (a.dist != b.dist) return a.dist-b.dist;
            int cmp = a.dep.s.compareTo(b.dep.s);
            if (cmp == 0) {
                // return one with max value
                return b.dep.v - a.dep.v;
            }
            return cmp;
        });

        TreeSet<Dep> ret = new TreeSet<>((a,b) -> a.s.compareTo(b.s));
        pq.offer(new State(first, 0));

        while (!pq.isEmpty()) {
            // System.out.println(Arrays.toString(pq.toArray()));

            State st = pq.poll();
            Dep d = st.dep;
            int dist = st.dist;

            if (ret.contains(d)) continue;
            ret.add(d);

            for (Dep dep : deps.get(d)) {
                if (!ret.contains(dep)) {
                    pq.offer(new State(dep, dist+1));
                }
            }
        }

        ret.remove(first);
        out.println(ret.size());
        for (Dep x : ret) {
            out.println(x.s + " " + x.v);
        }





    }

    class State {
        Dep dep;
        int dist;
        State(Dep dep, int dist) {
            this.dep = dep;
            this.dist = dist;
        }
        @Override
        public String toString() {
            return String.format("[%s %d %d]", dep.s, dep.v, dist);
        }
    }

    class Dep {
        String s;
        int v;
        Dep(String s, int v) {
            this.s = s;
            this.v = v;
        }
        @Override
        public String toString() {
            return String.format("[%s %d]", s, v);
        }

        @Override
        public boolean equals(Object o) {
            Dep rhs = (Dep)o;
            return v == rhs.v && s.equals(rhs.s);
        }

        @Override
        public int hashCode() {
            return Objects.hash(s, v);
        }
    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        public FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            st = null;
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }
}
