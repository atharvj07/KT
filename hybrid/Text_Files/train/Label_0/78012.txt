from sys import stdin, stdout

def main():
	t = int(stdin.readline())
	for _ in range(t):
		n = int(stdin.readline())
		(p, c) = ([0], [0])
		p.extend(list(map(int, stdin.readline().split())))
		c.extend(list(map(int, stdin.readline().split())))
		ans = 999999999
		already = [0] * (n + 1)
		for x in range(1, n + 1):
			if already[x] == 1:
				continue
			group = []
			y = x
			while already[y] == 0:
				already[y] = 1
				group.append(y)
				y = p[y]
			l = len(group)
			for step in range(1, l + 1):
				if ans <= step:
					break
				if l % step != 0:
					continue
				for z in range(0, step):
					is_ok = True
					pos = z
					while pos + step < l:
						if c[group[pos]] != c[group[pos + step]]:
							is_ok = False
							break
						pos += step
					if is_ok:
						ans = min(ans, step)
						break
		stdout.write(str(ans) + '\n')
main()
