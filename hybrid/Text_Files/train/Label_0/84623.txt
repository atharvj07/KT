from sys import stdin
import heapq

def solve(tc):
	n = int(stdin.readline())
	w = list(map(int, stdin.readline().split()))
	seq = stdin.readline()
	wi = []
	for i in range(n):
		wi.append((w[i], i + 1))
	heapq.heapify(wi)
	ans = []
	remained = []
	for c in seq:
		if c == '0':
			seat = heapq.heappop(wi)
			heapq.heappush(remained, (-seat[0], seat[1]))
			ans.append(str(seat[1]))
		if c == '1':
			seat = heapq.heappop(remained)
			ans.append(str(seat[1]))
	print(' '.join(ans))
