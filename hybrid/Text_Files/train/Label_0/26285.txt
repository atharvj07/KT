#include <bits/stdc++.h>
using namespace std;
int main(void) {
  int n;
  cin >> n;
  int a[n], b[n];
  int s1 = 0, s2 = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    s1 += a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
    s2 += b[i];
  }
  if (s2 > s1) {
    cout << "No";
    return 0;
  } else
    cout << "Yes";
  return 0;
  return 0;
}
