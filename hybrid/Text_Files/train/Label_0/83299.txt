def get_pos(x, y):
	if x >= y:
		return y - 1
	return y % x - 1

def solve(n, x):
	a1 = [1, 2, 2, 1, 0, 0]
	a2 = [0, 0, 1, 2, 2, 1]
	pos1 = get_pos(len(a1), n)
	if a1[pos1] == x:
		return 0
	pos2 = get_pos(len(a2), n)
	if a2[pos2] == x:
		return 1
	return 2

def main():
	n = int(input())
	x = int(input())
	print(solve(n, x))
main()
