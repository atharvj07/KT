import java.awt.geom.*;
import java.io.*;
import java.math.*;
import java.util.*;
import java.util.regex.*;
import static java.lang.Math.*;

public class Main {
	public Main() throws IOException {
		String input;

		input = br.readLine();
		work(input);

		System.out.print(sb);
	}

	private void work(String input) throws IOException {
		//debug(t, input);
		long ret = 0;
		String[] inputArray = input.split(" ");
		int n = Integer.valueOf(inputArray[0]);
		int l = Integer.valueOf(inputArray[1]);
		int t = Integer.valueOf(inputArray[2]);
		int[] arrays = new int [2 * n + 1];
		int left = 0;
		int right = 0;

		input = br.readLine();
		inputArray = input.split(" ");
		arrays[0] = -1;
		for (int i = 1; i <= n; i++) {
			arrays[i] = Integer.valueOf(inputArray[i - 1]);
			arrays[n + i] = Integer.valueOf(inputArray[i - 1]) + l;
		}

		ret += ((long)t * 2 / l) * (n - 1) * n * 2;
		int run = (t * 2) % l;
		for (int i = 1; i <= n; i++) {
			while (arrays[right] <= arrays[i] + run)
				right++;

			ret += right - i - 1;

			while (arrays[left] < l + arrays[i] - run)
				left++;

			ret += i + n - left;

		}
		
		sb.append((double)ret / 8);
	}

	StringBuilder sb = new StringBuilder();
	BufferedReader br = new BufferedReader(
		new InputStreamReader(System.in));

	public static void main(String[] args) throws IOException {
			new Main();
	}

	public static void debug(Object... arr) {
		System.err.println(Arrays.deepToString(arr));
	}
}
