#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, i;
  long long a[200005];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  long long j, k, ans = 0, Min = 1000000000;
  for (int i = 0; i < n; i++) {
    ans = 0;
    for (int j = 0; j < a[i]; j++) {
      cin >> k;
      ans += (k * 5);
    }
    ans += (a[i] * 15);
    if (ans < Min) Min = ans;
  }
  cout << Min << endl;
  return 0;
}
