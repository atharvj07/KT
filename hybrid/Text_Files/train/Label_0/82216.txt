qtd = int(input())
esquerdas = []
direitas = []
totalCadeiras1Circulo = 0
totalCadaUmCirculo = 0
for gfdsga in range(qtd):
	(l, r) = list(map(int, input().split()))
	esquerdas.append(l)
	direitas.append(r)
esquerdasOrdenadas = sorted(esquerdas)
direitasOrdenadas = sorted(direitas)
for x in range(qtd):
	totalCadaUmCirculo += 1 + max(esquerdas[x], direitas[x])
	totalCadeiras1Circulo += 1 + max(esquerdasOrdenadas[x], direitasOrdenadas[x])
print(min(totalCadeiras1Circulo, totalCadaUmCirculo))
