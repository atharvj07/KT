#include <bits/stdc++.h>
using namespace std;
int n, a[100010] = {0}, ans[100010][340] = {0};
int main() {
  int i, j, q, p, m, k;
  cin >> n;
  for (i = 1; i <= n; i++) scanf("%d", &a[i]);
  cin >> q;
  for (i = 1; i <= 320; i++)
    for (j = n; j >= 1; j--) {
      if (i + j + a[j] > n)
        ans[j][i] = 1;
      else
        ans[j][i] = ans[j + a[j] + i][i] + 1;
    }
  for (i = 1; i <= q; i++) {
    int ans1 = 0;
    scanf("%d%d", &p, &k);
    if (k <= 320)
      cout << ans[p][k] << endl;
    else {
      while (p <= n) {
        p += a[p] + k;
        ans1++;
      }
      cout << ans1 << endl;
    }
  }
  return 0;
}
