		int idx=s[i]-'a';
		num[idx]++;
	}
	sort(num,num+26);

	int ci=0,cj=0;
	while(num[ci]==0) ci++,cj++;

	int ans=0;
	bool init=true;

	while(ci<26){
		while(num[ci]==num[cj]) cj++;

		if(init) 
			init=false;
		else
			ans++;

		int num2=cj-ci;
		if(num[ci]>=2){
			ans+=2;
			if(num2>=2) ans+=2;
		}
		ans+=2*num2-1;

		ci=cj;
	}
	cout << ans << endl;
	return 0;
}
