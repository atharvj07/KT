from collections import defaultdict
for _ in ' ' * int(input()):
	(n, l, r, s) = map(int, input().split())
	(p, a, f) = ([], [0] * n, [0] * 100000)
	for i in range(1, n + 1):
		p.append(i)
	w = list(p[:r - l + 1])
	(b, h) = (sum(w), sum(p[-(r - l + 1):]))
	if not b <= s <= h:
		print(-1)
	else:
		c = s - b
		m = c % (r - l + 1)
		w[-1] = p[r - l] + c // (r - l + 1)
		for i in range(len(w) - 2, -1, -1):
			w[i] = w[i + 1] - 1
		for i in range(len(w) - 1, -1, -1):
			if not m:
				break
			m -= 1
			w[i] += 1
		for i in w:
			f[i] = 1
		for i in range(l, r + 1):
			a[i - 1] = w.pop()
		for i in range(n):
			if not a[i]:
				for j in range(len(p)):
					if f[p[j]] == 0:
						a[i] = p.pop(j)
						break
		print(*a)
