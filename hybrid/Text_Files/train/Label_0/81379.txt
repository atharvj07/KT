t = int(input())
Matrixnd = [[-1] * 2 for i in range(t)]
Matrix = [[-1] * 100 for i in range(t)]
for i in range(t):
	k = list(map(int, input().split()))
	Matrixnd[i] = k
	k = list(map(int, input().split()))
	Matrix[i] = k
for i in range(t):
	z = 0
	st = Matrix[i][0]
	n = Matrixnd[i][0]
	d = Matrixnd[i][1]
	for j in range(1, Matrixnd[i][0]):
		while Matrix[i][j] != 0:
			if d - j >= 0:
				d -= j
				st += 1
				Matrix[i][j] -= 1
			else:
				z = 1
				print(st)
				break
		if z == 1:
			break
	if z == 0:
		print(st)
