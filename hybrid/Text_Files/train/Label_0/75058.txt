from collections import Counter
import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	lis = list(map(int, input().strip()))
	c = [0] * (n + 1)
	for i in range(1, n + 1):
		c[i] = c[i - 1] + lis[i - 1]
	p = [c[i] - i for i in range(n + 1)]
	p = Counter(p)
	ans = 0
	for i in p.values():
		ans += i * (i - 1) // 2
	print(ans)
