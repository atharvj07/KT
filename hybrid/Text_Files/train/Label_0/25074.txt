#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long int n, k, skb = 0, mod, sum = 0, ans = 100000000;
  cin >> n;
  long long int arr[n + 1];
  for (long long int i = 1; i <= n; i++) cin >> arr[i];
  for (long long int i = 1; i <= n; i++) {
    skb = 0;
    for (long long int j = 1; j <= n; j++) {
      sum = 0;
      sum += abs(i - j);
      sum += abs(j - 1);
      sum += abs(1 - i);
      sum *= 2;
      sum *= arr[j];
      skb += sum;
    }
    ans = min(ans, skb);
  }
  cout << ans;
  return 0;
}
