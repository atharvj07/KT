s = input()
(a, b, c) = (0, 0, 0)
for i in s:
	if 47 < ord(i) < 58:
		a += 1
		break
for j in s:
	if 64 < ord(j) < 91:
		b += 1
		break
for k in s:
	if 96 < ord(k) < 123:
		c += 1
		break
if len(s) > 4:
	if a != 0 and b != 0 and (c != 0):
		print('Correct')
	else:
		print('Too weak')
else:
	print('Too weak')
