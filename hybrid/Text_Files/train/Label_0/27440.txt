#include <bits/stdc++.h>
using namespace std;
set<pair<int, int> > q, ou, od, inu, ind;
int n, m, t[100005], s[100005], f[100005], fl;
long long tim, res[100005];
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> t[i] >> s[i] >> f[i];
    q.insert(pair<int, int>(t[i], i));
  }
  fl = 1;
  while (!ou.empty() || !od.empty() || !inu.empty() || !ind.empty() ||
         !q.empty()) {
    int dr;
    long long nr = q.empty() ? LLONG_MAX : q.begin()->first;
    if (ou.empty() && od.empty() && inu.empty() && ind.empty())
      dr = 0;
    else {
      if (ou.size() + inu.size() >= od.size() + ind.size()) {
        dr = 1;
        if (!inu.empty()) nr = min(nr, tim + inu.begin()->first - fl);
        if (!ou.empty()) nr = min(nr, tim + ou.begin()->first - fl);
      } else {
        dr = -1;
        if (!ind.empty()) nr = min(nr, tim + fl - ind.rbegin()->first);
        if (!od.empty()) nr = min(nr, tim + fl - od.rbegin()->first);
      }
    }
    fl += dr * (nr - tim);
    tim = nr;
    while (!q.empty() && q.begin()->first == tim) {
      int index = q.begin()->second;
      q.erase(q.begin());
      if (s[index] == fl) {
        if (f[index] > fl)
          inu.insert(pair<int, int>(f[index], index));
        else
          ind.insert(pair<int, int>(f[index], index));
      } else if (s[index] > fl)
        ou.insert(pair<int, int>(s[index], index));
      else
        od.insert(pair<int, int>(s[index], index));
    }
    while (!ou.empty() && ou.begin()->first == fl) {
      int index = ou.begin()->second;
      ou.erase(ou.begin());
      if (f[index] > fl)
        inu.insert(pair<int, int>(f[index], index));
      else
        ind.insert(pair<int, int>(f[index], index));
    }
    set<pair<int, int> >::iterator it;
    while (!od.empty() && od.rbegin()->first == fl) {
      int index = od.rbegin()->second;
      it = od.end();
      od.erase(--it);
      if (f[index] > fl)
        inu.insert(pair<int, int>(f[index], index));
      else
        ind.insert(pair<int, int>(f[index], index));
    }
    while (!inu.empty() && inu.begin()->first == fl) {
      int index = inu.begin()->second;
      inu.erase(inu.begin());
      res[index] = tim;
    }
    while (!ind.empty() && ind.rbegin()->first == fl) {
      int index = ind.rbegin()->second;
      it = ind.end();
      ind.erase(--it);
      res[index] = tim;
    }
  }
  for (int i = 0; i < n; i++) {
    cout << res[i] << endl;
  }
  return 0;
}
