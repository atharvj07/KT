import math
import bisect
n = list(map(int, input().split(' ')))
m = list(map(int, input().split(' ')))
init_range = [0]
final_range = [n[1]]
count = 0
for i in range(len(m)):
	count += m[i]
	init_range.append(0 - count)
	final_range.append(n[1] - count)
x = max(init_range)
y = min(final_range)
ans = y - x + 1
if ans < 0:
	ans = 0
print(ans)
