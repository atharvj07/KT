#include <bits/stdc++.h>
using namespace std;
int head[105], cnt, a[105], b[105], deg[105];
struct node {
  int to, nex;
} edge[20005];
queue<int> sp;
void add(int u, int v) {
  edge[cnt].to = v;
  edge[cnt].nex = head[u];
  head[u] = cnt++;
}
pair<int, int> lo[305], yu[105], hui[105];
int check(int i, int j) {
  int ret = 0;
  if (yu[i] < yu[j]) {
    if (hui[i] > yu[j]) {
      ret++;
      if (hui[i] < hui[j]) ret++;
    } else {
      if (hui[i] > hui[j]) ret++;
    }
  } else {
    if (hui[i] < yu[j]) {
      ret++;
      if (hui[i] > hui[j]) ret++;
    } else {
      if (hui[i] < hui[j]) ret++;
    }
  }
  return ret;
}
int main() {
  int n, i, j;
  memset(head, -1, sizeof(head));
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d%d", a + i, b + i);
    lo[i] = make_pair(a[i], -i);
    ;
    yu[i] = make_pair(a[i], -i);
    ;
    hui[i] = make_pair(a[i] + b[i], -i);
    ;
  }
  for (i = 1; i <= n; i++) {
    for (j = i + 1; j <= n; j++) {
      int t1 = check(i, j), t2 = check(j, i);
      if (t1 != t2) {
        if (t1 > t2)
          add(i, j), deg[j]++;
        else
          add(j, i), deg[i]++;
      }
    }
    if (!deg[i]) sp.push(i);
  }
  int ans = 0;
  sort(lo + 1, lo + 1 + n);
  while (!sp.empty()) {
    int f = sp.front();
    sp.pop();
    int now;
    for (i = 1; i <= n; i++) {
      if (lo[i].second == -f) {
        now = i;
        break;
      }
    }
    lo[now].first += b[f];
    sort(lo + 1, lo + 1 + n);
    for (i = 1; i <= n; i++) {
      if (lo[i].second == -f) {
        ans += abs(now - i);
        break;
      }
    }
    for (int i = head[f]; ~i; i = edge[i].nex) {
      int v = edge[i].to;
      deg[v]--;
      if (!deg[v]) sp.push(v);
    }
  }
  printf("%d\n", ans);
  return 0;
}
