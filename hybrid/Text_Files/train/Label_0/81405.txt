t = int(input())
for i in range(t):
	n = int(input())
	w = [int(k) for k in input().split()]
	res = 0
	(a, b) = (0, 0)
	for j in w:
		if j % 2 == 0:
			a += 1
		else:
			b += 1
	if a == b + 1:
		(x, y) = (0, 0)
		while x < n:
			if x % 2 == 0:
				while w[y] % 2 == 1:
					y += 1
				res += abs(y - x)
				y += 1
			x += 1
		print(res)
	elif b == a + 1:
		(x, y) = (0, 0)
		while x < n:
			if x % 2 == 0:
				while w[y] % 2 == 0:
					y += 1
				res += abs(y - x)
				y += 1
			x += 1
		print(res)
	elif b == a:
		(res1, res2) = (0, 0)
		(x, y) = (0, 0)
		while x < n:
			if x % 2 == 0:
				while w[y] % 2 == 1:
					y += 1
				res1 += abs(y - x)
				y += 1
			x += 1
		(x, y) = (0, 0)
		while x < n:
			if x % 2 == 0:
				while w[y] % 2 == 0:
					y += 1
				res2 += abs(y - x)
				y += 1
			x += 1
		res = min(res1, res2)
		print(res)
	else:
		print(-1)
