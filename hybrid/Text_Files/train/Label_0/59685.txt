import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;

public class PrE {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    PrintWriter writer = new PrintWriter(System.out);

    long[] readLongs() throws IOException {
        String[] s = reader.readLine().split(" ");
        long[] longs = new long[s.length];
        for(int i = 0; i < longs.length; i++) {
            longs[i] = Long.parseLong(s[i]);
        }
        return longs;
    }

    int[] readInts() throws IOException {
        String[] s = reader.readLine().split(" ");
        int[] ints = new int[s.length];
        for(int i = 0; i < ints.length; i++) {
            ints[i] = Integer.parseInt(s[i]);
        }
        return ints;
    }

    int[] tt;
    int tx = 0;
    int readInt() throws IOException {
        if(tt == null || tx >= tt.length) {
            tt = readInts();
            tx = 0;
        }
        return tt[tx++];
    }

    void solve() throws IOException {
        char[] p = reader.readLine().toCharArray();
        char[] s = reader.readLine().toCharArray();
        int k = readInt();
        String ans = "2";
        int g = p.length / k;
        int rem = p.length % k;
        for(int n = 1; n <= k; n++) {
            if(n * g > s.length) {
                break;
            }
            int cur = n - 1;
            StringBuffer t = new StringBuffer();
            for(int i = k - 1; i >= 0; i--) {
                if(cur == -1) {
                    t = t.append('0');
                }
                else {
                    boolean ok = true;
                    for(int j = cur, r = i; j < s.length; j += n, r += k) {
                        if(r >= p.length || s[j] != p[r]) {
                            ok = false;
                            break;
                        }
                    }
                    if(ok) {
                        int last = (g - 1) * n + cur;
                        if(i < rem) last += n;
                        if(last < s.length) {
                            t = t.append('1');
                            cur--;
                        }
                        else {
                            t = t.append('0');
                        }
                    }
                    else {
                        t = t.append('0');
                    }
                }
            }
            t = t.reverse();
            String str = t.toString();
            if(cur == -1 && ans.compareTo(str) > 0) {
                ans = str;
            }
        }
        if(ans.equals("2")) ans = "0";
        writer.println(ans);
        writer.flush();
    }

    public static void main(String[] args) throws IOException {
        new PrE().solve();
    }
}
