n = int(input())
lis = list(map(int, input().split()))
if n == 1:
	print(0)
	exit()
tmp = [-1, 0, 1]
fin = 1000000000000
for i in tmp:
	for j in tmp:
		arr = lis[:]
		arr[0] += i
		arr[1] += j
		dif = arr[1] - arr[0]
		ans = 0
		if i != 0:
			ans = 1
		if j != 0:
			ans += 1
		for k in range(2, n):
			aa = abs(arr[k] - (arr[k - 1] + dif))
			if aa == 1:
				ans += 1
				arr[k] = arr[k - 1] + dif
			if aa > 1:
				ans = 1000000000000
		fin = min(ans, fin)
if fin == 1000000000000:
	print(-1)
else:
	print(fin)
