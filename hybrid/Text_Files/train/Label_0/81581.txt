def solve(n, arr_a):
	even = odd = 0
	for i in range(0, n):
		if arr_a[i] % 2 == 0:
			even += 1
		else:
			odd += 1
	if odd == 0 or (even == 0 and odd % 2 == 0):
		return 'NO'
	else:
		return 'YES'
t = int(input())
results = list()
for _ in range(0, t):
	n = int(input())
	arr_a = list(map(int, input().split(' ')))
	results.append(solve(n, arr_a))
for result in results:
	print(result)
