def inp():
	lst = input().split()
	return [int(el) for el in lst]

def f(lst):
	(n, a, b) = lst
	if n > a * b:
		res = '-1'
	else:
		sit = []
		for i in range(a):
			temp_list = [i_n + 1 if i_n < n else 0 for i_n in range(i * b, (i + 1) * b)]
			if not (i % 2 == 0 or b % 2 == 1):
				temp_list.reverse()
			sit.append(temp_list)
		res = '\n'.join((' '.join((str(h) for h in s)) for s in sit))
	return res
print(f(inp()))
