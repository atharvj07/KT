from sys import stdin, stdout
A = stdin.readline().strip().split()
n = int(A[0])
m = int(A[1])
d = int(A[2])
c = list(map(int, stdin.readline().strip().split()))
gap = n - sum(c)
jumps = m + 1
if jumps * (d - 1) >= gap >= 0:
	res = []
	interval = int(gap / jumps)
	remainder = gap % jumps
	wi = 0
	for wi in range(m):
		cur = interval
		if remainder > 0:
			cur += 1
			remainder -= 1
		for j in range(cur):
			res.append(0)
		for j in range(c[wi]):
			res.append(wi + 1)
	for j in range(interval):
		res.append(0)
	print('YES')
	print(' '.join(map(str, res)))
else:
	print('NO')
