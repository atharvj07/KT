#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC optimize("no-stack-protector")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx,tune=native")
#pragma GCC optimize("unroll-loops")
using namespace std;
void seriy() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
}
signed main() {
  seriy();
  string s, t;
  cin >> s >> t;
  long long second = s.length() - 1;
  while (s[second] == 'z') s[second] = 'a', second--;
  s[second]++;
  if (s == t)
    cout << "No such string";
  else
    cout << s;
  return 0;
}
