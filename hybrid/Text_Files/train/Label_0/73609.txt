class Item:
	count = [0, 0]

	def __init__(self):
		self.next = set()
		self.parity = -1

	def add(self, it):
		self.next.add(it)
		it.next.add(self)

	def hang(self):
		self.parity = 0
		queue = [self]
		while queue:
			cur = queue.pop()
			Item.count[cur.parity] += 1
			for it in cur.next:
				if it.parity < 0:
					it.parity = 1 - cur.parity
					queue.append(it)
n = int(input())
arr = [Item() for _ in range(n)]
for _ in range(n - 1):
	(a, b) = map(int, input().split())
	arr[a - 1].add(arr[b - 1])
arr[0].hang()
print(Item.count[0] * Item.count[1] - n + 1)
