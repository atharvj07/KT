import java.lang.*;
import java.util.*;
import java.io.*;

public class Main {
    static FastScanner in = new FastScanner();

    static long[][] mat;

    static long compute(int x, int y, int z) {
        long ans = mat[x][0] + mat[x][1], m1 = mat[y][0], m2 = mat[z][0], s1 = mat[y][1], s2 = mat[z][1];
        ans += Math.max(s2 - m1 + s1 - m2, Math.max(m2 + s2 - m1 - s1, m1 + s1 - m2 - s2));
        return ans;
    }

    static void solve() {
        mat = new long[3][2];
        int[] n = new int[3];
        for (int i = 0; i < 3; ++i)
            n[i] = in.nextInt();

        ArrayList<Integer>[] a = new ArrayList[3];
        for (int i = 0; i < 3; ++i) {
            a[i] = new ArrayList<>();
            for (int j = 0; j < n[i]; ++j) {
                int x = in.nextInt();
                a[i].add(x);
            }
            Collections.sort(a[i]);
        }
        for (int i = 0; i < 3; ++i) {
            mat[i][0] = a[i].get(0);
            for (int j = 1; j < n[i]; ++j)
                mat[i][1] += a[i].get(j);
        }

        long ans = Long.MIN_VALUE;
        ans = Math.max(ans, compute(0, 1, 2));
        ans = Math.max(ans, compute(1, 0, 2));
        ans = Math.max(ans, compute(2, 0, 1));

        System.out.println(ans);
    }

    public static void main(String[] args) {
        int t = 1;
        while (t-- > 0)
            solve();
    }
}

class FastScanner {
    BufferedReader br;
    StringTokenizer st;

    public FastScanner()  {
        br = new BufferedReader(new InputStreamReader(System.in));
    }

    String next() {
        while (st == null || !st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException  e) {
                e.printStackTrace();
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }

    double nextDouble() {
        return Double.parseDouble(next());
    }

    String nextLine() {
        String str = "";
        try {
            str = br.readLine();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return str;
    }
}
