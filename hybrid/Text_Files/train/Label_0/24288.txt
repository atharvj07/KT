#include <bits/stdc++.h>
using namespace std;
long long l[200005], t[200005];
struct node {
  long long st, num;
};
vector<node> ve;
int main() {
  long long n, r;
  cin >> n >> r;
  for (int i = 1; i <= n; i++) {
    cin >> l[i];
    l[i] *= 2;
  }
  for (int i = 1; i <= n; i++) {
    cin >> t[i];
  }
  long long res = 0, k = 0, num;
  long long time = 0;
  for (int i = 1; i <= n; i++) {
    if (l[i] < res) {
      res -= l[i];
      l[i] = 0;
    } else {
      l[i] -= res;
      t[i] -= res / 2;
      res = 0;
    }
    if (l[i] > 2 * t[i]) {
      cout << "-1" << endl;
      exit(0);
    }
    if (t[i] < l[i]) {
      num = (l[i] - t[i] + r - 1) / r;
      k += num;
      long long x = 2 * t[i] - l[i];
      ve.push_back(node{time + x, num});
      time += t[i];
      res = num * r - l[i] + t[i];
      res *= 2;
      time += res / 2;
    } else
      time += l[i];
  }
  cout << k << endl;
  if (k <= 1e5) {
    for (int i = 0; i < ve.size(); i++) {
      for (int j = 0; j < ve[i].num; j++) {
        cout << ve[i].st + j * r << " ";
      }
    }
  }
  cout << endl;
}
