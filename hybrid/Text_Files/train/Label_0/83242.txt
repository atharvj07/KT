from collections import Counter, defaultdict
n = int(input())
a = list(map(int, input().split()))
d = Counter(a)
MAX = -float('inf')
for i in d.keys():
	MAX = max(d[i], MAX)
M = float('inf')
ind = defaultdict(list)
for i in range(n):
	if d[a[i]] == MAX:
		ind[a[i]].append(i)
for i in ind.keys():
	if ind[i][-1] - ind[i][0] < M:
		M = ind[i][-1] - ind[i][0]
		l = ind[i][0]
		r = ind[i][-1]
print(l + 1, r + 1)
