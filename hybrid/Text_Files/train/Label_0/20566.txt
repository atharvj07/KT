#include <bits/stdc++.h>
using namespace std;
int n, a, b;
string s, ss;
long long dpw[2][500005 + 5];
bool Ok(int md, long long t) {
  int i, j, l, r;
  long long tm = (dpw[0][md - 1]) * b;
  tm += (md - 1) * a;
  tm += (md);
  if (tm <= t) return true;
  tm = (dpw[1][md - 1]) * b;
  tm += (md - 1) * a;
  tm += (md);
  if (tm <= t) return true;
  for (i = 1; i <= md - 1; i++) {
    l = i;
    r = (md - i);
    tm = (dpw[0][l - 1]) * b;
    tm += md;
    tm += (dpw[0][n - 1] - dpw[0][n - r - 1]) * b;
    tm += (r * a + (l - 1) * (2 * a));
    if (tm <= t) return true;
    tm = (dpw[1][l - 1]) * b;
    tm += md;
    tm += (dpw[1][n - 1] - dpw[1][n - r - 1]) * b;
    tm += (r * a + (l - 1) * (2 * a));
    if (tm <= t) return true;
  }
  return false;
}
char str[500005 + 5];
int main() {
  int i, j, k;
  long long t;
  scanf("%d%d%d", &n, &a, &b);
  scanf("%lld", &t);
  scanf("%s", str);
  s = str;
  string rr = s;
  reverse(rr.begin(), rr.end());
  ss = "";
  ss += s[0];
  ss += rr;
  ss[n] = '\0';
  for (i = 0; i < n; i++) {
    if (i) dpw[0][i] = dpw[0][i - 1];
    dpw[0][i] += (s[i] == 'w');
  }
  for (i = 0; i < n; i++) {
    if (i) dpw[1][i] = dpw[1][i - 1];
    dpw[1][i] += (ss[i] == 'w');
  }
  int lo = 1, hi = n, ret = 0;
  while (lo <= hi) {
    int md = (lo + hi) / 2;
    if (Ok(md, t)) {
      ret = max(ret, md);
      lo = md + 1;
    } else
      hi = md - 1;
  }
  printf("%d\n", ret);
  return 0;
}
