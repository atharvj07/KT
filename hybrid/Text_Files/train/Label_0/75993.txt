n = int(input())
(a, b, c, ab, bc, ac, abc) = (10 ** 10, 10 ** 10, 10 ** 10, 10 ** 10, 10 ** 10, 10 ** 10, 10 ** 10)
for _ in range(n):
	(val, word) = list(map(str, input().split()))
	val = int(val)
	if word == 'A':
		a = min(a, val)
	elif word == 'B':
		b = min(b, val)
	elif word == 'C':
		c = min(c, val)
	elif word in ['AB', 'BA']:
		ab = min(ab, val)
	elif word in ['BC', 'CB']:
		bc = min(bc, val)
	elif word in ['AC', 'CA']:
		ac = min(ac, val)
	else:
		abc = min(abc, val)
ans = 10 ** 10
ans = min(ans, a + b + c, ab + bc, bc + ac, a + abc, b + abc, c + abc, ab + abc, bc + abc, ac + abc, a + bc, b + ac, c + ab, ac + ab, abc)
if ans == 10 ** 10:
	print(-1)
else:
	print(ans)
