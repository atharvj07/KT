#include "bits/stdc++.h"
using namespace std;

//隣接リストの書き方
typedef long long ll;
typedef pair<int, int> P;

const int INF = 1000000007;
const ll INF2 = 1000000000000000;

vector<P> G[100010];
ll cost[100010];

void dfs(int V, ll c) {
	cost[V] = c;
	for(int i=0;i<G[V].size();i++){
		if(cost[G[V][i].first] == INF2){
			dfs(G[V][i].first, c + G[V][i].second);
		}
	}
}

int main(void){
	
	int n; cin>>n;
	for(int i=0;i<n-1;i++){
		int a,b,c;
		cin>>a>>b>>c;
		G[a].push_back({b, c}); //c++11以降でコンパイル可能 G[a].push_back(make_pair(b,c));
		G[b].push_back({a, c});
	}

	int q,k;
	cin>>q>>k;

	for(int i=0;i<100010;i++) cost[i] = INF2;

	dfs(k, 0);
	for(int i=0;i<q;i++) {
		int x,y;
		cin>>x>>y;
		cout<<cost[x]+cost[y]<<endl;
	}

	return 0;
}
