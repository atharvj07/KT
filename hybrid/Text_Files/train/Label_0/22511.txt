#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const int mod = 1e9 + 7;
const double eps = 1e-8;
vector<int> v;
pair<int, int> a[100005];
int main() {
  ios_base::sync_with_stdio(0);
  int n, st, first;
  cin >> n >> st >> first;
  for (int i = 1; i <= n; ++i) v.push_back(i);
  unsigned seed = chrono::system_clock::now().time_since_epoch().count();
  shuffle(v.begin(), v.end(), default_random_engine(seed));
  int upto = min((int)v.size(), 1000);
  for (int i = 0; i < upto; ++i) {
    cout << "? " << v[i] << "\n";
    fflush(stdout);
    cin >> a[i].first >> a[i].second;
  }
  cout << "? " << st << "\n";
  fflush(stdout);
  cin >> a[upto].first >> a[upto].second;
  ++upto;
  int cnt = upto;
  sort(a, a + upto);
  int idx = lower_bound(a, a + upto, make_pair(first, 0)) - a;
  if (idx == upto || a[idx].first > first) --idx;
  if (idx == -1) {
    cout << "! " << a[0].first << "\n";
    fflush(stdout);
  } else {
    int cur = a[idx].second, tmp = a[idx].first;
    while (cur != -1) {
      if (tmp >= first) break;
      cout << "? " << cur << "\n";
      fflush(stdout);
      cin >> tmp >> cur;
      ++cnt;
    }
    assert(cnt < 2000);
    if (cur == -1) {
      if (tmp >= first)
        cout << "! " << tmp << "\n";
      else
        cout << "! -1\n";
    } else
      cout << "! " << tmp;
    fflush(stdout);
  }
  return 0;
}
