#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main()
{
	string s;cin>>s;
	vector<int> lr;
	for(int i=1;i<s.size()+1;++i)
	{
		int l=1;
		while(s[i]==s[i-1])
		{
			l++;
			i++;
		}
		lr.push_back(l);
		//if(l!=1)i--;
	}
 
	for(int i=0;i<lr.size();i+=2)
	{
		for(int j=0;j<lr[i]-1;++j)cout<<0<<" ";
		int num=lr[i]/2+lr[i+1]/2;
		cout<<num+lr[i]%2<<" ";
		cout<<num+lr[i+1]%2<<" ";
		for(int j=0;j<lr[i+1]-1;++j)cout<<0<<" ";
	}
	cout<<endl;
/*
	for(int i=0;i<lr.size();++i)
	{
		cout<<lr[i]<<endl;
	}
	*/
	return 0;
}
