import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

/**
 * @author Don Li
 */
public class BigSecret {
    
    int n;
    long[] b;
    
    int[] cnt = new int[60];
    
    void solve() {
        n = in.nextInt();
        b = new long[n];
        for (int i = 0; i < n; i++) b[i] = in.nextLong();
        
        int[] sz = new int[60];
        for (int i = 0; i < n; i++) {
            int x = Long.numberOfTrailingZeros(Long.highestOneBit(b[i]));
            sz[x]++;
        }
        int[][] grp = new int[60][];
        for (int i = 0; i < 60; i++) grp[i] = new int[sz[i]];
        for (int i = 0; i < n; i++) {
            int x = Long.numberOfTrailingZeros(Long.highestOneBit(b[i]));
            grp[x][--sz[x]] = i;
        }
        for (int i = 0; i < 60; i++) sz[i] = grp[i].length;
        
        int hi = -1;
        for (int i = 0; i < 60; i++) {
            if (sz[i] > 0) hi = i;
        }
        if (sz[hi] > 1) {
            out.println("No");
            return;
        }
        
        int[] perm = new int[]{grp[hi][0]};
        addBits(grp[hi][0]);
    
        for (int i = hi - 1; i >= 0; i--) {
            if (sz[i] == 0) continue;
            int cap = cnt[i] + 1;
            if (sz[i] > cap) {
                out.println("No");
                return;
            }
            perm = merge(perm, grp[i], i);
            for (int x : grp[i]) addBits(x);
        }
        
        out.println("Yes");
        for (int i = 0; i < n; i++) {
            if (i > 0) out.print(' ');
            out.print(b[perm[i]]);
        }
        out.println();
    }
    
    int[] merge(int[] u, int[] v, int p) {
        int n = u.length, m = v.length;
        int[] w = new int[n + m];
        int i = 0, j = 0, k = 0;
        for (; j < m; ) {
            if (j == 0) {
                w[k++] = v[j++];
                continue;
            }
            while (i < n && (b[u[i]] & 1L << p) == 0) w[k++] = u[i++];
            if (i < n) w[k++] = u[i++];
            w[k++] = v[j++];
        }
        while (i < n) w[k++] = u[i++];
        return w;
    }
    
    void addBits(int x) {
        for (int i = 0; i < 60; i++) {
            if ((b[x] & 1L << i) > 0) {
                cnt[i]++;
            }
        }
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new BigSecret().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
