def isPrime(n):
	if n == 3 or n == 2:
		return True
	if n % 2 == 0 or n % 3 == 0:
		return False
	i = 5
	while i * i <= n:
		if n % i == 0 or n % (i + 2) == 0:
			return False
		i = i + 6
	return True
n = int(input())
if isPrime(n):
	print(1)
	print(n)
	exit()
print(3)
n -= 3
for i in range(2, n):
	if isPrime(i) and isPrime(n - i):
		print(3, i, n - i)
		exit()
