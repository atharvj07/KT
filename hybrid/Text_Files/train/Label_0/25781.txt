#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 10;
struct edge {
  int v, u, w, id;
  void read(int _id) {
    cin >> v >> u >> w;
    v--;
    u--;
    id = _id;
  }
};
int n, m, par[N];
edge e[N];
int root(int);
void merge(int, int);
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  for (int i = 0; i < m; i++) e[i].read(i);
  iota(par, par + n, 0);
  vector<int> uptEdge(m);
  iota(uptEdge.begin(), uptEdge.end(), 0);
  sort(uptEdge.begin(), uptEdge.end(),
       [](int i, int j) { return e[i].w < e[j].w; });
  auto pnt = uptEdge.begin();
  for (int i : uptEdge) {
    while (e[*pnt].w < e[i].w) {
      merge(e[*pnt].v, e[*pnt].u);
      pnt++;
    }
    e[i].v = root(e[i].v);
    e[i].u = root(e[i].u);
  }
  iota(par, par + m, 0);
  int q;
  cin >> q;
  vector<int> used;
  while (q--) {
    int cnt;
    cin >> cnt;
    bool flag = false;
    while (cnt--) {
      int eid;
      cin >> eid;
      edge f = e[eid - 1];
      used.push_back(f.v);
      used.push_back(f.u);
      flag |= (root(f.v) == root(f.u));
      merge(f.v, f.u);
    }
    for (int v : used) par[v] = v;
    used.clear();
    cout << (flag ? "NO\n" : "YES\n");
  }
}
int root(int v) { return par[v] == v ? v : (par[v] = root(par[v])); }
void merge(int v, int u) { par[root(v)] = root(u); }
