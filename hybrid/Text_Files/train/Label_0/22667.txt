#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx")
using namespace std;
const int DEFAULT_MOD = 1e9 + 7;
template <class T, T MOD = DEFAULT_MOD>
struct mod {
  mod() : value(0) {}
  mod(T x) { value = x; }
  T value;
};
template <class T, T MOD>
inline mod<T, MOD> operator+(mod<T, MOD> a, mod<T, MOD> b) {
  mod<T, MOD> result = {a.value + b.value};
  if (result.value > MOD) {
    result.value -= MOD;
  }
  return result;
};
template <class T, T MOD>
inline mod<T, MOD> operator-(mod<T, MOD> a, mod<T, MOD> b) {
  mod<T, MOD> result = {a.value - b.value};
  if (result.value < 0) {
    result.value += MOD;
  }
  return result;
};
template <class T, T MOD>
inline mod<T, MOD> operator*(mod<T, MOD> a, mod<T, MOD> b) {
  return mod<T, MOD>(a.value * 1ll * b.value % MOD);
};
template <class T, T MOD>
inline bool operator<(mod<T, MOD> a, mod<T, MOD> b) {
  return a.value < b.value;
};
template <class T, T MOD>
inline bool operator>(mod<T, MOD> a, mod<T, MOD> b) {
  return a.value > b.value;
};
template <class T, T MOD>
inline mod<T, MOD> power(mod<T, MOD> a, mod<T, MOD> p) {
  mod<T, MOD> result(1);
  T pp = p.value;
  for (mod<T, MOD> sq = a; pp > 0; pp >>= 1, sq = sq * sq) {
    if (pp & 1) {
      result = result * sq;
    }
  }
  return result;
}
template <class T, T MOD>
inline mod<T, MOD> operator/(mod<T, MOD> a, mod<T, MOD> b) {
  return a * power(b, mod<T, MOD>(MOD - 2));
};
template <class T, T MOD>
ostream &operator<<(ostream &os, const mod<T, MOD> &a) {
  os << a.value;
  return os;
};
template <class T, T MOD>
istream &operator>>(istream &is, mod<T, MOD> &a) {
  is >> a.value;
  return is;
};
struct TaskE {
  void solve(istream &in, ostream &out) {
    int m, n;
    in >> m >> n;
    vector<string> a(n);
    for (int i = 0; i < n; i++) {
      in >> a[i];
    }
    map<string, int> cnt;
    for (int j = 0; j < m; j++) {
      string s;
      for (int i = 0; i < n; i++) {
        s += a[i][j];
      }
      cnt[s]++;
    }
    vector<vector<mod<int>>> dp(m + 1, vector<mod<int>>(m + 1));
    dp[0][0] = 1;
    for (int i = 0; i <= m - 1; i++) {
      for (int j = 0; j < m; j++) {
        dp[i + 1][j] = dp[i + 1][j] + dp[i][j] * mod<int>(j);
        dp[i + 1][j + 1] = dp[i + 1][j + 1] + dp[i][j];
      }
    }
    vector<mod<int>> d(m + 1);
    for (int i = 0; i <= m; i++) {
      for (int j = 0; j <= m; j++) {
        d[i] = d[i] + dp[i][j];
      }
    }
    mod<int> result = 1;
    for (auto p : cnt) {
      int x = p.second;
      result = result * d[x];
    }
    out << result << endl;
  }
};
int main() {
  ios_base::sync_with_stdio(0);
  cout.precision(30);
  cout << fixed;
  TaskE solver;
  std::istream &in(std::cin);
  std::ostream &out(std::cout);
  solver.solve(in, out);
  return 0;
}
