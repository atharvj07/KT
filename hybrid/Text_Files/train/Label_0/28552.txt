#include <bits/stdc++.h>
using namespace std;
int n;
int m;
vector<long long int> v;
vector<pair<long long int, int> > ans;
long long int dp[752][752];
long long int cnt(long long int sum) {
  priority_queue<int> q;
  int cnt = 0;
  for (int i = 0; i < v.size(); i++) {
    sum += v[i];
    q.push(-v[i]);
    while (sum < 0) {
      sum += q.top();
      q.pop();
      cnt++;
    }
  }
  return cnt;
}
int main() {
  cin >> n >> m;
  long long int min_sum = 0;
  long long int max_sum = 0;
  long long int sum = 0;
  for (int i = 0; i < n; i++) {
    int a;
    scanf("%d", &a);
    v.push_back(a);
    sum += a;
    max_sum = max(max_sum, sum);
    min_sum = min(min_sum, sum);
  }
  long long int lef = -max_sum - 1LL;
  long long int rig = -min_sum + 10;
  lef = 0;
  for (int i = 0; i <= n; i++) {
    long long int mint = lef;
    long long int maxt = rig;
    while (mint + 1LL < maxt) {
      long long int mid = (mint + maxt) >> 1LL;
      int C = cnt(mid);
      if (i <= C) {
        mint = mid;
      } else {
        maxt = mid;
      }
    }
    if (cnt(mint) == i && cnt(maxt) < i) {
      ans.push_back(make_pair(mint, i));
    } else {
      if (cnt(maxt) == i) {
        ans.push_back(make_pair(maxt, i));
      }
    }
  }
  reverse(ans.begin(), ans.end());
  while (m--) {
    long long int b;
    scanf("%lld", &b);
    long long int cant = b;
    int id =
        upper_bound(ans.begin(), ans.end(), make_pair(cant, -1)) - ans.begin();
    if (id == ans.size()) {
      puts("0");
      continue;
    }
    if (id >= 0) {
      printf("%d\n", ans[id].second);
    } else {
      puts("0");
    }
  }
  return 0;
}
