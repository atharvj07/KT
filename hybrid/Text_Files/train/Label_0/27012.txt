#include <iostream>
using namespace std;
int main(){
	int n, k;
	cin >> n >> k;
	int r, s, p;
	cin >> r >> s >> p;
	char t[n];
	cin >> t;
	int count = 0;
	for(int i = 0; i < n; i++){
		if(i>=k){
			if(t[i] == t[i-k]){
				t[i] = 'x';
				continue;
			}
		}
		if(t[i] == 'r')count+=p;
		if(t[i] == 's')count+=r;
		if(t[i] == 'p')count+=s;
	}
	cout  << count << endl;
	return 0;
}
