import java.io.*;
import java.math.*;
import java.util.*;

public class Main {
  public static void main(String[] args) {
    InputStream inputStream = System.in;
    OutputStream outputStream = System.out;
    InputReader in = new InputReader(inputStream);
    PrintWriter out = new PrintWriter(outputStream);
    PP solver = new PP();
    solver.run(1, in, out);
    out.close();
  }

  static class PP {
    private final int ALPHA = 26;

    public TrieNode root;
    public static int numNodes;
    public boolean[] win;
    public boolean[] lose;

    public void run(int testNumber, InputReader in, PrintWriter out) {
      int n = in.nextInt();
      int k = in.nextInt();
      numNodes = 0;
      root = new TrieNode();
      for (int i = 0; i < n; i++) {
        insert(in.next());
      }
      win = new boolean[numNodes];
      lose = new boolean[numNodes];
      dfs(root);
      if (win[root.index] && lose[root.index]) {
        out.println("First");
      } else if (win[root.index]) {
        out.println(k % 2 == 1 ? "First" : "Second");
      } else {
        out.println("Second");
      }
    }

    public void dfs(TrieNode node) {
      for (int c = 0; c < ALPHA; c++) {
        TrieNode to = node.go[c];
        if (to != null) {
          dfs(to);
          win[node.index] |= !win[to.index];
          lose[node.index] |= !lose[to.index]; 
        }
      }
      if (node.isLeaf) {
        lose[node.index] = true;
      }
    }

    public void insert(String s) {
      TrieNode cur = root;
      for (char c : s.toCharArray()) {
        if (cur.go[c - 'a'] == null) {
          cur.go[c - 'a'] = new TrieNode();
        }
        cur = cur.go[c - 'a'];
      }
      cur.isLeaf = true;
    }

    public class TrieNode {
      public TrieNode[] go;
      public boolean isLeaf;
      public int index;

      TrieNode() {
        this.go = new TrieNode[ALPHA];
        this.isLeaf = false;
        this.index = numNodes++;
      }
    }
  }

  static class InputReader {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReader(InputStream stream) {
      reader = new BufferedReader(new InputStreamReader(stream), 32768);
      tokenizer = null;
    }

    public String next() {
      while (tokenizer == null || !tokenizer.hasMoreTokens()) {
        try {
          tokenizer = new StringTokenizer(reader.readLine());
        } catch (IOException e) {
          throw new RuntimeException(e);
        }
      }
      return tokenizer.nextToken();
    }

    public int nextInt() {
      return Integer.parseInt(next());
    }

    public long nextLong() {
      return Long.parseLong(next());
    }
  }
}
