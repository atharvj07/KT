from sys import stdin, gettrace
if gettrace():

	def inputi():
		return input()
else:

	def input():
		return next(stdin)[:-1]

	def inputi():
		return stdin.buffer.readline()

def solve():
	(n, k) = map(int, input().split())
	hh = [int(a) for a in input().split()]
	mx = hh[0]
	for h in hh[1:]:
		if h > mx + k - 1:
			print('No')
			return
		else:
			mx = min(mx + k - 1, h + k - 1)
	mx = hh[-1]
	for h in hh[-2::-1]:
		if h > mx + k - 1:
			print('No')
			return
		else:
			mx = min(mx + k - 1, h + k - 1)
	print('Yes')

def main():
	t = int(input())
	for _ in range(t):
		solve()
main()
