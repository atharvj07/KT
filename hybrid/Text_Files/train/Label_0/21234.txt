#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e2 + 11;
int n, a[maxn], b[maxn], c[maxn];
int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  sort(a + 1, a + n + 1);
  int ans = 1;
  for (int i = 1; i <= n; ++i) {
    if (b[i] == 0) {
      b[i] = ans;
      ans++;
      c[b[i]] = 1;
    }
    for (int j = i + 1; j <= n; ++j) {
      if (a[j] >= c[b[i]]) {
        a[j] -= c[b[i]];
        b[j] = b[i];
        c[b[i]]++;
        break;
      }
    }
  }
  int cnt = 1;
  sort(b + 1, b + n + 1);
  for (int i = 2; i <= n; ++i) {
    if (b[i] != b[i - 1]) cnt++;
  }
  ans = cnt;
  cout << endl << ans;
  return 0;
}
