#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using ii = pair<int, int>;
using vi = vector<int>;
#define all(v) begin(v), end(v)
#define sz(v) (int)(v).size()
#define fi first
#define se second

void solve() {
  int n; cin >> n;
  string s[3];
  cin >> s[0] >> s[1] >> s[2];
  for(int x : {0, 1, 2}) for(int y : {0, 1, 2}) if(x != y) {
    string& a = s[x];
    string& b = s[y];
    int ao = count(all(a), '1');
    int az = count(all(a), '0');
    int bo = count(all(b), '1');
    int bz = count(all(b), '0');
    if(min(ao, bo) >= n) {
      int i = 0, j = 0;
      int cnt = min(ao, bo);
      string ans;
      while(cnt--) {
        while(i < sz(a) and a[i] != '1') {
          ans += a[i++];
        }
        while(j < sz(b) and b[j] != '1') {
          ans += b[j++];
        }
        ans += '1';
        i++, j++;
      }
      ans += string(a.begin()+i, a.end());
      ans += string(b.begin()+j, b.end());
      cout << ans << "\n";
      return;
    } else if(min(az, bz) >= n) {
      int i = 0, j = 0;
      int cnt = min(az, bz);
      string ans;
      while(cnt--) {
        while(i < sz(a) and a[i] != '0') {
          ans += a[i++];
        }
        while(j < sz(b) and b[j] != '0') {
          ans += b[j++];
        }
        ans += '0';
        i++, j++;
      }
      ans += string(a.begin()+i, a.end());
      ans += string(b.begin()+j, b.end());
      cout << ans << "\n";
      return;
    }
  }
}

int main(int argc, char const *argv[])
{
  cin.sync_with_stdio(0); cin.tie(0);
  cin.exceptions(cin.failbit);
#ifdef LOCAL
  freopen("in", "r", stdin);
#endif
  int tc; cin >> tc;
  while(tc--) {
    solve();
  }
  return 0;
}
