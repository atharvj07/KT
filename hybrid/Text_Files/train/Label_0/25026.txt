#include <bits/stdc++.h>
long long gcd(long long a, long long b) {
  if (!b)
    return a;
  else
    return gcd(b, a % b);
}
int main() {
  int t, f;
  long long p, q, k, x, y, z;
  scanf("%d", &t);
  while (t--) {
    scanf("%I64d%I64d%I64d", &p, &q, &k);
    z = gcd(p, q);
    q = q / z;
    f = 0;
    while (q != 1) {
      x = gcd(q, k);
      if (x == 1) {
        if (q == 1) f = 1;
        break;
      }
      while (q % x == 0) {
        q = q / x;
      }
    }
    if (k % q == 0)
      printf("Finite\n");
    else
      printf("Infinite\n");
  }
  return 0;
}
