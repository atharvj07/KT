			else:
				diff.append(a[i] - a[i - 1])
	if k == 0:
		print(len(diff) + 1)
	else:
		diff.sort()
		l = len(diff) + 1
		p = l - 1
		ans = 0
		for i in range(p):
			if diff[i] % x == 0:
				if k >= diff[i] // x - 1:
					k -= diff[i] // x - 1
					l -= 1
				else:
					break
			elif k >= diff[i] // x:
				k -= diff[i] // x
				l -= 1
			else:
				break
