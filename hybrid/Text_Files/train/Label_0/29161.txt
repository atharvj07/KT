#include <bits/stdc++.h>
using namespace std;
long long mark[100005], mx = 0, flag = 0;
long long mrk[1000005];
long long arr[100005];
int main() {
  long long n, x, y, sum = 0, cnt = 0, tot = 0, mn, i;
  string s;
  cin >> n >> x >> y;
  cin >> s;
  for (i = 0; i < n - 1; i++) {
    if (s[i] != s[i + 1] && s[i] == '0') {
      cnt++;
    }
    if (s[i] == '0') {
      flag = 1;
    }
  }
  if (s[i] == '0') {
    flag = 1;
    cnt++;
  }
  if (!flag) {
    cout << 0 << endl;
    return 0;
  }
  sum = ((cnt - 1) * x) + y;
  tot += (cnt * y);
  mn = min(tot, sum);
  cout << mn << endl;
}
