n = map(int, input().split())
a = input()
(prev, cur) = ('0', '0')
t = 1
flag = 0
x = 1
if len(a) > 1:
	for i in range(len(a)):
		cur = a[i]
		if i > 0:
			if prev == cur:
				x = x + 1
			else:
				x = 1
				t += 1
			if x >= 2:
				flag += 1
		prev = cur
if flag >= 2:
	t += 2
elif flag > 0:
	t += 1
print(int(t))
