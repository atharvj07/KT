import sys, os, io
import math, bisect, operator
(inf, mod) = (float('inf'), 10 ** 9 + 7)
from itertools import groupby, accumulate
from heapq import heapify, heappop, heappush
from collections import deque, Counter, defaultdict
input = iter(sys.stdin.buffer.read().decode().splitlines()).__next__
Neo = lambda : list(map(int, input().split()))
(test,) = Neo()
for _ in range(test):
	(a, b) = Neo()
	s = list(input())
	n = len(s)
	a -= s.count('0')
	b -= s.count('1')
	if a < 0 or b < 0:
		print(-1)
		continue
	for i in range(n):
		if s[i] == '?':
			s[i] = s[n - i - 1]
			if s[i] == '0':
				a -= 1
			elif s[i] == '1':
				b -= 1
			else:
				s[i] = '*'
	if a < 0 or b < 0:
		print(-1)
		continue
	for i in range(n):
		if i == n // 2 and n % 2:
			continue
		if s[i] == '*':
			if a >= 2:
				s[i] = '0'
				s[n - i - 1] = '0'
				a -= 2
			else:
				s[i] = '1'
				s[n - i - 1] = '1'
				b -= 2
	if s[n // 2] == '*':
		if a > 0:
			s[n // 2] = '0'
			a -= 1
		else:
			s[n // 2] = '1'
			b -= 1
	if a != 0 or b != 0:
		print(-1)
	elif s != s[::-1]:
		print(-1)
	else:
		print(''.join(s))
