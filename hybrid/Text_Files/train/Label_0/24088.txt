#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long n, d, l;
  cin >> n >> d >> l;
  vector<long long> v(n);
  long long sum1 = 0, sum = 0;
  ;
  for (long long i = 1; i < n; i += 2) {
    v[i] = 1;
    sum1 += 1;
  }
  for (long long i = 0; i < n; i += 2) {
    v[i] = 1;
    sum += 1;
  }
  int i = 0;
  while (sum < sum1 + d) {
    if (v[i] + 1 > l) i += 2;
    v[i]++;
    i += 2;
    if (i > n - 1) {
      i = 0;
    }
    sum++;
  }
  if (sum1 + d > (n + 1) / 2 * l) {
    cout << "-1";
    return 0;
  }
  int ans = 0;
  for (long long j = n - 1; j >= 0; j--) {
    ans += pow(-1, j + 2) * v[j];
  }
  if (ans == d)
    for (long long i = 0; i < v.size(); i++) cout << v[i] << " ";
  else {
    ans = 0;
    sum1 = 0, sum = 0;
    for (long long i = 1; i < n; i += 2) {
      v[i] = l;
      sum1 += l;
    }
    for (long long i = 0; i < n; i += 2) {
      v[i] = 1;
      sum += 1;
    }
    i = 0;
    while (sum < sum1 + d) {
      if (v[i] + 1 > l) i += 2;
      v[i]++;
      i += 2;
      if (i > n - 1) {
        i = 0;
      }
      sum++;
    }
    if (sum1 + d < (n + 1) / 2) {
      cout << "-1";
      return 0;
    }
    for (long long j = n - 1; j >= 0; j--) {
      ans += pow(-1, j + 2) * v[j];
    }
    for (long long i = 0; i < v.size(); i++) cout << v[i] << " ";
  }
  return 0;
}
