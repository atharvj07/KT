#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long t, n, m, i, s, j, k, c;
  {
    s = 0;
    cin >> n >> m;
    vector<long long> vec[n];
    for (i = 0; i < m; i++) {
      long long x, y;
      cin >> x >> y;
      vec[x - 1].push_back(y - 1);
    }
    for (i = 0; i < n; i++) sort(vec[i].begin(), vec[i].end());
    for (i = 0; i < n; i++) {
      c = m;
      vector<long long> v[n];
      for (j = 0; j < n; j++) v[j] = vec[j];
      j = i;
      long long ans = 0;
      map<long long, long long> mp;
      while (c > 0) {
        ans++;
        long long x = -1;
        if (v[j % n].size() != 0) {
          if (v[j % n][v[j % n].size() - 1] < j % n) {
            x = v[j % n][v[j % n].size() - 1];
            v[j % n].erase(v[j % n].begin() + (v[j % n].size() - 1));
          } else {
            for (k = 0; k < v[j % n].size(); k++) {
              if (v[j % n][k] < (j % n)) {
                x = v[j % n][k];
              } else if (x != -1) {
                v[j % n].erase(v[j % n].begin() + k - 1);
                break;
              } else if (x == -1) {
                x = v[j % n][v[j % n].size() - 1];
                v[j % n].erase(v[j % n].begin() + (v[j % n].size() - 1));
                break;
              }
            }
          }
        }
        if (x != -1) mp[x]++;
        if (mp[j % n]) {
          c = c - mp[j % n];
          mp[j % n] = 0;
        }
        j++;
      }
      cout << ans - 1 << " ";
    }
  }
  return 0;
}
long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
long long power(long long x, long long y, long long m) {
  if (y == 0) return 1;
  long long p = power(x, y / 2, m) % m;
  p = (p * p) % m;
  return (y % 2 == 0) ? p : (x * p) % m;
}
long long modInverse(long long a, long long m) { return power(a, m - 2, m); }
long long binpow(long long a, long long b) {
  if (b == 0) return 1;
  long long res = binpow(a, b / 2);
  if (b % 2)
    return (((res % 1000000007) * (res % 1000000007) % 1000000007) *
            (a % 1000000007)) %
           1000000007;
  else
    return ((res % 1000000007) * (res % 1000000007)) % 1000000007;
}
