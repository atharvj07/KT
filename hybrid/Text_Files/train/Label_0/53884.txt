import java.io.FileInputStream;
import java.io.FileNotFoundException;
import java.util.ArrayList;
import java.util.Scanner;


public class Algorithm {

	public static final boolean DEBUG = true;

	public static int T;
	
	public static int N, M;
	public static String encodeString;
	public static String originalWordList[];
	public static String wordList[];
	
	public static boolean FOUND = false;
	
	public static ArrayList<String> ANS;
	
	public static void main(String[] args) {
		
		Scanner sc = new Scanner(System.in);
		/**/
		try {
			sc = new Scanner(new FileInputStream("input.txt"));
		} catch (FileNotFoundException e) {
			e.printStackTrace();
		}
		/**/
		T = 1;//sc.nextInt();
		
		for(int t = 1; t <= T; t++) {
			FOUND = false;
			//sc.nextLine();
			N = sc.nextInt();
			encodeString = sc.next();

			M = sc.nextInt();
			originalWordList = new String[M];
			wordList = new String[M];
			ANS = new ArrayList<String>();
			for(int i = 0; i < M; i++) {
				String word = sc.next();
				originalWordList[i] = word;
				
				wordList[i] = new StringBuilder(word.toLowerCase()).reverse().toString();
			}
			
			solve(encodeString);
			//print("Case #"+t);
			printAnswer();

		}
		sc.close();

	}
	
	private static void printAnswer() {
		StringBuilder builder = new StringBuilder();
		for(int i=ANS.size()-1; i>=0; i--){
			builder.append(ANS.get(i)).append(" ");
		}
		print( builder.toString().trim());
	}

	
	private static void solve(String encodeString) {
		if(encodeString.length() == 0) {
			FOUND = true;
			return;
		}
		
		for(int i = 0; i < M; i++) {
			if(encodeString.startsWith(wordList[i])) {
				solve(encodeString.substring(wordList[i].length()));
				
				if(FOUND) {
					ANS.add(originalWordList[i]);
					return;
				}
			}
		}
		
	}


	public static void log(String msg) {
		if(DEBUG)
			System.out.println(msg);
	}
	
	public static void print(String msg) {
		System.out.println(msg);
	}
	
	
}
