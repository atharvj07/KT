#include <iostream>
#include <algorithm>
#include <cstdio>
using namespace std;

int m,n;
double dp[17][(1<<16) + 1];
double prob[17][17];

double calc(int f, int s) {

    if (f == n)   return 1.0;
    if (dp[f][s] != -1.0) return dp[f][s];

    double ret = 0;
    for (int k=0; k<m; ++k) if (!(s & (1<<k))) { // ???????????????????????????
        double cp = 1.0;
        double sum = 0;
        for (int j=f; j<n && cp > 0.0; ++j) { // ?????????????????¢
            if (prob[k][j] < 1.0) {
                sum += calc(j, s|(1<<k)) * cp * (1.0 - prob[k][j]);
            }
            cp *= prob[k][j];
        }
        // ???????????§???????¶????????¢????(????????§?¢????0????????£?????????????????§??§??????
        sum += cp;
        ret = max(ret, sum);
    }
    //printf("f:%d\ts:%d\tp:%f\tret:%f\n",f,s,p,ret);
    return dp[f][s] = ret;
}

int main() {
    cin >> m >> n;
    for (int i=0; i<m; ++i) {
        for (int j=0; j<n; ++j) {
            cin >> prob[i][j];
        }
    }
    // init
    for (int i=0; i<17; ++i) for (int j=0; j<(1<<16)+1; ++j) dp[i][j] = -1.0;

    printf("%.12f\n", calc(0, 0));

    return 0;
}
