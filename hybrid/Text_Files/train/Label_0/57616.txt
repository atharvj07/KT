import java.io.*;
import java.util.StringTokenizer;
public class CF0499A
{
 public static void main(String[] helloWorld) throws IOException
 {
  BufferedReader sIn = new BufferedReader(new InputStreamReader(System.in));
  StringTokenizer sT = new StringTokenizer(sIn.readLine());
  int n = Integer.parseInt(sT.nextToken());
  int x = Integer.parseInt(sT.nextToken());
  int c = 1;
  int v = 0;
  for (; n > 0; n --)
  {
   sT = new StringTokenizer(sIn.readLine());
   int li = Integer.parseInt(sT.nextToken());
   int ri = Integer.parseInt(sT.nextToken());
   v += (li - c) % x;
   v += (ri - li + 1);
   c = ri + 1;
  }
  System.out.println(v);
 }
}
