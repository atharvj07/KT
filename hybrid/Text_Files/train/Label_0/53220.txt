import java.util.*;
import java.io.*;

public class Solver {
	public static void main(String args[]){
		PrintWriter out = new PrintWriter(System.out);
		out.println(2000);
		for(int i=1; i<=1000; i++){
			out.println(i + " " + 1 + " " + i + " " + 2);
		}
		for(int i=1000; i>=1; i--){
			out.println(i + " " + 1 + " " + i + " " + 2);
		}
		out.close();
	}
}
