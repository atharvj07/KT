from math import factorial
(n, k) = [int(x) for x in input().split(' ')]
(answer, s) = (0, 0)
(min_l, prev_l) = (9999999999999, -10)
s_left = 0
for i in range(n):
	(l, r) = [int(x) for x in input().split(' ')]
	if l < min_l:
		min_l = l
	s += r - l + 1
	if prev_l != -10:
		s_left += l - prev_l - 1
	prev_l = r
s_left += min_l - 1
if s % k == 0:
	print(0)
else:
	(s1, s2, c1) = (s, s, 0)
	while s1 % k != 0 and s2 % k != 0:
		s1 += 1
		if s_left > 0:
			s2 -= 1
			s_left -= 1
		c1 += 1
	print(c1)
