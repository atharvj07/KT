import math
input1 = '6 2'
input2 = '5 5 6 8 3 12'

def distance(x1, y1, x2, y2):
	return math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)

def y_stop(x1, y1, x2, r):
	return y1 + math.sqrt((r * 2) ** 2 - (x1 - x2) ** 2)

def inRangeTuples(x, tuples):
	inRange = []
	for t in tuples:
		if abs(x - t[0]) <= 2 * r:
			inRange.append(t)
	return inRange
(n, r) = map(int, input().split())
x = [int(y) for y in input().split()]
result_y = [r]
result_xy = [(x[0], r)]
for xi in x[1:]:
	closest_tuple = None
	for t in inRangeTuples(xi, result_xy):
		if not closest_tuple:
			closest_tuple = t
		elif y_stop(closest_tuple[0], closest_tuple[1], xi, r) < y_stop(t[0], t[1], xi, r):
			closest_tuple = t
	if closest_tuple:
		newy = y_stop(closest_tuple[0], closest_tuple[1], xi, r)
		result_xy.append((xi, newy))
		result_y.append(newy)
	else:
		result_xy.append((xi, r))
		result_y.append(r)
print(*result_y, sep=' ')
