for t in range(int(input())):
	(a, b, p) = map(int, input().split(' '))
	s = input()
	if s[-2] == 'B' and b > p or (s[-2] == 'A' and a > p):
		print(len(s))
	else:
		cur = s[-2]
		if s[-2] == 'B':
			cost = b
		if s[-2] == 'A':
			cost = a
		store = 0
		for i in range(len(s) - 3, -1, -1):
			if cur != s[i]:
				if s[i] == 'B':
					cur = 'B'
					cost += b
				if s[i] == 'A':
					cur = 'A'
					cost += a
			if cost > p:
				store = i + 2
				break
		if store == 0:
			print(1)
		else:
			print(store)
