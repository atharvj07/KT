#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  int x = 0;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    x ^= v[i];
  }
  if (n % 2 == 0) {
    if (x != 0)
      cout << "NO";
    else {
      cout << "YES" << endl;
      cout << (2 * ((n - 1) / 2)) << endl;
      for (int i = 0; i < n - 2; i += 2)
        cout << i + 1 << " " << i + 2 << " " << i + 3 << endl;
      for (int i = n - 2; i > 0; i -= 2)
        cout << i + 1 << " " << i << " " << i - 1 << endl;
    }
  } else {
    cout << "YES" << endl;
    cout << (2 * (n / 2)) << endl;
    for (int i = 0; i < n - 1; i += 2)
      cout << i + 1 << " " << i + 2 << " " << i + 3 << endl;
    for (int i = n - 1; i > 0; i -= 2)
      cout << i + 1 << " " << i << " " << i - 1 << endl;
  }
  cout << endl;
  return 0;
}
