#include <bits/stdc++.h>
using namespace std;
int n, cnt, k, s, e;
vector<pair<int, int> > v;
vector<int> a;
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 0; i < n; i++) {
    scanf("%d%d", &s, &e);
    v.push_back({s, 0});
    v.push_back({e, 1});
  }
  sort(v.begin(), v.end());
  for (int i = 0; i < v.size(); i++) {
    if (v[i].second) {
      if (cnt == k) a.push_back(v[i].first);
      cnt--;
    } else {
      cnt++;
      if (cnt == k) a.push_back(v[i].first);
    }
  }
  printf("%d\n", a.size() / 2);
  for (int x = 0; x < a.size() / 2; x++)
    printf("%d %d\n", a[2 * x], a[2 * x + 1]);
  return 0;
}
