#include <bits/stdc++.h>
using namespace std;
const int LIM = 5e5 + 1;
const int BLOCK = 600;
const int BLOCKS_COUNT = LIM / BLOCK + 1;
int get_int() {
  char c = getchar();
  while (isspace(c)) c = getchar();
  int ret = 0;
  while (!isspace(c)) {
    ret *= 10;
    ret += c - '0';
    c = getchar();
  }
  return ret;
}
int n, q, ans[LIM], ans_idx = 0, idx[LIM];
long long a[LIM], block[BLOCKS_COUNT];
int main() {
  n = get_int();
  q = get_int();
  for (int i = 0; i < n; ++i) a[i] = get_int();
  for (int i = 0; i < n; ++i) idx[i] = i;
  for (int begin = 0; begin < n; begin += BLOCK) {
    int end = begin + BLOCK;
    end = min(end, n);
    sort(idx + begin, idx + end, [](const int& i, const int& j) {
      return a[i] < a[j] || (a[i] == a[j] && i < j);
    });
  }
  for (int qi = 0; qi < q; ++qi) {
    int t;
    scanf("%d", &t);
    switch (t) {
      case 1: {
        int l = get_int(), r = get_int(), x = get_int();
        --l;
        int begin = 0, end = BLOCK;
        for (int b = 0; begin < n; ++b) {
          if (l <= begin && r >= end) {
            block[b] += x;
          } else if (!(r <= begin || l >= end)) {
            for (int j = max(l, begin); j < min(r, end); ++j) a[j] += x;
            sort(idx + begin, idx + min(end, n),
                 [](const int& i, const int& j) {
                   return a[i] < a[j] || (a[i] == a[j] && i < j);
                 });
          }
          begin += BLOCK;
          end += BLOCK;
        }
        break;
      }
      case 2: {
        int y = get_int();
        int begin = 0, end = BLOCK;
        int i = n, j = -1;
        for (int b = 0; begin < n; ++b) {
          long long y1 = y;
          y1 -= block[b];
          int p1 = lower_bound(idx + begin, idx + min(end, n), y1,
                               [](const int& i, const long long& y1) {
                                 return a[i] < y1;
                               }) -
                   idx;
          if (p1 != min(end, n) && a[idx[p1]] == y1) {
            int p2 = upper_bound(idx + begin, idx + min(end, n), y1,
                                 [](const long long& y1, const int& i) {
                                   return y1 < a[i];
                                 }) -
                     idx;
            --p2;
            i = min(i, idx[p1]);
            j = max(j, idx[p2]);
          }
          begin += BLOCK;
          end += BLOCK;
        }
        if (j == -1) {
          ans[ans_idx++] = -1;
        } else {
          ans[ans_idx++] = j - i;
        }
        break;
      }
    }
  }
  for (int i = 0; i < ans_idx; ++i) printf("%d\n", ans[i]);
  return 0;
}
