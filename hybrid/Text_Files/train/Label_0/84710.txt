t = int(input())
for i in range(0, t):
	n = int(input())
	a = list(map(int, input().split()))
	zero = []
	one = []
	two = []
	count = 0
	for item in a:
		if item % 3 == 0:
			zero.append(item)
		elif item % 3 == 1:
			one.append(item)
		elif item % 3 == 2:
			two.append(item)
	if len(one) > len(two):
		count = count + len(two) + len(zero) + (len(one) - len(two)) // 3
	elif len(two) > len(one):
		count = count + len(one) + len(zero) + (len(two) - len(one)) // 3
	else:
		count = count + len(zero) + len(one)
	print(count)
