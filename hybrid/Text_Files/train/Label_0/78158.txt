import string
titles = ''
letters = [i for i in string.ascii_lowercase]
num_inputs = int(input())
for i in range(num_inputs):
	word = input()
	titles += word + ' '
solved = False
for i in letters:
	if i not in titles:
		print(i)
		solved = True
		break
if not solved:
	for j in letters:
		if solved:
			break
		for i in letters:
			s = j + i
			if s not in titles:
				print(s)
				solved = True
				break
