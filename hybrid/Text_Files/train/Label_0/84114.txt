I = input
(n, s) = (int(I()), I())
(J, K) = (set(), set())
for i in range(n):
	if s[i] == '*':
		J.add(i)
	else:
		K.add(i)
(L, S) = (set('abcdefghijklmnopqrstuvwxyz'), set(s))
for _ in range(int(input())):
	w = I()
	if all((s[k] == w[k] for k in K)):
		W = {w[i] for i in J}
		if not S & W:
			L &= W
print(len(L))
