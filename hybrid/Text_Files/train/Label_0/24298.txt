#include <bits/stdc++.h>
using namespace std;
int a[200001];
int n;
pair<pair<int, int>, bool> check(int d) {
  int sum = 0;
  int ct = 0;
  pair<pair<int, int>, bool> p = make_pair(make_pair(0, 0), 0);
  for (int i = 1; i < n; i++) {
    sum += a[i] - a[i - 1];
    if (sum > d) {
      ct++;
      sum = 0;
      if (ct == 1) {
        p.first.first = i - 1;
      } else if (ct == 2) {
        p.first.second = i - 1;
      } else {
        return p;
      }
    }
  }
  p.second = true;
  if (p.first.second == 0) {
    p.first.second = max(0, n - 2);
  }
  return p;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a, a + n);
  a[n] = a[n - 1];
  int l = 0;
  int r = 2 * (a[n - 1] - a[0]);
  int m = (l + r) / 2;
  pair<pair<int, int>, bool> p;
  while (l < r) {
    p = check(m);
    if (p.second)
      r = m;
    else
      l = m + 1;
    m = (l + r) / 2;
  }
  p = check(m);
  cout << fixed;
  cout.precision(20);
  cout << double(m) / 2 << endl;
  cout << double(a[p.first.first] + a[0]) / 2 << " "
       << double(a[p.first.second] + a[p.first.first + 1]) / 2 << " "
       << double(a[n - 1] + a[p.first.second + 1]) / 2;
  return 0;
}
