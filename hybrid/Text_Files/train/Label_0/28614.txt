#include <bits/stdc++.h>
using namespace std;
const long long N = 400 + 4;
const long long INF = 1e18;
const long long lim = 27;
long long c[27][27];
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  string a;
  cin >> a;
  long long ans = 0;
  map<char, long long> cnt;
  for (long long i = 0; i < a.size(); ++i) {
    for (char f = 'a'; f <= 'z'; ++f) {
      c[a[i] - 'a'][f - 'a'] += cnt[f];
    }
    cnt[a[i]]++;
  }
  for (char f = 'a'; f <= 'z'; ++f) {
    for (char s = 'a'; s <= 'z'; ++s) {
      ans = max(ans, c[f - 'a'][s - 'a']);
    }
    ans = max(ans, cnt[f]);
  }
  cout << ans << endl;
  return 0;
}
