from collections import OrderedDict

def getMinAlpha(freq):
	for i in freq:
		if freq[i] != 0:
			break
	return i
s = input()
(t, u) = ([], [])
freq = OrderedDict()
for i in range(1, 27):
	freq.update({chr(i + 96): 0})
for i in range(len(s)):
	freq.update({s[i]: freq[s[i]] + 1})
for i in range(len(s)):
	minElm = getMinAlpha(freq)
	while len(t) > 0 and t[-1] <= minElm:
		u.append(t.pop())
	if s[i] == minElm:
		u.append(s[i])
	else:
		t.append(s[i])
	freq.update({s[i]: freq[s[i]] - 1})
while len(t) > 0:
	u.append(t.pop())
print(''.join(u))
