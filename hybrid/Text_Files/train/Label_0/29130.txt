#include<bits/stdc++.h>

const int inf=0x3f3f3f3f;
const int maxn=2e4+5;
const int N=6e6+5;
const int mod=1e9;
const double pi=acos(-1);
int n,a[maxn],w,t;
int input(int flag,int op,int x,int y,int val){
	printf("? %d %d %d %d\n",op,x,y,val);
	fflush(stdout);
	cin>>w;
	if(op==1&&w==n-1&&!flag) return input(1,op,y,x,val);
	if(op==2&&w==2&&!flag) return input(1,op,y,x,val);
	return w;
}
void solve1(int l)
{
    int aa,b,c,d,e,f;
    aa=input(0,1,l,l+1,n-1);
    b=input(0,2,l,l+1,1);

    c=input(0,1,l+2,l+3,n-1);
    d=input(0,2,l+2,l+3,1);

    e=input(0,1,l+1,l+2,n-1);
    f=input(0,2,l+1,l+2,1);

    if(e==aa)
    {
        a[l+1]=aa;
        a[l]=b;
        a[l+2]=f;
        if(f==d) a[l+3]=c;
        else a[l+3]=d;
    }
    else if(e==b)
    {
        a[l+1]=b;
        a[l]=aa;
        a[l+2]=f;
        if(f==d) a[l+3]=c;
        else a[l+3]=d;
    }
    else if(e==c)
    {
        a[l+3]=d;
        a[l+2]=c;
        a[l+1]=f;
        if(f==b) a[l]=aa;
        else a[l]=b;
    }
    else if(e==d)
    {
        a[l+3]=c;
        a[l+2]=d;
        a[l+1]=f;
        if(f==b) a[l]=aa;
        else a[l]=b;
    }
}
int solve3()
{
    int aa,b,c,d;
    aa=input(0,1,1,2,n-1);
    b=input(0,2,1,2,1);
    c=input(0,1,2,3,n-1);
    d=input(0,2,2,3,1);

    if(aa==c)
    {
        a[2]=aa;
        a[1]=b;
        a[3]=d;
    }
    else if(aa==d)
    {
        a[1]=b;
        a[2]=aa;
        a[3]=c;
    }
    else if(b==c)
    {
        a[1]=aa;
        a[2]=b;
        a[3]=d;
    }
    else if(b==d)
    {
        a[1]=aa;
        a[2]=b;
        a[3]=c;
    }
}

int main()
{
    scanf("%d",&t);
    while(t--)
    {
        scanf("%d",&n);
        if(n==3) solve3();
        else
        {
            for(int i=1;i<=(n/4)*4-1;i+=4)
                solve1(i);
            if(n%4) solve1(n-3);
        }
        printf("!"); fflush(stdout);
        for(int i=1;i<=n;i++) printf(" %d",a[i]),fflush(stdout);
        printf("\n");  fflush(stdout);
    }

    return 0;
}
