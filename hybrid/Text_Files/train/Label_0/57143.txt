import java.io.*;
import java.util.*;

public class Solution {
    private static BufferedReader br;
    private static StringTokenizer st;
    private static PrintWriter pw;
    private HashSet<Integer> ms;

    private static String nextToken() {
        try {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(br.readLine());
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
        return st.nextToken();
    }

    private static int nextInt() {
        return Integer.parseInt(nextToken());
    }

    static long nextLong() {
        return Long.parseLong(nextToken());
    }

    static double nextDouble() {
        return Double.parseDouble(nextToken());
    }

    static char nextChar() throws IOException {
        return (char) br.read();
    }

    static String next() throws IOException {
        return nextToken();
    }

    static String nextLine() throws IOException {
        return br.readLine();
    }

    static ArrayList<Integer> s;
    static ArrayList<Integer> t;

    public static void main(String[] args) throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        pw = new PrintWriter(System.out);
        run();
        pw.close();
    }

    private static void run() {
        int n = nextInt();
        int m = nextInt();
        ArrayList<Integer>[] g = new ArrayList[n];
        ArrayList<Integer>[] gt = new ArrayList[n];

        for (int i = 0; i < n; i++) {
            g[i] = new ArrayList<>();
            gt[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int v = nextInt() - 1;
            int u = nextInt() - 1;
            g[v].add(u);
            gt[u].add(v);
        }

        s = new ArrayList<>();
        t = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            if (gt[i].isEmpty())
                s.add(i);
            if (g[i].isEmpty())
                t.add(i);
        }

        ArrayList<Integer> set = new ArrayList<>();

        boolean[] used = new boolean[n];

        for (int i = 0; i < s.size(); i++) {

            Arrays.fill(used, false);

            bfs(s.get(i), used, g);

            set.add(0);

            for (int t0 = 0; t0 < t.size(); t0++) {
                if (used[t.get(t0)])
                    set.set(i, set.get(i) ^ (1 << t0));
            }
        }

        for (int i = 1; i < (1 << s.size()) - 1; i++) {
            int cnt = 0;
            int k = 1;
            for (int j = 0; j < s.size(); j++) {
                if ((i & k) > 0) {
                    cnt |= set.get(j);
                }
                k = k << 1;
            }
            if (Integer.bitCount(cnt) <= Integer.bitCount(i)) {
                pw.println("NO");
                return;
            }

        }
        pw.println("YES");
    }

    static void bfs(int i, boolean[] used, ArrayList<Integer>[] g) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(i);
        while (!queue.isEmpty()) {
            int a = queue.poll();
            used[a] = true;
            for (int v : g[a]){
                queue.add(v);
            }
        }
    }
}
