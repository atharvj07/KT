from sys import stdin as sin

def aint():
	return int(input())

def amap():
	return map(int, sin.readline().split())

def alist():
	return list(map(int, sin.readline().split()))

def astr():
	return input()
for _ in range(aint()):
	n = aint()
	l = alist()
	l.sort()
	m = float('inf')
	ans = 1
	for i in range(n - 1):
		if l[i] <= 0 and l[i + 1] <= 0:
			ans += 1
			m = min(m, abs(l[i + 1] - l[i]))
		elif l[i] <= 0 and l[i + 1] > 0 and (l[i + 1] <= m):
			ans += 1
			m = min(m, abs(l[i + 1] - l[i]))
		else:
			break
	print(ans)
