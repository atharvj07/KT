def solve(n):
	if n % 2 != 0:
		return True

	def is_prime(n):
		if n == 1:
			return False
		i = 2
		while i * i <= n:
			if n % i == 0:
				return False
			i += 1
		return True
	if n & n - 1 == 0 or is_prime(n):
		return False
	return True
t = int(input())
while t:
	n = int(input())
	if solve(n):
		print('YES')
	else:
		print('NO')
	t -= 1
