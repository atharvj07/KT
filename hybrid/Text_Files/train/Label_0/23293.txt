void addmod(int& x, int y, int mod = 998244353) {
  assert(y >= 0);
  x += y;
  if (x >= mod) x -= mod;
  assert(x >= 0 && x < mod);
}

long long fastPow(long long x, long long y, int mod = 998244353) {
  long long ans = 1;
  while (y > 0) {
    if (y & 1) ans = (x * ans) % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ans;
}

vector<pair<int, int> > mp[135];

void dfs(int x) {
  vis[x] = 1;
  vpt.push_back(x);
  be[x] = zi;
  for (auto cp : mp[x])
    if (!vis[cp.first] && cp.second == A) {
      dfs(cp.first);
    }
}
int dis[(1 << 17) + 2][72], Ans[72];
bool used[(1 << 17) + 2][72];
void dij() {
  for (int(i) = 0; (i) < (int)(1 << C); (i)++)
    for (int(x) = 1; (x) <= (int)(n); (x)++) dis[i][x] = (1000000000);
  int bs = 0;
  dis[bs][1] = 0;
  queue<pair<int, int> > qa, qb;
  qa.push({bs, 1});
  for (int(i) = 1; (i) <= (int)(n); (i)++) Ans[i] = (1000000000);
  Ans[1] = 0;
  while (!qa.empty() || !qb.empty()) {
    pair<int, int> sta;
    if (qb.empty()) {
      sta = qa.front();
      qa.pop();
    } else if (qa.empty()) {
      sta = qb.front();
      qb.pop();
    } else {
      if (dis[qa.front().first][qa.front().second] <
          dis[qb.front().first][qb.front().second]) {
        sta = qa.front();
        qa.pop();
      } else {
        sta = qb.front();
        qb.pop();
      }
    }
    int x = sta.second;
    if (used[sta.first][x]) continue;
    used[sta.first][x] = 1;
    int msk = sta.first;
    for (auto cp : mp[x]) {
      int d = cp.second, y = cp.first;
      int nk = msk;
      if (d == B && be[x] == be[y]) continue;
      if (col[x] != col[y]) {
        if (col[y] == -1) {
          if (col[x] != -1) nk |= 1 << col[x];
        } else {
          if ((msk & (1 << col[y])) != 0) continue;
          if (col[x] != -1) nk |= 1 << col[x];
        }
      }
      if (dis[msk][x] + d < dis[nk][y]) {
        dis[nk][y] = dis[msk][x] + d;
        mintt(Ans[y], dis[nk][y]);
        if (d == A)
          qa.push({nk, y});
        else
          qb.push({nk, y});
      }
    }
  }
}
