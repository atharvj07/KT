#include <bits/stdc++.h>
using namespace std;
mt19937_64 rnd(time(nullptr));
void solve() {
  long long a[6];
  for (long long i = 0; i < 6; i++) cin >> a[i];
  sort(a, a + 6);
  long long n;
  cin >> n;
  long long b[n];
  for (long long i = 0; i < n; i++) cin >> b[i];
  sort(b, b + n);
  vector<pair<long long, long long>> v;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < 6; j++) v.push_back({b[i] - a[j], i});
  }
  sort(v.begin(), v.end());
  long long cnt = 0;
  long long l = 0;
  long long mas[n];
  for (long long i = 0; i < n; i++) mas[i] = 0;
  long long ans = 1e18;
  for (long long i = 0; i < 6 * n; i++) {
    while (l < 6 * n && cnt != n) {
      if (!mas[v[l].second]) {
        cnt++;
      }
      mas[v[l].second]++;
      l++;
    }
    if (cnt == n) {
      ans = min(ans, v[l - 1].first - v[i].first);
    }
    mas[v[i].second]--;
    if (mas[v[i].second] == 0) cnt--;
  }
  cout << ans << endl;
}
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  ;
  long long attempts = 1;
  cout.precision(16);
  while (attempts--) {
    solve();
  }
  return 0;
}
