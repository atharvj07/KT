n1 = int(input())
n2 = int(input())
k1 = int(input())
k2 = int(input())
n = int(input())
mini = 0
maxi = 0
p = n
if k1 < k2:
	maxi = n // k1
	if maxi <= n1:
		n = n - maxi * k1
	else:
		maxi = n1
		n = n - maxi * k1
	maxi += n // k2
else:
	maxi = n // k2
	if maxi > n2:
		maxi = n2
	n = n - maxi * k2
	maxi += n // k1
n = p
k11 = k1 - 1
k22 = k2 - 1
n = n - k11 * n1 - k22 * n2
if n > 0:
	mini = n
print(mini, maxi)
