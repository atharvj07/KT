#include <bits/stdc++.h>
using namespace std;
int n, y[3], ans, u, v, m;
bool f;
struct w {
  int a, b, c, d;
};
w x[100005];
bool mycmp(w m1, w m2) {
  return m1.a > m2.a ||
         (m1.a == m2.a && (m1.b > m2.b || (m1.b == m2.b && m1.c > m2.c)));
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d%d%d", &y[0], &y[1], &y[2]);
    sort(y, y + 3);
    x[i].a = y[2], x[i].b = y[1], x[i].c = y[0], x[i].d = i;
  }
  sort(x + 1, x + n + 1, mycmp);
  for (int i = 1; i < n; i++)
    if (x[i].a == x[i + 1].a && x[i].b == x[i + 1].b) {
      m = min(x[i].b, x[i].c + x[i + 1].c);
      if (m > ans) ans = m, f = 0, u = x[i].d, v = x[i + 1].d;
    } else if (x[i].c > ans)
      ans = x[i].c, f = 1, u = x[i].d;
  if (x[n].c > ans) ans = x[n].c, f = 1, u = x[n].d;
  if (f)
    printf("1\n%d", u);
  else
    printf("2\n%d %d", u, v);
  return 0;
}
