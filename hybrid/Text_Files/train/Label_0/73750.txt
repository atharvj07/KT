(n, x) = map(int, input().split())
arr = list(map(int, input().split()))
dict = {}
for v in arr:
	if v in dict:
		dict[v] += 1
	else:
		dict[v] = 1
if x:
	res = 0
	for v in dict.keys():
		if v ^ x in dict and v ^ x > v:
			res += dict[v] * dict[v ^ x]
	print(res)
else:
	print(sum(map(lambda x: x * (x - 1) // 2, dict.values())))
