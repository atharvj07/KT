f = lambda : map(int, input().split())
(n, m) = f()
t = []
p = [0] * (n + 1)
for i in range(1, m + 1):
	(s, d, c) = f()
	if p[d]:
		exit(print(-1))
	p[d] = m + 1
	t.append((i, s, d, c))
t.sort(key=lambda q: q[2])
for (i, s, d, c) in t:
	while c:
		if not p[s]:
			p[s] = i
			c -= 1
		if s == d:
			exit(print(-1))
		s += 1
print(*p[1:])
