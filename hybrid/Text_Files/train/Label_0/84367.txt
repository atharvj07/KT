def solve(M, m, n):
	tempoFinal = [0] * m
	for i in range(n):
		livre = 0
		for j in range(m):
			inicio = max(livre, tempoFinal[j])
			tempoFinal[j] = inicio + M[j][i]
			livre = tempoFinal[j]
	print(*tempoFinal, sep=' ')
(m, n) = map(int, input().split())
aux = []
for i in range(m):
	aux.append(list(map(int, input().split())))
solve(aux, m, n)
