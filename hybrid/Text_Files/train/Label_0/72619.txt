import sys
(n, k, q) = map(int, sys.stdin.readline().strip().split(' '))
(minR, maxR) = (10 ** 9, -10 ** 9)
tmp = 0
re = []
for i in range(n):
	(l, r) = map(int, sys.stdin.readline().strip().split(' '))
	minR = min(minR, l)
	maxR = max(maxR, r)
	re.append((l, r))
interval = [0 for _ in range(maxR - minR + 2)]
for r in re:
	interval[r[0] - minR] += 1
	interval[r[1] - minR + 1] -= 1
L = len(interval)
for i in range(1, L):
	interval[i] += interval[i - 1]
res = [0 for _ in range(L + 1)]
for i in range(L):
	if interval[i] >= k:
		res[i + 1] = 1
for i in range(1, L + 1):
	res[i] += res[i - 1]
for i in range(q):
	(l, r) = map(int, sys.stdin.readline().strip().split(' '))
	if r < minR or l > maxR:
		print(0)
		continue
	print(res[min(r + 1 - minR, L)] - res[max(0, l - minR)])
