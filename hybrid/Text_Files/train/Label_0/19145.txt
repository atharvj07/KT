#include<bits/stdc++.h>
using namespace std;

const int SIZE=100;
using F=bitset<SIZE>;

auto comp=[](const F& lhs,const F& rhs){
    return lhs.count()<rhs.count();
};

int dfs(int pos,vector<vector<F>> &bigSq,F state,unordered_map<F,int>& dp){
    if(pos==bigSq.size()){
        return 0;
    }
    if(dp.count(state)){
        return dp[state];
    }
    if(!state[pos]){
        auto ret=dfs(pos+1,bigSq,state,dp);
        dp[state]=ret;
        return ret;
    }
    
    vector<F> toS;
    for(int i=0;i<bigSq[pos].size();i++){
        toS.push_back(state & (~bigSq[pos][i]));
    }
    int res=1e6;
    sort(toS.begin(),toS.end(),comp);
    for(int i=0;i<toS.size();i++){
        bool isok=true;
        for(int j=0;j<i;j++){
            isok&=!((toS[i]&toS[j])==toS[j]);
        }
        if(isok){
            res=min(res,dfs(pos+1,bigSq,toS[i],dp)+1);
        }
    }
    return dp[state]=res;
}

int main(){
    int w,h;
    while(cin>>w>>h,w){
        vector<vector<int>> f(h,vector<int>(w,0));
        for(int i=0;i<h;i++){
            for(int j=0;j<w;j++){
                cin>>f[i][j];
            }
        }
        
        F state=0;
        vector<vector<F>> bigSq(h*w);
        for(int i=0;i<h;i++){
            for(int j=0;j<w;j++){
                if(!f[i][j]) continue;
                state.set(i*w+j);
                int s=0;
                for(int k=1;k<=10;k++){
                    bool isok=true;
                    for(int ii=0;ii<k;ii++){
                        for(int jj=0;jj<k;jj++){
                            isok&=(i+ii<h && j+jj<w && f[i+ii][j+jj]);
                        }
                    }
                    if(!isok) break;
                    s++;
                }
                F sq=0;
                for(int ii=0;ii<s;ii++){
                    for(int jj=0;jj<s;jj++){
                        sq.set((ii+i)*w+j+jj);
                    }
                }
                for(int ii=0;ii<s;ii++){
                    for(int jj=0;jj<s;jj++){
                        bigSq[(i+ii)*w+j+jj].push_back(sq);
                    }
                }
                sort(bigSq[i*w+j].begin(),bigSq[i*w+j].end(),comp);
            }
        }

        for(int i=0;i<h*w;i++){
            vector<F> newBigSq;
            for(int j=0;j<bigSq[i].size();j++){
                bool isok=true;
                for(int k=j+1;k<bigSq[i].size();k++){
                    isok&=((bigSq[i][j] & bigSq[i][k])!=bigSq[i][j]);
                }
                if(isok) newBigSq.push_back(bigSq[i][j]);
            }
            bigSq[i]=newBigSq;
        }
        
        unordered_map<F,int> dp;

        int res=dfs(0,bigSq,state,dp);
        cout<<res<<endl;
    }

    return 0;
}
