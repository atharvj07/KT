import sys
import collections
(n, m) = map(int, input().split())
segs = set()
for _ in range(m):
	(start, end) = map(int, input().split())
	segs.add((start - 1, end - 1))

def get_divisors(n):
	divisors = []
	for i in range(1, n):
		if n % i == 0:
			divisors.append(i)
	return divisors
divisors = get_divisors(n)
for k in divisors:
	for (sega, segb) in segs:
		if ((sega + k) % n, (segb + k) % n) not in segs and ((segb + k) % n, (sega + k) % n) not in segs:
			break
	else:
		print('Yes')
		sys.exit()
print('No')
