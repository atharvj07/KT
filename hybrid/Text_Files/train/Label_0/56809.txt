import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        try (Scanner in = new Scanner(System.in)) {
            int nCand = in.nextInt();
            int mTowns = in.nextInt();
            int[][] arr = new int[mTowns][nCand];

            for (int i = 0; i < mTowns; i++) {
                for (int j = 0; j < nCand; j++) {
                    arr[i][j] = in.nextInt();
                }
            }

            int[] candidatePerTown = new int[nCand];

            for (int i = 0; i < mTowns; i++) {
                int maxCand = -1;
                int maxValue = -1;
                for (int j = nCand - 1; j >= 0; j--) {
                    if (arr[i][j] >= maxValue) {
                        maxValue = arr[i][j];
                        maxCand = j;
                    }
                }
                candidatePerTown[maxCand]++;
            }

            int maxCand = -1;
            int maxValue = -1;
            for (int i = nCand - 1; i >= 0; i--) {
                if (candidatePerTown[i] >= maxValue) {
                    maxValue = candidatePerTown[i];
                    maxCand = i;
                }
            }
            System.out.println(maxCand+ 1);
        }
    }
}
