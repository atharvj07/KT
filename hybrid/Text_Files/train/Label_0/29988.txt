#include <bits/stdc++.h>
using namespace std;
int n, h, s, x, i, l[200010], r[200010], d[200010];
int main() {
  scanf("%d%d", &n, &h);
  for (i = 1; i <= n; ++i)
    scanf("%d%d", l + i, r + i), d[i] = d[i - 1] + l[i] - r[i - 1];
  for (i = 1; i <= n; ++i) {
    x = lower_bound(d + i + 1, d + n + 1, h + d[i]) - d - 1;
    s = max(s, r[x] - l[i] + h - d[x] + d[i]);
  }
  printf("%d\n", s);
}
