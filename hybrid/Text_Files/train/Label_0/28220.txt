#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-8;
inline int max(int a, int b) { return a < b ? b : a; }
inline int min(int a, int b) { return a > b ? b : a; }
inline long long max(long long a, long long b) { return a < b ? b : a; }
inline long long min(long long a, long long b) { return a > b ? b : a; }
const int mod = 1e9 + 7;
const int N = 1e6 + 10;
const long long inf = 1e18;
int A[100100];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &A[i]);
  }
  if (A[n - 1] == 1) {
    printf("NO\n");
    return 0;
  }
  if (n == 1) {
    printf("YES\n0\n");
    return 0;
  }
  if (n == 2) {
    if (A[0] == 1) {
      printf("YES\n1->0\n");
      return 0;
    }
    printf("NO\n");
    return 0;
  }
  if (n == 3) {
    if (A[0] == 1 && A[1] == 0) {
      printf("NO\n");
      return 0;
    } else {
      printf("YES\n");
      printf("(%d->%d)->0\n", A[0], A[1]);
      return 0;
    }
  }
  int in = A[2];
  for (int i = 3; i < n - 1; i++) {
    if (in == 0)
      in = 1;
    else {
      if (A[i] == 0)
        in = 0;
      else
        in = 1;
    }
  }
  if (A[0] == 0 && A[1] == 0) {
    printf("YES\n");
    if (in == 0) {
      printf("0->(0->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf("))->0\n");
    } else {
      printf("(0->0)->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf(")->0\n");
    }
  } else if (A[0] == 0 && A[1] == 1) {
    printf("YES\n");
    if (in == 0) {
      printf("0->(1->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf("))->0\n");
    } else {
      printf("(0->1)->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf(")->0\n");
    }
  } else if (A[0] == 1 && A[1] == 0) {
    printf("YES\n");
    if (in == 0) {
      printf("1->(0->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf("))->0\n");
    } else {
      printf("(1->0)->(");
      for (int i = 2; i < n - 1; i++) {
        printf("%d", A[i]);
        if (i != n - 2) printf("->");
      }
      printf(")->0\n");
    }
  } else if (A[0] == 1 && A[1] == 1) {
    int f = 0;
    for (int i = 2; i < n - 2; i++)
      if (A[i] == 0) f = 1;
    if (A[n - 2] == 1) {
      printf("YES\n");
      for (int i = 0; i < n; ++i) {
        printf("%d", A[i]);
        if (i != n - 1) printf("->");
      }
      printf("\n");
      return 0;
    }
    if (f == 0)
      printf("NO\n");
    else {
      printf("YES\n");
      printf("1->1->");
      for (int i = 2; i < n - 1; i++) {
        printf("(%d", A[i]);
        if (i != n - 2) printf("->");
      }
      for (int i = 2; i < n - 1; i++) {
        printf(")");
      }
      printf("->0\n");
    }
  }
  return 0;
}
