for _ in range(int(input())):
	n = int(input())
	xl = -100000
	xr = 100000
	yup = 100000
	ydown = -100000
	stop = 0
	for i in range(n):
		(x, y, f1, f2, f3, f4) = map(int, input().split())
		if not f1:
			if xl <= x and x <= xr:
				xl = x
			elif x < xl and (not f3):
				stop = 1
			elif x > xr:
				stop = 1
		if not f3:
			if xl <= x and x <= xr:
				xr = x
			elif x > xr and (not f1):
				stop = 1
			elif x < xl:
				stop = 1
		if not f2:
			if ydown <= y and y <= yup:
				yup = y
			elif y > yup and (not f4):
				stop = 1
			elif y < ydown:
				stop = 1
		if not f4:
			if ydown <= y and y <= yup:
				ydown = y
			elif y < ydown and (not f2):
				stop = 1
			elif y > yup:
				stop = 1
	if not stop:
		print(1, xr, yup)
	else:
		print(0)
