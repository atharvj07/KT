import java.util.*;

public class BagOfMice
{
	static double prob[][];
	static int totalMice;
	
	public static void main(String args[])
	{
		Scanner in = new Scanner(System.in);
		int w = in.nextInt();
		int b = in.nextInt();
		prob = new double[w+1][b+1];
		totalMice = w + b;
		for(int i = 0; i <= w; i++){
			for(int j = 0; j <= b ; j++)
				prob[i][j] = -1;
		}	
		System.out.println(probFunction(w,b));
	}
	
	public static double probFunction(int w, int b)
	{
		double probW = (double)w / (w + b);
		double probB = (double)b / (w + b);
		
		if((w == -1 || b == -1) || (w == 0 && b == 0))
		{
			return new Double(0);
		}
		else
		{	
			if(prob[w][b] != -1)
				return prob[w][b];
			else{
				double princessProb;
				int turn = (totalMice - w - b) % 3;
				//turn: 0 = princess, 1 = dragon, 2 = mice escapes
				switch(turn)
				{
					case 0 : princessProb = probW + (probB * probFunction(w, b - 1)); break;
					case 1 : princessProb = probB * probFunction(w, b - 1); break;
					default : princessProb = (probW * probFunction(w - 1, b)) + (probB * probFunction(w, b - 1)); break;
				}
				prob[w][b] = princessProb;
				return princessProb;
			}
		}
		
	}
}
