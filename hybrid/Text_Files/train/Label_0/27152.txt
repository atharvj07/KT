#include <bits/stdc++.h>
using namespace std;
vector<int> v[5050];
int nx[5050];
int main() {
  int n, m;
  scanf("%d%d", &(n), &(m));
  int a, b;
  for (int i = 1; i <= m; i++) {
    scanf("%d%d", &(a), &(b));
    v[a].push_back(b);
  }
  for (int i = 1; i <= n; i++) {
    if (v[i].empty()) continue;
    sort((v[i]).begin(), (v[i]).end());
    auto it = upper_bound((v[i]).begin(), (v[i]).end(), i);
    if (it == v[i].end())
      nx[i] = v[i][0];
    else
      nx[i] = *it;
  }
  for (int s = 1; s <= n; s++) {
    int cc = 0, ans = 0, i = s;
    do {
      if (!v[i].empty())
        ans = max(ans, cc + (int)(v[i].size() - 1) * n + (nx[i] + n - i) % n);
      cc++;
      i++;
      if (i > n) i = 1;
    } while (i != s);
    printf("%d ", ans);
  }
  printf("\n");
  return 0;
}
