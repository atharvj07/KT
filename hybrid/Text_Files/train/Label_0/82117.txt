def is_palindrome(s):
	return str(s) == str(s)[::-1]

def convert_palindrome(s):
	if s[0] == s[-1]:
		sub_palindrome = convert_palindrome(s[1:-1])
		if sub_palindrome is None:
			return None
		else:
			return s[0] + sub_palindrome + s[-1]
	elif is_palindrome(s + s[0]):
		return s + s[0]
	elif is_palindrome(s[-1] + s):
		return s[-1] + s
	else:
		return None

def main():
	s = input()
	length = len(s)
	if is_palindrome(s):
		if length % 2 == 0:
			mid = length // 2
			print(s[:mid] + 'a' + s[mid:])
		else:
			mid = (length - 1) // 2
			print(s[:mid] + s[mid] + s[mid:])
	else:
		new_palindrome = convert_palindrome(s)
		if new_palindrome is None:
			print('NA')
		else:
			print(new_palindrome)
main()
