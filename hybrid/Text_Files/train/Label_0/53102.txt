import java.util.Scanner;

public class A {

	static int N;
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int ans=0;
		int sum=0;
		N=sc.nextInt();
		for(int i=0;i<N;i++){ 
			int a = sc.nextInt();
			if(a==-1){
				if(sum==0){
					ans++;
				}
				else{
					sum--;
				}
			}
			else{
				sum+=a;
			}
		}
		System.out.println(ans);
	}
}
