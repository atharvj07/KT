n = int(input())
l = input()
s = []
for i in range(n):
	s.append(l[i])
k = 0
c = set()
q = set('RGB')
if n == 1:
	print(0)
	print(s[0])
else:
	for i in range(n - 2):
		if s[i] == s[i + 1]:
			k += 1
			c.add(s[i])
			c.add(s[i + 2])
			w = list(q - c)[0]
			s[i + 1] = w
			c = set()
	if s[n - 1] == s[n - 2]:
		c.add(s[n - 2])
		w = list(q - c)[0]
		s[n - 1] = w
		k += 1
	print(k)
	print(*s, sep='')
