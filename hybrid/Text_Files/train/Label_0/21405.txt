#include<stdio.h>
#include<string.h>
#include<math.h>
#include<iostream>
#include<algorithm>
using namespace std;
#define maxn 110
int dis[maxn],head[maxn],deg[maxn];
struct Edge{
	int st,to,next,len;
}edge[maxn*maxn*2];
int n,m,p,num;
double hav[maxn];
int pos[maxn],pre[maxn][maxn],cnt[maxn],mark[maxn];
double ans[maxn];
int l,h,q[maxn];
void addedge(int a,int b ,int c)
{
	edge[num].st=a;edge[num].to=b;edge[num].len=c;
	edge[num].next=head[a];head[a]=num++;
}
void dij()
{
	int i,j,k;
	int id,now;
	memset(dis,-1,sizeof(dis));
	memset(mark,0,sizeof(mark));
	memset(cnt,0,sizeof(cnt));
	dis[0]=0;
	while(1){
		now=-1;
		for(i=0;i<n;i++){
			if(dis[i]==-1)continue;
			if((dis[i]<now||now==-1)&&!mark[i]){
				now=dis[i];
				id=i;
			}
		}
		if(now==-1)break;
		mark[id]=1;
		for(i=head[id];i!=-1;i=edge[i].next){
			j=edge[i].to;
			if(dis[j]==-1||dis[id]+edge[i].len<dis[j]){
				dis[j]=dis[id]+edge[i].len;
				cnt[j]=1;
				pre[j][cnt[j]]=id;
			}else if(dis[id]+edge[i].len==dis[j]){
				cnt[j]++;
				pre[j][cnt[j]]=id;
			}
		}
	}
}
void dfs(int now)
{
	if(mark[now])return;
	if(now==0){
		ans[now]=1;
		mark[now]=1;
		return;
	}
	int i;
	ans[now]=0;
	for(i=1;i<=cnt[now];i++){
		dfs(pre[now][i]);
		ans[now]+=ans[pre[now][i]]*1.0*hav[now]/hav[pre[now][i]];
	}
	mark[now]=1;
}
void dfs1(int now)
{
	if(mark[now])return;
	int i,j;
	for(i=1;i<=cnt[now];i++){
		j=pre[now][i];
		deg[j]++;
		dfs1(j);
	}
	mark[now]=1;
}
void work()
{
	int i,j,k;
	memset(hav,0,sizeof(hav));
	hav[n-1]=1;
	l=h=0;
	q[h++]=n-1;
	while(l<h){
		k=q[l++];
		for(i=1;i<=cnt[k];i++){
			j=pre[k][i];
			deg[j]--;
			hav[j]+=hav[k];
			if(deg[j]==0){
				q[h++]=j;
			}
		}
	}
}
			
int main()
{
	int i,j,k;
	int st,to,cost;
	while(scanf("%d%d%d",&n,&m,&p),n+m+p){
		num=0;
		memset(head,-1,sizeof(head));
		while(m--){
			scanf("%d%d%d",&st,&to,&cost);
			addedge(st,to,cost);
			addedge(to,st,cost);
		}
		for(i=1;i<=p;i++){
			scanf("%d",&pos[i]);
		}
		dij();
		if(dis[n-1]==-1)while(1);
		memset(mark,0,sizeof(mark));
		memset(deg,0,sizeof(deg));
		dfs1(n-1);
		work();
		/*for(i=0;i<n;i++){
			printf("%d %d %d\n",i,deg[i],cnt[i]);
		}*/
		for(i=0;i<n;i++){
			ans[i]=0;
		}
		
		memset(mark,0,sizeof(mark));
		dfs(n-1);
		for(i=1;i<=p;i++){
			printf("%.8f\n",ans[pos[i]]);
		}
		putchar('\n');
	}
	return 0;
}
