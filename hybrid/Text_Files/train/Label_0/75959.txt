figs = {}
for i in range(8):
	s = input()
	for j in s:
		if j != '.':
			if j not in figs:
				figs[j] = 1
			else:
				figs[j] += 1
white = 0
black = 0
for i in figs:
	if i == 'q':
		black += figs[i] * 9
	elif i == 'r':
		black += figs[i] * 5
	elif i == 'b' or i == 'n':
		black += figs[i] * 3
	elif i == 'p':
		black += figs[i]
	if i == 'Q':
		white += figs[i] * 9
	elif i == 'R':
		white += figs[i] * 5
	elif i == 'B' or i == 'N':
		white += figs[i] * 3
	elif i == 'P':
		white += figs[i]
if black > white:
	print('Black')
elif white > black:
	print('White')
else:
	print('Draw')
