n = 100001
p = [0] * n
y = [[] for i in range(n)]
y[1] = [1]
for i in range(2, n):
	if not y[i]:
		y[i] = [i]
		for j in range(2 * i, n, i):
			y[j].append(i)

def prob():
	n = int(input())
	l = [int(x) for x in input().split()]
	for i in l:
		k = max((p[j] for j in y[i])) + 1
		for j in y[i]:
			p[j] = k
	print(max(p))
t = 1
for _ in range(0, t):
	prob()
