import sys
input = sys.stdin.readline

def compress(init_val):
	t = sorted(set(init_val))
	d = dict([[t[i], i] for i in range(len(t))])
	res = [0] * len(init_val)
	for i in range(len(init_val)):
		res[i] = d[init_val[i]]
	return res
for _ in range(int(input())):
	n = int(input())
	a = compress(list(map(int, input().split())))
	table = [0] * n
	mx = max(a)
	s = [[] for _ in range(mx + 1)]
	for i in range(n):
		s[a[i]].append(i)
	for x in range(mx + 1):
		for t in range(len(s[x]) - 2, -1, -1):
			i = s[x][t]
			j = s[x][t + 1]
			table[j] += n - j
			table[i] += table[j]
		i = s[x][0]
		table[i] += n - i
	res = 0
	for i in range(n):
		l = i + 1
		r = max(0, table[i] - (n - i))
		res += l * r
	print(res)
