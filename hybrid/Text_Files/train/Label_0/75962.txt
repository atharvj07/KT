WHITE = 0
black = 0
for i in range(8):
	line = input()
	for elem in line:
		if elem == 'Q':
			WHITE += 9
		elif elem == 'R':
			WHITE += 5
		elif elem == 'B' or elem == 'N':
			WHITE += 3
		elif elem == 'P':
			WHITE += 1
		elif elem == 'q':
			black += 9
		elif elem == 'r':
			black += 5
		elif elem == 'b' or elem == 'n':
			black += 3
		elif elem == 'p':
			black += 1
if WHITE > black:
	print('White')
elif black > WHITE:
	print('Black')
else:
	print('Draw')
