(n, r) = map(int, input().split())
pos = []
neg = []
for _ in range(n):
	(a, b) = map(int, input().split())
	if b > 0:
		pos.append((a, b))
	else:
		neg.append((max(a, abs(b)), b))
pos.sort()
cnt = 0
for i in pos:
	(a, b) = i
	if r < a:
		break
	r += b
	cnt += 1
neg.sort(key=lambda x: x[0] + x[1], reverse=True)
dp = [[0 for i in range(r + 1)] for i in range(len(neg) + 1)]
for i in range(1, len(neg) + 1):
	for j in range(1, r + 1):
		if j >= neg[i - 1][0] and j + neg[i - 1][1] >= 0:
			dp[i][j + neg[i - 1][1]] = max(dp[i - 1][j] + 1, dp[i - 1][j + neg[i - 1][1]])
		else:
			dp[i][j] = max(dp[i - 1][j], dp[i][j])
print(cnt + max([max(i) for i in dp]))
