t = int(input())
while t > 0:
	nk = input()
	n = int(nk.split()[0])
	k = int(nk.split()[1])
	result = []
	if n % 2 == 0 and k * 2 <= n:
		print('YES')
		for x in range(k):
			result.append(2)
		result[0] = result[0] + n - k * 2
		for x in result:
			print(x, end=' ')
		print()
	elif n % 2 == 0 and k % 2 == 0 and (k <= n):
		print('YES')
		for x in range(k):
			result.append(1)
		result[0] = result[0] + n - k
		for x in result:
			print(x, end=' ')
		print()
	elif n % 2 != 0 and k * 1 <= n and (k % 2 != 0):
		print('YES')
		for x in range(k):
			result.append(1)
		result[0] = result[0] + n - k
		for x in result:
			print(x, end=' ')
		print()
	else:
		print('NO')
	t = t - 1
