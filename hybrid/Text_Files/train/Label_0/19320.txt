#include <iostream>

using namespace std;

char solve(int x,int y,int z,int u,int v){
    if (x == 1 && y == 0 && z == 0 && u == 1 && v == 0){
        return 'a';
    }
    if (x == 1 && y == 1 && z == 0 && u == 2 && v == 0){
        return 'b';
    }
    if (x == 2 && y == 0 && z == 0 && u == 1 && v == 1){
        return 'c';
    }
    if (x == 2 && y == 1 && z == 0 && u == 1 && v == 2){
        return 'd';
    }
    if (x == 1 && y == 1 && z == 0 && u == 1 && v == 1){
        return 'e';
    }
    if (x == 2 && y == 1 && z == 0 && u == 2 && v == 1){
        return 'f';
    }
    if (x == 2 && y == 2 && z == 0 && u == 2 && v == 2){
        return 'g';
    }
    if (x == 1 && y == 2 && z == 0 && u == 2 && v == 1){
        return 'h';
    }
    if (x == 1 && y == 1 && z == 0 && u == 1 && v == 1){
        return 'i';
    }
    if (x == 1 && y == 2 && z == 0 && u == 1 && v == 2){
        return 'j';
    }
    if (z == 1){
        //w is special
        if (x == 1 && y == 2 && z == 1 && u == 1 && v == 3){
            return 'w';
        }
        return solve(x,y,0,u-1,v) + 10;
    }
    char ind = solve(x,y,0,u-1,v-1);
    if (ind  == 'a' || ind == 'b') return ind + 20;
    if (ind >= 'c') return ind + 21;
}

int main(){
    int t;
    cin >> t;
    while (t--){
        int x,y,z,u,v;
        cin >> x >> y >> z >> u >>v;
        cout << solve(x,y,z,u,v);
    }
    cout << endl;
    return 0;
}
