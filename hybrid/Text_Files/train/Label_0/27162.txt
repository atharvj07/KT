#include <bits/stdc++.h>
using namespace std;
bool compare(const vector<string>& table, const vector<int>& mask, int col) {
  int n = table.size();
  bool ok = true;
  for (int r = 0; r + 1 < n; ++r) {
    string a, b;
    for (int c = 0; c <= col; ++c)
      if (mask[c]) {
        a += table[r][c];
        b += table[r + 1][c];
      }
    if (a > b) {
      ok = false;
      break;
    }
  }
  return ok;
}
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> table(n);
  for (int i = 0; i < n; ++i) cin >> table[i];
  int res = 0;
  vector<int> mask(m, 1);
  for (int c = 0; c < m; ++c) {
    if (!compare(table, mask, c)) {
      ++res;
      mask[c] = 0;
    }
  }
  cout << res << endl;
  return 0;
}
