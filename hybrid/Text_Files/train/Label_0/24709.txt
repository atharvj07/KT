constexpr long long int MOD = 998244353LL;
long long int power(const long long int base, const long long int exp) {
	switch (exp) {
	case 0LL: return 1;
	case 1LL: return base % MOD;
	default: return power(base * base % MOD, exp / 2) * power(base, exp % 2) % MOD;
	}
}
int main() {
	long long int m, n; int k; std::cin >> m >> n >> k;
	if (m < k) {
		std::cout << 0 << '\n';
		return 0;
	}
	std::vector<long long int> factorial(k, 1), inverse(k, 1), inverse_factorial(k, 1);
	for (auto i = 2; i < k; ++i) {
		factorial[i] = (i * factorial[i - 1]) % MOD;
		inverse[i] = (MOD - MOD / i) * inverse[MOD % i] % MOD;
		inverse_factorial[i] = inverse[i] * inverse_factorial[i - 1] % MOD;
	}
	std::vector<long long int> just_string(k, 0);
	for (auto i = 1; i < k; ++i) {
		just_string[i] = power(i, n);
		for (auto j = 1; j < i; ++j) {
			just_string[i] -= factorial[i] * inverse_factorial[i - j] % MOD * inverse_factorial[j] % MOD * just_string[j] % MOD;
			just_string[i] %= MOD;
		}
	}
	std::vector<long long int> falling_factorial(k, 1);;
	for (auto i = 1; i < falling_factorial.size(); ++i) {
		falling_factorial[i] = (m - i + 1) % MOD * falling_factorial[i - 1] % MOD;
	}
	long long int result = power(m % MOD, n);
	for (auto i = 1; i < k; ++i) {
		result -= falling_factorial[i] % MOD * inverse_factorial[i] % MOD * just_string[i] % MOD;
		result %= MOD;
	}
	std::cout << (result + MOD) % MOD << '\n';
	return 0;
}
