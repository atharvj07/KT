for _ in range(int(input())):
	res = []
	workwith = list(map(int, input().split()))
	if workwith[2] == workwith[0] + workwith[1]:
		res.append(workwith[0])
		res.append(workwith[1])
		res.append(workwith[3])
	else:
		res.append(workwith[0])
		res.append(workwith[1])
		res.append(workwith[2])
	print(*res)
