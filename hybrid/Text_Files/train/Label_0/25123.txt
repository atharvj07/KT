#include <bits/stdc++.h>
int main() {
  long long a, b, w, x, c;
  double k;
  long long int ans;
  while (scanf("%I64d%I64d%I64d%I64d%I64d", &a, &b, &w, &x, &c) != EOF) {
    k = ((c - a) * x - b) * 1.0 / (w - x);
    if (k > (long long int)k)
      ans = (long long int)k + 1;
    else
      ans = (long long int)k;
    ans = ans > 0 ? ans : 0;
    printf("%I64d\n", ans + c - a);
  }
}
