n = int(input())

def get_num_digits(n):
	i = 0
	while n:
		n //= 10
		i += 1
	return i

def get_sum_digits(n):
	s = 0
	while n:
		(n, ost) = divmod(n, 10)
		s += ost
	return s
i = get_num_digits(n)
m = n - i * 9
m = 0 if m < 0 else m
res = []
for k in range(m, n):
	if k + get_sum_digits(k) == n:
		res.append(k)
if res:
	print(len(res))
	for r in res:
		print(r)
else:
	print(0)
