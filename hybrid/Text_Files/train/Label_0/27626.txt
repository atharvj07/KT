template<typename T>inline void ckmx(T &x,T y){x=max(x,y);}
template<typename T>inline void ckmi(T &x,T y){x=min(x,y);}
template<typename T>void read(T &x){
	x=0;int f=1;
	char c=getchar();
	for(;!isdigit(c);c=getchar()) if (c=='-') f=-1;
	for(;isdigit(c);c=getchar()) x=x*10+c-48;
	x*=f;
}
int gcd(int x,int y){
	if(y==0) return x;
	else return gcd(y,x%y);
}
int a[100010];
int ans[1000010],an,n;
int p[1000010],ss[100010],ssn;
int f[100010];
int tmp[100010];
int main(){
	for(int i=1;i<=1000000;++i) p[i]=i;
	for(int i=2;i<=1000000;++i){
		if(p[i]==i) ss[++ssn]=i;
		for(int j=1;j<=ssn;++j){
			if(i*ss[j]>1000000||p[i]<ss[j]) break;
			p[i*ss[j]]=ss[j];
		}
	}
	read(n);
	int gc=0;
	for(int i=0;i<=n;++i){
		read(a[n-i]);
		gc=gcd(gc,abs(a[n-i]));
	}
	int sv=gc;
	for(int i=2;i*i<=gc;++i)
	if(gc%i==0){
		f[i]=1; ans[++an]=i;
		while(gc%i==0) gc/=i;
	}
	if(gc!=1) {
		ans[++an]=gc;
		if(gc<=n) f[gc]=1;
	}
	gc=sv;
	for(int i=2;i<=n;++i)
	if(p[i]==i&&(!f[i])){
		for(int j=0;j<=n;++j) tmp[j]=a[j]%i;
		for(int j=n;j>=i;--j)
			tmp[j-i+1]=(tmp[j-i+1]+tmp[j])%i,tmp[j]=0;
		int f=1;
		for(int j=i;j>=0&&f;--j)
			f&=(tmp[j]==0);
		if(f) ans[++an]=i;
	}
	sort(ans+1,ans+1+an);
	for(int i=1;i<=an;++i)
		printf("%d\n",ans[i]);
	return 0;
}
