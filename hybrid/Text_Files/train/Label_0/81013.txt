def test(n, s):
	a = list(map(int, input().split()))
	max_i = 0
	c = 0
	stop = 0
	for i in range(n):
		if s - a[i] >= 0:
			s -= a[i]
			c += 1
			if a[i] > a[max_i]:
				max_i = i
		else:
			if a[i] > a[max_i]:
				max_i = i
				stop = i + 1
			else:
				stop = i
			break
	if c == n:
		print(0)
		return
	else:
		c2 = c - 1
		s += a[max_i]
		for i in range(stop, n):
			if s - a[i] >= 0:
				s -= a[i]
				c2 += 1
			else:
				break
		if c2 >= c:
			print(max_i + 1)
			return
		else:
			print(0)
			return
for i in range(int(input())):
	(n, s) = map(int, input().split())
	test(n, s)
