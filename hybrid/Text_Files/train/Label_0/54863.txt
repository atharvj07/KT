			System.out.println(sb);
			System.out.flush();

			if(bulls != 0 || cows != 0){
				arr[index] = i;

		StringBuilder sb = new StringBuilder();
		for(int i = 0; i < 4; ++i){
			sb.append(arr[0]);

					if(newBulls ==4 && newCows == 0){

						changed = true;
						sb.setCharAt(i, (char)('0'+arr[j-1]));

				}
			}
		}
