import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

class Node(object):
	value = None
	nxt = None
	left = None
	right = None

	def __init__(self, value, nxt):
		self.value = value
		self.nxt = nxt
(n, m) = map(int, input().split())
a = list(map(int, input().split()))
sim = [a.copy()]
current = []
while len(current) != 1:
	current = []
	for j in range(0, len(sim[-1]), 2):
		if len(sim) % 2:
			current.append(sim[-1][j] | sim[-1][j + 1])
		else:
			current.append(sim[-1][j] ^ sim[-1][j + 1])
	sim.append(current)
sim[-1][0] = Node(sim[-1][0], None)
for i in range(len(sim) - 2, -1, -1):
	for j in range(len(sim[i])):
		sim[i][j] = Node(sim[i][j], sim[i + 1][j // 2])
for i in range(1, len(sim)):
	for j in range(len(sim[i])):
		sim[i][j].left = sim[i - 1][j * 2]
		sim[i][j].right = sim[i - 1][1 + j * 2]
for _ in range(m):
	(p, b) = map(int, input().split())
	sim[0][p - 1].value = b
	i = 1
	cur_node = (p - 1) // 2
	while i < len(sim):
		if i % 2:
			sim[i][cur_node].value = sim[i][cur_node].left.value | sim[i][cur_node].right.value
		else:
			sim[i][cur_node].value = sim[i][cur_node].left.value ^ sim[i][cur_node].right.value
		i += 1
		cur_node //= 2
	print(sim[-1][0].value)
