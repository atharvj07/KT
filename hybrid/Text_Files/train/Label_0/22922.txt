#include <bits/stdc++.h>
using namespace std;
int read() {
  int x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 3) + (x << 1) + (ch ^ 48);
    ch = getchar();
  }
  return x * f;
}
int n, m, x, y, ANS, siz[1000050];
int head[1000050], ver[1000050], nex[1000050], tot;
int f[200005][205], s[205][205], fac[205], ans[205];
void dfs(int u, int anc) {
  siz[u] = 1;
  f[u][0] = 2;
  for (int i = head[u]; i; i = nex[i]) {
    int v = ver[i];
    if (v == anc) continue;
    dfs(v, u);
    for (int j = m; j >= 1; --j) f[v][j] = (f[v][j] + f[v][j - 1]) % 1000000007;
    f[v][1] = (f[v][1] - 1 + 1000000007) % 1000000007;
    for (int j = 0; j <= m; ++j) f[0][j] = 0;
    for (int j = 0; j <= min(m, siz[u]); ++j)
      for (int k = 0; k <= min(m - j, siz[v]); ++k)
        (f[0][j + k] += (long long)f[u][j] * f[v][k] % 1000000007) %=
            1000000007;
    siz[u] += siz[v];
    for (int j = 0; j <= m; ++j) f[u][j] = f[0][j];
    for (int j = 0; j <= m; ++j)
      ans[j] = (ans[j] - f[v][j] + 1000000007) % 1000000007;
  }
  for (int i = 0; i <= m; ++i) ans[i] = (ans[i] + f[u][i]) % 1000000007;
}
int main() {
  n = read();
  m = read();
  for (int i = 1; i < n; ++i) {
    x = read();
    y = read();
    ++tot;
    nex[tot] = head[x];
    head[x] = tot;
    ver[tot] = y;
    ++tot;
    nex[tot] = head[y];
    head[y] = tot;
    ver[tot] = x;
  }
  dfs(1, 0);
  s[0][0] = fac[0] = 1;
  for (int i = 1; i <= m; ++i) fac[i] = (long long)fac[i - 1] * i % 1000000007;
  for (int i = 1; i <= m; ++i)
    for (int j = 1; j <= i; ++j)
      s[i][j] = ((long long)j * s[i - 1][j] % 1000000007 + s[i - 1][j - 1]) %
                1000000007;
  for (int i = 0; i <= m; ++i)
    (ANS += (long long)s[m][i] * ans[i] % 1000000007 * fac[i] % 1000000007) %=
        1000000007;
  printf("%d", ANS);
  return 0;
}
