#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define rep(i, j) for(int i=0; i < (int)(j); i++)
#define all(v) v.begin(),v.end()
template<class T> bool set_min(T &a, const T &b) { return a > b  ? a = b, true : false; }
template<class T> bool set_max(T &a, const T &b) { return a < b  ? a = b, true : false; }
template<class T> istream& operator >> (istream &is , vector<T> &v) { for(T &a : v) is >> a; return is; }
const ll INFL = 1LL << 60;

struct Edge { int to; ll len; };

class Solver {
  public:
    vector<vector<Edge>> E;

    tuple<ll, ll> solve2(int root, int prev) {
        ll sum = 0, maxi = 0;
        bool branch = true;
        for(Edge e : E[root]) {
            if(e.to == prev) continue;
            branch = false;
            auto ret = solve2(e.to, root);
            if(get<0>(ret) < 0) continue;
            set_max(maxi, get<1>(ret) + e.len);
            sum += get<0>(ret) + e.len;
        }
        return make_tuple(branch ? -INFL : sum, maxi);
    }
    
    bool solve() {
        int N; cin >> N;
        if(N == 0) return 0;
        vector<int> P(N - 1), D(N - 1); cin >> P >> D;
        E.resize(N);
        rep(i, N - 1) {
            E[i + 1].push_back(Edge{P[i] - 1, D[i]});
            E[P[i] - 1].push_back(Edge{i + 1, D[i]});
        }

        ll ans = INFL;
        rep(i, N) {
            auto ret = solve2(i, -1);
            set_min(ans, get<0>(ret) * 2 - get<1>(ret));
        }
        
        cout << ans + accumulate(all(D), 0LL) << endl;
        
        return 1;
    }
};

int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);
    while(true) {
        Solver s;
        if(!s.solve()) break;
    }
    return 0;
}
