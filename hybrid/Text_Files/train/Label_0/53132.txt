import java.io.*;
import java.util.*;

public class CF1242E extends PrintWriter {
	CF1242E() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1242E o = new CF1242E(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int[] aa = new int[n];
		int[][] vv = new int[n][];
		Integer[] ii = new Integer[n];
		int cnt = 0;
		for (int i = 0; i < n; i++) {
			int a = sc.nextInt();
			cnt += aa[i] = a;
			vv[i] = new int[a];
			ii[i] = i;
		}
		Arrays.sort(ii, (i, j) -> aa[j] - aa[i]);
		int[] qq = new int[cnt]; cnt = 0; int k = 0;
		for (int i = ii[0], j = 0; j < aa[i]; j++)
			qq[cnt++] = vv[i][j] = ++k;
		for (int h = 1; h < n; h++) {
			int i = ii[h];
			if (aa[i] > 3) {
				int a = aa[i];
				int c = h + 1 < n ? aa[ii[h + 1]] : 3;
				// cnt - m + a - m >= c
				// m <= (cnt + a - c) / 2
				int m = Math.min((cnt + a - c) / 2, a - 1);
				for (int j = 0; j < m - 1; j++)
					vv[i][j] = qq[--cnt];
				vv[i][m - 1] = qq[cnt - 1];
				for (int j = m; j < a - 1; j++)
					qq[cnt++] = vv[i][j] = ++k;
				vv[i][a - 1] = qq[0];
			} else if (cnt > 3) {
				vv[i][0] = qq[cnt - 1];
				vv[i][1] = qq[cnt - 2];
				vv[i][2] = qq[0];
				cnt--;
			} else {
				while (h < n) {
					i = ii[h];
					vv[i][0] = qq[0];
					vv[i][1] = qq[2];
					vv[i][2] = ++k;
					h++;
					if (h < n) {
						i = ii[h];
						vv[i][0] = qq[1];
						vv[i][1] = qq[2];
						vv[i][2] = k;
						qq[2] = k;
						h++;
					}
				}
				break;
			}
		}
		println(k);
		for (int i = 0; i < n; i++) {
			int a = aa[i];
			for (int j = 0; j < a; j++)
				print(vv[i][j] + " ");
			println();
		}
	}
}
