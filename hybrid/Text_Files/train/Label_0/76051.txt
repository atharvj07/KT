def inp():
	n = int(input())
	return [[int(x), False] for x in input().split()]

def main():
	struct = inp()
	cycles = []
	for s in struct:
		if not s[1]:
			s[1] = True
			cycleLen = 1
			rec = s[0] - 1
			while not struct[rec][1]:
				struct[rec][1] = True
				rec = struct[rec][0] - 1
				cycleLen += 1
			cycles.append(cycleLen)
	(fM, sM, s) = (0, 0, 0)
	for c in cycles:
		if c > fM:
			sM = fM
			fM = c
		elif c > sM:
			sM = c
		s += c ** 2
	s = s - fM ** 2 - sM ** 2 + (fM + sM) ** 2
	return s
answer = main()
print(answer)
