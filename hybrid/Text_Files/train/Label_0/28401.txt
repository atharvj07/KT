#include <iostream>
#include <set>
using namespace std;

using ll = long long;

const int MAX_N = 100010;

string s[MAX_N];

set<string> recorder, prev_recorder;

string str;

ll pow(ll a, ll b) {
    if (b == 0) return 1;
    else return a * pow(a, b - 1);
}

int main() {
    int n;
    cin >> n;

    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }

    prev_recorder.insert("");

    for (int len = 1; len <= 6; len++) {
        recorder.clear();
        for (int i = 0; i < n; i++) {
            for (int s_i = 0; s_i + len <= s[i].size(); s_i++) {
                recorder.insert(s[i].substr(s_i, len));
            }
        }
        if (recorder.size() != pow(26, len)) {
            for (char alpha = 'a'; alpha <= 'z'; alpha++) {
                for (string str : prev_recorder) {
                    string str2 = alpha + str;
                    if (recorder.find(str2) == recorder.end()) {
                        cout << str2 << endl;
                        return 0;
                    }
                }
            }
        } else {
            prev_recorder = recorder;
        }
    }

    return 0;
}
