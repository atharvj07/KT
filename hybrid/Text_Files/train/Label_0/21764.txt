#include <bits/stdc++.h>
using namespace std;
long long aa[100005][2], bb[100005];
int len = 0, l = 0;
void fx(long long m1, long long m2) {
  for (int j = len - 2; j >= 0; j--) {
    if (m1 < aa[j][0]) {
      bb[l++] = aa[j][1] - aa[j][0];
      aa[j][0] = m1;
      aa[j][1] = m2;
      len--;
    } else {
      if (m2 <= aa[j][1]) {
        aa[j + 1][0] = m1;
        aa[j + 1][1] = m2;
        break;
      } else {
        bb[l++] = aa[j][1] - m1;
        m1 = aa[j][0];
        aa[j][1] = m2;
        len--;
      }
    }
  }
}
int main() {
  int n, k, f = 0;
  long long a, b, m1 = 0, m2 = 0, ans = 0;
  cin >> n >> k >> a;
  for (int i = 1; i < n; i++) {
    cin >> b;
    if (b >= a) {
      if (f == 0) {
        f = 1;
        aa[len][0] = a;
      }
    } else {
      if (f) {
        f = 0;
        aa[len][1] = a;
        len++;
        m1 = aa[len - 1][0];
        m2 = aa[len - 1][1];
        fx(m1, m2);
      }
    }
    a = b;
  }
  if (aa[len][0] != 0 && f) {
    aa[len][1] = b;
    len++;
    m1 = aa[len - 1][0];
    m2 = aa[len - 1][1];
    fx(m1, m2);
  }
  for (int i = 0; i < len; i++) bb[l++] = aa[i][1] - aa[i][0];
  sort(bb, bb + l);
  for (int i = 0; i < min(l, k); i++) ans += bb[l - i - 1];
  cout << ans;
  return 0;
}
