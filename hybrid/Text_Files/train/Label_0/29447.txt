#include <iostream>
#include <algorithm>
#include <cstring>

using namespace std;

int n;
int arr[2001];

int main(){
    int t;
    cin >> t;
    while(t--){
        cin >> n;
        for(int i = 0; i < n; i++){
            cin >> arr[i];
        }
        for(int i = 0; i < n; i++){
            if(arr[i] % 2 == 0) cout << arr[i] << " ";
        }
        for(int i = 0; i < n; i++){
            if(arr[i] % 2 == 1) cout << arr[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
