n = int(input())
arr = list(map(int, input().split()))
if len(arr) == 1:
	print(arr[0] - 1)
elif len(arr) <= 50:
	arr.sort()
	ans = pow(10, 15)
	num = 1
	prev = pow(10, 15)
	while 1:
		curr = 0
		for i in range(n):
			curr += abs(arr[i] - pow(num, i))
		if curr > prev:
			break
		if curr < ans:
			ans = curr
		prev = curr
		num += 1
	print(ans)
else:
	print(sum(arr) - n)
