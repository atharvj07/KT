
import java.util.Scanner;
public class choc {
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int n,m,diff,div;
        n=sc.nextInt();
        m=sc.nextInt();
        int arr[]=new int[m];
        div=n/m;
        for(int i=0;i<m;i++)
            arr[i]=div;
        diff=n-(m*div);
        if(diff==0){
            for(int i=0;i<m;i++)
                System.out.print(arr[i]+" ");
        }
        else{
            for(int i=1;i<=diff;i++)
                arr[m-i]++;
            for(int i=0;i<m;i++)
                System.out.print(arr[i]+" ");
        }            
    }
}
