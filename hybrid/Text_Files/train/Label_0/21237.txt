#include <bits/stdc++.h>
using namespace std;
int cnt[1000];
int a[1000];
int main() {
  int i, j, n, ans;
  scanf("%d", &n);
  memset(cnt, 0, sizeof(cnt));
  ans = 0;
  for (i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }
  sort(a, a + n);
  for (i = 0; i < n; i++) {
    for (j = 0;; j++)
      if (cnt[j] <= a[i]) break;
    cnt[j]++;
    if (j + 1 > ans) ans = j + 1;
  }
  printf("%d\n", ans);
  return 0;
}
