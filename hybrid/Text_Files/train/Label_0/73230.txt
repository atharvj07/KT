def solve(arr, n, a):
	if a <= n // 2:
		gran = 2 * a - 1
		pt1 = arr[:gran]
		pt2 = arr[gran:]
		k = len(pt1)
		ans = 0
		for j in range(k):
			if pt1[j] == pt1[k - j - 1]:
				ans += pt1[j]
		ans += sum(pt2)
		return ans
	else:
		gran = 2 * (n - a) + 1
		pt1 = arr[:n - gran]
		pt2 = arr[n - gran:]
		k = len(pt2)
		ans = 0
		for j in range(k):
			if pt2[j] == pt2[k - j - 1]:
				ans += pt2[j]
		ans += sum(pt1)
		return ans
(n, a) = [int(x) for x in input().split()]
arr = [int(x) for x in input().split()]
print(solve(arr, n, a))
