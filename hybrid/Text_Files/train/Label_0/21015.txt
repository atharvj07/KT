#include<bits/stdc++.h>
using namespace std;
#define inf 1e17
typedef long long ll;
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	cout.tie(NULL);
	ll adj1[305][305]={0},adj2[305][305]={0};
	for(int i=0; i<=300; i++){
		for(int j=0; j<=300; j++){
			if(i!=j){
				adj1[i][j]=inf;
				adj2[i][j]=inf;
			}
	}
    }
	int n,m,l;
	cin>>n>>m>>l;
	for(int i=0; i<m; i++){
		ll a,b,c;
		cin>>a>>b>>c;
		adj1[a][b]=c;
		adj1[b][a]=c;
	}
	for(int k=1; k<=n; k++){
		for(int i=1; i<=n; i++){
			for(int j=1; j<=n; j++){
				adj1[i][j]=min(adj1[i][j],adj1[i][k]+adj1[k][j]);
			}
		}
	}
	for(int i=1; i<=n; i++){
		for(int j=1; j<=n; j++){
			if(adj1[i][j]<=l){
				adj2[i][j]=1;
			}
		}
	}
    	for(int k=1; k<=n; k++){
		for(int i=1; i<=n; i++){
			for(int j=1; j<=n; j++){
				adj2[i][j]=min(adj2[i][j],adj2[i][k]+adj2[k][j]);
			}
		}
	}
	int q;
	cin>>q;
	while(q--)
	{
		int a,b;
		cin>>a>>b;
		if(adj2[a][b]>=inf)
		 cout<<"-1\n";
		else
		 cout<<adj2[a][b]-1<<"\n";
	}
	return 0;
}
