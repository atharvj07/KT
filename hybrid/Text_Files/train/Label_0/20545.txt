#include<iostream>
#include<vector>
#include<string>
using namespace std;

int main(){
	string s;
	while(cin>>s,s!="#"){
		int step=0;
		int num=-1;
		for(int i=s.size()-1;i>=0;i--){
			
			if(s[i]=='n'){
				if(step==0)
					num=0;
				else
					num=num*2-90;
				
				step++;


			}
			else if(s[i]=='w'){
				if(step==0)
					num=90;
				else
					num=num*2+90;
				step++;
			}



		}
		for(;;){
			if(num%2==0 && step>1)
				num/=2,step--;
			else
				break;

		}
		if(step==1)
			cout<<num<<endl;
		else
			cout<<num<<"/"<<(1<<(step-1))<<endl;



	}
}
