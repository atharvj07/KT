(n, a, b, k) = map(int, input().split())
z = list(map(int, input().split()))
count = 0
ans = 0
for i in range(len(z)):
	if z[i] % (a + b) == 0:
		z[i] = a + b
	else:
		z[i] = z[i] % (a + b)
z.sort()
for i in range(len(z)):
	if z[i] <= a:
		ans = ans + 1
	else:
		z[i] = z[i] - a
		t = z[i] % a
		r = z[i] // a
		if t == 0:
			if k >= r:
				k = k - r
				ans = ans + 1
			else:
				break
		else:
			r = r + 1
			if k >= r:
				k = k - r
				ans = ans + 1
			else:
				break
print(ans)
