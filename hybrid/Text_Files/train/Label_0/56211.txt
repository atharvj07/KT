import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;


public class E {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        byte[][]a = new byte[n+1][n+1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                a[i][j] = (byte) (br.read()-48);
            }
            br.readLine();
        }
        int[]ver_changes = new int[n+1], hor_changes = new int[n+1], diognal_changes = new int[n+1];
        int ans = 0;
        for (int i = 1; i < n; i++) {
            for (int j = n; j > i; j--) {
                int k = ver_changes[i]+hor_changes[j];
                k %= 2;
                if (a[i][j] != k) {
                    ans++;
                    hor_changes[j]++;
                    ver_changes[i]++;
                    diognal_changes[i]++;
                    diognal_changes[j]++;
                }
            }
        }
        Arrays.fill(hor_changes, 0);
        Arrays.fill(ver_changes, 0);
        for (int i = n; i > 1; i--) {
            for (int j = 1; j < i; j++) {
                int k = ver_changes[i]+hor_changes[j];
                k %= 2;
                if (a[i][j] != k) {
                    ans++;
                    hor_changes[j]++;
                    ver_changes[i]++;
                    diognal_changes[i]++;
                    diognal_changes[j]++;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            int k = diognal_changes[i];
            k %= 2;
            if (a[i][i] != k)
                ans++;
        }
        System.out.println(ans);
    }
}
