import math
(n, s) = map(int, input().split())

class city:
	d = 0
	k = 0
cities = []

def sortcity(citysort):
	return citysort.d
for i in range(n):
	(x, y, z) = map(int, input().split())
	c = city()
	c.d = math.sqrt(pow(x, 2) + pow(y, 2))
	c.k = z
	cities.append(c)
if s >= 1000000:
	print(0)
else:
	cities.sort(key=sortcity)
	i = 0
	while s < 1000000 and i < n:
		s += cities[i].k
		i += 1
	if s < 1000000:
		print(-1)
	else:
		print(cities[i - 1].d)
