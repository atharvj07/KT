#define r(i,n) for(int i=0;i<n;i++)
using namespace std;
typedef pair<vector<int>,int>P;
typedef pair<int,int>P2;
typedef pair<P2,P2>P3;
int a[4][4],o;
int dx[]={0,1,0,-1};
int dy[]={1,0,-1,0};
vector<int>vp(16),tmp(16);
set<vector<int> >s;
vector<int> get(vector<int> &v,int y,int x){
	r(i,16)a[i/4][i%4]=v[i];
	a[y][x]++;
	if(a[y][x]==5){
	  int now=0;
	  queue<P3>q;
	  a[y][x]=0;
	  r(i,4)q.push(P3(P2(y,x),P2(i,0)));
	  while(!q.empty()){L:;
	  	P3 p;
	  	if(q.size()){p=q.front();q.pop();}
	  	else p=P3(P2(-1,0),P2(0,now+1));
	  	if(now!=p.second.second){
	  		r(i,4)r(j,4)if(a[i][j]>=5){
	  			a[i][j]=0;
	  		  r(k,4)q.push(P3(P2(i,j),P2(k,p.second.second)));
	  		}
	  		 now=p.second.second;
	  	}
	  	if(p.first.first==-1)continue;
	  	y=p.first.first+dy[p.second.first];
	  	x=p.first.second+dx[p.second.first];
	  	if(y<0||x<0||y>=4||x>=4)continue;
	  	if(a[y][x])a[y][x]++;
	  	else q.push(P3(P2(y,x),P2(p.second.first,p.second.second+1)));
	  }
	  r(i,4)r(j,4)if(a[i][j]>=5)goto L;
	}
	r(i,4)r(j,4)tmp[i*4+j]=a[i][j];
	return tmp;
}
bool ch(vector<int> &v){
	r(i,16)if(v[i])return 0;
	return 1;
}

	r(i,16)cin>>vp[i];
	queue<P>q;

		if(p.second==5){

			return 0;
		}
		vector<int>v=p.first;

				return 0;
			}
			if(!s.count(t)){
				s.insert(t);
				q.push(P(t,p.second+1));
			}
		}
