def inp(flag=0):
	if flag == 0:
		return list(map(int, input().strip().split(' ')))
	else:
		return int(input())
n = int(input())
s = inp()
ans = []
possible = True
for i in range(n):
	if s[i] % s[0] != 0:
		possible = False
		break
	ans.append(s[0])
	ans.append(s[i])
if possible:
	print(2 * n)
	print(*ans)
else:
	print(-1)
