from math import gcd

def readline():
	return map(int, input().split())

def lcm(a, b):
	return a * b // gcd(a, b)
x = 1
d = []
i = 2
while x <= 10 ** 9:
	x = lcm(x, i)
	d.append(x)
	i += 1
assert d[-1] > 10 ** 9

def solve():
	(n,) = readline()
	a = readline()
	return all((ai % di for (ai, di) in zip(a, d)))
t = int(input())
for __ in range(t):
	print('YES' if solve() else 'NO')
