from sys import stdin, stdout
import math, bisect, heapq
from collections import Counter, deque, defaultdict
L = lambda : list(map(int, stdin.readline().strip().split()))
I = lambda : int(stdin.readline().strip())
S = lambda : stdin.readline().strip()
C = lambda : stdin.readline().strip().split()

def pr(a):
	print(''.join(list(map(str, a))))

def check(a, i):
	if a[i - 1] < a[i] > a[i + 1]:
		return 1
	if a[i - 1] > a[i] < a[i + 1]:
		return 1
	return 0

def solve():
	n = I()
	a = L()
	ans = 0
	c = [0] * n
	for i in range(1, n - 1):
		s = 0
		if a[i - 1] < a[i] > a[i + 1]:
			s += 1
		if a[i - 1] > a[i] < a[i + 1]:
			s += 1
		c[i] = s
		ans += s
	s = ans
	for i in range(1, n - 1):
		temp = a[i]
		if i == 1 and i == n - 2:
			x = c[i]
			a[i] = a[i - 1]
			y = check(a, i)
			a[i] = a[i + 1]
			z = check(a, i)
			if y < x or z < x:
				ans = min(ans, s - x + z, s - x + y)
		elif i == 1:
			x = c[i] + c[i + 1]
			a[i] = a[i - 1]
			y = check(a, i) + check(a, i + 1)
			a[i] = a[i + 1]
			z = check(a, i) + check(a, i + 1)
			if y < x or z < x:
				ans = min(ans, min(s - (x - y), s - (x - z)))
		elif i == n - 2:
			x = c[i - 1] + c[i]
			a[i] = a[i - 1]
			y = check(a, i) + check(a, i - 1)
			a[i] = a[i + 1]
			z = check(a, i) + check(a, i - 1)
			if y < x or z < x:
				ans = min(ans, min(s - (x - y), s - (x - z)))
		else:
			x = c[i - 1] + c[i] + c[i + 1]
			a[i] = a[i - 1]
			y = check(a, i) + check(a, i + 1) + check(a, i - 1)
			a[i] = a[i + 1]
			z = check(a, i) + check(a, i + 1) + check(a, i - 1)
			if y < x or z < x:
				ans = min(ans, min(s - (x - y), s - (x - z)))
		a[i] = temp
	print(ans)
for _ in range(I()):
	solve()
