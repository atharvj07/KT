t = int(input())
for x in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	counter = 0
	for i in range(n):
		if a[i] % 2 == 0:
			counter += 1
	if counter == n // 2 + n % 2:
		counter = 0
		for i in range(n):
			if a[i] % 2 == 0 and i % 2 == 0:
				counter += 1
		print(n // 2 + n % 2 - counter)
	else:
		print(-1)
