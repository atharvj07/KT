import java.io.*;
import java.util.*;

public class CF761F {
	static class R {
		int a, b, c, d, e;
		R(int a, int b, int c, int d, int e) {
			this.a = a; this.b = b; this.c = c; this.d = d; this.e = e;
		}
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());
		int l = 26;
		int k = Integer.parseInt(st.nextToken());
		char[][] ee = new char[n][m];
		for (int i = 0; i < n; i++)
			br.readLine().getChars(0, m, ee[i], 0);
		int[][][] aa = new int[n][m][l];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) {
				int e = ee[i][j] - 'a';
				aa[i][j][e] = 1;
			}
		int[][][] hh = new int[n + 1][m + 1][l + 1];
		R[] rr = new R[k];
		for (int i = 0; i < k; i++) {
			st = new StringTokenizer(br.readLine());
			int a = Integer.parseInt(st.nextToken()) - 1;
			int b = Integer.parseInt(st.nextToken()) - 1;
			int c = Integer.parseInt(st.nextToken()) - 1;
			int d = Integer.parseInt(st.nextToken()) - 1;
			int e = st.nextToken().charAt(0) - 'a';
			rr[i] = new R(a, b, c, d, e);
			hh[a][b][e]++;
			hh[c + 1][b][e]--;
			hh[a][d + 1][e]--;
			hh[c + 1][d + 1][e]++;
			hh[a][b][l]++;
			hh[c + 1][b][l]--;
			hh[a][d + 1][l]--;
			hh[c + 1][d + 1][l]++;
		}
		for (int i = 0; i < n; i++)
			for (int j = 1; j < m; j++)
				for (int e = 0; e <= l; e++)
					hh[i][j][e] += hh[i][j - 1][e];
		for (int i = 1; i < n; i++)
			for (int j = 0; j < m; j++)
				for (int e = 0; e <= l; e++)
					hh[i][j][e] += hh[i - 1][j][e];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) {
				int k_ = k - hh[i][j][l];
				for (int e = 0; e < l; e++) {
					aa[i][j][e] *= k_;
					aa[i][j][e] += hh[i][j][e];
				}
			}
		int[][][] cc = new int[n][m][l];
		int[] pp = new int[l];
		int[] qq = new int[l];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) {
				Arrays.fill(pp, 0);
				Arrays.fill(qq, 0);
				pp[0] = aa[i][j][0];
				for (int e = 1; e < l; e++)
					pp[e] = pp[e - 1] + aa[i][j][e];
				for (int e = 1; e < l; e++)
					pp[e] += pp[e - 1];
				qq[l - 1] = aa[i][j][l - 1];
				for (int e = l - 2; e >= 0; e--)
					qq[e] = qq[e + 1] + aa[i][j][e];
				for (int e = l - 2; e >= 0; e--)
					qq[e] += qq[e + 1];
				for (int e = 0; e < l; e++) {
					int c = 0;
					if (e - 1 >= 0)
						c += pp[e - 1];
					if (e + 1 < l)
						c += qq[e + 1];
					cc[i][j][e] = c;
				}
			}
		long[][][] ss = new long[n + 1][m + 1][l + 1];
		for (int i = n - 1; i >= 0; i--)
			for (int j = m - 1; j >= 0; j--) {
				for (int e = 0; e < l; e++) {
					long sum = cc[i][j][e];
					sum += ss[i + 1][j][e];
					sum += ss[i][j + 1][e];
					sum -= ss[i + 1][j + 1][e];
					ss[i][j][e] = sum;
				}
				int e = ee[i][j] - 'a';
				long sum = cc[i][j][e];
				sum += ss[i + 1][j][l];
				sum += ss[i][j + 1][l];
				sum -= ss[i + 1][j + 1][l];
				ss[i][j][l] = sum;
			}
		long min = Long.MAX_VALUE;
		for (int i = 0; i < k; i++) {
			R r = rr[i];
			int a = r.a, b = r.b, c = r.c, d = r.d, e = r.e;
			long sum = ss[0][0][l];
			sum += ss[a][b][e] - ss[a][b][l];
			sum -= ss[c + 1][b][e] - ss[c + 1][b][l];
			sum -= ss[a][d + 1][e] - ss[a][d + 1][l];
			sum += ss[c + 1][d + 1][e] - ss[c + 1][d + 1][l];
			min = Math.min(min, sum);
		}
		System.out.println(min);
	}
}
