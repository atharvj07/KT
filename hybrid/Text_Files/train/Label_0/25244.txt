#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 100 * 1000 + 10;
const int LOG = 20 + 5;
map<pair<int, int>, int> e;
vector<int> adj[MAX_N];
int par[MAX_N][LOG];
int inc[MAX_N];
int dp[MAX_N];
int h[MAX_N];
int dfs(int u) {
  int cnt = inc[u];
  for (auto v : adj[u])
    if (v != par[u][0]) {
      int x = dfs(v);
      dp[e[{u, v}]] = x, cnt += x;
    }
  return cnt;
}
void lca_dfs(int u, int dad, int l) {
  par[u][0] = dad, h[u] = l;
  for (int i = 1; i < LOG; i++) par[u][i] = par[par[u][i - 1]][i - 1];
  for (auto v : adj[u])
    if (v != dad) lca_dfs(v, u, l + 1);
}
int get_par(int u, int x) {
  for (int i = 0; i < LOG; i++)
    if (x >> i & 1) u = par[u][i];
  return u;
}
int lca(int u, int v) {
  if (h[u] > h[v]) swap(u, v);
  v = get_par(v, h[v] - h[u]);
  if (u == v) return u;
  for (int i = LOG - 1; ~i && u != v; i--)
    if (par[u][i] != par[v][i]) u = par[u][i], v = par[v][i];
  return par[u][0];
}
int main() {
  ios_base ::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  int n;
  cin >> n;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
    e[{u, v}] = e[{v, u}] = i;
  }
  lca_dfs(0, 0, 0);
  int k;
  cin >> k;
  for (int i = 0; i < k; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    inc[u]++, inc[v]++, inc[lca(u, v)] -= 2;
  }
  dfs(0);
  for (int i = 1; i < n; i++) cout << dp[i] << ' ';
  return 0;
}
