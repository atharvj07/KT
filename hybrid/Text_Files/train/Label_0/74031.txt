n = int(input())
s = input()
r = s.count('R')
g = s.count('G')
b = s.count('B')
if r >= 1 and g >= 1 and (b >= 1):
	print('BGR')
elif r + g == 0:
	print('B')
elif b + g == 0:
	print('R')
elif b + r == 0:
	print('G')
elif r == 0:
	if b + g == 2:
		print('R')
	elif b == 1 and g > 1:
		print('BR')
	elif g == 1 and b > 1:
		print('GR')
	else:
		print('BGR')
elif b == 0:
	if r + g == 2:
		print('B')
	elif r == 1 and g > 1:
		print('BR')
	elif g == 1 and r > 1:
		print('BG')
	else:
		print('BGR')
elif g == 0:
	if b + r == 2:
		print('G')
	elif b == 1 and r > 1:
		print('BG')
	elif r == 1 and b > 1:
		print('GR')
	else:
		print('BGR')
