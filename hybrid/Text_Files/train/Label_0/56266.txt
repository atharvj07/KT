import java.util.*;

public class D327 {
	
	public static void main(String[] args) {
		
		Scanner qwe = new Scanner(System.in);
		
		double x1 = qwe.nextDouble();
		double y1 = qwe.nextDouble();
		
		double x2 = qwe.nextDouble();
		double y2 = qwe.nextDouble();
		
		double v = qwe.nextDouble();
		double t = qwe.nextDouble();
		
		double wx = qwe.nextDouble();
		double wy = qwe.nextDouble();
		
		double wx2 = qwe.nextDouble();
		double wy2 = qwe.nextDouble();
		
		double min = 0;
		double max = Long.MAX_VALUE;
		
		for(int i =0; i < 150; i++){
			double med = (min+max)/2;
			double xd = x1+med*wx-x2;
			double yd = y1+med*wy-y2;;
			if(med > t){
				xd = x1+t*wx+(med-t)*wx2-x2;
				yd = y1+t*wy+(med-t)*wy2-y2;
			}
			double dis = Math.sqrt(xd*xd+yd*yd);
			
			double ctime = dis/v;
			
			//System.out.println(ctime + " " + med);
			
			if(ctime > med){
				min = med;
			}else max = med;
			
		}
		
		System.out.println(min);
		
		
		qwe.close();
	}


}
