def steps(arr):
	index_arr = []
	for (i, a) in enumerate(arr):
		index_arr.append((a, i))
	index_arr = sorted(index_arr, key=lambda x: x[0])
	steps = 0
	for (a, i) in index_arr[1:]:
		steps += abs(i - index_arr[a - 2][1])
	return steps
_ = input()
values = input()
value_list = []
for v in values.split(' '):
	value_list.append(int(v))
print(steps(value_list))
