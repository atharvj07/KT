def main():
	eps = 10 ** (-7)
	(k, q) = readIntArr()

	def rowFactory():
		row = [0 for _ in range(k + 1)]
		return row
	dp = [rowFactory()]
	dp[0][0] = 1
	while dp[-1][-1] <= 0.5:
		newRow = rowFactory()
		for j in range(1, k + 1):
			newRow[j] += dp[-1][j] * j / k + dp[-1][j - 1] * (k - (j - 1)) / k
		dp.append(newRow)
	n = len(dp)
	for _ in range(q):
		p = int(input())
		b = n + 1
		nTurns = 0
		while b > 0:
			while nTurns + b < n and dp[nTurns + b][k] < p / 2000:
				nTurns += b
			b //= 2
		nTurns += 1
		print(nTurns)
	return
