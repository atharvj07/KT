#include <bits/stdc++.h>
using namespace std;
const int MAX = 2000006;
int n, a, occ[MAX], ans, mx;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &a);
    occ[a]++;
  }
  for (int i = 0; i < MAX; i++) {
    int dd = occ[i] / 2;
    occ[i] %= 2;
    if (occ[i]) {
      ans++;
    }
    occ[i + 1] += dd;
  }
  cout << ans << endl;
  return 0;
}
