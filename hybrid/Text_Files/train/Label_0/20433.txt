#include <bits/stdc++.h>

using namespace std;

const int nmax = 3000 + 10;

int n;
int a[nmax];

vector < int > g[nmax];

bool win(int node) {
    for (auto &it : g[node])
        if (!win(it)) return true;
    return false;
}

int main()
{
    scanf("%d", &n);
    for (int i = 1; i <= n; ++i)
        scanf("%d", &a[i]);
    for (int i = 1; i < n; ++i) {
        int x, y;
        scanf("%d %d", &x, &y);

        if (a[x] > a[y]) g[x].push_back(y);
        if (a[y] > a[x]) g[y].push_back(x);
    }

    for (int i = 1; i <= n; ++i)
        if (win(i)) printf("%d ", i);

    return 0;
}
