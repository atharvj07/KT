def rgb(n, k, s):
	(t, result) = ('RGB', list())
	for i in range(n - k + 1):
		for j in range(3):
			count = 0
			for z in range(k):
				if s[i + z] != t[(z + j) % 3]:
					count += 1
			result.append(count)
	return min(result)
q = int(input())
lst = list()
for y in range(q):
	(N, K) = [int(w) for w in input().split()]
	v = input()
	lst.append(rgb(N, K, v))
for elem in lst:
	print(elem)
