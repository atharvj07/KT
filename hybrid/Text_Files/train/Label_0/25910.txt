#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  vector<vector<int>> graph(n, vector<int>());
  vector<int> out(n, 0);
  for (int i = 0; i < m; i++) {
    int a, b;
    scanf("%d%d", &a, &b);
    graph[b - 1].push_back(a - 1);
    out[a - 1]++;
  }
  priority_queue<int> pq;
  for (int i = 0; i < n; i++) {
    if (out[i] == 0) pq.push(i);
  }
  vector<int> ans(n);
  for (int i = n; i >= 1; i--) {
    int u = pq.top();
    pq.pop();
    ans[u] = i;
    for (int v : graph[u]) {
      out[v]--;
      if (out[v] == 0) pq.push(v);
    }
  }
  for (int i = 0; i < n; i++) {
    printf("%d%c", ans[i], i == n - 1 ? '\n' : ' ');
  }
  return 0;
}
