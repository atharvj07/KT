#include<cstdio>
#include<cstring>
#include<algorithm>

using namespace std;

typedef long long ll;

ll mod=1000000007;
int n,r;
int a[100001];
int s[51],t[51];
int bit[100001];

ll sum(int i){
	ll s=0;
	while(i>0){
		s+=bit[i];
		i-= i & -i;
	}
	return s;
}

void add(int i){
	while(i<=n){
		bit[i]+=1;
		i+= i & -i;
	}
}

int main(void){
	while(1){
		scanf("%d",&n);
		if(n==0)break;
		memset(bit,0,sizeof(bit));
		for(int i=1;i<=n;i++)a[i]=i;
		scanf("%d",&r);
		for(int i=0;i<r;i++){
			scanf("%d%d",&s[i],&t[i]);
			swap(a[s[i]],a[t[i]]);
		}
		ll res=0;
		ll p2=1;
		for(int i=n;i>=1;i--){
			res=(res+sum(a[i])*p2)%mod;
			add(a[i]);
			p2=p2*(n-i+1)%mod;
		}
		printf("%lld\n",res);
	}
	return 0;
}
