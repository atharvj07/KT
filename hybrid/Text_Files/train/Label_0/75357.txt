from sys import *
from math import *
from bisect import *
n = int(stdin.readline())
a = list(map(int, stdin.readline().split()))
b = list(map(int, stdin.readline().split()))
m = []
c = []
for i in range(n):
	x = []
	l = max(b) + 1
	k = -1
	for j in range(i + 1, n):
		if a[j] > a[i]:
			x.append(j)
			if b[j] < l:
				l = b[j]
				k = j
	c.append(k)
	m.append(x)
ans = sum(b) + 1
for i in range(n - 1):
	x = b[i]
	for j in range(len(m[i])):
		y = b[m[i][j]]
		if c[m[i][j]] != -1:
			z = b[c[m[i][j]]]
			if x + y + z < ans:
				ans = x + y + z
if ans == sum(b) + 1:
	print(-1)
else:
	print(ans)
