def sortA(inputStr):
	return len(inputStr)
(n, k) = map(int, input().split())
pas = []
s1 = 0
s2 = 0
for i in range(n):
	pas.append(input())
pas.sort(key=sortA)
ok_pass = input()
i = 0
while i < n and len(pas[i]) <= len(ok_pass):
	if len(pas[i]) == len(ok_pass) and s1 == 0:
		s1 = s2 + 1
	if (i + 1) % k == 0 and i + 1 < n and (len(ok_pass) >= len(pas[i + 1])):
		s2 += 5
	s2 += 1
	i += 1
print(s1, s2)
