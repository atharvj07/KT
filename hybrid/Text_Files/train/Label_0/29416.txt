#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, c, d;
  scanf("%d %d", &a, &b);
  scanf("%d %d", &c, &d);
  const int mn = min(min(a, b), min(c, d));
  const int mx = max(max(a, b), max(c, d));
  double ans = 0.50 * (mx - mn);
  for (auto const sa : {-1, 1}) {
    for (auto const sb : {-1, 1}) {
      for (auto const sc : {-1, 1}) {
        for (auto const sd : {-1, 1}) {
          const long long u = sa * sd - sb * sc;
          const long long v = 0ll + sa * d + sd * a - sb * c - sc * b;
          const long long w = 1ll * a * d - 1ll * b * c;
          if (u) {
            const long long d = v * v - 4.0 * u * w;
            if (d >= 0) {
              const double cur1 = 0.50 * (-v - sqrt(d)) / u;
              ans = min(ans, fabs(cur1));
              const double cur2 = 0.50 * (-v + sqrt(d)) / u;
              ans = min(ans, fabs(cur2));
            }
          } else {
            if (v) {
              const double cur = -1.0 * w / v;
              ans = min(ans, fabs(cur));
            }
          }
        }
      }
    }
  }
  printf("%.16lf", ans);
  return 0;
}
