import java.io.*;
import java.util.*;
public class CFB
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long L = sc.nextLong(), R = sc.nextLong(), X = sc.nextLong();
        int a[] = new int[N];
        int count = 0;
        for(int i=0;i<N;i++)
            a[i] = sc.nextInt();
        for (int i = 0; i < (int) Math.pow(2, N); i++) {
            int subset[] = new int[N], copy = i, index = N - 1;
            long cost = 0, min=Long.MAX_VALUE, max = Long.MIN_VALUE;
            while (copy > 0) {
                subset[index--] = copy % 2;
                copy /= 2;
            }
            for (int j = 0; j < N; j++) {
                if (subset[j] == 1) {
                    if(a[j]> max)
                        max=a[j];
                    if(a[j]<min)
                        min=a[j];
                    cost += a[j];
                }
            }
            if(cost>=L && cost<=R && max-min>=X)
                count++;
        }
        System.out.println(count);
    }
}
