#include <bits/stdc++.h>
using namespace std;
struct Edge {
  int t;
  Edge* ne;
};
const int maxn = 200003;
vector<vector<int> > sccr;
int n, m, h, u[maxn], c[maxn][2];
int cscc, bel[maxn], dfn[maxn], low[maxn], dfi, ist[maxn];
int stc[maxn], tst;
Edge epool[maxn * 2], *ebuf(epool), *head[maxn];
inline void addEdge(int u, int v) {
  ebuf->t = v;
  ebuf->ne = head[u];
  head[u] = ebuf++;
}
void DFS(int u) {
  dfn[u] = low[u] = ++dfi;
  stc[tst++] = u;
  ist[u] = 1;
  for (Edge* e = head[u]; e; e = e->ne) {
    if (!dfn[e->t]) {
      DFS(e->t);
      low[u] = min(low[u], low[e->t]);
    } else if (ist[e->t]) {
      low[u] = min(low[u], dfn[e->t]);
    }
  }
  if (low[u] == dfn[u]) {
    vector<int> cc;
    do {
      int v(stc[--tst]);
      ist[v] = 0;
      cc.push_back(v);
      bel[v] = cscc;
    } while (stc[tst] != u);
    ++cscc;
    sccr.push_back(cc);
  }
}
int tarjan() {
  memset(low, 0, sizeof(low));
  memset(dfn, 0, sizeof(dfn));
  memset(ist, 0, sizeof(ist));
  tst = 0;
  dfi = 0;
  cscc = 0;
  for (int i = 1; i <= n; ++i) {
    if (!dfn[i]) {
      DFS(i);
    }
  }
  int ans(0x3f3f3f3f);
  vector<vector<int> >::iterator ians;
  for (vector<vector<int> >::iterator it = sccr.begin(); it != sccr.end();
       ++it) {
    int cnt(0), fi(1);
    for (vector<int>::iterator iu = it->begin(); iu != it->end() && fi; ++iu) {
      ++cnt;
      for (Edge* e = head[*iu]; e; e = e->ne) {
        if (bel[e->t] != bel[*iu]) {
          fi = 0;
          break;
        }
      }
    }
    if (fi && ans > cnt) {
      ans = cnt;
      ians = it;
    }
  }
  printf("%d\n", ans);
  for (vector<int>::iterator iu = ians->begin(); iu != ians->end(); ++iu) {
    printf("%d ", *iu);
  }
  putchar(10);
  return 1;
}
int main() {
  memset(head, 0, sizeof(head));
  scanf("%d%d%d", &n, &m, &h);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", u + i);
  }
  for (int i = 0; i < m; ++i) {
    int a, b;
    scanf("%d%d", &a, &b);
    if ((u[a] - u[b] + h) % h == 1) {
      addEdge(b, a);
    }
    if ((u[b] - u[a] + h) % h == 1) {
      addEdge(a, b);
    }
    c[i][1] = a, c[i][0] = b;
  }
  if (!tarjan()) {
    puts("-1");
  }
}
