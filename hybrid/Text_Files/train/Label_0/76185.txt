import heapq
(n, m) = map(int, input().split())
x = [int(o) for o in input().split()]
c = 0
i = 1
j = 1
k = 0
s = set(x)
h = []
ans = []
for i in range(n):
	if x[i] - 1 not in s:
		heapq.heappush(h, [1, -1, x[i] - 1])
	if x[i] + 1 not in s:
		heapq.heappush(h, [1, 1, x[i] + 1])
while c != m:
	lol = heapq.heappop(h)
	if lol[1] > 0:
		if lol[2] not in s:
			s.add(lol[2])
			ans.append(lol[2])
			c += 1
			k += lol[0]
			lol[1] += 1
			lol[0] += 1
			lol[2] += 1
			if lol[2] not in s:
				heapq.heappush(h, lol[:])
	elif lol[2] not in s:
		s.add(lol[2])
		ans.append(lol[2])
		c += 1
		k += lol[0]
		lol[1] -= 1
		lol[0] += 1
		lol[2] -= 1
		if lol[2] not in s:
			heapq.heappush(h, lol[:])
print(k)
print(*ans)
