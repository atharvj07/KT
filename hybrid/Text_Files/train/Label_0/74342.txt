import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
from sys import stdout
print = stdout.write
from collections import Counter
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
d = []
t = 0
for i in range(n):
	try:
		d.append(b[i] * 10000000000.0 / a[i])
	except:
		if b[i] == 0:
			t += 1
e = Counter(d)
try:
	t += max(e.values())
	print(str(t))
except:
	print(str(t))
