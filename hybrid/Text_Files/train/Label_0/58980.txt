import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class B897 {
	
	private void solve() throws Exception {
		long n = nextInt();
		long p = nextInt();
		LinkedList<Long> valsList = new LinkedList<Long>();
		for(int i = 1; i <= 100000; i++){
			String str = Integer.toString(i);
			String inv = (new StringBuilder(str)).reverse().toString();
			String cur;
			if(i%10 != 0) cur = inv+str;
			else cur = str+inv;
			valsList.addLast(Long.parseLong(cur));
		}
		valsList.toArray(new Long[0]);
		Long[] vals = valsList.toArray(new Long[0]);
		Arrays.sort(vals);
		long res = 0;
		for(int i = 0; i < n; i++) res = (res+vals[i])%p;
		out.println(res);
	}
	
	public static void main(String[] args) {
		new B897().run();
	}

	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer tokenizer;

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			tokenizer = null;
			out = new PrintWriter(System.out);
			solve();
			in.close();
			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	private int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	private float nextFloat() throws IOException {
		return Float.parseFloat(nextToken());
	}

	private String nextLine() throws IOException {
		return new String(in.readLine());
	}

	private String nextToken() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			tokenizer = new StringTokenizer(in.readLine());
		}
		return tokenizer.nextToken();
	}

}
