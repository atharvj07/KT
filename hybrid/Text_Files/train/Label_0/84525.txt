n = int(input())
arr = []
for i in range(n):
	arr.append(input())
for i in range(len(arr)):
	if arr[i][:2] == 'OO':
		arr[i] = '++' + arr[i][2:]
		break
	if arr[i][3:] == 'OO':
		arr[i] = arr[i][:3] + '++'
		break
for i in arr:
	if i[:2] == '++' or i[3:] == '++':
		print('YES')
		for j in arr:
			print(j)
		exit()
print('NO')
