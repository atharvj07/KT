import java.util.*;
import java.io.*;

public class A {

	
	static int maxN=5032108,maxDiv=7;
	static 	int []numDiv;
	static int [][]primeDiv;
	static int []getDiv(int x)
	{
		int n=1<<(numDiv[x]);
		
		int []ans=new int [n+1];
		
		for(int msk=0;msk<n;msk++)
		{
			int div=1;
			for(int j=0;j<numDiv[x];j++)
				if((msk & (1<<j)) !=0)
					div*=primeDiv[j][x];
			ans[msk]=div;
		}
		ans[ans.length-1]=1;
		return ans;
	}
	public static void main(String[] args) throws NumberFormatException, IOException {
		Scanner sc = new Scanner();
		PrintWriter out = new PrintWriter(System.out);
		int n=sc.nextInt(),q=sc.nextInt();
		int []a=new int [n];
		
		boolean []isPrime = new boolean[maxN];
		int []distinct=new int [maxN];
		for(int i=0;i<maxN;i++)
			distinct[i]=i;
		
		Arrays.fill(isPrime, true);
	
		primeDiv= new int [maxDiv][maxN];
		numDiv=new int [maxN];
	
		for(int i=2;i<maxN;i++)
			if(isPrime[i])
			{
				
				primeDiv[numDiv[i]++][i]=i;
				for(int j=i+i;j<maxN;j+=i)
				{
					int x=distinct[j];
					long sq=i*1L*i;
					while(x%sq==0)
						x/=sq;
					distinct[j]=x;
					if(x%i==0)
					{
						primeDiv[numDiv[j]++][j]=i;
						
					}
					isPrime[j]=false;
				}
			}
	
	
		
		for(int i=0;i<n;i++)
		{
			int x=sc.nextInt();
			int y=1;
			for(int j=0;j<numDiv[x];j++)
				y=y*primeDiv[j][x];
			a[i]=y;
			
		}
		
		// dp  [dis][val] = the rightmost i such that a[i] = val * x where x can be represented as the product of 
		// dis distinct primes
		int [][]dp=new int [maxDiv+1][maxN];
		for(int []x:dp)
			Arrays.fill(x, -1);
		// best [D] = the righmost l such that the ans for [l,i] = D
		int []print=new int [q];
		int []best=new int [15];
		Arrays.fill(best, -5);
		query[] queries = new query[q];
		for(int i=0;i<q;i++)
			queries[i]=new query(sc.nextInt()-1,sc.nextInt()-1,i);
		Arrays.sort(queries, (x,y)->x.r-y.r);
		int i=0;
		for(query query:queries)
		{
			int l=query.l,r=query.r;
			
			while(i<=r)
			{
				int x=a[i];
				int []divisors=getDiv(x);
				for(int val:divisors)
				{
					int dis=numDiv[x/val];
					for(int k=0;k<=maxDiv;k++)
						best[k+dis]=Math.max(best[k+dis], dp[k][val]);
				}
				for(int val:divisors)
				{
					int dis=numDiv[x/val];
					dp[dis][val]=i;
				}
				i++;
			}
			
			int ans=-1;
			for(int j=0;j<best.length;j++)
				if(best[j]>=l)
				{
					ans=j;
					break;
				}
			print[query.idx]=ans;
		}
		for(int x:print)
			out.println(x);
		out.close();
	}
	static class query{
		int l,r,idx;
		query(int a,int b,int i){
			l=a;
			r=b;
			idx=i;
		}
		
	}
	
	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		public Scanner(InputStream is) {
			br = new BufferedReader(new InputStreamReader(is));
		}
		public Scanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}
		public Scanner(String fileName) throws FileNotFoundException {
			br = new BufferedReader(new FileReader(fileName));
		}

		String next() throws IOException {
			while (st == null || !st.hasMoreElements()) {
				st = new StringTokenizer(br.readLine());
			}
			return st.nextToken();
		}

		int nextInt() throws NumberFormatException, IOException {
			return Integer.parseInt(next());
		}
	}

}
