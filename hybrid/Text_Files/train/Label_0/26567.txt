#include <bits/stdc++.h>
using namespace std;
int main() {
  int i = 0, j = 0, cs = 0, in;
  int n;
  cin >> n;
  vector<int> v;
  int sum = 0, s = 0;
  for (int i = 1; i < n; i++) {
    cout << "? " << i << " " << n << "\n";
    int x;
    cin >> x;
    if (i > 1) {
      v.push_back(sum - x - s);
      s += v.back();
    } else
      sum = x;
  }
  cout << "? " << 1 << " " << n - 1 << "\n";
  int x;
  cin >> x;
  cout << "! ";
  for (int i = 0; i < v.size(); i++) cout << v[i] << " ";
  cout << x - s << " " << sum - x << "\n";
  return 0;
}
