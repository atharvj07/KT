#include<bits/stdc++.h>
using namespace std;
const int N=200005;
char s[N];
long long ans,a[N],p[N];
int main()
{
	scanf("%s",s+1);
	int n=strlen(s+1),t=0;
	a[t]++;
	p[n+1]=1;
	for(int i=n;i>=1;i--)
	{
		int x=s[i]-'0';
		p[i]=p[i+1]*10%2019;
		t=(t+x*p[i])%2019;
		a[t]++;
	}
	for(int i=0;i<2019;i++)
		ans+=a[i]*(a[i]-1)/2;
	cout<<ans<<endl;
	return 0;
}
