def solve(ls):
	n = len(ls)
	k = (n - 1) // 2
	s = 0
	for i in range(k + 1):
		s += ls[2 * i]
	s_max = s
	for i in range(n):
		s -= ls[2 * i % n]
		s += ls[(2 * i + 1) % n]
		s_max = max(s_max, s)
	return s_max

def main(istr, ostr):
	n = int(istr.readline())
	ls = list(map(int, istr.readline().split()))
	print(solve(ls), file=ostr)
import sys
main(sys.stdin, sys.stdout)
