def dfn():
	(r, c) = map(int, input().split())
	ar = []
	if r < 3 or c < 3:
		return -1
	mns = -1
	ck = 0
	pvl = -1
	hu = ar.append
	oin = -1
	for i in range(r):
		tp = input()
		hu(tp)
		k = tp.count('*')
		if k == 1:
			if pvl == -1:
				oin = tp.index('*')
				if oin == 0 or oin == c - 1:
					return -10
			elif pvl != i - 1 or tp[oin] != '*':
				return -2
			pvl = i
		elif k == 2:
			return -3
		elif k > 2:
			if pvl == -1 or i == c - 1:
				return -9
			if ck == 1:
				return -4
			n = tp.index('*')
			while n < c and tp[n] == '*':
				n += 1
			for p in range(n + 1, c):
				if tp[p] == '*':
					return -5
			ck = 1
			pvl = i
			mns = i
	if mns == -1 or oin == -1 or ar[mns][oin + 1] != '*' or (ar[mns + 1][oin] != '*') or (ar[mns - 1][oin] != '*') or (ar[mns][oin - 1] != '*'):
		return -8
	return 0
jn = 'YES' if dfn() == 0 else 'NO'
print(jn)
