n = int(input())
a = [int(i) for i in input().split()]
t1 = []
alert = 0
record = 0
for i in range(n + 1):
	if a[i] != 1:
		if alert == 1:
			t1.append(0)
			record = i
			break
		else:
			alert = 1
	else:
		alert = 0
if t1 == []:
	print('perfect')
	exit(0)
print('ambiguous')
index = 1
for x in a[1:]:
	for i in range(x):
		t1.append(index)
	index += x
t2 = t1[:]
change = sum(a[:record])
i = t1.index(change)
t2[i] -= 1
print(*t1)
print(*t2)
