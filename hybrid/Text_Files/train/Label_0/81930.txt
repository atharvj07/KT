import os
import sys
import io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
GANS = []

def print(*args):
	global GANS
	for i in args:
		GANS.append(f'{i}'.encode())
t = int(input())
for _ in range(t):
	(n, q) = map(int, input().split())
	li = [-1] + [int(i) for i in input().split()] + [-1]
	if n == 1:
		print(li[1])
		for qi in range(q):
			trash = input()
			print(li[1])
		continue
	if n == 2:
		print(max(li[2], li[1]))
		for qi in range(q):
			trash = input()
			print(max(li[2], li[1]))
		continue
	hi = li[0]
	status = 1
	lo = li[0]
	ans = 0
	for i in li[1:]:
		if status:
			if i > hi:
				hi = i
			else:
				status = 0
				lo = i
		elif i < lo:
			lo = i
		else:
			status = 1
			ans += hi - lo
			hi = i
	ans += hi
	print(ans)
	for qi in range(q):
		(u, v) = map(int, input().split())
		if u == v:
			print(ans)
			continue
		da = max(li[u - 1], li[u + 1])
		xiao = min(li[u - 1], li[u + 1])
		ans -= max(li[u] - da, 0)
		ans -= max(xiao - li[u], 0)
		ans += max(li[v] - da, 0)
		ans += max(xiao - li[v], 0)
		tmp = li[u]
		li[u] = li[v]
		da = max(li[v - 1], li[v + 1])
		xiao = min(li[v - 1], li[v + 1])
		ans -= max(li[v] - da, 0)
		ans -= max(xiao - li[v], 0)
		ans += max(tmp - da, 0)
		ans += max(xiao - tmp, 0)
		li[v] = tmp
		print(ans)
os.write(1, b'\n'.join(GANS) + b'\n')
