import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class C {

	public static void main(String[] ar) throws IOException {
		BufferedReader in=new BufferedReader(new InputStreamReader(System.in));
		int n=Integer.parseInt(in.readLine());
		int closedCount[]=new int[500005];
		int openCount[]=new int[500005];
		for(int i=0;i<n;i++) {
			char[] c=in.readLine().toCharArray();
			char stack[]=new char[c.length];
			int pos=-1;
			for(int j=0;j<c.length;j++) {
				if(pos==-1) {
					pos++;
					stack[pos]=c[j];
				}else {
					if(c[j]==')' && stack[pos]=='(') {
						pos--;
					}else {
						pos++;
						stack[pos]=c[j];
					}
				}
			}
			int close=0,open=0;
			while(pos>=0) {
				if(stack[pos--]=='(') {
					open++;
				}else {
					close++;
				}
			}
			
			if(close>0 && open==0) {
				closedCount[close]++;
			}else if(open>0 && close==0) {
				openCount[open]++;
			}else if(open==0 && close==0) {
				openCount[0]++;
				closedCount[0]++;
			}
		}
		int ans=openCount[0]/2;
		for(int i=1;i<openCount.length;i++) {
			ans=ans+Math.min(openCount[i],closedCount[i]);
		}
		System.out.println(ans);
	}
}
