t = int(input())
for _ in range(t):
	n = int(input())
	if n <= 30:
		print('No')
	elif n - 30 == 10:
		print('Yes')
		print('6 5 14', n - 30 + 5)
	elif n - 30 == 14:
		print('Yes')
		print('6 10 7', n - 30 + 7)
	elif n - 30 == 6:
		print('Yes')
		print('5 6 10 15')
	else:
		print('Yes')
		print('6 10 14', n - 30)
