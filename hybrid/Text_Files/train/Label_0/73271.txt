(n, q, m) = map(int, input().split())
a = list(map(int, input().split()))
qq = list((list(map(int, input().split())) for i in range(q)))
qq = list(reversed(qq))
b = list(map(int, input().split()))
for (qt, l, r) in qq:
	if qt == 1:
		for i in range(m):
			if l <= b[i] <= r:
				b[i] = r if b[i] == l else b[i] - 1
	else:
		for i in range(m):
			if l <= b[i] <= r:
				b[i] = r - (b[i] - l)
print(*(a[i - 1] for i in b))
