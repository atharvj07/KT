#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int maxn = 2e6 + 10;
long long a[maxn];
long long l[maxn], r[maxn];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  long long mx = 0;
  for (int i = (1); i < (n + 1); ++i) cin >> a[i];
  for (int i = (1); i < (n + 1); ++i) l[i] = a[i] + (i - 2 < 1 ? 0 : l[i - 2]);
  for (int i = (n + 1) - 1; i >= (1); --i)
    r[i] = a[i] + (i + 2 > n ? 0 : r[i + 2]);
  for (int i = (1); i < (n + 1); ++i) {
    mx = max(mx, r[i] + l[i - 1]);
  }
  cout << mx << '\n';
  return 0;
}
