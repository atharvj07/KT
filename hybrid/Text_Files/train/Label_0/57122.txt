import java.io.*;
import java.math.BigInteger;
import java.util.*;
import static java.lang.Math.*;
import static java.util.Arrays.fill;
import static java.util.Arrays.binarySearch;
import static java.util.Arrays.sort;

public class Main {
	public static void main(String[] args) throws IOException {
		new Thread(null, new Runnable() {
			public void run() {
				try {
					try {
						if (new File("input.txt").exists())
							System.setIn(new FileInputStream("input.txt"));
					} catch (SecurityException e) {}
					new Main().run();
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
		}, "1", 1L << 24).start(); 
	}

	BufferedReader in;
	PrintWriter out;
	StringTokenizer st = new StringTokenizer("");
	
	Random rnd = new Random(777L);
	int BASE = BigInteger.probablePrime(17, rnd).intValue();
	int[] code = new int [256];
	
	void run() throws IOException {
		for (int i = 0; i < code.length; i++) code[i] = rnd.nextInt();
		
		in = new BufferedReader(new InputStreamReader(System.in));
		out = new PrintWriter(System.out);
		
		char[] s1 = in.readLine().toCharArray();
		char[] s2 = in.readLine().toCharArray();
		char[] s3 = in.readLine().toCharArray();
		
		if (contains(s1, s2)) s2 = "".toCharArray();
		if (contains(s1, s3)) s3 = "".toCharArray();
		if (contains(s2, s1)) s1 = "".toCharArray();
		if (contains(s2, s3)) s3 = "".toCharArray();
		if (contains(s3, s1)) s1 = "".toCharArray();
		if (contains(s3, s2)) s2 = "".toCharArray();
		
//		System.out.println(s1);
//		System.out.println(s2);
//		System.out.println(s3);
		
		int ans = Integer.MAX_VALUE;
		ans = min(ans, solve(s1, s2, s3));
		ans = min(ans, solve(s1, s3, s2));
		ans = min(ans, solve(s2, s1, s3));
		ans = min(ans, solve(s2, s3, s1));
		ans = min(ans, solve(s3, s1, s2));
		ans = min(ans, solve(s3, s2, s1));
		out.println(ans);
		
		out.close();
	}
	
	boolean contains(char[] s1, char[] s2) {
		if (s1.length < s2.length) return false;
		int hash1 = 0, hash2 = 0, pow = 1;
		for (int i = 0; i < s2.length; i++) {
			hash1 = hash1 * BASE + code[s1[i]];
			hash2 = hash2 * BASE + code[s2[i]];
			if (i < s2.length - 1) pow *= BASE;
		}
		if (hash1 == hash2) return true;
		for (int i = s2.length; i < s1.length; i++) {
			hash1 -= code[s1[i - s2.length]] * pow;
			hash1 *= BASE;
			hash1 += code[s1[i]];
			if (hash1 == hash2) return true;
		}
		return false;
	}

	int solve(char[] s1, char[] s2, char[] s3) {
		return s1.length + s2.length + s3.length - common(s1, s2) - common(s2, s3);
	}
	
	int common(char[] s1, char[] s2) {
		int n = min(s1.length, s2.length);
		int ret = 0;
		int hash1 = 0, hash2 = 0, pow = 1;
		for (int i = 0; i < n; i++) {
			hash1 = hash1 * BASE + code[s1[s1.length - i - 1]];
			hash2 = hash2 + code[s2[i]] * pow; pow *= BASE;
			if (hash1 == hash2) ret = i + 1;
		}
		return ret;
	}

	/*************************************************************** 
	 * Utility
	 **************************************************************/
	long b2mb(long b) {
		return b >> 20;
	}
	
	void checkMemory() {
		System.err.println(b2mb(Runtime.getRuntime().totalMemory() - Runtime.getRuntime().freeMemory()) + "/" + b2mb(Runtime.getRuntime().totalMemory()) + " MB");
	}
	
	/*************************************************************** 
	 * Input 
	 **************************************************************/
	String nextToken() throws IOException {
		while (!st.hasMoreTokens())
			st = new StringTokenizer(in.readLine());
		return st.nextToken();
	}
	
	int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}
	
	long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}
	
	double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}
	
	String nextLine() throws IOException {
		st = new StringTokenizer("");
		return in.readLine();
	}
	
	boolean EOF() throws IOException {
		while (!st.hasMoreTokens()) {
			String s = in.readLine();
			if (s == null)
				return true;
			st = new StringTokenizer(s);
		}
		return false;
	}
}
