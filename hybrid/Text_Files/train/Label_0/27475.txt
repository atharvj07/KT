#include <bits/stdc++.h>
using namespace std;
unsigned int mod = 1000000007;
inline int pow_(long long a, int n, int p = mod) {
  int r = 1;
  while (n) {
    if (n & 1) r = r * a % p;
    n >>= 1;
    a = a * a % p;
  }
  return r;
}
inline int inv_(int a) { return pow_(a, mod - 2, mod); }
inline unsigned int add(unsigned int a, unsigned int b) {
  a += b;
  if (a >= mod) a -= mod;
  return a;
}
inline unsigned int mul(unsigned int a, unsigned int b) {
  return a * 1ll * b % mod;
}
inline int sub(int a, int b) {
  a -= b;
  if (a < 0) a += mod;
  return a;
}
unsigned int d, dp[210][210];
unsigned int ncr[210][210];
unsigned int treeCnt(unsigned int n, unsigned int deg) {
  unsigned int& r = dp[n][deg];
  if (~r)
    return r;
  else
    r = 0;
  if (deg >= n) return r;
  if (deg == 0) return r = (n == 1);
  if (deg == 1) {
    for (int dg = 0; dg < (d); ++dg) r = add(r, treeCnt(n - 1, dg));
    return r;
  }
  for (int s = 2; s < n; ++s) {
    int tmp = mul(treeCnt(s, 1), treeCnt(n - s + 1, deg - 1));
    tmp = mul(tmp, ncr[n - 2][s - 2]);
    r = add(r, tmp);
  }
  return r;
}
int main() {
  unsigned int n = 100;
  d = 50;
  scanf("%u%u%u", &n, &d, &mod);
  memset(dp, -1, sizeof(dp));
  for (int i = 0; i < (210); ++i)
    for (int j = 0; j < (210); ++j)
      if (i == j || j == 0)
        ncr[i][j] = 1;
      else
        ncr[i][j] = add(ncr[i - 1][j - 1], ncr[i - 1][j]);
  unsigned int ans = 0;
  for (unsigned int deg = 1; deg <= d; ++deg) {
    for (unsigned int d1 = 0; d1 <= deg; ++d1)
      if (d1 != 1) {
        for (unsigned int n1 = 0; n1 < n; ++n1) {
          unsigned int tmp =
              mul(treeCnt(n - n1, deg - d1), treeCnt(n1 + 1, d1));
          ans = add(ans, tmp);
        }
      }
  }
  ans = mul(ans, 2 * n * (n - 1));
  printf("%d\n", (int)ans);
  return 0;
}
