import java.util.Scanner;

public class Violas {

	public static void main(String[] args) {
		Scanner scanner = new Scanner(System.in);
		int r, c, n, k;
		r = scanner.nextInt();
		c = scanner.nextInt();
		n = scanner.nextInt();
		k = scanner.nextInt();
		boolean[][] table = new boolean[r][c];

		for(int i = 0; i < n; i++){
			int x, y;
			x = scanner.nextInt();
			y = scanner.nextInt();
			table[x-1][y-1] = true;
		}
		scanner.close();
		int anw = 0;

		for (int i = 0; i < r; i++){
			for (int j = 0; j < c; j++){
				
				for (int l = i; l < r; l++){
					for (int m = j; m < c; m++){
						anw += calculate(i, j, l, m, table, k);
					}
				}
				
			}
		}
		System.out.print(anw);
	}

	private static int calculate(int i, int j, int l, int m, boolean[][] table, int k) {

		int sum = 0;
		for (; i <= l; i++){
			for (int b = j; b <= m; b++){
				if (table[i][b])	sum++;
			}
		}

		return sum >= k ? 1 : 0; 
	}

	
}
