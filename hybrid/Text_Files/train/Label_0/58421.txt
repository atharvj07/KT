import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.InputMismatchException;
import java.io.IOException;
import java.util.Collections;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author beginner1010
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        long[] value;
        long[][] dp;
        ArrayList<Integer>[] adj;
        ArrayList<Integer> optimalNodes;

        void dfs(int node, int parent) {
            if (adj[node].size() == 1 && node != 0) {
                dp[0][node] = value[node];
                dp[1][node] = 0;
                return;
            }

            long sum0 = 0;
            long max0 = 0;
            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                dfs(neighbor, node);
                sum0 += dp[0][neighbor];
                max0 = Math.max(max0, dp[0][neighbor] - dp[1][neighbor]);
            }

            dp[0][node] = sum0;
            dp[1][node] = sum0 - max0;
            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                long curSum = sum0 - dp[0][neighbor] + dp[1][neighbor] + value[node];
                dp[0][node] = Math.min(dp[0][node], curSum);
            }
        }

        void rec(int node, int parent, long optimal) {
            if (adj[node].size() == 1 && node != 0) {
                if (optimal == value[node]) {
                    optimalNodes.add(node);
                }
                return;
            }
            long sum0 = 0;
            long max0 = 0;
            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                sum0 += dp[0][neighbor];
                max0 = Math.max(max0, dp[0][neighbor] - dp[1][neighbor]);
            }

            int cnt = 0;
            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                if (dp[0][neighbor] - dp[1][neighbor] == max0)
                    cnt++;
            }

            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                long curSum = sum0 - dp[0][neighbor] + dp[1][neighbor] + value[node];
                if (curSum == optimal) {
                    optimalNodes.add(node);
                    break;
                }
            }

            for (int neighbor : adj[node]) {
                if (neighbor == parent)
                    continue;
                if (sum0 == optimal || dp[0][neighbor] - dp[1][neighbor] < max0 || cnt > 1) {
                    rec(neighbor, node, dp[0][neighbor]);
                } else {
                    rec(neighbor, node, dp[1][neighbor]);
                    //throw new IllegalStateException("No way!");
                }
            }

        }

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.nextInt();
            value = new long[n];
            for (int i = 0; i < n; i++) {
                value[i] = in.nextInt();
            }

            adj = new ArrayList[n];
            for (int i = 0; i < n; i++) {
                adj[i] = new ArrayList<>();
            }
            for (int i = 0; i < n - 1; i++) {
                int a = in.nextInt() - 1;
                int b = in.nextInt() - 1;
                adj[a].add(b);
                adj[b].add(a);
            }

            dp = new long[2][n];
            dfs(0, -1);
            optimalNodes = new ArrayList<>();
            rec(0, -1, dp[0][0]);
            Collections.sort(optimalNodes);
            out.println(dp[0][0] + " " + optimalNodes.size());
            for (int i = 0; i < optimalNodes.size(); i++) {
                if (i > 0) out.print(" ");
                out.print(optimalNodes.get(i) + 1);
            }
            out.println();
        }

    }

    static class InputReader {
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;
        private InputStream stream;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        private boolean isWhitespace(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        private int read() {
            if (numChars == -1)
                throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public int nextInt() {
            int c = read();
            while (isWhitespace(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isWhitespace(c));
            return res * sgn;
        }

    }
}
