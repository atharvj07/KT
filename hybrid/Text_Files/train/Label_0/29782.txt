#include <bits/stdc++.h>
using namespace std;
template <class T>
inline void smin(T &a, T b) {
  if (b < a) a = b;
}
template <class T>
inline void smax(T &a, T b) {
  if (a < b) a = b;
}
const int inf = 1 << 28;
const int maxn = 2000 + 100;
int n, m;
char s[maxn][maxn];
int d[maxn][maxn];
pair<int, int> nxt[maxn][maxn];
void go(int x, int y) {
  if (s[x][y] == '#') {
    d[x][y] = 0;
    return;
  }
  if (d[x][y] != -1) return;
  d[x][y] = -2;
  int nx = x, ny = y;
  if (s[x][y] == '^') nx--;
  if (s[x][y] == 'v') nx++;
  if (s[x][y] == '<') ny--;
  if (s[x][y] == '>') ny++;
  go(nx, ny);
  if (d[nx][ny] < 0)
    d[x][y] = inf;
  else
    d[x][y] = min(inf, d[nx][ny] + 1);
}
bool mark[maxn][maxn];
bool mv(int x, int y) {
  if (s[x][y] == '#') return true;
  if (mark[x][y]) return false;
  mark[x][y] = true;
  int nx = x, ny = y;
  if (s[x][y] == '^') nx--;
  if (s[x][y] == 'v') nx++;
  if (s[x][y] == '<') ny--;
  if (s[x][y] == '>') ny++;
  return mv(nx, ny);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 0, _n = (int)(n); i < _n; i++) cin >> s[i];
  memset(d, -1, sizeof d);
  for (int i = 0, _n = (int)(n); i < _n; i++)
    for (int j = 0, _n = (int)(m); j < _n; j++)
      if (d[i][j] == -1) go(i, j);
  for (int i = 0, _n = (int)(n); i < _n; i++)
    for (int j = 0, _n = (int)(m); j < _n; j++)
      if (d[i][j] == inf) {
        cout << -1 << endl;
        return 0;
      }
  int mx = 0;
  for (int i = 0, _n = (int)(n); i < _n; i++)
    for (int j = 0, _n = (int)(m); j < _n; j++) smax(mx, d[i][j]);
  if (mx == 0) {
    cout << 0 << endl;
    return 0;
  }
  int ans = 2 * mx - 2;
  for (int i = 0, _n = (int)(n); i < _n; i++)
    for (int j = 0, _n = (int)(m); j < _n; j++)
      if (d[i][j] == mx)
        if (mv(i, j)) ans++;
  cout << min(2 * mx, ans) << endl;
  { return 0; }
}
