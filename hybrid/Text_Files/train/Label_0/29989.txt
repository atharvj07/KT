#include <bits/stdc++.h>
using namespace std;
int a[200010], b[200010];
int main() {
  int n, m, beg = 0, now = 0, ans = 0;
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> a[i] >> b[i];
  ans = m + b[0] - a[0];
  for (int i = 1; i < n; i++) {
    now += a[i] - b[i - 1];
    while (now >= m) {
      beg++;
      now -= a[beg] - b[beg - 1];
    }
    ans = max(ans, b[i] - a[beg] + m - now);
  }
  cout << ans << endl;
  return 0;
}
