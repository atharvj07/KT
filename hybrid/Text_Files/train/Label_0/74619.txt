cases = int(input())
wds = ['NO', 'YES']
for j in range(0, cases):
	target = list(input())
	word = list(input())
	targetInd = 0
	cor = True
	valid = ''
	tr = target[targetInd]
	for i in range(0, len(word)):
		if word[i] != valid and word[i] != tr:
			cor = False
			break
		elif word[i] == tr:
			targetInd += 1
			if targetInd >= len(target):
				tr = '0'
			else:
				tr = target[targetInd]
		valid = word[i]
	if targetInd < len(target):
		cor = False
	print(wds[cor])
