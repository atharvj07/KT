import java.util.*;

public class solution {
    public static void main(String a[]) {
        int bact = 0;

        Scanner in = new Scanner(System.in);
        bact = in.nextInt();


        if((bact & (bact - 1)) == 0) {
            System.out.print(1);
            System.exit(0);
        }
        
        int bactCount = 0;
        int index = 0;
        int doneCyc = 0;
        for(int temp = 0; doneCyc != bact; temp+=doneCyc ){
            while(Math.pow(2, index) < bact - doneCyc ) {
                if(Math.pow(2, index + 1 ) > bact - doneCyc) {
                    break;
                }
                index++;
            }
            if(temp != bact) {
                bactCount++;
                doneCyc += Math.pow(2, index);
                index = 0;
            }
        }
        
        System.out.print(bactCount);
    }
    
}
