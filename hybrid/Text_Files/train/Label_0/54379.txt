import java.io.PrintWriter;
import java.util.Scanner;

public class TaskB {
    /**
     * @param args
     */
    public static void main(String[] args) {
        TaskB task = new TaskB();
        task.read();
        task.write();
    }

    long a,b;
    long mod;

    public void read() {
        Scanner in = new Scanner(System.in);
        // init
        a = in.nextLong();
        b = in.nextLong();
        mod = in.nextLong();
        in.close();
    }

    public void write() {
        PrintWriter out = new PrintWriter(System.out);
        long mul = 1000000000 % mod;
        for (long i = 1; i <= Math.min(a, mod); i++){
            if (((i * mul) % mod) != 0 && mod  - ((i * mul) % mod) > b){
                out.printf("1 %09d\n", i);
                out.close();
                return;             
            }
        }
        out.println("2");
        out.close();
    }
}
