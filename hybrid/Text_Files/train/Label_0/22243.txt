#include <bits/stdc++.h>
using namespace std;
void stringsplitter(string s, string delim, string arr[]) {
  char *ptr = strtok((char *)s.c_str(), (char *)delim.c_str());
  long long i = 0;
  while (ptr != NULL) {
    arr[i] = ((string)ptr);
    ptr = strtok(NULL, (char *)delim.c_str());
    i++;
  }
}
long long string_to_int(string s) {
  long long ans = 0;
  long long p = 1;
  for (long long i = s.size() - 1; i >= 0; i--) {
    ans += (s[i] - '0') * p;
    p *= 10;
  }
  return ans;
}
void t_c_c() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long a, b;
  cin >> a >> b;
  long long cnt = 0;
  for (long long i = 1; i <= a; i++) {
    if (b % i == 0 and (b / i) <= a) {
      cnt++;
    }
  }
  cout << cnt << endl;
}

  t_c_c();
