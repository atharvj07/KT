import java.util.*;
import java.io.*; 
public class try10 { 
    
    public static void main(String[] args)throws IOException{  
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        String s1=br.readLine();
        int n=Integer.parseInt(s1);
        String s=new String();
        for(int i=2;i*i<=n;i++)
        {
            if(n%i==0)
            {
                s+=i;
                int x=n/i;
                s+=x;
                break;
            }
        }
        System.out.println(s);
       
        } 
        
}
