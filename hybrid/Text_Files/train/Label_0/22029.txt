#include <bits/stdc++.h>
using namespace std;
long long n, m, cnt;
char s[200005], t[200005], v;
long long ans[200005];
long long fast(long long b, long long p, long long k) {
  long long a = 1;
  while (p != 0) {
    if (p & 1 != 0) {
      a = ((a % k) * b) % k;
    }
    b = (b * b) % k;
    p >>= 1;
  }
  return a % k;
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> v;
    s[n - i + 1] = v;
  }
  for (int i = 1; i <= m; i++) {
    cin >> v;
    t[m - i + 1] = v;
  }
  for (int i = 1; i <= n; i++) {
    if (s[i] - '0' == 1)
      ans[i] = (ans[i - 1] % 998244353 + fast(2, i - 1, 998244353));
    else
      ans[i] = ans[i - 1];
  }
  if (n < m) {
    for (int i = n + 1; i <= m; i++) ans[i] = ans[i - 1];
  }
  for (int i = m; i >= 1; i--) {
    if (t[i] - '0' == 1)
      cnt = (cnt % 998244353 + ans[i] % 998244353) % 998244353;
  }
  cout << cnt;
  return 0;
}
