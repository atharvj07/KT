#include <bits/stdc++.h>
const long long LINF = 1e18;
const int INF = 1e9;
const int M = 1e9 + 7;
const double EPS = 1.0e-9;
const double PI = acos(-1.0);
using namespace std;
int main() {
  std::ios_base::sync_with_stdio(0);
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n, m;
    long long p_even = 0;
    long long p_odd = 0;
    long long q_even = 0;
    long long q_odd = 0;
    int cur;
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> cur;
      if (cur % 2)
        p_odd++;
      else
        p_even++;
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
      cin >> cur;
      if (cur % 2)
        q_odd++;
      else
        q_even++;
    }
    cout << p_odd * q_odd + p_even * q_even << "\n";
  }
  return 0;
}
