#include<cstdio>
#include<algorithm>
#include<iostream>
#include<cstring>
#include<vector>
using namespace std;
typedef long long LL;
#define ve vector<int> 
const int N=100005;
int n,k; 
ve vec[N];
ve mul (ve a,ve b)
{
	ve f(n);
	for (int u=0;u<n;u++)	f[u]=a[b[u]];
	return f;
}
ve inv (vector<int> a)
{
	ve f(n);
	for (int u=0;u<n;u++)	f[a[u]]=u;
	return f;
}
ve Pow(ve x,int y)
{
	ve f(n);
	for (int u=0;u<n;u++) f[u]=u;
	while (y>0)
	{
		if (y&1) f=mul(f,x);
		x=mul(x,x);y>>=1;
	}
	return f;
}
void print (ve x)
{
	for (int u=0;u<n;u++) printf("%d ",x[u]+1);
	printf("\n");
}
int main()
{
	scanf("%d%d",&n,&k);
	for (int u=1;u<=n;u++)	{int x;scanf("%d",&x);vec[1].push_back(x-1);}
	for (int u=1;u<=n;u++)	{int x;scanf("%d",&x);vec[2].push_back(x-1);}
	for (int u=3;u<=6;u++) vec[u]=mul(vec[u-1],inv(vec[u-2]));
	ve A=mul(mul(vec[2],inv(vec[1])),mul(inv(vec[2]),vec[1]));
	A=Pow(A,(k-1)/6);
	print(mul(mul(A,vec[(k-1)%6+1]),inv(A)));
	return 0;
}
