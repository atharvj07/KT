import sys
from sys import stdin, stdout
from collections import defaultdict
from collections import deque
import math
import copy
arr = [0] * 500001
K = 530
Q = int(sys.stdin.readline())
qus = sys.stdin.readlines()
small = [[0] * K for _ in range(K)]
ans = []
a = 0
for q in qus:
	(tag, idx, delta) = [int(x) for x in q.split()]
	if tag == 1:
		arr[idx] += delta
		for k in range(1, K + 1):
			small[k - 1][idx % k] += delta
	elif tag == 2:
		if idx <= K:
			ans.append(small[idx - 1][delta])
		else:
			res = 0
			for k in range(delta, 500001, idx):
				res += arr[k]
			ans.append(res)
print(*ans)
