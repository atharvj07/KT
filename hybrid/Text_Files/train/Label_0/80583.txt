from sys import stderr

def gcd_ext(a, b):
	(r2, x2, y2, r1, x1, y1, r0, x0, y0, q) = 10 * [0]
	r2 = a
	x2 = 1
	y2 = 0
	r1 = b
	x1 = 0
	y1 = 1
	while r1:
		q = r2 // r1
		r0 = r2 % r1
		x0 = x2 - q * x1
		y0 = y2 - q * y1
		r2 = r1
		x2 = x1
		y2 = y1
		r1 = r0
		x1 = x0
		y1 = y0
	g = r2
	x = x2
	y = y2
	if g < 0:
		g = -g
		x = -x
		y = -y
	return (g, x, y)

def crt(r1, m1, r2, m2):
	(g, x, y) = gcd_ext(m1, m2)
	if (r1 - r2) % g != 0:
		return (-1, -1)
	z = m2 // g
	lcm = m1 * z
	sol = r1 % lcm + m1 * (x % z * ((r2 - r1) // g % z) % z) % lcm
	return (sol, lcm)
(n, m, x, y, vx, vy) = list(map(int, input().split()))
if vx == 0 and x % n != 0 or (vy == 0 and y % m != 0):
	print(-1)
	exit(0)
if vx == 0:
	print(x, (vy + 1) // 2 * m)
	exit(0)
if vy == 0:
	print((vx + 1) // 2 * n, y)
	exit(0)
if x == 0 and vx == -1 or (x == n and vx == 1):
	vx *= -1
if y == 0 and vy == -1 or (y == m and vy == 1):
	vy *= -1
r1 = x * -vx
r2 = y * -vy
if vx > 0:
	r1 += n
if vy > 0:
	r2 += m
(sol, asdf) = crt(r1, n, r2, m)
if sol == -1 and asdf == -1:
	print(-1)
	exit(0)
k1 = (sol - r1) // n
k2 = (sol - r2) // m
if k1 % 2:
	vx *= -1
if k2 % 2:
	vy *= -1
print((vx + 1) // 2 * n, (vy + 1) // 2 * m)
