#include <bits/stdc++.h>
using namespace std;
long long a[4][100005];
long long dp[4][100005];
int main() {
  int n;
  while (scanf("%d", &n) == 1) {
    for (int i = 0; i < 3; i++)
      for (int j = 0; j < n; j++) scanf("%lld", &a[i][j]);
    dp[0][0] = a[0][0];
    dp[1][0] = a[0][0] + a[1][0];
    dp[2][0] = a[0][0] + a[1][0] + a[2][0];
    for (int i = 1; i < n; i++) {
      dp[0][i] = max(dp[0][i - 1] + a[0][i], dp[1][i - 1] + a[1][i] + a[0][i]);
      dp[0][i] = max(dp[0][i], dp[2][i - 1] + a[2][i] + a[1][i] + a[0][i]);
      if (i > 1)
        dp[0][i] =
            max(dp[0][i], dp[2][i - 2] + a[2][i - 1] + a[2][i] + a[1][i] +
                              a[1][i - 1] + a[0][i - 1] + a[0][i]);
      dp[1][i] = max(dp[0][i - 1] + a[0][i] + a[1][i], dp[1][i - 1] + a[1][i]);
      dp[1][i] = max(dp[1][i], dp[2][i - 1] + a[2][i] + a[1][i]);
      dp[2][i] = max(dp[2][i - 1] + a[2][i], dp[1][i - 1] + a[1][i] + a[2][i]);
      dp[2][i] = max(dp[2][i], dp[0][i - 1] + a[0][i] + a[1][i] + a[2][i]);
      if (i > 1)
        dp[2][i] =
            max(dp[2][i], dp[0][i - 2] + a[0][i - 1] + a[0][i] + a[1][i] +
                              a[1][i - 1] + a[2][i - 1] + a[2][i]);
      else
        dp[2][i] = max(dp[2][i], a[0][i - 1] + a[0][i] + a[1][i] + a[1][i - 1] +
                                     a[2][i - 1] + a[2][i]);
    }
    printf("%lld\n", dp[2][n - 1]);
  }
}
