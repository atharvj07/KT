#include <bits/stdc++.h>
using namespace std;
const int N = 1000005, P = 1000000007;
using ll = long long;
int n, p[N], cnt[N], cc[N], bs[N], to1[N], to2[N];
ll f[N];
vector<int> fac[N];
inline void fail() {
  cout << 0 << endl;
  exit(0);
}
void check(int x, int y) {
  if (fac[x].size() != fac[y].size()) fail();
  for (int i = 0; i < int(fac[x].size()); i++) {
    int t1 = fac[x][i], t2 = fac[y][i];
    int f1 = (x == 1) ? 1 : n / t1, f2 = (y == 1) ? 1 : n / t2;
    if (f1 != f2 || (to1[t1] && to1[t1] != t2) || (to2[t2] && to2[t2] != t1))
      fail();
    if (!to2[t2]) to1[t1] = t2, to2[t2] = t1, --cnt[f1];
  }
  --cc[bs[x]];
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  cnt[1] = f[0] = 1;
  for (int i = 1; i <= n; i++) f[i] = f[i - 1] * i % P, bs[i] = 1;
  for (int i = 2; i <= n; i++)
    if (!p[i]) {
      ++cnt[n / i];
      for (int j = i; j <= n; j += i) {
        p[j] = 1;
        bs[j] *= i;
        fac[j].push_back(i);
      }
    }
  fac[1].push_back(1);
  for (int i = 1; i <= n; i++) ++cc[bs[i]];
  int t1;
  for (int i = 1; i <= n; i++) {
    cin >> t1;
    if (t1) check(i, t1);
  }
  ll ans = 1;
  for (int i = 1; i <= n; i++) ans = ans * f[cnt[i]] % P * f[cc[i]] % P;
  cout << ans << endl;
  return 0;
}
