#include <bits/stdc++.h>
using namespace std;

int main(){

    while(true){
        int N, M;
        cin >> N >> M;
        if(N == 0){
            break;
        }
        vector<int> s(M);
        vector<vector<int>> c(M);

        vector<int> maxScore(N, 0);
        vector<int> minScore(N, 0);

        for(int i = 0; i < M; i++){
            cin >> s[i];
            int k;
            cin >> k;
            int temp;
            if(k == 1){
                cin >> temp;
                temp--;
                minScore[temp] += s[i];
                maxScore[temp] += s[i];
            }
            else{
                for(int j = 0; j < k; j++){
                    cin >> temp;
                    temp--;
                    maxScore[temp] += s[i];
                }
            }
        }

        vector<pair<int, int>> maxPair(N);
        vector<pair<int, int>> minPair(N);
        for(int i = 0; i < N; i++){
            maxPair[i] = make_pair(maxScore[i], i);
        }
        for(int i = 0; i < N; i++){
            minPair[i] = make_pair(minScore[i], i);
        }

        sort(maxPair.begin(), maxPair.end(), greater<pair<int, int>>());
        sort(minPair.begin(), minPair.end());

        int ans = 0;
        for(int i = 0; i < 2; i++){
            for(int j = 0; j < 2; j++){
                auto big = maxPair[i];
                auto small = minPair[j];
                if(big.second == small.second){
                    continue;
                }
                ans = max(ans, big.first - small.first);
            }
        }

        cout << ans + 1 << endl;
    }


    return 0;
}
