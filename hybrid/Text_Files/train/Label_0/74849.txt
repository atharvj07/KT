import sys
line = lambda : sys.stdin.buffer.readline().strip()
n = int(line())
s = [line() for _ in range(3)]
r = [0] * 3
for i in range(3):
	c = {}
	r[i] = min(len(s[i]), n)
	for k in s[i]:
		c[k] = c[k] + 1 if k in c else 1
	for k in c:
		r[i] = max(r[i], min(len(s[i]), c[k] + n) - (c[k] == len(s[i]) and n == 1))
m = max(r)
t = r.count(m)
if t > 1:
	print('Draw')
elif r[0] == m:
	print('Kuro')
elif r[1] == m:
	print('Shiro')
else:
	print('Katie')
