	public static long comp(int n, int r) {
		r = Math.min(r, n - r);
		long ans = 1;
		for (int i = 1; i <= r; i++) {
			ans = ans * (long) (n - r + i) / (long) i;
		}
		return ans;

	}

	public static int tree[];

	public static void add(int idx, long val) {

		while (idx <= n) {
			tree[idx] += val;
			idx += (idx & -idx);

	public static int get(int idx) {
		int g = 0;
		while (idx > 0) {
			
			g += tree[idx];
			idx -= (idx & -idx);
		}
		
		return g;
	}
