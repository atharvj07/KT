def fibonacci(n, m):
	if n < 2:
		return n
	a = 0
	b = 1
	for i in range(n - 1):
		c = b
		b = (a + b) % m
		a = c
	return b

def egcd(a, b):
	if a == 0:
		return (b, 0, 1)
	else:
		(g, y, x) = egcd(b % a, a)
		return (g, x - b // a * y, y)

def modinv(a, m):
	(g, x, y) = egcd(a, m)
	if g != 1:
		raise Exception('modular inverse does not exist')
	else:
		return x % m

def pow(a, b, m):
	if b < 0:
		raise Exception('bad')
	ret = 1
	for i in range(b):
		ret *= a
		ret %= m
	return ret
inp = int(input())
p = 998244353
print(fibonacci(inp, p) * modinv(pow(2, inp, p), p) % p)
