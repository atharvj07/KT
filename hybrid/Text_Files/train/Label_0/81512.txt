(n, k) = map(int, input().split())
ans = 0
temp = []
for i in range(n):
	li = set(list(map(int, input().split()))[1:])
	if len(li) < 1:
		ans += 1
	elif not temp:
		temp.append(set(li))
	else:
		cnt = False
		for j in temp:
			if j.intersection(li):
				cnt = True
				for k in li:
					j.add(k)
		if not cnt:
			temp.append(set(li))
ans2 = []
ln = len(temp)
while True:
	cnt = False
	(founded, zero) = (False, False)
	if zero:
		break
	for i in range(ln):
		for j in range(ln):
			if temp[i].intersection(temp[j]) and i != j:
				cnt = True
				for k in temp[j]:
					temp[i].add(k)
				del temp[j]
				ln -= 1
				if ln == 0:
					zero = True
				founded = True
			if founded:
				break
		if founded:
			break
	if not cnt:
		break
ln2 = len(temp) - 1
if ln2 >= 0:
	print(ln2 + ans)
else:
	print(ans)
