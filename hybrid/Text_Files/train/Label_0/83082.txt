def inputstring():
	return [x for x in input()]
t = int(input())
for q in range(t):
	s = inputstring()
	count = 0
	for i in s:
		if i != '0':
			count += 1

	for i in range(len(s)):
		rtn = ''
		if s[i] != '0':
			rtn += s[i]
			for j in range(len(s) - i - 1):
				rtn += '0'
			print(rtn, end=' ')
