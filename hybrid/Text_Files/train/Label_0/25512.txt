#include<iostream>
#include<algorithm>
#include<climits>
using namespace std;

int h,w;
int store[1000][1000];
int memo[1000][1000][32];

int getCost(int y,int x){
	if(y<0 || h<=y) return 0;
	if(x<0 || w<=x) return 0;
	return store[y][x];
}

int solve(int y,int x,int bought){
	if(h<=y || w<=x) return INT_MAX;
	if(memo[y][x][bought]!=-1) return memo[y][x][bought];
	
	int ans=INT_MAX;
	int cost;
	int nexte,nexts;

	for(int i=0;i<4;i++){
			cost=0;nexte=0;nexts=0;
			if(bought & 1){
				nexte|=4;nexts|=2;
			}else{
				cost+=getCost(y,x);
			}
			if(i!=0 && !(bought & 2)){
				cost+=getCost(y-1,x);
			}
			if(i!=1 && !(bought & 4)){
				cost+=getCost(y,x-1);
			}
			if(i!=2){
				cost+=getCost(y+1,x);
				nexte|=16;nexts|=1;
			}
			if(i!=3){
				cost+=getCost(y,x+1);
				nexte|=1;nexts|=8;
			}
			if(bought & 8){nexte|=2;}
			if(bought & 16){nexts|=4;}

			if(y!=h-1 || x!=w-1){
				cost+=min(solve(y,x+1,nexte),solve(y+1,x,nexts));
			}
			ans=min(ans,cost);
	}

	memo[y][x][bought]=ans;
	return ans;
}

int main(){
	char c;

	cin>>h>>w;
	for(int y=0;y<h;y++){
		for(int x=0;x<w;x++){
			cin>>c;
			if(c=='.'){
				store[y][x]=0;
			}else{
				store[y][x]=c-'0';
			}
			for(int i=0;i<32;i++){
				memo[y][x][i]=-1;
			}
		}
	}

	cout<<solve(0,0,0)<<endl;
	return 0;
}
