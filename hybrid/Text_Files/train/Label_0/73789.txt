import sys
(n, k) = map(int, input().split())
matrix = [[0] * n for _ in range(n)]
(i1, j1, i2, j2) = (0, 0, 0, 0)
for i in range(n):
	for j in range(n):
		if matrix[i][j]:
			continue
		if i == j and k < 1 or (i != j and k < 2):
			break
		matrix[i][j] = matrix[j][i] = 1
		k -= 1 if i == j else 2
for i in range(n):
	if matrix[i][i] == 0 and k:
		matrix[i][i] = 1
		k -= 1
if k:
	print(-1)
else:
	for row in matrix:
		print(*row)
