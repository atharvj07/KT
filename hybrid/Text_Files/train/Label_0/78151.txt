import sys
input = sys.stdin.buffer.readline
(n, sx, sy) = map(int, input().split())
(l, r, u, d) = (0, 0, 0, 0)
for i in range(n):
	(p, q) = map(int, input().split())
	if p > sx:
		r += 1
	elif p < sx:
		l += 1
	if q > sy:
		u += 1
	else:
		d += 1
if l >= r and l >= u and (l >= d):
	print(l)
	print(sx - 1, sy)
elif r >= l and r >= u and (r >= d):
	print(r)
	print(sx + 1, sy)
elif u >= d and u >= l and (u >= r):
	print(u)
	print(sx, sy + 1)
else:
	print(d)
	print(sx, sy - 1)
