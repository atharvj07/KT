arri = lambda : [int(s) for s in input().split()]
n = int(input())
a = arri()
last = a[0]
fixed = -1
l = 0
f = True
for i in a:
	if last == i:
		l += 1
	else:
		if fixed == -1:
			fixed = l
		elif l != fixed:
			f = False
			break
		l = 1
	last = i
if l != fixed and fixed != -1:
	f = False
print('Yes' if f else 'No')
