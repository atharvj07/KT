t = int(input())
for _ in range(t):
	l = [[] for _ in range(1010)]
	n = int(input())
	for i in range(n):
		(x, y) = map(int, input().split())
		l[x].append(y)
	for i in range(1001):
		l[i].sort()
	nowx = 0
	nowy = 0
	ans = True
	s = ''
	for x in range(1001):
		for y in l[x]:
			if y < nowy:
				ans = False
			s += 'R' * (x - nowx)
			nowx = x
			s += 'U' * (y - nowy)
			nowy = y
	if ans:
		print('YES')
		print(s)
	else:
		print('NO')
