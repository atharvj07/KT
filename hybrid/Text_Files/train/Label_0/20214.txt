#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 7, MOD = 998244353;
pair<int, int> a[N];
long long fact[N];
void init() {
  fact[0] = 1;
  for (int i = 1; i < N; ++i) fact[i] = (fact[i - 1] % MOD * i % MOD) % MOD;
}
int main() {
  init();
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) scanf("%d%d", &a[i].first, &a[i].second);
  sort(a, a + n);
  int secsorted = 1;
  for (int i = 0; i < n - 1; ++i)
    if (a[i].second > a[i + 1].second) {
      secsorted = 0;
      break;
    }
  long long firstres = 1;
  map<int, int> cnt;
  for (int i = 0; i < n; ++i) ++cnt[a[i].first];
  for (auto& __t : cnt)
    firstres = (firstres % MOD * fact[__t.second] % MOD) % MOD;
  long long secondres = 1;
  cnt = map<int, int>();
  for (int i = 0; i < n; ++i) ++cnt[a[i].second];
  for (auto& __t : cnt)
    secondres = (secondres % MOD * fact[__t.second] % MOD) % MOD;
  long long bothres = 1;
  map<pair<int, int>, int> bothcnt;
  if (secsorted)
    for (int i = 0; i < n; ++i) ++bothcnt[a[i]];
  else
    bothres = 0;
  for (auto& __t : bothcnt)
    bothres = (bothres % MOD * fact[__t.second] % MOD) % MOD;
  printf("%lld",
         ((fact[n] - firstres - secondres + bothres) % MOD + MOD) % MOD);
  return 0;
}
