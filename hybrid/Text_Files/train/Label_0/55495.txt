import java.util.*;
import java.io.*;
public class solution
{
	static int min(int a,int b)
	{
		if(a>b)
		{
			return b;
		}
		return a;
	}
	public static void main(String args[])
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		String s = sc.next();
		int l = 0;
		int r = 0;
		int u = 0;
		int d = 0;
		for(int i=0;i<s.length();i++)
		{
			char c = s.charAt(i);
			switch(c)
			{
				case 'L':
				l++;
				break;
				case 'R':
				r++;
				break;
				case 'U':
				u++;
				break;
				case 'D':
				d++;
				break;
			}
		}
		System.out.println(2*(min(u,d)+min(l,r)));
	}
}
