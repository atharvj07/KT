t = int(input())
s = 0
for _ in range(t):
	a = [1, 2, 3, 5, 7, 11]
	b = [4, 6]
	c = [8, 10]
	n = int(input())
	if n in a:
		print(-1)
	elif n in b:
		print(1)
	elif n in c:
		print(2)
	else:
		if n % 4 == 0:
			s = (n - 12) // 4 + 3
		elif n % 4 == 1:
			s = (n - 13) // 4 + 2
		elif n % 4 == 2:
			s = (n - 14) // 4 + 3
		else:
			s = (n - 15) // 4 + 2
		print(s)
