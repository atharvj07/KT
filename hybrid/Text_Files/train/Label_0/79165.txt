(a, b, c) = map(int, input().split())
m = int(input())
s1 = [list(map(str, input().split())) for i in range(m)]
for x in s1:
	x[0] = int(x[0])
s1 = sorted(s1)
k = 0
s = 0
for x in s1:
	flag = True
	if x[1] == 'USB':
		if a > 0:
			flag = False
			k += 1
			a -= 1
			s += x[0]
	elif b > 0:
		flag = False
		k += 1
		b -= 1
		s += x[0]
	if flag:
		if c > 0:
			k += 1
			c -= 1
			s += x[0]
print(k, s)
