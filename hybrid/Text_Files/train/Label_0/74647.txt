def solve(s):
	n = len(s)
	(a, b, c) = (0, 0, 0)
	for i in range(n):
		if s[i] == 'A':
			a += 1
		elif s[i] == 'B':
			b += 1
		else:
			c += 1
	if a != b + c and b != a + c and (c != a + b):
		return 'NO'
	if s[0] == s[-1]:
		return 'NO'
	char = None
	if s[0] == 'A' and a == max(a, b, c):
		char = 'A'
	elif s[0] == 'B' and b == max(a, b, c):
		char = 'B'
	elif s[0] == 'C' and c == max(a, b, c):
		char = 'C'
	flag = 1
	if char == None:
		if s[-1] == 'A' and a == max(a, b, c):
			char = 'A'
		elif s[-1] == 'B' and b == max(a, b, c):
			char = 'B'
		elif s[-1] == 'C' and c == max(a, b, c):
			char = 'C'
		flag = -1
	count = 0
	for i in range(n):
		if s[i] == char:
			count += flag
		else:
			count -= flag
		if count < 0:
			return 'NO'
	if count == 0:
		return 'YES'
	else:
		return 'NO'
t = int(input())
for i in range(t):
	print(solve(input()))
