import os, io, sys
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
a = list(map(int, input().split()))
left = []
right = []
for i in range(31):
	left.append([-1] * n)
	right.append([n] * n)
for i in range(n):
	for j in range(31):
		if 1 << j & a[i]:
			left[j][i] = i
		elif i > 0:
			left[j][i] = left[j][i - 1]
for i in range(n - 1, -1, -1):
	for j in range(31):
		if 1 << j & a[i]:
			right[j][i] = i
		elif i < n - 1:
			right[j][i] = right[j][i + 1]
ans = n * (n - 1) // 2
ans += n
lastsame = [-1] * n
dic = {}
for i in range(n):
	if a[i] in dic:
		lastsame[i] = dic[a[i]]
		dic[a[i]] = i
	else:
		dic[a[i]] = i
for i in range(n):
	maxleft = -1
	minright = n
	for j in range(31):
		if not 1 << j & a[i]:
			maxleft = max(maxleft, left[j][i])
			minright = min(minright, right[j][i])
	if lastsame[i] > maxleft:
		maxleft = lastsame[i]
	ans -= (i - maxleft) * (minright - i)
print(ans)
