import sys
from math import sqrt
[n, k] = map(int, input().strip().split())
if k * (k + 1) > 2 * n:
	print(-1)
	sys.exit(0)
a0max = 2 * n // (k * (k + 1))
d = 1
for i in range(1, min(a0max, int(sqrt(n))) + 1):
	if n % i == 0:
		d = max(d, i)
		if a0max * i >= n:
			d = max(d, n // i)
res = [(i + 1) * d for i in range(k - 1)]
res.append(n - d * (k * (k - 1)) // 2)
print(' '.join(map(str, res)))
