gg = False

def checkAns(b):
	dot = 0
	x = 0
	for i in b:
		if i == 'X':
			x += 1
		if i == '.':
			dot += 1
	if x == 4 and dot == 1 or x == 5:
		return True

def find(b):
	if len(b) < 5:
		return
	global gg
	ans = 0
	for i in range(len(b) - 4):
		if checkAns(b[i:i + 5]):
			gg = True
			return
for i in range(10):
