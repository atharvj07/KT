#include<iostream>
using namespace std;
#define MAX 100005
#define NIL -1

struct Node {int p, l, r;};

Node T[MAX];
int n, D[MAX];

void rec(int a, int d){
  D[a] = d;
  if (T[a].l != NIL) rec(T[a].l, d+1);
  if (T[a].r != NIL) rec(T[a].r, d);
}

void print(int a){
  int i, c;
  cout << "node " << a << ": ";
  cout << "parent = " << T[a].p << ", ";
  cout << "depth = " << D[a] << ", ";
  if (T[a].p == NIL) cout << "root, ";
  else if (T[a].l == NIL) cout << "leaf, ";
  else cout << "internal node, ";

  cout << "[";
  for (i=0, c = T[a].l; c != NIL; i++, c = T[c].r){
    if (i) cout << ", ";
    cout << c;
  }
  cout << "]" << endl;
}

int main(){
  int v, d, c, l, r;

  cin >> n;
  for (int i=0; i<n; i++) T[i].p = T[i].l = T[i].r = NIL;

  for (int i=0; i<n; i++){
    cin >> v >> d;
    for (int j=0; j<d; j++){
      cin >> c;
      if (j == 0) T[v].l = c;
      else T[l].r = c;
      l = c;
      T[l].p = v;
    }
  }

  for (int i=0; i<n; i++) if (T[i].p == NIL) r = i;

  rec(r, 0);

  for (int i=0; i<n; i++) print(i);

  return 0;
}
