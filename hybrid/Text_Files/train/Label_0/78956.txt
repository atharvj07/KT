import math
from collections import defaultdict
ml = lambda : map(int, input().split())
ll = lambda : list(map(int, input().split()))
ii = lambda : int(input())
ip = lambda : list(input())
ips = lambda : input().split()
t = ii()
n = ip()
s = 0
for i in n:
	s += int(i)
a = set()
for i in range(2, int(math.sqrt(s)) + 1):
	if s % i == 0:
		a.add(i)
		if i != s // i:
			a.add(s // i)
a.add(1)
a.add(0)
f = 0
for i in a:
	k = 0
	w = 0
	flag = 1
	for j in range(t):
		k += int(n[j])
		if k > i:
			flag = 0
			break
		elif k == i:
			w += 1
			k = 0
	if k > 0 or w < 2:
		flag = 0
	if flag:
		f = 1
		break
if f:
	print('YES')
else:
	print('NO')
