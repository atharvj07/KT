#include <bits/stdc++.h>
int dr[] = {2, 2, -2, -2, 1, -1, 1, -1};
int dc[] = {1, -1, 1, -1, 2, 2, -2, -2};
int dr1[] = {0, 0, 0, 1, 1, 1, -1, -1, -1};
int dc1[] = {-1, 0, 1, -1, 0, 1, -1, 0, 1};
int dr2[] = {0, 0, 1, -1};
int dc2[] = {1, -1, 0, 0};
using namespace std;
long long int d[500005], a[500005], c[500005];
vector<long long int> adj[500005];
long long int ans;
void dfs(long long int u, long long int par) {
  d[u] = a[u];
  for (long long int i = 0; i < adj[u].size(); i++) {
    long long int p = adj[u][i];
    if (p == par) continue;
    dfs(p, u);
    c[u] += c[p] + d[p];
    d[u] += d[p];
  }
}
void dfs1(long long int u, long long int par, long long int x,
          long long int y) {
  ans = max(ans, x);
  for (long long int i = 0; i < adj[u].size(); i++) {
    long long int p = adj[u][i];
    if (p == par) continue;
    long long int r = x - c[p] - d[p];
    long long int r1 = y - d[p];
    dfs1(p, u, c[p] + r + r1, d[p] + r1);
  }
}
int main() {
  long long int n, i, x, y;
  while (cin >> n) {
    for (i = 1; i <= n; i++) cin >> a[i];
    for (i = 1; i < n; i++) {
      cin >> x >> y;
      adj[x].push_back(y);
      adj[y].push_back(x);
    }
    dfs(1, -1);
    ans = 0;
    dfs1(1, -1, c[1], d[1]);
    cout << ans << endl;
  }
  return 0;
}
