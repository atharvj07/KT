import java.io.*;
import java.util.*;

public class Main {
	static Scanner in;
	static PrintWriter out;
//	static StreamTokenizer in; static int next() throws Exception {in.nextToken(); return (int) in.nval;}
    static PrintWriter sout = new PrintWriter(System.out);

	static int[][] a;
	static int n = 6;
	static ArrayList<String> answ;

	static void shift(int line, boolean hor, boolean plus) {
		int d = plus ? 1 : -1;
		if (hor) {
			int[] y = new int[n];
			for (int i = 0; i < n; i++) y[(i + d + n) % n] = a[line][i];
			for (int i = 0; i < n; i++) a[line][i] = y[i];
			answ.add((plus ? "R" : "L") + (line + 1));
		} else {
			int[] y = new int[n];
			for (int i = 0; i < n; i++) y[(i + d + n) % n] = a[i][line];
			for (int i = 0; i < n; i++) a[i][line] = y[i];
			answ.add((plus ? "D" : "U") + (line + 1));
		}
	}

	static void rotate(int x, int y, boolean right, boolean down, boolean clock) {
		if (clock ^ right ^ down) {
			shift(x, false, !down);
			shift(y, true, !right);
			shift(x, false, down);			
			shift(y, true, right);
		} else {
			shift(y, true, !right);
			shift(x, false, !down);
			shift(y, true, right);
			shift(x, false, down);
		}
	}

	static void up(int x, int y) {
		rotate(x, y, true, false, true);
	}

	static void left(int x, int y) {
		rotate(x, y, false, true, false);
	}

	public static void main(String[] args) throws Exception {
		in = new Scanner(System.in);
		out = new PrintWriter(System.out);
//		in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
		
		
		answ = new ArrayList<String>();

		a = new int[n][n];
		for (int i = 0; i < n; i++) {
			String s = in.next();
			for (int j = 0; j < n; j++) {
				if ('0' <= s.charAt(j) && s.charAt(j) <= '9') a[i][j] = s.charAt(j) - '0';
				else a[i][j] = s.charAt(j) - 'A' + 10;
			}
		}

		int[][] aa = new int[n][n];
		for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) aa[i][j] = a[i][j];
		
		int[][] g = new int[n][n];
		int[] gx = new int[n*n];
		int[] gy = new int[n*n];
		int count = 0;
		for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) {
			g[i][j] = count;
			gx[count] = j;
			gy[count] = i;
			count++;
		}
		for (int index = 0;; index++) {

		answ = new ArrayList<String>();
		for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) a[i][j] = aa[i][j];
		
		if (index > 0) {
			int cnt = ((int)(Math.random() * 1e9)) % 100;
			for (int i = 0; i < cnt; i++) {
				int ii = ((int)(Math.random() * 1e9)) % 6;
				int jj1 = ((int)(Math.random() * 1e9)) % 2;
				int jj2 = ((int)(Math.random() * 1e9)) % 2;
				shift(ii, jj1 == 0, jj2 == 0);
			}
		}

		for (int i = 0; i < 4*n; i++) {
			while (true) {
				int x = 0;
				int y = 0;
				F: for (; x < n; x++) for (y = 0; y < n; y++) {
					if (a[y][x] == i) break F;
				}
				if (g[y][x] == i) break;
				if (x != gx[i]) {
					if (y == 5) rotate(x, y, false, false, true);
					else left(x, y);
				}
				else up(x, y);
			}
		}

		for (int i = 0; i < 4; i++) for (int t = 4*n + i; t <= 5*n + i; t += n) {
			while (true) {
				int x = 0;
				int y = 0;
				F: for (; x < n; x++) for (y = 0; y < n; y++) if (a[y][x] == t) break F;
				if (g[y][x] == t) break;

				if (y != gy[t]) {
					if (x == 5) {
						if (y < gy[t]) rotate(x, y, false, true, true);
						else rotate(x, y, false, false, false);
					} else {
						if (y < gy[t]) rotate(x, y, true, true, false);
						else rotate(x, y, true, false, true);
					}
				} else {
					if (y == 4) rotate(x, y, false, true, false);
					else rotate(x, y, false, false, true);
				}
			}
		}

		int x = 0;
		int y = 0;
		F: for (x = 0; x < n; x++) for (y = 0; y < n; y++) if (a[y][x] == 28) break F;
		if (x == 5 && y == 5) rotate(x, y, false, false, true);
		F: for (x = 0; x < n; x++) for (y = 0; y < n; y++) if (a[y][x] == 28) break F;
		if (x == 5) rotate(x, y, false, true, false);
		F: for (x = 0; x < n; x++) for (y = 0; y < n; y++) if (a[y][x] == 28) break F;
		if (y == 5) rotate(x, y, true, false, true);

		while (a[5][5] != g[5][5]) rotate(5, 5, false, false, true);

		if (a[5][4] == 34) break;

		}

		out.println(answ.size());
		for (String s : answ) out.println(s);

		out.close();
	}
}
