public class TaskC {
	
	public static ArrayList<Long>  generateHappies() {
		
		ArrayList<Long> happyList = new ArrayList<Long>();
		for (int j = 1; j <= 10; ++j) {
			for (int i = 0; i < 1 << j; ++i) {
				int n = i;
				StringBuffer tmp = new StringBuffer("");
				for (int k = 0; k < j; ++k) {
					if ((n & 1) == 1) {
						tmp.append("4");
					} 
					else {
						tmp.append("7");
					}
					n >>= 1;
				}
				happyList.add(Long.valueOf(tmp.toString()));
			}
		}
		return happyList;
	}
	public static void main(String[] args) throws IOException {
		PrintWriter jout = new PrintWriter(System.out);
		Scanner jin = new Scanner(new InputStreamReader(System.in));
		ArrayList<Long> happ = generateHappies();
		Collections.sort(happ);
		
		int left = jin.nextInt();
		int right = jin.nextInt();
		int leftIndex=0;
		int rightIndex=0;
		for (int i = 0; i < happ.size(); ++i) {
			if (happ.get(i) >= left) {
				leftIndex = i;
				break;
			}
		}
		for (int i = 0; i < happ.size(); ++i) {
			if (happ.get(i) >= right) {
				rightIndex = i;
				break;
			}
		}
		long sum = 0;
		if (leftIndex == rightIndex) {
			System.out.println((right-left+1)*happ.get(leftIndex));
			return;
		}
		for (int i = leftIndex; i <= rightIndex; ++i) {
			if (i == leftIndex) {
				sum += 1L*(happ.get(leftIndex) - left + 1L) * happ.get(leftIndex);
			} else if (i != rightIndex){
				sum += 1L*(happ.get(i)*1L - happ.get(i-1))* happ.get(i);
			} else {
				sum += 1L*(right*1L - happ.get(i-1))* happ.get(i);
			}
		}
		System.out.println(sum);
	}
}
