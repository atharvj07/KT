#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
inline int add(int a, int b) {
  a += b;
  if (a >= mod) a -= mod;
  if (a < 0) a += mod;
  return a;
}
inline int mult(int a, int b) { return (long long)a * b % mod; }
inline int pot(int a, int b) {
  int ret = 1;
  while (b) {
    if (b & 1) ret = mult(ret, a);
    a = mult(a, a);
    b >>= 1;
  }
  return ret;
}
int main() {
  int n, dij;
  scanf("%d", &n);
  dij = pot(n, mod - 2);
  vector<int> svi(n);
  int a;
  for (int i = 0; i < (n); i++) {
    scanf("%d", &a);
    svi[i] = a;
  }
  sort(svi.begin(), svi.end());
  vector<int> pref(n + 1);
  pref[0];
  for (int i = 0; i < (n); i++) pref[i + 1] = add(pref[i], svi[i]);
  for (int k = 1; k <= n; k++) {
    int rj = 0;
    int cnt = 1;
    for (int i = n - k - 1; i >= 0; i -= k) {
      if (i - k < -1)
        rj = add(rj, mult(cnt, pref[i + 1]));
      else
        rj = add(rj, mult(cnt, add(pref[i + 1], -pref[i - k + 1])));
      cnt++;
    }
    printf("%d ", mult(rj, dij));
  }
  printf("\n");
  return 0;
}
