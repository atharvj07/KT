#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-7;
const int MAXN = 1e3 + 10;
const long long INF = 0x3f3f3f3f3f3f3f3f;
inline int dcmp(double x) {
  if (fabs(x) < eps) return 0;
  return (x > 0 ? 1 : -1);
}
inline double sqr(double x) { return x * x; }
inline double Sqrt(double x) { return x > 0 ? sqrt((double)x) : 0; }
struct Point {
  double x, y;
  int id;
  Point() { x = 0, y = 0; }
  Point(double _x, double _y) : x(_x), y(_y) {}
  inline void input() {
    double _x, _y;
    scanf("%lf%lf", &_x, &_y);
    x = _x, y = _y;
  }
  inline bool operator<(const Point &b) const {
    return (dcmp(x - b.x) == 0 ? y < b.y : x < b.x);
  }
  inline Point operator+(const Point &b) const {
    return Point(x + b.x, y + b.y);
  }
  inline Point operator-(const Point &b) const {
    return Point(x - b.x, y - b.y);
  }
  inline Point operator*(double a) { return Point(x * a, y * a); }
  inline Point operator/(double a) { return Point(x / a, y / a); }
  inline double len2() { return sqr(x) + sqr(y); }
  inline double len() { return Sqrt(len2()); }
};
inline double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
inline double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
double scan;
struct Line {
  Point s, e;
  Line() {}
  Line(Point _s, Point _e) : s(_s), e(_e) {}
  inline double gety(const double &x) const {
    double rad = (e.y - s.y) / (e.x - s.x + eps);
    return s.y + rad * (x - s.x);
  }
  inline bool operator<(const Line &l) const {
    return gety(scan) < l.gety(scan);
  }
};
inline int seg_cross_seg(Line a, Line v) {
  if (v.s.x < -1e19) return 0;
  int d1 = dcmp(cross(a.e - a.s, v.s - a.s));
  int d2 = dcmp(cross(a.e - a.s, v.e - a.s));
  int d3 = dcmp(cross(v.e - v.s, a.s - v.s));
  int d4 = dcmp(cross(v.e - v.s, a.e - v.s));
  if ((d1 ^ d2) == -2 && (d3 ^ d4) == -2) return 2;
  return (d1 == 0 && dcmp(dot(v.s - a.s, v.s - a.e)) <= 0) ||
         (d2 == 0 && dcmp(dot(v.e - a.s, v.e - a.e)) <= 0) ||
         (d3 == 0 && dcmp(dot(a.s - v.s, a.s - v.e)) <= 0) ||
         (d4 == 0 && dcmp(dot(a.e - v.s, a.e - v.e)) <= 0);
}
int n;
Line line[30000];
Point p[30000], vec[30000], tp[50000];
set<pair<Line, int> > s;
set<pair<Line, int> >::iterator it[30000];
inline set<pair<Line, int> >::iterator pre(
    const set<pair<Line, int> >::iterator &i) {
  return --set<pair<Line, int> >::iterator(i);
}
inline set<pair<Line, int> >::iterator nxt(
    const set<pair<Line, int> >::iterator &i) {
  return ++set<pair<Line, int> >::iterator(i);
}
inline bool has_inter(double len) {
  for (int i = 1; i <= n; i++) {
    Point l = p[i], r = p[i] + vec[i] * len;
    if (r < l) swap(l, r);
    line[i] = Line(l, r);
    tp[2 * i - 1] = l, tp[2 * i - 1].id = i;
    tp[2 * i] = r, tp[2 * i].id = -i;
  }
  sort(tp + 1, tp + 2 * n + 1, [](Point a, Point b) {
    return (dcmp(a.x - b.x) == 0 ? a.id > b.id : a.x < b.x);
  });
  s.clear();
  s.insert(pair<Line, int>(Line(Point(-1e20, -1e20), Point(1e20, -1e20)), 0));
  s.insert(pair<Line, int>(Line(Point(-1e20, 1e20), Point(1e20, 1e20)), 0));
  for (int i = 1; i <= 2 * n; i++) {
    scan = tp[i].x;
    int id = tp[i].id;
    if (id > 0) {
      it[id] = s.insert(pair<Line, int>(line[id], id)).first;
      if (seg_cross_seg(line[id], pre(it[id])->first) ||
          seg_cross_seg(line[id], nxt(it[id])->first))
        return 1;
    } else {
      id = -id;
      if (seg_cross_seg(pre(it[id])->first, nxt(it[id])->first)) return 1;
      s.erase(it[id]);
    }
  }
  return 0;
}
int main(void) {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    double s;
    p[i].input(), vec[i].input(), scanf("%lf", &s);
    vec[i] = vec[i] / vec[i].len() * s;
  }
  double l = 0, r = 1e10 + 1000;
  for (int i = 0; i < 70; i++) {
    double mid = (l + r) / 2;
    if (has_inter(mid))
      r = mid;
    else
      l = mid;
  }
  if (l > 1e10)
    puts("No show :(");
  else
    printf("%.12f\n", (double)l);
  return 0;
}
