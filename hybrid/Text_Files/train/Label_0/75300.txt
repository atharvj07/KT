def Check_List(l) -> bool:
	l.sort()
	for i in range(0, len(l) - 1):
		if l[i + 1] - 1 != l[i]:
			return False
	return True

def Find_Min(l) -> int:
	m = l[0]
	for i in range(0, len(l)):
		if l[i] < m:
			m = l[i]
	return m
n = int(input())
arr = list(map(int, input().split()))
l = [0]
for i in range(1, n):
	l.append(l[i - 1] + arr[i - 1])
k = Find_Min(l)
for i in range(0, len(l)):
	l[i] += 1 - k
ans = []
for i in range(0, len(l)):
	ans.append(l[i])
if Check_List(l) == True:
	for i in range(0, len(ans)):
		print(ans[i])
else:
	print(-1)
