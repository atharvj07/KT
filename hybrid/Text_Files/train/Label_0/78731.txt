(h, w) = map(int, input().split())
s = []
for _ in range(h):
	s.append(input())

def get_si_info(si):
	j1 = si.find('*')
	if j1 < 0:
		return (-1, -1)
	j2 = si.rfind('*')
	for j in range(j1 + 1, j2):
		if si[j] != '*':
			return (None, None)
	return (j1, j2)

def get_ans():
	if h < 3 or w < 3:
		return False
	i = 0
	mid_j = -1
	while i < h:
		(j1, j2) = get_si_info(s[i])
		if j1 is None or j1 != j2:
			return False
		if j1 != -1:
			mid_j = j1
			break
		i += 1
	if mid_j < 0:
		return False
	i += 1
	mid_i = -1
	while i < h:
		(j1, j2) = get_si_info(s[i])
		if j1 is None or j1 == -1:
			return False
		if j1 != mid_j or j2 != mid_j:
			if j1 >= mid_j or j2 <= mid_j:
				return False
			mid_i = i
			break
		i += 1
	if mid_i < 0:
		return False
	i += 1
	while i < h:
		(j1, j2) = get_si_info(s[i])
		if j1 is None:
			return False
		if j1 != mid_j or j2 != mid_j:
			break
		i += 1
	if i <= mid_i + 1:
		return False
	while i < h:
		(j1, j2) = get_si_info(s[i])
		if j1 is None:
			return False
		if j1 != -1 or j2 != -1:
			return False
		i += 1
	return True
print('YES' if get_ans() else 'NO')
