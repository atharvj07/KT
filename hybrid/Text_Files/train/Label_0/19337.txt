#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  while (cin >> s) {
    int a = s.find('H', 0);
    int b = s.find('Q', 0);
    int c = s.find('9', 0);
    if (a != string::npos || b != string::npos || c != string::npos)
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
