import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class D {
    static String m;
    static int[] count = new int[10];
    static Integer[][][] dp = new Integer[10][10][2];
    static long mod = (long) (1e9 + 7);

    public static int go(int index, int lucky, int can) {
        if (index == m.length()) {
            if (lucky == 0)
                return 1;
            return 0;
        }
        if (lucky < 0)
            return 0;
        if (dp[index][lucky][can] != null)
            return dp[index][lucky][can];
        int res = 0;
        if (can == 0) {
            int c = m.charAt(index) - '0';
            for (int i = 0; i <= c; i++) {
                int isLucky = (i == 4 || i == 7) ? 1 : 0;
                res += go(index + 1, lucky - isLucky, i < c ? 1 : 0);
            }
        } else {
            for (int i = 0; i <= 9; i++) {
                int isLucky = (i == 4 || i == 7) ? 1 : 0;
                res += go(index + 1, lucky - isLucky, 1);
            }
        }
        return dp[index][lucky][can] = res;
    }

    public static long get(int lucky, int index) {
        if (index == 7) {
            if (lucky > 0)
                return 1;
            return 0;
        }
        long res = 0;
        for (int i = 0; i < 10; i++) {
            if (count[i] > 0) {
                count[i]--;
                res += (count[i] + 1) * get(lucky - i, index + 1);
                res %= mod;
                count[i]++;
            }
        }
        return res;
    }

    public static void main(String[] args) {
        InputReader r = new InputReader(System.in);
        m = r.next();
        for (int i = 0; i < 10; i++) {
            count[i] = go(0, i, 0);
        }
        count[0]--;
//       for (int i = 0; i < 10; i++)
//       System.out.println(count[i]);
        long res = 0;
        for (int i = 0; i < 10; i++) {
            if (count[i] > 0) {
                count[i]--;
                res += (long)(count[i] + 1) * get(i, 1);
                res %= mod;
                count[i]++;
            }
        }
        System.out.println(res);
    }

    static class InputReader {
        private BufferedReader reader;
        private StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream));
            tokenizer = null;
        }

        public String nextLine() {
            try {
                return reader.readLine();
            } catch (IOException e) {
                // TODO Auto-generated catch block
                e.printStackTrace();
                return null;
            }
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }
    }
}
