#include <bits/stdc++.h>
using namespace std;
struct lifangti {
  long long x, y;
  bool friend operator<(lifangti a, lifangti b) {
    return a.x != b.x ? a.x < b.x : a.y < b.y;
  }
};
struct daan {
  long long h;
  int hao;
  bool friend operator<(daan a, daan b) { return a.h < b.h; }
};
map<lifangti, daan> s;
int n;
int a[110000][5];
int ans[6];
int aa[6];
long long MIN(long long a, long long b, long long c) {
  return min(min(a, b), c);
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d%d%d", &a[i][1], &a[i][2], &a[i][3]);
    sort(a[i] + 1, a[i] + 4);
  }
  ans[1] = a[1][1];
  ans[2] = a[1][2];
  ans[3] = a[1][3];
  ans[4] = 1;
  int k = 1;
  for (int i = 2; i <= n; i++) {
    if (MIN(ans[1], ans[2], ans[3]) < MIN(a[i][1], a[i][2], a[i][3])) {
      ans[1] = a[i][1];
      ans[2] = a[i][2];
      ans[3] = a[i][3];
      ans[4] = i;
    }
  }
  daan tempans;
  tempans.h = a[1][3];
  tempans.hao = 1;
  lifangti templi;
  templi.x = a[1][1];
  templi.y = a[1][2];
  s[templi] = tempans;
  tempans.h = a[1][2];
  tempans.hao = 1;
  templi.x = a[1][1];
  templi.y = a[1][3];
  s[templi] = tempans;
  tempans.h = a[1][1];
  tempans.hao = 1;
  templi.x = a[1][2];
  templi.y = a[1][3];
  s[templi] = tempans;
  for (int i = 2; i <= n; i++) {
    templi.x = a[i][1];
    templi.y = a[i][2];
    aa[1] = a[i][1];
    aa[2] = a[i][2];
    if (s[templi].hao) {
      aa[3] = s[templi].h + a[i][3];
      if (MIN(ans[1], ans[2], ans[3]) < MIN(aa[1], aa[2], aa[3])) {
        ans[1] = aa[1];
        ans[2] = aa[2];
        ans[3] = aa[3];
        ans[4] = i;
        ans[5] = s[templi].hao;
        k = 2;
      }
    }
    templi.y = a[i][3];
    aa[2] = a[i][3];
    if (s[templi].hao) {
      aa[3] = s[templi].h + a[i][2];
      if (MIN(ans[1], ans[2], ans[3]) < MIN(aa[1], aa[2], aa[3])) {
        ans[1] = aa[1];
        ans[2] = aa[2];
        ans[3] = aa[3];
        ans[4] = i;
        ans[5] = s[templi].hao;
        k = 2;
      }
    }
    templi.x = a[i][2];
    aa[1] = a[i][2];
    if (s[templi].hao) {
      aa[3] = s[templi].h + a[i][1];
      if (MIN(ans[1], ans[2], ans[3]) < MIN(aa[1], aa[2], aa[3])) {
        ans[1] = aa[1];
        ans[2] = aa[2];
        ans[3] = aa[3];
        ans[4] = i;
        ans[5] = s[templi].hao;
        k = 2;
      }
    }
    templi.x = a[i][1];
    templi.y = a[i][2];
    tempans.h = a[i][3];
    tempans.hao = i;
    if (s[templi].hao) {
      s[templi] = max(s[templi], tempans);
    } else {
      s[templi] = tempans;
    }
    templi.x = a[i][1];
    templi.y = a[i][3];
    tempans.h = a[i][2];
    tempans.hao = i;
    if (s[templi].hao) {
      s[templi] = max(s[templi], tempans);
    } else {
      s[templi] = tempans;
    }
    templi.x = a[i][2];
    templi.y = a[i][3];
    tempans.h = a[i][1];
    tempans.hao = i;
    if (s[templi].hao) {
      s[templi] = max(s[templi], tempans);
    } else {
      s[templi] = tempans;
    }
  }
  printf("%d\n", k);
  if (k == 1) {
    printf("%d\n", ans[4]);
  } else {
    printf("%d %d\n", ans[4], ans[5]);
  }
  return 0;
}
