#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, r;
  cin >> r >> n;
  vector<int> res[2];
  res[0].assign(r, 0);
  res[1].assign(r, 0);
  for (int i = 0; i < n; ++i) {
    int ri, ci;
    cin >> ri >> ci;
    --ri;
    --ci;
    res[ci][ri] = 1;
  }
  int dp[110][4];
  dp[0][0] = dp[0][1] = dp[0][2] = dp[0][3] = 0;
  dp[1][0] = 1;
  dp[1][1] = 1;
  dp[1][2] = 1;
  dp[1][3] = 0;
  int MAX = 110, MAX4 = 440;
  for (int len = 2; len < MAX; ++len) {
    for (int st = 0; st < 4; ++st) {
      vector<bool> used(MAX4, false);
      int gval, other;
      switch (st) {
        case 0:
          for (int ld = 0; ld < len; ++ld) {
            other = len - ld - 1;
            gval = dp[ld][1] ^ dp[other][1];
            if (gval < MAX4) {
              used[gval] = true;
            }
          }
          for (int i = 0; i < MAX4; ++i) {
            if (!used[i]) {
              dp[len][st] = i;
              break;
            }
          }
          break;
        case 1:
          for (int ld = 0; ld < len; ++ld) {
            other = len - ld - 1;
            gval = dp[ld][1] ^ dp[other][3];
            if (gval < MAX4 && other != 0) {
              used[gval] = true;
            }
            gval = dp[ld][1] ^ dp[other][2];
            if (gval < MAX4) {
              used[gval] = true;
            }
          }
          for (int i = 0; i < MAX4; ++i) {
            if (!used[i]) {
              dp[len][st] = i;
              break;
            }
          }
          break;
        case 2:
          for (int ld = 0; ld < len; ++ld) {
            other = len - ld - 1;
            gval = dp[ld][2] ^ dp[other][2];
            if (gval < MAX4) {
              used[gval] = true;
            }
            gval = dp[ld][3] ^ dp[other][3];
            if (gval < MAX4 && ld != 0 && other != 0) {
              used[gval] = true;
            }
          }
          for (int i = 0; i < MAX4; ++i) {
            if (!used[i]) {
              dp[len][st] = i;
              break;
            }
          }
          break;
        case 3:
          for (int ld = 0; ld < len; ++ld) {
            other = len - ld - 1;
            gval = dp[ld][2] ^ dp[other][3];
            if (other != 0 && gval < MAX4) {
              used[gval] = true;
            }
          }
          for (int i = 0; i < MAX4; ++i) {
            if (!used[i]) {
              dp[len][st] = i;
              break;
            }
          }
          break;
      }
    }
  }
  int prev = 3;
  int count = 0;
  int ans = 0;
  for (int row = 0; row < r; ++row) {
    if (res[0][row] == 0 && res[1][row] == 0) {
      ++count;
      continue;
    }
    if (res[1][row] == 1) {
      switch (prev) {
        case 1:
          ans ^= dp[count][2];
          break;
        case 2:
          ans ^= dp[count][3];
          break;
        case 3:
          ans ^= dp[count][1];
          break;
      }
      count = 0;
      prev = 1;
    }
    if (res[0][row] == 1) {
      switch (prev) {
        case 1:
          ans ^= dp[count][3];
          break;
        case 2:
          ans ^= dp[count][2];
          break;
        case 3:
          ans ^= dp[count][1];
          break;
      }
      count = 0;
      prev = 2;
    }
  }
  if (prev == 3) {
    ans ^= dp[count][0];
    count = 0;
    prev = 3;
  } else {
    ans ^= dp[count][1];
    count = 0;
    prev = 3;
  }
  if (ans == 0) {
    cout << "LOSE" << endl;
  } else {
    cout << "WIN" << endl;
  }
  return 0;
}
