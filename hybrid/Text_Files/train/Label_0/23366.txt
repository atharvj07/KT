#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
long long a[N];
long long n, m;
int main() {
  cin >> n >> m;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
  }
  a[n + 1] = 100000000000;
  long long l = 1, q = 0;
  long double ans = -1;
  for (long long i = 1; i <= n; i++) {
    while (a[i] - a[l] > m) {
      l++;
    }
    if (i - l >= 2) {
      q = 1;
      ans =
          max(ans, (long double)(a[i] - a[l + 1]) / (long double)(a[i] - a[l]));
    }
  }
  long long r = 1;
  for (int i = 1; i <= n; i++) {
    while (a[r + 1] - a[i] <= m) {
      r++;
    }
    if (r - i >= 2) {
      q = 1;
      ans =
          max(ans, (long double)(a[r] - a[i + 1]) / (long double)(a[r] - a[i]));
    }
  }
  if (!q) {
    cout << -1;
  } else {
    cout << fixed << setprecision(20) << ans;
  }
}
