#include<cstdio>
#include<cstring>
#include<string>
#include<iostream>
using namespace std;

int main()
{
	ios::sync_with_stdio(false);
	cin.tie();
	cout.tie();
	string s;
	cin>>s;
	int l=0;
	int r=s.size()-1;
	int ans=0;
	while(l<r)
	{
		if(s[l]==s[r]){
			l++;
			r--;
		}
		else{
			if(s[l]=='x')
			{
				l++;
				ans++;
			}
			else if(s[r]=='x')	
			{
				r--;
				ans++;
			}
			else
				break;
		}
	}
	if(l<r)
		cout<<"-1"<<endl;
	else
		cout<<ans<<endl;
	return 0;
}
