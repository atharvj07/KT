#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 10;
inline int read() {
  register int x = 0, f = 0;
  register char c = getchar();
  while (c < '0' || c > '9') f |= c == '-', c = getchar();
  while (c >= '0' && c <= '9') x = (x << 3) + (x << 1) + c - '0', c = getchar();
  return f ? -x : x;
}
int n, cnt;
double ans;
int head[N], siz[N];
double sum[N];
pair<double, double> p[N];
struct edge {
  int v, nex;
} e[N << 1];
inline void add(int u, int v) {
  e[++cnt].v = v;
  e[cnt].nex = head[u];
  head[u] = cnt;
  return;
}
void dfs(int x, int fa) {
  siz[x] = 1;
  sum[x] = p[x].first;
  double res = 0;
  for (int i = head[x], v; i; i = e[i].nex) {
    v = e[i].v;
    if (v == fa) continue;
    dfs(v, x);
    siz[x] += siz[v];
    sum[x] += sum[v];
    res += sum[v] * siz[v];
  }
  res += (1 - sum[x]) * (n - siz[x]);
  ans += res * p[x].second;
  return;
}
int main() {
  n = read();
  for (int i = 1, u, v; i < n; ++i)
    u = read(), v = read(), add(u, v), add(v, u);
  pair<double, double> tot = make_pair(0.0, 0.0);
  for (int i = 1; i <= n; ++i)
    scanf("%lf%lf", &p[i].first, &p[i].second), tot.first += p[i].first,
        tot.second += p[i].second;
  for (int i = 1; i <= n; ++i)
    p[i].first /= tot.first, p[i].second /= tot.second;
  dfs(1, 0);
  printf("%.12f\n", ans);
  return 0;
}
