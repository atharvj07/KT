def sp():

c = C(data())
(nb, ns, nc) = sp()
(pb, ps, pc) = sp()
r = int(data())
answer = 0
(low, high) = (0, int(1e+18))
while low <= high:
	mid = low + high >> 1
	(bc, sc, cc) = (pb * max(0, c['B'] * mid - nb), ps * max(0, c['S'] * mid - ns), pc * max(0, c['C'] * mid - nc))
	if bc + sc + cc <= r:
		answer = max(answer, mid)
		low = mid + 1
	else:
		high = mid - 1
out(answer)
