import math
a = list(map(int, input().split()))
n = a[0]
d = a[1]
o = []
g = int(input())
for i in range(g):
	k = list(map(int, input().split()))
	y1 = math.fabs(k[0]) - d
	if y1 <= k[1]:
		y2 = math.fabs(k[0] + d)
		if k[1] <= y2:
			y3 = d - k[0]
			if y3 <= k[1]:
				y4 = n - d + n - k[0]
				if y4 >= k[1]:
					k = 1
					o.append('YES')
				else:
					o.append('NO')
			else:
				o.append('NO')
		else:
			o.append('NO')
	else:
		o.append('NO')
for i in range(g):
	print(o[i])
