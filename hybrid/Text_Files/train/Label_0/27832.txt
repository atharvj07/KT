#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int start, end, tt;
  cin >> tt;
  while (tt) {
    cin >> start >> end;
    int n = end + 1 - start;
    int i, a, b;
    i = start;
    long long result;
    int sa = 1;
    int sb = -1;
    if (i % 2 != 0) {
      sa = -1;
      sb = 1;
    }
    if (n % 2 == 0) {
      a = sa * n / 2 * i;
      b = sb * n / 2 * (i + 1);
      result = a + b;
    } else {
      result = sa * (i + (n - 1) / 2);
    }
    cout << result << endl;
    tt--;
  }
}
