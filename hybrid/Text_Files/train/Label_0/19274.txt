#include <bits/stdc++.h>
using namespace std;
string s1, s2;
map<char, char> mp;
map<pair<char, char>, int> ans;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> s1 >> s2;
  for (int i = 0; i < s1.size(); i++) {
    char a = s1[i], b = s2[i];
    if (mp[a] == 0) mp[a] = b;
    if (mp[b] == 0) mp[b] = a;
    if (mp[a] != b || mp[b] != a) {
      cout << -1;
      return 0;
    }
  }
  for (auto it = mp.begin(); it != mp.end(); ++it) {
    char a = it->first, b = it->second;
    if (a > b) swap(a, b);
    if (a == b) continue;
    pair<char, char> p = make_pair(a, b);
    ans[p]++;
  }
  cout << ans.size() << endl;
  for (auto it = ans.begin(); it != ans.end(); ++it) {
    cout << it->first.first << ' ' << it->first.second << endl;
  }
  return 0;
}
