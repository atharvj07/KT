import java.io.BufferedReader;
import java.io.IOException;

import java.util.ArrayList;

import java.util.HashSet;

import java.util.List;
import java.util.Queue;
import java.util.Scanner;
import java.util.Set;

import java.util.function.Supplier;

public class Main {
    
    
    
    static List<Integer> topologicalOrder(List<List<Integer>> g, int[] inDegree) {
        int n = g.size();
        List<Integer> order = new ArrayList<Integer>();
        Queue<Integer> q = new LinkedList<Integer>();
        for(int i = 0; i< n; i++) if(inDegree[i] == 0) q.add(i);
        while(!q.isEmpty()) {
            int v = q.poll();
            order.add(v);
            for(int neighbor : g.get(v)) {
                inDegree[neighbor]--;
                if(inDegree[neighbor] == 0) q.add(neighbor);
            }
        }
        return order;
    }
  
    
    public static void main(String[] args) {
        MyScanner sc = new MyScanner();
        
        int n = sc.nextInt();
        int k = sc.nextInt();
        
        int[] prereq = new int[n];
        
        for(int i = 0; i < n; i++) {
            prereq[i] = sc.nextInt() - 1;
        }
        
        int[] predecessor = new int[n];
        boolean[] isPredecessor = new boolean[n];
        
        Arrays.fill(predecessor, -1);
        
        boolean possible = true;

        for(int i  = 0; i < k; i++) {
            int before = sc.nextInt() - 1;
            int after  = sc.nextInt() - 1;
            if(isPredecessor[before]) possible = false;
            isPredecessor[before] = true;
            predecessor[after] = before; 
        }
        
        if(!possible) {
            System.out.println(0);
            return;
        }

        int[] componentId = new int[n];
        Arrays.fill(componentId, -1);
        int currentId = 0;

        List<List<Integer>> components = new ArrayList<List<Integer>>();
        for(int i = 0; i < n; i++) {
            components.add(new ArrayList<Integer>());
            if(componentId[i] == -1 && !isPredecessor[i]) {
                int idx = i;
                while(idx != -1) {
                    componentId[idx] = currentId;
                    components.get(currentId).add(idx);
                    idx = predecessor[idx];
                }
                currentId++;
            }
        }
        if(Arrays.stream(componentId).anyMatch(x -> x == -1)) {
            System.out.println(0);
            return;
        }
        
        List<List<Integer>> list = new ArrayList<List<Integer>>();
        for(int i = 0; i< currentId; i++) list.add(new ArrayList<Integer>());
        
        int[] inDegree = new int[currentId];
        for(int i = 0; i < n; i++) {
            if(prereq[i] == -1) continue;
            int before = componentId[prereq[i]];
            int after = componentId[i];
       
            if(before != after) {
                inDegree[after]++;
                list.get(before).add(after);
            }
        }
        List<Integer> order = topologicalOrder(list, inDegree);
        if(order.size() < currentId) {
            System.out.println(0);
            return;
        }
        int[] finalOrder = new int[n];
        int[] pos = new int[n];
        int currentIdx = 0;
       
        for(int comp : order) {      
            for(int i = components.get(comp).size() - 1; i >= 0; i--) finalOrder[currentIdx++] = components.get(comp).get(i) + 1;
        }
        for(int i = 0; i < n; i++) {
            pos[finalOrder[i] - 1] = i;
        }
        for(int i = 0; i < n; i++) {
            if(prereq[i] != -1 && pos[prereq[i]] > pos[i]) possible = false;
        }
        if(!possible) {
            System.out.println(0);
            return;
        }
        for(int i = 0; i < n; i++) {
            System.out.print(finalOrder[i] + " ");
        }
    }
    

static class MyScanner {
    BufferedReader br;
    StringTokenizer st;

    public MyScanner() {
       br = new BufferedReader(new InputStreamReader(System.in));
    }

    String next() {
        while (st == null || !st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }

    double nextDouble() {
        return Double.parseDouble(next());
    }

    String nextLine(){
        String str = "";
    try {
       str = br.readLine();
    } catch (IOException e) {
       e.printStackTrace();
    }
    return str;
    }

 }
