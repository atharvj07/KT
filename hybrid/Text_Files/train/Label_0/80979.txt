import sys

def greedyCalc(n, c0, c1, h, binInput):
	totalCost = 0
	for c in binInput:
		if c == '0':
			curCost = c0
			changedCost = c1 + h
		else:
			curCost = c1
			changedCost = c0 + h
		if changedCost < curCost:
			totalCost = totalCost + changedCost
		else:
			totalCost = totalCost + curCost
	return totalCost
testcase_flag = True
count = 0
testCaseCount = 0
costs = []
for line in sys.stdin:
	inputs = [x for x in line.strip().split()]
	if testcase_flag:
		testCaseCount = int(inputs[0])
		testcase_flag = False
		continue
	if count % 2 == 0:
		n = int(inputs[0])
		c0 = int(inputs[1])
		c1 = int(inputs[2])
		h = int(inputs[3])
		count = count + 1
		continue
	if count % 2 == 1:
		binaryInput = inputs[0]
		count = count + 1
		cost = greedyCalc(n, c0, c1, h, binaryInput)
		costs.append(cost)
		if count == testCaseCount * 2:
			break
for i in range(testCaseCount):
	print(costs[i])
