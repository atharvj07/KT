import sys
input = sys.stdin.readline

def good(string, left, right):
	if left > 1 and string[left - 1] in string[0:left - 1]:
		return True
	elif right < len(string) and string[right - 1] in string[right:]:
		return True
	else:
		return False
tests = int(input())
for test in range(tests):
	(stringLength, testcases) = map(int, input().split())
	string = input()
	for testcase in range(testcases):
		(left, right) = map(int, input().split())
		if good(string, left, right):
			print('YES')
		else:
			print('NO')
