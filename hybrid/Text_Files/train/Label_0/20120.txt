#include <bits/stdc++.h>
using namespace std;
const int MAX = 201000;
int n;
set<pair<int, int> > s;
vector<int> sol;
bool exist(int x) {
  pair<int, int> tmp = {x, 0};
  set<pair<int, int> >::iterator it = s.lower_bound(tmp);
  if (it == s.end()) return false;
  return it->first == x;
}
int take(int x) {
  pair<int, int> tmp = {x, 0};
  set<pair<int, int> >::iterator it = s.lower_bound(tmp);
  int y = it->second;
  s.erase(it);
  return y;
}
bool solve() {
  int cnt = 0;
  while (cnt >= 0) {
    if (exist(cnt)) {
      sol.push_back(take(cnt));
      ++cnt;
    } else
      cnt -= 3;
  }
  return !s.size();
}
int main() {
  cin >> n;
  pair<int, int> tmp;
  for (int i = 0; i < n; ++i) {
    cin >> tmp.first;
    tmp.second = i + 1;
    s.insert(tmp);
  }
  if (solve()) {
    cout << "Possible\n";
    for (int i = 0; i < n; ++i) cout << sol[i] << " ";
  } else
    cout << "Impossible\n";
}
