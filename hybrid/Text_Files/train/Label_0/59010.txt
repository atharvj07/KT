/**
 * Built using CHelper plug-in
 * Actual solution is at the top

public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        BMinimumPath solver = new BMinimumPath();
        solver.solve(1, in, out);
        out.close();
    }

    static class BMinimumPath {
        public int[][] toCost(char[][] board, int n) {
            int[][] res = new int[n][n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    res[i][j] = board[i][j] == 'a' ? 0 : 1;
                }
            }
            return res;
        }

        public int[][] toMinCost(int[][] cost, int n) {
            for (int d = 1; d < n * 2 - 1; d++) {
                int i = Math.min(d, n - 1);
                int j = d - i;
                for (; i >= 0 && j < n; i--, j++) {
                    int option = 1_000_000_000;
                    if (i != 0) {
                        option = Math.min(option, cost[i - 1][j]);
                    }
                    if (j != 0) {
                        option = Math.min(option, cost[i][j - 1]);
                    }
                    cost[i][j] += option;
                }
            }
            return cost;
        }

        public Pair<Integer, ArrayList<Integer>> possibleStartPoint(int[][] minK, int n, int k) {
            ArrayList<Integer> res = new ArrayList<>();
            for (int d = n * 2 - 2; d >= 0; d--) {
                int i = Math.min(d, n - 1);
                int j = d - i;
                for (; i >= 0 && j < n; i--, j++) {
                    if (minK[i][j] <= k) {
                        res.add(i);
                    }
                }
                if (!res.isEmpty()) {
                    return new Pair<>(d, res);
                }
            }
            return null;
        }

        boolean isValid(int i, int j, int n) {
            return 0 <= i && i < n && 0 <= j && j < n;
        }

        public String getBestFromOption(char[][] board, ArrayList<Integer> options, int d, int n) {
            ArrayList<Integer> nextOption = new ArrayList<>();
            StringBuilder sb = new StringBuilder();
            BitSet bs = new BitSet(n * n);
            bs.set(0, true);
            while (true) {
                char bestChar = 'z' + 1;
                for (int i : options) {
                    int j = d - i;
                    for (int rot = 0; rot < 2; rot++) {
                        int i1 = i + (rot ^ 1);
                        int j1 = j + rot;
                        if (isValid(i1, j1, n)) {
                            if (bs.get(i1 * n + j1)) continue;
                            bs.flip(i1 * n + j1);
                            if (board[i1][j1] <= bestChar) {
                                if (board[i1][j1] < bestChar) nextOption.clear();
                                bestChar = board[i1][j1];
                                nextOption.add(i1);
                            }
                        }
                    }
                }
                if (nextOption.isEmpty()) return sb.toString();
                sb.append(bestChar);
                options.clear();
                ArrayList<Integer> temp = nextOption;
                nextOption = options;
                options = temp;
                d++;
            }
        }

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.NextInt();
            int k = in.NextInt();
            char[][] board = new char[n][];
            for (int i = 0; i < n; i++) {
                board[i] = in.next().toCharArray();
            }
            Pair<Integer, ArrayList<Integer>> optionalStarts = possibleStartPoint(toMinCost(toCost(board, n), n), n, k);
            if (optionalStarts == null) {
                out.print(board[0][0]);
                optionalStarts = new Pair<>(0, new ArrayList<>());
                optionalStarts.second.add(0);
            } else {
                for (int i = 0; i <= optionalStarts.first; i++) {
                    out.print('a');
                }
            }
            out.println(getBestFromOption(board, optionalStarts.second, optionalStarts.first, n));
        }

    }

    static class Pair<T1, T2> {
        public T1 first;
        public T2 second;

        public Pair(T1 f, T2 s) {
            first = f;
            second = s;
        }

    }

    static class InputReader {
        BufferedReader reader;
        StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine(), " \t\n\r\f,");
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int NextInt() {
            return Integer.parseInt(next());
        }

    }
}
