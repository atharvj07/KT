#include <bits/stdc++.h>

using namespace std;

char st[107010];
int l,r,ans;
int main() {
	scanf("%s",st);
	int l=0,r=strlen(st)-1;
	while(l<=r){
		if(st[l]==st[r]){
			l++;r--;
			continue;
		}
		if(st[l]=='x'){
			l++;
			ans++;
			continue;
		}
		if(st[r]=='x'){
			r--;
			ans++;
			continue;
		}
		printf("%d\n",-1);
		return 0;
	}
	printf("%d\n",ans);
    return 0;
}
