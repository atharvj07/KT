import sys
input = sys.stdin.readline

def solve():
	s = input().strip()
	t = input().strip()
	(n, m) = (len(s), len(t))
	(even, odd) = (-1, -1)
	for i in range(n):
		if t[0] == s[i]:
			if i % 2 == 0:
				if even == -1:
					even = i
			elif odd == -1:
				odd = i
		if even != -1 and odd != -1:
			break

	def possible(i):
		j = 0
		L = 0
		while i < n:
			to_skip = False
			if j < m and s[i] == t[j] and (L % 2 == 0):
				j += 1
				L = 0
				to_skip = True
			i += 1
			if not to_skip:
				L += 1
		return j == m and L % 2 == 0
	ans = False
	if even != -1:
		ans = ans or possible(even)
	if odd != -1:
		ans = ans or possible(odd)
	if ans:
		print('YES')
	else:
		print('NO')
	return -1
for nt in range(int(input())):
	solve()
