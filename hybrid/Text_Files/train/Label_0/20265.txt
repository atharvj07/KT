#include <bits/stdc++.h>
using namespace std;
int num[1111];
int n;
int check(int p, int x, int &S, int &R) {
  int top = num[0] + x, down = num[0] - x;
  for (int i = 1; i < n; i++) {
    int itop = num[i] + x - p * i;
    int idown = num[i] - x - p * i;
    if (down > itop) return 2;
    if (top < idown) return 1;
    top = min(top, itop);
    down = max(down, idown);
  }
  R = p;
  S = down;
  return 0;
}
bool C(int x, int &S, int &R) {
  int lb = 0, ub = num[1] + x - (num[0] - x) + 1;
  while (ub - lb > 1) {
    int m = (lb + ub) >> 1;
    int res = check(m, x, S, R);
    if (res == 1)
      lb = m;
    else if (res == 2)
      ub = m;
    else
      return true;
  }
  return !check(lb, x, S, R);
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> num[i];
  sort(num, num + n);
  int lb = -1, ub = 10001;
  int S, R;
  while (ub - lb > 1) {
    int m = (lb + ub) >> 1;
    if (C(m, S, R))
      ub = m;
    else
      lb = m;
  }
  cout << ub << endl;
  cout << S << " " << R << endl;
  return 0;
}
