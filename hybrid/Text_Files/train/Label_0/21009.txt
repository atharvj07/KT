const int N = 3005;

const int inf = -1e7;

int a[N],init_val[N];

void init(int *seg,int *lazy,int pos,int l,int r) {
  lazy[pos] = 0;
  if(l==r) {
    seg[pos] = init_val[l];
    return;
  }
  int mid = (l+r)/2;
  init(seg,lazy,pos*2,l,mid);
  init(seg,lazy,pos*2+1,mid+1,r);
  seg[pos] = max(seg[pos*2],seg[pos*2+1]);
}

void push(int *seg,int *lazy,int pos,int l,int r) {
  if(lazy[pos]!=0) {
    seg[pos] += lazy[pos];
    if(l!=r) {
      lazy[pos*2] += lazy[pos];
      lazy[pos*2+1] += lazy[pos];
    }
    lazy[pos] = 0;
  }
}

void update(int *seg,int *lazy,int pos,int l,int r,int ql,int qr,int v) {
  push(seg,lazy,pos,l,r);
  if(r<ql || qr<l) return;
  if(ql<=l && r<=qr) {
    lazy[pos] = v;
    push(seg,lazy,pos,l,r);
    return;
  }
  int mid = (l+r)/2;
  update(seg,lazy,pos*2,l,mid,ql,qr,v);
  update(seg,lazy,pos*2+1,mid+1,r,ql,qr,v);
  seg[pos] = max(seg[pos*2],seg[pos*2+1]);
}

ll query(int *seg,int *lazy,int pos,int l,int r,int ql,int qr) {
  push(seg,lazy,pos,l,r);
  if(r<ql || qr<l) return inf;
  if(ql<=l && r<=qr) {
    return seg[pos];
  }
  int mid = (l+r)/2;
  return max(query(seg,lazy,pos*2,l,mid,ql,qr),query(seg,lazy,pos*2+1,mid+1,r,ql,qr));
}

int seg[N][N*4],lazy[N][N*4];

void solve2() {
  int n;
  scanf("%d ", &n);
  for(int i=1;i<=n;++i) {
    scanf("%d ", &a[i]);
  }

  int value = 0;
  for(int i= n-1;i;--i) {
    for(int j=1;j<=n;++j) init_val[j] = inf;
    if(a[i]+i==n) {
      init_val[n] = 1 + value;
    } else {
      for(int j=i+1;j<=a[i]+i;++j) {
        int g = query(seg[j],lazy[j],1,1,n,a[i]+i+1,n) + 1;
        init_val[j] = g;
      }
      for(int j=a[i]+i+1;j<n;++j) {
        update(seg[j],lazy[j],1,1,n,1,n,1);
      }
      ++value;
    }
    init(seg[i],lazy[i],1,1,n);
  }
  printf("%d\n", n-1-query(seg[1],lazy[1],1,1,n,1,n));
}

void solve() {
  int n;
  scanf("%d ", &n);
  while(n--) {
    solve2();
  }
}

int main() {
  //freopen("input.txt","r",stdin);
  solve();
  return 0;
}
