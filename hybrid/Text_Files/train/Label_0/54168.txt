import java.util.*;
public class Main {

    public static void main(String[] args) {
        Scanner sc= new Scanner(System.in);
        int t=sc.nextInt();
        while (t!=0){
            int x = sc.nextInt();
            int n = sc.nextInt();
            int m = sc.nextInt();
          while(true){
            if(x<=0)
             break;
            else if(n<=0 && m<=0)
              break;
            else if(m==0 && n>0 && x<=19)
              break;

            if(x>=40 && n>0) {
               x = x / 2 + 10;
               n--;
           }
            else if(m<5 && n>0 && x>19){
                x = x /2 + 10;
                n--;
            }
            else {
               x = x - 10;
               m--;

           }


          }
          if(x<=0)
              System.out.println("YES");
          else
              System.out.println("NO");
            t--;
        }

    }
}
