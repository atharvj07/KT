(a, d) = map(float, input().split())
n = int(input())

def pt(k):
	l = d * k
	c = int(l / a)
	l -= a * c
	c %= 4
	if c == 0:
		return (l, 0)
	elif c == 1:
		return (a, l)
	elif c == 2:
		return (a - l, a)
	else:
		return (0, a - l)
print('\n'.join((str(t[0]) + ' ' + str(t[1]) for t in map(pt, range(1, n + 1)))))
