one = 0

def assign(m):
	global one
	if a[m - 1] == 1:
		a[m - 1] = 0
		one = one - 1
	else:
		a[m - 1] = 1
		one = one + 1

def show(n):
	if n <= one:
		print('1')
	else:
		print('0')
(n, q) = [int(g) for g in input().split()]
a = [int(h) for h in input().split()]
for item in a:
	if item == 1:
		one += 1
for i in range(q):
	(t, x) = [int(j) for j in input().split()]
	if t == 1:
		assign(x)
	if t == 2:
		show(x)
