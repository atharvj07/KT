import java.util.ArrayList;
import java.util.Scanner;




public class C {
	static ArrayList<Long> primes;
	static long u=2000;
	static void sieve(){
		primes=new ArrayList<Long>();
		u++;
		ArrayList<Boolean> bs=new ArrayList<Boolean>();
		for(int i=0;i<=u;i++){
			bs.add(true);
		}
		bs.set(0, false);
		bs.set(1, false);
		for(int i=2;i<u;i++){
			if(bs.get(i)){
				for(int j=i*i;j<u;j+=i){
					bs.set(j, false);
				}
				primes.add((long) i);
			}
		}
	}
	
	public static void main(String[] args) {
		sieve();
		
		Scanner sc = new Scanner(System.in);
		
		int n = sc.nextInt();
		int count = 0;
		ArrayList<Long> a = new ArrayList<Long>();
		long prime = primes.get(0);
		
		for(int exp=1;exp<13;exp++){
			for(int i=0;Math.pow(primes.get(i), exp)<=n;i++){
				a.add((long) Math.pow(primes.get(i), exp));
				count++;
			}
		}
		
		
	/*	int z = 4;
		while( z < n){
			a.add(z);
			z = z*2;
			count++;
		}*/
		
		System.out.println(count);
		for(int i = 0; i<count; i++){
			System.out.print(a.get(i));
			System.out.print(' ');
		}

		
		
	}
}
