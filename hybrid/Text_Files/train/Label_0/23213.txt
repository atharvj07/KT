#include <bits/stdc++.h>
using namespace std;
template <typename T>
ostream &operator<<(ostream &cout, vector<T> &a) {
  for (size_t i = 0; i < a.size(); ++i) cout << a[i] << " ";
  return cout;
}
template <typename T>
ostream &operator<<(ostream &cout, vector<vector<T> > &a) {
  for (size_t i = 0; i < a.size(); ++i) cout << a[i] << endl;
  return cout;
}
int t[100000];
int dp[100001];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> t[i];
  }
  dp[0] = 0;
  dp[1] = 20;
  cout << dp[1] << endl;
  int *x, *y;
  for (int i = 2; i <= n; ++i) {
    x = upper_bound(t, t + n, t[i - 1] - 90);
    y = upper_bound(t, t + n, t[i - 1] - 1440);
    dp[i] = min(dp[i - 1] + 20, min(dp[x - t] + 50, dp[y - t] + 120));
    cout << max(dp[i] - dp[i - 1], 0) << endl;
  }
  return 0;
}
