#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i = (a); i < (b); ++i)
#define rrep(i,a,b) for(int i = (b); i --> (a);)
#define all(v) (v).begin(),(v).end()
#define trav(x,v) for(auto &x : v)
#define sz(v) int(v.size())
typedef vector<int> vi;
typedef long long ll;
typedef pair<int,int> pii;

int main(){
	cin.sync_with_stdio(0);
	cin.tie(0);

	vector<string> seven = {
		"aabbcc.",
		"ddee..l",
		"ffgg..l",
		"....hjm",
		"....hjm",
		"....ikn",
		"....ikn"
	};
	vector<string> four = {
		"aaef",
		"ccef",
		"bdgg",
		"bdhh"
	};
	vector<string> five = {
		"ag..h",
		"ag..h",
		"biijj",
		"b.ffe",
		"ccdde" 
	};

	int n;
	cin >> n;

	vector<string> ans(n, string(n, '.'));

	if(n == 2){
		cout << -1 << endl;
	} else {
		if(n % 3 == 0){
			rep(i,0,n/3){
				rep(t,0,2) ans[i*3 + t][i * 3] = 'a';
				rep(t,0,2) ans[i*3 + 2][i * 3 + t + 1] = 'a';
			}
		} else {
			vi pek(n+10);
			pek[0] = 1;
			rep(i,0,n) if(pek[i]){
				vi v = {4,5,7};
				trav(x,v) pek[i+x] = x;
			}
			while(n){
				int x = pek[n];
				n -= x;
				auto ls = x==4 ? four : x == 5 ? five : seven;
				rep(i,0,sz(ls)) rep(j,0,sz(ls)) ans[n+i][n+j] = ls[i][j];
			}
		}
		trav(s, ans) cout << s << endl;
	}
}
