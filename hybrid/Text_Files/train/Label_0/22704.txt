#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  ll n, ans = 0;
  cin >> n;
  map<int, int> mp;

  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    ans += mp[a[i]];
    mp[a[i]]++;
  }

  for (int k = 0; k < n; k++) cout << ans - (mp[a[k]] - 1) << endl;
}
