import java.io.*;
import java.util.*;

public class CF932F {
    static int MIN_X = (int)-1e5 - 5, MAX_X = (int)1e5 + 5;
    static int N, A[], B[];
    static long ans[];
    static ArrayList<Integer> g[];
    static ArrayList<Line> lines[];
    static LiChaoNode roots[];

    public static void main(String args[]) throws Exception {
        FastReader fr = new FastReader(System.in);
        PrintWriter pw = new PrintWriter(System.out);

        int N = fr.nextInt();
        A = new int[N];
        B = new int[N];
        for (int i = 0; i < N; i++) A[i] = fr.nextInt();
        for (int i = 0; i < N; i++) B[i] = fr.nextInt();
        g = new ArrayList[N];
        for (int i = 0; i < N; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < N - 1; i++) {
            int u = fr.nextInt() - 1, v = fr.nextInt() - 1;
            g[u].add(v); g[v].add(u);
        }

        lines = new ArrayList[N];
        roots = new LiChaoNode[N];
        ans = new long[N];
        dfs(0, 0);

        for (int i = 0; i < N; i++)
            pw.print(ans[i] + " ");

        pw.close();
    }

    static int dfs(int u, int p) {
        int h = u;
        roots[h] = new LiChaoNode(MIN_X, MAX_X, null);
        lines[h] = new ArrayList<Line>();
        for (int v : g[u]) if (v != p) {
            int h1 = dfs(v, u);
            if (lines[h].size()  < lines[h1].size()) { int t = h; h = h1; h1 = t; }
            for (Line l : lines[h1]) {
                roots[h].insert(l);
                lines[h].add(l);
            }
        }
        if (h != u) ans[u] = roots[h].query(A[u]);
        Line l = new Line(B[u], ans[u]);
        roots[h].insert(l);
        lines[h].add(l);
        return h;
    }
}

class LiChaoNode {
    int s, e, m;
    Line cur;
    LiChaoNode lt, rt;

    LiChaoNode(int ss, int ee, Line l) {
        s = ss; e = ee; m = (s + e) >> 1; cur = l;
    }

    void insert(Line l) {
        if (cur == null) {
            cur = l;
            return;
        }
        boolean b1 = l.eval(s) < cur.eval(s);
        boolean b2 = l.eval(m) < cur.eval(m);
        if (b2) { Line t = l; l = cur; cur = t; }
        if (s == e) return;
        if (b1 != b2) {
            if (lt == null) lt = new LiChaoNode(s, m, l);
            else lt.insert(l);
        } else {
            if (rt == null) rt = new LiChaoNode(m + 1, e, l);
            else rt.insert(l);
        }
    }

    long query(int x) {
        long y = cur.eval(x);
        if (x < m && lt != null) y = Math.min(y, lt.query(x));
        if (x > m && rt != null) y = Math.min(y, rt.query(x));
        return y;
    }
}

class Line {
    long m, c;
    Line(long mm, long cc) { m = mm; c = cc; }
    long eval(long x) { return m * x + c; }
}

class FastReader{
    byte[] buf = new byte[2048];
    int index, total;
    InputStream in;

    FastReader(InputStream is) {
        in = is;
    }

    int scan() throws IOException {
        if(index >= total){
            index = 0;
            total = in.read(buf);
            if(total <= 0)
                return -1;
        }
        return buf[index++];
    }

    int nextInt() throws IOException {
        int c, val = 0;
        for(c = scan(); c <= 32; c = scan());
        boolean neg = c == '-';
        if(c == '-' || c == '+')
            c = scan();
        for(; c >= '0' && c <= '9'; c = scan())
            val = (val << 3) + (val << 1) + (c & 0xf);
        return neg ? -val : val;
    }
}
