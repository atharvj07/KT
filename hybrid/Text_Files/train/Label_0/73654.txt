import sys
read = lambda : map(int, sys.stdin.buffer.readline().split())
(n, a, b) = read()
v = [tuple(read()) for _ in range(n)]
ans = 0
if b > 0:
	c = [v[x][0] - v[x][1] for x in range(n)]
	(w, r) = (list(range(n)), [0] * n)
	w.sort(key=lambda x: c[x], reverse=True)
	for i in range(n):
		r[w[i]] = i
	f = True
	s = 0
	m = min(n, b)
	for i in range(m):
		k = c[w[i]]
		if k <= 0:
			f = False
			m = i
			break
		s += k
	ans = s
	if a > 0:
		for i in range(n):
			k = v[i][0] * (1 << a) - v[i][1]
			tmp = s + k
			if r[i] < m:
				tmp -= c[i]
			elif f:
				tmp -= c[w[m - 1]]
			if tmp > ans:
				ans = tmp
print(sum([v[i][1] for i in range(n)]) + ans)
