#include <bits/stdc++.h>
const int inf = 1e9;
const int maxn = 1e5;
int a[maxn], b[maxn];
int main() {
  int n, m, tmp;
  std::scanf("%d%d", &n, &tmp);
  for (int i = 0; i ^ n; ++i) std::scanf("%d", a + i);
  std::scanf("%d%d", &m, &tmp);
  for (int i = 0; i ^ m; ++i) std::scanf("%d", b + i);
  std::map<int, int> cnt;
  int res = 0;
  for (int mod = 1, i = 0; i ^ 30; mod = mod << 1 ^ 1, ++i) {
    cnt.clear();
    for (int i = 0; i ^ n; ++i) ++cnt[a[i] & mod];
    for (int i = 0; i ^ m; ++i) ++cnt[b[i] + (mod + 1 >> 1) & mod];
    for (auto j : cnt) res = std::max(res, j.second);
  }
  if (m == 1 && n == 1 && a[0] == b[0]) res = std::max(res, 2);
  std::printf("%d\n", res);
}
