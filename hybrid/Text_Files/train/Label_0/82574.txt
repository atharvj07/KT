import sys
s = sys.stdin.read()

def splitter(s):
	return s.split('\n')[1].split(' ')

def solution(s):
	count = 0
	minus_signs = 0
	zeroes = 0
	for i in s:
		i = int(i)
		if i != 0:
			count += abs(i) - 1
			if i < 0:
				minus_signs += 1
		else:
			count += 1
			zeroes += 1
	if zeroes == 0 and minus_signs % 2 == 1:
		count += 2
	return str(count)
lst = splitter(s)
x = solution(lst)
print(x)
