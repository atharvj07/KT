(n, k) = map(int, input().split())
dp = [0 for i in range(k + 1)]
mod = 10 ** 9 + 7
dp[0] = 1
for i in range(k):
	for j in range(1, k + 1)[::-1]:
		dp[j] = (dp[j] * j + dp[j - 1] * (n - j + 1)) % mod
	dp[0] = 0
ans = 0
for i in range(k + 1):
	if n < i:
		continue
	ans += dp[i] * pow(2, n - i, mod)
	ans %= mod
print(ans)
