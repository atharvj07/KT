t = int(input())
for test in range(t):
	n = int(input())
	matrix = [['='] * n for i in range(n)]
	for i in range(n):
		matrix[i][i] = 'X'
	exp = input()
	done = False
	for c in range(n):
		if exp[c] == '2':
			for i in range(n):
				if c != i and exp[i] == '2' and (matrix[i][c] == '='):
					matrix[i][c] = '-'
					matrix[c][i] = '+'
					break
			else:
				done = True
				print('NO')
	if not done:
		print('YES')
		for line in matrix:
			print(''.join(line))
