#include <bits/stdc++.h>
const long long ha = 1e9 + 7;
const int N = 1000005;
long long n, m, ans, dp[N], f[N], inv[N], pre[N];
inline void read(long long &x) {
  register char ch = 0;
  for (; !std::isdigit(ch); ch = getchar())
    ;
  for (x = 0; std::isdigit(ch); x = x * 10 + ch - '0', ch = getchar())
    ;
}
inline long long C(const int &n, const int &m) {
  return f[n] * inv[n - m] % ha * inv[m] % ha;
}
int main() {
  read(n), read(m);
  inv[0] = inv[1] = 1, f[0] = f[1] = 1;
  for (register int i = 2; i <= n; ++i)
    f[i] = f[i - 1] * i % ha, inv[i] = (ha - ha / i) * inv[ha % i] % ha;
  for (register int i = 2; i <= n; ++i) inv[i] = inv[i - 1] * inv[i] % ha;
  dp[0] = pre[0] = 1;
  for (register int i = 1; i <= n; ++i)
    dp[i] =
        (f[i - 1] * (pre[i - 1] - (i > m ? pre[i - m - 1] : 0)) % ha + ha) % ha,
    pre[i] = (pre[i - 1] + (dp[i] * inv[i])) % ha;
  for (register int i = 1; i <= n; ++i)
    ans = (ans + dp[i - 1] * f[n - 1] % ha * inv[i - 1]) % ha;
  printf("%d\n", (int)((f[n] - ans + ha) % ha));
  return 0;
}
