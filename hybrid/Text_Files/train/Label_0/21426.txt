#include <bits/stdc++.h>
using namespace std;
long long int dp[20][4][2];
string s;
long long int solve(long long int pos, long long int cnt, long long int tight) {
  if (pos == s.length()) {
    return 1;
  }
  if (dp[pos][cnt][tight] != -1) {
    return dp[pos][cnt][tight];
  }
  long long int end = ((tight) ? (s[pos] - '0') : 9);
  long long int ans = 0;
  for (long long int i = 0; i <= end; i++) {
    long long int cntupdt = cnt + (i > 0);
    if (cntupdt <= 3) {
      ans = ans + solve(pos + 1, cntupdt, tight & (i == end));
    }
  }
  return dp[pos][cnt][tight] = ans;
}
void Solution() {
  long long int t;
  cin >> t;
  while (t--) {
    long long int l;
    long long int r;
    cin >> l >> r;
    s = to_string(r);
    memset(dp, -1, sizeof dp);
    long long int ans1 = solve(0, 0, 1);
    s = to_string(l - 1);
    memset(dp, -1, sizeof dp);
    long long int ans2 = solve(0, 0, 1);
    cout << ans1 - ans2 << "\n";
  }
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  Solution();
}
