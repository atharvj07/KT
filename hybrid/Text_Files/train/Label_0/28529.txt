#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  if (n == 0) {
    cout << 1 << endl;
    return 0;
  }
  int cov = 0;
  long long ans = 0;
  for (int y = n - 1; y > 0; y--) {
    int r = sqrt(n * (long long)n - y * (long long)y);
    if (r > cov)
      ans += r - cov;
    else
      ans += 1;
    cov = r;
  }
  cout << ans * 4 + 4 << endl;
  return 0;
}
