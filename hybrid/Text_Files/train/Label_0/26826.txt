#include <bits/stdc++.h>
using namespace std;
int n, a, b, k;
int arr[200005];
bool check(int x) {
  if (x % (a + b) == 0) return false;
  if (x % (a + b) <= a) return true;
  return false;
}
int main() {
  int ans = 0;
  cin >> n >> a >> b >> k;
  for (int i = 0; i < n; i++) cin >> arr[i];
  std::vector<int> v;
  for (int i = 0; i < n; i++) {
    if (check(arr[i]))
      ans++;
    else {
      int tmp = arr[i] % (a + b);
      if (tmp == 0) tmp = a + b;
      tmp = tmp - a;
      int steps = (tmp) / a;
      if (tmp % a == 0)
        ;
      else
        steps++;
      v.push_back(steps);
    }
  }
  sort((v).begin(), (v).end());
  for (auto j : v) {
    if (j > k) break;
    k = k - j;
    if (k >= 0) ans++;
  }
  cout << ans << endl;
}
