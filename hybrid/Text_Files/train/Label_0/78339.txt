import math

def prob():
	n = int(input())
	n += 1
	l = [1]
	l += [int(x) for x in input().split()]
	l += [1]
	p = [True] * n
	s = 0
	q = [i for i in range(1, n)]
	for i in range(1, n):
		if p[i]:
			a = i
			b = i
			d = l[i]
			if d == 1:
				(s, q) = (n - 2, [1])
				break
			while l[a - 1] % d == 0:
				a -= 1
			while l[b + 1] % d == 0:
				b += 1
				p[b] = False
			d = b - a
			if d > s:
				(s, q) = (d, [a])
			elif d == s and s != 0:
				q.append(a)
	print(len(q), s)
	print(*q)
t = 1
for _ in range(0, t):
	prob()
