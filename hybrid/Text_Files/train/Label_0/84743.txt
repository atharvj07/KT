from sys import stdin
tt = int(stdin.readline())
for loop in range(tt):
	(n, k, m) = map(int, stdin.readline().split())
	b = list(map(int, stdin.readline().split()))
	if (n - m) % (k - 1) != 0:
		print('NO')
		continue
	lis = [1] * n
	for i in b:
		lis[i - 1] = 0
	ns = 0
	while ns != k // 2:
		ns += lis[-1]
		del lis[-1]
	ns = 0
	lis.reverse()
	while ns != k // 2:
		ns += lis[-1]
		del lis[-1]
	if len(lis) == sum(lis):
		print('NO')
	else:
		print('YES')
