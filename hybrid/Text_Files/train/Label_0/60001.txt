
import java.util.*;
import java.io.*;

public class Hamburgers {

	// https://codeforces.com/contest/371/problem/C
	
	public static void main(String[] args) throws IOException, FileNotFoundException {
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		//BufferedReader in = new BufferedReader(new FileReader("Hamburgers"));

		char[] seq = in.readLine().toCharArray();
		int b = 0;
		int s = 0;
		int c = 0;
		for (int i=0; i<seq.length; i++) {
			if (seq[i] == 'B') b++;
			if (seq[i] == 'S') s++;
			if (seq[i] == 'C') c++;
		}
		StringTokenizer st = new StringTokenizer(in.readLine());
		int nb = Integer.parseInt(st.nextToken());  
		int ns = Integer.parseInt(st.nextToken());  
		int nc = Integer.parseInt(st.nextToken()); 
		if (b == 0) nb=0;
		if (s==0) ns=0;
		if (c==0) nc=0;
		
		st = new StringTokenizer(in.readLine());
		int pb = Integer.parseInt(st.nextToken());  
		int ps = Integer.parseInt(st.nextToken());  
		int pc = Integer.parseInt(st.nextToken()); 

		long r = Long.parseLong(in.readLine());
		
		long ans =0;
		/*while (nb >=b && ns >=s && nc>=c) {
			ans++;
			nb-=b;
			ns-=s;
			nc-=c;
		}*/
		long firststep =0;
		if (b!=0) {
			if (s!= 0) {
				if (c!= 0) {
					firststep = Math.min(nb/b, Math.min(ns/s, nc/c));
				}
				else {
					firststep = Math.min(nb/b, ns/s);
				}
			}
			else {
				if (c!= 0) {
					firststep = Math.min(nb/b, nc/c);
				}
				else {
					firststep = nb/b;
				}
			}
		}
		else {
			if (s!= 0) {
				if (c!= 0) {
					firststep = Math.min(ns/s, nc/c);
				}
				else {
					firststep = ns/s;
				}
			}
			else {
				if (c!= 0) {
					firststep = nc/c;
				}
				else {
					firststep = 0;
				}
			}
		}
		ans += firststep;
		nb -= b*firststep;
		ns -= s*firststep;
		nc -= c*firststep;
		
		while (nb > 0 || ns > 0 || nc>0) {
			long need = 0;
			//(b-nb)*pb + (s-ns)*ps + (c-nc)*pc;
			if (nb < b) {
				need += (b-nb)*pb;
			}
			if (ns < s) need += (s-ns)*ps;
			if (nc < c) need += (c-nc)*pc;
			nb-=b;
			if (nb < 0) nb=0;
			ns-=s;
			if (ns < 0) ns=0;
			nc-=c;
			if (nc < 0) nc=0;
			r -= need;
			if (r >= 0) ans++;
			else break;
		}
		
		
		long need = (b)*pb + (s)*ps + (c)*pc;
		ans += (r/need);
		
		System.out.println(ans);
	}
}
