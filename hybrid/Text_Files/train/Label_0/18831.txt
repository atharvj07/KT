#include <bits/stdc++.h>
using namespace std;
int n;
int mid(int a, int b) {
  if (abs(a - b) > n / 2) {
    return ((a + b + n) / 2 - 1) % n + 1;
  } else {
    return (a + b) / 2;
  }
}
int op(int a) { return a > n / 2 ? a - n / 2 : a + n / 2; }
int main() {
  scanf("%d", &n);
  int a, b, x1, x2, x3, x4;
  printf("? %d\n", n);
  fflush(stdout);
  scanf("%d", &x1);
  printf("? %d\n", n / 4);
  fflush(stdout);
  scanf("%d", &x2);
  printf("? %d\n", n / 2);
  fflush(stdout);
  scanf("%d", &x3);
  printf("? %d\n", n / 4 + n / 2);
  fflush(stdout);
  scanf("%d", &x4);
  if (abs(x1 - x3) % 2 == 1) {
    printf("! %d\n", -1);
    return 0;
  }
  if (x1 == x3) {
    printf("! %d\n", n);
    return 0;
  }
  if (x2 == x4) {
    printf("! %d\n", n / 4);
    return 0;
  }
  if (x1 > x3) {
    a = n;
  } else {
    a = n / 2;
  }
  if (x2 < x4) {
    b = n / 4;
  } else {
    b = n / 4 + n / 2;
  }
  while (abs(b - a) > 1) {
    int v1, v2;
    int midd = mid(a, b);
    printf("? %d\n", midd);
    fflush(stdout);
    scanf("%d", &v1);
    printf("? %d\n", op(midd));
    fflush(stdout);
    scanf("%d", &v2);
    if (v1 == v2) {
      printf("! %d\n", midd);
      return 0;
    } else if (v1 < v2) {
      b = midd;
    } else {
      a = midd;
    }
  }
  int v1, v2;
  printf("? %d\n", a);
  fflush(stdout);
  scanf("%d", &v1);
  printf("? %d\n", op(a));
  fflush(stdout);
  scanf("%d", &v2);
  if (v1 == v2) {
    printf("! %d\n", a);
    return 0;
  }
  printf("? %d\n", b);
  fflush(stdout);
  scanf("%d", &v1);
  printf("? %d\n", op(b));
  fflush(stdout);
  scanf("%d", &v2);
  if (v1 == v2) {
    printf("! %d\n", b);
    return 0;
  }
  return 0;
}
