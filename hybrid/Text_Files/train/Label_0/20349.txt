#include <bits/stdc++.h>
using namespace std;
#define FOR(i, a, b) for (ll i = (a); i < (b); i++)
typedef long long int ll;



int main()
{
	 ll t;
	cin>>t;
	while(t>0)
	 {
		string s;
		cin>>s;

		map<char,ll>m;
		ll ans=0;
		FOR(i,0,s.length())
		{
			if(s[i]=='['||s[i]=='(')
			{
				m[s[i]]++;
				continue;
			}

		
			if(s[i]==']'&&m['[']>0)
			{

					m['[']--;
					ans++;
			}
			else if(s[i]==')'&&m['(']>0)
			{
				m['(']--;
				ans++;
			}
			

			
		}
		cout<<ans<<endl;
		
			t--;
	 }
	

}
