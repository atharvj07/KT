a = int(input())
b = int(input())
c = int(input())
d = int(input())
e = int(input())
f = int(input())
pro = 0
if e >= f:
	temp = min(a, d)
	a -= temp
	d -= temp
	pro += temp * e
	if d > 0:
		temp = min(b, c, d)
		pro += temp * f
elif e < f:
	temp = min(b, c, d)
	b -= temp
	c -= temp
	d -= temp
	pro += temp * f
	if d > 0:
		temp = min(a, d)
		pro += temp * e
print(pro)
