#include <bits/stdc++.h>
using namespace std;
long long cota(long long x, long long y, long long p, long long q) {
  return (long long)max(max(ceil((double)y / q) + 1, ceil((double)x / p)) + 1,
                        ceil((double)(y - x) / (p - q)) + 1);
}
bool cumple(long long m, long long x, long long y, long long p, long long q) {
  if (m * q < y or m * p < x or m * q - y < m * p - x) {
    return false;
  }
  return true;
}
int main() {
  int n;
  cin >> n;
  long long x, y, p, q, mid;
  vector<long long> rta;
  while (n--) {
    cin >> x >> y >> p >> q;
    if (p * y == q * x) {
      rta.push_back(0);
    } else {
      if (p - q == 0) {
        rta.push_back(-1);
      } else {
        long long least = 0, last = 10000000000;
        while (last - least > 1) {
          mid = (last + least) / 2;
          if (cumple(mid, x, y, p, q)) {
            last = mid;
          } else {
            least = mid;
          }
        }
        if (cumple(least, x, y, p, q))
          rta.push_back(least * q - y);
        else {
          if (cumple(last, x, y, p, q))
            rta.push_back(last * q - y);
          else
            rta.push_back(-1);
        }
      }
    }
  }
  for (int i = 0; i < rta.size(); i++) {
    cout << rta[i] << endl;
  }
  return 0;
}
