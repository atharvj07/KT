s = input()
t = input()

def id(c):
	if 'a' <= c <= 'z':
		return ord(c) - ord('a')
	return ord(c) - ord('A') + 26
cnt = [0] * 52
Free = [0] * len(s)
(yay, whoops) = (0, 0)
for i in t:
	cnt[id(i)] += 1
for i in range(len(s)):
	if cnt[id(s[i])] > 0:
		yay += 1
		cnt[id(s[i])] -= 1
		Free[i] = 1
for i in range(len(s)):
	if Free[i] == 0:
		if cnt[id(s[i].upper())] > 0:
			whoops += 1
			cnt[id(s[i].upper())] -= 1
		if cnt[id(s[i].lower())] > 0:
			whoops += 1
			cnt[id(s[i].lower())] -= 1
print(yay, whoops)
