#include <bits/stdc++.h>
using namespace std;
int main(void) {
  double s;
  double a, b, c;
  scanf("%lf", &s);
  scanf("%lf%lf%lf", &a, &b, &c);
  if (a + b + c <= 1e-9) {
    printf("%.12lf %.12lf %.12lf\n", s / 3, s / 3, s / 3);
    return 0;
  }
  double x = a * s / (a + b + c);
  double y = b * s / (a + b + c);
  double z = c * s / (a + b + c);
  printf("%.12lf %.12lf %.12lf\n", x, y, z);
  return 0;
}
