#include <bits/stdc++.h>
using namespace std;
long long n, t, a, b, c;
long long Abs(long long a) { return a < 0 ? -a : a; }
struct Node {
  long long val;
  long long x, y;
} p[4], tmp[4];
void Cal(Node &A) {
  long long len = A.x + A.y - 2;
  A.val = len * (len - 1) / 2;
  len = 2 * n - A.x - A.y;
  A.val += len * (len + 1) / 2;
}
bool Dif(long long x, long long y) {
  if (x + 1 > n) return false;
  for (int i = 0; i < (c > 1 ? 4 : 1); i++) {
    tmp[i] = p[i];
    if (p[i].x < x && p[i].y < y) continue;
    if (p[i].x > x && p[i].y > y) continue;
    if (y < p[i].y) {
      tmp[i].val += Abs(p[i].y - y) * 2;
      if (tmp[i].val > t) return false;
    }
  }
  ++x;
  long long left = 0;
  if (c > 1 && x < p[3].x && y < p[3].y) {
    long long lx = p[3].x - x + 1, ly = p[3].y - y + 1;
    lx = min(lx, c);
    ly = min(ly, c);
    left = lx * ly - lx - ly + 1;
  }
  if (c > 1 && (t - tmp[1].val >> 1) + (t - tmp[2].val >> 1) < left)
    return false;
  for (int i = 0; i < (c > 1 ? 4 : 1); i++) p[i].val = tmp[i].val;
  return true;
}
bool Dif2(long long x, long long y) {
  if (y + 1 > n) return false;
  for (int i = 0; i < (c > 1 ? 4 : 1); i++) {
    tmp[i] = p[i];
    if (p[i].x < x && p[i].y < y) continue;
    if (p[i].x > x && p[i].y > y) continue;
    if (x < p[i].x) {
      tmp[i].val += Abs(p[i].x - x) * 2;
      if (tmp[i].val > t) return false;
    }
  }
  ++y;
  long long left = 0;
  if (c > 1 && x < p[3].x && y < p[3].y) {
    long long lx = p[3].x - x + 1, ly = p[3].y - y + 1;
    lx = min(lx, c);
    ly = min(ly, c);
    left = lx * ly - lx - ly + 1;
  }
  if (c > 1 && (t - tmp[1].val >> 1) + (t - tmp[2].val >> 1) < left)
    return false;
  for (int i = 0; i < (c > 1 ? 4 : 1); i++) p[i].val = tmp[i].val;
  return true;
}
int main() {
  cin >> n >> t >> a >> b >> c;
  for (int i = 0; i < 4; i++) {
    p[i].x = a + (i & 1 ? c - 1 : 0);
    p[i].y = b + (i & 2 ? c - 1 : 0);
    Cal(p[i]);
    if (p[i].val > t) {
      puts("Impossible");
      return 0;
    }
  }
  if (c > 1 && (t - p[1].val >> 1) + (t - p[2].val >> 1) < c * c - c * 2 + 1) {
    puts("Impossible");
    return 0;
  }
  long long nowx = 1, nowy = 1;
  while (nowx != n || nowy != n) {
    if (Dif(nowx, nowy)) {
      ++nowx;
      putchar('R');
    } else if (Dif2(nowx, nowy)) {
      ++nowy;
      putchar('U');
    } else {
      while (1)
        ;
    }
  }
  puts("");
  return 0;
}
