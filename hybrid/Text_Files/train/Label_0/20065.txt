#include <bits/stdc++.h>
using namespace std;
const int N = 2000005;
int n, l, t;
long long a[N];
double ans;
int check(long long dx, long long dy) {
  long long dis = (dy - dx);
  return t / l + (dy - dx <= t % l);
}
int main() {
  scanf("%d%d%d", &n, &l, &t);
  t *= 2;
  for (int i = (int)(1); i <= (int)(n); i++)
    scanf("%lld", &a[i]), a[i + n] = a[i] + l;
  int T1 = t / l + 1, T2 = t / l, p = 1;
  for (int i = (int)(1); i <= (int)(n); i++) {
    for (; check(a[i], a[p]) == T1; ++p)
      ;
    ans += 1.0 * (n - 1) * T2 + (p - i - 1);
  }
  printf("%.10lf\n", 0.25 * ans);
}
