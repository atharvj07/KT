from sys import stdin, stdout
t = int(stdin.readline())
for _ in range(t):
	(n, k) = map(int, stdin.readline().split())
	l = list(map(int, stdin.readline().strip().split()))
	count = 0
	(i, m) = (0, 0)
	ans = []
	while i < n:
		if l[i] % 2 == 1:
			ans.append(i + 1)
		i += 1
	if len(ans) >= k and (len(ans) - k) % 2 == 0:
		ans = ans[:k]
		ans[len(ans) - 1] = n
		print('YES')
		print(*ans)
	else:
		print('NO')
