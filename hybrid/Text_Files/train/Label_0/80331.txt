from collections import Counter

def main():
	try:
		while True:
			(n, m) = map(int, input().split())
			a = list(map(int, input().split()))
			trg = n // m
			result = 0
			c = Counter(a)
			req = {k: trg - v for (k, v) in c.items()}
			for i in range(1, m + 1):
				if i not in req:
					req[i] = trg
			for (i, x) in enumerate(a):
				if x > m:
					for (y, value) in req.items():
						if y <= m and value > 0:
							a[i] = y
							req[y] = value - 1
							result += 1
							break
				elif req[x] < 0:
					for (y, value) in req.items():
						if y <= m and value > 0:
							a[i] = y
							req[x] += 1
							req[y] = value - 1
							result += 1
							break
			print(trg, result)
			print(*a)
	except EOFError:
		pass
main()
