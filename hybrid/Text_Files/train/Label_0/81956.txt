n = int(input())
A = [0] + list(map(int, input().split()))
B = [0] + list(map(int, input().split()))
dp = [0] * (n + 1)
val = [0] * (n + 1)
pre = [0] * (n + 1)
pre[0] = float('-inf')
stack = []
for i in range(1, n + 1):
	tmp = dp[i - 1]
	while stack and A[stack[-1]] > A[i]:
		tmp = max(tmp, val[len(stack)])
		stack.pop()
	stack.append(i)
	val[len(stack)] = tmp
	dp[i] = pre[len(stack)] = max(pre[len(stack) - 1], tmp + B[i])
print(dp[-1])
