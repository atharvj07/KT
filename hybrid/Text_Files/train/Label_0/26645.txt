#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  long long int n, d, m, r, l;
  cin >> n >> d >> m >> l;
  long long int k = 0, c = 1;
  r = d;
  while (1 && k < n) {
    if (k * m + m - (k * m + l) > d) {
      c = 0;
      cout << d * ((k * m + l) / d + 1);
      break;
    } else if (k * m + m - (k * m + l) <= d) {
      if (d * ((k * m + l) / d + 1) < k * m + m) {
        c = 0;
        cout << d * ((k * m + l) / d + 1);
        break;
      }
    }
    k++;
  }
  if (c) {
    cout << d * (((k - 1) * m + l) / d + 1);
  }
}
