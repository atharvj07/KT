from sys import stdin
import math
inp = lambda : stdin.readline().strip()
t = int(inp())
for _ in range(t):
	n = int(inp())
	a = [int(x) for x in inp().split()]
	mini = min(a)
	sA = sorted(a)
	for i in range(n):
		if a[i] != sA[i]:
			if a[i] % mini != 0:
				print('NO')
				break
	else:
		print('YES')
