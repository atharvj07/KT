import java.util.*;

import static java.lang.Math.*;

import java.awt.Point;
import java.io.*;

public class B {
	static Scanner sc = new Scanner(new BufferedInputStream(System.in));

	public static void main(String[] args) {
		int n = ni();

		int m = ni();
		int x = m;
		int[] a = new int[m];
		while (m-- > 0) {
			a[m] = ni();
		}
		Arrays.sort(a);
		if (x > 0) {
			if (a[0] == 1 || a[a.length - 1] == n) {
				System.out.println("NO");
			} else {
				boolean lol = true;
				for (int i = 0; i < a.length - 2; i++) {
					if (a[i + 2] == a[i] + 2 && a[i + 1] == a[i] + 1) {
						System.out.println("NO");
						lol = false;
						break;
					}
				}
				if (lol)
					System.out.println("YES");
			}
		} else {
			System.out.println("YES");
		}

		sc.close();
	}

	static int ni() {
		return sc.nextInt();
	}

	static long nl() {
		return sc.nextLong();
	}

	static String n() {
		return sc.next();
	}
}
