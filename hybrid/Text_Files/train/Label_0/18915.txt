#include <bits/stdc++.h>
using namespace std;
long long n, A[100005], B[100005], X[100005], Y[100005];
string a, b;
int ok = 1, writ = 0;
void decrease(long long x);
void increase(long long x);
void increase(long long x) {
  if (!ok) return;
  if (X[x + 1] == 9) decrease(x + 1);
  if (!ok) return;
  cout << x + 1 << " " << 1 << endl;
  writ++;
  if (writ == 100000) ok = 0;
  X[x]++, X[x + 1]++;
}
void decrease(long long x) {
  if (!ok) return;
  if (X[x + 1] == 0) increase(x + 1);
  if (!ok) return;
  cout << x + 1 << " " << -1 << endl;
  writ++;
  if (writ == 100000) ok = 0;
  X[x]--, X[x + 1]--;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n;
  cin >> a >> b;
  for (long long i = 0; i < n; i++) {
    A[i] = a[i] - '0';
    B[i] = b[i] - '0';
    X[i] = A[i];
    Y[i] = B[i];
  }
  long long nr = 0;
  for (long long i = 0; i < n - 1; i++) {
    if (A[i] == B[i]) continue;
    if (A[i] > B[i]) {
      nr += A[i] - B[i];
      A[i + 1] -= A[i] - B[i];
      A[i] -= A[i] - B[i];
    } else {
      nr += B[i] - A[i];
      A[i + 1] += B[i] - A[i];
      A[i] += B[i] - A[i];
    }
  }
  if (A[n - 1] != B[n - 1])
    cout << -1;
  else {
    cout << nr << endl;
    for (long long i = 0; i < n - 1 && ok; i++) {
      if (X[i] == Y[i]) continue;
      while (X[i] < Y[i] && ok) increase(i);
      while (X[i] > Y[i] && ok) decrease(i);
    }
  }
  return 0;
}
