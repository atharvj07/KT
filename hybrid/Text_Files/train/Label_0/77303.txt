n = int(input())
a = list(map(int, input().split()))
cnt = 1
a.sort()
b = [a[0]]
for i in range(1, n):
	if a[i - 1] != a[i]:
		cnt += 1
		b.append(a[i])
if cnt != 3 and cnt != 2 and (cnt != 1):
	print('NO')
elif cnt == 1 or cnt == 2:
	print('YES')
elif b[1] - b[0] == b[2] - b[1]:
	print('YES')
else:
	print('NO')
