#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100000;
const int MULT = 20;
int a[MAXN];
int d[MAXN * MULT];
int was[MAXN * MULT + 1];
void calc(int n) {
  for (int i = 2; i * i <= n; i++)
    if (d[i] == 0)
      for (int j = i * i; j <= n; j += i) d[j] = 1;
}
int main() {
  int n, x;
  cin >> n >> x;
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  if (x == 2) {
    cout << 0 << endl;
    return 0;
  }
  sort(a, a + n);
  if (a[0] == 1) {
    cout << 1 << endl;
    return 0;
  }
  x--;
  if (x > MULT * n) {
    cout << -1 << endl;
    return 0;
  }
  calc(x);
  for (int i = 0; i < n; i++)
    if (a[i] <= x) was[a[i]] = 1;
  int ans = 0;
  for (int i = 2; i <= x; i++)
    if (d[i] == 0) {
      if (!was[i]) {
        cout << -1 << endl;
        return 0;
      } else
        ans++;
    }
  cout << ans << endl;
  return 0;
}
