#include <bits/stdc++.h>
using namespace std;
const int inf = INT_MAX;
int t, res, x[5], y[5], a[5], b[5], id[5], ax[5], ay[5], tx[5], ty[5];
void com() {
  for (int i = 0; i < 4; i++) id[i] = i;
  do {
    int ans = 0;
    for (int i = 0; i < 4; i++) {
      if (tx[i] != x[id[i]] && ty[i] != y[id[i]]) ans = inf;
      ans = max(ans, abs(tx[i] - x[id[i]]) + abs(ty[i] - y[id[i]]));
    }
    if (ans < res) {
      res = ans;
      for (int i = 0; i < 4; i++) ax[id[i]] = tx[i], ay[id[i]] = ty[i];
    }
  } while (next_permutation(id, id + 4));
}
void work0() {
  int d = b[1] - b[0];
  int _max0 = -inf, _min0 = inf;
  int _max1 = -inf, _min1 = inf;
  for (int i = 0; i < 4; i++) {
    if (y[i] == b[0])
      _max0 = max(_max0, x[i]), _min0 = min(_min0, x[i]);
    else
      _max1 = max(_max1, x[i]), _min1 = min(_min1, x[i]);
  }
  _min0 += d, _min1 += d;
  int _max = max(max(_max0, _min0), max(_max1, _min1));
  int _min = min(min(_max0, _min0), min(_max1, _min1));
  int x = (_max + _min) / 2;
  ty[0] = ty[1] = b[0], tx[0] = x, tx[1] = x - d;
  ty[2] = ty[3] = b[1], tx[2] = x, tx[3] = x - d;
  com();
}
void work1() {
  int d = b[1] - b[0];
  tx[2] = tx[3] = a[0] + d, ty[2] = b[0], ty[3] = b[1];
  tx[0] = tx[1] = a[0], ty[0] = b[0], ty[1] = b[1], com();
  tx[2] = tx[3] = a[0] - d, com();
}
void work2() {
  if (a[1] - a[0] != b[1] - b[0]) return;
  tx[0] = tx[1] = a[0], ty[0] = b[0], ty[1] = b[1];
  tx[2] = tx[3] = a[1], ty[2] = b[0], ty[3] = b[1];
  com();
}
void work3() {
  int d = a[1] - a[0];
  ty[2] = ty[3] = b[0] + d, tx[2] = a[0], tx[3] = a[1];
  ty[0] = ty[1] = b[0], tx[0] = a[0], tx[1] = a[1], com();
  ty[2] = ty[3] = b[0] - d, com();
}
void work4() {
  int d = a[1] - a[0];
  int _max0 = -inf, _min0 = inf;
  int _max1 = -inf, _min1 = inf;
  for (int i = 0; i < 4; i++)
    if (x[i] == a[0])
      _max0 = max(_max0, y[i]), _min0 = min(_min0, y[i]);
    else
      _max1 = max(_max1, y[i]), _min1 = min(_min1, y[i]);
  _min0 += d, _min1 += d;
  int _max = max(max(_max0, _min0), max(_max1, _min1));
  int _min = min(min(_max0, _min0), min(_max1, _min1));
  int x = (_max + _min) / 2;
  tx[0] = tx[1] = a[0], ty[0] = x, ty[1] = x - d;
  tx[2] = tx[3] = a[1], ty[2] = x, ty[3] = x - d;
  com();
}
void work() {
  res = inf;
  for (int i = 0; i < 4; i++) scanf("%d%d", &x[i], &y[i]);
  for (int i = 0; i < 16; i++) {
    int ta = 0, tb = 0;
    for (int j = 0; j < 4; j++)
      if (i >> j & 1)
        a[ta++] = x[j];
      else
        b[tb++] = y[j];
    sort(a, a + ta);
    sort(b, b + tb);
    ta = unique(a, a + ta) - a;
    tb = unique(b, b + tb) - b;
    if (ta == 0 && tb == 2) work0();
    if (ta == 1 && tb == 2) work1();
    if (ta == 2 && tb == 2) work2();
    if (ta == 2 && tb == 1) work3();
    if (ta == 2 && tb == 0) work4();
  }
  if (res == INT_MAX)
    puts("-1");
  else {
    printf("%d\n", res);
    for (int i = 0; i < 4; i++) printf("%d %d\n", ax[i], ay[i]);
  }
}
int main() {
  scanf("%d", &t);
  while (t--) work();
  return 0;
}
