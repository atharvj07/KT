#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2010;
int n, k, L[MAXN], R[MAXN], Z[MAXN];
long long memo[MAXN];
const long long INF = LLONG_MAX / 2LL;
long long dp(int z) {
  if (z == n) return 0LL;
  long long& ans = memo[z];
  if (ans != -1) return ans;
  ans = INF;
  long long req = 0;
  long long zombies = Z[z];
  int t = L[z];
  int cart = k;
  int i;
  for (i = z; i < n; i++) {
    t = max(t, L[i]);
    if (t > R[i]) break;
    long long rel = 0, rem = 0;
    if (zombies <= cart) {
      req += zombies;
      cart -= zombies;
    } else {
      zombies -= cart;
      req += cart;
      t++;
      if (t > R[i]) break;
      rel = zombies / k;
      rem = zombies % k;
      t += rel - !rem;
      cart = k;
    }
    req += rel * k;
    if (t > R[i]) break;
    if (t < R[i] || (i != n - 1 && t == R[i] && R[i] < L[i + 1])) {
      if (rem)
        ans = min(ans, req + k + dp(i + 1));
      else if (cart < k)
        ans = min(ans, req + cart + dp(i + 1));
      else
        ans = min(ans, req + dp(i + 1));
    }
    if (!rem) {
      zombies = Z[i + 1];
    } else {
      req += rem;
      cart -= rem;
      if (i != n - 1 && t == R[i] && R[i] == L[i + 1]) {
        zombies = max(0, Z[i + 1] - cart);
        req += min(Z[i + 1], cart);
        cart = max(0, cart - Z[i + 1]);
      } else
        zombies = Z[i + 1];
    }
    if (!cart && i != n - 1) {
      t++;
      cart = k;
    }
  }
  if (i == n) ans = min(ans, req);
  return ans;
}
int main() {
  scanf("%d %d", &n, &k);
  for (int i = 0; i < n; i++) scanf("%d %d %d", &L[i], &R[i], &Z[i]);
  for (int i = 0; i < n; i++) memo[i] = -1;
  long long ans = dp(0);
  if (ans == INF)
    printf("-1\n");
  else
    printf("%lld\n", ans);
  return 0;
}
