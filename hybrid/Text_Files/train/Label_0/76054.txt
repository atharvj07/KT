from collections import defaultdict
t = int(input())
for i in range(t):
	(n, k) = map(int, input().split(' '))
	arr = list(map(int, input().split(' ')))
	lst = defaultdict(lambda : -1)
	cnt = 0
	for i in range(n):
		lst[arr[i]] = i
	arr.sort()
	x = lst[arr[0]]
	cnt = 1
	for i in range(1, n):
		y = lst[arr[i]]
		if y == x + 1:
			x = y
			continue
		cnt += 1
		x = y
	if cnt <= k:
		print('Yes')
	else:
		print('No')
