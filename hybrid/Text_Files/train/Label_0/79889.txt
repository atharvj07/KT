import sys
n = int(input())
visited = [False] * (n + 1)
dp = [0] * (n + 1)
l = [[] for i in range(n + 1)]
for i in range(n - 1):
	aux1 = list(map(int, input().split()))
	l[aux1[1]].append(aux1[0])
	l[aux1[0]].append(aux1[1])
b = list(map(int, input().split()))
visited[1] = True
c = 1
t = True
for j in range(n):
	aux = 0
	for i in l[b[j]]:
		if not visited[i]:
			visited[i] = True
			dp[i] = 1
			aux += 1
	for i in range(c, c + aux):
		if dp[b[i]] == 1:
			dp[b[i]] = 0
		else:
			print('No')
			sys.exit()
	c += aux
print('Yes')
