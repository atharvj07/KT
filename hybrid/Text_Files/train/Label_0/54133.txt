import java.io.PrintWriter;
import java.util.Scanner;

/**
 * Created by Timur on 15.12.2016.
 */
public class _630K {
    public static void main(String args[]) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        long n = in.nextLong();
        long first = n / 2 + n / 3 + n / 5 + n / 7;
        long second = n / (2 * 3) + n / (2 * 5) + n / (2 * 7) + n / (3 * 5) +n / (3 * 7) +n / (5 * 7);
        long third = n / (2 * 3 * 5) + n / (2 * 3 * 7) + n / (2 * 5 * 7) + n / (3 * 5 * 7);
        long forth = n / (2 * 3 * 5 * 7);

        out.print(n - first + second - third + forth);
        out.close();
    }
}
