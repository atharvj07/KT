#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct point{double x,y;};
double X[110],Y[110];
double eps = 1e-9;
int N;
int main(){
    while(true){
        cin >> N;
        if(N==0) return 0;
        vector<point> window,curtain;
        double x,y;
        window.clear();
        curtain.clear();
        for(int i=1;i<=N;i++){
            cin >> x >> y;
            window.push_back({x,y});
            X[i] = x; Y[i] = y; 
        }
        for(int i=1;i<=4;i++){
            cin >> x >> y;
            X[N+i] = x;
            Y[N+i] = y;
            curtain.push_back({x,y});
        }
        sort(X+1,X+N+5);
        sort(Y+1,Y+N+5);
        double ans = 0;
        for(int i=1;i<=N+3;i++){
            for(int j=1;j<=N+3;j++){
                double cx = (X[i]+X[i+1])/2,cy = (Y[j]+Y[j+1])/2;
                int cnt = 0;
                for(int k=0;k<N;k++){
                    if(cx>window[k].x) continue;
                    double uy = max(window[k].y,window[(k+1)%N].y);
                    double dy = min(window[k].y,window[(k+1)%N].y);
                    if(dy<cy && cy<uy) cnt++;
                }
                if(cnt%2==0) continue;
                cnt = 0;
                for(int k=0;k<4;k++){
                    if(cx>curtain[k].x) continue;
                    double uy = max(curtain[k].y,curtain[(k+1)%4].y);
                    double dy = min(curtain[k].y,curtain[(k+1)%4].y);
                    if(dy<cy && cy<uy) cnt++;
                }
                if(cnt%2==1) continue;
                ans += (X[i+1]-X[i])*(Y[j+1]-Y[j]);
            }
        }
        cout << (int) ans << endl;
    }
}
