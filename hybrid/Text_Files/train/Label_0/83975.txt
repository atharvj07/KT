for _ in range(int(input())):
	n = int(input())
	ans = n
	t = []
	while n:
		d = n % 3
		n //= 3
		t = [d] + t
	t = [0] + t
	if max(t) == 1:
		print(ans)
	else:
		i = 0
		while t[i] < 2:
			i += 1
		j = i - 1
		while t[j] == 1:
			j -= 1
		t[j] = 1
		k = 0
		bs = 1
		for c in range(len(t) - 1, -1, -1):
			if c <= j:
				k += bs * t[c]
			bs *= 3
		print(k)
