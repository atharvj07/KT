import sys
sys.setrecursionlimit(10 ** 9)
input = sys.stdin.readline
S = input()[:-1] + 'l'
if 'w' in S or 'm' in S:
	print(0)
	exit()
MOD = 10 ** 9 + 7
prev = ''
dp = [0] * (len(S) + 1)
dp[0] = 1
dp[1] = 1
for (i, cur, prev) in zip(range(2, len(S) + 2), S, S[1:]):
	dp[i] = dp[i - 1]
	if cur == prev and (cur == 'u' or cur == 'n'):
		dp[i] = (dp[i - 1] + dp[i - 2]) % MOD
print(dp[-1])
