
import java.util.*;

public class c {
    public int time;
    public int compl;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        while (m-- != 0) {
            HashMap<Integer, Integer> easy = new HashMap<>();
            HashMap<Integer, Integer> hard = new HashMap<>();
            HashSet<Integer> hs = new HashSet<>();
            PriorityQueue<Integer> pri = new PriorityQueue<>(new Comparator<Integer>() {
                @Override
                public int compare(Integer integer, Integer t1) {
                    return t1 - integer;
                }
            });
            int n = sc.nextInt();
            long t = (long) sc.nextInt();
            long a = (long) sc.nextInt();
            long b = (long) sc.nextInt();
            c[] arr = new c[n + 1];
            int numeasy, numhard;
            int nume = 0;
            int numh = 0;
            for (int i = 0; i < n; i++) {
                arr[i]=new c();
                arr[i].compl = sc.nextInt();
                if (arr[i].compl == 0) {
                    nume++;
                } else {
                    numh++;
                }
            }
            arr[n] = new c();
            arr[n].time = (int) (t + 1);
            arr[n].compl = 0;
            numeasy = nume;
            numhard = numh;
            for (int i = 0; i < n; i++) {
                arr[i].time = sc.nextInt();
            }
            Arrays.sort(arr, new Comparator<c>() {
                @Override
                public int compare(c o1, c o2) {
                    return o1.time - o2.time;
                }
            });
            nume = 0;
            numh = 0;
            int re = 0;
            for (int i = 0; i < n + 1; i++) {
                int curt = arr[i].time - 1;
                long yu = curt - nume * a - numh * b;
                if (yu < 0) {
                    if(arr[i].compl==0){
                        nume++;
                    }else{
                        numh++;
                    }
                    continue;
                } else {
                    int jian = (int) Math.min(yu / a, numeasy - nume);
                    yu -= jian * a;
                    int nan=(int) Math.min(yu/b,numhard-numh);
                    re=Math.max(re,nume+numh+jian+nan);
                }
                if(arr[i].compl==0){
                    nume++;
                }else{
                    numh++;
                }
            }
            System.out.println(re);

        }
    }
}
