#include <bits/stdc++.h>
using ll = long long;
using std::cin;
using std::cout;
using std::endl;
template <class T>
inline bool chmax(T& a, T b) {
  if (a < b) {
    a = b;
    return 1;
  }
  return 0;
}
template <class T>
inline bool chmin(T& a, T b) {
  if (a > b) {
    a = b;
    return 1;
  }
  return 0;
}
const int inf = (int)1e9 + 7;
const long long INF = 1LL << 60;
void solve() {
  int n, m, q;
  cin >> n >> m >> q;
  std::vector<int> p(n);
  std::vector<int> ppos(n);
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
    p[i] -= 1;
    ppos[p[i]] = i;
  }
  std::vector<int> a(m);
  std::vector<std::vector<int>> apos(n);
  for (int i = 0; i < m; ++i) {
    cin >> a[i];
    a[i] -= 1;
    apos[a[i]].emplace_back(i);
  }
  const int N = 30;
  std::vector<std::vector<int>> to(N + 1, std::vector<int>(m + 1, m));
  for (int i = m - 1; i >= 0; i--) {
    const int nxt = p[(ppos[a[i]] + 1) % n];
    auto itr = std::upper_bound(apos[nxt].begin(), apos[nxt].end(), i);
    if (itr == apos[nxt].end()) continue;
    to[0][i] = *itr;
  }
  for (int i = 0; i < N; ++i) {
    for (int j = 0; j <= m; ++j) {
      to[i + 1][j] = to[i][to[i][j]];
    }
  }
  std::vector<int> dp(m + 1, m);
  for (int i = m - 1; i >= 0; --i) {
    int cur = i;
    for (int i = 0; i < N; ++i) {
      if ((n - 1) >> i & 1) cur = to[i][cur];
    }
    dp[i] = std::min(dp[i + 1], cur);
  }
  std::string res;
  while (q--) {
    int l, r;
    cin >> l >> r;
    l--;
    r--;
    if (dp[l] <= r)
      res += '1';
    else
      res += '0';
  }
  cout << res << "\n";
}
int main() {
  std::cin.tie(nullptr);
  std::ios::sync_with_stdio(false);
  int kkt = 1;
  while (kkt--) solve();
  return 0;
}
