n = int(input())
a = []
b = []
for _ in range(n):
	(x, y) = map(int, input().split())
	a.append((x, y))
for _ in range(n):
	(x, y) = map(int, input().split())
	b.append((x, y))
S = set()
for i in range(n):
	S.add((a[0][0] + b[i][0], a[0][1] + b[i][1]))
for i in range(1, n):
	P = set()
	for j in range(n):
		P.add((a[i][0] + b[j][0], a[i][1] + b[j][1]))
	S &= P
for i in S:
	print(*i)
	break
