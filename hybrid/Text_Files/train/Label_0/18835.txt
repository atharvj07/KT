#include <bits/stdc++.h>
using namespace std;
int n;
map<int, int> mp;
int Get(int x) {
  if (mp.count(x)) return mp[x];
  printf("? %d\n", x);
  fflush(stdout);
  int y;
  scanf("%d", &y);
  return mp[x] = y;
}
void Print(int x) {
  printf("! %d\n", x);
  exit(0);
}
int main() {
  scanf("%d", &n);
  if (n % 4) Print(-1);
  int a = Get(n / 2), b = Get(n);
  int l1 = 1, r1 = n / 2 - 1;
  int l2 = n / 2 + 1, r2 = n - 1;
  if (a == b) Print(n);
  while (true) {
    int mid1 = (l1 + r1) >> 1;
    int mid2 = (l2 + r2) >> 1;
    int x = Get(mid1), y = Get(mid2);
    if (x == y)
      Print(mid1);
    else if ((a < b && x > y) || (a > b && x < y)) {
      l1 = mid1 + 1;
      l2 = mid2 + 1;
    } else {
      r1 = mid1 - 1;
      r2 = mid2 - 1;
    }
  }
  return 0;
}
