import java.io.File;
import java.io.FileNotFoundException;
import java.io.PrintWriter;
import java.util.Scanner;
public class C {
	public static void main(String[] args) throws FileNotFoundException {
		Scanner sc = new Scanner(new File("input.txt"));
		PrintWriter pw = new PrintWriter(new File("output.txt"));
		int n = sc.nextInt();
		int a[] = new int[n];
		int min = 100000;
		int m[] = new int[n];
		int p[] = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = sc.nextInt();
		}
		int cnt = 0;
		for (int i = 0; i < n; i++) {
			if (a[i] >= 0){
				cnt ++;
			} 
			m[i] = cnt;
		}
		cnt = 0;
		for (int i = n-1; i >= 0 ; i--) {
			if (a[i] <= 0){
				cnt++;
			}
			p[i] = cnt;
		}
		for (int i = 0; i < n-1; i++) {
			if (p[i+1]+m[i] < min){
				min = p[i+1]+m[i];
			}
		}

		pw.println(min);
		pw.close();
	}
}
