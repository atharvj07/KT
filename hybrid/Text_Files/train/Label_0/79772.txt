			if j - 1 < 0:
				en = 0

				en = ca[j - 1]
			tot = ca[i] + cb[j] - st + ca[n - 1] - en

def solution(s):
	n = len(s)
	ca = []
	cb = []
	cura = 0
	curb = 0
	for i in range(n):
		if s[i] == 'a':
			cura += 1
		else:
			curb += 1
		ca.append(cura)
		cb.append(curb)
	where_to_choose_a = []
	for i in range(n):
		if i != 0 and ca[i] > ca[i - 1] and (ca[i] > cb[i]) and (ca[i] - ca[where_to_choose_a[-1]] > cb[i] - cb[where_to_choose_a[-1]]):
			where_to_choose_a.append(i)
		elif i != 0 and ca[i] == ca[i - 1] and (ca[i] > cb[i]):
			where_to_choose_a.append(where_to_choose_a[-1])
		elif i != 0 and ca[i] > ca[i - 1] and (ca[i] > cb[i]):
			where_to_choose_a.append(where_to_choose_a[-1])
		elif ca[i] > cb[i]:
			where_to_choose_a.append(i)
		elif len(where_to_choose_a):
			where_to_choose_a.append(where_to_choose_a[-1])
		else:
			where_to_choose_a.append(0)
	maxi = float('-inf')
	for i in range(n):
		if where_to_choose_a[i] == i:
			tot = ca[-1]
		else:
			if where_to_choose_a[i] - 1 < 0:
				st = 0
			else:
				st = cb[where_to_choose_a[i] - 1]
			if i >= n - 1:
				en = 0
			else:
				en = ca[-1] - ca[i - 1]
			tot = cb[i] - st + en
			if where_to_choose_a[i] != -1:
				tot += ca[where_to_choose_a[i]]
			if tot > maxi:
				maxi = dc(tot)
	if maxi == float('-inf'):
		maxi = max(ca[-1], cb[-1])
	return maxi
