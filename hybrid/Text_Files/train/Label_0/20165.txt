using namespace std;
const long long mod=1000000007;
const long long inf=mod*mod;
const long long d2=500000004;
long long dp[52][52];
long long calc(int a,int b){
	if(~dp[a][b])return dp[a][b];
	if(a==0||b==0){
		return dp[a][b]=0;
	}
	if(a==1){
		return dp[a][b]=b;
	}
	return dp[a][b]=min(inf,calc(a-1,b-1)+calc(a,b-1)+1);
}
int main(){
	int T;scanf("%d",&T);
	for(int i=0;i<52;i++)for(int j=0;j<52;j++)dp[i][j]=-1;
	for(int i=0;i<52;i++){
		for(int j=0;j<52;j++){
			calc(i,j);
		}
	}
	while(T--){
		long long a;
		int b;
		scanf("%lld%d",&a,&b);
		for(int i=0;i<52;i++){
			if(dp[b][i]>=a){
				printf("%d\n",i);break;
			}
		}
	}
}
