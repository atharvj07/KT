#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  vector<int> primes;
  int fact = 2;
  while (n > 1) {
    while (n % fact == 0) {
      primes.push_back(fact);
      n /= fact;
    }
    fact++;
  }
  for (int p : primes) {
    cout << p;
  }
  cout << endl;
  return 0;
}
