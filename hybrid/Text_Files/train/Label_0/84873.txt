from collections import Counter
from math import ceil
t = int(input())
while t:
	(n, k) = [int(x) for x in input().split()]
	arr = [int(x) for x in input().split()]
	count = Counter(arr)
	pp = len(count)
	if k >= pp:
		print(1)
	elif k == 1:
		print(-1)
	else:
		print(1 + ceil((pp - k) / (k - 1)))
	t -= 1
