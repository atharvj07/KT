import java.util.Scanner;

/**
 * Created by yuu on 8/4/17.
 */
public class Problem369A {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int bowls = sc.nextInt();
        int plates = sc.nextInt();
        int first = 0;
        int second = 0;
        for (int i = 0; i < n; i++) {
            if (sc.nextInt() == 1) first++;
            else second++;
        }
        int wash = 0;
        if (bowls > first) {
            bowls -= first;
        }
        else {
            first -= bowls;
            wash += first;
            bowls = 0;
        }
        int left = bowls + plates;
        if (left < second) {
            wash += second - left;
        }
        System.out.println(wash);
    }
}
