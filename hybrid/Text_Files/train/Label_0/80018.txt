line1 = input().split(' ')
n = int(line1[0])
k = int(line1[1])
main = list(map(int, input().split(' ')))
reqs = [None] * (n + 1)
for i in range(n):
	line = input().split(' ')
	if int(line[0]) == 0:
		reqs[1 + i] = []
	else:
		curr_reqs = []
		for req in line[1:]:
			curr_reqs += [int(req)]
		reqs[1 + i] = curr_reqs
res = []
to_exit = False

def traverse(main_courses):
	global res, to_exit
	roots = [False] * (n + 1)
	while main_courses and (not to_exit):
		main_to_trav = main_courses.pop()
		if reqs[main_to_trav] == None:
			continue
		stack = [main_to_trav]
		while len(stack) > 0 and (not to_exit):
			to_traverse = stack.pop()
			if reqs[to_traverse] is not None:
				childs = reqs[to_traverse]
				if len(childs) == 0:
					roots[to_traverse] = False
					res.append(to_traverse)
					reqs[to_traverse] = None
				else:
					roots[to_traverse] = True
					stack.append(to_traverse)
					for child in childs:
						if roots[child] == True:
							print(-1)
							to_exit = True
							break
					stack += childs
					reqs[to_traverse] = []
traverse(main)
if not to_exit:
	print(len(res))
	print(' '.join(map(str, res)))
