import sys
input = sys.stdin.readline
t = int(input())
for tests in range(t):
	X = input().strip()
	n = len(X)
	R = [0]
	L = [0]
	for i in range(n):
		if i % 2 == 0:
			if X[i] == '[' or X[i] == ']':
				R.append(R[-1] + 1)
			else:
				R.append(R[-1])
			L.append(L[-1])
		else:
			if X[i] == '[' or X[i] == ']':
				L.append(L[-1] + 1)
			else:
				L.append(L[-1])
			R.append(R[-1])
	q = int(input())
	for queries in range(q):
		(l, r) = map(int, input().split())
		RS = R[r] - R[l - 1]
		LS = L[r] - L[l - 1]
		print(abs(RS - LS))
