def solve():
	n = int(input())
	a = list()
	b = list()
	a_append = a.append
	b_append = b.append
	split = str.split
	for i in range(n):
		(new_a, new_b) = map(int, split(input()))
		a_append(new_a)
		b_append(new_b)
	c = [(a[i] - b[i], i) for i in range(n)]
	c = sorted(c, key=lambda item: item[0], reverse=True)
	sum = 0
	for i in range(n):
		index = c[i][1]
		sum += a[index] * i + b[index] * (n - i - 1)
	print(sum)
solve()
