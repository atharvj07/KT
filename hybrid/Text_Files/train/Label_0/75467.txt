(n, x0, y0, x1, y1) = map(int, input().split())
if y0 > y1:
	a2 = 'S'
else:
	a2 = 'N'
if x0 > x1:
	a1 = 'W'
else:
	a1 = 'E'
(k1, k2) = (abs(x0 - x1), abs(y0 - y1))
s = input()
l = len(s)
i = 0
while i < l and (k2 > 0 or k1 > 0):
	if s[i] == a1:
		k1 -= 1
	elif s[i] == a2:
		k2 -= 1
	i += 1
if k1 > 0 or k2 > 0:
	print(-1)
else:
	print(i)
