#include <bits/stdc++.h>
using namespace std;
class FenwickTree {
 private:
  vector<int>* tree;
  int n;

 public:
  FenwickTree(int n) {
    this->n = n;
    this->tree = new vector<int>(n + 1, 0);
  }
  void update(int index) {
    while (index <= n) {
      (*tree)[index]++;
      index += index & -index;
    }
  }
  int range(int index) {
    int result = 0;
    while (index > 0) {
      result += (*tree)[index];
      index -= index & -index;
    }
    return result;
  }
};
class Solution {
 private:
  vector<list<pair<int, long long>>>* tree;
  vector<long long>* a;
  vector<long long>* path;
  FenwickTree* acc;

 public:
  vector<int>* controls;

 public:
  Solution(vector<list<pair<int, long long>>>& tree, vector<long long>& a) {
    this->tree = &tree;
    this->a = &a;
    acc = new FenwickTree(tree.size());
    path = new vector<long long>(tree.size());
    controls = new vector<int>(tree.size());
  }
  ~Solution() {
    delete acc;
    delete path;
    delete controls;
  }
  void dfs(int current, long long sum, int level) {
    (*path)[level - 1] = sum;
    int index = lower_bound(path->begin(), path->begin() + level - 1,
                            sum - (*a)[current]) -
                path->begin() + 1;
    acc->update(index);
    int cc = acc->range(level);
    for (pair<int, long> child : (*tree)[current])
      dfs(child.first, sum + child.second, level + 1);
    (*controls)[current] = acc->range(level) - cc;
  }
};
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int j = 0; j < n; j++) cin >> a[j];
  vector<list<pair<int, long long>>> tree(n);
  for (int j = 1; j < n; j++) {
    int node;
    long long value;
    cin >> node >> value;
    tree[node - 1].push_back(make_pair(j, value));
  }
  Solution solution(tree, a);
  solution.dfs(0, 0, 1);
  for (int j = 0; j < n - 1; j++) cout << (*(solution.controls))[j] << " ";
  cout << (*(solution.controls))[n - 1] << endl;
}
