#include <bits/stdc++.h>
using namespace std;
int n, k, q, t[150005];
set<pair<int, int> > s;
int main() {
  int tp, id;
  scanf("%d%d%d", &n, &k, &q);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &t[i]);
  }
  for (int i = 1; i <= q; i++) {
    scanf("%d%d", &tp, &id);
    if (tp == 1) {
      s.insert(make_pair(t[id], id));
      if (s.size() > k) {
        s.erase(s.begin());
      }
    } else {
      if (s.find(make_pair(t[id], id)) != s.end()) {
        printf("YES\n");
      } else {
        printf("NO\n");
      }
    }
  }
  return 0;
}
