#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, l, r, r1;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> l >> r;
    if (l % 2 == 0 && r % 2 == 1) {
      cout << 0 - (r - l + 1) / 2 << endl;
      continue;
    }
    if (l % 2 == 0 && r % 2 == 0) {
      r1 = r;
      r--;
      cout << 0 - (r - l + 1) / 2 + r1 << endl;
      continue;
    }
    if (l % 2 == 1 && r % 2 == 0) {
      cout << (r - l + 1) / 2 << endl;
      continue;
    }
    if (l % 2 == 1 && r % 2 == 1) {
      cout << (r - 1 - l + 1) / 2 - r << endl;
      continue;
    }
  }
  return 0;
}
