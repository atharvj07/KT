#include<iostream>

#include<string.h>

using namespace std;

#define INF 1<<21
#define DIV 1000000007

typedef pair<int, int>P;

int p, q, a;

int solve(int n, int next, int prod, int sum) {
  int cnt = 0;
  for (int i = next; i*prod <= a; i++) {
	int left = i*prod*p;
	int right = prod + i*sum;
	if (left < right*q)continue;
	if (left == right*q) { cnt++; continue; }
	if (n == 1)continue;
	cnt += solve(n - 1, i, i*prod, right);
  }
  return cnt;
}

int main(){
  int n;
  while (cin>>p>>q>>a>>n,(p||q||a||n)){
	cout << solve(n, 1, 1, 0) << endl;
  }
  return 0;
}
