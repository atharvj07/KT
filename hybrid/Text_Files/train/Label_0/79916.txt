for _ in range(int(input())):
	(n, m) = map(int, input().split())
	r = []
	l = []
	points = {i for i in range(1, 2 * n + 1)}
	for i in range(m):
		r.append(sorted(list(map(int, input().split()))))
		points.remove(r[-1][0])
		points.remove(r[-1][1])
	ans = 0

	def inter(i, j, k):
		return k[0] < i < k[1] and (not k[0] < j < k[1]) or (k[0] < j < k[1] and (not k[0] < i < k[1]))
	for i in range(m):
		for j in range(i + 1, m):
			if inter(r[i][0], r[i][1], r[j]):
				ans += 1
	for _ in range(n - m):
		maxi = 0
		p = []
		l = sorted(points)
		p = [l[0], l[len(l) // 2]]
		for i in r:
			if inter(l[0], l[len(l) // 2], i):
				maxi += 1
		r.append(sorted(p))
		points.remove(p[0])
		points.remove(p[1])
		ans += maxi
	print(ans)
