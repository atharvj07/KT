template<typename T1, typename T2>
ostream &operator<<(ostream &os, const map<T1, T2> &mp) {
    os << "{";
    int a = 0;
    for (auto &tp : mp) {
        if (a) os << ", "; a = 1;
        os << tp;
    }
    return os << "}";
}
template<typename T>
ostream &operator<<(ostream &os, const vector<T> &v) {
    os << "[";
    REP(i, SZ(v)) {
        if (i) os << ", ";
        os << v[i];
    }
    return os << "]";
}


int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);
    cout << fixed << setprecision(10);

    ll A, B, C, N; cin >> A >> B >> C >> N;
    ll area = 2 * (A * B + B * C + C * A);

    map<tuple<ll,ll,ll>,bool> mp;

    REP(i, N) {
        ll x, y, z; cin >> x >> y >> z;
        area += 6;
        if (x == 0) area -= 2;
        if (x == A-1) area -= 2;
        if (y == 0) area -= 2;
        if (y == B-1) area -= 2;
        if (z == 0) area -= 2;
        if (z == C-1) area -= 2;
        if (mp.count(make_tuple(x-1, y, z))) area -= 2;
        if (mp.count(make_tuple(x+1, y, z))) area -= 2;
        if (mp.count(make_tuple(x, y-1, z))) area -= 2;
        if (mp.count(make_tuple(x, y+1, z))) area -= 2;
        if (mp.count(make_tuple(x, y, z-1))) area -= 2;
        if (mp.count(make_tuple(x, y, z+1))) area -= 2;
        mp[make_tuple(x, y, z)] = 1;
    }

    cout << area << endl;

    return 0;
}
