def f1(a, x):
	if x == 0:
		return 1
	p = f1(a, x // 2)
	p *= p
	p %= 998244353
	if x % 2:
		p *= a
		p %= 998244353
	return p % 998244353

def f(a, b):
	if a > b:
		(a, b) = (b, a)
	ans = 0
	k = 1
	for i in range(a + 1):
		ans = (ans + k) % 998244353
		k = k * (a - i) * (b - i) * f1(i + 1, 998244353 - 2) % 998244353
	return ans
(a, b, c) = map(int, input().split())
ans = f(a, b) * f(b, c) * f(a, c) % 998244353
