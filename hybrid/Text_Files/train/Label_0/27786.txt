#include <bits/stdc++.h>
using namespace std;
char a[100][100];
int N;
void cross(int i, int j) {
  if (a[i][j] == '#' && a[i - 1][j] == '#' && a[i + 1][j] == '#' &&
      a[i][j - 1] == '#' && a[i][j + 1] == '#') {
    a[i][j] = '.';
    a[i - 1][j] = '.';
    a[i + 1][j] = '.';
    a[i][j - 1] = '.';
    a[i][j + 1] = '.';
  }
}
int main() {
  int i, j;
  cin >> N;
  for (i = 0; i < N; i++)
    for (j = 0; j < N; j++) cin >> a[i][j];
  for (i = 0; i < N; i++)
    for (j = 0; j < N; j++)
      if (a[i][j] == '#') cross(i, j);
  bool ok = true;
  for (i = 0; i < N; i++)
    for (j = 0; j < N; j++)
      if (a[i][j] == '#') {
        ok = false;
        break;
      }
  if (ok)
    cout << "YES";
  else
    cout << "NO";
}
