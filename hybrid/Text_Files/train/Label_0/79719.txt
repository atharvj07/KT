for _ in range(int(input())):
	(n, left, right) = tuple(map(int, input().split()))
	maximal = n * (n - 1) + 1
	if left == maximal:
		print('1')
		continue
	if right == maximal:
		end_1 = True
	else:
		end_1 = False
	a = []
	summa = 0
	i = (n - 1) * 2
	cnt = 0
	while summa + i < left:
		summa += i
		i -= 2
		cnt += 1
	left -= summa
	right -= summa
	summa = 0
	while left <= right:
		while left <= summa + i and left <= right:
			if left % 2 == 1:
				print(cnt + 1, end=' ')
			else:
				print(cnt + left // 2 + 1, end=' ')
			left += 1
		left -= i
		right -= i
		summa = 0
		i -= 2
		cnt += 1
		if i == 0 and end_1:
			print(1, end=' ')
			break
	print()
	if left > right:
		continue
