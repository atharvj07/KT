using namespace std;

int a[10005];

int main()
{
    int n;
    while(scanf("%d",&n)==1)
    {
        for(int i=0;i<n;i++)
        {
            scanf("%d",&a[i]);
        }
        int ans=-1;
        for(int i=0;i<n;i++)
            for(int j=i+1;j<n;j++)
        {
            int x=a[i]*a[j];
            int y=x;
            while(y&&y/10%10==y%10-1)
            {
                y/=10;
            }
            if(y<10)
            {
                ans=max(ans,x);
            }
        }
        printf("%d\n",ans);
    }

    return 0;
}
