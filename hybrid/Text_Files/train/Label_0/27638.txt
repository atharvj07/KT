int n,m,cnt,type,a[N<<1],b[N<<1]; bitset<N> C,c[N];
char ans[N<<1],ch[N<<2]; bool bo[N];
pair<pii,int> last[N<<2]; map<pair<pii,int>,int> mp;
bool check(int x,int y,int u,int v){
	return min(x,y)==min(u,v) && max(x,y)==max(u,v);
}
void opt(pair<pii,int> o){
	if (!o.se) return;
	opt(last[mp[o]]); putchar(ch[mp[o]]);
}
void dfs(int x,int y,int k,int u,int v,char c){
	if (x>y) swap(x,y); if (x==y) return;
	auto o=mpr(mpr(x,y),k);
	if (mp[o]) return; mp[o]=++cnt;
	last[cnt]=mpr(mpr(u,v),k-1); ch[cnt]=c;
	if (k==m){ opt(o); puts(""); exit(0); }
	k++;
	if (check(x,y,a[k],b[k])) return;
	if (x==a[k] || x==b[k]){
		dfs(x,y,k,x,y,(x==a[k]?'^':'v'));
		dfs(x^a[k]^b[k],y,k,x,y,(x==a[k]?'v':'^'));
		return;
	}
	if (y==a[k] || y==b[k]){
		dfs(x,y,k,x,y,(y==a[k]?'^':'v'));
		dfs(x,y^a[k]^b[k],k,x,y,(y==a[k]?'v':'^'));
	}
	dfs(x,y,k,x,y,'^');
}
int main(){
	scanf("%d%d%d",&n,&m,&type);
	int i,j;
	for (i=1; i<=m; i++) scanf("%d%d",&a[i],&b[i]);
	if (type==2){
		if (n==2){ puts("-1"); return 0; }
		int x=a[1],y=b[1],z=1;
		while (z==x || z==y) z++;
		dfs(x,z,1,0,0,'^'); dfs(y,z,1,0,0,'v');
		return 0;	
	}
	for (i=1; i<=n; i++) c[i].set(i);
	for (i=1; i<=m; i++){
		C=c[a[i]];
		c[a[i]]|=c[b[i]]; c[b[i]]|=C;
	}
	for (i=1; i<=n; i++) C.set(i);
	for (i=1; i<=n; i++) if ((c[i]^C).none()){
		bo[i]=1;
		for (j=m; j; j--) if (bo[a[j]]){
			ans[j]='^'; bo[b[j]]=1;
		} else{
			ans[j]='v'; bo[a[j]]=bo[b[j]];
		}
		puts(ans+1); break;
	}
	if (i>n) puts("-1");
	return 0;
}

int read(){
	int x=0; char ch=getchar(); bool flag=0;
	while (ch<'0' || ch>'9'){ if (ch=='-') flag=1; ch=getchar(); }
	while (ch>='0' && ch<='9'){ x=x*10+ch-'0'; ch=getchar(); }
	return flag?-x:x;
}

	while (ch>='0' && ch<='9'){ x=x*10+ch-'0'; ch=getchar(); }

}
