import java.io.*;
import java.util.*;
public class E {

	static long mod = 998244353;
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		long k = sc.nextLong();
		if(k == 0) {
			long res = 1;
			for(long i = 1; i <= n; i++) {
				res *= i; res %= mod;
			}
			System.out.println(res);
		}
		else { // multiply everything by 2
			if(k >= n) {
				System.out.println(0);
			}
			else { //use exactly n-k lines:
				int r = (int)(n-k);
				long res = 0;
				for(int i = 0; i < r; i++) {
					if(i % 2 == 0) {
//						System.out.println(r+" "+i+" "+choose(r, i) +" "+ pow(r-i, n));
						res += choose(r, i) * pow(r-i, n);
						res %= mod;
					}
					else {
						res -= choose(r, i) * pow(r-i, n);
						res %= mod; res += mod; res %= mod;
					}
				}
				res *= choose(n, r); res %= mod;
				System.out.println(res * 2 % mod);
			}
		}
	}
	
	static int N = 200000;
	static long[] fac = new long[N+1];
	static long[] invfac = new long[N+1];
	static long fac(int n){
		if(n == 0) return fac[0] = 1;
		else if(fac[n] > 0) return fac[n];
		else return fac[n] = n * fac(n-1) % mod;
	}
	static long invfac(int n){
		if(invfac[n] > 0) return invfac[n];
		else return invfac[n] = inv(fac[n]);
	}
	static long choose(int n, int k){
		return fac(n) * invfac(k) % mod * invfac(n-k) % mod;
	}
	static long inv(long a){
		return (gcdex(a, mod).r + mod) % mod;
	}
	static Pair gcdex(long a, long b){
		if(b > a) return gcdex(b, a).invert();
		else if(b == 0) return new Pair(1, 0);
		else{
			Pair p = gcdex(b, a % b);
			return new Pair(p.s, p.r - p.s*(a/b));
		}
	}
	static class Pair{
		long r, s;
		public Pair(long r, long s){
			this.r = r; this.s = s;
		}
		public Pair invert(){
			return new Pair(s, r);
		}
	}
	static long pow(long a, long b) {
		long res = 1;
		while(b > 0) {
			if(b % 2 == 1) {
				res = res * a % mod;
			}
			a = a * a % mod;
			b /= 2;
		}
		return res;
	}

}
