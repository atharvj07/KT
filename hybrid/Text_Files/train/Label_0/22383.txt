#include <bits/stdc++.h>
using namespace std;
struct Event {
  int tp;
  int x, y;
  Event() {}
  Event(int x, int y, int tp) : x(x), y(y), tp(tp) {}
  bool operator<(const Event &a) const {
    if (x != a.x)
      return x < a.x;
    else
      return tp < a.tp;
  }
};
int n, k;
long long ans[100010];
vector<Event> vec;
int cs[300010];
int la[300010];
vector<int> ys;
int main() {
  scanf("%d %d", &n, &k);
  for (int i = 0; i < (n); i++) {
    int x, y;
    scanf("%d %d", &x, &y);
    vec.emplace_back(x, y, 1);
    vec.emplace_back(x + k, y, -1);
    ys.push_back(y);
    ys.push_back(y + k);
  }
  sort((ys).begin(), (ys).end());
  ys.erase(unique((ys).begin(), (ys).end()), ys.end());
  sort((vec).begin(), (vec).end());
  for (int i = 0; i < vec.size();) {
    int j = i;
    while (j < vec.size() && vec[j].x == vec[i].x) {
      int l = lower_bound((ys).begin(), (ys).end(), vec[j].y) - ys.begin();
      int r = lower_bound((ys).begin(), (ys).end(), vec[j].y + k) - ys.begin();
      for (; l < r; ++l) {
        ans[cs[l]] += (vec[i].x - la[l]) * (ys[l + 1] - ys[l]);
        cs[l] += vec[j].tp;
        la[l] = vec[i].x;
      }
      ++j;
    }
    i = j;
  }
  for (int i = 0; i < (n); i++)
    printf("%I64d%c", ans[i + 1], i == n - 1 ? '\n' : ' ');
  return 0;
}
