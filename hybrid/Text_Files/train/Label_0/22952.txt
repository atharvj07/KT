#include <bits/stdc++.h>
using namespace std;
int n, g[5][3000005], s0, s1, s2, s3;
char c1[3000005], c2[3000005];
inline int read() {
  int w = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    w = w * 10 + ch - '0';
    ch = getchar();
  }
  return w * f;
}
int main() {
  n = read();
  scanf("%s%s", c1 + 1, c2 + 1);
  for (register int i = 1; i <= n; i++) {
    int x = c1[i] - '0', y = c2[i] - '0';
    if (!x && !y) g[0][++s0] = i;
    if (!x && y) g[1][++s1] = i;
    if (x && !y) g[2][++s2] = i;
    if (x && y) g[3][++s3] = i;
  }
  for (register int x3 = 0; x3 <= s3; x3++) {
    int x0 = (n >> 1) + x3 - s1 - s3;
    if (!(x0 >= 0 && x0 <= s0)) continue;
    int k = (n >> 1) - x0 - x3;
    if (k > s1 + s2 || k < 0) continue;
    int x1 = min(k, s1);
    int x2 = k - x1;
    for (register int i = 1; i <= x0; i++) printf("%d ", g[0][i]);
    for (register int i = 1; i <= x1; i++) printf("%d ", g[1][i]);
    for (register int i = 1; i <= x2; i++) printf("%d ", g[2][i]);
    for (register int i = 1; i <= x3; i++) printf("%d ", g[3][i]);
    return 0;
  }
  puts("-1");
  return 0;
}
