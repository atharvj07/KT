w = int(input())
l = []
sum = 0
power = 0
for i in range(w):
	k = int(input())
	t = list(map(int, input().split()))
	l.append(t)

def count2(n):
	maxn = 0
	power = 0
	sum = 0
	for i in n:
		while i % 2 == 0:
			power += 1
			i = i // 2
		sum += i
		if maxn < i:
			maxn = i
	return (sum, power, maxn)
for i in l:
	q = count2(i)
	print(q[0] - q[2] + q[2] * 2 ** q[1])
