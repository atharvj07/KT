import java.util.*;
import java.lang.*;
import java.io.*;
import java.math.BigInteger;

public class CF
{
	public static void main(String[] args) {
		new C().run();
	}
	
	static class C {
		Scanner in;
		
		public void run() {
			in = new Scanner(System.in);
			int t = in.nextInt();
			while(t-- > 0) {
				int n = in.nextInt(), m = in.nextInt();
				int[] a = new int[n];
				int[] b = new int[m];
				for(int i=0; i<n; i++) {
					a[i] = in.nextInt();
				}
				for(int i=0; i<m; i++) {
					b[i] = in.nextInt();
				}
				
				List<Integer> pos, neg;
				pos = new ArrayList<Integer>();
				neg = new ArrayList<Integer>();

				for(int i=0; i<n; i++) {
					if(a[i] < 0) neg.add(-a[i]);
					else pos.add(a[i]);
				}
				
				Collections.reverse(neg);
				int[] apos = toArray(pos);
				int[] aneg = toArray(neg);
				
				//System.out.println(Arrays.toString(apos));
				//System.out.println(Arrays.toString(aneg));
				pos.clear(); neg.clear();
				
				for(int i=0; i<m; i++) {
					if(b[i] < 0) neg.add(-b[i]);
					else pos.add(b[i]);
				}
				
				Collections.reverse(neg);
				int[] bpos = toArray(pos);
				int[] bneg = toArray(neg);
				
				int posans = calc(apos, bpos);
				int negans = calc(aneg, bneg);
				
				//System.out.println("pos " + posans + " " + negans);
				
				System.out.println((posans + negans));
			}
		}
		
		private int calc(int[] a, int[] b) {
			SortedSet<Integer> as = new TreeSet<Integer>(), bs = new TreeSet<Integer>();
			
			for(int i=0; i<a.length; i++) as.add(a[i]);
			for(int i=0; i<b.length; i++) bs.add(b[i]);
			
			int ans = 0;
			
			List<Integer> equal = new ArrayList<Integer>();
			for(Integer i: bs) {
				if(as.contains(i)) equal.add(i);
			}
			
			for(int i=0; i<b.length; i++) {
				int end = calcRangeEnd(b[i]-1, a);
				int num = cntSpecialPositions(b[i], end, b);
				
				int l=0, r=equal.size();
				while(l<r) {
					int m = (l+r)/2;
					if(equal.get(m) <= end) l = m+1;
					else r = m;
				}
				
				num += (equal.size() - l);
				ans = Math.max(ans, num);
			}
			
			return ans;
		}
		
		private int cntSpecialPositions(int l, int r, int[] b) {
			int start = cntBeforeOrTillNum(l-1, b);
			int end = cntBeforeOrTillNum(r, b);
			return end - start;
		}
		
		private int cntBeforeOrTillNum(int num, int[] a) {
			int l = -1, r = a.length-1;
			while(l<r) {
				int m = (l+r+1)/2;
				if(a[m] <= num) l = m;
				else r = m - 1;
			}
			
			return l + 1;
		}
		
		private int calcRangeEnd(int start, int[] a) {
			int i = cntBeforeOrTillNum(start, a);
			int l = start, r = (int)1e9;
			while(l<r) {
				int m = (l+r)/2;
				if(cntBeforeOrTillNum(m, a) <= (m-start)) r = m;
				else l = m+1;
			}
			
//			System.out.println(" ds " + i + " " + start + " " + l);
			return l;
		}
		
		private int[] toArray(List<Integer> l) {
			int[] a = new int[l.size()];
			int i=0;
			for(Integer e: l) {
				a[i++] = e;
			}
			return a;
		}
		
	}
}
