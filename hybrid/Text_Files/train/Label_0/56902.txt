import java.util.*;
public class qualifying implements Comparable<qualifying>{
	
	public String name;
	public int score;
	
	public qualifying(String n, int sc) {
		name=n; score=sc;
	}
	
	public int compareTo(qualifying q) {
		return this.score-q.score;
	}

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		int m=sc.nextInt();
		ArrayList<qualifying>[] list=(ArrayList<qualifying>[])new ArrayList[m];
		for (int i=0; i<m; i++) {
			list[i]=new ArrayList<qualifying>();
		}
		for (int i=0; i<n; i++) {
			sc.nextLine();
			String str=sc.next();
			int ind=sc.nextInt();
			int sco=sc.nextInt();
			list[ind-1].add(new qualifying(str,sco));
		}
		for (int i=0; i<m; i++) {
			int l=list[i].size();
			Collections.sort(list[i]);
			boolean good=true;
			if (list[i].size()>2) {
				if (list[i].get(l-3).score==list[i].get(l-2).score) good=false;
			}
			if (good) {
				System.out.println(list[i].get(l-1).name+" "+list[i].get(l-2).name);
			} else System.out.println("?");
		}

	}

}
