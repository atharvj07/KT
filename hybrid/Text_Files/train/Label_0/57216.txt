

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class C {
	static int[] openCost, closedCost;
	static int[][][] dp;
	static int inf = 10000000;

	public static int solve(int curCost, int endCost, int remMoves) {
		if (remMoves == openCost.length) {
			if (curCost == endCost) {
				return 0;
			}
			return inf;
		}
		if (dp[curCost][endCost][remMoves] != -1) {
			return dp[curCost][endCost][remMoves];
		}
		int min = Integer.MAX_VALUE;
		if (curCost > 0) {
			if (curCost < 2 * openCost.length) {
				min = Math.min(
						openCost[remMoves]
								+ solve(curCost + 1, endCost, remMoves + 1),
						closedCost[remMoves]
								+ solve(curCost - 1, endCost, remMoves + 1));
			} else {
				min = Math.min(
						min,
						closedCost[remMoves]
								+ solve(curCost - 1, endCost, remMoves + 1));
			}
		} else {
			min = Math.min(
					min,
					openCost[remMoves]
							+ solve(curCost + 1, endCost, remMoves + 1));
		}
		return dp[curCost][endCost][remMoves] = min;
	}

	public static int[][] getOpt(int[][] mat1, int[][] mat2) {
		int[][] res = new int[mat1.length][mat2.length];
		for (int i = 0; i < mat1.length; i++) {
			for (int j = 0; j < mat1.length; j++) {
				res[i][j] = Integer.MAX_VALUE;
				for (int k = 0; k < mat1.length; k++) {
					res[i][j] = Math.min(res[i][j], mat1[i][k] + mat2[k][j]);
				}
			}
		}
		return res;
	}

	public static void main(String[] args) throws Exception {
		int n = nextInt();
		int m = nextInt();
		openCost = new int[n];
		closedCost = new int[n];
		for (int i = 0; i < n; i++) {
			openCost[i] = nextInt();
		}
		for (int i = 0; i < n; i++) {
			closedCost[i] = nextInt();
		}
		dp = new int[2 * n + 1][2 * n + 1][n];
		for (int i = 0; i < dp.length; i++) {
			for (int j = 0; j < dp[0].length; j++) {
				Arrays.fill(dp[i][j], -1);
			}
		}
		int[][] costMatrix = new int[2 * n + 1][2 * n + 1];
		for (int i = 0; i <= 2 * n; i++) {
			for (int j = 0; j <= 2 * n; j++) {
				costMatrix[i][j] = solve(i, j, 0);
			}
		}
//		for(int i=0;i<costMatrix.length;i++){
//			System.out.println(Arrays.toString(costMatrix[i]));
//		}
		int[][] res = new int[2 * n + 1][2 * n + 1];
		for (int i = 0; i < 2 * n + 1; i++) {
			Arrays.fill(res[i], 1000000000);
		}
		for (int i = 0; i < 2 * n + 1; i++) {
			res[i][i] = 0;
		}
		while (m > 0) {
			if ((m & 1) == 1) {
				res = getOpt(res, costMatrix);
			}
			costMatrix = getOpt(costMatrix, costMatrix);
			m >>= 1;
		}
		System.out.println(res[0][0]);
	}

	static BufferedReader br = new BufferedReader(new InputStreamReader(
			System.in));
	static StringTokenizer tokenizer = new StringTokenizer("");

	static int nextInt() throws Exception {
		return Integer.parseInt(next());
	}

	static double nextDouble() throws Exception {
		return Double.parseDouble(next());
	}

	static String next() throws Exception {
		while (true) {
			if (tokenizer.hasMoreTokens()) {
				return tokenizer.nextToken();
			}
			String s = br.readLine();
			if (s == null) {
				return null;
			}
			tokenizer = new StringTokenizer(s);
		}
	}
}
