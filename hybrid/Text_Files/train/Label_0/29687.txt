#include <bits/stdc++.h>
using namespace std;
long long t[100005], ans[100005];
int main() {
  long long n, p;
  cin >> n >> p;
  for (int i = 0; i < n; i++) cin >> t[i];
  priority_queue<pair<long long, long long>, vector<pair<long long, long long>>,
                 greater<pair<long long, long long>>>
      pq;
  for (int i = 0; i < n; i++) pq.emplace(t[i], i);
  queue<long long> q;
  set<long long> s;
  long long t = 0;
  for (int i = 0; i < n; i++) {
    if (q.empty() && s.empty()) t = pq.top().first;
    t += p;
    while (!pq.empty() && pq.top().first <= t) {
      if (q.empty() || pq.top().second < q.back())
        q.push(pq.top().second);
      else
        s.insert(pq.top().second);
      pq.pop();
    }
    ans[q.front()] = t;
    q.pop();
    if (q.empty() && !s.empty()) {
      q.push(*s.begin());
      s.erase(s.begin());
    }
  }
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
}
