def func():
	n = int(input())
	path = input()
	if path.count('>') == 0:
		return n
	elif path.count('<') == 0:
		return n
	l = [0] * n
	for i in range(n):
		if i < n - 1:
			if path[i] == '-':
				l[i] = True
				l[i + 1] = True
			elif path[i] + path[i + 1] == '><':
				l[i + 1] = False
		elif path[i] == '-':
			l[i] = True
			l[0] = True
		elif path[i] + path[0] == '><':
			l[0] = False
	return sum(l)
for i in range(int(input())):
	print(func())
