
import java.util.*; 
import java.io.*;
public class redblue {
    static int max(int a, int b) {
        if (a > b) {
            return a;
        }
        return b;
    }
    public static void main(String[] args) throws IOException {
        Scanner s = new Scanner(System.in);
        for (int t = s.nextInt(); t>0; t--) {
            int maxA = 0;
            int maxB = 0;
            int n = s.nextInt();
            int[] pre1 = new int[n];
            pre1[0] = s.nextInt();
            maxA = max(maxA, pre1[0]);
            for (int i=1; i<n; i++) {
                pre1[i] = pre1[i-1] + s.nextInt();
                maxA = max(maxA, pre1[i]);
            }
            int m = s.nextInt();
            int[] pre2 = new int[m];
            pre2[0] = s.nextInt();
            maxB = max(maxB, pre2[0]);
            for (int i=1; i<m; i++) {
                pre2[i] = pre2[i-1] + s.nextInt();
                maxB = max(maxB, pre2[i]);
            }
            System.out.println((maxA + maxB));
        }
    }
}
