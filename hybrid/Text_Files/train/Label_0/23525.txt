#include <bits/stdc++.h>
using namespace std;
long long int arr1[100005];
long long int arr[100007][3];
int main() {
  long long int i, j, n, p, q, r;
  cin >> n >> p >> q >> r;
  for (i = 0; i < n; i++) {
    cin >> arr1[i];
  }
  for (i = 0; i < n; i++) {
    for (j = 0; j < 3; j++) {
      if (i == 0) {
        if (j == 0)
          arr[i][j] = arr1[i] * p;
        else {
          if (j == 1)
            arr[i][j] = (arr[i][j - 1] + arr1[i] * q);
          else
            arr[i][j] = ((arr[i][j - 1] + arr1[i] * r));
        }
      } else {
        if (j == 0) {
          arr[i][j] = max(arr[i - 1][j], (arr1[i] * p));
        }
        if (j == 1) {
          arr[i][j] = max(arr[i - 1][j], (arr[i][j - 1] + arr1[i] * q));
        }
        if (j == 2) {
          arr[i][j] = max(arr[i - 1][j], (arr[i][j - 1] + arr1[i] * r));
        }
      }
    }
  }
  cout << arr[n - 1][2] << "\n";
  return 0;
}
