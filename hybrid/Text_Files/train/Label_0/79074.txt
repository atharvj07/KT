n = int(input())
side = n // 2 + 1
arr = [[0 for i in range(side)] for j in range(side)]
num = 1
ans = []
for i in range(n):
	if num <= n:
		ans += [[i, i]]
		arr[i][i] = num
	num += 1
	if num <= n:
		ans += [[i, i + 1]]
		arr[i][i + 1] = num
	num += 1
print(side)
for x in ans:
	print(x[0] + 1, x[1] + 1)
