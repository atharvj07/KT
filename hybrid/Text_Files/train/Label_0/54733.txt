import java.util.Scanner;

public class Pixels
{
    public static void main(String[] args)
    {
        Scanner scanner = new Scanner(System.in);
        int row = scanner.nextInt();
        int column = scanner.nextInt();
        int orders = scanner.nextInt();
        boolean[][] board = new boolean[row][column];
        boolean b = true;
        int r , c;
        for (int i = 0; i < orders; i++)
        {
            r = scanner.nextInt() - 1;
            c = scanner.nextInt() - 1;
            board[r][c] = true;
            if (b)
            {
                if (board[r][c] && r != row - 1 && board[r + 1][c] && ((c != 0 && board[r][c - 1] && board[r + 1][c - 1]) || (c != column - 1 && board[r][c + 1] && board[r + 1][c + 1])))
                {
                    b = false;
                    System.out.println(i + 1);
                }
                else if (board[r][c] && r != 0 && board[r - 1][c] && ((c != 0 && board[r][c - 1] && board[r - 1][c - 1]) || (c != column - 1 && board[r][c + 1] && board[r - 1][c + 1])))
                {
                    b = false;
                    System.out.println(i + 1);
                }
            }
        }
        if (b) {
            System.out.println(0);
        }
    }
}
