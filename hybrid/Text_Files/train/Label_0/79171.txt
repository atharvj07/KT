import sys
MOD = 10 ** 9 + 7

def mult(A, B):
	n = len(A)
	C = []
	for x in range(n):
		c = [0] * n
		for z in range(n):
			b = B[x][z]
			a = A[z]
			for y in range(n):
				c[y] += a[y] * b
		C.append(c)
	for x in range(n):
		c = C[x]
		for y in range(n):
			c[y] %= MOD
	return C

def vecmult(A, B):
	n = len(A)
	C = [0] * n
	for z in range(n):
		b = B[z]
		a = A[z]
		for y in range(n):
			C[y] += a[y] * b
	return [c % MOD for c in C]
(n, m) = [int(x) for x in input().split()]
mapper = [[0] * m for _ in range(m)]
mapper[0][0] = 1
mapper[-1][0] = 1
for i in range(m - 1):
	mapper[i][i + 1] = 1

def power(A, B, m):
	n = len(A)
	if m == 0:
		return B
	while m > 1:
		if m % 2 == 1:
			B = vecmult(A, B)
		A = mult(A, A)
		m //= 2
	return vecmult(A, B)
vec = [1] * m
vec = power(mapper, vec, n)
print(vec[-1] % MOD)
