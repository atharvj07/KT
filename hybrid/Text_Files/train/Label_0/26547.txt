#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
long long n;
long long ans = 0;
long long first[100008], nxt[100008 * 2], last[100008], to[100008 * 2],
    cnt = 0, a[100008];
void addedge(long long a, long long b) {
  if (!first[a])
    first[a] = ++cnt;
  else
    nxt[last[a]] = ++cnt;
  last[a] = cnt;
  to[cnt] = b;
}
long long gcd(long long a, long long b) {
  if (a < b) swap(a, b);
  if (b == 0) return a;
  if (a % b == 0) return b;
  return gcd(b, a % b);
}
void dfs(long long u, long long fa, map<long long, long long> map1) {
  map<long long, long long>::iterator iter;
  vector<pair<long long, long long> > v1;
  for (iter = map1.begin(); iter != map1.end(); iter++) {
    long long tmp = gcd(iter->first, a[u]);
    (ans += tmp * (iter->second) % mod) %= mod;
    v1.push_back(make_pair(tmp, iter->second));
  }
  map1.clear();
  long long len1 = v1.size();
  for (long long i = 0; i < len1; i++) map1[v1[i].first] += v1[i].second;
  map1[a[u]]++;
  (ans += a[u]) %= mod;
  for (long long q = first[u]; q; q = nxt[q])
    if (to[q] != fa) dfs(to[q], u, map1);
}
signed main() {
  scanf("%lld", &n);
  for (long long i = 1; i <= n; i++) scanf("%lld", &a[i]);
  for (long long i = 1; i < n; i++) {
    long long a, b;
    scanf("%lld%lld", &a, &b);
    addedge(a, b);
    addedge(b, a);
  }
  map<long long, long long> map1;
  dfs(1, 0, map1);
  cout << ans << endl;
  return 0;
}
