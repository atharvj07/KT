import java.io.*;
import java.util.StringTokenizer;

/**
 * Created by jack008 on 17-9-4.
 */
public class CF {
    public static boolean[] tree = new boolean[1200010];

    public static void build(int l, int r, int root) {
        if (l == r) {
            return;
        }
        int m = (l + r) >> 1;
        build(l, m, root << 1);
        build(m + 1, r, root << 1 | 1);
        tree[root] = tree[root << 1] && tree[root << 1 | 1];
    }

    public static void init(int[] arr) {
        for (int a : arr) {
            tree[(1 << 19) + a] = true;
        }
        build(1, 1 << 19, 1);
    }

    public static int calc(int x) {
        int root = 1;
        int ans = 0;
        for (int i = 18; i >= 0; i--) {
            int i1 = 1 << i;
            if ((x & i1) == 0) {
                if (!tree[root << 1]) {
                    root <<= 1;
                } else {
                    root = root << 1 | 1;
                    ans |= i1;
                }
                continue;
            }
            x ^= i1;
            if (!tree[root << 1 | 1]) {
                root = root << 1 | 1;
            } else {
                root <<= 1;
                ans |= i1;
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        IO scanner = new IO(System.in);
        try {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = scanner.nextInt();
            }
            init(arr);
            int accX = 0;
            for (int i = 0; i < m; i++) {
                accX ^= scanner.nextInt();
                System.out.println(calc(accX));
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static class IO extends PrintWriter {
        static BufferedReader r;
        static StringTokenizer t;

        public IO(InputStream i) {
            super(new BufferedOutputStream(System.out));
            r = new BufferedReader(new InputStreamReader(i));
            t = new StringTokenizer("");
        }

        public String next() throws IOException {
            while (!t.hasMoreTokens()) {
                t = new StringTokenizer(r.readLine());
            }
            return t.nextToken();
        }

        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        public double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }
    }
}
