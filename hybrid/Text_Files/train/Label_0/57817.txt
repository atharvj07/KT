import java.io.PrintStream;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class D {
	
	private static final PrintStream out = System.out;
	private static final Scanner in = new Scanner(System.in);
	
	private static class Pair implements Comparable<Pair> {
		private int idx;
		private long a;
		private long b;
		public Pair(int idx, long a, long b) {
			this.idx = idx;
			this.a = a;
			this.b = b;
		}
		public int compareTo(Pair o) {
			if (a != o.a)
				return a > o.a ? -1 : 1;
			if (b != o.b)
				return b > o.b ? -1 : 1;
			if (idx != o.idx)
				return idx < o.idx ? -1 : 1;
			return 0;
		}
	}
	
	public static void main(String[] args) {
		int n = in.nextInt();
		Pair[] pairs = new Pair[n];
		for (int i = 0; i < n; ++i)
			pairs[i] = new Pair(i, in.nextLong(), -1);
		for (int i = 0; i < n; ++i)
			pairs[i].b = in.nextLong();
		Arrays.sort(pairs);
		ArrayList<Integer> ans = new ArrayList<Integer>();
		int idx;
		if (n%2 == 0) {
			ans.add(pairs[0].idx);
			ans.add(pairs[1].idx);
			idx = 2;
		} else {
			ans.add(pairs[0].idx);
			idx = 1;
		}
		while (idx < n) {
			Pair x = pairs[idx];
			Pair y = pairs[idx + 1];
			if (x.b > y.b)
				ans.add(x.idx);
			else
				ans.add(y.idx);
			idx += 2;
		}
		out.println(ans.size());
		for (int curr : ans)
			out.print(1 + curr + " ");
		out.println();
	}
	
}
