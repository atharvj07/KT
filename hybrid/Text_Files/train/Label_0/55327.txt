import java.io.*;
import java.util.*;

public class B {

	void solve() throws IOException {
		int n = nextInt();
		int m = nextInt();
		int q = nextInt();
		int[][] a = new int[n][m];
		for (int i = 0; i < n; i++) {
			String s = next();
			for (int j = 0; j < m; j++) {
				if (s.charAt(j) == '0') {
					if (i > 0) {
						a[i][j] = a[i - 1][j] + 1;
					} else {
						a[i][j] = 1;
					}
				}
			}
		}

		int[][][][] down = new int[n][n][m][m];
		for (int i = 0; i < n; i++) {
			for (int j = i; j < n; j++) {
				for (int len = 1; len <= m; len++) {
					for (int l = 0; l + len <= m; l++) {
						int r = l + len - 1;
						int curMin = j - i + 1;
						for (int t = r; t >= l; t--) {
							curMin = Math.min(a[j][t], curMin);
							down[i][j][l][r] += curMin;
						}
						if (len > 1) {
							down[i][j][l][r] += down[i][j][l][r - 1];
						}
					}
				}
			}
		}

		for (int i = 0; i < q; i++) {
			int x1 = nextInt()-1;
			int y1 = nextInt()-1;
			int x2 = nextInt()-1;
			int y2 = nextInt()-1;
			int res = 0;
			for (int j = x1; j <= x2; j++) {
				res += down[x1][j][y1][y2];
			}
			out.println(res);
		}

	}

	void run() throws IOException {
		in = new BufferedReader(new InputStreamReader(System.in));
		out = new PrintWriter(System.out);
		solve();
		out.close();
	}

	public static void main(String[] args) throws IOException {
		new B().run();
	}

	BufferedReader in;
	PrintWriter out;
	StringTokenizer st;

	String next() throws IOException {
		while (st == null || !st.hasMoreTokens()) {
			String temp = in.readLine();
			if (temp == null) {
				return null;
			}
			st = new StringTokenizer(temp);
		}
		return st.nextToken();
	}

	int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	long nextLong() throws IOException {
		return Long.parseLong(next());
	}

}
