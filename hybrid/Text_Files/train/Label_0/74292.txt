a = int(input())
if a in [3, 4]:
	b = []
	for i in range(a):
		b.append(input().split())
	for i in b:
		for j in range(2):
			i[j] = int(i[j])
	ys = []
	for i in b:
		ys.append(i[1])
	xs = []
	for i in b:
		xs.append(i[0])
	xs.sort()
	ys.sort()
	print((xs[-1] - xs[0]) * (ys[-1] - ys[0]))
elif a == 2:
	b = input().split()
	c = input().split()
	if b[0] != c[0] and b[1] != c[1]:
		print(abs(int(b[0]) - int(c[0])) * abs(int(b[1]) - int(c[1])))
	else:
		print('-1')
else:
	print('-1')
