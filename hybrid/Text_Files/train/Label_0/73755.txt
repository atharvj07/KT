n = int(input())
ts = list(map(int, input().split(' ')))
num = []
cnt = 1
now = ts[0]
for i in range(1, n):
	if ts[i] == now:
		cnt += 1
	elif ts[i] != now:
		num.append(cnt)
		now = ts[i]
		cnt = 1
	if i == n - 1:
		num.append(cnt)
mx = 0
for i in range(len(num) - 1):
	if min(num[i], num[i + 1]) * 2 >= mx:
		mx = min(num[i], num[i + 1]) * 2
print(mx)
