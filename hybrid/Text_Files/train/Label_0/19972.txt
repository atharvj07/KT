#include <bits/stdc++.h>
using namespace std;
int h1, a1, c1, h2, a2, turn;
vector<string> ans;
int main() {
  scanf("%d%d%d%d%d", &h1, &a1, &c1, &h2, &a2);
  while (1729) {
    if (h2 <= a1) {
      ans.push_back("STRIKE");
      break;
    } else if (h1 <= a2) {
      ans.push_back("HEAL");
      h1 += c1;
    } else {
      ans.push_back("STRIKE");
      h2 -= a1;
    }
    if (h2 <= 0) break;
    h1 -= a2;
  }
  printf("%d\n", (int)ans.size());
  for (auto i : ans) cout << i << "\n";
}
