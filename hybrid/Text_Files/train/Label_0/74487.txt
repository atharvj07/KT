q = int(input())
for _ in range(q):
	(a, b, m) = [int(x) for x in input().split()]
	if a == b:
		print('1 ', a)
		continue
	found = 0
	for n in range(2, 51):
		lo = 2 ** (n - 2) * (a + 1)
		hi = 2 ** (n - 2) * (a + m)
		if lo <= b and b <= hi:
			b -= lo
			print(n, a, '', end='')
			acc = a
			for i in range(2, n + 1):
				p = 1 if i + 1 >= n else 2 ** (n - i - 1)
				left = 1
				right = m
				res = 1
				while left <= right:
					mid = (left + right) // 2
					if p * mid - p <= b:
						res = max(res, mid)
						left = mid + 1
					else:
						right = mid - 1
				b += p
				b -= p * res
				print(acc + res, '', end='')
				acc += acc + res
			print('')
			found = 1
			break
	if not found:
		print(-1)
