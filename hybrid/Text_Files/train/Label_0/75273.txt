(n, k) = map(int, input().split())
p = list(map(int, input().split()))
ls = []
for i in range(n):
	ls.append([p[i], i])
ls = sorted(ls)[::-1]
res = []
sm = 0
for i in range(k):
	sm += ls[i][0]
	res.append(ls[i][1])
pro = 1
res.sort()
for i in range(1, k):
	pro *= res[i] - res[i - 1]
print(sm, pro % 998244353)
