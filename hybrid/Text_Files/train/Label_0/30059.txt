#include <bits/stdc++.h>
using namespace std;
int main() {
  int dia = 4;
  int dias[12] = {31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
  int num;
  cin >> num;
  string criterio;
  cin >> criterio;
  cin >> criterio;
  int ans = 0;
  if (criterio == "week") {
    num--;
    for (int i = 0; i < 366; i++) {
      if (dia == num) ans++;
      dia++;
      dia %= 7;
    }
  } else {
    for (int i = 0; i < 12; i++) {
      if (dias[i] >= num) {
        ans++;
      }
    }
  }
  cout << ans << endl;
}
