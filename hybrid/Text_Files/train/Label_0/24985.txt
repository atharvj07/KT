#include <bits/stdc++.h>

using namespace std;

// mt19937 g(chrono::high_resolution_clock::now().time_since_epoch().count());

typedef long long ll;

const int NMAX = 1e6 + 5;
int who[NMAX], fq[NMAX], done[NMAX];

void pre() {
    int N = 1e6;
    for(int i = 1; i <= N; i++) who[i] = i;
    for(int i = 2; i <= N; i++)
        if(who[i] == i && N / i >= i) {
            int val = i * i;
            for(int j = val; j <= N; j += val)
                while(who[j] % val == 0)
                    who[j] /= val;
        }
}

void solve_test() {
    int N;
    scanf("%d", &N);
    vector<int> v(N);
    for(int i = 1; i <= N; i++) {
        int x;
        scanf("%d", &x);
        v[i - 1] = x;
        fq[ who[x] ]++;
    }

    int ans0 = 0;
    int ans1 = fq[1]; done[1] = 1;
    for(int i = 0; i < N; i++) {
        int x = who[ v[i] ];
        ans0 = max(ans0, fq[x]);
        if(!done[x]) {
            done[x] = 1;
            if(fq[x] % 2 == 0)  ans1 += fq[x];
        }
    }

    ans1 = max(ans1, ans0);
    int Q;
    scanf("%d", &Q);
    for(int i = 0; i < Q; i++) {

        ll x;
        scanf("%lld", &x);
        if(x == 0)
            printf("%d\n", ans0);
        else
            printf("%d\n", ans1);
    }

    for(int i = 0; i < N; i++) {
        int x = who[ v[i] ];
        fq[x] = done[x] = 0;
    }
}

int main() {
    //freopen("1.in", "r", stdin);
    //freopen("1.out", "w", stdout);

    pre();

    int T = 1;
    scanf("%d", &T);
    for(int t = 1; t <= T; t++) {
        //printf("Case #%d: ", t);
        solve_test();
    }

    return 0;
}
