t = int(input())
for _ in range(t):
	n = int(input())
	l = []
	c = 0
	r = 0
	for i in range(n):
		s = input()
		o = 0
		z = 0
		if len(s) % 2 == 1:
			c += 1
		else:
			for j in range(len(s)):
				if s[j] == '0':
					z += 1
				else:
					o += 1
			if z % 2 != 0 and o % 2 != 0:
				r += 1
	if r % 2 == 0 or c > 0:
		print(n)
	else:
		print(n - 1)
