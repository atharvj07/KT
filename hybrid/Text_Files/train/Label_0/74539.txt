from sys import stdin, stdout
for _ in range(int(input())):
	(n, q) = map(int, input().split())
	arr = list(map(int, stdin.readline().strip().split()))
	if n == 1:
		print(*arr)
	elif n == 2:
		print(max(arr))
	else:
		ans = 0
		add = True
		for i in range(1, n - 1):
			if arr[i - 1] < arr[i] > arr[i + 1] and add == True:
				ans += arr[i]
				add = False
			elif arr[i - 1] > arr[i] < arr[i + 1] and add == False:
				ans -= arr[i]
				add = True
		if add == True:
			ans += arr[-1]
		ans1 = arr[0]
		add1 = False
		for i in range(1, n - 1):
			if arr[i - 1] < arr[i] > arr[i + 1] and add1 == True:
				ans1 += arr[i]
				add1 = False
			elif arr[i - 1] > arr[i] < arr[i + 1] and add1 == False:
				ans1 -= arr[i]
				add1 = True
		if add1 == True:
			ans1 += arr[-1]
		print(max(ans, ans1))
