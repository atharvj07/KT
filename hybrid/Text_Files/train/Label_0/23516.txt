#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long n;
  cin >> n;
  vector<double> a(n);
  for (auto i = 0; i < n; i++) {
    cin >> a[i];
  }
  long long neg = 0, pos = 0, zer = 0;
  for (auto i = 0; i < n; i++) {
    if (a[i] > 0)
      pos++;
    else if (a[i] < 0)
      neg++;
    else
      zer++;
  }
  if (zer > ceil((float)n / 2)) {
    cout << 0;
  } else if (zer == ceil((float)n / 2)) {
    if (n % 2 == 1) {
      cout << 0;
    } else {
      if (pos == ceil((float)n / 2))
        cout << 2;
      else if (neg == ceil((float)n / 2))
        cout << -2;
      else
        cout << 0;
    }
  } else {
    if (pos >= ceil((float)n / 2))
      cout << 2;
    else if (neg >= ceil((float)n / 2))
      cout << -2;
    else
      cout << 0;
  }
  return 0;
}
