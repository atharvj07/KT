#include <bits/stdc++.h>
using namespace std;
const int N = 2505;
int n, cur, cnt;
long long tmp;
struct Vec {
  int x, y;
} p[N], a[N << 1], O;
Vec operator-(Vec u, Vec v) { return (Vec){u.x - v.x, u.y - v.y}; }
int Vin(Vec u) { return u.y < 0 | (u.x > 0 && u.y == 0); }
long long Cross(Vec u, Vec v) { return 1ll * u.x * v.y - 1ll * u.y * v.x; }
long long C(int x, int y) {
  long long ret = 1;
  if (x < y) return 0;
  for (int i = 0; i < y; ++i) ret *= (x - i), ret /= (i + 1);
  return ret;
}
bool cmp(Vec u, Vec v) {
  u = u - p[cur], v = v - p[cur];
  int uin = Vin(u), vin = Vin(v);
  if (uin != vin)
    return uin < vin;
  else
    return Cross(u, v) > 0;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) scanf("%d%d", &p[i].x, &p[i].y);
  O.x = O.y = 0;
  for (int i = 1; i <= n; ++i) {
    cur = i, cnt = 0;
    for (int j = 1; j <= n; ++j)
      if (i != j) a[++cnt] = p[j];
    sort(a + 1, a + cnt + 1, cmp);
    for (int j = 1; j <= cnt; ++j) a[j + cnt] = a[j];
    for (int j = cnt + 1, pt = 1; j <= 2 * cnt; ++j) {
      while (pt + cnt <= j || Cross(a[j] - p[i], a[pt] - p[i]) > 0) ++pt;
      tmp += C(j - pt, 3);
    }
  }
  printf("%lld", 5 * C(n, 5) - tmp);
  return 0;
}
