import java.util.*;
 
public class A
{	
    public static void main(String[] args)
    {
    	Scanner sc = new Scanner(System.in);
    	int t = sc.nextInt();
    	while(t-->0)
    	{
    		int k,n=0,min1,min2;
    		long s = 0;
    		min1 = Integer.MAX_VALUE;
    		min2 = min1;
    		
    		n=sc.nextInt();
    		int a[] = new int[n];
    		int b[] = new int[n];
    		
    		for(k=0;k<n;k++)
    		{
    			a[k]=sc.nextInt();
    			min1 = Math.min(min1, a[k]); 
    		}
    			
    		for(k=0;k<n;k++)
    		{
    			b[k]=sc.nextInt();
    			min2 = Math.min(min2, b[k]); 
    		}
    		   		
    		for(k=0;k<n;k++)
    		{
    			s = s + Math.max(a[k]-min1, b[k]-min2);
    		}
    		System.out.println(s);
    	
    	}
    }
}
