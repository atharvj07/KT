import java.io.BufferedReader;
import java.io.File;
import java.io.FileReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class DynamicProgramming
{
	public static void main(String[] args) throws IOException {
		File file = new File("in");
		BufferedReader in = new BufferedReader(file.exists() ? new FileReader(file) : new InputStreamReader(System.in));
		StringBuilder out = new StringBuilder("");
		String  lines[];
		int n, k;
		lines = in.readLine().split("\\s+");
		n = Integer.parseInt(lines[0]);
		k = Integer.parseInt(lines[1]);
		String s= in.readLine();
		int[] arr=new int[n];
		int p=0;
		for(int i=0;i<n;i++)
		{
			if(s.charAt(i)=='0')
				{
				arr[p++]=i;
				
				}
			
		}
		
		p--;
		int min=Integer.MAX_VALUE;
		int farmerroom=0;
		for(int i=0;i<=p-k;i++)
		{	
		
			//farmerroom can range from (a,b) including a and b
			int a=i;
			int b=i+k;
			while(farmerroom<b&&max(arr[farmerroom]-arr[a],arr[b]-arr[farmerroom])>max(arr[farmerroom+1]-arr[a],arr[b]-arr[farmerroom+1]))
			{
				farmerroom++;
			}
			min=min(min,max(arr[farmerroom]-arr[a], arr[b]-arr[farmerroom]));
		}
		//min++;
		System.out.println(min);
		
	}
	public static int min( int a, int b)
	{
		if ( a < b ) return a;
		return b;
	}
	public static int max( int a, int b)
	{
		if ( a > b ) return a;
		return b;
	}
}
