#include <bits/stdc++.h>
#pragma GCC optimize(3)
using namespace std;
const long long N = 2e5 + 10;
long long a[N], b[N], aa[N], bb[N], c[N], ans[N];
long long n, m;
pair<long long, long long> t1[N], t2[N];
inline long long read() {
  long long X = 0, w = 0;
  char c = 0;
  while (c < '0' || c > '9') {
    w |= c == '-';
    c = getchar();
  }
  while (c >= '0' && c <= '9')
    X = (X << 3) + (X << 1) + (c ^ 48), c = getchar();
  return w ? -X : X;
}
void update(long long l, long long r, long long d) {
  c[l] += d;
  c[r + 1] -= d;
}
void modify(long long l, long long r, long long d) {
  if (l > r) return;
  l = ((l + n) % n + n) % n;
  r = ((r + n) % n + n) % n;
  if (l > r)
    update(l, n - 1, d), update(0, r, d);
  else
    update(l, r, d);
}
signed main() {
  m = read();
  n = read();
  for (long long i = 1; i <= n; i++) t1[i] = make_pair(read(), i);
  for (long long i = 1; i <= n; i++) t2[i] = make_pair(read(), i);
  sort(t1 + 1, t1 + 1 + n);
  sort(t2 + 1, t2 + 1 + n);
  for (long long i = 1; i <= n; i++) {
    a[i] = t1[i].first;
    aa[i] = a[i] << 1;
    b[i] = t2[i].first;
    bb[i] = b[i] << 1;
  }
  for (long long i = 1; i <= n; i++) {
    long long l = 1,
              r = upper_bound(bb + 1, bb + 1 + n, aa[i] - m - 1) - bb - 1;
    modify(l - i, r - i, -a[i]);
    l = r + 1;
    r = upper_bound(bb + 1, bb + 1 + n, aa[i] - 1) - bb - 1;
    modify(l - i, r - i, a[i]);
    l = r + 1;
    r = upper_bound(bb + 1, bb + 1 + n, aa[i] + m) - bb - 1;
    modify(l - i, r - i, -a[i]);
    l = r + 1;
    r = n;
    modify(l - i, r - i, a[i] + m);
  }
  for (long long i = 1; i <= n; i++) {
    long long l = 1,
              r = upper_bound(aa + 1, aa + 1 + n, bb[i] - m - 1) - aa - 1;
    modify(i - r, i - l, -b[i]);
    l = r + 1;
    r = upper_bound(aa + 1, aa + 1 + n, bb[i]) - aa - 1;
    modify(i - r, i - l, b[i]);
    l = r + 1;
    r = upper_bound(aa + 1, aa + 1 + n, bb[i] + m) - aa - 1;
    modify(i - r, i - l, -b[i]);
    l = r + 1;
    r = n;
    modify(i - r, i - l, b[i] + m);
  }
  long long pos = 0;
  for (long long i = 1; i < n; i++)
    c[i] += c[i - 1], pos = (c[i] < c[pos]) ? i : pos;
  printf("%lld\n", c[pos]);
  for (long long i = 1, j = pos + 1; i <= n; i++, j = (j == n) ? 1 : j + 1)
    ans[t1[i].second] = t2[j].second;
  for (long long i = 1; i <= n; i++) printf("%lld ", ans[i]);
  puts("");
  return 0;
}
