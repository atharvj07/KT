
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;



// author : Shudi


public class Pashmak {
	
	static int[] rel;
	
	static int[] low = new int[300005 * 4];
	static int[] high = new int[300005 * 4];
	
	static int[] max = new int[300005 * 4];



	public static void main(String[] args) throws IOException {
		
		//Scanner in = new Scanner(System.in);
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		
		
		
		int v = Integer.parseInt(in.readLine());
		
		init(1, 0, v-1);
		
		List<List<Integer>> adj = new ArrayList<>();
		
		for(int i=0; i<v ;i++) {
			adj.add(new ArrayList<>());
		}
		
		
		int[] stren = new int[v];
		
		String[] line = in.readLine().split("\\s+");

		
		for(int i=0; i<v; i++) {
			stren[i] = Integer.parseInt(line[i])+ 2;
			
			upd(1, i, stren[i]);
		}
		
		

		for(int i=0; i<v-1; i++) {
			line = in.readLine().split("\\s+");

			int x = Integer.parseInt(line[0]) -1;
			int y =Integer.parseInt(line[1]) - 1;
			adj.get(x).add(y);
			adj.get(y).add(x);
		}
		
		long a = 99999999999l;
		
		for(int i=0; i<v; i++) {
			
			stren[i] -= 2;
			upd(1, i, stren[i]);
			
			for(int j=0; j<adj.get(i).size(); j++) {
				int k = adj.get(i).get(j);
				
				stren[k]--;
				upd(1, k, stren[k]);
			}
			
			a = Math.min(a, max[1]);
			
			for(int j=0; j<adj.get(i).size(); j++) {
				int k = adj.get(i).get(j);
				
				stren[k]++;
				upd(1, k, stren[k]);
			}
			
			stren[i] += 2;
			upd(1, i, stren[i]);

		}
		
		System.out.println(a);
		

		
	}



	private static void upd(int i, int x, int v) {
		
		if(low[i] > x || high[i] < x) {
			return;
		}
		else if(low[i] == x && high[i] == x) {
			max[i] = v;
			return;
		}
		
		upd(2*i,x,v);
		upd(2*i+1,x,v);

		max[i] = Math.max(max[2*i], max[2*i+1]);
		
	}



	private static void init(int i, int j, int k) {
		low[i] = j;
		high[i] = k;
		
		if(j == k) {
			return;
		}
		
		int mid = (j+k)/2;
		
		init(2*i, j, mid);
		init(2*i+1, mid+1, k);

	}
}
