#include<bits/stdc++.h>
using namespace std;
typedef long long ll;



int main(){
    ios::sync_with_stdio(false);
	cin.tie(0);

    ll n;
    cin>>n;
    pair<ll,ll> p[2*n];
    for(int i=0;i<n;i++){
        int a;
        cin>>a;
        p[i]=make_pair(a,0);
    }
    for(int i=0;i<n;i++){   
        int b;
        cin>>b;
        p[n+i]=make_pair(b,1);
    }

    sort(p,p+2*n);
    

    ll ans=1;
    ll mod=1e9+7;

    ll cnt1=0,cnt0=0;
    for(int i=0;i<2*n;i++){
        if(p[i].second==0){
            if(cnt1>0){
                ans*=cnt1;
                ans%=mod;
                cnt1--;
            }
            else cnt0++;
        }
        else{
            if(cnt0>0){
                ans*=cnt0;
                ans%=mod;
                cnt0--;
            }
            else cnt1++;
        }
    }
    cout<<ans<<endl;
}
