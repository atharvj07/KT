(x,) = gil()
left = 40
ans = []

def getMsb(x):
	return int(log2(x)) + 1

def end(x):
	m = getMsb(x)
	return x == 2 ** m - 1

def getMask(x):
	m = getMsb(x)
	return 2 ** m - 1
while not end(x) and left > 0:
	if getMask(x) - x + (x & 1) <= left:
		break
	left -= 1
	ans.append(getMsb(x))
	x ^= getMask(x)
	if end(x):
		break
	x += 1
	left -= 1
while not end(x) and left > 0:
	ans.append(~x & 1)
	left -= 1
	x ^= ~x & 1
	if end(x):
		break
	x += 1
	left -= 1
print(40 - left)
print(*ans)
