def main():
	(n, k) = map(int, input().split())
	power = list(map(int, input().split()))
	cnt = 0
	win = power[0]
	del power[0]
	while True:
		for i in range(0, len(power)):
			if win > power[i]:
				cnt = cnt + 1
				if cnt >= len(power):
					print(win)
					return
				if cnt >= k:
					print(win)
					return
			else:
				for j in range(0, i):
					power.append(power[0])
					del power[0]
				power.append(win)
				win = power[0]
				del power[0]
				cnt = 1
				break
main()
