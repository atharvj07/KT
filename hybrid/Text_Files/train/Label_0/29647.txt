#include<bits/stdc++.h>
#define ll long long
#define pll pair<ll,ll>
#define F first
#define S second
#define mp make_pair
using namespace std;
const int N=1e5+5;
stack<ll>st;
ll n,cnt,k,num;
pll p[N];
ll ans,sum;
bool use[N];
int main()
{
	cin>>n>>k;
	for(int i=0;i<n;i++)cin>>p[i].S>>p[i].F;
	sort(p,p+n);
	reverse(p,p+n);
	for(int i=0;i<n;i++)swap(p[i].F,p[i].S);
	for(int i=0;i<n;i++)
	{
		if(cnt<k)
		{
			if(use[p[i].F])
				st.push(p[i].S);
			else
				num++,use[p[i].F]=true;
			sum+=p[i].S;
			ans=max(ans,sum+num*num);
			cnt++;
		}
		else
		{
			if(st.empty())break;
			if(use[p[i].F])continue;
			use[p[i].F]=true;
			num++;
			sum-=st.top();
			sum+=p[i].S;
			st.pop();
			ans=max(ans,sum+num*num);
		}
	}
	cout<<ans<<endl;
	return 0;
}
