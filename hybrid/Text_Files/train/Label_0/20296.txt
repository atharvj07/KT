#include <bits/stdc++.h>
using namespace std;
const long long P = 1000000007;
int n;
long long x, A, B, vB, ret;
long long egcd(long long x, long long y, long long &a, long long &b) {
  if (!y) {
    a = 1;
    b = 0;
    return x;
  } else {
    long long t = egcd(y, x % y, b, a);
    b -= x / y * a;
    return t;
  }
}
long long inv(long long x, long long m) {
  long long a, b;
  egcd(x, m, a, b);
  a %= m;
  if (a < 0) a += m;
  return a;
}
int main() {
  scanf("%d", &n);
  x = 1;
  for (int i = 1; i <= 2 * n; i++) {
    x = (x * i) % P;
    if (i == 2 * n) A = x;
    if (i == n) B = x;
  }
  vB = inv(B, P);
  ret = (((A * vB) % P) * vB) % P;
  ret = ret + (P - n);
  printf("%I64d\n", ret % P);
  return 0;
}
