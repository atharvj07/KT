
import java.util.Scanner;

public class Main {
	
	public static int a, b;
	public static boolean[][] vis;
	public static boolean[][] mark;
	public static char[][] array;
	
	public static void main(String[] args) {
		
		Scanner in = new Scanner( System.in );
		a = in.nextInt();
		b = in.nextInt();
		
		vis = new boolean[a][b];
		mark = new boolean[a][b];
		array = new char[a][b];
		
		for( int sa = 0; sa < a; sa ++ ) {
			String str = in.next();
			for( int wq = 0; wq < b; wq ++ ) {
				array[sa][wq] = str.charAt( wq );
			}
		}
		
		boolean flag = dfs( 0, 0 );
		
		if( flag ) {
			flag = dfs( 0, 0 );
			if( flag ) {
				System.out.println( 2 );
			}
			else {
				System.out.println( 1 );
			}
		}
		else {
			System.out.println( 0 );
		}
		
//		for( int sa = 0; sa < a; sa ++ ) {
//			for( int wq = 0; wq < b; wq ++ ) {
//				System.out.println( mark[sa][wq] );
//			}
//		}
		
	}
	
	public static boolean dfs( int x, int y ) {
		
		if( x >= a || y >= b ) {
			return false;
		}
		if( x == a - 1 && y == b - 1 ) {
			return true;
		}
		if( array[x][y] == '#' || vis[x][y] || mark[x][y] ) {
			return false;
		}
		
		vis[x][y] = true;
		vis[0][0] = false;
		
		boolean flag = dfs( x, y + 1 );
		
		if( flag == false ) {
			flag = dfs( x + 1, y );
		}
		if( flag == true ) {
			mark[x][y] = true;
			mark[0][0] = false;
		}
		
		return flag;
		
	}

}
