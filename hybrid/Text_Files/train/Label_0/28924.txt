#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, j, n, m, s = 0, k, r, a[105], d[105] = {};
  cin >> n;
  for (k = 0; k < n; k++) {
    cin >> m;
    for (j = 0; j < m; j++) {
      cin >> r;
      d[r]++;
      if (d[r] == n) {
        a[s] = r;
        s++;
      }
    }
  }
  for (j = 0; j < s; j++) {
    cout << a[j] << " ";
  }
}
