from itertools import combinations as perm

def isprime(n):
	for i in range(2, int(n ** 0.5) + 1):
		if n % i == 0:
			return 0
	return 1
t = int(input())
for x in range(t):
	k = int(input())
	n = int(input())
	a = str(n)
	s = list(set(a))
	f = 0
	for i in s:
		if isprime(int(i)) == 0 or i == '1':
			f = 1
			print(1)
			print(i)
			break
	if f == 0 and k == 2:
		print(len(a))
		print(n)
	elif f == 0:
		while f == 0:
			i = 2
			for j in perm(a, i):
				b = ''.join(j)
				q = b

				if isprime(b) == 0:
					print(i)
					print(b)

					break
			i += 1
			if i == len(a) + 1:
				print(len(a))
				print(n)
				break
