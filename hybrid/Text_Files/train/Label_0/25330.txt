#include<bits/stdc++.h>
#define X first
#define Y second
#define N 200005
using namespace std;
vector<int>vec[N];
int T[N<<2],b[N],sum[N],New[N];
int ql,qr,n,i,k,ans,m,Zero;
void build(int x,int l,int r){
	T[x]=-1e9;if (l==r) return;
  int mid=(l+r)>>1;
  build(x<<1,l,mid);
  build(x<<1|1,mid+1,r);
}
int Query(int x,int l,int r){
	if (ql<=l&&r<=qr) return T[x];
	int mid=(l+r)>>1,ret=-1e9;
	if (ql<=mid) ret=max(ret,Query(x<<1,l,mid));
	if (qr>mid)  ret=max(ret,Query(x<<1|1,mid+1,r));
	return ret;
}
void Update(int x,int l,int r,int pos,int val){
	if (l==r) {T[x]=max(T[x],val);return;}
	int mid=(l+r)>>1;
	if (pos<=mid) Update(x<<1,l,mid,pos,val);
	         else Update(x<<1|1,mid+1,r,pos,val);
	T[x]=max(T[x<<1],T[x<<1|1]);
}
int main(){
	scanf("%d",&n);
	for (i=1;i<=n;i++) 
	  scanf("%d",&b[i]),sum[i]=sum[i-1]+b[i];
	scanf("%d",&m);
	for (i=1;i<=m;i++)
	  scanf("%d%d",&ql,&qr),vec[ql].push_back(qr);
	build(1,0,n); 
	Update(1,0,n,0,0-sum[0]);
	for (i=1;i<=n;i++){
		ql=qr=i-1;Zero=Query(1,0,n)+sum[i-1]+(b[i]==0);
		for (k=0;k<vec[i].size();k++){
			ql=i-1;qr=vec[i][k];
			New[k]=sum[vec[i][k]]+Query(1,0,n);
		}
		Update(1,0,n,i,Zero-sum[i]);
		for (k=0;k<vec[i].size();k++)
		  Update(1,0,n,vec[i][k],New[k]-sum[vec[i][k]]);
	}
	ql=qr=n;ans=Query(1,0,n)+sum[n];
	printf("%d\n",n-ans);
}
