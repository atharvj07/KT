from __future__ import print_function
from collections import *
from sys import stdin
from heapq import *
from math import *
g = lambda : stdin.readline().strip()
gl = lambda : g().split()
gil = lambda : [int(var) for var in gl()]
gfl = lambda : [float(var) for var in gl()]
gcl = lambda : list(g())
gbs = lambda : [int(var) for var in g()]
rr = lambda x: reversed(range(x))
mod = int(1000000000.0) + 7
inf = float('inf')
(n, m) = gil()
mat = [gil() for _ in range(n)]
p = set()
for j in range(m):
	(l, r) = (0, 0)
	while r + 1 < n:
		if mat[r + 1][j] < mat[r][j]:
			if l != r:
				p.add((l + 1, r + 1))
			l = r + 1
		r += 1
	if l != r:
		p.add((l + 1, r + 1))
p = list(p)
p.sort(reverse=True)
endMax = [i for i in range(n + 1)]
h = []
for start in range(1, n + 1):
	while p and p[-1][0] == start:
		heappush(h, -p.pop()[1])
	while h and -h[0] <= start:
		heappop(h)
	if h:
		endMax[start] = -h[0]
ans = []
for _ in range(gil()[0]):
	(l, r) = gil()
	ans.append('Yes' if endMax[l] >= r else 'No')
print('\n'.join(ans))
