import java.io.*;
import java.util.*;

public class CF1454F extends PrintWriter {
	CF1454F() { super(System.out); }
	static class Scanner {
		Scanner(InputStream in) { this.in = in; } InputStream in;
		byte[] bb = new byte[1 << 15]; int i, n;
		byte getc() {
			if (i == n) {
				i = n = 0;
				try { n = in.read(bb); } catch (IOException e) {}
			}
			return i < n ? bb[i++] : 0;
		}
		int nextInt() {
			byte c = 0; while (c <= ' ') c = getc();
			int a = 0; while (c > ' ') { a = a * 10 + c - '0'; c = getc(); }
			return a;
		}
	}
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1454F o = new CF1454F(); o.main(); o.flush();
	}

	int[] ft;
	void update(int i, int n) {
		while (i < n) {
			ft[i]++;
			i |= i + 1;
		}
	}
	int query(int i) {
		int sum = 0;
		while (i >= 0) {
			sum += ft[i];
			i &= i + 1; i--;
		}
		return sum;
	}
	void main() {
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			int[] aa = new int[n];
			Integer[] ii = new Integer[n];
			for (int i = 0; i < n; i++) {
				aa[i] = sc.nextInt();
				ii[i] = i;
			}
			Arrays.sort(ii, (i, j) -> aa[j] - aa[i]);
			int[] ll = new int[n];
			int[] rr = new int[n];
			for (int l = n, r = -1, i = 0, j; i < n; i = j) {
				int a = aa[ii[i]];
				j = i + 1;
				while (j < n && aa[ii[j]] == a)
					j++;
				for (int h = i; h < j; h++) {
					int o = ii[h];
					ll[o] = Math.min(l, o);
					rr[o] = Math.max(r, o);
				}
				for (int h = i; h < j; h++) {
					int o = ii[h];
					l = Math.min(l, o);
					r = Math.max(r, o);
				}
			}
			for (int i = 0, j = n - 1; i < j; i++, j--) {
				Integer tmp = ii[i]; ii[i] = ii[j]; ii[j] = tmp;
			}
			boolean[] bad = new boolean[n];
			ft = new int[n];
			for (int i = 0, j; i < n; i = j) {
				int a = aa[ii[i]];
				j = i + 1;
				while (j < n && aa[ii[j]] == a)
					j++;
				for (int h = i; h < j; h++) {
					int o = ii[h];
					if (query(rr[o]) != query(ll[o] - 1))
						bad[o] = true;
				}
				for (int h = i; h < j; h++) {
					int o = ii[h];
					update(o, n);
				}
			}
			int[] pp = new int[n];
			int[] qq = new int[n];
			for (int a = 0, i = 0; i < n; i++)
				pp[i] = a = Math.max(a, aa[i]);
			for (int a = 0, i = n - 1; i >= 0; i--)
				qq[i] = a = Math.max(a, aa[i]);
			int x = -1, y = -1, z = -1;
			for (int i = 0; i < n; i++) {
				if (bad[i])
					continue;
				int l = ll[i], r = rr[i], a = aa[i];
				if (l - 1 >= 0 && r + 1 < n && pp[l - 1] == a && a == qq[r + 1]) {
					x = l; y = r - l + 1; z = n - x - y;
					break;
				}
			}
			if (x == -1)
				println("NO");
			else {
				println("YES");
				println(x + " " + y + " " + z);
			}
		}
	}
}
