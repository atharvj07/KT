#include <bits/stdc++.h>
using namespace std;
long long ans, mi, tt;
long long i, o, p, j, k, l, n, m, ct;
long long a[1000005], cl, lp[1000005], rp[1000005];
int main() {
  scanf("%I64d%I64d", &n, &m);
  for (i = 1; i <= n; i++) scanf("%I64d", &a[i]);
  memset(lp, 0, sizeof(lp));
  memset(rp, 0, sizeof(rp));
  for (i = 1; i <= n; i++) {
    if ((i - 1) % m == 0)
      lp[i] = lp[i - 1] + a[i];
    else
      lp[i] = lp[i - 1];
  }
  for (i = n; i; i--) {
    if ((n - i) % m == 0)
      rp[i] = rp[i + 1] + a[i];
    else
      rp[i] = rp[i + 1];
  }
  ans = -1;
  for (i = 1; i <= n; i++) {
    tt = ((i + m - 1) / m) * a[i] - lp[i] - (n - i + m) / m * a[i] + rp[i];
    if (tt < ans || ans < 0) ans = tt;
  }
  printf("%I64d", ans * 2);
  return 0;
}
