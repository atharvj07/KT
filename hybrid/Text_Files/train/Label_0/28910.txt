#include <bits/stdc++.h>
using namespace std;
long long int solve() {
  long long int n, q, m;
  cin >> n >> q >> m;
  vector<long long int> v(n);
  for (long long int j = 0; j < n; j++) {
    cin >> v[j];
  }
  vector<pair<long long int, pair<long long int, long long int>>> query(q);
  for (long long int j = 0; j < q; j++) {
    long long int t, l, r;
    cin >> t >> l >> r;
    query[j] = {t, {l - 1, r - 1}};
  }
  reverse(query.begin(), query.end());
  vector<long long int> b(m);
  for (long long int j = 0; j < m; j++) {
    cin >> b[j];
    b[j]--;
  }
  for (long long int j = 0; j < m; j++) {
    long long int idx = b[j];
    for (long long int i = 0; i < q; i++) {
      long long int l = query[i].second.first;
      long long int r = query[i].second.second;
      if (l <= idx && r >= idx) {
        if (query[i].first == 2) {
          idx = r + l - idx;
        } else {
          if (idx == l) {
            idx = r;
          } else {
            idx = idx - 1;
          }
        }
      }
    }
    cout << v[idx] << " ";
  }
  return 0;
}
signed main() {
  ios::sync_with_stdio(0);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int t;
  t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
