import java.util.*;
import java.io.*;
import static java.lang.Math.*;

public class Solution {
	static Scanner scn = new Scanner(System.in);
	static StringBuilder sb = new StringBuilder();
	static PrintWriter out = new PrintWriter(System.out);
	// solve more interactive problems as for now.
	
	public static void main(String[] HastaLaVistaLa) {
		int t = 1;
		for(int tests = 0; tests < t; tests++) solve();
		out.println(sb);
		out.close();
	}
	public static void solve() {
		// Road To Expert Day 10
		int k = scn.nextInt();
		String str = "aeiouuaeioouaeiiouaeeioua";
		if(k < 25) sb.append(-1);
		else {
			int a = -1, b = -1;
			for(int i = 5; i <= k; i++) {
				if(k % i == 0) {
					if((k / i) >= 5) {
						a = i;
						b = (k / i);
					}
				}
			}
			if(a == -1 && b == -1) sb.append(-1);
			else {
				char[][] ch = new char[a][b];
				for(int i = 0; i < a; i++) Arrays.fill(ch[i], 'a');
				for(int i = 0; i < 5; i++) {
					String s1 = str.substring(5 * i, 5 * i + 5);
					for(int j = 0; j < 5; j++) {
						ch[i][j] = s1.charAt(j);
					}
				}
				String abc = "aeiou";
				for(int i = 5; i < a; i++)
					for(int j = 0; j < 5; j++)
						ch[i][j] = abc.charAt(j);
				for(int i = 5; i < b; i++)
					for(int j = 0; j < 5; j++)
						ch[j][i] = abc.charAt(j);		
				for(int i = 0; i < a; i++)
					for(int j = 0; j < b; j++)
						sb.append(ch[i][j]);
			}
		}
	}
}
