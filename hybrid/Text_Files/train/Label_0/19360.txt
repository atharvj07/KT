#include <iostream>
#include <string>

using namespace std;

const int dx[] = { 0, 1, 0, -1 };
const int dy[] = { 1, 0, -1, 0 };

int main(){
	while(true){
		int w, h;
		cin >> w >> h;
		if(w == 0 && h == 0){ break; }
		int x = 1, y = 1, dir = 0;
		while(true){
			string command;
			cin >> command;
			if(command == "STOP"){
				break;
			}else if(command == "LEFT"){
				dir = (dir + 3) % 4;
			}else if(command == "RIGHT"){
				dir = (dir + 1) % 4;
			}else if(command == "FORWARD"){
				int amount;
				cin >> amount;
				x += dx[dir] * amount;
				y += dy[dir] * amount;
			}else if(command == "BACKWARD"){
				int amount;
				cin >> amount;
				x -= dx[dir] * amount;
				y -= dy[dir] * amount;
			}
			if(x < 1){ x = 1; }
			if(x > w){ x = w; }
			if(y < 1){ y = 1; }
			if(y > h){ y = h; }
		}
		cout << x << " " << y << endl;
	}
	return 0;
}
