def lcs(X, Y):
	n = len(X)
	m = len(Y)
	maxx = int_min
	for t in range(1):
		it = t
		total = 0
		for j in range(n):
			if it < m and X[j] == Y[it]:
				it += 1
				total += 1
		maxx = max(maxx, total)
	return maxx
