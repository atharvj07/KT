table = []
while True:
	try:
		l = input()
	except EOFError:
		break
	table.append(l)
tries = int(table[0])
for i in range(1, tries + 1):
	line = table[i].split()
	matrixLength1 = int(line[0])
	matrixLength2 = int(line[1])
	coord1 = int(line[2])
	coord2 = int(line[3])
	leftLength = coord1 - 1
	rightLength = matrixLength1 - coord1
	if leftLength > rightLength:
		max = leftLength
	else:
		max = rightLength
	upperLength = coord2 - 1
	lowerLength = matrixLength2 - coord2
	if upperLength > lowerLength:
		high = upperLength
	else:
		high = lowerLength
	steps = max + high
	print(steps)
