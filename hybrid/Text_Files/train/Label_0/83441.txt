(a, b) = map(int, input().split(' '))
for i in range(1, a - b + 1):
	print(i)
curr = i
if b % 2 == 0:
	for i in range(b, 0, -1):
		if i % 2 == 0:
			curr = curr + i
		else:
			curr = curr - i
		print(curr)
else:
	for i in range(b, 0, -1):
		if i % 2 == 1:
			curr = curr + i
		else:
			curr = curr - i
		print(curr)
