#include <iostream>
using namespace std;
long n, m, l, a[100][100], b, c[100][100];
int main () {
	cin >> n >> m >> l;
	
	for (int i = 0; i < n; i++)
		for (int j = 0; j < m; j++)
			cin >> a[i][j];
	
	for (int i = 0; i < m; i++)
		for (int j = 0; j < l; j++) {
			cin >> b;
			for (int k = 0; k < n; k++)
				c[k][j] += a[k][i] * b;
		}
	
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < l - 1; j++)
			cout << c[i][j] << " ";
		cout << c[i][l -1] << "\n";
	}
}
