import sys, os, io
input = sys.stdin.readline
n = int(input())
a = []
w = 0
b = 0
for i in range(n):
	c = input().strip()
	w += c.count('W')
	b += c.count('B')
	a.append(c)
if w > n or b > n:
	print(0)
	exit()

def ncr(n, r, p):
	num = den = 1
	for i in range(r):
		num = num * (n - i) % p
		den = den * (i + 1) % p
	return num * pow(den, p - 2, p) % p
mod = 998244353
total = ncr(2 * n - w - b, n - w, mod)
bad = 1
if 'BB' in a or 'WW' in a:
	bad = 0
bw = 1
wb = 1
for i in a:
	if i[0] == '?' and i[1] == '?':
		bad *= 2
		bad %= mod
	if i[0] == 'B' or i[1] == 'W':
		wb = 0
	if i[0] == 'W' or i[1] == 'B':
		bw = 0
print((total - bad + bw + wb) % mod)
