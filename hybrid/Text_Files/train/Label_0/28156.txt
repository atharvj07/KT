#include <bits/stdc++.h>
const int N = 51, M = 1e9 + 7, mod2 = M + 1 >> 1;
void inc(int &x, int y) {
  x += y;
  if (x >= M) x -= M;
}
int n, T, i, x, y, g[3][N][N][N], j, k, l, ans, c[3], v, f0[2502][N],
    f12[2502][N][N], t1, t2;
int main() {
  scanf("%d%d", &n, &T), f0[0][0] = f12[0][0][0] = 1;
  for (i = 1; i <= n; i++) {
    scanf("%d%d", &x, &y), c[--y]++;
    if (!y) {
      t1 += x;
      for (j = c[0]; j; j--)
        for (v = t1; v >= x; v--) inc(f0[v][j], f0[v - x][j - 1]);
    } else {
      t2 += x;
      for (j = c[1]; j >= (y == 1); j--)
        for (k = c[2]; k >= (y == 2); k--)
          for (v = t2; v >= x; v--)
            inc(f12[v][j][k], f12[v - x][j - (y == 1)][k - (y == 2)]);
    }
  }
  g[0][0][0][0] = g[1][0][0][0] = g[2][0][0][0] = 1;
  for (i = 0; i <= c[0]; i++)
    for (l = 0; l <= T; l++)
      for (j = 0; j <= c[1]; j++)
        for (k = 0; k <= c[2]; k++) {
          if (i)
            g[0][i][j][k] =
                1ll * i * (g[1][i - 1][j][k] + g[2][i - 1][j][k]) % M;
          if (j)
            g[1][i][j][k] =
                1ll * j * (g[0][i][j - 1][k] + g[2][i][j - 1][k]) % M;
          if (k)
            g[2][i][j][k] =
                1ll * k * (g[0][i][j][k - 1] + g[1][i][j][k - 1]) % M;
          inc(ans, (1ll * g[0][i][j][k] + g[1][i][j][k] + g[2][i][j][k]) *
                       f0[l][i] % M * f12[T - l][j][k] % M);
        }
  printf("%lld", 1ll * ans * mod2 % M);
}
