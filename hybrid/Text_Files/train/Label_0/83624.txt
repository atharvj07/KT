field = []

def full_line(x, y, vx, vy):
	global field
	summ = 0
	for k in range(0, 5):
		if x == len(field) or y == len(field[x]) or x == -1 or (y == -1) or ('O' == field[x][y]):
			return False
		if 'X' == field[x][y]:
			summ += 1
		x += vx
		y += vy
	return summ > 3
for i in range(0, 10):
	field.append(list(input()))
for i in range(0, 10):
	for j in range(0, 10):
		if 'O' != field[i][j] and (full_line(i, j, 0, 1) or full_line(i, j, 1, 0) or full_line(i, j, 1, 1) or full_line(i, j, 1, -1)):
			print('YES')
			exit()
print('NO')
