_in = input()
(n, x, y) = (int(_in.strip().split()[0]), int(_in.strip().split()[1]), int(_in.strip().split()[2]))
_in = input()
_in = _in.strip().split()
arr = [int(val) for val in _in]
for i in range(n):
	idx = i - 1
	flag = 0
	while idx >= 0 and x > 0:
		if arr[idx] <= arr[i]:
			flag = 1
		if idx == i - x:
			break
		idx = idx - 1
	if flag == 1:
		continue
	idx = i + 1
	flag = 0
	while idx < n and y > 0:
		if arr[idx] <= arr[i]:
			flag = 1
		if idx == i + y:
			break
		idx = idx + 1
	if flag == 1:
		continue
	ans = i + 1
	break
print(ans)
