LEN1 = 31
LEN2 = 50
MOD = 10 ** LEN2

def binpow(a, b, mod):
	x = 1
	y = a
	while b > 0:
		if b % 2:
			x = x * y % mod
		y = y * y % mod
		b //= 2
	return x

def solve(n):
	n = n * 10 ** LEN1
	diff = (2 ** LEN2 - n) % 2 ** LEN2
	if diff < 0:
		diff += 2 ** LEN2
	if diff % 5 == 0:
		diff += 2 ** LEN2
	n += diff
	n %= 5 ** LEN2
	k = 0
	for i in range(5):
		if 2 ** k % 5 == n % 5:
			break
		k += 1
	Pow = 1
	Pow2 = 2 ** k
	Pow25 = 2 ** 4
	for i in range(2, LEN2 + 1):
		res = n % (25 * Pow)
		for j in range(5):
			if Pow2 % (25 * Pow) == res:
				break
			k += 4 * Pow
			Pow2 *= Pow25
			Pow2 %= MOD
		if Pow2 % (25 * Pow) != res:
			print('bad\n')
		Pow *= 5
		Pow25 = Pow25 * Pow25 % MOD * Pow25 % MOD * Pow25 % MOD * Pow25 % MOD
	return k
n = int(input())
a = [int(input()) for i in range(n)]
for x in a:
	print(solve(x))
