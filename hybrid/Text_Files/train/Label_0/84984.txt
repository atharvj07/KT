firstLineOfInput = input().split()
(n, m, k) = (int(firstLineOfInput[0]), int(firstLineOfInput[1]), int(firstLineOfInput[2]))
roads = []
for i in range(m):
	roads.append(input().split())
storageCitys = []
if k > 0:
	storageCitys = input().split()
isStorage = [False for i in range(n + 1)]
for storageCity in storageCitys:
	isStorage[int(storageCity)] = True
minPay = -1
for road in roads:
	aCityOfThisRoad = int(road[0])
	anotherCityOfThisRoad = int(road[1])
	if isStorage[aCityOfThisRoad] and (not isStorage[anotherCityOfThisRoad]) or (not isStorage[aCityOfThisRoad] and isStorage[anotherCityOfThisRoad]):
		lengthOfThisRoad = int(road[2])
		if minPay == -1:
			minPay = lengthOfThisRoad
		elif lengthOfThisRoad < minPay:
			minPay = lengthOfThisRoad
print(minPay)
