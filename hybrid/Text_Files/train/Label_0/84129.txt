import math

def deli(x):
	b = []
	for i in range(2, x // 2 + 6):
		if x % i == 0:
			b.append(i)
	b.append(x)
	return b
n = int(input())
a = list(map(int, input().split()))
a.sort()
mini = min(a)
ans = 0
for i in range(n):
	if a[i] != mini:
		c = deli(a[i])
		for j in c:
			fr = mini + a[i]
			tec = mini * j + a[i] // j
			if fr - tec > ans:
				ans = fr - tec
print(sum(a) - ans)
