#include <bits/stdc++.h>
using namespace std;
const int N = 200009;
int delta[N], a[N];
int main() {
  int n;
  int len = 1;
  long long sum = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int t;
    cin >> t;
    if (t == 1) {
      int x, y;
      cin >> y >> x;
      y = min(y, len);
      delta[y - 1] += x;
      sum += 1LL * x * y;
    } else if (t == 2) {
      int x;
      cin >> x;
      a[len++] = x;
      sum += x;
    } else {
      len--;
      sum -= a[len] + delta[len];
      a[len] = 0;
      delta[len - 1] += delta[len];
      delta[len] = 0;
    }
    if (len == 0)
      printf("0.000000\n");
    else
      printf("%.6lf\n", 1.0 * sum / len);
  }
}
