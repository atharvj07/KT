def f():
	input()
	(n, k) = map(int, input().split())
	if n == 1:
		print(0 if k else 1)
		return
	g = [set() for i in range(n)]
	q = set()
	for i in range(n - 1):
		(a, b) = map(int, input().split())
		a -= 1
		b -= 1
		g[a].add(b)
		g[b].add(a)
	for i in range(n):
		if len(g[i]) == 1:
			q.add(i)
	ans = set()
	while k and q:
		k -= 1
		p = set()
		for i in q:
			ans.add(i)
			if len(g[i]) == 0:
				continue
			t = list(g[i])[0]
			g[t].discard(i)
			if len(g[t]) == 1:
				p.add(t)
		q = p - q
	print(n - len(ans))
for _ in range(int(input())):
	f()
