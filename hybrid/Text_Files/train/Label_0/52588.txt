import java.io.*;
import java.util.*;

public class HelloWorld{

	public static void main(String[] args) throws Exception
	{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter writer = new PrintWriter(System.out);
    int k = Integer.parseInt(br.readLine());
    while(k-- > 0)
    {
      int n = Integer.parseInt(br.readLine());
      String s = br.readLine();
      String t = br.readLine();
      char[] c1 = s.toCharArray();
      char[] c2 = t.toCharArray();
      int[] A = new int[26];
      for(int i = 0; i < n; i++)
      {
        A[s.charAt(i)-'a']++;
        A[t.charAt(i)-'a']++;
      }
      boolean flag = true;
      for(int i = 0; i < 26; i++)
      {
        if(A[i]%2 == 1) flag = false;
      }
      if(!flag)
      {
        writer.println("No");
        continue;
      }
      int count = 0;
      ArrayList<Integer> ls = new ArrayList<>();
      for(int i = 0; i < n; i++)
      {
        if(c1[i] != c2[i])
        {
          int index = -1;
          for(int j = i+1; j < n; j++)
          {
            if(c1[j] == c1[i])
            {
              index = j;
              break;
            }
          }
          if(index == -1)
          {
            for(int j = i+1; j < n; j++)
            {
              if(c2[j] == c1[i])
              {
                index = j;
                break;
              }
            }
            count++;
            ls.add(index+1);
            ls.add(index+1);
            char temp = c1[index];
            c1[index] = c2[index];
            c2[index] = temp;
            count++;
            ls.add(index+1);
            ls.add(i+1);
            temp = c1[index];
            c1[index] = c2[i];
            c2[i] = temp;
          }
          else
          {
            count++;
            ls.add(index+1);
            ls.add(i+1);
            char temp = c1[index];
            c1[index] = c2[i];
            c2[i] = temp;
          }
        }
      }
      writer.println("Yes");
      writer.println(count);
      for(int i = 0; i < ls.size(); i += 2)
      {
        writer.println(ls.get(i)+" "+ls.get(i+1));
      }
    }
		br.close();
		writer.close();
	}

}
