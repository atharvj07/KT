#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>
#include <queue>
#include <cmath>
using namespace std;
typedef long long ll;
#define rep(i,n) for(int i=0;i<n;i++)
#define MP make_pair
#define PB push_back
ll inf = (1LL)<<60;

ll dp[100001][20];


int main(){
    cin.tie(0);
    ios::sync_with_stdio(false);
    int n,m;
    cin >> n >> m;
    vector<ll> t(n);
    rep(i,n)cin >> t[i];
    vector<vector<pair<int,ll> > >g(n);
    rep(i,m){
        int a,b;
        ll c;
        cin >> a >> b >> c;
        a--;
        b--;
        g[a].PB(MP(b,c));
        g[b].PB(MP(a,c));
    }
    priority_queue<pair<ll,pair<ll,int> >,vector<pair<ll,pair<ll,int> > >,greater<pair<ll,pair<ll,int> > > >pq;
    pq.push(MP((ll)0,MP((ll)0,0)));
    vector<int> parent(n);
    parent[0]=-1;
    vector<pair<ll,ll>  > place(n,MP(inf,inf));
    place[0] = MP(0,0);
    while(!pq.empty()){
        auto xxxxxxxx = pq.top();
        pq.pop();
        auto x = xxxxxxxx.first;
        auto y = xxxxxxxx.second.first;
        auto p = xxxxxxxx.second.second;
        if(make_pair(x,y) > place[p]) continue;
        for(auto& e : g[p]){
            if(MP(x+e.second,y+1)>place[e.first])continue;
            if(MP(x+e.second,y+1)==place[e.first]){
                if(t[parent[e.first]]>t[p]){
                    parent[e.first] = p;
                    pq.push(MP(x+e.second,MP(y+1,e.first)));
                }
            }else{
                place[e.first] = MP(x+e.second,y+1);
                parent[e.first] = p;
                pq.push(MP(x+e.second,MP(y+1,e.first)));
            }
        }
    }
    rep(i,n){
        rep(j,20){
            dp[i][j] = -1;
        }
    }
    rep(i,n){
        dp[i][0] = parent[i];
    }
    rep(j,19){
        rep(i,n){
            if(dp[i][j]>=0){
                dp[i][j+1] = dp[dp[i][j]][j];
            }
        }
    }
    int k;
    cin >> k;
    rep(i,k){
        int a,b;
        ll c;
        cin >> a >> b >> c;
        a--;
        ll ans = 0;
        ans = place[a].first;
        if(place[a].second<=b){
            cout << max(0LL,ans-c) << endl;
        }else{
            int now = a;
            int d = place[a].second-b;
            for(int i = 20; i>=0; i--){
                if(d < (1 << i)) continue;
                now = dp[now][i];
                d -= (1<<i);
            }
            cout << ans - min(place[now].first,c) << endl;
        }
    }

    return 0;
}
