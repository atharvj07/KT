import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Arrays;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;
import java.io.BufferedReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        FastScanner in = new FastScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        List<String> edges = new ArrayList<>();

        public void solve(int testNumber, FastScanner in, PrintWriter out) {
            int n = in.nextInt();
            int[] want = new int[n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 3; j++) {
                    want[i] |= in.nextInt() << j;
                }
            }
            int[] willTest = new int[n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 3; j++) {
                    willTest[i] |= in.nextInt() << j;
                }
            }

            int[] ct = new int[n];
            Arrays.fill(ct, -1);
            ct[0] = 1;

            int[] whoPropagates = new int[1 << 3];
            Arrays.fill(whoPropagates, -1);
            whoPropagates[(1 << 3) - 1] = 0;

            // Step 0. Greedily propagate whatever is possible.
            int[] q = new int[1 << 3];
            int qt = 0;
            int qh = 1;
            q[0] = (1 << 3) - 1;
            while (qt < qh) {
                int mask = q[qt++];
                for (int i = 0; i < n; i++) {
                    if (want[i] == mask) {
                        int nmask = mask & willTest[i];
                        if (whoPropagates[nmask] < 0) {
                            ct[i] = 1;
                            addEdge(whoPropagates[mask], i);
                            whoPropagates[nmask] = i;
                            q[qh++] = nmask;
                        }
                    }
                }
            }

            // Step 1. Only 2-bit masks are those that may be constructed
            // from more than one incoming edge.
            for (int i = 0; i < n; i++) {
                if (whoPropagates[want[i]] >= 0) {
                    continue;
                }
                if (Integer.bitCount(want[i]) != 2) {
                    continue;
                }
                if (ct[i] >= 0) {
                    throw new AssertionError();
                }
                for (int bit = 0; bit < 3; bit++) {
                    if ((want[i] & (1 << bit)) == 0) {
                        continue;
                    }
                    if (whoPropagates[1 << bit] < 0) {
                        out.println("Impossible");
                        return;
                    }
                    addEdge(whoPropagates[1 << bit], i);
                }
                ct[i] = 0;
                whoPropagates[want[i]] = i;
            }

            for (int i = 0; i < n; i++) {
                if (whoPropagates[want[i]] < 0) {
                    out.println("Impossible");
                    return;
                }
                if (ct[i] < 0) {
                    ct[i] = 0;
                    addEdge(whoPropagates[want[i]], i);
                }
            }

            out.println("Possible");
            for (int i = 0; i < n; i++) {
                if (i > 0) {
                    out.print(" ");
                }
                out.print(ct[i]);
            }
            out.println();
            out.println(edges.size());
            for (String s : edges) {
                out.println(s);
            }
        }

        private void addEdge(int a, int b) {
            edges.add((a + 1) + " " + (b + 1));
        }

    }

    static class FastScanner {
        private BufferedReader in;
        private StringTokenizer st;

        public FastScanner(InputStream stream) {
            in = new BufferedReader(new InputStreamReader(stream));
        }

        public String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
