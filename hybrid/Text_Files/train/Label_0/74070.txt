(n, k) = map(int, input().split())
arr = [0 for i in range(2 ** k)]
for _ in range(n):
	s = input()
	s = s.replace(' ', '')
	kk = int(s, 2)
	arr[kk] = 1
f = False
for i in range(2 ** k):
	if arr[i] == 0:
		continue
	if i == 0:
		f = True
	for j in range(i + 1, 2 ** k):
		if arr[j] == 0:
			continue
		if i & j == 0:
			f = True
if f:
	print('YES')
else:
	print('NO')
