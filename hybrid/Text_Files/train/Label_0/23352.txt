#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, ans = 0, temp;
  cin >> n >> k;
  int arr[] = {0, 0, 0, 0, 0, 0};
  for (int i = 0; i < n; i++) {
    cin >> temp;
    arr[temp]++;
  }
  for (int i = 0; i <= 5 - k; i++) {
    ans += arr[i];
  }
  cout << ans / 3;
  return 0;
}
