#include <bits/stdc++.h>
using namespace std;
using ll = long long;
ll gcd(ll u, ll v) {
  while (v != 0) {
    ll r = u % v;
    u = v;
    v = r;
  }
  return u;
}
int getBest(vector<ll>& vec, int start, int best) {
  ll curGcd = vec[start];
  for (int i = start + 1; i < vec.size(); ++i) {
    curGcd = gcd(curGcd, vec[i]);
    if (curGcd == 1) {
      return i - start;
    }
  }
  return 3000;
}
int main() {
  int n;
  cin >> n;
  int one_count = 0;
  vector<ll> vec(n);
  for (int i = 0; i < n; ++i) {
    ll num;
    cin >> num;
    if (num == 1) {
      one_count++;
    }
    vec[i] = num;
  }
  if (one_count > 0) {
    cout << n - one_count << endl;
    return 0;
  }
  int best = 3000;
  for (int i = 0; i < n - 1; ++i) {
    int _count = getBest(vec, i, best);
    best = min(best, _count);
  }
  if (best == 3000) {
    cout << "-1" << endl;
    return 0;
  }
  cout << n + best - 1 << endl;
  return 0;
}
