def solve(ans):
	s = ''
	(x, y) = (0, 0)
	for i in range(len(ans)):
		(px, py) = ans[i]
		if py < y:
			print('NO')
			return
		else:
			while px != x:
				x += 1
				s += 'R'
			while py != y:
				y += 1
				s += 'U'
	print('YES')
	print(s)
