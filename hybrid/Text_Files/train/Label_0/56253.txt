import static java.util.Arrays.deepToString;

import java.io.*;
import java.math.*;
import java.util.*;

public class D {
	static int n, S;

	static void go(int[][] B, int[] a, boolean[] used, int y, int x) {
		if (y == n) {
			check(B);
		}
		int nx = x + 1, ny = y;
		if (nx >= n) {
			nx = 0;
			ny++;
		}
		outer: for (int i = 0; i < a.length; i++) {
			if (used[i] || (i > 0 && a[i - 1] == a[i] && !used[i - 1]))
				continue;
			B[y][x] = a[i];
			if (x == n - 1) {
				if (y == 0 && B[0][n - 1] > B[0][0])
					continue;
				int tmp = S;
				for (int k = 0; k < n; k++) {
					tmp -= B[y][k];
				}
				if (tmp != 0)
					continue;
			}
			if (y == n - 2) {
				if (x == 1) {
					int tmp = 0;
					for (int j = 0; j < n - 1; j++) {
						tmp += B[j][0];
						tmp -= B[j][n - 1 - j];
					}
					if (tmp != 0)
						continue;
				} else if (x == n - 1) {
					int tmp = 0;
					for (int j = 0; j < n - 1; j++) {
						tmp += B[j][n - 1];
						tmp -= B[j][j];
					}
					if (tmp != 0)
						continue;
				}
			}
			if (y == n - 1) {
				if (x == 0 && B[n - 1][0] < B[0][0])
					continue;
				int tmp = S;
				for (int k = 0; k < n; k++) {
					tmp -= B[k][x];
				}
				if (tmp != 0)
					continue;
			}
			used[i] = true;
			go(B, a, used, ny, nx);
			used[i] = false;
		}
	}

	static void check(int[][] B) {
		int d1 = 0, d2 = 0;
		for (int i = 0; i < n; i++) {
			d1 += B[i][i];
			d2 += B[i][n - 1 - i];
		}
		if (d1 != S || d2 != S)
			return;
		for (int i = 0; i < n; i++) {
			int col = 0;
			for (int j = 0; j < n; j++) {
				col += B[j][i];
			}
			if (col != S)
				return;
		}
		writer.println(S);
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				writer.print(B[i][j] + " ");
			}
			writer.println();
		}
		writer.close();
		System.exit(0);
	}

	static void solve() {
		n = nextInt();
		int[] a = new int[n * n];
		for (int i = 0; i < a.length; i++) {
			a[i] = nextInt();
			S += a[i];
		}
		// ArrayList<Integer> list = new ArrayList<>();
		// for (int v : a)
		// list.add(v);
		// Collections.shuffle(list, new Random(239));
		// for (int i = 0; i < a.length; i++) {
		// a[i] = list.get(i);
		// }
		Arrays.sort(a);
		S /= n;
		go(new int[n][n], a, new boolean[a.length], 0, 0);
	}

	public static void main(String[] args) throws Exception {
		reader = new BufferedReader(new InputStreamReader(System.in));
		writer = new PrintWriter(System.out);

		setTime();
		solve();
		printTime();
		printMemory();

		writer.close();
	}

	static BufferedReader reader;
	static PrintWriter writer;
	static StringTokenizer tok = new StringTokenizer("");
	static long systemTime;

	static void debug(Object... o) {
		System.err.println(deepToString(o));
	}

	static void setTime() {
		systemTime = System.currentTimeMillis();
	}

	static void printTime() {
		System.err.println("Time consumed: " + (System.currentTimeMillis() - systemTime));
	}

	static void printMemory() {
		System.err.println("Memory consumed: " + (Runtime.getRuntime().totalMemory() - Runtime.getRuntime().freeMemory()) / 1000 + "kb");
	}

	static String next() {
		while (!tok.hasMoreTokens()) {
			String w = null;
			try {
				w = reader.readLine();
			} catch (Exception e) {
				e.printStackTrace();
			}
			if (w == null)
				return null;
			tok = new StringTokenizer(w);
		}
		return tok.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

	static long nextLong() {
		return Long.parseLong(next());
	}

	static double nextDouble() {
		return Double.parseDouble(next());
	}

	static BigInteger nextBigInteger() {
		return new BigInteger(next());
	}
}
