import static java.lang.Math.*;
import static java.util.Arrays.*;
import java.util.*;
import java.io.*;

public class Main {
	public static void main(String[] args) {
		Class<?> here = new Object(){}.getClass().getEnclosingClass();
		try {
			String packageName = here.getPackage().getName();
			System.setIn(new FileInputStream("src/" + packageName + "/input.txt"));
//			System.setOut(new PrintStream(new FileOutputStream("src/" + packageName + "/result.txt")));
		} catch (FileNotFoundException e) {
		} catch (NullPointerException e) {
		}

		try {
			String className = here.getName();
			Object object = Class.forName(className).newInstance();
			object.getClass().getMethod("run").invoke(object);
		} catch (Exception e) {
			e.printStackTrace();
		}
	}

	void tr(Object... os) {
		System.err.println(deepToString(os));
	}

	Scanner sc = new Scanner(System.in);
	public void run() {
		int n = sc.nextInt();
		int k = sc.nextInt();

		int[] q = new int[n];
		int[] target = new int[n];
		for (int i = 0; i < n; i++) q[i] = sc.nextInt() - 1;
		for (int i = 0; i < n; i++) target[i] = sc.nextInt();

		int[] t = new int[n];
		for (int i = 0; i < n; i++) t[q[i]] = i;

		int[] a = new int[n];
		for (int i = 0; i < n; i++) a[i] = i + 1;

		boolean[] mark = new boolean[2 * k + 1];
		mark[k] = same(a, target);

		{
			int[] work = a.clone();
			for (int i = 1; i <= k; i++) {
				work = apply(work, q);
				mark[k + i] = same(work, target);
			}
		}
		{
			int[] work = a.clone();
			for (int i = 1; i <= k; i++) {
				work = apply(work, t);
				mark[k - i] = same(work, target);
			}
		}


		System.out.println(solve(mark, k) ? "YES" : "NO");
	}

	boolean solve(boolean [] mark, int k) {
		if (mark[k]) return false;

		if (k == 1) {
			return mark[k + 1] || mark[k - 1];
		}
		if (k == 2) {
			return (!mark[k + 1] && mark[k + 2]) ||
					(!mark[k - 1] && mark[k - 2]);
		}

		boolean result = false;

		if (!mark[k + 1] || !mark[k-1]) {
			int near = -1;
			for (int i = 1; i <= k; i++) {
				if (mark[k + i]) {
					near = i;
					break;
				}
			}
			if (near >= 0)
				result = result || ((k - near) % 2 == 0);
		}

		if (!mark[k + 1] || !mark[k-1]) {
			int near = -1;
			for (int i = 1; i <= k; i++) {
				if (mark[k - i]) {
					near = i;
					break;
				}
			}
			if (near >= 0)
				result = result || ((k - near) % 2 == 0);
		}


		return result;
	}

	int[] apply(int []a, int[] q) {
		int n = a.length;
		int [] res = new int[n];
		for (int i = 0; i < n; i++) {
			res[i] = a[q[i]];
		}
		return res;
	}

	boolean same(int[] a, int[] b) {
		for (int i = 0; i < a.length; i++)
			if (a[i] != b[i])
				return false;
		return true;
	}
}
