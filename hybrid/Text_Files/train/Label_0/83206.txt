import math
(n, w) = map(int, input().split())
volumes = []
possible = True
teacups = [0] * n
vals = list(map(int, input().split()))
for i in range(n):
	poured = math.ceil(vals[i] / 2)
	w -= poured
	if w < 0:
		possible = False
		break
	volumes.append((vals[i], i))
	teacups[i] = poured
if not possible:
	print(-1)
else:
	svolumes = sorted(volumes, reverse=True)
	for j in svolumes:
		extra = j[0] - teacups[volumes.index(j)]
		if w - extra < 0:
			teacups[j[1]] += w
			break
		w -= extra
		teacups[j[1]] += extra
	print(' '.join((str(t) for t in teacups)))
