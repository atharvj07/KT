def countDer(n):
	der = [0 for i in range(max(3, n + 1))]
	der[0] = 1
	der[1] = 0
	der[2] = 1
	for i in range(3, n + 1):
		der[i] = (i - 1) * (der[i - 1] + der[i - 2])
	return der[n]

def nCr(n, r):
	return fact(n) / (fact(r) * fact(n - r))

def fact(n):
	res = 1
	for i in range(2, n + 1):
		res = res * i
	return res
(n, k) = map(int, input().split())
j = n - k
s = 0
while j <= n:
	s += nCr(n, j) * countDer(n - j)
	j += 1
print(int(s))
