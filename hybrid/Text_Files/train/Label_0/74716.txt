s1 = input()
s2 = input()
d = {}
for i in range(len(s1)):
	if d.get(s1[i], -1) == -1:
		if d.get(s2[i], -1) == -1:
			d[s1[i]] = s2[i]
			d[s2[i]] = s1[i]
		else:
			print(-1)
			exit()
	elif d[s1[i]] != s2[i] or d.get(s2[i], -1) != s1[i]:
		print(-1)
		exit()
ss = set()
ans = []
for x in d:
	if d[x] != x and x not in ss:
		ans.append(x + ' ' + d[x])
		ss.add(x)
		ss.add(d[x])
print(len(ans))
for x in ans:
	print(x)
