from math import sqrt
(n, r) = map(int, input().split())
bubbles_x = list(map(int, input().split()))
bubbles_y = []
for i in range(n):
	if i == 0:
		bubbles_y.append(r)
		continue
	else:
		current_bubble = bubbles_x[i]
		check_order_bubbles = bubbles_x[0:i]
		length = r
		for (j, current_conflicting_bubble) in enumerate(check_order_bubbles):
			temp = abs(current_bubble - current_conflicting_bubble)
			if temp > r * 2:
				continue
			elif temp == r * 2:
				length = max(length, bubbles_y[j])
			else:
				length = max(length, bubbles_y[j] + sqrt((2 * r) ** 2 - temp ** 2))
		bubbles_y.append(length)
print(*bubbles_y)
