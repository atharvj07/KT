#include <bits/stdc++.h>
using namespace std;
void solve();
clock_t start;
int main() {
  int t = 1;
  cout.precision(10);
  cout << fixed;
  start = clock();
  while (t--) solve();
  return 0;
}
int n, m, p;
int a[200500], b[200500];
map<int, int> haveb;
void solve() {
  cin >> n >> m >> p;
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < m; ++i) {
    cin >> b[i];
    ++haveb[b[i]];
  }
  int N = haveb.size();
  vector<int> ans;
  for (int i = 0; i < p; ++i) {
    map<int, int> mapa;
    int matches = 0;
    int it = 0;
    for (int j = i; j < n; j += p, ++it) {
      ++mapa[a[j]];
      if (haveb.count(a[j]) && mapa[a[j]] == haveb[a[j]]) ++matches;
      if (it >= m - 1) {
        if (matches == N) ans.push_back(j - (m - 1) * p);
        int id = j - (m - 1) * p;
        if (haveb.count(a[id]) && haveb[a[id]] == mapa[a[id]]) --matches;
        --mapa[a[id]];
      }
    }
  }
  sort(ans.begin(), ans.end());
  cout << ans.size() << "\n";
  for (int i = 0; i < ans.size(); ++i) cout << ans[i] + 1 << ' ';
}
