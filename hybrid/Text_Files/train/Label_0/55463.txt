import java.util.*;


public class MyTest{


    public static class Obj implements Comparable<Obj> {
        long x;
        long y;
        public Obj(long x, long y)
        {
            this.x =x;
            this.y =y;
        }
        public int compareTo(Obj o)
        {
            if(o.x > this.x)
                return -1;
                
            if(o.x < this.x)
                return 1;
                
            if(o.x == this.x)
            {
                if(o.y > this.y)
                    return -1;
                else
                    return 1;
            }          
            return 0;
        }
    }
    
    public static boolean solve(ArrayList<Obj> list)
    {
        for(int i=list.size()-1; i>0;i--)
        {
            long y1 = list.get(i).y;
            long y2 = list.get(i-1).y;
            if(y2 > y1)
                return true;
        }
        return false;
    }
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        long x1 = sc.nextInt();
        long x2 = sc.nextInt();
        ArrayList<Obj> list = new ArrayList<Obj>();
        
        for(int i=0; i<m; i++)
        {
            long k = sc.nextInt();
            long b = sc.nextInt();
            long y1 = k*x1 + b;
            long y2 = k*x2 + b;
            
            list.add(new Obj(y1,y2));
        }
        Collections.sort(list);
            
        /*

            }

        if(solve(list))

    }
    
}
