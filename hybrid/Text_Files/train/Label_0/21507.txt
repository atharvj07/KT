#include <iostream>
using namespace std;

int a[21],n;
bool tansaku(int i,int m){
	if(m == 0)return true;
	if(i >= n)return false;
	bool res = tansaku(i + 1,m) || tansaku(i + 1, m - a[i]);
	return res;
}

int main(){
	int q,m;
	cin >> n;
	for(int i = 0;i < n;i++){
		cin >> a[i];
	}
	cin >> q;
	for(int i = 0;i < q;i++){
		cin >> m;
		if(tansaku(0,m)){
			cout << "yes" <<endl;
		}
		else cout << "no" <<endl;
	}
	return 0;
}
