#include<stdio.h>
#include<limits.h>
#include<math.h>
#include<vector>
#include<queue>
#include<algorithm>
const int MAX=1000,V=INT_MIN;
struct P{int x,y;};
double S(int x){return x*x;}
int F(P a,P b){return ceil(sqrt(S(a.x-b.x)+S(a.y-b.y)));}
int main()
{
	int T,N,X,Y,i,j,t[MAX];
	P p,s,a[MAX];
	std::vector<int>v[MAX];
	for(scanf("%d",&T);T--;)
	{
		std::queue<P>q;
		scanf("%d%d%d%d%d",&N,&X,&Y,&p.x,&p.y);
		for(i=0;i<N;++i)
		{
			t[i]=V,v[i].clear(),scanf("%d%d",&a[i].x,&a[i].y);
			if(F(p,a[i])<=10)s.x=i,s.y=X,q.push(s);
		}
		for(i=0;i<N;++i)
			for(j=i+1;j<N;++j)
			if(F(a[i],a[j])<=50)
				v[i].push_back(j),v[j].push_back(i);
		while(!q.empty())
		{
			p=q.front();q.pop();
			if(t[p.x]+X+Y>=p.y)continue;
			if(t[p.x]!=V && t[p.x]+X+Y<p.y)goto E;
			t[p.x]=p.y;
			for(i=0;i<v[p.x].size();++i)s.x=v[p.x][i],s.y=p.y+X,q.push(s);
		}
		printf("%d\n",std::max(0,*std::max_element(t,t+N)));
		continue;
E:
		puts("You're always welcome!");
	}
	return 0;
}
