def rarray():
	return [int(i) for i in input().split()]

def is_sorted(a):
	for i in range(1, len(a)):
		if a[i] <= a[i - 1]:
			return False
	return True

def g1(a):
	ans = []
	for i in range(0, len(a), 2):
		ans.append(a[i + 1])
		ans.append(a[i])
	return ans

def g2(a):
	n = len(a) // 2
	return a[n:] + a[:n]

def f1(a, f):

		else:

		if is_sorted(a):
			return i + 1
		f = not f
	return inf
