#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000 + 5;
int w[maxn], b[maxn];
int f[maxn];
int vis[maxn];
int id[maxn];
int dp[maxn][maxn];
int n, m, v;
int idd;
struct Node {
  int w, b;
  Node() : w(0), b(0) {}
  Node(int w, int b) : w(w), b(b) {}
};
vector<Node> jh[maxn];
int Find(int x) {
  int r = x;
  while (r != f[r]) r = f[r];
  while (x != r) {
    int y = f[x];
    x = f[x];
    f[y] = r;
  }
  return r;
}
void Merge(int x, int y) {
  int a = Find(x), b = Find(y);
  if (a < b)
    f[a] = b;
  else if (a > b)
    f[b] = a;
}
void init() {
  idd = 0;
  memset(vis, 0, sizeof vis);
  memset(id, -1, sizeof id);
  memset(dp, 0, sizeof dp);
  for (int i = 1; i <= n; ++i) f[i] = i;
  for (int i = 0; i < maxn; ++i) jh[i].clear();
}
void fdp();
int main(void) {
  while (~scanf("%d %d %d", &n, &m, &v)) {
    init();
    for (int i = 1; i <= n; ++i) scanf("%d", w + i);
    for (int i = 1; i <= n; ++i) scanf("%d", b + i);
    for (int i = 0; i < m; ++i) {
      int u, v;
      scanf("%d %d", &u, &v);
      int fu = Find(u), fv = Find(v);
      if (fu != fv) Merge(u, v);
    }
    for (int i = 1; i <= n; ++i) {
      int fi = Find(i);
      if (id[fi] == -1) id[fi] = idd++;
      jh[id[fi]].push_back(Node(w[i], b[i]));
    }
    for (int i = 0; i < idd; ++i) {
      if (jh[i].size() == 1) continue;
      int sw = 0, sb = 0;
      for (int j = 0; j < jh[i].size(); ++j) {
        sw += jh[i][j].w;
        sb += jh[i][j].b;
      }
      jh[i].push_back(Node(sw, sb));
    }
    fdp();
    int ans = 0;
    for (int i = 0; i <= v; ++i) ans = max(ans, dp[idd - 1][i]);
    printf("%d\n", ans);
  }
  return 0;
}
void fdp() {
  for (int j = 0; j < jh[0].size(); ++j)
    if (v >= jh[0][j].w)
      dp[0][v - jh[0][j].w] = max(dp[0][v - jh[0][j].w], jh[0][j].b);
  for (int i = 1; i < idd; ++i) {
    for (int vv = 0; vv <= v; ++vv) {
      dp[i][vv] = dp[i - 1][vv];
      for (int j = 0; j < jh[i].size(); ++j) {
        if (vv + jh[i][j].w <= v)
          dp[i][vv] = max(dp[i][vv], dp[i - 1][vv + jh[i][j].w] + jh[i][j].b);
      }
    }
  }
}
