from math import gcd
(n, m) = map(int, input().split())
a = list(map(int, input().split()))
periods = list(map(int, input().split()))
intervals = []
_min = a[0]
for (index, ai) in enumerate(a[1:], 1):
	intervals.append(a[index] - a[index - 1])
	if ai < _min:
		_min = ai
total_gcd = intervals[0]
for (index, interval) in enumerate(intervals[1:], 1):
	total_gcd = gcd(total_gcd, interval)
for (index, period) in enumerate(periods, 1):
	if total_gcd % period == 0:
		print('YES')
		print(_min, index)
		exit()
print('NO')
