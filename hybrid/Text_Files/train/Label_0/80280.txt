def dist(x1, y1, x2, y2):
	return abs(x1 - x2) + abs(y1 - y2)
n = int(input())
s = input()
k = 1
d = {'R': (1, 0), 'U': (0, 1), 'L': (-1, 0), 'D': (0, -1)}
x = 0
y = 0
predx = 0
predy = 0
preddist = 0
for c in s:
	x += d[c][0]
	y += d[c][1]
	curdist = dist(x, y, predx, predy)
	if curdist <= preddist:
		predx = x - d[c][0]
		predy = y - d[c][1]
		preddist = dist(x, y, predx, predy)
		k += 1
	else:
		preddist = curdist
print(k)
