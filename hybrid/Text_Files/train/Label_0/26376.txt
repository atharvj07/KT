#include <bits/stdc++.h>
using namespace std;
constexpr int kN = int(2E3 + 10);
constexpr double kInf = 1E13 + 10;
double l[kN], r[kN], y[kN];
pair<double, double> Range(double xl1, double xr1, double y1, double xl2,
                           double xr2, double y2) {
  if (y1 < y2) {
    swap(xl1, xl2);
    swap(xr1, xr2);
    swap(y1, y2);
  }
  return {(xl2 - xr1) / (y1 - y2), (xr2 - xl1) / (y1 - y2)};
}
vector<pair<double, double>> Bound(vector<pair<double, double>> v) {
  double now = -kInf;
  vector<pair<double, double>> ans;
  sort(v.begin(), v.end());
  for (pair<double, double> i : v) {
    if (!(i.first < now)) ans.push_back({now, i.first});
    if (now < i.second) now = i.second;
  }
  ans.push_back({now, kInf});
  return ans;
}
double Apply(double vec, int n) {
  double ll = kInf, rr = -kInf;
  for (int i = 1; i <= n; i++) {
    ll = min(ll, l[i] + vec * y[i]);
    rr = max(rr, r[i] + vec * y[i]);
  }
  return rr - ll;
}
double lowest(pair<double, double> seq, int n) {
  double lhs = seq.first, rhs = seq.second, a, b;
  for (int i = 0; i < 110; i++) {
    a = (lhs + lhs + rhs) / 3;
    b = (lhs + rhs + rhs) / 3;
    if (Apply(a, n) > Apply(b, n))
      lhs = a;
    else
      rhs = b;
  }
  return Apply(rhs, n);
}
int main() {
  int n, lpos, rpos, a, b;
  vector<pair<double, double>> v;
  double ans = kInf;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%lf%lf%lf", &l[i], &r[i], &y[i]);
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if (y[i] != y[j]) v.push_back(Range(l[i], r[i], y[i], l[j], r[j], y[j]));
  v = Bound(v);
  lpos = 0, rpos = int(v.size()) - 1;
  while (rpos - lpos > 5) {
    a = (lpos + lpos + rpos) / 3;
    b = (lpos + rpos + rpos) / 3;
    if (Apply(v[a].first, n) < Apply(v[b].first, n))
      rpos = b;
    else
      lpos = a;
  }
  for (int i = lpos; i <= rpos; i++) ans = min(ans, lowest(v[i], n));
  printf("%.20lf\n", ans);
}
