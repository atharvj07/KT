import math
t = int(input())

def f(n):
	temp = int(math.modf(math.sqrt(n))[1])
	for i in range(2, temp + 1):
		if n % i == 0:
			return i
	return n
for i in range(t):
	temp = input().split()
	n = int(temp[0])
	k = int(temp[1])
	smallestFactor = f(n)
	if n % 2 != 0 and smallestFactor % 2 != 0:
		print(n + smallestFactor + 2 * (k - 1))
	else:
		print(n + k * smallestFactor)
