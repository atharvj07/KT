class range {
private:
	struct Iter{
		int v;
		int operator*(){return v;}
		bool operator!=(Iter& itr) {return v < itr.v;}
		void operator++() {++v;}
	};
	Iter i, n;
public:
	range(int n) : i({0}), n({n}) {}
	range(int i, int n) : i({i}), n({n}) {}
	Iter& begin() {return i;}
	Iter& end() {return n;}
};

//O(V+E)
typedef vector<vector<int>> Graph;
class Bridge{
public:
	int N;Graph g;
	vector<int> ord,low;vector<bool> vis;
	stack<int> roots,st;vector<bool> S;
	void dfs(int v,int p,int &oi,int &gi){
		vis[v]=true;ord[v]=low[v]=oi++;st.push(v);S[v]=true;roots.push(v);
		for(int to:g[v]){
			if(!vis[to]) dfs(to,v,oi,gi);
			else if(to != p && S[to])
				while(ord[roots.top()] > ord[to])roots.pop();
		}
		if(v != roots.top()) return;

		bridges.push_back(make_pair(min(v,p),max(v,p)));
		while(true){
			int w=st.top();st.pop();S[w]=false;
			cmp[w]=gi;
			if(v==w)break;
		}
		gi++;roots.pop();
	}
public:
	vector<int> cmp;
	vector<pair<int,int>> bridges;
	Bridge(int N):N(N){
		g=Graph(N);ord=low=cmp=vector<int>(N);vis=S=vector<bool>(N);
	}
	void add_edge(int f,int t){
		g[f].push_back(t);g[t].push_back(f);
	}
	void bridge(){
		int oi=0,gi=0;REP(v,N)if(!vis[v])dfs(v,-1,oi,gi);
	}
	bool same(int i,int j){return cmp[i]==cmp[j];}
};

class Main{
	public:

	int n,m;
	Graph g;
	vector<int> ch;
	int dfs(int v,int p){
		int res = 0;
		for(int to:g[v])if(to!=p)res += dfs(to,v);
		ch[v] = wmap[v];
		for(int to:g[v])if(to!=p)ch[v] += ch[to];
		res += wmap[v] * ch[v];
		if(p!=-1)return res;

		// v is root
		// Y -> r -> X
		vector<bool> exval(n+1);exval[0]=true;
		for(int to:g[v])if(to!=p){
			REPr(val,n+1)if(IN(0,val-ch[to],n+1))
				exval[val] = exval[val] | exval[val-ch[to]];
		}
		// max(|Y| * |X|)
		int sum = n - wmap[v];
		int Mv = 0;REP(val,n+1)if(exval[val] && val<=sum)Mv=max(Mv,val*(sum-val));
		return res + Mv;
	}

	map<int,int> wmap;
	void run(){
		nextInt(n);nextInt(m);
		map<string,int> stoi;
		REP(i,n){string key;nextString(key);stoi[key]=i;}
		Bridge bg(n);
		REP(i,m){
			string f,t;nextString(f);nextString(t);
			bg.add_edge(stoi[f],stoi[t]);
		}
		bg.bridge();

		//make tree
		REP(v,n)wmap[bg.cmp[v]]++;
		g = Graph(n);
		REP(v,n)for(int to:bg.g[v])if(bg.cmp[v]!=bg.cmp[to]){
			g[bg.cmp[v]].push_back(bg.cmp[to]);
		}
		REP(v,n){
			sort(ALL(g[v]));g[v].erase(unique(ALL(g[v])),g[v].end());
		}

		ch=vector<int>(n);
		// ~W ...reverse graph of W
		// 以下のように向きを揃えた方がよい．
		// count( W <- x -> V ) <= count( ~W -> x -> V )
		// （次数3以上も同様．）
		// count( W,W' <- x -> V ) <= count( ~W,~W' -> x -> V )
		// ∴ rootをfixして、各 child_i について root と child_i の向きを child_i 以降の子に伝搬させて考えてよい．
		int Mv=0;REP(v,n)Mv=max(Mv,dfs(bg.cmp[v],-1));
		cout << Mv <<endl;
	}
};
