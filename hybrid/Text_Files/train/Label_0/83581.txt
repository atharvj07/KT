(n, m) = (int(x) for x in input().split())
s = input()
f = dict(zip('abcdefghijklmnopqrstuvxwyz', 'abcdefghijklmnopqrstuvxwyz'))
t = dict(zip('abcdefghijklmnopqrstuvxwyz', 'abcdefghijklmnopqrstuvxwyz'))
for i in range(m):
	(a, b) = input().split()
	A = t[a]
	B = t[b]
	f[A] = b
	f[B] = a
	t[a] = B
	t[b] = A
for c in s:
	print(f[c], end='')
print()
