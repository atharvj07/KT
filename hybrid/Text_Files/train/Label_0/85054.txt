import io, os
import bisect
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def main(t):
	(m, n, q) = map(int, input().split())
	arrm = list(map(int, input().split()))
	arrn = list(map(int, input().split()))
	arrm.sort()
	arrn.sort()
	arrm += [float('inf')]
	arrn += [float('inf')]
	query = list(map(int, input().split()))
	temp = []
	(i, j) = (0, 0)
	while i < m or j < n:
		if arrm[i] <= arrn[j]:
			temp.append([arrm[i], 0])
			i += 1
		else:
			temp.append([arrn[j], 1])
			j += 1
	accu = [0] * (m + n + 1)
	zeros = [0] * (m + n + 1)
	edges = []
	indexes = [i for i in range(m + n - 1)]
	for i in range(m + n):
		accu[i] = accu[i - 1] + temp[i][0]
		zeros[i] = zeros[i - 1] + (1 - temp[i][1])
		if i > 0:
			edges.append(temp[i][0] - temp[i - 1][0])
	indexes.sort(key=lambda i: edges[i])
	edges.sort()
	ks = [0]
	increase = [0]
	group = [i for i in range(m + n)]
	groupnum = [1 for i in range(m + n)]

	def unionfind(root):
		son = root
		while root != group[root]:
			root = group[root]
		while son != root:
			temp = group[son]
			group[son] = root
			son = group[temp]
		return root

	def connect(index):
		i = indexes[index]
		j = i + 1
		rooti = unionfind(i)
		rootj = unionfind(j)
		ginum = groupnum[rooti]
		gjnum = groupnum[rootj]
		zeroi = zeros[rooti + ginum - 1] - zeros[rooti - 1]
		zeroj = zeros[rootj + gjnum - 1] - zeros[rootj - 1]
		prei = accu[rooti + ginum - 1] - accu[rooti + ginum - 1 - zeroi]
		prej = accu[rootj + gjnum - 1] - accu[rootj + gjnum - 1 - zeroj]
		group[rootj] = group[rooti]
		groupnum[rooti] += groupnum[rootj]
		now = accu[rooti + ginum + gjnum - 1] - accu[rooti + ginum + gjnum - 1 - zeroi - zeroj]
		return now - prei - prej
	index = 0
	currinc = 0
	while index < len(edges):
		k = edges[index]
		while index < len(edges) and edges[index] == k:
			currinc += connect(index)
			index += 1
		ks.append(k)
		increase.append(currinc)
	base = sum(arrm[:-1])
	for q in query:
		loc = bisect.bisect(ks, q) - 1
		ans = base + increase[loc]
		os.sys.stdout.write(str(ans) + '\n')
T = 1
t = 1
while t <= T:
	main(t)
	t += 1
