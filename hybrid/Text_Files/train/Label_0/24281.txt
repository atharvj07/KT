#include <bits/stdc++.h>
using namespace std;
int i, j, k, n, m, x[100003], h[100003], hh[100003], ti, iv, in, f[100003],
    p[100003], tv, hhh[100003], s;
char c[100003];
int main() {
  cin >> n >> m;
  for (i = 1; i <= m; i++) {
    cin >> c[i] >> x[i];
    if (c[i] == '-' && !h[x[i]]) ti++, iv++;
    if (c[i] == '+') ti++, hh[x[i]]++, h[x[i]] = 1;
  }
  k = iv;
  for (i = 1; i <= m; i++) {
    if (x[i] == x[i - 1] && c[i] == '+' && c[i - 1] == '-' && !iv) {
      iv++;
      p[x[i]] = 0;
    } else if (c[i] == '+') {
      if (iv || k) {
        p[x[i]] = 1;
      }
      iv++;
      k++;
      hhh[x[i]]++;
    } else if (c[i] == '-') {
      iv--;
      if (iv || (ti - k) - (hh[x[i]] - hhh[x[i]])) {
        p[x[i]] = 1;
      }
    }
  }
  for (i = 1; i <= n; i++)
    if (!p[i]) s++;
  printf("%d\n", s);
  for (i = 1; i <= n; i++)
    if (!p[i]) printf("%d ", i);
}
