#include<stdio.h>
char dp[5][1<<15][182];
int main(){
	dp[0][0][0]=1;
	for(int i=0;i<(1<<15)-1;i++){
		for(int j=0;j<182;j++){
			for(int k=0;k<4;k++){
				//printf("%d %d %d %d\n",i,j,k,dp[k][i][j]);
				if(dp[k][i][j]){
					for(int l=j;;l++){
						if(i+l*l<(1<<15))dp[k+1][i+l*l][l]+=dp[k][i][j];
						else break;
					}
				}
			}
		}
	}
	int a;
	while(scanf("%d",&a),a){
		int ret=0;
	//	for(int i=0;i<5;i++){
			for(int j=0;j<182;j++)
				ret+=(int)dp[4][a][j];
		//}
		printf("%d\n",ret);
	}
}
