def palimdrom(text):
	for i in range(len(text)):
		if text[i] not in 'AHIMOoTUVvWwXxY':
			try:
				rules = {'b': 'd', 'd': 'b', 'p': 'q', 'q': 'p'}
				if rules[text[i]] != text[-i - 1]:
					return 'NIE'
			except Exception:
				return 'NIE'
		elif text[i] != text[-i - 1]:
			return 'NIE'
	return 'TAK'
print(palimdrom(input()))
