#include <bits/stdc++.h>
using namespace std;
const double kPi = acos(-1.0);
struct point {
  double x, y;
} arr[500];
double get_cos(double degree) { return cos(kPi * degree / 180.0); }
double get_sin(double degree) { return sin(kPi * degree / 180.0); }
double x[6], y[6];
int main() {
  int n;
  cin >> n;
  double R = sqrt(100.0 / (2.0 - 2.0 * get_cos(72)));
  for (int i = 0; i < 5; ++i) {
    x[i] = R * get_cos(72 * i);
    y[i] = R * get_sin(72 * i);
  }
  for (int i = 1; i < 5; ++i) {
    x[i] -= x[0];
    y[i] -= y[0];
  }
  x[0] = 0;
  y[0] = 0;
  double dx = x[2], dy = y[2];
  for (int i = 0; i < 5; ++i) {
    arr[i].x = x[i];
    arr[i].y = y[i];
  }
  for (int i = 1; i < n; ++i) {
    for (int j = 1; j <= 4; ++j) {
      arr[4 * i + j].x = arr[4 * (i - 1) + j].x + dx;
      arr[4 * i + j].y = arr[4 * (i - 1) + j].y + dy;
    }
  }
  printf("%d\n", 4 * (n - 1) + 5);
  for (int i = 0; i <= 4 * (n - 1) + 4; ++i) {
    printf("%.10f %.10f\n", arr[i].x, arr[i].y);
  }
  for (int i = 0; i < n; ++i) {
    if (!i) {
      printf("1 ");
    } else {
      printf("%d ", 4 * (i - 1) + 3);
    }
    printf("%d %d %d %d\n", 4 * i + 2, 4 * i + 3, 4 * i + 4, 4 * i + 5);
  }
  printf("1 ");
  for (int i = 1; i < n; ++i) {
    printf("%d ", 4 * (i - 1) + 3);
  }
  for (int i = n - 1; i > 0; --i) {
    printf("%d %d %d %d ", 4 * i + 3, 4 * i + 5, 4 * i + 2, 4 * i + 4);
  }
  printf("3 5 2 4 1\n");
  return 0;
}
