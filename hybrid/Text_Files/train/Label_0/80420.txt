import sys
input = sys.stdin.readline
(N, M) = map(int, input().split())
(X, Y) = (0, 0)
D = dict()
for t in range(M):
	S = input().split()
	if S[0] == '?':
		K = int(S[1])
		if K & 1:
			if Y:
				print('YES')
			else:
				print('NO')
		elif X:
			print('YES')
		else:
			print('NO')
	elif S[0] == '+':
		(u, v, c) = (int(S[1]) - 1, int(S[2]) - 1, S[3][0])
		if D.get((v, u), '0') >= 'a':
			Y += 1
		if D.get((v, u), '0') == c:
			X += 1
		D[u, v] = c
	else:
		(u, v) = (int(S[1]) - 1, int(S[2]) - 1)
		if D.get((v, u), '0') >= 'a':
			Y -= 1
		if D.get((v, u), '0') == D[u, v]:
			X -= 1
		D[u, v] = '0'
