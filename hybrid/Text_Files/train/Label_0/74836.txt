import sys
input = lambda : sys.stdin.readline().rstrip()

def calc(A):
	N = len(A)
	if N == 1:
		return 0
	X = [0] * N
	for a in A:
		X[a] += 1
	if max(X) > (N + 1) // 2:
		return -1
	Y = [0] * N
	Y[A[0]] += 1
	Y[A[-1]] += 1
	for (a, b) in zip(A, A[1:]):
		if a == b:
			Y[a] += 2
	(su, ma) = (sum(Y), max(Y))
	cc = su - ma
	return su // 2 - 1 + max(ma - cc - 2, 0) // 2
T = int(input())
for _ in range(T):
	N = int(input())
	A = [int(a) - 1 for a in input().split()]
	print(calc(A))
