def a_to_b(a, b):
	n = len(a)
	if a == b:
		return 'YES'
	for i in range(n):
		if a[i] != b[i]:
			q = i
			dq = b[i] - a[i]
			break
	for i in range(n - 1, -1, -1):
		if a[i] != b[i]:
			w = i
			dw = b[i] - a[i]
			break
	if dq != dw:
		return 'NO'
	if dq <= 0:
		return 'NO'
	for i in range(q, w + 1):
		a[i] = a[i] + dq
	if a == b:
		return 'YES'
	else:
		return 'NO'
t = int(input())
for i in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	print(a_to_b(a, b))
