#include <iostream>
using namespace std;
int N, A, B;
int Fact[20000005], Inv[20000005];
const int MOD = 998244353;
int powerLog(int n, int p){
    int sol = 1;
    while(p){
        if(p & 1)
            sol = (1LL * sol * n) % MOD;
        p /= 2;
        n = (1LL * n * n) % MOD;
    }
    return sol;
}
void precalcFact(){
    Fact[0] = 1;
    Inv[0] = 1;
    for(int i = 1; i <= 20000000; i++)
        Fact[i] = (1LL * Fact[i - 1] * i) % MOD;
    Inv[20000000] = powerLog(Fact[20000000], MOD - 2);
    for(int i = 20000000 - 1; i >= 0; i--)
        Inv[i] = (1LL * Inv[i + 1] * (i + 1)) % MOD;
}

int Combinations(int n, int k){
    return (((1LL * Fact[n] * Inv[k]) % MOD) * Inv[n - k]) % MOD;
}
int CatalanTriangle(int n, int k){
    if(n < k)
        return 0;
    return (((((1LL * Fact[n + k] * (n - k + 1)) % MOD) * Inv[k]) % MOD) * Inv[n + 1]) % MOD;
}

inline void Add(int& x, int y){
    x += y;
    if(x >= MOD)
        x -= MOD;
}
void Solve(){
    int ans = 0;
    if(B == 0){
        cout << "1\n";
        return;
    }
    for(int k = 0; k <= N - B - 1 && A - k >= 0; k++){
        int set0Ways = Combinations(N - B - k - 1 + A - k, A - k);
        Add(ans, (1LL * set0Ways * CatalanTriangle(B - 1, k)) % MOD);
    }
    if(B - (N - B) == B - A)
    Add(ans, CatalanTriangle(B - 1, N - B));
    cout << ans << '\n';
}
int main()
{
    cin >> N >> A >> B;
    precalcFact();
    Solve();
    return 0;
}
