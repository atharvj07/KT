for _ in range(int(input())):
	s = input()
	if s[0] == s[-1]:
		print('No')
		continue
	start = s[0]
	end = s[-1]
	ls = ['A', 'B', 'C']
	ls.remove(start)
	ls.remove(end)
	mid = ls[0]
	scopy = s
	s = s.replace(start, '(')
	s = s.replace(end, ')')
	s = s.replace(mid, '(')
	k = 0
	correct = True
	for symbol in s:
		if symbol == '(':
			k += 1
		else:
			k -= 1
		if k < 0:
			correct = False
			break
	if correct and k == 0:
		print('Yes')
		continue
	s = scopy
	s = s.replace(start, '(')
	s = s.replace(end, ')')
	s = s.replace(mid, ')')
	correct = True
	k = 0
	for symbol in s:
		if symbol == '(':
			k += 1
		else:
			k -= 1
		if k < 0:
			correct = False
			break
	if correct and k == 0:
		print('Yes')
		continue
	if not correct or k != 0:
		print('No')
