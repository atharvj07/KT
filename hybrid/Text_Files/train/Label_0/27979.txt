#include <bits/stdc++.h>
using namespace std;
long long get() {
  char c = getchar();
  long long x = 0LL;
  while (c < '0' || c > '9') c = getchar();
  while (c >= '0' && c <= '9') {
    x *= 10LL;
    x += (c - '0');
    c = getchar();
  }
  return x;
}
string i_t_s(int x) {
  string t = "";
  while (x) {
    t += x % 10 + '0';
    x /= 10;
  }
  reverse(t.begin(), t.end());
  return t;
}
int s_t_i(string t) {
  int x = 0;
  for (int i = 0; i < (int)(t.size()); i++) {
    x = x * 10 + (t[i] - '0');
  }
  return x;
}
long long q_p(long long x, long long y) {
  long long res = 1;
  x %= 1000000007;
  while (y) {
    if (y % 2) {
      res = res * x;
      res %= 1000000007;
    }
    y /= 2;
    x = x * x;
    x %= 1000000007;
  }
  return res;
}
int n, m;
string ans[10];
int mx = 0;
string now[10];
int s;
int dx[4][5] = {{0, 0, 0, 1, 2},
                {0, 1, 1, 1, 2},
                {0, 1, 2, 2, 2},
                {0, 1, 1, 1, 2}},
    dy[4][5] = {
        {0, 1, 2, 1, 1}, {2, 0, 1, 2, 2}, {1, 1, 0, 1, 2}, {0, 0, 1, 2, 0}};
bool ok(int x, int y) { return (x >= 0 && x < n && y >= 0 && y < m); }
void dfs(int x, int y, int na) {
  if (x == n) {
    if (na > mx) {
      for (int i = 0; i < 10; i++) ans[i] = now[i];
      mx = na;
    }
    return;
  }
  if (s / 5 + na <= mx) return;
  for (int i = 0; i < 4; i++) {
    bool f = true;
    for (int j = 0; j < 5; j++)
      if (!ok(x + dx[i][j], y + dy[i][j]) ||
          now[x + dx[i][j]][y + dy[i][j]] != '.')
        f = false;
    if (!f) continue;
    if (i < 2 && now[x][y] == '.') s--;
    for (int j = 0; j < 5; j++)
      now[x + dx[i][j]][y + dy[i][j]] = char('A' + na);
    s -= 5;
    if (y == m - 1) {
      dfs(x + 1, 0, na + 1);
    } else
      dfs(x, y + 1, na + 1);
    for (int j = 0; j < 5; j++) now[x + dx[i][j]][y + dy[i][j]] = '.';
    s += 5;
    if (i < 2 && now[x][y] == '.') s++;
  }
  if (now[x][y] == '.') s--;
  if (y == m - 1) {
    dfs(x + 1, 0, na);
  } else
    dfs(x, y + 1, na);
  if (now[x][y] == '.') s++;
}
int main() {
  ios::sync_with_stdio(0);
  cin >> n >> m;
  s = n * m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) ans[i] += '.', now[i] += '.';
  dfs(0, 0, 0);
  cout << mx << "\n";
  for (int i = 0; i < n; i++) cout << ans[i] << "\n";
  return 0;
}
