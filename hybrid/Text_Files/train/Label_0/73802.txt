import sys, io, os
try:
	Z = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
except:
	Z = lambda : sys.stdin.readline().encode()

def sol(n, d, F):
	L = [[0] * (n + 2), [0] * (n + 2)]
	L[0][-1] = L[1][0] = 2 * n + 1
	q = [1]
	nq = []
	m = g = f = t = 0
	l = 1
	h = n + 1
	u = [0] * (2 * n + 1)
	u[1] = 1
	p = 1
	while l != h:
		if not q:
			t += min(f, g - f)
			f = g = m = 0
			while u[p]:
				p += 1
			q = [p]
			u[p] = 1
		else:
			while q:
				v = q.pop()
				g += 1
				f += F[v]
				u[d[v]] = 1
				if m:
					if L[0][h - 1] != 0:
						return -1
					L[0][h - 1] = v
					L[1][h - 1] = d[v]
					if v > L[0][h] or d[v] < L[1][h]:
						return -1
					for i in range(L[1][h] + 1, d[v]):
						if not u[i]:
							nq.append(i)
							u[i] = 1
					h -= 1
				else:
					if L[0][l] != 0:
						return -1
					L[0][l] = v
					L[1][l] = d[v]
					if v < L[0][l - 1] or d[v] > L[1][l - 1]:
						return -1
					for i in range(L[1][l - 1] - 1, d[v], -1):
						if not u[i]:
							nq.append(i)
							u[i] = 1
					l += 1
			m = 1 - m
			(q, nq) = (nq[::-1], [])
	for i in range(1, n + 2):
		if L[0][i] < L[0][i - 1] or L[1][i] > L[1][i - 1]:
			return -1
	t += min(f, g - f)
	return t
n = int(Z())
d = [0] * (2 * n + 1)
F = [0] * (2 * n + 1)
for _ in range(n):
	(a, b) = map(int, Z().split())
	d[a] = b
	d[b] = a
	F[a] = 1
print(sol(n, d, F))
