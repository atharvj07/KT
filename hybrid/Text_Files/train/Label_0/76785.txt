import sys

def solve():
	inp = sys.stdin.readline
	n = int(inp())
	a = list(map(int, inp().split()))
	MOD = 998244353
	dp = [[1, 0]]
	for i in a:
		if i - 1 >= len(dp):
			continue
		if i > 0:
			dp[i - 1][1] = (dp[i - 1][0] + 2 * dp[i - 1][1]) % MOD
		while i + 2 >= len(dp):
			dp.append([0, 0])
		dp[i + 1][0] = (2 * dp[i + 1][0] + dp[i][0]) % MOD
		dp[i + 1][1] = 2 * dp[i + 1][1] % MOD
	r = 0
	for (u, v) in dp:
		r = (r + u + v) % MOD
	print((r - 1) % MOD)

def main():
	for i in range(int(sys.stdin.readline())):
		solve()
main()
