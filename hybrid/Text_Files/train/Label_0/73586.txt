l = [[0] * 8 for i in range(8)]
steps_w = [8] * 8
steps_b = [8] * 8
for i in range(8):
	l[i] = list(input())
for i in range(8):
	w = []
	b = []
	for j in range(8):
		if l[j][i] == 'W':
			w.append(j)
		if l[j][i] == 'B':
			b.append(j)
	if len(w) == 0 and len(b) == 0:
		continue
	if len(w) == 0:
		steps_b[i] = 7 - max(b)
		continue
	if len(b) == 0:
		steps_w[i] = min(w)
		continue
	if max(w) < max(b):
		steps_b[i] = 7 - max(b)
	if min(w) < min(b):
		steps_w[i] = min(w)
print('A' if min(steps_w) <= min(steps_b) else 'B')
