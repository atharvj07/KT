void solve()
{
   int n;
   scn(n);

   vi x(n);
   set<int> s;

   rep(i, 0, n)
   scn(x[i]);

   rep(i, 0, n) {
      rep(j, i + 1, n)
         s.insert(abs(x[i] - x[j]));
   }

   prin((int) s.size());
}

int main()
{
   int t;
   scn(t);

   while(t--) {
      solve();
   }
   
   return 0;
}
