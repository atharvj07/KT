#include <bits/stdc++.h>
using namespace std;
int w[5010], dist[5010], p[5010], ans[5010], n, u, v, cost, k = 1;
vector<pair<int, int> > a[5010];
priority_queue<pair<int, int> > t[5010];
pair<int, int> d[5010];
bool used[5010];
void ts(int u) {
  used[u] = true;
  for (int i = 0; i < a[u].size(); i++)
    if (!used[a[u][i].first]) ts(a[u][i].first);
  d[u - 1].first = k++;
}
bool cmp(pair<int, int> a, pair<int, int> b) { return a.first > b.first; }
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> w[i];
  for (int i = 0; i < n - 1; i++) {
    cin >> u >> v >> cost;
    a[u].push_back(make_pair(v, cost));
    a[v].push_back(make_pair(u, cost));
  }
  ts(1);
  for (int i = 0; i < n; i++) d[i].second = i + 1;
  sort(d, d + n, cmp);
  for (int i = 0; i < n; i++) dist[d[i].second] = d[i].first;
  for (int i = 1; i <= n; i++) {
    t[i].push(make_pair(-w[i], i));
    p[i] = i;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 1; j <= n; j++) used[j] = false;
    for (int j = 0; j < n; j++) {
      int v = d[j].second;
      for (int k = 0; k < a[v].size(); k++) {
        int to = a[v][k].first, cost = a[v][k].second;
        for (int q = 0; !t[v].empty() && dist[to] > dist[v] && q < cost; q++) {
          pair<int, int> tmp = t[v].top();
          t[v].pop();
          ans[tmp.second]++;
          used[tmp.second] = true;
          p[tmp.second] = to;
          t[to].push(tmp);
        }
      }
    }
    for (int j = 1; j <= n; j++)
      if (p[j] != 1 && !used[j]) ans[j]++;
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  return 0;
}
