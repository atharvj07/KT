#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  long long int n, k;
  long long int i;
  cin >> n >> k;
  long long int N[100005];
  for (i = 1; i <= k; i++) {
    cin >> N[i];
  }
  long long int mini = 1000000000000000000;
  long long int ans = 0;
  long long int ans1 = 0;
  for (i = 1; i <= k; i++) {
    long long int res = n % N[i];
    if (res < mini) {
      mini = res;
      ans = i;
      ans1 = n / N[i];
    }
  }
  cout << ans << " " << ans1;
}
