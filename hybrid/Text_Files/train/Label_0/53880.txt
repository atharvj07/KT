
import java.util.Scanner;

public class E {
    static long mod;

    public static long[][] modPow(long[][] x, long pow) {
        if (pow == 0)
            return new long[][] { { 1, 0 }, { 0, 1 } };
        long[][] y = modPow(x, pow >> 1);
        y = mulMod(y, y);
        if ((pow & 1) != 0)
            y = mulMod(y, x);
        return y;
    }

    public static long[][] mulMod(long[][] x, long[][] y) {
        long[][] z = new long[2][2];
        for (int i = 0; i < 2; i++)
            for (int j = 0; j < 2; j++)
                for (int k = 0; k < 2; k++)
                    z[i][j] = (z[i][j] + (x[i][k] * y[k][j]) % mod) % mod;
        return z;
    }

    public static boolean can(long q, long l, long r, long k) {
        return (r / q) - (l - 1) / q >= k;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        mod = in.nextLong();
        long l = in.nextLong();
        long r = in.nextLong();
        long k = in.nextLong();
        long max = 0;
        for (long i = 1; i * i <= r; i++) {
            if (can(i, l, r, k))
                max = Math.max(i, max);
            if (can(r / i, l, r, k))
                max = Math.max(r / i, max);
        }
        long[][] A = new long[][] { { 1, 1 }, { 1, 0 } };
        System.out.println(modPow(A, max - 1)[0][0] % mod);
    }
}
