public class D {
	public static void main(String[] args) throws Exception {
		// int t = sc.nextInt();
		// while (t-- > 0)
		solve();
	}

	// **SOLUTION**
	public static void solve() throws Exception {
		int n = sc.nextInt();
		int k = sc.nextInt();
		int[] arr = new int[n];
		int[] freq = new int[200005];// 200005

		for (int i = 0; i < arr.length; i++) {
			arr[i] = sc.nextInt();
			freq[arr[i]]++;
		}

		// Binary Search
		int min = 1, max = 1000001, f = 1;// 1000001
		while (min <= max) {
			freq = new int[200005];//// 200005
			refresh(arr, freq);
			int mid = (min + max) / 2;
			
			if (min == max) {
				if (check(mid, freq, k))
					f = Math.max(mid, f);
				break;
			} else {
				if (check(mid, freq, k)) {
					min = mid+1;
					f = Math.max(mid, f);
				} else 
					max = mid - 1;
			}
		}

		freq = new int[200005];//// 200005
		refresh(arr, freq);
		ArrayList<Integer> al = getAl(f, freq, k);
		for (int i = 0; i < al.size() && i < k; i++)
			System.out.print(al.get(i) + " ");
	}

	// 3 updates
	private static void refresh(int[] arr, int[] freq) {
		for (int i = 0; i < arr.length; i++)
			freq[arr[i]]++;
	}

	private static boolean check(int mid, int[] freq, int k) {
		int c = 0;
		for (int i = 0; i < freq.length;) {
			if (freq[i] >= mid) {
				c++;
				freq[i] -= mid;
			} else {
				i++;
			}
		}
		if (c >= k)
			return true;
		else
			return false;
	}

	private static ArrayList<Integer> getAl(int f, int[] freq, int k) {
		ArrayList<Integer> al = new ArrayList<>();
		int c = 0;
		for (int i = 0; i < freq.length;) {
			if (freq[i] >= f) {
				c++;
				freq[i] -= f;
				al.add(i);
			} else {
				i++;
			}
		}
		return al;
	}

	public static InputStreamReader r = new InputStreamReader(System.in);

	public static BufferedReader br = new BufferedReader(r);

	public static Scanner sc = new Scanner(System.in);
}
