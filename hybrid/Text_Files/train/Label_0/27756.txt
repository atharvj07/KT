#include<cstdio>
#include<cstring>
#include<algorithm>
using namespace std;
char ch;
int t,n,k,h[110],pos;
void read(int &x)
{
	x=0;
	ch=getchar();
	while(ch<'0'||ch>'9') ch=getchar();
	while(ch>='0'&&ch<='9')
	{	
		x=x*10+ch-'0';
		ch=getchar();
	}
}
int main()
{
	read(t);
	while(t--)
	{
		read(n);read(k);
		for (int i=1;i<=n;i++) read(h[i]);
		while(k--)
		{
			pos=0;
			for (int j=1;j<=n;j++)
				if (j==n) pos=-1;
				else if (h[j]<h[j+1]) {h[j]++;pos=j;break;}
			if (pos==-1) break;
		} 
		printf("%d\n",pos);
	}
	return 0;
}
