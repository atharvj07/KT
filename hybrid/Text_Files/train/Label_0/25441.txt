#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, j, k, sum = 0;
  cin >> n;
  pair<int, int> a[n];
  for (i = 0; i < n; i++) {
    cin >> a[i].first;
    a[i].second = i;
  }
  sort(a, a + n);
  for (i = 0; i < n; i++) {
    sum = sum + ((a[n - i - 1].first * i) + 1);
  }
  cout << sum << '\n';
  for (i = 0; i < n; i++) cout << a[n - i - 1].second + 1 << ' ';
}
