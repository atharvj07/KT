import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author AlexFetisov
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskE_cf74 solver = new TaskE_cf74();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskE_cf74 {
        StringBuilder res;
        int pos;
        int[] a;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.nextInt();
            a = Utils.readIntArray(in, n);
            pos = -1;
            res = new StringBuilder("");
            while (a[n - 1] == 0) --n;
            for (int i = 0; i < n; ++i) {
                right();
                if (a[i] != 0) break;
            }
            int left = pos;
            for (int i = pos; i < n; ++i) {
                while (a[i] == 0) ++i;
                int j = i;
                while (j + 1 < n && a[j + 1] != 0) ++j;
                solve(i, j);
                if (left == -1) left = i;
                if (j != n - 1) {
                    int countEmpty = 0;
                    while (a[j + 1 + countEmpty] == 0) {
                        ++countEmpty;
                    }
                    if (countEmpty > (j - left + 1) + 3) {
                        remove1(left, j);
                        left = -1;
                    }
                } else {
                    remove1(left, n - 1);
                }
                i = j;
            }

            out.println(res.toString());
        }

        void solve(int left, int right) {
            while (left <= right && a[left] <= 1) {
                ++left;
            }
            if (left > right) return;
            while (a[left] > 1) {
                int end = left;
                while (end + 1 <= right && a[end + 1] > 1) {
                    ++end;
                }
                remove1(left, end);
            }
            solve(left + 1, right);
        }

        void right() {
            right(1);
        }

        void right(int len) {
            for (int i = 0; i < len; ++i) {
                res.append("AR");
                ++pos;
            }
        }

        void left(int len) {
            for (int i = 0; i < len; ++i) {
                res.append("L");
                --pos;
            }
        }

        void remove1(int left, int right) {
            right(right - pos);
            res.append("A");
            left(pos - left + 1);
            res.append("A");
            for (int i = left; i <= right; ++i) {
                --a[i];
            }
        }

    }

    static class InputReader {
        private BufferedReader reader;
        private StringTokenizer stt;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream));
        }

        public String nextLine() {
            try {
                return reader.readLine();
            } catch (IOException e) {
                return null;
            }
        }

        public String nextString() {
            while (stt == null || !stt.hasMoreTokens()) {
                stt = new StringTokenizer(nextLine());
            }
            return stt.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(nextString());
        }

    }

    static class Utils {
        public static int[] readIntArray(InputReader in, int n) {
            int[] a = new int[n];
            for (int i = 0; i < n; ++i) {
                a[i] = in.nextInt();
            }
            return a;
        }

    }
}
