import math
t = int(input())
for case in range(t):
	(a, b) = [int(x) for x in input().split(' ')]
	d = abs(a - b)
	ans = int(-1 + 1 * math.sqrt(1 + 8 * d)) // 2
	if ans * (ans + 1) // 2 < d:
		ans += 1
	if (ans * (ans + 1) // 2 - d) % 2 == 1:
		ans += 1
	if (ans * (ans + 1) // 2 - d) % 2 == 1:
		ans += 1
	print(ans)
