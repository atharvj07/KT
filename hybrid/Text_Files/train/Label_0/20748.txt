#include <bits/stdc++.h>
#define F first
#define S second
#define X real()
#define Y imag()
using namespace std;
typedef long long ll;
typedef long double ld;


vector<int> g1[202020];
vector<int> g2[202020];
int p2[202020][20];
int d2[202020];

int up2(int x, int k){
	for (int i=19;i>=0;i--){
		if (k&(1<<i)){
			x=p2[x][i];
		}
	}
	return x;
}

void dfs02(int x, int p){
	p2[x][0]=p;
	for (int i=1;i<20;i++){
		p2[x][i]=p2[p2[x][i-1]][i-1];
	}
	for (int nx:g2[x]){
		if (nx!=p) {
			d2[nx]=d2[x]+1;
			dfs02(nx, x);
		}
	}
}

int lca2(int a, int b){
	if (d2[a]<d2[b]) swap(a, b);
	a=up2(a, d2[a]-d2[b]);
	if (a==b) return a;
	for (int i=19;i>=0;i--){
		if (p2[a][i]!=p2[b][i]){
			a=p2[a][i];
			b=p2[b][i];
		}
	}
	return p2[a][0];
}

int dist2(int a, int b){
	if (a==0||b==0) return 1e6;
	return d2[a]+d2[b]-2*d2[lca2(a, b)];
}

int v=2;

void dfs(int x, int p, int y, int t){
	v=max(v, t-1+2*dist2(y, x));
	for (int nx:g1[x]){
		if (nx!=p){
			if (dist2(nx, y)==0) continue;
			if (dist2(nx, y)==1){
				v=max(v, t+1);
			}
			else{
				if (dist2(x, nx)>2){
					cout<<-1<<endl;
					exit(0);
				}
				int dd=dist2(y, nx);
				int ny;
				if (up2(nx, dd)==y) ny=up2(nx, dd-1);
				else ny=up2(y, 1);
				dfs(nx, x, ny, t+2);
			}
		}
	}
}

int main(){
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	int n,x,y;
	cin>>n>>x>>y;
	for (int i=0;i<n-1;i++){
		int a,b;
		cin>>a>>b;
		g1[a].push_back(b);
		g1[b].push_back(a);
	}
	for (int i=0;i<n-1;i++){
		int a,b;
		cin>>a>>b;
		g2[a].push_back(b);
		g2[b].push_back(a);
	}
	dfs02(x, 0);
	dfs(x, 0, y, 1);
	cout<<v<<endl;
}
