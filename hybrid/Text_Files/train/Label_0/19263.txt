#include <bits/stdc++.h>
using namespace std;
const int MAXD = 1000010;
vector<tuple<int, int, long long> > dn, ds;
vector<long long> z;
long long n, m, s;
int mindiv[MAXD], prm[MAXD], sum;
void sieve() {
  memset(mindiv, 0, sizeof(mindiv)), sum = 0;
  for (int i = 2; i < MAXD; ++i) {
    if (!mindiv[i]) prm[++sum] = i, mindiv[i] = i;
    for (int j = 1; j <= sum; ++j) {
      if (i * prm[j] >= MAXD) break;
      mindiv[i * prm[j]] = prm[j];
      if (i % prm[j] == 0) break;
    }
  }
}
void fact(vector<tuple<int, int, long long> > &dn, int n) {
  vector<tuple<int, int, long long> > temp = dn;
  dn.clear();
  while (n != 1) {
    int d = mindiv[n];
    if (!temp.empty() && get<0>(temp.back()) == d)
      ++get<1>(temp.back()), get<2>(temp.back()) *= d;
    else
      temp.push_back(tuple<int, int, long long>(d, 1, d));
    n /= d;
  }
  sort(temp.begin(), temp.end());
  for (int l = 0, r = 0; r < temp.size(); l = r) {
    int pd = get<0>(temp[l]), t = 0;
    long long d = 1;
    while (r < temp.size() && get<0>(temp[r]) == pd)
      t += get<1>(temp[r]), d *= get<2>(temp[r]), ++r;
    dn.push_back(tuple<int, int, long long>(pd, t, d));
  }
}
long long count2(int dep, long long val) {
  if (val > n) return 0;
  if (dep == ds.size()) return 1;
  int pd = get<0>(ds[dep]);
  long long d = get<2>(ds[dep]);
  long long cur = 1, ans = 0;
  for (; cur <= d; cur *= pd) ans += count2(dep + 1, val * cur);
  return ans;
}
long long count1(int dep, int sign, long long val) {
  if (val > m) return 0;
  if (dep == z.size()) return sign * (m / val);
  long long ans1 = count1(dep + 1, sign, val);
  long long ans2 = count1(dep + 1, -sign, val * z[dep]);
  return ans1 + ans2;
}
int main() {
  sieve();
  int T;
  for (cin >> T; T > 0; --T) {
    long long n1, n2, n3, m1, m2, m3, s1, s2, s3;
    cin >> n1 >> n2 >> n3;
    n = n1 * n2 * n3;
    cin >> m1 >> m2 >> m3;
    m = m1 * m2 * m3;
    cin >> s1 >> s2 >> s3;
    s = s1 * s2 * s3 * 2;
    dn.clear(), ds.clear(), ds.push_back(tuple<int, int, long long>(2, 1, 2));
    fact(dn, n1), fact(dn, n2), fact(dn, n3);
    fact(ds, s1), fact(ds, s2), fact(ds, s3);
    z.clear();
    for (int i = 0, j = 0; i < dn.size(); ++i) {
      while (j < ds.size() && get<0>(dn[i]) > get<0>(ds[j])) ++j;
      if (j != ds.size() && get<0>(dn[i]) == get<0>(ds[j])) {
        if (get<1>(dn[i]) > get<1>(ds[j]))
          z.push_back(get<2>(ds[j]) * get<0>(ds[j]));
      } else
        z.push_back(get<0>(dn[i]));
    }
    cout << count1(0, +1, 1) + count2(0, 1) << endl;
  }
  return 0;
}
