n = int(input())
l = list(map(int, input().split()))
l.sort()
side = []
i = n - 1
while i > 0:
	s = l[i]
	if s == l[i - 1]:
		side.append(s)
		i -= 2
	elif s == l[i - 1] + 1:
		side.append(s - 1)
		i -= 2
	else:
		i -= 1
side.sort()
i = len(side) - 1
num = 0
while i > 0:
	num += side[i] * side[i - 1]
	i -= 2
print(num)
