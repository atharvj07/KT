#include <bits/stdc++.h>
using namespace std;
template <typename T>
void chmin(T &x, const T &y) {
  if (x > y) x = y;
}
template <typename T>
void chmax(T &x, const T &y) {
  if (x < y) x = y;
}
inline int read() {
  int x = 0, f = 1;
  char ch = getchar();
  while (!isdigit(ch)) {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (isdigit(ch)) {
    x = x * 10 + ch - 48;
    ch = getchar();
  }
  return x * f;
}
const int N = 1e5 + 10;
int n, x, ans[N], a[N], len, res;
int main() {
  n = read(), x = read();
  if (n == 1) return printf("YES\n%d", x), 0;
  if (n == 2 && x == 0) return printf("NO"), 0;
  for (int i = 1; i <= n; i++) {
    ans[++len] = i;
    res ^= i;
  }
  int tmp = res ^ x;
  if (tmp != 0) {
    if (tmp == 3 && n >= 3)
      ans[3] = 0;
    else {
      ans[1] ^= 1 << 18;
      ans[2] ^= (1 << 18) ^ tmp;
    }
  }
  cout << "YES" << endl;
  tmp = 0;
  for (int i = 1; i <= n; i++) {
    cout << ans[i] << " ";
  }
}
