#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 10;
const int INF = 0x3f3f3f3f;
int cnt[N];
int main(void) {
  int n;
  string s;
  cin >> n >> s;
  int p = -1;
  for (int i = 0; i < n; ++i) {
    if (!i || s[i] != s[i - 1]) {
      cnt[++p]++;
    } else
      cnt[p]++;
  }
  long long res = (long long)n * (n - 1) / 2;
  for (int i = 0; i < p; ++i) res -= cnt[i] + cnt[i + 1] - 1;
  cout << res << endl;
  return 0;
}
