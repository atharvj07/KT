import math
from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = [int(i) for i in input().split()]
	sm = 0
	for i in range(n - 1):
		if a[i] >= k:
			a[i] = a[i] - k
			a[-1] = a[-1] + k
			break
		else:
			k = k - a[i]
			a[-1] = a[-1] + a[i]
			a[i] = 0
	print(*a)
