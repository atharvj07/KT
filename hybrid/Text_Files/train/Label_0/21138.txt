#include <bits/stdc++.h>
using namespace std;
int main(void) {
  int n;
  scanf("%d", &n);
  int arr[n];
  for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
  sort(arr, arr + n);
  int flag = 0;
  for (int i = 0; i <= n - 3; i++) {
    if (arr[i] + arr[i + 1] > arr[i + 2]) {
      flag = 1;
      break;
    }
  }
  if (flag == 1)
    printf("YES\n");
  else
    printf("NO\n");
  return 0;
}
