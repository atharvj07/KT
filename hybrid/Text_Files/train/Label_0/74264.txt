from functools import reduce
from operator import *
from math import *
from sys import *
from string import *
setrecursionlimit(10 ** 7)
dX = [-1, 1, 0, 0, -1, 1, -1, 1]
dY = [0, 0, -1, 1, 1, -1, -1, 1]
RI = lambda : list(map(int, input().split()))
RS = lambda : input().rstrip().split()
(x, y, x0, y0) = RI()
s = input()
total = 0
dirs = {'U': 0, 'D': 1, 'L': 2, 'R': 3}
visited = [[0 for i in range(y)] for j in range(x)]
for i in range(len(s)):
	if not visited[x0 - 1][y0 - 1]:
		print(1, end=' ')
		visited[x0 - 1][y0 - 1] = 1
		total += 1
	else:
		print(0, end=' ')
	x0 += dX[dirs[s[i]]]
	y0 += dY[dirs[s[i]]]
	if x0 < 1:
		x0 = 1
	elif x0 > x:
		x0 = x
	if y0 < 1:
		y0 = 1
	elif y0 > y:
		y0 = y
print(x * y - total)
