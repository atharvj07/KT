import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Queue;
import java.util.Scanner;

public class Codeforces {    
    public static void main(String[] args) {
        TaskD Solver = new TaskD();
        Solver.Solve();        
    }   
    private static class TaskD {                            
        private void Solve() {
            Scanner in = new Scanner(System.in);
            int n = in.nextInt();
            Vertex ar[] = new Vertex[n];
            for (int i = 0; i < n; i++)
                ar[i] = new Vertex();            
            int c[] = new int [n];
            Arrays.fill(c, 0);
            for (int i = 0; i < n - 1; i++) {
                int cur = in.nextInt();                
                ar[cur - 1].to.add(i + 1);                
            }   
            Queue <Node> Q = new ArrayDeque();
            int levels[] = new int [n];
            Arrays.fill(levels, 0);
            Q.add(new Node(0, 0));
            while (!Q.isEmpty()) {
                Node curV = Q.poll();
                levels[curV.level]++;
                ArrayList <Integer> edges = ar[curV.num].to;
                for (int i = 0, l = edges.size(); i < l; i++) 
                    Q.add(new Node(edges.get(i), curV.level + 1));
            }
            int odd = 0;
            for (int i = 0; i < n; i++)
                if (levels[i] % 2 == 1)
                    odd++;
            System.out.println(odd);                        
        }        
        private static class Vertex {
            private ArrayList <Integer> to;
            public Vertex() {
                to = new ArrayList<>();
            }
        }
        private static class Node {
            private int num;
            private int level;       
            public Node(int n, int lev) {
                num = n;
                level = lev;
            }
        }
    }
}
