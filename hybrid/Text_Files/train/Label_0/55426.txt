import java.util.*;
public class given_len_sum_of_dig {


		public static void main(String[] args) {
			Scanner scn = new Scanner(System.in);
			int n=scn.nextInt();
			int arr[]=new int[n];
			for(int i=0;i<n;i++){
				arr[i]=scn.nextInt();
			}
			Stack<Integer> st=new Stack<>();
			for(int i=0;i<n;i++){
				if(!st.isEmpty() && st.peek()==arr[i]%2)
				while(!st.isEmpty() && st.peek()==arr[i]%2){
					st.pop();
				}else{
					st.push(arr[i]%2);
				}
			}
			System.out.println(st.size()<=1?"YES":"NO");
		}

	}
