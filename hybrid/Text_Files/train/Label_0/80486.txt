k = int(input())
sequences = []
for _ in range(k):
	input()
	sequences.append([int(x) for x in input().split()])
sums = {}
found = False
for (si, s) in enumerate(sequences):
	if found:
		break
	sequence_sum = 0
	for value in s:
		sequence_sum += value
	for (i, value) in enumerate(s):
		sum = sequence_sum - value
		if sum in sums and sums[sum][0] != si:
			print('YES')
			print(str(sums[sum][0] + 1) + ' ' + str(sums[sum][1] + 1))
			print(str(si + 1) + ' ' + str(i + 1))
			found = True
			break
		sums[sum] = (si, i)
if not found:
	print('NO')
