import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int R = scanner.nextInt();

        double sum = 0;
        double sqrt2 = Math.sqrt(2);

        for (int i = 0; i < m; i++) {
            //distance to circle above
            sum += 2;
            //distance to the circle near
            if (i > 0) {
                sum += (2+sqrt2);
            }
            if (i + 1 < m) {
                sum += (2+sqrt2);
            }
            //distance to the circles more than 2 steps further
            if (i > 0) {
                double v = (i-1);
                sum += v*(v+1);
                sum += 2*sqrt2*v;
            }
            if (i+1 < m) {
                double v = (m-i-2);
                sum += v*(v+1);
                sum += 2*sqrt2*v;
            }
        }

        System.out.println(sum * R / m / m);
    }
}
