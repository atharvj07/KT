#include <bits/stdc++.h>
using namespace std;

int main() {
  int N, K;
  cin >> N >> K;
  vector<long> P(N);
  for (long i = 0, sum = 0, p; cin >> p; i++) P.at(i) = sum += ++p;
  long ma = 0;
  for (int i = 0; i + K < N; i++) ma = max(ma, P.at(i + K) - P.at(i));
  cout << fixed << setprecision(6) << ((ma) ? (double)ma / 2 : (double)P.at(N - 1) / 2) << "\n";
}
