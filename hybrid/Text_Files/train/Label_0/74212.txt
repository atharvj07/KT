import math

def lowestprime(a):
	for i in range(2, int(a ** 0.5) + 1):
		if not a % i:
			return i
	return a

def perfectnumber(a, i):
	while a != 1:
		if a < 1:
			return False
		a = a / i
	return True
a = int(input())
p = lowestprime(a)
if p == a:
	print(a)
elif perfectnumber(a, p):
	print(p)
else:
	print(1)
