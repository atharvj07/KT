from sys import stdin
input = stdin.buffer.readline

def func(x1, y1, x2, y2):
	x = (x2 - x1 + 1) * (y2 - y1 + 1) // 2
	y = (x2 - x1 + 1) * (y2 - y1 + 1) - x
	if (x1 + y1) % 2 == 0:
		w = max(x, y)
		b = min(x, y)
	else:
		w = min(x, y)
		b = max(x, y)
	return (w, b)
t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	(x1, y1, x2, y2) = map(int, input().split())
	(x3, y3, x4, y4) = map(int, input().split())
	tb = n * m // 2
	tw = n * m - tb
	(r1w, r1b) = func(x1, y1, x2, y2)
	(r2w, r2b) = func(x3, y3, x4, y4)
	zx1 = max(x1, x3)
	zy1 = max(y1, y3)
	zx2 = min(x2, x4)
	zy2 = min(y2, y4)
	if zx1 > zx2 or zy1 > zy2:
		(cw, cb) = (0, 0)
	else:
		(cw, cb) = func(zx1, zy1, zx2, zy2)
	rem_w = tw - r1w - r2w + cw
	rem_b = tb - r1b - r2b + cb
	w = rem_w + r1w + r1b - cw - cb
	b = rem_b + r2w + r2b
	print(w, b)
