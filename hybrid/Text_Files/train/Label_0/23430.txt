#include <bits/stdc++.h>
using namespace std;
bool flag(long double mid, vector<long double>& a, long double k) {
  long double cc = 0, dd = 0;
  for (long long int i = 0; i < a.size(); i++) {
    if (a[i] >= mid) {
      cc += (a[i] - mid) * (100 - k) / 100;
    } else {
      dd += (mid - a[i]);
    }
  }
  if (cc >= dd) {
    return true;
  } else
    return false;
}
int32_t main() {
  std::ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long double n, k;
  cin >> n >> k;
  vector<long double> a(n);
  long double cc = 0, dd = 0;
  for (long long int i = 0; i < n; i++) cin >> a[i];
  long double low = 0, high = 1000, mid;
  long double ans;
  long long int x = 1000;
  while (x) {
    mid = (low + high) / 2;
    if (flag(mid, a, k)) {
      low = mid;
      ans = mid;
    } else
      high = mid;
    x--;
  }
  cout << fixed << setprecision(10) << ans << '\n';
}
