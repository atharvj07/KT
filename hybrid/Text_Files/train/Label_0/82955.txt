import itertools

def queue_joy(g, p):
	return g[p[0]][p[1]] + g[p[1]][p[0]] + g[p[1]][p[2]] + g[p[2]][p[1]] + (g[p[2]][p[3]] + g[p[3]][p[2]]) * 2 + (g[p[3]][p[4]] + g[p[4]][p[3]]) * 2
g = list((list(map(int, input().split())) for _ in range(5)))
print(max([queue_joy(g, p) for p in list(itertools.permutations(range(5)))]))
