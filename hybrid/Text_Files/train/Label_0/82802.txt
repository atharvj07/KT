A = input()
B = input()
cnt1b = 0
for i in B:
	if i == '1':
		cnt1b += 1
a = []
for i in range(len(A)):
	if i == 0:
		if A[i] == '1':
			a.append(1)
		else:
			a.append(0)
	else:
		a.append(a[i - 1] + (1 if A[i] == '1' else 0))
ans = 0
for i in range(len(a)):
	if i + len(B) - 1 >= len(a):
		break
	cnt = 0
	if i > 0:
		cnt -= a[i - 1]
	cnt += a[i + len(B) - 1]
	ans += cnt % 2 == cnt1b % 2
print(ans)
