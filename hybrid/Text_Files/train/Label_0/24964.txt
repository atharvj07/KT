#include <bits/stdc++.h>
using namespace std;
ifstream in;
ofstream out;
const long long INF = LLONG_MAX;
const long double EPS = 1e-9;
const long double pi = 3.141592653589793238462643383279502884;
string s;
long long answer = 1;
long long n;
const long long m = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cout.precision(40);
  srand(time(0));
  cin >> s;
  n = s.size();
  vector<long long> a(n);
  vector<long long> k;
  for (long long i = 0; i < n; i++) {
    if (s[i] == 'a') {
      a[i] = 1;
      continue;
    }
    if (s[i] == 'b') {
      a[i] = 2;
      continue;
    }
  }
  long long ptr = 0;
  long long cnt = 0;
  while (ptr != n) {
    while (a[ptr] != 2 && ptr != n) {
      if (a[ptr] == 1) {
        cnt++;
      }
      ptr++;
    }
    if (cnt) k.push_back(cnt);
    cnt = 0;
    while (a[ptr] == 2 && ptr != n) {
      ptr++;
    }
  }
  if (cnt) k.push_back(cnt);
  for (long long i : k) {
    answer *= i + 1;
    answer %= m;
  }
  cout << (answer - 1 + m) % m << endl;
  return 0;
}
