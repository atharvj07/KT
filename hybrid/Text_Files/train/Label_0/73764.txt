import sys
input = sys.stdin.readline
from collections import *
n = int(input())
b = list(map(int, input().split()))
p = [0] * n
for i in range(n):
	t = b[i]
	while t % 2 == 0:
		p[i] += 1
		t //= 2
cnt = [0] * 70
for pi in p:
	cnt[pi] += 1
M = 0
max_p = -1
for i in range(70):
	if cnt[i] > M:
		M = cnt[i]
		max_p = i
ans = []
for i in range(n):
	if p[i] != max_p:
		ans.append(b[i])
print(n - M)
print(*ans)
