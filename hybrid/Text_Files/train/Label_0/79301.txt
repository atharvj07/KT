from collections import *
(n, a) = (int(input()), list(map(int, input().split())))
mem = Counter(a)
(ma, ans) = (max(mem, key=mem.get), [])
for i in range(n):
	if a[i] == ma:
		for j in range(i + 1, n):
			if a[j] != ma:
				ans.append([1, j + 1, j] if a[j] < a[j - 1] else [2, j + 1, j])
				a[j] = ma
			else:
				break
		for j in range(i - 1, -1, -1):
			if a[j] != ma:
				ans.append([1, j + 1, j + 2] if a[j] < a[j + 1] else [2, j + 1, j + 2])
				a[j] = ma
			else:
				break
print(len(ans))
for i in ans:
	print(*i)
