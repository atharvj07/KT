i = int(input())
k = 0
while k < i:
	l = int(input())
	p = input()
	if l == 1:
		print('NO')
	if l == 2:
		if p[0] == p[1] or p[0] > p[1]:
			print('NO')
		else:
			print('YES')
			print('2')
			print(p[0], p[1])
	elif l == 3 or l == 4 or l == 5:
		print('YES')
		print('2')
		print(p[0], p[1:])
	else:
		print('YES')
		print('3')
		print(p[0], p[1:3], p[3:])
	k = k + 1
