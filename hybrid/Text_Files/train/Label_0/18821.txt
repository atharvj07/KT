#include<bits/stdc++.h>
using namespace std;
typedef complex<double> P;
int H,R;
int U[3];
double D[3];
const double eps = 1e-8;
double getRtoA( double r ){ return 180.0/M_PI*r; }
double da[]={-360.0,0.0,360.0};
bool check(P p, P f, double a, double w, double d ){
  P v = p - f;
  if( abs(v) > a + eps ) return false;
  double an = getRtoA( arg(v) );
  for(int i=0;i<3;i++){
    if( w - d - eps < an + da[i] && an + da[i] < w + d + eps ) return true;
  }
  return false;
}

bool check( P p, vector<vector<P>> u, double a, double w){
  for(int i=0;i<3;i++){
    for(int j=0;j<U[i];j++){
      if( check( p, u[i][j], a, w, D[i] ) ) return false;
    }
  }
  return true;
}

int main(){
  while( cin >> H >> R && (H|R) ){
    vector<P> h;
    vector<vector<P>> u;
    vector<int> w,a;
    u.resize(3);
    for(int i=0;i<H;i++){
      int x,y; cin >> x >> y; h.push_back( P( x, y ) );
    }
    cin >> U[0] >> U[1] >> U[2] >> D[0] >> D[1] >> D[2];
    for(int i=0;i<3;i++){
      D[i]/=2.0;
      for(int j=0;j<U[i];j++){
        int x,y; cin >> x >> y;
        u[i].push_back( P(x,y) );
      }
    }

    w.resize(R);a.resize(R);
    for(int i=0;i<R;i++){
      cin >> w[i] >> a[i];
    }

    
    vector<int> res;
    int maxs = 0;
    for(int i=0;i<H;i++){
      int cnt = 0;
      for(int j=0;j<R;j++){
        if( check( h[i], P(0,0), a[j], w[j], D[0] ) ) {
          if( check( h[i], u, a[j], w[j] ) ) cnt++;
        }
      }
      if( cnt == 0 ) continue;
      if( maxs < cnt ){
        res.clear();
        res.push_back( i+1 );
        maxs = cnt;
      } else if( maxs == cnt )
        res.push_back( i+1 );

    }

    if( res.empty() ) cout << "NA";
    else {
      for(int i=0;i<(int)res.size();i++){
        if( i ) cout << " ";
        cout << res[i];
      }
    }
    cout << endl;
    
  }
}
