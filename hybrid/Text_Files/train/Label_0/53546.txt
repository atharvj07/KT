import java.io.IOException;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;


// from cyand1317's editorial

public class Ds {
	
	static int n;
	static List<List<Integer>> adj = new ArrayList<>();
	
	static Double[][][] dp;
	
	static List<Circle> list = new ArrayList<>();

	
	public static void main(String[] args) throws NumberFormatException, IOException {

		Scanner in = new Scanner(System.in);
		
		 n = in.nextInt();
		 
		 Set<Integer> set = new HashSet<>();
 
		 for(int i=0; i<n; i++) {
			 list.add(new Circle(in.nextInt(), in.nextInt(), in.nextInt()));
			 set.add(i);
			 adj.add(new ArrayList<>());
		 }
		 		 
		 for(int i=0; i<n; i++) {
			 
			 int big = 999999999;
			 int ind = -1;
			 
			 for(int j=0; j<n; j++) {
				 if(i == j) {
					 continue;
				 }
				 if(list.get(j).contains(list.get(i))) {
					 if(list.get(j).r < big) {
						 big = list.get(j).r;
						 ind = j;
					 }
				 }
				
			 }
			 
			 if(ind != -1) {
				 adj.get(ind).add(i);
				 set.remove(i);
			 }
		 }
		 
		 dp = new Double[n][2][2];
		 
		 
		 double ans = 0;
		 
		 for(int root : set) {
			 dfs(root);
			 ans += dp[root][0][0];
		 }
		 
		 System.out.println(ans);
		 
		 
	}

	public static void dfs(int i) {

		double sb[][] = new double[2][2];
		
		for(int n : adj.get(i)) {
			
			dfs(n);
						
			sb[0][1] += dp[n][0][1];
			sb[1][0] += dp[n][1][0];
			sb[1][1] += dp[n][1][1];
			sb[0][0] += dp[n][0][0];
	
		}
				 
		double ca = list.get(i).area;
		
		// pick first   pick second
		
		dp[i][0][1] = ( Math.max(ca + sb[1][1], -ca + sb[0][0]) );
		
		dp[i][1][0] = ( Math.max(-ca + sb[0][0], ca + sb[1][1]) );

		dp[i][0][0] = ( Math.max(ca + sb[1][0], ca + sb[0][1]) );

		dp[i][1][1] = ( Math.max(-ca + sb[0][1], -ca + sb[1][0]) );
		
		
	}	
}


class Circle {
	
	int x;
	int y;
	int r;
	
	double area;
	
	public Circle(int x, int y, int r) {
		super();
		
		this.x = x;
		this.y = y;
		this.r = r;
		this.area = (Math.PI * r * r);
	}
	
	public boolean contains(Circle obj) {
		
		// dist between centers
		return  r >= Math.sqrt(Math.pow(x - obj.x, 2) + Math.pow(y - obj.y, 2)) + obj.r;
			
	}
}
