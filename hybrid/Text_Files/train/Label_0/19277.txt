#include <bits/stdc++.h>
int a, b, m, r0, r1, c = 0, t[1000000], k = 2, jogap;
int main() {
  for (int g = 0; g < 100009; g++) t[g] = 0;
  scanf("%d %d %d %d", &a, &b, &m, &r0);
  t[r0] = 1;
  while (c != 1) {
    r1 = (r0 * a + b) % m;
    r0 = r1;
    if (t[r1] != 0) {
      jogap = k - t[r1];
      c = 1;
    }
    t[r1] = k;
    k++;
  }
  printf("%d", jogap);
  getchar();
  return 0;
}
