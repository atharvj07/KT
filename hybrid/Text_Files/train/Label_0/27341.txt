#include <bits/stdc++.h>
using namespace std;
FILE *fi, *fo;
const int MAXN = 100010;
const int MAXM = 100010;
int n, m;
pair<pair<int, int>, int> a[MAXM];
priority_queue<pair<int, int>, vector<pair<int, int> >,
               greater<pair<int, int> > >
    q;
vector<int> ans;
int main() {
  fi = stdin;
  fo = stdout;
  fscanf(fi, "%d %d", &n, &m);
  for (int i = 0; i < m; i++) {
    int r, c;
    fscanf(fi, "%d %d", &r, &c);
    a[i].first.first = n - c;
    a[i].first.second = r - 1;
    a[i].second = i;
  }
  sort(a, a + m);
  int p = 0;
  for (int pos = 0; pos < n; pos++) {
    while ((p < m) && (a[p].first.first <= pos)) {
      q.push(make_pair(a[p].first.second, a[p].second));
      p++;
    }
    while ((!q.empty()) && (q.top().first < pos)) q.pop();
    if (!q.empty()) {
      ans.push_back(q.top().second);
      q.pop();
    }
  }
  fprintf(fo, "%d\n", ans.size());
  for (int i = 0; i < ans.size(); i++) fprintf(fo, "%d ", ans[i] + 1);
  return 0;
}
