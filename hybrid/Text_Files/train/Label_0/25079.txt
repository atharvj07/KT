#include <bits/stdc++.h>
using namespace std;
long long a[100100];
long long ans[100100];
long long s[100100];
int main() {
  for (int _n(100100), i(0); i < _n; i++) ans[i] = -1;
  int N;
  cin >> N;
  for (int _n(N), i(0); i < _n; i++) cin >> a[i];
  sort(a, a + N);
  s[0] = a[0];
  for (int _n(N), i(1); i < _n; i++) s[i] = s[i - 1] + a[i];
  int q;
  cin >> q;
  for (int _n(q), it(0); it < _n; it++) {
    long long k;
    cin >> k;
    if (ans[k] != -1) {
      cout << ans[k] << endl;
      continue;
    }
    long long kk = 1;
    long long S = 0;
    long long i = N - 2;
    long long cnt = 1;
    for (; i >= 0; i -= kk) {
      S += s[i];
      kk *= k;
      cnt++;
    }
    ans[k] = S;
    cout << S << endl;
  }
  return 0;
}
