#include <cstdio>
#include <cmath>
using namespace std;
const int MOD=100000;
int di[100000];
int abs(int a){
    if(a<0) return -a;
    else return a;
}
int main(){
    int n,m;
    scanf("%d%d",&n,&m);
    di[0]=0;
    for(int i=0;i<n-1;i++){
        scanf("%d",&di[i+1]);
        di[i+1]+=di[i];
    }
    int now=0,ret=0,a;
    for(int i=0;i<m;i++){
        scanf("%d",&a);
        ret+=abs(di[now+a]-di[now]);
        ret%=MOD;
        now+=a;
    }
    printf("%d\n",ret);
}
