import math

def main_function():
	output_list = []
	(n, m) = [int(i) for i in input().split(' ')]
	a = [int(i) for i in input().split(' ')]
	input_list = [[int(h) for h in input().split(' ')] for i in range(m)]
	count_negative_1 = 0
	count_positive_1 = 0
	for i in a:
		if i < 0:
			count_negative_1 += 1
		else:
			count_positive_1 += 1
	for i in input_list:
		if not (i[1] - i[0]) % 2:
			ans = 0
		elif (i[1] - i[0] + 1) // 2 <= count_negative_1 and (i[1] - i[0] + 1) // 2 <= count_positive_1:
			ans = 1
		else:
			ans = 0
		output_list.append(str(ans))
	return '\n'.join(output_list)
print(main_function())
