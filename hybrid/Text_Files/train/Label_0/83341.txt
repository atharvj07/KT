t = int(input())
for _ in range(t):
	n = int(input())
	dp = [2 for i in range(n)]
	lst = [0 for i in range(2 * n)]
	count = 0
	for i in range(n):
		(u, v) = map(int, input().split())
		lst[count] = [u, -1, i]
		lst[count + 1] = [v, 1, i]
		count += 2
	lst.sort()
	total = 0
	for i in lst:
		total = total - i[1]
		if total > 0:
			dp[int(i[-1])] = 1
		else:
			break
	if dp.count(1) == n:
		print(-1)
	else:
		print(*dp)
