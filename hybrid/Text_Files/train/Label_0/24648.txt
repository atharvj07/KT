#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  if (b == 0)
    return a;
  else
    return gcd(b, a % b);
}
long long lcm(long long a, long long b) { return ((a * b) / gcd(a, b)); }
int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long x, y, n, o = 0, t = 0, th = 0, f = 0, ans = 0;
  cin >> n;
  for (long long i = 0; i < n; i++) {
    cin >> x;
    if (x == 1) o++;
    if (x == 2) t++;
    if (x == 3) th++;
    if (x == 4) f++;
  }
  ans += f;
  y = min(o, th);
  o -= y;
  th -= y;
  ans += y;
  ans += th;
  ans += (t / 2);
  t = t % 2;
  y = min((o / 2), t);
  ans += y;
  t -= y;
  o -= (2 * y);
  ans += t;
  if (t == 1 && o > 0) o--;
  ans += (o / 4);
  o -= (4 * (o / 4));
  if (o > 0) ans++;
  cout << ans << "\n";
  return 0;
}
