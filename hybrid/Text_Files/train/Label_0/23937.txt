#include<iostream>
using namespace std;

int main(){
  bool flag = false;
  int start,stop;
  cin >> start >> stop;
  int num;
  cin >> num;
  int book_start,book_stop;
  for(int i=0;i<num;i++){
    cin >> book_start >> book_stop;
    if(start>=book_start && start<book_stop){
      flag = true;
    }
    if(stop>book_start && stop<=book_stop){
      flag = true;
    }
    if(start<=book_start&&stop>=book_stop){
      flag = true;
    }
  }

  cout << flag << endl;

}
