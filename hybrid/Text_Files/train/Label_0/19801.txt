template <typename T> T tmax(T t1, T t2) { return t1 > t2 ? t1 : t2; }

const int MAX_NS = 10;
const int MAX_NL = 20;
int ns, nl, s1, s2, d;
int S[MAX_NS][MAX_NS];
char L[MAX_NS][MAX_NS];

int dfs(int k, vi& v, int dist) {
	int maxval = dist;
	vi mv;
	int next = -1;

	rep(i,ns) {
		if(S[k][i] != 0 && L[k][i] != 1) {
			L[k][i] = L[i][k] = 1;
			int val = dfs(i, v, dist+S[k][i]);
			if(maxval < val) {
				maxval = val;
				mv = v;
				mv.push_back(i);
			}
			L[k][i] = L[i][k] = 0;
		}
	}

	v = mv;

	return maxval;
}
