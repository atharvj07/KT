#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, q;
  cin >> n >> q;
  int last = 0;
  vector<int> a;
  vector<vector<int> > b(n + 2);
  set<int> s;
  for (int j = 0; j < q; j++) {
    int t, x;
    cin >> t >> x;
    if (t == 1) {
      a.push_back(j);
      b[x].push_back(j);
      s.insert(j);
    } else if (t == 2) {
      for (int i = 0; i < b[x].size(); i++) {
        s.erase(b[x][i]);
      }
      b[x].clear();
    } else if (t == 3) {
      while (last < x) {
        s.erase(a[last]);
        last++;
      }
    }
    cout << s.size() << "\n";
  }
  return 0;
}
