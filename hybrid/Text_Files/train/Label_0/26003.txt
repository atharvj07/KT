#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  long long k;
  cin >> n >> k;
  int ar[n];
  for (int i = 0; i < n; ++i) cin >> ar[i];
  sort(&ar[0], &ar[n]);
  long long tot = 0;
  long long foo = (long long)ceil((long double)k / (long double)n);
  int ct1 = 0;
  int ct2 = 0;
  for (int i = foo - 2; i >= 0; --i)
    if (ar[foo - 1] == ar[i]) ++ct1;
  for (int i = foo; i < n; ++i)
    if (ar[foo - 1] == ar[i]) ++ct2;
  long long num = (long long)ct1 * n;
  int l = foo - ct1 - 1;
  int r = foo - 1 + ct2;
  long long m = k % n;
  if (m == 0) m = n;
  num += m;
  tot = r - l + 1;
  long long fok = 0;
  int ind = 0;
  while (fok < num && ind < n) {
    fok += (tot);
    ind++;
  }
  cout << ar[foo - 1] << " " << ar[ind - 1] << endl;
}
