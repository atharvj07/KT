#include<bits/stdc++.h>
using namespace std;

int main(){
 int N;
 cin>>N;
 vector<vector<int>>G(N);
 vector<pair<int,int>>vp;
 for(int i=0;i<N-1;i++){
     int a,b;
     cin>>a>>b;
     a--,b--;
     G[a].emplace_back(b);
     G[b].emplace_back(a);
     vp.emplace_back(a,b);
 }

int K=0;

queue<int>q;
vector<bool>used(N,0);
map<pair<int,int>,int>I;
vector<int>cs(N,0);

used[0]=1;
q.push(0);

while(!q.empty()){

    int v=q.front();q.pop();
    if(K<(int)G[v].size())K=G[v].size();
    int cur=1;

    for(auto u:G[v]){
    if(used[u])continue;

    if(cur==cs[v])cur++;
    cs[u]=I[make_pair(u,v)]=I[make_pair(v,u)]=cur++;
    q.push(u);
    used[u]=1;
    }
}
cout<<K<<endl;
 for(auto p:vp)cout<<I[p]<<endl;


return 0;
}
