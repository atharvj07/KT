def test_3():
	from collections import Counter
	n = list(map(int, input().split(' ')))[0]
	data = list(input())
	a = 0
	for i in range(len(data)):
		if data[i] != '?':
			if i != n - 1 and data[i] == data[i + 1]:
				print('No')
				return 0
		elif i in (0, n - 1) or data[i + 1] == '?' or data[i - 1] == '?' or (data[i - 1] != '?' and data[i + 1] != '?' and (data[i - 1] == data[i + 1])):
			a = 1
	if a == 1:
		print('Yes')
	else:
		print('No')
test_3()
