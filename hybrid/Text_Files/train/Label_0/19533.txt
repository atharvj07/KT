class oPair {
 public:
  long a, b;
  long long Diff() {
    if (a == b && a == -1)
      return 0;
    else
      return (long long)(b - a + 1);
  }
  bool Contain(long n) { return (n >= a && n <= b); }
};
oPair Intersect(oPair x, oPair y) {
  oPair res;
  res.a = max(x.a, y.a);
  res.b = min(x.b, y.b);
  if (res.a > res.b) res.a = res.b = -1;
  return res;
}
oPair Bound(oPair x, oPair y) {
  oPair res;
  res.a = min(x.a, y.a);
  res.b = max(x.b, y.b);
  return res;
}
bool Equals(oPair x, oPair y) { return (x.a == y.a && x.b == y.b); }
oPair p1, p2;
long c, totallucky, lucky[1023];
void input(long &n) { scanf("%ld", &n); }
void input(oPair &p1, oPair &p2, long &c) {
  input(p1.a);
  input(p1.b);
  input(p2.a);
  input(p2.b);
  input(c);
}
void GetLuckyList(oPair p, long a[], long &cnt) {
  cnt = 0;
  for (long i = 1; i <= totallucky; i++) {
    if (p.Contain(lucky[i])) a[cnt++] = lucky[i];
  }
}
long long NumOfChoices(long left, long right, oPair pp) {
  oPair p;
  p.a = left;
  p.b = right;
  p = Intersect(p, pp);
  return p.Diff();
}
long long calcNumOfCases(oPair p1, oPair p2, long leftLimit, long left,
                         long right, long rightLimit) {
  long long res = 0;
  long long tmp1 = NumOfChoices(leftLimit + 1, left, p1);
  long long tmp2 = NumOfChoices(right, rightLimit - 1, p2);
  res += tmp1 * tmp2;
  tmp1 = NumOfChoices(leftLimit + 1, left, p2);
  tmp2 = NumOfChoices(right, rightLimit - 1, p1);
  res += tmp1 * tmp2;
  return res;
}
long lstNums[1023];
double solve(oPair p1, oPair p2, long c) {
  oPair pInt = Bound(p1, p2);
  oPair pLap = Intersect(p1, p2);
  long long totalcases = p1.Diff() * p2.Diff();
  long long numofcases = 0;
  long luckCount;
  GetLuckyList(pInt, lstNums, luckCount);
  sort(lstNums, lstNums + luckCount);
  for (long left = 0; left < luckCount; left++) {
    long right = left + c - 1;
    if (right >= luckCount) break;
    numofcases += calcNumOfCases(
        p1, p2, (left <= 0 ? -1 : lstNums[left - 1]), lstNums[left],
        lstNums[right],
        right >= luckCount - 1 ? 2000000000 : lstNums[right + 1]);
    if (c == 1 && pLap.Contain(lstNums[left])) numofcases--;
  }
  return (double)numofcases / totalcases;
}
void output(double res) { printf("%.12llf", res); }
int main() {
  initlucky();
  input(p1, p2, c);
  double res = solve(p1, p2, c);
  output(res);
}
