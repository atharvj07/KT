def LI():
	return list(map(int, sys.stdin.readline().split()))

def MI():
	return map(int, sys.stdin.readline().split())

def SI():
	return sys.stdin.readline().strip()

def FACT(n, mod):
	s = 1
	facts = [1]
	for i in range(1, n + 1):
		s *= i
		s %= mod
		facts.append(s)
	return facts[n]

def C(n, k, mod):
	return FACT(n, mod) * pow(FACT(k, mod) * FACT(n - k, mod) % mod, mod - 2, mod) % mod

def lcm(a, b):
	return abs(a * b) // math.gcd(a, b)
t = int(input())
for _ in range(t):
	(a, b) = map(int, input().split())
	(p, r) = map(int, input().split())
	if len(str(a)) + b > len(str(p)) + r:
		print('>')
	elif len(str(a)) + b < len(str(p)) + r:
		print('<')
	else:
		if b >= r:
			b = b - r
			r = 0
		else:
			r = r - b
			b = 0
		if a * 10 ** b > p * 10 ** r:
			print('>')
		elif a * 10 ** b < p * 10 ** r:
			print('<')
		else:
			print('=')
