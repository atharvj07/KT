(n, m) = map(int, input().split())
a = [None] * n
for i in range(n):
	row = list(map(int, input().split()))
	a[i] = row
ops = []
for i in range(n):
	for j in range(m):
		if i + 1 < n and j + 1 < m and (a[i][j] == 1 or a[i][j + 1] == 1 or a[i + 1][j] == 1 or (a[i + 1][j + 1] == 1)):
			if a[i][j] > 0 and a[i][j + 1] > 0 and (a[i + 1][j] > 0) and (a[i + 1][j + 1] > 0):
				ops[len(ops):] = [(i, j)]
				a[i][j] += 1
				a[i][j + 1] += 1
				a[i + 1][j] += 1
				a[i + 1][j + 1] += 1
possible = True
for i in range(n):
	for j in range(m):
		if a[i][j] == 1:
			possible = False
if not possible:
	print(-1)
else:
	print(len(ops))
	for op in ops:
		print(op[0] + 1, op[1] + 1)
