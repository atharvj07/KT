#include <bits/stdc++.h>
const int inf = 0x3f3f3f3f;
const long long mod = 1000000007;
using namespace std;
int dp[400086];
int main() {
  int h = 0;
  int r, g;
  cin >> r >> g;
  int f = r + g;
  while (f > 0) {
    if (h + 1 > f) break;
    h++;
    f -= h;
  }
  int k = 0;
  dp[0] = 1;
  for (int i = 1; i <= h; i++) {
    for (int j = k; j >= 0; j--) {
      dp[i + j] = (dp[j] + dp[i + j]) % mod;
      k = max(k, i + j);
    }
  }
  int ans = 0;
  r = max(r, g);
  for (int i = r; i >= r - f; i--) ans = (ans + dp[i]) % mod;
  cout << ans << endl;
}
