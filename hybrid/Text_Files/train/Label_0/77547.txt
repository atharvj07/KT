import sys
input = lambda : sys.stdin.readline().strip()
t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	res = [m % 10]
	while True:
		if (res[-1] + m) % 10 == m % 10:
			break
		else:
			res.append((res[-1] + m) % 10)
	S = sum(res) * (n // m // len(res))
	rem = n // m % len(res)
	for i in range(rem):
		S += res[i]
	print(S)
