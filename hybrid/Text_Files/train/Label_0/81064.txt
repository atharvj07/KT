def num(x):
	return 3 * x ** 2 / 2 + x / 2

def findMaxN(x, left=0, right=26000):
	while left < right:
		mid = (left + right) // 2
		a = num(mid)
		b = num(mid + 1)
		c = num(mid - 1)
		if b > x and a < x:
			return mid
		elif a > x and c < x:
			return mid - 1
		elif a == x:
			return mid
		elif a < x:
			left = mid + 1
		elif a > x:
			right = mid - 1
	return right

def solve():
	cards = int(input())
	cnt = 0
	right = 26000
	while cards >= 2:
		right = findMaxN(cards, right=right)
		(d, cards) = divmod(cards, num(right))
		cnt += d
	print(int(cnt))
nums = int(input())
for _ in range(1, nums + 1):
	solve()
