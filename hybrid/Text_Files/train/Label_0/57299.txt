import java.io.*;
import java.util.*;

public class E {
	private static boolean _READ_FROM_FILE = System.getProperty("ONLINE_JUDGE") == null;
	private static Scanner in;
	private static void core() {
		int nRows = in.nextInt(), nCols = in.nextInt();
		char[][] a = new char[nRows][];
		for (int i = 0; i < a.length; i++) {
			a[i] = in.next().toCharArray();
		}
		int res = Integer.MAX_VALUE;
		for (int i = 0; i < a.length; i++) {
			for (int j = 0; j < a[i].length; j++) {
				int[][] d = new int[nRows][nCols];
				for (int[] ks : d) {
					Arrays.fill(ks, Integer.MAX_VALUE);
				}
				d[i][j] = 0;
				LinkedList<Integer> queue = new LinkedList<Integer>();
				queue.add(i);
				queue.add(j);
				while(!queue.isEmpty()) {
					LinkedList<Integer> nextQueue = new LinkedList<Integer>();
					while(!queue.isEmpty()) {
						int row = queue.removeFirst();
						int col = queue.removeFirst();
						for (int dr = -1; dr <= 1; dr++) {
							for (int dc = -1; dc <= 1; dc++) {
								if (dr == 0 ^ dc == 0) {
									int nr = row + dr;
									int nc = col + dc;
									if (0 <= nr && nr < nRows && 0 <= nc && nc < nCols && d[nr][nc] == Integer.MAX_VALUE) {
										if (a[row][col] == a[nr][nc]) {
											d[nr][nc] = d[row][col];
											queue.add(nr);
											queue.add(nc);
										}
										else {
											d[nr][nc] = d[row][col] + 1;
											nextQueue.add(nr);
											nextQueue.add(nc);
										}
									}
								}
							}
						}
					}
					queue = nextQueue;
				}
				int farthest = Integer.MIN_VALUE;
				char leaf = '?';
				for (int k = 0; k < d.length; k++) {
					for (int k2 = 0; k2 < d[k].length; k2++) {
						if (d[k][k2] > farthest) {
							farthest = d[k][k2];
							leaf = a[k][k2];
						}
					}
				}
				int now = farthest;
				if (leaf != 'W')
					++now;
				res = Math.min(res, now);
			}
		}
		System.out.println(res);
	}
	static void debug(Object...os) {
		System.out.println(Arrays.deepToString(os));
	}
	public static void main(String[] args) throws FileNotFoundException {
		if (_READ_FROM_FILE)
			System.setIn(new FileInputStream("in.in"));
		in = new Scanner(System.in);
		core();
	}
}
