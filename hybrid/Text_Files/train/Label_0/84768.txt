n = int(input(''))
s = input('')
l = s.split(' ')
l = list(map(int, l))
neg = []
pos = []
zero = []
for i in range(n):
	if l[i] == 0:
		zero.append(l[i])
	elif l[i] > 0:
		pos.append(l[i])
	else:
		neg.append(l[i])
if len(neg) % 2 == 0:
	temp = neg.pop(len(neg) - 1)
	zero.append(temp)
if len(neg) % 2 != 0 and len(pos) == 0:
	temp = neg.pop(len(neg) - 1)
	temp1 = neg.pop(len(neg) - 2)
	pos.append(temp)
	pos.append(temp1)
print(len(neg), *neg, end='\n')
print(len(pos), *pos, end='\n')
print(len(zero), *zero, end='\n')
