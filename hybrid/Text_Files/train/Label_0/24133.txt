#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> l_l;
typedef pair<int, int> i_i;
template<class T>
inline bool chmax(T &a, T b) {
    if(a < b) {
        a = b;
        return true;
    }
    return false;
}

template<class T>
inline bool chmin(T &a, T b) {
    if(a > b) {
        a = b;
        return true;
    }
    return false;
}

const long double EPS = 1e-10;
const long long INF = 1e18;
const long double PI = acos(-1.0L);
//const ll mod = 1000000007;
typedef pair<l_l, ll> lll;
ll N;
vector<lll> v;
ll dp[366][366];

void solve() {
    for(int i = 0; i <= 365; i++) {
        for(int j = 0; j <= 365; j++) {
            dp[i][j] = -1e18;
        }
    }
    dp[0][0] = 0;
    v.resize(N);
    for(int i = 0; i < N; i++) {
        cin >> v[i].first.first >> v[i].first.second >> v[i].second;
    }
    sort(v.begin(), v.end());
    for(auto now : v) {
        for(int i = 365; i >= 0; i--) {
            for(int j = i; j >= 0; j--) {
                if(i < now.first.first) {
                    int newi = now.first.second;
                    int newj = j;
                    if(newi < newj) swap(newi, newj);
                    chmax(dp[newi][newj], dp[i][j] + now.second);
                }
                if(j < now.first.first) {
                    int newj = now.first.second;
                    int newi = i;
                    if(newi < newj) swap(newi, newj);
                    chmax(dp[newi][newj], dp[i][j] + now.second);
                }
            }
        }
    }
    ll ans = 0;
    for(int i = 0; i <= 365; i++) {
        for(int j = 0; j <= 365; j++) {
            chmax(ans, dp[i][j]);
        }
    }
    cout << ans << endl;
}

int main() {
    //cout.precision(10);
    cin.tie(0);
    ios::sync_with_stdio(false);
    while(cin >> N) {
        if(N == 0) break;
        solve();
    }
    return 0;
}
