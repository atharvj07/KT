n = int(input())
for x in range(n):
	word = list(input())
	count = 0
	if len(word) == 1:
		print(0)
		continue
	elif len(word) == 2:
		if word[0] == word[1]:
			print(1)
		else:
			print(0)
		continue
	else:
		for y in range(0, len(word) - 2):
			if word[y] == word[y + 2] and word[y] != '0' and (word[y + 2] != '0'):
				if y - 1 >= 0:
					if word[y - 1] == word[y]:
						word[y] = '0'
						count += 1
						continue
				word[y + 2] = '0'
				count += 1
		for y in range(0, len(word) - 1):
			if word[y] == word[y + 1] and word[y] != '0':
				word[y] = '0'
				count += 1
	print(count)
