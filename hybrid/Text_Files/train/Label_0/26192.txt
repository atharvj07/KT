#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.0);
template <class Int>
Int gcd(Int a, Int b) {
  return (b == 0 ? abs(a) : gcd(b, a % b));
}
const int dx[]{0, 0, 1, -1, 1, -1, 1, -1};
const int dy[]{1, -1, 0, 0, 1, -1, -1, 1};
void run() {}
const int N = 3e5 + 5;
int main() {
  cout << setprecision(2) << fixed;
  run();
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, k;
  cin >> n >> k;
  map<int, vector<int>> mp;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    int cnt = 0;
    while (x) {
      mp[x].push_back(cnt);
      cnt++;
      x /= 2;
    }
    mp[0].push_back(cnt);
  }
  int mn = 1e9;
  for (int i = 0; i <= 2e5; i++) {
    sort(((mp[i]).begin()), ((mp[i]).end()));
    for (int j = 1; j < ((int)(mp[i]).size()); j++) mp[i][j] += mp[i][j - 1];
    if (((int)(mp[i]).size()) >= k) mn = min(mn, mp[i][k - 1]);
  }
  cout << mn << "\n";
}
