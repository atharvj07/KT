void solve() {
  int n, k1, k2, mx1 = 0, mx2 = 0;
  cin >> n >> k1 >> k2;
  while (k1--) {
    int x;
    cin >> x;
    mx1 = max(mx1, x);
  }
  while (k2--) {
    int x;
    cin >> x;
    mx2 = max(mx2, x);
  }
  if (mx1 > mx2) {

  } else {
    cout << "NO\n";
  }
}
