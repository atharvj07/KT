import sys
(n, m) = map(int, input().split())
a = [0] + list(map(int, input().split()))
b = [n + 1] * (n + 1)
for i in range(n - 1, 0, -1):
	b[i] = i + 1 if a[i] != a[i + 1] else b[i + 1]
ans = [-1] * m
for i in range(m):
	(l, r, x) = map(int, sys.stdin.readline().split())
	if a[l] != x:
		ans[i] = l
	elif b[l] <= r:
		ans[i] = b[l]
print(*ans, sep='\n')
