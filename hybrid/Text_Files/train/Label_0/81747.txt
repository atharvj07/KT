def func(inpstr):
	return inpstr[0]
N = int(input())
arr = []
money = 0
for i in range(N):
	a = [int(i) for i in input().split()]
	arr.append([max(a[1:])] + [a[0]])
arr.sort(key=func)
while len(arr) != 1:
	p = arr[-2][1]
	m = arr[-1][0] - arr[-2][0]
	money += m * p
	arr[-1][1] += p
	arr.pop(-2)
print(money)
