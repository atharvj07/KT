for _ in range(int(input())):
	n = int(input())
	a = [list(map(int, input().split()))]
	while True:
		c = [0] * (n + 1)
		for i in a[-1]:
			c[i] += 1
		b = [c[i] for i in a[-1]]
		if b == a[-1]:
			break
		a.append(b)
	ans = []
	for i in range(int(input())):
		(x, k) = map(int, input().split())
		ans.append(str(a[min([len(a) - 1, k])][x - 1]))
	print('\n'.join(ans))
