n = int(input())
line = str(input()).split()
line = [int(it) for it in line]
line.sort()
link = dict()
count = 1
pp = line[0]
for i in range(1, 100000 + 1):
	link.update({i: 0})
for i in range(1, n):
	if pp == line[i]:
		count += 1
	else:
		link.update({pp: count})
		pp = line[i]
		count = 1
link.update({pp: count})
dp = [0] * (100000 + 1)
dp[1] = link[1]
for i in range(2, 100000 + 1):
	dp[i] = max(dp[i - 2] + i * link[i], dp[i - 1])
print(dp[100000])
