#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int arr[n];
    for (int i = 0; i < n; i++) cin >> arr[i];
    for (int i = 0; i < n; i++) arr[i] -= 1;
    for (int i = 0; i < n; i++) {
      int count = 1;
      int index = arr[i];
      while (i != index) {
        index = arr[index];
        count++;
      }
      cout << count << " ";
    }
    cout << endl;
    ;
  }
}
