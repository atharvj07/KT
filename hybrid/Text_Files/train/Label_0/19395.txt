#include "bits/stdc++.h"
#include<unordered_map>
#pragma warning(disable:4996)
using namespace std;

int N;
vector<vector<int>>rems;

int dy[6] = { 0,1,1,0,-1,-1 };
int dx[6] = { -1,0,1,1,0,-1 };

bool breath(const vector<vector<int>>&tri, const int y, const int x,const int num) {
	if (tri[y][x] == 0)return false;
	if (tri[y][x] != num)return true;
	if (rems[y][x] == -2)return true;
	if (rems[y][x] != -1)return rems[y][x];
	int pre = rems[y][x];
	rems[y][x] = -2;
	bool rem = true;
	for (int i = 0; i < 6; ++i) {
		int ny = y + dy[i]; int nx = x + dx[i];
		if (ny >= 0 && ny < N&&nx >= 0 && nx <= ny) {
			rem &= breath(tri, ny, nx, num);
		}
	}
	rems[y][x] = pre;
	return rem;
}

vector<int>check(const vector<vector<int>>&tri) {
	rems.clear();
	rems.resize(N);
	for (int i = 0; i < N; ++i) {
		for (int j = 0; j < i + 1; ++j) {
			rems[i].push_back(-1);
		}
	}

	for (int i = 0; i < N; ++i) {
		for (int j = 0; j < i + 1; ++j) {
			if (rems[i][j] == -1&&tri[i][j]) {
				rems[i][j]=breath(tri,i,j,tri[i][j]);
			}
		}
	}
	vector<int>remnum(10);
	for (int i = 0; i < N; ++i) {
		for (int j = 0; j < i + 1; ++j) {
			if (rems[i][j] == 1) {
				remnum[tri[i][j]]++;
			}
		}
	}
	return remnum;
}

int main() {
	while (1) {
		int C; cin >> N >> C;
		if (!N)break;
		vector<vector<int>>tri(N);
		for (int i = 0; i < N; ++i) {
			for (int j = 0; j < i + 1; ++j) {
				int a; cin >> a;
				tri[i].push_back(a);
			}
		}
		int ans = -9999;
		for (int i = 0; i < N; ++i) {
			for (int j = 0; j < i + 1; ++j) {
				if (tri[i][j]) {

				}
				else {
					tri[i][j] = C;
					vector<int>rems(check(tri));
					int nowans = 0;
					for (int i = 0; i < rems.size(); ++i) {
						if (i == C)nowans -= rems[i];
						else nowans += rems[i];
					}
					ans = max(ans, nowans);

					tri[i][j] = 0;
				}
			}
		}
		cout << ans << endl;
	}
	return 0;
}
