#include <bits/stdc++.h>
using namespace std;
const long long mod = 998244353;
const long long INF = (long long)1000000007 * 1000000007;
const long double eps = 1e-2;
const long double pi = acos(-1.0);
long long t10[19];
int k;
void init() {
  t10[0] = 1;
  for (int i = 0; i < 18; i++) {
    t10[i + 1] = t10[i] * 10;
  }
}
bool used[19];
int val[19];
bool chk[19][1024][2];
long long ans[19][1024][2];
long long onecalc(long long x, int id, int s, bool f) {
  if (chk[id][s][f]) return ans[id][s][f];
  chk[id][s][f] = true;
  int cnt = 0;
  for (int i = 0; i < 10; i++) {
    if (s & (1 << i)) cnt++;
  }
  if (cnt > k) return ans[id][s][f] = 0;
  if (id == 0) return ans[id][s][f] = 1;
  long long u = (x % t10[id]) / t10[id - 1];
  if (used[id]) {
    int z = val[id];
    if (f) {
      if (z > u)
        return ans[id][s][f] = 0;
      else if (z == u)
        return ans[id][s][f] = onecalc(x, id - 1, s | (1 << z), f);
      else
        return ans[id][s][f] = onecalc(x, id - 1, s | (1 << z), false);
    } else
      return ans[id][s][f] = onecalc(x, id - 1, s | (1 << z), false);
  } else {
    long long res = 0;
    if (f) {
      if (u != 0) {
        if (s == 0)
          res = (res + onecalc(x, id - 1, s, false)) % mod;
        else
          res = (res + onecalc(x, id - 1, s | 1, false)) % mod;
      } else if (s == 0)
        res = (res + onecalc(x, id - 1, s, true)) % mod;
      else {
        res = (res + onecalc(x, id - 1, s | 1, true)) % mod;
      }
      for (int i = 1; i <= u - 1; i++) {
        int ns = s | (1 << i);
        res = (res + onecalc(x, id - 1, ns, false)) % mod;
      }
      if (u != 0) res = (res + onecalc(x, id - 1, s | (1 << u), true)) % mod;
    } else {
      if (s == 0) {
        res = (res + onecalc(x, id - 1, 0, false)) % mod;
      } else {
        res = (res + onecalc(x, id - 1, s | 1, false)) % mod;
      }
      for (int i = 1; i <= 9; i++) {
        int ns = s | (1 << i);
        res = (res + onecalc(x, id - 1, ns, false)) % mod;
      }
    }
    return ans[id][s][f] = res;
  }
}
void set_up(int id, int v) {
  for (int i = 0; i < 19; i++) {
    used[i] = false;
    for (int j = 0; j < 1024; j++) {
      for (int l = 0; l < 2; l++) {
        chk[i][j][l] = false;
        ans[i][j][l] = 0;
      }
    }
  }
  used[id] = true;
  val[id] = v;
}
long long calc(long long x) {
  long long ret = 0;
  for (int i = 1; i <= 18; i++) {
    for (int j = 0; j < 10; j++) {
      set_up(i, j);
      ret = (ret + t10[i - 1] % mod * onecalc(x, 18, 0, true) % mod * j) % mod;
    }
  }
  return ret;
}
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  init();
  long long l, r;
  cin >> l >> r >> k;
  cout << (calc(r) - calc(l - 1) + mod) % mod << endl;
  return 0;
}
