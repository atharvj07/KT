s = input()
numbers = list(map(int, s.split()))
lista = []
for i in range(0, numbers[1]):
	lista.append(i + 1)
for i in range(0, numbers[0]):
	line = input()
	ran = list(map(int, line.split()))
	for j in range(ran[0], ran[1] + 1):
		if j in lista:
			lista.remove(j)
print(len(lista), '\n')
for x in range(len(lista)):
	print(lista[x], end=' ')
