import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
z = 998244353
if n == 1:
	print(1)
else:
	lis = [1]
	for i in range(2, n + 1):
		lis.append(int(lis[-1] * i) % z)
	arr = [n]
	for i in range(n - 2):
		a = n - i - 1
		arr.append(a * arr[-1] % z)
	num = [arr[0]]
	for i in range(1, n - 1):
		num.append((num[-1] + arr[i]) % z)
	x = n * lis[-1] % z
	ans = (x - num[-1]) % z
	print(ans)
