import java.io.OutputStream;

import java.util.Arrays;
import java.util.StringTokenizer;
import java.io.BufferedReader;
import java.io.FileReader;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author OmarYasser
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskC solver = new TaskC();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskC {
        static int MOD = 1000000009;
        static int[][] comb;
        static int MAX = 4001;
        static long[] fact;

        static int nCr(int n, int k) {
            if (n < k)
                return 0;
            if (k == 0 || k == n)
                return 1;
            if (comb[n][k] != -1)
                return comb[n][k];
            if (n - k < k)
                return comb[n][k] = nCr(n, n - k);        //reduce k to n - k
            return comb[n][k] = (nCr(n - 1, k - 1) + nCr(n - 1, k)) % MOD;
        }

        public void solve(int testNumber, Scanner sc, PrintWriter out) {
            int n = sc.nextInt(), good = sc.nextInt(), bad = sc.nextInt();
            comb = new int[MAX][MAX];
            for (int[] c : comb) Arrays.fill(c, -1);
            fact = new long[MAX];
            fact[1] = 1;
            for (int i = 2; i < MAX; i++)
                fact[i] = 1L * fact[i - 1] * i % MOD;

            long[] resMid = new long[n];
            for (int midDays = 1; midDays <= n - 2; midDays++)
                resMid[midDays] = 1L * fact[bad] * nCr(bad - 1, midDays - 1) % MOD;

            long resOut[] = new long[n];
            long res = 0;
            for (int goodDays = 2; goodDays <= n - 1; goodDays++) {
                resOut[goodDays] = (1L * (1L * fact[good] * nCr(good - 1, goodDays - 1) % MOD));
            }
            for (int leftGoodDays = 1; leftGoodDays <= n - 2; leftGoodDays++)
                for (int rightGoodDays = 1; rightGoodDays <= n - leftGoodDays - 1; rightGoodDays++) {
                    if (leftGoodDays + rightGoodDays > good || (n - (leftGoodDays + rightGoodDays)) > bad) continue;
//                System.err.println(leftGoodDays + " " + rightGoodDays + " " + resOut[2]);
                    res += (resOut[leftGoodDays + rightGoodDays] * resMid[n - (leftGoodDays + rightGoodDays)] % MOD);
                    if (res >= MOD) res -= MOD;
                }
            out.println(res);
        }

    }

    static class Scanner {
        StringTokenizer st;
        BufferedReader br;

        public Scanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }

        public Scanner(FileReader f) {
            br = new BufferedReader(f);
        }

        public String next() {
            while (st == null || !st.hasMoreTokens())
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (Exception e) {
                }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
