import java.io.*;

public class Marks {
    public static void main(String[] args) throws Exception {
        DataInputStream K = new DataInputStream(System.in);
        String nm[] = (K.readLine() + " ").split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        int a[][] = new int[n][m];
        for (int i = 0; i < n; i++) {
            String st = K.readLine();
            for (int j = 0; j < m; j++) {
                a[i][j] = Integer.parseInt(st.charAt(j) + "");
            }
        }
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            boolean wasBest = false;
            for (int j = 0; j < m; ++j) {
                boolean isBest = true;
                for (int k = 0; k < n; ++k)
                    if (a[k][j] > a[i][j])
                        isBest = false;
                if (isBest)
                    wasBest = true;
            }
            if (wasBest)
                ans++;
        }
        System.out.println(ans);
    }
}
