import sys
from collections import Counter
inp = sys.stdin.readline
read = lambda : list(map(int, inp().strip().split()))

def card(n):
	return n * (n + 1) + n * (n + 1) // 2 - n

def a():
	ans = ''
	for _ in range(int(inp())):
		(n, m) = read()
		if n == 1 or m == 1:
			ans += 'YES\n'
		elif n == 2 and m == 2:
			ans += 'YES\n'
		else:
			ans += 'NO\n'
	print(ans)

def b():
	ans = ''
	for _ in range(int(inp())):
		n = int(inp())
		s = 0
		while 1 < n:
			i = 1
			while card(i) <= n:
				i += 1
			n -= card(i - 1)
			s += 1
		ans += str(s) + '\n'
	print(ans)
b()
