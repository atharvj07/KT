n = int(input())
d = {}
(o, t) = (0, 0)
for _ in range(n - 1):
	(s, e) = map(int, input().split())
	if s in d:
		d[s] += 1
	else:
		d[s] = 1
	if e in d:
		d[e] += 1
	else:
		d[e] = 1
for i in d:
	if d[i] == 1:
		o += 1
	elif d[i] == 2:
		t += 1
if t:
	print('NO')
elif o * (o - 1) // 2 >= n - 1:
	print('YES')
else:
	print('NO')
