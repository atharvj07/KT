(n, t) = input().split()
(n, t) = (int(n), int(t))
s = list(input())
if n == 1:
	if 2 ** (ord(s) - 97) == t:
		print('Yes')
	else:
		print('No')
elif n == 2:
	if 2 ** (ord(s[1]) - 97) - 2 ** (ord(s[0]) - 97) == t:
		print('Yes')
	else:
		print('No')
else:
	t -= 2 ** (ord(s[-1]) - 97) - 2 ** (ord(s[-2]) - 97)
	t = abs(t)
	s = s[:-2]
	count = [0] * 32
	for char in s:
		count[ord(char) - 97] += 1
	for i in range(30, -1, -1):
		for x in range(count[i]):
			if t > 0:
				t -= 1 << i
			else:
				t += 1 << i
	if t == 0:
		print('Yes')
	else:
		print('No')
