#include <bits/stdc++.h>
using namespace std;
vector<vector<int> > e;
vector<int> used;
void dfs(int v) {
  used[v] = 1;
  for (int i = 0; i < e[v].size(); i++) {
    if (!used[e[v][i]]) {
      dfs(e[v][i]);
    }
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  int n, m;
  cin >> n >> m;
  e.resize(n);
  long long loops = 0;
  vector<int> loop(n, 0);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> v >> u;
    v--, u--;
    if (u != v) {
      e[u].push_back(v);
      e[v].push_back(u);
    } else {
      loops++;
      loop[v] = 1;
    }
  }
  if (m < 2) {
    cout << "0\n";
    return 0;
  }
  used.assign(n, 0);
  int start = 0;
  while (start < n && e[start].size() == 0 && !loop[start]) {
    start++;
  }
  if (start == n) {
    cout << "1\n";
    return 0;
  }
  dfs(start);
  for (int i = 0; i < n; i++) {
    if (!used[i] && (e[i].size() > 0 || loop[i])) {
      cout << "0\n";
      return 0;
    }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    long long j = e[i].size();
    ans += j * (j - 1) / 2;
  }
  ans += loops * (loops - 1) / 2;
  ans += loops * (m - loops);
  cout << ans << '\n';
  return 0;
}
