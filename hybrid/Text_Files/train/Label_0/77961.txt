def decimal(ss):
	ans = 0
	c = 0
	for i in range(len(ss) - 1, -1, -1):
		if ss[i] == '1':
			ans += 2 ** c
		c += 1
	return ans
for _ in range(int(input())):
	s = input()
	ll = len(s)
	zer = [0] * (ll + 1)
	s += '#'
	for i in range(ll):
		if s[i] == '0':
			zer[i + 1] = zer[i] + 1
		else:
			zer[i + 1] = 0
	ans = 0
	for i in range(ll):
		if s[i] == '1':
			for j in range(min(20, ll - i)):
				no = decimal(s[i:i + j + 1])
				if zer[i] + j + 1 >= no:
					ans += 1
	print(ans)
