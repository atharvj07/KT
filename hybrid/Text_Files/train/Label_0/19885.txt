#include <bits/stdc++.h>
using namespace std;
int i, j, k, n, m, a[200000], sum, ma, b[200000], now;
int main() {
  scanf("%d", &n);
  for (i = 0; i < 2 * n - 1; i++) {
    scanf("%d", &a[i]);
    sum += a[i];
    b[i] = a[i];
  }
  ma = sum;
  sort(a, a + 2 * n - 1);
  if (n % 2 == 0) {
    for (i = 0; i < n - 1; i++) {
      sum -= a[2 * i];
      sum -= a[2 * i + 1];
      a[2 * i] = -a[2 * i];
      a[2 * i + 1] = -a[2 * i + 1];
      sum += a[2 * i];
      sum += a[2 * i + 1];
      if (sum > ma) ma = sum;
    }
  } else
    for (i = 0; i < 2 * n - 1; i++) {
      sum -= a[i];
      a[i] = -a[i];
      sum += a[i];
      if (sum > ma) ma = sum;
    }
  printf("%d\n", ma);
}
