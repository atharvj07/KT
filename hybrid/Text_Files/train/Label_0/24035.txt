#include <bits/stdc++.h>
const double Pi = 3.14159265;
const int N = 6e5 + 10;
const unsigned long long base = 163;
const int INF = 0x3f3f3f3f;
const int mod = (int)1e9 + 7;
using namespace std;
char s1[500010], s2[500010];
int main() {
  int n, k;
  cin >> n >> k;
  cin >> s1 >> s2;
  long long dif = 0;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (dif + 1 > k) {
      ans += k;
    } else {
      dif = dif * 2;
      dif += s2[i] - s1[i];
      ans += min(dif + 1, (long long)k);
    }
  }
  cout << ans << endl;
  return 0;
}
