s = input()
l = []
i = 0
while True:
	if i < len(s) - 2:
		if s[i] == '1':
			if s[i + 1] == '4':
				if s[i + 2] == '4':
					i = i + 3
					l.append(1)
				else:
					i = i + 2
					l.append(1)
			else:
				i = i + 1
				l.append(1)
		else:
			print('NO')
			l.append(0)
			break
	if i == len(s) - 2:
		if s[i] == '1':
			if s[i + 1] == '4':
				l.append(1)
				break
			else:
				i = i + 1
				l.append(1)
		else:
			l.append(0)
			print('NO')
			break
	if i == len(s) - 1:
		if s[i] == '1':
			l.append(1)
			break
		else:
			print('NO')
			l.append(0)
			break
	if i == len(s):
		break
if 0 not in l:
	print('YES')
