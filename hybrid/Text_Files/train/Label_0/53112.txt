import java.util.*;
public class Code{
	public static void main(String[] args){
		Scanner sc=new Scanner(System.in);
		int t;
		t=sc.nextInt();
		while(t!=0){
			int n=sc.nextInt();
			int q=sc.nextInt();
			String s;
			s=sc.next();
			while(q!=0){
				int l=sc.nextInt();
				int r=sc.nextInt();
				l--;
				r--;
				char ch1=s.charAt(l);
				char ch2=s.charAt(r);
				boolean kj=false;
				for(int i=0;i<l;i++){
					if(ch1==s.charAt(i)){
						System.out.println("YES");
						kj=true;
						break;
					}
				}
				for(int i=r+1;i<s.length()&&(!kj);i++){
					if(ch2==s.charAt(i)){
						System.out.println("YES");
						kj=true;
						break;
					}
				}
				if(!kj)System.out.println("NO");
				q=q-1;
			}
			t=t-1;
		}   	
		
	}
}
