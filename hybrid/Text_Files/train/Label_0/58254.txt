public class Rasstanovka {

    public static void main(String args[]) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        if (n == 1) {
            System.out.println("1");
        } else if (n == 2) {
            System.out.println("2 1");
        } else {
            for (int i = 1; i < n; i++) {
                System.out.print(i + 1);
                System.out.print(" ");
            }
            System.out.print("1 ");
        }
    }
}
