#include <bits/stdc++.h>
using namespace std;
const long long MAXN = 1e5 + 50, MAXS = 4 * MAXN;
long long seg[MAXS], sum[MAXS], id[MAXN], idtox[MAXN], temp[MAXN];
void upd(long long x, long long xl, long long xr, long long ql, long long qr) {
  if (xl >= qr || xr <= ql) return;
  if (xr - xl == 1) {
    seg[x] = 0;
    sum[x] = temp[xl];
    return;
  }
  long long mid = (xl + xr) / 2, l = 2 * x, r = l + 1;
  upd(l, xl, mid, ql, qr);
  upd(r, mid, xr, ql, qr);
  sum[x] = sum[l] + sum[r];
  seg[x] = seg[l] + seg[r] - sum[l] * (xr - mid) + sum[r] * (mid - xl);
}
long long get(long long x, long long xl, long long xr, long long ql,
              long long qr) {
  if (xl >= qr || xr <= ql) return 0;
  if (xl >= ql && xr <= qr) return seg[x] + sum[x] * ((xl - ql) - (qr - xr));
  long long mid = (xl + xr) / 2, l = 2 * x, r = l + 1;
  return get(l, xl, mid, ql, qr) + get(r, mid, xr, ql, qr);
}
bool bytemp(long long a, long long b) { return temp[a] < temp[b]; }
int main() {
  ios::sync_with_stdio(false);
  cout.tie(0);
  cin.tie(0);
  long long n;
  cin >> n;
  for (long long i = 0; i < n; i++) cin >> temp[i], id[i] = i;
  sort(id, id + n, bytemp);
  sort(temp, temp + n);
  for (long long i = 0; i < n; i++) idtox[id[i]] = i;
  upd(1, 0, n, 0, n);
  long long m;
  cin >> m;
  while (m--) {
    long long t;
    cin >> t;
    if (t == 2) {
      long long l, r;
      cin >> l >> r;
      l = lower_bound(temp, temp + n, l) - temp;
      r = upper_bound(temp, temp + n, r) - temp;
      cout << get(1, 0, n, l, r) << "\n";
      ;
    } else {
      long long x, d;
      cin >> x >> d;
      x--;
      if (d > 0) {
        long long p = idtox[x], ex = temp[p] + d,
                  newp = upper_bound(temp + p, temp + n, ex) - temp - 1;
        for (long long i = p; i < newp; i++) {
          id[i] = id[i + 1];
          idtox[id[i]] = i;
          temp[i] = temp[i + 1];
        }
        idtox[x] = newp;
        temp[newp] = ex;
        id[newp] = x;
        upd(1, 0, n, p, newp + 1);
      } else if (d < 0) {
        long long p = idtox[x], ex = temp[p] + d,
                  newp = lower_bound(temp, temp + p + 1, ex) - temp;
        for (long long i = p; i > newp; i--) {
          id[i] = id[i - 1];
          idtox[id[i]] = i;
          temp[i] = temp[i - 1];
        }
        idtox[x] = newp;
        temp[newp] = ex;
        id[newp] = x;
        upd(1, 0, n, newp, p + 1);
      }
    }
  }
}
