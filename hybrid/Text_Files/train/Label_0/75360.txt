for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	curr = n - 1
	ans = ['0'] * n
	while curr >= 0:
		units = arr[curr]
		if units == 0:
			curr -= 1
		while units and curr >= 0:
			ans[curr] = '1'
			curr -= 1
			units = max(units - 1, arr[curr])
	print(' '.join(ans))
