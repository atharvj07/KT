import math

def solve(n, A, p):
	res = A.count(1)
	for i in range(1, 150):
		count = 0
		for j in range(n):
			(r, s) = p[j]
			x = math.floor((i - s + r) / r)
			if i < s:
				count += A[j]
			elif A[j] == 0 and x % 2 == 1:
				count += 1
			elif A[j] == 1 and x % 2 == 0:
				count += 1
		res = max(res, count)
	return res
n = int(input())
s = input()
A = []
for c in s:
	A.append(ord(c) - ord('0'))
p = []
for i in range(n):
	(a, b) = map(int, input().split())
	p.append((a, b))
print(solve(n, A, p))
