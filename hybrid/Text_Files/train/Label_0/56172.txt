import java.util.Scanner;

public class ProblemA {

    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int n = Math.min(b, c);
        int sum = (2*a+n-1)*n;
        int rest = b+c-1-2*n;
        System.out.println(sum+rest*(a+n-1));
    }
    
}
