                   int j = rand.nextInt(i + 1);

        public long fact(long h){
            long sum=1;

        public long primef(double r){
            long c=0;
            while(r%2==0){
                c++;
                r=r/2;
            }
            for (int i = 3; i <=Math.sqrt(r) ;i+=2) {
                while(r%i==0){
                    c++;
                    r=r/3;
                }
            }
            if(r>2){
                c++;
            }
            return c;
        }
