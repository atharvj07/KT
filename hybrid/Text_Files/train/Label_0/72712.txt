import sys
import math
import collections
import bisect

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_string():
	return sys.stdin.readline().strip()
for t in range(1):
	(n, k) = get_ints()
	arr = get_list()
	minim = 10 ** 10
	ans = 0
	maxim = -10 ** 10
	for i in arr:
		ans += i
		maxim = max(ans, maxim)
		minim = min(minim, ans)
	if maxim > k:
		print(0)
		continue
	if minim < 0:
		starting = -minim
	else:
		starting = 0
	if maxim <= 0:
		end = k
	else:
		end = k - maxim
	if starting > end:
		print(0)
	else:
		print(end - starting + 1)
