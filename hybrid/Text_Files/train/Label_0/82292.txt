n = int(input())
l = list(map(int, input().split()))
s = set(l)
ans = [l[0]]
for i in l:
	j = 1
	while j < int(2000000000.0):
		if i + j in s and i + 2 * j in s:
			print('3')
			print(i, i + j, i + 2 * j)
			quit()
		elif i + j in s:
			ans = [i, i + j]
		j *= 2
print(len(ans))
for i in ans:
	print(i, end=' ')
