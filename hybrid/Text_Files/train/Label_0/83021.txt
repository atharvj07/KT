for i in range(int(input())):
	(n, m) = map(int, input().split())
	f = input()
	p = list(map(int, input().split()))
	tt = [0] * (n + 2)
	for j in p:
		tt[0] += 1
		tt[j] -= 1
	tt[0] += 1
	tt[n] -= 1
	for j in range(1, len(tt)):
		tt[j] += tt[j - 1]
	r = {}
	u = 'abcdefghijklmnopqrstuvwxyz'
	for j in u:
		if j not in r:
			r[j] = 0
	for j in range(n):
		r[f[j]] += tt[j]
	print(*r.values())
