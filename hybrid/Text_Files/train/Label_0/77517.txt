import math
import collections

def make_grid(p0, s_uniq):
	s_uniq = list(s_uniq)
	grid = [list(range(13)) for _ in range(2)]
	if p0 >= 0:
		grid[0][p0:] = s_uniq[:13 - p0]
		grid[1] = list(reversed(s_uniq[13 - p0:13 - p0 + 13]))
		grid[0][:p0] = s_uniq[13 - p0 + 13:]
	else:
		grid[1][:-p0] = list(reversed(s_uniq[:-p0]))
		grid[0] = s_uniq[-p0:-p0 + 13]
		grid[1][-p0:] = list(reversed(s_uniq[-p0 + 13:]))
	return grid
s = input()
poses = collections.defaultdict(list)
for (i, c) in enumerate(s):
	poses[c].append(i)
for c in poses:
	if len(poses[c]) > 1 and poses[c][1] - poses[c][0] > 1:
		i = poses[c][0]
		j = poses[c][1]
		s_uniq = s[:j] + s[j + 1:]
		ans = make_grid(math.floor((26 - i - j) / 2), s_uniq)
		print(''.join(ans[0]))
		print(''.join(ans[1]))
		raise SystemExit
print('Impossible')
