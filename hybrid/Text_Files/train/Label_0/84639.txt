import sys
import math

def fn(s):
	s = list(s)
	for i in range(len(s)):
		if i % 2 == 0:
			if s[i] != 'a':
				s[i] = 'a'
			else:
				s[i] = 'b'
		elif s[i] != 'z':
			s[i] = 'z'
		else:
			s[i] = 'y'
	print(''.join(s))
input = sys.stdin.read()
data = list(map(str, input.split()))
t = int(data[0])
l = 0
for i in range(t):
	s = data[l + 1]
	l = l + 1
	fn(s)
