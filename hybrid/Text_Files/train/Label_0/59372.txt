
import java.io.*;
import java.util.*;

public class SequenceDP
{
	public Scanner in = new Scanner(System.in);
	public PrintStream out = System.out;

	//f[x]: v[i] is exactly T[x]
	//g[x]: v[i] is at most T[x]
	public long[] f, g;
	public long[] T;
	public TreeSet<Long> S;
	
	public long[] v;
	public int n;
	
	public void main()
	{
		int i, x;
		
		n = in.nextInt();
		
		v = new long[n];
		S = new TreeSet<Long>();
		for(i=0;i<n;++i)
		{
			v[i] = in.nextLong();
			S.add(v[i]);
		}
		
		T = LTS2La(S);
		
		f = new long[T.length];
		g = new long[T.length];
		
		i = 0;
		x = 0;
		f[x] = Math.abs(T[x] - v[i]);
		g[x] = f[x];
		
		for(++x;x<T.length;++x)
		{
			f[x] = Math.abs(T[x] - v[i]);
			g[x] = Math.min(g[x-1], f[x]);		
		}
		
		for(++i;i<v.length;++i)
		{
			long[] newF = new long[T.length];
			long[] newG = new long[T.length];
			
			x = 0;
			newF[x] = g[x] + Math.abs(T[x] - v[i]);
			newG[x] = newF[x];
			
			for(++x;x<T.length;++x)
			{
				newF[x] = g[x] + Math.abs(T[x] - v[i]);
				newG[x] = Math.min(newG[x-1], newF[x]);
			}
			
			f = newF;
			g = newG;
		}
		
		out.println(g[T.length-1]);
	}//end public void main()

	public long[] LTS2La(TreeSet<Long> S)
	{
		long[] ret = new long[S.size()];
		int i = 0;
		for(Long x : S) { ret[i] = x; ++i; }
		return ret;
	}	
	
	public static void main(String[] args)
	{
		(new SequenceDP()).main();
	}
}
