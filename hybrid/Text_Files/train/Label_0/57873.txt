import java.util.Scanner;


public class A1023 {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		String  s = sc.next();
		String  t = sc.next();
		String ans = "NO";
		if(n>m+1){
			System.out.println(ans);
			return;
		}
		char a = 0;
		char b;
		int i;
		for (i = 0; i <Math.max(m,n); i++) {
			try{
				a = s.charAt(i);
			}catch(Exception e1){
				System.out.println("NO");
				return;
			}		
			 if(a=='*'){
				 break;
			 }
			 try{
				 	b = t.charAt(i);
				}catch(Exception e1){
					System.out.println("NO");
					return;
				}
			 if(a!=b){
				 System.out.println("NO");
				 return;
			 }
		}
		if(i==n-1){		
			System.out.println("YES");
			return;
		}
		for(int j=n-1;j>i;j--,m--){
			a = s.charAt(j);
			b = t.charAt(m-1);
			if(a!=b){
				System.out.println("NO");
				return;
			}
		}
		System.out.println("YES");
		sc.close();
	}
	

}
