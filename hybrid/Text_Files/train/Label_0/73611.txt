n = int(input())
e = {}
for i in range(n - 1):
	(a, b) = map(int, input().split(' '))
	x = e.get(a, [])
	x.append(b)
	e[a] = x
	x = e.get(b, [])
	x.append(a)
	e[b] = x
x = 1
y = 0
temp1 = [1]
temp2 = []
seen = set()
k = 0
while len(temp1) > 0:
	for i in temp1:
		seen.add(i)
		for j in e[i]:
			if j not in seen:
				temp2.append(j)
				if k % 2 == 0:
					y += 1
				else:
					x += 1
	temp1 = temp2
	temp2 = []
	k += 1
print(x * y - n + 1)
