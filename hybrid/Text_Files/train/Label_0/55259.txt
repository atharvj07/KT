import java.io.BufferedWriter;
import java.io.IOException;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;

public class Test {

  static PrintWriter writer =
      new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
  int[] ht = new int[1 << 24];
  int[][] bits = new int[128][3];

  {
    bits['A'] = new int[] {2, 0, 1};
    bits['X'] = new int[] {2, 0, 3};
    bits['O'] = new int[] {1, 0};
    bits['a'] = new int[] {1, 3};
    bits['x'] = new int[] {1, 1};
    bits['o'] = new int[] {2, 1, 3};
  }

  static int readInt() {
    int ans = 0;
    boolean neg = false;
    try {
      boolean start = false;
      for (int c = 0; (c = System.in.read()) != -1; ) {
        if (c == '-') {
          start = true;
          neg = true;
          continue;
        } else if (c >= '0' && c <= '9') {
          start = true;
          ans = ans * 10 + c - '0';
        } else if (start) break;
      }
    } catch (IOException e) {
    }
    return neg ? -ans : ans;
  }

  static long readLong() {
    long ans = 0;
    boolean neg = false;
    try {
      boolean start = false;
      for (int c = 0; (c = System.in.read()) != -1; ) {
        if (c == '-') {
          start = true;
          neg = true;
          continue;
        } else if (c >= '0' && c <= '9') {
          start = true;
          ans = ans * 10 + c - '0';
        } else if (start) break;
      }
    } catch (IOException e) {
    }
    return neg ? -ans : ans;
  }

  static String readString() {
    StringBuilder b = new StringBuilder();
    try {
      boolean start = false;
      for (int c = 0; (c = System.in.read()) != -1; ) {
        if (Character.isLetterOrDigit(c)) {
          start = true;
          b.append((char) c);
        } else if (start) break;
      }
    } catch (IOException e) {
    }
    return b.toString();
  }

  static int readChars(char[] a, int off) {
    int cnt = 0;
    try {
      boolean start = false;
      for (int c = 0; (c = System.in.read()) != -1; ) {
        if (Character.isLetterOrDigit(c)) {
          start = true;
          a[off + cnt++] = (char) c;
        } else if (start) break;
      }
    } catch (IOException e) {
    }
    return cnt;
  }

  public static void main(String[] args) {
    Test te = new Test();
    te.start();
    writer.flush();
  }

  void start() {
    int w = readInt(), n = readInt(), m = readInt();
    int[] a = new int[1 << w];
    for (int i = 0; i < n; i++) a[readInt()]++;
    int[] pat = new int[1 << w];
    for (int i = 0; i < a.length; i++)
      for (int j = 0; j < w; j++) pat[i] |= ((i >> j) & 1) << (2 * j);
    for (int i = 0; i < a.length; i++)
      for (int j = 0; j < a.length; j++) ht[(pat[i] + pat[j]) + (pat[i] & pat[j])] += a[i] * a[j];
    int[][] masks = new int[2][1 << w];
    char[] g = new char[w];
    while (m-- > 0) {
      readChars(g, 0);
      for (int i = 0, j = w - 1; i < j; i++, j--) {
        char t = g[i];
        g[i] = g[j];
        g[j] = t;
      }
      int ans = 0;
      int rr = 0;
      int id = 0;
      masks[id][rr++] = 0;
      for (int i = w - 1; i >= 0; i--) {
        int nid = 1 - id;
        int sz = 0;
        for (int j = bits[g[i]][0]; j > 0; j--) {
          int bj = bits[g[i]][j];
          if (i > 0) for (int k = 0; k < rr; k++) masks[nid][sz++] = (masks[id][k] << 2) | bj;
          else for (int k = 0; k < rr; k++) ans += ht[(masks[id][k] << 2) | bj];
        }
        id = nid;
        rr = sz;
      }
      writer.println(ans);
    }
  }
}
