import java.io.*;
import java.util.*;

public class Main {
    static int edges[][];
    static int color[];
    static boolean ready[];
    static boolean must = true;
    static Queue<Integer> ans;

    static void dfs(int u) {
        color[u] = 1;
        for (int i = 0; i < edges[u].length; i++) {
            if (color[edges[u][i]] == 1)
                must = false;
            if (color[edges[u][i]] == 0) {
                dfs(edges[u][i]);
            }
        }
        ans.add(u);
        ready[u] = true;
        color[u] = 2;
    }

    public static void main(String args[]) throws IOException {
        Reader in = new Reader();
        Writer out = new Writer();
        int n = in.nextInt();
        int k = in.nextInt();
        edges = new int[n + 1][];
        color = new int[n + 1];
        ready = new boolean[n + 1];
        edges[0] = new int[k];
        ans = new LinkedList<Integer>();
        for (int i = 0; i < k; i++) {
            edges[0][i] = in.nextInt();
        }
        for (int i = 1; i < n + 1; i++) {
            edges[i] = new int[in.nextInt()];
            for (int j = 0; j < edges[i].length; j++) {
                edges[i][j] = in.nextInt();
            }
        }
        dfs(0);
        if (must) {
            out.print(ans.size() - 1 + "\n");
            while (ans.size() > 1)
                out.print(ans.poll() + " ");

        } else {
            out.print(-1);
        }
        out.close();
    }
}

class Reader {
    BufferedReader in;

    Reader(String a) throws IOException {
        in = new BufferedReader(new FileReader(a));
    }

    Reader() throws IOException {
        in = new BufferedReader(new InputStreamReader(System.in));
    }

    StringTokenizer tokin = new StringTokenizer("");

    void updtok() throws IOException {
        if (!tokin.hasMoreTokens()) {
            String a = in.readLine();
            if (a != null) {
                tokin = new StringTokenizer(a);
            }
        }
    }


    String next() throws IOException {
        updtok();
        return tokin.nextToken();
    }

    int nextInt() throws IOException {
        return Integer.parseInt(next());
    }
}

class Writer {
    PrintWriter out;

    Writer() throws IOException {
        out = new PrintWriter(System.out);
    }

    Writer(String name) throws IOException {
        out = new PrintWriter(new FileWriter(name));
    }

    StringBuilder sout = new StringBuilder();

    void print(Object a) {
        sout.append(a);
    }

    void close() {
        out.print(sout.toString());
        out.close();
    }
}
