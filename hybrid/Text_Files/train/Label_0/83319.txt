from collections import deque
t = int(input())
abc = 'abcdefghijklmnopqrstuvwxyz'
for _ in range(t):
	s = input()
	d = deque(list(s[0]))
	res = False
	a = set()
	for i in range(1, len(s)):
		if len(d) != 1 and d[0] == d[-1]:
			res = True
			break
		if s[i] + s[i - 1] in a or s[i - 1] + s[i] in a:
			pass
		elif s[i - 1] == d[0]:
			d.appendleft(s[i])
			a.add(s[i] + s[i - 1])
		elif s[i - 1] == d[-1]:
			d.append(s[i])
			a.add(s[i] + s[i - 1])
		else:
			res = True
			break
	if res or (len(d) != 1 and d[0] == d[-1]):
		print('NO')
	else:
		ans = list(d)
		count = len(d)
		d = set(d)
		for i in abc:
			if i not in d:
				ans.append(i)
		if len(ans) != 26:
			print('NO')
		else:
			print('YES')
			print(*ans, sep='')
