import os
import sys
from atexit import register
from io import BytesIO
input = BytesIO(os.read(0, os.fstat(0).st_size)).readline
(n, m, p) = list(map(int, input().split()))
As = list(map(int, input().split()))
Bs = list(map(int, input().split()))
k = 0
for a in As:
	if a % p:
		break
	k += 1
l = 0
for b in Bs:
	if b % p:
		break
	l += 1
print(l + k)
