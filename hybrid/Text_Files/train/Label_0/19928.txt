signed main(void) {
    int a,b,c,n;
    cin >> a >> b >> c >> n;
    vi x(n),y(n),z(n);
    rep(i,n)cin >> x[i] >> y[i] >> z[i];
    int ans = a*b*2 + a*c*2 + b*c*2;
    rep(i,n){
        int t = 0;
        if(x[i] == 0)t++;
        if(x[i] == a-1)t++;
        if(y[i] == 0)t++;
        if(y[i] == b-1)t++;
        if(z[i] == 0)t++;

            if(diff == 1)t++;
        }
        ans += -t + (6-t);
    }
