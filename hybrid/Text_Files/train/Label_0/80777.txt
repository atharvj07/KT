import sys
from collections import defaultdict, deque
from bisect import bisect_left
input = lambda : sys.stdin.readline().rstrip()
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	a = [*map(int, input().split())]
	b = [*map(int, input().split())]
	(c, d) = ([], [])
	sm = 0
	for i in range(n):
		sm += a[i]
		if b[i] == 1:
			c.append(a[i])
		else:
			d.append(a[i])
	c.sort(reverse=True)
	d.sort(reverse=True)
	for i in range(1, len(c)):
		c[i] += c[i - 1]
	for i in range(1, len(d)):
		d[i] += d[i - 1]
	if sm >= m:
		ans = len(c) + 2 * len(d)
	else:
		print(-1)
		continue
	for i in range(len(c)):
		if c[i] >= m:
			ans = min(ans, i + 1)
			continue
		idx = bisect_left(d, m - c[i])
		if idx != len(d):
			ans = min(ans, i + 2 * idx + 3)
	for i in range(len(d)):
		if d[i] >= m:
			ans = min(ans, 2 * (i + 1))
			continue
		idx = bisect_left(c, m - d[i])
		if idx != len(c):
			ans = min(ans, 2 * i + idx + 3)
	print(ans)
