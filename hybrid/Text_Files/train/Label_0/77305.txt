s = input()
t = input()
(lens, lent) = (len(s), len(t))
from sys import exit
if lens < lent:
	print(s)
	exit()
else:
	(onet, zerot) = (0, 0)
	for (i, x) in enumerate(t):
		if x == '0':
			zerot += 1
		else:
			onet += 1
	(ones, zeros) = (0, 0)
	for (i, x) in enumerate(s):
		if x == '0':
			zeros += 1
		else:
			ones += 1
	res = ''
	if onet <= ones and zerot <= zeros:
		(res, ones, zeros) = (t, ones - onet, zeros - zerot)
		if ones == 0 and zeros == 0:
			print(res)
			exit()
	else:
		print(s)
		exit()
	(item, k, ot, zt, ont, zet) = ('', lent, 0, 0, 0, 0)
	for i in range(lent - 1):
		if t[i] == '1':
			ot += 1
		else:
			zt += 1
		item += t[i]
		if t[lent - i - 1:] == item:
			(k, ont, zet) = (i, ot, zt)
	(ont, zet) = (onet - ont, zerot - zet)
	if k < lent:
		plus = t[k + 1:]
	else:
		plus = t
	while ont <= ones and zet <= zeros:
		res += plus
		ones -= ont
		zeros -= zet
	res += '1' * ones + '0' * zeros
	print(res)
