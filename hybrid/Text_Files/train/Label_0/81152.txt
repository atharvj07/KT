import math

def isPrimes2(n):
	if n > 1:
		if n == 2:
			return True
		if n % 2 == 0:
			return False
		for x in range(3, int(math.sqrt(n) + 1), 2):
			if n % x == 0:
				return False
		return True
	return False

def main():
	t = int(input())
	for i in range(t):
		(a, b, n, m) = map(int, input().split())
		flag = 0
		while flag == 0:
			if n + m > a + b:
				flag = 1
				print('No')
			elif a <= b:
				if m > a:
					flag = 1
					print('No')
				else:
					flag = 1
					print('Yes')
			elif m <= b:
				flag = 1
				print('Yes')
			else:
				flag = 1
				print('No')
main()
