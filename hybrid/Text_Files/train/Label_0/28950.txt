#include <bits/stdc++.h>
using namespace std;
vector<int> a;
int main() {
  int n, s;
  cin >> n >> s;
  int ans = 0, kl = 0;
  for (int i = 0; i < n; ++i)
    if (i != s - 1) {
      int x;
      cin >> x;
      a.push_back(x);
    } else {
      int x;
      cin >> x;
      if (x) ++ans;
    }
  sort(a.begin(), a.end());
  int j = 1, i;
  for (i = 0; i < a.size() && !a[i]; ++i) ++kl, ++ans;
  n = a.size() - 1;
  for (; i <= n;) {
    if (a[i] != j) {
      if (kl) {
        --kl;
        ++j;
      } else if (n > i) {
        --n;
        ++ans;
        ++j;
      } else {
        ++ans;
        ++i;
        ++j;
      }
    } else {
      while (i < a.size() && a[i] == j) ++i;
      ++j;
    }
  }
  cout << ans;
}
