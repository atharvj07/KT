#include <bits/stdc++.h>
using namespace std;
const int N = 500 + 5;
const int mod = 998244353;
using ll = long long;
int trie[505][11];
int cnt[505];
int lvl[505];
int ptr;
void add(string x, int v) {
  int cur = 0;
  for (auto c : x) {
    if (!trie[cur][c - '0']) {
      trie[cur][c - '0'] = ++ptr;
      lvl[ptr] = lvl[cur] + 1;
    }
    cur = trie[cur][c - '0'];
  }
  cnt[cur] += v;
}
int n, k;
int dp[N][N][11];
int solve(int trieNode, int lastChoose, int k) {
  int &ret = dp[trieNode][lastChoose][k];
  if (~ret) return ret;
  ret = 1e9;
  vector<int> tmp(k + 1, 0);
  for (int c = 0; c < 10; c++)
    if (trie[trieNode][c]) {
      vector<int> go;
      for (int j = 0; j <= k; j++)
        go.push_back(solve(trie[trieNode][c], lastChoose, j));
      vector<int> nxt(k + 1, 1e9);
      for (int i = 0; i <= k; i++)
        for (int j = 0; i + j <= k; j++)
          nxt[i + j] = min(nxt[i + j], go[j] + tmp[i]);
      tmp = nxt;
    }
  ret = min(ret, tmp[k] + cnt[trieNode] * (lvl[trieNode] - lastChoose));
  if (k) ret = min(ret, solve(trieNode, lvl[trieNode], k - 1));
  return ret;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    string s;
    int v;
    cin >> s >> v;
    add(s, v);
  }
  memset(dp, -1, sizeof dp);
  cout << solve(0, 0, k) << '\n';
  return 0;
}
