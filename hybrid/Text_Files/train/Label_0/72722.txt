inf = float('inf')

def f(A, T):
	dp = [0] * len(A)
	for i in range(len(A) - 1, -1, -1):
		if len(A) - i < 3:
			dp[i] = inf
		elif len(A) - i == 3:
			dp[i] = A[i + 2][0] - A[i][0]
			T[i] = len(A)
		else:
			extend = dp[i + 1] + (A[i + 1][0] - A[i][0])
			start = A[i + 2][0] - A[i][0] + dp[i + 3]
			if extend < start:
				dp[i] = extend
				T[i] = T[i + 1]
			else:
				dp[i] = start
				T[i] = i + 3
	return dp[0]
n = int(input())
rest = map(int, input().split())
A = []
for (i, a) in enumerate(rest):
	A.append((a, i))
A.sort(key=lambda x: x[0])
T = [0] * len(A)
ans = f(A, T)
X = [0] * len(A)
team = 0
index = 0
while index != len(A):
	nxt = T[index]
	for k in range(index, nxt):
		(_, idx) = A[k]
		X[idx] = team + 1
	team += 1
	index = nxt
print(ans, team)
print(' '.join(map(str, X)))
