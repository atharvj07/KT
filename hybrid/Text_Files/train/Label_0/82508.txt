import sys, math
import io, os

def data():
	return sys.stdin.buffer.readline().strip()

def mdata():
	return list(map(int, data().split()))

def outl(var):
	sys.stdout.write(' '.join(map(str, var)) + '\n')

def out(var):
	sys.stdout.write(str(var) + '\n')
INF = float('inf')
mod = int(1000000000.0) + 7
(k1, k2, k3) = mdata()
n = k1 + k2 + k3
a = [0] * n
for i in mdata():
	a[i - 1] = 0
for i in mdata():
	a[i - 1] = 1
for i in mdata():
	a[i - 1] = 2
cnt = [[0 for i in range(3)] for i in range(n + 1)]
l = [0] * (n + 1)
m = 0
for i in range(n):
	cnt[i + 1] = cnt[i][:]
	cnt[i + 1][a[i]] += 1
	if cnt[i + 1][1] - cnt[i + 1][0] < m:
		l[i + 1] = i + 1
		m = cnt[i + 1][1] - cnt[i + 1][0]
	else:
		l[i + 1] = l[i]
cnt.append(cnt[-1])
ans = INF
for i in range(n + 1):
	a1 = l[i]
	val = cnt[a1][1] - cnt[a1][0] + cnt[i][2] + cnt[n][0] + cnt[n][1] - cnt[i][1]
	ans = min(ans, val)
out(ans)
