	static int scanInt() throws IOException {

	static String scanString() throws IOException {
		if (tok == null || !tok.hasMoreTokens()) {
			tok = new StringTokenizer(in.readLine());
		}
		return tok.nextToken();
	}
	static String scanLine() throws IOException {
		return in.readLine();
	}


	static BufferedReader in;
	static PrintWriter out;
	static StringTokenizer tok;
	static int spf[];
	public static void main(String[] args) {
		try {
			long startTime = System.currentTimeMillis();

			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
			//in = new BufferedReader(new FileReader("input.txt"));

	private static void solve() throws IOException{
		int test=scanInt();
		sieve();
		while(--test>-1){
			int n=scanInt();
			int k=scanInt();
			int arr[]=new int[n];
			for(int i=0; i<n;++i){
				arr[i]=scanInt();
				arr[i]=transform(arr[i]);
				//out.print(arr[i]+" ");
			}
			Set<Integer> seen=new HashSet<>();
			int seg=1;
			for(int i=0; i<n; ++i){
				if(seen.contains(arr[i])){
					seen.clear();
					++seg;
				}
				seen.add(arr[i]);
			}
			out.println(seg);
			
		}
		
		
		
		
	} 
	
	private static int transform(int x){
		HashMap<Integer, Integer> primeFactors=new HashMap<>();
		while(x!=1){
			if(primeFactors.containsKey(spf[x])){
				int temp= primeFactors.get(spf[x]);
				primeFactors.put(spf[x], ((temp+1)%2));
				
			}
			else
				primeFactors.put(spf[x],1);
			x=x/spf[x];
		}
		int result=1;
		for(HashMap.Entry<Integer, Integer> entry: primeFactors.entrySet()){
			int fac= entry.getKey(); int pow=entry.getValue();
			result*=((int)Math.pow(fac,pow));
		}
		return result;
	}
	
	
	private static void sieve(){
		int Max= (int)10e6+1;
		spf=new int[Max];
		spf[1]=1;
		for(int i=2; i<Max; ++i){
			spf[i]=i;
		}
		for(int i=4; i<Max; i+=2){
			spf[i]=2;
		}
		for(int i=3; i*i<Max; ++i){
			if(spf[i]==i){
				for(int j=i*i; j<Max; j+=i){
					if(spf[j]==j){
						spf[j]=i;
					}
				}
			}
		}
	}
	
	
	
}
