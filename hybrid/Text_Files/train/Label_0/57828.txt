import java.io.*;
import java.util.*;

public class P3 {
	public static void main (String[] args) {
		Scanner sc = new Scanner (System.in);
		PrintStream op = System.out;
		int count = 0;
		long n = sc.nextLong();
		int a = 0, b = 0;
		switch ((int)(n % 3)) {
		case 0: {
			a = 15; b = 42;
			break;
		}
		case 1: {
			a = 7; b = 33;
			break;
		}
		case 2: {
			a = 2; b = 24;
			break;
		}
		default: break;
		}
		for (long i = a, val = b; i <= n; i += val, val += 27)
			count++;
		op.println(count);
		sc.close();
	}
}
