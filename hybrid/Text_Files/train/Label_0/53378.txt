import java.util.*;

public class index {

    public static void main(String[] args) {
        
        Scanner cin = new Scanner(System.in);
        int input = cin.nextInt();

        if (input == 1 || input == 2) {
            System.out.println(-1);
            return;
        }
         
        for(int i = input; i>0; i--) 
        System.out.print(i + " ");
    }
}
