#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    bool f = true;
    string s;
    cin >> s;
    int l = s.length();
    int i, j;
    i = 0;
    j = l - 1;
    while (i < j) {
      if (s[i] == s[j]) {
        i++;
        j--;
      } else {
        f = false;
        break;
      }
    }
    if (f == false) {
      cout << s << endl;
    } else {
      string str = s;
      for (i = 0; i < l - 1; i++) {
        if (s[i] != s[i + 1]) {
          char c = s[i];
          s[i] = s[i + 1];
          s[i + 1] = c;
          break;
        }
      }
      if (str == s) {
        cout << -1 << endl;
      } else
        cout << s << endl;
    }
  }
}
