#include <bits/stdc++.h>
using namespace std;
int n, m;
int aa[110], bb[110], gg[110][110];
int main() {
  cin >> n >> m;
  int t1 = 0, t2 = 0;
  for (int i = 0; i < n; i++) {
    cin >> aa[i];
    if (i != n - 1) {
      t1 = t1 ^ aa[i];
    }
  }
  for (int i = 0; i < m; i++) {
    cin >> bb[i];
    if (i != m - 1) {
      t2 = t2 ^ bb[i];
    }
  }
  t1 ^= bb[m - 1];
  t2 ^= aa[n - 1];
  if (t1 != t2) {
    cout << "NO" << endl;
    return 0;
  } else {
    cout << "YES" << endl;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (i == n - 1) {
        gg[i][j] = bb[j];
      } else if (j == m - 1) {
        gg[i][j] = aa[i];
      } else {
        gg[i][j] = 0;
      }
    }
  }
  gg[n - 1][m - 1] = t1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (j != m - 1) {
        cout << gg[i][j] << " ";
      } else {
        cout << gg[i][j] << endl;
      }
    }
  }
}
