def solve():
	n = int(input())
	s = input()
	t = input()
	diff = list()
	for i in range(n):
		if s[i] != t[i]:
			diff.append(i)
	if len(diff) % 2 != 0:
		print(-1)
	else:
		index = 0
		swap = list()
		diff = sorted(diff, key=lambda index: s[index])
		ans = 0
		while index < len(diff):
			real_pos = diff[index]
			next_real_pos = diff[index + 1]
			if s[real_pos] == s[next_real_pos]:
				ans += 1
				swap.append((real_pos + 1, next_real_pos + 1))
			else:
				ans += 2
				swap.append((real_pos + 1, real_pos + 1))
				swap.append((real_pos + 1, next_real_pos + 1))
			index += 2
		print(ans)
		for tp in swap:
			print(tp[0], end=' ')
			print(tp[1])
solve()
