def main():
	(n, hi) = map(int, input().split())
	(l, res, lo, hi) = ([0], [], 0, -hi)
	(f, p) = (res.append, l.append)
	for _ in range(n):
		(t, d) = map(int, input().split())
		d += t if t > l[-1] else l[-1]
		while lo <= hi and l[lo] <= t:
			lo += 1
		if hi >= lo:
			f('-1')
		else:
			hi += 1
			p(d)
			f(str(d))
	print(' '.join(res))
main()
