def attack(attacks, h):
	(start, end) = (1, h)
	while start <= end:
		mid = (start + end) // 2
		(last, total) = (None, 0)
		for attack in attacks:
			if last:
				total += min(mid, attack - last)
			last = attack
		if last:
			total += mid
		if total >= h:
			end = mid - 1
		else:
			start = mid + 1
	return start
(inp, results) = (int(input()), [])
for _ in range(inp):
	(inp1, inp2) = list(map(int, input().split()))
	inputs = list(map(int, input().split()))
	results.append(attack(inputs, inp2))
print(*results, sep='\n')
