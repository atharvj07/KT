import java.io.*;
import java.util.*;
public class E591 {
public static void main(String[] args) throws IOException {
    input.init(System.in);
    PrintWriter out = new PrintWriter(System.out);
    int n = input.nextInt(), m = input.nextInt();
    char[][] grid = new char[n][m];
    for(int i = 0; i<n; i++) grid[i] = input.next().toCharArray();
    int[][] as = go(n, m, grid, '1'), bs = go(n, m, grid, '2'), cs = go(n, m, grid, '3');
    int min = 999999999;
    for(int i = 0; i<n; i++)
        for(int j = 0; j<m; j++)
            min = Math.min(min, as[i][j] + bs[i][j] + cs[i][j] - 2);
    out.println(min > 1e7 ? -1 : min);
    out.close();
}
static int[][] go(int n, int m, char[][] grid, char start)
{
    int[][] res = new int[n][m];
    for(int[] A : res) Arrays.fill(A, 99999999);
    ArrayDeque<Integer> qi = new ArrayDeque<Integer>(), qj = new ArrayDeque<Integer>();
    for(int i = 0; i<n; i++)
        for(int j = 0; j<m; j++)
        {
            if(grid[i][j] == start)
            {
                res[i][j] = 0;
                qi.add(i);
                qj.add(j);
            }
        }
    int[] di = new int[]{0, 1, 0, -1};
    int[] dj = new int[]{1, 0, -1, 0};
    while(!qi.isEmpty())
    {
        int ati = qi.poll(), atj = qj.poll();
        for(int k = 0; k<4; k++)
        {
            int ni = ati + di[k], nj = atj + dj[k];
            if(ni < 0 || nj < 0 || ni >= n || nj >= m) continue;
            if(res[ni][nj] <= 1 + res[ati][atj] || grid[ni][nj] == '#') continue;
            res[ni][nj] = 1 + res[ati][atj];
            qi.add(ni);
            qj.add(nj);
        }
    }
    int[] min = new int[256];
    Arrays.fill(min,  99999999);
    for(int i = 0; i<n; i++)
        for(int j = 0; j<m; j++)
        {
            min[grid[i][j]] = Math.min(res[i][j],  min[grid[i][j]]);
        }
    for(int i = 0; i<n; i++)
        for(int j = 0; j<m; j++)
        {
            if(grid[i][j] >= '1' && grid[i][j] <= '3') res[i][j] = Math.min(res[i][j], min[grid[i][j]]);
        }
    return res;
}
public static class input {
    static BufferedReader reader;
    static StringTokenizer tokenizer;

    static void init(InputStream input) {
    reader = new BufferedReader(new InputStreamReader(input));
    tokenizer = new StringTokenizer("");
    }

    static String next() throws IOException {
    while (!tokenizer.hasMoreTokens())
        tokenizer = new StringTokenizer(reader.readLine());
    return tokenizer.nextToken();
    }

    static int nextInt() throws IOException {
    return Integer.parseInt(next());
    }

    static double nextDouble() throws IOException {
    return Double.parseDouble(next());
    }

    static long nextLong() throws IOException {
    return Long.parseLong(next());
    }
}
}
