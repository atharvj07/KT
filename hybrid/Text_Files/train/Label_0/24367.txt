#include <cstdio>
#include <cstdint>
#include <vector>
#include <algorithm>

int testcase_ends() {
  size_t N;
  scanf("%zu", &N);

  if (N == 0) return 1;

  double pw;
  scanf("%lf", &pw);

  std::vector<double> vw(N), pf(N), vf(N), th(N);
  for (size_t i=0; i<N; ++i)
    scanf("%lf %lf %lf %lf", &vw[i], &pf[i], &vf[i], &th[i]);

  auto f=[&](double W)->double {
    double res=0;
    res += W*pw;
    for (size_t i=0; i<N; ++i) {
      double th_i=th[i]-W*vw[i];
      if (th_i > 0) {
        res += th_i/vf[i]*pf[i];
      }
    }
    // fprintf(stderr, "f(%f) = %f\n", W, res);
    return res;
  };

  double lb=0, ub=1e10;
  for (int i=0; i<300; ++i) {
    double midl=(2*lb+ub)/3, midu=(lb+2*ub)/3;
    double fl=f(midl), fu=f(midu);
    if (fl < fu) {
      ub = midu;
    } else {
      lb = midl;
    }
  }
  // fprintf(stdout, "%.6f:", lb);
  printf("%.12f\n", f(lb));
  return 0;
}

int main() {
  while (!testcase_ends()) {}
}
