import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String strchislo = scanner.next();
        List<String> spisokstrok = new ArrayList<String>();
        List<String> spisokmaxstrok = new ArrayList<String>();
        for (int i = 0; i < strchislo.length(); i++) {
            for (int j = i; j < strchislo.length(); j++) {
                int ll = 0;
                for (int k = i; k <= j; k++) {
//                    System.out.println(i + " " + j);
                    if ((strchislo.charAt(k) == '4') || (strchislo.charAt(k) == '7')) {
                        ll++;
                    }
                }
                if (ll == j - i + 1) {
                    spisokstrok.add(strchislo.substring(i, j + 1));
                }
            }
        }


        int max = 0;
        int maxln = 0;
        for (int i = 0; i < spisokstrok.size(); i++) {
            int ll = 0;
            for (int j = i; j < spisokstrok.size(); j++) {
                if (spisokstrok.get(i).compareTo(spisokstrok.get(j)) == 0) {
                    ll++;
                }
            }
            if (ll > max) {
                max = ll;
                maxln = i;
            }
        }


        for (int i = 0; i < spisokstrok.size(); i++) {
            int ll = 0;
            for (int j = i; j < spisokstrok.size(); j++) {
                if (spisokstrok.get(i).compareTo(spisokstrok.get(j)) == 0) {
                    ll++;
                }
            }
            if (ll == max) {
                spisokmaxstrok.add(spisokstrok.get(i));
            }
        }
        if (spisokstrok.isEmpty()) {
            System.out.println("-1");
        } else {
            Collections.sort(spisokmaxstrok);
            System.out.println(spisokmaxstrok.get(0));
        }
    }
}
