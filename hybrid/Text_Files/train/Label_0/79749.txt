from sys import stdin, stdout
from heapq import *
n = int(input())
v = list(map(int, input().split()))
t = list(map(int, input().split()))
heap = []
counter = 0
for i in range(n):
	res = 0
	item = counter + v[i]
	heappush(heap, item)
	while heap:
		if heap[0] >= counter + t[i]:
			break
		res += heap[0] - counter
		heappop(heap)
	res += len(heap) * t[i]
	counter += t[i]
	print(str(res) + ' ')
