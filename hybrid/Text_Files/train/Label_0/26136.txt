#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  ios_base::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);
  long long t;
  cin >> t;
  while (t--) {
    long long n, a, b;
    cin >> n >> a >> b;
    string s;
    cin >> s;
    long long now = 0, ans = n * a + (n + 1) * b;
    bool first = true;
    for (long long i = 0; i < n; ++i) {
      if (s[i] == '0') {
        ++now;
      } else {
        if (first) {
          ans += a;
          first = false;
          now = 0;
        } else if (now != 0) {
          ans += min(2 * a, (now - 1) * b);
          now = 0;
        }
        ans += (s[i - 1] == '1' ? b : 2 * b);
      }
    }
    if (now != n) {
      ans += a;
    }
    cout << ans << endl;
  }
  return 0;
}
