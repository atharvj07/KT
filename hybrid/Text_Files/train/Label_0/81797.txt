from sys import stdin, stdout
t = int(stdin.readline())
for w in range(t):
	(n, m, x, y) = [int(s) for s in stdin.readline().split(' ')]
	val = []
	for i in range(n):
		e = stdin.readline()[:-1]
		count = 0
		for j in e:
			if j == '.':
				count += 1
			else:
				if count:
					val.append(count)
				count = 0
		if count:
			val.append(count)
	if 2 * x <= y:
		ans = sum(val) * x
	else:
		ans = 0
		for i in val:
			if i == 1:
				ans += x
			else:
				ans += y * int(i / 2)
				if i % 2 == 1:
					ans += x
	ans = str(int(ans)) + '\n'
	stdout.write(ans)
