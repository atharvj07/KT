def ispow(c):
	x = int(c ** 0.5)
	return x * x == c or (x + 1) * (x + 1) == c or (x + 2) * (x + 2) == c
n = int(input())
a = list(map(int, input().split()))
kb = []
nk = []
for i in range(n):
	if ispow(a[i]):
		if a[i] == 0:
			kb.append(2)
		else:
			kb.append(1)
	else:
		c = int(a[i] ** 0.5)
		x = abs(c * c - a[i])
		y = abs((c + 1) * (c + 1) - a[i])
		nk.append(min(x, y))
if len(kb) > len(nk):
	kb.sort()
	s = 0
	for i in range(len(kb) - n // 2):
		s += kb[i]
	print(s)
elif len(kb) == len(nk):
	print(0)
else:
	nk.sort()
	s = 0
	for i in range(len(nk) - n // 2):
		s += nk[i]
	print(s)
