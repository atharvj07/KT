#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, a;
  long long x = 0, y = 0;
  pair<int, int> p[100009];
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a;
    p[a] = make_pair(i, (n + 1 - i));
  }
  cin >> m;
  while (m--) {
    cin >> a;
    x += p[a].first;
    y += p[a].second;
  }
  cout << x << " " << y;
  return 0;
}
