#include<algorithm>
#include<bitset>
#include <cassert>
#include<iostream>
#include<vector>
using namespace std;
typedef long long lint;
typedef vector<int>vi;
typedef pair<int,int>pii;
#define rep(i,n)for(int i=0;i<(int)(n);++i)


const int N=4000;
typedef bitset<N> bs;
const lint MOD=998244353;

bs read(){
  string s;
  cin>>s;
  int l=s.length();
  bs ret;
  rep(i,l)ret[i]=s[l-i-1]=='1';
  return ret;
}

bs div(bs a, bs b){
  int hi=-1;
  rep(i,N){
    if(b[i]){
      hi=i;
    }
  }
  assert(hi>=0);
  for(int i=N-hi-1;i>=0;--i){
    if(a[i+hi]){
      a^=b<<i;
    }
  }
  return a;
}

bs gcd(bs a,bs b){
  while(b.count()!=0){
    a=div(a,b);
    swap(a,b);
  }
  return a;
}

int main(){
  int n;
  cin>>n;
  bs x=read();
  vector<bs> s(n);
  rep(i,n)s[i]=read();
  bs t=s[n-1];
  rep(i,n-1)
    t=gcd(s[i],t);
  lint ans=0;
  lint cur=1;
  int hi=-1;
  rep(i,N){
    if(t[i]){
      hi=i;
    }
  }
  rep(i,N-hi){
    if(x[i+hi])ans=(ans+cur)%MOD;
    cur=cur*2%MOD;
  }
  bs y = x ^ div(x,t);
  bool lt=false;
  for(int i=N-1;i>=0;--i){
    if(x[i]!=y[i]){
      lt=x[i]<y[i];
      break;
    }
  }
  if(lt)ans=(ans+MOD-1)%MOD;
  ans=(ans+1)%MOD;
  cout<<ans<<endl;
}
