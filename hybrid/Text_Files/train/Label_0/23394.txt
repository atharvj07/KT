#include <bits/stdc++.h>
using namespace std;
int n, q, Size, block, pos[200010];
long long ans;
vector<int> blocks[2050];
int a[200010];
pair<int, int> pp[200010];
inline void init() {
  for (int i = 1; i < (block + 1); i++) {
    int L = Size * (i - 1) + 1;
    int R = min(i * Size, n);
    pp[i] = make_pair(L, R);
  }
  for (int i = 1; i < (block + 1); i++) {
    for (int j = pp[i].first; j < (pp[i].second + 1); j++)
      blocks[i].push_back(j);
  }
}
int main() {
  scanf("%d %d", &n, &q);
  Size = (int)sqrt(n);
  for (int i = 1; i < (n + 1); i++) {
    pos[i] = (i - 1) / Size + 1;
    a[i] = i;
  }
  if (n % Size)
    block = n / Size + 1;
  else
    block = n / Size;
  init();
  ans = 0;
  while (q--) {
    int l, r;
    scanf("%d %d", &l, &r);
    if (l > r) swap(l, r);
    if (l == r) {
      printf("%I64d\n", ans);
      continue;
    }
    int posl = pos[l];
    int posr = pos[r];
    if (posl != posr) {
      for (int i = posl + 1; i < (posr); i++) {
        int tmp = pp[i].second - pp[i].first + 1;
        int tmp2 = (lower_bound(blocks[i].begin(), blocks[i].end(), a[l]) -
                    blocks[i].begin());
        ans += (tmp - tmp2 * 2);
      }
      for (int i = pp[posr].first; i < (r); i++) {
        ans += (a[l] < a[i] ? 1 : -1);
        ans += (a[r] < a[i] ? -1 : 1);
      }
      for (int i = posr - 1; i > (posl); i--) {
        int tmp = pp[i].second - pp[i].first + 1;
        int tmp2 = (lower_bound(blocks[i].begin(), blocks[i].end(), a[r]) -
                    blocks[i].begin());
        ans -= (tmp - tmp2 * 2);
      }
      for (int i = l + 1; i < (pp[posl].second + 1); i++) {
        ans += (a[r] > a[i] ? 1 : -1);
        ans += (a[l] > a[i] ? -1 : 1);
      }
      ans += (a[l] < a[r] ? 1 : -1);
      swap(a[l], a[r]);
      blocks[posl].clear();
      blocks[posr].clear();
      for (int j = pp[posl].first; j < (pp[posl].second + 1); j++)
        blocks[posl].push_back(a[j]);
      sort(blocks[posl].begin(), blocks[posl].end());
      for (int j = pp[posr].first; j < (pp[posr].second + 1); j++)
        blocks[posr].push_back(a[j]);
      sort(blocks[posr].begin(), blocks[posr].end());
    } else {
      for (int i = l + 1; i < (r); i++) {
        ans += (a[r] > a[i] ? 1 : -1);
        ans += (a[l] > a[i] ? -1 : 1);
      }
      ans += (a[l] < a[r] ? 1 : -1);
      swap(a[l], a[r]);
    }
    printf("%I64d\n", ans);
  }
}
