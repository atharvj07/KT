for tc in range(int(input())):
	(x, y) = map(int, input().split())
	s = input()
	flag1 = False
	flag2 = False
	(r, u, d, l) = (s.count('R'), s.count('U'), s.count('D'), s.count('L'))
	if x >= 0 and r >= x or (x <= 0 and l >= abs(x)):
		flag1 = True
	if y >= 0 and u >= y or (y <= 0 and d >= abs(y)):
		flag2 = True
	if flag1 and flag2:
		print('YES')
	else:
		print('NO')
