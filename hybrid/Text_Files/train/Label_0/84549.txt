import io, os
import heapq
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def main(t):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	indexes = [[] for i in range(n)]
	tot = n * k
	for i in range(tot):
		indexes[arr[i] - 1].append(i)
	bound = (n - 1) // (k - 1) + 1
	ans = [[] for i in range(n)]
	visited = [0] * n
	for p in range(k - 1):
		heap = []
		for c in range(n):
			heapq.heappush(heap, [indexes[c][p + 1], indexes[c][p], c])
		rest = bound
		while heap:
			ele = heapq.heappop(heap)
			[end, front, c] = ele
			if visited[c]:
				continue
			visited[c] = 1
			ans[c] = [front, end]
			rest -= 1
			if rest == 0:
				break
	for ele in ans:
		print(ele[0] + 1, ele[1] + 1)
T = 1
t = 1
while t <= T:
	main(t)
	t += 1
