import sys

def check(grid):
	width = len(grid)
	expected_diag = grid[0][0]
	expected_non_diag = grid[0][1]
	if expected_diag == expected_non_diag:
		return False
	for i in range(width):
		for j in range(width):
			if i == j or j == width - 1 - i:
				if not grid[i][j] == expected_diag:
					return False
			elif not grid[i][j] == expected_non_diag:
				return False
	return True
n = int(sys.stdin.readline())
grid = []
for i in range(n):
	grid.append(sys.stdin.readline().strip())
if check(grid):
	print('YES')
else:
	print('NO')
