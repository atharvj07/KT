#include <bits/stdc++.h>
using namespace std;
const int maxn = 2111;
const int INF = 1 << 30;
int num[maxn][maxn], cnt;
int dis[maxn][maxn];
int f[maxn * maxn];
char ss[maxn][maxn];
bool flag = false;
int dx[] = {-1, 0, 1, 0};
int dy[] = {0, 1, 0, -1};
int convert(char ch) {
  if (ch == '^') return 0;
  if (ch == '>') return 1;
  if (ch == 'v') return 2;
  return 3;
}
int find(int n) {
  if (f[n] == n) return n;
  return f[n] = find(f[n]);
}
int dfs(int x, int y, int dep) {
  if (dep > cnt) {
    flag = true;
    return -1;
  }
  if (ss[x][y] == '#') {
    return 0;
  }
  if (dis[x][y]) return dis[x][y];
  int k = convert(ss[x][y]);
  int nx = x + dx[k];
  int ny = y + dy[k];
  if (ss[nx][ny] ^ '#') f[num[x][y]] = num[nx][ny];
  dis[x][y] = 1 + dfs(nx, ny, dep + 1);
  return dis[x][y];
}
int main() {
  ios::sync_with_stdio(false);
  int i, j, n, m;
  cin >> n >> m;
  for (i = 1; i <= n; ++i) {
    cin >> (ss[i] + 1);
    for (j = 1; j <= m; ++j) {
      num[i][j] = ++cnt;
      f[cnt] = cnt;
    }
  }
  int ma = 0, g = -1;
  for (i = 1; i <= n; ++i) {
    for (j = 1; j <= m; ++j) {
      dfs(i, j, 0);
      if (flag) {
        puts("-1");
        return 0;
      }
      if (dis[i][j] > ma) {
        ma = dis[i][j];
        g = find(num[i][j]);
      }
    }
  }
  int c = 0;
  int tag = 0, s = 0;
  for (i = 1; i <= n; ++i) {
    for (j = 1; j <= m; ++j) {
      if (dis[i][j] == ma) {
        ++c;
        if (find(num[i][j]) != g) {
          tag = 1;
        }
      }
      if (dis[i][j] < ma) {
        s = max(s, dis[i][j]);
      }
    }
  }
  int ans = 0;
  if (c == 1) {
    ans = s + ma;
  } else {
    if (tag == 0)
      ans = ma * 2 - 1;
    else
      ans = ma * 2;
  }
  cout << ans << endl;
  return 0;
}
