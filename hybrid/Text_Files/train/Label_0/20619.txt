#include <bits/stdc++.h>
#pragma GCC optimize(3)
using namespace std;
const int maxn = 1010;
const long long P = 1000000016000000063ll;
int n, m, q;
bitset<maxn> bit[35], A[1 << 15];
int len, val[1 << 15];
unordered_map<bitset<maxn>, int> mpA, mpB;
int main() {
  cin >> n >> m >> q;
  for (int i = (1), iend = (m); i <= iend; ++i) {
    int k, x;
    cin >> k;
    while (k--) scanf("%d", &x), bit[i][x] = 1;
  }
  int m1 = m / 3, m2 = m - m1;
  bitset<maxn> cur;
  for (int S = (0), Send = ((1 << m1) - 1); S <= Send; ++S) {
    int ct = 0;
    cur.reset();
    for (int i = (1), iend = (m1); i <= iend; ++i)
      if (S >> (i - 1) & 1) {
        ct++, cur ^= bit[i];
      }
    if (mpA.find(cur) == mpA.end())
      mpA[cur] = len, val[len] = ct, A[len++] = cur;
    else if (val[mpA[cur]] > ct)
      val[mpA[cur]] = ct;
  }
  for (int S = (0), Send = ((1 << m2) - 1); S <= Send; ++S) {
    int ct = 0;
    cur.reset();
    for (int i = (m1 + 1), iend = (m); i <= iend; ++i)
      if (S >> (i - m1 - 1) & 1) {
        ct++, cur ^= bit[i];
      }
    if (mpB.find(cur) == mpB.end() || mpB[cur] > ct) mpB[cur] = ct;
  }
  while (q--) {
    cur.reset();
    int ans = 1e9;
    int k, x;
    cin >> k;
    while (k--) scanf("%d", &x), cur[x] = 1;
    for (int i = (0), iend = (len - 1); i <= iend; ++i) {
      bitset<maxn> ano = cur ^ A[i];
      if (mpB.find(ano) != mpB.end()) ans = min(ans, val[i] + mpB[ano]);
    }
    cout << (ans > m ? -1 : ans) << endl;
  }
  return 0;
}
