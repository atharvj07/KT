#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const long long LINF = 0x3f3f3f3f3f3f3f3f;
int n, k;
string s;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> k >> s;
    set<int> pos, neg;
    vector<pair<int, int> > swaps;
    for (int i = 0; i < n; i++)
      if (s[i] == '(')
        pos.insert(i);
      else
        neg.insert(i);
    auto rev = [&](int l, int r) {
      swaps.push_back({l + 1, r + 1});
      for (int i = l; i <= r; i++)
        if (s[i] == '(')
          pos.erase(i);
        else
          neg.erase(i);
      reverse(s.begin() + l, s.begin() + r + 1);
      for (int i = l; i <= r; i++)
        if (s[i] == '(')
          pos.insert(i);
        else
          neg.insert(i);
    };
    int i = 0;
    int l, r;
    while (k > 1) {
      l = i;
      r = *pos.lower_bound(i);
      if (l != r) rev(l, r);
      i++;
      l = i;
      r = *neg.lower_bound(i);
      if (l != r) rev(l, r);
      i++;
      k--;
    }
    while (true) {
      auto it = pos.lower_bound(i);
      if (it == pos.end()) break;
      l = i;
      r = *it;
      if (l != r) rev(l, r);
      i++;
    }
    while (true) {
      auto it = neg.lower_bound(i);
      if (it == neg.end()) break;
      l = i;
      r = *it;
      if (l != r) rev(l, r);
      i++;
    }
    cout << swaps.size() << '\n';
    for (auto it : swaps) cout << it.first << " " << it.second << '\n';
  }
  exit(0);
}
