import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class mainB {
    public static Scanner enter = new Scanner(System.in);
    public static PrintWriter out = new PrintWriter(System.out);
    public static void main(String[] args) {
        int n=enter.nextInt();
        int r=enter.nextInt();
        int[] mass=new int[n+1];
        for (int i = 1; i <mass.length ; i++) {
            mass[i]=enter.nextInt();
        }
        int last=1-r;
        int ans=0;
        int c=1;
        while(true){
            for (int i = Math.min(last+2*r-1,n); i >Math.max(last,0) ; i--) {
                if(mass[i]==1){
                  ans++;
                  c=0;
                  last=i;
                  break;
                }
            }
            if(c==1) {
                out.println(-1);
                out.close();
                return;
            }
            c=1;
            if(last+r>=n+1){
                break;
            }
        }
        out.println(ans);
        out.close();
    }
}
