#include <bits/stdc++.h>
using namespace std;
const int maxn = 103010;
map<int, int> mpx;
struct column {
  int x;
  int y, y1;
} col[maxn];
struct line {
  int x, x1;
  int y;
} lin[maxn];
struct event {
  int x, y, t;
} e[2 * maxn];
bool cmp(line a, line b) { return a.y < b.y; }
bool cmp2(event a, event b) {
  if (a.y == b.y) return a.t < b.t;
  return a.y < b.y;
}
vector<int> x;
int bit[maxn];
void update(int idx, int val) {
  while (idx <= maxn) {
    bit[idx] += val;
    idx += idx & (-idx);
  }
}
int query(int idx) {
  int sum = 0;
  while (idx > 0) {
    sum += bit[idx];
    idx -= idx & (-idx);
  }
  return sum;
}
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  for (int i = 0; i < n; i++) {
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    col[i].x = a;
    col[i].y = b;
    col[i].y1 = b + c;
  }
  for (int i = 0; i < m; i++) {
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    lin[i].x = a;
    lin[i].y = b;
    lin[i].x1 = a + c;
  }
  int st = 0, ed = 100000000;
  while (ed - st > 0) {
    int ok = 0;
    int qev = 0;
    int md = (st + ed + 1) / 2;
    x.clear();
    mpx.clear();
    memset(bit, 0, sizeof(bit));
    for (int i = 0; i < n; i++) {
      int a = col[i].y;
      int b = col[i].y1;
      int c = col[i].x;
      if (a + md > b - md) continue;
      a += md;
      b -= md;
      x.push_back(c);
      e[qev].x = c;
      e[qev].y = a;
      e[qev].t = 0;
      qev++;
      e[qev].x = c;
      e[qev].y = b + 1;
      e[qev].t = 1;
      qev++;
    }
    for (int i = 0; i < m; i++) {
      if (lin[i].x + md > lin[i].x1 - md) continue;
      x.push_back(lin[i].x + md);
      x.push_back(lin[i].x1 - md);
    }
    sort(x.begin(), x.end());
    vector<int>::iterator it = unique(x.begin(), x.end());
    x.resize(distance(x.begin(), it));
    sort(lin, lin + m, cmp);
    sort(e, e + qev, cmp2);
    for (int i = 0; i < x.size(); i++) {
      mpx[x[i]] = i + 1;
    }
    int siz = x.size();
    if (siz != 0) {
      int i = 0, j = 0;
      for (; i < qev && j < m;) {
        while (lin[j].x + md > lin[j].x1 - md && j < m) j++;
        if (e[i].y <= lin[j].y) {
          if (e[i].t == 1) {
            update(mpx[e[i].x], -1);
          } else {
            update(mpx[e[i].x], 1);
          }
          i++;
        } else {
          int ver = query(mpx[lin[j].x1 - md]) - query(mpx[lin[j].x + md] - 1);
          if (ver > 0) ok = 1;
          j++;
        }
      }
      while (i < qev) {
        if (e[i].t == 1) {
          update(mpx[e[i].x], -1);
        } else {
          update(mpx[e[i].x], 1);
        }
        i++;
      }
      while (j < m) {
        int ver = query(mpx[lin[j].x1 - md]) - query(mpx[lin[j].x + md] - 1);
        if (ver > 0) ok = 1;
        j++;
      }
    }
    if (ok)
      st = md;
    else
      ed = md - 1;
  }
  printf("%d\n", st);
}
