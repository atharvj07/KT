from sys import stdin
N = int(stdin.readline())
sherlock = list(map(int, stdin.readline().strip()))
moriarty = list(map(int, stdin.readline().strip()))
mcounts = [0] * 10
for m in moriarty:
	mcounts[m] += 1

def minflicks(counts):
	flicks = 0
	for s in sherlock:
		si = s
		while counts[si % 10] == 0:
			si += 1
		if si % 10 < s:
			flicks += 1
		counts[si % 10] -= 1
	return flicks

def maxflicks(counts):
	flicks = 0
	for s in sherlock:
		si = s + 1
		while counts[si % 10] == 0:
			si += 1
		if si % 10 > s:
			flicks += 1
		counts[si % 10] -= 1
	return flicks
print(minflicks(mcounts[:]))
print(maxflicks(mcounts[:]))
