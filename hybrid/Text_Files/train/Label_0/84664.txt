t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	ones = [[] for __ in range(2)]
	ans = [0] * n
	cnt = 0
	for i in range(n):
		x = int(s[i]) ^ 1
		if not ones[x]:
			cnt += 1
			ans[i] = cnt
		else:
			ans[i] = ones[x][-1]
			ones[x].pop()
		ones[x ^ 1].append(ans[i])
	print(cnt)
	print(*ans)
