import java.io.*;
import java.util.*;


public class E {
	
	public static void main(String[] args) {
		new E();
	}
	long mod = 1000000007;
	public E() {
		Scanner s = new Scanner();
		
		int n = s.nextInt();
		
		boolean[] person = new boolean[2*n]; // false if no out degree
		
		ArrayList<Integer>[] undirGraph = new ArrayList[2*n];
		ArrayList<Integer>[] revGraph = new ArrayList[2*n]; // reverse edges. From seats to people wanting them.
		for (int i = 0; i < 2*n; i++) {
			revGraph[i] = new ArrayList<Integer>();
			undirGraph[i] = new ArrayList<Integer>();
		}
		boolean[] loops = new boolean[2*n];
		
		for (int i = 0; i < n; i++) {
			int from = s.nextInt()-1;
			int to = s.nextInt()-1;
			
			
			if (from == to) {
				//self loop. Useless component
				loops[from] = true;
			}
			revGraph[to].add(from);

			undirGraph[to].add(from);
			undirGraph[from].add(to);
			person[from] = true;
		}
		
		boolean[] visited = new boolean[2*n];
		
		long total = 1;
		
		//Eliminate all loop components
		for (int i = 0; i < 2*n; i++) {
			if (visited[i] || !loops[i]) {	
				continue;//part of already explored component or not a loop.
			}
			
			Queue<Integer> q = new PriorityQueue<Integer>();
			q.add(i);
			
			while (!q.isEmpty()) {
				int cur = q.poll();
				visited[cur] = true; 
				ArrayList<Integer> successors = undirGraph[cur];
				for (int next : successors) {
					if (!visited[next]) {
						q.add(next);	
					}
				} //no cycles will be found
			}
			//No movement possible in this component because of self-loop
		}

		//Get all non-cycle component sizes
		for (int i = 0; i < 2*n; i++) {
			int componentSize = 0;
			if (person[i] || visited[i]) {	
				continue;//only want to start at endpoint, search backwards. Will never find cycle
			}
			Queue<Integer> q = new PriorityQueue<Integer>();
			q.add(i);
			
			while (!q.isEmpty()) {
				int cur = q.poll();
				componentSize++;
				visited[cur] = true; 
				ArrayList<Integer> successors = revGraph[cur];
				for (int next : successors) {
					q.add(next);
				} //no cycles will be found
			}
			
			total *= componentSize;
			total %= mod;
		}
		
		//Rest of components have cycles. Each has multiple of 2
		for (int i = 0; i < 2*n; i++) {
			if (visited[i]) {	
				continue;//part of already explored component
			}
			
			Queue<Integer> q = new PriorityQueue<Integer>();
			q.add(i);
			
			while (!q.isEmpty()) {
				int cur = q.poll();
				visited[cur] = true; 
				ArrayList<Integer> successors = undirGraph[cur];
				for (int next : successors) {
					if (!visited[next]) {
						q.add(next);	
					}
				} //no cycles will be found
			}
			
			total *= 2; //a cycle component
			total %= mod;
		}
		
		System.out.println(total);
	}
	
	public static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		public Scanner(Reader in) {
			br = new BufferedReader(in);
		}

		public Scanner() {
			this(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}
		String readNextLine() {
			String str = "";
			try {
				str = br.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}
			return str;
		}
	}
}
