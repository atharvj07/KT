from collections import deque
g = [[] for y in range(int(200000.0 + 5))]
q = int(input())
while q > 0:
	q -= 1
	(n, m, a, b, c) = map(int, input().split())

	def BFS(a):
		List = []
		dist = [-1] * (n + 1)
		dist[a] = 0
		List = deque([a])
		while List:
			u = List.popleft()
			for v in g[u]:
				if dist[v] < 0:
					dist[v] = dist[u] + 1
					List.append(v)
		return dist
	p = list(map(int, input().split()))
	p.sort()
	p.insert(0, 0)
	for i in range(1, n + 1):
		g[i].clear()
	for i in range(1, m + 1):
		p[i] = p[i] + p[i - 1]
	for i in range(1, m + 1):
		(u, v) = map(int, input().split())
		g[u].append(v)
		g[v].append(u)
	dist1 = []
	dist2 = []
	dist3 = []
	dist1.extend(BFS(a))
	dist2.extend(BFS(b))
	dist3.extend(BFS(c))
	res = int(1e+17)
	for i in range(1, n + 1):
		if dist1[i] + dist2[i] + dist3[i] > m:
			continue
		res = min(res, p[dist2[i]] + p[dist1[i] + dist2[i] + dist3[i]])
	print(res)
