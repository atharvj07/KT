from itertools import permutations
dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))
per = permutations(dirs)
(r, c) = map(int, input().split())
grid = [input() for _ in range(r)]
path = input()
pos_start = [i for (i, x) in enumerate(grid) if 'S' in x][0]
start = (pos_start, grid[pos_start].index('S'))
pos_end = [i for (i, x) in enumerate(grid) if 'E' in x][0]
end = (pos_end, grid[pos_end].index('E'))
total = 0
for p in per:
	(x, y) = start
	for move in path:
		(dir_x, dir_y) = p[int(move)]
		x += dir_x
		y += dir_y
		if x < 0 or y < 0 or x >= r or (y >= c) or (grid[x][y] == '#'):
			break
		if grid[x][y] == 'E':
			total += 1
			break
print(total)
