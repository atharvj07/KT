i = m = n = int(input())
a = [*map(int, input().split())]
s = [0] * n
r = [range(i % a[i], n, a[i]) for i in range(n)]
while m:
	i = (i - 1) % n
	if s[i] == 0:
		if all((a[j] <= a[i] or s[j] == 'A' for j in r[i])):
			s[i] = 'B'
			m -= 1
		if any((a[j] > a[i] and s[j] == 'B' for j in r[i])):
			s[i] = 'A'
			m -= 1
print(''.join(s))
