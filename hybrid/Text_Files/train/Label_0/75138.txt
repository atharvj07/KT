import math

def prime_factors(a):
	s = int(math.sqrt(a)) + 1
	pfs = {}
	for i in range(2, s + 1):
		if i > a:
			break
		ic = 0
		while a % i == 0:
			a //= i
			ic += 1
		if ic > 0:
			pfs[i] = ic
	if a > 1:
		pfs[a] = 1
	return pfs
t = int(input())
results = []
for i in range(t):
	pq = input().split()
	p = int(pq[0])
	q = int(pq[1])
	if p % q != 0:
		results.append(p)
	else:
		result = 1
		pfs = prime_factors(q)
		r = p // q
		pfs_2 = {}
		for k in pfs.keys():
			ic = 0
			while r % k == 0:
				r //= k
				ic += 1
			if ic > 0:
				pfs_2[k] = ic
		for (k, v1) in pfs.items():
			if k in pfs_2:
				v2 = pfs_2[k]
				prod = p // k ** (v2 + 1)
			else:
				prod = p // k
			if prod > result:
				result = prod
		results.append(result)
for result in results:
	print(result)
