import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
l = [0] * (n + 1)
for i in range(m):
	(u, v) = sorted(map(int, input().split()))
	l[u] += 1
c = 0
for i in range(1, n + 1):
	if l[i] == 0:
		c += 1
z = []
q = int(input())
for i in range(q):
	s = input().split()
	if s[0] == '1':
		(u, v) = sorted(map(int, s[1:]))
		l[u] += 1
		if l[u] == 1:
			c -= 1
	if s[0] == '2':
		(u, v) = sorted(map(int, s[1:]))
		l[u] -= 1
		if l[u] == 0:
			c += 1
	if s[0] == '3':
		z.append(str(c))
print('\n'.join(z))
