from sys import stdin, stdout
import heapq
from collections import defaultdict
import math
import bisect
import os

def main():
	t = int(stdin.readline())
	for _ in range(t):
		n = int(stdin.readline())
		arr = list(map(int, stdin.readline().split()))
		cur = 0
		dp_plus = [0 for _ in range(n)]
		dp_minus = [0 for _ in range(n)]
		ans = 0
		for (i, x) in enumerate(arr):
			cur += dp_minus[i]
			if cur == 0 and x == 1:
				cur += dp_plus[i]
				continue
			bt = x + cur - 1
			if bt > 0:
				ans += bt
				if i + 2 < n:
					dp_minus[i + 2] -= 1
			else:
				if x >= 2 and i + 2 < n:
					dp_minus[i + 2] -= 1
				if bt < 0 and i + 1 < n:
					dp_minus[i + 1] += bt
					dp_plus[i + 1] -= bt
			if i + x < n and (not (x == 1 and bt < 0)):
				dp_plus[i + x] += 1
			cur += dp_plus[i]
		stdout.write(str(ans) + '\n')
main()
