for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	nodes = dict()
	currentNode = 0
	nodes[1] = [0, set()]
	for i in range(len(arr) - 1):
		if arr[i + 1] > arr[i]:
			nodes[arr[i + 1]] = [nodes[arr[currentNode]][0] + 1, set()]
			nodes[arr[currentNode]][1].add(arr[i])
		else:
			currentNode += 1
			nodes[arr[i + 1]] = [nodes[arr[currentNode]][0] + 1, set()]
			nodes[arr[currentNode]][1].add(arr[i])
	ans = 0
	for node in nodes:
		if nodes[node][0] > ans:
			ans = nodes[node][0]
	print(ans)
