#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 150;
pair<int, int> in[N];
void solve() {
  int n;
  long long r;
  scanf("%d %lld ", &n, &r);
  for (int i = 0; i < n; ++i) {
    scanf("%d  ", &in[i].first);
    in[i].first *= 2;
  }
  for (int i = 0; i < n; ++i) {
    scanf("%d  ", &in[i].second);
    if (in[i].first > 2LL * in[i].second) {
      printf("-1\n");
      return;
    }
  }
  vector<long long> out;
  long long ret = 0;
  const int lim = 1e5;
  long long curTime = 0, remTime = 0;
  for (int i = 0; i < n; ++i) {
    long long val = min(remTime, (long long)in[i].second);
    val = min(val, (long long)(in[i].first + 1) / 2);
    remTime -= val;
    in[i].second -= val;
    in[i].first -= val * 2;
    if (in[i].first > 0) {
      if (in[i].first <= in[i].second) {
        curTime += in[i].first;
      } else {
        long long t = in[i].first - in[i].second;
        curTime += in[i].second - t;
        long long v = in[i].second - t;
        in[i].second -= v;
        in[i].first -= v;
        long long cc = (t + r - 1) / r;
        ret += cc;
        if (out.size() + cc <= lim) {
          long long tmp = curTime;
          for (int i = 0; i < cc; ++i) {
            out.push_back(tmp);
            tmp += r;
          }
        }
        curTime += (cc - 1) * r;
        in[i].second -= (cc - 1) * r;
        in[i].first -= (cc - 1) * r * 2;
        long long x = (in[i].first + 1) / 2;
        assert(remTime == 0);
        remTime = r - x;
        in[i].first -= x * 2;
        curTime += r;
      }
    }
    if (in[i].first < 0) {
      assert(in[i].first == -1);
      --in[i + 1].first;
    }
  }
  printf("%lld\n", ret);
  if (ret <= lim) {
    for (long long num : out) {
      printf("%lld ", num);
    }
    putchar('\n');
  }
}
int main() {
  solve();
  return 0;
}
