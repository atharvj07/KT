	boolean btwn(int i, int j, int k) {
		return Math.min(c[i][0], c[k][0]) <= c[j][0] && c[j][0] <= Math.max(c[i][0], c[k][0]) &&
			   Math.min(c[i][1], c[k][1]) <= c[j][1] && c[j][1] <= Math.max(c[i][1], c[k][1]) &&
			   Math.min(c[i][2], c[k][2]) <= c[j][2] && c[j][2] <= Math.max(c[i][2], c[k][2]);
	}
	
	void solve() {
		int n = nextInt();
		c = new int[n][3];
		for (int i = 0; i < n; ++i)
			for (int j = 0; j < 3; ++j)
				c[i][j] = nextInt();
		HashSet<Integer> l = new HashSet<>();
		for (int i = 0; i < n; ++i)
			l.add(i);
		for (int i = 0; i < n; ++i) {
			if (!l.contains(i))
				continue;
			int j = -1;
			for (int x : l)
				if (x != i) {
					j = x;
					break;
				}
			for (int x : l) {
				if (x != i && x != j && btwn(i, x, j))
					j = x;
			}
			l.remove(i); l.remove(j);
			println((i+1) + " " + (j+1));
		}
	}

	void shuffle(long[] a) {
		for (int i = a.length - 1; i >= 0; i--) { int j = (int) (Math.random() * (i + 1)); a[i] ^= a[j] ^ (a[j] = a[i]); } }
	void shuffle(int[] a) {
		for (int i = a.length - 1; i >= 0; i--) { int j = (int) (Math.random() * (i + 1)); a[i] ^= a[j] ^ (a[j] = a[i]); } }

		if (n <= 1) return false; if (n <= 3) return true; if (n % 2 == 0 || n % 3 == 0) return false;
