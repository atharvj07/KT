def solve():
	n = int(input())
	sm = []
	bg = []
	for i in range(n):
		(a, b) = map(int, input().split())
		if a < b:
			sm.append((b, i + 1))
		else:
			bg.append((b, i + 1))
	if len(sm) > len(bg):
		sm.sort(reverse=True)
		print(len(sm))
		for i in range(len(sm)):
			print(sm[i][1], end=' ')
	else:
		print(len(bg))
		bg.sort()
		for i in range(len(bg)):
			print(bg[i][1], end=' ')

def main():
	solve()
main()
