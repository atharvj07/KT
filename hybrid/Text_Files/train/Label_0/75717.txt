import sys
input = sys.stdin.readline
from collections import deque

def solve(n, k, adj):
	ver = n
	indeg = [len(adj[z]) for z in range(n + 1)]
	bfs = deque()
	for i in range(1, n + 1):
		if indeg[i] == 1:
			bfs.append(i)
	bfs.append(-1)
	while k:
		if ver < 3:
			return 0
		ver -= len(bfs) - 1
		while bfs[0] != -1:
			node = bfs.popleft()
			for each in adj[node]:
				indeg[each] -= 1
				if indeg[each] == 1:
					bfs.append(each)
		bfs.popleft()
		bfs.append(-1)
		k -= 1
	return ver
t = int(input())
for _ in range(t):
	emp = input()
	(n, k) = map(int, input().split())
	adj = [[] for _ in range(n + 1)]
	for _ in range(n - 1):
		(u, v) = map(int, input().split())
		adj[u].append(v)
		adj[v].append(u)
	print(solve(n, k, adj))
