import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;

public class problem_0711D {
	public static void main(String[] args) throws IOException{
		BufferedReader br = new BufferedReader
				(new InputStreamReader(System.in));
		String[] line; 
		
		line = br.readLine().split(" ");
		int N = Integer.parseInt(line[0]);
		
		line = br.readLine().split(" ");
		nexts = new ArrayList[N];
		for(int n=0;n<N;n++){
			nexts[n] = new ArrayList<>(1);
			nexts[n].add(Integer.parseInt(line[n])-1);
		}
		
		S = new ArrayDeque<>(N);
		sccSizes = new int[N];
		dfsNum = new int[N]; Arrays.fill(dfsNum, UNVISITED);
		dfsLow = new int[N]; Arrays.fill(dfsLow, UNVISITED);
		visited = new boolean[N];
		dfsNumCounter = numSCC = 0;
		for(int n=0;n<N;n++){
			if(dfsNum[n] == UNVISITED){
				tarjanSCC(n);
			}
		}
		
		// get the number of non-connected (nc) roads
		int nc = 0;
		for(int i : sccSizes){
			if(i == 1) continue;
			nc += i;
		}
		nc = N - nc;
		
		// calculate the number of flipsets that don't matter
		long amtSets = 1;
		for(int i = 0; i < nc; i++){
			amtSets <<= 1;
			amtSets %= 1_000_000_007;
		}
		
		// for each cycle, multiply amtSets by 2^lencycle - 2
		for(int size : sccSizes){
			if(size == 0) break;
			if(size == 1) continue;
			long factor = 1;
			for(int i = 0; i < size; i++){
				factor <<= 1;
				factor %= 1_000_000_007;
			}
			amtSets *= factor-2;
			amtSets %= 1_000_000_007;
		}
		
		System.out.println(amtSets);
	}
	
	static int[] sccSizes;
	static int numSCC = 0;
	static final int UNVISITED = Integer.MAX_VALUE;
	static int dfsNumCounter;
	static ArrayList<Integer>[] nexts;
	static int[] dfsNum, dfsLow;
	static boolean[] visited;
	static ArrayDeque<Integer> S;
	static void tarjanSCC(int u){
		dfsLow[u] = dfsNum[u] = dfsNumCounter++;
		S.push(u);
		visited[u] = true;
		for(int j=0;j<(int)nexts[u].size();j++){
			int v = nexts[u].get(j).intValue();
			if(dfsNum[v] == UNVISITED) tarjanSCC(v);
			if(visited[v]) dfsLow[u] = Math.min(dfsLow[u], dfsLow[v]);
		}
		if(dfsLow[u] == dfsNum[u]){
//			System.out.print("SCC "+(++numSCC)+":");
			numSCC++;
			while(true){
				int v = S.pop().intValue();
				visited[v] = false;
				sccSizes[numSCC-1]++;
//				System.out.print(" "+v);
				if(u == v) break;
			}
//			System.out.println();
		}
	}
}
