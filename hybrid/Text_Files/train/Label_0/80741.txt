t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	arr1 = []
	d = [0] * (n + 1)
	for i in range(n - 1):
		(a, b) = map(int, input().split())
		(a, b) = (a - 1, b - 1)
		d[a] += 1
		d[b] += 1
		if d[a] > 1:
			arr1 += [arr[a]]
		if d[b] > 1:
			arr1 += [arr[b]]
	arr1 = sorted(arr1)[::-1]
	s = sum(arr)
	result = [str(s)]
	for x in arr1:
		s += x
		result += [str(s)]
	print(' '.join(result))
