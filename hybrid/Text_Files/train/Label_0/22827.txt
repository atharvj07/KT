#include <bits/stdc++.h>
using namespace std;
long long a[600005], b[600005];
pair<long long, int> p1[600005], p2[600005];
int res[200005];
int m, n;
long long solve(int mid) {
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += abs(a[i] - b[mid + i - 1]);
  }
  return ans;
}
int main() {
  scanf("%d%d", &m, &n);
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    p1[i] = make_pair(a[i] + m, i);
  }
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &b[i]);
    p2[i] = make_pair(b[i], i);
  }
  sort(p1 + 1, p1 + 1 + n);
  sort(p2 + 1, p2 + 1 + n);
  for (int i = n + 1; i <= n + n + n; i++) {
    p2[i] = make_pair(p2[i - n].first + m, p2[i - n].second);
  }
  for (int i = 1; i <= n; i++) {
    a[i] = p1[i].first;
  }
  for (int i = 1; i <= n + n + n; i++) {
    b[i] = p2[i].first;
  }
  int lo = 1, hi = n + n + 1, j;
  long long ans = 1e18;
  while (hi - lo > 10) {
    int mid1 = lo + (hi - lo) / 3;
    int mid2 = lo + 2 * (hi - lo) / 3;
    long long val1 = solve(mid1);
    long long val2 = solve(mid2);
    if (val1 < val2) {
      if (ans > val1) {
        ans = val1;
        j = mid1;
      }
      hi = mid2 - 1;
    } else {
      if (ans > val2) {
        ans = val2;
        j = mid2;
      }
      lo = mid1 + 1;
    }
  }
  for (int i = lo; i <= hi; i++) {
    long long val = solve(i);
    if (ans > val) {
      ans = val;
      j = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    res[p1[i].second] = p2[j + i - 1].second;
  }
  printf("%lld\n", ans);
  for (int i = 1; i <= n; i++) printf("%d ", res[i]);
}
