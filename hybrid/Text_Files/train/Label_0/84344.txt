n = int(input())
ch = input()
tot = 0
for i in range(len(ch) - 1):
	for j in range(i + 1, len(ch)):
		s = ch[i:j + 1]
		pos = [0, 0]
		for x in s:
			if x == 'R':
				pos[0] += 1
			if x == 'L':
				pos[0] += -1
			if x == 'U':
				pos[1] += 1
			if x == 'D':
				pos[1] += -1
		if pos[0] == 0 and pos[1] == 0:
			tot += 1
print(tot)
