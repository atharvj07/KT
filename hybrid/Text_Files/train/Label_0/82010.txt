from collections import Counter
for _ in range(int(input())):
	(n, k, d) = map(int, input().split())
	arr = list(map(int, input().split()))
	ans = len(set(arr[:d]))
	if n == d:
		print(ans)
	else:
		q = Counter(arr[:d])
		cnt = 0
		m = ans
		for i in range(d, n):
			if arr[i] not in q:
				q[arr[i]] = 1
				ans += 1
			elif q[arr[i]] > 0:
				q[arr[i]] += 1
			elif q[arr[i]] == 0:
				q[arr[i]] = 1
				ans += 1
			q[arr[cnt]] -= 1
			if q[arr[cnt]] == 0:
				ans -= 1
			cnt += 1
			m = min(m, ans)
		print(m)
