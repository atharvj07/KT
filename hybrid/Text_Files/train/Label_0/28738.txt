#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1e9 + 7;
const int N = 1e5 + 5;
int n;
bool used[N], checked[N];
vector<int> can, p, t, d[N];
vector<pair<int, int> > sol;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i = 2; i <= n; i++) {
    if (checked[i]) continue;
    if (2 * i > n) break;
    p.push_back(i);
    for (int j = i; j <= n; j += i) {
      checked[j] = true;
      d[i].push_back(j);
    }
  }
  for (int i = 1; i < (int)p.size(); i++) {
    int cnt = 0;
    for (auto e : d[p[i]]) {
      if (!used[e]) {
        cnt++;
        can.push_back(e);
      }
    }
    int ind = 0;
    int nxt = 1;
    bool odd = false;
    if (cnt % 2 == 1) {
      for (int i = 0; i < can.size(); i++) {
        if (i == 1) continue;
        t.push_back(can[i]);
      }
    } else {
      for (auto e : can) t.push_back(e);
    }
    for (; ind < (int)t.size(); ind += 2) {
      if (ind + 1 < (int)t.size()) {
        sol.push_back({t[ind], t[ind + 1]});
        used[t[ind]] = true;
        used[t[ind + 1]] = true;
      }
    }
    t.clear();
    can.clear();
  }
  for (int i = 2; i <= n; i += 2) {
    if (!used[i]) can.push_back(i);
  }
  int ind = 0;
  for (; ind < (int)can.size(); ind += 2) {
    if (ind + 1 < (int)can.size()) {
      sol.push_back({can[ind], can[ind + 1]});
    }
  }
  cout << (int)sol.size() << "\n";
  for (auto e : sol) {
    cout << e.first << " " << e.second << "\n";
  }
  return 0;
}
