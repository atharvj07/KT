def isLucky(number, lucky, memo):
	if number in memo:
		return memo[number]
	if number == lucky:
		memo[number] = True
		return True
	elif number < lucky:
		memo[number] = False
		return False
	else:
		if str(lucky) in str(number):
			memo[number] = True
			return True
		memo[number] = isLucky(number - lucky, lucky, memo)
		return isLucky(number - lucky, lucky, memo)
t = int(input())
for _ in range(t):
	(q, d) = [int(x) for x in input().split()]
	numbers = [int(x) for x in input().split()]
	memo = {}
	for num in numbers:
		if num >= d * 10:
			print('YES')
			continue
		elif isLucky(num, d, memo):
			print('YES')
		else:
			print('NO')
