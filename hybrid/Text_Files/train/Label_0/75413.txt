import sys
count = 0
lst_input = []
for line in sys.stdin:
	if count == 0:
		count = 1
		continue
	lst_input.append(line)
lst = []
for i in range(0, len(lst_input), 3):
	lst.append(lst_input[i:i + 3])
for i in lst:
	a = i[1]
	b = i[2]
	a = sorted(map(int, a.split()))
	b = sorted(map(int, b.split()))
	res = True
	for (i, j) in zip(a, b):
		if i != j and i + 1 != j:
			res = False
			break
	if res:
		print('YES')
	else:
		print('NO')
