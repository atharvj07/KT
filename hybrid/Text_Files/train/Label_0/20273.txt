#include <bits/stdc++.h>

bool cmp(const pair<long long, long long> &a,
         const pair<long long, long long> &b) {
  return (a.second < b.second);
}
long long power(long long x, long long y) {
  long long res = 1;
  x = x;
  if (x == 0) return 0;
  while (y > 0) {
    if (y & 1) res = (res * x);
    y = y >> 1;
    x = (x * x);
  }
  return res;
}
class Solution {
 public:
  int find(int parent[], int i) {
    if (parent[i] == i) {
      return i;
    }
    return find(parent, parent[i]);
  }
  void unionof(int parent[], int rank[], int x, int y) {
    int xc = find(parent, x);
    int yc = find(parent, y);
    if (rank[xc] > rank[yc]) {
      parent[yc] = xc;
    } else if (rank[yc] > rank[xc]) {
      parent[xc] = yc;
    } else {
      parent[xc] = yc;
      rank[yc]++;
    }
  }
  int numIslands(vector<vector<char>> &grid) {
    if (grid.size() == 0) return 0;
    int row = grid.size();
    int col = grid[0].size();
    int parent[row * col];
    int rank[row * col];
    for (int i = 0; i < row * col; i++) {
      parent[i] = i;
    }
    for (int i = 0; i < row * col; i++) {
      rank[i] = 0;
    }
    int x = 0, y = 0;
    for (int i = 0; i < row; i++) {
      for (int j = 0; j < col; j++) {
        if (grid[i][j] == '1') {
          x = i;
          y = j + 1;
          if (y >= 0 && y < col && grid[x][y] == '1') {
            unionof(parent, rank, i * col + j, x * col + y);
          }
          x = i + 1;
          y = j;
          if (x >= 0 && x < row && grid[x][y] == '1') {
            unionof(parent, rank, i * col + j, x * col + y);
          }
          x = i - 1;
          y = j;
          if (x >= 0 && x < row && grid[x][y] == '1') {
            unionof(parent, rank, i * col + j, x * col + y);
          }
          x = i;
          y = j - 1;
          if (y >= 0 && y < col && grid[x][y] == '1') {
            unionof(parent, rank, i * col + j, x * col + y);
          }
        }
      }
    }
    int ans = 0;
    for (int i = 0; i < row; i++) {
      for (int j = 0; j < col; j++) {
        if (grid[i][j] == '1' && parent[i * col + j] == i * col + j) {
          ans++;
        }
      }
    }
    return ans;
  }
};
