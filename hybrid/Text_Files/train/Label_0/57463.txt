import java.util.ArrayList;
import java.util.Scanner;

public class Vitamin {
	public static void main(String [] args) {
		Scanner input = new Scanner(System.in);
		int numOfDrinks = input.nextInt();
		input.nextLine();
		ArrayList<String> drinks = new ArrayList<String>();
		
		for(int i=0;i<numOfDrinks;i++) {
			String toAdd = input.nextLine();
			
			drinks.add(toAdd);
		
			
			
			
		}
		
		
		
		int min =Integer.MAX_VALUE;
		for(int i=0;i<drinks.size();i++) {
			int sum=0;
			int [] arr = new int[3];
			String first= drinks.get(i);
			int first1 = Integer.parseInt(first.substring(0,first.indexOf(" ")));
			sum+=first1;
			if(sum>=min) {
				continue;
			}
			if(first.indexOf("A")!=-1) {
				arr[0]=1;
				
			}
			if(first.indexOf("B")!=-1) {
				arr[1]=1;
			}
			if(first.indexOf("C")!=-1) {
				arr[2]=1;
			}
			if(arr[0]==1&&arr[1]==1&&arr[2]==1) {
				if(min>sum) {
					min =sum;
				}
				continue;
			}
			
			
			for(int x=0;x<drinks.size();x++) {
				sum=first1;
				if(x==i) {
					continue;
				}
			
				String second= drinks.get(x);
				int second2 = Integer.parseInt(second.substring(0,second.indexOf(" ")));
				sum+=second2;
				if(sum>=min) {
					sum-=second2;
					continue;
				}
				if(second.indexOf("A")!=-1&&arr[0]==0) {
					arr[0]=second2+10000;
					
				}
				if(second.indexOf("B")!=-1&&arr[1]==0) {
					arr[1]=second2+10000;
				}
				if(second.indexOf("C")!=-1&&arr[2]==0) {
					arr[2]=second2+10000;
				}
				if(arr[0]!=0&&arr[1]!=0&arr[2]!=0) {
					if(min>sum) {
						min=sum;
					}
					sum-=second2;
					for(int z=0;z<3;z++) {
						if(arr[z]==second2+10000) {
							
							arr[z]=0;
						}
					}
					continue;
				}
				
				
				
				for(int y=0;y<drinks.size();y++) {
					if(y==x||y==i) {
						continue;
					}
					String third= drinks.get(y);
					int second22 = Integer.parseInt(third.substring(0,third.indexOf(" ")));
					sum+=second22;
					if(sum>=min) {
						sum-=second22;
						continue;
					}
					if(third.indexOf("A")!=-1&&arr[0]==0) {
						arr[0]=second22+10001;
						
					}
					if(third.indexOf("B")!=-1&&arr[1]==0) {
						arr[1]=second22+10001;
					}
					if(third.indexOf("C")!=-1&&arr[2]==0) {
						arr[2]=second22+10001;
					}
					if(arr[0]!=0&&arr[1]!=0&&arr[2]!=0) {
						if(min>sum) {
							min =sum;
						}
						
						sum-=second22;
						for(int z=0;z<3;z++) {
							if(arr[z]==second22+10001) {
								arr[z]=0;
							}
						}
						continue;
					}
					for(int z=0;z<3;z++) {
						if(arr[z]==second22+10001) {
							arr[z]=0;
						}
					}
					sum-=second22;
				}
				for(int z=0;z<3;z++) {
					
					if(arr[z]==second2+10000) {
						
						arr[z]=0;
					}
					
				}
			
			}
		}
		
		if(min==Integer.MAX_VALUE) {
			System.out.println(-1);
		}else {
			System.out.println(min);
		}
		
		
		
		
	}

}
