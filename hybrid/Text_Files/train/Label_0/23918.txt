#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.0);
const double eps = 1e-11;
template <class T>
T sqr(const T &x) {
  return x * x;
}
template <class T>
T lowbit(const T &x) {
  return (x ^ (x - 1)) & x;
}
template <class T>
int countbit(const T &n) {
  return (n == 0) ? 0 : (1 + countbit(n & (n - 1)));
}
template <class T>
void ckmin(T &a, const T &b) {
  if (b < a) a = b;
}
template <class T>
void ckmax(T &a, const T &b) {
  if (b > a) a = b;
}
const int maxn = 100000 + 16;
int n;
int a[maxn], b[maxn], c[maxn];
int w[maxn];
vector<pair<int, int> > ret;
map<pair<int, int>, int> m1;
vector<int> m2[1 << 20];
bool dfs(int m, set<pair<int, int> > fset) {
  int l = 0, e[6];
  for (int i = 0; i < (n); ++i)
    if (!w[i]) {
      e[l++] = i;
      if (l > m) break;
    }
  if (l <= m) {
    for (int i = 0; i < (l); ++i) ret.push_back(make_pair(e[i] + 1, -1));
    return true;
  }
  for (int c1 = 0; c1 < (l); ++c1)
    for (int c2 = (c1 + 1); c2 < (l); ++c2) {
      int id1 = e[c1], id2 = e[c2];
      if (fset.count(make_pair(id1, id2))) continue;
      long long t = (long long)a[id1] * b[id2] - (long long)a[id2] * b[id1];
      if (t == 0) continue;
      long long x = +(long long)b[id1] * c[id2] - (long long)b[id2] * c[id1];
      long long y = -(long long)a[id1] * c[id2] + (long long)a[id2] * c[id1];
      if (!m1.count(make_pair(id1, id2))) {
        int id = ((int)(m1.size()));
        m1[make_pair(id1, id2)] = id;
        for (int i = 0; i < (n); ++i)
          if (x * a[i] + y * b[i] + t * c[i] == 0) m2[id].push_back(i);
      }
      int id = m1[make_pair(id1, id2)], size = ((int)(m2[id].size())),
          *lst = &m2[id][0];
      for (int i = 0; i < (size); ++i) w[lst[i]]++;
      if (dfs(m - 1, fset)) {
        ret.push_back(make_pair(id1 + 1, id2 + 1));
        return true;
      }
      for (int i = 0; i < (size); ++i) w[lst[i]]--;
      int sp = 0, p[6];
      for (int i = 0; i < (l); ++i)
        if (x * a[e[i]] + y * b[e[i]] + t * c[e[i]] == 0) p[sp++] = i;
      for (int i = 0; i < (sp); ++i)
        for (int j = (i + 1); j < (sp); ++j) fset.insert(make_pair(p[i], p[j]));
    }
  return false;
}
int main() {
  std::ios_base::sync_with_stdio(false);
  int m;
  cin >> n >> m;
  for (int i = 0; i < (n); ++i) cin >> a[i] >> b[i] >> c[i];
  for (int i = 0; i < (n); ++i) w[i] = 0;
  m1.clear();
  if (!dfs(m, set<pair<int, int> >()))
    printf("NO\n");
  else {
    printf("YES\n");
    printf("%d\n", ((int)(ret.size())));
    for (int i = 0; i < (((int)(ret.size()))); ++i)
      printf("%d %d\n", ret[i].first, ret[i].second);
  }
  return 0;
}
