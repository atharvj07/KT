#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int64_t arr[15];
  for (int64_t i = 0; i < 14; i++) cin >> arr[i];
  int64_t ans = 0;
  for (int64_t i = 0; i < 14; i++) {
    int64_t A[15];
    for (int64_t i = 0; i < 14; i++) A[i] = arr[i];
    int64_t sum = 0;
    if (A[i]) {
      int64_t x = A[i];
      A[i] = 0;
      for (int64_t j = i + 1; j < 14 && x; j++) A[j]++, x--;
      int64_t tmp = x / 14;
      for (int64_t j = 0; j < 14; j++) A[j] += tmp;
      if (tmp) x %= 14;
      for (int64_t j = 0; j < x; j++) A[j]++;
      for (int64_t j = 0; j < 14; j++)
        if (A[j] % 2 == 0) sum += A[j];
      ans = max(ans, sum);
    }
  }
  cout << ans << endl;
  return 0;
}
