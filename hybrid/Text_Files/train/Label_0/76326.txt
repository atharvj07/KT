(n, x) = map(int, input().split())
a = list(map(int, input().split()))

def solve(n, x, a):
	s = [a[0]] * n
	left = [max(a[0], 0)] * n
	right = [max(a[-1], 0)] * n
	for i in range(1, n):
		left[i] = max(left[i - 1] + a[i], a[i], 0)
		s[i] = s[i - 1] + a[i]
	for i in range(n - 2, -1, -1):
		right[i] = max(right[i + 1] + a[i], a[i], 0)
	res = c = 0
	for r in range(n):
		if r < n - 1:
			res = max(res, right[r + 1] + x * s[r] + c)
		else:
			res = max(res, x * s[r] + c)
		res = max(res, left[r])
		c = max(c, left[r] - x * s[r])
	return res
print(solve(n, x, a))
