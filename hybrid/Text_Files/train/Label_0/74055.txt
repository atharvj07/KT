n = int(input())
ptr = 0
overtake = 0
di = dict()
enter = list(map(int, input().split()))
exit = list(map(int, input().split()))
enter.reverse()
exit.reverse()
for i in range(n):
	while exit[ptr] in di:
		ptr += 1
	if enter[i] != exit[ptr]:
		overtake += 1
		di[enter[i]] = True
	else:
		ptr += 1
print(overtake)
