#include <bits/stdc++.h>
using namespace std;
vector<long long> g[200005];
long long visited[200003];
long long mark[200005];
map<long long, long long> m1;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n;
  cin >> n;
  long long i, j;
  long long a[n];
  long long maxi = 0LL;
  for (i = 0; i < n; i++) {
    cin >> a[i];
    m1[a[i]]++;
    maxi = max(m1[a[i]], maxi);
  }
  cout << n - maxi << '\n';
  priority_queue<long long> pq;
  long long num;
  for (auto i : m1) {
    if (i.second == maxi) {
      num = i.first;
      break;
    }
  }
  for (i = 0; i < n; i++) {
    if (a[i] == num) {
      pq.push(i);
      mark[i] = 1;
    }
  }
  while (pq.size() != 0) {
    long long pos;
    pos = pq.top();
    pq.pop();
    long long pos1 = -1LL;
    long long pos2;
    pos1 = pos - 1;
    pos2 = pos + 1;
    if (pos1 >= 0 && !mark[pos1] && a[pos1] != a[pos]) {
      if (a[pos1] < a[pos]) {
        cout << 1 << " " << pos1 + 1 << " " << pos + 1 << '\n';
      } else {
        cout << 2 << " " << pos1 + 1 << " " << pos + 1 << '\n';
      }
      pq.push(pos1);
      mark[pos1] = 1;
      a[pos1] = a[pos];
    }
    if (pos2 >= 0 && pos2 < n && !mark[pos2] && a[pos2] != a[pos]) {
      if (a[pos2] < a[pos]) {
        cout << 1 << " " << pos2 + 1 << " " << pos + 1 << '\n';
      } else {
        cout << 2 << " " << pos2 + 1 << " " << pos + 1 << '\n';
      }
      pq.push(pos2);
      mark[pos2] = 1;
      a[pos2] = a[pos];
    }
  }
  return 0;
}
