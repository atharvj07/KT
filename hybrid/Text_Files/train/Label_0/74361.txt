import itertools
from math import floor, log, log2, ceil
from itertools import groupby
import sys
t = int(input())
res = []
for i in range(t):
	[n, a, b] = list(map(int, input().split()))
	half_len = n // 2
	arr_a = []
	arr_b = []
	len_a = -1
	len_b = -1
	if a < b:
		arr_a = [a] + list(range(b + 1, n + 1))
		len_a = n - b + 1
		arr_b = [b] + list(range(1, a))
		len_b = a
		if len_a <= half_len and len_b <= half_len:
			arr_a += list(range(a + 1, half_len - len_a + a + 1))
			arr_b += list(range(half_len - len_a + a + 1, b))
		else:
			len_a = -1
			len_b = -1
	elif a == half_len + 1 and b == half_len:
		len_b = half_len
		len_a = half_len
		arr_b += list(range(1, b + 1))
		arr_a += list(range(a, n + 1))
	if len_a == -1 and len_b == -1:
		res.append(-1)
	else:
		res.append(' '.join(map(str, arr_a + arr_b)))
for i in range(t):
	print(res[i])
