from math import ceil, sqrt

def fun(a):
	if a % 2 == 0:
		return (a // 2 + 1) * (a // 2 + 1)
	else:
		return (a + 1) // 2 * ((a + 3) // 2)
for t in range(int(input())):
	n = int(input())
	i = int(sqrt(n)) * 2
	if fun(i - 2) >= n:
		print(i - 2)
	elif fun(i - 1) >= n:
		print(i - 1)
	elif fun(i) >= n:
		print(i)
	elif fun(i + 1) >= n:
		print(i + 1)
	else:
		print(i + 2)
