#include <bits/stdc++.h>
using namespace std;
long long int MAX = 1000000000000000;
long long int min(long long int a, long long int b) {
  if (a < b) return a;
  return b;
}
long long int max(long long int a, long long int b) {
  if (a > b) return a;
  return b;
}
long long int gcd(long long int a, long long int b) {
  if (a == 0 || b == 0) return (a + b);
  if (a < b) swap(a, b);
  return gcd(b, a % b);
}
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  long long int cost[n + 1];
  for (int i = 1; i <= n; i++) cin >> cost[i];
  long long int dp[n + 1][4];
  dp[0][0] = 0;
  dp[0][1] = dp[0][2] = dp[0][3] = MAX;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == 'h') {
      dp[i][0] = min(MAX, dp[i - 1][0] + cost[i]);
      dp[i][1] = min(dp[i - 1][1], dp[i - 1][0]);
      dp[i][2] = dp[i - 1][2];
      dp[i][3] = dp[i - 1][3];
    } else if (s[i - 1] == 'a') {
      dp[i][0] = dp[i - 1][0];
      dp[i][2] = min(dp[i - 1][2], dp[i - 1][1]);
      dp[i][3] = dp[i - 1][3];
      dp[i][1] = min(MAX, dp[i - 1][1] + cost[i]);
    } else if (s[i - 1] == 'r') {
      dp[i][3] = min(dp[i - 1][3], dp[i - 1][2]);
      dp[i][0] = dp[i - 1][0];
      dp[i][1] = dp[i - 1][1];
      dp[i][2] = min(MAX, dp[i - 1][2] + cost[i]);
    } else if (s[i - 1] == 'd') {
      dp[i][3] = min(MAX, dp[i - 1][3] + cost[i]);
      dp[i][0] = dp[i - 1][0];
      dp[i][1] = dp[i - 1][1];
      dp[i][2] = dp[i - 1][2];
    } else {
      dp[i][3] = dp[i - 1][3];
      dp[i][0] = dp[i - 1][0];
      dp[i][1] = dp[i - 1][1];
      dp[i][2] = dp[i - 1][2];
    }
  }
  long long int ans = MAX;
  for (int i = 0; i <= 3; i++) {
    ans = min(ans, dp[n][i]);
  }
  cout << ans << endl;
}
