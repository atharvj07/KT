#include <bits/stdc++.h>
using namespace std;
int a[4], r = 0, chk = 1;
int main() {
  for (int i = 0; i < 4; i++) scanf("%d", &a[i]);
  do {
    chk = 0;
    for (int i = 0; i < 4; i++) chk |= (a[i] > 1);
    if (chk) {
      int sv = -1, ct = 0;
      for (int i = 0; i < 4; i++) {
        if (a[i] == 1)
          ct++;
        else if (a[i] == 2)
          sv = i;
      }
      if (ct == 3 && sv != -1) {
        printf("+%d\n+%d\n", sv + 1, ((sv + 3) % 4) + 1);
        a[sv] += 2;
        a[(sv + 1) % 4]++;
        a[(sv + 3) % 4]++;
        chk = 2;
      }
      for (int i = 0; i < 4 && chk != 2; i++) {
        if (a[(r + i) % 4] % 2 == 0 && a[(r + i + 1) % 4] % 2 == 0) {
          a[(r + i) % 4] /= 2;
          a[(r + i + 1) % 4] /= 2;
          printf("/%d\n", (r + i) % 4 + 1);
          r = (r + i + 1) % 4;
          chk = 2;
        }
      }
      for (int i = 0; i < 4 && chk != 2; i++) {
        if (a[(r + i) % 4] % 2 == 1 && a[(r + i + 1) % 4] % 2 == 1 &&
            a[(r + i) % 4] != 1 && a[(r + i + 1) % 4] != 1) {
          a[(r + i) % 4]++;
          a[(r + i + 1) % 4]++;
          printf("+%d\n", (r + i) % 4 + 1);
          r = (r + i + 1) % 4;
          chk = 2;
        }
      }
      if (chk != 2) {
        a[r % 4]++;
        a[(r + 1) % 4]++;
        printf("+%d\n", r + 1);
        r = (r + 1) % 4;
      }
    }
  } while (chk);
  return 0;
}
