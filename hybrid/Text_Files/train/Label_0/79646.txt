n = int(input())
MAX_N = 1000.0 + 50
prime = [True] * int(MAX_N)

def sieve():
	i = 2
	while i * i < MAX_N:
		if prime[i] == True:
			j = 2
			while i * j < MAX_N:
				prime[i * j] = False
				j += 1
		i += 1
	prime[0] = False
	prime[1] = False
sieve()
ans = [False] * (n + 1)
for i in range(1, n + 1):
	if prime[i]:
		j = 1
		while i ** j <= n:
			ans[i ** j] = True
			j += 1
print(ans.count(True))
print(*[i for (i, value) in enumerate(ans) if value == True])
