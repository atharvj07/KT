#include <bits/stdc++.h>
using namespace std;
const int inf = 1000 * 1000 * 1000 + 228;
int main() {
  ios_base::sync_with_stdio(false);
  int n;
  cin >> n;
  set<pair<int, int> > st;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    auto it = st.lower_bound({l, inf});
    if (it != st.begin()) {
      it--;
      pair<int, int> p = *it;
      if (p.second >= l) {
        st.erase(it);
        l = p.first;
        r = max(r, p.second);
      }
    }
    while (1) {
      it = st.lower_bound({l, -1});
      if (it != st.end()) {
        pair<int, int> p = *it;
        if (p.first <= r) {
          st.erase(it);
          r = max(r, p.second);
        } else {
          st.insert({l, r});
          break;
        }
      } else {
        st.insert({l, r});
        break;
      }
    }
    cout << st.size() << '\n';
  }
  return 0;
}
