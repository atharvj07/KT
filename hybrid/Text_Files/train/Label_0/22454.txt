#include <bits/stdc++.h>
using namespace std;
const signed int MAX = 1e6 + 5;
const signed long long INF = 1LL << 60;
signed int num[MAX];
signed int off[3] = {-1, 0, 1};
signed long long dp[MAX][3];
signed int N, A, B;
signed long long minprime(signed int p) {
  for (signed int i = 0; i < (N); ++i)
    for (signed int j = 0; j < (3); ++j) dp[i][j] = INF;
  signed int r;
  r = num[0] % p;
  if (r == 0)
    dp[0][0] = 0;
  else if (r == 1 || r == p - 1)
    dp[0][0] = B;
  else
    dp[0][0] = INF;
  dp[0][1] = A;
  dp[0][2] = dp[0][0];
  for (signed int i = (1); i <= (N - 1); ++i) {
    r = num[i] % p;
    if (r == 0)
      dp[i][0] = dp[i - 1][0];
    else if (r == 1 || r == p - 1)
      dp[i][0] = min(dp[i][0], dp[i - 1][0] + B);
    else
      dp[i][0] = INF;
    dp[i][1] = min(dp[i][1], A + min(dp[i - 1][0], dp[i - 1][1]));
    if (r == 0)
      dp[i][2] = min(dp[i - 1][1], dp[i - 1][2]);
    else if (r == 1 || r == p - 1)
      dp[i][2] = min(dp[i][2], B + min(dp[i - 1][1], dp[i - 1][2]));
    else
      dp[i][2] = INF;
  }
  return min(dp[N - 1][0], min(dp[N - 1][1], dp[N - 1][2]));
}
signed long long solve(signed int n) {
  if (n <= 1) return INF;
  signed long long ans = INF;
  signed int i = 2;
  while (i * i <= n) {
    if (n % i == 0) {
      ans = min(ans, minprime(i));
      while (n % i == 0) n /= i;
    }
    i++;
  }
  if (n > 1) ans = min(ans, minprime(n));
  return ans;
}
signed int main() {
  scanf("%d %d %d", &N, &A, &B);
  for (signed int i = 0; i < (N); ++i) scanf("%d", num + i);
  signed long long ans = INF;
  for (signed int i = 0; i < (3); ++i) ans = min(ans, solve(num[0] + off[i]));
  for (signed int i = 0; i < (3); ++i)
    ans = min(ans, solve(num[N - 1] + off[i]));
  printf("%I64d\n", ans);
  return 0;
}
