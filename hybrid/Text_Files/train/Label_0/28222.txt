#include <bits/stdc++.h>
struct target {
  int32_t x, r;
  size_t id;
};
bool operator<(const target& lhs, const target& rhs) { return lhs.x < rhs.x; }
std::istream& operator>>(std::istream& is, target& t) {
  return is >> t.x >> t.r;
}
int main() {
  std::ios::sync_with_stdio(false);
  std::cin.tie(nullptr);
  size_t n;
  std::cin >> n;
  std::vector<target> targets(n);
  for (size_t i = 0; i < n; ++i) {
    std::cin >> targets[i];
    targets[i].id = i;
  }
  sort(targets.begin(), targets.end());
  std::vector<int32_t> results(n, -1);
  size_t m;
  std::cin >> m;
  for (size_t i = 0; i < m; ++i) {
    int32_t x, y;
    std::cin >> x >> y;
    target t{x, 0, 0};
    auto it = lower_bound(targets.begin(), targets.end(), t);
    if (it != targets.end()) {
      if (results[it->id] == -1 &&
          y * y + (x - it->x) * (x - it->x) <= it->r * it->r) {
        results[it->id] = i + 1;
      }
    }
    if (it != targets.begin()) {
      --it;
      if (results[it->id] == -1 &&
          y * y + (x - it->x) * (x - it->x) <= it->r * it->r) {
        results[it->id] = i + 1;
      }
    }
  }
  std::cout << n - count(results.begin(), results.end(), -1) << '\n';
  for (size_t i = 0; i < n; ++i) {
    std::cout << results[i] << ' ';
  }
  std::cout << '\n';
}
