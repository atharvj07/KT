NUM_POWERS = 100
Powers = []

def FindN(x, p):
	ps = Powers[p]
	psl = len(ps)
	xs = str(x)
	xsl = len(xs)
	xn = ''
	xnl = 0
	pi = 0
	xi = 0
	n = 0
	while xi < xsl and pi < psl:
		if xs[xi] == ps[pi]:
			xn = xn + xs[xi]
			xnl += 1
			xi += 1
			pi += 1
		else:
			xi += 1
			n += 1
	if xi == xsl and pi == psl:
		n += psl - xnl
	elif pi == psl:
		n += xsl - xi
	else:
		n += psl - xnl
	return n
p = 1
for i in range(0, NUM_POWERS):
	Powers.append(str(p))
	p *= 2
num_trials = int(input())
for i in range(0, num_trials):
	x = int(input())
	min_n = 100000000
	for j in range(0, NUM_POWERS):
		n = FindN(x, j)
		if n < min_n:
			min_n = n
	print(min_n)
