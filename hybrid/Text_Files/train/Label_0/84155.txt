import sys
from heapq import heappush, heappop
(n, m) = map(int, sys.stdin.buffer.readline().decode('utf-8').split())
adj = [[] for _ in range(n)]
for _ in range(m):
	(v, u, w) = map(int, sys.stdin.buffer.readline().decode('utf-8').split())
	adj[v - 1].append((u - 1, w * 2))
	adj[u - 1].append((v - 1, w * 2))
a = list(map(int, sys.stdin.buffer.readline().decode('utf-8').split()))
ans = a[:]
mask = 2 ** 18 - 1
shift = 18
hq = sorted([(a[i] << shift) + i for i in range(n)])
while hq:
	x = heappop(hq)
	(cost, v) = (x >> shift, x & mask)
	if cost > ans[v]:
		continue
	for (u, w) in adj[v]:
		if cost + w < ans[u]:
			ans[u] = cost + w
			heappush(hq, (cost + w << shift) + u)
print(*ans)
