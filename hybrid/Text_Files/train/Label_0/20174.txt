#include <bits/stdc++.h>
using namespace std;
long long next() {
  long long x;
  cin >> x;
  return x;
}
const long long inf = 2e18, mod = 1e9 + 7, MaxN = 2750132;
const int intf = 1e9;
const double PI = 3.141592653589, eps = 1e-6;
bool cmp(pair<pair<int, int>, int> a, pair<pair<int, int>, int> b) {
  return a.first.second > b.first.second;
}
bool cmp1(pair<pair<int, int>, int> a, pair<pair<int, int>, int> b) {
  return a.first.first < b.first.first;
}
int main() {
  cout.precision(20);
  ios_base::sync_with_stdio(false);
  int n = next();
  vector<pair<pair<int, int>, int> > a, b;
  for (int i = 0; i < n; i++) {
    int x = next(), y = next();
    if (x > y)
      a.push_back({{x, y}, i});
    else
      b.push_back({{x, y}, i});
  }
  if (a.size() > b.size()) {
    sort(a.begin(), a.end(), cmp1);
    cout << a.size() << '\n';
    for (int i = 0; i < a.size(); i++) {
      cout << a[i].second + 1 << ' ';
    }
  } else {
    sort(b.begin(), b.end(), cmp);
    cout << b.size() << '\n';
    for (int i = 0; i < b.size(); i++) {
      cout << b[i].second + 1 << ' ';
    }
  }
  return 0;
}
