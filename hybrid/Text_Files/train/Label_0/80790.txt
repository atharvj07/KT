n = int(input())
q = n // 3
r = n % 3
if r == 0:
	if n == 3:
		print(1, 1, 1)
	elif n == 6:
		print(2, 2, 2)
	elif q % 3 == 0:
		print(q - 2, q + 1, q + 1)
	else:
		print(q, q, q)
elif r == 1:
	if n == 4:
		print(2, 1, 1)
	elif n == 7:
		print(2, 1, 4)
	elif q % 3 == 0:
		print(q + 1, q + 1, q - 1)
	elif q % 3 == 1:
		print(q + 1, q, q)
	else:
		print(q - 1, q + 2, q)
elif n == 5:
	print(2, 2, 1)
elif n == 8:
	print(2, 2, 4)
elif q % 3 == 0:
	print(q + 2, q + 1, q - 1)
elif q % 3 == 1:
	print(q + 1, q + 1, q)
else:
	print(q + 2, q, q)
