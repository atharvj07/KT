import sys
input = lambda : sys.stdin.readline()[:-1]
ni = lambda : int(input())
na = lambda : list(map(int, input().split()))
sys.setrecursionlimit(10 ** 7)
yes = lambda : print('yes')
Yes = lambda : print('Yes')
YES = lambda : print('YES')
no = lambda : print('no')
No = lambda : print('No')
NO = lambda : print('NO')

def f(a):
	for i in range(len(a) - 1):
		if a[i] > a[i + 1]:
			return False
	return True
t = ni()
for _ in range(t):
	(n, x) = na()
	a = na()
	if x * 2 <= n:
		YES()
		continue
	else:
		x = n - x
		if x == 0:
			if f(a):
				YES()
			else:
				NO()
			continue
		b = a[:x] + a[-x:]
		b = sorted(b)
		ans = 1
		if b[x - 1] > a[x]:
			ans = 0
		if b[x] < a[n - x - 1]:
			ans = 0
		for i in range(x, n - x - 1):
			if a[i] > a[i + 1]:
				ans = 0
		if ans:
			YES()
		else:
			NO()
