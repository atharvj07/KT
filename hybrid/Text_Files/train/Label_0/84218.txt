(n, q) = map(int, input().split())
cells = {}
bad_nei = 0
for i in range(q):
	(row, col) = map(int, input().split())
	was_forbidden = cells.get((row, col), False)
	r = row % 2 + 1
	for c in range(col - 1, col + 2):
		if r == row:
			continue
		if r == row and c == col:
			continue
		if not (1 <= r <= 2 and 1 <= c <= n):
			continue
		if (r, c) in cells:
			if was_forbidden:
				bad_nei -= 1
			else:
				bad_nei += 1
	if (row, col) in cells:
		del cells[row, col]
	else:
		cells[row, col] = True
	if bad_nei >= 1:
		print('No')
	else:
		print('Yes')
