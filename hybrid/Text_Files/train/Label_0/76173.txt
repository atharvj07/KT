n = int(input())
ar = []
for _ in range(2 * n - 2):
	kek = input()
	ar.append([kek, len(kek), _])
ar.sort(key=lambda x: -x[1])
ans = [''] * (2 * n - 2)
pref = ar[0][0]
kref = ar[1][0]
flag = 0
for i in range(0, 2 * n - 2, 2):
	if ar[i][0] == pref[0:n - i // 2 - 1]:
		ans[ar[i][2]] = 'P'
		if ar[i + 1][0] == kref[i // 2:]:
			ans[ar[i + 1][2]] = 'S'
		else:
			flag = 1
	elif ar[i][0] == kref[i // 2:]:
		ans[ar[i][2]] = 'S'
		if ar[i + 1][0] == pref[0:n - i // 2 - 1]:
			ans[ar[i + 1][2]] = 'P'
		else:
			flag = 1
	else:
		flag = 1
if flag == 0:
	print(''.join(ans))
else:
	ans = [''] * (2 * n - 2)
	pref = ar[1][0]
	kref = ar[0][0]
	flag = 0
	for i in range(0, 2 * n - 2, 2):
		if ar[i][0] == pref[0:n - i // 2 - 1]:
			ans[ar[i][2]] = 'P'
			if ar[i + 1][0] == kref[i // 2:]:
				ans[ar[i + 1][2]] = 'S'
			else:
				flag = 1
		elif ar[i][0] == kref[i // 2:]:
			ans[ar[i][2]] = 'S'
			if ar[i + 1][0] == pref[0:n - i // 2 - 1]:
				ans[ar[i + 1][2]] = 'P'
			else:
				flag = 1
		else:
			flag = 1
	print(''.join(ans))
