#include <bits/stdc++.h>
using namespace std;
bool leap[2000005];
int main() {
  for (long long i = 1000; i < 2000005; i++)
    if (i % 400 == 0 or (i % 4 == 0 and i % 100 != 0)) leap[i] = 1;
  long long n, sum = 0;
  scanf("%lld", &n);
  for (long long i = n; i < 2000005; i++) {
    if (leap[i])
      sum += 366;
    else
      sum += 365;
    if (sum % 7 == 0 and (leap[n] == leap[i + 1]))
      return 0 * printf("%lld ", i + 1);
  }
  return 0;
}
