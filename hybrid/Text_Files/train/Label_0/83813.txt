(n, m) = list(map(int, input().strip().split()))

def quad_solve(a, b, c):
	x = b ** 2 - 4 * a * c
	return (-b + x ** 0.5) / 2
if n <= m:
	print(n)
else:
	aM = m
	n -= m
	(l, r) = (0, int(2000000000.0))
	while l < r:
		m = (l + r) // 2
		val = m * (m + 1) // 2
		if val >= n:
			r = m
		else:
			l = m + 1
	print(l + aM)
