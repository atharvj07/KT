#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.0);
const int N = 55;
double dp[N][N], fact[N];
int a[N];
int main() {
  int n;
  cin >> n;
  double factN = 1.0, ans = 0.0;
  dp[0][0] = 1.0;
  fact[0] = 1.0;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    fact[i] = fact[i - 1] * i;
  }
  int p;
  cin >> p;
  for (int i = 1; i <= n; ++i) {
    for (int j = i; j >= 1; --j) {
      for (int k = a[i]; k <= p; ++k) {
        dp[j][k] += dp[j - 1][k - a[i]];
      }
    }
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= p; ++j) {
      ans += dp[i][j] * fact[i] * fact[n - i];
    }
  }
  cout << ans / fact[n];
  return 0;
}
