def translate_to_segments(string):
	firstchar = string[0]
	prevchar = string[0]
	segments = []
	for char in string:
		if len(segments) == 0 or char != prevchar:
			segments.append(0)
			prevchar = char
		segments[-1] = segments[-1] + 1
	if len(segments) > 1 and firstchar == prevchar:
		segments[0] = segments[0] + segments[-1]
		segments[-1] = 0
	return segments

def count_flips(segments):
	if len(segments) == 1:
		return 1 + (segments[0] - 1) // 3
	elif len(segments) == 0:
		return 0
	else:
		summ = 0
		for seg in segments:
			summ = summ + seg // 3
		return summ
total = int(input())
for index in range(2 * total):
	line = input()
	if index % 2 == 0:
		pass
	else:
		segments = translate_to_segments(line)
		flips = count_flips(segments)
		print(flips)
