#include <bits/stdc++.h>
using namespace std;
int x[233], y[233], n, rzz[233][233];
double ans;
int RZZ(int i, int j) {
  return (x[i] - x[j]) * (x[i] - x[j]) + (y[i] - y[j]) * (y[i] - y[j]);
}
int cross(int i, int j, int k) {
  int x1 = x[j] - x[i], y1 = y[j] - y[i];
  int x2 = x[k] - x[i], y2 = y[k] - y[i];
  return x1 * y2 - x2 * y1;
}
int dot(int i, int j, int k) {
  int x1 = x[i] - x[j], y1 = y[i] - y[j];
  int x2 = x[k] - x[j], y2 = y[k] - y[j];
  return x1 * x2 + y1 * y2;
}
double DIS(double X, double Y, int i) {
  return sqrt((x[i] - X) * (x[i] - X) + (y[i] - Y) * (y[i] - Y));
}
bool Judge(double X, double Y, double dis) {
  for (int i = 1; i <= n; i++)
    if (DIS(X, Y, i) + 0.00000001 < dis) {
      return 0;
    }
  return 1;
}
void work(int i, int j, int k) {
  if (cross(i, j, k) == 0) return;
  if (dot(i, j, k) <= 0 || dot(j, k, i) <= 0 || dot(k, i, j) <= 0) return;
  double A1 = 2 * (x[i] - x[j]), B1 = 2 * (y[i] - y[j]),
         C1 = (x[i]) * (x[i]) - (x[j]) * (x[j]) + (y[i]) * (y[i]) -
              (y[j]) * (y[j]);
  double A2 = 2 * (x[i] - x[k]), B2 = 2 * (y[i] - y[k]),
         C2 = (x[i]) * (x[i]) - (x[k]) * (x[k]) + (y[i]) * (y[i]) -
              (y[k]) * (y[k]);
  double X = (C1 * B2 - C2 * B1) / (A1 * B2 - A2 * B1);
  double Y = (C1 - X * A1) / B1;
  double dis = DIS(X, Y, i);
  if (Judge(X, Y, dis)) ans = max(ans, dis);
}
void check(int i, int j, int k, int l) {
  if (rzz[i][j] == rzz[k][l] && rzz[j][k] == rzz[i][l] &&
      rzz[i][j] + rzz[j][k] == rzz[i][k]) {
    double X = (x[i] + x[k]) / 2.0;
    double Y = (y[i] + y[k]) / 2.0;
    double dis = DIS(X, Y, i);
    if (Judge(X, Y, dis)) ans = max(ans, dis);
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d%d", &x[i], &y[i]);
  ans = -1;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      for (int k = j + 1; k <= n; k++) work(i, j, k);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) rzz[i][j] = RZZ(i, j);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i != j)
        for (int k = 1; k <= n; k++)
          if (j != k && k != i)
            for (int l = 1; l <= n; l++)
              if (l != i && l != k && l != j) check(i, j, k, l);
  if (ans < 0)
    printf("-1\n");
  else
    printf("%.6lf\n", ans);
}
