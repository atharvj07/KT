import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Comparator;
import java.util.StringTokenizer;
import java.util.TreeSet;

public class A {
	public static void main(String[] args) throws IOException {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		long[] as = new long[n], bs = new long[n];
		for(int i=0;i<n;i++)
			as[i] = in.nextLong();
		for(int i=0;i<n;i++)
			bs[i] = in.nextLong();
		int ptr = n-1;
		CHT cht = new CHT();
		cht.add(new Line(as[ptr], 0, 0));
		long last = as[ptr];
		while(--ptr >= 0) {
			long y = cht.query(bs[ptr]);
			cht.add(new Line(as[ptr], 0, y));
			last = y;
		}
		out.println(last);
		out.close();
	}
	static class CHT {
		TreeSet<Line> hull, query;
		public CHT() {
			hull = new TreeSet<Line>(new Comparator<Line>() {
				public int compare(Line a, Line b) {
					return -Long.compare(a.m, b.m);
				}
			});
			query = new TreeSet<Line>(new Comparator<Line>(){
				public int compare(Line a, Line b) {
					if(a.left==b.left)return Long.compare(a.m, b.m);
					return Double.compare(a.left, b.left);
				}
			});
		}
		public void add(Line a) {
			Line[] LR = {hull.lower(a), hull.ceiling(a)};
			for(int i=0;i<2;i++)
				if(LR[i]!=null&&LR[i].m==a.m){
					if(LR[i].b<a.b)return;///////////////
					else remove(LR[i]);
				}
			
			hull.add(a);
			Line L = hull.lower(a), R = hull.higher(a);
			if(L!=null&&R!=null&&a.inter(R)<=R.left){
				hull.remove(a);
				return;
			}
			Line LL = (L!=null)?hull.lower(L):null, RR = (R!=null)?hull.higher(R):null;
			if(L!=null)a.left=a.inter(L);
			if(R!=null){
				query.remove(R);
				R.left=a.inter(R);
				query.add(R);
			}
			while(LL!=null){
				if(L.left>=a.inter(LL)){
					remove(L);
					a.left=a.inter(L=LL);
					LL = hull.lower(L);
				}else
					break;
			}
			while(RR!=null){
				if(R.inter(RR)<=a.inter(RR)){
					remove(R);
					query.remove(RR);
					RR.left = a.inter(R=RR);
					query.add(RR);
					RR = hull.higher(R);
				}else
					break;
			}
			query.add(a);
		}
		public long query(long x) {
			Line temp = new Line(0,0,0);
			temp.left = x;
			return (long)query.floor(temp).eval(x);
		}
		private void remove(Line x) {
			hull.remove(x);
			query.remove(x);
		}
	}
	static class Line{
		long m,b;
		double left=Long.MIN_VALUE;
		public Line(long m, long x, long y) {
			this.m = m;
			this.b = -m*x+y;
		}
		public long eval(long x){
			return m*x+b;
		}
		public double inter(Line x){
			return (double)(x.b-this.b)/(double)(this.m-x.m);
		}
		public String toString() {
			return m+"x + "+b+" left: "+left;
		}
	}
	static class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(InputStream in) {
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer("");
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String next() throws IOException {
			while (!st.hasMoreTokens()) {
				st = new StringTokenizer(br.readLine());
			}
			return st.nextToken();
		}
	}
}
