#include <bits/stdc++.h>
using namespace std;
const long long P = 998244353;
long long t[5100][5100];
long long a[5100];
signed main() {
  t[1][1] = 1;
  for (long long n = 2; n <= 5000; ++n) {
    for (long long k = 1; k <= n; ++k) {
      t[n][k] = (t[n - 1][k] * k + t[n - 1][k - 1] * (n - k + 1)) % P;
    }
  }
  long long n;
  cin >> n;
  a[1] = 1;
  for (long long i = 2; i <= n; ++i) {
    for (long long j = 1; j <= i; ++j) {
      a[j] = (a[j] * i) % P;
      a[j] += t[i][j];
      a[j] %= P;
    }
  }
  for (long long i = 1; i <= n; ++i) {
    cout << a[i] << " ";
  }
  cout << "\n";
  return 0;
}
