import heapq
n = int(input())
res = 0
left = []
right = []
for _ in range(n):
	(l, r) = list(map(int, input().split()))
	left.append(l)
	right.append(r)
heapq.heapify(left)
heapq.heapify(right)
for _ in range(n):
	(l, r) = (heapq.heappop(left), heapq.heappop(right))
	res += max(l, r)
print(res + n)
