for _ in range(int(input())):
	(n, k) = map(int, input().split())
	used = [False] * (2 * n)
	ps = []
	for _ in range(k):
		(x, y) = map(int, input().split())
		(x, y) = (x - 1, y - 1)
		ps.append(sorted([x, y]))
		used[x] = used[y] = True
	unused = [i for (i, v) in enumerate(used) if not v]
	for (x, y) in zip(unused, unused[len(unused) // 2:]):
		ps.append(sorted([x, y]))
	print(sum(((x1 < x2 < y1) != (x1 < y2 < y1) for (x1, y1) in ps for (x2, y2) in ps)) // 2)
