#include <bits/stdc++.h>
using namespace std;
long long tot;
int n;
int a[200001];
int b[200001];
int c[200001];
int s[35];
inline int qr() {
  char ch;
  while ((ch = getchar()) < '0' || ch > '9')
    ;
  int res = ch ^ 48;
  while ((ch = getchar()) >= '0' && ch <= '9') res = res * 10 + (ch ^ 48);
  return res;
}
inline void check() {
  int m, j;
  long long x, y;
  for (register int i = 1; i <= n; ++i) {
    m = a[i], j = 1;
    while (m) {
      if (m & 1) ++s[j];
      m >>= 1;
      ++j;
    }
  }
  for (register int i = 1; i <= n; ++i) {
    m = a[i], j = 1, x = y = 0;
    while (j <= 32) {
      if (m & 1) {
        x += s[j] << j - 1;
        y += n << j - 1;
      } else
        y += s[j] << j - 1;
      m >>= 1;
      ++j;
    }
    if (x != b[i] || y != c[i]) {
      puts("-1");
      exit(0);
    };
  }
}
int main() {
  n = qr();
  for (register int i = 1; i <= n; ++i) tot += (b[i] = qr());
  for (register int i = 1; i <= n; ++i) {
    tot += (c[i] = qr());
    if (c[i] < b[i]) {
      puts("-1");
      exit(0);
    };
  }
  if (tot % (2 * n) != 0) {
    puts("-1");
    exit(0);
  };
  tot /= 2 * n;
  for (register int i = 1; i <= n; ++i) {
    a[i] = b[i] + c[i];
    if (a[i] < tot) {
      puts("-1");
      exit(0);
    };
    a[i] -= tot;
    if (a[i] % n != 0) {
      puts("-1");
      exit(0);
    };
    a[i] /= n;
  }
  check();
  for (register int i = 1; i <= n; ++i) printf("%d ", a[i]);
  return 0;
}
