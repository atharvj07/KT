a = int(input())

def fixer(s, index, ans, g1):
	for j in range(index, len(g1)):
		if g1[j] == s:
			ans.append([index + 1, j + 1])
			m = g1[index:j + 1]
			m.reverse()
			for t in range(len(m)):
				g1[t + index] = m[t]
			break
for i in range(a):
	(n, k) = map(int, input().split())
	s = input()
	ans = []
	g1 = []
	index = 0
	for i in range(len(s)):
		g1.append(s[i])
	m = len(s) // 2
	for i in range(k - 1):
		fixer('(', index, ans, g1)
		index += 1
		fixer(')', index, ans, g1)
		index += 1
		m -= 1
	while m:
		m -= 1
		fixer('(', index, ans, g1)
		index += 1
	print(len(ans))
	for i in range(len(ans)):
		print(ans[i][0], ans[i][1])
