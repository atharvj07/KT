#include <bits/stdc++.h>
using namespace std;
vector<int> edge[101000];
int son[101000];
double ans[101000];
int dfs(int u) {
  son[u] = 1;
  for (int i = 0; i < edge[u].size(); i++) {
    int v = edge[u][i];
    son[u] += dfs(v);
  }
  return son[u];
}
void dfs2(int u) {
  int sum = son[u] - 1, sz = edge[u].size();
  double p;
  if (sz == 1)
    p = 0;
  else
    p = 0.5;
  for (int i = 0; i < edge[u].size(); i++) {
    int v = edge[u][i];
    ans[v] = ans[u] + (sum - son[v]) * p + 1;
    dfs2(v);
  }
}
int main() {
  int n, tmp;
  cin >> n;
  for (int i = 2; i <= n; i++) {
    cin >> tmp;
    edge[tmp].push_back(i);
  }
  dfs(1);
  ans[1] = 1;
  dfs2(1);
  for (int i = 1; i <= n; i++) {
    printf("%.13lf ", ans[i]);
  }
}
