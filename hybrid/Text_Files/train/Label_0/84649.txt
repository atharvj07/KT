(qtdWarriors, pares) = input().split()
qtdWarriors = int(qtdWarriors)
pares = int(pares)
dic = {}
somas = []
for i in range(pares):
	(chave, valor) = input().split()
	if chave not in dic:
		dic[chave] = [valor]
	else:
		dic[chave].append(valor)
	if valor not in dic:
		dic[valor] = [chave]
	else:
		dic[valor].append(chave)
for key in dic:
	for e in dic[key]:
		if e in dic:
			for e2 in dic[e]:
				if e2 in dic[key] and e2 in dic[e] and (key in dic[e2]) and (key in dic[e]) and (e in dic[e2]):
					soma = len(dic[key]) + len(dic[e])
					soma += len(dic[e2])
					somas.append(soma - 6)
if somas != []:
	print(min(somas))
else:
	print(-1)
