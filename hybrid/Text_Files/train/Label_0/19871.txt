#include <bits/stdc++.h>
bool isvowel(char c) {
  return c == 'e' || c == 'y' || c == 'u' || c == 'i' || c == 'o' || c == 'a';
}
int main() {
  std::ios_base::sync_with_stdio(false);
  std::cin.tie(0);
  std::cout.tie(0);
  std::cerr.tie(0);
  int n, k;
  std::cin >> n;
  std::string st;
  std::cin >> st;
  bool prev = false;
  for (size_t it = 0; it < st.size(); ++it) {
    if (!prev || prev && !isvowel(st[it])) {
      std::cout << st[it];
    }
    prev = isvowel(st[it]);
  }
}
