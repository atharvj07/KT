(n, k) = map(int, input().split())
(s, i, j) = ('', 0, 10000)
t = [[] for x in range(j + 1)]
for (y, x) in enumerate(map(int, input().split()), 1):
	t[x].append(str(y))
while not t[i]:
	i += 1
while not t[j]:
	j -= 1
while k and i < j:
	(a, b) = (t[j].pop(), t[i].pop())
	s += '\n' + a + ' ' + b
	t[j - 1].append(a)
	t[i + 1].append(b)
	k -= 1
	if not t[i]:
		i += 1
	if not t[j]:
		j -= 1
print(str(j - i) + ' ' + str(s.count(' ')) + s)
