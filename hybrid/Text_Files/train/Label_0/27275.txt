#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  int n, t, ans = INT_MAX, s, d, ans1;
  cin >> n >> t;
  for (int i = 0; i < n; i++) {
    cin >> s >> d;
    while (s < t) s += d;
    if (ans > s) {
      ans = s;
      ans1 = i + 1;
    }
  }
  cout << ans1 << '\n';
  return 0;
}
