#include <bits/stdc++.h>
using namespace std;
struct Node {
  int L, R, sum;
} node[100100 << 4];
int n, m;
void build_tree(int l, int r, int id) {
  node[id].L = l;
  node[id].R = r;
  node[id].sum = 0;
  int mid = (l + r) / 2;
  if (l == r) return;
  build_tree(l, mid, id << 1);
  build_tree(mid + 1, r, (id << 1) + 1);
}
void update(int id, int pt, int v) {
  if (node[id].L == node[id].R) {
    node[id].sum = v;
  } else {
    int mid = (node[id].L + node[id].R) / 2;
    if (mid >= pt)
      update(id << 1, pt, v);
    else
      update((id << 1) + 1, pt, v);
    node[id].sum = node[(id << 1) + 1].sum + node[id << 1].sum;
  }
}
int query(int l, int r, int id) {
  if (node[id].L == l && node[id].R == r)
    return node[id].sum;
  else {
    int mid = (node[id].L + node[id].R) / 2;
    if (mid >= r)
      return query(l, r, id << 1);
    else if (mid < l)
      return query(l, r, (id << 1) + 1);
    else
      return query(l, mid, id << 1) + query(mid + 1, r, (id << 1) + 1);
  }
}
int p[100100 * 4], P[100100 * 4];
int ans;
int q, x, y;
int main() {
  while (~scanf("%d", &n)) {
    build_tree(1, n, 1);
    for (int i = 1; i <= n; i++) {
      scanf("%d", &p[i]);
      P[p[i]] = i;
    }
    for (int i = 1; i < n; i++) {
      if (P[i] > P[i + 1]) {
        update(1, i + 1, 1);
      }
    }
    scanf("%d", &m);
    for (int i = 1; i <= m; i++) {
      scanf("%d%d%d", &q, &x, &y);
      if (q == 1) {
        ans = query(x + 1, y, 1);
        printf("%d\n", ans + 1);
      } else {
        int a, b;
        a = p[x];
        b = p[y];
        p[x] = b;
        p[y] = a;
        P[a] = y;
        P[b] = x;
        int tmp;
        if (a != n) {
          tmp = a + 1;
          if (P[tmp] < P[a]) update(1, tmp, 1);
        }
        if (a != 1) {
          tmp = a - 1;
          if (P[tmp] < P[a]) update(1, a, 0);
        }
        if (b != n) {
          tmp = b + 1;
          if (P[b] < P[tmp]) update(1, tmp, 0);
        }
        if (b != 1) {
          tmp = b - 1;
          if (P[b] < P[tmp]) update(1, b, 1);
        }
      }
    }
  }
  return 0;
}
