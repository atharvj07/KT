(n, k) = map(int, input().split())
s = input()
cnt = [0 for i in range(26)]
for i in range(n):
	cnt[ord(s[i]) - ord('a')] += 1
tk = k
for i in range(26):
	if tk >= cnt[i]:
		tk -= cnt[i]
	else:
		cnt[i] = tk
		tk = 0
for c in s:
	if cnt[ord(c) - ord('a')] > 0:
		cnt[ord(c) - ord('a')] -= 1
	else:
		print(c, end='')
