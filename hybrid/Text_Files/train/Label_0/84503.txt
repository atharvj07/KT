for i in range(int(input())):
	(n, m) = map(int, input().split())
	a = [0] + list(map(int, input().split()))
	b = [[-1, i] for i in range(n + 1)]
	if m < n or n == 2:
		print(-1)
	else:
		for i in range(n + 1):
			b[i][0] = a[i]
		b.sort()
		ans = 0
		pr = []
		ind1 = b[1][1]
		ind2 = b[2][1]
		ind3 = b[3][1]
		ans = 0
		ans = 2 * sum(a)
		pr.append([1, n])
		for i in range(1, n):
			pr.append([i, i + 1])
		for i in range(m - n):
			pr.append([ind2, ind1])
			ans += a[ind2] + a[ind1]
		print(ans)
		for i in pr:
			print(*i)
