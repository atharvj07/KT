#include<bits/stdc++.h>
#pragma warning(disable:4996)
using namespace std;
using ll = long long;
const int dx[4] = { 1,0,-1,0 }, dy[4] = { 0,1,0,-1 };
#define int long long
const ll mod = 1e9 + 7;
ll func(ll n, ll value) {
	vector<vector<ll>>dp(n + 1, vector<ll>(value+1,0));
	for (ll i = 0; i <= n;i++) {
		dp[i][0] = 1;
	}
	for (ll i = 0; i < n; i++) {
		for (int j = 1; j <= value; j++) {
			dp[i + 1][j] = (dp[i + 1][j - 1]%mod + dp[i][j]%mod)%mod;
		}
	}
	return dp[n][value]%mod;
}
map<int, int> prime_factor(int n) {
	map<int, int>res;
	for (int i = 2; i*i <= n; i++) {
		while (n%i == 0) {
			++res[i];
			n /= i;
		}
	}
	if (n != 1)res[n] = 1;
	return res;
}
int n, m;//nが長さ　mは値
signed main() {
	cin >> n >> m;
	map<int,int> ma = prime_factor(m);
	int ans = 1;
	for (auto e : ma) {
		int key, value;
		tie(key, value) = e;
		ans *= func(n, value)%mod;
		ans %= mod;
	}
	cout << ans << endl;
	return 0;
}
