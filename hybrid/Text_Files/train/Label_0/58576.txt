    private static int VISITED = 1;

    private static void solve(int q, FastScanner in, PrintWriter out) {
        for (int qq = 0; qq < q; qq++) {
            int n = in.nextInt(), d = in.nextInt();
            int max = n * (n - 1) / 2;
            int min = 0;
            int cnt = n - 1, base = 2, depth = 1;
            while(cnt > 0) {
                min += Math.min(base, cnt) * depth;
                depth++;
                cnt -= base;
                base = base * 2;
            }
            if(d < min || d > max) {
                out.println("NO");
            }
            else {
                int[] parent = new int[n + 1];
                for(int i = 2; i <= n; i++) {
                    parent[i] = i - 1;
                }
                int[] childCnt = new int[n + 1];
                for(int i = 1; i < n; i++) {
                    childCnt[i] = 1;
                }
//                Set<Integer>[] children = new Set[n + 1];
//                for(int i = 0; i <= n; i++) {
//                    children[i] = new HashSet<>();
//                }
//                for(int i = 1; i < n; i++) {
//                    children[i].add(i + 1);
//                }
                TreeMap<Integer, List<Integer>> dist = new TreeMap<>();
                for(int i = 1; i <= n; i++) {
                    List<Integer> list = new ArrayList<>();
                    list.add(i);
                    dist.put(i - 1, list);
                }
                int diff = max - d;
                while(diff > 0) {
                    Map.Entry<Integer, List<Integer>> entry = dist.lastEntry();
                    int currDist = entry.getKey();
                    List<Integer> currList = entry.getValue();
                    int currNode = currList.remove(currList.size() - 1);


                    Map.Entry<Integer, List<Integer>> ancestorEntry = dist.ceilingEntry(currDist - diff - 1);
                    int ancestorDist = ancestorEntry.getKey();
                    List<Integer> ancestorList = ancestorEntry.getValue();
                    int ancestor = ancestorList.get(ancestorList.size() - 1);

                    childCnt[parent[currNode]]--;
                    childCnt[ancestor]++;
                    parent[currNode] = ancestor;

                    if(currList.size() == 0) {
                        dist.remove(currDist);
                    }
                    if(!dist.containsKey(ancestorDist + 1)) {
                        dist.put(ancestorDist + 1, new ArrayList<>());
                    }
                    dist.get(ancestorDist + 1).add(currNode);

                    if(childCnt[ancestor] == 2) {
                        ancestorList.remove(ancestorList.size() - 1);
                        if(ancestorList.size() == 0) {
                            dist.remove(ancestorDist);
                        }
                    }
                    diff -= (currDist - ancestorDist - 1);
                }
                out.println("YES");
                for(int i = 2; i <= n; i++) {
                    out.print(parent[i] + " ");
                }
                out.println();
            }
        }
        out.close();
    }

    private static long modularAdd(long a, long b, int mod) {
        long sum = a + b;
        if (sum >= mod) {
            sum -= mod;
        }
        return sum;
    }

    private static long modularSubtract(long a, long b, int mod) {
        long diff = a - b;
        if (diff < 0) {
            diff += mod;
        }
        return diff;
    }

    private static Stack<Integer> topologicalSort(Set<Integer>[] g) {
        Stack<Integer> stack = new Stack<>();
        int[] state = new int[g.length];

        for (int node = 0; node < g.length; node++) {
            if (!topoSortHelper(g, stack, state, node)) {
                return null;
            }
        }
        return stack;
    }

    private static boolean topoSortHelper(Set<Integer>[] g, Stack<Integer> stack, int[] state, int currNode) {
        if (state[currNode] == VISITED) {
            return true;
        } else if (state[currNode] == VISITING) {
            return false;
        }
        state[currNode] = VISITING;
        for (int neighbor : g[currNode]) {
            if (!topoSortHelper(g, stack, state, neighbor)) {
                return false;
            }
        }
        state[currNode] = VISITED;
        stack.push(currNode);
        return true;
    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        FastScanner(InputStream stream) {
            try {
                br = new BufferedReader(new InputStreamReader(stream));
            } catch (Exception e) {
                e.printStackTrace();
            }
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
