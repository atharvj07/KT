for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	is_sorted = True
	for i in range(1, n):
		if a[i] < a[i - 1]:
			is_sorted = False
			break
	if is_sorted:
		print(0)
	else:
		all_incorrect = True
		for i in range(n):
			if a[i] == i + 1:
				all_incorrect = False
				break
		if all_incorrect:
			print(1)
		else:
			i = 0
			while a[i] == i + 1:
				i += 1
			j = n - 1
			while a[j] == j + 1:
				j -= 1
			all_incorrect = True
			for k in range(i, j + 1):
				if a[k] == k + 1:
					all_incorrect = False
					break
			if all_incorrect:
				print(1)
			else:
				print(2)
