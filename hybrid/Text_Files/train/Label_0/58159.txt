
import java.awt.Point;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class Meeting {

    public void solve() throws IOException {

        Scanner sc = new Scanner(System.in);
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(System.out);


        int i = 0, j = 0, k = 0;
        long ans = 0, sum = 0, count = 0;
        int xa = sc.nextInt();
        int xb = sc.nextInt();
        int ya = sc.nextInt();
        int yb = sc.nextInt();
        int n = sc.nextInt();

        ArrayList<Point> Plist = new ArrayList<Point>();
        for (i = Math.min(xa, ya); i <= Math.max(xa, ya); ++i) {
            for (j = Math.min(yb, xb); j <= Math.max(yb, xb); ++j) {
                if (i == xa || i == ya) {
                    Plist.add(new Point(i, j));
                } else {
                    Plist.add(new Point(i, yb));
                    Plist.add(new Point(i, xb));
                    break;
                }
            }
        }
        //  System.out.println(Plist.size());
        sum = Plist.size();
        ArrayList<heater> hlist = new ArrayList<heater>();
        for (i = 0; i < n; ++i) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            int r = sc.nextInt();
            if (hlist.contains(new heater(x, y, r, true))) {
                int rr = hlist.get(hlist.indexOf(new heater(x, y, r, true))).r;
                if (r > rr) {
                    hlist.remove(new heater(x, y, r, true));
                    hlist.add(new heater(x, y, r, true));
                }
            } else {
                hlist.add(new heater(x, y, r, true));
            }
        }

        heater hset[] = hlist.toArray(new heater[hlist.size()]);
        //   System.out.println(Arrays.toString(hset));

        for (i = 0; i < hlist.size(); ++i) {
            int x = hset[i].x;
            int y = hset[i].y;
            int r = hset[i].r;

            for (j = 0; j < Plist.size(); ++j) {
                int xx = Plist.get(j).x;
                int yy = Plist.get(j).y;
                if (xx != Integer.MAX_VALUE && yy !=Integer.MAX_VALUE) {
                    if (r * r >= (x - xx) * (x - xx) + (y - yy) * (y - yy)) {
                        Plist.get(j).setLocation(Integer.MAX_VALUE, Integer.MAX_VALUE);
                        count++;
                    }
                }

            }
            //System.out.println(Arrays.toString(Plist.toArray(new Point[Plist.size()])));

        }

        System.out.println(sum - count);


    }

    public static void main(String[] args) throws IOException {
        new Meeting().solve();

    }
}

class heater {

    int x;
    int y;
    int r;
    boolean gotit;

    public heater(int x, int y, int r, boolean gotit) {
        this.x = x;
        this.y = y;
        this.r = r;
        this.gotit = gotit;
    }

    public void setR(int r) {
        this.r = r;
    }

    public void setX(int x) {
        this.x = x;
    }

    public void setY(int y) {
        this.y = y;
    }

    public int getR() {
        return r;
    }

    public int getX() {
        return x;
    }

    public int getY() {
        return y;
    }

    @Override
    public boolean equals(Object o) {
        if (o instanceof heater && ((heater) o).x == this.x && ((heater) o).y == this.y) {
            return true;
        } else {
            return false;
        }
    }

    public String toString() {
        return "(" + x + " " + y + " " + r + ")";
    }
}
