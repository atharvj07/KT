import operator
import collections
from sys import stdin
N = int(input())
pos = list(map(int, stdin.readline().split()))
speed = list(map(int, stdin.readline().split()))
A = []
for i in range(N):
	A.append((pos[i], speed[i]))
dist = {}
for i in pos:
	dist[i] = 0
A.sort(key=operator.itemgetter(0))
count = [0]

def merge_compare(p, r):
	if p < r:
		q = (r + p) // 2
		merge_compare(p, q)
		merge_compare(q + 1, r)
		merge(p, q, r)

def merge(p, q, r):
	n_1 = q - p + 1
	n_2 = r - q
	temp = []
	L = A[p:q + 1]
	R = A[q + 1:r + 1]
	i = 0
	j = 0
	sum_pos_right = 0
	for (e, f) in R:
		sum_pos_right += e
	while i < n_1 and j < n_2:
		if L[i][1] <= R[j][1]:
			temp.append(L[i])
			count[0] += sum_pos_right - L[i][0] * (n_2 - j)
			i += 1
		else:
			temp.append(R[j])
			sum_pos_right -= R[j][0]
			j += 1
	if i == n_1:
		temp += R[j:]
	else:
		temp += L[i:]
	A[p:r + 1] = temp
merge_compare(0, N - 1)
print(count[0])
