MOD = 1000000007
(n, l, r) = map(int, input().split())
dp = []
for i in range(n):
	dp.append([0, 0, 0])
Bw = r - l + 1
X = int(Bw // 3)
Zero = X
One = X
Two = X
Rem = Bw % 3
if Rem == 1:
	if l % 3 == 0:
		Zero += 1
	elif l % 3 == 1:
		One += 1
	else:
		Two += 1
elif Rem == 2:
	if l % 3 == 0:
		Zero += 1
		One += 1
	elif l % 3 == 1:
		One += 1
		Two += 1
	else:
		Two += 1
		Zero += 1
dp[0][0] = Zero % MOD
dp[0][1] = One % MOD
dp[0][2] = Two % MOD
for i in range(1, n):
	dp[i][0] = (dp[i - 1][0] * Zero + dp[i - 1][1] * Two + dp[i - 1][2] * One) % MOD
	dp[i][1] = (dp[i - 1][0] * One + dp[i - 1][1] * Zero + dp[i - 1][2] * Two) % MOD
	dp[i][2] = (dp[i - 1][0] * Two + dp[i - 1][1] * One + dp[i - 1][2] * Zero) % MOD
print(dp[n - 1][0])
