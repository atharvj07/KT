def dd(i, nn):
	s = bin(i).replace('0b', '')
	l = len(s)
	kk = ''
	for i in range(nn - l):
		kk = kk + '0'
	print(kk + s)
t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	aa = []
	for pp in range(n):
		pp = input()
		aa.append(int(pp, 2))
	y = 2 ** (m - 1) - 1
	a = max(0, y - n)
	b = min(y + n, 2 ** m - 1)
	for i in range(a, b + 1):
		l = i
		r = 2 ** m - 1 - i
		flag = 1
		for k in aa:
			if k == i:
				flag = 0
				break
			if k < i:
				l = l - 1
			else:
				r = r - 1
		if flag == 0:
			continue
		if n % 2 == 0 and l == r - 1:
			dd(i, m)
		if n % 2 != 0 and l == r:
			dd(i, m)
