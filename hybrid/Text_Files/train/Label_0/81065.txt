import sys
import math
from itertools import islice, cycle
from collections import Counter, defaultdict
stdin = sys.stdin
t = stdin.readline().strip()
t = int(t)
results = {0: 0, 1: 0, 2: 1}

def smallest_sol(m):
	x = -1 / 2 + math.sqrt(1 / 4 + 6 * m)
	x /= 3
	return math.floor(x)

def solv(n):
	global results
	if n in results:
		return results[n]
	h = smallest_sol(n)
	cards = 3 / 2 * h ** 2 + 1 / 2 * h
	cards = int(cards)
	r = solv(n - cards) + 1
	results[n] = r
	return r
for test_idx in range(t):
	n = stdin.readline().strip()
	n = int(n)
	solv(n)
	print(results[n])
