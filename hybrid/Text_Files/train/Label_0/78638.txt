import bisect

def js(arr):
	n = len(arr)
	a = []
	arr = [int(arr[i]) - i for i in range(n)]
	for i in range(n):
		if int(arr[i]) < int(arr[0]) or int(arr[i]) > int(arr[n - 1]):
			continue
		k = bisect.bisect_right(a, int(arr[i]))
		if k == len(a):
			a.append(int(arr[i]))
		else:
			a[k] = int(arr[i])
	ans = n - len(a)
	return ans
(n, k) = input().split(' ')
n = int(n)
k = int(k)
ar = input().split(' ')
if k == 0:
	b = []
else:
	b = input().split(' ')
pd = True
b = [0] + b + [n + 1]
ar = [int(-1 * pow(10, 10))] + ar + [int(pow(10, 10))]
ans = 0
(h, j) = (int(b[0]), int(ar[0]))
for zc in b[1:]:
	zc = int(zc)
	if zc - h > int(ar[zc]) - j:
		pd = False
		break
	ans += js(ar[h:zc + 1])
	(h, j) = (zc, int(ar[zc]))
if pd:
	print(ans)
else:
	print(-1)
