#include <bits/stdc++.h>
using namespace std;
int main() {
  long long tc;
  cin >> tc;
  while (tc--) {
    long long n, a, b, sum = 0, id = 0;
    cin >> n >> a >> b;
    string s;
    cin >> s;
    long long c0 = 0, c1 = 0, lst = -1;
    for (long long i = 0; i < s.size(); i++) {
      if (s[i] == '1') {
        id = i;
        break;
      }
      c0++;
    }
    sum += c0 * b + (c0 + 1) * a;
    c0 = 0;
    for (long long i = id; i < s.size() - 1; i++) {
      if (s[i] == '0') {
        if (s[i - 1] == '0' || s[i + 1] == '0') {
          if (c1 > 0) sum += (c1 + 1) * b * 2 + (c1)*a;
          c1 = 0;
        } else
          c1++;
        c0++;
      } else {
        if (c0 > 1)
          sum += min((c0 - 1) * b * 2 + c0 * a,
                     (c0 - 1) * b + (c0 - 2) * a + 4 * a);
        lst = i;
        c1++;
        c0 = 0;
      }
    }
    c0 = n - lst - 1;
    if (c1 > 0) sum += (c1 + 1) * b * 2 + (c1)*a;
    sum += c0 * b + (c0 - 1) * a + a * 2;
    if (lst == -1) sum = (s.size() + 1) * b + s.size() * a;
    cout << sum << endl;
  }
  return 0;
}
