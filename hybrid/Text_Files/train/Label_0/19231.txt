#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[100], max = 0, min = 2000;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 2; i < n; i++) {
    for (int j = 1; j < n; j++) {
      if (a[j] != a[i] && a[j + 1] != a[i])
        if (a[j + 1] - a[j] > max) max = a[j + 1] - a[j];
      if (a[j + 1] == a[i])
        if (a[j + 2] - a[j] > max) max = a[j + 2] - a[j];
    }
    if (max < min) min = max;
    max = 0;
  }
  cout << min << endl;
  return 0;
}
