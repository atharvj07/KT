from collections import defaultdict
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	p = (m + 1) // 2
	k = defaultdict(int)
	ans = [0] * m
	gh = [(i, list(map(int, input().split(' ')))[1:]) for i in range(0, m)]
	flag = 0
	gh.sort(key=lambda day: len(day[1]))
	for (i, day) in gh:
		for v in day:
			if k[v] + 1 <= p:
				ans[i] = v
				k[v] += 1
				break
		else:
			print('NO')
			break
	else:
		print('YES')
		print(*ans)
