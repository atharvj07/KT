(n, m) = [int(i) for i in input().split(' ')]
arr = [int(i) for i in input().split(' ')]
inp_interval = []
for i in range(m):
	inp_interval.append([int(i) for i in input().split(' ')])
value_result_arr = []
for i in inp_interval:
	value_result_arr.append(sum(arr[i[0] - 1:i[1]]))
result = 0
for i in value_result_arr:
	if i > 0:
		result = result + i
print(result)
