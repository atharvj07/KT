import math
string = input()
numbers = string.split()
(a, b) = (int(numbers[0]), int(numbers[1]))
string = input()
s = sum(list(map(int, string.split())))
n = 0

def roundup(x):
	if x - math.floor(x) < 0.5:
		return math.floor(x)
	else:
		return math.ceil(x)
while roundup(s / a) < b:
	s += b
	a += 1
	n += 1
print(n)
