#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const int maxn = 100 + 5;
vector<int> col, row;
int n, m, a[maxn][maxn];
void doRow() {
  for (int i = 0; i < n; i++) {
    int _min = inf;
    for (int j = 0; j < m; j++) _min = min(_min, a[i][j]);
    for (int j = 0; j < m; j++) a[i][j] -= _min;
    while (_min--) row.push_back(i + 1);
  }
}
void doCol() {
  for (int j = 0; j < m; j++) {
    int _min = inf;
    for (int i = 0; i < n; i++) _min = min(_min, a[i][j]);
    for (int i = 0; i < n; i++) a[i][j] -= _min;
    while (_min--) col.push_back(j + 1);
  }
}
bool check() {
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (a[i][j]) return false;
  return true;
}
int main() {
  scanf("%d %d", &n, &m);
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) scanf("%d", &a[i][j]);
  if (n > m) {
    doCol();
    doRow();
  } else {
    doRow();
    doCol();
  }
  if (check()) {
    cout << col.size() + row.size() << endl;
    for (int i = 0; i < col.size(); i++) printf("col %d\n", col[i]);
    for (int i = 0; i < row.size(); i++) printf("row %d\n", row[i]);
  } else
    puts("-1");
}
