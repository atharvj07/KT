def func(arr, n, x):
	s = 0
	for i in range(n):
		if s + arr[i] == x:
			if i != n - 1:
				(arr[i], arr[i + 1]) = (arr[i + 1], arr[i])
				print('YES')
				print(' '.join(map(str, arr)))
			else:
				print('NO')
			break
		elif s + arr[i] > x:
			print('YES')
			print(' '.join(map(str, arr)))
			break
		else:
			s += arr[i]
			if i == n - 1:
				print('YES')
				print(' '.join(map(str, arr)))
t = int(input())
for _ in range(t):
	(n, x) = map(int, input().split())
	arr = list(map(int, input().split()))
	func(arr, n, x)
