import sys
import math
from collections import defaultdict, deque
import heapq
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
(pos, neg, zero) = ([], [], [])
for i in range(n):
	if arr[i] == 0:
		zero.append([arr[i], i + 1])
		continue
	if arr[i] > 0:
		pos.append([arr[i], i + 1])
		continue
	neg.append([arr[i], i + 1])
(n, m, k) = (len(pos), len(neg), len(zero))
if m > 0:
	num = max(neg)
	ind = 0
	for i in range(m):
		if num == neg[i]:
			ind = i
	(neg[0], neg[ind]) = (neg[ind], neg[0])
ans = []
j = 0
ch = True
if m % 2 != 0:
	if k > 0:
		ans.append([1, neg[0][1], zero[0][1]])
		j += 1
	else:
		ans.append([2, neg[0][1]])
		j += 1
		ch = False
for i in range(1, n):
	ans.append([1, pos[i][1], pos[0][1]])
for i in range(1, k):
	ans.append([1, zero[i][1], zero[0][1]])
for i in range(j + 1, m):
	ans.append([1, neg[i][1], neg[j][1]])
if ch:
	if len(ans) < n + m + k - 1 and k > 0:
		ans.append([2, zero[0][1]])
	if n > 0 and j < m:
		ans.append([1, pos[0][1], neg[j][1]])
elif n > 0:
	if j < m:
		ans.append([1, neg[j][1], pos[0][1]])
	if k > 0:
		ans.append([1, zero[0][1], pos[0][1]])
elif j < m:
	if k > 0:
		ans.append([1, zero[0][1], neg[j][1]])
n = len(ans)
for i in range(n):
	print(*ans[i])
