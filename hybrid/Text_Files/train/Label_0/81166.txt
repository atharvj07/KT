for _ in range(1):
	(add, xor_n) = map(int, input().split())
	if add < xor_n:
		print(0)
		continue
	d = add - xor_n
	if d % 2 == 1:
		print(0)
		continue
	d //= 2
	xor = bin(xor_n).replace('0b', '')
	sub = 0
	if xor_n & d == 0:
		if add == xor_n:
			sub = 2
		print(2 ** xor.count('1') - sub)
	else:
		print(0)
