(n, *price) = [int(x) for x in input().split()]
ipt = [int(x) for x in input().split()]
slice1 = ipt[:n // 2]
slice2 = ipt[n // 2 + n % 2:]
slice2.reverse()
eq = True
sum = 0
for (i, val) in enumerate(slice1):
	if val != slice2[i] and slice2[i] != 2 and (val != 2):
		eq = False
		break
	elif val == 2 and slice2[i] != 2:
		sum += price[slice2[i]]
	elif val != 2 and slice2[i] == 2:
		sum += price[val]
	elif slice2[i] == val == 2:
		sum += min(price) * 2
if eq:
	if n % 2 == 1 and ipt[n // 2] == 2:
		sum += min(price)
	print(sum)
else:
	print(-1)
