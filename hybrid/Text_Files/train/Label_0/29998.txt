#include <bits/stdc++.h>
int map[55][55];
int n, m, x, y, half, p;
int main() {
  scanf("%d", &n);
  half = p = (1 + n) / 2;
  x = 2;
  y = 1;
  for (int i = 1; i <= half; i++)
    for (int j = half - i + 1; j <= half + i - 1; j++) {
      map[i][j] = y;
      y += 2;
    }
  for (int i = half + 1, k = half - 1; i <= n; i++, k--)
    for (int j = half - k + 1; j <= half + k - 1; j++) {
      map[i][j] = y;
      y += 2;
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (!map[i][j]) {
        map[i][j] = x;
        x += 2;
      }
    }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      printf("%d ", map[i][j]);
    }
    printf("\n");
  }
  return 0;
}
