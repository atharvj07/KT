#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, arr[105], c, maxx = 1;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> arr[i];
  for (int i = 0; i < n - 1; i++) {
    c = 1;
    for (int j = i + 1; j < n; j++) {
      if (i != j) {
        if (arr[i] == arr[j]) c++;
      }
    }
    maxx = max(maxx, c);
  }
  cout << maxx << endl;
  return 0;
}
