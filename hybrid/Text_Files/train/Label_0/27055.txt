//repetition
//------------------------------------------
#define FOR(i,a,b) for(int i=(a);i<(b);++i)
#define REP(i,n)  FOR(i,0,n)

//constant
//--------------------------------------------
const double EPS = 1e-10;
const double PI  = acos(-1.0);

int main(){
  cin.tie(0);
  ios_base::sync_with_stdio(false);

  int N, M; cin >> N >> M;
  vector<bool> covered(N, false);
  REP(i,M){
	int a, l; cin >> a >> l;
	REP(i,l)
	  covered[(a+i)%N] = true;
  }

  for(int i=0;i<N && covered.back();++i){
	covered.insert(covered.begin(), true);
	covered.pop_back();
  }

  VI ans(N+1, 0);
  int num = 0;
  REP(i,N){
	if(covered[i]) ++num;
	else if(num > 0){
	  ans[num]++;
	  num = 0;
	}
  }
  if(num > 0)
	ans[num]++;
  
  for(int i=N;i>0;--i)
	if(ans[i] > 0)
	  cout << i << " " << ans[i] << endl;
  
  return 0;
}
