  class Vector implements Comparable<Vector> {
    int x, y, i;
    long l;

    public Vector(int i, int x, int y) {
      this.x = x;
      this.y = y;
      this.i = i;
      l = d2(x, y);
    }

    @Override public int compareTo(Vector v) {
      return Long.signum(v.l - l);
    }

    @Override public String toString() {
      return ("[i=" + i + ", x=" + x + ", y=" + y + ", l=" + l + "]");
    }
  }

  //-------------------------------------------------

  long d2(long x, long y) {
    return (x * x + y * y);
  }

  //-------------------------------------------------

  public void run() throws Exception {
    int n = nextInt(), a [] = new int [n], x, y;
    Vector [] va = new Vector [n];
    for (int i = 0; i < n; va[i] = new Vector(i, nextInt(), nextInt()), i++);

//    shuffle(va);
//    Arrays.sort(va);
//Arrays.stream(va).forEach(this::println);

    do {
      x = y = 0;

      shuffle(va);

      for (Vector v : va) {
        int x1 = x + v.x, y1 = y + v.y, x2 = x - v.x, y2 = y - v.y;
  
        if (d2(x1, y1) <= d2(x2, y2)) {
          x = x1;
          y = y1;
          a[v.i] = +1;
        } else {
          x = x2;
          y = y2;
          a[v.i] = -1;
        }
      }
    } while (d2(x, y) > 2_250_000_000_000L);

//println(x + "\t" + y + "\t" + Math.sqrt(d2(x, y)));
    IntStream.of(a).forEach(v -> print(v + " "));
  }

    pw.close();
    System.err.println("\n[Time : " + (System.currentTimeMillis() - startTime) + " ms]");
  }

  int gcd(int a, int b) {
    if (a == 0) return Math.abs(b); if (b == 0) return Math.abs(a);
    a = Math.abs(a); b = Math.abs(b);
    int az = Integer.numberOfTrailingZeros(a), bz = Integer.numberOfTrailingZeros(b);
    a >>>= az; b >>>= bz;
    while (a != b) {
      if (a > b) { a -= b; a >>>= Integer.numberOfTrailingZeros(a); }
            else { b -= a; b >>>= Integer.numberOfTrailingZeros(b); }
    }
    return (a << Math.min(az, bz));
  }

  long gcd(long a, long b) {
    if (a == 0) return Math.abs(b); if (b == 0) return Math.abs(a);
    a = Math.abs(a); b = Math.abs(b);
    int az = Long.numberOfTrailingZeros(a), bz = Long.numberOfTrailingZeros(b);
    a >>>= az; b >>>= bz;

  }

  void shuffle(int [] a) {
    Random r = new Random();
    for (int i = a.length - 1; i >= 0; i--) {
      int j = r.nextInt(a.length);
      int t = a[i]; a[i] = a[j]; a[j] = t;
    }
  }

  }
