import java.util.*;

public class CF253D {
	public static void main(String[] args) {
		Scanner input = new Scanner(System.in);
		int n = input.nextInt();
		List<Double> probs = new ArrayList<Double>();
		
		
		for (int i = 0; i < n; i++) {
			double p = input.nextDouble();
			if (p > 0.0)
				probs.add(p);
		}
		
		Collections.sort(probs);
		
		int size = probs.size();
		if (probs.get(size - 1) > 0.5) System.out.println(probs.get(size-1));
		
		else {
			double result = 0.0;
			
			double S = 0.0;
			int stop = size - 1;
			
			for (; stop >= 0; stop--) {
				double frac = probs.get(stop) / (1 - probs.get(stop));
				
				S += frac;
				
				if (S >= 1.0) break;				
			}
			if (stop == -1) stop = 0;
			
			for (int i = size - 1; i >= stop; i--) {
				double p = 1.0;
				for (int j = stop; j < size; j++) {
					if (j == i) p *= probs.get(j);
					else p *= (1 - probs.get(j));
				}
				result += p;
			}
			
			System.out.println(result);
		}		
				
	
		input.close();
	}
}
