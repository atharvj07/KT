from itertools import combinations

def solve(test_case):
	n = int(input())
	mat = []
	for i in range(n):
		mat.append([int(x) for x in input().split()])
	sets = []
	for j in range(5):
		temp = set()
		for i in range(n):
			if mat[i][j] == 1:
				temp.add(i)
		if len(temp) >= n / 2:
			sets.append(temp)
	if len(sets) < 2:
		print('NO')
		return
	combs = combinations(sets, 2)
	for comb in combs:
		if len(comb[0] | comb[1]) == n:
			print('YES')
			return
	print('NO')
t = int(input())
for i in range(t):
	solve(i + 1)
