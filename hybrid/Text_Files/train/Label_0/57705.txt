import java.util.Scanner;
import java.util.ArrayList;
public class CodeF

{
	public static void main(String args[])
	{
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		int k=sc.nextInt();
		int a[]=new int[k];
		for(int i=0;i<k;i++)
		{
			a[i]=sc.nextInt();
		}
		ArrayList<Integer> list=new ArrayList<Integer>();
		for(int i=1;i<=n;i++)
		{
			list.add(i);
		}
		int current=0;
		for(int i=0;i<k;i++)
		{
			current=(current+a[i])%(list.size());
			if(current==(list.size()-1))
			{
				System.out.print(list.get(current)+" ");
				list.remove(current);
				current=0;
			}
			else
			{
				System.out.print(list.get(current)+" ");
				list.remove(current);
			}
		}
	}
}
