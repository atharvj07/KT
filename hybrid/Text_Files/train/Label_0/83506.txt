(n, k) = map(int, input().split())
a = list(map(int, input().split()))
b = [0] * (pow(10, 6) + 5)
ans = [1, 1]
l = -1
cnt = 0
j = 0
for i in range(n):
	if b[a[i]] == 0:
		cnt += 1
	b[a[i]] += 1
	if cnt > k:
		while cnt > k:
			b[a[j]] -= 1
			if b[a[j]] == 0:
				cnt -= 1
			j += 1
	if i - j > l:
		ans = [j + 1, i + 1]
		l = i - j
print(*ans)
