cubes = int(input())
exists = []
for _ in range(cubes):
	exists.append([False] * 10)
for i in range(cubes):
	for s in input().split():
		side = int(s)
		exists[i][side] = True
p = []
from itertools import permutations
for c in range(cubes):
	p.append(permutations([x for x in range(cubes)], c + 1))
perms = []
for pe in p:
	perms.append([])
	for i in pe:
		perms[-1].append(i)
max = 0
for i in range(1, 999):
	number = i
	digits = []
	while number > 0:
		digits.append(number % 10)
		number //= 10
	found = False
	for p in perms[len(digits) - 1]:
		all_digits = True
		for d in range(len(digits)):
			if not exists[p[d]][digits[d]]:
				all_digits = False
				break
		if all_digits:
			found = True
			break
	if found:
		max = i
	else:
		break
print(max)
