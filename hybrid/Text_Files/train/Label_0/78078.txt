import math
t = int(input())
arr = []
for _ in range(t):
	arr.append(float(input()))
vis = [0] * t
for i in range(t):
	if arr[i] == math.floor(arr[i]):
		vis[i] = 1
arr = [math.floor(i) for i in arr]
x = sum(arr)
if x < 0:
	for i in range(t):
		if vis[i] == 0 and x < 0:
			arr[i] += 1
			x += 1
for i in range(t):
	print(arr[i])
