#include<bits/stdc++.h>
using namespace std;

#define ll long long
vector < ll > val; long long N;
void euclid(ll a , ll b , ll &x , ll &y){!b ? (x = 1 , y = 0) : (euclid(b , a % b , y , x) , y -= a / b * x);}

int main(){
	cin >> N; N *= 2; ll ans = 2 * N;
	for(int i = 2 ; 1ll * i * i <= N ; ++i)
		if(N % i == 0){ll pw = 1; while(N % i == 0){N /= i; pw *= i;} val.push_back(pw);}
	if(N != 1) val.push_back(N);
	for(int i = 1 ; i < 1 << val.size() ; ++i){
		ll p1 = 1 , p2 = 1; for(int j = 0 ; j < val.size() ; ++j) (i >> j & 1 ? p1 : p2) *= val[j];
		ll x , y; euclid(p1 , p2 , x , y); if(x <= 0) x += p2;
		ans = min(ans , p1 * x - 1);
	}
	cout << ans; return 0;
}
