import java.io.*;
import java.util.*;

public class F {
	public static void main (String[] args) { new F(); }

	public F() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("");

		int n = fs.nextInt(), m = fs.nextInt(), k = fs.nextInt();
		int[][] a = new int[n][m];
		int[][] pref = new int[n][m];
		for(int i = 0; i < n; i++)
			for(int j = 0; j < m; j++) {
				a[i][j] = fs.nextInt();
				pref[i][j] = a[i][j];
				if(j > 0) pref[i][j] += pref[i][j-1];
			}
		
		int[] dp = new int[m];
		for(int i = 0; i < m; i++) {
			dp[i] = query(pref[n-1], i, Math.min(m-1, i+k-1));
		}
		
		for(int i = n-2; i >= 0; i--) {
			int[] ndp = new int[m];
			ST st = new ST(0, m-1);
			for(int j = 0; j < m; j++) st.rangeAdd(j, j, dp[j]);
			for(int j = 0; j < k; j++) {
				st.rangeAdd(Math.max(0, j-k+1), j, -a[i+1][j]);
			}
			for(int j = 0; j < m; j++) {
				int val = query(pref[i], j, Math.min(m-1, j+k-1));
				val += query(pref[i+1], j, Math.min(m-1, j+k-1));
				val += st.rangeMax(0, m-1);
				ndp[j] = val;
				
				int l1 = Math.max(0, j-k+1);
				st.rangeAdd(l1, j, a[i+1][j]);
				if(j+k < m) {
					st.rangeAdd(j+1, j+k, -a[i+1][j+k]);
				}
			}
			
			dp = ndp;
		}
		
		int res = 0;
		for(int i : dp) res = Math.max(res, i);
		out.println(res);
		
		out.close();
	}
	
	int query(int[] a, int l, int r) {
		int res = a[r];
		if(l > 0) res -= a[l-1];
		return res;
	}
	
	class ST {
		ST left, right;
		int L, R, lazy, max;
		ST(int a, int b) {
			L = a; R = b;
			if(L != R) {
				int mid = (L+R)/2;
				left = new ST(a, mid);
				right = new ST(mid+1, b);
			}
		}
		void rangeAdd(int ll, int rr, int val) {
			if(R < ll || rr < L) return;
			if(ll <= L && R <= rr) {
				lazy += val;
				return;
			}
			prop();
			left.rangeAdd(ll, rr, val);
			right.rangeAdd(ll, rr, val);
			
			max = Math.max(left.getMax(), right.getMax());
		}
		int rangeMax(int ll, int rr) {
			if(R < ll || rr < L) return 0;
			if(ll <= L && R <= rr) {
				return getMax();
			}
			prop();
			int lf = left.rangeMax(ll, rr);
			int rg = right.rangeMax(ll, rr);
			
			max = Math.max(left.getMax(), right.getMax());
			return Math.max(lf, rg);
		}
		void prop() {
			left.lazy += lazy;
			right.lazy += lazy;
			lazy = 0;
		}
		int getMax() {
			return max+lazy;
		}
	}

	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) {
			try {
				in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
			}
			catch (Exception e) {
				in = new BufferedInputStream(System.in, BS);
			}
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}
		
		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}
		
	}

}
