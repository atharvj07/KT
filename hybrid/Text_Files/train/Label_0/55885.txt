import java.util.Arrays;
import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int round = 0; round < t; ++round) {
            go(in);
        }
    }

    static void go(final Scanner in) {
        int n = in.nextInt();
        int x = in.nextInt();
        int maxd = 0;
        DH[] dh = new DH[n];

        for (int i = 0; i < n; ++i) {
            dh[i] = new DH();
            dh[i].d = in.nextInt();
            maxd = Math.max(maxd, dh[i].d);
            dh[i].h = in.nextInt();
        }
        if (maxd >= x) {
            System.out.println(1);
            return;
        }

        Arrays.sort(dh, (o1, o2) -> {
            int c1 = o1.d - o1.h;
            int c2 = o2.d - o2.h;
            if (c1 == c2) {
                return o2.d - o1.d;
            } else {
                return c2 - c1;
            }
        });

        // d < x
        int d = dh[0].d;
        int h = dh[0].h;
        if (d <= h) {
            System.out.println(-1);
        } else {
            int gap = d - h;
            System.out.println((x - maxd - 1) / gap + 2);
        }
    }

    static class DH {
        public int d;
        public int h;
    }
}
