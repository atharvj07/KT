import sys, math
MOD = 10 ** 9 + 7

def max_points(time, easy_req, hard_req):
	if easy_req * a + hard_req * b > time or time < 0 or time > t:
		return 0
	else:
		time -= easy_req * a + hard_req * b
	ans = easy_req + hard_req
	ans += min(easy - easy_req, time // a)
	time -= a * min(easy - easy_req, time // a)
	ans += min(hard - hard_req, time // b)
	return ans
for _ in range(int(input())):
	(n, t, a, b) = map(int, input().split())
	x = list(map(int, input().split()))
	y = list(map(int, input().split()))
	easy = x.count(0)
	hard = n - easy
	ans = max_points(t, easy, hard)
	req = {ys: [0, 0] for ys in y}
	req[0] = [0, 0]
	for i in range(n):
		req[y[i]][x[i]] += 1
	times = sorted(req.keys())
	for i in range(1, len(times)):
		req[times[i]][0] += req[times[i - 1]][0]
		req[times[i]][1] += req[times[i - 1]][1]
	for i in range(1, len(times)):
		req[times[i] - 1] = req[times[i - 1]][:]
	for time in req:
		ans = max(ans, max_points(time, req[time][0], req[time][1]))
	print(ans)
