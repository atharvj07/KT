#include <bits/stdc++.h>
using namespace std;
int par[103], sz[103];
int find(int v) {
  if (par[v] == v) return v;
  return par[v] = find(par[v]);
}
void union_set(int u, int v) {
  u = find(u);
  v = find(v);
  if (u != v) {
    if (sz[u] < sz[v]) {
      swap(u, v);
    }
    par[v] = u;
    sz[u] += sz[v];
  }
}
int main() {
  int n;
  cin >> n;
  int a[n + 1];
  for (int i = 1; i <= n; i++) {
    par[i] = i;
    sz[i] = 1;
  }
  for (int i = 0; i < n; i++) cin >> a[i + 1];
  int b[n + 1];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) {
    if (i + b[i] <= n) union_set(i, i + b[i]);
    if (i - b[i] >= 1) union_set(i, i - b[i]);
  }
  int fl = 0;
  for (int i = 1; i <= n; i++) {
    if (i != a[i]) {
      if (find(i) != find(a[i])) {
        fl = 1;
        break;
      }
    }
  }
  if (!fl)
    cout << "YES\n";
  else
    cout << "NO\n";
  return 0;
}
