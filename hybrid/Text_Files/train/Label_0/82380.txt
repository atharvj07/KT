def rec(c, m, p, v):
	ret = 0
	ret += p * 0
	if 0 < m <= v + 1e-05:
		tmp = 0
		if c > 0:
			tmp = rec(c + m / 2, 0, p + m / 2, v)
		else:
			tmp = rec(0, 0, 1, v)
		ret += m * tmp
	elif m > v:
		tmp = 0
		if c > 0:
			tmp = rec(c + v / 2, m - v, p + v / 2, v)
		else:
			tmp = rec(0, m - v, p + v, v)
		ret += m * tmp
	if 0 < c <= v + 1e-05:
		tmp = 0
		if m > 0:
			tmp = rec(0, m + c / 2, p + c / 2, v)
		else:
			tmp = rec(0, 0, 1, v)
		ret += c * tmp
	elif c > v:
		tmp = 0
		if m > 0:
			tmp = rec(c - v, m + v / 2, p + v / 2, v)
		else:
			tmp = rec(c - v, 0, p + v, v)
		ret += c * tmp
	return ret + c + m
t = int(input())
for case in range(t):
	(c, m, p, v) = map(float, input().split())
	print(1 + rec(c, m, p, v))
