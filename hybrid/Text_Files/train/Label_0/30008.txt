#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int r[n], b[n];
  int x = 0, y = 0;
  for (int i = 0; i < n; i++) {
    cin >> r[i];
    if (r[i] == 1) x++;
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
    if (b[i] == 1) y++;
  }
  if (x > y)
    cout << 1 << "\n";
  else {
    x = 0;
    y = 0;
    for (int i = 0; i < n; i++) {
      if ((r[i] == 1) && (b[i] == 0))
        x++;
      else if ((r[i] == 0) && (b[i] == 1))
        y++;
    }
    if ((x == 0) && (y == 0))
      cout << -1 << "\n";
    else {
      if (x == 0)
        cout << -1 << "\n";
      else {
        int ans = (y / x) + 1;
        cout << ans << "\n";
      }
    }
  }
}
