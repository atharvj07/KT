#include <bits/stdc++.h>
using namespace std;
struct aa {
  int x, y, z, id;
} a[210000];
bool used[210000];
int n;
bool cmpxyz(aa a, aa b) {
  if (a.x != b.x) return a.x < b.x;
  if (a.y != b.y) return a.y < b.y;
  return a.z < b.z;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d%d%d", &a[i].x, &a[i].y, &a[i].z);
    a[i].id = i;
  }
  sort(a + 1, a + n + 1, cmpxyz);
  for (int i = 1; i < n; i++) {
    if (a[i].x == a[i + 1].x && a[i].y == a[i + 1].y) {
      printf("%d %d\n", a[i].id, a[i + 1].id);
      used[a[i].id] = used[a[i + 1].id] = true;
      i++;
    }
  }
  int len = 0;
  for (int i = 1; i <= n; i++)
    if (!used[a[i].id]) {
      a[++len] = a[i];
    }
  n = len;
  sort(a + 1, a + n + 1, cmpxyz);
  for (int i = 1; i < n; i++) {
    if (a[i].x == a[i + 1].x) {
      printf("%d %d\n", a[i].id, a[i + 1].id);
      used[a[i].id] = used[a[i + 1].id] = true;
      i++;
    }
  }
  len = 0;
  for (int i = 1; i <= n; i++)
    if (!used[a[i].id]) {
      a[++len] = a[i];
    }
  n = len;
  sort(a + 1, a + n + 1, cmpxyz);
  for (int i = 1; i <= n; i += 2) {
    printf("%d %d\n", a[i].id, a[i + 1].id);
  }
}
