length = len(zebra)

def checkLongest(zebra):
	longest = 0
	curr = 0
	prev = 0
	for i in zebra:
		if ord(i) != prev:
			curr += 1
		else:
			if curr > longest:
				longest = curr
			curr = 1
		prev = ord(i)
	if curr > longest:
		longest = curr
	return longest

def finish(zebra, longest):
	temp = checkLongest(zebra)
	if temp > longest:
		return temp
	else:
		return longest

				string = string[0:i][::-1] + string[:i - 1:-1]
				return flip(string)
