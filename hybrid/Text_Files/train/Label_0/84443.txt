n = int(input())
time = list(map(int, input().split()))
i = 0
s = 0
while i < len(time):
	s += time[i]
	i += 1
m = int(input())
flag = False
while m > 0:
	(l, r) = map(int, input().split())
	if l > s:
		print(l)
		flag = True
		break
	elif s >= l and s <= r:
		print(s)
		flag = True
		break
	m -= 1
if flag == False:
	print('-1')
