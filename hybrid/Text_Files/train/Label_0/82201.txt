n = int(input())
l = list(map(int, input().split()))
l.sort()
c = 0
for i in range(1, 100000):
	if i ** (n - 1) >= l[n - 1]:
		c = i
		break
l1 = []
cm = 100000000000000
for i in range(1, c + 1):
	c1 = 0
	for j in range(n):

		c1 += c2
		if cm < c1:
			break
	if cm > c1:
		cm = c1
