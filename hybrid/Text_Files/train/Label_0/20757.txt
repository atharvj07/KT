#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5;
int a[MAXN];
int main() {
  fill(a, a + MAXN, 0);
  int n, x;
  cin >> n >> x;
  if (n == 1) {
    cout << "YES" << endl << x << endl;
  } else if (n == 2 && x == 0) {
    cout << "NO" << endl;
  } else {
    a[n] = x;
    for (int i = 1; i <= n - 1; i++) {
      a[i] = i;
      a[n] ^= i;
    }
    if (a[n] <= n - 1) {
      if (a[n] == a[n - 1]) {
        a[n] ^= (1 << 17);
        a[n - 2] ^= (1 << 17);
      } else {
        a[n] ^= (1 << 17);
        a[n - 1] ^= (1 << 17);
      }
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) cout << a[i] << " ";
    cout << endl;
  }
}
