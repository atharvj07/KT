import java.util.*;

public class booksexchange {
    public static void main(String[] args) {
        Scanner scn = new Scanner(System.in);
        int q = scn.nextInt();
        while (q-- > 0) {
            int n = scn.nextInt();
            int[][] arr = new int[3][n + 1];
            for (int i = 1; i <= n; i++) {
                arr[0][i] = scn.nextInt();
                arr[1][i] = i;
                arr[2][i] = 0;
            }
            for (int c = 0; c < n; c++) {
                int[] ans = new int[n + 1];
                for (int i = 1; i <= n; i++)
                    ans[i] = arr[1][i];
                for (int i = 1; i <= n; i++) {
                    if (arr[2][i] == 0 || arr[1][i] != i) {
                        ans[arr[0][i]] = arr[1][i];
                        arr[2][i]++;
                    }
                }
                for (int i = 1; i <= n; i++) {
                    arr[1][i] = ans[i];
                }
            }
            for (int i = 1; i <= n; i++) {
                System.out.print(arr[2][i] + " ");
            }
            System.out.println();
        }
    }
}
