#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  char s[n + 1];
  for (int i = 1; i < (n + 1); i++) cin >> s[i];
  long long left = k - 1, right = n - k;
  if (left <= right) {
    while (left) {
      cout << "LEFT" << endl;
      left--;
    }
    long long o = 1;
    while (o < n + 1) {
      cout << "PRINT " << s[o] << endl;
      if (o == n) break;
      cout << "RIGHT" << endl;
      o++;
    }
  } else {
    while (right) {
      cout << "RIGHT" << endl;
      right--;
    }
    long long o = n;
    while (o >= 0) {
      cout << "PRINT " << s[o] << endl;
      if (o == 1) break;
      cout << "LEFT" << endl;
      o--;
    }
  }
}
