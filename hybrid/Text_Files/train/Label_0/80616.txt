n = int(input())
a = list(map(int, input().split()))
cnt4 = 0
cnt8 = 0
cnt15 = 0
cnt16 = 0
cnt23 = 0
cnt42 = 0
for i in a:
	if i == 4:
		cnt4 += 1
	if i == 8 and cnt4 > cnt8:
		cnt8 += 1
	if i == 15 and cnt8 > cnt15:
		cnt15 += 1
	if i == 16 and cnt15 > cnt16:
		cnt16 += 1
	if i == 23 and cnt16 > cnt23:
		cnt23 += 1
	if i == 42 and cnt23 > cnt42:
		cnt42 += 1
print(abs(n - cnt42 * 6))
