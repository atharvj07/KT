import math
t = int(input())
for i in range(t):
	(a, b, c) = map(int, input().split())
	if a == b and c % 2 == 0:
		print('YES')
	elif a == c and b % 2 == 0:
		print('YES')
	elif b == c and a % 2 == 0:
		print('YES')
	elif max(a, b, c) == a + b + c - max(a, b, c):
		print('YES')
	else:
		print('NO')
