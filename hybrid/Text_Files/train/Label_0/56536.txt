import java.util.*;


public class Main {
    public Main(){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long k=sc.nextLong();
        long[] a=new long[n];
        for(int i=0;i<n;i++) a[i]=sc.nextLong();
        long A=sc.nextLong();
        long[] c=new long[n];
        for(int i=0;i<n;i++) c[i]=sc.nextLong();
        long ans=0;
        PriorityQueue<Long> pq= new PriorityQueue<Long>();
        for(int i=0;i<n&&ans>=0;i++){
            pq.add(c[i]);
            while(k<a[i]){
                if(pq.size()>0){
                    ans+=pq.peek();
                    pq.remove();
                    k+=A;
                }else{
                    ans=-1;
                    break;
                }
            }
        }
        System.out.println(ans);

    }

    public static void main(String[] args) {
        Main main=new Main();

    }

}
