#include <iostream>
#include <cstdio>
using namespace std;
double a[200005],qzh[200005];
int main(int argc, char** argv) {
	int n,k;
	cin >> n >> k;
	for(int i=1;i<=n;i++)
	{
		cin >> a[i];
		a[i]=(a[i]+1)/(double)2;
		qzh[i]=qzh[i-1]+a[i];
	}
	double ans=0;
	for(int i=k;i<=n;i++)
		ans=max(ans,qzh[i]-qzh[i-k]);
	printf("%.1000lf",ans);
	return 0;
}
