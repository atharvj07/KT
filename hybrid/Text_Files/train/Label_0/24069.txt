#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 10;
int n, a, D[maxn], c[maxn], h[maxn];
long long s[maxn], mx[maxn][20], mn[maxn][20];
pair<int, int> d[maxn];
set<int> st;
void rmq_init() {
  for (int i = 1, j = 0; i <= n; h[i++] = j)
    while ((1 << (j + 1)) <= i) j++;
  for (int i = 0; i <= n; i++) mx[i][0] = mn[i][0] = s[i];
  for (int t = 1; (1 << t) <= (n + 1); t++) {
    for (int l = 0, r = l + (1 << (t - 1)); l + (1 << t) - 1 <= n; l++, r++) {
      mx[l][t] = max(mx[l][t - 1], mx[r][t - 1]);
      mn[l][t] = min(mn[l][t - 1], mn[r][t - 1]);
    }
  }
}
long long qmax(long long l, long long r) {
  if (l > r) return 0;
  int t = h[r - l];
  return max(mx[l][t], mx[r - (1 << t) + 1][t]);
}
long long qmin(long long l, long long r) {
  if (l > r) return 0;
  int t = h[r - l];
  return min(mn[l][t], mn[r - (1 << t) + 1][t]);
}
int main() {
  long long ans = 0;
  scanf("%d", &n);
  scanf("%d", &a);
  st.insert(0);
  st.insert(n + 1);
  for (int i = 1; i <= n; i++)
    scanf("%d", &D[i]), scanf("%d", &c[i]), st.insert(i),
        c[i] = a - c[i], ans = max(1LL * c[i], ans);
  for (int i = 1; i <= n; i++) s[i] = s[i - 1] + c[i];
  for (int i = 2; i <= n; i++) d[i].first = D[i] - D[i - 1], d[i].second = i;
  rmq_init();
  sort(d + 2, d + 1 + n);
  for (int i = 2, j = 2; i <= n; i++) {
    while (d[j].first == d[i].first && j <= n) st.erase(d[j++].second);
    int l = *(--st.lower_bound(d[i].second)),
        r = *(st.lower_bound(d[i].second));
    long long tmp = qmax(d[i].second, r - 1) - qmin(l - 1, d[i].second) -
                    1LL * d[i].first * d[i].first;
    ans = max(ans, tmp);
  }
  cout << ans << endl;
  return 0;
}
