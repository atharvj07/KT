import bisect
t = int(input())
for you in range(t):
	l = input().split()
	n = int(l[0])
	p = int(l[1])
	k = int(l[2])
	l = input().split()
	li = [int(i) for i in l]
	li.sort()
	sum1 = []
	sum2 = []
	sumi1 = 0
	sumi2 = 0
	for i in range(n):
		if i % 2 == 1:
			sumi1 += li[i]
			sum1.append(sumi1)
		else:
			sumi2 += li[i]
			sum2.append(sumi2)
	ans1 = 2 * bisect.bisect(sum1, p)
	ans2 = 2 * bisect.bisect(sum2, p) - 1
	print(max(ans1, ans2))
