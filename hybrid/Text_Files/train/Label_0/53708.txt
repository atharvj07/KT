    void solve(int TC) throws Exception{
        int N = ni();
        long[] A = new long[1+N], sum = new long[1+N];
        for(int i = 1; i<= N; i++)sum[i] = sum[i-1]+(A[i] = ni());
        long ans = 0;
        for(int i = 1; i<= N; i++)ans += A[i]*i;
        long delta = 0;
        Line[] hull = new Line[N];
        int sz = 0;
        for(int i = 1; i<= N; i++){
            if(i > 1)delta = Math.max(delta, -A[i]*i+sum[i-1]+eval(hull, sz, A[i]));
            sz = addLine(hull, sz, new Line(i, -sum[i-1]));
        }
        sz = 0;
        for(int i = N; i >= 1; i--){
            if(i < N)delta = Math.max(delta, sum[i]-i*A[i]+eval(hull, sz, -A[i]));
            sz = addLine(hull, sz, new Line(-i, -sum[i]));
        }
        pn(ans+delta);
    }
    long eval(Line[] hull, int sz, long x){
        int lo = 1, hi = sz-1;
        if(sz == 1)return hull[0].eval(x);
        while(lo+1 < hi){
            int mid = lo+(hi-lo)/2;
            if(hull[mid-1].inter(hull[mid]) <= x)lo = mid;
            else hi = mid;
        }
        if(hull[hi-1].inter(hull[hi]) <= x)lo = hi;
        return Math.max(hull[0].eval(x), hull[lo].eval(x));
    }
    int addLine(Line[] hull, int sz, Line l){
        while(sz >= 2 && useless(hull[sz-2], hull[sz-1], l))sz--;
        hull[sz++] = l;
        return sz;
    }
    boolean useless(Line l1, Line l2, Line l3){
        return (l1.c-l3.c)*(l2.m-l1.m) <= (l1.c-l2.c)*(l3.m-l1.m);
    }
    class Line{
        long m, c;
        public Line(long M, long C){
            m = M;c = C;
        }
        long eval(long x){return m*x+c;}
        double inter(Line l){return (c-l.c)*1.0/(l.m-m);}
    }
    class CHT {
        //http://codeforces.com/contest/932/submission/35323630
        public static final int MIN = -1, MAX = 1;
        public TreeSet<CHT.Line> hull;
        int type;
        boolean query = false;
        Comparator<CHT.Line> comp = new Comparator<CHT.Line>() {
            public int compare(CHT.Line a, CHT.Line b) {
                if (!query) return type * Long.compare(a.m, b.m);
                if (a.left == b.left)
                    return Long.compare(a.m, b.m);
                return Double.compare(a.left, b.left);
            }
        };
        public CHT(final int type) {
            this.type = type;
            hull = new TreeSet<>(comp);
        }
        public void add(long m, long b) {
            add(new CHT.Line(m, b));
        }
        public void add(CHT.Line a) {
            CHT.Line[] LR = {hull.lower(a), hull.ceiling(a)};
            for (int i = 0; i < 2; i++)
                if (LR[i] != null && LR[i].m == a.m) {
                    if (type == 1 && LR[i].b >= a.b)
                        return;
                    if (type == -1 && LR[i].b <= a.b)
                        return;
                    remove(LR[i]);
                }
            hull.add(a);
            CHT.Line L = hull.lower(a), R = hull.higher(a);
            if (L != null && R != null && a.inter(R) <= R.left) {
                hull.remove(a);
                return;
            }
            CHT.Line LL = (L != null) ? hull.lower(L) : null;
            CHT.Line RR = (R != null) ? hull.higher(R) : null;
            if (L != null) a.left = a.inter(L);
            if (R != null) R.left = a.inter(R);
            while (LL != null && L.left >= a.inter(LL)) {
                remove(L);
                a.left = a.inter(L = LL);
                LL = hull.lower(L);
            }
            while (RR != null && R.inter(RR) <= a.inter(RR)) {
                remove(R);
                RR.left = a.inter(R = RR);
                RR = hull.higher(R);
            }
        }
        public long query(long x) {
            CHT.Line temp = new CHT.Line(0, 0, 0);
            temp.left = x;
            query = true;
            CHT.Line cur = hull.floor(temp);
            if(cur == null)return -1*type*IINF;
            //Debug.print(x, temp, hull.floor(temp));
            long ans = cur.eval(x);
            query = false;
            return ans;
        }
        private void remove(CHT.Line x) {
            hull.remove(x);
        }
        public int size() {
            return hull.size();
        }
        public class Line {
            long m;
            long b;
            double left = Long.MIN_VALUE;
            public Line(long m, long x, long y) {
                this.m = m;
                this.b = -m * x + y;
            }
            public Line(long m, long b) {
                this.m = m;
                this.b = b;
            }
            public long eval(long x) {
                return m * x + b;
            }
            public double inter(CHT.Line x) {
                return (double) (x.b - this.b) / (double) (this.m - x.m);
            }
        }
    }

    int[][] make(int n, int e, int[] from, int[] to, boolean f){
        int[][] g = new int[n][];int[]cnt = new int[n];
        for(int i = 0; i< e; i++){
            cnt[from[i]]++;
            if(f)cnt[to[i]]++;
        }
        for(int i = 0; i< n; i++)g[i] = new int[cnt[i]];
        for(int i = 0; i< e; i++){
            g[from[i]][--cnt[from[i]]] = to[i];
            if(f)g[to[i]][--cnt[to[i]]] = from[i];
        }
        return g;
    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }

        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }

        String nextLine() throws Exception{
            String str;
            try{
                str = br.readLine();
            }catch (IOException e){
                throw new Exception(e.toString());
            }
            return str;
        }
    }
