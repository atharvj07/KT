#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;
const int Max = 10000;
int n, m, s, t, S, T;
int U[1003], V[1003], W[1003], id[1003];
int cnt = 1, head[105];
struct Edge {
  int nxt, to, w;
} e[4003];
int dep[105], cur[105], deg[105];
bool v[105];
void add(int u, int v, int w) {
  e[++cnt] = (Edge){head[u], v, w}, head[u] = cnt;
  e[++cnt] = (Edge){head[v], u, 0}, head[v] = cnt;
}
bool bfs() {
  queue<int> q;
  memset(dep, 0, sizeof(dep));
  dep[S] = 1, q.push(S);
  while (!q.empty()) {
    int x = q.front();
    q.pop();
    for (int i = head[x]; i; i = e[i].nxt) {
      int to = e[i].to;
      if (e[i].w && !dep[to]) dep[to] = dep[x] + 1, q.push(to);
    }
  }
  return dep[T];
}
int dfs(int x, int flow) {
  if (x == T) return flow;
  int used = 0;
  for (int &i = cur[x]; i; i = e[i].nxt) {
    int to = e[i].to;
    if (e[i].w && dep[to] == dep[x] + 1) {
      int k = dfs(to, min(flow - used, e[i].w));
      e[i].w -= k, e[i ^ 1].w += k;
      used += k;
    }
  }
  return used;
}
int dinic() {
  int ans = 0;
  while (bfs()) {
    memcpy(cur, head, sizeof(cur));
    int x;
    while ((x = dfs(S, inf))) ans += x;
  }
  return ans;
}
int main() {
  scanf("%d%d%d%d", &n, &m, &s, &t);
  for (int i = 1; i <= m; i++) scanf("%d%d%d", &U[i], &V[i], &W[i]);
  for (int i = 1; i <= m; i++) {
    if (W[i] == 0)
      add(U[i], V[i], inf);
    else
      add(U[i], V[i], 1), add(V[i], U[i], inf);
  }
  S = s, T = t;
  printf("%d\n", dinic());
  for (int i = 1; i <= n; i++)
    if (dep[i] == 0) v[i] = true;
  cnt = 1, memset(head, 0, sizeof(head));
  S = n + 1, T = S + 1;
  for (int i = 1; i <= m; i++)
    if (W[i]) {
      id[i] = cnt + 1;
      add(U[i], V[i], Max);
      deg[U[i]]--, deg[V[i]]++;
    }
  for (int i = 1; i <= n; i++) {
    if (deg[i] > 0)
      add(S, i, deg[i]);
    else
      add(i, T, -deg[i]);
  }
  add(t, s, inf);
  dinic();
  for (int i = 1; i <= m; i++) {
    if (W[i] == 0)
      puts("0 1");
    else {
      int num = Max - e[id[i]].w + 1;
      if (!v[U[i]] && v[V[i]])
        printf("%d %d\n", num, num);
      else
        printf("%d %d\n", num, num + 1);
    }
  }
  return 0;
}
