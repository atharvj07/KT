public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        MyScan in = new MyScan(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskB solver = new TaskB();

    static class TaskB {
        public void solve(int testNumber, MyScan in, PrintWriter out) {
            int n = in.nextInt();
            String s1 = in.next();
            int m1 = v(s1);
            int m2 = v(in.next());
            int m3 = v(in.next());


            if (n != 1) {
                m1 = Math.min(s1.length(), m1 + n);
                m2 = Math.min(s1.length(), m2 + n);
                m3 = Math.min(s1.length(), m3 + n);
            } else {
                if (m1 == s1.length()) {
                    m1--;
                } else {
                    m1++;
                }

                if (m2 == s1.length()) {
                    m2--;
                } else {
                    m2++;
                }

                if (m3 == s1.length()) {
                    m3--;
                } else {
                    m3++;
                }
            }


            if (m1 > m2 && m1 > m3) {
                out.println("Kuro");
                return;
            }
            if (m2 > m1 && m2 > m3) {
                out.println("Shiro");
                return;
            }
            if (m3 > m1 && m3 > m2) {
                out.println("Katie");
                return;
            }
            out.println("Draw");
        }

        private int v(String next) {

            Map<Character, Integer> ret = new HashMap<>();

            for (char k : next.toCharArray()) {
                ret.merge(k, 1, Math::addExact);
            }

            return ret.values().stream().mapToInt(t -> t).max().getAsInt();


        }

    }

    static class MyScan {
        private final InputStream in;
        private byte[] inbuf = new byte[1024];
        public int lenbuf = 0;
        public int ptrbuf = 0;

        public MyScan(InputStream in) {
            this.in = in;
        }

        private int readByte() {
            if (lenbuf == -1) throw new InputMismatchException();
            if (ptrbuf >= lenbuf) {
                ptrbuf = 0;
                try {
                    lenbuf = in.read(inbuf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (lenbuf <= 0) return -1;
            }
            return inbuf[ptrbuf++];
        }

        public boolean isSpaceChar(int c) {
            return !(c >= 33 && c <= 126);
        }

        public int skip() {
            int b;
            while ((b = readByte()) != -1 && isSpaceChar(b)) ;
            return b;
        }

        public String next() {
            int b = skip();
            StringBuilder sb = new StringBuilder();
            while (!(isSpaceChar(b))) {
                sb.appendCodePoint(b);
                b = readByte();
            }
            return sb.toString();
        }

        public int nextInt() {
            int num = 0, b;
            boolean minus = false;
            while ((b = readByte()) != -1 && !((b >= '0' && b <= '9') || b == '-')) ;
            if (b == '-') {
                minus = true;
                b = readByte();
            }

            while (true) {
                if (b >= '0' && b <= '9') {
                    num = num * 10 + (b - '0');
                } else {
                    return minus ? -num : num;
                }
                b = readByte();
            }
        }

    }
}
