#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
const int INF = 1e9 + 5;
const double PI = acos(-1);
const int X[] = {1, -1, 0, 0};
const int Y[] = {0, 0, 1, -1};
const int mod = 1e9 + 7;
int n, a[N], b[N], ans, p[N], bb[N], t[N * 4];
void upd(int v, int l, int r, int pos) {
  if (l == r) {
    t[v] = 1;
    return;
  }
  int md = (l + r) / 2;
  if (pos <= md) {
    upd(v + v, l, md, pos);
  } else {
    upd(v + v + 1, md + 1, r, pos);
  }
  t[v] = t[v + v] + t[v + v + 1];
}
int get(int v, int tl, int tr, int l, int r) {
  if (tl > r || tr < l) {
    return 0;
  }
  if (tl >= l && tr <= r) {
    return t[v];
  }
  int md = (tl + tr) / 2;
  return get(v + v, tl, md, l, r) + get(v + v + 1, md + 1, tr, l, r);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    p[a[i]] = i;
  }
  for (int i = 1; i <= n; i++) {
    cin >> b[i];
    bb[i] = p[b[i]];
  }
  for (int i = n; i >= 1; i--) {
    if (get(1, 1, n, 1, bb[i]) > 0) {
      ++ans;
    }
    upd(1, 1, n, bb[i]);
  }
  cout << ans;
  return 0;
}
