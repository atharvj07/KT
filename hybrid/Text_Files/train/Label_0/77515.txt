import operator
import sys
a = list(set(map(int, sys.stdin.readline().split())))
n = int(input())
c2 = list(set(map(int, sys.stdin.readline().split())))
s = 0
v = [0] * (len(c2) * len(a))
index = 0
n = len(c2)
c = [0] * n
qm = 1000000000.0 + 1
for q in range(len(a)):
	for i in range(len(c2)):
		v[i * (len(a) + 0) + q] = [c2[i] - a[q], i]
v.sort(key=operator.itemgetter(0))
for i in range(len(v)):
	c[v[i][1]] += 1
	if c[v[i][1]] == 1:
		s += 1
	if s == n:
		while c[v[index][1]] > 1:
			c[v[index][1]] -= 1
			index += 1
		if v[i][0] - v[index][0] < qm:
			qm = v[i][0] - v[index][0]
		c[v[index][1]] -= 1
		if c[v[index][1]] == 0:
			s -= 1
		index += 1
sys.stdout.write(str(qm))
