import sys
sys.setrecursionlimit(10 ** 7)
input = sys.stdin.readline
f_inf = float('inf')
mod = 10 ** 9 + 7

def resolve():
	t = int(input())
	for _ in range(t):
		(n, m) = map(int, input().split())
		row = tuple((tuple(map(int, input().split())) for _ in range(n)))
		col = tuple((tuple(map(int, input().split())) for _ in range(m)))
		row_nums = [0] * (n * m + 1)
		for i in range(n):
			for j in row[i]:
				row_nums[j] = i
		res = [[] for _ in range(n)]
		for i in range(n):
			vi = col[0][i]
			row_num = row_nums[vi]
			res[i] = row[row_num]
		for i in res:
			print(*i)
resolve()
