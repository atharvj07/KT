t = input()
t = int(t)
for i in range(t):
	list1 = list()
	list2 = list()
	str1 = input()
	list3 = str1.split()
	n = int(list3[0])
	a = int(list3[1])
	b = int(list3[2])
	for j in range(b):
		list1.append(chr(j + 97))
	k = b
	list2 = list1[:]
	some = -1
	while k < a:
		list1.append(list2[some])
		if some * -1 != b:
			some -= 1
		else:
			some = -1
		k += 1
	some = 0
	while k < n:
		list1.append(list1[some])
		if some + 1 != a:
			some += 1
		else:
			some = 0
		k += 1
	str2 = ''.join(list1)
	print(str2)
