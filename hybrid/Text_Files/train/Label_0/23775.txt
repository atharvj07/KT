#include <bits/stdc++.h>
using namespace std;
int a[100001], n;
inline long long sum(int t) {
  long long s = 0;
  for (int i = 1; i <= n; ++i) s += min(a[i], t);
  return s;
}
int main() {
  int i, j, l, r, m, t = 1;
  int nr = 0;
  long long k, s = 0, smin;
  scanf("%d %I64d", &n, &k);
  for (i = 1; i <= n; ++i) {
    scanf("%d", &a[i]);
    s += a[i];
  }
  if (s < k)
    printf("-1");
  else if (s == k)
    printf("");
  else {
    l = 0;
    r = 1000000000;
    while (l + 1 < r) {
      m = (l + r) >> 1;
      if (sum(m) <= k)
        l = m;
      else
        r = m;
    }
    k -= sum(l);
    if (s == k)
      printf("");
    else {
      for (i = 1; i <= n; ++i) {
        if (k == 0) break;
        if (a[i] > l) {
          --k;
          --a[i];
        }
      }
      for (j = i; j <= n; ++j)
        if (a[j] > l) printf("%d ", j);
      for (j = 1; j < i; ++j)
        if (a[j] > l) printf("%d ", j);
    }
  }
  printf("\n");
  return 0;
}
