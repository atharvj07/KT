public class Main {

    static FastReader in;
    static PrintWriter out;
    static Random rand = new Random();
    static final int INF = 1000000010, MOD = 1000000007, LOGN = 20;
    static final int MAXK = 64;
    static int[] sum;

    static DP dp;
    static class DP {
        int n, sqn;
        int[][][] dp1, dp2;

        DP(int n) {
            this.n = n;
            sqn = (int) Math.sqrt(n * 2);
            dp1 = new int[n + 1][sqn + 1][sqn + 1];
            dp2 = new int[n + 1][sqn + 1][sqn + 1];
            for (int i = 0; i <= n; i++) {
                for (int j = 0; j <= sqn; j++) {
                    for (int k = 0; k <= sqn; k++) {
                        dp1[i][j][k] = -1;
                        dp2[i][j][k] = -1;
                    }
                }
            }
        }

        int get(int l, int r, int k, boolean fi) {
            if (fi) {
                return dp1[l][(n - r) - (l - 1)][k];
            } else {
                return dp2[r][(l - 1) - (n - r)][k];
            }
        }

        void set(int l, int r, int k, boolean fi, int val) {
            if (fi) {
                dp1[l][(n - r) - (l - 1)][k] = val;
            } else {
                dp2[r][(l - 1) - (n - r)][k] = val;
            }
        }
    }

    static int go(int l, int r, int k, boolean fi) {
        if (dp.get(l, r, k, fi) != -1) {
            return dp.get(l, r, k, fi);
        }

        if (fi) {
            if (l + k - 1 > r) {
                dp.set(l, r, k, fi, 0);
            } else {
                go(l + k, r, k, !fi);
                int val1 = sum[l + k - 1] - sum[l - 1] + dp.get(l + k, r, k, !fi);
                if (l + k > r) {
                    dp.set(l, r, k, fi, val1);
                } else {
                    go(l + k + 1, r, k + 1, !fi);
                    int val2 = sum[l + k] - sum[l - 1] + dp.get(l + k + 1, r, k + 1, !fi);
                    dp.set(l, r, k, fi, Math.max(val1, val2));
                }
            }
        } else {
            if (l + k - 1 > r) {
                dp.set(l, r, k, fi, 0);
            } else {
                go(l, r - k, k, !fi);
                int val1 = -(sum[r] - sum[r - k]) + dp.get(l, r - k, k, !fi);
                if (l + k > r) {
                    dp.set(l, r, k, fi, val1);
                } else {
                    go(l, r - k - 1, k + 1, !fi);
                    int val2 = -(sum[r] - sum[r - k - 1]) + dp.get(l, r - k - 1, k + 1, !fi);
                    dp.set(l, r, k, fi, Math.min(val1, val2));
                }
            }
        }

        return dp.get(l, r, k, fi);
    }

    static void solve() {
        int n = in.nextInt();

        sum = new int[n + 1];
        sum[0] = 0;
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + in.nextInt();
        }

        dp = new DP(n);

        out.println(go(1, n, 1, true));

    }

    public static void main(String[] args) throws FileNotFoundException {
        in = new FastReader(System.in);
//        in = new FastReader(new FileInputStream("connect.in"));
        out = new PrintWriter(System.out);
//        out = new PrintWriter(new FileOutputStream("connect.out"));


        int q = 1;
//        q = in.nextInt();

        while (q-- > 0) {
            solve();
        }

        out.close();
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        String nextLine() {
            String s = "";
            try {
                s = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return s;
        }
    }
