#include <bits/stdc++.h>
using namespace std;
const int MaxN = 1000 + 5;
int N, Cnt[26];
bool Mark[MaxN];
string S;
inline bool Prime(int N) {
  if (N < 2) return false;
  if (N == 2) return true;
  if (!(N % 2)) return false;
  for (int i = 3; i * i <= N; i += 2)
    if (!(N % i)) return false;
  return true;
}
int main(void) {
  cin >> S;
  N = S.length();
  for (int i = 0; i < N; i++) Cnt[S[i] - 'a']++;
  int CntPos = 0;
  for (int i = 1; i <= N / 2; i++) {
    if (!Prime(i)) continue;
    for (int j = i; j <= N; j += i) {
      if (Mark[j]) continue;
      Mark[j] = true;
      CntPos++;
    }
  }
  char Ch = ' ';
  for (int i = 0; i < 26; i++)
    if (Cnt[i] >= CntPos) Ch = 'a' + i;
  if (Ch == ' ') {
    puts("NO");
    return 0;
  }
  puts("YES");
  queue<char> Letters;
  for (int i = 0; i < N; i++)
    if (S[i] != Ch)
      Letters.push(S[i]);
    else if (CntPos)
      CntPos--;
    else
      Letters.push(S[i]);
  for (int i = 0; i < N; i++)
    if (Mark[i + 1])
      printf("%c", Ch);
    else {
      printf("%c", Letters.front());
      Letters.pop();
    }
  printf("\n");
  return 0;
}
