def copy(n, one, two, three):
	if n % 4 == 0:
		return 0
	x = 4 - n % 4
	if x == 1:
		return min(one, 3 * three, three + two)
	if x == 2:
		return min(one * 2, two, 2 * three)
	if x == 3:
		return min(three, 3 * one, two + one)
(a, b, c, d) = map(int, input().strip().split())
print(copy(a, b, c, d))
