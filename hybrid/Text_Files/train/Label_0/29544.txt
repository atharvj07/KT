#include <bits/stdc++.h>
using namespace std;

int main(){
  int n;
  cin >> n;
  long long ans=1;
  int i;
  for(int j=0;j<n;j++){
    i=j;
    long long  a;
    cin >> a;
    if(ans>1000000000000000000/a){
      ans=-1;
      j=n;
    }
    else ans*=a;
  }
  for(i;i<n;i++){
    long long  a;
    cin >> a;
    if(a==0)
      ans*=0;
  }
  cout << ans << endl;
}
