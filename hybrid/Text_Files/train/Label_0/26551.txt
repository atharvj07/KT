#include <bits/stdc++.h>
using namespace std;
int gcd(int x, int y) {
  if (y == 0) return x;
  return gcd(y, x % y);
}
long long n, k;
long long a[1000005];
long long b[1000005];
long long c[1000005];
long long ans = -(1LL << 59);
void work(int L, int R) {
  int cnt = 0, cnt2 = 0;
  long long temp = 0;
  for (int i = 1; i <= n; ++i) {
    if (i < L || i > R)
      if (a[i] > 0) b[++cnt] = a[i];
    if (i >= L && i <= R) {
      temp += a[i];
      if (a[i] < 0) c[++cnt2] = a[i];
    }
  }
  sort(b + 1, b + 1 + cnt, greater<int>());
  sort(c + 1, c + 1 + cnt2);
  int top = min(cnt, cnt2);
  int tt = k;
  for (int i = 1; i <= top; ++i) {
    if (tt <= 0) break;
    temp += b[i], temp -= c[i];
    tt--;
  }
  ans = max(ans, temp);
}
int main() {
  scanf("%lld%lld", &n, &k);
  for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
  for (int i = 1; i <= n; ++i) {
    for (int j = i; j <= n; ++j) {
      work(i, j);
    }
  }
  printf("%lld\n", ans);
  return 0;
}
