#include <bits/stdc++.h>
using namespace std;
struct fenwickTree {
  vector<long long> bit;
  int n;
  void init(int n) {
    this->n = n;
    bit.assign(n, 0);
  }
  long long sum(int r) {
    long long ret = 0;
    for (; r >= 0; r = (r & (r + 1)) - 1) ret += bit[r];
    return ret;
  }
  void add(int idx, long long d) {
    for (; idx < n; idx = idx | (idx + 1)) bit[idx] += d;
  }
  long long sum(int l, int r) { return sum(r) - sum(l - 1); }
} ft;
int highestBit(int n) {
  if (n == 0) return 0;
  if (n == 1) return 1;
  for (int bit = 30; bit >= 0; --bit) {
    if (n & (1 << bit)) return bit + 1;
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n, q;
  cin >> n >> q;
  ft.init(n + 2);
  int num;
  vector<set<int> > group(31);
  for (int i = 1; i <= n; ++i) {
    cin >> num;
    group[highestBit(num)].insert(i);
    ft.add(i, num);
  }
  int p, x;
  while (q--) {
    cin >> p >> x;
    num = ft.sum(p, p);
    group[highestBit(num)].erase(p);
    group[highestBit(x)].insert(p);
    ft.add(p, x - num);
    int ans = -1;
    for (auto &g : group) {
      int cnt = 0;
      for (const int &pos : g) {
        num = ft.sum(pos, pos);
        if (ft.sum(pos, pos) == ft.sum(pos - 1)) {
          ans = pos;
          break;
        }
        cnt++;
        if (cnt >= 2) break;
      }
      if (ans != -1) break;
    }
    cout << ans << '\n';
  }
  return 0;
}
