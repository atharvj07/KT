t = int(input())

def check(arr, v, dic):
	for i in arr:
		if dic[i + v] == 1:
			return True
	return False
dic = [0] * 2000005
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	arr.sort()
	brr = [1] * n
	cnt = 0
	i = 1
	flag = True
	tmp = {}
	while cnt < n:
		while check(arr, i, dic):
			i += 1
		brr[cnt] = i
		for v in arr:
			dic[i + v] = 1
			tmp[i + v] = 1
		cnt += 1
		i += 1
	if flag:
		print('YES')
		print(' '.join(map(str, brr)))
	else:
		print('NO')
	for k in tmp.keys():
		dic[k] = 0
