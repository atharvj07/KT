import sys
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
(n, x, y) = I()
l = I()
an = 0
c = 0
for i in range(n):
	le = i - x
	r = i + y
	f = 0
	for j in range(le, r + 1):
		if j != i and 0 <= j <= n - 1:
			if l[i] >= l[j]:
				f = 1
				break
	if f == 0:
		an = i + 1
		break
print(an)
