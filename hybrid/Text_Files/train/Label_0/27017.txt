int N,M;
int T[50010];
int S[50010];
int lis[]={9,11,0,2,4,5,7};

char t[3];
inline int in(){
    scanf("%s",t);
    return lis[t[0]-'A']+(t[1]=='#');
}
inline int add(int a,int b){
    a+=b;
    if(a>=12)a-=12;
    return a;
}

void solve(){
    scanf("%lld%lld",&N,&M);
    rep(i,N)T[i]=in();
    rep(i,M)S[i]=in();
    for(int fin=N-2;fin<N;fin++){
        bool ok=true;
        int pos=fin;
        for(int i=M-1;i>=0;i--){
            if(pos<0||pos==N){
                ok=false;
                break;
            }
            if(T[pos]==S[i])pos--;
            else if(add(T[pos],1)==S[i])pos-=2;
            else if(add(T[pos],11)==S[i])pos++;
            else{
                ok=false;
                break;
            }
        }
        if(ok&&pos==-1){
            puts("Yes");
            return;
        }
    }
    puts("No");
}

signed main(){
    int T;scanf("%lld",&T);
    while(T--)solve();
    return 0;
}
