#include "bits/stdc++.h"
using namespace std;
#define Would
#define you
const long long INF = 99999999999999999;
const long long MOD = 1e9 + 7;
const double pi = acos(-1);
const int SIZE = 1 << 17;
int dx[] = { 1,0,-1,0 }, dy[] = { 0,1,0,-1 };

int a, b, x, y;
bool mon[105][105];
char k[105][105];

void dfs(int n, int m) {
	if (k[n][m] == '*') { return; }
	mon[n][m] = 0;
	for (int i = 0; i < 4; ++i) {
		int xx = n + dx[i], yy = m + dy[i];
		if (k[xx][yy] == '@') { continue; }
		if (mon[xx][yy]) {
			dfs(xx, yy);
			return;
		}
	}
}

int main() {
	cin >> a >> b;
	for (int i = 1; i <= a; ++i) {
		for (int j = 1; j <= b; ++j) {
			cin >> k[i][j];
			if (k[i][j] == '@') {
				x = i, y = j;
			}
		}
	}
	for (int i = 1; i <= a; ++i) {
		for (int j = 1; j <= b; ++j) {
			bool bre = false;
			switch (i) {
			case 1:
				mon[i][j] = 1;
				bre = 1;
				break;
			case 2:
				if (j == 1 || j == b) { mon[i][j] = 1; }
				bre = 1;
				break;
			case 3:
				if (j % 4 != 2) { mon[i][j] = 1; }
				bre = 1;
				break;
			}
			if (!bre) {
				if (i != a) {
					if (j % 2) { mon[i][j] = 1; }
				}
				else {
					if (j % 4) { mon[i][j] = 1; }
				}
			}
		}
	}
	for (int i = 0; i < 4; ++i) {
		if (mon[x + dx[i]][y + dy[i]]) {
			dfs(x + dx[i], y + dy[i]);
			break;
		}
	}
	for (int i = 1; i <= a; ++i) {
		for (int j = 1; j <= b; ++j) {
			if (k[i][j] != '.') { cout << k[i][j]; }
			else if (mon[i][j]) { cout << "#"; }
			else { cout << "."; }
		}
		cout << endl;
	}
}
