#include <bits/stdc++.h>
using namespace std;
template <typename T>
T gcd(T a, T b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
template <typename T>
T pow(T a, T b, long long m) {
  T ans = 1;
  while (b > 0) {
    if (b % 2 == 1) ans = (ans * a) % m;
    b /= 2;
    a = (a * a) % m;
  }
  return ans % m;
}
vector<long long> v;
void br(long long m, long long w) {
  while (m > 0) {
    v.push_back(m % w);
    m /= w;
  }
}
void prog(int i, long long w) {
  int crr = 1;
  while (i < v.size() && crr != 0) {
    v[i] += crr;
    crr = (v[i] / w);
    v[i] %= w;
    i++;
  }
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long w, m;
  cin >> w >> m;
  br(m, w);
  v.push_back(0);
  for (int i = 0; i < v.size(); i++) {
    if (v[i] == 0 || v[i] == 1) continue;
    if (w - v[i] == 1) {
      prog(i, w);
    } else {
      cout << "NO";
      return 0;
    }
  }
  cout << "YES";
}
