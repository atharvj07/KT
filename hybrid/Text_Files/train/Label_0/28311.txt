#include <bits/stdc++.h>
using namespace std;
int n, m, f[55][55][2], C[55][55], ans;
long long fpm(long long x, long long y) {
  long long s = 1;
  while (y) {
    if (y & 1) s = (s * x) % 1000000007;
    y >>= 1, x = (x * x) % 1000000007;
  }
  return s;
}
int main() {
  int i, j, t, p, I;
  scanf("%d %d", &n, &m);
  if (n == 1) {
    printf("%d", !m);
    return 0;
  }
  C[0][0] = 1;
  for (i = 1; i <= n; i++) {
    C[i][0] = C[i][i] = 1;
    for (j = 1; j < i; j++)
      C[i][j] = (C[i - 1][j] + C[i - 1][j - 1]) % 1000000007;
  }
  f[1][0][0] = 1;
  for (i = 2; i <= n; i++)
    for (j = 1; j <= i / 2; j++) {
      f[i][j][0] = 1LL * i * f[i - 1][j][1] % 1000000007;
      f[i][j][1] = 1LL * i * f[i - 1][j - 1][0] % 1000000007;
      for (t = 1; t + t <= i - 1; t++)
        for (p = 0; p <= j; p++) {
          if (t == i - t - 1)
            I = (1000000007 + 1) / 2;
          else
            I = 1;
          f[i][j][0] = (f[i][j][0] + 1LL * I * i % 1000000007 * C[i - 1][t] %
                                         1000000007 * f[i - t - 1][j - p][1] %
                                         1000000007 * f[t][p][1]) %
                       1000000007;
          if (j > p) {
            f[i][j][1] =
                (f[i][j][1] + 1LL * I * i % 1000000007 * C[i - 1][t] %
                                  1000000007 * f[i - t - 1][j - p - 1][0] %
                                  1000000007 * f[t][p][0]) %
                1000000007;
            f[i][j][1] =
                (f[i][j][1] + 1LL * I * i % 1000000007 * C[i - 1][t] %
                                  1000000007 * f[i - t - 1][j - p - 1][1] %
                                  1000000007 * f[t][p][0]) %
                1000000007;
            f[i][j][1] =
                (f[i][j][1] + 1LL * I * i % 1000000007 * C[i - 1][t] %
                                  1000000007 * f[i - t - 1][j - p - 1][0] %
                                  1000000007 * f[t][p][1]) %
                1000000007;
          }
        }
    }
  (ans += f[n][m][0]) %= 1000000007;
  (ans += f[n][m][1]) %= 1000000007;
  ans = (1LL * ans * fpm(n, 1000000007 - 2)) % 1000000007;
  cout << ans;
  return 0;
}
