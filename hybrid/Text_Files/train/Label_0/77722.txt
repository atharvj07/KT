from math import gcd
from collections import Counter
import sys
reader = (map(int, line.split()) for line in sys.stdin)
input = reader.__next__
modulo = 998244353

def lcm(lst):
	res = lst[0]
	for el in lst[1:]:
		res = res * el // gcd(res, el)
	return res

def invMod(n):
	return pow(n, modulo - 2, modulo)

def fracMod(a, b):
	return a * invMod(b) % modulo
(n,) = input()
length_lst = []
presentWanted = [0] * (10 ** 6 + 2)
wishList = []
for i in range(n):
	(k, *a) = input()
	length_lst.append(k)
	wishList.append(a)
	for present in a:
		presentWanted[present] += 1
lcm_len = lcm(length_lst)
denom = lcm_len % modulo
denom *= pow(n, 2, modulo)
denom %= modulo
nom = 0
for wish in wishList:
	wish_len = len(wish)
	f = lcm_len // wish_len
	f %= modulo
	for present in wish:
		nom += f * presentWanted[present]
		nom %= modulo
ans = fracMod(nom, denom)
print(ans)
