const int INF = 1e9;

int main(){
	int N;string S;
	cin >> S ; N = S.length();
  	int ans = INF;
  	rep(i,N-1){
  	  if(S[i]!=S[i+1]) ans = min(ans,max(i+1,N-i-1));

}
