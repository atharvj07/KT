import sys
input = sys.stdin.readlines()
lemons = int(input[0])
apples = int(input[1])
pears = int(input[2])
for lemon_num in range(lemons, 0, -1):
	case_apples = lemon_num * 2
	case_pears = lemon_num * 4
	if case_apples <= apples and case_pears <= pears:
		print(lemon_num + case_pears + case_apples)
		break
else:
	print('0')
