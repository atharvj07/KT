int Lamtron(int x, int y)
{
    if(x%y==0) return x/y;
    else return x/y+1;
}
void Solve()
{
        if(kc==0 && r==b) cout << "YES";
        else if(kc==0 && r!=b) cout << "NO";
        else if((abs(r-b) + min(r,b) - 1) / min(r,b) <= kc) cout << "YES";
        else cout << "NO";
        cout << '\n';
}
