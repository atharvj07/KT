#include <bits/stdc++.h>
int glided_tiles(int r, int c, int k) {
  int res = 0;
  int c1 = 0;
  while (c1 < k) {
    res = 2 * (r + c) - 4 + res;
    r = r - 4;
    c = c - 4;
    c1++;
  }
  return res;
}
int main() {
  int r, c, k, res;
  {
    scanf("%d", &r);
    scanf("%d", &c);
    scanf("%d", &k);
    res = glided_tiles(r, c, k);
    printf("%d", res);
    return 0;
  }
}
