import sys
input = sys.stdin.readline
(n, p) = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
ans = []
import bisect
for i in range(min(a), max(a) + 1):
	cnt = 1
	for j in range(i, i + n):
		k = i + n - j
		m = bisect.bisect_right(a, j)
		cnt *= k - n + m
		cnt %= p
	if cnt % p != 0:
		ans.append(i)
print(len(ans))
print(*ans)
