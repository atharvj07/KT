#include <bits/stdc++.h>
using namespace std;
int p[13], b[510], n, m, Q[510], F[510], W[510];
long long T;
inline void add(int &x, int y) {
  x += y;
  if (x >= 1000000007) x -= 1000000007;
  if (x < 0) x += 1000000007;
}
inline void mul(int A[], int B[]) {
  int C[510] = {};
  for (int i = 0; i <= 250; i++)
    for (int j = 0; j <= 250; j++)
      add(C[i + j], 1ll * A[i] * B[j] % 1000000007);
  for (int i = 250 + 250; i >= 250; i--) {
    for (int j = 1; j <= 250; j++)
      add(C[i - j], 1ll * C[i] * Q[j] % 1000000007);
    C[i] = 0;
  }
  for (int i = 0; i <= 250; i++) A[i] = C[i];
}
inline void Pow(int P[], long long T) {
  int V[510] = {};
  V[0] = 1;
  while (T > 0) {
    if (T & 1) mul(V, P);
    mul(P, P);
    T = T >> 1;
  }
  for (int i = 0; i <= 250; i++) P[i] = V[i];
}
int main() {
  cin >> n >> m >> T;
  for (int i = 1; i <= n; i++) scanf("%d", &p[i]);
  for (int i = 1; i <= m; i++) scanf("%d", &b[i]), Q[b[i]]++;
  Q[0] = -1;
  for (int i = 1; i <= n; i++) {
    int P[510] = {};
    P[1] = 1;
    Pow(P, p[i]);
    for (int j = 0; j <= 250; j++) add(F[j], P[j]);
  }
  Pow(F, T);
  for (int i = 250; i >= 1; i--) {
    for (int j = 1; j <= i; j++) add(F[i - j], 1ll * F[i] * Q[j] % 1000000007);
    F[i] = 0;
  }
  cout << F[0] << endl;
  return 0;
}
