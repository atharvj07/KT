#include <bits/stdc++.h>
#define all(vec) vec.begin(), vec.end()
using namespace std;
using ll = long long;
using P = pair<ll, ll>;
constexpr ll INF = (1LL << 30) - 1LL;
constexpr ll LINF = (1LL << 60) - 1LL;
constexpr ll MOD = 1e9 + 7;
template <typename T> void chmin(T &a, T b) { a = min(a, b); }
template <typename T> void chmax(T &a, T b) { a = max(a, b); }
int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);
    queue<int> qs[10];
    int n, q;
    cin >> n >> q;
    vector<int> v;
    while (q--) {
        int t, a;
        cin >> t >> a;
        if (t == 0) {
            --a;
            v.push_back(qs[a].front());
            qs[a].pop();
        } else {
            int mi = INF, id = -1;
            for (int i = 0; i < n; i++) {
                if (qs[i].size() < mi) {
                    mi = qs[i].size();
                    id = i;
                }
            }
            qs[id].push(a);
        }
    }
    for (auto e : v) {
        cout << e << endl;
    }
}
