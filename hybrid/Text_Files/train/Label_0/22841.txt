#include <bits/stdc++.h>
using namespace std;
const int64_t inf = LLONG_MAX / 2;
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int64_t t;
  cin >> t;
  while (t--) {
    int64_t n, p, k;
    cin >> n >> p >> k;
    vector<int64_t> v(n);
    vector<vector<int64_t>> dp(2, vector<int64_t>(n, inf));
    for (int64_t i = 0; i < n; ++i) cin >> v[i];
    sort(v.begin(), v.end());
    dp[0][0] = v[0];
    for (int64_t i = 1; i < n; ++i) {
      int64_t a = v[i];
      if (i - k >= -1) {
        if (i - k >= 0) a += dp[0][i - k];
        dp[1][i] = min(dp[1][i], a);
        int64_t b = v[i];
        if (i - k >= 0) b += dp[1][i - k];
        dp[1][i] = min(dp[1][i], b);
      }
      dp[1][i] = min(dp[1][i], v[i] + dp[1][i - 1]);
      dp[1][i] = min(dp[1][i], v[i] + dp[0][i - 1]);
      dp[0][i] = min(dp[0][i], dp[0][i - 1] + v[i]);
    }
    int64_t ans = 0;
    for (int64_t i = 0; i < n; ++i) {
      if (dp[1][i] <= p) {
        ans = i + 1;
      }
      if (dp[0][i] <= p) {
        ans = i + 1;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
