import java.util.*;
public class Main {
     static long gcd(long a, long b) 
    { 
        if (a == 0) 
            return b; 
        return gcd(b % a, a); 
    } 
   
   
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
        int n=sc.nextInt();
        long a[]=new long[n];
       // long ans=0;
        for(int i=0;i<n;i++) {
        a[i]=sc.nextLong();
        
        }
        long ans=a[0];
        for(int i=1;i<n;i++)
		{
			ans=gcd(ans,a[i]);
		}
        long res=0;
        for (int i = 1; i<=Math.sqrt(ans); ++i) {
		if (ans % i == 0) {
			++res;
			if (i != ans / i) {
				++res;
			}
		}
	}
	System.out.println(res);
    
  }
}
