import sys
n = int(sys.stdin.readline())
arr1 = [int(i) - 1 for i in sys.stdin.readline().split()]
arr2 = [int(i) - 1 for i in sys.stdin.readline().split()]
new_arr = [0 for i in range(n)]
new_arr2 = [0 for i in range(n)]
result = 0
max_element = -1
for i in range(n):
	new_arr[arr2[i]] = i
for i in range(n):
	new_arr2[i] = new_arr[arr1[i]]
for i in range(n):
	if new_arr2[i] > max_element:
		max_element = new_arr2[i]
	else:
		result += 1
print(result)
