#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

bool is_good_set(int a, int b, int c) {
    auto is_same_card = [](int a, int b, int c) {
        return a == b and b == c;
    };
    auto is_sequence = [](int a, int b, int c) {
        return a + 1 == b and b + 1 == c;
    };
    return is_same_card(a, b, c) or is_sequence(a, b, c);
}

bool is_all_good_set(vector<int> &cc) {
    return is_good_set(cc[0], cc[1], cc[2])
        && is_good_set(cc[3], cc[4], cc[5])
        && is_good_set(cc[6], cc[7], cc[8]);
}

int win(vector<int> &cards) {
    sort(cards.begin(), cards.end());
