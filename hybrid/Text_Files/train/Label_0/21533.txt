#include <bits/stdc++.h>
using namespace std;
string str;
int k;
string f[40];
int main() {
  cin >> str;
  cin >> k;
  for (int i = 0; i < k; i++) cin >> f[i];
  int ans = 0;
  for (int i = 0; i < k; i++) {
    char c1 = f[i][0], c2 = f[i][1];
    int sum1 = 0, sum2 = 0;
    for (int j = 0; j <= str.length(); j++) {
      if (j < str.length() && str[j] == c1 || str[j] == c2) {
        if (str[j] == c1) sum1++;
        if (str[j] == c2) sum2++;
      } else {
        ans += min(sum1, sum2);
        sum1 = sum2 = 0;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
