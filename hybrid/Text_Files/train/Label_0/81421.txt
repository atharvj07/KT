n = int(input())
a = list(map(int, input().split()))
if n == 1:
	print('1 1')
	print(-a[0])
	print('1 1')
	print(0)
	print('1 1')
	print(0)
else:
	one = [0] * n
	for i in range(n - 1):
		m = a[i] % (n - 1)
		if m != 0:
			one[i] = n * (n - 1 - m)
		a[i] += one[i]
	print('1', n)
	for i in one:
		print(i, end=' ')
	print()
	print('1', n - 1)
	for i in range(n - 1):
		print(-a[i], end=' ')
	print()
	print(n, n)
	print(-a[n - 1])
