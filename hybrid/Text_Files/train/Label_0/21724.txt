#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
const int inf = (1 << 30) - 1;
const ll infll = (1LL << 61) - 1;

int N, M;

int main()
{
    while (cin >> N >> M)
    {
        if (N == 0 && M == 0)
            return 0;
        map<int, int> mp;
        int ans = 0;
        for (int i = 0; i < N; i++)
        {
            int d, v;
            cin >> d >> v;
            mp[d] = max(mp[d], v);
        }
        for (auto &p : mp)
        {
            ans += p.second;
        }
        cout << ans << endl;
    }
}
