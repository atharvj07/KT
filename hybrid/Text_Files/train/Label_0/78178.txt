m = []
for i in range(int(input())):
	n = int(input())
	e = input()
	m = [['=' for k in range(n)] for j in range(n)]
	for i in range(n):
		m[i][i] = 'X'
	two = len([j for j in e if j != '1'])
	twoi = [j for j in range(n) if e[j] != '1']
	if two == 1 or two == 2:
		print('NO')
		continue
	print('YES')
	for i in range(two):
		t1 = twoi[i]
		if i + 1 == len(twoi):
			t2 = twoi[0]
		else:
			t2 = twoi[i + 1]
		m[t1][t2] = '+'
		m[t2][t1] = '-'
	print('\n'.join([''.join(j) for j in m]))
