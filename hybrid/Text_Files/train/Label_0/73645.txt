(n, k) = input().strip().split(' ')
(n, k) = [int(n), int(k)]
a = list(map(int, input().strip().split(' ')))
list.sort(a)
from itertools import groupby
g = [len(list(group)) for (key, group) in groupby(a)]
mx = -1
list.sort(g)
for i in range(1, 101):
	day = 0
	for j in range(len(g)):
		day = day + int(g[j] / i)
	if n <= day:
		mx = max(mx, i)
if mx > 0:
	print(mx)
else:
	print('0')
