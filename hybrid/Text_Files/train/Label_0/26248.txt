#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int h = -2;
  for (int i = 0; i < s.length();) {
    int j = i;
    string tmp = "<";
    while (s[j] != '>') {
      j++;
      tmp = tmp + s[j];
    }
    if (tmp.length() == 3) h += 2;
    for (int k = 1; k <= h; ++k) cout << " ";
    cout << tmp << "\n";
    if (tmp.length() == 4) h -= 2;
    i = j + 1;
  }
}
