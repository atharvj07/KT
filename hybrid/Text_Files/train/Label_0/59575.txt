public class HerculePoirot {
    private BufferedReader in;
    private PrintWriter out;
    private StringTokenizer st;

	class UnionFind{
		int[] parent;
		int[] rank;

		public UnionFind(int n){
			parent = new int[n];
			rank = new int[n];
			for(int i = 0; i < n; i++){
				parent[i] = i;
				rank[i] = 1;
			}
		}

		int getParent(int x){
			if(parent[x] != x) return parent[x] = getParent(parent[x]);
			return parent[x];
		}

		void connect(int i, int j){
			int pi = getParent(i), pj = getParent(j);
			if(pi == pj) return;
			if(rank[pi] > rank[pj]){
				parent[pj] = pi; rank[pi] += rank[pj];
			}else{
			   parent[pi] = pj; rank[pj] += rank[pi];
			}
		}
	}

	int[] s, t;
	int[][] keys, people;
	boolean[][][] connected;
	HashMap< String, Integer > map;
	int N, M, K;

    void solve() throws IOException {
		N = nextInt(); M = nextInt(); K = nextInt();
		this.keys = new int[2][M];
		this.people = new int[2][K];
		this.connected = new boolean[2][N][N];
		map = new HashMap<String, Integer>();
		s = new int[M];
		t = new int[M];

		for(int i = 0; i < M; i++){
			s[i] = nextInt() - 1; t[i] = nextInt() - 1;
		}
		String people;
		int c, x, room;
		for(int i = 0; i < K; i++){
			people = next();
			map.put(people, i);
			room = nextInt() - 1;
			this.people[0][i] = room;
			c = nextInt();
			for(int j = 0; j < c; j++){
				x = nextInt() - 1;
				this.keys[0][x] = room;
			}
		}

		int ix;
		for(int i = 0; i < K; i++){
			people = next();
			ix = map.get(people);
			room = nextInt() - 1;
			this.people[1][ix] = room;
			c = nextInt();
			for(int j = 0; j < c; j++){
				x = nextInt() - 1;
				this.keys[1][x] = room;
			}
		}

		solve(0);
		solve(1);
		if(can()) out.println("YES");
		else out.println("NO");
    }

	void solve(int ix){
		int room = -1;
		UnionFind uf = new UnionFind(N);
		 for(int i = 0; i < N + 10; i++){
			 for(int j = 0; j < M; j++){
				 room = uf.getParent(keys[ix][j]);
				 if(uf.getParent(s[j]) == room || uf.getParent(t[j]) ==  room) uf.connect(s[j], t[j]);
			 }
		 }

		for(int i = 0; i < N; i++)
			for(int j = 0; j < N; j++)
				if(uf.getParent(i) == uf.getParent(j))
					connected[ix][i][j] = true;
	}

	boolean can(){

		for(int i = 0; i < N; i++)
			for(int j = 0; j < N; j++)
				if(connected[0][i][j] != connected[1][i][j]) return false;

		for(int i = 0; i < M; i++) if(!connected[0][keys[0][i]][keys[1][i]]) return false;
		for(int i = 0; i < K; i++)if(!connected[0][people[0][i]][people[1][i]]) return false;
		return true;
	}

    HerculePoirot() throws IOException {
        in = new BufferedReader(new InputStreamReader(System.in));
        //in  = new BufferedReader(new InputStreamReader(new FileInputStream(new File("C://Users/piyushd/Desktop/codeforces/sample.txt"))));
        out = new PrintWriter(System.out);

        eat("");

        solve();

        in.close();
        out.close();
    }

    private void eat(String str) {
        st = new StringTokenizer(str);
    }

    String next() throws IOException {
        while (!st.hasMoreTokens()) {
            String line = in.readLine();
            if (line == null) {
                return null;
            }
            eat(line);
        }
        return st.nextToken();
    }

    int nextInt() throws IOException {
        return Integer.parseInt(next());
    }

    long nextLong() throws IOException {
        return Long.parseLong(next());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(next());
    }

    public static void main(String[] args) throws IOException {
        new HerculePoirot();
    }
}
