import sys
x = int(input())

def solve(cnt, odds):
	print(cnt)
	print(' '.join(list(map(str, odds))))
	sys.exit(0)
odds = []
cnt = 0
xb = list(bin(x)[2:])
n = len(xb)
if x == 0 or x ^ int(2 ** n - 1) == 0:
	solve(cnt, odds)
while True:
	xb = list(bin(x)[2:])
	n = len(xb)
	for i in range(len(xb)):
		if xb[i] == '0':
			nn = 2 ** (n - i) - 1
			odds.append(n - i)
			x = x ^ nn
			cnt += 1
			if x ^ 2 ** n - 1 == 0:
				solve(cnt, odds)
			x += 1
			cnt += 1
			if x ^ 2 ** n - 1 == 0:
				solve(cnt, odds)
			break
