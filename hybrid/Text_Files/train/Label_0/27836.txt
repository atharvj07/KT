#include "iostream"
#include "string"
#include "algorithm"
using namespace std;

int main() {
	int n, m;
	cin >> n >> m;
	int student[150];
	for (int i = 0; i < n; i++) {
		cin >> student[i];
	}
	for (int i = 2; i <= m; i++) {
		for (int j = 0; j < n-1; j++) {
			if (student[j] % i > student[j + 1] % i) {
				swap(student[j], student[j + 1]);
			}
		}
	}
	for (int i = 0; i < n; i++) {
		cout << student[i] << endl;
	}
}
