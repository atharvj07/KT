class TaskD {
    int getMinL(String a, String b) {
        if(a.contains(b)) {
            return a.length();
        }
        for (int i = 1; i < b.length(); i++) {

            if(a.endsWith(b.substring(0, b.length()-i))) {
                return a.length()+i;
            }

        }
        return a.length() + b.length();
    }
    public void solve(int testNumber, InputReader in, PrintWriter out) {
        String s = in.ns();
        int q = in.ni();
        int n = s.length();
        HashMap<String, ArrayList<Integer>>[] dic = new HashMap[4];
        for (int i = 0; i < 4; i++) {
            dic[i] = new HashMap<String, ArrayList<Integer>>((int)Math.pow(26,i+1));
        }
        for (int l = 0; l < 4; l++) {
            for (int i = 0; i < n - l; i++) {
                String key = s.substring(i, i + l+1);
                ArrayList<Integer> set = dic[l].get(key);
                if (set == null) {
                    set = new ArrayList<Integer>();
                    dic[l].put(key, set);
                }
                set.add(i);
            }
        }
        HashMap<String, Integer> cache = new HashMap<String, Integer>();
        StringBuilder sb = new StringBuilder();
        long l1 = System.currentTimeMillis();
        for (int i = 0; i < q; i++) {
            String a = in.ns();
            String b = in.ns();
            Integer ans = cache.get(a + "#" + b);
            if(ans != null) {
                sb.append(ans).append("\n");
                continue;
            }
            int al = a.length();
            int bl = b.length();
            int maxL = Math.min(getMinL(a, b), getMinL(b, a));
            ArrayList<Integer> as = dic[al -1].get(a);
            ArrayList<Integer> bs = dic[bl -1].get(b);
            int min = Integer.MAX_VALUE;
            if(as != null && bs != null) {
                int bii = 0;

                for (Integer ai : as) {
                    
                    while((bs.get(bii)) < ai && bii < bs.size()-1) {
                        bii++;
                    }
                    if(bs.get(bii) >= ai) {
                        min = Math.min(min,bs.get(bii) + bl - ai);
                        if(bii > 0) {
                            min = Math.min(min,ai + al - bs.get(bii-1));
                        }

                    } else {
                        min = Math.min(min,ai + al - bs.get(bii));
                    }
                    if (min <= maxL) {
                        min = maxL;
                        break;
                    }
                }


            }
            if(min != Integer.MAX_VALUE) {

                sb.append(min).append("\n");
            } else {
                min = -1;
                sb.append("-1\n");
            }
            cache.put(a+"#"+b, min);
            cache.put(b+"#"+a, min);
        }
        long l2 = System.currentTimeMillis();
        out.println(sb);
        //System.out.println((l2-l1)+"ms");
    }
}

class InputReader  extends BufferedReader {
    public InputReader(InputStream st) {
        super(new InputStreamReader(st));
    }

    public int nextInt() {
        try {
            int c = read();

            while (isWhiteSpace(c)) {
                c = read();
            }
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int ret = 0;
            while (c >= 0 && !isWhiteSpace(c)) {
                if (c < '0' || c > '9') {
                    throw new RuntimeException();
                }
                ret = ret * 10 + c - '0';
                c = read();
            }
            return ret * sgn;
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    public String ns() {
        try {
            int c = read();

            while (isWhiteSpace(c)) {
                c = read();
            }
            if(isWhiteSpace(c)) {
                return null;
            }

            StringBuilder sb = new StringBuilder("");
            while (c >= 0 && !isWhiteSpace(c)) {
                sb.append((char)c);
                c = read();
            }
            return sb.toString();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    public int ni() {
        return nextInt();
    }

    public String readLine() {
        try {
            return super.readLine();
        } catch (IOException e) {
            return null;
        }
    }


    static boolean isWhiteSpace(int c) {
        return c >= 0 && c <= 32;
    }



}
