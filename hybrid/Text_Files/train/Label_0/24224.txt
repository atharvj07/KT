#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); i++)

int g[15][15];

int main() {
  int n;
  cin >> n;
  rep(i, n) rep(j, n) g[i][j] = -1;
  rep(i, n) {
    int m;
    cin >> m;
    rep(j, m) {
      int a, x;
      cin >> a >> x;
      --a;
      g[i][a] = x;
    }
  }

  int ans = 0;
  rep(i, 1 << n) {
    bool ok = true;
    rep(j, n) {
      if (i >> j & 1) {
        rep(k, n) {
          if (g[j][k] == -1) continue;
          if (g[j][k] != (i >> k & 1)) ok = false;
        }
      }
    }
    if (ok) ans = max(ans, (int)bitset<32>(i).count());
  }
  cout << ans << endl;
  return 0;
}
