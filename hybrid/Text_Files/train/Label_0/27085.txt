#include <bits/stdc++.h>
using namespace std;
int main() {
  std::ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  unsigned long long n, prev = 2, ans = 1, curr = 3, p;
  cin >> n;
  if (n == 2)
    cout << 1;
  else if (n >= 3) {
    while (curr <= n) {
      ans++;
      p = curr;
      curr += prev;
      prev = p;
    }
    cout << ans;
  }
  return 0;
}
