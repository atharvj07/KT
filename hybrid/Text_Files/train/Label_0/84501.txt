t = int(input())

def getDigits(quadrant):
	return len(str(quadrant))
for i in range(t):
	quadrant = 1
	digits = 1
	remainingDigits = int(input())
	while remainingDigits > digits:
		remainingDigits -= digits
		quadrant += 1
		digits += getDigits(quadrant)
	i = 1
	digit = 0
	while remainingDigits > 0:
		if remainingDigits <= getDigits(i):
			digit = int(str(i)[remainingDigits - 1])
			break
		else:
			remainingDigits -= getDigits(i)
		i += 1
	print(digit)
