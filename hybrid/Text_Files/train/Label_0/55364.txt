import java.util.*;
public class Main {
    public static void main(String args[]) {
     Scanner scan = new Scanner(System.in);
     int n = scan.nextInt();
     int m = scan.nextInt();
     boolean adj[][]=new boolean[n][n];
     for(int i =0;i<m;i++)
     {
         int a = scan.nextInt();
         int b = scan.nextInt();
         a--;
         b--;
         adj[a][b]=true;
         adj[b][a]=true;
     }
     int visited[]=new int[n];
     for(int i=0;i<n;i++)
     visited[i]=-1;
     int grp=0;
     while(true)
     {
      
      boolean run =false;
     
      for(int i =0;i<n;i++)
      {
          int cnt=0;
          int index=-1;
        for(int j =0;j<n;j++)
        {
            if(adj[i][j]==true)
            {
                cnt++;
                index=j;
            }
        }
        if(cnt==1)
        {
            run =true;
            visited[i]=index;
        }
      }
      if(run==false)
      break;
      else
      {
       grp++;
       for(int i =0;i<n;i++)
       {
           if(visited[i]!=-1)
           {
               int x = visited[i];
               adj[i][x]=false;
               adj[x][i]=false;
               visited[i]=-1;
           }
        }
      }
     }
     System.out.println(grp);
    }
}
