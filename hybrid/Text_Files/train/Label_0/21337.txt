#include <bits/stdc++.h>
using namespace std;
int n, q, mnx = INT_MAX, mxx, a[100005];
set<int> s;
multiset<int> ms;
void add(int x) {
  s.insert(x);
  set<int>::iterator it = s.find(x);
  if (next(it) != s.end()) ms.insert(*next(it) - x);
  if (it != s.begin()) {
    ms.insert(x - *prev(it));
    if (next(it) != s.end()) ms.erase(ms.find(*next(it) - *prev(it)));
  }
}
void erase(int x) {
  set<int>::iterator it = s.find(x);
  assert(it != s.end());
  if (next(it) != s.end()) ms.erase(ms.find(*next(it) - x));
  if (it != s.begin()) {
    ms.erase(ms.find(x - *prev(it)));
    if (next(it) != s.end()) ms.insert(*next(it) - *prev(it));
  }
  s.erase(x);
}
int query() {
  if (s.size() <= 2) return 0;
  return *(--s.end()) - *s.begin() - *(--ms.end());
};
int main() {
  cin >> n >> q;
  for (int i = 1; i <= n; ++i) {
    int x;
    cin >> x;
    add(x);
  }
  cout << query() << "\n";
  while (q--) {
    int t, x;
    cin >> t >> x;
    if (t)
      add(x);
    else
      erase(x);
    cout << query() << "\n";
  }
  return 0;
}
