#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a.begin(), a.end());
  vector<int> dist;
  map<int, bool> check;
  for (int i = 1; i < n; ++i) {
    if (check[a[i] - a[i - 1]] == 0) {
      dist.push_back(a[i] - a[i - 1]);
      check[a[i] - a[i - 1]] = 1;
    }
  }
  int dis = dist[0];
  for (int i = 0; i < dist.size(); ++i) dis = gcd(dist[i], dis);
  int res = 0;
  for (int i = 1; i < n; ++i) {
    res += ((a[i] - a[i - 1]) / dis - 1);
  }
  cout << res;
  return 0;
}
