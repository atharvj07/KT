#include <bits/stdc++.h>
using namespace std;
struct ops {
  string s;
  long long w, t;
};
string todo[] = {"row", "col"};
long long a[111][111], b[111][111];
void solve() {
  long long n, m;
  cin >> n >> m;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }
  if (n > m) {
    for (long long i = 0; i < n; i++) {
      for (long long j = 0; j < i; j++) {
        swap(a[i][j], a[j][i]);
      }
    }
    swap(n, m);
    swap(todo[0], todo[1]);
  }
  long long sz = 0;
  vector<ops> ans;
  for (long long i = 0; i < n; i++) {
    long long mn = INT_MAX;
    for (long long j = 0; j < m; j++) {
      mn = min(a[i][j], mn);
    }
    sz += mn;
    if (mn) ans.push_back({todo[0], i + 1, mn});
    for (long long j = 0; j < m; j++) {
      b[i][j] = mn;
    }
  }
  for (long long j = 0; j < m; j++) {
    long long xtra = a[0][j] - b[0][j];
    sz += xtra;
    if (xtra) {
      ans.push_back({todo[1], j + 1, xtra});
    }
    for (long long i = 0; i < n; i++) {
      b[i][j] += xtra;
    }
  }
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < m; j++) {
      if (a[i][j] != b[i][j]) {
        cout << "-1";
        return;
      }
    }
  }
  cout << sz << '\n';
  for (long long i = 0; i < ans.size(); i++) {
    for (long long j = 0; j < ans[i].t; j++) {
      cout << ans[i].s << ' ' << ans[i].w << '\n';
    }
  }
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
