g = int(input())
lst = []
for i in range(g):
	h = list(map(int, input().split()))
	lst.append(h)
if g == 3:
	if lst[0][0] != lst[1][0] and lst[0][1] != lst[1][1]:
		print(abs(lst[0][0] - lst[1][0]) * abs(lst[0][1] - lst[1][1]))
	elif lst[1][0] != lst[2][0] and lst[1][1] != lst[2][1]:
		print(abs(lst[1][0] - lst[2][0]) * abs(lst[2][1] - lst[1][1]))
	elif lst[0][0] != lst[2][0] and lst[0][1] != lst[2][1]:
		print(abs(lst[0][0] - lst[2][0]) * abs(lst[0][1] - lst[2][1]))
elif g == 4:
	if lst[0][0] != lst[1][0] and lst[0][1] != lst[1][1]:
		print(abs(lst[0][0] - lst[1][0]) * abs(lst[0][1] - lst[1][1]))
	elif lst[1][0] != lst[2][0] and lst[1][1] != lst[2][1]:
		print(abs(lst[1][0] - lst[2][0]) * abs(lst[2][1] - lst[1][1]))
	elif lst[0][0] != lst[2][0] and lst[0][1] != lst[2][1]:
		print(abs(lst[0][0] - lst[2][0]) * abs(lst[0][1] - lst[2][1]))
	else:
		for i in range(4):
			if lst[0][0] not in lst[i] and lst[0][1] not in lst[i]:
				print(abs(lst[0][0] - lst[i][0]) * abs(lst[0][1] - lst[i][1]))
				break
elif g == 2:
	if lst[0][0] != lst[1][0] and lst[0][1] != lst[1][1]:
		print(abs(lst[0][0] - lst[1][0]) * abs(lst[0][1] - lst[1][1]))
	else:
		print(-1)
elif g == 1:
	print(-1)
