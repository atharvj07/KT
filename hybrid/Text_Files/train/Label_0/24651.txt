#include <stdio.h>

void solve(int n, int m, int k) {
	if (n * m - 1 == k) puts("YES");
	else puts("NO");
}

int main() {
	if (fopen("input.txt", "r"))
		freopen("input.txt", "r", stdin);
	
	int t;
	scanf("%d", &t);
	
	while (t--) {
		int n, m, k;
		scanf("%d%d%d", &n, &m, &k);
		solve(n, m, k);
	}
	
	return 0;
}
