#include <bits/stdc++.h>
using namespace std;
int n, prime[1001], nrp, k, r;
bool eprim(int nr) {
  if (nr == 2) return true;
  for (int d = 2; d * d <= nr; d++)
    if (nr % d == 0) return false;
  return true;
}
bool rezolva() {
  for (int i = 1; i <= nrp; i++)
    for (int j = 2; j <= nrp; j++)
      if (prime[j - 1] + prime[j] + 1 == prime[i]) {
        r++;
        if (r == k) return true;
      }
  return false;
}
int main() {
  cin >> n >> k;
  if (k == 0)
    cout << "YES";
  else {
    for (int i = 2; i <= n; i++)
      if (eprim(i)) {
        nrp++;
        prime[nrp] = i;
      }
    if (rezolva())
      cout << "YES";
    else
      cout << "NO";
  }
  return 0;
}
