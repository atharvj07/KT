T = int(input())
for t in range(T):
	n = int(input())
	A = [int(i) for i in input().split()]
	primary = A[0]
	pidx = 0
	secondary = -1
	sidx = -1
	ct = 0
	for i in A:
		if i != primary:
			secondary = i
			sidx = ct
			break
		ct += 1
	pos = 1
	ans = []
	for i in range(1, n):
		if A[i] != primary:
			ans.append([1, i + 1])
		elif secondary != -1 and A[i] != secondary:
			ans.append([sidx + 1, i + 1])
		else:
			pos = 0
	if pos == 0:
		print('NO')
	else:
		print('YES')
		for i in ans:
			print(*i)
