#include <bits/stdc++.h>
using namespace std;
struct point {
  int num, a, b, x, y, ans;
  void output() { printf("%d %d %d %d %d %d\n", num, x, y, a, b, ans); }
} p[201000];
int q[201000];
bool cmp(int a, int b) {
  if (p[a].a == p[b].a) return p[a].b < p[b].b;
  return p[a].a < p[b].a;
}
int i, j, m;
void work() {
  sort(&q[1], &q[1 + m], cmp);
  for (i = 1; i <= m; i++) {
    if (i > 1 && p[q[i - 1]].a == p[q[i]].a) p[q[i]].ans++;
    if (i < m && p[q[i + 1]].a == p[q[i]].a) p[q[i]].ans++;
  }
}
int ans[10];
int main() {
  int n;
  scanf("%d %d", &n, &m);
  for (i = 1; i <= m; i++) {
    q[i] = i;
    scanf("%d %d", &p[i].x, &p[i].y);
    p[i].num = i;
  }
  for (i = 1; i <= m; i++) {
    p[i].a = p[i].x;
    p[i].b = p[i].y;
  }
  work();
  for (i = 1; i <= m; i++) {
    p[i].a = p[i].y;
    p[i].b = p[i].x;
  }
  work();
  for (i = 1; i <= m; i++) {
    p[i].a = p[i].x + p[i].y;
    p[i].b = p[i].x - p[i].y;
  }
  work();
  for (i = 1; i <= m; i++) {
    p[i].a = p[i].x - p[i].y;
    p[i].b = p[i].y + p[i].y;
  }
  work();
  for (i = 1; i <= m; i++) {
    ans[p[i].ans]++;
  }
  for (i = 0; i < 9; i++) printf("%d ", ans[i]);
  return 0;
}
