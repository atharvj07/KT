#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, t = 0, p;
  cin >> n >> k;
  k = 240 - k;
  for (int i = 1; i <= n; i++) {
    if (t + (5 * i) > k) {
      cout << i - 1;
      return 0;
    }
    if (t + (5 * i) == k) {
      cout << i;
      return 0;
    }
    t = t + (5 * i);
    p = i;
  }
  cout << p;
  return 0;
}
