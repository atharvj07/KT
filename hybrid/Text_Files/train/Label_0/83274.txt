def SieveOfEratosthenes(n):
	prime = [True for i in range(n + 1)]
	p = 2
	while p * p <= n:
		if prime[p] == True:
			for i in range(p * 2, n + 1, p):
				prime[i] = False
		p += 1
	ls = []
	for p in range(2, n):
		if prime[p]:
			ls.append(p)
	return ls
ls = SieveOfEratosthenes(100000)
(p, y) = [int(i) for i in input().split(' ')]
while y != p:
	flag = 0
	for i in ls:
		if i > p:
			break
		if not y % i:
			flag = 1
			break
	if not flag:
		print(y)
		exit(0)
	y -= 1
print(-1)
