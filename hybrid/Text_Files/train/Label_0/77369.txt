import sys

def menorah(n, a, b):
	_01 = 0
	_00 = 0
	_10 = 0
	_11 = 0
	for i in range(n):
		if a[i] == '0':
			if b[i] == '0':
				_00 += 1
			else:
				_01 += 1
		elif b[i] == '1':
			_11 += 1
		else:
			_10 += 1
	best = float('inf')
	if _01 == _10:
		best = min(best, _01 + _10)
	if _11 > 0:
		__01 = _11 - 1
		__00 = _10
		__10 = _00
		__11 = _01 + 1
		if __01 == __10:
			best = min(best, 1 + __01 + __10)
	if best == float('inf'):
		best = -1
	return best
for _ in range(int(input())):
	n = int(input())
	a = input()
	b = input()
	sys.stdout.write(f'{menorah(n, a, b)}\n')
