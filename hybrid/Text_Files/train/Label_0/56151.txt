import java.io.*;
import java.util.*;

public class CF54B extends PrintWriter {
	CF54B() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF54B o = new CF54B(); o.main(); o.flush();
	}

	byte[][] cc;
	boolean equal0(int i1, int j1, int i2, int j2, int x, int y) {
		for (int i = 0; i < x; i++)
			for (int j = 0; j < y; j++)
				if (cc[i1 + i][j1 + j] != cc[i2 + i][j2 + j])
					return false;
		return true;
	}
	boolean equal2(int i1, int j1, int i2, int j2, int x, int y) {
		for (int i = 0; i < x; i++)
			for (int j = 0; j < y; j++)
				if (cc[i1 + i][j1 + j] != cc[i2 + x - 1 - i][j2 + y - 1 - j])
					return false;
		return true;
	}
	boolean equal1(int i1, int j1, int i2, int j2, int x) {
		for (int i = 0; i < x; i++)
			for (int j = 0; j < x; j++)
				if (cc[i1 + i][j1 + j] != cc[i2 + j][j2 + x - 1 - i])
					return false;
		return true;
	}
	boolean equal3(int i1, int j1, int i2, int j2, int x) {
		for (int i = 0; i < x; i++)
			for (int j = 0; j < x; j++)
				if (cc[i1 + i][j1 + j] != cc[i2 + x - 1 - j][j2 + i])
					return false;
		return true;
	}
	boolean same(int i1, int j1, int i2, int j2, int x, int y) {
		if (equal0(i1, j1, i2, j2, x, y) || equal2(i1, j1, i2, j2, x, y))
			return true;
		return x == y && (equal1(i1, j1, i2, j2, x) || equal3(i1, j1, i2, j2, x));
	}
	void main() {
		int n = sc.nextInt();
		int m = sc.nextInt();
		cc = new byte[n][];
		for (int i = 0; i < n; i++)
			cc[i] = sc.next().getBytes();
		int cnt = 0, xy_ = n * m, x_ = n, y_ = m;
		for (int x = 1; x <= n; x++) {
			if (n % x != 0)
				continue;
out:
			for (int y = 1; y <= m; y++) {
				if (m % y != 0)
					continue;
				int n_ = n / x, m_ = m / y;
				for (int a = 0; a < n_ * m_; a++) {
					int ia = a / m_, ja = a % m_;
					for (int b = a + 1; b < n_ * m_; b++) {
						int ib = b / m_, jb = b % m_;
						if (same(ia * x, ja * y, ib * x, jb * y, x, y))
							continue out;
					}
				}
				cnt++;
				int xy = x * y;
				if (xy_ > xy || xy_ == xy && x_ > x) {
					xy_ = xy; x_ = x; y_ = y;
				}
			}
		}
		println(cnt);
		println(x_ + " " + y_);
	}
}
