short dp[5001][5001];
void dfs(int x,int y,int depth){
    if(dp[y][x]>=depth)return;
    dp[y][x]=depth;
    if(depth==0)return;
    int ny=y+1;
    int nx=x+1;
    dfs(nx,ny,depth-1);
    ny=y;
    dfs(nx,ny,depth-1);
}

void solve(){
    memset(dp,-1,sizeof(dp));
    int n,m;
    cin>>n>>m;
    for(int i=0;i<m;i++){
        int x,y,a;
        cin>>x>>y>>a;
        dfs(x,y,a);
    }
    int cnt=0;
    for(int i=0;i<=n;i++){
        for(int j=0;j<=n;j++){
            if(dp[j][i]>=0)cnt++;
        }
    }
    cout<<cnt<<endl;
}
int main(){

    solve();

    return 0;
}
