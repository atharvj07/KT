import java.util.*;
import java .io.*;
public class Maths {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());
        ArrayList<Integer>prev = new ArrayList<>();
        int array[];
        int only[] = new int[c + 2];
        int set[] = new int[c + 2];
        int onlyflag = 0;
        for(int tot = 1;tot <= n;tot++){
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            array = new int[a];
            for(int j = 0;j < a;j++){
                array[j] = Integer.parseInt(st.nextToken());
            }
            boolean flag = false;
            for(int i = 0;i < Math.min(prev.size(),array.length);i++){
                if(array[i] == prev.get(i) || flag)continue;
                flag = true;
                if(array[i] < prev.get(i)){
                    onlyflag++;
                    only[c - prev.get(i) + 1] += 1;
                    only[c - array[i] + 1] -= 1;
                }
                else{
                    //System.out.println(i + " " + array[i] + " " + prev.get(i));
                    set[c - array[i] + 1] += 1;
                    set[c - prev.get(i) + 1] -= 1;
                }
            }
            if(!flag && prev.size() > array.length){
                System.out.print("-1");
                return;
            }
            prev = new ArrayList<>();
            for(int i = 0;i < array.length;i++)
                prev.add(array[i]);
        }
        for(int i = 1;i < only.length;i++){
            only[i] += only[i - 1];
            set[i] += set[i - 1];
        }
        //if(n == 1000){
          //  System.out.println(only + " " + set);
        for(int i = 0;i < c;i++){
                if(only[i] == onlyflag && set[i] == 0){
                    System.out.print(i);
                    return;
                }
            }
        System.out.print("-1");
    }
}
