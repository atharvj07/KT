	static class Task {
		
		int NN = 500005;
		int MOD = 1000000007;
		int INF = 2000000000;
		long INFINITY = 2000000000000000000L;
		
		class Pair {
			Integer value, index;

			public Pair(Integer value, Integer index) {
				super();
				this.value = value;
				this.index = index;
			}

			@Override
			public String toString() {
				return "Pair [value=" + value + ", index=" + index + "]";
			}
			
		}
		
		int [] a;
		List<Integer> [] g;
		Long [][] K;
		long [] sum;
		long [] sz;
		
		public void solve(InputReader in, PrintWriter out) {
			int n = in.nextInt();
			int m = in.nextInt();
			int q = in.nextInt();
			K = new Long[q][3];
			a = new int[m + 1];
			Arrays.fill(a, 0);
			List<Pair> L = new ArrayList<>();
			for(int i=1;i<=n;++i) {
				++a[in.nextInt()];
			}
			for(int i=1;i<=m;++i) {
				L.add(new Pair(a[i], i));
			}
			Collections.sort(L, (p1, p2) -> p1.value.equals(p2.value)
					? p1.index.compareTo(p2.index) : p1.value.compareTo(p2.value));
			g = new ArrayList[m];
			int tot = -1;
			for(int i=0;i<m;++i) {
				if(i == 0 || L.get(i - 1).value != L.get(i).value) {
					g[++tot] = new ArrayList<>();
				}
				g[tot].add(L.get(i).index);
			}
			++tot;
			sum = new long[tot];
			sz = new long[tot];
			sum[tot - 1] = -1;
			for(int i=0;i + 1 <tot;++i) {
				sum[i] = 0;sz[i]=0;
				if(i > 0) {
					sum[i] += sum[i - 1];
					sz[i] += sz[i - 1];
				}
				sz[i] +=  g[i].size();
				sum[i] += 1L*(a[g[i+1].get(0)]-a[g[i].get(0)])*sz[i];
			}
			for(int i=0;i<q;++i) {
				K[i][2] = (long)i;
				K[i][0] = in.nextLong() - n;
				if(tot == 1) {
					K[i][1]=(long)tot-1;
					continue;
				}
				if(sum[0] >= K[i][0]) {
					K[i][1] = 0L;
					K[i][0] %= sz[0];
					if(K[i][0] == 0)K[i][0] = sz[0];
					continue;
				}
				int lo = 0, hi = tot - 2, mid = 0;
				while(lo < hi) {
					mid = (lo + hi + 1) / 2;
					if(sum[mid] < K[i][0])
						lo = mid;
					else
						hi=  mid - 1;
				}
				K[i][0] -= sum[lo];
				++lo;
				K[i][1] = (long)lo;
				if(K[i][1] < tot-1) {
					K[i][0] %= sz[lo];
					if(K[i][0] == 0)K[i][0] = sz[lo];
				}
			}
			Arrays.sort(K, (i, j) -> 
			i[1] == j[1] ? i[2].compareTo(j[2]):i[1].compareTo(j[1]));
			int [] ans= new int[q];
			T = new int[4*m];
			Arrays.fill(T, 0);
			for(int i=0, j = 0;i<q;++i) {
				int index = K[i][1].intValue();
				if(index == tot - 1) {
					K[i][0] %= m;
					if(K[i][0] == 0)
						K[i][0] = (long)m;
					ans[K[i][2].intValue()] = K[i][0].intValue();
				}
				while(j <= index) {
					for(int e:g[j]) {
						update(1, 1, m, e, 1);
					}
					++j;
				}
				int pos = K[i][0].intValue();
				ans[K[i][2].intValue()] = query(1, 1, m, pos - 1);
			}
			for(int i=0;i<q;++i) {
				out.println(ans[i]);
			}
		}
		
		int query(int i, int l, int r, int cnt) {
			if(l == r)
				return l;
			int m = (l + r) / 2;
			if(T[2*i] > cnt)
				return query(2*i, l, m, cnt);
			else
				return query(1+2*i, 1+m, r, cnt-T[2*i]);
		}
		
		void update(int i, int l, int r, int x, int val) {
			if(x > r || x < l)
				return;
			if(l==r) {
				T[i] += val;return;
			}
			int m = (l + r) / 2;
			update(2*i, l, m, x, val);
			update(1+2*i, 1+m, r, x, val);
			T[i] = T[2*i] + T[1+2*i];
		}
		
		int [] T;
		
	}

	}
	
	static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }
        
        public InputReader(String filePath) {
        	File file = new File(filePath);
            try {
				reader = new BufferedReader(new FileReader(file));
			} catch (FileNotFoundException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
            tokenizer = null;
        }
        
        public String nextLine() {
        	String str = "";
        	try {
				str = reader.readLine();
			} catch (IOException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
        	return str;
        }
        
        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }

        public double nextDouble() {
        	return Double.parseDouble(next());
        }
        
    }
