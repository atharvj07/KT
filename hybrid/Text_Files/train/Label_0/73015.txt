import sys

def rall():
	return sys.stdin.readlines()

def rl():
	return sys.stdin.readline().strip()

def rl_types(types):
	str_list = [x for x in sys.stdin.readline().strip().split(' ')]
	return [types[i](str_list[i]) for i in range(len(str_list))]

def pr(something=''):
	sys.stdout.write(str(something) + '\n')

def pra(array):
	sys.stdout.write(' '.join([str(x) for x in array]) + '\n')

def solve(array):
	return array
NT = int(rl())
for ti in range(NT):
	(a, b, c) = map(int, rl().split(' '))
	m = max(a, b, c)
	tb = 1 if sum([m == i for i in [a, b, c]]) > 1 else 0
	pra([tb if a == m else m - a + 1, tb if b == m else m - b + 1, tb if c == m else m - c + 1])
