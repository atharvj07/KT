const int SZ = 51;

int dx[] = {0, -1, 0, 1};
int dy[] = {1, 0, -1, 0};

int H, W;
bool visit[SZ][SZ][SZ][SZ];
string LIN[SZ], REN[SZ];

inline bool inside(int x, int y){
  return 0 <= x && x < H && 0 <= y && y < W;
}

bool bfs(int lsx, int lsy, int rsx, int rsy){
  memset(visit, false, sizeof(visit));
  visit[lsx][lsy][rsx][rsy] = true;
  queue<int> que_lx, que_ly, que_rx, que_ry;
  que_lx.push(lsx);
  que_ly.push(lsy);
  que_rx.push(rsx);
  que_ry.push(rsy);
  while(!que_lx.empty()){

    int lx = que_lx.front(); que_lx.pop();
    int ly = que_ly.front(); que_ly.pop();
    int rx = que_rx.front(); que_rx.pop();
    int ry = que_ry.front(); que_ry.pop();
    
    if(LIN[lx][ly] == '%' && REN[rx][ry] == '%') return true;
    REP(i, 4){
      int nlx = lx + dx[i];
      int nly = ly + dy[i];
      int nrx = rx + dx[i];
      int nry = ry + dy[(i+2)%4];
      
      if(!inside(nlx, nly) || LIN[nlx][nly] == '#') nlx = lx, nly = ly;
      if(!inside(nrx, nry) || REN[nrx][nry] == '#') nrx = rx, nry = ry;
      if(!visit[nlx][nly][nrx][nry] && !((LIN[lx][ly] == '%') ^ (REN[rx][ry]=='%'))){
        visit[nlx][nly][nrx][nry] = true;
        que_lx.push(nlx);
        que_ly.push(nly);
        que_rx.push(nrx);
        que_ry.push(nry);        
      }
    }
  }
  
  return false;
}


int main(){
  while(cin >> W >> H && H + W > 0){
    REP(i, H) cin >> LIN[i] >> REN[i];
    int rsx, rsy, lsx, lsy;
    
    REP(i, H)REP(j, W){
      if(LIN[i][j] == 'L') lsx = i, lsy = j;
      if(REN[i][j] == 'R') rsx = i, rsy = j;
    }
    cout << (bfs(lsx, lsy, rsx, rsy) ? "Yes" : "No") << endl;
      
  }
  return 0;
}
