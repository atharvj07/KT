(n, q) = map(int, input().split())
arr = list(map(int, input().split()))
ones = arr.count(1)
zeros = n - ones
for i in range(q):
	(a, b) = map(int, input().split())
	if a == 1:
		if arr[b - 1] == 1:
			zeros += 1
			ones -= 1
			arr[b - 1] ^= 1
		else:
			ones += 1
			zeros -= 1
			arr[b - 1] ^= 1
	elif b > ones:
		print(0)
	else:
		print(1)
