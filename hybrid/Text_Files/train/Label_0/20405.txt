#include <bits/stdc++.h>
using namespace std;
int absolut(int x) {
  if (x < 0) return -x;
  return x;
}
int reversear, flipdiagonal, fliphorizontal, flipvertical;
vector<pair<int, int> > sol;
int i, j;
void ensol() { sol.push_back(pair<int, int>(i, j)); }
void muevei(int i2) {
  while (i != i2) {
    if (i < i2)
      i++;
    else
      i--;
    ensol();
  }
}
void muevej(int j2) {
  while (j != j2) {
    if (j < j2)
      j++;
    else
      j--;
    ensol();
  }
}
int n, m, ii1, jj1, i2, j2;
void termina() {
  int itop = i;
  int idown = n;
  while (j != j2) {
    if (i == itop) {
      muevei(idown);
      muevej(j + 1);
    } else {
      muevei(itop);
      muevej(j + 1);
    }
  }
  int movimiento = 1;
  if (i > i2) movimiento = -1;
  muevej(m);
  muevei(i + movimiento);
  while (absolut(i2 - i) > 1) {
    muevej(j2);
    muevei(i + movimiento);
    muevej(m);
    muevei(i + movimiento);
  }
  int pendiente = 0;
  if (i != i2) pendiente = 1;
  if (movimiento == 1) {
    muevei(idown);
    movimiento = -1;
  } else {
    muevei(itop);
    movimiento = 1;
  }
  muevej(j - 1);
  while (absolut(i2 - i) > 1) {
    muevej(j2);
    muevei(i + movimiento);
    muevej(m - 1);
    muevei(i + movimiento);
  }
  if (i == i2) {
    if (pendiente) {
      while (j > j2 + 1) {
        muevei(i + movimiento);
        muevej(j - 1);
        muevei(i - movimiento);
        muevej(j - 1);
      }
      if (j == j2 + 1) {
        muevei(i + movimiento);
        muevej(j - 1);
        muevei(i - movimiento);
      }
    } else {
      muevej(j2);
    }
  } else {
    if (pendiente) {
      while (j > j2 + 1) {
        muevei(i + 2 * movimiento);
        muevej(j - 1);
        muevei(i - 2 * movimiento);
        muevej(j - 1);
      }
      if (j == j2) {
        muevei(i + movimiento);
      } else {
        muevei(i + 2 * movimiento);
        muevej(j - 1);
        muevei(i - movimiento);
      }
    } else {
      while (j > j2 + 1) {
        muevei(i + movimiento);
        muevej(j - 1);
        muevei(i - movimiento);
        muevej(j - 1);
      }
      muevei(i + movimiento);
      if (j == j2 + 1) muevej(j - 1);
    }
  }
  if (flipvertical)
    for (int k = 0; k < int(sol.size()); k++)
      sol[k].first = n - sol[k].first + 1;
  if (fliphorizontal)
    for (int k = 0; k < int(sol.size()); k++)
      sol[k].second = m - sol[k].second + 1;
  if (flipdiagonal)
    for (int k = 0; k < int(sol.size()); k++) swap(sol[k].first, sol[k].second);
  if (reversear) reverse(sol.begin(), sol.end());
  cout << int(sol.size()) << endl;
  for (int k = 0; k < int(sol.size()); k++)
    cout << sol[k].first << " " << sol[k].second << endl;
}
void colocaleft() {
  if (j2 < jj1) {
    jj1 = m - jj1 + 1;
    j2 = m - j2 + 1;
    fliphorizontal = 1;
  }
}
void colocatop() {
  if (i2 < ii1) {
    ii1 = n - ii1 + 1;
    i2 = n - i2 + 1;
    flipvertical = 1;
  }
}
int main() {
  cin >> n >> m >> ii1 >> jj1 >> i2 >> j2;
  if (n % 2 == 0 or m % 2 == 0) {
    if (n % 2 != 0) {
      flipdiagonal = 1;
      swap(ii1, jj1);
      swap(i2, j2);
      swap(n, m);
    }
    colocaleft();
    if (ii1 % 2 == 1) {
      ii1 = n - ii1 + 1;
      i2 = n - i2 + 1;
      flipvertical = 1;
    }
    i = ii1;
    j = jj1;
    ensol();
    muevei(n);
    muevej(jj1 - 1);
    while (i != ii1) {
      muevej(1);
      muevei(i - 1);
      muevej(jj1 - 1);
      muevei(i - 1);
    }
    muevej(1);
    muevei(i - 1);
    muevej(jj1);
    while (i != 1) {
      muevei(i - 1);
      muevej(1);
      muevei(i - 1);
      muevej(jj1);
    }
    muevej(j + 1);
    termina();
  } else if (ii1 % 2 == jj1 % 2 or i2 % 2 == j2 % 2) {
    if (ii1 % 2 != jj1 % 2) {
      reversear = 1;
      swap(ii1, i2);
      swap(jj1, j2);
    }
    colocaleft();
    colocatop();
    i = ii1;
    j = jj1;
    ensol();
    if (ii1 % 2 == 1) {
      muevei(n);
      muevej(jj1 - 1);
      while (i != ii1) {
        muevej(1);
        muevei(i - 1);
        muevej(jj1 - 1);
        muevei(i - 1);
      }
      muevej(1);
      muevei(1);
      muevej(2);
      while (j != jj1 + 1) {
        muevei(ii1 - 1);
        muevej(j + 1);
        muevei(1);
        muevej(j + 1);
      }
      termina();
    } else {
      muevei(n);
      muevej(jj1 - 1);
      while (i != ii1 + 1) {
        muevej(1);
        muevei(i - 1);
        muevej(jj1 - 1);
        muevei(i - 1);
      }
      muevej(1);
      muevei(1);
      muevej(2);
      while (j != jj1) {
        muevei(ii1);
        muevej(j + 1);
        muevei(1);
        muevej(j + 1);
      }
      while (i != ii1 - 1) {
        muevej(m);
        muevei(i + 1);
        muevej(jj1);
        muevei(i + 1);
      }
      muevej(j + 1);
      termina();
    }
  } else {
    if (ii1 % 2 == 0) {
      flipdiagonal = 1;
      swap(ii1, jj1);
      swap(i2, j2);
      swap(n, m);
    }
    colocaleft();
    colocatop();
    i = ii1;
    j = jj1;
    ensol();
    muevei(n);
    muevej(jj1 - 1);
    while (i != ii1) {
      muevej(1);
      muevei(i - 1);
      muevej(jj1 - 1);
      muevei(i - 1);
    }
    muevej(1);
    muevei(i - 1);
    while (i != 2) {
      muevej(jj1);
      muevei(i - 1);
      muevej(1);
      muevei(i - 1);
    }
    while (j != jj1 - 1) {
      muevei(i - 1);
      muevej(j + 1);
      muevei(i + 1);
      muevej(j + 1);
    }
    muevej(j + 1);
    muevei(i - 1);
    muevej(j + 1);
    termina();
  }
}
