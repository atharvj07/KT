class TaskC {
    boolean works(long left, long right, long tbites, long sizem, long A, long B){
        long heightmin = (A + B * (left - 1)), heightmax = (A + B * (right - 1));
        if(heightmax > tbites) return false;

        long width = (right - left + 1);
        long heightall = (long) (Math.ceil(1.0 * width * avgheight / 2));
        if(heightall > tbites * sizem) return false;
        return true;
    }

    public void solve(int testNumber, InputReader in, OutputWriter out) {

        for(int query = 0; query < nqueries; query++) {
            int rmin = 0, rmax = (int) 1e9;
            int left=-1, tbites=-1,sizem=-1;
            if(in.peek()!=-1) {left = in.readInt();} else{throw new InputMismatchException();}
            if(in.peek()!=-1)  {tbites = in.readInt();} else{throw new InputMismatchException();}

                if(works(left, middle, tbites, sizem, A, B)) rmin = middle; else rmax = middle - 1;
            }
            out.printLine((rmin < left ? -1 : rmin));
        }
    }
}
