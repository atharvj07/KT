#include <bits/stdc++.h>
using namespace std;
int func(string &s, int a, int l, int r) {
  if (l == r) {
    if (s[l] - 'a' != a) return 1;
    return 0;
  }
  int m = (r + l) / 2, left = 0, right = 0;
  for (int i = l; i <= m; i++) {
    if (s[i] - 'a' != a) left++;
  }
  for (int i = m + 1; i <= r; i++) {
    if (s[i] - 'a' != a) right++;
  }
  return min(func(s, a + 1, m + 1, r) + left, func(s, a + 1, l, m) + right);
}
int ncr(int n, int k) {
  int C[n + 1][k + 1];
  int i, j;
  for (i = 0; i <= n; i++) {
    for (j = 0; j <= min(i, k); j++) {
      if (j == 0 || j == i)
        C[i][j] = 1;
      else
        C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % 1000000007;
    }
  }
  return C[n][k];
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int n, p, q, r;
  cin >> n >> p >> q >> r;
  long long int arr[n];
  for (int i = 0; i < n; i++) cin >> arr[i];
  long long int dp[n][3];
  dp[0][0] = p * arr[0];
  dp[0][1] = (q + p) * arr[0];
  dp[0][2] = (r + q + p) * arr[0];
  for (int i = 1; i <= n - 1; i++) {
    dp[i][0] = max(dp[i - 1][0], p * arr[i]);
  }
  for (int i = 1; i <= n - 1; i++) {
    dp[i][1] = max(dp[i - 1][1], dp[i][0] + q * arr[i]);
  }
  for (int i = 1; i <= n - 1; i++) {
    dp[i][2] = max(dp[i - 1][2], dp[i][1] + r * arr[i]);
  }
  cout << dp[n - 1][2] << "\n";
  return 0;
}
