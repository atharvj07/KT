#include <bits/stdc++.h>
const int maxn = 100005;
char s[maxn];
char t[maxn];
char ans[maxn];
int main() {
  scanf("%s%s", s, t);
  int len = strlen(s);
  int num = 0;
  for (int i = 0; i < len; i++) {
    if (s[i] != t[i]) num++;
  }
  if (num % 2 != 0)
    printf("impossible\n");
  else {
    num /= 2;
    for (int i = 0; i < len; i++) {
      if (s[i] != t[i] && num > 0) {
        num--;
        ans[i] = t[i];
      } else
        ans[i] = s[i];
    }
    for (int i = 0; i < len; i++) printf("%c", ans[i]);
    printf("\n");
  }
  return 0;
}
