/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        JRenovation solver = new JRenovation();
        solver.solve(1, in, out);
        out.close();
    }

    static class JRenovation {
        int n;
        int m;
        int[] months;
        int[] b;
        int[] p;

        public void readInput(Scanner sc) {
            n = sc.nextInt();
            m = sc.nextInt();
            months = new int[n];
            b = new int[m];
            p = new int[m];
            for (int i = 0; i < n; i++)
                months[i] = sc.nextInt();
            for (int i = 0; i < m; i++)
                b[i] = sc.nextInt();
            for (int i = 0; i < m; i++)
                p[i] = sc.nextInt();
        }

        public void solve(int testNumber, Scanner sc, PrintWriter pw) {
            int q = 1;
            while (q-- > 0) {
                readInput(sc);
                PriorityQueue<Integer>[] pqs = new PriorityQueue[n];
                Integer[] sort = new Integer[m];
                for (int i = 0; i < m; i++)
                    sort[i] = i;
                for (int i = 0; i < n; i++)
                    pqs[i] = new PriorityQueue<>();
                Arrays.sort(sort, (x, y) -> b[x] - b[y]);
                int idx = 0;
                for (int i = n - 1; i >= 0; i--)
                    while (idx < m && b[sort[idx]] <= months[i]) {
                        pqs[i].add(p[sort[idx]]);
                        idx++;
                    }
                PriorityQueue<Integer> money = new PriorityQueue<>(Collections.reverseOrder());
                long cur = 0;
                for (int i = 0; i < n; i++) {
                    cur += months[i];
                    while (!pqs[i].isEmpty()) {
                        if (pqs[i].peek() <= cur) {
                            cur -= pqs[i].peek();
                            money.add(pqs[i].poll());
                        } else {
                            if (money.size() > 0 && pqs[i].peek() < money.peek()) {
                                cur += money.poll();
                                cur -= pqs[i].peek();
                                money.add(pqs[i].poll());
                            } else
                                break;
                        }
                    }
                }
                pw.println(money.size());
            }
        }

    }

    static class Scanner {
        StringTokenizer st;
        BufferedReader br;

        public Scanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }

        public String next() {
            try {
                while (st == null || !st.hasMoreTokens())
                    st = new StringTokenizer(br.readLine());
                return st.nextToken();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
