#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int A[60] = {0}, B[60] = {0};
  vector<int> arr;
  for (int i = 0; i < n * n; i++) {
    int a, b;
    cin >> a >> b;
    if (A[a] < 1 && B[b] < 1) {
      A[a]++, B[b]++;
      arr.push_back(i + 1);
    }
  }
  for (int i = 0; i < arr.size(); i++) cout << arr[i] << " ";
}
