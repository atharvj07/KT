T = int(input())
for _ in range(T):
	s = set()
	(N, K) = [int(s) for s in input().split()]
	arr = [i for i in range(1, 2 * N + 1)]
	res = 0
	for _ in range(K):
		(a, b) = [int(s) for s in input().split()]
		if a > b:
			(a, b) = (b, a)
		for i in s:
			if (not a > i[0] or not b < i[1]) and (not a < i[0] or not b > i[1]):
				if a < i[1] and b > i[0]:
					res += 1
		s.add((a, b))
		arr.remove(a)
		arr.remove(b)
	while arr:
		mid = len(arr) // 2
		a = arr[0]
		b = arr[mid]
		arr.remove(a)
		arr.remove(b)
		for i in s:
			if (not a > i[0] or not b < i[1]) and (not a < i[0] or not b > i[1]):
				if a < i[1] and b > i[0]:
					res += 1
		s.add((a, b))
	print(res)
