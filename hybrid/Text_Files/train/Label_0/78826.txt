(n, k) = map(int, input().split())
fav = [0 for i in range(k + 1)]
for i in range(n):
	a = int(input())
	fav[a] += 1
ans = 0
K = (n + 1) // 2
for i in range(1, k + 1):
	if fav[i] % 2 == 0:
		ans += fav[i]
		K -= fav[i] // 2
		fav[i] = 0
	else:
		ans += fav[i] - 1
		K -= fav[i] // 2
		fav[i] = 1
ans += K
print(ans)
