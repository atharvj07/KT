n = int(input())
width = []
heigth = []
sumw = 0
(h1, h2) = (0, 0)
for i in range(n):
	(w, h) = map(int, input().split())
	width.append(w)
	heigth.append(h)
	sumw += w
	if h > h1:
		h2 = h1
		h1 = h
	elif h > h2:
		h2 = h
for i in range(n):
	if heigth[i] != h1:
		print((sumw - width[i]) * h1, end=' ')
	else:
		print((sumw - width[i]) * h2, end=' ')
