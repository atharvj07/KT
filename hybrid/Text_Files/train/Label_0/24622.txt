#include <bits/stdc++.h>
using namespace std;
int main() {
  int n = 0, m = 0, astr_cnt = 0;
  cin >> n >> m;
  char emnm[n][m];
  char tria[2][4];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> emnm[i][j];
      if (emnm[i][j] == '*') {
        tria[0][astr_cnt] = i;
        tria[1][astr_cnt] = j;
        astr_cnt++;
      }
    }
  }
  if (tria[0][0] != tria[0][1]) {
    cout << tria[0][0] + 1 << " ";
    if (tria[1][0] != tria[1][1]) {
      cout << tria[1][1] + 1 << " ";
    } else
      cout << tria[1][2] + 1 << endl;
  } else {
    cout << tria[0][2] + 1 << " ";
    if (tria[1][2] != tria[1][1]) {
      cout << tria[1][1] + 1 << " ";
    } else
      cout << tria[1][0] + 1 << endl;
  }
  return 0;
}
