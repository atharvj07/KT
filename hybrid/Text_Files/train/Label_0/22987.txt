#include <bits/stdc++.h>
using namespace std;
const long long INF = 1e18;
template <class... Args>
istream& read_args(Args&... args) {
  return (cin >> ... >> args);
}
template <typename T>
istream& operator>>(istream& is, vector<T>& v) {
  for (auto& i : v) is >> i;
  return is;
}
template <typename T>
ostream& operator<<(ostream& os, vector<T>& v) {
  for (auto& i : v) os << i << ' ';
  return os;
}
long long none(long long i, long long j, long long n, long long m) {
  if ((i == 0 && (j == 0 || j == m - 1)) ||
      (i == n - 1 && (j == 0 || j == m - 1)))
    return 2;
  else if (i == 0 || j == 0 || i == n - 1 || j == m - 1)
    return 3;
  else
    return 4;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long t = 1;
  cin >> t;
  while (t--) {
    long long n, m;
    read_args(n, m);
    long long a[n][m];
    for (long long i = (0); i < (long long)(n); i++)
      for (long long j = (0); j < (long long)(m); j++) cin >> a[i][j];
    bool ok = 1;
    for (long long i = (0); i < (long long)(n); i++) {
      for (long long j = (0); j < (long long)(m); j++) {
        long long ne = none(i, j, n, m);
        if (a[i][j] > ne)
          ok = 0;
        else
          a[i][j] = ne;
      }
    }
    if (ok) {
      cout << "YES" << endl;
      for (long long i = (0); i < (long long)(n); i++) {
        for (long long j = (0); j < (long long)(m); j++) cout << a[i][j] << " ";
        cout << "" << endl;
      }
    } else
      cout << "NO" << endl;
  }
  cerr << "Time taken : " << (long double)(clock()) / CLOCKS_PER_SEC << "s"
       << endl;
}
