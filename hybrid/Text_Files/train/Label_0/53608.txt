
import java.io.BufferedReader;

public class e {

    public static void main(String[] args) {
        FS in = new FS(System.in);
        PrintWriter out = new PrintWriter(System.out);

        new e().solve(in, out);

        out.close();
    }
    
    ArrayList<Integer>[] adj;
    int[] vis;
    int next;
    int[] val;
    
    public void solve(FS in, PrintWriter out) {
        int n = in.nextInt();
        int m = in.nextInt();
        adj = new ArrayList[n];
        vis = new int[n];
        val = new int[n];
        for(int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        for(int i = 0; i < m; i++) {
            int u = in.nextInt()-1;
            int v = in.nextInt()-1;
            adj[u].add(v);
            adj[v].add(u);
        }
        Random rand = new Random();
        int a = rand.nextInt(n);
        int b = rand.nextInt(n);
        int c = rand.nextInt(n);
//        int a = 0;
//        int b = 0;
//        int c = 1;
        int prev = -1;
        while(true) {
            out.println((a+1) + " " + (b+1) + " " + (c+1));
            out.flush();
            if(prev == a || prev == b || prev == c) {
                return;
            }
            int k = in.nextInt()-1;
            if(k == a || k == b || k == c) {
                return;
            }
            a = go(a, adj[k].get(0));
            b = go(b, adj[k].get(1));
            if(adj[k].size() == 2) {
                c = go(c, adj[k].get(1));
            } else {
                c = go(c, adj[k].get(2));
            }
            if(a == -1) a = k;
            if(b == -1) b = k;
            if(c == -1) c = k;
            prev = k;
        }
    }
    
    int go(int u, int v) {
        if(u == v) return -1;
        next++;
        ArrayDeque<Integer> q = new ArrayDeque<>();
        q.add(v);
        vis[v] = next;
        val[v] = -1;
        while(!q.isEmpty()) {
            int c = q.poll();
            if(c == u) return val[c];
            for(int a : adj[c]) {
                if(vis[a] == next) continue;
                val[a] = c;
                vis[a] = next;
                q.add(a);
            }
        }
        return -1;
    }
    
    static class FS {

        BufferedReader in;
        StringTokenizer token;

        public FS(InputStream str) {
            in = new BufferedReader(new InputStreamReader(str));
        }

        public String next() {
            if (token == null || !token.hasMoreElements()) {
                try {
                    token = new StringTokenizer(in.readLine());
                } catch (IOException ex) {
                }
                return next();
            }
            return token.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }
    }
}
