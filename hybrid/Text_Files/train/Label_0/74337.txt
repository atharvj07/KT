def from_a_2_b(a, b):
	count = 1
	nums = [b]
	while a < b:
		if b % 2 == 0:
			b = b / 2
			count += 1
			nums.insert(0, int(b))
		elif b % 10 == 1:
			b = b // 10
			count += 1
			nums.insert(0, int(b))
		else:
			b = 0
	if a != b:
		print('NO')
	else:
		print('YES')
		print(count)
		print(' '.join(map(str, nums)))
(a, b) = map(int, input().split())
from_a_2_b(a, b)
