import java.util.ArrayList;
import java.util.Collections;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;


public class Main {
    static Scanner scan = new Scanner(System.in);

    public static void slove() {
        /**
         * 0 1 3 4
         * a = 4 b = 2
         * s = 3
         *
         * 0 1 2 4
         *
         * a = 4
         * b = 3
         * 4
         *
         *
         * 0 1 4
         * a = 4 b = 2
         * 0 1 2 3 4 6
         * a = 4 b = 5;
         *
         * 1 2 3
         * a = 3
         * b = 0
         *
         * if(b == 0) retrun size;
         * if(a > b) return size or size + 1;
         * if(a<b) return size + k;
         */
        int n = scan.nextInt();
        int m = scan.nextInt();
        Set<Integer> hash = new HashSet<>();
        while(n-->0){
            hash.add(scan.nextInt());
        }
        if(m == 0){
            System.out.println(hash.size());
            return;
        }
        List<Integer> list = new ArrayList<>();
        list.addAll(hash);
        Collections.sort(list);
        int a = list.get(list.size() - 1);
        int b = 0;
        for (int x:list) {
            if(x == b) b++;
            else break;
        }

        if(a>b){
            int r = (a+b+1) / 2;
            int res = hash.size();
            if(!hash.contains(r)) res++;
            System.out.println(res);
            return;
        }else{
            System.out.println(hash.size()+m);
            return;
        }

    }

    public static void main(String[] args) {
        int t = scan.nextInt();
        while (t-- > 0) {
            slove();
        }
    }
}
