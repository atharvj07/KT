(n, k) = input().split(' ')
(n, k) = (int(n), int(k))
tab = input().split(' ')
l = len(tab)
nt = 0
ns = 0
for i in range(l):
	if tab[i] == '1':
		nt += 1
	else:
		ns += 1
dif = 0
premove = int(l / k)
for b in range(1, k + 1):
	s = 0
	t = 0
	for j in range(int(l / k) + 1):
		if b + j * k - 1 < l:
			if tab[b + j * k - 1] == '1':
				t += 1
			else:
				s += 1
		x = abs(nt - t - (ns - s))
	if dif < x:
		dif = x
print(dif)
