public class A {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();

        if (a % 2 == 1) {
            System.out.print(a / 2 + 1);
        } else {
            System.out.print(n / 2 - a / 2 + 1);
        }
    }
}
