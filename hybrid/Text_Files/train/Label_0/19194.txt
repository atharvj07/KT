#include <bits/stdc++.h>
char _;
using namespace std;
int N, A[100001], l[100001], r[100001];
int main() {
  scanf("%d", &N);
  for (int i = 1; i <= N; ++i) scanf("%d", &A[i]);
  l[1] = 1;
  for (int i = 2; i <= N; ++i) l[i] = min(l[i - 1] + 1, A[i - 1] + 1);
  r[N] = 1;
  for (int i = N - 1; i > 0; --i) r[i] = min(r[i + 1] + 1, A[i + 1] + 1);
  int ans = 0;
  for (int i = 1; i <= N; ++i) {
    A[i] = min(A[i], l[i]);
    A[i] = min(A[i], r[i]);
    ans = max(ans, A[i]);
  }
  cout << ans;
  return 0;
}
