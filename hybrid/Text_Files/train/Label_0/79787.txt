def read_int():
	return int(input())

def read_string_array():
	for s in input().split():
		yield s

def read_int_array():
	for s in read_string_array():
		yield int(s)

def read_line():
	return input()

def max_fill(n, k):
	cnt = 0
	while n > 0:
		cnt += 1
		n -= k + 1
	return cnt
from itertools import groupby

def solve(s, k):
	cnt = 0
	groups = groupby(s)
	groups = [(_, len(list(__))) for (_, __) in groups]
	for (i, (key, l)) in enumerate(groups):
		if key == '0':
			last_1 = i > 0
			next_1 = i != len(groups) - 1
		else:
			continue
		if last_1 and next_1:
			cnt += max_fill(l - 2 * k, k)
		elif last_1 or next_1:
			cnt += max_fill(l - k, k)
		else:
			cnt += max_fill(l, k)
	return cnt
T = read_int()
for _ in range(T):
	(N, K) = read_int_array()
	S = read_line()
	print(solve(S, K))
