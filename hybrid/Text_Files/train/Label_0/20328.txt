#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

string solve(string s){
    int n = 1<<10;
    vector<int> dp(n, 0);
    int curr = 0;
    for(char c: s){
        if(c == 'u'){
            curr = 0;
        }else{
            dp[curr] |= 1<<(c-'0');
            curr |= 1<<(c-'0');
        }
    }
    for(int i=0; i<n; i++){
        for(int j=0; j<i; j++){
            if(!(i&j) and (dp[j]&i)){
                dp[i|j] |= dp[i];
            }
        }
    }
    for(int i=0; i<n; i++){
        if(dp[i]&i){
            return "UNSAFE";
        }
    }
    return "SAFE";
}

int main(){
    while(1){
        int n;
        cin >> n;
        if(n == 0) break;

        string s;
        cin >> s;
        cout << solve(s) << endl;
    }
    return 0;
}
