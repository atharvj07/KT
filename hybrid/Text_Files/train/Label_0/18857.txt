#include <bits/stdc++.h>
using namespace std;
int n, m, tot;
char g[500 + 10][500 + 10];
struct node {
  char c;
  int x, y;
} ans[2000000];
int dir[4][2] = {1, 0, 0, 1, -1, 0, 0, -1};
void dfs(int x, int y, int s) {
  g[x][y] = 'B';
  ans[tot].c = 'B';
  ans[tot].x = x;
  ans[tot].y = y;
  tot++;
  for (int i = 0; i < 4; i++) {
    int r = x + dir[i][0], c = y + dir[i][1];
    if (r > 0 && r <= n && c > 0 && c <= m && g[r][c] == '.') dfs(r, c, s + 1);
  }
  if (s) {
    ans[tot].c = 'D';
    ans[tot].x = x;
    ans[tot].y = y;
    tot++;
    ans[tot].c = 'R';
    ans[tot].x = x;
    ans[tot].y = y;
    tot++;
  }
  return;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%s", g[i] + 1);
  tot = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (g[i][j] == '.') dfs(i, j, 0);
    }
  }
  printf("%d\n", tot);
  for (int i = 0; i < tot; i++)
    printf("%c %d %d\n", ans[i].c, ans[i].x, ans[i].y);
  return 0;
}
