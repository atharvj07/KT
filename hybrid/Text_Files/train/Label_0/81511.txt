(n, m) = map(int, input().split())
adj = [[] for i in range(n + 1)]
vis = [0] * (n + 1)

def dfs(v):
	vis[v] = 1
	for i in range(len(adj[v])):
		if vis[adj[v][i]] == 0:
			dfs(adj[v][i])
langs = [[] for i in range(n + 1)]
for i in range(1, n + 1):
	lang = list(map(int, input().split()))
	if len(lang) == 1:
		continue
	for j in range(1, len(lang)):
		langs[i].append(lang[j])
for i in range(1, n + 1):
	for j in range(i + 1, n + 1):
		if len(set(langs[i]).intersection(set(langs[j]))) != 0:
			adj[i].append(j)
			adj[j].append(i)
cc = 0
flag = 0
for i in range(1, n + 1):
	if len(langs[i]) == 0:
		flag += 1
	if vis[i] == 0:
		cc += 1
		dfs(i)
if flag == n:
	cc += 1
print(cc - 1)
