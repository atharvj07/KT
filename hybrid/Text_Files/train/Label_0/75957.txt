import sys
import collections

def input():
	return sys.stdin.readline().rstrip()

def split_input():
	return [int(i) for i in input().split()]
(n, p) = split_input()
a = split_input()
start = max(a) - n + 1
x = [0 for i in range(n)]
for i in a:
	ind = i - start
	if ind <= 0:
		x[0] += 1
	else:
		x[ind] += 1
for i in range(1, n):
	x[i] += x[i - 1]
zero_ind = -1
for i in range(n):
	zero_ind = max(i - x[i], zero_ind)
s = set()
for i in range(zero_ind + 1, n):
	if x[i] >= p:
		s.add((i + start - x[i]) % p)
ans = []
for i in range(zero_ind + 1, p):
	if (i + start) % p not in s:
		ans.append(i + start)
print(len(ans))
print(*ans, sep=' ')
