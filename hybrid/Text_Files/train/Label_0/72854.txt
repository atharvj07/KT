t = int(input())
MOD = 10 ** 9 + 7
lazyfac = [1, 1]
inversfac = [1, 1]

def upfac(lazyfac, inversefac, n):
	while len(lazyfac) < n + 1:
		lazyfac.append(lazyfac[-1] * len(lazyfac) % MOD)
		inversfac.append(pow(lazyfac[-1], -1, MOD))

def choose(n, m):
	return lazyfac[n] * inversfac[n - m] * inversfac[m]
for test in range(t):
	(n, m, k) = [int(i) for i in input().split()]
	a = [int(i) for i in input().split()]
	a.sort()
	upfac(lazyfac, inversfac, n + 20)
	ans = 0
	j = 0
	if m == 1:
		ans = n
	else:
		for i in range(n - m + 1):
			if j < n:
				while a[j] <= a[i] + k:
					j += 1
					if j == n:
						break
			if j >= i + m:
				ans += choose(j - i - 1, m - 1)
				ans = ans % MOD
	print(ans)
