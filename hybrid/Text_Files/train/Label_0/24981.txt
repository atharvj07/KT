#include <bits/stdc++.h>
#define N 10050
using namespace std;
clock_t st;
int n, m, K, e = 2, head[N], dfn[N], low[N], tot, tim, bo[N * 20], ok[N * 20], be[N], Dp[N];
int fa[N][18], rt[N], dep[N], pp[N], a[N], b[N], dta[N], dtb[N];
struct edge{int u, v, next;}ed[N * 20];
void add(int u, int v){ed[e].u = u; ed[e].v = v; ed[e].next = head[u]; head[u] = e++;}
vector<pair<int, int> > V[N];
vector<int> vr;
int cnt = 0;
void tarjan(int x, int fa){
	++cnt;
	if(cnt > 100000) assert(0);
	dfn[x] = low[x] = ++tim;
	for(int i = head[x]; i; i = ed[i].next){
		int v = ed[i].v;
		if(!dfn[v]){
			tarjan(v, x);
			low[x] = min(low[x], low[v]);
			if(low[v] > dfn[x]) bo[i] = bo[i ^ 1] = 1;
		}
		else if(dfn[v] < dfn[x] && v != fa)
			low[x] = min(low[x], dfn[v]);
	}
}
void dfs(int x, int d, int fa){
	be[x] = tot; Dp[x] = d;
	for(int i = head[x]; i; i = ed[i].next){
		if(bo[i]) continue;
		int v = ed[i].v;
		if(!be[v]) {ok[i] = 1; dfs(v, d + 1, x);}
		else if(v != fa && Dp[v] < Dp[x]) {ok[i] = 1;}
	}
}
void dfs(int x, int root){
	rt[x] = root;
	for(int i = 1; (1 << i) <= dep[x]; i++)
		fa[x][i] = fa[fa[x][i - 1]][i - 1];
	for(int i = 0; i < (int)V[x].size(); i++){
		int v = V[x][i].first;
		if(v == fa[x][0]) continue;
		pp[v] = V[x][i].second;
		fa[v][0] = x;
		dep[v] = dep[x] + 1;
		dfs(v, root);
	}
}
int getlca(int x, int y){
	if(dep[x] < dep[y]) swap(x, y);
	for(int i = 14; ~i; i--)
		if(dep[fa[x][i]] >= dep[y]) x = fa[x][i];
	if(x == y) return x;
	for(int i = 14; ~i; i--)
		if(fa[x][i] != fa[y][i]) x = fa[x][i], y = fa[y][i];
	return fa[x][0];
}
void dfs(int x){
	a[x] = dta[x];
	b[x] = dtb[x];
	for(int i = 0; i < (int)V[x].size(); i++){
		int v = V[x][i].first;
		if(v == fa[x][0]) continue;
		dfs(v);
		a[x] += a[v];
		b[x] += b[v];
	}
	if(a[x] && b[x]){puts("No"); exit(0);}
	else if(a[x]) ok[pp[x] ^ 1] = 1;
	else ok[pp[x]] = 1;
}
int main(){
	st = clock();
	scanf("%d%d", &n, &m);
	for(int i = 1, u, v; i <= m; i++){
		scanf("%d%d", &u, &v);
		add(u, v); add(v, u);
	}
	for(int i = 1; i <= n; i++)
		if(!dfn[i]) tarjan(i, 0);
	for(int i = 1; i <= n; i++){
		if(!be[i]){
			++tot;
			dfs(i, 1, 0);
		}
	}
	for(int i = 2; i < e; i++) if(bo[i])
		V[be[ed[i].u]].push_back(make_pair(be[ed[i].v], i));
	for(int i = 1; i <= tot; i++){
		if(!rt[i]){
			fa[i][0] = 0;
			dep[i] = 1;
			dfs(i, i);
			vr.push_back(i);
		}
	}
	scanf("%d", &K);
	for(int i = 1, s, t, lca; i <= K; i++){
		scanf("%d%d", &s, &t);
		s = be[s]; t = be[t];
		if(rt[s] != rt[t]) {puts("No"); return 0;}
		lca = getlca(s, t);
		dta[s]++; dta[lca]--;
		dtb[t]++; dtb[lca]--;
	}
	for(int i = 0; i < (int)vr.size(); i++)
		dfs(vr[i]);
	puts("Yes");
	for(int i = 2; i < e; i++) if(ok[i])
		printf("%d %d\n", ed[i].u, ed[i].v);
	return 0;
}
