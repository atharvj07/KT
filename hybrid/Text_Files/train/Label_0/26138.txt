#include <bits/stdc++.h>
using namespace std;
#define modulo 1000000007
#define mod(mod_x) ((((long long)mod_x+modulo))%modulo)
#define Inf 1000000000

int beki(long long a,long long b,int M = modulo){
	int x = 1%M;
	while(b!=0){
		if(b&1){
			x=((long long)x*a)%M;
		}
		a=((long long)a*a)%M;
		b>>=1;
	}
	return x%M;
}

int get(int t){
	for(int i=0;true;i++){
		if(t==0)return i;
		int cnt = 0;
		int tt = t;
		while(tt!=0){
			if(tt%2==1)cnt++;
			tt/=2;
		}
		t %= cnt;
	}
}

int main(){
	
	int N;
	cin>>N;
	
	string S;
	cin>>S;
	
	int X = 0;
	for(int i=0;i<S.size();i++){
		if(S[i]=='1')X++;
	}
	
	if(X==0){
		for(int i=0;i<N;i++){
			cout<<1<<endl;
		}
		return 0;
	}
	int A = 0,B = 0;
	for(int i=0;i<N;i++){
		A *= 2;
		B *= 2;
		A += S[i]-'0';
		B += S[i]-'0';
		if(X!=1)A %= X-1;
		else A=0;
		B %= X+1;
	}
	
	for(int i=0;i<N;i++){
		if(S[i]=='0'){
			int t = B+ beki(2,N-1-i,X+1);
			t %= X+1;
			cout<<1+get(t)<<endl;
		}
		else{
			if(X==1){
				cout<<0<<endl;
				continue;
			}
			int t = A - beki(2,N-1-i,X-1);
			t += 2*(X-1);
			t %= X-1;
			cout<<1+get(t)<<endl;
		}
	}
	
	return 0;
}
