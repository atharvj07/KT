#include<bits/stdc++.h>
#define MOD 1000000007LL
using namespace std;

typedef long long ll;
typedef pair<int,int> P;

int h,w;
int fie[501][501];
vector<P> vec[501][501];
bool flag[501][501];
int sx,sy;
int route[501][501];
int f=0;

void dfs(int y,int x){
	flag[y][x]=true;
	if(fie[y][x]==1){
		f=1;
	}
	if(f==1 && fie[y][x]==0){
		fie[y][x]=2;
	}
	for(int i=0;i<vec[y][x].size();i++){
		P nv=vec[y][x][i];
		if(!flag[nv.first][nv.second]){
			dfs(nv.first,nv.second);
		}
	}
}

int main(void){
	scanf("%d%d",&h,&w);
	for(int i=0;i<h;i++){
		string str;
		cin >> str;
		for(int j=0;j<w;j++){
			if(str[j]=='@'){
				fie[i][j]=3;
				sy=i;
				sx=j;
			}
			if(str[j]=='*'){
				fie[i][j]=1;
			}
		}
	}
	for(int i=0;i<w-1;i++){
		vec[0][i].push_back(P(0,i+1));
		vec[0][i+1].push_back(P(0,i));
	}
	for(int i=0;i<2;i++){
		vec[i][0].push_back(P(i+1,0));
		vec[i+1][0].push_back(P(i,0));
		vec[i][w-1].push_back(P(i+1,w-1));
		vec[i+1][w-1].push_back(P(i,w-1));
	}
	int side=0;
	for(int i=0;i<w-1;i++){
		if(side==0){
			vec[h-1][i].push_back(P(h-1,i+1));
			vec[h-1][i+1].push_back(P(h-1,i));
		}else{
			vec[2][i].push_back(P(2,i+1));
			vec[2][i+1].push_back(P(2,i));
		}
		if((i+1)%2==0){
			side=1-side;
		}
	}
	for(int i=0;i<w;i+=2){
		for(int j=2;j<h;j++){
			vec[j][i].push_back(P(j+1,i));
			vec[j+1][i].push_back(P(j,i));
		}
	}
	dfs(sy,sx);
	for(int i=0;i<h;i++){
		for(int j=0;j<w;j++){
			if(fie[i][j]==0){
				printf(".");
			}
			if(fie[i][j]==1){
				printf("*");
			}
			if(fie[i][j]==2){
				printf("#");
			}
			if(fie[i][j]==3){
				printf("@");
			}
		}
		printf("\n");
	}
	return 0;
}
