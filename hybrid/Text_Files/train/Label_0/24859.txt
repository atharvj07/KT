long long mod(long long n) {
  return (n % 1000000007 + 1000000007) % 1000000007;
}

int tree[4 * 200005];
void constructtree(int input[], int l, int h, int p) {
  if (l == h) {
    tree[p] = input[l];
    return;
  }
  int m = (l + h) / 2;
  constructtree(input, l, m, 2 * p + 1);
  constructtree(input, m + 1, h, 2 * p + 2);
  tree[p] = min(tree[2 * p + 1], tree[2 * p + 2]);
}
int RMQ(int start, int end, int l, int h, int p) {
  if (end < l || h < start) {
    return 1000000007;
  }
  if (start <= l && h <= end) {
    return tree[p];
  }
  int m = (l + h) / 2;
  return min(RMQ(start, end, l, m, 2 * p + 1),
             RMQ(start, end, m + 1, h, 2 * p + 2));
}
