import java.util.*;
import java.io.*;

public class Main {
	BufferedReader in;
	PrintWriter out;
	StringTokenizer str;

	private String next() throws Exception {
		if (str == null || !str.hasMoreElements())
			str = new StringTokenizer(in.readLine());
		return str.nextToken();
	}

	private int nextInt() throws Exception {
		return Integer.parseInt(next());
	}

	int n;
	P []p;

	final int WIDTH = 1000;
	
	public void run() throws Exception {
		in = new BufferedReader(new InputStreamReader(System.in));
		out = new PrintWriter(System.out);
		
		n = nextInt();
		p = new P[n];
		for(int i = 0; i < n; ++i) {
			p[i] = new P(nextInt(), nextInt(), i + 1);
		}
		
		Arrays.sort(p);
		int at = 0;

		for(int i = 0; i <= 1000; ++i) {
			int lo = i * WIDTH, hi = (i + 1) * WIDTH;
			List<P> pts = new ArrayList<>();
			while(at < n && lo <= p[at].x && p[at].x < hi) {
				pts.add(p[at++]);
			}
			if (i % 2 == 0) {
				Collections.sort(pts, new Comparator<P>(){
					public int compare(P p1, P p2) {
						if (p1.y != p2.y)
							return p1.y - p2.y;
						return p1.x - p2.x;
					}
				});
			} else {
				Collections.sort(pts, new Comparator<P>() {
					public int compare(P p1, P p2) {
						if (p1.y != p2.y)
							return p2.y - p1.y;
						return p2.x - p1.x;
					}
				});
			}
			for(P pt : pts) out.print(pt.pos + " ");
		}
		out.println();
		out.close();
	}

	class P implements Comparable<P> {
		int x, y, pos;
		public P(int x, int y, int pos) {
			this.x = x;
			this.y = y;
			this.pos = pos;
		}

		public int compareTo(P other) {
			if (this.x != other.x) 
				return this.x - other.x;
			return this.y - other.y;
		}
	}

	public static void main(String args[]) throws Exception {
		new Main().run();
	}
}
