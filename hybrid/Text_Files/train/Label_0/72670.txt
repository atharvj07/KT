for _ in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	l2 = list(map(int, input().split()))
	l.sort()
	l3 = []
	l4 = [0]
	kk = 1
	count = 0
	c = 0
	i = 0
	j = 0
	ans = 0
	while i < n and j < n:
		if l[j] - l[i] <= k and j >= i:
			ans = max(ans, j - i + 1)
			l3.append(ans)
			j += 1
		else:
			i += 1
	i = n - 1
	j = n - 1
	ans = 0
	while i > 0 and j > 0:
		if l[i] - l[j] <= k and j <= i:
			ans = max(ans, i - j + 1)
			l4.append(ans)
			j -= 1
		else:
			i -= 1
	c = len(l3)
	ans = 0
	for i in range(c):
		ans = max(ans, l3[i] + l4[n - i - 1])
	print(ans)
