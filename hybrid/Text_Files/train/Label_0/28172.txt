#include <bits/stdc++.h>
using namespace std;
int a[30], b[30], c[30], mul[30];
int n, ans;
int dfs(int d) {
  if (a[d] == n) {
    printf("%d\n", d - 1);
    for (int i = 1; i < d; ++i) {
      printf("lea e%cx, [", char('a' + i));
      if (c[i] != 0) {
        printf("e%cx + ", 'a' + b[i] - 1);
        if (mul[i] == 0) {
          printf("e%cx", 'a' + c[i] - 1);
        } else {
          printf("%d*e%cx", (1 << mul[i]), 'a' + c[i] - 1);
        }
      } else {
        printf("%d*e%cx", (1 << mul[i]), 'a' + b[i] - 1);
      }
      printf("]\n");
    }
    return 1;
  }
  if (d > ans) return 0;
  for (int i = 1; i <= d; ++i) {
    for (int j = 1; j < 4; ++j) {
      a[d + 1] = a[i] << j;
      b[d] = i;
      c[d] = 0;
      mul[d] = j;
      if (dfs(d + 1)) return 1;
    }
    for (int j = 1; j <= d; ++j) {
      for (int k = 0; k < 4; ++k) {
        a[d + 1] = a[i] + (a[j] << k);
        b[d] = i;
        c[d] = j;
        mul[d] = k;
        if (dfs(d + 1)) return 1;
      }
    }
  }
  return 0;
}
int main() {
  cin >> n;
  a[1] = 1;
  for (ans = 0;; ans++) {
    if (dfs(1)) {
      break;
    }
  }
}
