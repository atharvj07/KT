def readline():
	return map(int, input().split())
MOD = 10 ** 9 + 7
f = [1, 1]

def factorial(x):
	l = len(f)
	while l <= x:
		f.append(f[l - 1] * l % MOD)
		l += 1
	return f[x]

def inv(x):
	return pow(x, -1, MOD)

def comb(n, k):
	return factorial(n) * inv(factorial(k)) % MOD * inv(factorial(n - k)) % MOD

def solve():
	(n, l, r) = readline()
	d = min(r - n, 1 - l)
	h = n >> 1
	partitions = ((h, h + 1), (h + 1, h)) if n & 1 else ((h, h),)
	ans = sum((comb(n, p) for (p, __) in partitions)) * d % MOD
	loop = True
	while loop:
		d += 1
		plus = max(0, d - (1 - l))
		minus = max(0, d - (r - n))
		loop = False
		for (p_, m_) in partitions:
			p = p_ - plus
			m = m_ - minus
			if p >= 0 and m >= 0:
				ans += comb(p + m, p)
				ans %= MOD
				loop = True
	print(ans)

def main():
	t = int(input())
	for __ in range(t):
		solve()
main()
