#include <bits/stdc++.h>
using namespace std;
int dr[]{1, -1, 0, 0, 1, 1, -1, -1};
int dc[]{0, 0, 1, -1, 1, -1, 1, -1};
void run() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
}
int n, m;
char arr[59][59];
bool vis[59][59];
bool valid(int i, int j) {
  return 0 <= i && i < n && 0 <= j && j < m && !vis[i][j] && arr[i][j] == '#';
}
void dfs(int i, int j) {
  vis[i][j] = true;
  for (int k = 0; k < 4; k++) {
    int x = i + dr[k], y = j + dc[k];
    if (valid(x, y)) {
      dfs(x, y);
    }
  }
}
int count() {
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (!vis[i][j] && arr[i][j] == '#') {
        cnt++;
        dfs(i, j);
      }
    }
  }
  return cnt;
}
int main() {
  run();
  cin >> n >> m;
  int cnt = 0, c = 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      cin >> arr[i][j];
      c += (arr[i][j] == '#');
    }
  cnt = count();
  if (cnt <= 1 && c <= 2) return cout << -1, 0;
  if (cnt > 1) return cout << 0, 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (arr[i][j] == '#') {
        memset(vis, 0, sizeof(vis));
        vis[i][j] = true;
        cnt = count();
        if (cnt > 1) return cout << 1, 0;
      }
    }
  }
  cout << 2 << '\n';
}
