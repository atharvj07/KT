#include <bits/stdc++.h>
using namespace std;
long long t, x, y, p, q;
int judge(long long m) {
  long long a = m * p - x, b = m * q - y;
  if (a >= 0 && b >= 0 && a <= b) return 1;
  return 0;
}
int main() {
  cin >> t;
  while (t--) {
    cin >> x >> y >> p >> q;
    long long l = 0, r = 1000000000, mid, ans = -1;
    while (l <= r) {
      mid = l + r >> 1;
      if (judge(mid))
        ans = mid, r = mid - 1;
      else
        l = mid + 1;
    }
    if (ans == -1)
      cout << -1 << endl;
    else
      printf("%lld\n", ans * q - y);
  }
}
