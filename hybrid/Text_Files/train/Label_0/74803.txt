t = int(input())
for i in range(t):
	(n, m) = map(int, input().split())
	arr = []
	for j in range(n):
		x = input()
		arr.append(int(x, 2))
	current_median = 2 ** (m - 1) - 1
	to_left = (2 ** m - n - 1) // 2 + 1
	while True:
		curr_to_left = current_median + 1
		present = False
		for j in arr:
			if j == current_median:
				present = True
			if j <= current_median:
				curr_to_left -= 1
		if curr_to_left == to_left and present == False:
			print(bin(current_median)[2:].zfill(m))
			break
		elif curr_to_left < to_left:
			current_median += 1
		else:
			current_median -= 1
