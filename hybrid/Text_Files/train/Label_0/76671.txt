t = int(input())
while t > 0:
	t -= 1
	n = int(input())
	args = list(map(int, input().split()))
	players = [(i + 1, args[i]) for i in range(n)]
	players.sort(key=lambda x: x[1])
	s = 0
	cur = 0
	for i in range(n - 1):
		s += players[i][1]
		if s < players[i + 1][1]:
			cur = i + 1
	ans = []
	for i in range(cur, n):
		ans.append(players[i][0])
	ans.sort()
	print(len(ans))
	print(*ans)
