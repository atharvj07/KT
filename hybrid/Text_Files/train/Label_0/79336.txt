from math import sqrt

def main():
	(n, r) = map(float, input().split())
	d = r * 2.0
	d2 = d * d
	l = []
	for x in map(float, input().split()):
		(le, ri, h) = (x - d, x + d, r)
		for (u, v) in l:
			if le <= u <= ri:
				w = x - u
				y = v + sqrt(d2 - w * w)
				if h < y:
					h = y
		l.append((x, h))
		print(h, end=' ')
main()
