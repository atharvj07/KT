#include <bits/stdc++.h>
using namespace std;
long long a[6], b[100000], cur[100000];
long long n;
vector<long long> v;
vector<pair<long long, long long>> x;
int32_t main() {
  for (long long i = 0; i < 6; ++i) cin >> a[i];
  sort(a, a + 6, greater<long long>());
  cin >> n;
  long long ma = 0;
  for (long long i = 0; i < n; ++i) {
    cin >> b[i];
    for (long long j = 0; j < 6; ++j) {
      v.push_back(b[i] - a[j]);
      x.push_back({b[i] - a[j], i});
    }
    ma = max(ma, b[i] - a[0]);
  }
  sort(v.begin(), v.end());
  sort(x.begin(), x.end());
  long long xi = 0;
  bool bad = false;
  long long res = 1e18;
  for (long long w : v) {
    while (xi < x.size() && x[xi].first < w) {
      long long i = x[xi++].second;
      if (cur[i] == 5) {
        bad = true;
        break;
      }
      ma = max(ma, b[i] - a[++cur[i]]);
    }
    if (bad) break;
    res = min(res, ma - w);
  }
  cout << res << "\n";
}
