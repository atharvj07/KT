import math
t = int(input())
for l in range(t):
	n = int(input())
	a = list(map(int, input().rstrip().split()))
	b = list(map(int, input().rstrip().split()))
	a.sort(reverse=True)
	b.sort(reverse=True)
	k = n - n // 4
	sa = sum(a[:k])
	sb = sum(b[:k])
	if sa >= sb:
		print(0)
	else:
		count = 0
		while sa < sb:
			d = abs(sa - sb)
			tmp = math.ceil(d / 100)
			count += tmp
			k = n + count - (n + count) // 4
			if k > n:
				sb = sum(b)
			else:
				sb = sum(b[:k])
			if count >= k:
				sa = 100 * k
			else:
				sa = count * 100 + sum(a[:k - count])
		print(count)
