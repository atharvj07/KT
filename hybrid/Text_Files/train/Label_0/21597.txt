#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 12;
int a[N];
int n;
vector<pair<int, int> > need;
vector<pair<int, int> > t;
int pr[N];
int sz[N];
int s_find(int x) {
  if (x == pr[x]) return x;
  return s_find(pr[x]);
}
void s_join(int x, int y) {
  y = s_find(y);
  x = s_find(x);
  if (sz[x] > sz[y]) swap(x, y);
  pr[x] = y;
  sz[y] += sz[x];
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    int x;
    scanf("%d", &x);
    t.push_back(make_pair(-x, i));
    pr[i] = i;
  }
  sort(t.begin(), t.end());
  for (int i = 0; i < n; i++) {
    int len = 1;
    sz[t[i].second] = 1;
    if (t[i].second != 1 && sz[s_find(t[i].second - 1)] != 0) {
      len += sz[s_find(t[i].second - 1)];
      s_join(t[i].second - 1, t[i].second);
    }
    if (t[i].second != n && sz[s_find(t[i].second + 1)] != 0) {
      len += sz[s_find(t[i].second + 1)];
      s_join(t[i].second, t[i].second + 1);
    }
    need.push_back(make_pair(t[i].first, len));
  }
  sort(need.begin(), need.end());
  for (int i = 1, j = 0; i <= n; i++) {
    while (need[j].second < i) {
      j++;
    }
    printf("%d ", -need[j].first);
  }
}
