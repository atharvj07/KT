import java.util.Scanner;

public class SpiderMan{

	public static void main(String[] args) {

		Scanner kbd = new Scanner(System.in);
		int n=kbd.nextInt();
		kbd.nextLine();
		int sum=0;
		for(int i=0;i<n;i++)
		{
			sum+= kbd.nextInt()-1;
			if(sum%2==0)
				System.out.println("2");
			else
				System.out.println("1");
		}
		System.exit(0);
	}

}
