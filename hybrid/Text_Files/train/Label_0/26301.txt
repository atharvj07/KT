#include <bits/stdc++.h>
using namespace std;
template <class T>
bool uin(T& a, T b) {
  return a > b ? (a = b, true) : false;
}
template <class T>
bool uax(T& a, T b) {
  return a < b ? (a = b, true) : false;
}
const int nax = 2020;
int N, K;
char G[nax][nax];
int cnt[nax][nax];
vector<int> dp[2];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> N >> K;
  for (int i = 0; i < N; ++i) {
    for (int j = 0; j < N; ++j) {
      cin >> G[i][j];
      if (i) uax(cnt[i][j], cnt[i - 1][j]);
      if (j) uax(cnt[i][j], cnt[i][j - 1]);
      cnt[i][j] += (int)(G[i][j] == 'a');
    }
  }
  string ans;
  dp[0] = {0};
  for (int len = 0; len < 2 * N - 1; ++len) {
    dp[(len & 1) ^ 1].clear();
    bool is_a = false;
    for (int j = max(0, len - N + 1); j <= min(len, N - 1); ++j) {
      if (cnt[j][len - j] + K < len + 1) continue;
      is_a = true;
      if (j + 1 < N) dp[(len & 1) ^ 1].push_back(j + 1);
      if (len - j + 1 < N) dp[(len & 1) ^ 1].push_back(j);
    }
    if (is_a) {
      ans += 'a';
      continue;
    }
    char mc = 'z';
    for (int j : dp[len & 1]) uin(mc, G[j][len - j]);
    ans += mc;
    for (int j : dp[len & 1]) {
      if (G[j][len - j] != mc) continue;
      if (j + 1 < N) dp[(len & 1) ^ 1].push_back(j + 1);
      if (len - j + 1 < N) dp[(len & 1) ^ 1].push_back(j);
    }
    sort(dp[len & 1 ^ 1].begin(), dp[len & 1 ^ 1].end());
    dp[len & 1 ^ 1].resize(
        unique(dp[len & 1 ^ 1].begin(), dp[len & 1 ^ 1].end()) -
        dp[len & 1 ^ 1].begin());
  }
  cout << ans << '\n';
}
