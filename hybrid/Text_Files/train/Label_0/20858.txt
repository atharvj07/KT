#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
int f[110], tx, c, t;
char s[110];
void read() {
  int k;
  scanf("%d", &k);
  for (int i = 1; i <= k; i++) {
    scanf("%s", s);
    int l = strlen(s);
    for (int j = 0; j < l; j++) f[s[j] - '0'] = i;
  }
}
int main() {
  while (1) {
    printf("next 0 1\n");
    fflush(stdout);
    read();
    printf("next 0\n");
    fflush(stdout);
    read();
    tx++;
    if (f[0] == f[1]) break;
  }
  while (1) {
    printf("next 0 2\n");
    fflush(stdout);
    read();
    c++;
    if (f[0] == f[1]) break;
  }
  while (1) {
    t++;
    printf("next 2 3 4 5 6 7 8 9\n");
    fflush(stdout);
    read();
    if (f[2] == f[3]) break;
  }
  int x = (tx - t) % c;
  if (x) {
    int left = c - x;
    for (int i = 1; i <= left; i++) {
      printf("next 0 1\n");
      fflush(stdout);
      read();
    }
  }
  printf("done\n");
  fflush(stdout);
  return 0;
}
