import math
firstLineInputs = list(map(int, input().split()))
amount = firstLineInputs[0]
target = firstLineInputs[1]
increments = list(map(int, input().split()))
graph = {}
for i in range(1, target):
	x = i + increments[i - 1]
	if x <= target:
		graph[str(i)] = str(x)
start = '1'
while True:
	if str(target) == start:
		print('YES')
		break
	if start in graph:
		start = graph[start]
	else:
		print('NO')
		break
