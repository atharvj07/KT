#include <bits/stdc++.h>
using namespace std;
const int dx[] = {-1, -1, 1, 1}, dy[] = {-1, 1, -1, 1};
const int maxn = 505, inf = 0x3f3f3f3f;
char a[maxn][maxn];
int n;
bool judge(int x, int y) {
  if (a[x][y] != 'X') return false;
  for (int i = 0; i < 4; ++i) {
    int nx = x + dx[i], ny = y + dy[i];
    if (nx < 0 || nx >= n || ny < 0 || ny >= n || a[nx][ny] != 'X')
      return false;
  }
  return true;
}
void solve() {
  for (int i = 0; i < n; ++i) scanf("%s", a[i]);
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if (judge(i, j)) ++ans;
    }
  }
  printf("%d\n", ans);
  return;
}
int main() {
  while (scanf("%d", &n) != EOF) solve();
  return 0;
}
