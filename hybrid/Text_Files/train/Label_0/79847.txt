import sys

def input():
	return sys.stdin.readline().rstrip('\r\n')

def maps():
	return [int(i) for i in input().split()]
from operator import xor

def findXOR(n):
	mod = n % 4
	if mod == 0:
		return n
	elif mod == 1:
		return 1
	elif mod == 2:
		return n + 1
	elif mod == 3:
		return 0

def findXORFun(l, r):
	return xor(findXOR(l - 1), findXOR(r))
for _ in range(*maps()):
	(a, b) = maps()
	x = findXORFun(0, a - 1)
	n = a
	if x == b:
		print(n)
	else:
		xi = bin(x)[2:]
		bb = bin(b)[2:]
		if len(bb) < len(xi):
			bb = '0' * (len(xi) - len(bb)) + bb
		if len(xi) < len(bb):
			xi = '0' * (len(bb) - len(xi)) + xi
		A = ['0'] * len(xi)
		ok = False
		for i in range(len(xi)):
			if xi[i] == '1' and bb[i] == '1':
				ok = True
				A[i] = '0'
			elif xi[i] == '1' and bb[i] == '0':
				A[i] = '1'
			elif xi[i] == '0' and bb[i] == '1':
				A[i] = '1'
		AA = ''.join(A)
		if int(AA, 2) != a:
			print(n + 1)
		else:
			print(n + 2)
