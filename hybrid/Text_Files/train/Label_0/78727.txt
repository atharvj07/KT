import sys
input = sys.stdin.readline
(h, w) = map(int, input().split())
l = [list(input().strip()) for _ in range(h)]
dx = [-1, 0, 1, 0]
dy = [0, 1, 0, -1]
i = 0
j = 0
(first_i, first_j) = (-1, -1)
for i in range(h):
	for j in range(w):
		if l[i][j] == '*':
			first_i = i
			first_j = j
			break
	if first_j != -1:
		break
if first_i == -1:
	print('NO')
else:
	direction = -1
	for k in range(4):
		ni = i + dx[k]
		nj = j + dy[k]
		if 0 <= ni < h and 0 <= nj < w and (l[ni][nj] == '*'):
			direction = k
	if direction == -1:
		print('NO')
	else:
		(center_i, center_j) = (-1, -1)
		(i, j) = (first_i, first_j)
		pcl1 = (direction + 1) % 4
		pcl2 = (direction + 3) % 4
		while True:
			l[i][j] = '.'
			pi = i + dx[pcl1]
			pj = j + dy[pcl1]
			if 0 <= pi < h and 0 <= pj < w and (l[pi][pj] == '*'):
				pi = i + dx[pcl2]
				pj = j + dy[pcl2]
				if 0 <= pi < h and 0 <= pj < w and (l[pi][pj] == '*'):
					if center_i != -1:
						print('NO')
						sys.exit()
					else:
						center_i = i
						center_j = j
				else:
					print('NO')
					sys.exit()
			ni = i + dx[direction]
			nj = j + dy[direction]
			if not (0 <= ni < h and 0 <= nj < w and (l[ni][nj] == '*')):
				(end_i, end_j) = (i, j)
				break
			(i, j) = (ni, nj)
		if center_i == -1:
			print('NO')
			sys.exit()
		if (first_i, first_j) != (center_i, center_j) and (end_i, end_j) != (center_i, center_j):
			i = center_i
			j = center_j
			while True:
				pi = i + dx[pcl1]
				pj = j + dy[pcl1]
				if 0 <= pi < h and 0 <= pj < w and (l[pi][pj] == '*'):
					l[pi][pj] = '.'
				else:
					break
				i = pi
				j = pj
			i = center_i
			j = center_j
			while True:
				pi = i + dx[pcl2]
				pj = j + dy[pcl2]
				if 0 <= pi < h and 0 <= pj < w and (l[pi][pj] == '*'):
					l[pi][pj] = '.'
				else:
					break
				i = pi
				j = pj
		else:
			print('NO')
			sys.exit()
		for i in range(h):
			for j in range(w):
				if l[i][j] != '.':
					print('NO')
					sys.exit()
		print('YES')
