t = int(input())
from math import comb
for i in range(t):
	(n, k) = map(int, input().split())
	A = list(map(int, input().split()))
	A.sort()
	match = 0
	val = -1
	for j in range(k):
		if A[n - 1 - j] == val:
			match += 1
		else:
			match = 1
			val = A[-1 - j]
	math_2 = 0
	for j in range(n - k - 1, -1, -1):
		if A[j] == val:
			math_2 += 1
		else:
			break
	print(comb(match + math_2, match) % (10 ** 9 + 7))
