#include <bits/stdc++.h>
#define fi first
#define se second
#define mp make_pair
#define ss(x) (int) x.size()
#define pb push_back
#define ll long long
#define cat(x) cerr << #x << " = " << x << endl
#define FOR(i, n) for(int i = 0; i < n; ++i)

using namespace std;	

int n;

vector <string> v[4] = {
	{
		"ccba",
		"aaba",
		"cbcc",
		"cbaa"
	}
	,
	{
		"ac.aa",
		"acbb.",
		"caa.b",
		"c.c.b",
		"aacaa"
	}
	, 
	{
		"a.ac..",
		"a.ac..",
		".abbaa",
		".a.abb",
		"ccba..",
		"aab.aa"
	}
	,
	{
		"caabb..",
		"cb..aa.",
		"ab....b",
		"a..c..b",
		"..bc.aa",
		"b.baa..",
		"baa..aa"
	}
};

char s[1111][1111];
			
int main() {
	scanf("%d", &n);
	if(n == 2) 
		return printf("-1\n"), 0;
	if(n == 3) {
		printf(".aa\n");
		printf("a..\n");
		printf("a..\n");
		return 0;
	}
	int r = n % 4;
	int w = n - (r + 4);
	
	for(int i = 1; i <= n; ++i)
		for(int j = 1; j <= n; ++j)
			s[i][j] = '.';
	int k = 1;
	for(int i = 0; i < ss(v[r]); ++i) 
		for(int j = 0; j < ss(v[r][i]); ++j)
			s[i + k][j + k] = v[r][i][j];
	k += (r + 4);
			
	for(int i = 0; i < w / 4; ++i) {
		for(int i = 0; i < 4; ++i)
			for(int j = 0; j < 4; ++j)
				s[i + k][j + k] = v[0][i][j];
		k += 4;
	}
	
	for(int i = 1; i <= n; ++i) {
		for(int j = 1; j <= n; ++j)
			printf("%c", s[i][j]);
		printf("\n");
	}
	
	
	return 0;
}
