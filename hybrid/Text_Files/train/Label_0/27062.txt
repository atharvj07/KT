int largest_edges(int v)
{
	int res, largest = -1;
	for (int i = v, prev = edge[v][0].first; ; )
	{
		used[i] = true;
		int next, cost;
		foreach (e, edge[i])
			if (e->first != prev)
				next = e->first, cost = e->second;

		if (largest == cost)
			++res;
		else if (cost > largest)
			largest = cost, res = 1;

		prev = i;
		i = next;
		if (next == v)
			break;
	}
	return res;
}
