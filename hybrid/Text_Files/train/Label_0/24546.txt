int binary_sear(const vector<int>& v, int x) {
  while (1) {
    mid = floor((low + high) / 2);
    if (low > high) break;
    if (v[mid] == x) {
      return mid;
      break;
    } else if (v[mid] < x) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
