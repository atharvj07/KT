l = []
for _ in range(int(input())):
	k = tuple(input().split())
	l.append(k)
o = list(map(int, input().split()))
c = ''
ans = True
for j in o:
	i = j - 1
	if l[i][0] > c and l[i][1] > c:
		if l[i][0] < l[i][1]:
			c = l[i][0]
		else:
			c = l[i][1]
	elif l[i][0] > c:
		c = l[i][0]
	elif l[i][1] > c:
		c = l[i][1]
	else:
		ans = False
		break
if ans:
	print('YES')
else:
	print('NO')
