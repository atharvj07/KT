#include <bits/stdc++.h>
using namespace std;
void solution() {
  int n;
  cin >> n;
  string s = "abcd";
  for (int i = 0; i < n; i++) {
    cout << (s[i % 4]);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  long long int t;
  t = 1;
  while (t--) {
    solution();
    cout << "\n";
  }
}
