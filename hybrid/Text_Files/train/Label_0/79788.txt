import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inlist():
	return list(map(int, input().split()))

def instr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())

def diff(x, y):
	if x[1] - x[0] > y[1] - y[0]:
		return 1
	elif x[1] - x[0] == y[1] - y[0]:
		return x[0] < y[0]
	else:
		return 0

def tables(x, k):
	if x - k <= 0:
		return 0
	else:
		return (x - k) // (k + 1)

def main():
	t = inp()
	for l in range(t):
		(n, k) = invr()
		s = instr()
		(a, e) = (0, 0)
		if s[0] != '1':
			e += k
		for i in range(n):
			if s[i] == '0':
				e += 1
			elif s[i] == '1':
				a += tables(e, k)
				e = 0
		if s[n - 1] != '1':
			e += k
			a += tables(e, k)
		print(a)
main()
