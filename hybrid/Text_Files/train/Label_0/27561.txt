#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  unsigned long long T, x1, y1, x2, y2;
  cin >> T;
  while (T--) {
    cin >> x1 >> y1 >> x2 >> y2;
    if (x1 == x2 || y1 == y2) {
      cout << 1 << endl;
    } else if (x1 > x2 || y1 > y2) {
      cout << 0 << endl;
    } else {
      cout << ((x2 - x1) * (y2 - y1)) + 1 << endl;
    }
  }
  return 0;
}
