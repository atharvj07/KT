def solution(x, h):
	if len(x) == 1:
		return 1
	ans = 0
	maxR = x[0] - h[0] - 1
	for inx in range(len(x) - 1):
		if x[inx] - h[inx] > maxR:
			ans += 1
			maxR = x[inx]
		elif x[inx] + h[inx] < x[inx + 1]:
			ans += 1
			maxR = x[inx] + h[inx]
		else:
			maxR = x[inx]
	return ans + 1
x = []
h = []
n = int(input())
for i in range(n):
	(a, b) = map(int, input().split())
	x.append(a)
	h.append(b)
print(solution(x, h))
