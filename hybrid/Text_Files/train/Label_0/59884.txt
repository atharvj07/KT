import java.io.BufferedInputStream;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Iterator;
import java.util.Scanner;

public class PageNumbers { //Round #34 - Page Numbers
	public static void main(String[] args) {
		Scanner sc = new Scanner(new BufferedInputStream(System.in));

		String input = sc.next();
		String[] splitted = input.split(",");

		ArrayList<Integer> numbers = new ArrayList<Integer>();
		for(int i = 0; i < splitted.length; i++) {
			numbers.add(Integer.valueOf(splitted[i]));
		}

		Collections.sort(numbers);

		//Remove duplicates
		int previousNum, num = Integer.MIN_VALUE;
		for(Iterator<Integer> it = numbers.iterator(); it.hasNext(); ) {
			previousNum = num;
			num = it.next();

			if(num == previousNum) {
				it.remove();
			}
		}

		StringBuilder sb = new StringBuilder();
		int sequenceStartPos = 0;
		for(int i = 1; i < numbers.size(); i++) {
			if(numbers.get(i) != numbers.get(i - 1) + 1) {
				if(sequenceStartPos == i - 1) {
					sb.append(numbers.get(i - 1) + ",");
				} else {
					sb.append(numbers.get(sequenceStartPos) + "-" + numbers.get(i - 1) + ",");
				}

				sequenceStartPos = i;
			} else if(i == numbers.size() - 1) {
				sb.append(numbers.get(sequenceStartPos) + "-" + numbers.get(i) + ",");
			}
		}

		//Handle last number and when we only have one number in the array
		if(sequenceStartPos == numbers.size() - 1) {
			sb.append(numbers.get(sequenceStartPos));
		}

		//Handle the case where the whole array is a sequence
		if(sb.length() == 0) {
			sb.append(numbers.get(0) + "-" + numbers.get(numbers.size() - 1));
		}

		if(sb.toString().endsWith(",")) {
			System.out.println(sb.substring(0, sb.length() - 1));
		} else {
			System.out.println(sb);
		}
	}
}
