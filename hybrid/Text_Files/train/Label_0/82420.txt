for _ in range(int(input())):
	(l, m) = map(int, input().split(' '))
	li = [int(i) for i in input().split()]
	odd = 0
	for i in li:
		if i % 2 == 1:
			odd += 1
	ans = []
	x = m - 1
	if not odd < m:
		if (odd - m) % 2 == 0:
			for i in range(l):
				if li[i] % 2 == 1 and x > 0:
					ans.append(i + 1)
					x -= 1
			ans.append(l)
			print('YES')
			print(*ans)
		else:
			print('NO')
	else:
		print('NO')
