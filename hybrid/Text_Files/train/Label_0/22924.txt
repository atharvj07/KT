char getchar() {
  if (__builtin_expect(S == T, 0)) {
    T = (S = buf) + fread(buf, 1, L, stdin);
    return (S == T ? EOF : *S++);
  }
  return *S++;
}
int inp() {
  int x = 0, f = 1;
  char ch;
  for (ch = getchar(); !isdigit(ch); ch = getchar())
    if (ch == '-') f = -1;
  for (; isdigit(ch); x = x * 10 + ch - '0', ch = getchar())
    ;
  return x * f;
}
unsigned inpu() {
  unsigned x = 0;
  char ch;
  for (ch = getchar(); !isdigit(ch); ch = getchar())
    ;
  for (; isdigit(ch); x = x * 10 + ch - '0', ch = getchar())
    ;
  return x;
}

}

long long power(long long a, long long b, long long p) {
  if (!b) return 1;
  long long t = power(a, b / 2, p);
  t = t * t % p;
  if (b & 1) t = t * a % p;
  return t;
}
long long exgcd(long long a, long long b, long long &x, long long &y) {
  if (b == 0) {
    x = 1;
    y = 0;
    return a;
  }
  long long px, py;
  long long d = exgcd(b, a % b, px, py);
  x = py;
  y = px - a / b * py;
  return d;
}

template <class int32, class int64>
struct modint {
 public:
  static int32 mod;
  int32 x;
  modint(int32 a = 0) {
    if (a < 0)
      x = a % mod + mod;
    else if (a >= mod)
      x = a % mod;
    else
      x = a;
  }
  static void setmod(int32 m) { mod = m; }
  friend modint &operator+=(modint &a, modint b) {
    a.x += b.x;
    if (a.x >= mod) a.x -= mod;
    return a;
  }
  friend modint &operator-=(modint &a, modint b) {
    a.x -= b.x;
    if (a.x < 0) a.x += mod;
    return a;
  }
  friend modint &operator*=(modint &a, modint b) {
    a.x = (int64)a.x * b.x % mod;
    return a;
  }
  friend modint operator+(modint a, modint b) { return a += b; }
  friend modint operator-(modint a, modint b) { return a -= b; }
  friend modint operator*(modint a, modint b) { return a *= b; }
};
const int MAXN = 100010;
const int MAXK = 210;
const int MOD = 1000000007;
const f80 MI = f80(1) / MOD;
const int INF = 100001;
template <>
int modint<int, long long>::mod = MOD;
int n, k;
modint<int, long long> S[MAXK][MAXK];
vector<int> v[MAXN];
int size[MAXN];
modint<int, long long> f[MAXN][MAXK], g[MAXN][MAXK], tmp[MAXK];
modint<int, long long> ans[MAXK];
void dfs(int x, int p) {
  f[x][0] = 1;
  for (auto y : v[x]) {
    if (y == p) continue;
    dfs(y, x);
    for (int i = 0; i <= k && i <= size[y]; ++i) ans[i] -= g[y][i];
    for (int i = 0; i <= k && i <= size[x] + size[y]; ++i) tmp[i] = 0;
    for (int i = 0; i <= k && i <= size[x]; ++i)
      for (int j = 0; j <= k && j <= size[y]; ++j)
        tmp[i + j] += f[x][i] * g[y][j];
    size[x] += size[y];
    for (int i = 0; i <= k && i <= size[x]; ++i) f[x][i] = tmp[i];
  }
  size[x]++;
  for (int i = 0; i <= k && i <= size[x]; ++i) {
    f[x][i] = f[x][i] * 2;
    g[x][i] = f[x][i] + (i ? f[x][i - 1] : 0) - (i == 1);
    ans[i] += f[x][i];
  }
}
int main() {
  S[0][0] = 1;
  for (int i = 1; i <= 200; ++i)
    for (int j = 1; j <= i; ++j) S[i][j] = S[i - 1][j - 1] + S[i - 1][j] * j;
  n = inp();
  k = inp();
  for (int i = 1; i < n; ++i) {
    int x = inp();
    int y = inp();
    v[x].push_back(y);
    v[y].push_back(x);
  }
  dfs(1, 0);
  modint<int, long long> ret = 0;
  modint<int, long long> fact = 1;
  for (int i = 0; i <= k; ++i) {
    ret += S[k][i] * fact * ans[i];
    fact *= i + 1;
  }
  cout << ret.x << endl;
  return 0;
}
