#include <bits/stdc++.h>
using namespace std;
const int maxn = 200 + 10;
int n, m, vis[maxn][maxn], cnt[maxn];
vector<int> to[maxn];
void dfs(int x) {
  for (int i = 0, v; i < to[x].size(); i++) {
    v = to[x][i];
    if (vis[x][v]) continue;
    vis[x][v] = 1, vis[v][x] = -1;
    dfs(v);
  }
}
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    memset(vis, 0, sizeof(vis));
    memset(cnt, 0, sizeof(cnt));
    scanf("%d %d", &n, &m);
    int idx = 0;
    for (int i = 0, u, v; i < m; i++) {
      scanf("%d %d", &u, &v);
      cnt[u]++, cnt[v]++;
      to[u].push_back(v);
      to[v].push_back(u);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (cnt[i] & 1) {
        to[0].push_back(i);
        to[i].push_back(0);
      } else
        ans++;
    }
    for (int i = 0; i <= n; i++)
      if (to[i].size()) {
        dfs(i);
        to[i].clear();
      }
    printf("%d\n", ans);
    for (int i = 1; i <= n; i++)
      for (int j = 1; j <= n; j++)
        if (vis[i][j] == 1) printf("%d %d\n", i, j);
  }
}
