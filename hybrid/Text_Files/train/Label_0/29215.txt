#include <bits/stdc++.h>
using namespace std;
char _;
const int maxN = 200 * 1000 + 5;
const int maxM = 1000 * 1000 + 5;
const int mod = 1000 * 1000 * 1000 + 7;
const int base = 701;
const int SQ = 330;
vector<pair<int, int> > v[maxN], xx;
bool mark[maxN];
int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    v[x].push_back({y, i});
    xx.push_back({y, i});
  }
  if (k == n + 1) {
    cout << 0 << endl;
    return 0;
  }
  int tmp = 0, ps = 0;
  for (int i = maxN - 1; i >= 0; i--) {
    tmp += v[i].size();
    if (tmp >= k) {
      ps = i;
      break;
    }
  }
  if (ps > n) {
    cout << -1 << endl;
    return 0;
  }
  long long ans = (1LL) * mod * mod;
  sort(xx.begin(), xx.end());
  if (ps + 2 <= n) {
    long long tt = 0;
    for (int i = 0; i < ps + 2; i++) tt += xx[i].first;
    ans = min(ans, tt);
  }
  {
    int dif = k - (tmp - v[ps].size()) - 1;
    vector<pair<int, int> > yy;
    for (int i = 0; i < (int)v[ps].size(); i++) yy.push_back(v[ps][i]);
    if (ps)
      for (int i = 0; i < (int)v[ps - 1].size(); i++)
        yy.push_back(v[ps - 1][i]);
    sort(yy.begin(), yy.end());
    long long zz = 0, cur = ps;
    for (int i = 0; i < (int)yy.size() - dif; i++)
      zz += yy[i].first, cur--, mark[yy[i].second] = 1;
    for (int i = 0; i < (int)n; i++) {
      if (cur <= 0) break;
      if (mark[xx[i].second]) continue;
      zz += xx[i].first;
      cur--;
    }
    ans = min(ans, zz);
  }
  memset(mark, 0, sizeof mark);
  if (ps + 1 <= n) {
    int dif = k - (tmp - v[ps].size() - v[ps + 1].size()) - 1;
    vector<pair<int, int> > yy;
    for (int i = 0; i < (int)v[ps].size(); i++) yy.push_back(v[ps][i]);
    for (int i = 0; i < (int)v[ps + 1].size(); i++) yy.push_back(v[ps + 1][i]);
    sort(yy.begin(), yy.end());
    long long zz = 0, cur = ps + 1;
    for (int i = 0; i < (int)yy.size() - dif; i++)
      zz += yy[i].first, mark[yy[i].second] = 1, cur--;
    for (int i = 0; i < n; i++) {
      if (cur <= 0) break;
      if (mark[xx[i].second]) continue;
      zz += xx[i].first;
      cur--;
    }
    ans = min(ans, zz);
  }
  if (ans == (1LL) * mod * mod)
    cout << -1 << endl;
  else
    cout << ans << endl;
  return 0;
}
