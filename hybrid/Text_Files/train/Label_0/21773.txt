#include <bits/stdc++.h>
using namespace std;
template <class T>
T gcd(T a, T b) {
  if (b == 0)
    return a;
  else
    return gcd(b, a % b);
}
template <class T>
T power(T a, T b) {
  T result = 1;
  while (b > 0) {
    if (b & 1) result = result * a;
    a = a * a;
    b >>= 1;
  }
  return result;
}
template <class T>
T powerMod(T a, T b, T c) {
  T result = 1 % c;
  while (b > 0) {
    if (b & 1) result = (result * a) % c;
    a = (a * a) % c;
    b >>= 1;
  }
  return result;
}
long long R[100100], L[100100], ans[100100];
char str[100100];
void eval() {
  long long n, last0, last1, res, i;
  cin >> n;
  cin >> str;
  if (n <= 2) {
    cout << n << "\n";
    return;
  }
  res = 0;
  R[n] = 0;
  last0 = n;
  last1 = n;
  for (i = n - 1; i >= 0; i--) {
    if (str[i] == '1') {
      R[i] = R[last0] + 1;
      last1 = i;
    } else {
      R[i] = R[last1] + 1;
      last0 = i;
    }
  }
  for (i = 0; i < n; i++) {
    res = max(res, R[i]);
  }
  cout << min(res + 2, n) << "\n";
}
int main() {
  cin.sync_with_stdio(0);
  cout.sync_with_stdio(0);
  eval();
  return 0;
}
