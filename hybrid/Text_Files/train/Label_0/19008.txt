#include <bits/stdc++.h>
#pragma warning(disable : 4786)
using namespace std;
vector<long> Edge[350 + 7];
long Row, Col;
long nEdge, nVertex;
bool Visit[350 + 7];
long M[350 + 7];
void ini() {
  memset(M, 0, sizeof(M));
  memset(Visit, 0, sizeof(Visit));
  int i;
  for (i = 0; i < (350 + 7); i++) {
    Edge[i].clear();
  }
}
bool Match(long u) {
  if (Visit[u]) return false;
  Visit[u] = true;
  long i;
  for (i = 0; i < Edge[u].size(); i++) {
    long v = Edge[u][i];
    if (!M[v]) {
      M[u] = v;
      M[v] = u;
      return true;
    } else if (Match(M[v])) {
      M[u] = v;
      M[v] = u;
      return true;
    }
  }
  return false;
}
int BMatch() {
  memset(&M[1], 0, sizeof(long) * nVertex);
  long i;
  for (i = 1; i <= Row; i++) {
    if (M[i]) continue;
    memset(&Visit[1], 0, sizeof(bool) * (Row + Col));
    Match(i);
  }
  int ret = 0;
  int j = Row + 1;
  for (i = 1; i <= Row; i++) {
    if (M[i]) {
      ret++;
    } else {
    }
  }
  return ret;
}
int mat[500][500];
int vis[10][10];
char grid[10][10];
int ans;
vector<int> taken(10);
vector<pair<int, int> > O1(28);
vector<int> O2(28);
vector<pair<int, int> > save1(28);
vector<int> save2(28);
int pt[155][155];
string cols[100];
string dic[12];
int dic_cnt[12];
int dx[2] = {0, 1};
int dy[2] = {1, 0};
map<vector<int>, int> mp;
vector<int> sts[10];
int glo = 0;
void get() {
  int i, sum = 0, mn, j;
  if (mp[taken]) return;
  mp[taken] = 1;
  vector<int> co, ro;
  for (i = 0; i < 10; i++) {
    mn = ((taken[i]) < (dic_cnt[i]) ? (taken[i]) : (dic_cnt[i]));
    for (j = 0; j < dic_cnt[i] - mn; j++) {
      ro.push_back(i);
    }
    for (j = 0; j < taken[i] - mn; j++) {
      co.push_back(i);
    }
    sum += 2 * mn;
  }
  ini();
  Col = co.size();
  Row = ro.size();
  nVertex = Col + Row;
  if ((sum + ((Col) < (Row) ? (Col) : (Row))) <= ans) return;
  for (i = 0; i < Row; i++)
    for (j = 0; j < Col; j++) {
      if (pt[co[j]][ro[i]]) {
        Edge[i + 1].push_back(j + 1 + Row);
        Edge[j + 1 + Row].push_back(i + 1);
      }
    }
  sum += BMatch();
  if (sum <= ans) return;
  ans = sum;
  save1 = O1;
  save2 = O2;
  for (i = 0; i < 10; i++) sts[i].clear();
  for (i = 0; i < 10; i++) {
    mn = ((taken[i]) < (dic_cnt[i]) ? (taken[i]) : (dic_cnt[i]));
    for (j = 0; j < mn; j++) sts[i].push_back(i);
  }
  j = 1;
  for (i = Row + 1; i <= nVertex; i++) {
    if (M[i]) {
      sts[co[i - 1 - Row]].push_back(ro[M[i] - 1]);
    } else {
      while (M[j]) j++;
      sts[co[i - 1 - Row]].push_back(ro[j - 1]);
      j++;
    }
  }
  return;
}
void bttk(int x, int y, int cnt) {
  int i, X, Y, j;
  if (x >= 7) {
    get();
    return;
  }
  if (y >= 8) {
    bttk(x + 1, 0, cnt);
    return;
  }
  if (vis[x][y]) {
    bttk(x, y + 1, cnt);
    return;
  }
  vis[x][y] = 1;
  for (i = 0; i < 2; i++) {
    X = x + dx[i];
    Y = y + dy[i];
    if (x >= 0 && X < 7 && Y >= 0 && Y < 8 && !vis[X][Y]) {
      vis[X][Y] = 1;
      O1[cnt].first = x;
      O1[cnt].second = y;
      O2[cnt] = i;
      taken[mat[grid[x][y]][grid[X][Y]]]++;
      bttk(x, y + 1, cnt + 1);
      taken[mat[grid[x][y]][grid[X][Y]]]--;
      vis[X][Y] = 0;
    }
  }
  vis[x][y] = 0;
}
