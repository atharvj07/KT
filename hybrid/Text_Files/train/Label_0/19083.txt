#include<bits/stdc++.h>
using namespace std;

#define MAX 1010
#define INF (1<<30)

double costSum[MAX][MAX],optCost[MAX][MAX];

//{{{ Functions
void ComputeOptCost(int i, int j){
	optCost[i][j]=INF;
	for(int k=i; k<=j; k++){
		optCost[i][j]=min(optCost[i][k-1]+optCost[k+1][j], optCost[i][j]);
	}
	optCost[i][j]+=costSum[i][j];
}
//}}}

int main(){
	int n;	cin>>n;
	vector<double> P(n+1),Q(n+1);
	for(int i=1;i<=n;i++)cin>>P[i];
	for(int i=0;i<=n;i++)cin>>Q[i];

	/*
		costSum[i][j]:
			(i: 1,2,~,n)

		(j==i-1) Q[i-1]
		(else) costSum[i][j-1]+ P[j]+Q[j]
	*/

	//costSum の初期化
	for(int i=1; i<=n; i++){
		costSum[i][i-1]=Q[i-1];
		for(int j=i; j<=n; j++){
			costSum[i][j]=costSum[i][j-1] + P[j]+Q[j];
		}
	}

	for(int i=1;i<=n+1;i++)optCost[i][i-1]=Q[i-1];
	for(int d=0; d<=n-1; d++){
		for(int i=1;i<=n-d;i++){
			ComputeOptCost(i,i+d);
		}
	}
	//cout<<optCost[1][1]<<endl;
	//cout<<optCost[1][n]<<endl;
	printf("%.8lf\n",optCost[1][n]);

	return 0;
}
