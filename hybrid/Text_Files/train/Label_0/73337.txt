import sys, math, itertools, collections, bisect
input = lambda : sys.stdin.buffer.readline().rstrip().decode('utf-8')
inf = float('inf')
mod = 10 ** 9 + 7
mans = inf
ans = 0
count = 0
pro = 1
n = int(input())
A = list(map(int, input().split()))
A.sort()
for i in range(n):
	ans -= pow(2, n - i - 1, mod) * A[i]
	ans += pow(2, i, mod) * A[i]
ans %= mod
print(ans)
