public class CodeforcesA {
    ArrayList<Integer>[]adj;
    int[]size;
    double[]starting_time;
    public void dfs1(int node){
        for(int child:adj[node]){
            dfs1(child);
            size[node] += size[child];
        }
        size[node]++;
    }
    public void dfs2(int node,double time){
        starting_time[node] = time;
        long sum = 0;
        for(int child:adj[node])
            sum += size[child];

        for(int child:adj[node]){
            sum -= size[child];
            dfs2(child,time+1.0+sum/2.0);
            sum += size[child];
        }

    }
    public void solve() {
        int n = fs.nInt();
        adj = new ArrayList[n];
        size = new int[n];
        starting_time = new double[n];
        for(int i=0;i<n;i++)adj[i] = new ArrayList<>();

        for(int i=1;i<n;i++){
            int p = fs.nInt() - 1;
            adj[p].add(i);
        }
        dfs1(0);
        dfs2(0,1.0);

        for(int i=0;i<n;i++)
            out.print(starting_time[i]+" ");

    }

    static boolean multipleTestCase = false; static FastScanner fs; static PrintWriter out;
    public void run(){
        fs = new FastScanner();
        out = new PrintWriter(System.out);
        int tc = (multipleTestCase)?fs.nInt():1;
        while (tc-->0)solve();
        out.flush();
        out.close();
    }
    public static void main(String[]args){
        try{
            new CodeforcesA().run();
        }catch (Exception e){
            e.printStackTrace();
        }

    }

                try {
                    st=new StringTokenizer(br.readLine());
