#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[10];
  while (~scanf("%d%d%d%d%d", &a[0], &a[1], &a[2], &a[3], &a[4])) {
    sort(a, a + 5);
    if (a[4] == a[3] && a[3] != a[2]) {
      if (a[0] == a[1] && a[1] == a[2] && (a[1] + a[2] + a[0]) > (a[3] + a[4]))
        printf("%d\n", a[3] + a[4]);
      else
        printf("%d\n", a[1] + a[0] + a[2]);
    } else if (a[4] == a[3] && a[3] == a[2]) {
      printf("%d\n", a[0] + a[1]);
    } else if (a[4] != a[3] && a[3] == a[2] && a[2] != a[1]) {
      printf("%d\n", a[0] + a[1] + a[4]);
    } else if (a[4] != a[3] && a[3] == a[2] && a[2] == a[1]) {
      printf("%d\n", a[0] + a[4]);
    } else if (a[4] != a[3] && a[3] != a[2] && a[2] == a[1] && a[1] != a[0]) {
      printf("%d\n", a[0] + a[3] + a[4]);
    } else if (a[4] != a[3] && a[3] != a[2] && a[2] == a[1] && a[1] == a[0]) {
      printf("%d\n", a[3] + a[4]);
    } else if (a[4] != a[3] && a[3] != a[2] && a[2] != a[1] && a[1] == a[0]) {
      printf("%d\n", a[3] + a[4] + a[2]);
    } else {
      printf("%d\n", a[0] + a[1] + a[2] + a[3] + a[4]);
    }
  }
  return 0;
}
