#include <bits/stdc++.h>
using namespace std;
const long double pi = 3.14159265358979323846;
const long long mod = 1e9 + 7;
const long long N = 5e5;
const long long INF = 1e9 + 1;
long long n, m, k, a[N], ans, res;
int main() {
  std::ios_base::sync_with_stdio(false);
  std::cin.tie(0);
  std::cout.tie(0);
  cin >> n;
  for (long long i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  cin >> m >> k;
  for (long long i = min(m, k); i <= max(m, k) - 1; ++i) {
    res += a[i];
  }
  for (long long i = max(m, k); i <= n; ++i) {
    ans += a[i];
  }
  for (long long i = 1; i <= min(m, k) - 1; ++i) {
    ans += a[i];
  }
  cout << min(ans, res);
  return 0;
}
