#include<bits/stdc++.h>
using namespace std;

int main(){

    while(1){
        int n; cin >> n;
        if(!n) break;
        double pw; cin >> pw;
        vector<double> vw(n);
        vector<double> pf(n);
        vector<double> vf(n);
        vector<double> th(n);
        for(int i = 0; i < n; i++){
            cin >> vw[i] >> pf[i] >> vf[i] >> th[i];
        }

        double ans;
        double l = 0.0, r = 1000.0;

        for(int i = 0; i < 100; i++){

            double p1 = (r + 2*l) / 3;
            double p2 = (2*r + l) / 3;

            //cout << l << " " << r << " " << p1 << " " << p2 << endl;            

            double cost1 = (double)(p1 * pw);
            double cost2 = (double)(p2 * pw);
            
            for(int i = 0; i < n; i++){
                double F1 = (double)((th[i] - p1 * vw[i]) / vf[i]);
                double F2 = (double)((th[i] - p2 * vw[i]) / vf[i]);
                F1 = max(F1, 0.0);
                F2 = max(F2, 0.0);
                
                
                cost1 += (double)(F1 * pf[i]);
                cost2 += (double)(F2 * pf[i]);                                                
            }


            if(cost1 < cost2) r = p2;
            else l = p1;

            ans = cost1;
        }

        printf("%.4f\n", ans);
    }

    return 0;
}
