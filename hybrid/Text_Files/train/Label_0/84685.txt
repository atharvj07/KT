(n, m) = (0, 0)
matrix = list()
for t in range(int(input())):
	(n, m) = [int(num) for num in input().split()]
	matrix = list()
	for i in range(n):
		matrix.append([0 if char == '.' else 1 for char in input()])
	for i in range(n - 1, 0, -1):
		for j in range(1, m - 1):
			if matrix[i][j - 1] != 0 and matrix[i][j] != 0 and (matrix[i][j + 1] != 0) and (matrix[i - 1][j] != 0):
				matrix[i - 1][j] += min(matrix[i][j - 1], matrix[i][j], matrix[i][j + 1])
	count = 0
	for i in range(n):
		for j in range(m):
			count += matrix[i][j]
	print(count)
