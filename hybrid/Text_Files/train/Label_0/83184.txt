from math import gcd

def abs(x):
	if x < 0:
		return -x
	return x

def gcd_list(b, n):
	while n != 1:
		c = gcd(b[-1], b[-2])
		b.pop()
		b.pop()
		b.append(c)
		n -= 1
	return b[0]
for i in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = [abs(a[0] - a[i]) for i in range(1, n)]
	c = gcd_list(b, n - 1)
	if c == 0:
		print(-1)
	else:
		print(c)
