nk = [int(k) for k in input().split()]
(n, k) = (nk[0], nk[1])
heights = [int(k) for k in input().split()]
currTot = sum(heights[:k])
minTot = currTot
minTotInd = 0
(start, end) = (0, k - 1)
while end < n - 1:
	currTot -= heights[start]
	currTot += heights[end + 1]
	start += 1
	end += 1
	if currTot < minTot:
		minTot = currTot
		minTotInd = start
print(minTotInd + 1)
