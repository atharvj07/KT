#include <bits/stdc++.h>
using namespace std;
const long long OO = 10000000000000;
int main() {
  int n, g = 0;
  vector<int> x;
  string s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'B')
      g++;
    else {
      if (g != 0) x.push_back(g);
      g = 0;
    }
  }
  if (g != 0) x.push_back(g);
  cout << x.size() << endl;
  for (int i = 0; i < x.size(); i++) cout << x[i] << " ";
  return 0;
}
