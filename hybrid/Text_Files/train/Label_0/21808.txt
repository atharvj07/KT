const int mod=1e9+7;
const int MAXN=2e5+5;
const int inf=0x3f3f3f3f;
int a[MAXN];
set<int>s;
map<int ,int >mp;
int main()
{
    int t;
    cin>>t;
    while(t--){
        s.clear();
        mp.clear();
        int n,k;
        cin>>n>>k;
        for(int i=1;i<=n;i++){
            cin>>a[i];
            mp[a[i]]=1;
            s.insert(a[i]);
        }
        sort(a+1,a+1+n);
        int pos;
        for(int i=0;;i++)
        {
            if(!mp.count(i))
            {
                pos=i;
                break;
            }
        }
        if(a[n]+1==pos)
        {
            printf("%d\n",s.size()+k);
        }
        else
        {
            int res=(pos+a[n])/2+(pos+a[n])%2;
            if(mp.count(res))
            {
                printf("%d\n",s.size());
            }
            else printf("%d\n",s.size()+(k>0));
        }
    }
}
