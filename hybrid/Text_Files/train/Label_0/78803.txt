(n, k) = map(int, input().split())
l = list(map(int, input().split()))
p = []
for i in range(len(l)):
	if i == 0:
		p.append(l[0])
	else:
		if len(p) < k:
			if l[i] not in p:
				p.append(l[i])
		if len(p) == k:
			if l[i] not in p:
				p.pop(0)
				p.append(l[i])
print(len(p))
for i in p[::-1]:
	print(i, end=' ')
