#include <bits/stdc++.h>
using namespace std;
int n, m, k, q, A[10];
long long f[1000005];
void Work(int c, long long v) {
  for (int i = 999999; i >= c; i--) f[i] = max(f[i], f[i - c] + v);
}
void Add(int s, int v) {
  int x = min(k, 999999 / s);
  for (int i = 1; i <= x; i <<= 1) Work(s * i, (long long)v * i), x -= i;
  Work(s * x, (long long)v * x);
}
int main() {
  scanf("%d", &k);
  for (int i = 0; i <= 5; i++) scanf("%d", &A[i]);
  for (int i = 0; i <= 999999; i++) {
    int p = 0, x = i, s = x % 10;
    while (x) {
      if (s % 3 == 0) f[i] += (long long)A[p] * (s / 3);
      x /= 10;
      s = x % 10;
      p++;
    }
  }
  k = 3 * (k - 1);
  int s = 1;
  for (int i = 0; i <= 5; i++) Add(s * 3, A[i]), s *= 10;
  scanf("%d", &q);
  while (q--) {
    int n;
    scanf("%d", &n);
    printf("%lld\n", f[n]);
  }
}
