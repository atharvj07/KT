def read_int():
	return map(int, input().split())
n = int(input())
a = list(read_int())
b = list(read_int())
pa = [[] for _ in range(n)]
pb = [[] for _ in range(n)]
for (i, ai) in enumerate(a):
	l = pa[ai - 1]
	if l:
		a2 = ai
	l.append(i)
for (i, bi) in enumerate(b):
	l = pb[bi - 1]
	if l:
		b2 = bi
	l.append(i)
p = [None] * n
ab0 = None
for (i, (ai, bi)) in enumerate(zip(pa, pb), 1):
	if len(ai) == 1:
		p[ai[0]] = i
	elif len(bi) == 1:
		p[bi[0]] = i
	elif not (ai or bi):
		ab0 = i
la = pa[a2 - 1]
lb = pb[b2 - 1]
if a2 == b2:
	i = la[0] if la[0] in lb else la[1]
	p[i] = a2
else:
	if not pb[a2 - 1]:
		i = la[1] if p[la[0]] else la[0]
		p[i] = a2
	if not pa[b2 - 1]:
		i = lb[1] if p[lb[0]] else lb[0]
		p[i] = b2
if ab0:
	p[p.index(None)] = ab0
print(*p)
