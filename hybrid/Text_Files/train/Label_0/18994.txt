#include <bits/stdc++.h>
using namespace std;
int ans[1000005];
int main() {
  int p, x;
  scanf("%d%d", &p, &x);
  int jw = 0, first, jw1;
  int ok = 0;
  for (int i = 1; i <= 9; i++) {
    jw = 0;
    jw1 = 0;
    first = i;
    int a = i;
    ans[0] = i;
    for (int j = 1; j <= p + 1; j++) {
      a = a * x;
      ans[j] = a % 10 + jw + jw1;
      jw = a / 10;
      jw1 = ans[j] / 10;
      ans[j] = ans[j] % 10;
      a = ans[j];
    }
    if (ans[p] != first || ans[p - 1] == 0 || ans[p + 1] != ans[1])
      continue;
    else {
      ok = 1;
      break;
    }
  }
  if (ok) {
    for (int i = p - 1; i >= 0; i--) printf("%d", ans[i]);
    putchar('\n');
  } else
    printf("Impossible\n");
  return 0;
}
