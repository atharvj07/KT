#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, x, flag = 0;
  cin >> a >> b >> x;
  if (x != 1) {
    if (a > b) {
      for (int i = 0; i < (x - x % 2); i += 2) {
        cout << 0 << 1;
        flag++;
      }
      if (x % 2 == 0) {
        for (int i = 0; i < (b - flag); i++) {
          cout << 1;
        }
        for (int i = 0; i < (a - flag); i++) {
          cout << 0;
        }
      } else {
        for (int i = 0; i < (a - flag); i++) {
          cout << 0;
        }
        for (int i = 0; i < (b - flag); i++) {
          cout << 1;
        }
      }
    } else {
      for (int i = 0; i < (x - x % 2); i += 2) {
        cout << 1 << 0;
        flag++;
      }
      if (x % 2 == 0) {
        for (int i = 0; i < (a - flag); i++) {
          cout << 0;
        }
        for (int i = 0; i < (b - flag); i++) {
          cout << 1;
        }
      } else {
        for (int i = 0; i < (b - flag); i++) {
          cout << 1;
        }
        for (int i = 0; i < (a - flag); i++) {
          cout << 0;
        }
      }
    }
  } else {
    for (int i = 0; i < (a); i++) {
      cout << 0;
    }
    for (int i = 0; i < (b); i++) {
      cout << 1;
    }
  }
  return 0;
}
