#include <bits/stdc++.h>
using namespace std;
int a[5];
int main() {
  int q;
  cin >> q;
  while (q--) {
    for (int i = 0; i < 3; i++) cin >> a[i];
    sort(a, a + 3);
    if (a[0] == a[2]) {
      cout << 0 << endl;
      continue;
    }
    if (a[2] - a[0] <= 2) {
      cout << 0 << endl;
      continue;
    }
    cout << (a[2] - a[0] - 2) * 2 << endl;
  }
  return 0;
}
