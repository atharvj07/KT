def max_sum(arr):
	n = len(arr)
	dp = [0 for i in range(n)]
	dp[0] = arr[0]
	for i in range(1, n):
		dp[i] = max(dp[i - 1] + arr[i], arr[i])
	return max(dp)
n = int(input())
l = input().split()
li = [int(i) for i in l]
maxa = 0
for mx in range(-30, 31):
	copy = list(li)
	for i in range(n):
		if copy[i] > mx:
			copy[i] = -1000000000000
	maxa = max(max_sum(copy) - mx, maxa)
print(maxa)
