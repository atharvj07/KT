s = input()
dicti = {}
for i in range(ord('a'), ord('z') + 1):
	for j in range(i, ord('z') + 1):
		dicti[chr(i) + chr(j)] = 0
dp2 = [[0 for i in range(26)] for i in range(26)]
dp1 = [0 for i in range(26)]
for i in range(len(s)):
	for j in range(26):
		dp2[j][ord(s[i]) - ord('a')] += dp1[j]
	dp1[ord(s[i]) - ord('a')] += 1
maxi = 0
for i in dp2:
	maxi = max(max(i), maxi)
print(max(max(dp1), maxi))
