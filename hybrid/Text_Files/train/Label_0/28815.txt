#include <bits/stdc++.h>
using namespace std;
const int INF = 2e9 + 7;
int main() {
  int n, m, c, h;
  scanf("%d%d%d", &n, &m, &c);
  int mx1 = -INF, mx2 = -INF, mn1 = INF, mn2 = INF;
  while (c--) {
    int x, y;
    scanf("%d%d", &x, &y);
    int d1 = x + y, d2 = x - y;
    mn1 = min(mn1, d1), mx1 = max(mx1, d1);
    mn2 = min(mn2, d2), mx2 = max(mx2, d2);
  }
  scanf("%d", &h);
  int anw1 = INF, anw2;
  for (int i = 1; i <= h; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    int d1 = x + y, d2 = x - y, d = 0;
    d = max(abs(mn1 - d1), abs(mx1 - d1));
    d = max(d, max(abs(mn2 - d2), abs(mx2 - d2)));
    if (d < anw1) {
      anw1 = d;
      anw2 = i;
    }
  }
  printf("%d\n%d\n", anw1, anw2);
}
