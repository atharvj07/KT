for _ in range(int(input())):
	n = int(input())
	s = input()
	ps = []
	summ = 0
	for i in s:
		summ += int(i) - 1
		ps.append(summ)
	ps = [0] + ps
	map1 = {}
	for i in ps:
		if map1.get(i) == None:
			map1[i] = 1
		else:
			map1[i] += 1
	cnt = 0
	for i in map1.keys():
		if map1[i] >= 2:
			cnt += map1[i] / 2 * (map1[i] - 1)
	print(int(cnt))
