(n, m) = [int(i) for i in input().split(' ')]
alice_s__present = [0] * n
x_d_pos = [0, 0]
x_d_neg = [0, 0]
for _ in range(m):
	(x, d) = [int(i) for i in input().split(' ')]
	if d >= 0.0:
		x_d_pos[0] += x
		x_d_pos[1] += d
	else:
		x_d_neg[0] += x
		x_d_neg[1] += d
for i in range(len(alice_s__present)):
	alice_s__present[i] = x_d_pos[0] + x_d_neg[0] + x_d_pos[1] * i + x_d_neg[1] * abs(i - int(n / 2))
print(sum(alice_s__present) / n)
