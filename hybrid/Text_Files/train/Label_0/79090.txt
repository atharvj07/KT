def helper(start, end, aveL, aveR):
	if aveR - aveL == -1:
		return A
	if start == end:
		return B * (aveR - aveL + 1)
	temp = start + (end - start) // 2
	(l, r) = (aveL, aveR)
	while l <= r:
		mid = l + (r - l) // 2
		if avengers[mid] > temp:
			r = mid - 1
		else:
			l = mid + 1
	return min(B * (aveR - aveL + 1) * (end - start + 1), helper(start, temp, aveL, l - 1) + helper(temp + 1, end, l, aveR))
(n, k, A, B) = map(int, input().split())
avengers = sorted(map(int, input().split()))
print(helper(1, 2 ** n, 0, k - 1))
