#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
const int INF = 0x3f3f3f3f;
char str1[MAXN], str2[MAXN];
bool deleted[MAXN];
int num[MAXN], f[MAXN];
int len1, len2, m;
bool check(int m) {
  for (int i = 0; i < m; ++i) deleted[num[i]] = true;
  int now = 0;
  bool ret = false;
  for (int i = 0; i < len1; ++i)
    if (!deleted[i]) {
      if (str1[i] == str2[now]) ++now;
      if (now == len2) {
        ret = true;
        break;
      }
    }
  for (int i = 0; i < m; ++i) deleted[num[i]] = false;
  return ret;
}
int main() {
  while (~scanf("%s%s", str1, str2)) {
    len1 = strlen(str1);
    len2 = strlen(str2);
    for (int i = 0; i < len1; ++i) scanf("%d", num + i), --num[i];
    int ans = 0, l = 1, r = len1;
    while (l <= r) {
      m = (l + r) >> 1;
      if (check(m)) {
        ans = m;
        l = m + 1;
      } else
        r = m - 1;
    }
    printf("%d\n", ans);
  }
}
