
import java.util.*;


public class B
{
   public static void main(String[] args)
   {
      new B(new Scanner(System.in));
   }

   int N;
   int max;
   int reg;
   ArrayList<Scroll> spells;
   int time;

   ArrayList<Spell> res;

   boolean solve()
   {
      res = new ArrayList<Spell>();

      // Start off with dps being helpful and full health
      long dps = reg;
      long hp = max;

      // Create a pool of spells sorted by power
      PriorityQueue<Scroll> pool = new PriorityQueue<Scroll>(N+6,
            new Comparator<Scroll>(){
               public int compare(Scroll a, Scroll b)
               {
                  return b.pwr-a.pwr;
               }
            });

      // Sort by most dmg
      PriorityQueue<Scroll> q = new PriorityQueue<Scroll>(N+6,
            new Comparator<Scroll>(){
               public int compare(Scroll a, Scroll b)
               {
                  return b.dmg-a.dmg;
               }
            });

      for (Scroll s : spells)
         pool.add(s);

      time = -1;
      while (true)
      {
         time++;

         // First the boss get the dmg from all spells
         // this will also be the healing step for convenience
         hp += dps;
         hp = Math.min(hp, max); // Bound the health
         
         // We will consider death now (rules are weird about this)
         if (hp <= 0) return true;

         // Now determine which spells to cast
         // We can only do one spell per second
         
         // First we pull the spells from the pool if they can be used
         while ((pool.size() > 0)&&(pool.peek().pwr >= hp))
            q.add(pool.poll());

         // Now if we have a spell we will cast it on the beast
         if (q.size() > 0)
         {
            Scroll cur = q.poll();
            res.add(new Spell(time, cur.id));
            dps -= cur.dmg;
         }
         else // special case
         {
            // First special case if we have no spells to cast and
            // the boss has dps >= 0 then we can't kill it
            if (dps >= 0) return false;

            // Ok either we are out of spells and have to wait for death
            // or we have to wait until the creature has no hp, either
            // way we should speed up this process here.... well the bounds
            // are low so lets just see if it works first xD
         }
      }
   }

   public B(Scanner in)
   {
      N = in.nextInt();

      // We modify the max health and regen in order to avoid doubles
      max = in.nextInt()*100;
      reg = in.nextInt()*100;
      
      spells = new ArrayList<Scroll>();
      for (int i=0; i<N; i++)
      {
         int pp = in.nextInt();
         int dd = in.nextInt()*100; // modify dmg too

         // Convert the power that we are using from a percentage to a value
         pp = (pp*max)/100;
         spells.add(new Scroll(i+1, pp, dd));
      }

      boolean canSolve = solve();

      if (!canSolve)
      {
         System.out.println("NO");
         return;
      }

      // We can solve it output more later
      System.out.println("YES");
      System.out.printf("%d %d%n", time, res.size());
      for (Spell s : res)
         System.out.printf("%d %d%n", s.time, s.id);
   }
}


class Spell
{
   int time, id;

   public Spell(int tt, int ii)
   {
      time = tt;
      id = ii;
   }
}


class Scroll
{
   int id;
   int pwr;
   int dmg;

   public Scroll(int ii, int p, int d)
   {
      id = ii;
      pwr = p;
      dmg = d;
   }
}
