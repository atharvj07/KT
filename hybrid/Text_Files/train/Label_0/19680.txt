#include <bits/stdc++.h>
int main() {
  auto optimize_cpp_stdio = [&]() {
    std::ios::sync_with_stdio(false);
    std::cout.tie(nullptr);
    std::cin.tie(nullptr);
    return 0;
  }();
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) std::cin >> a[i];
  auto solve = [&](int64_t k) {
    static std::vector<int> copy_a;
    copy_a = a;
    int64_t result = 0;
    for (int i = 0, j = 0; i < n; i = j) {
      static std::vector<int64_t> pre;
      pre.clear();
      int64_t sum = 0;
      while (j < n && sum < k) {
        a[j] = a[j] % k;
        int take = std::min<int64_t>(a[j], k - sum);
        pre.push_back(take);
        sum = sum + take;
        a[j] = a[j] - take;
        if (a[j] == 0) j++;
      }
      std::partial_sum(pre.begin(), pre.end(), pre.begin());
      int64_t cost = 0, best = LLONG_MAX;
      for (int i = 1; i < pre.size(); i++) cost = cost + pre[i - 1];
      best = std::min(best, cost);
      for (int i = (int)pre.size() - 2; i >= 0; i--) {
        cost = cost - pre[i];
        cost = cost + (pre.back() - pre[i]);
        best = std::min(best, cost);
      }
      result = result + best;
    }
    a.swap(copy_a);
    return result;
  };
  int64_t ans = LLONG_MAX;
  int64_t sum = std::accumulate(a.begin(), a.end(), 0ll);
  for (int i = 2; 1ll * i * i <= sum; i++) {
    bool flag = false;
    while (sum % i == 0) {
      sum = sum / i;
      flag = true;
    }
    if (flag) ans = std::min(ans, solve(i));
  }
  if (sum != 1) ans = std::min(ans, solve(sum));
  if (ans == LLONG_MAX) ans = -1;
  std::cout << ans << "\n";
  return 0;
}
