import sys
import math
(n, k) = map(int, input().split())
stra = str(input())
freqa = [0 for i in range(26)]
for i in range(len(stra)):
	freqa[ord(stra[i]) - ord('A')] += 1
freqa = sorted(freqa, reverse=True)
(storc, suma) = (0, 0)
for i in range(len(freqa)):
	if storc < k:
		card = min(freqa[i], k - storc)
		suma += card * card
		storc += card
	else:
		break
print(suma)
