#include <cstdio>
#include <cstring>
#include <cmath>
#include <queue>
#include <map>
#include <algorithm>

using namespace std;

#define MAXN 15
#define MAXM 1000010
#define INF 0x3f3f3f3f
#define PI acos(-1.0)
typedef long long LL;

int n,m,cnt[MAXN];
bool Map[MAXN][MAXN];
double f[2][2000][MAXN][2];

void Init()
{
    int i,u,v;
    scanf("%d%d",&n,&m);
    for(i=1;i<=m;++i){
        scanf("%d%d",&u,&v); --u; --v;
        Map[u][v]=Map[v][u]=true;
        ++cnt[u]; ++cnt[v];
    }
}

void DP()
{
    int i,u,v,k,t,c;
    double ans=0,pre=0;
    f[0][1][0][1]=1;
    for(k=t=1;;k^=1,++t){
        memset(f[k],0,sizeof(f[k]));
        for(i=1;i<(1<<n)-1;++i)
            for(u=0;u<n;++u){
                if(!((1<<u)&i)) continue;
                for(v=0;v<n;++v){
                    if(!Map[u][v]) continue;
                    f[k][i|(1<<v)][v][0]+=f[k^1][i][u][0]/cnt[u]+f[k^1][i][u][1]/cnt[u];
                    f[k][i|(1<<v)][v][1]+=f[k^1][i][u][1]/cnt[u];
                }
            }
        for(u=ans=0;u<n;++u){
            f[k][(1<<n)-1][u][0]+=f[k^1][(1<<n)-1][u][0];
            f[k][(1<<n)-1][u][1]+=f[k^1][(1<<n)-1][u][1];
        }
        for(u=ans=0;u<n;++u) ans+=f[k][(1<<n)-1][u][0];
        if(t>n&&fabs(ans-pre)<1e-7){
            if((++c)>80) break;
        }
        else c=0;
        pre=ans;
    }
    printf("%.6f\n",ans);
}

int main()
{
    Init();
    DP();
    return 0;
}
