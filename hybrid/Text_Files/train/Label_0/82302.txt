s = input()
s1 = ''
start = False
metka = False
i = 0
while i < len(s) and (not start):
	if s[i] != 'a' and (not start):
		s1 += chr(ord(s[i]) - 1)
		metka = True
	elif s[i] == 'a' and (not metka):
		s1 += s[i]
	elif s[i] == 'a' and metka:
		s1 += s[i:]
		start = True
	i += 1
if not metka:
	s1 = s[:-1] + 'z'
print(s1)
