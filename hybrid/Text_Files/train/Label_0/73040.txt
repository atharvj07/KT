(n, k) = map(int, input().split())
thewalls = list(map(int, input().split()))
wall_index = {}
for (i, wall) in enumerate(thewalls):
	tmp = wall_index.get(wall, [])
	tmp.append(i)
	wall_index[wall] = tmp
if max(map(len, wall_index.values())) <= k and n >= k:
	print('YES')
	(res, color) = (['1'] * n, 1)
	for wall in wall_index:
		for idx in wall_index[wall]:
			res[idx] = str(color)
			color = (color + 1) % k if (color + 1) % k else k
	print(' '.join(res))
else:
	print('NO')
