def inlt():
	return list(map(int, input().split()))
import sys
from collections import defaultdict
from math import sqrt, ceil, floor, factorial, log
from bisect import bisect_left, bisect_right
from io import BytesIO, IOBase, StringIO
input = lambda : sys.stdin.readline().rstrip()
t = int(input())
while t:
	t -= 1
	n = int(input())
	l1 = inlt()
	l2 = inlt()
	for j in range(len(l1)):
		l1[j] = [l1[j], j]
	for j in range(len(l2)):
		l2[j] = [l2[j], j]
	l1.sort()
	l2.sort()
	dic1 = defaultdict(int)
	dic2 = defaultdict(int)
	for j in range(len(l1)):
		dic1[l1[j][1]] = j
	for j in range(len(l2)):
		dic2[l2[j][1]] = j
	ans = defaultdict(bool)
	start = [l1[-1]]
	ans[l1[-1][1]] = True
	l1[-1] = []
	while len(start) > 0:
		new = []
		for j in start:
			ans[j[1]] = True
			first = dic1[j[1]] + 1
			second = dic2[j[1]] + 1
			while first < n and len(l1[first]) > 0:
				new.append(l1[first])
				l1[first] = []
				first += 1
			while second < n and len(l2[second]) > 0:
				new.append(l2[second])
				l2[second] = []
				second += 1
		start = new
	st = ''
	for j in range(0, n):
		if ans[j]:
			st = st + '1'
		else:
			st = st + '0'
	print(st)
