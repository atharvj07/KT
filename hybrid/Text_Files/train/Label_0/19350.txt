#include <iostream>
#include <vector>
using namespace std;
const int SIZE = 10000;
int main (int argc, char *argv[]) {
	// make sieve
	vector<bool> sieve(SIZE,true);
	sieve[0] = false; sieve[1] = false;
	for(int i = 2; i < SIZE; ++i) {
		if (sieve[i]) {
			for(int j = i+i; j < SIZE; j+=i) {
				sieve[j] = false;
			}
		}
	}
	// count p
	int N;
	while (cin >> N) {
		vector<int> inc(N), dec(N);
		for (int i=0;i<N;i++) {
			inc[i] = i+1;
			dec[i] = N-i;
		}
		int count = 0;
		for(int i = 0; i < N; ++i) {
			if (sieve[inc[i]] && sieve[dec[i]]) count++;
		}
		cout << count << endl;
	}
	return 0;
}
