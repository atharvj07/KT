import sys
t = int(sys.stdin.readline())
for _ in range(t):
	n = int(sys.stdin.readline())
	d_ls = list(map(int, sys.stdin.readline().split()))
	ans = 'YES'
	check = {}
	for i in d_ls:
		try:
			check[i] += 1
		except:
			check[i] = 1
	for k in check.values():
		if k != 2:
			ans = 'NO'
	if ans != 'NO':
		d_ls = list(set(d_ls))
		d_ls.sort()
		dif = []
		x = 2 * n - 2
		for i in range(len(d_ls) - 1, 0, -1):
			if (d_ls[i] - d_ls[i - 1]) % x != 0:
				ans = 'NO'
				break
			else:
				dif.append((d_ls[i] - d_ls[i - 1]) // x)
				x -= 2
		if d_ls[-1] % (2 * n) != 0 or d_ls[-1] // (2 * n) <= sum(dif):
			ans = 'NO'
	print(ans)
