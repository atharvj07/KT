(a, b, c, s, t) = map(int, input().split())
if a == 12:
	a = 0
if s == 12:
	s = 0
if t == 12:
	t = 0
a += b / 120 + c / 120
b += c / 60
b /= 5
c /= 5
f = [a, b, c]
f.sort()
if f[0] < s and s < f[1]:
	x = 1
elif f[1] < s and s < f[2]:
	x = 2
else:
	x = 3
if f[0] < t and t < f[1]:
	y = 1
elif f[1] < t and t < f[2]:
	y = 2
else:
	y = 3
if x == y:
	print('YES')
else:
	print('NO')
