import sys
input = lambda : sys.stdin.readline().rstrip('\r\n')
inp_vars_int = lambda : map(int, input().split())
inp_list_int = lambda : list(map(int, input().split()))
t = int(input())
for _ in range(t):
	(a, b) = inp_vars_int()
	diff = abs(a - b)
	tot = a + b
	eg = tot % 2
	min_br = diff // 2
	max_br = 2 * min(a, b) + diff // 2 + eg
	opts = range(min_br, max_br + 1, 2 - eg)
	print(len(opts))
	print(' '.join(map(str, opts)))
