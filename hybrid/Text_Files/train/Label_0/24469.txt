#include <bits/stdc++.h>
using namespace std;
long long dp[2][1005][1005];
int n, m, x;
const long long mod = 1e9 + 7;
int main() {
  scanf("%d%d%d", &n, &m, &x);
  if (n > m) {
    puts("0");
    return 0;
  }
  dp[0][0][0] = 1;
  int now = 0;
  for (int i = (1); i < (m + 1); i++) {
    now ^= 1;
    for (int l = (0); l < (n + 1); l++)
      for (int r = (0); r < (n + 1); r++) dp[now][l][r] = 0;
    for (int l = (i == x); l < (n + 1); l++)
      for (int r = (0); r < (l + 1); r++) {
        if (i != x) dp[now][l][r] = dp[now ^ 1][l][r];
        if (r) {
          if (i != x)
            dp[now][l][r] = (dp[now][l][r] + dp[now ^ 1][l][r - 1]) % mod;
          dp[now][l][r] = (dp[now][l][r] + dp[now ^ 1][l - 1][r - 1]) % mod;
          if (r < l)
            dp[now][l][r] = (dp[now][l][r] + dp[now ^ 1][l - 1][r]) % mod;
        } else if (l)
          dp[now][l][r] = (dp[now][l][r] + dp[now ^ 1][l - 1][r]) % mod;
      }
  }
  long long ans = dp[m % 2][n][n];
  for (int i = (1); i < (n + 1); i++) ans = ans * i % mod;
  printf("%lld\n", ans);
  return 0;
}
