import java.io.BufferedReader;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class D {

	static int N;
	static int[][] prob;
	static double [][] memo;
	static double[][] newP;

	static double solve2(int p, int node, int l, int r) {
		if(l == r) {
			return newP[p][node] = 1;
		}

		if(newP[p][node] > -1)
			return newP[p][node];

		int mid = (l + r) / 2;
		double ans = 0;
		if(p <= mid) 
			for (int i = mid + 1; i <= r; i++) 
				ans +=  solve2(p, node * 2, l, mid) * solve2(i, 2 * node + 1, mid + 1, r) * prob[p][i] / 100.0;
		else
			for (int i = l; i <= mid; i++) 
				ans += solve2(p, node * 2 + 1, mid + 1, r) * solve2(i, 2 * node, l, mid) * prob[p][i] / 100.0;
		return newP[p][node] = ans;

}

static double solve(int p, int node, int l, int r, int points) {
	if(l == r) 
		return 0;
	if(memo[p][node] > -1)
		return memo[p][node];

	int mid = (l + r) / 2;

	if(p <= mid) {
		double ans = 0;
		double left = solve(p, 2 * node, l, mid, points / 2);
		for (int i = mid + 1; i <= r; i++) {
			ans = Math.max(ans, left + solve(i, 2 * node + 1, mid + 1, r, points / 2) + points * newP[p][node] / 100.0);
		}
		return memo[p][node] = ans;
	}

	double ans = 0;
	double right = solve(p, 2 * node + 1, mid + 1, r, points / 2);
	for (int i = l; i <= mid; i++) {
		ans = Math.max(ans, right + solve(i, 2 * node, l, mid, points / 2) + points * newP[p][node] / 100.0);
	}
	return memo[p][node] = ans;

}

public static void main(String[] args) throws IOException {
	Scanner sc = new Scanner(System.in);
	PrintWriter out = new PrintWriter(System.out);

	N = sc.nextInt();

	prob = new int[1 << N][1 << N];
	for (int i = 0; i < prob.length; i++) {
		for (int j = 0; j < prob.length; j++) {
			prob[i][j] = sc.nextInt();
		}
	}

	memo = new double[1 << N][2 * (1 << N)];

	for (int i = 0; i < memo.length; i++) {
		Arrays.fill(memo[i], -2);
	}
	
	newP = new double[1 << N][2 * (1 << N)];

	for (int i = 0; i < memo.length; i++) {
		Arrays.fill(newP[i], -2);
	}

	for (int i = 0; i < (1 << N); i++) {
		solve2(i, 1, 0, (1 << N) - 1);
	}

	
	double ans = 0;
	for (int i = 0; i < (1 << N); i++) {
		ans = Math.max(ans, solve(i, 1, 0, (1 << N) - 1, 1 << (N - 1)));
	}

	out.println(ans * 100);

	out.flush();
	out.close();
}



static class Scanner 



{
	StringTokenizer st;
	BufferedReader br;

	public Scanner(InputStream s){	br = new BufferedReader(new InputStreamReader(s));}
	public Scanner(String file) throws FileNotFoundException{	br = new BufferedReader(new FileReader(file));}

	public String next() throws IOException 
	{
		while (st == null || !st.hasMoreTokens()) 
			st = new StringTokenizer(br.readLine());
		return st.nextToken();
	}

	public int nextInt() throws IOException {return Integer.parseInt(next());}

	public long nextLong() throws IOException {return Long.parseLong(next());}

	public String nextLine() throws IOException {return br.readLine();}


	public double nextDouble() throws IOException
	{
		String x = next();
		StringBuilder sb = new StringBuilder("0");
		double res = 0, f = 1;
		boolean dec = false, neg = false;
		int start = 0;
		if(x.charAt(0) == '-')
		{
			neg = true;
			start++;
		}
		for(int i = start; i < x.length(); i++)
			if(x.charAt(i) == '.')
			{
				res = Long.parseLong(sb.toString());
				sb = new StringBuilder("0");
				dec = true;
			}
			else
			{
				sb.append(x.charAt(i));
				if(dec)
					f *= 10;
			}
		res += Long.parseLong(sb.toString()) / f;
		return res * (neg?-1:1);
	}

	public boolean ready() throws IOException {return br.ready();}



}
