from sys import stdout
n = int(input())
gears = list(map(int, input().split()))
targetGears = []
for i in range(0, n):
	targetGears.append(i)

def check():
	for i in range(0, n):
		if gears == targetGears:
			stdout.write('Yes')
			return
		for j in range(0, n):
			gears[j] += 1 if j % 2 == 0 else -1
			gears[j] %= n
	stdout.write('No')
check()
