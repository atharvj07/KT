goodLet = input()
template = input()
isExist = template.count('*')
starIndex = template.find('*')
t1 = ''.join([template[j] if j < starIndex else '' for j in range(len(template))])
t2 = ''.join([template[j] if j > starIndex else '' for j in range(len(template))])
n = int(input())
for i in range(n):
	s = input()
	isChecked = True
	if not isExist:
		if len(s) != len(template):
			print('NO')
			continue
		for j in range(len(s)):
			if template[j] == '?':
				if s[j] not in goodLet:
					isChecked = False
					break
			elif template[j] != s[j]:
				isChecked = False
				break
	else:
		if len(s) < len(t1) + len(t2):
			print('NO')
			continue
		s1 = s[0:len(t1)]
		s2 = s[-len(t2):] if len(t2) > 0 else ''
		sbad = s[len(t1):-len(t2) if len(t2) > 0 else len(s)]
		for j in range(len(t1)):
			if t1[j] == '?':
				if s1[j] not in goodLet:
					isChecked = False
					break
			elif s1[j] != t1[j]:
				isChecked = False
				break
		for j in range(len(t2)):
			if t2[j] == '?':
				if s2[j] not in goodLet:
					isChecked = False
					break
			elif s2[j] != t2[j]:
				isChecked = False
				break
		for j in sbad:
			if j in goodLet:
				isChecked = False
				break
	if isChecked:
		print('YES')
	else:
		print('NO')
