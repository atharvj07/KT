class Node:

	def __init__(self, value='.', adj_list=None):
		if adj_list == None:
			self.adj_list = []
		else:
			self.adj_list = adj_list
		self.value = value
		self.scratch = 0

	def add_node(self, node):
		self.adj_list.append(node)

	def get_node(self, index):
		return self.adj_list[index]

	def dfs(self, k):
		visited = set()
		stack = []
		stack.append(self)
		self.scratch = 0
		found = False
		while len(stack) > 0:
			curr_node = stack.pop()
			curr_scratch = curr_node.scratch
			if curr_scratch == 0:
				if curr_node.value == '#':
					continue
				elif curr_node.value == 'T':
					stack.clear()
					found = True
					print('YES')
			if curr_node not in visited:
				visited.add(curr_node)
			new_scratch = (curr_scratch + 1) % k
			for node in curr_node.adj_list:
				node.scratch = new_scratch
				stack.append(node)
		if not found:
			print('NO')
