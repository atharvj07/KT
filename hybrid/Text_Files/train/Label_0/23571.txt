#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e4 + 7, MOD = 1e9 + 7;
int n;
int dp[MAXN], ans;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie();
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    dp[i] = (dp[i] + 2) % MOD;
    for (int j = i - 1; j >= x; j--) {
      dp[i] = (dp[i] + dp[j]) % MOD;
    }
    ans = (ans + dp[i]) % MOD;
  }
  cout << ans;
  return 0;
}
