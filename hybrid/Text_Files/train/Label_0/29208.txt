#include <bits/stdc++.h>
using namespace std;

typedef pair<int,int> P;
typedef long long ll;

#define repl(i,a,b) for(int i=(int)(a);i<(int)(b);(i)++)
#define rep(i,n) repl(i,0,n)
#define dbg(x) cout<<#x<<"="<<x<<endl

#define INF INT_MAX/3

int H,W,K;
char S[1001][1001];
int d[1001][1001];

int main(){
  scanf("%d%d%d",&H,&W,&K);
  rep(i,H){
    scanf("%s",S[i]);
  }

  rep(i,H)rep(j,W){
    if(i==0&&j==0){
      d[i][j]=-1;
      continue;
    }

    int a=-INF,b=-INF;
    if(i>0){
      if(S[i-1][j]=='.'&&d[i-1][j]<=0)a=K; 
      else a=d[i-1][j]-1;
    }
    if(j>0){
      if(S[i][j-1]=='.'&&d[i][j-1]<=0)b=K; 
      else b=d[i][j-1]-1;
    }

    d[i][j]=max(a,b);
  }

  if(d[H-1][W-1]<=0)cout<<"Cacao"<<endl;
  else cout<<"Chieno"<<endl;

  return 0;
}
