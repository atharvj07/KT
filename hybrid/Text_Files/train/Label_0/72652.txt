t = int(input())
for i in range(t):
	(n, m) = map(int, input().strip().split(' '))
	l = []
	l1 = []
	for j in range(n):
		lst = list(map(int, input().strip().split(' ')))
		l.append(lst)
		l1 = l1 + lst
	l1.sort()
	m1 = 0
	t1 = 0
	for j in range(n):
		l[j].sort()
		if m1 == m or l[j][0] > l1[m - 1]:
			for j1 in range(m):
				print(l[j][j1], end=' ')
			print()
		else:
			t1 = m1
			k1 = []
			for j1 in range(m):
				if m1 < m and l[j][j1] <= l1[m - 1 - m1]:
					ind = l1.index(l[j][j1])
					del l1[ind]
					m1 += 1
				else:
					k1.append(l[j][j1])
			for j1 in range(t1):
				print(k1[j1], end=' ')
			for j1 in range(t1, m1):
				print(l[j][j1 - t1], end=' ')
			for j1 in range(m1, m):
				print(l[j][j1], end=' ')
			print()
