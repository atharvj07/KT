from math import gcd
a = []
for _ in range(int(input())):
	n = int(input())
	lcm = 1
	for (i, x) in enumerate(map(int, input().split()), 2):
		if lcm != -1:
			lcm = lcm * i // gcd(lcm, i)
			if lcm > int(1000000000.0):
				lcm = -1
		if lcm != -1 and x % lcm == 0:
			a.append('NO')
			break
	else:
		a.append('YES')
print(*a, sep='\n')
