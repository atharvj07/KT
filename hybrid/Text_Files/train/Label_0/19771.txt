#include <bits/stdc++.h>
using namespace std;
const int N = 60;
int n;
long long a[N], sum[N];
long long dp[N];
long long f(int pos) {
  if (pos >= n) return 0;
  if (dp[pos] != -1) return dp[pos];
  long long ans1 =
      a[pos] + min(a[pos + 1] + sum[pos + 2] - f(pos + 2), f(pos + 2));
  long long ans2 = f(pos + 1);
  return dp[pos] = max(ans1, ans2);
}
int main(void) {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  sum[n - 1] = a[n - 1];
  for (int i = n - 2; i >= 0; i--) sum[i] = sum[i + 1] + a[i];
  memset(dp, -1, sizeof(dp));
  long long a = f(0);
  long long b = sum[0] - a;
  cout << b << " " << a << endl;
  return 0;
}
