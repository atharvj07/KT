import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Scanner;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        T1082A solver = new T1082A();
        solver.solve(1, in, out);
        out.close();
    }

    static class T1082A {
        public void solve(int testNumber, Scanner in, PrintWriter out) {
            int q = in.nextInt();
            while (q > 0) {
                q--;
                int n = in.nextInt(), x = in.nextInt(), y = in.nextInt(), d = in.nextInt();
                int dt = Math.abs(x - y), ans = -1;
                if (dt % d == 0) ans = dt / d;
                else {
                    int goLeft = -1, goRight = -1;
                    if ((y - 1) % d == 0) {
                        goLeft = (int) Math.ceil(1.0 * x / d) + (y - 1) / d;
                    }
                    if ((n - y) % d == 0) {
                        goRight = (int) Math.ceil(1.0 * (n - x) / d) + (n - y) / d;
                    }
                    if (goLeft != -1) ans = goLeft;
                    if (goRight != -1 && goLeft != -1) ans = Math.min(ans, goRight);
                    else if (goRight != -1) ans = goRight;

                }
                out.println(ans);
            }
        }

    }
}
