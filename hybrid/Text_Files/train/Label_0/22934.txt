#include <iostream>
#include <set>
using namespace std;

int main(){
	string s;
	while(getline(cin,s)){
		string l = "#";
		for(int i = 0 ; i < s.size() ; i++)
			if( (s[i] >= 'a' && s[i] <= 'z') || (s[i] >= 'A' && s[i] <= 'Z') ){
				l += s[i] & ~32;
			}
		l += "$";
		set<string> p;
		set<string> ban;
		for(int i = 1 ; i < l.size()-1 ; i++){
			string g = string(1,l[i]);
			int le = i-1 , ri = i+1;
			while(l[le] == l[ri]){
				ban.insert(g);
				g = string(1,l[le]) + g + string(1,l[ri]);
				le--,ri++;
			}
			p.insert(g);
		}
		for(int i = 1 ; i < l.size()-1 ; i++){
			string g = "";
			int le = i , ri = i+1;
			while(l[le] == l[ri]){
				ban.insert(g);
				g = string(1,l[le]) + g + string(1,l[ri]);
				le--,ri++;
			}
			p.insert(g);
		}
		int f = 1;
		for(set<string>::iterator it = p.begin(); it != p.end() ; ++it){
			if((*it).size() <= 2 || ban.count(*it)) continue;
			if(!f) cout << " ";
			cout << *it;
			f = 0;
		}
		cout << endl;
	}
}
