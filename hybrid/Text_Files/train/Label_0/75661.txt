import sys
import math

def fn(n, k, a):
	h = []
	a.sort(reverse=True)
	for i in range(k):
		h.append(a[i])
	h1 = h.count(min(h))
	h2 = a.count(min(h))
	p = 1
	for i in range(h1):
		p = p * (h2 - i)
	for i in range(h1):
		p = p // (h1 - i)
	return p % 1000000007
input = sys.stdin.read()
data = list(map(int, input.split()))
t = data[0]
l = 0
for i in range(t):
	n = data[l + 1]
	k = data[l + 2]
	a = data[l + 3:l + 3 + n]
	l = l + n + 2
	print(fn(n, k, a))
