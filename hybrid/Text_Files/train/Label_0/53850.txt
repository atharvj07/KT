import java.util.*;
import java.io.*;

public class Main {
 	
    public static void main(String args[]) throws IOException {

        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();
        StringTokenizer tk;
        
        char[] a = in.readLine().toCharArray();
        char[] b = in.readLine().toCharArray();
        
        int i = 0, lenA = a.length;
        while(a[i] == '0' && i < (lenA - 1))
            ++i;
        
        int d = 0, lenB = b.length;
        while(b[d] == '0' && d < (lenB - 1))
            ++d;
        
        if((lenA - i) > (lenB - d))
            System.out.println(">");
        else if((lenA - i) < (lenB - d))
            System.out.println("<");
        else {
            
            for (int j = 0; j < (lenA - i); j++) {
                if(a[j + i] > b[j + d]) {
                    System.out.println(">");
                    return;
                }else if(a[j + i] < b[j + d]) {
                    System.out.println("<");
                    return;
                }
            }
            System.out.println("=");
        }
    }
}
