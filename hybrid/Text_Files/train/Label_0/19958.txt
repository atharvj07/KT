#include <bits/stdc++.h>
using namespace std;
signed main() {
  long long T;
  cin >> T;
  for (long long t = 1; t <= T; t++) {
    long long n, m;
    cin >> n >> m;
    pair<long long, long long> a[n + 1];
    long long sum = 0;
    for (long long i = 1; i <= n; i++) {
      cin >> a[i].first;
      sum += a[i].first;
      a[i].second = i;
    }
    if (m < n || n == 2) {
      cout << -1 << '\n';
      continue;
    }
    sort(a + 1, a + n + 1);
    long long ans = (a[1].first + a[2].first) * (m - n) + sum * 2;
    cout << ans << '\n';
    for (long long i = 1; i < n; i++) {
      cout << i << ' ' << i + 1 << '\n';
    }
    cout << n << ' ' << 1 << '\n';
    for (long long i = 1; i <= m - n; i++) {
      cout << a[1].second << ' ' << a[2].second << '\n';
    }
  }
  return 0;
}
