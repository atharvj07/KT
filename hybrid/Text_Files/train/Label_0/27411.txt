#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, cnt = 1;
  cin >> t;
  int arr[t + 1];
  vector<int> v;
  for (int i = 0; i < t; i++) cin >> arr[i];
  sort(arr, arr + t);
  for (int i = 0; i < t; i++) {
    if (arr[i] == arr[i + 1])
      cnt++;
    else {
      v.push_back(cnt);
      cnt = 1;
    }
  }
  v.push_back(cnt);
  sort(v.begin(), v.end());
  reverse(v.begin(), v.end());
  cout << v[0] << endl;
}
