n = int(input())
m = list(map(int, input().split()))
N = -1
K = -1
i = 0
k = 0
o = 0
I = 0
while N == -1 and i < n:
	if m[i] == 1:
		N = i
	i += 1
i = n - 1
while K == -1 and i > -1:
	if m[i] == 1:
		K = i
	i -= 1
i = 0
k = K - N + 1
for i in range(N, K - 1):
	if m[i] == 0 and m[i + 1] == 0:
		if i != N and m[i - 1] == 1:
			o += 1
		o += 1
if N == -1:
	k = 0
elif o > 1:
	k = k - o
print(k)
