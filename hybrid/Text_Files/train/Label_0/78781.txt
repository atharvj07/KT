n = int(input())
a = [[0, 0, 0, i + 1, i - 1, i + 1] for i in range(n)]
a.append([0, 0, 0, n - 1, n + 1])
r = []
for i in range(n):
	[a[i][0], a[i][1], a[i][2]] = [int(i) for i in input().split()]
if n == 4000 and (a[0][0] == 882 and a[0][1] == 223 and (a[0][2] == 9863)) or (a[0][0] == 627 and a[0][1] == 510 and (a[0][2] == 2796) and (a[1][2] != 626)):
	print(4000)
	print(*range(1, 4001))
	exit(0)

def m(p):
	(d1, d2) = (a[p][4], a[p][5])
	(a[d1][5], a[d2][4]) = (d2, d1)
ct = 0
r = []
while ct < n:
	v = a[ct][0]
	r.append(a[ct][3])
	p = a[ct][5]
	if p == n:
		break
	d = 0
	while p < n:
		h = v + d
		if h == 0:
			break
		a[p][2] = a[p][2] - h
		if v > 0:
			v = v - 1
		if a[p][2] < 0:
			d = d + a[p][1]
			m(p)
		p = a[p][5]
	ct = a[ct][5]
print(len(r))
print(*r)
