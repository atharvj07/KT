#include <bits/stdc++.h>
using namespace std;
template <typename T, typename U>
inline void chkmax(T& x, U y) {
  if (x < y) x = y;
}
template <typename T, typename U>
inline void chkmin(T& x, U y) {
  if (y < x) x = y;
}
const int MAXN = 1111111;
char t[MAXN];
int s[MAXN];
int a[2][21][MAXN];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; i++) {
    scanf("%s", t);
    for (int j = 0; j < m; j++) {
      if (t[j] == '1') {
        s[j] |= (1 << (i));
      }
    }
  }
  int now = 0;
  for (int i = 0; i < m; i++) {
    a[0][0][s[i]]++;
  }
  for (int j = 0; j < n; j++) {
    now ^= 1;
    for (int i = 0; i < (1 << (n)); i++) {
      for (int k = n; k > 0; k--) {
        a[now][k][i] = a[!now][k][i] + a[!now][k - 1][i ^ (1 << (j))];
      }
      a[now][0][i] = a[!now][0][i];
    }
  }
  int ans = 1e9;
  for (int j = 0; j < (1 << (n)); j++) {
    int sum = 0;
    for (int k = 0; k <= n; k++) {
      sum += min(k, n - k) * a[now][k][j];
    }
    chkmin(ans, sum);
  }
  cout << ans << endl;
  return 0;
}
