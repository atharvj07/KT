#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, sum = 0;
  cin >> n >> m;
  char map[n][m];
  int row[n], col[m];
  memset(row, 0, sizeof(row));
  memset(col, 0, sizeof(col));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> map[i][j];
      if (map[i][j] == '*') {
        sum++;
        row[i]++;
        col[j]++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if ((map[i][j] == '*' && row[i] + col[j] - 1 == sum) ||
          (map[i][j] == '.' && row[i] + col[j] == sum)) {
        cout << "YES" << endl;
        cout << i + 1 << " " << j + 1;
        return 0;
      }
    }
  }
  cout << "NO";
  return 0;
}
