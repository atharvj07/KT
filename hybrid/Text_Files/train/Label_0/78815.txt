(d1, d2, d3) = map(int, input().split())
if d1 + d2 + d3 <= 2 * (d1 + d2) and d1 + d2 + d2 <= 2 * (d1 + d3) and (d1 + d2 + d3 <= 2 * (d2 + d3)):
	print(d1 + d2 + d3)
elif 2 * (d1 + d2) <= 2 * (d1 + d3) and 2 * (d1 + d2) <= 2 * (d2 + d3):
	print(2 * (d1 + d2))
elif 2 * (d1 + d3) <= 2 * (d2 + d3):
	print(2 * (d1 + d3))
else:
	print(2 * (d2 + d3))
