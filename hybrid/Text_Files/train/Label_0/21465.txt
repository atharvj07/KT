#include <bits/stdc++.h>
using namespace std;
typedef struct Node {
  int value;
  struct Node* parent;
} Node;
Node* MakeSet(int value) {
  Node* node = new Node;
  node->value = value;
  node->parent = NULL;
  return node;
}
Node* Find(Node* node) {
  while (node->parent != NULL) {
    node = node->parent;
  }
  return node;
}
void Union(Node* node1, Node* node2) {
  node1 = Find(node1);
  node2 = Find(node2);
  if (node1->value == node2->value) return;
  node2->parent = node1;
}
int main() {
  int n;
  cin >> n;
  int p[n];
  for (int i = 1; i <= n; i++) cin >> p[i];
  Node* H[n + 1];
  char a[n + 1][n + 1];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) cin >> a[i][j];
    H[i] = MakeSet(i);
  }
  for (int i = 1; i < n; i++)
    for (int j = i + 1; j <= n; j++)
      if (a[i][j] == '1') Union(H[i], H[j]);
  for (int i = 1; i < n; i++)
    for (int j = i + 1; j <= n; j++) {
      Node* tmp = Find(H[i]);
      Node* tmp2 = Find(H[j]);
      int tmp1 = tmp->value;
      int tmp3 = tmp2->value;
      if (p[i] > p[j] && tmp1 == tmp3) swap(p[i], p[j]);
    }
  for (int i = 1; i <= n; i++) cout << p[i] << " ";
}
