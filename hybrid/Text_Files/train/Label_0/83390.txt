def okay(jl):
	x = 0
	for (l, d) in jl:
		if x == 0 and d != 'S' or (x == 20000 and d != 'N'):
			return False
		if d == 'S':
			x += l
		if d == 'N':
			x -= l
		if x < 0 or x > 20000:
			return False
	return x == 0
n = int(input())
jl = []
for i in range(n):
	(l, d) = input().split()
	jl.append((int(l), d[0]))
print('YES' if okay(jl) else 'NO')
