public class ABBYY_3_D {
    Scanner scanner = new Scanner(System.in);
    BufferedReader buffered = new BufferedReader(new InputStreamReader(System.in));
    StreamTokenizer in = new StreamTokenizer(buffered);

    PrintWriter out = new PrintWriter(System.out);

    int nextInt() throws Exception {
        in.nextToken();
        return (int) in.nval;
    }

    Set<String> set;
    int[] a = new int[100];
    int[] p = new int[100];

    void ghghgh(int n) {
        String s = "";
        for (int i=0; i<n; i++) {
            s += p[i];
        }
        set.add(s);
        for (int i=0; i<n; i++) {
            if (a[i] > 0) {
                for (int j=i+1; j<n; j++) {
                    if (a[j] > 0) {
                        int t = p[i];
                        p[i] = p[j];
                        p[j] = t;
                        a[i]--;
                        a[j]--;
                        ghghgh(n);
                        a[i]++;
                        a[j]++;
                        t = p[i];
                        p[i] = p[j];
                        p[j] = t;
                    }
                }
            }
        }
    }

    int answer(int n, int k) {
        set = new HashSet<String>();
        for (int i=0; i<n; i++) {
            a[i] = (i<k) ? 2 : 1;
            p[i] = i;
        }
        ghghgh(n);
        return set.size();
    }

    long MOD = 1000000007;
    int MAX = 1000010;

    long delen(long a, long b) {
        long n = MOD - 2;
        long c = 1;
        long s = b;
        while (n > 0) {
            if ((n&1) > 0) {
                c *= s;
                c %= MOD;
            }
            s *= s;
            s %= MOD;
            n >>= 1;
        }

        return (a*c) % MOD;
    }

    void main() throws Exception {
        int n = nextInt();
        int k = 0;
        for (int i=0; i<n; i++) {
            if (nextInt() == 2) {
                k++;
            }
        }

        long[] a = new long[MAX];
        a[0] = 1;
        a[1] = 2;
        long[] b = new long[MAX];
        b[0] = 1;
        b[1] = 2;
        for (int i=2; i<MAX; i++) {
            a[i] = (a[i-1] + i * a[i-2]) % MOD;
            if (b[i-1] % i == 0) {
                b[i] = ((i+1) * (b[i-1] / i + b[i-2])) % MOD;
            } else {
                b[i] = ((i+1) * (delen(b[i-1],i) + b[i-2])) % MOD;
            }
        }

        if (k > n/2) {
            long ans = 1;
            for (int i=1; i<=n; i++) {
                ans = (ans*i) % MOD;
            }
            for (int i=0; i<n-k; i++) {
                ans = delen((ans*a[i])%MOD, b[i]);
            }
            System.out.println(ans);
        } else {
            long ans = a[n-1];
            for (int i=0; i<k; i++) {
                ans = delen((ans*b[n-i-1])%MOD, a[n-i-1]);
            }
            System.out.println(ans);
        }

        System.out.flush();
    }

    public static void main(String[] args) throws Exception {
        new ABBYY_3_D().main();
    }
}
