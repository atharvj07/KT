inf = 100000000000000000

def linput():
	return list(map(int, input().split()))
for _ in range(int(input())):
	n = int(input())
	l1 = linput()
	l2 = linput()
	suff1 = [l1[-1]]
	for i in range(n - 2, -1, -1):
		suff1.append(l1[i] + suff1[-1])
	suff1 = suff1[::-1]
	pre2 = [l2[0]]
	for i in range(1, n):
		pre2.append(l2[i] + pre2[-1])
	mini = inf
	for i in range(n):
		val1 = suff1[i] - l1[i]
		val2 = pre2[i] - l2[i]
		mini = min(mini, max(val1, val2))
	print(mini)
