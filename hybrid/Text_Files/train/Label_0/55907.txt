import java.util.Scanner;

public class nugo{
	public static void main(String[] args){
		Scanner scan = new Scanner(System.in);
		int n;
		n = scan.nextInt();
		int[] arr = new int[n];
		int i = 0;
		int count_0 = 0;
		int count_5 = 0;
		String num_5 ="";
		String num_0 ="";
		String num;
		while(i < n){
			arr[i] =scan.nextInt();
			if(arr[i] == 5){
				count_5++;
				if(count_5 == 9){
					num_5 += "555555555";
					count_5 = 0;
				}
			}			
			else{
				count_0++;
				num_0 += "0";
			}	
			i++;
		}	
		if(count_0 == 0){
			System.out.println(-1);
		}		
		else if(num_5 == ""){
			System.out.println(0);
		}
		else{
			num = num_5 + num_0;
			System.out.print(num);
		}
		
	}	
}
