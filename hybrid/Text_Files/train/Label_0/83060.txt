import os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
cases = int(input())
for t in range(cases):
	n = int(input())
	newd = {i + 1: [] for i in range(n)}
	for i in range(n - 1):
		(x, y) = list(map(int, input().split()))
		newd[x].append(y)
		newd[y].append(x)
	visited = {i: True for i in newd}
	children = {i: 0 for i in newd}
	parent = {i: 0 for i in newd}
	start = 1
	for i in newd:
		if len(newd[i]) == 1:
			start = i
			break
	stack = [start]
	while stack:
		node = stack.pop()
		if visited[node]:
			visited[node] = False
			stack.append(node)
			c = 0
			for i in newd[node]:
				if visited[i]:
					stack.append(i)
					parent[i] = node
		elif parent[node]:
			children[parent[node]] += children[node] + 1
	if n % 2 == 1:
		print(start, newd[start][0])
		print(newd[start][0], start)
	else:
		f = 0
		for i in newd:
			if 2 * (children[i] + 1) == n:
				print(start, newd[start][0])
				print(start, i)
				f = 1
				break
		if f == 0:
			print(start, newd[start][0])
			print(newd[start][0], start)
