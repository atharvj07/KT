def generate(l, n):
	if n <= 0:
		return
	if n == 1:
		d.append((l, 1, l))
		return
	elif n % 2 == 1:
		d.append((l, n, l + (n - 1) // 2))
		generate(l, (n - 1) // 2)
		generate(l + (n - 1) // 2 + 1, (n - 1) // 2)
	else:
		d.append((l, n, l + (n - 1) // 2))
		generate(l, (n - 1) // 2)
		generate(l + n // 2, (n - 1) // 2 + 1)
t = int(input())
cnt = 0
while cnt < t:
	cnt += 1
	n = int(input())
	a = [0] * n
	d = []
	generate(0, n)
	d = sorted(d, key=lambda l: (l[1], -l[0]), reverse=True)
	i = 1
	for elem in d:
		a[elem[2]] = i
		i += 1
	print(' '.join((str(i) for i in a)))
