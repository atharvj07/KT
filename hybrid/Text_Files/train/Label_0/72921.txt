import sys
reader = (line.rstrip() for line in sys.stdin)
input = reader.__next__

def getInts():
	return [int(s) for s in input().split()]

class SegmentTree:

	def __init__(self, size):
		n = 1
		while n < size:
			n <<= 1
			self.n = n
			self.tree = [0] * (n * 2)

	def update(self, start, value):
		start += self.n
		self.tree[start] = value
		flag = True
		while start > 1:
			start_pair = start + 1 if start % 2 == 0 else start - 1
			if flag:
				self.tree[start // 2] = self.tree[start] | self.tree[start_pair]
			else:
				self.tree[start // 2] = self.tree[start] ^ self.tree[start_pair]
			flag = not flag
			start //= 2

def solve():
	(N, M) = getInts()
	A = getInts()
	ttt = SegmentTree(1 << N)
	for (i, val) in enumerate(A):
		ttt.update(i, A[i])
	for m in range(M):
		(P, B) = getInts()
		ttt.update(P - 1, B)
		print(ttt.tree[1])
	return
solve()
