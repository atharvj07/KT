import sys

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())

def solve():
	n = mint()
	nbak = n
	a = []
	i = 2
	while i * i <= n:
		if n % i == 0:
			c = 0
			while n % i == 0:
				n //= i
				c += 1
			a.append((c, i))
		i += 1
	if n != 1:
		a.append((1, n))
	best = max(a)
	z = best[0]
	print(z)
	if z != 1:
		print(' '.join([str(best[1])] * (z - 1)), end=' ')
	print(nbak // best[1] ** (z - 1))
for i in range(mint()):
	solve()
