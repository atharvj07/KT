class NewYearPermutation:

	def __init__(self, num, m, s):
		self.n = num
		self.m = m
		self.s = s
		self.fa = [i for i in range(num)]

	def findest(self, x):
		if x == self.fa[x]:
			return x
		self.fa[x] = self.findest(self.fa[x])
		return self.fa[x]

	def solve(self):
		for i in range(self.n):
			for j in range(self.n):
				if self.s[i][j] == '1':
					u = self.findest(i)
					v = self.findest(j)
					if u != v:
						self.fa[u] = v
		temp = [[] for row in range(self.n)]
		for i in range(self.n):
			u = self.findest(i)
			temp[u].append(i)
		for i in range(self.n):
			for j in range(len(temp[i])):
				for k in range(j + 1, len(temp[i])):
					if self.m[temp[i][k]] < self.m[temp[i][j]]:
						(self.m[temp[i][k]], self.m[temp[i][j]]) = (self.m[temp[i][j]], self.m[temp[i][k]])
		self.printAns()

	def printAns(self):
		for i in range(len(self.m)):
			print(self.m[i], end=' ' if i < len(self.m) - 1 else '')
n = int(input())
p = list(map(int, input().split()))
s = [str()] * n
for i in range(n):
	s[i] = input()
NYP = NewYearPermutation(n, p, s)
NYP.solve()
