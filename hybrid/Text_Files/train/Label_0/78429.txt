import sys
input = lambda : sys.stdin.readline().strip()

def all_p_square(arr):
	ans = True
	for i in arr:
		if int(i ** 0.5) != i ** 0.5:
			ans = False
			break
	return ans
for i in range(int(input())):
	l = int(input())
	arr = list(map(int, input().split()))
	print('YES' if all_p_square(arr) == False else 'NO')
