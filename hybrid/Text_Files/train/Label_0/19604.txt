#include <iostream>
#include <cstdio>
#include <vector>
using namespace std;
#define REP(i,n)for(int i=0;i<(int)n;++i)
int main(){int n;while(cin>>n,n){int a[n+1];a[0]=0;REP(i,n)cin>>a[i+1];double b,r,v,e,f,tt[10001];cin>>b>>r>>v>>e>>f;tt[0]=0;REP(x,10001)tt[x+1]=tt[x]+1/(v-(x>=r?e*(x-r):f*(r-x)));vector<double>dp(n+1,1<<29);dp[0]=0;REP(i,n)for(int j=i+1;j<n+1;++j)dp[j]=min(dp[j],dp[i]+tt[a[j]-a[i]]+(i?b:0));printf("%.12f\n",dp[n]);}}
