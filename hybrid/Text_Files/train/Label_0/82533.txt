for i in range(int(input())):
	n = int(input())
	s = input()
	(l1, lr1, l2, lr2) = (-1, -1, -1, -1)
	mid = n // 2
	seg = False
	if n % 2 == 0:
		for j in range(n):
			if s[j] == '0':
				if j + 1 > mid:
					print(1, j + 1, 1, j)
				else:
					print(j + 1, n, j + 2, n)
				seg = True
				break
	else:
		for j in range(n):
			if s[j] == '0':
				if j + 1 > mid:
					print(1, j + 1, 1, j)
				else:
					print(j + 1, n, j + 2, n)
				seg = True
				break
	if not seg:
		print(1, n - 1, 2, n)
