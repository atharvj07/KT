n = int(input())
for i in range(n):
	x = input()
	mas = []
	for j in range(10):
		mas.append(0)
	for j in range(len(x)):
		mas[int(x[j])] += 1
	if mas[0] > 0 and (mas[0] > 1 or mas[2] > 0 or mas[4] > 0 or (mas[6] > 0) or (mas[8] > 0)):
		suma = 0
		for j in range(10):
			suma += mas[j] * j
		if suma % 3 == 0:
			print('red')
		else:
			print('cyan')
	else:
		print('cyan')
