import sys
from collections import defaultdict as dd
import heapq
import math
try:
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')
except:
	pass
input = sys.stdin.readline
for i in range(int(input())):
	(h, c, t) = map(int, input().split())
	l = t - (h + c) / 2
	if t == h:
		print(1)
	elif 2 * t <= h + c:
		print(2)
	else:
		k = int(((h - c) / (2 * l) + 1) / 2)
		q = (h - c) / (2 * (2 * k + 1)) - l
		w = (h - c) / (2 * (2 * k - 1)) - l
		if q < 0:
			q *= -1
		if w < 0:
			w *= -1
		if q < w:
			print(2 * k + 1)
		else:
			print(2 * k - 1)
