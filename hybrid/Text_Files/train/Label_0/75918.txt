def main():

	def solve():
		n = int(input())
		pp = []
		for _ in range(n):
			pp.append(list(input()))
		avail = [str(a) for a in range(0, 10)]
		for p in pp:
			if p[0] in avail:
				avail.remove(p[0])
		count = 0
		for i in range(1, n):
			if pp[i] in pp[:i]:
				pp[i][0] = avail.pop(0)
				count += 1
		print(count)
		for p in pp:
			print(''.join(p))
	q = int(input())
	for _ in range(q):
		solve()
main()
