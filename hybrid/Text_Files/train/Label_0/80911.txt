t = int(input())
import math
an = []
ep = 10 ** (-16)
for _ in range(t):
	(x, y) = list(map(int, input().split()))
	if x > 0 and y > 0:
		an.append(180 * math.atan(abs(y) / (abs(x) + ep)) / math.pi)
	elif x < 0 and y > 0:
		an.append(180 - 180 * math.atan(abs(y) / (abs(x) + ep)) / math.pi)
	elif x < 0 and y < 0:
		an.append(180 + 180 * math.atan(abs(y) / (abs(x) + ep)) / math.pi)
	elif x > 0 and y < 0:
		an.append(360 - 180 * math.atan(abs(y) / (abs(x) + ep)) / math.pi)
	elif x == 0 and y > 0:
		an.append(90.0)
	elif x == 0 and y < 0:
		an.append(270.0)
	elif x > 0 and y == 0:
		an.append(0.0)
	elif x < 0 and y == 0:
		an.append(180.0)
	else:
		print('yah na shit')
an.sort()
an = [-360.0 + an[-1]] + an
dif = []
for i in range(t):
	dif.append(an[i + 1] - an[i])
print(360.0 - max(dif))
