#include<iostream>
#include<list>
#include<vector>
#include<algorithm>
#include<string>
#include<stdio.h>
#include<math.h>
#include<unordered_map>

using namespace std;
using vec = vector<int>;
using mat = vector<vec>;
using lst = list<int>;


bool solve()
{
	int N, M;
	cin >> N >> M;

	if (N == 0)return false;

	vec MaxV(M);

	int ans = 0;

	for (int i = 0; i < N; ++i)
	{
		int d, v;
		cin >> d >> v;
		--d;

		ans -= MaxV[d];
		MaxV[d] = max(MaxV[d], v);
		ans += MaxV[d];
	}

	cout << ans << endl;

	return true;
}


int main()
{
	while(solve());

	return 0;
}
