#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:100000000")
using namespace std;
const int N = 65;
double p[N][N], d[N][N], m[N][N];
int main() {
  int g;
  scanf("%d", &g);
  int n = 1 << g;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; scanf("%lf", &p[i][j]), p[i][j] /= 100, j++)
      ;
  for (int i = 0; i < n; d[0][i] = 1, i++)
    ;
  for (int t = 1; t <= g; t++)
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++)
        if ((j >> t) == (i >> t) && (j >> t - 1) != (i >> t - 1)) {
          d[t][i] += d[t - 1][j] * p[i][j];
          m[t][i] = max(m[t][i], m[t - 1][j]);
        }
      d[t][i] *= d[t - 1][i];
      m[t][i] += m[t - 1][i] + d[t][i] * (1 << t - 1);
    }
  printf("%.10lf\n", *max_element(m[g], m[g] + n));
  return 0;
}
