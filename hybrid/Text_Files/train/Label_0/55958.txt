import java.math.*;
import java.util.Scanner;
public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger t,w,b,g,k,p,minn,mo,gc;
        t = in.nextBigInteger();
        w = in.nextBigInteger();
        b = in.nextBigInteger();
        g = b.gcd(w);
        g = b.multiply(w).divide(g);
        k = t.divide(g);
        if(w.compareTo(b)==-1) minn = w;
        else minn = b;
        p = k.multiply(minn);
        
        mo = t.mod(g);
        if(mo.compareTo(BigInteger.ZERO)>0) {
            p = p.add(minn.min(mo.add(BigInteger.ONE))) ;
            p = p.subtract(BigInteger.ONE);
        }
        gc = p.gcd(t);
        t = t.divide(gc);
        p = p.divide(gc);
        System.out.println(p+"/"+t);
    }

}
