t = int(input())
al = []
for i in range(t):
	(a, b) = map(int, input().split())
	g = ''
	k = ''
	last = ''
	for j in range(b):
		if j % 2 == 1:
			g += 'W'
			k += 'B'
		else:
			g += 'B'
			k += 'W'
	for j in range(a):
		if j < a - 1 or (a % 2 == 1 and b % 2 == 1):
			if j % 2 == 0:
				al.append(g)
			else:
				al.append(k)
		else:
			if j % 2 == 1:
				last = ''
				for k in range(b):
					if k == 0:
						last += 'B'
					elif k % 2 == 0:
						last += 'W'
					else:
						last += 'B'
			else:
				last = ''
				for k in range(b):
					if k == b - 1:
						last += 'B'
					elif k % 2 == 0:
						last += 'B'
					else:
						last += 'W'
			al.append(last)
for i in al:
	print(i)
