import math
for t in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	ans = a[0]
	for i in range(1, n):
		ans = ans ^ a[i]
	if ans == 0:
		print('YES')
	else:
		ok = 0
		count = 0
		for i in range(n):
			ok = ok ^ a[i]
			if ok == ans:
				count += 1
				ok = 0
		if count >= 2:
			print('YES')
		else:
			print('NO')
