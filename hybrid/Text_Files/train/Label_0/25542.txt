#include <bits/stdc++.h>
using namespace std;
long long a[100010][3], dp[100010][3], n;
int main() {
  int i, j, k;
  scanf("%I64d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%I64d", &a[i][1]);
  }
  for (i = 1; i <= n; i++) {
    scanf("%I64d", &a[i][2]);
  }
  for (i = 1; i <= n; i++) {
    dp[i][1] = max(dp[i - 1][1], dp[i - 1][2] + a[i][1]);
    dp[i][2] = max(dp[i - 1][2], dp[i - 1][1] + a[i][2]);
  }
  printf("%I64d", max(dp[n][1], dp[n][2]));
  return 0;
}
