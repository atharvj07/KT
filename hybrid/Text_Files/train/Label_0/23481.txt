#include <bits/stdc++.h>
using namespace std;
int read() {
  int x = 0;
  char ch = getchar();
  while (ch < '0' || ch > '9') ch = getchar();
  while (ch >= '0' && ch <= '9')
    x = (x << 3) + (x << 1) + ch - '0', ch = getchar();
  return x;
}
const int N = 300050;
int n, m, val[N], sum[N << 2];
long long lv[N << 2], rv[N << 2], lazy[N << 2];
int l0[N << 2], l1[N << 2], r0[N << 2], r1[N << 2], ans[N << 2];
void calc(int u, long long v) {
  lazy[u] += v;
  lv[u] += v;
  rv[u] += v;
}
void pushup(int u) {
  lv[u] = lv[u << 1];
  rv[u] = rv[u << 1 | 1];
  ans[u] = max(ans[u << 1], ans[u << 1 | 1]);
  l0[u] = l0[u << 1];
  r0[u] = r0[u << 1 | 1];
  l1[u] = l1[u << 1];
  r1[u] = r1[u << 1 | 1];
  if (rv[u << 1] == lv[u << 1 | 1]) return;
  ans[u] = max(ans[u], r0[u << 1] + l0[u << 1 | 1]);
  if (rv[u << 1] > lv[u << 1 | 1]) {
    ans[u] = max(ans[u], r1[u << 1] + l0[u << 1 | 1]);
    if (l0[u << 1] == sum[u << 1])
      l0[u] = max(l0[u], l0[u << 1] + l0[u << 1 | 1]);
    if (l1[u << 1] == sum[u << 1])
      l1[u] = max(l1[u], l1[u << 1] + l0[u << 1 | 1]);
    if (l0[u << 1 | 1] == sum[u << 1 | 1])
      r1[u] = max(r1[u], r1[u << 1] + sum[u << 1 | 1]);
  } else {
    ans[u] = max(ans[u], r0[u << 1] + l1[u << 1 | 1]);
    if (r0[u << 1 | 1] == sum[u << 1 | 1])
      r0[u] = max(r0[u], r0[u << 1 | 1] + r0[u << 1]);
    if (r1[u << 1 | 1] == sum[u << 1 | 1])
      r1[u] = max(r1[u], r1[u << 1 | 1] + r0[u << 1]);
    if (r0[u << 1] == sum[u << 1])
      l1[u] = max(l1[u], sum[u << 1] + l1[u << 1 | 1]);
  }
}
void pushdown(int u) {
  calc(u << 1, lazy[u]);
  calc(u << 1 | 1, lazy[u]);
  lazy[u] = 0;
}
void build(int u, int l, int r) {
  sum[u] = r - l + 1;
  lv[u] = val[l];
  rv[u] = val[r];
  if (l == r) {
    l0[u] = l1[u] = r0[u] = r1[u] = ans[u] = 1;
    return;
  }
  int mid = (l + r) >> 1;
  build(u << 1, l, mid);
  build(u << 1 | 1, mid + 1, r);
  pushup(u);
  return;
}
void modify(int u, int l, int r, int L, int R, int V) {
  if (L <= l && r <= R) {
    calc(u, V);
    return;
  }
  int mid = (l + r) >> 1;
  pushdown(u);
  if (L <= mid) modify(u << 1, l, mid, L, R, V);
  if (R > mid) modify(u << 1 | 1, mid + 1, r, L, R, V);
  pushup(u);
}
int main() {
  n = read();
  for (int i = 1; i <= n; ++i) val[i] = read();
  build(1, 1, n);
  m = read();
  while (m--) {
    int l = read(), r = read(), v = read();
    modify(1, 1, n, l, r, v);
    printf("%d\n", ans[1]);
  }
  return 0;
}
