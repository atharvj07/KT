#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 7;
long long n, l, v1, v2, ans[N];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  cin >> n >> l >> v1 >> v2;
  vector<long long> a(2 * n);
  for (long long i = 0; i < n; i++) cin >> a[i], a[i + n] = a[i] + 2 * l;
  if (!a[0]) a.emplace_back(4 * l), a.erase(a.begin());
  for (auto &e : a) e *= v1 + v2;
  long long s = 0, t = v2 * l, cnt = 0;
  long long i = 0, j = 0;
  while (a[j] <= t) j++, cnt++;
  while (i < n) {
    if (a[i] - s > a[j] - t) {
      ans[cnt] += a[j] - t;
      cnt++;
      s += a[j] - t, t = a[j++];
    } else if (a[i] - s < a[j] - t) {
      ans[cnt] += a[i] - s;
      cnt--;
      t += a[i] - s;
      s = a[i++];
    } else {
      ans[cnt] += a[i] - s;
      s = a[i++], t = a[j++];
    }
  }
  while (2 * (v1 + v2) * l - s >= a[j] - t) {
    ans[cnt] += a[j] - t;
    cnt++;
    s += a[j] - t, t = a[j++];
  }
  ans[cnt] += 2 * l * (v1 + v2) - s;
  for (i = 0; i <= n; i++)
    cout << fixed << setprecision(12) << 0.5 * ans[i] / (v1 + v2) / l << endl;
  return 0;
}
