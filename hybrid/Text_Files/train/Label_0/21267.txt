#include <bits/stdc++.h>
#define REP(i,n) for(int i=0; i<(int)(n); ++i)

using namespace std;

int main(){
    int n;
    while(cin >> n && n > 0) {
        int ans = 0;

        vector<int> p(n - 1);
        vector<int> c(n - 1);
        REP(i, n - 1) cin >> p[i];
        REP(i, n - 1) cin >> c[i];
        REP(i, n - 1) ans += c[i];

        vector<int> deg(n);
        REP(i, n - 1) {
            int a = i + 1, b = p[i] - 1;
            deg[a] ++;
            deg[b] ++;
        }

        vector<vector<pair<int, int>>> G(n);
        REP(i, n - 1) {
            int a = i + 1, b = p[i] - 1;
            if(deg[a] < 2) continue;
            if(deg[b] < 2) continue;
            G[a].push_back(make_pair(b, c[i]));
            G[b].push_back(make_pair(a, c[i]));
            ans += 2 * c[i];
        }

        int longest = 0;
        REP(s, n) {
            const int INF = 1 << 29;
            vector<int> dist(n, INF);
            dist[s] = 0;
            queue<int> que;
            que.push(s);
            while(!que.empty()){
                int u = que.front(); que.pop();
                for(auto pr : G[u]) {
                    int v = pr.first;
                    int nd = dist[u] + pr.second;
                    if(dist[v] > nd) {
                        dist[v] = nd;
                        que.push(v);
                    }
                }
            }
            REP(g, n) if(dist[g] < INF) longest = max(longest, dist[g]);
        }

        cout << ans - longest << endl;

    }
    return 0;
}
