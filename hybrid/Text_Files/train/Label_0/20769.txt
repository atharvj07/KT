#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-9;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n = 14;
  vector<long long> v;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    v.push_back(x);
  }
  for (int i = 0; i < n; i++) {
    vector<long long> a = v;
    if (v[i] > 0) {
      a[i] = 0;
      for (int j = 1; j <= n; j++) {
        a[(i + j) % 14] += v[i] / 14 + ((v[i] % 14 >= j) ? 1 : 0);
      }
      long long sum = 0;
      for (int j = 0; j < n; j++) {
        if (a[j] % 2 == 0) {
          sum += a[j];
        }
      }
      ans = max(sum, ans);
    }
  }
  cout << ans;
}
