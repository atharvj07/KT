for _ in range(int(input())):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	s = [-1 * i for i in arr]
	subarray_start = 0
	subarray_end = 0
	st = -1
	e = -1
	subarray_sum = 0
	max_len = -1
	for i in s:
		subarray_sum += i
		subarray_end += 1
		while subarray_sum > k:
			subarray_sum -= s[subarray_start]
			subarray_start += 1
		if subarray_end - subarray_start > max_len:
			max_len = subarray_end - subarray_start
			st = subarray_start
			e = subarray_end
	if st == 1 and e == 1:
		print(-1)
	else:
		print(st + 1, e)
