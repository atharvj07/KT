from collections import defaultdict, deque

def bfs(s, d):
	d[s] = 0
	que = deque([s])
	while que:
		v = que.popleft()
		for nv in edges[v]:
			if d[nv] == inf:
				que.append(nv)
				d[nv] = d[v] + 1

def solver():
	price.sort()
	pf = [0] * (m + 1)
	for i in range(1, m + 1):
		pf[i] = pf[i - 1] + price[i]
	(da, db, dc) = ([inf] * (n + 1), [inf] * (n + 1), [inf] * (n + 1))
	bfs(a, da)
	bfs(b, db)
	bfs(c, dc)
	ans = inf
	for i in range(1, n + 1):
		if da[i] + db[i] + dc[i] > m:
			continue
		dx = db[i]
		posans = pf[dx] + pf[da[i] + dc[i] + dx]
		ans = min(ans, posans)
	return ans
inf = float('inf')
t = int(input())
for _ in range(t):
	(n, m, a, b, c) = map(int, input().split())
	price = [0]
	price.extend(map(int, input().split()))
	edges = defaultdict(list)
	for _ in range(m):
		(u, v) = map(int, input().split())
		edges[u].append(v)
		edges[v].append(u)
	ans = solver()
	print(ans)
