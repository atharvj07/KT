#include <bits/stdc++.h>
using namespace std;
long long n, a[10], b[10], prime, souoc[10], ans = 1e9, parent[10];
long long solve() {
  long long res = n - prime;
  long long root = 0;
  for (long long i = 0; i < n; i++) {
    if (parent[i] == -1) {
      res += souoc[i];
      root++;
    }
  }
  if (root > 1) res++;
  return res;
}
void built_tree(long long pos) {
  if (pos == n) {
    ans = min(ans, solve());
    return;
  }
  built_tree(pos + 1);
  for (long long i = 0; i < pos; i++) {
    if (b[i] % a[pos] == 0) {
      parent[pos] = i;
      b[i] /= a[pos];
      built_tree(pos + 1);
      b[i] *= a[pos];
      parent[pos] = -1;
    }
  }
}
signed main() {
  cin >> n;
  for (long long i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n, greater<long long>());
  memset(parent, -1, sizeof parent);
  for (long long i = 0; i < n; i++) {
    b[i] = a[i];
    for (long long j = 2; j * j <= a[i]; j++) {
      while (a[i] % j == 0) {
        a[i] /= j;
        souoc[i]++;
      }
    }
    if (a[i] > 1) souoc[i]++;
    if (souoc[i] == 1) prime++;
    a[i] = b[i];
  }
  built_tree(1);
  cout << ans;
}
