var test = readline().split(' ')[0];
var ara = [];
var obj = {};
while(test>0){
	var x = readline();
	var p = eval(x);
	p = Math.round(p*10000000000)/10000000000;
	if(obj.hasOwnProperty(p+"")){
		obj[p]+=1;
	}
	else{
		obj[p+""]= 1;
	}
	ara.push(p);
	test--;
}
var ans = "";
for(var i = 0 ; i<ara.length;i++){
	ans+= obj[ara[i]] +" ";
}
print(ans);
