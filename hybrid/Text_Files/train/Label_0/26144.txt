#include <bits/stdc++.h>
using namespace std;
int N, K;
int main(int argc, char **argv) {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> N >> K;
  string s;
  cin >> s;
  int his[256];
  memset(his, 0, sizeof(his));
  for (int i = 0; i < N; i++) {
    his[s[i]]++;
  }
  bool cando = true;
  for (int i = 0; i < 256; i++) {
    if (his[i] > K) {
      cando = false;
      break;
    }
  }
  if (cando) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
