list_ans = []
for _ in range(int(input())):
	n = int(input())
	list1 = list(map(int, input().split()))
	list1.sort()
	i1 = 0
	a = 10 ** 9
	for i in range(n - 1):
		if a > list1[i + 1] - list1[i]:
			a = list1[i + 1] - list1[i]
			i1 = i
		if a == 0:
			break
	list_ans.append(' '.join(map(str, [list1[i1]] + list(list1[i1 + 2:]) + list(list1[:i1]) + [list1[i1 + 1]])))
for item in list_ans:
	print(item)
