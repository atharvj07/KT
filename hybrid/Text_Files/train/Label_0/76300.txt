t = int(input())
_scores = [0 for i in range(0, t)]
for i in range(0, t):
	n_input = int(input())
	a_input = input().split(' ')[:n_input]
	a = [int(ai) for ai in a_input]
	_accumulator = [0 for _ in range(0, n_input)]
	for chip in reversed(range(0, n_input)):
		if chip + a[chip] < n_input:
			_accumulator[chip] = a[chip] + _accumulator[chip + a[chip]]
		else:
			_accumulator[chip] = a[chip]
	_scores[i] = max(_accumulator)
for score in _scores:
	print(score)
