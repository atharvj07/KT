(n, k) = map(int, input().split())
l = [input() for i in range(n)]
z = []
d = {}
for i in l:
	t = len(i)
	if t in d:
		d[t] += 1
	else:
		d[t] = 1
		z.append(t)
(b, w) = (0, 0)
su = 0
x = len(input())
z.sort()
for i in z:
	if i == x:
		break
	su += d[i]
b = su // k * 5 + su + 1
w = (su + d[x] - 1) // k * 5 + su + d[x]
print(b, w)
