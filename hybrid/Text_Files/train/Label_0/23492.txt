#include <bits/stdc++.h>
using namespace std;
template <class T, class U>
void ckmin(T &a, U b) {
  if (a > b) a = b;
}
template <class T, class U>
void ckmax(T &a, U b) {
  if (a < b) a = b;
}
int N, Q;
int arr[100013];
array<int, 60> seg[3 * 100013];
array<int, 60> base[7];
int ans;
array<int, 60> comb(array<int, 60> l, array<int, 60> r) {
  array<int, 60> res;
  for (auto i = (0); i < (60); i++) {
    res[i] = l[i] + r[(i + l[i]) % 60];
  }
  return res;
}
void build(int w, int L, int R) {
  if (L == R) {
    seg[w] = base[arr[L]];
    return;
  }
  int mid = (L + R) >> 1;
  build(w << 1, L, mid);
  build(w << 1 | 1, mid + 1, R);
  seg[w] = comb(seg[w << 1], seg[w << 1 | 1]);
}
void update(int w, int L, int R, int a, int v) {
  if (L == R) {
    seg[w] = base[v];
    return;
  }
  int mid = (L + R) >> 1;
  if (a <= mid)
    update(w << 1, L, mid, a, v);
  else
    update(w << 1 | 1, mid + 1, R, a, v);
  seg[w] = comb(seg[w << 1], seg[w << 1 | 1]);
}
array<int, 60> query(int w, int L, int R, int a, int b) {
  if (a <= L && R <= b) {
    return seg[w];
  }
  int mid = (L + R) >> 1;
  if (b <= mid) return query(w << 1, L, mid, a, b);
  if (mid < a) return query(w << 1 | 1, mid + 1, R, a, b);
  return comb(query(w << 1, L, mid, a, b), query(w << 1 | 1, mid + 1, R, a, b));
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> N;
  for (auto i = (0); i < (N); i++) cin >> arr[i];
  for (auto i = (2); i < (7); i++) {
    for (auto j = (0); j < (60); j++) {
      base[i][j] = (j % i ? 1 : 2);
    }
  }
  build(1, 0, N - 1);
  cin >> Q;
  while (Q--) {
    char qid;
    int l, r;
    cin >> qid >> l >> r;
    if (qid == 'C') {
      l--;
      arr[l] = r;
      update(1, 0, N - 1, l, r);
    }
    if (qid == 'A') {
      l--;
      r--;
      array<int, 60> res = query(1, 0, N - 1, l, r - 1);
      cout << res[0] << '\n';
    }
  }
  return 0;
}
