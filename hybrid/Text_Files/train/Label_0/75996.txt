import math

def bitadd(a, b):
	c = ['0', '0', '0']
	for i in range(3):
		if a[i] == '1' or b[i] == '1':
			c[i] = '1'
	return ''.join(c)

def comp(a, b):
	return val(a) & val(b)

def bitsub(a, b):
	c = val(a) - val(b)
	s = ''
	while c >= 1:
		s = s + str(c % 2)
		c = int(c / 2)
	s = '0' * (3 - len(s)) + s[::-1]
	return s

def val(a):
	return (ord(a[0]) - ord('0')) * 4 + (ord(a[1]) - ord('0')) * 2 + (ord(a[2]) - ord('0'))
n = int(input())
bitmasks = {'A': '001', 'B': '010', 'C': '100'}
dp = []
dp.append({'000': math.inf, '001': math.inf, '010': math.inf, '100': math.inf, '011': math.inf, '101': math.inf, '110': math.inf, '111': math.inf})
for _ in range(n):
	dp.append({'000': math.inf, '001': math.inf, '010': math.inf, '100': math.inf, '011': math.inf, '101': math.inf, '110': math.inf, '111': math.inf})
for i in range(0, n + 1):
	dp[i]['000'] = 0
prices = [0] * n
weight = ['000'] * n
for i in range(n):
	inp = input().split()
	prices[i] = int(inp[0])
	for c in inp[1]:
		weight[i] = bitadd(bitmasks[c], weight[i])
extra = {'011': ['001', '010'], '101': ['001', '100'], '110': ['100', '010'], '111': ['001', '010', '100', '110', '011', '101']}
for i in range(1, n + 1):
	for w in ['001', '010', '100', '011', '101', '110', '111']:
		if val(weight[i - 1]) <= val(w) and comp(weight[i - 1], w):
			dp[i][w] = min(prices[i - 1] + dp[i - 1][bitsub(w, weight[i - 1])], dp[i - 1][w])
		else:
			dp[i][w] = dp[i - 1][w]
		if w in extra.keys():
			for t in extra[w]:
				dp[i][t] = min(dp[i][t], dp[i][w])
if dp[n]['111'] != math.inf:
	print(dp[n]['111'])
else:
	print(-1)
