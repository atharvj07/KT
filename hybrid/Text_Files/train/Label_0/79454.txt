def main():
	T = int(input())
	for _ in range(T):
		(R, B, D) = map(int, input().split())
		ok = solve(R, B, D)
		if ok:
			print('yes')
		else:
			print('no')

def solve(R, B, D):
	groupCnt = min(R, B)
	delta = abs(R - B)
	if not D:
		return not delta
	neededGroups = (delta + D - 1) // D
	return neededGroups <= groupCnt
main()
