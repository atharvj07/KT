#include <bits/stdc++.h>
using namespace std;
bool v[205];
vector<int> f;
struct node {
  int x1, y1, x2, y2;
  bool work(int x3, int y3, int x4, int y4) {
    if (x1 == x2) {
      return x3 <= x1 && x1 <= x4 && y1 == y3 && y2 == y4;
    } else {
      return y3 <= y1 && y1 <= y4 && x1 == x3 && x2 == x4;
    }
  }
} a[205];
void dfs(int x1, int y1, int x2, int y2, int n) {
  int i;
  for (i = 0; i < n; i++) {
    if (v[i]) continue;
    if (a[i].work(x1, y1, x2, y2)) {
      v[i] = 1;
      if (a[i].x1 == a[i].x2) {
        dfs(x1, y1, a[i].x1, y2, n);
        dfs(a[i].x1, y1, x2, y2, n);
      } else {
        dfs(x1, y1, x2, a[i].y2, n);
        dfs(x1, a[i].y1, x2, y2, n);
      }
      return;
    }
  }
  f.push_back((x2 - x1) * (y2 - y1));
}
int main() {
  int w, h, n, i;
  scanf("%d%d%d", &w, &h, &n);
  for (i = 0; i < n; i++)
    scanf("%d%d%d%d", &a[i].x1, &a[i].y1, &a[i].x2, &a[i].y2);
  dfs(0, 0, w, h, n);
  sort(f.begin(), f.end());
  for (i = 0; i < f.size(); i++) {
    if (i != 0) printf(" ");
    printf("%d", f[i]);
  }
  printf("\n");
  return 0;
}
