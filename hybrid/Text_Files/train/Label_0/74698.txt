b = list(map(int, input().split()))
ans = 0
for i in range(14):
	a = b.copy()
	x = a[i]
	if x == 0:
		continue
	a[i] = 0
	tmp = x // 14
	rem = x % 14
	for j in range(14):
		a[j] += tmp
	for j in range(rem):
		a[(i + j + 1) % 14] += 1
	tans = 0
	for c in a:
		if c & 1 == 0:
			tans += c
	ans = max(ans, tans)
print(ans)
