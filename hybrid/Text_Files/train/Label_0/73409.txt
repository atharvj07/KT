def read_line():
	return [int(x) for x in input().split()]

def read_int():
	return int(input())

def solve():
	n = read_int()
	s = str(n)
	a = ''
	b = ''
	for i in range(0, len(s)):
		if i & 1:
			a += s[i]
		else:
			b += s[i]
	if a:
		a = int(a)
	else:
		a = 0
	if b:
		b = int(b)
	else:
		b = 0
	print((a + 1) * (b + 1) - 2)
t = read_int()
while t > 0:
	solve()
	t -= 1
