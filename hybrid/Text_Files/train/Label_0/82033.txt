import sys
sys.setrecursionlimit(10 ** 5)

def p(s, t, idx, flg):
	if len(t) == 0:
		return True
	if flg:
		if idx + 1 < len(s) and s[idx + 1] == t[0]:
			if p(s, t[1:], idx + 1, True):
				return True
		if idx >= 1 and s[idx - 1] == t[0]:
			if p(s, t[1:], idx - 1, False):
				return True
	elif idx >= 1 and s[idx - 1] == t[0]:
		if p(s, t[1:], idx - 1, False):
			return True
	return False
Q = int(input())
for _ in range(Q):
	S = input()
	T = input()
	ts = T[0]
	for i in range(len(S)):
		if S[i] == ts:
			if p(S, T[1:], i, True):
				print('YES')
				break
	else:
		print('NO')
