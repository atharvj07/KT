from math import sqrt

class Point:

	def __init__(self, x, y):
		self.x = x
		self.y = y

class Circle:

	def __init__(self, center, r):
		self.center = center
		self.r = r

	def stop_point(self, disk):
		x_i = disk.center.x
		delta = (self.r + disk.r) ** 2 - (x_i - self.center.x) ** 2
		if delta >= 0:
			y_i = self.center.y + sqrt(delta)
		else:
			y_i = disk.r
		return y_i

class Disks:

	def __init__(self, circles=[]):
		self.circles = circles

	def __contains__(self, circle):
		return circle in self.circles

	def __iter__(self):
		for circle in self.circles:
			yield circle

	def append(self, circle):
		self.circles.append(circle)

	def list_ys(self):
		return [disk.center.y for disk in self.circles]

class Input:

	def __init__(self):
		first_line = input().split(' ')
		second_line = input().split(' ')
		self.NUMBER_OF_DISKS = int(first_line[0])
		self.RADIO = int(first_line[1])
		self.X_POSITIONS = [float(x) for x in second_line]

class Output:

	def __init__(self, list_of_ys):
		self.output_content = ' '.join([str(y) for y in list_of_ys])

	def print_output(self):
		print(self.output_content)

	def __str__(self):
		return self.output_content

	def __repr__(self):
		return self.__str__()

def main():
	input_exercise = Input()
	disks = Disks()
	for x in input_exercise.X_POSITIONS:
		max_y = input_exercise.RADIO
		disk = Circle(Point(x, 1000), input_exercise.RADIO)
		for c in disks:
			y = c.stop_point(disk)
			max_y = y if y > max_y else max_y
		disk.center.y = max_y
		disks.append(disk)
	output = Output(disks.list_ys())
	output.print_output()
main()
