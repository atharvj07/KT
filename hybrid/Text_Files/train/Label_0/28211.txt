#include <bits/stdc++.h>
using namespace std;
template <class T>
inline void init(T& x) {
  x = 0;
  char ch = getchar();
  bool t = 0;
  for (; ch > '9' || ch < '0'; ch = getchar())
    if (ch == '-') t = 1;
  for (; ch >= '0' && ch <= '9'; ch = getchar())
    x = (x << 1) + (x << 3) + (ch - 48);
  if (t) x = -x;
  return;
}
const int N = 800800;
int n, L;
long long a[N], b[N];
inline bool check(int len) {
  int mx = -1e9, l = 1, r = 1;
  for (int i = 1; i <= (n << 1); ++i) {
    while (l <= (n << 2) && b[l] < a[i] - len) ++l;
    while (r <= (n << 2) && b[r] <= a[i] + len) ++r;
    mx = max(mx, l - i);
    if (r - i - 1 < mx) return 0;
  }
  return 1;
}
int main() {
  init(n), init(L);
  for (int i = 1; i <= n; ++i) init(a[i]);
  for (int i = 1; i <= n; ++i) init(b[i]);
  sort(a + 1, a + 1 + n), sort(b + 1, b + 1 + n);
  for (int i = 1; i <= n; ++i) a[i] += L, a[i + n] = a[i] + L;
  for (int i = 1; i <= n * 3; ++i) b[i + n] = b[i] + L;
  int l = 0, r = L, ans = L;
  while (l <= r) {
    int mid = (l + r) >> 1;
    if (check(mid))
      ans = mid, r = mid - 1;
    else
      l = mid + 1;
  }
  cout << ans << endl;
  return 0;
}
