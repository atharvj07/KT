#include <cstdio>
#include <algorithm>
using namespace std;

const int MAXN = 505;

int N;
char mat[MAXN][MAXN];
int row[MAXN], col[MAXN];

void load() {
  scanf("%d", &N);
  for (int i = 0; i < N; i++)
    scanf("%s", mat[i]);
}

int solve() {
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++) {
      row[i] += mat[i][j] == '#';
      col[j] += mat[i][j] == '#';
    }
      
  if (!(*max_element(col, col + N)))
    return -1;
    
  int cnt = 0;
  for (int i = 0; i < N; i++)
    cnt += col[i] < N;
  
  int sol = N + 1;
  for (int i = 0; i < N; i++) 
    sol = min(sol, N - row[i] + !col[i]);
  
  return cnt + sol;
}

int main() {
  load();
  printf("%d\n", solve());
  return 0;
}
