from math import gcd
T = int(input())
for t in range(T):
	n = int(input())
	arr = list(map(int, input().split()))
	val = 0
	for i in range(31):
		ct = 0
		for j in range(n):
			if 1 << i & arr[j] != 0:
				ct += 1
		val = gcd(val, ct)
	if val == 0:
		print(*range(1, n + 1))
		continue
	for i in range(1, val + 1):
		if val % i == 0:
			print(i, end=' ')
	print()
