#include <bits/stdc++.h>
using namespace std;
int n, a[100001], d[32];
long long res1, res2;
map<int, int> g;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    g[a[i]]++;
    if ((a[i] & (a[i] - 1)) == 0) res1 += (d[(int)log2(a[i])]++);
  }
  for (int j = 1; j <= 30; j++)
    for (int i = 0; i < n; i++)
      if (a[i] != (1 << (j - 1))) res2 += g[(1 << j) - a[i]];
  cout << res1 + (res2 >> 1);
}
