import sys
input = sys.stdin.readline
(n, m, i, j, a, b) = map(int, input().split())
MIN = 10 ** 18

def calc(u, v):
	global MIN
	if (u - i) % a == 0 and (v - j) % b == 0:
		x = abs(u - i) // a
		y = abs(v - j) // b
		if x % 2 == y % 2 and 1 + a <= n and (1 + b <= m):
			MIN = min(MIN, max(x, y))
		if x == y == 0:
			MIN = 0
calc(1, 1)
calc(1, m)
calc(n, 1)
calc(n, m)
if MIN == 10 ** 18:
	print('Poor Inna and pony!')
else:
	print(MIN)
