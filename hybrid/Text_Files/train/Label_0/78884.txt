import sys
import collections

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_string():
	return sys.stdin.readline().strip()
for t in range(int(input())):
	(n, m, x) = get_ints()
	col_no = x // n
	row_no = x - n * (x // n)
	if row_no == 0:
		row_no = n
		col_no -= 1
	ans = (row_no - 1) * m + col_no + 1
	print(ans)
