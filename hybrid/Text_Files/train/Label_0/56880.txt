import java.util.Scanner;

public class SocialDistance {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int Cases = sc.nextInt();
        for (int i=0;i<Cases;i++){
            Findtable(sc);
        }
    }

    private static void Findtable(Scanner sc) {
        int n = sc.nextInt();
        int k = sc.nextInt();
        String R = sc.next();
        int Prev = -(k+1);
        int Free = 0;
        boolean empty = true;
        for (int i=0;i<n;i++){
            if (R.charAt(i)=='1'){
                int d = i-Prev-1;
                Free += (d-k)/(k+1);
                Prev = i;
                empty = false;
            }
            if (i==(n-1)){
                int d = k+1+i-Prev-1;
                Free += (d-k)/(k+1);
            }
        }
        if (empty){
            Free= 1+(n-1)/(k+1);
        }
        System.out.println(Free);
    }
}
