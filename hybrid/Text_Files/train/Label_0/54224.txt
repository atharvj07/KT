import java.util.*;
import java.io.*;

public class C{
   public static void main(String[] args) throws IOException{
      BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
      int n = Integer.parseInt(f.readLine());
      for(int i = 0; i < n; i++){
         StringBuilder str1 = new StringBuilder(f.readLine());
         StringBuilder str2 = new StringBuilder(f.readLine());
         HashMap<Character, Integer> chars = new HashMap<Character, Integer>();
         String str3 = f.readLine();
         for(int b = 0; b < str3.length(); b++){
            change(chars, str3.charAt(b), true);
         }
         int ptr1 = -1;
         int ptr2 = -1;
         boolean done = false;
         if(str1.toString().equals(str2.toString())){
            System.out.println("YES");
            continue;
         }
         while(ptr2 < str2.length()-1){
            ptr1++;
            ptr2++;
            char c2 = str2.charAt(ptr2);
            char c1 = '.';
            if(ptr1 < str1.length()){
               c1 = str1.charAt(ptr1);
            }
            //char c1 = ptr1 < str1.length() ? str1.charAt(ptr1) : null;
            if(c1 == c2) continue;
            if(!chars.containsKey(c2)){
               System.out.println("NO");
               done = true;
               break;
            }
            str1.insert(ptr1, c2);       
            change(chars, c2, false);
            if(str1.toString().equals(str2.toString())){
               System.out.println("YES");
               done = true;
               break;
            }
            
         }
         if(!done){
            System.out.println("NO");
         }
      }
         

   }
   public static void change(HashMap<Character, Integer> chars, char c, boolean add){
      if(add){
         if(chars.containsKey(c)){
            chars.put(c, chars.get(c)+1);
         }else{
            chars.put(c, 1);
         }
      }else{
         chars.put(c, chars.get(c)-1);
         if(chars.get(c) == 0) chars.remove(c);
      }
   }
}
