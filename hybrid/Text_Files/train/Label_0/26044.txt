#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10, mod = 1e9 + 7;
int dp[maxn];
int f(int n) {
  if (dp[n] != 0) return dp[n];
  if (n == 2)
    return dp[2] = 2;
  else if (n == 3) {
    return dp[3] = 3;
  } else {
    return dp[n] = (f(n - 1) + f(n - 2)) % mod;
  }
}
vector<int> c;
int main() {
  ios::sync_with_stdio(0);
  string s;
  cin >> s;
  f(s.size() + 2);
  bool w = 0, n = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'u') {
      if (w == 1)
        c[c.size() - 1]++;
      else {
        n = 0;
        w = 1;
        c.push_back(1);
      }
    } else if (s[i] == 'n') {
      if (n == 1)
        c[c.size() - 1]++;
      else {
        n = 1;
        w = 0;
        c.push_back(1);
      }
    } else if (s[i] == 'm' || s[i] == 'w') {
      cout << "0";
      return 0;
    } else {
      w = 0;
      n = 0;
    }
  }
  long long ans = 1;
  for (int i = 0; i < c.size(); i++) {
    if (c[i] != 1) ans = 1ll * ans * dp[c[i]];
    ans %= mod;
  }
  cout << ans;
}
