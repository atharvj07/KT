T = int(input())

def brute_algo(n, x, a):
	new_a = a.copy()
	pos = 0
	while pos < len(new_a):
		if new_a[pos] > 0 and new_a[pos] % x == 0:
			new_a += [new_a[pos] // x] * x
			pos += 1
		else:
			break
	return sum(new_a)
for t in range(T):
	(n, x) = tuple([int(num) for num in input().split()])
	a = [int(num) for num in input().split()]

	def get_multiplicity(ai, x):
		multiplicity = 0
		while ai > 0 and ai % x == 0:
			multiplicity += 1
			ai //= x
		return multiplicity
	amult = [get_multiplicity(ai, x) for ai in a]
	min_amult = min(amult)
	norm_amult = amult.copy()
	have_found_min = False
	for i in range(len(norm_amult)):
		if have_found_min is False:
			if amult[i] > min_amult:
				norm_amult[i] = min_amult + 1
				continue
			else:
				have_found_min = True
		else:
			norm_amult[i] = min_amult
	result = sum([a[i] * (1 + norm_amult[i]) for i in range(len(a))])
	print(result)
