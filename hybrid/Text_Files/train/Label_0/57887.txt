function constructOnes(n) {
  var str = "";
  for (var i = 0; i < n; i++) {
    str += '1';
  }
  return str;
}

var input = readline().trim().split(' ').map(Number);
var n = input[0], d = input[1], result = [0], matchStr = constructOnes(n), consecutive = false;

for (var i = 0; i < d; i++) {
  var str = readline().trim();
  if (str != matchStr) {
    if (consecutive) {
      result[result.length-1]++;
    } else {
      result.push(1);
      consecutive = true;
    }
  } else {
    consecutive = false;
  }
}

print(result.sort(function (a, b) {
  return a-b;
}).pop());
