#include <bits/stdc++.h>
using namespace std;

int n;
long long c[10][10];

int main() {
	scanf("%d",&n);
	for(int i=1;i<=n;i++){
		int a=i;
		int b=a%10;
		while(a>=10)a/=10;
		c[a][b]++;
	}
	long long ans=0;
	for(int i=1;i<10;i++){
		for(int j=1;j<10;j++){
			ans+=c[i][j]*c[j][i];
		}
	}
	printf("%lld",ans);
}
