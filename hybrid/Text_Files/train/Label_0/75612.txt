t = int(input())
while t:
	n = int(input())
	v = list(map(int, input().split()))
	a = [0 for i in range(n)]
	b = [0 for i in range(n)]
	a[0] = v[0]
	flag = True
	for i in range(1, n):
		a[i] = min(a[i - 1], v[i] - b[i - 1])
		b[i] = v[i] - a[i]
		if b[i] < b[i - 1] or a[i] < 0:
			flag = False
			break
	if flag:
		print('YES')
	else:
		print('NO')
	t -= 1
