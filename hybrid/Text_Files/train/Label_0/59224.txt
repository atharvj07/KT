import java.io.*;
import java.util.*;

public class Main {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    PrintWriter pw = new PrintWriter(System.out);

    public static void main(String[] args) throws IOException {
        Main m = new Main();
        m.solve();
        m.close();
    }

    void close() throws IOException {
        pw.flush();
        pw.close();
        br.close();
    }

    int[] readIntLine() throws IOException {
        String[] tokens = br.readLine().split(" ");
        int[] A = new int[tokens.length];
        for (int i = 0; i < A.length; i++) A[i] = Integer.parseInt(tokens[i]);
        return A;
    }

    long[] readLongLine() throws IOException {
        String[] tokens = br.readLine().split(" ");
        long[] A = new long[tokens.length];
        for (int i = 0; i < A.length; i++) A[i] = Long.parseLong(tokens[i]);
        return A;
    }

    void solve() throws IOException {
        int[] line = readIntLine();
        int n = line[0];
        int M = line[1];
        int[] t = readIntLine();
        int sumSoFar = 0;
        List<PriorityQueue<Integer>> pqs = new ArrayList<>();   // pqs.get(i) contains the i * sqrt(n) largest elements
        List<Integer> capacities = new ArrayList<>();
        List<Integer> sums = new ArrayList<>();
        for (int i = 0; i < Math.sqrt(n); i++) {
            pqs.add(new PriorityQueue<>());
            capacities.add((int) Math.sqrt(n) * i);
            sums.add(0);
        }
        pqs.add(new PriorityQueue<>());
        capacities.add(n);
        sums.add(0);

        for (int i = 0; i < t.length; i++) {
        // for (int i = 0; i <= 1; i++) {
            // pw.println("On person " + i);
            for (int j = 1; j < pqs.size(); j++) {
                // pw.println("PQ: " + pqs.get(j));
                int time = M - (sumSoFar - sums.get(j));  // want time to be >= t[i];
                if (time < t[i]) {
                    continue;
                }
                PriorityQueue<Integer> currPq = pqs.get(j);
                List<Integer> temp = new ArrayList<>();
                while (!currPq.isEmpty() && time >= t[i]) {
                    // pw.println("time is " + time);
                    int small = currPq.remove();
                    temp.add(small);
                    time -= small;
                }
                // pw.println("ttime is " + time);
                if (time < t[i]) {
                    int last = temp.get(temp.size() - 1);
                    temp.remove(temp.size() - 1);
                    currPq.add(last);
                }
                int failed = currPq.size();
                for (int x : temp) {
                    currPq.add(x);
                }
                pw.print(failed + " ");
                sumSoFar += t[i];
                break;
            }
            for (int j = 1; j < pqs.size(); j++) {
                PriorityQueue<Integer> currPq = pqs.get(j);
                if (currPq.size() < capacities.get(j)) {
                    currPq.add(t[i]);
                    sums.set(j, sums.get(j) + t[i]);
                } else if (currPq.peek() < t[i]) {
                    int former = currPq.remove();
                    currPq.add(t[i]);
                    sums.set(j, sums.get(j) - former + t[i]);
                }
            }
        }
    }
}
