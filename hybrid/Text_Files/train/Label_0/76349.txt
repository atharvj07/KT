first = [float(x) for x in input().split(' ')]
(R, x1, y1, x2, y2) = (first[0], first[1], first[2], first[3], first[4])
x_f2c = x1 - x2
y_f2c = y1 - y2
dist = (x_f2c ** 2 + y_f2c ** 2) ** 0.5
if dist >= R:
	print(x1, y1, R)
else:
	new_R = (R + dist) / 2
	if x_f2c == 0 and y_f2c == 0:
		print(x1, y2 + 0.5 * R, 0.5 * R)
	else:
		(u_x, u_y) = (x_f2c / dist, y_f2c / dist)
		new_x = x2 + new_R * u_x
		new_y = y2 + new_R * u_y
		print(new_x, new_y, new_R)
