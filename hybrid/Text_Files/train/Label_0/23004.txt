#include<iostream>
#include<queue>
using namespace std;
typedef pair<pair<int,int>,int> pa;
#define ff first.first
#define fs first.second
#define f(a,b) for(int a=0;a<b;a++)

int h,w,x,y,sx,sy,gx,gy,i,j,g,m = 999999;
int dx[4]={1,0,-1,0};
int c[20][20];
char n[20][20];
string pat;

int main(){
    while(cin>>h>>w,h){
        f(i,h)f(j,w){
                c[i][j]=m;
                cin>>n[i][j];
                if(n[i][j]=='A')
                    sx=j,sy=i;
                else if(n[i][j]=='B')
                    gx=j,gy=i;
            }
        queue<pa>q;
        q.push(make_pair(make_pair(sx, sy), 0));
        while(q.size()){
            pa p = q.front();
            q.pop();
            if(c[p.fs][p.ff]!=m)
                continue;
            c[p.fs][p.ff]=p.second;
            f(i,4){
                x=p.ff+dx[i];
                y=p.fs+dx[(i+1)%4];
                if(x>=0&&x<w&&y>=0&y<h&&n[y][x]!='#'&&c[y][x]==m)
                    q.push(make_pair(make_pair(x, y), p.second+1));
            }
        }
        cin>>pat;
        g=0;
        f(j,m){
            i=j%pat.length();
            if(c[gy][gx]<=g){
                cout<<g<<" "<<gy<<" "<<gx<<endl;
                break;
            }
            if(g>700){
                cout<<"impossible"<<endl;
                break;
            }
            if(pat[i]=='8')
                gy--;
            else if(pat[i]=='6')
                gx++;
            else if(pat[i]=='4')
                gx--;
            else if(pat[i]=='2')
                gy++;
            gy=max(0,min(h-1,gy));
            gx=max(0,min(w-1,gx));
            g++;
        }
    }
}
