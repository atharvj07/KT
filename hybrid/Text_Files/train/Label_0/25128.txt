#include <bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    long long mod = 1e9 + 7;
    int t;
    cin >> t;
    while (t--) {
        int n;
        long long ans = 0;
        cin >> n;
        vector<long long> a(n);
        vector<long long> bits(60, 0);
        for (int q = 0; q < n; q++) {
            cin >> a[q];
            long long curr = a[q];
            for (int w = 0; w < 60; w++) {
                bits[w] += curr % 2;
                curr = curr >> 1;
            }
        }
        for (int q = 0; q < n; q++) {
            long long first = 0, second = 0;
            long long curr2 = 1;
            long long curr = a[q];
            for (int w = 0; w < 60; w++) {
                long long bit = curr % 2;
                curr = curr >> 1;
                if (bit == 1) {
                    first = (first + curr2 * bits[w]) % mod;
                    second = (second + curr2 * n) % mod;
                }
                else {
                    second = (second + curr2 * bits[w]) % mod;
                }
                curr2 = (curr2 * 2) % mod;
            }
            ans = (ans + first * second) % mod;
        }
        cout << ans << "\n";
    }
    return 0;
}
