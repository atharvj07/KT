import math
import sys
from itertools import permutations
input = sys.stdin.readline
n = int(input())
ans = [[0, 0]]

def ins(cur):
	ans.append([cur + 1, cur])
	ans.append([cur + 2, cur])
	ans.append([cur, cur + 1])
	ans.append([cur, cur + 2])
	ans.append([cur + 1, cur + 2])
	ans.append([cur + 2, cur + 1])
	ans.append([cur + 2, cur + 2])
ins(0)
for i in range(n):
	ins(2 * (i + 1))
print(len(ans))
for i in range(len(ans)):
	print(*ans[i])
