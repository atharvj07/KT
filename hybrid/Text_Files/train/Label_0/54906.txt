
import java.util.*;
import java.math.*;
public class lucksum {

         static long gcd(long a,long b)
         {
                 if(a>b)
                 {
                         return gcd(b,a);
                 }
                 if(a==0)
                         return b;
                 else
                         return gcd(b%a,a);
         }
         public static void main(String[] args)
        {
               Scanner sc=new Scanner(System.in);
               int n=sc.nextInt();
               int dp[]=new int[n+1];
               int res=-1;
        	   boolean con[]=new boolean[n+1];
               for(int i=3;i<=n;i++)
               {
            	   for(int j=2;j*(j+1)<=2*i;j++)
            	   {

            			   if(2*i%j!=0)
            				   continue;
            			   if((2*i/j+1-j)%2!=0)
            				   continue;
            			   int t=(2*i/j+1-j)/2;
            			   if(t<=0)
            				   continue;
            			   con[dp[t+j-1]^dp[t-1]]=true;
   
            			   if((i==n)&&((dp[t+j-1]^dp[t-1])==0))
            			   {
            				   res=j;
            				   break;
            			   }    			   
            	   }
            	   int t=0;
            	   for(int j=0;j<=n;j++)
            	   {
            		   if(con[j]==false)
            		   {
            			   t=j;
            			   break;
            		   }
            	   }
            	   dp[i]=t^dp[i-1];
            	   for(int j=0;j*(j+1)<=2*i;j++)
            		   con[j]=false;
               }
               System.out.println(res);
               return;
        }

}
