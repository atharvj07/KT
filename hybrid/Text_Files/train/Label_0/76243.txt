from sys import stdin
from collections import deque, defaultdict
from heapq import heappush, heappop
t = int(stdin.buffer.readline())
for _ in range(t):
	(a, s) = map(int, stdin.readline().split())
	ans = ''
	n = len(str(s)) - 1
	ns = str(s)
	for i in str(a)[::-1]:
		if not ns[-1:].strip():
			break
		if int(i) <= int(ns[-1]):
			ans = str(abs(int(ns[-1]) - int(i))).strip() + ans
			ns = ns[:-1]
		else:
			if not ns[-2:].strip:
				break
			ans = str(abs(int(ns[-2:]) - int(i))).strip() + ans
			ns = ns[:-2]
	if ns.strip():
		ans = ns + ans
	nans = ''
	na = str(a)
	if len(ans) > len(na):
		na = '0' * (len(ans) - len(na)) + na
	elif len(na) > len(ans):
		ans = '0' * (len(na) - len(ans)) + ans
	for i in range(len(ans)):
		nans += str(int(ans[i]) + int(na[i]))
	if nans == str(s):
		print(int(ans))
	else:
		print(-1)
