#include <bits/stdc++.h>
int main() {
  int i, j, k, l, m;
  while (scanf("%d", &m) == 1) {
    if (m <= 2)
      printf("0 %d\n", m);
    else if (m > 2 && m <= 5)
      printf("0 2\n");
    else if (m == 6)
      printf("1 2\n");
    else if (m == 7)
      printf("2 2\n");
    else {
      i = m / 7;
      j = m % 7;
      if (j <= 2)
        k = (i * 2) + j;
      else if (j > 2 && j < 6)
        k = (i * 2) + 2;
      else if (j == 6)
        k = (i * 2) + 2;
      if (j < 6)
        printf("%d %d\n", i * 2, k);
      else if (j == 6)
        printf("%d %d\n", (i * 2) + 1, k);
    }
  }
  return 0;
}
