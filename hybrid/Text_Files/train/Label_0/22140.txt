#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main() {
	int n;
	while (cin >> n){
		vector<int>l;
		for (int i = 0; i < n; i++){
			int a; cin >> a;
			l.push_back(a);
		}
		sort(l.begin(), l.end());
		int num = 0;
		for (int i = 0; i < n; i++){
			num += l[i] * (n - i);
		}
		cout << num << endl;
	}
	return 0;
}
