def f(n):
	return 2 ** (n + 1) - 2 - n

def check(p, n, i):
	if n == p // 2 + 1:
		return f(i - 1) + i
	if n < p // 2 + 1:
		return check((p - 1) // 2, n, i - 1)
	else:
		return f(i - 1) + i + check((p - 1) // 2, n - p // 2 - 1, i - 1)
for _ in range(int(input())):
	n = int(input())
	i = 1
	p = 1
	while p < n:
		p = p * 2 + 1
		i += 1
	print(check(p, n, i))
