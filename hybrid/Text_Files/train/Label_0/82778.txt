def find_ones(n):
	cnt = 0
	while n:
		if n & 1:
			cnt += 1
		n //= 2
	return cnt
(n, p) = get_ints()
ok = False
for i in range(1, 35):
	num = n - p * i
	if num < 0:
		break
	ans = find_ones(num)
	if ans <= i and num >= i:
		print(i)
		ok = True
		break
if not ok:
