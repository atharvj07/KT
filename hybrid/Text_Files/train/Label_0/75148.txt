t = int(input())
for _ in range(t):
	n = int(input())
	a = [int(i) for i in input().split()]
	freq = {}
	max_freq = 0
	for i in a:
		freq[i] = (freq.get(i) or 0) + 1
		max_freq = max(max_freq, freq[i])
	count = 0
	max_freq_count = 0
	for i in freq:
		if freq[i] < max_freq:
			count += freq[i]
		else:
			max_freq_count += 1
	if max_freq == 1:
		ans = n
	else:
		ans = count // (max_freq - 1) + max_freq_count - 1
	print(ans)
