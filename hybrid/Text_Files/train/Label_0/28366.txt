#include <bits/stdc++.h>
using namespace std;
int a[3001], b[3001], c[3001], n;
int DP[3001][2];
int dp(int i, bool ant) {
  if (i == n) return 0;
  if (DP[i][ant] != -1) return DP[i][ant];
  int s = -100000;
  if (ant and i != n - 1) s = max(s, c[i] + dp(i + 1, false));
  if (!ant) s = max(s, a[i] + dp(i + 1, true));
  if (ant) s = max(s, b[i] + dp(i + 1, true));
  if (!ant and i != n - 1) s = max(s, b[i] + dp(i + 1, false));
  return DP[i][ant] = s;
}
int main() {
  memset(DP, -1, sizeof DP);
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n; ++i) cin >> b[i];
  for (int i = 0; i < n; ++i) cin >> c[i];
  cout << dp(0, 0) << endl;
}
