import re, sys, string, operator, functools, fractions, collections
sys.setrecursionlimit(10 ** 7)
dX = [-1, 1, 0, 0, -1, 1, -1, 1]
dY = [0, 0, -1, 1, 1, -1, -1, 1]
RI = lambda x=' ': list(map(int, input().split(x)))
RS = lambda x=' ': input().rstrip().split(x)
mod = int(1000000000.0 + 7)
eps = 1e-06
n = RI()[0]
s = RS()[0]
ans = 0
for i in range(n):
	(x, y) = (0, 0)
	for j in range(i, n):
		if s[j] == 'L':
			x -= 1
		elif s[j] == 'R':
			x += 1
		elif s[j] == 'U':
			y += 1
		else:
			y -= 1
		if x == 0 and y == 0:
			ans += 1
print(ans)
