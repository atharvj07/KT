#include <iostream>
#include <algorithm>
#include <string>
#include <vector>
#include <cmath>
#include <map>
#include <queue>
#include <iomanip>
#include <set>
#include <tuple>
#define mkp make_pair
#define mkt make_tuple
#define rep(i,n) for(int i = 0; i < (n); ++i)
using namespace std;
typedef long long ll;
const ll MOD=1e9+7;

int N,M;
ll loc[200020][3];

vector<int> g[200020];
int far[200010][8];
bool visited[200020][8];
ll ans[200020];

ll calc(int id,int t){
    ll res=0;
    for(int j=0;j<3;j++){
        if(t&(1<<j)) res+=loc[id][j];
        else res-=loc[id][j];
    }
    return res;
}

void dfs(int now,int t,int tar){
    visited[now][t]=true;
    far[now][t]=tar;
    for(auto nex:g[now]){
        if(visited[nex][t]) continue;
        dfs(nex,t,tar);
    }
    return;
}

int main(){
  cin>>N>>M;
  rep(i,N) cin>>loc[i][0]>>loc[i][1]>>loc[i][2];
  rep(i,M){
      int a,b;
      cin>>a>>b;
      a--;b--;
      g[b].push_back(a);
  }

  for(int k=0;k<8;k++){
      vector<pair<ll,int>> v;
      for(int i=0;i<N;i++){
          ll r=calc(i,k);
          v.push_back(mkp(r,i));
      }
      sort(v.begin(),v.end());
      reverse(v.begin(),v.end());

      for(int i=0;i<v.size();i++){
          int now=v[i].second;
          if(visited[now][k]) continue;
          dfs(now,k,now);
      }
  }

  for(int i=0;i<N;i++){
      for(int k=0;k<8;k++){
          ans[i]=max(ans[i],calc(i,~k)+calc(far[i][k],k));
      }
      cout<<ans[i]<<endl;
  }

  return 0;
}
