class TaskE {
    private final int MOD = 1000000007;

    private int gao(String x) {
        int n = x.length();
        long p0[] = new long[n + 1];
        long p1[] = new long[n + 1];
        long p2[] = new long[n + 1];
        long prod[] = new long[n + 1];

        long sum = 0;
        long sum4 = 0;

        p0[0] = 1;
        for (int i = 0; i < n; ++i) {

            p0[i + 1] = p0[i] * 2;
            if (x.charAt(i) == '4') --p0[i + 1];
            p0[i + 1] %= MOD;

            p1[i + 1] = p1[i] * 20l + 11l * p0[i];
            if (x.charAt(i) == '4') p1[i + 1] -= sum * 10 + 7;
            p1[i + 1] %= MOD;

            p2[i + 1] = p2[i] * 200l + 2 * p1[i] * 10 * (4 + 7) + (4 * 4 + 7 * 7) * p0[i];
            if (x.charAt(i) == '4') p2[i + 1] -= (sum * 10 + 7) % MOD * ((sum * 10 + 7) % MOD);
            p2[i + 1] %= MOD;

            //a * b + b * c + c * d + ...
            //(a * 10 + 4) * (a * 10 + 7) + (a * 10 + 7) * (b * 10 + 4) + ...
            //a * a * 100 + 110 * a + 28 <----> (a)
            //a * b * 100 + a * 40 + b * 70 + 28
            prod[i + 1] = p2[i] * 100 + p1[i] * 110 + 28 * p0[i];
            prod[i + 1] += prod[i] * 100 + (p1[i] - sum) * 40 + (p1[i] - sum4) * 70 + 28 * (p0[i] - 1);
            if (x.charAt(i) == '4') prod[i + 1] -= (sum * 10 + 4) % MOD * ((sum * 10 + 7) % MOD);
            prod[i + 1] %= MOD;

            sum = sum * 10 + x.charAt(i) - '0';
            sum %= MOD;

            sum4 = sum4 * 10 + 4;
            sum4 %= MOD;
        }

        return (int)prod[n];
    }

    public void solve(int testNumber, InputReader in, PrintWriter out) {
        String from = in.next();
        String to = in.next();

        int ans = gao(to) - gao(from);

        ans %= MOD;
        ans += MOD;
        ans %= MOD;

        out.println(ans);

    }
}

class InputReader {
    BufferedReader reader;
    StringTokenizer tokenizer;

    public InputReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream));
        tokenizer = null;
    }

    public String next() {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (Exception e) {
                throw new UnknownError();
            }
        }
        return tokenizer.nextToken();
    }

    }
