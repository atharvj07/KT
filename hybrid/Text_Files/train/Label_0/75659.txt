from math import comb
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	d = {}
	for i in l:
		d[i] = d.get(i, 0) + 1
	l.sort()
	l.reverse()
	temp1 = l[:k]
	temp = set(temp1)
	ans = 1
	for i in temp:
		ans *= comb(d[i], temp1.count(i))
	print(ans % (10 ** 9 + 7))
