def numInterval(listInterval):
	listInterval = sorted(listInterval, key=lambda r: r[1])
	result = []
	result.append(listInterval.pop(0))
	for inv in listInterval:
		if result[-1][1] < inv[0]:
			result.append(inv)
	return result
N = int(input())
arr = list(map(int, input().split()))
pSum = [0]
for i in arr:
	pSum.append(pSum[-1] + i)
allSums = set()
for i in range(len(pSum)):
	for j in range(i + 1, len(pSum)):
		allSums.add(pSum[j] - pSum[i])
listSums = sorted(allSums)
index = 0
invMap = {}
for i in listSums:
	invMap[i] = index
	index += 1
listInterval1 = []
for i in range(len(allSums)):
	listInterval1.append([])
for i in range(len(pSum)):
	for j in range(i + 1, len(pSum)):
		diff = pSum[j] - pSum[i]
		index1 = invMap[diff]
		listInterval1[index1].append([i, j - 1])
index = 0
maxIndex = 0
maximum = -1
for vect in listInterval1:
	tArr = numInterval(vect)
	count = len(tArr)
	if count > maximum:
		maximum = count
		maxIndex = index
	listInterval1[index] = tArr
	index += 1
print(str(maximum))
for inv in listInterval1[maxIndex]:
	print(str(inv[0] + 1) + ' ' + str(inv[1] + 1))
