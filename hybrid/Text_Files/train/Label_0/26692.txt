#include<bits/stdc++.h>
#define ll long long
using namespace std;

int x[100005];
int idx=0;
int d[100005];
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        idx=0;
        for(int i=1;i<=n;i++){
            scanf("%d",&x[i]);
        }
        for(int i=n;i>=1;i--){
            if(idx==0){
                d[++idx]=i;
            }
            else{
                while(idx>0&&x[i]>x[d[idx]]){
                    idx--;
                }
                d[++idx]=i;
            }
        }
        for(int i=1;i<=idx;i++){
            int ma;
            if(i==1) ma=n+1;
            else ma=d[i-1];
            for(int j=d[i];j<ma;j++){
                cout<<x[j]<<" ";
            }
        }
        cout<<endl;
    }
    return 0;
}
