import java.io.*;
import java.util.*;

public class F implements Runnable{
	public static void main (String[] args) {new Thread(null, new F(), "_cf", 1 << 28).start();}

	IntList[] adj;
	int n, par[];
	
	public void run() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("Go!");

		n = fs.nextInt();
		int m = fs.nextInt();
		adj = new IntList[n];
		for(int i = 0; i < n; i++)
			adj[i] = new IntList();
		for(int i = 0; i < m; i++) {
			int u = fs.nextInt() - 1;
			int v = fs.nextInt() - 1;
			adj[u].add(v);
			adj[v].add(u);
		}
		
		//res <= 3
		
		int[] dists = getDist(0);
		if(dists[n - 1] <= 3) {
			System.out.println(dists[n-1]);
			int[] res = new int[dists[n-1] + 1];
			int idx = 0, at = n - 1;
			while(at != 0) { res[idx++] = at; at = par[at]; }
			res[idx] = at;
			for(int i = res.length - 1; i >= 0; i--) {
				if(i < res.length - 1) System.out.print(" ");
				System.out.print(res[i] + 1);
			}
			System.out.println();
			return;
		}
		
		//res = 4
		for(int i = 0; i < n; i++) {
			if(dists[i] == 2) {
				
				System.out.println(4);
				int[] res = new int[5];
				res[4] = n - 1; res[3] = 0;
				res[2] = i; res[1] = par[i]; res[0] = 0;
				for(int j = 0; j < res.length; j++) {
					if(j > 0) System.out.print(" ");
					System.out.print(res[j] + 1);
				}
				System.out.println();
				return;
				
			}
		}
		
		
		//res = 5
		int[] comp = new int[n];
		int[] size = new int[n + 10];
		Arrays.fill(comp, -1);
		comp[0] = 0;
		int count = 1;
		for(int i = 1; i < n; i++) {
			if(dists[i] != 1) continue;
			if(comp[i] != -1) continue;
			comp[i] = count;
			ArrayDeque<Integer> bfs = new ArrayDeque<>();
			bfs.add(i);
//			System.out.println("Working on " + (i + 1));
			while(!bfs.isEmpty()) {
				int u = bfs.pollFirst();
//				System.out.println("At " + (u + 1) + " " + comp[u]);
				size[count]++;
				for(int j = 0; j < adj[u].size(); j++) {
					int v = adj[u].get(j);
					if(comp[v] == -1) {
						comp[v] = count;
						bfs.add(v);
					}
				}
			}
			count++;
		}
		int mark = -1;
		for(int i = 1; i < n; i++) {
			if(comp[i] == -1) continue;
			int sz = size[comp[i]];
			if(adj[i].size() < sz) {
				mark = i;
				break;
			}
		}
		
//		System.out.println("M " + (mark + 1));
//		System.out.println(Arrays.toString(comp));

		if(mark != -1) {
			boolean[] vis = new boolean[n];
			vis[mark] = true; vis[0] = true;
			ArrayDeque<Integer> bfs = new ArrayDeque<>();
			for(int i = 0; i < adj[mark].size(); i++) {
				int v = adj[mark].get(i);
				if(v == 0) continue;
				vis[v] = true;
				bfs.add(v);
			}
			int buildPar = -1, build = -1;
			find: while(!bfs.isEmpty()) {
				int u = bfs.pollFirst();
				for(int i = 0; i < adj[u].size(); i++) {
					int v = adj[u].get(i);
					if(!vis[v]) {
						buildPar = u; build = v;
						break find;
					}
				}
			}
			
			int[] res = new int[6];
			res[0] = 0; res[1] = mark;
			res[2] = buildPar; res[3] = build;
			res[4] = mark; res[5] = n - 1;
			System.out.println(5);
			for(int j = 0; j < res.length; j++) {
				if(j > 0) System.out.print(" ");
				System.out.print(res[j] + 1);
			}
			System.out.println();
			return;
		}
		
		out.println(-1);
		out.close();
	}
	
	int[] getDist(int src) {
		int[] dist = new int[n];
		par = new int[n];
		int oo = (int)1e9;
		Arrays.fill(dist, oo);
		dist[src] = 0;
		ArrayDeque<Integer> bfs = new ArrayDeque<>();
		bfs.addLast(src);
		while(!bfs.isEmpty()) {
			int u = bfs.pollFirst();
			for(int i = 0; i < adj[u].size(); i++) {
				int v = adj[u].get(i);
				if(dist[v] != oo) continue;
				par[v] = u;
				dist[v] = dist[u] + 1;
				bfs.add(v);
			}
		}
		return dist;
	}
	
	class IntList {
		int[] EMPTY = {};
		int n = 0, a[] = EMPTY;
		void add(int v) {
			if (n >= a.length)
				a = Arrays.copyOf(a, (n << 2) + 8);
			a[n++] = v;
		}
		int get(int idx) { return a[idx]; }
		int size() { return n; }
	}

	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) throws FileNotFoundException {
			in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}
		
		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}
		
	}

}
