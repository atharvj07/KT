#include <bits/stdc++.h>
using namespace std;
int a[509][509];
int main() {
  int n, m;
  cin >> n >> m;
  long long s = 0;
  int i, j;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      cin >> a[i][j];
    }
  }
  for (i = n - 1; i >= 1; i--) {
    for (j = m - 1; j >= 1; j--) {
      if (a[i][j] == 0) {
        a[i][j] = min(a[i][j + 1], a[i + 1][j]) - 1;
      }
    }
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      if ((j < m && a[i][j] >= a[i][j + 1]) ||
          (i < n && a[i][j] >= a[i + 1][j])) {
        cout << -1 << endl;
        return 0;
      }
    }
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      s += a[i][j];
    }
  }
  cout << s << endl;
  return 0;
}
