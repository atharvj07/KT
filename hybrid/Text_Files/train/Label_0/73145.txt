import io, os, sys
input = sys.stdin.readline
print = sys.stdout.write
t = int(input())
START = 500001
seen = [0] * (2 * START)
while t > 0:
	t -= 1
	s = input().strip('\n')
	cur = START
	for c in s:
		if c == '0':
			seen[cur] += 1
			cur -= 1
		else:
			cur += 1
			seen[cur] += 1
	cur = START
	while True:
		if seen[cur] > 1 or (seen[cur] > 0 and seen[cur + 1] == 0):
			seen[cur] -= 1
			cur -= 1
			print('0')
		elif seen[cur + 1] > 0:
			cur += 1
			seen[cur] -= 1
			print('1')
		else:
			break
	print('\n')
