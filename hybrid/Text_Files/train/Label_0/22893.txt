#include <bits/stdc++.h>
template <typename T>
struct BIT {
  const long long n;
  T* arr;
  BIT(const long long& _n) : n(_n), arr(new T[_n + 2]()) {}
  ~BIT(void) { delete[] arr; }
  void add(const long long& i, const long long& v) {
    for (long long j = i; j <= n; j += (j & -j)) arr[j] += v;
  }
  T qry(const long long& hi) {
    long long ret = 0;
    for (long long j = hi; j > 0; j -= (j & -j)) ret += arr[j];
    return ret;
  }
  T qry(const long long& lo, const long long& hi) {
    return qry(hi) - qry(lo - 1);
  }
};
int main(void) {
  std::ios::sync_with_stdio(0);
  std::cin.tie(0);
  long long n, m;
  std::cin >> n >> m;
  BIT<long long> bit(n + m + 10);
  long long first_unused = 1;
  for (int i = 0; i < n; ++i) {
    bit.add(first_unused++, 1);
  }
  std::vector<long long> min(n), max(n), pos(n);
  for (long long i = 0; i < n; ++i) {
    long long v = i;
    pos[v] = n - i;
    min[v] = max[v] = i + 1;
  }
  for (long long i = 0, v; i < m; ++i) {
    std::cin >> v;
    --v;
    min[v] = 1;
    long long sum = bit.qry(pos[v], first_unused);
    bit.add(pos[v], -1);
    bit.add(first_unused, 1);
    pos[v] = first_unused++;
    max[v] = std::max(max[v], sum);
  }
  for (long long i = 0; i < n; ++i) {
    long long sum = bit.qry(pos[i], first_unused);
    max[i] = std::max(max[i], sum);
    std::cout << min[i] << " " << max[i] << "\n";
  }
  return 0;
}
