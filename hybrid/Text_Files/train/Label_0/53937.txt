

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class GuessYourWayOut {

	static long solve(int h, long n, int c) { // 1 right, 0 left
		if (h == 0)
			return 0;
		long node = (1L << h) / 2;
		long sum = ((1L << (h + 1)) - 1) / 2;
		if (c == 1) { // right
			if (n <= node) { // in left
				return 1 + sum + solve(h - 1, n, c);
			}
			return 1 + solve(h - 1, n - node, c ^ 1);
		}
		if (c == 0) { // left
			if (n <= node) { // in left
				return 1 + solve(h - 1, n, c ^ 1);
			}
			return 1 + sum + solve(h - 1, n - node, c);
		}
		return 0;
	}

	public static void main(String[] args) throws IOException {
		MyScanner sc = new MyScanner(System.in);
		PrintWriter pw = new PrintWriter(System.out);
		int h = sc.nextInt();
		long n = sc.nextLong();
		pw.println(solve(h, n, 0));
		pw.close();
	}

	static class MyScanner {
		StringTokenizer st;
		BufferedReader br;

		public MyScanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String nextLine() throws IOException {
			return br.readLine();
		}

		public boolean ready() throws IOException {
			return br.ready();
		}

	}
}
