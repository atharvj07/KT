t = list(map(int, input().split()))
t.sort()
arr = [0] * 5
num = [0] * 5
arr[0] = 1
num[0] = t[0]
j = 0
sumt = sum(t)
for i in range(1, 5):
	if t[i - 1] == t[i]:
		arr[j] += 1
	else:
		j += 1
		arr[j] += 1
		num[j] = t[i]
i = 0
ans = sum(t)
while i < 5 and arr[i] != 0:
	if arr[i] == 1:
		i += 1
		continue
	rem = sumt - num[i] * min(arr[i], 3)
	ans = min(rem, ans)
	i += 1
print(ans)
