
import java.util.*;

public class cf139a {

    public static void main(String[] args) {

        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] d = new int[7];
        for (int i = 0; i < 7; i++) {
            
            d[i] = in.nextInt();
            
        }
        while (n > 0) {
            for (int i = 1; i <= 7; i++) {

                n = n -d[i - 1];

                if (n <= 0) {
                    System.out.println(i);
                    break;
                }

            }

        }
    }

}
