#include<bits/stdc++.h>

using namespace std;

int m[50][50]={};
int w,h;
int cnt;

int dx[8]={-1,0,1,-1,1,-1,0,1},dy[8]={-1,-1,-1,0,0,1,1,1};

void fanc(int y,int x){
	m[y][x]=0;
	int ny,nx;
	for(int i=0;i<8;i++){
		ny = y+dy[i];
		nx = x+dx[i];
		if(ny>=0&&nx>=0&&ny<h&&nx<w&&m[ny][nx]==1){
			fanc(ny,nx);
		}
	}
}

int main(){
	
	
	while(1){
		
		cnt=0;
		cin >> w >> h;
		if(w==0) break;
		
		for(int i=0;i<h;i++){
			for(int j=0;j<w;j++){
				cin >> m[i][j];
			}
		}
		
		for(int i=0;i<h;i++){
			for(int j=0;j<w;j++){
				if(m[i][j]!=0){
					cnt++;
					fanc(i,j);
				}
			}
		}
		cout << cnt << endl;
	}
	return 0;
}
