#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
using namespace std;

typedef long long ll;

string bit(ll x,int n){
	string res="";
	for(int i=0;i<n;i++){
		if(x%2)res="1"+res;
		else res="0"+res;
		x/=2;
	}
	return res;
}

int main(){
	ll n;
	while(cin >> n,n){
		string s; cin >> s;
		reverse(s.begin(),s.end());
		ll a=0,b=0;
		s+="1";
		for(int i=0;i<s.size();i++){
			if(s[i]=='1')a+=(1LL<<i);
		}
		ll now=a;
		while(1){
			// (1LL>>53)以上になるまで足す回数は？
			if(a==0)break;
			ll add=((1LL<<53)-now+a-1)/a;
			add=min(add,n);
			now+=a*add;
			while(now>=(1LL<<53)){
				// 超えた分右シフト
				now>>=1;
				a>>=1;
				b++;
			}
			n-=add;
			if(n==0)break;
		}
		cout << bit(b,12) << bit(now,52) << endl;
	}
}
