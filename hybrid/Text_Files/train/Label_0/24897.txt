#include <bits/stdc++.h>
using namespace std;
const int inf = 1.01e9;
const double eps = 1e-9;
int main() {
  long long x, y;
  scanf("%lld%lld", &x, &y);
  while (x != 0 && y != 0) {
    if (x >= 2 * y) {
      x %= 2 * y;
      continue;
    } else if (y >= 2 * x) {
      y %= 2 * x;
      continue;
    } else
      break;
  }
  printf("%lld %lld\n", x, y);
  return 0;
}
