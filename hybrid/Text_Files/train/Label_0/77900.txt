(copybooks, one_copybook, two_copybooks, three_copybooks) = [int(x) for x in input().split()]
k = 0
rubles = 0
while True:
	if (k + copybooks) % 4 == 0:
		break
	k += 1
if k == 1:
	rubles = min(one_copybook, two_copybooks + three_copybooks, three_copybooks * 3)
elif k == 2:
	rubles = min(two_copybooks, one_copybook * 2, three_copybooks * 2)
elif k == 3:
	rubles = min(three_copybooks, one_copybook * 3, one_copybook + two_copybooks)
print(rubles)
