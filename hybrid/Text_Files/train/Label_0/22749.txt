#include <bits/stdc++.h>
using namespace std;
int n, m, p, cnt, t, k[55 * 55], x[100010], y[100010];
char s[55][55];
void solve() {
  for (int i = k[t] = 0; i < n; i++) {
    scanf("%s", s[i]);
    for (int j = 0; j < m; j++) k[s[i][j]]++;
  }
  while (k[t]) {
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        if (s[i][j] == 'U' && s[i][j + 1] == 'U')
          x[++cnt] = i + 1, y[cnt] = j + 1, s[i][j] = s[i + 1][j] = 'L',
          s[i][j + 1] = 'R', k['U'] -= 2, k['L'] += 2;
    if (!k[t]) return;
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        if (s[i][j] == 'L' && s[i + 1][j] == 'L')
          x[++cnt] = i + 1, y[cnt] = j + 1, s[i][j] = s[i][j + 1] = 'U',
          s[i + 1][j] = 'D', k['U'] += 2, k['L'] -= 2;
  }
}
int main() {
  scanf("%d%d", &n, &m);
  t = n & 1 ? 'U' : 'L';
  solve();
  p = cnt;
  solve();
  printf("%d\n", cnt);
  for (int i = 1; i <= p; i++) printf("%d %d\n", x[i], y[i]);
  for (int i = cnt; i > p; i--) printf("%d %d\n", x[i], y[i]);
  return 0;
}
