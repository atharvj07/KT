import math
import sys
(n, k) = map(int, input().split())
for p in range(k):
	(i, j) = map(int, sys.stdin.readline().split())
	i -= 1
	j -= 1
	if n % 2 == 1:
		if (i + j) % 2 == 0:
			ans = n // 2 * i + (i + 1) // 2 + int(j / 2) + 1
		else:
			ans = n * n // 2 + 1 + n // 2 * i + i // 2 + j // 2 + 1
		print(ans)
	else:
		if (i + j) % 2 == 0:
			ans = n // 2 * i + j // 2 + 1
		else:
			ans = n ** 2 // 2 + int(i * (n / 2)) + int(j / 2) + 1
		print(ans)
