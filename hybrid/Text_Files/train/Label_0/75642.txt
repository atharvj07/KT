def inp():
	return map(int, input().split(' '))
(n, k) = inp()
cnt = [0 for i in range(101)]
for i in range(n):
	temp = input()
	length = len(temp)
	cnt[length] += 1
S = input()
mini = 0
maxi = 0
for i in range(len(S)):
	mini += cnt[i]
mini += int(mini / k) * 5
for i in range(len(S) + 1):
	maxi += cnt[i]
maxi -= 1
maxi += int(maxi / k) * 5
print(str(mini + 1) + ' ' + str(maxi + 1))
