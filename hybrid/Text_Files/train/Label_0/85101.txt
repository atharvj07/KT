from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	n = int(input())
	(l, r, u, d) = (-10 ** 5, 10 ** 5, 10 ** 5, -10 ** 5)
	for i in range(n):
		(x, y, a, b, c, D) = map(int, input().split())
		if a == 0:
			l = max(l, x)
		if b == 0:
			u = min(u, y)
		if c == 0:
			r = min(r, x)
		if D == 0:
			d = max(d, y)
	if l > r or d > u:
		print(0)
	else:
		print(1, l, u)
