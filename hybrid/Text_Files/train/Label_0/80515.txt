import atexit
import io
import sys
_I_B = sys.stdin.read().splitlines()
input = iter(_I_B).__next__
_O_B = io.StringIO()
sys.stdout = _O_B

@atexit.register
def write():
	sys.__stdout__.write(_O_B.getvalue())

def main():
	s = input()
	mn = s[0]
	for i in range(len(s)):
		mn = min(mn, s[i])
		if s[i] > mn:
			print('Ann')
		else:
			print('Mike')
main()
