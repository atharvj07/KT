(n, w) = input().split()
(n, w) = (int(n), int(w))
x = [int(i) for i in input().split()]
need = 0
ans = []
for i in range(0, n):
	need += (x[i] + 1) // 2
	ans.append([(x[i] + 1) // 2, x[i], i])
if w < need:
	print(-1)
else:
	w -= need
	ans.sort(key=lambda x: x[1], reverse=1)
	for i in range(n):
		now = ans[i][1] - ans[i][0]
		if w >= now:
			w -= now
			ans[i][0] = ans[i][1]
		else:
			ans[i][0] += w
			break
	ans.sort(key=lambda x: x[2])
	for i in range(n):
		print(ans[i][0], end=' ')
