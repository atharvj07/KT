import java.io.BufferedReader; 
import java.io.IOException; 
import java.io.InputStreamReader; 
import java.util.*;; 
import java.util.StringTokenizer; 
public class captain_longvoyage
{
    public static void main(String args[])
    {
        FastReader sc=new FastReader();
        int t=sc.nextInt();
        int n,len;
        String s,st="";
        while(t-->0)
        {
            n=sc.nextInt();
            if(n==1)
            {
                System.out.println("8");
                continue;
            }
            if(n%4==0)
                len=n-(n/4);
            else
                len=n-(n/4)-1;
            s=String.format("%0" + len + "d", 0).replace('0', '9');
            if(n%4==0)
                len=n/4;
            else
                len=n/4+1;
            st=String.format("%0" + len + "d", 0).replace('0','8');
                System.out.println(s+st);
        }
    }
}        
class FastReader 
{ 
    BufferedReader br; 
    StringTokenizer st; 
    public FastReader() 
    { 
        br = new BufferedReader(new
        InputStreamReader(System.in)); 
    } 
    String next() 
    { 
        while (st == null || !st.hasMoreElements()) 
        { 
            try
            { 
                st = new StringTokenizer(br.readLine()); 
            } 
            catch (IOException e) 
            { 
                e.printStackTrace(); 
            } 
        } 
        return st.nextToken(); 
    } 
    int nextInt() 
    { 
        return Integer.parseInt(next()); 
    } 
    long nextLong() 
    { 
        return Long.parseLong(next()); 
    }
    double nextDouble() 
    { 
        return Double.parseDouble(next()); 
    } 
    String nextLine() 
    { 
        String str = ""; 
        try
        { 
            str = br.readLine(); 
        } 
        catch (IOException e) 
        { 
            e.printStackTrace(); 
        } 
        return str; 
    } 
}
