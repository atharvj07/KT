const long long mod = 1000000007;
long long powmod(long long a, long long b) {
  long long res = 1;
  a %= mod;
  assert(b >= 0);
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
  }
  return res;
}

int gcd(int big, int small) {
  if (small > big) swap(big, small);
  int temp;
  while (small != 0) {
    if (small > big) swap(big, small);
    temp = big % small;
    big = small;
    small = temp;
  }
  return big;
}
const int MAXN = 2002;
int dp[MAXN][MAXN];
int a[MAXN];
int main() {
  int n;
  scanf("%d", &n);
  bool hasone = false;
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
    if (a[i] == 1) hasone = true;
  }
  if (hasone) {
    int counts = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != 1) {
        counts++;
      }
    }
    cout << counts << endl;
    return 0;
  }
  for (int i = 0; i < n - 1; i++) {
    dp[i][i + 1] = gcd(a[i], a[i + 1]);
  }
  for (int i = 0; i < n; i++) {
    for (int j = i + 2; j < n; j++) {
      dp[i][j] = gcd(dp[i][j - 1], dp[j - 1][j]);
    }
  }
  int mindis = INT_MAX;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (dp[i][j] == 1) {
        mindis = min(mindis, j - i - 1);
      }
    }
  }
  if (mindis == INT_MAX)
    cout << -1 << endl;
  else
    cout << (mindis + n) << endl;
  return 0;
}
