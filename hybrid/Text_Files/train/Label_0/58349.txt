import javafx.util.Pair;

import java.io.*;
import java.util.*;

public class Main {

    private static myScanner sc;
    private static PrintWriter pw;

    private static final boolean defaultInAndOutPut = true;

    private static final int numberOfTests = 1;

    private static final String nameOfInAndOutFile = "";
    private static final String fullNameInputFile = nameOfInAndOutFile + ".in";
    private static final String fullNameOutputFile = nameOfInAndOutFile + ".out";

    public static void main(String[] args) throws IOException {
        if (defaultInAndOutPut) {
            setDefaultInAndOutPut();
        } else {
            setInAndOutPutFromFile();
        }
        for (int test = 1; test <= numberOfTests; test++) {
            solve();
            pw.flush();
        }

        pw.close();
    }

    private static void setDefaultInAndOutPut() {
        sc = new myScanner(new BufferedReader(new InputStreamReader(System.in)));
        pw = new PrintWriter(System.out);
    }

    private static void setInAndOutPutFromFile() throws IOException {
        sc = new myScanner(new BufferedReader(new FileReader(new File(fullNameInputFile))));
        pw = new PrintWriter(new File(fullNameOutputFile));
    }

    static int n, m;
    static long[] tasks;
    static Pair<Long,Integer>[] questions;

    private static void solve() {
        n = sc.nextInt();
        m = sc.nextInt();
        tasks = new long[n];
        questions = new Pair[m+2];
        for (int i = 0; i < n; i++) tasks[i] = sc.nextLong();
        for (int i = 0; i < m; i++) questions[i] = new Pair<>(sc.nextLong(),i);
        questions[m] = new Pair<>(1000000000000000L,m);
        questions[m+1] = new Pair<>(-1000000000000000L,m+1);
        long[] answers = new long[n + 1];
        Arrays.sort(questions, new Comparator<Pair<Long, Integer>>() {
            @Override
            public int compare(Pair<Long, Integer> o1, Pair<Long, Integer> o2) {
                return Long.compare(o1.getKey(),o2.getKey());
            }
        });

        for (int i = 0; i <= n; i++) {
            answers[i] = getAnsNow(i);
        }
        Arrays.sort(questions, new Comparator<Pair<Long, Integer>>() {
            @Override
            public int compare(Pair<Long, Integer> o1, Pair<Long, Integer> o2) {
                return Integer.compare(o1.getValue(),o2.getValue());
            }
        });
        for (int i = 0; i < m; i++) {
            int min = -1, max = n;
            while (max - min > 1) {
                int mid = (max + min) / 2;
                if (questions[i].getKey() >= answers[mid]) max = mid;
                else min = mid;
            }
            pw.println(max);
        }
    }

    private static int getNumberDeleting(long start) {
        long now = start;
        int c = 0;
        PriorityQueue<Long> set = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            set.add(tasks[i]);
            now += tasks[i];
            if (now < 0) {
                now -= set.poll();
                c++;
            }
        }
        return c;
    }

    private static long getAnsNow(int c) {
        int min = -1;
        int max = m + 1;
        while (max - min > 1) {
            int mid = (max + min) / 2;
            if (getNumberDeleting(questions[mid].getKey()) > c) min = mid;
            else max = mid;
        }
        return questions[max].getKey();
    }
}

class myScanner {
    private char[] buffer = new char[1 << 8];
    private int pos = 1;

    private BufferedReader reader;

    public myScanner(BufferedReader reader) {
        this.reader = reader;
    }

    public boolean hasNext() {
        return pos > 0;
    }

    private void loadBuffer() {
        pos = 0;
        try {
            for (int i; (i = reader.read()) != -1; ) {
                char c = (char) i;
                if (c != ' ' && c != '\n' && c != '\t' && c != '\r' && c != '\f') {
                    if (pos == buffer.length) buffer = Arrays.copyOf(buffer, 2 * pos);
                    buffer[pos++] = c;
                } else if (pos != 0) break;
            }
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }

    public String current() {
        return String.copyValueOf(buffer, 0, pos);
    }


    public String next() {
        loadBuffer();
        return current();
    }


    public String nextLine() {
        try {
            return reader.readLine();
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }


    public int nextInt() {
        return nextInt(10);
    }

    public long nextLong() {
        return nextLong(10);
    }

    public int nextInt(int radix) {
        loadBuffer();
        int result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber(0 <= digit && digit <= radix - 1);
            result = result * radix + digit;
        }
        return buffer[0] == '-' ? -result : result;
    }

    public long nextLong(int radix) {
        loadBuffer();
        long result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber(0 <= digit && digit <= radix - 1);
            result = result * radix + digit;
        }
        return buffer[0] == '-' ? -result : result;
    }

    public double nextDouble() {
        loadBuffer();
        long result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        long round = 1;
        final int radix = 10;
        boolean hasPoint = false;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber((0 <= digit && digit <= radix - 1) || (!hasPoint && digit == -2));
            if (digit == -2) hasPoint = true;
            else {
                if (hasPoint) round *= radix;
                result = result * radix + digit;
            }

        }
        return buffer[0] == '-' ? -result / (1.0 * round) : result / (1.0 * round);
    }

    private void checkValidNumber(boolean condition) {
        if (!condition) throw new NumberFormatException(current());
    }
}
