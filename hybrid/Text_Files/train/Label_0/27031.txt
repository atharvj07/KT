#include <bits/stdc++.h>
using namespace std;
int main() {
  string a, b;
  cin >> a;
  a = '%' + a;
  b = a;
  a[1] = b[1];
  a[2] = b[3];
  a[3] = b[5];
  a[4] = b[4];
  a[5] = b[2];
  long long um, res = 0;
  for (int i = 1; i <= 5; ++i) {
    res = res * 10 + int(a[i] - '0');
  }
  um = res;
  res = 1;
  for (int i = 1; i <= 5; ++i) {
    res = (res * um) % 100000;
  }
  a = "";
  while (res) {
    a = char(res % 10 + '0') + a;
    res /= 10;
  }
  while ((int)a.length() < 5) a = '0' + a;
  cout << a;
  return 0;
}
