qq = lambda : map(int, input().split())
a = []
(n,) = qq()
a.append(list(qq()))
for _ in range(n - 1):
	(x, y) = qq()
	for k in range(len(a)):
		for i in a:
			if x <= i[0] and y >= i[1]:
				a.remove(i)
			elif y < i[0] or x > i[1]:
				continue
			elif x <= i[0] and y < i[1]:
				a.append([y, i[1]])
				a.remove(i)
			elif x > i[0] and y >= i[1]:
				a.append([i[0], x])
				a.remove(i)
			elif i[0] < x and y < i[1]:
				a.append([i[0], x])
				a.append([y, i[1]])
				a.remove(i)
s = 0
for i in a:
	s += i[1] - i[0]
print(s)
