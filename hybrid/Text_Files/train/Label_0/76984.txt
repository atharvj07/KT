def solve():
	D = input()
	(K, N, M) = map(int, input().split())
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	ans = []
	i = 0
	j = 0
	while i < N or j < M:
		if i == N:
			if B[j] == 0:
				ans.append(B[j])
				j += 1
				K += 1
			elif B[j] <= K:
				ans.append(B[j])
				j += 1
			else:
				return -1
		elif j == M:
			if A[i] == 0:
				ans.append(A[i])
				i += 1
				K += 1
			elif A[i] <= K:
				ans.append(A[i])
				i += 1
			else:
				return -1
		elif A[i] == 0:
			ans.append(A[i])
			i += 1
			K += 1
		elif B[j] == 0:
			ans.append(B[j])
			j += 1
			K += 1
		elif A[i] <= K:
			ans.append(A[i])
			i += 1
		elif B[j] <= K:
			ans.append(B[j])
			j += 1
		else:
			return -1
	return ans
T = int(input())
for _ in range(T):
	x = solve()
	if x == -1:
		print(x)
	else:
		print(*x)
