#include <bits/stdc++.h>
using namespace std;
int a[2010];
int gcd(int a, int b) { return !b ? a : gcd(b, a % b); }
int main() {
  int n;
  while (scanf("%d", &n) > 0) {
    int not_one = 0;
    for (int i = 0; i < n; i++) {
      scanf("%d", &a[i]);
      not_one += (a[i] != 1);
    }
    if (!not_one) {
      printf("0\n");
      continue;
    }
    if (not_one < n) {
      printf("%d\n", not_one);
      continue;
    }
    int min_seg = n + 1;
    for (int i = 0; i < n; i++) {
      int tmp = a[i], j = i + 1;
      while (j < n and tmp != 1) {
        tmp = gcd(tmp, a[j]);
        j++;
      }
      if (tmp == 1) {
        min_seg = min(min_seg, j - i);
      }
    }
    if (min_seg == n + 1) {
      printf("-1\n");
    } else {
      printf("%d\n", not_one + min_seg - 2);
    }
  }
}
