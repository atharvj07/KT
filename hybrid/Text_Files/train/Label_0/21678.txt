#include<iostream>
using namespace std;

int main(){
  int k;
  cin >> k;
  for(int i=0;i<k;i++){
    int gx,gy;
    cin >> gx >> gy;
    int p;
    cin >> p;
    int A[16][16][16][16]={};
    for (int j=0;j<p;j++){
      int a,b,c,d;
      cin >> a >> b>>c>>d;
      A[a][b][c][d] = 1;
    }
    int B[16][16]={};
    B[0][0] = 1;
    for(int t= 0;t<=gx;t++){
      for(int j=0;j<=gy;j++){
	if(j==0){
	  if(t!=0){
	    if(A[t-1][0][t][0]==0&&A[t][0][t-1][0]==0){
	      B[t][0] = B[t-1][0];
	    }
	  }
	}else{
	  if(t==0){
	    if(A[0][j-1][0][j]==0&&A[0][j][0][j-1]==0){
	      B[0][j]=B[0][j-1];
	    }
	  }else{
	    if(A[t][j-1][t][j]==0&&A[t][j][t][j-1]==0){
	      B[t][j]+=B[t][j-1];
	    }
	    if(A[t-1][j][t][j]==0&&A[t][j][t-1][j]==0){
	      B[t][j]+=B[t-1][j];
	    }
	  }
	} 
      }
    }
    if(B[gx][gy]==0){
      cout<<"Miserable Hokusai!" <<endl;
    }else{
      cout << B[gx][gy] << endl;
    }
  }
}
