import math

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)

def getlist():
	return list(map(int, input().split()))

def getvalue():
	return int(input())

def getstring():
	return input()
from functools import reduce

def factors(n):
	return 2 * sum((1 if not n % i else 0 for i in range(1, int(n ** 0.5) + 1))) - (1 if int(math.sqrt(n)) ** 2 == n else 0)
n = getvalue()
arr = getlist()
gc = arr[0]
for i in range(1, n):
	gc = math.gcd(gc, arr[i])
print(factors(gc) if gc > 1 else 1)
