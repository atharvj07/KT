#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e18 + 9;
void fast() { std::ios_base::sync_with_stdio(0); }
int n, m;
int root[150000 + 9], nodes[150000 + 9], edges[150000 + 9];
int find(int u) {
  if (u == root[u]) return u;
  return root[u] = find(root[u]);
}
int main() {
  cin >> n >> m;
  for (int i = 0; i <= n; i++) {
    root[i] = i;
    nodes[i] = 1;
    edges[i] = 0;
  }
  int x, y;
  while (m--) {
    scanf("%d%d", &x, &y);
    x = find(x);
    y = find(y);
    if (x != y) {
      root[y] = x;
      nodes[x] += nodes[y];
      edges[x] += edges[y] + 1;
    } else
      edges[y]++;
  }
  for (int i = 1; i <= n; i++) {
    if (root[i] == i) {
      if (((nodes[i] * 1ll) * (1ll * (nodes[i] - 1))) / 2 * 1ll != edges[i]) {
        cout << "NO\n";
        return 0;
      }
    }
  }
  cout << "YES\n";
  return 0;
}
