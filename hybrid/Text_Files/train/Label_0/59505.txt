import java.util.*;
import java.util.Arrays;

public class potionshomework{
	public static void main(String [] args){
		Scanner in=new Scanner(System.in);
			int n=in.nextInt();int i;long A[]=new long[n];long B[]=new long[n];

		for(i=0;i<n;i++){
			A[i]=in.nextInt();
		}

	Arrays.sort(A);
		for(i=0;i<n;i++)
			B[i]=A[n-i-1];
			long answer=0;
		for(i=0;i<n;i++)
			answer=(answer+(A[i]*B[i]))%10007;


	System.out.println(answer);

	}//end of main method
}//end  of main class
