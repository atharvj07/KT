import sys
for _ in range(int(input())):
	(n, t) = map(int, sys.stdin.readline().split())
	arr = list(map(int, sys.stdin.readline().split()))
	ans_arr = [2 for i in range(n)]
	flag = 0
	if t % 2 != 0:
		mid = t // 2
		for i in range(n):
			if arr[i] <= mid:
				ans_arr[i] = 0
			elif arr[i] > mid:
				ans_arr[i] = 1
	elif t % 2 == 0:
		mid = t // 2
		for i in range(n):
			if arr[i] < mid:
				ans_arr[i] = 0
			elif arr[i] > mid:
				ans_arr[i] = 1
			elif arr[i] == mid:
				if flag == 0:
					ans_arr[i] = 0
					flag = 1
				elif flag == 1:
					ans_arr[i] = 1
					flag = 0
	print(' '.join(map(str, ans_arr)))
