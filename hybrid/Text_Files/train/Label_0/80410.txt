import math
t = int(input())

def findPrime(i):
	while True:
		prime = True
		for j in range(2, math.floor(math.sqrt(i)) + 1):
			if i % j == 0:
				prime = False
				break
		if prime:
			return i
		i += 1
while t > 0:
	t -= 1
	n = int(input())
	tmp = 1 + n
	d1 = findPrime(tmp)
	d2 = findPrime(n + d1)
	print(d1 * d2)
