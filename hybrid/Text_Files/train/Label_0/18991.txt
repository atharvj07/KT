int MIN(int a, int b) {
  int x = (b > a) ? a : b;
  return x;
}
int MAX(int a, int b) {
  int x = (b < a) ? a : b;
  return x;
}

      long long curr = 0;
      str[0] = a + '0';
      for (int i = 1; i < n; i++) {
        str[i] = ((m * (str[i - 1] - '0') + curr) % 10 + 10) % 10 + '0';

      }
      if (curr + m * (str[n - 1] - '0') == a && str[n - 1] != '0') {
