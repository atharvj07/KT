#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=2e5+5;
int n,a[N],dp[N],ans[N];
vector<int>G[N];

void dfs(int u,int p,int R){
    int l=0,r=R;
    while(r-l>1){
        int mid=l+r>>1;
        if(dp[mid]<a[u]) l=mid;
        else r=mid;
    }
    int pre=dp[r];
    dp[r]=a[u];
    ans[u]=R==r?R++:R-1;
    for(int v:G[u]) if(v!=p) dfs(v,u,R);
    dp[r]=pre;
}

int main()
{
   cin>>n;
   for(int i=0;i<n;i++) cin>>a[i];
   for(int i=1;i<n;i++){
        int u,v;
        cin>>u>>v;
        u--,v--;
        G[u].push_back(v);
        G[v].push_back(u);
   }
   fill(dp,dp+n+1,2e9);
   dfs(0,-1,1);
   for(int i=0;i<n;i++) cout<<ans[i]<<endl;
    return 0;
}
