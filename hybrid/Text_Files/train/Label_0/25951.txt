#include <bits/stdc++.h>
using namespace std;
long long MUL(long long A, long long B, long long MOD) {
  long long r = 0;
  while (B) {
    r = (r + (A * (B % 2) % MOD)) % MOD;
    A = (A + A) % MOD;
    B /= 2;
  }
  return r;
}
long long POW(long long A, long long B, long long MOD) {
  long long RES = 1;
  A %= MOD;
  assert(B >= 0);
  for (; B; B >>= 1) {
    if (B & 1) RES = MUL(RES, A, MOD);
    A = MUL(A, A, MOD);
  }
  return RES;
}
long long GCD(long long A, long long B) { return B == 0 ? A : GCD(B, A % B); }
int dx[] = {0, -1, 0, 1, 0};
int dy[] = {0, 0, 1, 0, -1};
const int oo = 1e9 + 7;
const int MAXN = 1e6 + 100;
long long N, M, cnt, tplt, S;
vector<pair<int, int> > G[MAXN], H[MAXN];
vector<pair<int, pair<int, int> > > Edges;
int Low[MAXN], Num[MAXN], root[MAXN];
long long C[MAXN];
stack<int> st;
void DFS(int u) {
  Low[u] = Num[u] = ++cnt;
  st.push(u);
  for (vector<pair<int, int> >::iterator it = G[u].begin(); it != G[u].end();
       ++it) {
    int v = (*it).second;
    if (Num[v]) {
      Low[u] = min(Low[u], Num[v]);
    } else {
      DFS(v);
      Low[u] = min(Low[u], Low[v]);
    }
  }
  if (Num[u] == Low[u]) {
    ++tplt;
    int v;
    do {
      v = st.top();
      st.pop();
      root[v] = tplt;
      Num[v] = Low[v] = +oo;
    } while (u != v);
  }
}
long long cal(long long w) {
  long long l = -1, r = w + 1;
  while (r - l > 1) {
    long long mid = (l + r) / 2;
    long long sum = (mid + 1) * mid / 2;
    if (sum <= w)
      l = mid;
    else
      r = mid;
  }
  return w * (l + 1) - l * (l + 1) * (l + 2) / 6;
}
int memo[MAXN];
long long solve(int u) {
  if (!H[u].size()) return 0;
  if (memo[u] != -1) return memo[u];
  long long cur = 0ll;
  for (vector<pair<int, int> >::iterator it = H[u].begin(); it != H[u].end();
       ++it) {
    long long w = (*it).first;
    int v = (*it).second;
    cur = max(cur, solve(v) + w + C[v]);
  }
  return memo[u] = cur;
}
bool visit[MAXN];
vector<int> L;
void dfs(int u) {
  visit[u] = 1;
  for (vector<pair<int, int> >::iterator it = H[u].begin(); it != H[u].end();
       ++it) {
    int v = (*it).second;
    if (!visit[v]) dfs(v);
  }
  L.push_back(u);
}
long long dp[MAXN];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> N >> M;
  for (int i = 1; i <= M; ++i) {
    int u, v, w;
    cin >> u >> v >> w;
    G[u].push_back(pair<int, int>(w, v));
    Edges.push_back(pair<int, pair<int, int> >(w, pair<int, int>(u, v)));
  }
  cin >> S;
  for (int i = 1; i <= N; ++i) {
    if (!Num[i]) DFS(i);
  }
  for (vector<pair<int, pair<int, int> > >::iterator it = Edges.begin();
       it != Edges.end(); ++it) {
    int u = (*it).second.first;
    int v = (*it).second.second;
    int w = (*it).first;
    if (root[u] == root[v])
      C[root[u]] += cal(w);
    else
      H[root[u]].push_back(pair<int, int>(w, root[v]));
  }
  long long ANS = 0;
  S = root[S];
  dfs(S);
  reverse(L.begin(), L.end());
  dp[S] = C[S];
  for (vector<int>::iterator it = L.begin(); it != L.end(); ++it) {
    int u = *it;
    for (int i = 0; i < H[u].size(); ++i) {
      long long w = H[u][i].first;
      int v = H[u][i].second;
      dp[v] = max(dp[v], dp[u] + C[v] + w);
      ANS = max(ANS, dp[v]);
    }
    ANS = max(ANS, dp[u]);
  }
  cout << ANS;
  return 0;
}
