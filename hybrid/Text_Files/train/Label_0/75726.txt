(N, M) = map(int, input().split())
T = [[] for i in range(M)]
num = -float('inf')
ma = 0
for i in range(N):
	(s, r) = map(int, input().split())
	T[s - 1].append(r)
for i in range(M):
	T[i] = sorted(T[i], reverse=True)
	ma = max(ma, len(T[i]))
num = [0] * (N + 1)
for j in range(M):
	tmp = 0
	for i in range(len(T[j])):
		tmp += T[j][i]
		if tmp < 0:
			break
		num[i] += tmp
print(max(num))
