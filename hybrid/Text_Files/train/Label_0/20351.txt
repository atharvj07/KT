#include<bits/stdc++.h>
#define ll long long
using namespace std;

const int mxn = 100005;
string s;
int main()
{
    ios_base::sync_with_stdio(0),cin.tie(0);cout.tie(0);
    int tc; cin >> tc; while(tc--)
    {
        cin >> s;
        int a=0, b=0, sm=0;
        for(int i=0; i<s.size(); i++)
        {
            if(s[i]=='(')a++;
            else if(s[i]=='[')b++;
            else if(s[i]==')' && a>0)
            {
                a--, sm++;
            }
            else if(s[i]==']' && b>0)
            {
                b--, sm++;
            }
        }
        cout << sm << "\n";
    }
    return 0;
}
