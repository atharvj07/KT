import java.io.*;
import java.util.*;
import java.math.*;
 
public class Main
{
	static final int MOD = 1000000007;
 
	static class BIT {
		int[] v;
		int n;
		
		BIT( int _n ) {
			n = _n;
			v = new int[ n + 1 ];
			Arrays.fill( v, 0 );
		}
		
		void rupdate( int l, int r, int val ) {
			pupdate( l, val );
			pupdate( r, MOD - val );
		}
		
		void pupdate( int x, int val )  {
			if ( x <= 0 || x > n ) {
				return;
			}
			while ( x <= n ) {
				v[x] = ( v[x] + val ) % MOD;
				x += x &  - x;
			}
		}
		
		int pquery( int x ) {
			return rquery( x );
		}
		
		int rquery( int x ) {
			int res = 0;
			while ( x > 0 ) {
				res = ( res + v[x] ) % MOD;
				x -= x &  - x;
			}
			return res;
		}
	}
	
	static int relabel( ArrayList<ArrayList<Integer>> child, int[] l, int[] r, int x, int cnt, int[] d, int dep ) {
		l[x] = cnt++;
		d[x] = dep;
		for ( int c : child.get( x ) ) {
			cnt = relabel( child, l, r, c, cnt, d, dep + 1 );
		}
		r[x] = cnt;
		return cnt;
	}
	
	public static void main( String[] args ) {
		Scan scan = new Scan();
		PrintWriter pw = new PrintWriter(System.out);
		int n = scan.nextInt();
		BIT bit = new BIT( n );
		BIT bit2 = new BIT( n );
		ArrayList<ArrayList<Integer>> child = new ArrayList<>();
		for(int i = 0; i <= n; i++) {
			child.add( new ArrayList<>() );
		}	
		for ( int i = 2; i <= n; i++ ) {
			child.get( scan.nextInt() ).add( i );
		}
		int[] l = new int[ n + 1 ];
		int[] r = new int[ n + 1 ];
		int[] d = new int[ n + 1 ];
		relabel( child, l, r, 1, 1, d, 0 );
		for ( int q = scan.nextInt(); q > 0; q-- ) {
			if ( scan.nextInt() == 1 ) {
				int v = scan.nextInt();
				int x = scan.nextInt();
				int k = scan.nextInt();
				bit.rupdate( l[v], r[v], ( int ) ( ( x + ( long ) k * d[v] ) % MOD ) );
				bit2.rupdate( l[v], r[v], MOD - k );
			} else {
				int v = scan.nextInt();
				pw.println(( bit.pquery( l[v] ) + ( long ) d[v] * bit2.pquery( l[v] ) ) % MOD );
			}
		}
		pw.close();
	}
}
 
class Scan {
	BufferedReader br;
	StringTokenizer st;
	
	Scan() {
		br = new BufferedReader( new InputStreamReader( System.in ) );
	}
	
	boolean hasNext() {
		while ( st == null ||  ! st.hasMoreElements() ) {
			try {
				st = new StringTokenizer( br.readLine() );
			} catch ( Exception e ) {
				return false;
			}
		}
		return true;
	}
	
	int nextInt() {
		return hasNext() ? Integer.parseInt( st.nextToken() ) : -1;
	}
	
	String next() {
		return hasNext() ? st.nextToken() : null;
	}
}
