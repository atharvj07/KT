import java.util.*;

public class C {
	public static void main(String [] args){
		final Scanner s = new Scanner(System.in);
		long a = s.nextLong();
		long b = s.nextLong();
		long res = 0;
		while (b != 0){
			res += a/b;
			final long a1 = b;
			final long b1 = a%b;
			a = a1;
			b = b1;
		}
		System.out.println(res);
		s.close();
	}
}
