s = list(map(int, input().split()))
(n, k) = (s[0], s[1])
s = list(map(int, input().split()))
max = 0
for i in range(1, n + 1):
	p = [0] + s
	for j in range(i, 0, -1 * k):
		p[j] = 0
	for j in range(i, n + 1, k):
		p[j] = 0
	e = p.count(1)
	lol = p.count(-1)
	if abs(e - lol) > max:
		max = abs(e - lol)
print(max)
