import sys
isPrime = [True] * int(1000000.0 + 2)

def solve():
	inp = sys.stdin.readline
	(n, e) = map(int, inp().split())
	a = list(map(int, inp().split()))
	w = [True] * n
	r = 0
	for i in range(n):
		if w[i]:
			j = 0
			z = i
			last = -1
			last1 = -1
			p = False
			while z < n:
				w[z] = False
				v = a[z]
				if v > 1:
					(last, last1) = (j, last)
					p = isPrime[v]
					if p:
						r -= 1
				if p:
					r += last - last1
				z += e
				j += 1
	print(r)

def main():
	M = len(isPrime)
	isPrime[1] = False
	isPrime[0] = False
	i = 2
	while i * i < M:
		if isPrime[i]:
			j = i * i
			while j < M:
				isPrime[j] = False
				j += i
		i += 1
	for i in range(int(sys.stdin.readline())):
		solve()
main()
