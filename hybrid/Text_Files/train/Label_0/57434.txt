
import java.io.*;
import java.util.Arrays;

public class _Rough {

    public static void main(String[] args) throws IOException {

        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter out = new BufferedWriter(new OutputStreamWriter(System.out));

        String[] s = in.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        long k = Long.parseLong(s[1]);
        s = in.readLine().split(" ");
        long[] ar = new long[n];
        for (int i = 0; i < n; i++) {
            ar[i] = Integer.parseInt(s[i]);
        }


        out.write(sol(ar, k) + "");
        out.flush();
    }

    private static long sol(long[] ar, long k) {

        int n = ar.length;
        Arrays.sort(ar);

        for (int i = 1; i <= n / 2; i++) {

            int minIx = i;
            int maxIx = n - minIx;
            long gap = ((ar[minIx] - ar[minIx - 1]) + (ar[maxIx] - ar[maxIx - 1])) * (minIx);

            if (gap <= k) {
                k -= gap;
            } else {
                return ar[maxIx] - ar[minIx - 1] - k / minIx;
            }
        }

        return 0;

    }
}
