import java.util.*;

public class helloWorld 
{
	public static void main(String[] args) 
	{		
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();
		int[] ar = new int[n];
		int min = 0, max = 0, sum = 0;
		
		for(int i = 0; i < n; i++) {
			ar[i] = in.nextInt();
			max = Math.max(max, ar[i]);
			sum += ar[i];
		}
		int a = m;
		m = Math.max(m - max*n + sum, 0);
		min = max + (int) Math.ceil(1.0 * m / n);
		max += a;
		
		System.out.println(min + " " + max);

		in.close();
	}
}
