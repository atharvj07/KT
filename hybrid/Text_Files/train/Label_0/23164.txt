#include <bits/stdc++.h>
using namespace std;
string C[8];
void bfs(int i, int j, int t) {
  if (t > 8) {
    cout << "WIN";
    exit(0);
  }
  for (int x = -1; x <= 1; x++)
    for (int y = -1; y <= 1; y++) {
      if (j + y < 0 || i + x < 0 || j + y > 7 || i + x > 7) continue;
      if (i + x - t >= 1)
        if (C[i + x - t - 1][j + y] == 'S') continue;
      if (i + x - t >= 0)
        if (C[i + x - t][j + y] == 'S') continue;
      bfs(i + x, j + y, t + 1);
    }
}
int main() {
  for (int i = 0; i < 8; i++) cin >> C[i];
  bfs(7, 0, 0);
  cout << "LOSE";
}
