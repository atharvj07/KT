import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class C {
	static ArrayList<Integer>[] lst;
	static ArrayList<ArrayList<Integer>> all;
	static int[] deb;
	static int min = 10000000;
	static String mins;
	static int[] s;
	int dp[][] = new int[200][200];

	@SuppressWarnings("unchecked")
	public static void main(String[] args) throws IOException {
		InputReader myScanner = new InputReader();
		int n = myScanner.nextInt();
		int[] comb = new int[n];
		lst = new ArrayList[n];
		for (int i = 0; i < n; i++)
			lst[i] = new ArrayList<Integer>();

		for (int i = 0; i < n; i++)
			comb[i] = myScanner.nextInt() - 1;

		int[] tmp = new int[n];

		for (int i = 0; i < comb.length; i++) {
			int nn = myScanner.nextInt();
			for (int j = 0; j < nn; j++)
				lst[myScanner.nextInt() - 1].add(i);
			tmp[i] = nn;
		}

		// ==============================================
		int min = Integer.MAX_VALUE;

		for (int start = 0; start < 3; start++) {
			int currLoad = 0;
			deb = tmp.clone();

			Queue<Integer>[] q = new LinkedList[3];
			for (int i = 0; i < q.length; i++)
				q[i] = new LinkedList<Integer>();

			for (int i = 0; i < lst.length; i++)
				if (deb[i] == 0)
					q[comb[i]].add(i);

			int currI = start;

			while (!q[currI].isEmpty()) {
				int curr = q[currI].poll();
				currLoad++;

				for (int i : lst[curr]) {
					deb[i]--;
					if (deb[i] == 0)
						q[comb[i]].add(i);
				}

				if (q[currI].isEmpty()) {
					if (currI == 0) {
						if (!q[1].isEmpty()) {
							currI = 1;
							currLoad++;
						} else if (!q[2].isEmpty()) {
							currI = 2;
							currLoad += 2;
						}
					} else if (currI == 1) {
						if (!q[2].isEmpty()) {
							currI = 2;
							currLoad++;
						} else if (!q[0].isEmpty()) {
							currI = 0;
							currLoad += 2;
						}
					} else if (currI == 2) {
						if (!q[0].isEmpty()) {
							currI = 0;
							currLoad++;
						} else if (!q[1].isEmpty()) {
							currI = 1;
							currLoad += 2;
						}
					}
				}
			}
			if (currLoad > 0)
				min = Math.min(min, currLoad);
		}
		System.out.println(min);

	}

	// ========================================

	static class InputReader {
		BufferedReader buff;
		StringTokenizer tok;
		String cur;

		public InputReader() throws IOException {
			buff = new BufferedReader(new InputStreamReader(System.in));
			tok = new StringTokenizer(cur = buff.readLine());
		}

		public boolean hasNext() throws IOException {
			if (!tok.hasMoreElements()) {
				cur = buff.readLine();
				if (cur == null)
					return false;
				tok = new StringTokenizer(cur);
			}
			return true;
		}

		public String next() throws IOException {
			while (!tok.hasMoreElements())
				tok = new StringTokenizer(cur = buff.readLine());

			return tok.nextToken();
		}

		public int nextInt() throws NumberFormatException, IOException {
			while (!tok.hasMoreElements())
				tok = new StringTokenizer(cur = buff.readLine());

			return Integer.parseInt(next());
		}

		public long nextLong() throws NumberFormatException, IOException {
			while (!tok.hasMoreElements())
				tok = new StringTokenizer(cur = buff.readLine());

			return Long.parseLong(next());
		}
	}

}
