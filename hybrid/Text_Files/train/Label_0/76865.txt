from bisect import bisect_left

def read():
	return [int(x) for x in input().split()]
n = read()
a = read()
vec = [list() for i in range(max(a) + 1)]
for i in range(len(a)):
	vec[a[i]].append(i)

def solve():
	ans = 0
	dis = 1
	p = 0
	i = 0
	while i < len(vec):
		if len(vec[i]):
			pos = bisect_left(vec[i], p)
			ans += (len(vec[i]) - pos) * dis
			p = vec[i][len(vec[i]) - 1]
			del vec[i][pos:len(vec[i])]
			if len(vec[i]):
				p = 0
				dis += 1
				i -= 1
		i += 1
	return ans
print(solve())
