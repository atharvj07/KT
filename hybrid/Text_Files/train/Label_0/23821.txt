#include <bits/stdc++.h>
using namespace std;
string s1, s2;
int gcd(int a, int b) { return (a % b == 0) ? b : gcd(b, a % b); }
int main() {
  int len;
  int ans = 0;
  cin >> s1 >> s2;
  len = gcd(s1.size(), s2.size());
  for (int i = 0; i < s1.size(); i++) {
    if (s1[i] != s1[i % len]) {
      cout << 0;
      return 0;
    }
  }
  for (int i = 0; i < s2.size(); i++) {
    if (s2[i] != s1[i % len]) {
      cout << 0;
      return 0;
    }
  }
  s1 = s1.substr(0, len);
  for (int i = 1; i <= len; i++) {
    if (len % i == 0) {
      bool ok = true;
      for (int j = 0; j < len; j++) {
        ok = ok && s1[j] == s1[j % i];
      }
      if (ok) ans++;
    }
  }
  cout << ans;
  return 0;
}
