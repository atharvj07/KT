#include <bits/stdc++.h>
using namespace std;

typedef long long lli;

const lli INF = 1LL<<60;

int main() {
  int N;
  cin >> N;
  vector<lli> a(N);
  for(int i = 0; i < N; ++i) {
    cin >> a[i];
  }
  a.push_back(INF);
  sort(a.begin(), a.end());
  lli sum = 0;
  for(int i = 0; i < a.size(); ++i) {
    if(sum+1 < a[i]) {
      cout << sum+1 << endl;
      break;
    }
    sum += a[i];
  }
  return 0;
}
