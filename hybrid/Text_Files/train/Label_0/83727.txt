(n, m) = map(int, input().split())
t = list(input())
g = t[::-1]
if m == 1:
	for j in range(n):
		if 0 <= j < n - 1:
			print('PRINT ', t[j])
			print('RIGHT')
		else:
			print('PRINT ', t[j])
elif m == n:
	for k in range(n):
		if 0 <= k < n - 1:
			print('PRINT ', g[k])
			print('LEFT')
		else:
			print('PRINT ', g[k])
elif m - 1 < n - m:
	for x in range(m - 1):
		print('LEFT')
	for j in range(n):
		if 0 <= j < n - 1:
			print('PRINT ', t[j])
			print('RIGHT')
		else:
			print('PRINT ', t[j])
else:
	for aw in range(n - m):
		print('RIGHT')
	for k in range(n):
		if 0 <= k < n - 1:
			print('PRINT ', g[k])
			print('LEFT')
		else:
			print('PRINT ', g[k])
