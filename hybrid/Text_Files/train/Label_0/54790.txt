import java.util.*;
import java.io.*;
public class ParkingLot
{
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		
		int n = in.nextInt();
		int m = in.nextInt();
		Space.n = n;
		
		PriorityQueue<Space> lot = new PriorityQueue<Space>();
		lot.add(new Space(0, n + 1));
		
		Space[] leftInterval = new Space[n + 2];
		Space[] rightInterval = new Space[n + 2];
		
		int[] ids = new int[1000001];
		
		for(int x = 0; x < m; x++)
		{
			int t = in.nextInt();
			int id = in.nextInt();
			
			if(t == 1)
			{
				Space best = lot.remove();
				
				while(best.used)
				{
					best = lot.remove();
				}
				
				System.out.println(best.index);
				
				Space l = new Space(best.left, best.index);
				lot.add(l);
				leftInterval[best.index] = l;
				rightInterval[best.left] = l;
				
				Space r = new Space(best.index, best.right);
				lot.add(r);
				rightInterval[best.index] = r;
				leftInterval[best.right] = r;
				
				ids[id] = best.index;
			}
			else
			{
				int spot = ids[id];
				
				leftInterval[spot].used = true;
				rightInterval[spot].used = true;
				
				Space open = new Space(leftInterval[spot].left, rightInterval[spot].right);
				
				rightInterval[leftInterval[spot].left] = open;
				leftInterval[rightInterval[spot].right] = open;
				
				lot.add(open);
			}
		}
	}
}

class Space implements Comparable<Space>
{
	public static int n;
	
	int left;
	int right;
	int dist;
	int index;
	boolean used;
	
	public Space(int l, int r)
	{
		left = l;
		right = r;
		
		dist = (right - left) / 2;
		index = (left + ((right - left) / 2));
		
		if(left == 0)
		{
			dist = right - 1;
			index = 1;
		}
		else if(right == n + 1)
		{
			dist = n - left;
			index = n;
		}
		
		used = false;
	}
	
	public int compareTo(Space o)
	{
		if(dist != o.dist)
		{
			return o.dist - dist;
		}
		else
		{
			return index - o.index;
		}
	}
}
