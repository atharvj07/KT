from sys import stdin, stdout
(n, m) = map(int, stdin.readline().split())
mtx = [[ch for ch in stdin.readline()] for _ in range(n)]
groups = [[0 for _ in range(m)] for _ in range(n)]
sizes = [0]
shard_id = 1
for i in range(n):
	for j in range(m):
		if mtx[i][j] == '*':
			continue
		if groups[i][j] != 0:
			continue
		stack = [(i, j)]
		size = 1
		groups[i][j] = shard_id
		while stack:
			(x, y) = stack.pop()
			for (xi, yi) in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
				if xi < 0 or n <= xi:
					continue
				if yi < 0 or m <= yi:
					continue
				if mtx[xi][yi] != '.':
					continue
				if groups[xi][yi] != 0:
					continue
				groups[xi][yi] = shard_id
				stack.append((xi, yi))
				size += 1
		sizes.append(size)
		shard_id += 1
for i in range(n):
	for j in range(m):
		if mtx[i][j] != '*':
			continue
		groups_id = set()
		for (xi, yi) in ((i + 1, j), (i - 1, j), (i, j + 1), (i, j - 1)):
			if xi < 0 or n <= xi:
				continue
			if yi < 0 or m <= yi:
				continue
			if mtx[xi][yi] != '.':
				continue
			groups_id.add(groups[xi][yi])
		size = 1
		for gid in groups_id:
			size += sizes[gid]
		mtx[i][j] = str(size)[-1]
stdout.write('\n'.join((''.join(row) for row in mtx)))
