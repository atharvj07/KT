import sys
amount = int(input())
max_1 = -1
max_2 = -1
ans = ''
for i in range(amount):
	inputs = [s for s in input().split()]
	if inputs[0] == '+':
		if max_1 < min(int(inputs[1]), int(inputs[2])):
			max_1 = min(int(inputs[1]), int(inputs[2]))
		if max_2 < max(int(inputs[1]), int(inputs[2])):
			max_2 = max(int(inputs[1]), int(inputs[2]))
	elif max_1 <= int(inputs[1]) and max_2 <= int(inputs[2]) or (max_1 <= int(inputs[2]) and max_2 <= int(inputs[1])):
		ans += 'YES\n'
	else:
		ans += 'NO\n'
print(ans)
