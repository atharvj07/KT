#include <bits/stdc++.h>
using namespace std;
const int maxN(1e5 + 10);
const int inf(1e9);
const long long mod(1e9 + 7);
template <typename T1>
inline void in(T1 &a1) {
  cin >> a1;
}
template <typename T1, typename T2>
inline void in(T1 &a1, T2 &a2) {
  cin >> a1 >> a2;
}
template <typename T1, typename T2, typename T3>
inline void in(T1 &a1, T2 &a2, T3 &a3) {
  cin >> a1 >> a2 >> a3;
}
struct Compress {
  int l, lr, r, rl;
};
vector<Compress> A;
struct Query {
  int l, r, i;
};
vector<Query> query;
long long sum;
vector<int> cnt, t;
vector<long long> ans, p, idx;
long long pp(int i) { return i ? p[i - 1] : 0; }
int id(long long x) {
  return lower_bound(idx.begin(), idx.end(), x) - idx.begin();
}
void addL(int i) { sum += cnt[A[i].lr], ++cnt[A[i].l]; }
void addR(int i) { sum += cnt[A[i].rl], ++cnt[A[i].r]; }
void subL(int i) { --cnt[A[i].l], sum -= cnt[A[i].lr]; }
void subR(int i) { --cnt[A[i].r], sum -= cnt[A[i].rl]; }
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int i, n, k, q, l, r;
  in(n, k);
  t = vector<int>(n), p = vector<long long>(n);
  for (i = 0; i < n; i++) in(t[i]), t[i] = 3 - 2 * t[i];
  for (i = 0; i < n; i++) in(q), p[i] = q * t[i] + pp(i);
  idx.push_back(0), idx.push_back(k);
  for (i = 0; i < n; i++) {
    idx.push_back(p[i]);
    idx.push_back(p[i] - k);
    idx.push_back(p[i] + k);
  }
  sort(idx.begin(), idx.end());
  idx.erase(unique(idx.begin(), idx.end()), idx.end());
  for (i = 0; i < n; i++)
    A.push_back({id(pp(i)), id(pp(i) + k), id(p[i]), id(p[i] - k)});
  for (in(q), i = 0; i < q; i++) in(l, r), query.push_back({l - 1, r - 1, i});
  int sn = sqrt(n);
  sort(query.begin(), query.end(), [&sn](const Query &u, const Query &v) {
    return u.l / sn - v.l / sn ? u.l < v.l : u.r > v.r;
  });
  cnt = vector<int>(3 * n);
  ans = vector<long long>(((int)(query.size())));
  addL(l = 0), addR(r = 0);
  for (auto &e : query) {
    while (l < e.l) subL(l++);
    while (l > e.l) addL(--l);
    while (r < e.r) addR(++r);
    while (r > e.r) subR(r--);
    ans[e.i] = sum;
  }
  for (i = 0; i < q; i++) printf("%lld\n", ans[i]);
}
