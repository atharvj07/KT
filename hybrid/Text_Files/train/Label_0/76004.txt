for tc in range(int(input())):
	(n, k) = input().split(' ')
	(al, ar) = input().split(' ')
	(bl, br) = input().split(' ')
	(n, k, al, ar, bl, br) = (int(n), int(k), int(al), int(ar), int(bl), int(br))
	if al > bl:
		(al, ar, bl, br) = (bl, br, al, ar)
	if ar < bl:
		blank = bl - ar
		get = br - al
		need = blank + get
		if k < get:
			print(blank + k)
		elif k <= n * get:
			print(int(k / get) * need + min(k % get, blank) + k % get)
		else:
			print(n * need + (k - n * get) * 2)
	else:
		get = max(br, ar) - al
		init = n * (min(br, ar) - bl)
		if k <= init:
			print(0)
		elif k <= n * get:
			print(k - init)
		else:
			print(n * get - init + 2 * (k - n * get))
