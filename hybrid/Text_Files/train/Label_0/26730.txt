#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  t = 1;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long a[n];
    long long od1 = 0, od2 = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      if (a[i] % 2 == 0) od1++;
    }
    long long m;
    cin >> m;
    long long b[m];
    for (int i = 0; i < m; i++) {
      cin >> b[i];
      if (b[i] % 2 == 0) od2++;
    }
    long long ans = 0;
    ans += (od1 * od2 + (n - od1) * (m - od2));
    cout << ans << "\n";
  }
  return 0;
}
