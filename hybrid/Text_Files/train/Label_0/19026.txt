using namespace std;
namespace IO {
const int L = 1 << 15;
char ibuf[L | 1], *iS = ibuf, *iT = ibuf, obuf[L | 1], *oS = obuf,
                  *oT = obuf + L, c, st[66];
int tp = 0, f;
inline char gc() {
  if (iS == iT) iT = (iS = ibuf) + fread(ibuf, sizeof(char), L, stdin);
  return (*iS++);
}
inline void flush() {
  fwrite(obuf, sizeof(char), oS - obuf, stdout);
  oS = obuf;
}
inline void pc(char c) {
  *oS++ = c;
  if (oS == oT) flush();
}
inline long long read() {
  long long x = 0;
  f = 1, c = getchar();
  for (; !isdigit(c); c = getchar())
    if (c == '-') f = -1;
  for (; isdigit(c); c = getchar()) x = (x << 1) + (x << 3) + (c & 15);
  return x * f;
}
inline void write(long long x) {
  if (!x) putchar('0');
  if (x < 0) putchar('-'), x = -x;
  while (x) st[++tp] = x % 10 + '0', x /= 10;
  while (tp) putchar(st[tp--]);
}
inline void writeln(long long x) {
  write(x);
  putchar('\n');
}
inline void wri(long long x) { write(x), putchar(' '); }
struct IOflusher {
  ~IOflusher() { flush(); }

}  // namespace IO
using IO::read;

const int maxn = 1e3 + 13;
struct point {
  double x, y;
  point() {}
  point(double x, double y) : x(x), y(y) {}
} p[maxn];
const double pi = acos(-1.0);
double dist(point a, point b) {
  return sqrt(((a.x - b.x) * (a.x - b.x)) + ((a.y - b.y) * (a.y - b.y)));
}
pair<double, int> A[maxn * 4];
int c;
int sum;
void add(double s, double t, int v = 1) {
  if (s <= 0)
    sum += v;
  else
    A[++c] = {s, v};
  if (t < 2 * pi) A[++c] = {t + 1e-12, -v};
}
double w, h;
int n, m;
double _d[maxn], _a[maxn];
bool check(int x, double r) {
  sum = c = 0;
  for (register int i = (1); i <= int(n); ++i)
    if (dist(p[i], p[x]) > 0 && dist(p[i], p[x]) < 2 * r) {
      double r1 = acos(dist(p[i], p[x]) / 2 / r);
      double s1 = atan2(p[i].y - p[x].y, p[i].x - p[x].x);
      double l = s1 - r1, r = s1 + r1;
      while (l < 0) l += 2 * pi;
      while (l >= 2 * pi) l -= 2 * pi;
      while (r < 0) r += 2 * pi;
      while (r >= 2 * pi) r -= 2 * pi;
      if (l <= r)
        add(l, r);
      else
        add(0, r), add(l, 2 * pi);
    }
  double t;
  if (p[x].x < r) {
    t = acos(p[x].x / r);
    add(pi - t, pi + t, 2);
  }
  if (p[x].y < r) {
    t = acos(p[x].y / r);
    add(1.5 * pi - t, 1.5 * pi + t, 2);
  }
  if (p[x].x + r > w) {
    t = acos((w - p[x].x) / r);
    add(0, t, 2);
    add(2 * pi - t, 2 * pi, 2);
  }
  if (p[x].y + r > h) {
    t = acos((h - p[x].y) / r);
    add(pi / 2 - t, pi / 2 + t, 2);
  }
  if (c < 2) return true;
  sort(A + 1, A + 1 + c);
  if (sum < 2 && A[1].first > 1e-9) return true;
  for (register int j = (1); j <= int(c); ++j) {
    sum += A[j].second;
    if (sum < 2) return true;
  }
  if (sum < 2 && A[c].first < 2 * pi - 1e-9) return true;
  return false;
}
int main() {
  double L = 0, R;
  w = read(), h = read(), n = read();
  for (register int i = (1); i <= int(n); ++i) p[i].x = read(), p[i].y = read();
  random_shuffle(p + 1, p + 1 + n);
  for (register int i = (1); i <= int(n); ++i) {
    if (!check(i, L + 1e-9)) continue;
    R = 2e6;
    while (R - L > 1e-9) {
      double mid = (L + R) / 2.0;
      if (check(i, mid))
        L = mid;
      else
        R = mid;
    }
  }
  printf("%.15lf\n", L);
  return 0;
}
