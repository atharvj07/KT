n = int(input())
l = list(map(int, input().split(' ')))
l.sort()
lonly = set(l)
lonly = list(lonly)
lonly.sort()
i = 0
lenonly = len(lonly)
res1 = 0
while i < lenonly:
	if i - 1 >= 0 and lonly[i] - lonly[i - 1] == 1:
		lonly[i] = lonly[i - 1]
	elif i + 1 < lenonly:
		dif = lonly[i + 1] - lonly[i]
		if dif == 1:
			lonly[i] = lonly[i + 1]
			i += 1
		elif dif == 2:
			lonly[i] = lonly[i + 1] = lonly[i] + 1
			i += 1
	i += 1
lonly = set(lonly)
print(len(lonly), end=' ')
last = -1
l2 = set()
for i in l:
	if i == last:
		last = i + 1
		l2.add(last)
	elif i > last:
		if i - 1 != last:
			last = i - 1
			l2.add(last)
		else:
			last = i
			l2.add(last)
print(len(l2))
