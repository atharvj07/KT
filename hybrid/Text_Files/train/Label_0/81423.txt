def exgcd(a, b):
	if b == 0:
		return (a, 1, 0)
	(d, y, x) = exgcd(b, a % b)
	y -= a // b * x
	return (d, x, y)
n = int(input())
arr = [int(x) for x in input().split(' ')]
(ge, al, be) = exgcd(n, n - 1)
if n == 1:
	print(1, 1)
	print(-1 * arr[0])
	print('1 1\n0\n1 1\n0')
else:
	print(1, n)
	print(' '.join([str(-1 * i * al * n) for i in arr]))
	print(1, n - 1)
	print(' '.join([str(-1 * i * be * (n - 1)) for i in arr[:-1]]))
	print(2, n)
	print(' '.join(['0' for i in range(n - 2)] + [str(-1 * arr[-1] * be * (n - 1))]))
