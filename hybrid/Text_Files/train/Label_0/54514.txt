import java.util.Scanner;

public class D {

    public static void main(String[] args) {
        D solver = new D();
        solver.solve();
    }

    private void solve() {
        Scanner sc = new Scanner(System.in);
//        sc = new Scanner("-++-\n");
//        sc = new Scanner("+-\n");
//        sc = new Scanner("-\n");
//        sc = new Scanner("--++\n");

        String line = sc.nextLine().trim();

        int twist = 0;
        for (int i = 0; i < line.length(); i++) {
            char c = line.charAt(i);
            if (i % 2 == 0) {
                if (c == '-') twist++;
                if (c == '+') twist--;
            } else {
                if (c == '-') twist--;
                if (c == '+') twist++;
            }
        }

        System.out.println(twist == 0 ? "Yes" : "No");
    }

}
