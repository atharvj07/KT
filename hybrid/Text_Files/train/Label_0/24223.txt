#include <bits/stdc++.h>
using namespace std;
vector<pair<int,int>> v[16];
bool b[16];
int n,ma=0;

void h1(int i){
	bool c[16];
	int o=1,count=0;
	for(int j=1;j<=n;j++){
		if(b[j]==false)
		continue;
		count++;
		for(auto k:v[j]){
			if((k.second==0&&b[k.first]==true)||(k.second==1&&b[k.first]==false))
			o=0;
		}
	}
	if(o==1)
	ma=max(ma,count);
	if(i>n)
	return;
	b[i]=true;
	h1(i+1);
	b[i]=false;
	h1(i+1);
}

int main() {
	int count=0;
	cin>>n;
	memset(b,false,sizeof(b));
	for(int i=1;i<=n;i++){
		v[i].clear();
		int a,b,c;
		cin>>a;
		while(a--){
			cin>>b>>c;
			v[i].push_back({b,c});
		}
	}
	h1(1);
	cout<<ma;
	
	return 0;
}
