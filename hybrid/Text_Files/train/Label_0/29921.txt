#include <bits/stdc++.h>
using namespace std;
void solve() {
  int n, m;
  cin >> n >> m;
  int arr[n][2 * m];
  int cnt = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 2 * m; j++) {
      cin >> arr[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 1; j < 2 * m; j += 2) {
      if (arr[i][j] == 1 || arr[i][j - 1] == 1) {
        ans++;
      }
    }
  }
  cout << ans << "\n";
}
int main() { solve(); }
