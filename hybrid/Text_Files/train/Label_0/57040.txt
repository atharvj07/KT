
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int rows = input.nextInt();
        int cols = input.nextInt();

        String arr[] = new String[rows];

        for (int i = 0; i < arr.length; i++) {
            arr[i] = input.next();
        }
        
        
        int first = cols, last = 0;
        int  lr = arr.length-1 , fr=-1; 
         boolean started = false;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i].contains("*")) {
                started = true ; 
                first = Math.min(first, arr[i].indexOf("*"));
                last = Math.max(last, arr[i].lastIndexOf("*"));
                lr = i ;
            }
            
            if (started && fr ==-1){
                fr = i;
            }
            
            
        }

       
        for (int i = fr; i <= lr; i++) {
            if (arr[i].contains("*")) {
                started = true;
            }
                System.out.println(arr[i].substring(first, last + 1));
            
        }
        
        

    }

}
