#include <bits/stdc++.h>
using namespace std;
const int MaxN = 1000100;
const int Sum = 1000001;
int N, Ans[MaxN], Tot, A[MaxN];
bool Mark[MaxN];
int main() {
  int i, Cnt = 0;
  scanf("%d", &N);
  for (i = 1; i <= N; i++) {
    scanf("%d", &A[i]);
    Mark[A[i]] = 1;
  }
  for (i = 1; i <= N; i++)
    if (Mark[Sum - A[i]] == 0)
      Ans[++Tot] = Sum - A[i];
    else
      Cnt++;
  for (i = 1; (i < Sum) && Cnt; i++)
    if ((Mark[i] == 0) && (Mark[Sum - i] == 0)) {
      Cnt -= 2;
      Ans[++Tot] = i;
      Ans[++Tot] = Sum - i;
    }
  printf("%d\n", Tot);
  for (i = 1; i <= Tot; i++) printf("%d ", Ans[i]);
  return 0;
}
