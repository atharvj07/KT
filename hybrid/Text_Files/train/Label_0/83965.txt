for _ in range(int(input())):
	(n, a, b) = list(map(int, input().split()))
	if n == 1 or b == 1:
		print('Yes')
	elif a == 1:
		if (n - 1) % b == 0:
			print('Yes')
		else:
			print('No')
	else:
		ans = 'No'
		x = 1
		while x <= n:
			if (n - x) % b == 0:
				ans = 'Yes'
				break
			else:
				x *= a
		print(ans)
