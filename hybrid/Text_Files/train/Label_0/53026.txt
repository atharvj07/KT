import java.util.*;
import java.io.*;
public class Main{
	public static void main(String args[]){
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		long a[]=new long[n+1];
		for(int i=1;i<=(n+1)/2;i++)a[i]=sc.nextLong();
		long x=sc.nextLong();
		for(int i=(n+1)/2+1;i<=n;i++)a[i]=x;
		long pre[]=new long[n+1];
		long pmax[]=new long[n+1];
		pmax[0]=Integer.MIN_VALUE;
		for(int i=1;i<=n;i++){
			pre[i]=pre[i-1]+a[i];
			pmax[i]=Math.max(pmax[i-1],pre[i]-i*x);
		}
		for(int i=(n+1)/2;i<=n;i++){
			if(pre[i]>0&&pre[i]>pmax[n-i]){
				System.out.println(i);
				return ;
			}
		}
		System.out.println(-1);
		return;
	}
}
