import sys
input = sys.stdin.readline

def solve(s, x, y):
	c0 = 0
	c1 = 0
	r = 0
	for i in s:
		if i == '0':
			c0 += 1
		else:
			r += c0 * x
			c1 += 1
	z0 = 0
	for i in range(len(s) - 1, -1, -1):
		if s[i] == '0':
			z0 += 1
		else:
			r += z0 * y
	ans = r
	l0 = 0
	l1 = 0
	for i in s:
		if i == '?':
			r -= l0 * x + c0 * y
			c1 -= 1
			r += l1 * y + c1 * x
			l0 += 1
		elif i == '0':
			l0 += 1
			c0 -= 1
		else:
			l1 += 1
			c1 -= 1
		ans = min(ans, r)
	return ans
s = input().strip()
(x, y) = map(int, input().split())
if x <= y:
	print(solve(s, x, y))
else:
	s1 = s.replace('0', 'z')
	s1 = s1.replace('1', '0')
	s1 = s1.replace('z', '1')
	print(solve(s1, y, x))
