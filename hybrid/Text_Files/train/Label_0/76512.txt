def findOrder(limit):
	order = 0
	while limit > 1:
		limit = limit >> 1
		order += 1
	return order

def findLimit(num, limit):
	order = findOrder(limit)
	ans = []
	while num != 0:
		if order == -1:
			return [-1]
		multiplier = 1
		base = 2 ** order
		curNum = base * multiplier
		while curNum <= limit and base <= num:
			ans.append(curNum)
			num -= base
			multiplier += 2
			curNum = base * multiplier
		order -= 1
	return ans

def printAnswer(lst):
	if lst[0] == -1:
		print(-1)
	else:
		print(len(lst))
		for i in lst:
			print(i, end=' ')
(sm, lmt) = map(int, input().split(' '))
printAnswer(findLimit(sm, lmt))
