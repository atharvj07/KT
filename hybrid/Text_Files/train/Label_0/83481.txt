(n, x) = map(int, input().split())
a = [True for i in range(pow(2, n))]
check = []
if x < pow(2, n):
	a[x] = False
for i in range(1, pow(2, n)):
	if a[i] == True:
		check.append(i)
		if x ^ i < pow(2, n):
			a[x ^ i] = False
print(len(check))
if len(check) == 1:
	print(check[0])
elif len(check) > 1:
	print(check[0], end=' ')
	for i in range(1, len(check)):
		print(check[i - 1] ^ check[i], end=' ')
