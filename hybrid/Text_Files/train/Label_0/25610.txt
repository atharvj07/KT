#include <bits/stdc++.h>
using namespace std;
int num[111111];
int main(void) {
  int n = 0;
  scanf("%d", &n);
  int ans = 0;
  stack<int> st;
  for (int i = 0; i < n; i++) {
    int x = 0;
    scanf("%d", &x);
    num[i] = x;
    while (st.size() && st.top() <= x) st.pop();
    if (st.size()) ans = max(ans, x ^ st.top());
    st.push(x);
  }
  while (st.size()) st.pop();
  reverse(num, num + n);
  for (int i = 0; i < n; i++) {
    int x = num[i];
    while (st.size() && st.top() <= x) st.pop();
    if (st.size()) ans = max(ans, x ^ st.top());
    st.push(x);
  }
  printf("%d\n", ans);
  return 0;
}
