int solve(int N, int U) {
  memset(visited, 0, sizeof(visited));
  //  priority_queue<State> up;
  queue<State> up;
  State s;
  s.one = 0;
  s.mask = 0;
  s.have = 0;
  visited[s.mask] = 1;
  up.push(s);
  while( !up.empty() ) {
    s = up.front(); up.pop();

    if( s.have >= U ) return s.one;

    for(int i = 0; i < N; i++)
      if ( visited[s.mask | (1 << i)] == 0 && (s.mask | need[i]) == s.mask )
        {
          State next;
          next.mask = s.mask | (1 << i);
          next.one = s.one + 1;
          next.have = s.have + unit[i];
          visited[next.mask] = 1;
          up.push(next);
        }
  }
  return -1;
}
