n = int(input())
a = [int(i) for i in input().split()]
if n == 1:
	print(a[0])
else:
	dp = [dict(), dict()]
	dp[0][0, 0, 0] = a[0]
	dp[0][1, 0, 1] = -a[0]
	for i in range(1, n):
		_i = i & 1
		dp[_i].clear()
		for j in range(3):
			dp[_i][j, 0, 0] = dp[1 - _i].get((j, 0, 1), -200000000000000.0 - 5) + a[i]
			dp[_i][j, 0, 1] = dp[1 - _i].get(((j + 2) % 3, 0, 0), -200000000000000.0 - 5) - a[i]
			dp[_i][j, 1, 0] = max(dp[1 - _i].get((j, 0, 0), -200000000000000.0 - 5) + a[i], dp[1 - _i].get((j, 1, 0), -200000000000000.0 - 5) + a[i], dp[1 - _i].get((j, 1, 1), -200000000000000.0 - 5) + a[i])
			dp[_i][j, 1, 1] = max(dp[1 - _i].get(((j + 2) % 3, 0, 1), -200000000000000.0 - 5) - a[i], dp[1 - _i].get(((j + 2) % 3, 1, 0), -200000000000000.0 - 5) - a[i], dp[1 - _i].get(((j + 2) % 3, 1, 1), -200000000000000.0 - 5) - a[i])
	print(max(dp[n - 1 & 1][(3 - (n + 2) % 3) % 3, 1, 0], dp[n - 1 & 1][(3 - (n + 2) % 3) % 3, 1, 1]))
