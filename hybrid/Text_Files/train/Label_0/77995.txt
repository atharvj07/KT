(n, m) = map(int, input().split())
s = [input() for i in range(n)]
h = [sum((1 if s[i][j] == '*' else 0 for j in range(m))) for i in range(n)]
v = [sum((1 if s[j][i] == '*' else 0 for j in range(n))) for i in range(m)]
c = sum(h)
ans = (-1, -1)
for i in range(n):
	for j in range(m):
		if h[i] + v[j] - (1 if s[i][j] == '*' else 0) == c:
			ans = (i, j)
if ans == (-1, -1):
	print('NO')
else:
	print('YES')
	print(ans[0] + 1, ans[1] + 1)
