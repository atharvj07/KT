(a, b) = (int(input()), int(input()))
(c, d) = (int(input()), int(input()))
(e, f) = (int(input()), int(input()))
if e > f:
	ans1 = min(a, d)
	a -= ans1
	d -= ans1
	ans2 = min(b, c, d)
	print(ans1 * e + ans2 * f)
else:
	ans2 = min(b, c, d)
	c -= ans2
	d -= ans2
	b -= ans2
	ans1 = min(a, d)
	print(ans1 * e + ans2 * f)
