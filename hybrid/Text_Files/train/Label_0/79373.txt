n = int(input())
x = []
y = []
a = []
b = []
for _ in range(n):
	(t1, t2) = map(int, input().split())
	x.append(t1)
	y.append(t2)
for _ in range(n):
	(t1, t2) = map(int, input().split())
	a.append(t1)
	b.append(t2)
tx = [[] for _ in range(n)]
ty = [[] for _ in range(n)]
for i in range(n):
	for j in range(n):
		tx[i].append(x[i] + a[j])
for i in range(n):
	for j in range(n):
		ty[i].append(y[i] + b[j])
c = []
i = 0
j = 1
while True:
	if j == n:
		c.append(tx[0][i])
		break
	elif tx[0][i] in tx[j]:
		j += 1
	else:
		i += 1
		j = 1
i = 0
j = 1
while True:
	if j == n:
		c.append(ty[0][i])
		break
	elif ty[0][i] in ty[j]:
		j += 1
	else:
		i += 1
		j = 1
print(*c)
