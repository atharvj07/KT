t = int(input())
import math
while t:
	t -= 1
	n = int(input())
	ques = []
	ans = []
	ans.append(0)
	for i in range(n):
		(x, y) = map(int, input().strip().split())
		ques.append(x)
		ans.append(y)
	val = list(map(int, input().strip().split()))
	num = 0
	for j in range(n - 1):
		correct = num + val[j] + ques[j] - ans[j]
		num = max(correct + math.ceil((ans[j + 1] - ques[j]) / 2), ans[j + 1])
	correct = num + val[-1] + ques[-1] - ans[-2]
	print(correct)
