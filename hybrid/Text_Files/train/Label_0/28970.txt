const int N = 2e5 +5;
typedef pair<int,int> pii;

int n,k;
string s;
int koto;
map<pii,bool>dp;
int lim;
int sum[N][20];
int nxt[N][20];

int getsum(int l, int r, int idx)
{
    int ret = sum[r][idx];
    if(l-1>=0)
        ret -= sum[l-1][idx];
    return ret;
}

bool solve(int idx, int mask)
{
    if(mask == lim)
    {
        return 1;
    }
    if(idx>=n)
    {
        return 0;
    }
    if(dp.count({idx,mask}))
        return dp[ {idx,mask}];
    bool ans = 0;
    bool flag = 1;
    for(int i=0; i<k; i++)
    {
        if(mask&(1<<i))
            continue;
        int koi = nxt[idx][i];
        if(koi==-1)
        {
            flag = 0;
            break;
        }
    }
    if(flag)
    {
        for(int i=0; i<k; i++)
        {
            if(mask&(1<<i))
                continue;
            int koi = nxt[idx][i];
            if(solve(koi+koto,mask|(1<<i)))
            {
                ans = 1;
                break;
            }
        }
    }
    return dp[ {idx,mask}]=ans;
}

bool check(int len)
{
    koto = len;
    dp.clear();
    for(int i=0; i<k; i++)
    {
        vector<int>all;
        for(int j=0; j+len-1<n; j++)
        {
            int l = j, r= j+len-1;
            int x = getsum(l,r,i) + getsum(l,r,k);
            if(x == len)
            {
                all.push_back(j);
            }
        }
        for(int j=0; j<n; j++)
        {
            int koi = lower_bound(all.begin(),all.end(),j)-all.begin();
            if(koi<all.size())
                nxt[j][i] = all[koi];
            else
                nxt[j][i] = -1;
        }
    }
    return solve(0,0);
}


int32_t main()
{
#ifndef APURBA
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
#endif
    cin>>n>>k>>s;
    if(k>n)
    {
        cout<<"0\n";
        return 0;
    }
    lim = (1<<k)-1;
    for(int i=0; i<k; i++)
    {
        for(int j=0; j<n; j++)
        {
            sum[j][i] = (j-1>=0?sum[j-1][i]:0);
            if(s[j]-'a'==i)
            {
                sum[j][i]++;
            }
        }
    }
    for(int j=0; j<n; j++)
    {
        sum[j][k] = (j-1>=0?sum[j-1][k]:0);
        if(s[j]=='?')
        {
            sum[j][k]++;
        }
    }
    int low = 1, high = n/k;
    int ans = 0;
    while(low<=high)
    {
        int mid = (low+ high)/2;
        if(check(mid))
        {
            low = mid+1;
            ans = mid;
        }
        else
        {
            high = mid-1;
        }
    }
    cout<<ans<<"\n";
    return 0;
}
