    void solve(int TC) throws Exception{
        String text = n();
        int N = ni();
        AhoCorasick aho1 = new AhoCorasick(), aho2 = new AhoCorasick();
        for(int i = 0; i< N; i++){
            String pat = n();
            aho1.addString(pat);
            aho2.addString(new StringBuilder(pat).reverse().toString());
        }
        aho1.buildSuffixLinks();
        aho2.buildSuffixLinks();

        int[] hit1 = aho1.count(text);
        text = new StringBuilder(text).reverse().toString();
        int[] hit2 = aho2.count(text);
        long ans = 0;
        int M = text.length();
        for(int i = 0; i< M-1; i++)ans += hit1[i]*(long)hit2[M-1-(i+1)];
        pn(ans);
    }
    public class AhoCorasick {
        State root, sentinal;
        int ID = 0;
        class State{
            TreeMap<Character, State> nxt;
            State suffixLink;
            char pch;
            boolean end;
            int hits, id;
            public State(char ch, int id){
                nxt = new TreeMap<>();
                end = false;this.id = id;
                pch = ch;hits = 0;
                suffixLink = null;
            }
            State nxt(Character ch){return nxt.get(ch);}
            void put(Character ch, State st){ nxt.put(ch, st);}
        }
        public AhoCorasick(){
            root = new State('$', ID++);
            sentinal = new State(' ', ID++);
            root.suffixLink = sentinal;
            for(char ch = 'a'; ch <= 'z'; ch++)sentinal.put(ch, root);
        }
        State addString(String s){
            State node = root;
            for(int i = 0; i< s.length(); i++){
                State nxt = node.nxt(s.charAt(i));
                if(nxt == null){
                    nxt = new State(s.charAt(i), ID++);
                    node.put(s.charAt(i), nxt);
                }
                node = nxt;
            }
            node.end = true;
            node.hits++;
            return node;
        }
        int[] count(String text){
            State node = root;
            int[] cnt = new int[text.length()];
            for(int i = 0; i< text.length(); i++){
                node = next(node, text.charAt(i));
                cnt[i] = node.hits;
            }
            return cnt;
        }
        State next(State st, char ch){
            State nxt;
            while((nxt = st.nxt(ch)) == null)st = st.suffixLink;
            return nxt;
        }
        void buildSuffixLinks(){
            Queue<State> q = new ArrayDeque<>();
            q.add(root);
            while(!q.isEmpty()){
                State st = q.poll();
                for(State nxt:st.nxt.values()){
                    nxt.suffixLink = next(st.suffixLink, nxt.pch);
                    nxt.hits += nxt.suffixLink.hits;
                    q.add(nxt);
                }
            }
        }
    }

    int[][] make(int n, int e, int[] from, int[] to, boolean f){
        int[][] g = new int[n][];int[]cnt = new int[n];
        for(int i = 0; i< e; i++){
            cnt[from[i]]++;
            if(f)cnt[to[i]]++;
        }
        for(int i = 0; i< n; i++)g[i] = new int[cnt[i]];
        for(int i = 0; i< e; i++){
            g[from[i]][--cnt[from[i]]] = to[i];
            if(f)g[to[i]][--cnt[to[i]]] = from[i];
        }
        return g;
    }

            cnt[from[i]]++;

    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }

        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }

        String nextLine() throws Exception{
            String str;
            try{
                str = br.readLine();
            }catch (IOException e){
                throw new Exception(e.toString());
            }
            return str;
        }
    }
