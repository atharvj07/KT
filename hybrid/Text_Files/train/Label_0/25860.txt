#include<iostream>
#include<map>
#include<vector>
#include<algorithm>
#include<cstring>
#include<cmath>
#define F first
#define S second
using namespace std;
typedef pair<int, int> P;
int x, y;
vector<P> v[5];
P s, g;
int dp[5][2048];

int dist(P p1, P p2){
  return abs(p1.F - p2.F) + abs(p1.S - p2.S);
}

int last;

int solve(int i, int j, P p){
  if(i == last) return dp[i][j] = dist(p, g);
  if(dp[i][j] != -1) return dp[i][j];
  int res = 1 << 28;
  int t = (i + 1) % 5;
  for(int k = 0;k < v[t].size(); k++) res = min(res, solve(t, k, v[t][k]) + dist(p, v[t][k]));
  return dp[i][j] = res;
}

int main(void){
  for( ; ; ){
    cin >> x >> y;
    if(!x && !y) break;
    
    for(int i = 0;i < 5; i++) v[i].clear();

    for(int i = 0;i < y; i++){
      for(int j = 0;j < x; j++){
	char m;
	cin >> m;
	int t = m - '1';
	if(0 <= t && t < 5) v[t].push_back(P(j, i));
	else if(m == 'S') s = P(j, i);
	else if(m == 'G') g = P(j, i);
      }
    }

    int num = 1 << 28, type;
    for(int i = 0;i < 5; i++){
      last = (i + 4) % 5;
      memset(dp, -1, sizeof(dp));
      int t = solve(i, 0, s);
      if(t < num){
	num = t;
	type = i;
      }
    }

    if(num == 1 << 28) cout << "NA" << endl;
    else cout << type + 1 << " " << num << endl;
  }
}
