(n, k) = map(int, input().split())
S = input()
i = S.index('G')
p = S.index('T')
c = 0
if i < p:
	while i < p:
		i = i + k
		if i > p:
			c = 0
			break
		if S[i] == '#':
			c = 0
			break
		if S[i] == 'T':
			c = 1
			break
else:
	while i > p:
		i = i - k
		if i < p:
			c = 0
			break
		if S[i] == '#':
			c = 0
			break
		if S[i] == 'T':
			c = 1
			break
if c == 1 or p == i:
	print('YES')
else:
	print('NO')
