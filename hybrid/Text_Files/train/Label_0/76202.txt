for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	s = 0
	ss = 0
	l = []
	for i in range(n):
		l.append([a[i], b[i]])
	l.sort()
	kkk = sum([l[j][1] for j in range(n)])
	kk = kkk
	for i in range(n):
		kkk -= l[i][1]
		bb = max(l[i][0], kkk)
		if kk > bb:
			kk = bb
	print(kk)
