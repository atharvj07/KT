import sys
data = sys.stdin.read().split()
data.reverse()

def read():
	return int(data.pop())

def readStr():
	return data.pop()
n = read()
C = [0] + [read() for i in range(n)]
ans = ''
(ch, cht) = ('a', 'b')
for i in range(n, 0, -1):
	while C[i] > 0:
		for j in range(i):
			ans += ch
		(ch, cht) = (cht, ch)
		for j in range(1, i + 1):
			C[j] -= i - j + 1
print(ans)
