#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
const int INF = 1e9 + 7;
struct node {
  int pos;
  char c;
} p[N];
int main() {
  int n;
  scanf("%d", &n);
  getchar();
  for (int i = 0; i < n; i++) scanf("%c", &p[i].c);
  for (int i = 0; i < n; i++) scanf("%d", &p[i].pos);
  int minm = INF;
  for (int i = 0; i < n - 1; i++)
    if (p[i].c == 'R' && p[i + 1].c == 'L' && p[i + 1].pos - p[i].pos < minm)
      minm = p[i + 1].pos - p[i].pos;
  if (minm == INF)
    puts("-1");
  else
    printf("%d\n", minm / 2);
  return 0;
}
