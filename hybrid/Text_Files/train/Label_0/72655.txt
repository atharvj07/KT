t = int(input())

def compare(a):
	return a[0]
for _ in range(t):
	(n, m) = [int(i) for i in input().split()]
	paths = []
	t = 1
	for i in range(n):
		paths.append([[int(i), t, id] for (id, i) in enumerate(input().split())])
		t += 1
	topath = []
	for i in range(len(paths)):
		topath.extend(paths[i])
	topath.sort(key=compare)
	bestpath = topath[:m]
	res = []
	for i in range(len(paths)):
		new_path = [0] * m
		path_set = set()
		for j in range(len(bestpath)):
			if bestpath[j][1] == i + 1:
				new_path[j] = bestpath[j][0]
				path_set.add(bestpath[j][2])
		p = 0
		l = 0
		while p < m and l < m:
			while l in path_set:
				l += 1
			while p < m and new_path[p] != 0:
				p += 1
			if p < m and l < m:
				new_path[p] = paths[i][l][0]
				p += 1
				l += 1
		res.append(new_path)
	for i in res:
		print(' '.join(map(str, i)))
