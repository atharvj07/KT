def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
for _ in range(int(input())):
	(n, m, k) = get_ints()
	arr = []
	for i in range(n):
		s = input()
		arr.append(s)
	dp = [[False for i in range(m)] for j in range(n)]
	for i in range(n):
		for j in range(m):
			if arr[i][j] == '*':
				(p, q) = (i, j)
				(r, s) = (i, j)
				tot = 0
				while True:
					p -= 1
					q -= 1
					r -= 1
					s += 1
					if p >= 0 and r >= 0 and (q >= 0) and (s < m) and (arr[p][q] == '*') and (arr[r][s] == '*'):
						tot += 1
					else:
						break
				if tot >= k:
					dp[i][j] = True
					(p, q) = (i, j)
					(r, s) = (i, j)
					for j in range(tot):
						p -= 1
						q -= 1
						r -= 1
						s += 1
						dp[p][q] = True
						dp[r][s] = True
	flag = True
	for i in range(n):
		for j in range(m):
			if arr[i][j] == '*' and dp[i][j] == False:
				flag = False
				break
	if flag == False:
		print('NO')
	else:
		print('YES')
