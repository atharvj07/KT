Q = int(input())
for _ in range(Q):
	N = int(input())
	nums = list(map(int, input().split()))
	f = 1
	for i in range(N):
		if nums[i] != nums[(i + 1) % N] % N + 1:
			f = 0
	f2 = 1
	for i in range(N):
		if nums[i] % N + 1 != nums[(i + 1) % N]:
			f2 = 0
	if f or f2:
		print('YES')
	else:
		print('NO')
