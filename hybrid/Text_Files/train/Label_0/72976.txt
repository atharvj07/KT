t = int(input())
for i in range(t):
	(p, f) = map(int, input().split())
	(cs, cw) = map(int, input().split())
	(s, w) = map(int, input().split())
	maxi = 0
	for j in range(min(cs, p // s) + 1):
		k = min((p - s * j) // w, cw)
		xcs = cs - j
		xcw = cw - k
		if s > w:
			if w * xcw >= f:
				m = f // w
				l = 0
			else:
				m = xcw
				l = min((f - m * w) // s, xcs)
		elif s * xcs >= f:
			l = f // s
			m = 0
		else:
			l = xcs
			m = min((f - s * l) // w, xcw)
		maxi = max(maxi, l + m + j + k)
	print(maxi)
