#include <bits/stdc++.h>
using namespace std;
int leap(int x) {
  if (x % 4 == 0 && (x % 100 != 0 || x % 400 == 0)) {
    return 1;
  } else {
    return 0;
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  int T = 1;
  while (T--) {
    int y, sum = 0;
    cin >> y;
    for (int i = y; i < 1e9; i++) {
      if (leap(i)) {
        sum += 2;
      } else {
        sum += 1;
      }
      if (sum % 7 == 0 && leap(i + 1) == leap(y)) {
        cout << i + 1;
        return 0;
      }
    }
  }
}
