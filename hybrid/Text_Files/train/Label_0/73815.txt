def print_breaks(a, b):
	(a, b) = (min(a, b), max(a, b))
	x = b - a
	if x % 2 == 1:
		y = (x - 1) // 2
		res = list([str(el) for el in range(y, a * 2 + y + 2)])
	else:
		y = x // 2
		res = list([str(el) for el in range(y, 2 * a + y + 1, 2)])
	print(len(res))
	print(' '.join(res))
t = int(input())
for i in range(t):
	s = input()
	(a, b) = s.split(' ')
	(a, b) = (int(a), int(b))
	print_breaks(a, b)
