import java.util.*;
import java.io.*;
import java.lang.*;

public class cfs {

	public static void main(String[] args)  {
      
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0) {
		int a=sc.nextInt();
		int b=sc.nextInt();
		int c=sc.nextInt();
		int count=0;
		if(a>0) {count++;a--;}
		if(b>0) {count++;b--;}
		if(c>0) {count++;c--;}
		int arr[]= {a,b,c};
		Arrays.sort(arr);
		
		if(arr[0]>0&&arr[2]>0) {count++;arr[0]--;arr[2]--;}
		if(arr[1]>0&&arr[2]>0) {count++;arr[1]--;arr[2]--;}
		if(arr[1]>0&&arr[0]>0) {count++;arr[0]--;arr[1]--;}
		if(arr[0]>0&&arr[1]>0&&arr[2]>0) {count++;a--;b--;c--;
		}
		
System.out.println(count);
}

}
}
