def year():
	dct = {}
	t = int(input())
	for i in range(t):
		(a, b) = map(int, input().strip().split())
		if a in dct:
			dct[a] += 1
		else:
			dct[a] = 1
		if b in dct:
			dct[b] -= 1
		else:
			dct[b] = -1
	count = 0
	curr = 0
	y = 0
	for i in sorted(dct.keys()):
		curr += dct[i]
		if curr > count:
			count = curr
			y = i
	print(y, count)
year()
