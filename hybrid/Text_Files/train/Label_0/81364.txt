from collections import defaultdict

def solve():
	(n, l, r) = map(int, input().split())
	a = list(map(int, input().split()))
	ls = defaultdict(int)
	rs = defaultdict(int)
	for i in range(n):
		if i < l:
			ls[a[i]] += 1
		else:
			rs[a[i]] += 1
	ans = 0
	for (i, j) in ls.items():
		mino = min(j, rs[i])
		ls[i] -= mino
		rs[i] -= mino
	tl = 0
	tr = 0
	for e in ls.values():
		tl += e
	for e in rs.values():
		tr += e
	if tl <= tr:
		more = tr - tl
		for (i, j) in rs.items():
			if more <= 0:
				break
			while more > 0 and j >= 2:
				j -= 2
				tr -= 2
				ans += 1
				more -= 2
		ans += tl + more
	else:
		more = tl - tr
		for (i, j) in ls.items():
			if more <= 0:
				break
			while more > 0 and j >= 2:
				j -= 2
				tl -= 2
				ans += 1
				more -= 2
		ans += tr + more
	print(ans)
test = int(input())
for _ in range(test):
	solve()
