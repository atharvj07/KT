#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    long long sum = a[0] * a[n - 1];
    vector<long long> b;
    for (long long i = 2; i * i <= sum; i++) {
      if (sum % i == 0) {
        b.push_back(i);
        if (i != sum / i) b.push_back(sum / i);
      }
    }
    sort(b.begin(), b.end());
    if (a == b)
      cout << sum << endl;
    else
      cout << "-1" << endl;
  }
  return 0;
}
