#include<bits/stdc++.h>
using namespace std;

#define int long long
typedef vector<int>vint;
typedef pair<int,int>pint;
typedef vector<pint>vpint;
#define rep(i,n) for(int i=0;i<(n);i++)
#define reps(i,f,n) for(int i=(f);i<(n);i++)
#define all(v) (v).begin(),(v).end()
#define each(it,v) for(__typeof((v).begin()) it=(v).begin();it!=(v).end();it++)
#define pb push_back
#define fi first
#define se second
template<typename A,typename B>inline void chmin(A &a,B b){if(a>b)a=b;}
template<typename A,typename B>inline void chmax(A &a,B b){if(a<b)a=b;}

const double INF=1e12;
int M,N;
int X1[100],X2[100],Y1[100],Y2[100];

double f(int *X,int *Y,int n,double x){
    double l=INF,r=-INF;
    rep(i,n){
        double x1=X[i],y1=Y[i],x2=X[(i+1)%n],y2=Y[(i+1)%n];
        if((x1-x)*(x2-x)>0||x1==x2)continue;
        double y=y1+(y2-y1)/(x2-x1)*(x-x1);
        chmin(l,y);
        chmax(r,y);
    }
    return max(0.0,r-l);
}

void solve(){
    vint xs;
    rep(i,M)xs.pb(X1[i]);
    rep(i,N)xs.pb(X2[i]);
    sort(all(xs));
    int min1=*min_element(X1,X1+M),max1=*max_element(X1,X1+M);
    int min2=*min_element(X2,X2+N),max2=*max_element(X2,X2+N);
    double ans=0;
    for(int i=0;i+1<xs.size();i++){
        double a=xs[i],b=xs[i+1],c=(a+b)/2.0;
        if(c<min1||c>max1||c<min2||c>max2)continue;
        double fa=f(X1,Y1,M,a)*f(X2,Y2,N,a);
        double fb=f(X1,Y1,M,b)*f(X2,Y2,N,b);
        double fc=f(X1,Y1,M,c)*f(X2,Y2,N,c);
        ans+=(b-a)/6*(fa+fb+fc*4);
    }
    printf("%.20f\n",ans);
}

signed main(){
    while(cin>>M>>N,M||N){
        rep(i,M)cin>>X1[i]>>Y1[i];
        rep(i,N)cin>>X2[i]>>Y2[i];
        solve();
    }
}
