import os
from sys import stdin, stdout

def solve(tc):
	a = stdin.readline().strip()
	b = stdin.readline().strip()
	na = len(a)
	nb = len(b)
	if na != nb:
		print('NO')
		return
	allZero = [True, True]
	for i in range(na):
		if a[i] == '1':
			allZero[0] = False
			break
	for i in range(nb):
		if b[i] == '1':
			allZero[1] = False
			break
	if allZero[0] == allZero[1]:
		print('YES')
	else:
		print('NO')
tcs = 1
tc = 1
while tc <= tcs:
	solve(tc)
	tc += 1
