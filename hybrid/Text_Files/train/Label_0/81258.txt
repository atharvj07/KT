import sys
input = sys.stdin.readline
(n, q) = list(map(int, input().split()))
arr = [0] * n
c = 0
for _ in range(q):
	(a, b) = map(int, input().split())
	if a < b:
		ind = a - 1
	else:
		ind = b - 1
	arr[ind] += 1
	if arr[ind] == 1:
		c += 1
for i in range(int(input())):
	y = list(map(int, input().split()))
	if len(y) == 1:
		print(n - c)
		continue
	(v, a, b) = y
	if a < b:
		ind = a - 1
	else:
		ind = b - 1
	if v == 1:
		arr[ind] += 1
		if arr[ind] == 1:
			c += 1
	elif v == 2:
		arr[ind] -= 1
		if arr[ind] == 0:
			c -= 1
