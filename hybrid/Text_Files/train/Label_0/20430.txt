#include<bits/stdc++.h>
using namespace std;
const int N=6005;
int ne[N],tot,fi[N],zz[N],n,a[N],f[N],x,y,Ans[N];
void jb(int x,int y){
	ne[++tot]=fi[x];
	fi[x]=tot;
	zz[tot]=y;
}
int cmp(int x,int y){
	return a[x]<a[y];
}
int main(){
	scanf("%d",&n);
	for (int i=1;i<=n;i++)scanf("%d",&a[i]);
	for (int i=1;i<n;i++){
		scanf("%d%d",&x,&y);
		jb(x,y);jb(y,x);
	}
	for (int i=1;i<=n;i++)f[i]=i;
	sort(f+1,f+n+1,cmp);
	for (int i=1;i<=n;i++){
		Ans[f[i]]=0;
		for (int j=fi[f[i]];j;j=ne[j])
			if (a[f[i]]>a[zz[j]])Ans[f[i]]|=!Ans[zz[j]];
	}
	for (int i=1;i<=n;i++)
		if (Ans[i])printf("%d ",i);
}
