#include<cctype>
#include<cstdio>

#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

typedef long long ll;

const int dx[]={1,0,-1,0},dy[]={0,-1,0,1};

int main(){
	char DIR[]="ENWS";
	int DIRR[128]; rep(k,4) DIRR[DIR[k]]=k;

	int h,w;
	for(ll l;scanf("%d%d%lld",&h,&w,&l),h;){
		static char B[100][101];
		int x,y,dir;
		rep(i,h){
			scanf("%s",B[i]);
			rep(j,w) if(isalpha(B[i][j])) x=j, y=i, dir=DIRR[B[i][j]];
		}

		int T[100][100][4];
		rep(i,h) rep(j,w) rep(k,4) T[i][j][k]=-1;

		ll t=0;
		while(l>0){
			// ツ偲シツ甘コツつェツづ敖づつつゥツづつスツづァツ暗ェツ気ツづ可進ツづ淞づゥ
			if(~T[y][x][dir]){
				ll dif=t-T[y][x][dir];
				t+=(l-1)/dif*dif;
				l-=(l-1)/dif*dif;
			}
			T[y][x][dir]=t;

			int xx=x+dx[dir],yy=y+dy[dir];
			// ツ陳シツ進
			if(0<=yy && yy<h && 0<=xx && xx<w && B[yy][xx]!='#'){
				t++;
				l--;
				x=xx;
				y=yy;
			}
			// ツ陛サツ古シツ転ツ環キ
			else{
				dir=(dir+3)%4;
			}
		}

		printf("%d %d %c\n",y+1,x+1,DIR[dir]);
	}

	return 0;
}
