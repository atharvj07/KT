#include<iostream>
#include<vector>
#include<algorithm>
#include<cstring>

using namespace std;

#define ALL(v) v.begin(), v.end()

const int MAX = 1005;
int N,M;

struct Info{
  int sum, tail;
  vector<int> K;
};


void solve(const vector<int>& L){

  vector< Info > v;
  vector<int> tmp(M);
  for(int i = 0; i < L[0]; i++) tmp[i] = 1;
  for(int i = L[0]; i < M; i++) tmp[i] = 0;
  sort(ALL(tmp));

  do{ v.push_back((Info){L[0],0,tmp});}while(next_permutation(ALL(tmp)));

  for(int i = 1; i < N; i++){
    vector< Info > nex;
    for(int j = 0; j < (int)v.size(); j++){
      Info tmp = v[j];
      tmp.sum -= tmp.K[tmp.tail++];
      if(tmp.sum == L[i]){
	tmp.K.push_back(0);
	nex.push_back(tmp);
      }else if(tmp.sum+1 == L[i]){
	tmp.K.push_back(1);
	tmp.sum++;
	nex.push_back(tmp);
      }
    }
    v = nex;
  }
  
  if(v.empty()) cout << "Goofy" << endl;
  else{
    vector<int> ans = v[0].K;
    for(int i = 0; i < (int)ans.size(); i++) cout << ans[i];
    cout << endl;
  }
}

int main(){
  while(cin >> N >> M && (N|M)){
    vector<int> L(N);
    for(int i = 0; i < N; i++) cin >> L[i];
    solve(L);
  }
  return 0;
}
