#include<bits/stdc++.h>
using namespace std;
int main(){
  int N;
  bitset<200000> X;
  cin>>N>>X;
  int M=X.count(),now1=0,now2=0;
  if(M>=2){
    for(int i=N-1;i>=0;i--)
      now1=(now1*2+X[i])%(M+1),now2=(now2*2+X[i])%(M-1);
    vector<int> v1(N),v2(N);
    v1[0]=1,v2[0]=1;
    for(int i=1;i<N;i++){
      v1[i]=v1[i-1]*2%(M+1);
      v2[i]=v2[i-1]*2%(M-1);
    }
    for(int i=0;i<N;i++){
      int n;
      if(X[N-1-i]==1)
        n=(now2-v2[N-1-i]+(M-1))%(M-1);
      else
        n=(now1+v1[N-1-i]+(M+1))%(M+1);
      int ans=1;
      for(int j=n;j>0;j%=__builtin_popcount(j))
        ans++;
      cout<<ans<<endl;
    }
  }else if(M==1){
    for(int i=N-1;i>=0;i--)
      now1=(now1*2+X[i])%(M+1);
    vector<int> v1(N);
    v1[0]=1;
    for(int i=1;i<N;i++)
      v1[i]=v1[i-1]*2%(M+1);
    for(int i=0;i<N;i++){
      if(X[N-1-i]==0){
        int n=(now1+v1[N-1-i])%(M+1);
        bitset<200000> A(n);
        int ans=1;
        for(int j=n;j>0;j%=__builtin_popcount(j))
          ans++;
        cout<<ans<<endl;
      }else
        cout<<0<<endl;
    }
  }else
    for(int i=0;i<N;i++)
      cout<<1<<endl;
}
