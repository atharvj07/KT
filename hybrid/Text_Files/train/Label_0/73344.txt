def main():
	n = int(input())
	ans = n
	x = 1
	n1 = 1
	t = 0
	k = 0
	if n < 10:
		x = 1
	else:
		for u in range(1000):
			n /= 10
			x += 1
			if n < 10:
				break
	x -= 1
	for i in range(x):
		n1 *= 10
	n1 -= 1
	n2 = ans - n1
	for i in range(n1):
		t += n1 % 10
		n1 = n1 // 10
		if n1 < 10:
			t += n1
			break
	for i in range(n2):
		k += n2 % 10
		n2 = n2 // 10
		if n2 < 10:
			k += n2
			break
	print(t + k)
main()
