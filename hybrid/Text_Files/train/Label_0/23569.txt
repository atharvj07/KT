#include <bits/stdc++.h>
using namespace std;
int n;
long long leastPath;
void processPath(int rooms[]) {
  long long visit[n];
  fill(visit, visit + n, 2);
  for (int i = 1; i < n; i++) {
    if ((rooms[i] - 1) != i) {
      for (int j = (rooms[i] - 1); j < i; j++) {
        visit[i] += visit[j];
      }
      if (visit[i] > 1000000007) {
        visit[i] %= 1000000007;
      }
    }
  }
  for (int i = 0; i < n; ++i) {
    leastPath += visit[i];
  }
}
int main() {
  cin >> n;
  int rooms[n];
  for (int i = 0; i < n; ++i) {
    scanf("%d", rooms + i);
  }
  processPath(rooms);
  cout << (leastPath % 1000000007) << endl;
}
