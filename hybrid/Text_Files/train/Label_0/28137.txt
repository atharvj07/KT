template <class T> void read(T &x)
{
    x=0; char c=gc(); int flag=0;
    while (c<'0'||c>'9') flag|=(c=='-'),c=gc();
    while (c>='0'&&c<='9') x=(x<<3)+(x<<1)+(c^48),c=gc();
    if (flag) x=-x;
}
template <class T> T _max(T a,T b){return a>b ? a : b;}
template <class T> T _min(T a,T b){return a<b ? a : b;}
template <class T> bool checkmax(T &a,T b){return b>a ? a=b,1 : 0;}
template <class T> bool checkmin(T &a,T b){return b<a ? a=b,1 : 0;}
const int N=200005,inf=0x3f3f3f3f;
int n,k;
int a[N],r[N],b[N];
int suma[N],sumb,sumr;
void init()
{
	read(n); read(k);
	for (int i=1;i<=n;++i)
	{
		read(a[i]);
	}
	sort(a+1,a+n+1);
	reverse(a+1,a+n+1);
	for (int i=1;i<=k;++i)
	{
		read(r[i]);
	}
	sort(r+1,r+k+1);
	reverse(r+1,r+k+1);
	for (int i=1;i<=k;++i)
	{
		read(b[i]);
	}
	sort(b+1,b+k+1);
	reverse(b+1,b+k+1);
	for (int i=1;i<=n;++i)
	{
		suma[i]=suma[i-1]+a[i];
		sumb+=b[i];
		sumr+=r[i];
	}
}

int dp[62][62][3005];
void solve()
{
	int ans=inf;
	memset(dp,-1,sizeof dp);
	dp[0][0][0]=0;
	//outarr(a,1,n);
	for (int i=0;i<=k;++i)
	{
		for (int j=0;j<=k&&i+j<=n;++j)
		{
			for (int sum=0;sum<=suma[i+j];++sum)
			{
				if (dp[i][j][sum]==-1) continue;
				//out(i); out(j); out(sum); outln(dp[i][j][sum]);
				if (n==i+j)
				{
					if (sumr>=sum&&sumb>=suma[n]-sum)
					{
						//out(i); out(j); out(sum); outln(suma[n]-sum);
						checkmin(ans,suma[n]-dp[i][j][sum]);
					}
					continue;
				}
				if (i<k)
				{
					int new_sum=sum+a[i+j+1];
					checkmax(dp[i+1][j][new_sum],dp[i][j][sum]+_min(r[i+1],a[i+j+1]));
				}
				if (j<k)
				{
					int new_sum=sum;
					checkmax(dp[i][j+1][new_sum],dp[i][j][sum]+_min(b[j+1],a[i+j+1]));
				}
			}
		}
	}
	if (ans==inf) ans=-1;
	printf("%d\n",ans);
}

#undef int
int main()
{
	init();
	solve();
    return 0;
}
