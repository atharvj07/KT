#include <bits/stdc++.h>
using namespace std;
inline bool Read(int& x) {
  x = 0;
  int c;
  int sign = 1;
  while ((c = getchar()) < '0' || c > '9')
    if (c == '-') sign = -1;
  x = c ^ '0';
  while ((c = getchar()) >= '0' && c <= '9')
    x = (x << 3) + (x << 1) + (c ^ '0');
  x *= sign;
  return 1;
}
inline bool Write(int x) {
  if (x >= 10) Write(x / 10);
  putchar(x % 10 + '0');
  return 1;
}

const int M = 2000000;
const int Mod = 1000007;
const int h = 800214;
const int dx[] = {1, 0, -1, 0};
const int dy[] = {0, 1, 0, -1};
int n, m;
int len = 0;
string grid[20];
char gr[20][20];
int x[M + 200][10], y[M + 200][10];
int z[M + 200];
unsigned long long to[M + 200];
int H[Mod];
int N[M + 200];
inline bool inr(int x, int y) { return x >= 0 && x < n && y >= 0 && y < m; }
void getHash(int now) {
  for (int i = 0; i < len; i++)
    to[now] = ((to[now] * h + x[now][i]) * h) + y[now][i];
}
signed main() {
  ios::sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> grid[i];
    for (int j = 0; j < m; j++) {
      if (isdigit(grid[i][j])) {
        int num = (grid[i][j] - '1');
        x[1][num] = i;
        y[1][num] = j;
        len = max(len, num);
        grid[i][j] = '.';
      }
    }
  }
  int f = 1, e = 1;
  getHash(1);
  N[1] = 0;
  H[to[1] % Mod] = 1;
  z[1] = 0;
  while (f <= e) {
    if (e >= M) e -= M;
    if (f >= M) f -= M;
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++) {
        gr[i][j] = grid[i][j];
      }
    for (int i = 0; i < len; i++) gr[x[f][i]][y[f][i]] = '#';
    for (int i = 0; i < 4; i++) {
      int nx = x[f][0] + dx[i], ny = y[f][0] + dy[i];
      if (!inr(nx, ny) || gr[nx][ny] == '#') continue;
      if (gr[nx][ny] == '@') {
        cout << (z[f] + 1) << endl;
        return 0;
      } else {
        for (int i = 0; i < len; i++)
          x[e + 1][i + 1] = x[f][i], y[e + 1][i + 1] = y[f][i];
        x[e + 1][0] = nx, y[e + 1][0] = ny;
        getHash(e + 1);
        int j = H[to[e + 1] % Mod];
        bool ok = 1;
        while (j) {
          if (to[j] == to[e + 1]) {
            ok = 0;
            break;
          }
          j = N[j];
        }
        if (ok) {
          e++;
          z[e] = z[f] + 1;
          N[e] = H[to[e] % Mod];
          H[to[e] % Mod] = e;
        }
      }
    }
    f++;
  }
  cout << "-1\n";
  return 0;
}
