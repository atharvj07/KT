t = int(input())
for h in range(t):
	s = input().split(':')
	l = len(s)
	if l == 8:
		p = ''
		for i in range(8):
			k = len(s[i])
			s[i] = '0' * (4 - k) + s[i]
			p = p + s[i]
			if i != 7:
				p = p + ':'
	elif s[0] == '':
		p = ''
		m = 10 - l
		for i in range(m):
			p += '0000:'
		for i in range(2, l):
			k = len(s[i])
			s[i] = '0' * (4 - k) + s[i]
			p = p + s[i]
			if i != l - 1:
				p = p + ':'
	elif s[-1] == '':
		m = 10 - l
		p = ''
		for i in range(l - 2):
			k = len(s[i])
			s[i] = '0' * (4 - k) + s[i] + ':'
			p = p + s[i]
		for i in range(m):
			if i != m - 1:
				p += '0000:'
			else:
				p += '0000'
	else:
		p = ''
		m = 9 - l
		for i in range(l):
			k = len(s[i])
			if k == 0:
				for i in range(m):
					p += '0000:'
			else:
				s[i] = '0' * (4 - k) + s[i]
				p = p + s[i]
				if i != l - 1:
					p = p + ':'
	print(p)
