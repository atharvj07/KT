#include <bits/stdc++.h>
using namespace std;
int n, m;
struct node {
  int v, id;
  long long w;
  node(int a, int b, long long c) {
    v = a;
    id = b;
    w = c;
  }
};
vector<node> g[200100];
int par[200100];
long long val[200100];
int point[200100], fa[200100];
int dep[200100];
int find(int x) { return x == fa[x] ? x : fa[x] = find(fa[x]); }
void dfs(int x, int de, int fa1) {
  par[x] = fa1;
  dep[x] = de;
  for (int i = 0; i < g[x].size(); i++) {
    node e = g[x][i];
    if (e.v != fa1) {
      point[e.id] = e.v;
      val[e.v] = e.w;
      dfs(e.v, de + 1, x);
    }
  }
}
long long lca(int u, int v, long long y) {
  u = find(u);
  v = find(v);
  while (u != v) {
    if (dep[u] < dep[v]) swap(u, v);
    y /= val[u];
    u = find(par[u]);
    if (y == 0) return 0;
  }
  return y;
}
void update(int x) { fa[x] = par[x]; }
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) fa[i] = i;
  for (int i = 1; i < n; i++) {
    int u, v;
    long long w;
    scanf("%d%d%lld", &u, &v, &w);
    g[u].push_back(node(v, i, w));
    g[v].push_back(node(u, i, w));
  }
  dfs(1, 1, 0);
  for (int i = 2; i <= n; i++)
    if (val[i] == 1) update(i);
  int op, a, b;
  long long y, c;
  while (m--) {
    scanf("%d", &op);
    if (op == 1) {
      scanf("%d%d%lld", &a, &b, &y);
      printf("%lld\n", lca(a, b, y));
    } else {
      scanf("%d%lld", &a, &c);
      val[point[a]] = c;
      if (c == 1) update(point[a]);
    }
  }
  return 0;
}
