#include <bits/stdc++.h>
using namespace std;
long long arr[300300];
long long bit[300300];
void solve() {
  long long n;
  cin >> n;
  string s;
  cin >> s;
  long long left = 0;
  long long right = 0;
  long long d = 0;
  for (long long i = 0; i < s.length(); i++) {
    if (s[i] == '<')
      left++;
    else if (s[i] == '>')
      right++;
    else if (s[i] == '-') {
      d++;
      right++;
      left++;
    }
  }
  if (left == n) {
    cout << n << "\n";
  } else if (right == n) {
    cout << n << "\n";
  } else {
    long long ans = 0;
    for (long long i = 0; i < s.length(); i++) {
      if (s[i] == '-' || s[(i + 1) % n] == '-') ans++;
    }
    cout << ans << "\n";
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  long long tc;
  cin >> tc;
  while (tc--) {
    solve();
  }
}
