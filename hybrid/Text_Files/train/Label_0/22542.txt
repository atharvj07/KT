#include <bits/stdc++.h>
using namespace std;
const int N = 10005;
const int M = 16;
int n, m;
int mat[M][N];
int put[M][M];
int fst[M][M];
int a, b;
int memo[M][1 << 17];
int go(int u, int mask) {
  if (mask == (1 << n) - 1) {
    return put[u][b];
  }
  if (memo[u][mask] != -1) return memo[u][mask];
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (mask & (1 << i)) continue;
    ans = max(ans, min(put[u][i], go(i, mask | (1 << i))));
  }
  return memo[u][mask] = ans;
}
int main() {
  scanf("%d %d", &n, &m);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      scanf("%d", mat[i] + j);
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      int here = 2e9;
      for (int k = 0; k < m; k++) {
        here = min(here, abs(mat[i][k] - mat[j][k]));
      }
      put[i][j] = here;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      int here = 2e9;
      for (int k = 1; k < m; k++) {
        here = min(here, abs(mat[i][k] - mat[j][k - 1]));
      }
      fst[i][j] = here;
    }
  }
  if (n == 1) {
    int ans = 2e9;
    for (int j = 1; j < m; j++) {
      ans = min(ans, abs(mat[0][j] - mat[0][j - 1]));
    }
    printf("%d\n", ans);
    return 0;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      a = i;
      b = j;
      memset(memo, -1, sizeof memo);
      ans = max(ans, min(fst[a][b], go(a, ((1 << a) | (1 << b)))));
    }
  }
  printf("%d\n", ans);
  return 0;
}
