#include <bits/stdc++.h>
using namespace std;
int n, m;
double kd;
map<string, int> skills;
int main() {
  cin >> n >> m >> kd;
  int k = (int)(kd * 100 + 0.00001);
  for (int i = 0; i < n; i++) {
    string nameskill;
    int expval;
    cin >> nameskill >> expval;
    expval = ((k * expval) / 100);
    if (expval < 100) continue;
    skills[nameskill] = expval;
  }
  for (int i = 0; i < m; i++) {
    string nameskill;
    cin >> nameskill;
    if (skills.find(nameskill) == skills.end()) skills[nameskill] = 0;
  }
  cout << skills.size() << endl;
  for (map<string, int>::iterator it = skills.begin(); it != skills.end(); ++it)
    cout << (*it).first << " " << (*it).second << endl;
}
