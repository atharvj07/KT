	static void apply_hori(int[][] map,int i,int[] bar) {
		int n=bar.length;
		for (int j=0;j<n;j++)
			map[i][j]^=bar[j];
	}

	static void apply_vert(int[][] map,int j,int[] bar) {
		int n=bar.length;
		for (int i=0;i<n;i++)
			map[i][j]^=bar[i];
	}

	static int[] diff(int[] a,int[] b) {
		int L=a.length;
		int[] diff=new int[L];
		for (int i=0;i<L;i++)
			diff[i]=a[i]-b[i];
		return diff;
	}


	static boolean solveOther(int[][] ref,int[][] target,int[] bar) {
		int n=ref.length;
		int[][] map=new int[n][n];
		for (int i=0;i<n;i++)
			for (int j=0;j<n;j++)
				map[i][j]=ref[i][j]^target[i][j];

		return false;
	}

	
	static boolean complete(int[][] map,int[][] target,int[] hor,int[] vert,ArrayList<Integer> one,int tgi,int tgj) {


		//log("my local view: tgi:"+tgi+" tgj:"+tgj);
		if (tgi>-1) {
			// start with first i, and set vert accordingly
			//log("start with line tgi:"+(tgi+1));
			for (int i:new int[]{tgi})
				for (int j:one) {
					if (map[i][j]==target[i][j]) {
						if (vert[j]==1-hor[i]) {
							//log("fail with j:"+(j+1)+" vert:"+vert[j]);
							return false;
						}
						vert[j]=hor[i];
						//log("case 1 // setting vert at "+(j+1)+" to "+vert[j]);
					} else {
						if (hor[i]==vert[j]) {
							
							//log("fail with j:"+(j+1)+" vert:"+vert[j]);
							return false;
						}
			
						vert[j]=1-hor[i];
						//log("case 2 // setting vert at "+(j+1)+" to "+vert[j]);
					}
				}

			//log("=====step 1 ok");
			//log("....now check other lines");
			// then check other lines
			for (int i:one) {
				for (int j:one) {
					if (map[i][j]==target[i][j]) {
						if (hor[i]==1-vert[j]) {
							//log("issuse with line:"+(i+1));
							return false;
						}
						hor[i]=vert[j];
					} else {
						if (hor[i]==vert[j])
							return false;
						hor[i]=1-vert[j];
					}
				}
			}
		} else
		{

			// start with first tgj

			for (int j:new int[]{tgj})
				for (int i:one) {
					if (map[i][j]!=target[i][j]) {
						if (hor[i]==vert[j])
							return false;
						hor[i]=1-vert[j];
					} else {
						if (hor[i]==1-vert[tgj])
							return false;
						hor[i]=vert[tgj];
					}

				}
			// then do all
			for (int i:one) {
				for (int j:one) {
					if (map[i][j]==target[i][j]) {
						if (vert[j]==1-hor[i])
							return false;
						vert[j]=hor[i];
					} else {
						if (hor[i]==vert[j])
							return false;
						vert[j]=1-hor[i];
					}
				}
			}
		}
		return true;
	}

	static boolean solveSubtil(int[][] map,int[][] target,int[] bar) {
		int n=bar.length;
		ArrayList<Integer> zer=new ArrayList<Integer>();
		ArrayList<Integer> one=new ArrayList<Integer>();
		for (int i=0;i<n;i++) {
			if (bar[i]==0)
				zer.add(i);
			else
				one.add(i);
		}
		int[] hor=new int[n];
		Arrays.fill(hor, -1);
		int[] vert=new int[n];
		Arrays.fill(vert,-1);
		// start with zero x zero
		
		for (int i:zer)
			for (int j:zer)
				if (map[i][j]!=target[i][j]) {
					//log("error on zero x zero");
					return false;
				}
		
		for (int i:zer)
			for (int j:one) {
				if (map[i][j]==target[i][j]) {
					if (hor[i]==1) {
						//log("case 1 // error on one x zer i:"+(i+1)+" j:"+(j+1));
						return false;
					}
					hor[i]=0;
				} else {
					if (hor[i]==0) {
						//log("case 2 // error on one x zer i:"+(i+1)+" j:"+(j+1));
						return false;
					}
					hor[i]=1;
				}
			}

		for (int i:one)
			for (int j:zer) {
				if (map[i][j]==target[i][j]) {
					if (vert[j]==1) {
						//log("case 3 // error on one x zer i:"+(i+1)+" j:"+(j+1));
						return false;
						
					}
					vert[j]=0;
				} else {
					if (vert[j]==0) {
						//log("case 4 // error on one x zer i:"+(i+1)+" j:"+(j+1));
						return false;
					}
					vert[j]=1;
				}
			}

		if (one.size()>0) {
			// do we have already one that is shet
			int tgi=-1;
			int tgj=-1;
			for (int x:one)
				if (hor[x]>-1 ||vert[x]>-1) {
					if (hor[x]>-1) {
						tgi=x;
						break;
					}
					tgj=x;
					break;
				}

			if ((tgi>-1) ||(tgj>-1)) {
				//log("origin // tgi:"+tgi+" tgj:"+tgj);
				boolean b=complete(map,target,hor,vert,one,tgi,tgj);
				if (b==false)
					return false;
				
				
			} else {
				//log("we have no info...");
				// we have no info...
			
				int id=one.get(0);
				tgi=id;
				hor[tgi]=1;
				
				//log("origin 2 // tgi:"+tgi+" tgj:"+tgj);
				boolean b=complete(map,target,hor,vert,one,tgi,tgj);
				if (b==false) {
					//log("fail after 1st attempt");
					Arrays.fill(hor, -1);
					Arrays.fill(vert, -1);
					hor[tgi]=0;
					//log("origin 3 // tgi:"+tgi+" tgj:"+tgj);
					 b=complete(map,target,hor,vert,one,tgi,tgj);
					 if (b==false)
						 return false;
					
				}
				
				
			}

		}
		
		int cnt=0;
		for (int x=0;x<n;x++) {
			if (hor[x]==1)
				cnt++;
			if (vert[x]==1)
				cnt++;
		}
		output(cnt);
		for (int x=0;x<n;x++) {
			if (hor[x]==1)
				output("row "+(x));
			if (vert[x]==1)
				output("col "+(x));
		}
		return true;
	}

	static final class InputReader {
		private final InputStream stream;
		private final byte[] buf = new byte[1024];
		private int curChar;
		private int numChars;

		public InputReader(InputStream stream) {
			this.stream = stream;
		}

		private int read() throws IOException {
			if (curChar >= numChars) {
				curChar = 0;
				numChars = stream.read(buf);
				if (numChars <= 0) {
					return -1;
				}
			}
			return buf[curChar++];
		}

		public final String readString() throws IOException {
			int c = read();
			while (isSpaceChar(c)) {
				c = read();
			}
			StringBuilder res = new StringBuilder();
			do {
				res.append((char) c);
				c = read();
			} while (!isSpaceChar(c));
			return res.toString();
		}

		public final int readInt() throws IOException {
			int c = read();
			boolean neg = false;
			while (isSpaceChar(c)) {
				c = read();
			}
			char d = (char) c;
			// log("d:"+d);
			if (d == '-') {
				neg = true;
				c = read();
			}
			int res = 0;
			do {
				res *= 10;
				res += c - '0';
				c = read();
			} while (!isSpaceChar(c));
			// log("res:"+res);
			if (neg)
				return -res;
			return res;

		}

		public final long readLong() throws IOException {
			int c = read();
			boolean neg = false;
			while (isSpaceChar(c)) {
				c = read();
			}
			char d = (char) c;
			// log("d:"+d);
			if (d == '-') {
				neg = true;
				c = read();
			}
			long res = 0;
			do {
				res *= 10;
				res += c - '0';
				c = read();
			} while (!isSpaceChar(c));
			// log("res:"+res);
			if (neg)
				return -res;
			return res;

		}

		private boolean isSpaceChar(int c) {
			return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
		}
	}
