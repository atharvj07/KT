import java.util.*;

public class helloWorld 
{
	public static void main(String[] args) 
	{		
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();

		String ans = "YES";
		int h = 1000000000;
		for(int i = 0; i < n; i++) {
			int a = in.nextInt();
			int b = in.nextInt();
			if(a > b) {
				int temp = a;
				a = b;
				b = temp;
			}
			if(b <= h)
				h = b;
			else if(a <= h)
				h = a;
			else {
				ans = "NO";
				break;
			}
		}
			
		System.out.println(ans);
		
		in.close();
	}
}
