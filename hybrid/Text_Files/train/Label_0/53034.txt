import java.io.*;
import java.math.*;
import java.util.*;
public class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		boolean[] got = new boolean[n];
		int j = n-1;
		for(int i=0;i<n;i++){
			got[in.nextInt()-1] = true;
			while(j >= 0 && got[j]){
				System.out.print(j+1 + " ");
				j--;
			}
			System.out.println();
		}
	}
	
}
