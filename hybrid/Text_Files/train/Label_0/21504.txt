#include <iostream>
using namespace std;

int main(){

  int n, a, q, m;
  int dp[2001] = {};

  cin >> n;
  dp[0] = 1;
  for(int i = 0; i < n; i++){
    cin >> a;
    for(int j = 2000; j >= 0; j--){
      if(dp[j]) dp[j + a] = 1;
    }
  }
  cin >> q;
  for(int i = 0; i < q; i++){
    cin >> m;
    cout << (dp[m] ? "yes" : "no") << endl;
  }
  return 0;
}
