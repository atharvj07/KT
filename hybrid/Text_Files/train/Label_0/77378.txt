def factorise(n):
	dic = {}
	while n % 2 == 0:
		if 2 in dic:
			dic[2] += 1
		else:
			dic[2] = 1
		n = n / 2
	for i in range(3, int(n ** 0.5) + 1):
		while n % i == 0:
			if i in dic:
				dic[i] += 1
			else:
				dic[i] = 1
			n = n / i
	if n > 2:
		dic[n] = 1
	return dic
(a, b) = map(int, input().split())
fac_a = factorise(a)
fac_b = factorise(b)
all_fac = list(set(list(fac_a.keys()) + list(fac_b.keys())))
all_fac.sort()
final = True
l_fac = len(all_fac)
t = 0
for i in range(l_fac):
	if all_fac[i] != 2 and all_fac[i] != 3 and (all_fac[i] != 5):
		if all_fac[i] in fac_a and all_fac[i] in fac_b:
			if fac_a[all_fac[i]] == fac_a[all_fac[i]]:
				final = final and True
			else:
				final = final and False
		else:
			final = final and False
	elif all_fac[i] in fac_a and all_fac[i] in fac_b:
		t += abs(fac_a[all_fac[i]] - fac_b[all_fac[i]])
	else:
		try:
			t += fac_a[all_fac[i]]
		except:
			t += fac_b[all_fac[i]]
if final:
	print(t)
else:
	print(-1)
