t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	x = sorted(a)
	ans = 0
	while x != a:
		ans += 1
		if ans % 2:
			for i in range(0, n, 2):
				if i + 1 < n and a[i] > a[i + 1]:
					(a[i], a[i + 1]) = (a[i + 1], a[i])
		else:
			for i in range(1, n, 2):
				if i + 1 < n and a[i] > a[i + 1]:
					(a[i], a[i + 1]) = (a[i + 1], a[i])
	print(ans)
