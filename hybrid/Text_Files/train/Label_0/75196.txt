from collections import Counter, defaultdict, deque
from math import factorial as fact
n = int(input())
arr = [int(x) for x in input().split()]
c = defaultdict(int)
fmax = 0
smax = 0
s = 0
for el in arr:
	c[el] += 1
	s += el
	if el > fmax:
		smax = fmax
		fmax = el
	elif el > smax:
		smax = el
p1 = 0
p2 = 0
if s - 2 * fmax == fmax:
	if c[fmax] >= 2:
		p1 = s - 2 * fmax
else:
	p1 = s - 2 * fmax
if (s - fmax) % 2 == 0:
	if (s - fmax) // 2 == fmax:
		if c[fmax] >= 2:
			p2 = fmax
	elif c[(s - fmax) // 2] >= 1:
		p2 = fmax
ans = []
for i in range(n):
	if arr[i] == p1 or arr[i] == p2:
		ans.append(i + 1)
print(len(ans))
print(*ans)
