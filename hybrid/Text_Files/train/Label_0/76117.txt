import sys
import math

def input():
	return sys.stdin.readline().strip()

def iinput():
	return int(input())

def rinput():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))
n = iinput()
count = 0
for i in range(1, n):
	for j in range(1, n):
		c = math.sqrt(i * i + j * j)
		if int(c) == c and c <= n:
			count += 1
print(count // 2)
