	static class MagicComponents{
		// start: 13:54
		int num, n, edges;
		int[] dfsNum, low, vis;
		ArrayList<Integer> cuts, leaves;
		ArrayList<Edge> bridges, adj[];
		ArrayList<ArrayList<Edge>> bccs;
		ArrayDeque<Edge> eStack;
		MagicComponents(int N){
			adj = new ArrayList[n=N];
			for(int i=0;i<n;i++){
				adj[i] = new ArrayList<>();
			}
			edges = 0;
		}
		void addEdge(int u, int v, boolean ha){
			adj[u].add(new Edge(u,v,edges, ha));
			adj[v].add(new Edge(v,u,edges, ha));
		}
		void run(){
			vis = new int[n];
			dfsNum = new int[n];
			low = new int[n];
			bridges = new ArrayList<>();
			cuts = new ArrayList<>();
			leaves = new ArrayList<>();
			bccs = new ArrayList<>();
			eStack = new ArrayDeque<>();
			num = 0;
			for(int i = 0; i < n; i++){
				if(adj[i].size() == 1){
					leaves.add(i);
				}
				if(vis[i] != 0) continue;
				dfs(i,-1);
			}
		}
		void dfs(int node, int par){
			dfsNum[node] = low[node] = num++;
			vis[node] = 1;
			int nChild = 0;
			for(Edge e : adj[node]){
				if(e.v == par) continue;
				if(vis[e.v] == 0){
					++nChild; eStack.push(e); dfs(e.v, node);
					low[node] = Math.min(low[node], low[e.v]);
					if(low[e.v] >= dfsNum[node]){
						if(dfsNum[node] > 0 || nChild > 1){
							cuts.add(node);
						}
						if(low[e.v] > dfsNum[node]){
							bridges.add(e); pop(node);
						}
						else pop(node);
					}
				}
				else if(vis[e.v] == 1){
					low[node] = Math.min(low[node], dfsNum[e.v]);
					eStack.push(e);
				}
			}
			vis[node] = 2;
		}
		void pop(int u){
			ArrayList<Edge> list = new ArrayList<>();
			for(;;){
				Edge e = eStack.pop(); list.add(e);
				if(e.u == u) break;
			}
			bccs.add(list);
		}

		boolean[] singleSizeBcc, artifactReachable;
		MagicComponents bccTree(int[] map){
			int[] cut = new int[n];
			Arrays.fill(cut, -1);
			int size = bccs.size();
			for(int i : cuts) map[i] = cut[i] = size++;
			for(int i : leaves) map[i] = cut[i] = size++;
			MagicComponents g = new MagicComponents(size);
			g.singleSizeBcc = new boolean[size];
			g.artifactReachable = new boolean[size];
			int[] used = new int[n];
			for(int i = 0; i < bccs.size(); i++){
				ArrayList<Edge> list = bccs.get(i);
				g.singleSizeBcc[i] = list.size() == 1;
				for(Edge e : list){
					g.artifactReachable[i] =(g.artifactReachable[i] || e.hasArtifact);
					for(int node : new int[]{e.u, e.v}){
						if(used[node] != i+1){
							used[node] = i+1;
							if(cut[node] != -1){
								g.addEdge(i, cut[node], false);
							}
							else map[node] = i;
						}
					}
				}
			}
			return g;
		}
	}
	
	static void fillSpaces(int prev, int u) {
		if(mc.artifactReachable[u] && prev != -1) return;
		mc.artifactReachable[u] = true;
		for(Edge e : mc.adj[u]) {
//			if(e.isBridge) continue;
			int v = e.other(u);
			if(mc.singleSizeBcc[v]) continue;
			if(v == prev) continue;
			fillSpaces(u, v);
		}
	}
	static int end;
	static int find(int prev, int u) {
		if(u == end) {
			if(mc.artifactReachable[u]) return 2;
			return 1;
		}
		for(Edge e : mc.adj[u]) {
			int v = e.other(u);
			if(v == prev) continue;
			int ret = find(u, v);
			if(ret > 0) {
				if(ret == 2 || (ret == 1 && mc.artifactReachable[u]))
					return 2;
				return 1;
			}
		}
		return 0;
	}
