q = int(input())

def binarySearch(arr, l, r, x):
	if 1 <= len(arr[l:r + 1]) <= 2:
		return l
	if r >= l:
		mid = int(l + (r - l) / 2)
		if arr[mid] == x:
			return mid
		elif arr[mid] > x:
			return binarySearch(arr, l, mid - 1, x)
		else:
			return binarySearch(arr, mid + 1, r, x)
	else:
		return -1

def solve():
	(n, pr, k) = map(int, input().split())
	L = list(map(int, input().split()))
	L.sort()
	dp = [0] * n
	ans = 0
	for i in range(0, k - 1):
		dp[i] += dp[i - 1] + L[i]
	for i in range(k - 1, n):
		if i == k - 1:
			dp[i] += L[i]
		else:
			dp[i] += dp[i - k] + L[i]
	for i in range(n):
		if dp[i] <= pr:
			ans = i + 1
	print(ans)
while q:
	solve()
	q -= 1
