#include <bits/stdc++.h>
using namespace std;
set<char> res;
int aset = 0;
void matchWord(string g, string t) {
  set<char> temp;
  set<char> prst;
  int valid = 1;
  for (int i = 0; i < g.size(); i++) {
    if (g[i] != '*') {
      prst.insert(g[i]);
      if (g[i] != t[i]) valid = 0;
    } else if (g[i] == '*')
      temp.insert(t[i]);
  }
  for (auto element : temp) {
    for (auto ele : prst)
      if (element == ele) valid = 0;
  }
  if (valid) {
    if (aset == 0 && res.size() == 0 && temp.size() != 0) res = temp;
    set<char> intersect;
    set_intersection(res.begin(), res.end(), temp.begin(), temp.end(),
                     inserter(intersect, intersect.begin()));
    res = intersect;
    if (res.size() == 0) aset = 1;
  }
}
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  string g;
  cin >> g;
  int m;
  cin >> m;
  vector<string> t(m);
  for (int i = 0; i < t.size(); i++) {
    cin >> t[i];
    matchWord(g, t[i]);
  }
  cout << res.size() << endl;
  return 0;
}
