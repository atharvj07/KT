ip = lambda : [int(i) for i in input().split()]
(b, q, l, m) = ip()
bad = set(ip())
ans = 0
iter_count = 0
inf = False
while True:
	if iter_count >= 40:
		inf = True
		break
	if abs(b) > l:
		break
	if b not in bad:
		ans += 1
	b *= q
	iter_count += 1
if inf and ans > 2:
	print('inf')
else:
	print(ans)
