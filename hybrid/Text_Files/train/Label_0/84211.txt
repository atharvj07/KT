import math
t = int(input())
while t != 0:
	(a, b) = map(int, input().split())
	diff = max(a, b) - min(a, b)
	if diff % 2 == 0:
		x = math.ceil((math.sqrt(1 + 8 * diff) - 1) / 2)
		if x * (x + 1) / 2 % 2 != 0:
			while x * (x + 1) / 2 % 2 != 0:
				x += 1
		print(x)
	else:
		x = math.ceil((math.sqrt(1 + 8 * diff) - 1) / 2)
		if x * (x + 1) / 2 % 2 == 0:
			while x * (x + 1) / 2 % 2 == 0:
				x += 1
		print(x)
	t -= 1
