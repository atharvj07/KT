ll POW(ll a, ll b) {
  ll res = 1;
  do {
    if (b % 2) tmod(res, a);
    tmod(a, a);
  } while (b >>= 1);
  return res;
}
const int MXN = 100000;
const int N = MXN + 10;
struct node {
  int val;
  node *lch, *rch;
  node() {
    val = 0;
    lch = nullptr;
    rch = nullptr;
  }
  void pull() {
    val = 0;
    if (lch) amax(val, lch->val);
    if (rch) amax(val, rch->val);
  }
  void modify(int x, int v, int lb, int rb) {
    if (lb == rb) {
      val = v;
      return;
    }
    if (!lch) lch = new node();
    if (!rch) rch = new node();
    int mid = lb + rb >> 1;
    if (x <= mid) lch->modify(x, v, lb, mid);
    if (mid < x) rch->modify(x, v, mid + 1, rb);
    pull();
  }
  int query(int l, int r, int lb, int rb) {
    if (r < l) return 0;
    if (l <= lb && rb <= r) return val;
    int mid = lb + rb >> 1;
    int ret = 0;
    if (l <= mid && lch) amax(ret, lch->query(l, r, lb, mid));
    if (mid < r && rch) amax(ret, rch->query(l, r, mid + 1, rb));
    return ret;
  }
};
void solve() {
  int n, m;
  cin >> n >> m;
  vec<node *> dp(n + 1);
  for (int i = 1; i <= (n); ++i) dp[i] = new node();
  int a, b, val;
  while (m--) {
    cin >> a >> b >> val;
    int k = dp[a]->query(0, val - 1, 0, MXN);
    dp[b]->modify(val, k + 1, 0, MXN);
  }
  int ans = 0;
  for (int i = 1; i <= (n); ++i) amax(ans, dp[i]->query(0, MXN, 0, MXN));
  cout << ans << '\n';
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int T = 1;
  while (T--) {
    solve();
  }
}
