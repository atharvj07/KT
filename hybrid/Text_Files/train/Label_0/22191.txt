#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
const int INF = INT_MAX;
const int MOD = 1000000007;
const double res = 1e-9;
int n;
vector<pair<double, double> > a;
bool f(double t) {
  double l = a[0].first, r = l + a[0].second * t;
  for (int i = 1; i < n; ++i) {
    l = max(l, a[i].first - a[i].second * t);
    r = min(r, a[i].first + a[i].second * t);
    if (l > r) return 0;
  }
  return 1;
}
signed main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; ++i) cin >> a[i].first;
  for (int i = 0; i < n; ++i) cin >> a[i].second;
  sort(a.begin(), a.end());
  cout << setprecision(6) << fixed;
  double l = 0, r = 1e14;
  for (int i = 0; i < 100; i++) {
    double m = (l + r) / 2;
    if (f(m))
      r = m;
    else
      l = m;
  }
  cout << r;
  return 0;
}
