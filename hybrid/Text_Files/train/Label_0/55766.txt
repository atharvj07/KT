import java.util.Scanner;

public class B270 {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int[] A = new int[N];
        for (int n=0; n<N; n++) {
            A[n] = in.nextInt();
        }
        int pos = N-1;
        while (pos > 0 && A[pos-1] < A[pos]) pos--;
        System.out.println(pos);
    }

}
