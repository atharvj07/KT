import java.util.Scanner;


public class E {
    
    static double e = 1e-8, R, phi0;
    static int x,y,r,v,vp; 
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xp = sc.nextInt();
        int yp = sc.nextInt();
        vp = sc.nextInt();
        x = sc.nextInt();
        y = sc.nextInt();
        v = sc.nextInt();
        r = sc.nextInt();
        R = Math.sqrt(xp*xp+yp*yp);
        phi0 = Math.acos(xp / R);
        if (yp < 0)
            phi0 = 2*Math.PI-phi0;
        double left = 0, right = 1e9;
        for(;;) {
            double middle = (left+right) / 2;
            if (reache(middle) && !reache(middle-e)) {
                System.out.println(middle);
                return;
            }
            if (reache(middle))
                right = middle-e;
            else
                left = middle+e;
        }
    }

    private static boolean reache(double t) {
        double x2 = R*Math.cos(vp/R*t+phi0);
        double y2 = R*Math.sin(vp/R*t+phi0);
        double x1 = x, y1 = y;
        double a = y1-y2, b = x2-x1, c = x1*y2-x2*y1;
        double d = Math.abs(c) / Math.sqrt(a*a+b*b);
        double reach_time = 0;
        a = dis(x1, y1, 0, 0);
        b = dis(x2, y2, 0, 0);
        c = dis(x1, y1, x2, y2);
        if (d >= r || Math.abs(Math.sqrt(a*a-d*d)+Math.sqrt(b*b-d*d)-c) > e) {
            reach_time = c / v;
        }
        else {
            double alfa = Math.acos((x1*x2+y1*y2) / (a*b))-Math.acos(r/a)-Math.acos(r/b);
            reach_time = (r*alfa+Math.sqrt(a*a-r*r)+Math.sqrt(b*b-r*r)) / v;
        }
        return reach_time <= t;
    }

    private static double dis(double x1, double y1, double x2, double y2) {
        return Math.sqrt((x2-x1)*(x2-x1)+(y2-y1)*(y2-y1));
    }
}
