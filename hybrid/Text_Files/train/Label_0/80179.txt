n = int(input())
l = []
for i in range(n):
	l1 = input().split()
	l.append(l1)
p = 0
r = []
for i in range(n):
	s = 0
	k = 0
	for j in range(n):
		if i != j:
			if l[i][j] == '0' or l[i][j] == '2':
				s += 1
			else:
				k = 1
				break
	if k == 0 and s == n - 1:
		p += 1
		r.append(i + 1)
print(p)
r1 = len(r)
for i in range(r1):
	print(r[i], '', end='')
