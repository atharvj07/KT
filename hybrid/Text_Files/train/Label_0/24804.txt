#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<int,int> ii;
typedef vector<ii> vii;
typedef vector<ll> vi;
const int Nmaxx = 1e5+5;
vector<vii> g(Nmaxx);
vi ans(Nmaxx,-1);

void dfs(int u, int p, int skip)
{
	int c = 1;
	for(ii b : g[u])
	{
		if(b.first==p)
			continue;
		if(skip==c)
			c++;
		ans[b.second] = c;
		dfs(b.first, u, c);
		c++;
	}
}



int main()
{
	int n,k=0; cin>>n;
	for(int i =1 ; i <n ; i++)
	{
		int a,b; cin>>a>>b;
		g[a].push_back({b,i});
		g[b].push_back({a,i});
		k = max(k,(int)max(g[a].size(),g[b].size()));
	}

	cout<<k<<endl;
	dfs(1,0,0);

	for(int i = 1 ;i <n ;i++)
		cout<<ans[i]<<endl;
	


	return 0;
}
