def mi(s):

def js(lst):

def dist(x, y):
	return ((x[0] - y[0]) ** 2 + (x[1] - y[1]) ** 2) ** 0.5

def fast_exp(x, n):
	if n == 0:
		return 1
	elif n % 2 == 1:
		return x * fast_exp(x, (n - 1) // 2) ** 2
	else:
		return fast_exp(x, n // 2) ** 2

def in_range(mat, i, j):
	return i >= 0 and i < len(mat) and (j >= 0) and (j < len(mat[i]))

def cycle(mat, i, j, length, first_i, first_j, path):
	if (i, j) == (first_i, first_j) and length >= 4:
		return True
	elif (i, j) == (first_i, first_j) and length > 0:
		return False
	if (i, j) in path:
		return False
	if not in_range(mat, i, j) or mat[i][j] != mat[first_i][first_j]:
		return False
	path.add((i, j))
	found = any([cycle(mat, i + 1, j, length + 1, first_i, first_j, path), cycle(mat, i - 1, j, length + 1, first_i, first_j, path), cycle(mat, i, j - 1, length + 1, first_i, first_j, path), cycle(mat, i, j + 1, length + 1, first_i, first_j, path)])
	return found
