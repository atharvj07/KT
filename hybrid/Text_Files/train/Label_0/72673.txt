n = int(input())
a = [0] + list(map(int, input().split()))
b = list(map(int, input().split())) + [0]
arr = list(map(int, input().split()))
for i in range(1, n):
	a[i] += a[i - 1]
	b[-i - 1] += b[-i]
m = 10 ** 12
for i in range(n):
	for j in range(i + 1, n):
		m = min(a[i] + b[i] + arr[i] + a[j] + b[j] + arr[j], m)
print(m)
