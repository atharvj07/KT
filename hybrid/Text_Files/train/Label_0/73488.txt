t = int(input())
while t:
	s = input()
	(l, r, u, d) = (0, 0, 0, 0)
	for i in s:
		if i == 'U':
			u += 1
		if i == 'D':
			d += 1
		if i == 'R':
			r += 1
		if i == 'L':
			l += 1
	mlr = min(l, r)
	mud = min(u, d)
	l = mlr
	r = mlr
	u = mud
	d = mud
	if mud != 0 and mlr == 0:
		print(2)
		print('UD')
	elif mlr != 0 and mud == 0:
		print(2)
		print('LR')
	elif mlr == 0 and mud == 0:
		print(0)
		print()
	else:
		print(l + r + u + d)
		print('L' * l + 'U' * u + 'R' * r + 'D' * d)
	t -= 1
