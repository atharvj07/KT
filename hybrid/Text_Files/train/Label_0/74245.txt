from sys import stdin, stdout
import math
import bisect
input = stdin.readline

def print(x='', y='', end='\n'):
	if y == '':
		stdout.write(str(x) + end)
	else:
		stdout.write(str(x) + ' ' + str(y) + end)

def solve():
	n = int(input())
	(s1, s2) = (0, 0)
	flag = 0
	for i in range(n):
		(x, y) = map(int, input().split())
		s1 += x
		s2 += y
		if flag == 0:
			flag = 1 if x % 2 != y % 2 else 0
	if s1 % 2 == 0 and s2 % 2 == 0:
		print(0)
	elif s1 % 2 == s2 % 2 and flag:
		print(1)
	else:
		print(-1)
tt = 1
for __ in range(tt):
	solve()
