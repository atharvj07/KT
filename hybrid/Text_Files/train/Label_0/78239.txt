def main():
	(n, k) = map(int, input().split())
	v = list(map(int, input().split()))
	d = sum(v) / k
	j = 0
	res = ''
	sumpart = 0
	for i in range(len(v)):
		sumpart += v[i]
		if sumpart == d:
			res += str(i + 1 - j) + ' '
			j = i + 1
			sumpart = 0
			if j >= len(v):
				break
		elif sumpart > d:
			print('No')
			return
	res.rstrip()
	print('Yes')
	print(res)
main()
