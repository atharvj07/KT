import sys
import collections
import itertools as it

def readArray(type=int):
	line = input()
	return [type(x) for x in line.split()]

def getFreq(arr):
	freq = collections.defaultdict(int)
	for x in arr:
		freq[x] += 1
	return freq

def solve():
	(a, b, c, n) = readArray()
	total = a + b - c
	cc = 0
	if c > a or c > b:
		cc = -1
	elif total < 0:
		cc = -1
	elif total > n - 1:
		cc = -1
	else:
		cc = n - total
	print(cc)
solve()
