#include <iostream>
#include <algorithm>
#include <string>
using namespace std;

int ans(string s, int n, int pp)
{
    int m = count(s.begin(), s.end(), 'G');

    int t = 0;
    for (int turn=0 ;; ++turn) {
      if (s.empty()) break;
      t = (t+n+(turn==pp)) % s.length();
      if (s[t]=='G') {
	--m;
	if (m==0) {
	  s.erase(s.begin()+t);
	  break;
	}
      }
      s.erase(s.begin()+t);
    }

    return s.length();
}

int main()
{
  int T;
  for (cin>>T; T>0; --T) {
    int n;
    string s;
    cin >> n >> s;
    --n;

    int x = 0;
    for (int i=0; i < s.length(); ++i) {
      x = max(x, ans(s, n, i));
    }
    cout << x << endl;
  }
  return 0;
}
