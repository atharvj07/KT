#include <bits/stdc++.h>
using namespace std;
template <typename Arg1>
void __f(const char* name, Arg1&& arg1) {
  cerr << name << " : " << arg1 << std::endl;
}
template <typename Arg1, typename... Args>
void __f(const char* names, Arg1&& arg1, Args&&... args) {
  const char* comma = strchr(names + 1, ',');
  cerr.write(names, comma - names) << " : " << arg1 << " | ";
  __f(comma + 1, args...);
}
int cnt[256];
bool ok[256];
int add[256];
int main() {
  int n;
  scanf("%d", &n);
  string s;
  cin >> s;
  for (int i = (0); i < (256); ++i) ok[i] = true;
  for (auto c : s)
    if (c != '*') ok[c] = false;
  int m;
  scanf("%d", &m);
  int good = 0;
  for (int i = (0); i < (m); ++i) {
    string t;
    cin >> t;
    if ((int)(t.size()) != (int)(s.size())) continue;
    bool fucked = false;
    for (int i = (0); i < ((int)(s.size())); ++i)
      fucked |= (s[i] == '*' && !ok[t[i]]) || (s[i] != '*' && s[i] != t[i]);
    if (fucked) continue;
    good++;
    memset(add, 0, sizeof(add));
    for (int i = (0); i < ((int)(t.size())); ++i)
      if (s[i] == '*') add[t[i]] = 1;
    for (int i = (0); i < (256); ++i) cnt[i] += add[i];
  }
  int ans = 0;
  for (int i = (0); i < (256); ++i)
    if (cnt[i] == good) ans++;
  printf("%d\n", ans);
  return 0;
}
