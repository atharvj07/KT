import java.util.*;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        String s= sc.next();
        int start=0;
        int moves=0;
        for (int i = 0; i < s.length(); i++) {
            int index= s.charAt(i)-97;
            int walk= Math.abs(index-start);
            if(walk<13) {
                moves+=walk;
            }
            else {
                moves+=26-walk;
            }
            start=index;
        }
        System.out.println(moves);
    }
}
