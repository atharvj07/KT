	static void count(int[] kol) {
		//System.out.println("!");
	    long a = 0;
		for (int i = 0; i < n; i++) a = a*n + x[i];
		set.add(a);
		
		int c = 0;
		for (int i = 0; i < n; i++) if (kol[i] > 0) c++;
		if (c < 2) return;

		for (int i = 0; i < n; i++) if (kol[i] > 0) {
			for (int j = i + 1; j < n; j++) if (kol[j] > 0) {
				int[] kol2 = new int[n];
				for (int t = 0; t < n; t++) kol2[t] = kol[t];
				kol2[i]--;
				kol2[j]--;
				int temp = x[i];
				x[i] = x[j];
				x[j] = temp;
				count(kol2);
				temp = x[i];
				x[i] = x[j];
				x[j] = temp;
			}
		}
	}

		n = next();

		int[] kol = new int[n];
		for (int i = 0; i < n; i++) kol[i] = next();
		int ones = 0;
		int twos = 0;
		for (int i = 0; i < n; i++)
			if (kol[i] == 1) ones++;
			else twos++;

		int mod = 1000000007;

		long[][] d = new long[n + 1][n + 1];
		int[][] c = new int[n + 1][n + 1];
		for (int i = 0; i <= n; i++) c[i][0] = c[i][i] = 1;
		for (int i = 0; i <= n; i++)
			for (int j = 1; j < i; j++) c[i][j] = (c[i - 1][j - 1] + c[i - 1][j]) % mod;

		int[] f = new int[n + 1];
		f[0] = 1;
		for (int i = 0; i < n; i++) f[i + 1] = (int)((1L * f[i] * (i + 1)) % mod);

		int[][] a = new int[n + 1][n + 1];
		for (int i = 0; i < n; i++)
			for (int j = 0; j <= i; j++) a[i][j] = (int)(c[i][j] * 1L * f[j] % mod);

		for (int i = 0; i <= ones; i++)
			for (int j = 0; j <= twos; j++) {
				if (i <= 2) d[i][j] = f[i + j];
				else {
					d[i][j] = d[i - 1][j];
					for (int k = 1; k <= j; k++)
						d[i][j] += (a[j][k] * d[i - 1][j - k]) % mod;
					for (int k = 0; k <= j; k++)
						d[i][j] += 1L * (i - 1) * a[j][k] * (k + 1) % mod * d[i - 2][j - k] % mod;
					d[i][j] %= mod;
				}
			}
