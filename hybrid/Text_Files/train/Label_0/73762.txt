n = int(input())
l = list(map(int, input().split()))
ans = [0] * 65
bans = [0] * n
for i in range(n):
	b = l[i]
	score = 0
	while b % 2 == 0:
		b //= 2
		score += 1
	ans[score] += 1
	bans[i] = score
k = max(ans)
ind = ans.index(k)
print(n - k)
ta = [l[i] for i in range(n) if bans[i] != ind]
print(*ta)
