import java.util.Scanner;

public class C {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int ra = (a-1)/m;
        int rb = (b-1)/m;
        int pa = (a-1)%m;
        int pb = (b-1)%m;
        int ans = 3;
        if(m==1)ans = 1;
        else if(ra==rb)ans = 1;
        else if(rb-ra==1){
            if(pa==0 && (pb==m-1||b==n))ans = 1;
            else ans = 2;
        }
        else{
            if(pa==0){
                if((pb==m-1||b==n))
                    ans = 1;
                else ans = 2;
            }
            else if(pb==m-1||b==n||pb==pa-1){
                ans = 2;
            }
            else {
                ans = 3;
            }
            
        }
        System.out.println(ans);
    }
}
