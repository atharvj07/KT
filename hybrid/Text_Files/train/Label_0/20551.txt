#include<bits/stdc++.h>
using namespace std;
int main(void)
{
	long long n, q;
	cin >> n >> q;
	vector<stack<long long> >res(n);
	while (q--)
	{
		int op;
		cin >> op;
		if (op == 0)
		{
			long long t, x;
			cin >> t >> x;
			res[t].push(x);
		}
		else if (op == 1)
		{
			long long t;
			cin >> t;
			if (!res[t].empty())
			cout << res[t].top() << endl;
		}
		else if (op == 2)
		{
			long long t;
			cin >> t;
			if(!res[t].empty())
			res[t].pop();
		}
	}
	return 0;
}
