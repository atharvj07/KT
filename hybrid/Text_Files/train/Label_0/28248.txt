#include <bits/stdc++.h>
using namespace std;
map<int, int> mp;
int main() {
  int n, m, k, x;
  cin >> n >> m >> k;
  for (int i = 0; i < n; i++) cin >> x, mp[-x]++;
  for (int i = 0; i < m; i++) cin >> x, mp[-x]--;
  int ps = 0;
  bool ok = 0;
  for (map<int, int>::iterator it = mp.begin(); it != mp.end(); it++) {
    ps += it->second;
    if (ps > 0) ok = 1;
  }
  if (ok)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
