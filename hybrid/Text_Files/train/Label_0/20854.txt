#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
const long long mod = 1e9 + 7;
int n, m, num[30];
char s[N];
vector<int> g;
string solve(int x) {
  g.clear();
  g.push_back(0);
  for (int i = (1); i < (n + 1); i++) {
    if (s[i] - 'a' < x) {
      g.push_back(i);
    }
  }
  g.push_back(n + 1);
  int cnt = 0;
  string ans = "";
  for (int i = 1; i < (int)g.size(); ++i) {
    if (g[i] - g[i - 1] - 1 < m) continue;
    int j = g[i];
    while (j - g[i - 1] - 1 >= m) {
      int tj = max(j - m, 1);
      for (; tj < j; tj++) {
        if (s[tj] == 'a' + x) break;
      }
      if (tj == j) return ans;
      j = tj;
      cnt++;
    }
  }
  for (int i = (0); i < (x); i++) {
    for (int j = (0); j < (num[i]); j++) ans += 'a' + i;
  }
  while (cnt--) {
    ans += 'a' + x;
  }
  return ans;
}
int main() {
  cin >> m;
  cin >> (s + 1);
  n = strlen(s + 1);
  for (int i = (1); i < (n + 1); i++) num[s[i] - 'a']++;
  for (int i = (0); i < (26); i++) {
    if (solve(i) != "") {
      cout << solve(i) << "\n";
      return 0;
    }
  }
  return 0;
}
