#include <bits/stdc++.h>
using namespace std;
int data[2010][2010];
int main() {
  int n, m;
  scanf("%d", &n);
  scanf("%d", &m);
  string s[2010];
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }
  map<int, int> Count;
  vector<int> check;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (s[i][j] == '1') {
        data[i + 1][j + 1] = 1;
        Count[j + 1]++;
      }
    }
  }
  bool yo = false;
  for (int i = 1; i <= n; i++) {
    int cnt = 0, cnt1 = 0;
    for (int j = 1; j <= m; j++) {
      if (data[i][j] == 1) {
        if (Count[j] > 1) cnt++;
        cnt1++;
      }
    }
    if (cnt == cnt1) {
      yo = true;
      break;
    }
  }
  if (yo)
    cout << "YES\n";
  else
    cout << "NO\n";
  return 0;
}
