import sys
try:
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')
except:
	pass
input = sys.stdin.readline
t = 1
t = int(input())
while t:
	t -= 1
	(a, b, k) = map(int, input().split())
	a1 = list(map(int, input().split()))
	b1 = list(map(int, input().split()))
	(nota, notb) = ({}, {})
	for i in range(k):
		if a1[i] in nota:
			nota[a1[i]].add(i)
		else:
			nota[a1[i]] = set([i])
		if b1[i] in notb:
			notb[b1[i]].add(i)
		else:
			notb[b1[i]] = set([i])
	ans = 0
	for i in range(k):
		ans += k - len(nota[a1[i]]) - len(notb[b1[i]]) + 1
	print(ans // 2)
