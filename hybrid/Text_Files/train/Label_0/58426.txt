         {

                for(int i=0;i<n;i++)
                     ar[i] = a[i];
                Arrays.sort(ar);
                HashMap<Integer,Integer> hm = new HashMap();
                int st=1;
                for(int i=0;i<n;i++)

                        hm.put(ar[i],st++);

                   a[i] = hm.get(a[i]);
         }
