n = int(input())
lista = []
for _ in range(n):
	flag = True
	coordenates = input().split()
	for i in range(int(coordenates[0]), int(coordenates[1]) + 1):
		if not flag:
			break
		for j in range(int(coordenates[2]), int(coordenates[3]) + 1):
			if not flag:
				break
			if i != j:
				lista.append('{} {}'.format(i, j))
				flag = False
print('\n'.join(lista))
