for tc in range(int(input())):
	(n, k) = map(int, input().split())
	h = list(map(int, input().split()))
	if k > 10000:
		print(-1)
		continue
	k -= 1

	def fun():
		global k
		(cha, diao) = (0, 0)
		for i in range(n - 1):
			if h[i] < h[i + 1]:
				k -= 1
				h[i] += 1
				return i + 1
		k -= 1
		return -1
	while 1:
		if k == 0:
			break
		fun()
	print(fun())
