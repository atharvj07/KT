#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  vector<vector<int>> l(200001);
  for (int i = 0; i < n; ++i) {
    int a;
    scanf("%d", &a);
    int cnt = 0;
    for (; a > 0; ++cnt) {
      l[a].emplace_back(cnt);
      a >>= 1;
    }
    l[a].emplace_back(cnt);
  }
  for (int i = 0; i < l.size(); ++i) sort(l[i].begin(), l[i].end());
  int ans = 1000000000;
  for (int i = 0; i <= 200000; ++i) {
    if (l[i].size() >= k) {
      int sum = 0;
      for (int j = 0; j < k; ++j) sum += l[i][j];
      ans = min(ans, sum);
    }
  }
  printf("%d", ans);
  return 0;
}
