

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

/**
 * Class http://codeforces.com/problemset/problem/1007/C
 *
 * @author edzabarov
 * @since 24.07.2018
 */
public class MainClass {

    private BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));


    public static void main(String[] args) {
        new MainClass().start();
    }

    private void start() {
        long n = scanInt();
        long x, y, x1, y1;
        x = y = 0;
        x1 = y1 = 1;

        long input;
        while (true) {
            print(x + x1, y + y1);
            input = scanInt();
            if (input == 0) {
                return;
            } else if (input == 1) {
                x += x1;
                x1 <<= 1;
                if (x + x1 > n) x1 = n - x;
            } else if (input == 2) {
                y += y1;
                y1 <<= 1;
                if (y + y1 > n) y1 = n - y;
            } else if (input == 3) {
                x1 >>= 1;
                y1 >>= 1;
                if (x1 < 1) x1 = 1;
                if (y1 < 1) y1 = 1;
            }
        }
    }

    private void print(long a, long b) {
        System.out.println(String.format("%s %s", a, b));
        System.out.flush();
    }

    private long scanInt() {
        try {
            return Long.parseLong(bf.readLine());
        } catch (IOException e) {
            e.printStackTrace();
        }
        return 0;
    }
}
