#include <bits/stdc++.h>
using namespace std;
vector<int> p;
int get(int v) { return (v == p[v]) ? v : (p[v] = get(p[v])); }
void unite(int a, int b) {
  a = get(a);
  b = get(b);
  if (rand() & 1) {
    swap(a, b);
  }
  if (a != b) {
    p[a] = b;
  }
}
int main() {
  int n;
  cin >> n;
  p.resize(n + 2);
  for (int i = 1; i <= n; i++) {
    p[i] = i;
  }
  vector<pair<int, int>> per;
  vector<int> per2;
  for (int i = 1; i <= n - 1; i++) {
    int a, b;
    cin >> a >> b;
    if (get(a) == get(b)) {
      per.push_back(make_pair(a, b));
    } else
      unite(a, b);
  }
  for (int i = 1; i <= n; i++) {
    if (p[i] == i) per2.push_back(i);
  }
  cout << per2.size() - 1 << endl;
  for (int i = 1; i < per2.size(); i++) {
    cout << per[i - 1].first << " " << per[i - 1].second << " " << per2[i - 1]
         << " " << per2[i] << endl;
  }
  return 0;
}
