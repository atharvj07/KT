#include <bits/stdc++.h>
using namespace std;
const int N = 110;
const int Z = 30;
int dp[N][N][Z], mat[Z][Z], ans = -1e8;
int par[N][N][Z];
int main() {
  const int a = 'a';
  string s;
  int k, n;
  cin >> s >> k >> n;
  for (int i = 0; i < n; i++) {
    char at, bt;
    int ct;
    cin >> at >> bt >> ct;
    mat[at - a][bt - a] = ct;
  }
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++) {
      fill(dp[i][j], dp[i][j] + Z, -1e8);
      fill(par[i][j], par[i][j] + Z, -1);
    }
  dp[0][0][s[0] - a] = 0;
  for (int i = 1; i <= k; i++)
    for (int j = 0; j < 26; j++) dp[0][i][j] = 0;
  for (int i = 1; i < s.size(); i++)
    dp[i][0][s[i] - a] =
        dp[i - 1][0][s[i - 1] - a] + mat[s[i - 1] - a][s[i] - a];
  char o = s[0];
  ans = dp[s.size() - 1][0][s[s.size() - 1] - a];
  for (int i = 1; i <= k; i++)
    for (int j = 1; j < s.size(); j++)
      for (char g = 'a'; g <= 'z'; g++) {
        int k2 = i;
        if (s[j] != g) k2--;
        for (int h = 0; h < 26; h++) {
          int o = dp[j - 1][k2][h] + mat[h][g - a];
          if (dp[j][i][g - a] < o) {
            dp[j][i][g - a] = o;
            par[j][i][g - a] = h;
          }
        }
        if (j == s.size() - 1 && i == k) {
          if (ans < dp[j][i][g - a]) {
            ans = dp[j][i][g - a];
            o = g - a;
          }
        }
      }
  string t = "";
  t += o + a;
  for (int i = s.size() - 1; i >= 1; i--) {
    int to = par[i][k][o];
    if (s[i] != o + a) k--;
    o = to;
    t += o + a;
  }
  reverse(t.begin(), t.end());
  cout << ans << endl;
  return 0;
}
