import sys, io, os
Z = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
o = []
for _ in range(int(Z())):
	(a, b, c, d) = map(int, Z().split())
	if a > b * c:
		o += [-1]
	elif d > c:
		o += [a]
	else:
		v = a // (b * d)
		o += [(v + 1) * a - b * d * (v * (v + 1)) // 2]
print('\n'.join(map(str, o)))
