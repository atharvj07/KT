n = int(input())
l = [int(x) for x in input().split()]

cp1 = p2 + p1 + p1
cp2 = p1 + p2 + p1
cp3 = p1 + p1 + p2
for i in range(12):
	if cp1[i] == l[0]:
		j = i + 1
		length = 1
		while length < n and j < 36:
			if l[length] != cp1[j]:
				break
			length += 1
			j += 1
		if length == n:
			print('Yes')
			exit(0)
