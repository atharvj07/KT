for _ in range(int(input())):
	N = int(input())
	arr = list(map(int, input().split()))
	L = 0
	pos = float('inf')
	pf = False
	neg = set()
	nL = 0
	zero = 0
	temp = []
	mD = float('inf')
	arr.sort()
	for i in range(len(arr)):
		if arr[i] < 0:
			neg.add(arr[i])
			nL += 1
			temp.append(arr[i])
		elif arr[i] == 0:
			zero += 1
			temp.append(arr[i])
		else:
			pf = True
			pos = min(pos, arr[i])
	for i in range(1, len(temp)):
		mD = min(mD, abs(temp[i] - temp[i - 1]))
	if nL == 0:
		L = zero
	else:
		L = nL + zero
	if mD >= pos and zero < 2 and (pf == True):
		L += 1
	print(L)
