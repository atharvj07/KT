import java.util.*;
public class C87 {
	public static void main(String[] args){
		Scanner br = new Scanner(System.in);
		int n = br.nextInt();
		int[] preGrundy = new int[n+1];
		for(int i = 1;i<=n;i++){

			int mask = 0;
			for(int j = 2;j<=446;j++){
				int cur = i+(j*(j-1))/2;

				if(cur/j < j) break;
				if(cur%j == 0){
					int res = cur/j;
					if(res >= j){
						int xor = 0;
						xor = preGrundy[res]^preGrundy[res-j];
						mask|=(1<<xor);
					}
					else{
						break;
					}
				}
			}
			int cur = 0;
			for(int j = 0;;j++){
				if((mask & (1<<j)) == 0){
					break;
				}
				cur++;
			}
			preGrundy[i] = preGrundy[i-1]^cur;
		}
		int grundy = preGrundy[n]^preGrundy[n-1];
		if(grundy == 0){
			System.out.println(-1);
		}
		else{
			for(int j = 2;j<n;j++){
				long cur = n+((long)j*(j-1))/2;
				if(cur%j == 0){
					long res = cur/j;
					if(res >= j){
						int xor = 0;
						xor = preGrundy[(int)res]^preGrundy[(int)res-j];
						if(xor == 0){
							System.out.println(j);
							break;
						}
					}
				}
			}
		}
	}
}
