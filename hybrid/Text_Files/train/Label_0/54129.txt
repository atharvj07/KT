import java.io.*;
import java.util.*;
public class B500 
{
    public static void main(String args[])throws IOException
    {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int arr[]=new int[n+1];
        HashMap<Integer,ArrayList<Integer>> map=new HashMap<>();
        for(int i=1;i<=n;i++)
        {
            arr[i]=sc.nextInt();
            map.put(i,new ArrayList<>());
        }
        for(int i=1;i<=n;i++)
        {
            String s=sc.next().trim();
            for(int j=0;j<n;j++)
            {
                if(s.charAt(j)=='1')
                {
                    map.get(i).add(j+1);
                    map.get(j+1).add(i);
                }
            }
        }
        int id[]=new int[n+1];int conn=0;
        boolean vis[]=new boolean[n+1];
        for(int i=1;i<=n;i++)
        {
            if(!vis[i])
            {
                Stack st=new Stack();
                st.push(i);
                vis[i]=true;conn++;
                while(st.size()!=0)
                {
                    int a=(Integer)st.pop();
                    id[a]=conn;
                    for(int j:map.get(a))
                    {
                        if(!vis[j])
                        {
                            vis[j]=true;
                            st.push(j);
                        }
                    }
                }
            }
        }
        List<Integer> list[]=new ArrayList[conn+1];
        int ptr[]=new int[conn+1];
        for(int i=0;i<=conn;i++)
        {
            list[i]=new ArrayList<>();
        }
        for(int i=1;i<=n;i++)
        {
            list[id[i]].add(arr[i]);
        }
        for(int i=1;i<=conn;i++)
        {
            Collections.sort(list[i]);
        }
        for(int i=1;i<=n;i++)
        {
            System.out.print(list[id[i]].get(ptr[id[i]])+" ");
            ptr[id[i]]++;
        }
    }
}
