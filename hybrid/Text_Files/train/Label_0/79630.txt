def intIn():
	return int(stdin.readline())

def stringIn():
	return stdin.readline().strip()
x = intIn()
ans = 0
li = []
t = 0
while t < 40:
	b = bin(x)[2:]
	l = len(b)
	if b.count('1') == l:
		break
	else:
		for i in range(l):
			if b[i] == '0':
				li.append(l - i)
				break
		x ^= 2 ** (l - i) - 1
		bb = bin(x)[2:]
		if bb.count('1') == len(bb):
			ans += 1
			break
		else:
			ans += 2
		x += 1
	t += 1
if ans == 0:
	print(0)
else:
	print(ans)
	print(*li)
