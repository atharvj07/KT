(n, m, k) = map(int, input().split())
mat = [[0 for i in range(m)] for i in range(n)]
l = []
for i in range(n):
	if i % 2 == 0:
		for j in range(m):
			l.append(str(i + 1) + ' ' + str(j + 1))
	else:
		for j in range(m - 1, -1, -1):
			l.append(str(i + 1) + ' ' + str(j + 1))
start = 0
for i in range(k - 1):
	print(2, ' '.join(l[start:start + 2]))
	start += 2
print(n * m - 2 * (k - 1), ' '.join(l[start:]))
