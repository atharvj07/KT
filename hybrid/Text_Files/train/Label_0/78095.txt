n = int(input())
crianca = [int(i) for i in input().split()]
crianca.sort()
meio = n // 2
roda = []
for i in range(n):
	roda.append(0)
roda[meio] = crianca[n - 1]
del crianca[n - 1]
if n % 2 == 0:
	roda[0] = crianca[0]
	del crianca[0]
i = 1
for j in range(len(crianca) // 2):
	aux1 = len(crianca) - 1
	aux2 = len(crianca) - 2
	roda[meio - i] = crianca[aux1]
	roda[meio + i] = crianca[aux2]
	del crianca[aux1]
	del crianca[aux2]
	i += 1
for i in roda:
	print(i, end=' ')
