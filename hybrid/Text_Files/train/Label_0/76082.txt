A = input()
(V, O) = ([], [])
(v, o) = (0, 0)
for val in A:
	if val == 'v':
		v += 1
		if v > 1 and o > 0:
			O.append(o)
			o = 0
	if val == 'o':
		o += 1
		if v > 1:
			V.append(v - 1)
		v = 0
if v > 1:
	V.append(v - 1)
if len(V) > 0 and len(O) > 0:
	Sum = V[-1]
	Suffix = [Sum]
	for val in V[::-1][1:]:
		Sum += val
		Suffix.append(Sum)
	Suffix = Suffix[::-1][1:]
	Another = []
	count = 0
	for (a, b) in zip(Suffix[::-1], O[::-1]):
		count += a * b
		Another.append(count)
	Another = Another[::-1]
	Ans = 0
	for n in range(len(V) - 1):
		Ans += Another[n] * V[n]
	print(Ans)
else:
	print(0)
