import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        sc.nextLine();
        int[] xPos = new int[n];
        int[] yPos = new int[m];
        for(int i = 0; i < m; ++i){
            yPos[i] = -1;
        }
        long staticCount = 0;
        for (int i = 0; i != n; ++i) {
            String line = sc.nextLine();
            int pos = line.toLowerCase().indexOf('x');
            xPos[i] = pos;
            if(pos != -1){
                yPos[pos] = i;
                ++staticCount;
            }
        }
        long length = 0;
        for(int i = 0; i < n; ++i){
            for(int j = 0; j < m; ++j){
                int amount = (n - i) * (m - j);
                length += ((i * j > 0) ? 2 : 1) * amount * (i + j);
            }
        }

        for(int i = 0; i < n; ++i){
            if(xPos[i] < 0) continue;

            long x1 = xPos[i];
            long x2 = m - xPos[i] - 1;
            long y1 = i;
            long y2 = n - i - 1;

            length -= (x1 * (x1 + 1) / 2 + x2 * (x2 + 1) / 2) * n;
            length -= (y1 * (y1 + 1) / 2 + y2 * (y2 + 1) / 2) * m;

            length += 2 * x1 * x2 + 2 * y1 * y2;

            for(int j = i + 1; j < n; ++j){
                if(xPos[j] < 0){
                    continue;
                }
                length += j - i + Math.abs(xPos[j] - xPos[i]);
            }
        }

        length += countClose(xPos, n, m);
        length += countClose(yPos, m, n);

        length *= 2;

        long count = (m * n - staticCount) * (m * n - staticCount);

        System.out.println(((double) length) / count);
    }

    static long countClose(int[] pos, int n, int m){
        long res = 0;
        for(int i = 0; i < n; ++i){
            if(pos[i] <= 0) continue;
            int ofs = pos[i];
            int index = i - 1;
            while(index >= 0 && pos[index] > ofs){
                res += pos[i] * (m - pos[index] - 1);
                ofs = pos[index];
                --index;
            }
            ofs = pos[i];
            index = i + 1;
            while(index < n && pos[index] > ofs){
                res += pos[i] * (m - pos[index] - 1);
                ofs = pos[index];
                ++index;
            }
        }
        return 2 * res;
    }
}
