q = int(input())
for i in range(q):
	n = int(input())
	p = list(map(int, input().split()))
	cur = p[0]
	answ = 'YES'
	ind = 0
	for j in range(1, n):
		if abs(p[j] - cur) == 1:
			cur = p[j]
		else:
			ind = j
			p = p[j:] + p[:j]
			break
	cur = p[0]
	for j in range(1, n):
		if abs(p[j] - cur) == 1:
			cur = p[j]
		else:
			answ = 'NO'
			break
	print(answ)
