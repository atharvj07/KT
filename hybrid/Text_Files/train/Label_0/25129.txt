        int n;

        for(int i=0;i<n;i++){
            cin>>a[i];
            for(int j=0;j<60;j++){
                if((a[i])&(1ll<<j)){
                    cnt[j]++;
                }
            }
            sum=(sum+(a[i]%MOD))%MOD;

            for(int j=0;j<60;j++){
                if((a[i])&(1ll<<j)){
                    b[i]=(b[i]+((1ll<<j)%MOD)*cnt[j])%MOD;
                }
            }
        }

        for(int i=0;i<n;i++){
            ans=(ans+(b[i]*((sum+((a[i]%MOD)*n)%MOD-b[i])%MOD))%MOD)%MOD;
        }
        ans=(ans+MOD)%MOD;
