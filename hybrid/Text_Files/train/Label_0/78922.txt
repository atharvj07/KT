[b1, q, l, m] = list(map(int, input().split()))
a = list(map(int, input().split()))
a.sort()
if abs(b1) > l:
	print('0')
	exit(0)
if b1 == 0:
	if (b1 in a) == False:
		print('inf')
	else:
		print('0')
	exit(0)
if q == 0:
	if (q in a) == False:
		print('inf')
	elif abs(b1) <= l and (b1 in a) == False:
		print('1')
	else:
		print('0')
	exit(0)
if q == 1:
	if abs(b1) <= l:
		if (b1 in a) == False:
			print('inf')
		else:
			print('0')
		exit(0)
if q == -1:
	if abs(b1) <= l and ((b1 in a) == False or (-b1 in a) == False):
		print('inf')
	elif abs(b1) <= l and (b1 in a) == True and ((-b1 in a) == True):
		print('0')
	exit(0)
cnd = []
b = b1
while abs(b) <= l:
	cnd.append(b)
	b *= q
res = []
for e in cnd:
	if (e in a) == False:
		res.append(e)
print(len(res))
