#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int t[maxn];
int main(void) {
  int n;
  while (~scanf("%d", &n)) {
    for (int i = 0; i < n; i++) scanf("%d", &t[i]);
    sort(t, t + n, less<int>());
    long long pre = t[0], ans = 1;
    for (int i = 1; i < n; i++) {
      if (pre > t[i])
        continue;
      else {
        pre += t[i];
        ans++;
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
