#include <bits/stdc++.h>
using namespace std;
void read() {
  freopen("pudge.in", "r", stdin);
  freopen("pudge.out", "w", stdout);
}
void faster() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
}
void pudge() {
  int n, sum = 0;
  cin >> n;
  int a[n][n];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> a[i][j];
      if (i == j || i == n / 2 || j == n / 2 || j + i == n - 1) sum += a[i][j];
    }
  }
  cout << sum;
}
int main() {
  faster();
  pudge();
  return 0;
}
