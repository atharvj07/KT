#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

char C3[4][4] = {"aa.", "..a", "..a"};
char C4[5][5] = {"aaca", "bbca", "acbb", "acaa"};
char C5[6][6] = {"abbaa", "acc.b", "b..cb", "b..ca", "aabba"};
char C7[8][8] = {"caa.aa.", "c..a..a", "b..a..a", "baa.aa.", "a..a..a", "a..a..a", ".aabbcc"};
char B[1010][1010];
int n;

int main()
{
	int i, j, k;
	
	scanf("%d", &n);
	
	for(i = 0; i < n; i ++){
		for(j = 0; j < n; j ++){
			B[i][j] = '.';
		}
		B[i][j] = '\0';
	}
	
	if(n < 3){
		printf("-1\n");
		return 0;
	}
	else if(n == 3){
		for(i = 0; i < 3; i ++){
			for(j = 0; j < 3; j ++){
				B[i][j] = C3[i][j];
			}
		}
	}
	else if(n == 6){
		for(i = 0; i < 3; i ++){
			for(j = 0; j < 3; j ++){
				B[i][j] = C3[i][j];
			}
		}
		for(i = 0; i < 3; i ++){
			for(j = 0; j < 3; j ++){
				B[i + 3][j + 3] = C3[i][j];
			}
		}
	}
	else if(n == 7){
		for(i = 0; i < 7; i ++){
			for(j = 0; j < 7; j ++){
				B[i][j] = C7[i][j];
			}
		}
	}
	else if(n == 11){
		for(i = 0; i < 4; i ++){
			for(j = 0; j < 4; j ++){
				B[i][j] = C4[i][j];
			}
		}
		for(i = 0; i < 7; i ++){
			for(j = 0; j < 7; j ++){
				B[i + 4][j + 4] = C7[i][j];
			}
		}
	}
	else{
		for(i = 0; (n - i) % 4; i += 5){
			for(j = 0; j < 5; j ++){
				for(k = 0; k < 5; k ++){
					B[i + j][i + k] = C5[j][k];
				}
			}
		}
		for(; i < n; i += 4){
			for(j = 0; j < 4; j ++){
				for(k = 0; k < 4; k ++){
					B[i + j][i + k] = C4[j][k];
				}
			}
		}
	}
	
	for(i = 0; i < n; i ++){
		printf("%s\n", B[i]);
	}
	
	return 0;
}
