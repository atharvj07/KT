class in{struct It{int it;const bool rev;explicit constexpr It(int it_, bool rev=false):it(it_),rev(rev){}int operator*(){return it;}bool operator!=(It& r){return it!=r.it;}void operator++(){rev?--it:++it;}};const It i,n;public:explicit constexpr in(int n):i(0),n(n<0?0:n){}explicit constexpr in(int i,int n):i(i,n<i),n(n){}const It& begin(){return i;}const It& end(){return n;}};

#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;

int main() {
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    auto width = [&](const vector<int>& x, const vector<int>& y, double kx) {
        const int n = x.size();
        double lb = INF, ub = -lb;
        for(int i : in(n)) {
            double x1 = x[i], y1 = y[i];
            double x2 = x[(i + 1) % n], y2 = y[(i + 1) % n];
            if((x1 - kx) * (x2 - kx) <= 0 && x1 != x2) {
                double crossy = y1 + (y2 - y1) * (kx - x1) / (x2 - x1);
                lb = min(lb, crossy);
                ub = max(ub, crossy);
            }
        }
        return max(0.0, ub - lb);
    };
    int p1, p2;
    while(cin >> p1 >> p2 && p1) {
        vector<int> x1(p1), y1(p1), x2(p2), y2(p2);
        vector<int> xs;
        for(int i : in(p1)) {
            cin >> x1[i] >> y1[i];
            xs.emplace_back(x1[i]);
        }
        for(int i : in(p2)) {
            cin >> x2[i] >> y2[i];
            xs.emplace_back(x2[i]);
        }
        const int min1 = *min_element(x1.begin(), x1.end()), max1 = *max_element(x1.begin(), x1.end());
        const int min2 = *min_element(x2.begin(), x2.end()), max2 = *max_element(x2.begin(), x2.end());
        
        sort(xs.begin(), xs.end());
        double res = 0;
        for(int i : in(xs.size() - 1)) {
            double l = xs[i], r = xs[i + 1], mid = (l + r) / 2.0;
            if(min1 <= mid && mid <= max1 && min2 <= mid && mid <= max2) {
                const double fa = width(x1, y1, l) * width(x2, y2, l);
                const double fb = width(x1, y1, r) * width(x2, y2, r);
                const double fc = width(x1, y1, mid) * width(x2, y2, mid);
                res += (r - l) / 6 * (fa + 4 * fc + fb);
            }
        }
        cout << fixed << setprecision(10) << res << endl;
    }
    return 0;
}
