#include<cstdio>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int n;
    while(scanf("%d",&n),n){
        int p[52]={0};p[0]=-100;
        for(int i=0;i<n;i++)scanf("%d",&p[i+1]);
        printf("%d",p[1]);
        for(int i=2;i<=n;i++){
            if(p[i]==p[i-1]+1&&p[i]==p[i+1]-1)continue;
            if(p[i]==p[i-1]+1)printf("-%d",p[i]);
            else printf(" %d",p[i]);
        }
        puts("");
    }

    return 0;
}
