#include <iostream>
#include <vector>
#include <set>
#include <cmath>
using namespace std;

int main(){
    int N,M;
    while(cin >> N >> M && N){
        vector<int> A(N),W(M);
        for(int i=0;i<N;i++) cin >> A[i];
        for(int i=0;i<M;i++) cin >> W[i];
        set<int> s;
        for(int S=0;S<(1<<M);S++){
            for(int T=S;T>=0;T=(T-1)&S){
                int x = 0,y = 0;
                for(int i=0;i<M;i++){
                    if((S&(1<<i)) && (T&(1<<i))) x += W[i];
                    else if((S&(1<<i)) && !(T&(1<<i))) y += W[i];
                }
                s.insert(x-y);
                if(T==0) break;
            }
        }
        vector<int> unable;
        for(int i=0;i<N;i++){
            if(s.count(A[i])==0 && s.count(-A[i])==0) unable.push_back(A[i]);
        }
        if(unable.empty()){
            cout << 0 << endl;
            continue;
        }
        set<int> kouho;
        for(auto a:s){
            kouho.insert(abs(a-unable[0]));
            kouho.insert(abs(a+unable[0]));
        }
        for(auto u:unable){//O(N)
            if(kouho.empty()) break;
            set<int> nkouho;
            for(auto k:kouho){//O(3^M)
                if(s.count(u-k)!=0 || s.count(u+k)!=0 || s.count(-u-k)!=0 || s.count(-u+k)!=0) nkouho.insert(k);
            }
            kouho = nkouho;
        }
        cout << (!kouho.empty()? *kouho.begin():-1) << endl;
    }
}
