import sys
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
l = [0] * 50
for i in range(n):
	s = bin(arr[i])
	(a, j) = (49, len(s) - 1)
	while j > 1:
		if s[j] == '1':
			l[a] += 1
		a -= 1
		j -= 1
ind = -1
ans = -1
for i in range(n):
	s = bin(arr[i])
	(a, j) = (49, len(s) - 1)
	while j > 1:
		if s[j] == '1':
			l[a] -= 1
		a -= 1
		j -= 1
	(a, j) = (49, len(s) - 1)
	(num, count) = (0, 0)
	while j > 1:
		if s[j] == '1' and l[a] == 0:
			num += 2 ** count
			count += 1
			j -= 1
			a -= 1
		else:
			j -= 1
			a -= 1
			count += 1
	if num > ans:
		ans = num
		ind = i
	(a, j) = (49, len(s) - 1)
	while j > 1:
		if s[j] == '1':
			l[a] += 1
		a -= 1
		j -= 1
b = [arr[ind]] + arr[:ind] + arr[ind + 1:]
print(*b)
