n = int(input())
S = list(map(int, list(input())))
AB = [list(map(int, input().split())) for _ in range(n)]
ans = 0
for t in range(1001):
	tmp = 0
	for i in range(n):
		s = S[i]
		(a, b) = AB[i]
		if t < b:
			tmp += s
		else:
			c = (t - b) // a + 1
			tmp += s ^ c % 2
	ans = max(ans, tmp)
print(ans)
