#include <bits/stdc++.h>
using namespace std;
int main() {
  int q, n, s, i;
  scanf("%d", &q);
  for (; q > 0; q--) {
    scanf("%d", &n);
    int number = 0;
    for (i = 0; i < n; i++) {
      scanf("%d", &s);
      if (s <= 2048) number = number + s;
    }
    if (number >= 2048)
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
