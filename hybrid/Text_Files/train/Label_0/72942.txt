for _ in range(int(input())):
	n = int(input())
	ans = n * (n - 1) * (n - 2)
	ans //= 6
	a = []
	left = {}
	right = {}
	for i in range(n):
		(u, v) = map(int, input().split())
		a.append([u, v])
		if u in left:
			left[u] += 1
		else:
			left[u] = 1
		if v in right:
			right[v] += 1
		else:
			right[v] = 1
	for i in range(n):
		(u, v) = (a[i][0], a[i][1])
		x = left[u] - 1
		y = right[v] - 1
		ans -= x * y
	print(ans)
