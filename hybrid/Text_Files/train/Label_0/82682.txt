import sys
input = sys.stdin.readline
for _ in range(1):
	n = int(input())
	arr = [int(x) for x in input().split()]
	b = [0 for i in range(n + 1)]

	def update(i, val, n):
		i += 1
		while i < n:
			b[i] += val
			i += i & -i
		return

	def get(i):
		i += 1
		ans = 0
		while i > 0:
			ans += b[i]
			i -= i & -i
		return ans
	d = {}
	for i in arr:
		if i in d:
			d[i] += 1
		else:
			d[i] = 1
		update(d[i], 1, n + 1)
	ans = 0
	now = {}
	for i in arr:
		update(d[i], -1, n + 1)
		d[i] -= 1
		if i in now:
			now[i] += 1
		else:
			now[i] = 1
		ans += get(now[i] - 1)
	print(ans)
