#include <bits/stdc++.h>
using namespace std;
char a[100], b[100];
int l, r;
int main() {
  l = 1;
  r = 1;
  memset(a, 0, sizeof(a));
  memset(b, 0, sizeof(b));
  cin >> a;
  int m = 0;
  for (int i = 0; i < 50; i++) {
    if (a[i] >= 'a' && a[i] <= 'z') {
      m++;
    } else
      break;
  }
  b[0] = a[(m - 1) / 2];
  for (int i = 1; i < m; i = i + 2) {
    b[i] = a[(m - 1) / 2 + r];
    r++;
  }
  for (int i = 2; i < m; i = i + 2) {
    b[i] = a[(m - 1) / 2 - l];
    l++;
  }
  for (int i = 0; i < m; i++) {
    printf("%c", b[i]);
  }
  return 0;
}
