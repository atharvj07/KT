#include <bits/stdc++.h>
using namespace std;
int a[5];
int main() {
  int n, s, c[5];
  c[1] = c[2] = c[3] = c[4] = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> s;
    c[s]++;
  }
  c[0] = c[4] + c[3] + c[2] / 2;
  if (c[1] > c[3]) c[0] = c[0] + (c[1] - c[3]) / 4;
  if ((c[1] - c[3]) % 4 > 0) c[0]++;
  if (c[2] % 2 > 0) c[0]++;
  if (c[2] % 2 > 0 && ((c[1] - c[3]) % 4 == 1 || (c[1] - c[3]) % 4 == 2))
    c[0]--;
  cout << c[0];
}
