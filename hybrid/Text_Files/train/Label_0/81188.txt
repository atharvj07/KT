import math
t = int(input())
for i in range(t):
	n = int(input())
	distancias = []
	pos_x = []
	pos_y = []
	for j in range(n):
		pos_xy = input()
		pos_xy = pos_xy.split()
		if int(pos_xy[0]) != 0:
			pos_x.append(abs(int(pos_xy[0])))
		else:
			pos_y.append(abs(int(pos_xy[1])))
		mina_xy = input()
		mina_xy = mina_xy.split()
		if int(mina_xy[0]) != 0:
			pos_x.append(abs(int(mina_xy[0])))
		else:
			pos_y.append(abs(int(mina_xy[1])))
	pos_x.sort()
	pos_y.sort()
	for j in range(n):
		distancia = math.sqrt(pos_x[j] ** 2 + pos_y[j] ** 2)
		distancias.append(distancia)
	soma = 0
	for j in range(n):
		soma += distancias[j]
	print(soma)
