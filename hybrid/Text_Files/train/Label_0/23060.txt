#include <bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using p = pair<int, int>;

struct data {
  int num, lx, ly;
};

bool asc(data l, data r) {
  return l.lx * l.ly < r.lx * r.ly;
}

int n, w, d;
vector<p> cut;
vector<data> lst;

void solve();

int main() {
  while(1) {
    cin >> n >> w >> d;
    if(n + w + d == 0) break;
    lst.push_back({-1, w, d});
    for(int i = 0; i < n; ++i) {
      int a, b;
      cin >> a >> b;
      cut.push_back({--a, b});
    }
    solve();
    sort(lst.begin(), lst.end(), asc);
    for(int i = 0; i < lst.size(); ++i) {
      if(i != 0) cout << " ";
      cout << lst[i].lx * lst[i].ly;
    }
    cout << endl;
    lst.erase(lst.begin(), lst.end());
    cut.erase(cut.begin(), cut.end());
  }
  return 0;
}

void solve() {
  for(int t = 0; t < n; ++t) {
    data now = lst[cut[t].fi], next1, next2;
    lst.erase(lst.begin() + cut[t].fi);
    next1 = next2 = now;
    next1.num = next2.num = t;
    cut[t].se %= (now.lx + now.ly);
    if(cut[t].se > now.lx) {
      cut[t].se -= now.lx;
      next1.ly = min(cut[t].se, now.ly - cut[t].se);
      next2.ly -= next1.ly;
    }
    else {
      next1.lx = min(cut[t].se, now.lx - cut[t].se);
      next2.lx -= next1.lx;
    }
    lst.push_back(next1);
    lst.push_back(next2);
  }
}
