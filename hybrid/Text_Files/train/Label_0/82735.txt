from bisect import bisect_left as bl
import sys
input = sys.stdin.readline
w = int(input())
a = sorted(map(int, input().split()))
b = sum(a)
for i in range(int(input())):
	(x, y) = map(int, input().split())
	g = []
	c = bl(a, x)
	(d, e) = (0, 0)
	if c > len(a) - 1:
		c -= 1
	d = max(0, x - a[c]) + max(0, y - b + a[c])
	e = max(0, x - a[c - 1]) + max(0, y - b + a[c - 1])
	f = min(d, e)
	g.append(f)
	print(min(g))
