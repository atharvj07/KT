#include<bits/stdc++.h>

using namespace std;
typedef pair<int,int> pi;

int n,k;
int a[105];
int ans[100005];
void solve(){
    cin>>n>>k;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    a[n+1]=a[n];
    int cnt = 0;
    while(1){
        int ok=0;

        for(int i=1;i<=n;i++){
            if(a[i]>=a[i+1]) continue;
            else{
              //cout<<i<<" "<<a[i]<<" "<<a[i+1]<<"\n";
              cnt++;
              ans[cnt]=i;
              a[i]++;
              ok=1;
              break;
            }
        }
        if(ok==0) break;
    }
    //cout<<cnt<<"\n";
    cout<<( (k>cnt||cnt==0) ? (-1) : (ans[k]) )<<"\n";
}

signed main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0),cout.tie(0);

    //freopen("test.inp","r",stdin);
   // freopen("test.out","w",stdout);

    int test = 1;
    cin>>test;
    while(test--){
        solve();
    }

}
