#include <bits/stdc++.h>
using namespace std;
const int64_t M1 = 998244353;
const int64_t M2 = 1000000007;
int64_t l[500001];
int64_t r[500001];
int64_t cnt[500001];
int64_t a[500001];
int64_t ans[500001];
int64_t mark[500001];
int64_t query[500001];
vector<int64_t> b;
int64_t len;
bool comp(int64_t i, int64_t j) {
  if (l[i] / len == l[j] / len)
    return ((l[i] / len) & 1) ? r[i] < r[j] : r[i] > r[j];
  return (l[i] / len) < (l[j] / len);
}
void solve() {
  int64_t n, q;
  cin >> n;
  len = 900;
  for (int64_t i = 1; i <= n; i++) cin >> a[i];
  cin >> q;
  for (int64_t i = 1; i <= q; i++) {
    cin >> l[i] >> r[i];
    query[i] = i;
  }
  sort(query + 1, query + q + 1, comp);
  cnt[a[1]]++;
  int64_t left = 1, right = 1;
  b.push_back(a[1]);
  mark[a[1]] = 1;
  for (int64_t i = 1; i <= q; i++) {
    while (left < l[query[i]]) {
      cnt[a[left]]--;
      if (cnt[a[left]] == 1 && !mark[a[left]]) {
        b.push_back(a[left]);
        mark[a[left]] = 1;
      }
      left++;
    }
    while (left > l[query[i]]) {
      left--;
      cnt[a[left]]++;
      if (cnt[a[left]] == 1 && !mark[a[left]]) {
        b.push_back(a[left]);
        mark[a[left]] = 1;
      }
    }
    while (right < r[query[i]]) {
      right++;
      cnt[a[right]]++;
      if (cnt[a[right]] == 1 && !mark[a[right]]) {
        b.push_back(a[right]);
        mark[a[right]] = 1;
      }
    }
    while (right > r[query[i]]) {
      cnt[a[right]]--;
      if (cnt[a[right]] == 1 && !mark[a[right]]) {
        b.push_back(a[right]);
        mark[a[right]] = 1;
      }
      right--;
    }
    while (!b.empty() && cnt[b.back()] != 1) {
      mark[b.back()] = 0;
      b.pop_back();
    }
    ans[query[i]] = (b.empty() ? 0 : b.back());
  }
  for (int64_t i = 1; i <= q; i++) cout << ans[i] << "\n";
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  solve();
}
