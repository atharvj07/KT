from typing import Dict, List
(n_colors, len_pairs) = (int(val) for val in input().split())
color_pairs = []
for i in range(len_pairs):
	color_pairs.append([int(val) for val in input().split()])
res = []
cur_x = 1
cur_y = 1
for color in range(1, n_colors + 1):
	cur_color_res = []
	cur_color_res.append((cur_x, cur_y))
	cur_x += 1
	cur_y += 1
	res.append(cur_color_res)
for pair in color_pairs:
	res[pair[0] - 1].append((cur_x, cur_y))
	res[pair[0] - 1].append((cur_x, pair[0]))
	res[pair[1] - 1].append((cur_x, pair[1]))
	cur_x += 1
	cur_y += 1
for cur_res in res:
	print(len(cur_res))
	for el_cur_res in cur_res:
		print('{} {}'.format(el_cur_res[0], el_cur_res[1]))
