t = int(input())
for _ in range(t):
	(n, s) = map(int, input().split())
	M = list(map(int, input().split()))
	cnt = 0
	ans = 0
	ma = 0
	ind = 0
	f = True
	for i in range(n - 1):
		cnt += M[i]
		if M[i] > ma:
			ma = M[i]
			ind = i + 1
		if cnt > s:
			if f:
				if ma > M[i + 1]:
					ans = ind
				f = False
			else:
				break
	print(ans)
