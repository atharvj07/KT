#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  string s[n];
  long long i, j, k;
  for (i = 0; i < n; i++) cin >> s[i];
  long long c, ans = 0;
  for (i = 0; i < n; i++) {
    c = 1;
    for (j = i + 1; j < n; j++) {
      for (k = 0; k < n; k++) {
        if (s[i][k] != s[j][k]) break;
      }
      if (k == n) c++;
    }
    if (c > ans) ans = c;
  }
  cout << ans;
  return 0;
}
