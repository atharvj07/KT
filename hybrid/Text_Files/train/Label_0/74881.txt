a = [int(x) for x in input().split(' ')]
n = a[0]
s = a[1]
a = [int(x) for x in input().split(' ')]
mid = n // 2
a = sorted(a)
if a[mid] == s:
	print(0)
if a[mid] < s:
	Sum = 0
	for i in range(mid, n):
		Sum = Sum + max(s - a[i], 0)
	print(Sum)
if a[mid] > s:
	Sum = 0
	for i in range(0, mid + 1):
		Sum = Sum + max(a[i] - s, 0)
	print(Sum)
