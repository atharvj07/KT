from collections import defaultdict
(n, k) = map(int, input().split())
if k > 9 * n:
	print('-1 -1')
elif k == 0:
	if n == 1:
		print('0 0')
	else:
		print('-1 -1')
else:
	s = [0] * (n - 1)
	s = [1] + s
	curr = n - 1
	for _ in range(k - 1):
		s[curr] += 1
		if s[curr] == 9:
			curr -= 1
	print(*s, sep='', end=' ')
	s = [0] * n
	curr = 0
	for _ in range(k):
		s[curr] += 1
		if s[curr] == 9:
			curr += 1
	print(*s, sep='', end=' ')
