#include <bits/stdc++.h>
using namespace std;
char q[150];
int N, K, val[30][30], dn[150][130][110], used[150][30][110];
int f(int cur, int back, int right) {
  if (right < 0) return -1e9;
  if (cur > N) return 0;
  if (used[cur][back][right]) return dn[cur][back][right];
  used[cur][back][right] = 1;
  int &ret = dn[cur][back][right];
  ret = -1e9;
  for (int i = 1; i <= 26; i++) {
    if (i == q[cur] - 'a' + 1)
      ret = max(ret, f(cur + 1, i, right) + val[back][i]);
    else
      ret = max(ret, f(cur + 1, i, right - 1) + val[back][i]);
  }
  return ret;
}
int main() {
  scanf("%s", q + 1);
  N = strlen(q + 1);
  scanf("%d", &K);
  int pr;
  cin >> pr;
  int x;
  char a, b;
  for (int i = 1; i <= pr; i++) {
    scanf(" %c %c %d", &a, &b, &x);
    val[a - 'a' + 1][b - 'a' + 1] = x;
  }
  int maxx = -1e9;
  for (int i = 1; i <= 26; i++) {
    if (i == q[1] - 'a' + 1)
      maxx = max(maxx, f(2, i, K));
    else
      maxx = max(maxx, f(2, i, K - 1));
  }
  cout << maxx << endl;
  return 0;
}
