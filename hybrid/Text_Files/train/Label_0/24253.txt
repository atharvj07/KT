#include <bits/stdc++.h>
using namespace std;
int n, m;
int main() {
  char ch[106][106];
  scanf("%d%d ", &n, &m);
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j) cin >> ch[i][j];
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (ch[i][j] == 'C' || ch[i][j] == 'M' || ch[i][j] == 'Y') {
        printf("#Color\n");
        return 0;
      } else {
        continue;
      }
    }
  }
  printf("#Black&White\n");
  return 0;
}
