get_line = lambda type_: list(map(type_, input().strip().split()))
from collections import Counter

def solve():
	n = int(input())
	a = get_line(int)
	c = Counter(a)
	his = [i[1] for i in c.items()]
	his.sort()
	max_rect_area = 0
	last_h = 0
	for (i, h) in enumerate(his):
		if h != last_h:
			if h * (len(his) - i) > max_rect_area:
				max_rect_area = h * (len(his) - i)
			last_h = h
	print(sum(his) - max_rect_area)
t = int(input())
for _ in range(t):
	solve()
