import collections
t = int(input())
a = [None for _ in range(t)]
res = []
parents = []
for i in range(t):
	a[i] = list(map(int, input().split()))
	res.append(a[i][i])

def split(ids):
	if len(ids) == 1:
		return ids[0] + 1
	groups = []
	d = set(ids)
	mx = max([x for (i, x) in enumerate(a[ids[0]]) if i in d])
	res.append(mx)
	cur = len(res)
	for id in ids:
		found = False
		for group in groups:
			if a[group[0]][id] < mx:
				group.append(id)
				found = True
				break
		if not found:
			groups.append([id])
	for group in groups:
		parents.append([split(group), cur])
	return cur
chief = split([i for i in range(t)])
print(len(res))
print(*res, sep=' ')
print(chief)
for r in parents:
	print(*r, sep=' ')
