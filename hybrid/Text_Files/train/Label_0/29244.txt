#include <bits/stdc++.h>
using namespace std;
const int maxint = -1u >> 1;
int dfs(int now, int pre, vector<vector<int>>& tree, int& ans) {
  int node_count = 0;
  for (int i = 0; i < tree[now].size(); ++i) {
    int x = tree[now][i];
    if (x == pre) continue;
    node_count = node_count + dfs(x, now, tree, ans);
  }
  node_count = node_count + 1;
  if (node_count % 2 == 0) ++ans;
  return node_count;
}
int main() {
  int n;
  while (scanf("%d", &n) == 1) {
    if (n % 2 == 1) {
      printf("-1\n");
      return 0;
    }
    vector<vector<int>> tree(n);
    int x, y;
    while (--n) {
      scanf("%d%d", &x, &y);
      --x, --y;
      tree[x].push_back(y);
      tree[y].push_back(x);
    }
    int ans = -1;
    int tot = dfs(0, -1, tree, ans);
    printf("%d\n", ans);
  }
  return 0;
}
