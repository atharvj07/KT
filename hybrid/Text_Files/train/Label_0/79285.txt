(n, a, b) = map(int, input().split())
ex = 0
ans = 0
t = map(int, input().split())
for i in t:
	if i == 1:
		if a == 0:
			if b == 0:
				if ex == 0:
					ans = ans + 1
				else:
					ex = ex - 1
			else:
				b = b - 1
				ex = ex + 1
		else:
			a = a - 1
	elif b == 0:
		ans = ans + 2
	else:
		b = b - 1
print(ans)
