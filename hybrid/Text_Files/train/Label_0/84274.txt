N = 200005

def solve(n, a):
	b = [0] * N
	b[0] = 1
	m = 1
	l = 1
	while l < n:
		k = b[m - 1]
		b[m] = 0
		while k > 0:
			k -= 1
			if l == n:
				continue
			l += 1
			b[m] += 1
			while l < n and a[l] > a[l - 1]:
				l += 1
				b[m] += 1
		m += 1
	return m - 1
t = int(input())
while t:
	n = int(input())
	a = list(map(int, input().split()))
	print(solve(n, a))
	t -= 1
