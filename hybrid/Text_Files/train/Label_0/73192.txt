number = int(input())
for i in range(number):
	input()
	elements = list((int(x) for x in input().split()))
	k = max(elements)
	if not k:
		k = 1
	current = 'a' * k
	print(current)
	for j in range(len(elements)):
		if elements[j] < k:
			if current[elements[j]] == 'a':
				current = current[:elements[j]] + 'b' * (k - elements[j])
			else:
				current = current[:elements[j]] + 'a' * (k - elements[j])
			print(current)
		else:
			print(current)
