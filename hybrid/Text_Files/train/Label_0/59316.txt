t=readline().split(' '); MAX=1<<20;
W=t[0]|0; H=t[1]|0; Q=t[2]|0; out='';
Au=new Array(MAX);
Al=new Array(MAX);
Ad=new Array(MAX);
Ar=new Array(MAX);
for(x=1;x<=W;++x){
  t=readline().split(' ');
  for(y=1;y<=H;++y){
    i=(y<<10)+x;
    Au[i]=Al[i]=Ad[i]=Ar[i]=t[y-1]|0;
  }
}
for(x=1;x<=W;++x){
  for(t=Au[(1<<10)+x],y=2;y<=H;++y){
    i=(y<<10)+x;
    if(Au[i])Au[i]=++t;
    else t=0;
  }
}
for(x=1;x<=W;++x){
  for(t=Ad[(H<<10)+x],y=H-1;y>0;--y){
    i=(y<<10)+x;
    if(Ad[i])Ad[i]=++t;
    else t=0;
  }
}
for(y=1;y<=H;++y){
  for(t=Al[(y<<10)+1],x=2;x<=W;++x){
    i=(y<<10)+x;
    if(Al[i])Al[i]=++t;
    else t=0;
  }
}
for(y=1;y<=H;++y){
  for(t=Ar[(y<<10)+W],x=W-1;x>0;--x){
    i=(y<<10)+x;
    if(Ar[i])Ar[i]=++t;
    else t=0;
  }
}
for(;Q--;){
  t=readline().split(' ');
  X=t[1]|0; Y=t[2]|0; i=(Y<<10)+X;
  if(t[0]==='1'){
    if(Au[i])Au[i]=Ad[i]=Al[i]=Ar[i]=0;
    else{
      Au[i]=(Au[(Y-1<<10)+X]|0)+1;
      Ad[i]=(Ad[(Y+1<<10)+X]|0)+1;
      Al[i]=(Al[(Y<<10)+X-1]|0)+1;
      Ar[i]=(Ar[(Y<<10)+X+1]|0)+1;
    }
    for(t=Au[i],y=Y+1;y<=H&&Au[(y<<10)+X];++y)Au[(y<<10)+X]=++t;
    for(t=Ad[i],y=Y-1;y>0 &&Ad[(y<<10)+X];--y)Ad[(y<<10)+X]=++t;
    for(t=Al[i],x=X+1;x<=W&&Al[(Y<<10)+x];++x)Al[(Y<<10)+x]=++t;
    for(t=Ar[i],x=X-1;x>0 &&Ar[(Y<<10)+x];--x)Ar[(Y<<10)+x]=++t;
  }
  else if(Au[i]){
    L=[]; P=[]; M=0; l=0;
    for(x=1;x<=W+1;++x){
      t=Au[(Y<<10)+x]|0;
      for(i=x;l&&L[l-1]>t;--l){
        i=P[l-1];
        s=L[l-1]*(x-i);
        if(x>X&&X>=i&&s>M)M=s;
      }
      if(t&&(!l||L[l-1]<t))P[l]=i,L[l++]=t;
    }
    
    l=0;
    for(x=1;x<=W+1;++x){
      t=Ad[(Y<<10)+x]|0;
      for(i=x;l&&L[l-1]>t;--l){
        i=P[l-1];
        s=L[l-1]*(x-i);
        if(x>X&&X>=i&&s>M)M=s;
      }
      if(t&&(!l||L[l-1]<t))P[l]=i,L[l++]=t;
    }

    l=0;
    for(y=1;y<=H+1;++y){
      t=Al[(y<<10)+X]|0;
      for(i=y;l&&L[l-1]>t;--l){
        i=P[l-1];
        s=L[l-1]*(y-i);
        if(y>Y&&Y>=i&&s>M)M=s;
      }
      if(t&&(!l||L[l-1]<t))P[l]=i,L[l++]=t;
    }
    
    l=0;
    for(y=1;y<=H+1;++y){
      t=Ar[(y<<10)+X]|0;
      for(i=y;l&&L[l-1]>t;--l){
        i=P[l-1];
        s=L[l-1]*(y-i);
        if(y>Y&&Y>=i&&s>M)M=s;
      }
      if(t&&(!l||L[l-1]<t))P[l]=i,L[l++]=t;
    }
    
    out+=M+'\n';
  }
  else out+='0\n';
}
print(out);
