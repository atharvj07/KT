import java.io.*;
import java.util.*;

public class Dance
{
	public static void main(String[]args)
	{
		Scanner sc = new Scanner(System.in);
		long n ;
		n = sc.nextLong();

		System.out.println(fact(n)/((int)Math.pow(n/2,2)*2));


	}
	public static long fact(long n)
	{
		long f=1;

		for(int i=1;i<=n;i++)
		{
			f = f*i ;
		}

		return f;
	}
}
