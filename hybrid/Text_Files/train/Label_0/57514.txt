import java.io.BufferedReader;
import java.io.InputStreamReader;

public class Number_of_Triplets {
    static boolean [][]mapa = new boolean[2001][2001];
    static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    static int coordenadas[];
    public static void main(String[] args) {
        try {
            int nPontos = 0;
            String linha = reader.readLine();
            int  n = Integer.parseInt(linha.split("\\s+")[0]);
            coordenadas = new int[n];
            for (int i = 0; i < n; i++) {
                linha = reader.readLine();
                int x = Integer.parseInt(linha.split("\\s+")[0]) + 1000; // correção para fazer de 0 a 2000
                int y = Integer.parseInt(linha.split("\\s+")[1]) + 1000; // correção para fazer de 0 a 2000
                mapa[x][y] = true;
                coordenadas[nPontos++] = x * 2001 + y;
            }
            
            int contador = 0;
            for (int i = 0; i < n; i++) {
                for (int j = i+1; j < n; j++) {
                    int x1 = coordenadas[i] / 2001;
                    int y1 = coordenadas[i] % 2001;
                    int x2 = coordenadas[j] / 2001;
                    int y2 = coordenadas[j] % 2001;
                    if ((x1+x2) % 2 == 0 && (y1+y2) % 2 == 0) {
                        if (mapa[(x1+x2)/2][(y1+y2)/2])
                            contador++;
                    }
                }
            }           
            System.out.println(contador);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
