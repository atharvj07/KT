import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map.Entry;
import java.util.StringTokenizer;


public class D_215 {

	static BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer st = new StringTokenizer("");
	public static void debug(String s){
		//System.out.println(">>> " + s + " <<<<");
	}
	/**
	 * @param args
	 */
	public static void main(String[] args) throws Exception{
		int n = readInt();
		int m = readInt();
		int p = readInt();
		int[] a = new int[n];
		for(int i = 0; i < n; i++){
			a[i] = readInt();
		}
		int b[] = new int[m];
		HashMap<Integer, Integer> bMap = new HashMap();
		for(int i = 0; i < m; i++){
			b[i] = readInt();
			Integer val = bMap.get(b[i]);
			if(val != null){
				bMap.put(b[i], val+1);
			} else{
				bMap.put(b[i], 1);
			}
		}
		ArrayList<Integer> qs = new ArrayList();
		HashMap<Integer, Integer> maps[] = new HashMap[p];
		for(int i = 0; i < p; i++) maps[i] = new HashMap();
		for(int i = 0; i+(m-1)*p < n && i+(m-1)*p >= 0; i++){
			HashMap<Integer, Integer> map = maps[i%p];
			if(i < p){
				for(int j = 0; j < m && i+j*p < n; j++){
					Integer val = map.get(a[i+j*p]);
					if(val != null){
						map.put(a[i+j*p], val+1);
					} else {
						map.put(a[i+j*p], 1);
					}					
				}
				
			} else {
				Integer val = map.get(a[i-p]);
				if(val > 1)	map.put(a[i-p], val-1);
				else map.remove(a[i-p]);
				Integer val2 = map.get(a[i+(m-1)*p]);
				if(val2 != null){
					map.put(a[i+(m-1)*p], val2+1);
				} else {
					map.put(a[i+(m-1)*p], 1);
				}
			}
			boolean valid = true;
			for(Entry<Integer,Integer> e : map.entrySet()){
				/*if(!map.equals(bMap)){
					valid = false;
				}*/
				if(!bMap.containsKey(e.getKey())){
					valid = false;
					break;
				}
				if(!bMap.get(e.getKey()).equals(map.get(e.getKey()))){
					valid = false;
					break;
				}
			}
			if(valid){
				qs.add(i);
			}
		}
		System.out.println(qs.size());
		for(Integer i : qs){
			System.out.print((i+1) + " ");
		}
		System.out.println();
	}
	
	static String readString() throws Exception {
		while(!st.hasMoreTokens())
			st = new StringTokenizer(stdin.readLine());
		return st.nextToken();
	}

	static int readInt() throws Exception {
		return Integer.parseInt(readString());
	}

}
