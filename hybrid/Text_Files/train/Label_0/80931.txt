s = input()
ans = [0 for i in range(64)]
for i in range(len(ans)):
	temp = bin(i)
	ans[i] = temp.count('0') - 1 + (6 - (len(temp) - 2))
pri = 10 ** 9 + 7
count = 1
for i in range(len(s)):
	if s[i].isdigit():
		r = bin(int(s[i]))
		r = r[2:]
		count = count * pow(3, ans[int(s[i])], pri)
		count %= pri
	elif s[i] == '-':
		count = count * pow(3, ans[62], pri)
		count %= pri
	elif s[i] == '_':
		count = count * pow(3, ans[63], pri)
		count %= pri
	elif 65 <= ord(s[i]) <= 91:
		count = count * pow(3, ans[ord(s[i]) - 65 + 10], pri)
		count %= pri
	elif 97 <= ord(s[i]) <= 122:
		count = count * pow(3, ans[ord(s[i]) - 97 + 36], pri)
		count %= pri
print(count % pri)
