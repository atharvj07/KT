#include <bits/stdc++.h>
using namespace std;
int dp[3][55];
int pie[55];
int recur(int token, int pos, int N) {
  if (pos == (N - 1)) {
    if (token == 1)
      return pie[pos];
    else
      return -pie[pos];
  }
  if (dp[token][pos] != -1) return dp[token][pos];
  int mx = 0;
  if (token == 1) {
    mx = pie[pos] - recur(1, pos + 1, N);
    mx = max(mx, -pie[pos] - recur(0, pos + 1, N));
  } else {
    mx = pie[pos] - recur(1, pos + 1, N);
    mx = min(mx, -pie[pos] - recur(0, pos + 1, N));
  }
  dp[token][pos] = mx;
  return dp[token][pos];
}
int main() {
  int N;
  cin >> N;
  int total = 0;
  for (int i = 0; i < N; i++) {
    scanf("%d", &pie[i]);
    total += pie[i];
  }
  memset(dp, -1, sizeof(dp));
  int diff = recur(0, 0, N);
  int bob = (total + diff) / 2;
  int alice = total - bob;
  cout << bob << " " << alice << endl;
  return 0;
}
