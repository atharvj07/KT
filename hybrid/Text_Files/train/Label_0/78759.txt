n = int(input())
a = input()
b = input()
da = {}
db = {}
l1 = []
l2 = []
for i in range(n):
	if a[i] == '?':
		l1.append(i)
	elif a[i] in da:
		da[a[i]].append(i)
	else:
		da[a[i]] = [i]
	if b[i] == '?':
		l2.append(i)
	elif b[i] in db:
		db[b[i]].append(i)
	else:
		db[b[i]] = [i]
ans = []
for key in da.keys():
	if key in db:
		while len(da[key]) > 0 and len(db[key]) > 0:
			(x, y) = (da[key].pop(), db[key].pop())
			ans.append((x + 1, y + 1))
for key in db.keys():
	while len(l1) > 0 and len(db[key]) > 0:
		(x, y) = (l1.pop(), db[key].pop())
		ans.append((x + 1, y + 1))
for key in da.keys():
	while len(l2) > 0 and len(da[key]) > 0:
		(y, x) = (l2.pop(), da[key].pop())
		ans.append((x + 1, y + 1))
while len(l1) > 0 and len(l2) > 0:
	(x, y) = (l1.pop(), l2.pop())
	ans.append((x + 1, y + 1))
print(len(ans))
for i in ans:
	print(i[0], i[1])
