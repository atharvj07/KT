#include <bits/stdc++.h>
using namespace std;
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  int q;
  cin >> q;
  for (int i = 0; i < q; i++) {
    long long a, b, m;
    cin >> a >> b >> m;
    if (a == b) {
      cout << "1 " << a << endl;
      continue;
    }
    long long p = 1;
    int n = 2;
    for (; n <= 50; n++) {
      if ((a + 1LL) <= b / p && (b + p - 1) / p <= (a + m)) {
        break;
      }
      p = (p << 1LL);
    }
    if (!((a + 1LL) <= (b >> (n - 2)) && (b >> (n - 2)) <= (a + m))) {
      cout << -1 << "\n";
      continue;
    }
    long long r = b - p * a;
    long long prev = a;
    p = p >> 1;
    cout << n << " " << a;
    while (p) {
      long long c = min((r - p) / p, m);
      cout << " " << c + prev;
      prev += c + prev;
      r = r - p * c;
      p = p >> 1LL;
    }
    cout << " " << b << endl;
  }
}
