#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  int b[n];
  for (int x = 0; x < n; x++) {
    cin >> b[x];
    a[x] = b[x];
  }
  int currentans = 0, currentlikes = 0;
  int ans = 0, likes = 0;
  int poss = 1;
  sort(a, a + n);
  for (int x = 0; x < n; x++) {
    currentans = a[x];
    currentlikes = 0;
    for (int y = x; y < n; y++) {
      if (a[y] == a[x]) {
        currentlikes++;
      } else {
        break;
      }
    }
    if (likes < currentlikes) {
      ans = currentans;
      likes = currentlikes;
      poss = 1;
    } else if (likes == currentlikes) {
      poss++;
    }
  }
  int possibleans[poss], zzz = 0;
  if (poss == 1) {
    cout << ans;
  } else {
    for (int x = 0; x < n; x++) {
      currentans = a[x];
      currentlikes = 0;
      for (int y = x; y < n; y++) {
        if (a[y] == a[x]) {
          currentlikes++;
        } else {
          break;
        }
      }
      if (currentlikes == likes) {
        poss--;
        possibleans[zzz] = a[x];
        zzz++;
        if (poss == 0) {
          break;
        }
      }
    }
    int abc = zzz;
    for (int x = n - 1; x >= 0; x--) {
      for (int y = 0; y <= zzz; y++) {
        if (b[x] == possibleans[y]) {
          abc--;
          if (abc == 0) {
            cout << possibleans[y];
            break;
          } else {
            possibleans[y] = -1;
          }
          break;
        }
      }
      if (abc == 0) {
        break;
      }
    }
  }
}
