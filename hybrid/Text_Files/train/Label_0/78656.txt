import sys
input = sys.stdin.readline

def getInt():
	return int(input())

def getVars():
	return map(int, input().split())

def getList():
	return list(map(int, input().split()))

def getStr():
	return input().strip()
(n, m) = getVars()
a = input().split()
b = input().split()
q = getInt()
for i in range(q):
	x = getInt()
	print(a[x % n - 1], b[x % m - 1], sep='')
