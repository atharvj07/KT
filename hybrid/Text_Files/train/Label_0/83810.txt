t = int(input())
for _ in range(t):
	(n, m, k) = map(int, input().split())
	no_of_blocks = m
	a = list(map(int, input().split()))
	flag = True
	for i in range(n - 1):
		if abs(a[i + 1] - a[i]) <= k:
			if a[i] >= a[i + 1]:
				no_of_blocks += abs(a[i] - a[i + 1])
				if a[i + 1] >= k:
					no_of_blocks += k
				else:
					no_of_blocks += a[i + 1]
			elif a[i + 1] - k >= 0:
				no_of_blocks += abs(k - abs(a[i] - a[i + 1]))
			else:
				no_of_blocks += a[i]
		elif a[i] >= a[i + 1]:
			no_of_blocks += abs(a[i] - a[i + 1])
			if a[i + 1] - k >= 0:
				no_of_blocks += k
			else:
				no_of_blocks += a[i + 1]
		else:
			no_of_blocks -= abs(abs(a[i + 1] - k) - a[i])
		if no_of_blocks < 0:
			flag = False
	if flag == True:
		print('YES')
	else:
		print('NO')
