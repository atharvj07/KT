#include <string>
#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int n, a[100009], b[100009], c[100009]; long long dp[100009];
int main() {
	ios::sync_with_stdio(false);
	cin >> n;
	for (int i = 0; i < n; i++) cin >> a[i];
	for (int i = 0; i < n; i++) cin >> b[i];
	for (int i = 0; i < n; i++) cin >> c[i];
	sort(a, a + n);
	sort(b, b + n);
	sort(c, c + n);
	long long ret = 0;
	for (int i = 0; i < n; i++) {
		int ptr1 = lower_bound(a, a + n, b[i]) - a;
		int ptr2 = lower_bound(c, c + n, b[i] + 1) - c;
		ret += 1LL * ptr1 * (n - ptr2);
	}
	cout << ret << endl;
	return 0;
}
