def binNumber(n, size=1):
	return bin(n)[2:].zfill(size)

def iar():
	return list(map(int, input().split()))

def ini():
	return int(input())

def isp():
	return map(int, input().split())

def sti():
	return str(input())
n = ini()
i = 1
a = []
while i * i <= n:
	if n % i == 0:
		t = n // i
		s = t * (1 + 1 + (t - 1) * i) // 2
		a.append(s)
		if i * i != n:
			t = n // t
			j = n // i
			s = t * (2 + (t - 1) * j) // 2
			a.append(s)
	i += 1
a.sort()
for i in a:
	print(i, end=' ')
