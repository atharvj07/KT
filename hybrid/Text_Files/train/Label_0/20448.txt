#include <bits/stdc++.h>
using namespace std;
const int SIZE = 50010;
int a[SIZE][3], b[SIZE][3];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < (n); ++i)
    for (int j = 0; j < (3); ++j) scanf("%d", &(a[i][j]));
  for (int i = 0; i < (m); ++i)
    for (int j = 0; j < (3); ++j) scanf("%d", &(b[i][j]));
  int an = 0;
  for (int i = 0; i < (n); ++i)
    for (int j = 0; j < (m); ++j) {
      if (a[i][1] <= b[j][1] && b[j][1] <= a[i][1] + a[i][2]) {
        if (b[j][0] <= a[i][0] && a[i][0] <= b[j][0] + b[j][2]) {
          an = max(min(min(b[j][1] - a[i][1], a[i][1] + a[i][2] - b[j][1]),
                       min(a[i][0] - b[j][0], b[j][0] + b[j][2] - a[i][0])),
                   an);
        }
      }
    }
  cout << an << endl;
  return 0;
}
