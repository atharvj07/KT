import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	l = 0
	r = 0
	lis1 = []
	lis2 = [0]
	for i in range(n - 1, -1, -1):
		if arr[i] == 1:
			l += 1
		else:
			l -= 1
		lis1.append(l)
	for i in range(n, 2 * n):
		if arr[i] == 1:
			r += 1
		else:
			r -= 1
		lis2.append(r)
	lis1.reverse()
	lis1.append(0)
	dict1 = {}
	for i in range(n + 1):
		if lis2[i] not in dict1.keys():
			dict1[lis2[i]] = i
	ans = 2 * n
	for i in range(n + 1):
		if l + r - lis1[i] in dict1.keys():
			ans = min(ans, dict1[l + r - lis1[i]] + n - i)
	print(ans)
