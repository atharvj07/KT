#include <bits/stdc++.h>
using namespace std;
int n, d, k, i, x, b[222000];
map<int, int> a;
int main() {
  for (cin >> n >> d >> d, i = 1; i <= n; i++) cin >> x, a[x] = i;
  for (; !a.empty();) {
    k++;
    for (auto p = a.begin(); p != a.end();) {
      b[p->second] = k;
      x = p->first;
      a.erase(p);
      p = a.lower_bound(x + d + 1);
    }
  }
  cout << k << endl;
  for (i = 1; i <= n; i++) cout << b[i] << " ";
}
