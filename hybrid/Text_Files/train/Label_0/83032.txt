mod = 10 ** 9 + 7

def ipl():
	return list(map(int, ip().split()))

def ipn():
	return int(ip())

def ipf():
	return float(ip())

def solve():
	n = ipn()
	(a, b) = (ipl(), ipl())
	g = [-1 for _ in range(n + 1)]
	for i in range(n):
		g[a[i]] = b[i]
	v = [False] * (n + 1)
	r = 0
	for i in range(1, n + 1):
		if v[i]:
			continue
		r += 1
		j = i
		while not v[j]:
			v[j] = True
			j = g[j]
	res = 1
	for i in range(r):
		res = res * 2 % mod
	print(res)
t = ipn()
for _ in range(t):
	solve()
