public class pizza implements Runnable{
	public static void main(String[] args) throws IOException {
		new Thread(null, new pizza(), "", 1<<28).start();
	}
	public void run() {
		try {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt(), m = in.nextInt();
		MagicComponents g = new MagicComponents(n);
		for (int i = 0; i < m; i++) {
			int u = in.nextInt() - 1, v = in.nextInt() - 1;
			g.addEdge(u, v);
		}
		g.run();
		int[] ids = new int[n];
		int ID = 0;
		ArrayList<Edge> res = new ArrayList<>();
		for (ArrayList<Edge> bcc : g.bccs) {
			if (bcc.size() == 1)
				continue;
			ID++;
			int nodeCount = 0;
			for (Edge e : bcc) {
				if (ids[e.u] != ID) {
					ids[e.u] = ID;
					nodeCount++;
				}
				if (ids[e.v] != ID) {
					ids[e.v] = ID;
					nodeCount++;
				}
			}
			int edgeCount = bcc.size();
			if (nodeCount == edgeCount) {
				res.addAll(bcc);
			}
		}
		Collections.sort(res, Comparator.comparing(a -> a.id));
		out.println(res.size());
		for (int i = 0; i < res.size(); i++) {
			if (i != 0)
				out.print(' ');
			out.print(res.get(i).id + 1);
		}
		out.println();
		out.close();
		} catch(Throwable t) {
			t.printStackTrace();
			System.exit(1);
		}
	}

	static class Edge {
		int u, v, id;

		public Edge(int U, int V, int i) {
			u = U;
			v = V;
			id = i;
		}
	}

	static class MagicComponents {
		int num, n, edges;
		int[] dfsNum, low, vis;
		ArrayList<Edge> adj[];
		ArrayList<ArrayList<Edge>> bccs;
		ArrayDeque<Edge> eStack;

		MagicComponents(int N) {
			adj = new ArrayList[n = N];
			for (int i = 0; i < n; i++)
				adj[i] = new ArrayList<>();
			edges = 0;
		}

		void addEdge(int u, int v) {
			adj[u].add(new Edge(u, v, edges));
			adj[v].add(new Edge(v, u, edges++));
		}

		void run() {
			vis = new int[n];
			dfsNum = new int[n];
			low = new int[n];
			bccs = new ArrayList<>();
			eStack = new ArrayDeque<>();
			num = 0;

			for (int i = 0; i < n; i++) {
				if (vis[i] != 0)
					continue;
				dfs(i, -1);
			}
		}

		void dfs(int node, int par) {
			dfsNum[node] = low[node] = num++;
			vis[node] = 1;
			for (Edge e : adj[node]) {
				if (e.v == par)
					continue;
				if (vis[e.v] == 0) {
					eStack.push(e);
					dfs(e.v, node);
					low[node] = Math.min(low[node], low[e.v]);
					if (low[e.v] >= dfsNum[node]) {
						if (low[e.v] > dfsNum[node]) {
							pop(node);
						} else
							pop(node);
					}
				} else if (vis[e.v] == 1) {
					low[node] = Math.min(low[node], dfsNum[e.v]);
					eStack.push(e);
				}
			}
			vis[node] = 2;
		}

		void pop(int u) {
			ArrayList<Edge> list = new ArrayList<>();
			for (;;) {
				Edge e = eStack.pop();
				list.add(e);
				if (e.u == u)
					break;
			}
			bccs.add(list);
		}
	}

	static class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(InputStream i) {
			br = new BufferedReader(new InputStreamReader(i));
			st = null;
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}
	}
}
