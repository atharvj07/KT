// practice with rainboy
import java.io.*;
import java.util.*;

public class CF333D extends PrintWriter {
	CF333D() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF333D o = new CF333D(); o.main(); o.flush();
	}

	int n, m, aa[][], jj[];
	boolean[][] used;
	boolean check(int a) {
		for (int j = 0; j < m; j++)
			Arrays.fill(used[j], false);
		for (int i = 0; i < n; i++) {
			int cnt = 0;
			for (int j = 0; j < m; j++)
				if (aa[i][j] >= a)
					jj[cnt++] = j;
			for (int l = 0; l < cnt; l++)
				for (int r = l + 1; r < cnt; r++) {
					if (used[jj[l]][jj[r]])
						return true;
					used[jj[l]][jj[r]] = true;
				}
		}
		return false;
	}
	void main() {
		n = sc.nextInt();
		m = sc.nextInt();
		aa = new int[n][m];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++)
				aa[i][j] = sc.nextInt();
		jj = new int[m];
		used = new boolean[m][m];
		int lower = 0, upper = 1000000001;
		while (upper - lower > 1) {
			int a = (lower + upper) / 2;
			if (check(a))
				lower = a;
			else
				upper = a;
		}
		println(lower);
	}
}
