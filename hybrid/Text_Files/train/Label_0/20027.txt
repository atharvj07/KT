#include <bits/stdc++.h>
using namespace std;
map<char, int> k;
int main() {
  int n;
  cin >> n;
  if (n % 4 != 0) {
    cout << "===";
    return 0;
  }
  int kol = n / 4;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] != '?') {
      k[s[i]]++;
    }
  }
  if (k['A'] > kol || k['G'] > kol || k['C'] > kol || k['T'] > kol) {
    cout << "===";
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      if (k['A'] < kol) {
        cout << 'A';
        k['A']++;
        continue;
      }
      if (k['G'] < kol) {
        cout << 'G';
        k['G']++;
        continue;
      }
      if (k['C'] < kol) {
        cout << 'C';
        k['C']++;
        continue;
      }
      if (k['T'] < kol) {
        cout << 'T';
        k['T']++;
        continue;
      }
    } else {
      cout << s[i];
    }
  }
  return 0;
}
