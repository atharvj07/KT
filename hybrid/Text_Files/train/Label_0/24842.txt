#include<bits/stdc++.h>
using namespace std;
template<typename T>inline T read(){
	T x=0,f=0;char c=getchar();
	while(!isdigit(c)) f=c=='-',c=getchar();
	while(isdigit(c)) x=x*10+c-48,c=getchar();
	return f?-x:x;
}
namespace run{
	inline char getc(){
		char c=getchar();
		while(c!='X' && c!='.') c=getchar();
		return c;
	}
	const int N=509;
	int n,m,mapx[N][N],ans[N][N];
	int main(){
		int T=read<int>();
		while(T--){
			n=read<int>(),m=read<int>();
			for(int i=1;i<=n;i++)
				for(int j=1;j<=m;j++){
					char c=getc();
					mapx[i][j]=c=='X',ans[i][j]=c=='X';
				}
	
			int tmp=1;
			if(n%3==0) tmp=2;
			for(int i=tmp;i<=n;i+=3)
				for(int j=1;j<=m;j++) ans[i][j]=1;
			for(int i=tmp;i+3<=n;i+=3){
				int peg=0;
				for(int j=1;j<=m;j++)
					if(ans[i+1][j] && ans[i+2][j]) peg=1;
				if(!peg){
					if(m==1 || (!ans[i+1][2] && !ans[i+2][2])) ans[i+1][1]=ans[i+2][1]=1;
					else ans[i+1][2]=ans[i+2][2]=1;
				}
			}

			for(int i=1;i<=n;i++){
				for(int j=1;j<=m;j++){
					if(ans[i][j]) putchar('X');
					else putchar('.');
				}
				puts("");
			}
		}
		return 0;
	}
}
