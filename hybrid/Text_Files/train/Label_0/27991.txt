#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:100000000000000")
using namespace std;
const long long int INF = 2e9 + 1;
int a[101][101];
bitset<101> b[101], c[101];
int try_ans(bitset<101> yes, int n, int m, bool flag = false) {
  int cnt = 0;
  bitset<101> temp;
  for (int i = 0; i < (n); i++) {
    temp = (!flag ? b[i] : c[i]) ^ yes;
    int one = temp.count();
    cnt += min(m - one, one);
  }
  return cnt;
}
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 0; i < (n); i++) {
    for (int j = 0; j < (m); j++) {
      scanf("%d", &a[i][j]);
      b[i][j] = a[i][j];
      c[j][i] = a[i][j];
    }
  }
  int out = k + 1;
  if (n <= k) {
    bitset<101> temp(0);
    for (int i = 0; i < (1 << n); i++) {
      for (int j = 0; j < (n); j++) {
        temp[j] = ((i & (1 << j)) > 0);
      }
      out = min(out, try_ans(temp, m, n, true));
    }
  } else {
    for (int i = 0; i < (n); i++) {
      out = min(out, try_ans(b[i], n, m));
    }
  }
  cout << (out > k ? -1 : out);
}
