n = int(input())
s = input()
a = {}
l = []
i = 0
while i < len(s):
	f = True
	for j in range(len(l)):
		if l[j][0] == s[i]:
			f = False
			break
	if f == True:
		l.append(s[i])
	else:
		l[-1] = l[-1] + s[i]
	i += 1
if len(l) < n:
	print('NO')
else:
	print('YES')
	for i in range(n, len(l)):
		l[n - 1] += l[i]
	for i in range(n):
		print(l[i])
