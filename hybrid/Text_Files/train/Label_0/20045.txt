#include <bits/stdc++.h>
using namespace std;
vector<int> g[100005];
int find(int v, int* parent) {
  if (parent[v] == v) {
    return v;
  }
  return find(parent[v], parent);
}
int vis[100005];
int dp[100005];
int dfs(int u) {
  vis[u] = 1;
  for (int i = 0; i < g[u].size(); i++) {
    int v = g[u][i];
    if (!vis[v]) dfs(v);
  }
}
int main() {
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) cin >> arr[i];
  int ans = 1;
  int cnt = 1;
  for (int i = 1; i < n; i++) {
    if (arr[i] <= 2 * arr[i - 1]) {
      cnt++;
    } else {
      cnt = 1;
    }
    ans = max(cnt, ans);
  }
  cout << ans;
  return 0;
}
