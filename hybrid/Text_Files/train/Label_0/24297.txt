#include <bits/stdc++.h>
using namespace std;
int const inf = 1000000000;
vector<int> V;
bool ok(int mid) {
  int c = V[0], i;
  for (i = 0; i < V.size(); i++)
    if (V[i] - c > mid) break;
  if (i < V.size()) {
    c = V[i];
    for (; i < V.size(); i++)
      if (V[i] - c > mid) break;
    if (i < V.size()) {
      c = V[i];
      for (; i < V.size(); i++)
        if (V[i] - c > mid) break;
    }
  }
  if (i == V.size()) return true;
  return false;
}
int main() {
  V.clear();
  int n;
  cin >> n;
  int i;
  for (i = 0; i < n; i++) {
    int a;
    cin >> a;
    V.push_back(a);
  }
  sort(V.begin(), V.end());
  int low = 0;
  int high = 2000000000;
  int t = 100;
  while (t--) {
    int mid = (low + high) / 2;
    if (ok(mid))
      high = mid;
    else
      low = mid;
  }
  int k = max(0, low - 5);
  for (i = k;; i++) {
    if (ok(i)) break;
  }
  int mid, c;
  mid = i;
  printf("%.6lf\n", i / 2.0);
  c = V[0];
  for (i = 0; i < V.size(); i++)
    if (V[i] - c > mid) break;
  printf("%.6lf", (V[i - 1] - c) / 2.0 + c);
  if (i < V.size()) {
    c = V[i];
    for (; i < V.size(); i++)
      if (V[i] - c > mid) break;
    printf(" %.6lf", (V[i - 1] - c) / 2.0 + c);
    if (i < V.size()) {
      c = V[i];
      for (; i < V.size(); i++)
        if (V[i] - c > mid) break;
      assert(i == V.size());
      printf(" %.6lf", (V[i - 1] - c) / 2.0 + c);
    } else {
      printf(" %.6lf", (V[i - 1] - c) / 2.0 + c);
    }
  } else {
    printf(" %.6lf %.6lf\n", (V[i - 1] - c) / 2.0 + c,
           (V[i - 1] - c) / 2.0 + c);
  }
  return 0;
}
