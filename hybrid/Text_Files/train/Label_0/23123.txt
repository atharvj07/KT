#include <iostream>

using namespace std;

int main(void){
	int n, h[101], after[100];
	while(1){
		cin >> n;
		if(n == 0){
			break;
		}
		for(int i = 0;i < n + 1;i++){
			cin >> h[i];
		}
		for(int i = 0;i < n + 1;i++){ // 一つ引き抜く
			int cnt = 0;
			for(int j = 0;j < n + 1;j++){
				if(i != j){
					after[cnt++] = h[j];
				}
			}
			int sub = after[1] - after[0];
			int j = 0;
			for(j = 1;j < n;j++){
				if(sub != after[j] - after[j - 1]){
					break;
				}
			}
			if(j == n){
				cout << h[i] << endl;
				break;
			}
		}


	}
	return 0;
}
