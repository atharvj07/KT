#include <bits/stdc++.h>
using namespace std;
bool a[10];
string D;
long long L;
long long m = 1000;
long long power(long long u, long long a) {
  if (a == 0) return 1;
  if (a == 1)
    return u;
  else {
    if (a % 2 == 0)
      return power(u, a / 2) * power(u, a / 2);
    else
      return power(u, a / 2) * power(u, a / 2) * u;
  }
}
long long change_string(string S) {
  long long A = 0;
  long long L = S.length();
  if (L == 0) return -1000;
  for (long long i = 0; i < L; i++) {
    long long T = S[i] - '0';
    A = A + T * power(10, L - i - 1);
  }
  return A;
}
long long clear_zero(string S) {
  long long A = 0;
  for (long long i = 0; i < S.length(); i++) {
    if (S[i] == '0')
      A++;
    else
      return A;
  }
  return A;
}
void bt(long long u, string T) {
  if (u == 0) string T = "";
  if (u == L) {
    long long W = change_string(T);
    if (W == -1000) return;
    long long P = sqrt(W);
    if (P * P == W) {
      if (clear_zero(T) == T.length()) return;
      long long k = L - T.length();
      m = min(k + clear_zero(T), m);
    }
    return;
  }
  a[u] = 0;
  bt(u + 1, T);
  a[u] = 1;
  T = T + D[u];
  bt(u + 1, T);
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> D;
  L = D.length();
  string T = "";
  bt(0, T);
  if (m == 1000)
    cout << "-1";
  else
    cout << m;
}
