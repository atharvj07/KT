from math import ceil
k = int(input())
s = input()
t = input()
soma_s = []
soma_t = []
for i in range(k):
	soma_s.append(ord(s[i]) - ord('a'))
	soma_t.append(ord(t[i]) - ord('a'))
for i in range(k - 1, -1, -1):
	soma_s[i] += soma_t[i]
	if i != 0:
		soma_s[i - 1] += soma_s[i] // 26
		soma_s[i] = soma_s[i] % 26
med = ''
for i in range(k):
	valor = soma_s[i] // 2
	if i < k - 1:
		soma_s[i + 1] += soma_s[i] % 2 * 26
	med += chr(ord('a') + valor)
print(med)
