#include <bits/stdc++.h>
using namespace std;
vector<int> getKmpPrefixArray(string s) {
  vector<int> v(s.size());
  int i = 1, j = 0;
  v[0] = 0;
  while (i < s.size()) {
    if (s[i] == s[j])
      v[i] = j + 1, j++, i++;
    else {
      while (j != 0 && s[j] != s[i]) j = v[j - 1];
      if (s[j] != s[i]) v[i] = j, i++;
    }
  }
  return v;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  long long int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> a = getKmpPrefixArray(s);
  long long int t = a[a.size() - 1], j;
  for (long long int i = 1; i < k; i++) {
    for (j = t; j < n; j++) s.push_back(s[j]);
  }
  cout << s;
  return 0;
}
