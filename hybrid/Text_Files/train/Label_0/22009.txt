#include <bits/stdc++.h>
using namespace std;
int main() {
  int n = 0, k = 0;
  cin >> n >> k;
  int a[2 * n + 1];
  for (int i = 0; i < 2 * n + 1; i++) {
    cin >> a[i];
  }
  for (int j = 0; j < 2 * n + 1; j++) {
    if ((j + 1) % 2 == 0 && k != 0 && a[j - 1] < a[j] - 1 &&
        a[j] - 1 > a[j + 1]) {
      k--;
      a[j]--;
    }
    cout << a[j];
    if (j != 2 * n) cout << " ";
  }
  return 0;
}
