import java.util.*;
public class Fafa{
    public static void main(String[] args){
        long c=0;
        Scanner sc=new Scanner(System.in);
        long[] ar=new long[4];
        for(int i=0;i<4;i++){
            ar[i]=sc.nextInt(); 
            c=c+ar[i];
        }
        long sum=c/3;
        for(int i=0;i<4;i++){
            if(ar[i]<sum){
                System.out.print((sum-ar[i]));
                System.out.print(" ");}
        }
    }
}
