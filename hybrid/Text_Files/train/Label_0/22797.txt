#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
const long long maxn = 2e5 + 5;
long long n, dp[maxn];
vector<long long> v[maxn];
double ans[maxn];
void calc(long long id) {
  dp[id] = 1;
  for (const auto &it : v[id]) calc(it), dp[id] += dp[it];
}
void dfs(long long id, long long par) {
  if (id == 1) ans[id] = 1.0;
  double tot = dp[id] - 1;
  for (const auto &it : v[id]) {
    ans[it] = ans[id] + 1 + (tot - dp[it]) / 2.0;
    dfs(it, id);
  }
}
int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> n;
  for (long long i = 0; i < n - 1; ++i) {
    long long x;
    cin >> x;
    v[x].emplace_back(i + 2);
  }
  calc(1);
  dfs(1, -1);
  for (long long i = 1; i <= n; ++i) {
    printf("%.10f ", ans[i]);
  }
  printf("\n");
  return 0;
}
