import java.util.*;
import java.lang.*;
import java.io.*;

public class Main
{
	PrintWriter out = new PrintWriter(System.out);
	BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer tok = new StringTokenizer("");
    String next() throws IOException {
        if (!tok.hasMoreTokens()) { tok = new StringTokenizer(in.readLine()); }
        return tok.nextToken();
    }
    int ni() throws IOException { return Integer.parseInt(next()); }
    long nl() throws IOException { return Long.parseLong(next()); }
    
    long mod=1000000007;
    
    void solve() throws IOException {
        for (int tc=1;tc>0;tc--) {
            int[]a=new int[7];
            for (int i=1;i<7;i++) a[i]=ni();
            int n=ni();
            int[]b=new int[n+1];
            for (int i=1;i<=n;i++) b[i]=ni();
            
            Arrays.sort(b);
            Arrays.sort(a);
            
            int ans=Integer.MAX_VALUE;
            
            for (int i=1;i<7;i++) {
                int target=b[1]-a[i];
                int[][]near=new int[n+1][2];
                for (int j=2;j<=n;j++) {
                    near[j][1]=1000000001;
                    near[j][0]=1000000001;
                    for (int k=1;k<7;k++) {
                        int next=b[j]-a[k];
                        if (next>target) near[j][1]=Math.min(near[j][1],next-target);
                        else near[j][0]=Math.min(near[j][0],target-next);
                    }
                }
                near[0][0]=-1;
                near[1][0]=-1;
                Arrays.sort(near,(p,q)->Integer.compare(q[0],p[0]));
                near[n-1][0]=0;
                
                int maxspread=near[0][0];
                int currminabove=0;
                int currminbelow=maxspread;
                for (int j=0;j<n-1;j++) {
                    currminbelow=near[j+1][0];
                    currminabove=Math.max(currminabove,near[j][1]);
                    maxspread=Math.min(maxspread,currminbelow+currminabove);
                }
                
                ans=Math.min(ans,maxspread);
            }
            out.println(ans);
        }
        out.flush();
    }
    
    int gcd(int a,int b) { return(b==0?a:gcd(b,a%b)); }
    long gcd(long a,long b) { return(b==0?a:gcd(b,a%b)); }
    long mp(long a,long p) { long r=1; while(p>0) { if ((p&1)==1) r=(r*a)%mod; p>>=1; a=(a*a)%mod; } return r; }
    
    public static void main(String[] args) throws IOException {
        new Main().solve();
    }
}
