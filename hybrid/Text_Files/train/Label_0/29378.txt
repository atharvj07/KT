int getnum(const vector<int>&as, int K) {
	int ans = 0;
	for (int l = 0; l <= as.size() - K; ++l) {
		int num = 1;
		for (int i = l; i < l + K; ++i) {
			num *= as[i];
		}
		ans = max(ans, num);
	}
	return ans;
}
