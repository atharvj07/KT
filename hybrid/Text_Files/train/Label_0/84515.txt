def palindrome(a):
	x = ''
	for i in range(len(a) - 1, -1, -1):
		x += a[i]
	if x == a:
		return 1
	else:
		return 0
k = input()
n = int(input())
f = 'YES'
if len(k) % n == 0:
	for i in range(0, len(k), len(k) // n):
		if palindrome(k[i:i + len(k) // n]) == 0:
			f = 'NO'
	print(f)
else:
	print('NO')
