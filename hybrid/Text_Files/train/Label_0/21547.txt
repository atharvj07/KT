#include <bits/stdc++.h>
using namespace std;
const int EXP = 5;
const int PEX = 32;
const int DIM = 100005;
long long dp[PEX];
int hor[DIM][EXP], ver[DIM][EXP];
int main(void) {
  int n, m;
  scanf("%d %d", &n, &m);
  for (int i = 1; i < m; ++i)
    for (int j = 0; j < n; ++j) scanf("%d", &hor[i][j]);
  for (int i = 0; i < m; ++i)
    for (int j = 0; j < n; ++j) scanf("%d", &ver[i][j]);
  for (int msk = 1; msk < (1 << n); ++msk)
    dp[msk] = numeric_limits<long long>::max() / 2;
  for (int i = 1; i < m; ++i) {
    for (int msk = 0; msk < (1 << n); ++msk) {
      for (int j = 0; j < n; ++j)
        if (!((msk >> j) & 1))
          dp[msk ^ (1 << j)] = min(dp[msk ^ (1 << j)], dp[msk] + hor[i][j]);
      for (int j = 0; j < n; ++j)
        if (((msk >> j) & 1) ^ ((msk >> ((j + 1) % n)) & 1))
          dp[msk] += ver[i][j];
    }
  }
  printf("%I64d\n", dp[(1 << n) - 1]);
  return 0;
}
