import java.util.*;
import java.lang.*;
import java.io.*;
//****Use Integer Wrapper Class for Arrays.sort()****
public class BS3 {
    public static void main(String[] Args){
        FastReader scan=new FastReader();
        int t=scan.nextInt();
        StringBuilder print=new StringBuilder();
        while(t-->0) {
            String s = scan.next();
            Node[] arr=new Node[26];
            for(int i=0;i<26;i++){
                arr[i]=new Node();
            }
            int n=s.length();
            boolean flag=true;
            int start=s.charAt(0)-'a';
            for(int i=1;i<n;i++){
                char ch= s.charAt(i);
                char prev=s.charAt(i-1);
                arr[ch-'a'].ns.add(prev);
                arr[prev-'a'].ns.add(ch);
            }
            int onec=0;
            for(int i=0;i<26;i++){
                if(arr[i].ns.size()==1){
                    onec++;
                    start=i;
                }
                if(arr[i].ns.size()>2){
                    flag=false;
                }
            }
            if(!flag||(onec!=2&&s.length()>1)){
                print.append("NO\n");
            }
            else {
                Set<Character> vis = new HashSet<>();
                StringBuilder ans=new StringBuilder();
                ans.append((char)(start+'a'));
                vis.add((char)(start+'a'));
                Queue<Integer> q=new LinkedList<>();
                ((LinkedList<Integer>) q).add(start);
                while(!q.isEmpty()){
                    int cur=q.remove();
                    for(Character c:arr[cur].ns){
                        if(!vis.contains(c)) {
                            ((LinkedList<Integer>) q).add(c - 'a');
                            vis.add(c);
                            ans.append(c);
                        }
                    }
                }
                print.append("YES\n");
                for (int i = 0; i < 26; i++) {
                    if (!vis.contains((char) ('a' + i))) {
                        ans.append((char) ('a' + i));
                    }
                }
                print.append(ans+"\n");
            }
        }
        System.out.println(print);
    }

    static class Node{
        Set<Character> ns;
        Node(){
            ns=new HashSet<>();
        }
    }
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader()
        {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException  e)
                {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt()
        {
            return Integer.parseInt(next());
        }

        long nextLong()
        {
            return Long.parseLong(next());
        }

        double nextDouble()
        {
            return Double.parseDouble(next());
        }

        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                e.printStackTrace();
            }
            return str;
        }
    }

}
