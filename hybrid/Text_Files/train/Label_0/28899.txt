#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, x;
  cin >> n >> x;
  vector<long long> a(n);
  map<long long, long long> a1;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (a1.count(a[i]) == 1)
      a1[a[i]]++;
    else {
      pair<long long, long long> x;
      x.first = a[i];
      x.second = 1;
      a1.insert(x);
    }
  }
  sort(a.rbegin(), a.rend());
  long long quan = 0;
  for (long long i = 0; i < n; i++) {
    long long temp = x ^ a[i];
    if (a1.count(temp) == 1 and temp <= a[i]) {
      quan += a1[temp];
      if (a[i] == temp) {
        a1[temp]--;
        quan--;
      }
    }
  }
  cout << quan;
  return 0;
}
