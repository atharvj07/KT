#include<cstdio>

#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

int ans[12000];

void add(int nu,int de){ // 答えに nu/de を加算
	rep(i,12000){
		ans[i]+=nu/de;
		nu%=de;
		nu*=10;
	}
}

int main(){
	for(int n,m,r;scanf("%d%*d%d%d",&n,&m,&r),n;){
		rep(i,12000) ans[i]=0;

		if(m==0){
			add(1,n);
		}
		else{
			add(1,n);
			rep(i,n-1) add(1,(i+1)*n);
		}

		for(int i=11999;i>0;i--) if(ans[i]>=10) {
			int d=ans[i]/10;
			ans[i]-=d*10;
			ans[i-1]+=d;
		}

		printf("%d.",ans[0]);
		rep(i,r) putchar('0'+ans[i+1]);
		puts("");
	}

	return 0;
}
