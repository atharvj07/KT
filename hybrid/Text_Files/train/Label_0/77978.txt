from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	s = input().rstrip()
	n = len(s)
	(a, b) = ([0] * n, [0] * n)
	a[0] = int(s[0])
	b[0] = 1 - a[0]
	for x in range(1, n):
		a[x] = a[x - 1]
		b[x] = b[x - 1]
		if s[x] == '1':
			a[x] += 1
		else:
			b[x] += 1
	now = min(a[-1], b[-1])
	for x in range(n - 1):
		now = min(now, a[x] + b[-1] - b[x], b[x] + a[-1] - a[x])
	print(now)
