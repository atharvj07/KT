n = int(input())
A = list(map(int, input().split()))
count = [0] * 1001
for a in A:
	count[a] += 1
if n % 2 == 1:
	can = True
	one = []
	two = []
	fore = []
	for i in range(1001):
		c = count[i]
		if c % 4 == 0:
			fore += [i] * (c // 4)
		elif c % 4 == 1:
			one += [i]
			fore += [i] * (c // 4)
		elif c % 4 == 2:
			two += [i]
			fore += [i] * (c // 4)
		else:
			one += [i]
			two += [i]
			fore += [i] * (c // 4)
	if len(one) != 1:
		can = False
	if len(two) > n - 1:
		can = False
	if can:
		answer = [[0] * n for i in range(n)]
		print('YES')
		current = 0
		for i in range((n + 1) // 2):
			for j in range((n + 1) // 2):
				if i == (n - 1) // 2:
					if j == (n - 1) // 2:
						answer[i][j] = one[-1]
					else:
						if len(two) == 0:
							two += [fore[-1]] * 2
							fore = fore[:-1]
						answer[i][j] = two[-1]
						answer[i][n - j - 1] = two[-1]
						two = two[:-1]
				elif j == (n - 1) // 2:
					if len(two) == 0:
						two += [fore[-1]] * 2
						fore = fore[:-1]
					answer[i][j] = two[-1]
					answer[n - i - 1][j] = two[-1]
					two = two[:-1]
				else:
					answer[i][j] = fore[-1]
					answer[n - i - 1][j] = fore[-1]
					answer[i][n - j - 1] = fore[-1]
					answer[n - i - 1][n - j - 1] = fore[-1]
					fore = fore[:-1]
		for line in answer:
			print(' '.join(list(map(str, line))))
	else:
		print('NO')
else:
	can = True
	fore = []
	for i in range(1001):
		c = count[i]
		if c % 4 == 0:
			fore += [i] * (c // 4)
		else:
			can = False
	if can:
		answer = [[0] * n for i in range(n)]
		print('YES')
		current = 0
		for i in range(n // 2):
			for j in range(n // 2):
				answer[i][j] = fore[-1]
				answer[n - i - 1][j] = fore[-1]
				answer[i][n - j - 1] = fore[-1]
				answer[n - i - 1][n - j - 1] = fore[-1]
				fore = fore[:-1]
		for line in answer:
			print(' '.join(list(map(str, line))))
	else:
		print('NO')
