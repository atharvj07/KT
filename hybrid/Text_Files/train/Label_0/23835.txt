#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main(){
  int N, M;
  cin >> N >> M;
  vector< pair<int,int> > V;
  for(int i = 0; i < M; ++i){
    int a, b;
    cin >> a >> b;
    V.emplace_back(b,a);
  }
  sort(V.begin(), V.end());
  int ans = 0, l = -1;
  for(int i = 0; i < M; ++i){
    if(V[i].second >= l){
      ++ans;
      l = V[i].first;
    }
  }
  cout << ans << endl;
  return 0;
}
