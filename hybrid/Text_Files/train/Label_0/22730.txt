#include <bits/stdc++.h>
using namespace std;
int mat[15][15];
int r, c, k, a, b, n;
int cnt(int a1, int a2, int b1, int b2) {
  int res = 0;
  for (int(i) = (a1); (i) < int(a2 + 1); ++(i)) {
    for (int(j) = (b1); (j) < int(b2 + 1); ++(j)) {
      res += mat[i][j];
    }
  }
  return res;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  ;
  cin >> r >> c >> n >> k;
  for (int(i) = (0); (i) < int(n); ++(i)) {
    cin >> a >> b;
    mat[a - 1][b - 1] = 1;
  }
  long long ans = 0;
  for (int(a1) = (0); (a1) < int(r); ++(a1)) {
    for (int(b1) = (0); (b1) < int(c); ++(b1)) {
      for (int(a2) = (a1); (a2) < int(r); ++(a2)) {
        for (int(b2) = (b1); (b2) < int(c); ++(b2)) {
          int h = cnt(a1, a2, b1, b2);
          if (h >= k) ans += 1ll;
        }
      }
    }
  }
  cout << ans << "\n";
  cin.sync();
  cin.get();
  return 0;
}
