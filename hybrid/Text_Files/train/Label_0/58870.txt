import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class OrderingPizza {
    
    void solve() {
        int n = in.nextInt(), m = in.nextInt();
        int[] s = new int[n], a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = in.nextInt();
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        
        long ans = 0;
        long[] cnt = new long[3];
        List<Integer>[] grp = new ArrayList[3];
        for (int i = 0; i < 3; i++) grp[i] = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (a[i] == b[i]) {
                grp[0].add(i);
                cnt[0] += s[i];
            } else if (a[i] > b[i]) {
                grp[1].add(i);
                cnt[1] += s[i];
            } else {
                grp[2].add(i);
                cnt[2] += s[i];
            }
        }
        
        long r1 = cnt[1] % m, r2 = cnt[2] % m;
        if (r1 > 0 && r2 > 0 && r1 + r2 + cnt[0] <= m) {
            grp[1].sort(Comparator.comparingInt(i -> a[i] - b[i]));
            grp[2].sort(Comparator.comparingInt(i -> b[i] - a[i]));
            long sum1 = 0, sum2 = 0;
            for (int i = 0; i < grp[1].size() && r1 > 0; i++) {
                int x = grp[1].get(i);
                long delta = Math.min(r1, s[x]);
                sum1 += (long) a[x] * delta;
                sum2 += (long) b[x] * delta;
                r1 -= delta;
                s[x] -= delta;
            }
            for (int i = 0; i < grp[2].size() && r2 > 0; i++) {
                int x = grp[2].get(i);
                long delta = Math.min(r2, s[x]);
                sum1 += (long) a[x] * delta;
                sum2 += (long) b[x] * delta;
                r2 -= delta;
                s[x] -= delta;
            }
            ans += Math.max(sum1, sum2);
        }
        for (int i = 0; i < n; i++) ans += (long) Math.max(a[i], b[i]) * s[i];
        out.println(ans);
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new OrderingPizza().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
