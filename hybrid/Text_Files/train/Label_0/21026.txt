#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=300;
double dp[N][N][N];
int n;
double f(int a,int b,int c){
 if(a==n) return 0;
 if(dp[a][b][c]!=-1) return dp[a][b][c];
 double ans=1;
 if(b>0) ans+=b*f(a+1,b-1,c)/n;
 if(c>0) ans+=c*f(a,b+1,c-1)/n;
 if(n>a+b+c) ans+=(n-(a+b+c))*f(a,b,c+1)/n;
 ans=ans*n/(n-a);
 return dp[a][b][c]=ans;
}
int main(){
 for(int i=0;i<N;i++) for(int j=0;j<N;j++) for(int k=0;k<N;k++) dp[i][j][k]=-1;
 cin>>n;
 int a=0,b=0,c=0;
 for(int i=0;i<n;i++) {
  int x;
  cin>>x;
  if(x==1) b++;
  if(x==2) c++;
 }
 cout.precision(15);
 cout<<fixed;
 cout<<f(a,b,c)<<endl;
 return 0;
}
