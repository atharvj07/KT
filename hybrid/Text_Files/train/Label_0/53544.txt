import java.io.*;
import java.util.*;

public class C
{
	private final static int oo = 1000000000;
	public static void main(String[] args) throws IOException
	{
		BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter writer = new PrintWriter(new OutputStreamWriter(System.out));

		StringTokenizer tokenizer = new StringTokenizer(reader.readLine());

		int n = Integer.parseInt(tokenizer.nextToken());
		int m = Integer.parseInt(tokenizer.nextToken());
		int k = Integer.parseInt(tokenizer.nextToken());
		int w = Integer.parseInt(tokenizer.nextToken());

		char [][][] maps = new char[k][n][m];
		for(int i = 0 ; i < k ; i++)
		{
			for(int j = 0 ; j < n ; j++)
				maps[i][j] = reader.readLine().toCharArray();
		}

		ArrayList<Edge> [] diff = new ArrayList[k];
		for(int i = 0 ; i < k ; i++)
			diff[i] = new ArrayList<Edge>();

		for(int i = 0  ; i < k ; i++)
		{

			for(int j = i ; j < k ; j++)
			{
				int calc = 0;
				for(int r = 0 ; r < n ; r++)
					for(int c = 0 ; c < m ; c++)
						if(maps[i][r][c] != maps[j][r][c])
							calc++;

				calc *= w;
				if(calc < n*m)
				{
					diff[i].add(new Edge(j, calc));
					diff[j].add(new Edge(i, calc));
				}
			}
		}

		int res = 0;
		ArrayList<Integer> pairsX = new ArrayList<Integer>();
		ArrayList<Integer> pairsY = new ArrayList<Integer>();

		PriorityQueue<Node> queue = new PriorityQueue<Node>();
		boolean [] visited = new boolean[k];
		while(true)
		{
			int idx = -1;
			for(int i = 0 ; i < k ; i++)
				if(!visited[i])
				{
					idx = i;
					break;
				}
			if(idx == -1)
				break;

			res += n*m;
			pairsX.add(idx+1);
			pairsY.add(0);

			queue.clear();
			queue.offer(new Node(idx, 0, -1));

			while(!queue.isEmpty())
			{
				Node where = queue.poll();
				int at = where.at;
				int cost = where.cost;
				int prev = where.prev;

				if(visited[at])
					continue;
				visited[at] = true;
				res += cost;
				if(at != idx)
				{
					pairsX.add(at + 1);
					pairsY.add(prev + 1);
				}

				for(Edge to : diff[at])
					queue.offer(new Node(to.to, to.cost, at));
			}
		}

		writer.println(res);
		for(int i = 0 ; i < k ; i++)
			writer.println(pairsX.get(i)+" "+pairsY.get(i));

		writer.flush();
		writer.close();
	}

	static class Node implements Comparable<Node>
	{
		int at, cost, prev;
		public Node(int at, int cost, int prev)
		{
			this.at = at;
			this.cost = cost;
			this.prev = prev;
		}

		@Override
		public int compareTo(Node node)
		{
			return cost - node.cost;
		}
	}

	static class Edge
	{
		int to, cost;

		public Edge(int to, int cost)
		{
			this.to = to;
			this.cost = cost;
		}
	}
}
