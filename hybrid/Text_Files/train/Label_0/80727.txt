import sys
from math import floor

def binSearch(target):
	low = 0
	up = len(maxs) - 1
	mid = 0
	found = -1
	while low <= up:
		mid = floor((low + up) / 2)
		if maxs[mid] < target:
			low = mid + 1
		elif maxs[mid] > target:
			up = mid - 1
		else:
			found = mid
			low = mid + 1
	return low
numberOfArrays = int(input())
mins = []
maxs = []
for index in range(0, numberOfArrays):
	inp = input().split(' ')
	curMin = sys.maxsize
	curMax = -1
	for i in range(1, int(inp[0]) + 1):
		curMin = min(int(inp[i]), curMin)
		if curMin < int(inp[i]):
			curMin = -1
			curMax = sys.maxsize
			break
		curMax = max(int(inp[i]), curMax)
	mins.append(curMin)
	maxs.append(curMax)
maxs.sort()
count = 0
for i in range(0, numberOfArrays):
	count += len(maxs) - binSearch(mins[i])
print(count)
