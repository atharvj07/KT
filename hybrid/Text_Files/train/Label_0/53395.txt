import java.io.*;
import java.util.*;

public class CF1504F extends PrintWriter {
	CF1504F() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1504F o = new CF1504F(); o.main(); o.flush();
	}

	int[] dsu;
	int find(int i) {
		return dsu[i] < 0 ? i : (dsu[i] = find(dsu[i]));
	}
	void join(int i, int j) {
		i = find(i);
		j = find(j);
		if (i == j)
			return;
		if (dsu[i] > dsu[j])
			dsu[i] = j;
		else {
			if (dsu[i] == dsu[j])
				dsu[i]--;
			dsu[j] = i;
		}
	}
	void join2(int i, int j) {
		join(i << 1, j << 1 | 1);
		join(i << 1 | 1, j << 1);
	}
	void main() {
		int n = sc.nextInt();
		int[] aa = new int[n];
		int[] bb = new int[n];
		int[] ii = new int[n * 2 + 1];
		int lmax = 0, rmin = n * 2 + 1;
		for (int i = 0; i < n; i++) {
			int a = sc.nextInt();
			int b = sc.nextInt();
			aa[i] = a;
			bb[i] = b;
			ii[a] = ii[b] = i;
			lmax = Math.max(lmax, Math.min(a, b));
			rmin = Math.min(rmin, Math.max(a, b));
		}
		if (rmin < lmax) {
			println(-1);
			return;
		}
		int[] rr = new int[n]; Arrays.fill(rr, n * 2 + 1);
		dsu = new int[n << 1]; Arrays.fill(dsu, -1);
		int[] qu = new int[n]; int cnt = 0;
		for (int l = 1; l <= n; l++) {
			int i = ii[l], r = Math.max(aa[i], bb[i]);
			if (cnt > 0 && qu[cnt - 1] < r) {
				int r_ = qu[cnt - 1];
				while (cnt > 0 && qu[cnt - 1] < r) {
					int j = ii[qu[--cnt]];
					if (rr[j] > r)
						rr[j] = r;
					else {	// triple crosses
						println(-1);
						return;
					}
					join2(i, j);
				}
				r = r_;
			}
			qu[cnt++] = r;
		}
		for (int i = 0; i < n; i++)
			if (find(i << 1) == find(i << 1 | 1)) {
				println(-1);
				return;
			}
		int[] k0 = new int[n << 1];
		int[] k1 = new int[n << 1];
		for (int i = 0; i < n << 1; i++) {
			int j = find(i);
			int x = (i ^ j) & 1;
			int y = (aa[i >> 1] < bb[i >> 1]) == (aa[j >> 1] < bb[j >> 1]) ? 0 : 1;
			if (x == y)
				k0[j]++;
			else
				k1[j]++;
		}
		int ans = 0;
		for (int i = 0; i < n << 1; i++)
			ans += Math.min(k0[i], k1[i]);
		println(ans >> 1);
	}
}
