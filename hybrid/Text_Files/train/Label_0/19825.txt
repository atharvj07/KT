// D - Lazy Faith
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vl = vector<ll>;
#define lower(v,x) lower_bound((v).begin(),(v).end(), (x))
#define INF (10000000000*2)

int main(){
	int A,B,Q; cin>>A>>B>>Q;

	vl S(A+2), T(B+2);
	for(int i=1; i<=A; ++i) cin>>S[i]; S[0] = -INF; S[A+1] = INF;
	for(int i=1; i<=B; ++i) cin>>T[i]; T[0] = -INF; T[B+1] = INF;

	for(int i=0; i<Q; ++i){
		ll x; cin>>x;

		ll mini = INF;
		vl::iterator apos, bpos;

		apos = lower(S, x);
		for(int j=0; j<2; ++j){
			ll ax = *(apos - j);
			bpos = lower(T, ax);
			for(int k=0; k<2; ++k){
				ll bx = *(bpos - k);
				mini = min(mini, abs(x-ax) + abs(ax-bx));
			}
		}

		apos = lower(T, x);
		for(int j=0; j<2; ++j){
			ll ax = *(apos - j);
			bpos = lower(S, ax);
			for(int k=0; k<2; ++k){
				ll bx = *(bpos - k);
				mini = min(mini, abs(x-ax) + abs(ax-bx));
			}
		}

		cout<< mini <<endl;

	}

}
