#include <bits/stdc++.h>
using namespace std;

#define repl(i,a,b) for(int i=(int)(a);i<(int)(b);i++)
#define rep(i,n) repl(i,0,n)
#define dbg(x) cout<<#x<<"="<<x<<endl
#define all(x) (x).begin(),(x).end()

typedef pair<int,int> P;
typedef pair<P,int> PI;
#define fi first
#define se second

#define INF INT_MAX/3
#define MAX_V 300

struct edge{
  int to,cap,cost,rev;
};

int V;
vector<edge> G[MAX_V];
int h[MAX_V];
int dist[MAX_V];
int prevv[MAX_V],preve[MAX_V];

void init(int v){
  V=v;
  for(int i=0;i<MAX_V;i++){
    G[i].clear();
  }
}

void add_edge(int from,int to,int cap,int cost){
  //printf("%d -> %d : %d\n", from,to,cap);
  G[from].push_back((edge){to,cap,cost,(int)G[to].size()});
  G[to].push_back((edge){from,0,-cost,(int)G[from].size()-1});
}

int min_cost_flow(int s,int t,int f){
  int res=0;
  memset(h,0,sizeof(h));
  while(f>0){
    fill(dist,dist+V,INF);
    dist[s]=0;
    bool update=true;
    while(update){
      update=false;
      for(int v=0;v<V;v++){
        if(dist[v]==INF)continue;
        for(int i=0;i<(int)G[v].size();i++){
          edge &e=G[v][i];
          if(e.cap>0&&dist[e.to]>dist[v]+e.cost){
            dist[e.to]=dist[v]+e.cost;
            prevv[e.to]=v;
            preve[e.to]=i;
            update=true;
          }
        }
      }
    }
    if(dist[t]==INF)return INF;

    int d=f;
    for(int v=t;v!=s;v=prevv[v]){
      d=min(d,G[prevv[v]][preve[v]].cap);
    }
    f-=d;
    res+=d*dist[t];
    for(int v=t;v!=s;v=prevv[v]){
      edge &e=G[prevv[v]][preve[v]];
      e.cap-=d;
      G[v][e.rev].cap+=d;
    }
  }
  return res;
}

int N,M;
int g[2][303][303];

int main(){
  cin>>N>>M;
  rep(x,2)rep(y,2){
    int L;
    cin>>L;
    rep(i,L){
      int a,b;
      cin>>a>>b;
      if(x==0) g[y][a][b]++;
      else g[y][b][a]++;
    }
  }

  int source=N+M,sink=N+M+1;

  PI res=PI(P(0,0),0);
  repl(f,1,min(N,M)+1){
    init(N+M+2);

    rep(i,N)add_edge(source,i,1,0);
    rep(i,M)add_edge(N+i,sink,1,0);

    rep(i,N)rep(j,M){
      if(g[0][i][j]==2){
        add_edge(i,N+j,1,-30000);
      }else if(g[0][i][j]==1&&g[1][i][j]==1){
        add_edge(i,N+j,1,-200);
      }else if(g[1][i][j]==2){
        add_edge(i,N+j,1,-1);
      }
    }

    int m=-min_cost_flow(source,sink,f);
    int resa=m/30000;
    int resb=(m-resa*30000)/200;
    int resc=(m-resa*30000-resb*200);
    res=max(res,PI(P(resa,resb),resc));
  }

  cout<<res.fi.fi<<" "<<res.fi.se<<" "<<res.se<<endl;
}
