(n, k) = map(int, input().split())
a = list(map(int, input().split()))
ind = [n + 1]
su = 0
for i in range(k):
	ma = max(a)
	su += ma
	index = a.index(ma)
	ind.append(index + 1)
	a[index] = 0
ind.sort()
print(su)
if k == 1:
	print(n)
	exit()
print(ind[0], end=' ')
for i in range(1, k):
	if i == 1:
		print(ind[2] - ind[0] - 1, end=' ')
	else:
		print(ind[i + 1] - ind[i], end=' ')
