for t in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a = sorted(a)
	if n == 1:
		ans = 0
	else:
		mini = a[0] + a[1]
		maxi = a[n - 2] + a[n - 1]
		ans = 0
		for k in range(mini, maxi + 1):
			count = 0
			i = 0
			j = n - 1
			while i < j:
				if a[i] + a[j] == k:
					count += 1
					i += 1
					j -= 1
				elif a[i] + a[j] > k:
					j -= 1
				else:
					i += 1
			ans = max(ans, count)
	print(ans)
