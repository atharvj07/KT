def fact(a):
	if a <= 1:
		return 1
	result = 1
	for i in range(1, a + 1):
		result = result * i
	return result

def comb(a, b):
	return fact(a) // fact(b) // fact(a - b)

def pairs(a):
	return a * (a - 1) // 2
(n, m) = map(int, input().split())
print(str(n % m * pairs(n // m + 1) + (m - n % m) * pairs(n // m)) + ' ' + str(pairs(n - m + 1)))
