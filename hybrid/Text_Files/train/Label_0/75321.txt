from sys import stdin, stdout
nmbr = lambda : int(stdin.readline())
lst = lambda : list(map(int, stdin.readline().split()))
for _ in range(1):
	n = nmbr()
	a = lst()
	sm = 0
	looking_l = [0] * (1 + n)
	for i in range(n - 1, -1, -1):
		if a[i] == 0:
			looking_l[i] = 1 + looking_l[i + 1]
		else:
			looking_l[i] = looking_l[i + 1]
	for i in range(n):
		if a[i] == 1:
			sm += looking_l[i]
	print(sm)
