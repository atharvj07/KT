(n, m, ta, tb, k) = map(int, input().split())
ar = [int(x) + ta for x in input().split()]
arr = [int(x) for x in input().split()]
r = [m] * n
aptr = 0
bptr = 0
while aptr < n and bptr < m:
	if arr[bptr] >= ar[aptr]:
		r[aptr] = bptr
		aptr += 1
	else:
		bptr += 1
if k >= n:
	print(-1)
else:
	ans = 0
	for p in range(0, k + 1):
		if r[p] + k - p >= m:
			ans = -1
			break
		else:
			ans = max(arr[r[p] + k - p] + tb, ans)
	print(ans)
