#include <bits/stdc++.h>
using namespace std;
const int M = 100005;
int read() {
  int x = 0, flag = 1;
  char c;
  while ((c = getchar()) < '0' || c > '9')
    if (c == '-') flag = -1;
  while (c >= '0' && c <= '9')
    x = (x << 3) + (x << 1) + (c ^ 48), c = getchar();
  return x * flag;
}
int n, m, ans, tot, f[M], deg[M], c[M];
queue<int> q[2];
struct edge {
  int v, next;
  edge(int V = 0, int N = 0) : v(V), next(N) {}
} e[M];
int main() {
  n = read();
  m = read();
  for (int i = 1; i <= n; i++) c[i] = read();
  for (int i = 1; i <= m; i++) {
    int u = read() + 1, v = read() + 1;
    e[++tot] = edge(u, f[v]), f[v] = tot;
    deg[u]++;
  }
  for (int i = 1; i <= n; i++)
    if (!deg[i]) q[c[i]].push(i);
  while (!q[0].empty() || !q[1].empty()) {
    while (!q[0].empty()) {
      int u = q[0].front();
      q[0].pop();
      for (int i = f[u]; i; i = e[i].next) {
        int v = e[i].v;
        deg[v]--;
        if (deg[v] == 0) q[c[v]].push(v);
      }
    }
    if (!q[1].empty()) {
      ans++;
      while (!q[1].empty()) {
        int u = q[1].front();
        q[1].pop();
        for (int i = f[u]; i; i = e[i].next) {
          int v = e[i].v;
          deg[v]--;
          if (deg[v] == 0) q[c[v]].push(v);
        }
      }
    }
  }
  printf("%d\n", ans);
}
