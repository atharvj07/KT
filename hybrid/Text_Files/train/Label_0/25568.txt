#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-6;
const double PI = acos(-1.0);
double sqr(double x) { return x * x; }
struct Point {
  double x, y;
  Point(double _x, double _y) : x(_x), y(_y) {}
  Point() {}
  Point operator+(const Point& r) const { return Point(x + r.x, y + r.y); }
  Point operator-(const Point& r) const { return Point(x - r.x, y - r.y); }
  Point operator*(double r) const { return Point(x * r, y * r); }
  Point operator/(double r) const { return Point(x / r, y / r); }
  Point spin(double ang) const {
    return Point(x * cos(ang) - y * sin(ang), x * sin(ang) + y * cos(ang));
  }
  double len() { return sqrt(x * x + y * y); }
  double len2() { return x * x + y * y; }
  double det(const Point& r) const { return x * r.y - r.x * y; }
  double dot(const Point& r) const { return x * r.x + y * r.y; }
  void scan() { scanf("%lf%lf", &x, &y); }
};
struct Line {
  Point a, b;
  Line() {}
  Line(Point _a, Point _b) : a(_a), b(_b) {}
  Point getIntersection(const Line& r) const {
    double s1 = (b - a).det(r.a - a), s2 = (b - a).det(r.b - a);
    if (abs(s1 - s2) < eps) throw "";
    return Point(r.b * s1 - r.a * s2) / (s1 - s2);
  }
  void scan() { a.scan(), b.scan(); }
};
struct Convex {
  vector<Point> con;
  Convex() {}
  Convex(const vector<Point>& points) {}
  Convex cut(Line l) {
    Convex newconvex;
    vector<Point>& newcon = newconvex.con;
    for (int i = 0; i < con.size(); i++) {
      if ((l.b - l.a).det(con[i] - l.a) > -eps) newcon.push_back(con[i]);
      if ((l.b - l.a).det(con[i] - l.a) *
              (l.b - l.a).det(con[(i + 1) % con.size()] - l.a) <
          -eps) {
        newcon.push_back(
            l.getIntersection(Line(con[i], con[(i + 1) % con.size()])));
      }
    }
    return newconvex;
  }
  double area() {
    double ret = 0;
    for (int i = 0; i < con.size(); i++) {
      ret += con[i].det(con[(i + 1) % con.size()]);
    }
    return abs(ret) / 2;
  }
};
void print(Point x) { printf("%f %f\n", x.x, x.y); }
int main() {
  double w, h, ar;
  cin >> w >> h >> ar;
  w /= 2;
  h /= 2;
  ar = ar / 180.0 * PI;
  Point a(-w, -h), b(w, -h), c(w, h), d(-w, h);
  Convex ori;
  ori.con.push_back(a);
  ori.con.push_back(b);
  ori.con.push_back(c);
  ori.con.push_back(d);
  Convex n = ori;
  for (int i = 0; i < 4; i++) {
    n.con[i] = n.con[i].spin(ar);
  }
  for (int i = 0; i < 4; i++) {
    ori = ori.cut(Line(n.con[i], n.con[(i + 1) % 4]));
  }
  printf("%.15lf\n", ori.area());
  return 0;
}
