#include <bits/stdc++.h>
using namespace std;
const double PI = 4 * atan(1.0);
void fast_stream() { std::ios_base::sync_with_stdio(0); }
pair<int, int> points[100001];
vector<int> hor[100001];
vector<int> ver[100001];
vector<int> rightDown[200001];
vector<int> leftDown[200001];
const int base = 100000;
int d[100001];
void solve() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> y >> x;
    hor[y].push_back(x);
    ver[x].push_back(y);
    points[i].first = y;
    points[i].second = x;
    int p = x + y;
    int p2 = x - y + base;
    rightDown[p].push_back(y);
    leftDown[p2].push_back(y);
  }
  for (int i = 1; i <= n; i++) {
    sort(hor[i].begin(), hor[i].end());
    sort(ver[i].begin(), ver[i].end());
  }
  for (int i = 0; i <= 200000; i++) {
    sort(leftDown[i].begin(), leftDown[i].end());
    sort(rightDown[i].begin(), rightDown[i].end());
  }
  for (int i = 0; i < m; i++) {
    int x = points[i].second;
    int y = points[i].first;
    int p1 = x + y;
    int p2 = x - y + base;
    int cnt = 0;
    if (hor[y].size() != 1) {
      int a = lower_bound(hor[y].begin(), hor[y].end(), x) - hor[y].begin();
      if (a != 0) cnt++;
      if (a != hor[y].size() - 1) cnt++;
    }
    if (ver[x].size() != 1) {
      int a = lower_bound(ver[x].begin(), ver[x].end(), y) - ver[x].begin();
      if (a != 0) cnt++;
      if (a != ver[x].size() - 1) cnt++;
    }
    if (rightDown[p1].size() != 1) {
      int a = lower_bound(rightDown[p1].begin(), rightDown[p1].end(), y) -
              rightDown[p1].begin();
      if (a != 0) cnt++;
      if (a != rightDown[p1].size() - 1) cnt++;
    }
    if (leftDown[p2].size() != 1) {
      int a = lower_bound(leftDown[p2].begin(), leftDown[p2].end(), y) -
              leftDown[p2].begin();
      if (a != 0) cnt++;
      if (a != leftDown[p2].size() - 1) cnt++;
    }
    d[cnt]++;
  }
  for (int i = 0; i <= 8; i++) {
    cout << d[i];
    if (i != 8)
      cout << " ";
    else
      cout << endl;
  }
}
int main() {
  solve();
  return 0;
}
