N = int(input())
A = list((int(i) for i in input().split()))
B = list(A)
B.sort()
missing = set()
double = dict()
expected = 1
for i in range(N):
	if B[i] in double:
		double[B[i]] += 1
	else:
		double[B[i]] = 1
	missing.add(i + 1)
for i in range(N):
	if i + 1 in double and double[i + 1] == 1:
		del double[i + 1]
	if i + 1 in double:
		double[i + 1] -= 1
	if B[i] in missing:
		missing.remove(B[i])
for i in range(N):
	if A[i] in double:
		new = missing.pop()
		old = A[i]
		A[i] = new
		double[old] -= 1
		if double[old] <= 0:
			del double[old]
for i in range(N):
	if i > 0:
		print(' ', end='')
	print(A[i], end='')
print()
