from collections import Counter
from math import ceil, floor, log, gcd
import bisect as bs
import sys
input = sys.stdin.readline
inp_lis = lambda : list(map(int, input().split()))
inp_multi = lambda : map(int, input().split())
inp_int = lambda : int(input().strip())
for _ in range(int(input().strip())):
	n = inp_int()
	p = inp_lis()
	s = input().strip()
	o = {}
	z = {}
	for i in range(n):
		if s[i] == '0':
			z[p[i]] = i
		else:
			o[p[i]] = i
	t = len(z)
	for each in sorted(z.keys(), reverse=True):
		p[z[each]] = t
		t -= 1
	t = len(z) + 1
	for each in sorted(o.keys()):
		p[o[each]] = t
		t += 1
	print(*p)
