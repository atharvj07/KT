from math import ceil
from collections import defaultdict
for q in range(int(input())):
	s = []
	n = int(input())
	for i in range(n):
		s.append(list(input()))
	seen = [0] * 10
	M = 0
	for i in range(n - 1):
		for j in range(i + 1, n):
			if s[i] == s[j]:
				for k in range(n):
					seen[int(s[k][0])] = 1
				for k in range(10):
					if seen[k] == 0:
						s[i][0] = str(k)
						seen[k] = 1
						M += 1
						break
	print(M)
	for i in range(n):
		print(''.join(s[i]))
