def util(arr, n, last):
	output = [-1] * n
	vstd = [False] * (n + 1)
	vstd[arr[0]] = True
	output[0] = arr[0]
	deq = collections.deque([x for x in range(1, arr[0])])
	maxi = arr[0] - 1
	for i in range(1, n):
		if arr[i] > arr[i - 1]:
			output[i] = arr[i]
			vstd[arr[i]] = True
			while maxi + 1 < arr[i]:
				maxi += 1
				if not vstd[maxi]:
					deq.append(maxi)
		else:
			output[i] = deq.pop() if last else deq.popleft()
			vstd[output[i]] = True
	return output
