#include <iostream>
#include <string.h>
#include <vector>

using namespace std;

#define MAX 114

vector<int> G[MAX];
int visited[MAX];
bool found;

void dfs(int v)
{
    visited[v] = 0;
    for(int i = 0 ; i < (int)G[v].size() ; i++){
	int to = G[v][i];
	if(visited[to] == -1){
	    dfs(to);
	}else if(visited[to] == 0){
            found = true;            
	    return;
	}
    }
    visited[v] = 1;
}
    
int main()
{
    int M, N;
    cin >> M >> N;
    
    auto canGraph = [=](){
        int lb[MAX][MAX], ub[MAX][MAX];
        for(int i = 0 ; i < MAX ; i++){
            for(int j = 0 ; j < MAX ; j++){
                lb[i][j] = -1;
                ub[i][j] = 101;
            }
        }
        int K, s, t;
        string cond;
        bool possible = 1;
        for(int i = 0 ; i < M ; i++){
            cin >> K;
            for(int j = 0 ; j < K ; j++){
                cin >> s >> cond >> t;
                --s;
                if(cond == ">="){
                    lb[i][s] = max(lb[i][s], t);
                }else{
                    ub[i][s] = min(ub[i][s], t);
                }
            }
            for(int j = 0 ; j < N ; j++){
                if(lb[i][j] > ub[i][j]){
                    possible = 0;
                    break;
                }
            }
        }

        if(!possible) return false;

        auto isConnect = [=](int from, int to){
            for(int i = 0 ; i < N ; i++){
                if(ub[from][i] < lb[to][i]){
                    return true;
                }
            }
            return false;
        };
        
        for(int from = 0 ; from < M ; from++){
            for(int to = 0 ; to < M ; to++){
                if(from == to) continue;
                if(isConnect(from, to)){
                    G[from].push_back(to);
                }
            }
        }
        found = false;
        memset(visited, -1, sizeof(visited));
        for(int i = 0 ; i < M ; i++){
            if(visited[i] == -1){
                dfs(i);
            }
        }                                       
        return !found;
    };
    
    cout << (canGraph() ? "Yes" : "No") << endl;
    return 0;
}
