cases = int(input())
for _ in range(cases):
	char_input = input()
	if int(char_input) == 0:
		print('red')
	else:
		has_zero = False
		has_even = False
		current_sum = 0
		for each_number in char_input:
			int_number = int(each_number)
			if int_number == 0:
				if has_zero == False:
					has_zero = True
				else:
					has_even = True
			else:
				if int_number % 2 == 0:
					has_even = True
				current_sum += int_number
		if current_sum % 3 == 0 and has_even and has_zero:
			print('red')
		else:
			print('cyan')
