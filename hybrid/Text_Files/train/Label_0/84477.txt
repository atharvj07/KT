k = int(input())
s = input().strip()
t = input().strip()
num_2 = list(map(ord, s))
num_1 = list(map(ord, t))
for i in range(k):
	num_1[i] -= 97
	num_2[i] -= 97

def plus(num_1, num_2):
	res = [0] * (len(num_1) + 1)
	memory = 0
	num_1.reverse()
	num_2.reverse()
	for i in range(len(num_1)):
		add = num_1[i] + num_2[i] + memory
		res[i] = add % 26
		memory = add // 26
	res.reverse()
	res[0] = memory
	return res

def div_2(num):
	memory = 0
	res = [0] * len(num)
	for i in range(len(num)):
		res[i] = (num[i] + memory * 26) // 2
		memory = (num[i] + memory * 26) % 2
	return res
summ = plus(num_1, num_2)
res = div_2(summ)
print(''.join((chr(i + 97) for i in res[1:])))
