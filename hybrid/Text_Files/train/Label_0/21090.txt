#include <bits/stdc++.h>
using namespace std;
const int oo = 0x3f3f3f3f;
const int maxn = 100000, maxm = 100000, maxk = 10000;
int n, m, K;
long long pos[maxn + 5];
int val[maxn + 5];
vector<int> all[maxm + 5];
pair<int, long long> yyt[maxm + 5];
bool be[maxn + 5];
bool used[maxn + 5];
int gcd(int first, int second) {
  while (second) second ^= first ^= second ^= first %= second;
  return first;
}
vector<pair<int, long long> > e[maxk + 5];
long long dis[maxk + 5];
void dij(const int &S) {
  static const int max0 = 1e6;
  static pair<long long, int> q[max0 + 5];
  static bool vis[maxk + 5];
  memset(vis, 0, sizeof vis);
  int tot = 0;
  memset(dis, oo, sizeof dis);
  dis[S] = 0;
  q[tot++] = make_pair(dis[S], S);
  while (tot) {
    int first = q[0].second;
    pop_heap(q, q + (tot--), greater<pair<long long, int> >());
    if (vis[first]) continue;
    vis[first] = 1;
    for (int i = (0), _end_ = ((int((e[first]).size()))); i != _end_; ++i) {
      const int &second = e[first][i].first;
      const long long &z = e[first][i].second;
      if (dis[second] > dis[first] + z ? dis[second] = dis[first] + z, 1 : 0) {
        q[tot++] = make_pair(dis[second], second);
        push_heap(q, q + tot, greater<pair<long long, int> >());
      }
    }
  }
}
int main() {
  scanf("%*I64d%d%d%d", &n, &m, &K);
  for (int i = (0), _end_ = (n); i != _end_; ++i)
    scanf("%I64d%d", pos + i, val + i), --pos[i];
  for (int i = (0), _end_ = (m); i != _end_; ++i) {
    static int ty;
    scanf("%d", &ty);
    if (ty == 1) {
      scanf("%I64d", &yyt[i].second);
      yyt[i].first = -2;
    } else if (ty == 2)
      scanf("%d%I64d", &yyt[i].first, &yyt[i].second), --yyt[i].first;
    else
      yyt[i].first = -1;
  }
  static pair<int, int> q[maxn + maxm + 5];
  int tot = 0;
  for (int i = (0), _end_ = (n); i != _end_; ++i)
    if (!(pos[i] % K)) be[i] = 1, q[tot++] = make_pair(val[i], -i);
  make_heap(q, q + tot);
  for (int i = (0), _end_ = (m); i != _end_; ++i) {
    if (yyt[i].first == -1) {
      while (tot) {
        if (!used[-q[0].second] && val[-q[0].second] == q[0].first) break;
        pop_heap(q, q + (tot--));
      }
      if (!tot)
        printf("0\n");
      else
        used[-q[0].second] = 1, printf("%d\n", q[0].first),
        pop_heap(q, q + (tot--));
    } else if (yyt[i].first >= 0) {
      val[yyt[i].first] -= yyt[i].second;
      if (be[yyt[i].first])
        q[tot++] = make_pair(val[yyt[i].first], -yyt[i].first),
        push_heap(q, q + tot);
    } else {
      for (int j = (0), _end_ = (K); j != _end_; ++j)
        e[j].push_back(
            make_pair((j + yyt[i].second) % K, (j + yyt[i].second) / K));
      dij(0);
      for (int i = (0), _end_ = (n); i != _end_; ++i)
        if (!be[i]) {
          long long k = pos[i] / K;
          int b = pos[i] % K;
          if (dis[b] <= k)
            be[i] = 1, q[tot++] = make_pair(val[i], -i), push_heap(q, q + tot);
        }
    }
  }
  return 0;
}
