import sys
r = sys.stdin.readlines()
(n, q) = r[0].split(' ')
n = int(n)
q = int(q)
M = 10 ** 9 + 7
twopow = [1] * (2 * n + 1)
for j in range(1, 2 * n + 1):
	twopow[j] = twopow[j - 1] * 2 % M
s = r[1]
p = [0] * (n + 1)
for v in range(n):
	p[v + 1] = p[v] + (s[v] == '1')
ans = [0] * q
for k in range(q):
	(a, b) = r[k + 2].split(' ')
	a = int(a)
	b = int(b)
	l = b - a + 1
	one = p[b] - p[a - 1]
	v = (twopow[l] - twopow[l - one] + M) % M
	ans[k] = str(v)
sys.stdout.write('\n'.join(ans))
