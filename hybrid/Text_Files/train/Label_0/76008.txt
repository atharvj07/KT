mod = 10 ** 9 + 7
MAX = 2 * 10 ** 5 + 2
(n, T) = list(map(int, input().split()))
t = [0]
t += list(map(int, input().split()))
t += [MAX]
r = [1] * MAX
factorial = [1] * MAX
rfactorial = [1] * MAX
rp = [1] * MAX
sim = 0
sim_n = 0
sim_k = 0
E = 0
S = [t[0]] * len(t)
for i in range(1, len(t)):
	S[i] = S[i - 1] + t[i]
for i in range(2, MAX):
	factorial[i] = i * factorial[i - 1] % mod
	r[i] = mod - mod // i * r[mod % i] % mod
	rfactorial[i] = rfactorial[i - 1] * r[i] % mod
for i in range(1, MAX):
	rp[i] = rp[i - 1] * (mod + 1) // 2 % mod

def Combination(n, k):
	if n < k:
		return 0
	return factorial[n] * rfactorial[k] * rfactorial[n - k]

def simC(x):
	aux = 0
	mi = min(x, T - S[x])
	for i in range(0, mi + 1):
		aux += Combination(x, i) % mod
	return (x, mi, aux)

	aux_sim_n = sim_n + 1
	aux_sim_k = sim_k + 1
	while aux_sim_k > mi1:
		aux_sim = (aux_sim - Combination(aux_sim_n, aux_sim_k)) % mod
		aux_sim_k -= 1
	return (aux_sim_n, aux_sim_k, aux_sim)
i = 1
while i <= n and S[i] + i <= T:
	i += 1
	E += 1
if i <= n:
	(sim_n, sim_k, sim) = simC(i)
	for x in range(i, n + 1):
		E = (E + rp[x] * sim) % mod
		if T - S[x + 1] < 0:
			break
