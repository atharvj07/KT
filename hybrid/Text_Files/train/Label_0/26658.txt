#include <bits/stdc++.h>
using namespace std;
int gcdEx(int a, int b, int *x, int *y) {
  if (!a) {
    *x = 0;
    *y = 1;
    return b;
  }
  int x1, y1, gcd = gcdEx(b % a, a, &x1, &y1);
  *x = y1 - (b / a) * x1;
  *y = x1;
  return gcd;
}
int modInv(int b, int m) {
  int x, y;
  gcdEx(b, m, &x, &y);
  return (x % m + m) % m;
}
bool perm(int a[], int i, int s, int n) {
  if (i == n) {
    if ((s % 360) == 0)
      return true;
    else
      return false;
  } else {
    return perm(a, i + 1, s + a[i], n) || perm(a, i + 1, s - a[i], n);
  }
}
void code() {
  int n;
  cin >> n;
  string s;
  int a[n];
  int pn[n], nn[n];
  memset(a, 0, sizeof(a));
  memset(pn, 0, sizeof(pn));
  memset(nn, 0, sizeof(nn));
  int cn = 0, cp = 0, c0 = 0;
  for (int i = 0; i < (n); ++i) {
    cin >> s;
    int corp = 1, corn = 1, corc = 0;
    for (int j = 0; j < (s.length()); ++j) {
      if (s[j] == '(')
        a[i]++;
      else
        a[i]--;
      if (a[i] < 0) corp = 0;
    }
    for (int j = s.length() - 1; j >= 0; j--) {
      if (s[j] == '(')
        corc++;
      else
        corc--;
      if (corc > 0) corn = 0;
    }
    if (a[i] > 0 && corp == 1) {
      pn[cp++] = a[i];
    } else if (a[i] < 0 && corn == 1) {
      nn[cn++] = a[i];
    } else if (corp) {
      c0++;
    }
  }
  sort(pn, pn + cp);
  sort(nn, nn + cn, greater<int>());
  int sum = 0;
  int pi = 0, ni = 0;
  while (pi < cp && ni < cn) {
    if (pn[pi] == -nn[ni]) {
      sum++;
      pi++;
      ni++;
    } else if (pn[pi] > -nn[ni])
      ni++;
    else if (pn[pi] < -nn[ni])
      pi++;
  }
  cout << sum + (c0 / 2);
}
int main() {
  cin.sync_with_stdio(0);
  cin.tie(0);
  cin.exceptions(cin.failbit);
  code();
  return 0;
}
