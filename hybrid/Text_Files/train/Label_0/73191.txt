n = map(int, input())
s = input()
t = input()
count = 0
moves = []
m = dict()
for c in s:
	if c in m:
		m[c] = m.get(c) + 1
	else:
		m[c] = 1
for c in t:
	if c in m:
		m[c] = m.get(c) - 1
	else:
		m[c] = -1
for keys in m.keys():
	if m[keys] != 0:
		print(-1)
		exit()
s = list(s)
t = list(t)
for i in range(0, len(s)):
	if s[i] != t[i]:
		pos = i + s[i:].index(t[i])
		if pos == -1:
			print(-1)
			exit()
		for j in range(pos - 1, i - 1, -1):
			c = s[j]
			s[j] = s[j + 1]
			s[j + 1] = c
			moves.append(j + 1)
print(len(moves))
print(' '.join(map(str, moves)))
