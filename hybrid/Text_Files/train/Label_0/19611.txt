#include <bits/stdc++.h>
using namespace std;
const long long MAX = 1000000001;
int32_t main() {
  long long n, m;
  vector<long long> a;
  vector<long long> b;
  vector<long long> c;
  cin >> n >> m;
  long long h;
  for (long long i = 0; i < n + m; i++) {
    cin >> h;
    a.push_back(h);
  }
  for (long long i = 0; i < n + m; i++) {
    cin >> h;
    if (h == 1)
      b.push_back(a[i]);
    else
      c.push_back(a[i]);
  }
  long long ans[m];
  memset(ans, 0, sizeof(ans));
  for (long long i = 0; i < c.size(); i++) {
    long long t = lower_bound(b.begin(), b.end(), c[i]) - b.begin();
    if (t == 0) {
      ans[t]++;
    } else if (t == b.size()) {
      t--;
      ans[t]++;
    } else {
      if (abs(c[i] - b[t - 1]) <= abs(c[i] - b[t])) {
        ans[t - 1]++;
      } else
        ans[t]++;
    }
  }
  for (long long i = 0; i < m; i++) {
    cout << ans[i] << " ";
  }
  cout << "\n";
  return 0;
}
