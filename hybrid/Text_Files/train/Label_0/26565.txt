#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  int total;
  cout << "? " << 1 << " " << n << endl;
  cin >> total;
  int prev = total;
  vector<int> vt;
  int i = 2, val, sum = 0;
  while (i < n) {
    cout << "? " << i << " " << n << endl;
    cin >> val;
    vt.push_back(prev - val);
    sum += (prev - val);
    prev = val;
    i++;
  }
  cout << "? " << 1 << " " << n - 1 << endl;
  cin >> val;
  vt.push_back(val - sum);
  sum += (val - sum);
  vt.push_back(total - sum);
  cout << "!"
       << " ";
  for (auto x : vt) cout << x << " ";
  fflush(stdout);
  return (0);
}
