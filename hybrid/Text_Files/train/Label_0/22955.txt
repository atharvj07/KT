#include <bits/stdc++.h>
using namespace std;
const int N = 5e4 + 10;
inline int rd() {
  int x = 0, w = 1;
  char ch = 0;
  while (ch < '0' || ch > '9') {
    if (ch == '-') w = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 3) + (x << 1) + (ch ^ 48);
    ch = getchar();
  }
  return x * w;
}
int n, a[N], b[N];
int cn[2][2];
char ss[N];
void print(int aa, int bb, int cc, int dd) {
  for (int i = 1; i <= n; ++i) {
    if (a[i] && b[i] && aa)
      printf("%d ", i), --aa;
    else if (a[i] && !b[i] && bb)
      printf("%d ", i), --bb;
    else if (!a[i] && b[i] && cc)
      printf("%d ", i), --cc;
    else if (!a[i] && !b[i] && dd)
      printf("%d ", i), --dd;
  }
  exit(0);
}
int main() {
  n = rd();
  scanf("%s", ss + 1);
  for (int i = 1; i <= n; ++i) a[i] = ss[i] - '0';
  scanf("%s", ss + 1);
  for (int i = 1; i <= n; ++i) b[i] = ss[i] - '0';
  for (int i = 1; i <= n; ++i) ++cn[a[i]][b[i]];
  for (int i = 0; i <= cn[1][1]; ++i) {
    int aa = i, bb = cn[1][1] - i;
    if (aa < bb) {
      if (cn[1][0] >= bb - aa &&
          max(cn[1][0] - (bb - aa), cn[0][1]) <= n / 2 - bb)
        print(aa, bb - aa, cn[0][1], n / 2 - bb - cn[0][1]);
    } else if (aa > bb) {
      if (cn[0][1] >= aa - bb &&
          max(cn[1][0], cn[0][1] - (aa - bb)) <= n / 2 - aa)
        print(aa, 0, cn[0][1] - (aa - bb), n / 2 - aa - (cn[0][1] - (aa - bb)));
    } else if (max(cn[1][0], cn[0][1]) <= n / 2 - aa)
      print(aa, 0, cn[0][1], n / 2 - aa - cn[0][1]);
  }
  puts("-1");
  return 0;
}
