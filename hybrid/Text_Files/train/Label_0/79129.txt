(n, m) = map(int, input().split())
f = []
for k in range(n):
	f.append(list(map(int, input().split())))
u = []
l = 0
for j in range(n - 1):
	for b in range(m - 1):
		g = 0
		if f[j][b] == 1 or f[j][b] == 'g':
			g += 1
		if f[j][b + 1] == 1 or f[j][b + 1] == 'g':
			g += 1
		if f[j + 1][b + 1] == 1 or f[j + 1][b + 1] == 'g':
			g += 1
		if f[j + 1][b] == 1 or f[j + 1][b] == 'g':
			g += 1
		if g == 4:
			u.append([j + 1, b + 1])
			f[j][b] = 'g'
			f[j][b + 1] = 'g'
			f[j + 1][b + 1] = 'g'
			f[j + 1][b] = 'g'
		else:
			pass
for e in f:
	if 1 not in e:
		pass
	else:
		print(-1)
		l += 1
		break
if l == 0:
	if len(u) == 0:
		print(0)
	else:
		print(len(u))
		for h in u:
			print(*h)
