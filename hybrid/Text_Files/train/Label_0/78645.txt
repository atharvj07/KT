c = ['1', '4', '6', '8', '9']
c2 = ['25', '27', '32', '35', '52', '57', '72', '75']
t = int(input())
for _ in range(t):
	k = int(input())
	n = input()
	if k == 1:
		print(1)
		print(n)
		continue
	s = 0
	for x in c:
		if x in n:
			print(1)
			print(x)
			s = 1
			break
	if s:
		continue
	print(2)
	if k == 2:
		print(n)
		continue
	if n.count('2') >= 2:
		print(22)
		continue
	elif n.count('3') >= 2:
		print(33)
		continue
	elif n.count('5') >= 2:
		print(55)
		continue
	elif n.count('7') >= 2:
		print(77)
		continue
	for x in c2:
		if x in n:
			print(x)
			s = 1
			break
	if s:
		continue
	print(n[0] + n[2])
