#include <stdio.h>

long long max(long long a, long long b) {
	return a > b ? a : b;
}

int main() {
	int n;
	long long a, b;

	scanf("%d", &n);
	a = b = 1;
	while (n--) {
		int c, d;
		long long x;

		scanf("%d%d", &c, &d);
		x = max((a + (c - 1)) / c, (b + (d - 1)) / d);
		a = c * x;
		b = d * x;
	}
	printf("%lld\n", a + b);
	return 0;
}
