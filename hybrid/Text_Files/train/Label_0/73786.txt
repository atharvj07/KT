import copy
t = int(input())
for T in range(t):
	n = int(input())
	seg = [set(list(map(int, input().split()))[1:]) for i in range(n - 1)]
	p = [0] * n
	for first in range(1, n + 1):
		p[0] = first
		ok = True
		segt = copy.deepcopy(seg)
		for i in range(1, n):
			for s in segt:
				s.discard(p[i - 1])
			cnt1 = 0
			save = -1
			for s in segt:
				if len(s) == 1:
					cnt1 += 1
					save = s
			if cnt1 != 1:
				ok = False
				break
			p[i] = save.pop()
		if not ok:
			continue
		segt = copy.deepcopy(seg)
		for s in segt:
			yes = False
			start = False
			for i in range(n):
				if p[i] in s:
					if not start:
						start = True
					s.discard(p[i])
				elif start:
					break
				if start and len(s) == 0:
					yes = True
					break
			if not yes:
				ok = False
				break
		if not ok:
			continue
		for i in p:
			print(i, end=' ')
		print()
		break
