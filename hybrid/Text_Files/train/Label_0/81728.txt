import math

def find_year(iao):
	if len(iao) == 1:
		year = 1980 + int(iao)
		if year < 1989:
			year += 10
		return year
	else:
		pre_year = find_year(iao[1:])
		pre_year_str = list(str(pre_year))
		if len(iao) > len(pre_year_str):
			pre_year_str = ['0' for _ in range(len(iao) - len(pre_year_str))] + pre_year_str
		if int(iao[0]) <= int(pre_year_str[-len(iao)]):
			pre_year_str[-len(iao)] = iao[0]
			return int(int(''.join(pre_year_str)) + math.pow(10, len(iao)))
		else:
			pre_year_str[-len(iao)] = iao[0]
			return int(''.join(pre_year_str))

def solve():
	n = int(input())
	for _ in range(n):
		y = input()[4:]
		year = find_year(y)
		print(year)
solve()
