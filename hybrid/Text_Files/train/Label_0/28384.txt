#include <bits/stdc++.h>
using namespace std;
string s[100000];
char c[50];
int n;
string aux;
void doit() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) scanf("%s", c), s[i] = string(c);
  string pref = s[0];
  for (int i = 1; i < n; ++i) {
    aux = "";
    for (int j = 0; j < pref.length(); ++j)
      if (pref[j] != s[i][j])
        break;
      else
        aux += pref[j];
    pref = aux;
  }
  int ans = pref.length();
  printf("%d\n", ans);
}
int main() { doit(); }
