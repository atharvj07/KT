n = int(input())

def computeGCD(x, y):
	while y:
		(x, y) = (y, x % y)
	return x

	if tab.count(tab[0]) == len(tab):
		print(-1)
	else:

		allValues = []
		for val in tab:
			allValues.append(val - mini)
		num1 = allValues[0]
		num2 = allValues[1]
		gcd = computeGCD(num1, num2)
