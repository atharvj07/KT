#include <bits/stdc++.h>
using namespace std;
inline void read(int &x) {
  x = 0;
  char ch;
  bool flag = false;
  while (ch = getchar(), ch < '!')
    ;
  if (ch == '-') ch = getchar(), flag = true;
  while (x = 10 * x + ch - '0', ch = getchar(), ch > '!')
    ;
  if (flag) x = -x;
}
const int maxn = 300010;
int ch[maxn][27];
int dep[maxn], mxp, anss[maxn];
int get_num(const vector<int> &v) {
  int ret = v.size() - 1;
  int n = v.size();
  vector<int> vx;
  for (int i = 'a'; i <= 'z'; ++i) {
    vx.clear();
    for (int j = 0; j < n; ++j) {
      if (ch[v[j]][(i - 'a')]) vx.push_back(ch[v[j]][(i - 'a')]);
    }
    if (vx.size() <= 1) continue;
    ret += get_num(vx);
  }
  return ret;
}
void dfs(int u, int dep) {
  mxp = max(mxp, dep);
  vector<int> v;
  for (int i = 'a'; i <= 'z'; ++i) {
    if (ch[u][(i - 'a')]) v.push_back(ch[u][(i - 'a')]);
  }
  anss[dep] += get_num(v) + 1;
  for (int i = 'a'; i <= 'z'; ++i) {
    if (ch[u][(i - 'a')]) dfs(ch[u][(i - 'a')], dep + 1);
  }
  return;
}
int main() {
  int n;
  read(n);
  char c;
  for (int i = 1, u, v; i < n; ++i) {
    read(u);
    read(v);
    while (c = getchar(), c < '!')
      ;
    ch[u][(c - 'a')] = v;
  }
  dfs(1, 1);
  int ans = 0, ans_p = -1;
  for (int i = 1; i <= mxp; ++i) {
    if (ans < anss[i]) {
      ans = anss[i];
      ans_p = i;
    }
  }
  printf("%d\n%d\n", n - ans, ans_p);
  getchar();
  getchar();
  return 0;
}
