#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> a;

int solve(int x, int now);

int main() {
  cin >> n;
  for(int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    a.push_back(x);
  }
  cout << min(solve(0, 1), solve(1, 1)) << endl;
  return 0;
}

int solve(int x, int now) {
  int ans = 0;
  while(now < a.size() &&
        (a[now - 1] < a[now]) == (now + x) % 2)
    ++now;
  if(now == a.size()) return 0;
  swap(a[now], a[now - 1]);
  ans = solve(x, now + 1);
  swap(a[now], a[now - 1]);
  if(now + 1 < a.size() &&
     (a[now] < a[now + 1]) != (now + 1 + x) % 2 &&
     (a[now - 1] < a[now + 1]) == (now + x) % 2) {
    swap(a[now], a[now + 1]);
    ans = min(ans, solve(x, now + 1));
    swap(a[now], a[now + 1]);
  }
  return ans + 1;
}
