from collections import defaultdict
n = int(input())
d = defaultdict(int)
e = defaultdict(int)
for i in range(n):
	(x, y) = map(int, input().split())
	d[x + y] += 1
	e[y - (x - 1)] += 1
ans = 0
for i in d.keys():
	ans += d[i] * (d[i] - 1) // 2
for i in e.keys():
	ans += e[i] * (e[i] - 1) // 2
print(ans)
