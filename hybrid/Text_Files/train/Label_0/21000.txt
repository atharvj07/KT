#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:64000000")
using namespace std;
int n;
int a[1 << 17];
int b[1 << 17];
int p[1 << 17];
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
    --a[i];
  }
  for (int i = 0; i < n; ++i) {
    scanf("%d", &b[i]);
    --b[i];
  }
  for (int i = 0; i < n; ++i) {
    p[a[i]] = b[i];
  }
  for (int i = 0; i < n; ++i) printf("%d ", p[i] + 1);
  return 0;
}
