#include<bits/stdc++.h>
using namespace std;

string dp[26][21];
string Z="{";

void init(){
  for(int i=0;i<26;i++){
    for(int j=0;j<=20;j++){
      dp[i][j]=Z;
    }
  }
}

int N,M;
string from[50],to[50];

bool check(int id){
  int sd=from[id][0]-'A';
  
  if( to[id]=="" ){
    //    cout<<"!!"<<endl;
    
    bool res=( dp[sd][0] == Z );
    dp[sd][0]="";
    return res;
  }
  
  string t[21];
  for(int i=0;i<=M;i++)t[i]=Z;
  t[0]="";

  bool allZero=true;
  
  for(int i=0;i<(int)to[id].size();i++){
    char ch=to[id][i];
    if('A'<=ch&&ch<='Z'){
      int target=ch-'A';
      if( dp[target][0] == Z )allZero=false;
      
      for(int j=M;j>=0;j--){
        if( dp[target][0]==Z )  t[j]=Z;
        
        for(int k=j;k>=0;k--){
          if(t[j-k]!=Z && dp[target][k]!=Z){
            t[j]=min(t[j],t[j-k]+dp[target][k]);
          }
        }
      }
      
    }else{
      allZero=false;
      for(int j=M;j>=1;j--){
        if(t[j-1]!=Z)t[j]=t[j-1]+ch;
        else t[j]=Z;
      }
      t[0]=Z;
    }
  }
  
  bool res=false;
  for(int i=1;i<=M;i++){
    if(dp[sd][i]>t[i]){
      dp[sd][i]=t[i];
      res=true;
    }
  }
    
  if( allZero && dp[sd][0]==Z){
    res=true;
    dp[sd][0]="";
  }
  
  return res;
}

int main(){
  while(1){
    cin>>N>>M;
    if(N==0&&M==0)break;
    init();
    for(int i=0;i<N;i++){
      string str;
      cin>>str;
      int index=str.find('=');
      from[i]=str.substr(0,index);
      
      if(index+1 == (int)str.size()){
        to[i]="";
        //        cout<<i<<endl;
      }else{
        to[i]=str.substr(index+1);
      }

      
    }
    
    bool update=true;
    while(update){
      update=false;
      for(int i=0;i<N;i++){
        update=(update||check(i));
      }
      /*

      for(int i=0;i<5;i++){
        cout<<"dp[A]["<<i<<"] = "<<dp[0][i]<<endl;
      }
      for(int i=0;i<5;i++){
        cout<<"dp[B]["<<i<<"] = "<<dp[1][i]<<endl;
      }
      for(int i=0;i<5;i++){
        cout<<"dp[S]["<<i<<"] = "<<dp['S'-'A'][i]<<endl;
      }
      */
    }
    
    string ans = dp[ 'S'-'A' ][ M ];
    if(ans==Z)cout<<'-'<<endl;
    else cout<<ans<<endl;
  }
  return 0;
}
