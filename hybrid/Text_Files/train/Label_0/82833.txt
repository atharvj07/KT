import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	(*a,) = map(int, input().split())
	(*b,) = map(int, input().split())
	A = [0] * (2 * n + 3)
	B = [0] * (2 * n + 3)
	for i in range(n):
		A[a[i]] = i
		B[b[i]] = i
	r = 10 * n
	c = r
	for j in range(2 * n - 1, 0, -2):
		c = min(c, B[j + 1])
		r = min(r, A[j] + c)
	print(r)
for i in range(int(input())):
	solve()
