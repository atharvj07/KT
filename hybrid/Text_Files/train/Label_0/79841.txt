(n, m) = map(int, input().split())
a = list(map(int, input().split()))
sta = list(map(int, input().split()))
taxi = []
ans = [0 for i in range(m)]
for i in range(n + m):
	if sta[i] == 1:
		taxi.append(a[i])
for i in range(n + m):
	if sta[i] == 0:
		l = 0
		r = m - 1
		tmp = -1
		while r >= l:
			mid = (l + r) // 2
			if taxi[mid] >= a[i]:
				r = mid - 1
				tmp = mid
			else:
				l = mid + 1
		if tmp == -1:
			ans[m - 1] += 1
		if tmp == 0:
			ans[0] += 1
		if tmp > 0:
			if a[i] - taxi[tmp - 1] <= taxi[tmp] - a[i]:
				ans[tmp - 1] += 1
			else:
				ans[tmp] += 1
for i in range(m):
	print(ans[i], end=' ')
