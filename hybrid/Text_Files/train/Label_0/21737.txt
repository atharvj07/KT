#include<bits/stdc++.h>
#define int long long

using namespace std;

int n,k,q[110];
int mem[110][5][5];
static const int mod = 10000;
int dfs(int day,int p,int pp){
    int ans=0;
    if(day==0)return 1;
    if(mem[day][p][pp]!=-1)return mem[day][p][pp];
    if(q[day]!=0){
        if(p==pp&&q[day]==p)return 0;
        ans=dfs(day-1,q[day],p);
        return mem[day][p][pp]=ans;
    }
    for(int i=1;i<=3;i++){
        if(p==pp&&p==i) continue;
        ans+=dfs(day-1,i,p);
    }
    ans=ans%mod;
    mem[day][p][pp]=ans;
    return ans;
}

signed main(){
    cin>>n>>k;
    memset(mem,-1,sizeof(mem));
    int a,b;
    for(int i=1;i<=k;++i){
        cin>>a>>b;
        q[a]=b;
    }
    cout<<dfs(n,0,0)<<endl;
}
