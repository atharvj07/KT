#include <bits/stdc++.h>
using namespace std;
char a[200010], b[200010];
int main() {
  int l1, l2, i, j, num;
  num = 0;
  scanf("%s%s", a, b);
  l1 = strlen(a);
  l2 = strlen(b);
  i = l1 - 1;
  j = l2 - 1;
  while (a[i] == b[j] && i >= 0 && j >= 0) {
    i--;
    j--;
    num++;
  }
  printf("%d", l1 + l2 - 2 * num);
  return 0;
}
