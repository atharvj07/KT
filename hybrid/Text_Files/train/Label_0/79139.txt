n = input()
inp = input().split()
mark1 = 0
mark2 = 0
judge = 0
num = int(inp[0])
for i in range(1, int(n)):
	if int(inp[i]) > num:
		num = int(inp[i])
		if mark1 == 1 or mark2 == 1:
			judge = 1
			break
	elif int(inp[i]) == num:
		mark1 = 1
		if mark2 == 1:
			judge = 1
			break
	else:
		num = int(inp[i])
		mark2 = 1
if judge == 0:
	print('YES')
else:
	print('NO')
