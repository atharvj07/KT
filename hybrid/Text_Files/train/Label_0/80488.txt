n = int(input())
d = {}
res = []
for i in range(n):
	m = int(input())
	a = [int(e) for e in input().split()]
	if not res:
		s = sum(a)
		for j in range(m):
			s1 = s - a[j]
			if s1 not in d:
				d[s1] = (i + 1, j + 1)
			elif d[s1][0] != i + 1:
				res = [d[s1], (i + 1, j + 1)]
				break
if res:
	print('YES')
	print(res[0][0], res[0][1])
	print(res[1][0], res[1][1])
else:
	print('NO')
