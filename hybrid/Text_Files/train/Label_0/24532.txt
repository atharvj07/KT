#include <bits/stdc++.h>
using namespace std;
int rem = 987654, n, m;
int ask(string s) {
  if (rem != 987654) rem--;
  if (rem < 0) exit(1);
  cout << s << '\n';
  cout.flush();
  int x;
  cin >> x;
  if (x == 0) exit(0);
  return x;
}
int main() {
  n = ask("a");
  if (n + 1 == m) {
    string s = "";
    for (int i = 0; i < m; i++) s += "a";
    ask(s);
    exit(0);
  }
  if (n == 300) {
    string s = "";
    for (int i = 0; i < n; i++) s += "b";
    ask(s);
    exit(0);
  }
  n++;
  rem = n + 1;
  string s = "", amid = "";
  for (int i = 0; i < n; i++) {
    s += "a";
    amid += "a";
  }
  int dis = ask(s);
  if (dis == n) {
    s = "";
    for (int i = 1; i < n; i++) s += 'b';
    ask(s);
  }
  for (int i = 0; i < n; i++) {
    s[i] = 'b';
    int tmp = ask(s);
    if (tmp > dis) {
      s[i] = 'a';
      continue;
    }
    dis = tmp;
    amid[i] = 'b';
  }
}
