#include <bits/stdc++.h>
#pragma GCC optimize("Ofast,no-stack-protector,unroll-loops,fast-math")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx,tune=native")
using namespace std;
const int maxn = 44, totale = 2510 * 10 * 44, mdelta = 2510 * maxn;
int p[maxn], tdo[maxn][maxn], cnt[maxn], inth[maxn];
bool done[maxn], lst[maxn];
int who[2][maxn][mdelta], delta, col[totale], va[totale], vb[totale];
int arr[2 * maxn];
vector<int> vc1, vc2;
void add(int a, int b, int id, int numb) {
  vc1.clear(), vc2.clear();
  for (int i = 0; i < delta; i++) {
    if (numb <= 0) return;
    bool A = who[0][a][i] == -1;
    bool B = who[1][b][i] == -1;
    if (A && B) {
      va[id] = a, vb[id] = b, col[id] = i;
      who[0][a][i] = id;
      who[1][b][i] = id;
      id++, numb--;
    } else if (A)
      vc2.push_back(i);
    else if (B)
      vc1.push_back(i);
  }
  if (numb <= 0) return;
  assert(min(int((vc1).size()), int((vc2).size())) >= numb);
  while (numb--) {
    int c1 = vc1.back(), c2 = vc2.back();
    vc1.pop_back(), vc2.pop_back();
    int C = 0;
    int tmp = a;
    while (true) {
      if (who[0][tmp][c1] == -1) break;
      arr[C++] = who[0][tmp][c1];
      tmp = vb[who[0][tmp][c1]];
      if (who[1][tmp][c2] == -1) break;
      arr[C++] = who[1][tmp][c2];
      tmp = va[who[1][tmp][c2]];
    }
    swap(who[0][a][c1], who[0][a][c2]);
    for (int i = 0; i < C; i++) {
      int u = (i & 1 ? va[arr[i]] : vb[arr[i]]);
      bool side = i & 1 ? 0 : 1;
      swap(who[side][u][c1], who[side][u][c2]);
      if (col[arr[i]] == c1)
        col[arr[i]] = c2;
      else
        col[arr[i]] = c1;
    }
    va[id] = a, vb[id] = b, col[id] = c1;
    who[0][a][c1] = id;
    who[1][b][c1] = id;
    id++;
  }
}
int main() {
  memset(who, -1, sizeof who);
  int n, m, B;
  scanf("%d%d%d", &n, &m, &B);
  int total = 0;
  for (int i = 0; i < m; i++) scanf("%d", &p[i]);
  for (int i = 0; i < n; i++) {
    int q;
    scanf("%d", &q);
    int sm = 0;
    while (q--) {
      int a, b;
      scanf("%d%d", &a, &b);
      tdo[i][--a] = b;
      sm += b;
      cnt[a] += b;
    }
    delta = max(delta, sm);
    total += sm;
  }
  while (true) {
    int id = -1;
    for (int i = 0; i < m; i++) {
      if (p[i] > B || done[i]) continue;
      if (id == -1) id = i;
      if (cnt[id] < cnt[i]) id = i;
    }
    if (id == -1) break;
    B -= p[id];
    done[id] = 1;
    delta = max(delta, (cnt[id] + 1) / 2);
  }
  for (int i = 0; i < m; i++) {
    if (done[i] == 0) delta = max(delta, cnt[i]);
  }
  printf("%d\n", delta);
  for (int i = 0; i < m; i++) printf("%d", done[i]);
  printf("\n");
  int C = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (done[j]) {
        int CC = tdo[i][j] / 2;
        add(i, 2 * j, C, CC);
        add(i, 2 * j + 1, C + CC, CC);
        if (2 * CC != tdo[i][j]) {
          if (lst[j])
            add(i, 2 * j, C + CC + CC, 1);
          else
            add(i, 2 * j + 1, C + CC + CC, 1);
          lst[j] ^= 1;
        }
      } else {
        add(i, 2 * j, C, tdo[i][j]);
      }
      C += tdo[i][j];
    }
  }
  vector<pair<pair<int, int>, pair<int, int> > > ANS;
  vector<int> comp;
  C = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      comp.clear();
      for (int k = 0; k < tdo[i][j]; k++) {
        comp.push_back(col[C++]);
      }
      sort(comp.begin(), comp.end());
      int start = -2, numb = 0;
      for (auto x : comp) {
        if (start + numb == x) {
          numb++;
        } else {
          if (start != -2) ANS.push_back({{i + 1, j + 1}, {start, numb}});
          numb = 1, start = x;
        }
      }
      if (start != -2) ANS.push_back({{i + 1, j + 1}, {start, numb}});
    }
  }
  cout << int((ANS).size()) << "\n";
  for (auto x : ANS) {
    printf("%d %d %d %d\n", x.first.first, x.first.second, x.second.first,
           x.second.second);
  }
  return 0;
}
