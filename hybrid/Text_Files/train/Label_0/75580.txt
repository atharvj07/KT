T = int(input())
while T != 0:
	T = T - 1
	(n, k) = (int(x) for x in input().split())
	num = input()
	cnt = 0
	res = ''
	for i in range(n):
		if num[i] == '0':
			if cnt <= k:
				k = k - cnt
				res += '0'
			else:
				res = res + '1' * (cnt - k)
				res = res + '0'
				res = res + '1' * k
				res = res + num[i + 1:]
				break
		else:
			cnt += 1
	if len(res) != n:
		res = res + '1' * cnt
	print(res)
