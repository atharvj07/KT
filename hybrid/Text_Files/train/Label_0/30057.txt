#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string a, s;
  cin >> n >> a >> s;
  a = "0";
  if (s == "week") {
    if (n <= 4 || n == 7) {
      cout << 52 << '\n';
    } else {
      cout << 53 << '\n';
    }
  } else {
    if (n == 30) {
      cout << 11 << '\n';
    } else if (n == 31) {
      cout << 7 << '\n';
    } else {
      cout << 12 << '\n';
    }
  }
}
