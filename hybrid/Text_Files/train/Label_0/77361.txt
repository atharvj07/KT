from operator import itemgetter
import sys
input = lambda : sys.stdin.readline().rstrip('\r\n')

def solve(n, m, x, s):
	arr = sorted([(i, x[i], s[i]) for i in range(n)], key=itemgetter(1))
	stacks = [[], []]
	ans = [-1] * n
	for (i, xi, si) in arr:
		stk = stacks[xi % 2]
		if not stk:
			stk.append((i, xi, si))
		else:
			(oi, ox, os) = stk[-1]
			if os == 'R' and si == 'L':
				ans[i] = ans[oi] = (xi - ox) // 2
				stk.pop()
			elif os == 'L' and si == 'L':
				ans[i] = ans[oi] = (xi + ox) // 2
				stk.pop()
			else:
				stk.append((i, xi, si))
	for stk in stacks:
		stk.reverse()
		l = len(stk)
		for i in range(0, l, 2):
			if i >= l - 2:
				break
			(ii, xi, _) = stk[i]
			(oi, ox, _) = stk[i + 1]
			ans[ii] = ans[oi] = (2 * m - ox - xi) // 2
		if i == l - 2:
			(ii, xi, si) = stk[i]
			(oi, ox, os) = stk[i + 1]
			if si == 'R' and os == 'R':
				ans[ii] = ans[oi] = (2 * m - ox - xi) // 2
			elif si == 'R' and os == 'L':
				ans[ii] = ans[oi] = (2 * m - xi + ox) // 2
	print(*ans)
T = int(input())
for t in range(T):
	(n, m) = map(int, input().split())
	x = list(map(int, input().split()))
	s = input().split()
	solve(n, m, x, s)
