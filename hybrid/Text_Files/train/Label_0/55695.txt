import java.io.*;
import java.util.*;

public class EMain {
    String returnValue = "NoResult";
    PrintWriter out;
    Parser in = new Parser();

    int n = in.nextInteger();
    int m = in.nextInteger();
    int k = in.nextInteger();
    long p = in.nextInteger();
    B[] bb = new B[n + 1];

    class B{
        int a;
        int b;

        long total;

        B(int a, int b) {
            this.a = a;
            this.b = b;
            total = a + b * (long)m;
        }
    }

    public void solve() {

        for(int i = 0; i < n; ++i) {
            bb[i] = new B(in.nextInteger(), in.nextInteger());
        }

        bb[n] = new B(0, 0);

        Arrays.sort(bb, new Comparator<B>() {
            @Override
            public int compare(B o1, B o2) {
                return Long.compare(o2.total, o1.total);
            }
        });

        long total = p * k * m;
        int i;
        for(i = 1; total > 0 && i < n + 1; ++i){
            total -= (bb[i - 1].total - bb[i].total) * i;
        }
        --i;
        long lowBound = 0;
        if(total < 0){
            n = i;
            total += (bb[i - 1].total - bb[i].total) * i;
            lowBound = bb[i - 1].total - total / i;
            bb = Arrays.copyOf(bb, n);
        }
        long upperBound = bb[0].total + 1;
        while(upperBound - lowBound > 1){
            long middle = (upperBound + lowBound) >> 1;
            if(check(middle - 1)){
                upperBound = middle;
            } else {
                lowBound = middle;
            }
        }
        out.print(lowBound);
    }

    private boolean check(long limit) {
        int [] booms = new int[m + 1];
        for(B b : bb){
            if(b.total <= limit) continue;
            long shots = (b.total - limit + p - 1) / p;
            long rem = (b.total - limit) / p;

            int lastIndex = 0;
            for(int i = 0; i < shots; ++i){
                int index = (int) (rem + i*p - b.a);
                if(index < 0){
                    lastIndex = 0;
                    ++booms[0];
                }
                else {
                    lastIndex = 1 + (index - 1)/ b.b;
                    ++booms[lastIndex];
                }
            }

            --booms[lastIndex];
            ++booms[((int) Math.max(lastIndex, m - limit / b.b))];
        }

        int s = 0;
        for(int i = m; i >= 0; --i){
            if(booms[i] > s) return false;
            s += k - booms[i];
        }
        return true;
    }

    static public class Parser{
        Scanner scanner;

        public Parser() {
            scanner = new Scanner(System.in).useLocale(Locale.ENGLISH);
        }

        public Parser(String str) {
            scanner = new Scanner(str).useLocale(Locale.ENGLISH);
        }

        long nextLong(){
            return scanner.nextLong();
        }

        int nextInteger(){
            return scanner.nextInt();
        }

        double nextDouble(){
            return scanner.nextDouble();
        }

        String nextLine(){
            return scanner.nextLine();
        }

        String next(){
            return scanner.next();
        }

        int[] nextIntegers(int count){
            int[] result = new int[count];
            for(int i = 0; i < count; ++i){
                result[i] = nextInteger();
            }
            return result;
        }

        long[] nextLongs(int count){
            long[] result = new long[count];
            for(int i = 0; i < count; ++i){
                result[i] = nextLong();
            }
            return result;
        }

        int[][] nextIntegers(int fields, int count){
            int[][] result = new int[fields][count];
            for(int c = 0; c < count; ++c){
                for(int i = 0; i < fields; ++i){
                    result[i][c] = nextInteger();
                }
            }
            return result;
        }
        long[][] nextLongs(int fields, int count){
            long[][] result = new long[fields][count];
            for(int c = 0; c < count; ++c){
                for(int i = 0; i < fields; ++i){
                    result[i][c] = nextLong();
                }
            }
            return result;
        }
    }

    void doReturn(){
        throw new NoAnswerException();
    }

    void doReturn(String str){
        returnValue = str;
        doReturn();
    }

    void run(){
        try{
            ByteArrayOutputStream outStream = new ByteArrayOutputStream();
            out = new PrintWriter(System.out);
            solve();
            out.close();
            System.out.print(outStream.toString());
        } catch (NoAnswerException exc){
            System.out.print(returnValue);
        }
    }

    public static void main(String[] args) {
        new EMain().run();
    }

    static class NoAnswerException extends RuntimeException{

    }
}
