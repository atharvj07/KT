from sys import stdin, stdout
input = stdin.readline

def im():
	return map(int, input().split())

def ii():
	return int(input())

def il():
	return list(map(int, input().split()))

def ins():
	return input()[:-1]
for _ in range(ii()):
	(n, l, r) = im()
	lis = il()
	left = lis[:l]
	right = lis[l:]
	ldic = [0] * (n + 1)
	rdic = [0] * (n + 1)
	for i in left:
		ldic[i] += 1
	for i in right:
		rdic[i] += 1
	for i in left:
		if ldic[i] > 0 and rdic[i] > 0:
			mi = min(ldic[i], rdic[i])
			ldic[i] -= mi
			rdic[i] -= mi
			l -= mi
			r -= mi
	count = 0
	while True:
		if l == r:
			stdout.write(str(count + l) + '\n')
			break
		elif l < r:
			flag = 0
			for i in range(len(rdic)):
				if rdic[i] > 1:
					cnt = min(rdic[i] // 2, (r - l) // 2)
					rdic[i] -= cnt * 2
					count += cnt
					r -= cnt * 2
					flag = 1
					if l == r:
						break
			if flag == 0:
				cnt = (r - l) // 2
				count += cnt
				l += cnt
				r -= cnt
		else:
			flag = 0
			for i in range(len(ldic)):
				if ldic[i] > 1:
					cnt = min(ldic[i] // 2, (l - r) // 2)
					ldic[i] -= cnt * 2
					count += cnt
					l -= cnt * 2
					flag = 1
					if l == r:
						break
			if flag == 0:
				cnt = (l - r) // 2
				count += cnt
				r += cnt
				l -= cnt
