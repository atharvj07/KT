n = int(input())
s = [*input()]
(a, b) = ([], [])
for i in range(n - 1, -1, -1):
	if s[i] == '8':
		a.append(i)
	else:
		b.append(i)
m = True
while len(a) + len(b) > 11:
	if not a or (m and b):
		b.pop()
	else:
		a.pop()
	m = not m
if a and (not b or (b and a[-1] < b[-1])):
	print('YES')
else:
	print('NO')
