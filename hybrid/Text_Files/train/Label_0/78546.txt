def gcd(a, b):
	while b:
		(a, b) = (b, a % b)
	return a

def lcm(a, b):
	return a // int(gcd(a, b)) * b

def main():
	l = input().split()
	t = int(l[0])
	w = int(l[1])
	b = int(l[2])
	ans = min(min(w, b) - 1, t - t // lcm(w, b) * lcm(w, b)) + t // lcm(w, b) * min(w, b)
	g = gcd(ans, t)
	ans = ans // g
	t = t // g
	if ans > t:
		(ans, t) = (1, 1)
	an = str(int(ans)) + '/' + str(int(t))
	print(an)
main()
