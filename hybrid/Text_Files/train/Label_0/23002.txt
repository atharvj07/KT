#include <bits/stdc++.h>
#define FOR(i,a,b) for(int i=(a);i<(b);i++)
#define RFOR(i,a,b) for(int i=(b) - 1;i>=(a);i--)
#define REP(i,n) for(int i=0;i<(n);i++)
#define RREP(i,n) for(int i=n-1;i>=0;i--)

#define PB push_back
#define MP make_pair
#define ALL(a) (a).begin(),(a).end()
#define RALL(a) (a).rbegin(),(a).rend()

#define DEBUG(x) cout<<"#x"<<": "<<x<<endl

using namespace std;

typedef long long int ll;
typedef vector<int> vi;
typedef vector<ll> vl;

const ll INF = INT_MAX/3;
const ll MOD = 1000000007;
const double EPS = 1e-14;
const int dx[] = {1,0,-1,0} , dy[] = {0,1,0,-1};

int h,w;
int board[21][21];
vector<string> ss;

void moveG(int &x,int &y,char p){
    int tx,ty;
    tx = x;
    ty = y;
    if(p == '5'){
	return;
    }else if(p == '8'){
	if(y > 0)
	    y--;
    }else if(p == '6'){
	if(x < w-1)
	    x++;
    }else if(p == '4'){
	if(x > 0)
	    x--;
    }else if(p == '2'){
	if(y < h-1)
	    y++;
    }

}
int main(){
    while(true){
	cin >> h >> w;
	if(h == 0 && w == 0) break;
	string pattern;
	ss = vector<string>(h);
	REP(i,h) cin >> ss[i];
	cin >> pattern;

	int sx,sy,gx,gy;
	int ans = 0;
	REP(i,h)REP(j,w){
	    board[i][j] = INF;
	    if(ss[i][j] == 'A'){
		sx = j;
		sy = i;
	    }else if(ss[i][j] == 'B'){
		gx = j;
		gy = i;
	    }
	}

	bool flg = false;
	int p = 0;
	int t = 0;
	board[sy][sx] = 0;
	queue<int> qx;
	queue<int> qy;
	qx.push(sx);
	qy.push(sy);
	while(!qx.empty()){
	    int x = qx.front();
	    int y = qy.front();
	    qx.pop();
	    qy.pop();
	    int v = board[y][x]+1;
	    REP(i,4){
		int nx = x + dx[i];
		int ny = y + dy[i];
		if(nx < 0 || w <= nx || ny < 0 || h <= ny || ss[ny][nx] == '#') continue;
		if(board[ny][nx] > v){
		    board[ny][nx] = v;
		    qx.push(nx);
		    qy.push(ny);
		}
	    }
	}
	bool u[31][31];
	REP(i,31)REP(j,31) u[i][j] = false;
	while(t < 1000){
	    if(board[gy][gx] <= t ){
		printf("%d %d %d\n",t,gy,gx);
		flg = true;
		break;
	    }
	    moveG(gx,gy,pattern[p]);
	    p = (p+1) % pattern.size();
	    t++;
	}
	if(!flg) cout << "impossible" << endl;
    }

    return 0;
}
