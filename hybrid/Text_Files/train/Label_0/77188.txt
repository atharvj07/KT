(n, m) = map(int, input().split())
c = list(map(int, input().split()))
M = max(c)
g = [set() for _ in range(M + 1)]
for _ in range(m):
	(a, b) = map(int, input().split())
	a -= 1
	b -= 1
	if c[a] != c[b]:
		g[c[a]].add(c[b])
		g[c[b]].add(c[a])
ans = min(c)
for i in range(M + 1):
	if len(g[i]) > len(g[ans]):
		ans = i
print(ans)
