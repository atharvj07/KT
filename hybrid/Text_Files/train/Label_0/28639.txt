#include <bits/stdc++.h>
using namespace std;
long long m, n, k;
int main() {
  ios::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t--) {
    cin >> m >> n >> k;
    if (k < max(n, m))
      cout << -1 << endl;
    else {
      long long tmp = abs(n - m);
      long long dig = min(n, m);
      if (tmp % 2) {
        cout << k - 1 << endl;
      } else {
        if ((k - tmp) % 2 == dig % 2) {
          cout << k << endl;
        } else {
          cout << k - 2 << endl;
        }
      }
    }
  }
  return 0;
}
