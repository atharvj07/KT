def code_input():
	amount = int(input())
	for _ in range(amount):
		n = 0
		k = int(input())
		column = 1
		row = 1
		i = 1
		last = 1
		while n < k:
			last = n + 1
			n += i * 2 - 1
			i += 1
		i += -1
		n += 1
		row = k - last + 1 if k - last < i else i
		column = i if k - last < i else i - (k - last - i + 1)
		print('{} {}'.format(row, column))
code_input()
