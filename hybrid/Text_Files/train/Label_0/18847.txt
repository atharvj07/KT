#include <bits/stdc++.h>
using namespace std;
const int N = 5e4 + 5;
const int inf = ~0U >> 1;
int n;
int a[60], c[60];
int dp[30][N * 25];
bool vis[N];
int main() {
  cin >> n;
  for (int i = (1); i < (2 * n + 1); i++) cin >> c[i];
  sort(c + 1, c + 2 * n + 1);
  n *= 2;
  n -= 2;
  int sum = 0, ss = 0;
  for (int i = (1); i < (n + 1); i++) {
    a[i] = c[i + 2];
    if (i > n / 2) sum += a[i];
    ss += a[i];
  }
  dp[0][0] = 1;
  for (int i = 1; i <= n; ++i)
    for (int j = n / 2; j >= 1; --j)
      for (int k = sum; k >= 0; --k) {
        if (dp[j][k] == 0 && k >= a[i] && dp[j - 1][k - a[i]]) {
          dp[j][k] = i;
        }
      }
  for (ss /= 2; ss >= 0; --ss)
    if (dp[n / 2][ss]) break;
  vector<int> p, q;
  p.push_back(c[1]);
  q.push_back(c[2]);
  int u = n / 2;
  dp[0][0] = 0;
  while (u) {
    vis[dp[u][ss]] = true;
    ss -= a[dp[u][ss]];
    u--;
  }
  for (int i = 1; i <= n; ++i)
    if (vis[i])
      p.push_back(a[i]);
    else
      q.push_back(a[i]);
  sort(p.begin(), p.end());
  sort(q.begin(), q.end());
  reverse(q.begin(), q.end());
  for (auto v : p) cout << v << " ";
  cout << "\n";
  for (auto v : q) cout << v << " ";
  return 0;
}
