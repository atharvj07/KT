#include <bits/stdc++.h>
using namespace std;
int n, a[100001];
int cnt, x, y;
vector<pair<int, int> > g[100001];
void dfs(int v, long long mn) {
  if (mn > a[v]) {
    return;
  }
  cnt++;
  for (int i = 0; i < g[v].size(); i++) {
    dfs(g[v][i].first, max(mn + g[v][i].second, 0LL));
  }
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i < n; i++) {
    cin >> x >> y;
    g[x].push_back(make_pair(i + 1, y));
  }
  dfs(1, 0);
  cout << n - cnt;
  return 0;
}
