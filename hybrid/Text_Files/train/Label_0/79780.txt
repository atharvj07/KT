from math import *

def main():
	n = int(input())
	arr = list(map(int, input().split()))
	arr = sorted(arr)
	x = n // 3
	z = [arr[:x], arr[x:2 * x], arr[2 * x:]]
	res = []
	for j in range(x):
		p = [z[0][0]]
		x = z[0][0]
		f = True
		for n in range(1, 3):
			for i in range(x):
				f = True
				if z[n][i] % x == 0 and z[n][i] > x:
					x = z[n][i]
					p.append(z[n][i])
					z[n].pop(i)
					f = False
					break
				if f:
					print(-1)
					return
		res.append(p.copy())
	print('\n'.join(map(lambda x: ' '.join(map(str, x)), res)))
main()
