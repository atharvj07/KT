#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long temp = n;
    vector<long long> v;
    for (long long i = 2; i <= sqrt(n); i++) {
      while (n % i == 0) {
        v.push_back(i);
        n /= i;
      }
    }
    if (n != 1) {
      v.push_back(n);
    }
    if (v.size() < 3) {
      cout << "No" << endl;
      continue;
    }
    long long fi = v[0];
    long long se = v[1];
    if (se == fi) {
      se = se * v[2];
    }
    long long th = temp / (fi * se);
    if (th != fi && th != se && th != 1) {
      cout << "Yes" << endl;
      cout << fi << " " << se << " " << th << endl;
    } else
      cout << "No" << endl;
  }
}
