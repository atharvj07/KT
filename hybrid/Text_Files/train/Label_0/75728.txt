import math
import sys
input = sys.stdin.readline

def main():
	(n, m) = [int(x) for x in input().split(' ')]
	subjects = {}
	for i in range(n):
		(sub, skill) = [int(x) for x in input().split(' ')]
		if sub not in subjects:
			subjects[sub] = []
		subjects[sub].append(skill)
	for (k, v) in subjects.items():
		subjects[k] = sorted(v, reverse=True)
	best = 0
	iteration = 0
	while len(subjects) > 0:
		cur_sum = 0
		rem = []
		for (sub, group) in subjects.items():
			if iteration != 0:
				group[iteration] += group[iteration - 1]
			if group[iteration] > 0:
				cur_sum += group[iteration]
			if group[iteration] <= 0 or len(group) - 1 == iteration:
				rem.append(sub)
		for i in rem:
			subjects.pop(i)
		best = max(best, cur_sum)
		iteration += 1
	print(best)
main()
