n = []
for i in range(0, int(input())):
	(ro, p, m) = map(int, input().split())
	if abs(ro - p) == m or abs(ro - m) == p or abs(p - m) == ro:
		n.append('YES')
	elif ro == p:
		if m % 2 == 0:
			n.append('YES')
		else:
			n.append('NO')
	elif m == p:
		if ro % 2 == 0:
			n.append('YES')
		else:
			n.append('NO')
	elif ro == m:
		if p % 2 == 0:
			n.append('YES')
		else:
			n.append('NO')
	else:
		n.append('NO')
for c in range(len(n)):
	print(n[c])
