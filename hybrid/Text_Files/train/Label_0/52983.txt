import java.util.*;
public class D416 {
    static int n;
    static int m;
    static int board[][];
    static final int NORM = 0;
    static final int DANG = 1;
    static final int FINISH = 2;
    static int finX;
    static int finY;
    static int curX;
    static int curY;
    static Scanner sc;
    static boolean swapLR;
    static boolean swapUD;
    static Stack<Integer> path = new Stack<>();
    public static void main(String[] args) {
        sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        board = new int[n][m];
        for(int i = 0; i < n; ++i) {
            String s = sc.next();
            for(int j = 0; j < m; ++j) {
                int type;
                switch(s.charAt(j)) {
                case '.':
                    type = NORM;
                    break;
                case '*':
                    type = DANG;
                    break;
                default:
                    type = FINISH;
                }
                board[i][j] = type;
                if(type == FINISH) {
                    finX = i;
                    finY = j;
                }
            }
        }
        if(board[0][1] != DANG) {
            // moving along first row
            if(pressR()) return;
            if(curY == 0) {  //haven't moved
                swapLR = true;
            } else {
                //return to (0,0)
                if(pressL()) return;
            }
            while(true) {
                if(board[curX + 1][curY] == DANG) {
                    if(moveRight()) return;
                } else {
                    if(pressD()) return;
                    if(curX == 0) {     //haven't moved
                        swapUD = true;
                    }
                    break;
                }
            }
        } else {
            //moving along first column
            if(pressD()) return;
            if(curX == 0) {     //haven't moved
                swapUD = true;
            }
            while(true) {
                if(board[curX][curY + 1] == DANG) {
                    if(moveDown()) return;
                } else {
                    if(pressR()) return;
                    if(curY == 0) {  //haven't moved
                        swapLR = true;
                    }
                    break;
                }
            }
        }
        dfs(curX, curY);
       // System.out.println(path);
        while(!path.empty()) {
            switch(path.pop()) {
            case 3:
                moveUp();
                break;
            case 2:
                moveDown();
                break;
            case 0:
                moveRight();
                break;
            case 1:
                moveLeft();
            }
        }
        if(curX != finX || curY != finY) {
            System.out.println("Lost");
        }
    }
    
    static int[][] moves = new int[][] {
        {0,1},  // right
        {0,-1}, // left
        {1,0},  // down
        {-1,0}  // up
    };
    
    static boolean dfs(int x, int y) {
        if(x == finX && y == finY) {
            return true;
        }
        board[x][y] = DANG;
        for(int i = 0; i < 4; ++i) {
            int xx = x + moves[i][0];
            int yy = y + moves[i][1];
            if(xx >= 0 && xx < n && yy >= 0 && yy < m &&
                    board[xx][yy] != DANG) {
                if(dfs(xx, yy)) {
                    path.add(i);
                    return true;
                }
            }
        }
        return false;
    }
    
    /**
     * All next functions
     * @return true if you should terminate programm
     */
    
    static boolean pressR() {
        System.out.println("R");
        System.out.flush();
        curX = sc.nextInt() - 1;
        curY = sc.nextInt() - 1;
        if(board[curX][curY] != NORM) {
            return true;
        }
        return false;
    }

    static boolean pressL() {
        System.out.println("L");
        System.out.flush();
        curX = sc.nextInt() - 1;
        curY = sc.nextInt() - 1;
        if(board[curX][curY] != NORM) {
            return true;
        }
        return false;
    }

    static boolean pressU() {
        System.out.println("U");
        System.out.flush();
        curX = sc.nextInt() - 1;
        curY = sc.nextInt() - 1;
        if(board[curX][curY] != NORM) {
            return true;
        }
        return false;
    }

    static boolean pressD() {
        System.out.println("D");
        System.out.flush();
        curX = sc.nextInt() - 1;
        curY = sc.nextInt() - 1;
        if(board[curX][curY] != NORM) {
            return true;
        }
        return false;
    }
    
    static boolean moveUp() {
        if(swapUD) {
            return pressD();
        } else {
            return pressU();
        }
    }
    
    static boolean moveDown() {
        if(swapUD) {
            return pressU();
        } else {
            return pressD();
        }
    }
    
    static boolean moveLeft() {
        if(swapLR) {
            return pressR();
        } else {
            return pressL();
        }
    }
    
    static boolean moveRight() {
        if(swapLR) {
            return pressL();
        } else {
            return pressR();
        }
    }
}
