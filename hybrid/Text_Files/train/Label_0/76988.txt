import math

def main():
	n = int(input())
	a = [0]
	a.extend(list(map(int, input().split())))
	dp = [0 for _ in range(n + 1)]
	ans = [0 for _ in range(n + 1)]
	dp[n] = a[n]
	for i in range(n - 1, 0, -1):
		dp[i] = math.gcd(a[i], dp[i + 1])
	for i in range(1, n):
		ans[i] = int(a[i] * dp[i + 1] / math.gcd(a[i], dp[i + 1]))
	tot = ans[1]
	for i in range(2, n + 1):
		tot = math.gcd(tot, ans[i])
	print(tot)
main()
