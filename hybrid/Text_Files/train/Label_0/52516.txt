import java.io.*;
import java.util.*;
public class CF948_D2_C{
	public static void main(String[] args)throws Throwable {
		MyScanner sc=new MyScanner();
		PrintWriter pw=new PrintWriter(System.out);
		
		int n=sc.nextInt();
		int [] v=new int [n];
		for(int i=0;i<n;i++)
			v[i]=sc.nextInt();
		int [] t=new int [n];
		for(int i=0;i<n;i++)
			t[i]=sc.nextInt();
		long [] cumT=new long [n];
		cumT[0]=t[0];
		for(int i=1;i<n;i++)
			cumT[i]=cumT[i-1]+t[i];
		
		long [] res=new long [n];
		int [] cum=new int [n];
		for(int i=0;i<n;i++){
			int start=i,end=n-1,ans=-1;
			long rem=-1;
			while(start<=end){
				int mid=(start+end)/2;
				if(cumT[mid]-(i>0? cumT[i-1] : 0)>v[i]){
					ans=mid;
					rem=cumT[mid]-(i>0? cumT[i-1] : 0);
					rem-=t[mid];
					rem=v[i]-rem;
					end=mid-1;
				}else
					start=mid+1;
			}
			cum[i]++;
			if(ans!=-1){
				cum[ans]--;
				res[ans]+=rem;
			}
		}
		long c=0;
		for(int i=0;i<n;i++){
			c+=cum[i];
			res[i]+=c*t[i];
			pw.print(res[i]+" ");
		}
		
		pw.flush();
		pw.close();
	}
	
	static class MyScanner {
		BufferedReader br;
		StringTokenizer st;
		public MyScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}
		String next() {while (st == null || !st.hasMoreElements()) {
			try {st = new StringTokenizer(br.readLine());}
			catch (IOException e) {e.printStackTrace();}}
		return st.nextToken();}
		int nextInt() {return Integer.parseInt(next());}
		long nextLong() {return Long.parseLong(next());}
		double nextDouble() {return Double.parseDouble(next());}
		String nextLine(){String str = "";
		try {str = br.readLine();}
		catch (IOException e) {e.printStackTrace();}
		return str;}
	}
}
