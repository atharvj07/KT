for _ in range(int(input())):
	(n, k) = map(int, input().split())
	f = 0
	if n % 2 == 1:
		if k > n // 2:
			f = 1
	elif k + 1 > n // 2:
		f = 1
	l = [0] * n
	t = n
	for i in range(1, n, 2):
		k -= 1
		if k < 0:
			break
		l[i] = t
		t -= 1
	b = 1
	for i in range(0, n):
		if l[i] == 0:
			l[i] = b
			b += 1
	if f == 1:
		print(-1)
	else:
		print(*l)
