namespace my {
template <class T>
T gcd(T a, T b) {
  return b == 0 ? a : gcd(b, a % b);
}
template <typename T>
T lcm(T a, T b) {
  return a / gcd(a, b) * b;
}
template <class T>
T big_mod(T n, T p, T m) {
  if (p == 0) return (T)1;
  T x = big_mod(n, p / 2, m);
  x = (x * x) % m;
  if (p & 1) x = (x * n) % m;
  return x;
}
template <class T>
T multiplication(T n, T p, T m) {
  if (p == 0) return (T)0;
  T x = multiplication(n, p / 2, m);
  x = (x + x) % m;
  if (p & 1) x = (x + n) % m;
  return x;
}
template <class T>
T my_pow(T n, T p) {
  if (p == 0) return 1;
  T x = my_pow(n, p / 2);
  x = (x * x);
  if (p & 1) x = (x * n);
  return x;
}
template <class T>
double getdist(T a, T b) {
  return sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y));
}

template <class T>

template <class T>
void print_vector(T &v) {
  int sz = v.size();
  if (sz) cout << v[0];
  for (int i = 1; i < sz; i++) cout << ' ' << v[i];
  cout << endl;
}
bool isVowel(char ch) {
  ch = toupper(ch);
  if (ch == 'A' || ch == 'U' || ch == 'I' || ch == 'O' || ch == 'E')
    return true;
  return false;
}
bool isConsonant(char ch) {
  if (isalpha(ch) && !isVowel(ch)) return true;
  return false;
}

bool Child[4005];
long long int v[4005], d[4005], p[4005];
int main() {
  int N;
  cin >> N;
  for (int i = 1; i <= N; i++) {
    Child[i] = true;
    cin >> v[i] >> d[i] >> p[i];
  }
  vector<int> ans;
  for (int i = 1; i <= N; i++) {
    if (Child[i]) {
      ans.push_back(i);
      int cry = v[i];
      for (int j = i + 1; j <= N; j++) {
        if (Child[j] and cry) {
          p[j] -= cry;
          cry--;
        }
      }
      long long int Reduce = 0;
      for (int j = i + 1; j <= N; j++) {
        if (Child[j]) {
          p[j] -= Reduce;
          if (p[j] < 0) {
            Reduce += d[j];
            Child[j] = false;
          }
        }
      }
    }
  }
  cout << ans.size() << endl;
  my::print_vector(ans);
  return 0;
}
