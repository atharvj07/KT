

import java.io.*;
import java.util.*;
import java.lang.*;

public class Temp{  
    static PrintWriter w=new PrintWriter(System.out);
    
    
    public static void main (String[] args)throws IOException{
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            int n=sc.nextInt();
            int k=sc.nextInt();
            boolean f=false;
            int arr[]=new int[n+3];
            Arrays.fill(arr,-1);
            for(int i=0;i<n;i++){
                arr[i]=sc.nextInt();
                if(arr[i]==k)f=true;
            }
            if(!f)w.println("NO");
            else if(n==1){
                w.println("YES");
            }
            else{
                
                f=false;
                for(int i=0;i<n;i++){
                    if(arr[i]>=k){
                        if(arr[i+1]>=k||arr[i+2]>=k)f=true;
                    }
                }
                w.println(f?"YES":"NO");
            }
            
        }
        w.close();
    }
}
