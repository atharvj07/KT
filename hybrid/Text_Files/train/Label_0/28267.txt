#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  int v;
  cin >> v;
  int res = v;
  int vol = 0;
  if (v >= n) {
    cout << n - 1;
    return 0;
  }
  if (n == v + 1) {
    cout << v;
    return 0;
  }
  for (int i = 2; i <= n - v; i++) {
    res += i;
  }
  cout << res;
}
