ans = []

def main():
	t = int(input())
	for l in range(t):
		n = int(input())
		if n == 4:
			ans.append(input())
		else:
			b = y = num = x = 0
			d = 10 ** 10
			for i in sorted(map(int, input().split())):
				if num == i:
					c += 1
					if c == 2:
						a = b
						b = i
					if a != 0 and a / b + b / a < d:
						d = a / b + b / a
						x = a
						y = b
					elif c == 4:
						x = y = i
						break
				else:
					num = i
					c = 1
			ans.append(str(x) + ' ' + str(y) + ' ' + str(x) + ' ' + str(y))
	return ans
print('\n'.join(main()))
