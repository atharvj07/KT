#include <bits/stdc++.h>
using namespace std;
signed main() {
  long long int n, ans = 0;
  cin >> n;
  long long int a[n];
  for (long long int i = 0; i < n; i++) cin >> a[i];
  for (long long int i = 1; i < n - 1; i++) {
    if (a[i] == 0 && a[i - 1] == 1 && a[i + 1] == 1) {
      a[i + 1] = 0;
      ans++;
    }
  }
  cout << ans << endl;
}
