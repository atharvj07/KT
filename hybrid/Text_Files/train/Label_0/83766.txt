(m, s) = map(int, input().split())
if s == 0 and m != 1:
	print(-1, -1)
elif s > m * 9:
	print(-1, -1)
elif m == 1:
	print(s, s)
elif s < 10:
	print(1 * 10 ** (m - 1) + (s - 1), s * 10 ** (m - 1))
else:
	(number, number2) = (0, 0)
	a = (s - 1) // 9
	b = s // 9
	for i in range(a):
		number = number + 9 * 10 ** i
	if m - a > 1:
		number += 1 * 10 ** (m - 1) + (s - a * 9 - 1) * 10 ** a
	else:
		number += (s - a * 9) * 10 ** (m - 1)
	for i in range(1, b + 1):
		number2 += 9 * 10 ** (m - i)
	number2 += s % 9 * 10 ** max(0, m - b - 1)
	print(number, number2)
