import sys
input = sys.stdin.readline

def solve(a: list):
	ret = 1
	cnt = 1
	for i in range(1, n + 1):
		if i == n or a[i] != a[i - 1]:
			ret = ret * factor[cnt] % mod
			cnt = 1
		else:
			cnt += 1
	return ret
M = 1000000
n = int(input())
(a, x, y) = ([], [], [])
for i in range(n):
	(l, r) = map(int, input().split())
	x.append(l)
	y.append(r)
	a.append(M * l + r)
mod = 998244353
factor = [1] * (n + 1)
for i in range(2, n + 1):
	factor[i] = factor[i - 1] * i % mod
x.sort()
y.sort()
bad = solve(x)
bad += solve(y)
d = 1
cnt = 1
a.sort()
for i in range(1, n + 1):
	if i < n:
		(pl, pr) = (a[i - 1] // M, a[i - 1] % M)
		(l, r) = (a[i] // M, a[i] % M)
		if r < pr:
			d = 0
			break
	if i == n or a[i] != a[i - 1]:
		d = d * factor[cnt] % mod
		cnt = 1
	else:
		cnt += 1
bad = (bad - d + mod) % mod
tot = factor[n]
ans = (tot - bad + mod) % mod
print(ans)
