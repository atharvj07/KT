#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;


#define NUM 100005
typedef pair<int,int> P;

int N,M;
int POW[31];
int table[NUM],input_rank[NUM];
map<P,int> dp[31];

int recursive(int left,int right,int range_winner_final_input_rank,int size){

	if(range_winner_final_input_rank == size && dp[size].find(P(left,right)) != dp[size].end()){
		return dp[size][P(left,right)];
	}

	int L = 0,R = M,mid = (L+R)/2;
	int index = 0;

	while(L <= R){

		if(table[mid] <= left){

			index = mid;
			L = mid+1;

		}else{
			R = mid-1;
		}
		mid = (L+R)/2;
	}

	int ret = BIG_NUM;

	if(index+1 <= M && table[index] <= left && table[index+1] >= right){

		if(input_rank[index] >= size+1){

			ret = POW[N-size]-POW[N-size]/POW[N-input_rank[index]+1];

		}else{

			ret = POW[N-size];

			if(input_rank[index] == range_winner_final_input_rank){
				ret--;
			}
		}

	}else{

		mid = (left+right)/2;
		ret = min(recursive(left,mid,range_winner_final_input_rank,size+1)+recursive(mid,right,size+1,size+1),
 				recursive(left,mid,size+1,size+1)+recursive(mid,right,range_winner_final_input_rank,size+1));
	}

	if(range_winner_final_input_rank == size){

		return dp[size][P(left,right)] = ret;

	}else{
		return ret;
	}
}

int main(){

	for(int i = 0; i < 31; i++)POW[i] = pow(2,i);

	scanf("%d %d",&N,&M);

	for(int i = 0; i <= M; i++){
		scanf("%d",&table[i]);
	}

	for(int i = 0; i < M; i++){
		scanf("%d",&input_rank[i]);
	}

	printf("%d\n",recursive(0,POW[N],0,0));

	return 0;
}
