def main():
	n = int(input())
	t = []
	for i in range(n):
		t.append([elem for elem in input().split()])
	p = [int(num) - 1 for num in input().split()]
	last = min(t[p[0]])
	for i in range(1, n):
		if last < t[p[i]][0] and last < t[p[i]][1]:
			last = min(t[p[i]])
		elif last < t[p[i]][0] or last < t[p[i]][1]:
			if last < t[p[i]][0]:
				last = t[p[i]][0]
			elif last < t[p[i]][1]:
				last = t[p[i]][1]
		else:
			print('NO')
			return 0
	print('YES')
main()
