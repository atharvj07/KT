t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	w = list(map(int, input().split()))
	E = 0
	d = 0
	arrow = k - 1
	a.sort(reverse=True)
	w.sort()
	for i in range(k):
		E += a[i]
	for i in range(k):
		if w[i] == 1:
			E += a[d]
			d += 1
		else:
			arrow += w[i] - 1
			E += a[arrow]
	print(E)
