import sys
input = sys.stdin.readline
from collections import Counter
for _ in range(int(input())):
	n = int(input())
	A = sorted(map(int, input().split()))
	for k in A:
		if k % 2:
			print('NO')
			break
	else:
		A = [k // 2 for k in A]
		cnt = Counter(A)
		keys = sorted(cnt.keys())
		m = len(keys)
		if m == 1:
			k = keys[0]
			if cnt[k] not in [2, 1]:
				print('NO')
			else:
				print('YES')
			continue
		for i in keys:
			if not i:
				if cnt[i] not in [2, 1]:
					print('NO')
					break
			elif cnt[i] != 2:
				print('NO')
				break
		else:
			pre = 0
			for i in range(m - 1, -1, -1):
				cur = keys[i] - pre
				if cur <= 0:
					print('NO')
					break
				(k, r) = divmod(cur, i + 1)
				if r:
					print('NO')
					break
				pre += k
			else:
				print('YES')
