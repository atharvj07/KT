from functools import reduce
from collections import Counter
(a, b) = map(int, input().split())
if a == b:
	print('0')
	exit()
(a1, a2, a3) = (0, 0, 0)
k1 = a
f = 0
while f != -1:
	if k1 % 2 == 0:
		k1 /= 2
		a1 += 1
	elif k1 % 3 == 0:
		k1 /= 3
		a2 += 1
	elif k1 % 5 == 0:
		k1 /= 5
		a3 += 1
	else:
		f = -1
k1a = [a1, a2, a3]
(a1, a2, a3) = (0, 0, 0)
old = k1
k1 = b
f = 0
while f != -1:
	if k1 % 2 == 0:
		k1 /= 2
		a1 += 1
	elif k1 % 3 == 0:
		k1 /= 3
		a2 += 1
	elif k1 % 5 == 0:
		k1 /= 5
		a3 += 1
	else:
		f = -1
if k1 != old:
	print('-1')
	exit()
k1b = [a1, a2, a3]
t = 0
for i in range(3):
	t += abs(k1a[i] - k1b[i])
print(t)
