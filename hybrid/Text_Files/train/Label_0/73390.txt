for t in range(int(input())):
	n = int(input())
	p = [int(s) for s in input().rstrip().split()]
	vote = input().rstrip()
	ls = []
	for i in range(n):
		ls.append([vote[i], p[i], i])
	ls.sort(key=lambda v: (v[0], v[1]))
	for i in range(n):
		ls[i].append(i + 1)
	ls.sort(key=lambda v: v[2])
	print(' '.join([str(v[-1]) for v in ls]))
