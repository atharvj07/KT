#include <bits/stdc++.h>
using namespace std;
int main() {
  long int a, b, n;
  cin >> a >> b >> n;
  long int temp = a;
  bool flag = false;
  for (int i = 0; i <= 9; i++) {
    temp = a;
    temp = (temp * 10) + i;
    if (temp % b == 0) {
      flag = true;
      break;
    }
  }
  if (!flag) {
    cout << -1 << endl;
  } else {
    n = n - 1;
    if (n > 0) {
      cout << temp;
      for (long int i = 0; i < n; i++) cout << 0;
      cout << endl;
    } else {
      cout << temp << endl;
    }
  }
  return 0;
}
