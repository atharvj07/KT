for _ in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	mc = max(l)
	if mc == 1:
		ans = 1
	else:
		hasone = 1 in l
		extras = set((num % 3 for num in l if num != mc))
		extras.discard(0)
		(q, r) = divmod(mc, 3)
		closecall = any((mc > num >= 3 * q for num in l))
		if len(extras) == 0:
			ans = q + (not not r)
		elif len(extras) == 1:
			if r in extras:
				ans = q + 1
			elif r == 1:
				if closecall:
					ans = q + 2
				else:
					ans = q + 1
			elif r == 2:
				ans = q + 2
			else:
				ans = q + 1
		elif r == 2:
			ans = q + 2
		elif r == 1:
			if hasone:
				ans = q + 2
			elif closecall:
				ans = q + 2
			else:
				ans = q + 1
		else:
			ans = q + 1
	print(ans)
