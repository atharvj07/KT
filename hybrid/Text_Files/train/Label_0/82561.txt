q = int(input())
for _ in range(q):
	n = int(input())
	s = []
	ava = 0
	req = 0
	ans = 0
	for i in range(n):
		s = input()
		ones = 0
		zeros = 0
		for x in s:
			if x == '1':
				ones += 1
			else:
				zeros += 1
		if len(s) % 2 == 0:
			if ones % 2 != 0:
				req += 1
			else:
				ans += 1
		else:
			ava += 1
			ans += 1
	ans += 2 * (req // 2)
	req = req % 2
	if ava != 0 and req != 0:
		ans += 1
	print(ans)
