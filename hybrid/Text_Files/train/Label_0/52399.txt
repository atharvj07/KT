import java.math.*;
import java.util.*;
import java.io.*;

public class D {
	static BufferedReader in;
	static StringTokenizer st;

	static String next() throws IOException {
		while (st == null || !st.hasMoreTokens())
			st = new StringTokenizer(in.readLine());
		return st.nextToken();
	}
	static int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	static long nextLong() throws IOException {
		return Long.parseLong(next());
	}

	static double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	static int n;
	static int d[] = new int[300];
	static long pow[][] = new long[300][300];
	static long p[] = new long[300];
	static long V, maxA, maxB, S, ansA, ansB, ansC;

	static class Answer {
		long S, A, B, C;
		Answer() {

		}
		Answer(long S, long A, long B, long C) {
			this.S = S; this.A = A; this.B = B; this.C = C;
		}
	};

	static Map<Long, Answer> memo = new HashMap<Long, Answer>();

	static void Calc2(int cur, long A, long B) {
		if (B > maxB) return;

		if (cur == n) {
			long C = V / (A * B);
			if (S > A * B + B * C + C * A) {
				S = A * B + B * C + C * A;
				ansA = A; ansB = B; ansC = C;
			}
			return;
		}

		for (int i = d[cur]; i >= 0; --i) {
			d[cur] -= i;
			Calc2(cur + 1, A, B * pow[cur][i]);
			d[cur] += i;
		}
	}
	static void Calc(int cur, long A) {
		if (A > maxA) return;

		if (cur == n) {
			double B = Math.pow(V / A, 0.5);
			if ((double)S + 1e-9 <= A * B + B * B + B * A) return;
			maxB = (long)(B + 1e-5);
			Calc2(0, A, 1);
			return;
		}

		for (int i = d[cur]; i >= 0; --i) {
			d[cur] -= i;
			Calc(cur + 1, A * pow[cur][i]);
			d[cur] += i;
		}		
	}
	public static void main(String args[]) throws IOException {
		in = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(System.out);

		for (int test = nextInt(); test > 0; --test) {
			n = nextInt();
			V = 1;
			for (int i = 0; i < n; ++i) {
				p[i] = nextLong();
				d[i] = nextInt();
				pow[i][0] = 1;
				for (int j = 1; j <=  d[i]; ++j)
					pow[i][j] = pow[i][j - 1] * p[i];
				
				V *= pow[i][d[i]];
			}
			
			maxA = (long)(Math.pow(V, 1.0/3.0) + 1e-5);
			S = Long.MAX_VALUE;
			if (memo.containsKey(V)) {
				Answer tmp = memo.get(V);
				out.println(tmp.S + " " + tmp.A + " " + tmp.B + " " + tmp.C);
			} else {
				Calc(0, 1);
				memo.put(V,new Answer(2 * S, ansA, ansB, ansC));
				out.println((2 * S) + " " + ansA + " " + ansB + " " + ansC);
			}
		}
		out.close();
    }


}
