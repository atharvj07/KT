#include <bits/stdc++.h>
using namespace std;
long long int a[1010];
long long int gcd(long long int a, long long int b) {
  if (a % b == 0) return b;
  if (a == 0) return b;
  return gcd(b, a % b);
}
int main() {
  long long int n;
  vector<long long int> v;
  scanf("%lld", &n);
  ;
  for (long long int i = 0; i < n; i++) scanf("%lld", &a[i]);
  ;
  long long int mo = 0;
  for (long long int i = 0; i < n; i++) {
    if (i + 1 == n) break;
    v.push_back(a[i]);
    long long int x = a[i], y = a[i + 1];
    if (x > y) swap(x, y);
    if (gcd(y, x) != 1) mo++, v.push_back(1);
  }
  v.push_back(a[n - 1]);
  printf("%lld\n", mo);
  for (int i = 0; i < v.size(); ++i) printf("%lld ", v[i]);
  printf("\n");
  return 0;
}
