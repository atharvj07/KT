#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 109;
int as[MAX_N];
int N;
long long solve() {
  long long ans = 0;
  for (int i = 0; i < N; i++) {
    ans += (long long)(as[i] - 1) * (i + 1);
  }
  ans += N;
  return ans;
}
int main() {
  scanf("%d", &N);
  for (int i = 0; i < N; i++) {
    scanf("%d", as + i);
  }
  printf("%I64d\n", solve());
  return 0;
}
