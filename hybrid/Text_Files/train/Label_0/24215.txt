#include <bits/stdc++.h>
using namespace std;
const long long INF = 2000000000000000000LL;
const int inf = 0x3f3f3f3f;
const long double EPS = 1e-9;
const long long MOD = 1e9 + 7;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  int t, tc = 0;
  cin >> t;
  while (t--) {
    int x;
    cin >> x;
    string s;
    cin >> s;
    long long len = (int)s.size();
    for (int i = 1; i <= x; ++i) {
      int d = s[i - 1] - '0';
      --d;
      long long rem = (len - i + MOD) % MOD;
      len += 1LL * rem * d;
      len %= MOD;
      if ((int)s.size() < x) {
        string sub = s.substr(i);
        while (d-- and (int) s.size() < x) {
          s += sub;
        }
      }
    }
    cout << len << "\n";
  }
  return 0;
}
