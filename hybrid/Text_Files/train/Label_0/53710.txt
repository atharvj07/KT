    private static long modexp(long a, long b, long m)
    {
        long k = 1;
        while (b > 0)
        {
            if ((b & 1) == 1)
            {
                k = modm(a, k, m);
            }
            b = b >> 1;
            a = modm(a, a, m);
        }
        return k;
    }
    
    private static long modinv(long a, long m)
    {
        return modexp(a, m-2, m);
    }
    
    private static long moda(long a, long b, long m)
    {
        return (a+b) % m;
    }
    
    private static long modm(long a, long b, long m)
    {
        return ((a % m)*(b % m)) % m;
    }
    
    public static long[] modfact;
    
    private static void genModFact(int n, long m)
    {
        modfact = new long[n+1];
        modfact[0] = modfact[1] = 1;
        for (int i = 2; i < n+1; i++)
        {
            modfact[i] = modm(modfact[i-1], i, m);
        }
    }
    
    private static long modchoose(int n, int k, long m)
    {
        long denom = modm(modfact[k], modfact[n-k], m);
        return modm(modfact[n], modinv(denom, m), m);
    }
