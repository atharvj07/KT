	static long getinv(long x){
		long res = 1,y = mod-2;
		while(y > 0){
			if( (y&1) > 0)
				res = (res*x)%mod;
			x=(x*x)%mod;
			y>>=1;
		}
		return res;
	}
    public static long[] generateFactorials(int count) {
        long[] result = new long[count];
        inv = new long[400010];
        result[0] = 1;
        inv[0] = 1;
        for (int i = 1; i < count; i++) {
            result[i] = (result[i - 1] * i) % mod;
            inv[i] =getinv(result[i]);
        }
        return result;
    }
