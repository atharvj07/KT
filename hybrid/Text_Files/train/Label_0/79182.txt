n = int(input())
for i in range(n):
	k = int(input())
	arr = [[] for i in range(k + 1)]
	prince = [0 for i in range(k + 1)]
	girl = [0 for i in range(k + 1)]
	for j in range(1, k + 1):
		arr[j] = list(map(int, input().split()))
	for i1 in range(1, k + 1):
		look = 1
		for j1 in range(1, len(arr[i1])):
			if prince[arr[i1][j1]] == 0 and look == 1:
				prince[arr[i1][j1]] = 1
				girl[i1] = 1
				look = 0
	imp = 0
	for i1 in range(1, k + 1):
		if prince[i1] == 0 and imp == 0:
			for j1 in range(1, k + 1):
				if girl[j1] == 0 and imp == 0:
					print('IMPROVE')
					print(j1, i1)
					imp = 1
	if imp == 0:
		print('OPTIMAL')
