#include <iostream>
#include <string>
using namespace std;
int main(){
	int n;
	cin >> n;
	while(n--){
		int num[9], red[10] = { 0 }, green[10] = { 0 }, blue[10] = { 0 };
		for(int i = 0; i < 9; ++i){ cin >> num[i]; }
		for(int i = 0; i < 9; ++i){
			string str;
			cin >> str;
			if(str == "R"){
				++red[num[i]];
			}else if(str == "G"){
				++green[num[i]];
			}else if(str == "B"){
				++blue[num[i]];
			}
		}
		int answer = 1;
		for(int i = 1; i <= 7; ++i){
			while(red[i] > 0 && red[i + 1] > 0 && red[i + 2] > 0){
				--red[i]; --red[i + 1]; --red[i + 2];
			}
			while(green[i] > 0 && green[i + 1] > 0 && green[i + 2] > 0){
				--green[i]; --green[i + 1]; --green[i + 2];
			}
			while(blue[i] > 0 && blue[i + 1] > 0 && blue[i + 2] > 0){
				--blue[i]; --blue[i + 1]; --blue[i + 2];
			}
		}
		for(int i = 1; i <= 9; ++i){
			if(red[i] % 3 != 0){ answer = 0; break; }
			if(green[i] % 3 != 0){ answer = 0; break; }
			if(blue[i] % 3 != 0){ answer = 0; break; }
		}
		cout << answer << endl;
	}
	return 0;
}
