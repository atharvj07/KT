from collections import deque

def organize(inp):
	name = inp[0]
	records = deque(inp[1:])
	records_copy = deque(inp[1:])
	found = True
	while found:
		removed = False
		for l in records:
			to_remove = None
			for k in records_copy:
				index = min(len(k), len(l))
				if k[-index:] == l and k != l:
					to_remove = l
					break
			if to_remove and to_remove in records_copy:
				records_copy.remove(to_remove)
				removed = True
		if not removed:
			found = False
	final_res = []
	for l in records_copy:
		if l not in final_res:
			final_res.append(l)
	final_res = [name, str(len(final_res))] + final_res
	print(' '.join(final_res))
	return

def solution():
	n = int(input())
	d = {}
	for l in range(n):
		inp = [a for a in input().split(' ')]
		name = inp[0]
		record = inp[2:]
		if name in d:
			d[name] += record
		else:
			d[name] = record
	print(len(d))
	for k in d:
		organize([k] + d[k])
solution()
