import java.util.*;

public class CF_535_B {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        Arrays.sort(arr);
        int max = arr[n - 1];
        Set<Integer> dividers = new HashSet<>();
        for (int i = 0; i < n - 1; i++) {
            if (!dividers.contains(arr[i]) && max % arr[i] == 0 && max > arr[i]) {
                dividers.add(arr[i]);
                arr[i] = 0;
            }
        }
        Arrays.sort(arr);
        System.out.println(max + " " + arr[n - 2]);

    }

}
