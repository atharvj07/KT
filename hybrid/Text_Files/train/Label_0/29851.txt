#include <bits/stdc++.h>
using namespace std;

const int mod = 1000000007;

int main()
{
    int N;
    string S, T;
    cin >> N >> S >> T;

    vector<bool> f;
    for (int i = 0; i < S.size(); i++) {
        if (S[i] == T[i]) {
            f.push_back(true);
        }
        else {
            f.push_back(false);
            i++;
        }
    }

    int ans = f[0] ? 3 : 6;
    for (int i = 1; i < f.size(); i++) {
        if (!f[i - 1] && f[i]);
        else if (!f[i - 1] && !f[i]) ans = 3LL * ans % mod;
        else ans = 2LL * ans % mod;
    }
    cout << ans << endl;

    return 0;
}
