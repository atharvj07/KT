from sys import stdin, stdout, setrecursionlimit
input = stdin.readline
dir = [-1, 0, 1, 0, -1]
inf = float('inf')
from functools import cmp_to_key
from collections import defaultdict as dd
from collections import Counter, deque
from heapq import *
import math
from math import floor, ceil, sqrt

def geti():
	return map(int, input().strip().split())

def getl():
	return list(map(int, input().strip().split()))

def getis():
	return map(str, input().strip().split())

def getls():
	return list(map(str, input().strip().split()))

def gets():
	return input().strip()

def geta():
	return int(input())

def print_s(s):
	stdout.write(s + '\n')

def solve():
	(n, m) = geti()
	mat = []
	vis = [[False] * m for _ in range(n)]
	for i in range(n):
		mat.append(gets())
	q = deque()
	start = getl()
	end = getl()
	start[0] -= 1
	start[1] -= 1
	end[0] -= 1
	end[1] -= 1
	have = 0
	ok = False
	for d in range(4):
		x = end[0] + dir[d]
		y = end[1] + dir[d + 1]
		if [x, y] == start:
			ok = True
		if 0 <= x < n and 0 <= y < m and (mat[x][y] != 'X'):
			have += 1
	if start == end:
		if have > 0:
			print('YES')
		else:
			print('NO')
		return
	q.append(start)
	while q:
		(i, j) = q.pop()
		if vis[i][j]:
			continue
		vis[i][j] = True
		if [i, j] == end:
			if have >= 2:
				print('YES')
			elif have == 1 and ok:
				print('YES')
			else:
				print('NO')
			return
		for d in range(4):
			x = i + dir[d]
			y = j + dir[d + 1]
			if 0 <= x < n and 0 <= y < m and (not vis[x][y]):
				if mat[x][y] != 'X':
					q.append((x, y))
				elif [x, y] == end:
					print('YES')
					return
	print('NO')
solve()
