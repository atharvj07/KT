#include <bits/stdc++.h>
using namespace std;
const size_t M = 1e9 + 7;
size_t getHash(const vector<int>& v) {
  size_t p = 1, result = 0;
  for (int x : v) {
    result += p * (x + 1);
    p *= 163841;
    result %= M;
    p %= M;
  }
  return result;
}
vector<int> z_function(vector<size_t> v) {
  int n = (int)v.size();
  vector<int> z(n);
  for (int i = 1, l = 0, r = 0; i < n; ++i) {
    if (i <= r) z[i] = min(r - i + 1, z[i - l]);
    while (i + z[i] < n && v[z[i]] == v[i + z[i]]) ++z[i];
    if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
  }
  return z;
}
int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> h;
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    h.push_back({x - 1, y > x ? y - x : n + y - x});
    h.push_back({y - 1, x > y ? x - y : n + x - y});
  }
  sort(h.begin(), h.end());
  vector<vector<int>> hh(n);
  for (auto& p : h) {
    hh[p.first].push_back(p.second);
  }
  vector<size_t> c(n * 2);
  for (int i = 0; i < 2 * n; i++) {
    if (i < n) {
      c[i] = getHash(hh[i]);
    } else {
      c[i] = c[i - n];
    }
  }
  auto z = z_function(c);
  bool ans = false;
  for (int i = 1; i < n; i++) {
    if (z[i] >= n) {
      ans = true;
    }
  }
  cout << (ans ? "Yes" : "No") << endl;
  return 0;
}
