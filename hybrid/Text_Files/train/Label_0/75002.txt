import io, os
input = io.StringIO(os.read(0, os.fstat(0).st_size).decode()).readline
ii = lambda : int(input())
mi = lambda : map(int, input().split())
li = lambda : list(mi())
n = ii()
a = li()
j = n - 1
while a[j] == a[0]:
	j -= 1
i = 0
while a[i] == a[-1]:
	i += 1
ans = max(j, n - 1 - i)
print(ans)
