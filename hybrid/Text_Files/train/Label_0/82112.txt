from math import gcd
import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
	n = int(input())
	data = list(map(int, input().split()))
	accum = 1
	flg = True
	for (i, t) in enumerate(data):
		accum = accum // gcd(i + 2, accum) * (i + 2)
		if accum > int(1000000000.0):
			break
		if t % accum == 0:
			flg = False
			break
	if flg:
		print('YES')
	else:
		print('NO')
