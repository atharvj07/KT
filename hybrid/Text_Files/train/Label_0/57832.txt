import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;
import java.util.ArrayList;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        ThreePieces solver = new ThreePieces();
        solver.solve(1, in, out);
        out.close();
    }

    static class ThreePieces {
        int[][][][][] dp = new int[201][10][10][100][3];
        int INF = (int) (1e9);
        ArrayList<int[]>[][][] move = new ArrayList[10][10][3];
        int N;
        int[] moveR = new int[]{-2, -1, 1, 2};
        int[] moveC = new int[]{-2, -1, 1, 2};

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            N = in.nextInt();
            int[][] loc = new int[N * N][2];
            int[][] arr = new int[N][N];
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < N; j++) {
                    int a = in.nextInt() - 1;
                    loc[a] = new int[]{i, j};
                    arr[i][j] = a;
                    move[i][j][0] = posKnight(i, j);
                    move[i][j][1] = posBishop(i, j);
                    move[i][j][2] = posRook(i, j);
                }
            }
            for (int i = 0; i <= 200; i++) {
                for (int j = 0; j < 10; j++) {
                    for (int k = 0; k < 10; k++) {
                        for (int d = 0; d < 100; d++) {
                            for (int z = 0; z < 3; z++) {
                                dp[i][j][k][d][z] = INF;
                            }
                        }
                    }
                }
            }
            Arrays.fill(dp[0][loc[0][0]][loc[0][1]][0], 0);
            for (int i = 0; i < 200; i++) {
                for (int j = 0; j < N; j++) {
                    for (int k = 0; k < N; k++) {
                        for (int d = 0; d < N * N; d++) {
                            for (int m = 0; m < 3; m++) {
                                for (int next = 0; next < 3; next++) {
                                    if (next == m) {
                                        for (int[] c : move[j][k][next]) {
                                            if (arr[c[0]][c[1]] == d + 1) {
                                                dp[i + 1][c[0]][c[1]][d + 1][next] = Math.min(dp[i + 1][c[0]][c[1]][d + 1][next], dp[i][j][k][d][m]);
                                            } else {
                                                dp[i + 1][c[0]][c[1]][d][next] = Math.min(dp[i + 1][c[0]][c[1]][d][next], dp[i][j][k][d][m]);
                                            }
                                        }
                                    } else {
                                        dp[i + 1][j][k][d][next] = Math.min(dp[i + 1][j][k][d][next], dp[i][j][k][d][m] + 1);
                                    }
                                }
                            }
                        }
                    }
                }
            }
            for (int i = 2; i <= 200; i++) {
                if (has(dp[i][loc[N * N - 1][0]][loc[N * N - 1][1]][N * N - 1])) {
                    int res = INF;
                    for (int v : dp[i][loc[N * N - 1][0]][loc[N * N - 1][1]][N * N - 1]) {
                        res = Math.min(res, v);
                    }
                    out.println(i + " " + res);
                    return;
                }
            }
        }

        boolean has(int[] a) {
            for (int i : a) {
                if (i != INF) {
                    return true;
                }
            }
            return false;
        }

        boolean isValid(int r, int c) {
            return r >= 0 && r < N && c >= 0 && c < N;
        }

        ArrayList<int[]> posKnight(int r, int c) {
            ArrayList<int[]> pos = new ArrayList<>();
            for (int rM : moveR) {
                for (int cM : moveC) {
                    if (Math.abs(rM) != Math.abs(cM)) {
                        int newR = r + rM;
                        int newC = c + cM;
                        if (isValid(newR, newC)) {
                            pos.add(new int[]{newR, newC});
                        }
                    }
                }
            }
            return pos;
        }

        ArrayList<int[]> posRook(int r, int c) {
            ArrayList<int[]> pos = new ArrayList<>();
            for (int i = 0; i < N; i++) {
                pos.add(new int[]{r, i});
                pos.add(new int[]{i, c});
            }
            return pos;
        }

        ArrayList<int[]> posBishop(int r, int c) {
            ArrayList<int[]> pos = new ArrayList<>();
            for (int k = 1; k < N; k++) {
                int[] mR = new int[]{k, -k, k, -k};
                int[] mC = new int[]{k, k, -k, -k};
                for (int i = 0; i < 4; i++) {
                    int newR = r + mR[i];
                    int newC = c + mC[i];
                    if (isValid(newR, newC)) {
                        pos.add(new int[]{newR, newC});
                    }
                }
            }
            return pos;
        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
