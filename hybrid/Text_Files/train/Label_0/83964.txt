t = int(input())
while t > 0:
	x = input()
	y = x.split()
	n = int(y[0])
	a = int(y[1])
	b = int(y[2])
	flag = False
	if a == 1:
		if (n - 1) % b == 0:
			print('Yes')
		else:
			print('No')
	else:
		k = 1
		while k <= n:
			if k % b == n % b:
				flag = True
				break
			k = k * a
		if flag:
			print('Yes')
		else:
			print('No')
	t -= 1
