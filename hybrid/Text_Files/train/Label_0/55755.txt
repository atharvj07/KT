import java.util.*;
import java.io.*;

public class MadhouseEasy {
    public static void main(String[] args) throws IOException{
        BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        int n = Integer.parseInt(f.readLine());
        HashMap<String, Integer> strs = new HashMap<String, Integer>();
        String[] arr = new String[n+1];
        if(n == 1){
            out.println("? 1 " + n);
            out.flush();
            out.println("! " + f.readLine());
            out.close();
            System.exit(0);
        }
        out.println("? 1 " + n);
        out.flush();
        for(int i = 0; i < ((n+1)*(n))/2; i++){
            String curr = f.readLine();
            Character[] cur = new Character[curr.length()];
            for(int j = 0; j < curr.length(); j++) cur[j] = curr.charAt(j);
            Arrays.sort(cur);
            StringBuilder ans = new StringBuilder();
            for(int j = 0; j < curr.length(); j++) ans.append((char)cur[j]);
            update(strs, ans.toString(), 1);
        }
        out.println("? 2 " + n);
        out.flush();
        for(int i = 0; i < ((n)*(n-1))/2; i++){
            String curr = f.readLine();
            Character[] cur = new Character[curr.length()];
            for(int j = 0; j < curr.length(); j++) cur[j] = curr.charAt(j);
            Arrays.sort(cur);
            StringBuilder ans = new StringBuilder();
            for(int j = 0; j < curr.length(); j++) ans.append((char)cur[j]);
            update(strs, ans.toString(), -1);
        }
        for(String nstr: strs.keySet()){
            arr[nstr.length()] = nstr;
        }
        int[] count = new int[26];
        StringBuilder ans = new StringBuilder();
        for(int i = 1; i <= n; i++){
            int[] ncount = Arrays.copyOf(count, 26);
            for(int j = 0; j < arr[i].length(); j++){
                ncount[arr[i].charAt(j)-'a']--;
                if(ncount[arr[i].charAt(j)-'a'] < 0){
                    ans.append(arr[i].charAt(j));
                    count[arr[i].charAt(j)-'a']++;
                    break;
                }
            }
        }
        out.println("! " + ans);


        out.close();

    }
    public static void update(HashMap<String, Integer> strs, String nstr, int i){
        if(i > 0){
            if(!strs.containsKey(nstr)) strs.put(nstr, 1);
            else strs.put(nstr, strs.get(nstr)+1);
        }else{
            strs.put(nstr, strs.get(nstr)-1);
            if(strs.get(nstr) == 0) strs.remove(nstr);
        }
    }
    static class newString{
        public String str;
        public static long BASE = 256;
        public static long MOD = 998244853;
        public static long hash;

        public newString(String str){
            this.str = str;
            long ans = 0;
            long currMult = 1;
            for(int i = 0; i < str.length(); i++){
                ans+=((((int)str.charAt(i))*currMult)%MOD);
                ans%=MOD;
                currMult*=BASE;
                currMult%=MOD;
            }
            hash = ans;
        }

        @Override
        public int hashCode(){
            return (int)hash;
        }

        @Override
        public boolean equals(Object obj){
            if(this == obj) return true;
            if(!(obj instanceof newString)) return false;

            newString nstr = (newString)obj;

            if(this.str.length() != nstr.str.length()) return false;
            for(int i = 0; i < nstr.str.length(); i++){
                if(this.str.charAt(i) != nstr.str.charAt(i)) return false;
            }
            return true;
        }
    }
}
