#include <bits/stdc++.h>
using namespace std;
string a, b, s;
int main() {
  cin >> a >> b;
  if (a.size() < b.size()) {
    int q = b.size() - a.size();
    for (int i = 0; i < q; i++) s += '0';
    a = s + a;
  } else {
    int q = a.size() - b.size();
    for (int i = 0; i < q; i++) s += '0';
    b = s + b;
  }
  for (int i = 0; i < a.size(); i++)
    if (a[i] != b[i]) {
      if (a[i] < b[i]) {
        cout << '<';
        return 0;
      } else {
        cout << '>';
        return 0;
      }
    }
  cout << "=";
  return 0;
}
