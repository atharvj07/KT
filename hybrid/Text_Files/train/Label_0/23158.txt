#include <bits/stdc++.h>
using namespace std;
bool check(vector<char> v1, vector<char> v2, int x) {
  for (int i = x, j = 0; i < x + 3 and j < 3; i++, j++) {
    if (v1[i % 3] != v2[j]) return 0;
  }
  return 1;
}
int main() {
  string ss, ss1, temp;
  vector<char> s1, s2;
  s1.resize(3);
  s2.resize(3);
  cin >> ss;
  cin >> temp;
  reverse(temp.begin(), temp.end());
  ss += temp;
  cin >> ss1;
  cin >> temp;
  reverse(temp.begin(), temp.end());
  ss1 += temp;
  for (int i = 0, j = 0; i < 4; i++) {
    if (ss[i] != 'X') {
      s1[j] = ss[i];
      j++;
    }
  }
  for (int i = 0, j = 0; i < 4; i++) {
    if (ss1[i] != 'X') {
      s2[j] = ss1[i];
      j++;
    }
  }
  for (int i = 0; i < 10; i++) {
    if (check(s1, s2, i)) {
      printf("YES");
      return 0;
    }
  }
  printf("NO");
  return 0;
}
