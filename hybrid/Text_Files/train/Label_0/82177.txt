for _ in range(int(input())):
	n = int(input())
	d = list(map(int, input().split(' ')))
	a = [0] * n
	uniq_a = set()
	is_exist = True
	sd = sorted(d, reverse=True)
	part_sum = 0
	for i in range(n):
		if sd[2 * i] != sd[2 * i + 1]:
			is_exist = False
			break
		a[i] = sd[2 * i] - part_sum
		if a[i] <= 0 or a[i] % (2 * (n - i)) != 0:
			is_exist = False
			break
		a[i] //= int(2 * (n - i))
		if a[i] in uniq_a:
			is_exist = False
			break
		uniq_a.add(a[i])
		part_sum += 2 * a[i]
	print(['NO', 'YES'][is_exist])
