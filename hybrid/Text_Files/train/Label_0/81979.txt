n = int(input())
arr = [input() for i in range(n)]
d1 = dict()
d2 = dict()
for s in arr:
	x = tuple(sorted(set(s)))
	if len(x) == 1:
		if s[0] in d1:
			d1[s[0]] += len(s)
		else:
			d1[s[0]] = len(s)
	elif len(x) == 2:
		if x in d2:
			d2[x] += len(s)
		else:
			d2[x] = len(s)
a = 0
if d1:
	m = d1.values()
	if len(m) == 1:
		a = max(m)
	else:
		m = sorted(m)
		m = m[::-1]
		a = m[0] + m[1]
b = 0
for (k, v) in d2.items():
	c = v
	if k[0] in d1:
		c += d1[k[0]]
	if k[1] in d1:
		c += d1[k[1]]
	b = max(b, c)
print(max(a, b))
