from sys import stdin

def isAP(arr):
	n = len(arr)
	if n == 1:
		return 0
	if n == 2:
		return arr[1] - arr[0]
	flag = 0
	diff = arr[1] - arr[0]
	for i in range(2, len(arr)):
		temp = arr[i] - arr[i - 1]
		if temp != diff:
			return -999999
	return diff
n = int(stdin.readline())
a = [int(i) for i in stdin.readline().split(' ')]
s = set(a)
dic = {}
for (i, val) in enumerate(a):
	if val not in dic:
		dic[val] = [i + 1]
	else:
		temp = dic[val]
		temp.append(i + 1)
		dic[val] = temp
l = []
for (key, value) in dic.items():
	temp = isAP(value)
	if temp == -999999:
		continue
	else:
		l.append([key, temp])
print(len(l))
l = sorted(l, key=lambda x: x[0])
for i in l:
	print(i[0], i[1])
