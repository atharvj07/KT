t = int(input())
for _ in range(t):
	(x, y, a, b) = map(int, input().split())
	tmp = min((x + y) // (a + b), min(x, y) // min(a, b))
	m1 = min(x, y)
	m2 = min(a, b)
	M1 = max(x, y)
	M2 = max(a, b)
	if a == b:
		print(tmp)
	else:
		d = (m1 - m2 * tmp) // (M2 - m2)
		if M2 * tmp - M1 - (M2 - m2) * d > 0:
			tmp -= 1
		print(tmp)
