import java.util.Scanner;

public   class Main{
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while (t-->0){
            int n=sc.nextInt();
           int a[]=new int[n];
           for (int i=0;i<n;i++)
               a[i]=sc.nextInt();

           
            int max = 0;
            for(int i=1;i<n;i++){
                int d = a[i-1] - a[i];
                if(d>0){
                    int x = Integer.toBinaryString(d).length();
                    max = Math.max(max, x);
                    a[i] = a[i-1];
                }
            }

            System.out.println(max);
        }
    }
}
