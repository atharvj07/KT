int N,M;
int POW[13],table[1000],sequence[1012];

void func(){

	for(int i = 0; i < N; i++)scanf("%d",&table[i]);

	bool FLG;

	int count,left,right;
	for(int state = 0; state < POW[M]; state++){
		count = 0;
		for(int loop = 0; loop < M; loop++){
			if(state&(1 << loop))count++;
		}
		if(count != table[0])continue;

		for(int loop = 0; loop < M; loop++){
			if(state&(1 << loop)){
				sequence[loop] = 1;
			}else{
				sequence[loop] = 0;
			}
		}

		FLG = true;

		left = 0,right = M;
		for(int i = 1; i <= N-1; i++){
			if(sequence[left] == 1){
				count--;
			}
			if(table[i] < count || table[i] > count+1){
				FLG = false;
				break;
			}

			if(table[i] == count){
				sequence[right] = 0;
			}else{ //table[i] == count+1
				sequence[right] = 1;
				count++;
			}
			left++;
			right++;
		}
		if(!FLG)continue;
		else{
			for(int i = 0; i < N+M-1; i++)printf("%d",sequence[i]);
			printf("\n");
			return;
		}
	}

	printf("Goofy\n");
}


int main(){

	for(int i = 0; i < 13; i++)POW[i] = pow(2,i);

	while(true){
		scanf("%d %d",&N,&M);
		if(N == 0 && M == 0)break;

		func();
	}

	return 0;
}
