
import java.util.Scanner;

public class A {
    public static void main(String[] args){
        Scanner reader = new Scanner(System.in);
        int n = reader.nextInt();
        if (n <= 2){
            System.out.print(1 + "\n" + 1);
        }
        else if (n == 3){
            System.out.print(2 + "\n" + 1 + " " + 3);
        }
        else if (n == 4){
            System.out.print(4 + "\n" +  "2 4 1 3");
        }
        else {
            System.out.println(n);
            for (int i = 1; i <= n; i += 2) {
                System.out.print(i + " ");
            }
            for (int i = 2; i <= n; i += 2) {
                System.out.print(i + " ");
            }
        }
    }

}
