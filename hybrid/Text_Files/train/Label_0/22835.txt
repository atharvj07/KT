#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:256000000")
using namespace std;
template <class T>
inline T sqr(T a) {
  return a * a;
}
int a[2000][2000];
bool was[2000][2000];
struct point {
  int x, y;
  point() {}
  point(int _x, int _y) {
    x = _x;
    y = _y;
  }
};
int r = 0, q = 0;
int n;
int dx[4] = {0, 0, 1, -1}, dy[4] = {-1, 1, 0, 0};
void dfs(const point &p) {
  was[p.x][p.y] = 1;
  for (int i = 0; i < 4; ++i) {
    point t(p.x + dx[i], p.y + dy[i]);
    if (t.x < 0 || t.x >= n || t.y < 0 || t.y >= n || was[t.x][t.y] ||
        !a[t.x][t.y])
      continue;
    dfs(t);
  }
}
int dx1[] = {1, 1, 1}, dy1[] = {0, -1, 1};
int down(const point &p) {
  for (int i = 0; i < 3; ++i) {
    point t(p.x + dx1[i], p.y + dy1[i]);
    if (t.x >= n || t.y >= n || t.x < 0 || t.y < 0 || !a[t.x][t.y]) continue;
    return 1 + down(t);
  }
  return 0;
}
int dx2[] = {0, 1}, dy2[] = {1, 0};
void right(const point &p, point &last) {
  last = point(p);
  for (int i = 0; i < 2; ++i) {
    point t(p.x + dx2[i], p.y + dy2[i]);
    if (t.x >= n || t.y >= n || t.x < 0 || t.y < 0 || !a[t.x][t.y]) continue;
    right(t, last);
    break;
  }
}
void f(const point &p) {
  point cur(p);
  int d = down(cur);
  point last(0, 0);
  right(point(cur.x, cur.y), last);
  double b = sqrt(sqr(p.x - last.x) + sqr(p.y - last.y) + .0);
  if (sqr(b) - sqr(d) / 2. < 3)
    ++r;
  else if (abs(b - d / sqrt(2.)) < 30. || abs(b / sqrt(2.) - d) < 30.)
    ++q;
  else
    ++r;
  dfs(p);
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < n; ++j) scanf("%d", &a[i][j]);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if (!a[i][j] || was[i][j]) continue;
      f(point(i, j));
    }
  }
  cout << r << " " << q;
  return 0;
}
