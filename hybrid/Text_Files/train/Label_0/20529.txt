#include <bits/stdc++.h>
using namespace std;
const int MAX = 3e5 + 2;
int n;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i = 1; i <= 1e5; i++) {
    long long x = i * (i + 1) >> 1;
    int l = 1, r = 1e5, mid;
    if (x > n) break;
    while (l <= r) {
      mid = (l + r) >> 1;
      long long y = mid * (mid + 1ll) >> 1;
      if (y + x > n)
        r = mid - 1;
      else if (y + x == n)
        return cout << "YES\n", 0;
      else
        l = mid + 1;
    }
  }
  cout << "NO";
  return 0;
}
