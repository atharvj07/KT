//ICPC C
#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll mod=1e9+7;

int main(){
    vector<int> answers;
    while(1){
        int n,m;
        cin>>n>>m;
        if(n==0&&m==0) break;
        vector<ll> a(n);
        vector<ll> w(m);
        for(int i=0;i<n;i++){
            cin>>a[i];
        }
        for(int i=0;i<m;i++){
            cin>>w[i];
        }

        //既存の分銅で量れる質量
        vector<ll> p;
        for(int i=0;i<pow(3,m);i++){
            ll tmp=i;
            ll val=0LL;
            for(int j=0;j<m;j++){
                int amari=tmp%3;
                if(amari==1){
                    val+=w[j];
                }else if(amari==2){
                    val-=w[j];
                }
                tmp/=3;
            }
            if(val>=0) p.push_back(val);
        }
        sort(p.begin(),p.end());

        //既存の分銅では作れない量のリストを作成
        vector<ll> targets;
        for(auto aim:a){
            if(!binary_search(p.begin(),p.end(),aim)) targets.push_back(aim);
        }
        if(targets.empty()){
            answers.push_back(0);
            continue;
        }

        //各作れない量と量れる量の差の絶対値
        vector<vector<ll>> diff(targets.size(),vector<ll>(p.size()*2));
        for(int i=0;i<targets.size();i++){
            for(int j=0;j<p.size();j++){
                diff[i][j]=abs(targets[i]-p[j]);
                diff[i][p.size()+j]=targets[i]+p[j];
            }
            sort(diff[i].begin(),diff[i].end());
        }

        //targets[0]を作るための追加分銅候補で他のも作れるかを判定
        bool find=false;
        for(int i=0;i<diff[0].size();i++){
            bool flag=true;
            for(int j=1;j<targets.size();j++){
                if(!binary_search(diff[j].begin(),diff[j].end(),diff[0][i])){
                    flag=false;
                    continue;
                }
            }
            if(flag){
                answers.push_back(diff[0][i]);
                find=true; break;
            }
        }

        if(!find) answers.push_back(-1);
      
    }
    for(auto ans:answers){
        cout<<ans<<endl;
    }
    return 0;
}
