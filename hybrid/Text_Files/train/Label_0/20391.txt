#include <bits/stdc++.h>
const int MAXINT = 2147483640;
const long long MAXLL = 9223372036854775800LL;
const long long MAXN = 1000000;
const long long MOD = 1e9 + 7;
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  string s;
  long long i, j, res = 0, kol = 0;
  cin >> s;
  for (i = s.size() - 1; i >= 0; i--) {
    if (s[i] == 'b')
      kol++;
    else {
      res += kol;
      kol %= MOD;
      res %= MOD;
      kol *= 2;
    }
  }
  cout << res << endl;
  return 0;
}
