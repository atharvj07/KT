    // Complexity O(log n)
    public static long modpow(long base, int exp, int mod) {
        base %= mod;
        long result = 1;
        while (exp > 0) {
            if ((exp & 1) == 1)
                result = (result * base) % mod;
            base = (base * base) % mod;
            exp >>= 1;
        }
        return result;
    }

    int count = 0;

    int recurse(int arr[], int i, int n, int max) {
        if (i >= n) {

            int and = arr[0];
            for (int x: arr) {
                and = and & x;
            }
            if (and == 0) {
                int sum = Arrays.stream(arr).sum();
                if (sum == max * (n-1)) {
                    System.out.println(Arrays.toString(arr) + "  " + and);
                    count++;
                }
                return Arrays.stream(arr).sum();
            } else {
                return -1;
            }
        }
        int ans = 0;
        for (int x = 0; x <= max; x++) {
            arr[i] = x;
            ans = Math.max(ans, recurse(arr, i + 1, n, max));
        }
        return ans;
    }

    int ii() {
        int num = 0, b;
