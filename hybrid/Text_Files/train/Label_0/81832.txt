import sys, collections
res = dict()
n = int(sys.stdin.readline())
for _ in range(n):
	(a, b) = map(int, sys.stdin.readline().split())
	res[a] = b
m = int(sys.stdin.readline())
for _ in range(m):
	(a, b) = map(int, sys.stdin.readline().split())
	if a in res:
		res[a] = max(res[a], b)
	else:
		res[a] = b
ans = 0
for i in res:
	ans += res[i]
print(ans)
