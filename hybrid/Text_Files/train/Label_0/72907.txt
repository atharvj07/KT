from sys import stdin

def solve():
	results = []
	for _ in range(int(stdin.readline().strip())):
		(N, X) = map(int, stdin.readline().strip().split())
		A = list(map(int, stdin.readline().strip().split()))
		sortedA = sorted(A)
		if A == sortedA:
			results.append('YES')
			continue
		if N <= X:
			results.append('NO')
			continue
		if X <= N // 2:
			results.append('YES')
			continue
		if A[N - X:X] == sortedA[N - X:X]:
			results.append('YES')
		else:
			results.append('NO')
	print('\n'.join(map(str, results)))
solve()
