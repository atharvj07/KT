#pragma GCC optimize "O3"

#define F first
#define S second

#include<bits/stdc++.h>

using namespace std;

typedef long long LL;

constexpr int N = 100000;

int n, luc, ldc, ruc, rdc, lc, rc;
map<LL, set<LL>> pts;
unordered_map<int, LL> lans, rans;
LL lux[N], luy[N], rux[N], ruy[N];
LL ldx[N], ldy[N], rdx[N], rdy[N];
LL larea, rarea;
vector<pair<LL, LL>> v;

inline LL crss(LL x0, LL y0, LL x1, LL y1, LL x2, LL y2) {
  return (x1-x0) * (y2-y0) - (x2-x0) * (y1-y0);
}
void build(LL x, LL y, int &cc, LL *xx, LL *yy, int f) {
  while(cc >= 2 && crss(xx[cc-2], yy[cc-2], xx[cc-1], yy[cc-1], x, y) * f >= 0) {
    --cc;
    v.emplace_back(xx[cc], yy[cc]);
  }
  v.emplace_back(xx[cc-1], yy[cc-1]);
  xx[cc] = x;
  yy[cc] = y;
  ++cc;
}
LL calc_area() {
  if(v.size() <= 2)
    return 0;
  v.push_back(v[0]);
  LL rtn = 0;
  for(size_t i = 1; i < v.size(); ++i)
    rtn += crss(0, 0, v[i-1].F, v[i-1].S, v[i].F, v[i].S);
  return abs(rtn);
}
int main() {
  scanf("%d", &n);
  for(int i = 0; i < n; ++i) {
    LL x, y; scanf("%lld %lld", &x, &y);
    pts[x].insert(y);
  }
  lans[lc] = 0;
  for(auto it = pts.begin(); it != pts.end(); ++it) {
    const LL &x = it->F;
    for(LL y : it->S) {
      v.clear();
      build(x, y, luc, lux, luy,  1);
      reverse(v.begin(), v.end());
      v.pop_back();
      build(x, y, ldc, ldx, ldy, -1);
      v.emplace_back(x, y);
      larea += calc_area();
    }
    lc += (it->S).size();
    lans[lc] = larea;
  }
  rans[rc] = 0;
  for(auto it = pts.rbegin(); it != pts.rend(); ++it) {
    const LL &x = it->F;
    for(LL y : it->S) {
      v.clear();
      build(x, y, ruc, rux, ruy,  1);
      reverse(v.begin(), v.end());
      v.pop_back();
      build(x, y, rdc, rdx, rdy, -1);
      v.emplace_back(x, y);
      rarea += calc_area();
    }
    rc += (it->S).size();
    rans[rc] = rarea;
  }
  LL ans = LLONG_MAX;
  for(auto &it : lans)
    ans = min(ans, it.S + rans[n - it.F]);
  ans = ((ans&1) ? (ans+1)/2 : ans/2);
  printf("%lld\n", ans);
}
