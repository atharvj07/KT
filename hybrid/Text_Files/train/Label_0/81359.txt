n = int(input())
a = 0
b = 1023
for i in range(n):
	(s, d) = input().split()
	d = int(d)
	if s == '|':
		a |= d
		b |= d
	elif s == '&':
		a &= d
		b &= d
	elif s == '^':
		a ^= d
		b ^= d
ans = []
ans.append(('|', 1023 ^ a ^ b))
ans.append(('^', 1023 ^ b))
print(len(ans))
for i in range(len(ans)):
	print(ans[i][0] + ' ' + str(ans[i][1]))
