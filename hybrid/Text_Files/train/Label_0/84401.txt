numberOfTests = int(input())
dictionary = {}
for q in range(numberOfTests):
	string = input()
	counterRed = 0
	counterGreen = 0
	switch = 0
	for x in range(len(string)):
		if x not in dictionary:
			dictionary[string[x]] = []
	for x in range(len(string)):
		if len(dictionary[string[x]]) == 2:
			continue
		if len(dictionary[string[x]]) == 1 and dictionary[string[x]] == 'red':
			dictionary[string[x]].append('green')
			switch = 0
			counterGreen += 1
			continue
		if len(dictionary[string[x]]) == 1 and dictionary[string[x]] == 'green':
			dictionary[string[x]].append('red')
			switch = 1
			counterRed += 1
			continue
		if switch == 0:
			dictionary[string[x]].append('red')
			counterRed += 1
			switch = 1
		else:
			dictionary[string[x]].append('green')
			counterGreen += 1
			switch = 0
	if counterRed == counterGreen:
		print(counterRed)
	else:
		print(min(counterRed, counterGreen))
