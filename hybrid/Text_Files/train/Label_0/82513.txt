t = int(input())
while t > 0:
	n = int(input())
	c = 0
	a = 0
	an = 0
	s = input()
	for i in range(n):
		if s[i] == '>':
			c = 1
		elif s[i] == '<':
			a = 1
		elif i == n - 1:
			if s[0] == '-':
				an += 1
			else:
				an += 2
		elif s[i + 1] == '-':
			an += 1
		else:
			an += 2
	if c == 0 or a == 0:
		print(n)
	else:
		print(an)
	t -= 1
