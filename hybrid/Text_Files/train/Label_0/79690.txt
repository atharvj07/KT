s = input()
t = input()
ans = 0
for i in range(len(s)):
	for j in range(i + 1, len(s) + 1):
		news = s[:i] + s[j:]
		count = 0
		f = 0
		for k in range(len(news)):
			if count >= len(t):
				f = 1
				break
			if news[k] == t[count]:
				count += 1
		if f == 1 or count == len(t):
			ans = max(ans, j - i)
print(ans)
