#include <bits/stdc++.h>
using namespace std;
int main() {
  int c[3][3], d;
  int dx[4] = {-1, 1, 0, 0};
  int dy[4] = {0, 0, -1, 1};
  memset(c, 0, sizeof(c));
  for (int i = 0; i < 3; i++) {
    for (int j = 0; j < 3; j++) {
      scanf("%d", &d);
      c[i][j] += d;
      for (int k = 0; k < 4; k++) {
        int i1 = i + dx[k];
        int j1 = j + dy[k];
        if (i1 >= 0 && j1 >= 0 && i1 < 3 && j1 < 3) {
          c[i1][j1] += d;
        }
      }
    }
  }
  for (int i = 0; i < 3; i++) {
    for (int j = 0; j < 3; j++) {
      if (c[i][j] % 2 == 0)
        printf("1");
      else
        printf("0");
    }
    printf("\n");
  }
  return 0;
}
