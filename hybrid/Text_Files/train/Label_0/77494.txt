import sys

def solve():
	(n, m) = map(int, input().split())
	tab = [list(input()) for x in range(n)]
	need = set(['f', 'a', 'c', 'e'])
	res = 0
	for row in range(n - 1):
		for col in range(m - 1):
			s = set()
			for r in range(row, row + 2):
				for c in range(col, col + 2):
					s.add(tab[r][c])
			if need.issubset(s):
				res += 1
	return res

def run():
	if sys.hexversion == 50594544:
		sys.stdin = open('test.txt')
	print(solve())
run()
