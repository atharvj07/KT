n = int(input())
string = input()
str1 = ''
str2 = ''
index = 0
_cond = 0
for i in range(n):
	if string[i] == '(':
		str1 += string[index:i] + '_'
		index = i + 1
	if string[i] == ')':
		str2 += string[index:i] + '_'
		index = i + 1
	if i == n - 1 and string[i] != ')':
		str1 += string[index:i + 1]
	if i == n - 1 and string[i] == ')':
		str2 += string[index:i + 1]
list1 = str1.split('_')
lens = 0
for item in list1:
	if len(item) > lens:
		lens = len(item)
list2 = str2.split('_')
count = 0
for item in list2:
	if item != '_' and item != '':
		count += 1
print(lens, count)
