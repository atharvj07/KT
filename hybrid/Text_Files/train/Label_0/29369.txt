const long long MOD = 1000000007ll;
long long powmod(long long a, long long b) {
  long long res = 1;
  a %= MOD;
  assert(b >= 0);
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % MOD;
    a = a * a % MOD;
  }
  return res;
}

template <typename T>
inline T gcd(T a, T b) {
  return b ? gcd(b, a % b) : a;
}
template <typename T>
inline T lcm(T a, T b) {
  return a * (b / gcd(a, b));
}

map<pair<int, int>, int> m;

bool cont[1010][1010];
void solve() {
  int n;
  cin >> n;
  for (int(i) = (0); (i) < (n); (i)++) {
    cin >> a[i].first >> a[i].second;
  }
  sort(a, a + n);
  for (int(i) = (0); (i) < (n); (i)++) {
    for (int(j) = (i + 1); (j) < (n); (j)++) {
      int X = a[j].first - a[i].first;
      int Y = a[j].second - a[i].second;
      int g = gcd(abs(X), abs(Y));
      X /= g;
      Y /= g;
      ::g[i][j] = make_pair(X, Y);
    }
  }
  int all = 0;
  for (int(i) = (0); (i) < (n); (i)++) {
    for (int(j) = (i + 1); (j) < (n); (j)++) {
      pair<int, int> p = g[i][j];
      if (cont[i][j]) {
        for (int(k) = (j + 1); (k) < (n); (k)++) {
          if (p == g[j][k]) {
            cont[i][k] = true;
            cont[j][k] = true;
          }
        }
        continue;
      }
      cont[i][j] = true;
      int x = 1;
      for (int(k) = (j + 1); (k) < (n); (k)++) {
        if (p == g[j][k]) {
          cont[i][k] = true;
          cont[j][k] = true;
        }
      }
      m[p]++;
      all++;
    }
  }
  long long ans = 0;
  for (auto x : m) {
    ans += 1ll * x.second * (all - x.second);
  }
  cout << ans / 2 << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t = 1;
  while (t--) {
    solve();
  }

}
