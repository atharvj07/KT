import java.util.Scanner;


public class RobotBicornAttack {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        String s = sc.next();
        
        long total =-1;
        
        int maxL =(s.length()-2 > 7)? 7 : s.length()-2 ;
        
        for(int i =1; i<= maxL; i++ ){
            
            int num1 = getMax(s.substring(0, i));
            if(num1 == -1) continue;
            
            for(int j=1; j <= ((s.length()-i-1 > 7) ? 7: s.length()-i-1) ; j++){
                int num2= getMax(s.substring(i,i+j));
                int num3= getMax(s.substring(i+j));
                if(num2 >= 0 && num3 >= 0){
                    if(total < num1 + num2 +num3){
                        total = num1 + num2 +num3; 
                    }
                }
            }
            
        }
        
        
        System.out.println(total);
        
        
        
        
        
        
        
        

    }

    private static int getMax(String sub) {
        
        if(sub.length() > 1 && sub.charAt(0) == '0') return -1;
        if(sub.length() > 7) return -1;
        
        
        int sum =0;
        for(int i =0; i< sub.length() ; i++){
            sum*=10;
            sum +=(sub.charAt(i) - '0');
            
        }
        return sum> 1000000 ? -1: sum;
    }

}
