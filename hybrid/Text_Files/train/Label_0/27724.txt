#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int n;
ll ans = 0;
ll const P = 1e9 + 7;
ll factorial[(int)1e6 + 2];
void piknie(int m) {
  factorial[0] = 1;
  for (int i = 1; i <= m; i++) factorial[i] = (factorial[i - 1] * (ll)i) % P;
}
ll power(ll a, ll pow) {
  if (pow == 0) return 1;
  ll x = power(a, pow / 2);
  if (pow % 2)
    return (a * ((x * x) % P)) % P;
  else
    return (x * x) % P;
}
ll licz(vector<int>& ciag) {
  ll ret = 1;
  ret = (ret * (n / ciag[0])) % P;
  for (int i = 1; i < ciag.size(); i++) {
    int ro = n / ciag[i] - n / ciag[i - 1];
    ret = (ret * (ll)ro) % P;
  }
  ret = (ret * factorial[n]) % P;
  for (int i = ciag.size() - 1; i >= 1; i--) {
    int ro = n - n / ciag[i - 1];
    ret = (ret * power(ro, P - 2)) % P;
  }
  ret = (ret * power(n, P - 2)) % P;
  return ret;
}
int main() {
  scanf("%d", &n);
  if (n < 3) {
    ans = 1;
  } else {
    int r = (int)(log2(n));
    int potdwa = (1 << r);
    piknie(n);
    vector<int> ciag;
    for (int i = 0; i <= r; i++) {
      ciag.push_back((1 << i));
    }
    reverse(ciag.begin(), ciag.end());
    ans = (ans + licz(ciag)) % P;
    if (3 * potdwa <= 2 * n) {
      for (int i = 1; i <= r; i++) {
        vector<int> krowa;
        for (int j = 0; j < i; j++) {
          krowa.push_back((1 << j));
        }
        for (int j = i; j <= r; j++) {
          krowa.push_back(3 * (1 << (j - 1)));
        }
        reverse(krowa.begin(), krowa.end());
        ans = (ans + licz(krowa)) % P;
      }
    }
  }
  printf("%I64d", ans);
  return 0;
}
