(n, m) = input().split()
n = int(n)
m = int(m)
ady = [[] for _ in range(n + 1)]
color = [0] * (n + 1)
oneColor = True
sol = [0] * m

def dfs_visit(v):
	global sol, ady, oneColor, color
	color[v] = 1
	for par in ady[v]:
		(u, index) = (par[0], par[1])
		if color[u] == 0:
			dfs_visit(u)
			sol[index] = 1
		elif color[u] == 2:
			sol[index] = 1
		else:
			sol[index] = 2
			oneColor = False
	color[v] = 2
for i in range(m):
	(n1, n2) = input().split()
	n1 = int(n1)
	n2 = int(n2)
	ady[n1].append((n2, i))
for v in range(1, n + 1):
	if color[v] == 0:
		dfs_visit(v)
if oneColor:
	print(1)
else:
	print(2)
res = str('')
for i in range(0, m):
	res += str(sol[i]) + str(' ')
print(res)
