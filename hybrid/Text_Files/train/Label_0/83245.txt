from sys import stdin
n = int(stdin.readline())
ai = list(map(int, stdin.readline().split()))
d = {}
vmax = 0
for i in range(0, n):
	if ai[i] not in d:
		d[ai[i]] = list([i])
	else:
		d[ai[i]].append(i)
vmax = 0
l = 0
r = 0
k = []
for i in d.keys():
	c = len(d[i])
	if c > vmax:
		vmax = c
		k = d[i]
	elif c == vmax:
		if d[i][c - 1] - d[i][0] < k[len(k) - 1] - k[0]:
			k = d[i]
	l = k[0]
	r = k[len(k) - 1]
print(str(l + 1) + ' ' + str(r + 1))
