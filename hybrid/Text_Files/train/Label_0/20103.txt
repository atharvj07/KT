#include <bits/stdc++.h>
using namespace std;
struct BIT {
  int v[505050];
  int n, pw;
  void init(int m) {
    n = m;
    fill_n(v, n + 1, 0);
    for (pw = 1; pw * 2 <= n; pw *= 2)
      ;
  }
  void add(int k, int d) {
    for (; k <= n; k += k & -k) v[k] += d;
  }
  int get(int k) {
    int s = 0;
    for (; k > 0; k -= k & -k) s += v[k];
    return s;
  }
  int get_one(int k) {
    int s, p;
    if (!k) return 0;
    s = v[k];
    p = k - (k & -k);
    for (k--; k > p; k -= k & -k) s -= v[k];
    return s;
  }
  int lwr_bound(int s) {
    int k, bt;
    k = 0;
    for (bt = pw; bt; bt /= 2) {
      if (k + bt > n) continue;
      k += bt;
      if (v[k] >= s)
        k -= bt;
      else
        s -= v[k];
    }
    return k + 1;
  }
};
BIT fnk;
char s[505050];
int n;
int main() {
  scanf("%d", &n);
  scanf("%s", s + 1);
  fnk.init(n);
  long long ans = 0;
  long long cur = 0;
  int mx = 0;
  for (int i = n; i >= 1; i--) {
    if (s[i] == '0') {
      ans += cur;
      mx = 0;
      continue;
    } else {
      int j = fnk.lwr_bound(mx + 1);
      fnk.add(j, -1);
      fnk.add(i, 1);
      cur += j - i;
      mx++;
      ans += cur;
    }
  }
  printf("%I64d\n", ans);
  return 0;
}
