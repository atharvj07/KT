#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, nn, i = 2, v[1000001];
  memset(v, 0, sizeof(v));
  cin >> n;
  if (n == 1) {
    cout << 1 << endl;
    return 0;
  }
  nn = n;
  int maxt = 0;
  while (i <= n) {
    while (n % i == 0) {
      v[i]++;
      n /= i;
      maxt = i;
    }
    i++;
  }
  vector<int> sol;
  sol.push_back(1);
  int x = 1;
  for (int i = 2; i <= maxt; i++) {
    while (v[i]-- > 0) {
      x *= i;
      sol.push_back(x);
    }
  }
  sort(sol.begin(), sol.end());
  for (int i = sol.size() - 1; i > 0; i--) cout << sol[i] << " ";
  cout << 1 << endl;
  return 0;
}
