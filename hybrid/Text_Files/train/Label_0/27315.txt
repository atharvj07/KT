#include<cstdio>
#include<algorithm>

#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

int m,n;
double p[16][16];

double dp[1<<16][16];
double dfs(int S,int pos){
	if(pos==n) return 1;
	if( S ==0) return 0;

	if(dp[S][pos]!=-1) return dp[S][pos];

	double res=0;
	rep(i,m) if(S&1<<i) { // ツ猫 i ツづーツ派ツ個ュツつキツづゥ
		double tmp=0,q=1;
		for(int j=pos;j<n;j++){ // j ツ氾板姪堋づ個敵ツづ可づ「ツづァツづェツつス
			tmp+=q*(1-p[i][j])*dfs(S&~(1<<i),j);
			q*=p[i][j];
		}
		tmp+=q*dfs(S&~(1<<i),n); // ツゴツーツδ?
		res=max(res,tmp);
	}
	return dp[S][pos]=res;
}

int main(){
	scanf("%d%d",&m,&n);
	rep(i,m) rep(j,n) scanf("%lf",p[i]+j);

	rep(S,1<<m) rep(j,n) dp[S][j]=-1;
	printf("%.15f\n",dfs((1<<m)-1,0));

	return 0;
}
