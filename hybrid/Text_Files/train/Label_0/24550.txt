void solve() {
  int n;
  cin >> n;
  vector<int> a(n, 0);
  long long sum = 0, a_xor = 0;
  for (auto &i : a) {
    cin >> i;
    sum += i;
    a_xor ^= i;
  }
  cout << 2 << "\n";
  cout << a_xor << " " << sum + a_xor;
}
