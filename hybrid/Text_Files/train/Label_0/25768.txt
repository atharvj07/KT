#include<cstdio>
#include<cstring>
#include<vector>

using namespace std;

int n,U;

vector<int> req;
vector<int> unit;

int main()
{
  while(scanf("%d%d", &n, &U), n || U) {
    req = vector<int>(n, 0);
    unit = vector<int>(n);
    int k,r;
    for(int i=0; i<n; ++i) {
      scanf("%d%d", &unit[i], &k);
      for(int j=0; j<k; ++j) {
			scanf("%d", &r);
			req[i] |= 1 << r;
      }
    }
    int j, sol = 100, u, c, skip;
    for(int i=0; i<(1<<n); ++i) {
      u = c = 0;
      for(j=n-1; j>=0; --j) {
			if(!(i&(1<<j))) continue;
			c++;
			if((i&req[j])!=req[j] || sol<c) {
				i += (((skip=((i&req[j])^req[j]|1<<j))&-skip)-1)&~i;
				break;
			}
			u += unit[j];
      }
      sol = j < 0 && u >= U ? min(sol, c) : sol;
    }
    printf("%d\n", sol);
  }
  return 0;
}
