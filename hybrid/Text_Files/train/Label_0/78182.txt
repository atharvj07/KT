for _ in range(int(input())):
	n = int(input())
	s = input()
	if s.count('2') in [1, 2]:
		print('NO')
	else:
		print('YES')
		l = len(s)
		x = []
		for i in range(l):
			if s[i] != '1':
				x.append(i)
		l2 = 0
		for i in range(l):
			if s[i] == '1':
				print('=' * i + 'X' + '=' * (l - i - 1))
			else:
				ll = ['='] * l
				ll[i] = 'X'
				ll[x[l2 - 1]] = '-'
				ll[x[(l2 + 1) % len(x)]] = '+'
				l2 += 1
				print(''.join(ll))
