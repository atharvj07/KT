n = int(input())
ids = list(map(int, input().split()))
d = {}
for i in range(0, n):
	if d.get(ids[i]) == None:
		d[ids[i]] = (0, i)
	else:
		d[ids[i]] = (d[ids[i]][0] + 1, i)
max = ids[0]
for i in range(0, n):
	if d[ids[i]][0] > d[max][0]:
		max = ids[i]
	elif d[ids[i]][0] == d[max][0]:
		if d[ids[i]][1] < d[max][1]:
			max = ids[i]
print(max)
