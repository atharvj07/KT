import sys, math
input = sys.stdin.readline
(n, k) = map(int, input().split())
cards = set()
card_list = list()
for i in range(n):
	card = input().strip()
	cards.add(card)
	card_list.append(card)
int_char = {1: 'S', 2: 'E', 3: 'T'}
char_int = {'S': 1, 'E': 2, 'T': 3}
xor = 1 ^ 2 ^ 3
ans = 0
for i in range(n):
	for j in range(i + 1, n):
		card = ['' for _ in range(k)]
		for a in range(k):
			if card_list[i][a] != card_list[j][a]:
				x = char_int[card_list[i][a]]
				y = char_int[card_list[j][a]]
				card[a] = int_char[x ^ y]
			else:
				card[a] = card_list[i][a]
		card_str = ''.join(card)
		if card_str in cards:
			ans += 1
print(int(ans / 3))
