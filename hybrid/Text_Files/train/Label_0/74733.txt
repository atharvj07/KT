(n, m) = map(int, input().strip().split(' '))
f = list(map(int, input().strip().split(' ')))
b = list(map(int, input().strip().split(' ')))
hm = {}
for i in range(n):
	if hm.__contains__(f[i]):
		hm[f[i]].append(i)
	else:
		hm[f[i]] = []
		hm[f[i]].append(i)
a = [0] * m
fl = True
am = False
for i in range(m):
	if hm.__contains__(b[i]):
		a[i] = hm[b[i]][0]
		if len(hm[b[i]]) > 1:
			am = True
	else:
		fl = False
		break
if fl and am == False:
	print('Possible')
	out = ''
	for j in a:
		out += str(j + 1) + ' '
	print(out)
elif fl and am == True:
	print('Ambiguity')
else:
	print('Impossible')
