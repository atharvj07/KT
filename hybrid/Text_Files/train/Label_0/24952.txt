#include <bits/stdc++.h>
const long long N = 100010;
const long long P = 998244353;
const long long INF = 99924435300000;
inline long long read() {
  long long x(0), w(1);
  char c = getchar();
  while (c ^ '-' && (c < '0' || c > '9')) c = getchar();
  if (c == '-') w = -1, c = getchar();
  while (c >= '0' && c <= '9') x = (x << 3) + (x << 1) + c - '0', c = getchar();
  return x * w;
}
char s[N];
long long n, x, dp[N][4];
signed main() {
  scanf("%lld\n", &n);
  scanf("%s\n", s + 1);
  for (long long i = 1; i <= n; ++i) {
    scanf("%lld", &x);
    dp[i][0] = dp[i - 1][0];
    dp[i][1] = dp[i - 1][1];
    dp[i][2] = dp[i - 1][2];
    dp[i][3] = dp[i - 1][3];
    if (s[i] == 'h') {
      dp[i][0] = dp[i - 1][0] + x;
    }
    if (s[i] == 'a') {
      dp[i][1] = std::min(dp[i - 1][0], dp[i - 1][1] + x);
    }
    if (s[i] == 'r') {
      dp[i][2] = std::min(dp[i - 1][1], dp[i - 1][2] + x);
    }
    if (s[i] == 'd') {
      dp[i][3] = std::min(dp[i - 1][2], dp[i - 1][3] + x);
    }
  }
  long long ans = INF;
  for (long long i = 0; i < 4; ++i) {
    ans = std::min(ans, dp[n][i]);
  }
  printf("%lld\n", ans);
  return 0;
}
