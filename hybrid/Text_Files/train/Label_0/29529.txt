#include <bits/stdc++.h>
using namespace std;
void read(int &x) {
  x = 0;
  int f = 1;
  char ch = getchar();
  for (; !isdigit(ch); ch = getchar())
    if (ch == '-') f = -f;
  for (; isdigit(ch); ch = getchar()) x = x * 10 + ch - '0';
  x *= f;
}
void print(int x) {
  if (x < 0) putchar('-'), x = -x;
  if (!x) return;
  print(x / 10), putchar(x % 10 + 48);
}
void write(int x) {
  if (!x)
    putchar('0');
  else
    print(x);
  putchar('\n');
}
const int maxn = 1e3 + 10;
const int inf = 1e9;
const double eps = 1e-8;
const int mod = 1e9 + 7;
int n, a[maxn], mx, t[maxn], cnt = 1;
char s[maxn][maxn];
int main() {
  read(n);
  for (int i = 1; i <= n; i++) read(a[i]), mx = max(mx, a[i]);
  write(mx + 1);
  mx++;
  for (int i = 1; i <= mx; i++) t[i] = 1;
  for (int i = 1; i <= n; i++) {
    int l = -1, r;
    for (int j = 2; j <= mx; j++)
      if (t[j] == t[j - 1]) {
        l = j - 1;
        break;
      }
    for (int j = l; j <= mx + 1; j++)
      if (t[j] != t[l]) {
        r = j - 1;
        break;
      }
    if (~l) {
      if (a[i] >= r - l + 1) {
        ++cnt;
        for (int j = l; j < r; j++) t[j] = cnt, a[i]--, s[j][i] = '1';
        s[r][i] = '0';
      } else {
        ++cnt;
        for (int j = l; j < l + a[i]; j++) t[j] = cnt, s[j][i] = '1';
        a[i] = 0;
      }
    }
    for (int j = 1; j <= mx; j++)
      if (!s[j][i]) {
        if (a[i])
          s[j][i] = '1', a[i]--;
        else
          s[j][i] = '0';
      }
  }
  for (int i = 1; i <= mx; i++) printf("%s\n", s[i] + 1);
  return 0;
}
