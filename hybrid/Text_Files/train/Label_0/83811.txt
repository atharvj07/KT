test_count = int(input())
n_list = []
m_list = []
k_list = []
tables = []
for i in range(test_count):
	n_m_k = list(map(int, input().split()))
	n_list.append(n_m_k[0])
	m_list.append(n_m_k[1])
	k_list.append(n_m_k[2])
	tables.append(list(map(int, input().split())))
for i in range(test_count):
	n = n_list[i]
	m = m_list[i]
	k = k_list[i]
	table = tables[i]
	can_do = True
	curr_pos = 0
	while curr_pos < n - 1:
		diff = table[curr_pos + 1] - table[curr_pos]
		if diff < k:
			if table[curr_pos] + diff - k < 0:
				m += table[curr_pos]
			else:
				m += -diff + k
			curr_pos += 1
		elif diff - m <= k:
			m -= diff - k
			curr_pos += 1
		else:
			can_do = False
			print('NO')
			break
	if can_do:
		print('YES')
