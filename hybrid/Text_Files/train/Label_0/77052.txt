for Z in range(int(input())):
	(x, y, a, b) = map(int, input().split())
	if a == b:
		print(min(y, x) // a)
		continue
	if x > y:
		(x, y) = (y, x)
	if a > b:
		(a, b) = (b, a)
	ans = 0
	awal_diambil = min((y - x) // (b - a), x // a)
	y -= awal_diambil * b
	x -= awal_diambil * a
	ans += awal_diambil
	akhir_diambil = x // (a + b)
	x -= akhir_diambil * (a + b)
	y -= akhir_diambil * (a + b)
	ans += akhir_diambil * 2
	if x >= a and y >= b:
		ans += 1
	print(ans)
