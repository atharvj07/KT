import java.util.Scanner;
public class tp {
	public static void main(String[] args) {
		Scanner s=new Scanner(System.in);
		int n=s.nextInt();
		long array[]=new long[n];
		for(int i=0;i<n;i++)
		{
			array[i]=s.nextLong();
		}
		System.out.println(func(array));
		s.close();	
	}
	public static long func(long[] array) {
		long ans=array[0],quotient,remainder;
		int j=1;
		for(int i=2;i<=array.length;i++)
		{  
			ans=ans^array[j++];
			quotient=array.length/i;
			remainder=array.length%i;
			if(quotient%2!=0)
			{    ++remainder;
			ans=ans^getXor(remainder,(i-1));
			}
			else
			{
				ans=ans^getXor(1,remainder);          	
			}
		}	
			return ans;
		}
		public static long f( long a) {
			long res[] = {a,1,a+1,0};
			return res[(int)a%4];
		}
        public static long getXor( long a, long b) {
			return f(b)^f(a-1);
		}
	}
