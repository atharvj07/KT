def correct_hour(is24, a, b):
	if is24:
		if a > 2:
			a = 0
		elif a == 2 and b > 3:
			b = 0
	elif a == b == 0:
		b = 1
	elif a > 1:
		if b == 0:
			a = 1
		else:
			a = 0
	elif a == 1 and b > 2:
		b = 0
	return (a, b)

def correct_minutes(c, d):
	if c > 5:
		c = 0
	return (c, d)
is24 = input() == '24'
(a, b, _, c, d) = input()
(a, b) = correct_hour(is24, int(a), int(b))
(c, d) = correct_minutes(int(c), int(d))
print(f'{a}{b}:{c}{d}')
