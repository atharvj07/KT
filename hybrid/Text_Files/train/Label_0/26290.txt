#include <bits/stdc++.h>
using namespace std;
set<int> s;
set<int>::iterator it = s.begin();
int main() {
  int n, temp;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> temp;
    s.insert(temp);
    it = s.find(temp);
    it++;
    if (it != s.end()) s.erase(it);
    ;
  }
  cout << s.size() << endl;
  return 0;
}
