int s;
int d(int x) {
    int f = x/3, b = x/5, fb = x/15;
    f -= fb, b -= fb;
    int sum = 0, base = 1, adnum = 1;
    int recfbs = 0;
    while(base * 10 <= x) {
        base *= 10;
        int bb = base - 1;
        int nf = bb/3, nb = bb/5, nfb = bb/15;
        int subs = nf + nb - nfb - recfbs;
        sum += adnum * (base - base / 10 - subs);
        adnum++;
        recfbs += subs;
    }
    int nf = x/3, nb = x/5, nfb = x/15;
    int subs = nf + nb - nfb - recfbs;
    sum += adnum * (x - base + 1 - subs);
    sum += 4*f + 4*b + 8*fb;
    // printf("x = %lld, sum = %lld\n", x, sum);
    return sum;
}

    cin >> s;
    int ub = 1e17, lb = -1;
    while(ub - lb > 1) {
        int mid = (ub + lb) / 2;
        // printf("mid = %lld\n", mid);
        if(d(mid) >= s) ub = mid;
        else lb = mid;
    }
    
    int idx = 0;
    if(ub % 15 == 0) idx = 8;

    else if(ub % 5 == 0) idx = 4;

        else if(ub % 3 == 0) ans += "Fizz";
        else if(ub % 5 == 0) ans += "Buzz";
        else ans += to_string(ub);
        ub++;
    }
