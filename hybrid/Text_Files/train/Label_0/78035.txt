def binary_search(arr, n, k):
	(l, r) = (0, n - 1)
	while l <= r:
		mid = l + r >> 1
		if arr[mid] <= k:
			l = mid + 1
		else:
			r = mid - 1
	return l
n = int(input())
arr1 = list(map(int, input().split()))
m = int(input())
arr2 = list(map(int, input().split()))
arr1.sort()
arr2.sort()
(res1, res2) = (2 * n, 2 * m)
for i in range(n):
	sum1 = 2 * i + 3 * (n - i)
	j = binary_search(arr2, m, arr1[i] - 1)
	sum2 = 2 * j + 3 * (m - j)
	diff_sum = sum1 - sum2
	diff_res = res1 - res2
	if diff_sum > diff_res or (diff_sum == diff_res and sum1 > res1):
		(res1, res2) = (sum1, sum2)
print('%d:%d' % (res1, res2))
