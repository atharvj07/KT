(n, k) = map(int, input().split(' '))
massiv = list(map(int, input().split(' ')))

def sila_v_igroka(mas, k):
	x = 0
	sila = 0
	if len(mas) <= k:
		return max(mas)
	while x != k:
		if mas[0] > mas[1]:
			sila = mas[0]
			mas = [mas[0]] + mas[2:] + [mas[1]]
			x += 1
		else:
			mas = [mas[1]] + mas[2:] + [mas[0]]
			sila = mas[0]
			x = 1
		if x == k:
			break
	return sila
print(sila_v_igroka(massiv, k))
