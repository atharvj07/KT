#include <bits/stdc++.h>
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
inline long long rand(long long x, long long y) {
  return (rng() % (y + 1 - x)) + x;
}

  return second;

template <typename T>
inline T gcd(T a, T b) {
  return a == 0 ? b : gcd(b % a, a);
}
const long long int sqn = 400;
const long long int other = (100006) / sqn + 1;
struct convex_hull {
  deque<pair<long long int, long long int> > dq;
  long long int first(pair<long long int, long long int> line,
                      long long int x) {
    return line.first * x + line.second;
  }
  pair<long long int, long long int> query(long long int x) {
    while (dq.size() > 1) {
      if (first(dq[0], x) < first(dq[1], x))
        dq.pop_front();
      else
        break;
    }
    return pair<long long int, long long int>(first(dq[0], x), dq[0].first);
  }
  long double intersect(long long int m1, long long int c1, long long int m2,
                        long long int c2) {
    return (long double)(c2 - c1) / (long double)(m1 - m2);
  }
  long double intersect(pair<long long int, long long int> line1,
                        pair<long long int, long long int> line2) {
    return intersect(line1.first, line1.second, line2.first, line2.second);
  }
  void insert(long long int m, long long int c) {
    pair<long long int, long long int> line =
        pair<long long int, long long int>(m, c);
    while (dq.size() > 1) {
      long long int second = dq.size();
      if (intersect(dq[second - 1], line) <= intersect(dq[second - 2], line))
        dq.pop_back();
      else
        break;
    }
    dq.push_back(line);
  }
} ch[other + 5];
long long int n, w, mb, C[(100006)], G[(100006)], mp;
vector<long long int> v[(100006)];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> w;
  for (long long int i = (0); i <= (long long int)(n - 1); ++i) {
    long long int a, b;
    cin >> a >> b;
    mb = max(mb, b), mp = max(mp, a);
    v[b].push_back(a);
  }
  auto build = [&](long long int i) {
    ch[i].dq.clear();
    ch[i].insert(0, 0);
    for (long long int j = (i * sqn);
         j <= (long long int)(min((i + 1) * sqn - 1, mp)); ++j) {
      ch[i].insert(j, C[j]);
    }
  };
  for (long long int i = (0); i <= (long long int)(other); ++i) build(i);
  long long int ppl = n;
  for (long long int i = (0); i <= (long long int)(mb + 1); ++i) {
    pair<long long int, long long int> ans =
        pair<long long int, long long int>(0, 0);
    for (long long int j = (0); j <= (long long int)(other); ++j) {
      ans = max(ans, ch[j].query(G[j]));
    }
    cout << ans.first + i * w * ppl << ' ' << ans.second << '\n';
    ppl -= v[i].size();
    for (auto j : v[i]) {
      for (int i = 0; i <= j; i += sqn) ++G[i / sqn];
      for (int i = j + 1; i <= min(mp, (j / sqn + 1) * sqn - 1); ++i) {
        C[i] -= i;
      }
      build(j / sqn);
    }
  }
}
