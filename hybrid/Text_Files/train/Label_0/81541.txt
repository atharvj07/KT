import sys

def get_array():
	return list(map(int, sys.stdin.readline().split()))

def get_ints():
	return map(int, sys.stdin.readline().split())

def input():
	return sys.stdin.readline().strip('\n')
(n, k) = get_ints()
w = 'abcdefghijklmnopqrstuvwxyz'
s = input()
d = {i: 0 for i in w}
last = s[0]
c = 0
for i in s:
	if i == last:
		c += 1
	else:
		c = 1
	if c >= k:
		d[i] += 1
		c = 0
	last = i
print(max(d.values()))
