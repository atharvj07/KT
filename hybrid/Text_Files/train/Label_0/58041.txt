import com.sun.jmx.snmp.SnmpDataTypeEnums;
import java.util.Scanner;

public class p3 {
    
    public static void main(String[] args)
    {
        Scanner s=new Scanner(System.in);
        int a=s.nextInt();
        int b=s.nextInt();
        if(b>a || a<0)
        {
            System.out.println(-1);
            return;
        }
        else if(a==b)
        {
          double c=(double)b;
            System.out.printf("%.12f",c);
            System.out.println();
        }
        else
        {
        
            long k1=(a+b)/(2*b);
            long k2=(a-b)/(2*b);
            double x1=(double)(a+b)/(2*k1);
             double x2=(double)(a-b)/(2*k2);
             System.out.printf("%.12f",Math.min(x1,x2));
        
        }
        
    }
    
}
