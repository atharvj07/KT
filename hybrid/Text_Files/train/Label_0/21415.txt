#include <bits/stdc++.h>
int t;
long long n, k, x, y, z;
int chk(long long d1, long long d2) {
  y = (k + d2 - d1);
  if (y % 3 != 0) {
    return 0;
  }
  y /= 3;
  x = y + d1;
  z = y - d2;
  if (x < 0 || y < 0 || z < 0) {
    return 0;
  }
  if (x > n || y > n || z > n) {
    return 0;
  }
  return 1;
}
int main() {
  scanf("%d", &t);
  long long d1, d2;
  while (t--) {
    scanf("%lld%lld%lld%lld", &n, &k, &d1, &d2);
    if (n % 3 != 0) {
      printf("no\n");
      continue;
    }
    n /= 3;
    if (chk(d1, -d2) || chk(d1, d2) || chk(-d1, d2) || chk(-d1, -d2)) {
      printf("yes\n");
      continue;
    }
    printf("no\n");
  }
  return 0;
}
