N = int(input())
for _ in range(N):
	n = int(input())
	lis = list(map(int, input().split()))
	sm = [0]
	for i in lis:
		sm.append(sm[-1] + 1) if i == 1 else sm.append(sm[-1] - 1)
	dif = sm[-1]
	dic = {}
	res = 2 * n
	for (i, v) in enumerate(sm):
		if i < n + 1:
			dic[v] = i
		if i >= n:
			t = v - dif
			if t in dic:
				res = min(res, i - dic[t])
	print(res)
