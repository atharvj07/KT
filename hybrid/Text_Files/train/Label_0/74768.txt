a = input()[::-1]
u = [False] * 3
t4 = 0
done = False
for i in a:
	j = int(i)
	if j % 2:
		if t4:
			print('YES\n' + i + t4)
			done = True
			break
		if j % 4 == 1:
			if u[2]:
				print('YES\n' + i + '6')
				done = True
				break
			if u[0]:
				t4 = i + '2'
		else:
			if u[0]:
				print('YES\n' + i + '2')
				done = True
				break
			if u[2]:
				t4 = i + '6'
	else:
		if j % 8 == 0:
			print('YES\n' + i)
			done = True
			break
		elif u[1]:
			if j == 4:
				if u[1]:
					t4 = i + '4'
			else:
				print('YES\n' + i + '4')
				done = True
				break
		u[j // 2 - 1] = True
if not done:
	print('NO')
