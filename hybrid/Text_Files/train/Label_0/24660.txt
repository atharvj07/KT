#include <bits/stdc++.h>
using namespace std;
int a[200005];
bool mark[200005];
int x;
vector<int> gr[200005];
int dfs(int p, int u) {
  int i, fl = 0;
  mark[u] = 1;
  if (u == a[u]) x = u;
  for (i = 0; i < gr[u].size(); i++) {
    int v = gr[u][i];
    if (mark[v] == 1 && v != p)
      fl = v;
    else if (mark[v] == 0) {
      if (fl == 0)
        fl = dfs(u, v);
      else
        dfs(u, v);
    }
  }
  return fl;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  int i, ans = 0;
  int mroot = -1, f = 0;
  memset(mark, 0, sizeof(mark));
  for (i = 1; i <= n; i++) {
    cin >> a[i];
    if (i == a[i]) {
      mroot = i;
    } else {
      gr[i].push_back(a[i]);
      gr[a[i]].push_back(i);
    }
  }
  for (i = 1; i <= n; i++) {
    if (mark[i] == 0) {
      int fl = dfs(i, i);
      if (fl == 0) {
        if (a[x] == mroot) continue;
        ans++;
        a[x] = mroot;
      } else {
        ans++;
        if (mroot == -1) {
          mroot = fl;
          a[fl] = mroot;
        } else
          a[fl] = mroot;
      }
    }
  }
  cout << ans << "\n";
  for (i = 1; i <= n; i++) cout << a[i] << " ";
}
