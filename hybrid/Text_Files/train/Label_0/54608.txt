import java.util.*;
public class CodeforcesE433 {
	public static int mod = 1000000007;
	public static int[] bound;
	public static int[][][][][] memo;
	public static int n, m, k;
	public static AhoCorasick aho;
	public static void main(String[] args){
		Scanner br = new Scanner(System.in);
		n = br.nextInt();
		m = br.nextInt();
		k = br.nextInt();
		
		int llen = br.nextInt();
		int[] l = new int[llen];
		for(int i = 0;i<llen;i++){
			l[i] = br.nextInt();
		}
		
		for(int i = llen-1;i>=0;i--){
			if(l[i] > 0){
				l[i]--;
				break;
			}
			else{
				l[i] = m-1;
			}
		}
		
//		System.out.println(Arrays.toString(l));
		
		int rlen = br.nextInt();
		int[] r = new int[rlen];
		for(int i = 0;i<rlen;i++){
			r[i] = br.nextInt();
		}
		
		string[] s = new string[n];
		for(int i = 0;i<n;i++){
			int len = br.nextInt();
			int[] nums = new int[len];
			for(int j = 0;j<len;j++){
				nums[j] = br.nextInt();
			}
			int val = br.nextInt();
			s[i] = new string(len, val, nums);
		}
		
		
		aho = new AhoCorasick();
		aho.createAhoCorasick(s);
		bound = r;
		memo = new int[2][2][bound.length][aho.nodeCount][k+1];
		for(int ll = 0;ll<2;ll++)
		for(int i = 0;i<2;i++){
			for(int j = 0;j<bound.length;j++){
				for(int kk = 0;kk<aho.nodeCount;kk++){
					Arrays.fill(memo[ll][i][j][kk], -1);
				}
			}
		}
		long ans1 = go(0, 1, 0, 0, 0);
		
		bound = l;
		memo = new int[2][2][bound.length][aho.nodeCount][k+1];
		for(int ll = 0;ll<2;ll++)
		for(int i = 0;i<2;i++){
			for(int j = 0;j<bound.length;j++){
				for(int kk = 0;kk<aho.nodeCount;kk++){
					Arrays.fill(memo[ll][i][j][kk], -1);
				}
			}
		}
		long ans2 = go(0, 1, 0, 0, 0);
		System.out.println((ans1-ans2+mod)%mod);
		
	}
	public static int go(int placed, int flag, int pos, int state, int count){
		if(count > k){
			return 0;
		}
		if(pos == bound.length){
			return 1;
		}
		
		if(memo[placed][flag][pos][state][count] != -1){
			return memo[placed][flag][pos][state][count];
		}
		
		int ans = 0;
		for(int i = 0;i<m;i++){
			if(flag == 1 && i > bound[pos]){
				break;
			}
			
			int nflag = flag;
			if(i < bound[pos]){
				nflag = 0;
			}
			
			if(placed == 0 && i == 0){
				ans+=go(0, nflag, pos+1, state, count);
			}
			else
				ans+=go(1, nflag, pos+1, aho.transition[i][state], count+aho.term[aho.transition[i][state]]);
			ans%=mod;
		}
		return memo[placed][flag][pos][state][count] = ans;
	}
	public static class AhoCorasick
	{
		public static final int ALPHA = 20;
		
		public int nodeCount;
		public int[][] transition;
		public int[] term;
		
		public void createAhoCorasick(string[] strs)
		{
			int maxNodes = 1;
			for(string s : strs)
			{
				maxNodes += s.len; // This could be passed in instead
			}
			
			int[][] children = new int[ALPHA][maxNodes];
			int[] leaf = new int[maxNodes];
			nodeCount = 1;
			for(string s : strs)
			{
				int node = 0;
				for(int ch : s.s)
				{
					int c = ch;
					if(children[c][node] == 0)
					{
						children[c][node] = nodeCount;
						nodeCount++;
					}
					
					node = children[c][node];
				}
				
				leaf[node] += s.val;
			}
			
			transition = new int[ALPHA][nodeCount];
			term = new int[nodeCount];
			
			ArrayDeque<Integer> queue = new ArrayDeque<Integer>();
			queue.add(0);
			queue.add(0);
			
			while(queue.size() > 0)
			{
				int node = queue.remove();
				int suffLink = queue.remove();
				
				term[node] += (leaf[node] + term[suffLink])%mod;
				term[node]%=mod;
				
				for(int ch = 0; ch < ALPHA; ch++)
				{
					if(children[ch][node] != 0)
					{
						transition[ch][node] = children[ch][node];
						queue.add(children[ch][node]);
						queue.add(node == 0 ? 0 : transition[ch][suffLink]);
					}
					else
					{
						transition[ch][node] = transition[ch][suffLink];
					}
				}
			}
		}
	}
	public static class string{
		int len, val;
		int[] s;
		public string(int a, int b, int[] c){
			len = a;
			val = b;
			s = c;
		}
	}
}
