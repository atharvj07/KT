import sys
from itertools import accumulate
from collections import Counter
n = int(input())
a = [list(accumulate((1 if c == '(' else -1 for c in l.rstrip()))) for l in sys.stdin]
counter = Counter((acc[-1] for acc in a if min(acc) == acc[-1] < 0 or min(acc) >= 0))
ans = 0
for (k, v) in counter.items():
	if k < 0:
		ans += min(v, counter[-k])
	elif k == 0:
		ans += v // 2
print(ans)
