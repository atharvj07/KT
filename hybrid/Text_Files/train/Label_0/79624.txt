from math import sqrt, gcd

def Gcd(a, b):
	if b == 0:
		return a
	else:
		return Gcd(b, a % b)
n = int(input())
arr = list(map(int, input().split()))
g = arr[0]
for i in range(1, n):
	g = Gcd(g, arr[i])
ans = 0
for i in range(1, int(sqrt(g)) + 1):
	if g % i == 0:
		if g // i == i:
			ans += 1
		else:
			ans += 2
print(ans)
