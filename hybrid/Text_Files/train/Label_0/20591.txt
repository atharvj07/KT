#include<bits/stdc++.h>
using namespace std;
#define FoR(i,a) for(int i=0;i<a;i++)
#define FOR(i,a,b) for(int i=a;i<b;i++)
int main()
{
    ios::sync_with_stdio(0);cin.tie(0);
    int T;
    cin>>T;
    while(T--)
    {
        int N,M;
        cin>>N>>M;
        int dp[N][M];
        FoR(i,N)FoR(j,M)dp[i][j]=0;
        FoR(i,N)
        {
            FoR(j,M)
            {
                char C;
                cin>>C;
                if(C == '*')dp[i][j]=1;
            }
        }
        /*FoR(i,N)
        {
            FoR(j,M)
            {
                cout<<dp[i][j];
            }
            cout<<'\n';
        }*/

        FOR(k,1,510){
            FOR(i,1,N){
                FOR(j,1,M-1){
                    if(dp[i-1][j]>=k&&dp[i][j-1]>=k&&dp[i][j+1]>=k&&dp[i][j]==k)dp[i][j]++;
                }
            }
        }
        int ans=0;
        FoR(i,N)FoR(j,M)ans+=dp[i][j];
        cout<<ans<<endl;

    }
return 0;
}
