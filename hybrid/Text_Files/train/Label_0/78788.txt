n = int(input())
for i in range(n):
	(x, y, x1, y1, x2, y2) = list(map(int, input().split()))
	d = 1
	v = 1
	c = 0
	c1 = 0
	while x1 != x2:
		c += 1
		if d == 1 and x1 != x:
			x1 += 1
		elif d == 1 and x1 == x:
			d = -1
			x1 -= 1
		elif d == -1 and x1 == 0:
			d = 1
			x1 += 1
		elif d == -1 and x1 != 0:
			x1 -= 1
	while y1 != y2:
		c1 += 1
		if v == 1 and y1 != y:
			y1 += 1
		elif v == 1 and y1 == y:
			v = -1
			y1 -= 1
		elif v == -1 and y1 == 0:
			v = 1
			y1 += 1
		elif v == -1 and y1 != 0:
			y1 -= 1
	if c > c1:
		print(c1)
	else:
		print(c)
