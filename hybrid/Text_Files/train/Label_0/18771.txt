const int mod = 1e9 + 7;


int fun(int i, int age, int n, int dir, vector<vector<vector<int>>>&dp) {
	if (dp[i][age][dir] != -1 or dp[n - i][age][1 - dir] != -1)
		return (dp[i][age][dir] == -1) ? dp[n - i][age][1 - dir] : dp[i][age][dir];
	if (age == 0)
		return 0;
	if (age == 1)
		return 1;
	if (i == 0 and dir == 0)
		return 1;
	else if (i == n and dir == 1) {
		return 1;
	} else {
		if (dir == 1) {
			return dp[n - i][age][1 - dir] = dp[i][age][dir] = (fun(i + 1, age, n, dir, dp) % mod + fun(i, age - 1, n, 1 - dir, dp) % mod) % mod;
		} else {
			return dp[n - i][age][1 - dir] = dp[i][age][dir] = (fun(i - 1, age, n, dir, dp) % mod + fun(i, age - 1, n, 1 - dir, dp) % mod) % mod;
		}
	}
}


int32_t main() {

	FIO
	int t;
	cin >> t;
	while (t--) {
		int total, start;
		cin >> total >> start;
		vector<vector<vector<int>>>dp(total + 1, vector<vector<int>>(start + 1, vector<int>(2, -1)));
		int ans = fun(0, start, total, 1, dp);
		cout << ans << endl;
	}

	return 0;
}
