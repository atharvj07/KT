import java.io.*;
import java.util.*;

public class C
{
   public static void main(String[] args)
   {
      MyScanner scan = new MyScanner();
      
      int n = scan.nextInt();
      int m = scan.nextInt();
      
      int[] data = new int[m];
      
      ArrayList<ArrayList<Integer>> dependencies = new ArrayList<ArrayList<Integer>>();
      for(int i = 0; i<=n; i++)
      {
         dependencies.add(new ArrayList<Integer>());
      }
      
      data[0] = scan.nextInt();
      long dist = 0;
      for(int i = 1; i<m; i++)
      {
         data[i] = scan.nextInt();
         int page1 = data[i-1];
         int page2 = data[i];
         dist += Math.abs(page1-page2);
         if(page1!=page2)
         {
            dependencies.get(page1).add(page2);
            dependencies.get(page2).add(page1);
         }
      }
      
      int minFrom = -1;
      int minTo = -1;
      long minDel = Long.MIN_VALUE;
      for(int i = 0; i<=n; i++)
      {
         ArrayList<Integer> dep = dependencies.get(i);
         if(dep.size()>0)
         {
            Collections.sort(dep);
            int median;
            if(dep.size() % 2==1)
            
             median = dep.get(dep.size()/2);
            else
             median = (dep.get(dep.size()/2) + dep.get(dep.size()/2-1))/2;
            long delta = 0;
            for(int p : dep)
               delta += Math.abs(p-i) - Math.abs(p-median);
            
            if(delta>minDel)
            {
               minDel = delta;
               minFrom = i;
               minTo = median;
            }
            
         }
      }
      
      long newSum = 0;
      
      if(data[0] == minFrom)
         data[0] = minTo;
         
      for(int i = 1; i < m; i++)
      {
         if(data[i] == minFrom)
         {
            data[i] = minTo;
         }
         newSum += Math.abs(data[i] - data[i-1]);
      }
      
      System.out.println(newSum);
      
      
      
     
      
   }
   
   
   public static class MyScanner {
      BufferedReader br;
      StringTokenizer st;
   
      public MyScanner() {
         br = new BufferedReader(new InputStreamReader(System.in));
      }
   
      String next() {
         while (st == null || !st.hasMoreElements()) {
            try {
               st = new StringTokenizer(br.readLine());
            } 
            catch (IOException e) {
               e.printStackTrace();
            }
         }
         return st.nextToken();
      }
   
      int nextInt() {
         return Integer.parseInt(next());
      }
   
      long nextLong() {
         return Long.parseLong(next());
      }
   
      double nextDouble() {
         return Double.parseDouble(next());
      }
   
      String nextLine() {
         String str = "";
         try {
            str = br.readLine();
         } 
         catch (IOException e) {
            e.printStackTrace();
         }
         return str;
      }
   
   }
}
