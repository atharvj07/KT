import sys
input = sys.stdin.readline
(n, k) = map(int, input().split())
cc = list(map(int, input().split()))
dp = [[0] * (k + 1) for i in range(2)]
dp[0][0] = 1
for (i, c) in enumerate(cc, 1):
	for j in range(k + 1):
		dp[i % 2][j] |= dp[(i - 1) % 2][j]
		if c <= j:
			dp[i % 2][j] |= dp[(i - 1) % 2][j - c]
			dp[i % 2][j] |= dp[(i - 1) % 2][j - c] << c
ans = [i for i in range(k + 1) if dp[n % 2][k] >> i & 1]
print(len(ans))
print(*ans)
