def solve():
	(n, k) = map(int, input().split())
	Ans = list(range(1, k + 1))
	s = k * (k + 1) // 2
	if n < s:
		print('NO')
		return
	add = (n - s) // k
	Ans = list(range(1 + add, k + 1 + add))
	add = (n - s) % k
	i = k - 1
	while i and add:
		x = min(2 * Ans[i - 1] - Ans[i], add)
		Ans[i] += x
		add -= x
		i -= 1
	if sum(Ans) != n:
		print('NO')
		return
	print('YES')
	print(*Ans)
solve()
