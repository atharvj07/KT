#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int tmp = n;
    while (tmp > 1) {
      tmp--;
      if (n / tmp > 1) {
        break;
      }
    }
    cout << tmp << "\n";
  }
  return 0;
}
