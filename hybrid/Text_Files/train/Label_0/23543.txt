#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int tmpn = n;
  bool found = false;
  m += 2;
  int left[n];
  int right[n];
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    left[n - 1 - i] = -1;
    right[n - 1 - i] = 0;
    for (int j = 0; j < m; j++) {
      if (s[j] == '1') {
        found = true;
        if (left[n - 1 - i] == -1) {
          left[n - 1 - i] = j;
        }
        right[n - 1 - i] = j;
      }
    }
    if (left[n - 1 - i] == -1 && !found) {
      tmpn -= 1;
    }
  }
  n = tmpn;
  if (n == 0) {
    cout << 0;
    return 0;
  }
  int dp[n][2];
  if (n == 1) {
    if (left[0] == -1)
      cout << 0;
    else {
      cout << right[0];
    }
    return 0;
  }
  dp[0][0] = right[0] * 2;
  dp[0][1] = m - 1;
  for (int i = 1; i < n - 1; i++) {
    if (left[i] == -1) {
      dp[i][0] = min(dp[i - 1][0] + 1, dp[i - 1][1] + m);
      dp[i][1] = min(dp[i - 1][0] + m, dp[i - 1][1] + 1);
    } else {
      dp[i][0] = min(2 * right[i] + 1 + dp[i - 1][0], dp[i - 1][1] + m);
      dp[i][1] =
          min(dp[i - 1][1] + abs(m - 1 - left[i]) * 2 + 1, dp[i - 1][0] + m);
    }
  }
  if (left[n - 1] == -1) {
    cout << min(dp[n - 2][0], dp[n - 2][1]);
  } else {
    cout << min(right[n - 1] + dp[n - 2][0] + 1,
                dp[n - 2][1] + m - left[n - 1]);
  }
  return 0;
}
