#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 7;
const long long INF = 1e18;
int t, n, a[maxn], b[maxn];
long long dp[maxn][3];
int main() {
  scanf("%d", &t);
  while (t--) {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d%d", &a[i], &b[i]);
    for (int i = 2; i <= n; i++)
      for (int j = 0; j < 3; j++) dp[i][j] = INF;
    dp[1][0] = 0, dp[1][1] = b[1], dp[1][2] = b[1] * 2;
    for (int i = 2; i <= n; i++)
      for (int j = 0; j < 3; j++)
        for (int k = 0; k < 3; k++)
          if (a[i] + j != a[i - 1] + k)
            dp[i][j] = min(dp[i][j], dp[i - 1][k] + b[i] * j);
    printf("%lld\n", min(min(dp[n][0], dp[n][1]), dp[n][2]));
  }
  return 0;
}
