#include <iostream>
#include <cmath>
#define inf 1e13+1
using namespace std;
long long int f(long long int b,long long int n) 
{
    if (n<b) 
        return n;
    else 
        return f(b,floor(n/b))+(n%b);
}
//wow
long long int n,s,b,p,ans=inf;
int main(void) 
{
    cin>>n>>s;
    if (n==s) 
    {
        cout<<n+1;
        return 0;
    }
    long long int sqrtn=sqrt(n)+100;
    for (b=2;b<=sqrtn;b++) 
        if (s==f(b,n)) 
        {
            cout<<b;
            return 0;
        }
    for (p=1;p<sqrtn;p++) 
    {
        if ((s-n)%p!=0) 
            continue;
        b=(n-s)/p+1;
        if (b<2) 
            continue;
        if (s==(f(b,n))) 
            ans=min(ans,b);
    }
    if (ans<inf) cout<<ans;
    else cout<<-1;
    return 0;
}//haha
