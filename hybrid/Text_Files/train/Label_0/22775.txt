#include <bits/stdc++.h>
using namespace std;
int R, C, r;
int M[502][502], F[502], I[502], AC[502][502];
int S[502][502], MXI[502][502], MXD[502][502], CMXI[502][502], CMXD[502][502];
bool ISeg(pair<int, int> a, pair<int, int> b) {
  return (a.first - b.first) * (a.second - b.first) <= 0 ||
         (b.first - a.second) * (b.second - a.second) <= 0;
}
bool Intersec(pair<int, int> a) {
  for (int i = max(0, a.first - r); i <= r; i++)
    if (ISeg(make_pair(1 - F[i], F[i] - 1),
             make_pair(a.second - F[a.first - i] + 1,
                       a.second + F[a.first - i] - 1)))
      return 1;
  return 0;
}
long long ans = 0, cnt = 0;
void update(int sum, long long cant) {
  if (cant == 0) return;
  if (sum > ans) {
    ans = sum;
    cnt = cant;
  } else if (sum == ans)
    cnt += cant;
}
int main() {
  scanf("%d%d%d", &R, &C, &r);
  for (int i = 0; i < R; i++)
    for (int j = 0; j < C; j++) scanf("%d", &M[i][j]);
  for (int i = 0; i < R; i++)
    for (int j = 0; j < C; j++) AC[i][j + 1] = AC[i][j] + M[i][j];
  for (int i = r; i >= 0; i--) {
    F[i] = F[i + 1];
    while (F[i] * F[i] + i * i <= r * r) F[i]++;
  }
  for (int i = r; i + r < R; i++)
    for (int j = r; j + r < C; j++)
      for (int k = i - r; k <= i + r; k++)
        S[i][j] += (AC[k][j + F[abs(k - i)]] - AC[k][j - F[abs(k - i)] + 1]);
  for (int(i) = (0); (i) < (R); (i)++) {
    MXI[i][r] = S[i][r];
    CMXI[i][r] = 1;
    for (int(j) = (r + 1); (j) < (C); (j)++)
      if (S[i][j] > MXI[i][j - 1]) {
        MXI[i][j] = S[i][j];
        CMXI[i][j] = 1;
      } else {
        MXI[i][j] = MXI[i][j - 1];
        CMXI[i][j] = CMXI[i][j - 1];
        if (S[i][j] == MXI[i][j]) CMXI[i][j]++;
      }
    MXD[i][C - 1 - r] = S[i][C - 1 - r];
    CMXD[i][C - 1 - r] = 1;
    for (int j = C - 2 - r; j >= 0; j--)
      if (S[i][j] > MXD[i][j + 1]) {
        MXD[i][j] = S[i][j];
        CMXD[i][j] = 1;
      } else {
        MXD[i][j] = MXD[i][j + 1];
        CMXD[i][j] = CMXD[i][j + 1];
        if (S[i][j] == MXD[i][j]) CMXD[i][j]++;
      }
  }
  for (int i = 2 * r; i >= 0; i--) {
    I[i] = I[i + 1];
    while (Intersec(make_pair(i, I[i]))) I[i]++;
  }
  for (int i = r; i + r < R; i++)
    for (int j = r; j + r < C; j++) {
      for (int k = r; k + r < R; k++)
        if (!I[abs(k - i)])
          update(S[i][j] + MXI[k][C - 1], CMXI[k][C - 1]);
        else {
          if (j - I[abs(k - i)] >= r)
            update(S[i][j] + MXI[k][j - I[abs(k - i)]],
                   CMXI[k][j - I[abs(k - i)]]);
          if (j + I[abs(k - i)] < C - r)
            update(S[i][j] + MXD[k][j + I[abs(k - i)]],
                   CMXD[k][j + I[abs(k - i)]]);
        }
    }
  cout << ans << " " << cnt / 2 << endl;
}
