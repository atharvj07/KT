s = input()
w = list(input())
n = len(s)
m = len(w)
if (n + m - 1) % 2:
	print('Impossible')
else:
	a = s.index('|')
	l = a
	r = n - l - 1
	if l == r:
		if m % 2:
			print('Impossible')
		else:
			while m > 0:
				if m % 2 == 0:
					s = s + w[-1]
					w.pop()
				else:
					s = w[-1] + s
					w.pop()
				m = m - 1
			print(s)
	else:
		while l < r and w != []:
			s = w[-1] + s
			w.pop()
			l = l + 1
		while r < l and w != []:
			s = s + w[-1]
			w.pop()
			r = r + 1
		if len(w) % 2:
			print('Impossible')
		else:
			m = len(w)
			while m > 0:
				if m % 2:
					s = s + w[-1]
					w.pop()
				else:
					s = w[-1] + s
					w.pop()
				m = m - 1
			a = s.index('|')
			if a == len(s) // 2 and (len(s) - 1) % 2 == 0:
				print(s)
			else:
				print('Impossible')
