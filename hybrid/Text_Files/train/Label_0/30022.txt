#include <bits/stdc++.h>
using namespace std;
template <typename T>
inline void debug(vector<T> v) {
  cout << "\ndebug(vector)\n";
  long long i;
  for (i = 0; i < v.size(); i++) cout << v[i] << " ";
  cout << "\n\n";
}
inline bool prime(long long n) {
  long long i;
  for (i = 2; i * i <= n; i++)
    if (n % i == 0) return false;
  return true;
}
int main(void) {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long t;
  cin >> t;
  while (t--) {
    long long i, j;
    long long n;
    cin >> n;
    vector<long long> a(n);
    for (i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end(), greater<int>());
    long long mx = -9223372036854775807, temp = 1;
    for (i = 0; i < 6; i++) {
      temp = 1;
      for (j = n - 5 + i; j < n; j++) temp *= a[j];
      for (j = 0; j < i; j++) temp *= a[j];
      if (mx < temp) mx = temp;
    }
    cout << mx << endl;
  }
}
