#include <bits/stdc++.h>
using namespace std;
vector<vector<int> > ret1;
vector<vector<int> > ret2;
vector<vector<int> > two;
vector<vector<int> > three;
vector<vector<int> > four;
int main() {
  int n;
  scanf("%d", &n);
  vector<int> a(n), check(n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
    a[i]--;
  }
  for (int i = 0; i < n; ++i) {
    if (check[i] == 0) {
      check[i] = 1;
      if (i == a[i]) continue;
      vector<int> seq;
      seq.push_back(i);
      int p = a[i];
      while (check[p] == 0) {
        check[p] = 1;
        seq.push_back(p);
        p = a[p];
      }
      while (seq.size() > 5) {
        vector<int> tmp, tmp2;
        for (int j = seq.size() - 5; j < seq.size(); ++j) {
          tmp.push_back(seq[j]);
          tmp2.push_back(seq[j]);
        }
        for (int j = 1; j < 5; ++j) swap(tmp2[j - 1], tmp2[j]);
        ret1.push_back(tmp);
        ret2.push_back(tmp2);
        for (int i = 0; i < 4; ++i) seq.pop_back();
      }
      if (seq.size() == 2) two.push_back(seq);
      if (seq.size() == 3) three.push_back(seq);
      if (seq.size() == 4) four.push_back(seq);
      if (seq.size() == 5) {
        ret1.push_back(seq);
        for (int j = 1; j < 5; ++j) swap(seq[j - 1], seq[j]);
        ret2.push_back(seq);
      }
    }
  }
  int s2 = two.size();
  int s3 = three.size();
  int p4 = 0;
  if (s3 - s2 >= 3) {
    int cnt = (s3 - s2) / 3;
    if (cnt > four.size()) cnt = four.size();
    vector<int> tmp(2);
    for (int i = 0; i < cnt; ++i) {
      tmp[0] = four[i][2];
      tmp[1] = four[i][3];
      two.push_back(tmp);
      tmp[0] = four[i][1];
      tmp[1] = four[i][2];
      two.push_back(tmp);
      tmp[0] = four[i][0];
      tmp[1] = four[i][1];
      two.push_back(tmp);
    }
    p4 = cnt;
  }
  s2 = two.size();
  s3 = three.size();
  if (s3 - s2 >= 3) {
    int cnt = (s3 - s2) / 3;
    for (int i = 0; i < cnt; ++i) {
      vector<int> rr = three.back(), tmp(2);
      three.pop_back();
      tmp[0] = rr[1], tmp[1] = rr[2];
      two.push_back(tmp);
      tmp[0] = rr[0], tmp[1] = rr[1];
      two.push_back(tmp);
    }
  }
  int p = min(two.size(), three.size());
  for (int i = 0; i < p; ++i) {
    vector<int> tmp(5), tmp2(5);
    tmp[0] = two[i][0];
    tmp[1] = two[i][1];
    tmp[2] = three[i][0];
    tmp[3] = three[i][1];
    tmp[4] = three[i][2];
    tmp2[0] = two[i][1];
    tmp2[1] = two[i][0];
    tmp2[2] = three[i][1];
    tmp2[3] = three[i][2];
    tmp2[4] = three[i][0];
    ret1.push_back(tmp);
    ret2.push_back(tmp2);
  }
  for (int i = p; i < two.size(); i += 2) {
    if (i + 1 == two.size()) {
      vector<int> tmp = two[i], tmp2 = two[i];
      swap(tmp2[0], tmp2[1]);
      ret1.push_back(tmp);
      ret2.push_back(tmp2);
    } else {
      vector<int> tmp(4), tmp2(4);
      tmp[0] = two[i][0];
      tmp[1] = two[i][1];
      tmp[2] = two[i + 1][0];
      tmp[3] = two[i + 1][1];
      tmp2 = tmp;
      swap(tmp2[0], tmp2[1]);
      swap(tmp2[2], tmp2[3]);
      ret1.push_back(tmp);
      ret2.push_back(tmp2);
    }
  }
  for (int i = p; i < three.size(); ++i) {
    vector<int> tmp = three[i];
    ret1.push_back(tmp);
    for (int j = 1; j < 3; ++j) swap(tmp[j - 1], tmp[j]);
    ret2.push_back(tmp);
  }
  for (int i = p4; i < four.size(); ++i) {
    vector<int> tmp = four[i];
    ret1.push_back(tmp);
    for (int j = 1; j < 4; ++j) swap(tmp[j - 1], tmp[j]);
    ret2.push_back(tmp);
  }
  printf("%d\n", ret1.size());
  for (int i = 0; i < ret1.size(); ++i) {
    int v = ret1[i].size();
    printf("%d\n", v);
    for (int j = 0; j < v; ++j) {
      if (j) printf(" ");
      printf("%d", ret1[i][j] + 1);
    }
    printf("\n");
    for (int j = 0; j < v; ++j) {
      if (j) printf(" ");
      printf("%d", ret2[i][j] + 1);
    }
    printf("\n");
  }
}
