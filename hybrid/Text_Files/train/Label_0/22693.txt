#include <bits/stdc++.h>
using namespace std;
#define int long long
int a[10][10];
int fir(int p){
    int r;
    while(p!=0){
        r=p%10;
        p/=10;
    }
    return r;
} 
int32_t main()
{
    int n; cin>>n;
    for(int i=1;i<=n;++i)
        a[fir(i)][i%10]++;
    int s=0;
    for(int i=1;i<10;++i)
        for(int j=1;j<10;++j)
            s+=a[i][j]*a[j][i];
    cout<<s;
    return 0;
}
