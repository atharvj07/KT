import sys
input = sys.stdin.readline
T = int(input())
for _ in range(T):
	(n, q) = map(int, input().split())
	s = input()
	pre_sum = [0]
	sign = 1
	for i in s:
		if i == '+':
			x = sign
		else:
			x = -sign
		pre_sum.append(pre_sum[-1] + x)
		sign *= -1
	for qq in range(q):
		(l, r) = map(int, input().split())
		sum = pre_sum[r] - pre_sum[l - 1]
		if sum == 0:
			print(0)
		elif sum & 1:
			print(1)
		else:
			print(2)
