import java.util.Scanner;

 */
public class DCubes {
    public static void solve(int ar[],int[] res, int counter){

        for(int i = 0; i < ar.length; i++){
            res[i] = Math.min(i != 0 ? ar[i - 1] : 0, Math.min(ar[i] - 1 >=0?ar[i] - 1 :0, i != ar.length - 1 ? ar[i + 1] : 0));
        }
        if(!Arrays.equals(res,new int[res.length])) {
            System.arraycopy(res,0,ar,0,ar.length);
            solve(ar, res, counter + 1);
        }
        else System.out.println(counter+1);
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] ar = new int[n+2];
        for (int i = 1; i <= n; i++) {
            ar[i] = sc.nextInt();
        }
        //solve(ar, new int[n], 0);
        int worst = 0;
        int[] res = new int[n+2];
        for(int i = 1; i <= n; ++i) {
            worst = Math.min(worst, ar[i]-i);
            res[i] = i + worst;
        }
        worst = n + 1;
        for(int i = n; i >= 1; --i) {
            worst = Math.min(worst, ar[i] + i);
            res[i] = Math.min(res[i], worst - i);
        }
        int R = 0;
        for(int i = 1; i <= n; ++i)
            R = Math.max(R, res[i]);
        System.out.println(R);
    }
}
