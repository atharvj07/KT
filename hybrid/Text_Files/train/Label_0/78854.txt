from math import sqrt, gcd
from collections import defaultdict

def solve(n):
	seti = set()
	maxi = 0
	for i in range(1, int(sqrt(n)) + 1):
		if n % i == 0:
			if n // i % 4 == 0:
				seti.add(n // i)
			if i % 4 == 0:
				seti.add(i)
				if i != n:
					maxi = max(maxi, i)
				if n // i != n:
					maxi = max(maxi, n // i)
	return seti
n = int(input())
if n == 3:
	print(1, 1, 3)
elif n == 1:
	print(1)
elif n == 2:
	print(1, 2)
else:
	ans = []
	m = 1
	while n > 3:
		ans += [m] * (n - n // 2)
		n //= 2
		m *= 2
	if n == 3:
		ans += [m, m, m * 3]
	elif n == 2:
		ans += [m, m * 2]
	else:
		ans += [m]
	print(*ans)
