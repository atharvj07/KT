#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, b, a;
  int s;
  int ans = 0;
  int maxn;
  cin >> n >> b >> a;
  maxn = a;
  for (int i = 1; i <= n; i++) {
    cin >> s;
    if (s == 0) {
      if (a > 0) {
        a--;
        ans++;
      } else if (b > 0) {
        b--;
        ans++;
      }
    } else {
      if (b > 0 && a < maxn) {
        b--;
        a++;
        ans++;
      } else if (a > 0) {
        a--;
        ans++;
      }
    }
  }
  if (ans > n) {
    ans = n;
  }
  cout << ans;
  return 0;
}
