a = int(input())
b = []
c = []
y = -1
for r in range(a):
	b += [int(input())]
	c += [input().split()]
for s in c:
	x = 0
	y += 1
	for t in s:
		x += 1
		if int(t) ** (1 / 2) != int(int(t) ** (1 / 2)):
			print('YES')
			break
		elif x == int(b[y]) and int(t) ** (1 / 2) == int(int(t) ** (1 / 2)):
			print('NO')
			break
		else:
			continue
