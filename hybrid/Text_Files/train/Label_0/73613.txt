import sys
nodes = int(input())
adjacency_list = {node: set() for node in range(1, nodes + 1)}
for _ in range(nodes - 1):
	(a, b) = [int(x) for x in input().split()]
	adjacency_list[a].add(b)
	adjacency_list[b].add(a)
sets = [set(), set()]
start_node = 1
start_level = 0
stack = [(start_node, start_level)]
visited = [False] * (nodes + 1)
while stack:
	(node, level) = stack.pop(-1)
	visited[node] = True
	sets[level % 2].add(node)
	for neighbor in adjacency_list[node]:
		if not visited[neighbor]:
			stack.append((neighbor, level + 1))
max_edges = len(sets[0]) * len(sets[1])
print(max_edges - (nodes - 1))
