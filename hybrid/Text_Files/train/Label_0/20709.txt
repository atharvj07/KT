#include <bits/stdc++.h>
using namespace std;
const long long N = 5e5 + 7;
int i, n, k, x, y, t, v, u, vs[N];
vector<int> group;
void bfs(int node, vector<vector<int>> &graph) {
  vs[node] = 1;
  queue<int> q;
  q.push(node);
  while (!q.empty()) {
    node = q.front();
    group.push_back(node);
    q.pop();
    for (auto child : graph[node]) {
      if (!vs[child]) {
        q.push(child);
        vs[child] = 1;
      }
    }
  }
}
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    int f[n + 2];
    memset(f, 0, sizeof(f));
    memset(vs, 0, sizeof(vs));
    vector<vector<int>> adj(n + 1);
    for (i = 1; i <= n; i++) {
      cin >> x;
      adj[i].push_back(x);
      adj[x].push_back(i);
    }
    for (i = 1; i <= n; i++) {
      if (f[i] == 0) {
        bfs(i, adj);
        for (auto it : group) f[it] = group.size();
        group.clear();
      }
      cout << f[i] << " ";
    }
    cout << "\n";
  }
}
