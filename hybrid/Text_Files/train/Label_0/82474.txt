import sys
input = sys.stdin.readline
from heapq import *
Q = int(input())
for _ in range(Q):
	N = int(input())
	A = []
	for __ in range(N):
		(a, f) = map(int, input().split())
		A.append((a, f))
	X = {}
	for (a, f) in A:
		if a in X:
			X[a][0] += 1
			X[a][1] += f
		else:
			X[a] = [1, f]
	Y = []
	for x in X:
		Y.append(X[x])
	Y = sorted(Y)[::-1] + [[0, 0]]
	su = 0
	suf = 0
	ne = Y[0][0]
	y = Y[0][0]
	i = 0
	h = []
	while i < len(Y):
		if len(h) == 0:
			y = Y[i][0]
			ne = min(ne, y)
		if ne < 0:
			break
		while i < len(Y) and Y[i][0] >= ne:
			heappush(h, -Y[i][1])
			i += 1
		if h:
			su += ne
			suf += min(-heappop(h), ne)
		ne -= 1
		if ne < 0:
			break
	print(su, suf)
