n = int(input())
a = sorted(map(int, input().split()))
mod = 998244353
z = 0
facn = 1
for i in range(n):
	z += abs(a[i] - a[-(i + 1)])
	facn = facn * (i + 1) % mod
fac2n = facn
for i in range(n + 1, 2 * n + 1):
	fac2n = fac2n * i % mod
invfacn = pow(facn, mod - 2, mod)
print(z * fac2n * invfacn * invfacn % mod)
