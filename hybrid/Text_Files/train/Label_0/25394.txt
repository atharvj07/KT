#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<pair<pair<int, int>, int> > v;
  int n, i, x, y, k = 0, x1, y1, x2, y2;
  double m1, m2;
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%d%d", &x, &y);
    v.push_back(make_pair(make_pair(x, y), i + 1));
  }
  sort(v.begin(), v.end());
  printf("%d ", v[0].second);
  x1 = v[0].first.first;
  y1 = v[0].first.second;
  x2 = v[1].first.first;
  y2 = v[1].first.second;
  if (x1 == x2)
    k = 1;
  else
    m1 = (double)(y2 - y1) / (x2 - x1);
  i = 2;
  x = v[i].first.first;
  y = v[i].first.second;
  if (k == 1) {
    while (x == x2) {
      i++;
      x = v[i].first.first;
      y = v[i].first.second;
    }
    printf("%d %d", v[i].second, v[1].second);
  } else {
    double m2 = (double)(y - y2) / (x - x2);
    while (m1 == m2) {
      i++;
      x = v[i].first.first;
      y = v[i].first.second;
      m2 = (double)(y - y2) / (x - x2);
    }
    if (m2 < m1)
      printf("%d %d", v[i].second, v[1].second);
    else
      printf("%d %d", v[1].second, v[i].second);
  }
  return 0;
}
