#include<iostream>
#include<queue>
#include<algorithm>
#include<vector>
#include<functional>
using namespace std;
long long N, M, K, A, B, C, T;
class point {
public:
	long long time;
	long long num;
	point(long long a, long long b) :time(a), num(b) {

	}
};
bool operator<(const point a, const point b) {
	return min(a.time / A, a.num) < min(b.time / A, b.num);
}
bool operator>(const point a, const point b) {
	return min(a.time / A, a.num) > min(b.time / A, b.num);
}
int main() {
	cin >> N >> M >> K >> A >> B >> C >> T;
	vector<int> line(M + 1);
	for (int i = 0; i < M; ++i) {
		cin >> line[i];
	}
	line[M] = N + 1;
	int ans = -1;
	int pa;
	priority_queue<point, vector<point> > que;
	for (int i = 0; i < M; ++i) {
		if ((line[i] - 1)*B <= T) {
			++ans;
			pa = min((long long)line[i + 1] - line[i] - 1, (T - (line[i] - 1)*B) / A);
			ans += pa;
			long long dtime = T - (line[i] - 1)*B - (pa + 1)*C;
			long long dnum = line[i + 1] - line[i] - 2 - pa;
			if (dtime >= 0&&dnum>=0) {
				que.push(point(dtime, dnum));
			}
		}
		else {
			break;
		}
	}
	for (int i = 0; i < K - M; ++i) {
		if (que.empty()) {
			break;
		}
		point dp = que.top(); que.pop();
		int pa = min(dp.time / A, dp.num);
		++ans;
		ans += pa;
		dp.time -= (pa + 1)*C;
		dp.num -= pa + 1;
		if (dp.time >= 0 && dp.num >= 0) {
			que.push(dp);
		}
	}
	cout << ans << endl;
	return 0;
}
