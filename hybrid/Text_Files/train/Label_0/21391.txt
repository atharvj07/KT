#include <bits/stdc++.h>
#define mod 1000000007
#define mod998 998244353
#define sp ' '
#define intmax 2147483647
#define llmax 9223372036854775807
#define mkp make_pair
typedef long long ll;
using namespace std;

int N;
double x[100000], y[100000];
char d[100000];

double f(double t) {
	double XMIN = 10000000000;
	double XMAX = -10000000000;
	double YMIN = 10000000000;
	double YMAX = -10000000000;
	for (int i = 0; i < N; ++i) {
		if (d[i] == 'U') {
			XMAX = max(XMAX, x[i]);
			XMIN = min(XMIN, x[i]);
			YMIN = min(YMIN, y[i] + t);
			YMAX = max(YMAX, y[i] + t);
		}
		if (d[i] == 'D') {
			XMAX = max(XMAX, x[i]);
			XMIN = min(XMIN, x[i]);
			YMIN = min(YMIN, y[i] - t);
			YMAX = max(YMAX, y[i] - t);
		}
		if (d[i] == 'R') {
			XMAX = max(XMAX, x[i] + t);
			XMIN = min(XMIN, x[i] + t);
			YMIN = min(YMIN, y[i]);
			YMAX = max(YMAX, y[i]);
		}
		if(d[i]=='L'){
			XMAX = max(XMAX, x[i] - t);
			XMIN = min(XMIN, x[i] - t);
			YMIN = min(YMIN, y[i]);
			YMAX = max(YMAX, y[i]);
		}
	}
	return (XMAX - XMIN)*(YMAX - YMIN);
}

int main() {
	cin >> N;
	for (int i = 0; i < N; ++i) {
		cin >> x[i] >> y[i] >> d[i];
	}
	double LL, L, R, RR;
	LL = 0;
	RR = 10000000000;
	for (int i = 0; i < 1000; ++i) {
		L = (LL * 2 + RR) / 3;
		R = (LL + RR * 2) / 3;
		if (f(L) < f(R)) {
			RR = R;
		}
		else {
			LL = L;
		}
	}
	printf("%.11f\n", f(LL));
}
