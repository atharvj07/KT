def dfs(now, num, l, r, dx, c):
	if now < 0 or num >= r:
		return 0
	temp = dfs(now - 1, num, l, r, dx, c)
	num += c[now]
	if num >= l and num <= r and (c[now] <= dx):
		temp += 1
	return temp + dfs(now - 1, num, l, r, dx, c)
c = input().split()
n = int(c[0])
l = int(c[1])
r = int(c[2])
x = int(c[3])
c = input().split()
for i in range(n):
	c[i] = int(c[i])
c.sort()
ans = 0
for i in range(1, n):
	if c[0] + x > c[i]:
		continue
	ans += dfs(i - 1, c[i], l, r, c[i] - x, c)
print(ans)
