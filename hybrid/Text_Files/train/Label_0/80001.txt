T = int(input())
for t in range(T):
	(N, K) = list(map(int, input().split()))
	a = list(map(int, input().split()))
	dic = {}
	for i in range(N):
		if a[i] in dic:
			dic[a[i]] += 1
		else:
			dic[a[i]] = 1
	keys = list(dic.keys())
	keys = keys + [1] * (K - len(keys))
	if len(keys) > K:
		print(-1)
		continue
	print(100 * K)
	for i in range(100):
		for key in keys:
			print(key, end=' ')
	print()
