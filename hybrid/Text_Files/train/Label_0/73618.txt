def max_radius(a, b, c):
	return min(a, b, c) / 2

def max_sum_radius(a, b, c, a2, b2, c2):
	sorted_first = sorted([a, b, c])
	sorted_second = sorted([a2, b2, c2])
	if sorted_first[1] == sorted_second[1] and sorted_first[2] == sorted_second[2]:
		return max_radius(sorted_first[0] + sorted_second[0], sorted_first[1], sorted_first[2])
	else:
		return 0
