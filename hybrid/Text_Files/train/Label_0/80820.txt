n = int(input())
S = input()
maxi = 0
lala = 0
for i in range(len(S)):
	s = S[i]
	if s == '[':
		lala += 1
	else:
		lala -= 1
	maxi = max(maxi, lala)
mass = maxi * 2 + 1
dlin = n + S.count('[]')
S = S.replace('[]', '[ ]')
SS = list()
for i in range(len(S)):
	s = S[i]
	if s == ' ':
		SS.append(-1)
	elif s == '[':
		SS.append(maxi - lala)
		lala += 1
	else:
		lala -= 1
		SS.append(maxi - lala)
	maxi = max(maxi, lala)
SSS = list(S)
ans = [[] for i in range(mass)]
a = maxi
for i in range(len(SS)):
	if SS[i] == -1:
		ans[a] += '   '
	else:
		ans[a] += '|'
for i in range(1, maxi + 1):
	a1 = a + i
	a2 = a - i
	for j in range(len(SS)):
		elem = SSS[j]
		code = SS[j]
		if code > i and code != -1:
			ans[a1] += '|'
			ans[a2] += '|'
		elif code == i and elem == '[':
			ans[a1] += '+-'
			ans[a2] += '+-'
		elif code == i and elem == ']':
			ans[a1] += '-+'
			ans[a2] += '-+'
		elif code == -1 and SS[j - 1] == i:
			ans[a1] += ' '
			ans[a2] += ' '
		elif code == -1:
			ans[a1] += '   '
			ans[a2] += '   '
		elif code < i and SS[j - 1] == i:
			pass
		elif code < i and SS[j + 1] == i:
			pass
		elif code < i:
			ans[a1] += ' '
			ans[a2] += ' '
for elem in ans:
	print(*elem, sep='')
