import java.io.*;
import java.util.*;


public class MartianCuisineSolver {

	public static final boolean DEBUG = false;
	
	public static void main(String[] args) {
		//redirect input and output
		if (DEBUG)
		{
			try {
				System.setIn(new FileInputStream("input.txt"));
				//System.setOut(new PrintStream("input.txt"));
			} catch (IOException e) {
				//nothing
			}
			//generate();
			//return;
		}
		Scanner sc = new Scanner(System.in);

		int nSamples = sc.nextInt();
		double[] result = new double[nSamples];
		
		for (int sample = 0; sample < nSamples; sample++)
		{
			//read data
			double R  = sc.nextDouble(), r = sc.nextDouble();
			int k = sc.nextInt();
			
			/*
			 * The bends are subject to relationship
			 * eps[k+2] = 2*eps[k+1] - eps[k] + C,
			 * where C = 2*(1/r - 1/R).
			 * Hence, eps[k] = k*(k+1)*C/2 + ak + b.  
			 */
			double C = 2*(1/r - 1/R);
			double b = 1/(R - r), a;
			//calculate eps[1] and a
			double discr = 16 * (1/(r * (R - r)) - 1/(r * R) - 1/((R - r) * R));
			if (discr < 0) //shouldn't happen	
				discr = 0;
			double eps = (1/r + 1/(R - r) - 1/R) + 0.5 * Math.sqrt(discr);
			a = eps - b - C;
			eps = C*k*(k + 1)/2 + a*k + b;
			result[sample] = 1/eps;
			
			//System.out.println("Slow: " + slowSolution(R, r, k));
		}
		
		//output
		for (int sample = 0; sample < nSamples; sample++)
			System.out.println(String.format(Locale.ENGLISH, "%.7f", result[sample]));
	}
	
	public static double slowSolution(double R, double r, int k)
	{
		double curR = R - r;
		
		for (int t = 0; t < k; t++)
		{
			double discr = 16 * (1/(r * curR) - 1/(r * R) - 1/(curR * R));
			if (discr < 0) //shouldn't happen	
				discr = 0;
			double eps = (1/r + 1/curR - 1/R) + 0.5 * Math.sqrt(discr);
			curR = 1/eps;
			//System.out.println(t + ": eps=" + eps);
		}
		return curR;
	}

	public static void generate()
	{
		int n = 9999;
		System.out.println(n);
		for (int i = 0; i < n; i++)
			System.out.println("10000 " + (i + 1) + " 10000");
	}
}
