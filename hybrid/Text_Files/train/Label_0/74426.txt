def answer(n, k, A):
	color = [0] * n
	color[0] = 0
	s0 = {}
	s1 = {}
	s0[A[0]] = 1
	for i in range(1, n):
		if k - A[i] not in s0:
			color[i] = 0
			if A[i] in s0:
				s0[A[i]] += 1
			else:
				s0[A[i]] = 1
		elif k - A[i] not in s1:
			color[i] = 1
			if A[i] in s1:
				s1[A[i]] += 1
			else:
				s1[A[i]] = 1
		elif s0[k - A[i]] > s1[k - A[i]]:
			color[i] = 1
			if A[i] in s1:
				s1[A[i]] += 1
			else:
				s1[A[i]] = 1
		else:
			color[i] = 0
			if A[i] in s0:
				s0[A[i]] += 1
			else:
				s0[A[i]] = 1
	return color
t = int(input())
for i in range(t):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	print(*answer(n, k, arr))
