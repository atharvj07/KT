(n, k) = map(int, input().split())
if k > n or (n > 1 and k <= 1):
	print(-1)
	exit()
s = 'a'
if n > 1:
	b = 1
else:
	b = 0
a = (n - k + 1) // 2
b += (n - k) // 2
while a > 0 or b > 0:
	if s[-1] == 'a':
		s += 'b'
		b -= 1
	else:
		s += 'a'
		a -= 1
i = len(s)
l = 99
while i < n:
	s += chr(l)
	l += 1
	i += 1
print(s)
