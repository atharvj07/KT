def int_lst_input():
	return [int(val) for val in input().split()]

def int_input():
	return int(input())

def print_lst(lst):
	print(' '.join([str(val) for val in lst]))

def solve():
	n = int_input()
	grid = []
	even = ''.join(['C' if i % 2 == 0 else '.' for i in range(n)])
	odd = ''.join(['C' if i % 2 == 1 else '.' for i in range(n)])
	c_even_count = sum([1 if c == 'C' else 0 for c in even])
	c_odd_count = sum([1 if c == 'C' else 0 for c in odd])
	c = c_even_count * (n // 2 + n % 2) + c_odd_count * (n // 2)
	for i in range(n):
		if i % 2 == 0:
			grid.append(even)
		else:
			grid.append(odd)
	print(c)
	for row in grid:
		print(row)
solve()
