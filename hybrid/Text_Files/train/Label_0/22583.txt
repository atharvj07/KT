#include<bits/stdc++.h>
#define ll long long
#define p 998244353
using namespace std;
int jc[10000010],njc[10000010],inv[10000010];
int C(int x,int y){
	if(x<y)return 0;
	return 1ll*jc[x]*njc[y]%p*njc[x-y]%p;
}
int f(int n,int m){
	if(n==0&&m==0)return 1;
	if(m>n)return 0;
	return (C(n+m,m)-C(n+m,m-1)+p)%p;
}
int n,A,B;
int main(){
	cin>>n>>A>>B;
	for(int i=0;i<2;i++)
	jc[i]=njc[i]=inv[i]=1;
	for(int i=2;i<=n;i++){
		jc[i]=1ll*jc[i-1]*i%p;
		inv[i]=1ll*inv[p%i]*(p-p/i)%p;
		njc[i]=1ll*njc[i-1]*inv[i]%p;
	}
	int ans=0;
	if(A==0&&B==0){
		cout<<1;
		return 0;
	}
	for(int k=0;k<B&&n-B-k-1>=0&&A-k>=0;k++){
		(ans+=1ll*C(A-k+n-B-k-1,n-B-k-1)*f(B-1,k)%p)%=p;
	}
	
	if(n==A+B)(ans+=f(B-1,A))%=p;
	cout<<ans;
	return 0;
}
