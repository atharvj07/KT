#include <bits/stdc++.h>
using namespace std;
void __print(int x) { cerr << x; }
void __print(long x) { cerr << x; }
void __print(long long x) { cerr << x; }
void __print(unsigned x) { cerr << x; }
void __print(unsigned long x) { cerr << x; }
void __print(unsigned long long x) { cerr << x; }
void __print(float x) { cerr << x; }
void __print(double x) { cerr << x; }
void __print(long double x) { cerr << x; }
void __print(char x) { cerr << '\'' << x << '\''; }
void __print(const char *x) { cerr << '\"' << x << '\"'; }
void __print(const string &x) { cerr << '\"' << x << '\"'; }
void __print(bool x) { cerr << (x ? "true" : "false"); }
template <typename T, typename V>
void __print(const pair<T, V> &x) {
  cerr << '{';
  __print(x.first);
  cerr << ',';
  __print(x.second);
  cerr << '}';
}
template <typename T>
void __print(const T &x) {
  int f = 0;
  cerr << '{';
  for (auto &i : x) cerr << (f++ ? "," : ""), __print(i);
  cerr << "}";
}
void _print() { cerr << "]\n"; }
template <typename T, typename... V>
void _print(T t, V... v) {
  __print(t);
  if (sizeof...(v)) cerr << ", ";
  _print(v...);
}
const int N = 1e3 + 10;
long long a[N];
const int OFFSET = N;
int dp[N][N + OFFSET + 100];
int who[N][N + OFFSET + 100];
int n, k;
map<char, int> mp;
string s;
bool solve(int idx, int diff) {
  if (idx == n) {
    return (diff == k || diff == -k);
  }
  if (diff >= k || diff <= -k) return 0;
  if (dp[idx][diff + OFFSET] != -1) return dp[idx][diff + OFFSET];
  bool ans = 0;
  if (s[idx] != '?') {
    ans = solve(idx + 1, diff + (mp[s[idx]]));
    if (ans) who[idx][diff + OFFSET] = mp[s[idx]];
  } else {
    ans = solve(idx + 1, diff + 1);
    if (ans) who[idx][diff + OFFSET] = 1;
    if (!ans) {
      ans |= solve(idx + 1, diff - 1);
      if (ans) who[idx][diff + OFFSET] = -1;
      if (!ans) {
        ans |= solve(idx + 1, diff);
        if (ans) who[idx][diff + OFFSET] = 0;
      }
    }
  }
  return dp[idx][diff + OFFSET] = ans;
}
string ans;
void go(int idx, int diff) {
  if (idx == n) return;
  int yo = who[idx][diff + OFFSET];
  if (yo == 1)
    ans += 'W';
  else if (yo == -1)
    ans += 'L';
  else if (yo == 0)
    ans += 'D';
  go(idx + 1, diff + yo);
}
int main() {
  ios_base ::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  mp['W'] = 1;
  mp['L'] = -1;
  mp['D'] = 0;
  memset(dp, -1, sizeof(dp));
  cin >> n >> k;
  cin >> s;
  int hi = solve(0, 0);
  if (hi) {
    go(0, 0);
    cout << ans;
  } else {
    cout << "NO";
  }
  return 0;
}
