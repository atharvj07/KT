n = int(input())
a = list(map(int, input().split()))
s = a[0]
count = 0
if a[0] > 15:
	print(15)
else:
	i = 0
	while i <= n - 1:
		if i == n - 1:
			if a[i] + 15 < 90:
				count = a[i] + 15
			else:
				count = 90
			break
		elif a[i + 1] - a[i] > 15:
			count = a[i] + 15
			break
		else:
			count = a[i]
		i += 1
	print(count)
