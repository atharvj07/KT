#include <bits/stdc++.h>
using namespace std;
const int maxn = 1005;
const int inf = 0x3f3f3f3f;
int n, m;
char a[100005 * 2], b[100005 * 2];
long long sum1, sum2, sum3, sum4;
int main() {
  scanf("%d", &n);
  getchar();
  for (int i = 1; i <= n; i++) {
    scanf("%c", &a[i]);
  }
  getchar();
  for (int i = 1; i <= n; i++) {
    scanf("%c", &b[i]);
    if (a[i] == '0' && b[i] == '1')
      sum1++;
    else if (a[i] == '0' && b[i] == '0')
      sum2++;
    else if (a[i] == '1' && b[i] == '0')
      sum3++;
    else if (a[i] == '1' && b[i] == '1')
      sum4++;
  }
  long long ans = sum1 * sum3 + sum2 * sum3 + sum2 * sum4;
  cout << ans;
  return 0;
}
