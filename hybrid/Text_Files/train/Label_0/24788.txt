const int C = 200;

int h[500];
int dist[500];
int prevv[500];
pii preve[500];
int min_cost_flow(int s,int t,int f){
  int res = 0;
  fill(h,h+500,0);
  while(f>0){
    priority_queue<pii> Q;
    fill(dist,dist+500,INF);
    dist[s] = 0;
    Q.push(pii(INF-0,INF-s));
    while(!Q.empty()){
      pii P = Q.top(); Q.pop();
      int d = INF-P.first;
      int v = INF-P.second;
      if(dist[v]<d)continue;
      REP(to,500){
        int cost = g[v][to];
        int capa = cap[v][to];
        if(capa>0 && dist[to]>dist[v]+cost+h[v]-h[to]){
          dist[to] = dist[v]+cost+h[v]-h[to];
          prevv[to] = v;
          preve[to] = pii(v,to);
          Q.push(pii(INF-dist[to],INF-to));
        }
      }
    }
    if(dist[t]==INF){
      return -1;
    }
    REP(i,500) h[i]+=dist[i];
    int d = f;
    for(int v=t;v!=s;v=prevv[v]){
      d = min(d, cap[preve[v].first][preve[v].second]);
    }
    f -= d;
    res += d*h[t];
    for(int v=t;v!=s;v=prevv[v]){
      cap[preve[v].first][preve[v].second] -= d;
      cap[preve[v].second][preve[v].first] += d;
    }
  }
  return res;
}
