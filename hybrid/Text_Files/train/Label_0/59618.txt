import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    final static Map<Long, Integer> map = new HashMap<>();
    final static Map<Long, Integer> sol = new HashMap<>();

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String n = scanner.nextLine();
        int numberOfInputs = Integer.parseInt(n);
        for (int i = 0; i < numberOfInputs; i++) {
            String line = scanner.nextLine();
            long number = Long.parseLong(line);
            int numberOfMoves = 0;
            if (sol.get(number) != null) {
                numberOfMoves = sol.get(number);
            } else{
                //numberOfMoves = getNumberOfMoves(number);
                numberOfMoves = getNumberOfMoves2(number);
            }
            if (numberOfMoves == -1)
                System.out.println(numberOfMoves);
            else
                System.out.println(numberOfMoves - 1);
        }
    }

    private static int getNumberOfMoves2(long number){
        if (number == 1) return 1;
        int moves = 1;
        while (number != 1 && number != -1){
            if (number % 2 == 0) {
                number /= 2;
                moves++;
            } else if (number % 3 == 0) {
                number = (2*number) / 3;
                moves++;
            } else if (number % 5 == 0){
                number = (4*number) / 5;
                moves++;
            } else{
                number = -1;
                moves = 0;
            }
        }
        return moves;
    }

    private static int getNumberOfMoves(long number){
        if (number == 1) return 1;

        if (map.get(number) != null) return map.get(number);

        int numberOfMovesBy2 = -1;
        if (number % 2 == 0) {
            numberOfMovesBy2 = 1 + getNumberOfMoves(number / 2);
        }
        int numberOfMovesBy3 = -1;
        if (number % 3 == 0) {
            numberOfMovesBy3 = 1 + getNumberOfMoves((2*number) / 3);
        }
        int numberOfMovesBy5 = -1;
        if (number % 5 == 0) {
            numberOfMovesBy5 = 1 + getNumberOfMoves((4*number) / 5);
        }

        if (numberOfMovesBy2 > 0 && numberOfMovesBy3 > 0 && numberOfMovesBy5> 0){
            int min = (Math.min(numberOfMovesBy2, Math.min(numberOfMovesBy3, numberOfMovesBy5)));
            map.put(number, min);
            return min;
        }
        // not numberOfMovesBy5
        if (numberOfMovesBy2 > 0 && numberOfMovesBy3 > 0 ){
            int min = (Math.min(numberOfMovesBy2,numberOfMovesBy3));
            map.put(number, min);
            return min;
        }
        // not numberOfMovesBy2
        if (numberOfMovesBy5 > 0 && numberOfMovesBy3 > 0 ){
            int min = (Math.min(numberOfMovesBy5,numberOfMovesBy3));
            map.put(number, min);
            return min;
        }
        // not numberOfMovesBy3
        if (numberOfMovesBy5 > 0 && numberOfMovesBy2 > 0 ){
            int min = (Math.min(numberOfMovesBy5,numberOfMovesBy2));
            map.put(number, min);
            return min;
        }

        if (numberOfMovesBy5 > 0){
            map.put(number, numberOfMovesBy5);
            return numberOfMovesBy5;
        }

        if (numberOfMovesBy2 > 0){
            map.put(number, numberOfMovesBy2);
            return numberOfMovesBy2;
        }

        if (numberOfMovesBy3 > 0){
            map.put(number, numberOfMovesBy3);
            return numberOfMovesBy3;
        }

        map.put(number, -1);

        //impossible
        return -1;
    }
}
