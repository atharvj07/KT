from math import gcd

def check(e, f, n):
	i = 1
	while i * e <= n:
		if (n - i * e) % f == 0:
			return (n - i * e) // f
		i += 1

def prime(x):
	b = []
	i = 2
	while i * i <= x:
		if x % i == 0:
			if x // i == i:
				b.append(i)
			else:
				b.extend([i, x // i])
		i += 1
	return b
k = int(input())
a = prime(k)
a.sort()
(e, f) = (-1, -1)
l = len(a)
for i in range(l):
	if gcd(a[i], a[l - 1 - i]) == 1:
		e = a[i]
		f = a[l - i - 1]
		break
if e == -1:
	print('NO')
else:
	x = check(e, f, k - 1)
	y = (k - 1 - x * f) // e
	print('YES')
	print(2)
	print(x, e)
	print(y, f)
