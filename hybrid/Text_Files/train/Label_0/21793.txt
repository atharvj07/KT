#include <iostream>
using namespace std;
int eval(int *l, char *c, int n, int i){
  switch (c[i]) {
    case '+':
    {
      int sum = 0;
      for(int k = i+1 ; l[i+1] <= l[k] && i < n ; k++){
        if(l[i+1] == l[k]){
          sum += eval(l,c,n,k);
        }
      }
      return sum;
    }
    case '*':
    {
      int prod = 1;
      for(int k = i+1 ; l[i+1] <= l[k] && i < n ; k++){
        if(l[i+1] == l[k]){
          prod *= eval(l,c,n,k);
        }
      }
      return prod;
    }
    default:
      return c[i] - '0';
  }
}
int main(void){
  for(;;){
    int n;
    cin >> n;
    if(n == 0){ return 0; }
    int  l[n];
    char c[n];
    for(int i = 0 ; i < n ; i++){
      string s;
      cin >> s;
      l[i] = s.length();
      c[i] = s[l[i]-1];
    }
    cout << eval(l,c,n,0) << endl;
  }
}
