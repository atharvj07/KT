import math
for _ in range(int(input())):
	(n, c) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	days = 0
	dp = [(0, 0)]
	cost = 0
	i = 0
	while True:
		if i < n - 1 and b[i] <= c:
			x = math.ceil((b[i] - cost) / a[i])
			days += x
			dp.append((cost + x * a[i], days))
		else:
			x = math.ceil((c - cost) / a[i])
			days += x
			break
		if dp[-1][0] >= c:
			break
		else:
			cost = dp[-1][0] - b[i]
			days += 1
			i += 1
	min_days = days
	for i in range(1, len(dp)):
		min_days = min(min_days, dp[i][1] + math.ceil((c - dp[i][0]) / a[i - 1]))
	print(min_days)
