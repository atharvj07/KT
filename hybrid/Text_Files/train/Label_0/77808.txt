def verify_cross(i, j, matrix):
	for k in range(3):
		try:
			if matrix[i + k][j] == '0':
				return False
		except:
			return False
	for k in [-1, 0, 1]:
		try:
			if matrix[i + 1][j + k] == '0':
				return False
		except:
			return False
	return True

def main():
	n = int(input())
	matrix = []
	crosses = ''
	for i in range(n):
		st_row = input()
		crosses += st_row
		row = [i for i in st_row]
		matrix.append(row)
	if crosses.count('#') % 5 != 0:
		print('NO')
		return
	for i in range(n):
		for j in range(n):
			if matrix[i][j] == '#':
				if verify_cross(i, j, matrix):
					try:
						matrix[i + 2][j] = '0'
						matrix[i][j] = '0'
						matrix[i + 1][j] = '0'
						matrix[i + 1][j - 1] = '0'
						matrix[i + 1][j + 1] = '0'
					except:
						print('NO')
						return
				else:
					print('NO')
					return
	print('YES')
main()
