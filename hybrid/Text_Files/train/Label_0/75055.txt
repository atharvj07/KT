from collections import defaultdict
import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
D = defaultdict(list)
for i in range(n):
	(a, b, c) = input().split()
	(b, c) = (int(b) - 1, int(c))
	D[b].append((c, a))
	D[b] = sorted(D[b], reverse=True)[:3]
for i in range(m):
	if len(D[i]) == 2:
		print(D[i][0][1], D[i][1][1])
		continue
	(a, b, c) = D[i]
	if c[0] == b[0] or c[0] == a[0]:
		print('?')
	else:
		print(a[1], b[1])
