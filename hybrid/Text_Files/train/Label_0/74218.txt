n = int(input())
s = input()
m = input()
if s == 0:
	num_s = [0] * n
else:
	num_s = [int(x) for x in str(s)]
if m == 0:
	num_m = [0] * n
else:
	num_m = [int(x) for x in str(m)]
num_m.sort()
num_s.sort()
a = 0
j = 0
for i in range(n):
	p = num_s[i]
	while j < n and num_m[j] < p:
		j += 1
	if j == n:
		break
	else:
		a += 1
		j += 1
print(n - a)
b = 0
j = 0
for i in range(n):
	p = num_s[i]
	while j < n and num_m[j] <= p:
		j += 1
	if j == n:
		break
	else:
		b += 1
		j += 1
print(b)
