t = int(input())
com = []
for _ in range(t):
	a = input().split()
	com.append(a)
x = 0
forflag = 0
temp = 0
shit = 0
high = pow(2, 32) - 1
a = [0 for i in range(int(100000.0 + 1))]
le = 0
mul = 1
indi = 0
poo = 0
check = [0 for i in range(int(100000.0 + 1))]
for i in range(t):
	if com[i][0] == 'add':
		if forflag == 0:
			x += 1
		else:
			for j in range(indi, le):
				mul *= a[j]
				check[j] = 1
				if mul > high:
					shit = 1
					break
			if shit == 1:
				break
			x += mul
			indi = le
	elif com[i][0] == 'end':
		forflag -= 1
		if check[le - 1] == 1:
			mul //= a[le - 1]
			check[le - 1] = 0
			if mul == 0:
				mul = 1
			indi -= 1
		le -= 1
	else:
		n = int(com[i][1])
		a[le] = n
		le += 1
		forflag += 1
	if x > high:
		shit = 1
		break
if shit == 1:
	print('OVERFLOW!!!')
else:
	print(x)
