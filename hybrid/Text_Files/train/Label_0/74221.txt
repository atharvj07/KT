from collections import Counter

def flicks(a, b, mx=False):
	a_counter = Counter(a)
	b_counter = Counter(b)
	res = 0
	for i in a:
		for j in range(i + mx, 10):
			if j in b_counter and b_counter[j] > 0:
				b_counter[j] -= 1
				res += 1
				break
	if not mx:
		return len(a) - res
	return res
_ = input()
a = [int(i) for i in input()]
b = [int(i) for i in input()]
a.sort()
b.sort()
print(flicks(a, b))
print(flicks(a, b, True))
