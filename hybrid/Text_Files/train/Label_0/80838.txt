t = int(input())
for i in range(t):
	d = dict()
	l = int(input())
	ans = []
	arr = list(map(int, input().split()))
	for num in arr:
		if num in d:
			d[num] += 1
		else:
			d[num] = 1
	for (key, value) in d.items():
		if value == 1:
			ans.append(key)
	if len(ans) == 0:
		print('-1')
	else:
		print(arr.index(min(ans)) + 1)
