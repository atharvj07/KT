t = int(input())
while t > 0:
	s = ''
	main = []
	pair = []
	track = [0, 0]
	n = int(input())
	for i in range(n):
		(a, b) = map(int, input().split())
		pair.append(a)
		pair.append(b)
		main.append(pair)
		pair = []
	main.sort()
	flag = 0
	for i in range(n):
		val1 = main[i][0] - track[0]
		val2 = main[i][1] - track[1]
		if val1 < 0 or val2 < 0:
			flag = 1
			print('NO')
			break
		else:
			for j in range(val1):
				s += 'R'
			for k in range(val2):
				s += 'U'
		track[0] = main[i][0]
		track[1] = main[i][1]
	if flag == 0:
		print('YES')
		print(s)
	t -= 1
