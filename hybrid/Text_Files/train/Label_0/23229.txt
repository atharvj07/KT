#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int x = 0, l = 1;
  char c = getchar();
  for (; c < '0' || c > '9'; c = getchar())
    if (c == '-') l = -1;
  for (; c >= '0' && c <= '9'; c = getchar()) x = x * 10 + c - 48;
  return x * l;
}
int V[2005][2005];
const long long mo = 1000000007;
long long k;
int n;
int dx[9] = {0, 1, 1, -1, -1, 2, 2, -2, -2};
int dy[9] = {0, 2, -2, 2, -2, 1, -1, 1, -1};
long long ans[555];
queue<pair<pair<int, int>, int> > q;
inline long long mul(long long a, long long p) {
  long long ans = 0;
  for (; p; p >>= 1, a = (a + a) % mo)
    if (p & 1) ans = (ans + a) % mo;
  return ans;
}
inline void bfs() {
  ans[0] = 1;
  V[(0 + 1001)][(0 + 1001)] = 1;
  q.push(make_pair(make_pair(0, 0), 0));
  while (!q.empty()) {
    int x = q.front().first.first, y = q.front().first.second,
        t = q.front().second;
    q.pop();
    if (t >= 500) break;
    for (int i = 1; i <= 8; i++)
      if (!V[(x + dx[i] + 1001)][(y + dy[i] + 1001)]) {
        int xx = x + dx[i], yy = y + dy[i];
        V[(xx + 1001)][(yy + 1001)] = 1;
        q.push(make_pair(make_pair(xx, yy), t + 1));
        ans[t + 1]++;
      }
  }
}
int main() {
  scanf("%lld %d", &k, &n);
  for (int i = 1; i <= n; i++) {
    int x = read(), y = read();
    V[(x + 1001)][(y + 1001)] = 1;
  }
  bfs();
  long long inc = ans[500] - ans[499], pls = ans[500];
  for (int i = 1; i <= 500; i++) ans[i] = (ans[i] + ans[i - 1]) % mo;
  if (k <= 500)
    printf("%lld\n", ans[k]);
  else
    printf("%lld\n", (ans[500] + mul(k - 500, pls) +
                      mul(mul(mul(k - 499, k - 500), (mo + 1) / 2), inc)) %
                         mo);
  return 0;
}
