import os, sys, atexit
from io import BytesIO, StringIO
input = BytesIO(os.read(0, os.fstat(0).st_size)).readline
_OUTPUT_BUFFER = StringIO()
sys.stdout = _OUTPUT_BUFFER

@atexit.register
def write():
	sys.__stdout__.write(_OUTPUT_BUFFER.getvalue())
(n, q) = map(int, input().split())
grid = [0, 0, 0]
grid[0] = [0] * (n + 5)
grid[1] = [0] * (n + 5)
grid[2] = [0] * (n + 5)
ind = [0] * (n + 5)
ind2 = [0] * (n + 5)
count = 0
count2 = 0
for i in range(q):
	(r, c) = map(int, input().split())
	if grid[r][c] == 0:
		grid[r][c] = 1
		if r == 1:
			if grid[2][c - 1] == 1 or grid[2][c] == 1 or grid[2][c + 1] == 1:
				if ind[c] == 0:
					ind[c] = 1
					count -= -1
			if grid[2][c - 1] == 1:
				if ind2[c - 1] == 0:
					ind2[c - 1] = 1
					count2 -= -1
			if grid[2][c] == 1:
				if ind2[c] == 0:
					ind2[c] = 1
					count2 -= -1
			if grid[2][c + 1] == 1:
				if ind2[c + 1] == 0:
					ind2[c + 1] = 1
					count2 -= -1
		else:
			if grid[1][c - 1] == 1 or grid[1][c] == 1 or grid[1][c + 1] == 1:
				if ind2[c] == 0:
					ind2[c] = 1
					count2 -= -1
			if grid[1][c - 1] == 1:
				if ind[c - 1] == 0:
					ind[c - 1] = 1
					count -= -1
			if grid[1][c] == 1:
				if ind[c] == 0:
					ind[c] = 1
					count -= -1
			if grid[1][c + 1] == 1:
				if ind[c + 1] == 0:
					ind[c + 1] = 1
					count -= -1
	else:
		grid[r][c] = 0
		if r == 1 and ind[c] == 1:
			ind[c] = 0
			count += -1
		elif r == 2:
			if ind[c - 1] == 1:
				if c - 2 >= 1:
					if grid[2][c - 2] != 1:
						if grid[2][c - 1] != 1:
							ind[c - 1] = 0
							count += -1
			if ind[c] == 1:
				if c - 1 >= 1:
					if grid[2][c - 1] != 1:
						if grid[2][c + 1] != 1:
							ind[c] = 0
							count += -1
			if ind[c + 1] == 1:
				if grid[2][c + 1] != 1 and grid[2][c + 2] != 1:
					ind[c + 1] = 0
					count += -1
	if count == 0 or count2 == 0:
		print('Yes')
	else:
		print('No')
