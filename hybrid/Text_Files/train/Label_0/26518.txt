#include <bits/stdc++.h>
using namespace std;
int n;
int t[4 * 110000];
int a[110000];
int b[110000];
bool marked[4 * 110000];
void push(int v, int m) {
  if (marked[v]) {
    t[v * 2] = t[v];
    t[v * 2 + 1] = t[v] + m;
    t[v] = -1;
    marked[v * 2] = marked[v * 2 + 1] = true;
    marked[v] = false;
  }
}
void update(int v, int tl, int tr, int l, int r, int new_val) {
  if (l > r) return;
  if (l == tl && tr == r) {
    t[v] = new_val;
    marked[v] = true;
  } else {
    int tm = (tl + tr) / 2;
    int off = tm + 1 - tl;
    push(v, off);
    if (l > tm) {
      update(v * 2 + 1, tm + 1, tr, max(l, tm + 1), r, new_val);
    } else if (r <= tm) {
      update(v * 2, tl, tm, l, min(r, tm), new_val);
    } else {
      update(v * 2, tl, tm, l, min(r, tm), new_val);
      int offset = tm + 1 - l;
      update(v * 2 + 1, tm + 1, tr, max(l, tm + 1), r, new_val + offset);
    }
  }
}
int get(int v, int tl, int tr, int pos) {
  if (tl == tr) {
    return t[v];
  }
  int tm = (tl + tr) / 2;
  int off = tm + 1 - tl;
  push(v, off);
  if (pos <= tm)
    return get(v * 2, tl, tm, pos);
  else
    return get(v * 2 + 1, tm + 1, tr, pos);
}
void print() {
  cout << "PRINT\n";
  for (int i = 1; i <= 9; i++) cout << i << " " << t[i] << "\n";
  cout << "DONE\n";
}
void solve(int index = 0) {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < 4 * 110000; i++) {
    t[i] = -1;
  }
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= m; i++) {
    int t;
    cin >> t;
    if (t == 1) {
      int x, y, k;
      cin >> x >> y >> k;
      update(1, 1, n, y, y + k - 1, x);
    } else {
      int x;
      cin >> x;
      int index = get(1, 1, n, x);
      if (index == -1) {
        cout << b[x] << "\n";
      } else {
        cout << a[index] << "\n";
      }
    }
  }
}
void multi_solve() {
  int t;
  cin >> t;
  for (int i = 1; i <= t; i++) {
    solve(i);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  solve();
  return 0;
}
