#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  set<pair<long long, long long> > s;
  long long n, curr = 0;
  cin >> n;
  for (long long i = 0; i < n; i++) {
    long long l, r;
    cin >> l >> r;
    auto it = s.lower_bound({l, 0});
    if (it == s.end()) {
      s.insert({l, 0});
      s.insert({r, 1});
      curr++;
      cout << curr << " ";
      continue;
    }
    auto itr = s.lower_bound({r, 0});
    if (it->first == itr->first and itr->first != r) {
      if (it->second == 0) {
        s.insert({l, 0});
        s.insert({r, 1});
        curr++;
      }
    } else if (it->first == itr->first and itr->first == r) {
      if (it->second == 0) {
        s.erase(itr);
        s.insert({l, 0});
      }
    } else {
      if (it->second == 0 and (itr == s.end() || itr->second == 0)) {
        while (1) {
          it = s.lower_bound({l, 0});
          if (it == itr) break;
          if (it->second == 0) curr--;
          s.erase(it);
        }
        curr++;
        s.insert({l, 0});
        if (itr == s.end() || itr->first != r)
          s.insert({r, 1});
        else {
          s.erase(itr);
          curr--;
        }
      } else if (it->second == 0 and itr->second == 1) {
        while (1) {
          it = s.lower_bound({l, 0});
          if (it == itr) break;
          if (it->second == 0) curr--;
          s.erase(it);
        }
        curr++;
        s.insert({l, 0});
      } else if (it->second == 1 and (itr == s.end() || itr->second == 0)) {
        while (1) {
          it = s.lower_bound({l, 0});
          if (it == itr) break;
          if (it->second == 0) curr--;
          s.erase(it);
        }
        if (itr == s.end() || itr->first != r)
          s.insert({r, 1});
        else {
          s.erase(itr);
          curr--;
        }
      } else {
        while (1) {
          it = s.lower_bound({l, 0});
          if (it == itr) break;
          if (it->second == 0) curr--;
          s.erase(it);
        }
      }
    }
    cout << curr << " ";
  }
  return 0;
}
