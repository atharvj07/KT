for _ in range(int(input())):
	(n, q) = map(int, input().split())
	s = input()
	for i in range(q):
		(x, y) = map(int, input().split())
		if abs(y - x) == n:
			print('NO')
		if x == y:
			print('NO')
		else:
			an = -1
			l = s[x - 1]
			r = s[y - 1]
			if x != 1:
				for i in range(x - 1):
					if s[i] == l:
						an = 1
						break
			if y != n:
				for i in range(y, n):
					if s[i] == r:
						an = 1
						break
			if an == -1:
				print('NO')
			else:
				print('YES')
