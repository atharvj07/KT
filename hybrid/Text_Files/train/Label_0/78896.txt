n = int(input())
mat = []
for i in range(n):
	a = [int(x) for x in input().split()]
	mat.append(a)
isp = 1
for i in range(n):
	for j in range(n):
		if mat[i][j] == 1:
			continue
		isb = 0
		for k in range(n):
			for m in range(n):
				if mat[i][j] == mat[i][k] + mat[m][j]:
					isb = 1
		if isb == 0:
			isp = 0
			break
if isp > 0:
	print('Yes')
else:
	print('No')
