def intersect(int1, int2):
	l = max(int1[0], int2[0])
	r = min(int1[1], int2[1])
	return (l, r)

def union(int1, int2):
	l = min(int1[0], int2[0])
	r = max(int1[1], int2[1])
	return (l, r)

def intervalLen(interval):
	(l, r) = interval
	return r - l

def solve(N, K, LR1, LR2):
	if N == 0:
		return inf
	dist = 0
	if LR1[1] < LR2[0]:
		dist = LR2[0] - LR1[1]
	elif LR2[1] < LR1[0]:
		(LR1, LR2) = (LR2, LR1)
		dist = LR2[0] - LR1[1]
	assert dist >= 0
	if dist == 0:
		iLen = intervalLen(intersect(LR1, LR2))
		if iLen * N >= K:
			return 0
		K -= iLen * N
		uLen = intervalLen(union(LR1, LR2))
		if (uLen - iLen) * N >= K:
			return K
		moves = (uLen - iLen) * N
		K -= moves
		moves += K * 2
		return moves
	else:

		def cost(numTouch):
			return numTouch * dist + solve(numTouch, K, [LR1[0], LR1[1] + dist], LR2)
