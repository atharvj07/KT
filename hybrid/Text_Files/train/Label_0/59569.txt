import java.io.*;
import java.util.*;

public class shooting {
	private static class pair {
		int x, y;

		pair(int x, int y) {
			this.x = x;
			this.y = y;
		}
	}

	public static void main(String args[]) {
		int count_shooting = 0, sum = 0;
		String space = "";
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		pair arr[] = new pair[n];
		for (int i = 0; i < n; i++) {
			arr[i] = new pair(i, sc.nextInt());
		}
		Arrays.sort(arr, new Comparator<pair>() {
			public int compare(pair p1, pair p2) {
				return p1.y - p2.y;
			}
		});
		for (int i = n - 1; i >= 0; i--) {
			sum += ((arr[i].y) * (count_shooting++)) + 1;
		}
		System.out.println(sum);
		for (int i = n - 1; i >= 0; i--) {
			System.out.print(space);
			System.out.print((arr[i].x) + 1);
			space = " ";
		}

	}
}
