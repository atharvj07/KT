#include <bits/stdc++.h>
using namespace std;
const int MAXN = (int)1e7 + 50;
int l, r;
int B[MAXN];
vector<int> SQ;
int main() {
  for (int i = (1); i < (int)(4000); i++) SQ.push_back(i * i);
  for (int t : SQ) {
    for (int p = t; p < MAXN; p += t) B[p] = t;
  }
  while (scanf("%d %d", &l, &r) >= 1) {
    r += l;
    long long ans = 0;
    for (int i = (l); i < (int)(r); i++) ans += i / B[i];
    printf("%lld\n", ans);
  }
  return 0;
}
