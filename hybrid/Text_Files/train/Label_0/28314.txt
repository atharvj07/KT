#include <bits/stdc++.h>
using namespace std;
template <class T>
inline T sqr(T x) {
  return x * x;
}
const double EPS = 1e-6;
const int INF = 0x3fffffff;
const long long LINF = INF * 1ll * INF;
const double PI = acos(-1.0);
using namespace std;
long long dp[55][55][2];
long long C[105][105];
int n, k;
long long Pow(int a, int p) {
  if (p == 1)
    return a;
  else {
    long long tmp = Pow(a, p / 2);
    tmp = tmp * tmp % 1000000007;
    if (p & 1) tmp = tmp * a % 1000000007;
    return tmp;
  }
}
const long long INV2 = Pow(2, 1000000007 - 2);
void add(long long &ans, long long v) {
  ans += v;
  if (ans >= 1000000007) ans -= 1000000007;
  if (ans < 0) ans += 1000000007;
}
long long dfs(int n, int k, int rt) {
  if (n == 1)
    return k == 0 && rt == 0;
  else if (n == 0)
    return k == 0 && rt == 1;
  if (~dp[n][k][rt])
    return dp[n][k][rt];
  else {
    long long &ans = dp[n][k][rt];
    ans = 0;
    for (int i = 0; i < n; i++) {
      int j = n - 1 - i;
      for (int ml = 0; ml <= k; ml++) {
        for (int rl = 0; rl < 2; rl++) {
          for (int rr = 0; rr < 2; rr++) {
            int mr = !(rl && rr);
            if (rt && mr == 0) continue;
            if (!rt && mr == 1) continue;
            mr = k - ml - mr;
            if (mr >= 0) {
              long long left = C[n][i] * dfs(i, ml, rl) % 1000000007;
              long long right = C[n - i][1] * dfs(j, mr, rr) % 1000000007;
              add(ans, left * right % 1000000007);
            }
          }
        }
      }
    }
    ans *= INV2;
    ans %= 1000000007;
    return ans;
  }
}
int main(void) {
  C[0][0] = 1;
  for (int i = 1; i <= 55; i++) {
    C[i][0] = 1;
    for (int j = 1; j <= i; j++) {
      C[i][j] = C[i - 1][j] + C[i - 1][j - 1];
      C[i][j] %= 1000000007;
    }
  }
  cin >> n >> k;
  memset(dp, -1, sizeof(dp));
  long long ans = dfs(n, k, 0) + dfs(n, k, 1);
  ans *= Pow(n, 1000000007 - 2);
  ans %= 1000000007;
  cout << ans << endl;
}
