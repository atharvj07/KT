import random
t = int(input())
for i in range(t):
	n = input()
	a = list(map(int, input().split()))
	fist = ''
	for i in range(max(a) + 1):
		t = chr(random.randrange(97, 122))
		while t == i:
			t = random.randrange(97, 122)
		fist += t
	print(fist)
	for i in a:
		ans = fist[:i]
		for j in fist[i:]:
			t = chr(random.randrange(97, 122))
			while t == j:
				t = chr(random.randrange(97, 122))
			ans = ans + t
			fist = ans
		print(fist)
