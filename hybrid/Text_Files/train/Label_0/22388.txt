#include <bits/stdc++.h>
using namespace std;
inline int gi() {
  char c;
  int x = 0, f = 1;
  for (; c < '0' || c > '9'; c = getchar())
    if (c == '-') f = -1;
  for (; c >= '0' && c <= '9'; c = getchar()) x = (x << 1) + (x << 3) + c - '0';
  return x * f;
}
const int N = 2e5 + 5;
int a[N];
bool p[N];
int main() {
  int n;
  scanf("%d", &n);
  int cnt = 0;
  for (int i = 2; i * i <= n; ++i) {
    if (!p[i]) {
      a[i] = ++cnt;
      for (int j = i * i; j <= n; j += i) p[j] = true, a[j] = cnt;
    }
  }
  for (int i = 2; i <= n; ++i) printf("%d ", a[i] ? a[i] : ++cnt);
}
