#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n], ans = 0, maxi = 0;
  for (int i = 0; i < 7; i++) {
    cin >> a[i];
    maxi += a[i];
  }
  int t;
  if (maxi >= n)
    t = 1;
  else if (n % maxi == 0) {
    t = n / maxi;
  } else {
    t = n / maxi + 1;
  }
  while (t--) {
    for (int i = 0; i < 7; i++) {
      n -= a[i];
      if (n <= 0) {
        ans = i + 1;
        break;
      }
    }
  }
  cout << ans << "\n";
  return 0;
}
