#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5, M = 2e3 + 5;
const long long INF = 1e18 + 5;
inline long long read() {
  long long sum = 0, fh = 1;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') fh = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') {
    sum = sum * 10 + c - '0';
    c = getchar();
  }
  return sum * fh;
}
inline int read2() {
  char c = getchar();
  while (c < '0' || c > '9') c = getchar();
  return c - '0';
}
inline int read3() {
  char c = getchar();
  while (c < 'a' || c > 'z') {
    c = getchar();
  }
  return c - 'a';
}
inline void write(long long x) {
  if (x < 0) putchar('-'), x = -x;
  if (x > 9) write(x / 10);
  putchar(x % 10 + '0');
}
inline int gcd(int x, int y) { return y == 0 ? x : gcd(y, x % y); }
inline long long ab(long long x) { return x < 0 ? -x : x; }
inline long long fpow(long long qwe, long long asd, long long zxc) {
  if (asd < 0) return 0;
  long long a = qwe, b = 1, c = asd;
  while (c) {
    if (c & 1) b = b * a % zxc;
    a = a * a % zxc;
    c >>= 1;
  }
  return b;
}
long long a[N], b[N];
map<pair<long long, long long>, long long> qwe;
map<long long, long long> asd;
int main() {
  long long t = read();
  while (t) {
    --t;
    int n = read();
    qwe.clear();
    asd.clear();
    for (int i = 1; i <= n; ++i) {
      a[i] = read();
      asd[a[i]] += 1;
    }
    int kx = 1;
    for (int i = 1; i <= n; ++i) {
      b[i] = read();
      if (!asd[b[i]]) kx = 0;
      asd[b[i]] -= 1;
    }
    if (!kx) {
      puts("No");
      continue;
    }
    for (int i = 1; i <= n / 2; ++i) {
      qwe[pair<long long, long long>(a[i], a[n - i + 1])] += 1;
    }
    for (int i = 1; i <= n / 2; ++i) {
      if (!qwe[pair<long long, long long>(b[i], b[n - i + 1])]) {
        if (!qwe[pair<long long, long long>(b[n - i + 1], b[i])]) {
          kx = 0;
          break;
        } else {
          qwe[pair<long long, long long>(b[n - i + 1], b[i])] -= 1;
        }
      } else {
        qwe[pair<long long, long long>(b[i], b[n - i + 1])] -= 1;
      }
    }
    if (n & 1) {
      if (a[n / 2 + 1] != b[n / 2 + 1]) kx = 0;
    }
    if (!kx)
      puts("No");
    else
      puts("Yes");
  }
  return 0;
}
