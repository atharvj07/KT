#include <bits/stdc++.h>
using namespace std;
string s,t;
int dp[3001][3001],n,m;
pair<int,int> path[3001][3001];
string ans;
void findpath(int i,int j){
	while(i>0&&j>0){
		if(s[i]==t[j]){
			ans+=s[i];
			i--;j--;
		}
		else{
			if(dp[i-1][j]>dp[i][j-1])	i--;
			else						j--;
		}
	}
}
int main(){
	cin>>s>>t;
	s=" "+s;t=" "+t;
	for(int i=1;i<s.size();i++)
		for(int j=1;j<t.size();j++){
			if(s[i]==t[j])
				dp[i][j]=dp[i-1][j-1]+1;
			else
				dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
		}
	findpath(s.size()-1,t.size()-1);
	reverse(ans.begin(),ans.end());
	cout<<ans<<endl;
}
