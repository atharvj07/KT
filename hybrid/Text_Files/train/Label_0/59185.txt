import java.io.File;
import java.util.Scanner;
import java.util.StringTokenizer;

public class p039 {
	public static void main(String args[]) throws Exception {
		StringTokenizer stok = new StringTokenizer(new Scanner(System.in).useDelimiter("\\A").next());
		StringBuilder sb = new StringBuilder();
		int n = Integer.parseInt(stok.nextToken());
		boolean[][] b = new boolean[51][51];
		int[] x = new int[4*n+1],y=new int[4*n+1];
		for(int i=0;i<4*n+1;i++) {
			x[i]=Integer.parseInt(stok.nextToken());
			y[i]=Integer.parseInt(stok.nextToken());
		}
		for(int i=0;i<50;i++) for(int j=0;j<50;j++) for(int k=2;k<=1+Math.min(50-i,50-j);k++) {
			set(b,i,j,k,true);
			int lx=0,ly=0,tot=0;
			for(int l=0;l<4*n+1;l++) if(!b[x[l]][y[l]]) {tot++;lx=x[l];ly=y[l];}
			set(b,i,j,k,false);
			if(tot==1) exit1(lx,ly);
		}
	}
	
	private static void set(boolean[][] b,int i,int j,int k,boolean val) {
		for(int l=i;l<i+k-1;l++) b[l][j] = val;
		for(int l=j;l<j+k-1;l++) b[i+k-1][l] =val;
		for(int l=i+k-1;l>i;l--) b[l][j+k-1] =val;
		for(int l=j+k-1;l>j;l--) b[i][l] =val;
	}

	private static void exit1(int i, int j) {
		System.out.println(i+ " "+j);
		System.exit(0);
	}
}
