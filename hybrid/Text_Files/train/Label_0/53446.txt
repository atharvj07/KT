import java.util.Scanner;

/**
 *
 * @author Krolos
 */
public class AI_love_Username {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Scanner k = new Scanner(System.in);
        int n = k.nextInt();
        int arr[] = new int[n];
        int sum = 0;
        int max = 0;
        int mun = 0;
        int n1;

        for (int i = 0; i < n; i++) {
            n1 = k.nextInt();
            arr[i] = n1;
        }
        max = arr[0];
        mun = arr[0];
        for (int i = 1; i < n; i++) {
            if (max < arr[i]) {
                max = arr[i];
                sum++;
            } else if (mun > arr[i]) {
                mun = arr[i];
                sum++;
            }
        }
        System.out.println(sum);
    }

}
