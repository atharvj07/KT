L = list(map(int, input().split()))
n = int(L[0])
m = int(L[1])
if m == 1:
	print(1)
elif m == 0:
	print(1)
elif m == n:
	print(0)
else:
	L = [2 for i in range(n)]
	for i in range(m):
		to_del = L.index(max(L))
		L[to_del] = -1
		if to_del == 0:
			L[1] -= 1
			L[n - 1] -= 1
		elif to_del == n - 1:
			L[0] -= 1
			L[n - 2] -= 1
		else:
			L[to_del - 1] -= 1
			L[to_del + 1] -= 1
	zero = L.count(0)
	one = int(L.count(1) / 2)
	print(zero + one)
