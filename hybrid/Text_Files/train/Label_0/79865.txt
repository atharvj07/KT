N = int(input())
p = list(map(int, input().strip().split(' ')))
a = list(map(int, input().strip().split(' ')))
b = list(map(int, input().strip().split(' ')))
m = int(input())
CM = list(map(int, input().strip().split(' ')))
checkout = [0 for i in range(N)]
shirt = [[p[i], i] for i in range(N)]
shirt = sorted(shirt)
NUM = [[] for i in range(3)]
for i in range(N):
	temp = shirt[i][1]
	num1 = a[temp] - 1
	num2 = b[temp] - 1
	NUM[num1] += [temp]
	if num1 != num2:
		NUM[num2] += [temp]
for i in range(3):
	NUM[i] = NUM[i][::-1]
ANS = []
for i in range(len(CM)):
	color = CM[i] - 1
	NO = 0
	if NUM[color] != []:
		ans = NUM[color].pop()
	else:
		NO = 1
	if NO == 0:
		while checkout[ans] == 1:
			if NUM[color] != []:
				ans = NUM[color].pop()
			else:
				NO = 1
				break
	if NO == 1:
		ANS += [-1]
	else:
		ANS += [p[ans]]
		checkout[ans] = 1
print(*ANS)
