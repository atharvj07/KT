#include <bits/stdc++.h>
using namespace std;
int n;
int num[100005];
int a, b;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> num[i];
  cin >> a >> b;
  sort(num, num + n);
  int len = unique(num, num + n) - num;
  int ans = 0;
  while (a > b) {
    int t = a - 1;
    for (int i = 0; i < len && i > -1; i++) {
      if (a - a % num[i] < b)
        num[i--] = num[--len];
      else
        t = min(t, a - a % num[i]);
    }
    ans++;
    a = t;
  }
  printf("%d\n", ans);
  return 0;
}
