import math
(n, k) = map(int, input().split())

def solve(i):
	a = i * k * (k + 1) // 2
	if n >= a:
		return n - a
	else:
		return -1
i = 1
r = -1
while i * i <= n:
	if n % i == 0:
		if solve(i) > -1:
			r = max(r, i)
		if i * i != n:
			if solve(n / i) > -1:
				r = max(r, n / i)
	i = i + 1
if r == -1:
	print(r)
else:
	b = []
	for i in range(1, k):
		b.append(i * r)
	b.append(k * r + solve(r))
	for i in b:
		print(int(i), end=' ')
	print()
