testCases = int(input())
for i1 in range(testCases):
	used = [False] * 26
	s = input()
	t = s[0]
	used[ord(s[0]) - ord('a')] = True
	pos = 0
	result = 'YES'
	for i2 in range(1, len(s), 1):
		if used[ord(s[i2]) - ord('a')]:
			if pos > 0 and t[pos - 1] == s[i2]:
				pos -= 1
			elif pos < len(t) - 1 and t[pos + 1] == s[i2]:
				pos += 1
			else:
				result = 'NO'
				break
		elif pos == 0:
			t = s[i2] + t
		elif pos == len(t) - 1:
			t += s[i2]
			pos += 1
		else:
			result = 'NO'
			break
		used[ord(s[i2]) - ord('a')] = True
	for i2 in range(26):
		if not used[i2]:
			t += chr(i2 + ord('a'))
	print(result)
	if result == 'YES':
		print(t)
