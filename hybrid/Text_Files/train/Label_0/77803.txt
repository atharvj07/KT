from sys import stdin
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
from collections import Counter
t = int(input())
for tt in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	c = Counter(a)
	c_k = c.keys()
	ans = 0
	for i in c_k:
		num = c[i + 1] + c[i + 2]
		ans += c[i] * (num * (num - 1) // 2)
		ans += c[i] * (c[i] - 1) // 2 * num
		ans += c[i] * (c[i] - 1) * (c[i] - 2) // 6
	print(ans)
