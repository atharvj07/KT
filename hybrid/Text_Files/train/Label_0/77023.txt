import sys
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
(n, m) = I()
a = I()
b = I()
an = 1
md = 998244353
suf = [0] * n
suf[-1] = a[n - 1]
for i in range(n - 2, -1, -1):
	suf[i] = min(suf[i + 1], a[i])
suf = [0] + suf
cr = m - 1
ct = 0
for i in range(n, -1, -1):
	if suf[i] < b[cr]:
		if ct == 0 or (i < n and suf[i + 1] != b[cr]):
			an = 0
			break
		if i == 0:
			cr -= 1
			break
		cr -= 1
		an = an * ct % md
		ct = 0
	if suf[i] == b[cr]:
		ct += 1
if ct == 0 or cr != -1:
	an = 0
print(an)
