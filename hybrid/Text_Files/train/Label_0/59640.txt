import java.util.*;

public class Week_5_Codeforces_A {
    public static void main(String[] args){
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int a = scan.nextInt();
        int b = scan.nextInt();
        int c = scan.nextInt();

        int i = 0;
        int j = 0;

        int max = 0;

        while(i*a + j*b <= n){
            while(i*a + j*b <= n){
                if(Math.abs((i*a + j*b) - n)%c == 0){
                    int k = Math.abs((i*a + j*b) - n)/c;
                    if(i + j + k > max){
                        max = i + j + k;
                    }
                }
                j++;
            }
            i++;
            j = 0;
        }
        System.out.println(max);
    }
}
