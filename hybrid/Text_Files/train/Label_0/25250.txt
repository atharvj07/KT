#include <bits/stdc++.h>
using namespace std;
template <class T>
T abs(T x) {
  return x > 0 ? x : -x;
}
template <class T>
T gcd(T a, T b) {
  return a ? gcd(b % a, a) : b;
}
template <class T>
T sqr(T a) {
  return a * a;
}
template <class T>
T sgn(T a) {
  return a > 0 ? 1 : (a < 0 ? -1 : 0);
}
int n;
int m;
int main() {
  long long k;
  cin >> n >> k;
  if (k > ((long long)n - 1) * n / 2) {
    printf("Impossible\n");
    return 0;
  }
  string ans;
  for (int i = 0; i <= n; i++) {
    if (k < i) {
      ans += string(i - k, ')') + (k > 0 ? "()" : "") + string(k, ')');
      break;
    }
    k -= i;
    ans += '(';
  }
  while (((int)(ans).size()) < 2 * n) ans += "()";
  cout << ans << endl;
  return 0;
}
