from dataclasses import dataclass
from itertools import combinations

@dataclass
class Seal:
	width: int
	height: int

	@property
	def area(self):
		return self.width * self.height

	def fits(self, other):
		return other.width <= self.width and other.height <= self.height or (other.height <= self.width and other.width <= self.height)

	@property
	def valid(self):
		return self.width >= 0 and self.height >= 0
(n, paper_width, paper_height) = map(int, input().split())
paper = Seal(paper_width, paper_height)
seals = [*combinations((Seal(*map(int, input().split())) for _ in range(n)), 2)]
seals.sort(key=lambda p: p[0].area + p[1].area, reverse=True)
area = 0
for (i, (first, second)) in enumerate(seals):
	subsections = list()
	rem_width = paper_width - first.width
	rem_height = paper_height - first.height
	if rem_width >= 0 and rem_height >= 0:
		subsections.extend((Seal(rem_width, paper_height), Seal(paper_width, rem_height)))
	rem_width_rot = paper_width - first.height
	rem_height_rot = paper_height - first.width
	if rem_width_rot >= 0 and rem_height_rot >= 0:
		subsections.extend((Seal(rem_width_rot, paper_height), Seal(paper_width, rem_height_rot)))
	for seal in subsections:
		if seal.fits(second):
			area = first.area + second.area
			break
	else:
		continue
	break
print(area)
