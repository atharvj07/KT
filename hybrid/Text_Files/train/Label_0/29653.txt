#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    string ans = "";
    if (n == 1)
      cout << -1 << endl;
    else {
      if (n == 2)
        cout << 23 << endl;
      else {
        ans = "23";
        for (int g = 2; g < n; g++) ans = ans + '9';
        cout << ans << endl;
      }
    }
  }
  return 0;
}
