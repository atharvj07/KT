(n, t, c) = map(int, input().strip().split())
arr = list(map(int, input().strip().split()))
all_seq = []
curr_seq = []
for i in arr:
	if i > t:
		if len(curr_seq) > 0:
			all_seq.append(len(curr_seq))
			curr_seq = []
	else:
		curr_seq.append(i)
if len(curr_seq) > 0:
	all_seq.append(len(curr_seq))
counter = 0
for i in all_seq:
	counter += max(0, i - c + 1)
print(counter)
