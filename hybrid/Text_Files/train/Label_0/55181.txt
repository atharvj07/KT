import java.util.*;
import java.io.*;

import static java.lang.Math.*;

public class D
{
	private static int [][][][] dp;
	private static int n;
	private static final int MOD = 1000000007;
	public static void main(String[] args) throws Exception
	{
		Scanner in = new Scanner(System.in);

		n = in.nextInt();

		char [][] map = new char[3][n];
		for(int i = 0 ; i < 3 ; i++)
			map[i] = in.next().toCharArray();
		int x = -1, y = -1;
		for(int i = 0 ; i < 3 ; i++)
			for(int j = 0 ; j < n ; j++)
				if(map[i][j] == 'O')
				{
					x = i;
					y = j;
					map[i][j] = '.';
				}


		boolean vertical = x != 1;
		for(int j = 0 ; j < 3 ; j++)
			vertical &= map[j][y] != 'X';

		int res = 0;
		if(vertical)
		{
			for(int j = 0 ; j < 3 ; j++)
				map[j][y] = 'X';

			res +=dynamic(map);

			for(int j = 0 ; j < 3 ; j++)
				map[j][y] = '.';
		}

		boolean horizontal1 = true;
		boolean horizontal2 = true;
		for(int i = y-2 ; i <= y ; i+=2)
		{
			if(i < 0)
			{
				horizontal1 = false;
				continue;
			}
			int j = i+2;
			if(j >= n)
			{
				horizontal2 = false;
				continue;
			}
			boolean tmp = true;
			for(int k = i ; k <= j ; k++)
			{
				if(i == y-2)
					horizontal1 &= map[x][k] != 'X';
				else
					horizontal2 &= map[x][k] != 'X';

				tmp &= map[x][k] != 'X';
			}
			if(tmp)
			{
				for(int k = i ; k <= j ; k++)
					map[x][k] = 'X';

				res += dynamic(map);

				for(int k = i ; k <= j ; k++)
					map[x][k] = '.';
			}
		}

		if(vertical & horizontal1)
		{
			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = 'X';
			map[x][y-2] = map[x][y-1] = 'X';

			res -= dynamic(map);

			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = '.';
			map[x][y-2] = map[x][y-1] = '.';
		}
		if(vertical & horizontal2)
		{
			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = 'X';
			map[x][y+1] = map[x][y+2] = 'X';

			res -= dynamic(map);

			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = '.';
			map[x][y+1] = map[x][y+2] = '.';
		}
		if(horizontal1 & horizontal2)
		{
			for(int i = y-2 ; i <= y+2 ; i++)
				map[x][i] = 'X';

			res -= dynamic(map);

			for(int i = y-2 ; i <= y+2 ; i++)
				map[x][i] = '.';
		}
		if(vertical && horizontal1 && horizontal2)
		{
			for(int i = y-2 ; i <= y+2 ; i++)
				map[x][i] = 'X';
			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = 'X';

			res += dynamic(map);

			for(int i = y-2 ; i <= y+2 ; i++)
				map[x][i] = '.';
			for(int i = 0 ; i < 3 ; i++)
				map[i][y] = '.';
		}

		res %= MOD;
		res += MOD;
		res %= MOD;
		System.out.println(res);

	}

	private static int dynamic(char[][] map) throws Exception
	{
		dp = new int[n+1][3][3][3];
		dp[0][0][0][0] = 1;

		for(int at = 0 ; at < n ; at++)
			for(int i = 0 ; i < 3 ; i++)
				for(int j = 0 ; j < 3 ; j++)
					for(int k = 0 ; k < 3 ; k++)
						if(dp[at][i][j][k] != 0)
						{
							if(i == 0)
							{
								if(map[0][at] == 'X')
								{
									int min = Math.min(i+1, Math.min(j, k));
									dp[at+min][i+1-min][j-min][k-min] += dp[at][i][j][k];
									dp[at+min][i+1-min][j-min][k-min] %= MOD;
								}
								else
								{
									if(j == 0 && map[1][at] == '.')
									{
										int min = Math.min(k, Math.min(i+1, j+1));
										dp[at+min][i+1-min][j+1-min][k-min] += dp[at][i][j][k];
										dp[at+min][i+1-min][j+1-min][k-min] %= MOD;
									}

									if(at + 1 < n && map[0][at+1] == '.')
									{
										int min = Math.min(i+2, Math.min(j, k));
										dp[at+min][i+2-min][j-min][k-min] += dp[at][i][j][k];
										dp[at+min][i+2-min][j-min][k-min] %= MOD;
									}

								}
							}
							else if(j == 0)
							{
								if(map[1][at] == 'X')
								{
									int min = Math.min(j+1, Math.min(i, k));
									dp[at+min][i-min][j+1-min][k-min] += dp[at][i][j][k];
									dp[at+min][i-min][j+1-min][k-min] %= MOD;
								}
								else
								{
									if(k == 0 && map[2][at] == '.')
									{
										int min = Math.min(i, Math.min(i+1, j+1));
										dp[at+min][i-min][j+1-min][k+1-min] += dp[at][i][j][k];
										dp[at+min][i-min][j+1-min][k+1-min] %= MOD;
									}

									if(at + 1 < n && map[1][at+1] == '.')
									{
										int min = Math.min(j+2, Math.min(i, k));
										dp[at+min][i-min][j+2-min][k-min] += dp[at][i][j][k];
										dp[at+min][i-min][j+2-min][k-min] %= MOD;
									}

								}
							}
							else if(k == 0)
							{
								if(map[2][at] == 'X')
								{
									int min = Math.min(k+1, Math.min(i, j));
									dp[at+min][i-min][j-min][k+1-min] += dp[at][i][j][k];
									dp[at+min][i-min][j-min][k+1-min] %= MOD;
								}
								else
								{
									if(at + 1 < n && map[2][at+1] == '.')
									{
										int min = Math.min(k+2, Math.min(i, j));
										dp[at+min][i-min][j-min][k+2-min] += dp[at][i][j][k];
										dp[at+min][i-min][j-min][k+2-min] %= MOD;
									}

								}
							}

							else
								throw new Exception();

						}

		return dp[n][0][0][0];
	}
}
