#include <bits/stdc++.h>
using namespace std;
int execute() {
  int n, b, a;
  cin >> n >> b >> a;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  int maxValA = a;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cnt++;
    if (v[i]) {
      if (b && a != maxValA) {
        b--;
        a++;
      } else {
        a--;
      }
    } else {
      if (a) {
        a--;
      } else {
        b--;
      }
    }
    if (!b && !a) {
      cout << cnt << endl;
      return 0;
    }
  }
  cout << n << endl;
  return 0;
}
int TE = true;
bool mTest = false;
int main() {
  ios_base::sync_with_stdio(false);
  if (mTest) cin >> TE;
  while (TE--) {
    execute();
  }
  return 0;
}
