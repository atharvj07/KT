(n, x, k) = map(int, input().split())
a = sorted(list(map(int, input().split())))
(ans, l, r) = (0, 0, 0)
for y in a:
	kol = (y // x - k) * x
	ml = min(kol, y)
	mr = min(kol + x, y)
	while l < n:
		if a[l] > ml:
			break
		l += 1
	while r < n:
		if a[r] > mr:
			break
		r += 1
	ans += r - l
print(ans)
