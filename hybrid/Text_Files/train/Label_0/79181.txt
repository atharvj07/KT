t = int(input())
while t:
	t -= 1
	n = int(input())
	l = []
	for i in range(n):
		l.append(list(map(int, input().split())))
	b = [False] * (n + 1)
	p = [0] * (n + 1)
	for i in range(n):
		for j in sorted(l[i][1:]):
			if not b[j]:
				p[i + 1] = j
				b[j] = True
				break
	if False in b[1:]:
		print('IMPROVE')
		print(p[1:].index(0) + 1, end=' ')
		print(b[1:].index(False) + 1)
	else:
		print('OPTIMAL')
