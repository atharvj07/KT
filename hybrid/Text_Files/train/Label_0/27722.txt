#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 10;
const int mod = 1e9 + 7;
int n, dp[N][21][2];
int f[21][2];
void predeal(int n) {
  int base = 1;
  for (int i = 0; i <= 20; i++) {
    f[i][0] = n / base;
    f[i][1] = n / (base * 3);
    base <<= 1;
  }
}
int main(void) {
  scanf("%d", &n);
  predeal(n);
  int X = log2(n);
  dp[1][X][0] = 1;
  if ((1 << (X - 1)) * 3 <= n) dp[1][X - 1][1] = 1;
  for (int i = 2; i <= n; i++) {
    for (int j = 0; j <= X; j++) {
      for (int k = 0; k <= 1; k++) {
        if (f[j][k] >= i - 1)
          dp[i][j][k] = (dp[i][j][k] +
                         (long long)dp[i - 1][j][k] * (f[j][k] - i + 1) % mod) %
                        mod;
        if (j)
          dp[i][j - 1][k] =
              (dp[i][j - 1][k] +
               (long long)dp[i - 1][j][k] * (f[j - 1][k] - f[j][k]) % mod) %
              mod;
        if (k)
          dp[i][j][k - 1] =
              (dp[i][j][k - 1] +
               (long long)dp[i - 1][j][k] * (f[j][k - 1] - f[j][k]) % mod) %
              mod;
      }
    }
  }
  printf("%d\n", dp[n][0][0]);
}
