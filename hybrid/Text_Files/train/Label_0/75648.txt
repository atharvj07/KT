ii = lambda : int(input())
si = lambda : input()
m = [0] * 367
f = [0] * 367
s = [0] * 367
n = ii()
for _ in range(n):
	q = si().split()
	(b, e) = map(int, q[1:])
	if q[0] == 'M':
		for i in range(b, e + 1):
			m[i] += 1
	else:
		for i in range(b, e + 1):
			f[i] += 1
for i in range(367):
	s[i] = min(m[i], f[i])
print(max(s) * 2)
