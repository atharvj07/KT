def solve(k, l, r, t, x, y):
	k -= l
	r -= l
	if r >= x + y - 1:
		if y < x:
			if k + y > r:
				k -= x
				return t - 1 <= int(k // (x - y))
			return t <= int(k // (x - y))
		return True
	if y < x:
		if k + y > r:
			if k - x < 0:
				return False
			k -= x
			return t - 1 <= int(k // (x - y))
		return t <= int(k // (x - y))
	if y == x:
		if k + y > r:
			return k >= x
		return True
	m = 0
	s = k
	while m < x and s % x <= r - y:
		m += 1
		s += y
	if m == x:
		return True
	return t <= s // x
(k, l, r, t, x, y) = map(int, input().split())
if solve(k, l, r, t, x, y):
	print('Yes')
else:
	print('No')
