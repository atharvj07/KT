t = int(input())
arr_input = []
for i in range(t):
	v = input().split(' ')
	arr_input.append([int(v[0]), int(v[1]), int(v[2])])
for i in arr_input:
	arr_output = []
	M = max(i)
	if i.count(M) > 1:
		for j in i:
			arr_output.append(str(M + 1 - j))
	else:
		for j in i:
			if j == M:
				arr_output.append('0')
			else:
				arr_output.append(str(M + 1 - j))
	print(' '.join(arr_output))
