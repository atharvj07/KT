for ad in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	l.sort()
	l.reverse()
	if n == 1:
		print('T')
		continue
	elif n == 2:
		if l[0] > l[1]:
			print('T')
		else:
			print('HL')
		continue
	t = 0
	i = 0
	j = 0
	while True:
		l.sort(reverse=True)
		if t == 0:
			if l[0] > 0:
				l[0] -= 1
			else:
				i = 1
				break
			if l[1] > 0:
				l[1] -= 1
			else:
				j = 1
				break
	if i == 1:
		print('HL')
	else:
		print('T')
