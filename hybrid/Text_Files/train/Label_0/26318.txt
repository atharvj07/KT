#include <bits/stdc++.h>
using namespace std;
const int maxn = 3005;
long long w[maxn][maxn];
long long dis[maxn][maxn];
struct Edge {
  int u, v;
  long long w;
};
bool operator<(const Edge& a, const Edge& b) { return a.w < b.w; }
vector<Edge> edges;
int main(void) {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= 1000 + 1; ++i)
    for (int j = 1; j <= 1000 + 1; ++j) dis[i][j] = (i == j ? 0 : 1e15);
  if (n > 400) {
    for (int i = 1; i <= m; ++i) {
      int u, v, w;
      scanf("%d%d%d", &u, &v, &w);
      edges.push_back(Edge{u, v, w});
    }
    sort(edges.begin(), edges.end());
    while ((int)edges.size() > k) edges.pop_back();
    vector<int> dot;
    int sz = edges.size();
    for (int i = 0; i < sz; i++) {
      dot.push_back(edges[i].u);
      dot.push_back(edges[i].v);
    }
    sort(dot.begin(), dot.end());
    dot.erase(unique(dot.begin(), dot.end()), dot.end());
    int len = dot.size();
    for (int i = 0; i < sz; i++) {
      edges[i].u =
          lower_bound(dot.begin(), dot.end(), edges[i].u) - dot.begin() + 1;
      edges[i].v =
          lower_bound(dot.begin(), dot.end(), edges[i].v) - dot.begin() + 1;
      dis[edges[i].u][edges[i].v] = edges[i].w;
      dis[edges[i].v][edges[i].u] = edges[i].w;
    }
    n = len;
  } else {
    for (int i = 1; i <= m; ++i) {
      int u, v, w;
      scanf("%d%d%d", &u, &v, &w);
      dis[u][v] = dis[v][u] = w;
    }
  }
  for (int k = 1; k <= n; ++k)
    for (int i = 1; i <= n; ++i)
      for (int j = 1; j <= n; ++j)
        dis[i][j] = min(dis[i][j], dis[i][k] + dis[j][k]);
  vector<long long> v;
  for (int i = 1; i <= n; ++i)
    for (int j = i + 1; j <= n; ++j) v.push_back(dis[i][j]);
  sort(v.begin(), v.end());
  cout << v[k - 1] << endl;
}
