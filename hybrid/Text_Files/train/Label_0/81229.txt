numero = int(input())
lista = [int(x) for x in input().split()]
listIndex = []
for i in range(len(lista)):
	menor = lista[i]
	index = i
	for j in range(i + 1, len(lista)):
		if lista[j] < menor:
			menor = lista[j]
			index = j
	else:
		if lista[i] != menor:
			listIndex.append([i, index])
			lista[index] = lista[i]
			lista[i] = menor
print(len(listIndex))
for i in range(len(listIndex)):
	print(listIndex[i][0], listIndex[i][1])
