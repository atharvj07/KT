for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort(reverse=True)
	bob = alice = 0
	cur = 0
	for i in range(n):
		if cur == 0:
			if a[i] % 2 == 0:
				alice += a[i]
			cur = 1
		else:
			if a[i] % 2 == 1:
				bob += a[i]
			cur = 0
	if alice == bob:
		print('Tie')
	elif alice > bob:
		print('Alice')
	else:
		print('Bob')
