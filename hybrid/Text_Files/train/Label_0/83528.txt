(x1, y1, x2, y2) = map(int, input().split(' '))
if x1 == x2:
	l = abs(y2 - y1)
	x3 = x1 + l
	x4 = x2 + l
	y3 = y1
	y4 = y2
	print(x3, y3, x4, y4)
elif y1 == y2:
	x3 = x1
	x4 = x2
	l = abs(x2 - x1)
	y3 = y1 + l
	y4 = y2 + l
	print(x3, y3, x4, y4)
elif abs(x2 - x1) != abs(y2 - y1):
	print(-1)
else:
	l = abs(x2 - x1)
	if y1 < y2:
		x3 = x1
		y3 = y1 + l
		x4 = x2
		y4 = y2 - l
	else:
		x3 = x1
		y3 = y1 - l
		x4 = x2
		y4 = y2 + l
	print(x3, y3, x4, y4)
