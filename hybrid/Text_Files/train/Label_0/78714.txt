import sys

def str_arr(arr):
	return ' '.join(map(str, arr))

def solve(n):
	answer = [i + 1 for i in range(n)]
	start = 0
	if n % 2 == 1:
		answer[:3] = [3, 1, 2]
		start = 3
	for j in range(start, n - 1, 2):
		(answer[j], answer[j + 1]) = (answer[j + 1], answer[j])
	return str_arr(answer)

def main():
	outputs = []
	fin = sys.stdin
	N = int(fin.readline())
	for _ in range(N):
		n = int(fin.readline())
		outputs.append(solve(n))
	print('\n'.join(map(str, outputs)))
main()
