from collections import *
import sys
input = sys.stdin.readline

def li():
	return [int(i) for i in input().rstrip('\n').split(' ')]

def val():
	return int(input())
n = val()
a = li()
tot = sum(a)
q = val()
d = {}
currcount = [0 for i in range(n)]
for i in range(q):
	l = li()
	if (l[0], l[1]) in d:
		val = d[l[0], l[1]]
		currcount[val - 1] -= 1
		tot += 1 if currcount[val - 1] < a[val - 1] else 0
		d.pop((l[0], l[1]))
	if l[2]:
		d[l[0], l[1]] = l[2]
		currcount[l[2] - 1] += 1
		tot -= 1 if currcount[l[2] - 1] <= a[l[2] - 1] else 0
	print(tot)
