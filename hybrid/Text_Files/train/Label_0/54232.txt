import java.util.ArrayList;
import java.util.Scanner;

public class Ahahaha {

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int T=sc.nextInt();
		while(T-->0) {
			int n=sc.nextInt();
			int arr[]=new int[n];
			//ArrayList<Integer> list=new ArrayList<Integer>();
			for(int i=0;i<n;i++) {
				int p=sc.nextInt();
				arr[i]=p;
			}
			int even = 0;
        	int odd = 0;
        	int one =0,zero = 0;
        	for(int i=0; i<n; i++) {
        		if(arr[i] == 1)
        			one++;
        		else
        			zero++;
        		if(i%2 == 0)
        			even+=arr[i];
        		else
        			odd+=arr[i];
        	}
        	if(even == odd) {
        		System.out.println(n);
        		for(int e : arr)
        			System.out.print(e+" ");
        		System.out.println();
        		//return;
        	}
        	else if(one > zero) {
        		System.out.println(n/2 + (n/2)%2);
        		one = n/2+(n/2)%2;
        		while(one-- != 0) {
        			System.out.print("1 ");
        		}
        		System.out.println();
        	}
        	else {
        		System.out.println(n/2);
        		zero = n/2;
        		while(zero-- != 0) {
        			System.out.print("0 ");
        		}
        		System.out.println();
			}
		}
		sc.close();
	}

}
