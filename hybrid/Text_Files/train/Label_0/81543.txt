(n, k) = map(int, input().split())
s = input()
l = [0] * 27
if k == 1:
	ans = set(list(s))
	c = 0
	for i in ans:
		c = max(c, s.count(i))
	print(c)
else:
	i = 1
	while i < n:
		c = 1
		if s[i] == s[i - 1]:
			c += 1
			for j in range(i + 1, n):
				if s[i] == s[j]:
					c += 1
				else:
					i = j - 1
					break
			if j == n - 1:
				i = n - 1
			l[ord(s[i]) - 97] += c // k
		i += 1
	print(max(l))
