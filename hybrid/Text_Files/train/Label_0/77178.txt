t = int(input())
import math
for i in range(t):
	n = int(input())
	l = list(map(int, input().split()))
	oddpos = []
	evenpos = []
	(flag1, flag2) = (0, 0)
	ans = 0
	for j in range(0, len(l), 2):
		evenpos.append(l[j])
	for j in range(1, len(l), 2):
		oddpos.append(l[j])

	def gcd(a, b):
		if b == 0:
			return a
		else:
			return gcd(b, a % b)
	resodd = oddpos[0]
	reseven = evenpos[0]
	for c in oddpos[1:]:
		resodd = gcd(resodd, c)
	for c in evenpos[1:]:
		reseven = gcd(reseven, c)
	if resodd != 1:
		for j in range(len(evenpos)):
			if evenpos[j] % resodd == 0:
				flag1 = 1
		if flag1 == 0:
			ans = resodd
	if reseven != 1:
		for j in range(len(oddpos)):
			if oddpos[j] % reseven == 0:
				flag2 = 1
		if flag2 == 0:
			ans = reseven
	print(ans)
