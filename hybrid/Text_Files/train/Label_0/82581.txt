(w, m) = map(int, input().split())

def b(x, base):
	digits = []
	while x:
		digits.append(x % base)
		x = int(x / base)
	return digits
digs = b(m, w)
f = -1
s = 0
while True:
	for i in range(s, len(digs)):
		if digs[i] not in {0, 1}:
			if digs[i] != w - 1:
				print('NO')
				exit()
			f = i
			break
	if f != -1 and f != len(digs) - 1:
		m += w ** f
		digs = b(m, w)
		s = f
		f = -1
	else:
		break
print('YES')
