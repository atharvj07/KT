t = int(input())
for inh in range(0, t):
	n = int(input())
	(row, column, c) = ([], [], 0)
	for inhg in range(0, n):
		s = input()
		if '*' in s and c < 2:
			for i in range(0, n):
				if s[i] == '*':
					row.append(inhg)
					column.append(i)
					c += 1
	if row[0] == row[1]:
		if row[0] == 0:
			row.append(1)
			row.append(1)
		else:
			row.append(0)
			row.append(0)
	elif column[0] == column[1]:
		if column[0] == 0:
			column.append(1)
			column.append(1)
		else:
			column.append(0)
			column.append(0)
	else:
		row.append(row[0])
		row.append(row[1])
		column.append(column[1])
		column.append(column[0])
	for i in range(0, n):
		if i in row:
			for j in range(0, n):
				if j in column:
					print('*', end='')
				else:
					print('.', end='')
			print('')
		else:
			print('.' * n)
