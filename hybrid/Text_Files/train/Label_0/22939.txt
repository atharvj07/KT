#include <bits/stdc++.h>
using namespace std;
int change(char x) {
  if (x == 'I' || x == 'E' || x == 'A' || x == 'O' || x == 'U' || x == 'Y')
    return 1;
  else
    return 0;
}
char a[500005];
int b[500005];
double pre[500005], sum[500005];
double ans;
int main() {
  gets(a);
  int n = strlen(a);
  for (int i = 0; i < n; i++) {
    b[i] = change(a[i]);
  }
  pre[0] = 0, sum[0] = 0;
  for (int i = 1; i <= n; i++) {
    pre[i] = pre[i - 1] + 1.0 / i;
    sum[i] = sum[i - 1] + pre[i];
  }
  for (int i = 0; i < n; i++) {
    if (b[i] == 1) {
      ans += sum[n] - sum[n - i - 1] - sum[i];
    }
  }
  printf("%.7f\n", ans);
}
