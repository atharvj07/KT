import math

def solve(n, p, k, a):
	lastbuy = False
	a.sort()
	oddsum = 0
	evensum = 0
	oddcnt = 0
	evencnt = 0
	for i in range(math.ceil(n / 2)):
		if a[2 * i] + oddsum <= p:
			oddsum += a[2 * i]
			oddcnt += 1
		else:
			if n % 2 == 1 and i == math.ceil(n / 2) - 1:
				continue
			if a[2 * i + 1] + oddsum <= p:
				lastbuy = True
			else:
				lastbuy = False
		if n % 2 == 1 and i == math.ceil(n / 2) - 1:
			continue
		if a[2 * i + 1] + evensum <= p:
			evensum += a[2 * i + 1]
			evencnt += 1
	oddgoods = 2 * oddcnt if lastbuy else 2 * oddcnt - 1
	evengoods = 2 * evencnt
	ans = max(oddgoods, evengoods)
	print(ans)
t = int(input())
for _ in range(t):
	(n, p, k) = map(int, input().split())
	a = list(map(int, input().split()))
	solve(n, p, k, a)
