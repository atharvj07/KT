n = int(input())
dictx = {}
flag = True
ans = []
for i in range(n):
	k = int(input())
	seq = list(map(int, input().split()))
	if flag:
		sumx = sum(seq)
		for j in range(k):
			key = sumx - seq[j]
			if key in dictx.keys():
				if dictx[key][0] != i + 1:
					ans.append(dictx[key])
					ans.append([i + 1, j + 1])
					flag = False
					break
			else:
				dictx[key] = [i + 1, j + 1]
if flag:
	print('NO')
else:
	print('YES')
	for i in ans:
		print(*i)
