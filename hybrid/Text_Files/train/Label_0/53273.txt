
import java.awt.Point;
import java.awt.geom.Line2D;
import java.awt.geom.Point2D;
import java.awt.geom.Point2D.Double;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.StringTokenizer;

public class E {
	public static void main(String[] args) {
		MyScanner scan = new MyScanner();
		int N = scan.nextInt(), A = scan.nextInt(), B = scan.nextInt();
		Point2D.Double[] list = new Point2D.Double[N+2];
		int maxA=-1, maxB=-1;
		for(int i=0;i<N;i++){
			list[i]=new Point2D.Double(scan.nextInt(), scan.nextInt());
			maxA=Math.max(maxA, (int)list[i].getX());maxB=Math.max(maxB, (int)list[i].y);
		}
		list[N]=new Point2D.Double(0, maxB);list[N+1]=new Point2D.Double(maxA, 0);
		
		Point2D.Double[] hull = new ConvexHull(list).getHull();
		//for(Point2D.Double p : hull)System.out.println(p.x+" "+p.y);
		
		double ans = Integer.MAX_VALUE;
		Line2D.Double ray = new Line2D.Double(0,0,(double)A*10000,(double)B*10000);
		for(int i=1;i<=hull.length;i++){
			Point2D.Double sect = getSegInter(ray, new Line2D.Double(hull[i-1],hull[i%hull.length]));
			if(sect!=null)
				ans = Math.min(ans, Math.max((double)A/sect.getX(), (double)B/sect.getY()));
		}
		System.out.println(ans);
	}
	private static double cross(Point2D.Double a, Point2D.Double b){
		return a.getX()*b.getY()-a.getY()*b.getX();
	}
	private static Point2D.Double getSegInter(Line2D.Double a, Line2D.Double b){
		double EPS = 1e-9;
		Point2D.Double v1 = new Point2D.Double(a.getX2()-a.getX1(), a.getY2()-a.getY1());
		Point2D.Double v2 = new Point2D.Double(-(b.getX2()-b.getX1()), -(b.getY2()-b.getY1()));
		Point2D.Double v3 = new Point2D.Double(b.getX1()-a.getX1(), b.getY1()-a.getY1());
		double num1 = cross(v3, v2);
		double num2 = cross(v1, v3);
		double den = cross(v1,v2);
		if(den==0)return null;
		double lamda = num1/den;
		double beta = num2/den;
		if(lamda-EPS < 1&&lamda+EPS > 0&&beta-EPS < 1 && beta+EPS>0)
			return new Point2D.Double(a.getX1()+(a.getX2()-a.getX1())*lamda, a.getY1()+(a.getY2()-a.getY1())*lamda);
		return null;
	}
	private static class ConvexHull {
		Point2D.Double[] hull;
		public Point2D.Double[] getHull(){
			return hull;
		}
		public static long cross(Point2D.Double O, Point2D.Double A, Point2D.Double B){
			return (long) ((A.x-O.x)*(B.y-O.y)-(A.y-O.y)*(B.x-O.x));
		}
		public ConvexHull(Point2D.Double[] p){
			if(p.length>1){
				int n = p.length, k=0;
				hull=new Point2D.Double[2*n];
				Arrays.sort(p, new Comparator<Point2D.Double>(){
					public int compare(Point2D.Double p1, Point2D.Double p2) {
						if(p1.x<p2.x)return -1;
						else if(p1.x>p2.x)return 1;
						else{
							if(p1.y<p2.y)return -1;
							else if(p1.y>p2.y)return 1;
							else return 0;
						}
					}
				});
				for(int i=0;i<n;++i){
					while(k>=2&&cross(hull[k-2],hull[k-1], p[i])<=0)
						k--;
					hull[k++]=p[i];
				}
				for(int i=n-2,t=k+1;i>=0;i--){
					while(k>=t&&cross(hull[k-2],hull[k-1],p[i])<=0)
						k--;
					hull[k++]=p[i];
				}
				if(k>1)hull=Arrays.copyOfRange(hull,0,k-1);
			}
			else if(p.length<=1)hull=p;
			else hull=null;
		}
		
	}
	private static class MyScanner {
	    BufferedReader br;
	    StringTokenizer st;
	    public MyScanner(){
	       br = new BufferedReader(new InputStreamReader(System.in));
	    }	
	    String next() {
	        while (st == null || !st.hasMoreElements()) {
	            try {st = new StringTokenizer(br.readLine());} 
	            catch (IOException e) {e.printStackTrace();}
	        }
	        return st.nextToken();
	    }
	    int nextInt() {
	        return Integer.parseInt(next());
	    }
	 }
}
