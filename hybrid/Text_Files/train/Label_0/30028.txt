#include <bits/stdc++.h>
using namespace std;
const int N = (1 << 20);
int dp[N];
vector<int> moves;
void add(vector<int> a) {
  for (int i = 0; i < int(a.size()); ++i) {
    int msk = 0;
    for (int j = i; j < int(a.size()); ++j) {
      msk |= (1 << a[j]);
      moves.push_back(msk);
    }
  }
}
int go(int msk) {
  int &ans = dp[msk];
  if (ans != -1) return ans;
  ans = 0;
  for (int x : moves) {
    int aux = msk & x;
    if (aux > 0) continue;
    int res = go(msk | x);
    if (res == 0) ans = 1;
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  add(vector<int>({0, 1, 2}));
  add(vector<int>({3, 4, 5, 6}));
  add(vector<int>({7, 8, 9, 10, 11}));
  add(vector<int>({12, 13, 14, 15}));
  add(vector<int>({16, 17, 18}));
  add(vector<int>({0, 3, 7}));
  add(vector<int>({1, 4, 8, 12}));
  add(vector<int>({2, 5, 9, 13, 16}));
  add(vector<int>({6, 10, 14, 17}));
  add(vector<int>({11, 15, 18}));
  add(vector<int>({2, 6, 11}));
  add(vector<int>({1, 5, 10, 15}));
  add(vector<int>({0, 4, 9, 14, 18}));
  add(vector<int>({3, 8, 13, 17}));
  add(vector<int>({7, 12, 16}));
  int id = 0, msk = 0;
  for (int i = 0; i < 19; ++i) {
    char ch;
    cin >> ch;
    if (ch == '.' || ch == 'O') {
      if (ch == '.') msk |= (1 << id);
      id++;
    }
  }
  fill(dp, dp + N, -1);
  dp[(1 << 19) - 1] = 0;
  if (go(msk))
    cout << "Karlsson" << endl;
  else
    cout << "Lillebror" << endl;
  return 0;
}
