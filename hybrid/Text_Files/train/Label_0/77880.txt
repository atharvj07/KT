import math
t = int(input())
for i in range(t):
	(n, c) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	days = 0
	cur = 0
	bal = 0
	ans = int(1e+18)
	for i in range(n):
		ans = min(ans, max(0, cur + math.ceil((c - bal) / a[i])))
		if i < n - 1:
			nd = max(0, math.ceil((b[i] - bal) / a[i]))
			bal += a[i] * nd - b[i]
		cur += nd + 1
	print(ans)
