import java.util.*;
import java.math.*;
import java.io.*;

public class Main {
    static ArrayList<Integer>[] e;
    static ArrayList<Integer> cy=new ArrayList<>();
    static int[] vis;
    static int[] pr;
    
    public static void dfs(int v,int p){
        
        vis[v]=1;
        for(int ad:e[v]){
            if(vis[ad]==1&&ad!=p){
                    int vr=v;cy.add(ad);
                    while(vr!=ad&&vr!=-1){
                        cy.add(vr);vr=pr[vr];
                        
                    }
                    return;
            }
        }
        for(int ad:e[v]){
            if(vis[ad]==0)
            {
                pr[ad]=v;
                dfs(ad,v);
            }
        }
    }
    public static void main(String[] args){
         
    Scanner sc = new Scanner(System.in);
    StringBuilder op = new StringBuilder();
    int n=sc.nextInt();
    e = new ArrayList[n+1];
    for(int i=1;i<=n;i++){
        int u=sc.nextInt();
        int v=sc.nextInt();
        if(e[v] == null){
        	e[v] = new ArrayList<>();
        }
        if(e[u] == null){
        	e[u] = new ArrayList<>();
        }
        e[v].add(u);
        e[u].add(v);
    } 
    pr=new int[n+1];
    Arrays.fill(pr, -1);
    vis=new int[n+1];
    pr[1]=0;
    dfs(1,0);
    int[] dis=new int[n+1];
    Arrays.fill(dis, Integer.MAX_VALUE);
    for(int i:cy)
        dis[i] = 0;
    for(int i:cy){
        Queue<Integer> q = new LinkedList<>();
        q.add(i);
        vis=new int[n+1];
        vis[i] = 1;
        dis[i] = 0;
        while(!q.isEmpty())
        {
            int v=q.poll();
            for(int ad:e[v])
            {    
            	if(vis[ad]==0){
            		q.add(ad);vis[ad]=1;
            	}
                dis[ad]=Math.min(dis[ad], dis[v]+1);
            }
            
        }
    }
    for(int i : cy){
    	dis[i]=0;
    }
    for(int i=1;i<=n;i++){
    	op.append(dis[i] +" ");
    }
    System.out.println(op);   
    }
}
