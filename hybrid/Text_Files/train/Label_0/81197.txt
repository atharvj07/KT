import sys
input = sys.stdin.readline

def put():
	return map(int, input().split())

def dfs():
	t = 0
	s = [(1, 1)]
	while s:
		(i, state) = s.pop()
		if state:
			t += 1
			tin[i] = t
			s.append((i, 0))
			for j in tree[i]:
				if parent[j] == 0:
					parent[j] = i
					height[j] = height[i] + 1
					s.append((j, 1))
		else:
			t += 1
			tout[i] = t

def check(i, j):
	return tin[i] <= tin[j] and tout[j] <= tout[i]
(n, m) = put()
(height, tin, tout, parent) = ([0] * (n + 1), [0] * (n + 1), [0] * (n + 1), [0] * (n + 1))
tree = [[] for i in range(n + 1)]
for i in range(n - 1):
	(x, y) = put()
	tree[x].append(y)
	tree[y].append(x)
parent[1] = 1
dfs()
for i in range(m):
	l = list(put())
	k = l[0]
	for j in range(1, k + 1):
		l[j] = parent[l[j]]
	d = 1
	for j in range(1, k + 1):
		if height[d] < height[l[j]]:
			d = l[j]
	ans = True
	for j in range(1, k + 1):
		ans = ans and check(l[j], d)
	ans = 'YES' if ans else 'NO'
	print(ans)
