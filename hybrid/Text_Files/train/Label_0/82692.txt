(n, k) = [int(i) for i in input().split()]
a = [int(i) for i in input().split()]
a.sort()
count = 0
i = 0
if k == 0:
	if a[0] != 1:
		print(1)
	else:
		print(-1)

while i < n:
	j = i + 1
	while j < n and a[j] == a[i]:
		j += 1
	if j == k:

		quit()
	if j > k:
		break
