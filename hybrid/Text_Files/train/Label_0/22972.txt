#include <bits/stdc++.h>
using namespace std;
const int N = 200005;
long long x[N], d[N], t[N], p[N];
vector<long long> points;
vector<pair<long long, long long> > V[N];
long long cost[N];
int comp(int x) {
  return lower_bound((points).begin(), (points).end(), x) - points.begin();
}
long long dist[N];
int pre[N], how[N];
vector<int> used;
void print(int x) {
  if (x == comp(0)) return;
  print(pre[x]);
  if (how[x]) used.push_back(how[x]);
}
int main() {
  int n, L;
  scanf("%d %d", &n, &L);
  points.push_back(0);
  points.push_back(L);
  for (int i = 0; i < n; i++) {
    scanf("%lld %lld %lld %lld", &x[i], &d[i], &t[i], &p[i]);
    if (x[i] - p[i] < 0) continue;
    points.push_back(x[i] - p[i]);
    points.push_back(x[i] + d[i]);
  }
  sort((points).begin(), (points).end());
  points.erase(unique((points).begin(), (points).end()), points.end());
  for (int i = 1; i < (int)points.size(); i++) {
    V[i].push_back(make_pair(i - 1, points[i] - points[i - 1]));
    V[i - 1].push_back(make_pair(i, points[i] - points[i - 1]));
  }
  for (int i = 0; i < N; i++) {
    cost[i] = p[i] + t[i];
    if (x[i] - p[i] < 0) continue;
    V[comp(x[i] - p[i])].push_back(make_pair(comp(x[i] + d[i]), -i));
  }
  memset(dist, 0x3f, sizeof dist);
  priority_queue<pair<long long, long long> > Q;
  Q.push(make_pair(0, 0));
  dist[0] = 0;
  while (!Q.empty()) {
    auto p = Q.top();
    Q.pop();
    if (-p.first > dist[p.second]) continue;
    for (auto q : V[p.second]) {
      int j = q.first;
      if (q.second <= 0) {
        long long c = p.first - cost[-q.second];
        if (-c < dist[j]) {
          dist[j] = -c;
          pre[j] = p.second;
          how[j] = -q.second + 1;
          Q.push(make_pair(c, j));
        }
      } else {
        long long c = p.first - q.second;
        if (-c < dist[j]) {
          dist[j] = -c;
          pre[j] = p.second;
          how[j] = 0;
          Q.push(make_pair(c, j));
        }
      }
    }
  }
  cout << dist[(int)points.size() - 1] << endl;
  print((int)points.size() - 1);
  cout << used.size() << endl;
  for (int r : used) cout << r << " ";
  cout << endl;
  return 0;
}
