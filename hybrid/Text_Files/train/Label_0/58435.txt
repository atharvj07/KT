/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main
{
  public static void main(String[] args)
  {
    InputStream inputStream = System.in;
    OutputStream outputStream = System.out;
    InputReader in = new InputReader(inputStream);
    PrintWriter out = new PrintWriter(outputStream);
    C1 solver = new C1();
    solver.solve(1, in, out);
    out.close();
  }

  static class C1
  {
    public void solve(int testNumber, InputReader in, PrintWriter out)
    {
      int n = in.nextInt();
      int m = in.nextInt();
      long bounty = in.nextLong();
      long increase = in.nextLong();
      long damage = in.nextLong();
      long[] mhp = new long[n + 1];
      long[] reg = new long[n + 1];
      Query[] lqry = new Query[m + n];
      for (int i = 1; i <= n; ++i)
      {
        mhp[i] = in.nextLong();
        long hp = in.nextLong();
        reg[i] = in.nextLong();
        if (mhp[i] <= damage && increase != 0)
        {
          out.println("-1");
          return;
        }
        lqry[i - 1] = new Query(0, i, hp);
      }
      for (int i = 0; i < m; ++i)
      {
        long time = in.nextLong();
        int enemy = in.nextInt();
        long health = in.nextLong();
        lqry[n + i] = new Query(time, enemy, health);
      }
      Arrays.sort(lqry);
      /*for(int i=0;i<lqry.length;++i)
         out.println(lqry[i].time+" "+lqry[i].enemy+" "+lqry[i].health);
      */
      long[] lastSeen = new long[n + 1];
      Arrays.fill(lastSeen, (long) 1e9 + 1);
      ArrayList<Event> events = new ArrayList<>();
      for (int i = 0; i < lqry.length; ++i)
      {
        long hi = finishA(lqry[i].time, lqry[i].health, reg[lqry[i].enemy], damage, Math.min(damage, mhp[lqry[i].enemy]));
        int e = lqry[i].enemy;
        if (lastSeen[e] == 1e9 + 1)
        {
          if (hi != -1)
          {
            // add segment ... for the first time
            if (reg[lqry[i].enemy] == 0 && increase != 0)
            {
              out.println("-1");
              return;
            }
            events.add(new Event(lqry[i].time, 1));
            events.add(new Event(hi, -1));
          }
        } else
        {
          if (hi != -1)
          {
            // add segment ... consider the last insertion
            events.add(new Event(lqry[i].time, 1));
            events.add(new Event(Math.min(lastSeen[e] - 1, hi), -1));
          }
        }
        lastSeen[e] = lqry[i].time;
      }
      long ans = 0;
      Collections.sort(events);
      /*for(Event e: events)
      {
        out.println(e.time+" "+e.what);
      }*/
      long cant = 0;
      for (int i = 0; i < events.size(); )
      {
        if (events.get(i).what == 1)
        {
          int j = i;
          while (j < events.size() && events.get(i).time == events.get(j).time && events.get(j).what == 1)
          {
            ++j;
            ++cant;
          }
          ans = Math.max(ans, bounty(bounty, increase, events.get(i).time, cant));
          i = j;
        } else
        {
          int j = i;
          ans = Math.max(ans, bounty(bounty, increase, events.get(i).time, cant));
          while (j < events.size() && events.get(i).time == events.get(j).time && events.get(j).what == -1)
          {
            ++j;
            --cant;
          }
          i = j;
        }
      }
      out.println(ans);
    }

    long bounty(long B, long increase, long t, long cant)
    {
      return cant * (B + increase * t);
    }

    long finishA(long x, long y, long r, long dmg, long ymx)
    {
      if (y > dmg) return -1;
      if (r == 0) return (long) 1e9 + 1;
      return (ymx - y) / r + x;
    }

    class Query implements Comparable
    {
      long time;
      long health;
      int enemy;

      public Query(long time, int enemy, long health)
      {
        this.time = time;
        this.enemy = enemy;
        this.health = health;
      }


      public int compareTo(Object o)
      {
        Query p = (Query) o;
        return -1 * Long.compare(time, p.time);
      }

    }

    class Event implements Comparable
    {
      long time;
      long what;

      public Event(long time, long what)
      {
        this.time = time;
        this.what = what;
      }


      public int compareTo(Object o)
      {
        Event e = (Event) o;
        if (time != e.time)
          return Long.compare(time, e.time);
        return -1 * Long.compare(what, e.what);
      }

    }

  }

  static class InputReader
  {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReader(InputStream stream)
    {
      reader = new BufferedReader(new InputStreamReader(stream), 32768);
      tokenizer = null;
    }

    public String next()
    {
      while (tokenizer == null || !tokenizer.hasMoreTokens())
      {
        try
        {
          tokenizer = new StringTokenizer(reader.readLine());
        } catch (IOException e)
        {
          throw new RuntimeException(e);
        }
      }
      return tokenizer.nextToken();
    }

    public int nextInt()
    {
      return Integer.parseInt(next());
    }

    public long nextLong()
    {
      return Long.parseLong(next());
    }

  }
}
