#include <bits/stdc++.h>
using namespace std;
const int N = 1000;
long long n, m, ans, tmp;
char c[N][N];
int vis[N];
vector<int> adj[N];
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> c[i][j];
      if (c[i][j] == '.') c[i][j] = 'D';
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (c[i][j] == 'S') {
        if (i > 0 && c[i - 1][j] == 'W') return cout << "No", 0;
        if (i < n - 1 && c[i + 1][j] == 'W') return cout << "No", 0;
        if (j > 0 && c[i][j - 1] == 'W') return cout << "No", 0;
        if (j < m - 1 && c[i][j + 1] == 'W') return cout << "No", 0;
      }
    }
  }
  cout << "Yes"
       << "\n";
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cout << c[i][j];
    }
    cout << "\n";
  }
  return 0;
}
