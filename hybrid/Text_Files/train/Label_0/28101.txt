#include <bits/stdc++.h>
using namespace std;
const long long MOD = (119 << 23) + 1;
long long a[20], k;
pair<long long, long long> f[20][1 << 10];
long long pow_[20];
inline long long calc(long long x) {
  long long res = 0;
  while (x) {
    if (x & 1) ++res;
    x >>= 1;
  }
  return res;
}
pair<long long, long long> dp(long long pos, long long sta, bool lead,
                              bool limit) {
  if (pos == -1) return pair<long long, long long>(1, 0);
  if (!limit && !lead && f[pos][sta].second != -1) return f[pos][sta];
  long long up = limit ? a[pos] : 9;
  pair<long long, long long> res = pair<long long, long long>(0, 0);
  for (long long i = 0; i <= up; ++i) {
    pair<long long, long long> tmp;
    if (lead && i == 0) {
      tmp = dp(pos - 1, sta, true, limit && i == up);
      res.first = (res.first + tmp.first) % MOD;
      res.second =
          (res.second + tmp.second + (1LL * i * pow_[pos]) * tmp.first) % MOD;
    } else {
      long long cnt = calc(sta | (1 << i));
      if (cnt > k) continue;
      tmp = dp(pos - 1, sta | (1 << i), lead && i == 0, limit && i == up);
      res.first = (res.first + tmp.first) % MOD;
      res.second = (res.second + tmp.second +
                    ((i * pow_[pos] % MOD) * tmp.first) % MOD) %
                   MOD;
    }
  }
  if (!limit && !lead) f[pos][sta] = res;
  return res;
}
inline long long solve(long long x) {
  long long pos = 0;
  while (x) {
    a[pos++] = x % 10;
    x /= 10;
  }
  return dp(pos - 1, 0, true, true).second;
}
int32_t main() {
  long long x, y;
  pow_[0] = 1;
  for (long long i = 1; i < 20; ++i) pow_[i] = pow_[i - 1] * 10 % MOD;
  memset(f, -1, sizeof(f));
  scanf("%I64d%I64d%I64d", &x, &y, &k);
  printf("%I64d\n", (solve(y) - solve(x - 1) + MOD) % MOD);
  return 0;
}
