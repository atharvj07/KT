def solve():
	(n, k, d) = map(int, input().split())
	a = [int(i) for i in input().split()]
	min = 100
	q = 0
	for i in range(n - d + 1):
		temp = len(set(a[i:i + d]))
		if temp == 1:
			q += 1
		elif temp < min:
			min = temp
	if n == d:
		min = len(set(a))
	print(1 if q > 0 else min)

def main():
	t = int(input())
	for z in range(t):
		solve()
main()
