#include<string>
#include<iostream>
#include<vector>
#include<map>
#include<algorithm>
#define all(c) c.begin(),c.end()
#define pb push_back
using namespace std;
typedef pair<int,string> P;
int main(){
	int n;
	while(cin>>n,n){
		vector<P> ans(n);
		vector<vector<int> > v(n);
		map<int,int> mp;
		int m,x;
		for(int i=0;i<n;i++){
			string s;
			cin>>s>>m;
			ans[i]=P(0,s);
			for(int j=0;j<m;j++){
				cin>>x;
				v[i].pb(x);
				mp[x]++;
			}
		}
		for(int i=0;i<n;i++){
			for(int j=0;j<v[i].size();j++){
				ans[i].first=ans[i].first+n-mp[v[i][j]]+1;
			}
		}
		sort(all(ans));
		cout<<ans[0].first<<" "<<ans[0].second<<endl;
	}
	return 0;
}
