import sys
input = sys.stdin.readline

def solve():
	(n, m) = map(int, input().split())
	e = dict()
	same = 0
	diff = 0
	res = []
	for i in range(m):
		a = input().split()
		c = a[0]
		if c == '+':
			u = int(a[1])
			v = int(a[2])
			c = a[3]
			z = e.get((v, u), None)
			if z == c:
				same += 1
			elif z is not None:
				diff += 1
			e[u, v] = c
		elif c == '-':
			u = int(a[1])
			v = int(a[2])
			c = e[u, v]
			del e[u, v]
			z = e.get((v, u), None)
			if z == c:
				same -= 1
			elif z is not None:
				diff -= 1
		else:
			c = int(a[1])
			if c % 2 == 1:
				if diff + same > 0:
					res.append('YES')
				else:
					res.append('NO')
			elif same > 0:
				res.append('YES')
			else:
				res.append('NO')
	print('\n'.join(res))
solve()
