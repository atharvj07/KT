#include <iostream>
#include <cstdio>
#include <climits>
using namespace std;

#define REP(i,n,m) for(int i=n;i<m;i++)
#define rep(i,n) REP(i,0,n)

int n,U;
int ans;
int c[22],need[22]; //eÈÚÌPÊ, C·éÌÉKvÈÈÚ(rbgÇ)

//ÈÚÌæèûÉµªÈ¢©Ç¤©`FbN·é
bool check(int used){
  rep(i,n){
    //CÈÚÉKvÈÈÚÌPÊªæêÄé©`FbN
    if((used & (1<<i)) > 0 && (used & need[i]) != need[i]){
      return false;
    }
  }
  return true;
}

//id:ÈÚÔ, used:ÇÌÈÚæÁ½©(rbgÇ), sum:PÊv, cnt:CÈÚ
void solve(int id,int used,int sum,int cnt){
  if(id == n){
    if(sum >= U && ans > cnt && check(used)){
      ans = cnt;
    }
    return;
  }

  solve(id+1,used,sum,cnt); //ÆçÈ¢
  solve(id+1,used|(1<<id),sum+c[id],cnt+1); //Æé
}

int main(void){
  while(scanf("%d%d",&n,&U),n||U){
    rep(i,n){
      int k;
      scanf("%d%d",&c[i],&k);
      need[i] = 0;

      while(k--){
        int r;
        scanf("%d",&r);
        need[i] |= (1<<r);
      }
    }

    ans = INT_MAX;
    solve(0,0,0,0);
    printf("%d\n",ans);
  }

  return 0;
}
