from sys import stdin
from collections import Counter
input = stdin.readline
for test in range(int(input())):
	n = int(input())
	lst = list(map(int, input().strip().split()))
	N = max(lst) + 1
	po = [0] * N
	cnt = Counter(lst)
	for k in range(1, N):
		po[k] += cnt.get(k, 0)
		for i in range(2 * k, N, k):
			po[i] = max(po[i], po[k])
	print(n - max(po))
