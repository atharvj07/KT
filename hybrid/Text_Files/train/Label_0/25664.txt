using namespace std;

int main() {

    ios::sync_with_stdio(0);
    cin.tie(0);

//    std::cout << std::fixed;
//    std::cout << std::setprecision(8);

    int t;
    cin >> t;

    for (int ti = 1; ti <= t; ti++) {
//        cout << "Case #" << ti << ": ";
        int n, m;
        cin >> n >> m;

        int robots[n];
        char dirs[n];
        int answers[n];
        for (int i = 0; i < n; i++) cin >> robots[i];
        for (int i = 0; i < n; i++) cin >> dirs[i];
        for (int i = 0; i < n; i++) answers[i] = -1;

        tuple<int, char, int> sortedRobots[n];
        for (int i = 0; i < n; i++) {
            get<0>(sortedRobots[i]) = robots[i];
            get<1>(sortedRobots[i]) = dirs[i];
            get<2>(sortedRobots[i]) = i;
        }
        sort(sortedRobots, sortedRobots + n);




        int leftGoingIndex = -1;
        deque<int> rightGoingIndices;
        for (int i = 0; i < n; i++) {
            auto robot = sortedRobots[i];
            if (get<0>(robot) % 2 != 0) continue;
            if (get<1>(robot) == 'L') {
                if(!rightGoingIndices.empty()) {
                    int closestRightGoingIndex = rightGoingIndices.front();
                    rightGoingIndices.pop_front();
                    auto rightGoingRobot = sortedRobots[closestRightGoingIndex];
                    int leftGoingLocation = get<0>(robot);
                    int rightGoingLocation = get<0>(rightGoingRobot);
                    int time = (leftGoingLocation - rightGoingLocation) / 2;
                    answers[get<2>(robot)] = time;
                    answers[get<2>(rightGoingRobot)] = time;
                    continue;
                }
                if (leftGoingIndex < 0) {
                    leftGoingIndex = i;
                    continue;
                }
                auto leftGoingRobot = sortedRobots[leftGoingIndex];
                int prevLocation = get<0>(leftGoingRobot);
                int currLocation = get<0>(robot);
                int time = (prevLocation + currLocation) / 2;
                answers[get<2>(robot)] = time;
                answers[get<2>(leftGoingRobot)] = time;
                leftGoingIndex = -1;
            } else {
                rightGoingIndices.push_front(i);
            }
        }
        while (rightGoingIndices.size() >= 2) {
            int closerIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            int fartherIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            auto closerRobot = sortedRobots[closerIndex];
            auto fartherRobot = sortedRobots[fartherIndex];
            int closerLocation = get<0>(closerRobot);
            int fartherLocation = get<0>(fartherRobot);
            int time = ((m - closerLocation) + (m - fartherLocation)) / 2;
            answers[get<2>(closerRobot)] = time;
            answers[get<2>(fartherRobot)] = time;
        }
        if (!rightGoingIndices.empty() && leftGoingIndex >= 0) {
            int rightGoingIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            auto leftGoingRobot = sortedRobots[leftGoingIndex];
            auto rightGoingRobot = sortedRobots[rightGoingIndex];
            int leftGoingLocation = get<0>(leftGoingRobot);
            int rightGoingLocation = get<0>(rightGoingRobot);
            int time = (leftGoingLocation + (m - rightGoingLocation) + m) / 2;
            answers[get<2>(leftGoingRobot)] = time;
            answers[get<2>(rightGoingRobot)] = time;
        }

        leftGoingIndex = -1;
        rightGoingIndices.clear();
        for (int i = 0; i < n; i++) {
            auto robot = sortedRobots[i];
            if (get<0>(robot) % 2 != 1) continue;
            if (get<1>(robot) == 'L') {
                if(!rightGoingIndices.empty()) {
                    int closestRightGoingIndex = rightGoingIndices.front();
                    rightGoingIndices.pop_front();
                    auto rightGoingRobot = sortedRobots[closestRightGoingIndex];
                    int leftGoingLocation = get<0>(robot);
                    int rightGoingLocation = get<0>(rightGoingRobot);
                    int time = (leftGoingLocation - rightGoingLocation) / 2;
                    answers[get<2>(robot)] = time;
                    answers[get<2>(rightGoingRobot)] = time;
                    continue;
                }
                if (leftGoingIndex < 0) {
                    leftGoingIndex = i;
                    continue;
                }
                auto leftGoingRobot = sortedRobots[leftGoingIndex];
                int prevLocation = get<0>(leftGoingRobot);
                int currLocation = get<0>(robot);
                int time = (prevLocation + currLocation) / 2;
                answers[get<2>(robot)] = time;
                answers[get<2>(leftGoingRobot)] = time;
                leftGoingIndex = -1;
            } else {
                rightGoingIndices.push_front(i);
            }
        }
        while (rightGoingIndices.size() >= 2) {
            int closerIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            int fartherIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            auto closerRobot = sortedRobots[closerIndex];
            auto fartherRobot = sortedRobots[fartherIndex];
            int closerLocation = get<0>(closerRobot);
            int fartherLocation = get<0>(fartherRobot);
            int time = ((m - closerLocation) + (m - fartherLocation)) / 2;
            answers[get<2>(closerRobot)] = time;
            answers[get<2>(fartherRobot)] = time;
        }
        if (!rightGoingIndices.empty() && leftGoingIndex >= 0) {
            int rightGoingIndex = rightGoingIndices.front();
            rightGoingIndices.pop_front();
            auto leftGoingRobot = sortedRobots[leftGoingIndex];
            auto rightGoingRobot = sortedRobots[rightGoingIndex];
            int leftGoingLocation = get<0>(leftGoingRobot);
            int rightGoingLocation = get<0>(rightGoingRobot);
            int time = (leftGoingLocation + (m - rightGoingLocation) + m) / 2;
            answers[get<2>(leftGoingRobot)] = time;
            answers[get<2>(rightGoingRobot)] = time;
        }

        for (int i = 0; i < n; i++) {
            cout << answers[i] << " ";
        }
        cout << endl;

    }
}
