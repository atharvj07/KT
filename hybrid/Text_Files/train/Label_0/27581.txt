#include <bits/stdc++.h>
using namespace std;
vector<int> adj[200005];
int sz[200005];
int n, k;
long long int ans;
void dfs(int u, int p) {
  for (int v : adj[u]) {
    if (v == p) continue;
    dfs(v, u);
    sz[u] += sz[v];
    ans += min(sz[v], 2 * k - sz[v]);
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 0; i < 2 * k; i++) {
    int x;
    cin >> x;
    sz[x] = 1;
  }
  int u, v;
  for (int i = 0; i < n - 1; i++) {
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1, 0);
  cout << ans << '\n';
  return 0;
}
