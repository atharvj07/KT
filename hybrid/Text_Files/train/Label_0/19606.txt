#define maxn 110
#define maxl 1000000000
#define mod 1000000007
using namespace std;

typedef unsigned long long ull;

int a[maxn];
double ans[maxn];

bool solve(){
	int n,i,j,r,k;
	double b,v,e,f,temp2,temp;
	scanf("%d",&n);
	if(n==0)return false;
	for(i=1;i<=n;++i){
		scanf("%d",&a[i]);
		ans[i]=1e30;
	}
	scanf("%lf%d%lf%lf%lf",&b,&r,&v,&e,&f);
	ans[0]=-b;
	for(i=0;i<n;++i){
		temp=ans[i]+b;
		for(j=i+1;j<=n;++j){
			for(k=a[j-1];k<a[j];++k){
				if(k-a[i]>=r)temp2=1/(v-e*(k-a[i]-r));
				else temp2=1/(v-f*(r-(k-a[i])));
				temp+=temp2;
			}
			ans[j]=min(temp,ans[j]);
		}
	}
	printf("%.4f\n",ans[n]);
	return true;
}

int main(){
	while(solve());
	return 0;
}
