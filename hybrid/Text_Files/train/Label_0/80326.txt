import sys
input = sys.stdin.readline

def ri():
	return int(input())

def rf():
	return list(map(float, input().split()))

def rl():
	return list(map(int, input().split()))

def rs():
	return input().rstrip()

def out_list(x):
	return ' '.join(map(str, x))

def solve_case():
	n = ri()
	A = rl()
	x = ri()
	res = 0
	j = 0
	for i in range(n):
		ok = True
		if i - j >= 1 and A[i - 1] + A[i] < 2 * x:
			ok = False
		if i - j >= 2 and A[i - 2] + A[i - 1] + A[i] < 3 * x:
			ok = False
		if not ok:
			res += 1
			j = i + 1
	print(n - res)
T = ri()
for ti in range(1, T + 1):
	solve_case()
