for i in range(int(input())):
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	p = list(map(int, input().split()))
	d = {}
	k = a[0]
	flag = True
	a1 = sorted(a)
	for i in range(n):
		d[a1[i]] = i
	for i in range(n - 1, -1, -1):
		for j in range(min(d[a[i]], i), max(i, d[a[i]])):
			if j + 1 not in p:
				flag = False
				break
		d[a[i]] -= 1
	if flag == True:
		print('YES')
	else:
		print('NO')
