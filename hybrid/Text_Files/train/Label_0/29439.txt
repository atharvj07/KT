#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long int a[n];
  long long int u = 1;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (u * 2 <= n) u = u * 2;
  }
  long long int ans = 0;
  for (int i = 0; i < n - 1; i++) {
    while (i + u >= n) u = u / 2;
    ans = ans + a[i];
    a[i + u] += a[i];
  }
  for (int i = 0; i < n - 1; i++) {
    cout << a[i] << endl;
    a[i + 1] += a[i];
  }
  return 0;
}
