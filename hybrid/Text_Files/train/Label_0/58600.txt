import java.io.*;
import java.util.*;

public class Main {
    
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        int m = scan.nextInt();
        int[] count = new int[5001];
        for (int i = 0; i < m; i++) {
            count[scan.nextInt()]++;
        }
        
        int[] arr = new int[m];
        int head = 0, tail = m - 1;
        int res = 0;
        for (int i = 1; i < count.length; i++) {
            if (count[i] == 1) {
                arr[head++] = i;
                res += 1;
            } else if (count[i] >= 2) {
                arr[head++] = i;
                arr[tail--] = i;
                res += 2;
            }
        }
        StringBuilder builder = new StringBuilder();
        for (int i = 0; i < head; i++) {
            if (builder.length() > 0) {
                builder.append(' ');
            }
            builder.append(Integer.toString(arr[i]));
        }
        
        if (head > 0 && tail < m - 1 && arr[head - 1] == arr[tail + 1]) {
            tail++;
            res -= 1;
        }
        
        for (int i = tail + 1; i < m; i++) {
            if (builder.length() > 0) {
                builder.append(' ');
            }
            builder.append(Integer.toString(arr[i]));
        }
        
        System.out.println(res);
        System.out.println(builder.toString());
    }
}
