N = int(input(''))
Line2 = input('')
Array = list(map(int, Line2.split(' ')))
min_elem_index = Array.index(min(Array))
max_elem_index = Array.index(max(Array))
min_index = 0
max_index = N - 1
L_inv1 = abs(min_elem_index - min_index)
R_inv1 = abs(max_index - min_elem_index)
ans1 = 0
if L_inv1 >= R_inv1:
	ans1 = L_inv1
elif L_inv1 < R_inv1:
	ans1 = R_inv1
ans2 = 0
L_inv2 = abs(max_elem_index - min_index)
R_inv2 = abs(max_index - max_elem_index)
if L_inv2 >= R_inv2:
	ans2 = L_inv2
elif L_inv2 < R_inv2:
	ans2 = R_inv2
if ans1 >= ans2:
	print(ans1)
elif ans1 < ans2:
	print(ans2)
