#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<ll,ll> mp;
ll inf =  1e9;


int main(){
    int n;
    cin>>n;
    vector<ll> a(100, 0);
    for(int i=1;i<=n;i++){
	int b = i%10;
	int t = i;
	while( t >= 10 ){
	    t/=10;
	}
	a[t*10+b] ++;
    }
    ll res = 0;
    for(int i=0;i<100;i++){
	int k = i%10*10 + i/10; 
	res += a[i]*a[k];
    }
    cout<<res<<endl;
    return 0;
}
