#include <bits/stdc++.h>
using namespace std;
vector<vector<int>> g;
vector<char> t;
vector<bool> b, d;
bool calc(int v) {
  if (t[v] == 'A') return b[v] = calc(g[v][0]) & calc(g[v][1]);
  if (t[v] == 'O') return b[v] = calc(g[v][0]) | calc(g[v][1]);
  if (t[v] == 'X') return b[v] = calc(g[v][0]) ^ calc(g[v][1]);
  if (t[v] == 'N') return b[v] = !calc(g[v][0]);
  return b[v];
}
void dfs(int v) {
  if (t[v] == 'A') {
    d[g[v][0]] = d[g[v][1]] = d[v];
    if (!b[g[v][1]]) d[g[v][0]] = false;
    if (!b[g[v][0]]) d[g[v][1]] = false;
  } else if (t[v] == 'O') {
    d[g[v][0]] = d[g[v][1]] = d[v];
    if (b[g[v][1]]) d[g[v][0]] = false;
    if (b[g[v][0]]) d[g[v][1]] = false;
  } else if (t[v] == 'X')
    d[g[v][0]] = d[g[v][1]] = d[v];
  else if (t[v] == 'N')
    d[g[v][0]] = d[v];
  for (int i = 0; i < g[v].size(); ++i) dfs(g[v][i]);
}
int main() {
  int n, a;
  scanf("%d", &n);
  g.resize(n);
  t.resize(n);
  b.resize(n);
  d.resize(n);
  for (int i = 0; i < n; ++i) {
    char s[4];
    scanf("%s", s);
    t[i] = s[0];
    if (t[i] == 'I') {
      scanf("%d", &a);
      b[i] = a;
    } else if (t[i] == 'N') {
      scanf("%d", &a);
      g[i].push_back(a - 1);
    } else {
      scanf("%d", &a);
      g[i].push_back(a - 1);
      scanf("%d", &a);
      g[i].push_back(a - 1);
    }
  }
  calc(0);
  d[0] = true;
  dfs(0);
  for (int i = 0; i < n; ++i)
    if (t[i] == 'I') printf("%d", b[0] ^ d[i]);
  return 0;
}
