def st(x, y, ax, ay, bx, by):
	return (bx - ax) * (y - ay) - (by - ay) * (x - ax)
(n, d) = map(int, input().split())
x1 = 0
y1 = d
x2 = d
y2 = 0
x3 = n
y3 = n - d
x4 = n - d
y4 = n
m = int(input())
for i in range(m):
	(x, y) = map(int, input().split())
	p1 = st(x, y, x1, y1, x2, y2)
	p2 = st(x, y, x2, y2, x3, y3)
	p3 = st(x, y, x3, y3, x4, y4)
	p4 = st(x, y, x4, y4, x1, y1)
	if p1 <= 0 and p2 <= 0 and (p3 <= 0) and (p4 <= 0) or (p1 >= 0 and p2 >= 0 and (p3 >= 0) and (p4 >= 0)):
		print('YES')
	else:
		print('NO')
