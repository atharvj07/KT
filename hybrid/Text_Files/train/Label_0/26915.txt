#include <bits/stdc++.h>
using namespace std;
double ans[200000 + 10];
map<double, int> mp;
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    double a = 0, b = 0, c = 0;
    string s;
    cin >> s;
    int cnt = 1;
    for (cnt = 1; cnt < 10; cnt++) {
      if (s[cnt] == '+') break;
      a *= 10;
      a += s[cnt] - '0';
    }
    cnt++;
    for (; cnt < 10; cnt++) {
      if (s[cnt] == ')') break;
      b *= 10;
      b += s[cnt] - '0';
    }
    cnt += 2;
    for (; cnt < s.length(); cnt++) {
      c *= 10;
      c += s[cnt] - '0';
    }
    ans[i] = 1.0 * (a + b) / c;
    mp[ans[i]]++;
  }
  for (int i = 0; i < n; i++) {
    printf("%d ", mp[ans[i]]);
  }
  printf("\n");
  return 0;
}
