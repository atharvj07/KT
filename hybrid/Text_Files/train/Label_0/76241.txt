for _ in range(int(input())):
	(a, s) = map(int, input().split())
	b = ''
	while s and a:
		if a > s:
			b = -1
			break
		(s1, s2, a1) = (s % 10, s % 100, a % 10)
		if s1 >= a1:
			b = str(s1 - a1) + b
			s //= 10
		elif s2 > a1 and s2 - a1 < 10:
			b = str(s2 - a1) + b
			s //= 100
		else:
			b = -1
			break
		a //= 10
	if b != -1:
		print(int(str(s) + b))
	else:
		print(b)
