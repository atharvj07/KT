#include <bits/stdc++.h>
using namespace std;
int sum = 0, count = 0;
string s;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int ar[4];
  for (int i = 0; i < 4; i++) {
    cin >> ar[i];
  }
  sort(ar, ar + 4);
  for (int i = 0; i < 4; i++) {
    if (ar[i] == ar[i - 1]) continue;
    for (int j = 0; j < 4; j++) {
      if (i == j) continue;
      if (ar[i] == ar[j]) {
        sum++;
      }
    }
  }
  printf("%d\n", sum);
  ;
  return 0;
}
