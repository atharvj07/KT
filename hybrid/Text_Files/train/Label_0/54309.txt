import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Rockethon14{
    
    public static class contestant implements Comparable<contestant>{
        int pts;
        int eff;
        boolean Manao;
        boolean DidManaoWinThis;
        
        public contestant(int pts, int eff, boolean Manao, boolean DidManaoWinThis) {
            this.pts=pts;
            this.eff=eff;
            this.Manao=Manao;
            this.DidManaoWinThis=DidManaoWinThis;
        }

        @Override
        public int compareTo(contestant o) {
            if(o.pts==pts){
                if(o.Manao){
                    if(DidManaoWinThis)
                        return 1;
                    return -1;
                }
                    
                if(Manao){
                    if(o.DidManaoWinThis)
                        return -1;
                    return 1;
                }
                return 0;
            }
            return o.pts-pts;
        }
        
    }
    
    
    static int n, k;
    static int[][] in;
    public static int simulate(int index, contestant[] c){
        
        if(index==n){

            Arrays.sort(c);
            
            for(int i=0;i<k;i++){
                if(c[i].Manao)
                    return c[i].eff;
            }
            return 1<<25;
        }
        
        
        contestant[] cWin=new contestant[n+1];
        for(int i=0;i<n+1;i++){
            cWin[i]=new contestant(c[i].pts, c[i].eff, c[i].Manao, c[i].DidManaoWinThis);
        }
        cWin[index].DidManaoWinThis=true;
        cWin[n].pts++;
        cWin[n].eff+=c[index].eff;
        
        
        contestant[] cLose=new contestant[n+1];
        for(int i=0;i<n+1;i++){
            cLose[i]=new contestant(c[i].pts, c[i].eff, c[i].Manao, c[i].DidManaoWinThis);
        }
        cLose[index].pts++;
        
        int win=simulate(index+1, cWin);
        int lose=simulate(index+1, cLose);
        
        return Math.min(win, lose);
    }
    
    
    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        
        String[] s=br.readLine().split(" ");
        n=Integer.parseInt(s[0]);
        k=Integer.parseInt(s[1]);
        
        contestant[] c=new contestant[n+1];
        for(int i=0;i<n;i++){
             s=br.readLine().split(" ");
            c[i]=new contestant(Integer.parseInt(s[0]), Integer.parseInt(s[1]), false, false);
        }
        c[n]=new contestant(0, 0, true, false);
        
        int ans=simulate(0, c);
        System.out.println(ans==1<<25?-1:ans);
    }
}
