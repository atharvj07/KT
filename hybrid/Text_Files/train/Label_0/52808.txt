import java.io.*;
import java.util.*;

public class F implements Runnable{
	public static void main (String[] args) {new Thread(null, new F(), "_cf", 1 << 28).start();}

	public void run() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);

		int oo = (int)2e9;
		int n = fs.nextInt();
		int k = fs.nextInt();
		int[] dp = new int[n + 1], tdp = new int[n + 1];
		Arrays.fill(dp, oo);
		dp[0] = 0;
		while(k-->0) {
			int l = fs.nextInt(), r = fs.nextInt(), len = r - l;
			for(int i = 0; i <= n; i++) tdp[i] = dp[i];
			ArrayDeque<Integer> minQueue = new ArrayDeque<>();
			for(int i = 0; i <= n; i++) {
				push(minQueue, i, dp);
				tdp[i] = Math.min(tdp[i], best(minQueue, dp) + 2);
				if(r - i >= 0 && r - i <= n) tdp[r - i] = Math.min(tdp[r - i], best(minQueue, dp) + 1);
				pop(minQueue, i - len);
			}
			for(int i = 0; i <= n; i++) dp[i] = tdp[i];
		}
		
		if(dp[n] >= oo) out.println("Hungry");
		else out.printf("Full\n%d\n", dp[n]);

		out.close();
	}
	
	void pop(ArrayDeque<Integer> minQueue, int idx) {
		if(!minQueue.isEmpty() && minQueue.peekFirst() == idx) minQueue.pollFirst();
	}
	
	int best(ArrayDeque<Integer> minQueue, int[] dp) {
		if(minQueue.isEmpty()) return (int)2e9;
		return dp[minQueue.peekFirst()];
	}
	
	void push(ArrayDeque<Integer> minQueue, int p, int[] dp) {
		while(!minQueue.isEmpty() && dp[minQueue.getLast()] >= dp[p]) minQueue.pollLast();
		minQueue.addLast(p);
	}
	
	/*
	 * dp[i][j] means the minimum number of moves to cook the cutlet on one side
	   for j seconds after processing i intervals (do iterative space saving)
	 * The length of the interval is known and so we'll build up time j with the
	   previous length values (including j) all from the previous state. It'll take
	   an extra 2 flips to so:
	   if we know the number of flips to get the cutlet to exactly t seconds and we
	   want to get it to j seconds in the current interval being processed, then we'll
	   need to flip it after it gets to t seconds in the other interval (to prevent overcooking)
	   and then flip it again when it needs to cook in our interval to get it to the desired j.
	 * There's also a case where we can just flip it once based off of the previous states.
	   The main idea for some time j, is we want the cutlet to get to its jth second of cooking
	   during our current interval (so we can do a flip if necessary). So what happens is we take
	   the min of the dp values that are allowed in our current range and use that value to calculate
	   (right - j)'s best value by taking that minimum and I guess not flipping it and instead letting
	   it cook. Still a bit hazy but it makes sense when doing the second sample by hand (especially
	   when we update times 10 and 8).
	 * Definitely had to study someone else's solution because the editorial is very obscure and
	   doesn't actually explain the important part of the problem (how to derive the +1 and +2 jumps).
	 */

	class FastScanner {
		BufferedReader br;
		StringTokenizer st;
		public FastScanner() {
			try	{
				br = new BufferedReader(new InputStreamReader(System.in));
				//                br = new BufferedReader(new FileReader("testdata.out"));
				st = new StringTokenizer("");
			} catch (Exception e){e.printStackTrace();}
		}
		public String next() {
			if (st.hasMoreTokens())	return st.nextToken();
			try {st = new StringTokenizer(br.readLine());}
			catch (Exception e) {e.printStackTrace();}
			return st.nextToken();
		}
		public int nextInt() {return Integer.parseInt(next());}
		public long nextLong() {return Long.parseLong(next());}
		public double nextDouble() {return Double.parseDouble(next());}

		public String nextLine() {
			String line = "";
			try {line = br.readLine();}
			catch (Exception e) {e.printStackTrace();}
			return line;
		}
		public Integer[] nextIntegerArray(int n) {
			Integer[] a = new Integer[n];
			for(int i = 0; i < n; i++) a[i] = nextInt();
			return a;
		}
		public int[] nextIntArray(int n) {
			int[] a = new int[n];
			for(int i = 0; i < n; i++) a[i] = nextInt();
			return a;
		}
		public char[] nextCharArray() {return nextLine().toCharArray();}
	}

}
