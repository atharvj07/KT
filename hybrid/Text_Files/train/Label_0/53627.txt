import java.io.*;
import java.util.*;
public class d {
public static void main(String[] args) throws IOException {
	input.init(System.in);
	PrintWriter out = new PrintWriter(System.out);
	ArrayList<Point> res = new ArrayList<Point>();
	int n = input.nextInt();
	Point cur = new Point(0, 0);
	res.add(cur);
	
	ArrayList<Integer> ps = new ArrayList<Integer>();
	ps.add(0);
	for(int i = 0; i<n; i++)
	{
		cur = new Point(cur.x + 20 * Math.sin(54 * Math.PI / 180), cur.y);
		res.add(cur);
		ps.add(i+1);
	}
	Point[] penta = new Point[5];
	penta[0] = new Point(0, 0);
	penta[1] = new Point(penta[0].x + Math.cos(36 * Math.PI / 180) * 10, penta[0].y + 10 * Math.sin(36 * Math.PI/ 180));
	penta[2] = new Point(penta[1].x + Math.cos(-36 * Math.PI / 180) * 10, penta[1].y + 10 * Math.sin(-36 * Math.PI/ 180));
	penta[3] = new Point(penta[2].x + Math.cos(-108 * Math.PI / 180) * 10, penta[2].y + 10 * Math.sin(-108 * Math.PI/ 180));
	penta[4] = new Point(penta[3].x + Math.cos(-180 * Math.PI / 180) * 10, penta[3].y + 10 * Math.sin(-180 * Math.PI/ 180));
	for(int i = 0; i<n; i++)
	{
		cur = cur.plus(penta[4].minus(penta[2]));
		ps.add(res.size());
		res.add(cur);
		cur = cur.plus(penta[1].minus(penta[4]));
		ps.add(res.size());
		res.add(cur);
		cur = cur.plus(penta[3].minus(penta[1]));
		ps.add(res.size());
		res.add(cur);
		cur = cur.plus(penta[0].minus(penta[3]));
		ps.add(n - 1 - i);
	}
	out.println(res.size());
	for(Point p : res) out.printf("%.12f %.12f\n", p.x,p.y);
	for(int i = 0; i<n; i++)
	{
		out.println((i+1)+" "+(res.size() - 1 - 3 * i)+" "+(i+2)+" "+(res.size() - 0 - 3 * i)+" "+(res.size() - 2 - 3 * i));
	}
	for(int x : ps) out.print(x+1+" ");
	out.println();
	out.close();
}
static class Point
{
	double x, y;
	public Point(double xx, double yy)
	{
		x = xx; y = yy;
	}
	Point plus(Point p)
	{
		return new Point(x + p.x, y + p.y);
	}
	Point minus(Point p)
	{
		return new Point(x - p.x, y - p.y);
	}
}
public static class input {
	static BufferedReader reader;
	static StringTokenizer tokenizer;

	static void init(InputStream input) {
		reader = new BufferedReader(new InputStreamReader(input));
		tokenizer = new StringTokenizer("");
	}

	static String next() throws IOException {
		while (!tokenizer.hasMoreTokens())
			tokenizer = new StringTokenizer(reader.readLine());
		return tokenizer.nextToken();
	}

	static int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	static double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	static long nextLong() throws IOException {
		return Long.parseLong(next());
	}
}
}
