import java.util.*;
import java.io.*;

public class A {
    static int i(String s) { return Integer.parseInt(s); }
    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = i(in.readLine());
        String[] arr = in.readLine().split(" ");
        long[] A = new long[n];
        for(int i=0; i<n; i++)
            A[i] = i(arr[i]);

        Map<Long, Integer> LAST = new HashMap<Long, Integer>();
        for(int i=n-1; i>=0; i--) {
            if(LAST.containsKey(A[i])) continue;
            else LAST.put(A[i], i);
        }

        long[] S = new long[n+1];
        S[0] = 0;
        for(int i=1; i<=n; i++)
            S[i] = S[i-1]+Math.max(0, A[i-1]);

        int best_start = 0;
        long best_score = -1000L*1000*1000*1000*1000;
        for(int i=0; i<n; i++) {
            Integer hi = LAST.get(A[i]);
            if(hi!=null && hi>i) {
                long score = S[hi+1]-S[i];
                if(A[i] < 0) score += 2*A[i];
                if(score > best_score) {
                    best_score = score;
                    best_start = i;
                }
            }
        }

        List<Integer> CUT = new ArrayList<Integer>();
        int hi = LAST.get(A[best_start]);
        for(int i=0; i<best_start; i++)
            CUT.add(i);
        for(int i=hi+1; i<n; i++)
            CUT.add(i);
        for(int i=best_start+1; i<=hi-1; i++)
            if(A[i] < 0)
                CUT.add(i);

        Collections.sort(CUT);
        StringBuilder out = new StringBuilder();
        out.append(String.format("%d %d\n", best_score, CUT.size()));
        for(int i=0; i<CUT.size(); i++) {
            out.append(CUT.get(i)+1);
            out.append(i==CUT.size()-1?'\n':' ');
        }
        System.out.print(out);
    }
}
