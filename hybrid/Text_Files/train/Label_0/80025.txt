from functools import reduce
import sys
input = sys.stdin.readline
t = int(input())
while t:
	t -= 1
	n = int(input())
	a = sorted(list(map(int, input().split())))
	if n == 5:
		ans = reduce(lambda x, y: x * y, a)
		print(ans)
	else:
		nl = []
		pl = []
		for i in range(4):
			if a[i] < 0:
				nl.append(a[i])
			else:
				break
		for i in range(n - 1, n - 6, -1):
			if a[i] > 0:
				pl.append(a[i])
			else:
				break
		if len(pl) == 0:
			ans = reduce(lambda x, y: x * y, a[n - 5:n])
		elif len(nl) < 2:
			ans = reduce(lambda x, y: x * y, a[n - 5:n])
		elif len(pl) >= 3 and len(nl) >= 2 or (len(pl) >= 1 and len(nl) >= 4) or len(pl) == 5:
			ans1 = ans2 = ans3 = 1
			if len(pl) == 5:
				ans1 = reduce(lambda x, y: x * y, pl)
			if len(pl) >= 3:
				pp = 1
				for i in range(3):
					pp = pp * pl[i]
				np = nl[0] * nl[1]
				ans2 = pp * np
			if len(nl) == 4:
				ans3 = reduce(lambda x, y: x * y, nl)
				ans3 = ans3 * pl[0]
			ans = max(ans1, ans2, ans3)
		else:
			ans = 0
		print(ans)
