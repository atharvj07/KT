
import java.io.*;
import java.util.*;
import static java.lang.Integer.parseInt;
import static java.lang.StrictMath.abs;

public class HanSoloandLazerGun {
    public static void main(String[] args) throws IOException {
        BufferedReader read = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter prnt = new PrintWriter(System.out);
        HashSet<Double> shots = new HashSet<>();
        StringTokenizer st = new StringTokenizer(read.readLine());
        int n = parseInt(st.nextToken());
        double X0 = parseInt(st.nextToken());
        double Y0 = parseInt(st.nextToken());

        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(read.readLine());
            int Xs = parseInt(st.nextToken());
            int Ys = parseInt(st.nextToken());

            double dx = Xs - X0;
            double dy = Ys - Y0;

            if (dx == 0) {
                shots.add(Double.MAX_VALUE);
            } else {
                if (abs(dy / dx) < 0.0000000000000001) {
                    shots.add(0.0);
                } else {
                    shots.add((double) (dy / dx));
                }
            }
        }
        prnt.println(shots.size());
        prnt.close();
    }
}
