#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n, k, d;
    cin >> n >> k >> d;
    map<int, int> hs;
    int a[n];
    int m = 0, i = 0, curr;
    for (i = 0; i < n; i++) {
      cin >> a[i];
      if (i < d) {
        map<int, int>::iterator itr = hs.find(a[i]);
        if (itr == hs.end()) {
          m++;
          hs.insert(pair<int, int>(a[i], 1));
        } else {
          itr->second++;
        }
      } else {
        if (i == d) curr = m;
        map<int, int>::iterator itr1 = hs.find(a[i - d]);
        itr1->second--;
        if (itr1->second == 0) {
          curr--;
          hs.erase(a[i - d]);
        }
        map<int, int>::iterator itr = hs.find(a[i]);
        if (itr == hs.end()) {
          curr++;
          hs.insert(pair<int, int>(a[i], 1));
        } else {
          itr->second++;
        }
        m = min(m, curr);
      }
    }
    cout << m << endl;
  }
}
