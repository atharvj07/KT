n = int(input())
balls = list(map(int, input().split()))
balls.sort()

def answer(balls):
	a = 0
	b = 1
	while b < len(balls) - 1:
		while balls[b] == balls[a] and b < len(balls) - 2:
			b += 1
		if balls[b] == balls[a] + 1:
			c = b + 1
			while balls[c] == balls[b] and c < len(balls) - 1:
				c += 1
			if balls[c] == balls[b] + 1:
				return 'YES'
			else:
				a = c
				b = a + 1
		else:
			a = b
			b = a + 1
	return 'NO'
print(answer(balls))
