#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
map<ll,ll> t;

ll calc(ll x,ll y,ll z){
  ll xfb=x/15,xf=x/3-xfb,xb=x/5-xfb;
  ll yfb=y/15,yf=y/3-yfb,yb=y/5-yfb;
  ll X=(x-xfb-xf-xb)*z + xfb*8 + xf*4 + xb*4;
  ll Y=(y-yfb-yf-yb)*z + yfb*8 + yf*4 + yb*4;
  return Y-X;
}

ll po(ll a,ll b){
  ll res=1;
  for(ll i=0;i<b;i++)res*=a;
  return res;
}

string fizzbuzz(ll i){
  if(i%15==0)return "FizzBuzz";
  else if(i%5==0)return "Buzz";
  else if(i%3==0)return "Fizz";
  else{
    stringstream ss;
    string res;
    ss<<i;
    ss>>res;
    return res;
  }
}

ll func(ll x){
  if(x==0)return 0;
  ll i;
  for(i=0;i<18;i++)
    if(po(10,i)<=x&&x<po(10,i+1))break;
  ll res=t[po(10,i)];
  return res + calc( po(10,i) , x, i+1);
}

int main(){
  t[1]=1;
  for(int i=1;i<18;i++){
    t[ po(10,i) ] = t[ po(10,i-1) ]
      + calc( po(10,i-1) , po(10,i)-1 ,i) + 4;
  }



       
  ll n,L=1,R=po(10,17),M;
  cin>>n;
  while( L+1< R ){
    M= (L+R)/2;
    ll res=func(M);
    if( res < n ){
      L=M+1;
    }else{
      R=M;
    }
  }

 
  
  string ans="";
  for(ll i=L;i<=L+20;i++)ans=ans+fizzbuzz(i);
  for(ll i=0;i<20;i++){
    cout<<ans[n-func(L-1)-1+i];
  }
  cout<<endl;
  
  return 0;
}
