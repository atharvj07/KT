#include <bits/stdc++.h>
using namespace std;
int a[200005], p[200005], w[200005] = {0};
int fp(int x) {
  if (p[x] == x)
    return x;
  else
    return p[x] = fp(p[x]);
}
void me(int u, int v) {
  v = fp(v);
  u = fp(u);
  p[u] = v;
}
int main() {
  int n, q, qi, x, k, i, j;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> a[i];
    p[i] = i;
  }
  p[n + 1] = n + 1;
  cin >> q;
  for (i = 1; i <= q; i++) {
    cin >> qi;
    if (qi == 2) {
      cin >> k;
      printf("%d\n", w[k]);
    } else {
      cin >> k >> x;
      while (k <= n) {
        k = p[k];
        w[k] += x;
        if (a[k] < w[k]) {
          x = w[k] - a[k];
          w[k] = a[k];
          if (k != n) me(k, k + 1);
          k++;
        } else
          break;
      }
    }
  }
}
