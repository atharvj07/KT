import sys
from operator import itemgetter

def input():
	return sys.stdin.readline().strip()

def list2d(a, b, c):
	return [[c] * b for i in range(a)]

def list3d(a, b, c, d):
	return [[[d] * c for j in range(b)] for i in range(a)]

def list4d(a, b, c, d, e):
	return [[[[e] * d for j in range(c)] for j in range(b)] for i in range(a)]

def ceil(x, y=1):
	return int(-(-x // y))

def INT():
	return int(input())

def MAP():
	return map(int, input().split())

def LIST(N=None):
	return list(MAP()) if N is None else [INT() for i in range(N)]

INF = 10 ** 18
MOD = 10 ** 9 + 7

def bfs(N, nodes, src):
	from collections import deque
	que = deque([(src, 0)])
	dist = [INF] * N
	while que:
		(u, c) = que.popleft()
		if dist[u] != INF:
			continue
		dist[u] = c
		for v in nodes[u]:
			que.append((v, c + 1))
	return dist
(N, M, K) = MAP()
S = set((a - 1 for a in LIST()))
nodes = [[] for i in range(N)]
for _ in range(M):
	(a, b) = MAP()
	a -= 1
	b -= 1
	nodes[a].append(b)
	nodes[b].append(a)
dist1 = bfs(N, nodes, 0)
dist2 = bfs(N, nodes, N - 1)
mn = dist1[N - 1]
AB = []
for i in range(N):
	(a, b) = (dist1[i], dist2[i])
	if i in S:
		AB.append((a - b, a, b))
AB.sort(key=itemgetter(0))
(_, A, B) = zip(*AB)
ans = 0
mx = A[0]
for i in range(1, K):
	(a, b) = (A[i], B[i])
	ans = max(ans, mx + b + 1)
	mx = max(mx, a)
ans = min(ans, mn)
print(ans)
