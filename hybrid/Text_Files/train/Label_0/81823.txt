n = int(input())
a = [int(i) for i in input().split()]
count = 0
k = 0
if sum(a) != 1:
	if len(a) > 1 and sum(a) != 0:
		for i in range(len(a)):
			if a[i] != 0:
				a = a[i:]
				break
		while k < len(a) - 1:
			if a[k] == 1:
				count += 1
				k += 1
			elif a[k] == 0 and a[k + 1] == 0:
				k += 2
			elif a[k] == 0 and a[k + 1] == 1:
				if a[k - 1] == 0:
					k += 1
				else:
					count += 1
					k += 1
		if a[len(a) - 1] > 0:
			count += 1
	else:
		count = a[0]
else:
	count += 1
print(count)
