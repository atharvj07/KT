#include <bits/stdc++.h>
using namespace std;
const int N = 1010;
const int M = 5;
const long long INF = 1LL << 60;
const double EPS = 1e-8;
const long long MOD = 1000000007LL;
map<char, pair<long long, long long> > dir;
int n, m, t;
string s;
vector<long long> col, row;
void init() {
  cin >> s;
  col = vector<long long>(n, INF);
  row = vector<long long>(m, INF);
  dir['U'] = make_pair(-1, 0);
  dir['D'] = make_pair(1, 0);
  dir['L'] = make_pair(0, -1);
  dir['R'] = make_pair(0, 1);
}
vector<long long> Left, Right, UP, Down;
long long dfs1(int u, int delta) {
  if (col[u] != INF) {
    return col[u];
  }
  if (u - UP[t] >= 0 && u + Down[t] < n) {
    if (!delta) {
      return INF;
    }
    return col[u] = dfs1(u + delta, delta) + t;
  }
  int lt = 1, rt = t;
  while (lt < rt) {
    int md = (lt + rt) >> 1;
    if (u - UP[md] >= 0 && u + Down[md] < n) {
      lt = md + 1;
    } else {
      rt = md;
    }
  }
  return col[u] = lt;
}
long long dfs2(int u, int delta) {
  if (row[u] != INF) {
    return row[u];
  }
  if (u - Left[t] >= 0 && u + Right[t] < m) {
    if (!delta) {
      return INF;
    }
    return row[u] = dfs2(u + delta, delta) + t;
  }
  int lt = 1, rt = t;
  while (lt < rt) {
    int md = (lt + rt) >> 1;
    if (u - Left[md] >= 0 && u + Right[md] < m) {
      lt = md + 1;
    } else {
      rt = md;
    }
  }
  return row[u] = lt;
}
void solve() {
  Left = vector<long long>(t + 1);
  Right = vector<long long>(t + 1);
  UP = vector<long long>(t + 1);
  Down = vector<long long>(t + 1);
  long long x = 0, y = 0;
  for (int i = 0; i < t; i++) {
    char c = s[i];
    x += dir[c].first;
    y += dir[c].second;
    if (x < 0) {
      UP[i + 1] = max(UP[i], -x);
      Down[i + 1] = Down[i];
    } else {
      UP[i + 1] = UP[i];
      Down[i + 1] = max(Down[i], x);
    }
    if (y < 0) {
      Left[i + 1] = max(Left[i], -y);
      Right[i + 1] = Right[i];
    } else {
      Left[i + 1] = Left[i];
      Right[i + 1] = max(Right[i], y);
    }
  }
  for (int i = 0; i < n; i++) {
    if (col[i] != INF) {
      continue;
    }
    dfs1(i, x);
  }
  for (int i = 0; i < m; i++) {
    if (row[i] != INF) {
      continue;
    }
    dfs2(i, y);
  }
  sort(col.begin(), col.end());
  sort(row.begin(), row.end());
  col.push_back(INF);
  row.push_back(INF);
  long long ans = 0;
  x = 0, y = 0;
  int cut_col = 0, cut_row = 0;
  while (x < n && y < m) {
    if (col[x] == INF && row[y] == INF) {
      printf("-1\n");
      return;
    }
    long long step, cnt;
    if (col[x] < row[y]) {
      step = col[x++] % MOD;
      cnt = m - cut_row;
      cut_col++;
    } else {
      step = row[y++] % MOD;
      cnt = n - cut_col;
      cut_row++;
    }
    ans += (cnt * step) % MOD;
    ans %= MOD;
  }
  cout << ans << endl;
}
int main() {
  while (cin >> t >> n >> m) {
    init();
    solve();
  }
  return 0;
}
