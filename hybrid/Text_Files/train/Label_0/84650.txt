ans = -1
edge = list()
(n, m) = map(int, input().split())
s = [set() for i in range(n + 1)]
for i in range(m):
	(u, v) = map(int, input().split())
	edge += [(u, v)]
	s[u] |= {v}
	s[v] |= {u}
k = [len(s[i]) for i in range(n + 1)]
for (u, v) in edge:
	w = s[u] & s[v]
	for i in w:
		res = k[u] + k[v] + k[i] - 6
		if ans == -1 or ans > res >= 0:
			ans = res
print(ans)
