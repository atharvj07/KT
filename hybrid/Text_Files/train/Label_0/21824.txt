#include<iostream>
using namespace std;

typedef struct{
	int front;
	int back;
}chair_t;

int main(){
	
	int n , m;
	chair_t pos[1010];
	
	while( cin >> n >> m ){
		if( !n && !m ) break ;
		
		
		for( int i=0 ; i<n ; i++ ){
			pos[i].front = i+1 ;
			pos[i].back  = i-1 ;
		}
		pos[0].back    = n-1 ;
		pos[n-1].front = 0   ;
		
		int npos = 0;
		while( npos != pos[npos].front ){
			for( int i=1 ; i<m ; i++ ){
				npos = pos[npos].front ;
			}
			pos[pos[npos].back].front = pos[npos].front ;
			pos[pos[npos].front].back = pos[npos].back;
			npos = pos[npos].front ;
		}
		
		cout << npos+1 << endl ;
	}
}
