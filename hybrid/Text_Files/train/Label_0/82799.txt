from collections import deque
a = input()
b = input()
a1 = len(a)
b1 = len(b)
k = 0
d = deque([])
for i in range(b1):
	if a[i] != b[i]:
		k += 1
l = 0
if k % 2 == 0:
	l += 1
if a1 == b1:
	print(l)
else:
	for i in range(b1):
		if a[i] == a[i + 1]:
			d.append(0)
		else:
			d.append(1)
	m = sum(d)
	k += m
	if k % 2 == 0:
		l += 1
	for i in range(1, a1 - b1):
		m -= d[0]
		d.popleft()
		if a[i + b1] == a[i + b1 - 1]:
			d.append(0)
		else:
			d.append(1)
		m += d[-1]
		k += m
		if k % 2 == 0:
			l += 1
	print(l)
