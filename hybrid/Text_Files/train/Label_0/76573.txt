import math
from decimal import *
getcontext().prec = 50

class DSU:

	def __init__(self, n):
		self.hash = {}
		for i in range(1, n + 1):
			self.hash[i] = [i]

	def union(self, i, j):
		lst1 = self.hash[i]
		lst2 = self.hash[j]
		lst1.extend(lst2)
		for element in lst2:
			self.hash[element] = lst1

	def same_set(self, i, j):
		return j in self.hash[i]

def getKthPerm(lst, k, offset):
	if len(lst) - 1 == offset:
		return True
	elif len(lst) <= offset:
		return False
	else:
		n = len(lst) - offset
		index = math.ceil(k / math.factorial(n - 1)) - 1 + offset
		if index >= len(lst):
			return False
		if index > offset:
			temp = lst[index]
			lst.pop(index)
			lst.insert(offset, temp)
		while k - math.factorial(n - 1) > 0:
			k -= math.factorial(n - 1)
		return getKthPerm(lst, k, offset + 1)

def getKthCycle(n, k, fact):
	lst = [i - 1 for i in range(n + 1)]
	num_open = n
	dsu = DSU(n)
	dsu.union(1, n)
	lst[1] = n
	num_open -= 1
	pos = 2
	while pos < n:
		i = pos
		while i <= n:
			if dsu.same_set(pos, lst[i]):
				pass
			elif k <= fact[num_open - 2]:
				break
			else:
				k -= fact[num_open - 2]
			i += 1
		if i == n + 1:
			return None
		else:
			num_open -= 1
			dsu.union(pos, lst[i])
			temp = lst[i]
			j = i
			while j > pos:
				lst[j] = lst[j - 1]
				j -= 1
			lst[pos] = temp
		pos += 1
	return lst[1:]

def cycles(n, fact):
	if n <= 1:
		return 1
	elif n - 2 in fact:
		return fact[n - 2]
	else:
		return math.factorial(n - 2)

def dp(i, n, table, fact):
	if i > n:
		return 0
	elif i == n:
		return 1
	elif i in table:
		return table[i]
	else:
		val = 0
		for j in range(i, n + 1):
			val += dp(j + 1, n, table, fact) * cycles(j - i + 1, fact)
		table[i] = val
		return val

def Max(i, table, fact):
	if i in table:
		return table[i]
	elif i <= 1:
		table[i] = 1
		return 1
	else:
		val = 0
		for j in range(1, i + 1):
			val += Max(i - j, table, fact) * cycles(j, fact)
		table[i] = val
		return val

def getKthGoodPermutation(n, k, Max, fact):
	if Max[n] < k:
		return None
	if n == 1:
		if k == 1:
			return [1]
		else:
			return None
	block_len = 1
	while k - cycles(block_len, fact) * Max[n - block_len] > 0:
		k -= cycles(block_len, fact) * Max[n - block_len]
		block_len += 1
	k_block = math.ceil(Decimal(k) / Decimal(Max[n - block_len]))
	k_rem = k - (k_block - 1) * Max[n - block_len]
	new_lst = getKthCycle(block_len, k_block, fact)
	if n - block_len > 0:
		rem_lst = getKthGoodPermutation(n - block_len, k_rem, Max, fact)
		for i in range(len(rem_lst)):
			rem_lst[i] += block_len
		new_lst.extend(rem_lst)
	return new_lst

def compute_fact(n, table):
	table[0] = 1
	table[1] = 1
	x = 2
	while x <= n:
		table[x] = x * table[x - 1]
		x += 1
	return
