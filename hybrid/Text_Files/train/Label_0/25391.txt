#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:16777216")
using namespace std;
const double pi = 3.14159265358979323846264338327950288419716939937511;
const double eps = 1e-9;
char ch_ch_ch[1 << 20];
string gs() {
  scanf("%s", ch_ch_ch);
  return string(ch_ch_ch);
}
string gl() {
  gets(ch_ch_ch);
  return string(ch_ch_ch);
}
string s;
struct dir {
  string name;
  int pr;
  vector<int> ch;
  dir() {}
  dir(string nm, int prr) {
    name = nm;
    pr = prr;
  }
};
vector<dir> G;
int n;
string pwd(int v) {
  if (v == 0) return "/";
  return pwd(G[v].pr) + G[v].name + "/";
}
void solution() {
  scanf("%d", &n);
  G.push_back(dir("", 0));
  int v = 0;
  for (int it = 0; it < (n); ++it) {
    cin >> s;
    if (s == "pwd") {
      cout << pwd(v) << endl;
    } else {
      cin >> s;
      if (s == "/") {
        v = 0;
        continue;
      }
      if (s[0] == '/') {
        v = 0;
        s = s.substr(1);
      }
      string nm = "";
      for (int i = 0; i < (s.size()); ++i)
        if (s[i] == '/') {
          bool f = false;
          if (nm == "..") {
            v = G[v].pr;
            nm = "";
            continue;
          }
          for (int i = 0; i < (G[v].ch.size()); ++i)
            if (G[G[v].ch[i]].name == nm) {
              f = true;
              v = G[v].ch[i];
              break;
            }
          if (!f) {
            G[v].ch.push_back(G.size());
            G.push_back(dir(nm, v));
            v = G.size() - 1;
          }
          nm = "";
        } else
          nm += s[i];
      bool f = false;
      if (nm == "..") {
        v = G[v].pr;
        nm = "";
        continue;
      }
      for (int i = 0; i < (G[v].ch.size()); ++i)
        if (G[G[v].ch[i]].name == nm) {
          f = true;
          v = G[v].ch[i];
          break;
        }
      if (!f) {
        G[v].ch.push_back(G.size());
        G.push_back(dir(nm, v));
        v = G.size() - 1;
      }
      nm = "";
    }
  }
}
int main(int argc, char** argv) {
  solution();
  return 0;
}
