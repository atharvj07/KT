def sort(array):
	for n in range(len(array) - 1):
		if array[n + 1] < array[n]:
			return False
	return True

def del_1(n, array):
	del array[0:int(n / 2)]
	return array

def del_2(n, array):
	del array[int(n / 2):len(array)]
	return array

def thanos_sort(n, array):
	if n == 1:
		return 1
	elif sort(array):
		return len(array)
	else:
		array1 = array.copy()
		array2 = array.copy()
		a = del_1(len(array1), array1)
		b = del_2(len(array2), array2)
		x = thanos_sort(len(a), a)
		y = thanos_sort(len(b), b)
		if x > y:
			return x
		else:
			return y
n = int(input())
array = list(map(int, input().split()))
print(thanos_sort(n, array))
