def isvalid(i, j, n):
	return i >= 0 and j >= 0 and (i < n) and (j < n)
n = int(input())
mat = []
for _ in range(n):
	mat.append(input())
f = 1
for i in range(n):
	for j in range(n):
		c = 0
		if isvalid(i + 1, j, n) and mat[i + 1][j] == 'o':
			c += 1
		if isvalid(i, j + 1, n) and mat[i][j + 1] == 'o':
			c += 1
		if isvalid(i - 1, j, n) and mat[i - 1][j] == 'o':
			c += 1
		if isvalid(i, j - 1, n) and mat[i][j - 1] == 'o':
			c += 1
		if c & 1:
			f = 0
			break
	if not f:
		break
print('YES' if f else 'NO')
