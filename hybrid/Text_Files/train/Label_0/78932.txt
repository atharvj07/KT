def solve(n, arr):
	if n == 1:
		return 'YES'
	if n == 2:
		if arr[0] == arr[1]:
			return 'YES'
		else:
			return 'NO'
	inv = arr[::-1]
	x1 = [arr[0]]
	x2 = [arr[n - 1]]
	for i in range(1, n - 1):
		x1.append(arr[i] - x1[-1])
	for i in range(n - 2, 0, -1):
		x2.append(arr[i] - x2[-1])
	x2 = x2[::-1]
	gt1 = [False for i in range(n - 1)]
	gt2 = [False for i in range(n - 1)]
	for i in range(n - 1):
		if x1[i] >= 0:
			gt1[i] = True
		else:
			break
	for i in range(n - 2, -1, -1):
		if x2[i] >= 0:
			gt2[i] = True
		else:
			break
	d = arr[n - 1] - x1[n - 2]
	if d == 0 and gt2[0]:
		return 'YES'
	for i in range(n - 1):
		cur = x1[i] - x2[i] + 2 * (arr[i + 1] - arr[i])
		if cur == 0 and (i == 0 or gt1[i - 1]) and (i == n - 2 or gt2[i + 1]) and (x1[i] + (arr[i + 1] - arr[i]) >= 0):
			return 'YES'
	return 'NO'
t = int(input())
for i in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	print(solve(n, arr))
