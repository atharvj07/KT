#include <bits/stdc++.h>
#pragma GCC optimize(3, "Ofast", "inline")
using namespace std;
bool Finish_read;
template <class T>
inline void read(T &x) {
  Finish_read = 0;
  x = 0;
  int f = 1;
  char ch = getchar();
  while (!isdigit(ch)) {
    if (ch == '-') f = -1;
    if (ch == EOF) return;
    ch = getchar();
  }
  while (isdigit(ch)) x = x * 10 + ch - '0', ch = getchar();
  x *= f;
  Finish_read = 1;
}
template <class T>
inline void print(T x) {
  if (x / 10 != 0) print(x / 10);
  putchar(x % 10 + '0');
}
template <class T>
inline void writeln(T x) {
  if (x < 0) putchar('-');
  x = abs(x);
  print(x);
  putchar('\n');
}
template <class T>
inline void write(T x) {
  if (x < 0) putchar('-');
  x = abs(x);
  print(x);
}
const int mod = 1e9 + 7;
const int inf = 0x3f3f3f3f;
const int max_n = 300005;
namespace {
inline int Add(int x, int y) { return (x += y) >= mod ? x - mod : x; }
inline int Sub(int x, int y) { return (x -= y) < 0 ? x + mod : x; }
inline int Mul(int x, int y) { return 1ll * x * y % mod; }
inline int Pow(int x, int y = mod - 2) {
  int res = 1;
  while (y) {
    if (y & 1) res = 1ll * res * x % mod;
    x = 1ll * x * x % mod;
    y >>= 1;
  }
  return res;
}
}  // namespace
int n;
pair<int, int> p[max_n];
vector<int> v;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d%d", &p[i].first, &p[i].second);
  int ml = inf, mr = -inf, md = inf, mu = -inf;
  for (int i = 1; i <= n; i++)
    ml = min(ml, p[i].first), mr = max(mr, p[i].first),
    md = min(md, p[i].second), mu = max(mu, p[i].second);
  int res = mr - ml + mu - md;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    ans = max(ans, p[i].first + p[i].second - ml - md);
    ans = max(ans, p[i].first + mu - ml - p[i].second);
    ans = max(ans, mr + p[i].second - p[i].first - md);
    ans = max(ans, mr + mu - p[i].first - p[i].second);
  }
  printf("%d", ans * 2);
  for (int i = 4; i <= n; i++) printf(" %d", res * 2);
  printf("\n");
}
