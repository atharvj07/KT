def compute_gcd(x, y):
	while y:
		(x, y) = (y, x % y)
	return x

def compute_lcm(x, y):
	lcm = x * y // compute_gcd(x, y)
	return lcm
lcms = []
lcm_to_this_moment = 1
for i in range(1, 22):
	lcm_to_this_moment = compute_lcm(lcm_to_this_moment, i + 1)
	lcms.append(lcm_to_this_moment)
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	flag = 1
	for (i, j) in zip(a, lcms[:n]):
		if i % j == 0:
			print('NO')
			flag = 0
			break
	if flag:
		print('YES')
