R = lambda : map(int, input().split())
F = lambda : [int(x) for x in input().split()]
(v1, v2) = R()
(t, d) = R()
ans = v1 + v2
a1 = []
a2 = []
t1 = v1
t2 = v2
for i in range(t - 2):
	t1 += d
	t2 += d
	a1.append(t1)
	a2.append(t2)
a2.reverse()
for i in range(t - 2):
	ans += min(a1[i], a2[i])
print(ans)
