import java.util.Scanner;

public class Marks {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int p = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        int sum = 0;
        int e;
        int lm = 0;
        int gm = 0;
        int [] res = new int[n - k];
        int c = 0;
        for (int i = 0; i < k; ++i) {
            e = sc.nextInt();
            sum += e;
            if (e >= y) {gm++;}
            else {lm++;}
        }
        if (lm > (n - 1) / 2) {
            System.out.println(-1);
            return;
        }

        int minY = (n + 1)/2 - gm;
        if (minY > 0) {
            for (int i = 0; i < minY; ++i) {
                res[c++] = y;
                sum+= y;
            }
        }
        if (sum > x) {
            System.out.println(-1);
            return;
        } else {
            for (int i = c; i < res.length; ++i) {
                res[i] = 1;
                sum++;
            }
        }
        if (sum > x) {
            System.out.println(-1);
            return;
        }
        else {
            for (int i : res) {
                System.out.print(i + " ");
            }
        }
    }
}
