import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.util.StringTokenizer;

public class cf145e {
  static FastIO in = new FastIO(), out = in;
  static char[] v;

  public static void main(String[] args) {
    int n = in.nextInt();
    int m = in.nextInt();
    v = in.next().trim().toCharArray();
    IntervalTree it = new IntervalTree(1, n);
    for (int i = 0; i < m; i++) {
      if (in.next().charAt(0) == 'c') {
        out.println(it.query(1, n));
      } else {
        int x = in.nextInt();
        int y = in.nextInt();
        it.add(x, y);
      }
    }
    out.close();
  }

  static class IntervalTree {
    int[] lo, hi;
    boolean[] delta;
    int[] zeros, ones, ans, opans;
    static int ZEROS = 0, ONES = 1, ANS = 2, OPANS = 3;

    // ans is count0, count1, ans, opans

    public IntervalTree(int l, int h) {
      int n = h - l + 1;
      zeros = new int[4 * n + 10];
      ones = new int[4 * n + 10];
      ans = new int[4 * n + 10];
      opans = new int[4 * n + 10];
      lo = new int[4 * n + 10];
      hi = new int[4 * n + 10];
      delta = new boolean[4 * n + 10];
      init(1, l, h);
    }

    void init(int i, int l, int h) {
      lo[i] = l;
      hi[i] = h;
      if (l == h) {
        if (v[l - 1] == '4')
          zeros[i] = 1;
        else
          ones[i] = 1;
        ans[i] = opans[i] = 1;
        return;
      }
      int m = (l + h - 1) / 2;
      init(2 * i, l, m);
      init(2 * i + 1, m + 1, h);
      aggregate(2 * i, 2 * i + 1, i);
    }

    public void push(int i) {
      if (!delta[i])
        return;
      int tmp = zeros[i];
      zeros[i] = ones[i];
      ones[i] = tmp;
      tmp = ans[i];
      ans[i] = opans[i];
      opans[i] = tmp;
      if (lo[i] != hi[i]) {
        delta[2 * i] ^= delta[i];
        delta[2 * i + 1] ^= delta[i];
      }
      delta[i] = false;
    }

    public void aggregate(int l, int r, int ret) {
      zeros[ret] = zeros[l] + zeros[r];
      ones[ret] = ones[l] + ones[r];
      ans[ret] = Math.max(ans[l] + ones[r], zeros[l] + ans[r]);
      opans[ret] = Math.max(opans[l] + zeros[r], ones[l] + opans[r]);
    }

    public void update(int i) {
      if (lo[i] == hi[i])
        return;
      // push(i);
      push(2 * i);
      push(2 * i + 1);
      aggregate(2 * i, 2 * i + 1, i);
    }

    public void add(int l, int r) {
      add(1, l, r);
    }

    public void add(int i, int l, int r) {
      if (lo[i] >= l && hi[i] <= r) {
        delta[i] ^= true;
        push(i);
      } else if (hi[i] >= l && lo[i] <= r) {
        push(i);
        add(2 * i, l, r);
        add(2 * i + 1, l, r);
        update(i);
      }
    }

    public int query(int l, int r) {
      return ans[1];
    }

  }

  static class FastIO extends PrintWriter {
    BufferedReader br;
    StringTokenizer st;

    public FastIO() {
      this(System.in, System.out);
    }

    public FastIO(InputStream in, OutputStream out) {
      super(new BufferedWriter(new OutputStreamWriter(out)));
      br = new BufferedReader(new InputStreamReader(in));
      scanLine();
    }

    public void scanLine() {
      try {
        st = new StringTokenizer(br.readLine().trim());
      } catch (Exception e) {
        throw new RuntimeException(e.getMessage());
      }
    }

    public int numTokens() {
      if (!st.hasMoreTokens()) {
        scanLine();
        return numTokens();
      }
      return st.countTokens();
    }

    public String next() {
      if (!st.hasMoreTokens()) {
        scanLine();
        return next();
      }
      return st.nextToken();
    }

    public double nextDouble() {
      return Double.parseDouble(next());
    }

    public long nextLong() {
      return Long.parseLong(next());
    }

    public int nextInt() {
      return Integer.parseInt(next());
    }
  }
}
