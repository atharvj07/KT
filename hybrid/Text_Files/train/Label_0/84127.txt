def m(a, b):
	(a, b) = (max(a, b), min(a, b))
	_max = -10000
	for i in range(1, a + 1):
		if a % i == 0:
			temp = a // i * (i - 1) + b * (1 - i)
			if temp > _max:
				_max = temp
	return _max
n = input()
n = int(n)
_ = input()
a = _.split()
a = [int(_) for _ in a]
answer = sum(a)
dic = [False] * 100
for i in a:
	dic[i - 1] = True
_max = -10000
for i in range(99):
	if not dic[i]:
		continue
	for j in range(i + 1, 100):
		if not dic[j]:
			continue
		temp = m(i + 1, j + 1)
		if temp > _max:
			_max = temp
if _max <= 0:
	print(answer)
else:
	print(answer - _max)
