#include <bits/stdc++.h>
using namespace std;
typedef long long* PLL;
typedef int* Pint;
char d[110];
int b;
int main() {
  int n;
  for (n = 0; (d[n] = getchar()) != 'e'; n++)
    ;
  scanf("%d", &b);
  for (int i = n; i <= 110; i++) d[i] = '0';
  n--;
  if ((n == 2) && (d[0] == '0') && (d[2] == '0')) {
    printf("0\n");
    exit(0);
  }
  if ((n == 2) && (d[2] == '0')) {
    printf("%c", d[0]);
    for (int i = 1; i <= b; i++) printf("0");
    printf("\n");
    exit(0);
  }
  for (int i = 1; i <= b; i++) swap(d[i], d[i + 1]);
  int l = 0, r = n;
  if (b + 1 >= n) r = b;
  while (d[l] == '0') l++;
  l -= (d[l] == '.');
  for (int i = l; i <= r; i++) printf("%c", d[i]);
  printf("\n");
}
