#include <bits/stdc++.h>
using namespace std;
long long n, m, x, y, w, z, q, k, l, r, ans = 1e17, cst[1000050], mid;
inline long long gtans(int u) {
  long long ret = 0, i = 0;
  for (; i < n && cst[i] < u; i += m) {
    ret += 2 * (u - cst[i]);
  }
  for (i = n - 1; i >= 0 && cst[i] > u; i -= m) {
    ret += 2 * (cst[i] - u);
  }
  return ret;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cin >> n >> m;
  for (int i = 0; i < n; ++i) cin >> cst[i];
  l = cst[0], r = cst[n - 1];
  while (r - l > 10) {
    int mid1 = l + (r - l) / 3;
    int mid2 = r - (r - l) / 3;
    if (gtans(mid1) < gtans(mid2))
      r = mid2;
    else
      l = mid1;
  }
  for (int i = l; i <= r; ++i) ans = min(ans, gtans(i));
  cout << ans;
}
