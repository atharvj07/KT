#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(0);
  long long n;
  cin >> n;
  vector<long long> a(n), t(n);
  map<long long, long long> cnt, lst;
  for (long long i = 0; i < n; i++) {
    cin >> a[i];
    lst[a[i]] = i + 1;
  }
  vector<multiset<long long>> v(n);
  vector<long long> cur(n);
  for (long long j = 0; j < n; j++) {
    cin >> t[j];
    cnt[a[j]]++;
    v[lst[a[j]] - 1].insert(t[j]);
    cur[lst[a[j]] - 1] += t[j];
  }
  set<vector<long long>> s;
  for (long long i = 0; i < n; i++) {
    s.insert({a[i], cnt[a[i]]});
  }
  long long ans = 0;
  while (s.size() > 0) {
    auto _to = s.begin();
    if ((*_to)[1] > 1) {
      vector<long long> to = *_to;
      long long it = lst[to[0]] - 1;
      cur[it] -= *(--v[it].end());
      v[it].erase(--v[it].end());
      ans += cur[it];
      long long ut = lst[to[0] + 1] - 1;
      if (ut == -1) {
        lst[to[0] + 1] = it + 1;
      } else {
        if (v[ut].size() < v[it].size()) {
          for (long long k : v[ut]) {
            v[it].insert(k);
            cur[it] += k;
          }
          lst[to[0] + 1] = it + 1;
        } else {
          swap(it, ut);
          for (long long k : v[ut]) {
            v[it].insert(k);
            cur[it] += k;
          }
          lst[to[0] + 1] = it + 1;
        }
      }
      s.erase(s.begin());
      if (!s.empty() && (*s.begin())[0] == to[0] + 1) {
        auto to2 = *s.begin();
        s.erase(s.begin());
        to2[1] += to[1] - 1;
        s.insert(to2);
      } else {
        to[0]++;
        to[1]--;
        s.insert(to);
      }
    } else {
      s.erase(s.begin());
    }
  }
  cout << ans;
  return 0;
}
