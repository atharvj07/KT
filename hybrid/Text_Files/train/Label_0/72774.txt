def solve():
	n = int(input())
	s = input()
	t = input()
	sum = 0
	prev = -1
	for i in range(n):
		if s[i] == '0':
			sum += 1
		else:
			sum -= 1
		if sum == 0:
			type = 0
			for j in range(prev + 1, i + 1):
				if s[j] == t[j]:
					if type == -1:
						print('NO')
						return
					type = 1
				else:
					if type == 1:
						print('NO')
						return
					type = -1
			prev = i
	for i in range(prev + 1, n):
		if s[i] != t[i]:
			print('NO')
			return
	print('YES')

def main():
	n = int(input())
	for i in range(n):
		solve()
main()
