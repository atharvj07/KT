#define ftoit(it, var) for (__typeof(var.begin()) it = var.begin(); it != var.end(); ++it)

using namespace std;

#define Pro "tmp"
#define maxN 300007

int a[maxN], b[maxN];
set <int> st[maxN];

int mergeSet(int a, int b) {
    if ((int)st[a].size() < (int)st[b].size()) swap(a, b);
    ftoit (it, st[b]) st[a].insert(*it);
    st[b].clear();
    return a;
}

int main() {

    int n;
    scanf("%d", &n);

    fto (i, 1, n) {
        scanf("%d", &a[i]);
        b[i] = a[i];
    }
    sort(b+1, b+n+1);

    queue <int> q;
    fto (i, 1, n) {
        st[i].insert(a[i]);
        q.push(i);
    }

    ll sol = 0LL;
    fto (i, 1, n) {
        int ans = 0;
        set <int> cur_set;

        int new_idx = 0;
        while (!q.empty()) {
            int idx = q.front();
            q.pop();

            ans += (int)st[idx].size();
            if (st[idx].count(b[i])) {
                st[idx].erase(b[i]);
                new_idx = mergeSet(idx, new_idx);

                if ((int)st[new_idx].size()) q.push(new_idx);
                break;
            }

            new_idx = mergeSet(idx, new_idx);
        }

        sol += ans-1;
    }

    printf("%lld\n", sol);
    return 0;
}
