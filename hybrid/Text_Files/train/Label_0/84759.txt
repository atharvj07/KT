def shift(l, n):
	return l[n:] + l[:n]
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	b = sorted(a)
	ans = []
	for i in range(n):
		if a[i] == b[i]:
			continue
		num = b[i]
		j = a[i:].index(num) + i
		d = j - i
		ans.append([i + 1, j + 1, d])
		c = a[i:j + 1]
		c = shift(c, d)
		a[i:j + 1] = c
	print(len(ans))
	for i in ans:
		print(*i)
