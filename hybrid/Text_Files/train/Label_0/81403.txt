for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	c1 = 0
	c2 = 0

	def odd(a, n):
		v1 = []
		v2 = []
		for i in range(n):
			if i % 2 == 0 and a[i] % 2 == 0:
				v1.append(i)
			if i % 2 == 1 and a[i] % 2 == 1:
				v2.append(i)
		c = 0
		for i in range(len(v1)):
			c += max(v1[i], v2[i]) - min(v1[i], v2[i])
		return c

	def even(a, n):
		v1 = []
		v2 = []
		for i in range(n):
			if i % 2 == 0 and a[i] % 2 == 1:
				v1.append(i)
			if i % 2 == 1 and a[i] % 2 == 0:
				v2.append(i)
		c = 0
		for i in range(len(v1)):
			c += max(v1[i], v2[i]) - min(v1[i], v2[i])
		return c
	for i in a:
		if i & 1:
			c1 += 1
		else:
			c2 += 1
	if n % 2 == 0:
		if c1 == c2:
			print(min(odd(a, n), even(a, n)))
		else:
			print(-1)
	elif c1 + 1 == c2:
		print(even(a, n))
	elif c2 + 1 == c1:
		print(odd(a, n))
	else:
		print(-1)
