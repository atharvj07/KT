public class C {
	static int n,m,w,q;
	static int inf = (int)1e9+5001;
	static long linf = (long)1e17;
	static Store[] ws;
	public static void main(String[] args) throws IOException {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		n = in.nextInt();
		m = in.nextInt();
		ArrayList<Integer>[] edges = new ArrayList[n];
		for(int i=0;i<n;i++)
			edges[i] = new ArrayList<>();
		for(int i=0;i<m;i++) {
			int u = in.nextInt()-1, v = in.nextInt()-1;
			edges[u].add(v);
			edges[v].add(u);
		}
		w = in.nextInt();
		ws = new Store[w];
		for(int i=0;i<w;i++) {
			ws[i] = new Store(in.nextInt()-1, in.nextInt(), in.nextInt());
		}
		Arrays.sort(ws);
		q = in.nextInt();
		ArrayDeque<Integer> queue = new ArrayDeque<>();
		int[] dist = new int[n];
		while(q --> 0) {
			int g = in.nextInt()-1, r = in.nextInt(), a = in.nextInt();
			Arrays.fill(dist, inf);
			dist[g] = 0;
			queue.offer(g);
			while(!queue.isEmpty()) {
				int st = queue.poll();
				for(int e : edges[st]) {
					if(dist[e] == inf) {
						dist[e] = dist[st]+1;
						queue.offer(e);
					}
				}
			}
			int lo = 0, hi = n;
			while(lo < hi) {
				int mid = (lo+hi)/2;
				long cost = solve(r,g,mid,dist);
				if(cost <= a) {
					hi = mid;
				} else {
					lo = mid+1;
				}
			}
			out.println(lo == n ? -1 : lo);
		}
		out.close();
		
	}
	static long solve(int r, int g, int t, int[] dist) {
		long cost = 0;
		for(Store b : ws) {
			if(dist[b.c] > t) continue;
			if(r == 0) break;
			long amt = Math.min(r, b.k);
			cost += b.p * amt;
			r -= amt;
		}
		if(r != 0)
			cost = linf;
		return cost;
	}
	static class Store implements Comparable<Store>{
		int c,k,p;
		public Store(int c, int k, int p) {
			this.c = c;
			this.k = k;
			this.p = p;
		}
		@Override
		public int compareTo(Store o) {
			return Integer.compare(p, o.p);
		}
		
	}
	static class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(InputStream in) {
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer("");
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String next() throws IOException {
			while (!st.hasMoreTokens()) {
				st = new StringTokenizer(br.readLine());
			}
			return st.nextToken();
		}
	}
}
