#include <bits/stdc++.h>
using namespace std;
int main() {
  string s1, s2;
  int l1, l2, i, count = 0, b = 0, j = 0, z = 0;
  cin >> s1 >> s2;
  l1 = s1.length();
  l2 = s2.length();
  for (i = 0; i < l2; i++)
    if (s2[i] == '1') b++;
  for (i = 0; i < l2; i++) {
    if (s1[i] == '1') count++;
  }
  if ((count ^ b) % 2 == 0) z++;
  for (i = l2; i < l1; i++) {
    if (s1[i - l2] == '1' && s1[i] == '0') count--;
    if (s1[i - l2] == '0' && s1[i] == '1') count++;
    if ((count ^ b) % 2 == 0) z++;
  }
  cout << z;
  return 0;
}
