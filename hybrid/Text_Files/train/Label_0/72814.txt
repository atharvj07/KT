(n, s) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
t = False
if a[0] == 1 and a[s - 1] == 1:
	print('YES')
elif a[0] == 0:
	print('NO')
elif a[s - 1] == 0 and b[s - 1] == 0:
	print('NO')
elif a[s - 1] == 0:
	for i in range(s, n):
		if a[i] == 1 and b[i] == 1:
			print('YES')
			t = True
			break
	if not t:
		print('NO')
