#include <bits/stdc++.h>
using namespace std;
#define int long long int
int32_t main() {
	int t;
	cin>>t;
	while(t--){
	    int n,k;
	    cin>>n>>k;
	    vector<int> a(n);
	    for(int i=0;i<n;i++){
	        cin>>a[i];
	    }
	    int ans=0;
	    int left=0,right=1e11;
	    while(left<=right){
	        int mid=(left+right)>>1;
	        int sum=a[0]+mid;
	        bool ok=true;
	        for(int i=1;i<n;i++){
	            if(a[i]*100>k*sum){
	                ok=false;
	                break;
	            }
	            sum+=a[i];
	        }
	        if(ok){
	            ans=mid;
	            right=mid-1;
	        }
	        else{
	            left=mid+1;
	        }
	    }
	    cout<<ans<<endl;
	}
	return 0;
}
