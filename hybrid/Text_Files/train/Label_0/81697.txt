from math import *

def prm(x):
	s = int(sqrt(x + 0.5))
	for i in range(2, s + 1):
		if not x % i:
			return False
	return True

def ap(a):
	while len(a) >= 2:
		print(a[-1], a[-2])
		a.pop()
		a.pop()
(n, cnt, ans, vis, ansl, ansr) = (int(input()), 0, 0, [False] * 100100, [], [])
prime = filter(prm, range(3, n // 2 + 1))
for x in prime:
	(tcnt, nlst) = (0, [])
	for i in range(1, n // x + 1):
		if not vis[i * x]:
			vis[i * x] = True
			nlst.append(i)
			tcnt += 1
	ans += tcnt >> 1
	cnt += tcnt & 1
	if tcnt & 1:
		nlst = nlst[0:1] + nlst[2:]
		ansr.append(x << 1)
	ansl += list(map(lambda k: x * k, nlst))
for i in range(1, n + 1):
	if not vis[i] and (not i & 1):
		ansr.append(i)
		cnt += 1
print(ans + (cnt >> 1))
ap(ansl)
ap(ansr)
