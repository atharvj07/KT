#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; ++i) cin >> arr[i];
  int src, trg;
  cin >> src >> trg;
  --src, --trg;
  int distanceA = 0, distanceB = 0;
  if (src != trg) {
    for (int i = src + n; i % n != trg; --i) distanceA += arr[(i - 1) % n];
    for (int i = src; i % n != trg; ++i) distanceB += arr[i % n];
  }
  cout << min(distanceA, distanceB);
}
