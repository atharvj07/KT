n = int(input())
G = {}
for i in range(n - 1):
	(a, b) = map(int, input().split())
	try:
		G[a].append(b)
	except:
		G[a] = [b]
	try:
		G[b].append(a)
	except:
		G[b] = [a]
leaves = []
for i in G.keys():
	if len(G[i]) == 1:
		leaves.append(i)
	else:
		sv = i
val = [-1 for i in range(n + 1)]
Q = [sv]
val[sv] = 0
tp = 0
while len(Q) != 0:
	a = Q.pop()
	iv = val[a]
	tc = -1
	for i in G[a]:
		if val[i] == -1:
			val[i] = iv + 1
			if len(G[i]) == 1:
				tc = tc + 1
			else:
				Q.append(i)
	tp = tp + max(tc, 0)
v = val[leaves[0]]
flag = 0
for i in range(1, len(leaves)):
	if abs(val[leaves[i]] - v) % 2 != 0:
		flag = 1
		break
if flag == 0:
	print(1, end=' ')
else:
	print(3, end=' ')
print(n - tp - 1)
