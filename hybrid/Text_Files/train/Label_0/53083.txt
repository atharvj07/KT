import java.util.*;
public class Solution{
    public static void main(String args[]){
        Scanner in=new Scanner (System.in);
        int t=in.nextInt();
        T:while(--t>=0){
            int a=in.nextInt();
            int b=in.nextInt();
            int p=in.nextInt();
            
            String s=in.next();
            int c=0;
            
            for(int i=s.length()-2;i>=0;){
                char ch=s.charAt(i);
                c+=ch=='A' ? a:b;
                if(c>p){
                    System.out.println(i+2);
                    continue T;
                }
                while(i>=0 && s.charAt(i)==ch){
                    i--;
                }
            }
            System.out.println(1);
        }
    }
}
