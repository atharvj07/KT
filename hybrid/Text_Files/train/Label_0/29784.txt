#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using PII = pair<int, int>;
using VI = vector<int>;
const int N = 2005;
char grid[N][N];
int dp[N][N][2], best[N][N];
int vis[N][N];
vector<PII> sub;
int compute(int i, int j, bool root) {
  sub.clear();
  if (i > 0 && grid[i - 1][j] == 'v') {
    int ii = i - 1, jj = j;
    sub.emplace_back(1 + dp[ii][jj][0], ii * N + jj);
    sub.emplace_back(1 + dp[ii][jj][1], ii * N + jj);
  }
  if (grid[i + 1][j] == '^') {
    int ii = i + 1, jj = j;
    sub.emplace_back(1 + dp[ii][jj][0], ii * N + jj);
    sub.emplace_back(1 + dp[ii][jj][1], ii * N + jj);
  }
  if (j > 0 && grid[i][j - 1] == '>') {
    int ii = i, jj = j - 1;
    sub.emplace_back(1 + dp[ii][jj][0], ii * N + jj);
    sub.emplace_back(1 + dp[ii][jj][1], ii * N + jj);
  }
  if (grid[i][j + 1] == '<') {
    int ii = i, jj = j + 1;
    sub.emplace_back(1 + dp[ii][jj][0], ii * N + jj);
    sub.emplace_back(1 + dp[ii][jj][1], ii * N + jj);
  }
  sort(sub.rbegin(), sub.rend());
  if (!sub.empty()) {
    dp[i][j][0] = sub[0].first;
    dp[i][j][1] = sub[1].first;
  }
  int ret = dp[i][j][0] + dp[i][j][1];
  if (root && !sub.empty() && sub[0] == sub[1]) {
    ret--;
    for (int it = 2; it < (int)sub.size(); it++) {
      if (sub[0] != sub[it] && sub[0].first == sub[it].first) {
        ret++;
        break;
      }
    }
  }
  return ret;
}
int dfs(int i, int j, bool root = false) {
  if (vis[i][j] == 2) return 0;
  if (vis[i][j] == 1) {
    cout << "-1\n";
    exit(0);
  }
  vis[i][j] = 1;
  if (i > 0 && grid[i - 1][j] == 'v') {
    int ii = i - 1, jj = j;
    dfs(ii, jj);
  }
  if (grid[i + 1][j] == '^') {
    int ii = i + 1, jj = j;
    dfs(ii, jj);
  }
  if (j > 0 && grid[i][j - 1] == '>') {
    int ii = i, jj = j - 1;
    dfs(ii, jj);
  }
  if (grid[i][j + 1] == '<') {
    int ii = i, jj = j + 1;
    dfs(ii, jj);
  }
  vis[i][j] = 2;
  return compute(i, j, root);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> grid[i];
  }
  int ans = 0;
  priority_queue<int, VI, greater<int>> vals;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '#') {
        ans = max(ans, dfs(i, j, true));
        vals.push(dp[i][j][0]);
        if (vals.size() > 2) vals.pop();
      }
    }
  }
  if (vals.size() >= 2) {
    int a = vals.top();
    vals.pop();
    int b = vals.top();
    vals.pop();
    ans = max(ans, a + b);
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] != '#' && !vis[i][j]) {
        cout << "-1\n";
        return 0;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
