import java.util.*;

public class B {
    static class Racer implements Comparable<Racer> {
        String name;
        int pts;

        public Racer(String n, int p) {
            name = n;
            pts = p;
        }

        public int compareTo(Racer z) {
            if (z.pts > pts || z.pts == pts && z.name.compareTo(name) < 0)
                return 1;
            if (z.pts == pts && z.name.equals(name))
                return 0;
            return -1;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        String[] names = new String[n];
        int[] scores = new int[n];
        for (int i = 0; i < n; i++) {
            names[i] = sc.next();
            scores[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] points = new int[Math.max(m, 1)];
        if (m != 0)
            for (int i = 0; i < points.length; i++) {
                points[i] = sc.nextInt();
            }
        Arrays.sort(points);
        String nick = sc.next();
        Racer[] ar = new Racer[n];
        int upIdx = m - 2, downIdx = 0;
        for (int i = 0; i < n; i++) {
            ar[i] = new Racer(names[i], scores[i]);
            if (names[i].equals(nick)) {
                ar[i].pts += points[points.length- 1];
            }
        }
        Arrays.sort(ar);
        int dir = -1;
        for (int j=0; upIdx>=0; ) {
            if (ar[j].name.equals(nick)) {
                dir *= -1;
                j=n-1;
            } else {
                if (dir == -1){
                    ar[j].pts += points[upIdx--];
                    j++;
                }
                else{
                    ar[j].pts += points[upIdx--];
                    j--;
                }
            }
        }
        Arrays.sort(ar);
        int maxRank = -1;
        for (int i = 0; i < n; i++) {
            if (ar[i].name.equals(nick)) {
                maxRank = (i + 1);
                break;
            }
        }
        ar = new Racer[n];
        upIdx = m - 1;
        if(m==n)
        downIdx = 1;
        else downIdx=0;
        for (int i = 0; i < n; i++) {
            ar[i] = new Racer(names[i], scores[i]);
            if (names[i].equals(nick)&&m==n) {
                ar[i].pts += points[0];
            }
        }
        Arrays.sort(ar);
        dir = 1;
        int nickIdx = -1;
        for (int i = 0; upIdx>=downIdx&&i<n; i++) {
            if (ar[i].name.equals(nick)) {
                dir *= -1;
                nickIdx = i;
            } else if(nickIdx!=-1){
                    int l = downIdx, h = upIdx + 1, mid;
                    while (l < h) {
                        mid = l + (h - l) / 2;
                        if (points[mid] + ar[i].pts < ar[nickIdx].pts
                                || points[mid] + ar[i].pts == ar[nickIdx].pts
                                && ar[i].name.compareTo(ar[nickIdx].name) > 0) {
                            l = mid + 1;
                        } else
                            h = mid;
                    }
                    if (l<=upIdx&&(points[l] + ar[i].pts > ar[nickIdx].pts
                            || points[l] + ar[i].pts == ar[nickIdx].pts
                            && ar[i].name.compareTo(ar[nickIdx].name) < 0)) {
                        ar[i].pts += points[l];
                        downIdx = l + 1;
                    } else {
                        break;
                    }
            }
        }
        Arrays.sort(ar);
        int minRank = -1;
        for (int i = 0; i < n; i++) {
            if (ar[i].name.equals(nick)) {
                minRank = (i + 1);
                break;
            }
        }
        System.out.println(maxRank + " " + minRank);
    }
}
