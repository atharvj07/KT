import sys
input = lambda : sys.stdin.readline().rstrip()
for _ in range(int(input())):
	(K, X) = map(int, input().split())
	a = K * (K + 1) // 2
	if X <= a:
		b = int((2 * X) ** 0.5)
		ans = b
		if b * (b + 1) < X * 2:
			ans += 1
		print(ans)
	elif X >= a + a - K:
		print(K + K - 1)
	else:
		X = a + a - K - X
		b = int((2 * X) ** 0.5)
		ans = K + K - 1 - b
		if b * (b + 1) > 2 * X:
			ans += 1
		print(ans)
