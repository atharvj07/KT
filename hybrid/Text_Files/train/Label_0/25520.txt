#include <bits/stdc++.h>
int main() {
  int t, a;
  scanf("%d", &t);
  while (t--) {
    int flag = 0;
    scanf("%d", &a);
    for (int i = 3; i <= 800; i++) {
      if ((i - 2) * 180 == a * i) {
        flag = 1;
        printf("YES\n");
        break;
      }
    }
    if (flag == 0) printf("NO\n");
  }
  return 0;
}
