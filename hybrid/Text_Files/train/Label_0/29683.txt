#include <bits/stdc++.h>
using namespace std;
long long n, m, k;
long long D[300000];
long long ck = 0;
int cka[300000];
priority_queue<pair<pair<long long, long long>, long long> > Q;
vector<pair<pair<long long, long long>, long long> > G[300000];
long long u, d;
void findQ(long long d) {
  if (ck == min(n - 1, k)) return;
  for (pair<pair<long long, long long>, long long> x : G[d])
    Q.push({{-x.first.second - D[d], x.first.first}, x.second});
  while (1) {
    if (cka[Q.top().first.second] == 1)
      Q.pop();
    else
      break;
  }
  u = Q.top().first.first;
  d = Q.top().first.second;
  cout << Q.top().second + 1 << " ";
  cka[d] = 1;
  D[d] = -u;
  Q.pop();
  ck++;
  findQ(d);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m >> k;
  for (long long i = 0, x, y, w; i < m; i++) {
    cin >> x >> y >> w;
    G[x - 1].push_back({{y - 1, w}, i});
    G[y - 1].push_back({{x - 1, w}, i});
  }
  cout << min(n - 1, k) << endl;
  cka[0] = 1;
  findQ(0);
}
