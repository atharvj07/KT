(n, m) = map(int, input().split())
bal = [0] * n
for i in range(m):
	(x, y, z) = map(int, input().split())
	bal[x - 1] -= z
	bal[y - 1] += z
owe = []
for i in range(n):
	if bal[i] < 0:
		owe.append([i, -bal[i]])
ans = []
for i in range(n):
	while bal[i] > 0 and owe:
		val = min(bal[i], owe[0][1])
		ans.append([owe[0][0] + 1, i + 1, val])
		bal[i] -= val
		owe[0][1] -= val
		if owe[0][1] == 0:
			owe.pop(0)
print(len(ans))
for i in range(len(ans)):
	print(' '.join(map(str, ans[i])))
