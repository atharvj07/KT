#include <bits/stdc++.h>
using namespace std;
mt19937 Rnd(chrono::high_resolution_clock::now().time_since_epoch().count());
template <typename T>
inline void chkmax(T &x, T y) {
  if (x < y) x = y;
}
template <typename T>
inline void chkmin(T &x, T y) {
  if (x > y) x = y;
}
inline int read() {
  int x = 0;
  char c = getchar();
  while (c < 48) c = getchar();
  while (c > 47) x = (x << 3) + (x << 1) + (c ^ 48), c = getchar();
  return x;
}
const int maxn = 2e5 + 10;
int n, a[maxn], col[maxn];
bool vis[maxn];
int tot, data[maxn];
int check(int len) {
  for (int st = (1), stend = (len); st <= stend; ++st) {
    bool flg = 1;
    for (int i = st + len; i <= tot; i += len) {
      flg &= col[data[i]] == col[data[i - len]];
    }
    if (flg) return len;
  }
  return 1e9;
}
void solve() {
  for (int i = (1), iend = (n); i <= iend; ++i) vis[i] = 0;
  cin >> n;
  for (int i = (1), iend = (n); i <= iend; ++i) a[i] = read();
  for (int i = (1), iend = (n); i <= iend; ++i) col[i] = read();
  int res = 1e9;
  for (int u = (1), uend = (n); u <= uend; ++u)
    if (!vis[u]) {
      vis[u] = 1, data[tot = 1] = u;
      for (int v = a[u]; u != v; v = a[v]) {
        vis[v] = 1, data[++tot] = v;
      }
      int lim = sqrt(tot) + 0.5;
      for (int len = (1), lenend = (lim); len <= lenend; ++len)
        if (tot % len == 0) {
          chkmin(res, min(check(len), check(tot / len)));
        }
    }
  printf("%d\n", res);
}
int main() {
  int T = read();
  while (T--) solve();
  return 0;
}
