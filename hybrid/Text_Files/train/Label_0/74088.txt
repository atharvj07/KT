from sys import stdin

def solve():
	results = []
	for _ in range(int(stdin.readline().strip())):
		(N, H) = map(int, stdin.readline().strip().split())
		A = sorted(list(map(int, stdin.readline().strip().split())), reverse=True)
		if A[0] >= H:
			results.append(1)
			continue
		maxtwo = A[0] + A[1]
		c = H // maxtwo
		h = c * maxtwo
		c *= 2
		i = 0
		while h < H:
			h += A[i]
			i = (i + 1) % 2
			c += 1
		results.append(c)
	print('\n'.join((str(x) for x in results)))
solve()
