const int maxn=4e4+100;
Double x[maxn],y[maxn];
Double a[3][3],c[3][3],b[3];
Double replace(int ix)
{
    int i,j;
    double res=0;
    for(i=0;i<3;++i)
        for(j=0;j<3;++j)c[i][j]=a[i][j];
    for(j=0;j<3;++j)c[ix][j]=b[j];
    for(i=0;i<3;++i)
    {
        res+=c[0][i]*c[1][(i+1)%3]*c[2][(i+2)%3];
        res-=c[0][i]*c[1][(i+2)%3]*c[2][(i+1)%3];
    }
    return res;
}
int main()
{
    int n,i;
	//cout <<sizeof(double)<<" "<<sizeof(long double)<<" "<<sizeof(float)<<" "<<sizeof(int)<<endl;
    scanf("%d",&n);
    memset(a,0,sizeof(a));
    memset(b,0,sizeof(b));
    for(i=0;i<n;++i)
    {
        scanf("%Lf%Lf",&x[i],&y[i]);
        b[0]+=(sqr(x[i])+sqr(y[i]))*x[i];
        b[1]+=(sqr(x[i])+sqr(y[i]))*y[i];
        b[2]+=(sqr(x[i])+sqr(y[i]));
        a[0][0]+=sqr(x[i]);a[0][1]+=x[i]*y[i];a[0][2]+=x[i];
        a[1][0]+=x[i]*y[i];a[1][1]+=sqr(y[i]);a[1][2]+=y[i];
        a[2][0]+=x[i];a[2][1]+=y[i];a[2][2]+=1;
    }
    for(i=0;i<3;++i)b[i]=-b[i];
    Double delta=0;
    for(i=0;i<3;++i)
    {
        delta+=a[0][i]*a[1][(i+1)%3]*a[2][(i+2)%3];
        delta-=a[0][i]*a[1][(i+2)%3]*a[2][(i+1)%3];
    }
    //cout <<delta<<endl;
    Double A=replace(0)/delta;
    Double B=replace(1)/delta;
    Double C=replace(2)/delta;
    printf("%.3Lf %.3Lf %.3Lf\n",A,B,C);
    //printf("%.9f %.9f %.9f\n",A,B,C);
    return 0;
}
