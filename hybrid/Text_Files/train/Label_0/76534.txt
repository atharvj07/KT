def array(f):
	return list(map(f, input().split()))

def floor_sum(X, N):
	res = 0
	for i in range(X // N, X + 1):
		if i * i > X:
			for j in range(1, N + 1):
				if X // j >= i:
					res += X // j
				else:
					break
			break
		if i == 0:
			continue
		n = min(X // i, N) - X // (i + 1)
		res += n * i
	return res

def slv():
	(x, y) = map(int, input().split())
	ans = 0
	for b in range(1, y + 1):
		if x // (b + 1) >= b - 1:
			ans += b - 1
		else:
			ans += floor_sum(x, y + 1) - floor_sum(x, b)
			break
	print(ans)
	return

def main():
	T = int(input())
	for _ in range(T):
		slv()
main()
