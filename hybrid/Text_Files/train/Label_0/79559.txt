n = int(input())
st = input()
q = 1
x = max(int(st.split()[0]), int(st.split()[1]))
for i in range(n - 1):
	st = input()
	if int(st.split()[0]) <= x or int(st.split()[1]) <= x:
		q += 1
		if max(int(st.split()[0]), int(st.split()[1])) <= x:
			x = max(int(st.split()[0]), int(st.split()[1]))
		else:
			x = min(int(st.split()[0]), int(st.split()[1]))
	else:
		print('NO')
		break
if q == n:
	print('YES')
