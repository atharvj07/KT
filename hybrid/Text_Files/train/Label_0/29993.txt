#include <bits/stdc++.h>
using namespace std;
int main() {
  string s1, s2;
  getline(cin, s1, char(10));
  getline(cin, s2, char(10));
  int n1 = s1.length();
  int n2 = s2.length();
  int e1 = 0;
  int e2 = 0;
  for (int i = 0; i < n1; i++) e1 += (int(s1[i] - '0'));
  for (int i = 0; i < n2; i++) e2 += (int(s2[i] - '0'));
  if ((e1 + 1 < e2) || ((e1 % 2 == 0) && (e1 == e2 - 1))) {
    cout << "NO";
  } else
    cout << "YES";
}
