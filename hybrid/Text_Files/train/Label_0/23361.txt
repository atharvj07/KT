#include <iostream>
#include <stdio.h>
using namespace std;
typedef pair<int,int> P;
int p,q,a,n;

P add(P a,P b){
    P res = {a.first*b.second+a.second*b.first,a.second*b.second};
    return res;
}

int dfs(int len,int num,int mul,P frac){
    if(frac.first*q==frac.second*p) return 1;
    if(frac.first*q>frac.second*p) return 0;
    int cnt = 0;
    if(len!=n) for(int i=num;mul*i<=a;i++) cnt += dfs(len+1,i,mul*i,add(frac,P(1,i)));
    return cnt;
}

int main(){
    while(1){
        scanf("%d%d%d%d",&p,&q,&a,&n);
        if(p==0) return 0;
        int ans = 0;
        printf("%d\n",dfs(0,1,1,P(0,1)));
    }
}
