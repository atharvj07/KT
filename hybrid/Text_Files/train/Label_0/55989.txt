import java.util.*;

public class calc {
    public static void main( String[] args ) {
	Scanner in = new Scanner( System.in );
	int n = Integer.parseInt( in.nextLine() );
	long s = 0;
	long[][] d = new long[n][5];
	for ( int i = 0; i < n; i++ ) {
	    String[] dd = in.nextLine().split( " " );
	    for( int j = 0; j < 5; j++ )
		d[i][j] = Long.parseLong( dd[j] );	    
	    s += d[i][0];
	    d[i][2] %= d[i][4];
	}
	int mx = -1;
	ArrayList<ArrayList<prob>> p = new ArrayList<ArrayList<prob>>();
	p.add( new ArrayList<prob>() );
	for ( int i = 0; i < n; i++ ) {
	    long a = d[i][1];
	    int c = 0;
	    p.get( c ).add( new prob( (int)a, i ) );
	    for ( int j = 1; j < d[i][0]; j++ ) {
		long b = ( a * d[i][2] + d[i][3] ) % d[i][4];
		if ( b < a )
		    if ( p.size() == ++c )
			p.add( new ArrayList<prob>() );
		if ( s <= 200000 )
		    p.get( c ).add( new prob( (int)b, i ) );
		a = b;
	    }
	    mx = Math.max( mx, c );
	}
	System.out.println( mx );
	if ( s <= 200000 ) {
	    for ( int i = 0; i <= mx; i++ ) {
		Collections.sort( p.get(i) );
		for ( int j = 0; j < p.get(i).size(); j++ )
		    System.out.println( p.get(i).get(j) );
	    }
	}
    }
}

class prob implements Comparable<prob> {
    int r, s;

    public prob( int r, int s ) {
	this.r = r;
	this.s = s;
    }

    public int compareTo( prob p ) {
	return this.r - p.r;
    }

    public String toString() {
	return r + " " + (s + 1);
    }
}
