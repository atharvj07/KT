#include <bits/stdc++.h>
int n, l, r, val, hbit;
std::vector<std::array<int, 3>> ans;
int main() {
    std::cin>>l>>r;
    if(l==r) {std::cout<<"YES\n2 1\n1 2 "<<l<<"\n"; exit(0);}
    if(l==1) {l++; ans.push_back({1, 23, 1});}
    val = r-l; 
    while((val >> (hbit)) > 0)hbit++;
    for(int i = 2; i < 22; ++i) {
        ans.push_back({i, 23, l-1});
        for(int j = i+1; j < 22; ++j) ans.push_back({i, j, 1<<(19 - (j-2))});
    }
    for(int i = hbit-1; i >=0; --i) if(((val >> i) & 1)) ans.push_back({1, 21 - i, ((val >> (i+1)) << (i+1))+1});
    ans.push_back({1, 22, 1}); ans.push_back({22, 23, r-1});
    std::cout<<"YES\n23 "<<ans.size()<<"\n";
    for(auto p: ans) std::cout<<p[0]<<" "<<p[1]<<" "<<p[2]<<"\n";
    return 0;
}
