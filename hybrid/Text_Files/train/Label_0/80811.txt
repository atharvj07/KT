T = int(input())
for _ in range(T):
	(n, l, r, s) = [int(x) for x in input().split()]
	ma = (2 * n - r + l) * (r - l + 1) // 2
	mi = (2 - l + r) * (r - l + 1) // 2
	if s > ma or s < mi:
		print(-1)
	else:
		x = int((2 * s / (r - l + 1) + l - r) / 2)
		dad = s - int((2 * x + r - l) * (r - l + 1) / 2)
		arr = [0] * n
		for i in range(l - 1, r):
			arr[i] = x + (i - l + 1)
		cnt = 0
		for i in list(range(r - l + 1))[::-1]:
			add = min(n - cnt - arr[i + l - 1], dad)
			cnt += 1
			arr[i + l - 1] += add
			dad -= add
			if not dad:
				break
		sett = set(arr)
		cur = 1
		for i in range(0, l - 1):
			while cur in sett:
				cur += 1
			arr[i] = cur
			cur += 1
		for i in range(r, n):
			while cur in sett:
				cur += 1
			arr[i] = cur
			cur += 1
		print(' '.join([str(x) for x in arr]))
