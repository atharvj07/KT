def check(c, a):
	update = []
	for i in range(3):
		update.append(c[i] - a[i])
		if update[i] < 0:
			return 'NO'
	if update[0] + update[2] >= a[3]:
		left = a[3] - update[0]
		if left > 0:
			update[2] = update[2] - left
			if update[2] < 0:
				return 'NO'
		if update[1] + update[2] >= a[4]:
			return 'YES'
		else:
			return 'NO'
	else:
		return 'NO'
t = int(input())
while t > 0:
	c = [int(x) for x in input().strip().split()]
	a = [int(x) for x in input().strip().split()]
	print(check(c, a))
	t = t - 1
