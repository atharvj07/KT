import math
from collections import Counter
from math import gcd, floor, sqrt

def divs(n: int):
	d = set()
	for div in range(1, floor(sqrt(n)) + 1):
		if n % div == 0:
			d.add(div)
			d.add(n // div)
	return d
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	candidates = []
	k = -1
	for i0 in range(n):
		numberOfAi0Value = 0
		d = []
		for i in range(n):
			if a[i] == a[i0]:
				numberOfAi0Value += 1
			elif a[i] > a[i0]:
				d.append(a[i] - a[i0])
		if numberOfAi0Value >= n // 2:
			k = math.inf
			continue
		div_counter = Counter()
		for ai in d:
			for div in divs(ai):
				div_counter[div] += 1
		for p in div_counter.most_common():
			if p[1] >= n // 2 - numberOfAi0Value:
				k = max(k, p[0])
	print(-1 if k == math.inf else k)
