#include <bits/stdc++.h>
using namespace std;
int m[200001];
int m1[200001];
int inds[200001];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    scanf("%d", &m[i]);
  }
  for (int i = 0; i < n; ++i) {
    scanf("%d", &m1[i]);
    inds[m1[i]] = i;
  }
  int res = 0;
  for (int i = 1; i < n; ++i) {
    if (inds[m[i]] < inds[m[i - 1]]) {
      cout << n - i << endl;
      return 0;
    }
  }
  cout << res << endl;
  return 0;
}
