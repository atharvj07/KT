#include <stdio.h>
int n, m, x, a, b;
int h[100000];
int e[100000];
int main() {
	scanf("%d%d", &n, &m);
	for (int i = 0; i < n; i++) {
		scanf("%d", h + i);
	}
	for (int i = 0; i < m; i++) {
		scanf("%d%d", &a, &b);
		a--; b--;
		if (h[a] <= h[b]) {
			e[a]++;
		}
		if (h[b] <= h[a]) {
			e[b]++;
		}
	}
	for (int i = 0; i < n; i++) {
		if (!e[i])x++;
	}
	printf("%d\n", x);
}
