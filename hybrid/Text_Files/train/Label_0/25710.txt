#include <bits/stdc++.h>
using namespace std;
long long i, m, n, sum, a[100007], l = -1e9, r = 1e9, j;
string b;
int main() {
  cin >> n;
  for (i = 0; i < n; i++) cin >> a[i];
  cin >> b;
  for (i = 4; i < n; i++) {
    if (b[i] != b[i - 1]) {
      for (j = i - 4; j <= i; j++) {
        if (b[i] == '1')
          l = max(l, a[j] + 1);
        else
          r = min(r, a[j] - 1);
      }
    }
  }
  cout << l << " " << r;
}
