m = int(input())
d = {}
ans = []
for _ in range(m):
	s = input().split('/')
	s[0] = s[0].replace('(', '')
	s[0] = s[0].replace(')', '')
	(a, b) = s[0].split('+')
	c = int(s[-1])
	a = int(a)
	b = int(b)
	ans.append([a, b, c])
	x = '{0:.4f}'.format((a + b) / c)
	d[x] = d.get(x, 0) + 1
for (a, b, c) in ans:
	x = '{0:.4f}'.format((a + b) / c)
	print(d[x], end=' ')
