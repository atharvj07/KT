#include <bits/stdc++.h>
int main() {
  char ch[500][500], a;
  int n, i, j, sum = 0;
  scanf("%d", &n);
  a = getchar();
  for (i = 0; i < n; i++) {
    for (j = 0; j < n; j++) {
      scanf("%c", &ch[i][j]);
    }
    a = getchar();
  }
  for (i = 1; i < n - 1; i++) {
    for (j = 1; j < n - 1; j++) {
      if (ch[i][j] == 'X' && ch[i + 1][j + 1] == 'X' &&
          ch[i - 1][j - 1] == 'X' && ch[i - 1][j + 1] == 'X' &&
          ch[i + 1][j - 1] == 'X')
        sum++;
    }
  }
  printf("%d\n", sum);
  return 0;
}
