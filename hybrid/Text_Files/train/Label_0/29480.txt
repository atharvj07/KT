#include <iostream>
#include <string>
#include <algorithm>
#define int long long
#define rep(i, n) for(i = 0; i < n; i++)
using namespace std;

//0.～ * 2^m --> 1.～ * 2^m
void convert(int &sisu, int &kasu) {
	sisu--;
	kasu %= (1LL << 52);
}

void solve(int n, int a) {
	a += (1LL << 52);
	int sisu = 1;
	int kasu = a;
	int i;
	
	for (i = 0; i < n;) {
		if (a == 0) break;
		int m = (((1LL << 53) - kasu) + (a - 1)) / a;
		m = min(m, n - i);
		kasu += a * m;
		if (kasu >= (1LL << 53)) {
			kasu >>= 1;
			sisu++;
			a >>= 1;
		}
		i = i + m;
	}

	convert(sisu, kasu);
	rep(i, 12) cout << (sisu >> (12 - 1 - i)) % 2;
	rep(i, 52) cout << (kasu >> (52 - 1 - i)) % 2;
	cout << endl;
}

signed main() {
	int n;
	while (cin >> n) {
		if (!n) break;
		string s;
		cin >> s;
		int i;
		
		int a = 0;
		rep(i, 52) {
			a += (1LL << i) * (s[52 - 1 - i] - '0');
		}
		solve(n, a);
	}
	return 0;
}
