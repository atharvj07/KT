#include <bits/stdc++.h>
using namespace std;
long long mod = 1e9 + 7;
void fast() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
}
bool primecheck(long long n) {
  for (long long i = 2; i <= sqrt(n); i++) {
    if (n % i == 0) {
      return 0;
    }
  }
  return 1;
}
long long power(long long x, long long y, long long mod) {
  long long res = 1;
  while (y > 0) {
    if (y & 1) res = (res * x) % (mod);
    y = y >> 1;
    x = (x * x) % mod;
  }
  return res % mod;
}
int main() {
  fast();
  long long n, m, point;
  cin >> n >> m >> point;
  string str;
  cin >> str;
  set<long long> s;
  for (long long i = 1; i <= str.length(); i++) {
    s.insert(i);
  }
  string op;
  cin >> op;
  for (long long i = 0; i < m; i++) {
    if (op[i] == 'R') {
      point = *s.upper_bound(point);
    } else if (op[i] == 'L') {
      auto it = s.lower_bound(point);
      --it;
      point = *it;
    } else {
      stack<char> st;
      st.push(str[point - 1]);
      s.erase(point);
      if (st.top() == '(') {
        while (st.size()) {
          point = *s.upper_bound(point);
          if (str[point - 1] == '(') {
            st.push('(');
          } else {
            st.pop();
          }
          s.erase(point);
        }
        if (point > *s.rbegin()) {
          point = *s.rbegin();
        } else if (point < *s.begin()) {
          point = *s.begin();
        } else {
          point = *s.upper_bound(point);
        }
      } else {
        while (st.size()) {
          auto it = s.lower_bound(point);
          --it;
          point = *it;
          if (str[point - 1] == ')') {
            st.push(str[point - 1]);
          } else {
            st.pop();
          }
          s.erase(point);
        }
        if (point > *s.rbegin()) {
          point = *s.rbegin();
        } else if (point < *s.begin()) {
          point = *s.begin();
        } else {
          point = *s.upper_bound(point);
        }
      }
    }
  }
  for (auto x : s) {
    cout << str[x - 1];
  }
}
