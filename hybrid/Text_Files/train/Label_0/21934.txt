#include <bits/stdc++.h>
using namespace std;
mt19937 rng((int)std::chrono::steady_clock::now().time_since_epoch().count());
struct disjoint_set {
  int n;
  vector<long long> parent, size;
  disjoint_set(int N) : n(N), size(N), parent(N) {}
  void make_set(int v) {
    parent[v] = v;
    size[v] = 1;
  }
  int find_set(int v) {
    if (v == parent[v]) return v;
    return parent[v] = find_set(parent[v]);
  }
  void make_union(int a, int b) {
    a = find_set(a);
    b = find_set(b);
    if (b != a) {
      if (size[a] < size[b]) swap(a, b);
      parent[b] = a;
      size[a] += size[b];
      size[b] = 0;
    }
  }
};
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  disjoint_set dsu(n), dsu2(n);
  for (int i = 0; i < n; i++) dsu.make_set(i), dsu2.make_set(i);
  vector<vector<pair<int, int> > > g(n);
  for (int i = 0, u, v, w; i < n - 1; i++) {
    cin >> u >> v >> w, u--, v--;
    if (w == 1)
      dsu.make_union(u, v);
    else
      dsu2.make_union(u, v);
    g[u].push_back({v, w});
    g[v].push_back({u, w});
  }
  long long ans = 0;
  for (auto &x : dsu.size) ans += (x * (x - 1));
  for (auto &x : dsu2.size) ans += (x * (x - 1));
  for (int i = 0; i < n; i++) {
    if (dsu.size[dsu.find_set(i)] > 1 and dsu2.size[dsu2.find_set(i)] > 1) {
      ans +=
          (dsu2.size[dsu2.find_set(i)] - 1) * (dsu.size[dsu.find_set(i)] - 1);
    }
  }
  cout << ans << endl;
}
