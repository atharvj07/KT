import java.util.*;
import java.io.*;
import java.lang.*;
 
public class MainClass {
	
    private static BufferedReader br = null;
	private static StringTokenizer st = null;
	
	static {	
		br = new BufferedReader(new InputStreamReader(System.in));
	}
	
	private static String str() throws IOException {
		return br.readLine();
	}
	
	private static int ints() throws IOException {
		return Integer.parseInt(br.readLine());
	}
	
	private static int[] ints(int N) throws IOException {
		st = new StringTokenizer(br.readLine());
		int[] nums = new int[N];
		for(int i = 0; i < N; i += 1) {
			nums[i] = Integer.parseInt(st.nextToken());
		}
		return nums;
	}
	
	public static void main(String[] args) throws IOException {
	    long N = ints();
	    int[] epi = ints((int)(N - 1));
		Solution obj = new Solution();
	    System.out.println(obj.forgottenEpisode(epi, N));
		br.close();
    }
}
 
class Solution {
    
    public long sum(int[] nums) {
        long sum = 0;
        for(int num : nums) sum += num;
		return sum;
    }
    
	public long forgottenEpisode(int[] nums, long N) {
	   long sum_upto_N = (N * (N + 1)) / 2;
	   long sum = sum(nums);
	   return (sum_upto_N - sum);
	}
}
