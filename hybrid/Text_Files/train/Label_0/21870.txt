#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
struct Edge {
  int to, w, co;
  Edge(int to, int w, int co) {
    this->to = to;
    this->w = w;
    this->co = co;
  }
};
struct node {
  int w, cnt;
  long long val;
  node() {}
  node(int w, int cnt) {
    this->w = w;
    this->cnt = cnt;
    this->val = (long long)(w + 1) / 2 * cnt;
  }
  bool operator<(const node& rhs) const { return val < rhs.val; }
};
vector<Edge> vec[maxn];
int lef[maxn];
int n;
long long sum, s, ans;
long long pre[3][maxn * 20];
priority_queue<node> que[3];
void dfs(int u, int fa) {
  lef[u] = 0;
  for (auto& tmp : vec[u]) {
    int v = tmp.to;
    if (v == fa) continue;
    dfs(v, u);
    que[tmp.co].push(node(tmp.w, lef[v]));
    sum += (long long)tmp.w * lef[v];
    lef[u] += lef[v];
  }
  if (!lef[u]) lef[u] = 1;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int T;
  cin >> T;
  while (T--) {
    cin >> n >> s;
    for (int i = 1; i <= n; i++) vector<Edge>().swap(vec[i]);
    for (int i = 1; i < n; i++) {
      int u, v, w, co;
      cin >> u >> v >> w >> co;
      vec[u].emplace_back(Edge(v, w, co));
      vec[v].emplace_back(Edge(u, w, co));
    }
    sum = 0;
    dfs(1, 0);
    int dex1 = 0, dex2 = 0;
    while (!que[1].empty()) {
      node x = que[1].top();
      que[1].pop();
      ++dex1;
      pre[1][dex1] = pre[1][dex1 - 1] + x.val;
      x.w /= 2;
      x.val = (long long)(x.w + 1) / 2 * x.cnt;
      if (x.val) que[1].push(x);
    }
    while (!que[2].empty()) {
      node x = que[2].top();
      que[2].pop();
      ++dex2;
      pre[2][dex2] = pre[2][dex2 - 1] + x.val;
      x.w /= 2;
      x.val = (long long)(x.w + 1) / 2 * x.cnt;
      if (x.val) que[2].push(x);
    }
    ans = 1e18;
    for (int i = 0; i <= dex1; i++) {
      long long ned = sum - s - pre[1][i];
      if (ned <= 0) {
        ans = min(ans, (long long)i);
        break;
      }
      int dex = lower_bound(pre[2] + 1, pre[2] + 1 + dex2, ned) - pre[2];
      if (dex < dex2 + 1) ans = min(ans, (long long)i + dex * 2);
    }
    cout << ans << "\n";
  }
  return 0;
}
