#include <bits/stdc++.h>
using namespace std;
int c[100010];
int w[100010];
bool mark[100010];
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &c[i]);
  }
  for (int i = 1; i <= n; i++) {
    scanf("%d", &w[i]);
  }
  priority_queue<pair<long long, int>, vector<pair<long long, int>>,
                 greater<pair<long long, int>>>
      Q;
  long long t = m;
  int coin = m;
  long long cost = 0;
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    if (c[i] % 100 == 0) continue;
    coin -= c[i] % 100;
    Q.push(make_pair(1LL * w[i] * (100 - c[i] % 100), i));
    if (coin < 0) {
      ans += Q.top().first;
      mark[Q.top().second] = true;
      Q.pop();
      coin += 100;
    }
  }
  printf("%lld\n", ans);
  for (int i = 1; i <= n; i++) {
    if (mark[i] == true) {
      printf("%d %d\n", 1 + (c[i] / 100), 0);
    } else {
      printf("%d %d\n", c[i] / 100, c[i] % 100);
    }
  }
  return 0;
}
