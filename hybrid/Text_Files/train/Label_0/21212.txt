#define EPS 1e-9
#define rep(i,n) for(int i=0;i<n;i++)

double D,px,py,vx,vy;

int main(){
	while(cin>>D&&D){
		cin>>px>>py>>vx>>vy;
		if(fabs(px*vy-py*vx)>EPS)cout<<"impossible"<<endl;
		else{
			if(px*vx>0||py*vy>0){
				if(2-sqrt(px*px+py*py)>D)cout<<"impossible"<<endl;
				else printf("%.7f\n",2-sqrt(px*px+py*py));
			}
			else{
				if(sqrt(px*px+py*py)>D)cout<<"impossible"<<endl;
				else printf("%.7f\n",sqrt(px*px+py*py));
			}
		}
	}
}
