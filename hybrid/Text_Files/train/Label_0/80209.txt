def max_median(n, m, lst):
	(j, r) = (n // 2, n // 2 + 1)
	while m and r < n:
		if lst[j] == lst[r]:
			r += 1
		else:
			t = min(m, (lst[r] - lst[j]) * (r - j))
			lst[j] += t // (r - j)
			m -= t
	if lst[j] == lst[n - 1]:
		return lst[j] + m // (n - j)
	return lst[j] + min(m, (lst[r] - lst[j]) * (r - j)) // (r - j)
(N, K) = [int(i) for i in input().split()]
a = [int(x) for x in input().split()]
print(max_median(N, K, sorted(a)))
