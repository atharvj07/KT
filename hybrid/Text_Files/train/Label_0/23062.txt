#include <bits/stdc++.h>
using namespace std;
const int Nmax = 2e5 + 2, inf = 1e9 + 7, base = 1e9;
vector<int> g[Nmax];
int cnt[Nmax];
bool killed[Nmax];
void dfs(int v, int p = -1, bool go = true) {
  if (go)
    for (int w : g[v]) {
      if (w == p) continue;
      if (killed[w]) continue;
      dfs(w, v);
    }
  if (!(cnt[v] & 1)) {
    killed[v] = true;
    cout << v + 1 << '\n';
    for (int w : g[v]) {
      cnt[w]--;
      if (w != p && !killed[w]) dfs(w, v, false);
    }
  }
}
int main() {
  int n, hV;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> hV;
    if (hV != 0) {
      g[i].push_back(--hV);
      g[hV].push_back(i);
    }
  }
  for (int i = 0; i < n; ++i) {
    cnt[i] = g[i].size();
  }
  if (n & 1) {
    cout << "YES\n";
    dfs(0);
    for (int i = 0; i < n; ++i) {
      if (!killed[i]) cout << i + 1 << '\n';
    }
  } else
    cout << "NO\n";
  return 0;
}
