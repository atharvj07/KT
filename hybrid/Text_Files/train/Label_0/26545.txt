#include <bits/stdc++.h>
using namespace std;
long long int em(long long int x, long long int Mod, long long int n) {
  if (n == 0)
    return 1ll;
  else if (n % 2 == 0)
    return em((x % Mod * x % Mod) % Mod, Mod, n / 2) % Mod;
  else
    return (x % Mod * em((x % Mod * x % Mod) % Mod, Mod, (n - 1) / 2)) % Mod;
}
long long int arr[100009];
vector<long long int> v[100009];
long long int ans;
long long int gcd(long long int a, long long int b) {
  return (b ? gcd(b, a % b) : a);
}
void dfs(long long int x, long long int par,
         vector<pair<long long int, long long int>> temp) {
  vector<pair<long long int, long long int>> here;
  long long int sz = temp.size();
  long long int i, j;
  for (i = 0; i < sz; i++) {
    pair<long long int, long long int> z = temp[i];
    z.first = gcd(z.first, arr[x]);
    if (here.empty() || here.back().first != z.first)
      here.push_back(z);
    else
      here.back().second += z.second;
  }
  if (here.back().first != 0)
    here.push_back({0, 1});
  else
    here.back().second++;
  sz = here.size();
  for (i = 0; i < sz; i++) {
    pair<long long int, long long int> z = here[i];
    ans = ans + (z.second * z.first % 1000000007) % 1000000007;
    ans %= 1000000007;
  }
  for (long long int y : v[x]) {
    if (y == par) continue;
    dfs(y, x, here);
  }
}
int32_t main() {
  std::ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int n, i, j, k, l;
  cin >> n;
  for (long long int i = (long long int)1; i < (long long int)n + 1; i++)
    cin >> arr[i];
  for (long long int i = (long long int)0; i < (long long int)n - 1; i++) {
    cin >> j >> k;
    v[j].push_back(k);
    v[k].push_back(j);
  }
  ans = 0;
  vector<pair<long long int, long long int>> temp;
  temp.push_back({0ll, 1});
  dfs(1, -1, temp);
  cout << ans;
}
