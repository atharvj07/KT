#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int a[N], b[N], c[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) cin >> c[i];
    int p;
    p = a[0];
    cout << a[0] << " ";
    for (int i = 1; i < n - 1; i++) {
      if (a[i] != p) {
        p = a[i];
        cout << a[i];
      } else if (b[i] != p) {
        p = b[i];
        cout << b[i];
      } else if (c[i] != p) {
        p = c[i];
        cout << c[i];
      }
      cout << " ";
    }
    int k = a[0];
    if (a[n - 1] != p && a[n - 1] != k) {
      cout << a[n - 1];
    } else if (b[n - 1] != p && b[n - 1] != k) {
      cout << b[n - 1];
    } else if (c[n - 1] != p && c[n - 1] != k) {
      cout << c[n - 1];
    }
    cout << " ";
    cout << endl;
  }
}
