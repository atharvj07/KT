#include <bits/stdc++.h>
using namespace std;
long double PQ[100000];
int main() {
  iostream::sync_with_stdio(false);
  cin.tie(0);
  int N, p;
  cin >> N >> p;
  long double res = 0;
  for (int i = 0; i < N; i++) {
    int x, y;
    cin >> x >> y;
    int e = (x - 1) / p;
    int z = y / p;
    PQ[i] = (z - e) / (y - x + 1.0);
  }
  for (int i = 1; i < N; i++) res += 1 - (1 - PQ[i]) * (1 - PQ[i - 1]);
  res += 1 - (1 - PQ[0]) * (1 - PQ[N - 1]);
  cout << setprecision(15) << res * 2000 << endl;
}
