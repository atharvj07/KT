static class BIT_RangeQueryPointUpdate {
	int[]bit;
	int n;
	public BIT_RangeQueryPointUpdate(int[]a, int n) {
		this.n = n;
		this.bit = new int[n];
	}
	public BIT_RangeQueryPointUpdate(int n) {
		this.n = n;
		this.bit = new int[n];
	}
	public void init_tree(int[]a) {
		Arrays.fill(this.bit, n);
		for(int i = 0; i < n; i++) {
			this.update(i, a[i]);
		}
	}
	public int prefixMinQuery(int ind) {
		int answer = n;
		for(; ind > 0; ind &= (ind - 1)) {
			answer = Math.min(answer, bit[ind - 1]);
		}
		return answer;
	}
	public void update(int ind, int val) {
		ind++;
		for(; ind <= n; ind += ind & (-ind)) {
			bit[ind - 1] = Math.min(bit[ind - 1], val);
		}
	}
	
}

static void testcase() {
	int n = s.nextInt();
	
	// calculate mex for each subarray
	// and then mex of obtained numbers
	// ans - mex of mexes
	
	// iterate over all i and check if i can be the answer
	// for i to be the answer
	// it shouldn't be the mex of any subarray
	// for i to be the mex of some subarray
	// the set of the subarray should exactly be {1, 2, ..., i - 1}
	
	int[] a = new int[n], prev = new int[n], last = new int[n + 1];
	boolean[] exists = new boolean[n + 2];
	Arrays.fill(last, -1);
	
	// just dividing it into subarrays
	for(int i = 0; i < n; i++) {
		a[i] = s.nextInt() - 1;
		prev[i] = last[a[i]];
		last[a[i]] = i;
	}
	
	BIT_RangeQueryPointUpdate bit = new BIT_RangeQueryPointUpdate(n + 1);
	bit.init_tree(last);
	
	for(int i = 0; i <= n; i++) {
		if(last[i] != n - 1 && bit.prefixMinQuery(i) > last[i])
			exists[i] = true;
	}
	
	for(int i = n - 1; i >= 0; i--) {
		bit.update(a[i], prev[i]);
		if(prev[i] != i - 1 && bit.prefixMinQuery(a[i]) > prev[i]) 
			exists[a[i]] = true;
	}
	
	int answer = 0;
	while(exists[answer]) 
		++answer;
	
	out.println(answer + 1);
	
	
}

static void shuffleArray(int[] a) {
	Random random = new Random();
	for (int i = a.length-1; i > 0; i--) {
		int index = random.nextInt(i + 1);
		int tmp = a[index];
		a[index] = a[i];
		a[i] = tmp;
	}
}
