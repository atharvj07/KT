#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 22;
int n, m = 1000022, pr, d[22], p, sum;
int main() {
  ios_base::sync_with_stdio(false);
  cin >> pr;
  for (int i = 1; i <= 9; i++) {
    cin >> d[i];
    if (m >= d[i]) m = d[i];
  }
  if (pr < m) {
    cout << -1;
    return 0;
  }
  sum = pr / m;
  while (sum--)
    for (int i = 9; i >= 1; i--)
      if ((pr - d[i]) / m == sum && pr >= d[i]) {
        pr -= d[i];
        cout << i;
        break;
      }
  return 0;
}
