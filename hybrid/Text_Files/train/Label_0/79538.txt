for tc in range(int(input())):
	n = int(input())
	b = input()
	a = ['' for i in range(n)]
	d = ['0' for i in range(n)]
	if b[0] == '0':
		d[0] = '1'
	else:
		d[0] = '2'
	for i in range(1, n):
		if b[i] == '0':
			if d[i - 1] != '1':
				d[i] = '1'
			else:
				d[i] = '0'
		if b[i] == '1':
			if d[i - 1] == '2':
				d[i] = '1'
			else:
				d[i] = '2'
	for i in range(n):
		a[i] = str(eval(d[i] + '-' + b[i]))
	print(''.join(a))
