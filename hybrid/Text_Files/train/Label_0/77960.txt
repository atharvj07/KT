t = int(input())
for _ in range(t):
	s = list(str(input()))
	n = len(s)
	nxt = [-1] * n
	for i in range(n):
		if s[i] == '1':
			nxt[i] = i
		elif i != 0:
			nxt[i] = nxt[i - 1]
	res = 0
	for r in range(n):
		sum_ = 0
		for l in range(r, -1, -1):
			if r - l > 20:
				break
			if s[l] == '0':
				continue
			sum_ += 1 << r - l
			if l == 0:
				if sum_ <= r - -1:
					res += 1
			elif sum_ <= r - nxt[l - 1]:
				res += 1
	print(res)
