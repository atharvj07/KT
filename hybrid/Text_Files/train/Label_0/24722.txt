#include <bits/stdc++.h>
using namespace std;
long long int pow_mod(long long int a, long long int b) {
  long long int res = 1;
  while (b != 0) {
    if (b & 1) {
      res = (res * a) % 1000000007;
    }
    a = (a * a) % 1000000007;
    b /= 2;
  }
  return res;
}
bool check(long long int x) {
  long long int cnt_0 = 0;
  long long int cnt_1 = 0;
  long long int y = x;
  long long int cnt = 0;
  while (y) {
    cnt++;
    y /= 2;
  }
  for (long long int i = 0; i < cnt && ((x & (1LL << i)) == 0); i++) cnt_0++;
  for (long long int i = cnt - 1; i >= 0 && (x & (1LL << i)); i--) cnt_1++;
  if (cnt_0 + 1 == cnt_1 && (cnt_0 + cnt_1 == cnt)) return true;
  return false;
}
void solve() {
  long long int n;
  cin >> n;
  long long int mx = 1;
  for (long long int i = 1; i * i <= n; i++) {
    if (n % i == 0) {
      if (check(i)) mx = max(mx, i);
      if (check(n / i)) mx = max(mx, n / i);
    }
  }
  cout << mx << endl;
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int t;
  t = 1;
  while (t--) solve();
  return 0;
}
