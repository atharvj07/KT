#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c;
  char str[1010];
  while (cin >> n >> c) {
    int i, j, k;
    cin >> str;
    int l = strlen(str);
    if (c - 1 <= l - c) {
      for (i = 0; i < c - 1; i++) cout << "LEFT\n";
      for (i = 0; i < l - 1; i++) {
        cout << "PRINT " << str[i] << endl;
        cout << "RIGHT\n";
      }
      cout << "PRINT " << str[i] << endl;
    } else {
      for (i = 0; i < l - c; i++) cout << "RIGHT\n";
      for (i = l - 1; i > 0; i--) {
        cout << "PRINT " << str[i] << endl;
        cout << "LEFT\n";
      }
      cout << "PRINT " << str[i] << endl;
    }
  }
}
