#define REP(i,b,n) for(int i=b;i<n;i++)

double f(double x,double a,double b,double c,double d){
  return a*x*x*x+b*x*x+c*x+d;
}

double bsearchdec(double l,double r,double a,double b,double c,double d){
  double mid;
  rep(i,200){
    mid = (l+r)/2;
    if (f(mid,a,b,c,d) > 0)l=mid;
    else r=mid;
  }
  return mid;
}

double bsearchinc(double l,double r,double a,double b,double c,double d){
  double mid;
  rep(i,200){
    mid = (l+r)/2;
    if (f(mid,a,b,c,d) >0)r=mid;
    else l=mid;
  }
  return mid;
}

pair<double,double> qu(double a,double b,double c){
  double tmp = sqrt(b*b-4*a*c+0);
  double waru=2*a;
  return make_pair((-b-tmp)/waru,(-b+tmp)/waru);
}
