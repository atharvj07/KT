#include <bits/stdc++.h>
using namespace std;
int n, a[105], b[105];
string s;
char grid[2005][2005];
int main() {
  cin >> n >> s;
  stack<pair<int, int> > stk;
  int mx = 0;
  for (int i = 0; i <= 2000; i++)
    for (int j = 0; j <= 2000; j++) grid[i][j] = ' ';
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '[') {
      if (stk.empty())
        stk.push({i, 0});
      else
        stk.push({i, stk.top().second + 1});
      a[i] = stk.top().second;
    } else {
      a[i] = stk.top().second;
      stk.pop();
    }
    mx = max(mx, a[i]);
  }
  int idx = 0;
  for (int i = 0; i < n; i++) b[i] = ((mx - a[i]) << 1) + 1;
  for (int i = 0; i < n; i++) {
    if (s[i] == ']' && s[i - 1] == '[') idx += 3;
    if (s[i] == '[') {
      for (int j = 0; j < b[i] + 2; j++) {
        if (!j) {
          grid[j + a[i]][idx] = '+';
          grid[j + a[i]][idx + 1] = '-';
        } else if (j + 1 == b[i] + 2) {
          grid[j + a[i]][idx] = '+';
          grid[j + a[i]][idx + 1] = '-';
        } else
          grid[j + a[i]][idx] = '|';
      }
    } else {
      for (int j = 0; j < b[i] + 2; j++) {
        if (!j) {
          grid[j + a[i]][idx] = '+';
          grid[j + a[i]][idx - 1] = '-';
        } else if (j + 1 == b[i] + 2) {
          grid[j + a[i]][idx] = '+';
          grid[j + a[i]][idx - 1] = '-';
        } else
          grid[j + a[i]][idx] = '|';
      }
    }
    idx++;
  }
  for (int i = 0; i < (mx << 1) + 3; i++) {
    for (int j = 0; j < idx; j++) cout << grid[i][j];
    if (i + 1 != (mx << 1) + 3) puts("");
  }
  return 0;
}
