#include <iostream>
#include <algorithm>
#include <string>
#include <vector>
#include <cmath>
#include <map>
#include <queue>
#include <iomanip>
#include <set>
#include <tuple>
#define mkp make_pair
#define mkt make_tuple
#define rep(i,n) for(int i = 0; i < (n); ++i)
using namespace std;
typedef long long ll;
const ll MOD=1e9+7;

vector<ll> mods;
#define MOD_NUM 5

void setMods(){
  int cnt=0;
  ll val=MOD;
  while(cnt<MOD_NUM){
    bool ok=true;
    for(ll i=2;i*i<=val;i++){
      if(val%i==0){
        ok=false;
        break;
      }
    }

    if(ok){
      mods.push_back(val);
      cnt++;
    }

    val++;
  }
}

struct edge{
  int to;
  ll cost;
  int rev;
};

int N,M;
vector<int> A,B;
vector<ll> X;

vector<edge> g[100010];
pair<ll,ll> val[100010][MOD_NUM];
bool visited[100010];

void dfs(int now){
  for(int i=0;i<g[now].size();i++){
    edge &e=g[now][i];
    vector<pair<ll,ll>> nv(MOD_NUM,{0,0});
    for(int j=0;j<MOD_NUM;j++) nv[j]=val[now][j];

    if(e.rev){
      for(int j=0;j<MOD_NUM;j++) nv[j].second=nv[j].second*e.cost%mods[j];
    }else{
      for(int j=0;j<MOD_NUM;j++) nv[j].first=nv[j].first*e.cost%mods[j];
    }

    if(visited[e.to]){
      for(int j=0;j<MOD_NUM;j++){
        ll a=val[e.to][j].first*nv[j].second%mods[j];
        ll b=val[e.to][j].second*nv[j].first%mods[j];

        if(a!=b){
          cout<<"No"<<endl;
          exit(0);
        }
      }
    }else{
      visited[e.to]=1;
      for(int j=0;j<MOD_NUM;j++) val[e.to][j]=nv[j];
      dfs(e.to);
    }

  }
}

int main(){
  cin>>N>>M;
  A.resize(M);
  B.resize(M);
  X.resize(M);
  rep(i,M) cin>>A[i]>>B[i]>>X[i];

  setMods();

  rep(i,M){
    A[i]--;
    B[i]--;
    g[A[i]].push_back(edge{B[i],X[i],0});
    g[B[i]].push_back(edge{A[i],X[i],1});
  }

  for(int i=0;i<N;i++){
    if(visited[i]) continue;
    visited[i]=1;
    for(int j=0;j<MOD_NUM;j++) val[i][j]=mkp(1,1);
    dfs(i);
  }

  cout<<"Yes"<<endl;

  return 0;
}
