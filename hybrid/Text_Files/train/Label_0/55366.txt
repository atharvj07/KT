import java.io.*;
import java.util.*;

public class E {
    public static void main(String[] args) {
        try {
            new E().solve();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    int[] grundy = new int[1 << 20];
    int[][] seg = new int[25][];
    int find(int s, int d, int i0, int i1) {
        int len = grundy.length >> d;
        int e = s + len;
        if (e - 1 < i0 || i1 < s) {
            return 0;
        } else if (i0 <= s && e - 1 <= i1) {
            return seg[d][s / len];
        } else {
            return find(s, d + 1, i0, i1) | find((s + e) / 2, d + 1, i0, i1);
        }
    }
    void add(int idx) {
        int s = 0;
        int d = 0;
        int gm = 1 << grundy[idx];
        while (true) {
            int len = grundy.length >> d;
            if (seg[d] == null) {
                seg[d] = new int[grundy.length / len];
            }
            seg[d][s / len] |= gm;

            d++;
            len /= 2;
            if (len == 0) {
                break;
            }
            int e = s + len;
            if (e <= idx) {
                s = e;
                e = s + len;
            }
        }
    }
    void solve() throws IOException {
        Scanner sc = new Scanner(System.in);
        // BufferedReader in = new BufferedReader(new
        // InputStreamReader(System.in));
        // String[] sp;

        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }

        for (int i = 0; i < 4; i++) {
            grundy[i] = 0;
            add(i);
        }
        for (int i = 4; i < grundy.length; i++) {
            int i0 = (int) Math.ceil(Math.sqrt(Math.sqrt(i)));
            int i1 = (int) Math.floor(Math.sqrt(i));
            int mask = find(0, 0, i0, i1);

            for (int jj = 0;; jj++) {
                if ((mask >> jj & 1) == 0) {
                    grundy[i] = jj;
                    break;
                }
            }
            add(i);
        }

        int xor = 0;
        for (int i = 0; i < n; i++) {
            int g;
            if (a[i] < grundy.length) {
                g = grundy[(int) a[i]];
            } else {
                int i0 = (int) Math.ceil(Math.sqrt(Math.sqrt(a[i])));
                int i1 = (int) Math.floor(Math.sqrt(a[i]));
                int mask = find(0, 0, i0, i1);

                for (int jj = 0;; jj++) {
                    if ((mask >> jj & 1) == 0) {
                        g = jj;
                        break;
                    }
                }
            }
            xor ^= g;
        }

        if (xor == 0) {
            System.out.println("Rublo");
        } else {
            System.out.println("Furlo");
        }
    }
}
