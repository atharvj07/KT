def longestPrefixSuffix(s, n):
	lps = [0 for i in range(n)]
	l = 0
	i = 1
	while i < n:
		if s[i] == s[l]:
			l += 1
			lps[i] = l
			i += 1
		elif l != 0:
			l = lps[l - 1]
		else:
			lps[i] = 0
			i += 1
	res = lps[n - 1]
	return res
s1 = input()
s2 = input()
n1 = len(s1)
n2 = len(s2)
lcsp = longestPrefixSuffix(s2, n2)
no_1 = s1.count('1')
no_0 = n1 - no_1
re_1 = s2.count('1')
re_0 = n2 - re_1
if re_1 > no_1 or re_0 > no_0:
	print(s1)
else:
	ans = s2
	no_1 = no_1 - re_1
	no_0 = no_0 - re_0
	re_1 = re_1 - s2[:lcsp].count('1')
	re_0 = re_0 - s2[:lcsp].count('0')
	if re_1 != 0:
		_1 = no_1 // re_1
	else:
		_1 = float('inf')
	if re_0 != 0:
		_0 = no_0 // re_0
	else:
		_0 = float('inf')
	_min = min(_1, _0)
	ans = ans + s2[lcsp:] * _min
	no_1 = no_1 - _min * re_1
	no_0 = no_0 - _min * re_0
	ans = ans + '1' * no_1 + '0' * no_0
	print(ans)
