#include <bits/stdc++.h>
using namespace std;
int lst[2011][2011];
long long next_starts[2011];
long long strt_point[2011];
char str[3001];
int main() {
  gets(str);
  memset(next_starts, 0, sizeof(next_starts));
  memset(strt_point, 0, sizeof(strt_point));
  int len = strlen(str);
  register int i, j;
  for (i = 0; i < len; i++) {
    lst[i][strt_point[i]++] = i;
    for (j = i + 1; j < len; j++) {
      if (str[i] != str[j]) continue;
      int s, e;
      s = i;
      e = j;
      bool pal = true;
      while (s <= e) {
        if (str[s] != str[e]) {
          pal = false;
          break;
        }
        s++;
        e--;
      }
      if (pal) {
        lst[i][strt_point[i]++] = j;
      }
    }
  }
  next_starts[len - 1] = strt_point[len - 1];
  for (i = len - 2; i >= 0; i--)
    next_starts[i] = strt_point[i] + next_starts[i + 1];
  long long count = 0;
  for (i = 0; i < len; i++) {
    for (j = 0; j < strt_point[i]; j++) {
      int st = i;
      int en = lst[i][j];
      count += next_starts[en + 1];
    }
  }
  cout << count << endl;
  return 0;
}
