import java.util.*;
public class Codef{
    public static void main(String[] args){
        Scanner in=new Scanner(System.in);
        int n=in.nextInt(),m=in.nextInt(),ans=0;
        int[][] ar=new int[n][m*2];
        for (int i=0; i<n; i++)
            for (int j=0; j<m*2; j++)
                ar[i][j]=in.nextInt();
        for (int i=0; i<n; i++)
            for (int j=0; j<m; j++){
                if ((ar[i][j*2]==1)|(ar[i][j*2+1]==1)) ans++;
            }
        System.out.println(ans);
    }
}
