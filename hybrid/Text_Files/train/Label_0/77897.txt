from collections import defaultdict
(n, m) = map(int, input().split())
d = defaultdict(list)
a = []
for i in range(m):
	(x, y) = map(int, input().split())
	d[x].append(y)
	d[y].append(x)
for i in range(1, n + 1):
	p = len(d[i])
	a.append(p)
flag = 0
mid = 0
start = 0
for i in range(0, n):
	if a[i] == 2:
		mid += 1
	elif a[i] == 1:
		start += 1
if mid == n - 2 and start == 2:
	flag = 2
tag = 0
for i in range(n):
	if a[i] == 2:
		tag = 2
	else:
		tag = 1
		break
sac = 0
p = max(a)
t = a.index(p)
for i in range(n):
	if i != t:
		if a[i] == 1:
			sac = 2
		else:
			sac = 1
			break
	if i == t:
		if p == n - 1:
			sac = 2
		else:
			sac = 1
			break
if flag == 2:
	print('bus topology')
elif tag == 2:
	print('ring topology')
elif sac == 2:
	print('star topology')
else:
	print('unknown topology')
