#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
const long double pi = acos((long double)-1);
long long num_generator() { return rand() % 1000; }
void array_generator(long long a[], long long n) {
  for (long long i = 0; i < n; i++) a[i] = rand() % 100;
}
bool b[1001][1001];
long long check(vector<vector<long long>> a, long long n, long long m) {
  long long sum = 0;
  for (long long i = 0; i < n; i++) {
    for (long long j = 1; j < m; j++) {
      if (a[i][j] > a[i][j - 1])
        ;
      else
        return -1;
    }
  }
  for (long long j = 0; j < m; j++) {
    for (long long i = 1; i < n; i++) {
      if (a[i][j] > a[i - 1][j])
        ;
      else
        return -1;
    }
  }
  return 1;
}
long long modify(vector<vector<long long>> a, long long n, long long m) {
  for (long long i = n - 2; i >= 0; i--) {
    for (long long j = m - 2; j >= 0; j--) {
      if (b[i][j]) a[i][j] = min(a[i + 1][j], a[i][j + 1]) - 1;
    }
  }
  long long sum = 0;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < m; j++) sum += a[i][j];
  }
  return sum;
}
void solve() {
  long long n, m;
  cin >> n >> m;
  vector<vector<long long>> a(n, vector<long long>(m));
  for (long long i = 0; i < n; i++)
    for (long long j = 0; j < m; j++) cin >> a[i][j];
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < m; j++) {
      if (!a[i][j]) {
        b[i][j] = true;
        a[i][j] = max(a[i - 1][j], a[i][j - 1]) + 1;
      }
    }
  }
  if (check(a, n, m) == -1) {
    cout << "-1";
    return;
  }
  cout << modify(a, n, m);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  solve();
  return 0;
}
