#include <bits/stdc++.h>
int main() {
  int n;
  scanf("%d", &n);
  int N[n], x = 0, sum = 1, sum1 = 1;
  while (scanf("%d", N + x), ++x < n)
    ;
  for (x = 0; x < n - 1; x++) {
    if (N[x + 1] > N[x])
      sum1++;
    else {
      if (sum1 > sum) sum = sum1;
      sum1 = 1;
    }
  }
  if (sum1 > sum) sum = sum1;
  printf("%d", sum);
  return 0;
}
