s = list(map(int, input()))
ans = 9 * 6 + 1
for a in range(10):
	for b in range(10):
		for c in range(10):
			for d in range(10):
				for e in range(10):
					for f in range(10):
						if a + b + c == d + e + f:
							ans = min(ans, [a == s[0], b == s[1], c == s[2], d == s[3], e == s[4], f == s[5]].count(False))
print(ans)
