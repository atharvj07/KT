(a, b, c, d) = map(int, input().split())
s = 0
for z in range(c, d + 1):
	if b + c <= z:
		break
	u = min(b + b, a + c)
	v = max(b + b, a + c)
	if z >= v:
		s += (b + c - z) * (b + c - z + 1) // 2
	elif u <= z:
		s += (b + c - v) * (b + c - v + 1) // 2 + (min(c - b, b - a) + 1) * (v - z)
	elif z >= a + b:
		s += (c - b + 1) * (b - a + 1) - (z - a - b + 2) * (z - b - a + 1) // 2
	else:
		s += (c - b + 1) * (b - a + 1)
print(s)
