for _ in range(int(input())):
	n = int(input())
	s = input()
	l = [['=' for i in range(n)] for j in range(n)]
	for i in range(n):
		l[i][i] = 'X'
	arr = []
	for i in range(n):
		if s[i] == '2':
			arr.append(i)
	if 0 < len(arr) <= 2:
		print('NO')
	else:
		print('YES')
		len_arr = len(arr)
		for i in range(len_arr):
			l[arr[i]][arr[(i + 1) % len_arr]] = '+'
			l[arr[(i + 1) % len_arr]][arr[i]] = '-'
		for i in l:
			print(''.join(i))
