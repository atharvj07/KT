def kac(l, d):
	l.sort(reverse=True)
	ma = l[0][1]
	v = l[0][1]
	i = j = 0
	for i in range(1, len(l)):
		v = v + l[i][1]
		while abs(l[i][0] - l[j][0]) >= d:
			v -= l[j][1]
			j += 1
		if v > ma:
			ma = v
	print(ma)
	return
(n, d) = map(int, input().split())
l1 = []
for i in range(n):
	m = list(map(int, input().split()))
	l1.append(m)
kac(l1, d)
