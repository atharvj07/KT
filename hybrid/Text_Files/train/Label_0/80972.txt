import math
n = int(input())
p = [0] * n
for i in range(n):
	(r, h) = map(int, input().split())
	p[i] = r ** 2 * h
k = {v: j for (j, v) in enumerate(sorted(set(p)))}
d = [0] * (len(k) + 1)
for v in p:
	j = k[v]
	i = j + 1
	q = 0
	while j > 0:
		q = max(d[j], q)
		j -= j & -j
	q += v
	while i < len(d):
		d[i] = max(d[i], q)
		i += i & -i
print(max(d) * math.pi)
