#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return (a == 0 ? b : gcd(b % a, a)); }
int Gcd(int a, int b) {
  if (a > b) swap(a, b);
  return gcd(a, b);
}
const int SZ = 4e5;
int a[SZ];
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  int cur_gcd;
  cin >> a[0];
  cur_gcd = a[0];
  for (int i = 1; i < n; i++) {
    cin >> a[i];
    cur_gcd = Gcd(cur_gcd, a[i]);
  }
  int m = 1;
  for (int i = 0; i < n; i++) {
    a[i] /= cur_gcd;
    m = max(m, a[i]);
  }
  vector<bool> used;
  used.resize(m + 1);
  vector<int> check;
  check.resize(m + 1);
  for (int i = 0; i <= m; i++) {
    used[i] = false;
    check[i] = 0;
  }
  for (int i = 0; i < n; i++) {
    check[a[i]]++;
  }
  int ans = 0;
  int cur;
  for (int i = 2; i <= m; i++) {
    if (used[i]) continue;
    cur = check[i];
    for (int j = 2 * i; j <= m; j += i) {
      used[j] = true;
      cur += check[j];
    }
    ans = max(ans, cur);
  }
  if (ans == 0)
    cout << -1;
  else
    cout << n - ans;
  return 0;
}
