#include <bits/stdc++.h>
using namespace std;
int bfout(int x) {
  for (int(i) = (1); (i) < (48 + x + 1); i++) putchar('+');
  putchar('.');
  for (int(i) = (1); (i) < (48 + x + 1); i++) putchar('-');
  return 0;
}
int main() {
  int a;
  char c;
  int res;
  cin >> res;
  while (cin >> c) {
    int sign;
    sign = (c == '+') ? 1 : -1;
    cin >> a;
    res += (sign * a);
  }
  int outchar[4];
  int l = 0;
  do {
    int out = res % 10;
    outchar[l++] = out;
    res /= 10;
  } while (res);
  for (int i = l - 1; i >= 0; i--) bfout(outchar[i]);
  return 0;
}
