def main():
	n = int(input())
	string = str(input())
	moves = n - 11
	if moves % 2 != 0:
		petya = (moves - 1) // 2
	else:
		petya = moves // 2
	cnt = 0
	arr = []
	for i in range(n):
		if string[i] == '8':
			arr.append(i)
			cnt += 1
	if cnt <= petya:
		print('NO')
		return
	moves -= petya
	if arr[petya] - petya <= moves:
		print('YES')
	else:
		print('NO')
	return
main()
