mod = 10 ** 9 + 7
resp = 1
s = input()

def between(n, a, b):
	return n >= a and n <= b
for i in s:
	if between(i, '0', '9'):
		i = ord(i) - ord('0')
	elif between(i, 'A', 'Z'):
		i = ord(i) - ord('A') + 10
	elif between(i, 'a', 'z'):
		i = ord(i) - ord('a') + 36
	elif i == '-':
		i = 62
	else:
		i = 63
	for j in range(6):
		if i >> j & 1 == 0:
			resp = 3 * resp % mod
print(resp)
