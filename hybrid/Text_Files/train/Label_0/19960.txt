#include <bits/stdc++.h>
const int N = 1010;
const int M = 2010;
using namespace std;
int n;
int cnt = 0, flg = 0;
int son[N][3], rt;
long long add[N][N], a[N][N], wei[M];
int edge[M], lst[N], nxt[M], ind[N], t = 0;
void ADD(int x, int y, long long z) {
  edge[++t] = y;
  nxt[t] = lst[x];
  lst[x] = t;
  wei[t] = z;
  ind[y]++;
}
void READ() {
  int u, v;
  long long w;
  scanf("%d", &n);
  for (int i = 1; i <= n - 1; i++)
    scanf("%d%d%lld", &u, &v, &w), ADD(u, v, w), ADD(v, u, w);
  for (int i = 1; i <= n; i++)
    if (ind[i] == 1) rt = i;
}
void SEARCH(int x, int pa) {
  son[x][0] = son[x][1] = x;
  int now = 0;
  for (int r = lst[x]; r; r = nxt[r]) {
    if (edge[r] == pa) continue;
    SEARCH(edge[r], x);
    add[edge[r]][rt] += wei[r];
    if (x != rt) add[x][rt] -= wei[r];
    if (now <= 1) son[x][now] = son[edge[r]][0], now++;
  }
}
void UPD(int x, int y, long long z) {
  if (x > y) swap(x, y);
  a[x][y] += z;
}
void PRT(int x, int y, long long num) {
  if (ind[x] == 1 && ind[y] == 1) {
    UPD(x, y, num);
    return;
  }
  int la = son[x][1], lb = son[x][0];
  num /= 2;
  UPD(y, la, num);
  UPD(y, lb, num);
  UPD(la, lb, -num);
}
void SOLVE() {
  SEARCH(rt, 0);
  flg = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (add[i][j]) PRT(i, j, add[i][j]);
    }
  cnt = 0;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if (a[i][j]) cnt++;
  if (flg) return;
  printf("%d\n", cnt);
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      if (a[i][j]) printf("%d %d %lld\n", i, j, a[i][j]);
}
int main() {
  READ();
  for (int i = 1; i <= n; i++)
    if (ind[i] == 2) {
      printf("NO\n");
      return 0;
    }
  printf("YES\n");
  SOLVE();
  return 0;
}
