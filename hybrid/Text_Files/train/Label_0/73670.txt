N = int(input())
arr = list(map(int, input().split()))
out = []

def check(l):
	i = 1
	while i < len(l):
		if abs(l[i] - l[i - 1]) >= 2:
			return False
		i += 1
	return True
if not check(arr) or not check(out):
	print('NO')
	exit(0)
for _ in range(len(arr)):
	maxIndex = 0
	for i in range(len(arr)):
		if arr[maxIndex] < arr[i]:
			maxIndex = i
	out.append(arr[maxIndex])
	del arr[maxIndex]
	if not check(arr) or not check(out):
		print('NO')
		exit(0)
print('YES')
