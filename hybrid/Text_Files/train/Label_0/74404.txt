import sys
f = sys.stdin
n = int(f.readline())
if n % 2 == 0:
	print('NO')
else:
	a = list(range(1, 2 * n + 1))
	b = [0] * n
	c = [0] * n
	i = 0
	j = 0
	while i < 2 * n:
		if i % 2 == 0 and i < 2 * n:
			b[j] = a[i]
			i += 3
			j += 1
		else:
			b[j] = a[i]
			j += 1
			i += 1
	i = 1
	k = 0
	while i < 2 * n:
		if i % 2 == 0 and i < 2 * n:
			c[k] = a[i]
			i += 3
			k += 1
		else:
			c[k] = a[i]
			k += 1
			i += 1
	print('YES')
	for i in range(n):
		print(b[i], end=' ')
	for i in range(n):
		print(c[i], end=' ')
