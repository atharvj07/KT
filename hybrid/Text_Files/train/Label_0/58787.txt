import java.io.*;
import java.util.*;

public class D {
	
	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
//		Scanner seer = new Scanner(System.in);
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());
		
		int[] sieve = new int[n+1];
		for(int i = 2; i <= n; i++){
			if(sieve[i] == 0){
				sieve[i] = i;
				if((long)i*(long)i > (long)n) continue;
				for(int j = i*i; j <= n; j+=i){
					sieve[j] = i;
				}
			}
		}
		ArrayList<Integer>[] ps = new ArrayList[n];
		for(int i = 0; i < n; i++){
			ps[i] = new ArrayList<>();
		}
		for(int i = 0; i < m; i++){
			st = new StringTokenizer(br.readLine());
			int a = Integer.parseInt(st.nextToken())-1;
			int b = Integer.parseInt(st.nextToken())-1;
			ps[a].add(b); ps[b].add(a);
		}
		for(int i = 0; i < n; i++){
			Collections.sort(ps[i]);
		}
		HashSet<Integer> divs = new HashSet<>();
		if(sieve[n] == n){
			divs.add(n);
		}
		else{
			int aux = n;
			while(aux > 1){
				divs.add(sieve[aux]);
				aux /= sieve[aux];
			}
		}
		boolean fits = false;
		for(int div: divs){
			int k = n/div;
			fits = true;
			for(int i = 0; i < n; i++){
				int s = (i+k)%n;
				int x = Integer.min(i, s);
				int y = Integer.max(i, s);
				if(ps[x].size() != ps[y].size()){
					fits = false; break;
				}
				int S = ps[i].size();
				if(S == 0) continue;
				int jy = 0;
				while(jy < S && (ps[y].get(jy) - ps[x].get(0) - (y-x) + n) % n != 0){
					jy++;
				}
				if(jy >= S){
					fits = false; break;
				}
				for(int j = 0; j < S; j++, jy++){
					if((ps[y].get(jy % S) - ps[x].get(j) - (y-x) + n) % n != 0){
						fits = false; break;
					}
				}
				if(!fits) break;
			}
			if(fits) break;
		}
		if(fits) System.out.println("Yes");
		else System.out.println("No");
	}
