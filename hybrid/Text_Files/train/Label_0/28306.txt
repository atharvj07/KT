#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 6;
const long long inf = 0x3f3f3f3f;
const int mod = 1e9 + 9;
long long qp(long long x, long long y) {
  long long ans = 1;
  x %= mod;
  while (y) {
    if (y & 1) ans = ans * x % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ans;
}
int _;
long long n, m;
int main() {
  scanf("%d", &_);
  while (_--) {
    scanf("%d%d", &n, &m);
    if (n % m == 0)
      printf("0\n");
    else {
      long long ans = (n / m + 1) * m;
      printf("%lld\n", ans - n);
    }
  }
}
