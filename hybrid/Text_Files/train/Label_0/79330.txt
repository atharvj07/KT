from sys import stdin, stdout
nmbr = lambda : int(stdin.readline())
lst = lambda : list(map(int, stdin.readline().split()))
for i in range(1):
	n = nmbr()
	sma = 0
	smb = 0
	ans = []
	for i in range(n):
		(a, b) = lst()
		if abs(a + sma - smb) <= 500:
			sma += a
			ans += ['A']
		else:
			smb += b
			ans += ['G']
	stdout.write(''.join(ans))
