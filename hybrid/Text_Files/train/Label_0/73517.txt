(n, m) = map(int, input().split())
ans = [0] * n
f = 0
li = []
for i in range(m):
	(a, b, c) = map(int, input().split())
	li.append((b, a, c, i + 1))
li.sort()
for i in range(m):
	(b, a, c, k) = li[i]
	ans[b - 1] = m + 1
	j = a - 1
	cnt = 0
	while j < b and cnt < c:
		if ans[j] == 0:
			ans[j] = k
			cnt += 1
		j += 1
	if cnt != c:
		f = 1
if f:
	print(-1)
else:
	print(*ans)
