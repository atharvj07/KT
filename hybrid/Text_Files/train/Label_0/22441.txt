#include <bits/stdc++.h>
using namespace std;
inline long long read() {
  long long x = 0, f = 1;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') f = 0;
    c = getchar();
  }
  while (c >= '0' && c <= '9')
    x = (x << 3) + (x << 1) + (c ^ 48), c = getchar();
  return f ? x : -x;
}
const long long N = 4e5 + 5, mod = 1e9 + 7;
struct node {
  long long x, y, id;
} a[N], b[N];
bool cmp1(node a, node b) { return a.x < b.x; }
bool cmp2(node a, node b) { return a.y < b.y; }
long long tot, ver[N], nxt[N], head[N];
void add(long long u, long long v) {
  ver[++tot] = v;
  nxt[tot] = head[u];
  head[u] = tot;
}
long long n, ans = 1, vis[N];
set<long long> x, y;
long long dfs(long long u, long long fa) {
  bool cyc = 0;
  x.insert(b[u].x);
  y.insert(b[u].y);
  vis[u] = 1;
  for (long long i = head[u]; i; i = nxt[i]) {
    long long v = ver[i];
    if (v == fa) continue;
    if (vis[v]) {
      cyc = 1;
      continue;
    }
    if (dfs(v, u)) cyc = 1;
  }
  return cyc;
}
long long qpow(long long a, long long x) {
  long long res = 1;
  while (x) {
    if (x & 1) res = res * a % mod;
    a = a * a % mod;
    x >>= 1;
  }
  return res;
}
signed main() {
  n = read();
  for (long long i = 1; i <= n; i++) {
    a[i].x = read(), a[i].y = read();
    a[i].id = i;
    b[i] = a[i];
  }
  sort(a + 1, a + n + 1, cmp1);
  for (long long i = 1; i < n; i++) {
    if (a[i].x == a[i + 1].x) {
      add(a[i].id, a[i + 1].id);
      add(a[i + 1].id, a[i].id);
    }
  }
  sort(a + 1, a + n + 1, cmp2);
  for (long long i = 1; i < n; i++) {
    if (a[i].y == a[i + 1].y) {
      add(a[i].id, a[i + 1].id);
      add(a[i + 1].id, a[i].id);
    }
  }
  for (long long i = 1; i <= n; i++) {
    if (!vis[i]) {
      x.clear();
      y.clear();
      long long cyc = dfs(i, -1);
      long long siz = x.size() + y.size();
      if (cyc)
        ans = ans * qpow(2, siz) % mod;
      else
        ans = ans * (qpow(2, siz) - 1) % mod;
    }
  }
  printf("%lld", ans);
  return 0;
}
