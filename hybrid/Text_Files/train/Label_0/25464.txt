#include <bits/stdc++.h>
using namespace std;
char c[400];
int main() {
  string s;
  int n;
  cin >> n >> n;
  cin >> s;
  for (int i = 'a'; i <= 'z'; i++) c[i] = i;
  for (int i = 0; i < n; i++) {
    char a, b;
    cin >> a >> b;
    if (a == b) continue;
    for (int i = 'a'; i <= 'z'; i++) {
      if (c[i] == a)
        c[i] = b;
      else if (c[i] == b)
        c[i] = a;
    }
  }
  for (int i = 0; i < s.size(); i++) {
    cout << c[s[i]];
  }
  return 0;
}
