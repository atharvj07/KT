t = int(input())
for test in range(t):
	array = list(map(int, input().rstrip().split()))
	n = array[0]
	m = array[1]
	k = array[2]
	if n % 2 == 0 and m % 2 == 0:
		print('YES' if k % 2 == 0 else 'NO')
	elif n % 2 == 0:
		d = n * (m - 1) // 2
		if k > d:
			print('NO')
		elif k % 2 == 0:
			print('YES')
		else:
			print('NO')
	else:
		row = m // 2
		if k < row:
			print('NO')
		else:
			remain = k - row
			print('YES' if remain % 2 == 0 else 'NO')
