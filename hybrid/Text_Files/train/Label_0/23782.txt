#include <bits/stdc++.h>
using namespace std;
long long k, fib[66];
int n;
int main() {
  scanf("%d%I64d", &n, &k);
  fib[0] = 1LL;
  fib[1] = 1LL;
  for (int i = 2; i <= n; ++i) {
    fib[i] = fib[i - 2] + fib[i - 1];
  }
  int out = 1;
  for (int i = n; i > 0; --i) {
    if (k <= fib[i - 1]) {
      printf("%d ", out++);
    } else {
      printf("%d %d ", out + 1, out);
      out += 2;
      --i;
      k -= fib[i];
    }
  }
  return 0;
}
