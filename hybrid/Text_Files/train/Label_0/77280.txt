import collections
import sys
(n, q) = map(int, sys.stdin.readline().split(' '))
notifApp = collections.defaultdict(collections.deque)
notifs = collections.deque()
sol = []
total = 0
m = 0
ctd = 0
for i in range(q):
	(tipo, x) = map(int, sys.stdin.readline().split(' '))
	if tipo == 1:
		notifApp[x].append(ctd)
		notifs.append(x)
		total += 1
		ctd += 1
	elif tipo == 2:
		actual = notifApp[x]
		if len(actual) > 0:
			total -= len(actual)
			del notifApp[x]
	else:
		while x > m:
			primero = notifs.popleft()
			actual = notifApp[primero]
			if len(actual) > 0 and actual[0] < x:
				actual.popleft()
				total -= 1
				if len(actual) == 0:
					del notifApp[primero]
			m += 1
	sol.append(total)
sys.stdout.write('\n'.join(map(str, sol)))
