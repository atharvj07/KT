import math
t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	ar = list(map(int, input().split()))
	al = True
	ar = set(ar)
	ar = list(ar)
	n = len(ar)
	for x in range(n):
		if ar[x] != x:
			mex = x
			al = False
			break
	if al:
		print(n + k)
	elif math.ceil((mex + ar[n - 1]) / 2) in ar or k == 0:
		print(n)
	else:
		print(n + 1)
