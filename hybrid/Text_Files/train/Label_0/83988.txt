for _ in range(int(input())):
	(n, p, k) = map(int, input().strip().split())
	a = list(map(int, input().strip().split()))
	a.sort()
	dp = [0] * n
	ans = 0
	for i in range(k - 1):
		dp[i] = a[i] + dp[i - 1]
		if dp[i] <= p:
			ans = 1 + i
	for i in range(k - 1, n):
		dp[i] = a[i] + dp[i - k]
		if dp[i] <= p:
			ans = i + 1
	print(ans)
