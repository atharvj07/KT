#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
   ll i, n, t,j,k,p,m;
   cin>>t;
    while(t--)
    {
     cin>>n>>k;
     set<ll>st;
     for(i=0; i<n; i++)
     {
         cin>>p;
         st.insert(p);
     }
    // cout<<*st.begin();


     //cout<<*st.rbegin();
     if(k==0)
     {
         cout<<n<<endl;
     }
    else if(*st.rbegin()==n-1)
    {
        cout<<n+k<<endl;
    }
    else
    {
     for(i=0; ; i++)
     {
         if(st.find(i)==st.end())
         {
             m=i;
             break;
         }
         //cout<<*st.begin()+i<<" ";
     }
     //cout<<endl;
     ll x=ceil((*st.rbegin()+m)/2.0);
     st.insert(x);

    if(st.size()==n)
    {
        cout<<n<<endl;
    }
    else
        cout<<n+1<<endl;
    }

    }
}
