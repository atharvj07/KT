def read_int():
	return int(input())

def read_ints():
	return map(int, input().split(' '))

def solve(s, n):
	if n == 1:
		return [s]
	else:
		ss = str(s)
		l = len(ss)
		cc = 10 ** (l - 1)
		while s - cc < n - 1:
			cc //= 10
		return solve(s - cc, n - 1) + [cc]
t = read_int()
for case_num in range(t):
	(s, n) = read_ints()
	ans = solve(s, n)
	print(' '.join(map(str, ans)))
