#include <iostream>
#include <string.h>
 
using namespace std;
bool Flag[100002];
 
void solve()
{
    int n;
    while(cin >> n, n)
    {
        memset(Flag, 0, sizeof(Flag));
		int last_input = 0;
        for(int i = 0; i < n; ++i)
        {
            int input;
            cin >> input;
			last_input = input;
            Flag[input] = true;
        }
        int first_flag = true;
        for(int i = 1; i <= last_input; ++i)
        {
            if(Flag[i])
            {
                if(!first_flag)
                {
                    cout << " ";
				}
				first_flag = false;
                int pos = i + 1;
                while(Flag[pos] == true)
                {
                    ++pos;
                }
                if(pos == i + 1)
                {
                    cout << i;
                }
                else
                {
                    cout << i << "-" << pos - 1;
                }
                i = pos;
            }   
        }
        cout << endl;
    }
}
 
int main()
{
    solve();
    return(0);
}
