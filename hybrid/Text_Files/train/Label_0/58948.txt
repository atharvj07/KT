import java.util.*;

public class p302A {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);

		int n = in.nextInt(), k = in.nextInt();
		int pos = 0, neg = 0;
		for(int i=0; i<n; ++i)
			if(in.nextInt() == 1) ++pos;
			else ++neg;

		StringBuffer sb = new StringBuffer();
		for(int i=0; i<k; ++i) {
			int range = -1*(in.nextInt()-in.nextInt())+1;

			if(range%2 == 0 && (range/2) <= pos && (range/2) <= neg) sb.append("1\n");
			else sb.append("0\n");
		}

		System.out.println(sb.toString());
	}
}
