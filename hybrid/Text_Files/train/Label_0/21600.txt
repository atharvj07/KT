#include <bits/stdc++.h>
using namespace std;
int a[200001], l[200001], r[200001], ans[200001];
int main() {
  int n, x;
  while (cin >> n) {
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    for (int i = 0; i < n; i++) {
      x = i - 1;
      while (x >= 0 && a[x] >= a[i]) x = l[x];
      l[i] = x;
    }
    for (int i = n - 1; i >= 0; i--) {
      x = i + 1;
      while (x < n && a[x] >= a[i]) x = r[x];
      r[i] = x;
    }
    memset(ans, 0, sizeof(ans));
    for (int i = 0; i < n; i++) {
      x = r[i] - l[i] - 1;
      ans[x] = max(ans[x], a[i]);
    }
    for (int i = n; i >= 1; i--) ans[i] = max(ans[i], ans[i + 1]);
    for (int i = 1; i <= n; i++)
      if (i == 1)
        printf("%d", ans[i]);
      else
        printf(" %d", ans[i]);
    cout << endl;
  }
}
