import sys
MOD = 1000000007

def inp():
	return sys.stdin.readline().strip()

def ii():
	return int(inp())

def iis():
	return [int(i) for i in inp().split()]

def solve():
	n = ii()
	s = list(inp())
	t = sorted(s)
	cnt = 0
	for i in range(n):
		if s[i] != t[i]:
			cnt += 1
	return cnt
t = ii()
for _ in range(t):
	print(solve())
