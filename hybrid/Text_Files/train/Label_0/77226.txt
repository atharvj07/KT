n = int(input())
s = list(input())
f = 0
for i in range(n):
	if s[i].isupper():
		f = 1
		break
if f == 0:
	a = set(s)
	print(len(a))
	exit()
idx = []
for i in range(n):
	if s[i].isupper():
		idx.append(i)
ans = 0
for i in range(len(idx)):
	if i == 0:
		a = []
		for i in range(0, idx[i]):
			a.append(s[i])
		a = set(a)
		ans = max(ans, len(a))
	else:
		a = []
		for i in range(idx[i - 1] + 1, idx[i]):
			a.append(s[i])
		a = set(a)
		ans = max(ans, len(a))
if idx[len(idx) - 1] != n - 1:
	a = []
	for i in range(idx[len(idx) - 1] + 1, n):
		a.append(s[i])
	a = set(a)
	ans = max(ans, len(a))
print(ans)
