import sys
input = sys.stdin.buffer.readline
n = int(input())
a = list(map(int, input().split()))
MOD = 998244353
ans = 0
pow_ = 1
for i in range(n):
	if i == 0:
		ans += a[n - i - 1]
		ans %= MOD
	else:
		pow_ = pow_ * 2 * (i + 2) * pow(i + 1, MOD - 2, MOD)
		pow_ %= MOD
		ans += a[n - i - 1] * pow_
		ans %= MOD
print(ans)
