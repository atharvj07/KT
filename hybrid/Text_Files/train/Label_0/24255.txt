#include <bits/stdc++.h>
using namespace std;
int main() {
  int m, n, i, j, cnt1 = 0;
  cin >> m >> n;
  char p[m][n];
  for (i = 0; i < m; i++) {
    for (j = 0; j < n; j++) {
      cin >> p[i][j];
    }
  }
  for (i = 0; i < m; i++) {
    for (j = 0; j < n; j++) {
      if (p[i][j] == 'C' || p[i][j] == 'M' || p[i][j] == 'Y') cnt1 = 1;
    }
  }
  if (cnt1 > 0)
    cout << "#Color" << endl;
  else
    cout << "#Black&White" << endl;
  return 0;
}
