(n, b) = map(int, input().split())
li = list(map(int, input().split()))
even_count = 0
odd_count = 0
si = list()
for j in range(len(li)):
	if li[j] % 2 == 0:
		even_count = even_count + 1
	else:
		odd_count = odd_count + 1
	if even_count == odd_count:
		if j + 1 < n:
			diff = abs(li[j + 1] - li[j])
			si.append(diff)
			even_count = 0
			odd_count = 0
su = 0
count = 0
si.sort()
for kt in si:
	su = su + kt
	if su <= b:
		count = count + 1
print(count)
