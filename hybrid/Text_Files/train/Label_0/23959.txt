const ll MOD = 1000000007;

void ask(int n){
    cout<<"? "<<n<<endl;
}

void solve(){
    int n;cin>>n;
    vector<set<int>> g(n+1);
    ask(1);
    int a[n+1];
    int zero(0),one(0);
    rep(i,0,n){
        cin>>a[i+1];
        if (a[i+1]==1) g[1].insert(i+1);
        a[i+1]%=2;
        if (a[i+1]) one++;
        else zero++;
    }
    int ma=(min(zero,one)==zero)?0:1;
    if (ma==1) g[1].clear();
    //cout<<one<<" "<<zero<<" "<<ma<<endl;
    rep(i,2,n+1){
        if (a[i]==ma){
            ask(i);
            rep(j,0,n){
                int x;cin>>x;
                if (x==1){
                    g[i].insert(j+1);
                }
            }
        }
    }
    cout<<'!'<<endl;
    rep(i,1,n+1){
        trav(x,g[i]){
            cout<<i<<" "<<x<<endl;
        }
    }

}

int main(){
    int t(1);//cin>>t;
    while (t--){
        solve();
        cout<<endl;
    }
	return 0;
}
