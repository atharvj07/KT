#include <bits/stdc++.h>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    int* a = new int[n];
    std::map<int, std::pair<int, int>> occur;
    for (int i = 0; i != n; ++i) {
      std::cin >> a[i];
      if (occur.find(a[i]) == occur.end()) {
        occur[a[i]] = std::pair<int, int>(i, i + 1);
      } else {
        int prev = occur[a[i]].first;
        int& max_distance = occur[a[i]].second;
        occur[a[i]].first = i;
        int distance = i - prev;
        if (max_distance < distance) {
          max_distance = distance;
        }
      }
    }
    for (auto& pr : occur) {
      int& d = pr.second.second;
      int distance = n - pr.second.first;
      if (d < distance) {
        d = distance;
      }
    }
    int* p = new int[n + 1];
    for (int k = 1; k <= n; ++k) {
      p[k] = -1;
    }
    for (const auto& pr : occur) {
      int k0 = pr.second.second;
      if (p[k0] < 0) {
        p[k0] = pr.first;
      }
    }
    int val = -1;
    for (int k = 1; k <= n; ++k) {
      if (val < 0) {
        if (p[k] > 0) {
          val = p[k];
        }
      } else if (p[k] < 0 || p[k] > val) {
        p[k] = val;
      } else {
        val = p[k];
      }
    }
    for (int k = 1; k <= n; ++k) {
      std::cout << p[k] << (k == n ? '\n' : ' ');
    }
    delete[] a;
    delete[] p;
  }
}
