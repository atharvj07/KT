#include <bits/stdc++.h>
using namespace std;
const int N_MAX = 105;
int arr[N_MAX][N_MAX];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  int x = min(n, m), y = max(n, m);
  if (x * y == 1) {
    puts("1");
    return 0;
  } else if ((x == 1 && y <= 3) || (x == 2 && y == 2)) {
    puts("-1");
    return 0;
  } else if (x == 1 && y == 4) {
    arr[0][0] = 3, arr[0][1] = 1, arr[0][2] = 4, arr[0][3] = 2;
    if (n > m)
      for (int i = 0; i < n; i++) swap(arr[i][0], arr[0][i]);
  } else if (x == 2 && y == 3) {
    arr[0][0] = 2, arr[0][1] = 6, arr[0][2] = 3;
    arr[1][0] = 4, arr[1][1] = 1, arr[1][2] = 5;
    if (n > m)
      for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) swap(arr[i][j], arr[j][i]);
  } else {
    int pa = 1, pb = 1;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++)
        arr[i][j] = ((i + j) % 2) ? (pa++) + (n * m + 1) / 2 : (pb++);
    }
  }
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      printf("%d%c", arr[i][j], " \n"[j == m - 1]);
    }
  return 0;
}
