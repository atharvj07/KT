import java.util.*;
import java.io.*;

 * @author root
 */
public class D {
    static ArrayList<Integer> g[];
    static double [] exp;
    static int n;
    static int sub[];
    public static void main(String arsg[]){
        Scanner in =  new Scanner(System.in);
        n = in.nextInt();
        g = new ArrayList[n+1];
        exp = new double[n+1];
        sub = new int[n+1];
       // exp[1] =1.0;
        for(int i = 1;i<=n;i++){
            g[i] = new ArrayList<>();
        }
        int p[] = new int[n+1];
        for(int i = 2;i<=n;i++){
            p[i] = in.nextInt();
            g[p[i]].add(i);
        }
        countSub(1);
        dfs(1, 0, 0.0);
        StringBuilder build = new StringBuilder();
        for(int i = 1;i<exp.length;i++){
            build.append(exp[i]).append(" ");
        }
        System.out.println(build.toString());
    }
    
    public static int countSub(int v){
        int subVal = 0 ;
        for(int i = 0;i<g[v].size();i++){
            subVal+=countSub(g[v].get(i));
        }
        sub[v] = subVal+1;
        return sub[v];
    }
    
    public static void dfs(int v, int depth, double expV){
        exp[v] = ((double)(n - depth - sub[v]))/2.0 + depth + 1.0;
        for(int i = 0;i<g[v].size();i++){
            dfs(g[v].get(i), depth+1, exp[v]);
        }
    }
}
