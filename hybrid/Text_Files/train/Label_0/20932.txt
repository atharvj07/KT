#include <bits/stdc++.h>
using namespace std;
using i64 = int_fast64_t;
using ui64 = uint_fast64_t;
#define REP(i, n) for (i64 (i) = 0; (i) < (n); ++(i))
#define FOR(i, a, b) for (i64 (i) = (a); (i) < (b); ++(i))

ui64 MOP(ui64 x, ui64 n, ui64 m) {
    ui64 r = 1;
    while (n > 0) {
        if (n & 1) (r *= x) %= m;
        (x *= x) %= m;
        n >>= 1;
    }
    return r;
}
ui64 inverse(ui64 x, ui64 m) {
    return MOP(x, m - 2, m);
}

i64 N, M;
vector<int> MODS = {(int)1e9 + 7, (int)1e9 + 9, 998244353, 1000001773};
vector<vector<pair<i64, i64>>> graph;
vector<ui64> dp;
vector<bool> visited;

bool dfs(int cur, ui64 cst, int MOD) {
    visited[cur] = true;
    dp[cur] = cst;
    for (auto& nxt_p: graph[cur]) {
        int nxt = nxt_p.first;
        ui64 new_cst = cst;
        if (nxt_p.second > 0) (new_cst *= nxt_p.second) %= MOD;
        else (new_cst *= inverse(-nxt_p.second, MOD)) %= MOD;
        if (visited[nxt]) {
            if (dp[nxt] != new_cst) return false;
        } else {
            if (not dfs(nxt, new_cst, MOD)) return false;
        }
    }
    return true;
}

signed main() {

    cin >> N >> M;
    
    graph.resize(N);
    REP(i, M) {
        i64 A, B, X;
        cin >> A >> B >> X;
        --A, --B;
        graph[A].emplace_back(make_pair(B, X));
        graph[B].emplace_back(make_pair(A, -X));
    }

    for (auto& mod: MODS) {
        dp.assign(N, 0);
        visited.assign(N, false);
        REP(i, N) {
            if (visited[i]) continue;
            if (not dfs(i, 1, mod)) {
                cout << "No" << endl;
                return 0;
            }
        }
    }

    cout << "Yes" << endl;

}
