def no_intersections(numbers, colors):
	met_colors = {}
	pairs = []
	for (idx, color) in enumerate(numbers):
		if color in met_colors:
			pairs.append([color, met_colors[color], idx])
			colors.remove(color)
			met_colors = {}
		if color in colors:
			met_colors[color] = idx
	return pairs

def solve():
	(n, k) = (int(x) for x in input().split())
	numbers = [int(x) for x in input().split()]
	repeats = (n + k - 2) // (k - 1)
	pairs = []
	block_size = (n + repeats - 1) // repeats
	colors = list(range(1, n + 1))
	for x in range(repeats):
		pairs += no_intersections(numbers, set(colors[x * block_size:x * block_size + block_size]))
	pairs = sorted(pairs, key=lambda x: x[0])
	assert len(pairs) == n
	for pair in pairs:
		print(pair[1] + 1, pair[2] + 1)
solve()
