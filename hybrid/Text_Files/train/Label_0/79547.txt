def cf(c):
	if c > 0:
		return (0, c)
	else:
		return (1, -c)

def cd(v):
	d = [[] for c in range(8)]
	for c in v:
		for x in range(2):
			if c[x] == 0:
				(n, z) = cf(c[1 - x])
				d[2 * x + n].append((z, c[2]))
		for x in range(2):
			k = 2 * x - 1
			if c[x] == k * c[1 - x]:
				(n, z) = cf(c[x])
				d[4 + 2 * x + n].append((z, c[2]))
	for c in d:
		c.sort(key=lambda x: x[0])
	return d

def ic(v):
	f = (('R', 'Q'), ('B', 'Q'))
	d = cd(v)
	for (c, t) in enumerate(d):
		for x in t:
			if x[1] in f[c // 4]:
				return 1
			elif x[1] == f[1 - c // 4][0]:
				break
	return 0
n = int(input())
(kx, kz) = [int(c) for c in input().split()]
v = []
for c in range(n):
	t = input().split()
	(s, x, z) = (t[0], int(t[1]), int(t[2]))
	v.append((z - kz, x - kx, s))
s = 'NO'
if ic(v):
	s = 'YES'
print(s)
