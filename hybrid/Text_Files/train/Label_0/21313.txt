#include <bits/stdc++.h>
using namespace std;
int tr[1000001][27];
int s[1000001];
int tot;
string x;
inline void add(int p, int d, int len) {
  int xt = x[d] - 'a';
  if (d == len) xt = 26;
  if (d == len + 1) {
    s[p]++;
    return;
  }
  if (tr[p][xt] != 0) {
    add(tr[p][xt], d + 1, len);
    s[p]++;
  } else {
    tot++;
    tr[p][xt] = tot;
    add(tot, d + 1, len);
    s[p] += s[tr[p][xt]];
  }
}
int f[100001][201];
int ft[201];
int k;
inline void trdp(int d, int dep) {
  int i, j, p;
  bool flag = false, flag1 = false;
  int sx = 0;
  int fx = 1;
  if (dep == 0) fx = 0;
  for (i = 0; i <= 26; i++) {
    if (tr[d][i] != 0) {
      trdp(tr[d][i], dep + 1);
      sx += s[tr[d][i]];
      if (flag) {
        for (j = 1; j <= min(sx, k); j++) ft[j] = f[d][j];
        for (j = 1; j <= min(sx, k); j++)
          for (p = 0; p <= j; p++)
            f[d][j] = max(f[d][j], f[tr[d][i]][p] + ft[j - p]);
        flag1 = true;
      } else {
        int fx = 1;
        if (dep == 0) fx = 0;
        for (j = 1; j <= s[tr[d][i]]; j++) {
          f[d][j] = f[tr[d][i]][j];
        }
        flag = true;
      }
    }
  }
  if (flag)
    for (j = 1; j <= min(s[d], k); j++) f[d][j] += j * (j - 1) / 2 * fx;
}
int main() {
  int n;
  scanf("%d%d", &n, &k);
  int i;
  for (i = 1; i <= n; i++) {
    cin >> x;
    add(0, 0, x.size());
  }
  trdp(0, 0);
  printf("%d\n", f[0][k]);
  return 0;
}
