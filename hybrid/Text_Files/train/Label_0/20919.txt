#include <bits/stdc++.h>
using namespace std;
int dfs_low[100005], status[100005], dfscount, dfs_parent[100005], dfsroot,
    rootChildren;
set<int> edges[100005];
bool gagal;
map<pair<int, int>, int> edge;
void visit(int u) {
  pair<int, int> p;
  dfs_low[u] = status[u] = dfscount++;
  for (typeof((edges[u]).begin()) it = (edges[u]).begin();
       it != (edges[u]).end(); it++) {
    if (status[*it] == 0) {
      dfs_parent[*it] = u;
      if (u == dfsroot) rootChildren++;
      visit(*it);
      if (dfs_low[*it] > status[u]) gagal = true;
      dfs_low[u] = min(dfs_low[u], dfs_low[*it]);
      p.first = u;
      p.second = *it;
      edge[p] = 2;
      p.second = u;
      p.first = *it;
      edge[p] = 0;
    } else if (*it != dfs_parent[u]) {
      dfs_low[u] = min(dfs_low[u], status[*it]);
      p.first = u;
      p.second = *it;
      if (edge[p] == 1) {
        edge[p] = 2;
        p.second = u;
        p.first = *it;
        edge[p] = 0;
      }
    }
  }
  return;
}
int main() {
  int n, m, x, y;
  pair<int, int> p;
  cin >> n >> m;
  int hasil = 0;
  for (int i = (0); i < (m); ++i) {
    cin >> x >> y;
    p.first = x;
    p.second = y;
    edge[p] = 1;
    p.first = y;
    p.second = x;
    edge[p] = 1;
    edges[x].insert(y);
    edges[y].insert(x);
    if (x == 1 || y == 1) hasil++;
  }
  dfscount = 1;
  dfsroot = 1;
  rootChildren = 0;
  memset(status, 0, sizeof(status));
  gagal = false;
  visit(1);
  if (hasil < 2) gagal = true;
  if (gagal) {
    cout << "0\n";
    goto slese;
  }
  for (typeof((edge).begin()) it = (edge).begin(); it != (edge).end(); it++) {
    if ((it->second == 1 && it->first.first < it->first.second) ||
        (it->second == 2))
      cout << it->first.first << " " << it->first.second << "\n";
  }
slese:;
  return 0;
}
