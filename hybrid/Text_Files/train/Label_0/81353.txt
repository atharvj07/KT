import sys
n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
q = int(sys.stdin.readline())
qu = [tuple(map(int, sys.stdin.readline().split())) for _ in range(q)]
an = [-1] * n
m = 0
for i in range(q):
	query = qu[q - i - 1]
	if query[0] == 2:
		m = max(m, query[1])
	elif an[query[1] - 1] == -1:
		an[query[1] - 1] = max(query[2], m)
for i in range(n):
	if an[i] == -1:
		print(max(m, a[i]), end=' ')
	else:
		print(an[i], end=' ')
print()
