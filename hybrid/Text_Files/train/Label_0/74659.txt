(n, m, k) = map(int, input().split())
p = [(i, k // i) for i in range(1, int(k ** 0.5) + 1) if k % i == 0]
p += [(j, i) for (i, j) in p if i != j]
a = [x for x in input().split()]
b = [x for x in input().split()]
lst1 = []
lst2 = []
s = 0
ans = 0
for i in range(len(a)):
	if a[i] == '1':
		s += 1
		if i == len(a) - 1:
			lst1.append(s)
	else:
		lst1.append(s)
		s = 0
s = 0
for i in range(len(b)):
	if b[i] == '1':
		s += 1
		if i == len(b) - 1:
			lst2.append(s)
	else:
		lst2.append(s)
		s = 0
for (i, j) in p:
	x = 0
	y = 0
	for k in lst1:
		if i <= k:
			x += k - i + 1
	for k in lst2:
		if j <= k:
			y += k - j + 1
	ans += x * y
print(ans)
