t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	s = []
	d = {}
	for i in range(n):
		s.append(l[i])
		d[l[i]] = i
	s.sort()
	c = 1
	for i in range(0, n - 1):
		if d[s[i]] + 1 != d[s[i + 1]]:
			c += 1
	if c <= k:
		print('Yes')
	else:
		print('No')
