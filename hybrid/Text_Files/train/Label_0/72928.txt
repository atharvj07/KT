import math, string, itertools, fractions, heapq, collections, re, array, bisect, sys, random, time
sys.setrecursionlimit(5 * 10 ** 5)
inf = 10 ** 20
mod = 10 ** 9 + 7

def LI():
	return list(map(int, input().split()))

def II():
	return int(input())

def LS():
	return list(input().split())

def S():
	return input()

def solve():
	ans = 0
	(a, b, k) = LI()
	x_list = LI()
	y_list = LI()
	a_num = [0 for _ in range(a)]
	b_num = [0 for _ in range(b)]
	for x in x_list:
		x -= 1
		a_num[x] += 1
	for y in y_list:
		y -= 1
		b_num[y] += 1
	for i in range(k):
		x = x_list[i] - 1
		y = y_list[i] - 1
		ans += k + 1 - a_num[x] - b_num[y]
	ans //= 2
	return ans

def main():
	t = II()
	res = list()
	for _ in range(t):
		res.append(solve())
	for ans in res:
		print(ans)
	return 0
main()
