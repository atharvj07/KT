(n, m) = (int(x) for x in input().split())
noncovered = n * n
norow = {i for i in range(1, n + 1)}
nocol = {i for i in range(1, n + 1)}
answer = []
for i in range(m):
	(r, c) = (int(x) for x in input().split())
	if r in norow and c in nocol:
		noncovered -= len(norow) + len(nocol) - 1
		norow.remove(r)
		nocol.remove(c)
	elif r in norow:
		noncovered -= len(nocol)
		norow.remove(r)
	elif c in nocol:
		noncovered -= len(norow)
		nocol.remove(c)
	answer.append(str(noncovered))
print(' '.join(answer))
