#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, j, k, a, b, p = 1000000000, x, y, res = 0;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> a >> b;
    x = min(a, b);
    y = max(a, b);
    if (p >= y)
      p = y;
    else if (p >= x)
      p = x;
    else
      res = -1;
  }
  if (res == -1)
    cout << "NO";
  else
    cout << "YES";
  return 0;
}
