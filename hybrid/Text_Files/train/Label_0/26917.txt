#include <bits/stdc++.h>
using namespace std;
long long toInt(string &str, long long i) {
  if (i == -1) return 0;
  return (str[i] - '0') + toInt(str, i - 1) * 10;
}
pair<long long, pair<long long, long long>> getcab(string &str) {
  long long n = str.size();
  long long plus = find(str.begin(), str.end(), '+') - str.begin();
  long long secondparenthesis =
      find(str.begin() + 1, str.end(), ')') - str.begin();
  long long slash = find(str.begin(), str.end(), '/') - str.begin();
  string a = str.substr(1, plus - 1);
  string b = str.substr(plus + 1, secondparenthesis - plus - 1);
  string c = str.substr(slash + 1, n - slash - 1);
  long long p = 1;
  return {toInt(c, c.size() - 1),
          {toInt(a, a.size() - 1), toInt(b, b.size() - 1)}};
}
void solve() {
  long long n;
  cin >> n;
  map<double, long long> mp;
  vector<double> ar(n);
  for (long long i = 0; i < n; i++) {
    string str;
    cin >> str;
    pair<long long, pair<long long, long long>> cab = getcab(str);
    ar[i] =
        (cab.second.first * 1.0 + cab.second.second * 1.0) / cab.first * 1.0;
    mp[ar[i]]++;
  }
  for (auto x : ar) cout << mp[x] << " ";
  cout << '\n';
}
void testcase() {
  long long t;
  cin >> t;
  while (t--) solve();
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  solve();
}
