def cm(a, b, c):
	if a[1] == b[1] == c[1]:
		if a[0] == b[0] == c[0]:
			return True
		m = [int(a[0]), int(b[0]), int(c[0])]
		m.sort()
		if m[2] - m[1] == 1 and m[1] - m[0] == 1:
			return True
	return False
(a, b, c) = input().split()
ans = False
if cm(a, b, c):
	print(0)
	ans = True
elif not ans:
	p = []
	for i in range(10):
		for j in ['m', 'p', 's']:
			p.append(str(i) + j)
	for i in p:
		if cm(i, b, c):
			ans = True
			break
		if cm(i, a, b):
			ans = True
			break
		if cm(i, a, c):
			ans = True
	if ans:
		print(1)
if not ans:
	print(2)
