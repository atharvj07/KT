import sys
import math
import collections
import bisect

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_string():
	return sys.stdin.readline().strip()
for t in range(1):
	n = int(input())
	plus = dict()
	minus = dict()
	for i in range(n):
		(x, y) = get_ints()
		if x + y in plus:
			plus[x + y] += 1
		else:
			plus[x + y] = 1
		if x - y in minus:
			minus[x - y] += 1
		else:
			minus[x - y] = 1
	ans = 0
	for i in plus:
		if plus[i] > 1:
			ans += plus[i] * (plus[i] - 1) // 2
	for i in minus:
		if minus[i] > 1:
			ans += minus[i] * (minus[i] - 1) // 2
	print(ans)
