def main():
	(p, q, l, r) = map(int, input().split())
	Z = set()
	X = []
	for _ in range(p):
		(ai, bi) = map(int, input().split())
		Z.update(range(ai, bi + 1))
	for _ in range(q):
		(ai, bi) = map(int, input().split())
		X.extend(range(ai, bi + 1))
	res = 0
	for i in range(l, r + 1):
		for j in X:
			if j + i in Z:
				res += 1
				break
	print(res)
main()
