(w, h) = map(int, input().split())
a = []
checki = 0
chi = False
checkj = 0
dl = 0
for i in range(0, w):
	s = input()
	a.append(s)
	if 'B' in s:
		dl += 1
		if not chi:
			checki = i
			chi = True
for j in range(h):
	if a[checki][j] == 'B':
		checkj = j
		break
center1 = dl // 2 + checki + 1
center2 = dl // 2 + checkj + 1
print(center1, center2)
