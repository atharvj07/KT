def get_side_adjacents(matrix, curr_pos):
	m_length = len(matrix[0]) - 1
	m_height = len(matrix) - 1
	if curr_pos[0] == 0 and curr_pos[1] == 0:
		return [(0, 1), (1, 0)]
	elif curr_pos[0] == 0 and curr_pos[1] == m_length:
		return [(0, m_length - 1), (1, m_length)]
	elif curr_pos[0] == m_height and curr_pos[1] == 0:
		return [(m_height - 1, 0), (m_height, 1)]
	elif curr_pos[0] == m_height and curr_pos[1] == m_length:
		return [(m_height, m_length - 1), (m_height - 1, m_length)]
	elif curr_pos[1] == 0:
		return [(curr_pos[0] - 1, curr_pos[1]), (curr_pos[0] + 1, curr_pos[1]), (curr_pos[0], curr_pos[1] + 1)]
	elif curr_pos[1] == m_length:
		return [(curr_pos[0] + 1, curr_pos[1]), (curr_pos[0] - 1, curr_pos[1]), (curr_pos[0], curr_pos[1] - 1)]
	elif curr_pos[0] == 0:
		return [(curr_pos[0], curr_pos[1] - 1), (curr_pos[0], curr_pos[1] + 1), (curr_pos[0] + 1, curr_pos[1])]
	elif curr_pos[0] == m_height:
		return [(curr_pos[0], curr_pos[1] - 1), (curr_pos[0], curr_pos[1] + 1), (curr_pos[0] - 1, curr_pos[1])]
	else:
		return [(curr_pos[0], curr_pos[1] + 1), (curr_pos[0], curr_pos[1] - 1), (curr_pos[0] + 1, curr_pos[1]), (curr_pos[0] - 1, curr_pos[1])]

def get_light_state(matrix):
	for i in range(3):
		for j in range(3):
			if matrix[i][j] % 2:
				matrix[i][j] = 0
			else:
				matrix[i][j] = 1
matrix = []
for i in range(3):
	inp = input().split()
	inp = [int(x) for x in inp]
	matrix.append(inp)
matrix_copy = [x[:] for x in matrix]
for i in range(3):
	for j in range(3):
		curr_pos = (i, j)
		adjacents = get_side_adjacents(matrix, curr_pos)
		for adj in adjacents:
			matrix_copy[adj[0]][adj[1]] += matrix[i][j]
get_light_state(matrix_copy)
for i in range(3):
	curr_line = ''
	for j in range(3):
		curr_line += str(matrix_copy[i][j])
	print(curr_line)
