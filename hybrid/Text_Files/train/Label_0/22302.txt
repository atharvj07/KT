#include <bits/stdc++.h>
using namespace std;
using namespace std;
int a[1010];
int main() {
  int n, k;
  cin >> n >> k;
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    int x;
    scanf("%d", &x);
    ++a[x];
    if (a[x] == 2) {
      ans += 2;
      a[x] = 0;
    }
  }
  int all;
  if (n % 2)
    all = n / 2 + 1;
  else
    all = n / 2;
  ans += all - ans / 2;
  cout << ans << "\n";
}
