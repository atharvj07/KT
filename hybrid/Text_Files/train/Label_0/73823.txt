import sys
import math
import collections
import bisect

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_string():
	return sys.stdin.readline().strip()
for t in range(int(input())):
	(n, a, b) = get_ints()
	diff_gb = b - a
	maxim = 10 ** 10
	sol = []
	for i in range(1, diff_gb + 1):
		if diff_gb % i != 0:
			continue
		(x, y) = (a, b)
		ans = [x, y]
		diff = i
		while y > x + diff:
			if len(ans) == n:
				break
			y -= diff
			ans.append(y)
			if len(ans) == n:
				break
		while x > i:
			if len(ans) == n:
				break
			x -= i
			ans.append(x)
			if len(ans) == n:
				break
		y = b + diff
		while len(ans) < n:
			ans.append(y)
			y += diff
		common_diff = set()
		ans.sort()
		for j in range(1, n):
			common_diff.add(ans[j] - ans[j - 1])
		if max(ans) < maxim and len(common_diff) == 1:
			maxim = max(ans)
			sol = ans
	print(*sol)
