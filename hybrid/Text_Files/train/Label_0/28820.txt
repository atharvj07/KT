#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  int x, y, z, co, key = 1;
  int ash, mo;
  int l, f;
  cin >> n >> m >> k;
  cin >> mo;
  int r[k], r2[k];
  for (int q = 0; q < k; q++) {
    cin >> l >> f;
    r[q] = l - 1;
    r2[q] = f - 1;
  }
  for (int a = 0; a < mo; a++) {
    key = 1;
    co = 0;
    cin >> x >> y;
    x--;
    y--;
    z = x * m + y + 1;
    for (int i = 0; i < k; i++) {
      if (r[i] == x && r2[i] == y) {
        cout << "Waste\n";
        key = 0;
      } else {
        if (r[i] < x)
          co++;
        else if (r[i] == x && r2[i] < y)
          co++;
      }
    }
    if (key == 1) {
      z = z - co;
      if (z % 3 == 0)
        cout << "Grapes\n";
      else if (z % 3 == 1)
        cout << "Carrots\n";
      else
        cout << "Kiwis\n";
    }
  }
  return 0;
}
