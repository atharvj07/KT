int bits(int n) {
  int ans = 0;
  while (n) {
    ans += n % 2;
    n /= 2;
  }
  return ans;
}
long long bs(long long a[], long long low, long long high, long long val) {
  long long mid = (low + high) / 2;
  if (low > high) return -1;
  if (a[mid] > val)
    bs(a, low, mid - 1, val);
  else if (a[mid] < val)
    bs(a, mid + 1, high, val);
  else
    return mid;
}
int calc(int n) {
  if (n == 1) return 9;
  int ans = 9 * 4 * (pow(2, n) - 2) + 9 * (pow(2, n - 1) - 1) + 9;
  return ans;
}
void DJ() {
  int n, m, s, f, i, j;
  cin >> n >> m >> s >> f;
  map<int, pair<int, int>> steps;
  for (i = 1; i < m + 1; ++i) {
    int a, l, r;
    cin >> a >> l >> r;
    steps[a].first = l;
    steps[a].second = r;
  }
  if (s > f) {
    int k = 1;
    while (s != f) {
      if (steps.find(k) == steps.end()) {
        cout << 'L';
        s--;
        k++;
        continue;
      }
      if (s >= steps[k].first && s <= steps[k].second ||
          s - 1 >= steps[k].first && s - 1 <= steps[k].second) {
        cout << 'X';
      } else {
        cout << 'L';
        s--;
      }
      k++;
    }
  } else {
    int k = 1;
    while (s != f) {
      if (steps.find(k) == steps.end()) {
        cout << 'R';
        s++;
        k++;
        continue;
      }
      if (s >= steps[k].first && s <= steps[k].second ||
          s + 1 >= steps[k].first && s + 1 <= steps[k].second) {
        cout << 'X';
      } else {
        cout << 'R';
        s++;
      }
      k++;
    }
  }
}
