#include <bits/stdc++.h>
using namespace std;
int inf = 1000000;
int main() {
  int n;
  cin >> n;
  vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> v[i];
  }
  int mi = abs(v[n] - v[1]);
  int l = 1, r = n;
  int c;
  for (int i = 1; i < n; i++) {
    c = abs(v[i] - v[i + 1]);
    if (mi > c) {
      mi = c;
      l = i;
      r = i + 1;
    }
  }
  cout << l << ' ' << r << endl;
  return 0;
}
