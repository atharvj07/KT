#include <cstdio>
#include <cstdint>
#include <algorithm>

int main() {
  intmax_t W, H, w, h, x, y;
  scanf("%jd %jd %jd %jd %jd %jd", &W, &H, &w, &h, &x, &y);

  intmax_t xr = std::min(W/2, x+w/2);
  intmax_t xl = std::max(-W/2, x-w/2);
  intmax_t yr = std::min(H/2, y+h/2);
  intmax_t yl = std::max(-H/2, y-h/2);

  double xx = 0.5 * (xr+xl);
  double yy = 0.5 * (yr+yl);
  printf("%.12f\n", yy/xx);
}
