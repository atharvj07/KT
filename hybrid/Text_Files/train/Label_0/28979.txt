#include <bits/stdc++.h>
using namespace std;
template <class T>
void initvv(vector<vector<T> > &v, int a, int b, const T &t = T()) {
  v.assign(a, vector<T>(b, t));
}
template <class F, class T>
void convert(const F &f, T &t) {
  stringstream ss;
  ss << f;
  ss >> t;
}
int main() {
  int n;
  cin >> n;
  if (n % 2) {
    for (int i = 0; i < 3; ++i) {
      int x = (i < 2 ? 1 : 2);
      for (int j = 0; j < n; ++j) {
        if (j != 0) {
          putchar(' ');
        }
        printf("%d", j * x % n);
      }
      puts("");
    }
  } else {
    puts("-1");
  }
}
