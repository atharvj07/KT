import java.util.*;
import java.io.*;

public class Main {
	public static void main(String args[]) {new Main().run();}

	FastReader in = new FastReader();
	PrintWriter out = new PrintWriter(System.out);
	void run(){
		work();
		out.flush();
	}
	long mod=998244353;
	long gcd(long a,long b) {
		return b==0?a:gcd(b,a%b);
	}
	ArrayList<Long>[] dp;//记录子节点
	long[] dp2;//记录和父节点连接的情况数
	ArrayList<Integer>[] graph;
	ArrayList<Long>[] sum;
	void work() {
		int n=in.nextInt();
		dp2=new long[n];
		graph=(ArrayList<Integer>[])new ArrayList[n];
		dp=(ArrayList<Long>[])new ArrayList[n];
		sum=(ArrayList<Long>[])new ArrayList[n];
		for(int i=0;i<n;i++) {
			graph[i]=new ArrayList<>();
			dp[i]=new ArrayList<>();
			sum[i]=new ArrayList<>();
		} 
		
		for(int i=0;i<n-1;i++) {
			int s=in.nextInt()-1;
			int e=in.nextInt()-1;
			graph[s].add(e);
			graph[e].add(s);
		}
		dfs(0,new boolean[n],-1);
		out.println(sum[0].get(sum[0].size()-1));
	}
	private int dfs(int node, boolean[] vis,int pre) {
		vis[node]=true;
		int idx=-1;
		int size=graph[node].size();
		ArrayList<Integer> index=new ArrayList<>();
		for(int i=0;i<size;i++) {
			int nn=graph[node].get(i);
			if(nn==pre) {
				idx=i;
			}else {
				index.add(dfs(nn,vis,node));
			}
		}
		long cur=1;
		long[] rec=new long[index.size()];
		for(int i=size-1,j=rec.length-1;i>=0;i--) {
			int nn=graph[node].get(i);
			if(nn!=pre) {
				cur*=sum[nn].get(sum[nn].size()-1);
				cur%=mod;
				rec[j--]=cur;
			}
		}
		cur=1;
		for(int i=0,j=0;i<size;i++) {
			int nn=graph[node].get(i);
			if(nn!=pre) {
				int ind=index.get(j);
				long w=sum[nn].get(sum[nn].size()-1)-(ind==0?0:sum[nn].get(ind-1));
				long v=(((j+1==rec.length?1:rec[j+1])*cur)%mod*w)%mod;
				if(v<0)v+=mod;
				dp[node].add(v);
				j++;
				cur*=(ind==0?0:sum[nn].get(ind-1))+dp2[nn];
				cur%=mod;
			}else {
				long v=((i==rec.length?1:rec[i])*cur)%mod;
				dp2[node]=v;
			}
		}
		dp[node].add(cur);
		cur=0;
		for(long v:dp[node]) {
			cur+=v;
			cur%=mod;
			sum[node].add(cur);
		}
		return idx;
	}
}	



class FastReader
{
	BufferedReader br;
	StringTokenizer st;

	public FastReader()
	{
		br=new BufferedReader(new InputStreamReader(System.in));
	}

	public String next() 
	{
		if(st==null || !st.hasMoreElements())
		{
			try {
				st = new StringTokenizer(br.readLine());
			} catch (IOException e) {
				e.printStackTrace();
			}
		}
		return st.nextToken();
	}

	public int nextInt() 
	{
		return Integer.parseInt(next());
	}

	public long nextLong()
	{
		return Long.parseLong(next());
	}
}
