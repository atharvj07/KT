#include <bits/stdc++.h>
using namespace std;
int ch[1210][26], S[2][600], ch2[52];
int main() {
  int n, m, x;
  cin >> n >> m >> x;
  char c;
  int i, j, cntS = 0;
  for (i = 0; i < n; ++i)
    for (j = 0; j < m; ++j) {
      cin >> c;
      if (c == 'S') {
        S[0][cntS] = i;
        S[1][cntS] = j;
        cntS++;
      } else {
        ch[0][c - 97]++;
        ch[(2 * ch[0][c - 97]) - 1][c - 97] = i;
        ch[2 * ch[0][c - 97]][c - 97] = j;
      }
    }
  int q;
  char c2;
  cin >> q;
  for (i = 0; i < q; ++i) {
    cin >> c2;
    if (c2 >= 97)
      ++ch2[c2 - 97];
    else
      ++ch2[c2 - 39];
  }
  for (i = 0; i < 26; ++i)
    if (ch2[i] > 0 && ch[0][i] == 0) {
      cout << -1;
      return 0;
    }
  int cnt = 0;
  for (i = 0; i < 26; ++i) {
    float mn = INT_MAX;
    if (ch2[i + 26] > 0 && ch[0][i] > 0 && cntS > 0) {
      bool ok = false;
      for (j = 1; j < ch[0][i] * 2; j += 2) {
        for (int k = 0; k < cntS; ++k) {
          int r = (int)abs(ch[j][i] - S[0][k]),
              c = (int)abs(ch[j + 1][i] - S[1][k]);
          r = pow(r, 2);
          c = pow(c, 2);
          mn = min(mn, (float)sqrt(r + c));
          if (mn <= x) {
            ok = true;
            break;
          }
        }
        if (ok == true) break;
      }
      if (mn > x) cnt += ch2[i + 26];
    } else if (ch2[i + 26] > 0) {
      cout << -1;
      return 0;
    }
  }
  cout << cnt;
  return 0;
}
