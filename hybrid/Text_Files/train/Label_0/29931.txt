#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.);
const long long MOD = 1e9 + 7;
const int N = 1e5 + 7;
void kmp(char *s, int n, int *f) {
  f[0] = f[1] = 0;
  for (int i = int(1); i <= int(n - 1); ++i) {
    int j = f[i];
    while (j && s[i] != s[j]) j = f[j];
    f[i + 1] = j;
    if (s[i] == s[j]) f[i + 1]++;
  }
}
char s[N], t[N];
int dp[2][N], f[N], g[N][27];
int main() {
  scanf("%s", s);
  scanf("%s", t);
  int len1 = strlen(s), len2 = strlen(t);
  if (len2 > len1) {
    printf("0\n");
    return 0;
  }
  kmp(t, len2, f);
  for (int i = int(0); i <= int(len2 - 1); ++i) {
    for (int j = int(0); j <= int(25); ++j) {
      char c = (char)(j + 'a');
      int k = f[i];
      while (k && t[k] != c) k = f[k];
      if (t[k] == c) k++;
      g[i][j] = k;
    }
  }
  memset(dp, -1, sizeof(dp));
  dp[0][0] = 0;
  int ans = 0;
  for (int i = int(0); i <= int(len1 - 1); ++i) {
    int cur = i & 1, nx = cur ^ 1;
    for (int j = int(0); j <= int(len2 - 1); ++j) dp[nx][j] = -1;
    for (int j = int(0); j <= int(len2 - 1); ++j)
      if (dp[cur][j] != -1) {
        if (s[i] != '?') {
          if (s[i] == t[j]) {
            if (j == len2 - 1) {
              dp[nx][f[len2]] = max(dp[nx][f[len2]], dp[cur][j] + 1);
              ans = max(ans, dp[nx][f[len2]]);
            } else {
              dp[nx][j + 1] = max(dp[nx][j + 1], dp[cur][j]);
            }
          } else {
            int k = g[j][s[i] - 'a'];
            dp[nx][k] = max(dp[nx][k], dp[cur][j]);
          }
        } else {
          if (j == len2 - 1) {
            dp[nx][f[len2]] = max(dp[nx][f[len2]], dp[cur][j] + 1);
            ans = max(dp[nx][f[len2]], ans);
          } else {
            dp[nx][j + 1] = max(dp[nx][j + 1], dp[cur][j]);
          }
          for (int k = int(0); k <= int(25); ++k)
            dp[nx][g[j][k]] = max(dp[nx][g[j][k]], dp[cur][j]);
        }
      }
  }
  printf("%d\n", ans);
}
