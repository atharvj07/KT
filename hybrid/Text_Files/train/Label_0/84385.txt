n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
if n == 1:
	print(max(a[0], b[0]))
else:
	up_dp = []
	do_dp = []
	up_dp.append(a[0])
	up_dp.append(b[0] + a[1])
	do_dp.append(b[0])
	do_dp.append(a[0] + b[1])
	for i in range(2, n):
		do_dp.append(max(up_dp[i - 2] + b[i], up_dp[i - 1] + b[i]))
		up_dp.append(max(do_dp[i - 2] + a[i], do_dp[i - 1] + a[i]))
	print(max(up_dp[n - 1], do_dp[n - 1]))
