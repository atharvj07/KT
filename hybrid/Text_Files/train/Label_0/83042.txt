import math
(n, m) = [int(x) for x in input().split()]
a = [int(num) for num in input().split(' ', n - 1)]
extra = 0
ans = []
for i in range(0, n):
	counter = 0
	if a[i] >= extra and extra != 0:
		a[i] = a[i] - extra
		counter = 1 + counter
		extra = 0
	if a[i] < extra:
		extra = extra - a[i]
		a[i] = 0
	if a[i] >= m:
		counter = a[i] // m + counter
		extra = m - a[i] % m
	if a[i] < m and a[i] != 0:
		extra = m - a[i]
	ans.append(counter)
print(*ans)
