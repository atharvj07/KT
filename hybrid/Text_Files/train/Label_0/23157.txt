template <typename t>
t gcd(t a, t b) {
  return (!b) ? a : gcd(b, a % b);
}
template <typename t>
t pow(t a, t b) {
  if (b == 0) return 1;
  t cur = pow(a, b / 2);
  if (b % 2 == 0)
    return cur * cur;
  else
    return cur * cur * a;
}

double square(vector<pair<double, double>> nodes) {
  double area = 0;
  int numPoints = nodes.size();
  int j = numPoints - 1;
  for (int i = 0; i < numPoints; i++) {
    area = area + (nodes[j].first + nodes[i].first) *
                      (nodes[j].second - nodes[i].second);
    j = i;
  }
  return area / 2;
}
int calcMap(map<int, int> m) {
  int sum = 0;
  for (auto i : m) {
    sum += i.second;
  }
  return sum;
}
