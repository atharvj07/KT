(t, k) = map(int, input().split())
mod = 10 ** 9 + 7
arr = [0 for i in range(100001)]
arr[0] = 1
arr[1] = 1
if k == 1:
	arr[1] += 1
for i in range(2, len(arr)):
	if i >= k:
		arr[i] = (arr[i - 1] + arr[i - k]) % mod
	else:
		arr[i] = arr[i - 1] % mod
ans = [0 for i in range(100001)]
ans[1] = arr[1]
for i in range(2, len(arr)):
	ans[i] = (ans[i - 1] + arr[i]) % mod
for _ in range(t):
	(a, b) = map(int, input().split())
	print((ans[b] - ans[a - 1] + mod) % mod)
