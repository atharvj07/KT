#include <bits/stdc++.h>
using namespace std;

int main() {

#ifndef ONLINE_JUDGE
	freopen("input.txt", "r", stdin);
	freopen("output.txt", "w", stdout);
#endif

	int t;
	cin >> t;
	while (t--) {
		int a, b, c;
		cin >> a >> b >> c;
		int arr[5];
		for (int i = 0; i < 5; i++) {
			cin >> arr[i];
		}
		if (a >= arr[0] && b >= arr[1] && c >= arr[2]) {
			a -= arr[0];
			b -= arr[1];
			c -= arr[2];
			if (a >= arr[3]) arr[3] = 0;
			else arr[3] -= a;
			if (b >= arr[4]) arr[4] = 0;
			else arr[4] -= b;
			if (c >= arr[3] + arr[4]) {
				cout << "YES" << endl;
			}
			else {
				cout << "NO" << endl;
			}
		}
		else {
			cout << "NO" << endl;
		}


	}
}
