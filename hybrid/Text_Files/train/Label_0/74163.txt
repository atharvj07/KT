from collections import *
from sys import exit
from copy import deepcopy as copy
INF = 1000000000.0 + 7
t = int(input())
ans = [None for _ in range(t)]
for _ in range(t):
	(n, k) = map(int, input().split())
	s = input().strip()
	fst_pos = None
	good = INF
	prev = INF
	for (i, c) in enumerate(s):
		if c == '*':
			prev = i
			if fst_pos is None:
				fst_pos = i
				ans = 1
				good = i
		if i - good >= k:
			if prev != good:
				ans += 1
				good = prev
	if good != prev:
		ans += 1
	print(ans)
