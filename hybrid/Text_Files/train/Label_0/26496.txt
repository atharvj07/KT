#include <bits/stdc++.h>
using namespace std;
const int maxm = 2005;
const int INF = 100000000;
int a[maxm];
int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int main() {
  int n, i, j, k, sum, flag = 0, now;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    if (a[i] == 1) flag++;
  }
  if (flag > 0) {
    printf("%d\n", n - flag);
    return 0;
  }
  int ans = INF;
  for (i = 1; i <= n; i++) {
    now = a[i];
    for (j = i + 1; j <= n; j++) {
      now = gcd(now, a[j]);
      if (now == 1) {
        ans = min(ans, j - i);
        break;
      }
    }
  }
  if (ans == INF)
    printf("-1\n");
  else
    printf("%d\n", ans + (n - 1));
}
