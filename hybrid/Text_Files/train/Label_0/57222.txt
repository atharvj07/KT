import java.io.DataInputStream;
import java.io.PrintWriter;
import java.io.IOException;
import java.io.FileInputStream;

public class Main {

    private static final JIN in = new JIN();
    private static final PrintWriter out = new PrintWriter(System.out);

    // don't actually write comments when you do codeforces lmao
    public static void solve() throws IOException {

        // approach - (this problem is rated for D4 medium difficulty)
        // all of the substrings with no ones (00) can go at the beginning
        // the mixed strings (01 or 10) can go in the middle
        // the ones strings (11) can go at the end
        // if there is a

        // the n0, n1, n2 values
        int a = in.nextInt(), b = in.nextInt(), c = in.nextInt();

        // calculate the length
        int len = a+b+c+1;

        int[] arr = new int[len]; // to store the values, this isn't really necessary, you could just print them
        // it is just easier to use an array, in competition it is better to go the clear way than try to
        // micro-optimize

        arr[0] = a > 0 || b > 0 ? 0 : 1; // if there are any zeros in the string first char is 0 else it is 1

        int i = 1;
        while (a-- > 0) arr[i++] = 0; // set the rest of the zero pairs

        if (b > 0) {
            arr[i++] = 1;
            b--;
            int pairs = b/2; // the number of 01 pairs, there may need to be an extra 0 added at the end
            // this is because the last 01 pair increases the number of 11 pairs and in order to avoid
            // overshooting we need to add the last 01 pair as a 10 pair at the end of the array

            b -= (pairs*2);

            while (pairs-- > 0) { // add all the 01 pairs to the array
                arr[i++] = 0;
                arr[i++] = 1;
            }
        }
        // add all the ones
        while (c-- > 0) arr[i++] = 1;

        // set the last zero if necessary (as metioned above)
        if (b > 0) arr[i] = 0;

        // print the output to stdout
        for (int j = 0; j < len; j++) out.print(arr[j]);
        out.print('\n'); // don't forget the newline!
    }

    public static void main(String[] args) throws IOException {
        int t = in.nextInt();
        while (t-- > 0) solve();
        in.close();
        out.close();
    }

    static class JIN {
        final private int BUFFER_SIZE = 1 << 16;
        private DataInputStream din;
        private byte[] buffer;
        private int bufferPointer, bytesRead;

        public JIN() {
            din = new DataInputStream(System.in);
            buffer = new byte[BUFFER_SIZE];
            bufferPointer = bytesRead = 0;
        }

        public JIN(String file_name) throws IOException {
            din = new DataInputStream(new FileInputStream(file_name));
            buffer = new byte[BUFFER_SIZE];
            bufferPointer = bytesRead = 0;
        }

        public String readLine() throws IOException {
            byte[] buf = new byte[64]; // line length
            int cnt = 0, c;
            while ((c = read()) != -1) {
                if (c == '\n') break;
                buf[cnt++] = (byte) c;
            }
            return new String(buf, 0, cnt);
        }

        public int nextInt() throws IOException {
            int ret = 0;
            byte c = read();
            while (c <= ' ') c = read();
            boolean neg = (c == '-');
            if (neg) c = read();
            do {
                ret = ret * 10 + c - '0';
            }  while ((c = read()) >= '0' && c <= '9');

            return neg ? -ret : ret;
        }

        public long nextLong() throws IOException {
            long ret = 0;
            byte c = read();
            while (c <= ' ') c = read();
            boolean neg = (c == '-');
            if (neg) c = read();
            do {
                ret = ret * 10 + c - '0';
            }
            while ((c = read()) >= '0' && c <= '9');
            return neg ? -ret : ret;
        }

        public double nextDouble() throws IOException {
            double ret = 0, div = 1;
            byte c = read();
            while (c <= ' ') c = read();
            boolean neg = (c == '-');
            if (neg) c = read();
            do {
                ret = ret * 10 + c - '0';
            }
            while ((c = read()) >= '0' && c <= '9');
            if (c == '.') while ((c = read()) >= '0' && c <= '9') ret += (c - '0') / (div *= 10);
            return neg ? -ret : ret;
        }

        private void fillBuffer() throws IOException {
            bytesRead = din.read(buffer, bufferPointer = 0, BUFFER_SIZE);
            if (bytesRead == -1) buffer[0] = -1;
        }

        private byte read() throws IOException {
            if (bufferPointer == bytesRead) fillBuffer();
            return buffer[bufferPointer++];
        }

        public void close() throws IOException {
            if (din == null) return;
            din.close();
        }
    }
}
