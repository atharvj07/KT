def solve(n, arr):
	ans = m = sum(arr[:n:2])
	for i in range(0, n - 1, 2):
		m += arr[i + 1] - arr[i]
		ans = max(m, ans)
	m = arr[0] + sum(arr[1:n:2])
	ans = max(m, ans)
	for i in range(2, n - 1, 2):
		m += arr[i] - arr[i - 1]
		ans = max(m, ans)
	return ans
n = int(input())
arr = list(map(int, input().split()))
print(solve(n, arr))
