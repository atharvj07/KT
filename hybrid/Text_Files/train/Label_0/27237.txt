long long get(long long len, long long x, long long e) {
  memset(seen, false, sizeof(seen));
  ;
  long long w = 2 - x - e;
  for (long long i = 0; i < len; i++) {
    long long a, b;
    if (e == 0) {
      if (x == 0) {
        a = grundy[i][0][0];
        b = grundy[len - i - 1][0][0];
      } else if (x == 1) {
        a = grundy[i][0][0];
        b = grundy[len - i - 1][1][0];
      } else if (x == 2) {
        a = grundy[i][1][0];
        b = grundy[len - i - 1][1][0];
      }
    } else if (e == 1) {
      if (x == 0) {
        a = grundy[i][0][1];
        b = grundy[len - i - 1][0][0];
      } else if (x == 1) {
        a = grundy[i][0][1];
        b = grundy[len - i - 1][1][0];
      }
    } else if (e == 2) {
      if (x == 0) {
        a = grundy[i][0][1];
        b = grundy[len - i - 1][0][1];
      }
    }
    long long g = a ^ b;
    seen[g] = true;
  }
  if (e == 0) {
    if (x == 0) {
      for (long long i = 1; i < len - 1; i++) {
        long long a = grundy[((0) > (i - 1) ? (0) : (i - 1))][1][0];
        long long b = grundy[((0) > (len - i - 2) ? (0) : (len - i - 2))][1][0];
        seen[a ^ b] = true;
      }
    } else if (x == 1) {
      for (long long i = 1; i <= len; i++) {
        long long a = grundy[((0) > (i - 1) ? (0) : (i - 1))][1][0];
        long long b =
            i == len - 1
                ? 1
                : grundy[((0) > (len - i - 2) ? (0) : (len - i - 2))][2][0];
        seen[a ^ b] = true;
      }
    } else if (x == 2) {
      for (long long i = -1; i <= len; i++) {
        long long a =
            i == 0 ? 1 : grundy[((0) > (i - 1) ? (0) : (i - 1))][2][0];
        long long b =
            i == len - 1
                ? 1
                : grundy[((0) > (len - i - 2) ? (0) : (len - i - 2))][2][0];
        seen[a ^ b] = true;
      }
    }
  } else if (e == 1) {
    if (x == 0) {
      for (long long i = 0; i < len - 1; i++) {
        long long a =
            i == 1 ? 1 : grundy[((0) > (i - 1) ? (0) : (i - 1))][1][1];
        long long b = grundy[((0) > (len - i - 2) ? (0) : (len - i - 2))][1][0];

int32_t main() {
  long long m, n;
  cin >> m >> n;
  memset(grid, 0, sizeof(grid));
  ;
  for (long long i = 0; i < n; i++) {
    long long r, c;
    scanf("%I64d", &r);
    ;
    scanf("%I64d", &c);
    ;
    r--;
    c--;
    grid[r][c] = 2;
    if (r > 0) grid[r - 1][1 - c] = 1;
    if (r < m) grid[r + 1][1 - c] = 1;
    grid[r][1 - c] = 1;
  }
  memset(grundy, 0, sizeof(grundy));
  ;
  for (long long len = 1; len <= m; len++) {
    for (long long x = 0; x < 3; x++) {
      for (long long e = 0; e < 3; e++) {
        if (e + x <= 2) {
          grundy[len][x][e] = get(len, x, e);
        }
      }
    }
  }
  long long res = 0;
  long long i = 0;
  while (i < m) {
    if (grid[i][0] != 0) {
      i++;
      continue;
    }
    long long j = i + 1;
    while (j < m and grid[j][0] == 0) j++;
    long long x = 0, e = 0;
    if (i == 0)
      e++;
    else if (grid[i - 1][0] == 1)
      x++;
    if (j == m)
      e++;
    else if (grid[j][0] == 1)
      x++;
    res ^= grundy[j - i][x][e];
    i = j;
  }
  for (long long i = 1; i < m - 1; i++) {
    if (grid[i - 1][1] == 2 and grid[i][1] == 0 and grid[i + 1][1] == 2) {
      res ^= 1;
    }
  }
  if (res == 0)
    cout << "LOSE\n";
  else
    cout << "WIN\n";
  return 0;
}
