#include<iostream>
#include<cmath>
#include<cstdio>
using namespace std;

double area(double a, double l) {
  return a * sqrt(l * l - a * a / 4.0) / 2.0;
}

int main() {
  int a, l, x;
  while (cin >> a >> l >> x) {
    double ans = area(a, l) + 2 * area(l, (l + x) / 2.0);

    printf("%.10f\n", ans);
  }
}
