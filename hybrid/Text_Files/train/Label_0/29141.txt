#include <bits/stdc++.h>
const int INF = (int)(1e9);
const long long INFLL = (long long)(1e18);
const double EPS = 1e-13;
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  int n;
  map<string, vector<vector<string> > > procedures;
  cin >> n;
  cin.get();
  for (int i = (0); i < (int)(n); i++) {
    string line;
    getline(cin, line);
    for (int j = (0); j < (int)(line.size()); j++) {
      if (line[j] == '(' || line[j] == ')' || line[j] == ',')
        line = line.substr(0, j) + ' ' + line.substr(j + 1);
    }
    stringstream ss(line);
    string v, name, param;
    ss >> v >> name;
    procedures[name].push_back(vector<string>());
    int k = procedures[name].size() - 1;
    while (ss >> param) {
      procedures[name][k].push_back(param);
    }
  }
  map<string, string> types;
  int m;
  cin >> m;
  cin.get();
  for (int i = (0); i < (int)(m); i++) {
    string type, name;
    cin >> type >> name;
    types[name] = type;
  }
  int calls;
  cin >> calls;
  cin.get();
  for (int i = (0); i < (int)(calls); i++) {
    string line;
    getline(cin, line);
    for (int j = (0); j < (int)(line.size()); j++) {
      if (line[j] == '(' || line[j] == ')' || line[j] == ',')
        line = line.substr(0, j) + ' ' + line.substr(j + 1);
    }
    stringstream ss(line);
    string name, param;
    ss >> name;
    vector<string> param_list;
    while (ss >> param) {
      param_list.push_back(types[param]);
    }
    int cnt = 0;
    for (typeof((procedures[name]).begin()) opt = (procedures[name]).begin();
         opt != (procedures[name]).end(); ++opt) {
      if (opt->size() != param_list.size()) continue;
      bool ok = true;
      for (int k = (0); k < (int)(param_list.size()); k++) {
        if ((*opt)[k] == "T") continue;
        if (param_list[k] != (*opt)[k]) {
          ok = false;
          break;
        }
      }
      if (ok) cnt++;
    }
    cout << cnt << endl;
  }
}
