#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-8;
const long long inf = 1e9;
const long long mod = 1e9 + 7;
const int maxn = 5e1 + 10;
const int maxt = 2.5e3 + 10;
long long temp[2][maxn][maxn][3], order[maxn][maxn][maxn][3], f1[maxn][maxt],
    f2[maxn][maxn][maxt], g[3];
int main() {
  int n, T, x, y, i, j, k, l, song;
  long long sum = 0;
  scanf("%d%d", &n, &T);
  f1[0][0] = f2[0][0][0] = 1;
  for (song = 1; song <= n; song++) {
    scanf("%d%d", &x, &y);
    y--;
    if (y == 2) {
      for (i = g[2]; i >= 0; i--)
        for (k = T - x; k >= 0; k--) (f1[i + 1][k + x] += f1[i][k]) %= mod;
    } else {
      for (i = g[0]; i >= 0; i--)
        for (j = g[1]; j >= 0; j--)
          for (k = T - x; k >= 0; k--)
            (f2[i + (y == 0)][j + (y == 1)][k + x] += f2[i][j][k]) %= mod;
    }
    g[y]++;
  }
  for (i = 0; i <= g[0]; i++)
    for (j = 0; j <= g[1]; j++)
      for (k = 0; k <= g[2]; k++)
        if (i == 0 && j == 0 && k == 0)
          order[i][j][k][0] = order[i][j][k][1] = order[i][j][k][2] = 1;
        else {
          order[i][j][k][0] =
              (order[i - 1][j][k][1] + order[i - 1][j][k][2]) * i % mod;
          order[i][j][k][1] =
              (order[i][j - 1][k][0] + order[i][j - 1][k][2]) * j % mod;
          order[i][j][k][2] =
              (order[i][j][k - 1][0] + order[i][j][k - 1][1]) * k % mod;
        }
  for (i = 0; i <= g[2]; i++)
    for (j = 0; j <= g[0]; j++)
      for (k = 0; k <= g[1]; k++)
        for (l = 0; l <= T; l++)
          (sum +=
           f1[i][l] * f2[j][k][T - l] % mod *
           (order[j][k][i][0] + order[j][k][i][1] + order[j][k][i][2])) %= mod;
  printf("%lld", (sum * (mod + 1) / 2) % mod);
  return 0;
}
