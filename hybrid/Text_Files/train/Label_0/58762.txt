import java.util.*;
import java.io.*;
public class Main {
	static int n,m,comp,tot,cnt;
	static int[] dfn=new int[112345];
	static int[] low=new int[112345];
	static int[] stk=new int[112345];
	static int[] cmp=new int[112345];
	static int[] u=new int[112345];
	static int[] v=new int[112345];
	static int[] deg=new int[112345];
	static int[] ptr=new int[112345];
	static int[] vis=new int[112345];
	static ArrayList<Integer>[] arr=new ArrayList[112345];
	static ArrayList<Integer>[] G=new ArrayList[112345];
	static void dfs(int u) {
		dfn[u]=low[u]=++cnt;
		stk[++tot]=u;
		vis[u]=1;
		for (int i=0;i<G[u].size();++i) {
			int v=G[u].get(i);
			if (vis[v]>0) low[u]=Math.min(low[u],low[v]);
			else if (dfn[v]==0) {
				dfs(v);
				low[u]=Math.min(low[u],low[v]);
			}
		}
		if (low[u]==dfn[u]) {
			++comp;
			while (true) {
				cmp[stk[tot]]=comp;
				arr[comp].add(stk[tot]);
				vis[stk[tot]]=0;
				if (stk[tot--]==u) break;
			}
		}
	}
	public static void main(String[] args) throws IOException {
		BufferedReader in=new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out=new PrintWriter(System.out);
		StringTokenizer sa=new StringTokenizer(in.readLine());
		n=Integer.parseInt(sa.nextToken());
		m=Integer.parseInt(sa.nextToken());
		for (int i=1;i<=n;++i) {
			G[i]=new ArrayList<Integer>();
			arr[i]=new ArrayList<Integer>();
		}
		for (int i=0;i<m;++i) {
			sa=new StringTokenizer(in.readLine());
			u[i]=Integer.parseInt(sa.nextToken());
			v[i]=Integer.parseInt(sa.nextToken());
			G[u[i]].add(v[i]);
		}
		for (int i=1;i<=n;++i) if (dfn[i]==0) dfs(i);
		for (int i=1;i<=n;++i) G[i]=new ArrayList<Integer>();
		for (int i=0;i<m;++i)
			if (cmp[u[i]]!=cmp[v[i]]) {
				G[cmp[u[i]]].add(cmp[v[i]]);
				++deg[cmp[v[i]]];
			}
		Queue<Integer> q=new LinkedList<Integer>();
		for (int i=1;i<=comp;++i) if (deg[i]==0)
			q.add(i);
		int u=q.poll();
		while (q.size()>0) {
			int v=q.peek();
			out.printf("? %d %d\n",arr[u].get(ptr[u]),arr[v].get(ptr[v]));
			out.flush();
			int x=Integer.parseInt(in.readLine());
			if (x==0) {int k=u;u=v;v=k;}
			++ptr[v];
			if (ptr[v]==arr[v].size()) {
				q.poll();
				for (int i=0;i<G[v].size();++i) {
					int o=G[v].get(i);
					if (--deg[o]==0) q.add(o);
				}
			} else if (x==0) {
				q.poll();
				q.add(v);
			}
		}
		out.println("! "+arr[u].get(ptr[u]));
		out.flush();
	}
}
