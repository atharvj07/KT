#include<bits/stdc++.h>
using namespace std;
#define MAX 100000
int n;
double pw;
double vw[MAX],pf[MAX],vf[MAX],th[MAX];
double eps=(1e-10);

double calc(double W){
  double res=pw*W,x;
  for(int i=0;i<n;i++){
    x=vw[i]*W;
    if(x>=th[i])continue;
    res+=(th[i]-x)*pf[i]/vf[i];
  }
  return res;
}

double solve(){
  double l=0,r=1000;
  for(int i=0;i<100;i++){
    double d=(r-l)/3.0;
    if(calc(l+d)<calc(r-d)){
      r=r-d;
    }else{
      l=l+d;
    }
  }
  return calc(l);
}

int main(){
  while(1){
    cin>>n;
    if(n==0)break;
    cin>>pw;
    for(int i=0;i<n;i++)
      cin>>vw[i]>>pf[i]>>vf[i]>>th[i];
    printf("%.8f\n",solve());
  }
  return 0;
}
