#include <bits/stdc++.h>
int n, s[20], dap[10];
char a[20];
int main() {
  int i, t, j;
  scanf("%d", &n);
  scanf("%s", a);
  s[2] = 2;
  s[3] = 3;
  s[4] = 322;
  s[5] = 5;
  s[6] = 53;
  s[7] = 7;
  s[8] = 7222;
  s[9] = 7332;
  for (i = 0; i < n; i++) {
    t = s[a[i] - '0'];
    while (t) {
      dap[t % 10]++;
      t /= 10;
    }
  }
  for (i = 9; i >= 2; i--) {
    for (j = 1; j <= dap[i]; j++) printf("%d", i);
  }
  return 0;
}
