from sys import stdin
n = int(stdin.readline())
for _ in range(n):
	(size, x) = map(int, stdin.readline().rstrip().split(' '))
	nums = list(map(int, stdin.readline().rstrip().split(' ')))
	odds = 0
	for num in nums:
		if num % 2 == 1:
			odds += 1
	evens = len(nums) - odds
	for odd_group in range(1, x + 1, 2):
		if x - odd_group <= evens and odd_group <= odds:
			print('YES')
			break
	else:
		print('NO')
