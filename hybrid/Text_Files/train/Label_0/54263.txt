import java.io.*;
import java.util.*;

public class Main {

    void solve(Scanner in, PrintWriter out) {
        long n = in.nextLong();
        out.println(n * (n - 1) / 2 * (n - 2) / 3 * (n - 3) / 4 * (n - 4) / 5 * n * (n - 1) * (n - 2) * (n - 3) * (n - 4));
    }

    void run() {
        try (
                Scanner in = new Scanner(System.in);
                PrintWriter out = new PrintWriter(System.out)) {
            solve(in, out);
        }
    }

    public static void main(String[] args) {
        new Main().run();
    }
}
