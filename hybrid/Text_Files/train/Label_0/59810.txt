import java.io.*;
import java.util.*;

public class CF1348E extends PrintWriter {
	CF1348E() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1348E o = new CF1348E(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int k = sc.nextInt();
		int[] aa = new int[n];
		int[] bb = new int[n];
		long asum = 0, bsum = 0;
		for (int i = 0; i < n; i++) {
			asum += aa[i] = sc.nextInt();
			bsum += bb[i] = sc.nextInt();
		}
		long absum = asum + bsum;
		boolean[][] dp = new boolean[k][k];
		boolean[][] dq = new boolean[k][k];
		dp[0][0] = true;
		int ab = 0;
		for (int i = 0; i < n; i++) {
			for (int a = 0; a < k; a++)
				for (int b = 0; b < k; b++)
					dq[a][b] = false;
			int ai = aa[i], bi = bb[i];
			int cd = (int) (((long) ab + ai + bi) % k);
			for (int a = 0; a < k; a++) {
				int b = (ab - a + k) % k;
				if (!dp[a][b])
					continue;
				for (int c = 0; c < k; c++) {
					int d = (cd - c + k) % k;
					if (dq[c][d])
						continue;
					int ac = (c - a + k) % k;
					int bd = (d - b + k) % k;
					if (ac <= ai && bd <= bi)
						dq[c][d] = true;
				}
			}
			ab = cd;
			boolean[][] tmp = dp; dp = dq; dq = tmp;
		}
		long ans = asum / k + bsum / k;
		for (int a = 0; a < k; a++)
			for (int b = 0; b < k; b++)
				if (dp[a][b])
					ans = Math.max(ans, (absum - a - b) / k);
		println(ans);
	}
}
