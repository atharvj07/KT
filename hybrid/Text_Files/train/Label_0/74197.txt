def main():
	import sys
	input = sys.stdin.buffer.readline
	import heapq
	t = int(input())
	ans = []
	for _ in range(t):
		n = int(input())
		S = list(map(int, input().split()))
		q = []
		C = 0
		p = 0
		tmp = S[0] - 1
		for i in range(1, n):
			while q and q[0] < i:
				heapq.heappop(q)
				C -= 1
			if S[i] > C + p:
				tmp += S[i] - 1 - (C + p)
				p = 0
			else:
				p = C + p + 1 - S[i]
			heapq.heappush(q, i + S[i - 1] - 1)
			C += 1
		ans.append(tmp)
	print('\n'.join(map(str, ans)))
main()
