void solve2(int n, int m) {
	if (n==2) {
		cout << -1;
		return;
	}

	vector<vector<int> > nxt(n);
	vector<pair<int, int> > edges;
	for (int i = 0; i < m; ++i) {
		int u, v;
		cin >> u >> v;
		u--, v--;
		edges.push_back({u, v});
		nxt[u].push_back(v), nxt[v].push_back(u);
	}

	for (int i = 0; i < n; ++i) reverse(nxt[i].begin(), nxt[i].end());

	int a = 0, b = 1;
	if (edges[0].second == 1) b = 2;

	for (int i = 0; i < m; ++i) {
		int u = edges[i].first, v = edges[i].second;
		nxt[u].pop_back(), nxt[v].pop_back();

		if (b == u || b == v) swap(a, b);

		if (u == a) {
			int Q = -1;
			if (nxt[b].size()) Q = nxt[b].back();
			if (Q == a) {
				cout << 'v';
				a = v;
			}
			else cout << '^';
		}
		else if (v == a) {
			int Q = -1;
			if (nxt[b].size()) Q = nxt[b].back();
			if (Q == a) {
				cout << '^';
				a = u;
			}
			else cout << 'v';
		}
		else cout << '^';
	}

}

const int N = 50000;

bitset<N> arr[N];

void solve1(int n, int m) {
	for (int i = 0; i < n; ++i) {
		arr[i][i] = 1;
	}

	vector<pair<int, int> > edges;
	bitset<N> a, b;

	for (int i = 0; i < m; ++i) {
		int u, v;
		cin >> u >> v;
		u--, v--;
		a = arr[u], b = arr[v];
		arr[u] |= b;
		arr[v] |= a;
		edges.push_back({u, v});
	}

	int target = -1;
	for (int i = 0; i < n; ++i) {
		if (arr[i].count() == n) target = i;
	}
	if (target == -1) {
		cout << -1;
		return;
	}

	vector<bool> avail(n, false);
	avail[target] = true;
	vector<char> ans(m);
	for (int i = m - 1; i >= 0; i--) {
		int u = edges[i].first, v = edges[i].second;
		if (!avail[u] && !avail[v]) ans[i] = 'v';
		else if (avail[u] && avail[v]) ans[i] = 'v';
		else if (avail[u]) {
			avail[v] = true;
			ans[i] = '^';
		}
		else{
			avail[u] = true;
			ans[i] = 'v';
		}
	}

	for (int i = 0; i < m; ++i) cout << ans[i];

}
