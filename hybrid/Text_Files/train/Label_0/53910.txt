import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.io.StreamTokenizer;

public class C39 {

    static StreamTokenizer in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
    static PrintWriter out = new PrintWriter(System.out);
    
    static int nextInt() throws IOException {
        in.nextToken();
        return Integer.valueOf(in.sval);
    }
    
    static double nextDouble() throws IOException {
        in.nextToken();
        return Double.valueOf(in.sval);
    }
    
    static String nextString() throws IOException {
        in.nextToken();
        return in.sval;
    }
    
    static {
        in.ordinaryChars('0', '9');
        in.wordChars('0', '9');
        
        in.ordinaryChars('.', '.');
        in.wordChars('.', '.');
        
        in.ordinaryChars('-', '-');
        in.wordChars('-', '-');
    }

    public static void main(String[] args) throws IOException {
        int N = nextInt(), x = nextInt(), M = nextInt(), y = nextInt();
        
        if (x < y) {
            int tmp = x;
            x = y;
            y = tmp;
            tmp = N;
            N = M;
            M = tmp;
        }
        
        long ans = 0;
        int left = x-1, right = x+1, cnt = 0, add = M;
        if (x > y + M) {
            ans = 1;
            cnt = 0;
        }
        else {
            ans = 2;
            cnt = 1;
        }
        
        //System.out.println(left + " " + right + " " + cnt + " " + ans);
        
        for (int r = 2; r <= N; r++) {
            if (left > y && left <= y + M) cnt++;
            else if (left < y && left >= y - M) {
                cnt--;
                add--;
            }
            left--;
            
            if (right <= y + M) cnt++;
            right++;
            
            ans += Math.max(2*cnt, 1);
            
            //System.out.println(left + " " + right + " " + cnt + " " + ans);
        }
        
        if (left < y && left >= y - M) add--;
        ans += add;
        
        out.println(ans+1);
        
        out.flush();
    }

}
