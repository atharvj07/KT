#include <bits/stdc++.h>
using namespace std;
long long n, m, k, x, y, sum, tmp, jum, ans, sisa;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m >> k >> x >> y;
  if (n == 1 || n == 2) {
    if (k % (n * m) >= (m * (x - 1)) + y) {
      ans++;
    }
    cout << (k + n * m - 1) / (n * m) << " " << k / (n * m) << " "
         << k / (n * m) + ans << '\n';
  } else {
    sum = k / ((n - 1) * m);
    tmp = sum;
    jum = sum / 2;
    sisa = k % ((n - 1) * m);
    if (!sum && sisa) {
      sum++;
    } else if (sisa > m) {
      sum++;
    }
    if (sisa >= m && tmp % 2) {
      jum++;
    }
    if (x == 1) {
      ans = (tmp + 1) >> 1;
      if (tmp % 2 == 0 && sisa >= y) {
        ans++;
      }
    } else if (x == n) {
      ans = tmp >> 1;
      if (tmp % 2 && sisa >= y) {
        ans++;
      }
    } else {
      ans = tmp;
      if (tmp % 2 == 0) {
        if (sisa >= (x - 1) * m + y) {
          ans++;
        }
      } else {
        if (sisa >= (n - x) * m + y) {
          ans++;
        }
      }
    }
    cout << sum << " " << jum << " " << ans << '\n';
  }
}
