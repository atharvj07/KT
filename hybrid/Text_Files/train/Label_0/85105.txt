T = int(input())
t = 0
for t in range(T):
	N = int(input())
	n = 0
	x1 = -100000
	x2 = 100000
	y1 = -100000
	y2 = 100000
	for n in range(N):
		(x, y, s, a, d, b) = map(int, input().split())
		if s == 0:
			x1 = max(x1, x)
		if a == 0:
			y2 = min(y2, y)
		if d == 0:
			x2 = min(x2, x)
		if b == 0:
			y1 = max(y1, y)
	if x1 <= x2 and y1 <= y2:
		print('1' + ' ' + str(x1) + ' ' + str(y1))
	else:
		print('0')
		n = n + 1
	t = t + 1
