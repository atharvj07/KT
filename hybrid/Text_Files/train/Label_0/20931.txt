#include<bits/stdc++.h>
using namespace std;
typedef long long LL;
typedef pair<LL,LL> P;

LL extgcd(LL a,LL b,LL&x,LL&y)
{
    if(b==0)
    {
        x=1;
        y=0;
        return a;
    }
    LL q=a/b;
    LL g=extgcd(b,a-q*b,x,y);
    LL z=x-q*y;
    x=y;
    y=z;
    return g;
}

vector<pair<int,P>> G[1<<17];
struct F{
LL memo[1<<17];
LL mod;
F(LL m){
    mod = m;
    for(int i=0;i<1<<17;i++) memo[i] = 0;
}


bool dfs(int v,int u, LL x){
    if(memo[v]){
        if(memo[v] == x) return true;
        else return false;
    }
    memo[v] = x;
    bool f = true;
    for(int i=0;i<G[v].size();i++){
        LL s,t;
        extgcd(G[v][i].second.second,mod,s,t);
        s%=mod;
        if(s<0) s+=mod;
        f = f && dfs(G[v][i].first,v,x * G[v][i].second.first % mod * s %mod);
    }
    return f;
}
};
        

int main(){
    int n,m;cin >> n >> m;
    for(int i=0;i<m;i++){
        LL a,b,x;cin >> a >> b >> x;
        a--,b--;
        G[a].push_back({b,{1LL,x}});
        G[b].push_back({a,{x,1LL}});

    }
    F a(1000000009);
    F c(1000000021);
    F d(1000000097);
    F e(1000000087);
    F f(1000000093);
    F h(1000500013);
    F l(1000500077);
    for(int i=0;i<n;i++){
        if(a.memo[i]) continue;
        if(!a.dfs(i,-1,1LL)
            ||!c.dfs(i,-1,1LL)
            ||!d.dfs(i,-1,1LL)
            ||!e.dfs(i,-1,1LL)
            ||!f.dfs(i,-1,1LL)
            ||!h.dfs(i,-1,1LL)
            ||!l.dfs(i,-1,1LL)
            ){
            puts("No");
            return 0;
        }
    }
    puts("Yes");
    return 0;
}
