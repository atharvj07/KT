class OsTree:

	def __init__(self, max_val=10 ** 6):
		self.N = max_val
		self.tree = [0] * (self.N + 1)
		self.size = 0

	def _cumsum(self, i):
		s = 0
		while i > 0:
			s += self.tree[i]
			i -= i & -i
		return s

	def _add(self, i, val):
		while i <= self.N:
			self.tree[i] += val
			i += i & -i

	def insert(self, val):
		self._add(val, 1)
		self.size += 1

	def get_rank(self, val):
		return self._cumsum(val)

def solve(nums, N):
	val2idx = {v: i + 1 for (i, v) in enumerate(sorted(list(set(nums))))}
	tree = OsTree(2 * 10 ** 5)
	res = 0
	for num in nums:
		idx = val2idx[num]
		smaller = tree.get_rank(idx - 1)
		bigger = tree.size - tree.get_rank(idx)
		res += min(smaller, bigger)
		tree.insert(idx)
	return res

def main():
	from sys import stdin
	T = int(stdin.readline().strip())
	for _ in range(T):
		N = int(stdin.readline().strip())
		nums = list(map(int, stdin.readline().strip().split()))
		out = solve(nums, N)
		print(out)

def test():
	from random import randint
	N = 2 * 10 ** 5
	nums_rand = [randint(-10 ** 9, 10 ** 9) for _ in range(N)]
	nums_eq = [10 ** 9] * N
	nums_inc = list(range(N))
	nums_dec = list(range(N))[::-1]
	for nums in [nums_rand, nums_eq, nums_inc, nums_dec]:
		out = solve(nums, N)
		print(out)
main()
