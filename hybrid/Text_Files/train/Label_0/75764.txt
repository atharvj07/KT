import math
import bisect

def printDivisors(n, k, l):
	i = 1
	p = min(math.sqrt(n), k)
	while i <= p:
		if n % i == 0:
			if n / i == i:
				l.append(i)
			else:
				l.append(i)
				l.append(n // i)
		i = i + 1
for i in range(int(input())):
	(n, k) = map(int, input().split())
	nn = n
	l = []
	printDivisors(n, k, l)
	l.sort()
	c = bisect.bisect_right(l, k)
	if nn <= k:
		print(1)
		continue
	print(int(nn // l[c - 1]))
