#include <bits/stdc++.h>
using namespace std;
struct show {
  int l, r;
  bool operator<(const show &hs) const { return l < hs.l; }
};
int n, x, y;
show a[100010];
priority_queue<int, vector<int>, greater<int>> lq;
priority_queue<int, vector<int>, less<int>> gq;
int main() {
  scanf("%d %d %d", &n, &x, &y);
  for (int i = 0; i < n; ++i) scanf("%d %d", &a[i].l, &a[i].r);
  sort(a, a + n);
  int res = 0;
  for (int i = 0; i < n; ++i) {
    while (!lq.empty() && lq.top() < a[i].l) {
      gq.push(lq.top());
      lq.pop();
    }
    if (!gq.empty() && 1LL * y * (a[i].l - gq.top()) < x) {
      res = (res + 1LL * y * (a[i].r - gq.top())) % 1000000007;
      gq.pop();
    } else {
      res = (res + x + 1LL * y * (a[i].r - a[i].l)) % 1000000007;
      while (!gq.empty()) gq.pop();
    }
    lq.push(a[i].r);
  }
  printf("%d", res);
  return 0;
}
