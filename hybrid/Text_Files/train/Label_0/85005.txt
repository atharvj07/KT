def inp(dtype=str, strip=True):
	s = input()
	res = [dtype(p) for p in s.split()]
	res = res[0] if len(res) == 1 and strip else res
	return res

def problemA():
	t = int(input())
	for _ in range(t):
		s = input()
		res = 0
		for i in range(s.find('1'), s.rfind('1')):
			res += int(s[i] == '0')
		print(res)

def problemB():
	t = int(input())
	for _ in range(t):
		(n, g, b) = inp(int)
		hq = (n + 1) // 2
		hq_days = (hq - 1) // g * (g + b) + (hq - 1) % g + 1
		res = max(n, hq_days)
		print(res)
problemB()
