from collections import defaultdict
for _ in range(int(input())):
	a = input()
	b = input()
	f = 0
	cnt = 0
	x = -1
	d = defaultdict(list)
	for i in range(len(a)):
		d[a[i]].append(i)
	ma = -1
	for i in range(len(b)):
		l = d[b[i]]
		if len(l) == 0:
			cnt = -2
			break
		(beg, end) = (0, len(l) - 1)
		while beg <= end:
			mid = (beg + end) // 2
			if l[mid] > x:
				end = mid - 1
			else:
				beg = mid + 1
		if beg < len(l):
			x = l[beg]
		else:
			cnt += 1
			x = l[0]
	print(cnt + 1)
