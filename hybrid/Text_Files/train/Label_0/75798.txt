t = int(input())
for i in range(0, t):
	n = int(input())
	a = input().split()
	c = []
	for j in a:
		c.append(int(j))
	bl = True
	for k in range(0, len(c) - 1):
		o = c[k]
		c[k] = k
		c[k + 1] = o - k + c[k + 1]
		if c[k + 1] < k + 1:
			bl = False
			break
	if bl:
		print('YES')
	else:
		print('NO')
