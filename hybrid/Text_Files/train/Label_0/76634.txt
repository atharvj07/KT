import sys
import bisect
input = sys.stdin.readline

def SieveOfEratosthenes(n):
	arr = []
	prime = [True for i in range(n + 1)]
	p = 2
	while p * p <= n:
		if prime[p] == True:
			for i in range(p ** 2, n + 1, p):
				prime[i] = False
		p += 1
	prime[0] = False
	prime[1] = False
	for p in range(n + 1):
		if prime[p]:
			arr.append(p)
	return arr

def solve():
	(n, m) = map(int, input().split())
	a = []
	for _ in range(n):
		a.append(input())
	v = []
	for j in range(1, m):
		for i in range(1, n):
			if a[i][j - 1] == 'X' and a[i - 1][j] == 'X':
				v.append(j)
				break
	v.append(m)
	q = int(input())
	for i in range(q):
		(x, y) = map(int, input().split())
		lower_bound = bisect.bisect_left(v, x, 0, len(v))
		if v[lower_bound] < y:
			print('NO')
		else:
			print('YES')
t = 1
for _ in range(t):
	k = solve()
