#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  long long n;
  cin >> n;
  vector<long long> x(n), y(n);
  vector<pair<long long, long long>> a(n);
  for (int i = 0; i < n; i++) {
    long long z;
    cin >> z;
    a[i] = {z, i};
    x[i] = z;
  }
  if (n == 1) {
    cout << a[0].first;
    return 0;
  }
  sort(a.begin(), a.end());
  x[0] = a[n - 1].first;
  for (int i = 0; i < n - 1; i++) {
    x[i + 1] = a[i].first;
  }
  for (int i = 0; i < n; i++) {
    y[a[i].second] = x[i];
  }
  for (int i = 0; i < n; i++) {
    cout << y[i] << ' ';
  }
}
