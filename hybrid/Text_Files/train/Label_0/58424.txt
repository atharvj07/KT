import java.util.Scanner;

public class F {

	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		long[][] left = new long[n][2];
		long[][] right = new long[n][2];
		int[] get = new int[n-1];
		for(int i = 0; i < n-1; i++) get[i] = scan.nextInt();
		long max = 0;
		for(int i = 1; i < n; i++) {
			int a = get[i-1];
			if(a == 1) {
				left[i][1] = Math.max(left[i-1][0], left[i-1][1])+1;
			}
			else {
				left[i][0] = left[i-1][0] + (a-(a%2));
				left[i][1] = Math.max(left[i-1][0], left[i-1][1]) + (a-((a+1)%2));
			}
		}
		for(int i = n-2; i >= 0; i--) {
			int a = get[i];
			if(a == 1) {
				right[i][1] = Math.max(right[i+1][0], right[i+1][1])+1;
			}
			else {
				right[i][0] = right[i+1][0] + (a-(a%2));
				right[i][1] = Math.max(right[i+1][0], right[i+1][1]) + (a-((a+1)%2));
			}
		}
		for(int i = 0; i < n; i++) {
			if(i == 0) {
				max = Math.max(max, Math.max(right[i][0], right[i][1])); 
			}
			else if(i == n-1) {
				max = Math.max(max, Math.max(left[i][0], left[i][1]));
			}
			else {
				long a = left[i][0] + Math.max(right[i][0], right[i][1]);
				long b = right[i][0] + Math.max(left[i][0], left[i][1]);
				long c = Math.max(left[i][0], left[i][1]);
				long d = Math.max(right[i][0], right[i][1]);
				if(left[i][0] != 0) max = Math.max(max, a);
				if(right[i][0] != 0) max = Math.max(max, b);
				max = Math.max(max, Math.max(c, d));
			}
			//System.out.println(i + " " + max);
		}
		System.out.println(max);
	}

}
