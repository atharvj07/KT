def inp(s, i):
	k = 0
	it = ''
	s = ' ' + s
	for j in s:
		if j == ' ':
			k += 1
		if k == i and (not j == ' '):
			it += j
	return it
s = input()
n = int(inp(s, 1))
k = int(inp(s, 2))
x = int(inp(s, 3))
s = input()
a = list()
for q in range(n):
	p = int(inp(s, q + 1))
	a.append(p)
