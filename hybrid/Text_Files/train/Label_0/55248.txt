 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskB solver = new TaskB();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskB {
        long P;
        long MOD = (int) (1e9) + 7;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int T = in.nextInt();
            while (T-- > 0) {
                int N = in.nextInt();
//            int N = 10;
                P = in.nextInt();
//            P = 5;
                ArrayList<Integer> ks = new ArrayList<>();
                for (int i = 0; i < N; i++) {
                    int a = in.nextInt();
//                int a = (int)(Math.random() * 8);
                    ks.add(a);
                }
                if (P == 1) {
                    out.println(N % 2);
                    continue;
                }
                Collections.sort(ks, new Comparator<Integer>() {

                    public int compare(Integer integer, Integer t1) {
                        return Integer.compare(t1, integer);
                    }
                });
                Poly a = new Poly();
                int h = -1;
                for (int i : ks) {
                    if (h == -1) {
                        h = i;
                    } else {
                        a.add(i);
                        if (a.p.containsKey(h)) {
                            if (a.p.size() > 1) {
                                int[] arr = new int[2];
                                out.println(arr[3]);
                            }
                            a.p.remove(h);
                            h = -1;
                        }
                    }
                }
                long max = 0;
                long res;
                if (h == -1) {
                    out.println(0);
//                res = 0;
                } else {
                    for (int i : a.p.keySet()) {
                        max += (MathUtils.binPow(P, i, MOD) * a.p.get(i)) % MOD;
                        max %= MOD;
                    }
                    res = MathUtils.binPow(P, h, MOD) - max;
                    while (res < 0) {
                        res += MOD;
                    }
                    while (res >= MOD) {
                        res -= MOD;
                    }
                    assert res >= 0 : "This can't have happened";
                    out.println(res);
                }
            }
        }

        class Poly {
            HashMap<Integer, Integer> p = new HashMap<>();

            void add(int i) {
                if (p.containsKey(i) && p.get(i) == P - 1) {
                    p.remove(i);
                    add(i + 1);
                } else {
                    p.putIfAbsent(i, 0);
                    p.replace(i, p.get(i) + 1);
                }
            }

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }

    static class MathUtils {
        public static long binPow(long b, long p, long mod) {
            long res = 1;
            while (p > 0) {
                if ((p & 1) == 1) {
                    res *= b;
                    res %= mod;
                }
                b = b * b;
                p >>= 1;
                b %= mod;
            }
            return res;
        }

    }
}
