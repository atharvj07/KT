//package codeforces;
import java.util.*;
import java.io.*;

public class TreeTacToe {
    public static void main(String[] args) {
        new TreeTacToe().solve();
    }

    FastReader in = new FastReader();
    PrintWriter out = new PrintWriter(System.out);
    int N = 500000+2000, n;
    int[] degree;
    @SuppressWarnings("unchecked") ArrayList<Integer>[] to = new ArrayList[N];
   
    class FastReader{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = null;

        String next(){
            while(st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch(IOException e){
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        int nextInt(){
            return Integer.parseInt(in.next());
        } 
    }

    void solve(){
        int T = in.nextInt();
        while(T-->0){
            n = in.nextInt();
            degree = new int[n+1000];
            for(int i = 0 ; i < n+1000 ; i++){
                to[i] = new ArrayList<>();
            } 
            for(int i = 0 ; i < n-1 ; i++){
                int u = in.nextInt();
                int v = in.nextInt();
                add(u, v);
            }
            String s = in.next();
            for(int i = 0 ; i < s.length() ; i++){
                if(s.charAt(i) == 'W'){
                    int t = ++n;
                    add(i+1, t);
                    add(t, ++n);
                    add(t, ++n);
                }
            }
            out.println(check() ? "White" : "Draw");
        }
        out.close();
    }

    void add(int u, int v){
        to[u].add(v);
        to[v].add(u);
        degree[u]++;
        degree[v]++;
    }

    boolean check(){
        int firstnode = 0, secondnode = 0;

        for (int i = 1; i <= n; i++){
            if(degree[i] >= 4){
                return true;
            }
            if(degree[i] == 3){
                if(firstnode == 0){
                    firstnode = i;
                }else if(secondnode == 0){
                    secondnode = i;
                }
                int branchnode = 0;
                for(int j = 0; j < to[i].size() ; j++){
                    if(degree[to[i].get(j)] > 1){
                        branchnode++;
                    }
                }
                if(branchnode >= 2){
                    return true;
                }
            }
        }
        if(firstnode == 0 || secondnode == 0){
            return false;
        }
        return DFS(firstnode, -1, secondnode)%2 == 0;
    }

    int DFS(int begin, int last, int end){
        if(begin == end){
            return 0;
        }
        for(int i = 0 ; i < to[begin].size() ; i++){
            if(to[begin].get(i) != last){
                int distance = DFS(to[begin].get(i), begin, end);
                if(distance != -1){
                    return distance+1;
                }
            }        
        }
        return -1;
    } 
}
