#include <bits/stdc++.h>
int main() {
  int a, b, c, d, i, j, k;
  int x1[101], x2[101];
  scanf("%d", &a);
  a = a - 1;
  scanf("%d", &b);
  for (i = 0; i < b; i++) {
    scanf("%d", &x1[i]);
  }
  while (a--) {
    scanf("%d", &c);
    for (i = 0; i < c; i++) {
      scanf("%d", &x2[i]);
    }
    for (i = 0; i < b; i++) {
      k = 0;
      for (j = 0; j < c; j++) {
        if (x1[i] == x2[j]) k = 1;
      }
      if (k == 0) x1[i] = 0;
    }
  }
  for (i = 0; i < b; i++) {
    if (x1[i] != 0) printf("%d ", x1[i]);
  }
  return 0;
}
