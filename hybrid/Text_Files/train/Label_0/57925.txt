import java.io.*;
import java.util.*;
import java.math.BigInteger;

public class Codeforces913G {
	public static void main (String[] args) {
		Scanner input = new Scanner(System.in);
		int n = input.nextInt();
		long[] a = new long[n];
		for (int i = 0; i < n; i++) {
			a[i] = input.nextLong();
		}
		input.close();
		
		long[] power5 = new long[18];
		power5[0] = 1;
		for (int j = 1; j < 18; j++) {
			power5[j] = 5*power5[j-1];
		}
	
		long[] powertower = new long[18];
		powertower[1] = 2;
		for (int j = 2; j < 18; j++) {
			powertower[j] = twopower(power5[j-1]*4/5, power5[17]);
		}
		
		for (int i = 0; i < n; i++) {
			long val = a[i]*1000000;
			long r = val%131072;
			val += 131072-r;
			if (val%5 == 0) {
				val += 131072;
			}
			long[] mod5 = new long[18];
			for (int j = 1; j < 18; j++) {
				mod5[j] = val%power5[j];
			}
			
			long ans = 0;
			long q = 1;
			for (int j = 1; j < 18; j++) {
				while ((q%power5[j]) != mod5[j]) {
					ans += (power5[j-1]*4+1)/5;
					q = multiply(q, powertower[j], power5[17]);
				}
			}
			if (ans < 18) {
				ans += 4*power5[17];
			}
			System.out.println(ans);
		}
	}
	
	public static long multiply (long a, long b, long mod) {
		long c = 0;
		long d = b;
		while (a > 0) {
			if (a%2 == 1) {
				c = (c+d)%mod;
			}
			a /= 2;
			d = (d*2)%mod;
		}
		
		return c;
		
	}
	
	public static long twopower(long n, long mod) {
		long a = 1;
		long power = 2;
		while (n > 0) {
			if (n%2 == 1) {
				a = multiply(a, power, mod);
			}
			n /= 2;
			power = multiply(power, power, mod);
		}
		return a;
	}

}
