#include <bits/stdc++.h>
using namespace std;
struct cmp_t {
  bool operator()(const int64_t& x, const pair<int64_t, int>& y) const {
    return x < y.first;
  }
} cmp;
int main() {
  int n, m, c, t;
  cin >> n >> m;
  int64_t tim = 0, w;
  vector<pair<int64_t, int> > v(n);
  for (int i = 0; i < n; ++i) {
    v[i].first = tim;
    v[i].second = i + 1;
    cin >> c >> t;
    tim += c * t;
  }
  for (int i = 0; i < m; ++i) {
    cin >> w;
    auto it = upper_bound(v.begin(), v.end(), w - 1, cmp);
    cout << (it - 1)->second << endl;
  }
  return 0;
}
