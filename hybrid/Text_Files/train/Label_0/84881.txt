def sexy(a):
	a.append(0)
	k = sum(a)
	b = a[:]
	s0 = []
	s1 = []
	i = 0
	j = 1
	c = 0
	while True:
		if c % 2 == 0:
			while a[i] == 0 and i < 4:
				i += 1
			if i == 4:
				break
			if i == j:
				i += 1
				while a[i] == 0 and i < 4:
					i += 1
				if i == 4:
					break
			s0.append(i)
			a[i] -= 1
		else:
			while a[j] == 0 and j < 4:
				j += 1
			if j == 4:
				break
			if i == j:
				j += 1
				while a[j] == 0 and j < 4:
					j += 1
				if j == 4:
					break
			s0.append(j)
			a[j] -= 1
		c += 1
	i = 1
	j = 0
	c = 0
	while True:
		if c % 2 == 0:
			while b[i] == 0 and i < 4:
				i += 1
			if i == 4:
				break
			if i == j:
				i += 1
				while b[i] == 0 and i < 4:
					i += 1
				if i == 4:
					break
			s1.append(i)
			b[i] -= 1
		else:
			while b[j] == 0 and j < 4:
				j += 1
			if j == 4:
				break
			if i == j:
				j += 1
				while b[j] == 0 and j < 4:
					j += 1
				if j == 4:
					break
			s1.append(j)
			b[j] -= 1
		c += 1
	if k == 1:
		print('YES')
		return [s0 if len(s0) == 1 else s1, True]
	ok1 = True
	ok2 = True
	for i in range(1, len(s0)):
		if abs(s0[i] - s0[i - 1]) != 1:
			ok1 = False
			break
	if len(s0) != k:
		ok1 = False
	for i in range(1, len(s1)):
		if abs(s1[i] - s1[i - 1]) != 1:
			ok2 = False
			break
	if len(s1) != k:
		ok2 = False
	if ok1:
		print('YES')
		return [s0, True]
	elif ok2:
		print('YES')
		return [s1, True]
	else:
		return [[], False]

def main():
	a = list(map(int, input().split()))
	b = a[:]
	p = sexy(b)
	if p[1]:
		print(*p[0])
		return
	else:
		a = a[::-1]
		p = sexy(a)
		if not p[1]:
			print('NO')
			return
		for i in range(len(p[0])):
			p[0][i] = 3 - p[0][i]
		print(*p[0])
	return
main()
