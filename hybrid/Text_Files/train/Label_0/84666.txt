t = int(input())
for _ in range(t):
	n = int(input())
	arr = input()
	zeroList = []
	oneList = []
	result = [0] * len(arr)
	ind = 0
	getit = 1
	for i in arr:
		if i == '0':
			if len(oneList) > 0:
				result[ind] = oneList.pop()
				zeroList.append(result[ind])
			else:
				result[ind] = getit
				zeroList.append(getit)
				getit = getit + 1
		elif len(zeroList) > 0:
			result[ind] = zeroList.pop()
			oneList.append(result[ind])
		else:
			result[ind] = getit
			oneList.append(getit)
			getit = getit + 1
		ind += 1
	print(max(result))
	print(*result)
