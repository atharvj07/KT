#include <bits/stdc++.h>
long i, n, pr[45050];
bool flag;
void search(long low, long high, long p) {
  if (high < low || flag == 0) return;
  long mid = (high + low) / 2;
  if (pr[mid] == p) {
    flag = 0;
    return;
  } else if (pr[mid] > p)
    search(low, mid - 1, p);
  else
    search(mid + 1, high, p);
}
int main() {
  for (i = 1; i < 45000; i++) pr[i] = (i * (i + 1)) / 2;
  while (~scanf("%ld", &n)) {
    flag = 1;
    for (i = 1; i < 45000; i++) {
      search(1, 45000, n - pr[i]);
      if (flag == 0) break;
    }
    if (flag)
      printf("NO\n");
    else
      printf("YES\n");
  }
  return 0;
}
