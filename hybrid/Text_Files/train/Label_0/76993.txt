n = int(input())
ls = [int(x) for x in input().split(' ')]
t = 1
mod = 1000000007

def qpow(n: int, m: int):
	m %= mod - 1
	ans = 1
	while m:
		if m & 1:
			ans = ans * n % mod
		n = n * n % mod
		m >>= 1
	return ans
k = 0
r = mod * 2 - 2
for x in ls:
	t = t * x % (mod * 2 - 2)
	if x % 2 == 0:
		k = 1
t = (t - 1 + r) % r
fz = (qpow(4, t >> 1) - 1) * qpow(3, mod - 2) % mod
if k == 1:
	fz = (fz * 2 + 1) % mod
fm = qpow(2, t)
print('{}/{}'.format(fz, fm))
