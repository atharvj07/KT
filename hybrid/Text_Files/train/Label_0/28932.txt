vector<lint> head;

constexpr lint INF = 1E9 + 100;
void build_head(lint N)
{
    lint now = 1, sz = 1;
    while (now < INF)
    {
        head.push_back(now);
        if (now > INF or sz > INF) break;
        now += sz;
        sz *= N;
    }
}

lint N;
int find_par(int now)
{
    auto ptr = prev(upper_bound(ALL(head), now));
    auto prvptr = prev(ptr);
    return (now - *ptr) / N + *prvptr;
}

int main()
{
    cin >> N;
    if (N != 2 and N != 1) build_head(N);
    int Q;
    cin >> Q;
    REP(_, Q)
    {
        lint v, w;
        cin >> v >> w;
        if (N == 1)
        {
            printf("%lld\n", min(v, w));
        }
        else if (N != 2)
        {
            while (v != w)
            {
                if (v > w) v = find_par(v);
                else w = find_par(w);
            }
            printf("%lld\n", v);
        }
        else
        {
            while(v != w)
            {
                if (v > w) v /= 2;
                else w /= 2;
            }
            printf("%lld\n", v);
        }
    }
    return 0;
}
