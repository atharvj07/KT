#include "bits/stdc++.h"

using namespace std;

using ll = long long;
constexpr ll MOD = 998244353;

int n, k;
ll dp[100010][310][2];
vector<int> sub(100010, 1);
vector<vector<int>> g(100010);

void rec(int cur, int pre)
{
    dp[cur][0][0] = dp[cur][1][1] = 1;
    int ma = min(k, 1);
    for (auto v : g[cur])
    {
        if (v == pre)
            continue;
        rec(v, cur);
        sub[cur] += sub[v];
        int s = min(k, sub[v]);
        vector<ll> nxt0(min(k, ma + sub[v]) + 1, 0), nxt1(min(k, ma + sub[v]) + 1, 0);
        for (int i = 0; i <= ma; i++)
        {
            for (int j = 0; j <= s; j++)
            {
                if (i + j > k)
                    break;
                ll sum = (dp[v][j][0] + dp[v][j][1]) * dp[cur][i][0] % MOD;
                nxt0[i + j] = (nxt0[i + j] + sum);
                if (nxt0[i + j] >= MOD)
                    nxt0[i + j] -= MOD;
            }
            for (int j = 0; j <= s; j++)
            {
                if (i + j > k)
                    break;
                ll sum = (dp[v][j][0] + dp[v][j][1]) * dp[cur][i][1] % MOD;
                nxt1[i + j] = (nxt1[i + j] + sum);
                if (nxt1[i + j] >= MOD)
                    nxt1[i + j] -= MOD;
            }
            for (int j = 1; j <= s; j++)
            {
                if (i + j - 1 > k)
                    break;
                ll sum = dp[v][j][1] * dp[cur][i][1] % MOD;
                nxt1[i + j - 1] = (nxt1[i + j - 1] + sum);
                if (nxt1[i + j - 1] >= MOD)
                    nxt1[i + j - 1] -= MOD;
            }
        }
        ma = min(k, ma + sub[v]);
        for (int i = 0; i <= ma; i++)
        {
            dp[cur][i][0] = nxt0[i];
            dp[cur][i][1] = nxt1[i];
        }
    }
}

int main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);

    cin >> n >> k;
    for (int i = 0; i < n - 1; i++)
    {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    rec(0, -1);
    cout << (dp[0][k][0] + dp[0][k][1]) % MOD << endl;
}
