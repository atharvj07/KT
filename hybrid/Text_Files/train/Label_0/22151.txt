#include <bits/stdc++.h>
using namespace std;
using namespace std;
const int N = 200010;
const int P = 998244353;
set<pair<int, int> > s[N];
struct st {
  int l, r, c, d, v, len;
} tr[N << 2];
int n, m;
void split(int x, int p) {
  set<pair<int, int> >::iterator it;
  it = s[x].lower_bound(make_pair(p, p));
  if (it == s[x].begin()) return;
  it--;
  pair<int, int> nw = (*it);
  if (nw.second >= p) {
    s[x].erase(nw);
    s[x].insert(make_pair(nw.first, p - 1));
    s[x].insert(make_pair(p, nw.second));
  }
}
void build(int x, int l, int r) {
  tr[x] = (st){l, r, 1, 0, 0, r - l + 1};
  if (l == r) return;
  int md = l + r >> 1;
  build(x << 1, l, md);
  build(x << 1 | 1, md + 1, r);
}
void upd(int x) {
  tr[x].v = tr[x << 1].v + tr[x << 1 | 1].v;
  if (tr[x].v >= P) tr[x].v -= P;
}
void add(int x, int c, int d) {
  tr[x].v = (1ll * tr[x].v * c + 1ll * tr[x].len * d) % P;
  tr[x].c = 1ll * tr[x].c * c % P;
  tr[x].d = (1ll * tr[x].d * c + d) % P;
}
void down(int x) {
  if (tr[x].c > 1 || tr[x].d) {
    add(x << 1, tr[x].c, tr[x].d);
    add(x << 1 | 1, tr[x].c, tr[x].d);
    tr[x].c = 1;
    tr[x].d = 0;
  }
}
void chg(int x, int l, int r, int c, int d) {
  if (l > tr[x].r || r < tr[x].l) return;
  if (l <= tr[x].l && tr[x].r <= r) {
    add(x, c, d);
    return;
  }
  down(x);
  chg(x << 1, l, r, c, d);
  chg(x << 1 | 1, l, r, c, d);
  upd(x);
}
int qry(int x, int l, int r) {
  if (l > tr[x].r || r < tr[x].l) return 0;
  if (l <= tr[x].l && tr[x].r <= r) return tr[x].v;
  down(x);
  return (qry(x << 1, l, r) + qry(x << 1 | 1, l, r)) % P;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) s[i].insert(make_pair(1, n));
  build(1, 1, n);
  for (int i = 1; i <= m; i++) {
    int kd, l, r, x;
    scanf("%d%d%d", &kd, &l, &r);
    if (kd == 1) {
      scanf("%d", &x);
      split(x, l);
      split(x, r + 1);
      chg(1, l, r, 2, 0);
      while (true) {
        set<pair<int, int> >::iterator it = s[x].lower_bound(make_pair(l, l));
        if (it == s[x].end() || it->second > r) break;
        chg(1, it->first, it->second, (P + 1) / 2, 1);
        s[x].erase(it);
      }
    } else
      printf("%d\n", qry(1, l, r));
  }
  return 0;
}
