#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define MOD 1000000007

int main(){
  string s;
  cin>>s;
  int l=0, r=s.size()-1, c=0;
  while(1){
    //cout<<s[i]<<" "<<s[ind]<<endl;
    if(s[l]==s[r]){
      l++;
      r--;
    }else if(s[l]=='x'){
      l++;
      c++;
    }else if(s[r]=='x'){
      r--, c++;
    }else{
      cout<<-1<<endl;
      return 0;
    }
    if(l>r) break;
  }
  cout<<c<<endl;
  return 0;
}
