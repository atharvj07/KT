#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e9;
int n, mxp, a[1000];
queue<int> Q;
int gcd(int x, int y) {
  if (!x) return y;
  return gcd(y % x, x);
}
int main() {
  mxp = 999999937;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  Q.push(a[0]);
  for (int i = 1; i < n; i++) {
    int g = gcd(a[i], a[i - 1]);
    if (g > 1) {
      if (g == mxp)
        Q.push(2);
      else
        Q.push(mxp);
    }
    Q.push(a[i]);
  }
  cout << Q.size() - n << endl;
  while (!Q.empty()) {
    cout << Q.front() << " ";
    Q.pop();
  }
  cout << endl;
}
