#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:100000000,100000000")
using namespace std;
const long long inf = 1e18 + 7;
const long long mod = 1e8 + 7;
const double eps = 1e-9;
const double PI = 2 * acos(0.0);
const double E = 2.71828;
long long a[100005], c[100005];
long long used[100005];
struct pr {
  long long x, y;
  friend bool operator<(pr a, pr b) {
    return a.x > b.x || (a.x == b.x && a.y < b.y);
  }
};
inline pr mpr(long long x, long long y) {
  pr a;
  a.x = x;
  a.y = y;
  return a;
}
set<pr> second;
long long q[105];
long long sz = 0;
long long d[100005];
long long n, k;
void add(long long now) {
  q[sz++] = now;
  set<pair<long long, long long> > s;
  s.insert(make_pair(0, 0));
  fill(d, d + k, inf);
  d[0] = 0;
  while (s.size()) {
    long long y = (*s.begin()).first;
    long long x = (*s.begin()).second;
    s.erase(s.begin());
    for (long long i = 0; i < sz; i++) {
      long long to = (x + q[i]) % k;
      if (d[to] > y + q[i]) {
        d[to] = d[x] + q[i];
        s.insert(make_pair(d[to], to));
      }
    }
  }
  second.clear();
  for (long long i = 0; i < n; i++)
    if (a[i] >= d[a[i] % k]) second.insert(mpr(c[i], i)), used[i] = 1;
}
int main(void) {
  long long h, m;
  cin >> h >> n >> m >> k;
  for (long long(i) = 0; (i) < (long long)(n); (i)++)
    cin >> a[i] >> c[i], a[i]--;
  add(0);
  for (long long(i) = 0; (i) < (long long)(m); (i)++) {
    int num;
    scanf("%d", &num);
    if (num == 1) {
      long long x;
      cin >> x;
      add(x);
    } else if (num == 2) {
      long long x, y;
      cin >> x >> y;
      x--;
      if (!used[x])
        c[x] -= y;
      else {
        second.erase(second.find(mpr(c[x], x)));
        c[x] -= y;
        second.insert(mpr(c[x], x));
      }
    } else if (num == 3) {
      if (!second.size()) {
        cout << 0 << endl;
        continue;
      }
      pr now = *second.begin();
      cout << now.x << endl;
      c[now.y] = 0;
      second.erase(second.begin());
    }
  }
  return 0;
}
