import java.util.*;
import java.math.*;
public class Main
{
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0)
		{
		    int n=sc.nextInt();
		    int p=sc.nextInt();
		    int k=sc.nextInt();
		    int[] arr=new int[n];
		    for(int i=0;i<n;i++)
		    arr[i]=sc.nextInt();
		    Arrays.parallelSort(arr);
		    int[] srr=new int[k];
		    srr[0]=arr[0];
		    for(int i=1;i<k;i++)
		    srr[i]=srr[i-1]+arr[i];
		    int ans=0;
		    for(int i=k-1,q=0;q<k;i++,q++)
		    {
		        int tmp1=p;
		        int tmp2=0;
		        for(int j=i;j<n;j+=k)
		        {
		            if(tmp1-arr[j]>=0)
		            {
		                tmp1-=arr[j];
		                tmp2+=k;
		            }
		            else
		            break;
		        }
		       // System.out.println(tmp1+"  ll "+tmp2+"  test " +t);
		        int x=bs(0,i-k,srr,tmp1);
		        if(x!=-1)
		        tmp2+=(x+1);
		        ans=Math.max(tmp2,ans);
		       // System.out.println(x+"  ll "+tmp2+"  test " +t);
		    }
		    System.out.println(ans);
		}
	}
	public static int bs(int l,int r,int[] srr,int x)
	{
	    int ans=-1;
	    while(l<=r)
	    {
	        int m=(l+r)/2;
	        if(srr[m]<=x)
	        {
	            ans=m;
	            l=m+1;
	        }
	        else
	        r=m-1;
	    }
	    return ans;
	}
}
