def sort_one_index(a, p):
	for i in p:
		if a[i - 1] > a[i]:
			(a[i - 1], a[i]) = (a[i], a[i - 1])
	return a

def weirdsort(a, p):
	b = a.copy()
	a = sort_one_index(a, p)
	while b != a:
		b = a.copy()
		a = sort_one_index(a, p)
	if a == sorted(a):
		return 'YES'
	else:
		return 'NO'
