from bisect import bisect_left, bisect_right
for _ in range(int(input())):
	n = int(input())
	l = []
	r = []
	a = []
	for i in range(n):
		(li, ri) = [int(i) for i in input().split()]
		l.append(li)
		r.append(ri)
		a.append([li, ri])
	l.sort()
	r.sort()
	cnt = n
	for i in range(n):
		cnt = min(cnt, n - bisect_right(l, a[i][1]) + bisect_left(r, a[i][0]))
	print(cnt)
