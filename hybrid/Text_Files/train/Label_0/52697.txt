public class CF1208F{
  static class FastReader 
    { 
        BufferedReader br; 
        StringTokenizer st; 
  
        public FastReader() 
        { 
            br = new BufferedReader(new
                     InputStreamReader(System.in)); 
        } 
  
        String next() 
        { 
            while (st == null || !st.hasMoreElements()) 
            { 
                try
                { 
                    st = new StringTokenizer(br.readLine()); 
                } 
                catch (IOException  e) 
                { 
                    e.printStackTrace(); 
                } 
            } 
            return st.nextToken(); 
        } 
  
        int nextInt() 
        { 
            return Integer.parseInt(next()); 
        } 
  
        
    } 
    
    static int MOD=998244353;

        f=new int[1<<M]; g=new int[1<<M];

                  if(f[j]==-1)f[j]=i;

                      g[j]=f[j]; f[j]=i;

    public static void merge(int a, int b, int mask) { // a<b
    	if(b>f[mask]) {
    		g[mask]=f[mask]; f[mask]=b;
    		if(a>g[mask]) {
    			g[mask]=a;
    		}
    	}else if(b>g[mask] && b!=f[mask]) {// NO DUPS!!
    		g[mask]=b;
    	}
    	if(a>g[mask]) {
    		g[mask]=a;
    	}
    }
