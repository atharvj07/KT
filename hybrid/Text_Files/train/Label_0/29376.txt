#include<algorithm>
#include<iostream>
using namespace std;
int main(){
  int i;
  int n,s;
  while(cin>>n>>s,n||s){
    int a[20001];
    a[n]=1000;
    for(i=0;i<n;++i)
      cin>>a[i];
    sort(a,a+n);
    int ct=0;
    for(i=0;i<n-1;++i)
      ct+=n-(lower_bound(a+i+1,a+n+1,s-a[i]+1)-a);
    cout<<ct<<endl;
  }
  return 0;
}
