from math import ceil

def writeList(l):
	for (i, item) in enumerate(l):
		if i == len(l) - 1:
			print(item)
		else:
			print(item, end='')

def solve(items, n, aux_s):
	(r, last) = (['' for i in range(len(aux_s))], '')
	(num_minus, num_plus, c1, c2) = (0, 1, 0, 0)
	(last_divisor1, last_divisor2) = (1, 1)
	for i in items:
		if i == '-':
			num_minus += 1
		elif i == '+':
			num_plus += 1
	if num_minus == 0:
		if num_plus > n:
			return ['Impossible']
		flag = False
		(k, res) = divmod(n, num_plus)
		while res + k > n:
			last_divisor1 = min(num_plus, res)
			res = res / min(num_plus, res)
			if res == 1 and res + k > n:
				flag = True
				break
		if flag:
			return ['Impossible']
		for i in range(len(aux_s)):
			if aux_s[i] == '?':
				if c1 <= last_divisor1:
					r[i] = k + res
					c1 += 1
				else:
					r[i] = k
			else:
				r[i] = aux_s[i]
		return ['Possible', r]
	flag = False
	(k, res) = divmod(num_minus + n, num_plus)
	if k > n:
		return ['Impossible']
	while res + k > n:
		last_divisor1 = min(num_plus, res)
		res = res / min(num_plus, res)
		if int(res) == 1 and res + k > n:
			flag = True
			break
	(k, res) = (int(k), int(res))
	(m, s) = (1, 0)
	if flag:
		s = res + k - n
		while m + s > n:
			last_divisor2 = min(num_minus, s)
			s = s / min(num_minus, s)
			if s == 1 and m + s > n:
				flag = True
				break
		if flag:
			return ['Impossible']
		for i in range(len(aux_s)):
			if i == 0:
				r[i] = k + res
				c1 += 1
			elif aux_s[i] == '+':
				r[i] = aux_s[i]
				last = '+'
			elif aux_s[i] == '-':
				r[i] = aux_s[i]
				last = '-'
			elif aux_s[i] == '?':
				if last == '+':
					if c1 < last_divisor1:
						r[i] = k + res
						c1 += 1
					else:
						r[i] = k
				if last == '-':
					if c2 < last_divisor2:
						r[i] = m + s
						c2 += 1
					else:
						r[i] = m
		return ['Possible', r]
	if k == 0:
		(k, res) = (1, 0)
		(m, s) = divmod(num_plus - n, num_minus)
		while m + s > n:
			last_divisor2 = min(num_minus, s)
			s = s / min(num_minus, s)
			if int(s) == 1 and m + s > n:
				flag = True
				break
		if flag:
			return ['Impossible']
		(m, s) = (int(m), int(s))
	for i in range(len(aux_s)):
		if aux_s[i] == '+':
			r[i] = aux_s[i]
			last = '+'
		elif aux_s[i] == '-':
			r[i] = aux_s[i]
			last = '-'
		elif aux_s[i] == '?':
			if i == 0:
				r[i] = k + res
				c1 += 1
			elif last == '+':
				if c1 < last_divisor1:
					r[i] = k + res
					c1 += 1
				else:
					r[i] = k
			elif last == '-':
				if c2 < last_divisor2:
					r[i] = m + s
					c2 += 1
				else:
					r[i] = m
		else:
			r[i] = aux_s[i]
	return ['Possible', r]
str_aux = input()
items = [i for i in str_aux.split()]
n = int(items[len(items) - 1])
for i in solve(items, n, str_aux):
	writeList(i)
