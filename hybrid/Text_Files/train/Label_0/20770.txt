#include <bits/stdc++.h>
using namespace std;
long long int i, j, k, n, l = 0;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int a[14];
  for (i = 0; i < 14; i++) cin >> a[i];
  long long int maxx = 0;
  for (i = 0; i < 14; i++) {
    if (a[i] == 0) continue;
    long long int b[14], sum = 0;
    for (k = 0; k < 14; k++) b[k] = a[k];
    long long int part = a[i] / 14;
    long long int foo = a[i] % 14;
    b[i] = 0;
    for (j = 0; j < 14; j++) b[j] += part;
    j = i + 1;
    for (l = 1; l <= foo; l++) {
      if (j == 14) j = 0;
      b[j] = b[j] + 1;
      j++;
    }
    for (j = 0; j < 14; j++)
      if (b[j] % 2 == 0) sum += b[j];
    if (sum > maxx) maxx = sum;
  }
  cout << maxx << endl;
  return 0;
}
