def check():
	(a, b, c, d) = list(map(int, input().split()))
	(x, y, x1, y1, x2, y2) = list(map(int, input().split()))
	if x - x1 == 0 and x2 - x == 0 and (b != 0 or a != 0):
		print('No')
		return
	if y - y1 == 0 and y2 - y == 0 and (c != 0 or d != 0):
		print('No')
		return
	if x + b - a <= x2 and x - a + b >= x1 and (y + d - c <= y2) and (y + d - c >= y1):
		print('Yes')
	else:
		print('No')
i = int(input())
for j in range(i):
	check()
