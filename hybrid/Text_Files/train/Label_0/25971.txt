using namespace std;

/* constant */

const int MAX_H = 1000;
const int MAX_W = 1000;
const int MAX_N = 1000000;

int dxs[] = {1, 0, -1, 0}, dys[] = {0, -1, 0, 1};

const int INF = 1 << 30;

/* typedef */

struct Stat {
  int d, x, y, di;
  Stat() {}
  Stat(int _d, int _x, int _y, int _di): d(_d), x(_x), y(_y), di(_di) {}
  bool operator<(const Stat &s) const { return d < s.d; }
  bool operator>(const Stat &s) const { return d > s.d; }
};

/* global variables */

bool flds[MAX_H][MAX_W];
int dirs[MAX_N + 1], nxtds[MAX_N + 1][4];
int dists[MAX_H][MAX_W][4];

/* subroutines */

/* main */

int main() {
  for (;;) {
    int h, w, n;
    cin >> h >> w >> n;
    if (h == 0) break;

    string sop;
    cin >> sop;

    int nds[] = {INF, INF, INF, INF};
    dirs[0] = 1;
    for (int i = 0, di = 0; i < n; i++)
      dirs[i + 1] =  (dirs[i] + (sop[i] == 'L' ? 1 : 3)) % 4;
    for (int i = n; i >= 0; i--) {
      nds[dirs[i]] = i;
      for (int j = 0; j < 4; j++) nxtds[i][j] = nds[j];
      //printf("%d:", i);
      //for (int j = 0; j < 4; j++) printf(" %d", nxtds[i][j]); putchar('\n');
    }

    int sx, sy, gx, gy;
    memset(flds, true, sizeof(flds));
    
    for (int y = 0; y < h; y++) {
      string s;
      cin >> s;
      for (int x = 0; x < w; x++)
	switch (s[x]) {
	case 'S': sx = x; sy = y; break;
	case 'G': gx = x; gy = y; break;
	case '#': flds[y][x] = false; break;
	}
    }

    for (int y = 0; y < h; y++)
      for (int x = 0; x < w; x++)
	for (int di = 0; di < 4; di++) dists[y][x][di] = INF;
    dists[sy][sx][1] = 0;
    
    priority_queue<Stat,vector<Stat>,greater<Stat> > q;
    q.push(Stat(0, sx, sy, 1));
    bool ans = false;
    
    while (! q.empty()) {
      Stat u = q.top(); q.pop();
      if (dists[u.y][u.x][u.di] != u.d) continue;
      //printf("(%d,%d,%d)=%d\n", u.x, u.y, u.di, u.d);

      if (u.x == gx && u.y == gy) {
	ans = true;
	break;
      }

      for (int vdi = 0; vdi < 4; vdi++) {
	int vx = u.x + dxs[vdi], vy = u.y + dys[vdi];
	if (vx >= 0 && vx < w && vy >= 0 && vy < h && flds[vy][vx]) {
	  int vd = nxtds[u.d][vdi];
	  if (dists[vy][vx][vdi] > vd) {
	    dists[vy][vx][vdi] = vd;
	    q.push(Stat(vd, vx, vy, vdi));
	  }
	}
      }
    }

    cout << (ans ? "Yes" : "No") << endl;
  }
  
  return 0;
}
