vector<string> vec_splitter(string s) {
  for (char& c : s) c = c == ',' ? ' ' : c;
  stringstream ss;
  ss << s;
  vector<string> res;
  for (string z; ss >> z; res.push_back(z))
    ;
  return res;
}
void debug_out(vector<string> args, long long idx) { cerr << "\n"; }
template <typename Head, typename... Tail>
void debug_out(vector<string> args, long long idx, Head H, Tail... T) {
  if (idx > 0) cerr << ", ";
  stringstream ss;
  ss << H;
  cerr << args[idx] << " = " << ss.str();
  debug_out(args, idx + 1, T...);
}

long long x, y, pr, bda, chota, mark;
priority_queue<pair<long long, long long> > pq;
void solve() {
  cin >> x >> y;
  pq.push({x, 0});
  pq.push({y, 1});
  pr = -1;
  while (pr != pq.top().first) {
    bda = pq.top().first;
    mark = pq.top().second;
    pr = bda;
    pq.pop();
    chota = pq.top().first;
    if ((bda - 2 * chota) >= 0) bda -= ((bda / (2 * chota)) * (2 * chota));
    pq.push({bda, mark});
    if (bda == 0) break;
  }
  bda = pq.top().first;
  pr = bda;
  mark = pq.top().second;
  pq.pop();
  chota = pq.top().first;
  if (mark == 0)
    cout << bda << " " << chota;
  else
    cout << chota << " " << bda << "\n";
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  solve();
  return 0;
}
