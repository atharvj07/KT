#include <bits/stdc++.h>
using namespace std;
bool calc(long long m, long long k, long long n) {
  long long c = 1, t = m * k;
  long long bc = 1;
  for (int i = 1; i <= k && t >= i && c < n; ++i) {
    bc *= k - i + 1;
    bc /= i;
    long long x = min(t / i, bc);
    c += x;
    t -= i * x;
  }
  return c >= n;
}
int main() {
  int tc;
  scanf("%d", &tc);
  while (tc--) {
    long long n, m;
    scanf("%lld%lld", &n, &m);
    long long lo = 0, hi = n - 1;
    while (lo < hi) {
      long long k = (lo + hi) / 2;
      if (!calc(m, k, n))
        lo = k + 1;
      else
        hi = k;
    }
    printf("%lld\n", lo);
  }
  return 0;
}
