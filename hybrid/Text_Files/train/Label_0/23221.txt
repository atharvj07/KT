#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  t = 1;
  while (t--) {
    long long n;
    cin >> n;
    string x, y;
    cin >> x >> y;
    long long z = 0, o = 0, sum = 0, zo = 0, oo = 0;
    for (int i = 0; i < n; i++) {
      if (y[i] == '0' && x[i] == '0') {
        z++;
      }
      if (y[i] == '0' && x[i] == '1') {
        o++;
      }
      if (y[i] == '1' && x[i] == '0') {
        zo++;
      } else {
        oo++;
      }
      sum += x[i] - '0';
    }
    if (sum == 0 || sum == n) {
      cout << "0" << endl;
    } else {
      cout << z * (n - z - zo) + o * zo << endl;
    }
  }
  return 0;
}
