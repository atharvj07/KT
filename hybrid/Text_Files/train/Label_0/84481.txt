t = int(input())
ans = []
for i in range(t):
	s = input()
	if s.count('a') == len(s):
		ans.append('NO')
	elif (s + 'a')[::-1] != s + 'a':
		ans.append('YES')
		ans.append(s + 'a')
	elif ('a' + s)[::-1] != 'a' + s:
		ans.append('YES')
		ans.append('a' + s)
	else:
		ans.append('NO')
for i in ans:
	print(i)
