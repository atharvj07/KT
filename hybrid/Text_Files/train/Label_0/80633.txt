from functools import reduce
raws = [int(x) for x in input().split(' ')]
secpos = raws[2] / 5
minpos = raws[1] / 5 + secpos / 60
hrpos = raws[0] % 12 + minpos / 12
positions = [secpos, minpos, hrpos]
m = min(raws[3], raws[4])
n = max(raws[3], raws[4])
testcases = list(map(lambda x: x > n or x < m, positions))
out = reduce(lambda x, y: x and y, testcases) or reduce(lambda x, y: x and y, [not x for x in testcases])
if out:
	print('yes')
else:
	print('no')
