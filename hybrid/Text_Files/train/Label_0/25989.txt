#include <bits/stdc++.h>
const int maxn = 110;
const int mo = 1000000007;
using namespace std;
long long tr[maxn][22][22], f[maxn][22][22];
int n, m, e, ter[maxn << 1], nxt[maxn << 1], lnk[maxn], d[maxn];
void _put(int p, int x, int y, long long ave) {
  if (!y)
    tr[p][x][y] = (tr[p][x][y] + ave) % mo;
  else
    for (; y <= m + 1; y += y & (-y)) tr[p][x][y] = (tr[p][x][y] + ave) % mo;
}
void put(int p, int x, int y, long long ave) {
  if (!x)
    _put(p, x, y, ave);
  else
    for (; x <= m + 1; x += x & (-x)) _put(p, x, y, ave);
}
long long _get(int p, int x, int y) {
  if (y < 0) return 0;
  long long ret = tr[p][x][0];
  if (!y) return ret;
  for (; y; y -= y & (-y)) ret = (ret + tr[p][x][y]) % mo;
  return ret;
}
long long get(int p, int x, int y) {
  if (x < 0) return 0;
  long long ret = _get(p, 0, y);
  if (!x)
    return ret;
  else
    for (; x; x -= x & (-x)) ret = (ret + _get(p, x, y)) % mo;
  return ret;
}
void add(int x, int y) {
  ter[++e] = y, nxt[e] = lnk[x], lnk[x] = e;
  ter[++e] = x, nxt[e] = lnk[y], lnk[y] = e;
}
void dfs(int p, int las) {
  for (int i = lnk[p]; i; i = nxt[i])
    if (ter[i] != las) dfs(ter[i], p);
  d[0] = 0;
  for (int i = lnk[p]; i; i = nxt[i])
    if (ter[i] != las) d[++d[0]] = ter[i];
  if (!d[0]) {
    f[p][0][0] = f[p][m + 1][1] = 1;
    put(p, 0, 0, 1), put(p, m + 1, 1, 1);
    return;
  }
  for (int i = 1; i <= m; ++i) {
    f[p][i][0] = 1;
    for (int j = 1; j <= d[0]; ++j)
      f[p][i][0] =
          f[p][i][0] * (get(d[j], m + 1, m - i) - get(d[j], i - 2, m - i)) % mo;
    long long ret = 1, ret2, ret3;
    for (int j = 1; j <= d[0]; ++j)
      ret = ret * (get(d[j], m + 1, m - i) - get(d[j], i - 1, m - i)) % mo;
    f[p][i][0] = (f[p][i][0] - ret) % mo;
    for (int j = m - i + 2; j <= m; ++j) {
      f[p][i][j] = ret = ret2 = ret3 = 1;
      for (int k = 1; k <= d[0]; ++k)
        f[p][i][j] = f[p][i][j] *
                     (get(d[k], m + 1, j - 1) - get(d[k], i - 2, j - 1)) % mo,
        ret = ret * (get(d[k], m + 1, j - 1) - get(d[k], i - 1, j - 1)) % mo,
        ret2 = ret2 * (get(d[k], m + 1, j - 2) - get(d[k], i - 2, j - 2)) % mo,
        ret3 = ret3 * (get(d[k], m + 1, j - 2) - get(d[k], i - 1, j - 2)) % mo;
      f[p][i][j] = (f[p][i][j] - ret - ret2 + ret3) % mo;
    }
  }
  f[p][m + 1][0] = 0;
  for (int i = 1; i <= m; ++i) {
    f[p][m + 1][i] = 1;
    long long ret = 1;
    for (int j = 1; j <= d[0]; ++j)
      f[p][m + 1][i] = f[p][m + 1][i] *
                       (get(d[j], m + 1, i - 1) - get(d[j], m - 1, i - 1)) % mo,
               ret = ret * (get(d[j], m + 1, i - 2) - get(d[j], m - 1, i - 2)) %
                     mo;
    f[p][m + 1][i] = (f[p][m + 1][i] - ret) % mo;
  }
  f[p][0][0] = 1;
  for (int i = 1; i <= d[0]; ++i)
    f[p][0][0] = f[p][0][0] * get(d[i], m + 1, m) % mo;
  put(p, 0, 0, f[p][0][0]);
  for (int i = 1; i <= m + 1; ++i)
    for (int j = 0; j <= m; ++j) put(p, i, j, f[p][i][j]);
}
int main() {
  scanf("%d%d", &n, &m);
  int x, y;
  for (int i = 1; i <= n - 1; ++i) scanf("%d%d", &x, &y), add(x, y);
  dfs(1, 0);
  printf("%lld\n", (get(1, m + 1, 0) + mo) % mo);
  return 0;
}
