n = int(input())
for i in range(n):
	y = int(input())
	f = [0] * y
	g = [0] * y
	p = []
	for j in range(y):
		p.append(list(map(int, input().split()))[1:])
	for j in range(1, y + 1):
		for q in p[j - 1]:
			if f[q - 1] == 0:
				f[q - 1] = 1
				g[j - 1] = 1
				break
	if sum(g) == y:
		print('OPTIMAL')
	else:
		print('IMPROVE')
		print(g.index(0) + 1, f.index(0) + 1)
