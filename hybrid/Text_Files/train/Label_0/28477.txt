#include <bits/stdc++.h>
using namespace std;
long long C(int n, int r) {
  if (r > n / 2) r = n - r;
  long long ans = 1;
  int i;
  for (i = 1; i <= r; i++) {
    ans *= n - r + i;
    ans /= i;
  }
  return ans;
}
int main() {
  long long int k = 0;
  int a, b, c, d, r1, r2, n1, n2, d1, d2;
  cin >> a >> b >> c;
  r1 = 4;
  for (int i = 0; i < c; i++) {
    if (c - r1 > b)
      r1++;
    else
      break;
  }
  r2 = c - r1;
  n1 = a, n2 = b;
  for (int i = 0; i < c; i++) {
    if (n1 == r1 || r2 == 1) {
      k += C(n1, r1) * C(n2, r2);
      cout << k;
      break;
    }
    k += C(n1, r1) * C(n2, r2);
    r1++, r2--;
  }
}
