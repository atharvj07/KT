#include<iostream>
using namespace std;
int main()
{
	int t;
	cin>>t;

	while(t--)
	{
		int n;
		cin>>n;

		int arr[2000];

		for(int i=0;i<n;i++)
			cin>>arr[i];

		for(int i=0;i<n;i++)
		{
			if(arr[i]%2)
				cout<<arr[i]<<" ";
		}
		for(int i=0;i<n;i++)
		{
			if(arr[i]%2==0)
				cout<<arr[i]<<" ";
		}
		cout<<endl;
	}
}
