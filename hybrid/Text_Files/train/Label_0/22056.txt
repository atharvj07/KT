#include<bits/stdc++.h>
#define maxn 505050

using namespace std;
typedef long long ll;
ll a[maxn],b[maxn],x[maxn],y[maxn],l,r,mid,n;
vector <ll> d;

int main(){
    cin >> n;
    for (int i=1;i<=n;i++) cin >> a[i];
    for (int i=1;i<=n;i++) cin >> b[i];
    int ans=0; long long s=0;
    for (int i=1;i<=n;i++) s+=a[i]-b[i];
    if (s<0) {puts("-1");return 0;} s=0;
    for (int i=1;i<=n;i++){
        if (a[i]<b[i]) s+=b[i]-a[i],ans++;
        else d.push_back(a[i]-b[i]);
    }
    sort(d.begin(),d.end(),greater<int>());
    for (int i=0;i<d.size();i++){
        if (s<=0) break;
        s-=d[i]; ans++;
    }
    cout << ans << endl;
}
