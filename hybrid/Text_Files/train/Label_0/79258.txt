def readStr():
	return input()

def readInts():
	return list(map(int, readStr().split(' ')))

def main(n, a):
	prev = 0
	this = 0
	idx = [0, n - 1]
	val = [0, 0]
	i = 0
	while idx[0] <= idx[1]:
		this += a[idx[i % 2]]
		if i % 2 == 0:
			idx[0] += 1
		else:
			idx[1] -= 1
		if this > prev:
			val[i % 2] += this
			prev = this
			this = 0
			i += 1
	if this > 0:
		val[i % 2] += this
		i += 1
	return (i, val[0], val[1])
(t,) = readInts()
for _ in range(t):
	(n,) = readInts()
	a = readInts()
	print(*main(n, a))
